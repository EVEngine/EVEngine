21Result<UvPaintRegion> prepareUvPaintRegionResult(int targetWidth, int targetHeight, float u, float v, float radiusU,
22 float radiusV, float r, float g, float b, float a, bool flipV) {
23 const auto invalid = [](const char* message, const char* path) {
24 return Result<UvPaintRegion>::failure(
25 Diagnostic::error(DiagnosticCode::InvalidArgument, message, path, {}, "image.uvPaintRegion"));
26 };
27 if (targetWidth <= 0 || targetHeight <= 0) return invalid("paint target dimensions must be positive", "targetSize");
28 if (!std::isfinite(u) || !std::isfinite(v) || !std::isfinite(radiusU) || !std::isfinite(radiusV) || radiusU < 0.f ||
29 radiusV < 0.f)
30 return invalid("UV center and radii must be finite; radii must be non-negative", "brush");
31 if (!std::isfinite(r) || !std::isfinite(g) || !std::isfinite(b) || !std::isfinite(a) || r < 0.f || r > 1.f ||
32 g < 0.f || g > 1.f || b < 0.f || b > 1.f || a < 0.f || a > 1.f)
33 return invalid("paint color channels must be finite values in [0,1]", "color");
34
35 UvPaintRegion region;
36 region.targetWidth = targetWidth;
37 region.targetHeight = targetHeight;
38 region.u = std::clamp(u, 0.f, 1.f);
39 region.v = std::clamp(v, 0.f, 1.f);
40 region.radiusU = radiusU;
41 region.radiusV = radiusV;
42 region.r = r;
43 region.g = g;
44 region.b = b;
45 region.a = a;
46
47 const float pixelU = region.u * float(targetWidth - 1);
48 const float mappedV = flipV ? 1.f - region.v : region.v;
49 const float pixelV = mappedV * float(targetHeight - 1);
50 region.centerX = static_cast<int>(std::lround(pixelU));
51 region.centerY = static_cast<int>(std::lround(pixelV));
52 region.minX = std::clamp(static_cast<int>(std::floor(pixelU - radiusU * targetWidth)), 0, targetWidth - 1);
53 region.maxX = std::clamp(static_cast<int>(std::ceil(pixelU + radiusU * targetWidth)), 0, targetWidth - 1);
54 region.minY = std::clamp(static_cast<int>(std::floor(pixelV - radiusV * targetHeight)), 0, targetHeight - 1);
55 region.maxY = std::clamp(static_cast<int>(std::ceil(pixelV + radiusV * targetHeight)), 0, targetHeight - 1);
56 return Result<UvPaintRegion>::success(region);
57}
