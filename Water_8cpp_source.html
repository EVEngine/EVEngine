41Shader *newWaterShader(Graphics *gfx) {
42 // Use precompiled SPIR-V (not runtime glslc) so the water shader also works
43 // on platforms without a runtime compiler (e.g. Windows).
44 Shader *sh = nullptr;
45 if (gfx->getBackendName() == "webgpu") {
46 sh = gfx->newMeshShaderFromWgsl(shaders::kWaterVertWgsl, shaders::kWaterFragWgsl);
47 } else {
48 std::vector<uint32_t> vert(water_vert_spv, water_vert_spv + water_vert_spv_count);
49 std::vector<uint32_t> frag(water_frag_spv, water_frag_spv + water_frag_spv_count);
50 sh = gfx->newMeshShaderFromSpv(vert, frag);
51 }
52 gfx->configureMeshShaderSurface(*sh, BlendMode::Alpha, false, false)
53 .expect("water requires alpha blending with depth writes disabled");
54 for (int i = 0; i < kUniformCount; ++i) {
55 const std::string name(kUniformNames[i]);
56 if (name == "deepColor" || name == "shallowColor" || name == "foamColor" || name == "reflTint")
57 sh->declareVec3(kUniformNames[i]);
58 else
59 sh->declareFloat(kUniformNames[i]);
60 }
61 return sh;
62}

84void Water::createPlane(float sizeX, float sizeZ, int segX, int segZ) {
85 segX = std::max(1, segX);
86 segZ = std::max(1, segZ);
87 std::vector<float> pos, nrm, uv;
88 std::vector<uint32_t> idx;
89 for (int z = 0; z <= segZ; ++z) {
90 for (int x = 0; x <= segX; ++x) {
91 pos.push_back(-sizeX * 0.5f + sizeX * float(x) / segX);
92 pos.push_back(0.f);
93 pos.push_back(-sizeZ * 0.5f + sizeZ * float(z) / segZ);
94 nrm.push_back(0.f);
95 nrm.push_back(1.f);
96 nrm.push_back(0.f);
97 uv.push_back(float(x) / segX);
98 uv.push_back(float(z) / segZ);
99 }
100 }
101 for (int z = 0; z < segZ; ++z) {
102 for (int x = 0; x < segX; ++x) {
103 const uint32_t a = uint32_t(z * (segX + 1) + x);
104 const uint32_t b = a + 1;
105 const uint32_t c = a + uint32_t(segX + 1);
106 const uint32_t d = c + 1;
107 idx.push_back(a);
108 idx.push_back(c);
109 idx.push_back(b);
110 idx.push_back(b);
111 idx.push_back(c);
112 idx.push_back(d);
113 }
114 }
115 mesh_ = gfx_->newMeshFromArrays(pos.data(), nrm.data(), uv.data(), int(pos.size() / 3),
116 idx.data(), int(idx.size()));
117}

197void Water::drawWithReflection(Texture* requestedReflection, float requestedStrength) {
198 if (!gfx_ || !mesh_ || !shader_) return;
199 bindParams();
200 Texture *reflection = requestedReflection;
201 float reflectionStrength = requestedReflection ? requestedStrength : -1.0F;
202 RenderControl *rc = gfx_->getRenderControl();
203 if (!reflection && config_.screenSpaceReflection && rc && rc->isEnabled("ssr")) {
204 ScreenSpaceReflection *ssr = gfx_->pipelineScreenSpaceReflection();
205 if (ssr->hasValidHistory()) {
206 reflection = ssr->getReflectionTexture();
207 reflectionStrength = config_.screenSpaceReflectionStrength;
208 }
209 }
210 shader_->sendFloat("ssrStrength", reflection ? reflectionStrength : -1.0F);
211 Texture *depth = rc && rc->getGBuffer() ? rc->getGBuffer()->getDepthTexture() : nullptr;
212 Texture *sceneColor = gfx_->getSceneColorTexture();
213 shader_->sendFloat("opacityAndDepth", depth ? config_.opacity : -(config_.opacity + 1.0F));
214 shader_->sendFloat("refractionStrength", sceneColor ? config_.refractionStrength : 0.0F);
215 gfx_->setMesh3DHeightTexture(reflection);
216 gfx_->setMesh3DSceneDepth(depth);
217 gfx_->drawMeshShader(mesh_, glm::mat4(1.f), sceneColor, glm::vec4(1.f), shader_);
218 gfx_->setMesh3DHeightTexture(nullptr);
219 gfx_->setMesh3DSceneDepth(nullptr);
220}

222void Water::drawReflectionCapture() {
223 if (!gfx_ || !mesh_ || !shader_) return;
224 bindParams();
225 // Probe captures must never sample the main-view SSR history: that creates
226 // view-dependent feedback and recursively bakes an old reflection into the cube.
227 shader_->sendFloat("ssrStrength", -1.0F);
228 shader_->sendFloat("opacityAndDepth", -(config_.opacity + 1.0F));
229 shader_->sendFloat("refractionStrength", 0.0F);
230 gfx_->setMesh3DHeightTexture(nullptr);
231 gfx_->drawMeshShader(mesh_, glm::mat4(1.f), nullptr, glm::vec4(1.f), shader_);
232 // Restore the authored state for a subsequent main-view draw in the same frame.
233 shader_->sendFloat("ssrStrength", config_.screenSpaceReflection ? config_.screenSpaceReflectionStrength : -1.0F);
234 shader_->sendFloat("refractionStrength", config_.refractionStrength);
235}
