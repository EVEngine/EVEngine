15// Push-constant layout (data[32]):

31Shader *newWaterShader(Graphics *gfx) {
32 // Use precompiled SPIR-V (not runtime glslc) so the water shader also works
33 // on platforms without a runtime compiler (e.g. Windows).
34 std::vector<uint32_t> frag(water_frag_spv, water_frag_spv + water_frag_spv_count);
35 Shader *sh = gfx->newMeshShaderFromSpv({}, frag);
36 for (int i = 0; i < kUniformCount; ++i) {
37 if (std::string(kUniformNames[i]) == "waterCol")
38 sh->declareVec3(kUniformNames[i]);
39 else if (std::string(kUniformNames[i]) == "reflTint")
40 sh->declareVec3(kUniformNames[i]);
41 else
42 sh->declareFloat(kUniformNames[i]);
43 }
44 return sh;
45}

59void Water::createPlane(float sizeX, float sizeZ, int segX, int segZ) {
60 segX = std::max(1, segX);
61 segZ = std::max(1, segZ);
62 std::vector<float> pos, nrm, uv;
63 std::vector<uint32_t> idx;
64 for (int z = 0; z <= segZ; ++z) {
65 for (int x = 0; x <= segX; ++x) {
66 pos.push_back(-sizeX * 0.5f + sizeX * float(x) / segX);
67 pos.push_back(0.f);
68 pos.push_back(-sizeZ * 0.5f + sizeZ * float(z) / segZ);
69 nrm.push_back(0.f);
70 nrm.push_back(1.f);
71 nrm.push_back(0.f);
72 uv.push_back(float(x) / segX);
73 uv.push_back(float(z) / segZ);
74 }
75 }
76 for (int z = 0; z < segZ; ++z) {
77 for (int x = 0; x < segX; ++x) {
78 const uint32_t a = uint32_t(z * (segX + 1) + x);
79 const uint32_t b = a + 1;
80 const uint32_t c = a + uint32_t(segX + 1);
81 const uint32_t d = c + 1;
82 idx.push_back(a);
83 idx.push_back(c);
84 idx.push_back(b);
85 idx.push_back(b);
86 idx.push_back(c);
87 idx.push_back(d);
88 }
89 }
90 mesh_ = gfx_->newMeshFromArrays(pos.data(), nrm.data(), uv.data(), int(pos.size() / 3),
91 idx.data(), int(idx.size()));
92}
