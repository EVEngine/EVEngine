67Result<glm::ivec2> waterPointCounts(const WaterMeshSettings& settings) {
68 if ((settings.type != WaterMeshType::Plane && settings.type != WaterMeshType::Circle) ||
69 !std::isfinite(settings.sizeX) || !std::isfinite(settings.sizeY) || !std::isfinite(settings.sizeZ) ||
70 !std::isfinite(settings.densityX) || !std::isfinite(settings.densityY) || !std::isfinite(settings.height) ||
71 settings.sizeX <= 0.0F || settings.sizeY <= 0.0F || settings.sizeZ <= 0.0F || settings.densityX <= 0.0F ||
72 settings.densityY <= 0.0F) {
73 return Result<glm::ivec2>::failure(
74 Diagnostic::error(DiagnosticCode::InvalidArgument,
75 "water.mesh: finite positive size/density and a supported shape are required"));
76 }
77 const double scale = settings.type == WaterMeshType::Plane ? 1.0 : 0.5;
78 const double rawX = double(settings.densityX) / 40.0 * double(settings.sizeX) * scale;
79 const double rawY = double(settings.densityY) / 40.0 * double(settings.sizeZ) * scale;
80 if (rawX > 4096.0 || rawY > 4096.0)
81 return Result<glm::ivec2>::failure(
82 Diagnostic::error(DiagnosticCode::InvalidArgument, "water.mesh: generated axis point count exceeds 4096"));
83 int x = static_cast<int>(std::nearbyint(rawX));
84 int y = static_cast<int>(std::nearbyint(rawY));
85 if (settings.type == WaterMeshType::Plane) {
86 ++x;
87 ++y;
88 if (x < 2 || y < 2)
89 return Result<glm::ivec2>::failure(
90 Diagnostic::error(DiagnosticCode::InvalidArgument, "water.mesh: plane requires at least one polygon"));
91 } else if (x < 1 || y < 1) {
92 return Result<glm::ivec2>::failure(
93 Diagnostic::error(DiagnosticCode::InvalidArgument, "water.mesh: circle requires at least one ring"));
94 }
95 return Result<glm::ivec2>::success({x, y});
96}

99 if (ring < 0) return 0;

102 if (extent < 0) extent += count;
103 return 3 * ring * (ring + 1) + extent + 1;

106template <typename Stop, typename Value, typename Project>
107Value sampleStops(const std::vector<Stop>& stops, float time, Project project) {
108 if (time <= stops.front().time) return project(stops.front());
109 if (time >= stops.back().time) return project(stops.back());
110 for (std::size_t i = 1; i < stops.size(); ++i)
111 if (time <= stops[i].time) {
112 const auto& a = stops[i - 1];
113 const auto& b = stops[i];
114 const float t = (time - a.time) / (b.time - a.time);
115 return project(a) + (project(b) - project(a)) * t;
116 }
117 return project(stops.back());
118}

151 if (!counts) return Result<int>::failure(counts.status());
152 const std::int64_t triangles = settings.type == WaterMeshType::Plane

164 Shader* sh = nullptr;
165 if (gfx->getBackendName() == "webgpu") {

172 gfx->configureMeshShaderSurface(*sh, BlendMode::Alpha, false, false)

177 sh->declareVec3(kUniformNames[i]);
178 else
179 sh->declareFloat(kUniformNames[i]);
180 }
181 return sh;

238Result<int> Water::createProceduralMesh(const WaterMeshSettings& settings) {
239 auto counts = waterPointCounts(settings);
240 if (!counts) return Result<int>::failure(counts.status());
241 auto triangleCount = calculateWaterMeshTriangles(settings);
242 if (!triangleCount) return Result<int>::failure(triangleCount.status());
243 if (!gfx_)
244 return Result<int>::failure(
245 Diagnostic::error(DiagnosticCode::PreconditionViolation, "water.mesh: graphics owner is unavailable"));
246
247 std::vector<float> pos, nrm, uv;
248 std::vector<uint32_t> idx;
249 if (settings.type == WaterMeshType::Plane) {
250 const int nx = counts.value().x, nz = counts.value().y;
251 pos.reserve(std::size_t(nx) * nz * 3);
252 nrm.reserve(std::size_t(nx) * nz * 3);
253 uv.reserve(std::size_t(nx) * nz * 2);
254 idx.reserve(std::size_t(triangleCount.value()) * 3);
255 const float sx = settings.sizeX / float(nx - 1), sz = settings.sizeZ / float(nz - 1);
256 for (int z = 0; z < nz; ++z)
257 for (int x = 0; x < nx; ++x) {
258 pos.insert(pos.end(), {(float(nx - 1) * 0.5F - x) * sx, settings.height * settings.sizeY,
259 (float(nz - 1) * 0.5F - z) * sz});
260 nrm.insert(nrm.end(), {0.0F, 1.0F, 0.0F});
261 uv.insert(uv.end(), {(float(nx / 2) - x) / float(nx), (float(nz / 2) - z) / float(nz)});
262 }
263 for (int z = 0; z < nz - 1; ++z)
264 for (int x = 0; x < nx - 1; ++x) {
265 const uint32_t a = uint32_t(z * nx + x), b = a + 1, c = a + uint32_t(nx), d = c + 1;
266 idx.insert(idx.end(), {a, c, b, c, d, b});
267 }
268 } else {
269 const int rings = counts.value().x;
270 const std::size_t vertexCount = 1u + 3u * std::size_t(rings) * std::size_t(rings + 1);
271 pos.reserve(vertexCount * 3);
272 nrm.reserve(vertexCount * 3);
273 uv.reserve(vertexCount * 2);
274 idx.reserve(std::size_t(triangleCount.value()) * 3);
275 const float scaleX = (settings.sizeX * 0.5F) / float(rings);
276 const float scaleZ = (settings.sizeZ * 0.5F) / float(counts.value().y);
277 pos.insert(pos.end(), {0.0F, 0.0F, 0.0F});
278 nrm.insert(nrm.end(), {0.0F, 1.0F, 0.0F});
279 uv.insert(uv.end(), {0.0F, 0.0F});
280 for (int ring = 0; ring < rings; ++ring) {
281 const int ringPoints = (ring + 1) * 6;
282 const float step = 2.0F * 3.14159265358979323846F / float(ringPoints);
283 for (int point = 0; point < ringPoints; ++point) {
284 const float cx = std::cos(step * point) * float(ring + 1);
285 const float cz = std::sin(-step * point) * float(ring + 1);
286 pos.insert(pos.end(), {cx * scaleX, 0.0F, cz * scaleZ});
287 nrm.insert(nrm.end(), {0.0F, 1.0F, 0.0F});
288 uv.insert(uv.end(), {cx / settings.densityX, cz / settings.densityX});
289 }
290 }
291 for (int ring = 0; ring < rings; ++ring) {
292 int other = 0;
293 for (int point = 0; point < (ring + 1) * 6; ++point) {
294 if (point % (ring + 1) != 0) {
295 idx.insert(
296 idx.end(),
297 {uint32_t(waterCirclePoint(ring - 1, other + 1)), uint32_t(waterCirclePoint(ring - 1, other)),
298 uint32_t(waterCirclePoint(ring, point)), uint32_t(waterCirclePoint(ring, point)),
299 uint32_t(waterCirclePoint(ring, point + 1)), uint32_t(waterCirclePoint(ring - 1, other + 1))});
300 ++other;
301 } else {
302 idx.insert(idx.end(),
303 {uint32_t(waterCirclePoint(ring, point)), uint32_t(waterCirclePoint(ring, point + 1)),
304 uint32_t(waterCirclePoint(ring - 1, other))});
305 }
306 }
307 }
308 }
309 Mesh* candidate =
310 gfx_->newMeshFromArrays(pos.data(), nrm.data(), uv.data(), int(pos.size() / 3), idx.data(), int(idx.size()));
311 if (!candidate)
312 return Result<int>::failure(
313 Diagnostic::error(DiagnosticCode::Failed, "water.mesh: graphics mesh allocation failed"));
314 mesh_ = candidate;
315 return Result<int>::success(int(pos.size() / 3));
316}

318Result<int> Water::setDepthGradient(const WaterDepthGradient& gradient, int resolution) {
319 if (!gfx_ || resolution < 2 || resolution > 4096 || gradient.colorStops.empty() || gradient.alphaStops.empty())
320 return Result<int>::failure(
321 Diagnostic::error(DiagnosticCode::InvalidArgument,
322 "water.gradient: graphics, color/alpha keys and resolution 2..4096 required"));
323 for (std::size_t i = 1; i < gradient.colorStops.size(); ++i)
324 if (gradient.colorStops[i - 1].time == gradient.colorStops[i].time)
325 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
326 "water.gradient: duplicate color times are ambiguous"));
327 for (std::size_t i = 1; i < gradient.alphaStops.size(); ++i)
328 if (gradient.alphaStops[i - 1].time == gradient.alphaStops[i].time)
329 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
330 "water.gradient: duplicate alpha times are ambiguous"));
331 image::ImageData pixels(resolution, resolution, "RGBA8");
332 for (int x = 0; x < resolution; ++x) {
333 const float t = float(x) / float(resolution);
334 const glm::vec3 color = sampleStops<WaterGradientColorStop, glm::vec3>(
335 gradient.colorStops, t, [](const auto& stop) { return stop.color; });
336 const float alpha = sampleStops<WaterGradientAlphaStop, float>(gradient.alphaStops, t,
337 [](const auto& stop) { return stop.alpha; });
338 for (int y = 0; y < resolution; ++y)
339 pixels.setPixel(x, y, image::ImageData::Colorf{color.r, color.g, color.b, alpha});
340 }
341 Texture* candidate = gfx_->newTextureFromImageData(&pixels, false, false);
342 if (!candidate)
343 return Result<int>::failure(
344 Diagnostic::error(DiagnosticCode::Failed, "water.gradient: texture allocation failed"));
345 depthGradientTexture_ = candidate;
346 return Result<int>::success(resolution * resolution);
347}

437void Water::drawWithReflection(Texture* requestedReflection, float requestedStrength) {
438 if (!gfx_ || !mesh_ || !shader_) return;
439 bindParams();
440 Texture* reflection = requestedReflection;
441 float reflectionStrength = requestedReflection ? requestedStrength : -1.0F;
442 RenderControl* rc = gfx_->getRenderControl();
443 if (!reflection && config_.screenSpaceReflection && rc && rc->isEnabled("ssr")) {
444 ScreenSpaceReflection* ssr = gfx_->pipelineScreenSpaceReflection();
445 if (ssr->hasValidHistory()) {
446 reflection = ssr->getReflectionTexture();
447 reflectionStrength = config_.screenSpaceReflectionStrength;
448 }
449 }
450 shader_->sendFloat("ssrStrength", reflection ? reflectionStrength : -1.0F);
451 Texture* depth = rc && rc->getGBuffer() ? rc->getGBuffer()->getDepthTexture() : nullptr;
452 Texture* sceneColor = gfx_->getSceneColorTexture();
453 shader_->sendFloat("opacityAndDepth", depth ? config_.opacity : -(config_.opacity + 1.0F));
454 shader_->sendFloat("refractionStrength", sceneColor ? config_.refractionStrength : 0.0F);

499Result<WaterPlanarReflectionPlan> buildWaterPlanarReflectionPlan(const WaterPlanarReflectionSettings& settings,
500 const WaterPlanarReflectionInput& input) {
501 const float multiplier = resolutionScale(settings.resolutionMultiplier);
502 if (multiplier == 0.0F || settings.textureResolution <= 0 || settings.textureResolution > 16384 ||
503 !std::isfinite(settings.clipPlaneOffset) || !std::isfinite(settings.customRenderDistance) ||
504 settings.customRenderDistance < 0.0F || !std::isfinite(settings.lodBias) || settings.lodBias <= 0.0F ||
505 !finite3(input.cameraPosition) || !finite3(input.cameraForward) ||
506 glm::dot(input.cameraForward, input.cameraForward) <= 0.0F || !std::isfinite(input.waterPlaneY) ||
507 !std::isfinite(input.renderScale) || input.renderScale <= 0.0F) {
508 return Result<WaterPlanarReflectionPlan>::failure(
509 Diagnostic::error(DiagnosticCode::InvalidArgument,
510 "water.planarReflection: finite camera, plane, scale and valid settings required"));
511 }
512 for (float distance : settings.customRenderDistances) {
513 if (!std::isfinite(distance) || distance < 0.0F)
514 return Result<WaterPlanarReflectionPlan>::failure(
515 Diagnostic::error(DiagnosticCode::InvalidArgument,
516 "water.planarReflection: layer distances must be finite and nonnegative"));
517 }
518
519 WaterPlanarReflectionPlan plan;
520 const float planeY = input.waterPlaneY + settings.clipPlaneOffset;
521 plan.reflectionMatrix = glm::mat4(1.0F);
522 plan.reflectionMatrix[1][1] = -1.0F;
523 plan.reflectionMatrix[3][1] = 2.0F * planeY;
524 plan.cameraPosition = {input.cameraPosition.x, 2.0F * planeY - input.cameraPosition.y, input.cameraPosition.z};
525 plan.cameraForward =
526 glm::normalize(glm::vec3(input.cameraForward.x, -input.cameraForward.y, input.cameraForward.z));
527 plan.worldClipPlane = {0.0F, 1.0F, 0.0F, -planeY};
528 plan.cullingMask = settings.enabled ? (settings.reflectLayers & ~(1U << 4U)) : 0U;
529 plan.renderShadows = settings.shadows;
530 plan.shouldRender = settings.enabled && !settings.disableSkyboxReflections && !input.orthographic &&
531 !input.reflectionOrPreviewCamera;
532 const double pixels = static_cast<double>(settings.textureResolution) * input.renderScale * multiplier;
533 plan.textureWidth = plan.textureHeight = std::max(1, static_cast<int>(pixels));
534 if (settings.enableRenderDistance) {
535 if (settings.enablePerLayerDistances) {
536 for (std::size_t i = 0; i < plan.layerCullDistances.size(); ++i)
537 plan.layerCullDistances[i] = settings.customRenderDistances[i] * settings.lodBias;
538 } else
539 plan.layerCullDistances.fill(settings.customRenderDistance * settings.lodBias);
540 }
541 return Result<WaterPlanarReflectionPlan>::success(plan);
542}

668Result<WaterReflectionMaskTransition> WaterReflectionMasker::evaluate(float playerX, float playerZ, float terrainX,
669 float terrainZ, float terrainWidth,
670 float terrainDepth,
671 WaterPlanarReflectionSettings& settings) {
672 if (!finite6(playerX, playerZ, terrainX, terrainZ, terrainWidth, terrainDepth) || terrainWidth <= 0.f ||
673 terrainDepth <= 0.f)
674 return Result<WaterReflectionMaskTransition>::failure(Diagnostic::error(
675 DiagnosticCode::InvalidArgument, "finite player, terrain origin and positive terrain size are required", {},
676 {}, "graphics.waterReflectionMasker"));
677 bool next = false;
678 sampleX_ = sampleY_ = -1;
679 const float u = (playerX - terrainX) / terrainWidth, v = (playerZ - terrainZ) / terrainDepth;
680 if (!pixels_.empty() && u >= 0.f && u <= 1.f && v >= 0.f && v <= 1.f) {
681 const int x = static_cast<int>(std::lround(u * static_cast<float>(width_)));
682 const int y = static_cast<int>(std::lround(v * static_cast<float>(height_)));
683 sampleX_ = x;
684 sampleY_ = y;
685 if (x >= 0 && x < width_ && y >= 0 && y < height_) {
686 const auto& p = pixels_[static_cast<size_t>(y) * static_cast<size_t>(width_) + static_cast<size_t>(x)];
687 const auto hit = [&](float value) { return value >= minimum_ && value <= maximum_; };
688 switch (channel_) {
689 case WaterReflectionMaskChannel::R: next = hit(p.r); break;
690 case WaterReflectionMaskChannel::G: next = hit(p.g); break;
691 case WaterReflectionMaskChannel::B: next = hit(p.b); break;
692 case WaterReflectionMaskChannel::A: next = hit(p.a); break;
693 case WaterReflectionMaskChannel::RGBA: next = hit(p.r) || hit(p.g) || hit(p.b) || hit(p.a); break;
694 }
695 }
696 }
697 if (next == enabled_)
698 return Result<WaterReflectionMaskTransition>::success(WaterReflectionMaskTransition::Unchanged);
699 enabled_ = next;
700 heightFeaturesEnabled_ = next;
701 settings.enabled = next;
702 return Result<WaterReflectionMaskTransition>::success(next ? WaterReflectionMaskTransition::Enabled
703 : WaterReflectionMaskTransition::Disabled);
704}
