19// Push-constant layout (data[32]):

35Shader *newWaterShader(Graphics *gfx) {
36 // Use precompiled SPIR-V (not runtime glslc) so the water shader also works
37 // on platforms without a runtime compiler (e.g. Windows).
38 Shader *sh = nullptr;
39 if (gfx->getBackendName() == "webgpu") {
40 sh = gfx->newMeshShaderFromWgsl({}, shaders::kWaterFragWgsl);
41 } else {
42 std::vector<uint32_t> frag(water_frag_spv, water_frag_spv + water_frag_spv_count);
43 sh = gfx->newMeshShaderFromSpv({}, frag);
44 }
45 for (int i = 0; i < kUniformCount; ++i) {
46 if (std::string(kUniformNames[i]) == "waterCol")
47 sh->declareVec3(kUniformNames[i]);
48 else if (std::string(kUniformNames[i]) == "reflTint")
49 sh->declareVec3(kUniformNames[i]);
50 else
51 sh->declareFloat(kUniformNames[i]);
52 }
53 return sh;
54}

76void Water::createPlane(float sizeX, float sizeZ, int segX, int segZ) {
77 segX = std::max(1, segX);
78 segZ = std::max(1, segZ);
79 std::vector<float> pos, nrm, uv;
80 std::vector<uint32_t> idx;
81 for (int z = 0; z <= segZ; ++z) {
82 for (int x = 0; x <= segX; ++x) {
83 pos.push_back(-sizeX * 0.5f + sizeX * float(x) / segX);
84 pos.push_back(0.f);
85 pos.push_back(-sizeZ * 0.5f + sizeZ * float(z) / segZ);
86 nrm.push_back(0.f);
87 nrm.push_back(1.f);
88 nrm.push_back(0.f);
89 uv.push_back(float(x) / segX);
90 uv.push_back(float(z) / segZ);
91 }
92 }
93 for (int z = 0; z < segZ; ++z) {
94 for (int x = 0; x < segX; ++x) {
95 const uint32_t a = uint32_t(z * (segX + 1) + x);
96 const uint32_t b = a + 1;
97 const uint32_t c = a + uint32_t(segX + 1);
98 const uint32_t d = c + 1;
99 idx.push_back(a);
100 idx.push_back(c);
101 idx.push_back(b);
102 idx.push_back(b);
103 idx.push_back(c);
104 idx.push_back(d);
105 }
106 }
107 mesh_ = gfx_->newMeshFromArrays(pos.data(), nrm.data(), uv.data(), int(pos.size() / 3),
108 idx.data(), int(idx.size()));
109}

169void Water::draw() {
170 if (!gfx_ || !mesh_ || !shader_) return;
171 bindParams();
172 const int drawableW = gfx_->getPixelWidth() > 0 ? gfx_->getPixelWidth() : gfx_->getWidth();
173 const int drawableH = gfx_->getPixelHeight() > 0 ? gfx_->getPixelHeight() : gfx_->getHeight();
174 const float viewportW = viewportW_ > 0.f ? viewportW_ : float(drawableW);
175 const float viewportH = viewportH_ > 0.f ? viewportH_ : float(drawableH);
176 shader_->sendFloat("viewportW", std::max(viewportW, 1.f));
177 shader_->sendFloat("viewportH", std::max(viewportH, 1.f));
178
179 Texture *reflection = nullptr;
180 RenderControl *rc = gfx_->getRenderControl();
181 if (ssrEnabled_ && rc && rc->isEnabled("ssr")) {
182 ScreenSpaceReflection *ssr = gfx_->pipelineScreenSpaceReflection();
183 if (ssr->hasValidHistory()) reflection = ssr->getReflectionTexture();
184 }
185 shader_->sendFloat("ssrEnabled", reflection ? 1.f : 0.f);
186 gfx_->setMesh3DHeightTexture(reflection);
187 gfx_->drawMeshShader(mesh_, glm::mat4(1.f), nullptr, glm::vec4(1.f), shader_);
188 gfx_->setMesh3DHeightTexture(nullptr);
189}

191void Water::drawReflectionCapture() {
192 if (!gfx_ || !mesh_ || !shader_) return;
193 bindParams();
194 // Probe captures must never sample the main-view SSR history: that creates
195 // view-dependent feedback and recursively bakes an old reflection into the cube.
196 shader_->sendFloat("ssrEnabled", 0.f);
197 gfx_->setMesh3DHeightTexture(nullptr);
198 gfx_->drawMeshShader(mesh_, glm::mat4(1.f), nullptr, glm::vec4(1.f), shader_);
199 // Restore the authored state for a subsequent main-view draw in the same frame.
200 shader_->sendFloat("ssrEnabled", ssrEnabled_ ? 1.f : 0.f);
201 shader_->sendFloat("ssrStrength", ssrStrength_);
202}
