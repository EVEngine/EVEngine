12Result<std::size_t> RTS::stepAttackSlots() {
13 EV_PROFILE_MODULE("rts", "RTS::stepAttackSlots");
14 try {
15 // Scope: Unit infantry with current Attack/AttackMove and a live Unit target.
16 // Read: Identity, Orders, Motion, Combat, Crowd, Durability, Containment; canonical map.
17 // Write: Navigation combatSlotTarget/combatSlotWaiting, after closing the view.
18 // No structural mutations/events/callbacks; simulation phase before both motion backends.
19 struct Projection {
20 ecs::EntityHandle handle;
21 std::optional<WorldPosition> target;
22 bool waiting = false;
23 };
24 std::vector<Projection> plan;
25 struct Member {
26 ecs::EntityHandle handle;
27 SubjectRef subject;
28 float range, radius, distance;
29 std::size_t projection;
30 };
31 using Key = std::tuple<ecs::Table*, std::type_index, std::uint32_t, std::uint32_t>;
32 struct Group {
33 ecs::EntityHandle target;
34 std::vector<Member> members;
35 };
36 std::map<Key, Group> groups;
37 {
38 auto view = ecs::View<Unit, Unit::Identity, Unit::Navigation, Unit::Motion, Unit::Orders, Unit::Combat,
39 Unit::Crowd, Unit::Durability, Unit::Containment>();
40 for (auto it = view.begin(); it != view.end(); ++it) {
41 auto [identity, nav, motion, orders, combat, crowd, durability, containment] = *it;
42 (void)nav;
43 const auto projection = plan.size();
44 plan.push_back({identity->self, {}, false});
45 if (!durability->alive || containment->container.isBound()) continue;
46 auto current = systems_internal::readCurrent(orders->values);
47 if (!current) return Result<std::size_t>::failure(current.status());
48 if (!current.value() ||
49 (current.value()->kind != OrderKind::Attack && current.value()->kind != OrderKind::AttackMove))
50 continue;
51 const auto handle =
52 current.value()->kind == OrderKind::Attack ? current.value()->targetEntity : combat->target;
53 auto* target = dynamic_cast<Unit*>(ecs::try_get(handle));
54 if (!target || !target->durability()->alive || target->containment()->container.isBound()) continue;
55 const float range = combat->engagementRange;
56 const float touch = crowd->radius + target->crowd()->radius;
57 if (!std::isfinite(touch) || crowd->radius < 0.f || target->crowd()->radius < 0.f ||
58 !std::isfinite(motion->x) || !std::isfinite(motion->y) || !std::isfinite(target->motion()->x) ||
59 !std::isfinite(target->motion()->y))
60 return Result<std::size_t>::failure(
61 Diagnostic::error(DiagnosticCode::InvalidArgument,
62 "Infantry slot geometry must be finite and non-negative", "combat.slots"));
63 // Rifle/artillery standoff remains owned by the weapon policy; infantry melee alone needs front
64 // admission.
65 if (!std::isfinite(range) || range <= 0.f || touch <= 0.f || range > touch * 2.f) continue;
66 const float distance = std::hypot(motion->x - target->motion()->x, motion->y - target->motion()->y);
67 if (distance > range + touch * 2.f) continue;
68 const Key key{handle.table, handle.type, handle.id, handle.generation};
69 auto& group = groups[key];
70 group.target = handle;
71 group.members.push_back(
72 {identity->self, identity->subject, range, crowd->radius, distance, projection});
73 }
74 }
75 std::size_t processed = 0;
76 for (auto& [key, group] : groups) {
77 (void)key;
78 auto* target = dynamic_cast<Unit*>(ecs::try_get(group.target));
79 if (!target) continue;
80 std::sort(group.members.begin(), group.members.end(), [](const Member& a, const Member& b) {
81 if (a.distance != b.distance) return a.distance < b.distance;
82 return a.subject.persistentId() < b.subject.persistentId();
83 });
84 float radius = 0.f, maximumFrontRadius = 0.f, maximumRange = 0.f;
85 for (const auto& member : group.members) {
86 radius = std::max(radius, member.radius);
87 maximumRange = std::max(maximumRange, member.range);
88 maximumFrontRadius = std::max(maximumFrontRadius,
89 std::max(target->crowd()->radius + member.radius, member.range * 0.8f));
90 }
91 const float spacing = 2.f * radius + 0.1f;
92 const int capacity = std::clamp(
93 int(std::clamp(std::floor(2.0 * std::numbers::pi * double(maximumFrontRadius) / spacing), 1.0, 64.0)),
94 1, 64);
95 std::vector<bool> reserved(size_t(capacity), false);
96 std::vector<std::pair<WorldPosition, float>> footprints;
97 footprints.reserve(size_t(capacity));
98 std::size_t waiting = 0;
99 for (const auto& member : group.members) {
100 auto* unit = dynamic_cast<Unit*>(ecs::try_get(member.handle));
101 if (!unit) continue;
102 auto& projected = plan[member.projection];
103 const WorldPosition from{unit->motion()->x, unit->motion()->y};
104 int best = -1;
105 float bestDistance = 0.f;
106 WorldPosition destination{};
107 const float frontRadius = std::max(target->crowd()->radius + member.radius, member.range * 0.8f);
108 for (int slot = 0; slot < capacity && frontRadius <= member.range; ++slot) {
109 if (reserved[size_t(slot)]) continue;
110 const float angle = 2.f * std::numbers::pi_v<float> * float(slot) / float(capacity);
111 const WorldPosition candidate{target->motion()->x + std::cos(angle) * frontRadius,
112 target->motion()->y + std::sin(angle) * frontRadius};
113 const bool occupied = std::any_of(footprints.begin(), footprints.end(), [&](const auto& footprint) {
114 return std::hypot(candidate.x - footprint.first.x, candidate.y - footprint.first.y) <
115 member.radius + footprint.second + 0.1f;
116 });
117 if (occupied) continue;
118 if (pathfinder_ && !systems_internal::isFormationSegmentClear(*pathfinder_, navigationGrid_, from,
119 candidate, member.radius))
120 continue;
121 const float distance = std::hypot(candidate.x - from.x, candidate.y - from.y);
122 if (best < 0 || distance < bestDistance) {
123 best = slot;
124 bestDistance = distance;
125 destination = candidate;
126 }
127 }
128 if (best >= 0) {
129 reserved[size_t(best)] = true;
130 footprints.emplace_back(destination, member.radius);
131 } else {
132 projected.waiting = true;
133 const std::size_t row = waiting / size_t(capacity) + 1, slot = waiting++ % size_t(capacity);
134 const float angle = 2.f * std::numbers::pi_v<float> * float(slot) / float(capacity);
135 const float rearRadius =
136 std::max(maximumFrontRadius + float(row) * spacing, maximumRange + spacing);
137 destination = {target->motion()->x + std::cos(angle) * rearRadius,
138 target->motion()->y + std::sin(angle) * rearRadius};
139 if (pathfinder_ && !systems_internal::isFormationSegmentClear(*pathfinder_, navigationGrid_, from,
140 destination, member.radius))
141 destination = from;
142 }
143 projected.target = destination;
144 ++processed;
145 }
146 }
147 for (const auto& projected : plan) {
148 auto* unit = dynamic_cast<Unit*>(ecs::try_get(projected.handle));
149 if (!unit) continue;
150 unit->navigation()->combatSlotTarget = projected.target;
151 unit->navigation()->combatSlotWaiting = projected.waiting;
152 }
153 return Result<std::size_t>::success(processed);
154 } catch (const std::bad_alloc&) {
155 return Result<std::size_t>::failure(Diagnostic::error(
156 DiagnosticCode::Failed, "Infantry attack-slot preparation allocation failed", "combat.slots"));
157 }
158}
