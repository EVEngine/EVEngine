25bool fromTo(float ax, float ay, float az, float bx, float by, float bz, Quat& out) {
26 const float al = std::sqrt(ax * ax + ay * ay + az * az);
27 const float bl = std::sqrt(bx * bx + by * by + bz * bz);
28 if (al < 1e-6f || bl < 1e-6f) return false;
29 ax /= al;
30 ay /= al;
31 az /= al;
32 bx /= bl;
33 by /= bl;
34 bz /= bl;
35 const float dot = clampf(ax * bx + ay * by + az * bz, -1.f, 1.f);
36 if (dot < -0.9999f) {
37 float ox = std::fabs(ax) < 0.8f ? 1.f : 0.f;
38 float oy = std::fabs(ax) < 0.8f ? 0.f : 1.f;
39 float oz = 0.f;
40 const float cx = ay * oz - az * oy;
41 const float cy = az * ox - ax * oz;
42 const float cz = ax * oy - ay * ox;
43 const float cl = std::sqrt(cx * cx + cy * cy + cz * cz);
44 out = {cx / cl, cy / cl, cz / cl, 0.f};
45 return true;
46 }
47 const float cx = ay * bz - az * by;
48 const float cy = az * bx - ax * bz;
49 const float cz = ax * by - ay * bx;
50 out = {cx, cy, cz, 1.f + dot};
51 const float ql = std::sqrt(out.x * out.x + out.y * out.y + out.z * out.z + out.w * out.w);
52 out.x /= ql;
53 out.y /= ql;
54 out.z /= ql;
55 out.w /= ql;
56 return true;
57}

71TransformTRS mulTRS(const TransformTRS& parent, const TransformTRS& local) {
72 // world = parent * local (TRS, scale ignored in rotation path for FK positions)
73 TransformTRS out;
74 // Rotate local translation by parent rotation, then add parent translation * scales.
75 const float qx = parent.qx, qy = parent.qy, qz = parent.qz, qw = parent.qw;
76 const float lx = local.px * parent.sx;
77 const float ly = local.py * parent.sy;
78 const float lz = local.pz * parent.sz;
79 // q * v
80 const float ix = qw * lx + qy * lz - qz * ly;
81 const float iy = qw * ly + qz * lx - qx * lz;
82 const float iz = qw * lz + qx * ly - qy * lx;
83 const float iw = -qx * lx - qy * ly - qz * lz;
84 out.px = parent.px + (ix * qw + iw * -qx + iy * -qz - iz * -qy);
85 out.py = parent.py + (iy * qw + iw * -qy + iz * -qx - ix * -qz);
86 out.pz = parent.pz + (iz * qw + iw * -qz + ix * -qy - iy * -qx);
87
88 // Quaternion multiply parent * local
89 out.qw = parent.qw * local.qw - parent.qx * local.qx - parent.qy * local.qy - parent.qz * local.qz;
90 out.qx = parent.qw * local.qx + parent.qx * local.qw + parent.qy * local.qz - parent.qz * local.qy;
91 out.qy = parent.qw * local.qy - parent.qx * local.qz + parent.qy * local.qw + parent.qz * local.qx;
92 out.qz = parent.qw * local.qz + parent.qx * local.qy - parent.qy * local.qx + parent.qz * local.qw;
93 out.normalizeRotation();
94
95 out.sx = parent.sx * local.sx;
96 out.sy = parent.sy * local.sy;
97 out.sz = parent.sz * local.sz;
98 return out;
99}

124void AnimPose::blendFrom(const AnimPose* a, const AnimPose* b, float t) {
125 if (!a || !b) throw Exception("AnimPose.blendFrom: pose is null");
126 if (a->getBoneCount() != b->getBoneCount()) {
127 throw Exception("AnimPose.blendFrom: bone count mismatch");
128 }
129 resize(a->getBoneCount());
130 for (int i = 0; i < getBoneCount(); ++i) {
131 const TransformTRS& av = a->locals_[static_cast<size_t>(i)];
132 const TransformTRS& bv = b->locals_[static_cast<size_t>(i)];
133 TransformTRS& out = locals_[static_cast<size_t>(i)];
134#if defined(__SSE2__) || defined(_M_X64) || defined(_M_IX86_FP)
135 const __m128 weight = _mm_set1_ps(clampf(t, 0.f, 1.f));
136 const __m128 oneMinus = _mm_sub_ps(_mm_set1_ps(1.f), weight);
137 alignas(16) float values[4];
138 _mm_store_ps(values, _mm_add_ps(_mm_mul_ps(_mm_setr_ps(av.px, av.py, av.pz, 0.f), oneMinus),
139 _mm_mul_ps(_mm_setr_ps(bv.px, bv.py, bv.pz, 0.f), weight)));
140 out.px = values[0]; out.py = values[1]; out.pz = values[2];
141 _mm_store_ps(values, _mm_add_ps(_mm_mul_ps(_mm_setr_ps(av.sx, av.sy, av.sz, 0.f), oneMinus),
142 _mm_mul_ps(_mm_setr_ps(bv.sx, bv.sy, bv.sz, 0.f), weight)));
143 out.sx = values[0]; out.sy = values[1]; out.sz = values[2];
144#else
145 out.px = lerpf(av.px, bv.px, t); out.py = lerpf(av.py, bv.py, t); out.pz = lerpf(av.pz, bv.pz, t);
146 out.sx = lerpf(av.sx, bv.sx, t); out.sy = lerpf(av.sy, bv.sy, t); out.sz = lerpf(av.sz, bv.sz, t);
147#endif
148 slerpQuat(av.qx, av.qy, av.qz, av.qw, bv.qx, bv.qy, bv.qz, bv.qw, clampf(t, 0.f, 1.f),

247bool AnimPose::aimBone(const AnimSkeleton* skeleton, int boneIndex, float targetX, float targetY, float targetZ,
248 float weight) {
249 if (!skeleton) throw Exception("AnimPose.aimBone: skeleton is null");
250 requireBone(boneIndex);
251 computeWorld(skeleton);
252 const auto& bone = world(boneIndex);
253 const Mat4 mat = Mat4::fromTRS(bone);
254 Quat delta;
255 if (!fromTo(mat.m[8], mat.m[9], mat.m[10], targetX - bone.px, targetY - bone.py, targetZ - bone.pz, delta)) {
256 return false;

259 blendLocalRotation(local(boneIndex), worldToLocalRotation(skeleton, this, boneIndex, desiredWorld), weight);
260 computeWorld(skeleton);

264bool AnimPose::solveTwoBoneIK(const AnimSkeleton* skeleton, int rootBone, int midBone, int tipBone, float targetX,
265 float targetY, float targetZ, float weight) {
266 if (!skeleton) throw Exception("AnimPose.solveTwoBoneIK: skeleton is null");
267 requireBone(rootBone);
268 requireBone(midBone);
269 requireBone(tipBone);
270 if (skeleton->getParent(midBone) != rootBone || skeleton->getParent(tipBone) != midBone) return false;
271
272 bool changed = false;
273 for (int pass = 0; pass < 2; ++pass) {
274 for (int joint : {midBone, rootBone}) {
275 computeWorld(skeleton);
276 const auto& j = world(joint);
277 const auto& t = world(tipBone);
278 Quat delta;
279 if (!fromTo(t.px - j.px, t.py - j.py, t.pz - j.pz, targetX - j.px, targetY - j.py, targetZ - j.pz, delta)) {
280 continue;
281 }
282 const Quat desiredWorld = quatMul(delta, {j.qx, j.qy, j.qz, j.qw});
283 blendLocalRotation(local(joint), worldToLocalRotation(skeleton, this, joint, desiredWorld), weight);
284 changed = true;
285 }
286 }
287 computeWorld(skeleton);
288 return changed;
289}
