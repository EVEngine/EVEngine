11std::string quote(const std::string& s) {
12 std::ostringstream o;
13 o << '"';
14 for (unsigned char c : s) {
15 switch (c) {
16 case '"': o << "\\\""; break;
17 case '\\': o << "\\\\"; break;
18 case '\n': o << "\\n"; break;
19 case '\r': o << "\\r"; break;
20 case '\t': o << "\\t"; break;
21 default:
22 if (c < 32)
23 o << "\\u" << std::hex << std::setw(4) << std::setfill('0') << int(c) << std::dec;
24 else
25 o << char(c);
26 }
27 }
28 return o.str() + '"';
29}
30void properties(std::ostringstream& o, const std::unordered_map<std::string, std::string>& p, bool tiled = false) {
31 o << (tiled ? '[' : '{');
32 bool first = true;
33 for (const auto& v : p) {
34 if (!first) o << ',';
35 first = false;
36 if (tiled)
37 o << "{\"name\":" << quote(v.first) << ",\"type\":\"string\",\"value\":" << quote(v.second) << '}';
38 else
39 o << quote(v.first) << ':' << quote(v.second);
40 }
41 o << (tiled ? ']' : '}');
42}
43void readProperties(eve::json::Value v, std::unordered_map<std::string, std::string>& out) {
44 if (v.isObject())
45 for (const auto& k : v.keys()) out[k] = v.get(k.c_str()).asString();
46 else if (v.isArray())
47 for (size_t i = 0; i < v.size(); ++i) {
48 auto p = v.at(i);
49 out[p.getString("name")] = p.get("value").asString();
50 }

65 static eve::Result<std::unique_ptr<LevelDocument>> readJson(const std::string& s, bool tiled) {
66 std::string error;
67 auto json = eve::json::Document::parse(s, &error);
68 if (!json.valid())
69 return eve::Result<std::unique_ptr<LevelDocument>>::failure(
70 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, std::move(error), {}, {}, "editor.level"));
71 auto r = json.root();
72 int w = r.getInt("width"), h = r.getInt("height");
73 float tw = r.getFloat("tilewidth", 32), th = r.getFloat("tileheight", 32);
74 if (w < 1 || h < 1) {
75 return eve::Result<std::unique_ptr<LevelDocument>>::failure(eve::Diagnostic::error(
76 eve::DiagnosticCode::InvalidArgument, "map dimensions must be positive", {}, {}, "editor.level"));
77 }
78 auto out = std::make_unique<LevelDocument>(w, h, tw, th);
79 out->setOrientation(r.getString("orientation", "orthogonal"));
80 readProperties(r.get("properties"), out->properties());
81 auto ls = r.get("layers");
82 for (size_t i = 0; i < ls.size(); ++i) {
83 auto v = ls.at(i);
84 std::string kind = v.getString("type");
85 bool obj = kind == "objectgroup" || kind == "objects";
86 int li = obj ? out->addObjectLayer(v.getString("name")) : out->addTileLayer(v.getString("name"));
87 auto layerRef = out->layer(li);
88 auto* l = &layerRef->get();
89 l->id = tiled ? std::to_string(v.getInt("id", int(i + 1))) : v.getString("id", l->id);
90 l->visible = v.getBool("visible", true);
91 l->opacity = v.getFloat("opacity", 1);
92 l->offsetX = v.getFloat("offsetx", 0);
93 l->offsetY = v.getFloat("offsety", 0);
94 readProperties(v.get("properties"), l->properties);
95 if (obj) {
96 auto os = v.get("objects");
97 for (size_t j = 0; j < os.size(); ++j) {
98 auto q = os.at(j);
99 int oi =
100 out->addObject(li, q.getString("type", q.getString("class")), q.getFloat("x"), q.getFloat("y"));
101 auto objectRef = out->object(li, oi);
102 auto* ob = &objectRef->get();
103 ob->id = std::to_string(q.getInt("id", int(j + 1)));
104 ob->name = q.getString("name");
105 ob->width = q.getFloat("width");
106 ob->height = q.getFloat("height");
107 ob->rotation = q.getFloat("rotation");
108 ob->visible = q.getBool("visible", true);
109 readProperties(q.get("properties"), ob->properties);
110 }
111 } else {
112 auto data = v.get("data");
113 for (size_t n = 0; n < data.size() && n < size_t(w * h); ++n)
114 l->tiles->setGid(int(n) % w, int(n) / w, data.at(n).asInt());
115 }
116 }
117 return eve::Result<std::unique_ptr<LevelDocument>>::success(std::move(out));
118 }
119 static std::string writeJson(const LevelDocument& d, bool tiled) {
120 std::ostringstream o;
121 o << "{\n ";
122 if (!tiled)
123 o << "\"format\":\"eve.level\",\n \"version\":1,\n ";
124 else
125 o << "\"type\":\"map\",\n \"version\":\"1.10\",\n ";
126 o << "\"orientation\":" << quote(d.getOrientation()) << ",\n \"width\":" << d.getWidth()
127 << ",\"height\":" << d.getHeight() << ",\"tilewidth\":" << d.getTileWidth()
128 << ",\"tileheight\":" << d.getTileHeight() << ",\n \"properties\":";
129 properties(o, d.properties(), tiled);
130 o << ",\n \"layers\":[";
131 for (int i = 0; i < d.getLayerCount(); ++i) {
132 if (i) o << ',';
133 const auto layerRef = d.layer(i);
134 const auto* l = &layerRef->get();
135 o << "\n {\"id\":" << (tiled ? std::to_string(i + 1) : quote(l->id)) << ",\"name\":" << quote(l->name)
136 << ",\"type\":" << quote(l->kind == LevelLayer::Kind::Tiles ? "tilelayer" : "objectgroup")
137 << ",\"visible\":" << (l->visible ? "true" : "false") << ",\"opacity\":" << l->opacity
138 << ",\"offsetx\":" << l->offsetX << ",\"offsety\":" << l->offsetY << ",\"properties\":";
139 properties(o, l->properties, tiled);
140 if (l->tiles) {
141 o << ",\"width\":" << d.getWidth() << ",\"height\":" << d.getHeight() << ",\"data\":[";
142 for (int y = 0; y < d.getHeight(); ++y)
143 for (int x = 0; x < d.getWidth(); ++x) {
144 if (x || y) o << ',';
145 o << l->tiles->getGid(x, y);
146 }
147 o << ']';
148 } else {
149 o << ",\"objects\":[";
150 for (size_t j = 0; j < l->objects.size(); ++j) {
151 if (j) o << ',';
152 auto& q = l->objects[j];
153 o << "{\"id\":" << (tiled ? std::to_string(j + 1) : quote(q.id)) << ",\"name\":" << quote(q.name)
154 << ",\"type\":" << quote(q.type) << ",\"x\":" << q.x << ",\"y\":" << q.y
155 << ",\"width\":" << q.width << ",\"height\":" << q.height << ",\"rotation\":" << q.rotation
156 << ",\"visible\":" << (q.visible ? "true" : "false") << ",\"properties\":";
157 properties(o, q.properties, tiled);
158 o << '}';
159 }
160 o << ']';
161 }
162 o << '}';
163 }
164 o << "\n ]\n}\n";
165 return o.str();
166 }

209std::string LevelFormatRegistry::detect(const std::string& p, const std::string& t) const {
210 for (auto& i : formats_)

212 for (auto& i : formats_)

232eve::Result<std::unique_ptr<LevelDocument>> LevelFormatRegistry::load(const std::string& p,
233 const std::string& id) const {
234 std::ifstream f(p, std::ios::binary);
235 if (!f) {
236 return eve::Result<std::unique_ptr<LevelDocument>>::failure(
237 eve::Diagnostic::error(eve::DiagnosticCode::NotFound, "cannot open level", p, {}, "editor.level"));
238 }
239 std::ostringstream s;
240 s << f.rdbuf();
241 auto fmt = id.empty() ? detect(p, s.str()) : id;
242 return decode(fmt, s.str());
243}
244eve::Result<void> LevelFormatRegistry::save(const std::string& p, const LevelDocument& d, const std::string& id) const {
245 auto fmt = id.empty() ? detect(p, "") : id;
246 if (fmt.empty()) fmt = "eve.level";
247 auto encoded = encode(fmt, d);
248 if (!encoded) return eve::Result<void>::failure(encoded.status());
249 std::ofstream f(p, std::ios::binary);
250 if (!f) {
251 return eve::Result<void>::failure(
252 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "cannot write level", p, {}, "editor.level"));
253 }
254 f << encoded.value();
255 if (!f)
256 return eve::Result<void>::failure(
257 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "failed while writing level", p, {}, "editor.level"));
258 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
259}
