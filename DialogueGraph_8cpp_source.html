68GraphConnectionDecision DialogueGraphDomain::canConnect(const GraphPinRecord& from,
69 const GraphPinRecord& to) const {
70 GraphConnectionDecision decision;
71 decision.allowed = from.direction == GraphPinDirection::Output &&
72 to.direction == GraphPinDirection::Input && from.type == "dialogue.flow" &&
73 to.type == "dialogue.flow" && from.node != to.node;
74 if (!decision.allowed)
75 decision.diagnostics.push_back(eve::editing::ruleDiagnostic(
76 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.dialogue.invalid-connection"),
77 DiagnosticSeverity::Error, "Dialogue flow requires output-to-input pins on distinct nodes"));
78 return decision;
79}

81EditorResult<GraphNodeRecord> DialogueGraphDomain::makeNode(const GraphNodeId& id,
82 const std::string& kind) const {
83 if (id.empty() || !isDialogueKind(kind))

85 "Dialogue node requires an id and supported kind");
86 GraphNodeRecord node;

88 node.type = "dialogue." + kind;
89 node.properties = EditorValue(defaultProperties(kind));
90 node.pins.push_back({GraphPinId(id.value() + ".in"), id, "dialogue.flow", GraphPinDirection::Input});
91 if (kind != "end")
92 node.pins.push_back({GraphPinId(id.value() + ".out"), id, "dialogue.flow",
93 GraphPinDirection::Output});
94 return eve::editing::applied<GraphNodeRecord>(std::move(node));
95}

111DialogueGraphCompileResult DialogueGraphDomain::compile(const GraphDocumentData& graph) const {
112 DialogueGraphCompileResult result;
113 result.documentRevision = graph.revision;
114 if (graph.domain != domain()) {
115 result.status = EditorStatus::Rejected;
116 diagnostic(result, "editor.dialogue.wrong-domain", "Graph is not a dialogue conversation document");
117 return result;
118 }
119 if (graph.schemaVersion != 1) {
120 result.status = EditorStatus::Unsupported;
121 diagnostic(result, "editor.dialogue.unsupported-schema", "Dialogue graph schema version is unsupported");
122 return result;
123 }
124 const auto* idValue = field(graph.parameters, "id");
125 const auto* versionValue = field(graph.parameters, "version");
126 const auto* entryValue = field(graph.parameters, "entry");
127 const auto* parametersValue = field(graph.parameters, "parameters");
128 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
129 const auto* version = versionValue ? versionValue->getIf<int64_t>() : nullptr;
130 const auto* entry = entryValue ? entryValue->getIf<std::string>() : nullptr;
131 const auto* parameters = parametersValue ? parametersValue->getIf<EditorValue::Array>() : nullptr;
132 if (!id || id->empty() || !version || *version <= 0 || !entry || entry->empty() || !parameters) {
133 result.status = EditorStatus::Failed;
134 diagnostic(result, "editor.dialogue.invalid-document",
135 "Dialogue parameters require id, positive version, entry and a parameter array");
136 return result;
137 }
138
139 std::map<GraphNodeId, const GraphNodeRecord*> nodes;
140 std::map<GraphPinId, const GraphNodeRecord*> pinOwners;
141 for (const GraphNodeRecord& node : graph.nodes) {
142 const std::string kind = node.type.starts_with("dialogue.") ? node.type.substr(9) : "";
143 if (!isDialogueKind(kind) && kind != "route")
144 diagnostic(result, "editor.dialogue.unknown-node", "Unknown dialogue node type: " + node.type);
145 if (!nodes.emplace(node.id, &node).second)
146 diagnostic(result, "editor.dialogue.duplicate-node", "Duplicate dialogue node id: " + node.id.value());
147 for (const GraphPinRecord& pin : node.pins)
148 if (!pinOwners.emplace(pin.id, &node).second)
149 diagnostic(result, "editor.dialogue.duplicate-pin", "Duplicate dialogue pin id: " + pin.id.value());
150 }
151 const auto entryNode = nodes.find(GraphNodeId(*entry));
152 if (entryNode == nodes.end() || entryNode->second->type == "dialogue.route")
153 diagnostic(result, "editor.dialogue.invalid-entry", "Entry must reference a dialogue business node");
154
155 std::map<GraphNodeId, std::vector<GraphNodeId>> outgoing;
156 std::map<GraphNodeId, std::vector<GraphNodeId>> incoming;
157 for (const GraphEdgeRecord& edge : graph.edges) {
158 const auto from = pinOwners.find(edge.from);
159 const auto to = pinOwners.find(edge.to);
160 if (from == pinOwners.end() || to == pinOwners.end()) {
161 diagnostic(result, "editor.dialogue.dangling-edge", "Dialogue edge references a missing pin");
162 continue;
163 }
164 const GraphPinRecord* fromPin = nullptr;
165 const GraphPinRecord* toPin = nullptr;
166 for (const GraphPinRecord& pin : from->second->pins)
167 if (pin.id == edge.from) fromPin = &pin;
168 for (const GraphPinRecord& pin : to->second->pins)
169 if (pin.id == edge.to) toPin = &pin;
170 if (!fromPin || !toPin || !canConnect(*fromPin, *toPin).allowed) {
171 diagnostic(result, "editor.dialogue.invalid-edge", "Dialogue edge has invalid direction or type");
172 continue;
173 }
174 outgoing[from->second->id].push_back(to->second->id);
175 incoming[to->second->id].push_back(from->second->id);
176 }
177
178 EditorValue::Array compiledNodes;
179 for (const auto& [nodeId, node] : nodes) {
180 const std::string kind = node->type.substr(9);
181 const auto* properties = node->properties.getIf<EditorValue::Object>();
182 if (!properties) {
183 diagnostic(result, "editor.dialogue.invalid-properties",
184 "Dialogue node properties must be an object: " + nodeId.value());
185 } else {
186 const auto& allowed = propertyKeys(kind);
187 for (const auto& [key, value] : *properties)
188 if (!allowed.contains(key) || !value.getIf<std::string>())
189 diagnostic(result, "editor.dialogue.invalid-property",
190 "Dialogue property is unknown or not a string: " + nodeId.value() + "." + key);
191 for (const std::string& key : allowed)
192 if (!properties->contains(key))
193 diagnostic(result, "editor.dialogue.missing-property",
194 "Dialogue node is missing property: " + nodeId.value() + "." + key);
195 }
196 if (kind == "route") {
197 if (incoming[nodeId].size() != 1 || outgoing[nodeId].size() != 1)
198 diagnostic(result, "editor.dialogue.incomplete-route",
199 "Route requires exactly one source and target: " + nodeId.value());
200 else {
201 const auto source = nodes.at(incoming[nodeId].front());
202 if (source->type != "dialogue.branch" && source->type != "dialogue.choice")
203 diagnostic(result, "editor.dialogue.invalid-route-source",
204 "Routes may only originate from branch or choice nodes: " + nodeId.value());
205 if (nodes.at(outgoing[nodeId].front())->type == "dialogue.route")
206 diagnostic(result, "editor.dialogue.invalid-route-target",
207 "A route must target a business node: " + nodeId.value());
208 }
209 continue;
210 }
211 const bool routed = kind == "branch" || kind == "choice";
212 for (const GraphNodeId& target : outgoing[nodeId]) {
213 const bool targetIsRoute = nodes.at(target)->type == "dialogue.route";
214 if (routed != targetIsRoute)
215 diagnostic(result, "editor.dialogue.invalid-flow",
216 routed ? "Branch and choice outputs must connect through route nodes"
217 : "Only branch and choice nodes may connect to route nodes");
218 }
219 if (kind == "end" && !outgoing[nodeId].empty())
220 diagnostic(result, "editor.dialogue.end-has-output", "End node cannot have outgoing flow");
221 if (!routed && kind != "end" && outgoing[nodeId].size() != 1)
222 diagnostic(result, "editor.dialogue.next-required",
223 "Dialogue node requires exactly one next target: " + nodeId.value());
224 if (routed && outgoing[nodeId].empty())
225 diagnostic(result, "editor.dialogue.route-required",
226 "Branch or choice requires at least one route: " + nodeId.value());
227
228 EditorValue::Object compiled{{"id", nodeId.value()}, {"kind", kind}, {"properties", node->properties}};
229 if (!routed && kind != "end" && outgoing[nodeId].size() == 1)
230 compiled["next"] = outgoing[nodeId].front().value();
231 EditorValue::Array routes;
232 if (routed) {
233 for (const GraphNodeId& routeId : outgoing[nodeId]) {
234 if (nodes.at(routeId)->type != "dialogue.route" || outgoing[routeId].size() != 1) continue;
235 const EditorValue* labelValue = field(nodes.at(routeId)->properties, "label");
236 const auto* label = labelValue ? labelValue->getIf<std::string>() : nullptr;
237 if (!label)
238 diagnostic(result, "editor.dialogue.invalid-route-label",
239 "Route label must be a string: " + routeId.value());
240 else
241 routes.emplace_back(EditorValue::Object{{"id", routeId.value()},
242 {"label", *label},
243 {"target", outgoing[routeId].front().value()}});
244 }
245 }
246 compiled["routes"] = std::move(routes);
247 compiledNodes.emplace_back(std::move(compiled));
248 }
249 std::set<std::string> parameterNames;
250 for (const EditorValue& parameter : *parameters) {
251 const auto* name = parameter.getIf<std::string>();
252 if (!name || name->empty() || !parameterNames.insert(*name).second)
253 diagnostic(result, "editor.dialogue.invalid-parameter",
254 "Conversation parameters must be unique non-empty strings");
255 }
256 if (!result.diagnostics.empty()) {
257 result.status = EditorStatus::Failed;
258 return result;
259 }
260 result.definition = EditorValue::Object{{"id", *id}, {"version", *version}, {"entry", *entry},
261 {"parameters", *parameters}, {"nodes", std::move(compiledNodes)}};
262 result.status = EditorStatus::Applied;
263 return result;
264}
