cmake_minimum_required(VERSION 3.21)
project(McpToolFamilyContract LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(EVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(EVE_DEPENDENCIES "" CACHE PATH "Pinned Poco install")
option(MCP_WITH_PROVIDER "Include optional provider" ON)
option(MCP_SHARED "Test DLL exports" OFF)
set(_kind OBJECT)
if(MCP_SHARED)
  set(_kind SHARED)
endif()
add_library(EVDevTools ${_kind} "${EVE_ROOT}/src/engine/devtools/McpToolFamilies.cpp")
if(MCP_WITH_PROVIDER)
  target_sources(EVDevTools PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/provider.cpp")
endif()
target_compile_features(EVDevTools PUBLIC cxx_std_20)
target_include_directories(EVDevTools PUBLIC "${EVE_ROOT}/src/engine")
target_include_directories(EVDevTools SYSTEM PUBLIC "${EVE_DEPENDENCIES}/include")
target_compile_definitions(EVDevTools PUBLIC POCO_STATIC POCO_NO_AUTOMATIC_LIBS NOMINMAX)
if(MCP_SHARED)
  target_compile_definitions(EVDevTools PUBLIC EVENGINE_MODULE_DLL PRIVATE EVENGINE_EXPORTS_FOUNDATION)
endif()
if(MSVC)
  target_compile_options(EVDevTools PUBLIC /utf-8 /EHsc /W4 /WX /wd4251 /wd4275)
endif()
find_library(POCO_JSON NAMES PocoJSONmdd PocoJSON REQUIRED PATHS "${EVE_DEPENDENCIES}/lib" NO_DEFAULT_PATH)
find_library(POCO_FOUNDATION NAMES PocoFoundationmdd PocoFoundation REQUIRED PATHS "${EVE_DEPENDENCIES}/lib" NO_DEFAULT_PATH)
target_link_libraries(EVDevTools PUBLIC "${POCO_JSON}" "${POCO_FOUNDATION}")
include("${EVE_ROOT}/src/engine/devtools/McpToolFamilies.cmake")
add_executable(mcp_family_contract contract.cpp)
target_link_libraries(mcp_family_contract PRIVATE EVDevTools)
target_compile_definitions(mcp_family_contract PRIVATE MCP_EXPECT_PROVIDER=$<BOOL:${MCP_WITH_PROVIDER}>)
add_test(NAME mcp.family-contract COMMAND mcp_family_contract)
