cmake_minimum_required(VERSION 3.21)
project(EVEngineShaderAssetTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
include(../../cmake/shader_asset_validation.cmake)
# Reuse the Unity import CPU pipeline and run its regression suites unchanged.
add_subdirectory(../asset_import unity-import)
include(../../cmake/TestLevels.cmake)
eve_configure_test_levels(EVE_TEST_LEVEL_FILE)
include("${EVE_TEST_LEVEL_FILE}")
eve_target_test_sources(unity_import_core PRIVATE ../../src/modules/asset/import/ShaderImporter.cpp)
eve_add_test_executable(shader_asset_tests ../asset_shader.cpp)
target_link_libraries(shader_asset_tests PRIVATE unity_import_core)
if(MSVC)
    target_compile_options(shader_asset_tests PRIVATE /utf-8 /EHsc /W4)
endif()
add_test(NAME asset.shader.contract COMMAND shader_asset_tests)
eve_label_test_group(asset.shader.contract asset_shader.cpp)
set_tests_properties(asset.shader.contract PROPERTIES FAIL_REGULAR_EXPRESSION "Assertion Failed")
eve_add_test_executable(shader_validation_tests ShaderValidationTests.cpp ../../src/modules/asset/graphics/ShaderAssetValidation.cpp)
target_link_libraries(shader_validation_tests PRIVATE unity_import_core)
eve_configure_shader_asset_validation(shader_validation_tests)
if(MSVC)
    target_compile_options(shader_validation_tests PRIVATE /utf-8 /EHsc)
endif()
add_test(NAME asset.shader.gpuAdmission COMMAND shader_validation_tests)
eve_label_test_group(asset.shader.gpuAdmission ShaderValidationTests.cpp)
eve_add_test_executable(shader_validation_absent ShaderValidationAbsent.cpp ../../src/modules/asset/graphics/ShaderAssetValidation.cpp)
target_link_libraries(shader_validation_absent PRIVATE unity_import_core)
if(MSVC)
    target_compile_options(shader_validation_absent PRIVATE /utf-8 /EHsc)
endif()
add_test(NAME asset.shader.validationProviderAbsent COMMAND shader_validation_absent)
eve_label_test_group(asset.shader.validationProviderAbsent ShaderValidationAbsent.cpp)
eve_add_test_executable(shader_package_probe ShaderPackageProbe.cpp)
include(../../cmake/TestImpact.cmake)
eve_enable_test_impact(TARGETS shader_asset_tests shader_validation_tests shader_validation_absent
    unity_source_tests unity_collection_tests unity_fbx_tests unity_fbx_unavailable)
target_link_libraries(shader_package_probe PRIVATE unity_import_core)
if(MSVC)
    target_compile_options(shader_package_probe PRIVATE /utf-8 /EHsc)
endif()
