54ui_editing::Result<void> UiThemeEditor::configureWorkspace(editor::EditorWorkspace& workspace) const {
55 editor::EditorWorkspace candidate = workspace;
56 struct Panel {
57 const char* id;
58 const char* title;
59 const char* region;
60 const char* context;
61 int order;
62 };
63 constexpr Panel panels[] = {
64 {"ui.themes", "Themes", "left", "list", 100},
65 {"ui.preview", "Theme Preview", "center", "preview", 100},
66 {"ui.inspector", "Theme Inspector", "right", "inspector", 100},
67 };
68 for (const auto& panel : panels) {
69 if (!candidate.registerPanel(panel.id, panel.title, panel.region, panel.order) ||
70 !candidate.setPanelCapability(panel.id, "ui.theme") || !candidate.setPanelContext(panel.id, panel.context))
71 return editorError(ui_editing::EditorStatus::Rejected, "editor.ui-theme.workspace-conflict",
72 "Could not install the UI theme workspace composition");
73 }
74 if (!candidate.activatePanel("ui.preview"))
75 return editorError(ui_editing::EditorStatus::Rejected, "editor.ui-theme.workspace-activate",
76 "Could not activate the UI theme preview panel");
77 workspace = std::move(candidate);
78 return eve::editing::applied<void>();
79}

94ui_editing::Result<void> UiThemeEditor::commit(ui_editing::Result<ui_editing::DomainOperation> operation,
95 std::string label) {
96 if (!operation.ok()) return ui_editing::Result<void>::failure(operation.status());
97 editor::TransactionSpec spec;
98 spec.id = editor::TransactionId("ui.theme.tx." + std::to_string(++txSequence_));
99 spec.label = std::move(label);
100 spec.target = editor::TargetId(target_.targetId());
101 spec.baseRevision = target_.revision();
102 auto begun = transactions_.begin(std::move(spec));
103 if (!begun.ok()) return editorError(begun.code(), "editor.ui-theme.begin", "Could not begin the theme transaction");
104 auto appended = transactions_.append(std::move(operation).takeValue());
105 if (!appended.ok()) {
106 auto discarded = transactions_.discard();
107 if (!discarded.ok()) discarded.ignore("pending theme transaction already inactive");
108 return ui_editing::Result<void>::failure(appended.status());
109 }
110 auto committed = transactions_.commit();
111 if (!committed.ok()) return ui_editing::Result<void>::failure(committed.status());
112 return refreshPreview();
113}

231float UiThemeEditor::getColorChannel(const std::string& path, int channel) const {
232 auto value = target_.read(selection(), ui_editing::PropertyPath(path));
233 if (value.state != ui_editing::PropertyReadState::Value) return 0.0f;
234 const auto* array = value.value.getIf<ui_editing::EditorValue::Array>();
235 if (!array || channel < 0 || static_cast<std::size_t>(channel) >= array->size()) return 0.0f;
236 const auto* number = (*array)[static_cast<std::size_t>(channel)].getIf<double>();
237 return number ? static_cast<float>(*number) : 0.0f;
