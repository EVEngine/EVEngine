25 auto payload = reader_.read(image, "eve.image/2", capabilities, limits.maximumDecodedBytes);
26 if (!payload) return Result<LoadedGraphicsImage>::failure(payload.status());
27 const asset::RuntimeAssetChunk* bulk = nullptr;
28 for (const auto& chunk : payload.value().chunks) {
29 if (chunk.kind != asset::EvpackChunkKind::Bulk) continue;
30 if (bulk)
31 return failure<LoadedGraphicsImage>(DiagnosticCode::ParseError,
32 "eve.image/2 must contain exactly one bulk chunk");
33 bulk = &chunk;
34 }
35 static constexpr std::uint8_t magic[] = {'E', 'V', 'I', 'M', 'G', 0, 1, 0};
36 if (!bulk || bulk->bytes.size() < 24 ||
37 !std::equal(std::begin(magic), std::end(magic), bulk->bytes.begin()))
38 return failure<LoadedGraphicsImage>(DiagnosticCode::ParseError,
39 "EVIMG bulk header is missing or invalid");
40 const std::uint32_t width = little32(bulk->bytes, 8);
41 const std::uint32_t height = little32(bulk->bytes, 12);
42 const std::uint32_t flags = little32(bulk->bytes, 16);
43 const std::uint32_t reserved = little32(bulk->bytes, 20);
44 const std::uint64_t pixels = std::uint64_t(width) * height;
45 if (width == 0 || height == 0 || width > limits.maximumDimension ||
46 height > limits.maximumDimension || pixels > limits.maximumPixels || flags > 1 ||
47 reserved != 0 || pixels > (std::numeric_limits<std::uint64_t>::max() - 24) / 4 ||
48 pixels * 4 + 24 != bulk->bytes.size())
49 return failure<LoadedGraphicsImage>(DiagnosticCode::InvalidArgument,
50 "EVIMG dimensions, flags, or byte size are invalid");
51 auto uploaded = factory_.uploadRgba8(width, height, bulk->bytes.data() + 24, flags == 1);
52 if (!uploaded) return Result<LoadedGraphicsImage>::failure(uploaded.status());
53 return Result<LoadedGraphicsImage>::success(

55}
