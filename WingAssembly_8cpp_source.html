48Result<ForceBatch> prepareSurfaces(W& world, PhysicsLink link, std::span<const WingSurface> surfaces,
49 std::span<const double> controls, ForceVector wind) {
50 constexpr bool planar = std::is_same_v<W, World>;
51 if (controls.size() != surfaces.size() || !isFinite(wind) || (planar && wind.z != 0)) return invalid<ForceBatch>();
52 auto resolved = link.resolve(world);
53 if (!resolved) return Result<ForceBatch>::failure(resolved.status());
54 auto& body = *resolved.value();
55 if (!body.isActive() || body.getType() != "dynamic") return invalid<ForceBatch>();
56 std::vector<ForceLoad> loads;
57 loads.reserve(surfaces.size());
58 for (std::size_t i = 0; i < surfaces.size(); ++i) {
59 const auto& surface = surfaces[i];
60 if (!isFinite(controls[i]) ||
61 (planar && (surface.center.z != 0 || surface.model.chord.z != 0 || surface.model.normal.z != 0)))
62 return invalid<ForceBatch>();
63 const double angle = std::remainder(controls[i], 2 * std::numbers::pi);
64 const double c = std::cos(angle), s = std::sin(angle);
65 auto model = surface.model;
66 model.chord = direction(body, add(scale(surface.model.chord, c), scale(surface.model.normal, s)));
67 model.normal = direction(body, add(scale(surface.model.normal, c), scale(surface.model.chord, -s)));
68 const auto center = point(body, surface.center);
69 if (!fits(center)) return invalid<ForceBatch>();
70 const auto relative = add(velocity(body, center), scale(wind, -1));
71 auto sampled = ForceModels::wing(model, relative);
72 if (!sampled) return Result<ForceBatch>::failure(sampled.status());
73 loads.push_back({link, sampled.value().force, center, {}});
74 }
75 return ForceBatch::prepare(world, loads);
76}
