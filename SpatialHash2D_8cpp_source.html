85bool SpatialHash2D::update(int id, float minX, float minY, float maxX, float maxY) {
86 auto it = items_.find(id);
87 if (it == items_.end()) return false;
88 const AABB2 next = makeAABB2(minX, minY, maxX, maxY);
89 if (!next.valid()) return false;
90 int oldMinX, oldMinY, oldMaxX, oldMaxY, nextMinX, nextMinY, nextMaxX, nextMaxY;
91 cellRange(it->second, oldMinX, oldMinY, oldMaxX, oldMaxY);
92 cellRange(next, nextMinX, nextMinY, nextMaxX, nextMaxY);
93 if (oldMinX == nextMinX && oldMinY == nextMinY && oldMaxX == nextMaxX && oldMaxY == nextMaxY) {
94 // Retain the legacy update-to-end order without rebuilding occupied buckets.
95 for (int y = oldMinY; y <= oldMaxY; ++y) {
96 for (int x = oldMinX; x <= oldMaxX; ++x) {
97 auto &bucket = cells_.at(cellKey2(x, y));
98 const auto position = std::find(bucket.begin(), bucket.end(), id);
99 std::rotate(position, position + 1, bucket.end());
100 }
101 }
102 it->second = next;
103 return true;
104 }
105 eraseCells(id, it->second);
106 it->second = next;
107 insertCells(id, next);
108 return true;
109}

111void SpatialHash2D::queryCells(int minCX, int minCY, int maxCX, int maxCY, const AABB2 *rect, float cx, float cy,
112 float radius, bool useCircle, bool usePoint) {
113 results_.clear();
114 for (int y = minCY; y <= maxCY; ++y) {
115 for (int x = minCX; x <= maxCX; ++x) {
116 auto it = cells_.find(cellKey2(x, y));
117 if (it == cells_.end()) continue;
118 for (int id : it->second) {
119 auto item = items_.find(id);
120 if (item == items_.end()) continue;
121 const AABB2 &b = item->second;
122 // Visit an item only in its first bucket intersecting this query.
123 // This preserves traversal order without allocating a set for every query.
124 const int firstX = std::max(minCX, static_cast<int>(std::floor(b.minX / cellSize_)));
125 const int firstY = std::max(minCY, static_cast<int>(std::floor(b.minY / cellSize_)));
126 if (x != firstX || y != firstY) continue;
127 bool hit = false;
128 if (rect) {
129 hit = b.intersectsAABB(*rect);
130 } else if (useCircle) {
131 hit = b.intersectsCircle(cx, cy, radius);
132 } else if (usePoint) {
133 hit = b.containsPoint(cx, cy);
134 }
135 if (hit) results_.addUnchecked(id);
136 }
137 }
138 }
139}
