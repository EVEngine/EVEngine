49eve::Result<int> ShopCatalogue::replaceFromJsonStrict(const std::string &json) {
50 std::string parseError;
51 const auto document = eve::json::Document::parse(json, &parseError);
52 if (!document.valid()) return failure(parseError.empty() ? "invalid JSON" : parseError, "$");
53 const auto root = document.root();
54 if (!root.isArray() || root.size() == 0) return failure("shop catalogue must be a non-empty array", "$");
55
56 std::vector<ShopOffer> proposed;
57 std::unordered_map<std::string, std::size_t> proposedIndex;
58 proposed.reserve(root.size());
59 for (std::size_t index = 0; index < root.size(); ++index) {
60 const auto object = root.at(index);
61 const std::string path = "$[" + std::to_string(index) + "]";
62 if (!object.isObject()) return failure("shop offer must be an object", path);
63 if (!onlyKnownFields(object)) return failure("shop offer contains an unknown field", path);
64 const auto id = object.get("id");
65 const auto itemIdValue = object.get("itemId");
66 const auto name = object.get("name");
67 const auto description = object.get("desc");
68 const auto buyPrice = object.get("buyPrice");
69 const auto sellPrice = object.get("sellPrice");
70 if (!id.isString() || !validId(id.asString()))
71 return failure("offer id must be a stable non-empty id", path + ".id");
72 const std::string itemId = itemIdValue.isString() ? itemIdValue.asString() : std::string{};
73 if (!validId(itemId)) return failure("itemId must be a stable non-empty id", path + ".itemId");
74 if (!inventory::ItemRegistry::find(itemId))
75 return failure("shop offer references an unknown inventory item", path + ".itemId");
76 if (!name.isString() || name.asString().empty() || name.asString().size() > 512)
77 return failure("name must be a non-empty string of at most 512 bytes", path + ".name");
78 if (!description.isString() || description.asString().size() > 2048)
79 return failure("desc must be a string of at most 2048 bytes", path + ".desc");
80 if (!buyPrice.isInt64() || buyPrice.asInt64() < 0 || buyPrice.asInt64() > INT_MAX)
81 return failure("buyPrice must be a non-negative integer", path + ".buyPrice");
82 if (!sellPrice.isInt64() || sellPrice.asInt64() < 0 || sellPrice.asInt64() > buyPrice.asInt64())
83 return failure("sellPrice must be a non-negative integer no greater than buyPrice", path + ".sellPrice");
84 if (proposedIndex.count(id.asString()) != 0) return failure("duplicate offer id", path + ".id");
85 ShopOffer offer;
86 offer.id = id.asString();
87 offer.itemId = itemId;
88 offer.displayName = name.asString();
89 offer.description = description.asString();
90 offer.buyPrice = static_cast<int>(buyPrice.asInt64());
91 offer.sellPrice = static_cast<int>(sellPrice.asInt64());
92 proposedIndex.emplace(offer.id, proposed.size());
93 proposed.push_back(std::move(offer));
94 }
95 offers() = std::move(proposed);
96 offerIndex() = std::move(proposedIndex);
97 return eve::Result<int>::success(static_cast<int>(offers().size()));
98}

128eve::Result<double> validateTrade(const GameState &gameState, const std::string &currencyId, const std::string &itemId,
129 int quantity, double unitPrice, bool purchase) {
130 if (currencyId.empty() || itemId.empty() || quantity <= 0 || !std::isfinite(unitPrice) || unitPrice < 0.0)
131 return eve::Result<double>::failure(eve::Diagnostic::error(
132 eve::DiagnosticCode::InvalidArgument,
133 "shop trade requires currency/item ids, positive quantity, and finite non-negative unit price", "trade",
134 {{"itemId", itemId}}, "rpg.shop-transaction"));
135 const double total = unitPrice * static_cast<double>(quantity);
136 const double balance = gameState.getVariable(currencyId);
137 const double result = purchase ? balance - total : balance + total;
138 if (!std::isfinite(total) || !std::isfinite(balance) || !std::isfinite(result))
139 return eve::Result<double>::failure(eve::Diagnostic::error(
140 eve::DiagnosticCode::InvalidArgument, "shop trade would produce a non-finite currency value", "unitPrice",
141 {{"itemId", itemId}}, "rpg.shop-transaction"));
142 if (purchase && balance < total)
143 return eve::Result<double>::failure(eve::Diagnostic::error(
144 eve::DiagnosticCode::PreconditionViolation, "shop purchase has insufficient currency", "currency",
145 {{"itemId", itemId}}, "rpg.shop-transaction"));
146 return eve::Result<double>::success(total);
147}

177eve::Result<int> ShopTransaction::buy(GameState &gameState, inventory::Bag &bag, const std::string &currencyId,
178 const std::string &itemId, int quantity, double unitPrice) {
179 auto validated = validateTrade(gameState, currencyId, itemId, quantity, unitPrice, true);
180 if (!validated.ok()) return eve::Result<int>::failure(validated.status());
181 const double total = std::move(validated).takeValue();
182 auto prepared = inventory::InventorySystem::prepareAddBatch(&bag, {{itemId, quantity}});
183 if (!prepared.ok()) return eve::Result<int>::failure(prepared.status());
184 auto before = gameState.snapshotJson();
185 if (!before.ok()) return eve::Result<int>::failure(before.status());
186 std::string snapshot = std::move(before).takeValue();
187
188 gameState.addVariable(currencyId, -total);
189 auto committed = inventory::InventorySystem::commitAddBatch(std::move(prepared).takeValue());
190 if (!committed.ok()) return rollbackFailure(gameState, snapshot, committed.status(), itemId);
191 return committed;
192}
