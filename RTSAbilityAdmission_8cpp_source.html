79 Result<void> prepare(const transaction::TransactionContext&) override {
80 if (phase_ != Phase::Idle) return conflict("ability stage is already in flight");
81 auto* caster = resolve();
82 if (!caster) return conflict("ability caster is stale");
83 auto valid = AbilitySystem::validateCast(*caster, spec_, target_, point_);
84 if (!valid) return valid;
85 before_ = *caster->abilities();
86 after_ = before_;
87 auto found = std::find_if(after_.cooldowns.begin(), after_.cooldowns.end(),
88 [&](const auto& c) { return c.id == spec_.id; });
89 if (found == after_.cooldowns.end())
90 after_.cooldowns.push_back({spec_.id, spec_.cooldown});
91 else
92 found->remaining = spec_.cooldown;
93 after_.channel = Unit::Abilities::Channel{
94 spec_, target_,
95 point_, caster->durability()->state.health,
96 spec_.castTime, spec_.channelTickInterval > 0 ? spec_.channelTickInterval : spec_.castTime};
97 phase_ = Phase::Prepared;
98 return Result<void>::success();
99 }
100 Result<void> commit(const transaction::TransactionContext&) override {
101 if (phase_ != Phase::Prepared) return conflict("ability stage is not prepared");
102 auto* caster = resolve();
103 if (!caster) return conflict("ability caster is stale");
104 auto valid = AbilitySystem::validateCast(*caster, spec_, target_, point_);
105 if (!valid) return valid;
106 if (!sameCooldowns(caster->abilities()->cooldowns, before_.cooldowns))
107 return conflict("ability state changed during admission");
108 std::swap(*caster->abilities(), after_);
109 phase_ = Phase::Committed;
110 return Result<void>::success(Status::success(StatusCode::Applied));
111 }
112 Result<void> rollback(const transaction::TransactionContext&) override {
113 if (phase_ != Phase::Prepared) return conflict("ability stage is not prepared");
114 phase_ = Phase::Terminal;
115 return Result<void>::success();
116 }
117 Result<void> compensate(const transaction::TransactionContext&) override {
118 if (phase_ != Phase::Committed) return conflict("ability stage is not committed");
119 auto* caster = resolve();
120 if (!caster) return conflict("ability caster is stale during compensation");
121 const auto& channel = caster->abilities()->channel;
122 if (!channel || channel->spec.id != spec_.id || channel->remaining != spec_.castTime)
123 return conflict("ability was advanced before compensation");
124 std::swap(*caster->abilities(), after_);
125 phase_ = Phase::Terminal;
126 return Result<void>::success();
127 }
