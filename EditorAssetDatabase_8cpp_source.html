17bool containsInsensitive(std::string value, std::string text) {
18 auto lower = [](unsigned char character) { return static_cast<char>(std::tolower(character)); };
19 std::transform(value.begin(), value.end(), value.begin(), lower);
20 std::transform(text.begin(), text.end(), text.begin(), lower);
21 return value.find(text) != std::string::npos;
22}

26Result<AssetRecord> MemoryAssetDatabase::publish(AssetRecord record, std::vector<AssetDependency> dependencies) {
27 if (record.guid.empty() || record.logicalUri.empty() || record.typeId.empty())
28 return eve::editing::failed<AssetRecord>(EditorStatus::Rejected, RuleId("editor.asset.invalid-record"),
29 "Asset GUID, logical URI and type are required");
30 const auto uriOwner = uriToGuid_.find(record.logicalUri);
31 if (uriOwner != uriToGuid_.end() && uriOwner->second != record.guid)
32 return eve::editing::failed<AssetRecord>(EditorStatus::Conflict, RuleId("editor.asset.uri-conflict"),
33 "Another asset already owns this logical URI");
34 for (const AssetDependency& dependency : dependencies) {
35 if (dependency.from != record.guid || dependency.to.empty())
36 return eve::editing::failed<AssetRecord>(EditorStatus::Rejected, RuleId("editor.asset.invalid-dependency"),
37 "Published dependencies must originate from the product asset");
38 }
39 record.status = AssetStatus::Ready;
40 const auto existing = records_.find(record.guid);
41 if (existing != records_.end() && existing->second.logicalUri != record.logicalUri)
42 uriToGuid_.erase(existing->second.logicalUri);
43 uriToGuid_.insert_or_assign(record.logicalUri, record.guid);
44 records_.insert_or_assign(record.guid, record);
45 std::erase_if(dependencies_, [&](const AssetDependency& dependency) { return dependency.from == record.guid; });
46 dependencies_.insert(dependencies_.end(), dependencies.begin(), dependencies.end());
47 ++generation_;
48 return eve::editing::applied<AssetRecord>(std::move(record));
49}

51Result<std::vector<AssetRecord>> MemoryAssetDatabase::publishBatch(std::vector<AssetPublication> publications) {
52 if (publications.empty())
53 return eve::editing::failed<std::vector<AssetRecord>>(EditorStatus::Rejected,
54 RuleId("editor.asset.empty-publication"),
55 "An asset publication batch must not be empty");
56 auto stagedRecords = records_;
57 auto stagedUris = uriToGuid_;
58 auto stagedDependencies = dependencies_;
59 std::vector<AssetRecord> published;
60 std::unordered_map<AssetGuid, bool, StrongEditorIdHash<AssetGuid>> batchGuids;
61 published.reserve(publications.size());
62 for (AssetPublication& publication : publications) {
63 AssetRecord& record = publication.record;
64 if (record.guid.empty() || record.logicalUri.empty() || record.typeId.empty())
65 return eve::editing::failed<std::vector<AssetRecord>>(EditorStatus::Rejected,
66 RuleId("editor.asset.invalid-record"),
67 "Asset GUID, logical URI and type are required");
68 if (!batchGuids.emplace(record.guid, true).second)
69 return eve::editing::failed<std::vector<AssetRecord>>(
70 EditorStatus::Conflict, RuleId("editor.asset.duplicate-batch-guid"),
71 "An asset publication batch contains a duplicate GUID");
72 const auto uriOwner = stagedUris.find(record.logicalUri);
73 if (uriOwner != stagedUris.end() && uriOwner->second != record.guid)
74 return eve::editing::failed<std::vector<AssetRecord>>(EditorStatus::Conflict,
75 RuleId("editor.asset.uri-conflict"),
76 "Another asset already owns this logical URI");
77 for (const AssetDependency& dependency : publication.dependencies) {
78 if (dependency.from != record.guid || dependency.to.empty())
79 return eve::editing::failed<std::vector<AssetRecord>>(
80 EditorStatus::Rejected, RuleId("editor.asset.invalid-dependency"),
81 "Published dependencies must originate from the product asset");
82 }
83 record.status = AssetStatus::Ready;
84 const auto existing = stagedRecords.find(record.guid);
85 if (existing != stagedRecords.end() && existing->second.logicalUri != record.logicalUri)
86 stagedUris.erase(existing->second.logicalUri);
87 stagedUris.insert_or_assign(record.logicalUri, record.guid);
88 stagedRecords.insert_or_assign(record.guid, record);
89 std::erase_if(stagedDependencies,
90 [&](const AssetDependency& dependency) { return dependency.from == record.guid; });
91 stagedDependencies.insert(stagedDependencies.end(), publication.dependencies.begin(),
92 publication.dependencies.end());
93 published.push_back(std::move(record));
94 }
95 records_ = std::move(stagedRecords);
96 uriToGuid_ = std::move(stagedUris);
97 dependencies_ = std::move(stagedDependencies);
98 ++generation_;
99 return eve::editing::applied<std::vector<AssetRecord>>(std::move(published));
100}

117Result<AssetPage<AssetRecord>> MemoryAssetDatabase::query(const AssetQuery& query, std::size_t offset,
118 std::size_t limit,
119 std::optional<std::uint64_t> generation) const {
120 if (generation && *generation != generation_)
121 return eve::editing::failed<AssetPage<AssetRecord>>(EditorStatus::Conflict, RuleId("editor.asset.page-expired"),
122 "Asset index changed while paging");
123 std::vector<AssetRecord> matches;
124 for (const auto& [guid, record] : records_) {
125 (void)guid;
126 if (query.status && record.status != *query.status) continue;
127 if (!query.typeIds.empty() &&
128 std::find(query.typeIds.begin(), query.typeIds.end(), record.typeId) == query.typeIds.end())
129 continue;
130 if (!query.pathPrefixes.empty()) {
131 bool prefix = false;
132 for (const std::string& candidate : query.pathPrefixes)
133 if (record.logicalUri.starts_with(candidate)) prefix = true;
134 if (!prefix) continue;
135 }
136 bool tags = true;
137 for (const std::string& tag : query.tagsAll)
138 if (std::find(record.tags.begin(), record.tags.end(), tag) == record.tags.end()) tags = false;
139 if (!tags) continue;
140 if (!query.text.empty() && !containsInsensitive(record.logicalUri, query.text) &&
141 !containsInsensitive(record.typeId, query.text))
142 continue;
143 matches.push_back(record);
144 }
145 std::sort(matches.begin(), matches.end(),
146 [](const AssetRecord& left, const AssetRecord& right) { return left.logicalUri < right.logicalUri; });
147 AssetPage<AssetRecord> page;
148 page.generation = generation_;
149 if (offset < matches.size()) {
150 const std::size_t end = std::min(matches.size(), offset + limit);
151 page.values.assign(matches.begin() + static_cast<std::ptrdiff_t>(offset),
152 matches.begin() + static_cast<std::ptrdiff_t>(end));
153 page.nextOffset = end;
154 page.hasMore = end < matches.size();
155 }
156 return eve::editing::applied<AssetPage<AssetRecord>>(std::move(page));
157}

186Result<AssetRecord> ImportCoordinator::publish(const ImportTicket& ticket, ImportProduct product) {
187 if (!database_)
188 return eve::editing::failed<AssetRecord>(EditorStatus::Failed, RuleId("editor.import.missing-database"),
189 "Import coordinator has no asset database");
190 const auto generation = generations_.find(ticket.asset);
191 if (generation == generations_.end() || generation->second != ticket.generation)
192 return eve::editing::failed<AssetRecord>(EditorStatus::Conflict, RuleId("editor.import.stale-product"),
193 "Importer product was superseded by a newer source generation");
194 AssetRecord& record = product.record;
195 if (record.guid != ticket.asset || record.sourceHash != ticket.sourceHash ||
196 record.importerId != ticket.importerId || record.importerVersion != ticket.importerVersion)
197 return eve::editing::failed<AssetRecord>(EditorStatus::Conflict, RuleId("editor.import.ticket-mismatch"),
198 "Importer product identity does not match its request ticket");
199 if (record.sourceUri.empty() || record.logicalUri.empty() || record.typeId.empty())
200 return eve::editing::failed<AssetRecord>(EditorStatus::Rejected, RuleId("editor.import.incomplete-product"),
201 "Importer product lacks source, logical URI or asset type");
202 std::set<std::string> artifacts;
203 for (const std::string& artifact : record.artifacts)
204 if (artifact.empty() || !artifacts.insert(artifact).second)
205 return eve::editing::failed<AssetRecord>(EditorStatus::Rejected, RuleId("editor.import.invalid-artifact"),
206 "Importer artifacts must be unique and non-empty");
207 if (record.artifacts.empty())
208 return eve::editing::failed<AssetRecord>(EditorStatus::Rejected, RuleId("editor.import.missing-artifact"),
209 "Importer product must contain at least one published artifact");
210 if (std::any_of(record.diagnostics.begin(), record.diagnostics.end(), [](const EditorDiagnostic& diagnostic) {
211 return diagnostic.severity() == DiagnosticSeverity::Error ||
212 diagnostic.severity() == DiagnosticSeverity::Fatal;
213 }))
214 return eve::editing::failed<AssetRecord>(EditorStatus::Rejected, RuleId("editor.import.product-errors"),
215 "Importer product contains error diagnostics");
216 auto published = database_->publish(std::move(record), std::move(product.dependencies));
217 if (published.ok()) generations_.erase(ticket.asset);
218 return published;
219}

237Result<void> EditorCommandService::checkExecutionPolicy(const CommandDescriptor& descriptor, const CommandId& id,
238 CommandSource source, const EditorValue& payload,
239 const HostProfile& profile) const {
240 if (!profile.allowsCommand(id))
241 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.command.profile-denied"),
242 "Command is unavailable in this host: " + id.value());
243 if (!profile.hasFeatures(descriptor.requiredFeatures))
244 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.command.feature-denied"),

248 "Command is not available to automation: " + id.value());
249 if (!payload.isWithinLimits(kMaxPayloadDepth, kMaxPayloadElements, profile.maxPayloadBytes()))
250 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.command.payload-limit"),
251 "Command payload exceeds host limits: " + id.value());
252 return eve::editing::applied<void>();

342Result<EditorValue> EditorCommandService::execute(const CommandId& id, const CommandContext& context,
343 const EditorValue& payload) const {
344 auto it = std::find_if(commands_.begin(), commands_.end(),
345 [&](const Registration& entry) { return entry.descriptor.id == id; });
346 if (it == commands_.end())
347 return error(EditorStatus::NotFound, "editor.command.not-found", "Command is not registered: " + id.value());
348 if (!context.profile)
349 return error(EditorStatus::Failed, "editor.command.missing-profile", "Command context has no host profile");
350 if (auto policy = checkExecutionPolicy(it->descriptor, id, context.source, payload, *context.profile); !policy.ok())
351 return Result<EditorValue>::failure(std::move(policy).status());
352
353 if (!it->handler)
354 return error(EditorStatus::Unsupported, "editor.command.requires-plan",
355 "Command must be planned before it can be executed: " + id.value());
356 try {
357 return it->handler(context, payload);
358 } catch (const std::exception& exception) {
359 return error(EditorStatus::Failed, "editor.command.handler-exception",
360 "Command handler failed: " + std::string(exception.what()));
361 } catch (...) {
362 return error(EditorStatus::Failed, "editor.command.handler-exception",
363 "Command handler failed with an unknown exception");
364 }
365}

367Result<CommandPlan> EditorCommandService::plan(const CommandRequest& request, const HostProfile& profile) const {
368 auto failure = [](EditorStatus status, const char* rule, std::string message) {
369 return eve::editing::failed<CommandPlan>(status, RuleId(rule), std::move(message));
370 };
371 auto it = std::find_if(commands_.begin(), commands_.end(),
372 [&](const Registration& entry) { return entry.descriptor.id == request.id; });
373 if (it == commands_.end())
374 return failure(EditorStatus::NotFound, "editor.command.not-found",
375 "Command is not registered: " + request.id.value());
376 if (auto policy = checkExecutionPolicy(it->descriptor, request.id, request.source, request.payload, profile);
377 !policy.ok())
378 return Result<CommandPlan>::failure(std::move(policy).status());
379 if (request.expectedRevision && *request.expectedRevision != request.context.targetRevision)
380 return failure(EditorStatus::Conflict, "editor.command.revision-conflict",
381 "Expected revision does not match the captured context");
382 if (!it->planner)
383 return failure(EditorStatus::Unsupported, "editor.command.planning-unsupported",
384 "Command does not provide a planning handler");
385 try {
386 Result<CommandPlan> result = it->planner(request);
387 if (result.ok()) {
388 result.value().command = request.id;
389 result.value().target = request.context.target;
390 result.value().baseRevision = request.context.targetRevision;
391 if (result.value().targetGeneration == 0)
392 result.value().targetGeneration = request.context.targetGeneration;
393 result.value().commandGeneration = it->generation;
394 result.value().plannedPayload = request.payload;
395 result.value().plannedSource = request.source;
396 if (result.value().id.empty())
397 result.value().id = PlanId(request.id.value() + ".plan." + std::to_string(++planSequence_));
398 }
399 return result;
400 } catch (const std::exception& exception) {
401 return failure(EditorStatus::Failed, "editor.command.planner-exception", exception.what());
402 } catch (...) {
403 return failure(EditorStatus::Failed, "editor.command.planner-exception",
404 "Command planner failed with an unknown exception");
405 }

408Result<TransactionReceipt> EditorCommandService::executePlan(const CommandRequest& request, const CommandPlan& plan,
409 const HostProfile& profile) const {
410 auto failure = [](EditorStatus status, const char* rule, std::string message) {
411 return eve::editing::failed<TransactionReceipt>(status, RuleId(rule), std::move(message));
412 };
413 if (plan.id.empty() || plan.command != request.id || plan.target != request.context.target)
414 return failure(EditorStatus::Rejected, "editor.command.invalid-plan",
415 "Plan identity or target does not match the request");
416 if (plan.baseRevision != request.context.targetRevision)
417 return failure(EditorStatus::Conflict, "editor.command.stale-plan",
418 "Plan was produced for another target revision");
419 if (plan.targetGeneration != request.context.targetGeneration)
420 return failure(EditorStatus::Conflict, "editor.command.stale-target-generation",
421 "Plan was produced for another target lifetime");
422 if (request.expectedRevision && *request.expectedRevision != request.context.targetRevision)
423 return failure(EditorStatus::Conflict, "editor.command.revision-conflict",
424 "Expected revision does not match the captured context");
425 auto it = std::find_if(commands_.begin(), commands_.end(),
426 [&](const Registration& entry) { return entry.descriptor.id == request.id; });
427 if (it == commands_.end())
428 return failure(EditorStatus::NotFound, "editor.command.not-found",
429 "Command is not registered: " + request.id.value());
430 if (plan.commandGeneration != it->generation)
431 return failure(EditorStatus::Conflict, "editor.command.stale-registration",
432 "Command registration changed after planning");
433 if (auto policy = checkExecutionPolicy(it->descriptor, request.id, request.source, request.payload, profile);
434 !policy.ok())
435 return Result<TransactionReceipt>::failure(std::move(policy).status());
436 if (request.payload != plan.plannedPayload || request.source != plan.plannedSource)
437 return failure(EditorStatus::Rejected, "editor.command.plan-input-mismatch",
438 "Plan payload or source differs from the validated request");
439 if (!it->planExecutor)
440 return failure(EditorStatus::Unsupported, "editor.command.execution-unsupported",
441 "Command does not provide a plan executor");
442 try {
443 return it->planExecutor(request, plan);
444 } catch (const std::exception& exception) {
445 return failure(EditorStatus::Failed, "editor.command.executor-exception", exception.what());
446 } catch (...) {
447 return failure(EditorStatus::Failed, "editor.command.executor-exception",
448 "Command executor failed with an unknown exception");
449 }

543void EditorDock::layout(float screenW, float screenH) {
544 if (screenW <= 0.f || screenH <= 0.f) {
545 throw Exception("EditorDock::layout: screen size must be > 0");
546 }
547 float left = sizeOf("left");
548 float right = sizeOf("right");
549 float top = sizeOf("top");
550 float bottom = sizeOf("bottom");
551 // Clamp so center remains non-negative
552 if (left + right > screenW) {
553 float s = screenW / (left + right);
554 left *= s;
555 right *= s;
556 }
557 if (top + bottom > screenH) {
558 float s = screenH / (top + bottom);
559 top *= s;
560 bottom *= s;
561 }
562
563 rects_["top"] = {0.f, 0.f, screenW, top};
564 rects_["bottom"] = {0.f, screenH - bottom, screenW, bottom};
565 rects_["left"] = {0.f, top, left, screenH - top - bottom};
566 rects_["right"] = {screenW - right, top, right, screenH - top - bottom};
567 rects_["center"] = {left, top, screenW - left - right, screenH - top - bottom};
568}
