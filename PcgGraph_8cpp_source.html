33 for (const auto& node : graph.nodes) {

60GraphConnectionDecision PcgPointGraphDomain::canConnect(const GraphPinRecord& from,
61 const GraphPinRecord& to) const {
62 GraphConnectionDecision decision;
63 decision.allowed = from.direction == GraphPinDirection::Output &&
64 to.direction == GraphPinDirection::Input && from.type == "point" &&
65 to.type == "point" && from.node != to.node;
66 if (!decision.allowed)
67 decision.diagnostics.push_back(pcgDiagnostic(
68 "editor.pcg.invalid-connection", DiagnosticSeverity::Error,
69 "PCG graph connections require point output-to-input pins on distinct nodes"));
70 return decision;
71}

73EditorResult<GraphNodeRecord> PcgPointGraphDomain::makeNode(const GraphNodeId& id,
74 const std::string& operation) const {
75 if (id.empty() || procgen::PointGraph::getOperationInputCount(operation) < 0)
76 return pcgGraphError<GraphNodeRecord>("editor.pcg.unknown-operation",
77 "Unknown PointGraph operation");
78 GraphNodeRecord node;
79 node.id = id;
80 node.type = operation;
81 EditorValue::Object properties;
82 const int parameterCount = procgen::PointGraph::getOperationParamCount(operation);
83 for (int i = 0; i < parameterCount; ++i) {
84 const std::string key = procgen::PointGraph::getOperationParamKey(operation, i);
85 properties[key] = defaultValue(procgen::PointGraph::getOperationParamKind(operation, i),
86 procgen::PointGraph::getOperationParamDefault(operation, i));
87 }
88 if (operation == "subgraph") {
89 properties["definition"] = EditorValue("");
90 properties["inputNode"] = EditorValue("input");
91 properties["outputNode"] = EditorValue("output");
92 }
93 node.properties = EditorValue(std::move(properties));
94 const int inputs = procgen::PointGraph::getOperationInputCount(operation);
95 for (int i = 0; i < inputs; ++i)
96 node.pins.push_back({GraphPinId(id.value() + ".in" + std::to_string(i)), id, "point",
97 GraphPinDirection::Input});
98 node.pins.push_back(
99 {GraphPinId(id.value() + ".out"), id, "point", GraphPinDirection::Output});
100 return eve::editing::applied<GraphNodeRecord>(std::move(node));
101}

103PcgGraphMigrationResult PcgPointGraphDomain::migrate(const GraphDocumentData& graph) const {
104 PcgGraphMigrationResult result;
105 result.fromVersion = graph.schemaVersion;
106 result.graph = graph;
107 if (graph.domain != domain()) {
108 result.status = EditorStatus::Rejected;
109 result.diagnostics.push_back(pcgDiagnostic("editor.pcg.wrong-domain",
110 DiagnosticSeverity::Error,
111 "Graph is not a procgen.point document"));
112 return result;
113 }
114 if (graph.schemaVersion == 1) {
115 result.status = EditorStatus::NoOp;
116 return result;
117 }
118 if (graph.schemaVersion != 0) {
119 result.status = EditorStatus::Unsupported;
120 result.diagnostics.push_back(pcgDiagnostic(
121 "editor.pcg.unsupported-schema", DiagnosticSeverity::Error,
122 "Unsupported procgen.point graph schemaVersion: " +
123 std::to_string(graph.schemaVersion)));
124 return result;
125 }
126
127 std::unordered_map<std::string, GraphPinId> pinRemap;
128 std::vector<GraphNodeRecord> migratedNodes;
129 migratedNodes.reserve(graph.nodes.size());
130 for (const auto& legacy : graph.nodes) {
131 auto canonical = makeNode(legacy.id, legacy.type);
132 if (!canonical.ok()) {
133 result.status = EditorStatus::Rejected;
134 result.diagnostics.push_back(pcgDiagnostic(
135 "editor.pcg.migration-unknown-operation", DiagnosticSeverity::Error,
136 "Cannot migrate unknown PCG operation: " + legacy.type));
137 return result;
138 }
139 auto* defaults = canonical.value().properties.getIf<EditorValue::Object>();
140 const auto* properties = legacy.properties.getIf<EditorValue::Object>();
141 if (legacy.properties.type() != EditorValue::Type::Null && !properties) {
142 result.status = EditorStatus::Rejected;
143 result.diagnostics.push_back(pcgDiagnostic(
144 "editor.pcg.migration-invalid-properties", DiagnosticSeverity::Error,
145 "Legacy PCG node properties must be an object: " + legacy.id.value()));
146 return result;
147 }
148 if (properties)
149 for (const auto& [key, value] : *properties) (*defaults)[key] = value;
150
151 int input = 0;
152 int output = 0;
153 for (const auto& pin : legacy.pins) {
154 const int ordinal = pin.direction == GraphPinDirection::Input ? input++ : output++;
155 int canonicalOrdinal = 0;
156 const auto replacement = std::find_if(
157 canonical.value().pins.begin(), canonical.value().pins.end(),
158 [&](const GraphPinRecord& candidate) {
159 if (candidate.direction != pin.direction) return false;
160 return canonicalOrdinal++ == ordinal;
161 });
162 if (replacement == canonical.value().pins.end()) continue;
163 if (!pinRemap.emplace(pin.id.value(), replacement->id).second) {
164 result.status = EditorStatus::Rejected;
165 result.diagnostics.push_back(pcgDiagnostic(
166 "editor.pcg.migration-duplicate-pin", DiagnosticSeverity::Error,
167 "Legacy PCG graph contains duplicate pin id: " + pin.id.value()));
168 return result;
169 }
170 }
171 migratedNodes.push_back(std::move(canonical.value()));
172 }
173 result.graph.nodes = std::move(migratedNodes);
174 for (auto& edge : result.graph.edges) {
175 const auto from = pinRemap.find(edge.from.value());
176 const auto to = pinRemap.find(edge.to.value());
177 if (from != pinRemap.end()) edge.from = from->second;
178 if (to != pinRemap.end()) edge.to = to->second;
179 }
180 result.graph.schemaVersion = 1;
181 result.status = EditorStatus::Applied;
182 result.diagnostics.push_back(pcgDiagnostic("editor.pcg.migrated-v0-v1",
183 DiagnosticSeverity::Info,
184 "Migrated procgen.point graph schemaVersion 0 to 1"));
185 return result;
186}

188PcgGraphCompileResult PcgPointGraphDomain::compile(const GraphDocumentData& graph) const {
189 PcgGraphCompileResult result;
190 result.documentRevision = graph.revision;
191 const auto migration = migrate(graph);
192 if (migration.status != EditorStatus::Applied && migration.status != EditorStatus::NoOp) {
193 result.status = migration.status;
194 result.diagnostics = migration.diagnostics;
195 return result;
196 }
197 result.diagnostics = migration.diagnostics;
198 const auto& source = migration.graph;
199 procgen::PointGraph compiled;
200 for (const auto& node : source.nodes) {
201 if (!compiled.addNode(node.id.value(), node.type)) {
202 result.status = EditorStatus::Failed;
203 result.diagnostics.push_back(pcgDiagnostic("editor.pcg.invalid-node",
204 DiagnosticSeverity::Error,
205 "Unknown or duplicated PCG node: " +
206 node.id.value()));
207 return result;
208 }
209 const auto* properties = node.properties.getIf<EditorValue::Object>();
210 if (!properties) {
211 result.status = EditorStatus::Failed;
212 result.diagnostics.push_back(pcgDiagnostic(
213 "editor.pcg.invalid-properties", DiagnosticSeverity::Error,
214 "PCG node properties must be an object: " + node.id.value()));
215 return result;
216 }
217 for (const auto& [key, value] : *properties) {
218 const bool subgraphProperty =
219 node.type == "subgraph" &&
220 (key == "definition" || key == "inputNode" || key == "outputNode");
221 if (subgraphProperty) continue;
222 int parameterIndex = -1;
223 const int parameterCount = procgen::PointGraph::getOperationParamCount(node.type);
224 for (int index = 0; index < parameterCount; ++index)
225 if (procgen::PointGraph::getOperationParamKey(node.type, index) == key) {
226 parameterIndex = index;
227 break;
228 }
229 const std::string kind = parameterIndex >= 0
230 ? procgen::PointGraph::getOperationParamKind(
231 node.type, parameterIndex)
232 : std::string();
233 bool applied = false;
234 if (kind == "bool") {
235 if (const auto* boolean = value.getIf<bool>())
236 applied = compiled.setNodeInt(node.id.value(), key, *boolean ? 1 : 0);
237 } else if (kind == "int") {
238 if (const auto* integer = value.getIf<int64_t>();
239 integer && *integer >= std::numeric_limits<int>::min() &&
240 *integer <= std::numeric_limits<int>::max())
241 applied = compiled.setNodeInt(node.id.value(), key, int(*integer));
242 } else if (kind == "float") {
243 if (const auto* number = value.getIf<double>();
244 number && std::isfinite(*number) &&
245 std::abs(*number) <= std::numeric_limits<float>::max())
246 applied = compiled.setNodeFloat(node.id.value(), key, float(*number));
247 } else if (kind == "string") {
248 if (const auto* text = value.getIf<std::string>())
249 applied = compiled.setNodeString(node.id.value(), key, *text);
250 }
251 if (!applied) {
252 result.status = EditorStatus::Failed;
253 result.diagnostics.push_back(pcgDiagnostic(
254 parameterIndex < 0 ? "editor.pcg.unknown-property"
255 : "editor.pcg.property-type-mismatch",
256 DiagnosticSeverity::Error,
257 "Invalid PCG property " + node.id.value() + "." + key));
258 return result;
259 }
260 }
261 if (node.type == "subgraph") {
262 const auto definition = properties->find("definition");
263 const auto inputNode = properties->find("inputNode");
264 const auto outputNode = properties->find("outputNode");
265 const auto* definitionText =
266 definition == properties->end() ? nullptr : definition->second.getIf<std::string>();
267 const auto* inputText =
268 inputNode == properties->end() ? nullptr : inputNode->second.getIf<std::string>();
269 const auto* outputText =
270 outputNode == properties->end() ? nullptr : outputNode->second.getIf<std::string>();
271 procgen::PointGraph nested;
272 if (!definitionText || !inputText || !outputText || definitionText->empty() ||
273 !nested.deserializeDefinition(*definitionText) ||
274 !compiled.setNodeSubgraph(node.id.value(), &nested, *inputText, *outputText)) {
275 result.status = EditorStatus::Failed;
276 result.diagnostics.push_back(pcgDiagnostic(
277 "editor.pcg.invalid-subgraph", DiagnosticSeverity::Error,
278 "PCG subgraph node requires a valid definition and input/output node ids"));
279 return result;
280 }
281 }
282 }
283 if (source.parameters.type() != EditorValue::Type::Null) {
284 const auto* parameters = source.parameters.getIf<EditorValue::Object>();
285 if (!parameters) {
286 result.status = EditorStatus::Failed;
287 result.diagnostics.push_back(pcgDiagnostic("editor.pcg.invalid-parameters",
288 DiagnosticSeverity::Error,
289 "PCG graph parameters must be an object"));
290 return result;
291 }
292 for (const auto& [name, value] : *parameters) {
293 const auto* binding = value.getIf<EditorValue::Object>();
294 const auto node = binding ? binding->find("node") : EditorValue::Object::const_iterator{};
295 const auto key = binding ? binding->find("key") : EditorValue::Object::const_iterator{};
296 const auto* nodeId = binding && node != binding->end()
297 ? node->second.getIf<std::string>()
298 : nullptr;
299 const auto* parameterKey = binding && key != binding->end()
300 ? key->second.getIf<std::string>()
301 : nullptr;
302 if (!nodeId || !parameterKey ||
303 !compiled.exposeParameter(name, *nodeId, *parameterKey)) {
304 result.status = EditorStatus::Failed;
305 result.diagnostics.push_back(pcgDiagnostic(
306 "editor.pcg.invalid-parameter-binding", DiagnosticSeverity::Error,
307 "Invalid PCG graph parameter binding: " + name));
308 return result;
309 }
310 }
311 }
312 std::unordered_map<std::string, StableId> connectedInputs;
313 for (const auto& edge : source.edges) {
314 const GraphPinRecord* from = findPin(source, edge.from);
315 const GraphPinRecord* to = findPin(source, edge.to);
316 const GraphNodeRecord* toNode = to ? findNode(source, to->node) : nullptr;
317 const int slot = toNode && to ? inputIndex(*toNode, to->id) : -1;
318 const std::string inputKey = to ? to->id.value() : std::string();
319 if (!inputKey.empty() && connectedInputs.find(inputKey) != connectedInputs.end()) {
320 result.status = EditorStatus::Failed;
321 result.diagnostics.push_back(pcgDiagnostic(
322 "editor.pcg.duplicate-input", DiagnosticSeverity::Error,
323 "PCG input pin has more than one incoming edge: " + inputKey));
324 return result;
325 }
326 if (!from || !to || !toNode || !canConnect(*from, *to).allowed || slot < 0 ||
327 !compiled.connect(from->node.value(), to->node.value(), slot)) {
328 result.status = EditorStatus::Failed;
329 result.diagnostics.push_back(pcgDiagnostic("editor.pcg.invalid-edge",
330 DiagnosticSeverity::Error,
331 "Invalid PCG graph edge: " +
332 edge.id.value()));
333 return result;
334 }
335 connectedInputs.emplace(inputKey, edge.id);
336 }
337 result.definition = compiled.serializeDefinition();
338 procgen::PointGraph structuralCheck;
339 if (!structuralCheck.deserializeDefinition(result.definition)) {
340 result.status = EditorStatus::Failed;
341 result.diagnostics.push_back(pcgDiagnostic("editor.pcg.cycle", DiagnosticSeverity::Error,
342 structuralCheck.getError()));
343 result.definition.clear();
344 return result;
345 }
346 result.status = EditorStatus::Applied;
347 return result;
348}

350PcgGraphPreviewResult PcgPointGraphDomain::preview(const GraphDocumentData& graph,
351 const std::string& inputNode,
352 procgen::PointSet* previewInput,
353 const std::string& outputNode,
354 int nodeBudget, int pointBudget) const {
355 PcgGraphPreviewResult result;
356 result.documentRevision = graph.revision;
357 const auto compiled = compile(graph);
358 result.diagnostics = compiled.diagnostics;
359 if (compiled.status != EditorStatus::Applied) {
360 result.status = compiled.status;
361 return result;
362 }
363 if (!previewInput || inputNode.empty() || outputNode.empty()) {
364 result.status = EditorStatus::Rejected;
365 result.diagnostics.push_back(pcgDiagnostic(
366 "editor.pcg.preview-invalid-input", DiagnosticSeverity::Error,
367 "PCG preview requires input/output node ids and point input"));
368 return result;
369 }
370 procgen::PointGraph runtime;
371 if (!runtime.deserializeDefinition(compiled.definition)) {
372 result.status = EditorStatus::Failed;
373 result.diagnostics.push_back(pcgDiagnostic("editor.pcg.preview-bind-failed",
374 DiagnosticSeverity::Error,
375 "PCG preview definition could not be loaded"));
376 return result;
377 }
378 runtime.setExecutionNodeBudget(nodeBudget);
379 runtime.setMaxNodeOutputPoints(pointBudget);
380 if (!runtime.setNodePoints(inputNode, previewInput)) {
381 result.status = EditorStatus::Failed;
382 result.diagnostics.push_back(pcgDiagnostic(
383 "editor.pcg.preview-bind-failed", DiagnosticSeverity::Error,
384 runtime.getError().empty() ? "PCG preview input node is invalid: " + inputNode
385 : runtime.getError()));
386 return result;
387 }
388 std::unique_ptr<procgen::PointSet> output(runtime.execute(outputNode));
389 if (!output) {
390 result.status = runtime.wasCancelled() ? EditorStatus::Cancelled : EditorStatus::Failed;
391 result.diagnostics.push_back(pcgDiagnostic(
392 runtime.wasCancelled() ? "editor.pcg.preview-cancelled"
393 : "editor.pcg.preview-execution-failed",
394 runtime.wasCancelled() ? DiagnosticSeverity::Info : DiagnosticSeverity::Error,
395 runtime.getError()));
396 return result;
397 }
398 result.outputCount = output->getCount();
399 result.metrics.reserve(size_t(runtime.getMetricCount()));
400 for (int index = 0; index < runtime.getMetricCount(); ++index) {
401 PcgGraphPreviewMetric metric;
402 metric.nodeId = runtime.getMetricNodeId(index);
403 metric.outputCount = runtime.getMetricOutputCount(index);
404 metric.milliseconds = runtime.getMetricMilliseconds(index);
405 metric.cacheHit = runtime.isMetricCacheHit(index);
406 metric.minX = runtime.getMetricMinX(index);
407 metric.minY = runtime.getMetricMinY(index);
408 metric.minZ = runtime.getMetricMinZ(index);
409 metric.maxX = runtime.getMetricMaxX(index);
410 metric.maxY = runtime.getMetricMaxY(index);
411 metric.maxZ = runtime.getMetricMaxZ(index);
412 metric.averageDensity = runtime.getMetricAverageDensity(index);
413 result.metrics.push_back(std::move(metric));
414 }
415 result.status = EditorStatus::Applied;
416 return result;
417}
