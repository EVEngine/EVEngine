21void SphereVolumeBrushKernel::sample(float x, float y, float z, float radius,
22 std::vector<VolumeBrushPoint>& out) const {
23 out.clear();
24 radius = std::max(0.5f, radius);
25 const int minX = static_cast<int>(std::floor(x - radius));
26 const int maxX = static_cast<int>(std::ceil(x + radius));
27 const int minY = static_cast<int>(std::floor(y - radius));
28 const int maxY = static_cast<int>(std::ceil(y + radius));
29 const int minZ = static_cast<int>(std::floor(z - radius));
30 const int maxZ = static_cast<int>(std::ceil(z + radius));
31 for (int iz = minZ; iz <= maxZ; ++iz)
32 for (int iy = minY; iy <= maxY; ++iy)
33 for (int ix = minX; ix <= maxX; ++ix) {
34 const float dx = static_cast<float>(ix) - x;
35 const float dy = static_cast<float>(iy) - y;
36 const float dz = static_cast<float>(iz) - z;
37 const float distance = std::sqrt(dx * dx + dy * dy + dz * dz);
38 if (distance <= radius) out.push_back({ix, iy, iz, falloffWeight(falloff_, distance / radius)});
39 }
40}
