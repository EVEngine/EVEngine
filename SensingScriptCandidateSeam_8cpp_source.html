76bool parseName(const std::array<NameEntry<T>, N>& table, std::string_view text, T* out) noexcept {
77 for (const NameEntry<T>& entry : table) {
78 if (entry.name != text) continue;
79 if (out != nullptr) *out = entry.value;

81 }
82 return false;

109bool readInteger(const Value& value, std::int64_t& out) {
110 if (const auto* integer = value.getIf<std::int64_t>()) {
111 out = *integer;
112 return true;
113 }
114 if (const auto* number = value.getIf<double>()) {
115 if (!std::isfinite(*number) || std::floor(*number) != *number) return false;
116 out = static_cast<std::int64_t>(*number);
117 return true;
118 }

123bool readStrings(const Value* value, std::vector<std::string>& out) {
124 if (isNull(value)) return true;
125 const auto* entries = value->getIf<Value::Array>();
126 if (entries == nullptr) return false;
127 out.reserve(entries->size());
128 for (const Value& entry : *entries) {
129 const auto* text = entry.getIf<std::string>();
130 if (text == nullptr) return false;
131 out.push_back(*text);
132 }
133 return true;

173Value areaValue(const TargetingSpec& spec) {
174 if (spec.worldArea) {
175 const WorldArea& area = *spec.worldArea;
176 const std::string shape = std::string(nameOf(kWorldAreaShapes, area.shape()));
177 Value::Object out{{"shape", Value::string(shape)}};
178 switch (area.shape()) {
179 case WorldArea::Shape::Circle2D:
180 case WorldArea::Shape::Sphere3D:
181 out.emplace("center", worldPointValue(area.first()));
182 out.emplace("radius", Value::number(double(area.radius())));
183 break;
184 case WorldArea::Shape::Box2D:
185 case WorldArea::Shape::Box3D:
186 out.emplace("min", worldPointValue(area.first()));
187 out.emplace("max", worldPointValue(area.second()));
188 break;
189 case WorldArea::Shape::Cone2D:
190 // 方向存在 second()、范围存在 radius()、半角是这一片才补上的出口。
191 out.emplace("apex", worldPointValue(area.first()));
192 out.emplace("dirX", Value::number(double(area.second().x())));
193 out.emplace("dirY", Value::number(double(area.second().y())));
194 out.emplace("halfAngle", Value::number(double(area.halfAngle())));
195 out.emplace("range", Value::number(double(area.radius())));
196 break;
197 }
198 return Value::object(std::move(out));
199 }
200 if (spec.gridArea) {
201 const GridArea& gridArea = *spec.gridArea;
202 Value::Object out{{"shape", Value::string(std::string(nameOf(kGridAreaShapes, gridArea.shape())))},
203 {"min", gridPointValue(gridArea.minimum())},
204 {"max", gridPointValue(gridArea.maximum())}};
205 return Value::object(std::move(out));
206 }
207 return {};
208}

216Value::Object queryValue(const TargetingQuery& query) {
217 const TargetingSpec& spec = query.spec;
218 Value::Object out;
219 out.emplace("origin", Value::string(query.origin.format()));
220 out.emplace("space", Value::string(std::string(nameOf(kCoordinateSpaces, spec.space))));
221 out.emplace("originLocation", locationValue(query.originLocation));
222 out.emplace("domain", Value::string(std::string(nameOf(kTargetDomains, spec.domain))));
223 out.emplace("minRange", Value::number(double(spec.minRange)));
224 // +inf 过不去（编组拒绝非有限数）：无界用 null 表示，这也是脚本侧唯一能表达"无界"的值。
225 out.emplace("maxRange", std::isfinite(spec.maxRange) ? Value::number(double(spec.maxRange)) : Value{});
226 out.emplace("requiredTags", stringsValue(spec.requiredTags));
227 out.emplace("excludedTags", stringsValue(spec.excludedTags));
228 out.emplace("zone", spec.zone ? Value::string(spec.zone->logicalId().format()) : Value{});
229 out.emplace("area", areaValue(spec));
230 return out;
231}

242Result<TargetLocation> locationFromScript(const Value& value, CoordinateSpace querySpace) {
243 const Value* spaceField = field(value, "space");
244 const auto* spaceText = spaceField != nullptr ? spaceField->getIf<std::string>() : nullptr;
245 if (spaceText == nullptr)
246 return scriptFailure<TargetLocation>(DiagnosticCode::InvalidArgument,
247 "candidate location must state its 'space'", "candidate.location.space");
248 CoordinateSpace space = CoordinateSpace::World2D;
249 if (!parseName(kCoordinateSpaces, *spaceText, &space))
250 return scriptFailure<TargetLocation>(DiagnosticCode::InvalidArgument,
251 "candidate location has an unknown space: " + *spaceText,
252 "candidate.location.space");
253 if (space != querySpace)
254 return scriptFailure<TargetLocation>(
255 DiagnosticCode::Unsupported,
256 "candidate location space '" + *spaceText + "' does not match the query space '" +
257 std::string(nameOf(kCoordinateSpaces, querySpace)) + "'; convert explicitly instead of mixing spaces",
258 "candidate.location.space");
259
260 const Value* xField = field(value, "x");
261 const Value* yField = field(value, "y");
262 const Value* zField = field(value, "z");
263 const bool isGrid = space == CoordinateSpace::Grid2D || space == CoordinateSpace::Grid3D;
264 if (isGrid) {
265 std::int64_t x = 0;
266 std::int64_t y = 0;
267 std::int64_t z = 0;
268 if (xField == nullptr || yField == nullptr || !readInteger(*xField, x) || !readInteger(*yField, y) ||
269 (zField != nullptr && !isNull(zField) && !readInteger(*zField, z)))
270 return scriptFailure<TargetLocation>(DiagnosticCode::InvalidArgument,
271 "grid candidate location requires integral x/y", "candidate.location");
272 if (space == CoordinateSpace::Grid2D)
273 return Result<TargetLocation>::success(TargetLocation(GridPoint::grid2D(int32_t(x), int32_t(y))));
274 return Result<TargetLocation>::success(TargetLocation(GridPoint::grid3D(int32_t(x), int32_t(y), int32_t(z))));
275 }
276
277 double x = 0.0;
278 double y = 0.0;
279 double z = 0.0;
280 if (xField == nullptr || yField == nullptr || !readNumber(*xField, x) || !readNumber(*yField, y) ||
281 (zField != nullptr && !isNull(zField) && !readNumber(*zField, z)))
282 return scriptFailure<TargetLocation>(DiagnosticCode::InvalidArgument,
283 "world candidate location requires finite x/y", "candidate.location");
284 if (space == CoordinateSpace::World2D) {
285 auto point = WorldPoint::world2D(float(x), float(y));
286 if (!point.ok()) return Result<TargetLocation>::failure(point.status());
287 return Result<TargetLocation>::success(TargetLocation(std::move(point).takeValue()));
288 }
289 auto point = WorldPoint::world3D(float(x), float(y), float(z));
290 if (!point.ok()) return Result<TargetLocation>::failure(point.status());
291 return Result<TargetLocation>::success(TargetLocation(std::move(point).takeValue()));
292}

294Result<TargetCandidate> candidateFromScript(const Value& entry, CoordinateSpace querySpace) {
295 if (entry.type() != Value::Type::Object)
296 return scriptFailure<TargetCandidate>(DiagnosticCode::InvalidArgument, "every candidate must be a table",
297 "candidate");
298
299 const Value* subjectField = field(entry, "subject");
300 const auto* subjectText = subjectField != nullptr ? subjectField->getIf<std::string>() : nullptr;
301 if (subjectText == nullptr)
302 return scriptFailure<TargetCandidate>(DiagnosticCode::InvalidArgument, "candidate requires a 'subject' string",
303 "candidate.subject");
304 const auto subjectId = PersistentId::parse(*subjectText);
305 if (!subjectId)
306 return scriptFailure<TargetCandidate>(DiagnosticCode::InvalidArgument,
307 "candidate subject is not a canonical UUID: " + *subjectText,
308 "candidate.subject");
309
310 const Value* locationField = field(entry, "location");
311 if (locationField == nullptr)
312 return scriptFailure<TargetCandidate>(DiagnosticCode::InvalidArgument, "candidate requires a 'location' table",
313 "candidate.location");
314 auto location = locationFromScript(*locationField, querySpace);
315 if (!location.ok()) return Result<TargetCandidate>::failure(location.status());
316
317 TargetCandidate candidate;
318 candidate.subject = SubjectRef::fromPersistentId(*subjectId);
319 candidate.location = std::move(location).takeValue();
320
321 if (const Value* domainField = field(entry, "domain"); !isNull(domainField)) {
322 const auto* domainText = domainField->getIf<std::string>();
323 TargetDomain domain = TargetDomain::Neutral;
324 if (domainText == nullptr || !parseName(kTargetDomains, *domainText, &domain))
325 return scriptFailure<TargetCandidate>(DiagnosticCode::InvalidArgument, "candidate has an unknown domain",
326 "candidate.domain");
327 candidate.domain = domain;
328 }
329 if (!readStrings(field(entry, "tags"), candidate.tags))
330 return scriptFailure<TargetCandidate>(DiagnosticCode::InvalidArgument,
331 "candidate 'tags' must be an array of strings", "candidate.tags");
332
333 std::vector<std::string> zoneIds;
334 if (!readStrings(field(entry, "zones"), zoneIds))
335 return scriptFailure<TargetCandidate>(DiagnosticCode::InvalidArgument,
336 "candidate 'zones' must be an array of strings", "candidate.zones");
337 candidate.zones.reserve(zoneIds.size());
338 for (const std::string& zoneId : zoneIds) {
339 const auto parsed = LogicalId::parse(zoneId);
340 if (!parsed)
341 return scriptFailure<TargetCandidate>(DiagnosticCode::InvalidArgument,
342 "candidate zone is not a valid logical id: " + zoneId,
343 "candidate.zones");
344 const auto zone = ZoneRef::fromLogicalId(*parsed);
345 if (!zone)
346 return scriptFailure<TargetCandidate>(
347 DiagnosticCode::InvalidArgument, "candidate zone is not a valid ZoneRef: " + zoneId, "candidate.zones");
348 candidate.zones.push_back(*zone);
349 }
350 return Result<TargetCandidate>::success(std::move(candidate));
351}

358class ScriptSensingCandidateProvider final : public ISensingCandidateProvider {
359public:
360 explicit ScriptSensingCandidateProvider(ScriptProviderHandle handle) : handle_(std::move(handle)) {}
361
363 [[nodiscard]] const ScriptProviderHandle& handle() const noexcept { return handle_; }
364
365 Result<std::vector<TargetCandidate>> query(const TargetingQuery& query) const override {
366 const ScriptArg args[] = {ScriptArg::fromTable(queryValue(query))};
367 auto answer = ScriptProviderRegistry::instance().callValue(handle_, "queryCandidates", args);
368 if (!answer.ok()) return Result<std::vector<TargetCandidate>>::failure(answer.status());
369
370 const auto* entries = answer.value().getIf<Value::Array>();
371 if (entries == nullptr)
372 return scriptFailure<std::vector<TargetCandidate>>(
373 DiagnosticCode::InvalidArgument, "queryCandidates must return an array of candidate tables",
374 "sensing.script-candidates");
375
376 std::vector<TargetCandidate> candidates;
377 candidates.reserve(entries->size());
378 for (const Value& entry : *entries) {
379 auto candidate = candidateFromScript(entry, query.spec.space);
380 if (!candidate.ok()) return Result<std::vector<TargetCandidate>>::failure(candidate.status());
381 candidates.push_back(std::move(candidate).takeValue());
382 }
383 return Result<std::vector<TargetCandidate>>::success(std::move(candidates));
384 }

386private:
387 ScriptProviderHandle handle_;
388};

403EVENGINE_API_PLATFORM void declareScriptSensingCandidateSeam() {
404 ScriptSensingCandidateProviders& owned = scriptSensingCandidateProviders();
405 ScriptProviderRegistry::instance().declareCapability(
406 ISensingCandidateProvider::capabilityName,
407 [&owned](const ScriptProviderHandle& handle) -> Result<void> {
408 // 单实现槽位：`provide` 会顶掉当前实现，顶掉必须可见（否则原生 provider 会"悄悄失效"，
409 // 而症状只是候选变少）。
410 if (cap::query<ISensingCandidateProvider>() != nullptr) {
411 recordLogEvent("warn", "script sensing candidate provider '" + handle.name +
412 "' replaced the provider that owned ISensingCandidateProvider");
413 }
414 auto adapter = std::make_unique<ScriptSensingCandidateProvider>(handle);
415 ScriptSensingCandidateProvider* raw = adapter.get();
416 // 先注册再交所有权：从这一刻起能力槽位持有裸指针，`owned.live` 是让它有效的唯一原因。
417 cap::provide<ISensingCandidateProvider>(raw);
418 owned.live.push_back(std::move(adapter));
419 return Result<void>::success();
420 },
421 [&owned](const ScriptProviderHandle& handle) -> Result<void> {
422 const auto match = std::find_if(owned.live.begin(), owned.live.end(),
423 [&handle](const std::unique_ptr<ScriptSensingCandidateProvider>& adapter) {
424 return adapter->handle() == handle;
425 });
426 if (match == owned.live.end()) {
427 // 幂等：后置条件是"这一代没有适配器"，模块退出时可能已经撤过了。
428 return Result<void>::success();
429 }
430 cap::revoke<ISensingCandidateProvider>(match->get());
431 owned.retired.push_back(std::move(*match));
432 owned.live.erase(match);
433 return Result<void>::success();
434 });
435}

437EVENGINE_API_PLATFORM void withdrawScriptSensingCandidateSeam() {
438 ScriptSensingCandidateProviders& owned = scriptSensingCandidateProviders();
439 for (auto& adapter : owned.live) {
440 cap::revoke<ISensingCandidateProvider>(adapter.get());
441 owned.retired.push_back(std::move(adapter));
442 }
443 owned.live.clear();
444
445 // 连声明一起撤：模块不在了还留着接缝，只会让脚本发布出一个永远不会被调用的适配器。
446 ScriptProviderRegistry::instance()
447 .removeCapability(ISensingCandidateProvider::capabilityName)
448 .ignore("a module may be destroyed without ever having declared the seam");
449}
