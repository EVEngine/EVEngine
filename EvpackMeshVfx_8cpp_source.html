25Result<std::unique_ptr<EvpackMeshVfx>> EvpackMeshVfx::load(const asset::EvpackResourceReader& reader,
26 const AssetRef& asset,
27 const asset::EvpackCapabilities& capabilities,
28 graphics::Graphics& graphics) {
29 auto payload = reader.read(asset, "eve.stylize.mesh-vfx/1", capabilities, 1024 * 1024);
30 if (!payload) return Result<std::unique_ptr<EvpackMeshVfx>>::failure(payload.status());
31 if (payload.value().chunks.size() != 1 || payload.value().chunks[0].kind != asset::EvpackChunkKind::Definition)
32 return failure<std::unique_ptr<EvpackMeshVfx>>(DiagnosticCode::InvalidArgument,
33 "Effect requires one definition");
34 auto value = asset::decodeRuntimeDefinition(payload.value().chunks[0].bytes);
35 if (!value) return Result<std::unique_ptr<EvpackMeshVfx>>::failure(value.status());
36 auto json = value.value().toJson();
37 if (!json) return Result<std::unique_ptr<EvpackMeshVfx>>::failure(json.status());
38 auto definition = stylize::MeshVfxAsset::fromJson(json.value());
39 if (!definition) return Result<std::unique_ptr<EvpackMeshVfx>>::failure(definition.status());
40 if (definition.value().layers.size() > 32 || definition.value().trail || definition.value().trailBinding ||
41 !definition.value().animationTriggers.play.empty() || !definition.value().animationTriggers.stop.empty() ||
42 !definition.value().animationTriggers.trailBreak.empty())
43 return failure<std::unique_ptr<EvpackMeshVfx>>(
44 DiagnosticCode::Unsupported,
45 "Package renderer supports up to 32 mesh layers; trail/animation attachments require a gameplay adapter");
46 std::map<std::string, std::map<std::string, float>> defaults;
47 std::vector<AssetRef> references;
48 for (const auto& layer : definition.value().layers) {
49 const float cycle = layer.playback.fadeIn + layer.playback.duration + layer.playback.fadeOut;
50 if (!std::isfinite(cycle) || (layer.playback.loop && cycle <= 0))
51 return failure<std::unique_ptr<EvpackMeshVfx>>(DiagnosticCode::InvalidArgument, "Invalid playback cycle");
52 auto reference = AssetRef::parse(layer.style);
53 if (!reference) return Result<std::unique_ptr<EvpackMeshVfx>>::failure(reference.status());
54 auto shader = asset::loadShaderAsset(reader, reference.value(), capabilities);
55 if (!shader) return Result<std::unique_ptr<EvpackMeshVfx>>::failure(shader.status());
56 if (shader.value().interface != asset::ShaderAssetInterface::Mesh3D)
57 return failure<std::unique_ptr<EvpackMeshVfx>>(DiagnosticCode::InvalidArgument,
58 "Mesh effect requires mesh shader");
59 auto layout = asset_graphics::validateShaderAssetGpu(shader.value());
60 if (!layout) return Result<std::unique_ptr<EvpackMeshVfx>>::failure(layout.status());
61 auto& parameters = defaults[layer.style];
62 for (const auto& parameter : shader.value().parameters) {
63 if (parameter.defaults.size() != 1)
64 return failure<std::unique_ptr<EvpackMeshVfx>>(DiagnosticCode::Unsupported,
65 "Mesh VFX curves require scalar shader parameters");
66 parameters[parameter.name] = parameter.defaults.front();
67 }
68 references.push_back(std::move(reference).takeValue());
69 }
70 auto playback = stylize::MeshVfxAssetInstance::create(definition.value(), defaults);
71 if (!playback) return Result<std::unique_ptr<EvpackMeshVfx>>::failure(playback.status());
72 auto impl = std::make_unique<Impl>(Impl{graphics, std::move(playback).takeValue(), {}});
73 for (const auto& reference : references) {
74 auto shader = asset_graphics::EvpackShader::load(reader, reference, capabilities, graphics);
75 if (!shader) return Result<std::unique_ptr<EvpackMeshVfx>>::failure(shader.status());
76 impl->shaders.push_back(std::move(shader).takeValue());
77 }
78 auto result = std::unique_ptr<EvpackMeshVfx>(new EvpackMeshVfx(std::move(impl)));
79 result->play();
80 return Result<std::unique_ptr<EvpackMeshVfx>>::success(std::move(result));
81}

106Result<void> EvpackMeshVfx::draw(graphics::Mesh& mesh, const std::array<float, 16>& transform,
107 OptionalRef<graphics::Texture> texture, const std::array<float, 4>& tint) {
108 for (std::size_t i = 0; i < impl_->shaders.size(); ++i) {
109 auto& layer = impl_->playback->layer(i);
110 if (layer.intensity() <= 0) continue;
111 auto& shader = *impl_->shaders[i];
112 for (const auto& parameter : shader.definition().parameters) {
113 const float value = layer.style().getFloat(parameter.name);
114 auto applied = shader.setParameter(parameter.name, std::span<const float>(&value, 1));
115 if (!applied) return applied;
116 }
117 auto color = tint;
118 color[3] *= layer.intensity();
119 auto drawn = shader.drawMesh(mesh, transform, texture, color);
120 if (!drawn) return drawn;
121 }
122 return Result<void>::success();
123}
