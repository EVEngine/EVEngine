14 const auto owner = integerField(table, "owner"), index = integerField(table, "index"),
15 generation = integerField(table, "generation");
16 if (owner <= 0 || index < 0 || index >= AgentHandle::invalidIndex || generation <= 0 ||

53AgentUpdate parseUpdate(Crowd& crowd, const ssq::Table& table) {
54 AgentUpdate update;
55 update.reference = parseReference(table.find("reference").to<ssq::Table>());
56 const auto flow = optionalField<ssq::Table>(table, "flowRequest");
57 const auto defaultFlow = optionalField<bool>(table, "useDefaultFlow");
58 if (flow && defaultFlow) throw ssq::TypeException("flowRequest and useDefaultFlow cannot be combined");
59 if (flow) {
60 auto resolved = resolveFlowRequestLease(crowd, *flow);
61 if (!resolved) throw ssq::TypeException("Invalid, stale or foreign request-owned lease");
62 update.flow = std::move(resolved).takeValue();
63 }
64 if (defaultFlow && *defaultFlow) update.flow = FlowLease{};
65
66 if (auto item = optionalField<ssq::Table>(table, "movement")) update.movement = parseMovementBinding(*item);
67 update.maxSpeed = optionalField<float>(table, "maxSpeed");
68 update.maxAcceleration = optionalField<float>(table, "maxAcceleration");
69 update.headingTurnRate = optionalField<float>(table, "headingTurnRate");
70 update.radius = optionalField<float>(table, "radius");
71 update.avoidancePriority = optionalField<int>(table, "avoidancePriority");
72 update.data = optionalField<int>(table, "data");
73 update.holdPosition = optionalField<bool>(table, "holdPosition");
74 if (auto item = optionalField<ssq::Table>(table, "interaction")) {
75 update.interaction = {item->get<float>("pushability"), item->get<bool>("holdPosition"), item->get<int>("layer"),
76 item->get<int>("mask")};
77 }
78 if (auto item = optionalField<ssq::Table>(table, "policy")) update.policy = parseInteractionPolicy(*item);
79 if (auto item = optionalField<ssq::Table>(table, "intent")) {
80 const auto action = item->get<std::string>("action");
81 AgentAction mode;
82 if (action == "idle")
83 mode = AgentAction::Idle;
84 else if (action == "flow")
85 mode = AgentAction::Flow;
86 else if (action == "seek")
87 mode = AgentAction::Seek;
88 else if (action == "boids")
89 mode = AgentAction::Boids;
90 else
91 throw ssq::TypeException("Unknown steering action");
92 update.intent = {mode, item->get<bool>("hasTarget"), item->get<float>("targetX"), item->get<float>("targetY")};
93 }
94 return update;
95}

116 return {MovementClassId(static_cast<std::uint64_t>(identity)), static_cast<std::uint64_t>(revision)};

119InteractionPolicy parseInteractionPolicy(ssq::Table table) {
