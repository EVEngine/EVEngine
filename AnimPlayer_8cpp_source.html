86void AnimPlayer::update(float dt) {
87 if (dt < 0.f) throw Exception("AnimPlayer.update: dt must be >= 0");
88 if (!playing_ || paused_ || !clip_) return;
89
90 time_ += dt * speed_;
91 if (!effectiveLoop() && clip_->getDuration() > 0.f && time_ >= clip_->getDuration()) {
92 time_ = clip_->getDuration();
93 playing_ = false;
94 }
95
96 AnimPose sampled;
97 clip_->sample(time_, &sampled, skeleton_);
98
99 if (blending_ && prevClip_) {
100 blendElapsed_ += dt;
101 float t = blendDuration_ > 1e-8f ? blendElapsed_ / blendDuration_ : 1.f;
102 if (t >= 1.f) {
103 blending_ = false;
104 prevClip_ = nullptr;
105 pose_.copyFrom(&sampled);
106 } else {
107 // Advance previous clip time during fade for continuity.
108 prevTime_ += dt * speed_;
109 prevClip_->sample(prevTime_, &prevPose_, skeleton_);
110 pose_.blendFrom(&prevPose_, &sampled, t);
111 }
112 } else {
113 pose_.copyFrom(&sampled);
114 }
115}
