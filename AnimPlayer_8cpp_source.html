128void AnimPlayer::dispatchEvents(float oldTime, float newTime) {
129 if (!clip_ || clip_->getEventCount() == 0 || newTime <= oldTime) return;
130 const float duration = clip_->getDuration();
131 if (effectiveLoop() && duration > 1e-8f) {
132 const int firstCycle = static_cast<int>(std::floor(oldTime / duration));
133 const int lastCycle = static_cast<int>(std::floor(newTime / duration));
134 for (int cycle = firstCycle; cycle <= lastCycle; ++cycle) {
135 for (int i = 0; i < clip_->getEventCount(); ++i) {
136 const float absoluteTime = static_cast<float>(cycle) * duration + clip_->getEventTime(i);
137 if (absoluteTime > oldTime && absoluteTime <= newTime)
138 events_.push_back({clip_->getEventName(i), clip_->getEventPayload(i)});
139 }
140 }
141 return;
142 }
143 const float endTime = duration > 0.f ? std::min(newTime, duration) : newTime;
144 for (int i = 0; i < clip_->getEventCount(); ++i) {
145 const float eventTime = clip_->getEventTime(i);
146 if (eventTime > oldTime && eventTime <= endTime)
147 events_.push_back({clip_->getEventName(i), clip_->getEventPayload(i)});
148 }
149}

151void AnimPlayer::updateUnchecked(float dt) {
152 if (dt < 0.f) throw Exception("AnimPlayer.update: dt must be >= 0");
153 events_.clear();
154 if (!playing_ || paused_ || !clip_) return;
155 if (updateRate_ > 0.f) {
156 updateAccumulator_ += dt;
157 const float interval = 1.f / updateRate_;
158 if (updateAccumulator_ + 1e-8f < interval) {
159 rootMotion_ = TransformTRS::identity();
160 return;
161 }
162 dt = updateAccumulator_;
163 updateAccumulator_ = 0.f;
164 }
165
166 rootMotion_ = TransformTRS::identity();
167 const float previousTime = time_;
168 clip_->sample(previousTime, &rootPreviousPose_, skeleton_);
169 time_ += dt * speed_;
170 clip_->collectEvents(previousTime, time_, effectiveLoop(), pendingEvents_);
171 dispatchEvents(previousTime, time_);
172 if (!effectiveLoop() && clip_->getDuration() > 0.f && time_ >= clip_->getDuration()) {
173 time_ = clip_->getDuration();
174 playing_ = false;
175 }
176
177 clip_->sample(time_, &sampledPose_, skeleton_);
178 const TransformTRS& fromRoot = rootPreviousPose_.local(rootMotionBone_);
179 const TransformTRS& toRoot = sampledPose_.local(rootMotionBone_);
180 rootMotion_.px = toRoot.px - fromRoot.px;
181 rootMotion_.py = toRoot.py - fromRoot.py;
182 rootMotion_.pz = toRoot.pz - fromRoot.pz;
183 if (effectiveLoop() && clip_->getDuration() > 1e-8f) {
184 const float duration = clip_->getDuration();
185 const int cycles = static_cast<int>(std::floor(time_ / duration) - std::floor(previousTime / duration));
186 if (cycles != 0) {
187 clip_->sample(0.f, &rootStartPose_, skeleton_);
188 clip_->sampleClamped(duration, &rootEndPose_, skeleton_);
189 const TransformTRS& startRoot = rootStartPose_.local(rootMotionBone_);
190 const TransformTRS& endRoot = rootEndPose_.local(rootMotionBone_);
191 rootMotion_.px += static_cast<float>(cycles) * (endRoot.px - startRoot.px);
192 rootMotion_.py += static_cast<float>(cycles) * (endRoot.py - startRoot.py);
193 rootMotion_.pz += static_cast<float>(cycles) * (endRoot.pz - startRoot.pz);
194 }
195 }
196 rootMotion_.qx =
197 fromRoot.qw * toRoot.qx - fromRoot.qx * toRoot.qw - fromRoot.qy * toRoot.qz + fromRoot.qz * toRoot.qy;
198 rootMotion_.qy =
199 fromRoot.qw * toRoot.qy + fromRoot.qx * toRoot.qz - fromRoot.qy * toRoot.qw - fromRoot.qz * toRoot.qx;
200 rootMotion_.qz =
201 fromRoot.qw * toRoot.qz - fromRoot.qx * toRoot.qy + fromRoot.qy * toRoot.qx - fromRoot.qz * toRoot.qw;
202 rootMotion_.qw =
203 fromRoot.qw * toRoot.qw + fromRoot.qx * toRoot.qx + fromRoot.qy * toRoot.qy + fromRoot.qz * toRoot.qz;
204 rootMotion_.normalizeRotation();
205
206 if (blending_ && prevClip_) {
207 blendElapsed_ += dt;
208 float t = blendDuration_ > 1e-8f ? blendElapsed_ / blendDuration_ : 1.f;
209 if (t >= 1.f) {
210 blending_ = false;
211 prevClip_ = nullptr;
212 pose_.copyFrom(&sampledPose_);
213 } else {
214 // Advance previous clip time during fade for continuity.
215 prevTime_ += dt * speed_;
216 prevClip_->sample(prevTime_, &prevPose_, skeleton_);
217 pose_.blendFrom(&prevPose_, &sampledPose_, t);
218 }
219 } else {
220 pose_.copyFrom(&sampledPose_);
221 }
222}

224eve::Result<void> AnimPlayer::advance(const eve::SimulationStep& step) {
225 if (step.delta.nanoseconds() < 0)
226 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
227 "animation player delta must be non-negative"));
228 if (hasLastTick_ && step.tick <= lastTick_)
229 return eve::Result<void>::failure(
230 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "animation player tick must advance monotonically"));
231
232 const float dt = static_cast<float>(step.delta.seconds());
233 if (!std::isfinite(dt))
234 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
235 "animation player delta is outside float range"));
236 updateUnchecked(dt);
237 lastTick_ = step.tick;
238 hasLastTick_ = true;
239 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
240}

242void AnimPlayer::update(float dt) {
243 auto duration = eve::Duration::fromSeconds(dt);
244 if (!duration) {
245 duration.ignore("legacy animation player update received an invalid duration");
246 return;
247 }
248 auto nextTick = hasLastTick_ ? lastTick_.incremented() : std::optional<eve::SimulationTick>(eve::SimulationTick(1));
249 if (!nextTick) {
250 eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
251 "animation player simulation tick overflow"))
252 .ignore("legacy animation player update tick overflow");
253 return;
254 }
255 advance({*nextTick, std::move(duration).takeValue()}).ignore("legacy animation player update facade");
256}
