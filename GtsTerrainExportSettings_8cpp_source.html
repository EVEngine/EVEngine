548Result<std::string> encodeGtsMaskedTerrainObj(const Heightmap& heightmap, const Heightmap& maskmap,
549 GtsTerrainSaveResolution resolution, float sizeX, float sizeY,
550 float sizeZ, GtsTerrainObjFaceMode faceMode, float threshold,
551 bool invert) {
552 if (!enumRange(int(faceMode), 1) || !std::isfinite(threshold) || threshold < 0.f || threshold > 1.f)
553 return invalid<std::string>("GTS masked terrain OBJ settings are invalid");
554 MeshBuild validated;
555 auto built = buildGtsTerrainBaseMesh(validated, heightmap, resolution, sizeX, sizeY, sizeZ);
556 if (!built) return Result<std::string>::failure(built.status());
557 if (maskmap.getWidth() < 1 || maskmap.getHeight() < 1)
558 return invalid<std::string>("GTS masked terrain OBJ mask is empty");
559 const int terrainResolution = heightmap.getWidth(), stride = 1 << int(resolution);
560 std::vector<int> vertexIndices(std::size_t(terrainResolution) * std::size_t(terrainResolution), -1);
561 struct Vertex {
562 float x, y, z, u, v;
563 };
564 std::vector<Vertex> vertices;
565 std::vector<int> indices;
566 auto sampleNormalized = [](const Heightmap& map, float x, float z) {
567 return map.sampleBilinear(x * float(map.getWidth()), z * float(map.getHeight()));
568 };
569 auto vertexIndex = [&](int x, int z, float height) {
570 auto& index = vertexIndices[std::size_t(z) * std::size_t(terrainResolution) + std::size_t(x)];
571 if (index < 0) {
572 index = int(vertices.size());
573 vertices.push_back({-float(x) * sizeX / float(terrainResolution - 1), height * sizeY,
574 float(z) * sizeZ / float(terrainResolution - 1), float(x) / float(terrainResolution),
575 float(z) / float(terrainResolution)});
576 }
577 return index;
578 };
579 auto addTriangle = [&](int a, int b, int c) {
580 indices.push_back(c);
581 indices.push_back(b);
582 indices.push_back(a);
583 };
584 auto addQuad = [&](int a, int b, int c, int d) {
585 indices.push_back(a);
586 indices.push_back(c);
587 indices.push_back(d);
588 indices.push_back(b);
589 };
590 for (int z = 0; z < terrainResolution - stride; z += stride) {
591 const float z0 = float(z) / float(terrainResolution), z1 = float(z + stride) / float(terrainResolution);
592 for (int x = 0; x < terrainResolution - stride; x += stride) {
593 const float x0 = float(x) / float(terrainResolution), x1 = float(x + stride) / float(terrainResolution);
594 const bool outside = sampleNormalized(maskmap, x0, z0) < threshold;
595 if (outside != invert) continue;
596 const int v1 = vertexIndex(x, z + stride, sampleNormalized(heightmap, z1, x0));
597 const int v2 = vertexIndex(x + stride, z + stride, sampleNormalized(heightmap, z1, x1));
598 const int v3 = vertexIndex(x, z, sampleNormalized(heightmap, z0, x0));
599 const int v4 = vertexIndex(x + stride, z, sampleNormalized(heightmap, z0, x1));
600 if (faceMode == GtsTerrainObjFaceMode::Triangles) {
601 addTriangle(v3, v1, v2);
602 addTriangle(v3, v2, v4);
603 } else
604 addQuad(v1, v2, v3, v4);
605 }
606 }
607 std::ostringstream stream;
608 stream.imbue(std::locale::classic());
609 stream << std::setprecision(std::numeric_limits<float>::max_digits10);
610 stream << "# Unity terrain OBJ File\n";
611 for (const auto& vertex : vertices) stream << "v " << vertex.x << ' ' << vertex.y << ' ' << vertex.z << '\n';
612 for (const auto& vertex : vertices) stream << "vt " << vertex.u << ' ' << vertex.v << '\n';
613 const int faceSize = faceMode == GtsTerrainObjFaceMode::Triangles ? 3 : 4;
614 for (std::size_t i = 0; i < indices.size(); i += std::size_t(faceSize)) {
615 stream << "f";
616 for (int j = 0; j < faceSize; ++j) {
617 const int index = indices[i + std::size_t(j)] + 1;
618 stream << ' ' << index << '/' << index;
619 }
620 stream << '\n';
621 }
622 auto text = stream.str();
623 if (text.size() > 512U * 1024U * 1024U) return invalid<std::string>("GTS masked terrain OBJ exceeds size limit");
624 return Result<std::string>::success(std::move(text));
625}
