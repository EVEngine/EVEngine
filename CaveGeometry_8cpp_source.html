40float ellipsoidDistance(Vec3 point, const Chamber& shape) {
41 const Vec3 q = sub(point, shape.center);
42 const float normalizedX = q.x / shape.radii.x;
43 const float normalizedY = q.y / shape.radii.y;
44 const float normalizedZ = q.z / shape.radii.z;
45 const float base =
46 (std::sqrt(normalizedX * normalizedX + normalizedY * normalizedY + normalizedZ * normalizedZ) - 1.f) *
47 std::min({shape.radii.x, shape.radii.y, shape.radii.z});
48 if (shape.irregularity <= 0.f) return base;
49 const float azimuth = std::atan2(normalizedZ, normalizedX);
50 const float lobes = std::sin(azimuth * 3.f + shape.phase) * 0.58f +
51 std::sin(azimuth * 2.f - normalizedY * 2.4f + shape.phase * 0.7f) * 0.42f;
52 return base - lobes * shape.irregularity * std::min({shape.radii.x, shape.radii.y, shape.radii.z}) * 0.13f;
53}

55float taperedSegmentDistance(Vec3 point, const Dripstone& shape, float horizontalX, float horizontalZ) {
56 const Vec3 axis = sub(shape.end, shape.start);
57 const float length2 = dot(axis, axis);
58 const float t = length2 > 1e-8f ? std::clamp(dot(sub(point, shape.start), axis) / length2, 0.f, 1.f) : 0.f;
59 Vec3 delta = sub(point, add(shape.start, mul(axis, t)));
60 delta.x *= horizontalX;
61 delta.z *= horizontalZ;
62 float radius = shape.startRadius + (shape.endRadius - shape.startRadius) * t;
63 const float envelope = std::sin(t * 3.1415926535f);
64 radius *= 1.f + shape.profileAmplitude * envelope *
65 std::sin(t * shape.profileFrequency * 6.283185307f + shape.profilePhase);
66 return std::sqrt(dot(delta, delta)) - radius;
67}

69float flowstoneDistance(Vec3 point, const Flowstone& shape) {
70 const Vec3 q = sub(point, shape.center);
71 const float u = dot(q, shape.tangent);
72 const float v = q.y;
73 const float ripple = std::sin((v / shape.halfHeight) * shape.rippleFrequency + shape.phase) * shape.rippleAmplitude;
74 const float n = dot(q, shape.normal) - ripple;
75 const float ellipsoid =
76 std::sqrt((u * u) / (shape.halfWidth * shape.halfWidth) + (v * v) / (shape.halfHeight * shape.halfHeight) +
77 (n * n) / (shape.thickness * shape.thickness)) -
78 1.f;
79 return ellipsoid * std::min({shape.halfWidth, shape.halfHeight, shape.thickness});
80}

82float curtainDistance(Vec3 point, const Curtain& shape) {
83 const Vec3 q = sub(point, shape.anchor);
84 const float u = dot(q, shape.tangent);
85 const float downward = -q.y;
86 const float wave = std::sin((u / shape.halfWidth) * shape.waveFrequency + shape.phase) * shape.waveAmplitude;
87 const float n = dot(q, shape.normal) - wave;
88 const float side = std::fabs(u) - shape.halfWidth;
89 const float vertical = std::max(-downward, downward - shape.length);
90 const float sheet = std::fabs(n) - shape.thickness;
91 return std::max({side, vertical, sheet});
92}

102float latticeNoise(float x, float y, float z, uint32_t seed) {
103 const int ix = int(std::floor(x));
104 const int iy = int(std::floor(y));
105 const int iz = int(std::floor(z));
106 const float fx = x - float(ix);
107 const float fy = y - float(iy);
108 const float fz = z - float(iz);
109 auto smooth = [](float t) { return t * t * (3.f - 2.f * t); };
110 auto sample = [seed](int sx, int sy, int sz) {
111 uint32_t h = hash(uint32_t(sx) * 73856093u ^ uint32_t(sy) * 19349663u ^ uint32_t(sz) * 83492791u ^ seed);
112 return float(h & 0xffffu) / 32767.5f - 1.f;
113 };
114 const float ux = smooth(fx), uy = smooth(fy), uz = smooth(fz);
115 auto lerp = [](float a, float b, float t) { return a + (b - a) * t; };
116 const float x00 = lerp(sample(ix, iy, iz), sample(ix + 1, iy, iz), ux);
117 const float x10 = lerp(sample(ix, iy + 1, iz), sample(ix + 1, iy + 1, iz), ux);
118 const float x01 = lerp(sample(ix, iy, iz + 1), sample(ix + 1, iy, iz + 1), ux);
119 const float x11 = lerp(sample(ix, iy + 1, iz + 1), sample(ix + 1, iy + 1, iz + 1), ux);
120 return lerp(lerp(x00, x10, uy), lerp(x01, x11, uy), uz);
121}

130 const Vec3 ab = sub(b, a), ac = sub(c, a);
131 Vec3 normal{ab.y * ac.z - ab.z * ac.y, ab.z * ac.x - ab.x * ac.z, ab.x * ac.y - ab.y * ac.x};
132 if (dot(normal, referenceNormal) < 0.f) {
133 std::swap(b, c);

137 if (length > 1e-8f) normal = mul(normal, 1.f / length);
138 const uint32_t base = uint32_t(mesh.getVertexCount());

140 mesh.addVertex(b.x, b.y, b.z, normal.x, normal.y, normal.z, b.x + 0.5f, b.y + 0.5f);
141 mesh.addVertex(c.x, c.y, c.z, normal.x, normal.y, normal.z, c.x + 0.5f, c.y + 0.5f);
142 mesh.addTriangle(base, base + 1, base + 2);
