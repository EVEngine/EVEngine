27eve::Result<eve::StateMutation> decodeMutation(const eve::Value& value, std::size_t index) {
28 const std::string path = "stateMutations." + std::to_string(index);
29 if (!value.isObject()) return payloadFailure<eve::StateMutation>("state mutation must be an object", path);
30 for (const auto& key : value.keys())
31 if (key != "subject" && key != "key" && key != "kind" && key != "value" && key != "persistent")
32 return payloadFailure<eve::StateMutation>("unknown state mutation field '" + key + "'", path + "." + key);
33
34 const eve::Value* subject = value.find("subject");
35 const eve::Value* key = value.find("key");
36 const eve::Value* kind = value.find("kind");
37 const eve::Value* rawValue = value.find("value");
38 const eve::Value* persistent = value.find("persistent");
39 if (!subject || !subject->isString() || subject->asString().empty())
40 return payloadFailure<eve::StateMutation>("state mutation subject must be a non-empty string",
41 path + ".subject");
42 if (!key || !key->isString() || key->asString().empty())
43 return payloadFailure<eve::StateMutation>("state mutation key must be a non-empty string", path + ".key");
44 if (!kind || !kind->isString())
45 return payloadFailure<eve::StateMutation>("state mutation kind must be a string", path + ".kind");
46
47 eve::StateMutation mutation;
48 mutation.subject = subject->asString();
49 mutation.key = key->asString();
50 if (kind->asString() == "set")
51 mutation.kind = eve::MutationKind::Set;
52 else if (kind->asString() == "remove")
53 mutation.kind = eve::MutationKind::Remove;
54 else if (kind->asString() == "addTag")
55 mutation.kind = eve::MutationKind::AddTag;
56 else if (kind->asString() == "removeTag")
57 mutation.kind = eve::MutationKind::RemoveTag;
58 else if (kind->asString() == "addNumber")
59 mutation.kind = eve::MutationKind::AddNumber;
60 else
61 return payloadFailure<eve::StateMutation>("unsupported state mutation kind '" + kind->asString() + "'",
62 path + ".kind");
63
64 const bool needsValue = mutation.kind == eve::MutationKind::Set || mutation.kind == eve::MutationKind::AddNumber;
65 if (needsValue && !rawValue)
66 return payloadFailure<eve::StateMutation>("state mutation requires value", path + ".value");
67 if (!needsValue && rawValue)
68 return payloadFailure<eve::StateMutation>("state mutation kind does not accept value", path + ".value");
69 if (rawValue) mutation.value = *rawValue;
70 if (persistent) {
71 if (!persistent->isBool())
72 return payloadFailure<eve::StateMutation>("state mutation persistent must be boolean",
73 path + ".persistent");
74 mutation.persistent = persistent->asBool();
75 }
76 return eve::Result<eve::StateMutation>::success(std::move(mutation));
77}

79eve::Result<void> validateChoice(const eve::dnut::SequenceNode& node) {

89eve::Result<void> validateCommand(const eve::dnut::SequenceNode& node) {
90 const eve::Value* name = node.payload.find("name");
91 if (!name) name = node.payload.find("target");
92 if (!name || !name->isString() || name->asString().empty())
93 return payloadFailure("dialogue command requires a non-empty name", "payload.name");
94 const eve::Value* kind = node.payload.find("kind");
95 if (kind && (!kind->isString() || (kind->asString() != "operation" && kind->asString() != "gameplay")))
96 return payloadFailure("dialogue command kind must be operation or gameplay", "payload.kind");
97 auto payment = decodeSequencePayment(node.payload);
98 if (!payment.ok()) return eve::Result<void>::failure(payment.status());
99 auto mutations = decodeSequenceStateMutations(node.payload);
100 if (!mutations.ok()) return eve::Result<void>::failure(mutations.status());
101 return eve::Result<void>::success();
102}

123eve::Result<std::vector<eve::StateMutation>> decodeSequenceStateMutations(const eve::Value& payload) {
124 if (!payload.isObject())
125 return payloadFailure<std::vector<eve::StateMutation>>("dialogue payload must be an object", "payload");
126 const eve::Value* values = payload.find("stateMutations");
127 if (!values) return eve::Result<std::vector<eve::StateMutation>>::success({});
128 if (!values->isArray())
129 return payloadFailure<std::vector<eve::StateMutation>>("stateMutations must be an array",
130 "payload.stateMutations");
131 std::vector<eve::StateMutation> mutations;
132 mutations.reserve(values->arraySize());
133 for (std::size_t index = 0; index < values->arraySize(); ++index) {
134 auto mutation = decodeMutation(values->at(index), index);
135 if (!mutation.ok()) return eve::Result<std::vector<eve::StateMutation>>::failure(mutation.status());
136 mutations.push_back(std::move(mutation).takeValue());
137 }
138 return eve::Result<std::vector<eve::StateMutation>>::success(std::move(mutations));
139}

141eve::Value encodeSequenceStateMutations(std::span<const eve::StateMutation> mutations) {
142 eve::Value::Array values;
143 values.reserve(mutations.size());
144 for (const auto& mutation : mutations) {
145 eve::Value value = eve::Value::Object{};
146 value.set("subject", eve::Value::string(mutation.subject));
147 value.set("key", eve::Value::string(mutation.key));
148 value.set("kind", eve::Value::string(mutationKindName(mutation.kind)));
149 if (mutation.kind == eve::MutationKind::Set || mutation.kind == eve::MutationKind::AddNumber)
150 value.set("value", mutation.value);
151 if (mutation.persistent) value.set("persistent", eve::Value::boolean(true));
152 values.push_back(std::move(value));
153 }
154 return eve::Value(std::move(values));
155}

157eve::Result<void> registerDialogueSequenceSteps(eve::dnut::StepKindRegistry& registry) {
158 eve::dnut::StepKindRegistry candidate = registry;
159 using eve::dnut::StepField;
160 using eve::dnut::StepFieldType;
161 using eve::dnut::StepKindDescriptor;
162 using eve::dnut::StepShape;
163
164 auto line = candidate.registerStep(StepKindDescriptor{
165 "line",
166 "Line",
167 "dialogue",
168 StepShape::Await,
169 {StepField{"speaker", StepFieldType::String, false}, StepField{"text", StepFieldType::String, false},
170 StepField{"pool", StepFieldType::String, false}, StepField{"i18n", StepFieldType::String, false},
171 StepField{"voice", StepFieldType::String, false}, StepField{"expression", StepFieldType::String, false}}});
172 if (!line.ok()) return line;
173 auto choice = candidate.registerStep(
174 StepKindDescriptor{"choice", "Choice", "dialogue", StepShape::Await, {}, validateChoice});
175 if (!choice.ok()) return choice;
176 auto command = candidate.registerStep(StepKindDescriptor{
177 "command",
178 "Command",
179 "dialogue",
180 StepShape::Await,
181 {StepField{"name", StepFieldType::String, false}, StepField{"target", StepFieldType::String, false},
182 StepField{"kind", StepFieldType::String, false}, StepField{"arguments", StepFieldType::Any, false},
183 StepField{"payment", StepFieldType::Any, false}, StepField{"stateMutations", StepFieldType::Any, false},
184 StepField{"resultLocal", StepFieldType::String, false}, StepField{"result", StepFieldType::String, false}},
185 validateCommand});
186 if (!command.ok()) return command;
187 registry = std::move(candidate);
188 return eve::Result<void>::success();

191eve::Result<void> validateDialogueSequenceAsset(const eve::dnut::SequenceAsset& asset,
192 const eve::dnut::StepKindRegistry& registry) {
193 auto graph = asset.validate();
194 if (!graph.ok()) return graph;
195 for (const auto& node : asset.nodes) {
196 if (registry.contains(node.type)) {
197 auto validated = registry.validate(node);
198 if (!validated.ok()) return validated;
199 }
200 if (node.type != "command" && (node.payload.find("payment") || node.payload.find("stateMutations")))
201 return payloadFailure("payment and stateMutations are valid only on command nodes",
202 "nodes." + node.id + ".payload");
203 for (const auto& route : node.routes)
204 if (node.type != "choice" && (route.payload.find("payment") || route.payload.find("stateMutations")))
205 return payloadFailure("payment and stateMutations are valid only on choice routes",
206 "nodes." + node.id + ".routes");
207 }
208 return eve::Result<void>::success();
209}

211eve::Result<CommandRequest> decodeDialogueCommandRequest(const eve::dnut::SequenceCommandRequest& request) {
212 CommandRequest decoded;
213 decoded.requestId = request.requestId;
214 decoded.name = request.name;
215 decoded.arguments = request.arguments;
216 decoded.bindings = request.bindings;
217 decoded.locals = request.locals;
218 const eve::Value* kind = request.payload.find("kind");
219 decoded.kind = kind && kind->isString() && kind->asString() == "gameplay" ? CommandRequestKind::GameplayAction
220 : CommandRequestKind::Operation;
221 auto payment = decodeSequencePayment(request.payload);
222 if (!payment.ok()) return eve::Result<CommandRequest>::failure(payment.status());
223 decoded.payment = std::move(payment).takeValue();
224 auto mutations = decodeSequenceStateMutations(request.payload);
225 if (!mutations.ok()) return eve::Result<CommandRequest>::failure(mutations.status());
226 decoded.stateMutations = std::move(mutations).takeValue();
227 return eve::Result<CommandRequest>::success(std::move(decoded));
228}
