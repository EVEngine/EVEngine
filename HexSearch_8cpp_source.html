63std::int32_t HexSearchContext::beginPhase() noexcept {
64 phase_ += 2;
65 // Clearing the frontier is only this: the bucket array holds no owned state,
66 // and the per-cell records are invalidated lazily by the phase check, so a
67 // new search never pays for resetting the grid.
68 buckets_.clear();
69 minimum_ = 0;
70 return phase_;
71}

73void HexSearchContext::enqueue(std::int32_t cellIndex) noexcept {
74 if (data_.empty() || cellIndex < 0 || cellIndex >= cellCount()) return;
75 std::int32_t priority = data_[cellIndex].priority();
76 if (priority < 0) priority = 0;
77 if (priority < minimum_) minimum_ = priority;
78 if (static_cast<std::size_t>(priority) >= buckets_.size())
79 buckets_.resize(static_cast<std::size_t>(priority) + 1, -1);
80 // Link the cell at the head of its bucket: equal-priority cells form an
81 // intrusive singly linked list through `nextWithSamePriority`.
82 data_[cellIndex].nextWithSamePriority = buckets_[static_cast<std::size_t>(priority)];
83 buckets_[static_cast<std::size_t>(priority)] = cellIndex;
84}
85
86HexSearchPop HexSearchContext::dequeue(std::int32_t& outCellIndex) noexcept {
87 const std::int32_t bucketCount = static_cast<std::int32_t>(buckets_.size());
88 for (; minimum_ < bucketCount; ++minimum_) {
89 const std::int32_t head = buckets_[static_cast<std::size_t>(minimum_)];
90 if (head < 0) continue;
91 buckets_[static_cast<std::size_t>(minimum_)] = data(head).nextWithSamePriority;
92 outCellIndex = head;
93 return HexSearchPop::Cell;
94 }
95 return HexSearchPop::Empty;
96}

98void HexSearchContext::unlink(std::int32_t cellIndex, std::int32_t bucket) noexcept {
99 if (bucket < 0 || static_cast<std::size_t>(bucket) >= buckets_.size()) return;
100 std::int32_t current = buckets_[static_cast<std::size_t>(bucket)];
101 if (current < 0) return;
102 if (current == cellIndex) {
103 buckets_[static_cast<std::size_t>(bucket)] = data(current).nextWithSamePriority;
104 return;
105 }
106 while (true) {
107 const std::int32_t next = data(current).nextWithSamePriority;
108 if (next < 0) return;
109 if (next == cellIndex) {
110 data(current).nextWithSamePriority = data(next).nextWithSamePriority;
111 return;
112 }
113 current = next;
114 }
115}

124bool isValidDestination(const HexMap& map, HexCoordinates coordinates, const HexOccupancyQuery& occupied) {
125 if (!map.contains(coordinates)) return false;
126 // `canHoldUnit` is the shared "explored, explorable, above water" terrain rule;
127 // occupancy by another actor is the caller's extra clause.
128 const HexCellData* cell = map.cell(coordinates);
129 if (cell == nullptr || !canHoldUnit(*cell)) return false;
130 if (occupied && occupied(coordinates)) return false;
131 return true;
132}

134std::int32_t moveCost(const HexMap& map, HexCoordinates from, HexCoordinates to, HexDirection direction,
135 const HexOccupancyQuery& occupied) {
136 if (!isValidDestination(map, to, occupied)) return -1;
137
138 const HexEdgeType type = edgeType(map.elevation(from), map.elevation(to));
139 if (type == HexEdgeType::Cliff) return -1;
140
141 if (map.flags(from).hasRoad(direction)) return 1;
142 if (map.isWalled(from) != map.isWalled(to)) return -1;
143
144 const std::int32_t base = type == HexEdgeType::Flat ? 5 : 10;
145 return base + map.urbanLevel(to) + map.farmLevel(to) + map.plantLevel(to);
146}

150Result<HexPath> findPath(const HexMap& map, HexSearchContext& scratch, HexCoordinates from, HexCoordinates to,
151 const HexMoveRules& rules, const HexOccupancyQuery& occupied) {
152 if (map.empty()) return pathInvalidArgument("cannot search an empty hex map");
153 if (!map.contains(from)) return pathInvalidArgument("path origin is outside the hex map");
154 if (!map.contains(to)) return pathInvalidArgument("path goal is outside the hex map");
155 if (scratch.cellCount() != map.cellCount())
156 return pathInvalidArgument("search scratch must be resized to map.cellCount() before findPath");
157 if (!isValidDestination(map, to, occupied)) return unreachable("path goal is not a valid destination");
158
159 const std::int32_t speed = std::max<std::int32_t>(1, rules.speed);
160 const std::int32_t start = map.indexOf(from);
161 const std::int32_t goal = map.indexOf(to);
162 const HexCoordinates goalCoord = to;
163 const std::int32_t phase = scratch.beginPhase();
164
165 scratch.data(start).searchPhase = phase;
166 scratch.enqueue(start);
167
168 bool found = false;
169 while (true) {
170 std::int32_t current = -1;
171 if (scratch.dequeue(current) != HexSearchPop::Cell) break;
172
173 const std::int32_t currentDistance = scratch.data(current).distance;
174 // Marking the record with `phase + 1` is how the reference says "expanded":
175 // an untouched record still carries the older phase, so the distinction
176 // needs no clearing pass.
177 scratch.data(current).searchPhase = phase + 1;
178
179 if (current == goal) {
180 found = true;
181 break;
182 }
183
184 const std::int32_t currentTurn = turnOf(currentDistance, speed);
185 const HexCoordinates currentCoord = map.coordinatesAt(current);
186
187 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
188 const HexDirection direction = static_cast<HexDirection>(i);
189 HexCoordinates neighbour{};
190 if (!map.getNeighbor(currentCoord, direction, neighbour)) continue;
191
192 const std::int32_t neighbourIndex = map.indexOf(neighbour);
193 if (neighbourIndex < 0) continue;
194 // Already expanded: its bucket entry was consumed and it must not be
195 // relaxed again within this phase.
196 if (scratch.data(neighbourIndex).searchPhase == phase + 1) continue;
197
198 const std::int32_t step = moveCost(map, currentCoord, neighbour, direction, occupied);
199 if (step < 0) continue;
200
201 std::int32_t distance = currentDistance + step;
202 const std::int32_t turn = turnOf(distance, speed);
203 if (turn > currentTurn) distance = turn * speed + step;
204
205 HexSearchData& neighbourData = scratch.data(neighbourIndex);
206 if (neighbourData.searchPhase != phase) {
207 // A fresh record is fully overwritten; it carries no bucket link yet.
208 neighbourData.searchPhase = phase;
209 neighbourData.distance = distance;
210 neighbourData.pathFrom = current;
211 neighbourData.heuristic = neighbour.distanceTo(goalCoord);
212 scratch.enqueue(neighbourIndex);
213 } else if (distance < neighbourData.distance) {
214 // The priority must be read before the record is updated: `change`
215 // unlinks the cell from the bucket it is still sitting in.
216 const std::int32_t oldPriority = neighbourData.priority();
217 neighbourData.distance = distance;
218 neighbourData.pathFrom = current;
219 scratch.change(neighbourIndex, oldPriority);
220 }
221 }
222 }
223
224 if (!found) return unreachable("no path exists between the two cells");
225
226 HexPath path;
227 for (std::int32_t index = goal; index != start; index = scratch.data(index).pathFrom) {
228 if (index < 0) return unreachable("path reconstruction lost the chain to the origin");
229 const std::int32_t distance = scratch.data(index).distance;
230 path.cells.push_back(index);
231 path.turns.push_back(turnOf(distance, speed));
232 }
233 path.cells.push_back(start);
234 path.turns.push_back(turnOf(scratch.data(start).distance, speed));
235 std::reverse(path.cells.begin(), path.cells.end());
236 std::reverse(path.turns.begin(), path.turns.end());
237 return Result<HexPath>::success(std::move(path));
238}

242Result<void> collectVisibleCells(const HexMap& map, HexSearchContext& scratch, HexCoordinates from,
243 std::int32_t range, std::vector<std::int32_t>& out) {
244 out.clear();
245 if (map.empty()) return voidInvalidArgument("cannot collect visibility from an empty hex map");
246 if (!map.contains(from)) return voidInvalidArgument("view origin is outside the hex map");
247 if (scratch.cellCount() != map.cellCount())
248 return voidInvalidArgument("search scratch must be resized to map.cellCount() before collectVisibleCells");
249
250 const std::int32_t searchRange = range + map.values(from).viewElevation();
251 const std::int32_t start = map.indexOf(from);
252 const std::int32_t phase = scratch.beginPhase();
253
254 scratch.data(start).searchPhase = phase;
255 scratch.data(start).distance = 0;
256 scratch.enqueue(start);
257
258 while (true) {
259 std::int32_t current = -1;
260 if (scratch.dequeue(current) != HexSearchPop::Cell) break;
261
262 // Nearest first: the frontier always pops the cheapest step count, so the
263 // produced order is stable for a given map.
264 scratch.data(current).searchPhase = phase + 1;
265 out.push_back(current);
266
267 const std::int32_t currentDistance = scratch.data(current).distance;
268 const HexCoordinates currentCoord = map.coordinatesAt(current);
269
270 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
271 const HexDirection direction = static_cast<HexDirection>(i);
272 HexCoordinates neighbour{};
273 if (!map.getNeighbor(currentCoord, direction, neighbour)) continue;
274 if (!map.isExplorable(neighbour)) continue;
275
276 const std::int32_t neighbourIndex = map.indexOf(neighbour);
277 if (neighbourIndex < 0) continue;
278 if (scratch.data(neighbourIndex).searchPhase == phase + 1) continue;
279
280 const std::int32_t distance = currentDistance + 1;
281 if (distance + map.values(neighbour).viewElevation() > searchRange) continue;
282 if (distance > from.distanceTo(neighbour)) continue;
283
284 HexSearchData& neighbourData = scratch.data(neighbourIndex);
285 if (neighbourData.searchPhase != phase) {
286 neighbourData.searchPhase = phase;
287 neighbourData.distance = distance;
288 neighbourData.pathFrom = current;
289 neighbourData.heuristic = 0;
290 scratch.enqueue(neighbourIndex);
291 } else if (distance < neighbourData.distance) {
292 const std::int32_t oldPriority = neighbourData.priority();
293 neighbourData.distance = distance;
294 neighbourData.pathFrom = current;
295 scratch.change(neighbourIndex, oldPriority);
296 }
297 }
298 }
299 return Result<void>::success();
300}
