119void FluidSimulator::stepSolver(float dt, int substeps) {
120 if (sim_.particleCount() <= 0) return;
121 if (preferGpu_ && !gpuOk_) ensureGpu();
122 if (!gpuOk_) {
123 sim_.step(dt, substeps);
124 return;
125 }
126
127 const int iters = std::max(1, substeps);
128 const float sub = dt / float(iters);
129
130 seq_->begin();
131 uploadParticles();
132 const int pbf = std::max(0, sim_.params().pbfIterations);
133 for (int it = 0; it < iters; ++it) {
134 setCommonConstants(shClear_, sub);
135 seq_->recordDispatch(shClear_, groupsFor(grid_.cellCount()));
136 setCommonConstants(shBuild_, sub);
137 seq_->recordDispatch(shBuild_, groupsFor(sim_.maxParticles()));
138 setCommonConstants(shIntegrate_, sub);
139 seq_->recordDispatch(shIntegrate_, groupsFor(sim_.maxParticles()));
140 for (int k = 0; k < pbf; ++k) {
141 setCommonConstants(shClear_, sub);
142 seq_->recordDispatch(shClear_, groupsFor(grid_.cellCount()));
143 setCommonConstants(shBuild_, sub);
144 seq_->recordDispatch(shBuild_, groupsFor(sim_.maxParticles()));
145 setCommonConstants(shDensityLambda_, sub);
146 seq_->recordDispatch(shDensityLambda_, groupsFor(sim_.maxParticles()));
147 setCommonConstants(shDelta_, sub);
148 seq_->recordDispatch(shDelta_, groupsFor(sim_.maxParticles()));
149 setCommonConstants(shApply_, sub);
150 seq_->recordDispatch(shApply_, groupsFor(sim_.maxParticles()));
151 }
152 }
153 downloadParticles();
154 seq_->submit();
155
156 // Update the CPU mirror so reads reflect GPU state.
157 std::vector<float> posF(size_t(sim_.maxParticles()) * 4u);
158 std::vector<float> velF(size_t(sim_.maxParticles()) * 4u);
159 std::vector<float> densF(size_t(sim_.maxParticles()), 0.f);
160 stagePos_->downloadBytes(posF.data(), uint64_t(posF.size()) * sizeof(float));
161 stageVel_->downloadBytes(velF.data(), uint64_t(velF.size()) * sizeof(float));
162 stageDens_->downloadBytes(densF.data(), uint64_t(densF.size()) * sizeof(float));
163 std::vector<FluidParticle>& cpu = sim_.particles();
164 for (int i = 0; i < sim_.particleCount(); ++i) {
165 const size_t b = size_t(i) * 4u;
166 cpu[size_t(i)].pos = glm::vec3(posF[b], posF[b + 1], posF[b + 2]);
167 cpu[size_t(i)].vel = glm::vec3(velF[b], velF[b + 1], velF[b + 2]);
168 }
169 std::vector<float>& dens = sim_.densities();
170 for (int i = 0; i < sim_.particleCount(); ++i) dens[size_t(i)] = densF[size_t(i)];
171}

173eve::Result<void> FluidSimulator::step(const eve::SimulationStep& stepValue,
174 const eve::physics::SimulationSettings& settings) {
175 auto valid = eve::physics::detail::validateSimulationStep(stepValue, settings, observation_);
176 if (!valid) return valid;
177 auto next = eve::physics::detail::advanceSimulationObservation(observation_, stepValue);
178 if (!next) return eve::Result<void>::failure(next.status());
179
180 try {
181 if (preferGpu_ && !gpuOk_) {
182 bool available = false;
183 try {
184 available = ensureGpu();
185 } catch (...) {
186 available = false;
187 }
188 if (!available) {
189 backendFallback_ = true;
190 backendSelectionStatus_ = eve::Status(
191 eve::StatusCode::Applied,
192 {eve::Diagnostic::warning(
193 eve::DiagnosticCode::Unsupported, "Fluid GPGPU accelerator unavailable; CPU reference selected",
194 "fluids.simulationBackend", {{"selected", "cpu"}, {"fallback", "explicit"}})});
195 }
196 }
197 stepSolver(static_cast<float>(stepValue.delta.seconds()), settings.subStepCount);
198 } catch (const std::exception& error) {
199 return eve::Result<void>::failure(eve::Diagnostic::error(
200 eve::DiagnosticCode::Failed, std::string("Fluid simulation step failed: ") + error.what(),
201 "fluids.simulationBackend.step"));
202 } catch (...) {
203 return eve::Result<void>::failure(eve::Diagnostic::error(
204 eve::DiagnosticCode::Failed, "Fluid simulation step failed with an unknown exception",
205 "fluids.simulationBackend.step"));
206 }
207 observation_ = std::move(next).takeValue();
208 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
209}

248bool FluidSimulator::ensureGpu() {
249 if (gpuOk_) return true;
250 gpgpu_ = eve::gpgpu::Gpgpu::create();
251 if (!gpgpu_ || !gpgpu_->isAvailable()) return false;
252 if (sim_.sdf().voxelCount() <= 0) return false;
253
254 const int max = sim_.maxParticles();
255 try {
256 shClear_ = gpgpu_->newShader(kFluidClearGrid);
257 shBuild_ = gpgpu_->newShader(kFluidBuildGrid);
258 shDensityLambda_ = gpgpu_->newShader(kFluidDensityLambda);
259 shDelta_ = gpgpu_->newShader(kFluidComputeDelta);
260 shApply_ = gpgpu_->newShader(kFluidApplyDelta);
261 shIntegrate_ = gpgpu_->newShader(kFluidIntegrate);
262 bufPos_ = gpgpu_->newBuffer(max * 4 * int(sizeof(float)), "storage");
263 bufVel_ = gpgpu_->newBuffer(max * 4 * int(sizeof(float)), "storage");
264 bufHead_ = gpgpu_->newBuffer(grid_.cellCount() * int(sizeof(int)), "storage");
265 bufNext_ = gpgpu_->newBuffer(max * int(sizeof(int)), "storage");
266 bufDens_ = gpgpu_->newBuffer(max * int(sizeof(float)), "storage");
267 bufLambda_ = gpgpu_->newBuffer(max * int(sizeof(float)), "storage");
268 bufGrad_ = gpgpu_->newBuffer(max * 4 * int(sizeof(float)), "storage");
269 bufSdf_ = gpgpu_->newBuffer(sim_.sdf().voxelCount() * int(sizeof(float)), "storage");
270 stagePos_ = gpgpu_->newBuffer(max * 4 * int(sizeof(float)), "staging");
271 stageVel_ = gpgpu_->newBuffer(max * 4 * int(sizeof(float)), "staging");
272 stageDens_ = gpgpu_->newBuffer(max * int(sizeof(float)), "staging");
273 seq_ = gpgpu_->newSequence();
274 } catch (...) {
275 return false;
276 }
277 if (!seq_ || !seq_->isAvailable()) return false;
278
279 gpuOk_ = true; // uploadSdf() guards on gpuOk_; set it before uploading.
280 uploadSdf();
281 shClear_->bindBuffer(2, bufHead_);
282 shBuild_->bindBuffer(0, bufPos_);
283 shBuild_->bindBuffer(2, bufHead_);
284 shBuild_->bindBuffer(3, bufNext_);
285 shDensityLambda_->bindBuffer(0, bufPos_);
286 shDensityLambda_->bindBuffer(2, bufHead_);
287 shDensityLambda_->bindBuffer(3, bufNext_);
288 shDensityLambda_->bindBuffer(4, bufDens_);
289 shDensityLambda_->bindBuffer(6, bufLambda_);
290 shDensityLambda_->bindBuffer(7, bufGrad_);
291 shDelta_->bindBuffer(0, bufPos_);
292 shDelta_->bindBuffer(2, bufHead_);
293 shDelta_->bindBuffer(3, bufNext_);
294 shDelta_->bindBuffer(6, bufLambda_);
295 shDelta_->bindBuffer(7, bufGrad_);
296 shApply_->bindBuffer(0, bufPos_);
297 shApply_->bindBuffer(2, bufHead_);
298 shApply_->bindBuffer(3, bufNext_);
299 shApply_->bindBuffer(7, bufGrad_);
300 shApply_->bindBuffer(5, bufSdf_);
301 shIntegrate_->bindBuffer(0, bufPos_);
302 shIntegrate_->bindBuffer(1, bufVel_);
303 shIntegrate_->bindBuffer(2, bufHead_);
304 shIntegrate_->bindBuffer(3, bufNext_);
305 shIntegrate_->bindBuffer(5, bufSdf_);
306 return true;
307}

316void FluidSimulator::uploadParticles() {
317 if (!gpuOk_) return;
318 const int max = sim_.maxParticles();
319 std::vector<float> posF(size_t(max) * 4u, 0.f);
320 std::vector<float> velF(size_t(max) * 4u, 0.f);
321 const auto& particles = sim_.particles();
322 for (int i = 0; i < sim_.particleCount(); ++i) {
323 const size_t b = size_t(i) * 4u;
324 posF[b] = particles[size_t(i)].pos.x;
325 posF[b + 1] = particles[size_t(i)].pos.y;
326 posF[b + 2] = particles[size_t(i)].pos.z;
327 velF[b] = particles[size_t(i)].vel.x;
328 velF[b + 1] = particles[size_t(i)].vel.y;
329 velF[b + 2] = particles[size_t(i)].vel.z;
330 }
331 seq_->recordUpload(bufPos_, posF.data(), uint64_t(posF.size()) * sizeof(float));
332 seq_->recordUpload(bufVel_, velF.data(), uint64_t(velF.size()) * sizeof(float));
333}
