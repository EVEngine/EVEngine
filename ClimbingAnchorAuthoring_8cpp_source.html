41std::string hexHash(const ClimbingAnchorBakeRequest& request) {
42 std::uint64_t hash = 1469598103934665603ull;
43 const auto append = [&hash](std::string_view value) {
44 for (const unsigned char byte : value) {
45 hash ^= byte;
46 hash *= 1099511628211ull;
47 }
48 };
49 const auto appendNumber = [&append](auto value) {
50 std::array<char, 64> buffer{};
51 const int count = std::snprintf(buffer.data(), buffer.size(), "%.9g", static_cast<double>(value));
52 if (count > 0) append(std::string_view(buffer.data(), static_cast<std::size_t>(count)));
53 append("|");
54 };
55 append("evengine.climbing-anchor-bake-v1|");
56 append(request.sourceGeometryContentId);
57 append("|");
58 appendNumber(request.settings.handSpacing);
59 appendNumber(request.settings.hangingFeetDrop);
60 appendNumber(request.settings.ladderMountOffset);
61 appendNumber(request.settings.occupancySlots);
62 std::vector<const ClimbingLedgeBakeSource*> ledges;
63 ledges.reserve(request.ledges.size());
64 for (const auto& ledge : request.ledges) ledges.push_back(&ledge);
65 std::sort(ledges.begin(), ledges.end(), [](const auto* lhs, const auto* rhs) {
66 return lhs->id < rhs->id;
67 });
68 for (const auto* source : ledges) {
69 const auto& ledge = *source;
70 append(ledge.id);
71 appendNumber(ledge.closed ? 1 : 0);
72 appendNumber(ledge.localNormal.x);
73 appendNumber(ledge.localNormal.y);
74 appendNumber(ledge.localNormal.z);
75 for (const Vec3 normal : ledge.localNormals) {
76 appendNumber(normal.x);
77 appendNumber(normal.y);
78 appendNumber(normal.z);
79 }
80 for (const Vec3 point : ledge.points) {
81 appendNumber(point.x);
82 appendNumber(point.y);
83 appendNumber(point.z);
84 }
85 auto tags = ledge.tags;
86 std::sort(tags.begin(), tags.end());
87 for (const auto& tag : tags) append(tag);
88 }
89 std::vector<const ClimbingLadderBakeSource*> ladders;
90 ladders.reserve(request.ladders.size());
91 for (const auto& ladder : request.ladders) ladders.push_back(&ladder);
92 std::sort(ladders.begin(), ladders.end(), [](const auto* lhs, const auto* rhs) {
93 return lhs->id < rhs->id;
94 });
95 for (const auto* source : ladders) {
96 const auto& ladder = *source;
97 append(ladder.id);
98 appendNumber(ladder.bottomCenter.x);
99 appendNumber(ladder.bottomCenter.y);
100 appendNumber(ladder.bottomCenter.z);
101 appendNumber(ladder.localUp.x);

103 appendNumber(ladder.localUp.z);

106 appendNumber(ladder.localNormal.z);

110 auto tags = ladder.tags;
111 std::sort(tags.begin(), tags.end());
112 for (const auto& tag : tags) append(tag);

289eve::Result<ClimbingAnchorBakeRequest> decodeClimbingAnchorBakeRequest(const eve::Value& value) {
290 const auto* root = value.getIf<eve::Value::Object>();
291 if (!root)
292 return failure<ClimbingAnchorBakeRequest>(eve::DiagnosticCode::ParseError,
293 "anchor bake request must be an object");
294 ClimbingAnchorBakeRequest request;
295 std::string schemaId;
296 std::int64_t schemaVersion = -1;
297 const auto* schema = field(*root, "schemaVersion");
298 if (schema) {
299 if (const auto* integer = schema->getIf<std::int64_t>()) schemaVersion = *integer;
300 }
301 const auto* settingsValue = field(*root, "settings");
302 const auto* settings = settingsValue ? settingsValue->getIf<eve::Value::Object>() : nullptr;
303 const auto* ledgesValue = field(*root, "ledges");
304 const auto* ledges = ledgesValue ? ledgesValue->getIf<eve::Value::Array>() : nullptr;
305 const auto* laddersValue = field(*root, "ladders");
306 const auto* ladders = laddersValue ? laddersValue->getIf<eve::Value::Array>() : nullptr;
307 if (!readString(*root, "schemaId", schemaId) || schemaId != ClimbingAnchorBakeRequest::SchemaId ||
308 schemaVersion != ClimbingAnchorBakeRequest::SchemaVersion || !readString(*root, "graphId", request.graphId) ||
309 !readString(*root, "sourceGeometryContentId", request.sourceGeometryContentId) || !settings || !ledges ||
310 !ladders || !readFloat(*settings, "handSpacing", request.settings.handSpacing) ||
311 !readFloat(*settings, "hangingFeetDrop", request.settings.hangingFeetDrop) ||
312 !readFloat(*settings, "ladderMountOffset", request.settings.ladderMountOffset) ||
313 !readUInt32(*settings, "occupancySlots", request.settings.occupancySlots)) {
314 if (schemaVersion > ClimbingAnchorBakeRequest::SchemaVersion)
315 return failure<ClimbingAnchorBakeRequest>(eve::DiagnosticCode::UnknownVersion,
316 "anchor bake request schema version is unsupported",
317 "schemaVersion");
318 return failure<ClimbingAnchorBakeRequest>(eve::DiagnosticCode::ParseError,
319 "anchor bake request has missing or invalid fields");
320 }
321 request.ledges.reserve(ledges->size());
322 for (std::size_t index = 0; index < ledges->size(); ++index) {
323 const auto* object = (*ledges)[index].getIf<eve::Value::Object>();
324 ClimbingLedgeBakeSource source;
325 if (!object || !readString(*object, "id", source.id) || !readVecArray(*object, "points", source.points) ||
326 !readVecField(*object, "localNormal", source.localNormal) ||
327 !readVecArray(*object, "localNormals", source.localNormals) ||
328 !readBool(*object, "closed", source.closed) || !readStrings(*object, "tags", source.tags))
329 return failure<ClimbingAnchorBakeRequest>(eve::DiagnosticCode::ParseError,
330 "ledge bake source has missing or invalid fields",
331 "ledges." + std::to_string(index));
332 request.ledges.push_back(std::move(source));
333 }
334 request.ladders.reserve(ladders->size());
335 for (std::size_t index = 0; index < ladders->size(); ++index) {
336 const auto* object = (*ladders)[index].getIf<eve::Value::Object>();
337 ClimbingLadderBakeSource source;
338 if (!object || !readString(*object, "id", source.id) ||
339 !readVecField(*object, "bottomCenter", source.bottomCenter) ||
340 !readVecField(*object, "localUp", source.localUp) ||
341 !readVecField(*object, "localNormal", source.localNormal) ||
342 !readFloat(*object, "rungSpacing", source.rungSpacing) || !readFloat(*object, "width", source.width) ||
343 !readUInt32(*object, "rungCount", source.rungCount) || !readStrings(*object, "tags", source.tags))
344 return failure<ClimbingAnchorBakeRequest>(eve::DiagnosticCode::ParseError,
345 "ladder bake source has missing or invalid fields",
346 "ladders." + std::to_string(index));
347 request.ladders.push_back(std::move(source));
348 }
349 return eve::Result<ClimbingAnchorBakeRequest>::success(std::move(request));
350}

352eve::Result<ClimbingAnchorBakeResult> bakeClimbingAnchorGraph(const ClimbingAnchorBakeRequest& request) {
353 if (request.graphId.empty() || request.sourceGeometryContentId.empty())
354 return failure<ClimbingAnchorBakeResult>(eve::DiagnosticCode::InvalidArgument,
355 "graph and source geometry ids are required", "identity");
356 if (!finite(request.settings.handSpacing) || request.settings.handSpacing <= 0.f ||
357 !finite(request.settings.hangingFeetDrop) || request.settings.hangingFeetDrop <= 0.f ||
358 !finite(request.settings.ladderMountOffset) || request.settings.ladderMountOffset < 0.f ||
359 request.settings.occupancySlots == 0 || request.settings.occupancySlots > 8)
360 return failure<ClimbingAnchorBakeResult>(eve::DiagnosticCode::InvalidArgument,
361 "bake settings are outside supported bounds", "settings");
362 if (request.ledges.empty() && request.ladders.empty())
363 return failure<ClimbingAnchorBakeResult>(eve::DiagnosticCode::InvalidArgument,
364 "at least one ledge or ladder source is required", "sources");
365
366 ClimbingAnchorBakeResult result;
367 result.graph.id = request.graphId;
368 result.graph.sourceGeometryContentId = request.sourceGeometryContentId;
369 result.graph.buildSettingsHash = hexHash(request);
370 std::unordered_set<std::string> sourceIds;
371
372 for (std::size_t sourceIndex = 0; sourceIndex < request.ledges.size(); ++sourceIndex) {
373 const auto& source = request.ledges[sourceIndex];
374 const std::string path = "ledges." + std::to_string(sourceIndex);
375 if (source.id.empty() || !sourceIds.emplace(source.id).second || source.points.size() < 2 ||
376 (!source.localNormals.empty() && source.localNormals.size() != source.points.size()) ||
377 !uniqueNonEmpty(source.tags))
378 return failure<ClimbingAnchorBakeResult>(eve::DiagnosticCode::InvalidArgument,
379 "ledge id/tags/normals are invalid or it needs two points", path);
380 std::vector<std::string> ids;
381 ids.reserve(source.points.size());
382 for (std::size_t index = 0; index < source.points.size(); ++index) {
383 if (!finite(source.points[index]))
384 return failure<ClimbingAnchorBakeResult>(eve::DiagnosticCode::InvalidArgument,
385 "ledge points must be finite", path + ".points");
386 const std::size_t previous = index == 0 ? (source.closed ? source.points.size() - 1 : 0) : index - 1;
387 const std::size_t next = index + 1 == source.points.size() ? (source.closed ? 0 : index) : index + 1;
388 const Vec3 normalValue = source.localNormals.empty() ? source.localNormal : source.localNormals[index];
389 auto normalResult = normalized(normalValue, path + ".localNormals." + std::to_string(index));
390 if (!normalResult) return eve::Result<ClimbingAnchorBakeResult>::failure(normalResult.status());
391 const Vec3 normal = normalResult.value();
392 const Vec3 segment = index + 1 < source.points.size() || source.closed
393 ? subtract(source.points[next], source.points[index])
394 : subtract(source.points[index], source.points[previous]);
395 auto tangentResult = normalized(segment,
396 path + ".points." + std::to_string(index));
397 if (!tangentResult) return eve::Result<ClimbingAnchorBakeResult>::failure(tangentResult.status());
398 Vec3 tangent = tangentResult.value();
399 tangent = subtract(tangent, multiply(normal, dot(tangent, normal)));
400 tangentResult = normalized(tangent, path + ".frame." + std::to_string(index));
401 if (!tangentResult) return eve::Result<ClimbingAnchorBakeResult>::failure(tangentResult.status());
402 tangent = tangentResult.value();
403 ClimbingAnchorKind kind = ClimbingAnchorKind::Ledge;
404 if (index != 0 && (index + 1 != source.points.size() || source.closed)) {
405 auto incoming = normalized(subtract(source.points[index], source.points[previous]), path);
406 auto outgoing = normalized(subtract(source.points[next], source.points[index]), path);
407 if (!incoming) return eve::Result<ClimbingAnchorBakeResult>::failure(incoming.status());
408 if (!outgoing) return eve::Result<ClimbingAnchorBakeResult>::failure(outgoing.status());
409 const float turn = cross(incoming.value(), outgoing.value()).y;
410 if (std::fabs(turn) > 0.05f)
411 kind = turn > 0.f ? ClimbingAnchorKind::CornerOuter : ClimbingAnchorKind::CornerInner;
412 }
413 const std::string id = source.id + "." + std::to_string(index);
414 ids.push_back(id);
415 result.graph.nodes.push_back(ledgeNode(id, kind, source.points[index], normal, tangent,
416 request.settings, source.tags));
417 ++result.ledgeNodeCount;
418 }
419 const std::size_t edgeCount = source.closed ? ids.size() : ids.size() - 1;
420 for (std::size_t index = 0; index < edgeCount; ++index) {
421 const std::size_t next = (index + 1) % ids.size();
422 const auto& target = result.graph.nodes[result.graph.nodes.size() - ids.size() + next];
423 result.graph.edges.push_back({ids[index], ids[next],
424 target.kind == ClimbingAnchorKind::Ledge
425 ? ClimbingAnchorEdgeKind::Shimmy
426 : ClimbingAnchorEdgeKind::Corner,
427 true, {}, {}});
428 }
429 }
430
431 for (std::size_t sourceIndex = 0; sourceIndex < request.ladders.size(); ++sourceIndex) {
432 const auto& source = request.ladders[sourceIndex];
433 const std::string path = "ladders." + std::to_string(sourceIndex);
434 if (source.id.empty() || !sourceIds.emplace(source.id).second || !finite(source.bottomCenter) ||
435 !finite(source.rungSpacing) || source.rungSpacing <= 0.f || !finite(source.width) ||
436 source.width <= 0.f || source.rungCount < 2 || source.rungCount > 1024 ||
437 !uniqueNonEmpty(source.tags))
438 return failure<ClimbingAnchorBakeResult>(eve::DiagnosticCode::InvalidArgument,
439 "ladder source is outside supported bounds", path);
440 auto upResult = normalized(source.localUp, path + ".localUp");
441 auto normalResult = normalized(source.localNormal, path + ".localNormal");
442 if (!upResult) return eve::Result<ClimbingAnchorBakeResult>::failure(upResult.status());
443 if (!normalResult) return eve::Result<ClimbingAnchorBakeResult>::failure(normalResult.status());
444 const Vec3 up = upResult.value();
445 const Vec3 normal = normalResult.value();
446 auto tangentResult = normalized(cross(up, normal), path + ".frame");
447 if (!tangentResult) return eve::Result<ClimbingAnchorBakeResult>::failure(tangentResult.status());
448 const Vec3 tangent = tangentResult.value();
449 std::vector<std::string> ids;
450 ids.reserve(source.rungCount);
451 for (std::uint32_t index = 0; index < source.rungCount; ++index) {
452 ClimbingAnchorNodeDefinition node;
453 node.id = source.id + ".rung." + std::to_string(index);
454 node.kind = ClimbingAnchorKind::LadderRung;
455 node.localPosition = add(source.bottomCenter, multiply(up, source.rungSpacing * static_cast<float>(index)));
456 node.localNormal = normal;
457 node.localTangent = tangent;
458 node.leftHandSocket = add(node.localPosition, multiply(tangent, -source.width * 0.5f));
459 node.rightHandSocket = add(node.localPosition, multiply(tangent, source.width * 0.5f));
460 node.feetSocket = add(node.localPosition, multiply(up, -source.rungSpacing * 2.f));
461 node.occupancySlots = request.settings.occupancySlots;
462 node.tags = source.tags;
463 node.tags.push_back("ladder");

469 }

471 endpointTags.push_back("ladder_mount");

478 request.settings, endpointTags));

484 const Vec3 top = add(source.bottomCenter,
485 multiply(up, source.rungSpacing * static_cast<float>(source.rungCount - 1)));
486 result.graph.nodes.push_back(ledgeNode(
487 dismountId, ClimbingAnchorKind::Ledge,
488 add(add(top, multiply(up, request.settings.ladderMountOffset)),
489 multiply(normal, request.settings.ladderMountOffset)),
490 normal, tangent, request.settings, endpointTags));
491 result.graph.edges.push_back({mountId, ids.front(), ClimbingAnchorEdgeKind::Mount, false, {}, {}});
492 for (std::size_t index = 0; index + 1 < ids.size(); ++index)
493 result.graph.edges.push_back({ids[index], ids[index + 1], ClimbingAnchorEdgeKind::Climb, true, {}, {}});
494 result.graph.edges.push_back(
495 {ids.back(), dismountId, ClimbingAnchorEdgeKind::Dismount, false, {}, {}});
496 }
497
498 auto valid = validateClimbingAnchorGraphDefinition(result.graph);
499 if (!valid) return eve::Result<ClimbingAnchorBakeResult>::failure(valid.status());
500 auto encoded = encodeClimbingAnchorGraphDefinition(result.graph);
501 if (!encoded) return eve::Result<ClimbingAnchorBakeResult>::failure(encoded.status());
502 auto canonical = decodeClimbingAnchorGraphDefinition(encoded.value());
503 if (!canonical) return eve::Result<ClimbingAnchorBakeResult>::failure(canonical.status());
504 result.graph = std::move(canonical.value());
505 return eve::Result<ClimbingAnchorBakeResult>::success(std::move(result));
506}

508eve::Result<ClimbingAnchorAuthoringOverlay> inspectClimbingAnchorGraphAuthoring(
509 const ClimbingAnchorGraphDefinition& graph) {
510 auto valid = validateClimbingAnchorGraphDefinition(graph);
511 if (!valid) return eve::Result<ClimbingAnchorAuthoringOverlay>::failure(valid.status());
512 ClimbingAnchorAuthoringOverlay overlay;
513 overlay.graphId = graph.id;
514 overlay.buildSettingsHash = graph.buildSettingsHash;
515 overlay.nodes.reserve(graph.nodes.size());
516 for (const auto& node : graph.nodes)
517 overlay.nodes.push_back({node.id, node.kind, node.localPosition, node.localNormal, node.localTangent,
518 node.leftHandSocket, node.rightHandSocket, node.feetSocket,
519 node.occupancySlots});
520 overlay.edges.reserve(graph.edges.size());
521 for (const auto& edge : graph.edges)
522 overlay.edges.push_back({edge.from, edge.to, edge.kind, edge.bidirectional});
523 std::sort(overlay.nodes.begin(), overlay.nodes.end(),
524 [](const auto& lhs, const auto& rhs) { return lhs.id < rhs.id; });
525 std::sort(overlay.edges.begin(), overlay.edges.end(), [](const auto& lhs, const auto& rhs) {
526 if (lhs.from != rhs.from) return lhs.from < rhs.from;
527 if (lhs.to != rhs.to) return lhs.to < rhs.to;
528 return lhs.kind < rhs.kind;
529 });
530 return eve::Result<ClimbingAnchorAuthoringOverlay>::success(std::move(overlay));
531}
