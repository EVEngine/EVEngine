31template <class T>
33struct Field {
34 std::string name;
36 std::function<Value(const T&)> read;
38 std::function<void(T&, const Value&)> write;
39 std::string sqlType = "TEXT";
40 bool primaryKey = false;
41};

43template <class T>
45struct Mapping {
46 std::string table;
47 std::vector<Field<T>> fields;
48
50 Row toRow(const T& object) const {
51 Row row;
52 for (const auto& field : fields) row[field.name] = field.read(object);
53 return row;
54 }
55
57 T fromRow(const Row& row) const {
58 T object{};
59 for (const auto& field : fields) {
60 auto it = row.find(field.name);
61 if (it != row.end()) field.write(object, it->second);
62 }
63 return object;
64 }
65};

400template <class T>
402class Repository {
403public:
405 Repository(Connection& connection, Mapping<T> mapping) : connection_(connection), mapping_(std::move(mapping)) {}
406
408 void createTable(bool ifNotExists = true) {
409 if (mapping_.fields.empty()) throw std::invalid_argument("ORM mapping has no fields");
410 std::string sql = "CREATE TABLE ";
411 if (ifNotExists) sql += "IF NOT EXISTS ";
412 sql += mapping_.table + " (";
413 for (std::size_t i = 0; i < mapping_.fields.size(); ++i) {
414 const auto& field = mapping_.fields[i];
415 if (i) sql += ',';
416 sql += field.name + " " + field.sqlType;
417 if (field.primaryKey) sql += " PRIMARY KEY";
418 }
419 connection_.execute(sql + ")");
420 }
421
423 int insert(const T& object) { return connection_.insert(mapping_.table, mapping_.toRow(object)); }
424
426 std::vector<T> all() {
427 std::vector<T> objects;
428 for (const auto& row : connection_.query("SELECT * FROM " + mapping_.table))
429 objects.push_back(mapping_.fromRow(row));
430 return objects;
431 }
432
434 bool find(const Value& id, T& object) {
435 auto rows = connection_.queryByKey(mapping_.table, primaryKey().name, id);
436 if (rows.empty()) return false;
437 object = mapping_.fromRow(rows.front());
438 return true;
439 }
440
442 int update(const T& object) {
443 const auto& key = primaryKey();
444 Row values = mapping_.toRow(object);
445 Value id = values.at(key.name);
446 values.erase(key.name);
447 return connection_.updateByKey(mapping_.table, values, key.name, id);
448 }
449
451 int remove(const Value& id) { return connection_.removeByKey(mapping_.table, primaryKey().name, id); }
452
454 std::vector<T> select(const std::function<void(Query&)>& configure) {
456 Query query(&connection_, mapping_.table);
458 configure(query);
459 std::vector<T> objects;
460 for (const auto& row : query.all()) objects.push_back(mapping_.fromRow(row));
461 return objects;
462 }
463
464private:
465 const Field<T>& primaryKey() const {
466 for (const auto& field : mapping_.fields)
467 if (field.primaryKey) return field;
468 throw std::logic_error("ORM mapping requires one primary key field");
469 }

471 Connection& connection_;
472 Mapping<T> mapping_;
