28template <class T>
29struct Field {
30 std::string name;
31 std::function<Value(const T&)> read;
32 std::function<void(T&, const Value&)> write;
33 std::string sqlType = "TEXT";
34 bool primaryKey = false;
35};

37template <class T>
38struct Mapping {
39 std::string table;
40 std::vector<Field<T>> fields;
41
42 Row toRow(const T& object) const {
43 Row row;
44 for (const auto& field : fields) row[field.name] = field.read(object);
45 return row;
46 }
47
48 T fromRow(const Row& row) const {
49 T object{};
50 for (const auto& field : fields) {
51 auto it = row.find(field.name);
52 if (it != row.end()) field.write(object, it->second);
53 }
54 return object;
55 }
56};

157template <class T>
158class Repository {
159public:
160 Repository(Connection& connection, Mapping<T> mapping) : connection_(connection), mapping_(std::move(mapping)) {}
161
162 void createTable(bool ifNotExists = true) {
163 if (mapping_.fields.empty()) throw std::invalid_argument("ORM mapping has no fields");
164 std::string sql = "CREATE TABLE ";
165 if (ifNotExists) sql += "IF NOT EXISTS ";
166 sql += mapping_.table + " (";
167 for (std::size_t i = 0; i < mapping_.fields.size(); ++i) {
168 const auto& field = mapping_.fields[i];
169 if (i) sql += ',';
170 sql += field.name + " " + field.sqlType;
171 if (field.primaryKey) sql += " PRIMARY KEY";
172 }
173 connection_.execute(sql + ")");
174 }
175
176 int insert(const T& object) { return connection_.insert(mapping_.table, mapping_.toRow(object)); }
177
178 std::vector<T> all() {
179 std::vector<T> objects;
180 for (const auto& row : connection_.query("SELECT * FROM " + mapping_.table))
181 objects.push_back(mapping_.fromRow(row));
182 return objects;
183 }
184
185 bool find(const Value& id, T& object) {
186 auto rows = connection_.queryByKey(mapping_.table, primaryKey().name, id);
187 if (rows.empty()) return false;
188 object = mapping_.fromRow(rows.front());
189 return true;
190 }
191
192 int update(const T& object) {
193 const auto& key = primaryKey();
194 Row values = mapping_.toRow(object);
195 Value id = values.at(key.name);
196 values.erase(key.name);
197 return connection_.updateByKey(mapping_.table, values, key.name, id);
198 }
199
200 int remove(const Value& id) { return connection_.removeByKey(mapping_.table, primaryKey().name, id); }
201
202 std::vector<T> select(const std::function<void(Query&)>& configure) {
203 Query query(&connection_, mapping_.table);
204 configure(query);
205 std::vector<T> objects;
206 for (const auto& row : query.all()) objects.push_back(mapping_.fromRow(row));
207 return objects;
208 }
209
210private:
211 const Field<T>& primaryKey() const {
212 for (const auto& field : mapping_.fields)
213 if (field.primaryKey) return field;
214 throw std::logic_error("ORM mapping requires one primary key field");
215 }
216
217 Connection& connection_;
218 Mapping<T> mapping_;
219};
