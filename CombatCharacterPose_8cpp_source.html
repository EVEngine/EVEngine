33double distance3(double ax, double ay, double az, double bx, double by, double bz) {
34 const double dx = ax - bx;
35 const double dy = ay - by;
36 const double dz = az - bz;
37 return std::sqrt(dx * dx + dy * dy + dz * dz);
38}

96Result<std::optional<action::AbilityIntent>> CombatEnemyIntentSource::nextIntent(SimulationTick tick) {
97 if (!targets_)
98 return Result<std::optional<action::AbilityIntent>>::failure(
99 Diagnostic::error(DiagnosticCode::NotFound, "enemy target runtime is unavailable", "targets"));
100
101 std::optional<action::AbilityIntent> emitted;
102 for (auto& [key, enemy] : enemies_) {
103 (void)key;
104 if (enemy.phase == CombatEnemyPhase::Recover &&
105 tick.value() >= enemy.phaseStartTick + enemy.definition.recoverTicks)
106 enemy.phase = CombatEnemyPhase::Idle;
107 if (emitted) continue;
108 if (!enemy.hasAction) continue;
109 if (enemy.definition.lightGrant.isZero()) continue;
110
111 const auto emitAttack = [&]() {
112 action::AbilityIntent intent;
113 intent.grantId = enemy.definition.lightGrant;
114 intent.request.actionId = enemy.lightAction;
115 enemy.lastAttackTick = tick.value();
116 enemy.phase = enemy.definition.recoverTicks > 0 ? CombatEnemyPhase::Recover : CombatEnemyPhase::Idle;
117 enemy.phaseStartTick = tick.value();
118 emitted = std::move(intent);
119 };
120
121 if (enemy.phase == CombatEnemyPhase::Telegraph) {
122 if (tick.value() >= enemy.phaseStartTick + enemy.definition.telegraphTicks) emitAttack();
123 continue;
124 }
125 if (enemy.phase == CombatEnemyPhase::Recover) continue;
126
127 const auto selfPose = poses_.find(enemy.definition.subject.format());
128 if (selfPose == poses_.end()) continue;
129 auto lock = targets_->state(enemy.definition.subject);
130 if (!lock || !lock.value().target) continue;
131 const auto targetPose = poses_.find(lock.value().target->format());
132 if (targetPose == poses_.end()) continue;
133 const double dist = distance3(selfPose->second.x, selfPose->second.y, selfPose->second.z, targetPose->second.x,
134 targetPose->second.y, targetPose->second.z);
135 if (dist > enemy.definition.nearRadius) continue;
136 if (enemy.lastAttackTick != 0 && tick.value() < enemy.lastAttackTick + enemy.definition.attackCooldownTicks)
137 continue;
138
139 if (enemy.definition.telegraphTicks > 0) {
140 enemy.phase = CombatEnemyPhase::Telegraph;
141 enemy.phaseStartTick = tick.value();
142 continue;
143 }
144 emitAttack();
145 }
146 return Result<std::optional<action::AbilityIntent>>::success(std::move(emitted));
147}

149Result<std::vector<CombatEnemySteering>> CombatEnemyIntentSource::nextSteering(SimulationTick) const {
150 if (!targets_)
151 return Result<std::vector<CombatEnemySteering>>::failure(
152 Diagnostic::error(DiagnosticCode::NotFound, "enemy target runtime is unavailable", "targets"));
153
154 std::vector<CombatEnemySteering> result;
155 for (const auto& [key, enemy] : enemies_) {
156 (void)key;
157 if (enemy.phase != CombatEnemyPhase::Idle) continue;
158 const auto selfPose = poses_.find(enemy.definition.subject.format());
159 if (selfPose == poses_.end()) continue;
160 auto lock = targets_->state(enemy.definition.subject);
161 if (!lock || !lock.value().target) continue;
162 const auto targetPose = poses_.find(lock.value().target->format());
163 if (targetPose == poses_.end()) continue;
164 const double dist = distance3(selfPose->second.x, selfPose->second.y, selfPose->second.z, targetPose->second.x,
165 targetPose->second.y, targetPose->second.z);
166 if (dist <= enemy.definition.nearRadius) continue;
167
168 CombatVector3 delta{targetPose->second.x - selfPose->second.x, 0.0, targetPose->second.z - selfPose->second.z};
169 const double magnitude = std::hypot(delta.x, delta.z);
170 if (magnitude <= 1e-6) continue;
171 CombatEnemySteering steering;
172 steering.subject = enemy.definition.subject;
173 steering.moveDirection = {delta.x / magnitude, 0.0, delta.z / magnitude};
174 steering.speedFraction = dist <= enemy.definition.midRadius ? 0.7 : 1.0;
175 result.push_back(steering);
176 }
177 return Result<std::vector<CombatEnemySteering>>::success(std::move(result), Status::success(StatusCode::Applied));
178}

180CombatEnemyBand CombatEnemyIntentSource::band(SubjectRef subject) const {
181 const auto found = enemies_.find(subject.format());
182 if (found == enemies_.end() || !targets_) return CombatEnemyBand::Far;
183 const auto selfPose = poses_.find(subject.format());
184 if (selfPose == poses_.end()) return CombatEnemyBand::Far;
185 auto lock = targets_->state(subject);
186 if (!lock || !lock.value().target) return CombatEnemyBand::Far;
187 const auto targetPose = poses_.find(lock.value().target->format());
188 if (targetPose == poses_.end()) return CombatEnemyBand::Far;
189 const double dist = distance3(selfPose->second.x, selfPose->second.y, selfPose->second.z, targetPose->second.x,
190 targetPose->second.y, targetPose->second.z);
191 if (dist <= found->second.definition.nearRadius) return CombatEnemyBand::Near;
192 if (dist <= found->second.definition.midRadius) return CombatEnemyBand::Mid;
193 return CombatEnemyBand::Far;
194}

348Result<CombatLocomotionAdvance> CombatLocomotionRuntime::advance(const SimulationStep& step) {
349 if (step.delta < Duration::zero() || step.tick < lastTick_)
350 return Result<CombatLocomotionAdvance>::failure(Diagnostic::error(
351 DiagnosticCode::InvalidArgument, "locomotion step is negative or moves tick backwards", "step"));
352 auto candidate = states_;
353 CombatLocomotionAdvance result;
354 result.tick = step.tick;
355 const double seconds = step.delta.seconds();
356 for (auto& [key, state] : candidate) {
357 (void)key;
358 bool arrived = false;
359 std::optional<CombatNavigationGoal> activeGoal = state.navigationGoal;
360 if (state.navigationGoal) {
361 if (!navigation_)
362 return Result<CombatLocomotionAdvance>::failure(
363 Diagnostic::error(DiagnosticCode::Unsupported, "active goal has no combat navigation provider",
364 state.subject.format()));
365 auto steering = navigation_->steer(state, *state.navigationGoal, step.tick);
366 if (!steering) return Result<CombatLocomotionAdvance>::failure(steering.status());
367 auto valid = validateSteering(steering.value());
368 if (!valid) return Result<CombatLocomotionAdvance>::failure(valid.status());
369 arrived = steering.value().phase == CombatNavigationPhase::Arrived;
370 state.moveDirection = arrived ? CombatVector2{} : normalized(steering.value().direction);
371 state.moveSpeedFraction = arrived ? 0.0 : steering.value().speedFraction;
372 if (arrived) {
373 state.navigationGoal.reset();
374 state.velocity = {};
375 }
376 }
377 const CombatVector2 previous = state.position;
378 const CombatVector2 desired{state.moveDirection.x * state.maximumSpeed * state.moveSpeedFraction,
379 state.moveDirection.z * state.maximumSpeed * state.moveSpeedFraction};
380 const CombatVector2 velocityDelta{desired.x - state.velocity.x, desired.z - state.velocity.z};
381 const double deltaLength = length(velocityDelta);
382 const double maximumDelta = state.acceleration * seconds;
383 const double scale = deltaLength > maximumDelta && deltaLength > 0.0 ? maximumDelta / deltaLength : 1.0;
384 state.velocity.x += velocityDelta.x * scale;
385 state.velocity.z += velocityDelta.z * scale;
386 state.position.x += state.velocity.x * seconds;
387 state.position.z += state.velocity.z * seconds;
388 if (activeGoal && !arrived) {
389 const CombatVector2 before{activeGoal->position.x - previous.x, activeGoal->position.z - previous.z};
390 const CombatVector2 after{activeGoal->position.x - state.position.x,
391 activeGoal->position.z - state.position.z};
392 if (length(after) <= activeGoal->acceptanceRadius || before.x * after.x + before.z * after.z <= 0.0) {
393 state.position = activeGoal->position;
394 state.velocity = {};
395 state.moveDirection = {};
396 state.moveSpeedFraction = 0.0;
397 state.navigationGoal.reset();
398 arrived = true;
399 }
400 }
401 if (length(state.velocity) > 0.0) state.facing = normalized(state.velocity);
402 if (state.position.x != previous.x || state.position.z != previous.z)
403 result.events.push_back(
404 {state.subject, CombatLocomotionEventKind::Moved, previous, state.position, step.tick});
405 if (arrived)
406 result.events.push_back(
407 {state.subject, CombatLocomotionEventKind::Arrived, previous, state.position, step.tick});
408 }
409 states_ = std::move(candidate);
410 lastTick_ = step.tick;
411 return Result<CombatLocomotionAdvance>::success(std::move(result), Status::success(StatusCode::Applied));
412}

446Result<CombatLoopFrame> CombatLoopRuntime::advance(const SimulationStep& step) {
447 if (!characters_ || !melee_)
448 return Result<CombatLoopFrame>::failure(
449 Diagnostic::error(DiagnosticCode::NotFound, "combat loop requires character and melee runtimes", "loop"));
450
451 CombatLoopFrame frame;
452 frame.tick = step.tick;
453
454 if (enemies_) {
455 for (const auto& state : characters_->states()) {
456 auto posed = enemies_->setPosition(state.subject, state.position.x, state.position.y, state.position.z);
457 if (!posed) return Result<CombatLoopFrame>::failure(posed.status());
458 }
459 auto steering = enemies_->nextSteering(step.tick);
460 if (!steering) return Result<CombatLoopFrame>::failure(steering.status());
461 frame.enemySteering = steering.value();
462 for (const auto& steer : frame.enemySteering) {
463 auto moved = characters_->setMoveIntent(steer.subject, steer.moveDirection, steer.speedFraction);
464 if (!moved) return Result<CombatLoopFrame>::failure(moved.status());
465 }
466 }
467
468 if (cancelResolver_ && cancelSubject_.isValid() && !cancelAbility_.format().empty()) {
469 CombatCancelResolveRequest request;
470 request.subject = cancelSubject_;
471 request.currentAbility = cancelAbility_;
472 request.tick = step.tick;
473 auto resolved = cancelResolver_->resolve(request);
474 if (resolved)
475 frame.cancels.push_back(std::move(resolved).takeValue());
476 else if (resolved.status().code() != StatusCode::NotFound)
477 return Result<CombatLoopFrame>::failure(resolved.status());
478 }
479
480 if (feel_) {
481 for (const auto& state : characters_->states()) {
482 auto frozen = characters_->setTimeFrozen(state.subject, feel_->isFrozen(state.subject));
483 if (!frozen) return Result<CombatLoopFrame>::failure(frozen.status());
484 }
485 }
486
487 if (targets_) {
488 for (const auto& state : characters_->states()) {
489 if (state.mode != CombatCharacterMode::Attacking) continue;
490 if (feel_ && feel_->isFrozen(state.subject)) continue;
491 auto lock = targets_->state(state.subject);
492 if (!lock || !lock.value().target) continue;
493 auto target = characters_->state(*lock.value().target);
494 if (!target) continue;
495 CombatWarpRequest warp;
496 warp.attacker = state.position;
497 warp.target = target.value().position;
498 warp.attackerFacing = state.facing;
499 warp.desiredDistance = 1.4;
500 warp.maxTranslation = 0.12;
501 warp.remainingBudget = 0.6;
502 auto solved = CombatMotionWarp::solve(warp);
503 if (!solved) {
504 if (solved.status().code() == StatusCode::Conflict) continue;
505 return Result<CombatLoopFrame>::failure(solved.status());
506 }
507 auto faced = characters_->setFacing(state.subject, solved.value().facing);
508 if (!faced) return Result<CombatLoopFrame>::failure(faced.status());
509 if (solved.value().translation.x != 0.0 || solved.value().translation.z != 0.0) {
510 auto applied = characters_->setRootMotionDelta(state.subject, solved.value().translation);
511 if (!applied) return Result<CombatLoopFrame>::failure(applied.status());
512 }
513 }
514 }
515
516 auto locomotion = characters_->advance(step);
517 if (!locomotion) return Result<CombatLoopFrame>::failure(locomotion.status());
518 frame.locomotion = std::move(locomotion).takeValue();
519
520 for (const auto& state : characters_->states()) {
521 auto posed = melee_->setHurtboxPose(state.subject, "torso",
522 {{state.position.x, state.position.y + 1.0, state.position.z}, 0.0});
523 if (!posed && posed.status().code() != StatusCode::NotFound)
524 return Result<CombatLoopFrame>::failure(posed.status());

526 }
527

529 if (!melee) return Result<CombatLoopFrame>::failure(melee.status());
530 frame.melee = std::move(melee).takeValue();
531
532 if (states_) {
533 for (const auto& hit : frame.melee.hits) {
534 if (dodging(characters_, hit.target)) {
535 frame.perfectDodges.push_back({hit.target, hit.source, hit.hitboxId});
536 continue;
537 }
538 if (invulnerable(windows_, characters_, hit.target)) continue;
539 DamageRequest request;
540 request.source = hit.source;
541 request.target = hit.target;
542 request.actionExecution = hit.actionExecution;
543 request.damageType = hit.damageType;
544 request.healthDamage = hit.healthDamage;
545 request.poiseDamage = hit.poiseDamage;
546 request.knockback = hit.knockback;
547 if (guards_) {
548 auto guarded = guards_->mitigate(hit.target, request);
549 if (!guarded) return Result<CombatLoopFrame>::failure(guarded.status());
550 frame.guards.push_back(guarded.value());
551 if (guarded.value().result != GuardResult::None) continue;
552 }
553 const auto found = states_->find(hit.target.format());
554 if (found == states_->end())
555 return Result<CombatLoopFrame>::failure(
556 Diagnostic::error(DiagnosticCode::NotFound, "loop damage target missing", hit.target.format()));
557 DamageRuntime damage;
558 auto outcome = damage.apply(found->second, request);
559 if (!outcome) return Result<CombatLoopFrame>::failure(outcome.status());
560 if (feel_) {
561 auto felt = feel_->applyFromOutcome(outcome.value());
562 if (!felt) return Result<CombatLoopFrame>::failure(felt.status());
563 }
564 auto reacted = characters_->applyDamageReaction(
565 hit.target, outcome.value().reaction,
566 feel_ ? feel_->state(hit.target).hitstunRemaining : Duration::zero(), outcome.value().knockback);
567 if (!reacted) return Result<CombatLoopFrame>::failure(reacted.status());
568 frame.outcomes.push_back(std::move(outcome).takeValue());
569 }
570 }
571
572 if (feel_) {
573 auto felt = feel_->advance(step);
574 if (!felt) return Result<CombatLoopFrame>::failure(felt.status());
575 }
576
577 if (cameraFocus_.isValid()) {
578 auto focus = characters_->state(cameraFocus_);
579 if (focus) {
580 CombatCameraFramingRequest framing;
581 framing.player = focus.value().position;
582 framing.playerFacing = focus.value().facing;
583 if (targets_) {
584 auto lock = targets_->state(cameraFocus_);
585 if (lock && lock.value().target) {
586 auto locked = characters_->state(*lock.value().target);
587 if (locked) framing.lockTarget = locked.value().position;
588 }
589 }
590 auto view = CombatCameraFraming::solve(framing);
591 if (!view) return Result<CombatLoopFrame>::failure(view.status());
592 frame.camera = view.value();
593 }
594 }
595 return Result<CombatLoopFrame>::success(std::move(frame), Status::success(StatusCode::Applied));
596}
