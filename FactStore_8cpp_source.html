38eve::Result<decision::ConditionReasonCode> parseReasonCode(std::string_view text) {
39 using R = decision::ConditionReasonCode;
40 static const std::pair<std::string_view, R> names[] = {
41 {"passed", R::Passed},
42 {"child_failed", R::ChildFailed},
43 {"no_child_passed", R::NoChildPassed},
44 {"negated", R::Negated},
45 {"missing_value", R::MissingValue},
46 {"value_mismatch", R::ValueMismatch},
47 {"tag_missing", R::TagMissing},
48 {"tag_unavailable", R::TagUnavailable},
49 {"attribute_missing", R::AttributeMissing},
50 {"resource_missing", R::ResourceMissing},
51 {"state_missing", R::StateMissing},
52 {"state_mismatch", R::StateMismatch},
53 {"authority_denied", R::AuthorityDenied},
54 {"authority_unavailable", R::AuthorityUnavailable},
55 {"policy_rejected", R::PolicyRejected},
56 {"policy_unavailable", R::PolicyUnavailable},
57 {"invalid_condition", R::InvalidCondition},
58 };
59 for (const auto& [name, code] : names) {

62 return eve::Result<decision::ConditionReasonCode>::failure(eve::Diagnostic::error(
63 eve::DiagnosticCode::InvalidArgument, "unknown condition reason code", "reason", {}, "emergence.fact"));

66eve::Result<decision::ConditionResult> decodePolicyResult(const eve::Value& value) {
67 const auto* object = value.getIf<eve::Value::Object>();
68 if (object == nullptr)
69 return eve::Result<decision::ConditionResult>::failure(eve::Diagnostic::error(
70 eve::DiagnosticCode::InvalidArgument, "policy result must be an object", {}, {}, "emergence.fact"));
71 if (!exactFields(*object, {"passed", "reason", "evidence", "details"}))
72 return eve::Result<decision::ConditionResult>::failure(eve::Diagnostic::error(
73 eve::DiagnosticCode::InvalidArgument, "policy result contains an unknown field", {}, {}, "emergence.fact"));
74 const auto* passed = field(*object, "passed");
75 const auto* reason = field(*object, "reason");
76 if (passed == nullptr || !passed->isBool())
77 return eve::Result<decision::ConditionResult>::failure(eve::Diagnostic::error(
78 eve::DiagnosticCode::InvalidArgument, "policy passed must be a boolean", "passed", {}, "emergence.fact"));
79 if (reason == nullptr || !reason->isString())
80 return eve::Result<decision::ConditionResult>::failure(eve::Diagnostic::error(
81 eve::DiagnosticCode::InvalidArgument, "policy reason must be a string", "reason", {}, "emergence.fact"));
82 auto code = parseReasonCode(reason->asString());
83 if (!code) return eve::Result<decision::ConditionResult>::failure(code.status());
84 eve::Value evidence;
85 eve::Value details = eve::Value::Object{};
86 if (const auto* evidenceValue = field(*object, "evidence")) evidence = *evidenceValue;
87 if (const auto* detailsValue = field(*object, "details")) details = *detailsValue;
88 if (passed->asBool())
89 return eve::Result<decision::ConditionResult>::success(
90 decision::ConditionResult::success(std::move(evidence), std::move(details)));
91 return eve::Result<decision::ConditionResult>::success(
92 decision::ConditionResult::failed(code.value(), std::move(evidence), std::move(details)));
93}

113eve::Result<std::pair<FactDomain, std::string>> parseFactKey(std::string_view key) {
114 const auto fail = [](const char* message) {
115 return eve::Result<std::pair<FactDomain, std::string>>::failure(
116 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, message, "key", {}, "emergence.fact"));
117 };
118 if (key.empty()) return fail("fact key must be non-empty");
119 const auto slash = key.find(':');
120 if (slash == std::string_view::npos || slash == 0 || slash + 1 >= key.size())
121 return fail("fact key must use domain:name form");
122 const auto domainText = key.substr(0, slash);
123 const auto name = key.substr(slash + 1);
124 FactDomain domain = FactDomain::Value;
125 if (domainText == "value")
126 domain = FactDomain::Value;
127 else if (domainText == "tag")
128 domain = FactDomain::Tag;
129 else if (domainText == "attr")
130 domain = FactDomain::Attribute;
131 else if (domainText == "resource")
132 domain = FactDomain::Resource;
133 else if (domainText == "state")
134 domain = FactDomain::State;
135 else if (domainText == "authority")
136 domain = FactDomain::Authority;
137 else if (domainText == "policy")
138 domain = FactDomain::Policy;
139 else
140 return fail("unknown fact domain prefix");
141 return eve::Result<std::pair<FactDomain, std::string>>::success(std::make_pair(domain, std::string(name)));
142}

282eve::Result<void> FactStore::restoreJson(std::string_view json) {
283 auto parsed = eve::Value::fromJson(json);
284 if (!parsed) return eve::Result<void>::failure(parsed.status());
285 const auto* object = parsed.value().getIf<eve::Value::Object>();
286 if (object == nullptr)
287 return eve::Result<void>::failure(eve::Diagnostic::error(
288 eve::DiagnosticCode::InvalidArgument, "fact snapshot must be an object", {}, {}, "emergence.fact"));
289 if (!exactFields(*object, {"schema", "version", "values", "tags", "attributes", "resources", "states",
290 "authorities", "policies"}))
291 return eve::Result<void>::failure(eve::Diagnostic::error(
292 eve::DiagnosticCode::InvalidArgument, "fact snapshot contains an unknown field", {}, {}, "emergence.fact"));
293 const auto* schema = field(*object, "schema");
294 const auto* version = field(*object, "version");
295 if (schema == nullptr || !schema->isString() || schema->asString() != "eve.emergence.facts")
296 return eve::Result<void>::failure(eve::Diagnostic::error(
297 eve::DiagnosticCode::InvalidArgument, "unexpected fact snapshot schema", "schema", {}, "emergence.fact"));
298 if (version == nullptr || !version->isInt64() || version->asInt() != 1)
299 return eve::Result<void>::failure(eve::Diagnostic::error(
300 eve::DiagnosticCode::UnknownVersion, "unsupported fact snapshot version", "version", {}, "emergence.fact"));
301
302 FactStore next;
303 auto loadObject = [&](const char* fieldName, auto&& setter) -> eve::Result<void> {
304 const auto it = object->find(fieldName);
305 if (it == object->end()) return eve::Result<void>::success();
306 const auto* map = it->second.getIf<eve::Value::Object>();
307 if (map == nullptr)
308 return eve::Result<void>::failure(eve::Diagnostic::error(
309 eve::DiagnosticCode::InvalidArgument, "fact field must be an object", fieldName, {}, "emergence.fact"));
310 for (const auto& [key, value] : *map) {
311 if (!setter(next, key, value))
312 return eve::Result<void>::failure(eve::Diagnostic::error(
313 eve::DiagnosticCode::InvalidArgument, "invalid fact entry", fieldName, {}, "emergence.fact"));
314 }
315 return eve::Result<void>::success();
316 };
317
318 if (auto r = loadObject("values",
319 [](FactStore& store, const std::string& key, const eve::Value& value) {
320 (void)store.setValue(key, value);
321 return true;
322 });
323 !r)
324 return r;
325 if (auto r = loadObject("tags",
326 [](FactStore& store, const std::string& key, const eve::Value& value) {
327 if (!value.isBool()) return false;
328 (void)store.setTag(key, value.asBool());
329 return true;
330 });
331 !r)
332 return r;
333 if (auto r = loadObject("attributes",
334 [](FactStore& store, const std::string& key, const eve::Value& value) {
335 (void)store.setAttribute(key, value);
336 return true;
337 });
338 !r)
339 return r;
340 if (auto r = loadObject("resources",
341 [](FactStore& store, const std::string& key, const eve::Value& value) {
342 (void)store.setResource(key, value);
343 return true;
344 });
345 !r)
346 return r;
347 if (auto r = loadObject("states",
348 [](FactStore& store, const std::string& key, const eve::Value& value) {
349 (void)store.setState(key, value);
350 return true;
351 });
352 !r)
353 return r;
354 if (auto r = loadObject("authorities",
355 [](FactStore& store, const std::string& key, const eve::Value& value) {
356 if (!value.isBool()) return false;
357 (void)store.setAuthority(key, value.asBool());
358 return true;
359 });
360 !r)
361 return r;
362 if (auto r = loadObject("policies",
363 [](FactStore& store, const std::string& key, const eve::Value& value) {
364 auto decoded = decodePolicyResult(value);
365 if (!decoded) return false;
366 (void)store.setPolicy(key, std::move(decoded).takeValue());
367 return true;
368 });
369 !r)
370 return r;
371
372 *this = std::move(next);
373 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
374}
