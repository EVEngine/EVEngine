67eve::Result<condition::ConditionResult> decodePolicyResult(const eve::Value& value) {
68 const auto* object = value.getIf<eve::Value::Object>();
69 if (object == nullptr)
70 return eve::Result<condition::ConditionResult>::failure(eve::Diagnostic::error(
71 eve::DiagnosticCode::InvalidArgument, "policy result must be an object", {}, {}, "emergence.fact"));
72 if (!exactFields(*object, {"passed", "reason", "evidence", "details"}))
73 return eve::Result<condition::ConditionResult>::failure(eve::Diagnostic::error(
74 eve::DiagnosticCode::InvalidArgument, "policy result contains an unknown field", {}, {}, "emergence.fact"));
75 const auto* passed = field(*object, "passed");
76 const auto* reason = field(*object, "reason");
77 if (passed == nullptr || !passed->isBool())
78 return eve::Result<condition::ConditionResult>::failure(eve::Diagnostic::error(
79 eve::DiagnosticCode::InvalidArgument, "policy passed must be a boolean", "passed", {}, "emergence.fact"));
80 if (reason == nullptr || !reason->isString())
81 return eve::Result<condition::ConditionResult>::failure(eve::Diagnostic::error(
82 eve::DiagnosticCode::InvalidArgument, "policy reason must be a string", "reason", {}, "emergence.fact"));
83 auto code = parseReasonCode(reason->asString());
84 if (!code) return eve::Result<condition::ConditionResult>::failure(code.status());
85 eve::Value evidence;
86 eve::Value details = eve::Value::Object{};
87 if (const auto* evidenceValue = field(*object, "evidence")) evidence = *evidenceValue;
88 if (const auto* detailsValue = field(*object, "details")) details = *detailsValue;
89 if (passed->asBool())
90 return eve::Result<condition::ConditionResult>::success(
91 condition::ConditionResult::success(std::move(evidence), std::move(details)));
92 return eve::Result<condition::ConditionResult>::success(
93 condition::ConditionResult::failed(code.value(), std::move(evidence), std::move(details)));
94}

114eve::Result<std::pair<FactDomain, std::string>> parseFactKey(std::string_view key) {
115 const auto fail = [](const char* message) {
116 return eve::Result<std::pair<FactDomain, std::string>>::failure(
117 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, message, "key", {}, "emergence.fact"));
118 };
119 if (key.empty()) return fail("fact key must be non-empty");
120 const auto slash = key.find(':');
121 if (slash == std::string_view::npos || slash == 0 || slash + 1 >= key.size())
122 return fail("fact key must use domain:name form");
123 const auto domainText = key.substr(0, slash);
124 const auto name = key.substr(slash + 1);
125 FactDomain domain = FactDomain::Value;
126 if (domainText == "value")
127 domain = FactDomain::Value;
128 else if (domainText == "tag")
129 domain = FactDomain::Tag;
130 else if (domainText == "attr")
131 domain = FactDomain::Attribute;
132 else if (domainText == "resource")
133 domain = FactDomain::Resource;
134 else if (domainText == "state")
135 domain = FactDomain::State;
136 else if (domainText == "authority")
137 domain = FactDomain::Authority;
138 else if (domainText == "policy")
139 domain = FactDomain::Policy;
140 else
141 return fail("unknown fact domain prefix");
142 return eve::Result<std::pair<FactDomain, std::string>>::success(std::make_pair(domain, std::string(name)));
143}

280eve::Result<void> FactStore::restoreJson(std::string_view json) {
281 auto parsed = eve::Value::fromJson(json);
282 if (!parsed) return eve::Result<void>::failure(parsed.status());
283 const auto* object = parsed.value().getIf<eve::Value::Object>();
284 if (object == nullptr)
285 return eve::Result<void>::failure(eve::Diagnostic::error(
286 eve::DiagnosticCode::InvalidArgument, "fact snapshot must be an object", {}, {}, "emergence.fact"));
287 if (!exactFields(*object, {"schema", "version", "values", "tags", "attributes", "resources", "states",
288 "authorities", "policies"}))
289 return eve::Result<void>::failure(eve::Diagnostic::error(
290 eve::DiagnosticCode::InvalidArgument, "fact snapshot contains an unknown field", {}, {}, "emergence.fact"));
291 const auto* schema = field(*object, "schema");
292 const auto* version = field(*object, "version");
293 if (schema == nullptr || !schema->isString() || schema->asString() != "eve.emergence.facts")
294 return eve::Result<void>::failure(eve::Diagnostic::error(
295 eve::DiagnosticCode::InvalidArgument, "unexpected fact snapshot schema", "schema", {}, "emergence.fact"));
296 if (version == nullptr || !version->isInt64() || version->asInt() != 1)
297 return eve::Result<void>::failure(eve::Diagnostic::error(
298 eve::DiagnosticCode::UnknownVersion, "unsupported fact snapshot version", "version", {}, "emergence.fact"));
299
300 FactStore next;
301 auto loadObject = [&](const char* fieldName, auto&& setter) -> eve::Result<void> {
302 const auto it = object->find(fieldName);
303 if (it == object->end()) return eve::Result<void>::success();
304 const auto* map = it->second.getIf<eve::Value::Object>();
305 if (map == nullptr)
306 return eve::Result<void>::failure(eve::Diagnostic::error(
307 eve::DiagnosticCode::InvalidArgument, "fact field must be an object", fieldName, {}, "emergence.fact"));
308 for (const auto& [key, value] : *map) {
309 if (!setter(next, key, value))
310 return eve::Result<void>::failure(eve::Diagnostic::error(
311 eve::DiagnosticCode::InvalidArgument, "invalid fact entry", fieldName, {}, "emergence.fact"));
312 }
313 return eve::Result<void>::success();
314 };
315
316 if (auto r = loadObject("values",
317 [](FactStore& store, const std::string& key, const eve::Value& value) {
318 (void)store.setValue(key, value);
319 return true;
320 });
321 !r)
322 return r;
323 if (auto r = loadObject("tags",
324 [](FactStore& store, const std::string& key, const eve::Value& value) {
325 if (!value.isBool()) return false;
326 (void)store.setTag(key, value.asBool());
327 return true;
328 });
329 !r)
330 return r;
331 if (auto r = loadObject("attributes",
332 [](FactStore& store, const std::string& key, const eve::Value& value) {
333 (void)store.setAttribute(key, value);
334 return true;
335 });
336 !r)
337 return r;
338 if (auto r = loadObject("resources",
339 [](FactStore& store, const std::string& key, const eve::Value& value) {
340 (void)store.setResource(key, value);
341 return true;
342 });
343 !r)
344 return r;
345 if (auto r = loadObject("states",
346 [](FactStore& store, const std::string& key, const eve::Value& value) {
347 (void)store.setState(key, value);
348 return true;
349 });
350 !r)
351 return r;
352 if (auto r = loadObject("authorities",
353 [](FactStore& store, const std::string& key, const eve::Value& value) {
354 if (!value.isBool()) return false;
355 (void)store.setAuthority(key, value.asBool());
356 return true;
357 });
358 !r)
359 return r;
360 if (auto r = loadObject("policies",
361 [](FactStore& store, const std::string& key, const eve::Value& value) {
362 auto decoded = decodePolicyResult(value);
363 if (!decoded) return false;
364 (void)store.setPolicy(key, std::move(decoded).takeValue());
365 return true;
366 });
367 !r)
368 return r;
369
370 *this = std::move(next);
371 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
372}
