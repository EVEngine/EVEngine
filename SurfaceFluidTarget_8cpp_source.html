94EditorResult<SurfaceFluidSettings> parse(const EditorValue& value) {
95 SurfaceFluidSettings s;
96 const auto number = [&](const char* key, double& destination) {
97 const auto* entry = field(value, key);
98 const auto* v = entry ? entry->getIf<double>() : nullptr;
99 if (v) destination = *v;
100 return v != nullptr;
101 };
102 const auto* gravityValue = field(value, "gravity");
103 const auto* gravity = gravityValue ? gravityValue->getIf<EditorValue::Array>() : nullptr;
104 const auto* crossingsValue = field(value, "maxCrossings");
105 const auto* crossings = crossingsValue ? crossingsValue->getIf<int64_t>() : nullptr;
106 if (!gravity || gravity->size() != 3 || !crossings)
107 return fail<SurfaceFluidSettings>(EditorStatus::Rejected, "editor.surface-fluid.fields",
108 "Surface fluid settings are incomplete");
109 double* gravityOut[]{&s.gravityX, &s.gravityY, &s.gravityZ};
110 for (int i = 0; i < 3; ++i) {
111 const auto* v = (*gravity)[i].getIf<double>();
112 if (!v)
113 return fail<SurfaceFluidSettings>(EditorStatus::Rejected, "editor.surface-fluid.gravity",
114 "Gravity must be a Vec3");
115 *gravityOut[i] = *v;
116 }
117 s.maxCrossings = static_cast<int>(*crossings);
118 bool complete = true;
119 complete &= number("friction", s.friction);
120 complete &= number("maxSpeed", s.maxSpeed);
121 complete &= number("adhesionAcceleration", s.adhesionAcceleration);
122 complete &= number("contactAngleDegrees", s.contactAngleDegrees);
123 complete &= number("mergeRadiusScale", s.mergeRadiusScale);
124 complete &= number("trailDeposition", s.trailDeposition);
125 complete &= number("airDrag", s.airDrag);
126 complete &= number("reattachDistance", s.reattachDistance);
127 complete &= number("diffusion", s.diffusion);
128 complete &= number("evaporation", s.evaporation);
129 complete &= number("maxWetness", s.maxWetness);
130 complete &= number("velocityStretch", s.velocityStretch);
131 complete &= number("maxAspectRatio", s.maxAspectRatio);
132 complete &= number("surfaceOffset", s.surfaceOffset);
133 complete &= number("dryRoughness", s.dryRoughness);
134 complete &= number("wetRoughness", s.wetRoughness);
135 complete &= number("drySpecular", s.drySpecular);
136 complete &= number("wetSpecular", s.wetSpecular);
137 complete &= number("wetDarkening", s.wetDarkening);
138 complete &= number("normalStrength", s.normalStrength);
139 if (!complete)
140 return fail<SurfaceFluidSettings>(EditorStatus::Rejected, "editor.surface-fluid.fields",
141 "Surface fluid settings are incomplete");
142 const auto diagnostics = validateSettings(s);
143 if (std::any_of(diagnostics.begin(), diagnostics.end(),
144 [](const auto& d) { return d.severity() == DiagnosticSeverity::Error; }))
145 return fail<SurfaceFluidSettings>(EditorStatus::Rejected, "editor.surface-fluid.invalid",
146 "Surface fluid settings are invalid");
147 return eve::editing::applied<SurfaceFluidSettings>(s);
148}

219EditorResult<DomainOperation> SurfaceFluidTarget::makeSet(const SelectionSnapshot& s, const PropertyPath& p,
220 const EditorValue& v, PropertySetMode m) const {
221 if (m == PropertySetMode::Reset) return makeReset(s, p);
222 auto d = schema(s).find(p);
223 if (!matches(s) || !d || m != PropertySetMode::Absolute)
224 return fail<DomainOperation>(EditorStatus::Rejected, "editor.surface-fluid.set",
225 "Surface fluid property requires a matching absolute edit");
226 auto checked = validatePropertyValue(*d, v);
227 if (!checked.ok()) return EditorResult<DomainOperation>::failure(checked.status());
228 EditorValue candidate = settingsValue(settings_);
229 (*candidate.getIf<EditorValue::Object>())[p.value()] = v;
230 auto parsed = parse(candidate);
231 if (!parsed.ok()) return EditorResult<DomainOperation>::failure(parsed.status());
232 DomainOperation op;
233 op.type = "surface-fluid.settings.replace.v1";
234 op.inverseType = op.type;
235 op.target = TargetId(id_);
236 op.payload = settingsValue(parsed.value());
237 op.inverse = settingsValue(settings_);
238 op.hasInverse = true;
239 op.affectedProperties.push_back(p.value());
240 op.mergeKey = "surface-fluid:" + id_ + ":" + p.value();
241 return eve::editing::applied<DomainOperation>(std::move(op));
242}
