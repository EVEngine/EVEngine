60 Impl() = default;
61 Impl(const Impl& other) : settings(other.settings), last(other.last), history(other.history), cursor(other.cursor) {
62 tiles.reserve(other.tiles.size());
63 for (const auto& source : other.tiles) {
64 Tile tile;
65 tile.name = source.name; tile.heights = source.heights; tile.splat = source.splat;
66 tile.detail = source.detail;
67 tile.trees = source.trees; tile.objects = source.objects; tile.probes = source.probes;
68 tile.originX = source.originX; tile.originZ = source.originZ;
69 tiles.push_back(std::move(tile));
70 }
71 }
72 Snapshot snapshot() const {
73 Snapshot result;
74 for (const auto& tile : tiles) {
75 result.heights.push_back(tile.heights); result.splats.push_back(tile.splat);
76 result.details.push_back(tile.detail);
77 result.trees.push_back(tile.trees); result.objects.push_back(tile.objects);
78 result.probes.push_back(tile.probes);
79 }
80 result.report = last;
81 return result;
82 }
83 void restore(const Snapshot& value) {
84 for (std::size_t i = 0; i < tiles.size(); ++i) {
85 tiles[i].heights = value.heights[i]; tiles[i].splat = value.splats[i];
86 tiles[i].detail = value.details[i];
87 tiles[i].trees = value.trees[i]; tiles[i].objects = value.objects[i];
88 tiles[i].probes = value.probes[i];
89 }
90 last = value.report;
91 }
92 void record() {
93 history.resize(static_cast<std::size_t>(cursor + 1));
94 history.push_back(snapshot());
95 ++cursor;
96 }

104Result<int> TerrainWorldWorkspace::create(const TerrainWorldCreationSettings& s) {
105 if (!impl_ || impl_->runStatus == TerrainWorldRunStatus::Pending || s.tilesX <= 0 || s.tilesZ <= 0 ||
106 s.tilesX > 4096 / s.tilesZ || !finitePositive(s.tileSize) ||
107 !finitePositive(s.tileHeight) || !std::isfinite(s.centerX) || !std::isfinite(s.centerZ) ||
108 s.heightmapResolution < 2 || !powerOfTwo(s.heightmapResolution - 1) ||
109 !powerOfTwo(s.controlTextureResolution) || !powerOfTwo(s.detailResolution) ||
110 s.treeResolution <= 0 || s.objectResolution <= 0 || s.splatLayers <= 0 ||
111 s.defaultSplatLayer < 0 || s.defaultSplatLayer >= s.splatLayers || s.defaultDetailDensity < 0 ||
112 s.namePrefix.empty())
113 return invalidWorld<int>("terrain.world.create: valid Pcg tile topology and resolutions required");
114 auto candidate = std::make_unique<Impl>();
115 candidate->settings = s;
116 candidate->tiles.reserve(static_cast<std::size_t>(s.tilesX * s.tilesZ));
117 const double baseX = s.centerX - s.tileSize * s.tilesX * 0.5;
118 const double baseZ = s.centerZ - s.tileSize * s.tilesZ * 0.5;
119 if (!std::isfinite(baseX) || !std::isfinite(baseZ))
120 return invalidWorld<int>("terrain.world.create: world origin is not representable");
121 for (int x = 0; x < s.tilesX; ++x) for (int z = 0; z < s.tilesZ; ++z) {
122 Impl::Tile tile;
123 tile.name = (s.worldMap ? "World Map" : s.namePrefix) + "_" + std::to_string(x) + "_" +
124 std::to_string(z) + (s.nameSuffix.empty() ? "" : "-" + s.nameSuffix);
125 tile.originX = baseX + s.tileSize * x; tile.originZ = baseZ + s.tileSize * z;
126 if (!std::isfinite(tile.originX) || !std::isfinite(tile.originZ))
127 return invalidWorld<int>("terrain.world.create: tile origin is not representable");
128 tile.heights = Heightmap(s.heightmapResolution, s.heightmapResolution);
129 auto splat = tile.splat.initialize(s.controlTextureResolution, s.controlTextureResolution,
130 s.splatLayers, s.defaultSplatLayer);
131 if (!splat.ok()) return Result<int>::failure(splat.status());
132 auto detail = tile.detail.reset(s.detailResolution, s.detailResolution, s.defaultDetailDensity);
133 if (!detail.ok()) return Result<int>::failure(detail.status());
134 candidate->tiles.push_back(std::move(tile));
135 }
136 candidate->history.push_back(candidate->snapshot());
137 impl_.swap(candidate);
138 return Result<int>::success(static_cast<int>(impl_->tiles.size()));
139}

143Result<int> mutateWorld(std::unique_ptr<TerrainWorldWorkspace::Impl>& owner, Apply&& apply) {
144 if (!owner || owner->tiles.empty() || owner->runStatus == TerrainWorldRunStatus::Pending)
145 return invalidWorld<int>("terrain.world: created workspace without pending spawn required");
146 auto candidate = std::make_unique<TerrainWorldWorkspace::Impl>(*owner);

151 candidate->record();
152 owner.swap(candidate);

156Result<TerrainMultiTileReport> applySpawnRule(TerrainWorldWorkspace::Impl& world,
157 const TerrainSpawnRule& variant) {
158 return std::visit(
159 [&](const auto& rule) -> Result<TerrainMultiTileReport> {
160 using Rule = std::decay_t<decltype(rule)>;
161 if (!rule.enabled) return Result<TerrainMultiTileReport>::success({});
162 if constexpr (std::is_same_v<Rule, TerrainSplatSpawnRule>) {
163 std::vector<TerrainSplatTile> tiles;
164 for (auto& tile : world.tiles)
165 tiles.push_back({tile.name, &tile.splat, tile.originX, tile.originZ,
166 world.settings.tileSize, world.settings.tileSize, world.settings.worldMap});
167 return paintTerrainSplatLayerMultiTile(tiles, rule.paint, rule.targetLayer, rule.operation,
168 world.settings.worldMap);
169 } else if constexpr (std::is_same_v<Rule, TerrainDetailSpawnRule>) {
170 auto settings = rule.settings;
171 settings.mode = rule.mode;
172 std::vector<TerrainDetailTile> tiles;
173 for (auto& tile : world.tiles)
174 tiles.push_back({tile.name, &tile.detail, tile.originX, tile.originZ,
175 world.settings.tileSize, world.settings.tileSize, world.settings.worldMap});
176 return applyTerrainDetailMultiTile(tiles, rule.fitness, settings, rule.operation, rule.seed,
177 world.settings.worldMap);
178 } else if constexpr (std::is_same_v<Rule, TerrainTreeSpawnRule>) {
179 std::vector<TerrainTreeTile> tiles;
180 for (auto& tile : world.tiles)
181 tiles.push_back({tile.name, &tile.trees, &tile.heights, tile.originX, tile.originZ,
182 world.settings.tileSize, world.settings.tileSize,
183 world.settings.treeResolution, world.settings.treeResolution,
184 world.settings.worldMap});
185 return applyTerrainTreesMultiTile(tiles, rule.fitness, rule.settings, rule.operation, rule.mode,
186 world.settings.worldMap);
187 } else if constexpr (std::is_same_v<Rule, TerrainObjectSpawnRule>) {
188 std::vector<TerrainObjectTile> tiles;
189 for (auto& tile : world.tiles)
190 tiles.push_back({tile.name, &tile.objects, &tile.heights, tile.originX, tile.originZ,
191 world.settings.tileSize, world.settings.tileSize,
192 world.settings.objectResolution, world.settings.objectResolution,
193 world.settings.worldMap});
194 return applyTerrainObjectsMultiTile(tiles, rule.fitness, rule.settings, rule.operation, rule.mode,
195 world.settings.worldMap);
196 } else if constexpr (std::is_same_v<Rule, TerrainProbeSpawnRule>) {
197 std::vector<TerrainProbeTile> tiles;
198 for (auto& tile : world.tiles)
199 tiles.push_back({tile.name, &tile.probes, &tile.heights, tile.originX, tile.originZ,
200 world.settings.tileSize, world.settings.tileSize,
201 world.settings.objectResolution, world.settings.objectResolution,
202 world.settings.worldMap});
203 return applyTerrainProbesMultiTile(tiles, rule.fitness, rule.settings, rule.operation, rule.mode,
204 world.settings.worldMap);
205 } else {
206 std::vector<TerrainHeightTile> tiles;
207 for (auto& tile : world.tiles)
208 tiles.push_back({tile.name, &tile.heights, tile.originX, tile.originZ,
209 world.settings.tileSize, world.settings.tileSize, world.settings.worldMap});
210 return applyTerrainStampMultiTile(tiles, rule.stamp, rule.operation, rule.localMask,
211 rule.globalMask, world.settings.worldMap);
212 }
213 },
214 variant);
215}

291Result<int> TerrainWorldWorkspace::spawn(const TerrainSpawnPlan& plan) {
292 if (plan.rules().empty()) return invalidWorld<int>("terrain.world.spawn: nonempty spawn plan required");
293 bool hasEnabled = false;
294 for (const auto& rule : plan.rules())
295 std::visit([&](const auto& value) { hasEnabled = hasEnabled || value.enabled; }, rule);
296 if (!hasEnabled) return invalidWorld<int>("terrain.world.spawn: at least one enabled rule required");
297 return mutateWorld(impl_, [&](Impl& world) -> Result<TerrainMultiTileReport> {
298 TerrainMultiTileReport combined;
299 std::unordered_set<std::string> affectedNames;
300 for (const auto& variant : plan.rules()) {
301 auto applied = applySpawnRule(world, variant);
302 if (!applied.ok()) return Result<TerrainMultiTileReport>::failure(applied.status());
303 auto report = std::move(applied.value());
304 if (report.changedSamples > std::numeric_limits<int>::max() - combined.changedSamples)
305 return Result<TerrainMultiTileReport>::failure(Diagnostic::error(
306 DiagnosticCode::InvalidArgument, "terrain.world.spawn: changed count exceeds integer range"));
307 combined.changedSamples += report.changedSamples;
308 for (const auto& mapping : report.mappings) affectedNames.insert(mapping.terrainName);
309 combined.mappings.insert(combined.mappings.end(), report.mappings.begin(), report.mappings.end());
310 }
311 combined.affectedTiles = static_cast<int>(affectedNames.size());
312 return Result<TerrainMultiTileReport>::success(std::move(combined));
313 });
314}

359Result<TerrainWorldRunStatus> TerrainWorldWorkspace::stepSpawn(int maxRules) {
360 if (!impl_ || impl_->runStatus != TerrainWorldRunStatus::Pending || !impl_->runCandidate || maxRules <= 0)
361 return Result<TerrainWorldRunStatus>::failure(Diagnostic::error(
362 DiagnosticCode::InvalidArgument,
363 "terrain.world.stepSpawn: pending spawn and positive rule budget required"));
364 int visited = 0;
365 while (impl_->runNext < impl_->runPlan.rules().size() && visited < maxRules) {
366 auto applied = applySpawnRule(*impl_->runCandidate, impl_->runPlan.rules()[impl_->runNext]);
367 if (!applied.ok()) {
368 impl_->runCandidate.reset();
369 impl_->runStatus = TerrainWorldRunStatus::Failed;
370 return Result<TerrainWorldRunStatus>::failure(applied.status());
371 }
372 const auto& report = applied.value();
373 if (report.changedSamples > std::numeric_limits<int>::max() - impl_->runReport.changedSamples) {
374 impl_->runCandidate.reset();
375 impl_->runStatus = TerrainWorldRunStatus::Failed;
376 return Result<TerrainWorldRunStatus>::failure(Diagnostic::error(
377 DiagnosticCode::InvalidArgument, "terrain.world.stepSpawn: changed count exceeds integer range"));
378 }
379 impl_->runReport.changedSamples += report.changedSamples;
380 impl_->runReport.mappings.insert(impl_->runReport.mappings.end(), report.mappings.begin(),
381 report.mappings.end());
382 ++impl_->runNext;
383 ++visited;
384 }
385 if (impl_->runNext == impl_->runPlan.rules().size()) {
386 std::unordered_set<std::string> affected;
387 for (const auto& mapping : impl_->runReport.mappings) affected.insert(mapping.terrainName);
388 impl_->runReport.affectedTiles = static_cast<int>(affected.size());
389 auto completed = std::move(impl_->runCandidate);
390 completed->last = std::move(impl_->runReport);
391 completed->record();
392 completed->runStatus = TerrainWorldRunStatus::Completed;
393 completed->runNext = impl_->runNext;
394 impl_.swap(completed);
395 }
396 return Result<TerrainWorldRunStatus>::success(impl_->runStatus);
397}

456Result<int> TerrainWorldWorkspace::clearSpawns(const TerrainWorldClearSettings& settings) {
457 if (!settings.details && !settings.trees && !settings.objects && !settings.probes)
458 return invalidWorld<int>("terrain.world.clearSpawns: at least one domain must be selected");
459 return mutateWorld(impl_, [&](Impl& world) {
460 TerrainMultiTileReport report;
461 const std::string source = std::to_string(settings.sourceNamespace);
462 auto clearPoints = [&](PointSet& points) -> Result<int> {
463 if (settings.sourceNamespace == 0) {
464 const int changed = points.getCount();
465 points.clear();
466 return Result<int>::success(changed);
467 }
468 PointSet retained;
469 retained.reserve(static_cast<std::size_t>(points.getCount()));
470 int changed = 0;
471 for (int i = 0; i < points.getCount(); ++i) {
472 if (points.getStringAttribute(i, "spawnNamespace", "") == source) {

491 } else {

493 if (!cleared.ok()) return Result<TerrainMultiTileReport>::failure(cleared.status());
494 tileChanges += cleared.value();
495 }
496 }
497 if (settings.trees) {

499 if (!cleared.ok()) return Result<TerrainMultiTileReport>::failure(cleared.status());

501 }

505 tileChanges += cleared.value();
506 }
507 if (settings.probes) {
508 auto cleared = clearPoints(tile.probes);
509 if (!cleared.ok()) return Result<TerrainMultiTileReport>::failure(cleared.status());
510 tileChanges += cleared.value();
511 }
512 report.changedSamples += tileChanges;
513 report.affectedTiles += tileChanges > 0;
514 }
515 return Result<TerrainMultiTileReport>::success(report);
516 });
