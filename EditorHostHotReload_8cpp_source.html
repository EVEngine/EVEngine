62std::string normalizedRelativePath(const std::string& root, const std::string& input, std::filesystem::path* absolute) {
63 if (input.empty()) return {};
64 std::error_code ec;
65 const std::filesystem::path canonicalRoot = std::filesystem::weakly_canonical(root, ec);
66 if (ec) return {};
67 std::filesystem::path candidate(input);
68 if (!candidate.is_absolute()) candidate = canonicalRoot / candidate;
69 candidate = std::filesystem::weakly_canonical(candidate, ec);
70 if (ec) return {};
71
72 std::filesystem::path relative = std::filesystem::relative(candidate, canonicalRoot, ec);
73 if (ec || relative.empty()) return {};
74 for (const auto& part : relative) {
75 if (part == "..") return {};
76 }
77 if (absolute) *absolute = candidate;
78 std::string path = relative.generic_string();
79 while (path.rfind("./", 0) == 0) path.erase(0, 2);
80 return path;
81}

155std::string EditorHost::reloadResource(const std::string& input) {
156 if (!running()) return "error: host not started";
157 std::filesystem::path absolute;
158 const std::string path = normalizedRelativePath(gameRoot(), input, &absolute);
159 if (path.empty()) return "error: resource path is outside project root";
160 if (!std::filesystem::is_regular_file(absolute)) {
161 const std::string result = "error: resource not found: " + path;
162 recordReload(path, result);
163 return result;
164 }
165
166 const std::string source = readText(absolute);
167 if (source.empty()) {
168 const std::string result = "error: resource is empty: " + path;
169 recordReload(path, result);
170 return result;
171 }
172
173 std::string result;
174 const std::string editorSuffix = ".editor.json";
175 const std::string vmSuffix = ".vm.nut";
176 const std::string editorId = editorIdFromPath(path, editorSuffix);
177 const std::string vmEditorId = editorIdFromPath(path, vmSuffix);
178 if (!editorId.empty()) {
179 const std::string sourceId = editorIdFromSource(source);
180 if (sourceId.empty()) {
181 result = "error: editor resource requires a valid id";
182 } else if (sourceId != editorId) {
183 result = "error: editor id must match resource filename: " + editorId;
184 } else {
185 const auto values = editorValues(*this, editorId);
186 result = applyEditor(source);
187 if (result.rfind("error:", 0) != 0) restoreEditorValues(*this, editorId, values);
188 }
189 } else if (!vmEditorId.empty()) {
190 const std::string vmName = editorVmName(*this, vmEditorId);
191 if (vmName.empty()) {
192 result = "error: editor ViewModel not found for: " + vmEditorId;
193 } else {
194 const auto values = editorValues(*this, vmEditorId);
195 result = registerVM(vmName, source);
196 if (result.rfind("error:", 0) != 0) restoreEditorValues(*this, vmEditorId, values);
197 }
198 } else if (path == "mcp.nut" || (path.rfind("mcp/", 0) == 0 && endsWith(path, ".nut"))) {
199 result = runScript(source);
200 } else {
201 result = "error: unsupported MCP host resource: " + path;
202 }
203 recordReload(path, result);
204 return result;
205}
