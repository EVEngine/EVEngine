229Var stackVar(HSQUIRRELVM vm, SQInteger index) {
230 switch (sq_gettype(vm, index)) {
231 case OT_NULL:
232 return Var();
233 case OT_BOOL: {
234 SQBool b = SQFalse;
235 if (SQ_SUCCEEDED(sq_getbool(vm, index, &b))) return Var(bool(b));
236 return Var();
237 }
238 case OT_INTEGER: {
239 SQInteger i = 0;
240 if (SQ_SUCCEEDED(sq_getinteger(vm, index, &i)))
241 return Var(static_cast<Poco::Int64>(i));
242 return Var();
243 }
244 case OT_FLOAT: {
245 SQFloat f = 0.f;
246 if (SQ_SUCCEEDED(sq_getfloat(vm, index, &f)))
247 return Var(static_cast<double>(f));
248 return Var();
249 }
250 case OT_STRING: {
251 const SQChar* s = nullptr;
252 if (SQ_SUCCEEDED(sq_getstring(vm, index, &s)) && s) return Var(std::string(s));
253 return Var();
254 }
255 case OT_ARRAY: {
256 HSQOBJECT obj;
257 if (SQ_FAILED(sq_getstackobj(vm, index, &obj))) return Var();
258 sq_pushobject(vm, obj);
259 sq_pushnull(vm);
260 Array::Ptr arr = new Array();
261 while (SQ_SUCCEEDED(sq_next(vm, -2))) {
262 arr->add(stackVar(vm, -1));
263 sq_pop(vm, 2);
264 }
265 sq_pop(vm, 1);
266 return Var(arr);
267 }
268 default:
269 return Var();
270 }
271}

273void pushVar(HSQUIRRELVM vm, const Var& v) {
274 if (v.isEmpty()) {
275 sq_pushnull(vm);
276 return;
277 }
278 try {
279 if (v.isBoolean()) {
280 sq_pushbool(vm, v.convert<bool>() ? SQTrue : SQFalse);
281 return;
282 }
283 if (v.isNumeric()) {
284 sq_pushfloat(vm, static_cast<SQFloat>(v.convert<double>()));
285 return;
286 }
287 if (v.isString()) {
288 const std::string s = v.convert<std::string>();
289 sq_pushstring(vm, s.c_str(), static_cast<SQInteger>(s.size()));
290 return;
291 }
292 if (v.isArray()) {
293 Array::Ptr arr;
294 try {
295 arr = v.extract<Array::Ptr>();
296 } catch (...) {
297 sq_pushnull(vm);
298 return;
299 }
300 sq_newarray(vm, 0);
301 if (arr) {
302 for (size_t i = 0; i < arr->size(); ++i) {
303 pushVar(vm, arr->get(static_cast<unsigned int>(i)));
304 if (SQ_FAILED(sq_arrayappend(vm, -2))) break;
305 }
306 }
307 return;
308 }
309 } catch (...) {
310 }
311 sq_pushnull(vm);
312}

316 const SQInteger top = sq_gettop(vm);

318 name, SQTrue))) {

320 return "compile failed";
321 }

324 sq_settop(vm, top);
325 return "runtime failed";
326 }
327 sq_settop(vm, top);
328 return {};

331void callRootFunc(HSQUIRRELVM vm, const std::string& name, const std::vector<Var>& args) {
332 const SQInteger top = sq_gettop(vm);
333 sq_pushroottable(vm);
334 sq_pushstring(vm, name.c_str(), static_cast<SQInteger>(name.size()));
335 if (SQ_FAILED(sq_get(vm, -2))) {
336 sq_settop(vm, top);
337 return; // hook not defined
338 }
339 if (sq_gettype(vm, -1) != OT_CLOSURE) {
340 sq_settop(vm, top);
341 return;
342 }
343 sq_pushroottable(vm);
344 for (const auto& a : args) pushVar(vm, a);
345 if (SQ_FAILED(sq_call(vm, static_cast<SQInteger>(args.size()) + 1, SQFalse, SQTrue)))
346 fprintf(stderr, "eve.host: %s() failed\n", name.c_str());
347 sq_settop(vm, top);
348}

350bool callTableFunc(HSQUIRRELVM vm, const ssq::Table& tbl, const std::string& name,
351 const std::vector<Var>& args, std::string* err) {
352 if (name.empty()) return true;
353 const SQInteger top = sq_gettop(vm);
354 try {
355 ssq::Function fn = tbl.findFunc(name.c_str());
356 sq_pushobject(vm, fn.getRaw());
357 sq_pushobject(vm, tbl.getRaw());
358 for (const auto& a : args) pushVar(vm, a);
359 if (SQ_FAILED(sq_call(vm, static_cast<SQInteger>(args.size()) + 1, SQFalse, SQTrue))) {
360 if (err) *err = "runtime failed: " + name;
361 sq_settop(vm, top);
362 return false;
363 }
364 sq_settop(vm, top);
365 return true;
366 } catch (const ssq::NotFoundException&) {
367 sq_settop(vm, top);
368 return true; // optional callback / command absent
369 } catch (const std::exception& e) {
370 if (err) *err = e.what();
371 sq_settop(vm, top);
372 return false;
373 }

377bool splitBindPath(const std::string& bind, std::string& slot, std::string& nested) {
378 if (bind.rfind("vm.", 0) != 0) return false;
379 std::string rest = bind.substr(3);
380 const size_t dot = rest.find('.');
381 if (dot == std::string::npos) {
382 slot = rest;
383 nested.clear();
384 } else if (rest.find('.', dot + 1) == std::string::npos) {
385 slot = rest.substr(0, dot);
386 nested = rest.substr(dot + 1);
387 } else {
388 return false; // only one nesting level supported in v1
389 }
390 return !slot.empty();
391}

404 std::string slot, nested;
405 if (ok) *ok = false;

414 return objectToVar(vm, o);

421 std::string slot, nested;

437 } catch (...) {
438 }

571std::string EditorHost::openWindow(const std::string& title, int width, int height) {
572 if (!impl_) return "error: host not started";
573 auto& I = *impl_;
574 if (!I.allowWindow) return "error: window creation disabled";
575 try {
576 if (!I.win) I.win = ModuleManager::requireInstance<eve::window::Window>("Window");
577 if (!I.gfx) I.gfx = ModuleManager::requireInstance<eve::graphics::Graphics>("Graphics");
578 eve::window::WindowSettings s;
579 s.width = static_cast<uint16_t>(width > 0 ? width : 1280);
580 s.height = static_cast<uint16_t>(height > 0 ? height : 800);
581 s.centered = true;
582 s.resizable = true;
583 if (!I.win->setWindowSettings(s)) return "error: setWindowSettings failed";
584 I.win->setWindowTitle(title.empty() ? "EVEngine AI Host" : title);
585 if (!I.ui) I.ui = ModuleManager::requireInstance<eve::ui::UI>("UI");
586 I.windowOpen = true;
587 I.windowTitle = title;
588 // Keep screen readback on while the host window is open: the AI's whole
589 // feedback loop is "render -> capture", so every presented frame should
590 // be available to eve_host_capture immediately (the Vulkan backend
591 // installs the present-copy hook as soon as readback is enabled).
592 if (I.gfx) I.gfx->setScreenReadbackEnabled(true);
593 return "ok";
594 } catch (const std::exception& e) {
595 return std::string("error: ") + e.what();
596 }
597}
