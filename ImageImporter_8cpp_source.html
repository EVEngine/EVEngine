20Result<ImageInfo> inspectImage(std::span<const std::uint8_t> bytes) {
21 static constexpr std::uint8_t png[] = {0x89, 'P', 'N', 'G', 0x0d, 0x0a, 0x1a, 0x0a};
22 if (bytes.size() >= 24 && std::equal(std::begin(png), std::end(png), bytes.begin()) &&
23 std::string_view(reinterpret_cast<const char*>(bytes.data() + 12), 4) == "IHDR") {
24 ImageInfo info{"png", big32(bytes.data() + 16), big32(bytes.data() + 20)};
25 if (info.width == 0 || info.height == 0)
26 return detail::failure<ImageInfo>(DiagnosticCode::ParseError, "PNG dimensions must be positive");
27 return Result<ImageInfo>::success(std::move(info));
28 }
29 if (bytes.size() >= 4 && bytes[0] == 0xff && bytes[1] == 0xd8) {
30 std::size_t cursor = 2;
31 while (cursor + 4 <= bytes.size()) {
32 if (bytes[cursor] != 0xff) return detail::failure<ImageInfo>(DiagnosticCode::ParseError,
33 "JPEG marker stream is malformed");
34 while (cursor < bytes.size() && bytes[cursor] == 0xff) ++cursor;
35 if (cursor >= bytes.size()) break;
36 const std::uint8_t marker = bytes[cursor++];
37 if (marker == 0xd9 || marker == 0xda) break;
38 if (marker >= 0xd0 && marker <= 0xd7) continue;
39 if (cursor + 2 > bytes.size()) break;
40 const std::uint16_t length = (std::uint16_t(bytes[cursor]) << 8) | bytes[cursor + 1];
41 if (length < 2 || length > bytes.size() - cursor) break;
42 const bool startOfFrame = (marker >= 0xc0 && marker <= 0xc3) ||
43 (marker >= 0xc5 && marker <= 0xc7) ||
44 (marker >= 0xc9 && marker <= 0xcb) ||
45 (marker >= 0xcd && marker <= 0xcf);
46 if (startOfFrame) {
47 if (length < 7) break;
48 ImageInfo info{"jpeg", (std::uint32_t(bytes[cursor + 5]) << 8) | bytes[cursor + 6],
49 (std::uint32_t(bytes[cursor + 3]) << 8) | bytes[cursor + 4]};
50 if (info.width == 0 || info.height == 0) break;
51 return Result<ImageInfo>::success(std::move(info));
52 }
53 cursor += length;
54 }
55 return detail::failure<ImageInfo>(DiagnosticCode::ParseError, "JPEG dimensions are missing or malformed");
56 }
57 return detail::failure<ImageInfo>(DiagnosticCode::Unsupported,
58 "image importer v1 supports PNG and JPEG encoded sources");
59}

122 result.entries.push_back({definitionPath, {definitionText.begin(), definitionText.end()}});

127 result.findings.push_back({request.sourceName, "Image." + info.value().encoding,
128 ImportDisposition::Translated,
129 "encoded source and explicit colour semantics retained"});
130 auto report = detail::finalizeImportReport(
131 result, request.package, "image", info.value().encoding,
132 {{"colorSpace", Value(request.colorSpace == ImageColorSpace::Srgb ? "srgb" : "linear")},
133 {"usage", Value(request.usage)}});
134 if (!report) return Result<PreparedAssetImport>::failure(report.status());
135 return Result<PreparedAssetImport>::success(std::move(result));
136}
