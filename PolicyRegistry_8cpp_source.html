88eve::Result<PolicyDescriptor> makeDescriptor(const std::string& domain, const std::string& name, int version,
89 int priority, bool enabled, const std::string& kindName,
90 const std::string& schemaId, const std::string& metadataJson) {
91 if (domain.empty() || name.empty())
92 return failure<PolicyDescriptor>(eve::DiagnosticCode::InvalidArgument,
93 "policy domain and name must not be empty");
94 if (version <= 0)
95 return failure<PolicyDescriptor>(eve::DiagnosticCode::InvalidArgument,
96 "policy schema version must be positive");
97
98 ImplementationKind kind = ImplementationKind::Builtin;
99 if (!parseImplementationKind(kindName, kind))
100 return failure<PolicyDescriptor>(eve::DiagnosticCode::InvalidArgument, "invalid implementation kind");
101
102 auto metadata = eve::Value::fromJson(metadataJson);
103 if (!metadata.ok()) return eve::Result<PolicyDescriptor>::failure(metadata.status());
104 auto metadataValue = std::move(metadata).takeValue();
105 if (!metadataValue.isObject())
106 return failure<PolicyDescriptor>(eve::DiagnosticCode::InvalidArgument, "policy metadata must be a JSON object");
107 auto canonical = metadataValue.toJson();
108 if (!canonical.ok()) return eve::Result<PolicyDescriptor>::failure(canonical.status());
109
110 if (!schemaId.empty() && eve::schema::SchemaRegistry::resolve(schemaId, version)) {
111 const auto errors = eve::schema::SchemaRegistry::validate(schemaId, version, metadataJson);
112 if (!errors.empty()) {
113 const auto& error = errors.front();
114 return eve::Result<PolicyDescriptor>::failure(eve::Diagnostic::error(
115 eve::DiagnosticCode::InvalidArgument, error.message, error.path,
116 eve::DiagnosticDetails{
117 {"schemaId", schemaId}, {"schemaVersion", std::to_string(version)}, {"validationCode", error.code}},
118 "policyregistry.schema"));
119 }
120 }
121
122 PolicyDescriptor descriptor;
123 descriptor.domain = domain;
124 descriptor.name = name;
125 descriptor.version = eve::SchemaVersion(static_cast<std::uint64_t>(version));
126 descriptor.priority = priority;
127 descriptor.enabled = enabled;
128 descriptor.kind = kind;
129 descriptor.schemaId = schemaId;
130 descriptor.metadataJson = std::move(canonical).takeValue();
131 return eve::Result<PolicyDescriptor>::success(std::move(descriptor));
132}

189PolicyEvent PolicyRegistry::projectEvent(const Storage::Event& event) {
190 PolicyEvent result;

202 result.version = event.data.version;
203 result.generation = event.generation;
204 result.enabled = event.data.enabled;

280 if (!current.ok()) {

336const PolicyDescriptor* PolicyRegistry::select(const std::string& domain) const {
337 const PolicyDescriptor* selected = nullptr;
338 for (std::size_t index = 0;; ++index) {
339 const auto* key = storage_.keyAt(index);
340 if (!key) break;
341 if (key->first != domain) continue;
342 const auto* candidate = storage_.at(index);
343 if (candidate && candidate->enabled &&
344 (!selected || candidate->priority > selected->priority ||
345 (candidate->priority == selected->priority && candidate->name < selected->name)))
346 selected = candidate;
347 }
348 return selected;
349}

399std::string PolicyRegistry::snapshotJson() const {
400 const auto state = storage_.snapshotState();
401 eve::Value::Array descriptors;
402 eve::Value::Array events;
403 eve::Value::Array generations;
404 descriptors.reserve(storage_.size());
405 events.reserve(storage_.eventCount());
406 generations.reserve(state.entries.size());
407
408 for (const auto& [key, entry] : state.entries) {
409 eve::Value::Object generation;
410 generation.emplace("domain", eve::Value(key.first));
411 generation.emplace("generation", eve::Value(std::to_string(entry.generation.value())));
412 generation.emplace("name", eve::Value(key.second));
413 generations.emplace_back(std::move(generation));
414
415 if (!entry.value.has_value()) continue;
416 const auto& descriptor = *entry.value;
417 eve::Value::Object item;
418 item.emplace("domain", eve::Value(descriptor.domain));
419 item.emplace("enabled", eve::Value(descriptor.enabled));
420 item.emplace("generation", eve::Value(std::to_string(entry.generation.value())));
421 item.emplace("kind", eve::Value(implementationKindName(descriptor.kind)));
422 item.emplace("metadata", parseStoredJson(descriptor.metadataJson));
423 item.emplace("name", eve::Value(descriptor.name));
424 item.emplace("priority", eve::Value(static_cast<std::int64_t>(descriptor.priority)));
425 item.emplace("schemaId", eve::Value(descriptor.schemaId));
426 item.emplace("version", eve::Value(static_cast<std::int64_t>(descriptor.version.value())));
427 descriptors.emplace_back(std::move(item));
428 }
429
430 for (const auto& raw : state.events) {
431 const auto event = projectEvent(raw);
432 eve::Value::Object item;
433 item.emplace("domain", eve::Value(event.domain));
434 item.emplace("enabled", eve::Value(event.enabled));
435 item.emplace("generation", eve::Value(std::to_string(event.generation.value())));
436 item.emplace("name", eve::Value(event.name));
437 item.emplace("policyName", eve::Value(event.policyName));
438 item.emplace("sequence", eve::Value(std::to_string(event.sequence.value())));
439 item.emplace("version", eve::Value(static_cast<std::int64_t>(event.version.value())));
440 events.emplace_back(std::move(item));
441 }
442
443 eve::Value::Object root;
444 root.emplace("descriptors", eve::Value(std::move(descriptors)));
445 root.emplace("events", eve::Value(std::move(events)));
446 root.emplace("generations", eve::Value(std::move(generations)));
447 root.emplace("nextEventSequence", eve::Value(std::to_string(state.nextEventSequence.value())));
448 root.emplace("version", eve::Value(std::int64_t(1)));
449 auto encoded = eve::Value(std::move(root)).toJson();
450 if (!encoded.ok()) std::terminate();
451 return std::move(encoded).takeValue();
452}

454eve::Result<void> PolicyRegistry::restoreJson(const std::string& json) {
455 auto parsed = eve::Value::fromJson(json);
456 if (!parsed.ok()) {
457 return eve::Result<void>::failure(parsed.status());
458 }
459 const eve::Value value = std::move(parsed).takeValue();
460 const auto* root = value.getIf<eve::Value::Object>();
461 if (!root) {
462 return failure<void>(eve::DiagnosticCode::ParseError, "policy snapshot must be an object");
463 }
464 const auto* snapshotVersion = field(*root, "version");
465 const auto* versionNumber = snapshotVersion ? snapshotVersion->getIf<std::int64_t>() : nullptr;
466 const auto* descriptorValue = field(*root, "descriptors");
467 const auto* generationValue = field(*root, "generations");
468 const auto* eventValue = field(*root, "events");
469 const auto* nextSequenceValue = field(*root, "nextEventSequence");
470 const auto* descriptorArray = descriptorValue ? descriptorValue->getIf<eve::Value::Array>() : nullptr;
471 const auto* generationArray = generationValue ? generationValue->getIf<eve::Value::Array>() : nullptr;
472 const auto* eventArray = eventValue ? eventValue->getIf<eve::Value::Array>() : nullptr;
473 std::uint64_t nextSequence = 0;
474 if (!versionNumber || *versionNumber != 1 || !descriptorArray || !generationArray || !eventArray ||
475 !nextSequenceValue || !readUint64String(*nextSequenceValue, nextSequence) || nextSequence == 0) {
476 return failure<void>(eve::DiagnosticCode::ParseError, "invalid policy snapshot fields");
477 }
478
479 Storage::State candidate;
480 for (std::size_t index = 0; index < generationArray->size(); ++index) {
481 const auto* item = (*generationArray)[index].getIf<eve::Value::Object>();
482 std::string domain;
483 std::string name;
484 std::uint64_t generation = 0;
485 const auto* generationField = item ? field(*item, "generation") : nullptr;
486 if (!item || !readString(*item, "domain", domain) || !readString(*item, "name", name) || domain.empty() ||
487 name.empty() || !generationField || !readUint64String(*generationField, generation) || generation == 0 ||
488 !candidate.entries.emplace(Key{domain, name}, Storage::Entry{eve::Generation(generation), std::nullopt})
489 .second) {
490 return failure<void>(eve::DiagnosticCode::ParseError,
491 "invalid policy generation at index " + std::to_string(index));
492 }
493 }
494
495 for (std::size_t index = 0; index < descriptorArray->size(); ++index) {
496 const auto* item = (*descriptorArray)[index].getIf<eve::Value::Object>();
497 std::string domain;
498 std::string name;
499 std::string kindName;
500 std::string schemaId;
501 eve::SchemaVersion version;
502 int priority = 0;
503 std::uint64_t generation = 0;
504 bool enabled = false;
505 const auto* versionField = item ? field(*item, "version") : nullptr;
506 const auto* priorityField = item ? field(*item, "priority") : nullptr;
507 const auto* enabledField = item ? field(*item, "enabled") : nullptr;
508 const auto* generationField = item ? field(*item, "generation") : nullptr;
509 const auto* metadata = item ? field(*item, "metadata") : nullptr;
510 ImplementationKind kind = ImplementationKind::Builtin;
511 if (!item || !readString(*item, "domain", domain) || !readString(*item, "name", name) || domain.empty() ||
512 name.empty() || !versionField || !readPositiveSchemaVersion(*versionField, version) || !priorityField ||
513 !readInt(*priorityField, priority) || !enabledField || !enabledField->isBool() || !generationField ||
514 !readUint64String(*generationField, generation) || generation == 0 ||
515 !readString(*item, "kind", kindName) || !parseImplementationKind(kindName, kind) || !metadata ||
516 !metadata->isObject() || !readString(*item, "schemaId", schemaId)) {
517 return failure<void>(eve::DiagnosticCode::ParseError,
518 "invalid policy descriptor at index " + std::to_string(index));
519 }
520 enabled = *enabledField->getIf<bool>();
521 auto entry = candidate.entries.find(Key{domain, name});
522 if (entry == candidate.entries.end() || entry->second.generation != eve::Generation(generation) ||
523 entry->second.value.has_value()) {
524 return failure<void>(eve::DiagnosticCode::InvariantViolation,
525 "inconsistent policy generation at index " + std::to_string(index));
526 }
527 auto metadataJson = metadata->toJson();
528 if (!metadataJson.ok()) {
529 return eve::Result<void>::failure(metadataJson.status());
530 }
531 PolicyDescriptor descriptor;
532 descriptor.domain = domain;
533 descriptor.name = name;
534 descriptor.version = version;
535 descriptor.priority = priority;
536 descriptor.enabled = enabled;
537 descriptor.kind = kind;
538 descriptor.schemaId = schemaId;
539 descriptor.generation = eve::Generation(generation);
540 descriptor.metadataJson = std::move(metadataJson).takeValue();
541 entry->second.value = std::move(descriptor);
542 }
543
544 eve::EventSequence previousSequence;
545 for (std::size_t index = 0; index < eventArray->size(); ++index) {
546 const auto* item = (*eventArray)[index].getIf<eve::Value::Object>();
547 std::string name;
548 std::string domain;
549 std::string policyName;
550 std::uint64_t sequence = 0;
551 std::uint64_t generation = 0;
552 const auto* sequenceField = item ? field(*item, "sequence") : nullptr;
553 const auto* generationField = item ? field(*item, "generation") : nullptr;
554 const auto* enabledField = item ? field(*item, "enabled") : nullptr;
555 const auto* versionField = item ? field(*item, "version") : nullptr;
556 eve::SchemaVersion version{1};
557 if (!item || !readString(*item, "name", name) || !readString(*item, "domain", domain) ||
558 !readString(*item, "policyName", policyName) || domain.empty() || policyName.empty() ||
559 (name != "policy_registered" && name != "policy_replaced" && name != "policy_removed" &&
560 name != "policy_enabled") ||
561 !sequenceField || !readUint64String(*sequenceField, sequence) || sequence == 0 ||
562 (!previousSequence.isZero() && eve::EventSequence(sequence) <= previousSequence) || !generationField ||
563 !readUint64String(*generationField, generation) || generation == 0 || !enabledField ||
564 !enabledField->isBool()) {
565 return failure<void>(eve::DiagnosticCode::ParseError,
566 "invalid policy event at index " + std::to_string(index));
567 }
568 if (versionField && !readPositiveSchemaVersion(*versionField, version)) {
569 return failure<void>(eve::DiagnosticCode::ParseError,
570 "invalid policy event version at index " + std::to_string(index));
571 }
572 Storage::Event event;
573 event.sequence = eve::EventSequence(sequence);
574 event.operation = name == "policy_removed"
575 ? eve::RegistryOperation::Remove
576 : (generation == 1 ? eve::RegistryOperation::Insert : eve::RegistryOperation::Replace);
577 event.key = Key{domain, policyName};
578 event.generation = eve::Generation(generation);
579 event.tombstone = name == "policy_removed";
580 event.label = name == "policy_enabled" ? "enabled" : std::string{};
581 event.data.version = version;
582 event.data.enabled = *enabledField->getIf<bool>();
583 candidate.events.push_back(std::move(event));
584 previousSequence = eve::EventSequence(sequence);
585 }
586 if (!previousSequence.isZero() && eve::EventSequence(nextSequence) <= previousSequence) {
587 return failure<void>(eve::DiagnosticCode::InvariantViolation,
588 "next event sequence must exceed retained events");
589 }
590 candidate.nextEventSequence = eve::EventSequence(nextSequence);
591
592 auto restored = storage_.restoreState(std::move(candidate));
593 if (!restored.ok()) {
594 return eve::Result<void>::failure(restored.status());
595 }
596 clearEventProjection();
597 revision_ = eve::Revision(nextSequence - 1);
598 tick_ = eve::SimulationTick{};
599 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
600}

610 const eve::SnapshotHashProvider& hashProvider) {
611 if (source.type != "policyregistry.registry" || source.schema != policySchema())
612 return failure<void>(eve::DiagnosticCode::InvalidArgument,
613 "snapshot does not belong to policyregistry::PolicyRegistry");
614 if (!instanceId_.isNil() && source.instanceId != instanceId_)
615 return failure<void>(eve::DiagnosticCode::Conflict,
616 "snapshot instanceId does not match policyregistry::PolicyRegistry");
617 auto migrated = policyMigrations().migrate(source, eve::SchemaVersion(1), hashProvider);
618 if (!migrated.ok()) return eve::Result<void>::failure(migrated.status());

622 if (!restored.ok()) return eve::Result<void>::failure(restored.status());
623 const auto& candidate = migrated.value();
624 instanceId_ = candidate.instanceId;
625 revision_ = candidate.revision;
626 tick_ = candidate.tick;
627 return eve::Result<void>::success();
