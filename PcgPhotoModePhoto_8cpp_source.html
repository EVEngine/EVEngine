30Result<void> PcgPhotoModePhotoAuthority::applyPhotoModeField(const PhotoModeAssignment& assignment) {
31 auto next = state_;
32 const auto invalid = [&](const char* message) {
33 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, message,
34 assignment.field, {}, "ui.pcgPhotoModePhoto"));
35 };
36 if (assignment.field == "m_isUsingPcgLighting") {
37 const auto* value = std::get_if<bool>(&assignment.value);
38 if (!value) return invalid("Pcg lighting identity requires a bool");
39 next.usingPcgLighting = *value;
40 } else if (assignment.field == "m_selectedPcgLightingProfile") {
41 const auto* value = std::get_if<int64_t>(&assignment.value);
42 if (!value || *value < -1) return invalid("Pcg lighting profile must be -1 or non-negative");
43 next.lightingProfile = static_cast<int>(*value);
44 } else if (assignment.field == "m_lastSceneName") {
45 const auto* value = std::get_if<std::string>(&assignment.value);
46 if (!value) return invalid("last scene name requires a string");
47 next.lastSceneName = *value;
48 } else if (assignment.field == "m_screenshotResolution") {
49 const auto* value = std::get_if<int64_t>(&assignment.value);
50 if (!value || *value < 0 || *value > 9) return invalid("screenshot resolution index must be in [0,9]");
51 next.screenshotResolution = static_cast<int>(*value);
52 } else if (assignment.field == "m_screenshotImageFormat") {
53 const auto* value = std::get_if<int64_t>(&assignment.value);
54 if (!value || *value < 0 || *value > 3) return invalid("screenshot format index must be in [0,3]");
55 next.screenshotFormat = static_cast<PcgScreenshotFormat>(*value);
56 } else {
57 const auto* value = std::get_if<bool>(&assignment.value);
58 if (!value) return invalid("photo option requires a bool");
59 if (assignment.field == "m_loadSavedSettings") next.loadSavedSettings = *value;
60 else if (assignment.field == "m_revertOnDisabled") next.revertOnDisabled = *value;
61 else if (assignment.field == "m_showFPS") next.showFps = *value;
62 else if (assignment.field == "m_showReticle") next.showReticle = *value;
63 else if (assignment.field == "m_showRuleOfThirds") next.showRuleOfThirds = *value;
64 else return invalid("unsupported Photo-domain field");
65 }
66 state_ = std::move(next);
67 return Result<void>::success();
68}
