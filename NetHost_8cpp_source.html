160void NetRpc::call(uint16_t msgId, const void* data, size_t n, bool reliable) {
161 if (!link_ || (data == nullptr && n > 0)) return;
162 std::vector<char> env;
163 env.push_back(static_cast<char>(msgId & 0xff));
164 env.push_back(static_cast<char>((msgId >> 8) & 0xff));
165 const char* p = static_cast<const char*>(data);
166 env.insert(env.end(), p, p + n);
167 link_->send(reliable ? UdpLink::MsgType::Reliable : UdpLink::MsgType::Unreliable, 0, env.data(), env.size());
168}

183void NetRpc::onLinkMessage(UdpLink::MsgType type, uint8_t channel, const char* data, size_t n) {
184 (void)type;
185 if (channel != 0 || n < 2) return;
186 const uint16_t msgId = static_cast<uint16_t>(uint8_t(data[0])) | (static_cast<uint16_t>(uint8_t(data[1])) << 8);
187 const char* payload = data + 2;
188 const size_t payloadLen = n - 2;
189
190 auto sit = scriptHandlers_.find(msgId);
191 if (sit != scriptHandlers_.end()) {
192 callScriptFn(sit->second, std::string(payload, payloadLen));
193 return;
194 }
195 auto it = handlers_.find(msgId);
196 if (it == handlers_.end()) return;
197 NetReader reader(payload, payloadLen);
198 it->second(reader);
199}

442void NetWorker::threadMain() {
443 while (true) {
444 std::vector<std::function<void()>> batch;
445 {
446 std::unique_lock<std::mutex> lock(mu_);
447 cv_.wait_for(lock, std::chrono::milliseconds(5), [this] { return !running_ || !jobs_.empty(); });
448 if (!running_ && jobs_.empty()) break;
449 batch.swap(jobs_);
450 }

452 if (job) job();
453 }
454 if (owner_) owner_->pollSockets();
