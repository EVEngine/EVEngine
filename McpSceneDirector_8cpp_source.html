22std::string snippetErrorText(HSQUIRRELVM vm, bool compile, const SQChar* source) {
23 eve::script::ScriptErrorContext ctx =
24 compile ? eve::script::captureCompileError(vm) : eve::script::takeLastScriptError(vm);
25 if (ctx.source.empty()) ctx.source = source;
26 if (ctx.message.empty()) ctx.message = compile ? "compile failed" : "runtime failed";
27 if (compile) ctx.stack.clear();
28 std::string report = std::string("type: ") + (compile ? "compile_error" : "runtime_error") +
29 "\norigin: " + (source == kMcpSnippetSource ? "eve_run_script" : "scene_director") +
30 "\nsource: " + ctx.source + "\nline: " + std::to_string(ctx.line) +
31 "\ncolumn: " + std::to_string(ctx.column) +
32 "\nmessage: " + (compile ? ctx.message : eve::script::formatScriptError(ctx));
33 mcpRecordSnippetError(report);
34 return report;
35}

54bool runMcpVmSnippet(HSQUIRRELVM vm, const std::string& source, std::string* err) {
55 eve::script::clearLastScriptError(vm);
56 const SQInteger top = sq_gettop(vm);
57 // MCP snippets are transient input, not project scripts. Compile them
58 // directly from memory so the unified project compiler cannot register a
59 // file-shaped source that hot reload may later treat as user content.
60 if (SQ_FAILED(
61 sq_compilebuffer(vm, source.c_str(), static_cast<SQInteger>(source.size()), kMcpSnippetSource, SQTrue))) {
62 sq_settop(vm, top);
63 if (err) *err = snippetErrorText(vm, true, kMcpSnippetSource);
64 return false;
65 }
66 sq_pushroottable(vm);
67 if (SQ_FAILED(sq_call(vm, 1, SQFalse, SQTrue))) {
68 sq_settop(vm, top);
69 if (err) *err = snippetErrorText(vm, false, kMcpSnippetSource);
70 return false;
71 }
72 sq_settop(vm, top);
73 return true;
74}

76std::string callSceneDirectorReturn(HSQUIRRELVM vm, const std::string& snippet, std::string* err) {
77 eve::script::clearLastScriptError(vm);
78 const SQInteger top = sq_gettop(vm);
79 if (SQ_FAILED(sq_compilebuffer(vm, snippet.c_str(), static_cast<SQInteger>(snippet.size()), kMcpSceneDirectorSource,
80 SQTrue))) {
81 sq_settop(vm, top);
82 if (err) *err = snippetErrorText(vm, true, kMcpSceneDirectorSource);
83 return {};
84 }
85 sq_pushroottable(vm);
86 if (SQ_FAILED(sq_call(vm, 1, SQTrue, SQTrue))) {
87 sq_settop(vm, top);
88 std::string text = snippetErrorText(vm, false, kMcpSceneDirectorSource);
89 if (err)
90 *err = text == "runtime failed" ? "runtime failed (is the scene_director kit installed?)" : std::move(text);
91 return {};
92 }
93 std::string json = sqValueToJson(vm, -1);
94 sq_settop(vm, top);
95 return json;
96}

102bool ensureSceneDirectorInstalled(HSQUIRRELVM vm, std::string* err, bool allowHeavyWork) {
103 if (!vm) {
104 if (err) *err = "no VM";
105 return false;
106 }
107 if (sceneDirectorPresent(vm)) {
108 g_installPending.store(false);
109 return true;
110 }
111 if (!allowHeavyWork) {
112 g_installPending.store(true);
113 if (err) *err = "deferred-to-update";
114 return false;
115 }
116 std::lock_guard lock(g_installMu);
117 if (sceneDirectorPresent(vm)) {
118 g_installPending.store(false);
119 return true;
120 }
121 const bool ok = installSceneDirectorNow(vm, err);
122 if (ok) g_installPending.store(false);
123 return ok;
124}
