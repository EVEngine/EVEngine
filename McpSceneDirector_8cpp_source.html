24bool runMcpVmSnippet(HSQUIRRELVM vm, const std::string& source, std::string* err) {
25 const SQInteger top = sq_gettop(vm);
26 // MCP snippets are transient input, not project scripts. Compile them
27 // directly from memory so the unified project compiler cannot register a
28 // file-shaped source that hot reload may later treat as user content.
29 if (SQ_FAILED(
30 sq_compilebuffer(vm, source.c_str(), static_cast<SQInteger>(source.size()), kMcpSnippetSource, SQTrue))) {
31 sq_settop(vm, top);
32 if (err) *err = snippetErrorText(vm, true);
33 return false;
34 }
35 sq_pushroottable(vm);
36 if (SQ_FAILED(sq_call(vm, 1, SQFalse, SQTrue))) {
37 sq_settop(vm, top);
38 if (err) *err = snippetErrorText(vm, false);
39 return false;
40 }
41 sq_settop(vm, top);
42 return true;
43}

45std::string callSceneDirectorReturn(HSQUIRRELVM vm, const std::string& snippet, std::string* err) {
46 const SQInteger top = sq_gettop(vm);
47 if (SQ_FAILED(sq_compilebuffer(vm, snippet.c_str(), static_cast<SQInteger>(snippet.size()), kMcpSceneDirectorSource,
48 SQTrue))) {
49 sq_settop(vm, top);
50 if (err) *err = snippetErrorText(vm, true);
51 return {};
52 }
53 sq_pushroottable(vm);
54 if (SQ_FAILED(sq_call(vm, 1, SQTrue, SQTrue))) {
55 sq_settop(vm, top);
56 std::string text = snippetErrorText(vm, false);
57 if (err)
58 *err = text == "runtime failed" ? "runtime failed (is the scene_director kit installed?)" : std::move(text);
59 return {};
60 }
61 std::string json = sqValueToJson(vm, -1);
62 sq_settop(vm, top);
63 return json;
64}

66bool ensureSceneDirectorInstalled(HSQUIRRELVM vm, std::string* err) {
67 const std::string check = "return (\"scene_director\" in getroottable()) ? (scene_director != null) : false;";
68 const std::string out = callSceneDirectorReturn(vm, check, nullptr);
69 if (!out.empty() && out.find("true") != std::string::npos) return true;
70 const char* kit = eve::scene_director_content;
71 if (!kit || !*kit) {
72 if (err) *err = "scene_director.nut not embedded (rebuild EVScripts)";
73 return false;
74 }
75 return runMcpVmSnippet(vm, kit, err);
76}
