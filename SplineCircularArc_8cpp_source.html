35 if (chord < 1e-7 || direction < 1e-7) return invalid();

37 if (reference == CircularArcReference::End) cross = -cross;
38 const double theta = std::atan2(cross, dx * x + dz * z);

41 arc.ax = ax;
42 arc.az = az;
43 arc.bx = bx;
44 arc.bz = bz;

46 const double bulge = std::tan(theta * .5), offset = (1 - bulge * bulge) / (4 * bulge);
47 arc.line = false;
48 arc.cx = (ax + bx) * .5 - z * offset;
49 arc.cz = (az + bz) * .5 + x * offset;
50 arc.radius = chord * (1 + bulge * bulge) / (4 * std::abs(bulge));
51 arc.angle = std::atan2(az - arc.cz, ax - arc.cx);
52 arc.sweep = 4 * std::atan(bulge);
53 const double maximum = std::numeric_limits<float>::max();
54 if (!std::isfinite(arc.radius) || arc.radius * std::abs(arc.sweep) > maximum ||
55 std::abs(arc.cx) + arc.radius > maximum || std::abs(arc.cz) + arc.radius > maximum)
