30const char* transformKernel() {
31#ifdef EVENGINE_WEBGPU
32 return R"(
33struct Data { values: array<f32> };
34struct Push { data: array<vec4f, 8> };
35@group(0) @binding(0) var<storage, read_write> points: Data;
36@group(0) @binding(8) var<uniform> push: Push;
37fn parameter(index: u32) -> f32 {
38 let value = push.data[index / 4u];
39 switch index % 4u {
40 case 0u: { return value.x; }
41 case 1u: { return value.y; }
42 case 2u: { return value.z; }
43 default: { return value.w; }
44 }
45}
46@compute @workgroup_size(64)
47fn main(@builtin(global_invocation_id) gid: vec3u) {
48 if (gid.x >= u32(parameter(0u))) { return; }
49 let base = gid.x * 11u;
50 for (var operation = 0u; operation < min(u32(parameter(1u)), 4u); operation++) {
51 let offset = 2u + operation * 7u;
52 let sx = parameter(offset + 3u);
53 let sy = parameter(offset + 4u);
54 let sz = parameter(offset + 5u);
55 let yaw = parameter(offset + 6u);
56 let radians = yaw * 0.017453292519943295;
57 let c = cos(radians);
58 let s = sin(radians);
59 let x = points.values[base] * sx;
60 let z = points.values[base + 2u] * sz;
61 points.values[base] = x * c - z * s + parameter(offset);
62 points.values[base + 1u] = points.values[base + 1u] * sy + parameter(offset + 1u);
63 points.values[base + 2u] = x * s + z * c + parameter(offset + 2u);
64 var nx = points.values[base + 3u] / select(1.0, sx, abs(sx) > 0.000001);
65 var ny = points.values[base + 4u] / select(1.0, sy, abs(sy) > 0.000001);
66 var nz = points.values[base + 5u] / select(1.0, sz, abs(sz) > 0.000001);
67 let rotated = vec3f(nx * c - nz * s, ny, nx * s + nz * c);
68 let normal = select(rotated, normalize(rotated), dot(rotated, rotated) > 0.0);
69 points.values[base + 3u] = normal.x;
70 points.values[base + 4u] = normal.y;
71 points.values[base + 5u] = normal.z;
72 points.values[base + 6u] += yaw;
73 points.values[base + 7u] *= sx;
74 points.values[base + 8u] *= sy;
75 points.values[base + 9u] *= sz;
76 }
77}
78)";
79#else
80 return R"(#version 450
81layout(local_size_x = 64) in;
82layout(set = 0, binding = 0) buffer Data { float values[]; } points;
83layout(push_constant) uniform Push { float data[32]; } push;
84void main() {
85 uint i = gl_GlobalInvocationID.x;
86 if (i >= uint(push.data[0])) return;
87 uint base = i * 11u;
88 for (int operation = 0; operation < min(int(push.data[1]), 4); ++operation) {
89 int offset = 2 + operation * 7;
90 float sx = push.data[offset + 3], sy = push.data[offset + 4], sz = push.data[offset + 5];
91 float yaw = push.data[offset + 6];
92 float radians = yaw * 0.017453292519943295;
93 float c = cos(radians), s = sin(radians);
94 float x = points.values[base] * sx;
95 float z = points.values[base + 2u] * sz;
96 points.values[base] = x * c - z * s + push.data[offset];
97 points.values[base + 1u] = points.values[base + 1u] * sy + push.data[offset + 1];
98 points.values[base + 2u] = x * s + z * c + push.data[offset + 2];
99 float nx = points.values[base + 3u] / (abs(sx) > 0.000001 ? sx : 1.0);
100 float ny = points.values[base + 4u] / (abs(sy) > 0.000001 ? sy : 1.0);
101 float nz = points.values[base + 5u] / (abs(sz) > 0.000001 ? sz : 1.0);
102 vec3 normal = vec3(nx * c - nz * s, ny, nx * s + nz * c);
103 float normalLength = length(normal);
104 if (normalLength > 0.0) normal /= normalLength;
105 points.values[base + 3u] = normal.x;
106 points.values[base + 4u] = normal.y;
107 points.values[base + 5u] = normal.z;
108 points.values[base + 6u] += yaw;
109 points.values[base + 7u] *= sx;
110 points.values[base + 8u] *= sy;
111 points.values[base + 9u] *= sz;
112 }
113}
114)";
115#endif

151bool PointCompute::transformChain(const PointSet& input, PointSet& output, const std::vector<Transform>& transforms) {
152 error_.clear();
153 if (transforms.empty() || transforms.size() > kMaxTransforms) {
154 error_ = "transform chain must contain one to four operations";
155 return false;
156 }
157 if (input.empty()) {
158 output = input;
159 return true;
160 }
161 constexpr size_t bytesPerPoint = size_t(kPointFloats) * sizeof(float);
162 if (size_t(input.getCount()) > size_t(std::numeric_limits<int>::max()) / bytesPerPoint) {
163 error_ = "point data exceeds GPU buffer address range";
164 return false;
165 }
166 try {
167 if (forcedFailure("compile")) {
168 error_ = "forced shader compilation failure";
169 return false;
170 }
171 auto* gpu = eve::ModuleManager::getInstance<eve::gpgpu::Gpgpu>("Gpgpu");
172 if (!gpu) gpu = eve::gpgpu::Gpgpu::create();
173 if (!gpu || !gpu->isAvailable()) {
174 error_ = "compute device unavailable";
175 return false;
176 }
177
178 impl_->packed.resize(size_t(input.getCount()) * kPointFloats);
179 std::vector<float>& packed = impl_->packed;
180 for (int index = 0; index < input.getCount(); ++index) {
181 const ProcgenPoint& point = input.points()[size_t(index)];
182 float* values = packed.data() + size_t(index) * kPointFloats;
183 values[0] = point.x;
184 values[1] = point.y;
185 values[2] = point.z;
186 values[3] = point.normalX;
187 values[4] = point.normalY;
188 values[5] = point.normalZ;
189 values[6] = point.yaw;
190 values[7] = point.scaleX;
191 values[8] = point.scaleY;
192 values[9] = point.scaleZ;
193 values[10] = point.density;
194 }
195 const int byteSize = int(packed.size() * sizeof(float));
196 if (!impl_->shader) impl_->shader.reset(gpu->newShader(transformKernel()));
197 if (!impl_->sequence) impl_->sequence.reset(gpu->newSequence());
198 const bool reusedBuffers = byteSize <= impl_->capacityBytes;
199 if (byteSize > impl_->capacityBytes) {
200 if (forcedFailure("allocation")) {
201 error_ = "forced GPU allocation failure";
202 return false;
203 }
204 int capacity = std::max(byteSize, impl_->capacityBytes + impl_->capacityBytes / 2);
205 impl_->storage.reset(gpu->newBuffer(capacity, "storage"));
206 impl_->staging.reset(gpu->newBuffer(capacity, "staging"));
207 impl_->capacityBytes = capacity;
208 impl_->peakBufferBytes = std::max(impl_->peakBufferBytes, uint64_t(capacity));
209 }
210 if (!impl_->storage || !impl_->staging || !impl_->shader || !impl_->sequence ||
211 !impl_->sequence->isAvailable()) {
212 error_ = "compute resources unavailable";
213 return false;
214 }
215 impl_->shader->bindBuffer(0, impl_->storage.get());
216 impl_->shader->setFloat(0, float(input.getCount()));
217 impl_->shader->setFloat(1, float(transforms.size()));
218 for (size_t index = 0; index < transforms.size(); ++index) {
219 const Transform& transform = transforms[index];
220 const int offset = 2 + int(index) * 7;
221 impl_->shader->setFloat(offset, transform.x);
222 impl_->shader->setFloat(offset + 1, transform.y);
223 impl_->shader->setFloat(offset + 2, transform.z);
224 impl_->shader->setFloat(offset + 3, transform.scaleX);
225 impl_->shader->setFloat(offset + 4, transform.scaleY);
226 impl_->shader->setFloat(offset + 5, transform.scaleZ);
227 impl_->shader->setFloat(offset + 6, transform.yaw);
228 }
229 impl_->sequence->begin();
230 impl_->sequence->recordUpload(impl_->storage.get(), packed.data(), uint64_t(byteSize));
231 impl_->sequence->recordDispatch(impl_->shader.get(), (input.getCount() + kWorkgroupSize - 1) / kWorkgroupSize);
232 impl_->sequence->recordDownload(impl_->storage.get(), impl_->staging.get(), uint64_t(byteSize));
233 if (forcedFailure("submit")) {
234 error_ = "forced GPU submission failure";
235 impl_->reset();
236 return false;
237 }
238 impl_->sequence->submit();
239 if (forcedFailure("readback")) {
240 error_ = "forced GPU readback failure";
241 impl_->reset();
242 return false;
243 }
244 impl_->staging->downloadBytes(packed.data(), uint64_t(byteSize));
245 ++impl_->uploadCount;
246 ++impl_->dispatchCount;
247 ++impl_->readbackCount;
248 if (reusedBuffers) ++impl_->bufferReuseCount;
249 impl_->lastFusedTransformCount = int(transforms.size());
250
251 output = input;
252 for (int index = 0; index < output.getCount(); ++index) {
253 ProcgenPoint& point = output.mutablePoint(size_t(index));
254 const float* values = packed.data() + size_t(index) * kPointFloats;
255 point.x = values[0];
256 point.y = values[1];
257 point.z = values[2];
258 point.normalX = values[3];
259 point.normalY = values[4];
260 point.normalZ = values[5];
261 point.yaw = values[6];
262 point.scaleX = values[7];
263 point.scaleY = values[8];
264 point.scaleZ = values[9];
265 point.density = values[10];
266 }
267 return true;
268 } catch (const std::exception& exception) {
269 error_ = exception.what();
270 impl_->reset();
271 } catch (...) {
272 error_ = "unknown compute failure";
273 impl_->reset();
274 }
275 return false;
276}
