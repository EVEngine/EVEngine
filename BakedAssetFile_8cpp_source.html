64Result<void> publishBakedAsset(std::string_view path, std::string_view bytes) {
65 if (path.empty() || path.find('\0') != std::string_view::npos)
66 return Result<void>::failure(
67 Diagnostic::error(DiagnosticCode::InvalidArgument, "asset path is empty or contains NUL"));
68 const std::filesystem::path target(path);
69 if (target.filename().empty())
70 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "asset filename is empty"));
71#if defined(__linux__)
72 const auto parent = target.parent_path().empty() ? std::filesystem::path(".") : target.parent_path();
73 PublicationDescriptor directory{::open(parent.c_str(), O_RDONLY | O_DIRECTORY | O_CLOEXEC)};
74 if (directory.value < 0)
75 return Result<void>::failure(
76 Diagnostic::error(DiagnosticCode::Failed, "cannot open asset publication directory", std::string(path)));
77 // No visible staging name: the kernel reclaims unpublished data even after SIGKILL.
78 PublicationDescriptor data{::openat(directory.value, ".", O_TMPFILE | O_WRONLY | O_CLOEXEC, 0666)};
79 if (data.value < 0) {
80 const int error = errno;
81 const auto code = error == EOPNOTSUPP || error == EINVAL || error == EISDIR ? DiagnosticCode::Unsupported
82 : DiagnosticCode::Failed;
83 return Result<void>::failure(Diagnostic::error(code, "anonymous asset staging unavailable", std::string(path),
84 {{"system_error", std::to_string(error)}}));
85 }
86 std::size_t offset = 0;
87 while (offset < bytes.size()) {
88 const auto count =
89 std::min(bytes.size() - offset, static_cast<std::size_t>(std::numeric_limits<ssize_t>::max()));
90 const auto written = ::write(data.value, bytes.data() + offset, count);
91 if (written < 0 && errno == EINTR) continue;
92 if (written <= 0)
93 return Result<void>::failure(
94 Diagnostic::error(DiagnosticCode::Failed, "asset staging write failed", std::string(path)));
95 offset += static_cast<std::size_t>(written);
96 }
97 auto synced = syncPublicationDescriptor(data.value, path, "not_published");
98 if (!synced) return synced;
99 // procfs descriptor linking works without CAP_DAC_READ_SEARCH (unlike AT_EMPTY_PATH).
100 const auto source = "/proc/self/fd/" + std::to_string(data.value);
101 if (::linkat(AT_FDCWD, source.c_str(), directory.value, target.filename().c_str(), AT_SYMLINK_FOLLOW) < 0) {
102 const int error = errno;
103 return Result<void>::failure(
104 Diagnostic::error(error == EEXIST ? DiagnosticCode::Conflict : DiagnosticCode::Failed,
105 "asset publication failed (requires procfs descriptor links and a new path)",
106 std::string(path), {{"system_error", std::to_string(error)}}));
107 }
108 // Never roll back a complete target that readers may already have observed.
109 return syncPublicationDescriptor(directory.value, path, "published");
110#else
