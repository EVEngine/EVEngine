34eve::Result<void> WeaponEffectExecutor::applyImmediate(WeaponEffectTarget& target,
35 const effects::EffectInstance& effect) const {
36 auto valid = validate(target);
37 if (!valid) return valid;
38 if (effect.hasTag("weapon:jam")) target.jammed = true;
39 if (effect.hasTag("weapon:recoil")) {
40 if (effect.magnitude > std::numeric_limits<double>::max() - target.recoil)

42 "weapon recoil overflows", "target.recoil"));
43 target.recoil += effect.magnitude;
44 }
45 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));

48eve::Result<WeaponEffectUpdate> WeaponEffectExecutor::settle(WeaponEffectTarget& target,
49 effects::EffectUpdateSummary lifecycle) const {
50 auto valid = validate(target);
51 if (!valid) return eve::Result<WeaponEffectUpdate>::failure(valid.status());
52 WeaponEffectUpdate result;
53 result.lifecycle = std::move(lifecycle);
54 for (const auto& tick : result.lifecycle.periodicTicks) {
55 if (tick.stackCount == 0 || !std::isfinite(tick.magnitude) || tick.magnitude < 0.0)
56 return eve::Result<WeaponEffectUpdate>::failure(eve::Diagnostic::error(
57 eve::DiagnosticCode::InvalidArgument, "weapon periodic effect value is invalid", "periodic"));
58 const double amount = tick.magnitude * static_cast<double>(tick.stackCount);
59 if (!std::isfinite(amount))
60 return eve::Result<WeaponEffectUpdate>::failure(eve::Diagnostic::error(
61 eve::DiagnosticCode::InvalidArgument, "weapon periodic effect overflows", "periodic"));
62 if (has(tick, "weapon:recoil")) {
63 if (amount > std::numeric_limits<double>::max() - target.recoil)
64 return eve::Result<WeaponEffectUpdate>::failure(eve::Diagnostic::error(
65 eve::DiagnosticCode::InvalidArgument, "weapon recoil settlement overflows", "target.recoil"));
66 target.recoil += amount;
67 } else if (has(tick, "weapon:jam")) {
68 target.jammed = true;
69 } else {
70 target.heat = std::min(target.maxHeat, target.heat + amount);
71 if (target.heat >= target.maxHeat) {
72 target.jammed = true;
73 ++target.blockedShots;
74 result.jammed = true;
75 }
76 }
77 ++result.settled;
78 }
79 result.jammed = result.jammed || target.jammed;
80 return eve::Result<WeaponEffectUpdate>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
81}

124eve::Result<WeaponEffectUpdate> WeaponEffectAdapter::advance(const eve::SimulationStep& step) {
125 auto candidate = container_.clone();
126 auto lifecycle = effects::EffectExecutor{}.advance(candidate, step);
127 if (!lifecycle) return eve::Result<WeaponEffectUpdate>::failure(lifecycle.status());
128 auto targetCandidate = target_;
129 auto settled = executor_.settle(targetCandidate, std::move(lifecycle).takeValue());
130 if (!settled) return settled;
131 auto result = std::move(settled).takeValue();
132 target_ = std::move(targetCandidate);
133 container_ = std::move(candidate);
134 return eve::Result<WeaponEffectUpdate>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
