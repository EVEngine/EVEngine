61Result<DamageAmounts> DamageRuntime::preview(const CombatState& target, const DamageRequest& request) const {
62 auto stateValid = target.validate();
63 if (!stateValid) return Result<DamageAmounts>::failure(stateValid.status());
64 auto requestValid = request.validate();
65 if (!requestValid) return Result<DamageAmounts>::failure(requestValid.status());
66 if (target.subject != request.target)
67 return invalid<DamageAmounts>("Damage request target does not match combat state", "target");
68 DamageAmounts amounts{request.healthDamage, request.poiseDamage, 1.0};
69 if (rule_) {
70 auto evaluated = rule_->evaluate(request, target);
71 if (!evaluated) return Result<DamageAmounts>::failure(evaluated.status());
72 amounts = std::move(evaluated).takeValue();
73 }
74 auto valid = validateAmounts(amounts);
75 if (!valid) return Result<DamageAmounts>::failure(valid.status());
76 return Result<DamageAmounts>::success(amounts);
77}

79Result<DamageOutcome> DamageRuntime::apply(CombatState& target, const DamageRequest& request) const {
80 auto policyValid = policy_.validate();
81 if (!policyValid) return Result<DamageOutcome>::failure(policyValid.status());
82 auto previewed = preview(target, request);
83 if (!previewed) return Result<DamageOutcome>::failure(previewed.status());
84 DamageAmounts amounts = std::move(previewed).takeValue();
85 DamageRuleSource source = DamageRuleSource::Default;
86 if (rule_) source = DamageRuleSource::Provider;
87
88 CombatState candidate = target;
89 candidate.health = std::max(0.0, candidate.health - amounts.healthDamage);
90 candidate.poise = std::max(0.0, candidate.poise - amounts.poiseDamage);
91 const double appliedHealth = target.health - candidate.health;
92 const double appliedPoise = target.poise - candidate.poise;
93
94 HitReaction reaction = HitReaction::None;
95 if (candidate.health == 0.0 && target.health > 0.0) {
96 reaction = HitReaction::Death;
97 } else if (candidate.poise == 0.0 && appliedPoise > 0.0) {
98 reaction = HitReaction::Knockdown;
99 } else if (target.maxPoise > 0.0 && appliedPoise >= target.maxPoise * policy_.staggerPoiseFraction) {
100 reaction = HitReaction::Stagger;
101 } else if (appliedHealth > 0.0 && appliedHealth >= policy_.flinchDamageThreshold) {
102 reaction = HitReaction::Flinch;
103 }
104
105 DamageOutcome outcome{request.source,
106 request.target,
107 target.health,
108 candidate.health,
109 target.poise,
110 candidate.poise,
111 appliedHealth,
112 appliedPoise,
113 {request.knockback.x * amounts.knockbackScale, request.knockback.y * amounts.knockbackScale,
114 request.knockback.z * amounts.knockbackScale},
115 reaction,
116 source};
117 target = candidate;
118 return Result<DamageOutcome>::success(std::move(outcome), Status::success(StatusCode::Applied));
119}

121Result<double> DamageRuntime::recoverPoise(CombatState& target, double amount) const {
122 auto valid = target.validate();
123 if (!valid) return Result<double>::failure(valid.status());
124 if (!finiteNonNegative(amount)) return invalid<double>("Poise recovery amount is invalid", "amount");
125 const double recovered = std::min(target.maxPoise, target.poise + amount);
126 const bool changed = recovered != target.poise;
127 target.poise = recovered;
128 return Result<double>::success(recovered, Status::success(changed ? StatusCode::Applied : StatusCode::NoOp));
129}
