27class DamageSettlementPolicy final : public settlement::ISettlementPolicy {
28public:
29 DamageSettlementPolicy(CombatState& target, const DamageRequest& request, DamageAmounts amounts,
30 const HitReactionPolicy& reactionPolicy)
31 : target_(target), request_(request), reactionPolicy_(reactionPolicy), amounts_(std::move(amounts)) {}
32
33 Result<void> validate(settlement::SettlementContext&) override {
34 auto policyValid = reactionPolicy_.validate();
35 if (!policyValid) return policyValid;
36 return Result<void>::success();
37 }
38
39 Result<void> sourceModifiers(settlement::SettlementContext& context) override {
40 return context.setMagnitude(amounts_.healthDamage);
41 }
42 Result<void> targetMitigation(settlement::SettlementContext& context) override {
43 const double before = context.magnitude();
44 const double after = before * request_.incomingDamageMultiplier;
45 if (after < before) {
46 auto resisted = context.addResisted(before - after);
47 if (!resisted) return resisted;
48 }
49 return context.setMagnitude(after);
50 }
51 Result<void> armorShield(settlement::SettlementContext& context) override {
52 shieldAbsorbed_ = std::min(context.magnitude(), request_.availableShield);
53 auto recorded = context.addAbsorbed(shieldAbsorbed_);
54 if (!recorded) return recorded;
55 return context.setMagnitude(context.magnitude() - shieldAbsorbed_);
56 }
57 Result<void> clamp(settlement::SettlementContext& context) override { return context.setClampMax(target_.health); }
58
59 Result<settlement::PreparedApply> prepareApply(const settlement::SettlementContext& context) override {
60 CombatState candidate = target_;
61 candidate.health = std::max(0.0, candidate.health - context.magnitude());
62 candidate.poise = std::max(0.0, candidate.poise - amounts_.poiseDamage);
63 const CombatState before = target_;
64 return Result<settlement::PreparedApply>::success(settlement::PreparedApply(
65 [this, candidate]() {
66 target_ = candidate;
67 return Result<void>::success(Status::success(StatusCode::Applied));
68 },
69 [this, before]() { target_ = before; }));
70 }
71
72 Result<std::vector<settlement::SettlementRequest>> prepareTrigger(
73 const settlement::SettlementContext& context, const settlement::SettlementResult& result) override {
74 std::vector<settlement::SettlementRequest> transitions;
75 const auto append = [&](std::string key) {
76 settlement::SettlementRequest transition = context.request();
77 transition.kind = "trigger";
78 transition.resource.clear();
79 transition.magnitude = 0.0;
80 transition.decisions.clear();
81 transition.trigger = std::move(key);
82 transition.chain = {};
83 transition.tags.push_back("trigger:" + transition.trigger);
84 transitions.push_back(std::move(transition));
85 };
86
87 if (request_.availableShield > 0.0 && shieldAbsorbed_ >= request_.availableShield)
88 append("shield_break");
89 const bool died = target_.health > 0.0 && result.applied >= target_.health;
90 if (died) {
91 append("death");
92 if (context.request().source.isValid() && context.request().source != context.request().target)
93 append("kill");
94 }
95 return Result<std::vector<settlement::SettlementRequest>>::success(std::move(transitions));
96 }
97
98 [[nodiscard]] const DamageAmounts& amounts() const noexcept { return amounts_; }
99 [[nodiscard]] double shieldAbsorbed() const noexcept { return shieldAbsorbed_; }
100
101private:
102 CombatState& target_;
103 const DamageRequest& request_;
104 const HitReactionPolicy& reactionPolicy_;
105 DamageAmounts amounts_;
106 double shieldAbsorbed_ = 0.0;
107};

171Result<DamageAmounts> DamageRuntime::preview(const CombatState& target, const DamageRequest& request) const {
172 auto stateValid = target.validate();
173 if (!stateValid) return Result<DamageAmounts>::failure(stateValid.status());
174 auto requestValid = request.validate();
175 if (!requestValid) return Result<DamageAmounts>::failure(requestValid.status());
176 if (target.subject != request.target)
177 return Result<DamageAmounts>::failure(Diagnostic::error(
178 DiagnosticCode::InvalidArgument, "Damage request target does not match combat state", "target"));
179 DamageAmounts amounts{request.healthDamage, request.poiseDamage, 1.0};
180 if (rule_) {
181 auto evaluated = rule_->evaluate(request, target);
182 if (!evaluated) return Result<DamageAmounts>::failure(evaluated.status());
183 amounts = std::move(evaluated).takeValue();
184 }
185 auto valid = validateAmounts(amounts);
186 if (!valid) return Result<DamageAmounts>::failure(valid.status());
187 return Result<DamageAmounts>::success(amounts);
188}

190Result<DamageOutcome> DamageRuntime::apply(CombatState& target, const DamageRequest& request) const {
191 auto previewed = preview(target, request);
192 if (!previewed) return Result<DamageOutcome>::failure(previewed.status());
193 const CombatState before = target;
194 DamageSettlementPolicy adapter(target, request, std::move(previewed).takeValue(), policy_);
195 settlement::SettlementRequest settlementRequest;
196 settlementRequest.source = request.source;
197 settlementRequest.target = request.target;
198 settlementRequest.kind = "damage";
199 settlementRequest.magnitude = request.healthDamage;
200 settlementRequest.tags = {request.damageType};
201 Value::Object context;
202 context["damage_type"] = request.damageType;
203 context["poise_damage"] = request.poiseDamage;
204 context["incoming_damage_multiplier"] = request.incomingDamageMultiplier;
205 context["available_shield"] = request.availableShield;
206 context["target_health_ratio"] =
207 target.maxHealth > 0.0 ? target.health / target.maxHealth : 0.0;
208 context["target_poise_ratio"] = target.maxPoise > 0.0 ? target.poise / target.maxPoise : 0.0;
209 settlementRequest.context = Value(std::move(context));
210 auto settled = settlement_.settle(settlementRequest, adapter);
211 if (!settled) return Result<DamageOutcome>::failure(settled.status());
212 const auto settlementResult = std::move(settled).takeValue();
213 const DamageAmounts amounts = adapter.amounts();
214 const double appliedHealth = before.health - target.health;
215 const double appliedPoise = before.poise - target.poise;
216
217 HitReaction reaction = HitReaction::None;
218 if (target.health == 0.0 && before.health > 0.0) {
219 reaction = HitReaction::Death;
220 } else if (target.poise == 0.0 && appliedPoise > 0.0) {
221 reaction = HitReaction::Knockdown;
222 } else if (before.maxPoise > 0.0 && appliedPoise >= before.maxPoise * policy_.staggerPoiseFraction) {
223 reaction = HitReaction::Stagger;
224 } else if (appliedHealth > 0.0 && appliedHealth >= policy_.flinchDamageThreshold) {
225 reaction = HitReaction::Flinch;
226 }
227
228 DamageOutcome outcome{request.source,
229 request.target,
230 before.health,
231 target.health,
232 before.poise,
233 target.poise,
234 appliedHealth,
235 appliedPoise,
236 adapter.shieldAbsorbed(),
237 {request.knockback.x * amounts.knockbackScale, request.knockback.y * amounts.knockbackScale,
238 request.knockback.z * amounts.knockbackScale},
239 reaction,
240 rule_ ? DamageRuleSource::Provider : DamageRuleSource::Default,
241 std::move(settlementRequest),
242 std::move(settlementResult)};
243 return Result<DamageOutcome>::success(std::move(outcome), Status::success(StatusCode::Applied));
244}
