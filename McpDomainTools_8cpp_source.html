75std::string decalProject(Poco::JSON::Object::Ptr args) {
76 auto* decals = eve::cap::query<eve::IDecalQuery>();
77 if (!decals) return unavailablePayload(eve::IDecalQuery::capabilityName);
78
79 // A decal needs a surface normal; up is the least surprising default, and a
80 // caller that omits it almost always means "on the ground".
81 const float nx = getArgFloat(args, "nx", 0.f);
82 const float ny = getArgFloat(args, "ny", 1.f);
83 const float nz = getArgFloat(args, "nz", 0.f);
84
85 // Zero means "engine default" for every size/time field: the decal module
86 // owns those defaults, so the tool must not invent a second set.
87 const int id =
88 decals->project(getArgFloat(args, "x"), getArgFloat(args, "y"), getArgFloat(args, "z"), nx, ny, nz,
89 /*albedoTexture=*/nullptr, getArgString(args, "kind"), getArgFloat(args, "size"),
90 getArgFloat(args, "depth"), getArgBool(args, "randomYaw"), getArgInt(args, "seed"),
91 getArgFloat(args, "fadeIn"), getArgFloat(args, "lifetime"), getArgFloat(args, "fadeOut"));
92 if (id <= 0) return errorPayload("decal projection was rejected");
93
94 Poco::JSON::Object::Ptr out = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
95 out->set("ok", true);
96 out->set("id", id);
97 out->set("count", decals->count());
98 return textPayload(out);
99}

148std::string physicsSphereCast(Poco::JSON::Object::Ptr args) {
149 auto* cast = eve::cap::query<eve::ICameraObstructionQuery>();
150 if (!cast) return unavailablePayload(eve::ICameraObstructionQuery::capabilityName);
151 if (!args || !args->has("from") || !args->has("to")) return errorPayload("from and to are required");
152
153 const std::array<float, 3> from = getArgVec3(args, "from", {0.f, 0.f, 0.f});
154 const std::array<float, 3> to = getArgVec3(args, "to", {0.f, 0.f, 0.f});
155 const float radius = getArgFloat(args, "radius", 0.f);
156 const long long mask = getArgInt64(args, "maskBits", static_cast<long long>(kAllCategoryBits));
157
158 eve::CameraObstructionHit hit;
159 const bool swept = cast->sphereCast(from[0], from[1], from[2], to[0], to[1], to[2], radius,
160 static_cast<std::uint64_t>(mask), getArgInt(args, "ignoredBodyId", -1), &hit);
161 if (!swept) return errorPayload("sphere cast is unavailable (no live 3D physics world)");
162
163 Poco::JSON::Object::Ptr out = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
164 out->set("ok", true);
165 out->set("hit", hit.hit);
166 out->set("radius", radius);
167 if (hit.hit) {
168 out->set("bodyId", hit.bodyId);
169 out->set("fraction", hit.fraction);
170 Poco::JSON::Array::Ptr point = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
171 point->add(hit.x);
172 point->add(hit.y);
173 point->add(hit.z);
174 out->set("point", point);
175 Poco::JSON::Array::Ptr normal = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
176 normal->add(hit.normalX);
177 normal->add(hit.normalY);
178 normal->add(hit.normalZ);
179 out->set("normal", normal);
180 }
181 return textPayload(out);
182}

233std::string sensingLastQuery(Poco::JSON::Object::Ptr args) {
234 auto* sensing = eve::cap::query<eve::ISensingQuery>();
235 if (!sensing) return unavailablePayload(eve::ISensingQuery::capabilityName);
236
237 const bool hasIndex = args && args->has("world");
238 const int index = getArgInt(args, "world", -1);
239
240 Poco::JSON::Object::Ptr out = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
241 out->set("ok", true);
242 out->set("worldCount", sensing->worldCount());
243
244 Poco::JSON::Array::Ptr worlds = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
245 bool found = false;
246 for (const auto& entry : sensing->lastQueries()) {
247 if (hasIndex && entry.index != index) continue;
248 found = true;
249 Poco::JSON::Object::Ptr item = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
250 item->set("index", entry.index);
251 // The world owns this document's schema (eve.sensing.lastQuery v1); nest
252 // it as an object so the agent does not have to parse JSON-in-JSON.
253 try {
254 Poco::JSON::Parser parser;
255 Poco::Dynamic::Var parsed = parser.parse(entry.lastQueryJson);
256 item->set("query", parsed);
257 } catch (...) {
258 item->set("query", entry.lastQueryJson);
259 }
260 worlds->add(item);
261 }
262 if (hasIndex && !found) {
263 Poco::JSON::Object::Ptr missing = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
264 missing->set("ok", false);
265 missing->set("error", "no live sensing world with index " + std::to_string(index));
266 return textPayload(missing, true);
267 }
268 out->set("worlds", worlds);
269 return textPayload(out);
270}

281std::string callMcpDomainTool(std::string_view name, Poco::JSON::Object::Ptr args) {
282 if (name == "eve_decal_status") return decalStatus();
283 if (name == "eve_decal_project") return decalProject(args);
284 if (name == "eve_decal_remove") return decalRemove(args);
285 if (name == "eve_decal_clear") return decalClear();
286 if (name == "eve_decal_set_limit") return decalSetLimit(args);
287 if (name == "eve_physics_sphere_cast") return physicsSphereCast(args);
288 if (name == "eve_world_project_down") return worldProjectDown(args);
289 if (name == "eve_profiler_frame") return profilerFrame();
290 if (name == "eve_profiler_report") return profilerReport();
291 if (name == "eve_sensing_last_query") return sensingLastQuery(args);
292 return errorPayload("unknown domain tool '" + std::string(name) + "'");
293}
