35std::string canonicalValue(const eve::Value& value) {
36 switch (value.type()) {
37 case eve::Value::Type::Null: return "null";
38 case eve::Value::Type::Bool: return std::string("bool:") + (value.asBool() ? "1" : "0");
39 case eve::Value::Type::Int64: return "int:" + std::to_string(value.asInt());
40 case eve::Value::Type::Double: return "double:" + doubleToken(value.asDouble());
41 case eve::Value::Type::String: {
42 const std::string& text = value.asString();
43 std::string result = "string:";
44 appendLength(result, text);
45 return result;
46 }
47 case eve::Value::Type::Array: {
48 std::string result = "array:";
49 const auto* array = value.getIf<eve::Value::Array>();
50 result += std::to_string(array ? array->size() : 0);
51 result.push_back(':');
52 if (array) {
53 for (const eve::Value& element : *array) {
54 const std::string encoded = canonicalValue(element);
55 appendLength(result, encoded);
56 }
57 }
58 return result;
59 }
60 case eve::Value::Type::Object: {
61 std::string result = "object:";
62 const auto* object = value.getIf<eve::Value::Object>();
63 result += std::to_string(object ? object->size() : 0);
64 result.push_back(':');
65 if (object) {
66 for (const auto& [key, member] : *object) {
67 appendLength(result, key);
68 const std::string encoded = canonicalValue(member);
69 appendLength(result, encoded);
70 }
71 }
72 return result;
73 }
74 }
75 return "null";
76}

83std::optional<int> asInt(const eve::Value& value) {
84 if (const auto* integer = value.getIf<std::int64_t>()) {
85 if (*integer < std::numeric_limits<int>::lowest() || *integer > std::numeric_limits<int>::max())
86 return std::nullopt;
87 return static_cast<int>(*integer);
88 }
89 if (const auto* boolean = value.getIf<bool>()) return *boolean ? 1 : 0;
90 if (const auto* number = value.getIf<double>()) {
91 if (!std::isfinite(*number) || std::trunc(*number) != *number ||
92 *number < static_cast<double>(std::numeric_limits<int>::lowest()) ||
93 *number > static_cast<double>(std::numeric_limits<int>::max()))
94 return std::nullopt;
95 return static_cast<int>(*number);
96 }
97 return std::nullopt;
98}

100std::optional<float> asFloat(const eve::Value& value) {
101 if (const auto* boolean = value.getIf<bool>()) return *boolean ? 1.f : 0.f;
102 if (const auto* integer = value.getIf<std::int64_t>()) {
103 const float converted = static_cast<float>(*integer);
104 return std::isfinite(converted) ? std::optional<float>(converted) : std::nullopt;
105 }
106 if (const auto* number = value.getIf<double>()) {
107 if (!std::isfinite(*number)) return std::nullopt;
108 const float converted = static_cast<float>(*number);
109 return std::isfinite(converted) ? std::optional<float>(converted) : std::nullopt;
110 }
111 return std::nullopt;
112}

114std::optional<bool> asBool(const eve::Value& value) {
115 if (const auto* boolean = value.getIf<bool>()) return *boolean;
116 if (const auto* integer = value.getIf<std::int64_t>()) {
117 if (*integer == 0) return false;
118 if (*integer == 1) return true;
119 return std::nullopt;
120 }
121 if (const auto* number = value.getIf<double>()) {
122 if (!std::isfinite(*number)) return std::nullopt;
123 if (*number == 0.0) return false;
124 if (*number == 1.0) return true;
125 }
126 return std::nullopt;
127}

209std::string Params::canonicalString() const {
210 std::string result = "params:v2;";
211 // Dimensions live in their own domain so an algorithm parameter named
212 // "width" or "height" cannot collide with setSize().
213 appendField(result, "dimension", kHeightKey, eve::Value(static_cast<std::int64_t>(height_)));
214 appendField(result, "dimension", kSeedKey, eve::Value(static_cast<std::int64_t>(seed_)));
215 appendField(result, "dimension", kWidthKey, eve::Value(static_cast<std::int64_t>(width_)));
216 for (const auto& [key, value] : values_) appendField(result, "parameter", key, value);
217 return result;
218}
