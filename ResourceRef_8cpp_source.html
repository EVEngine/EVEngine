31Result<AssetRef> AssetRef::fromUri(const ResourceUri& uri) {
32 if (uri.scheme() != UriScheme::Asset || !uri.query().empty() || !uri.fragment().empty()) {
33 return parseFailure<AssetRef>("AssetRef requires an asset:// UUID without query or fragment", uri.format());
34 }
35 const auto parsed = PersistentId::parse(uri.path());
36 if (!parsed || parsed->isNil()) {
37 return parseFailure<AssetRef>("asset URI path must be a non-nil canonical UUID", uri.format());
38 }
39 return Result<AssetRef>::success(AssetRef(*parsed));

66Result<ResourceRef> ResourceRef::parse(std::string_view text) {
67 auto uri = ResourceUri::parse(text);
68 if (!uri.ok()) return statusFailure<ResourceRef>(uri.status());
69 auto value = std::move(uri).takeValue();
70 if (value.scheme() == UriScheme::Asset) {
71 auto asset = AssetRef::fromUri(value);
72 if (!asset.ok()) return statusFailure<ResourceRef>(asset.status());
73 return Result<ResourceRef>::success(ResourceRef(std::move(asset).takeValue()));
74 }
75 return Result<ResourceRef>::success(ResourceRef(std::move(value)));
76}
