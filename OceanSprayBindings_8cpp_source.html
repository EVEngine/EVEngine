45Result<OceanSpraySettings> decodeSettings(const Value& value) {
46 using namespace ocean_codec;
47 try {
48 keys(value, {"seed", "capacity", "gravity", "lifetime", "radius", "emissionPerVolumeSpeed"});
49 OceanSpraySettings s;
50 const auto& seed = field(value, "seed");
51 if (!seed.isString()) throw std::runtime_error("seed requires decimal text");
52 const auto& text = seed.asString();
53 if (text.empty() || (text.size() > 1 && text[0] == '0')) throw std::runtime_error("noncanonical seed");
54 auto parsed = std::from_chars(text.data(), text.data() + text.size(), s.seed);
55 if (parsed.ec != std::errc{} || parsed.ptr != text.data() + text.size())
56 throw std::runtime_error("seed outside uint64");
57 const auto& capacity = field(value, "capacity");
58 if (!capacity.isInt64() || capacity.asInt() < 1 || capacity.asInt() > 16384)
59 throw std::runtime_error("capacity range");
60 s.capacity = static_cast<unsigned>(capacity.asInt());
61 s.gravity = number(field(value, "gravity"));
62 s.lifetime = number(field(value, "lifetime"));
63 s.radius = number(field(value, "radius"));
64 s.emissionPerVolumeSpeed = number(field(value, "emissionPerVolumeSpeed"));
65 auto checked = OceanSprayFrame::create(s);
66 if (!checked) return Result<OceanSpraySettings>::failure(checked.status());
67 return Result<OceanSpraySettings>::success(s);
68 } catch (const std::runtime_error& e) {
69 return Result<OceanSpraySettings>::failure(invalid(e.what()));
70 }
71}
72Result<OceanSprayView> decodeView(const ssq::Object& object) {
73 auto v = script::valueFromSquirrel(object);
74 if (!v) return Result<OceanSprayView>::failure(v.status());
75 try {
76 using namespace ocean_codec;
77 keys(v.value(), {"eye", "forward", "up", "origin"});
78 return Result<OceanSprayView>::success({vector3(field(v.value(), "eye")), vector3(field(v.value(), "forward")),
79 vector3(field(v.value(), "up")), vector3(field(v.value(), "origin"))});
80 } catch (const std::runtime_error& e) {
81 return Result<OceanSprayView>::failure(invalid(e.what()));
82 }
83}
