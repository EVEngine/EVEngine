28ContinuousArtOutput ContinuousArt::stylize(const FogRayResult& physical, float viewZ,
29 float neighborOpacity) const noexcept {
30 ContinuousArtOutput out;
31 const float opacity = std::clamp(1.f - physical.transmittance, 0.f, 1.f);
32 out.opacity = opacity;
33 out.color = physical.inScatter * params_.colorTint;
34
35 const float depthCue = 1.f - std::exp(-std::max(viewZ, 0.f) * 0.05f);
36 out.edgeMask = std::clamp(std::fabs(opacity - neighborOpacity) * (0.5f + params_.edgeBoost), 0.f, 1.f);
37 out.edgeMask *= std::clamp(1.f - params_.silhouetteSoftness * depthCue, 0.f, 1.f);
38
39 const float glow = params_.internalGlow * opacity * (0.35f + 0.65f * depthCue);
40 out.color += glm::vec3(glow) * params_.colorTint;
41
42 // Silver dust: cool highlight on bright in-scatter without touching optics.
43 const float luma = 0.2126f * physical.inScatter.r + 0.7152f * physical.inScatter.g + 0.0722f * physical.inScatter.b;
44 out.dustHighlight = std::clamp(params_.silverDust * luma, 0.f, 1.f);
45 out.color += glm::vec3(0.85f, 0.9f, 1.f) * out.dustHighlight * 0.15f;
46 return out;
47}

55[[nodiscard]] float approach(float current, float target, float maxDelta) noexcept {
56 const float delta = target - current;
57 if (std::fabs(delta) <= maxDelta) return target;
58 return current + std::copysign(maxDelta, delta);
59}

61[[nodiscard]] glm::vec3 approach3(const glm::vec3& current, const glm::vec3& target, float maxDelta) noexcept {
62 return {approach(current.x, target.x, maxDelta), approach(current.y, target.y, maxDelta),
63 approach(current.z, target.z, maxDelta)};

128glm::vec3 SceneWind::sample(const glm::vec3& world, float timeSeconds) const noexcept {
129 const float t = timeSeconds * curlTimeScale_;
130 const float f = curlFrequency_;
131 // Analytic curl of a smooth potential → locally swirling, nearly div-free.
132 const float sx = std::sin(f * world.y + 0.7f * t);
133 const float sy = std::sin(f * world.z + 1.1f * t);
134 const float sz = std::sin(f * world.x + 0.5f * t);
135 const float cx = std::cos(f * world.z + 0.9f * t);
136 const float cy = std::cos(f * world.x + 0.3f * t);
137 const float cz = std::cos(f * world.y + 1.3f * t);
138 const glm::vec3 curl(currentCurl_ * (sy - cz), currentCurl_ * (sz - cx), currentCurl_ * (sx - cy));
139 return currentMain_ + curl;
140}
