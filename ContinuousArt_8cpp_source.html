31ContinuousArtOutput ContinuousArt::stylize(const FogRayResult& physical, float viewZ,
32 float neighborOpacity) const noexcept {
33 ContinuousArtOutput out;
34 const float opacity = std::clamp(1.f - physical.transmittance, 0.f, 1.f);
35 out.opacity = opacity;
36 out.color = physical.inScatter * params_.colorTint;
37
38 const float depthCue = 1.f - std::exp(-std::max(viewZ, 0.f) * 0.05f);
39 out.edgeMask = std::clamp(std::fabs(opacity - neighborOpacity) * (0.5f + params_.edgeBoost), 0.f,
40 1.f);
41 out.edgeMask *= std::clamp(1.f - params_.silhouetteSoftness * depthCue, 0.f, 1.f);
42
43 const float glow = params_.internalGlow * opacity * (0.35f + 0.65f * depthCue);
44 out.color += glm::vec3(glow) * params_.colorTint;
45
46 // Silver dust: cool highlight on bright in-scatter without touching optics.
47 const float luma =
48 0.2126f * physical.inScatter.r + 0.7152f * physical.inScatter.g + 0.0722f * physical.inScatter.b;
49 out.dustHighlight = std::clamp(params_.silverDust * luma, 0.f, 1.f);
50 out.color += glm::vec3(0.85f, 0.9f, 1.f) * out.dustHighlight * 0.15f;
51 return out;
52}
