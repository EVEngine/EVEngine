181int Weapon::registerWeaponsFromJson(const std::string& json) {
182 auto canonical = eve::Value::fromJson(json);
183 if (!canonical) {
184 canonical.ignore("weapon definition input could not be parsed as canonical Value");
185 return 0;
186 }
187
188 int n = 0;
189 const auto publish = [this, &n](const eve::Value& value) {
190 const auto* object = value.getIf<eve::Value::Object>();
191 if (object == nullptr) return;
192 const auto id = object->find("id");
193 const auto* name = id == object->end() ? nullptr : id->second.getIf<std::string>();
194 if (name == nullptr || name->empty()) return;
195
196 auto encoded = value.toJson();
197 if (!encoded) {
198 encoded.ignore("weapon definition value could not be serialized canonically");
199 return;
200 }
201 eve::definitions::Definition source;
202 source.type = "weapon";
203 source.id = *name;
204 source.version = eve::SchemaVersion(1);
205 source.json = encoded.value();
206 auto typed = parseWeaponDefinition(source);
207 if (!typed) {
208 typed.ignore("weapon definition failed typed projection validation");
209 return;
210 }
211
212 auto current = definitionRegistry_.resolve("weapon", *name);
213 const bool exists = current.ok();
214 if (!exists && current.status().code() != eve::StatusCode::NotFound) return;
215 auto stored = exists ? definitionRegistry_.replace("weapon", *name, 1, source.json)
216 : definitionRegistry_.insert("weapon", *name, 1, source.json);
217 if (stored) {
218 ++n;
219 } else {
220 stored.ignore("weapon definition registry rejected canonical registration");
221 }
222 };
223 const eve::Value& root = canonical.value();
224 if (const auto* array = root.getIf<eve::Value::Array>()) {
225 for (const auto& value : *array) publish(value);
226 } else {
227 publish(root);
228 }
229 return n;
230}

286WeaponEntity* Weapon::newWeapon(const std::string& defId) {
287 auto definition = findDef(defId);
288 if (!definition) {
289 definition.ignore("legacy nullable factory returns null for an unknown weapon definition");
290 return nullptr;
291 }
292 WeaponDefinition d = std::move(definition).takeValue();
293
294 WeaponEntity* w = WeaponEntity::createWeapon();
295 w->identity()->id = defId + "#" + std::to_string(nextInstance_++);
296 w->identity()->defId = defId;
297 w->definition()->owned = std::make_shared<const WeaponDefinition>(std::move(d));
298 w->definition()->def = w->definition()->owned.get();
299 w->state()->stages = &w->definition()->def->stages;
300 const WeaponDefinition& storedDefinition = *w->definition()->def;
301 w->state()->selector = storedDefinition.fireMode;
302 w->state()->currentSpread = storedDefinition.spreadMin;
303
304 // 运行时资源：热武器 ammo 映射到弹匣/备用；其余形态从模板 resource 拷贝。
305 Resource& r = w->state()->resource;
306 r.kind = storedDefinition.resource.kind;
307 r.max = storedDefinition.resource.max;
308 r.regen = storedDefinition.resource.regen;
309 r.cost = storedDefinition.resource.cost;
310 r.infinite = storedDefinition.resource.infinite;
311 if (storedDefinition.kind == WeaponKind::Ranged) {
312 r.kind = ResourceKind::Ammo;
313 r.max = static_cast<float>(storedDefinition.magSize);
314 r.value = static_cast<float>(storedDefinition.magSize);
315 r.cost = 1.f;
316 r.reserve = storedDefinition.reserveSize < 0 ? 0 : storedDefinition.reserveSize;
317 r.infinite = storedDefinition.reserveSize < 0;
318 } else {
319 r.value = r.infinite ? 0.f : r.max;
320 }
321
322 weapons_.push_back(ecs::handle_of(w));
323 return w;
324}
