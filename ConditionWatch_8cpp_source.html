17eve::Result<void> walk(const decision::Condition& condition, std::set<std::string>& out) {
18 using K = decision::ConditionKind;
19 switch (condition.kind()) {
20 case K::All:
21 case K::Any:
22 for (const auto& child : condition.children()) {
23 auto nested = walk(child, out);
24 if (!nested) return nested;
25 }
26 return eve::Result<void>::success();
27 case K::Not:
28 if (condition.children().empty())
29 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
30 "not condition requires a child", "children",
31 {}, "emergence.watch"));
32 return walk(condition.children().front(), out);
33 case K::Compare:
34 insertUnique(out, makeFactKey(FactDomain::Value, condition.key()));
35 return eve::Result<void>::success();
36 case K::HasTag:
37 insertUnique(out, makeFactKey(FactDomain::Tag, condition.key()));
38 return eve::Result<void>::success();
39 case K::HasAttribute:
40 insertUnique(out, makeFactKey(FactDomain::Attribute, condition.key()));
41 return eve::Result<void>::success();
42 case K::HasResource:
43 insertUnique(out, makeFactKey(FactDomain::Resource, condition.key()));
44 return eve::Result<void>::success();
45 case K::StateEquals:
46 insertUnique(out, makeFactKey(FactDomain::State, condition.key()));
47 return eve::Result<void>::success();
48 case K::AuthorityCheck:
49 insertUnique(out, makeFactKey(FactDomain::Authority, condition.key()));
50 return eve::Result<void>::success();
51 case K::PolicyCall: {
52 insertUnique(out, makeFactKey(FactDomain::Policy, condition.key()));
53 const auto& declaration = condition.scriptDeclaration();
54 if (declaration) {
55 for (const auto& dependency : declaration->dependencies) {
56 auto parsed = parseFactKey(dependency);
57 if (!parsed)

59 eve::DiagnosticCode::InvalidArgument,

61 "emergence.watch"));
62 insertUnique(out, dependency);
63 }
64 }
65 if (out.empty() || (declaration && declaration->dependencies.empty() && condition.key().empty())) {
66 return eve::Result<void>::failure(eve::Diagnostic::error(
67 eve::DiagnosticCode::InvalidArgument,
68 "policy_call conditions require a policy name or declared dependencies for indexing", "policy", {},
69 "emergence.watch"));
70 }
71 return eve::Result<void>::success();
72 }

82 if (!condition.isValid())
83 return eve::Result<std::vector<std::string>>::failure(eve::Diagnostic::error(
84 eve::DiagnosticCode::InvalidArgument, "condition tree is invalid", "condition", {}, "emergence.watch"));
85 std::set<std::string> unique;
