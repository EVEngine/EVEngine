29 if (!owned.ok()) return script::projectStatusResult(vm, owned.status());
30 auto result = script::projectStatusResult(vm, Status::success(), std::move(owned).takeValue());
31 result.set("ownership", std::string("owned"));
32 return result;
33 });
34 cls.addFunc("restore", [vm](VegetationField* self, ssq::Object document) {

38 return script::projectResult(vm, self->restore(input.value()));
39 });
40 cls.addFunc("snapshot", [vm](VegetationField* self) {
41 if (!self) return script::projectResult(vm, missing());

43 });
44 cls.addFunc("sample", [vm](VegetationField* self, float x, float y, float z, int color, int extras, int motion,
