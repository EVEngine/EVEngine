210Result<void> ActionTimeline::validate() const {
211 if (schemaVersion.value() != kActionTimelineSchemaVersion)
212 return invalid("unsupported action timeline schema version", "schemaVersion");
213 if (!actionId.isValid()) return invalid("action timeline requires a valid action id", "actionId");
214 if (duration.nanoseconds() < 0) return invalid("timeline duration must be non-negative", "durationNs");
215 if (!std::isfinite(montage.basePlayRate) || montage.basePlayRate <= 0.0)
216 return invalid("montage base play rate must be positive and finite", "montage.basePlayRate");
217 if (montage.defaultBlendIn < Duration::zero() || montage.defaultBlendOut < Duration::zero() ||
218 montage.blendOutOffset < Duration::zero())
219 return invalid("montage blend durations must be non-negative", "montage");
220 for (std::size_t index = 0; index < splitTimestamps.size(); ++index) {
221 if (splitTimestamps[index] <= Duration::zero() || splitTimestamps[index] >= duration ||
222 (index > 0 && splitTimestamps[index - 1] >= splitTimestamps[index]))
223 return invalid("physical section splits must be strictly ordered inside the timeline",
224 "splitTimestampsNs[" + std::to_string(index) + "]");
225 }
226
227 std::set<std::string> ids;
228 for (std::size_t index = 0; index < animationSections.size(); ++index) {
229 const auto& section = animationSections[index];
230 const std::string path = "animationSections[" + std::to_string(index) + "]";
231 if (!section.id.isValid()) return invalid("animation section id is invalid", path + ".id");
232 if (!ids.insert(section.id.format()).second) return invalid("timeline item ids must be unique", path + ".id");
233 if (section.animationUri.empty()) return invalid("animation section requires a URI", path + ".animationUri");
234 if (section.start < Duration::zero() || section.end <= section.start || section.end > duration)
235 return invalid("animation section range is invalid", path);
236 if (section.blendIn < Duration::zero() ||
237 section.blendIn.nanoseconds() > section.end.nanoseconds() - section.start.nanoseconds())
238 return invalid("animation section blend-in is outside its range", path + ".blendInNs");
239 if (section.sourceStart < Duration::zero() || section.sourceEnd < Duration::zero() ||
240 (!section.sourceEnd.isZero() && section.sourceEnd <= section.sourceStart))
241 return invalid("animation section source trim is invalid", path + ".sourceStartNs");
242 if (index > 0) {
243 const auto& previous = animationSections[index - 1];
244 if (previous.start > section.start)
245 return invalid("animation sections must be sorted by start time", path + ".startNs");
246 if (previous.end > section.start) return invalid("animation sections must not overlap", path + ".startNs");
247 }
248 }
249 for (std::size_t trackIndex = 0; trackIndex < tracks.size(); ++trackIndex) {
250 const auto& track = tracks[trackIndex];
251 const std::string path = "tracks[" + std::to_string(trackIndex) + "]";
252 if (!track.id.isValid()) return invalid("track id is invalid", path + ".id");
253 if (!ids.insert(track.id.format()).second) return invalid("timeline item ids must be unique", path + ".id");
254 for (std::size_t index = 0; index < track.notifies.size(); ++index) {
255 const auto& notify = track.notifies[index];
256 const std::string itemPath = path + ".notifies[" + std::to_string(index) + "]";
257 if (!notify.id.isValid() || !notify.type.isValid()) return invalid("notify ids are invalid", itemPath);
258 if (!ids.insert(notify.id.format()).second)
259 return invalid("timeline item ids must be unique", itemPath + ".id");
260 if (notify.time < Duration::zero() || notify.time > duration)
261 return invalid("notify time is outside the timeline", itemPath + ".timeNs");
262 if (index > 0 && track.notifies[index - 1].time > notify.time)
263 return invalid("notifies must be sorted by time", itemPath + ".timeNs");
264 }
265 for (std::size_t index = 0; index < track.states.size(); ++index) {
266 const auto& state = track.states[index];
267 const std::string itemPath = path + ".states[" + std::to_string(index) + "]";
268 if (!state.id.isValid() || !state.type.isValid()) return invalid("notify-state ids are invalid", itemPath);
269 if (!ids.insert(state.id.format()).second)
270 return invalid("timeline item ids must be unique", itemPath + ".id");
271 if (state.start < Duration::zero() || state.end <= state.start || state.end > duration)
272 return invalid("notify-state range is invalid", itemPath);
273 if (index > 0 && track.states[index - 1].start > state.start)
274 return invalid("notify states must be sorted by start time", itemPath + ".startNs");
275 }
276 }
277 return Result<void>::success();
278}

280Result<std::vector<ActionTimelineEvent>> ActionTimeline::sample(Duration previous, Duration current,
281 bool includePrevious) const {
282 auto valid = validate();
283 if (!valid) return Result<std::vector<ActionTimelineEvent>>::failure(valid.status());
284 if (previous < Duration::zero() || current < previous || current > duration)
285 return invalid<std::vector<ActionTimelineEvent>>("sample range is outside the timeline", "sample");
286
287 std::vector<ActionTimelineEvent> out;
288 for (const auto& track : tracks) {
289 if (track.muted) continue;
290 for (const auto& notify : track.notifies) {
291 if (!notify.enabled) continue;
292 if (inWindow(notify.time, previous, current, includePrevious))
293 out.push_back(
294 {ActionTimelineEventKind::Notify, track.id, notify.id, notify.type, notify.time, notify.payload});
295 }
296 for (const auto& state : track.states) {
297 if (!state.enabled) continue;
298 if (inWindow(state.start, previous, current, includePrevious))
299 out.push_back(
300 {ActionTimelineEventKind::StateEnter, track.id, state.id, state.type, state.start, state.payload});
301 if (inWindow(state.end, previous, current, includePrevious))
302 out.push_back(
303 {ActionTimelineEventKind::StateExit, track.id, state.id, state.type, state.end, state.payload});
304 }
305 }
306 std::stable_sort(out.begin(), out.end(), [](const auto& left, const auto& right) {
307 if (left.time != right.time) return left.time < right.time;
308 if (left.trackId != right.trackId) return left.trackId.format() < right.trackId.format();
309 if (left.itemId != right.itemId) return left.itemId.format() < right.itemId.format();
310 return static_cast<std::uint8_t>(left.kind) < static_cast<std::uint8_t>(right.kind);
311 });
312 return Result<std::vector<ActionTimelineEvent>>::success(std::move(out));
313}

315Result<std::vector<ActionActiveBlock>> ActionTimeline::activeBlocks(Duration time) const {
316 auto valid = validate();
317 if (!valid) return Result<std::vector<ActionActiveBlock>>::failure(valid.status());
318 if (time < Duration::zero() || time > duration)
319 return invalid<std::vector<ActionActiveBlock>>("active-block sample is outside the timeline", "time");
320
321 std::vector<ActionActiveBlock> out;
322 for (const auto& track : tracks) {
323 if (track.muted) continue;
324 for (const auto& state : track.states) {
325 if (!state.enabled) continue;
326 if (time < state.start || time >= state.end) continue;
327 out.push_back({track.id,
328 state.id,
329 state.type,
330 Duration::fromNanoseconds(time.nanoseconds() - state.start.nanoseconds()),
331 Duration::fromNanoseconds(state.end.nanoseconds() - state.start.nanoseconds()),
332 state.payload});
333 }
334 }
335 std::stable_sort(out.begin(), out.end(), [](const auto& left, const auto& right) {
336 if (left.trackId != right.trackId) return left.trackId.format() < right.trackId.format();
337 return left.itemId.format() < right.itemId.format();
338 });
339 return Result<std::vector<ActionActiveBlock>>::success(std::move(out));
340}

342Result<Value> ActionTimeline::toValue() const {
343 auto valid = validate();
344 if (!valid) return Result<Value>::failure(valid.status());
345 Value::Array encodedTracks;
346 Value::Array encodedSections;
347 Value::Array encodedSplits;
348 encodedSections.reserve(animationSections.size());
349 for (const auto& section : animationSections) encodedSections.push_back(encodeAnimationSection(section));
350 encodedSplits.reserve(splitTimestamps.size());
351 for (const Duration split : splitTimestamps) encodedSplits.emplace_back(split.nanoseconds());
352 encodedTracks.reserve(tracks.size());
353 for (const auto& track : tracks) {
354 Value::Array notifies;
355 Value::Array states;
356 for (const auto& notify : track.notifies) notifies.push_back(encodeNotify(notify));
357 for (const auto& state : track.states) states.push_back(encodeState(state));
358 Value::Object object;
359 object["id"] = track.id.format();
360 object["label"] = track.label;
361 object["kind"] = std::string(actionTrackKindName(track.kind));
362 object["muted"] = track.muted;
363 object["locked"] = track.locked;
364 object["notifies"] = Value(std::move(notifies));
365 object["states"] = Value(std::move(states));
366 encodedTracks.emplace_back(std::move(object));
367 }
368 Value::Object root;
369 root["schema"] = std::string(kActionTimelineSchemaId);
370 root["schemaVersion"] = static_cast<std::int64_t>(schemaVersion.value());
371 root["actionId"] = actionId.format();
372 root["durationNs"] = duration.nanoseconds();
373 root["animationUri"] = animationSections.empty() ? animationUri : animationSections.front().animationUri;
374 root["animationSections"] = Value(std::move(encodedSections));
375 root["splitTimestampsNs"] = Value(std::move(encodedSplits));
376 root["montage"] = Value::Object{{"basePlayRate", montage.basePlayRate},
377 {"looping", montage.looping},
378 {"footIk", montage.footIk},
379 {"animationLayer", static_cast<std::int64_t>(montage.animationLayer)},
380 {"defaultBlendInNs", montage.defaultBlendIn.nanoseconds()},
381 {"defaultBlendOutNs", montage.defaultBlendOut.nanoseconds()},
382 {"blendOutOffsetNs", montage.blendOutOffset.nanoseconds()},
383 {"rootMotionHorizontal", montage.rootMotionHorizontal},
384 {"rootMotionVertical", montage.rootMotionVertical},
385 {"rootMotionRotation", montage.rootMotionRotation}};
386 root["tracks"] = Value(std::move(encodedTracks));
387 root["metadata"] = Value(metadata);
388 return Result<Value>::success(Value(std::move(root)));
389}

391Result<ActionTimeline> ActionTimeline::fromValue(const Value& value) {
392 if (!value.isObject()) return invalid<ActionTimeline>("expected action timeline object", "timeline");
393 auto schema = stringField(value, "schema", "schema");
394 if (!schema) return Result<ActionTimeline>::failure(schema.status());
395 if (schema.value() != kActionTimelineSchemaId)
396 return invalid<ActionTimeline>("unexpected action timeline schema", "schema");
397 auto version = intField(value, "schemaVersion", "schemaVersion");
398 if (!version) return Result<ActionTimeline>::failure(version.status());
399 auto actionId = logicalIdField(value, "actionId", "actionId");
400 if (!actionId) return Result<ActionTimeline>::failure(actionId.status());
401 auto duration = intField(value, "durationNs", "durationNs");
402 if (!duration) return Result<ActionTimeline>::failure(duration.status());
403 auto animationUri = stringField(value, "animationUri", "animationUri");
404 if (!animationUri) return Result<ActionTimeline>::failure(animationUri.status());
405 const Value* tracksValue = field(value, "tracks");
406 const Value* metadataValue = field(value, "metadata");
407 if (!tracksValue || !tracksValue->isArray()) return invalid<ActionTimeline>("expected tracks array", "tracks");
408 if (!metadataValue || !metadataValue->isObject())
409 return invalid<ActionTimeline>("expected metadata object", "metadata");
410
411 ActionTimeline candidate;
412 if (version.value() < 1 || version.value() > static_cast<std::int64_t>(kActionTimelineSchemaVersion))
413 return invalid<ActionTimeline>("unsupported action timeline schema version", "schemaVersion");
414 candidate.schemaVersion = SchemaVersion(kActionTimelineSchemaVersion);
415 candidate.actionId = std::move(actionId).takeValue();
416 candidate.duration = Duration::fromNanoseconds(duration.value());
417 candidate.animationUri = std::move(animationUri).takeValue();
418 candidate.metadata = *metadataValue->getIf<Value::Object>();
419 if (version.value() == 1) {
420 if (!candidate.animationUri.empty() && !candidate.duration.isZero()) {
421 candidate.animationSections.push_back({*LogicalId::fromParts("animation-section", "legacy"),
422 candidate.animationUri, Duration::zero(), candidate.duration,
423 Duration::zero()});
424 }
425 } else {
426 const Value* sectionsValue = field(value, "animationSections");
427 if (!sectionsValue || !sectionsValue->isArray())
428 return invalid<ActionTimeline>("expected animationSections array", "animationSections");
429 const auto& sections = *sectionsValue->getIf<Value::Array>();
430 for (std::size_t index = 0; index < sections.size(); ++index) {
431 auto decoded = decodeAnimationSection(sections[index], "animationSections[" + std::to_string(index) + "]",
432 version.value());
433 if (!decoded) return Result<ActionTimeline>::failure(decoded.status());
434 candidate.animationSections.push_back(std::move(decoded).takeValue());
435 }
436 if (!candidate.animationSections.empty())
437 candidate.animationUri = candidate.animationSections.front().animationUri;
438 if (version.value() >= 3) {
439 const Value* splitsValue = field(value, "splitTimestampsNs");
440 if (!splitsValue || !splitsValue->isArray())
441 return invalid<ActionTimeline>("expected splitTimestampsNs array", "splitTimestampsNs");
442 const auto& splits = *splitsValue->getIf<Value::Array>();
443 for (std::size_t index = 0; index < splits.size(); ++index) {
444 if (!splits[index].isInt64())
445 return invalid<ActionTimeline>("expected physical section timestamp",
446 "splitTimestampsNs[" + std::to_string(index) + "]");
447 candidate.splitTimestamps.push_back(Duration::fromNanoseconds(splits[index].asInt()));
448 }
449 const Value* montageValue = field(value, "montage");
450 if (!montageValue || !montageValue->isObject())
451 return invalid<ActionTimeline>("expected montage settings object", "montage");
452 const Value* basePlayRate = field(*montageValue, "basePlayRate");
453 auto looping = boolField(*montageValue, "looping", "montage.looping");

481 const Value& trackValue = tracks[trackIndex];

485 auto label = stringField(trackValue, "label", path + ".label");

490 if (!label) return Result<ActionTimeline>::failure(label.status());
491 if (!kindText) return Result<ActionTimeline>::failure(kindText.status());
492 if (!muted) return Result<ActionTimeline>::failure(muted.status());
493 if (!locked) return Result<ActionTimeline>::failure(locked.status());

497 const Value* states = field(trackValue, "states");
498 if (!notifies || !notifies->isArray())
499 return invalid<ActionTimeline>("expected notifies array", path + ".notifies");
500 if (!states || !states->isArray()) return invalid<ActionTimeline>("expected states array", path + ".states");
501 ActionTrack track;
502 track.id = std::move(id).takeValue();
503 track.label = std::move(label).takeValue();
504 track.kind = std::move(kind).takeValue();
505 track.muted = muted.value();
506 track.locked = locked.value();
507 const auto& notifyValues = *notifies->getIf<Value::Array>();
508 for (std::size_t index = 0; index < notifyValues.size(); ++index) {
509 auto decoded = decodeNotify(notifyValues[index], path + ".notifies[" + std::to_string(index) + "]",
510 version.value());

512 track.notifies.push_back(std::move(decoded).takeValue());

517 version.value());

519 track.states.push_back(std::move(decoded).takeValue());
520 }
521 candidate.tracks.push_back(std::move(track));
522 }
523 auto valid = candidate.validate();
524 if (!valid) return Result<ActionTimeline>::failure(valid.status());
525 return Result<ActionTimeline>::success(std::move(candidate));
526}
