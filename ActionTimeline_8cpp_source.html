136Result<void> ActionTimeline::validate() const {
137 if (schemaVersion.value() != kActionTimelineSchemaVersion)
138 return invalid("unsupported action timeline schema version", "schemaVersion");
139 if (!actionId.isValid()) return invalid("action timeline requires a valid action id", "actionId");
140 if (duration.nanoseconds() < 0) return invalid("timeline duration must be non-negative", "durationNs");
141
142 std::set<std::string> ids;
143 for (std::size_t trackIndex = 0; trackIndex < tracks.size(); ++trackIndex) {
144 const auto& track = tracks[trackIndex];
145 const std::string path = "tracks[" + std::to_string(trackIndex) + "]";
146 if (!track.id.isValid()) return invalid("track id is invalid", path + ".id");
147 if (!ids.insert(track.id.format()).second) return invalid("timeline item ids must be unique", path + ".id");
148 for (std::size_t index = 0; index < track.notifies.size(); ++index) {
149 const auto& notify = track.notifies[index];
150 const std::string itemPath = path + ".notifies[" + std::to_string(index) + "]";
151 if (!notify.id.isValid() || !notify.type.isValid()) return invalid("notify ids are invalid", itemPath);
152 if (!ids.insert(notify.id.format()).second)
153 return invalid("timeline item ids must be unique", itemPath + ".id");
154 if (notify.time < Duration::zero() || notify.time > duration)
155 return invalid("notify time is outside the timeline", itemPath + ".timeNs");
156 if (index > 0 && track.notifies[index - 1].time > notify.time)
157 return invalid("notifies must be sorted by time", itemPath + ".timeNs");
158 }
159 for (std::size_t index = 0; index < track.states.size(); ++index) {
160 const auto& state = track.states[index];
161 const std::string itemPath = path + ".states[" + std::to_string(index) + "]";
162 if (!state.id.isValid() || !state.type.isValid()) return invalid("notify-state ids are invalid", itemPath);
163 if (!ids.insert(state.id.format()).second)
164 return invalid("timeline item ids must be unique", itemPath + ".id");
165 if (state.start < Duration::zero() || state.end <= state.start || state.end > duration)
166 return invalid("notify-state range is invalid", itemPath);
167 if (index > 0 && track.states[index - 1].start > state.start)
168 return invalid("notify states must be sorted by start time", itemPath + ".startNs");
169 }
170 }
171 return Result<void>::success();
172}

174Result<std::vector<ActionTimelineEvent>> ActionTimeline::sample(Duration previous, Duration current,
175 bool includePrevious) const {
176 auto valid = validate();
177 if (!valid) return Result<std::vector<ActionTimelineEvent>>::failure(valid.status());
178 if (previous < Duration::zero() || current < previous || current > duration)
179 return invalid<std::vector<ActionTimelineEvent>>("sample range is outside the timeline", "sample");
180
181 std::vector<ActionTimelineEvent> out;
182 for (const auto& track : tracks) {
183 if (track.muted) continue;
184 for (const auto& notify : track.notifies)
185 if (inWindow(notify.time, previous, current, includePrevious))
186 out.push_back(
187 {ActionTimelineEventKind::Notify, track.id, notify.id, notify.type, notify.time, notify.payload});
188 for (const auto& state : track.states) {
189 if (inWindow(state.start, previous, current, includePrevious))
190 out.push_back(
191 {ActionTimelineEventKind::StateEnter, track.id, state.id, state.type, state.start, state.payload});
192 if (inWindow(state.end, previous, current, includePrevious))
193 out.push_back(
194 {ActionTimelineEventKind::StateExit, track.id, state.id, state.type, state.end, state.payload});
195 }
196 }
197 std::stable_sort(out.begin(), out.end(), [](const auto& left, const auto& right) {
198 if (left.time != right.time) return left.time < right.time;
199 if (left.trackId != right.trackId) return left.trackId.format() < right.trackId.format();
200 if (left.itemId != right.itemId) return left.itemId.format() < right.itemId.format();
201 return static_cast<std::uint8_t>(left.kind) < static_cast<std::uint8_t>(right.kind);
202 });
203 return Result<std::vector<ActionTimelineEvent>>::success(std::move(out));
204}

206Result<Value> ActionTimeline::toValue() const {
207 auto valid = validate();
208 if (!valid) return Result<Value>::failure(valid.status());
209 Value::Array encodedTracks;
210 encodedTracks.reserve(tracks.size());
211 for (const auto& track : tracks) {
212 Value::Array notifies;
213 Value::Array states;
214 for (const auto& notify : track.notifies) notifies.push_back(encodeNotify(notify));
215 for (const auto& state : track.states) states.push_back(encodeState(state));
216 Value::Object object;
217 object["id"] = track.id.format();
218 object["label"] = track.label;
219 object["kind"] = std::string(actionTrackKindName(track.kind));
220 object["muted"] = track.muted;
221 object["locked"] = track.locked;
222 object["notifies"] = Value(std::move(notifies));
223 object["states"] = Value(std::move(states));
224 encodedTracks.emplace_back(std::move(object));
225 }
226 Value::Object root;
227 root["schema"] = std::string(kActionTimelineSchemaId);
228 root["schemaVersion"] = static_cast<std::int64_t>(schemaVersion.value());
229 root["actionId"] = actionId.format();
230 root["durationNs"] = duration.nanoseconds();
231 root["animationUri"] = animationUri;
232 root["tracks"] = Value(std::move(encodedTracks));
233 root["metadata"] = Value(metadata);
234 return Result<Value>::success(Value(std::move(root)));
235}

237Result<ActionTimeline> ActionTimeline::fromValue(const Value& value) {
238 if (!value.isObject()) return invalid<ActionTimeline>("expected action timeline object", "timeline");
239 auto schema = stringField(value, "schema", "schema");
240 if (!schema) return Result<ActionTimeline>::failure(schema.status());
241 if (schema.value() != kActionTimelineSchemaId)
242 return invalid<ActionTimeline>("unexpected action timeline schema", "schema");
243 auto version = intField(value, "schemaVersion", "schemaVersion");
244 if (!version) return Result<ActionTimeline>::failure(version.status());
245 auto actionId = logicalIdField(value, "actionId", "actionId");
246 if (!actionId) return Result<ActionTimeline>::failure(actionId.status());
247 auto duration = intField(value, "durationNs", "durationNs");
248 if (!duration) return Result<ActionTimeline>::failure(duration.status());
249 auto animationUri = stringField(value, "animationUri", "animationUri");
250 if (!animationUri) return Result<ActionTimeline>::failure(animationUri.status());

252 const Value* metadataValue = field(value, "metadata");
253 if (!tracksValue || !tracksValue->isArray()) return invalid<ActionTimeline>("expected tracks array", "tracks");
254 if (!metadataValue || !metadataValue->isObject())
255 return invalid<ActionTimeline>("expected metadata object", "metadata");
256
257 ActionTimeline candidate;
258 candidate.schemaVersion = SchemaVersion(static_cast<std::uint64_t>(version.value()));
259 candidate.actionId = std::move(actionId).takeValue();
260 candidate.duration = Duration::fromNanoseconds(duration.value());
261 candidate.animationUri = std::move(animationUri).takeValue();
262 candidate.metadata = *metadataValue->getIf<Value::Object>();
263 const auto& tracks = *tracksValue->getIf<Value::Array>();
264 for (std::size_t trackIndex = 0; trackIndex < tracks.size(); ++trackIndex) {
265 const Value& trackValue = tracks[trackIndex];
266 const std::string path = "tracks[" + std::to_string(trackIndex) + "]";
267 if (!trackValue.isObject()) return invalid<ActionTimeline>("expected track object", path);
268 auto id = logicalIdField(trackValue, "id", path + ".id");
269 auto label = stringField(trackValue, "label", path + ".label");
270 auto kindText = stringField(trackValue, "kind", path + ".kind");
271 auto muted = boolField(trackValue, "muted", path + ".muted");
272 auto locked = boolField(trackValue, "locked", path + ".locked");
273 if (!id) return Result<ActionTimeline>::failure(id.status());
274 if (!label) return Result<ActionTimeline>::failure(label.status());
275 if (!kindText) return Result<ActionTimeline>::failure(kindText.status());
276 if (!muted) return Result<ActionTimeline>::failure(muted.status());
277 if (!locked) return Result<ActionTimeline>::failure(locked.status());
278 auto kind = parseActionTrackKind(kindText.value());
279 if (!kind) return Result<ActionTimeline>::failure(kind.status());
280 const Value* notifies = field(trackValue, "notifies");
281 const Value* states = field(trackValue, "states");
282 if (!notifies || !notifies->isArray())
283 return invalid<ActionTimeline>("expected notifies array", path + ".notifies");
284 if (!states || !states->isArray()) return invalid<ActionTimeline>("expected states array", path + ".states");
285 ActionTrack track;
286 track.id = std::move(id).takeValue();
287 track.label = std::move(label).takeValue();
288 track.kind = std::move(kind).takeValue();
289 track.muted = muted.value();
290 track.locked = locked.value();
291 const auto& notifyValues = *notifies->getIf<Value::Array>();
292 for (std::size_t index = 0; index < notifyValues.size(); ++index) {
293 auto decoded = decodeNotify(notifyValues[index], path + ".notifies[" + std::to_string(index) + "]");
294 if (!decoded) return Result<ActionTimeline>::failure(decoded.status());
295 track.notifies.push_back(std::move(decoded).takeValue());
296 }
297 const auto& stateValues = *states->getIf<Value::Array>();
298 for (std::size_t index = 0; index < stateValues.size(); ++index) {
299 auto decoded = decodeState(stateValues[index], path + ".states[" + std::to_string(index) + "]");
300 if (!decoded) return Result<ActionTimeline>::failure(decoded.status());
301 track.states.push_back(std::move(decoded).takeValue());
302 }
303 candidate.tracks.push_back(std::move(track));
304 }
305 auto valid = candidate.validate();
306 if (!valid) return Result<ActionTimeline>::failure(valid.status());
307 return Result<ActionTimeline>::success(std::move(candidate));
308}
