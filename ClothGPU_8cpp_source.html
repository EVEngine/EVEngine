23// One thread owns one particle: integrate, then relax its fixed-size link list.
24// Push constant layout (float indices):
25// 0 dt | 1 gx | 2 gy | 3 stiffness | 4 iterations | 5 count | 6 damping
26// 7 boundX | 8 boundY | 9 boundW | 10 boundH | 11 hasBounds
27// 12 forceX | 13 forceY | 14 interactX | 15 interactY
28// 16 interactRadius | 17 interactStrength | 18 links-per-particle
29// 19 mode (0 = integrate, 1 = constraints + bounds, 2 = build hash,
30// 3 = self-collision, 4 = clear hash counters)
31// 20 self-collision min distance | 21 self-collision on/off
32// 22 hash nx | 23 hash ny | 24 cell count
33const char *kClothKernel = R"glsl(#version 450
34layout(local_size_x = 64) in;
35
36layout(set = 0, binding = 0) buffer Pos {
37 vec4 posIn[]; // x, y, prevX, prevY (read)
38};
39layout(set = 0, binding = 1) buffer PosOut {
40 vec4 posOut[]; // x, y, prevX, prevY (write)
41};
42layout(set = 0, binding = 2) buffer Links {
43 vec4 links[]; // otherIndex, restLen, 0, 0 per particle slot
44};
45layout(set = 0, binding = 3) buffer Flags {
46 vec4 flags[]; // x = 1.0 when pinned
47};
48layout(set = 0, binding = 4) buffer CellHead {
49 uint cellHead[]; // per-cell linked-list head; 0xFFFFFFFF = empty
50};
51layout(set = 0, binding = 5) buffer CellNext {
52 uint cellNext[]; // slot -> next slot in the same cell
53};
54layout(set = 0, binding = 6) buffer CellItems {
55 uint cellItems[]; // slot -> particle index
56};
57layout(set = 0, binding = 7) buffer CellSlot {
58 uint cellSlot[1]; // global slot counter for hash building
59};
60
61layout(push_constant) uniform PC {
62 float data[32];
63} pc;
64
65// Keep two non-adjacent particles at least minDist apart. Pinned endpoints are
66// handled by their own thread being skipped: the free side takes the full
67// correction; two free sides split it.
68void separatePair(uint i, uint j, inout vec4 p) {
69 if (j == i) return;
70 const uint K = uint(pc.data[18]);
71 bool linked = false;
72 for (uint k = 0u; k < K; ++k) {
73 vec4 l = links[i * K + k];
74 if (int(l.x) < 0) break;
75 if (int(l.x) == int(j)) {
76 linked = true;
77 break;
78 }
79 }
80 if (linked) return;
81 bool jPinned = flags[j].x > 0.5;
82 vec4 q = posIn[j];
83 float dx = q.x - p.x;
84 float dy = q.y - p.y;
85 float d2 = dx * dx + dy * dy;
86 float minDist = pc.data[20];
87 if (d2 >= minDist * minDist || d2 < 1e-8) return;
88 float d = sqrt(d2);
89 float corr = min(2.0, (minDist - d) / d);
90 float w = jPinned ? 1.0 : 0.5;
91 p.x -= dx * corr * w;
92 p.y -= dy * corr * w;
93}

95void main() {
96 // Reset the hash: every cell head becomes empty, the global slot counter
97 // goes back to zero. Runs once per substep before building.
98 if (int(pc.data[19]) == 4) {
99 uint cid = gl_GlobalInvocationID.x;
100 if (cid < uint(pc.data[24])) cellHead[cid] = 0xFFFFFFFFu;
101 if (cid == 0u) cellSlot[0] = 0u;
102 return;
103 }
104
105 uint i = gl_GlobalInvocationID.x;
106 uint n = uint(pc.data[5]);
107 if (i >= n) return;
108
109 const float dt = pc.data[0];
110 const float damp = 1.0 - pc.data[6];
111 const bool pinned = flags[i].x > 0.5;
112 const uint K = uint(pc.data[18]);
113 const int mode = int(pc.data[19]);
114 vec4 p = posIn[i];
115
116 if (mode == 0 && !pinned) {
117 // Verlet integrate.
118 float vx = (p.x - p.z) * damp;
119 float vy = (p.y - p.w) * damp;
120 p.z = p.x;
121 p.w = p.y;
122 float ax = pc.data[1] + pc.data[12];
123 float ay = pc.data[2] + pc.data[13];
124 p.x += vx + ax * dt * dt;
125 p.y += vy + ay * dt * dt;
126
127 // Pointer field (Fluid2D-style attract/repel).
128 float ir = pc.data[16];
129 if (ir > 0.0 && abs(pc.data[17]) > 1e-9) {
130 float dx = pc.data[14] - p.x;
131 float dy = pc.data[15] - p.y;
132 float r2 = dx * dx + dy * dy;
133 if (r2 < ir * ir && r2 > 1e-6) {
134 float r = sqrt(r2);
135 float w = 1.0 - r / ir;
136 p.x += (dx / r) * pc.data[17] * w * dt * dt;
137 p.y += (dy / r) * pc.data[17] * w * dt * dt;
138 }
139 }
140 }
141
142 // Distance-constraint relaxation over this particle's link slots. The
143 // owning thread never moves a pinned particle: a pinned endpoint leaves
144 // the full correction to the free partner's thread. Runs in a separate
145 // dispatch so integration results are visible before solving.
146 if (mode == 1 && !pinned) {
147 // One Jacobi pass per dispatch: each thread reads the previous pass's
148 // positions (posIn) and writes its own corrected position (posOut).
149 // The outer loop in update() repeats the dispatch to converge.
150 for (uint k = 0u; k < K; ++k) {
151 vec4 l = links[i * K + k];
152 int other = int(l.x);
153 if (other < 0) break;
154 vec4 q = posIn[uint(other)];
155 bool otherPinned = flags[uint(other)].x > 0.5;
156 float dx = q.x - p.x;
157 float dy = q.y - p.y;
158 float dist = sqrt(dx * dx + dy * dy);
159 if (dist < 1e-5) continue;
160 float diff = (dist - l.y) / dist * pc.data[3];
161 if (otherPinned) {
162 p.x += dx * diff;
163 p.y += dy * diff;
164 } else {
165 p.x += dx * diff * 0.5;
166 p.y += dy * diff * 0.5;
167 }
168 }
169 }
170
171 // Build the spatial hash: each particle atomically claims a global slot and
172 // prepends it to its cell's linked list (overflow-free).
173 if (mode == 2 && pc.data[21] > 0.5) {
174 int nx = int(pc.data[22]);
175 int ny = int(pc.data[23]);
176 if (nx > 0 && ny > 0) {
177 float cellSize = pc.data[20];
178 float bx = pc.data[7];
179 float by = pc.data[8];
180 int cx = clamp(int(floor((p.x - bx) / cellSize)), 0, nx - 1);
181 int cy = clamp(int(floor((p.y - by) / cellSize)), 0, ny - 1);
182 uint cid = uint(cy * nx + cx);
183 uint slot = atomicAdd(cellSlot[0], 1u);
184 if (slot >= n) return;
185 cellNext[slot] = atomicExchange(cellHead[cid], slot);
186 cellItems[slot] = i;
187 }
188 }
189
190 // Particle-level self-collision via the hash (3x3 neighbor cells) or an
191 // O(n) scan fallback when no bounds define a hash grid.
192 if (mode == 3 && pc.data[21] > 0.5 && !pinned) {
193 int nx = int(pc.data[22]);
194 int ny = int(pc.data[23]);
195 if (nx > 0 && ny > 0 && pc.data[20] > 0.0) {
196 float cellSize = pc.data[20];
197 float bx = pc.data[7];
198 float by = pc.data[8];
199 int cx = clamp(int(floor((p.x - bx) / cellSize)), 0, nx - 1);
200 int cy = clamp(int(floor((p.y - by) / cellSize)), 0, ny - 1);
201 for (int oy = -1; oy <= 1; ++oy) {
202 for (int ox = -1; ox <= 1; ++ox) {
203 int ncx = clamp(cx + ox, 0, nx - 1);
204 int ncy = clamp(cy + oy, 0, ny - 1);
205 uint cid = uint(ncy * nx + ncx);
206 uint s = cellHead[cid];
207 while (s != 0xFFFFFFFFu) {
208 separatePair(i, cellItems[s], p);
209 s = cellNext[s];
210 }
211 }
212 }
213 } else {
214 for (uint j = 0u; j < n; ++j) separatePair(i, j, p);
215 }
216 }
217
218 // Axis-aligned bounds with a soft bounce (mirrors the CPU cloth).
219 if (mode == 1 && pc.data[11] > 0.5 && !pinned) {
220 float bx = pc.data[7];
221 float by = pc.data[8];
222 float bw = pc.data[9];
223 float bh = pc.data[10];
224 if (p.x < bx) {
225 float vx = p.x - p.z;
226 p.x = bx;
227 p.z = p.x + vx * 0.35;
228 } else if (p.x > bx + bw) {
229 float vx = p.x - p.z;
230 p.x = bx + bw;
231 p.z = p.x + vx * 0.35;
232 }
233 if (p.y < by) {
234 float vy = p.y - p.w;
235 p.y = by;
236 p.w = p.y + vy * 0.35;
237 } else if (p.y > by + bh) {
238 float vy = p.y - p.w;
239 p.y = by + bh;
240 p.w = p.y + vy * 0.35;
241 }
242 }
243
244 posOut[i] = p;
245}
246)glsl";

252 : gpgpu_(gpgpu), cols_(cols), rows_(rows), spacing_(spacing), originX_(originX),
253 originY_(originY) {
254 if (!gpgpu_) throw Exception("ClothGPU: Gpgpu module required");
255 if (cols_ < 2 || rows_ < 2) throw Exception("ClothGPU: cols and rows must be >= 2");
256 if (spacing_ <= 0.f) throw Exception("ClothGPU: spacing must be > 0");
257
258 const int count = getParticleCount();
259 posCpu_.assign(static_cast<size_t>(count) * 4, 0.f);
260 flagsCpu_.assign(static_cast<size_t>(count), 0.f);
261 for (int r = 0; r < rows_; ++r) {
262 for (int c = 0; c < cols_; ++c) {
263 const size_t i = static_cast<size_t>(r * cols_ + c);
264 posCpu_[i * 4 + 0] = originX_ + float(c) * spacing_;
265 posCpu_[i * 4 + 1] = originY_ + float(r) * spacing_;
266 posCpu_[i * 4 + 2] = posCpu_[i * 4 + 0];
267 posCpu_[i * 4 + 3] = posCpu_[i * 4 + 1];
268 }
269 }
270 rebuildLinks();
271 // Pin the top row in the CPU flag mirror only; buffers are created next and
272 // uploadInitialState() uploads the flags together with positions/links.

321void ClothGPU::rebuildLinks() {
322 links_.assign(static_cast<size_t>(getParticleCount()) * kMaxLinksPerParticle, Link{});
323 linkCpu_.assign(static_cast<size_t>(getParticleCount()) * kMaxLinksPerParticle * 4, 0.f);
324
325 const auto findFreeSlot = [&](int i) {
326 for (int k = 0; k < kMaxLinksPerParticle; ++k) {
327 Link &s = links_[static_cast<size_t>(i) * kMaxLinksPerParticle + k];
328 if (s.other < 0) return k;
329 }
330 return kMaxLinksPerParticle - 1;
331 };
332 auto add = [&](int r, int c, int nr, int nc) {
333 if (nr < 0 || nc < 0 || nr >= rows_ || nc >= cols_) return;
334 const int i = r * cols_ + c;
335 const int j = nr * cols_ + nc;
336 Link &slot = links_[static_cast<size_t>(i) * kMaxLinksPerParticle +
337 findFreeSlot(i)];
338 slot.other = j;
339 const float dx = (float(nc) - float(c)) * spacing_;
340 const float dy = (float(nr) - float(r)) * spacing_;
341 slot.rest = std::sqrt(dx * dx + dy * dy);
342 };
343
344 for (int r = 0; r < rows_; ++r) {
345 for (int c = 0; c < cols_; ++c) {
346 add(r, c, r, c + 1); // right
347 add(r, c, r, c - 1); // left
348 add(r, c, r + 1, c); // down
349 add(r, c, r - 1, c); // up
350 add(r, c, r + 1, c + 1);
351 add(r, c, r + 1, c - 1);
352 add(r, c, r - 1, c + 1);
353 add(r, c, r - 1, c - 1);
354 add(r, c, r, c + 2); // bend
355 add(r, c, r, c - 2);
356 add(r, c, r + 2, c);
357 add(r, c, r - 2, c);
358 }
359 }
360
361 for (size_t i = 0; i < links_.size(); ++i) {
362 const Link &link = links_[i];
363 linkCpu_[i * 4 + 0] = float(link.other);
364 linkCpu_[i * 4 + 1] = link.rest;
365 linkCpu_[i * 4 + 2] = 0.f;
366 linkCpu_[i * 4 + 3] = 0.f;
367 }
368}

537eve::Result<void> ClothGPU::stepGpu(float dt, int substeps) {
538 if (destroyed_ || !shader_ || !gpgpu_ || !seq_)
539 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
540 "GPU cloth resources are not available",
541 "physics.clothGpu.step"));
542 if (substeps < 1 || substeps > 1024)
543 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
544 "GPU cloth substep count must be in [1, 1024]",
545 "physics.clothGpu.step.substeps"));
546
547 const int count = getParticleCount();
548 ensureHashBuffers();
549 const float h = dt / float(substeps);
550 shader_->bindBuffer(2, linkBuf_);
551 shader_->bindBuffer(3, flagBuf_);
552 if (cellHeadBuf_ && cellNextBuf_ && cellItemsBuf_ && cellSlotBuf_) {
553 shader_->bindBuffer(4, cellHeadBuf_);
554 shader_->bindBuffer(5, cellNextBuf_);
555 shader_->bindBuffer(6, cellItemsBuf_);
556 shader_->bindBuffer(7, cellSlotBuf_);
557 }
558 shader_->setFloat(0, h);
559 shader_->setFloat(1, gravityX_);
560 shader_->setFloat(2, gravityY_);
561 shader_->setFloat(3, stiffness_);
562 shader_->setFloat(4, float(iterations_));
563 shader_->setFloat(5, float(count));
564 shader_->setFloat(6, damping_);
565 shader_->setFloat(7, boundX_);
566 shader_->setFloat(8, boundY_);
567 shader_->setFloat(9, boundW_);
568 shader_->setFloat(10, boundH_);
569 shader_->setFloat(11, hasBounds_ ? 1.f : 0.f);
570 shader_->setFloat(12, forceX_);
571 shader_->setFloat(13, forceY_);
572 shader_->setFloat(14, interactX_);
573 shader_->setFloat(15, interactY_);
574 shader_->setFloat(16, interactRadius_);
575 shader_->setFloat(17, interactStrength_);
576 shader_->setFloat(18, float(kMaxLinksPerParticle));
577 shader_->setFloat(20, particleSize_ * 2.f);
578 shader_->setFloat(21, selfCollision_ ? 1.f : 0.f);
579 shader_->setFloat(22, float(hashNx_));
580 shader_->setFloat(23, float(hashNy_));
581 const int nCells = hashNx_ > 0 && hashNy_ > 0 ? hashNx_ * hashNy_ : 0;
582 shader_->setFloat(24, float(nCells));
583
584 const int groups = (count + 63) / 64;
585 // Double-buffered passes: every pass reads posIn and writes posOut, then the
586 // roles swap. Keeps constraint iterations Jacobi-stable (no in-place
587 // feedback between threads). The whole frame is recorded into one Sequence
588 // submission (the Sequence inserts memory barriers between dispatches).
589 GpuBuffer *in = posBuf_;
590 GpuBuffer *out = posBufB_;
591 const auto pass = [&](float mode) {
592 shader_->bindBuffer(0, in);
593 shader_->bindBuffer(1, out);
594 shader_->setFloat(19, mode);
595 seq_->recordDispatch(shader_, groups, 1, 1);
596 std::swap(in, out);
597 };
598 seq_->begin();
599 for (int s = 0; s < substeps; ++s) {
600 pass(0.f); // integrate
601 if (selfCollision_ && nCells > 0) {
602 // Clear counters, build the hash from the freshly integrated
603 // positions, then constraints, then hash-based self-collision.
604 shader_->setFloat(19, 4.f);
605 seq_->recordDispatch(shader_, (nCells + 63) / 64, 1, 1);
606 pass(2.f);
607 }
608 for (int it = 0; it < iterations_; ++it) pass(1.f); // constraints + bounds
609 if (selfCollision_) {
610 for (int sc = 0; sc < 2; ++sc)
611 pass(3.f); // self-collision (hash or O(n²) fallback)
612 }
613 }
614
615 // The pass count is even in both branches, so the result lands in posBuf_.
616 seq_->recordDownload(posBuf_, staging_, uint64_t(count) * 4 * sizeof(float));
617 seq_->submit();
618 staging_->downloadBytes(posCpu_.data(), uint64_t(count) * 4 * sizeof(float));
619 forceX_ = 0.f;
620 forceY_ = 0.f;
621 interactStrength_ = 0.f;
622 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
623}

626 if (destroyed_)
627 return eve::Result<void>::failure(eve::Diagnostic::error(
628 eve::DiagnosticCode::PreconditionViolation, "Cannot step a destroyed GPU cloth", "physics.clothGpu.step"));
629 auto valid = detail::validateSimulationStep(stepValue, settings, observation_);
630 if (!valid) return valid;
631 auto next = detail::advanceSimulationObservation(observation_, stepValue);
632 if (!next) return eve::Result<void>::failure(next.status());
633 try {
634 auto applied = stepGpu(static_cast<float>(stepValue.delta.seconds()), settings.subStepCount);
635 if (!applied) return applied;
636 } catch (const std::exception &error) {
637 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::Failed,
638 std::string("GPU cloth step failed: ") + error.what(),

640 } catch (...) {
641 return eve::Result<void>::failure(eve::Diagnostic::error(
642 eve::DiagnosticCode::Failed, "GPU cloth step failed with an unknown exception", "physics.clothGpu.step"));
643 }
644 observation_ = std::move(next).takeValue();
645 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
646}

660 if (!gfx || destroyed_) return;
661 const Color linkColor(colorR_, colorG_, colorB_, colorA_ * 0.75f);
662 const Color nodeColor(colorR_, colorG_, colorB_, colorA_);

665 for (int k = 0; k < kMaxLinksPerParticle; ++k) {
666 const Link &link = links_[static_cast<size_t>(i) * kMaxLinksPerParticle + k];
667 if (link.other < 0) break;
668 const float x1 = posCpu_[static_cast<size_t>(i) * 4 + 0];
669 const float y1 = posCpu_[static_cast<size_t>(i) * 4 + 1];
670 const float x2 = posCpu_[static_cast<size_t>(link.other) * 4 + 0];
671 const float y2 = posCpu_[static_cast<size_t>(link.other) * 4 + 1];
672 const float dx = x2 - x1;
673 const float dy = y2 - y1;
674 const float len = std::sqrt(dx * dx + dy * dy);
675 const int steps = std::max(1, int(len / 4.f));
676 for (int s = 0; s <= steps; ++s) {
677 const float t = float(s) / float(steps);
678 gfx->drawSolidRect(x1 + dx * t - 1.f, y1 + dy * t - 1.f, 2.f, 2.f, linkColor);
679 }
680 }
681 }
682 for (int i = 0; i < count; ++i) {
683 const float s = flagsCpu_[static_cast<size_t>(i)] > 0.5f
684 ? std::max(5.f, particleSize_ + 2.f)
685 : particleSize_;
686 gfx->drawSolidRect(posCpu_[static_cast<size_t>(i) * 4 + 0] - s * 0.5f,
687 posCpu_[static_cast<size_t>(i) * 4 + 1] - s * 0.5f, s, s,
688 nodeColor);
689 }
690}
