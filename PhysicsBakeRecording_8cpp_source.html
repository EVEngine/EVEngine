23template <class T>
24ResultRef<T> sourceArgument(const ssq::Object& object) {
25 if (object.getType() == ssq::Type::INSTANCE && object.getTypeTag() == script::detail::squirrelTypeHash<T*>()) {
26 try {
27 if (auto* source = object.to<T*>()) return ResultRef<T>::success(std::ref(*source));
28 } catch (const ssq::TypeException&) {
29 // Malformed wrappers are rejected before any frame admission.
30 }
31 }
32 return ResultRef<T>::failure(invalid("expected a live source of the requested native type"));
33}

39template <class WorldType, class BodyType>
40Result<void> bindRigid(ScriptRigidRecording& session, const ssq::Object& worldObject, const ssq::Object& bodyObject,
41 const std::string& identity, RigidTrackSpace expectedSpace) {
42 if (session.recording.space() != expectedSpace || session.recording.frameCount() != 0 ||
43 session.bindings.size() >= 16384)
44 return Result<void>::failure(invalid("wrong coordinate space, locked roster or binding budget exceeded"));
45 auto world = sourceArgument<WorldType>(worldObject);
46 if (!world) return Result<void>::failure(world.status());
47 auto body = sourceArgument<BodyType>(bodyObject);
48 if (!body) return Result<void>::failure(body.status());
49 const auto id = PersistentId::parse(identity);
50 if (!id || id->isNil()) return Result<void>::failure(invalid("expected a non-nil persistent UUID"));
51 const auto& sourceWorld = world.value().get();
52 const auto& sourceBody = body.value().get();
53 if (!sourceWorld.isValid() || !sourceBody.isValid() || sourceBody.getWorld() != &sourceWorld ||
54 sourceWorld.findBody(sourceBody.runtimeHandle()) != &sourceBody)
55 return Result<void>::failure(Diagnostic::error(DiagnosticCode::StaleHandle,
56 "source body is stale or belongs to another world",
57 "physicsBakeRecording.script"));
58 const RigidBakeBinding candidate{*id, sourceWorld.runtimeHandle(), sourceBody.runtimeHandle()};
59 if ((!session.bindings.empty() && session.bindings.front().world != candidate.world) ||
60 std::any_of(session.bindings.begin(), session.bindings.end(),
61 [&](const auto& binding) { return binding.id == candidate.id || binding.body == candidate.body; }))
62 return Result<void>::failure(invalid("mixed worlds or duplicate body/identity binding"));
63 try {
64 session.bindings.push_back(candidate);
65 } catch (const std::bad_alloc&) {
66 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Failed, "rigid binding allocation failed",
67 "physicsBakeRecording.script"));
68 }
69 return Result<void>::success();
70}
71template <class WorldType>
72ssq::Table captureRigid(HSQUIRRELVM vm, ScriptRigidRecording& session, const ssq::Object& worldObject, float seconds,
73 RigidTrackSpace expectedSpace) {
74 if (session.recording.space() != expectedSpace)
75 return script::projectStatusResult(vm, Status::failure(invalid("wrong capture coordinate space")));
76 auto world = sourceArgument<WorldType>(worldObject);
77 if (!world) return script::projectStatusResult(vm, world.status());
78 auto frame = captureRigidFrame(world.value().get(), session.bindings, seconds);
79 if (!frame) return script::projectStatusResult(vm, frame.status());
80 return script::projectResult(vm, session.recording.append(std::move(frame).takeValue()));
81}
