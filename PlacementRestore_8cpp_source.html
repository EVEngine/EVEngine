179eve::Result<EdgeCurveGroup> PlacementSystem::restoreEdgeCurveGroupExact(PlacementWorld *world,
180 const EdgeCurveGroup &group,
181 const std::vector<PlacedBuilding> &members) {
182 if (!world || !group.id || group.controlPoints.size() != 4 || group.subdivisions < 2 || group.subdivisions > 4096 ||
183 members.empty() || group.instanceIds.size() != members.size()) {
184 return eve::Result<EdgeCurveGroup>::failure(eve::Diagnostic::error(
185 eve::DiagnosticCode::InvalidArgument,
186 "curve group restore requires a world, identity, four controls, valid subdivisions and matching members",
187 {}, {}, "building.edge-curve-group.restore"));
188 }
189 std::unordered_set<int> memberIds;
190 for (size_t index = 0; index < members.size(); ++index) {
191 const PlacedBuilding &member = members[index];
192 if (member.instanceId <= 0 || member.instanceId != group.instanceIds[index] || member.placementKind != "edge" ||
193 member.buildingId != group.buildingId || member.level != group.level ||
194 !memberIds.insert(member.instanceId).second) {
195 return eve::Result<EdgeCurveGroup>::failure(eve::Diagnostic::error(
196 eve::DiagnosticCode::InvalidArgument,
197 "curve group members must be unique ordered edges with matching definition and level",
198 std::to_string(member.instanceId), {}, "building.edge-curve-group.restore"));
199 }
200 }
201 for (const EdgeCurveControlPoint &point : group.controlPoints) {
202 if (!std::isfinite(point.x) || !std::isfinite(point.y)) {
203 return eve::Result<EdgeCurveGroup>::failure(
204 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "curve group controls must be finite", {},
205 {}, "building.edge-curve-group.restore"));
206 }
207 }
208 if ((!group.surfaceSamples.empty() && group.surfaceSamples.size() != static_cast<size_t>(group.subdivisions + 1)) ||
209 (group.surfaceSamples.empty() &&
210 (!group.surfaceProviderName.empty() || !group.surfaceId.empty() || group.surfaceRevision != 0))) {
211 return eve::Result<EdgeCurveGroup>::failure(
212 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
213 "curve group surface metadata requires exactly subdivisions plus one frames", {}, {},
214 "building.edge-curve-group.restore"));
215 }
216 for (const EdgeCurveSurfaceSample &sample : group.surfaceSamples) {
217 const float normalLength = std::sqrt(sample.normalX * sample.normalX + sample.normalY * sample.normalY +
218 sample.normalZ * sample.normalZ);
219 if (!std::isfinite(sample.worldX) || !std::isfinite(sample.worldY) || !std::isfinite(sample.worldZ) ||
220 !std::isfinite(normalLength) || normalLength <= 1e-5f) {
221 return eve::Result<EdgeCurveGroup>::failure(
222 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
223 "curve group surface frames require finite positions and non-zero normals", {},
224 {}, "building.edge-curve-group.restore"));
225 }
226 }
227
228 std::unique_ptr<PlacementWorld> candidate = world->cloneState();
229 if (!candidate || candidate->edgeCurveGroups_.contains(group.id.value)) {
230 return eve::Result<EdgeCurveGroup>::failure(
231 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "curve group identity is already in use",
232 std::to_string(group.id.value), {}, "building.edge-curve-group.restore"));
233 }
234 for (const PlacedBuilding &member : members) {
235 std::string reason;
236 if (restoreExact(candidate.get(), member, &reason) != PlacementRestoreStatus::Restored) {
237 return eve::Result<EdgeCurveGroup>::failure(eve::Diagnostic::error(
238 eve::DiagnosticCode::Conflict, "curve group member restore was rejected without destination mutation",
239 reason, {}, "building.edge-curve-group.restore"));
240 }
241 }
242 for (int instanceId : group.instanceIds) candidate->buildings_.at(instanceId).edgeCurveGroupId = group.id;
243 candidate->edgeCurveGroups_.emplace(group.id.value, group);
244 candidate->nextEdgeCurveGroupId_ = std::max(candidate->nextEdgeCurveGroupId_, group.id.value + 1);
245 world->swapState(*candidate);
246 return eve::Result<EdgeCurveGroup>::success(group);
247}
