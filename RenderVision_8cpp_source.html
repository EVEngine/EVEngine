38// Splits "http://host[:port][/basepath]" into host, port and base path.
39void splitBaseUrl(const std::string& url, std::string& host, Poco::UInt16& port,
40 std::string& basePath) {
41 host = "127.0.0.1";
42 port = 80;
43 basePath = "";
44 std::string rest = url;
45 const auto scheme = rest.find("://");
46 if (scheme != std::string::npos) rest = rest.substr(scheme + 3);
47 const auto slash = rest.find('/');
48 std::string authority = (slash == std::string::npos) ? rest : rest.substr(0, slash);
49 if (slash != std::string::npos) basePath = rest.substr(slash);
50 // Strip trailing slash from basePath.
51 while (basePath.size() > 1 && basePath.back() == '/') basePath.pop_back();
52 const auto colon = authority.rfind(':');
53 if (colon != std::string::npos) {
54 host = authority.substr(0, colon);
55 try {
56 port = static_cast<Poco::UInt16>(std::stoi(authority.substr(colon + 1)));
57 } catch (...) {
58 port = 80;
59 }
60 } else {
61 host = authority;
62 }
63 if (host.empty()) host = "127.0.0.1";
64}

183std::string RenderVision::doDescribe(eve::IRenderCapture* cap, const std::string& renderDataJson,
184 const std::string& reason) {
185 std::string baseUrl, apiKey, model, path;
186 int timeoutMs = timeoutMs_;
187 {
188 std::lock_guard<std::mutex> lock(mu_);
189 ensureEnvLocked();
190 baseUrl = baseUrl_;
191 apiKey = apiKey_;
192 model = model_;
193 path = path_;
194 timeoutMs = timeoutMs_;
195 }
196 if (baseUrl.empty() || model.empty()) {
197 const std::string err = "error: RenderVision not configured (set EVE_VISION_BASE_URL/MODEL or call eve_render_vision_config)";
198 std::lock_guard<std::mutex> lock(mu_);
199 lastError_ = err;
200 return err;
201 }
202
203 // --- Capture frame to PNG bytes in memory (via the render-capture interface) ---
204 const std::string dataUrl = cap->capturePngDataUrl();
205 if (dataUrl.empty()) {
206 const std::string err = "error: frame readback returned no image";
207 std::lock_guard<std::mutex> lock(mu_);
208 lastError_ = err;
209 return err;
210 }
211
212 // --- Build OpenAI-compatible chat/completions body ---
213 std::string context = reason.empty() ? "Engine render parameters:" : "Engine render parameters (dump reason: " + reason + "):";
214 if (!renderDataJson.empty()) context += "\n" + renderDataJson;
215
216 Poco::JSON::Object::Ptr root = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
217 root->set("model", model);
218 Poco::JSON::Array::Ptr messages = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
219
220 Poco::JSON::Object::Ptr sys = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
221 sys->set("role", "system");
222 sys->set("content", kSystemPrompt);
223 messages->add(sys);
224
225 Poco::JSON::Object::Ptr user = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
226 user->set("role", "user");
227 Poco::JSON::Array::Ptr uc = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
228
229 Poco::JSON::Object::Ptr textPart = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
230 textPart->set("type", "text");
231 textPart->set("text", context);
232 uc->add(textPart);
233
234 Poco::JSON::Object::Ptr imgPart = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
235 imgPart->set("type", "image_url");
236 Poco::JSON::Object::Ptr iu = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
237 iu->set("url", dataUrl);
238 imgPart->set("image_url", iu);
239 uc->add(imgPart);
240
241 user->set("content", uc);
242 messages->add(user);
243 root->set("messages", messages);
244 root->set("max_tokens", 1024);
245
246 std::ostringstream bodyStream;
247 Poco::JSON::Stringifier::stringify(Poco::Dynamic::Var(root), bodyStream, 0, 0);
248 const std::string body = bodyStream.str();
249
250 // --- HTTP POST ---
251 std::string host;
252 Poco::UInt16 port = 80;
253 std::string basePath;
254 splitBaseUrl(baseUrl, host, port, basePath);
255 std::string endpoint = basePath + path;
256 if (endpoint.empty()) endpoint = "/chat/completions";
257
258 std::string responseBody;
259 try {
260 Poco::Net::HTTPClientSession session(host, port);
261 session.setTimeout(
262 Poco::Timespan(static_cast<long long>(timeoutMs) * 1000));
263 session.setKeepAlive(false);
264 Poco::Net::HTTPRequest req(Poco::Net::HTTPRequest::HTTP_POST, endpoint,
265 Poco::Net::HTTPMessage::HTTP_1_1);
266 req.setContentType("application/json");
267 req.setChunkedTransferEncoding(false);
268 req.setContentLength(static_cast<int>(body.size()));
269 if (!apiKey.empty()) req.set("Authorization", "Bearer " + apiKey);
270 std::ostream& os = session.sendRequest(req);
271 os << body;
272 os.flush();
273 Poco::Net::HTTPResponse res;
274 std::istream& is = session.receiveResponse(res);
275 responseBody.assign(std::istreambuf_iterator<char>(is), std::istreambuf_iterator<char>());
276 if (res.getStatus() != Poco::Net::HTTPResponse::HTTP_OK) {
277 std::string snippet = responseBody.size() > 300 ? responseBody.substr(0, 300) : responseBody;
278 const std::string err = "error: vision HTTP " + std::to_string(res.getStatus()) + ": " + snippet;
279 std::lock_guard<std::mutex> lock(mu_);
280 lastError_ = err;
281 return err;
282 }
283 } catch (const Poco::Exception& e) {
284 const std::string err = std::string("error: vision request failed: ") + e.displayText();
285 std::lock_guard<std::mutex> lock(mu_);
286 lastError_ = err;
287 return err;
288 } catch (const std::exception& e) {
289 const std::string err = std::string("error: vision request failed: ") + e.what();
290 std::lock_guard<std::mutex> lock(mu_);
291 lastError_ = err;
292 return err;
293 }
294
295 // --- Parse response ---
296 try {
297 Poco::JSON::Parser parser;
298 auto var = parser.parse(responseBody);
299 auto obj = var.extract<Poco::JSON::Object::Ptr>();
300 auto choices = obj ? obj->getArray("choices") : nullptr;
301 if (!choices || choices->size() == 0) {
302 const std::string err = "error: vision response had no choices";
303 std::lock_guard<std::mutex> lock(mu_);
304 lastError_ = err;
305 return err;
306 }
307 auto c0 = choices->getObject(0);
308 auto msg = c0 ? c0->getObject("message") : nullptr;
309 std::string content = msg ? msg->optValue<std::string>("content", "") : "";
310 if (content.empty()) {
311 const std::string err = "error: vision response had empty content";
312 std::lock_guard<std::mutex> lock(mu_);
313 lastError_ = err;
314 return err;
315 }
316 std::lock_guard<std::mutex> lock(mu_);
317 latest_ = content;
318 lastError_.clear();
319 return content;
320 } catch (const Poco::Exception& e) {
321 const std::string err = std::string("error: vision response parse failed: ") + e.displayText();
322 std::lock_guard<std::mutex> lock(mu_);
323 lastError_ = err;
324 return err;
325 } catch (const std::exception& e) {
326 const std::string err = std::string("error: vision response parse failed: ") + e.what();
327 std::lock_guard<std::mutex> lock(mu_);
328 lastError_ = err;
329 return err;
330 }
331}
