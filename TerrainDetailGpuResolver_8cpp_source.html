58Result<std::unique_ptr<TerrainDetailRuntime>> TerrainDetailRuntime::load(
59 const asset::EvpackResourceReader& reader, const asset::EvpackCapabilities& capabilities,
60 graphics::Graphics& graphics, const AssetRef& asset, const TerrainVegetationLimits& limits) {
61 EvpackInstanceSetLoader loader(reader);
62 auto loaded = loader.load(asset, capabilities);
63 if (!loaded) return Result<std::unique_ptr<TerrainDetailRuntime>>::failure(loaded.status());
64 auto realization = realizeTerrainVegetation(nullptr, &loaded.value(), limits);
65 if (!realization)
66 return Result<std::unique_ptr<TerrainDetailRuntime>>::failure(realization.status());
67
68 auto runtime = std::unique_ptr<TerrainDetailRuntime>(
69 new TerrainDetailRuntime(reader, capabilities, graphics));
70 // Resolve and validate every prototype before RenderSystem3D opens a frame. GPU uploads from the collector
71 // callback can wait on the active frame and deadlock; publication is therefore deferred until preparation passes.
72 auto prepared = buildTerrainVegetationGpuPlan(realization.value(), *runtime->resolver_, {});
73 if (!prepared) return Result<std::unique_ptr<TerrainDetailRuntime>>::failure(prepared.status());
74 runtime->renderer_ = std::make_unique<TerrainVegetationRenderer>(
75 std::move(realization).takeValue(), *runtime->resolver_);
76 return Result<std::unique_ptr<TerrainDetailRuntime>>::success(std::move(runtime));
77}

104Result<TerrainVegetationGpuPrototype> TerrainDetailGpuResolver::resolve(
105 const TerrainVegetationGpuPrototypeRequest& request) {
106 if (!request.detail)
107 return failure<TerrainVegetationGpuPrototype>(DiagnosticCode::NotFound,
108 "terrain Detail metadata is required", std::string(request.prototype));
109 const auto& detail = *request.detail;
110 if (detail.resourceAsset.empty())
111 return failure<TerrainVegetationGpuPrototype>(DiagnosticCode::NotFound,
112 "terrain Detail resource is unresolved", detail.prototype);
113 const std::string key = detail.prototype + "\n" + detail.renderMode + "\n" + detail.resourceAsset;
114 if (const auto found = impl_->resources.find(key); found != impl_->resources.end()) {
115 if (found->second.slots.parts.empty() &&
116 (found->second.slots.meshId == graphics::kInvalidGpuDrivenSlot ||
117 found->second.slots.materialId == graphics::kInvalidGpuDrivenSlot))
118 return failure<TerrainVegetationGpuPrototype>(DiagnosticCode::Conflict,
119 "terrain Detail resource cleanup is pending", detail.prototype);
120 return Result<TerrainVegetationGpuPrototype>::success(found->second.slots);
121 }
122 auto reference = AssetRef::parse(detail.resourceAsset);
123 if (!reference) return Result<TerrainVegetationGpuPrototype>::failure(reference.status());
124 if (detail.usePrototypeMesh || detail.renderMode == "VertexLit") {
125 auto prefab = asset_graphics::EvpackStaticPrefab::load(impl_->reader, impl_->meshes, impl_->images,
126 reference.value(), impl_->capabilities);
127 if (!prefab) return Result<TerrainVegetationGpuPrototype>::failure(prefab.status());
128 auto parts = prefab.value()->prepareGpuDriven(impl_->graphics);
129 if (!parts) return Result<TerrainVegetationGpuPrototype>::failure(parts.status());
130 Impl::Resource resource;
131 resource.prefab = std::move(prefab).takeValue();
132 resource.slots.parts.reserve(parts.value().size());
133 for (const auto& source : parts.value())
134 resource.slots.parts.push_back({source.localTransform, source.meshId, source.materialId, source.flags,
135 source.lodGroupId});
136 const auto slots = resource.slots;
137 impl_->resources.emplace(key, std::move(resource));
138 return Result<TerrainVegetationGpuPrototype>::success(slots);
139 }
140 auto geometry = buildTerrainDetailCard(detail);
141 if (!geometry) return Result<TerrainVegetationGpuPrototype>::failure(geometry.status());
142 auto mesh = impl_->meshes.uploadMesh(geometry.value().positions.data(), geometry.value().normals.data(),
143 geometry.value().texcoords.data(),
144 static_cast<int>(geometry.value().positions.size() / 3),
145 geometry.value().indices.data(), static_cast<int>(geometry.value().indices.size()));
146 if (!mesh) return Result<TerrainVegetationGpuPrototype>::failure(mesh.status());
147 asset_graphics::EvpackImageLoader imageLoader(impl_->reader, impl_->images);
148 auto image = imageLoader.load(reference.value(), impl_->capabilities);
149 if (!image) {
150 auto released = impl_->meshes.releaseMesh(mesh.value());
151 if (!released) {
152 Impl::Resource cleanup;
153 cleanup.mesh = mesh.value();
154 impl_->resources.emplace(key, std::move(cleanup));
155 return Result<TerrainVegetationGpuPrototype>::failure(released.status());
156 }
157 return Result<TerrainVegetationGpuPrototype>::failure(image.status());
158 }
159 Impl::Resource resource;
160 resource.mesh = mesh.value();
161 resource.texture = image.value().texture;
162 resource.material = std::make_unique<graphics::Material>();
163 resource.material->setSurfaceMode("masked");
164 resource.material->setAlphaCutoff(.5f);
165 resource.material->setDoubleSided(true);
166 resource.material->setCameraFacing(geometry.value().cameraFacing);
167 resource.material->setAlbedoTexture(resource.texture);
168 resource.material->setRoughness(1.f);
169 if (!impl_->graphics.gpuDrivenMaterialUsable(resource.material.get())) {
170 auto imageReleased = impl_->images.releaseImage(resource.texture);
171 auto meshReleased = impl_->meshes.releaseMesh(resource.mesh);
172 if (imageReleased) resource.texture = nullptr;
173 if (meshReleased) resource.mesh = nullptr;
174 if (!imageReleased || !meshReleased) {
175 const auto status = !imageReleased ? imageReleased.status() : meshReleased.status();
176 impl_->resources.emplace(key, std::move(resource));
177 return Result<TerrainVegetationGpuPrototype>::failure(status);
178 }
179 return failure<TerrainVegetationGpuPrototype>(DiagnosticCode::Unsupported,
180 "terrain Detail material is unavailable to GPU-driven rendering",
181 detail.prototype);
182 }
183 resource.slots.meshId = impl_->graphics.gpuDrivenMeshRecord(resource.mesh);
184 resource.slots.materialId = impl_->graphics.gpuDrivenMaterialRecord(resource.material.get());
185 if (resource.slots.meshId == graphics::kInvalidGpuDrivenSlot ||
186 resource.slots.materialId == graphics::kInvalidGpuDrivenSlot) {
187 auto materialReleased = impl_->graphics.gpuDrivenReleaseMaterialRecord(resource.material.get());
188 auto imageReleased = impl_->images.releaseImage(resource.texture);
189 auto meshReleased = impl_->meshes.releaseMesh(resource.mesh);
190 if (materialReleased) resource.material.reset();
191 if (imageReleased) resource.texture = nullptr;
192 if (meshReleased) resource.mesh = nullptr;
193 if (!materialReleased || !imageReleased || !meshReleased) {
194 const auto status = !materialReleased ? materialReleased.status()
195 : !imageReleased ? imageReleased.status()
196 : meshReleased.status();
197 impl_->resources.emplace(key, std::move(resource));
198 return Result<TerrainVegetationGpuPrototype>::failure(status);
199 }
200 return failure<TerrainVegetationGpuPrototype>(DiagnosticCode::Failed,
201 "terrain Detail GPU table registration failed", detail.prototype);
202 }
203 const auto slots = resource.slots;
204 impl_->resources.emplace(key, std::move(resource));
205 return Result<TerrainVegetationGpuPrototype>::success(slots);
206}

208Result<void> TerrainDetailGpuResolver::drawShadow(
209 const TerrainVegetationGpuPrototypeRequest& request, const std::array<float, 16>& transform,
210 const std::array<float, 16>& lightViewProjection) {
211 if (!request.detail)
212 return failure<void>(DiagnosticCode::NotFound, "terrain Detail shadow metadata is required",
213 std::string(request.prototype));
214 const auto& detail = *request.detail;
215 const std::string key = detail.prototype + "\n" + detail.renderMode + "\n" + detail.resourceAsset;
216 const auto found = impl_->resources.find(key);
217 // Shadow passes precede the frame's GPU collector. The first frame prepares the resource in that collector;
218 // subsequent shadow passes consume it. Uploading inside an active shadow render pass can deadlock the backend.
219 if (found == impl_->resources.end()) return Result<void>::success();
220 if (found->second.prefab)
221 return found->second.prefab->drawShadow(impl_->graphics, transform, lightViewProjection);
222 if (!found->second.mesh || !found->second.texture)
223 return failure<void>(DiagnosticCode::Conflict, "terrain Detail shadow resource cleanup is pending",
224 detail.prototype);
225 const glm::mat4 lightMvp = glm::make_mat4(lightViewProjection.data()) * glm::make_mat4(transform.data());
226 impl_->graphics.drawMeshShadowAlpha(found->second.mesh, lightMvp, found->second.texture, true);
227 return Result<void>::success();
228}

230Result<void> TerrainDetailGpuResolver::release() {
231 Status failed;
232 for (auto iterator = impl_->resources.begin(); iterator != impl_->resources.end();) {
233 if (iterator->second.material) {
234 auto material = impl_->graphics.gpuDrivenReleaseMaterialRecord(iterator->second.material.get());
235 if (material)
236 iterator->second.material.reset();
237 else
238 failed = material.status();
239 }
240 if (iterator->second.prefab) {
241 auto prefab = iterator->second.prefab->release();
242 if (prefab)
243 iterator->second.prefab.reset();
244 else
245 failed = prefab.status();
246 }
247 Result<void> image = Result<void>::success();
248 Result<void> mesh = Result<void>::success();
249 if (iterator->second.texture) {
250 image = impl_->images.releaseImage(iterator->second.texture);
251 if (image) iterator->second.texture = nullptr;
252 }
253 if (iterator->second.mesh) {
254 mesh = impl_->meshes.releaseMesh(iterator->second.mesh);
255 if (mesh) iterator->second.mesh = nullptr;
256 }
257 if (!iterator->second.texture && !iterator->second.mesh && !iterator->second.material &&
258 !iterator->second.prefab)
259 iterator = impl_->resources.erase(iterator);
260 else {
261 failed = !image ? image.status() : mesh.status();
262 ++iterator;
263 }
264 }
265 if (failed.isFailure()) return Result<void>::failure(failed);
266 return Result<void>::success();
267}
