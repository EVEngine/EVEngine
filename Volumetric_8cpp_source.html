372void Volumetric::setCamera(float eyeX, float eyeY, float eyeZ, float targetX, float targetY,
373 float targetZ, float upX, float upY, float upZ, float fovYDeg,
374 float aspect, float nearZ, float farZ) {
375 nearZ_ = nearZ > 1e-4f ? nearZ : 0.1f;
376 farZ_ = farZ > nearZ_ ? farZ : nearZ_ + 1.f;
377 const float aspectSafe = aspect > 1e-4f ? aspect : 1.f;
378 const float fovRad = fovYDeg * 0.017453292519943295f;
379 const glm::mat4 view =
380 glm::lookAtRH(glm::vec3(eyeX, eyeY, eyeZ), glm::vec3(targetX, targetY, targetZ),
381 glm::vec3(upX, upY, upZ));
382 const glm::mat4 proj = perspectiveVulkanRH_ZO(fovRad, aspectSafe, nearZ_, farZ_);
383 setInvViewProj(glm::inverse(proj * view));
384 rayShader_->sendFloat("nearZ", nearZ_);
385 rayShader_->sendFloat("farZ", farZ_);
386 fogShader_->sendFloat("nearZ", nearZ_);
387 fogShader_->sendFloat("farZ", farZ_);
388}

496float Volumetric::getFloat(const std::string &name) const {
497 // Prefer the active mode's shader so quality/sample queries match setMode.
498 if (mode_ == "raymarch" && rayShader_ && rayShader_->hasUniform(name)) {
499 float v = 0.f;
500 if (rayShader_->getFromVar(name, &v, sizeof(v)) == int(sizeof(v))) return v;
501 }
502 if (mode_ == "fog" && fogShader_ && fogShader_->hasUniform(name)) {
503 float v = 0.f;
504 if (fogShader_->getFromVar(name, &v, sizeof(v)) == int(sizeof(v))) return v;
505 }
506 if (mode_ == "cloud" && cloudShader_ && cloudShader_->hasUniform(name)) {
507 float v = 0.f;
508 if (cloudShader_->getFromVar(name, &v, sizeof(v)) == int(sizeof(v))) return v;
509 }
510 if (shader_ && shader_->hasUniform(name)) {
511 float v = 0.f;
512 if (shader_->getFromVar(name, &v, sizeof(v)) == int(sizeof(v))) return v;
513 }
514 if (rayShader_ && rayShader_->hasUniform(name)) {
515 float v = 0.f;
516 if (rayShader_->getFromVar(name, &v, sizeof(v)) == int(sizeof(v))) return v;
517 }
518 if (fogShader_ && fogShader_->hasUniform(name)) {
519 float v = 0.f;
520 if (fogShader_->getFromVar(name, &v, sizeof(v)) == int(sizeof(v))) return v;
521 }
522 if (cloudShader_ && cloudShader_->hasUniform(name)) {
523 float v = 0.f;
524 if (cloudShader_->getFromVar(name, &v, sizeof(v)) == int(sizeof(v))) return v;
525 }
526 throw eve::Exception("Volumetric.getFloat: missing param '%s'", name.c_str());
527}

733void Volumetric::uploadFroxel(Graphics *gfx) {
734 if (!gfx) throw eve::Exception("Volumetric.uploadFroxel: null graphics");
735 const int w = atmosphereVolume_->getWidth();
736 const int h = atmosphereVolume_->getHeight();
737 const int d = atmosphereVolume_->getDepth();
738 if (w <= 0 || h <= 0 || d <= 0)
739 throw eve::Exception("Volumetric.uploadFroxel: grid is not configured");
740 const int atlasW = w * froxelAtlasCols_;
741 const int atlasH = h * froxelAtlasRows_;
742 std::vector<uint8_t> rgba(std::size_t(atlasW) * std::size_t(atlasH) * 4u, 0u);
743 for (int z = 0; z < d; ++z) {
744 const int ox = (z % froxelAtlasCols_) * w;
745 const int oy = (z / froxelAtlasCols_) * h;
746 for (int y = 0; y < h; ++y) {
747 for (int x = 0; x < w; ++x) {
748 const glm::vec4 value = atmosphereVolume_->integratedAt(x, y, z);
749 const std::size_t offset = (std::size_t(oy + y) * std::size_t(atlasW) +
750 std::size_t(ox + x)) * 4u;
751 const glm::vec3 mapped = glm::vec3(value) / (glm::vec3(1.f) + glm::vec3(value));
752 rgba[offset + 0] = uint8_t(std::lround(glm::clamp(mapped.r, 0.f, 1.f) * 255.f));
753 rgba[offset + 1] = uint8_t(std::lround(glm::clamp(mapped.g, 0.f, 1.f) * 255.f));
754 rgba[offset + 2] = uint8_t(std::lround(glm::clamp(mapped.b, 0.f, 1.f) * 255.f));
755 rgba[offset + 3] = uint8_t(std::lround(glm::clamp(value.a, 0.f, 1.f) * 255.f));
756 }
757 }
758 }
759 if (!froxelAtlas_ || froxelAtlas_->getWidth() != atlasW || froxelAtlas_->getHeight() != atlasH)
760 froxelAtlas_ = gfx->newTexture(atlasW, atlasH, rgba.data());
761 else
762 gfx->updateTexture(froxelAtlas_, atlasW, atlasH, rgba.data());
763 froxelShader_->sendFloat("atlasCols", float(froxelAtlasCols_));
764 froxelShader_->sendFloat("atlasRows", float(froxelAtlasRows_));
765 froxelShader_->sendFloat("sliceCount", float(d));
766 froxelShader_->sendFloat("nearDistance", atmosphereVolume_->getNearDistance());
767 froxelShader_->sendFloat("farDistance", atmosphereVolume_->getFarDistance());
768}

789 float (*depth01)(int x, int y, void *userdata),
790 void *userdata) {
791 if (!gfx) throw eve::Exception("Volumetric.newLinearDepthTexture: null graphics");
792 if (width < 1 || height < 1)

799 if (d < 0.f) d = 0.f;

804 rgba[i + 1] = b;
805 rgba[i + 2] = b;
806 rgba[i + 3] = 255;
807 }
808 }
809 return gfx->newTexture(width, height, rgba.data());
