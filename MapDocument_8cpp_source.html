29EditorResult<MapLayerRecord> parseLayer(const EditorValue& value) {
30 const auto* idValue = field(value, "id");
31 const auto* nameValue = field(value, "name");
32 const auto* kindValue = field(value, "kind");
33 const auto* visibleValue = field(value, "visible");
34 const auto* lockedValue = field(value, "locked");
35 const auto* opacityValue = field(value, "opacity");
36 const auto* orderValue = field(value, "order");
37
38 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
39 const auto* name = nameValue ? nameValue->getIf<std::string>() : nullptr;
40 const auto* kind = kindValue ? kindValue->getIf<std::string>() : nullptr;
41
42 const auto* visible = visibleValue ? visibleValue->getIf<bool>() : nullptr;
43 const auto* locked = lockedValue ? lockedValue->getIf<bool>() : nullptr;
44 const auto* opacity = opacityValue ? opacityValue->getIf<double>() : nullptr;
45 const auto* order = orderValue ? orderValue->getIf<int64_t>() : nullptr;
46
47 static const std::set<std::string> kinds{"tile", "object", "road", "group"};
48 if (!id || id->empty() || !name || name->empty() || !kind || !kinds.contains(*kind) || !visible || !locked ||
49 !opacity || *opacity < 0.0 || *opacity > 1.0 || !order)
50 return mapError<MapLayerRecord>(EditorStatus::Rejected, "editor.map.invalid-layer",
51 "Map layer has invalid identity, kind, opacity or order");
52 return eve::editing::applied<MapLayerRecord>(
53 {StableId(*id), *name, *kind, *visible, *locked, *opacity, static_cast<int>(*order)});
54}

87EditorResult<MapRoadRecord> parseRoad(const EditorValue& value) {
88 const auto* idValue = field(value, "id");
89 const auto* layerValueEntry = field(value, "layer");
90 const auto* nameValue = field(value, "name");
91 const auto* materialValue = field(value, "materialAsset");
92 const auto* closedValue = field(value, "closed");
93 const auto* pointsValue = field(value, "points");
94
95 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
96 const auto* layer = layerValueEntry ? layerValueEntry->getIf<std::string>() : nullptr;
97 const auto* name = nameValue ? nameValue->getIf<std::string>() : nullptr;
98 const auto* material = materialValue ? materialValue->getIf<std::string>() : nullptr;
99 const auto* closed = closedValue ? closedValue->getIf<bool>() : nullptr;
100 const auto* points = pointsValue ? pointsValue->getIf<EditorValue::Array>() : nullptr;
101
102 if (!id || id->empty() || !layer || layer->empty() || !name || name->empty() || !material || !closed || !points ||
103 points->size() < 2)
104 return mapError<MapRoadRecord>(EditorStatus::Rejected, "editor.map.invalid-road",
105 "Road requires identity, layer, name and at least two points");
106 MapRoadRecord result{StableId(*id), StableId(*layer), *name, *material, *closed, {}};
107 std::set<StableId> pointIds;
108 for (const EditorValue& pointValueEntry : *points) {
109 auto point = parsePoint(pointValueEntry);
110 if (!point.ok() || !pointIds.insert(point.value().id).second)
111 return mapError<MapRoadRecord>(EditorStatus::Rejected, "editor.map.duplicate-spline-point",
112 "Road spline point ids must be unique");
113 result.points.push_back(std::move(point).value());
114 }
115 if (result.closed && result.points.size() < 3)
116 return mapError<MapRoadRecord>(EditorStatus::Rejected, "editor.map.closed-road-point-count",
117 "Closed road requires at least three points");
118 return eve::editing::applied<MapRoadRecord>(std::move(result));
119}

136EditorResult<MapPlacementRecord> parsePlacement(const EditorValue& value) {
137 const auto string = [&](const char* key) -> const std::string* {
138 const EditorValue* entry = field(value, key);
139 return entry ? entry->getIf<std::string>() : nullptr;
140 };
141 const auto number = [&](const char* key) -> const double* {
142 const EditorValue* entry = field(value, key);
143 return entry ? entry->getIf<double>() : nullptr;
144 };
145 const auto* id = string("id");
146 const auto* layer = string("layer");
147 const auto* asset = string("asset");
148 const auto* x = number("x");
149 const auto* y = number("y");
150 const auto* z = number("z");
151 const auto* rx = number("rotationX");
152 const auto* ry = number("rotationY");
153 const auto* rz = number("rotationZ");
154 const auto* sx = number("scaleX");
155 const auto* sy = number("scaleY");
156 const auto* sz = number("scaleZ");
157 if (!id || id->empty() || !layer || layer->empty() || !asset || asset->empty() || !x || !y || !z || !rx || !ry ||
158 !rz || !sx || !sy || !sz || *sx <= 0.0 || *sy <= 0.0 || *sz <= 0.0)
159 return mapError<MapPlacementRecord>(EditorStatus::Rejected, "editor.map.invalid-placement",
160 "Placement requires ids, asset, complete TRS and positive scale");
161 for (const double* component : {x, y, z, rx, ry, rz, sx, sy, sz})
162 if (!std::isfinite(*component))
163 return mapError<MapPlacementRecord>(EditorStatus::Rejected, "editor.map.nonfinite-placement",
164 "Placement transform must be finite");
165 return eve::editing::applied<MapPlacementRecord>(
166 {StableId(*id), StableId(*layer), *asset, *x, *y, *z, *rx, *ry, *rz, *sx, *sy, *sz});
167}

200EditorResult<void> MapDocumentTarget::applyDomainOperation(const DomainOperation& domainOperation) {
201 if (domainOperation.target != TargetId(id_))
202 return mapError<void>(EditorStatus::Rejected, "editor.map.target-mismatch",
203 "Map operation targets another document");
204 if (domainOperation.type == "map.layer.create.v1") {
205 auto layer = parseLayer(domainOperation.payload);
206 if (!layer.ok()) return mapError<void>(layer.code(), "editor.map.invalid-layer", "Layer payload is invalid");
207 if (layers_.contains(layer.value().id))
208 return mapError<void>(EditorStatus::Conflict, "editor.map.layer-exists", "Map layer already exists");
209 for (const auto& [id, current] : layers_) {
210 static_cast<void>(id);
211 if (current.order == layer.value().order)
212 return mapError<void>(EditorStatus::Conflict, "editor.map.layer-order-conflict",
213 "Map layer visual order is already occupied");
214 }
215 auto layerValue = std::move(layer).value();
216 layers_.emplace(layerValue.id, std::move(layerValue));
217 } else if (domainOperation.type == "map.layer.delete.v1") {
218 auto layer = parseLayer(domainOperation.payload);
219 if (!layer.ok() || !layers_.contains(layer.value().id))
220 return mapError<void>(EditorStatus::NotFound, "editor.map.layer-not-found", "Map layer was not found");
221 if (layers_.at(layer.value().id).locked)
222 return mapError<void>(EditorStatus::Rejected, "editor.map.layer-locked", "Map layer is locked");
223 for (const auto& [id, road] : roads_) {
224 static_cast<void>(id);
225 if (road.layer == layer.value().id)
226 return mapError<void>(EditorStatus::Rejected, "editor.map.layer-not-empty",
227 "Map layer still contains roads");
228 }
229 for (const auto& [id, placement] : placements_) {
230 static_cast<void>(id);
231 if (placement.layer == layer.value().id)
232 return mapError<void>(EditorStatus::Rejected, "editor.map.layer-not-empty",
233 "Map layer still contains placements");
234 }
235 layers_.erase(layer.value().id);
236 } else if (domainOperation.type == "map.layer.set.v1") {
237 auto layer = parseLayer(domainOperation.payload);
238 if (!layer.ok() || !layers_.contains(layer.value().id))
239 return mapError<void>(EditorStatus::NotFound, "editor.map.layer-not-found", "Map layer was not found");
240 for (const auto& [id, current] : layers_)
241 if (id != layer.value().id && current.order == layer.value().order)
242 return mapError<void>(EditorStatus::Conflict, "editor.map.layer-order-conflict",
243 "Map layer visual order is already occupied");
244 if (layers_.at(layer.value().id).kind != layer.value().kind) {
245 for (const auto& [id, road] : roads_) {
246 static_cast<void>(id);
247 if (road.layer == layer.value().id)
248 return mapError<void>(EditorStatus::Rejected, "editor.map.layer-kind-in-use",
249 "Cannot change kind of a layer containing roads");
250 }
251 for (const auto& [id, placement] : placements_) {
252 static_cast<void>(id);
253 if (placement.layer == layer.value().id)
254 return mapError<void>(EditorStatus::Rejected, "editor.map.layer-kind-in-use",
255 "Cannot change kind of a layer containing placements");
256 }
257 }
258 auto layerValue = std::move(layer).value();
259 layers_[layerValue.id] = std::move(layerValue);
260 } else if (domainOperation.type == "map.road.set.v1") {
261 auto road = parseRoad(domainOperation.payload);
262 if (!road.ok()) return mapError<void>(road.code(), "editor.map.invalid-road", "Road payload is invalid");
263 const auto layer = layers_.find(road.value().layer);
264 if (layer == layers_.end() || layer->second.kind != "road")
265 return mapError<void>(EditorStatus::Rejected, "editor.map.road-layer-required",
266 "Road must reference a road layer");
267 if (layer->second.locked)
268 return mapError<void>(EditorStatus::Rejected, "editor.map.layer-locked", "Road layer is locked");
269 const auto existing = roads_.find(road.value().id);
270 if (existing != roads_.end() && layers_.at(existing->second.layer).locked)
271 return mapError<void>(EditorStatus::Rejected, "editor.map.layer-locked",
272 "Road cannot be moved out of a locked layer");
273 auto roadValue = std::move(road).value();
274 roads_[roadValue.id] = std::move(roadValue);
275 } else if (domainOperation.type == "map.road.delete.v1") {
276 auto road = parseRoad(domainOperation.payload);
277 if (!road.ok() || !roads_.contains(road.value().id))
278 return mapError<void>(EditorStatus::NotFound, "editor.map.road-not-found", "Road was not found");
279 if (layers_.at(roads_.at(road.value().id).layer).locked)
280 return mapError<void>(EditorStatus::Rejected, "editor.map.layer-locked", "Road layer is locked");
281 roads_.erase(road.value().id);
282 } else if (domainOperation.type == "map.placement.set.v1") {
283 auto placement = parsePlacement(domainOperation.payload);
284 if (!placement.ok())
285 return mapError<void>(placement.code(), "editor.map.invalid-placement", "Placement payload is invalid");
286 const auto layer = layers_.find(placement.value().layer);
287 if (layer == layers_.end() || layer->second.kind != "object")
288 return mapError<void>(EditorStatus::Rejected, "editor.map.object-layer-required",
289 "Placement must reference an object layer");
290 if (layer->second.locked)
291 return mapError<void>(EditorStatus::Rejected, "editor.map.layer-locked", "Object layer is locked");
292 const auto existing = placements_.find(placement.value().id);
293 if (existing != placements_.end() && layers_.at(existing->second.layer).locked)
294 return mapError<void>(EditorStatus::Rejected, "editor.map.layer-locked",
295 "Placement cannot be moved out of a locked layer");
296 auto placementValue = std::move(placement).value();
297 placements_[placementValue.id] = std::move(placementValue);
298 } else if (domainOperation.type == "map.placement.delete.v1") {
299 auto placement = parsePlacement(domainOperation.payload);
300 if (!placement.ok() || !placements_.contains(placement.value().id))
301 return mapError<void>(EditorStatus::NotFound, "editor.map.placement-not-found", "Placement was not found");
302 if (layers_.at(placements_.at(placement.value().id).layer).locked)
303 return mapError<void>(EditorStatus::Rejected, "editor.map.layer-locked", "Object layer is locked");
304 placements_.erase(placement.value().id);
305 } else {
306 return mapError<void>(EditorStatus::Unsupported, "editor.map.operation-unsupported",
307 "Map operation is unsupported: " + domainOperation.type);
308 }
309 ++revision_;
310 dirty_.include(0, 0);
311 return eve::editing::applied<void>();
312}

467 return mapError<DomainOperation>(EditorStatus::Rejected, "editor.map.layer-locked", "Object layer is locked");
468 const auto found = placements_.find(placement.id);
469 if (found != placements_.end() && layers_.at(found->second.layer).locked)
470 return mapError<DomainOperation>(EditorStatus::Rejected, "editor.map.layer-locked",
471 "Placement cannot be moved out of a locked layer");
472 const EditorValue inverse = found == placements_.end() ? placementValue(placement) : placementValue(found->second);
473 const std::string inverseType = found == placements_.end() ? "map.placement.delete.v1" : "map.placement.set.v1";
474 return eve::editing::applied<DomainOperation>(
475 operation("map.placement.set.v1", inverseType, id_, placementValue(placement), inverse, placement.id));
476}

490std::vector<EditorDiagnostic> MapDocumentTarget::validate() const {
491 std::vector<EditorDiagnostic> diagnostics;
492 std::set<int> orders;
493 for (const auto& [id, layer] : layers_) {
494 static_cast<void>(id);
495 if (!orders.insert(layer.order).second)
496 diagnostics.push_back(eve::editing::ruleDiagnostic(
497 eve::DiagnosticCode::InvariantViolation, RuleId("editor.map.duplicate-layer-order"),
498 DiagnosticSeverity::Error, "Multiple map layers share visual order " + std::to_string(layer.order)));
499 }
500 for (const auto& [id, road] : roads_) {
501 const auto layer = layers_.find(road.layer);
502 if (layer == layers_.end() || layer->second.kind != "road")
503 diagnostics.push_back(eve::editing::ruleDiagnostic(
504 eve::DiagnosticCode::InvariantViolation, RuleId("editor.map.invalid-road-layer"),
505 DiagnosticSeverity::Error, "Road references a missing or non-road layer: " + id.value()));
506 double lengthSquared = 0.0;
507 for (size_t index = 1; index < road.points.size(); ++index) {
508 const double dx = road.points[index].x - road.points[index - 1].x;
509 const double dy = road.points[index].y - road.points[index - 1].y;
510 const double dz = road.points[index].z - road.points[index - 1].z;
511 lengthSquared += dx * dx + dy * dy + dz * dz;
512 }
513 if (lengthSquared <= 1e-12)
514 diagnostics.push_back(eve::editing::ruleDiagnostic(
515 eve::DiagnosticCode::InvariantViolation, RuleId("editor.map.zero-length-road"),
516 DiagnosticSeverity::Error, "Road has no geometric length: " + id.value()));
517 if (road.materialAsset.empty())
518 diagnostics.push_back(eve::editing::ruleDiagnostic(
519 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.map.road-material-missing"),
520 DiagnosticSeverity::Warning, "Road has no material asset: " + id.value()));
521 }
522 for (const auto& [id, placement] : placements_) {
523 const auto layer = layers_.find(placement.layer);
524 if (layer == layers_.end() || layer->second.kind != "object")
525 diagnostics.push_back(eve::editing::ruleDiagnostic(
526 eve::DiagnosticCode::InvariantViolation, RuleId("editor.map.invalid-placement-layer"),
527 DiagnosticSeverity::Error, "Placement references a missing or non-object layer: " + id.value()));
528 }
529 return diagnostics;
530}

532MapRoadPreviewResult MapDocumentTarget::previewRoad(const StableId& roadId, int triangleBudget) const {
533 MapRoadPreviewResult result;
534 result.documentRevision = revision_;
535 result.road = roadId;
536 const auto found = roads_.find(roadId);
537 if (found == roads_.end()) {
538 result.status = EditorStatus::NotFound;
539 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
540 eve::DiagnosticCode::NotFound, RuleId("editor.map.road-not-found"), DiagnosticSeverity::Error,
541 "Road preview target was not found"));
542 return result;
543 }
544 if (triangleBudget <= 0) {
545 result.status = EditorStatus::Rejected;
546 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
547 eve::DiagnosticCode::InvalidArgument, RuleId("editor.map.invalid-preview-budget"),
548 DiagnosticSeverity::Error, "Road preview triangle budget must be positive"));
549 return result;
550 }
551 const MapRoadRecord& road = found->second;
552 const int segmentCount = static_cast<int>(road.points.size()) - (road.closed ? 0 : 1);
553 const int triangleCount = segmentCount * 2;
554 if (triangleCount > triangleBudget) {
555 result.status = EditorStatus::Rejected;
556 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
557 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.map.road-preview-budget"),
558 DiagnosticSeverity::Error, "Road strip exceeds preview triangle budget"));
559 return result;
560 }
561 EditorValue::Array vertices;
562 EditorValue::Array indices;
563 for (size_t index = 0; index < road.points.size(); ++index) {
564 const MapSplinePointRecord& point = road.points[index];
565 const MapSplinePointRecord& previous =
566 road.points[index == 0 ? (road.closed ? road.points.size() - 1 : 0) : index - 1];
567 const MapSplinePointRecord& next =
568 road.points[index + 1 == road.points.size() ? (road.closed ? 0 : road.points.size() - 1) : index + 1];
569 double tangentX = next.x - previous.x;
570 double tangentZ = next.z - previous.z;
571 const double length = std::sqrt(tangentX * tangentX + tangentZ * tangentZ);
572 if (length <= 1e-9) {
573 result.status = EditorStatus::Failed;
574 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
575 eve::DiagnosticCode::InvariantViolation, RuleId("editor.map.degenerate-road-tangent"),
576 DiagnosticSeverity::Error,
577 "Road preview contains a degenerate tangent at point " + point.id.value()));
578 return result;
579 }
580 tangentX /= length;
581 tangentZ /= length;
582 const double halfWidth = point.width * 0.5;
583 vertices.emplace_back(
584 EditorValue::Array{point.x - tangentZ * halfWidth, point.y, point.z + tangentX * halfWidth});
585 vertices.emplace_back(
586 EditorValue::Array{point.x + tangentZ * halfWidth, point.y, point.z - tangentX * halfWidth});
587 if (index > 0) {
588 const int64_t base = static_cast<int64_t>(index * 2);
589 for (const int64_t value : {base - 2, base - 1, base, base, base - 1, base + 1})
590 indices.emplace_back(value);
591 const double dx = point.x - previous.x;
592 const double dy = point.y - previous.y;
593 const double dz = point.z - previous.z;
594 result.centerlineLength += std::sqrt(dx * dx + dy * dy + dz * dz);
595 }
596 }
597 if (road.closed) {
598 const int64_t last = static_cast<int64_t>(road.points.size() * 2 - 2);
599 for (const int64_t value : {last, last + 1, int64_t{0}, int64_t{0}, last + 1, int64_t{1}})
600 indices.emplace_back(value);
601 const MapSplinePointRecord& first = road.points.front();
602 const MapSplinePointRecord& lastPoint = road.points.back();
603 const double dx = first.x - lastPoint.x;
604 const double dy = first.y - lastPoint.y;
605 const double dz = first.z - lastPoint.z;
606 result.centerlineLength += std::sqrt(dx * dx + dy * dy + dz * dz);
607 }
608 result.mesh = EditorValue::Object{{"primitive", "triangles"},
609 {"vertices", std::move(vertices)},
610 {"indices", std::move(indices)},
611 {"materialAsset", road.materialAsset}};
612 result.status = EditorStatus::Applied;
613 return result;
614}

629EditorResult<void> MapDocumentTarget::loadSnapshot(const EditorValue& snapshot) {
630 const auto* versionValue = field(snapshot, "schemaVersion");
631 const auto* layersValue = field(snapshot, "layers");
632 const auto* roadsValue = field(snapshot, "roads");
633 const auto* placementsValue = field(snapshot, "placements");
634 const auto* version = versionValue ? versionValue->getIf<int64_t>() : nullptr;
635 const auto* layers = layersValue ? layersValue->getIf<EditorValue::Array>() : nullptr;
636 const auto* roads = roadsValue ? roadsValue->getIf<EditorValue::Array>() : nullptr;
637 const auto* placements = placementsValue ? placementsValue->getIf<EditorValue::Array>() : nullptr;
638 if (!version || *version != 1 || !layers || !roads || !placements)
639 return mapError<void>(EditorStatus::Unsupported, "editor.map.invalid-snapshot",
640 "Map snapshot schema is invalid or unsupported");
641 MapDocumentTarget candidate(id_);
642 for (const EditorValue& value : *layers) {
643 auto layer = parseLayer(value);
644 if (!layer.ok()) return mapError<void>(EditorStatus::Rejected, "editor.map.invalid-layer", "Invalid layer");
645 auto create = candidate.makeCreateLayer(layer.value());
646 if (!create.ok() || !candidate.applyDomainOperation(create.value()).ok())
647 return mapError<void>(EditorStatus::Rejected, "editor.map.invalid-layer-set",
648 "Map snapshot contains conflicting layers");
649 }
650 for (const EditorValue& value : *roads) {
651 auto road = parseRoad(value);
652 if (!road.ok()) return mapError<void>(EditorStatus::Rejected, "editor.map.invalid-road", "Invalid road");
653 auto set = candidate.makeSetRoad(road.value());
654 if (!set.ok() || !candidate.applyDomainOperation(set.value()).ok())
655 return mapError<void>(EditorStatus::Rejected, "editor.map.invalid-road-set",
656 "Map snapshot contains an invalid road layer reference");
657 }
658 for (const EditorValue& value : *placements) {
659 auto placement = parsePlacement(value);
660 if (!placement.ok())
661 return mapError<void>(EditorStatus::Rejected, "editor.map.invalid-placement", "Invalid placement");
662 auto set = candidate.makeSetPlacement(placement.value());
663 if (!set.ok() || !candidate.applyDomainOperation(set.value()).ok())
664 return mapError<void>(EditorStatus::Rejected, "editor.map.invalid-placement-set",
665 "Map snapshot contains an invalid placement layer reference");
666 }
667 layers_.swap(candidate.layers_);
668 roads_.swap(candidate.roads_);
669 placements_.swap(candidate.placements_);
670 ++revision_;
671 dirty_.clear();
672 return eve::editing::applied<void>();
673}
