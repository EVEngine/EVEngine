82MeshBuild buildTiles(const Grid2D& grid, float cellSize, float height, std::string_view group, int minX, int minY,
83 int maxX, int maxY) {
84 MeshBuild mesh;
85 minX = std::clamp(minX, 0, grid.getWidth());
86 minY = std::clamp(minY, 0, grid.getHeight());
87 maxX = std::clamp(maxX, minX, grid.getWidth());
88 maxY = std::clamp(maxY, minY, grid.getHeight());
89 for (int y = minY; y < maxY; ++y) {
90 for (int x = minX; x < maxX; ++x) {
91 if (grid.getCell(x, y) == int(Semantic::Empty)) continue;
92 const auto preset = resolveTilePreset(std::uint8_t(grid.getDetail(x, y)));
93 const std::string cellGroup =
94 (group.empty() ? std::string("tiles") : std::string(group)) + "/" + std::to_string(grid.getCell(x, y)) +
95 "/" + std::string(tilePresetKindName(preset.kind)) + "/r" + std::to_string(preset.rotationDegrees) +
96 (preset.mirrorX ? "/mx/" : "/normal/") + std::to_string(preset.configuration);
97 mesh.setActiveGroup(cellGroup);
98 const float x0 = float(x) * cellSize;
99 const float x1 = x0 + cellSize;
100 const float z0 = float(y) * cellSize;
101 const float z1 = z0 + cellSize;
102 addQuad(mesh, x0, height, z0, x1, height, z0, x1, height, z1, x0, height, z1, 0.f, 1.f, 0.f);
103 if (grid.getCell(x - 1, y) == int(Semantic::Empty))
104 addQuad(mesh, x0, 0.f, z1, x0, 0.f, z0, x0, height, z0, x0, height, z1, -1.f, 0.f, 0.f);
105 if (grid.getCell(x + 1, y) == int(Semantic::Empty))
106 addQuad(mesh, x1, 0.f, z0, x1, 0.f, z1, x1, height, z1, x1, height, z0, 1.f, 0.f, 0.f);
107 if (grid.getCell(x, y - 1) == int(Semantic::Empty))
108 addQuad(mesh, x0, 0.f, z0, x1, 0.f, z0, x1, height, z0, x0, height, z0, 0.f, 0.f, -1.f);
109 if (grid.getCell(x, y + 1) == int(Semantic::Empty))
110 addQuad(mesh, x1, 0.f, z1, x0, 0.f, z1, x0, height, z1, x1, height, z1, 0.f, 0.f, 1.f);
111 }
112 }
113 mesh.setMeta("source", "mesh.grid_tiles");
114 return mesh;
115}

132Result<void> MeshGraph::connect(std::string_view fromId, std::string_view toId, int inputIndex) {
133 const auto from = nodes_.find(std::string(fromId));
134 const auto to = nodes_.find(std::string(toId));
135 if (from == nodes_.end() || to == nodes_.end())
136 return failureVoid(DiagnosticCode::NotFound, "connection references an unknown node", std::string(toId));
137 const Spec* source = specFor(from->second.operation);
138 const Spec* target = specFor(to->second.operation);
139 if (inputIndex < 0 || inputIndex >= target->inputs)
140 return failureVoid(DiagnosticCode::InvalidArgument, "input slot is outside the operation contract",
141 std::string(toId));
142 if (source->output != target->input[inputIndex])
143 return failureVoid(DiagnosticCode::TypeMismatch, "typed graph ports are incompatible", std::string(toId));
144 to->second.inputs[inputIndex] = std::string(fromId);
145 invalidateFrom(toId);
146 ++revision_;
147 return Result<void>::success();
148}

209Result<MeshGraphValue> MeshGraph::evaluate(std::string_view id, std::unordered_map<std::string, int>& states) {
210 const auto found = nodes_.find(std::string(id));
211 if (found == nodes_.end())
212 return failure<MeshGraphValue>(DiagnosticCode::NotFound, "unknown node", std::string(id));
213 Node& node = found->second;
214 if (node.cacheValid) return Result<MeshGraphValue>::success(node.cache);
215 if (states[node.id] == 1)
216 return failure<MeshGraphValue>(DiagnosticCode::Conflict, "graph contains a cycle", node.id);
217 states[node.id] = 1;
218 const Spec* spec = specFor(node.operation);
219 std::vector<MeshGraphValue> inputs;
220 for (int input = 0; input < spec->inputs; ++input) {
221 if (node.inputs[input].empty())
222 return failure<MeshGraphValue>(DiagnosticCode::NotFound, "required input is disconnected", node.id);
223 auto value = evaluate(node.inputs[input], states);
224 if (!value.ok()) return value;
225 if (typeOf(value.value()) != spec->input[input])
226 return failure<MeshGraphValue>(DiagnosticCode::TypeMismatch, "runtime port type mismatch", node.id);
227 inputs.push_back(std::move(value).takeValue());
228 }
229 if (spec->inputs == 0) {
230 if (!node.hasValue)
231 return failure<MeshGraphValue>(DiagnosticCode::NotFound, "input node has no value", node.id);
232 node.cache = node.value;
233 } else if (node.operation == "mesh.grid_tiles") {
234 const auto& grid = std::get<Grid2D>(inputs[0]);
235 node.cache = buildTiles(
236 grid, node.floats.contains("cellSize") ? node.floats["cellSize"] : 1.f,
237 node.floats.contains("height") ? node.floats["height"] : 1.f,
238 node.strings.contains("group") ? node.strings["group"] : "tiles",
239 node.floats.contains("minX") ? int(node.floats["minX"]) : 0,
240 node.floats.contains("minY") ? int(node.floats["minY"]) : 0,
241 node.floats.contains("maxX") ? int(node.floats["maxX"]) : grid.getWidth(),
242 node.floats.contains("maxY") ? int(node.floats["maxY"]) : grid.getHeight());
243 } else if (node.operation == "mesh.merge") {
244 MeshBuild result = std::get<MeshBuild>(inputs[0]);
245 const MeshBuild& other = std::get<MeshBuild>(inputs[1]);
246 if (!result.appendTransformed(&other, 0.f, 0.f, 0.f, 0.f, 1.f, 1.f, 1.f))
247 return failure<MeshGraphValue>(DiagnosticCode::Failed, "mesh merge failed", node.id);
248 node.cache = std::move(result);
249 } else if (node.operation == "mesh.transform") {
250 MeshBuild result;
251 const MeshBuild& source = std::get<MeshBuild>(inputs[0]);
252 if (!result.appendTransformed(&source, node.floats["x"], node.floats["y"], node.floats["z"], node.floats["yaw"],
253 node.floats.contains("sx") ? node.floats["sx"] : 1.f,
254 node.floats.contains("sy") ? node.floats["sy"] : 1.f,
255 node.floats.contains("sz") ? node.floats["sz"] : 1.f))
256 return failure<MeshGraphValue>(DiagnosticCode::InvalidArgument, "mesh transform failed", node.id);
257 node.cache = std::move(result);
258 } else if (node.operation == "mesh.instance_points") {
259 MeshBuild result;
260 const MeshBuild& source = std::get<MeshBuild>(inputs[0]);
261 const PointSet& points = std::get<PointSet>(inputs[1]);
262 const float scale = node.floats.contains("scale") ? node.floats["scale"] : 1.f;
263 for (int index = 0; index < points.getCount(); ++index)
264 if (!result.appendTransformed(&source, points.getX(index), points.getY(index), points.getZ(index), 0.f,
265 scale, scale, scale))
266 return failure<MeshGraphValue>(DiagnosticCode::InvalidArgument, "point instancing failed", node.id);
267 node.cache = std::move(result);
268 }
269 states[node.id] = 2;
270 node.cacheValid = true;
271 return Result<MeshGraphValue>::success(node.cache);
272}

295void MeshGraph::invalidateFrom(std::string_view id) {
296 std::queue<std::string> pending;
297 std::unordered_set<std::string> visited;
298 pending.push(std::string(id));
299 while (!pending.empty()) {
300 const std::string current = pending.front();
301 pending.pop();
302 if (!visited.insert(current).second) continue;
303 const auto found = nodes_.find(current);
304 if (found != nodes_.end()) found->second.cacheValid = false;
305 for (const auto& [otherId, node] : nodes_)
306 if (node.inputs[0] == current || node.inputs[1] == current) pending.push(otherId);
307 }

310std::string MeshGraph::serializeDefinition() const {
311 std::ostringstream out;
312 out << "EVPCG_MESH_GRAPH 1\n" << std::setprecision(9);
313 for (const auto& id : order_) {
314 const auto& node = nodes_.at(id);
315 out << "NODE " << std::quoted(id) << ' ' << std::quoted(node.operation) << '\n';
316 std::vector<std::string> keys;
317 for (const auto& [key, value] : node.floats) keys.push_back(key);
318 std::sort(keys.begin(), keys.end());
319 for (const auto& key : keys)
320 out << "FLOAT " << std::quoted(id) << ' ' << std::quoted(key) << ' ' << node.floats.at(key) << '\n';
321 keys.clear();
322 for (const auto& [key, value] : node.strings) keys.push_back(key);
323 std::sort(keys.begin(), keys.end());
324 for (const auto& key : keys)
325 out << "STRING " << std::quoted(id) << ' ' << std::quoted(key) << ' ' << std::quoted(node.strings.at(key))
326 << '\n';
327 }
328 for (const auto& id : order_) {
329 const auto& node = nodes_.at(id);
330 for (int input = 0; input < 2; ++input)
331 if (!node.inputs[input].empty())
332 out << "EDGE " << std::quoted(node.inputs[input]) << ' ' << std::quoted(id) << ' ' << input << '\n';
333 }
334 out << "END\n";
335 return out.str();
336}

338Result<void> MeshGraph::deserializeDefinition(std::string_view definition) {
339 MeshGraph replacement;
340 std::istringstream input{std::string(definition)};
341 std::string magic;
342 int version = 0;
343 if (!(input >> magic >> version) || magic != "EVPCG_MESH_GRAPH" || version != 1)
344 return failureVoid(DiagnosticCode::ParseError, "invalid mesh graph header");
345 std::string line;
346 std::getline(input, line);
347 bool ended = false;
348 while (std::getline(input, line)) {
349 if (line.empty()) continue;
350 std::istringstream record(line);
351 std::string kind;
352 record >> kind;
353 if (kind == "END") {
354 ended = true;
355 break;
356 }
357 std::string id;
358 auto result = [&]() -> Result<void> {
359 if (kind == "NODE") {
360 std::string operation;
361 if (!(record >> std::quoted(id) >> std::quoted(operation)))
362 return failureVoid(DiagnosticCode::ParseError, "invalid NODE record");
363 return replacement.addNode(std::move(id), std::move(operation));
364 }
365 if (kind == "EDGE") {
366 std::string target;
367 int slot = -1;
368 if (!(record >> std::quoted(id) >> std::quoted(target) >> slot))
369 return failureVoid(DiagnosticCode::ParseError, "invalid EDGE record");
370 return replacement.connect(id, target, slot);
371 }
372 if (kind == "FLOAT") {
373 std::string key;
374 float value = 0.f;
375 if (!(record >> std::quoted(id) >> std::quoted(key) >> value))
376 return failureVoid(DiagnosticCode::ParseError, "invalid FLOAT record");
377 return replacement.setNodeFloat(id, std::move(key), value);
378 }
379 if (kind == "STRING") {
380 std::string key;
381 std::string value;
382 if (!(record >> std::quoted(id) >> std::quoted(key) >> std::quoted(value)))
383 return failureVoid(DiagnosticCode::ParseError, "invalid STRING record");
384 return replacement.setNodeString(id, std::move(key), std::move(value));
385 }
386 return failureVoid(DiagnosticCode::ParseError, "unknown mesh graph record: " + kind);
387 }();
388 if (!result.ok()) return result;
389 record >> std::ws;
390 if (!record.eof()) return failureVoid(DiagnosticCode::ParseError, "trailing mesh graph record data");
391 }
392 if (!ended) return failureVoid(DiagnosticCode::ParseError, "mesh graph END record is missing");
393
394 std::unordered_map<std::string, int> states;
395 const auto visit = [&](const auto& self, const std::string& id) -> Result<void> {
396 if (states[id] == 2) return Result<void>::success();
397 if (states[id] == 1) return failureVoid(DiagnosticCode::Conflict, "cycle in serialized graph", id);
398 states[id] = 1;
399 const auto found = replacement.nodes_.find(id);
400 if (found == replacement.nodes_.end()) return failureVoid(DiagnosticCode::NotFound, "unknown node", id);
401 for (const auto& dependency : found->second.inputs) {
402 if (dependency.empty()) continue;
403 auto result = self(self, dependency);
404 if (!result.ok()) return result;
405 }
406 states[id] = 2;
407 return Result<void>::success();
408 };
409 for (const auto& id : replacement.order_) {
410 auto result = visit(visit, id);
411 if (!result.ok()) return result;
412 }
413 *this = std::move(replacement);
414 return Result<void>::success();
415}
