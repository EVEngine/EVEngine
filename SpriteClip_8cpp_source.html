51float SpriteClip::getFPS() const {
52 if (frames_.empty()) return 0.f;
53 const float duration = frames_.front().duration;
54 if (duration <= 0.f) return 0.f;
55 for (const Entry &entry : frames_) {
56 if (std::fabs(entry.duration - duration) > 1e-6f) return 0.f;
57 }
58 return 1.f / duration;
59}

89float SpriteClip::wrapTime(float timeSeconds) const {
90 if (timeSeconds < 0.f) timeSeconds = 0.f;
91 const float dur = getDuration();
92 if (dur <= 0.f) return 0.f;
93 if (loop_) {
94 float t = std::fmod(timeSeconds, dur);
95 if (t < 0.f) t += dur;
96 return t;
97 }
98 return timeSeconds > dur ? dur : timeSeconds;
99}

101int SpriteClip::frameAtTime(float timeSeconds) const {
102 if (frames_.empty()) return -1;
103 float t = wrapTime(timeSeconds);
104 const float dur = getDuration();
105 if (!loop_ && timeSeconds >= dur) return getFrameCount() - 1;
106
107 float acc = 0.f;
108 for (int i = 0; i < getFrameCount(); ++i) {
109 acc += frames_[static_cast<size_t>(i)].duration;
110 if (t < acc || i == getFrameCount() - 1) return i;
111 }
112 return getFrameCount() - 1;
113}
