35 [[nodiscard]] Result<ContentId> hash(std::string_view canonicalInput) const override {
36 std::uint64_t left = 14695981039346656037ull;
37 std::uint64_t right = 1099511628211ull;
38 for (const unsigned char byte : canonicalInput) {
39 left = (left ^ byte) * 1099511628211ull;
40 right = (right ^ (static_cast<std::uint64_t>(byte) + 0x9e3779b97f4a7c15ull)) * 14029467366897019727ull;
41 }
42 ContentId::Bytes bytes{};
43 for (int index = 0; index < 8; ++index) {
44 bytes[static_cast<std::size_t>(index)] =
45 static_cast<std::uint8_t>(left >> static_cast<unsigned>(56 - index * 8));
46 bytes[static_cast<std::size_t>(index + 8)] =
47 static_cast<std::uint8_t>(right >> static_cast<unsigned>(56 - index * 8));
48 }
49 return Result<ContentId>::success(ContentId(bytes));
50 }
51};
