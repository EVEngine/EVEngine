64eve::Result<void> AnimCurveLibrary::load(std::span<const std::byte> bytes) {
65 try {
66 if (bytes.size() > 64 * 1024 * 1024) throw std::runtime_error("animation curves exceed 64 MiB");
67 Reader reader(bytes);
68 if (reader.integer() != 0x43465645 || reader.integer() != 1)
69 throw std::runtime_error("unsupported animation curves format");
70 auto count = reader.integer();
71 if (count > 100000) throw std::runtime_error("too many animation curve sources");
72 Impl next;
73 for (std::uint32_t i = 0; i < count; ++i) {
74 auto source = reader.string();
75 Clip clip;
76 clip.duration = reader.number();
77 auto channelCount = reader.integer();
78 if (channelCount > 64 || clip.duration < 0 || (channelCount && clip.duration <= 0))
79 throw std::runtime_error("invalid animation curve duration or channel count");
80 for (std::uint32_t j = 0; j < channelCount; ++j) {
81 auto name = reader.string();
82 auto keyCount = reader.integer();
83 if (keyCount < 2 || keyCount > 1000000) throw std::runtime_error("invalid animation curve key count");
84 reader.require(std::size_t(keyCount) * 8);
85 std::vector<Key> keys;
86 keys.reserve(keyCount);
87 for (std::uint32_t k = 0; k < keyCount; ++k) {
88 Key key{reader.number(), reader.number()};
89 if (key.time < 0 || key.time > clip.duration || (!keys.empty() && key.time <= keys.back().time))
90 throw std::runtime_error("unordered animation curve time");
91 keys.push_back(key);
92 }
93 if (keys.front().time != 0 || keys.back().time != clip.duration)
94 throw std::runtime_error("animation curve endpoints do not span duration");
95 if (!clip.channels.emplace(name, std::move(keys)).second)
96 throw std::runtime_error("duplicate animation curve channel");
97 }
98 if (!next.clips.emplace(source, std::move(clip)).second)
99 throw std::runtime_error("duplicate animation curve source");
100 }
101 if (!reader.finished()) throw std::runtime_error("trailing animation curve bytes");
102 impl_->clips.swap(next.clips);
103 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
104 } catch (const std::exception& error) {
105 return eve::Result<void>::failure(invalid(error.what()));
106 }
