29Pixel sampleRepeat(const image::ImageData& image, float u, float v) {
30 const int width = image.getWidth();
31 const int height = image.getHeight();
32 const float x = wrap01(u) * float(width) - 0.5f;
33 const float y = wrap01(v) * float(height) - 0.5f;
34 const int x0 = int(std::floor(x));
35 const int y0 = int(std::floor(y));
36 const float tx = x - float(x0);
37 const float ty = y - float(y0);
38 auto pixel = [&](int px, int py) {
39 px = (px % width + width) % width;
40 py = (py % height + height) % height;
41 return image.getPixel(px, py);
42 };
43 const Pixel a = pixel(x0, y0);
44 const Pixel b = pixel(x0 + 1, y0);
45 const Pixel c = pixel(x0, y0 + 1);
46 const Pixel d = pixel(x0 + 1, y0 + 1);
47 auto mix = [tx, ty](float av, float bv, float cv, float dv) {
48 return (av + (bv - av) * tx) + ((cv + (dv - cv) * tx) - (av + (bv - av) * tx)) * ty;
49 };
50 return Pixel{mix(a.r, b.r, c.r, d.r), mix(a.g, b.g, c.g, d.g), mix(a.b, b.b, c.b, d.b), mix(a.a, b.a, c.a, d.a)};
51}

55std::size_t VirtualTextureStreaming::PageKeyHash::operator()(const PageKey& key) const noexcept {
56 std::size_t seed = std::hash<int>{}(key.x);
57 seed ^= std::hash<int>{}(key.y) + 0x9e3779b9u + (seed << 6u) + (seed >> 2u);
58 return seed;

95eve::Result<void> VirtualTextureStreaming::initializePhysicalCache() {
96 if (producer_->pageCountX() <= 0 || producer_->pageCountY() <= 0)
97 return eve::Result<void>::failure(invalid("virtual texture producer has no pages"));
98 auto bootstrap = producer_->buildResidentSet({});
99 if (!bootstrap.ok()) return eve::Result<void>::failure(bootstrap.status());
100
101 const auto& set = bootstrap.value();
102 pageExtent_ = set.pageExtent;
103 borderFraction_ = set.borderFraction;
104 atlasSlotsX_ = set.atlasSlotsX;
105 atlasSlotsY_ = set.atlasSlotsY;
106 const int slotCount = atlasSlotsX_ * atlasSlotsY_;
107
108 auto makeState = [&](PhysicalState& state) {
109 state.albedo = std::make_unique<image::ImageData>(*set.albedoAtlas);
110 state.normal = std::make_unique<image::ImageData>(*set.normalAtlas);
111 state.pageTable = std::make_unique<image::ImageData>(*set.pageTable);
112 state.slots.assign(std::size_t(slotCount), SlotRecord{});
113 state.lru.clear();
114 state.pageToSlot.clear();
115 // Slot 0 is reserved for the whole-surface substitute and is never LRU-managed.
116 state.slots[0].occupied = true;
117 state.slots[0].pageX = -1;
118 state.slots[0].pageY = -1;
119 };
120 makeState(published_);
121 makeState(working_);
122 pending_.reset();
123 gpuAlbedo_ = gpuNormal_ = gpuPageTable_ = nullptr;
124 generation_ = 0;
125 initialized_ = true;
126 lifetime_ = {};
127 lifetime_.generation = 0;
128 lifetime_.residentPages = 0;
129 return eve::Result<void>::success();
130}

182eve::Result<void> VirtualTextureStreaming::ingestPackedFeedback(std::span<const std::uint32_t> packed) {
183 auto ready = ensureInitialized();
184 if (!ready.ok()) return ready;
185 for (std::uint32_t word : packed) {
186 auto decoded = unpackFeedback(word);
187 if (!decoded.ok()) continue;
188 auto sample = decoded.value();
189 // Higher mips collapse onto their mip-zero parents for the current physical pool.
190 if (sample.mipLevel != 0) {
191 sample.pageX >>= sample.mipLevel;
192 sample.pageY >>= sample.mipLevel;
193 sample.mipLevel = 0;
194 }
195 auto recorded = recordFeedback(sample.pageX, sample.pageY, sample.mipLevel);
196 if (!recorded.ok()) return recorded;
197 }
198 return eve::Result<void>::success();
199}

227eve::Result<void> VirtualTextureStreaming::invalidateRect(float u0, float v0, float u1, float v1, int mipLevel) {
228 auto ready = ensureInitialized();
229 if (!ready.ok()) return ready;
230 if (!std::isfinite(u0) || !std::isfinite(v0) || !std::isfinite(u1) || !std::isfinite(v1))
231 return eve::Result<void>::failure(invalid("invalidate rect coordinates must be finite"));
232 if (u1 < u0) std::swap(u0, u1);
233 if (v1 < v0) std::swap(v0, v1);
234 const int minX = std::clamp(int(std::floor(wrap01(u0) * float(pageCountX()))), 0, pageCountX() - 1);
235 const int maxX = std::clamp(int(std::floor(wrap01(u1) * float(pageCountX()))), 0, pageCountX() - 1);
236 const int minY = std::clamp(int(std::floor(wrap01(v0) * float(pageCountY()))), 0, pageCountY() - 1);
237 const int maxY = std::clamp(int(std::floor(wrap01(v1) * float(pageCountY()))), 0, pageCountY() - 1);
238 for (int y = minY; y <= maxY; ++y)
239 for (int x = minX; x <= maxX; ++x) {
240 auto invalidated = invalidatePage(x, y, mipLevel);
241 if (!invalidated.ok()) return invalidated;
242 }
243 return eve::Result<void>::success();
244}

279eve::Result<int> VirtualTextureStreaming::allocateSlot(PhysicalState& state,
280 std::vector<VirtualTextureRegionPixels>& pageTableClears) {
281 for (int slot = 1; slot < int(state.slots.size()); ++slot) {
282 if (!state.slots[std::size_t(slot)].occupied) {
283 state.slots[std::size_t(slot)].occupied = true;
284 state.lru.push_front(slot);
285 state.slots[std::size_t(slot)].lru = state.lru.begin();
286 return eve::Result<int>::success(slot);
287 }
288 }
289 if (state.lru.empty())
290 return eve::Result<int>::failure(failed("virtual texture physical cache has no evictable slots"));
291 const int victim = state.lru.back();
292 state.lru.pop_back();
293 auto& record = state.slots[std::size_t(victim)];
294 if (record.occupied && record.pageX >= 0 && record.pageY >= 0) {
295 clearPageMapping(state, record.pageX, record.pageY);
296 pageTableClears.push_back(copyPageTableTexel(*state.pageTable, record.pageX, record.pageY));
297 state.pageToSlot.erase(PageKey{record.pageX, record.pageY});
298 ++lifetime_.pagesEvicted;
299 }
300 record.occupied = true;
301 record.dirty = false;
302 record.pageX = -1;
303 record.pageY = -1;
304 state.lru.push_front(victim);
305 record.lru = state.lru.begin();
306 return eve::Result<int>::success(victim);
307}

309eve::Result<void> VirtualTextureStreaming::decodeIntoSlot(PhysicalState& state, int slot, int pageX, int pageY,
310 std::vector<VirtualTextureRegionPixels>& albedoRegions,
311 std::vector<VirtualTextureRegionPixels>& normalRegions,
312 std::vector<VirtualTextureRegionPixels>& pageTableSets) {
313 auto albedo = producer_->requestAlbedoPage(pageX, pageY, 0);
314 if (!albedo.ok()) return eve::Result<void>::failure(albedo.status());
315 auto normal = producer_->requestNormalPage(pageX, pageY, 0);
316 if (!normal.ok()) return eve::Result<void>::failure(normal.status());
317
318 const int slotX = slot % atlasSlotsX_;
319 const int slotY = slot / atlasSlotsX_;
320 const int dstX = slotX * pageExtent_;
321 const int dstY = slotY * pageExtent_;
322 state.albedo->paste(const_cast<image::ImageData*>(albedo.value().get()), dstX, dstY, 0, 0, pageExtent_,
323 pageExtent_);
324 state.normal->paste(const_cast<image::ImageData*>(normal.value().get()), dstX, dstY, 0, 0, pageExtent_,
325 pageExtent_);
326 albedoRegions.push_back(copyRegion(*state.albedo, dstX, dstY, pageExtent_, pageExtent_));
327 normalRegions.push_back(copyRegion(*state.normal, dstX, dstY, pageExtent_, pageExtent_));
328
329 auto& record = state.slots[std::size_t(slot)];
330 record.pageX = pageX;
331 record.pageY = pageY;
332 record.dirty = false;
333 record.occupied = true;
334 state.pageToSlot[PageKey{pageX, pageY}] = slot;
335 touchSlot(state, slot);
336
337 const Pixel entry{(float(slotX) + 0.5f) / float(atlasSlotsX_), (float(slotY) + 0.5f) / float(atlasSlotsY_), 1.f,
338 1.f};
339 state.pageTable->setPixel(pageX, pageY, entry);
340 pageTableSets.push_back(copyPageTableTexel(*state.pageTable, pageX, pageY));
341 ++lifetime_.pagesDecoded;
342 return eve::Result<void>::success();
343}

345eve::Result<VirtualTextureStreamingStats> VirtualTextureStreaming::update() {
346 auto ready = ensureInitialized();
347 if (!ready.ok()) return eve::Result<VirtualTextureStreamingStats>::failure(ready.status());
348 if (pending_.has_value())
349 return eve::Result<VirtualTextureStreamingStats>::failure(
350 failed("commit or abandon the pending virtual texture upload before update"));
351
352 // Compact + dedupe feedback (UE feedback analyze).
353 std::unordered_map<PageKey, int, PageKeyHash> unique;
354 unique.reserve(feedback_.size());
355 for (const auto& sample : feedback_) {
356 if (sample.mipLevel != 0) continue;
357 ++unique[PageKey{sample.pageX, sample.pageY}];
358 }
359 feedback_.clear();
360 lifetime_.uniqueFeedback += unique.size();
361 lifetime_.pendingFeedback = 0;
362
363 std::vector<std::pair<PageKey, int>> ordered(unique.begin(), unique.end());
364 std::sort(ordered.begin(), ordered.end(), [](const auto& a, const auto& b) {
365 if (a.second != b.second) return a.second > b.second;
366 if (a.first.y != b.first.y) return a.first.y < b.first.y;
367 return a.first.x < b.first.x;
368 });
369
370 VirtualTextureUploadBatch batch;
371 batch.expectedGeneration = generation_;
372 std::size_t decoded = 0;
373
374 for (const auto& [page, weight] : ordered) {
375 (void)weight;
376 auto found = working_.pageToSlot.find(page);
377 if (found != working_.pageToSlot.end()) {
378 touchSlot(working_, found->second);
379 if (!working_.slots[std::size_t(found->second)].dirty) continue;
380 if (decoded >= config_.maxDecodesPerUpdate) break;
381 auto decodedOk = decodeIntoSlot(working_, found->second, page.x, page.y, batch.albedoRegions,
382 batch.normalRegions, batch.pageTableSets);
383 if (!decodedOk.ok()) return eve::Result<VirtualTextureStreamingStats>::failure(decodedOk.status());
384 ++decoded;
385 continue;
386 }
387 if (decoded >= config_.maxDecodesPerUpdate) break;
388 auto slot = allocateSlot(working_, batch.pageTableClears);
389 if (!slot.ok()) return eve::Result<VirtualTextureStreamingStats>::failure(slot.status());
390 auto decodedOk = decodeIntoSlot(working_, slot.value(), page.x, page.y, batch.albedoRegions,
391 batch.normalRegions, batch.pageTableSets);
392 if (!decodedOk.ok()) return eve::Result<VirtualTextureStreamingStats>::failure(decodedOk.status());
393 ++decoded;
394 }
395
396 // Also process dirty residents that were not in this frame's feedback.
397 if (decoded < config_.maxDecodesPerUpdate) {
398 for (int slot = 1; slot < int(working_.slots.size()) && decoded < config_.maxDecodesPerUpdate; ++slot) {
399 auto& record = working_.slots[std::size_t(slot)];
400 if (!record.occupied || !record.dirty || record.pageX < 0) continue;
401 auto decodedOk = decodeIntoSlot(working_, slot, record.pageX, record.pageY, batch.albedoRegions,
402 batch.normalRegions, batch.pageTableSets);
403 if (!decodedOk.ok()) return eve::Result<VirtualTextureStreamingStats>::failure(decodedOk.status());
404 ++decoded;
405 }
406 }
407
408 const bool hasWork = !batch.pageTableClears.empty() || !batch.albedoRegions.empty() ||
409 !batch.normalRegions.empty() || !batch.pageTableSets.empty();
410 if (hasWork) {
411 pending_ = std::move(batch);
412 lifetime_.pendingUploads = pending_->pageTableClears.size() + pending_->albedoRegions.size() +
413 pending_->normalRegions.size() + pending_->pageTableSets.size();
414 } else {
415 lifetime_.pendingUploads = 0;
416 }
417 lifetime_.generation = generation_;
418 lifetime_.residentPages = working_.pageToSlot.size();
419 return eve::Result<VirtualTextureStreamingStats>::success(stats());
420}

422eve::Result<VirtualTextureUploadBatch> VirtualTextureStreaming::takePendingUploads() {
423 if (!pending_.has_value())
424 return eve::Result<VirtualTextureUploadBatch>::failure(failed("no pending virtual texture upload batch"));
425 VirtualTextureUploadBatch batch = std::move(*pending_);
426 // Keep the optional engaged with an empty marker? We need commit/abandon to know a batch was taken.
427 // Store a sentinel empty batch with the same expectedGeneration.
428 pending_ = VirtualTextureUploadBatch{};
429 pending_->expectedGeneration = batch.expectedGeneration;
430 lifetime_.pendingUploads = 0;

460VirtualTextureStreaming::PhysicalState VirtualTextureStreaming::cloneState(const PhysicalState& source) const {
461 PhysicalState copy;
462 copy.albedo = std::make_unique<image::ImageData>(*source.albedo);
463 copy.normal = std::make_unique<image::ImageData>(*source.normal);
464 copy.pageTable = std::make_unique<image::ImageData>(*source.pageTable);
465 copy.slots = source.slots;
466 copy.pageToSlot = source.pageToSlot;
467 copy.lru.clear();
468 // Rebuild LRU iterators against the new list.
469 for (int slot : source.lru) {
470 copy.lru.push_back(slot);
471 copy.slots[std::size_t(slot)].lru = std::prev(copy.lru.end());
472 }
473 // Slot 0 has no LRU entry.
474 if (!copy.slots.empty()) copy.slots[0].lru = copy.lru.end();
475 return copy;
476}

515eve::Result<void> VirtualTextureStreaming::syncToMaterial(Graphics& gfx, Material& material) {
516 auto ready = ensureInitialized();
517 if (!ready.ok()) return ready;
518
519 if (!gpuAlbedo_ || !gpuNormal_ || !gpuPageTable_) {
520 TextureCreateInfo info{};
521 info.generateMipmaps = false;
522 gpuAlbedo_ = gfx.newTexture(const_cast<image::ImageData*>(published_.albedo.get()), info);
523 gpuNormal_ = gfx.newTexture(const_cast<image::ImageData*>(published_.normal.get()), info);
524 gpuPageTable_ = gfx.newTexture(const_cast<image::ImageData*>(published_.pageTable.get()), info);
525 if (!gpuAlbedo_ || !gpuNormal_ || !gpuPageTable_)
526 return eve::Result<void>::failure(failed("failed to create virtual texture GPU atlases"));
527 auto bound = material.setVirtualTexture(gpuAlbedo_, gpuNormal_, gpuPageTable_, pageCountX(), pageCountY(),
528 atlasSlotsX_, atlasSlotsY_, borderFraction_);
529 if (!bound.ok()) return bound;
530 }
531
532 if (!pending_.has_value()) return eve::Result<void>::success();
533
534 auto batchResult = takePendingUploads();
535 if (!batchResult.ok()) return eve::Result<void>::failure(batchResult.status());
536 auto batch = std::move(batchResult).value();
537
538 auto restoreGpuFromPublished = [&]() -> eve::Result<void> {
539 TextureRegionUpload albedoUpload;
540 albedoUpload.width = published_.albedo->getWidth();
541 albedoUpload.height = published_.albedo->getHeight();
542 albedoUpload.rgba = std::span<const std::uint8_t>(
543 static_cast<const std::uint8_t*>(published_.albedo->getData()), published_.albedo->getSize());
544 albedoUpload.bytesPerRow = std::size_t(albedoUpload.width) * 4u;
545 auto albedoStatus =
546 gfx.updateTextureRegions(gpuAlbedo_, std::span<const TextureRegionUpload>(&albedoUpload, 1));
547 if (!albedoStatus.ok()) return albedoStatus;
548 TextureRegionUpload normalUpload = albedoUpload;
549 normalUpload.rgba = std::span<const std::uint8_t>(
550 static_cast<const std::uint8_t*>(published_.normal->getData()), published_.normal->getSize());
551 auto normalStatus =
552 gfx.updateTextureRegions(gpuNormal_, std::span<const TextureRegionUpload>(&normalUpload, 1));
553 if (!normalStatus.ok()) return normalStatus;
554 TextureRegionUpload tableUpload = albedoUpload;
555 tableUpload.width = published_.pageTable->getWidth();
556 tableUpload.height = published_.pageTable->getHeight();
557 tableUpload.rgba = std::span<const std::uint8_t>(
558 static_cast<const std::uint8_t*>(published_.pageTable->getData()), published_.pageTable->getSize());
559 tableUpload.bytesPerRow = std::size_t(tableUpload.width) * 4u;
560 return gfx.updateTextureRegions(gpuPageTable_, std::span<const TextureRegionUpload>(&tableUpload, 1));
561 };
562
563 auto clears = uploadRegions(gfx, gpuPageTable_, batch.pageTableClears);
564 if (!clears.ok()) {
565 (void)abandonPublish();
566 (void)restoreGpuFromPublished();
567 return clears;
568 }
569 auto albedo = uploadRegions(gfx, gpuAlbedo_, batch.albedoRegions);
570 if (!albedo.ok()) {
571 (void)abandonPublish();
572 (void)restoreGpuFromPublished();
573 return albedo;
574 }
575 auto normal = uploadRegions(gfx, gpuNormal_, batch.normalRegions);
576 if (!normal.ok()) {
577 (void)abandonPublish();
578 (void)restoreGpuFromPublished();
579 return normal;
580 }
581 auto sets = uploadRegions(gfx, gpuPageTable_, batch.pageTableSets);
582 if (!sets.ok()) {
583 (void)abandonPublish();
584 (void)restoreGpuFromPublished();
585 return sets;
586 }
587 return commitPublish(batch.expectedGeneration);
588}
