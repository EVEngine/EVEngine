18 return Result<Output>::failure(

21 const int bones = skeleton.getBoneCount(), joints = skin.getBoneCount();
22 if (times_.empty() || times_.size() > 16 || bones <= 0 || bones > 4096 || joints <= 0 || joints > 4096 ||
23 clip.getTrackCount() > bones)
24 return fail("Inspection requires 1..16 samples, 1..4096 bones/joints and compatible clip tracks");
25 if (!std::isfinite(clip.getDuration()) || clip.getDuration() <= 0.f)
26 return fail("Clip duration must be finite and positive");

31 time.clipTime > double(clip.getDuration()))
32 return fail("Sample times must be finite, strictly ordered on the timeline and inside the clip");
33 previous = time.timelineTime;
34 }
35 std::vector<Mat4> inverseBinds(size_t(joints), Mat4::identity());
36 for (int joint = 0; joint < joints; ++joint) {
37 const int bone = skin.getSkeletonBone(joint);
38 if (bone < 0 || bone >= bones || skin.getSkinBoneName(joint) != skeleton.getBoneName(bone))
39 return fail("Skin joint identity does not match the supplied skeleton");
40 for (int element = 0; element < 16; ++element) {
41 const float value = skin.getInverseBindElement(joint, element);
42 if (!std::isfinite(value)) return fail("Inverse bind matrix is non-finite");
43 inverseBinds[size_t(joint)].m[element] = value;
44 }
45 }
46 Output frames;

48 AnimPose pose;
49 for (const auto& time : times_) {
50 // Use the canonical sampler with already-resolved time: do not change clip.loop
51 // or call AnimPlayer::setTime/advance, which would affect live playback.
52 clip.sampleClamped(float(time.clipTime), &pose, &skeleton);
53 pose.computeWorld(&skeleton);
54 AnimInspectionFrame frame{time, std::vector<float>(size_t(joints) * 16)};
55 for (int joint = 0; joint < joints; ++joint) {
56 const auto matrix =
57 Mat4::mul(Mat4::fromTRS(pose.world(skin.getSkeletonBone(joint))), inverseBinds[size_t(joint)]);
58 for (float value : matrix.m)
59 if (!std::isfinite(value)) return fail("Sampled palette contains non-finite values");
60 if (matrix.m[3] != 0.f || matrix.m[7] != 0.f || matrix.m[11] != 0.f || matrix.m[15] != 1.f)
61 return fail("Sampled skin matrix must be affine");
62 std::copy_n(matrix.m, 16, frame.palette.begin() + size_t(joint) * 16);
63 }
64 frames.push_back(std::move(frame));
65 }
