19bool number(const Value* value, double& out) {
20 if (!value) return false;
21 if (const auto* real = value->getIf<double>())
22 out = *real;
23 else if (const auto* integer = value->getIf<std::int64_t>())
24 out = static_cast<double>(*integer);
25 else
26 return false;
27 return std::isfinite(out);
28}
29bool vector(const Value* value, glm::dvec3& out) {
30 const auto* values = value ? value->getIf<Value::Array>() : nullptr;

34bool boundsOf(const Value* value, AuthoringBounds& out) {
35 const auto* values = value ? value->getIf<Value::Array>() : nullptr;
36 if (!values || values->size() != 6) return false;
37 for (std::size_t i = 0; i < 6; ++i)
38 if (!number(&(*values)[i], out[i])) return false;
39 return out[0] <= out[3] && out[1] <= out[4] && out[2] <= out[5];

42Result<glm::dmat4> worldMatrix(const ISceneHierarchyEditTarget& target, ObjectId id) {
43 glm::dmat4 world(1.0);
44 std::set<ObjectId> visited;
45 while (!id.empty()) {
46 if (!visited.insert(id).second)
47 return failed<glm::dmat4>(Status::Rejected, RuleId("scene.authoring.cycle"), "Hierarchy contains a cycle");
48 auto object = target.sceneObject(id);
49 if (!object.ok()) return Result<glm::dmat4>::failure(object.status());
50 const auto& t = object.value().transform;
51 auto local = glm::translate(glm::dmat4(1.0), glm::dvec3(t.x, t.y, t.z));
52 local = glm::rotate(local, t.rotationY, glm::dvec3(0, 1, 0));
53 local = glm::rotate(local, t.rotationX, glm::dvec3(1, 0, 0));
54 local = glm::rotate(local, t.rotationZ, glm::dvec3(0, 0, 1));
55 local = glm::scale(local, glm::dvec3(t.scaleX, t.scaleY, t.scaleZ));
56 world = local * world;
57 id = object.value().parent;
58 }
59 return applied<glm::dmat4>(world);
60}
61struct Geometry {
62 glm::dvec3 minimum, maximum;
63 Value::Array corners;
64 glm::dmat4 matrix;
65};
66Result<Geometry> geometry(const ISceneHierarchyEditTarget& target, const ObjectId& id, const AuthoringBounds& bounds) {
67 if (id.empty())
68 return failed<Geometry>(Status::Rejected, RuleId("scene.authoring.object"), "Object id is required");
69 for (double value : bounds)
70 if (!std::isfinite(value))
71 return failed<Geometry>(Status::Rejected, RuleId("scene.authoring.bounds"), "Bounds must be finite");
72 for (std::size_t axis = 0; axis < 3; ++axis)
73 if (bounds[axis] > bounds[axis + 3])
74 return failed<Geometry>(Status::Rejected, RuleId("scene.authoring.bounds"), "Bounds must be ordered");
75 auto world = worldMatrix(target, id);
76 if (!world.ok()) return Result<Geometry>::failure(world.status());
77 Geometry result{glm::dvec3(std::numeric_limits<double>::max()),
78 glm::dvec3(std::numeric_limits<double>::lowest()),
79 {},
80 world.value()};
81 for (int i = 0; i < 8; ++i) {
82 const glm::dvec3 local(bounds[(i & 1) ? 3 : 0], bounds[(i & 2) ? 4 : 1], bounds[(i & 4) ? 5 : 2]);
83 const auto p = glm::dvec3(result.matrix * glm::dvec4(local, 1));
84 for (int axis = 0; axis < 3; ++axis)
85 if (!std::isfinite(p[axis]))
86 return failed<Geometry>(Status::Rejected, RuleId("scene.authoring.world-overflow"),
87 "World transform overflow");
88 result.minimum = glm::min(result.minimum, p);
89 result.maximum = glm::max(result.maximum, p);
90 result.corners.push_back(point(p));
91 }
92 return applied<Geometry>(std::move(result));
93}
94Result<CommandPlan> planPlacement(IEditableTarget& target, const CommandRequest& request, const std::string& mode) {
95 auto hierarchy = target.capability<ISceneHierarchyEditTarget>();
96 auto transforms = target.capability<ITransformEditTarget>();
97 const auto* idValue = field(request.payload, "object");
98 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
99 const auto reject = [](const char* message) {
100 return failed<CommandPlan>(Status::Rejected, RuleId("scene.authoring.placement-payload"), message);
101 };
102 if (!hierarchy || !transforms)
103 return failed<CommandPlan>(Status::Unsupported, RuleId("scene.authoring.target"),
104 "Hierarchy and transform editing capabilities are required");
105 if (!id || id->empty()) return reject("Object id is required");
106 const auto* payload = request.payload.getIf<Value::Object>();
107 std::set<std::string> keys{"object", "bounds"};
108 if (mode == "ground") keys.insert("height");
109 if (mode == "align") keys.insert({"reference", "referenceBounds", "axis", "gap"});
110 if (mode == "screen") keys = {"object", "anchor", "eye", "target", "up", "fov", "aspect", "screen"};
111 for (const auto& [key, value] : *payload) {
112 (void)value;
113 if (!keys.contains(key)) return reject("Unknown placement field");
114 }
115 auto current = hierarchy->get().sceneObject(ObjectId(*id));
116 if (!current.ok()) return Result<CommandPlan>::failure(current.status());
117 auto world = worldMatrix(hierarchy->get(), ObjectId(*id));
118 if (!world.ok()) return Result<CommandPlan>::failure(world.status());
119 glm::dvec3 delta(0);
120 if (mode == "screen") {
121 glm::dvec3 eye, targetPoint, up, anchor;
122 double fov = 0, aspect = 0;
123 const auto* screenValue = field(request.payload, "screen");
124 const auto* screen = screenValue ? screenValue->getIf<Value::Array>() : nullptr;
125 double u = 0, v = 0;
126 if (!vector(field(request.payload, "eye"), eye) || !vector(field(request.payload, "target"), targetPoint) ||
127 !vector(field(request.payload, "up"), up) || !vector(field(request.payload, "anchor"), anchor) ||
128 !number(field(request.payload, "fov"), fov) || !number(field(request.payload, "aspect"), aspect) ||
129 !screen || screen->size() != 2 || !number(&(*screen)[0], u) || !number(&(*screen)[1], v) || fov <= 0 ||
130 fov >= 179 || aspect <= 0 || u < 0 || u > 1 || v < 0 || v > 1)
131 return reject("Screen placement requires a valid perspective camera and normalized UV");
132 auto forward = targetPoint - eye;
133 if (glm::length(forward) < 1e-8 || glm::length(up) < 1e-8) return reject("Camera basis is degenerate");
134 forward = glm::normalize(forward);
135 auto right = glm::cross(forward, glm::normalize(up));
136 if (glm::length(right) < 1e-8) return reject("Camera up is parallel to viewing direction");
137 right = glm::normalize(right);
138 up = glm::cross(right, forward);
139 const auto p = glm::dvec3(world.value() * glm::dvec4(anchor, 1));
140 const double depth = glm::dot(p - eye, forward);
141 if (depth <= 1e-8) return reject("Anchor must be in front of the camera");
142 const double halfHeight = depth * std::tan(glm::radians(fov) * 0.5);
143 delta =
144 eye + forward * depth + right * ((2 * u - 1) * halfHeight * aspect) + up * ((1 - 2 * v) * halfHeight) - p;
145 } else {
146 AuthoringBounds bounds{};
147 if (!boundsOf(field(request.payload, "bounds"), bounds)) return reject("Asset bounds are required");
148 auto objectGeometry = geometry(hierarchy->get(), ObjectId(*id), bounds);
149 if (!objectGeometry.ok()) return Result<CommandPlan>::failure(objectGeometry.status());
150 if (mode == "ground") {
151 double height = 0;
152 if (!number(field(request.payload, "height"), height)) return reject("Ground height must be finite");
153 delta.y = height - objectGeometry.value().minimum.y;
154 } else {
155 const auto* refValue = field(request.payload, "reference");
156 const auto* ref = refValue ? refValue->getIf<std::string>() : nullptr;
157 const auto* axisValue = field(request.payload, "axis");
158 const auto* axis = axisValue ? axisValue->getIf<std::string>() : nullptr;
159 double gap = 0;
160 AuthoringBounds referenceBounds{};
161 if (!ref || *ref == *id || !axis || (*axis != "x" && *axis != "y" && *axis != "z") ||
162 !number(field(request.payload, "gap"), gap) ||
163 !boundsOf(field(request.payload, "referenceBounds"), referenceBounds))
164 return reject("Alignment requires a different reference, its bounds, axis and finite gap");
165 // Moving an ancestor would move the reference too, invalidating the solved gap.
166 auto ancestor = ObjectId(*ref);
167 while (!ancestor.empty()) {
168 if (ancestor == ObjectId(*id)) return reject("Reference cannot descend from the moved object");
169 auto entry = hierarchy->get().sceneObject(ancestor);
170 if (!entry.ok()) return Result<CommandPlan>::failure(entry.status());
171 ancestor = entry.value().parent;
172 }
173 auto referenceGeometry = geometry(hierarchy->get(), ObjectId(*ref), referenceBounds);
174 if (!referenceGeometry.ok()) return Result<CommandPlan>::failure(referenceGeometry.status());
175 const int index = *axis == "x" ? 0 : *axis == "y" ? 1 : 2;
176 delta[index] = referenceGeometry.value().maximum[index] + gap - objectGeometry.value().minimum[index];
177 }
178 }
179 auto parent = worldMatrix(hierarchy->get(), current.value().parent);
180 if (!parent.ok()) return Result<CommandPlan>::failure(parent.status());
181 const double determinant = glm::determinant(parent.value());
182 if (!std::isfinite(determinant) || std::abs(determinant) < 1e-12) return reject("Parent transform is singular");
183 const auto localDelta = glm::dvec3(glm::inverse(parent.value()) * glm::dvec4(delta, 0));
184 auto transform = current.value().transform;
185 transform.x += localDelta.x;
186 transform.y += localDelta.y;
187 transform.z += localDelta.z;
188 auto operation = transforms->get().makeSetTransform(ObjectId(*id), transform);
189 if (!operation.ok()) return Result<CommandPlan>::failure(operation.status());
190 CommandPlan plan;
191 plan.operations.push_back(std::move(operation.value()));
192 plan.summary = Value::Object{{"object", *id}, {"constraint", mode}, {"worldDisplacement", point(delta)}};
193 return applied<CommandPlan>(std::move(plan));
194}

197Result<EditorValue> inspectAuthoringObject(const ISceneHierarchyEditTarget& target, const ObjectId& object,
198 const AuthoringBounds& bounds) {
199 auto result = geometry(target, object, bounds);
200 if (!result.ok()) return Result<EditorValue>::failure(result.status());
201 const auto& g = result.value();
202 EditorValue::Array matrix;
203 for (int column = 0; column < 4; ++column)
204 for (int row = 0; row < 4; ++row) matrix.emplace_back(g.matrix[column][row]);
205 const auto center = (g.minimum + g.maximum) * 0.5;
206 return editing::applied<EditorValue>(
207 EditorValue::Object{{"worldMatrix", std::move(matrix)},
208 {"worldMinimum", point(g.minimum)},
209 {"worldMaximum", point(g.maximum)},
210 {"worldSize", point(g.maximum - g.minimum)},
211 {"worldCorners", g.corners},
212 {"groundAnchor", point(glm::dvec3(center.x, g.minimum.y, center.z))}});
213}
