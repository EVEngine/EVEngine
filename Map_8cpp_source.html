28void retireReplacedLayers(const std::vector<TileLayer *> &previous,
29 const std::vector<TileLayer *> &replacement) {
30 for (TileLayer *layer : previous) {
31 if (!layer || std::find(replacement.begin(), replacement.end(), layer) != replacement.end())
32 continue;
33 layer->clear();
34 layer->setVisible(false);
35 }
36}

127int Map::publishCollision(TileLayer *layer) {
128 collisionRects_.clear();
129 if (!layer) return 0;
130 const auto config = layer->config();
131 const auto tiles = layer->tiles();
132 const auto tileset = layer->tileset();
133 if (config->orientation != MapOrientation::Orthogonal) return 0;
134
135 std::vector<uint8_t> solid(size_t(config->mapW * config->mapH), 0);
136 for (int y = 0; y < config->mapH; ++y) {
137 for (int x = 0; x < config->mapW; ++x) {
138 const int gid = int(tileGid(tiles->gids[size_t(y * config->mapW + x)]));
139 const auto it = std::find_if(tileset->visuals.begin(), tileset->visuals.end(),
140 [gid](const auto &v) { return v.gid == gid; });
141 if (gid != 0 && it != tileset->visuals.end() && !it->collisionShapes.empty()) {
142 const float originX = config->originX + x * (config->tileW + config->cellGapX);
143 const float originY = config->originY + y * (config->tileH + config->cellGapY);
144 for (const auto &shape : it->collisionShapes)
145 collisionRects_.push_back({originX + shape.x, originY + shape.y, shape.width, shape.height});
146 } else {
147 solid[size_t(y * config->mapW + x)] = gid != 0 && it != tileset->visuals.end() && !it->walkable;
148 }
149 }
150 }
151 for (int y = 0; y < config->mapH; ++y) {
152 for (int x = 0; x < config->mapW; ++x) {
153 if (!solid[size_t(y * config->mapW + x)]) continue;
154 int width = 1;
155 while (x + width < config->mapW && solid[size_t(y * config->mapW + x + width)])
156 ++width;
157 int height = 1;
158 bool canGrow = true;
159 while (y + height < config->mapH && canGrow) {
160 for (int dx = 0; dx < width; ++dx)
161 if (!solid[size_t((y + height) * config->mapW + x + dx)]) canGrow = false;
162 if (canGrow) ++height;
163 }
164 for (int dy = 0; dy < height; ++dy)
165 for (int dx = 0; dx < width; ++dx)
166 solid[size_t((y + dy) * config->mapW + x + dx)] = 0;
167 collisionRects_.push_back({config->originX + x * (config->tileW + config->cellGapX),
168 config->originY + y * (config->tileH + config->cellGapY),
169 width * (config->tileW + config->cellGapX),
170 height * (config->tileH + config->cellGapY)});
171 }
172 }
173 cap::forEach<ITileCollisionSink>([&](ITileCollisionSink *sink) {
174 sink->replaceTileCollision(layer->layerRef(), collisionRects_.data(), collisionRects_.size());
175 });
176 return int(collisionRects_.size());
177}

253eve::Result<int> Map::loadFromFileWithObjectContract(const std::string &path,
254 std::string_view contractJson) {
255 std::vector<MapObject> objects;
256 std::string error;
257 auto layers = loadMapFile(path, &objects, &error);
258 if (layers.empty()) {
259 return eve::Result<int>::failure(eve::Diagnostic::error(
260 eve::DiagnosticCode::ParseError, error.empty() ? "map could not be loaded" : error, path, {},
261 "map.load"));
262 }
263 auto admitted = validateMapObjects(objects, contractJson);
264 if (!admitted.ok()) {
265 discardCandidateLayers(layers);
266 return eve::Result<int>::failure(admitted.status());
267 }
268 retireReplacedLayers(loadedLayers_, layers);
269 setObjects(std::move(objects));
270 loadedLayers_ = std::move(layers);
271 return eve::Result<int>::success(static_cast<int>(loadedLayers_.size()),
272 eve::Status::success(eve::StatusCode::Applied));
273}

329std::optional<std::size_t> Map::findObjectAt(float x, float y, std::string_view type) const {
330 if (!std::isfinite(x) || !std::isfinite(y)) return std::nullopt;
331 for (std::size_t index = 0; index < objects_.size(); ++index) {
332 const MapObject &object = objects_[index];
333 if (!type.empty() && object.type != type) continue;
334 const bool point = object.width <= 0.f && object.height <= 0.f;
335 if (point ? (object.x == x && object.y == y)
336 : (x >= object.x && y >= object.y && x <= object.x + std::max(0.f, object.width) &&
337 y <= object.y + std::max(0.f, object.height)))
338 return index;
339 }
340 return std::nullopt;
341}
