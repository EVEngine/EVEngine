174void Camera3D::screenToRay(float screenX, float screenY, float viewW, float viewH) {
175 auto d = data();
176 d->screenRayOx = d->eyeX;
177 d->screenRayOy = d->eyeY;
178 d->screenRayOz = d->eyeZ;
179 d->screenRayDx = 0.f;
180 d->screenRayDy = 0.f;
181 d->screenRayDz = -1.f;
182 if (viewW <= 0.f || viewH <= 0.f) return;
183
184 const glm::vec3 eye(d->eyeX, d->eyeY, d->eyeZ);
185 const glm::vec3 target(d->targetX, d->targetY, d->targetZ);
186 const glm::vec3 up(d->upX, d->upY, d->upZ);
187 const glm::mat4 viewM = glm::lookAtRH(eye, target, up);
188 const float aspect = viewW / viewH;
189 const float fovRad = d->fovYDeg * 0.017453292519943295f;
190 const glm::mat4 projM = perspectiveVulkanRH_ZO(fovRad, aspect, d->nearZ, d->farZ);
191 const glm::mat4 invVP = glm::inverse(projM * viewM);
192
193 // Screen pixel → Vulkan NDC (Y-down; matches perspectiveVulkanRH_ZO).
194 const float ndcX = (screenX / viewW) * 2.f - 1.f;
195 const float ndcY = (screenY / viewH) * 2.f - 1.f;
196 auto unproject = [&](float ndcZ) -> glm::vec3 {
197 glm::vec4 w = invVP * glm::vec4(ndcX, ndcY, ndcZ, 1.f);
198 if (std::fabs(w.w) < 1e-8f) return eye;
199 w /= w.w;
200 return glm::vec3(w);
201 };
202 // ZO depth: near = 0, far = 1.
203 const glm::vec3 nearPt = unproject(0.f);
204 const glm::vec3 farPt = unproject(1.f);
205 glm::vec3 dir = farPt - nearPt;
206 const float len = glm::length(dir);
207 if (len > 1e-8f) dir /= len;
208 else dir = glm::normalize(target - eye);
209
210 d->screenRayOx = eye.x;
211 d->screenRayOy = eye.y;
212 d->screenRayOz = eye.z;
213 d->screenRayDx = dir.x;
214 d->screenRayDy = dir.y;
215 d->screenRayDz = dir.z;
216}

455FrustumPlanes extractFrustum(const glm::mat4 &m) {
456 FrustumPlanes f;
457 const glm::vec4 r0(m[0][0], m[1][0], m[2][0], m[3][0]);
458 const glm::vec4 r1(m[0][1], m[1][1], m[2][1], m[3][1]);
459 const glm::vec4 r2(m[0][2], m[1][2], m[2][2], m[3][2]);
460 const glm::vec4 r3(m[0][3], m[1][3], m[2][3], m[3][3]);
461 auto norm = [](glm::vec4 &v) {
462 const float l = glm::length(glm::vec3(v));
463 if (l > 1e-8f) v /= l;
464 };
465 f.p[0] = r3 + r0; // left
466 f.p[1] = r3 - r0; // right
467 f.p[2] = r3 + r1; // bottom
468 f.p[3] = r3 - r1; // top
469 // Vulkan clip space uses zero-to-one depth: near plane is z_clip = 0
470 // (plane r2), not the z = -w plane used by OpenGL-style [-1,1] depth.
471 f.p[4] = r2; // near
472 f.p[5] = r3 - r2; // far
473 for (auto &pl : f.p) norm(pl);
474 return f;
475}

544void RenderSystem3D::render(Graphics &gfx) {
545 eve::debug::RenderPassScope pass3d("RenderSystem3D");
546 Camera3D *defaultCam = findDefaultCamera3D();
547
548 RenderControl *rc = gfx.getRenderControl();
549 rc->ensureCompiled();
550 const bool doShadow = rc->hasPass("shadow");
551 const bool doGBuffer = rc->hasPass("gbuffer");
552 const bool doForward = rc->hasPass("forward");
553 const bool doHair = rc->hasPass("hair");
554 const bool allowClustered = rc->isEnabled("clustered");
555
556 std::vector<PackedLight3D> packed;
557 collectLights3D(packed, size_t(ClusteredLightConfig::kMaxLights));
558 promoteDirectional(packed);
559 Light3D::Data *shadowCaster = doShadow ? findShadowCasterDir(packed) : nullptr;
560 prioritizeShadowCaster(packed, shadowCaster);
561 const bool haveExtraShadowCasters = doShadow && !g_shadowDrawers.empty();
562
563 const float aspect = (gfx.getHeight() > 0) ? float(gfx.getWidth()) / float(gfx.getHeight()) : 1.f;
564
565 ShadowUpload shadowUpload{};
566 shadowUpload.active = false;
567 // Per-cascade light frustums for caster culling (sphere vs frustum).
568 FrustumPlanes cascadeFrustums[ShadowConfig::kCascades];
569 if ((shadowCaster || haveExtraShadowCasters) && defaultCam) {
570 auto cd = defaultCam->data();
571 glm::vec3 dir = shadowCaster ? glm::vec3(shadowCaster->dx, shadowCaster->dy, shadowCaster->dz) : gLightDir;
572 if (glm::length(dir) < 1e-6f)
573 dir = glm::vec3(0.f, 1.f, 0.f);
574 else
575 dir = glm::normalize(dir);
576 const float shadowBias = shadowCaster ? shadowCaster->shadowBias : 0.003f;
577 const float shadowStrength = shadowCaster ? shadowCaster->shadowStrength : 1.f;
578 const float fovRad = cd->fovYDeg * 0.017453292519943295f;
579 shadowUpload = buildDirectionalCSM(
580 dir, glm::vec3(cd->eyeX, cd->eyeY, cd->eyeZ), glm::vec3(cd->targetX, cd->targetY, cd->targetZ),
581 glm::vec3(cd->upX, cd->upY, cd->upZ), fovRad, aspect, cd->nearZ, cd->farZ, shadowBias, shadowStrength);
582 for (int c = 0; c < ShadowConfig::kCascades; ++c)
583 cascadeFrustums[c] = extractFrustum(shadowUpload.ubo.lightVP[c]);
584 }
585 gfx.setMesh3DShadows(shadowUpload);
586
587 const bool useClustered = allowClustered && packed.size() > size_t(Lighting3DPack::kMaxLights);
588 // Light split / directional promotion is camera-independent: compute once,
589 // then each camera view bakes its own clustered table from these lists.
590 std::vector<ClusteredLightGpu> clusteredPoints;
591 std::vector<ClusteredLightGpu> clusteredDirs;
592 if (useClustered) {
593 splitLights(packed, clusteredPoints, clusteredDirs);
594 if (clusteredDirs.empty()) {
595 ClusteredLightGpu d{};
596 d.posRadius = glm::vec4(gLightDir, 0.f);
597 d.color = glm::vec4(gLightColor, 1.f);
598 clusteredDirs.push_back(d);
599 }
600 if (shadowCaster && !clusteredDirs.empty()) {
601 glm::vec3 dir(shadowCaster->dx, shadowCaster->dy, shadowCaster->dz);
602 if (glm::length(dir) < 1e-6f)
603 dir = glm::vec3(0.f, 1.f, 0.f);
604 else
605 dir = glm::normalize(dir);
606 for (size_t i = 0; i < clusteredDirs.size(); ++i) {
607 if (glm::length(glm::vec3(clusteredDirs[i].posRadius) - dir) < 1e-3f) {
608 if (i != 0) std::swap(clusteredDirs[0], clusteredDirs[i]);
609 break;
610 }
611 }
612 }
613 }
614
615 const bool haveManager = ecs::current()->getManager<Renderable3D>() != nullptr;
616 const bool shadowActive = doShadow && shadowUpload.active;
617
618 // Per-camera constants (matrices, frustum, lighting, clustered table).
619 // The default camera is slot 0 so the G-buffer pass reuses its view.
620 std::vector<CameraView> cams;
621 cams.reserve(2);
622 if (defaultCam) {
623 cams.push_back(buildCameraView(defaultCam->data().operator->(), packed, useClustered, aspect, clusteredPoints,
624 clusteredDirs, gfx));
625 }
626 auto findOrAddCam = [&](Camera3D *camEnt) -> int {
627 Camera3D::Data *d = camEnt->data().operator->();
628 for (size_t i = 0; i < cams.size(); ++i) {
629 if (cams[i].data == d) return int(i);
630 }
631 cams.push_back(buildCameraView(d, packed, useClustered, aspect, clusteredPoints, clusteredDirs, gfx));
632 return int(cams.size()) - 1;
633 };
634
635 // Single ECS traversal: one model matrix, one LOD pick, one set of
636 // sphere-vs-frustum tests per part. Passes below only replay the list.
637 std::vector<CulledItem> items;
638 items.reserve(64);
639 if (haveManager) {
640 auto view = ecs::View<Renderable3D, Renderable3D::Transform3D, Renderable3D::MeshRenderer>();
641 for (auto it = view.begin(); it != view.end(); ++it) {
642 auto [xf, mr] = *it;
643 if (!mr->visible) continue;
644 Camera3D *camEnt = mr->camera ? mr->camera : defaultCam;
645 if (!camEnt) continue;
646 const int camIdx = findOrAddCam(camEnt);
647 const CameraView &cv = cams[size_t(camIdx)];
648 const float dx = xf->x - cv.eye.x;
649 const float dy = xf->y - cv.eye.y;
650 const float dz = xf->z - cv.eye.z;
651 const float distSq = dx * dx + dy * dy + dz * dz;
652 const float dist = std::sqrt(distSq);
653 const glm::mat4 model = modelFromTransform(*xf);
654 const float maxScale = std::max(std::abs(xf->sx), std::max(std::abs(xf->sy), std::abs(xf->sz)));
655 const bool shadowOk = mr->effectiveCastShadow();
656
657 auto pushPart = [&](Mesh *drawMesh, Material *mat, bool asHair, bool castsShadow) {
658 if (!drawMesh) return;
659 CulledItem item;
660 item.mr = mr;
661 item.mesh = drawMesh;
662 item.material = mat;
663 item.shader = mat ? mat->effectiveShader() : mr->shader;
664 item.model = model;
665 item.distSq = distSq;
666 item.camIdx = camIdx;
667 item.hair = asHair;
668 item.xray = mr->xrayHighlight;
669 if (drawMesh->hasBounds()) {
670 const glm::vec4 c4 =
671 model * glm::vec4(drawMesh->boundsCx, drawMesh->boundsCy, drawMesh->boundsCz, 1.f);
672 item.worldC = glm::vec3(c4);
673 item.worldR = drawMesh->boundsRadius * maxScale;
674 item.inView = cv.frustum.sphereVisible(item.worldC, item.worldR);
675 item.inDefaultView = defaultCam ? cams[0].frustum.sphereVisible(item.worldC, item.worldR) : true;
676 if (shadowActive && castsShadow) {
677 for (int c = 0; c < ShadowConfig::kCascades; ++c) {
678 if (cascadeFrustums[c].sphereVisible(item.worldC, item.worldR))
679 item.cascadeMask |= (1u << c);
680 }
681 }
682 } else {
683 // No bounds (e.g. legacy/imported mesh): never cull.
684 item.inView = true;
685 item.inDefaultView = true;
686 if (shadowActive && castsShadow) item.cascadeMask = (1u << ShadowConfig::kCascades) - 1u;
687 }
688 items.push_back(item);
689 };
690
691 if (mr->usesParts()) {
692 for (int p = 0; p < mr->partCount; ++p) {
693 Material *mat = mr->parts[p].material ? mr->parts[p].material : mr->material;
694 const bool asHair = mat ? mat->isTransparentHair() : mr->isHair;
695 const bool casts = shadowOk && !(mat && !mat->getCastShadow());
696 pushPart(mr->parts[p].mesh, mat, asHair, casts);
697 }
698 } else {
699 Mesh *drawMesh = mr->meshForDistance(dist);
700 pushPart(drawMesh, mr->material, mr->effectiveHair(), shadowOk);
701 }
702 }
703 }
704
705 // CSM shadow passes — replay the collected casters, culled per cascade.
706 if (shadowActive && (haveManager || haveExtraShadowCasters)) {
707 auto cd = defaultCam->data();
708 for (int c = 0; c < ShadowConfig::kCascades; ++c) {
709 eve::debug::rtPassBegin("ShadowPass");
710 gfx.beginShadowPass(c);
711 for (const auto &item : items) {
712 if ((item.cascadeMask & (1u << c)) == 0) continue;
713 eve::debug::rtBind("mesh", "shadowCaster");
714 eve::debug::rtDraw("drawMeshShadow", "cascade");
715 gfx.drawMeshShadow(item.mesh, shadowUpload.ubo.lightVP[c] * item.model);
716 }
717 // Extra shadow casters (billboard/card geometry not in the ECS).
718 for (const auto &drawer : g_shadowDrawers) drawer(gfx, shadowUpload.ubo.lightVP[c], *cd);
719 gfx.endShadowPass();
720 eve::debug::rtPassEnd("ShadowPass");
721 }
722 }
723
724 // G-buffer fill (sampleable depth/normal) — before the forward swapchain pass.
725 if (doGBuffer && defaultCam && (haveManager || !g_gbufferDrawers.empty())) {
726 eve::debug::rtPassBegin("GBufferPass");
727 const CameraView &cv = cams[0]; // default camera is slot 0
728 const int gw = std::max(1, gfx.getPixelWidth() > 0 ? gfx.getPixelWidth() : gfx.getWidth());
729 const int gh = std::max(1, gfx.getPixelHeight() > 0 ? gfx.getPixelHeight() : gfx.getHeight());
730 gfx.beginGBufferPass(gw, gh);
731 for (const auto &item : items) {
732 // X-ray targets are skipped so their pixels record the occluder depth
733 // behind them; the X-ray shader samples that to detect occlusion.
734 if (item.xray) continue;
735 if (item.hair) continue;
736 if (!item.inDefaultView) continue;
737 Texture *alb = item.material ? item.material->getAlbedoTexture() : item.mr->texture;
738 const float tr = item.material ? item.material->getTintR() : item.mr->r;
739 const float tg = item.material ? item.material->getTintG() : item.mr->g;
740 const float tb = item.material ? item.material->getTintB() : item.mr->b;
741 eve::debug::rtDraw("drawMeshGBuffer", "gbuffer");
742 gfx.drawMeshGBuffer(item.mesh, cv.viewProj * item.model, item.model, cv.data->nearZ, cv.data->farZ, alb, tr,
743 tg, tb);
744 }
745 // Extra G-buffer contributors (billboard/card geometry not in the ECS).
746 for (const auto &drawer : g_gbufferDrawers) drawer(gfx, *cv.data, cv.viewProj, aspect);
747 gfx.endGBufferPass();
748 eve::debug::rtPassEnd("GBufferPass");
749 } else if (!doGBuffer) {
750 rc->getGBuffer()->clear();
751 }
752
753 if (!doForward && !doHair) return;
754
755 // GPU-driven intent must be set BEFORE begin3DFrame: when the stage-2 cull
756 // chain is live, begin3DFrame defers opening the scene color pass so the
757 // compute section can be recorded before the opaque draws.
758 const bool gpuDrivenWanted =
759 rc->isEnabled("gpuDriven") && gfx.supportsGpuDriven3D();
760 gfx.gpuDrivenSetEnabled(gpuDrivenWanted);
761
762 gfx.begin3DFrame();
763 if (!gfx.had3DThisFrame()) return;
764
765 if (!haveManager) return;
766
767 // Replay the items collected above: opaque first, hair back-to-front.
768 std::vector<const CulledItem *> opaque;
769 std::vector<const CulledItem *> hairItems;
770 opaque.reserve(items.size());
771 hairItems.reserve(items.size() / 4);
772 for (const auto &item : items) {
773 if (!item.inView) continue;
774 (item.hair ? hairItems : opaque).push_back(&item);
775 }
776 // Opaque: group by (camera, shader, material, mesh) so the backend sees
777 // long runs of identical pipeline/descriptor state instead of thrashing
778 // between materials. Hair stays sorted back-to-front by distance below.
779 std::stable_sort(opaque.begin(), opaque.end(), [](const CulledItem *a, const CulledItem *b) {
780 if (a->camIdx != b->camIdx) return a->camIdx < b->camIdx;
781 if (a->shader != b->shader) return a->shader < b->shader;
782 if (a->material != b->material) return a->material < b->material;
783 return a->mesh < b->mesh;
784 });
785 std::stable_sort(hairItems.begin(), hairItems.end(),
786 [](const CulledItem *a, const CulledItem *b) { return a->distSq > b->distSq; });
787
788 auto bindLegacyMaterial = [&](Renderable3D::MeshRenderer *mr) {
789 gfx.setMesh3DMaterial(mr->metallic, mr->roughness);
790 gfx.setMesh3DTexCellBomb(mr->texBombScale, mr->texBombStrength, mr->texBombRot);
791 gfx.setMesh3DNormalTexture(mr->normalTexture);
792 gfx.setMesh3DHeightTexture(mr->heightTexture);
793 gfx.setMesh3DParallax(mr->parallaxScale, mr->parallaxMinLayers, mr->parallaxMaxLayers);
794 gfx.setMesh3DShadowReceive(mr->receiveShadow);
795 };
796
797 // Per-camera / per-lighting state. The clustered SSBO table is uploaded at
798 // most once per camera; afterwards only the cheap active flag toggles.
799 int curCam = -1;
800 bool curLit = false;
801 bool curClustered = false;
802
803 auto drawMeshWithMaterial = [&](const CulledItem &item, CameraView &cv) {
804 auto *mr = item.mr;
805 Mesh *drawMesh = item.mesh;
806 Material *mat = item.material;
807 if (!drawMesh) return;
808
809 Texture *albedo = mr->texture;
810 Color tint(mr->r, mr->g, mr->b, mr->a);
811 Shader *shader = mr->shader;
812 if (mat) {
813 mat->bind(gfx);
814 albedo = mat->getAlbedoTexture();
815 tint = Color(mat->getTintR(), mat->getTintG(), mat->getTintB(), mat->getTintA());
816 shader = mat->effectiveShader();
817 } else {
818 bindLegacyMaterial(mr);
819 }
820
821 const bool lit = mat ? mat->getReceiveLight() : mr->receiveLight;
822 const bool clustered = lit && useClustered && !shader;
823 if (item.camIdx != curCam || lit != curLit || clustered != curClustered) {
824 if (item.camIdx != curCam) {
825 gfx.setMesh3DViewProj(cv.viewProj);
826 gfx.setMesh3DView(cv.view);
827 gfx.setMesh3DClip(cv.data->nearZ, cv.data->farZ);
828 gfx.setMesh3DCameraPos(cv.eye);
829 gfx.setMesh3DEnv(cv.data->envMap, cv.data->envIntensity);
830 curCam = item.camIdx;
831 }
832 if (clustered && cv.clusteredValid) {
833 if (!cv.clusteredUploaded) {
834 gfx.setMesh3DClusteredLighting(cv.clustered);
835 cv.clusteredUploaded = true;
836 } else {
837 gfx.setMesh3DClusteredActive(true);
838 }
839 } else {
840 gfx.setMesh3DClusteredActive(false);
841 }
842 if (lit) {
843 gfx.setMesh3DLighting(cv.lighting);
844 } else {
845 Lighting3DPack none{};
846 none.count = 0;
847 none.ambient = glm::vec4(1.f, 1.f, 1.f, 0.f);
848 gfx.setMesh3DLighting(none);
849 }
850 curLit = lit;
851 curClustered = clustered;
852 }
853
854 const glm::mat4 &model = item.model;
855 if (albedo) eve::debug::rtBind("texture", "albedo");
856 if (shader) eve::debug::rtBind("shader", item.hair ? "hair" : "mesh");
857 eve::debug::rtBind("mesh", "renderable3d");
858 eve::debug::rtDraw("drawMeshShader", shader ? "custom" : "default");
859 gfx.drawMeshShader(drawMesh, model, albedo, tint, shader);
860
861 // X-ray second pass: paint only the occluded (behind-building) part over
862 // the scene. The pipeline runs with depth test/write off + alpha blend and
863 // the shader discards visible fragments by sampling the G-buffer depth.
864 if (mr->xrayHighlight && mr->xrayShader) {
865 float sw = gfx.getPixelWidth() > 0 ? float(gfx.getPixelWidth()) : float(gfx.getWidth());
866 float shh =
867 gfx.getPixelHeight() > 0 ? float(gfx.getPixelHeight()) : float(gfx.getHeight());
868 if (mr->xrayShader->hasUniform("screenW")) mr->xrayShader->sendFloat("screenW", sw);
869 if (mr->xrayShader->hasUniform("screenH")) mr->xrayShader->sendFloat("screenH", shh);
870 eve::debug::rtBind("shader", "xray");
871 eve::debug::rtDraw("drawMeshShader", "xray");
872 gfx.drawMeshShader(drawMesh, model, albedo, tint, mr->xrayShader);
873 }
874 };
875
876 // Provide the G-buffer depth to X-ray shaders for the occlusion test.
877 if (doGBuffer && rc->getGBuffer() && rc->getGBuffer()->isValid()) {
878 if (Texture *depth = rc->getGBuffer()->getHwDepthTexture()) gfx.setMesh3DSceneDepth(depth);
879 }
880
881 if (doForward) {
882 bool gpuDrivenUsed = false;
883 if (gpuDrivenWanted && !useClustered && defaultCam && !opaque.empty()) {
884 bool eligible = true;
885 for (const CulledItem *it : opaque) {
886 if (it->camIdx != 0 || !it->material || it->mr->camera != nullptr ||
887 it->material->effectiveShader() != nullptr ||
888 !gfx.gpuDrivenMaterialUsable(it->material)) {
889 eligible = false;
890 break;
891 }
892 }
893 if (eligible) {
894 const CameraView &cv = cams[0]; // default camera is slot 0
895 const Camera3D::Data *cd = cv.data;
896 const glm::vec3 eye = cv.eye;
897 gfx.setMesh3DViewProj(cv.viewProj);
898 gfx.setMesh3DView(cv.view);
899 gfx.setMesh3DClip(cd->nearZ, cd->farZ);
900 gfx.setMesh3DCameraPos(cv.eye);
901 gfx.setMesh3DEnv(cd->envMap, cd->envIntensity);
902 ClusteredLightingUpload off{};
903 off.active = false;
904 gfx.setMesh3DClusteredLighting(off);
905 gfx.setMesh3DLighting(cv.lighting);
906
907 std::vector<eve::graphics::GpuInstance> instances;
908 instances.reserve(opaque.size());
909 bool recordsOk = true;
910 bool vgAny = false;
911 const bool resolveWanted = gfx.gpuDrivenResolveWanted();
912 for (const CulledItem *it : opaque) {
913 // Stage 3 VG: meshes with a virtual-geometry asset are culled /
914 // drawn through the cluster path, not the instance chain.
915 const uint32_t vgAsset =
916 resolveWanted ? gfx.gpuDrivenVgAssetId(it->mesh)
917 : eve::graphics::kInvalidGpuDrivenSlot;
918 if (vgAsset != eve::graphics::kInvalidGpuDrivenSlot) {
919 const uint32_t matId = gfx.gpuDrivenMaterialRecord(it->material);
920 if (matId == eve::graphics::kInvalidGpuDrivenSlot) {
921 recordsOk = false;
922 break;
923 }
924 vgAny |= gfx.gpuDrivenVgSetInstance(vgAsset, it->model, matId);
925 continue;
926 }
927 eve::graphics::GpuInstance inst{};
928 inst.model = it->model;
929 inst.meshId = gfx.gpuDrivenMeshRecord(it->mesh);
930 inst.materialId = gfx.gpuDrivenMaterialRecord(it->material);
931 if (inst.meshId == eve::graphics::kInvalidGpuDrivenSlot ||
932 inst.materialId == eve::graphics::kInvalidGpuDrivenSlot) {
933 recordsOk = false;
934 break;
935 }
936 instances.push_back(inst);
937 }
938 if (recordsOk) {
939 // Stage 2: GPU frustum/HZB cull + GPU-written indirect commands.
940 if (gfx.gpuDrivenCullEnabled() && !instances.empty() &&
941 gfx.gpuDrivenCullBegin(instances.data(), uint32_t(instances.size()))) {
942 gfx.gpuDrivenCullEmit(cv.viewProj, eye, cd->fovYDeg, cd->nearZ, cd->farZ);
943 if (gfx.gpuDrivenResolveWanted()) {
944 // Stage 3: opaque goes to the GBuffer vis pass, the
945 // scene color pass runs the fullscreen resolve.
946 gfx.gpuDrivenRecordVisPass();
947 gfx.gpuDrivenOpenScenePass();
948 gfx.gpuDrivenResolve();
949 } else {
950 gfx.gpuDrivenOpenScenePass();
951 gfx.gpuDrivenDrawOpaque();
952 }
953 gpuDrivenUsed = true;
954 } else if (gfx.gpuDrivenCullEnabled() && instances.empty() && vgAny) {
955 // VG-only frame: no instance chain; the vis pass runs the
956 // VG cluster cull + draws + resolve.
957 gfx.gpuDrivenVgComputeSection(cv.viewProj, eye, cd->fovYDeg, cd->nearZ,
958 cd->farZ);
959 gfx.gpuDrivenRecordVisPass();
960 gfx.gpuDrivenOpenScenePass();
961 gfx.gpuDrivenResolve();
962 gpuDrivenUsed = true;
963 } else {
964 // Deferred scene pass must be open before stage-1 recording.
965 gfx.gpuDrivenOpenScenePass();
966 if (gfx.gpuDrivenSubmitOpaque(instances.data(),
967 uint32_t(instances.size())))
968 gpuDrivenUsed = true;
969 }
970 }
971 }
972 }
973 if (!gpuDrivenUsed) {
974 if (gfx.gpuDrivenScenePassPending()) gfx.gpuDrivenOpenScenePass();
975 for (const CulledItem *item : opaque)
976 drawMeshWithMaterial(*item, cams[size_t(item->camIdx)]);
977 }
978 }
979 if (doHair) {
980 // Hair-only frames never reach the GPU-driven forward block above;
981 // open the deferred scene pass (if any) before recording hair draws.
982 if (gfx.gpuDrivenScenePassPending()) gfx.gpuDrivenOpenScenePass();
983 for (const CulledItem *item : hairItems) drawMeshWithMaterial(*item, cams[size_t(item->camIdx)]);
984 }
985
986 const bool doAO = rc->isEnabled("ao");
987 // WebGPU has no SPIR-V AO shaders; the X-ray path (below) still reads the
988 // G-buffer depth directly.
989 if (doAO && gfx.supportsGBufferPost() && defaultCam && gfx.had3DThisFrame()) {
990 GBuffer *gb = rc->getGBuffer();
991 if (gb && gb->isValid()) {
992 auto cd = defaultCam->data();
993 const float aspectSafe = aspect > 1e-4f ? aspect : 1.f;
994 auto bindCam = [&](auto *fx) {
995 fx->setCamera(cd->eyeX, cd->eyeY, cd->eyeZ, cd->targetX, cd->targetY, cd->targetZ,
996 cd->upX, cd->upY, cd->upZ, cd->fovYDeg, aspectSafe, cd->nearZ, cd->farZ);
997 };
998 AmbientOcclusion *ao = gfx.pipelineAmbientOcclusion();
999 ao->setQuality("medium");
1000 ao->setIntensity(0.16f);
1001 ao->setPower(1.1f);
1002 ao->setRadius(std::clamp(cd->farZ * 0.006f, 0.18f, 0.35f));
1003 bindCam(ao);
1004 if (Texture *depth = gb->getHwDepthTexture())
1005 ao->applyFromGBuffer(&gfx, depth, gb->getNormalTexture());
1006 // Fullscreen SSGI from lit scene color reprints nearby props
1007 // (curtains, planters) onto the floor as multiple swimming ghosts.
1008 // Mesh shaders still add hemispheric sky/ground + wrap fill.
1009 }
1010 }
1011
1012 const bool doOutline = rc->isEnabled("outline");
1013 if (doOutline && defaultCam && gfx.had3DThisFrame()) {
1014 GBuffer *gb = rc->getGBuffer();
1015 if (gb && gb->isValid()) {
1016 Outline *outline = gfx.pipelineOutline();
1017 auto cd = defaultCam->data();
1018 outline->setClip(cd->nearZ, cd->farZ);
1019 if (Texture *depth = gb->getHwDepthTexture())
1020 outline->apply(&gfx, depth, gb->getNormalTexture());
1021 }
1022 }
1023}

1025void RenderSystem3D::renderToCanvas(Graphics &gfx, Canvas *target, Camera3D *camera) {
1026 if (!target || !camera) return;
1027 auto cd = camera->data();
1028 const float aspect = target->getWidth() > 0
1029 ? float(target->getWidth()) / float(target->getHeight())
1030 : 1.f;
1031
1032 // Preview-quality forward pass: no shadow / G-buffer / AO passes.
1033 gfx.begin3DFrameToCanvas(target);
1034
1035 const glm::vec3 eye(cd->eyeX, cd->eyeY, cd->eyeZ);
1036 const glm::vec3 look(cd->targetX, cd->targetY, cd->targetZ);
1037 const glm::vec3 up(cd->upX, cd->upY, cd->upZ);
1038 const glm::mat4 viewM = glm::lookAtRH(eye, look, up);
1039 const float fovRad = cd->fovYDeg * 0.017453292519943295f;
1040 const glm::mat4 projM = perspectiveVulkanRH_ZO(fovRad, aspect, cd->nearZ, cd->farZ);
1041 gfx.setMesh3DViewProj(projM * viewM);
1042 gfx.setMesh3DView(viewM);
