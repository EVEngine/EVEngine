270static ReflectionProbeUpload selectReflectionProbes(const Camera3D::Data &camera,
271 const glm::vec3 &samplePosition) {
272 struct Candidate {
273 int slot = 0;
274 int priority = 0;
275 float distance2 = 0.f;
276 };
277 std::array<Candidate, Camera3D::Data::kMaxReflectionProbes> candidates{};
278 int candidateCount = 0;
279 for (int slot = 0; slot < Camera3D::Data::kMaxReflectionProbes; ++slot) {
280 const auto &probe = camera.reflectionProbes[static_cast<size_t>(slot)];
281 if (!probe.enabled || !probe.cubemap) continue;
282 const glm::vec3 outside = glm::max(glm::abs(samplePosition - probe.center) - probe.extent,
283 glm::vec3(0.f));
284 candidates[static_cast<size_t>(candidateCount++)] =
285 Candidate{slot, probe.priority, glm::dot(outside, outside)};
286 }
287 std::stable_sort(candidates.begin(), candidates.begin() + candidateCount,
288 [](const Candidate &a, const Candidate &b) {
289 if (a.priority != b.priority) return a.priority > b.priority;
290 if (a.distance2 != b.distance2) return a.distance2 < b.distance2;
291 return a.slot < b.slot;
292 });
293
294 ReflectionProbeUpload upload;
295 upload.count = std::min(candidateCount, ReflectionProbeUpload::kMaxProbes);
296 for (int i = 0; i < upload.count; ++i) {
297 const auto &source = camera.reflectionProbes[static_cast<size_t>(candidates[i].slot)];
298 auto &target = upload.probes[i];
299 target.cubemap = source.cubemap;
300 target.center = source.center;
301 target.extent = source.extent;
302 target.intensity = source.intensity;
303 target.blendDistance = source.blendDistance;
304 target.priority = source.priority;
305 }
306 return upload;
307}

309void Camera3D::screenToRay(float screenX, float screenY, float viewW, float viewH) {
310 auto d = data();
311 d->screenRayOx = d->eyeX;
312 d->screenRayOy = d->eyeY;
313 d->screenRayOz = d->eyeZ;
314 d->screenRayDx = 0.f;
315 d->screenRayDy = 0.f;
316 d->screenRayDz = -1.f;
317 if (viewW <= 0.f || viewH <= 0.f) return;
318
319 const glm::vec3 eye(d->eyeX, d->eyeY, d->eyeZ);
320 const glm::vec3 target(d->targetX, d->targetY, d->targetZ);
321 const glm::vec3 up(d->upX, d->upY, d->upZ);
322 const glm::mat4 viewM = glm::lookAtRH(eye, target, up);
323 const float aspect = viewW / viewH;
324 const float fovRad = d->fovYDeg * 0.017453292519943295f;
325 const glm::mat4 projM = cameraProjectionVulkanRH_ZO(
326 d->orthographic, fovRad, d->orthoHeight, aspect, d->nearZ, d->farZ);
327 const glm::mat4 invVP = glm::inverse(projM * viewM);
328
329 // Screen pixel → Vulkan NDC (Y-down; matches perspectiveVulkanRH_ZO).
330 const float ndcX = (screenX / viewW) * 2.f - 1.f;
331 const float ndcY = (screenY / viewH) * 2.f - 1.f;
332 auto unproject = [&](float ndcZ) -> glm::vec3 {
333 glm::vec4 w = invVP * glm::vec4(ndcX, ndcY, ndcZ, 1.f);
334 if (std::fabs(w.w) < 1e-8f) return eye;
335 w /= w.w;
336 return glm::vec3(w);
337 };
338 // ZO depth: near = 0, far = 1.
339 const glm::vec3 nearPt = unproject(0.f);
340 const glm::vec3 farPt = unproject(1.f);
341 glm::vec3 dir = farPt - nearPt;
342 const float len = glm::length(dir);
343 if (len > 1e-8f) dir /= len;
344 else dir = glm::normalize(target - eye);
345
346 d->screenRayOx = eye.x;
347 d->screenRayOy = eye.y;
348 d->screenRayOz = eye.z;
349 if (d->orthographic) {
350 d->screenRayOx = nearPt.x;
351 d->screenRayOy = nearPt.y;
352 d->screenRayOz = nearPt.z;
353 }
354 d->screenRayDx = dir.x;
355 d->screenRayDy = dir.y;
356 d->screenRayDz = dir.z;
357}

614Light3D::Data *findShadowCasterDir(const std::vector<PackedLight3D> &packed) {
615 Light3D::Data *best = nullptr;
616 float bestI = -1.f;
617 for (const auto &pl : packed) {
618 if (pl.isPoint || !pl.data || !pl.data->castShadow) continue;
619 if (pl.data->intensity > bestI) {
620 bestI = pl.data->intensity;
621 best = pl.data;
622 }
623 }
624 return best;
625}

627void prioritizeShadowCaster(std::vector<PackedLight3D> &packed, Light3D::Data *caster) {
628 if (!caster || packed.empty()) return;
629 for (size_t i = 0; i < packed.size(); ++i) {
630 if (packed[i].data != caster) continue;
631 if (i != 0) std::swap(packed[0], packed[i]);
632 return;
633 }

640struct FrustumPlanes {
641 glm::vec4 p[6]{};
642
643 bool sphereVisible(const glm::vec3 &center, float radius) const {
644 for (const auto &pl : p) {
645 const float d = pl.x * center.x + pl.y * center.y + pl.z * center.z + pl.w;
646 if (d < -radius) return false;
647 }
648 return true;
649 }
650};
651
652FrustumPlanes extractFrustum(const glm::mat4 &m) {
653 FrustumPlanes f;
654 const glm::vec4 r0(m[0][0], m[1][0], m[2][0], m[3][0]);
655 const glm::vec4 r1(m[0][1], m[1][1], m[2][1], m[3][1]);
656 const glm::vec4 r2(m[0][2], m[1][2], m[2][2], m[3][2]);
657 const glm::vec4 r3(m[0][3], m[1][3], m[2][3], m[3][3]);
658 auto norm = [](glm::vec4 &v) {
659 const float l = glm::length(glm::vec3(v));
660 if (l > 1e-8f) v /= l;
661 };
662 f.p[0] = r3 + r0; // left
663 f.p[1] = r3 - r0; // right
664 f.p[2] = r3 + r1; // bottom
665 f.p[3] = r3 - r1; // top
666 // Vulkan clip space uses zero-to-one depth: near plane is z_clip = 0
667 // (plane r2), not the z = -w plane used by OpenGL-style [-1,1] depth.
668 f.p[4] = r2; // near
669 f.p[5] = r3 - r2; // far
670 for (auto &pl : f.p) norm(pl);
671 return f;
672}

679struct CameraView {
680 Camera3D::Data *data = nullptr;
681 glm::vec3 eye{0.f};
682 glm::mat4 view{1.f};
683 glm::mat4 proj{1.f};
684 glm::mat4 viewProj{1.f};
685 glm::mat4 cullViewProj{1.f};
686 FrustumPlanes frustum;
687 float fovRad = 1.f;
688 Lighting3DPack lighting{};
689 ClusteredLightingUpload clustered{};
690 bool clusteredValid = false;
692 bool clusteredUploaded = false;
693};
694
695CameraView buildCameraView(Camera3D::Data *cd, const std::vector<PackedLight3D> &packed, bool useClustered,
696 float aspect, const std::vector<ClusteredLightGpu> &clusteredPoints,
697 const std::vector<ClusteredLightGpu> &clusteredDirs, Graphics &gfx,
698 const glm::vec2 &jitterNdc) {
699 CameraView cv;
700 cv.data = cd;
701 cv.eye = glm::vec3(cd->eyeX, cd->eyeY, cd->eyeZ);
702 const glm::vec3 look(cd->targetX, cd->targetY, cd->targetZ);
703 const glm::vec3 up(cd->upX, cd->upY, cd->upZ);
704 cv.view = glm::lookAtRH(cv.eye, look, up);
705 cv.fovRad = cd->fovYDeg * 0.017453292519943295f;
706 cv.proj = cameraProjectionVulkanRH_ZO(cd->orthographic, cv.fovRad,
707 cd->orthoHeight, aspect,
708 cd->nearZ, cd->farZ);
709 cv.cullViewProj = cv.proj * cv.view;
710 cv.frustum = extractFrustum(cv.cullViewProj);
711 cv.proj[2][0] += jitterNdc.x;
712 cv.proj[2][1] += jitterNdc.y;
713 cv.viewProj = cv.proj * cv.view;
714 cv.lighting = packLights3D(packed, cd);
715 if (useClustered && !cd->orthographic) {
716 cv.clustered = buildClusteredLighting(clusteredPoints, clusteredDirs, cv.view, cd->nearZ, cd->farZ,
717 gfx.getWidth(), gfx.getHeight(), cv.fovRad,
718 glm::vec4(cd->ambientR, cd->ambientG, cd->ambientB, 0.f));
719 cv.clusteredValid = true;
720 }
721 return cv;
722}

752void RenderSystem3D::render(Graphics &gfx) {
753 eve::debug::RenderPassScope pass3d("RenderSystem3D");
754 Camera3D *defaultCam = findDefaultCamera3D();
755
756 RenderControl *rc = gfx.getRenderControl();
757 rc->ensureCompiled();
758 const bool doShadow = rc->hasPass("shadow");
759 const bool doGBuffer = rc->hasPass("gbuffer");
760 const bool doDecal = rc->hasPass("decal");
761 const bool doForward = rc->hasPass("forward");
762 const bool doHair = rc->hasPass("hair");
763 const bool allowClustered = rc->isEnabled("clustered");
764
765 std::vector<PackedLight3D> packed;
766 collectLights3D(packed, size_t(ClusteredLightConfig::kMaxLights));
767 promoteDirectional(packed);
768 Light3D::Data *shadowCaster = doShadow ? findShadowCasterDir(packed) : nullptr;
769 prioritizeShadowCaster(packed, shadowCaster);
770 const bool haveExtraShadowCasters = doShadow && !g_shadowDrawers.empty();
771
772 const float aspect = (gfx.getHeight() > 0) ? float(gfx.getWidth()) / float(gfx.getHeight()) : 1.f;
773
774 ShadowUpload shadowUpload{};
775 shadowUpload.active = false;
776 // Per-cascade light frustums for caster culling (sphere vs frustum).
777 FrustumPlanes cascadeFrustums[ShadowConfig::kCascades];
778 if ((shadowCaster || haveExtraShadowCasters) && defaultCam) {
779 auto cd = defaultCam->data();
780 glm::vec3 dir = shadowCaster ? glm::vec3(shadowCaster->dx, shadowCaster->dy, shadowCaster->dz) : gLightDir;
781 if (glm::length(dir) < 1e-6f)
782 dir = glm::vec3(0.f, 1.f, 0.f);
783 else
784 dir = glm::normalize(dir);
785 const float shadowBias = shadowCaster ? shadowCaster->shadowBias : 0.003f;
786 const float shadowStrength = shadowCaster ? shadowCaster->shadowStrength : 1.f;
787 const float fovRad = cd->fovYDeg * 0.017453292519943295f;
788 shadowUpload = buildDirectionalCSM(
789 dir, glm::vec3(cd->eyeX, cd->eyeY, cd->eyeZ), glm::vec3(cd->targetX, cd->targetY, cd->targetZ),
790 glm::vec3(cd->upX, cd->upY, cd->upZ), fovRad, aspect, cd->nearZ, cd->farZ, shadowBias, shadowStrength);
791 for (int c = 0; c < ShadowConfig::kCascades; ++c)
792 cascadeFrustums[c] = extractFrustum(shadowUpload.ubo.lightVP[c]);
793 }
794 gfx.setMesh3DShadows(shadowUpload);
795
796 const bool useClustered = allowClustered && packed.size() > size_t(Lighting3DPack::kMaxLights);
797 // Light split / directional promotion is camera-independent: compute once,
798 // then each camera view bakes its own clustered table from these lists.
799 std::vector<ClusteredLightGpu> clusteredPoints;
800 std::vector<ClusteredLightGpu> clusteredDirs;
801 if (useClustered) {
802 splitLights(packed, clusteredPoints, clusteredDirs);
803 if (clusteredDirs.empty()) {
804 ClusteredLightGpu d{};
805 d.posRadius = glm::vec4(gLightDir, 0.f);
806 d.color = glm::vec4(gLightColor, 1.f);
807 clusteredDirs.push_back(d);
808 }
809 if (shadowCaster && !clusteredDirs.empty()) {
810 glm::vec3 dir(shadowCaster->dx, shadowCaster->dy, shadowCaster->dz);
811 if (glm::length(dir) < 1e-6f)
812 dir = glm::vec3(0.f, 1.f, 0.f);
813 else
814 dir = glm::normalize(dir);
815 for (size_t i = 0; i < clusteredDirs.size(); ++i) {
816 if (glm::length(glm::vec3(clusteredDirs[i].posRadius) - dir) < 1e-3f) {
817 if (i != 0) std::swap(clusteredDirs[0], clusteredDirs[i]);
818 break;
819 }
820 }
821 }
822 }
823
824 const bool haveManager = ecs::current()->getManager<Renderable3D>() != nullptr;
825 const bool shadowActive = doShadow && shadowUpload.active;
826
827 // Per-camera constants (matrices, frustum, lighting, clustered table).
828 // The default camera is slot 0 so the G-buffer pass reuses its view.
829 std::vector<CameraView> cams;
830 cams.reserve(2);
831 glm::vec2 temporalJitter(0.f);
832 AntiAliasing *temporalAA = nullptr;
833 if (defaultCam && rc->isEnabled("taa")) {
834 auto cd = defaultCam->data();
835 temporalAA = gfx.pipelineAntiAliasing();
836 temporalAA->setTemporalCamera(glm::vec3(cd->eyeX, cd->eyeY, cd->eyeZ),
837 glm::vec3(cd->targetX, cd->targetY, cd->targetZ),
838 cd->fovYDeg);
839 temporalJitter = temporalAA->prepareTemporalJitter(gfx.getWidth(), gfx.getHeight());
840 }
841 if (defaultCam) {
842 cams.push_back(buildCameraView(defaultCam->data().operator->(), packed, useClustered, aspect, clusteredPoints,
843 clusteredDirs, gfx, temporalJitter));
844 if (temporalAA) {
845 auto cd = defaultCam->data();
846 temporalAA->setTemporalViewProjection(cams.front().viewProj, cd->nearZ, cd->farZ);
847 }
848 }
849 auto findOrAddCam = [&](Camera3D *camEnt) -> int {
850 Camera3D::Data *d = camEnt->data().operator->();
851 for (size_t i = 0; i < cams.size(); ++i) {
852 if (cams[i].data == d) return int(i);
853 }
854 cams.push_back(buildCameraView(d, packed, useClustered, aspect, clusteredPoints,
855 clusteredDirs, gfx, temporalJitter));
856 return int(cams.size()) - 1;
857 };
858
859 // Single ECS traversal: one model matrix, one LOD pick, one set of
860 // sphere-vs-frustum tests per part. Passes below only replay the list.
861 std::vector<CulledItem> items;
862 items.reserve(64);
863 if (haveManager) {
864 auto view = ecs::View<Renderable3D, Renderable3D::Transform3D, Renderable3D::MeshRenderer>();
865 for (auto it = view.begin(); it != view.end(); ++it) {
866 auto [xf, mr] = *it;
867 if (!mr->visible) continue;
868 Camera3D *camEnt = mr->camera ? mr->camera : defaultCam;
869 if (!camEnt) continue;
870 const int camIdx = findOrAddCam(camEnt);
871 const CameraView &cv = cams[size_t(camIdx)];
872 const float dx = xf->x - cv.eye.x;
873 const float dy = xf->y - cv.eye.y;
874 const float dz = xf->z - cv.eye.z;
875 const float distSq = dx * dx + dy * dy + dz * dz;
876 const float dist = std::sqrt(distSq);
877 const glm::mat4 model = modelFromTransform(*xf);
878 const float maxScale = std::max(std::abs(xf->sx), std::max(std::abs(xf->sy), std::abs(xf->sz)));
879 const bool shadowOk = mr->effectiveCastShadow();
880
881 auto pushPart = [&](Mesh *drawMesh, Material *mat, bool asHair, bool castsShadow,
882 const ModelPart* modelPart) {
883 if (!drawMesh) return;
884 CulledItem item;
885 item.mr = mr;
886 item.mesh = drawMesh;
887 item.material = mat;
888 item.shader = mat ? mat->effectiveShader() : mr->shader;
889 item.model = model;
890 if (temporalAA)
891 item.temporalMotion = temporalAA->prepareTemporalObjectMotion(mr, model);
892 item.distSq = distSq;
893 const glm::vec4 viewCenter = cv.view * glm::vec4(xf->x, xf->y, xf->z, 1.f);
894 item.projectedDepth = -viewCenter.z;
895 item.camIdx = camIdx;
896 item.hair = asHair;
897 item.surfaceMode = mat ? mat->surfaceMode()
898 : (asHair ? SurfaceMode::Transparent
899 : SurfaceMode::Opaque);
900 item.sortPriority = modelPart && modelPart->hasSortPriority
901 ? modelPart->sortPriority
902 : (mat ? mat->getSortPriority() : 0);
903 item.xray = mr->xrayHighlight;
904 if (drawMesh->hasBounds()) {
905 const glm::vec4 c4 =
906 model * glm::vec4(drawMesh->boundsCx, drawMesh->boundsCy, drawMesh->boundsCz, 1.f);
907 item.worldC = glm::vec3(c4);
908 item.worldR = drawMesh->boundsRadius * maxScale;
909 item.inView = cv.frustum.sphereVisible(item.worldC, item.worldR);
910 item.inDefaultView = defaultCam ? cams[0].frustum.sphereVisible(item.worldC, item.worldR) : true;
911 if (shadowActive && castsShadow) {
912 for (int c = 0; c < ShadowConfig::kCascades; ++c) {
913 if (cascadeFrustums[c].sphereVisible(item.worldC, item.worldR))
914 item.cascadeMask |= (1u << c);
915 }
916 }
917 } else {
918 // No bounds (e.g. legacy/imported mesh): never cull.
919 item.inView = true;
920 item.inDefaultView = true;
921 if (shadowActive && castsShadow) item.cascadeMask = (1u << ShadowConfig::kCascades) - 1u;
922 }
923 items.push_back(item);
924 };
925
926 if (mr->usesParts()) {
927 for (int p = 0; p < mr->partCount; ++p) {
928 Material *mat = mr->parts[p].material ? mr->parts[p].material : mr->material;
929 const bool asHair = mat ? mat->isTransparentHair() : mr->isHair;
930 const bool casts = shadowOk && !(mat && !mat->getCastShadow());
931 pushPart(mr->parts[p].mesh, mat, asHair, casts, &mr->parts[p]);
932 }
933 } else {
934 Mesh *drawMesh = mr->meshForDistance(dist);
935 pushPart(drawMesh, mr->material, mr->effectiveHair(), shadowOk, nullptr);
936 }
937 }
938 }
939
940 // CSM shadow passes — replay the collected casters, culled per cascade.
941 if (shadowActive && (haveManager || haveExtraShadowCasters)) {
942 auto cd = defaultCam->data();
943 for (int c = 0; c < ShadowConfig::kCascades; ++c) {
944 eve::debug::rtPassBegin("ShadowPass");
945 gfx.beginShadowPass(c);
946 for (const auto &item : items) {
947 if ((item.cascadeMask & (1u << c)) == 0) continue;
948 eve::debug::rtBind("mesh", "shadowCaster");
949 eve::debug::rtDraw("drawMeshShadow", "cascade");
950 Texture *shadowAlbedo =
951 item.material ? item.material->getAlbedoTexture() : item.mr->texture;
952 if (item.surfaceMode == SurfaceMode::Masked)
953 gfx.drawMeshShadowAlpha(item.mesh, shadowUpload.ubo.lightVP[c] * item.model,
954 shadowAlbedo);
955 else if (item.surfaceMode != SurfaceMode::Transparent)
956 gfx.drawMeshShadow(item.mesh, shadowUpload.ubo.lightVP[c] * item.model);
957 }
958 // Extra shadow casters (billboard/card geometry not in the ECS).
959 for (const auto &drawer : g_shadowDrawers) drawer(gfx, shadowUpload.ubo.lightVP[c], *cd);
960 gfx.endShadowPass();
961 eve::debug::rtPassEnd("ShadowPass");
962 }
963 }
964
965 // G-buffer fill (sampleable depth/normal) — before the forward swapchain pass.
966 if (doGBuffer && defaultCam && (haveManager || !g_gbufferDrawers.empty())) {
967 eve::debug::rtPassBegin("GBufferPass");
968 const CameraView &cv = cams[0]; // default camera is slot 0
969 const int gw = std::max(1, gfx.getPixelWidth() > 0 ? gfx.getPixelWidth() : gfx.getWidth());
970 const int gh = std::max(1, gfx.getPixelHeight() > 0 ? gfx.getPixelHeight() : gfx.getHeight());
971 gfx.beginGBufferPass(gw, gh);
972 for (const auto &item : items) {
973 // X-ray targets are skipped so their pixels record the occluder depth
974 // behind them; the X-ray shader samples that to detect occlusion.
975 if (item.xray) continue;
976 if (item.surfaceMode == SurfaceMode::Transparent) continue;
977 if (!item.inDefaultView) continue;
978 Texture *alb = item.material ? item.material->getAlbedoTexture() : item.mr->texture;
979 const float tr = item.material ? item.material->getTintR() : item.mr->r;
980 const float tg = item.material ? item.material->getTintG() : item.mr->g;
981 const float tb = item.material ? item.material->getTintB() : item.mr->b;
982 const float roughness =
983 item.material ? item.material->getRoughness() : item.mr->roughness;
984 const float metallic =
985 item.material ? item.material->getMetallic() : item.mr->metallic;
986 eve::debug::rtDraw("drawMeshGBuffer", "gbuffer");
987 if (item.surfaceMode == SurfaceMode::Masked)
988 gfx.drawMeshGBufferAlpha(item.mesh, cv.viewProj * item.model, item.model,
989 cv.data->nearZ, cv.data->farZ, alb, tr, tg, tb,
990 item.temporalMotion.x, item.temporalMotion.y, roughness,
991 metallic);
992 else
993 gfx.drawMeshGBuffer(item.mesh, cv.viewProj * item.model, item.model,
994 cv.data->nearZ, cv.data->farZ, alb, tr, tg, tb,
995 item.temporalMotion.x, item.temporalMotion.y, roughness,
996 metallic);
997 }
998 // Extra G-buffer contributors (billboard/card geometry not in the ECS).
999 for (const auto &drawer : g_gbufferDrawers) drawer(gfx, *cv.data, cv.viewProj, aspect);
1000 gfx.endGBufferPass();
1001 eve::debug::rtPassEnd("GBufferPass");
1002 } else if (!doGBuffer) {
1003 rc->getGBuffer()->clear();
1004 }
1005
1006 // Screen-space decal layer: box-projected decals read the G-buffer
1007 // depth/normal and write albedo/normal/params targets sampled by
1008 // mesh3d.frag before lighting. Skipped when no decals are registered or
1009 // the backend cannot run the pass (WebGPU).
1010 if (doDecal && defaultCam && !g_decalDrawers.empty() && gfx.supportsDecal()) {
1011 eve::debug::rtPassBegin("DecalPass");
1012 const CameraView &cv = cams[0];
1013 const int dw = std::max(1, gfx.getPixelWidth() > 0 ? gfx.getPixelWidth() : gfx.getWidth());
1014 const int dh = std::max(1, gfx.getPixelHeight() > 0 ? gfx.getPixelHeight() : gfx.getHeight());
1015 gfx.beginDecalPass(dw, dh);
1016 gfx.setDecalCamera(cv.viewProj, cv.data->nearZ, cv.data->farZ);
1017 for (const auto &drawer : g_decalDrawers) drawer(gfx, *cv.data, cv.viewProj, aspect);
1018 gfx.endDecalPass();
1019 eve::debug::rtPassEnd("DecalPass");
1020 }
1021
1022 if (!doForward && !doHair) return;
1023
1024 // GPU-driven intent must be set BEFORE begin3DFrame: when the stage-2 cull
1025 // chain is live, begin3DFrame defers opening the scene color pass so the
1026 // compute section can be recorded before the opaque draws.
1027 const bool gpuDrivenWanted =
1028 rc->isEnabled("gpuDriven") && gfx.supportsGpuDriven3D();
1029 gfx.gpuDrivenSetEnabled(gpuDrivenWanted);
1030
1031 if (defaultCam) {
1032 auto cd = defaultCam->data();
1033 gfx.setSceneExposure(std::exp2(cd->exposureEV));
1034 gfx.setSceneAutoExposure(cd->autoExposure, cd->autoExposureMinEV,
1035 cd->autoExposureMaxEV);
1036 gfx.setSceneBloom(cd->bloomIntensity, cd->bloomThreshold);
1037 }
1038 gfx.begin3DFrame();
1039 if (!gfx.had3DThisFrame()) return;
1040
1041 auto drawPersistentPrimitives = [&]() {
1042 if (cams.empty()) return;
1043 const CameraView &camera = cams.front();
1044 SceneDrawContext context;
1045 context.view = camera.view;
1046 context.projection = camera.proj;
1047 context.cameraPosition = camera.eye;

1213 off.active = false;
1214 gfx.setMesh3DClusteredLighting(off);
1215 gfx.setMesh3DLighting(cv.lighting);

1217 std::vector<eve::graphics::GpuInstance> instances;
1218 instances.reserve(opaque.size());

1220 bool vgAny = false;

1224 selectReflectionProbes(*cd, it->worldC);

1226 // drawn through the cluster path, not the instance chain.

1228 resolveWanted ? gfx.gpuDrivenVgAssetId(it->mesh)
1229 : eve::graphics::kInvalidGpuDrivenSlot;

1234 break;

1237 continue;

1240 inst.model = it->model;

1242 inst.materialId = gfx.gpuDrivenMaterialRecord(it->material);
1243 inst.reflectionProbeSlots.z = uint32_t(probes.count);
1244 for (int probeIndex = 0; probeIndex < probes.count; ++probeIndex) {

1247 if (slot == eve::graphics::kInvalidGpuDrivenSlot) {
1248 recordsOk = false;
1249 break;

1253 glm::vec4(probe.center, probe.intensity);
1254 inst.reflectionProbeExtent[probeIndex] =
1255 glm::vec4(probe.extent, probe.blendDistance);

1259 inst.materialId == eve::graphics::kInvalidGpuDrivenSlot) {
1260 recordsOk = false;

1273 // scene color pass runs the fullscreen resolve.
1274 gfx.gpuDrivenRecordVisPass();
1275 gfx.gpuDrivenOpenScenePass();
1276 gfx.gpuDrivenResolve();

1280 }
1281 gpuDrivenUsed = true;
1282 } else if (gfx.gpuDrivenCullEnabled() && instances.empty() && vgAny) {

1284 // VG cluster cull + draws + resolve.
1285 gfx.gpuDrivenVgComputeSection(cv.cullViewProj, eye, cd->fovYDeg, cd->nearZ,
1286 cd->farZ);
1287 gfx.gpuDrivenRecordVisPass();
1288 gfx.gpuDrivenOpenScenePass();
1289 gfx.gpuDrivenResolve();

1291 } else {

1293 gfx.gpuDrivenOpenScenePass();
1294 if (gfx.gpuDrivenSubmitOpaque(instances.data(),

1300 }

1309 // hair pass remains independently switchable for hair materials.
1310 if (gfx.gpuDrivenScenePassPending()) gfx.gpuDrivenOpenScenePass();
1311 for (const CulledItem *item : transparentItems) {
1312 if ((item->hair && doHair) || (!item->hair && doForward))

1314 }

1320 const bool doRTGI = rc->isEnabled("rtgi") || rc->isEnabled("reflectionChain");
1321 const bool doSSR = rc->isEnabled("ssr") || rc->isEnabled("reflectionChain");
1322 bool aoApplied = false;

1325 GBuffer *gb = rc->getGBuffer();

1329 auto bindCam = [&](auto *fx) {
1330 fx->setCamera(cd->eyeX, cd->eyeY, cd->eyeZ, cd->targetX, cd->targetY, cd->targetZ,
1331 cd->upX, cd->upY, cd->upZ, cd->fovYDeg, aspectSafe, cd->nearZ, cd->farZ);
1332 };

1335 ao->setIntensity(0.16f);
1336 ao->setPower(1.1f);
1337 ao->setRadius(std::clamp(cd->farZ * 0.006f, 0.18f, 0.35f));
1338 bindCam(ao);

1340 // post-process layout. Its G-buffer already carries equivalent
1341 // linear depth in a filterable RGBA target.
1342 Texture *depth = gfx.getBackendName() == "webgpu" ? gb->getDepthTexture()
1343 : gb->getHwDepthTexture();
1344 if (depth)
1345 ao->applyFromGBuffer(&gfx, depth, gb->getNormalTexture());
1346 // Fullscreen SSGI from lit scene color reprints nearby props
1347 // (curtains, planters) onto the floor as multiple swimming ghosts.
1348 // Mesh shaders still add hemispheric sky/ground + wrap fill.
1349 }
1350 aoApplied = true;

1356 if (gb && gb->isValid()) {

1359 : gb->getHwDepthTexture();

1362 const std::string reflectionQuality = rc->getReflectionQuality();

1364 reflectionQuality == "medium" ? 6 : 8;

1367 auto cd = defaultCam->data();
1368 const float aspectSafe = aspect > 1e-4f ? aspect : 1.f;
1369 const float dw =
1370 gfx.getCanvas() ? float(gfx.getCanvas()->getWidth()) : float(gfx.getWidth());
1371 const float dh =
1372 gfx.getCanvas() ? float(gfx.getCanvas()->getHeight()) : float(gfx.getHeight());
1373 auto bindCam = [&](auto *fx) {
1374 fx->setCamera(cd->eyeX, cd->eyeY, cd->eyeZ, cd->targetX, cd->targetY,
1375 cd->targetZ, cd->upX, cd->upY, cd->upZ, cd->fovYDeg, aspectSafe,
1376 cd->nearZ, cd->farZ);
1377 if (!cams.empty()) fx->setInvViewProj(glm::inverse(cams.front().viewProj));
1378 };

1381 Texture *ssrTexture = nullptr;
1382
1383 if (doRTGI) {
1384 GlobalIllumination *gi = gfx.pipelineGlobalIllumination();

1387 gi->setWorldNormalTexture(gb->getNormalTexture());
1388 gi->setAlbedoTexture(gb->getAlbedoTexture());
1389 gi->setTemporalMotionTexture(gb->getDepthTexture());
1390 gi->setDepthPyramid(depthAtlas, depthLevels);

1393 if (rtgiCanvas) {
1394 gi->applyFromSceneTo(&gfx, sceneColor, depth, rtgiCanvas);
1395 rtgiTexture = gi->getWorkingTexture();
1396 }
1397 }
1398
1399 if (doSSR) {
1400 ScreenSpaceReflection *ssr = gfx.pipelineScreenSpaceReflection();
1401 if (ssr->getQuality() != rc->getReflectionQuality())
1402 ssr->setQuality(rc->getReflectionQuality());
1403 ssr->setEnabled(true);
1404 ssr->setTemporalMotionTexture(gb->getDepthTexture());
1405 ssr->setDepthPyramid(depthAtlas, depthLevels);
1406 bindCam(ssr);
1407 Canvas *reflCanvas = ssr->getReflectionCanvas();
1408 if (reflCanvas) {
1409 ssr->applyFromSceneTo(&gfx, sceneColor, depth, gb->getNormalTexture(),
1410 gb->getAlbedoTexture(), reflCanvas);
1411 ssrTexture = ssr->getReflectionTexture();
1412 }
1413 }
1414
1415 // All offscreen reflection work is complete. Queue AO and the
1416 // reflection overlays only now, so a later setCanvas() cannot
1417 // force an early swapchain present with pending overlays.

1420 if (gfx.getBackendName() == "vulkan") {
1421 if (rtgiTexture)
1422 gfx.drawTexturedRectShaderUV(
1423 rtgiTexture, nullptr, 0.f, 0.f, dw, dh, 0.f, 0.f, 1.f, 1.f,
1424 Color(1.f, 1.f, 1.f, 1.f), false, BlendMode::Additive);
1425 if (ssrTexture)
1426 gfx.drawTexturedRectShaderUV(
1427 ssrTexture, nullptr, 0.f, 0.f, dw, dh, 0.f, 0.f, 1.f, 1.f,
1428 Color(1.f, 1.f, 1.f, 1.f), false, BlendMode::Additive);
1429 } else if (Canvas *composite =
1430 gfx.pipelineReflectionComposite(int(dw), int(dh))) {
1431 // Register before setCanvas(): switching away from the scene target
1432 // closes its pass and queues the final resolve immediately.
1433 gfx.setFinalSceneTexture(composite->getTexture());
1434 Canvas *previous = gfx.getCanvas();
1435 gfx.setCanvas(composite);
1436 gfx.drawTexturedRectShaderUV(
1437 sceneColor, nullptr, 0.f, 0.f, dw, dh, 0.f, 0.f, 1.f, 1.f,
1438 Color(1.f, 1.f, 1.f, 1.f), false, BlendMode::Opaque);
1439 if (rtgiTexture)
1440 gfx.drawTexturedRectShaderUV(
1441 rtgiTexture, nullptr, 0.f, 0.f, dw, dh, 0.f, 0.f, 1.f, 1.f,
1442 Color(1.f, 1.f, 1.f, 1.f), false, BlendMode::Additive);
1443 if (ssrTexture)
1444 gfx.drawTexturedRectShaderUV(
1445 ssrTexture, nullptr, 0.f, 0.f, dw, dh, 0.f, 0.f, 1.f, 1.f,
1446 Color(1.f, 1.f, 1.f, 1.f), false,
1447 BlendMode::Premultiplied);
1448 gfx.setCanvas(previous);
1449 }
1450 }
1451 }
1452 }
1453 }
1454 if (!aoApplied) applyAO();
1455
1456 const bool doOutline = rc->isEnabled("outline");
1457 if (doOutline && defaultCam && gfx.had3DThisFrame()) {
1458 GBuffer *gb = rc->getGBuffer();
1459 if (gb && gb->isValid()) {
1460 Outline *outline = gfx.pipelineOutline();
1461 auto cd = defaultCam->data();
1462 outline->setClip(cd->nearZ, cd->farZ);
1463 Texture *depth = gfx.getBackendName() == "webgpu" ? gb->getDepthTexture()
1464 : gb->getHwDepthTexture();
1465 if (depth)
1466 outline->apply(&gfx, depth, gb->getNormalTexture());
1467 }
1468 }

1500 gfx.setSceneAutoExposure(cd->autoExposure, cd->autoExposureMinEV,

1515 direction.posRadius = glm::vec4(gLightDir, 0.f);

1520 clusteredPoints, clusteredDirections, viewM, cd->nearZ, cd->farZ,
1521 target->getWidth(), target->getHeight(),
1522 cd->fovYDeg * 0.017453292519943295f,
1523 glm::vec4(cd->ambientR, cd->ambientG, cd->ambientB, 0.f));
1524 clusteredCapture = clustered.active;

1528 noClustered.active = false;
1529 gfx.setMesh3DClusteredLighting(noClustered);
