38 std::vector<std::pair<int, int>> result;
39 for (int y = 0; y < grid.getHeight(); ++y)

45std::vector<std::vector<std::pair<int, int>>> islands(const Grid2D& grid) {
46 std::vector<std::vector<std::pair<int, int>>> result;
47 std::vector<unsigned char> visited(std::size_t(grid.getWidth() * grid.getHeight()));
48 constexpr int dx[] = {-1, 1, 0, 0};
49 constexpr int dy[] = {0, 0, -1, 1};
50 for (int y = 0; y < grid.getHeight(); ++y) {
51 for (int x = 0; x < grid.getWidth(); ++x) {
52 const int start = y * grid.getWidth() + x;
53 if (!occupied(grid, x, y) || visited[std::size_t(start)]) continue;
54 result.emplace_back();
55 std::queue<std::pair<int, int>> pending;
56 pending.emplace(x, y);
57 visited[std::size_t(start)] = 1;
58 while (!pending.empty()) {
59 const auto current = pending.front();
60 pending.pop();
61 result.back().push_back(current);
62 for (int direction = 0; direction < 4; ++direction) {
63 const int nx = current.first + dx[direction];
64 const int ny = current.second + dy[direction];
65 if (nx < 0 || ny < 0 || nx >= grid.getWidth() || ny >= grid.getHeight()) continue;
66 const int index = ny * grid.getWidth() + nx;
67 if (occupied(grid, nx, ny) && !visited[std::size_t(index)]) {
68 visited[std::size_t(index)] = 1;
69 pending.emplace(nx, ny);
70 }
71 }
72 }
73 }
74 }
75 return result;
76}

78Grid2D cellular(const GenerateSettings& settings) {
79 Grid2D grid = makeGrid(settings);
80 std::mt19937_64 rng(settings.seed);
81 std::uniform_real_distribution<float> probability(0.f, 1.f);
82 for (int y = 0; y < settings.height; ++y)
83 for (int x = 0; x < settings.width; ++x)
84 if (probability(rng) < settings.x) grid.setCell(x, y, settings.semantic);
85 for (int step = 0; step < settings.a; ++step) {
86 Grid2D next = grid;
87 for (int y = 0; y < settings.height; ++y) {
88 for (int x = 0; x < settings.width; ++x) {
89 int neighbors = 0;
90 for (int oy = -1; oy <= 1; ++oy)
91 for (int ox = -1; ox <= 1; ++ox)
92 if ((ox != 0 || oy != 0) && occupied(grid, x + ox, y + oy)) ++neighbors;
93 next.setCell(
94 x, y,
95 neighbors > 4 ? settings.semantic : (neighbors < 4 ? int(Semantic::Empty) : grid.getCell(x, y)));
96 }
97 }
98 grid = std::move(next);
99 }
100 return grid;
101}

103Grid2D randomWalk(const GenerateSettings& settings) {
104 Grid2D grid = makeGrid(settings);
105 std::mt19937_64 rng(settings.seed);
106 std::uniform_int_distribution<int> startX(0, settings.width - 1);
107 std::uniform_int_distribution<int> startY(0, settings.height - 1);
108 std::uniform_int_distribution<int> direction(0, 3);
109 int x = settings.b != 0 ? std::clamp(settings.b, 0, settings.width - 1) : startX(rng);
110 int y = settings.c != 0 ? std::clamp(settings.c, 0, settings.height - 1) : startY(rng);
111 const int iterations = std::max(1, settings.a);
112 const int length = std::max(1, int(settings.x));
113 constexpr int dx[] = {-1, 1, 0, 0};
114 constexpr int dy[] = {0, 0, -1, 1};
115 for (int iteration = 0; iteration < iterations; ++iteration) {
116 for (int step = 0; step < length; ++step) {
117 setOccupied(grid, x, y, settings.semantic);
118 const int d = direction(rng);
119 x = std::clamp(x + dx[d], 0, settings.width - 1);
120 y = std::clamp(y + dy[d], 0, settings.height - 1);
121 }
122 if (settings.y > 0.5f) {
123 x = startX(rng);
124 y = startY(rng);
125 }
126 }
127 return grid;
128}

130Grid2D maze(const GenerateSettings& settings) {
131 Grid2D grid = makeGrid(settings);
132 if (settings.width < 3 || settings.height < 3) return grid;
133 std::mt19937_64 rng(settings.seed);
134 std::vector<std::pair<int, int>> stack{{1, 1}};
135 setOccupied(grid, 1, 1, settings.semantic);
136 constexpr int dx[] = {-2, 2, 0, 0};
137 constexpr int dy[] = {0, 0, -2, 2};
138 while (!stack.empty()) {
139 const auto current = stack.back();
140 std::vector<int> candidates;
141 for (int d = 0; d < 4; ++d) {
142 const int nx = current.first + dx[d];
143 const int ny = current.second + dy[d];
144 if (nx > 0 && ny > 0 && nx < settings.width - 1 && ny < settings.height - 1 && !occupied(grid, nx, ny))
145 candidates.push_back(d);
146 }
147 if (candidates.empty()) {
148 stack.pop_back();
149 continue;
150 }
151 std::shuffle(candidates.begin(), candidates.end(), rng);
152 const int d = candidates.front();
153 const int nx = current.first + dx[d];
154 const int ny = current.second + dy[d];
155 setOccupied(grid, current.first + dx[d] / 2, current.second + dy[d] / 2, settings.semantic);
156 setOccupied(grid, nx, ny, settings.semantic);
157 stack.emplace_back(nx, ny);
158 }
159 return grid;
160}

164Result<Grid2D> generate(std::string_view operation, const GenerateSettings& settings) {
165 if (settings.width <= 0 || settings.height <= 0)
166 return Result<Grid2D>::failure(Diagnostic::error(
167 DiagnosticCode::InvalidArgument, "generator dimensions must be positive", {}, {}, "procgen.gridGraph"));
168 Grid2D grid = makeGrid(settings);
169 std::mt19937_64 rng(settings.seed);
170 if (operation == "generate.fill") {
171 grid.fill(settings.semantic);
172 } else if (operation == "generate.random_noise") {
173 std::uniform_real_distribution<float> probability(0.f, 1.f);
174 for (int y = 0; y < settings.height; ++y)
175 for (int x = 0; x < settings.width; ++x)
176 if (probability(rng) < settings.x) grid.setCell(x, y, settings.semantic);
177 } else if (operation == "generate.checkerboard" || operation == "generate.dot_grid") {
178 const int spacing = std::max(1, settings.a);
179 for (int y = 0; y < settings.height; ++y)
180 for (int x = 0; x < settings.width; ++x)
181 if (operation == "generate.checkerboard" ? ((x / spacing + y / spacing) % 2 == 0)
182 : (x % spacing == 0 && y % spacing == 0))
183 grid.setCell(x, y, settings.semantic);
184 } else if (operation == "generate.shape") {
185 const int cx = settings.a;
186 const int cy = settings.b;
187 const int radius = std::max(1, settings.c);
188 for (int y = 0; y < settings.height; ++y)
189 for (int x = 0; x < settings.width; ++x) {
190 bool inside = settings.x < 0.5f ? ((x - cx) * (x - cx) + (y - cy) * (y - cy) <= radius * radius)
191 : (std::abs(x - cx) <= radius && std::abs(y - cy) <= radius);
192 if (inside) grid.setCell(x, y, settings.semantic);
193 }
194 } else if (operation == "generate.cellular") {
195 grid = cellular(settings);
196 } else if (operation == "generate.random_walk") {
197 grid = randomWalk(settings);
198 } else if (operation == "generate.maze") {
199 grid = maze(settings);
200 } else if (operation == "generate.poisson") {
201 const float radius = std::max(1.f, settings.x);
202 std::vector<std::pair<int, int>> accepted;
203 std::vector<std::pair<int, int>> candidates;
204 for (int y = 0; y < settings.height; ++y)
205 for (int x = 0; x < settings.width; ++x) candidates.emplace_back(x, y);
206 std::shuffle(candidates.begin(), candidates.end(), rng);
207 for (const auto& candidate : candidates) {
208 const bool clear = std::none_of(accepted.begin(), accepted.end(), [&](const auto& point) {
209 const float dx = float(point.first - candidate.first);
210 const float dy = float(point.second - candidate.second);
211 return dx * dx + dy * dy < radius * radius;
212 });
213 if (clear) {
214 accepted.push_back(candidate);
215 grid.setCell(candidate.first, candidate.second, settings.semantic);
216 }
217 }
218 } else {
219 return Result<Grid2D>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
220 "unsupported generator: " + std::string(operation), {}, {},
221 "procgen.gridGraph"));
222 }
223 return Result<Grid2D>::success(std::move(grid));
224}

226Result<Grid2D> select(const Grid2D& input, std::string_view operation, int mode, int count, float weight,
227 std::uint64_t seed, std::string_view rule) {
228 Grid2D out;
229 out.resize(input.getWidth(), input.getHeight());
230 auto active = cells(input);
231 if (operation == "select.random") {
232 std::mt19937_64 rng(seed);
233 std::shuffle(active.begin(), active.end(), rng);
234 const int amount = count > 0 ? std::min(count, int(active.size()))
235 : int(std::round(std::clamp(weight, 0.f, 1.f) * float(active.size())));
236 active.resize(std::size_t(amount));
237 for (const auto& point : active)
238 out.setCell(point.first, point.second, input.getCell(point.first, point.second));
239 } else if (operation == "select.border" || operation == "select.fill") {
240 for (const auto& point : active) {
241 bool border = false;
242 constexpr int dx[] = {-1, 1, 0, 0};
243 constexpr int dy[] = {0, 0, -1, 1};
244 for (int d = 0; d < 4; ++d) border = border || !occupied(input, point.first + dx[d], point.second + dy[d]);
245 if ((operation == "select.border" && border) || (operation == "select.fill" && !border))
246 out.setCell(point.first, point.second, input.getCell(point.first, point.second));
247 }
248 } else if (operation == "select.neighbors") {
249 for (const auto& point : active) {
250 int neighbors = 0;
251 for (int oy = -1; oy <= 1; ++oy)
252 for (int ox = -1; ox <= 1; ++ox)
253 if ((ox != 0 || oy != 0) && occupied(input, point.first + ox, point.second + oy)) ++neighbors;
254 if ((mode == 0 && neighbors == count) || (mode == 1 && neighbors >= count) ||
255 (mode == 2 && neighbors <= count))
256 out.setCell(point.first, point.second, input.getCell(point.first, point.second));
257 }
258 } else if (operation == "select.rule") {
259 if (rule.size() != 9)
260 return Result<Grid2D>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
261 "rule must contain exactly nine characters (0/1/*)", {},
262 {}, "procgen.gridGraph"));
263 for (const auto& point : active) {
264 bool match = true;
265 for (int oy = -1; oy <= 1 && match; ++oy)
266 for (int ox = -1; ox <= 1; ++ox) {
267 const char expected = rule[std::size_t((oy + 1) * 3 + ox + 1)];
268 if (expected != '*' &&
269 (occupied(input, point.first + ox, point.second + oy) != (expected == '1'))) {
270 match = false;
271 break;
272 }
273 }
274 if (match) out.setCell(point.first, point.second, input.getCell(point.first, point.second));
275 }
276 } else if (operation == "select.islands" || operation == "select.island_centers") {
277 for (const auto& island : islands(input)) {
278 const bool keep = mode == 0 ? int(island.size()) < count
279 : (mode == 1 ? int(island.size()) > count : int(island.size()) == count);
280 if (!keep) continue;
281 if (operation == "select.island_centers") {
282 long long sx = 0;
283 long long sy = 0;
284 for (const auto& point : island) {
285 sx += point.first;
286 sy += point.second;
287 }
288 const int x = int(std::llround(double(sx) / double(island.size())));
289 const int y = int(std::llround(double(sy) / double(island.size())));
290 out.setCell(x, y, input.getCell(island.front().first, island.front().second));
291 } else {
292 for (const auto& point : island)
293 out.setCell(point.first, point.second, input.getCell(point.first, point.second));
294 }
295 }
296 } else {
297 return Result<Grid2D>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
298 "unsupported selector: " + std::string(operation), {}, {},
299 "procgen.gridGraph"));
300 }
301 return Result<Grid2D>::success(std::move(out));
302}

304Result<Grid2D> findPath(const Grid2D& navigation, const Grid2D& starts, const Grid2D& targets, int semantic) {
305 if (navigation.getWidth() != starts.getWidth() || navigation.getHeight() != starts.getHeight() ||
306 navigation.getWidth() != targets.getWidth() || navigation.getHeight() != targets.getHeight())
307 return Result<Grid2D>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
308 "pathfinding grids must have equal dimensions", {}, {},
309 "procgen.gridGraph"));
310 const auto startCells = cells(starts);
311 const auto targetCells = cells(targets);
312 if (startCells.empty() || targetCells.empty())
313 return Result<Grid2D>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
314 "pathfinding requires start and target cells", {}, {},
315 "procgen.gridGraph"));
316 const int width = navigation.getWidth();
317 const int height = navigation.getHeight();
318 std::vector<int> previous(std::size_t(width * height), -1);
319 std::queue<int> pending;
320 const int start = startCells.front().second * width + startCells.front().first;
321 pending.push(start);
322 previous[std::size_t(start)] = start;
323 int destination = -1;
324 while (!pending.empty() && destination < 0) {
325 const int current = pending.front();
326 pending.pop();
327 const int x = current % width;
328 const int y = current / width;
329 if (occupied(targets, x, y)) {
330 destination = current;
331 break;
332 }
333 constexpr int dx[] = {-1, 1, 0, 0};
334 constexpr int dy[] = {0, 0, -1, 1};
335 for (int d = 0; d < 4; ++d) {
336 const int nx = x + dx[d];
337 const int ny = y + dy[d];
338 if (nx < 0 || ny < 0 || nx >= width || ny >= height || !occupied(navigation, nx, ny)) continue;
339 const int next = ny * width + nx;
340 if (previous[std::size_t(next)] < 0) {
341 previous[std::size_t(next)] = current;
342 pending.push(next);
343 }
344 }
345 }
346 if (destination < 0)
347 return Result<Grid2D>::failure(
348 Diagnostic::error(DiagnosticCode::InvalidArgument, "no path found", {}, {}, "procgen.gridGraph"));
349 Grid2D out;
350 out.resize(width, height);
351 for (int current = destination;; current = previous[std::size_t(current)]) {
352 out.setCell(current % width, current / width, semantic);
353 if (current == start) break;
354 }
355 return Result<Grid2D>::success(std::move(out));
356}
