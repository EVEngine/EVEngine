94PointGraphNodeMetric makeMetric(const std::string& id, const PointSet& points, float milliseconds, bool cacheHit,
95 const std::string& backend = "cpu") {
96 PointGraphNodeMetric metric;
97 metric.id = id;
98 metric.outputCount = points.getCount();
99 metric.milliseconds = milliseconds;
100 metric.cacheHit = cacheHit;
101 metric.backend = backend;
102 if (points.empty()) return metric;
103 const auto& first = points.points().front();
104 metric.minX = metric.maxX = first.x;
105 metric.minY = metric.maxY = first.y;
106 metric.minZ = metric.maxZ = first.z;
107 double densitySum = 0.0;
108 for (const auto& point : points.points()) {
109 metric.minX = std::min(metric.minX, point.x);
110 metric.minY = std::min(metric.minY, point.y);
111 metric.minZ = std::min(metric.minZ, point.z);
112 metric.maxX = std::max(metric.maxX, point.x);
113 metric.maxY = std::max(metric.maxY, point.y);
114 metric.maxZ = std::max(metric.maxZ, point.z);
115 densitySum += point.density;
116 }
117 metric.averageDensity = float(densitySum / double(points.getCount()));
118 return metric;
119}

121uint64_t spatialSampleUpperBound(const SpatialData& spatial, float spacing) {
122 if (!spatial.hasBounds() || spacing <= 0.f) return 0;
123 const auto axisCount = [spacing](float minimum, float maximum) {
124 if (maximum <= minimum) return uint64_t(1);
125 return uint64_t(std::floor(double(maximum - minimum) / double(spacing) + 0.001)) + 1;
126 };
127 const uint64_t x = axisCount(spatial.getMinX(), spatial.getMaxX());
128 const uint64_t y = spatial.getKind() == "surface.heightfield"
129 ? uint64_t(1)
130 : axisCount(spatial.getMinY(), spatial.getMaxY());
131 const uint64_t z = axisCount(spatial.getMinZ(), spatial.getMaxZ());
132 const uint64_t maximum = std::numeric_limits<uint64_t>::max();
133 if (x > maximum / y) return maximum;
134 const uint64_t xy = x * y;
135 return xy > maximum / z ? maximum : xy * z;
136}

154 nodes_.erase(id);

157 const auto binding = parameters_.find(*it);

159 floatOverrides_.erase(*it);

169 for (auto& input : node.inputs)

286 const std::string& key) {

288 if (name.empty() || key.empty() || node == nodes_.end() || parameters_.count(name) != 0)

815const PointSet* PointGraph::evaluateTransformSegment(const std::string& id,
816 std::unordered_map<std::string, int>& states) {
817 std::vector<std::string> chain;
818 std::string cursor = id;
819 while (chain.size() < 4) {
820 const auto found = nodes_.find(cursor);
821 if (found == nodes_.end() || found->second.operation != "transform" || found->second.cacheValid ||
822 found->second.inputs[0].empty())
823 break;
824 chain.push_back(cursor);
825 cursor = found->second.inputs[0];
826 }
827 if (chain.size() < 2) return nullptr;
828 std::reverse(chain.begin(), chain.end());
829
830 const PointSet* input = evaluate(cursor, states);
831 if (!input) return nullptr;
832 const bool gpuEligible =
833 !input->empty() &&
834 (computePolicy_ == "gpu" || (computePolicy_ == "auto" && input->getCount() >= computeMinimumPoints_));
835 if (!gpuEligible) return nullptr;
836 if (executionNodeBudget_ > 0 && evaluatedNodes_ + int(chain.size()) > executionNodeBudget_) {
837 error_ = "execution node budget exceeded at node: " + id;
838 lastCancelled_ = true;
839 return nullptr;
840 }
841
842 std::vector<PointCompute::Transform> transforms;
843 transforms.reserve(chain.size());
844 for (const std::string& nodeId : chain) {
845 const Node& transformNode = nodes_.at(nodeId);
846 transforms.push_back({floatValue(transformNode, "x", 0.f), floatValue(transformNode, "y", 0.f),
847 floatValue(transformNode, "z", 0.f), floatValue(transformNode, "yaw", 0.f),
848 floatValue(transformNode, "scaleX", 1.f), floatValue(transformNode, "scaleY", 1.f),
849 floatValue(transformNode, "scaleZ", 1.f)});
850 }
851
852 PointSet output;
853 const uint64_t started = nowNanoseconds();
854 if (!pointCompute_->transformChain(*input, output, transforms)) {
855 computeFallbackReason_ = pointCompute_->getError();
856 return nullptr;
857 }
858
859 evaluatedNodes_ += int(chain.size());
860 for (const std::string& nodeId : chain) {
861 Node& transformNode = nodes_.at(nodeId);
862 transformNode.deferredTransformValid = true;
863 states[nodeId] = 2;
864 }
865 Node& finalNode = nodes_.at(id);
866 finalNode.cache = std::move(output);
867 finalNode.cacheValid = true;
868 finalNode.deferredTransformValid = false;
869#ifdef EVENGINE_WEBGPU
870 const std::string backend = "webgpu";
871#else
872 const std::string backend = "vulkan";
873#endif
874 const float elapsed = float(nowNanoseconds() - started) * 0.000001f;
875 finalNode.cacheMetric = makeMetric(id, finalNode.cache, elapsed, false, backend);
876 metrics_.push_back(finalNode.cacheMetric);
877 return &finalNode.cache;
878}

880bool PointGraph::validateNode(const std::string& id, std::unordered_map<std::string, int>& states) {
881 if (states[id] == 2) return true;
882 if (states[id] == 1) {
883 error_ = "cycle at node: " + id;
884 return false;
885 }
886 const auto found = nodes_.find(id);
887 if (found == nodes_.end() || !operationKnown(found->second.operation)) {
888 error_ = "invalid node: " + id;
889 return false;
890 }
891 states[id] = 1;
892 for (const auto& input : found->second.inputs) {
893 if (!input.empty() && !validateNode(input, states)) return false;
894 }
895 const auto& node = found->second;
896 if (node.operation == "input" && !node.hasPoints) error_ = "input node has no points: " + id;
897 else if ((node.operation == "spatial.sample" || node.operation == "spatial.filter" ||
898 node.operation == "spatial.project" || node.operation == "biome.generate") &&
899 !node.spatial)
900 error_ = "node has no spatial data: " + id;
901 else if (node.operation == "biome.generate" && !node.biomeRules)
902 error_ = "node has no biome rules: " + id;
903 else if (node.operation == "grammar.generate" && !node.shapeGrammar)
904 error_ = "node has no shape grammar: " + id;
905 else if (node.operation == "subgraph" && (!node.subgraph || node.inputs[0].empty()))
906 error_ = "subgraph is not configured: " + id;
907 else if (node.operation == "subgraph") {
908 PointGraph nested = *node.subgraph;
909 PointSet externalInput;
910 if (!nested.setNodePoints(node.subgraphInput, &externalInput) || !nested.validate())
911 error_ = "invalid subgraph at " + id + ": " + nested.getError();
912 } else {
913 const auto* spec = operationSpec(node.operation);
914 for (int index = 0; spec && index < spec->inputs; ++index)
915 if (node.inputs[index].empty()) {
916 error_ = node.operation + " requires input " + std::to_string(index) + ": " + id;
917 break;
918 }
919 }
920 if (!error_.empty()) return false;
921 states[id] = 2;
922 return true;
923}
