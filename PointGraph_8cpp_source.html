103bool PointGraph::removeNode(const std::string& id) {
104 if (nodes_.find(id) == nodes_.end()) return false;
105 invalidateFrom(id);
106 nodes_.erase(id);
107 nodeOrder_.erase(std::remove(nodeOrder_.begin(), nodeOrder_.end(), id), nodeOrder_.end());
108 for (auto it = parameterOrder_.begin(); it != parameterOrder_.end();) {
109 const auto binding = parameters_.find(*it);
110 if (binding != parameters_.end() && binding->second.nodeId == id) {
111 floatOverrides_.erase(*it);
112 intOverrides_.erase(*it);
113 stringOverrides_.erase(*it);
114 parameters_.erase(binding);
115 it = parameterOrder_.erase(it);
116 } else {
117 ++it;
118 }
119 }
120 for (auto& [otherId, node] : nodes_)
121 for (auto& input : node.inputs)
122 if (input == id) input.clear();
123 invalidateTopology(id);
124 return true;
125}

240bool PointGraph::exposeParameter(const std::string& name, const std::string& nodeId,
241 const std::string& key) {
242 const auto node = nodes_.find(nodeId);
243 if (name.empty() || key.empty() || node == nodes_.end() || parameters_.count(name) != 0)
244 return false;
245 const OperationSpec* spec = operationSpec(node->second.operation);
246 if (!spec) return false;
247 const auto* parameter = operationParam(spec, key);
248 if (!parameter) return false;
249 for (const auto& [existingName, binding] : parameters_)
250 if (binding.nodeId == nodeId && binding.key == key) return false;
251 parameters_.emplace(name, ParameterBinding{nodeId, key, parameter->kind});
252 parameterOrder_.push_back(name);
253 ++revision_;
254 return true;
255}

341Result<PointSet> PointGraph::executeResult(std::string_view outputId) {
342 const std::string output(outputId);
343 metrics_.clear();
344 cacheHitCount_ = 0;
345 evaluatedNodes_ = 0;
346 lastCancelled_ = false;
347 computeFallbackReason_.clear();
348 ++executionCount_;
349 if (cancelRequested_) {
350 lastCancelled_ = true;
351 return Result<PointSet>::failure(
352 Diagnostic::error(DiagnosticCode::Cancelled, "execution cancelled", output, {}, "procgen.pointGraph"));
353 }
354 auto compiled = compileExecutionPlan(output);
355 if (!compiled.ok()) return Result<PointSet>::failure(compiled.status());
356 std::unordered_map<std::string, int> states;
357 auto result = evaluate(output, states);
358 if (!result.ok()) {
359 lastCancelled_ = result.status().code() == StatusCode::Cancelled;
360 return Result<PointSet>::failure(result.status());
361 }
362 return Result<PointSet>::success(result.value().get());
363}

472PointSet* PointGraph::materializeNodeOutput(const std::string& id) const {
473 const auto found = nodes_.find(id);
474 if (found == nodes_.end()) return nullptr;
475 const Node& node = found->second;
476 if (node.cacheValid) return new PointSet(node.cache);
477 if (!node.deferredTransformValid || node.operation != "transform" || node.inputs[0].empty()) return nullptr;
478 std::unique_ptr<PointSet> input(materializeNodeOutput(node.inputs[0]));
479 if (!input) return nullptr;
480 return new PointSet(transformPointSet(*input, floatValue(node, "x", 0.f), floatValue(node, "y", 0.f),
481 floatValue(node, "z", 0.f), floatValue(node, "yaw", 0.f),
482 floatValue(node, "scaleX", 1.f), floatValue(node, "scaleY", 1.f),
483 floatValue(node, "scaleZ", 1.f)));
484}

533void PointGraph::invalidateFrom(const std::string& id) {
534 ++revision_;
535 std::vector<std::string> dirty{id};

537 for (size_t cursor = 0; cursor < dirty.size(); ++cursor) {
538 const std::string current = dirty[cursor];
539 if (visited[current]) continue;
540 visited[current] = true;
541 const auto found = nodes_.find(current);
542 if (found != nodes_.end()) {
543 found->second.cacheValid = false;
544 found->second.deferredTransformValid = false;
545 }
546 for (const auto& [candidateId, candidate] : nodes_) {
547 if (candidate.inputs[0] == current || candidate.inputs[1] == current)
548 dirty.push_back(candidateId);
549 }
550 }
551 metrics_.clear();
