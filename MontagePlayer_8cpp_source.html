21TransformTRS filtered(TransformTRS delta, MontageRootMotionMask mask) {
22 if (!mask.translationX) delta.px = 0.0f;
23 if (!mask.translationY) delta.py = 0.0f;
24 if (!mask.translationZ) delta.pz = 0.0f;
25 if (!mask.rotation) {
26 delta.qx = delta.qy = delta.qz = 0.0f;
27 delta.qw = 1.0f;
28 }
29 return delta;

32void multiplyRotation(TransformTRS& total, float x, float y, float z, float w) {
33 const float tx = total.qx;
34 const float ty = total.qy;
35 const float tz = total.qz;
36 const float tw = total.qw;
37 total.qx = tw * x + tx * w + ty * z - tz * y;
38 total.qy = tw * y - tx * z + ty * w + tz * x;
39 total.qz = tw * z + tx * y - ty * x + tz * w;
40 total.qw = tw * w - tx * x - ty * y - tz * z;
41}

64Result<void> MontagePlayer::prepare(action::ActionTimeline timeline, std::vector<MontageClipAsset>&& clips) {
65 auto valid = timeline.validate();
66 if (!valid) return Result<void>::failure(valid.status());
67
68 std::set<std::string> uris;
69 for (std::size_t index = 0; index < clips.size(); ++index) {
70 if (clips[index].uri.empty() || !clips[index].clip)
71 return invalid<void>("montage clip asset is incomplete", "clips[" + std::to_string(index) + "]");
72 if (!std::isfinite(clips[index].clip->getDuration()) || clips[index].clip->getDuration() <= 0.0f)
73 return invalid<void>("montage clip duration must be positive", "clips[" + std::to_string(index) + "]");
74 if (!uris.insert(clips[index].uri).second)
75 return invalid<void>("montage clip URI is duplicated", "clips[" + std::to_string(index) + "].uri");
76 }
77 for (std::size_t index = 0; index < timeline.animationSections.size(); ++index) {
78 const auto& section = timeline.animationSections[index];
79 if (!uris.contains(section.animationUri))
80 return invalid<void>("animation section clip was not supplied",
81 "animationSections[" + std::to_string(index) + "].animationUri");
82 AnimClip* clip = nullptr;
83 const auto asset = std::find_if(clips.begin(), clips.end(),
84 [&](const auto& candidate) { return candidate.uri == section.animationUri; });
85 if (asset != clips.end()) clip = asset->clip.get();
86 const double sourceStart = section.sourceStart.seconds();
87 const double sourceEnd = section.sourceEnd.isZero() ? clip->getDuration() : section.sourceEnd.seconds();
88 if (sourceStart >= clip->getDuration() || sourceEnd > clip->getDuration() || sourceEnd <= sourceStart)
89 return invalid<void>("animation section trim is outside the supplied clip",
90 "animationSections[" + std::to_string(index) + "].sourceStartNs");
91 }
92
93 timeline_ = std::move(timeline);
94 clips_ = std::move(clips);
95 activeSection_ = nullptr;
96 player_->stop();
97 time_ = Duration::zero();
98 lastTick_ = SimulationTick::zero();
99 hasLastTick_ = false;
100 started_ = false;
101 playing_ = false;
102 blendingOut_ = false;
103 weight_ = 1.0;
104 executionId_ = {};
105 rootMotionMask_.translationX = timeline_->montage.rootMotionHorizontal;
106 rootMotionMask_.translationZ = timeline_->montage.rootMotionHorizontal;
107 rootMotionMask_.translationY = timeline_->montage.rootMotionVertical;
108 rootMotionMask_.rotation = timeline_->montage.rootMotionRotation;
109 return Result<void>::success(Status::success(StatusCode::Applied));
110}

236void MontagePlayer::activate(const action::ActionAnimationSection& section, Duration localTime) {
237 AnimClip* clip = clipFor(section.animationUri);
238 const float authoredSeconds = static_cast<float>(
239 Duration::fromNanoseconds(section.end.nanoseconds() - section.start.nanoseconds()).seconds());
240 const float sourceStart = static_cast<float>(section.sourceStart.seconds());
241 const float sourceEnd =
242 section.sourceEnd.isZero() ? clip->getDuration() : static_cast<float>(section.sourceEnd.seconds());
243 const float speed = (sourceEnd - sourceStart) / authoredSeconds;
244 player_->setBlendCurve(section.blendCurve == action::ActionBlendCurve::Linear ? AnimBlendCurve::Linear
245 : AnimBlendCurve::EaseInOut);
246 if (!player_->isPlaying())
247 player_->play(clip);
248 else
249 player_->crossFade(clip, static_cast<float>(section.blendIn.seconds()));
250 player_->setLoop(false);
251 player_->setSpeed(speed);
252 player_->setTime(sourceStart + static_cast<float>(localTime.seconds()) * speed);
253 activeSection_ = &section;
254}

264Result<MontageAdvance> MontagePlayer::present(const action::ActionAdvance& advance, SimulationTick tick) {
265 if (!timeline_) return invalid<MontageAdvance>("montage has not been prepared", "montage");
266 if (!playing_) return invalid<MontageAdvance>("montage is not playing", "montage");
267 if (advance.id != executionId_)
268 return invalid<MontageAdvance>("action advance belongs to another execution", "advance.id");
269 if (advance.totalElapsed < time_ || advance.totalElapsed > timeline_->duration)
270 return invalid<MontageAdvance>("action advance time is outside the presentation cursor",
271 "advance.totalElapsed");
272 if (hasLastTick_ && tick <= lastTick_)
273 return Result<MontageAdvance>::failure(
274 Diagnostic::error(DiagnosticCode::Conflict, "montage tick must advance monotonically", "tick"));
275
276 MontageAdvance result;
277 result.previous = time_;
278 result.current = advance.totalElapsed;
279
280 if (!started_ && result.current == time_) {
281 if (const auto* section = sectionAt(time_)) activate(*section, Duration::zero());
282 }
283
284 Duration cursor = time_;
285 while (cursor < result.current) {
286 const action::ActionAnimationSection* section = sectionAt(cursor);
287 if (!section) {
288 auto next = std::find_if(timeline_->animationSections.begin(), timeline_->animationSections.end(),
289 [&](const auto& candidate) { return candidate.start > cursor; });
290 cursor =
291 next == timeline_->animationSections.end() ? result.current : std::min(next->start, result.current);
292 activeSection_ = nullptr;
293 if (gapPolicy_ == MontageGapPolicy::BindPose) player_->stop();
294 continue;
295 }
296 if (activeSection_ != section)
297 activate(*section, Duration::fromNanoseconds(cursor.nanoseconds() - section->start.nanoseconds()));
298 const Duration sliceEnd = std::min(section->end, result.current);
299 const Duration slice = Duration::fromNanoseconds(sliceEnd.nanoseconds() - cursor.nanoseconds());
300 player_->update(static_cast<float>(slice.seconds()));
301 accumulateRootMotion(result.rootMotion);
302 cursor = sliceEnd;
303 if (cursor == section->end) activeSection_ = nullptr;
304 }
305
306 result.events = advance.timelineEvents;
307 result.activeBlocks = advance.activeBlocks;
308 result.sectionId = activeSection_ ? std::optional<LogicalId>(activeSection_->id) : std::nullopt;
309 result.completed = advance.phase == action::ActionPhase::Completed;
310 const double currentSeconds = result.current.seconds();
311 if (!timeline_->montage.defaultBlendIn.isZero()) {
312 const double t = std::clamp(currentSeconds / timeline_->montage.defaultBlendIn.seconds(), 0.0, 1.0);
313 weight_ = t * t * (3.0 - 2.0 * t);
314 } else {
315 weight_ = 1.0;
316 }
317 if (!timeline_->montage.looping && !timeline_->montage.defaultBlendOut.isZero()) {
318 const double start = timeline_->duration.seconds() - timeline_->montage.defaultBlendOut.seconds() +
319 timeline_->montage.blendOutOffset.seconds();
320 if (currentSeconds >= start) {
321 const double t =
322 std::clamp((currentSeconds - start) / timeline_->montage.defaultBlendOut.seconds(), 0.0, 1.0);
323 weight_ = std::min(weight_, 1.0 - t * t * (3.0 - 2.0 * t));
324 }
325 }
326 result.weight = weight_;
327 result.rootMotion = filtered(result.rootMotion, rootMotionMask_);
328 if (receiver_) receiver_->applyMontageRootMotion(result.rootMotion);
329
330 time_ = result.current;
331 lastTick_ = tick;
332 hasLastTick_ = true;
333 started_ = true;
334 if (result.completed) playing_ = false;
335 return Result<MontageAdvance>::success(
336 std::move(result), Status::success(result.completed ? StatusCode::Applied : StatusCode::Pending));
337}

339std::vector<action::ActionTimelineEvent> MontagePlayer::realignStateEvents(Duration target) const {
340 std::vector<action::ActionTimelineEvent> events;
341 for (const auto& track : timeline_->tracks) {
342 if (track.muted) continue;
343 for (const auto& state : track.states) {
344 const bool wasActive = time_ >= state.start && time_ < state.end;
345 const bool isActive = target >= state.start && target < state.end;
346 if (wasActive == isActive) continue;
347 events.push_back(
348 {isActive ? action::ActionTimelineEventKind::StateEnter : action::ActionTimelineEventKind::StateExit,
349 track.id, state.id, state.type, target, state.payload});
350 }
351 }
352 return events;
353}

360 SimulationTick tick) {
361 if (!timeline_) return invalid<MontageAdvance>("montage has not been prepared", "montage");

503 return Result<MontageAdvance>::failure(
504 Diagnostic::error(DiagnosticCode::Conflict, "montage tick must advance monotonically", "tick"));
505 auto elapsed = blendOutElapsed_.tryAdd(delta);
506 if (!elapsed) return Result<MontageAdvance>::failure(elapsed.status());
507 blendOutElapsed_ = std::min(std::move(elapsed).takeValue(), blendOutDuration_);
508 const double t = std::clamp(blendOutElapsed_.seconds() / blendOutDuration_.seconds(), 0.0, 1.0);
509 weight_ = blendOutStartWeight_ * (1.0 - t * t * (3.0 - 2.0 * t));
510 MontageAdvance result;
511 result.previous = result.current = time_;
512 result.weight = weight_;
513 result.completed = t >= 1.0;
514 lastTick_ = tick;
515 if (result.completed) {
516 player_->stop();
517 activeSection_ = nullptr;
518 blendingOut_ = false;
519 playing_ = false;
520 }
521 return Result<MontageAdvance>::success(
522 std::move(result), Status::success(result.completed ? StatusCode::Applied : StatusCode::Pending));
523}
