46RootMotionLockAxes locksFromMask(MontageRootMotionMask mask) noexcept {
47 RootMotionLockAxes locks = RootMotionLockAxes::None;
48 if (!mask.translationX) locks = locks | RootMotionLockAxes::X;
49 if (!mask.translationY) locks = locks | RootMotionLockAxes::Y;
50 if (!mask.translationZ) locks = locks | RootMotionLockAxes::Z;

54MontageRootMotionMask maskFromPolicy(const RootMotionPolicy& policy) noexcept {
55 MontageRootMotionMask mask;
56 mask.translationX = !hasLock(policy.lockAxes, RootMotionLockAxes::X);
57 mask.translationY = !hasLock(policy.lockAxes, RootMotionLockAxes::Y);
58 mask.translationZ = !hasLock(policy.lockAxes, RootMotionLockAxes::Z);
59 mask.rotation = !policy.lockRotation;
60 return mask;
61}

63void multiplyRotation(TransformTRS& total, float x, float y, float z, float w) {
64 const float tx = total.qx;
65 const float ty = total.qy;
66 const float tz = total.qz;
67 const float tw = total.qw;
68 total.qx = tw * x + tx * w + ty * z - tz * y;
69 total.qy = tw * y - tx * z + ty * w + tz * x;
70 total.qz = tw * z + tx * y - ty * x + tz * w;
71 total.qw = tw * w - tx * x - ty * y - tz * z;
72}

74Result<std::unique_ptr<PreparedMontage>> buildPreparedMontage(const action::ActionTimeline& timeline,
75 const std::vector<MontageClipAsset>& clips) {
76 auto prepared = std::make_unique<PreparedMontage>();
77 prepared->clipIndices.reserve(clips.size());
78 for (std::size_t index = 0; index < clips.size(); ++index)
79 prepared->clipIndices.emplace(clips[index].uri, index);
80
81 prepared->sections.reserve(timeline.animationSections.size());
82 for (std::size_t index = 0; index < timeline.animationSections.size(); ++index) {
83 const auto& section = timeline.animationSections[index];
84 const auto clip = prepared->clipIndices.find(section.animationUri);
85 if (clip == prepared->clipIndices.end())
86 return invalid<std::unique_ptr<PreparedMontage>>("animation section clip was not supplied",
87 "animationSections[" + std::to_string(index) +
88 "].animationUri");
89 prepared->sections.push_back(
90 {section.start, section.end, clip->second,
91 index > 0 && timeline.animationSections[index - 1].end == section.start,
92 index + 1 < timeline.animationSections.size() &&
93 timeline.animationSections[index + 1].start == section.end});
94 }
95
96 prepared->physicalSections.reserve(timeline.splitTimestamps.size() + 1U);
97 Duration start = Duration::zero();
98 for (const Duration split : timeline.splitTimestamps) {
99 prepared->physicalSections.emplace_back(start, split);
100 start = split;
101 }
102 prepared->physicalSections.emplace_back(start, timeline.duration);
103 return Result<std::unique_ptr<PreparedMontage>>::success(std::move(prepared));
104}

129Result<void> MontagePlayer::prepare(action::ActionTimeline timeline, std::vector<MontageClipAsset>&& clips) {
130 auto valid = timeline.validate();
131 if (!valid) return Result<void>::failure(valid.status());
132
133 std::set<std::string> uris;
134 for (std::size_t index = 0; index < clips.size(); ++index) {
135 if (clips[index].uri.empty() || !clips[index].clip)
136 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
137 "montage clip asset is incomplete",
138 "clips[" + std::to_string(index) + "]"));
139 if (!std::isfinite(clips[index].clip->getDuration()) || clips[index].clip->getDuration() <= 0.0f)
140 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
141 "montage clip duration must be positive",
142 "clips[" + std::to_string(index) + "]"));
143 if (!uris.insert(clips[index].uri).second)
144 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
145 "montage clip URI is duplicated",
146 "clips[" + std::to_string(index) + "].uri"));
147 }
148 for (std::size_t index = 0; index < timeline.animationSections.size(); ++index) {
149 const auto& section = timeline.animationSections[index];
150 if (!uris.contains(section.animationUri))
151 return Result<void>::failure(
152 Diagnostic::error(DiagnosticCode::InvalidArgument, "animation section clip was not supplied",
153 "animationSections[" + std::to_string(index) + "].animationUri"));
154 AnimClip* clip = nullptr;
155 const auto asset = std::find_if(clips.begin(), clips.end(),
156 [&](const auto& candidate) { return candidate.uri == section.animationUri; });
157 if (asset != clips.end()) clip = asset->clip.get();
158 const double sourceStart = section.sourceStart.seconds();
159 const double sourceEnd = section.sourceEnd.isZero() ? clip->getDuration() : section.sourceEnd.seconds();
160 if (sourceStart >= clip->getDuration() || sourceEnd > clip->getDuration() || sourceEnd <= sourceStart)
161 return Result<void>::failure(Diagnostic::error(
162 DiagnosticCode::InvalidArgument, "animation section trim is outside the supplied clip",
163 "animationSections[" + std::to_string(index) + "].sourceStartNs"));
164 }
165
166 auto prepared = buildPreparedMontage(timeline, clips);
167 if (!prepared) return Result<void>::failure(prepared.status());
168
169 timeline_ = std::move(timeline);
170 clips_ = std::move(clips);
171 prepared_ = std::move(prepared).takeValue();
172 activeSection_ = nullptr;
173 player_->stop();
174 time_ = Duration::zero();
175 lastTick_ = SimulationTick::zero();
176 hasLastTick_ = false;
177 started_ = false;
178 playing_ = false;
179 blendingOut_ = false;
180 weight_ = 1.0;
181 executionId_ = {};
182 applyTimelineRootMotionChannels();
183 return Result<void>::success(Status::success(StatusCode::Applied));
184}

320std::optional<std::size_t> MontagePlayer::sectionIndexAt(Duration time) const noexcept {
321 if (!prepared_ || prepared_->sections.empty()) return std::nullopt;
322 const auto found = std::upper_bound(
323 prepared_->sections.begin(), prepared_->sections.end(), time,
324 [](Duration value, const PreparedMontage::Section& section) { return value < section.start; });
325 if (found == prepared_->sections.begin()) return std::nullopt;
326 const auto candidate = std::prev(found);
327 if (time < candidate->start || time >= candidate->end) return std::nullopt;
328 return static_cast<std::size_t>(std::distance(prepared_->sections.begin(), candidate));
329}

344double MontagePlayer::sectionEnvelopeAt(Duration time) const noexcept {
345 if (!timeline_ || !prepared_ || prepared_->sections.empty()) return 0.0;
346 auto index = sectionIndexAt(time);
347 if (!index && time == prepared_->sections.back().end) index = prepared_->sections.size() - 1U;
348 if (!index) return 0.0;
349 const auto& section = timeline_->animationSections[*index];
350 const auto& topology = prepared_->sections[*index];
351 double envelope = 1.0;
352 if (!topology.contiguousPrevious && !section.blendIn.isZero()) {
353 const auto elapsed = Duration::fromNanoseconds(time.nanoseconds() - section.start.nanoseconds());
354 envelope =
355 action::evaluateActionBlendCurve(section.blendInCurve, elapsed.seconds() / section.blendIn.seconds());
356 }
357 if (!topology.contiguousNext && !section.blendOut.isZero()) {
358 const auto remaining = Duration::fromNanoseconds(section.end.nanoseconds() - time.nanoseconds());
359 envelope *=
360 action::evaluateActionBlendCurve(section.blendOutCurve, remaining.seconds() / section.blendOut.seconds());
361 }
362 return envelope;
363}

365void MontagePlayer::activate(const action::ActionAnimationSection& section, Duration localTime) {
366 const auto sectionIndex = static_cast<std::size_t>(&section - timeline_->animationSections.data());
367 AnimClip* clip = clips_[prepared_->sections[sectionIndex].clipIndex].clip.get();
368 const float authoredSeconds = static_cast<float>(
369 Duration::fromNanoseconds(section.end.nanoseconds() - section.start.nanoseconds()).seconds());
370 const float sourceStart = static_cast<float>(section.sourceStart.seconds());
371 const float sourceEnd =
372 section.sourceEnd.isZero() ? clip->getDuration() : static_cast<float>(section.sourceEnd.seconds());
373 const float speed = (sourceEnd - sourceStart) / authoredSeconds;
374 const auto blendCurve = [](action::ActionBlendCurve curve) {
375 switch (curve) {
376 case action::ActionBlendCurve::Linear: return AnimBlendCurve::Linear;
377 case action::ActionBlendCurve::EaseIn: return AnimBlendCurve::EaseIn;
378 case action::ActionBlendCurve::EaseOut: return AnimBlendCurve::EaseOut;
379 case action::ActionBlendCurve::EaseInOut: return AnimBlendCurve::EaseInOut;
380 case action::ActionBlendCurve::SmoothStep: return AnimBlendCurve::SmoothStep;
381 }
382 return AnimBlendCurve::Linear;
383 };
384 player_->setBlendCurve(blendCurve(section.blendInCurve));
385 if (!player_->isPlaying())
386 player_->play(clip);
387 else
388 player_->crossFade(clip, static_cast<float>(section.blendIn.seconds()));
389 player_->setLoop(false);
390 player_->setSpeed(speed);
391 player_->setTime(sourceStart + static_cast<float>(localTime.seconds()) * speed);
392 activeSection_ = &section;
393}

454Result<MontageAdvance>
455MontagePlayer::presentImpl(const action::ActionAdvance& advance, std::vector<action::ActionTimelineEvent> events,
456 std::vector<MontageActiveBlock> activeBlocks, SimulationTick tick) {
457 if (!timeline_) return invalid<MontageAdvance>("montage has not been prepared", "montage");
458 if (!playing_) return invalid<MontageAdvance>("montage is not playing", "montage");
459 if (advance.id != executionId_)
460 return Result<MontageAdvance>::failure(Diagnostic::error(
461 DiagnosticCode::InvalidArgument, "action advance belongs to another execution", "advance.id"));
462 if (advance.totalElapsed < time_ || advance.totalElapsed > timeline_->duration)
463 return Result<MontageAdvance>::failure(
464 Diagnostic::error(DiagnosticCode::InvalidArgument, "action advance time is outside the presentation cursor",
465 "advance.totalElapsed"));
466 if (hasLastTick_ && tick <= lastTick_)
467 return Result<MontageAdvance>::failure(
468 Diagnostic::error(DiagnosticCode::Conflict, "montage tick must advance monotonically", "tick"));
469
470 MontageAdvance result;
471 result.previous = time_;
472 result.current = advance.totalElapsed;
473
474 if (!started_ && result.current == time_) {
475 if (const auto* section = sectionAt(time_)) activate(*section, Duration::zero());
476 }
477
478 Duration cursor = time_;
479 while (cursor < result.current) {
480 const action::ActionAnimationSection* section = sectionAt(cursor);
481 if (!section) {
482 const auto next = nextSectionStartAfter(cursor);
483 cursor = next ? std::min(*next, result.current) : result.current;
484 activeSection_ = nullptr;
485 if (gapPolicy_ == MontageGapPolicy::BindPose) player_->stop();
486 continue;
487 }
488 if (activeSection_ != section)
489 activate(*section, Duration::fromNanoseconds(cursor.nanoseconds() - section->start.nanoseconds()));
490 const Duration sliceEnd = std::min(section->end, result.current);
491 const Duration slice = Duration::fromNanoseconds(sliceEnd.nanoseconds() - cursor.nanoseconds());
492 player_->update(static_cast<float>(slice.seconds()));
493 accumulateRootMotion(result.rootMotion);
494 cursor = sliceEnd;
495 if (cursor == section->end) activeSection_ = nullptr;
496 }
497
498 result.events = std::move(events);
499 result.activeBlocks = std::move(activeBlocks);
500 result.sectionId = activeSection_ ? std::optional<LogicalId>(activeSection_->id) : std::nullopt;
501 result.completed = advance.phase == action::ActionPhase::Completed;
502 const double currentSeconds = result.current.seconds();
503 if (!timeline_->montage.defaultBlendIn.isZero()) {
504 const double t = std::clamp(currentSeconds / timeline_->montage.defaultBlendIn.seconds(), 0.0, 1.0);
505 weight_ = t * t * (3.0 - 2.0 * t);
506 } else {
507 weight_ = 1.0;
508 }
509 if (!timeline_->montage.looping && !timeline_->montage.defaultBlendOut.isZero()) {
510 const double start = timeline_->duration.seconds() - timeline_->montage.defaultBlendOut.seconds() +
511 timeline_->montage.blendOutOffset.seconds();
512 if (currentSeconds >= start) {
513 const double t =
514 std::clamp((currentSeconds - start) / timeline_->montage.defaultBlendOut.seconds(), 0.0, 1.0);
515 weight_ = std::min(weight_, 1.0 - t * t * (3.0 - 2.0 * t));
516 }
517 }
518 weight_ *= sectionEnvelopeAt(result.current);
519 result.weight = weight_;
520 // Root motion is already policy-filtered (locks / facing / rotation) by the owned AnimPlayer.
521 if (receiver_) receiver_->applyMontageRootMotion(result.rootMotion);
522
523 time_ = result.current;
524 lastTick_ = tick;
525 hasLastTick_ = true;
526 started_ = true;
527 if (result.completed) playing_ = false;
528 return Result<MontageAdvance>::success(

552Result<MontageAdvance> MontagePlayer::jumpToTime(action::ActionExecutionId executionId, Duration target,
553 SimulationTick tick) {
554 if (!timeline_)
555 return Result<MontageAdvance>::failure(
556 Diagnostic::error(DiagnosticCode::InvalidArgument, "montage has not been prepared", "montage"));
557 if (!playing_)
558 return Result<MontageAdvance>::failure(
559 Diagnostic::error(DiagnosticCode::InvalidArgument, "montage is not playing", "montage"));
560 if (executionId != executionId_)
561 return Result<MontageAdvance>::failure(
562 Diagnostic::error(DiagnosticCode::InvalidArgument, "jump belongs to another execution", "executionId"));
563 if (target < Duration::zero() || target > timeline_->duration)
564 return Result<MontageAdvance>::failure(
565 Diagnostic::error(DiagnosticCode::InvalidArgument, "montage jump is outside the timeline", "target"));
566 if (hasLastTick_ && tick <= lastTick_)
567 return Result<MontageAdvance>::failure(
568 Diagnostic::error(DiagnosticCode::Conflict, "montage tick must advance monotonically", "tick"));
569
570 MontageAdvance result;
571 result.previous = time_;
572 result.current = target;
573 result.events = realignStateEvents(target);
574 auto active = activeBlocksAt(target);
575 if (!active) return Result<MontageAdvance>::failure(active.status());
576 result.activeBlocks = std::move(active).takeValue();
577 player_->stop();
578 activeSection_ = nullptr;
579 if (const auto* section = sectionAt(target)) {
580 activate(*section, Duration::fromNanoseconds(target.nanoseconds() - section->start.nanoseconds()));
581 result.sectionId = section->id;
582 }
583 time_ = target;
584 lastTick_ = tick;
585 hasLastTick_ = true;
586 started_ = true;
587 return Result<MontageAdvance>::success(std::move(result), Status::success(StatusCode::Applied));
588}

593 if (!prepared_ || sectionIndex >= prepared_->physicalSections.size())

598Result<MontageAdvance> MontagePlayer::evaluateSectionProgress(action::ActionExecutionId executionId,
599 std::size_t sectionIndex, double progress,
600 SimulationTick tick) {
601 if (!std::isfinite(progress) || progress < 0.0 || progress > 1.0)
602 return invalid<MontageAdvance>("physical section progress must be in [0, 1]", "progress");
603 if (!timeline_) return invalid<MontageAdvance>("montage has not been prepared", "montage");
604 if (!prepared_ || sectionIndex >= prepared_->physicalSections.size())
605 return invalid<MontageAdvance>("physical section index is out of range", "sectionIndex");

608 const Duration target = Duration::fromNanoseconds(range.first.nanoseconds() +
609 static_cast<std::int64_t>(std::llround(span * progress)));
610 return jumpToTime(executionId, target, tick);
611}

664Result<MontageAdvance> MontagePlayer::beginBlendOut(Duration duration, SimulationTick tick) {
665 if (duration < Duration::zero())
666 return Result<MontageAdvance>::failure(Diagnostic::error(
667 DiagnosticCode::InvalidArgument, "montage blend-out duration must be non-negative", "duration"));
668 if (!timeline_)
669 return Result<MontageAdvance>::failure(
670 Diagnostic::error(DiagnosticCode::InvalidArgument, "montage has not been prepared", "montage"));
671 if (duration.isZero() && playing_) return interrupt(tick);
672 if (!playing_ && weight_ <= 0.0)
673 return Result<MontageAdvance>::success(MontageAdvance{}, Status::success(StatusCode::NoOp));
674 if (hasLastTick_ && tick <= lastTick_)
675 return Result<MontageAdvance>::failure(
676 Diagnostic::error(DiagnosticCode::Conflict, "montage tick must advance monotonically", "tick"));
677 MontageAdvance result;
678 result.previous = result.current = time_;
679 result.weight = weight_;
680 if (duration.isZero()) {
681 player_->stop();
682 activeSection_ = nullptr;
683 playing_ = false;
684 blendingOut_ = false;
685 weight_ = 0.0;
686 result.weight = 0.0;
687 result.completed = true;
688 lastTick_ = tick;
689 hasLastTick_ = true;
690 return Result<MontageAdvance>::success(std::move(result), Status::success(StatusCode::Applied));
691 }
692 for (const auto& track : timeline_->tracks) {
693 if (track.muted) continue;
694 for (const auto& state : track.states)
695 if (time_ >= state.start && time_ < state.end)
696 result.events.push_back(
697 {action::ActionTimelineEventKind::StateExit, track.id, state.id, state.type, time_, state.payload});
698 }
699 blendingOut_ = true;
700 blendOutDuration_ = duration;
701 blendOutElapsed_ = Duration::zero();
702 blendOutStartWeight_ = weight_;
703 lastTick_ = tick;
704 hasLastTick_ = true;
705 return Result<MontageAdvance>::success(std::move(result), Status::success(StatusCode::Pending));
706}

708Result<MontageAdvance> MontagePlayer::advanceBlendOut(Duration delta, SimulationTick tick) {
709 if (!blendingOut_)
710 return Result<MontageAdvance>::failure(
711 Diagnostic::error(DiagnosticCode::InvalidArgument, "montage is not blending out", "montage"));
712 if (delta < Duration::zero())
713 return Result<MontageAdvance>::failure(Diagnostic::error(
714 DiagnosticCode::InvalidArgument, "montage blend-out delta must be non-negative", "delta"));
715 if (tick <= lastTick_)
716 return Result<MontageAdvance>::failure(
717 Diagnostic::error(DiagnosticCode::Conflict, "montage tick must advance monotonically", "tick"));
718 auto elapsed = blendOutElapsed_.tryAdd(delta);
719 if (!elapsed) return Result<MontageAdvance>::failure(elapsed.status());
720 blendOutElapsed_ = std::min(std::move(elapsed).takeValue(), blendOutDuration_);
721 const double t = std::clamp(blendOutElapsed_.seconds() / blendOutDuration_.seconds(), 0.0, 1.0);
722 weight_ = blendOutStartWeight_ * (1.0 - t * t * (3.0 - 2.0 * t));
723 MontageAdvance result;
724 result.previous = result.current = time_;
725 result.weight = weight_;
726 result.completed = t >= 1.0;
727 lastTick_ = tick;
728 if (result.completed) {
729 player_->stop();
730 activeSection_ = nullptr;
731 blendingOut_ = false;
732 playing_ = false;
733 }
734 return Result<MontageAdvance>::success(
735 std::move(result), Status::success(result.completed ? StatusCode::Applied : StatusCode::Pending));
736}
