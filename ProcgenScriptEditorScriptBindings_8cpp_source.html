32ssq::Table loadModuleFromScript(const editor::ScriptBind& bind, ScriptProcgenScriptEditor* self, const std::string& uri,
33 const std::string& id, const std::string& displayName, const std::string& kind,
34 const ssq::Object& schema) {
35 if (!self) return bind.fail(DiagnosticCode::InvalidArgument, "procgen editor must not be null");
36 auto converted = script::valueFromSquirrel(schema);
37 if (!converted.ok()) return script::projectStatusResult(bind.vm(), converted.status());
38 return editor::project(
39 bind.vm(), self->editor().loadModule(uri, id, displayName, kind, editor::toEditorValue(converted.value())));
40}
