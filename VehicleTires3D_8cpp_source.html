54 // Only this synchronous native callback sees the borrowed wrappers. No user callbacks or query-cache mutation.
55 static float collect(b3ShapeId id, b3Pos point, b3Vec3 normal, float fraction, uint64_t, int triangle, int,
56 void* opaque) {
57 auto& context = *static_cast<RayContext*>(opaque);
58 if (b3Shape_IsSensor(id)) return -1;
59 auto* shape = static_cast<physics::Shape3D*>(b3Shape_GetUserData(id));
60 auto* ground = static_cast<physics::Body3D*>(b3Body_GetUserData(b3Shape_GetBody(id)));
61 if (!shape || !ground || ground == &context.chassis || !ground->isValid() || !shape->isValid()) return -1;
62 const auto n = vec(normal);
63 if (dot(n, context.up) <= 1e-4) return -1;
64 auto& hit = context.contact;
65 if (!hit.ground || fraction < hit.fraction || (fraction == hit.fraction && shape->getId() < hit.shapeId)) {
66 auto material = b3Shape_GetSurfaceMaterial(id);
67 if (triangle >= 0 && triangle < shape->getTriangleMeshTriangleCount()) {
68 const int slot = shape->getTriangleMeshMaterialIndex(triangle);
69 if (slot >= 0 && slot < b3Shape_GetMeshMaterialCount(id))
70 material = b3Shape_GetMeshSurfaceMaterial(id, slot);
71 }
72 // Conveyor velocity is local to the shape, then projected onto the hit plane.
73 const b3Quat local{shape->getLocalRotX(), shape->getLocalRotY(), shape->getLocalRotZ(),
74 shape->getLocalRotW()};
75 material.tangentVelocity = b3Body_GetWorldVector(eve::physics::box3d::Access::bodyId(*ground),
76 b3RotateVector(local, material.tangentVelocity));
77 hit = {ground, vec(point), n, material, fraction, shape->getId()};
78 }
79 return hit.fraction;
80 }

82double response(physics::Body3D& body, V point, V a, V b) {
83 if (b3Body_GetType(eve::physics::box3d::Access::bodyId(body)) != b3_dynamicBody) return 0;
84 const auto r = add(point, scale(vec(b3Body_GetWorldCenterOfMass(eve::physics::box3d::Access::bodyId(body))), -1));
85 const auto inertia = b3Body_GetWorldInverseRotationalInertia(eve::physics::box3d::Access::bodyId(body));
86 return b3Body_GetInverseMass(eve::physics::box3d::Access::bodyId(body)) * dot(a, b) +
87 dot(cross(r, a), multiply(inertia, cross(r, b)));
88}

90struct PredictedBody {
91 physics::Body3D* body;
92 V center, freeLinear, freeAngular, linear, angular;
93 b3Matrix3 inverseInertia;
94 double inverseMass;
95 V velocity(V point) const { return add(linear, cross(angular, add(point, scale(center, -1)))); }
96 void impulse(V point, V impulse) {
97 linear = add(linear, scale(impulse, inverseMass));
98 angular = add(angular, multiply(inverseInertia, cross(add(point, scale(center, -1)), impulse)));
99 }

111Result<VehiclePhysicsStatus> submit3D(VehicleEntity& vehicle, physics::Body3D& body,
112 std::shared_ptr<VehicleWheelRuntime>& runtime, const SimulationStep& step) {
113 using namespace physics;
114 auto pending = submissionState(runtime, step);
115 if (!pending || pending.value() != VehiclePhysicsStatus::Unavailable) return pending;
116 const auto* definition = vehicle.definition()->def;
117 auto* world = body.getWorld();
118 if (!definition || !world || !body.isValid()) return reject("Vehicle definition or spatial body is unavailable");
119 auto phase = ForceBatch::prepare(*world, {});
120 if (!phase) return Result<VehiclePhysicsStatus>::failure(phase.status());
121 if (world->rigidBackendKind() != physics::RigidBackend3D::Box3D)
122 return Result<VehiclePhysicsStatus>::failure(
123 Diagnostic::error(DiagnosticCode::Unsupported,
124 "Ray tire contact integration currently requires the Box3D backend", "vehicle.tires3d"));
125 const auto& config = definition->suspension;
126 const auto input = *vehicle.input();
127 const double dt = step.delta.seconds(), mass = body.getMass();
128 if (!positive(dt) || dt > .05 || !positive(mass) || !nonnegative(definition->maxSpeed) ||
129 !nonnegative(definition->accel) || !nonnegative(config.maxTravel) || !nonnegative(config.driveForce) ||
130 !nonnegative(config.brakeTorque) || !nonnegative(config.maxSteerAngle) || config.maxSteerAngle > 1.4 ||
131 !std::isfinite(input.throttle) || !std::isfinite(input.steer) || !std::isfinite(input.brake))
132 return reject("Invalid spatial tire input, suspension travel or fixed duration");
133 auto wheels = config.wheels;
134 if (wheels.empty()) {
135 if (!positive(definition->radius))
136 return reject("Default spatial wheel layout requires a positive chassis size");
137 for (int x : {-1, 1})
138 for (int z : {-1, 1}) {
139 SuspensionWheel wheel;
140 wheel.x = float(x) * definition->radius * .45f;
141 wheel.z = float(z) * definition->radius * .6f;
142 wheel.y = -.175f;
143 wheel.steer = z > 0;
144 wheels.push_back(wheel);
145 }
146 }
147 std::vector<VehicleEntity::SuspensionState::WheelState> candidate(wheels.size());
148 if (vehicle.suspension()->wheels.size() == wheels.size()) candidate = vehicle.suspension()->wheels;
149 const double count = static_cast<double>(wheels.size());
150 const auto driven =
151 static_cast<std::size_t>(std::count_if(wheels.begin(), wheels.end(), [](const auto& w) { return w.drive; }));
152 const auto up = vec(b3Body_GetWorldVector(eve::physics::box3d::Access::bodyId(body), {0, 1, 0}));
153 if (!finite(up)) return reject("Invalid suspension orientation");
154 auto chassisLink = PhysicsLink::fromBody(body);
155 if (!chassisLink) return Result<VehiclePhysicsStatus>::failure(chassisLink.status());
156 std::vector<ForceLoad> loads;
157 loads.reserve(wheels.size() * 2);
158 std::vector<PredictedBody> predicted;
159 auto predict = [&](physics::Body3D& source) {
160 for (std::size_t i = 0; i < predicted.size(); ++i)
161 if (predicted[i].body == &source) return i;
162 const auto raw = physics::box3d::Access::bodyId(source);
163 const bool dynamic = b3Body_GetType(raw) == b3_dynamicBody;
164 const V center = vec(b3Body_GetWorldCenterOfMass(raw));
165 V linear = vec(b3Body_GetLinearVelocity(raw)), angular = vec(b3Body_GetAngularVelocity(raw));
166 if (dynamic)
167 linear = add(linear, scale(V{world->getGravityX(), world->getGravityY(), world->getGravityZ()},
168 dt * source.getGravityScale()));
169 predicted.push_back({&source, center, linear, angular, linear, angular,
170 dynamic ? b3Body_GetWorldInverseRotationalInertia(raw) : b3Matrix3{},
171 dynamic ? b3Body_GetInverseMass(raw) : 0});
172 return predicted.size() - 1;
173 };
174 predict(body);
175 std::vector<PreparedWheel> contacts;
176 contacts.reserve(wheels.size());
177 for (std::size_t i = 0; i < wheels.size(); ++i) {
178 const auto& wheel = wheels[i];
179 const V localMount{wheel.x, double(wheel.y) + wheel.restLength, wheel.z};
180 const double rayLength = double(wheel.restLength) + wheel.radius + config.maxTravel;
181 if (!fits(localMount) || !positive(wheel.radius) || !nonnegative(wheel.restLength) ||
182 !nonnegative(wheel.stiffness) || !nonnegative(wheel.damping) || !positive(rayLength))
183 return reject("Invalid spatial wheel geometry or spring parameters");
184 const auto mount = b3Body_GetWorldPoint(eve::physics::box3d::Access::bodyId(body), native(localMount));
185 const auto translation = scale(up, -rayLength);
186 if (!fits(vec(mount)) || !fits(translation) || !fits(add(vec(mount), translation)))
187 return reject("Suspension ray exceeds supported coordinate range");
188 RayContext ray{body, up, {}};
189 b3World_CastRay(eve::physics::box3d::Access::worldId(*world), mount, native(translation),
190 b3DefaultQueryFilter(), &RayContext::collect, &ray);
191 const auto& contact = ray.contact;
192 const double steer = wheel.steer ? std::clamp(double(input.steer), -1.0, 1.0) * config.maxSteerAngle : 0;
193 auto forward = vec(b3Body_GetWorldVector(eve::physics::box3d::Access::bodyId(body),
194 {float(std::sin(steer)), 0, float(std::cos(steer))}));
195 V normal = contact.ground ? contact.normal : up;
196 forward = add(forward, scale(normal, -dot(forward, normal)));
197 const double norm = length(forward);
198 if (!positive(norm) || norm < 1e-6) return reject("Wheel heading is perpendicular to the contact plane");
199 forward = scale(forward, 1 / norm);
200 const auto lateral = cross(normal, forward);
201 const V point = contact.ground ? contact.point : vec(mount);
202 auto relative = vec(b3Body_GetWorldPointVelocity(eve::physics::box3d::Access::bodyId(body),
203 {float(point.x), float(point.y), float(point.z)}));
204 if (contact.ground) {
205 relative = add(relative,
206 scale(vec(b3Body_GetWorldPointVelocity(eve::physics::box3d::Access::bodyId(*contact.ground),
207 {float(point.x), float(point.y), float(point.z)})),
208 -1));
209 const auto conveyor = vec(contact.material.tangentVelocity);
210 relative = add(relative, scale(add(conveyor, scale(normal, -dot(conveyor, normal))), -1));
211 }
212 std::array<V, 3> axes{forward, lateral, normal};
213 double matrix[3][3]{};
214 for (int a = 0; a < 3; ++a)
215 for (int b = 0; b < 3; ++b) {
216 matrix[a][b] = response(body, point, axes[a], axes[b]);
217 if (contact.ground) matrix[a][b] += response(*contact.ground, point, axes[a], axes[b]);
218 if (!std::isfinite(matrix[a][b])) return reject("Invalid spatial contact inertia");
219 }
220 double diagonal[3]{};
221 for (int a = 0; a < 3; ++a) {
222 diagonal[a] = matrix[a][a];
223 for (int b = 0; b < 3; ++b)
224 if (b != a) diagonal[a] += std::abs(matrix[a][b]);
225 diagonal[a] *= count;
226 if (!nonnegative(diagonal[a])) return reject("Invalid contact response bound");
227 }
228 PreparedWheel prepared;
229 prepared.contact = contact;
230 prepared.point = point;
231 prepared.forward = forward;
232 prepared.lateral = lateral;
233 prepared.normal = normal;
234 prepared.diagonal = {diagonal[0], diagonal[1], diagonal[2]};
235 if (contact.ground) {
236 prepared.ground = predict(*contact.ground);
237 const auto conveyor = vec(contact.material.tangentVelocity);
238 prepared.conveyor = add(conveyor, scale(normal, -dot(conveyor, normal)));
239 prepared.compression = std::clamp(double(wheel.restLength) - (contact.fraction * rayLength - wheel.radius),
240 0.0, double(config.maxTravel));
241 const double alpha = dot(normal, up);
242 prepared.spring = wheel.stiffness * prepared.compression / alpha;
243 prepared.gain = (wheel.damping + wheel.stiffness * dt) / (alpha * alpha);
244 }
245 prepared.input.angularVelocity = candidate[i].angularVelocity;
246 prepared.input.longitudinalVelocity = dot(relative, forward);
247 prepared.input.lateralVelocity = dot(relative, lateral);
248 prepared.input.longitudinalInverseMass = diagonal[0];
249 prepared.input.lateralInverseMass = diagonal[1];
250 driveInputs(prepared.input, config, input, wheel, driven, definition->maxSpeed,
251 std::min(double(config.driveForce), mass * definition->accel));
252 prepared.model = model(config, wheel.radius);
253 if (contact.ground)
254 prepared.model.friction = std::min(prepared.model.friction, double(contact.material.friction));
255 contacts.push_back(prepared);
256 }
257 // Bounded Jacobi iterations use the diagonal majorant as a preconditioner.
258 // Every sample starts with the original wheel speed: drive/brake torque is integrated once.
259 for (int iteration = 0; iteration < 64; ++iteration) {
260 for (std::size_t i = 0; i < contacts.size(); ++i) {
261 auto& c = contacts[i];
262 V relative = predicted[0].velocity(c.point);
263 if (c.contact.ground) relative = add(relative, scale(predicted[c.ground].velocity(c.point), -1));
264 relative = add(relative, scale(c.conveyor, -1));
265 const double vn = dot(relative, c.normal) - c.diagonal[2] * c.impulse[2];
266 const double load =
267 c.contact.ground ? std::max(0.0, (c.spring - c.gain * vn) / (1 + dt * c.gain * c.diagonal[2])) : 0;
268 if (!nonnegative(load)) return reject("Suspension force is not representable");
269 auto ti = c.input;
270 ti.longitudinalVelocity = dot(relative, c.forward) - c.diagonal[0] * c.impulse[0];
271 ti.lateralVelocity = dot(relative, c.lateral) - c.diagonal[1] * c.impulse[1];
272 ti.normalLoad = load;
273 auto sampled = ForceModels::tire(c.model, ti, dt);
274 if (!sampled) return Result<VehiclePhysicsStatus>::failure(sampled.status());
275 const auto& value = sampled.value();
276 c.impulse = {value.longitudinalForce * dt, value.lateralForce * dt, load * dt};
277 auto& state = candidate[i];
278 state.angularVelocity = value.angularVelocity;
279 state.longitudinalForce = value.longitudinalForce;
280 state.lateralForce = value.lateralForce;
281 state.slipRatio = value.slipRatio;
282 state.slipAngle = value.slipAngle;
283 state.normalLoad = load;
284 state.prevCompression = float(c.compression);
285 state.grounded = load > 0;
286 }
287 for (auto& value : predicted) {
288 value.linear = value.freeLinear;
289 value.angular = value.freeAngular;
290 }
291 for (const auto& c : contacts) {
292 const auto impulse =
293 add(scale(c.normal, c.impulse[2]), add(scale(c.forward, c.impulse[0]), scale(c.lateral, c.impulse[1])));
294 predicted[0].impulse(c.point, impulse);
295 if (c.contact.ground) predicted[c.ground].impulse(c.point, scale(impulse, -1));
296 }
297 }
298 for (const auto& c : contacts) {
299 const auto force = scale(
300 add(scale(c.normal, c.impulse[2]), add(scale(c.forward, c.impulse[0]), scale(c.lateral, c.impulse[1]))),
301 1 / dt);
302 loads.push_back({chassisLink.value(), force, c.point, {}});
303 if (c.contact.ground && predicted[c.ground].inverseMass > 0) {
304 auto link = PhysicsLink::fromBody(*c.contact.ground);
305 if (!link) return Result<VehiclePhysicsStatus>::failure(link.status());
306 loads.push_back({link.value(), scale(force, -1), c.point, {}});
307 }
308 }
309 auto batch = ForceBatch::prepare(*world, loads);
310 if (!batch) return Result<VehiclePhysicsStatus>::failure(batch.status());
311 return commit(*world, batch.value(), runtime, std::move(candidate), step);
312}
