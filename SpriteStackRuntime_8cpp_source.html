26SpriteStackLayerArtifact artifact(const image::ImageData& image, int index) {
27 std::size_t covered = 0;
28 for (int y = 0; y < image.getHeight(); ++y)
29 for (int x = 0; x < image.getWidth(); ++x)
30 if (image.getPixel(x, y).a > 0.f) ++covered;
31 const std::size_t pixels = static_cast<std::size_t>(image.getWidth()) * image.getHeight();
32 return {index, image.getWidth(), image.getHeight(), checksum(image),
33 pixels ? static_cast<double>(covered) / pixels : 0.0};
34}

40EditorResult<std::vector<SpriteStackLayerArtifact>> SpriteStackBakeRuntime::bake(
41 const SpriteStackDocumentTarget& document, const ISpriteStackModelResolver* resolver) {
42 const auto diagnostics = document.validate();
43 for (const auto& d : diagnostics)
44 if (d.severity() == DiagnosticSeverity::Error)
45 return EditorResult<std::vector<SpriteStackLayerArtifact>>::failure(
46 eve::Status(EditorStatus::Rejected, diagnostics));
47 std::vector<image::ImageData*> raw;
48 try {
49 if (document.value().sourceKind == "primitive")
50 raw = spritestack::slicePrimitiveToLayers(document.value().source, document.value().bake);
51 else {
52 if (!resolver)
53 return fail<std::vector<SpriteStackLayerArtifact>>(EditorStatus::Rejected,
54 "editor.spritestack.resolver",
55 "Model SpriteStack bake requires an asset resolver");
56 auto resolved = resolver->resolveModel(document.value().source);
57 if (!resolved.ok() || !resolved.value())
58 return EditorResult<std::vector<SpriteStackLayerArtifact>>::failure(resolved.status());
59 raw = spritestack::sliceModelToLayers(resolved.value(), document.value().bake);
60 }
61 } catch (const std::exception& exception) {
62 for (auto* layer : raw) delete layer;
63 return fail<std::vector<SpriteStackLayerArtifact>>(EditorStatus::Failed, "editor.spritestack.bake",
64 exception.what());
65 }
66 if (raw.size() != static_cast<std::size_t>(document.value().bake.layerCount) ||
67 std::any_of(raw.begin(), raw.end(), [](auto* p) { return p == nullptr; })) {
68 for (auto* layer : raw) delete layer;
69 return fail<std::vector<SpriteStackLayerArtifact>>(EditorStatus::Failed, "editor.spritestack.layer-count",
70 "SpriteStack baker returned an incomplete generation");
71 }
72 std::vector<std::unique_ptr<image::ImageData>> candidate;
73 std::vector<SpriteStackLayerArtifact> artifacts;
74 candidate.reserve(raw.size());
75 artifacts.reserve(raw.size());
76 for (std::size_t i = 0; i < raw.size(); ++i) {
77 candidate.emplace_back(raw[i]);
78 artifacts.push_back(artifact(*raw[i], static_cast<int>(i)));
79 }
80 layers_ = std::move(candidate);
81 stack_.reset();
82 value_ = document.value();
83 revision_ = document.revision();
84 return eve::editing::applied<std::vector<SpriteStackLayerArtifact>>(std::move(artifacts), diagnostics);
85}
