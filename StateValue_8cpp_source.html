13bool parseSegment(const std::string& seg, std::string& key, std::vector<size_t>& indexes) {
14 key.clear();
15 indexes.clear();
16 if (seg.empty()) return false;
17
18 size_t i = 0;
19 if (seg[0] != '[') {
20 while (i < seg.size() && seg[i] != '[') key.push_back(seg[i++]);
21 }
22 if (key.empty() && (i >= seg.size())) return false;
23
24 while (i < seg.size()) {
25 if (seg[i] != '[') return false;
26 const size_t close = seg.find(']', i);
27 if (close == std::string::npos) return false;
28 const std::string idx = seg.substr(i + 1, close - i - 1);
29 if (idx.empty() || idx.find_first_not_of("0123456789") != std::string::npos) return false;
30 indexes.push_back(static_cast<size_t>(std::stoull(idx)));
31 i = close + 1;
32 }
33 return true;
34}

93const StateValue* StateValue::get(const std::string& dottedPath) const {
94 const StateValue* cur = this;
95 size_t start = 0;
96 while (true) {
97 const size_t dot = dottedPath.find('.', start);
98 const bool last = (dot == std::string::npos);
99 const std::string seg = dottedPath.substr(start, last ? std::string::npos : dot - start);
100
101 std::string key;
102 std::vector<size_t> indexes;
103 if (!parseSegment(seg, key, indexes)) return nullptr;
104 if (!key.empty()) {
105 cur = cur->find(key);
106 if (!cur) return nullptr;
107 }
108 for (const size_t idx : indexes) {
109 if (!cur->isArray() || idx >= cur->arr_.size()) return nullptr;
110 cur = &cur->arr_[idx];
111 }
112 if (last) return cur;
113 start = dot + 1;
114 }
115}

121bool StateValue::setPath(const std::string& dottedPath, StateValue v) {
122 std::vector<std::string> segs;
123 size_t start = 0;
124 while (true) {
125 const size_t dot = dottedPath.find('.', start);
126 if (dot == std::string::npos) {
127 segs.push_back(dottedPath.substr(start));
128 break;
129 }
130 segs.push_back(dottedPath.substr(start, dot - start));
131 start = dot + 1;
132 }
133 if (segs.empty() || (segs.size() == 1 && segs[0].empty())) return false;
134
135 StateValue* cur = this;
136 for (size_t si = 0; si + 1 < segs.size(); ++si) {
137 std::string key;
138 std::vector<size_t> indexes;
139 // Intermediate segments must be plain object keys; array traversal is
140 // only supported on the final segment.
141 if (!parseSegment(segs[si], key, indexes) || key.empty() || !indexes.empty()) return false;
142 if (!cur->isObject()) return false;
143 StateValue* next = cur->find(key);
144 if (!next) {
145 cur->set(key, StateValue::object());
146 next = cur->find(key);
147 }
148 cur = next;
149 }
150
151 std::string key;
152 std::vector<size_t> indexes;
153 if (!parseSegment(segs.back(), key, indexes)) return false;
154 if (!indexes.empty()) {
155 if (!key.empty()) {
156 if (!cur->isObject()) return false;
157 const StateValue* arr = cur->find(key);
158 if (!arr || !arr->isArray()) return false;
159 cur = cur->find(key);
160 }
161 if (!cur->isArray() || indexes.size() != 1 || indexes[0] >= cur->arr_.size()) return false;
162 cur->arr_[indexes[0]] = std::move(v);
163 return true;
164 }
165 if (!cur->isObject() || key.empty()) return false;
166 cur->set(key, std::move(v));
167 return true;
168}

170bool StateValue::mergeDefaults(const StateValue& defaults) {
171 if (!isObject() || !defaults.isObject()) return false;
172 bool changed = false;
173 for (const auto& kv : defaults.obj_) {
174 StateValue* mine = find(kv.first);
175 if (!mine) {
176 set(kv.first, kv.second);
177 changed = true;
178 } else if (mine->isObject() && kv.second.isObject()) {
179 changed = mine->mergeDefaults(kv.second) || changed;
180 }
181 }
182 return changed;
183}
