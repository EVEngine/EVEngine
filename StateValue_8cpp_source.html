13bool parseSegment(const std::string& seg, std::string& key, std::vector<size_t>& indexes) {
14 key.clear();
15 indexes.clear();
16 if (seg.empty()) return false;
17
18 size_t i = 0;
19 if (seg[0] != '[') {
20 while (i < seg.size() && seg[i] != '[') key.push_back(seg[i++]);
21 }
22 if (key.empty() && (i >= seg.size())) return false;
23
24 while (i < seg.size()) {
25 if (seg[i] != '[') return false;
26 const size_t close = seg.find(']', i);
27 if (close == std::string::npos) return false;
28 const std::string idx = seg.substr(i + 1, close - i - 1);
29 if (idx.empty() || idx.find_first_not_of("0123456789") != std::string::npos) return false;
30 indexes.push_back(static_cast<size_t>(std::stoull(idx)));
31 i = close + 1;
32 }
33 return true;
34}

38Value toCanonicalStateValue(const StateValue& value) {
39 switch (value.kind()) {
40 case StateValue::Kind::Null: return Value::null();
41 case StateValue::Kind::Int: return Value::integer(value.asInt());
42 case StateValue::Kind::Float: return Value::number(value.asDouble());
43 case StateValue::Kind::Bool: return Value::boolean(value.asBool());
44 case StateValue::Kind::String: return Value::string(value.asString());
45 case StateValue::Kind::Array: {
46 Value::Array array;
47 array.reserve(value.arraySize());
48 for (size_t i = 0; i < value.arraySize(); ++i) array.push_back(toCanonicalStateValue(value.at(i)));
49 return Value::array(std::move(array));
50 }
51 case StateValue::Kind::Object: {
52 Value::Object object;
53 for (const auto& key : value.keys()) object.emplace(key, toCanonicalStateValue(*value.find(key)));
54 return Value::object(std::move(object));
55 }
56 }
57 return Value::null();
58}

136const StateValue* StateValue::get(const std::string& dottedPath) const {
137 const StateValue* cur = this;
138 size_t start = 0;
139 while (true) {
140 const size_t dot = dottedPath.find('.', start);
141 const bool last = (dot == std::string::npos);
142 const std::string seg = dottedPath.substr(start, last ? std::string::npos : dot - start);
143
144 std::string key;
145 std::vector<size_t> indexes;
146 if (!parseSegment(seg, key, indexes)) return nullptr;
147 if (!key.empty()) {
148 cur = cur->find(key);
149 if (!cur) return nullptr;
150 }
151 for (const size_t idx : indexes) {
152 if (!cur->isArray() || idx >= cur->arr_.size()) return nullptr;
153 cur = &cur->arr_[idx];
154 }
155 if (last) return cur;
156 start = dot + 1;
157 }

164bool StateValue::setPath(const std::string& dottedPath, StateValue v) {
165 std::vector<std::string> segs;
166 size_t start = 0;
167 while (true) {
168 const size_t dot = dottedPath.find('.', start);
169 if (dot == std::string::npos) {
170 segs.push_back(dottedPath.substr(start));
171 break;
172 }
173 segs.push_back(dottedPath.substr(start, dot - start));
174 start = dot + 1;
175 }
176 if (segs.empty() || (segs.size() == 1 && segs[0].empty())) return false;
177
178 StateValue* cur = this;
179 for (size_t si = 0; si + 1 < segs.size(); ++si) {
180 std::string key;
181 std::vector<size_t> indexes;
182 // Intermediate segments must be plain object keys; array traversal is
183 // only supported on the final segment.
184 if (!parseSegment(segs[si], key, indexes) || key.empty() || !indexes.empty()) return false;
185 if (!cur->isObject()) return false;
186 StateValue* next = cur->find(key);
187 if (!next) {
188 cur->set(key, StateValue::object());
189 next = cur->find(key);
190 }
191 cur = next;
192 }
193
194 std::string key;
195 std::vector<size_t> indexes;
196 if (!parseSegment(segs.back(), key, indexes)) return false;
197 if (!indexes.empty()) {
198 if (!key.empty()) {
199 if (!cur->isObject()) return false;
200 const StateValue* arr = cur->find(key);
201 if (!arr || !arr->isArray()) return false;
202 cur = cur->find(key);
203 }
204 if (!cur->isArray() || indexes.size() != 1 || indexes[0] >= cur->arr_.size()) return false;
205 cur->arr_[indexes[0]] = std::move(v);
206 return true;
207 }
208 if (!cur->isObject() || key.empty()) return false;
209 cur->set(key, std::move(v));
210 return true;
211}

213bool StateValue::mergeDefaults(const StateValue& defaults) {
214 if (!isObject() || !defaults.isObject()) return false;
215 bool changed = false;
216 for (const auto& kv : defaults.obj_) {
217 StateValue* mine = find(kv.first);
218 if (!mine) {
219 set(kv.first, kv.second);
220 changed = true;
221 } else if (mine->isObject() && kv.second.isObject()) {
222 changed = mine->mergeDefaults(kv.second) || changed;
223 }
224 }
225 return changed;
226}
