12OffscreenCanvas::OffscreenCanvas(Graphics *owner, int width, int height, bool hdr)

14 ASSERT(owner != nullptr);
15 ASSERT_GT(width, 0);

22 hdr ? vk::Format::eR16G16B16A16Sfloat : vk::Format::eR8G8B8A8Unorm;

30 .minFilter(vk::Filter::eNearest)
31 .addressModeU(vk::SamplerAddressMode::eClampToEdge)
32 .addressModeV(vk::SamplerAddressMode::eClampToEdge)
33 .build(device);
34
35 auto sets = vkb::DescriptorSetBuilder()
36 .layout(owner->getTexSetLayout())
37 .build(device.instance, owner->getDescriptorPool());
38 sampleGpu.width = width;
39 sampleGpu.height = height;
40 sampleGpu.viewOverride = color.imageView();
41
42 vkb::UnboundSet unbound{sets[0]};
43 vkb::DescriptorSetUpdater updater;
44 updater.beginDescriptorSet(unbound)
45 .beginImages(0, 0, vk::DescriptorType::eCombinedImageSampler)
46 .image(vkb::SampledImage::forLaterSample(sampleGpu.sampler, color.imageView()))
47 .beginImages(1, 0, vk::DescriptorType::eCombinedImageSampler)
48 .image(vkb::SampledImage::forLaterSample(sampleGpu.sampler, color.imageView()))
49 .update(device.instance);
50 sampleGpu.descriptorSet = std::move(unbound).publish();

53 sampleTexture.height = height;

55 sampleTexture.pixelHeight = height;
56 sampleTexture.gpuHandle = &sampleGpu;

98void OffscreenCanvas::readAllPixels(std::vector<uint8_t> &outRgba) {
99 if (hdr) throw Exception("HDR Canvas pixel readback requires an explicit tone-map pass");
100 auto &device = owner->getDevice();
101 device->waitIdle();
102
103 const vk::DeviceSize byteSize = vk::DeviceSize(width) * vk::DeviceSize(height) * 4;
104 vkb::GenericBuffer staging(device, vk::BufferUsageFlagBits::eTransferDst, byteSize,
105 vk::MemoryPropertyFlagBits::eHostVisible |
106 vk::MemoryPropertyFlagBits::eHostCoherent);
107
108 vkb::executeImmediately(device.instance, owner->getUploadPool(),
109 device.getQueue(vkb::QueueType::graphics), [&](vk::CommandBuffer cb) {
110 color.setLayout(cb, vk::ImageLayout::eTransferSrcOptimal);
111 vk::BufferImageCopy region{};
112 region.imageSubresource = {vk::ImageAspectFlagBits::eColor, 0, 0, 1};
113 region.imageExtent = vk::Extent3D{uint32_t(width), uint32_t(height), 1};
114 cb.copyImageToBuffer(color.image(), vk::ImageLayout::eTransferSrcOptimal,
115 staging.buffer, region);
116 color.setLayout(cb, vk::ImageLayout::eShaderReadOnlyOptimal);
117 });
118
119 outRgba.resize(size_t(byteSize));
120 void *mapped = staging.map();
121 // Vulkan FB row 0 is NDC -Y (top). Batcher maps logical y=0 → NDC -1, so
122 // logical top-left matches FB top-left — copy as-is into Y-down buffer.
123 std::memcpy(outRgba.data(), mapped, size_t(byteSize));
124 staging.unmap();
125 staging.release();
126}

156Color OffscreenCanvas::getPixel(int x, int y) {
157 ASSERT_GE(x, 0);
158 ASSERT_GE(y, 0);
159 ASSERT_LT(x, width);
160 ASSERT_LT(y, height);
161 if (x < 0 || y < 0 || x >= width || y >= height)
162 throw Exception("OffscreenCanvas::getPixel: out of bounds (%d,%d)", x, y);
163 std::vector<uint8_t> px;
164 readAllPixels(px);
165 size_t i = (size_t(y) * size_t(width) + size_t(x)) * 4;
166 return Color(px[i + 0] / 255.f, px[i + 1] / 255.f, px[i + 2] / 255.f, px[i + 3] / 255.f);
167}
