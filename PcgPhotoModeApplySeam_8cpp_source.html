69Value photoModeValueToScript(const PhotoModeValue& value) {
70 if (const auto* boolean = std::get_if<bool>(&value)) return Value::boolean(*boolean);

126EVENGINE_API_WORLD void declareScriptPhotoModeApplySeam() {
127 ScriptPhotoModeApplyProviders& owned = scriptPhotoModeApplyProviders();
128 ScriptProviderRegistry::instance().declareCapability(
129 IPhotoModeApplySink::capabilityName,
130 [&owned](const ScriptProviderHandle& handle) -> Result<void> {
131 // 单实现槽位：`provide` 会顶掉当前实现，顶掉必须可见（否则原生提供方会"悄悄失效"，
132 // 而症状只是照片模式的字段不再生效）。
133 if (cap::query<IPhotoModeApplySink>() != nullptr) {
134 recordLogEvent("warn", "script photo-mode provider '" + handle.name +
135 "' replaced the provider that owned IPhotoModeApplySink");
136 }
137 auto adapter = std::make_unique<ScriptPhotoModeApplySink>(handle);
138 ScriptPhotoModeApplySink* raw = adapter.get();
139 // 先注册再交所有权：从这一刻起能力槽位持有裸指针，`owned.live` 是让它有效的唯一原因。
140 cap::provide<IPhotoModeApplySink>(raw);
141 owned.live.push_back(std::move(adapter));
142 return Result<void>::success();
143 },
144 [&owned](const ScriptProviderHandle& handle) -> Result<void> {
145 const auto match = std::find_if(owned.live.begin(), owned.live.end(),
146 [&handle](const std::unique_ptr<ScriptPhotoModeApplySink>& adapter) {
147 return adapter->handle() == handle;
148 });
149 if (match == owned.live.end()) {
150 // 幂等：后置条件是"这一代没有适配器"，模块退出时可能已经撤过了。
151 return Result<void>::success();
152 }
153 cap::revoke<IPhotoModeApplySink>(match->get());
154 owned.retired.push_back(std::move(*match));
155 owned.live.erase(match);
156 return Result<void>::success();
157 });
158}

160EVENGINE_API_WORLD void withdrawScriptPhotoModeApplySeam() {
161 ScriptPhotoModeApplyProviders& owned = scriptPhotoModeApplyProviders();
162 for (auto& adapter : owned.live) {
163 cap::revoke<IPhotoModeApplySink>(adapter.get());
164 owned.retired.push_back(std::move(adapter));
165 }
166 owned.live.clear();
167
168 ScriptProviderRegistry::instance()
169 .removeCapability(IPhotoModeApplySink::capabilityName)
170 .ignore("module teardown may already have dropped this seam");
171}
