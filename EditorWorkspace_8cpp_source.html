48EditorResult<WorkspacePanelDescriptor> EditorWorkspace::registerPanel(WorkspacePanelDescriptor descriptor) {
49 if (descriptor.id.empty())
50 return eve::editing::failed<WorkspacePanelDescriptor>(
51 EditorStatus::Rejected, RuleId("editor.workspace.empty-panel-id"), "Workspace panel id must not be empty");
52 if (descriptor.title.empty())
53 return eve::editing::failed<WorkspacePanelDescriptor>(EditorStatus::Rejected,
54 RuleId("editor.workspace.empty-panel-title"),
55 "Workspace panel title must not be empty");
56 if (!parseWorkspaceRegion(descriptor.region))
57 return eve::editing::failed<WorkspacePanelDescriptor>(EditorStatus::Rejected,
58 RuleId("editor.workspace.invalid-region"),
59 "Workspace panel uses an unsupported dock region");
60 if (findPanel(descriptor.id))
61 return eve::editing::failed<WorkspacePanelDescriptor>(
62 EditorStatus::Conflict, RuleId("editor.workspace.duplicate-panel"),
63 "Workspace panel id is already registered: " + descriptor.id);
64 const std::string id = descriptor.id;
65 panels_.push_back(std::move(descriptor));
66 sortPanels();
67 if (activePanel_.empty()) activePanel_ = id;
68 changed();
69 return eve::editing::applied<WorkspacePanelDescriptor>(*findPanel(id));
70}

77EditorResult<WorkspacePanelDescriptor> EditorWorkspace::removePanel(const StableId& id) {
78 const auto found = std::find_if(panels_.begin(), panels_.end(), [&](const auto& panel) { return panel.id == id.value(); });
79 if (found == panels_.end())
80 return eve::editing::failed<WorkspacePanelDescriptor>(EditorStatus::NotFound,
81 RuleId("editor.workspace.panel-not-found"),
82 "Workspace panel is not registered: " + id.value());
83 WorkspacePanelDescriptor removed = *found;
84 panels_.erase(found);
85 if (activePanel_ == id.value()) activePanel_ = panels_.empty() ? std::string{} : panels_.front().id;
86 changed();
87 return eve::editing::applied<WorkspacePanelDescriptor>(std::move(removed));

102EditorResult<WorkspacePanelDescriptor> EditorWorkspace::movePanel(const StableId& id, WorkspaceRegion region,

361void EditorWorkspace::sortPanels() {
362 std::stable_sort(panels_.begin(), panels_.end(), [](const auto& left, const auto& right) {
363 if (left.region != right.region) return left.region < right.region;
364 if (left.order != right.order) return left.order < right.order;
365 return left.id < right.id;
366 });
367}
