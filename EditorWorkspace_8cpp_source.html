53EditorResult<WorkspacePanelDescriptor> EditorWorkspace::registerPanel(WorkspacePanelDescriptor descriptor) {
54 if (descriptor.id.empty())
55 return workspaceError<WorkspacePanelDescriptor>(EditorStatus::Rejected, "editor.workspace.empty-panel-id",
56 "Workspace panel id must not be empty");
57 if (descriptor.title.empty())
58 return workspaceError<WorkspacePanelDescriptor>(EditorStatus::Rejected, "editor.workspace.empty-panel-title",
59 "Workspace panel title must not be empty");
60 if (!parseWorkspaceRegion(descriptor.region))
61 return workspaceError<WorkspacePanelDescriptor>(EditorStatus::Rejected, "editor.workspace.invalid-region",
62 "Workspace panel uses an unsupported dock region");
63 if (findPanel(descriptor.id))
64 return workspaceError<WorkspacePanelDescriptor>(EditorStatus::Conflict, "editor.workspace.duplicate-panel",
65 "Workspace panel id is already registered: " + descriptor.id);
66 const std::string id = descriptor.id;
67 panels_.push_back(std::move(descriptor));
68 sortPanels();
69 if (activePanel_.empty()) activePanel_ = id;
70 changed();
71 return eve::editing::applied<WorkspacePanelDescriptor>(*findPanel(id));
72}

79EditorResult<WorkspacePanelDescriptor> EditorWorkspace::removePanel(const StableId& id) {
80 const auto found = std::find_if(panels_.begin(), panels_.end(), [&](const auto& panel) { return panel.id == id.value(); });
81 if (found == panels_.end())
82 return workspaceError<WorkspacePanelDescriptor>(EditorStatus::NotFound, "editor.workspace.panel-not-found",
83 "Workspace panel is not registered: " + id.value());
84 WorkspacePanelDescriptor removed = *found;
85 panels_.erase(found);
86 if (activePanel_ == id.value()) activePanel_ = panels_.empty() ? std::string{} : panels_.front().id;
87 changed();
88 return eve::editing::applied<WorkspacePanelDescriptor>(std::move(removed));
89}

358void EditorWorkspace::sortPanels() {
359 std::stable_sort(panels_.begin(), panels_.end(), [](const auto& left, const auto& right) {
360 if (left.region != right.region) return left.region < right.region;
361 if (left.order != right.order) return left.order < right.order;
362 return left.id < right.id;
363 });
