41Grid2D makeHexTopology(const Params &params) {
42 const int width = std::clamp(params.getInt("width", 32), 1, 256);
43 const int height = std::clamp(params.getInt("height", 24), 1, 256);
44 Grid2D topology;
45 topology.resize(width, height);
46 const std::uint32_t seed = params.getSeed();
47 for (int y = 0; y < height; ++y) {
48 for (int x = 0; x < width; ++x) {
49 // This is a compact cell-topology projection. Elevation and dense
50 // geometry remain in the mesh recipe; no per-cell ECS objects are
51 // created merely to publish topology.
52 const std::uint32_t value = seed ^ (std::uint32_t(x) * 0x9e3779b9u) ^ (std::uint32_t(y) * 0x85ebca6bu);
53 topology.setCell(x, y, int((value ^ (value >> 16u)) % 10u));
54 }
55 }
56 topology.setMeta("role", "topology");
57 topology.setMeta("algorithm", "mesh.hexterrain");
58 topology.setMeta("coordinateSpace", "hex_axial");
59 return topology;
60}

62PointSet makeHexAnchors(const Params &params) {
63 const float radius = params.getFloat("radius", 1.f);
64 const int width = std::max(2, params.getInt("width", 32));
65 const int height = std::max(2, params.getInt("height", 24));
66 const float xStep = radius * 1.5f;
67 const float zStep = radius * 1.7320508076f;
68 PointSet anchors;
69 anchors.add(0.f, 0.f, 0.f);
70 anchors.add(float(width - 1) * xStep, 0.f, 0.f);
71 anchors.add(0.f, 0.f, float(height - 1) * zStep);
72 anchors.add(float(width - 1) * xStep, 0.f, (float(height - 1) + float((width - 1) & 1) * .5f) * zStep);
73 anchors.setStringAttribute(0, "role", "spawn");
74 anchors.setStringAttribute(1, "role", "edge");
75 anchors.setStringAttribute(2, "role", "edge");
76 anchors.setStringAttribute(3, "role", "edge");
77 return anchors;
78}

91PointSet makeCastleAnchors(const Params &params) {
92 const float width = std::max(16.f, params.getFloat("width", 42.f));
93 const float depth = std::max(16.f, params.getFloat("depth", 36.f));
94 const float gateWidth = std::clamp(params.getFloat("gateWidth", 5.f), 2.f, width * .3f);
95 const float keepWidth = std::clamp(params.getFloat("keepWidth", width * .25f), 6.f, width * .55f);
96 const float keepDepth = std::clamp(params.getFloat("keepDepth", depth * .24f), 6.f, depth * .55f);
97 PointSet anchors;
98 anchors.add(0.f, 0.f, -depth * .5f - 1.f);
99 anchors.add(-width * .5f + gateWidth, 0.f, -depth * .5f);
100 anchors.add(width * .5f - gateWidth, 0.f, -depth * .5f);
101 anchors.add(-keepWidth * .5f, 0.f, -keepDepth * .5f);
102 anchors.add(keepWidth * .5f, 0.f, keepDepth * .5f);
103 anchors.setStringAttribute(0, "role", "gate");
104 anchors.setStringAttribute(1, "role", "outer_gate_left");
105 anchors.setStringAttribute(2, "role", "outer_gate_right");
106 anchors.setStringAttribute(3, "role", "keep_min");
107 anchors.setStringAttribute(4, "role", "keep_max");
108 return anchors;

166eve::Result<GeneratedArtifact> generateComposite(const Params &params, ArtifactId id, std::string_view recipe,
167 bool (*generateMesh)(const Params &, MeshBuild &, std::string &),
168 Grid2D topology, PointSet anchors) {
169 if (id.isNil())
170 return rejectedArtifact(eve::DiagnosticCode::InvalidArgument, "top-level artifact identity must not be nil");
171 const auto key = BuildKey::forRecipe(recipe, params);
172 if (!key)
173 return rejectedArtifact(eve::DiagnosticCode::InvalidArgument, "recipe or parameters cannot form a build key");
174
175 MeshBuild mesh;
176 std::string error;
177 if (!generateMesh(params, mesh, error)) {
178 return rejectedArtifact(eve::DiagnosticCode::Failed,
179 error.empty() ? "procedural mesh generation failed" : error);
180 }
181 const Collider collider = colliderForMesh(mesh);
182 std::vector<ArtifactPart> parts;
183 parts.reserve(4);
184 eve::Status failure;
185 eve::Value::Object meshMeta;
186 meshMeta.emplace("role", eve::Value("mesh"));
187 meshMeta.emplace("densePayload", eve::Value(true));
188 auto meshPart = partFromMesh("mesh", id, *key, std::move(mesh), std::move(meshMeta));
189 if (!appendPart<ArtifactPart>(std::move(meshPart), parts, failure))
190 return eve::Result<GeneratedArtifact>::failure(std::move(failure));
191
192 eve::Value::Object colliderMeta;
193 colliderMeta.emplace("role", eve::Value("collider"));
194 colliderMeta.emplace("source", eve::Value("mesh"));
195 auto colliderPart = partFromCollider("collider", id, *key, collider, std::move(colliderMeta));
196 if (!appendPart<ArtifactPart>(std::move(colliderPart), parts, failure))
197 return eve::Result<GeneratedArtifact>::failure(std::move(failure));
198
199 eve::Value::Object topologyMeta;
200 topologyMeta.emplace("role", eve::Value("topology"));
201 auto topologyPart = partFromGrid("topology", id, *key, std::move(topology), std::move(topologyMeta));
202 if (!appendPart<ArtifactPart>(std::move(topologyPart), parts, failure))
203 return eve::Result<GeneratedArtifact>::failure(std::move(failure));
204
205 eve::Value::Object anchorsMeta;
206 anchorsMeta.emplace("role", eve::Value("anchors"));
207 anchorsMeta.emplace("coordinateSpace", eve::Value("world"));
208 auto anchorsPart = partFromPoints("anchors", id, *key, std::move(anchors), std::move(anchorsMeta));
209 if (!appendPart<ArtifactPart>(std::move(anchorsPart), parts, failure))
210 return eve::Result<GeneratedArtifact>::failure(std::move(failure));
211
212 CompositeArtifact composite;
213 composite.children = std::move(parts);
214 eve::Value::Object metadata;
215 metadata.emplace("algorithm", eve::Value(std::string(recipe)));
216 metadata.emplace("childCount", eve::Value(std::int64_t(composite.children.size())));
217 metadata.emplace("determinism", eve::Value("seeded_cpu"));
218 metadata.emplace("payloadPolicy", eve::Value("dense data stays typed"));
219 const Bounds bounds = composite.children.front().bounds;
220 return makeArtifact(id, ArtifactType::Composite, eve::SchemaVersion(1), *key, bounds, {}, std::move(metadata),
221 GeneratedArtifact::Payload(std::move(composite)));
222}

236eve::Result<GeneratedArtifact> generateMeshArtifact(std::string_view recipeId, const Params &params, ArtifactId id) {
237 if (recipeId == "mesh.hexterrain") return generateHexTerrainArtifact(params, id);
238 if (recipeId == "mesh.castle") return generateCastleArtifact(params, id);
239 if (recipeId == "mesh.roadNetwork") return generateRoadNetworkArtifact(params, id);
240 if (id.isNil())
241 return rejectedArtifact(eve::DiagnosticCode::InvalidArgument, "top-level artifact identity must not be nil");
242 const auto key = BuildKey::forRecipe(recipeId, params);
243 if (!key)
244 return rejectedArtifact(eve::DiagnosticCode::InvalidArgument, "recipe or parameters cannot form a build key");
245 MeshRecipeRegistry::instance().registerBuiltins();
246 MeshBuild mesh;
247 std::string error;
248 if (!MeshRecipeRegistry::instance().generate(std::string(recipeId), params, mesh, error))
249 return rejectedArtifact(eve::DiagnosticCode::Failed,
250 error.empty() ? "procedural mesh generation failed" : error);
251 eve::Value::Object metadata;
252 metadata.emplace("algorithm", eve::Value(std::string(recipeId)));
253 metadata.emplace("determinism", eve::Value("seeded_cpu"));
254 const Bounds bounds = boundsForMesh(mesh);
255 return makeArtifact(id, ArtifactType::MeshData, eve::SchemaVersion(1), *key, bounds, {}, std::move(metadata),
256 std::move(mesh));
257}
