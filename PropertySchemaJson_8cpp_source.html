31Value propertyDescriptorToJsonSchema(const PropertyDescriptor& descriptor) {
32 Value::Object node;
33 node["type"] = propertyTypeName(descriptor.type);
34 node["title"] = descriptor.displayNameKey;
35 node["description"] = descriptor.descriptionKey;
36 node["category"] = descriptor.category;
37 if (descriptor.type == PropertyType::Enum && !descriptor.enumItems.empty()) {
38 Value::Array items;
39 items.reserve(descriptor.enumItems.size());
40 for (const std::string& item : descriptor.enumItems) items.emplace_back(item);
41 node["enum"] = std::move(items);
42 }
43 if (descriptor.type == PropertyType::Vec2 || descriptor.type == PropertyType::Vec3 ||
44 descriptor.type == PropertyType::Vec4) {
45 node["items"] = Value::Object{{"type", "number"}};
46 const int count = descriptor.type == PropertyType::Vec2 ? 2 : (descriptor.type == PropertyType::Vec3 ? 3 : 4);
47 node["minItems"] = count;
48 node["maxItems"] = count;
49 }
50 if (descriptor.numeric.minimum) node["minimum"] = *descriptor.numeric.minimum;
51 if (descriptor.numeric.maximum) node["maximum"] = *descriptor.numeric.maximum;
52 if (hasPropertyFlag(descriptor.flags, PropertyFlag::ReadOnly)) node["readOnly"] = true;
53 return node;
54}

65Result<Value> propertySchemaToJsonSchema(const PropertySchema& schema, std::string_view schemaId) {
66 if (schemaId.empty())
67 return failed<Value>(Status::Rejected, RuleId("editing.schema.empty-id"),
68 "JSON Schema projection requires a non-empty schema id");
69
70 if (schema.documentSchema.type() != Value::Type::Null) {
71 if (!schema.documentSchema.getIf<Value::Object>() || !schema.properties.empty())
72 return failed<Value>(Status::Rejected, RuleId("editing.schema.conflicting-shapes"),
73 "Document schema must be an object and cannot coexist with flat properties");
74 return applied<Value>(schema.documentSchema);
75 }
76
77 Value::Object root;
78 root["$schema"] = "https://json-schema.org/draft/2020-12/schema";
79 root["$id"] = std::string(schemaId);
80 root["type"] = "object";
81 root["typeId"] = schema.typeId.empty() ? std::string(schemaId) : schema.typeId;
82 root["version"] = static_cast<std::int64_t>(schema.version);
83
84 Value::Object properties;
85 Value::Array required;
86 for (const PropertyDescriptor& descriptor : schema.properties) {
87 const std::string key = descriptor.path.value();
88 if (key.empty())
89 return failed<Value>(Status::Rejected, RuleId("editing.schema.empty-path"),
90 "PropertySchema entries must use a non-empty path");
91 properties.emplace(key, propertyDescriptorToJsonSchema(descriptor));
92 if (!hasPropertyFlag(descriptor.flags, PropertyFlag::ReadOnly)) required.emplace_back(key);
93 }
94 root["properties"] = std::move(properties);
95 root["required"] = std::move(required);
96 root["additionalProperties"] = false;
97 return applied<Value>(Value(std::move(root)));
98}
