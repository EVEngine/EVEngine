15bool isBareWord(const std::string& value) {
16 if (value.empty()) return false;
17 for (const char c : value) {
18 const bool accepted = (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') ||
19 c == '_' || c == '.' || c == ':';
20 if (!accepted) return false;
21 }
22 return true;

57eve::Result<void> StepKindRegistry::registerStep(StepKindDescriptor descriptor) {
58 if (!isBareWord(descriptor.type))
59 return registryFailure(eve::DiagnosticCode::InvalidArgument,
60 "step type must be a non-empty bare word", "type");
61 if (descriptors_.contains(descriptor.type))
62 return registryFailure(eve::DiagnosticCode::AlreadyExists,
63 "step type '" + descriptor.type + "' is already registered", descriptor.type);
64 for (const auto& field : descriptor.fields) {
65 if (!isBareWord(field.name))
66 return registryFailure(eve::DiagnosticCode::InvalidArgument,
67 "step '" + descriptor.type + "' declares an invalid field name", field.name);
68 }
69 if (descriptor.displayName.empty()) descriptor.displayName = descriptor.type;
70 descriptors_.emplace(descriptor.type, std::move(descriptor));
71 return eve::Result<void>::success();
72}

108eve::Result<void> StepKindRegistry::validate(const SequenceNode& node) const {
109 const auto* contract = descriptor(node.type);
110 if (!contract)
111 return registryFailure(eve::DiagnosticCode::NotFound,
112 "node '" + node.id + "' uses unregistered step type '" + node.type + "'",
113 "nodes." + node.id + ".type");
114 if (!node.payload.isObject())
115 return registryFailure(eve::DiagnosticCode::InvalidArgument,
116 "node '" + node.id + "' payload must be an object",
117 "nodes." + node.id + ".payload");
118
119 for (const auto& field : contract->fields) {
120 const eve::Value* value = node.payload.find(field.name);
121 if (!value) {
122 if (field.required)
123 return registryFailure(eve::DiagnosticCode::InvalidArgument,
124 "step '" + node.type + "' on node '" + node.id + "' requires field '" +
125 field.name + "'",
126 "nodes." + node.id + ".payload." + field.name);
127 continue;
128 }
129 if (!matchesFieldType(*value, field.type))
130 return registryFailure(eve::DiagnosticCode::InvalidArgument,
131 "step '" + node.type + "' field '" + field.name + "' must be " +
132 fieldTypeName(field.type),
133 "nodes." + node.id + ".payload." + field.name);
134 }
135
136 for (const auto& key : node.payload.keys()) {
137 bool declared = false;
138 for (const auto& field : contract->fields) {
139 if (field.name == key) {
140 declared = true;
141 break;
142 }
143 }
144 if (!declared)
145 return registryFailure(eve::DiagnosticCode::InvalidArgument,
146 "step '" + node.type + "' does not accept field '" + key + "'",
147 "nodes." + node.id + ".payload." + key);
148 }
149
150 if (contract->shape == StepShape::Instant && !hasHandler(node.type))
151 return registryFailure(eve::DiagnosticCode::PreconditionViolation,
152 "step '" + node.type + "' is instant but has no handler bound",
153 "nodes." + node.id + ".type");
154 if (contract->validate) {
155 auto domainResult = contract->validate(node);
156 if (!domainResult.ok()) return domainResult;
157 }
158 return eve::Result<void>::success();
159}

161StepOutcome StepKindRegistry::dispatch(const SequenceNode& node, const StepContext& context) const {
162 const auto* contract = descriptor(node.type);
163 if (!contract) {
164 StepOutcome outcome;
165 outcome.status = StepStatus::Failed;
166 outcome.error = "step type '" + node.type + "' is not registered";
167 return outcome;
168 }
169 const auto handler = handlers_.find(node.type);
170 if (handler == handlers_.end()) {
171 if (contract->shape == StepShape::Await) {
172 StepOutcome outcome;
173 outcome.status = StepStatus::Blocked;
174 return outcome;
175 }
176 StepOutcome outcome;
177 outcome.status = StepStatus::Failed;
178 outcome.error = "step type '" + node.type + "' has no handler";
179 return outcome;
180 }
181 return handler->second(node, context);
182}
