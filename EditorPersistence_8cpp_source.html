18DomainOperation decodeOperation(const EditorValue& value) {
19 DomainOperation operation;
20 const auto* object = value.getIf<EditorValue::Object>();
21 if (!object) return operation;
22 const auto text = [&](const char* key) {
23 auto found = object->find(key);
24 const auto* result = found == object->end() ? nullptr : found->second.getIf<std::string>();
25 return result ? *result : std::string{};
26 };
27 operation.type = text("type");
28 operation.inverseType = text("inverseType");
29 operation.target = TargetId(text("target"));
30 operation.mergeKey = text("mergeKey");
31 auto payload = object->find("payload");
32 if (payload != object->end()) operation.payload = payload->second;
33 auto inverse = object->find("inverse");
34 if (inverse != object->end()) operation.inverse = inverse->second;
35 auto hasInverse = object->find("hasInverse");
36 const auto* flag = hasInverse == object->end() ? nullptr : hasInverse->second.getIf<bool>();
37 operation.hasInverse = flag && *flag;
38 return operation;
39}

63EditorResult<EditorPersistenceSnapshot> EditorPersistenceAdapter::commit(const EditorPersistenceSnapshot& base,
64 EditorValue content,
65 std::vector<DomainOperation> journal) {
66 if (!store_ || base.kind != kind_)
67 return persistenceError(EditorStatus::Rejected, "editor.persistence.kind-mismatch",
68 "Persistence checkpoint belongs to another backend kind");
69 EditorValue::Array serializedJournal;
70 for (const DomainOperation& operation : journal) serializedJournal.push_back(operationValue(operation));
71 EditorValue::Object envelope;
72 envelope["schemaVersion"] = EditorValue(1);
73 envelope["content"] = content;
74 envelope["journal"] = EditorValue(std::move(serializedJournal));
75 auto stored =
76 store_->compareAndSwap(resourceUri_, base.revision, base.contentHash, EditorValue(std::move(envelope)));
77 if (!stored.ok()) return EditorResult<EditorPersistenceSnapshot>::failure(stored.status());

81EditorResult<EditorPersistenceSnapshot> EditorPersistenceAdapter::decode(EditorPersistenceKind kind,
82 const StoredDocument& stored) {
83 const auto* envelope = stored.content.getIf<EditorValue::Object>();
84 if (!envelope)
85 return persistenceError(EditorStatus::Failed, "editor.persistence.invalid-envelope",
86 "Persistence checkpoint must be an object");
87 const auto content = envelope->find("content");
88 const auto journal = envelope->find("journal");
89 const auto* journalValues = journal == envelope->end() ? nullptr : journal->second.getIf<EditorValue::Array>();
90 if (content == envelope->end() || !journalValues)
91 return persistenceError(EditorStatus::Failed, "editor.persistence.invalid-envelope",
92 "Persistence checkpoint content or journal is missing");
93 EditorPersistenceSnapshot result;
94 result.kind = kind;
95 result.revision = stored.revision;
96 result.contentHash = stored.contentHash;
97 result.content = content->second;
98 for (const EditorValue& value : *journalValues) result.journal.push_back(decodeOperation(value));
99 return eve::editing::applied<EditorPersistenceSnapshot>(std::move(result));
100}
