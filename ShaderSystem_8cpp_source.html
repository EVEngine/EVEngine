100void ShaderSystem::dispatch(int entityCount, float dt) {
101 if (!gpu_ || !shader_ || entityCount <= 0) return;
102
103 for (int i = 0; i < kMaxBindings; ++i) {
104 if (buffers_[i]) shader_->bindBuffer(i, buffers_[i]);
105 }
106
107 shader_->setFloat(0, dt);
108 shader_->setFloat(1, float(entityCount));
109
110 const int ls = localSize_ > 0 ? localSize_ : kDefaultLocalSize;
111 const int groups = (entityCount + ls - 1) / ls;
112 gpu_->dispatch(shader_, groups, 1, 1);
