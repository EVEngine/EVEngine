46 if (binding < 0 || binding >= kMaxBindings)
47 throw Exception("ShaderSystem.ensureBuffer: binding out of range");
48 if (!gpu_) throw Exception("ShaderSystem.ensureBuffer: Gpgpu is null");
49 if (floatCount <= 0) floatCount = 1;
