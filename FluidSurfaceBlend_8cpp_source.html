7glm::vec4 FluidSurfaceRenderer::blendFluidColor(FluidBlendFactor factor, const glm::vec4& source,
8 const glm::vec4& destination) {
9 switch (factor) {
10 case FluidBlendFactor::Zero: return glm::vec4(0.f);
11 case FluidBlendFactor::One: return glm::vec4(1.f);
12 case FluidBlendFactor::DestinationColor: return destination;
13 case FluidBlendFactor::SourceColor: return source;
14 case FluidBlendFactor::OneMinusDestinationColor: return glm::vec4(1.f) - destination;
15 case FluidBlendFactor::SourceAlpha: return glm::vec4(source.a);
16 case FluidBlendFactor::OneMinusSourceColor: return glm::vec4(1.f) - source;
17 case FluidBlendFactor::DestinationAlpha: return glm::vec4(destination.a);
18 case FluidBlendFactor::OneMinusDestinationAlpha: return glm::vec4(1.f - destination.a);
19 case FluidBlendFactor::SourceAlphaSaturate: {
20 const float saturated = std::min(source.a, 1.f - destination.a);
21 return glm::vec4(saturated, saturated, saturated, 1.f);
22 }
23 case FluidBlendFactor::OneMinusSourceAlpha: return glm::vec4(1.f - source.a);
24 }
25 return glm::vec4(0.f);
26}

39Result<void> FluidSurfaceRenderer::compositeConfiguredSurfaceBlend(std::span<const uint8_t> sceneColor) {
40 const size_t expected = size_t(params_.width) * size_t(params_.height) * 4u;
41 if (!auxiliaryCurrent_ || residentColorCurrent_)
42 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
43 "Current fluid frame has no host-visible surface color",
44 "fluids.surface.blend"));
45 if (sceneColor.size() != expected)
46 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
47 "Surface blend requires a matching RGBA8 scene image",
48 "fluids.surface.blend"));
49
50 for (size_t offset = 0; offset < expected; offset += 4u) {
51 glm::vec4 source, destination;
52 for (size_t channel = 0; channel < 4u; ++channel) {
53 const auto component = static_cast<glm::vec4::length_type>(channel);
54 source[component] = float(color_[offset + channel]) / 255.f;
55 destination[component] = float(sceneColor[offset + channel]) / 255.f;
56 }
57 const glm::vec4 blended =
58 glm::clamp(source * blendFluidColor(surfaceBlendSource_, source, destination) +
59 destination * blendFluidColor(surfaceBlendDestination_, source, destination),
60 glm::vec4(0.f), glm::vec4(1.f));
61 for (size_t channel = 0; channel < 4u; ++channel)
62 color_[offset + channel] =
63 uint8_t(blended[static_cast<glm::vec4::length_type>(channel)] * 255.f + .5f);
64 }
65 residentColorCurrent_ = false;
66 return Result<void>::success();
67}
