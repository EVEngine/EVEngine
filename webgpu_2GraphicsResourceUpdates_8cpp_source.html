21ResultRef<Texture> Graphics::requestFileTexture(const std::string& filename) {
22 if (filename.empty())
23 return ResultRef<Texture>::failure(Diagnostic::error(
24 DiagnosticCode::InvalidArgument, "file texture path is empty", {}, {}, "graphics.texture.request"));
25 try {
26 const std::string key = normalizeTexPath(filename);
27 auto it = texturesByPath.find(key);
28 if (it != texturesByPath.end() && it->second) return ResultRef<Texture>::success(std::ref(*it->second));
29 if (!fileTextureSourceExists(filename) && !fileTextureSourceExists(key))
30 return ResultRef<Texture>::failure(Diagnostic::error(DiagnosticCode::NotFound,
31 "file texture source is absent: " + filename, {}, {},
32 "graphics.texture.request"));
33 requestFileImageDecode(key);
34 auto texture = std::make_unique<Texture>();
35 texture->markDeferredFilePixels(this);
36 Texture* raw = texture.get();
37 DeferredFileTexture pending{key, raw};
38 if (ownedTextures.size() == ownedTextures.capacity())
39 ownedTextures.reserve(std::max(size_t(1), ownedTextures.capacity() * 2));
40 if (deferredFileTextures_.size() == deferredFileTextures_.capacity())
41 deferredFileTextures_.reserve(std::max(size_t(1), deferredFileTextures_.capacity() * 2));
42 texturesByPath.insert_or_assign(key, raw);
43 ownedTextures.push_back(std::move(texture));
44 deferredFileTextures_.push_back(std::move(pending));
45 return ResultRef<Texture>::success(std::ref(*raw));
46 } catch (const std::exception& error) {
47 return ResultRef<Texture>::failure(
48 Diagnostic::error(DiagnosticCode::Failed, error.what(), {}, {}, "graphics.texture.request"));
49 }
50}

52bool Graphics::uploadDeferredFileTexture(Texture* texture, image::ImageData* data) {
53 if (!texture || !data) return false;
54 if (texture->gpuHandle)
55 return updateTexture(texture, data->getWidth(), data->getHeight(),
56 static_cast<const uint8_t*>(data->getData()));
57 TextureCreateInfo info;
58 info.sampler = texture->sampler;
59 Texture* fresh = newTexture(data, info);
60 if (!fresh || !fresh->gpuHandle) return false;
61 texture->gpuHandle = fresh->gpuHandle;
62 texture->width = fresh->width;
63 texture->height = fresh->height;
64 texture->pixelWidth = fresh->pixelWidth;
65 texture->pixelHeight = fresh->pixelHeight;
66 texture->mipmapCount = fresh->mipmapCount;
67 texture->sampler = fresh->sampler;
68 fresh->gpuHandle = nullptr;
69 auto texIt = std::find_if(ownedTextures.begin(), ownedTextures.end(),
70 [&](const std::unique_ptr<Texture>& t) { return t.get() == fresh; });
71 if (texIt == ownedTextures.end()) return false;
72 (void)texIt->release();
73 ownedTextures.erase(texIt);
74 delete fresh;
75 return true;
76}

78bool Graphics::reloadTextureFromFile(const std::string& filename) {
79 const std::string key = normalizeTexPath(filename);
80 auto it = texturesByPath.find(key);
81 if (it == texturesByPath.end()) return false;
82
83 ensureFileTexturesReady();
84 applyPendingResourceChanges();
85
86 // The provider hands back a cache-owned ImageData; the pin keeps it alive until
87 // the pixels have been copied out of it.
88 image::ImageData* data = nullptr;
89 eve::ResourcePin keepAlive;
90 try {
91 auto* imgMod = image::Image::create();
92 data = imgMod->newImageDataFromFile(filename);
93 if (data != nullptr) {
94 auto pinned = eve::ResourceManager::getInstance().pin(data);
95 if (!pinned.ok()) return false;
96 keepAlive = std::move(pinned).takeValue();
97 // The pin is the authority from here on; the borrowed pointer may have gone
98 // stale before the pin was taken.
99 data = static_cast<image::ImageData*>(keepAlive.get());
100 }
101 } catch (...) {
102 return false;
103 }
104 if (!data) return false;
105
106 Texture* tex = it->second;
107 return updateTexture(tex, data->getWidth(), data->getHeight(), static_cast<const uint8_t*>(data->getData()));
108}

110bool Graphics::updateMeshVertices(Mesh* mesh, const float* posXYZ, const float* nrmXYZ, const float* uvST,
111 int vertexCount, const uint32_t* indices, int indexCount) {
112 if (!mesh || !mesh->gpuHandle || !posXYZ || vertexCount <= 0 || indexCount < 0) return false;
113 if (indexCount > 0 && (!indices || indexCount % 3 != 0)) return false;
114 for (int i = 0; i < indexCount; ++i) {
115 if (indices[i] >= uint32_t(vertexCount)) return false;
116 }
117 auto* gpu = static_cast<GpuMesh*>(mesh->gpuHandle);
118 const auto owned =
119 std::find_if(ownedGpuMeshes.begin(), ownedGpuMeshes.end(),
120 [gpu](const std::unique_ptr<GpuMesh>& candidate) { return candidate.get() == gpu; });
121 if (owned == ownedGpuMeshes.end()) return false;
122
123 auto& verts = gpu->updateVertices;
124 verts.clear();
125 verts.reserve(size_t(vertexCount) * 8u);
126 for (int i = 0; i < vertexCount; ++i) {
127 verts.insert(verts.end(), posXYZ + size_t(i) * 3u, posXYZ + size_t(i) * 3u + 3u);
128 if (nrmXYZ)
129 verts.insert(verts.end(), nrmXYZ + size_t(i) * 3u, nrmXYZ + size_t(i) * 3u + 3u);
130 else
131 verts.insert(verts.end(), {0.f, 0.f, 1.f});
132 if (uvST)
133 verts.insert(verts.end(), uvST + size_t(i) * 2u, uvST + size_t(i) * 2u + 2u);
134 else
135 verts.insert(verts.end(), {0.f, 0.f});
136 }
137
138 const uint64_t vertexBytes = verts.size() * sizeof(float);
139 if (vertexBytes > gpu->vertexCapacity) {
140 WGPUBufferDescriptor desc{};
141 desc.label = sv("eve_mesh_dynamic_vb");
142 desc.size = vertexBytes;
143 desc.usage = WGPUBufferUsage_CopyDst | WGPUBufferUsage_Vertex | WGPUBufferUsage_Storage;
144 gpu->vertexBuffer = device.CreateBuffer(reinterpret_cast<const wgpu::BufferDescriptor*>(&desc));
145 gpu->vertexCapacity = vertexBytes;
146 }
147 queue.WriteBuffer(gpu->vertexBuffer, 0, verts.data(), vertexBytes);
148 gpu->vertexCount = uint32_t(vertexCount);
149
150 if (indexCount > 0) {
151 const wgpu::IndexFormat format = vertexCount <= 65535 ? wgpu::IndexFormat::Uint16 : wgpu::IndexFormat::Uint32;
152 auto& idx16 = gpu->updateIndices16;
153 idx16.clear();
154 const void* indexData = indices;
155 uint64_t indexBytes = uint64_t(indexCount) * sizeof(uint32_t);
156 if (format == wgpu::IndexFormat::Uint16) {
157 idx16.reserve(size_t(indexCount) + 1u);
158 for (int i = 0; i < indexCount; ++i) idx16.push_back(uint16_t(indices[i]));
159 if (idx16.size() % 2 != 0) idx16.push_back(0);
160 indexData = idx16.data();
161 indexBytes = idx16.size() * sizeof(uint16_t);
162 }
163 if (format != gpu->indexFormat || indexBytes > gpu->indexCapacity) {
164 WGPUBufferDescriptor desc{};
165 desc.label = sv("eve_mesh_dynamic_ib");
166 desc.size = indexBytes;
167 desc.usage = WGPUBufferUsage_CopyDst | WGPUBufferUsage_Index | WGPUBufferUsage_Storage;
168 gpu->indexBuffer = device.CreateBuffer(reinterpret_cast<const wgpu::BufferDescriptor*>(&desc));
169 gpu->indexCapacity = indexBytes;
170 }
171 queue.WriteBuffer(gpu->indexBuffer, 0, indexData, indexBytes);
172 gpu->indexFormat = format;
173 gpu->indexCount = uint32_t(indexCount);
174 }
175
176 mesh->computeBounds(posXYZ, vertexCount);
177 mesh->gpuVertexCount = int(gpu->vertexCount);
178 mesh->indexCount = int(gpu->indexCount);
179 return true;
180}
