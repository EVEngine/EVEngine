20 const RoadBakeOptions& options,
21 std::size_t& samplingWork) {
22 const auto& points = edge.controlPoints;
23 const int segments = path.segmentCount();
24 EV_ASSERT((!path.isClosed() && static_cast<std::size_t>(segments) + 1 == points.size()),
25 "road parameter scan must consume its owning open edge path");
26 std::vector<float> parameters{0.f};
27 V3 anchor{points.front().x, points.front().y, points.front().z};
28 V3 previous{points[1].x - points[0].x, 0.f, points[1].z - points[0].z};
29 const double spacing = static_cast<double>(edge.style.samplingSpacingMeters) * options.samplingScale;
30 const double threshold =
31 std::min(180., static_cast<double>(edge.style.samplingAngleDegrees) * options.samplingScale);
32 auto append = [&](float parameter) -> Result<void> {
33 if (parameter < parameters.back())
34 return bakeFail<void>(DiagnosticCode::PreconditionViolation, "parameter scan lost monotonic precision",
35 "sampling");
36 if (parameter == parameters.back()) return Result<void>::success();
37 if (parameters.size() == 4097u)
38 return bakeFail<void>(DiagnosticCode::PreconditionViolation, "parameter scan exceeds the 4097 frame budget",
39 "sampling");
40 parameters.push_back(parameter);
41 return Result<void>::success();
42 };
43 for (int segment = 0; segment < segments; ++segment) {
44 const auto& a = points[static_cast<std::size_t>(segment)];
45 const auto& b = points[static_cast<std::size_t>(segment + 1)];
46 const double chord =
47 std::hypot(static_cast<double>(b.x) - a.x, static_cast<double>(b.y) - a.y, static_cast<double>(b.z) - a.z);
48 if (chord <= 1e-7)
49 return bakeFail<std::vector<SplineFrameSample>>(
50 DiagnosticCode::InvalidArgument, "parameter scan requires separated marker positions", "sampling");
51 const float step = static_cast<float>(1. / (chord * 5.));
52 if (!std::isfinite(step) || step <= 0.f)
53 return bakeFail<std::vector<SplineFrameSample>>(DiagnosticCode::InvalidArgument,
54 "parameter scan step is not representable", "sampling");
55 for (float u = step; u < 1.f;) {
56 if (samplingWork >= options.maximumSamplingEvaluations)
57 return bakeFail<std::vector<SplineFrameSample>>(
58 DiagnosticCode::PreconditionViolation, "parameter scan exceeds the shared sampling work budget",
59 "samplingWorkBudget");
60 ++samplingWork;
61 const float parameter = (static_cast<float>(segment) + u) / static_cast<float>(segments);
62 auto sample = path.evaluateResult(parameter);
63 if (!sample.ok()) return Result<std::vector<SplineFrameSample>>::failure(sample.status());
64 V3 current{sample.value().x, sample.value().y, sample.value().z};
65 const V3 direction = current - anchor;
66 const double distance = size(direction);
67 if (distance >= spacing || (distance >= 1. && turn(previous, direction) >= threshold)) {
68 auto added = append(parameter);
69 if (!added.ok()) return Result<std::vector<SplineFrameSample>>::failure(added.status());
70 previous = direction;
71 anchor = current;
72 }
73 const float next = u + step;
74 if (next <= u)
75 return bakeFail<std::vector<SplineFrameSample>>(DiagnosticCode::PreconditionViolation,
76 "parameter scan cannot advance at float precision",
77 "sampling");
78 u = next;
79 }
80 auto added = append(static_cast<float>(segment + 1) / static_cast<float>(segments));
81 if (!added.ok()) return Result<std::vector<SplineFrameSample>>::failure(added.status());
82 const V3 endpoint{b.x, b.y, b.z};
83 const V3 incoming = endpoint - anchor;
84 if (size(incoming) > 1e-7) previous = incoming;
85 anchor = endpoint;
86 }
87 return path.sampleFramesAtParametersResult(parameters, 24);
88}
