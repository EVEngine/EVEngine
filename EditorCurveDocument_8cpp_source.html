29EditorResult<EditorCurveKey> parseKey(const EditorValue& value) {
30 const auto string = [&](const char* name) { const auto* entry = field(value, name); return entry ? entry->getIf<std::string>() : nullptr; };
31 const auto number = [&](const char* name) { const auto* entry = field(value, name); return entry ? entry->getIf<double>() : nullptr; };
32 const auto* id = string("id"); const auto* time = number("time"); const auto* assigned = number("value");
33 const auto* in = number("inTangent"); const auto* out = number("outTangent"); const auto* interpolation = string("interpolation");
34 const std::set<std::string> modes{"constant", "linear", "cubic"};
35 if (!id || id->empty() || !time || !assigned || !in || !out || !interpolation ||
36 !modes.contains(*interpolation) || !std::isfinite(*time) || !std::isfinite(*assigned) ||
37 !std::isfinite(*in) || !std::isfinite(*out) || *time < 0.0 || *time > 1.0)
38 return curveError<EditorCurveKey>(EditorStatus::Rejected, "editor.curve.invalid-key",
39 "Curve key requires stable id, normalized time and finite values");
40 return eve::editing::applied<EditorCurveKey>({StableId(*id), *time, *assigned, *in, *out, *interpolation});
41}

48EditorResult<EditorGradientStop> parseStop(const EditorValue& value) {
49 const auto* idValue = field(value, "id"); const auto* timeValue = field(value, "time"); const auto* colorValue = field(value, "color");
50 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr; const auto* time = timeValue ? timeValue->getIf<double>() : nullptr;
51 const auto* color = colorValue ? colorValue->getIf<EditorValue::Array>() : nullptr;
52 if (!id || id->empty() || !time || !std::isfinite(*time) || *time < 0.0 || *time > 1.0 || !color || color->size() != 4)
53 return curveError<EditorGradientStop>(EditorStatus::Rejected, "editor.curve.invalid-stop",
54 "Gradient stop requires stable id, normalized time and RGBA");
55 EditorGradientStop result; result.id = StableId(*id); result.time = *time;
56 for (std::size_t i = 0; i < 4; ++i) {
57 const auto* number = (*color)[i].getIf<double>();
58 if (!number || !std::isfinite(*number) || *number < 0.0 || *number > 1.0)
59 return curveError<EditorGradientStop>(EditorStatus::Rejected,
60 "editor.curve.invalid-color",
61 "Gradient color must be normalized");
62 result.color[i] = *number;
63 }
64 return eve::editing::applied<EditorGradientStop>(std::move(result));
65}

75std::vector<T> timeline(const std::map<StableId, T>& values) {

96EditorResult<void> EditorCurveDocument::applyDomainOperation(const DomainOperation& operationValue) {
97 if (operationValue.target != TargetId(id_))
98 return curveError<void>(EditorStatus::Rejected, "editor.curve.target-mismatch", "Curve operation targets another document");
99 if (operationValue.type == "curve.key.set.v1") {

101 keys_[parsed.value().id] = std::move(parsed.value());
102 } else if (operationValue.type == "curve.key.delete.v1") {

104 } else if (operationValue.type == "curve.stop.set.v1") {
105 auto parsed = parseStop(operationValue.payload); if (!parsed.ok()) return curveError<void>(parsed.code(), "editor.curve.invalid-stop", "Gradient stop is invalid");
106 stops_[parsed.value().id] = std::move(parsed.value());
107 } else if (operationValue.type == "curve.stop.delete.v1") {
108 auto parsed = parseStop(operationValue.payload); if (!parsed.ok() || !stops_.erase(parsed.value().id)) return curveError<void>(EditorStatus::NotFound, "editor.curve.stop-not-found", "Gradient stop was not found");
109 } else return curveError<void>(EditorStatus::Unsupported, "editor.curve.operation-unsupported", "Curve operation is unsupported");
110 ++revision_; dirty_.include(0, 0); return eve::editing::applied<void>();

148double EditorCurveDocument::sampleCurve(double time) const {
149 const auto values = keys(); if (values.empty()) return 0.0;
150 time = std::clamp(time, 0.0, 1.0);
151 const auto upper = std::upper_bound(values.begin(), values.end(), time, [](double t, const auto& key) { return t < key.time; });
152 if (upper == values.begin()) return upper->value;
153 if (upper == values.end()) return values.back().value;
154 const auto& a = *(upper - 1); const auto& b = *upper; const double span = b.time - a.time;
155 const double t = span > 0.0 ? (time - a.time) / span : 0.0;
156 if (a.interpolation == "constant") return a.value;
157 if (a.interpolation == "linear") return mix(a.value, b.value, t);
158 const double t2 = t * t, t3 = t2 * t;
159 return (2 * t3 - 3 * t2 + 1) * a.value + (t3 - 2 * t2 + t) * a.outTangent * span +
160 (-2 * t3 + 3 * t2) * b.value + (t3 - t2) * b.inTangent * span;
161}

163std::array<double, 4> EditorCurveDocument::sampleGradient(double time) const {
164 const auto values = stops(); if (values.empty()) return {1.0, 1.0, 1.0, 1.0};
165 time = std::clamp(time, 0.0, 1.0);
166 const auto upper = std::upper_bound(values.begin(), values.end(), time, [](double t, const auto& stop) { return t < stop.time; });
167 if (upper == values.begin()) return upper->color;
168 if (upper == values.end()) return values.back().color;
169 const auto& a = *(upper - 1); const auto& b = *upper; const double t = b.time > a.time ? (time - a.time) / (b.time - a.time) : 0.0;
170 std::array<double, 4> result; for (std::size_t i = 0; i < 4; ++i) result[i] = mix(a.color[i], b.color[i], t); return result;
171}

173EditorCurvePreview EditorCurveDocument::preview(int sampleCount, int maximumSamples) const {
174 EditorCurvePreview result; result.documentRevision = revision_;
175 if (sampleCount < 2 || maximumSamples < 2 || sampleCount > maximumSamples) {
176 result.status = EditorStatus::Rejected;
177 result.diagnostics.push_back(editing::ruleDiagnostic(
178 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.curve.preview-budget"),
179 DiagnosticSeverity::Error, "Curve preview sample count exceeds its bounded budget"));
180 return result;
181 }
182 result.curveSamples.reserve(static_cast<std::size_t>(sampleCount)); result.gradientSamples.reserve(static_cast<std::size_t>(sampleCount));
183 for (int i = 0; i < sampleCount; ++i) { const double t = static_cast<double>(i) / (sampleCount - 1); result.curveSamples.push_back(sampleCurve(t)); result.gradientSamples.push_back(sampleGradient(t)); }
184 result.status = EditorStatus::Applied; return result;
