71bool BSPTree2D::insertInto(Node &node, int id, const AABB2 &bounds) {
72 if (!node.bounds.containsAABB(bounds) && node.depth == 0) {
73 if (!node.bounds.intersectsAABB(bounds)) return false;
74 } else if (node.depth > 0 && !node.bounds.containsAABB(bounds)) {
75 return false;
76 }
77
78 if (node.isLeaf()) {
79 node.itemIds.push_back(id);
80 if (static_cast<int>(node.itemIds.size()) > maxPerNode_ && node.depth < maxDepth_) {
81 split(node);
82 }
83 return true;
84 }
85
86 // Fully on one side of the split plane?
87 if (node.axis == 0) {
88 if (bounds.maxX <= node.split && node.left) {
89 return insertInto(*node.left, id, bounds);
90 }
91 if (bounds.minX >= node.split && node.right) {
92 return insertInto(*node.right, id, bounds);
93 }
94 } else {
95 if (bounds.maxY <= node.split && node.left) {
96 return insertInto(*node.left, id, bounds);
97 }
98 if (bounds.minY >= node.split && node.right) {
99 return insertInto(*node.right, id, bounds);
100 }
101 }
102 node.itemIds.push_back(id);
103 return true;
104}

106void BSPTree2D::split(Node &node) {
107 node.axis = node.depth % 2;
108 node.split = (node.axis == 0) ? node.bounds.centerX() : node.bounds.centerY();
109
110 node.left = std::make_unique<Node>();
111 node.right = std::make_unique<Node>();
112 node.left->depth = node.depth + 1;
113 node.right->depth = node.depth + 1;
114 node.left->axis = node.left->depth % 2;
115 node.right->axis = node.right->depth % 2;
116
117 if (node.axis == 0) {
118 node.left->bounds = makeAABB2(node.bounds.minX, node.bounds.minY, node.split, node.bounds.maxY);
119 node.right->bounds = makeAABB2(node.split, node.bounds.minY, node.bounds.maxX, node.bounds.maxY);
120 } else {
121 node.left->bounds = makeAABB2(node.bounds.minX, node.bounds.minY, node.bounds.maxX, node.split);
122 node.right->bounds = makeAABB2(node.bounds.minX, node.split, node.bounds.maxX, node.bounds.maxY);
123 }
124
125 std::vector<int> remain;
126 remain.reserve(node.itemIds.size());
127 for (int id : node.itemIds) {
128 auto it = items_.find(id);
129 if (it == items_.end()) continue;
130 const AABB2 &b = it->second;
131 if (node.axis == 0) {
132 if (b.maxX <= node.split) {
133 node.left->itemIds.push_back(id);
134 } else if (b.minX >= node.split) {
135 node.right->itemIds.push_back(id);
136 } else {
137 remain.push_back(id);
138 }
139 } else {
140 if (b.maxY <= node.split) {
141 node.left->itemIds.push_back(id);
142 } else if (b.minY >= node.split) {
143 node.right->itemIds.push_back(id);
144 } else {
145 remain.push_back(id);
146 }
147 }
148 }
149 node.itemIds.swap(remain);
150}

152void BSPTree2D::collect(Node &node, const AABB2 *rect, float cx, float cy, float radius, bool useCircle) {
153 if (rect && !node.bounds.intersectsAABB(*rect)) return;
154 if (useCircle && !node.bounds.intersectsCircle(cx, cy, radius)) return;
155
156 for (int id : node.itemIds) {
157 auto it = items_.find(id);
158 if (it == items_.end()) continue;
159 const AABB2 &b = it->second;
160 bool hit = false;
161 if (rect) {
162 hit = b.intersectsAABB(*rect);
163 } else if (useCircle) {
164 hit = b.intersectsCircle(cx, cy, radius);
165 } else {
166 hit = b.containsPoint(cx, cy);
167 }
168 if (hit) results_.addUnique(id);
169 }
170
171 if (!node.isLeaf()) {
172 if (node.left) collect(*node.left, rect, cx, cy, radius, useCircle);
173 if (node.right) collect(*node.right, rect, cx, cy, radius, useCircle);
174 }
175}
