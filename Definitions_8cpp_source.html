202eve::Result<Definition> makeDefinition(const std::string& type, const std::string& id, int version,
203 const std::string& json) {
204 if (type.empty() || id.empty())
205 return failure<Definition>(eve::DiagnosticCode::InvalidArgument, "definition type and id must not be empty");
206 if (!eve::LogicalId::fromParts(type, id))
207 return failure<Definition>(eve::DiagnosticCode::InvalidArgument,
208 "definition type and id must form a valid namespace:name reference");
209 if (version <= 0)
210 return failure<Definition>(eve::DiagnosticCode::InvalidArgument, "definition schema version must be positive");
211
212 auto parsed = eve::Value::fromJson(json);
213 if (!parsed.ok()) return eve::Result<Definition>::failure(parsed.status());
214 auto value = std::move(parsed).takeValue();
215 auto canonical = value.toJson();
216 if (!canonical.ok()) return eve::Result<Definition>::failure(canonical.status());
217
218 if (eve::schema::SchemaRegistry::resolve(type, version)) {
219 const auto errors = eve::schema::SchemaRegistry::validate(type, version, json);
220 if (!errors.empty()) {
221 const auto& error = errors.front();
222 return eve::Result<Definition>::failure(eve::Diagnostic::error(
223 eve::DiagnosticCode::InvalidArgument, error.message, error.path,
224 eve::DiagnosticDetails{
225 {"schemaId", type}, {"schemaVersion", std::to_string(version)}, {"validationCode", error.code}},
226 "definitions.schema"));
227 }
228 }
229
230 Definition definition;
231 definition.type = type;
232 definition.id = id;
233 definition.version = eve::SchemaVersion(static_cast<std::uint64_t>(version));
234 definition.json = std::move(canonical).takeValue();
235 return eve::Result<Definition>::success(std::move(definition));
236}

404std::string DefinitionRegistry::snapshotJson() const {
405 const auto state = storage_.snapshotState();
406 eve::Value::Array definitions;
407 eve::Value::Array events;
408 eve::Value::Array generations;
409 definitions.reserve(storage_.size());
410 events.reserve(storage_.eventCount());
411 generations.reserve(state.entries.size());
412
413 for (const auto& [key, entry] : state.entries) {
414 eve::Value::Object generation;
415 generation.emplace("generation", eve::Value(std::to_string(entry.generation.value())));
416 generation.emplace("id", eve::Value(key.second));
417 generation.emplace("type", eve::Value(key.first));
418 generations.emplace_back(std::move(generation));
419
420 if (!entry.value.has_value()) continue;
421 const auto& definition = *entry.value;
422 eve::Value::Object item;
423 item.emplace("generation", eve::Value(std::to_string(entry.generation.value())));
424 item.emplace("id", eve::Value(definition.id));
425 item.emplace("json", parseStoredJson(definition.json));
426 item.emplace("type", eve::Value(definition.type));
427 item.emplace("version", eve::Value(static_cast<std::int64_t>(definition.version.value())));
428 definitions.emplace_back(std::move(item));
429 }
430
431 for (const auto& raw : state.events) {
432 const auto event = projectEvent(raw);
433 eve::Value::Object item;
434 item.emplace("generation", eve::Value(std::to_string(event.generation.value())));
435 item.emplace("id", eve::Value(event.id));
436 item.emplace("name", eve::Value(event.name));
437 item.emplace("sequence", eve::Value(std::to_string(event.sequence.value())));
438 item.emplace("type", eve::Value(event.type));
439 item.emplace("version", eve::Value(static_cast<std::int64_t>(event.version.value())));
440 events.emplace_back(std::move(item));
441 }
442
443 eve::Value::Object root;
444 root.emplace("definitions", eve::Value(std::move(definitions)));
445 root.emplace("events", eve::Value(std::move(events)));
446 root.emplace("generations", eve::Value(std::move(generations)));
447 root.emplace("nextEventSequence", eve::Value(std::to_string(state.nextEventSequence.value())));
448 root.emplace("version", eve::Value(std::int64_t(1)));
449 auto encoded = eve::Value(std::move(root)).toJson();
450 if (!encoded.ok()) std::terminate();
451 return std::move(encoded).takeValue();
452}

454eve::Result<void> DefinitionRegistry::restoreJson(const std::string& json) {
455 auto schemaValidation = validateDefinitionsSnapshot(json);
456 if (!schemaValidation.ok()) return schemaValidation;
457 auto parsed = eve::Value::fromJson(json);
458 if (!parsed.ok()) {
459 return eve::Result<void>::failure(parsed.status());
460 }
461 const eve::Value value = std::move(parsed).takeValue();
462 const auto* root = value.getIf<eve::Value::Object>();
463 if (!root) {
464 return failure<void>(eve::DiagnosticCode::ParseError, "definitions snapshot must be an object");
465 }
466 const auto* snapshotVersion = field(*root, "version");
467 const auto* versionNumber = snapshotVersion ? snapshotVersion->getIf<std::int64_t>() : nullptr;
468 const auto* definitions = field(*root, "definitions");
469 const auto* generations = field(*root, "generations");
470 const auto* events = field(*root, "events");
471 const auto* nextSequenceValue = field(*root, "nextEventSequence");
472 const auto* definitionArray = definitions ? definitions->getIf<eve::Value::Array>() : nullptr;
473 const auto* generationArray = generations ? generations->getIf<eve::Value::Array>() : nullptr;
474 const auto* eventArray = events ? events->getIf<eve::Value::Array>() : nullptr;
475 std::uint64_t nextSequence = 0;
476 if (!versionNumber || *versionNumber != 1 || !definitionArray || !generationArray || !eventArray ||
477 !nextSequenceValue || !readUint64String(*nextSequenceValue, nextSequence) || nextSequence == 0) {
478 return failure<void>(eve::DiagnosticCode::ParseError, "invalid definitions snapshot fields");
479 }
480
481 Storage::State candidate;
482 for (std::size_t index = 0; index < generationArray->size(); ++index) {
483 const auto* item = (*generationArray)[index].getIf<eve::Value::Object>();
484 std::string type;
485 std::string id;
486 std::uint64_t generation = 0;
487 const auto* generationValue = item ? field(*item, "generation") : nullptr;
488 if (!item || !readString(*item, "type", type) || !readString(*item, "id", id) || type.empty() || id.empty() ||
489 !generationValue || !readUint64String(*generationValue, generation) || generation == 0 ||
490 !candidate.entries.emplace(Key{type, id}, Storage::Entry{eve::Generation(generation), std::nullopt})
491 .second) {
492 return failure<void>(eve::DiagnosticCode::ParseError,
493 "invalid definition generation at index " + std::to_string(index));
494 }
495 }
496
497 for (std::size_t index = 0; index < definitionArray->size(); ++index) {
498 const auto* item = (*definitionArray)[index].getIf<eve::Value::Object>();
499 std::string type;
500 std::string id;
501 eve::SchemaVersion versionValue;
502 std::uint64_t generation = 0;
503 const auto* jsonValue = item ? field(*item, "json") : nullptr;
504 const auto* versionField = item ? field(*item, "version") : nullptr;
505 const auto* generationField = item ? field(*item, "generation") : nullptr;
506 if (!item || !readString(*item, "type", type) || !readString(*item, "id", id) || type.empty() || id.empty() ||
507 !versionField || !readPositiveSchemaVersion(*versionField, versionValue) || !generationField ||
508 !readUint64String(*generationField, generation) || generation == 0 || !jsonValue) {
509 return failure<void>(eve::DiagnosticCode::ParseError,
510 "invalid definition at index " + std::to_string(index));
511 }
512 const Key key{type, id};
513 auto entry = candidate.entries.find(key);
514 if (entry == candidate.entries.end() || entry->second.generation != eve::Generation(generation) ||
515 entry->second.value.has_value()) {
516 return failure<void>(eve::DiagnosticCode::InvariantViolation,
517 "inconsistent definition generation at index " + std::to_string(index));
518 }
519 auto canonical = jsonValue->toJson();
520 if (!canonical.ok()) {
521 return eve::Result<void>::failure(canonical.status());
522 }
523 Definition definition;
524 definition.type = type;
525 definition.id = id;
526 definition.version = versionValue;
527 definition.generation = eve::Generation(generation);
528 definition.json = std::move(canonical).takeValue();
529 entry->second.value = std::move(definition);
530 }
531
532 eve::EventSequence previousSequence;
533 for (std::size_t index = 0; index < eventArray->size(); ++index) {
534 const auto* item = (*eventArray)[index].getIf<eve::Value::Object>();
535 std::string name;
536 std::string type;
537 std::string id;
538 eve::SchemaVersion versionValue;
539 std::uint64_t sequence = 0;
540 std::uint64_t generation = 0;
541 const auto* versionField = item ? field(*item, "version") : nullptr;
542 const auto* sequenceField = item ? field(*item, "sequence") : nullptr;
543 const auto* generationField = item ? field(*item, "generation") : nullptr;
544 if (!item || !readString(*item, "name", name) || !readString(*item, "type", type) ||
545 !readString(*item, "id", id) || (name != "definition_reloaded" && name != "definition_removed") ||
546 type.empty() || id.empty() || !versionField || !readPositiveSchemaVersion(*versionField, versionValue) ||
547 !sequenceField || !readUint64String(*sequenceField, sequence) || sequence == 0 ||
548 (!previousSequence.isZero() && eve::EventSequence(sequence) <= previousSequence) || !generationField ||
549 !readUint64String(*generationField, generation) || generation == 0) {
550 return failure<void>(eve::DiagnosticCode::ParseError,
551 "invalid definition event at index " + std::to_string(index));
552 }
553 const bool removed = name == "definition_removed";
554 Storage::Event event;
555 event.sequence = eve::EventSequence(sequence);
556 event.operation = removed
557 ? eve::RegistryOperation::Remove
558 : (generation == 1 ? eve::RegistryOperation::Insert : eve::RegistryOperation::Replace);
559 event.key = Key{type, id};
560 event.generation = eve::Generation(generation);
561 event.tombstone = removed;
562 event.data.version = versionValue;
563 candidate.events.push_back(std::move(event));
564 previousSequence = eve::EventSequence(sequence);
565 }
566 if (!previousSequence.isZero() && eve::EventSequence(nextSequence) <= previousSequence) {
567 return failure<void>(eve::DiagnosticCode::InvariantViolation,
568 "next event sequence must exceed retained events");
569 }
570 candidate.nextEventSequence = eve::EventSequence(nextSequence);
571
572 auto restored = storage_.restoreState(std::move(candidate));
573 if (!restored.ok()) {
574 return eve::Result<void>::failure(restored.status());
575 }
576 clearEventProjection();
577 revision_ = eve::Revision(nextSequence - 1);
578 tick_ = eve::SimulationTick{};
579 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
580}

589eve::Result<void> DefinitionRegistry::restoreSnapshot(const eve::SnapshotEnvelope& source,
590 const eve::SnapshotHashProvider& hashProvider) {
591 if (source.type != "definitions.registry" || source.schema != definitionsSchema())
592 return failure<void>(eve::DiagnosticCode::InvalidArgument,
593 "snapshot does not belong to definitions::DefinitionRegistry");
594 if (!instanceId_.isNil() && source.instanceId != instanceId_)
595 return failure<void>(eve::DiagnosticCode::Conflict,
596 "snapshot instanceId does not match definitions::DefinitionRegistry");
597
598 auto migrated = definitionsMigrations().migrate(source, eve::SchemaVersion(1), hashProvider);
599 if (!migrated.ok()) return eve::Result<void>::failure(migrated.status());
600 const auto& candidate = migrated.value();
601 auto metadata = eve::validateSnapshotPayloadMetadata(candidate.payload, candidate.revision, candidate.tick);
602 if (!metadata.ok()) return eve::Result<void>::failure(metadata.status());
603 auto payload = candidate.payload.toJson();
604 if (!payload.ok()) return eve::Result<void>::failure(payload.status());
605 const std::string payloadJson = std::move(payload).takeValue();
606
607 // restoreJson validates into a candidate State before the single storage swap.
608 auto restored = restoreJson(payloadJson);
609 if (!restored.ok()) return eve::Result<void>::failure(restored.status());
610
611 instanceId_ = candidate.instanceId;
612 revision_ = candidate.revision;
613 tick_ = candidate.tick;
614 return eve::Result<void>::success();
615}
