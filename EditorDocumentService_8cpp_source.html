22Result<StoredDocument> MemoryAtomicDocumentStore::compareAndSwap(const std::string& resourceUri,
23 Revision expectedRevision,
24 const std::string& expectedContentHash,
25 const EditorValue& content) {
26 auto found = documents_.find(resourceUri);
27 const Revision current = found == documents_.end() ? 0 : found->second.revision;
28 const std::string currentHash = found == documents_.end() ? std::string{} : found->second.contentHash;
29 if (current != expectedRevision || currentHash != expectedContentHash)
30 return eve::editing::failed<StoredDocument>(EditorStatus::Conflict, RuleId("editor.document.disk-conflict"),
31 "Document store revision changed before save");
32 StoredDocument stored;
33 stored.content = content;
34 stored.revision = current + 1;
35 stored.contentHash = editorValueContentHash(content);
36 documents_.insert_or_assign(resourceUri, stored);
37 return eve::editing::applied<StoredDocument>(std::move(stored));
38}

40Result<DocumentSnapshot> DocumentService::open(DocumentKey key, std::string title, std::string resourceUri,
41 EditorValue initialContent) {
42 if (!store_ || key.asset.empty() || resourceUri.empty())
43 return error(EditorStatus::Rejected, "editor.document.invalid-open",
44 "Document store, asset identity and resource URI are required");
45 DocumentId id("document:" + key.asset.value());
46 if (open_.contains(id)) return eve::editing::applied<DocumentSnapshot>(open_.at(id).snapshot);
47

63 } else {

65 }

99Result<DocumentSnapshot> DocumentService::executeSave(const SaveTicket& ticket) {
100 auto pending = pendingSaves_.find(ticket.id);
101 auto document = open_.find(ticket.document);
102 if (pending == pendingSaves_.end() || document == open_.end())
103 return error(EditorStatus::NotFound, "editor.document.save-ticket-not-found",
104 "Save ticket or document is no longer available");
105 document->second.snapshot.state = DocumentState::Saving;
106 Result<StoredDocument> stored =
107 store_->compareAndSwap(document->second.snapshot.resourceUri, ticket.expectedDiskRevision,
108 ticket.expectedContentHash, pending->second.content);
109 pendingSaves_.erase(pending);
110 if (!stored.ok()) {
111 document->second.snapshot.state =
112 stored.code() == EditorStatus::Conflict ? DocumentState::Conflict : DocumentState::Failed;
113 document->second.snapshot.diagnostics = stored.diagnostics();
114 return Result<DocumentSnapshot>::failure(stored.status());
115 }
116 document->second.snapshot.revision.disk = stored.value().revision;
117 document->second.snapshot.revision.saved = ticket.capturedEditRevision;
118 document->second.snapshot.diskContentHash = stored.value().contentHash;
119 document->second.snapshot.state = DocumentState::Ready;
120 document->second.snapshot.diagnostics.clear();
121 return eve::editing::applied<DocumentSnapshot>(document->second.snapshot);
122}

154Result<DocumentSnapshot> DocumentService::reconcileExternal(const DocumentId& document, ContentValidator validator) {
155 auto found = open_.find(document);
156 if (found == open_.end()) return error(EditorStatus::NotFound, "editor.document.not-open", "Document is not open");
157
158 Result<StoredDocument> stored = store_->read(found->second.snapshot.resourceUri);
159 if (!stored.ok()) {
160 const bool expectedMissing = stored.code() == EditorStatus::NotFound &&
161 found->second.snapshot.revision.disk == 0 &&
162 found->second.snapshot.diskContentHash.empty();
163 if (expectedMissing) return eve::editing::applied<DocumentSnapshot>(found->second.snapshot);
164 found->second.snapshot.state =
165 stored.code() == EditorStatus::NotFound || stored.code() == EditorStatus::Conflict ? DocumentState::Conflict
166 : DocumentState::Failed;
167 found->second.snapshot.diagnostics = stored.diagnostics();
168 if (found->second.snapshot.diagnostics.empty())
169 found->second.snapshot.diagnostics.push_back(
170 eve::editing::ruleDiagnostic(eve::DiagnosticCode::NotFound, RuleId("editor.document.external-missing"),
171 DiagnosticSeverity::Error, "Persisted document was removed externally"));
172 const EditorStatus status =
173 found->second.snapshot.state == DocumentState::Conflict ? EditorStatus::Conflict : stored.code();
174 return Result<DocumentSnapshot>::failure(eve::Status(status, found->second.snapshot.diagnostics));
175 }
176
177 const bool changed = stored.value().revision != found->second.snapshot.revision.disk ||
178 stored.value().contentHash != found->second.snapshot.diskContentHash;
179 if (!changed) {
180 if (found->second.snapshot.state == DocumentState::Conflict) {
181 found->second.snapshot.state = DocumentState::Ready;
182 found->second.snapshot.diagnostics.clear();
183 }
184 return eve::editing::applied<DocumentSnapshot>(found->second.snapshot);
185 }
186 if (found->second.snapshot.dirty()) {
187 found->second.snapshot.state = DocumentState::Conflict;
188 found->second.snapshot.diagnostics = {eve::editing::ruleDiagnostic(
189 eve::DiagnosticCode::Conflict, RuleId("editor.document.external-conflict"), DiagnosticSeverity::Error,
190 "Document changed on disk while the session has unsaved edits")};
191 return Result<DocumentSnapshot>::failure(
192 eve::Status(EditorStatus::Conflict, found->second.snapshot.diagnostics));
193 }
194
195 if (validator) {
196 auto validated = validator(stored.value().content);
197 if (!validated.ok()) {
198 found->second.snapshot.state = DocumentState::Conflict;
199 found->second.snapshot.diagnostics = validated.diagnostics();
200 if (found->second.snapshot.diagnostics.empty())
201 found->second.snapshot.diagnostics.push_back(eve::editing::ruleDiagnostic(
202 eve::DiagnosticCode::InvalidArgument, RuleId("editor.document.external-invalid"),
203 DiagnosticSeverity::Error, "External document content failed domain validation"));
204 return Result<DocumentSnapshot>::failure(
205 eve::Status(EditorStatus::Conflict, found->second.snapshot.diagnostics));
206 }
207 }
208
209 found->second.content = stored.value().content;
210 const Revision synchronizedEdit = std::max(found->second.snapshot.revision.edit + 1, stored.value().revision);
211 found->second.snapshot.revision.edit = synchronizedEdit;
212 found->second.snapshot.revision.saved = synchronizedEdit;
213 found->second.snapshot.revision.disk = stored.value().revision;
214 found->second.snapshot.diskContentHash = stored.value().contentHash;
215 found->second.snapshot.state = DocumentState::Ready;
216 found->second.snapshot.diagnostics.clear();
217 return eve::editing::applied<DocumentSnapshot>(found->second.snapshot);
218}

245Result<EnvironmentAssetCard> EnvironmentAssetBrowser::card(const AssetRecord& record) const {
246 EnvironmentAssetCard result;
247 result.asset = record.guid;
248 result.logicalUri = record.logicalUri;
249 result.status = record.status;
250 result.diagnostics = record.diagnostics;
251 if (!record.artifacts.empty()) result.previewArtifact = record.artifacts.front();
252 const auto* layoutEntry = field(record.metadata, "layout");
253 const auto* widthEntry = field(record.metadata, "width");
254 const auto* heightEntry = field(record.metadata, "height");
255 const auto* facesEntry = field(record.metadata, "faceCount");
256 const auto* layout = layoutEntry ? layoutEntry->getIf<std::string>() : nullptr;
257 const auto* width = widthEntry ? widthEntry->getIf<int64_t>() : nullptr;
258 const auto* height = heightEntry ? heightEntry->getIf<int64_t>() : nullptr;
259 const auto* faces = facesEntry ? facesEntry->getIf<int64_t>() : nullptr;
260 static const std::set<std::string> layouts{"cubemap", "horizontal-cross", "vertical-cross", "equirectangular"};
261 if (!layout || !layouts.contains(*layout) || !width || !height || *width <= 0 || *height <= 0) {
262 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
263 eve::DiagnosticCode::InvalidArgument, RuleId("editor.environment.invalid-map-metadata"),
264 DiagnosticSeverity::Error, "Environment map requires supported layout and positive dimensions"));
265 } else {
266 result.layout = *layout;
267 result.width = static_cast<int>(*width);
268 result.height = static_cast<int>(*height);
269 result.faceCount = faces ? static_cast<int>(*faces) : (*layout == "cubemap" ? 6 : 1);
270 if (*layout == "cubemap" && result.faceCount != 6)
271 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
272 eve::DiagnosticCode::InvalidArgument, RuleId("editor.environment.cubemap-face-count"),
273 DiagnosticSeverity::Error, "Cubemap assets require exactly six faces"));
274 if (*layout == "equirectangular" && *width != *height * 2)
275 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
276 eve::DiagnosticCode::InvalidArgument, RuleId("editor.environment.equirectangular-aspect"),
277 DiagnosticSeverity::Warning, "Equirectangular maps normally use a 2:1 aspect ratio"));
278 }
279 if (record.artifacts.empty())
280 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
281 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.environment.missing-artifact"),
282 DiagnosticSeverity::Error, "Environment asset has no imported artifact"));
283 if (record.status != AssetStatus::Ready && record.status != AssetStatus::Warning)
284 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
285 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.environment.asset-not-ready"),
286 DiagnosticSeverity::Error, "Environment asset is not ready for preview or assignment"));
287 const bool error =
288 std::any_of(result.diagnostics.begin(), result.diagnostics.end(), [](const EditorDiagnostic& diagnostic) {
289 return diagnostic.severity() == DiagnosticSeverity::Error ||
290 diagnostic.severity() == DiagnosticSeverity::Fatal;
291 });
292 if (error) return Result<EnvironmentAssetCard>::failure(eve::Status(EditorStatus::Rejected, result.diagnostics));
293 return eve::editing::applied<EnvironmentAssetCard>(std::move(result));
294}

334std::size_t nodeCount(const EditorValue& value, std::size_t limit) {
335 std::size_t result = 1;
336 if (const auto* array = value.getIf<EditorValue::Array>()) {
337 for (const auto& child : *array) {
338 if (result > limit) break;
339 result += nodeCount(child, limit - std::min(result, limit));
340 }
341 } else if (const auto* object = value.getIf<EditorValue::Object>()) {
342 for (const auto& [key, child] : *object) {
343 (void)key;
344 if (result > limit) break;
345 result += nodeCount(child, limit - std::min(result, limit));
346 }
347 }
348 return result;
349}

367Result<void> EditorEventTimeline::setCapacity(std::size_t capacity) {
368 if (capacity == 0 || capacity > 1000000)
369 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.timeline.capacity"),
370 "Timeline capacity must be between 1 and 1,000,000");

373 events_.pop_front();
374 ++dropped_;
375 }
376 ++generation_;

380Result<std::uint64_t> EditorEventTimeline::append(EditorTimelineEvent event, std::size_t maximumPayloadNodes) {
381 if (event.domain.empty() || event.type.empty() || maximumPayloadNodes == 0 ||
382 nodeCount(event.payload, maximumPayloadNodes) > maximumPayloadNodes)
383 return eve::editing::failed<std::uint64_t>(EditorStatus::Rejected, RuleId("editor.timeline.invalid-event"),
384 "Timeline event identity or payload budget is invalid");
385 if (event.sequence == 0)
386 event.sequence = nextSequence_++;
387 else {
388 if (event.sequence < nextSequence_)
389 return eve::editing::failed<std::uint64_t>(EditorStatus::Conflict, RuleId("editor.timeline.sequence"),
390 "Timeline event sequence is stale or duplicated");
391 nextSequence_ = event.sequence + 1;
392 }
393 const std::uint64_t sequence = event.sequence;
394 if (events_.size() == capacity_) {
395 events_.pop_front();
396 ++dropped_;
397 }
398 events_.push_back(std::move(event));
399 ++generation_;
400 return eve::editing::applied<std::uint64_t>(sequence);
401}

403Result<EditorTimelinePage> EditorEventTimeline::query(const EditorTimelineQuery& filter, std::size_t offset,
404 std::size_t limit,
405 std::optional<std::uint64_t> generation) const {
406 if (limit == 0 || limit > 10000)
407 return eve::editing::failed<EditorTimelinePage>(EditorStatus::Rejected, RuleId("editor.timeline.page-size"),
408 "Timeline page size must be between 1 and 10,000");
409 if (generation && *generation != generation_)
410 return eve::editing::failed<EditorTimelinePage>(
411 EditorStatus::Conflict, RuleId("editor.timeline.stale-generation"), "Timeline changed while paging");
412 std::vector<EditorTimelineEvent> matched;
413 for (const auto& event : events_) {
414 if (!includes(filter.domains, event.domain) || !includes(filter.types, event.type) ||
415 (!filter.source.empty() && event.source != filter.source) ||
416 (!filter.correlation.empty() && event.correlation != filter.correlation) ||
417 (filter.minimumSeverity && severityRank(event.severity) < severityRank(*filter.minimumSeverity)) ||
418 event.tick < filter.firstTick || (filter.lastTick > 0 && event.tick > filter.lastTick))
419 continue;
420 matched.push_back(event);
421 }
422 EditorTimelinePage result;
423 result.generation = generation_;
424 result.droppedEvents = dropped_;
425 const std::size_t begin = std::min(offset, matched.size());
426 const std::size_t end = std::min(begin + limit, matched.size());
427 result.values.assign(matched.begin() + static_cast<std::ptrdiff_t>(begin),
428 matched.begin() + static_cast<std::ptrdiff_t>(end));
429 result.nextOffset = end;
430 result.hasMore = end < matched.size();
431 return eve::editing::applied<EditorTimelinePage>(std::move(result));
432}

444Result<void> EditorExtensionRegistry::load(IGameEditorExtension& extension) {
445 if (!commands_)
446 return invalid("editor.extension.missing-command-service", "Extension registry has no command service");
447 const std::string owner = extension.ownerModule();
448 if (owner.empty()) return invalid("editor.extension.missing-owner", "Game editor extension owner is required");
449 if (std::any_of(commandVisibility_.begin(), commandVisibility_.end(),
450 [&](const CommandVisibility& command) { return command.ownerModule == owner; }) ||
451 std::any_of(tools_.begin(), tools_.end(),
452 [&](const ExtensionToolDescriptor& tool) { return tool.ownerModule == owner; }))
453 return invalid("editor.extension.duplicate-owner", "Game editor extension owner is already loaded");
454 activeOwner_ = owner;
455 try {
456 extension.registerEditor(*this);
457 } catch (const std::exception& exception) {
458 unload(owner);
459 activeOwner_.clear();
460 return invalid("editor.extension.exception", exception.what());
461 } catch (...) {
462 unload(owner);
463 activeOwner_.clear();
464 return invalid("editor.extension.exception", "Game editor extension threw an unknown exception");
465 }
466 activeOwner_.clear();
467 return eve::editing::applied<void>();
468}

470std::size_t EditorExtensionRegistry::unload(const std::string& ownerModule) {
471 std::size_t removed = commands_ ? commands_->unregisterOwner(ownerModule) : 0;
472 auto eraseOwner = [&](auto& values) {
473 const std::size_t before = values.size();
474 std::erase_if(values, [&](const auto& value) { return value.ownerModule == ownerModule; });
475 removed += before - values.size();
476 };
477 eraseOwner(commandVisibility_);
478 eraseOwner(tools_);
479 eraseOwner(palettes_);
480 eraseOwner(rules_);
481 return removed;
482}

514Result<EditorValue> EditorExtensionRegistry::registerCommand(CommandDescriptor descriptor, EditorCommandHandler handler,
