16EditorResult<StoredDocument> MemoryAtomicDocumentStore::compareAndSwap(const std::string& resourceUri,
17 Revision expectedRevision,
18 const std::string& expectedContentHash,
19 const EditorValue& content) {
20 auto found = documents_.find(resourceUri);
21 const Revision current = found == documents_.end() ? 0 : found->second.revision;
22 const std::string currentHash = found == documents_.end() ? std::string{} : found->second.contentHash;
23 if (current != expectedRevision || currentHash != expectedContentHash)
24 return eve::editing::failed<StoredDocument>(EditorStatus::Conflict, RuleId("editor.document.disk-conflict"),
25 "Document store revision changed before save");
26 StoredDocument stored;
27 stored.content = content;
28 stored.revision = current + 1;
29 stored.contentHash = editorValueContentHash(content);
30 documents_.insert_or_assign(resourceUri, stored);
31 return eve::editing::applied<StoredDocument>(std::move(stored));
32}

34EditorResult<DocumentSnapshot> DocumentService::open(DocumentKey key, std::string title, std::string resourceUri,
35 EditorValue initialContent) {
36 if (!store_ || key.asset.empty() || resourceUri.empty())
37 return error(EditorStatus::Rejected, "editor.document.invalid-open",
38 "Document store, asset identity and resource URI are required");
39 DocumentId id("document:" + key.asset.value());
40 if (open_.contains(id)) return eve::editing::applied<DocumentSnapshot>(open_.at(id).snapshot);
41
42 OpenDocument document;
43 document.snapshot.id = id;
44 document.snapshot.key = std::move(key);
45 document.snapshot.title = std::move(title);
46 document.snapshot.resourceUri = std::move(resourceUri);
47 document.snapshot.state = DocumentState::Ready;
48 EditorResult<StoredDocument> stored = store_->read(document.snapshot.resourceUri);
49 if (stored.ok()) {
50 document.content = stored.value().content;
51 document.snapshot.revision.disk = stored.value().revision;
52 document.snapshot.revision.edit = stored.value().revision;
53 document.snapshot.revision.saved = stored.value().revision;
54 document.snapshot.diskContentHash = stored.value().contentHash;
55 } else if (stored.code() == EditorStatus::NotFound) {
56 document.content = std::move(initialContent);
57 } else {
58 return error(stored.code(), "editor.document.store-read", "Document store could not be read");
59 }
60 DocumentSnapshot snapshot = document.snapshot;
61 open_.emplace(id, std::move(document));
62 return eve::editing::applied<DocumentSnapshot>(std::move(snapshot));
63}

93EditorResult<DocumentSnapshot> DocumentService::executeSave(const SaveTicket& ticket) {
94 auto pending = pendingSaves_.find(ticket.id);
95 auto document = open_.find(ticket.document);
96 if (pending == pendingSaves_.end() || document == open_.end())
97 return error(EditorStatus::NotFound, "editor.document.save-ticket-not-found",
98 "Save ticket or document is no longer available");
99 document->second.snapshot.state = DocumentState::Saving;
100 EditorResult<StoredDocument> stored =
101 store_->compareAndSwap(document->second.snapshot.resourceUri, ticket.expectedDiskRevision,
102 ticket.expectedContentHash, pending->second.content);
103 pendingSaves_.erase(pending);
104 if (!stored.ok()) {
105 document->second.snapshot.state =
106 stored.code() == EditorStatus::Conflict ? DocumentState::Conflict : DocumentState::Failed;
107 document->second.snapshot.diagnostics = stored.diagnostics();
108 return EditorResult<DocumentSnapshot>::failure(stored.status());
109 }
110 document->second.snapshot.revision.disk = stored.value().revision;
111 document->second.snapshot.revision.saved = ticket.capturedEditRevision;
112 document->second.snapshot.diskContentHash = stored.value().contentHash;
113 document->second.snapshot.state = DocumentState::Ready;
114 document->second.snapshot.diagnostics.clear();
115 return eve::editing::applied<DocumentSnapshot>(document->second.snapshot);
116}

144EditorResult<DocumentSnapshot> DocumentService::reconcileExternal(const DocumentId& document) {
145 auto found = open_.find(document);
146 if (found == open_.end()) return error(EditorStatus::NotFound, "editor.document.not-open", "Document is not open");
147
148 EditorResult<StoredDocument> stored = store_->read(found->second.snapshot.resourceUri);
149 if (!stored.ok()) {
150 const bool expectedMissing = stored.code() == EditorStatus::NotFound &&
151 found->second.snapshot.revision.disk == 0 &&
152 found->second.snapshot.diskContentHash.empty();
153 if (expectedMissing) return eve::editing::applied<DocumentSnapshot>(found->second.snapshot);
154 found->second.snapshot.state = stored.code() == EditorStatus::NotFound ||
155 stored.code() == EditorStatus::Conflict
156 ? DocumentState::Conflict
157 : DocumentState::Failed;
158 found->second.snapshot.diagnostics = stored.diagnostics();
159 if (found->second.snapshot.diagnostics.empty())
160 found->second.snapshot.diagnostics.push_back(eve::editing::ruleDiagnostic(
161 eve::DiagnosticCode::NotFound, RuleId("editor.document.external-missing"),
162 DiagnosticSeverity::Error, "Persisted document was removed externally"));
163 const EditorStatus status = found->second.snapshot.state == DocumentState::Conflict
164 ? EditorStatus::Conflict
165 : stored.code();
166 return EditorResult<DocumentSnapshot>::failure(eve::Status(status, found->second.snapshot.diagnostics));
167 }
168
169 const bool changed = stored.value().revision != found->second.snapshot.revision.disk ||
170 stored.value().contentHash != found->second.snapshot.diskContentHash;
171 if (!changed) {
172 if (found->second.snapshot.state == DocumentState::Conflict) {
173 found->second.snapshot.state = DocumentState::Ready;
174 found->second.snapshot.diagnostics.clear();
175 }
176 return eve::editing::applied<DocumentSnapshot>(found->second.snapshot);
177 }
178 if (found->second.snapshot.dirty()) {
179 found->second.snapshot.state = DocumentState::Conflict;
180 found->second.snapshot.diagnostics = {eve::editing::ruleDiagnostic(
181 eve::DiagnosticCode::Conflict, RuleId("editor.document.external-conflict"), DiagnosticSeverity::Error,
182 "Document changed on disk while the session has unsaved edits")};
183 return EditorResult<DocumentSnapshot>::failure(
184 eve::Status(EditorStatus::Conflict, found->second.snapshot.diagnostics));
185 }
186
187 found->second.content = stored.value().content;
188 const Revision synchronizedEdit =
189 std::max(found->second.snapshot.revision.edit + 1, stored.value().revision);
190 found->second.snapshot.revision.edit = synchronizedEdit;
191 found->second.snapshot.revision.saved = synchronizedEdit;
192 found->second.snapshot.revision.disk = stored.value().revision;
193 found->second.snapshot.diskContentHash = stored.value().contentHash;
194 found->second.snapshot.state = DocumentState::Ready;
195 found->second.snapshot.diagnostics.clear();
196 return eve::editing::applied<DocumentSnapshot>(found->second.snapshot);
197}
