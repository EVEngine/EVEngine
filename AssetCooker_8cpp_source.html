32void appendString(std::vector<std::uint8_t>& bytes, std::string_view text) {
33 const std::uint64_t size = text.size();
34 for (unsigned shift = 0; shift != 64; shift += 8) bytes.push_back(static_cast<std::uint8_t>(size >> shift));
35 bytes.insert(bytes.end(), text.begin(), text.end());
36}

38PersistentId buildIdentity(const EvaArchive& source, const AssetCookProfile& profile,
39 std::string_view canonicalManifest) {
40 std::vector<std::uint8_t> key;
41 appendString(key, "eve.asset-cooker/1");
42 appendString(key, evpackCodecBuildIdentity(profile.chunkCodec));
43 const auto packageBytes = source.manifest.packageId.bytes();
44 key.insert(key.end(), packageBytes.begin(), packageBytes.end());
45 appendString(key, canonicalManifest);
46 appendString(key, profile.variant.os); appendString(key, profile.variant.arch);
47 appendString(key, profile.variant.graphics); appendString(key, profile.variant.shaderFormat);
48 appendString(key, profile.variant.quality);
49 for (const auto& family : profile.variant.textureFamilies) appendString(key, family);
50 for (const auto& feature : profile.variant.features) appendString(key, feature);
51 for (unsigned shift = 0; shift != 32; shift += 8)
52 key.push_back(static_cast<std::uint8_t>(profile.bulkAlignment >> shift));
53 for (const auto& entry : source.entries) {
54 appendString(key, entry.path);
55 const auto hash = sha256(entry.bytes);
56 key.insert(key.end(), hash.begin(), hash.end());
57 }
58 auto digest = sha256(key);
59 PersistentId::Bytes bytes{};
60 std::copy_n(digest.begin(), bytes.size(), bytes.begin());
61 bytes[6] = static_cast<std::uint8_t>((bytes[6] & 0x0f) | 0x50);
62 bytes[8] = static_cast<std::uint8_t>((bytes[8] & 0x3f) | 0x80);
63 return PersistentId(bytes);
64}

104Result<std::vector<std::uint8_t>> binaryDefinition(
105 std::span<const std::uint8_t> json, std::uint64_t maximumBytes,
106 std::span<const EvaDependency> dependencies = {}) {
107 auto parsed = Value::fromJson(std::string_view(
108 reinterpret_cast<const char*>(json.data()), json.size()));
109 if (!parsed) return Result<std::vector<std::uint8_t>>::failure(parsed.status());
110 auto* object = parsed.value().getIf<Value::Object>();
111 if (!object)
112 return cookFailure<std::vector<std::uint8_t>>(
113 DiagnosticCode::ParseError, "runtime definition root must be an object");
114 Value::Array fallbacks;
115 for (const auto& dependency : dependencies) {
116 if (dependency.kind != EvaDependencyKind::RuntimeOptional) continue;
117 Value::Object fallback;
118 fallback["path"] = Value(dependency.path);
119 fallback["to"] = Value(dependency.to.format());
120 fallback["expectedType"] = Value(dependency.expectedType);
121 fallback["policy"] = Value(dependency.fallback);
122 fallbacks.emplace_back(std::move(fallback));
123 }
124 if (!fallbacks.empty()) (*object)["dependencyFallbacks"] = Value(std::move(fallbacks));
125 RuntimeDefinitionLimits limits;
126 limits.maximumBytes = maximumBytes;
127 return encodeRuntimeDefinition(parsed.value(), limits);
128}

153Result<AssetCookReceipt> cookEvaToEvpack(const EvaArchive& source, const AssetCookProfile& profile,
154 const EvaArchiveLimits& evaLimits,
155 const EvpackLimits& evpackLimits) {
156 auto migrated = migrateEvaArchive(source, evaLimits);
157 if (!migrated) return Result<AssetCookReceipt>::failure(migrated.status());
158 auto rebuilt = buildEvaArchive(migrated.value().manifest, migrated.value().entries, evaLimits);
159 if (!rebuilt) return Result<AssetCookReceipt>::failure(rebuilt.status());
160 auto admitted = parseEvaArchive(rebuilt.value(), evaLimits);
161 if (!admitted) return Result<AssetCookReceipt>::failure(admitted.status());
162 EvaArchive canonical = std::move(admitted).takeValue();
163 EvaManifest dependencyManifest = canonical.manifest;
164 std::erase_if(dependencyManifest.dependencies, [&](const EvaDependency& dependency) {
165 return dependency.kind == EvaDependencyKind::Platform &&
166 !platformDependencyApplies(dependency, profile.variant);
167 });
168 auto dependencyValidation = validateEvaDependencies(dependencyManifest,
169 profile.availableDependencies);
170 if (!dependencyValidation)
171 return Result<AssetCookReceipt>::failure(dependencyValidation.status());
172 auto canonicalManifest = serializeEvaManifest(canonical.manifest);
173 if (!canonicalManifest) return Result<AssetCookReceipt>::failure(canonicalManifest.status());
174 auto publication = validatePublication(canonical.manifest, profile.publication);
175 if (!publication) return Result<AssetCookReceipt>::failure(publication.status());
176
177 std::map<PersistentId, std::vector<PersistentId>> runtimeDependencies;
178 for (const auto& dependency : canonical.manifest.dependencies) {
179 if (dependency.kind == EvaDependencyKind::RuntimeOptional ||
180 dependency.kind == EvaDependencyKind::Build || dependency.kind == EvaDependencyKind::Editor ||
181 dependency.kind == EvaDependencyKind::Source || !platformDependencyApplies(dependency, profile.variant))
182 continue;
183 const bool present = std::binary_search(dependencyValidation.value().presentAssets.begin(),
184 dependencyValidation.value().presentAssets.end(),
185 dependency.to.id());
186 if (present) runtimeDependencies[dependency.from.id()].push_back(dependency.to.id());
187 }
188 for (auto& [asset, dependencies] : runtimeDependencies) {
189 (void)asset;
190 std::sort(dependencies.begin(), dependencies.end());
191 dependencies.erase(std::unique(dependencies.begin(), dependencies.end()), dependencies.end());
192 }
193
194 EvpackBuild build;
195 build.packageId = canonical.manifest.packageId;
196 build.buildId = buildIdentity(canonical, profile, canonicalManifest.value());
197 build.variants.push_back(profile.variant);
198 for (const auto& asset : canonical.manifest.assets) {
199 std::vector<EvaDependency> assetDependencyPolicies;
200 for (const auto& dependency : canonical.manifest.dependencies)
201 if (dependency.from.id() == asset.asset.id() &&
202 platformDependencyApplies(dependency, profile.variant))
203 assetDependencyPolicies.push_back(dependency);
204 const auto definition = std::lower_bound(canonical.entries.begin(), canonical.entries.end(), asset.definition,
205 [](const EvaArchiveEntry& entry, std::string_view path) {
206 return entry.path < path;
207 });
208 if (definition == canonical.entries.end() || definition->path != asset.definition)
209 return cookFailure<AssetCookReceipt>(DiagnosticCode::NotFound, "asset definition is missing",
210 asset.definition);
211 const auto dependencies = runtimeDependencies[asset.asset.id()];
212 if (asset.type == "eve.image") {
213 if (std::find(profile.variant.textureFamilies.begin(),
214 profile.variant.textureFamilies.end(), "rgba8") ==
215 profile.variant.textureFamilies.end())
216 return cookFailure<AssetCookReceipt>(
217 DiagnosticCode::Unsupported,
218 "eve.image/2 requires an explicit rgba8 target texture family", asset.definition);
219 auto parsedDefinition = Value::fromJson(std::string_view(
220 reinterpret_cast<const char*>(definition->bytes.data()), definition->bytes.size()));
221 if (!parsedDefinition)
222 return Result<AssetCookReceipt>::failure(parsedDefinition.status());
223 const auto* definitionObject = parsedDefinition.value().getIf<Value::Object>();
224 const Value* blobValue = definitionObject ? field(*definitionObject, "blob") : nullptr;
225 if (!blobValue || !blobValue->isString())
226 return cookFailure<AssetCookReceipt>(DiagnosticCode::ParseError,
227 "eve.image/2 source blob path is missing",
228 asset.definition);
229 const auto sourceBlob = std::lower_bound(
230 canonical.entries.begin(), canonical.entries.end(), blobValue->asString(),
231 [](const EvaArchiveEntry& entry, std::string_view path) { return entry.path < path; });
232 if (sourceBlob == canonical.entries.end() || sourceBlob->path != blobValue->asString())
233 return cookFailure<AssetCookReceipt>(DiagnosticCode::NotFound,
234 "eve.image/2 source blob is missing",
235 blobValue->asString());
236 auto image = cookCanonicalImageRgba8(definition->bytes, sourceBlob->bytes,
237 evpackLimits.maximumChunkBytes);
238 if (!image) return Result<AssetCookReceipt>::failure(image.status());
239 auto runtimeDefinition = binaryDefinition(image.value().definition,
240 evpackLimits.maximumChunkBytes,
241 assetDependencyPolicies);
242 if (!runtimeDefinition)
243 return Result<AssetCookReceipt>::failure(runtimeDefinition.status());
244 build.chunks.push_back({asset.asset.id(), asset.type, asset.schemaVersion, 0,
245 EvpackChunkKind::Definition, 0, profile.chunkCodec, 8,
246 dependencies, std::move(runtimeDefinition).takeValue()});
247 build.chunks.push_back({asset.asset.id(), asset.type, asset.schemaVersion, 0,
248 EvpackChunkKind::Bulk, 1, EvpackCodec::None,
249 profile.bulkAlignment, dependencies,
250 std::move(image).takeValue().bulk});
251 continue;
252 }
253 if (asset.type == "eve.pcg-graph") {
254 auto pcg = cookCanonicalPcgGraph(definition->bytes, 4096,
255 evpackLimits.maximumChunkBytes);
256 if (!pcg) return Result<AssetCookReceipt>::failure(pcg.status());
257 auto runtimeDefinition = binaryDefinition(pcg.value().definition,
258 evpackLimits.maximumChunkBytes,
259 assetDependencyPolicies);
260 if (!runtimeDefinition)
261 return Result<AssetCookReceipt>::failure(runtimeDefinition.status());
262 build.chunks.push_back({asset.asset.id(), asset.type, asset.schemaVersion, 0,
263 EvpackChunkKind::Definition, 0, profile.chunkCodec, 8,
264 dependencies, std::move(runtimeDefinition).takeValue()});
265 build.chunks.push_back({asset.asset.id(), asset.type, asset.schemaVersion, 0,
266 EvpackChunkKind::Bulk, 1, profile.chunkCodec,
267 profile.bulkAlignment, dependencies,
268 std::move(pcg).takeValue().executionPlan});
269 continue;
270 }
271 auto runtimeDefinition = binaryDefinition(definition->bytes,
272 evpackLimits.maximumChunkBytes,
273 assetDependencyPolicies);
274 if (!runtimeDefinition)
275 return Result<AssetCookReceipt>::failure(runtimeDefinition.status());
276 build.chunks.push_back({asset.asset.id(), asset.type, asset.schemaVersion, 0,
277 EvpackChunkKind::Definition, 0, profile.chunkCodec, 8,
278 dependencies, std::move(runtimeDefinition).takeValue()});
279 const std::string prefix = "assets/" + asset.asset.id().format() + "/";
280 std::uint32_t chunkId = 1;
281 for (const auto& entry : canonical.entries) {
282 if (entry.path == asset.definition || !entry.path.starts_with(prefix)) continue;
283 build.chunks.push_back({asset.asset.id(), asset.type, asset.schemaVersion, 0,
284 kindForPath(entry.path), chunkId++, profile.chunkCodec,
285 profile.bulkAlignment, dependencies, entry.bytes});
286 }
287 }
288 const std::uint32_t chunkCount = static_cast<std::uint32_t>(build.chunks.size());
289 const PersistentId buildId = build.buildId;
290 auto packageBytes = buildEvpack(std::move(build), evpackLimits);
291 if (!packageBytes) return Result<AssetCookReceipt>::failure(packageBytes.status());
292 auto verified = parseEvpack(packageBytes.value(), evpackLimits);
293 if (!verified) return Result<AssetCookReceipt>::failure(verified.status());
294 return Result<AssetCookReceipt>::success(
295 {canonical.manifest.packageId, buildId, chunkCount, std::move(packageBytes).takeValue()});
296}
