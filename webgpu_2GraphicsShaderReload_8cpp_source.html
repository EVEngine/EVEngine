27Result<void> Graphics::replaceShaderFromWgsl(Shader &shader, const std::string &vertWgsl,
28 const std::string &fragWgsl) {
29 auto shaderIt = std::find_if(ownedShaders.begin(), ownedShaders.end(),
30 [&](const std::unique_ptr<Shader> &owned) {
31 return owned.get() == &shader;
32 });
33 if (shaderIt == ownedShaders.end() || !shader.gpuHandle)
34 return reloadFailure(DiagnosticCode::StaleHandle,
35 "shader is not a live resource owned by this Graphics instance",
36 "shader");
37 if (fragWgsl.empty())
38 return reloadFailure(DiagnosticCode::InvalidArgument,
39 "fragment WGSL must not be empty", "fragWgsl");
40
41 auto *current = static_cast<GpuShader *>(shader.gpuHandle);
42 auto gpuIt = std::find_if(ownedGpuShaders.begin(), ownedGpuShaders.end(),
43 [&](const std::unique_ptr<GpuShader> &owned) {
44 return owned.get() == current;
45 });
46 if (gpuIt == ownedGpuShaders.end())
47 return reloadFailure(DiagnosticCode::StaleHandle,
48 "shader GPU resource is no longer owned by this Graphics instance",
49 "shader.gpuHandle");
50
51 GpuShader candidate;
52 candidate.isMesh3D = current->isMesh3D;
53 candidate.isHair3D = current->isHair3D;
54 candidate.pipelineLayout = current->pipelineLayout;
55 candidate.setLayout = current->setLayout;
56 candidate.wgslVert = vertWgsl.empty() ? current->wgslVert : vertWgsl;
57 candidate.wgslFrag = fragWgsl;
58 if (candidate.wgslVert.empty())
59 return reloadFailure(DiagnosticCode::PreconditionViolation,
60 "shader has no reusable default vertex WGSL", "vertWgsl");
61
62 try {
63 if (!candidate.isMesh3D) {
64 candidate.swapchainPipeline = createPipelineForShader(
65 &candidate, wgpu::TextureFormat(surfaceFormat), false, false, false, false,
66 false, tex2DPipelineLayout);
67 candidate.offscreenPipeline = createPipelineForShader(
68 &candidate, wgpu::TextureFormat::RGBA8Unorm, false, false, false, false,
69 false, tex2DPipelineLayout);
70 } else {
71 candidate.mesh3dPipeline = createPipelineForShader(
72 &candidate, static_cast<wgpu::TextureFormat>(sceneColorFormat), true, true,
73 candidate.isHair3D, false, false, mesh3dPipelineLayout);
74 if (!candidate.isHair3D)
75 candidate.mesh3dXrayPipeline = createPipelineForShader(
76 &candidate, static_cast<wgpu::TextureFormat>(sceneColorFormat), false, true,
77 false, false, false, mesh3dPipelineLayout);
78 }
79 } catch (const std::exception &error) {
80 return reloadFailure(DiagnosticCode::Failed,
81 std::string("failed to prepare replacement pipeline: ") + error.what(),
82 "pipeline");
83 } catch (...) {
84 return reloadFailure(DiagnosticCode::Failed,
85 "failed to prepare replacement pipeline", "pipeline");
86 }
87
88 current->swapchainPipeline = std::move(candidate.swapchainPipeline);
89 current->offscreenPipeline = std::move(candidate.offscreenPipeline);
90 current->mesh3dPipeline = std::move(candidate.mesh3dPipeline);
91 current->mesh3dXrayPipeline = std::move(candidate.mesh3dXrayPipeline);
92 current->wgslVert = std::move(candidate.wgslVert);
93 current->wgslFrag = std::move(candidate.wgslFrag);
94 return Result<void>::success();
95}
