54Result<OceanBoatReceipt> OceanBoat::apply(const OceanWaterView& frame, double dt, double throttle, double rudder) {
55 if (!std::isfinite(dt) || dt <= 0.0 || dt > 1.0 / 30.0 || !std::isfinite(throttle) || std::abs(throttle) > 1.0 ||
56 !std::isfinite(rudder) || std::abs(rudder) > 1.0 || frame.time() <= impl_->lastTime)
57 return failure<OceanBoatReceipt>(DiagnosticCode::InvalidArgument, "Invalid or repeated boat step");
58 auto* world = physics::World3D::findWorld(impl_->link.world);
59 if (!world) return failure<OceanBoatReceipt>(DiagnosticCode::StaleHandle, "Boat world no longer exists");
60 auto resolved = impl_->link.resolve(*world);
61 if (!resolved) return Result<OceanBoatReceipt>::failure(resolved.status());
62 auto& body = *resolved.value();
63 const auto& h = impl_->hull;
64 const glm::dquat rotation{body.getRotW(), body.getRotX(), body.getRotY(), body.getRotZ()};
65 const auto inverse = glm::conjugate(rotation);
66 const double mass = body.getMass();
67 if (!std::isfinite(mass) || mass <= 0.0 || body.getType() != "dynamic")
68 return failure<OceanBoatReceipt>(DiagnosticCode::InvalidArgument, "Boat body is no longer dynamic");
69 OceanBoatReceipt receipt;
70 receipt.time = frame.time();
71 std::vector<OceanCellForce> forces;
72 forces.reserve(h.cells.size() + 1);
73 receipt.contacts.reserve(h.cells.size());
74 for (const auto& cell : h.cells) {
75 auto p = body.localToWorldPointOwned(float(cell.center.x), float(cell.center.y), float(cell.center.z));
76 if (!p) return Result<OceanBoatReceipt>::failure(p.status());
77 auto v = body.getLocalPointVelocityOwned(float(cell.center.x), float(cell.center.y), float(cell.center.z));
78 if (!v) return Result<OceanBoatReceipt>::failure(v.status());
79 const auto position = vector(p.value()), velocity = vector(v.value());
80 auto water = frame.sample({position.x, position.z});
81 if (!water) return Result<OceanBoatReceipt>::failure(water.status());
82 OceanBuoyancyCell probe{position, velocity, cell.radius, cell.volume, 0.0};
83 auto force = evaluateOceanCellForce(probe, water.value(), h.waterDensity, h.gravity);
84 if (!force) return Result<OceanBoatReceipt>::failure(force.status());
85 const auto relative = velocity - water.value().velocity;
86 const auto local = inverse * relative;
87 const double fraction = force.value().submergedVolume / cell.volume;
88 const double area = std::numbers::pi * cell.radius * cell.radius * fraction;
89 auto drag = -0.5 * h.waterDensity * area * h.drag * glm::abs(local) * local;
90 // Cap each cell's dissipative impulse to its share of body momentum.
91 for (int axis = 0; axis < 3; ++axis) {
92 const double bound = mass * std::abs(local[axis]) / (dt * double(h.cells.size()));
93 drag[axis] = std::clamp(drag[axis], -bound, bound);
94 }
95 force.value().force += rotation * drag;
96 receipt.submergedVolume += force.value().submergedVolume;
97 receipt.contacts.push_back({position, relative, force.value().submergedVolume});
98 forces.push_back(force.value());
99 }
100 auto propeller = body.localToWorldPointOwned(float(h.propeller.x), float(h.propeller.y), float(h.propeller.z));
101 if (!propeller) return Result<OceanBoatReceipt>::failure(propeller.status());
102 const auto p = vector(propeller.value());
103 auto water = frame.sample({p.x, p.z});
104 if (!water) return Result<OceanBoatReceipt>::failure(water.status());
105 auto propVelocity =
106 body.getLocalPointVelocityOwned(float(h.propeller.x), float(h.propeller.y), float(h.propeller.z));
107 if (!propVelocity) return Result<OceanBoatReceipt>::failure(propVelocity.status());
108 const auto relative = inverse * (vector(propVelocity.value()) - water.value().velocity);
109 const double wet = std::clamp((water.value().height - p.y) / 0.3, 0.0, 1.0);
110 const double rudderForce =
111 std::clamp(-rudder * h.rudderCoefficient * relative.z * std::abs(relative.z), -mass * 10.0, mass * 10.0);
112 forces.push_back({p, rotation * (glm::dvec3(rudderForce, 0.0, throttle * h.thrust) * wet), 0.0});
113 for (const auto& force : forces) {
114 if (!finite(force.force) || glm::length(force.force) > 1e12)
115 return failure<OceanBoatReceipt>(DiagnosticCode::InvalidArgument, "Hull force exceeds supported range");
116 receipt.totalForce += force.force;
117 }
118 // No fallible operations or external callbacks after the first force is applied.
119 for (const auto& f : forces)
120 body.applyForceAt(float(f.force.x), float(f.force.y), float(f.force.z), float(f.position.x),
121 float(f.position.y), float(f.position.z));
122 impl_->lastTime = frame.time();
123 return Result<OceanBoatReceipt>::success(std::move(receipt));
124}
