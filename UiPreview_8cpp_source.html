10UiPreviewSnapshot UiDocumentPreviewService::build(const UiDocumentTarget& document, double width, double height) const {
11 UiPreviewSnapshot result;
12 result.documentRevision = document.revision();
13 result.viewportWidth = width;
14 result.viewportHeight = height;
15 if (!std::isfinite(width) || !std::isfinite(height) || width <= 0.0 || height <= 0.0) {
16 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
17 eve::DiagnosticCode::InvalidArgument, RuleId("editor.ui.invalid-preview-viewport"),
18 DiagnosticSeverity::Error, "UI preview viewport must have finite positive dimensions"));
19 return result;
20 }
21 result.diagnostics = document.validate();
22 std::map<ObjectId, UiPreviewWidget> computed;
23 std::function<void(const ObjectId&, const UiPreviewWidget&, int)> visit;
24 visit = [&](const ObjectId& id, const UiPreviewWidget& parent, int depth) {
25 auto value = document.widget(id);
26 if (!value.ok()) return;
27 const UiWidgetSnapshot& widget = value.value();
28 const double contentX = parent.x + widget.style.marginLeft + (depth == 0 ? 0.0 : 0.0);
29 const double contentY = parent.y + widget.style.marginTop;
30 const double parentWidth = std::max(0.0, parent.width - widget.style.marginLeft - widget.style.marginRight);
31 const double parentHeight = std::max(0.0, parent.height - widget.style.marginTop - widget.style.marginBottom);
32 UiPreviewWidget preview;
33 preview.id = widget.id;
34 preview.parent = widget.parent;
35 preview.depth = depth;
36 preview.width = widget.layout.width;
37 preview.height = widget.layout.height;
38 preview.x =
39 contentX + parentWidth * widget.layout.anchorX + widget.layout.x - preview.width * widget.layout.pivotX;
40 preview.y =
41 contentY + parentHeight * widget.layout.anchorY + widget.layout.y - preview.height * widget.layout.pivotY;
42 preview.visible = widget.visible && parent.visible;
43 computed[id] = preview;
44 result.widgets.push_back(preview);
45 UiPreviewWidget childParent = preview;
46 childParent.x += widget.style.paddingLeft;
47 childParent.y += widget.style.paddingTop;
48 childParent.width = std::max(0.0, childParent.width - widget.style.paddingLeft - widget.style.paddingRight);
49 childParent.height = std::max(0.0, childParent.height - widget.style.paddingTop - widget.style.paddingBottom);
50 for (const ObjectId& child : document.children(id)) visit(child, childParent, depth + 1);
51 };
52 const UiPreviewWidget viewport{ObjectId(), ObjectId(), 0.0, 0.0, width, height, -1, true};
53 for (const ObjectId& root : document.children(ObjectId())) visit(root, viewport, 0);
54 const bool hasError = std::any_of(
55 result.diagnostics.begin(), result.diagnostics.end(),
56 [](const EditorDiagnostic& diagnostic) { return diagnostic.severity() == DiagnosticSeverity::Error; });
57 result.status = hasError ? EditorStatus::Failed : EditorStatus::Applied;
58 return result;
59}

61EditorResult<ObjectId> UiDocumentPreviewService::pick(const UiPreviewSnapshot& preview, double x, double y) const {
62 if (!std::isfinite(x) || !std::isfinite(y))
63 return eve::editing::failed<ObjectId>(EditorStatus::Rejected, RuleId("editor.ui.invalid-pick-point"),
64 "UI pick coordinates must be finite");
65 const UiPreviewWidget* best = nullptr;
66 for (const UiPreviewWidget& widget : preview.widgets) {
67 if (!widget.visible || x < widget.x || y < widget.y || x > widget.x + widget.width ||
68 y > widget.y + widget.height)
69 continue;
70 if (!best || widget.depth > best->depth || (widget.depth == best->depth && widget.id > best->id))
71 best = &widget;
72 }
73 if (!best)
74 return eve::editing::failed<ObjectId>(EditorStatus::NotFound, RuleId("editor.ui.pick-miss"),
75 "No visible widget contains the preview point");
76 return eve::editing::applied<ObjectId>(best->id);
77}
