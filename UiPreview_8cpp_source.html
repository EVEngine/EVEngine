15UiPreviewSnapshot UiDocumentPreviewService::build(const UiDocumentTarget& document, double width, double height) const {
16 UiPreviewSnapshot result;
17 result.documentRevision = document.revision();
18 result.viewportWidth = width;
19 result.viewportHeight = height;
20 if (!std::isfinite(width) || !std::isfinite(height) || width <= 0.0 || height <= 0.0) {
21 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
22 eve::DiagnosticCode::InvalidArgument, RuleId("editor.ui.invalid-preview-viewport"),
23 DiagnosticSeverity::Error, "UI preview viewport must have finite positive dimensions"));
24 return result;
25 }
26 result.diagnostics = document.validate();
27 std::map<ObjectId, UiPreviewWidget> computed;
28 std::function<void(const ObjectId&, const UiPreviewWidget&, int)> visit;
29 visit = [&](const ObjectId& id, const UiPreviewWidget& parent, int depth) {
30 auto value = document.widget(id);
31 if (!value.ok()) return;
32 const UiWidgetSnapshot& widget = value.value();
33 const double contentX = parent.x + widget.style.marginLeft + (depth == 0 ? 0.0 : 0.0);
34 const double contentY = parent.y + widget.style.marginTop;
35 const double parentWidth = std::max(0.0, parent.width - widget.style.marginLeft - widget.style.marginRight);
36 const double parentHeight = std::max(0.0, parent.height - widget.style.marginTop - widget.style.marginBottom);
37 UiPreviewWidget preview;
38 preview.id = widget.id;
39 preview.parent = widget.parent;
40 preview.depth = depth;
41 preview.width = widget.layout.width;
42 preview.height = widget.layout.height;
43 preview.x =
44 contentX + parentWidth * widget.layout.anchorX + widget.layout.x - preview.width * widget.layout.pivotX;
45 preview.y =
46 contentY + parentHeight * widget.layout.anchorY + widget.layout.y - preview.height * widget.layout.pivotY;
47 preview.visible = widget.visible && parent.visible;
48 computed[id] = preview;
49 result.widgets.push_back(preview);
50 UiPreviewWidget childParent = preview;
51 childParent.x += widget.style.paddingLeft;
52 childParent.y += widget.style.paddingTop;
53 childParent.width = std::max(0.0, childParent.width - widget.style.paddingLeft - widget.style.paddingRight);
54 childParent.height = std::max(0.0, childParent.height - widget.style.paddingTop - widget.style.paddingBottom);
55 for (const ObjectId& child : document.children(id)) visit(child, childParent, depth + 1);
56 };
57 const UiPreviewWidget viewport{ObjectId(), ObjectId(), 0.0, 0.0, width, height, -1, true};
58 for (const ObjectId& root : document.children(ObjectId())) visit(root, viewport, 0);
59 const bool hasError = std::any_of(
60 result.diagnostics.begin(), result.diagnostics.end(),
61 [](const EditorDiagnostic& diagnostic) { return diagnostic.severity() == DiagnosticSeverity::Error; });
62 result.status = hasError ? EditorStatus::Failed : EditorStatus::Applied;
63 return result;
64}

66Result<ObjectId> UiDocumentPreviewService::pick(const UiPreviewSnapshot& preview, double x, double y) const {
67 if (!std::isfinite(x) || !std::isfinite(y))
68 return eve::editing::failed<ObjectId>(EditorStatus::Rejected, RuleId("editor.ui.invalid-pick-point"),
69 "UI pick coordinates must be finite");
70 const UiPreviewWidget* best = nullptr;
71 for (const UiPreviewWidget& widget : preview.widgets) {
72 if (!widget.visible || x < widget.x || y < widget.y || x > widget.x + widget.width ||
73 y > widget.y + widget.height)
74 continue;
75 if (!best || widget.depth > best->depth || (widget.depth == best->depth && widget.id > best->id))
76 best = &widget;
77 }
78 if (!best)
79 return eve::editing::failed<ObjectId>(EditorStatus::NotFound, RuleId("editor.ui.pick-miss"),
80 "No visible widget contains the preview point");
81 return eve::editing::applied<ObjectId>(best->id);
82}

84Result<EditorValue> UiDocumentPreviewService::anchorGizmo(const UiDocumentTarget& document,
85 const UiPreviewSnapshot& preview, const ObjectId& id) const {
86 if (preview.documentRevision != document.revision())
87 return eve::editing::failed<EditorValue>(EditorStatus::Conflict, RuleId("editor.ui.stale-preview"),
88 "UI preview revision no longer matches the document");
89 auto widget = document.widget(id);
90 if (!widget.ok())
91 return eve::editing::failed<EditorValue>(EditorStatus::NotFound, RuleId("editor.ui.widget-not-found"),
92 "UI widget does not exist");
93 double parentX = 0.0, parentY = 0.0, parentW = preview.viewportWidth, parentH = preview.viewportHeight;
94 if (!widget.value().parent.empty()) {
95 const auto parent =
96 std::find_if(preview.widgets.begin(), preview.widgets.end(),
97 [&](const UiPreviewWidget& value) { return value.id == widget.value().parent; });
98 if (parent == preview.widgets.end())
99 return eve::editing::failed<EditorValue>(EditorStatus::NotFound,
100 RuleId("editor.ui.preview-parent-not-found"),
101 "Parent is absent from UI preview");
102 auto parentWidget = document.widget(parent->id);
103 parentX = parent->x + parentWidget.value().style.paddingLeft;
104 parentY = parent->y + parentWidget.value().style.paddingTop;
105 parentW = std::max(
106 0.0, parent->width - parentWidget.value().style.paddingLeft - parentWidget.value().style.paddingRight);
107 parentH = std::max(
108 0.0, parent->height - parentWidget.value().style.paddingTop - parentWidget.value().style.paddingBottom);
109 }
110 const auto previewWidget = std::find_if(preview.widgets.begin(), preview.widgets.end(),
111 [&](const UiPreviewWidget& value) { return value.id == id; });
112 if (previewWidget == preview.widgets.end())
113 return eve::editing::failed<EditorValue>(EditorStatus::NotFound, RuleId("editor.ui.preview-widget-not-found"),
114 "Widget is absent from UI preview");
115 const double anchorX = parentX + parentW * widget.value().layout.anchorX;
116 const double anchorY = parentY + parentH * widget.value().layout.anchorY;
117 const double pivotX = previewWidget->x + previewWidget->width * widget.value().layout.pivotX;
118 const double pivotY = previewWidget->y + previewWidget->height * widget.value().layout.pivotY;
119 return eve::editing::applied<EditorValue>(
120 EditorValue::Object{{"anchor", EditorValue::Array{anchorX, anchorY}},
121 {"pivot", EditorValue::Array{pivotX, pivotY}},
122 {"line", EditorValue::Array{anchorX, anchorY, pivotX, pivotY}}});
123}

139 Revision expectedRevision) const {

141 snapshot.documentRevision = catalog.revision();

156 snapshot.theme = asset.value().tokens;

158 snapshot.status = EditorStatus::Applied;
159 return snapshot;

298const EditorValue* field(const EditorValue& value, const char* key) {
299 const auto* object = value.getIf<EditorValue::Object>();
300 if (!object) return nullptr;
301 const auto found = object->find(key);
302 return found == object->end() ? nullptr : &found->second;
303}

305bool readColor(const EditorValue& value, float dest[4]) {
306 const auto* array = value.getIf<EditorValue::Array>();
307 if (!array || array->size() < 4) return false;
308 for (int i = 0; i < 4; ++i) {
309 const auto* number = (*array)[static_cast<std::size_t>(i)].getIf<double>();
310 if (!number || !std::isfinite(*number) || *number < 0.0 || *number > 1.0) return false;
311 dest[i] = static_cast<float>(*number);
312 }
313 return true;
314}

366EditorValue themeTokensValue(const ui::Theme& theme) {
367 EditorValue::Object object;
368 for (const ColorTok& token : kColors) object[token.path] = colorValue(theme.*(token.member));
369 for (const FloatTok& token : kFloats) object[token.path] = static_cast<double>(theme.*(token.member));
370 for (const LayoutTok& token : kLayout) object[token.path] = static_cast<double>(theme.layout.*(token.member));
371 object["nav.keyboard"] = theme.navEnableKeyboard;
372 object["nav.gamepad"] = theme.navEnableGamepad;
373 return EditorValue(std::move(object));

376Result<ui::Theme> parseThemeTokens(const EditorValue& value) {
377 if (!value.getIf<EditorValue::Object>())
378 return eve::editing::failed<ui::Theme>(EditorStatus::Rejected, RuleId("editor.ui-theme.tokens"),
379 "Theme tokens must be an object");
380 ui::Theme theme;
381 for (const ColorTok& token : kColors) {
382 const EditorValue* entry = field(value, token.path);
383 if (!entry || !readColor(*entry, theme.*(token.member)))
384 return eve::editing::failed<ui::Theme>(EditorStatus::Rejected, RuleId("editor.ui-theme.color"),
385 std::string("Theme color is missing or invalid: ") + token.path);
386 }
387 for (const FloatTok& token : kFloats) {
388 const EditorValue* entry = field(value, token.path);
389 const auto* number = entry ? entry->getIf<double>() : nullptr;
390 if (!number || !std::isfinite(*number))
391 return eve::editing::failed<ui::Theme>(EditorStatus::Rejected, RuleId("editor.ui-theme.float"),
392 std::string("Theme number is missing or invalid: ") + token.path);
393 theme.*(token.member) = static_cast<float>(*number);
394 }
395 for (const LayoutTok& token : kLayout) {
396 const EditorValue* entry = field(value, token.path);
397 const auto* number = entry ? entry->getIf<double>() : nullptr;
398 if (!number || !std::isfinite(*number))
399 return eve::editing::failed<ui::Theme>(
400 EditorStatus::Rejected, RuleId("editor.ui-theme.layout"),
401 std::string("Theme layout number is missing or invalid: ") + token.path);
402 theme.layout.*(token.member) = static_cast<float>(*number);
403 }
404 const EditorValue* keyboard = field(value, "nav.keyboard");
405 const EditorValue* gamepad = field(value, "nav.gamepad");
406 const auto* keyboardFlag = keyboard ? keyboard->getIf<bool>() : nullptr;
407 const auto* gamepadFlag = gamepad ? gamepad->getIf<bool>() : nullptr;
408 if (!keyboardFlag || !gamepadFlag)
409 return eve::editing::failed<ui::Theme>(EditorStatus::Rejected, RuleId("editor.ui-theme.nav"),
410 "Theme navigation flags are required");
411 theme.navEnableKeyboard = *keyboardFlag;
412 theme.navEnableGamepad = *gamepadFlag;
413 auto diagnostics = validateThemeTokens(theme);
414 if (!diagnostics.empty())
415 return eve::editing::failed<ui::Theme>(EditorStatus::Rejected, RuleId("editor.ui-theme.range"),
416 diagnostics.front().message());
417 return eve::editing::applied<ui::Theme>(theme);
418}

420Result<void> assignThemeToken(ui::Theme& theme, const PropertyPath& path, const EditorValue& value) {
421 for (const ColorTok& token : kColors) {
422 if (path != PropertyPath(token.path)) continue;
423 if (!readColor(value, theme.*(token.member)))
424 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.ui-theme.token-color"),
425 std::string("Theme color token is invalid: ") + token.path);
426 return eve::editing::applied<void>();
427 }
428 for (const FloatTok& token : kFloats) {
429 if (path != PropertyPath(token.path)) continue;
430 const auto* number = value.getIf<double>();
431 if (!number || !inRange(static_cast<float>(*number), token.minimum, token.maximum))
432 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.ui-theme.token-float"),
433 std::string("Theme number token is invalid: ") + token.path);
434 theme.*(token.member) = static_cast<float>(*number);
435 return eve::editing::applied<void>();
436 }
437 for (const LayoutTok& token : kLayout) {
438 if (path != PropertyPath(token.path)) continue;
439 const auto* number = value.getIf<double>();
440 if (!number || !inRange(static_cast<float>(*number), token.minimum, token.maximum))
441 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.ui-theme.token-layout"),
442 std::string("Theme layout token is invalid: ") + token.path);
443 theme.layout.*(token.member) = static_cast<float>(*number);
444 return eve::editing::applied<void>();
445 }
446 if (path == PropertyPath("nav.keyboard")) {
447 const auto* flag = value.getIf<bool>();
448 if (!flag)
449 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.ui-theme.token-nav"),
450 "Theme navigation token is invalid");
451 theme.navEnableKeyboard = *flag;
452 return eve::editing::applied<void>();
453 }
454 if (path == PropertyPath("nav.gamepad")) {
455 const auto* flag = value.getIf<bool>();
456 if (!flag)
457 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.ui-theme.token-nav"),
458 "Theme navigation token is invalid");
459 theme.navEnableGamepad = *flag;
460 return eve::editing::applied<void>();
461 }
462 return eve::editing::failed<void>(EditorStatus::NotFound, RuleId("editor.ui-theme.token"),
463 "Unknown theme token: " + path.value());

466EditorValue readThemeToken(const ui::Theme& theme, const PropertyPath& path) {
467 for (const ColorTok& token : kColors) {

512std::vector<EditorDiagnostic> validateThemeTokens(const ui::Theme& theme) {
513 std::vector<EditorDiagnostic> diagnostics;
514 for (const ColorTok& token : kColors) {
515 const float* color = theme.*(token.member);
516 for (int i = 0; i < 4; ++i) {
517 if (!inRange(color[i], 0.f, 1.f)) {
518 diagnostics.push_back(tokenDiagnostic(
519 "editor.ui-theme.color-range", std::string("Theme color channel is out of range: ") + token.path));
520 break;
521 }
522 }
523 }
524 for (const FloatTok& token : kFloats) {
525 if (!inRange(theme.*(token.member), token.minimum, token.maximum))
526 diagnostics.push_back(tokenDiagnostic("editor.ui-theme.float-range",
527 std::string("Theme number is out of range: ") + token.path));
528 }
529 for (const LayoutTok& token : kLayout) {
530 if (!inRange(theme.layout.*(token.member), token.minimum, token.maximum))
531 diagnostics.push_back(tokenDiagnostic("editor.ui-theme.layout-range",
532 std::string("Theme layout number is out of range: ") + token.path));
533 }
534 return diagnostics;
535}
