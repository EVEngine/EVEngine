17ssq::Table bakeArraysTable(HSQUIRRELVM vm, const MeshArrays& arrays) {
18 ssq::Array positions(vm);
19 for (float v : arrays.positions) positions.push(v);
20 ssq::Array normals(vm);
21 for (float v : arrays.normals) normals.push(v);
22 ssq::Array uvs(vm);
23 for (float v : arrays.uvs) uvs.push(v);
24 ssq::Array indices(vm);
25 for (std::uint32_t v : arrays.indices) indices.push(static_cast<std::int64_t>(v));
26 ssq::Table table(vm);
27 table.set("positions", positions);
28 table.set("normals", normals);
29 table.set("uvs", uvs);
30 table.set("indices", indices);
31 table.set("vertexCount", static_cast<std::int64_t>(arrays.vertexCount()));
32 table.set("indexCount", static_cast<std::int64_t>(arrays.indexCount()));
33 table.set("triangleCount", static_cast<std::int64_t>(arrays.triangleCount()));
34 return table;
35}
