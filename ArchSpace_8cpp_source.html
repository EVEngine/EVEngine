16ssq::Table bakeArraysTable(HSQUIRRELVM vm, const MeshArrays& arrays) {
17 ssq::Array positions(vm);
18 for (float v : arrays.positions) positions.push(v);
19 ssq::Array normals(vm);
20 for (float v : arrays.normals) normals.push(v);
21 ssq::Array uvs(vm);
22 for (float v : arrays.uvs) uvs.push(v);
23 ssq::Array indices(vm);
24 for (std::uint32_t v : arrays.indices) indices.push(static_cast<std::int64_t>(v));
25 ssq::Table table(vm);
26 table.set("positions", positions);
27 table.set("normals", normals);
28 table.set("uvs", uvs);
29 table.set("indices", indices);
30 table.set("vertexCount", static_cast<std::int64_t>(arrays.vertexCount()));
31 table.set("indexCount", static_cast<std::int64_t>(arrays.indexCount()));
32 table.set("triangleCount", static_cast<std::int64_t>(arrays.triangleCount()));
33 return table;
34}
