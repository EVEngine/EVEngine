25graphics::Texture* resolveAlbedo(const eve::stylize::AttackVfxLayerStartRequest& request) {
26 auto* graphics = eve::ModuleManager::getInstance<graphics::Graphics>("Graphics");
27 if (!graphics) return nullptr;
28
29 const std::string& uri = request.layer ? request.layer->uri : std::string{};
30 if (uri.rfind("solid:", 0) == 0 || uri.rfind("decal://", 0) == 0 || uri.empty()) {
31 const float r = layerParam(request, "r", 0.85f);
32 const float g = layerParam(request, "g", 0.25f);
33 const float b = layerParam(request, "b", 0.08f);
34 const float a = layerParam(request, "a", 0.85f);
35 const auto toByte = [](float v) -> std::uint8_t {
36 if (v <= 0.f) return 0;
37 if (v >= 1.f) return 255;
38 return static_cast<std::uint8_t>(v * 255.f + 0.5f);
39 };
40 const std::uint8_t rgba[4] = {toByte(r), toByte(g), toByte(b), toByte(a)};
41 return graphics->newTexture(1, 1, rgba);
42 }
43 try {
44 return graphics->newTextureFromFile(uri);
45 } catch (...) {
46 return nullptr;
47 }
48}

71 eve::Result<eve::stylize::AttackVfxLayerHandle> start(
72 const eve::stylize::AttackVfxLayerStartRequest& request) override {
73 if (!request.layer)
74 return eve::Result<eve::stylize::AttackVfxLayerHandle>::failure(eve::Diagnostic::error(
75 eve::DiagnosticCode::InvalidArgument, "AttackVfx decal layer is null", "layer"));
76 if (request.layer->uri.empty())
77 return eve::Result<eve::stylize::AttackVfxLayerHandle>::failure(eve::Diagnostic::error(
78 eve::DiagnosticCode::InvalidArgument, "AttackVfx decal layer uri is empty", "uri"));
79
80 graphics::Texture* albedo = resolveAlbedo(request);
81 if (!albedo)
82 return eve::Result<eve::stylize::AttackVfxLayerHandle>::failure(eve::Diagnostic::error(
83 eve::DiagnosticCode::NotFound, "decal albedo texture could not be resolved", "uri"));
84
85 // World placement comes from authored floatParams (or spatial adapters later).
86 // Entity ids must never be treated as coordinates.
87 const float x = layerParam(request, "x", 0.f);
88 const float y = layerParam(request, "y", 0.05f);
89 const float z = layerParam(request, "z", 0.f);
90 const float size = layerParam(request, "size", 1.2f);
91 const float depth = layerParam(request, "depth", 0.35f);
92 const float lifetime = layerParam(request, "lifetime", 1.5f);
93 const float fadeIn = layerParam(request, "fadeIn", 0.05f);
94 const float fadeOut = layerParam(request, "fadeOut", 0.35f);
95 const int seed =
96 request.playRequest
97 ? static_cast<int>(request.playRequest->sourceId ^ request.playRequest->targetId)
98 : 0;
99
100 const int decalId = DecalManager::inst().project(
101 x, y, z, 0.f, 1.f, 0.f, albedo, "attackvfx", size, depth, true, seed, fadeIn, lifetime,
102 fadeOut, 0.f, 0.f, 0.f, 0.f);
103 if (decalId <= 0) {
104 releaseTexture(albedo);
105 return eve::Result<eve::stylize::AttackVfxLayerHandle>::failure(eve::Diagnostic::error(
106 eve::DiagnosticCode::Failed, "DecalManager.project returned an invalid id", "uri"));
107 }
108
109 Live owned;
110 owned.decalId = decalId;
111 owned.albedo = albedo;
112 const auto id = ++nextId_;
113 live_.emplace(id, std::move(owned));
114 return eve::Result<eve::stylize::AttackVfxLayerHandle>::success(
115 eve::stylize::AttackVfxLayerHandle{id});
116 }

118 eve::Result<void> update(eve::stylize::AttackVfxLayerHandle handle, double dtSeconds,
119 const eve::stylize::AttackVfxLayerStartRequest& request) override {
120 const auto found = live_.find(handle.id);
121 if (found == live_.end())
122 return eve::Result<void>::failure(eve::Diagnostic::error(
123 eve::DiagnosticCode::StaleHandle, "decal AttackVfx layer handle is stale", "handle"));
124
125 // Coalesce DecalManager::update to once per runtime tickSerial.
126 if (request.tickSerial != lastTickSerial_) {
127 DecalManager::inst().update(static_cast<float>(dtSeconds));
128 lastTickSerial_ = request.tickSerial;
129 }
130
131 if (found->second.draining && !DecalManager::inst().contains(found->second.decalId)) {
132 releaseTexture(found->second.albedo);
133 live_.erase(found);
134 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
135 }
136 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
137 }

139 eve::Result<void> stop(eve::stylize::AttackVfxLayerHandle handle,
140 eve::stylize::AttackVfxStopBehavior behavior) override {
141 const auto found = live_.find(handle.id);
142 if (found == live_.end())
143 return eve::Result<void>::failure(eve::Diagnostic::error(
144 eve::DiagnosticCode::StaleHandle, "decal AttackVfx layer handle is stale", "handle"));
145 if (behavior == eve::stylize::AttackVfxStopBehavior::ClearImmediately) {
146 (void)DecalManager::inst().remove(found->second.decalId);
147 releaseTexture(found->second.albedo);
148 live_.erase(found);
149 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
150 }
151 // StopEmitting: leave the projected decal to age out via update()/DecalManager.
152 found->second.draining = true;
153 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
154 }
