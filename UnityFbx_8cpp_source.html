36std::optional<double> setting(const std::string& text, const std::string& name) {
37 std::smatch match;
38 if (!std::regex_search(text, match, std::regex("(?:^|\\n) " + name + ": *([^\\r\\n]+)"))) return {};
39 try {
40 std::size_t end = 0;
41 const auto token = match[1].str();
42 const auto value = std::stod(token, &end);
43 if (end != token.size() || !std::isfinite(value)) return {};
44 return value;
45 } catch (...) {
46 return {};
47 }
48}

54Result<PreparedAssetImport> meshImport(const UnityProjectImportRequest& request, const UnitySourceAsset& source,
55 const aiMesh& mesh, std::int64_t fileId, float scale) {
56 if (!mesh.HasPositions() || !mesh.HasNormals() || !mesh.HasTextureCoords(0) || mesh.HasBones() ||
57 mesh.mNumAnimMeshes)
58 return detail::failure<PreparedAssetImport>(
59 DiagnosticCode::Unsupported,
60 "FBX static mesh requires positions, imported normals and UV0; skin/morph conversion is unavailable",
61 source.path);
62 if (mesh.mNumVertices > request.limits.maximumVerticesPerPrimitive ||
63 std::uint64_t(mesh.mNumFaces) * 3 > request.limits.maximumIndicesPerPrimitive ||
64 std::uint64_t(mesh.mNumVertices) * 32 + std::uint64_t(mesh.mNumFaces) * 12 > request.limits.maximumDecodedBytes)
65 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument, "FBX mesh exceeds budget",
66 source.path);
67 std::vector<std::uint8_t> buffer;
68 auto scalar = [&](float value) { put32(buffer, std::bit_cast<std::uint32_t>(value)); };
69 // Unity's FBX importer reflects X. The canonical right-handed basis then reflects Z.
70 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
71 scalar(-mesh.mVertices[i].x * scale);
72 scalar(mesh.mVertices[i].y * scale);
73 scalar(-mesh.mVertices[i].z * scale);
74 }
75 const auto normalOffset = buffer.size();
76 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
77 scalar(-mesh.mNormals[i].x);
78 scalar(mesh.mNormals[i].y);
79 scalar(-mesh.mNormals[i].z);
80 }
81 const auto uvOffset = buffer.size();
82 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
83 // Canonical image rows/glTF UV origin are top-left; Unity mesh UV origin is bottom-left.
84 scalar(mesh.mTextureCoords[0][i].x);
85 scalar(1.f - mesh.mTextureCoords[0][i].y);
86 }
87 const auto indexOffset = buffer.size();
88 for (unsigned i = 0; i < mesh.mNumFaces; ++i) {
89 if (mesh.mFaces[i].mNumIndices != 3)
90 return detail::failure<PreparedAssetImport>(DiagnosticCode::Unsupported,
91 "FBX contains non-triangle primitives", source.path);
92 for (unsigned j = 0; j < 3; ++j) put32(buffer, mesh.mFaces[i].mIndices[j]);
93 }
94 Value::Array views, accessors;
95 auto stream = [&](std::size_t offset, std::size_t length, std::uint64_t count, const char* type, int component) {
96 const auto index = std::int64_t(views.size());
97 views.emplace_back(Value::Object{{"buffer", Value(std::int64_t(0))},
98 {"byteOffset", Value(std::int64_t(offset))},
99 {"byteLength", Value(std::int64_t(length))}});
100 accessors.emplace_back(Value::Object{{"bufferView", Value(index)},
101 {"componentType", Value(std::int64_t(component))},
102 {"count", Value(std::int64_t(count))},
103 {"type", Value(type)}});
104 };
105 stream(0, normalOffset, mesh.mNumVertices, "VEC3", 5126);
106 stream(normalOffset, uvOffset - normalOffset, mesh.mNumVertices, "VEC3", 5126);
107 stream(uvOffset, indexOffset - uvOffset, mesh.mNumVertices, "VEC2", 5126);
108 stream(indexOffset, buffer.size() - indexOffset, std::uint64_t(mesh.mNumFaces) * 3, "SCALAR", 5125);
109 Value primitive(Value::Object{{"attributes", Value(Value::Object{{"POSITION", Value(std::int64_t(0))},
110 {"NORMAL", Value(std::int64_t(1))},
111 {"TEXCOORD_0", Value(std::int64_t(2))}})},
112 {"indices", Value(std::int64_t(3))}});
113 Value document(Value::Object{
114 {"asset", Value(Value::Object{{"version", Value("2.0")}})},
115 {"buffers", Value(Value::Array{Value(Value::Object{{"uri", Value("mesh.bin")},
116 {"byteLength", Value(std::int64_t(buffer.size()))}})})},
117 {"bufferViews", Value(std::move(views))},
118 {"accessors", Value(std::move(accessors))},
119 {"meshes",
120 Value(Value::Array{Value(Value::Object{{"primitives", Value(Value::Array{std::move(primitive)})}})})}});
121 auto json = document.toJson();
122 if (!json) return Result<PreparedAssetImport>::failure(json.status());
123 auto identity = request.package;
124 identity.packageId = identity.packageId.child("unity:" + source.guid + ":" + std::to_string(fileId));
125 auto result = prepareGltfImport({identity,
126 source.path,
127 {json.value().begin(), json.value().end()},
128 {{"mesh.bin", std::move(buffer)}},
129 request.limits});
130 if (!result) return result;
131 for (auto& mapping : result.value().sourceMappings) mapping.sourceObject = std::to_string(fileId);
132 return result;
133}

136Result<PreparedAssetImport> prepareUnityFbx(const UnityProjectImportRequest& request, const UnitySourceAsset& source) {
137 const auto& input = request.files.at(source.path);
138 const auto& metadata = request.files.at(source.path + ".meta");
139 const std::string meta(metadata.begin(), metadata.end());
140 const auto scale = setting(meta, "globalScale");
141 if (!std::regex_search(meta, std::regex(R"(internalIDToNameTable:\s*\[\])")) ||
142 setting(meta, "fileIdsGeneration") != 2 || !scale || *scale <= 0 || setting(meta, "useFileScale") != 1 ||
143 setting(meta, "normalImportMode") != 0 || setting(meta, "swapUVChannels") != 0)
144 return detail::failure<PreparedAssetImport>(
145 DiagnosticCode::Unsupported,
146 "FBX requires hashed file IDs, positive scale, file units, imported normals and original UV0", source.path);
147 if (input.empty() || input.size() > request.limits.maximumSourceBytes)
148 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument, "FBX source exceeds budget",
149 source.path);
150 Assimp::Importer importer;
151 importer.SetIOHandler(new NoExternalIO());
152 const auto* scene = importer.ReadFileFromMemory(input.data(), input.size(), aiProcess_Triangulate, "fbx");
153 if (!scene || !scene->mRootNode)
154 return detail::failure<PreparedAssetImport>(DiagnosticCode::ParseError, importer.GetErrorString(), source.path);
155 if (scene->mNumAnimations || scene->mNumMeshes > request.limits.maximumAssets)
156 return detail::failure<PreparedAssetImport>(DiagnosticCode::Unsupported,
157 "FBX animation or mesh budget is unsupported", source.path);
158 std::uint64_t meshBytes = 0;
159 for (unsigned i = 0; i < scene->mNumMeshes; ++i) {
160 const auto& mesh = *scene->mMeshes[i];
161 const auto bytes = std::uint64_t(mesh.mNumVertices) * 32 + std::uint64_t(mesh.mNumFaces) * 12;
162 if (bytes > request.limits.maximumDecodedBytes || meshBytes > request.limits.maximumDecodedBytes - bytes)
163 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
164 "aggregate FBX mesh budget exceeded", source.path);
165 meshBytes += bytes;
166 }
167 double units = 0;
168 float floatUnits = 0;
169 if (scene->mMetaData) {
170 if (!scene->mMetaData->Get("UnitScaleFactor", units) && scene->mMetaData->Get("UnitScaleFactor", floatUnits))
171 units = floatUnits;
172 }
173 const double factor = units * 0.01 * *scale;
174 if (!std::isfinite(factor) || factor <= 0 || factor > std::numeric_limits<float>::max())
175 return detail::failure<PreparedAssetImport>(DiagnosticCode::Unsupported, "FBX file unit scale is unavailable",
176 source.path);
177 auto manifest = detail::baseManifest(request.package, "eve.unity-fbx/1");
178 if (!manifest) return Result<PreparedAssetImport>::failure(manifest.status());
179 PreparedAssetImport out;
180 out.manifest = std::move(manifest).takeValue();
181 std::set<std::string> names;
182 std::set<std::int64_t> ids;
183 std::uint32_t nodeCount = 0;
184 std::function<Result<void>(const aiNode&, unsigned)> visit = [&](const aiNode& node,
185 unsigned depth) -> Result<void> {
186 if (++nodeCount > request.limits.maximumAssets || depth > 256)
187 return detail::failure<void>(DiagnosticCode::InvalidArgument, "FBX hierarchy exceeds budget", source.path);
188 if (node.mNumMeshes) {
189 const std::string name(node.mName.C_Str());
190 if (node.mNumMeshes != 1 || name.empty() || !names.insert(name).second)
191 return detail::failure<void>(DiagnosticCode::Unsupported,
192 "FBX requires unique names and one mesh per node", source.path);
193 const std::string key = "Type:Mesh->" + name + "0";
194 const auto id = std::bit_cast<std::int64_t>(std::uint64_t(XXH64(key.data(), key.size(), 0)));
195 if (id == 0 || !ids.insert(id).second)
196 return detail::failure<void>(DiagnosticCode::Conflict, "FBX subasset identity collision", source.path);
197 if (node.mMeshes[0] >= scene->mNumMeshes)
198 return detail::failure<void>(DiagnosticCode::ParseError, "FBX mesh index is invalid", source.path);
199 auto converted = meshImport(request, source, *scene->mMeshes[node.mMeshes[0]], id, float(factor));
200 if (!converted) return Result<void>::failure(converted.status());
201 auto& part = converted.value();
202 out.manifest.entrypoints.emplace(std::to_string(id), part.manifest.assets.front().asset);
203 for (auto& asset : part.manifest.assets) out.manifest.assets.push_back(std::move(asset));
204 for (auto& entry : part.entries)
205 if (entry.path.starts_with("assets/")) out.entries.push_back(std::move(entry));
206 for (auto& mapping : part.sourceMappings) out.sourceMappings.push_back(std::move(mapping));
207 out.findings.push_back(
208 {source.path, "FBX.mesh:" + std::to_string(id), ImportDisposition::Translated, name});
209 }
210 for (unsigned i = 0; i < node.mNumChildren; ++i) {
211 auto result = visit(*node.mChildren[i], depth + 1);
212 if (!result) return result;
213 }
214 return Result<void>::success();
215 };
216 auto traversed = visit(*scene->mRootNode, 0);
217 if (!traversed) return Result<PreparedAssetImport>::failure(traversed.status());
218 if (out.manifest.assets.empty())
219 return detail::failure<PreparedAssetImport>(DiagnosticCode::Unsupported, "FBX contains no static meshes",
220 source.path);
221 out.findings.push_back({source.path, "FBX.additionalStreams", ImportDisposition::Unsupported,
222 "only position, imported normal and UV0 streams are converted; tangents, vertex colors and "
223 "secondary UVs are not applied"});
224 return Result<PreparedAssetImport>::success(std::move(out));
225}
