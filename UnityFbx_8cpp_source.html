38std::optional<double> setting(const std::string& text, const std::string& name) {
39 std::smatch match;
40 if (!std::regex_search(text, match, std::regex("(?:^|\\n) " + name + ": *([^\\r\\n]+)"))) return {};
41 try {
42 std::size_t end = 0;
43 const auto token = match[1].str();
44 const auto value = std::stod(token, &end);
45 if (end != token.size() || !std::isfinite(value)) return {};
46 return value;
47 } catch (...) {
48 return {};
49 }

52std::set<std::int64_t> referencedMeshIds(const UnityProjectImportRequest& request,
53 std::string_view sourceGuid) {
54 std::set<std::int64_t> result;
55 const std::regex reference("m_Mesh: *\\{fileID: *(-?[0-9]+), *guid: *" +
56 std::string(sourceGuid) + R"(, *type: *[23]\})",
57 std::regex::icase);
58 for (const auto& [path, bytes] : request.files) {
59 if (unity_detail::foldAscii(std::filesystem::path(path).extension().string()) != ".prefab" || bytes.empty())
60 continue;
61 const std::string text(bytes.begin(), bytes.end());
62 for (std::sregex_iterator match(text.begin(), text.end(), reference), end; match != end; ++match) {
63 try {
64 std::size_t parsed = 0;
65 const auto id = std::stoll((*match)[1].str(), &parsed);
66 if (parsed == static_cast<std::size_t>((*match)[1].length()) && id != 0) result.emplace(id);
67 } catch (...) {
68 }
69 }
70 }
71 return result;
72}

78Result<PreparedAssetImport> meshImport(const UnityProjectImportRequest& request, const UnitySourceAsset& source,
79 const aiMesh& mesh, std::int64_t fileId, std::optional<unsigned> submesh,
80 float scale) {
81 if (!mesh.HasPositions() || !mesh.HasNormals() || !mesh.HasTextureCoords(0) || mesh.HasBones() ||
82 mesh.mNumAnimMeshes)
83 return detail::failure<PreparedAssetImport>(
84 DiagnosticCode::Unsupported,
85 "FBX static mesh requires positions, imported normals and UV0; skin/morph conversion is unavailable",
86 source.path);
87 std::uint64_t vertexFloats = 6;
88 for (unsigned set = 0; set < AI_MAX_NUMBER_OF_TEXTURECOORDS; ++set)
89 if (mesh.HasTextureCoords(set)) vertexFloats += 2;
90 if (mesh.HasTangentsAndBitangents()) vertexFloats += 4;
91 for (unsigned set = 0; set < AI_MAX_NUMBER_OF_COLOR_SETS; ++set)
92 if (mesh.HasVertexColors(set)) vertexFloats += 4;
93 if (mesh.mNumVertices > request.limits.maximumVerticesPerPrimitive ||
94 std::uint64_t(mesh.mNumFaces) * 3 > request.limits.maximumIndicesPerPrimitive ||
95 vertexFloats > request.limits.maximumDecodedBytes / 4 / mesh.mNumVertices ||
96 std::uint64_t(mesh.mNumVertices) * vertexFloats * 4 + std::uint64_t(mesh.mNumFaces) * 12 >
97 request.limits.maximumDecodedBytes)
98 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument, "FBX mesh exceeds budget",
99 source.path);
100 std::vector<std::uint8_t> buffer;
101 auto scalar = [&](float value) { put32(buffer, std::bit_cast<std::uint32_t>(value)); };
102 // Unity's FBX importer reflects X. The canonical right-handed basis then reflects Z.
103 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
104 scalar(-mesh.mVertices[i].x * scale);
105 scalar(mesh.mVertices[i].y * scale);
106 scalar(-mesh.mVertices[i].z * scale);
107 }
108 const auto normalOffset = buffer.size();
109 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
110 scalar(-mesh.mNormals[i].x);
111 scalar(mesh.mNormals[i].y);
112 scalar(-mesh.mNormals[i].z);
113 }
114 Value::Array views, accessors;
115 auto stream = [&](std::size_t offset, std::size_t length, std::uint64_t count, const char* type, int component) {
116 const auto index = std::int64_t(views.size());
117 views.emplace_back(Value::Object{{"buffer", Value(std::int64_t(0))},
118 {"byteOffset", Value(std::int64_t(offset))},
119 {"byteLength", Value(std::int64_t(length))}});
120 accessors.emplace_back(Value::Object{{"bufferView", Value(index)},
121 {"componentType", Value(std::int64_t(component))},
122 {"count", Value(std::int64_t(count))},
123 {"type", Value(type)}});
124 };
125 stream(0, normalOffset, mesh.mNumVertices, "VEC3", 5126);
126 stream(normalOffset, buffer.size() - normalOffset, mesh.mNumVertices, "VEC3", 5126);
127 Value::Object attributes{{"POSITION", Value(std::int64_t(0))}, {"NORMAL", Value(std::int64_t(1))}};
128 for (unsigned set = 0; set < AI_MAX_NUMBER_OF_TEXTURECOORDS; ++set) {
129 if (!mesh.HasTextureCoords(set)) continue;
130 const auto offset = buffer.size();
131 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
132 // Canonical image rows/glTF UV origin are top-left; Unity mesh UV origin is bottom-left.
133 scalar(mesh.mTextureCoords[set][i].x);
134 scalar(1.f - mesh.mTextureCoords[set][i].y);
135 }
136 const auto accessor = std::int64_t(accessors.size());
137 stream(offset, buffer.size() - offset, mesh.mNumVertices, "VEC2", 5126);
138 attributes["TEXCOORD_" + std::to_string(set)] = Value(accessor);
139 }
140 if (mesh.HasTangentsAndBitangents()) {
141 const auto offset = buffer.size();
142 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
143 scalar(-mesh.mTangents[i].x);
144 scalar(mesh.mTangents[i].y);
145 scalar(-mesh.mTangents[i].z);
146 const auto& n = mesh.mNormals[i];
147 const auto& t = mesh.mTangents[i];
148 const auto& b = mesh.mBitangents[i];
149 const auto handedness =
150 ((n.y * t.z - n.z * t.y) * b.x + (n.z * t.x - n.x * t.z) * b.y + (n.x * t.y - n.y * t.x) * b.z) < 0.f
151 ? -1.f
152 : 1.f;
153 scalar(handedness);
154 }
155 const auto accessor = std::int64_t(accessors.size());
156 stream(offset, buffer.size() - offset, mesh.mNumVertices, "VEC4", 5126);
157 attributes["TANGENT"] = Value(accessor);
158 }
159 for (unsigned set = 0; set < AI_MAX_NUMBER_OF_COLOR_SETS; ++set) {
160 if (!mesh.HasVertexColors(set)) continue;
161 const auto offset = buffer.size();
162 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
163 scalar(mesh.mColors[set][i].r);
164 scalar(mesh.mColors[set][i].g);
165 scalar(mesh.mColors[set][i].b);
166 scalar(mesh.mColors[set][i].a);
167 }
168 const auto accessor = std::int64_t(accessors.size());
169 stream(offset, buffer.size() - offset, mesh.mNumVertices, "VEC4", 5126);
170 attributes["COLOR_" + std::to_string(set)] = Value(accessor);
171 }
172 const auto indexOffset = buffer.size();
173 for (unsigned i = 0; i < mesh.mNumFaces; ++i) {
174 if (mesh.mFaces[i].mNumIndices != 3)
175 return detail::failure<PreparedAssetImport>(DiagnosticCode::Unsupported,
176 "FBX contains non-triangle primitives", source.path);
177 for (unsigned j = 0; j < 3; ++j) put32(buffer, mesh.mFaces[i].mIndices[j]);
178 }
179 const auto indexAccessor = std::int64_t(accessors.size());
180 stream(indexOffset, buffer.size() - indexOffset, std::uint64_t(mesh.mNumFaces) * 3, "SCALAR", 5125);
181 Value primitive(Value::Object{{"attributes", Value(std::move(attributes))}, {"indices", Value(indexAccessor)}});
182 Value document(Value::Object{
183 {"asset", Value(Value::Object{{"version", Value("2.0")}})},
184 {"buffers", Value(Value::Array{Value(Value::Object{{"uri", Value("mesh.bin")},
185 {"byteLength", Value(std::int64_t(buffer.size()))}})})},
186 {"bufferViews", Value(std::move(views))},
187 {"accessors", Value(std::move(accessors))},
188 {"meshes",
189 Value(Value::Array{Value(Value::Object{{"primitives", Value(Value::Array{std::move(primitive)})}})})}});
190 auto json = document.toJson();
191 if (!json) return Result<PreparedAssetImport>::failure(json.status());
192 auto identity = request.package;
193 std::string identityKey = "unity:" + source.guid + ":" + std::to_string(fileId);
194 if (submesh) identityKey += ":submesh:" + std::to_string(*submesh);
195 identity.packageId = identity.packageId.child(identityKey);
196 auto result = prepareGltfImport({identity,
197 source.path,
198 {json.value().begin(), json.value().end()},
199 {{"mesh.bin", std::move(buffer)}},
200 request.limits});
201 if (!result) return result;
202 for (auto& mapping : result.value().sourceMappings)
203 mapping.sourceObject = std::to_string(fileId) +
204 (submesh ? "/submesh/" + std::to_string(*submesh) : std::string{});
205 return result;
206}

209Result<PreparedAssetImport> prepareUnityFbx(const UnityProjectImportRequest& request, const UnitySourceAsset& source) {
210 const auto& input = request.files.at(source.path);
211 const auto& metadata = request.files.at(source.path + ".meta");
212 const std::string meta(metadata.begin(), metadata.end());
213 const auto scale = setting(meta, "globalScale");
214 const auto generationSetting = setting(meta, "fileIdsGeneration");
215 const auto generation = generationSetting.value_or(1.0);
216 if ((generation != 1 && generation != 2) || !scale || *scale <= 0 || setting(meta, "useFileScale") != 1 ||
217 setting(meta, "normalImportMode") != 0 || setting(meta, "swapUVChannels") != 0)
218 return detail::failure<PreparedAssetImport>(
219 DiagnosticCode::Unsupported,
220 "FBX requires supported file IDs, positive scale, file units, imported normals and original UV0", source.path);
221 if (input.empty() || input.size() > request.limits.maximumSourceBytes)
222 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument, "FBX source exceeds budget",
223 source.path);
224 Assimp::Importer importer;
225 importer.SetIOHandler(new NoExternalIO());
226 const auto* scene = importer.ReadFileFromMemory(input.data(), input.size(),
227 aiProcess_Triangulate | aiProcess_CalcTangentSpace, "fbx");
228 if (!scene || !scene->mRootNode)
229 return detail::failure<PreparedAssetImport>(DiagnosticCode::ParseError, importer.GetErrorString(), source.path);
230 if (scene->mNumAnimations || scene->mNumMeshes > request.limits.maximumAssets)
231 return detail::failure<PreparedAssetImport>(DiagnosticCode::Unsupported,
232 "FBX animation or mesh budget is unsupported", source.path);
233 std::uint64_t meshBytes = 0;
234 for (unsigned i = 0; i < scene->mNumMeshes; ++i) {
235 const auto& mesh = *scene->mMeshes[i];
236 const auto bytes = std::uint64_t(mesh.mNumVertices) * 32 + std::uint64_t(mesh.mNumFaces) * 12;
237 if (bytes > request.limits.maximumDecodedBytes || meshBytes > request.limits.maximumDecodedBytes - bytes)
238 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
239 "aggregate FBX mesh budget exceeded", source.path);
240 meshBytes += bytes;
241 }
242 double units = 0;
243 float floatUnits = 0;
244 if (scene->mMetaData) {
245 if (!scene->mMetaData->Get("UnitScaleFactor", units) && scene->mMetaData->Get("UnitScaleFactor", floatUnits))
246 units = floatUnits;
247 }
248 const double factor = units * 0.01 * *scale;
249 if (!std::isfinite(factor) || factor <= 0 || factor > std::numeric_limits<float>::max())
250 return detail::failure<PreparedAssetImport>(DiagnosticCode::Unsupported, "FBX file unit scale is unavailable",
251 source.path);
252 auto manifest = detail::baseManifest(request.package, "eve.unity-fbx/1");
253 if (!manifest) return Result<PreparedAssetImport>::failure(manifest.status());
254 PreparedAssetImport out;
255 out.manifest = std::move(manifest).takeValue();
256 const auto referencedIds = generation == 1 ? referencedMeshIds(request, source.guid) : std::set<std::int64_t>{};
257 std::uint32_t meshNodeCount = 0;
258 std::function<void(const aiNode&)> countMeshNodes = [&](const aiNode& node) {
259 if (node.mNumMeshes) ++meshNodeCount;
260 for (unsigned index = 0; index < node.mNumChildren; ++index) countMeshNodes(*node.mChildren[index]);
261 };
262 countMeshNodes(*scene->mRootNode);
263 if (generation == 1 && meshNodeCount > 1 &&
264 std::any_of(referencedIds.begin(), referencedIds.end(), [](std::int64_t id) {
265 return id < 4300000 || id >= 4400000 || ((id - 4300000) & 1) != 0;
266 }))
267 return detail::failure<PreparedAssetImport>(
268 DiagnosticCode::Unsupported,
269 "legacy FBX has non-indexed mesh references that cannot be mapped to multiple nodes", source.path);
270 std::set<std::string> names;
271 std::set<std::int64_t> ids;
272 std::uint32_t nodeCount = 0;
273 std::function<Result<void>(const aiNode&, unsigned)> visit = [&](const aiNode& node,
274 unsigned depth) -> Result<void> {
275 if (++nodeCount > request.limits.maximumAssets || depth > 256)
276 return detail::failure<void>(DiagnosticCode::InvalidArgument, "FBX hierarchy exceeds budget", source.path);
277 if (node.mNumMeshes) {
278 const std::string name(node.mName.C_Str());
279 if (name.empty() || !names.insert(name).second)
280 return detail::failure<void>(DiagnosticCode::Unsupported,
281 "FBX requires unique mesh-bearing node names", source.path);
282 std::int64_t id = 0;
283 if (generation == 2) {
284 const std::string key = "Type:Mesh->" + name + "0";
285 id = std::bit_cast<std::int64_t>(std::uint64_t(XXH64(key.data(), key.size(), 0)));
286 } else if (meshNodeCount == 1 && referencedIds.size() == 1) {
287 id = *referencedIds.begin();
288 } else {
289 id = std::int64_t(4300000) + std::int64_t(node.mMeshes[0]) * 2;
290 }
291 if (id == 0 || !ids.insert(id).second)
292 return detail::failure<void>(DiagnosticCode::Conflict, "FBX subasset identity collision", source.path);
293 for (unsigned slot = 0; slot < node.mNumMeshes; ++slot) {
294 if (node.mMeshes[slot] >= scene->mNumMeshes)
295 return detail::failure<void>(DiagnosticCode::ParseError, "FBX mesh index is invalid", source.path);
296 const auto submesh = node.mNumMeshes > 1 ? std::optional<unsigned>(slot) : std::nullopt;
297 auto converted = meshImport(request, source, *scene->mMeshes[node.mMeshes[slot]], id, submesh,
298 float(factor));
299 if (!converted) return Result<void>::failure(converted.status());
300 auto& part = converted.value();
301 const std::string entrypoint = std::to_string(id) +
302 (submesh ? "/submesh/" + std::to_string(*submesh) : std::string{});
303 out.manifest.entrypoints.emplace(entrypoint, part.manifest.assets.front().asset);
304 for (auto& asset : part.manifest.assets) out.manifest.assets.push_back(std::move(asset));
305 for (auto& entry : part.entries)
306 if (entry.path.starts_with("assets/")) out.entries.push_back(std::move(entry));
307 for (auto& mapping : part.sourceMappings) out.sourceMappings.push_back(std::move(mapping));
308 }
309 out.findings.push_back(
310 {source.path, "FBX.mesh:" + std::to_string(id), ImportDisposition::Translated, name});
311 }
312 for (unsigned i = 0; i < node.mNumChildren; ++i) {
313 auto result = visit(*node.mChildren[i], depth + 1);
314 if (!result) return result;
315 }
316 return Result<void>::success();
317 };
318 auto traversed = visit(*scene->mRootNode, 0);
319 if (!traversed) return Result<PreparedAssetImport>::failure(traversed.status());
320 if (out.manifest.assets.empty())
321 return detail::failure<PreparedAssetImport>(DiagnosticCode::Unsupported, "FBX contains no static meshes",
322 source.path);
323 out.findings.push_back({source.path, "FBX.vertexStreams", ImportDisposition::Translated,
324 "positions, imported normals, tangents, vertex colors and all available UV sets retained"});
325 return Result<PreparedAssetImport>::success(std::move(out));
326}
