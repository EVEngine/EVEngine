83template <class Ref, class T>
84std::optional<Ref> nativeProxyReference(T* native) {
85 if (!native) return std::nullopt;
86 std::lock_guard lock(NativeProxyReleases<T>::mutex);
87 const auto found = NativeProxyReleases<T>::records.find(native);
88 if (found == NativeProxyReleases<T>::records.end()) return std::nullopt;
89 if (const auto* reference = std::any_cast<Ref>(&found->second.reference)) return *reference;
90 return std::nullopt;
91}

93template <class T>
94SQInteger releaseNativeProxy(SQUserPointer pointer, SQInteger) {
95 auto native = static_cast<T*>(pointer);
96 if (!native) return 0;
97
98 std::function<eve::Result<void>()> release;
99 {
100 std::lock_guard lock(NativeProxyReleases<T>::mutex);
101 const auto found = NativeProxyReleases<T>::records.find(native);
102 if (found == NativeProxyReleases<T>::records.end()) return 0;
103 release = std::move(found->second.release);
104 NativeProxyReleases<T>::records.erase(found);
105 }
106 release().ignore("release Squirrel procgen native proxy");
107 return 0;
108}

110template <class T, class Ref, class Resolve, class Release>
111ssq::Table makeOwnedNativeProxy(HSQUIRRELVM vm, eve::Result<Ref>&& reference, Resolve&& resolve, Release&& release) {
112 if (!reference) return eve::script::projectStatusResult(vm, reference.status());
113
114 const auto ref = std::move(reference).takeValue();
115 const auto view = std::invoke(std::forward<Resolve>(resolve), ref);
116 if (!view.isBound()) {
117 std::invoke(release, ref).ignore("rollback unbound Squirrel procgen proxy");
118 return eve::script::projectStatusResult(
119 vm, eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle,
120 "procgen proxy could not resolve its owned object",
121 "procgen", {}, "procgen.squirrel"))
122 .status());
123 }
124
125 const SQInteger top = sq_gettop(vm);
126 const size_t hashCode = eve::script::detail::squirrelTypeHash<T*>();
127 sq_pushobject(vm, ssq::detail::getClassObj(vm, hashCode));
128 if (SQ_FAILED(sq_createinstance(vm, -1))) {
129 sq_settop(vm, top);
130 std::invoke(release, ref).ignore("rollback failed Squirrel procgen instance");
131 return eve::script::projectStatusResult(
132 vm, eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::Failed,
133 "failed to create Squirrel procgen proxy", "procgen",
134 {}, "procgen.squirrel"))
135 .status());
136 }
137 sq_remove(vm, -2);
138 auto native = view.get();
139 std::optional<typename NativeProxyReleases<T>::Record> retired;
140 {
141 std::lock_guard lock(NativeProxyReleases<T>::mutex);
142 auto& records = NativeProxyReleases<T>::records;
143 if (auto old = records.find(native); old != records.end()) {
144 retired.emplace(std::move(old->second));
145 records.erase(old);
146 }
147 }
148 if (retired) {
149 // An explicitly released allocation can be reused while its script proxy is still alive.
150 // Disarm that proxy before publishing the new generation at the same native address.
151 sq_pushobject(retired->vm, retired->instance);
152 sq_setreleasehook(retired->vm, -1, nullptr);
153 sq_setinstanceup(retired->vm, -1, nullptr);
154 sq_pop(retired->vm, 1);
155 }
156 HSQOBJECT instance;
157 sq_getstackobj(vm, -1, &instance);
158 {
159 std::lock_guard lock(NativeProxyReleases<T>::mutex);
160 NativeProxyReleases<T>::records.emplace(
161 native,
162 typename NativeProxyReleases<T>::Record{
163 ref, [ref, release = std::forward<Release>(release)]() mutable { return std::invoke(release, ref); },
164 vm, instance});
165 }
166 sq_setinstanceup(vm, -1, native);
167 sq_settypetag(vm, -1, reinterpret_cast<SQUserPointer>(hashCode));
168 sq_setreleasehook(vm, -1, &releaseNativeProxy<T>);
169
170 ssq::Instance value(vm);
171 sq_getstackobj(vm, -1, &value.getRaw());
172 sq_addref(vm, &value.getRaw());
173 sq_settop(vm, top);
174
175 auto result = eve::script::projectStatusResult(vm, eve::Status::success(eve::StatusCode::Applied));
176 result.set("value", value);
177 result.set("ownership", std::string("owned"));
178 result.set("ownerEpoch", static_cast<std::int64_t>(ref.ownerEpoch));
179 result.set("handle", static_cast<std::int64_t>(ref.packed()));
180 return result;
181}

197template <class Ref, class Proxy, class Release>
198ssq::Table makeOwnedProxy(HSQUIRRELVM vm, eve::Result<Ref>&& reference, Release&& release) {
199 if (!reference) return eve::script::projectStatusResult(vm, reference.status());
200 const Ref ref = std::move(reference).takeValue();
201 auto object = eve::script::makeOwnedSquirrelInstance<Proxy>(vm, std::make_unique<Proxy>(ref));
202 if (!object) {
203 const eve::Status status = object.status();
204 object.ignore("failed to create owned procgen proxy");
205 std::invoke(std::forward<Release>(release), ref).ignore("rollback failed owned procgen allocation");
206 return eve::script::projectStatusResult(vm, status);
207 }
208 ssq::Object owned = std::move(object).takeValue();
209 auto result = eve::script::projectStatusResult(vm, eve::Status::success(eve::StatusCode::Applied));
210 result.set("value", owned);
211 result.set("ownership", std::string("owned"));
212 result.set("ownerEpoch", static_cast<std::int64_t>(ref.ownerEpoch));
213 result.set("handle", static_cast<std::int64_t>(ref.packed()));
214 return result;
