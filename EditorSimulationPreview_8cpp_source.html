18 std::uint64_t tick) const {

20 if (!stepped.ok()) return EditorResult<SimulationPreviewFrame>::failure(stepped.status());
21 auto captured = backend.capture();
22 if (!captured.ok()) return EditorResult<SimulationPreviewFrame>::failure(captured.status());
23 std::set<std::string> ids;
24 for (const auto& object : captured.value()) {
25 if (object.object.empty() || !ids.insert(object.object).second)
26 return eve::editing::failed<SimulationPreviewFrame>(
27 EditorStatus::Conflict, RuleId("editor.simulation.unstable-object-id"),
28 "Simulation preview samples require unique stable object ids");
29 const double values[]{object.positionX, object.positionY, object.positionZ, object.rotationX,
30 object.rotationY, object.rotationZ, object.rotationW};
31 if (std::any_of(std::begin(values), std::end(values), [](double value) { return !std::isfinite(value); }))
32 return eve::editing::failed<SimulationPreviewFrame>(EditorStatus::Failed,
33 RuleId("editor.simulation.nonfinite-sample"),
34 "Simulation preview produced a non-finite transform");
35 }
36 std::sort(captured.value().begin(), captured.value().end(),
37 [](const auto& a, const auto& b) { return a.object < b.object; });
38 return eve::editing::applied<SimulationPreviewFrame>(

67SimulationBakeResult SimulationPreviewController::bake(const IEditorSimulationBackend& source,
68 std::uint64_t sourceRevision, int steps, int maxSteps) const {
69 SimulationBakeResult result;
70 result.sourceRevision = sourceRevision;
71 result.fixedDelta = fixedDelta_;
72 if (steps < 0 || maxSteps <= 0 || steps > maxSteps) {
73 result.status = EditorStatus::Rejected;
74 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
75 eve::DiagnosticCode::InvalidArgument, RuleId("editor.simulation.bake-budget"),
76 DiagnosticSeverity::Error, "Simulation bake step count exceeds the configured budget"));
77 return result;
78 }
79 auto backend = source.cloneForPreview();
80 if (!backend) {
81 result.status = EditorStatus::Unsupported;
82 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
83 eve::DiagnosticCode::Unsupported, RuleId("editor.simulation.clone-unsupported"),
84 DiagnosticSeverity::Error, "Simulation backend cannot create an isolated preview"));
85 return result;
86 }
87 auto initial = backend->capture();
88 if (!initial.ok()) {
89 result.status = initial.code();
90 result.diagnostics = initial.diagnostics();
91 return result;
92 }
93 std::sort(initial.value().begin(), initial.value().end(),
94 [](const auto& a, const auto& b) { return a.object < b.object; });
95 result.frames.push_back({0, 0.0, std::move(initial.value())});
96 for (int index = 1; index <= steps; ++index) {
97 auto frame = advance(*backend, static_cast<std::uint64_t>(index));
98 if (!frame.ok()) {
99 result.status = frame.code();
100 result.diagnostics = frame.diagnostics();
101 return result;
102 }
103 result.frames.push_back(std::move(frame.value()));
104 }
105 result.status = EditorStatus::Applied;
106 return result;
107}
