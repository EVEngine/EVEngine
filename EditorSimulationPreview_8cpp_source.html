22Result<SimulationPreviewFrame> SimulationPreviewController::advance(IEditorSimulationBackend& backend,
23 std::uint64_t tick) const {
24 auto stepped = backend.step(tick, fixedDelta_);
25 if (!stepped.ok()) return Result<SimulationPreviewFrame>::failure(stepped.status());
26 auto captured = backend.capture();
27 if (!captured.ok()) return Result<SimulationPreviewFrame>::failure(captured.status());
28 std::set<std::string> ids;
29 for (const auto& object : captured.value()) {
30 if (object.object.empty() || !ids.insert(object.object).second)
31 return eve::editing::failed<SimulationPreviewFrame>(
32 EditorStatus::Conflict, RuleId("editor.simulation.unstable-object-id"),
33 "Simulation preview samples require unique stable object ids");
34 const double values[]{object.positionX, object.positionY, object.positionZ, object.rotationX,
35 object.rotationY, object.rotationZ, object.rotationW};
36 if (std::any_of(std::begin(values), std::end(values), [](double value) { return !std::isfinite(value); }))
37 return eve::editing::failed<SimulationPreviewFrame>(EditorStatus::Failed,
38 RuleId("editor.simulation.nonfinite-sample"),
39 "Simulation preview produced a non-finite transform");
40 }
41 std::sort(captured.value().begin(), captured.value().end(),
42 [](const auto& a, const auto& b) { return a.object < b.object; });
43 return eve::editing::applied<SimulationPreviewFrame>(
44 {tick, static_cast<double>(tick) * fixedDelta_, std::move(captured.value())});
45}

72SimulationBakeResult SimulationPreviewController::bake(const IEditorSimulationBackend& source,
73 std::uint64_t sourceRevision, int steps, int maxSteps) const {
74 SimulationBakeResult result;
75 result.sourceRevision = sourceRevision;
76 result.fixedDelta = fixedDelta_;
77 if (steps < 0 || maxSteps <= 0 || steps > maxSteps) {
78 result.status = EditorStatus::Rejected;
79 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
80 eve::DiagnosticCode::InvalidArgument, RuleId("editor.simulation.bake-budget"), DiagnosticSeverity::Error,
81 "Simulation bake step count exceeds the configured budget"));
82 return result;
83 }
84 auto backend = source.cloneForPreview();
85 if (!backend) {
86 result.status = EditorStatus::Unsupported;
87 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
88 eve::DiagnosticCode::Unsupported, RuleId("editor.simulation.clone-unsupported"), DiagnosticSeverity::Error,
89 "Simulation backend cannot create an isolated preview"));
90 return result;
91 }
92 auto initial = backend->capture();
93 if (!initial.ok()) {
94 result.status = initial.code();
95 result.diagnostics = initial.diagnostics();
96 return result;
97 }
98 std::sort(initial.value().begin(), initial.value().end(),
99 [](const auto& a, const auto& b) { return a.object < b.object; });
100 result.frames.push_back({0, 0.0, std::move(initial.value())});
101 for (int index = 1; index <= steps; ++index) {
102 auto frame = advance(*backend, static_cast<std::uint64_t>(index));
103 if (!frame.ok()) {
104 result.status = frame.code();
105 result.diagnostics = frame.diagnostics();
106 return result;
107 }
108 result.frames.push_back(std::move(frame.value()));
109 }
110 result.status = EditorStatus::Applied;
111 return result;
112}

194TransactionState transactionState(eve::transaction::CoordinatorState state) noexcept {
195 switch (state) {
196 case eve::transaction::CoordinatorState::Committed: return TransactionState::Committed;
197 case eve::transaction::CoordinatorState::Compensated: return TransactionState::RolledBack;
198 case eve::transaction::CoordinatorState::CompensationFailed:
199 case eve::transaction::CoordinatorState::RollbackFailed:
200 case eve::transaction::CoordinatorState::PartiallyCommitted: return TransactionState::Failed;
201 case eve::transaction::CoordinatorState::RolledBack: return TransactionState::RolledBack;
202 }
203 return TransactionState::Failed;
204}

206TransactionReceipt projectRecord(const EditorTransactionRecord& record) {
207 TransactionReceipt result;
208 result.id = record.specification.id;
209 result.state = transactionState(record.coordinator.state);
210 if (record.authorityReceipt) {
211 result.beforeRevision = record.authorityReceipt->beforeRevision;
212 result.afterRevision = record.authorityReceipt->afterRevision;
213 result.affectedObjects = record.authorityReceipt->affectedObjects;
214 result.diagnostics = record.authorityReceipt->diagnostics;
215 result.authorityReceipt = record.authorityReceipt->authorityReceipt;
216 }
217 return result;
218}
