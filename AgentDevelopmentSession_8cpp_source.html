16bool canAdvance(AgentDevelopmentPhase from, AgentDevelopmentPhase to) {
17 if (to == AgentDevelopmentPhase::Recover)
18 return from != AgentDevelopmentPhase::Recover && from != AgentDevelopmentPhase::Complete &&
19 from != AgentDevelopmentPhase::Aborted;
20 switch (from) {
21 case AgentDevelopmentPhase::Discover: return to == AgentDevelopmentPhase::Modify;
22 case AgentDevelopmentPhase::Modify: return to == AgentDevelopmentPhase::Run;
23 case AgentDevelopmentPhase::Run: return to == AgentDevelopmentPhase::Observe;
24 case AgentDevelopmentPhase::Observe: return to == AgentDevelopmentPhase::Verify;
25 case AgentDevelopmentPhase::Verify: return to == AgentDevelopmentPhase::Modify;
26 case AgentDevelopmentPhase::Recover:
27 return to == AgentDevelopmentPhase::Modify || to == AgentDevelopmentPhase::Run;
28 default: return false;
29 }
30}

39AgentDevelopmentResult AgentDevelopmentSession::start(
40 std::string objective, std::vector<AgentAcceptanceCriterion> criteria) {
41 if (active()) return rejected("active-session", "A development session is already active");
42 if (objective.empty()) return rejected("invalid-objective", "objective must not be empty");
43 if (criteria.empty()) return rejected("invalid-criteria", "at least one acceptance criterion is required");
44 std::set<std::string> ids;
45 for (const auto& criterion : criteria) {
46 if (criterion.id.empty() || criterion.description.empty())
47 return rejected("invalid-criterion", "criterion id and description must not be empty");
48 if (!ids.insert(criterion.id).second)
49 return rejected("duplicate-criterion", "criterion ids must be unique");
50 }
51 reset();
52 sessionId_ = "agent.dev." + std::to_string(nextSession_++);
53 objective_ = std::move(objective);
54 criteria_ = std::move(criteria);
55 phase_ = AgentDevelopmentPhase::Discover;
56 return applied();
57}

59AgentDevelopmentResult AgentDevelopmentSession::advance(std::string_view sessionId,
60 AgentDevelopmentPhase phase) {
61 if (sessionId != sessionId_) return rejected("session-not-found", "development session was not found");
62 if (!active()) return rejected("session-terminal", "development session is already terminal");
63 if (!canAdvance(phase_, phase))
64 return rejected("invalid-transition", "requested phase does not follow the development workflow");
65 phase_ = phase;
66 return applied();

69AgentDevelopmentResult AgentDevelopmentSession::record(std::string_view sessionId,
70 AgentDevelopmentEvidence evidence) {
71 if (sessionId != sessionId_) return rejected("session-not-found", "development session was not found");
72 if (!active()) return rejected("session-terminal", "development session is already terminal");
73 if (evidence.kind.empty() || evidence.summary.empty())
74 return rejected("invalid-evidence", "evidence kind and summary must not be empty");
75 static const std::set<std::string> kEvidenceKinds{
76 "runtime-observation", "test", "screenshot", "checkpoint", "artifact", "manual"};
77 if (!kEvidenceKinds.contains(evidence.kind))
78 return rejected("invalid-evidence-kind", "evidence kind is not part of schema version one");
79 if (evidence.status != "pass" && evidence.status != "fail" && evidence.status != "pending")
80 return rejected("invalid-evidence-status", "evidence status must be pass, fail, or pending");
81 auto criterion = std::find_if(criteria_.begin(), criteria_.end(), [&](const auto& value) {
82 return value.id == evidence.criterionId;
83 });
84 if (criterion == criteria_.end())
85 return rejected("criterion-not-found", "evidence criterion was not declared by the session");
86 evidence.sequence = nextEvidence_++;
87 if (evidence.status == "pass" && evidence.kind != "manual") criterion->passed = true;
88 if (evidence.status == "fail") criterion->passed = false;
89 evidence_.push_back(std::move(evidence));
90 return applied();
91}

152AgentDevelopmentResult parseAgentDevelopmentPhase(std::string_view name,
153 AgentDevelopmentPhase* phase) {
154 if (!phase) return rejected("invalid-output", "phase output must not be null");
155 if (name == "discover") *phase = AgentDevelopmentPhase::Discover;

158 else if (name == "observe") *phase = AgentDevelopmentPhase::Observe;
