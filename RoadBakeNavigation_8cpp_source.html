19Result<void> bakeTurnOverlays(MeshBuild& mesh, RoadOverlay& overlay, const RoadNetwork& network,
20 const RoadBakeOptions& options,
21 const std::unordered_set<std::uint32_t>& activeJunctionNodes) {
22 std::unordered_map<std::uint32_t, SplinePath> paths;
23 for (const auto& edge : network.edges()) {
24 auto spline = edgeToSpline(edge);
25 if (!spline.ok()) return Result<void>::failure(spline.status());
26 paths.emplace(edge.id, std::move(spline).takeValue());
27 }
28
29 for (const auto& link : network.laneLinks()) {
30 auto inEdge = network.edgeResult(link.inEdge);
31 auto outEdge = network.edgeResult(link.outEdge);
32 if (!inEdge.ok() || !outEdge.ok()) continue;
33 const std::uint32_t nodeId =
34 link.inDirection == RoadLaneDirection::Forward ? inEdge.value().to : inEdge.value().from;
35 auto node = network.nodeResult(nodeId);
36 if (!node.ok()) continue;
37
38 auto& inPath = paths[link.inEdge];
39 auto& outPath = paths[link.outEdge];
40 auto inLen = inPath.lengthResult(16);
41 auto outLen = outPath.lengthResult(16);
42 if (!inLen.ok() || !outLen.ok()) continue;
43
44 const bool activeJunction = activeJunctionNodes.contains(node.value().id);
45 const float inTrim =
46 activeJunction ? junctionSocketDistance(network, node.value(), inEdge.value(), inLen.value()) : 0.f;
47 const float outTrim =
48 activeJunction ? junctionSocketDistance(network, node.value(), outEdge.value(), outLen.value()) : 0.f;
49 const float inDist = link.inDirection == RoadLaneDirection::Forward ? inLen.value() - inTrim : inTrim;
50 const float outDist = link.outDirection == RoadLaneDirection::Forward ? outTrim : outLen.value() - outTrim;
51 auto inFrame = inPath.travelFrameResult(inDist, "clamp", 16);
52 auto outFrame = outPath.travelFrameResult(outDist, "clamp", 16);
53 if (!inFrame.ok() || !outFrame.ok()) continue;
54 if (activeJunction) {
55 const JunctionPlane plane = fitJunctionPlane(network, node.value());
56 blendFrameToJunctionPlane(inFrame.value(), plane, 1.f);
57 blendFrameToJunctionPlane(outFrame.value(), plane, 1.f);
58 }
59
60 const float inLat = laneCenterOffset(inEdge.value().style, inEdge.value().lanesForward,
61 inEdge.value().lanesBackward, link.inLane, link.inDirection);
62 const float outLat = laneCenterOffset(outEdge.value().style, outEdge.value().lanesForward,
63 outEdge.value().lanesBackward, link.outLane, link.outDirection);
64
65 const auto& fi = inFrame.value();
66 const auto& fo = outFrame.value();
67 const V3 p0{fi.sample.x, fi.sample.y, fi.sample.z};
68 const V3 p1{fo.sample.x, fo.sample.y, fo.sample.z};
69 const V3 si{fi.sideX, fi.sideY, fi.sideZ};
70 const V3 so{fo.sideX, fo.sideY, fo.sideZ};
71 const V3 ui{fi.upX, fi.upY, fi.upZ};
72 const V3 uo{fo.upX, fo.upY, fo.upZ};
73 V3 ti{fi.forwardX, fi.forwardY, fi.forwardZ};
74 V3 to{fo.forwardX, fo.forwardY, fo.forwardZ};
75 if (link.inDirection == RoadLaneDirection::Backward) ti = ti * -1.f;
76 if (link.outDirection == RoadLaneDirection::Backward) to = to * -1.f;
77 const V3 a = p0 + si * (inLat * fi.sample.scaleX) + ui * 0.09f;
78 const V3 d = p1 + so * (outLat * fo.sample.scaleX) + uo * 0.09f;
79 // A radius-sized handle works for ordinary junctions, but can overshoot
80 // badly when two short/acute arms leave their lane endpoints close
81 // together. Bound it by the endpoint chord so the cubic stays local.
82 const float handle = std::min(node.value().junctionRadius * 0.65f, length(d - a) * 0.5f);
83 const V3 b = a + ti * handle;
84 const V3 c = d - to * handle;
85
86 RoadPolyline poly;
87 poly.r = 0.25f;
88 poly.g = 0.95f;
89 poly.b = 0.45f;
90 poly.width = options.navRibbonHalfWidth * 2.f;
91 poly.inEdge = link.inEdge;
92 poly.outEdge = link.outEdge;
93 poly.inLane = link.inLane;
94 poly.outLane = link.outLane;
95 poly.inDirection = link.inDirection;
96 poly.outDirection = link.outDirection;
97 poly.speedLimitMps = std::min(inEdge.value().style.speedLimitMps, outEdge.value().style.speedLimitMps);
98 poly.trafficPriority = inEdge.value().style.trafficPriority;
99 const int samples = std::max(4, options.turnSamples);
100 V3 prev{};
101 bool hasPrev = false;
102 for (int i = 0; i <= samples; ++i) {
103 const float t = static_cast<float>(i) / static_cast<float>(samples);
104 const float u = 1.f - t;
105 const V3 p = a * (u * u * u) + b * (3.f * u * u * t) + c * (3.f * u * t * t) + d * (t * t * t);
106 poly.xyz.push_back(p.x);
107 poly.xyz.push_back(p.y);
108 poly.xyz.push_back(p.z);
109 if (hasPrev) {
110 const V3 delta = p - prev;
111 if (length(delta) > 1e-5f) {
112 const V3 fwd = normalize(delta);
113 const V3 up = normalize(ui + uo);
114 const V3 lat = normalize(cross(up, fwd));
115 const float hw = options.navRibbonHalfWidth * 0.9f;
116 appendStripQuad(mesh, prev + lat * -hw, p + lat * -hw, p + lat * hw, prev + lat * hw, up, 0.f, 1.f,
117 0.f, 1.f, RoadMaterial::Nav);
118 }
119 }
120 if (i == samples / 2 && length(d - a) > 1e-5f) {
121 // Follow the curve at the arrow position. Using the endpoint chord
122 // points across a hairpin and can visibly cross the turn ribbon.
123 const V3 derivative = (b - a) * (3.f * u * u) + (c - b) * (6.f * u * t) + (d - c) * (3.f * t * t);
124 const V3 fwd = length(derivative) > 1e-5f ? normalize(derivative) : normalize(d - a);
125 appendArrow(mesh, p, fwd, normalize(ui + uo), 0.7f);
126 }
127 prev = p;
128 hasPrev = true;
129 }
130 overlay.turns.push_back(std::move(poly));
131 if (isMeshBudgetExceeded(mesh, options))
132 return bakeFail<void>(DiagnosticCode::PreconditionViolation,
133 "road navigation exceeds the mesh element budget", "mesh");
134 }
135 return Result<void>::success();
136}
