45Result<std::vector<Cell>> Reachability::pathTo(Cell target) const {
46 if (!contains(target))
47 return failure<std::vector<Cell>>(DiagnosticCode::NotFound, "tactics target is not reachable", "target");
48 std::vector<Cell> result{target};
49 while (result.back() != origin_) {
50 const auto found = predecessor_.find(result.back());
51 if (found == predecessor_.end())
52 return failure<std::vector<Cell>>(DiagnosticCode::InvariantViolation,
53 "tactics reachability predecessor chain is incomplete", "path");
54 result.push_back(found->second);
55 }
56 std::reverse(result.begin(), result.end());
57 return Result<std::vector<Cell>>::success(std::move(result));

60Result<Reachability> PathQuery::reachable(const BoardState& board, SubjectRef subject, int budget) {
61 if (!subject.isValid())
62 return failure<Reachability>(DiagnosticCode::InvalidArgument,
63 "tactics reachability requires a valid subject", "subject");
64 if (budget < 0)
65 return failure<Reachability>(DiagnosticCode::InvalidArgument,
66 "tactics reachability budget must be non-negative", "budget");
67 const auto origin = board.position(subject);
68 if (!origin)
69 return failure<Reachability>(DiagnosticCode::NotFound, "tactics reachability subject is not placed",
70 "subject");
71
72 std::map<Cell, int> best;
73 std::map<Cell, Cell> predecessor;
74 std::priority_queue<FrontierNode, std::vector<FrontierNode>, FrontierLater> frontier;
75 best.emplace(*origin, 0);
76 frontier.push({0, *origin});
77
78 while (!frontier.empty()) {
79 const FrontierNode current = frontier.top();
80 frontier.pop();
81 const auto known = best.find(current.cell);
82 if (known == best.end() || known->second != current.cost) continue;
83 for (const Cell next : board.neighbours(current.cell)) {
84 const auto occupied = board.occupant(next);
85 if (occupied && *occupied != subject) continue;
86 auto state = board.cell(next);
87 if (!state) return Result<Reachability>::failure(state.status());
88 const CellState cellState = std::move(state).takeValue();
89 if (!cellState.passable || current.cost > std::numeric_limits<int>::max() - cellState.moveCost) continue;
90 const int candidate = current.cost + cellState.moveCost;
91 if (candidate > budget) continue;
92 const auto old = best.find(next);
93 if (old != best.end() && candidate >= old->second) continue;
94 best[next] = candidate;
95 predecessor[next] = current.cell;
96 frontier.push({candidate, next});
97 }
98 }
99
100 Reachability result;
101 result.origin_ = *origin;
102 result.predecessor_ = std::move(predecessor);
103 result.cells_.reserve(best.size());
104 for (const auto& [cellValue, cost] : best) result.cells_.push_back({cellValue, cost});
105 return Result<Reachability>::success(std::move(result));
106}

114Result<std::vector<Cell>> PathQuery::cellsInRange(const BoardState& board, Cell origin, int minimum, int maximum,
115 CellRangeMetric metric) {
116 if (minimum < 0 || maximum < minimum)
117 return failure<std::vector<Cell>>(DiagnosticCode::InvalidArgument,
118 "tactics range bounds must be ordered and non-negative", "range");
119 std::vector<Cell> result;
120 for (const Cell cell : board.cells()) {
121 const std::int64_t deltaX = static_cast<std::int64_t>(cell.x) - origin.x;
122 const std::int64_t deltaY = static_cast<std::int64_t>(cell.y) - origin.y;
123 const std::int64_t dx = std::abs(deltaX);
124 const std::int64_t dy = std::abs(deltaY);
125 const std::int64_t dz =
126 std::abs(static_cast<std::int64_t>(cell.layer) - origin.layer);
127 std::int64_t distance = 0;
128 switch (metric) {
129 case CellRangeMetric::Manhattan: distance = dx + dy + dz; break;
130 case CellRangeMetric::Chebyshev: distance = std::max({dx, dy, dz}); break;
131 case CellRangeMetric::Hex:
132 if (cell.layer != origin.layer) continue;
133 distance = std::max({dx, dy, std::abs(deltaX + deltaY)});
134 break;
135 }
136 if (distance >= minimum && distance <= maximum) result.push_back(cell);
137 }
138 return Result<std::vector<Cell>>::success(std::move(result));
139}
