30MeshParticleRenderInputs buildMeshParticleRenderInputs(
31 std::span<const MeshParticleInstance> instances, MeshEffectInstance* effect,
32 graphics::Mesh* mesh, graphics::Texture* albedo, const MeshVfxBatchKey& key,
33 std::uint64_t emitterStableId, const glm::vec3& cameraPosition,
34 float projectionScalePixels, float maximumDistance, std::int32_t priority) {
35 MeshParticleRenderInputs result;
36 result.lodCandidates.reserve(instances.size());
37 result.renderItems.reserve(instances.size());
38 result.commands.reserve(instances.size());
39 const float safeProjection = std::max(0.f, projectionScalePixels);
40 const float safeMaximumDistance = std::max(0.f, maximumDistance);
41 for (const auto& instance : instances) {
42 const glm::vec3 position(instance.model[3]);
43 const float distance = glm::length(position - cameraPosition);
44 const float scale = glm::length(glm::vec3(instance.model[0]));
45 const float projectedRadius = safeProjection * scale / std::max(distance, 1e-4f);
46 const std::uint64_t stableId =
47 (emitterStableId * 0x9e3779b97f4a7c15ull) ^ instance.stableId;
48 result.lodCandidates.push_back(
49 {stableId, distance, projectedRadius, priority,
50 std::isfinite(distance) && distance <= safeMaximumDistance});
51 result.renderItems.push_back({stableId, key, distance});
52 MeshVfxRendererCommand command;
53 command.stableInstanceId = stableId;
54 command.kind = MeshVfxRendererCommand::Kind::Particle;
55 command.effect = effect;
56 command.particle = {mesh, albedo, instance};
57 result.commands.push_back(std::move(command));
58 }
59 return result;
60}

62eve::Result<TrailUploadData> prepareTrailUpload(const TrailMeshSnapshot& snapshot) {
63 TrailUploadData upload;
64 upload.positions.reserve(snapshot.vertices.size() * 3u);
65 upload.normals.assign(snapshot.vertices.size() * 3u, 0.f);
66 upload.uvs.reserve(snapshot.vertices.size() * 2u);
67 upload.indices = snapshot.indices;
68
69 for (const TrailVertex& vertex : snapshot.vertices) {
70 if (!finite(vertex.position) || !std::isfinite(vertex.uv.x) || !std::isfinite(vertex.uv.y))
71 return failure<TrailUploadData>(eve::DiagnosticCode::InvalidArgument,
72 "trail snapshot contains a non-finite vertex");
73 upload.positions.insert(upload.positions.end(),
74 {vertex.position.x, vertex.position.y, vertex.position.z});
75 upload.uvs.insert(upload.uvs.end(), {vertex.uv.x, vertex.uv.y});
76 }
77 if (upload.indices.size() % 3u != 0u)
78 return failure<TrailUploadData>(eve::DiagnosticCode::InvalidArgument,
79 "trail snapshot index count is not triangular");
80
81 for (std::size_t i = 0; i < upload.indices.size(); i += 3u) {
82 const std::uint32_t ia = upload.indices[i];
83 const std::uint32_t ib = upload.indices[i + 1u];
84 const std::uint32_t ic = upload.indices[i + 2u];
85 if (ia >= snapshot.vertices.size() || ib >= snapshot.vertices.size() || ic >= snapshot.vertices.size())
86 return failure<TrailUploadData>(eve::DiagnosticCode::InvalidArgument,
87 "trail snapshot contains an out-of-range index");
88 const glm::vec3 a = snapshot.vertices[ia].position;
89 const glm::vec3 b = snapshot.vertices[ib].position;
90 const glm::vec3 c = snapshot.vertices[ic].position;
91 glm::vec3 normal = glm::cross(b - a, c - a);
92 const float length = glm::length(normal);
93 if (length > 1e-6f) normal /= length;
94 for (const std::uint32_t index : {ia, ib, ic}) {
95 upload.normals[index * 3u] += normal.x;
96 upload.normals[index * 3u + 1u] += normal.y;
97 upload.normals[index * 3u + 2u] += normal.z;
98 }
99 }
100 for (std::size_t i = 0; i < snapshot.vertices.size(); ++i) {
101 glm::vec3 normal(upload.normals[i * 3u], upload.normals[i * 3u + 1u],
102 upload.normals[i * 3u + 2u]);
103 const float length = glm::length(normal);
104 normal = length > 1e-6f ? normal / length : glm::vec3(0.f, 0.f, 1.f);
105 upload.normals[i * 3u] = normal.x;
106 upload.normals[i * 3u + 1u] = normal.y;
107 upload.normals[i * 3u + 2u] = normal.z;
108 }
109 return eve::Result<TrailUploadData>::success(std::move(upload));
110}

145eve::Result<MeshEffectSubmitStatus> MeshEffectRenderer::submitOverlay(
146 MeshEffectInstance& effect, const MeshEffectDrawSource& source) {
147 if (!effect.isBound())
148 return failure<MeshEffectSubmitStatus>(eve::DiagnosticCode::PreconditionViolation,
149 "mesh effect has no bound target");
150 if (source.target != effect.target())
151 return failure<MeshEffectSubmitStatus>(eve::DiagnosticCode::StaleHandle,
152 "resolved mesh source does not match the effect target");
153 if (!source.mesh)
154 return failure<MeshEffectSubmitStatus>(eve::DiagnosticCode::InvalidArgument,
155 "resolved mesh source has no mesh");
156 if (effect.intensity() <= 0.f)
157 return eve::Result<MeshEffectSubmitStatus>::success(MeshEffectSubmitStatus::SkippedInactive);
158
159 try {
160 graphics::Color tint = source.tint;
161 tint.a *= effect.intensity();
162 if (!batchActive_)
163 graphics_.setMesh3DSurface(graphics::SurfaceMode::Transparent, graphics::BlendMode::Additive,
164 false, true, 0.f);
165 requestSceneColorForRefraction(graphics_, effect);
166 graphics_.drawMeshShader(source.mesh, source.model, source.albedo, tint, shaderFor(effect));
167 } catch (const std::exception& error) {
168 return failure<MeshEffectSubmitStatus>(eve::DiagnosticCode::Failed,
169 std::string("mesh overlay submission failed: ") + error.what());
170 }
171 return eve::Result<MeshEffectSubmitStatus>::success(MeshEffectSubmitStatus::Drawn);

174eve::Result<MeshEffectSubmitStatus> MeshEffectRenderer::submitTrail(
175 MeshEffectInstance& effect, const TrailMeshSnapshot& snapshot, const glm::mat4& model,
176 const graphics::Color& tint) {
177 if (effect.intensity() <= 0.f)
178 return eve::Result<MeshEffectSubmitStatus>::success(MeshEffectSubmitStatus::SkippedInactive);
179 if (snapshot.vertices.empty() || snapshot.indices.empty())
180 return eve::Result<MeshEffectSubmitStatus>::success(MeshEffectSubmitStatus::SkippedEmpty);
181
182 auto prepared = prepareTrailUpload(snapshot);
183 if (!prepared.ok())
184 return eve::Result<MeshEffectSubmitStatus>::failure(prepared.status());
185 TrailUploadData upload = std::move(prepared).takeValue();
186
187 try {
188 if (!trailMesh_) {
189 trailMesh_ = graphics_.newMeshFromArrays(
190 upload.positions.data(), upload.normals.data(), upload.uvs.data(),
191 static_cast<int>(upload.positions.size() / 3u), upload.indices.data(),
192 static_cast<int>(upload.indices.size()));
193 if (!trailMesh_)
194 return failure<MeshEffectSubmitStatus>(eve::DiagnosticCode::Failed,
195 "graphics failed to create the trail mesh");
196 } else if (!graphics_.updateMeshVertices(
197 trailMesh_, upload.positions.data(), upload.normals.data(), upload.uvs.data(),
198 static_cast<int>(upload.positions.size() / 3u), upload.indices.data(),
199 static_cast<int>(upload.indices.size()))) {
200 return failure<MeshEffectSubmitStatus>(eve::DiagnosticCode::Unsupported,
201 "graphics backend rejected the dynamic trail upload");
202 }
203 graphics::Color drawTint = tint;
204 drawTint.a *= effect.intensity();
205 if (!batchActive_)
206 graphics_.setMesh3DSurface(graphics::SurfaceMode::Transparent, graphics::BlendMode::Additive,
207 false, true, 0.f);
208 requestSceneColorForRefraction(graphics_, effect);
209 graphics_.drawMeshShader(trailMesh_, model, nullptr, drawTint, shaderFor(effect));
210 } catch (const std::exception& error) {
211 return failure<MeshEffectSubmitStatus>(eve::DiagnosticCode::Failed,
212 std::string("trail submission failed: ") + error.what());
213 }
214 return eve::Result<MeshEffectSubmitStatus>::success(MeshEffectSubmitStatus::Drawn);
215}

243 if (batchActive_ || !activeCommands_.empty())
244 return failure<MeshVfxSubmissionReport>(eve::DiagnosticCode::PreconditionViolation,
245 "mesh VFX queue submission is reentrant");

253 "mesh VFX renderer commands contain a duplicate stable ID");
254 }
255 }

258 activeCommands_.clear();
259 batchActive_ = false;
260 return eve::Result<MeshVfxSubmissionReport>::success(std::move(report));

277MeshVfxSubmitStatus MeshEffectRenderer::submitDraw(const MeshVfxBatchedDraw& draw) {
278 if (!batchActive_)
279 return MeshVfxSubmitStatus::Failed;
280 const auto found = activeCommands_.find(draw.stableInstanceId);
281 if (found == activeCommands_.end())
282 return MeshVfxSubmitStatus::Skipped;
283 const auto& command = *found->second;
284 eve::Result<MeshEffectSubmitStatus> submitted =
285 command.kind == MeshVfxRendererCommand::Kind::Trail
286 ? submitTrail(*command.effect, command.trail, command.trailModel, command.trailTint)
287 : command.kind == MeshVfxRendererCommand::Kind::Particle
288 ? submitParticle(*command.effect, command.particle)
289 : submitOverlay(*command.effect, command.overlay);
290 if (!submitted.ok())
291 return MeshVfxSubmitStatus::Failed;
292 return submitted.value() == MeshEffectSubmitStatus::Drawn ? MeshVfxSubmitStatus::Accepted
293 : MeshVfxSubmitStatus::Skipped;
294}
