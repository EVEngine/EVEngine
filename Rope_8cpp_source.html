120eve::Result<Rope3D*> Rope::newRope3DFromJson(const std::string& json) {
121 auto registered = registerRope3DCreateSchema();
122 if (!registered.ok()) return eve::Result<Rope3D*>::failure(registered.status());
123 const auto errors = eve::schema::SchemaRegistry::validate(kRopeCreateSchemaId, 1, json);
124 if (!errors.empty())
125 return ropeCreateFailure<Rope3D*>(eve::DiagnosticCode::InvalidArgument, errors.front().message,
126 errors.front().path);
127 std::string parseError;
128 auto document = eve::json::Document::parse(json, &parseError);
129 if (!document.valid()) return ropeCreateFailure<Rope3D*>(eve::DiagnosticCode::ParseError, std::move(parseError));
130 const auto root = document.root();
131 try {
132 auto rope = std::make_unique<Rope3D>(root.getInt("particleCount"), root.getFloat("startX"),
133 root.getFloat("startY"), root.getFloat("startZ"), root.getFloat("endX"),
134 root.getFloat("endY"), root.getFloat("endZ"));
135 rope->setGravity(root.getFloat("gravityX", 0.f), root.getFloat("gravityY", -9.81f),
136 root.getFloat("gravityZ", 0.f));
137 rope->setStretchCompliance(root.getFloat("stretchCompliance", 0.f));
138 rope->setDistanceConstraintsEnabled(root.getBool("distanceConstraintsEnabled", true));
139 rope->setBendCompliance(root.getFloat("bendCompliance", 0.002f));
140 rope->setBendConstraintsEnabled(root.getBool("bendConstraintsEnabled", true));
141 rope->setMaxBending(root.getFloat("maxBending", 0.025f));
142 rope->setPlasticity(root.getFloat("plasticYield", 0.f), root.getFloat("plasticCreep", 0.f));
143 rope->setMaxCompression(root.getFloat("maxCompression", 0.f));
144 rope->setDamping(root.getFloat("damping", 0.01f));
145 rope->setParticleMass(root.getFloat("particleMass", 0.1f));
146 rope->setRadius(root.getFloat("radius", 0.04f));
147 rope->setCollisionFriction(root.getFloat("collisionFriction", 0.f));
148 rope->setCollisionRestitution(root.getFloat("collisionRestitution", 0.f));
149 rope->setContinuousCollision(root.getBool("continuousCollision", true));
150 rope->setSelfCollision(root.getBool("selfCollision", true));
151 if (root.getBool("pinStart", false)) {
152 auto pinned = rope->pin(0);
153 if (!pinned.ok()) return eve::Result<Rope3D*>::failure(pinned.status());
154 }
155 if (root.getBool("pinEnd", false)) {
156 auto pinned = rope->pin(rope->getParticleCount() - 1);
157 if (!pinned.ok()) return eve::Result<Rope3D*>::failure(pinned.status());
158 }
159 if (root.getBool("tearingEnabled", false))
160 rope->setTearing(root.getFloat("tearResistance", 1000.f), root.getInt("maxTearsPerStep", 1));
161 return eve::Result<Rope3D*>::success(rope.release(), eve::Status::success(eve::StatusCode::Applied));
162 } catch (const std::exception& exception) {
163 return ropeCreateFailure<Rope3D*>(eve::DiagnosticCode::InvalidArgument, exception.what());
164 }
165}
