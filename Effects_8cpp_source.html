30eve::Result<EffectDefinition> effectDefinitionFromBinding(const std::string& type, int priority, float duration,
31 const std::string& stackKey,
32 const std::string& policyNameValue) {
33 StackPolicy policy;
34 if (!parsePolicy(policyNameValue, policy))
35 return effectsBindingFailure<EffectDefinition>(eve::DiagnosticCode::InvalidArgument,
36 "unknown effect stack policy", "policy");
37
38 EffectDefinition definition;
39 definition.id = type;
40 definition.stackKey = stackKey;
41 definition.priority = priority;
42 definition.duration = duration;
43 switch (policy) {
44 case StackPolicy::Replace: definition.policy.stackMode = StackMode::Replace; break;
45 case StackPolicy::Stack: definition.policy.stackMode = StackMode::NewInstance; break;
46 case StackPolicy::Refresh: definition.policy.stackMode = StackMode::Reuse; break;
47 }
48 auto validation = definition.validate();
49 if (!validation.ok()) return eve::Result<EffectDefinition>::failure(validation.status());
50 return eve::Result<EffectDefinition>::success(std::move(definition));
51}

61template <class Ref, class Proxy, class Release>
62ssq::Table makeOwnedProxy(HSQUIRRELVM vm, eve::Result<Ref>&& reference, Release&& release) {
63 if (!reference) return eve::script::projectStatusResult(vm, reference.status(), false, false);
64 const Ref ref = std::move(reference).takeValue();
65 auto object = eve::script::makeOwnedSquirrelInstance<Proxy>(vm, std::make_unique<Proxy>(ref));
66 if (!object) {
67 const eve::Status status = object.status();
68 object.ignore("failed to create owned effects proxy");
69 std::invoke(std::forward<Release>(release), ref).ignore("rollback failed owned effects allocation");
70 return eve::script::projectStatusResult(vm, status, false, false);
71 }
72 ssq::Object owned = std::move(object).takeValue();
73 auto result = eve::script::projectStatusResult(vm, eve::Status::success(eve::StatusCode::Applied), true, false);
74 result.set("value", owned);
75 result.set("ownership", std::string("owned"));
76 result.set("ownerEpoch", static_cast<std::int64_t>(ref.ownerEpoch));
77 result.set("handle", static_cast<std::int64_t>(ref.packed()));
78 return result;
79}
