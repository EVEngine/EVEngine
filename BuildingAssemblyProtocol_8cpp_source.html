9Result<void> checkWirePayload(const Value& value, size_t& remaining, size_t depth = 0) {
10 if (remaining == 0 || (depth >= 64 && (value.isArray() || value.isObject())) ||
11 (value.isDouble() && !std::isfinite(value.asDouble())))
12 return fail<void>("House protocol response exceeds its finite 100000-value / 64-depth budget");
13 --remaining;
14 if (value.isArray()) {
15 for (size_t i = 0; i < value.arraySize(); ++i) {
16 auto result = checkWirePayload(value.at(i), remaining, depth + 1);
17 if (!result.ok()) return result;
18 }
19 } else if (value.isObject()) {
20 for (const auto& key : value.keys()) {
21 auto result = checkWirePayload(*value.find(key), remaining, depth + 1);
22 if (!result.ok()) return result;
23 }
24 }
25 return Result<void>::success();
26}

29Result<Value> executeHouseRequest(PlacementWorld& w, const Value& request) {
30 const auto* op = request.find("op");
31 const auto* version = request.find("version");
32 if (!op || !op->isString() || !version || !version->isInt64() || version->asInt() != 1)
33 return fail<Value>("House request requires version 1 and operation");
34 auto name = op->asString();
35 if (name == "encode") {
36 auto value = request.find("value");
37 if (!value) return fail<Value>("Missing value to encode");
38 auto json = value->toJson();
39 if (!json.ok()) return Result<Value>::failure(json.status());
40 return Result<Value>::success(std::move(json).takeValue());
41 }
42 if (name == "decode") {
43 auto json = request.find("json");
44 if (!json || !json->isString()) return fail<Value>("Missing JSON text");
45 return Value::fromJson(json->asString());
46 }
47 if (name == "validateBlueprint") {
48 auto value = request.find("blueprint");
49 if (!value) return fail<Value>("Missing blueprint");
50 auto blueprint = AssemblyService::decodeBlueprint(*value);
51 if (!blueprint.ok()) return Result<Value>::failure(blueprint.status());
52 return AssemblyService::encodeBlueprint(blueprint.value());
53 }
54 if (name == "snapshot") return AssemblyService::snapshotWorld(w);
55 if (name == "restore" || name == "applyReceipt") {
56 auto desired = request.find("state");
57 if (!desired) return fail<Value>("Missing desired state");
58 auto expected = request.find("expected");
59 if (name == "applyReceipt" && !expected) return fail<Value>("Receipt needs expected state");
60 auto result = expected ? AssemblyService::applyReceipt(w, *expected, *desired)
61 : AssemblyService::restoreWorld(w, *desired);
62 if (!result.ok()) return Result<Value>::failure(result.status());
63 return Result<Value>::success(Value::Object{});
64 }
65 if (name == "capture") {
66 auto id = request.find("blueprintId"), ids = request.find("members"), origin = request.find("pose");
67 if (!id || !id->isString() || !ids || !ids->isArray() || !origin)
68 return fail<Value>("Capture needs blueprintId, members and pose");
69 auto guid = AssetGuid::parse(id->asString());
70 auto pose = parsePose(*origin);
71 std::vector<int> members;
72 const bool poseOk = pose.ok();
73 if (!guid || !poseOk) return fail<Value>("Invalid blueprint identity or capture pose");
74 for (const auto& member : *ids->getIf<Value::Array>()) {
75 if (!member.isInt64() || member.asInt() <= 0 || member.asInt() > std::numeric_limits<int>::max())
76 return fail<Value>("Invalid member key");
77 members.push_back(int(member.asInt()));
78 }
79 auto b = AssemblyService::capture(w, members, *guid, pose.value());
80 if (!b.ok()) return Result<Value>::failure(b.status());
81 return AssemblyService::encodeBlueprint(b.value());
82 }
83 auto id = request.find("assemblyId");
84 if (!id || !id->isString()) return fail<Value>("Missing house identity");
85 auto identity = AssemblyId::parse(id->asString());
86 if (!identity) return fail<Value>("Invalid house identity");
87 if (name == "members") {
88 auto a = AssemblyService::assembly(w, *identity);
89 if (!a.ok()) return Result<Value>::failure(a.status());
90 Value::Array parts;
91 for (const auto& [key, member] : a.value().members)
92 parts.emplace_back(codec::instanceValue(codec::snapshot(w.buildings().at(member))));
93 return Result<Value>::success(std::move(parts));
94 }
95 auto prepare = [&]() -> Result<PreparedAssembly> {
96 if (name == "remove") return AssemblyService::prepareRemoval(w, *identity);
97 if (name == "remodel") {
98 auto members = request.find("members");
99 if (!members || !members->isArray()) return fail<PreparedAssembly>("Missing remodel members");
100 std::vector<PlacedBuilding> parts;
101 for (const auto& value : *members->getIf<Value::Array>()) {
102 // Canonical instance decoder requires a positive record id; new members use
103 // explicit new=true and discard that temporary wire key before admission.
104 auto p = codec::parseInstance(value);
105 if (!p.ok()) return Result<PreparedAssembly>::failure(p.status());
106 auto runtime = codec::runtime(p.value());
107 if (auto add = value.find("new"); add && add->isBool() && add->asBool()) runtime.instanceId = 0;
108 parts.push_back(std::move(runtime));
109 }
110 return AssemblyService::prepareRemodel(w, *identity, parts);
111 }
112 auto poseValue = request.find("pose");
113 if (!poseValue) return fail<PreparedAssembly>("Missing target pose");
114 auto pose = parsePose(*poseValue);
115 if (!pose.ok()) return Result<PreparedAssembly>::failure(pose.status());
116 if (name == "move") return AssemblyService::prepareMove(w, *identity, pose.value());
117 if (name == "place" || name == "preview") {
118 auto blueprint = request.find("blueprint");
119 if (!blueprint) return fail<PreparedAssembly>("Missing blueprint");
120 auto b = AssemblyService::decodeBlueprint(*blueprint);
121 if (!b.ok()) return Result<PreparedAssembly>::failure(b.status());
122 return AssemblyService::preparePlacement(w, b.value(), *identity, pose.value());
123 }
124 return fail<PreparedAssembly>("Unknown house operation", DiagnosticCode::Unsupported);
125 };
126 auto token = prepare();
127 if (!token.ok()) return Result<Value>::failure(token.status());
128 if (name == "preview") {
129 Value::Array parts;
130 for (const auto& p : token.value().preview()) parts.emplace_back(codec::instanceValue(codec::snapshot(p)));
131 return Result<Value>::success(std::move(parts));
132 }
133 // Reserve the complete wire receipt before publishing the mutation. A script must
134 // never receive ok/null after the native world has already changed.
135 auto after = AssemblyService::snapshotWorld(*token.value().data_->candidate);
136 if (!after.ok()) return Result<Value>::failure(after.status());
137 size_t remaining = 99998; // receipt object and assemblyId
138 auto beforeBudget = checkWirePayload(token.value().data_->before, remaining, 1);
139 if (!beforeBudget.ok()) return Result<Value>::failure(beforeBudget.status());
140 auto afterBudget = checkWirePayload(after.value(), remaining, 1);
141 if (!afterBudget.ok()) return Result<Value>::failure(afterBudget.status());
142 auto committed = AssemblyService::commit(w, std::move(token).takeValue());
143 if (!committed.ok()) return Result<Value>::failure(committed.status());
144 auto receipt = std::move(committed).takeValue();
145 return Result<Value>::success(Value::Object{{"assemblyId", receipt.assembly.format()},
146 {"before", std::move(receipt.before)},
147 {"after", std::move(receipt.after)}});
148}
