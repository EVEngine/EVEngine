35eve::Result<double> numberField(const eve::Value::Object& object, std::string_view name, double fallback,
36 std::string path) {
37 const auto* value = field(object, name);
38 if (value == nullptr) return eve::Result<double>::success(fallback);
39 double number = 0.0;
40 if (const auto* real = value->getIf<double>())
41 number = *real;
42 else if (const auto* integer = value->getIf<std::int64_t>())
43 number = static_cast<double>(*integer);
44 else
45 return invalid<double>("field must be a number", std::move(path));
46 if (!std::isfinite(number)) return invalid<double>("field must be finite", std::move(path));
47 return eve::Result<double>::success(number);
48}

50eve::Result<std::vector<std::string>> stringsField(const eve::Value::Object& object, std::string_view name,
51 std::string path) {
52 const auto* value = field(object, name);
53 if (value == nullptr) return eve::Result<std::vector<std::string>>::success({});
54 const auto* array = value->getIf<eve::Value::Array>();
55 if (array == nullptr) return invalid<std::vector<std::string>>("field must be an array of strings", path);
56 std::vector<std::string> result;
57 result.reserve(array->size());
58 for (std::size_t index = 0; index < array->size(); ++index) {
59 const auto* string = (*array)[index].getIf<std::string>();
60 if (string == nullptr)
61 return invalid<std::vector<std::string>>("array item must be a string",
62 path + "[" + std::to_string(index) + "]");
63 result.push_back(*string);
64 }
65 return eve::Result<std::vector<std::string>>::success(std::move(result));
66}

91eve::Result<eve::Value::Object> ruleObject(const effects::EffectInstance& effect, std::string path) {
92 const auto* encoded = field(effect.payload.object(), "settlement.rule");
93 if (encoded == nullptr) return eve::Result<eve::Value::Object>::success({});
94 if (const auto* object = encoded->getIf<eve::Value::Object>())
95 return eve::Result<eve::Value::Object>::success(*object);
96 if (const auto* json = encoded->getIf<std::string>()) {
97 auto parsed = eve::Value::fromJson(*json);
98 if (!parsed) return eve::Result<eve::Value::Object>::failure(parsed.status());
99 auto value = std::move(parsed).takeValue();
100 const auto* object = value.getIf<eve::Value::Object>();
101 if (object == nullptr) return invalid<eve::Value::Object>("settlement.rule JSON must be an object", path);
102 return eve::Result<eve::Value::Object>::success(*object);
103 }
104 return invalid<eve::Value::Object>("settlement.rule must be an object or JSON object string", std::move(path));
105}

109eve::Result<SettlementRuleSet> projectEffectRules(const effects::EffectContainer& effects, std::string_view scope) {
110 if (scope.empty()) return invalid<SettlementRuleSet>("effect rule scope must not be empty", "scope");
111 std::vector<SettlementRule> rules;
112 for (int index = 0; index < effects.effectCount(); ++index) {
113 const auto* effect = effects.effectAt(index);
114 if (effect == nullptr)
115 return invalid<SettlementRuleSet>("effect container returned an empty active slot",
116 "effects[" + std::to_string(index) + "]");
117 if (!effect->payload.has("settlement.rule")) continue;
118 const auto path = "effects[" + std::to_string(index) + "].payload.settlement.rule";
119 auto objectResult = ruleObject(*effect, path);
120 if (!objectResult) return eve::Result<SettlementRuleSet>::failure(objectResult.status());
121 auto object = std::move(objectResult).takeValue();
122
123 auto stageName = stringField(object, "stage", path + ".stage", true);
124 if (!stageName) return eve::Result<SettlementRuleSet>::failure(stageName.status());
125 auto operationName = stringField(object, "operation", path + ".operation", true);
126 if (!operationName) return eve::Result<SettlementRuleSet>::failure(operationName.status());
127 auto stage = parseStage(stageName.value(), path + ".stage");
128 if (!stage) return eve::Result<SettlementRuleSet>::failure(stage.status());
129 auto operation = parseOperation(operationName.value(), path + ".operation");
130 if (!operation) return eve::Result<SettlementRuleSet>::failure(operation.status());
131 auto value = numberField(object, "value", effect->magnitude, path + ".value");
132 if (!value) return eve::Result<SettlementRuleSet>::failure(value.status());
133 auto perStack = numberField(object, "value_per_extra_stack", 0.0, path + ".value_per_extra_stack");
134 if (!perStack) return eve::Result<SettlementRuleSet>::failure(perStack.status());
135 auto when = stringField(object, "when", path + ".when");
136 if (!when) return eve::Result<SettlementRuleSet>::failure(when.status());
137 auto kinds = stringsField(object, "kinds", path + ".kinds");
138 if (!kinds) return eve::Result<SettlementRuleSet>::failure(kinds.status());
139 auto required = stringsField(object, "required_tags", path + ".required_tags");
140 if (!required) return eve::Result<SettlementRuleSet>::failure(required.status());
141 auto excluded = stringsField(object, "excluded_tags", path + ".excluded_tags");
142 if (!excluded) return eve::Result<SettlementRuleSet>::failure(excluded.status());
143
144 SettlementRule rule;
145 rule.id = std::string(scope) + "." + effect->id;
146 rule.source = effect->source.empty() ? effect->type : effect->source;
147 rule.stage = stage.value();
148 rule.priority = effect->priority;
149 rule.operation = operation.value();
150 rule.value = value.value();
151 rule.valuePerExtraStack = perStack.value();
152 rule.stacks = effect->stackCount;
153 rule.filter.kinds = std::move(kinds).takeValue();
154 rule.filter.requiredTags = std::move(required).takeValue();
155 rule.filter.excludedTags = std::move(excluded).takeValue();
156 rule.when = std::move(when).takeValue();
157 rules.push_back(std::move(rule));
158 }
159 SettlementRuleSet result;
160 auto configured = result.configure(std::move(rules));
161 if (!configured) return eve::Result<SettlementRuleSet>::failure(configured.status());
162 return eve::Result<SettlementRuleSet>::success(std::move(result));
163}
