50Result<int> PcgLoadingScreen::addMissingScene(const std::string& terrainName, const std::string& impostorName) {
51 if (!timeoutOpen_)
52 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
53 "beginTimeout must be called before adding a missing scene", {}, {}, "ui.pcgLoadingScreen"));
54 if ((!terrainName.empty() && !validName(terrainName)) || (!impostorName.empty() && !validName(impostorName)))
55 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
56 "missing-scene names must not contain null characters", {}, {}, "ui.pcgLoadingScreen"));
57 pendingMissingScenes_.push_back({terrainName, impostorName, {}, {}});
58 return Result<int>::success(static_cast<int>(pendingMissingScenes_.size() - 1));
59}

61Result<void> PcgLoadingScreen::addReference(int sceneIndex, const std::string& objectName, bool impostor) {
62 if (!timeoutOpen_ || sceneIndex < 0 || static_cast<size_t>(sceneIndex) >= pendingMissingScenes_.size() ||
63 !validName(objectName))
64 return invalidLoadingScreen("an open timeout, valid scene index and nonempty object name are required");
65 auto& scene = pendingMissingScenes_[static_cast<size_t>(sceneIndex)];
66 (impostor ? scene.impostorReferences : scene.regularReferences).push_back(objectName);
67 return Result<void>::success();

78Result<void> PcgLoadingScreen::endTimeout() {
79 if (!timeoutOpen_) return invalidLoadingScreen("beginTimeout must be called before endTimeout");
80 std::string message = "Pcg Loading Screen Timeout, closing down loading screen.\r\n";
81 message += "If you feel that there is no error but the terrain loading simply needs more time, try to increase the Timeout value in the Terrain Loader Manager.\r\n";
82 message += "The following scenes are still not loaded:\r\n";
83 for (const auto& scene : pendingMissingScenes_) {
84 if (!scene.regularReferences.empty()) {
85 message += "\r\n " + scene.terrainName + "\r\n Referencing Objects:\r\n";
86 for (const auto& name : scene.regularReferences) message += name + ", ";
87 }
88 if (!scene.impostorReferences.empty()) {
89 message += "\r\n " + scene.impostorName + "\r\n Referencing Objects:\r\n";
90 for (const auto& name : scene.impostorReferences) message += name + ", ";
91 }
92 message += "\r\n";
93 }
94 timeoutMessage_ = std::move(message);
95 pendingMissingScenes_.clear();
96 timeoutOpen_ = false;
97 onLoadProgressEnded();
98 return Result<void>::success();
99}

101Result<void> PcgLoadingScreen::tick(float deltaTime) {
102 if (!std::isfinite(deltaTime) || deltaTime < 0.f)
103 return invalidLoadingScreen("frame delta time must be finite and non-negative");
104 if (fading_) {
105 backgroundAlpha_ -= deltaTime * fadeOutSpeed_;
106 if (backgroundAlpha_ <= 0.f) {
107 backgroundAlpha_ = 0.f;
108 canvasVisible_ = false;
109 fading_ = false;
110 }
111 }
112 return Result<void>::success();
113}
