45std::vector<std::string> Scene::collectFrustumIdsAt(const std::string &hostName, graphics::Camera3D *cam, float viewW,
46 float viewH) const {
47 SceneHost *h = resolveHost(hostName);
48 std::vector<std::string> out;
49 if (!h || !cam || viewW <= 0.f || viewH <= 0.f) return out;
50 auto *projection = eve::cap::query<ISceneCameraProjection>();
51 if (!projection) return out;
52 const auto clip = projection->clipForViewport(*cam, viewW, viewH);
53 if (!clip) return out;
54
55 h->walkDepthFirst([&](SceneHost *, int, SceneNode &n) {
56 if (!n.hasBounds) return;
57 if (aabbIntersectsFrustum(clip->clip, clip->inverse, worldBoundsOf(n))) {
58 out.push_back(n.id);
59 }
60 });
61 return out;
62}

64Result<int> Scene::applyPcgTerrainCullingAt(const std::string &hostName, graphics::Camera3D *cam, float viewW,
65 float viewH, const std::string &tag) {
66 SceneHost *host = resolveHost(hostName);
67 if (!host || !cam || !std::isfinite(viewW) || !std::isfinite(viewH) || viewW <= 0.f || viewH <= 0.f || tag.empty())
68 return Result<int>::failure(Diagnostic::error(
69 DiagnosticCode::InvalidArgument, "Pcg terrain culling requires host, camera, positive viewport and tag",
70 "scene.pcgTerrainCulling"));
71 auto *projection = eve::cap::query<ISceneCameraProjection>();
72 if (!projection)
73 return Result<int>::failure(Diagnostic::error(
74 DiagnosticCode::Unsupported, "Pcg terrain culling needs the graphics camera projection capability",
75 "scene.pcgTerrainCulling"));
76 const auto clip = projection->clipForViewport(*cam, viewW, viewH);
77 if (!clip)
78 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
79 "Pcg terrain culling requires a usable camera viewport",
80 "scene.pcgTerrainCulling"));
81
82 std::vector<std::pair<std::string, bool>> changes;
83 host->walkDepthFirst([&](SceneHost *, int, SceneNode &node) {
84 if (!node.hasBounds || std::find(node.tags.begin(), node.tags.end(), tag) == node.tags.end()) return;
85 const bool visible = aabbIntersectsFrustum(clip->clip, clip->inverse, worldBoundsOf(node));
86 if (node.visible != visible) changes.emplace_back(node.id, visible);
87 });
88 for (const auto &[id, visible] : changes) {
89 auto node = host->findById(id);
90 if (!node) continue;
91 node.value()->visible = visible;
92 host->markSubtreeDirtyById(id);
93 }
94 return Result<int>::success(static_cast<int>(changes.size()));
95}

111 SceneComponent *component = nullptr; // Owner invalidates before destruction; never exposed as a handle.
112 bool queued = false;

119void flushPendingComponents() {
120 if (flushing) return;
121 flushing = true;
122 struct FlushGuard {
123 ~FlushGuard() {
124 draining.clear();
125 flushing = false;
126 }
127 } guard;
128 draining.swap(pending);
129 for (size_t i = 0; i < draining.size(); ++i) {
130 auto &entry = draining[i];
131 auto registration = entry.lock();
132 if (!registration || !registration->component) continue;
133 registration->queued = false;
134 try {
135 registration->component->updateIfDirty();
136 } catch (...) {
137 pending.insert(pending.end(), draining.begin() + i + 1, draining.end());
138 throw;
139 }
140 }
141}
142} // namespace detail

231int registerLinkKind(const LinkOps &ops) {
232 if (!ops.kind || !ops.pushWorld) return -1;
233 const int existing = findLinkKind(ops.kind);
234 if (existing >= 0) return existing;
235 table().push_back({ops.kind, ops});
236 // Point the stored ops at our own copy of the name so a caller may pass a
237 // temporary.
238 Entry &e = table().back();
239 e.ops.kind = e.name.c_str();
240 return int(table().size()) - 1;
241}
242
243int findLinkKind(const char *kind) {
244 if (!kind) return -1;
245 const auto &t = table();
246 for (size_t i = 0; i < t.size(); ++i)
247 if (t[i].name == kind) return int(i);
248 return -1;
249}
250
251const LinkOps *linkOps(int kindId) {
252 const auto &t = table();
253 if (kindId < 0 || size_t(kindId) >= t.size()) return nullptr;
254 return &t[size_t(kindId)].ops;
255}
256
257const char *linkKindName(int kindId) {
258 const LinkOps *ops = linkOps(kindId);
259 return ops ? ops->kind : "";
260}
