1// Picking / culling entry points whose camera math belongs to graphics.

3// The traversal and mutation stay here because the scene owns them; only the

8// graphics (L4) cannot be imported by scene (L1).

28std::string Scene::pickScreenAt(const std::string &hostName, graphics::Camera3D *cam, float screenX, float screenY,
29 float viewW, float viewH) const {
30 if (!cam) return {};
31 auto *projection = eve::cap::query<ISceneCameraProjection>();
32 if (!projection) return {};
33 const auto ray = projection->screenRay(*cam, screenX, screenY, viewW, viewH);
34 if (!ray) return {};
35 return pickRayAt(hostName, ray->origin.x, ray->origin.y, ray->origin.z, ray->direction.x, ray->direction.y,
36 ray->direction.z);
37}

39std::vector<std::string> Scene::collectFrustumIdsAt(const std::string &hostName, graphics::Camera3D *cam, float viewW,
40 float viewH) const {
41 SceneHost *h = resolveHost(hostName);
42 std::vector<std::string> out;
43 if (!h || !cam || viewW <= 0.f || viewH <= 0.f) return out;
44 auto *projection = eve::cap::query<ISceneCameraProjection>();
45 if (!projection) return out;
46 const auto clip = projection->clipForViewport(*cam, viewW, viewH);
47 if (!clip) return out;
48
49 h->walkDepthFirst([&](SceneHost *, int, SceneNode &n) {
50 if (!n.hasBounds) return;
51 if (aabbIntersectsFrustum(clip->clip, clip->inverse, worldBoundsOf(n))) {
52 out.push_back(n.id);
53 }
54 });
55 return out;
56}

58Result<int> Scene::applyPcgTerrainCullingAt(const std::string &hostName, graphics::Camera3D *cam, float viewW,
59 float viewH, const std::string &tag) {
60 SceneHost *host = resolveHost(hostName);
61 if (!host || !cam || !std::isfinite(viewW) || !std::isfinite(viewH) || viewW <= 0.f || viewH <= 0.f || tag.empty())
62 return Result<int>::failure(Diagnostic::error(
63 DiagnosticCode::InvalidArgument, "Pcg terrain culling requires host, camera, positive viewport and tag",
64 "scene.pcgTerrainCulling"));
65 auto *projection = eve::cap::query<ISceneCameraProjection>();
66 if (!projection)
67 return Result<int>::failure(Diagnostic::error(
68 DiagnosticCode::Unsupported, "Pcg terrain culling needs the graphics camera projection capability",
69 "scene.pcgTerrainCulling"));
70 const auto clip = projection->clipForViewport(*cam, viewW, viewH);
71 if (!clip)
72 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
73 "Pcg terrain culling requires a usable camera viewport",
74 "scene.pcgTerrainCulling"));
75
76 std::vector<std::pair<std::string, bool>> changes;
77 host->walkDepthFirst([&](SceneHost *, int, SceneNode &node) {
78 if (!node.hasBounds || std::find(node.tags.begin(), node.tags.end(), tag) == node.tags.end()) return;
79 const bool visible = aabbIntersectsFrustum(clip->clip, clip->inverse, worldBoundsOf(node));
80 if (node.visible != visible) changes.emplace_back(node.id, visible);
81 });
82 for (const auto &[id, visible] : changes) {
83 auto node = host->findById(id);
84 if (!node) continue;
85 node.value()->visible = visible;
86 host->markSubtreeDirtyById(id);
87 }
88 return Result<int>::success(static_cast<int>(changes.size()));
89}
