60WidgetDesc ScenePanel::nodeTree(const eve::SceneNodeInfo& node,
61 const std::vector<eve::SceneNodeInfo>& all) {
62 std::vector<WidgetDesc> children;
63 children.push_back(button("select " + node.name + "##sel_" + node.id,
64 "sel_" + node.id,
65 [this, id = node.id]() { selectNode(id); }));
66 for (const std::string& childId : node.children) {
67 const auto found =
68 std::find_if(all.begin(), all.end(),
69 [&](const eve::SceneNodeInfo& n) { return n.id == childId; });
70 if (found == all.end()) continue;
71 children.push_back(nodeTree(*found, all));
72 }
73 return collapsingHeader(
74 node.name + "##hdr_" + node.id, std::move(children), "node_" + node.id,
75 false);

78WidgetDesc ScenePanel::build() {
79 std::vector<WidgetDesc> children;
80 eve::ISceneQuery* scene = eve::cap::query<eve::ISceneQuery>();
81 if (!scene) {
82 children.push_back(text(
83 "Scene module unavailable. This build has no scene support.",
84 "scene_missing"));
85 return window("Scene", std::move(children), "root");
86 }
87
88 const std::vector<eve::SceneNodeInfo> nodes = scene->nodes(512);
89 const std::string hostName = scene->activeHost();
90 children.push_back(row(
91 {
92 text("Scene: " + (hostName.empty() ? std::string("(none)") : hostName),
93 "scene_host"),
94 spacer("scene_spacer"),
95 button("Refresh##scene_refresh", "scene_refresh",
96 [this]() { refresh(); }),
97 },
98 "scene_toolbar"));
99 children.push_back(separator("scene_sep"));
100
101 // Tree roots (nodes without a parent present in the snapshot).
102 for (const eve::SceneNodeInfo& node : nodes) {
103 const bool hasParent =
104 !node.parent.empty() &&
105 std::find_if(nodes.begin(), nodes.end(),
106 [&](const eve::SceneNodeInfo& n) {
107 return n.id == node.parent;
108 }) != nodes.end();
109 if (!hasParent) children.push_back(nodeTree(node, nodes));
110 }
111
112 // Selected node property sheet.
113 if (!selectedId_.empty()) {
114 eve::SceneNodeInfo info;
115 if (scene->getNode(selectedId_, &info)) {
116 const std::string base = "node_" + info.id;
117 children.push_back(separator("scene_sel_sep"));
118 children.push_back(text("id: " + info.id, base + "_id"));
119 children.push_back(text("path: " + info.path, base + "_path"));
120 children.push_back(row(
121 {
122 text("x", base + "_lx"),
123 inputText("##" + base + "_x", std::to_string(info.x),
124 base + "_x",
125 [id = info.id, scene](const std::string& t) {
126 eve::SceneNodeInfo cur;
127 if (!scene->getNode(id, &cur)) return;
128 cur.x = static_cast<float>(std::atof(t.c_str()));
129 scene->setNodeTransform(id, cur.x, cur.y, cur.z);
130 }),
131 text("y", base + "_ly"),
132 inputText("##" + base + "_y", std::to_string(info.y),
133 base + "_y",
134 [id = info.id, scene](const std::string& t) {
135 eve::SceneNodeInfo cur;
136 if (!scene->getNode(id, &cur)) return;
137 cur.y = static_cast<float>(std::atof(t.c_str()));
138 scene->setNodeTransform(id, cur.x, cur.y, cur.z);
139 }),
140 text("z", base + "_lz"),
141 inputText("##" + base + "_z", std::to_string(info.z),
142 base + "_z",
143 [id = info.id, scene](const std::string& t) {
144 eve::SceneNodeInfo cur;
145 if (!scene->getNode(id, &cur)) return;
146 cur.z = static_cast<float>(std::atof(t.c_str()));
147 scene->setNodeTransform(id, cur.x, cur.y, cur.z);
148 }),
149 },
150 base + "_transform"));
151 children.push_back(row(
152 {
153 checkbox("visible##" + base + "_vis", info.visible,
154 base + "_visible",
155 [id = info.id, scene](bool v) {
156 scene->setNodeVisible(id, v);
157 }),
158 spacer(base + "_vis_spacer"),
159 button("Pick##" + base + "_pick", base + "_pick",
160 [this, id = info.id]() {
161 if (pickHandler_) pickHandler_(id);
162 }),
163 },
164 base + "_actions"));
165 }
166 }
167 return window("Scene", std::move(children), "root");
168}
