36editing::Result<int> applyHeightmapBrush(procgen::Heightmap* heightmap, float centerX,
37 float centerY, float radius, float strength) {
38 if (!heightmap || radius < 0.0F)
39 return eve::editing::failed<int>(
40 editing::Status::Rejected, editing::RuleId("procgen.heightmap.invalid-brush"),
41 "Heightmap brush requires a target and non-negative radius");
42 if (strength == 0.0F) {
43 return editing::Result<int>::success(0, eve::Status::success(editing::Status::NoOp));
44 }
45 const int minX = std::max(0, static_cast<int>(std::floor(centerX - radius)));
46 const int maxX = std::min(heightmap->getWidth() - 1, static_cast<int>(std::ceil(centerX + radius)));
47 const int minY = std::max(0, static_cast<int>(std::floor(centerY - radius)));
48 const int maxY = std::min(heightmap->getHeight() - 1, static_cast<int>(std::ceil(centerY + radius)));
49 const float edge = radius + 0.5F;
50 int changed = 0;
51 for (int y = minY; y <= maxY; ++y) {
52 for (int x = minX; x <= maxX; ++x) {
53 const float dx = static_cast<float>(x) - centerX;
54 const float dy = static_cast<float>(y) - centerY;
55 const float distance = std::sqrt(dx * dx + dy * dy);
56 if (distance > radius) continue;
57 const float oldHeight = heightmap->height(x, y);
58 const float newHeight = std::clamp(oldHeight + strength * (1.0F - distance / edge), 0.0F, 1.0F);
59 if (newHeight == oldHeight) continue;
60 heightmap->setHeight(x, y, newHeight);
61 ++changed;
62 }
63 }
64 return eve::editing::applied<int>(changed);
65}
