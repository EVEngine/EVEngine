18std::string sha256(std::span<const std::uint8_t> bytes) {
19 data::HashFunction::Value digest{};
20 data::HashFunction::getHashFunction("sha256")
21 ->hash("sha256", reinterpret_cast<const char*>(bytes.data()), bytes.size(), digest);
22 static constexpr char hex[] = "0123456789abcdef";
23 std::string result = "sha256:";
24 result.reserve(71);
25 for (std::size_t index = 0; index < 32; ++index) {
26 const auto byte = static_cast<std::uint8_t>(digest.data[index]);
27 result.push_back(hex[byte >> 4]);
28 result.push_back(hex[byte & 0x0f]);
29 }
30 return result;
31}

33Result<Value> migrateImageV1ToV2(const Value& input) {
34 const auto* object = input.getIf<Value::Object>();
35 if (!object)
36 return migrationFailure<Value>(DiagnosticCode::ParseError,
37 "eve.image definition must be an object");
38 const auto schema = object->find("schema");
39 const auto version = object->find("schemaVersion");
40 const auto colorSpace = object->find("colorSpace");
41 if (schema == object->end() || !schema->second.isString() || schema->second.asString() != "eve.image" ||
42 version == object->end() || !version->second.isInt64() || version->second.asInt() != 1 ||
43 colorSpace == object->end() || !colorSpace->second.isString() ||
44 (colorSpace->second.asString() != "srgb" && colorSpace->second.asString() != "linear"))
45 return migrationFailure<Value>(DiagnosticCode::ParseError,
46 "eve.image/1 definition is malformed");
47 Value::Object migrated = *object;
48 migrated.erase("colorSpace");
49 migrated["schemaVersion"] = Value(std::int64_t(2));
50 migrated["color"] = Value(Value::Object{
51 {"primaries", Value("srgb")},
52 {"transfer", Value(colorSpace->second.asString())},
53 });
54 return Result<Value>::success(Value(std::move(migrated)));
55}

57Result<Value> migrateDefinition(std::string_view type, SchemaVersion from,
58 SchemaVersion current, const Value& definition) {
59 if (from == current) return Result<Value>::success(definition);
60 if (from.value() > current.value())
61 return migrationFailure<Value>(DiagnosticCode::UnknownVersion,
62 "asset definition is newer than this reader", std::string(type));
63 if (from.value() + 1 != current.value())
64 return migrationFailure<Value>(DiagnosticCode::Unsupported,
65 "asset definition is older than the N-1 compatibility window",
66 std::string(type));
67 if (type == "eve.image" && from.value() == 1 && current.value() == 2)
68 return migrateImageV1ToV2(definition);
69 return migrationFailure<Value>(DiagnosticCode::Unsupported,
70 "asset definition has no registered migration", std::string(type));
71}

73Result<void> refreshImportReport(EvaArchive& archive) {
74 const auto path = archive.manifest.provenance.find("path");
75 if (path == archive.manifest.provenance.end()) return Result<void>::success();
76 if (!path->second.isString() || path->second.asString() != "reports/import.json")
77 return migrationFailure<void>(DiagnosticCode::ParseError,
78 "migration found an invalid import report path");
79 const auto entry = std::lower_bound(archive.entries.begin(), archive.entries.end(),
80 path->second.asString(),
81 [](const EvaArchiveEntry& candidate, std::string_view value) {
82 return candidate.path < value;
83 });
84 if (entry == archive.entries.end() || entry->path != path->second.asString())
85 return migrationFailure<void>(DiagnosticCode::NotFound,
86 "migration import report is missing", path->second.asString());
87 auto parsed = Value::fromJson(std::string(entry->bytes.begin(), entry->bytes.end()));
88 if (!parsed) return Result<void>::failure(parsed.status());
89 auto* report = parsed.value().getIf<Value::Object>();
90 if (!report)
91 return migrationFailure<void>(DiagnosticCode::ParseError,
92 "migration import report root is invalid", entry->path);
93 Value::Array assets;
94 for (const auto& asset : archive.manifest.assets)
95 assets.emplace_back(Value::Object{{"asset", Value(asset.asset.format())},
96 {"type", Value(asset.type + "/" +
97 std::to_string(asset.schemaVersion.value()))},
98 {"contentHash", Value(asset.contentHash)}});
99 (*report)["canonicalAssets"] = Value(std::move(assets));
100 report->erase("importKey");
101 auto facts = parsed.value().toJson();
102 if (!facts) return Result<void>::failure(facts.status());
103 const std::string key = sha256(std::span<const std::uint8_t>(
104 reinterpret_cast<const std::uint8_t*>(facts.value().data()), facts.value().size()));
105 (*report)["importKey"] = Value(key);
106 auto encoded = parsed.value().toJson();
107 if (!encoded) return Result<void>::failure(encoded.status());
108 entry->bytes.assign(encoded.value().begin(), encoded.value().end());
109 archive.manifest.provenance["importKey"] = Value(key);
110 return Result<void>::success();
111}

130Result<EvaArchive> migrateEvaArchive(EvaArchive source, const EvaArchiveLimits& limits) {
131 for (auto& asset : source.manifest.assets) {
132 auto current = currentAssetSchemaVersion(asset.type);
133 if (!current) return Result<EvaArchive>::failure(current.status());
134 if (asset.schemaVersion.value() > current.value().value())
135 return migrationFailure<EvaArchive>(DiagnosticCode::UnknownVersion,
136 "asset definition is newer than this reader", asset.type);
137 const auto entry = std::lower_bound(source.entries.begin(), source.entries.end(), asset.definition,
138 [](const EvaArchiveEntry& candidate, std::string_view path) {
139 return candidate.path < path;
140 });
141 if (entry == source.entries.end() || entry->path != asset.definition)
142 return migrationFailure<EvaArchive>(DiagnosticCode::NotFound,
143 "asset definition is missing", asset.definition);
144 const std::string_view text(reinterpret_cast<const char*>(entry->bytes.data()), entry->bytes.size());
145 auto parsed = Value::fromJson(text);
146 if (!parsed) return Result<EvaArchive>::failure(parsed.status());
147 auto migrated = migrateDefinition(asset.type, asset.schemaVersion, current.value(), parsed.value());
148 if (!migrated) return Result<EvaArchive>::failure(migrated.status());
149 auto encoded = migrated.value().toJson();
150 if (!encoded) return Result<EvaArchive>::failure(encoded.status());
151 entry->bytes.assign(encoded.value().begin(), encoded.value().end());
152 asset.schemaVersion = current.value();
153 asset.contentHash = sha256(entry->bytes);
154 }
155 auto report = refreshImportReport(source);
156 if (!report) return Result<EvaArchive>::failure(report.status());
157 auto rebuilt = buildEvaArchive(source.manifest, std::move(source.entries), limits);
158 if (!rebuilt) return Result<EvaArchive>::failure(rebuilt.status());
159 return parseEvaArchive(rebuilt.value(), limits);
160}
