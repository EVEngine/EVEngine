12std::string sha256(std::span<const std::uint8_t> bytes) {
13 data::HashFunction::Value digest{};
14 data::HashFunction::getHashFunction("sha256")->hash("sha256", reinterpret_cast<const char*>(bytes.data()),
15 bytes.size(), digest);
16 static constexpr char hex[] = "0123456789abcdef";
17 std::string result = "sha256:";
18 result.reserve(71);
19 for (std::size_t index = 0; index < 32; ++index) {
20 const auto byte = static_cast<std::uint8_t>(digest.data[index]);
21 result.push_back(hex[byte >> 4]);
22 result.push_back(hex[byte & 0x0f]);
23 }
24 return result;
25}

467Result<void> refreshImportReport(EvaArchive& archive) {
468 const auto path = archive.manifest.provenance.find("path");
469 if (path == archive.manifest.provenance.end()) return Result<void>::success();
470 if (!path->second.isString() || path->second.asString() != "reports/import.json")
471 return Result<void>::failure(Diagnostic::error(
472 DiagnosticCode::ParseError, "migration found an invalid import report path", {}, {}, "asset.migration"));
473 const auto entry = std::lower_bound(
474 archive.entries.begin(), archive.entries.end(), path->second.asString(),
475 [](const EvaArchiveEntry& candidate, std::string_view value) { return candidate.path < value; });
476 if (entry == archive.entries.end() || entry->path != path->second.asString())
477 return Result<void>::failure(Diagnostic::error(DiagnosticCode::NotFound, "migration import report is missing",
478 path->second.asString(), {}, "asset.migration"));
479 auto parsed = Value::fromJson(std::string(entry->bytes.begin(), entry->bytes.end()));
480 if (!parsed) return Result<void>::failure(parsed.status());
481 auto* report = parsed.value().getIf<Value::Object>();
482 if (!report)
483 return Result<void>::failure(Diagnostic::error(
484 DiagnosticCode::ParseError, "migration import report root is invalid", entry->path, {}, "asset.migration"));
485 Value::Array assets;
486 for (const auto& asset : archive.manifest.assets)
487 assets.emplace_back(
488 Value::Object{{"asset", Value(asset.asset.format())},
489 {"type", Value(asset.type + "/" + std::to_string(asset.schemaVersion.value()))},
490 {"contentHash", Value(asset.contentHash)}});
491 (*report)["canonicalAssets"] = Value(std::move(assets));
492 report->erase("importKey");
493 auto facts = parsed.value().toJson();
494 if (!facts) return Result<void>::failure(facts.status());
495 const std::string key = sha256(std::span<const std::uint8_t>(
496 reinterpret_cast<const std::uint8_t*>(facts.value().data()), facts.value().size()));
497 (*report)["importKey"] = Value(key);
498 auto encoded = parsed.value().toJson();
499 if (!encoded) return Result<void>::failure(encoded.status());
500 entry->bytes.assign(encoded.value().begin(), encoded.value().end());
501 archive.manifest.provenance["importKey"] = Value(key);
502 return Result<void>::success();
503}

537Result<EvaArchive> migrateEvaArchive(EvaArchive source, const EvaArchiveLimits& limits) {
538 for (auto& asset : source.manifest.assets) {
539 auto current = currentAssetSchemaVersion(asset.type);
540 if (!current) return Result<EvaArchive>::failure(current.status());
541 if (asset.schemaVersion.value() > current.value().value())
542 return Result<EvaArchive>::failure(Diagnostic::error(DiagnosticCode::UnknownVersion,
543 "asset definition is newer than this reader",
544 asset.type, {}, "asset.migration"));
545 const auto entry = std::lower_bound(
546 source.entries.begin(), source.entries.end(), asset.definition,
547 [](const EvaArchiveEntry& candidate, std::string_view path) { return candidate.path < path; });
548 if (entry == source.entries.end() || entry->path != asset.definition)
549 return Result<EvaArchive>::failure(Diagnostic::error(
550 DiagnosticCode::NotFound, "asset definition is missing", asset.definition, {}, "asset.migration"));
551 const std::string_view text(reinterpret_cast<const char*>(entry->bytes.data()), entry->bytes.size());
552 auto parsed = Value::fromJson(text);
553 if (!parsed) return Result<EvaArchive>::failure(parsed.status());
554 auto migrated = migrateDefinition(asset.type, asset.schemaVersion, current.value(), parsed.value());
555 if (!migrated) return Result<EvaArchive>::failure(migrated.status());
556 auto encoded = migrated.value().toJson();
557 if (!encoded) return Result<EvaArchive>::failure(encoded.status());
558 entry->bytes.assign(encoded.value().begin(), encoded.value().end());
559 if (asset.schemaVersion != current.value()) {
560 const std::string oldType = asset.type + "/" + std::to_string(asset.schemaVersion.value());
561 const std::string newType = asset.type + "/" + std::to_string(current.value().value());
562 for (auto& dependency : source.manifest.dependencies)
563 if (dependency.to.id() == asset.asset.id() && dependency.expectedType == oldType)
564 dependency.expectedType = newType;
565 }
566 asset.schemaVersion = current.value();
567 asset.contentHash = sha256(entry->bytes);
568 }
569 auto report = refreshImportReport(source);
570 if (!report) return Result<EvaArchive>::failure(report.status());
571 auto rebuilt = buildEvaArchive(source.manifest, std::move(source.entries), limits);
572 if (!rebuilt) return Result<EvaArchive>::failure(rebuilt.status());
573 return parseEvaArchive(rebuilt.value(), limits);
574}
