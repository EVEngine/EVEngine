18std::string sha256(std::span<const std::uint8_t> bytes) {
19 data::HashFunction::Value digest{};
20 data::HashFunction::getHashFunction("sha256")
21 ->hash("sha256", reinterpret_cast<const char*>(bytes.data()), bytes.size(), digest);
22 static constexpr char hex[] = "0123456789abcdef";
23 std::string result = "sha256:";
24 result.reserve(71);
25 for (std::size_t index = 0; index < 32; ++index) {
26 const auto byte = static_cast<std::uint8_t>(digest.data[index]);
27 result.push_back(hex[byte >> 4]);
28 result.push_back(hex[byte & 0x0f]);
29 }
30 return result;
31}

33Result<Value> migrateImageV1ToV2(const Value& input) {
34 const auto* object = input.getIf<Value::Object>();
35 if (!object)
36 return migrationFailure<Value>(DiagnosticCode::ParseError,
37 "eve.image definition must be an object");
38 const auto schema = object->find("schema");
39 const auto version = object->find("schemaVersion");
40 const auto colorSpace = object->find("colorSpace");
41 if (schema == object->end() || !schema->second.isString() || schema->second.asString() != "eve.image" ||
42 version == object->end() || !version->second.isInt64() || version->second.asInt() != 1 ||
43 colorSpace == object->end() || !colorSpace->second.isString() ||
44 (colorSpace->second.asString() != "srgb" && colorSpace->second.asString() != "linear"))
45 return migrationFailure<Value>(DiagnosticCode::ParseError,
46 "eve.image/1 definition is malformed");
47 Value::Object migrated = *object;
48 migrated.erase("colorSpace");
49 migrated["schemaVersion"] = Value(std::int64_t(2));
50 migrated["color"] = Value(Value::Object{
51 {"primaries", Value("srgb")},
52 {"transfer", Value(colorSpace->second.asString())},
53 });
54 return Result<Value>::success(Value(std::move(migrated)));

57Result<Value> migrateDefinition(std::string_view type, SchemaVersion from,
58 SchemaVersion current, const Value& definition) {
59 if (from == current) return Result<Value>::success(definition);
60 if (from.value() > current.value())
61 return migrationFailure<Value>(DiagnosticCode::UnknownVersion,
62 "asset definition is newer than this reader", std::string(type));
63 if (from.value() + 1 != current.value())
64 return migrationFailure<Value>(DiagnosticCode::Unsupported,
65 "asset definition is older than the N-1 compatibility window",
66 std::string(type));
67 if (type == "eve.mesh" && from.value() == 1 && current.value() == 2) {
68 const auto* object = definition.getIf<Value::Object>();
69 if (!object || !object->contains("schema") || !object->at("schema").isString() ||
70 object->at("schema").asString() != "eve.mesh" || !object->contains("schemaVersion") ||
71 !object->at("schemaVersion").isInt64() || object->at("schemaVersion").asInt() != 1 ||
72 object->contains("texcoordSets"))
73 return migrationFailure<Value>(DiagnosticCode::ParseError, "eve.mesh/1 is malformed");
74 auto migrated = *object;
75 if (auto uv = migrated.find("texcoord0"); uv != migrated.end()) {
76 if (!uv->second.isBool())
77 return migrationFailure<Value>(DiagnosticCode::ParseError, "invalid legacy UV metadata");
78 Value::Array sets;
79 if (uv->second.asBool()) sets.emplace_back(int64_t(0));
80 migrated.erase(uv);
81 migrated["texcoordSets"] = Value(std::move(sets));
82 }
83 migrated["schemaVersion"] = Value(int64_t(2));
84 return Result<Value>::success(Value(std::move(migrated)));
85 }
86 if (type == "eve.material" && from.value() == 1 && current.value() == 2) {
87 const auto* object = definition.getIf<Value::Object>();
88 if (!object || !object->contains("schema") || !object->at("schema").isString() ||
89 object->at("schema").asString() != "eve.material" || !object->contains("schemaVersion") ||
90 !object->at("schemaVersion").isInt64() || object->at("schemaVersion").asInt() != 1)
91 return migrationFailure<Value>(DiagnosticCode::ParseError, "eve.material/1 is malformed");
92 auto migrated = *object;
93 migrated["schemaVersion"] = Value(int64_t(2));
94 return Result<Value>::success(Value(std::move(migrated)));
95 }
96 if (type == "eve.image" && from.value() == 1 && current.value() == 2)
97 return migrateImageV1ToV2(definition);
98 if (type == "eve.scene-template" && from.value() == 1 && current.value() == 2) {
99 const auto* object = definition.getIf<Value::Object>();
100 if (!object || !object->contains("nodes") || object->contains("renderers") || !object->contains("schema") ||
101 !object->at("schema").isString() || object->at("schema").asString() != "eve.scene-template" ||
102 !object->contains("schemaVersion") || !object->at("schemaVersion").isInt64() ||
103 object->at("schemaVersion").asInt() != 1)
104 return migrationFailure<Value>(DiagnosticCode::ParseError,
105 "scene-template/1 is malformed or has unversioned renderers");
106 auto migrated = *object;
107 migrated["schemaVersion"] = Value(std::int64_t(2));
108 migrated["renderers"] = Value(Value::Array{});
109 return Result<Value>::success(Value(std::move(migrated)));
110 }
111 return migrationFailure<Value>(DiagnosticCode::Unsupported,
112 "asset definition has no registered migration", std::string(type));
113}

115Result<void> refreshImportReport(EvaArchive& archive) {
116 const auto path = archive.manifest.provenance.find("path");
117 if (path == archive.manifest.provenance.end()) return Result<void>::success();
118 if (!path->second.isString() || path->second.asString() != "reports/import.json")
119 return migrationFailure<void>(DiagnosticCode::ParseError,
120 "migration found an invalid import report path");
121 const auto entry = std::lower_bound(archive.entries.begin(), archive.entries.end(),
122 path->second.asString(),
123 [](const EvaArchiveEntry& candidate, std::string_view value) {
124 return candidate.path < value;
125 });
126 if (entry == archive.entries.end() || entry->path != path->second.asString())
127 return migrationFailure<void>(DiagnosticCode::NotFound,
128 "migration import report is missing", path->second.asString());
129 auto parsed = Value::fromJson(std::string(entry->bytes.begin(), entry->bytes.end()));
130 if (!parsed) return Result<void>::failure(parsed.status());
131 auto* report = parsed.value().getIf<Value::Object>();
132 if (!report)
133 return migrationFailure<void>(DiagnosticCode::ParseError,
134 "migration import report root is invalid", entry->path);
135 Value::Array assets;
136 for (const auto& asset : archive.manifest.assets)
137 assets.emplace_back(Value::Object{{"asset", Value(asset.asset.format())},
138 {"type", Value(asset.type + "/" +
139 std::to_string(asset.schemaVersion.value()))},
140 {"contentHash", Value(asset.contentHash)}});
141 (*report)["canonicalAssets"] = Value(std::move(assets));
142 report->erase("importKey");
143 auto facts = parsed.value().toJson();
144 if (!facts) return Result<void>::failure(facts.status());
145 const std::string key = sha256(std::span<const std::uint8_t>(
146 reinterpret_cast<const std::uint8_t*>(facts.value().data()), facts.value().size()));
147 (*report)["importKey"] = Value(key);
148 auto encoded = parsed.value().toJson();
149 if (!encoded) return Result<void>::failure(encoded.status());
150 entry->bytes.assign(encoded.value().begin(), encoded.value().end());
151 archive.manifest.provenance["importKey"] = Value(key);
152 return Result<void>::success();
153}

184Result<EvaArchive> migrateEvaArchive(EvaArchive source, const EvaArchiveLimits& limits) {
185 for (auto& asset : source.manifest.assets) {
186 auto current = currentAssetSchemaVersion(asset.type);
187 if (!current) return Result<EvaArchive>::failure(current.status());
188 if (asset.schemaVersion.value() > current.value().value())
189 return migrationFailure<EvaArchive>(DiagnosticCode::UnknownVersion,
190 "asset definition is newer than this reader", asset.type);
191 const auto entry = std::lower_bound(source.entries.begin(), source.entries.end(), asset.definition,
192 [](const EvaArchiveEntry& candidate, std::string_view path) {
193 return candidate.path < path;
194 });
195 if (entry == source.entries.end() || entry->path != asset.definition)
196 return migrationFailure<EvaArchive>(DiagnosticCode::NotFound,
197 "asset definition is missing", asset.definition);
198 const std::string_view text(reinterpret_cast<const char*>(entry->bytes.data()), entry->bytes.size());
199 auto parsed = Value::fromJson(text);
200 if (!parsed) return Result<EvaArchive>::failure(parsed.status());
201 auto migrated = migrateDefinition(asset.type, asset.schemaVersion, current.value(), parsed.value());
202 if (!migrated) return Result<EvaArchive>::failure(migrated.status());
203 auto encoded = migrated.value().toJson();
204 if (!encoded) return Result<EvaArchive>::failure(encoded.status());
205 entry->bytes.assign(encoded.value().begin(), encoded.value().end());
206 if (asset.schemaVersion != current.value()) {
207 const std::string oldType = asset.type + "/" + std::to_string(asset.schemaVersion.value());
208 const std::string newType = asset.type + "/" + std::to_string(current.value().value());
209 for (auto& dependency : source.manifest.dependencies)
210 if (dependency.to.id() == asset.asset.id() && dependency.expectedType == oldType)
211 dependency.expectedType = newType;
212 }
213 asset.schemaVersion = current.value();
214 asset.contentHash = sha256(entry->bytes);
215 }
216 auto report = refreshImportReport(source);
217 if (!report) return Result<EvaArchive>::failure(report.status());
218 auto rebuilt = buildEvaArchive(source.manifest, std::move(source.entries), limits);
219 if (!rebuilt) return Result<EvaArchive>::failure(rebuilt.status());
220 return parseEvaArchive(rebuilt.value(), limits);
221}
