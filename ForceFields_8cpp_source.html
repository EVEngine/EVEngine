8bool validRegion(const FieldRegion& r) {
9 if (!isFinite(r.center) || !isFinite(r.radius) || !isNonnegative(r.edgeWidth) || !isFinite(r.halfExtents))
10 return false;
11 switch (r.kind) {
12 case FieldRegionKind::Global: return true;
13 case FieldRegionKind::Sphere: return r.radius < 0 || r.edgeWidth <= r.radius;
14 case FieldRegionKind::RectangleXY:
15 return isPositive(r.halfExtents.x) && isPositive(r.halfExtents.y) && r.halfExtents.z == 0 &&
16 r.edgeWidth <= std::min(r.halfExtents.x, r.halfExtents.y);
17 case FieldRegionKind::Box:
18 return isPositive(r.halfExtents.x) && isPositive(r.halfExtents.y) && isPositive(r.halfExtents.z) &&
19 r.edgeWidth <= std::min({r.halfExtents.x, r.halfExtents.y, r.halfExtents.z});
20 }
21 return false;
22}
23double weight(const FieldRegion& r, ForceVector position) {
24 if (r.kind == FieldRegionKind::Global || (r.kind == FieldRegionKind::Sphere && r.radius < 0)) return 1;
25 const auto delta = add(position, scale(r.center, -1));
26 double interior = 0;
27 if (r.kind == FieldRegionKind::Sphere)
28 interior = r.radius - length(delta);
29 else {
30 interior = std::min(r.halfExtents.x - std::abs(delta.x), r.halfExtents.y - std::abs(delta.y));
31 if (r.kind == FieldRegionKind::Box) interior = std::min(interior, r.halfExtents.z - std::abs(delta.z));
32 }
33 if (interior <= 0) return 0;
34 if (r.edgeWidth == 0) return 1;
35 return std::clamp(interior / r.edgeWidth, 0.0, 1.0);
36}

47Result<ForceVector> ForceModels::radial(const RadialField& m, ForceVector p, double mass) {
48 if (!validRegion(m.region) || !isFinite(m.strength) || !isPositive(m.softening) || !isFinite(p) ||
49 !isPositive(mass) || !isNonnegative(m.maxAcceleration) || !isNonnegative(m.maxForce))
50 return invalid<ForceVector>();
51 const auto delta = add(m.region.center, scale(p, -1));
52 const double distance = length(delta), radius = std::hypot(distance, m.softening);
53 if (!isFinite(delta) || !isFinite(distance) || !isFinite(radius)) return invalid<ForceVector>();
54 const double fade = weight(m.region, p);
55 if (distance == 0 || m.strength == 0 || fade == 0 || m.maxAcceleration == 0 || m.maxForce == 0) return checked({});
56 // Compare magnitudes in log space so tiny softened cores cannot overflow before the caps apply.
57 const double logMass = std::log(mass);
58 const double logForce =
59 std::log(std::abs(m.strength)) + std::log(distance) - 3 * std::log(radius) + logMass + std::log(fade);
60 const double cappedLog = std::min({logForce, std::log(m.maxAcceleration) + logMass, std::log(m.maxForce)});
61 const double magnitude = std::min(std::exp(cappedLog), m.maxForce);
62 const double sign = m.strength < 0 ? -1 : 1;
63 // Divide components separately: reciprocal distance may overflow for subnormal positions.
64 return checked({sign * (delta.x / distance) * magnitude, sign * (delta.y / distance) * magnitude,
65 sign * (delta.z / distance) * magnitude});
66}

72Result<ForceVector> ForceModels::damping(const DampingModel& m, ForceVector velocity, double mass, double dt) {
73 if (!isNonnegative(m.linear) || !isNonnegative(m.quadratic) || !isFinite(velocity) || !isPositive(mass) ||
74 !isPositive(dt))
75 return invalid<ForceVector>();
76 const double coefficient = m.linear + m.quadratic * length(velocity);
77 const double rate = coefficient / mass * dt;
78 if (!isFinite(coefficient) || !isFinite(rate)) return invalid<ForceVector>();
79 return checked(scale(velocity, -coefficient / (1 + rate)));
80}
81Result<ForceVector> ForceModels::friction(const FrictionModel& m, ForceVector velocity, ForceVector demand,
82 double load) {
83 if (!isNonnegative(m.kineticCoefficient) || !isFinite(m.staticCoefficient) ||
84 m.staticCoefficient < m.kineticCoefficient || !isFinite(velocity) || !isFinite(demand) || !isNonnegative(load))
85 return invalid<ForceVector>();
86 const double speed = length(velocity), required = length(demand);
87 if (!isFinite(speed) || !isFinite(required) || !isFinite(m.staticCoefficient * load)) return invalid<ForceVector>();
88 if (speed > 0) return checked(scale(velocity, -m.kineticCoefficient * load / speed));
89 if (required <= m.staticCoefficient * load) return checked(scale(demand, -1));
90 return checked(scale(demand, -m.kineticCoefficient * load / required));
91}
