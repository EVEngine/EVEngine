42double flowingRatio(double t) { return 0.08 + 0.92 * std::sin(t * std::numbers::pi * 0.5); }
43struct PathSample {
44 double progress, lateral;
45};
46PathSample samplePath(V p, const std::vector<V>& path, const std::vector<double>& arc) {
47 double best = std::numeric_limits<double>::max(), result = 1.;
48 for (std::size_t i = 0; i + 1 < path.size(); ++i) {
49 const V delta = sub(path[i + 1], path[i]);
50 const double t = std::clamp(dot(sub(p, path[i]), delta) / dot(delta, delta), 0., 1.);
51 const V difference = sub(p, add(path[i], mul(delta, t)));
52 const double squared = dot(difference, difference);
53 if (squared < best) {
54 best = squared;
55 result = (arc[i] + (arc[i + 1] - arc[i]) * t) / arc.back();
56 }
57 }
58 const double z = p[2] - path.front()[2];
59 return {result, std::sqrt(std::max(0., best - z * z))};
60}

63Result<MeshBuild> flatArrow(const std::vector<V>& path, const std::vector<V>& normals, V tip, double shaft, double head,
64 double thickness, double radius, int steps, bool flowing) {
65 if (radius >= shaft * 0.25) return Result<MeshBuild>::failure(invalid());
66 std::vector<double> arc{0.};
67 for (std::size_t i = 1; i < path.size(); ++i) arc.push_back(arc.back() + length(sub(path[i], path[i - 1])));
68 const auto taper = [&](double t) { return flowing ? flowingRatio(t) : 1.; };
69 const auto progress = [&](V p) { return samplePath(p, path, arc).progress; };
70 std::vector<V> outline;
71 for (std::size_t i = 0; i < path.size(); ++i)
72 outline.push_back(sub(path[i], mul(normals[i], shaft * taper(arc[i] / arc.back()) * 0.5)));
73 outline.push_back(sub(path.back(), mul(normals.back(), head * 0.5)));
74 outline.push_back(tip);
75 outline.push_back(add(path.back(), mul(normals.back(), head * 0.5)));
76 for (std::size_t i = path.size(); i-- > 0;)
77 outline.push_back(add(path[i], mul(normals[i], shaft * taper(arc[i] / arc.back()) * 0.5)));
78 // Remove collinear samples before rounding so straight shafts remain inexpensive.
79 for (std::size_t i = 0; outline.size() > 3 && i < outline.size();) {
80 const std::size_t n = outline.size();
81 if (std::abs(turn(outline[(i + n - 1) % n], outline[i], outline[(i + 1) % n])) < 1e-12) {
82 outline.erase(outline.begin() + static_cast<std::ptrdiff_t>(i));
83 if (i > 0) --i;
84 } else
85 ++i;
86 }
87 if (flowing) {
88 // Reserve a visible fillet span at both shoulders instead of letting
89 // dense curve samples force the corner radius down to one tiny edge.
90 const V necks[]{sub(path.back(), mul(normals.back(), shaft * 0.5)),
91 add(path.back(), mul(normals.back(), shaft * 0.5))};
92 const double span = std::min(shaft * 0.7, radius * 10.);
93 for (V neck : necks) {
94 bool shortened = true;
95 while (shortened && outline.size() > 7) {
96 shortened = false;
97 const auto found = std::find(outline.begin(), outline.end(), neck);
98 const std::size_t i = static_cast<std::size_t>(found - outline.begin());
99 for (std::size_t j : {(i + outline.size() - 1) % outline.size(), (i + 1) % outline.size()}) {
100 if (length(sub(outline[j], neck)) < span && outline[j] != necks[0] && outline[j] != necks[1] &&
101 outline[j] != tip) {
102 outline.erase(outline.begin() + static_cast<std::ptrdiff_t>(j));
103 shortened = true;
104 break;
105 }
106 }
107 }
108 }
109 }
110 std::vector<V> contour;
111 for (std::size_t i = 0; i < outline.size(); ++i) {
112 const V p = outline[(i + outline.size() - 1) % outline.size()], v = outline[i],
113 q = outline[(i + 1) % outline.size()];
114 const V incoming = sub(p, v), outgoing = sub(q, v);
115 const double cosine = dot(unit(incoming), unit(outgoing));
116 const double desiredCut = 2. * radius * (flowing && v != tip ? 3. : 1.) *
117 std::sqrt(std::max(0., (1. + cosine) * 0.5)) / std::max(1e-8, (1. - cosine) * 0.5);
118 const double cut = std::min({desiredCut, length(incoming) * 0.45, length(outgoing) * 0.45});
119 const V a = add(v, mul(unit(incoming), cut)), b = add(v, mul(unit(outgoing), cut));
120 // Already smooth path samples need no additional corner subdivision.
121 if (dot(unit(incoming), unit(outgoing)) < -0.99) {
122 contour.push_back(v);
123 continue;
124 }
125 for (int j = 0; j <= steps; ++j) {
126 const double t = static_cast<double>(j) / steps;
127 contour.push_back(add(add(mul(a, (1 - t) * (1 - t)), mul(v, 2 * t * (1 - t))), mul(b, t * t)));
128 }
129 }
130 const std::size_t count = contour.size();
131 std::vector<V> insetDirections;
132 for (std::size_t i = 0; i < count; ++i) {
133 const V prev = unit(sub(contour[i], contour[(i + count - 1) % count]));
134 const V next = unit(sub(contour[(i + 1) % count], contour[i]));
135 const V n0{-prev[1], prev[0], 0.}, n1{-next[1], next[0], 0.};
136 const double denominator = 1. + dot(n0, n1);
137 if (denominator < 1e-6) return Result<MeshBuild>::failure(invalid());
138 insetDirections.push_back(mul(add(n0, n1), 1. / denominator));
139 }
140 std::vector<double> radii, thicknesses;
141 for (V point : contour) {
142 const double ratio = flowing ? taper(progress(point)) : 1.;
143 const double h = thickness;
144 thicknesses.push_back(h);
145 radii.push_back(flowing ? std::min({radius, shaft * ratio * 0.2, h * 0.45}) : radius);
146 }
147 std::vector<std::vector<V>> rings;
148 const auto appendRing = [&](double insetFactor, double sign, double riseFactor) {
149 std::vector<V> ring;
150 for (std::size_t i = 0; i < count; ++i) {
151 V v = add(contour[i], mul(insetDirections[i], radii[i] * insetFactor));
152 v[2] += sign * thicknesses[i] * 0.5 + radii[i] * riseFactor;
153 ring.push_back(v);
154 }
155 rings.push_back(std::move(ring));
156 };
157 for (int j = 0; j <= steps; ++j) {
158 const double angle = std::numbers::pi * 0.5 * j / steps;
159 appendRing(1. - std::sin(angle), -1., 1. - std::cos(angle));
160 }
161 appendRing(0., 1., -1.);
162 for (int j = 1; j <= steps; ++j) {
163 const double angle = std::numbers::pi * 0.5 * j / steps;
164 appendRing(1. - std::cos(angle), 1., std::sin(angle) - 1.);
165 }
166 // Reject overlapping inset contours instead of publishing a broken bevel.
167 for (const auto& ring : rings) {
168 for (std::size_t i = 0; i < count; ++i)
169 for (std::size_t j = i + 2; j < count; ++j) {
170 if (i == 0 && j + 1 == count) continue;
171 const V a = ring[i], b = ring[(i + 1) % count], c = ring[j], d = ring[(j + 1) % count];
172 if (turn(a, b, c) * turn(a, b, d) < 0. && turn(c, d, a) * turn(c, d, b) < 0.)
173 return Result<MeshBuild>::failure(invalid("Flat arrow inset intersects itself at edges " +
174 std::to_string(i) + ", " + std::to_string(j)));
175 }
176 }
177 MeshBuild mesh;
178 mesh.setActiveGroup("bevel");
179 for (std::size_t r = 0; r + 1 < rings.size(); ++r)
180 for (std::size_t i = 0; i < count; ++i) {
181 const std::size_t j = (i + 1) % count;
182 quad(mesh, rings[r][i], rings[r][j], rings[r + 1][j], rings[r + 1][i], 0., 1.);
183 }
184 mesh.setActiveGroup("face");
185 std::vector<std::size_t> remaining;
186 for (std::size_t i = 0; i < count; ++i) remaining.push_back(i);
187 const auto& top = rings.back();
188 const auto& bottom = rings.front();
189 while (remaining.size() > 2) {
190 bool clipped = false;
191 for (std::size_t i = 0; i < remaining.size(); ++i) {
192 const auto a = remaining[(i + remaining.size() - 1) % remaining.size()], b = remaining[i],
193 c = remaining[(i + 1) % remaining.size()];
194 if (turn(top[a], top[b], top[c]) <= 1e-12) continue;
195 bool occupied = false;
196 for (auto k : remaining) {
197 if (k == a || k == b || k == c) continue;
198 if (turn(top[a], top[b], top[k]) >= -1e-12 && turn(top[b], top[c], top[k]) >= -1e-12 &&
199 turn(top[c], top[a], top[k]) >= -1e-12) {
200 occupied = true;
201 break;
202 }
203 }
204 if (occupied) continue;
205 triangle(mesh, top[a], top[b], top[c], 0., 0.5, 1.);
206 triangle(mesh, bottom[c], bottom[b], bottom[a], 1., 0.5, 0.);
207 remaining.erase(remaining.begin() + static_cast<std::ptrdiff_t>(i));
208 clipped = true;
209 break;
210 }
211 if (!clipped) return Result<MeshBuild>::failure(invalid("Flat arrow cap cannot be triangulated"));
212 }
213 return Result<MeshBuild>::success(std::move(mesh));
214}

218 V position;
219 double u, v;

221MeshBuild shapeFlatSurface(const MeshBuild& source, V base, V axis, double length, double centerZ, double scale,
222 const std::vector<V>& path, bool flowing, double shaftWidth, double headWidth) {
223 MeshBuild output;
224 int active = -2;
225 const auto distance = [&](V p) { return dot(sub(p, base), axis); };
226 std::vector<double> arc{0.};
227 for (std::size_t i = 1; i < path.size(); ++i)
228 arc.push_back(arc.back() + std::sqrt(dot(sub(path[i], path[i - 1]), sub(path[i], path[i - 1]))));
229 std::vector<std::pair<V, double>> cuts;
230 if (flowing) {
231 const V chord = sub(path.back(), path.front());
232 const double chordLength = std::sqrt(dot(chord, chord));
233 const V direction = mul(chord, 1. / chordLength);
234 for (int i = 1; i < 16; ++i) cuts.emplace_back(direction, dot(path.front(), direction) + chordLength * i / 16.);
235 }
236 if (flowing) {
237 // Sample the crowned head in both directions, not only along its perimeter.
238 for (int i = 1; i < 13; ++i) cuts.emplace_back(axis, dot(base, axis) + length * i / 13.);
239 const V transverse{-axis[1], axis[0], 0.};
240 for (int i = 1; i < 10; ++i) cuts.emplace_back(transverse, dot(base, transverse) + headWidth * (i / 10. - 0.5));
241 }
242 if (scale != 1.)
243 for (int i = 0; i <= 4; ++i) cuts.emplace_back(axis, dot(base, axis) + length * (-0.25 + 0.25 * i / 4.));
244 for (int i = 0; i < source.getIndexCount(); i += 3) {
245 const int group = source.getTriangleGroup(i / 3);
246 if (group != active) {
247 output.setActiveGroup(source.getGroupName(group));
248 active = group;
249 }
250 std::vector<std::vector<HeadVertex>> polygons(1);
251 for (int j = 0; j < 3; ++j) {
252 const int index = source.getIndex(i + j);
253 polygons[0].push_back({{source.getPositionX(index), source.getPositionY(index), source.getPositionZ(index)},
254 source.getUvU(index),
255 source.getUvV(index)});
256 }
257 for (const auto& [clipAxis, plane] : cuts) {
258 const auto projection = [&](V point) { return dot(point, clipAxis); };
259 const auto intersection = [&](HeadVertex a, HeadVertex b) {
260 if (b.position < a.position) std::swap(a, b);
261 const double t = (plane - projection(a.position)) / (projection(b.position) - projection(a.position));
262 V point = add(a.position, mul(sub(b.position, a.position), t));
263 for (auto& value : point) value = static_cast<float>(value);
264 return HeadVertex{point, a.u + (b.u - a.u) * t, a.v + (b.v - a.v) * t};
265 };
266 std::vector<std::vector<HeadVertex>> next;
267 for (auto polygon : polygons) {
268 const double tolerance = length * 1e-6;
269 for (auto& vertex : polygon) {
270 const double delta = projection(vertex.position) - plane;
271 if (std::abs(delta) <= tolerance) {
272 vertex.position = sub(vertex.position, mul(clipAxis, delta));
273 for (auto& value : vertex.position) value = static_cast<float>(value);
274 }
275 }
276 double minimum = projection(polygon[0].position), maximum = minimum;
277 for (const auto& vertex : polygon) {
278 minimum = std::min(minimum, projection(vertex.position));
279 maximum = std::max(maximum, projection(vertex.position));
280 }
281 if (minimum >= plane - tolerance || maximum <= plane + tolerance) {
282 next.push_back(polygon);
283 continue;
284 }
285 std::vector<HeadVertex> low, high;
286 for (std::size_t k = 0; k < polygon.size(); ++k) {
287 const auto a = polygon[k], b = polygon[(k + 1) % polygon.size()];
288 double da = projection(a.position) - plane, db = projection(b.position) - plane;
289 if (std::abs(da) <= tolerance) da = 0.;
290 if (std::abs(db) <= tolerance) db = 0.;
291 if (da <= 0.) low.push_back(a);
292 if (da >= 0.) high.push_back(a);
293 if ((da < 0. && db > 0.) || (da > 0. && db < 0.)) {
294 const auto cut = intersection(a, b);
295 low.push_back(cut);
296 high.push_back(cut);
297 }
298 }
299 if (low.size() >= 3) next.push_back(std::move(low));
300 if (high.size() >= 3) next.push_back(std::move(high));
301 }
302 polygons = std::move(next);
303 }
304 for (auto& polygon : polygons) {
305 for (auto& vertex : polygon) {
306 const double t = std::clamp((distance(vertex.position) / length + 0.25) * 4., 0., 1.);
307 const double weight = t * t * (3. - 2. * t);
308 double taper = 1., crown = 1.;
309 if (flowing) {
310 const auto sample = samplePath(vertex.position, path, arc);
311 taper = flowingRatio(sample.progress);
312 const double headDistance = distance(vertex.position);
313 const double headHalfWidth = headWidth * 0.5 * std::clamp(1. - headDistance / length, 0., 1.);
314 const double halfWidth = (1. - weight) * shaftWidth * 0.5 * taper + weight * headHalfWidth;
315 const double lateral =
316 headDistance >= 0. ? std::abs(cross(axis, sub(vertex.position, base))[2]) : sample.lateral;
317 const double fraction = halfWidth > 1e-8 ? lateral / halfWidth : 1.;
318 crown = 1. + 0.35 * std::max(0., 1. - fraction * fraction);
319 }
320 vertex.position[2] =
321 centerZ + (vertex.position[2] - centerZ) * taper * crown * (1. + (scale - 1.) * weight);
322 vertex.position[2] = static_cast<float>(vertex.position[2]);
323 }
324 for (std::size_t j = 1; j + 1 < polygon.size(); ++j) {
325 const HeadVertex tri[]{polygon[0], polygon[j], polygon[j + 1]};
326 const V normalVector =
327 cross(sub(tri[1].position, tri[0].position), sub(tri[2].position, tri[0].position));
328 if (dot(normalVector, normalVector) < 1e-32) continue;
329 const V n = unit(normalVector);
330 const auto first = static_cast<uint32_t>(output.getVertexCount());
331 for (const auto& vertex : tri)
332 output.addVertex(static_cast<float>(vertex.position[0]), static_cast<float>(vertex.position[1]),
333 static_cast<float>(vertex.position[2]), static_cast<float>(n[0]),
334 static_cast<float>(n[1]), static_cast<float>(n[2]), static_cast<float>(vertex.u),
335 static_cast<float>(vertex.v));
336 output.addTriangle(first, first + 1, first + 2);
337 }
338 }
339 }
340 return output;
341}
342// Contract the front nose edge/face into one geometric apex at the target.
343// Thickness and lateral width both taper; merely sharpening the XY silhouette
344// would retain the old vertical edge at the end of an extruded arrow.
345Result<MeshBuild> pointFlatTip(const MeshBuild& source, V base, V axis, V tip, double headLength) {
346 const double start = headLength * 0.55;
347 const double tolerance = headLength * 1e-6;
348 const auto distance = [&](V p) { return dot(sub(p, base), axis); };
349 double front = 0.;
350 for (int i = 0; i < source.getVertexCount(); ++i)
351 front = std::max(front, distance({source.getPositionX(i), source.getPositionY(i), source.getPositionZ(i)}));
352 if (front <= start + tolerance || front > headLength + tolerance)
353 return Result<MeshBuild>::failure(invalid("Flat arrow has no valid nose interval before its target"));
354 std::vector<V> positions;
355 for (int i = 0; i < source.getVertexCount(); ++i) {
356 V p{source.getPositionX(i), source.getPositionY(i), source.getPositionZ(i)};
357 const double d = distance(p);
358 if (d >= front - tolerance)
359 p = tip;
360 else if (d > start) {
361 const double t = std::clamp((d - start) / (front - start), 0., 1.);
362 const double forward = d + (headLength - front) * t * t * (3. - 2. * t);
363 const V lateral = sub(sub(p, base), mul(axis, d));
364 p = add(base, add(mul(axis, forward), mul(V{lateral[0], lateral[1], 0.}, std::sqrt(1. - t))));
365 p[2] = tip[2] + (lateral[2]) * (1. - t * t);
366 }
367 for (auto& value : p) value = static_cast<float>(value);
368 positions.push_back(p);
369 }
370 MeshBuild mesh;
371 int active = -2;
372 for (int i = 0; i < source.getIndexCount(); i += 3) {
373 const int indices[]{source.getIndex(i), source.getIndex(i + 1), source.getIndex(i + 2)};
374 const V a = positions[indices[0]], b = positions[indices[1]], c = positions[indices[2]];
375 // These faces belonged to the contracted nose edge and carry no remaining area.
376 if (a == b || b == c || c == a) continue;
377 const V area = cross(sub(b, a), sub(c, a));
378 if (length(area) < 1e-16)
379 return Result<MeshBuild>::failure(invalid("Pointed arrow tip exceeds float mesh precision"));
380 const V normal = unit(area);
381 const int group = source.getTriangleGroup(i / 3);
382 if (group != active) {
383 mesh.setActiveGroup(source.getGroupName(group));
384 active = group;
385 }
386 const auto first = static_cast<uint32_t>(mesh.getVertexCount());
387 for (int index : indices) {
388 const V p = positions[index];
389 mesh.addVertex(static_cast<float>(p[0]), static_cast<float>(p[1]), static_cast<float>(p[2]),
390 static_cast<float>(normal[0]), static_cast<float>(normal[1]), static_cast<float>(normal[2]),
391 source.getUvU(index), source.getUvV(index));
392 }
393 mesh.addTriangle(first, first + 1, first + 2);
394 }
395 return Result<MeshBuild>::success(std::move(mesh));
396}

451Result<MeshBuild> ArrowGenerator::buildMeshResult() const {
452 const V chord = sub(b_, a_);
453 const double chordLength = length(chord);
454 // A strictly forward control polygon excludes cusps and repeated samples.
455 if (std::abs(dot(bend_, unit(chord))) >= chordLength * 0.499 || (planar_ && (a_[2] != b_[2] || bend_[2] != 0.)))
456 return Result<MeshBuild>::failure(invalid());
457 const V control = add(mul(add(a_, b_), 0.5), bend_);
458 std::vector<V> points{a_};
459 std::vector<double> arc{0.};
460 for (int i = 1; i <= segments_; ++i) {
461 const double t = static_cast<double>(i) / segments_;
462 const V p = i == segments_
463 ? b_
464 : add(add(mul(a_, (1. - t) * (1. - t)), mul(control, 2. * t * (1. - t))), mul(b_, t * t));
465 arc.push_back(arc.back() + length(sub(p, points.back())));
466 points.push_back(p);
467 }
468 const double total = arc.back();
469 if (total <= headLength_ + 1e-5 || headLength_ >= chordLength * 0.9) return Result<MeshBuild>::failure(invalid());
470 // Trim by distance from B: the final cone has the requested axial length.
471 std::size_t cut = points.size() - 2;
472 while (cut > 0 && length(sub(points[cut], b_)) < headLength_) --cut;
473 V lo = points[cut], hi = points[cut + 1];
474 for (int i = 0; i < 48; ++i) {
475 const V mid = mul(add(lo, hi), 0.5);
476 if (length(sub(mid, b_)) > headLength_)
477 lo = mid;
478 else
479 hi = mid;
480 }
481 const V base = mul(add(lo, hi), 0.5);
482 points.resize(cut + 1);
483 arc.resize(cut + 1);
484 const double last = length(sub(base, points.back()));
485 if (last > 1e-8) {
486 points.push_back(base);
487 arc.push_back(arc.back() + last);
488 } else
489 points.back() = base;
490 const V headAxis = unit(sub(b_, base));
491 if (flat_ && flowing_) {
492 // Align the neck with the head while retaining the tail and distant curve.
493 const double neckSpan = std::min(headLength_ * 0.5, length(sub(base, points.front())) * 0.5);
494 for (auto& point : points) {
495 const V offset = sub(point, base);
496 const double t = std::clamp(1. - length(offset) / neckSpan, 0., 1.);
497 const double weight = t * t * (3. - 2. * t);
498 const V aligned = add(base, mul(headAxis, dot(offset, headAxis)));
499 point = add(mul(point, 1. - weight), mul(aligned, weight));
500 }
501 }
502 std::vector<V> tangents, normals;
503 for (std::size_t i = 0; i < points.size(); ++i) {
504 V t = i + 1 == points.size() ? headAxis : unit(sub(points[i + 1], points[i]));
505 if (i > 0 && i + 1 < points.size()) t = unit(add(t, unit(sub(points[i], points[i - 1]))));
506 tangents.push_back(t);
507 if (planar_)
508 normals.push_back({-t[1], t[0], 0.});
509 else {
510 const V reference = i == 0 ? (std::abs(t[2]) < 0.9 ? V{0., 0., 1.} : V{0., 1., 0.}) : normals.back();
511 const V projected = sub(reference, mul(t, dot(reference, t)));
512 if (length(projected) < 1e-8) return Result<MeshBuild>::failure(invalid());
513 normals.push_back(unit(projected));
514 }
515 }
516 MeshBuild mesh;
517 const int sides = planar_ ? 2 : sides_;
518 const auto ringPoint = [&](std::size_t i, int j, double radius) {
519 if (planar_) return add(points[i], mul(normals[i], j == 0 ? radius : -radius));
520 const double angle = 2. * std::numbers::pi * j / sides;
521 return add(
522 points[i],
523 mul(add(mul(normals[i], std::cos(angle)), mul(cross(tangents[i], normals[i]), std::sin(angle))), radius));
524 };
525 if (flat_) {
526 auto result =
527 flatArrow(points, normals, b_, shaftWidth_, headWidth_, thickness_, bevelRadius_, bevelSegments_, flowing_);
528 if (!result.ok()) return Result<MeshBuild>::failure(result.status());
529 mesh = std::move(result).takeValue();
530 if (headThicknessScale_ != 1. || flowing_)
531 mesh = shapeFlatSurface(mesh, base, headAxis, headLength_, a_[2], headThicknessScale_, points, flowing_,
532 shaftWidth_, headWidth_);
533 } else {
534 mesh.setActiveGroup("shaft");
535 for (std::size_t i = 0; i + 1 < points.size(); ++i)
536 for (int j = 0; j < (planar_ ? 1 : sides); ++j) {
537 const int k = (j + 1) % sides;
538 quad(mesh, ringPoint(i, j, shaftWidth_ / 2.), ringPoint(i, k, shaftWidth_ / 2.),
539 ringPoint(i + 1, k, shaftWidth_ / 2.), ringPoint(i + 1, j, shaftWidth_ / 2.), arc[i] / total,
540 arc[i + 1] / total);
541 }
542 const std::size_t end = points.size() - 1;
543 mesh.setActiveGroup("head");
544 if (planar_)
545 triangle(mesh, ringPoint(end, 1, headWidth_ / 2.), b_, ringPoint(end, 0, headWidth_ / 2.),
546 arc.back() / total, 1., arc.back() / total);
547 else {
548 for (int j = 0; j < sides; ++j) {
549 const int k = (j + 1) % sides;
550 triangle(mesh, ringPoint(end, j, headWidth_ / 2.), ringPoint(end, k, headWidth_ / 2.), b_,
551 arc.back() / total, arc.back() / total, 1.);
552 quad(mesh, ringPoint(end, j, shaftWidth_ / 2.), ringPoint(end, k, shaftWidth_ / 2.),
553 ringPoint(end, k, headWidth_ / 2.), ringPoint(end, j, headWidth_ / 2.), arc.back() / total,
554 arc.back() / total);
555 triangle(mesh, points[0], ringPoint(0, k, shaftWidth_ / 2.), ringPoint(0, j, shaftWidth_ / 2.), 0., 0.,
556 0.);
557 }
558 }
559 }
560 if (flat_) {
561 auto pointed = pointFlatTip(mesh, base, headAxis, b_, headLength_);
562 if (!pointed.ok()) return Result<MeshBuild>::failure(pointed.status());
563 mesh = std::move(pointed).takeValue();
564 }
565 // Float mesh streams must still represent every generated face at the chosen world scale.
566 for (int i = 0; i < mesh.getIndexCount(); i += 3) {
567 V p[3];
568 for (int j = 0; j < 3; ++j) {
569 const int index = mesh.getIndex(i + j);
570 p[j] = {mesh.getPositionX(index), mesh.getPositionY(index), mesh.getPositionZ(index)};
571 }
572 if (length(cross(sub(p[1], p[0]), sub(p[2], p[0]))) < 1e-16)
573 return Result<MeshBuild>::failure(invalid("Mesh contains a degenerate float triangle"));
574 }
575 if (flat_ && flowing_) {
576 // Angle-weighted normals are stable across uneven cap tessellation and preserve face UV seams.
577 std::map<V, V> accumulated;
578 for (int i = 0; i < mesh.getIndexCount(); i += 3) {
579 V p[3];
580 for (int j = 0; j < 3; ++j) {
581 const int k = mesh.getIndex(i + j);
582 p[j] = {mesh.getPositionX(k), mesh.getPositionY(k), mesh.getPositionZ(k)};
583 }
584 const V normal = unit(cross(sub(p[1], p[0]), sub(p[2], p[0])));
585 for (int j = 0; j < 3; ++j) {
586 const V a = unit(sub(p[(j + 1) % 3], p[j])), b = unit(sub(p[(j + 2) % 3], p[j]));
587 const double angle = std::acos(std::clamp(dot(a, b), -1., 1.));
588 accumulated[p[j]] = add(accumulated[p[j]], mul(normal, angle));
589 }
590 }
591 for (int i = 0; i < mesh.getVertexCount(); ++i) {
592 const V point{mesh.getPositionX(i), mesh.getPositionY(i), mesh.getPositionZ(i)};
593 const V sum = accumulated.at(point);
594 if (length(sum) < 1e-16)
595 return Result<MeshBuild>::failure(invalid("Flowing arrow has an undefined normal"));
596 const V average = unit(sum);
597 const V face{mesh.getNormalX(i), mesh.getNormalY(i), mesh.getNormalZ(i)};
598 // Preserve a crease when a shared average exceeds 75 degrees from this face.
599 const V normal = dot(average, face) >= 0.2588190451 ? average : face;
600 for (int j = 0; j < 3; ++j)
601 mesh.normals()[static_cast<std::size_t>(i) * 3 + j] = static_cast<float>(normal[j]);
602 }
603 }
604 return Result<MeshBuild>::success(std::move(mesh));
605}
