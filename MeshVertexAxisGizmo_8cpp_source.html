33Result<std::string> MeshVertexAxisGizmoBuilder::pickAxisResult(const editing::GizmoSnapshot& snapshot, double originX,
34 double originY, double originZ, double directionX,
35 double directionY, double directionZ) const {
36 const double rayLength = std::sqrt(directionX * directionX + directionY * directionY + directionZ * directionZ);
37 if (!std::isfinite(rayLength) || rayLength <= 1e-9)
38 return Result<std::string>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
39 "axis pick requires a finite non-zero ray", "ray", {},
40 "procgen_editing.meshVertexAxisGizmo"));
41 const std::array<double, 3> ray{directionX / rayLength, directionY / rayLength, directionZ / rayLength};
42 double bestDistance = std::numeric_limits<double>::infinity();
43 std::string best;
44 for (const auto& primitive : snapshot.primitives) {
45 if (primitive.kind != "arrow" || primitive.id.rfind("mesh.axis.", 0) != 0) continue;
46 const std::array<double, 3> a = primitive.position;
47 const std::array<double, 3> b{a[0] + primitive.direction[0] * primitive.length,
48 a[1] + primitive.direction[1] * primitive.length,
49 a[2] + primitive.direction[2] * primitive.length};
50 const std::array<double, 3> segment{b[0] - a[0], b[1] - a[1], b[2] - a[2]};
51 const std::array<double, 3> offset{originX - a[0], originY - a[1], originZ - a[2]};
52 const double rr = 1.0;
53 const double ss = segment[0] * segment[0] + segment[1] * segment[1] + segment[2] * segment[2];
54 const double rs = ray[0] * segment[0] + ray[1] * segment[1] + ray[2] * segment[2];
55 const double ro = ray[0] * offset[0] + ray[1] * offset[1] + ray[2] * offset[2];
56 const double so = segment[0] * offset[0] + segment[1] * offset[1] + segment[2] * offset[2];
57 const double denominator = rr * ss - rs * rs;
58 double rayT = denominator > 1e-12 ? (rs * so - ss * ro) / denominator : -ro;
59 double segmentT = denominator > 1e-12 ? (rr * so - rs * ro) / denominator : 0.0;
60 rayT = std::max(0.0, rayT);
61 segmentT = std::clamp(segmentT, 0.0, 1.0);
62 const double dx = originX + ray[0] * rayT - (a[0] + segment[0] * segmentT);
63 const double dy = originY + ray[1] * rayT - (a[1] + segment[1] * segmentT);
64 const double dz = originZ + ray[2] * rayT - (a[2] + segment[2] * segmentT);
65 const double distance = std::sqrt(dx * dx + dy * dy + dz * dz);
66 if (distance <= primitive.length * 0.12 && rayT < bestDistance) {
67 bestDistance = rayT;
68 best = primitive.id.substr(10);
69 }
70 }
71 if (best.empty())
72 return Result<std::string>::failure(Diagnostic::error(DiagnosticCode::NotFound, "ray did not hit an axis",
73 "ray", {}, "procgen_editing.meshVertexAxisGizmo"));
74 return Result<std::string>::success(std::move(best));
75}
