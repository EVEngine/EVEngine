60std::vector<V2> samplePlaneCentres(std::uint32_t seed, float radius, float minDistance, int maxLeaves) {
61 std::vector<V2> centres;
62 if (radius <= 0.f || minDistance <= 0.f || maxLeaves <= 0) return centres;
63
64 // Hexagonal packing fits about 2A / (sqrt(3) d^2) centres in area A. Solving
65 // for the disk makes the *disk* land on the leaf cap; the enclosing square
66 // then saturates above it, so the budget below is only a safety cap and the
67 // fill is never truncated (a truncated Bridson fill is spatially biased).
68 const float capSpacing = std::sqrt(2.f * kPi * radius * radius / (std::sqrt(3.f) * float(maxLeaves)));
69 const float spacing = std::max(minDistance, capSpacing);
70 const int squareSide = std::max(2, int(std::ceil(2.f * radius)));
71 const float squareArea = float(squareSide) * float(squareSide);
72 const int squareBudget = int(1.3f * float(maxLeaves) * squareArea / std::max(kPi * radius * radius, 1e-6f)) + 8;
73
74 const PointSet square = poissonDiskPoints(squareSide, squareSide, spacing, seed, squareBudget);
75 const float half = 0.5f * float(squareSide);
76 // `poissonDiskPoints` samples the XZ plane, so the second area axis is z.
77 for (const ProcgenPoint &point : square.points()) {
78 const float dx = point.x - half;
79 const float dz = point.z - half;
80 if (dx * dx + dz * dz > radius * radius) continue;
81 centres.push_back({dx, dz});
82 if (int(centres.size()) >= maxLeaves) break;
83 }
84 return centres;
85}
94template <typename NormalAt>
95void addPetal(MeshBuild &out, V3 center, V3 right, V3 forward, float size, float roll, float u0, float u1,
96 bool doubleSided, const NormalAt &normalAt) {
97 const float c = std::cos(roll), s = std::sin(roll);
98 const V3 axisX = add(mul(right, c), mul(forward, s));
99 const V3 axisY = add(mul(right, -s), mul(forward, c));
100
101 // Lanceolate outline: pointed base, widest just above the middle, single
102 // tip. Six points read as a leaf only when the shoulders sit low and the
103 // taper towards the tip stays long; a symmetric hexagon reads as a pebble.
104 constexpr float kOutlineX[6] = {0.f, -0.42f, -0.50f, 0.f, 0.50f, 0.42f};
105 constexpr float kOutlineY[6] = {-0.50f, -0.24f, 0.06f, 0.50f, 0.06f, -0.24f};
106
107 const uint32_t base = uint32_t(out.getVertexCount());
108 for (int i = 0; i < 6; ++i) {
109 const V3 p = add(center, add(mul(axisX, kOutlineX[i] * size), mul(axisY, kOutlineY[i] * size)));
110 const V3 n = normalAt(p);
111 const float u = u0 + (0.5f + kOutlineX[i]) * (u1 - u0);
112 const float v = 0.5f + kOutlineY[i];
113 out.addVertex(p.x, p.y, p.z, n.x, n.y, n.z, u, v);
114 }
115 for (int i = 1; i < 5; ++i) {
116 out.addTriangle(base, base + uint32_t(i), base + uint32_t(i + 1));
117 }
118 if (doubleSided) {
119 for (int i = 1; i < 5; ++i) {
120 out.addTriangle(base, base + uint32_t(i + 1), base + uint32_t(i));
121 }
122 }
123}

127int addFoliageCluster(MeshBuild &out, const FoliageClusterDesc &desc) {
128 const float radius = std::max(1e-3f, desc.radius);
129 const float leafSize = std::max(1e-3f, desc.leafSize);
130 const float leafSpacing = std::max(0.05f, desc.leafSpacing);
131 const int planes = std::clamp(desc.planes, 1, 24);
132 const int capPlanes = std::clamp(desc.capPlanes, 0, 8);
133 const float tiltMax = std::clamp(desc.tiltDegrees, 0.f, 80.f) * kPi / 180.f;
134 const float planeOffset = std::clamp(desc.planeOffset, 0.f, 0.6f) * radius;
135 const float scaleVar = std::clamp(desc.planeScaleVariation, 0.f, 0.6f);
136 const float leafJitter = std::clamp(desc.leafJitter, 0.f, 0.5f) * radius;
137 const float sizeVar = std::clamp(desc.leafScaleVariation, 0.f, 0.6f);
138 const int maxLeaves = std::clamp(desc.maxLeavesPerPlane, 1, 256);
139 const float rounding = std::clamp(desc.normalRounding, 0.f, 1.f);
140 const float uvMin = std::min(desc.uvMin, desc.uvMax);
141 const float uvMax = std::max(desc.uvMin, desc.uvMax);
142
143 const V3 center{desc.centerX, desc.centerY, desc.centerZ};
144 std::mt19937 rng(desc.seed);
145
146 // Requested blue-noise spacing. `samplePlaneCentres` widens it further when
147 // it would otherwise overshoot the per-plane leaf cap.
148 const float minDistance = leafSize * leafSpacing;
149
150 // Vertex normals come from the cluster sphere so a bundle of flat cards
151 // still shades as one rounded mass. Near the centre the sphere direction is
152 // ill-conditioned, so the blend falls back to the plane normal there.
153 const float roundingDistance = std::max(1e-4f, radius * 0.35f);
154 const auto normalAt = [&](V3 p, V3 planeNormal) {
155 const V3 radial = sub(p, center);
156 const float length = std::sqrt(dot(radial, radial));
157 const float weight = rounding * std::clamp(length / roundingDistance, 0.f, 1.f);
158 if (length < 1e-5f || weight <= 0.f) return planeNormal;
159 const V3 blended = add(mul(mul(radial, 1.f / length), weight), mul(planeNormal, 1.f - weight));
160 if (dot(blended, blended) < 1e-6f) return planeNormal;
161 return norm(blended);
162 };
163
164 const int planeCount = planes + capPlanes;
165 const float yawPhase = randomRange(rng, 0.f, 2.f * kPi);
166 const float yawStep = 2.f * kPi / float(planeCount);
167 int leaves = 0;
168 for (int index = 0; index < planeCount; ++index) {
169 const bool cap = index >= planes;
170
171 // Ring planes take an even share of the circle plus a jitter, so the
172 // bundle never reads as a regular polygon from any angle.
173 const float yaw =
174 cap ? randomRange(rng, 0.f, 2.f * kPi) : yawPhase + (float(index) + randomRange(rng, 0.2f, 0.8f)) * yawStep;
175 // Ring planes lean a little off vertical; the caps lean near-horizontal
176 // so the cluster's poles are not left hollow.
177 const float tilt = cap ? ((index & 1) ? 1.f : -1.f) * randomRange(rng, 62.f, 86.f) * kPi / 180.f
178 : randomRange(rng, -tiltMax, tiltMax);
179 const V3 planeNormal{std::sin(yaw) * std::cos(tilt), std::sin(tilt), std::cos(yaw) * std::cos(tilt)};
180
181 V3 right, forward;
182 basisFor(planeNormal, right, forward);
183
184 const float radialScale = 1.f - scaleVar * randomRange(rng, 0.f, 1.f) - (cap ? 0.35f : 0.f);
185 const float planeRadius = radius * std::max(0.2f, radialScale);
186 const float offsetAngle = randomRange(rng, 0.f, 2.f * kPi);
187 const float offsetR = planeOffset * std::sqrt(randomRange(rng, 0.f, 1.f));
188 const V3 origin = add(
189 center, add(mul(right, std::cos(offsetAngle) * offsetR), mul(forward, std::sin(offsetAngle) * offsetR)));
190
191 const std::vector<V2> centers = samplePlaneCentres(desc.seed ^ (std::uint32_t(index + 1) * 2654435761u),
192 planeRadius, minDistance, maxLeaves);
193 for (const V2 &point : centers) {
194 const float jitter = randomRange(rng, -1.f, 1.f) * leafJitter;
195 const V3 position =
196 add(origin, add(add(mul(right, point.x), mul(forward, point.y)), mul(planeNormal, jitter)));
197
198 // Point the petals outward from the cluster centre so the bundle
199 // reads as growth radiating from the branch rather than confetti.
200 float roll = randomRange(rng, 0.f, 2.f * kPi);
201 const V3 radial = sub(position, center);
202 const V3 tangent = sub(radial, mul(planeNormal, dot(radial, planeNormal)));
203 if (dot(tangent, tangent) > 1e-8f) {
204 const V3 outward = norm(tangent);
205 roll = std::atan2(dot(outward, forward), dot(outward, right)) - 0.5f * kPi +
206 randomRange(rng, -0.85f, 0.85f);
207 }
208
209 const float size = leafSize * (1.f + randomRange(rng, -sizeVar, sizeVar));
210 const float u0 = uvMin + randomRange(rng, 0.f, 0.35f) * (uvMax - uvMin);
211 const float u1 = std::min(uvMax, u0 + 0.65f * (uvMax - uvMin));
212 const auto shadedNormal = [&](V3 p) { return normalAt(p, planeNormal); };
213 addPetal(out, position, right, forward, size, roll, u0, u1, desc.doubleSided, shadedNormal);
214 ++leaves;
215 }
216 }
217 return leaves;
218}
