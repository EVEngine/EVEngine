68Result<std::uint32_t> RoadNetwork::addEdge(std::uint32_t from, std::uint32_t to,
69 std::vector<RoadControlPoint> controlPoints, int lanesForward,
70 int lanesBackward, RoadStyle style) {
71 if (findNodeIndex(from) < 0 || findNodeIndex(to) < 0)
72 return Result<std::uint32_t>::failure(
73 Diagnostic::error(DiagnosticCode::NotFound, "edge endpoints must reference live nodes", "edge"));
74 if (controlPoints.size() < 2)
75 return Result<std::uint32_t>::failure(
76 Diagnostic::error(DiagnosticCode::InvalidArgument, "edge needs at least two control points", "edge"));
77 for (const auto& p : controlPoints) {
78 if (!finite3(p.x, p.y, p.z))
79 return Result<std::uint32_t>::failure(
80 Diagnostic::error(DiagnosticCode::InvalidArgument, "control points must be finite", "edge"));
81 }
82 auto styleOk = validateStyle(style);
83 if (!styleOk.ok()) return Result<std::uint32_t>::failure(styleOk.status());
84 auto profile = makeRoadProfile(style, lanesForward, lanesBackward);
85 if (!profile.ok()) return Result<std::uint32_t>::failure(profile.status());
86
87 const std::uint32_t id = nextEdgeId_++;
88 edgeIndex_[id] = static_cast<int>(edges_.size());
89 RoadEdge edge;
90 edge.id = id;
91 edge.from = from;
92 edge.to = to;
93 edge.controlPoints = std::move(controlPoints);
94 edge.lanesForward = lanesForward;
95 edge.lanesBackward = lanesBackward;
96 edge.style = style;
97 edges_.push_back(std::move(edge));
98 ++revision_;
99 return Result<std::uint32_t>::success(id);
100}

102Result<void> RoadNetwork::addLaneLink(RoadLaneConnection link) {
103 const int inIdx = findEdgeIndex(link.inEdge);
104 const int outIdx = findEdgeIndex(link.outEdge);
105 if (inIdx < 0 || outIdx < 0)
106 return Result<void>::failure(
107 Diagnostic::error(DiagnosticCode::NotFound, "lane link references missing edges", "laneLink"));
108 const auto& inEdge = edges_[static_cast<std::size_t>(inIdx)];
109 const auto& outEdge = edges_[static_cast<std::size_t>(outIdx)];
110 if (inEdge.to != outEdge.from)
111 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
112 "lane link requires inEdge.to == outEdge.from", "laneLink"));
113 if (link.inLane < 0 || link.inLane >= inEdge.lanesForward)
114 return Result<void>::failure(
115 Diagnostic::error(DiagnosticCode::InvalidArgument, "inLane out of range", "laneLink"));
116 if (link.outLane < 0 || link.outLane >= outEdge.lanesForward)
117 return Result<void>::failure(
118 Diagnostic::error(DiagnosticCode::InvalidArgument, "outLane out of range", "laneLink"));
119 laneLinks_.push_back(link);
120 ++revision_;
121 return Result<void>::success();
122}

124Result<int> RoadNetwork::connectAllTurns(std::uint32_t nodeId) {
125 if (findNodeIndex(nodeId) < 0)
126 return Result<int>::failure(Diagnostic::error(DiagnosticCode::NotFound, "unknown node", "nodeId"));
127 int added = 0;
128 for (const auto& inEdge : edges_) {
129 if (inEdge.to != nodeId) continue;
130 for (const auto& outEdge : edges_) {
131 if (outEdge.from != nodeId) continue;
132 if (outEdge.id == inEdge.id) continue;
133 for (int il = 0; il < inEdge.lanesForward; ++il) {
134 const int ol = std::min(il, outEdge.lanesForward - 1);
135 auto r = addLaneLink(RoadLaneConnection{inEdge.id, il, outEdge.id, ol});
136 if (!r.ok()) return Result<int>::failure(r.status());
137 ++added;
138 }
139 }
140 }
141 return Result<int>::success(added);
142}

235Result<RoadNetwork> RoadNetwork::makeCross(float span, int lanes) {
236 if (!std::isfinite(span) || span < 16.f || lanes < 1 || lanes > 4)
237 return Result<RoadNetwork>::failure(
238 Diagnostic::error(DiagnosticCode::InvalidArgument, "span>=16, lanes in [1,4] required", "cross"));
239 RoadNetwork network;
240 const float half = span * 0.5f;
241 RoadStyle style = groundStyle();
242 style.deckThickness = 0.08f;
243 style.sidewalkWidth = 1.2f;
244 style.curbWidth = 0.35f;
245 style.curbHeight = 0.45f;
246 // Push arms outward past the asphalt half-width so sidewalks do not overlap;
247 // the leftover ring is filled by quarter-circle curb returns.
248 const float asphaltHalf = 0.5f * style.laneWidth * static_cast<float>(lanes);
249 const float cornerR = 2.8f;
250 const float jr = asphaltHalf + cornerR;
251 auto nC = network.addNode(0.f, 0.f, 0.f, jr);
252 auto nN = network.addNode(0.f, 0.f, -half, 2.f);
253 auto nS = network.addNode(0.f, 0.f, half, 2.f);
254 auto nW = network.addNode(-half, 0.f, 0.f, 2.f);
255 auto nE = network.addNode(half, 0.f, 0.f, 2.f);
256 for (auto* r : {&nC, &nN, &nS, &nW, &nE}) {
257 if (!r->ok()) return Result<RoadNetwork>::failure(r->status());
258 }
259 // Arms run into the hub; bake trims each end by junctionRadius so strips
260 // stop outside the corner arcs.
261 auto e1 = network.addEdge(nN.value(), nC.value(), {P(0.f, 0.f, -half), P(0.f, 0.f, 0.f)}, lanes, 0, style);
262 auto e2 = network.addEdge(nC.value(), nS.value(), {P(0.f, 0.f, 0.f), P(0.f, 0.f, half)}, lanes, 0, style);
263 auto e3 = network.addEdge(nW.value(), nC.value(), {P(-half, 0.f, 0.f), P(0.f, 0.f, 0.f)}, lanes, 0, style);
264 auto e4 = network.addEdge(nC.value(), nE.value(), {P(0.f, 0.f, 0.f), P(half, 0.f, 0.f)}, lanes, 0, style);
265 for (auto* e : {&e1, &e2, &e3, &e4}) {
266 if (!e->ok()) return Result<RoadNetwork>::failure(e->status());
267 }
268 auto turns = network.connectAllTurns(nC.value());
269 if (!turns.ok()) return Result<RoadNetwork>::failure(turns.status());
270 return Result<RoadNetwork>::success(std::move(network));
271}

284Result<RoadNetwork> RoadNetwork::makeInterchange(float span, float bridgeHeight, int lanes, std::uint32_t seed) {
285 if (!std::isfinite(span) || span < 16.f || !std::isfinite(bridgeHeight) || bridgeHeight < 1.f || lanes < 1 ||
286 lanes > 4)
287 return Result<RoadNetwork>::failure(Diagnostic::error(
288 DiagnosticCode::InvalidArgument, "span>=16, bridgeHeight>=1, lanes in [1,4] required", "interchange"));
289
290 RoadNetwork network;
291 const float half = span * 0.5f;
292 const float h = bridgeHeight;
293 // Tiny seed wobble on elevated mid-handles only — ground cross stays axis-aligned
294 // so bakeJunction keeps the same outward-center curb returns as makeCross.
295 const float wobble = 0.25f * static_cast<float>((seed % 7u) - 3);
296
297 RoadStyle ground = groundStyle();
298 ground.deckThickness = 0.08f;
299 ground.sidewalkWidth = 1.2f;
300 ground.curbWidth = 0.35f;
301 ground.curbHeight = 0.45f;
302 const float asphaltHalf = 0.5f * ground.laneWidth * static_cast<float>(lanes);
303 const float cornerR = 2.8f;
304 const float jr = asphaltHalf + cornerR;
305
306 RoadStyle bridge = bridgeStyle();
307 bridge.pierSpacing = 14.0f;
308 bridge.sidewalkWidth = 1.0f;
309 bridge.curbWidth = 0.38f;
310 bridge.curbHeight = 0.50f;
311
312 auto nGround = network.addNode(0.f, 0.f, 0.f, jr);
313 auto nN = network.addNode(0.f, 0.f, -half, 2.f);
314 auto nS = network.addNode(0.f, 0.f, half, 2.f);
315 auto nW = network.addNode(-half, 0.f, 0.f, 2.f);
316 auto nE = network.addNode(half, 0.f, 0.f, 2.f);
317
318 // Elevated ring corners + hub (overpass).
319 const float r = half * 0.58f;
320 auto nElev = network.addNode(0.f, h, 0.f, 4.0f);
321 auto nNE = network.addNode(r, h, -r, 2.5f);
322 auto nSE = network.addNode(r, h, r, 2.5f);
323 auto nSW = network.addNode(-r, h, r, 2.5f);
324 auto nNW = network.addNode(-r, h, -r, 2.5f);
325 for (auto* res : {&nGround, &nN, &nS, &nW, &nE, &nElev, &nNE, &nSE, &nSW, &nNW}) {
326 if (!res->ok()) return Result<RoadNetwork>::failure(res->status());
327 }
328
329 auto add = [&](std::uint32_t a, std::uint32_t b, std::vector<RoadControlPoint> pts, const RoadStyle& style) {
330 return network.addEdge(a, b, std::move(pts), lanes, 0, style);
331 };
332
333 // Ground cross — straight axis arms (same fillet path as makeCross).
334 const auto g = nGround.value();
335 auto e1 = add(nN.value(), g, {P(0.f, 0.f, -half), P(0.f, 0.f, 0.f)}, ground);
336 auto e2 = add(g, nS.value(), {P(0.f, 0.f, 0.f), P(0.f, 0.f, half)}, ground);
337 auto e3 = add(nW.value(), g, {P(-half, 0.f, 0.f), P(0.f, 0.f, 0.f)}, ground);
338 auto e4 = add(g, nE.value(), {P(0.f, 0.f, 0.f), P(half, 0.f, 0.f)}, ground);
339
340 // Elevated ring — quarter arcs with mid handles so the loft reads circular.
341 auto e5 = add(nNW.value(), nNE.value(),
342 {P(-r, h, -r), P(-r * 0.35f + wobble, h, -r * 1.12f), P(r * 0.35f, h, -r * 1.12f), P(r, h, -r)},
343 bridge);
344 auto e6 = add(nNE.value(), nSE.value(),
345 {P(r, h, -r), P(r * 1.12f, h, -r * 0.35f), P(r * 1.12f, h, r * 0.35f + wobble), P(r, h, r)},
346 bridge);
347 auto e7 = add(nSE.value(), nSW.value(),
348 {P(r, h, r), P(r * 0.35f, h, r * 1.12f), P(-r * 0.35f - wobble, h, r * 1.12f), P(-r, h, r)},
349 bridge);
350 auto e8 = add(nSW.value(), nNW.value(),
351 {P(-r, h, r), P(-r * 1.12f, h, r * 0.35f), P(-r * 1.12f, h, -r * 0.35f - wobble), P(-r, h, -r)},
352 bridge);
353
354 // Diagonal overpass through the elevated hub.
355 auto e9 = add(nNW.value(), nElev.value(),
356 {P(-r, h, -r), P(-r * 0.45f, h, -r * 0.45f), P(-5.5f, h, -5.5f)}, bridge);
357 auto e10 = add(nElev.value(), nSE.value(),
358 {P(5.5f, h, 5.5f), P(r * 0.45f, h, r * 0.45f), P(r, h, r)}, bridge);
359
360 // Ramps: ground east → elevated SE; elevated NW → ground west.
361 auto e11 = add(nE.value(), nSE.value(),
362 {P(half, 0.f, 0.f), P(half * 0.82f, h * 0.22f, half * 0.12f),
363 P(half * 0.68f, h * 0.55f, half * 0.32f), P(half * 0.58f, h * 0.85f, half * 0.48f), P(r, h, r)},
364 bridge);
365 auto e12 = add(nNW.value(), nW.value(),
366 {P(-r, h, -r), P(-half * 0.58f, h * 0.85f, -half * 0.42f),
367 P(-half * 0.68f, h * 0.55f, -half * 0.28f), P(-half * 0.82f, h * 0.22f, -half * 0.10f),
368 P(-half, 0.f, 0.f)},
369 bridge);
370
371 for (auto* edge : {&e1, &e2, &e3, &e4, &e5, &e6, &e7, &e8, &e9, &e10, &e11, &e12}) {
372 if (!edge->ok()) return Result<RoadNetwork>::failure(edge->status());
373 }
374
375 // Turns only where they help nav overlay — ground hub (filleted cross) + elevated hub.
376 for (std::uint32_t node : {g, nElev.value()}) {
377 auto turns = network.connectAllTurns(node);
378 if (!turns.ok()) return Result<RoadNetwork>::failure(turns.status());
379 }
380
381 return Result<RoadNetwork>::success(std::move(network));
382}
