27 point.widthScale <= 16.f && std::isfinite(point.rollDegrees) && std::fabs(point.rollDegrees) <= 360.f &&

33Result<void> validateCircularPoints(const std::vector<RoadControlPoint>& points) {
34 if (std::none_of(points.begin(), points.end(),
35 [](const auto& p) { return p.pathControl == RoadPathControl::Circular; }))
36 return Result<void>::success();
37 eve::procgen::SplinePath path;
38 for (const auto& p : points) {
39 eve::procgen::SplinePoint q;
40 q.x = p.x;
41 q.y = p.y;
42 q.z = p.z;
43 q.spanControl = static_cast<eve::procgen::SplineSpanControl>(p.pathControl);
44 q.circularFromEnd = p.circularFromEnd;
45 auto added = path.addPointResult(q);
46 if (!added.ok()) return added;
47 }
48 auto evaluated = path.evaluateResult(0.f);
49 if (!evaluated.ok()) return Result<void>::failure(evaluated.status());
50 return Result<void>::success();
51}

59bool hasUsableGeometry(const std::vector<RoadControlPoint>& points) {
60 constexpr float minimumSegmentLengthSquared = 1e-8f;
61 for (std::size_t i = 1; i < points.size(); ++i) {
62 const float dx = points[i].x - points[i - 1].x;
63 const float dy = points[i].y - points[i - 1].y;
64 const float dz = points[i].z - points[i - 1].z;
65 if (dx * dx + dy * dy + dz * dz > minimumSegmentLengthSquared) return true;
66 }
67 return false;
68}

116int mapLaneByLateralRank(int inLane, int inLaneCount, int outLaneCount) {

119 return static_cast<int>(std::lround(rank * static_cast<float>(outLaneCount - 1)));

170Result<std::uint32_t> RoadNetwork::addEdge(std::uint32_t from, std::uint32_t to,
171 std::vector<RoadControlPoint> controlPoints, int lanesForward,
172 int lanesBackward, RoadStyle style) {
173 if (findNodeIndex(from) < 0 || findNodeIndex(to) < 0)
174 return Result<std::uint32_t>::failure(
175 Diagnostic::error(DiagnosticCode::NotFound, "edge endpoints must reference live nodes", "edge"));
176 if (!nodes_[static_cast<std::size_t>(findNodeIndex(from))].connection.ports.empty() ||
177 !nodes_[static_cast<std::size_t>(findNodeIndex(to))].connection.ports.empty())
178 return Result<std::uint32_t>::failure(
179 Diagnostic::error(DiagnosticCode::PreconditionViolation,
180 "clear a custom connection before adding a new incident edge", "edge"));
181 if (from == to)
182 return Result<std::uint32_t>::failure(
183 Diagnostic::error(DiagnosticCode::PreconditionViolation, "edge endpoints must be distinct", "edge"));
184 if (controlPoints.size() < 2)
185 return Result<std::uint32_t>::failure(
186 Diagnostic::error(DiagnosticCode::InvalidArgument, "edge needs at least two control points", "edge"));
187 for (const auto& p : controlPoints) {
188 if (!validPoint(p))
189 return Result<std::uint32_t>::failure(Diagnostic::error(
190 DiagnosticCode::InvalidArgument, "control point positions and profiles must be valid", "edge"));
191 }
192 auto styleOk = validateStyle(style);
193 if (!styleOk.ok()) return Result<std::uint32_t>::failure(styleOk.status());
194 auto profile = makeRoadProfile(style, lanesForward, lanesBackward);
195 if (!profile.ok()) return Result<std::uint32_t>::failure(profile.status());
196
197 // A node is the authoritative connection position. Canonicalizing the two
198 // endpoint samples prevents authored handles from leaving invisible gaps or
199 // sending turn curves toward a stale position.
200 const auto& fromNode = nodes_[static_cast<std::size_t>(findNodeIndex(from))];
201 const auto& toNode = nodes_[static_cast<std::size_t>(findNodeIndex(to))];
202 anchorPoint(controlPoints.front(), fromNode.x, fromNode.y, fromNode.z);
203 anchorPoint(controlPoints.back(), toNode.x, toNode.y, toNode.z);
204 auto circular = validateCircularPoints(controlPoints);
205 if (!circular.ok()) return Result<std::uint32_t>::failure(circular.status());
206 if (!hasUsableGeometry(controlPoints))
207 return Result<std::uint32_t>::failure(Diagnostic::error(
208 DiagnosticCode::PreconditionViolation, "edge centerline must contain a non-degenerate segment", "edge"));
209
210 const std::uint32_t id = nextEdgeId_++;
211 edgeIndex_[id] = static_cast<int>(edges_.size());
212 RoadEdge edge;
213 edge.id = id;
214 edge.from = from;
215 edge.to = to;
216 edge.controlPoints = std::move(controlPoints);
217 edge.lanesForward = lanesForward;
218 edge.lanesBackward = lanesBackward;
219 edge.style = style;
220 edges_.push_back(std::move(edge));
221 ++revision_;
222 return Result<std::uint32_t>::success(id);
223}

266Result<void> RoadNetwork::validateLaneConnection(const RoadLaneConnection& link) const {
267 const int inIdx = findEdgeIndex(link.inEdge);
268 const int outIdx = findEdgeIndex(link.outEdge);
269 if (inIdx < 0 || outIdx < 0)
270 return Result<void>::failure(
271 Diagnostic::error(DiagnosticCode::NotFound, "lane link references missing edges", "laneLink"));
272 const auto& inEdge = edges_[static_cast<std::size_t>(inIdx)];
273 const auto& outEdge = edges_[static_cast<std::size_t>(outIdx)];
274 const auto sharedNode = laneEntryNode(inEdge, link.inDirection);
275 if (sharedNode != laneExitNode(outEdge, link.outDirection))
276 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
277 "lane link directions must meet at the same node", "laneLink"));
278 if (link.inEdge == link.outEdge)
279 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
280 "lane link cannot make an immediate U-turn on one edge",
281 "laneLink"));
282 if (link.inLane < 0 || link.inLane >= laneCount(inEdge, link.inDirection))
283 return Result<void>::failure(
284 Diagnostic::error(DiagnosticCode::InvalidArgument, "inLane out of range", "laneLink"));
285 if (link.outLane < 0 || link.outLane >= laneCount(outEdge, link.outDirection))
286 return Result<void>::failure(
287 Diagnostic::error(DiagnosticCode::InvalidArgument, "outLane out of range", "laneLink"));
288 return Result<void>::success();
289}

291Result<void> RoadNetwork::addLaneLink(RoadLaneConnection link) {
292 auto valid = validateLaneConnection(link);
293 if (!valid.ok()) return valid;
294 if (std::any_of(laneLinks_.begin(), laneLinks_.end(),
295 [&](const auto& existing) { return sameLaneLink(existing, link); }))
296 return Result<void>::failure(
297 Diagnostic::error(DiagnosticCode::AlreadyExists, "lane link already exists", "laneLink"));
298 if (std::any_of(blockedLaneLinks_.begin(), blockedLaneLinks_.end(),
299 [&](const auto& blocked) { return sameLaneLink(blocked, link); }))
300 return Result<void>::failure(
301 Diagnostic::error(DiagnosticCode::PreconditionViolation, "lane link is persistently blocked", "laneLink"));
302 laneLinks_.insert(std::lower_bound(laneLinks_.begin(), laneLinks_.end(), link, laneOrderLess), link);
303 ++revision_;
304 return Result<void>::success();
305}

307Result<bool> RoadNetwork::blockLaneLink(RoadLaneConnection link) {
308 auto valid = validateLaneConnection(link);
309 if (!valid.ok()) return Result<bool>::failure(valid.status());
310 if (std::any_of(blockedLaneLinks_.begin(), blockedLaneLinks_.end(),
311 [&](const auto& blocked) { return sameLaneLink(blocked, link); }))
312 return Result<bool>::failure(
313 Diagnostic::error(DiagnosticCode::AlreadyExists, "lane link is already blocked", "laneLink"));
314 auto candidate = *this;
315 const auto active = std::find_if(candidate.laneLinks_.begin(), candidate.laneLinks_.end(),
316 [&](const auto& current) { return sameLaneLink(current, link); });
317 const bool removed = active != candidate.laneLinks_.end();
318 if (removed) candidate.laneLinks_.erase(active);
319 candidate.blockedLaneLinks_.insert(
320 std::lower_bound(candidate.blockedLaneLinks_.begin(), candidate.blockedLaneLinks_.end(), link, laneOrderLess),
321 link);
322 auto checked = candidate.validate();
323 if (!checked.ok()) return Result<bool>::failure(checked.status());
324 candidate.revision_ = revision_ + 1;
325 *this = std::move(candidate);
326 return Result<bool>::success(removed);
327}

351Result<int> RoadNetwork::connectAllTurns(std::uint32_t nodeId) {
352 if (findNodeIndex(nodeId) < 0)
353 return Result<int>::failure(Diagnostic::error(DiagnosticCode::NotFound, "unknown node", "nodeId"));
354 auto candidate = *this;
355 int added = 0;
356 constexpr RoadLaneDirection directions[] = {RoadLaneDirection::Forward, RoadLaneDirection::Backward};
357 for (const auto& inEdge : candidate.edges_) {
358 for (const auto inDirection : directions) {
359 const int inLanes = laneCount(inEdge, inDirection);
360 if (inLanes == 0 || laneEntryNode(inEdge, inDirection) != nodeId) continue;
361 for (const auto& outEdge : candidate.edges_) {
362 if (outEdge.id == inEdge.id) continue;
363 for (const auto outDirection : directions) {
364 const int outLanes = laneCount(outEdge, outDirection);
365 if (outLanes == 0 || laneExitNode(outEdge, outDirection) != nodeId) continue;
366 for (int il = 0; il < inLanes; ++il) {
367 // Preserve normalized lateral rank when lane counts change. This
368 // distributes merges/expansions across the full destination width
369 // instead of collapsing every overflow lane onto one side.
370 const int ol = mapLaneByLateralRank(il, inLanes, outLanes);
371 RoadLaneConnection link{inEdge.id, il, outEdge.id, ol, inDirection, outDirection};
372 const bool blocked = std::any_of(candidate.blockedLaneLinks_.begin(),
373 candidate.blockedLaneLinks_.end(),
374 [&](const auto& current) {
375 return sameLaneLink(current, link);
376 });
377 if (blocked) continue;
378 // Any authored link for the same incoming lane and outgoing
379 // physical route is an override, even when it selects another
380 // destination lane. Do not re-add an automatic parallel turn.
381 const bool overridden =
382 std::any_of(candidate.laneLinks_.begin(), candidate.laneLinks_.end(), [&](const auto& current) {
383 return current.inEdge == link.inEdge && current.inLane == link.inLane &&
384 current.outEdge == link.outEdge && current.inDirection == link.inDirection &&
385 current.outDirection == link.outDirection;
386 });
387 if (overridden) continue;
388 auto r = candidate.addLaneLink(link);
389 if (!r.ok()) return Result<int>::failure(r.status());
390 ++added;
391 }
392 }
393 }
394 }
395 }
396 if (added == 0) return Result<int>::success(0);
397 auto valid = candidate.validate();
398 if (!valid.ok()) return Result<int>::failure(valid.status());
399 candidate.revision_ = revision_ + 1;
400 *this = std::move(candidate);
401 return Result<int>::success(added);
402}

404Result<void> RoadNetwork::validate() const {
405 for (const auto& node : nodes_) {
406 if (!finite3(node.x, node.y, node.z) || !std::isfinite(node.junctionRadius) || node.junctionRadius <= 0.f ||
407 !validJunctionControl(node.junctionControl))
408 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
409 "network contains an invalid junction node", "node"));
410 }
411 for (const auto& node : nodes_) {
412 auto connection = detail::validateConnectionSeams(*this, node);
413 if (!connection.ok()) return connection;
414 auto routes = detail::validateFacilityRoutes(*this, node);
415 if (!routes.ok()) return routes;
416 }
417 for (const auto& edge : edges_) {
418 for (const auto& point : edge.controlPoints) {
419 if (!validPoint(point))
420 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
421 "edge contains invalid control-point profiles", "edge"));
422 }
423 auto circular = validateCircularPoints(edge.controlPoints);
424 if (!circular.ok()) return circular;
425 auto styleOk = validateStyle(edge.style);
426 if (!styleOk.ok()) return styleOk;
427 auto profile = makeRoadProfile(edge.style, edge.lanesForward, edge.lanesBackward);
428 if (!profile.ok()) return Result<void>::failure(profile.status());
429 if (findNodeIndex(edge.from) < 0 || findNodeIndex(edge.to) < 0)
430 return Result<void>::failure(
431 Diagnostic::error(DiagnosticCode::InvariantViolation, "edge references a missing node", "edge"));
432 if (edge.from == edge.to)
433 return Result<void>::failure(
434 Diagnostic::error(DiagnosticCode::InvariantViolation, "edge forms a self-loop", "edge"));
435 if (edge.controlPoints.size() < 2)
436 return Result<void>::failure(
437 Diagnostic::error(DiagnosticCode::InvariantViolation, "edge has fewer than two samples", "edge"));
438 if (!hasUsableGeometry(edge.controlPoints))
439 return Result<void>::failure(Diagnostic::error(
440 DiagnosticCode::InvariantViolation, "edge centerline has no usable segment", "edge"));
441 const auto& a = nodes_[static_cast<std::size_t>(findNodeIndex(edge.from))];
442 const auto& b = nodes_[static_cast<std::size_t>(findNodeIndex(edge.to))];
443 const auto& p0 = edge.controlPoints.front();
444 const auto& p1 = edge.controlPoints.back();
445 if (p0.x != a.x || p0.y != a.y || p0.z != a.z || p1.x != b.x || p1.y != b.y || p1.z != b.z)
446 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
447 "edge samples are not anchored to endpoint nodes", "edge"));
448 }
449 for (std::size_t i = 0; i < laneLinks_.size(); ++i) {
450 const auto& link = laneLinks_[i];
451 const int inIdx = findEdgeIndex(link.inEdge);
452 const int outIdx = findEdgeIndex(link.outEdge);
453 if (inIdx < 0 || outIdx < 0)

566Result<int> RoadNetwork::mergeNodes(std::uint32_t keepNodeId, std::uint32_t removeNodeId, float maxDistance) {
567 if (keepNodeId == removeNodeId)
568 return Result<int>::failure(
569 Diagnostic::error(DiagnosticCode::InvalidArgument, "merge nodes must be distinct", "removeNodeId"));
570 const int keepIndex = findNodeIndex(keepNodeId), removeIndex = findNodeIndex(removeNodeId);
571 if (keepIndex < 0 || removeIndex < 0)
572 return Result<int>::failure(
573 Diagnostic::error(DiagnosticCode::NotFound, "merge references an unknown node", "nodeId"));
574 if (!std::isfinite(maxDistance) || maxDistance < 0.f)
575 return Result<int>::failure(
576 Diagnostic::error(DiagnosticCode::InvalidArgument, "merge distance must be finite and non-negative",
577 "maxDistance"));
578 const auto& keep = nodes_[static_cast<std::size_t>(keepIndex)];
579 const auto& removed = nodes_[static_cast<std::size_t>(removeIndex)];
580 const float dx = keep.x - removed.x, dy = keep.y - removed.y, dz = keep.z - removed.z;
581 if (dx * dx + dy * dy + dz * dz > maxDistance * maxDistance)
582 return Result<int>::failure(Diagnostic::error(
583 DiagnosticCode::PreconditionViolation, "nodes are outside the merge distance", "maxDistance"));
584 if (std::any_of(edges_.begin(), edges_.end(), [&](const auto& edge) {
585 return (edge.from == keepNodeId && edge.to == removeNodeId) ||
586 (edge.from == removeNodeId && edge.to == keepNodeId);
587 }))
588 return Result<int>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
589 "merging nodes would create a self-loop edge", "removeNodeId"));
590
591 if (!nodes_[static_cast<std::size_t>(keepIndex)].connection.ports.empty() ||
592 !nodes_[static_cast<std::size_t>(removeIndex)].connection.ports.empty() ||
593 !nodes_[static_cast<std::size_t>(keepIndex)].connection.facilityRoutes.empty() ||
594 !nodes_[static_cast<std::size_t>(removeIndex)].connection.facilityRoutes.empty())
595 return Result<int>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
596 "clear custom connection assets before merging their nodes",
597 "nodeId"));
598 auto candidate = *this;
599 int rewired = 0;
600 for (auto& edge : candidate.edges_) {
601 if (edge.from == removeNodeId) {
602 edge.from = keepNodeId;
603 anchorPoint(edge.controlPoints.front(), keep.x, keep.y, keep.z);
604 ++rewired;
605 }
606 if (edge.to == removeNodeId) {
607 edge.to = keepNodeId;
608 anchorPoint(edge.controlPoints.back(), keep.x, keep.y, keep.z);
609 ++rewired;
610 }
611 }
612 candidate.nodes_.erase(candidate.nodes_.begin() + removeIndex);
613 candidate.nodeIndex_.clear();
614 for (std::size_t i = 0; i < candidate.nodes_.size(); ++i)
615 candidate.nodeIndex_[candidate.nodes_[i].id] = static_cast<int>(i);
616 auto valid = candidate.validate();
617 if (!valid.ok()) return Result<int>::failure(valid.status());
618 candidate.revision_ = revision_ + 1;
619 *this = std::move(candidate);
620 return Result<int>::success(rewired);
621}

687Result<void> RoadNetwork::reverseEdge(std::uint32_t edgeId) {
688 const int index = findEdgeIndex(edgeId);
689 if (index < 0) return Result<void>::failure(Diagnostic::error(DiagnosticCode::NotFound, "unknown edge", "edgeId"));
690 auto candidate = *this;
691 auto& edge = candidate.edges_[static_cast<std::size_t>(index)];
692 std::swap(edge.from, edge.to);
693 std::reverse(edge.controlPoints.begin(), edge.controlPoints.end());
694 // Strength belongs to outgoing spans, not the geometric marker after reversal.
695 const float terminalStrength = edge.controlPoints.front().splineStrength;
696 const auto terminalControl = edge.controlPoints.front().pathControl;
697 const bool terminalStart = edge.controlPoints.front().alignStraightStart,
698 terminalEnd = edge.controlPoints.front().alignStraightEnd;
699 const bool terminalCircle = edge.controlPoints.front().circularFromEnd;
700 for (std::size_t i = 0; i + 1 < edge.controlPoints.size(); ++i) {
701 edge.controlPoints[i].splineStrength = edge.controlPoints[i + 1].splineStrength;
702 edge.controlPoints[i].pathControl = edge.controlPoints[i + 1].pathControl;
703 edge.controlPoints[i].circularFromEnd = edge.controlPoints[i].pathControl == RoadPathControl::Circular
704 ? !edge.controlPoints[i + 1].circularFromEnd
705 : edge.controlPoints[i + 1].circularFromEnd;
706 edge.controlPoints[i].alignStraightStart = edge.controlPoints[i + 1].alignStraightEnd;
707 edge.controlPoints[i].alignStraightEnd = edge.controlPoints[i + 1].alignStraightStart;
708 }
709 edge.controlPoints.back().splineStrength = terminalStrength;
710 edge.controlPoints.back().pathControl = terminalControl;
711 edge.controlPoints.back().circularFromEnd =
712 terminalControl == RoadPathControl::Circular ? !terminalCircle : terminalCircle;
713 edge.controlPoints.back().alignStraightStart = terminalEnd;
714 edge.controlPoints.back().alignStraightEnd = terminalStart;
715 for (auto& point : edge.controlPoints) {
716 if (point.rollDegrees != 0.f) point.rollDegrees = -point.rollDegrees;
717 if (point.rollCenterOffset != 0.f) point.rollCenterOffset = -point.rollCenterOffset;
718 }
719 detail::reverseRoadAuthoring(edge.style);
720 std::swap(edge.lanesForward, edge.lanesBackward);
721 std::swap(edge.style.sideObjectStartOffset, edge.style.sideObjectEndOffset);
722 std::swap(edge.style.sideObjectsLeft, edge.style.sideObjectsRight);
723 const auto flip = [](RoadLaneDirection direction) {
724 return direction == RoadLaneDirection::Forward ? RoadLaneDirection::Backward : RoadLaneDirection::Forward;
725 };
726 for (auto& link : candidate.laneLinks_) {
727 if (link.inEdge == edgeId) link.inDirection = flip(link.inDirection);
728 if (link.outEdge == edgeId) link.outDirection = flip(link.outDirection);
729 }
730 for (auto& link : candidate.blockedLaneLinks_) {
731 if (link.inEdge == edgeId) link.inDirection = flip(link.inDirection);
732 if (link.outEdge == edgeId) link.outDirection = flip(link.outDirection);
733 }
734 candidate.normalizeLaneLinkOrder();
735 auto valid = candidate.validate();
736 if (!valid.ok()) return valid;
737 candidate.revision_ = revision_ + 1;
738 *this = std::move(candidate);
739 return Result<void>::success();
740}

744}
