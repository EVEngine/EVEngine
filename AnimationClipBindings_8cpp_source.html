20void exposeAnimClipBindings(ssq::Table& table) {

22 table.addFunc("loadAnimationTrackBatch", [vm = table.getHandle()](ssq::Array clips, ssq::Array paths,
23 AnimSkeleton* skeleton, int workers) {
24 ssq::Table result(vm);
25 try {
26 if (!skeleton || clips.size() != paths.size() || clips.size() > 64)
27 throw std::runtime_error("invalid animation batch arguments");
28 auto* fs = eve::ModuleManager::getInstance<eve::filesystem::Filesystem>("Filesystem");
29 if (!fs) throw std::runtime_error("filesystem unavailable");
30 std::vector<Owned<eve::filesystem::FileData>> buffers;
31 std::vector<AnimationTrackInput> inputs;
32 std::size_t total = 0;
33 for (std::size_t i = 0; i < clips.size(); ++i) {
34 auto* destination = clips.get<AnimClip*>(i);
35 if (!destination) throw std::runtime_error("missing animation batch destination");
36 auto* raw = fs->read(paths.get<std::string>(i));
37 if (!raw) throw std::runtime_error("animation track file unavailable");
38 Owned<eve::filesystem::FileData> data(raw);
39 if (data->getSize() > 256 * 1024 * 1024 - total)
40 throw std::runtime_error("animation batch exceeds 256 MiB limit");
41 total += data->getSize();
42 inputs.push_back({*destination, {static_cast<const std::byte*>(data->getData()), data->getSize()}});
43 buffers.push_back(std::move(data));
44 }
45 auto loaded = loadAnimationTrackBatch(inputs, *skeleton, workers);
46 result.set("ok", loaded.ok());
47 result.set("message", loaded.status().describe());
48 result.set("workers", loaded.ok() ? loaded.value() : 0);
49 } catch (const std::exception& error) {
50 result.set("ok", false);
51 result.set("message", std::string(error.what()));
52 result.set("workers", 0);
53 }
54 return result;
55 });
