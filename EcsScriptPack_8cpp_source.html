89std::vector<std::string> collectFieldNames(ssq::Object fieldsObj) {
90 std::vector<std::string> names;
91 HSQUIRRELVM vm = fieldsObj.getHandle();
92 if (!vm) return names;
93 HSQOBJECT arr = fieldsObj.getRaw();
94 if (arr._type != OT_ARRAY) return names;
95
96 const SQInteger top = sq_gettop(vm);
97 sq_pushobject(vm, arr);
98 const SQInteger len = sq_getsize(vm, -1);
99 names.reserve(size_t(std::max<SQInteger>(0, len)));
100 for (SQInteger i = 0; i < len; ++i) {
101 sq_pushinteger(vm, i);
102 if (SQ_FAILED(sq_get(vm, -2))) continue;
103 std::string name;
104 if (readStringField(vm, -1, name)) names.push_back(name);
105 sq_pop(vm, 1);
106 }
107 sq_settop(vm, top);
108 return names;
109}

121int packScriptEntityFloats(ssq::Object entitiesObj, const std::string &slot,
122 ssq::Object fieldsObj, GpuBuffer *buf) {
123 if (!buf) throw Exception("packScriptEntityFloats: buffer is null");
124 if (slot.empty()) throw Exception("packScriptEntityFloats: empty slot");
125
126 HSQUIRRELVM vm = entitiesObj.getHandle();
127 if (!vm) return 0;
128 HSQOBJECT ents = entitiesObj.getRaw();
129 if (ents._type != OT_ARRAY) return 0;
130
131 const std::vector<std::string> fields = collectFieldNames(fieldsObj);
132 if (fields.empty()) return 0;
133
134 const int n = entityArrayLen(vm, ents);
135 if (n <= 0) return 0;
136
137 const int floatsPer = int(fields.size());
138 std::vector<float> tmp(size_t(n) * size_t(floatsPer), 0.f);
139
140 const SQInteger top = sq_gettop(vm);
141 sq_pushobject(vm, ents);
142 for (int i = 0; i < n; ++i) {
143 sq_pushinteger(vm, i);
144 if (SQ_FAILED(sq_get(vm, -2))) {
145 sq_settop(vm, top);
146 throw Exception("packScriptEntityFloats: bad entity index %d", i);
147 }
148 if (sq_gettype(vm, -1) != OT_INSTANCE) {
149 sq_settop(vm, top);
150 throw Exception("packScriptEntityFloats: entity[%d] is not an instance", i);
151 }
152 HSQOBJECT entity{};
153 sq_getstackobj(vm, -1, &entity);
154
155 HSQOBJECT comp{};
156 sq_resetobject(&comp);
157 if (!getInstanceSlot(vm, entity, slot, comp) || comp._type != OT_INSTANCE) {
158 if (comp._type != OT_NULL) sq_release(vm, &comp);
159 sq_settop(vm, top);
160 throw Exception("packScriptEntityFloats: missing component slot '%s'", slot.c_str());
161 }
162
163 for (int f = 0; f < floatsPer; ++f) {
164 float v = 0.f;
165 if (!getNumberField(vm, comp, fields[size_t(f)], v)) {
166 sq_release(vm, &comp);
167 sq_settop(vm, top);
168 throw Exception("packScriptEntityFloats: missing field '%s.%s'", slot.c_str(),
169 fields[size_t(f)].c_str());
170 }
171 tmp[size_t(i) * size_t(floatsPer) + size_t(f)] = v;
172 }
173 sq_release(vm, &comp);
174 sq_pop(vm, 1); // entity
175 }
176 sq_settop(vm, top);
177
178 const int needBytes = n * floatsPer * int(sizeof(float));
179 if (buf->getSize() < needBytes)
180 throw Exception("packScriptEntityFloats: buffer too small (%d < %d)", buf->getSize(),
181 needBytes);
182 buf->writeFloat32s(tmp.data(), n * floatsPer, 0);
183 return n;
184}
