66std::vector<std::string> collectFieldNames(ssq::Object fieldsObj) {
67 std::vector<std::string> names;
68 HSQUIRRELVM vm = fieldsObj.getHandle();
69 if (!vm) return names;
70 HSQOBJECT arr = fieldsObj.getRaw();
71 if (arr._type != OT_ARRAY) return names;
72
73 const SQInteger top = sq_gettop(vm);
74 sq_pushobject(vm, arr);
75 const SQInteger len = sq_getsize(vm, -1);
76 names.reserve(size_t(std::max<SQInteger>(0, len)));
77 for (SQInteger i = 0; i < len; ++i) {
78 sq_pushinteger(vm, i);
79 if (SQ_FAILED(sq_get(vm, -2))) continue;
80 std::string name;
81 if (readStringField(vm, -1, name)) names.push_back(name);
82 sq_pop(vm, 1);
83 }
84 sq_settop(vm, top);
85 return names;
86}

103int packScriptEntityFloatsRange(ssq::Object entitiesObj, const std::string &slot, ssq::Object fieldsObj, GpuBuffer *buf,
104 int firstEntity, int entityCount) {
105 if (!buf) throw Exception("packScriptEntityFloats: buffer is null");
106 if (slot.empty()) throw Exception("packScriptEntityFloats: empty slot");
107
108 HSQUIRRELVM vm = entitiesObj.getHandle();
109 if (!vm) return 0;
110 HSQOBJECT ents = entitiesObj.getRaw();
111 if (ents._type != OT_ARRAY) return 0;
112
113 const std::vector<std::string> fields = collectFieldNames(fieldsObj);
114 if (fields.empty()) return 0;
115
116 const int total = entityArrayLen(vm, ents);
117 const int first = std::clamp(firstEntity, 0, total);
118 const int n = std::clamp(entityCount, 0, total - first);
119 if (n <= 0) return 0;
120
121 const int floatsPer = int(fields.size());
122 std::vector<float> tmp(size_t(n) * size_t(floatsPer), 0.f);
123
124 const SQInteger top = sq_gettop(vm);
125 sq_pushobject(vm, ents);
126 for (int i = 0; i < n; ++i) {
127 sq_pushinteger(vm, first + i);
128 if (SQ_FAILED(sq_get(vm, -2))) {
129 sq_settop(vm, top);
130 throw Exception("packScriptEntityFloats: bad entity index %d", first + i);
131 }
132 if (sq_gettype(vm, -1) != OT_INSTANCE) {
133 sq_settop(vm, top);
134 throw Exception("packScriptEntityFloats: entity[%d] is not an instance", first + i);
135 }
136 HSQOBJECT entity{};
137 sq_getstackobj(vm, -1, &entity);
138
139 HSQOBJECT comp{};
140 sq_resetobject(&comp);
141 if (!getInstanceSlot(vm, entity, slot, comp) || comp._type != OT_INSTANCE) {
142 if (comp._type != OT_NULL) sq_release(vm, &comp);
143 sq_settop(vm, top);
144 throw Exception("packScriptEntityFloats: missing component slot '%s'", slot.c_str());
145 }
146 // `getInstanceSlot` 已经 addref 过一次；把这一份引用交给 `ssq::Object`（它的析构一定会
147 // `sq_release`），下面就不再手工释放——否则就是双重释放（见 ECS.cpp §9.64.3 的坑）。
148 ssq::Object compObj(vm);
149 compObj.getRaw() = comp;
150
151 for (int f = 0; f < floatsPer; ++f) {
152 float v = 0.f;
153 if (!getNumberField(compObj, fields[size_t(f)], v)) {
154 sq_settop(vm, top);
155 throw Exception("packScriptEntityFloats: missing field '%s.%s'", slot.c_str(),
156 fields[size_t(f)].c_str());
157 }
158 tmp[size_t(i) * size_t(floatsPer) + size_t(f)] = v;
159 }
160 sq_pop(vm, 1); // entity
161 }
162 sq_settop(vm, top);
163
164 const int needBytes = (first + n) * floatsPer * int(sizeof(float));
165 if (buf->getSize() < needBytes)
166 throw Exception("packScriptEntityFloats: buffer too small (%d < %d)", buf->getSize(),
167 needBytes);
168 buf->writeFloat32s(tmp.data(), n * floatsPer, first * floatsPer);
169 return n;
170}

178 GpuBuffer *buf, int firstEntity, int entityCount) {
179 if (!buf) throw Exception("unpackScriptEntityFloats: buffer is null");
180 if (slot.empty()) throw Exception("unpackScriptEntityFloats: empty slot");
181 if (entityCount <= 0) return 0;
182
183 HSQUIRRELVM vm = entitiesObj.getHandle();
184 if (!vm) return 0;
185 HSQOBJECT ents = entitiesObj.getRaw();
186 if (ents._type != OT_ARRAY) return 0;
187
188 const std::vector<std::string> fields = collectFieldNames(fieldsObj);
189 if (fields.empty()) return 0;
190
191 const int nArr = entityArrayLen(vm, ents);
192 const int first = std::clamp(firstEntity, 0, nArr);
193 const int n = std::clamp(entityCount, 0, nArr - first);
194 if (n <= 0) return 0;
195
196 const int floatsPer = int(fields.size());
197 std::vector<float> tmp(size_t(n) * size_t(floatsPer), 0.f);
198 const int needBytes = (first + n) * floatsPer * int(sizeof(float));
199 if (buf->getSize() < needBytes)
200 throw Exception("unpackScriptEntityFloats: buffer too small (%d < %d)", buf->getSize(), needBytes);
201 buf->readFloat32s(tmp.data(), n * floatsPer, first * floatsPer);
202
203 const SQInteger top = sq_gettop(vm);
204 sq_pushobject(vm, ents);
205 for (int i = 0; i < n; ++i) {
206 sq_pushinteger(vm, first + i);
207 if (SQ_FAILED(sq_get(vm, -2))) {
208 sq_settop(vm, top);
209 throw Exception("unpackScriptEntityFloats: bad entity index %d", first + i);
210 }
211 HSQOBJECT entity{};
212 sq_getstackobj(vm, -1, &entity);
213
214 HSQOBJECT comp{};
215 sq_resetobject(&comp);
216 if (!getInstanceSlot(vm, entity, slot, comp) || comp._type != OT_INSTANCE) {
217 if (comp._type != OT_NULL) sq_release(vm, &comp);
218 sq_settop(vm, top);
219 throw Exception("unpackScriptEntityFloats: missing component slot '%s'", slot.c_str());
220 }
221 // 归属同 pack：addref 交给 `ssq::Object` 的析构释放，不再手工 `sq_release`。
222 ssq::Object compObj(vm);
223 compObj.getRaw() = comp;
224
225 for (int f = 0; f < floatsPer; ++f) {
226 const float v = tmp[size_t(i) * size_t(floatsPer) + size_t(f)];
227 if (!setNumberField(compObj, fields[size_t(f)], v)) {
228 sq_settop(vm, top);
229 throw Exception("unpackScriptEntityFloats: cannot set '%s.%s'", slot.c_str(),
230 fields[size_t(f)].c_str());
231 }
232 }
233 sq_pop(vm, 1);
234 }
235 sq_settop(vm, top);
236 return n;
