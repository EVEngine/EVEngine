50Result<void> BattleReplay::replay(Battle& battle, std::span<const BattleCommand> commands) {
51 if (commands.empty()) return Result<void>::success(Status::success(StatusCode::NoOp));
52 std::uint64_t previousSequence = 0;
53 for (std::size_t index = 0; index < commands.size(); ++index) {
54 const BattleCommand& command = commands[index];
55 const std::string path = "commands[" + std::to_string(index) + "]";
56 if (command.sequence == 0 || (index > 0 && command.sequence != previousSequence + 1))
57 return fail(DiagnosticCode::Conflict, "tactics replay command sequence is not contiguous", path);
58 if (battle.commands()->nextSequence != command.sequence)
59 return fail(DiagnosticCode::Conflict, "tactics replay command sequence differs from target", path);
60 if (battle.turn()->revision != command.expectedRevision)
61 return fail(DiagnosticCode::Conflict, "tactics replay expected revision differs from target", path);
62 auto applied = apply(battle, command);
63 if (!applied) return Result<void>::failure(applied.status());
64 if (battle.turn()->revision != command.resultingRevision)
65 return fail(DiagnosticCode::InvariantViolation, "tactics replay produced a different revision", path);
66 const auto& recorded = battle.commands()->values.back();
67 if (recorded.kind != command.kind || recorded.sequence != command.sequence ||
68 recorded.expectedRevision != command.expectedRevision ||
69 recorded.resultingRevision != command.resultingRevision)
70 return fail(DiagnosticCode::InvariantViolation, "tactics replay recorded a different command", path);
71 previousSequence = command.sequence;
72 }
73 return Result<void>::success(Status::success(StatusCode::Applied));
74}
