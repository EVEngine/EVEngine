73 Result<Value> registerFighter(const std::string& subjectText, const std::string& ownerId,
74 double x, double z, double health, double poise,
75 double maximumSpeed, double acceleration) {
76 auto subject = parseSubject(subjectText, "subject");
77 if (!subject) return Result<Value>::failure(subject.status());
78 CombatState combatState{subject.value(), health, health, poise, poise};
79 auto valid = combatState.validate();
80 if (!valid) return Result<Value>::failure(valid.status());
81 if (states_.contains(subjectText))
82 return failure<Value>(DiagnosticCode::AlreadyExists, "combat fighter already exists", subjectText);
83 auto registered = locomotion_.registerSubject(
84 {subject.value(), ownerId, {x, z}, maximumSpeed, acceleration});
85 if (!registered) return Result<Value>::failure(registered.status());
86 states_.emplace(subjectText, std::move(combatState));
87 return state(subjectText);
88 }

116 Result<Value> advance(std::int64_t tick, double seconds) {
117 if (tick < 0)
118 return failure<Value>(DiagnosticCode::InvalidArgument, "combat tick must be non-negative", "tick");
119 auto delta = Duration::fromSeconds(seconds);
120 if (!delta) return Result<Value>::failure(delta.status());
121 auto advanced = locomotion_.advance(
122 {SimulationTick(static_cast<std::uint64_t>(tick)), std::move(delta).takeValue()});
123 if (!advanced) return Result<Value>::failure(advanced.status());
124 Value::Array events;
125 events.reserve(advanced.value().events.size());
126 for (const auto& event : advanced.value().events) {
127 Value::Object value;
128 value["subject"] = event.subject.format();
129 value["kind"] = event.kind == CombatLocomotionEventKind::Arrived ? "arrived" : "moved";
130 value["position"] = vectorValue(event.position);
131 events.emplace_back(std::move(value));
132 }
133 Value::Object result;
134 result["tick"] = tick;
135 result["events"] = std::move(events);
136 return Result<Value>::success(Value(std::move(result)), Status::success(StatusCode::Applied));

139 Result<Value> applyDamage(const std::string& sourceText, const std::string& targetText,
140 const std::string& damageType, double healthDamage,
141 double poiseDamage, double knockbackX, double knockbackY,
142 double knockbackZ) {
143 auto source = parseSubject(sourceText, "source");
144 if (!source) return Result<Value>::failure(source.status());
145 auto target = parseSubject(targetText, "target");
146 if (!target) return Result<Value>::failure(target.status());
147 const auto found = states_.find(targetText);
148 if (found == states_.end())
149 return failure<Value>(DiagnosticCode::NotFound, "combat target was not found", targetText);
150 DamageRequest request;
151 request.source = source.value();
152 request.target = target.value();
153 request.damageType = damageType;
154 request.healthDamage = healthDamage;
155 request.poiseDamage = poiseDamage;
156 request.knockback = {knockbackX, knockbackY, knockbackZ};
157 auto outcome = damage_.apply(found->second, request);
158 if (!outcome) return Result<Value>::failure(outcome.status());
159 Value::Object result;
160 result["source"] = outcome.value().source.format();
161 result["target"] = outcome.value().target.format();
162 result["health"] = outcome.value().health;
163 result["poise"] = outcome.value().poise;
164 result["appliedHealthDamage"] = outcome.value().appliedHealthDamage;
165 result["appliedPoiseDamage"] = outcome.value().appliedPoiseDamage;
166 result["reaction"] = reactionName(outcome.value().reaction);
167 return Result<Value>::success(Value(std::move(result)), Status::success(StatusCode::Applied));
168 }

217 Result<Value> setTimelineAbility(const std::string& definitionText,
218 const std::string& timelineJson, double cooldownSeconds,
219 const std::string& instancingText,
220 const std::string& groupText,
221 const std::string& triggerTag, bool replace) {
222 auto definitionId = LogicalId::parse(definitionText);
223 if (!definitionId)
224 return failure<Value>(DiagnosticCode::InvalidArgument, "ability definition id is invalid",
225 "definitionId");
226 auto decoded = Value::fromJson(timelineJson);
227 if (!decoded) return Result<Value>::failure(decoded.status());
228 auto timeline = action::ActionTimeline::fromValue(decoded.value());
229 if (!timeline) return Result<Value>::failure(timeline.status());
230 auto cooldown = Duration::fromSeconds(cooldownSeconds);
231 if (!cooldown) return Result<Value>::failure(cooldown.status());
232
233 action::AbilityDefinition definition;
234 definition.id = *definitionId;
235 definition.action.id = timeline.value().actionId;
236 definition.action.timeline = timeline.value();
237 definition.cooldown = cooldown.value();
238 if (instancingText == "non-instanced")
239 definition.instancing = action::AbilityInstancingPolicy::NonInstanced;
240 else if (instancingText == "per-owner")
241 definition.instancing = action::AbilityInstancingPolicy::PerOwner;
242 else if (instancingText == "per-execution")
243 definition.instancing = action::AbilityInstancingPolicy::PerExecution;
244 else
245 return failure<Value>(DiagnosticCode::InvalidArgument, "ability instancing policy is invalid",
246 "instancing");
247 if (groupText == "independent")
248 definition.activationGroup = action::AbilityActivationGroup::Independent;
249 else if (groupText == "exclusive-replaceable")
250 definition.activationGroup = action::AbilityActivationGroup::ExclusiveReplaceable;
251 else if (groupText == "exclusive-blocking")
252 definition.activationGroup = action::AbilityActivationGroup::ExclusiveBlocking;
253 else
254 return failure<Value>(DiagnosticCode::InvalidArgument, "ability activation group is invalid",
255 "activationGroup");
256 if (!triggerTag.empty()) {
257 if (!tags::isValidGameplayTagName(triggerTag))
258 return failure<Value>(DiagnosticCode::InvalidArgument, "ability trigger tag is invalid",
259 "triggerTag");
260 definition.triggers.push_back({triggerTag, tags::GameplayTagMatch::Exact});
261 }
262
263 const auto& splits = timeline.value().splitTimestamps;
264 if (splits.empty()) {
265 definition.action.timing.active = timeline.value().duration;
266 } else if (splits.size() == 1) {
267 definition.action.timing.windup = splits[0];
268 definition.action.timing.active = Duration::fromNanoseconds(
269 timeline.value().duration.nanoseconds() - splits[0].nanoseconds());
270 } else {
271 definition.action.timing.windup = splits[0];
272 definition.action.timing.active = Duration::fromNanoseconds(
273 splits[1].nanoseconds() - splits[0].nanoseconds());
274 definition.action.timing.recover = Duration::fromNanoseconds(
275 timeline.value().duration.nanoseconds() - splits[1].nanoseconds());
276 }
277 auto changed = replace ? abilities_.replaceDefinition(definition)
278 : abilities_.registerDefinition(definition);
279 if (!changed) return Result<Value>::failure(changed.status());
280 abilityActions_[definitionText] = definition.action.id;
281 Value::Object result;
282 result["definitionId"] = definitionText;
283 result["actionId"] = definition.action.id.format();
284 result["durationSeconds"] = timeline.value().duration.seconds();
285 result["sectionCount"] = static_cast<std::int64_t>(timeline.value().sectionCount());
286 return Result<Value>::success(Value(std::move(result)), Status::success(StatusCode::Applied));
287 }

289 Result<Value> activateAbility(std::int64_t grantValue, std::int64_t tickValue) {
290 auto grant = parseGrant(grantValue);
291 if (!grant) return Result<Value>::failure(grant.status());
292 if (tickValue < 0)
293 return failure<Value>(DiagnosticCode::InvalidArgument, "ability tick must be non-negative", "tick");
294 auto grantState = abilities_.findGrant(grant.value());
295 if (!grantState) return Result<Value>::failure(grantState.status());
296 const auto action = abilityActions_.find(grantState.value().definitionId.format());
297 if (action == abilityActions_.end())
298 return failure<Value>(DiagnosticCode::NotFound, "ability definition action is unavailable",
299 "definitionId");
300 action::ActionRequest request;
301 request.actionId = action->second;
302 request.requestedTick = SimulationTick(static_cast<std::uint64_t>(tickValue));
303 const auto tick = request.requestedTick;
304 auto activated = abilities_.activate(grant.value(), std::move(request), tick);
305 if (!activated) return Result<Value>::failure(activated.status());
306 return Result<Value>::success(abilityActivationValue(activated.value()),
307 Status::success(StatusCode::Applied));
308 }

310 Result<Value> advanceAbilities(std::int64_t tickValue, double seconds) {
311 if (tickValue < 0)
312 return failure<Value>(DiagnosticCode::InvalidArgument, "ability tick must be non-negative", "tick");
313 auto delta = Duration::fromSeconds(seconds);
314 if (!delta) return Result<Value>::failure(delta.status());
315 Value::Array advances;
316 for (const auto& activation : abilities_.activeActivations()) {
317 auto advanced = actions_.advance(activation.executionId,
318 SimulationTick(static_cast<std::uint64_t>(tickValue)), delta.value());
319 if (!advanced) return Result<Value>::failure(advanced.status());
320 Value::Object value;
321 value["activationId"] = static_cast<std::int64_t>(activation.id.value());
322 value["executionId"] = static_cast<std::int64_t>(activation.executionId.value());
323 value["phase"] = std::string(action::actionPhaseName(advanced.value().phase));
324 value["elapsedSeconds"] = advanced.value().totalElapsed.seconds();
325 advances.emplace_back(std::move(value));
326 }
327 auto cooled = abilities_.advanceCooldowns(delta.value());
328 if (!cooled) return Result<Value>::failure(cooled.status());
329 auto synchronized = abilities_.synchronize();
330 if (!synchronized) return Result<Value>::failure(synchronized.status());
331 Value::Object result;
332 result["advances"] = std::move(advances);
333 result["completedCount"] = static_cast<std::int64_t>(synchronized.value());
334 result["activeCount"] = static_cast<std::int64_t>(abilities_.activeActivations().size());
335 return Result<Value>::success(Value(std::move(result)), Status::success(StatusCode::Applied));
336 }

338 Result<Value> cancelAbility(std::int64_t activationValue, std::int64_t tickValue) {
339 if (activationValue <= 0 || tickValue < 0)
340 return failure<Value>(DiagnosticCode::InvalidArgument, "ability activation or tick is invalid",
341 "activation");
342 const action::AbilityActivationId activationId(static_cast<std::uint64_t>(activationValue));
343 const auto active = abilities_.activeActivations();
344 const auto found = std::find_if(active.begin(), active.end(),
345 [&](const auto& item) { return item.id == activationId; });
346 if (found == active.end())
347 return failure<Value>(DiagnosticCode::NotFound, "ability activation was not found", "activationId");
348 auto cancelled = actions_.cancel(found->executionId,
349 SimulationTick(static_cast<std::uint64_t>(tickValue)));
350 if (!cancelled) return Result<Value>::failure(cancelled.status());
351 auto synchronized = abilities_.synchronize();
352 if (!synchronized) return Result<Value>::failure(synchronized.status());
353 Value::Object result;
354 result["activationId"] = activationValue;
355 result["phase"] = std::string("cancelled");
356 return Result<Value>::success(Value(std::move(result)), Status::success(StatusCode::Applied));
357 }
