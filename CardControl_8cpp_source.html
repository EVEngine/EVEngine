166Value CardControl::handState(const Entry& entry) const {
167 Value::Array hand;
168 if (entry.hand != nullptr) {
169 const int count = entry.hand->count();
170 for (int index = 0; index < count; ++index) {
171 CardData* card = entry.hand->get(index);
172 if (card == nullptr) continue;
173 hand.emplace_back(Value(Value::Object{
174 {"card", Value(card->identity()->id)},
175 {"definition", Value(card->identity()->definitionId)},
176 {"name", Value(card->identity()->name)},
177 {"kind", Value(card->identity()->kind)},
178 {"cost", Value(static_cast<std::int64_t>(card->stats()->cost))},
179 {"attack", Value(static_cast<std::int64_t>(card->stats()->attack))},
180 {"health", Value(static_cast<std::int64_t>(card->stats()->health))},
181 {"state",
182 Value(cardStateName(card->state()->phase) != nullptr ? std::string(cardStateName(card->state()->phase))
183 : std::string{})},
184 {"disabled", Value(card->visual()->disabled)},
185 }));
186 }
187 }
188 std::int64_t deckSize = 0;
189 Deck* deck = module_ != nullptr ? module_->getDeck() : nullptr;
190 if (deck != nullptr) deckSize = static_cast<std::int64_t>(deck->count());
191 return Value(Value::Object{
192 {"owner", Value(entry.hand != nullptr ? entry.hand->meta()->owner : std::string{})},
193 {"hand", Value(std::move(hand))},
194 {"handSize", Value(static_cast<std::int64_t>(entry.hand != nullptr ? entry.hand->count() : 0))},
195 {"deck", Value(deckSize)},
196 // `unbound` means this publication carries no payment account, so
197 // `card:play` is deliberately unavailable rather than silently absent.
198 {"payment", Value(entry.account != nullptr ? std::string("bound") : std::string("unbound"))},
199 });
200}

202Result<GameplayObservation> CardControl::observeGameplay(const GameplaySession& session, SubjectRef instance) const {
203 const Entry* entry = find(instance);
204 if (entry == nullptr)
205 return failure<GameplayObservation>(DiagnosticCode::NotFound, "card gameplay instance was not found",
206 "instance");
207 if (!controls(session, *entry))
208 return failure<GameplayObservation>(DiagnosticCode::PreconditionViolation,
209 "session does not control the card owner", "instance");
210 GameplayObservation observation;
211 observation.domain = id("gameplay:card");
212 observation.instance = entry->instance;
213 observation.tick = entry->tick;
214 observation.revision = entry->revision;
215 observation.state = handState(*entry);
216 return Result<GameplayObservation>::success(std::move(observation));

219Result<std::vector<GameplayActionDescriptor>> CardControl::availableGameplayActions(const GameplaySession& session,
220 SubjectRef instance,
221 SubjectRef subject) const {
222 auto observed = observeGameplay(session, instance);
223 if (!observed) return Result<std::vector<GameplayActionDescriptor>>::failure(observed.status());
224 const Entry* entry = find(instance);
225 if (entry == nullptr)
226 return failure<std::vector<GameplayActionDescriptor>>(DiagnosticCode::NotFound,
227 "card gameplay instance was not found", "instance");
228 if (subject != entry->owner)
229 return failure<std::vector<GameplayActionDescriptor>>(DiagnosticCode::PreconditionViolation,
230 "card action subject must be its owner", "subject");
231
232 const Value stringType(Value::Object{{"type", Value("string")}});
233 const Value numberType(Value::Object{{"type", Value("number")}});
234 std::vector<GameplayActionDescriptor> actions{
235 {id(kDraw), Value(Value::Object{})},
236 // Cost-free cards play through the unbound payment boundary; a charging
237 // card fails there with an explicit "no payment account bound" diagnostic,
238 // so the gap is reported instead of the action vanishing from discovery.
239 {id(kPlay), Value(Value::Object{{"card", stringType}})},
240 };
241 if (session.access != GameplayAccess::PlayerEquivalent) {
242 actions.push_back(
243 {id(kSetAttribute),
244 Value(Value::Object{{"card", stringType}, {"attribute", stringType}, {"value", numberType}})});
245 }
246 return Result<std::vector<GameplayActionDescriptor>>::success(std::move(actions));
247}

249Result<GameplayCommandReceipt> CardControl::submitGameplay(const GameplaySession& session, SubjectRef instance,
250 const GameplayCommand& command) {
251 Entry* entry = find(instance);
252 if (entry == nullptr)
253 return failure<GameplayCommandReceipt>(DiagnosticCode::NotFound, "card gameplay instance was not found",
254 "instance");
255 if (!controls(session, *entry) || command.subject != entry->owner)
256 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation,
257 "session does not control the card owner", "command.subject");
258 if (command.id.empty())
259 return failure<GameplayCommandReceipt>(DiagnosticCode::InvalidArgument, "command id must not be empty",
260 "command.id");
261 if (command.observedTick != entry->tick || command.expectedRevision != entry->revision)
262 return failure<GameplayCommandReceipt>(
263 DiagnosticCode::Conflict, "card command was based on a stale observation", "command.expectedRevision");
264 auto values = object(command.parameters);
265 if (!values) return Result<GameplayCommandReceipt>::failure(values.status());
266 const Value::Object& parameters = *values.value();
267 if (entry->hand == nullptr || module_ == nullptr)
268 return failure<GameplayCommandReceipt>(DiagnosticCode::Failed, "card instance lost its hand", "instance");
269
270 const std::string action = command.action.format();
271 std::string detail;
272 std::string eventType;
273 std::int64_t applied = 0;
274 if (action == kDraw) {
275 CardData* drawn = module_->drawCard(entry->hand->meta()->owner);
276 if (drawn == nullptr)
277 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation,
278 "the deck has no card to draw", "command.action");
279 detail = "drew " + drawn->identity()->definitionId;
280 eventType = "card.drawn";
281 applied = 1;
282 } else if (action == kPlay) {
283 auto cardId = text(parameters, "card");
284 if (!cardId) return Result<GameplayCommandReceipt>::failure(cardId.status());
285 CardData* card = cardId.value().empty() ? nullptr : entry->hand->find(cardId.value());
286 if (card == nullptr)
287 return failure<GameplayCommandReceipt>(DiagnosticCode::NotFound, "that card is not in the hand",
288 "command.parameters.card");
289 eve::resource::IResourceAccount& account = entry->account != nullptr ? *entry->account : unboundAccount();
290 const auto played = module_->play(*card, account, {}, command.id);
291 if (!played) return Result<GameplayCommandReceipt>::failure(played.status());
292 detail = "played " + card->identity()->definitionId;
293 eventType = "card.played";
294 applied = 1;
295 } else if (action == kSetAttribute) {
296 if (session.access == GameplayAccess::PlayerEquivalent)
297 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation,
298 "rewriting card attributes requires the test-driver or "
299 "developer-cheat profile",
300 "session.access");
301 auto cardId = text(parameters, "card");
302 auto attribute = text(parameters, "attribute");
303 auto value = number(parameters, "value", 0.0);
304 if (!cardId) return Result<GameplayCommandReceipt>::failure(cardId.status());
305 if (!attribute) return Result<GameplayCommandReceipt>::failure(attribute.status());
306 if (!value) return Result<GameplayCommandReceipt>::failure(value.status());
307 if (cardId.value().empty() || attribute.value().empty())
308 return failure<GameplayCommandReceipt>(DiagnosticCode::InvalidArgument,
309 "set-attribute needs a card and an attribute", "command.parameters");
310 CardData* card = entry->hand->find(cardId.value());
311 if (card == nullptr)
312 return failure<GameplayCommandReceipt>(DiagnosticCode::NotFound, "that card is not in the hand",
313 "command.parameters.card");
314 const auto appliedAttribute = module_->setCardAttribute(*card, attribute.value(), value.value());
315 if (!appliedAttribute) return Result<GameplayCommandReceipt>::failure(appliedAttribute.status());
316 detail = "set " + attribute.value() + " of " + card->identity()->definitionId;
317 eventType = "card.attribute-changed";
318 applied = 1;
319 } else {
320 return failure<GameplayCommandReceipt>(DiagnosticCode::Unsupported, "unsupported card action",
321 "command.action");
322 }
323
324 ++entry->revision;
325 GameplayEvent event;
326 event.sequence = entry->nextEventSequence++;
327 event.tick = entry->tick;
328 event.type = eventType;
329 event.subject = entry->owner;
330 event.causationCommandId = command.id;
331 event.correlationId = command.id;
332 event.payload = Value(Value::Object{
333 {"instance", Value(entry->instance.format())}, {"detail", Value(detail)}, {"quantity", Value(applied)}});
334 entry->events.push_back(std::move(event));
335
336 GameplayCommandReceipt receipt;
337 receipt.commandId = command.id;
338 receipt.executionId = command.id + ".executed";
339 receipt.acceptedTick = entry->tick;
340 receipt.resultingRevision = entry->revision;
341 receipt.details =
342 Value(Value::Object{{"action", Value(action)}, {"detail", Value(detail)}, {"quantity", Value(applied)}});
343 return Result<GameplayCommandReceipt>::success(std::move(receipt), Status::success(StatusCode::Applied));
344}

361Result<std::vector<GameplayEvent>> CardControl::gameplayEvents(const GameplaySession& session, SubjectRef instance,
362 std::uint64_t afterSequence) const {
363 auto observed = observeGameplay(session, instance);
364 if (!observed) return Result<std::vector<GameplayEvent>>::failure(observed.status());
365 const Entry* entry = find(instance);
366 if (entry == nullptr)
367 return failure<std::vector<GameplayEvent>>(DiagnosticCode::NotFound, "card gameplay instance was not found",

369 std::vector<GameplayEvent> result;
370 for (const auto& event : entry->events)

374 Status::success(empty ? StatusCode::NoOp : StatusCode::Applied));
