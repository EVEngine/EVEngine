53std::string quote(const std::string& value) {
54 std::ostringstream out;
55 out << '"';
56 for (unsigned char c : value) {
57 switch (c) {
58 case '"': out << "\\\""; break;
59 case '\\': out << "\\\\"; break;
60 case '\n': out << "\\n"; break;
61 case '\r': out << "\\r"; break;
62 case '\t': out << "\\t"; break;
63 default:
64 if (c < 0x20)
65 out << "\\u" << std::hex << std::setw(4) << std::setfill('0') << int(c) << std::dec;
66 else
67 out << static_cast<char>(c);
68 }
69 }
70 return out.str() + '"';
71}

84bool parseRuleId(const std::string& id, uint64_t& output) {
85 constexpr const char* prefix = "authority-";
86 if (!id.starts_with(prefix)) return false;
87 const std::string suffix = id.substr(std::char_traits<char>::length(prefix));
88 if (suffix.size() != 16) return false;
89 try {
90 size_t pos = 0;
91 output = std::stoull(suffix, &pos);
92 return pos == suffix.size() && output != 0;
93 } catch (...) {
94 return false;
95 }

98bool wins(const Rule& candidate, const Rule& current) {
99 if (candidate.priority != current.priority) return candidate.priority > current.priority;
100 if (candidate.effect != current.effect) return candidate.effect == RuleEffect::Deny;
101 return candidate.order < current.order;

154std::string Store::add(RuleEffect effect, const std::string& actor, const std::string& scope,
155 const std::string& capability, const std::string& source, int priority, double duration) {
156 if (actor.empty() || scope.empty() || capability.empty() || !std::isfinite(duration)) return {};
157 std::ostringstream id;
158 id << "authority-" << std::setw(16) << std::setfill('0') << nextId_++;
159 Rule rule;
160 rule.id = id.str();
161 rule.actor = actor;
162 rule.scope = scope;
163 rule.capability = capability;
164 rule.source = source;
165 rule.effect = effect;
166 rule.priority = priority;
167 rule.duration = std::max(0.0, duration);
168 rule.remaining = duration > 0.0 ? duration : -1.0;
169 rule.order = nextOrder_++;
170 rules_.push_back(std::move(rule));
171 emit(effect == RuleEffect::Grant ? EventKind::Granted : EventKind::Denied, rules_.back());
172 advanceRevision(revision_);
173 return rules_.back().id;
174}

215void Store::update(double dtSeconds) {
216 if (dtSeconds <= 0.0 || !std::isfinite(dtSeconds)) return;
217 bool changed = false;
218 for (auto it = rules_.begin(); it != rules_.end();) {
219 if (it->remaining < 0.0) {
220 ++it;
221 continue;
222 }
223 it->remaining = std::max(0.0, it->remaining - dtSeconds);
224 changed = true;
225 if (it->remaining == 0.0) {
226 emit(EventKind::Expired, *it, "duration_elapsed");
227 it = rules_.erase(it);
228 } else {
229 ++it;
230 }
231 }
232 query_.clear();
233 if (changed) advanceRevision(revision_);
234}

236Decision Store::explain(const std::string& actor, const std::string& scope, const std::string& capability) const {
237 const Rule* winner = nullptr;
238 for (const auto& rule : rules_)
239 if (rule.actor == actor && rule.scope == scope && rule.capability == capability &&
240 (!winner || wins(rule, *winner)))
241 winner = &rule;
242 if (!winner) return {false, "no_matching_rule", {}, {}, 0};
243 return {winner->effect == RuleEffect::Grant, winner->effect == RuleEffect::Grant ? "granted" : "denied", winner->id,
244 winner->source, winner->priority};
245}

279std::string Store::snapshotJson() const {
280 std::ostringstream out;
281 out << "{\"version\":1,\"nextId\":" << quote(std::to_string(nextId_))
282 << ",\"nextOrder\":" << quote(std::to_string(nextOrder_))

292 << ",\"order\":" << quote(std::to_string(rule.order)) << '}';

297eve::Result<void> Store::restoreJson(const std::string& json) {
298 std::string parseError;
299 auto document = eve::json::Document::parse(json, &parseError);
300 const auto root = document.root();
301 uint64_t nextId = 0, nextOrder = 0, nextSequence = 0;
302 if (!document.valid() || !root.isObject() || root.getInt("version") != 1 || !parseU64(root.get("nextId"), nextId) ||
303 !parseU64(root.get("nextOrder"), nextOrder) || !parseU64(root.get("nextSequence"), nextSequence) ||
304 nextId == 0 || nextOrder == 0 || nextSequence == 0 || !root.get("rules").isArray()) {
305 return eve::Result<void>::failure(
306 eve::Diagnostic::error(eve::DiagnosticCode::ParseError,
307 parseError.empty() ? "invalid authority snapshot" : parseError, "authority.store"));
308 }
309 std::deque<Rule> restored;
310 const auto values = root.get("rules");
311 for (size_t i = 0; i < values.size(); ++i) {
312 const auto item = values.at(i);
313 Rule rule;
314 uint64_t order = 0, numericId = 0;
315 const auto effect = item.getString("effect");
316 if (!item.isObject() || item.getString("id").empty() || item.getString("actor").empty() ||
317 item.getString("scope").empty() || item.getString("capability").empty() ||
318 (effect != "grant" && effect != "deny") || !parseU64(item.get("order"), order) || order == 0 ||
319 order >= nextOrder || !parseRuleId(item.getString("id"), numericId) || numericId >= nextId ||
320 !item.get("priority").isNumber() || !item.get("duration").isNumber() || !item.get("remaining").isNumber()) {
321 return eve::Result<void>::failure(
322 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "invalid rule at index " + std::to_string(i),
323 "authority.store.rules[" + std::to_string(i) + "]"));
324 }
325 rule.id = item.getString("id");
326 rule.actor = item.getString("actor");
327 rule.scope = item.getString("scope");
328 rule.capability = item.getString("capability");
329 rule.source = item.getString("source");
330 rule.effect = effect == "grant" ? RuleEffect::Grant : RuleEffect::Deny;
331 rule.priority = item.getInt("priority");
332 rule.duration = item.getDouble("duration");
333 rule.remaining = item.getDouble("remaining");
334 rule.order = order;
335 if (!std::isfinite(rule.duration) || !std::isfinite(rule.remaining) || rule.duration < 0.0 ||
336 rule.remaining == 0.0 || rule.remaining < -1.0 ||
337 std::any_of(
338 restored.begin(), restored.end(),
339 [&](const Rule& old) { return old.id == rule.id || old.order == rule.order; })) {
340 return eve::Result<void>::failure(eve::Diagnostic::error(
341 eve::DiagnosticCode::InvariantViolation, "invalid rule state at index " + std::to_string(i),
342 "authority.store.rules[" + std::to_string(i) + "]"));
343 }
344 restored.push_back(std::move(rule));
345 }
346 rules_ = std::move(restored);
347 nextId_ = nextId;
348 nextOrder_ = nextOrder;
349 nextSequence_ = nextSequence;
350 events_.clear();
351 query_.clear();
352 revision_ = eve::Revision(nextSequence > 0 ? nextSequence - 1 : 0);
353 tick_ = eve::SimulationTick{};
354 return eve::Result<void>::success();
355}

364eve::Result<void> Store::restoreSnapshot(const eve::SnapshotEnvelope& source,
365 const eve::SnapshotHashProvider& hashProvider) {
366 if (source.type != "authority.store" || source.schema != authoritySchema())
367 return snapshotFailure<void>(eve::DiagnosticCode::InvalidArgument,
368 "snapshot does not belong to authority::Store");
369 if (!instanceId_.isNil() && source.instanceId != instanceId_)
370 return snapshotFailure<void>(eve::DiagnosticCode::Conflict,
371 "snapshot instanceId does not match authority::Store");
372
373 auto migrated = authorityMigrations().migrate(source, eve::SchemaVersion(1), hashProvider);
374 if (!migrated.ok()) return eve::Result<void>::failure(migrated.status());
375 const auto& candidate = migrated.value();
376 auto metadata = eve::validateSnapshotPayloadMetadata(candidate.payload, candidate.revision, candidate.tick);
377 if (!metadata.ok()) return eve::Result<void>::failure(metadata.status());
378 auto payload = candidate.payload.toJson();
379 if (!payload.ok()) return eve::Result<void>::failure(payload.status());
380 const std::string payloadJson = std::move(payload).takeValue();
381
382 // restoreJson parses into local containers before replacing rules/counters;
383 // it is therefore the domain candidate-builder and has no partial writes.
384 auto restored = restoreJson(payloadJson);
385 if (!restored.ok()) return eve::Result<void>::failure(restored.status());
386
387 instanceId_ = candidate.instanceId;
388 revision_ = candidate.revision;
389 tick_ = candidate.tick;
390 return eve::Result<void>::success();
391}
