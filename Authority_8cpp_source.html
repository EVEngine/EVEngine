47std::string quote(const std::string& value) {
48 std::ostringstream out;
49 out << '"';
50 for (unsigned char c : value) {
51 switch (c) {
52 case '"': out << "\\\""; break;
53 case '\\': out << "\\\\"; break;
54 case '\n': out << "\\n"; break;
55 case '\r': out << "\\r"; break;
56 case '\t': out << "\\t"; break;
57 default:
58 if (c < 0x20)
59 out << "\\u" << std::hex << std::setw(4) << std::setfill('0') << int(c) << std::dec;
60 else
61 out << static_cast<char>(c);
62 }
63 }
64 return out.str() + '"';
65}

78bool parseRuleId(const std::string& id, uint64_t& output) {
79 constexpr const char* prefix = "authority-";
80 if (!id.starts_with(prefix)) return false;
81 const std::string suffix = id.substr(std::char_traits<char>::length(prefix));
82 if (suffix.size() != 16) return false;
83 try {
84 size_t pos = 0;
85 output = std::stoull(suffix, &pos);
86 return pos == suffix.size() && output != 0;
87 } catch (...) {
88 return false;
89 }

182void Store::indexInsert(const Rule& rule) {
183 if (hasWildcard(rule)) {
184 wildcardIds_.push_back(rule.id);
185 return;
186 }
187 exactIndex_[exactKey(rule.actor, rule.scope, rule.capability)].push_back(rule.id);

190void Store::indexErase(const Rule& rule) {
191 if (hasWildcard(rule)) {
192 wildcardIds_.erase(std::remove(wildcardIds_.begin(), wildcardIds_.end(), rule.id), wildcardIds_.end());
193 return;
194 }
195 const auto key = exactKey(rule.actor, rule.scope, rule.capability);
196 const auto it = exactIndex_.find(key);
197 if (it == exactIndex_.end()) return;
198 auto& ids = it->second;
199 ids.erase(std::remove(ids.begin(), ids.end(), rule.id), ids.end());
200 if (ids.empty()) exactIndex_.erase(it);
201}

219eve::Result<std::string> Store::add(RuleEffect effect, const std::string& actor, const std::string& scope,
220 const std::string& capability, const std::string& source, int priority,
221 double duration) {
222 if (actor.empty() || scope.empty() || capability.empty())
223 return rejectedRule("authority actor, scope, and capability must be non-empty");
224 if (!std::isfinite(duration)) return rejectedRule("authority rule duration must be finite");
225 std::ostringstream id;
226 id << "authority-" << std::setw(16) << std::setfill('0') << nextId_++;
227 Rule rule;
228 rule.id = id.str();
229 rule.actor = actor;
230 rule.scope = scope;
231 rule.capability = capability;
232 rule.source = source;
233 rule.effect = effect;
234 rule.priority = priority;
235 rule.duration = std::max(0.0, duration);
236 rule.remaining = duration > 0.0 ? duration : -1.0;
237 rule.order = nextOrder_++;
238 rules_.push_back(std::move(rule));
239 indexInsert(rules_.back());
240 emit(effect == RuleEffect::Grant ? EventKind::Granted : EventKind::Denied, rules_.back());
241 advanceRevision(revision_);
242 query_.clear();
243 return eve::Result<std::string>::success(rules_.back().id, eve::Status::success(eve::StatusCode::Applied));
244}

266int Store::revokeMatching(const std::string& reason, const std::function<bool(const Rule&)>& predicate) {
267 int removed = 0;
268 for (auto it = rules_.begin(); it != rules_.end();) {
269 if (predicate(*it)) {
270 emit(EventKind::Revoked, *it, reason);
271 indexErase(*it);
272 it = rules_.erase(it);
273 ++removed;
274 } else {
275 ++it;
276 }
277 }
278 query_.clear();
279 if (removed > 0) advanceRevision(revision_);
280 return removed;
281}

299void Store::update(double dtSeconds) {
300 if (dtSeconds <= 0.0 || !std::isfinite(dtSeconds)) return;
301 bool changed = false;
302 for (auto it = rules_.begin(); it != rules_.end();) {
303 if (it->remaining < 0.0) {
304 ++it;
305 continue;
306 }
307 it->remaining = std::max(0.0, it->remaining - dtSeconds);
308 changed = true;
309 if (it->remaining == 0.0) {
310 emit(EventKind::Expired, *it, "duration_elapsed");
311 indexErase(*it);
312 it = rules_.erase(it);
313 } else {
314 ++it;
315 }
316 }
317 query_.clear();
318 if (changed) advanceRevision(revision_);
319}

321Decision Store::explain(const std::string& actor, const std::string& scope, const std::string& capability) const {
322 const Rule* winner = nullptr;
323 std::vector<std::string> candidateIds;
324 if (const auto it = exactIndex_.find(exactKey(actor, scope, capability)); it != exactIndex_.end())
325 candidateIds.insert(candidateIds.end(), it->second.begin(), it->second.end());
326 candidateIds.insert(candidateIds.end(), wildcardIds_.begin(), wildcardIds_.end());
327 for (const auto& id : candidateIds) {
328 const Rule* rule = find(id);
329 if (!rule || !ruleMatches(*rule, actor, scope, capability)) continue;
330 if (!winner || wins(*rule, *winner)) winner = rule;
331 }
332 if (!winner) return {false, "no_matching_rule", {}, {}, 0};
333 return {winner->effect == RuleEffect::Grant, winner->effect == RuleEffect::Grant ? "granted" : "denied", winner->id,
334 winner->source, winner->priority};
335}

401eve::Result<void> Store::restoreJson(const std::string& json) {
402 std::string parseError;
403 auto document = eve::json::Document::parse(json, &parseError);
404 const auto root = document.root();
405 uint64_t nextId = 0, nextOrder = 0, nextSequence = 0;
406 if (!document.valid() || !root.isObject() || root.getInt("version") != 1 || !parseU64(root.get("nextId"), nextId) ||
407 !parseU64(root.get("nextOrder"), nextOrder) || !parseU64(root.get("nextSequence"), nextSequence) ||
408 nextId == 0 || nextOrder == 0 || nextSequence == 0 || !root.get("rules").isArray()) {
409 return eve::Result<void>::failure(
410 eve::Diagnostic::error(eve::DiagnosticCode::ParseError,
411 parseError.empty() ? "invalid authority snapshot" : parseError, "authority.store"));
412 }
413 std::deque<Rule> restored;
414 const auto values = root.get("rules");
415 for (size_t i = 0; i < values.size(); ++i) {
416 const auto item = values.at(i);
417 Rule rule;
418 uint64_t order = 0, numericId = 0;
419 const auto effect = item.getString("effect");
420 if (!item.isObject() || item.getString("id").empty() || item.getString("actor").empty() ||
421 item.getString("scope").empty() || item.getString("capability").empty() ||
422 (effect != "grant" && effect != "deny") || !parseU64(item.get("order"), order) || order == 0 ||
423 order >= nextOrder || !parseRuleId(item.getString("id"), numericId) || numericId >= nextId ||
424 !item.get("priority").isNumber() || !item.get("duration").isNumber() || !item.get("remaining").isNumber()) {
425 return eve::Result<void>::failure(
426 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "invalid rule at index " + std::to_string(i),
427 "authority.store.rules[" + std::to_string(i) + "]"));
428 }
429 rule.id = item.getString("id");
430 rule.actor = item.getString("actor");
431 rule.scope = item.getString("scope");
432 rule.capability = item.getString("capability");
433 rule.source = item.getString("source");
434 rule.effect = effect == "grant" ? RuleEffect::Grant : RuleEffect::Deny;
435 rule.priority = item.getInt("priority");
436 rule.duration = item.getDouble("duration");
437 rule.remaining = item.getDouble("remaining");
438 rule.order = order;
439 if (!std::isfinite(rule.duration) || !std::isfinite(rule.remaining) || rule.duration < 0.0 ||
440 rule.remaining == 0.0 || rule.remaining < -1.0 ||
441 std::any_of(
442 restored.begin(), restored.end(),
443 [&](const Rule& old) { return old.id == rule.id || old.order == rule.order; })) {
444 return eve::Result<void>::failure(eve::Diagnostic::error(
445 eve::DiagnosticCode::InvariantViolation, "invalid rule state at index " + std::to_string(i),
446 "authority.store.rules[" + std::to_string(i) + "]"));
447 }
448 restored.push_back(std::move(rule));
449 }
450 rules_ = std::move(restored);
451 nextId_ = nextId;
452 nextOrder_ = nextOrder;
453 nextSequence_ = nextSequence;
454 events_.clear();
455 query_.clear();
456 rebuildIndex();
457 revision_ = eve::Revision(nextSequence > 0 ? nextSequence - 1 : 0);
458 tick_ = eve::SimulationTick{};
459 return eve::Result<void>::success();
460}

469eve::Result<void> Store::restoreSnapshot(const eve::SnapshotEnvelope& source,
470 const eve::SnapshotHashProvider& hashProvider) {
471 if (source.type != "authority.store" || source.schema != authoritySchema())
472 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
473 "snapshot does not belong to authority::Store"));
474 if (!instanceId_.isNil() && source.instanceId != instanceId_)
475 return eve::Result<void>::failure(eve::Diagnostic::error(
476 eve::DiagnosticCode::Conflict, "snapshot instanceId does not match authority::Store"));
477
478 auto migrated = authorityMigrations().migrate(source, eve::SchemaVersion(1), hashProvider);
479 if (!migrated.ok()) return eve::Result<void>::failure(migrated.status());
480 const auto& candidate = migrated.value();
481 auto metadata = eve::validateSnapshotPayloadMetadata(candidate.payload, candidate.revision, candidate.tick);
482 if (!metadata.ok()) return eve::Result<void>::failure(metadata.status());
483 auto payload = candidate.payload.toJson();
484 if (!payload.ok()) return eve::Result<void>::failure(payload.status());
485 const std::string payloadJson = std::move(payload).takeValue();
486
487 // restoreJson parses into local containers before replacing rules/counters;
488 // it is therefore the domain candidate-builder and has no partial writes.
489 auto restored = restoreJson(payloadJson);
490 if (!restored.ok()) return eve::Result<void>::failure(restored.status());
491
492 instanceId_ = candidate.instanceId;
493 revision_ = candidate.revision;
494 tick_ = candidate.tick;
495 return eve::Result<void>::success();
496}
