15bool readVec3(const editing::Value& value, glm::vec3& output) {
16 const auto* array = value.getIf<editing::Value::Array>();
17 if (!array || array->size() != 3) return false;
18 for (std::size_t index = 0; index < 3; ++index) {
19 const auto* number = (*array)[index].getIf<double>();
20 if (!number || !std::isfinite(*number)) return false;
21 output[static_cast<int>(index)] = static_cast<float>(*number);
22 }
23 return true;

64editing::Result<std::unique_ptr<VegetationFieldTarget>> VegetationFieldTarget::create(
65 std::string id, graphics::VegetationField& fieldValue) {
66 if (id.empty())
67 return fail<std::unique_ptr<VegetationFieldTarget>>(
68 editing::Status::Rejected, "editor.vegetation-field.empty-id", "Vegetation field target ID is empty");
69 const auto revision = fieldValue.revision();
70 auto elements = fieldValue.snapshotElements();
71 if (!elements.ok()) return editing::Result<std::unique_ptr<VegetationFieldTarget>>::failure(elements.status());
72 std::vector<Entry> entries;
73 try {
74 entries.reserve(elements.value().size());
75 for (std::size_t index = 0; index < elements.value().size(); ++index)
76 entries.push_back(
77 {editing::StableId("element-" + std::to_string(index + 1)), std::move(elements.value()[index])});
78 return editing::applied<std::unique_ptr<VegetationFieldTarget>>(
79 std::unique_ptr<VegetationFieldTarget>(new VegetationFieldTarget(
80 std::move(id), &fieldValue, fieldValue.globalValues(), std::move(entries), revision)));
81 } catch (const std::bad_alloc&) {
82 return fail<std::unique_ptr<VegetationFieldTarget>>(
83 editing::Status::Failed, "editor.vegetation-field.allocation", "Vegetation field target allocation failed");
84 }

174 if (!valid.ok()) return editing::Result<editing::DomainOperation>::failure(valid.status());
175 return replacement(*before, after, {});

178editing::Result<editing::DomainOperation> VegetationFieldTarget::makeSet(const editing::SelectionSnapshot& selection,
179 const editing::PropertyPath& path,
180 const editing::Value& value,
181 editing::PropertySetMode mode) const {
182 if (mode == editing::PropertySetMode::Reset) return makeReset(selection, path);
183 const Entry* before = selected(selection);
184 const auto descriptor = schema(selection).find(path);
185 if (!before || !descriptor || mode != editing::PropertySetMode::Absolute)
186 return fail<editing::DomainOperation>(editing::Status::Rejected, "editor.vegetation-field.property",
187 "Vegetation field property edit is invalid");
188 Entry after = *before;
189 if (path == editing::PropertyPath("element.center")) {
190 if (!readVec3(value, after.value.center))
191 return fail<editing::DomainOperation>(editing::Status::Rejected, "editor.vegetation-field.center",
192 "Vegetation field center must be a finite vec3");
193 } else if (path == editing::PropertyPath("element.extents")) {
194 if (!readVec3(value, after.value.extents))
195 return fail<editing::DomainOperation>(editing::Status::Rejected, "editor.vegetation-field.extents",
196 "Vegetation field extents must be a finite vec3");
197 } else {
198 const auto* number = value.getIf<double>();
199 if (!number || !std::isfinite(*number))
200 return fail<editing::DomainOperation>(editing::Status::Rejected, "editor.vegetation-field.yaw",
201 "Vegetation field yaw must be finite");
202 after.value.yaw = static_cast<float>(*number);
203 }
204 graphics::VegetationField validator;
205 std::vector<graphics::VegetationElement> values;
206 values.reserve(entries_.size());
207 for (const Entry& entry : entries_) values.push_back(entry.id == after.id ? after.value : entry.value);
208 const auto valid = validator.replace(globals_, values);
209 if (!valid.ok()) return editing::Result<editing::DomainOperation>::failure(valid.status());
210 return replacement(*before, after, path.value());
211}

222editing::Result<void> VegetationFieldTarget::validateAndPublish(const std::vector<Entry>& entries) {
223 std::vector<graphics::VegetationElement> values;
224 try {
225 values.reserve(entries.size());
226 for (const Entry& entry : entries) values.push_back(entry.value);
227 } catch (const std::bad_alloc&) {
228 return fail<void>(editing::Status::Failed, "editor.vegetation-field.allocation",
229 "Vegetation field publication allocation failed");
230 }
231 if (!field_) {
232 graphics::VegetationField validator;
233 const auto result = validator.replace(globals_, values);
234 return result.ok() ? editing::applied<void>() : editing::Result<void>::failure(result.status());
235 }
236 if (field_->revision() != fieldRevision_)
237 return fail<void>(editing::Status::Conflict, "editor.vegetation-field.runtime-stale",
238 "Vegetation field changed outside this editor target");
239 const auto result = field_->replace(globals_, values);
240 if (!result.ok()) return editing::Result<void>::failure(result.status());
241 fieldRevision_ = field_->revision();
242 return editing::applied<void>();
243}

245editing::Result<void> VegetationFieldTarget::applyDomainOperation(const editing::DomainOperation& operation) {
246 if (operation.target != targetId() || operation.type != "vegetation-field.element.transform.v1")
247 return fail<void>(editing::Status::Rejected, "editor.vegetation-field.operation",
248 "Vegetation field operation mismatch");
249 const auto* idValue = field(operation.payload, "id");
250 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
251 Entry* current = id ? selected(*id) : nullptr;
252 if (!current)
253 return fail<void>(editing::Status::NotFound, "editor.vegetation-field.element",
254 "Vegetation field element was not found");
255 Entry replacementEntry = *current;
256 const auto* center = field(operation.payload, "center");
257 const auto* extents = field(operation.payload, "extents");
258 if (!center || !extents || !readVec3(*center, replacementEntry.value.center) ||
259 !readVec3(*extents, replacementEntry.value.extents))
260 return fail<void>(editing::Status::Rejected, "editor.vegetation-field.transform",
261 "Vegetation field transform payload is invalid");
262 const auto* yawValue = field(operation.payload, "yaw");
263 const auto* yaw = yawValue ? yawValue->getIf<double>() : nullptr;
264 if (!yaw || !std::isfinite(*yaw))
265 return fail<void>(editing::Status::Rejected, "editor.vegetation-field.transform",
266 "Vegetation field transform payload is invalid");
267 replacementEntry.value.yaw = static_cast<float>(*yaw);
268 auto candidate = entries_;
269 *std::find_if(candidate.begin(), candidate.end(), [&](const Entry& entry) { return entry.id == current->id; }) =
270 std::move(replacementEntry);
271 auto published = validateAndPublish(candidate);
272 if (!published.ok()) return published;
273 entries_ = std::move(candidate);
274 ++revision_;
275 dirty_.include(0, 0);
276 return editing::applied<void>();
277}
