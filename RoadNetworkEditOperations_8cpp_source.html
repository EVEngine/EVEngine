87EditorResult<editing::DomainOperation> RoadNetworkEditTarget::makeMergeNodesImpl(std::uint32_t keepNodeId,
88 std::uint32_t removeNodeId,
89 float maxDistance,
90 bool connectTurns) const {
91 auto removedNode = network_->nodeResult(removeNodeId);
92 if (!removedNode.ok()) return roadFailure<editing::DomainOperation>(removedNode.status());
93 std::vector<procgen::road::RoadEdge> affectedEdges;
94 for (const auto& edge : network_->edges())
95 if (edge.from == removeNodeId || edge.to == removeNodeId) affectedEdges.push_back(edge);
96 std::vector<procgen::road::RoadLaneConnection> affectedLinks;
97 for (const auto& link : network_->laneLinks()) {
98 const bool referencesAffectedEdge =
99 std::any_of(affectedEdges.begin(), affectedEdges.end(),
100 [&](const auto& edge) { return link.inEdge == edge.id || link.outEdge == edge.id; });
101 if (referencesAffectedEdge) affectedLinks.push_back(link);
102 }
103 std::vector<procgen::road::RoadLaneConnection> affectedBlockedLinks;
104 for (const auto& link : network_->blockedLaneLinks()) {
105 const bool referencesAffectedEdge =
106 std::any_of(affectedEdges.begin(), affectedEdges.end(),
107 [&](const auto& edge) { return link.inEdge == edge.id || link.outEdge == edge.id; });
108 if (referencesAffectedEdge) affectedBlockedLinks.push_back(link);
109 }
110 auto candidate = *network_;
111 auto merged = candidate.mergeNodes(keepNodeId, removeNodeId, maxDistance);
112 if (!merged.ok()) return roadFailure<editing::DomainOperation>(merged.status());
113 if (connectTurns) {
114 auto connected = candidate.connectAllTurns(keepNodeId);
115 if (!connected.ok()) return roadFailure<editing::DomainOperation>(connected.status());
116 }
117
118 editing::DomainOperation operation;
119 operation.type = connectTurns ? kMergeNodesAndConnect : kMergeNodes;
120 operation.inverseType = kUnmergeNodes;
121 operation.target = targetId();
122 operation.payload = encodeMergeNodes(keepNodeId, removeNodeId, maxDistance);
123 operation.inverse = encodeUnmergeNodes(removedNode.value(), affectedEdges, affectedLinks, affectedBlockedLinks);
124 operation.hasInverse = true;
125 operation.affectedObjects.emplace_back(targetId(), "node/" + std::to_string(keepNodeId));
126 operation.affectedObjects.emplace_back(targetId(), "node/" + std::to_string(removeNodeId));
127 for (const auto& edge : affectedEdges)
128 operation.affectedObjects.emplace_back(targetId(), "edge/" + std::to_string(edge.id));
129 operation.affectedProperties.emplace_back("topology");
130 return editing::applied(std::move(operation));

309EditorResult<editing::DomainOperation> RoadNetworkEditTarget::makeSnapEdgeEndpoint(std::uint32_t edgeId,
310 bool fromEndpoint, float maxDistance,
311 bool connectTurns) const {
312 if (!std::isfinite(maxDistance) || maxDistance < 0.f)
313 return reject<editing::DomainOperation>("editor.road.snap-distance",
314 "Road endpoint snap distance must be finite and non-negative");
315 auto edge = network_->edgeResult(edgeId);
316 if (!edge.ok()) return roadFailure<editing::DomainOperation>(edge.status());
317 const std::uint32_t oldNodeId = fromEndpoint ? edge.value().from : edge.value().to;
318 const std::uint32_t otherNodeId = fromEndpoint ? edge.value().to : edge.value().from;
319 auto endpoint = network_->nodeResult(oldNodeId);
320 if (!endpoint.ok()) return roadFailure<editing::DomainOperation>(endpoint.status());
321
322 const float maximumSquared = maxDistance * maxDistance;
323 float bestSquared = std::numeric_limits<float>::max();
324 std::uint32_t bestNodeId = 0;
325 for (const auto& node : network_->nodes()) {
326 if (node.id == oldNodeId || node.id == otherNodeId) continue;
327 const float dx = node.x - endpoint.value().x;
328 const float dy = node.y - endpoint.value().y;
329 const float dz = node.z - endpoint.value().z;
330 const float distanceSquared = dx * dx + dy * dy + dz * dz;
331 if (distanceSquared > maximumSquared) continue;
332 if (distanceSquared < bestSquared || (distanceSquared == bestSquared && node.id < bestNodeId)) {
333 bestSquared = distanceSquared;
334 bestNodeId = node.id;
335 }
336 }
337 if (bestNodeId == 0)
338 return reject<editing::DomainOperation>("editor.road.snap-not-found",
339 "No distinct road node is inside the endpoint snap distance",
340 EditorStatus::NotFound);
341 return connectTurns ? makeReconnectEdgeEndpointAndConnect(edgeId, fromEndpoint, bestNodeId)
342 : makeReconnectEdgeEndpoint(edgeId, fromEndpoint, bestNodeId);
343}
