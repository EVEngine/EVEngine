65eve::Result<RuleDefinition> RuleEngine::validateAndNormalize(RuleDefinition rule) {
66 if (rule.id.empty())
67 return eve::Result<RuleDefinition>::failure(eve::Diagnostic::error(
68 eve::DiagnosticCode::InvalidArgument, "rule id must be non-empty", "id", {}, "emergence.rule_engine"));
69 if (!rule.condition.isValid())
70 return eve::Result<RuleDefinition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
71 "rule condition is invalid", "condition", {},
72 "emergence.rule_engine"));
73 for (const auto& action : rule.actions) {
74 if (action.kind.empty())
75 return eve::Result<RuleDefinition>::failure(
76 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "action kind must be non-empty",
77 "actions.kind", {}, "emergence.rule_engine"));
78 }
79 if (rule.watchKeys.empty()) {
80 auto keys = collectWatchKeys(rule.condition);
81 if (!keys) return eve::Result<RuleDefinition>::failure(keys.status());
82 rule.watchKeys = std::move(keys).takeValue();
83 } else {
84 std::sort(rule.watchKeys.begin(), rule.watchKeys.end());
85 rule.watchKeys.erase(std::unique(rule.watchKeys.begin(), rule.watchKeys.end()), rule.watchKeys.end());
86 for (const auto& key : rule.watchKeys) {
87 auto parsed = parseFactKey(key);
88 if (!parsed) return eve::Result<RuleDefinition>::failure(parsed.status());
89 }
90 }
91 if (rule.watchKeys.empty())
92 return eve::Result<RuleDefinition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
93 "rule must declare at least one watch key",
94 "watchKeys", {}, "emergence.rule_engine"));
95 return eve::Result<RuleDefinition>::success(std::move(rule));
96}

98eve::Result<int> RuleEngine::replaceCatalogue(std::vector<RuleDefinition> rules) {
99 if (draining_)
100 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
101 "cannot replace catalogue during drain", {}, {},
102 "emergence.rule_engine"));
103 std::vector<RuleDefinition> next;
104 next.reserve(rules.size());
105 std::unordered_map<std::string, std::uint32_t> ids;
106 WatchIndex index;
107 for (auto& rule : rules) {
108 auto normalized = validateAndNormalize(std::move(rule));
109 if (!normalized) return eve::Result<int>::failure(normalized.status());
110 auto value = std::move(normalized).takeValue();
111 if (ids.contains(value.id))
112 return eve::Result<int>::failure(eve::Diagnostic::error(
113 eve::DiagnosticCode::AlreadyExists, "duplicate rule id", "id", {}, "emergence.rule_engine"));
114 const auto slot = static_cast<std::uint32_t>(next.size());
115 ids.emplace(value.id, slot);
116 index.addRule(slot, value.watchKeys);
117 next.push_back(std::move(value));
118 }
119 rules_ = std::move(next);
120 runtime_.assign(rules_.size(), RuleRuntime{});
121 idToIndex_ = std::move(ids);
122 watchIndex_ = std::move(index);
123 dirty_.clear();
124 activations_.clear();
125 return eve::Result<int>::success(static_cast<int>(rules_.size()), eve::Status::success(eve::StatusCode::Applied));
126}

128eve::Result<int> RuleEngine::replaceCatalogueJson(std::string_view json) {
129 auto parsed = eve::Value::fromJson(json);
130 if (!parsed) return eve::Result<int>::failure(parsed.status());
131 const auto* root = parsed.value().getIf<eve::Value::Object>();
132 if (root == nullptr)
133 return eve::Result<int>::failure(eve::Diagnostic::error(
134 eve::DiagnosticCode::InvalidArgument, "rules document must be an object", {}, {}, "emergence.rule_engine"));
135 const auto* schema = objField(*root, "schema");
136 const auto* version = objField(*root, "version");
137 if (schema == nullptr || !schema->isString() || schema->asString() != "eve.emergence.rules")
138 return eve::Result<int>::failure(eve::Diagnostic::error(
139 eve::DiagnosticCode::InvalidArgument, "unexpected rules schema", "schema", {}, "emergence.rule_engine"));
140 if (version == nullptr || !version->isInt64() || version->asInt() != 1)
141 return eve::Result<int>::failure(eve::Diagnostic::error(
142 eve::DiagnosticCode::UnknownVersion, "unsupported rules version", "version", {}, "emergence.rule_engine"));
143 const auto* rulesValue = objField(*root, "rules");
144 const auto* rulesArray = rulesValue == nullptr ? nullptr : rulesValue->getIf<eve::Value::Array>();
145 if (rulesArray == nullptr)
146 return eve::Result<int>::failure(eve::Diagnostic::error(
147 eve::DiagnosticCode::InvalidArgument, "rules must be an array", "rules", {}, "emergence.rule_engine"));
148
149 std::vector<RuleDefinition> rules;
150 rules.reserve(rulesArray->size());
151 for (std::size_t i = 0; i < rulesArray->size(); ++i) {
152 const auto* object = (*rulesArray)[i].getIf<eve::Value::Object>();
153 if (object == nullptr)
154 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
155 "rule entry must be an object", "rules", {},
156 "emergence.rule_engine"));
157 RuleDefinition rule;
158 const auto* id = objField(*object, "id");
159 if (id == nullptr || !id->isString() || id->asString().empty())
160 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
161 "rule id must be a non-empty string", "rules.id",
162 {}, "emergence.rule_engine"));
163 rule.id = id->asString();
164 if (const auto* priority = objField(*object, "priority")) {
165 auto parsed = requireInt(*priority, "rules.priority");
166 if (!parsed) return eve::Result<int>::failure(parsed.status());
167 rule.priority = parsed.value();
168 }
169 if (const auto* once = objField(*object, "once")) {
170 if (!once->isBool())
171 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
172 "once must be a boolean", "rules.once", {},
173 "emergence.rule_engine"));
174 rule.once = once->asBool();
175 }
176 if (const auto* cooldown = objField(*object, "cooldownTicks")) {
177 auto parsed = requireNonNegativeU64(*cooldown, "rules.cooldownTicks");
178 if (!parsed) return eve::Result<int>::failure(parsed.status());
179 rule.cooldownTicks = parsed.value();
180 }
181 if (const auto* fireMode = objField(*object, "fireMode")) {
182 if (!fireMode->isString())
183 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
184 "fireMode must be a string", "rules.fireMode",
185 {}, "emergence.rule_engine"));
186 if (fireMode->asString() == "rising")
187 rule.fireMode = FireMode::Rising;
188 else if (fireMode->asString() == "level")
189 rule.fireMode = FireMode::Level;
190 else
191 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
192 "unknown fireMode", "rules.fireMode", {},
193 "emergence.rule_engine"));
194 }
195 const auto* conditionValue = objField(*object, "condition");
196 if (conditionValue == nullptr)
197 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
198 "rule requires condition", "rules.condition", {},
199 "emergence.rule_engine"));
200 auto condition = decodeCondition(*conditionValue);
201 if (!condition) return eve::Result<int>::failure(condition.status());
202 rule.condition = std::move(condition).takeValue();
203 if (const auto* watch = objField(*object, "watchKeys")) {
204 const auto* array = watch->getIf<eve::Value::Array>();
205 if (array == nullptr)
206 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
207 "watchKeys must be an array", "rules.watchKeys",
208 {}, "emergence.rule_engine"));
209 for (const auto& item : *array) {
210 if (!item.isString() || item.asString().empty())
211 return eve::Result<int>::failure(eve::Diagnostic::error(
212 eve::DiagnosticCode::InvalidArgument, "watch key must be a non-empty string", "rules.watchKeys",
213 {}, "emergence.rule_engine"));
214 rule.watchKeys.push_back(item.asString());
215 }
216 }
217 if (const auto* actions = objField(*object, "actions")) {
218 const auto* array = actions->getIf<eve::Value::Array>();
219 if (array == nullptr)
220 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
221 "actions must be an array", "rules.actions", {},
222 "emergence.rule_engine"));
223 for (const auto& item : *array) {
224 const auto* actionObject = item.getIf<eve::Value::Object>();
225 if (actionObject == nullptr)
226 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
227 "action must be an object", "rules.actions",
228 {}, "emergence.rule_engine"));
229 EmergenceAction action;
230 const auto* kind = objField(*actionObject, "kind");
231 if (kind == nullptr || !kind->isString() || kind->asString().empty())
232 return eve::Result<int>::failure(eve::Diagnostic::error(
233 eve::DiagnosticCode::InvalidArgument, "action kind must be a non-empty string",
234 "rules.actions.kind", {}, "emergence.rule_engine"));
235 action.kind = kind->asString();
236 if (const auto* args = objField(*actionObject, "args")) action.args = *args;
237 rule.actions.push_back(std::move(action));
238 }
239 }
240 rules.push_back(std::move(rule));
241 }
242 return replaceCatalogue(std::move(rules));
243}

331eve::Result<void> RuleEngine::executeOne(const EmergenceAction& action, std::uint64_t /*tick*/) {
332 if (action.kind == "fact.set") {
333 const auto* object = action.args.getIf<eve::Value::Object>();
334 if (object == nullptr)
335 return fail(eve::DiagnosticCode::InvalidArgument, "fact.set args must be an object", "args");
336 const auto* domain = objField(*object, "domain");
337 const auto* key = objField(*object, "key");
338 const auto* value = objField(*object, "value");
339 if (domain == nullptr || !domain->isString() || key == nullptr || !key->isString() || value == nullptr)
340 return fail(eve::DiagnosticCode::InvalidArgument, "fact.set requires domain, key, value", "args");
341 const auto& domainText = domain->asString();
342 const auto& keyText = key->asString();
343 if (domainText == "value")
344 return setValue(keyText, *value).andThen([](bool) { return eve::Result<void>::success(); });
345 if (domainText == "tag") {
346 if (!value->isBool()) return fail(eve::DiagnosticCode::InvalidArgument, "tag value must be bool", "value");
347 return setTag(keyText, value->asBool()).andThen([](bool) { return eve::Result<void>::success(); });
348 }
349 if (domainText == "attr")
350 return setAttribute(keyText, *value).andThen([](bool) { return eve::Result<void>::success(); });
351 if (domainText == "resource")
352 return setResource(keyText, *value).andThen([](bool) { return eve::Result<void>::success(); });
353 if (domainText == "state")
354 return setState(keyText, *value).andThen([](bool) { return eve::Result<void>::success(); });
355 if (domainText == "authority") {
356 if (!value->isBool())
357 return fail(eve::DiagnosticCode::InvalidArgument, "authority value must be bool", "value");
358 return setAuthority(keyText, value->asBool()).andThen([](bool) { return eve::Result<void>::success(); });
359 }
360 return fail(eve::DiagnosticCode::InvalidArgument, "unknown fact.set domain", "domain");
361 }
362 if (action.kind == "economy.credit" || action.kind == "economy.debit") {
363 auto* economy = eve::cap::query<eve::economy::IEconomy>();
364 if (economy == nullptr)
365 return fail(eve::DiagnosticCode::PreconditionViolation, "IEconomy provider is absent", action.kind.c_str());
366 const auto* object = action.args.getIf<eve::Value::Object>();
367 if (object == nullptr)
368 return fail(eve::DiagnosticCode::InvalidArgument, "economy action args must be an object", "args");
369 const auto* player = objField(*object, "player");
370 const auto* type = objField(*object, "type");
371 const auto* amount = objField(*object, "amount");
372 if (player == nullptr || type == nullptr || !type->isString() || amount == nullptr)
373 return fail(eve::DiagnosticCode::InvalidArgument, "economy action requires player, type, amount", "args");
374 auto playerId = requireInt(*player, "player");
375 if (!playerId) return eve::Result<void>::failure(playerId.status());
376 auto qty = requireInt(*amount, "amount");
377 if (!qty) return eve::Result<void>::failure(qty.status());
378 if (qty.value() < 0) return fail(eve::DiagnosticCode::InvalidArgument, "amount must be non-negative", "amount");
379 if (action.kind == "economy.credit") {
380 economy->credit(playerId.value(), type->asString(), qty.value());
381 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
382 }
383 if (!economy->debit(playerId.value(), type->asString(), qty.value()))
384 return fail(eve::DiagnosticCode::Conflict, "economy debit rejected", "amount");
385 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
386 }
387 if (action.kind == "emit") {
388 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
389 }
390
391 eve::Result<void> handlerFailure = eve::Result<void>::success();
392 bool consumed = false;
393 eve::cap::forEach<eve::IEmergenceActionHandler>([&](eve::IEmergenceActionHandler* handler) {
394 if (consumed || !handlerFailure.ok()) return;
395 auto handled = handler->tryHandle(action.kind, action.args);
396 if (!handled) {
397 handlerFailure = eve::Result<void>::failure(handled.status());
398 return;
399 }
400 if (handled.value()) consumed = true;
401 });
402 if (!handlerFailure.ok()) return handlerFailure;
403 if (!consumed && action.kind != "emit") {
404 // Unhandled domain actions remain in the activation log for scripts.
405 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
406 }
407 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
408}

418eve::Result<int> RuleEngine::drain(std::uint64_t tick) {
419 if (draining_)
420 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
421 "nested drain is not allowed", {}, {},
422 "emergence.rule_engine"));
423 draining_ = true;
424 lastDrainEvaluations_ = 0;
425 int produced = 0;

427 struct Guard {
428 bool& flag;
429 ~Guard() { flag = false; }
430 } guard{draining_};

432 struct PendingFire {
433 Activation activation;
434 std::uint32_t index = 0;
435 bool nowPassed = false;
436 bool once = false;
437 std::uint64_t cooldownTicks = 0;
438 };
439
440 // Fixed-point: evaluate current dirty set, execute activations (which may dirty more), repeat.
441 constexpr int kMaxPasses = 64;
442 for (int pass = 0; pass < kMaxPasses; ++pass) {
443 if (dirty_.empty()) break;
444 std::vector<std::uint32_t> batch(dirty_.begin(), dirty_.end());
445 dirty_.clear();
446 std::sort(batch.begin(), batch.end(), [&](std::uint32_t left, std::uint32_t right) {
447 if (rules_[left].priority != rules_[right].priority) return rules_[left].priority > rules_[right].priority;
448 return rules_[left].id < rules_[right].id;
449 });
450
451 std::vector<PendingFire> pending;
452 for (const auto index : batch) {
453 if (index >= rules_.size()) continue;
454 auto& runtime = runtime_[index];
455 const auto& rule = rules_[index];
456 if (runtime.disabled) continue;
457 if (runtime.cooldownUntil > tick) continue;
458 ++lastDrainEvaluations_;
459 const auto result = rule.condition.evaluate(facts_);
460 const bool nowPassed = result.passed();
461 bool fire = false;
462 if (rule.fireMode == FireMode::Rising)
463 fire = !runtime.passed && nowPassed;
464 else
465 fire = nowPassed;
466 if (!fire) {
467 runtime.passed = nowPassed;
468 continue;
469 }
470 PendingFire entry;
471 entry.activation.sequence = nextSequence_++;
472 entry.activation.tick = tick;
473 entry.activation.ruleId = rule.id;
474 entry.activation.priority = rule.priority;
475 entry.activation.actions = rule.actions;
476 entry.index = index;
477 entry.nowPassed = nowPassed;
478 entry.once = rule.once;
479 entry.cooldownTicks = rule.cooldownTicks;
480 pending.push_back(std::move(entry));
481 }
482
483 for (std::size_t i = 0; i < pending.size(); ++i) {
484 auto& entry = pending[i];
485 auto executed = executeActions(entry.activation.actions, tick);
486 if (!executed) {
487 for (std::size_t j = i; j < pending.size(); ++j) dirty_.insert(pending[j].index);
488 return eve::Result<int>::failure(executed.status());
489 }
490 auto& runtime = runtime_[entry.index];
491 runtime.passed = entry.nowPassed;
492 runtime.fired = true;
493 if (entry.once) runtime.disabled = true;
494 if (entry.cooldownTicks > 0) runtime.cooldownUntil = tick + entry.cooldownTicks;
495 activations_.push_back(std::move(entry.activation));
496 ++produced;
497 }
498 }
499 if (!dirty_.empty())
500 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
501 "emergence drain exceeded cascade pass budget", {}, {},
502 "emergence.rule_engine"));
503 return eve::Result<int>::success(
504 produced, eve::Status::success(produced > 0 ? eve::StatusCode::Applied : eve::StatusCode::NoOp));
505}

526std::string RuleEngine::snapshotJson() const {
527 eve::Value::Object root;
528 root.emplace("schema", eve::Value("eve.emergence.runtime"));
529 root.emplace("version", eve::Value(std::int64_t{1}));
530 auto factsJson = eve::Value::fromJson(facts_.snapshotJson());
531 if (factsJson) root.emplace("facts", std::move(factsJson).takeValue());
532 eve::Value::Array rules;
533 for (std::size_t i = 0; i < rules_.size(); ++i) {
534 eve::Value::Object row;
535 row.emplace("id", eve::Value(rules_[i].id));
536 row.emplace("passed", eve::Value(runtime_[i].passed));
537 row.emplace("fired", eve::Value(runtime_[i].fired));
538 row.emplace("disabled", eve::Value(runtime_[i].disabled));
539 row.emplace("cooldownUntil", eve::Value(static_cast<std::int64_t>(runtime_[i].cooldownUntil)));
540 rules.push_back(eve::Value(std::move(row)));
541 }
542 root.emplace("rules", eve::Value(std::move(rules)));
543 auto encoded = eve::Value(std::move(root)).toJson();
544 return encoded ? std::move(encoded).takeValue() : std::string("{}");
545}

547eve::Result<void> RuleEngine::restoreJson(std::string_view json) {
548 auto parsed = eve::Value::fromJson(json);
549 if (!parsed) return eve::Result<void>::failure(parsed.status());
550 const auto* root = parsed.value().getIf<eve::Value::Object>();
551 if (root == nullptr) return fail(eve::DiagnosticCode::InvalidArgument, "runtime snapshot must be an object");
552 if (!exactFields(*root, {"schema", "version", "facts", "rules"}))
553 return fail(eve::DiagnosticCode::InvalidArgument, "runtime snapshot contains an unknown field");
554 const auto* schema = objField(*root, "schema");
555 const auto* version = objField(*root, "version");
556 if (schema == nullptr || !schema->isString() || schema->asString() != "eve.emergence.runtime")
557 return fail(eve::DiagnosticCode::InvalidArgument, "unexpected runtime schema", "schema");
558 if (version == nullptr || !version->isInt64() || version->asInt() != 1)
559 return fail(eve::DiagnosticCode::UnknownVersion, "unsupported runtime version", "version");
560 const auto* factsValue = objField(*root, "facts");
561 const auto* rulesValue = objField(*root, "rules");
562 if (factsValue == nullptr)
563 return fail(eve::DiagnosticCode::InvalidArgument, "runtime snapshot requires facts", "facts");
564 if (rulesValue == nullptr)
565 return fail(eve::DiagnosticCode::InvalidArgument, "runtime snapshot requires rules", "rules");
566
567 FactStore nextFacts;
568 {
569 auto encoded = factsValue->toJson();
570 if (!encoded) return eve::Result<void>::failure(encoded.status());
571 auto restored = nextFacts.restoreJson(encoded.value());
572 if (!restored) return restored;
573 }
574
575 std::vector<RuleRuntime> nextRuntime(runtime_.size());
576 const auto* array = rulesValue->getIf<eve::Value::Array>();
577 if (array == nullptr) return fail(eve::DiagnosticCode::InvalidArgument, "rules must be an array", "rules");
578 std::unordered_set<std::string> seenIds;
579 for (const auto& item : *array) {
580 const auto* object = item.getIf<eve::Value::Object>();
581 if (object == nullptr) return fail(eve::DiagnosticCode::InvalidArgument, "rule latch must be an object");
582 if (!exactFields(*object, {"id", "passed", "fired", "disabled", "cooldownUntil"}))
583 return fail(eve::DiagnosticCode::InvalidArgument, "rule latch contains an unknown field");
584 const auto* id = objField(*object, "id");
585 if (id == nullptr || !id->isString() || id->asString().empty())
586 return fail(eve::DiagnosticCode::InvalidArgument, "rule latch requires id", "id");
587 if (!seenIds.insert(id->asString()).second)
588 return fail(eve::DiagnosticCode::AlreadyExists, "duplicate rule latch id", "id");
589 auto it = idToIndex_.find(id->asString());
590 if (it == idToIndex_.end())
591 return fail(eve::DiagnosticCode::NotFound, "runtime rule id missing from catalogue", "id");
592 auto& runtime = nextRuntime[it->second];
593 const auto* passed = objField(*object, "passed");
594 const auto* fired = objField(*object, "fired");
595 const auto* disabled = objField(*object, "disabled");
596 const auto* cooldown = objField(*object, "cooldownUntil");
597 if (passed == nullptr || !passed->isBool())
598 return fail(eve::DiagnosticCode::InvalidArgument, "passed must be bool", "passed");
599 if (fired == nullptr || !fired->isBool())
600 return fail(eve::DiagnosticCode::InvalidArgument, "fired must be bool", "fired");
601 if (disabled == nullptr || !disabled->isBool())
602 return fail(eve::DiagnosticCode::InvalidArgument, "disabled must be bool", "disabled");
603 if (cooldown == nullptr)
604 return fail(eve::DiagnosticCode::InvalidArgument, "cooldownUntil is required", "cooldownUntil");
605 auto cooldownValue = requireNonNegativeU64(*cooldown, "cooldownUntil");
606 if (!cooldownValue) return eve::Result<void>::failure(cooldownValue.status());
607 runtime.passed = passed->asBool();
608 runtime.fired = fired->asBool();
609 runtime.disabled = disabled->asBool();
610 runtime.cooldownUntil = cooldownValue.value();
611 }
612 if (seenIds.size() != rules_.size())
613 return fail(eve::DiagnosticCode::InvalidArgument, "runtime snapshot must include every catalogue rule",
614 "rules");
615
616 facts_ = std::move(nextFacts);
617 runtime_ = std::move(nextRuntime);
618 dirty_.clear();
619 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
620}
