102Result<void> validateInstancedRenderable(const Renderable3D::MeshRenderer& mr) {
103 auto* shader = mr.material ? mr.material->effectiveShader() : mr.shader;
104 if (!mr.mesh || !shader || shader->getKind() != Shader::Kind::eMesh3D || shader->isXray() || mr.usesParts() ||
105 mr.effectiveHair() || mr.xrayHighlight || mr.lodCount || mr.mesh->hasGpuSkinning() ||
106 mr.effectiveCastShadow() || mr.castOcclusion)
107 return Result<void>::failure(Diagnostic::error(
108 DiagnosticCode::Unsupported,
109 "Instanced renderables require one custom static mesh without shadow/occlusion casting, LOD, hair or Xray",
110 "graphics.instances"));
111 return Result<void>::success();
112}
113} // namespace detail
114Result<void> Renderable3D::setInstanceRange(const MeshInstanceRange& range) {
115 auto valid = validateMeshInstanceRange(range);
116 if (!valid) return valid;
117 auto mr = meshRenderer();
118 auto supported = detail::validateInstancedRenderable(*mr);
119 if (!supported) return supported;
120 mr->instances = range;
121 return Result<void>::success();
122}
123void Renderable3D::clearInstanceRange() { meshRenderer()->instances.reset(); }
