17 // Construct names before publication. Existing uniform writes only check schema and copy bytes.
18 static const std::array<std::string, 5> names = {"windGlobals", "windPhaseDistance", "windFlex", "windFrequency",
19 "windSineTime"};
20 constexpr std::array<int, 5> offsets{18, 22, 24, 27, 30};
21 constexpr std::array<int, 5> sizes{4, 2, 3, 3, 2};
22 for (size_t i = 0; i < names.size(); ++i) {
23 if (shader.usedFloats() != 32 || !shader.hasUniform(names[i]) ||

26 "grass.wind: canonical grass wind uniform layout required"));
27 }
28 for (size_t i = 0; i < names.size(); ++i) {
29 const size_t bytes = size_t(sizes[i]) * sizeof(float);

31 EV_ASSERT(written == int(bytes));
32 }

41Result<void> applyTreeWind(Shader& shader, const VegetationWindState& state, const VegetationWindProfile& vegetation,
42 const TreeWindProfile& tree, double seconds) {
43 if (!std::isfinite(tree.widthHeight.x) || tree.widthHeight.x <= 0 || !std::isfinite(tree.widthHeight.y) ||
44 tree.widthHeight.y <= 0 || !std::isfinite(tree.bendFactor) || tree.bendFactor < 0)
45 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
46 "tree.wind: positive finite dimensions and bend required"));

48 scaled.flex *= tree.bendFactor;
49 auto packet = packVegetationWind(state, scaled, seconds);
50 if (!packet.ok()) return Result<void>::failure(packet.status());
51 static const std::array<std::string, 6> names = {"windGlobals", "windPhaseDistance", "windFlex",
52 "windFrequency", "windSineTime", "treeDimensions"};
53 constexpr std::array<int, 6> offsets{0, 4, 6, 9, 12, 14};
54 constexpr std::array<int, 6> sizes{4, 2, 3, 3, 2, 2};
55 for (size_t i = 0; i < names.size(); ++i)
56 if (shader.usedFloats() != 16 || !shader.hasUniform(names[i]) ||
57 shader.getUniformIndex(names[i]) != offsets[i] || shader.getUniformFloatCount(names[i]) != sizes[i])
58 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
59 "tree.wind: canonical 16-float shader layout required"));
60 std::array<float, 16> values{};
61 std::copy(packet.value().begin(), packet.value().end(), values.begin());
62 values[14] = tree.widthHeight.x;
63 values[15] = tree.widthHeight.y;
64 for (size_t i = 0; i < names.size(); ++i) {
65 const auto bytes = size_t(sizes[i]) * sizeof(float);
66 const int written = shader.sendToVar(names[i], values.data() + offsets[i], bytes);

68 }
