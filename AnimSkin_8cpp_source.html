24Mat4 fromAiMatrix(const aiMatrix4x4& m) {
25 // Assimp is row-major; convert to column-major Mat4.
26 Mat4 out;
27 out.m[0] = m.a1;
28 out.m[1] = m.b1;
29 out.m[2] = m.c1;
30 out.m[3] = m.d1;
31 out.m[4] = m.a2;
32 out.m[5] = m.b2;
33 out.m[6] = m.c2;

35 out.m[8] = m.a3;
36 out.m[9] = m.b3;

38 out.m[11] = m.d3;
39 out.m[12] = m.a4;

41 out.m[14] = m.c4;

51void normalMatrixCofactor(const Mat4& m, float out[9]) {
52 // Row-major 3x3 linear part of the skin matrix.
53 const float r0x = m.m[0], r0y = m.m[4], r0z = m.m[8];
54 const float r1x = m.m[1], r1y = m.m[5], r1z = m.m[9];
55 const float r2x = m.m[2], r2y = m.m[6], r2z = m.m[10];
56 out[0] = r1y * r2z - r1z * r2y;
57 out[1] = r1z * r2x - r1x * r2z;
58 out[2] = r1x * r2y - r1y * r2x;
59 out[3] = r2y * r0z - r2z * r0y;
60 out[4] = r2z * r0x - r2x * r0z;
61 out[5] = r2x * r0y - r2y * r0x;
62 out[6] = r0y * r1z - r0z * r1y;
63 out[7] = r0z * r1x - r0x * r1z;
64 out[8] = r0x * r1y - r0y * r1x;
65}

81AnimSkin* AnimSkin::fromModel(const model3d::ModelData* model, int meshIndex, const AnimSkeleton* skeleton) {
82 if (!model || !skeleton) {
83 throw Exception("AnimSkin.fromModel: null model/skeleton");
84 }
85 const aiMesh* mesh = model->getMesh(meshIndex);
86 if (!mesh) {
87 throw Exception("AnimSkin.fromModel: invalid mesh index %d", meshIndex);
88 }
89 if (!mesh->HasBones() || mesh->mNumBones == 0) {
90 throw Exception("AnimSkin.fromModel: mesh %d has no bones", meshIndex);
91 }
92 if (mesh->mNumVertices == 0 || !mesh->mVertices) {
93 throw Exception("AnimSkin.fromModel: mesh %d has no vertices", meshIndex);
94 }
95
96 auto* skin = new AnimSkin();
97 skin->vertexCount_ = static_cast<int>(mesh->mNumVertices);
98 skin->bindPos_.resize(static_cast<size_t>(skin->vertexCount_) * 3u);
99 for (int v = 0; v < skin->vertexCount_; ++v) {
100 const aiVector3D& p = mesh->mVertices[v];
101 skin->bindPos_[static_cast<size_t>(v) * 3u + 0] = p.x;
102 skin->bindPos_[static_cast<size_t>(v) * 3u + 1] = p.y;
103 skin->bindPos_[static_cast<size_t>(v) * 3u + 2] = p.z;
104 }
105 if (mesh->HasNormals() && mesh->mNormals) {
106 skin->bindNrm_.resize(static_cast<size_t>(skin->vertexCount_) * 3u);
107 for (int v = 0; v < skin->vertexCount_; ++v) {
108 const aiVector3D& n = mesh->mNormals[v];
109 skin->bindNrm_[static_cast<size_t>(v) * 3u + 0] = n.x;
110 skin->bindNrm_[static_cast<size_t>(v) * 3u + 1] = n.y;
111 skin->bindNrm_[static_cast<size_t>(v) * 3u + 2] = n.z;
112 }
113 }
114
115 // Gather skin joints that map onto the skeleton by name.
116 std::vector<int> meshBoneToSkin(mesh->mNumBones, -1);
117 int matched = 0;
118 for (unsigned b = 0; b < mesh->mNumBones; ++b) {
119 const aiBone* bone = mesh->mBones[b];
120 if (!bone) continue;
121 const std::string name = bone->mName.C_Str();
122 const int skBone = skeleton->findBone(name);
123 if (skBone < 0) continue;
124 meshBoneToSkin[b] = static_cast<int>(skin->skeletonBone_.size());
125 skin->skeletonBone_.push_back(skBone);
126 skin->skinBoneNames_.push_back(name);
127 skin->inverseBind_.push_back(fromAiMatrix(bone->mOffsetMatrix));
128 ++matched;
129 }
130 if (matched == 0) {
131 delete skin;
132 throw Exception("AnimSkin.fromModel: no mesh bones matched skeleton names");
133 }
134
135 // Accumulate all influences then keep the top-k per vertex (renormalize).
136 struct Acc {
137 int bone = -1;
138 float weight = 0.f;
139 };
140 std::vector<std::vector<Acc>> acc(static_cast<size_t>(skin->vertexCount_));
141
142 for (unsigned b = 0; b < mesh->mNumBones; ++b) {
143 const int skinBone = meshBoneToSkin[b];
144 if (skinBone < 0) continue;
145 const aiBone* bone = mesh->mBones[b];
146 if (!bone) continue;
147 for (unsigned w = 0; w < bone->mNumWeights; ++w) {
148 const aiVertexWeight& vw = bone->mWeights[w];
149 if (vw.mVertexId >= mesh->mNumVertices) continue;
150 if (vw.mWeight <= 0.f) continue;
151 acc[vw.mVertexId].push_back(Acc{skinBone, vw.mWeight});
152 }
153 }
154
155 skin->influences_.assign(static_cast<size_t>(skin->vertexCount_) * kMaxInfluences, Influence{});
156 for (int v = 0; v < skin->vertexCount_; ++v) {
157 auto& list = acc[static_cast<size_t>(v)];
158 std::sort(list.begin(), list.end(), [](const Acc& a, const Acc& b) { return a.weight > b.weight; });
159 // Merge duplicate bone entries.
160 std::unordered_map<int, float> merged;
161 for (const Acc& a : list) {
162 if (a.bone < 0) continue;
163 merged[a.bone] += a.weight;
164 }
165 list.clear();
166 for (const auto& kv : merged) list.push_back(Acc{kv.first, kv.second});
167 std::sort(list.begin(), list.end(), [](const Acc& a, const Acc& b) { return a.weight > b.weight; });
168
169 float sum = 0.f;
170 const int take = std::min(kMaxInfluences, static_cast<int>(list.size()));
171 for (int i = 0; i < take; ++i) sum += list[static_cast<size_t>(i)].weight;
172 const float inv = (sum > 1e-8f) ? (1.f / sum) : 0.f;
173 for (int i = 0; i < take; ++i) {
174 Influence& dst = skin->influences_[static_cast<size_t>(v) * kMaxInfluences + i];
175 dst.bone = list[static_cast<size_t>(i)].bone;
176 dst.weight = list[static_cast<size_t>(i)].weight * inv;
177 }
178 // Vertices with no weights stay at bind (weight sum 0 → identity path in skin).
179 }
180
181 return skin;
182}

278void AnimSkin::skinPositions(const AnimPose* pose, float* outPosXYZ) const {
279 if (!pose) throw Exception("AnimSkin.skinPositions: pose is null");
280 if (!outPosXYZ) throw Exception("AnimSkin.skinPositions: outPosXYZ is null");
281 if (vertexCount_ <= 0) return;
282
283 updateMatrixPalette(pose);
284
285 for (int v = 0; v < vertexCount_; ++v) {
286 const float bx = bindPos_[static_cast<size_t>(v) * 3u + 0];
287 const float by = bindPos_[static_cast<size_t>(v) * 3u + 1];
288 const float bz = bindPos_[static_cast<size_t>(v) * 3u + 2];
289 float ox = 0.f, oy = 0.f, oz = 0.f;

291 for (int i = 0; i < kMaxInfluences; ++i) {

293 if (inf.bone < 0 || inf.weight <= 0.f) continue;

295 skinMatrices_[static_cast<size_t>(inf.bone)].transformPoint(bx, by, bz, px, py, pz);
296 ox += px * inf.weight;
297 oy += py * inf.weight;
298 oz += pz * inf.weight;
299 wsum += inf.weight;
300 }
301 if (wsum <= 1e-8f) {
302 ox = bx;
303 oy = by;
304 oz = bz;
305 }
306 outPosXYZ[static_cast<size_t>(v) * 3u + 0] = ox;
307 outPosXYZ[static_cast<size_t>(v) * 3u + 1] = oy;

351bool AnimSkin::updateSkinnedNormals(const AnimPose* pose) {
352 skinnedNrmValid_ = false;
353 if (!pose || bindNrm_.empty() || vertexCount_ <= 0) return false;
354
355 // n' = sum_i w_i * (M_i^{-1})^T * n, M_i = boneWorld * inverseBind.
356 normalMatrices_.resize(inverseBind_.size() * 9u);
357 for (size_t j = 0; j < inverseBind_.size(); ++j) {
358 const int skelBone = skeletonBone_[j];
359 const Mat4 world = Mat4::fromTRS(pose->world(skelBone));
360 const Mat4 skinMat = Mat4::mul(world, inverseBind_[j]);
361 normalMatrixCofactor(skinMat, normalMatrices_.data() + j * 9u);
362 }
363
364 skinnedNrm_.resize(static_cast<size_t>(vertexCount_) * 3u);
365 for (int v = 0; v < vertexCount_; ++v) {
366 const size_t base = static_cast<size_t>(v) * 3u;
367 const float bx = bindNrm_[base + 0];
368 const float by = bindNrm_[base + 1];
369 const float bz = bindNrm_[base + 2];
370 float nx = 0.f, ny = 0.f, nz = 0.f;
371 float wsum = 0.f;
372 for (int i = 0; i < kMaxInfluences; ++i) {
373 const Influence& inf = influences_[static_cast<size_t>(v) * kMaxInfluences + i];
374 if (inf.bone < 0 || inf.weight <= 0.f) continue;
375 const float* c = normalMatrices_.data() + static_cast<size_t>(inf.bone) * 9u;
376 const float tx = c[0] * bx + c[1] * by + c[2] * bz;
377 const float ty = c[3] * bx + c[4] * by + c[5] * bz;
378 const float tz = c[6] * bx + c[7] * by + c[8] * bz;
379 nx += tx * inf.weight;
380 ny += ty * inf.weight;
381 nz += tz * inf.weight;
382 wsum += inf.weight;
383 }
384 if (wsum <= 1e-8f) {
385 nx = bx;
386 ny = by;
387 nz = bz;
388 } else {
389 const float len = std::sqrt(nx * nx + ny * ny + nz * nz);
390 if (len > 1e-8f) {
391 nx /= len;
392 ny /= len;
393 nz /= len;
394 } else {
395 nx = bx;
396 ny = by;
397 nz = bz;
398 }
399 }
400 skinnedNrm_[base + 0] = nx;
401 skinnedNrm_[base + 1] = ny;
402 skinnedNrm_[base + 2] = nz;
403 }
404 skinnedNrmValid_ = true;
405 return true;
406}
