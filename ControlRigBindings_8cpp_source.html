27Result<void> edit(ControlRigEditor* self, F function) {
28 auto value = self->snapshot();
29 if (!value.ok()) return Result<void>::failure(value.status());
30 auto definition = animation::decodeControlRig(value.value());
31 if (!definition.ok()) return Result<void>::failure(definition.status());
32 auto changed = function(definition.value());
33 if (!changed.ok()) return changed;
34 return self->replace(definition.value());
35}
