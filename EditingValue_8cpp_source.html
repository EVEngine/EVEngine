42bool checkValue(const Value& value, size_t depth, size_t maxDepth, size_t maxElements, size_t maxStringBytes,
43 ValueBudget& budget) {
44 if (depth > maxDepth) return false;
45 return std::visit(
46 [&](const auto& current) -> bool {
47 using T = std::decay_t<decltype(current)>;
48 if constexpr (std::is_same_v<T, std::string>) {
49 budget.stringBytes += current.size();
50 return budget.stringBytes <= maxStringBytes;
51 } else if constexpr (std::is_same_v<T, Value::Array>) {
52 budget.elements += current.size();
53 if (budget.elements > maxElements) return false;
54 for (const auto& entry : current)
55 if (!checkValue(entry, depth + 1, maxDepth, maxElements, maxStringBytes, budget)) return false;
56 return true;
57 } else if constexpr (std::is_same_v<T, Value::Object>) {
58 budget.elements += current.size();
59 if (budget.elements > maxElements) return false;
60 for (const auto& [key, entry] : current) {
61 budget.stringBytes += key.size();
62 if (budget.stringBytes > maxStringBytes ||
63 !checkValue(entry, depth + 1, maxDepth, maxElements, maxStringBytes, budget))
64 return false;
65 }
66 return true;
67 } else {
68 return true;
69 }
70 },
71 value.storage());
72}
