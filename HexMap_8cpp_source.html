26Result<void> HexMap::reset(std::int32_t cellCountX, std::int32_t cellCountZ, std::uint32_t seed) {
27 if (cellCountX <= 0 || cellCountZ <= 0) return invalidArgument("hex map size must be positive");
28 if (cellCountX % HexMetrics::kChunkSizeX != 0 || cellCountZ % HexMetrics::kChunkSizeZ != 0)
29 return invalidArgument("hex map size must be a multiple of the 5x5 chunk size");
30
31 cellCountX_ = cellCountX;
32 cellCountZ_ = cellCountZ;
33 chunkCountX_ = cellCountX / HexMetrics::kChunkSizeX;
34 chunkCountZ_ = cellCountZ / HexMetrics::kChunkSizeZ;
35 // `HexValues` stores the elevation biased by +15, so a zeroed record would read
36 // back as -15. Every cell therefore starts explicitly at elevation 0 and water 0.
37 // Every cell also starts *explorable but unexplored*, which is the fog-of-war
38 // precondition of the reference project: a viewer may reveal it, nothing has yet.
39 HexCellData initial{};
40 initial.values = initial.values.withElevation(0).withWaterLevel(0);
41 initial.flags = initial.flags.withExplorable(true).withExplored(false);
42 cells_.assign(static_cast<std::size_t>(cellCountX_) * static_cast<std::size_t>(cellCountZ_), initial);
43 chunkDirty_.assign(static_cast<std::size_t>(chunkCount()), 0u);
44 dirtyQueue_.clear();
45 noise_.reset(seed);
46 revision_ = 0;
47 markAllChunksDirty();
48 return Result<void>::success();
49}

125void HexMap::markChunkDirtyAndNeighbors(std::int32_t chunkIndex) noexcept {
126 if (chunkIndex < 0 || chunkIndex >= chunkCount()) return;
127 markChunkDirty(chunkIndex);
128 const std::int32_t column = chunkColumnOf(chunkIndex);
129 const std::int32_t row = chunkRowOf(chunkIndex);
130 for (std::int32_t dz = -1; dz <= 1; ++dz) {
131 for (std::int32_t dx = -1; dx <= 1; ++dx) {
132 if (dx == 0 && dz == 0) continue;
133 const std::int32_t cx = column + dx;
134 const std::int32_t cz = row + dz;
135 if (cx < 0 || cx >= chunkCountX_ || cz < 0 || cz >= chunkCountZ_) continue;
136 markChunkDirty(cx + cz * chunkCountX_);
137 }
138 }

166Result<HexCoordinates> HexMap::pickCell(HexVec3 rayOrigin, HexVec3 rayDirection) const noexcept {
167 if (empty())
168 return Result<HexCoordinates>::failure(
169 Diagnostic::error(DiagnosticCode::NotFound, "hex map is empty", "hexmap.pick"));
170
171 const float dirLength =
172 std::sqrt(rayDirection.x * rayDirection.x + rayDirection.y * rayDirection.y + rayDirection.z * rayDirection.z);
173 if (dirLength <= 1e-6f)
174 return Result<HexCoordinates>::failure(
175 Diagnostic::error(DiagnosticCode::InvalidArgument, "ray direction must be non-zero", "hexmap.pick"));
176 rayDirection.x /= dirLength;
177 rayDirection.y /= dirLength;
178 rayDirection.z /= dirLength;
179
180 // Bounding box of the whole map, padded by one cell and one elevation step.
181 HexVec3 low = cellGroundPosition(HexCoordinates::fromOffset(0, 0));
182 HexVec3 high = cellGroundPosition(HexCoordinates::fromOffset(cellCountX_ - 1, cellCountZ_ - 1));
183 const float pad = HexMetrics::kOuterRadius;
184 low.x -= pad;
185 low.z -= pad;
186 low.y = HexMetrics::elevationY(HexMetrics::kMinElevation) - HexMetrics::kElevationStep;
187 high.x += pad;
188 high.z += pad;
189 high.y = HexMetrics::elevationY(HexMetrics::kMaxElevation) + HexMetrics::kElevationStep;
190
191 float tEnter = 0.f;
192 float tExit = std::numeric_limits<float>::max();
193 const float origin[3] = {rayOrigin.x, rayOrigin.y, rayOrigin.z};
194 const float direction[3] = {rayDirection.x, rayDirection.y, rayDirection.z};
195 const float boxLow[3] = {low.x, low.y, low.z};
196 const float boxHigh[3] = {high.x, high.y, high.z};
197 for (int axis = 0; axis < 3; ++axis) {
198 if (std::fabs(direction[axis]) < 1e-8f) {
199 if (origin[axis] < boxLow[axis] || origin[axis] > boxHigh[axis]) {
200 return Result<HexCoordinates>::failure(
201 Diagnostic::error(DiagnosticCode::NotFound, "ray misses the hex map bounds", "hexmap.pick"));
202 }
203 continue;
204 }
205 const float inv = 1.f / direction[axis];
206 float t0 = (boxLow[axis] - origin[axis]) * inv;
207 float t1 = (boxHigh[axis] - origin[axis]) * inv;
208 if (t0 > t1) std::swap(t0, t1);
209 tEnter = std::max(tEnter, t0);
210 tExit = std::min(tExit, t1);
211 }
212 if (tExit < tEnter)
213 return Result<HexCoordinates>::failure(
214 Diagnostic::error(DiagnosticCode::NotFound, "ray misses the hex map bounds", "hexmap.pick"));
215
216 const float step = HexMetrics::kElevationStep * 0.5f;
217 const float maxRayLength = tExit;
218 const int maxSteps = 512;
219 float previousT = tEnter;
220 HexCoordinates previousCoordinates{};
221 bool havePrevious = false;
222
223 for (int i = 0; i <= maxSteps; ++i) {
224 const float t = std::min(tEnter + static_cast<float>(i) * step, maxRayLength);
225 const HexVec3 point{rayOrigin.x + rayDirection.x * t, rayOrigin.y + rayDirection.y * t,
226 rayOrigin.z + rayDirection.z * t};
227 const HexCoordinates coordinates = HexCoordinates::fromWorldPosition(point);
228 if (contains(coordinates)) {
229 const float surfaceY = cellPosition(coordinates).y;
230 if (point.y <= surfaceY) {
231 // Bisect between the last sample above the surface and this one.
232 float lo = havePrevious ? previousT : tEnter;
233 float hi = t;
234 HexCoordinates best = contains(previousCoordinates) ? previousCoordinates : coordinates;
235 for (int iteration = 0; iteration < 8; ++iteration) {
236 const float mid = (lo + hi) * 0.5f;
237 const HexVec3 sample{rayOrigin.x + rayDirection.x * mid, rayOrigin.y + rayDirection.y * mid,
238 rayOrigin.z + rayDirection.z * mid};
239 const HexCoordinates candidate = HexCoordinates::fromWorldPosition(sample);
240 if (!contains(candidate)) break;
241 best = candidate;
242 if (sample.y <= cellPosition(candidate).y)
243 hi = mid;
244 else
245 lo = mid;
246 }
247 if (contains(best)) return Result<HexCoordinates>::success(best);
248 return Result<HexCoordinates>::success(coordinates);
249 }
250 previousCoordinates = coordinates;
251 havePrevious = true;
252 }
253 previousT = t;
254 if (t >= maxRayLength) break;
255 }
256
257 return Result<HexCoordinates>::failure(
258 Diagnostic::error(DiagnosticCode::NotFound, "ray never reached the hex surface", "hexmap.pick"));
259}

335void HexMap::validateRivers(HexCoordinates coordinates) noexcept {
336 const HexCellData* data = cell(coordinates);
337 if (!data || !data->flags.hasRiver()) return;
338 // A river may only leave a cell that is at least as high as its neighbour,
339 // unless the cell itself is a lake at exactly its own elevation.
340 const std::int32_t fromElevation = elevation(coordinates);
341 const std::int32_t fromWater = waterLevel(coordinates);
342 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
343 const HexDirection direction = static_cast<HexDirection>(i);
344 if (!data->flags.hasRiverOut(direction)) continue;
345 HexCoordinates neighbour{};
346 const bool canFlow = getNeighbor(coordinates, direction, neighbour) &&
347 (fromElevation >= elevation(neighbour) || fromWater == fromElevation);
348 if (!canFlow) {
349 removeRiver(coordinates).ignore("river validation must keep the map consistent");
350 return;
351 }
352 }
353}

355Result<void> HexMap::setElevation(HexCoordinates c, std::int32_t value) {
356 HexCellData* data = mutableCell(c);
357 if (!data) return invalidArgument("cell is outside the hex map");
358 const std::int32_t clamped = clampInt(value, HexMetrics::kMinElevation, HexMetrics::kMaxElevation);
359 if (data->values.elevation() == clamped) return Result<void>::success();
360 data->values = data->values.withElevation(clamped);
361
362 // Roads cannot span a difference of more than one elevation step.
363 bool hasInvalidRoad = false;
364 for (std::int32_t i = 0; i < kHexDirectionCount && !hasInvalidRoad; ++i) {
365 const HexDirection direction = static_cast<HexDirection>(i);
366 if (!data->flags.hasRoad(direction)) continue;
367 HexCoordinates neighbour{};
368 if (!getNeighbor(c, direction, neighbour)) continue;
369 const std::int32_t delta = elevation(neighbour) - clamped;
370 if (delta > 1 || delta < -1) hasInvalidRoad = true;
371 }
372 if (hasInvalidRoad) removeRoads(c).ignore("elevation change must drop invalid roads");
373 refreshCellDependents(c);
374 validateRivers(c);
375 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
376 HexCoordinates neighbour{};
377 if (getNeighbor(c, static_cast<HexDirection>(i), neighbour)) validateRivers(neighbour);
378 }
379 return Result<void>::success();
380}

470Result<void> HexMap::setOutgoingRiver(HexCoordinates c, HexDirection direction) {
471 HexCellData* data = mutableCell(c);
472 if (!data) return invalidArgument("cell is outside the hex map");
473 HexCoordinates neighbour{};
474 if (!getNeighbor(c, direction, neighbour)) return invalidArgument("river direction leaves the hex map");
475 if (data->flags.hasAnyRiverOut() && data->flags.hasRiverOut(direction)) return Result<void>::success();
476 if (elevation(c) < elevation(neighbour) && waterLevel(c) != elevation(c))
477 return invalidArgument("a river cannot flow uphill");
478
479 // Drop only the links this call replaces, mirroring the reference cell's
480 // `RemoveOutgoingRiver` + same-edge `RemoveIncomingRiver`. Calling the full
481 // `removeRiver` here also erased the *incoming* link from the upstream cell, so
482 // carving a channel step by step wiped the step behind it and only the last
483 // edge survived; that is why a 40-cell walk produced a 2-cell river.
484 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
485 const HexDirection previous = static_cast<HexDirection>(i);
486 if (!data->flags.hasRiverOut(previous)) continue;
487 data->flags = data->flags.withoutRiverOut(previous);
488 HexCoordinates previousNeighbour{};
489 if (getNeighbor(c, previous, previousNeighbour)) {
490 HexCellData* previousData = mutableCell(previousNeighbour);
491 previousData->flags = previousData->flags.withoutRiverIn(opposite(previous));
492 refreshCellDependents(previousNeighbour);
493 }
494 }
495 data = mutableCell(c);
496 // A river may not leave and enter through the same edge.
497 if (data->flags.hasRiverIn(direction)) {
498 data->flags = data->flags.withoutRiverIn(direction);
499 HexCellData* otherData = mutableCell(neighbour);
500 otherData->flags = otherData->flags.withoutRiverOut(opposite(direction));
501 }
502
503 data = mutableCell(c);
504 data->flags = data->flags.withRiverOut(direction).withoutRoad(direction);
505 data->values = data->values.withSpecialIndex(0);
506 HexCellData* other = mutableCell(neighbour);
507 other->flags = other->flags.withRiverIn(opposite(direction)).withoutRoad(opposite(direction));
508 other->values = other->values.withSpecialIndex(0);
509 refreshCellDependents(c);
510 refreshCellDependents(neighbour);
511 return Result<void>::success();
512}

514Result<void> HexMap::removeRiver(HexCoordinates c) {
515 HexCellData* data = mutableCell(c);
516 if (!data) return invalidArgument("cell is outside the hex map");
517 if (!data->flags.hasRiver()) return Result<void>::success();
518 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
519 const HexDirection direction = static_cast<HexDirection>(i);
520 data->flags = data->flags.withoutRiverIn(direction).withoutRiverOut(direction);
521 }
522 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
523 const HexDirection direction = static_cast<HexDirection>(i);
524 HexCoordinates neighbour{};
525 if (!getNeighbor(c, direction, neighbour)) continue;
526 HexCellData* other = mutableCell(neighbour);
527 other->flags = other->flags.withoutRiverIn(opposite(direction)).withoutRiverOut(opposite(direction));
528 refreshCellDependents(neighbour);
529 }
530 refreshCellDependents(c);
531 return Result<void>::success();
532}
533
534Result<void> HexMap::addRoad(HexCoordinates c, HexDirection direction) {
535 HexCellData* data = mutableCell(c);
536 if (!data) return invalidArgument("cell is outside the hex map");
537 HexCoordinates neighbour{};
538 if (!getNeighbor(c, direction, neighbour)) return invalidArgument("road direction leaves the hex map");
539 if (data->flags.hasRoad(direction)) return Result<void>::success();
540 if (data->flags.hasRiverThrough(direction)) return Result<void>::success();
541 if (data->flags.hasRiver() && !data->flags.hasRiverBeginOrEnd()) {
542 // A road may only join a river at a begin or end cell.
543 return Result<void>::success();
544 }
545 HexCellData* other = mutableCell(neighbour);
546 if (other->values.specialIndex() != 0 || data->values.specialIndex() != 0) return Result<void>::success();
547 if (other->flags.hasRiverThrough(opposite(direction))) return Result<void>::success();
548 const std::int32_t delta = elevation(neighbour) - elevation(c);
549 if (delta > 1 || delta < -1) return Result<void>::success();
550
551 data->flags = data->flags.withRoad(direction);
552 other->flags = other->flags.withRoad(opposite(direction));
553 refreshCellDependents(c);
554 refreshCellDependents(neighbour);
555 return Result<void>::success();
556}
557
558Result<void> HexMap::removeRoads(HexCoordinates c) {
559 HexCellData* data = mutableCell(c);
560 if (!data) return invalidArgument("cell is outside the hex map");
561 if (!data->flags.hasRoad()) return Result<void>::success();
562 data->flags = data->flags.without(HexFlags::roadMask());
563 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
564 const HexDirection direction = static_cast<HexDirection>(i);
565 HexCoordinates neighbour{};
566 if (!getNeighbor(c, direction, neighbour)) continue;
567 HexCellData* other = mutableCell(neighbour);
568 other->flags = other->flags.withoutRoad(opposite(direction));
569 refreshCellDependents(neighbour);
570 }
571 refreshCellDependents(c);
572 return Result<void>::success();
573}

577void HexMap::collectBrush(HexCoordinates center, std::int32_t radius, std::vector<std::int32_t>& out) const {
578 out.clear();
579 if (!contains(center) || radius < 0) return;
580 // The brush reaches `radius` steps, so its offset-space footprint is at most
581 // `2 * radius` columns wide and `radius` rows tall.
582 const std::int32_t centerColumn = center.offsetX();
583 const std::int32_t minColumn = std::max(0, centerColumn - radius * 2);
584 const std::int32_t maxColumn = std::min(cellCountX_ - 1, centerColumn + radius * 2);
585 const std::int32_t minZ = std::max(0, center.z - radius);
586 const std::int32_t maxZ = std::min(cellCountZ_ - 1, center.z + radius);
587 for (std::int32_t z = minZ; z <= maxZ; ++z) {
588 for (std::int32_t column = minColumn; column <= maxColumn; ++column) {
589 const HexCoordinates candidate = HexCoordinates::fromOffset(column, z);
590 if (center.distanceTo(candidate) <= radius) out.push_back(column + z * cellCountX_);
591 }
592 }
593}

632Result<void> HexMap::editFeatureLevel(HexCoordinates center, std::int32_t radius, std::int32_t feature,
633 std::int32_t delta) {
634 if (!contains(center)) return invalidArgument("brush centre is outside the hex map");
635 if (radius < 0) return invalidArgument("brush radius must be non-negative");
636 std::vector<std::int32_t> cells;
637 collectBrush(center, radius, cells);
638 const std::int32_t layer = layerIndex(feature);
639 for (std::int32_t index : cells) {
640 const HexCoordinates coordinates = coordinatesAt(index);
641 const HexCellData* data = cellAt(index);
642 if (!data || data->flags.hasRiver()) continue;
643 if (data->values.specialIndex() != 0) continue;
644 const std::int32_t current = layer == 0 ? data->values.urbanLevel()
645 : layer == 1 ? data->values.farmLevel()
646 : data->values.plantLevel();
647 const std::int32_t next = clampInt(current + delta, 0, 3);
648 if (layer == 0)
649 setUrbanLevel(coordinates, next).ignore("brush urban edit clamps per cell");
650 else if (layer == 1)
651 setFarmLevel(coordinates, next).ignore("brush farm edit clamps per cell");
652 else
653 setPlantLevel(coordinates, next).ignore("brush plant edit clamps per cell");
654 }
655 return Result<void>::success();
656}
