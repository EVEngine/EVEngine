20Quat multiply(const Quat& a, const Quat& b) {
21 return {a.w * b.x + a.x * b.w + a.y * b.z - a.z * b.y,

43Result<void> UpperBodyOrientationCompensator::configure(const UpperBodyOrientationProfile& profile) {
44 const int root = skeleton_.findBone(profile.rootBone);
45 const int hips = skeleton_.findBone(profile.hipsBone);
46 const int spine = skeleton_.findBone(profile.spineBone);
47 if (root < 0) return profileError("upper-body orientation root bone was not found", "profile.rootBone");
48 if (hips < 0) return profileError("upper-body orientation hips bone was not found", "profile.hipsBone");
49 if (spine < 0) return profileError("upper-body orientation spine bone was not found", "profile.spineBone");
50 if (!isDescendant(skeleton_, hips, root))
51 return profileError("upper-body orientation hips bone must descend from root", "profile.hipsBone");
52 if (skeleton_.getParent(spine) != hips)
53 return profileError("upper-body orientation spine bone must be a direct child of hips", "profile.spineBone");
54
55 rootBone_ = root;
56 hipsBone_ = hips;
57 spineBone_ = spine;
58 configured_ = true;
59 return Result<void>::success(Status::success(StatusCode::Applied));
60}

67Result<void> UpperBodyOrientationCompensator::apply(AnimPose& pose, AnimPose& upperBodyPose, float weight) const {
68 if (!configured_)
69 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Unsupported,
70 "upper-body orientation profile is not configured",
71 "profile"));
72 if (pose.getBoneCount() != skeleton_.getBoneCount() ||
73 upperBodyPose.getBoneCount() != skeleton_.getBoneCount())
74 return profileError("upper-body orientation poses must match the configured skeleton", "pose");
75 if (!std::isfinite(weight) || weight < 0.0f || weight > 1.0f)
76 return profileError("upper-body orientation weight must be finite and within [0,1]", "weight");
77 if (weight <= 0.0f) return Result<void>::success(Status::success(StatusCode::NoOp));
78
79 upperBodyPose.computeWorld(&skeleton_);
80 pose.computeWorld(&skeleton_);
81 const Quat targetInRoot = multiply(inverse(rotation(upperBodyPose.world(rootBone_))),
82 rotation(upperBodyPose.world(spineBone_)));
83 const Quat targetWorld = multiply(rotation(pose.world(rootBone_)), targetInRoot);
84 const Quat targetLocal = multiply(inverse(rotation(pose.world(hipsBone_))), targetWorld);
85
86 TransformTRS& spine = pose.local(spineBone_);
87 slerpQuat(spine.qx, spine.qy, spine.qz, spine.qw, targetLocal.x, targetLocal.y, targetLocal.z, targetLocal.w,
88 weight, spine.qx, spine.qy, spine.qz, spine.qw);
89 spine.normalizeRotation();
90 pose.computeWorld(&skeleton_);
91 return Result<void>::success(Status::success(StatusCode::Applied));
92}
