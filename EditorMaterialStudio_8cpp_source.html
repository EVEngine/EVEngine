35EditorResult<void> MaterialStudioController::beginInteraction(PropertyPath path) {
36 if (activeProperty_ || transactions_.active())
37 return studioError<void>(EditorStatus::Conflict, "editor.material.studio-interaction-active",
38 "Finish or cancel the active material edit first");
39 const auto selection = selectionFor(target_.authoringTarget());
40 const auto descriptor = target_.authoringTarget().schema(selection).find(path);
41 if (!descriptor)
42 return studioError<void>(EditorStatus::NotFound, "editor.material.studio-property",
43 "Material property is not present in the schema: " + path.value());
44 draft_ = std::make_unique<MaterialDocumentTarget>(target_.authoringTarget());
45 activeProperty_ = std::move(path);
46 finalValue_.reset();
47 diagnostics_.clear();
48 return eve::editing::applied<void>();
49}

51EditorResult<void> MaterialStudioController::updateInteraction(EditorValue value) {
52 if (!draft_ || !activeProperty_)
53 return studioError<void>(EditorStatus::Rejected, "editor.material.studio-no-interaction",
54 "Begin a material interaction before updating it");
55 auto operation = draft_->makeSet(selectionFor(*draft_), *activeProperty_, value, PropertySetMode::Absolute);
56 if (!operation.ok()) {
57 diagnostics_ = operation.diagnostics();
58 return EditorResult<void>::failure(operation.status());
59 }
60 auto applied = draft_->applyDomainOperation(operation.value());
61 if (!applied.ok()) {
62 diagnostics_ = applied.diagnostics();
63 return applied;
64 }
65 finalValue_ = std::move(value);
66 diagnostics_ = draft_->validate();
67 previewDirty_ = true;
68 return eve::editing::applied<void>();
69}

71EditorResult<TransactionReceipt> MaterialStudioController::commitInteraction() {
72 if (!draft_ || !activeProperty_ || !finalValue_)
73 return studioError<TransactionReceipt>(EditorStatus::Rejected, "editor.material.studio-no-final-value",
74 "The material interaction has no value to commit");
75 auto operation = target_.authoringTarget().makeSet(selectionFor(target_.authoringTarget()), *activeProperty_,
76 *finalValue_, PropertySetMode::Absolute);
77 if (!operation.ok())
78 return EditorResult<TransactionReceipt>::failure(operation.status());
79 TransactionSpec specification;
80 specification.id = TransactionId("material-studio-" + std::to_string(++transactionSequence_));
81 specification.label = "Edit material " + activeProperty_->value();
82 specification.target = TargetId(target_.targetId());
83 specification.baseRevision = target_.revision();
84 specification.mergeKey = operation.value().mergeKey;
85 auto begun = transactions_.begin(std::move(specification));
86 if (!begun.ok())
87 return studioError<TransactionReceipt>(begun.code(), "editor.material.studio-begin",
88 "Could not begin the material transaction");
89 auto appended = transactions_.append(std::move(operation).takeValue());
90 if (!appended.ok()) {
91 [[maybe_unused]] auto discarded = transactions_.discard();
92 return studioError<TransactionReceipt>(appended.code(), "editor.material.studio-append",
93 "Could not append the material edit");
94 }
95 auto previewed = transactions_.preview();
96 if (!previewed.ok()) {
97 [[maybe_unused]] auto discarded = transactions_.discard();
98 return studioError<TransactionReceipt>(previewed.code(), "editor.material.studio-preflight",
99 "Material transaction preflight failed");
100 }
101 auto committed = transactions_.commit();
102 if (!committed.ok()) return committed;
103 clearInteraction();
104 previewDirty_ = true;
105 return committed;
106}

116EditorResult<void> MaterialStudioController::tick(std::uint64_t monotonicMilliseconds) {
117 if (!previewDirty_) return noStudioChange();
118 if (hasPreviewTimestamp_ && monotonicMilliseconds < lastPreviewMilliseconds_)
119 return studioError<void>(EditorStatus::Rejected, "editor.material.studio-time-regressed",
120 "Material preview time must be monotonic");
121 if (hasPreviewTimestamp_ && monotonicMilliseconds - lastPreviewMilliseconds_ < previewIntervalMilliseconds_)
122 return noStudioChange();
123 auto rendered = refreshPreview();
124 if (rendered.ok()) {
125 lastPreviewMilliseconds_ = monotonicMilliseconds;
126 hasPreviewTimestamp_ = true;
127 }
128 return rendered;
129}

168EditorResult<void> MaterialStudioController::renderPreview(const MaterialDocumentTarget& material) {
169 auto task = previews_.render(document_, material, settings_, renderer_);
170 if (!task.ok()) {
171 diagnostics_ = task.diagnostics();
172 return studioError<void>(task.code(), "editor.material.studio-preview",
173 "Material preview request was rejected");
174 }
175 auto result = previews_.result(task.value());
176 if (!result.ok()) {
177 diagnostics_ = result.diagnostics();
178 return studioError<void>(result.code(), "editor.material.studio-preview-result",
179 "Material preview result is unavailable");
180 }
181 auto published = previews_.publish(document_, material.revision(), task.value());
182 diagnostics_ = result.value().diagnostics;
183 if (!published.ok()) {
184 diagnostics_.insert(diagnostics_.end(), published.diagnostics().begin(),
185 published.diagnostics().end());
186 return published;
187 }
188 previewArtifact_ = result.value().artifact;
189 previewRevision_ = material.revision();
190 previewDirty_ = false;
191 return eve::editing::applied<void>();
192}
