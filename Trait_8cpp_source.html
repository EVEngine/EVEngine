46int TraitRegistry::loadFromJson(const std::string &json, std::string *error) {
47 std::string err;
48 const eve::json::Document doc = eve::json::Document::parse(json, &err);
49 if (!doc.valid()) {
50 if (error) *error = err.empty() ? "invalid json" : err;
51 return 0;
52 }
53 const Value root = doc.root();
54 int n = 0;
55 const auto parseOne = [&](Value o) {
56 if (!o.isObject()) return;
57 TraitDefinition def;
58 def.id = o.getString("id");
59 if (def.id.empty()) return;
60 def.tags = o.getStringArray("tags");
61 def.extra = o.getStringMap("extra");
62 const Value traits = o.get("traits");
63 for (size_t i = 0; i < traits.size(); ++i) {
64 const Value t = traits.at(i);
65 TraitSpec spec;
66 spec.kind = t.getString("kind");
67 spec.target = t.getString("target");
68 spec.value = t.getDouble("value", 0.0);
69 if (spec.kind.empty()) continue;
70 def.traits.push_back(std::move(spec));
71 }
72 traitTable()[def.id] = std::move(def);
73 ++n;
74 };
75 if (root.isArray()) {
76 for (size_t i = 0; i < root.size(); ++i) parseOne(root.at(i));
77 } else if (root.isObject()) {
78 parseOne(root);
79 }
80 return n;
81}

96int TraitSystem::apply(RPGActor *actor, const std::string &traitId, const std::string &source) {
97 if (!actor) return 0;
98 const TraitDefinition *def = findDef(traitId);
99 if (!def) return 0;
100 auto &ts = *actor->traits();
101 RPGActor::TraitInstance inst;
102 inst.instanceId = ts.nextInstanceId++;
103 inst.traitId = traitId;
104 inst.source = source;
105 // paramRate → 属性修改器（mulAdd 倍率），source="trait:<instanceId>"
106 for (const auto &spec : def->traits) {
107 if (spec.kind != "paramRate" || spec.target.empty()) continue;
108 AttributeSystem::addModifier(actor, spec.target, "trait:" + std::to_string(inst.instanceId), "mulAdd",
109 spec.value - 1.0, 0);
110 }
111 ts.active.push_back(std::move(inst));
112 return inst.instanceId;
113}
