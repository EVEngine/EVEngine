74template <class Pointer>
75SQInteger invokePointer(HSQUIRRELVM vm) {
76 try {
77 void* data = nullptr;
78 if (SQ_FAILED(sq_getuserdata(vm, -1, &data, nullptr)) || !data)
79 return sq_throwerror(vm, "Missing direct binding userdata");
80 Pointer pointer;
81 // Member pointers may be larger than void*. Copy their complete object
82 // representation, without imposing alignment/lifetime assumptions on VM memory.
83 std::memcpy(&pointer, data, sizeof(pointer));
84 using R = typename Call<Pointer>::Result;
85 if constexpr (std::is_void_v<R>) {
86 Call<Pointer>::invoke(vm, pointer);
87 return 0;
88 } else {
89 PushResult<R>::push(vm, Call<Pointer>::invoke(vm, pointer));
90 return 1;
91 }
92 } catch (const std::exception& e) {
93 return sq_throwerror(vm, e.what());
94 }
95}
