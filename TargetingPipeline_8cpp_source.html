20bool insideCone(float px, float py, float apexX, float apexY, float dirX, float dirY, float halfAngle,
21 float range) noexcept {
22 const float dx = px - apexX;
23 const float dy = py - apexY;
24 const float distSq = dx * dx + dy * dy;
25 if (distSq > range * range) return false;
26 if (distSq == 0.f) return true;
27 const float facingLen = std::hypot(dirX, dirY);
28 if (!(facingLen > 0.f)) return false;
29 const float invDist = 1.f / std::sqrt(distSq);
30 const float nx = dx * invDist;
31 const float ny = dy * invDist;
32 const float fx = dirX / facingLen;
33 const float fy = dirY / facingLen;
34 const float dot = std::clamp(nx * fx + ny * fy, -1.f, 1.f);
35 return std::acos(dot) <= halfAngle;
36}

38[[nodiscard]] bool matchesFactFilters(const Subject& subject, float distance, const QuerySpec& spec) {
39 for (const auto& tag : spec.requiredTags)
40 if (!subject.tags.count(tag)) return false;
41 for (const auto& tag : spec.excludedTags)
42 if (subject.tags.count(tag)) return false;
43 if (!spec.includeFactions.empty()) {
44 bool included = false;
45 for (const auto& faction : spec.includeFactions)
46 if (subject.faction == faction) {
47 included = true;
48 break;
49 }
50 if (!included) return false;
51 }
52 for (const auto& faction : spec.excludeFactions)
53 if (subject.faction == faction) return false;
54 if (spec.visibleTo && !subject.visibleTo.count(*spec.visibleTo)) return false;
55 if (distance < spec.minRange || distance > spec.maxRange) return false;
56 return true;
57}

95 std::vector<RankedCandidate> kept;
96 kept.reserve(inout.size());

169 std::sort(inout.begin(), inout.end(), [](const RankedCandidate& a, const RankedCandidate& b) {

171 return a.id < b.id;

242Result<void> TargetingPipeline::registerTask(std::unique_ptr<ITargetingTask> task) {
243 if (!task) {
244 return pipelineFailure<void>(DiagnosticCode::InvalidArgument, "targeting task must not be null", "task");
245 }
246 const std::string id{task->id()};
247 if (id.empty()) {
248 return pipelineFailure<void>(DiagnosticCode::InvalidArgument, "targeting task id must not be empty", "task.id");
249 }
250 if (tasks_.count(id) != 0) {
251 return pipelineFailure<void>(DiagnosticCode::AlreadyExists, "targeting task id already registered", "task.id");
252 }
253 tasks_.emplace(id, std::move(task));
254 return Result<void>::success(Status::success(StatusCode::Applied));
255}

257Result<void> TargetingPipeline::registerPreset(TargetingPreset preset) {
258 if (preset.id.empty()) {
259 return pipelineFailure<void>(DiagnosticCode::InvalidArgument, "targeting preset id must not be empty",
260 "preset.id");
261 }
262 if (preset.steps.empty()) {
263 return pipelineFailure<void>(DiagnosticCode::InvalidArgument, "targeting preset must contain at least one step",
264 "preset.steps");
265 }
266 for (const auto& step : preset.steps) {
267 if (step.taskId.empty() || tasks_.count(step.taskId) == 0) {
268 return pipelineFailure<void>(DiagnosticCode::NotFound, "targeting preset references unknown task id",
269 "preset.steps.taskId");
270 }
271 }
272 presets_.insert_or_assign(preset.id, std::move(preset));
273 return Result<void>::success(Status::success(StatusCode::Applied));
274}

280Result<CandidateQueryResult> TargetingPipeline::execute(TargetingSourceContext& context,
281 const TargetingPreset& preset) const {
282 if (context.world == nullptr) {
283 return pipelineFailure<CandidateQueryResult>(DiagnosticCode::InvalidArgument,
284 "TargetingPipeline requires a SensingWorld", "pipeline.world");
285 }
286 if (!finite(context.origin.x) || !finite(context.origin.y)) {
287 return pipelineFailure<CandidateQueryResult>(DiagnosticCode::InvalidArgument,
288 "TargetingSourceContext origin must be finite", "pipeline.origin");
289 }
290 if (preset.steps.empty()) {
291 return pipelineFailure<CandidateQueryResult>(DiagnosticCode::InvalidArgument,
292 "TargetingPreset must contain at least one step", "preset.steps");
293 }
294
295 std::vector<RankedCandidate> working;
296 for (const auto& step : preset.steps) {
297 const auto found = tasks_.find(step.taskId);
298 if (found == tasks_.end() || !found->second) {
299 return pipelineFailure<CandidateQueryResult>(DiagnosticCode::NotFound,
300 "TargetingPreset step task is not registered",
301 "preset.steps.taskId");
302 }
303 auto ran = found->second->execute(context, working, step);
304 if (!ran) return Result<CandidateQueryResult>::failure(ran.status());
305 }
306 return Result<CandidateQueryResult>::success(CandidateQueryResult{std::move(working)});
307}
