125EditorResult<HouseKitComponentValue> parseComponent(const EditorValue& value) {
126 HouseKitComponentValue out;
127 const auto *editorId = field(value, "editorId"), *id = field(value, "id"), *model = field(value, "model"),
128 *category = field(value, "category");
129 const auto* editorIds = editorId ? editorId->getIf<std::string>() : nullptr;
130 const auto* ids = id ? id->getIf<std::string>() : nullptr;
131 const auto* models = model ? model->getIf<std::string>() : nullptr;
132 const auto* categories = category ? category->getIf<std::string>() : nullptr;
133 if (!editorIds || editorIds->empty() || !ids || !models || !categories)
134 return fail<HouseKitComponentValue>(EditorStatus::Rejected, "editor.housegen.component",
135 "House component identity is invalid");
136 out.id = ObjectId(*editorIds);
137 out.component.id = *ids;
138 out.component.modelPath = *models;
139 out.component.category = *categories;
140 int* values[] = {&out.component.width, &out.component.depth, &out.component.height, &out.component.weight};
141 const char* fields[] = {"width", "depth", "height", "weight"};
142 for (int i = 0; i < 4; ++i) {
143 const auto* v = field(value, fields[i]);
144 const auto* n = v ? v->getIf<int64_t>() : nullptr;
145 if (!n)
146 return fail<HouseKitComponentValue>(EditorStatus::Rejected, "editor.housegen.dimension",
147 "House component dimension is invalid");
148 *values[i] = static_cast<int>(*n);
149 }
150 if (!readIntegers(field(value, "rotations"), out.component.rotations) ||
151 !readStrings(field(value, "tags"), out.component.tags))
152 return fail<HouseKitComponentValue>(EditorStatus::Rejected, "editor.housegen.arrays",
153 "House component rotations or tags are invalid");
154 const auto* socketValues = field(value, "sockets");
155 const auto* socketArray = socketValues ? socketValues->getIf<EditorValue::Array>() : nullptr;
156 if (!socketArray)
157 return fail<HouseKitComponentValue>(EditorStatus::Rejected, "editor.housegen.sockets",
158 "House sockets are invalid");
159 for (const auto& socketValue : *socketArray) {
160 housegen::HouseSocket socket;
161 const auto * d = field(socketValue, "direction"), *t = field(socketValue, "type");
162 const auto* ds = d ? d->getIf<std::string>() : nullptr;
163 const auto* ts = t ? t->getIf<std::string>() : nullptr;
164 if (!ds || !ts || !direction(*ds, socket.direction) ||
165 !readStrings(field(socketValue, "accepts"), socket.accepts))
166 return fail<HouseKitComponentValue>(EditorStatus::Rejected, "editor.housegen.socket",
167 "House socket is invalid");
168 socket.type = *ts;
169 out.component.sockets.push_back(std::move(socket));
170 }
171 const auto* material = field(value, "material");
172 if (!material)
173 return fail<HouseKitComponentValue>(EditorStatus::Rejected, "editor.housegen.material",
174 "House material is missing");
175 auto& m = out.component.material;
176 const auto *hbc = field(*material, "hasBaseColor"), *hm = field(*material, "hasMetallic"),
177 *hr = field(*material, "hasRoughness");
178 const auto* hbcb = hbc ? hbc->getIf<bool>() : nullptr;
179 const auto* hmb = hm ? hm->getIf<bool>() : nullptr;
180 const auto* hrb = hr ? hr->getIf<bool>() : nullptr;
181 const auto* colorValue = field(*material, "baseColor");
182 const auto* color = colorValue ? colorValue->getIf<EditorValue::Array>() : nullptr;
183 if (!hbcb || !hmb || !hrb || !color || color->size() != 4)
184 return fail<HouseKitComponentValue>(EditorStatus::Rejected, "editor.housegen.material",
185 "House material flags or color are invalid");
186 m.hasBaseColor = *hbcb;
187 m.hasMetallic = *hmb;
188 m.hasRoughness = *hrb;
189 float* colors[] = {&m.baseColorR, &m.baseColorG, &m.baseColorB, &m.baseColorA};
190 for (int i = 0; i < 4; ++i)
191 if (!number(&(*color)[i], *colors[i]))
192 return fail<HouseKitComponentValue>(EditorStatus::Rejected, "editor.housegen.color",
193 "House material color is invalid");
194 std::string* paths[] = {&m.baseColorTexture, &m.normalTexture, &m.heightTexture};
195 const char* pathFields[] = {"baseColorTexture", "normalTexture", "heightTexture"};
196 for (int i = 0; i < 3; ++i) {
197 const auto* p = field(*material, pathFields[i]);
198 const auto* s = p ? p->getIf<std::string>() : nullptr;
199 if (!s)
200 return fail<HouseKitComponentValue>(EditorStatus::Rejected, "editor.housegen.texture",
201 "House material texture is invalid");
202 *paths[i] = *s;
203 }
204 float* scalars[] = {&m.metallic, &m.roughness, &m.parallaxScale, &m.parallaxMinLayers,
205 &m.parallaxMaxLayers, &m.cellBombScale, &m.cellBombStrength, &m.cellBombRotation};
206 const char* scalarFields[] = {"metallic", "roughness", "parallaxScale", "parallaxMinLayers",
207 "parallaxMaxLayers", "cellBombScale", "cellBombStrength", "cellBombRotation"};
208 for (int i = 0; i < 8; ++i)
209 if (!number(field(*material, scalarFields[i]), *scalars[i]))
210 return fail<HouseKitComponentValue>(EditorStatus::Rejected, "editor.housegen.material-number",
211 "House material number is invalid");
212 return eve::editing::applied<HouseKitComponentValue>(std::move(out));

321PropertyReadResult HouseGenDocumentTarget::read(const SelectionSnapshot& s, const PropertyPath& path) const {
322 if (!matches(s) || !schema(s).find(path)) return {};
323 std::optional<EditorValue> common;
324 for (const auto& item : s.items) {
325 const auto& c = std::find_if(components_.begin(), components_.end(), [&](const auto& v) {
326 return v.id.value() == item.item.value();
327 })->component;
328 EditorValue value;
329 if (path == PropertyPath("component.model"))
330 value = c.modelPath;
331 else if (path == PropertyPath("component.category"))
332 value = c.category;
333 else if (path == PropertyPath("component.width"))
334 value = int64_t{c.width};
335 else if (path == PropertyPath("component.depth"))
336 value = int64_t{c.depth};
337 else if (path == PropertyPath("component.height"))
338 value = int64_t{c.height};
339 else if (path == PropertyPath("component.weight"))
340 value = int64_t{c.weight};
341 else if (path == PropertyPath("material.metallic"))
342 value = static_cast<double>(c.material.metallic);
343 else
344 value = static_cast<double>(c.material.roughness);
345 if (!common)
346 common = value;
347 else if (*common != value)
348 return {PropertyReadState::Mixed, {}, {}};
349 }
350 return {PropertyReadState::Value, *common, {}};

370EditorResult<DomainOperation> HouseGenDocumentTarget::makeSet(const SelectionSnapshot& s, const PropertyPath& path,
371 const EditorValue& value, PropertySetMode mode) const {
372 if (mode == PropertySetMode::Reset) return makeReset(s, path);
373 auto d = schema(s).find(path);
374 if (!matches(s) || !d || mode != PropertySetMode::Absolute || !validatePropertyValue(*d, value).ok())
375 return fail<DomainOperation>(EditorStatus::Rejected, "editor.housegen.set",
376 "House component property is invalid");
377 auto candidate = *this;
378 for (const auto& item : s.items) {
379 auto& c = std::find_if(candidate.components_.begin(), candidate.components_.end(), [&](const auto& v) {
380 return v.id.value() == item.item.value();
381 })->component;
382 if (path == PropertyPath("component.model"))
383 c.modelPath = *value.getIf<std::string>();
384 else if (path == PropertyPath("component.category"))
385 c.category = *value.getIf<std::string>();
386 else if (path == PropertyPath("component.width"))
387 c.width = static_cast<int>(*value.getIf<int64_t>());
388 else if (path == PropertyPath("component.depth"))
389 c.depth = static_cast<int>(*value.getIf<int64_t>());
390 else if (path == PropertyPath("component.height"))
391 c.height = static_cast<int>(*value.getIf<int64_t>());
392 else if (path == PropertyPath("component.weight"))
393 c.weight = static_cast<int>(*value.getIf<int64_t>());
394 else if (path == PropertyPath("material.metallic")) {
395 c.material.hasMetallic = true;
396 c.material.metallic = static_cast<float>(*value.getIf<double>());
397 } else {
398 c.material.hasRoughness = true;
399 c.material.roughness = static_cast<float>(*value.getIf<double>());
400 }
401 }
402 if (hasErrors(candidate.validate()))
403 return fail<DomainOperation>(EditorStatus::Rejected, "editor.housegen.invalid",
404 "House component edit invalidates the kit");
405 return replacement(candidate.contentValue(), path.value());
406}

446std::vector<EditorDiagnostic> HouseGenDocumentTarget::validate() const {
447 std::vector<EditorDiagnostic> out;
448 if (components_.size() > 2048)
449 out.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::PreconditionViolation,
450 RuleId("editor.housegen.component-budget"), DiagnosticSeverity::Error,
451 "House kit exceeds 2048 components"));
452 housegen::HouseComponentLibrary library;
453 std::set<std::string> editorIds;
454 std::set<std::string> categories;
455 for (const auto& c : components_) {
456 if (c.id.empty() || !editorIds.insert(c.id.value()).second)
457 out.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::Conflict,
458 RuleId("editor.housegen.editor-id"), DiagnosticSeverity::Error,
459 "House editor component IDs must be unique"));
460 auto registered = library.registerComponent(c.component);
461 if (!registered.ok())
462 out.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument,
463 RuleId("editor.housegen.component"), DiagnosticSeverity::Error,
464 "Invalid or duplicate runtime component: " + c.component.id));
465 categories.insert(c.component.category);
466 }
467 for (const char* required : {"foundation", "floor", "wall", "door", "roof"})
468 if (!categories.contains(required))
469 out.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::PreconditionViolation,
470 RuleId("editor.housegen.category"), DiagnosticSeverity::Warning,
471 "House kit is not generation-ready; missing category: " + std::string(required)));
472 const bool footprint = request_.footprint == "auto" || request_.footprint == "rectangle" ||
473 request_.footprint == "l_shape" || request_.footprint == "t_shape";
474 const bool roof =
475 request_.roof == "auto" || request_.roof == "gable" || request_.roof == "flat" || request_.roof == "shed";
476 const bool entrance = request_.entrance == "auto" || request_.entrance == "north" || request_.entrance == "east" ||

513 return fail<void>(EditorStatus::Rejected, "editor.housegen.invalid", "House document validation failed");

515 request_ = std::move(candidate.request_);

517 dirty_.include(0, 0);
