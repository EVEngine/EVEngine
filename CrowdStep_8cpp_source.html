12 Diagnostic::error(DiagnosticCode::InvalidArgument, "Crowd dt must be finite and non-negative", "dt"));
13 StepReport report;
14 report.navigation = getFieldStatus();
15 report.navigationClassMismatches = impl_->navigationMismatchCount();

20 d.avoidanceTruncations = 0;
21 double maxStep = 1.0 / 60.0;
22 float travelScale = d.field.valid() ? d.field.getCellSize() : 0.f;
23 float maxSpeed = 0.f;
24 for (size_t i = 0; i < d.xs.size(); ++i) {
25 if (d.radii[i] > 0.f) travelScale = travelScale > 0.f ? std::min(travelScale, d.radii[i]) : d.radii[i];
26 maxSpeed = std::max(maxSpeed, d.maxSpeeds[i]);
27 }
28 if (travelScale > 0.f && maxSpeed > 0.f) maxStep = std::min(maxStep, 0.5 * double(travelScale) / double(maxSpeed));
29 // Float dt=1/60 rounds slightly upward. Do not create an extra substep for
30 // representation error at an otherwise exact boundary.
31 const double count = std::ceil(double(dt) / maxStep - 1e-6);
32 if (!std::isfinite(count) || count > budget.maxSubsteps)
33 return Result<StepReport>::failure(Diagnostic::error(
34 DiagnosticCode::PreconditionViolation, "Crowd step exceeds its substep budget; reduce dt or speed", "dt"));
35 report.substeps = std::max(1, static_cast<int>(count));
36 d.workReport = {};

38 const auto start = std::chrono::steady_clock::now();
39 work();
40 duration +=
41 std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::steady_clock::now() - start).count();
42 };
43 const float substep = dt / float(report.substeps);
44 for (int sub = 0; sub < report.substeps; ++sub) {
45 d.simTime += substep;
46 measure(report.gridNanoseconds, [&] { d.rebuildGrid(); });
47 measure(report.motionNanoseconds, [&] { d.stepAgents(substep); });
48 measure(report.wallNanoseconds, [&] { d.resolveWalls(); });
49 if (d.resolveOverlaps) {
50 for (int pass = 0; pass < budget.maxContactPasses; ++pass) {
51 measure(report.gridNanoseconds, [&] { d.rebuildGrid(); });
52 float penetration = 0.f;
53 measure(report.contactNanoseconds, [&] { penetration = d.resolveOverlapsPass(); });
54 measure(report.wallNanoseconds, [&] { d.resolveWalls(); });
55 if (penetration < 1e-4f) break;
56 }
57 }
58 }

61 const auto residualStart = std::chrono::steady_clock::now();
62 const float maxRadius = *std::max_element(d.radii.begin(), d.radii.end());
63 for (size_t i = 0; i < d.xs.size(); ++i) {
64 d.forEachNeighbor(d.xs[i], d.ys[i], d.radii[i] + maxRadius, [&](int neighbor) {
65 const size_t j = size_t(neighbor);
66 if (j <= i || !d.canContact(i, j)) return;
67 if (report.residualPairsChecked >= budget.maxContactChecks - d.workReport.contactPairsChecked)
68 throw WorkLimit{"Crowd final contact inspection exceeds its contact budget"};

71 report.maxPenetration = std::max(report.maxPenetration, depth);
72 if (depth > 0.001f) ++report.unresolvedContacts;
73 });
74 if (d.field.valid()) {
75 float x = d.xs[i], y = d.ys[i];
76 d.field.resolvePenetration(x, y, d.radii[i]);
77 bool outside = false;
78 if (d.clampToField) {
79 outside = d.xs[i] - d.radii[i] < d.field.getOriginX() - 0.001f ||
80 d.ys[i] - d.radii[i] < d.field.getOriginY() - 0.001f ||
81 d.xs[i] + d.radii[i] >
82 d.field.getOriginX() + float(d.field.getWidth()) * d.field.getCellSize() + 0.001f ||
83 d.ys[i] + d.radii[i] >
84 d.field.getOriginY() + float(d.field.getHeight()) * d.field.getCellSize() + 0.001f;
85 }
86 if (outside || std::hypot(x - d.xs[i], y - d.ys[i]) > 0.001f) ++report.unresolvedWalls;
87 }
88 }
89 report.residualNanoseconds =
90 std::chrono::duration_cast<std::chrono::nanoseconds>(std::chrono::steady_clock::now() - residualStart).count();
91 report.gridBuilds = d.workReport.gridBuilds;
92 report.gridCellsPeak = d.workReport.gridCellsPeak;
93 report.cellsVisited = d.workReport.cellsVisited;
94 report.gridEntriesPeak = d.workReport.gridEntriesPeak;
95 report.gridSortComparisons = d.workReport.gridSortComparisons;
96 report.spatialVisits = d.workReport.spatialVisits;
97 report.contactPairsPeak = d.workReport.contactPairsPeak;
98 report.pairVisits = d.workReport.pairVisits;
99 report.preferredVisits = d.workReport.preferredVisits;
100 report.preferredNeighborsPeak = d.workReport.preferredNeighborsPeak;
101 report.candidatesVisited = d.workReport.candidatesVisited;
102 report.contactPairsChecked = d.workReport.contactPairsChecked;
103 report.contactPasses = d.workReport.contactPasses;
104 report.avoidanceChecks = d.avoidanceChecks;
105 report.avoidanceTruncations = d.avoidanceTruncations;
106 std::uint32_t quality = 0;
107 if (report.avoidanceTruncations > 0) quality |= static_cast<std::uint32_t>(SolveQuality::PredictionTruncated);
108 if (report.unresolvedContacts > 0) quality |= static_cast<std::uint32_t>(SolveQuality::ContactResidual);
109 if (report.unresolvedWalls > 0) quality |= static_cast<std::uint32_t>(SolveQuality::WallResidual);
110 report.quality = static_cast<SolveQuality>(quality);
111 return Result<StepReport>::success(report, Status::success(StatusCode::Applied));
112}
