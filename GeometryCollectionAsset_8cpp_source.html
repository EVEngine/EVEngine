181eve::Result<void> GeometryCollectionAsset::validate() const {
182 if (bones.empty())
183 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
184 "geometry-collection requires at least one bone",
185 "bones"));
186 if (bones.size() > 4096)
187 return eve::Result<void>::failure(eve::Diagnostic::error(
188 eve::DiagnosticCode::InvalidArgument, "geometry-collection exceeds the 4096-bone limit", "bones"));
189 if (edges.size() > 16384)
190 return eve::Result<void>::failure(eve::Diagnostic::error(
191 eve::DiagnosticCode::InvalidArgument, "geometry-collection exceeds the 16384-edge limit", "edges"));
192 const auto finitePositive = [](float value) { return std::isfinite(value) && value > 0.f; };
193 const auto finiteNonNeg = [](float value) { return std::isfinite(value) && value >= 0.f; };
194 for (std::size_t i = 0; i < bones.size(); ++i) {
195 const auto& bone = bones[i];
196 if (!finitePositive(bone.halfExtentX) || !finitePositive(bone.halfExtentY) || !finitePositive(bone.halfExtentZ))
197 return eve::Result<void>::failure(eve::Diagnostic::error(
198 eve::DiagnosticCode::InvalidArgument, "bone half-extents must be finite and positive", "bones"));
199 if (!std::isfinite(bone.localX) || !std::isfinite(bone.localY) || !std::isfinite(bone.localZ))
200 return eve::Result<void>::failure(eve::Diagnostic::error(
201 eve::DiagnosticCode::InvalidArgument, "bone local pose must be finite", "bones"));
202 if (!finitePositive(bone.mass) || !finitePositive(bone.density))
203 return eve::Result<void>::failure(eve::Diagnostic::error(
204 eve::DiagnosticCode::InvalidArgument, "bone mass and density must be finite and positive", "bones"));
205 if (!finiteNonNeg(bone.friction) || bone.friction > 1.f || !finiteNonNeg(bone.restitution) ||
206 bone.restitution > 1.f)
207 return eve::Result<void>::failure(eve::Diagnostic::error(
208 eve::DiagnosticCode::InvalidArgument, "bone friction/restitution must be in [0, 1]", "bones"));
209 if (bone.clusterId < 0 || bone.fractureLevel < 0)
210 return eve::Result<void>::failure(eve::Diagnostic::error(
211 eve::DiagnosticCode::InvalidArgument, "bone clusterId and fractureLevel must be non-negative",
212 "bones"));
213 (void)i;
214 }
215 for (const auto& edge : edges) {
216 if (edge.boneA < 0 || edge.boneB < 0 || static_cast<std::size_t>(edge.boneA) >= bones.size() ||
217 static_cast<std::size_t>(edge.boneB) >= bones.size() || edge.boneA == edge.boneB)
218 return eve::Result<void>::failure(eve::Diagnostic::error(
219 eve::DiagnosticCode::InvalidArgument, "edge bone indices must name two distinct bones", "edges"));
220 if (!std::isfinite(edge.strainThreshold) || edge.strainThreshold <= 0.f)
221 return eve::Result<void>::failure(eve::Diagnostic::error(
222 eve::DiagnosticCode::InvalidArgument, "edge strainThreshold must be finite and positive", "edges"));
223 }
224 return eve::Result<void>::success();
225}
