55Result<int> PcgRuntimeStamper::execute(Heightmap& target, double originX, double originZ,
56 double spacingX, double spacingZ) {
57 if (status_ != PcgRuntimeStampStatus::Loaded || !validRaster(stamp_))
58 return invalid<int>("a valid stamp resource must be loaded before execution");
59 if (!validRaster(target) || !std::isfinite(originX) || !std::isfinite(originZ) ||
60 !std::isfinite(spacingX) || !std::isfinite(spacingZ) || spacingX <= 0.0 || spacingZ <= 0.0)
61 return invalid<int>("finite nonempty terrain and positive grid spacing are required");
62
63 Heightmap candidate = target;
64 std::fill(candidate.data().begin(), candidate.data().end(), 0.0F);
65 Heightmap distanceMask(stamp_.getWidth(), stamp_.getHeight());
66 const double halfX = std::max(0.5, double(stamp_.getWidth() - 1) * 0.5);
67 const double halfY = std::max(0.5, double(stamp_.getHeight() - 1) * 0.5);
68 for (int y = 0; y < distanceMask.getHeight(); ++y) {
69 for (int x = 0; x < distanceMask.getWidth(); ++x) {
70 const double nx = (x - double(stamp_.getWidth() - 1) * 0.5) / halfX;
71 const double ny = (y - double(stamp_.getHeight() - 1) * 0.5) / halfY;
72 distanceMask.setHeight(x, y, float(std::max(0.0, 1.0 - std::sqrt(nx * nx + ny * ny))));
73 }
74 }
75 Heightmap one(1, 1);
76 one.setHeight(0, 0, 1.0F);
77 TerrainStampSettings settings;
78 settings.originX = originX;
79 settings.originZ = originZ;
80 settings.spacingX = spacingX;
81 settings.spacingZ = spacingZ;
82 settings.centerX = originX + double(target.getWidth() - 1) * spacingX * 0.5;
83 settings.centerZ = originZ + double(target.getHeight() - 1) * spacingZ * 0.5;
84 settings.width = std::max(spacingX, double(target.getWidth() - 1) * spacingX);
85 settings.depth = std::max(spacingZ, double(target.getHeight() - 1) * spacingZ);
86 settings.rotation = 0.0;
87 settings.amplitude = 6.0F;
88 settings.operation = TerrainStampOperation::Set;
89 auto result = applyTerrainStamp(candidate, stamp_, settings, one, distanceMask);
90 if (!result.ok()) {
91 progress_ = "Failed to load stamp";
92 status_ = PcgRuntimeStampStatus::Failed;
93 return result;
94 }
95 target = std::move(candidate);
96 progress_ = "Stamp progress: 1";
97 status_ = PcgRuntimeStampStatus::Completed;
98 return result;
99}
