33 Result<Policy> train(const Policy& p, const Observation& o, std::uint32_t action, double rate) override {
34 auto result = execute(p, o, action, rate, true);
35 if (!result) return Result<Policy>::failure(result.status());
36 Policy next = p;
37 next.weights = std::move(result).takeValue();
38 auto validated = validatePolicy(next);
39 if (!validated) return Result<Policy>::failure(validated.status());
40 return Result<Policy>::success(std::move(next));
41 }
42
43private:
44 Result<std::vector<double>> execute(const Policy& p, const Observation& o, std::uint32_t action, double rate,
45 bool training) {
46 auto ready = available();
47 if (!ready) return Result<std::vector<double>>::failure(ready.status());
48 try {
49 // Cache only the two shapes, never caller-owned weights or observation pointers.
50 if (f_ != p.featureCount || h_ != p.hiddenWidth || a_ != p.actionCount) {
51 inference_.reset();
52 training_.reset();
53 f_ = p.featureCount;
54 h_ = p.hiddenWidth;
55 a_ = p.actionCount;
56 }
57 auto& program = training ? training_ : inference_;
58 if (!program) {
59 auto recipe = detail::makePolicyGraph(int(f_), int(h_), int(a_), training);
60 auto optimized = tensor::optimizeGraph(recipe.graph, recipe.output);
61 program.reset(tensor::GpuProgram::tryBuild(recipe.graph, optimized, recipe.output));
62 if (!program)
63 return gpuError<std::vector<double>>("Policy graph cannot execute on this GPU; no CPU fallback");
64 }
65 std::vector<float> weights;
66 weights.reserve(p.weights.size());
67 for (double weight : p.weights) weights.push_back(static_cast<float>(weight));
68 std::vector<float> mask(a_, -std::numeric_limits<float>::infinity()), target(a_, 0);
69 for (auto legal : o.legalActions) mask[legal] = 0;
70 std::vector<const float*> feeds{weights.data(), o.features.data(), mask.data()};
71 float learningRate = float(rate);
72 if (training) {
73 target[action] = 1;
74 feeds.push_back(target.data());
75 feeds.push_back(&learningRate);
76 }
77 auto output = program->run(feeds);
78 if (output.size() != (training ? p.weights.size() : p.actionCount))
79 return gpuError<std::vector<double>>("GPU returned an invalid output shape");
80 for (auto value : output)
81 if (!std::isfinite(value)) return gpuError<std::vector<double>>("GPU returned non-finite values");
82 return Result<std::vector<double>>::success({output.begin(), output.end()});
83 } catch (const std::exception& e) {
84 inference_.reset();
85 training_.reset();
86 return gpuError<std::vector<double>>(e.what());
87 }
88 }
89 std::uint32_t f_ = 0, h_ = 0, a_ = 0;
90 std::unique_ptr<tensor::GpuProgram> inference_, training_;
91};
