66Result<WaveEntry> decodeEntry(const eve::json::Value& object) {
67 WaveEntry entry;
68 entry.archetypeId = object.getString("archetypeId");
69 entry.count = object.getInt("count", 1);
70 entry.intervalSeconds = object.getDouble("intervalSeconds", 0.5);
71 entry.delaySeconds = object.getDouble("delaySeconds", 0.0);

79 if (!std::isfinite(entry.intervalSeconds) || entry.intervalSeconds < 0.0)

81 "intervalSeconds must be finite and >= 0",

87 return Result<WaveEntry>::success(std::move(entry));

90Result<WaveDefinition> decodeWave(const eve::json::Value& object) {
91 WaveDefinition wave;
92 wave.id = object.getString("id");
93 wave.prepSeconds = object.getDouble("prepSeconds", 3.0);
94 wave.rewardGold = object.getInt("rewardGold", 0);
95 if (wave.id.empty())
96 return Result<WaveDefinition>::failure(
97 Diagnostic::error(DiagnosticCode::InvalidArgument, "wave requires id", "wave.id", {}, "towerdefense"));
98 if (!std::isfinite(wave.prepSeconds) || wave.prepSeconds < 0.0)
99 return Result<WaveDefinition>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
100 "prepSeconds must be finite and >= 0",
101 "wave.prepSeconds", {}, "towerdefense"));
102 if (wave.rewardGold < 0)
103 return Result<WaveDefinition>::failure(Diagnostic::error(
104 DiagnosticCode::InvalidArgument, "rewardGold must be >= 0", "wave.rewardGold", {}, "towerdefense"));
105 const eve::json::Value entries = object.get("entries");
106 if (!entries.isArray() || entries.size() == 0)
107 return Result<WaveDefinition>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
108 "wave requires a non-empty entries array",
109 "wave.entries", {}, "towerdefense"));
110 wave.entries.reserve(entries.size());
111 for (std::size_t i = 0; i < entries.size(); ++i) {
112 const eve::json::Value entry = entries.at(i);
113 if (!entry.isObject())
114 return Result<WaveDefinition>::failure(Diagnostic::error(
115 DiagnosticCode::InvalidArgument, "wave entry must be an object", "wave.entries", {}, "towerdefense"));
116 auto decoded = decodeEntry(entry);
117 if (!decoded) return Result<WaveDefinition>::failure(decoded.status());
118 wave.entries.push_back(std::move(decoded).takeValue());
119 }
120 return Result<WaveDefinition>::success(std::move(wave));
121}

138Result<void> WaveDirector::registerWave(WaveDefinition definition) {
139 if (definition.id.empty())
140 return Result<void>::failure(
141 Diagnostic::error(DiagnosticCode::InvalidArgument, "wave requires id", "wave.id", {}, "towerdefense"));
142 if (definition.entries.empty())
143 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
144 "wave requires a non-empty entries array", "wave.entries", {},
145 "towerdefense"));

147 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,

149 "towerdefense"));

152 entry.intervalSeconds < 0.0 || !std::isfinite(entry.delaySeconds) || entry.delaySeconds < 0.0)

154 DiagnosticCode::InvalidArgument, "wave entry fields are invalid", "wave.entries", {}, "towerdefense"));
155 }
156 for (auto& existing : waves_) {
157 if (existing.id == definition.id) {
158 existing = std::move(definition);
159 return Result<void>::success(Status::success(StatusCode::Applied));
160 }
161 }
162 waves_.push_back(std::move(definition));
163 return Result<void>::success(Status::success(StatusCode::Applied));

166Result<int> WaveDirector::loadScheduleJson(std::string_view json) {
167 std::string error;
168 const eve::json::Document doc = eve::json::Document::parse(std::string(json), &error);
169 if (!doc.valid())
170 return Result<int>::failure(Diagnostic::error(DiagnosticCode::ParseError,
171 error.empty() ? "invalid wave schedule json" : error, "schedule",
172 {}, "towerdefense"));
173 const eve::json::Value root = doc.root();
174 if (!root.isArray())
175 return Result<int>::failure(Diagnostic::error(
176 DiagnosticCode::InvalidArgument, "wave schedule must be a JSON array", "schedule", {}, "towerdefense"));
177 clear();
178 int loaded = 0;
179 for (std::size_t i = 0; i < root.size(); ++i) {
180 const eve::json::Value entry = root.at(i);
181 if (!entry.isObject())
182 return Result<int>::failure(Diagnostic::error(
183 DiagnosticCode::InvalidArgument, "schedule entries must be objects", "schedule", {}, "towerdefense"));
184 auto decoded = decodeWave(entry);
185 if (!decoded) return Result<int>::failure(decoded.status());
186 auto registered = registerWave(std::move(decoded).takeValue());
187 if (!registered) return Result<int>::failure(registered.status());
188 ++loaded;
189 }
190 return Result<int>::success(loaded);
191}

251Result<WaveAdvanceResult> WaveDirector::advance(std::uint64_t tick, double dt) {
252 WaveAdvanceResult out;
253 out.phase = phase_;
254 if (!std::isfinite(dt) || dt < 0.0)
255 return Result<WaveAdvanceResult>::failure(Diagnostic::error(
256 DiagnosticCode::InvalidArgument, "dt must be finite and >= 0", "advance.dt", {}, "towerdefense"));
257 if (hasTick_ && tick <= lastTick_)
258 return Result<WaveAdvanceResult>::failure(Diagnostic::error(
259 DiagnosticCode::PreconditionViolation, "tick must strictly increase", "advance.tick", {}, "towerdefense"));
260 hasTick_ = true;
261 lastTick_ = tick;
262
263 if (phase_ == WavePhase::Idle || phase_ == WavePhase::Completed || phase_ == WavePhase::Failed) {
264 out.phase = phase_;
265 return Result<WaveAdvanceResult>::success(std::move(out));
266 }
267
268 double budget = dt;
269 int spawnsThisAdvance = 0;
270 constexpr int kMaxSpawnsPerAdvance = 256;
271 while (true) {
272 if (phase_ == WavePhase::Preparing || phase_ == WavePhase::BetweenWaves) {
273 if (remainingPrep_ > budget) {
274 remainingPrep_ -= budget;
275 break;
276 }
277 budget -= remainingPrep_;
278 remainingPrep_ = 0.0;
279 auto began = beginWave(waveIndex_, out);
280 if (!began) {
281 phase_ = WavePhase::Failed;
282 emitLifecycle(out, "failed", {}, waveIndex_, 0, began.status().describe());
283 out.phase = phase_;
284 return Result<WaveAdvanceResult>::success(std::move(out));
285 }
286 continue;
287 }
288
289 if (phase_ != WavePhase::Spawning) break;
290 if (waveIndex_ < 0 || waveIndex_ >= static_cast<int>(waves_.size())) {
291 phase_ = WavePhase::Failed;
292 emitLifecycle(out, "failed", {}, waveIndex_, 0, "active wave missing");
293 break;
294 }
295 auto& wave = waves_[static_cast<std::size_t>(waveIndex_)];
296 if (entryIndex_ >= static_cast<int>(wave.entries.size())) {
297 emitLifecycle(out, "waveCompleted", wave.id, waveIndex_, wave.rewardGold, {});
298 const int next = waveIndex_ + 1;
299 if (next >= static_cast<int>(waves_.size())) {
300 phase_ = WavePhase::Completed;
301 emitLifecycle(out, "scheduleCompleted", {}, next, 0, {});
302 break;
303 }
304 waveIndex_ = next;
305 remainingPrep_ = waves_[static_cast<std::size_t>(next)].prepSeconds;
306 phase_ = WavePhase::BetweenWaves;
307 continue;
308 }
309
310 auto& entry = wave.entries[static_cast<std::size_t>(entryIndex_)];
311 if (spawnCooldown_ > budget) {
312 spawnCooldown_ -= budget;
313 break;
314 }
315 budget -= spawnCooldown_;
316 spawnCooldown_ = 0.0;
317 if (spawnsThisAdvance >= kMaxSpawnsPerAdvance) break;
318
319 WaveSpawnEvent spawn;
320 spawn.sequence = ++eventSequence_;
321 spawn.waveId = wave.id;
322 spawn.waveIndex = waveIndex_;
323 spawn.archetypeId = entry.archetypeId;
324 spawn.entryIndex = entryIndex_;
325 spawn.spawnIndex = spawnedInEntry_;
326 out.spawns.push_back(std::move(spawn));
327 ++spawnsThisAdvance;
328 ++spawnedInEntry_;
329 if (spawnedInEntry_ >= entry.count) {
330 ++entryIndex_;
331 spawnedInEntry_ = 0;
332 if (entryIndex_ < static_cast<int>(wave.entries.size()))
333 spawnCooldown_ = wave.entries[static_cast<std::size_t>(entryIndex_)].delaySeconds;
334 } else {
335 spawnCooldown_ = entry.intervalSeconds;
336 }
337 // Zero-interval batches may consume no dt; stop once the time budget is exhausted.
338 if (budget <= 0.0 && spawnCooldown_ <= 0.0) break;
339 }
340
341 out.phase = phase_;
342 return Result<WaveAdvanceResult>::success(std::move(out));
343}
