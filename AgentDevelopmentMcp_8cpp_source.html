85std::string callAgentDevelopmentTool(std::string_view name, Poco::JSON::Object::Ptr args) {
86 auto& session = AgentDevelopmentSession::instance();
87 AgentDevelopmentResult result{"applied", {}};
88 if (name == "eve_agent_session_start") {
89 std::vector<AgentAcceptanceCriterion> criteria;
90 try {
91 auto array = args ? args->getArray("criteria") : nullptr;
92 if (array) {
93 for (std::size_t index = 0; index < array->size(); ++index) {
94 auto item = array->getObject(static_cast<unsigned>(index));
95 if (!item) {
96 result = invalid("invalid-criterion", "each criterion must be an object");
97 break;
98 }
99 criteria.push_back({stringField(item, "id"), stringField(item, "description"),
100 item->optValue<bool>("required", true), false});
101 }
102 }
103 } catch (...) {
104 result = invalid("invalid-criteria", "criteria must be an array of objects");
105 }
106 if (result.isAccepted()) result = session.start(stringField(args, "objective"), std::move(criteria));
107 } else if (name == "eve_agent_session_advance") {
108 AgentDevelopmentPhase phase = AgentDevelopmentPhase::Aborted;
109 result = parseAgentDevelopmentPhase(stringField(args, "phase"), &phase);
110 if (result.isAccepted()) result = session.advance(stringField(args, "sessionId"), phase);
111 } else if (name == "eve_agent_session_evidence") {
112 AgentDevelopmentEvidence evidence;
113 evidence.criterionId = stringField(args, "criterionId");
114 evidence.kind = stringField(args, "kind");
115 evidence.status = stringField(args, "status");
116 evidence.summary = stringField(args, "summary");
117 evidence.artifact = stringField(args, "artifact");
118 result = session.record(stringField(args, "sessionId"), std::move(evidence));
119 } else if (name == "eve_agent_session_complete") {
120 result = session.complete(stringField(args, "sessionId"), stringField(args, "summary"));
121 } else if (name == "eve_agent_session_abort") {
122 result = session.abort(stringField(args, "sessionId"), stringField(args, "reason"));
123 } else if (name != "eve_agent_session_status") {
124 result = invalid("unknown-tool", "unknown Agent development-session tool");
125 }
126 return stringify(Poco::Dynamic::Var(snapshot(result)));
127}
