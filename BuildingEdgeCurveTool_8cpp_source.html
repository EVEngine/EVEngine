18OverlayStyle styleFor(const editing::GizmoPrimitive& primitive) {
19 const auto channel = [&](size_t index) {
20 return static_cast<unsigned int>(std::clamp(primitive.color[index], 0.0, 1.0) *
21 255.0 + 0.5);
22 };
23 const unsigned int color = (channel(3) << 24U) | (channel(0) << 16U) |
24 (channel(1) << 8U) | channel(2);
25 return {color, primitive.dashed ? 1.f : 2.f, false};
26}

146ToolResponse BuildingEdgeCurveTool::finish(EditorContext& context,
147 const EditorPointerEvent& event) {
148 if (!isDragging()) return ToolResponse::ignored();
149 (void)move(event);
150 if (!isDragging()) return ToolResponse::release();
151 auto operation = drag_->finishDrag();
152 drag_.reset();
153 if (!operation.ok()) {
154 recordFailure(operation.code(), operation.diagnostics());
155 return ToolResponse::release();
156 }
157 auto* target = dynamic_cast<BuildingPlacementTarget*>(context.target());
158 if (!target || !authority_) {
159 auto failure = toolFailure(editing::Status::Failed,
160 "editor.building.curve-tool-authority-missing",
161 "Curve commit requires its selected target and authority");
162 recordFailure(failure.code(), failure.diagnostics());
163 return ToolResponse::release();
164 }
165 editing::TransactionSpec transaction;
166 transaction.id = editing::TransactionId(
167 descriptor_.id + "." + std::to_string(++transactionSequence_));
168 transaction.label = descriptor_.label;
169 transaction.origin = editing::ActionOrigin::User;
170 transaction.target = target->targetId();
171 transaction.baseRevision = target->revision();
172 transaction.mergeKey = "building.edge-curve.handle";
173 const std::array<editing::DomainOperation, 1> operations{std::move(operation.value())};
174 auto plan = authority_->preflight(transaction, std::span<const editing::DomainOperation>(operations));
175 if (!plan.ok()) {
176 recordFailure(plan.code(), plan.diagnostics());
177 return ToolResponse::release();
178 }
179 auto receipt = authority_->commit(plan.value());
180 if (!receipt.ok()) {
181 recordFailure(receipt.code(), receipt.diagnostics());
182 return ToolResponse::release();
183 }
184 lastStatus_ = editing::Status::Applied;
185 diagnostics_ = receipt.value().diagnostics;
186 lastReceipt_ = std::move(receipt.value());
187 if (selection_ && preview_) {
188 selection_->controlPoints = preview_->curve.controlPoints;
189 if (!lastReceipt_->affectedObjects.empty()) {
190 const std::string& value = lastReceipt_->affectedObjects.front().object.value();
191 int member = 0;
192 const auto parsed = std::from_chars(value.data(), value.data() + value.size(), member);
193 if (parsed.ec == std::errc{} && parsed.ptr == value.data() + value.size() &&
194 member > 0)
195 selection_->memberInstanceId = member;
196 }
197 }
198 return ToolResponse::release();
199}

219void BuildingEdgeCurveTool::drawOverlay(EditorContext&, IEditorOverlay& overlay) {
220 if (!preview_ || !viewport_) return;
221 for (const editing::GizmoPrimitive& primitive : preview_->gizmo.primitives) {
222 const OverlayStyle style = styleFor(primitive);
223 if (primitive.kind == "sphere") {
224 auto center = viewport_->projectWorld(primitive.position);
225 if (center.ok()) overlay.circle(center.value(), 7.f, style);
226 continue;
227 }
228 if (primitive.kind != "line") continue;
229 std::array<double, 3> from = primitive.position;
230 std::array<double, 3> to = primitive.position;
231 for (size_t axis = 0; axis < 3; ++axis) {
232 const double half = primitive.direction[axis] * primitive.length * 0.5;
233 from[axis] -= half;
234 to[axis] += half;
235 }
236 auto projectedFrom = viewport_->projectWorld(from);
237 auto projectedTo = viewport_->projectWorld(to);
238 if (projectedFrom.ok() && projectedTo.ok())
239 overlay.line(projectedFrom.value(), projectedTo.value(), style);
240 }
241}
