11Result<void> AudioAuditionTransport::bind(StableId asset, Revision revision, IAudioTransportBackend* backend) {
12 if (asset.empty() || revision == 0 || !backend)
13 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.audio.transport-bind"),
14 "Audition requires asset identity, revision and playback backend");
15 const double duration = backend->duration();
16 if (!std::isfinite(duration) || duration <= 0.0)
17 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.audio.transport-duration"),
18 "Audition source requires a finite positive duration");
19 unbind();
20 asset_ = std::move(asset);
21 revision_ = revision;
22 backend_ = backend;
23 backend_->setNativeLooping(false);
24 return eve::editing::applied<void>();
25}
26Result<void> AudioAuditionTransport::validateRevision(Revision expected) const {
27 if (!backend_)

35Result<void> AudioAuditionTransport::setLoop(Revision expected, bool enabled, double start, double end) {
36 auto valid = validateRevision(expected);
37 if (!valid.ok()) return valid;
38 const double duration = backend_->duration();
39 if (end == 0.0) end = duration;
40 if (!std::isfinite(start) || !std::isfinite(end) || start < 0.0 || end <= start || end > duration)
41 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.audio.transport-loop-range"),
42 "Loop range must be finite, ordered and inside the clip");
43 loopEnabled_ = enabled;

49Result<void> AudioAuditionTransport::play(Revision expected) {
50 auto valid = validateRevision(expected);
51 if (!valid.ok()) return valid;
52 if (loopEnabled_ && (backend_->tell() < loopStart_ || backend_->tell() >= loopEnd_)) {

106Result<AudioTransportSnapshot> AudioAuditionTransport::update(Revision expected) {
107 auto valid = validateRevision(expected);
108 if (!valid.ok()) {
109 const EditorStatus status = valid.code();
110 if (status == EditorStatus::Conflict) unbind();
111 return eve::editing::failed<AudioTransportSnapshot>(status, RuleId("editor.audio.transport-stale"),
112 "Audition source is absent or stale");
113 }
114 if (state_ == AudioTransportState::Playing && loopEnabled_ && backend_->tell() >= loopEnd_) {
115 auto seekResult = backend_->seek(loopStart_);
116 if (!seekResult.ok())
117 return eve::editing::failed<AudioTransportSnapshot>(
118 seekResult.code(), RuleId("editor.audio.transport-loop-seek"), "Playback backend cannot wrap the loop");
119 backend_->play();
120 } else if (state_ == AudioTransportState::Playing && !backend_->playing() &&
121 backend_->tell() >= backend_->duration()) {
122 state_ = AudioTransportState::Stopped;
123 }
124 return eve::editing::applied<AudioTransportSnapshot>(observe());
125}

141AudioWaveformResult AudioWaveformService::generate(const AudioWaveformRequest& request, const EditorAudioPcm& pcm,
142 int maxPixels, std::size_t maxSamples) const {
143 AudioWaveformResult result;
144 result.asset = request.asset;
145 result.sourceRevision = request.sourceRevision;
146 result.sampleRate = pcm.sampleRate;
147 result.channels = pcm.channels;
148 if (request.asset.empty() || request.pixelWidth <= 0 || request.pixelWidth > maxPixels || maxPixels <= 0 ||
149 pcm.sampleRate <= 0 || pcm.channels <= 0 || pcm.channels > 32 || pcm.samples.empty() ||
150 pcm.samples.size() > maxSamples || pcm.samples.size() % static_cast<std::size_t>(pcm.channels) != 0) {
151 result.status = EditorStatus::Rejected;
152 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
153 eve::DiagnosticCode::InvalidArgument, RuleId("editor.audio.invalid-waveform-input"),
154 DiagnosticSeverity::Error, "Waveform requires bounded, aligned PCM and a valid asset/sample format"));
155 return result;
156 }
157 for (float sample : pcm.samples)
158 if (!std::isfinite(sample)) {
159 result.status = EditorStatus::Rejected;
160 result.diagnostics.push_back(
161 eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument, RuleId("editor.audio.nonfinite-pcm"),
162 DiagnosticSeverity::Error, "Decoded PCM contains a non-finite sample"));
163 return result;
164 }
165 const std::size_t frameCount = pcm.samples.size() / static_cast<std::size_t>(pcm.channels);
166 result.duration = static_cast<double>(frameCount) / pcm.sampleRate;
167 const int buckets = std::min<int>(request.pixelWidth, static_cast<int>(frameCount));
168 result.envelopes.assign(static_cast<std::size_t>(pcm.channels), {});
169 for (auto& channel : result.envelopes) channel.reserve(static_cast<std::size_t>(buckets));
170 for (int bucket = 0; bucket < buckets; ++bucket) {
171 const std::size_t begin = static_cast<std::size_t>(bucket) * frameCount / buckets;
172 const std::size_t end = std::max(begin + 1, static_cast<std::size_t>(bucket + 1) * frameCount / buckets);
173 for (int channel = 0; channel < pcm.channels; ++channel) {
174 float minimum = std::numeric_limits<float>::infinity();
175 float maximum = -std::numeric_limits<float>::infinity();
176 double squares = 0.0;
177 for (std::size_t frame = begin; frame < end; ++frame) {
178 const float sample = pcm.samples[frame * static_cast<std::size_t>(pcm.channels) + channel];
179 minimum = std::min(minimum, sample);
180 maximum = std::max(maximum, sample);
181 squares += static_cast<double>(sample) * sample;
182 }
183 result.envelopes[static_cast<std::size_t>(channel)].push_back(
184 {static_cast<double>(begin) / pcm.sampleRate, static_cast<double>(end) / pcm.sampleRate, minimum,
185 maximum, static_cast<float>(std::sqrt(squares / static_cast<double>(end - begin)))});
186 }
187 }
188 if (request.loopEnd > 0.0 || request.loopStart > 0.0) {
189 if (!std::isfinite(request.loopStart) || !std::isfinite(request.loopEnd) || request.loopStart < 0.0 ||
190 request.loopEnd <= request.loopStart || request.loopEnd > result.duration) {
191 result.status = EditorStatus::Rejected;
192 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
193 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.audio.invalid-loop-range"),
194 DiagnosticSeverity::Error, "Loop range must be ordered and within decoded clip duration"));
195 return result;
196 }
197 const auto bucketAt = [&](double seconds) {
198 return std::min(buckets - 1, static_cast<int>(seconds / result.duration * buckets));
199 };
200 result.loopStartBucket = bucketAt(request.loopStart);
201 result.loopEndBucket = bucketAt(request.loopEnd);
202 const std::size_t startFrame =
203 std::min(frameCount - 1, static_cast<std::size_t>(request.loopStart * pcm.sampleRate));
204 const std::size_t endFrame =
205 std::min(frameCount - 1, static_cast<std::size_t>(request.loopEnd * pcm.sampleRate));
206 double maximumDelta = 0.0;
207 for (int channel = 0; channel < pcm.channels; ++channel) {
208 const float start = pcm.samples[startFrame * static_cast<std::size_t>(pcm.channels) + channel];
209 const float end = pcm.samples[endFrame * static_cast<std::size_t>(pcm.channels) + channel];
210 maximumDelta = std::max(maximumDelta, std::abs(static_cast<double>(end - start)));
211 }
212 result.loopSeamDelta = maximumDelta;
213 if (maximumDelta > 0.1)
214 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
215 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.audio.loop-seam-discontinuity"),
216 DiagnosticSeverity::Warning, "Loop endpoints have a large amplitude discontinuity and may click"));
217 }
218 result.status = EditorStatus::Applied;
219 return result;
220}
