110EditorResult<AudioTransportSnapshot> AudioAuditionTransport::update(Revision expected) {
111 auto valid = validateRevision(expected);
112 if (!valid.ok()) {
113 const EditorStatus status = valid.code();
114 if (status == EditorStatus::Conflict) unbind();
115 return transportError<AudioTransportSnapshot>(status, "editor.audio.transport-stale",
116 "Audition source is absent or stale");
117 }
118 if (state_ == AudioTransportState::Playing && loopEnabled_ && backend_->tell() >= loopEnd_) {
119 auto seekResult = backend_->seek(loopStart_);
120 if (!seekResult.ok())
121 return transportError<AudioTransportSnapshot>(seekResult.code(),
122 "editor.audio.transport-loop-seek",
123 "Playback backend cannot wrap the loop");
124 backend_->play();
125 } else if (state_ == AudioTransportState::Playing && !backend_->playing() &&
126 backend_->tell() >= backend_->duration()) {
127 state_ = AudioTransportState::Stopped;
128 }
129 return eve::editing::applied<AudioTransportSnapshot>(observe());
130}
