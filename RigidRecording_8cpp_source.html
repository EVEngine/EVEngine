42 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
43 "rigid recording sample budget exceeded", "rigidRecording"));
44 auto timing = impl_->recipe.validateSampleTime(static_cast<std::uint32_t>(impl_->frames.size()), frame.seconds);
45 if (!timing) return timing;
46 const std::span<const RigidTrackSample> roster =
47 impl_->frames.empty() ? std::span<const RigidTrackSample>{} : impl_->frames.front().samples;
48 auto valid = bake_detail::validateRigidFrame(impl_->space, frame, roster,
49 impl_->frames.empty() ? -1 : impl_->frames.back().seconds);
50 if (!valid) return valid;
