22EditorResult<PluginPermissionGrant> parseGrant(const EditorValue& value) {
23 auto string = [&](const char* key) -> const std::string* {
24 const auto* entry = field(value, key);
25 return entry ? entry->getIf<std::string>() : nullptr;
26 };
27 const auto *id = string("id"), *plugin = string("plugin"), *capability = string("capability"),
28 *scope = string("scope"), *decision = string("decision");
29 static const std::set<std::string> capabilities{"filesystem.read", "filesystem.write", "network.connect",
30 "database.connect", "script.native", "process.execute"};
31 static const std::set<std::string> decisions{"allow", "deny", "ask"};
32 if (!id || id->empty() || !plugin || plugin->empty() || !capability || !capabilities.contains(*capability) ||
33 !scope || scope->empty() || !decision || !decisions.contains(*decision) || *scope == "*" || *scope == "/")
34 return permissionError<PluginPermissionGrant>(
35 EditorStatus::Rejected, "editor.plugins.invalid-permission",
36 "Permission requires plugin, known capability, explicit narrow scope and decision");
37 return eve::editing::applied<PluginPermissionGrant>({StableId(*id), *plugin, *capability, *scope, *decision});
38}

57EditorResult<void> PluginPermissionTarget::applyDomainOperation(const DomainOperation& operation) {
58 if (operation.target != TargetId(id_))
59 return permissionError<void>(EditorStatus::Rejected, "editor.plugins.permission-target",
60 "Permission operation targets another document");
61 if (operation.type == "plugin.permission.set.v1") {
62 auto parsed = parseGrant(operation.payload);
63 if (!parsed.ok()) return EditorResult<void>::failure(parsed.status());
64 for (const auto& [id, current] : grants_)
65 if (id != parsed.value().id && current.plugin == parsed.value().plugin &&
66 current.capability == parsed.value().capability && current.scope == parsed.value().scope)
67 return permissionError<void>(EditorStatus::Conflict, "editor.plugins.duplicate-permission",
68 "Equivalent permission already exists");
69 grants_.insert_or_assign(parsed.value().id, parsed.value());
70 } else if (operation.type == "plugin.permission.remove.v1") {
71 const auto* id = operation.payload.getIf<std::string>();
72 if (!id || !grants_.erase(StableId(*id)))
73 return permissionError<void>(EditorStatus::NotFound, "editor.plugins.permission-not-found",
74 "Permission grant was not found");
75 } else {
76 return permissionError<void>(EditorStatus::Rejected, "editor.plugins.permission-operation",
77 "Unsupported permission operation");
78 }
79 ++revision_; dirty_.include(0, 0);
80 return eve::editing::applied<void>();
81}

133EditorResult<void> PluginPermissionTarget::loadSnapshot(const EditorValue& snapshot) {
134 const auto *versionEntry = field(snapshot, "schemaVersion"), *grantsEntry = field(snapshot, "grants");
135 const auto* version = versionEntry ? versionEntry->getIf<int64_t>() : nullptr;
136 const auto* grants = grantsEntry ? grantsEntry->getIf<EditorValue::Array>() : nullptr;
137 if (!version || *version != 1 || !grants)
138 return permissionError<void>(EditorStatus::Rejected, "editor.plugins.permission-snapshot",
139 "Invalid permission snapshot envelope");
140 PluginPermissionTarget candidate(id_);
141 for (const auto& entry : *grants) {
142 auto parsed = parseGrant(entry);
143 if (!parsed.ok()) return EditorResult<void>::failure(parsed.status());
144 auto planned = candidate.makeSet(parsed.value());
145 if (!planned.ok()) return EditorResult<void>::failure(planned.status());
146 auto applied = candidate.applyDomainOperation(planned.value());
147 if (!applied.ok()) return applied;
148 }
149 grants_ = std::move(candidate.grants_); ++revision_; dirty_.include(0, 0);
150 return eve::editing::applied<void>();
