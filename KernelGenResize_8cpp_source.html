86std::string emitQuantizedBVal(DType dt, int group, const char *bufName = "b") {
87 std::ostringstream os;
88 const int g = group > 0 ? group : 1;
89 switch (dt) {
90 case DType::Int8:
91 os << "fn bval(idx: u32) -> f32 {\n"
92 << " var word: u32 = " << bufName << "[idx >> 2u];\n"
93 << " var byte: u32 = (word >> ((idx & 3u) * 8u)) & 0xFFu;\n"
94 << " var v: i32 = i32(byte);\n"
95 << " if (v >= 128) { v -= 256; }\n"
96 << " return f32(v) * bs[idx / " << g << "u];\n"
97 << "}\n";
98 break;
99 case DType::Int4:
100 os << "fn bval(idx: u32) -> f32 {\n"
101 << " var word: u32 = " << bufName << "[idx >> 3u];\n"
102 << " var byte: u32 = (word >> (((idx >> 1u) & 3u) * 8u)) & 0xFFu;\n"
103 << " var nib: u32 = select(byte >> 4u, byte & 0xFu, (idx & 1u) == 0u);\n"
104 << " var v: i32 = i32(nib);\n"
105 << " if (v >= 8) { v -= 16; }\n"
106 << " return f32(v) * bs[idx / " << g << "u];\n"
107 << "}\n";
108 break;
109 case DType::Fp16:
110 os << "fn bval(idx: u32) -> f32 {\n"
111 << " var word: u32 = " << bufName << "[idx >> 1u];\n"
112 << " var hb: u32 = (word >> ((idx & 1u) * 16u)) & 0xFFFFu;\n"
113 << " return unpack2x16float(hb).x;\n"
114 << "}\n";
115 break;
116 case DType::Fp8E4M3:
117 os << "fn bval(idx: u32) -> f32 {\n"
118 << " var word: u32 = " << bufName << "[idx >> 2u];\n"
119 << " var byte: u32 = (word >> ((idx & 3u) * 8u)) & 0xFFu;\n"
120 << " var s: i32 = select(1, -1, (byte & 0x80u) != 0u);\n"
121 << " var e: i32 = i32((byte >> 3u) & 0xFu);\n"
122 << " var m: i32 = i32(byte & 0x7u);\n"
123 << " var v: f32 = select(exp2(f32(e - 7)) * (1.0 + f32(m) / 8.0),\n"
124 << " exp2(-6.0) * f32(m) / 8.0, e == 0);\n"
125 << " return f32(s) * v * bs[idx / " << g << "u];\n"
126 << "}\n";
127 break;
128 case DType::Fp4E2M1:
129 os << "fn bval(idx: u32) -> f32 {\n"
130 << " var word: u32 = " << bufName << "[idx >> 3u];\n"
131 << " var byte: u32 = (word >> (((idx >> 1u) & 3u) * 8u)) & 0xFFu;\n"
132 << " var nib: u32 = select(byte >> 4u, byte & 0xFu, (idx & 1u) == 0u);\n"
133 << " var s: i32 = select(1, -1, (nib & 8u) != 0u);\n"
134 << " var e: i32 = i32((nib >> 1u) & 3u);\n"
135 << " var m: i32 = i32(nib & 1u);\n"
136 << " var v: f32 = select(exp2(f32(e - 1)) * (1.0 + 0.5 * f32(m)),\n"

142 }

183 std::string emit(int nodeId) {
184 const GraphNode &nd = graph.node(nodeId);
185 const std::string x =
186 (nd.in0 == group.nodes.front() && rootVar != "") ? rootVar : (nd.in0 >= 0 ? operand(nd.in0) : rootVar);
187 std::string expr;
188 switch (nd.type) {
189 case OpType::Add: expr = "(" + x + " + " + operand(nd.in1) + ")"; break;
190 case OpType::Sub: expr = "(" + x + " - " + operand(nd.in1) + ")"; break;
191 case OpType::Multiply: expr = "(" + x + " * " + operand(nd.in1) + ")"; break;
192 case OpType::Divide: expr = "(" + x + " / " + operand(nd.in1) + ")"; break;
193 case OpType::AddScalar: expr = "(" + x + " + " + scalarStr(nd.s0) + ")"; break;
194 case OpType::SubScalar: expr = "(" + x + " - " + scalarStr(nd.s0) + ")"; break;
195 case OpType::MulScalar: expr = "(" + x + " * " + scalarStr(nd.s0) + ")"; break;
196 case OpType::DivScalar: expr = "(" + x + " / " + scalarStr(nd.s0) + ")"; break;
197 case OpType::PowScalar: expr = "pow(" + x + ", " + scalarStr(nd.s0) + ")"; break;
198 case OpType::Neg: expr = "(-" + x + ")"; break;
199 case OpType::Abs: expr = "abs(" + x + ")"; break;
200 case OpType::Sqrt: expr = "sqrt(" + x + ")"; break;
201 case OpType::Exp: expr = "exp(" + x + ")"; break;
202 case OpType::Log: expr = "log(" + x + ")"; break;
203 case OpType::Sin: expr = "sin(" + x + ")"; break;
204 case OpType::Cos: expr = "cos(" + x + ")"; break;
205 case OpType::Tanh: expr = "tanh(" + x + ")"; break;
206 case OpType::Relu: expr = "max(" + x + ", 0.0)"; break;
207 case OpType::Sigmoid: expr = "(1.0 / (1.0 + exp(-" + x + ")))"; break;
208 case OpType::Gelu:
209 expr = "(0.5 * " + x + " * (1.0 + tanh(0.7978845608028654 * (" + x + " + 0.044715 * " + x + " * " + x +
210 " * " + x + "))))";
211 break;
212 case OpType::Silu: expr = "(" + x + " / (1.0 + exp(-" + x + ")))"; break;
213 case OpType::Clamp: {
214 float lo = nd.s0, hi = nd.s1;
215 if (lo > hi) std::swap(lo, hi);
216 expr = "clamp(" + x + ", " + scalarStr(lo) + ", " + scalarStr(hi) + ")";
217 break;
218 }
219 case OpType::MaximumScalar: expr = "max(" + x + ", " + scalarStr(nd.s0) + ")"; break;
220 case OpType::MinimumScalar: expr = "min(" + x + ", " + scalarStr(nd.s0) + ")"; break;
221 case OpType::Where:
222 expr = "select(" + operand(nd.in2) + ", " + operand(nd.in1) + ", " + operand(nd.in0) + " > 0.5)";
223 break;
224 default: return "";
225 }
226 const std::string var = "t" + std::to_string(temp++);
227 os << " var " << var << ": f32 = " << expr << ";\n";
228 if (vars.size() <= static_cast<size_t>(nodeId)) vars.resize(static_cast<size_t>(nodeId) + 1);
229 vars[static_cast<size_t>(nodeId)] = var;
230 return var;
231 }

239void emitInputIndexExprs(std::ostringstream &os, const Graph &g, const FusedGroup &grp,
240 std::vector<std::string> &indexExprs) {
241 const GraphNode &on = g.node(grp.outputNode);
242 const int rank = on.rank;
243 std::vector<int> S(static_cast<size_t>(rank), 1);
244 if (rank > 0) {
245 S[static_cast<size_t>(rank - 1)] = 1;
246 for (int k = rank - 2; k >= 0; --k) S[static_cast<size_t>(k)] = S[static_cast<size_t>(k + 1)] * on.dims[k + 1];
247 }
248 indexExprs.resize(grp.inputs.size());
249 for (size_t k = 0; k < grp.inputs.size(); ++k) {
250 const GraphNode &inN = g.node(grp.inputs[k]);
251 bool identical = inN.rank == rank;
252 if (identical) {
253 for (int d = 0; d < rank; ++d)
254 if (inN.dims[d] != on.dims[d]) {
255 identical = false;
256 break;
257 }
258 }
259 if (identical) {
260 indexExprs[k] = "i_";
261 continue;
262 }
263 const int pad = rank - inN.rank;
264 std::vector<int> stride(static_cast<size_t>(rank), 0);
265 for (int d = 0; d < rank; ++d) {
266 const int dim = d < pad ? 1 : inN.dims[d - pad];
267 if (dim == 1) continue;
268 int s = 1;
269 for (int t = d + 1; t < rank; ++t) {
270 const int dt = t < pad ? 1 : inN.dims[t - pad];
271 if (dt != 1) s *= dt;
272 }
273 stride[static_cast<size_t>(d)] = s;
274 }
275 const std::string var = "idx" + std::to_string(k);
276 os << " var " << var << ": u32 = 0u;\n";
277 for (int d = 0; d < rank; ++d) {
278 if (stride[static_cast<size_t>(d)] == 0) continue;
279 os << " " << var << " += ((i_ / " << S[static_cast<size_t>(d)] << "u) % " << on.dims[d] << "u) * "
280 << stride[static_cast<size_t>(d)] << "u;\n";
281 }
282 indexExprs[k] = var;
283 }

316 const GraphNode &B = g.node(mm.in1);

373 } else {

383 os << "}\n";

414 grp.epilogue.empty() ? std::string("r") : ctx.vars[static_cast<size_t>(grp.outputNode)];
415 os << " o[i_] = " << finalExpr << ";\n}\n";
416 out.groupsX2 = gx;
417 out.groupsY2 = gy;
418 }
419 out.pass1.clear();
420 out.pass2 = os.str();
421 out.inputCount = hasBias ? 3 : 2;
422 out.qDtype = bQuant ? B.dtype : 0;
423 out.qGroup = B.qGroup;
424 out.scalesBinding = bQuant && B.dtype != static_cast<int>(DType::Fp16) ? scalesBinding : -1;
425 out.outputBinding = bQuant ? bindingOut : -1;
426 if (tiled && batched)
427 throw eve::Exception(
428 "Tensor WGSL: unsupported kernel variant or binding count"); // batched matmul uses the naive variant
429 return;
430}

432void genConv(const Graph &g, const FusedGroup &grp, KernelSpec &out) {
433 const GraphNode &cn = g.node(grp.nodes.front());
434 const GraphNode &X = g.node(cn.in0);
435 const GraphNode &Wt = g.node(cn.in1);
436 const bool is1d = cn.type == OpType::Conv1d;
437 const int stride = cn.i0, pad = cn.i1;
438 const bool hasBias = grp.biasNode >= 0;
439 const bool hasConvBias = cn.in2 >= 0;
440 const int bindingOut = 2 + int(hasConvBias) + int(hasBias);
441
442 std::ostringstream os;
443 os << header(kLocalSize);
444 os << bufferDecl(0, "x");
445 os << bufferDecl(1, "w");
446 if (hasConvBias) os << bufferDecl(2, "convBias");
447 if (hasBias) os << bufferDecl(hasConvBias ? 3 : 2, "bias");
448 os << bufferDecl(bindingOut, "o");
449 os << pushConstant();
450 std::vector<std::string> indexExprs(grp.inputs.size(), "i_");
451 ChainContext ctx{g, grp, indexExprs, "r", os, 0, ""};
452 ctx.biasIndexExpr = is1d ? "ol" : "ow";
453
454 os << "@compute @workgroup_size(workgroupX, workgroupY)\nfn main(@builtin(global_invocation_id) globalId: "
455 "vec3<u32>, @builtin(local_invocation_id) localId: vec3<u32>, @builtin(workgroup_id) groupId: vec3<u32>) {\n";
456 os << " var i_: u32 = globalId.x;\n";
457 os << " if (i_ >= " << g.node(grp.outputNode).size << "u) { return; }\n";
458 if (is1d) {
459 const int C = X.dims[1], L = X.dims[2];
460 const int F = Wt.dims[0], K = Wt.dims[2];
461 os << " var n_: u32 = i_ / (" << F << "u * " << cn.dims[2] << "u);\n";
462 os << " var rem: u32 = i_ % (" << F << "u * " << cn.dims[2] << "u);\n";
463 os << " var f: u32 = rem / " << cn.dims[2] << "u;\n";
464 os << " var ol: u32 = rem % " << cn.dims[2] << "u;\n";
465 os << " var r: f32 = " << (hasConvBias ? "convBias[f]" : "0.0") << ";\n";
466 os << " for (var c: u32 = 0u; c < " << C << "u; c++) {\n";
467 os << " for (var kk: u32 = 0u; kk < " << K << "u; kk++) {\n";
468 os << " var il: i32 = i32(ol) * " << stride << " + i32(kk) - " << pad << ";\n";
469 os << " if (il < 0 || il >= " << L << ") { continue; }\n";
470 os << " r += x[(n_ * " << C << "u + c) * " << L << "u + u32(il)] * w[(f * " << C << "u + c) * " << K
471 << "u + kk];\n";
472 os << " }\n";
473 os << " }\n";
474 } else {
475 const int C = X.dims[1], H = X.dims[2], W = X.dims[3];
476 const int F = Wt.dims[0], KH = Wt.dims[2], KW = Wt.dims[3];
477 os << " var n_: u32 = i_ / (" << F << "u * " << cn.dims[2] << "u * " << cn.dims[3] << "u);\n";
478 os << " var rem: u32 = i_ % (" << F << "u * " << cn.dims[2] << "u * " << cn.dims[3] << "u);\n";
479 os << " var f: u32 = rem / (" << cn.dims[2] << "u * " << cn.dims[3] << "u);\n";
480 os << " var rem2: u32 = rem % (" << cn.dims[2] << "u * " << cn.dims[3] << "u);\n";
481 os << " var oh: u32 = rem2 / " << cn.dims[3] << "u;\n";
482 os << " var ow: u32 = rem2 % " << cn.dims[3] << "u;\n";
483 os << " var r: f32 = " << (hasConvBias ? "convBias[f]" : "0.0") << ";\n";
484 os << " for (var c: u32 = 0u; c < " << C << "u; c++) {\n";
485 os << " for (var kh: u32 = 0u; kh < " << KH << "u; kh++) {\n";
486 os << " var ih: i32 = i32(oh) * " << stride << " + i32(kh) - " << pad << ";\n";
487 os << " if (ih < 0 || ih >= " << H << ") { continue; }\n";
488 os << " for (var kw: u32 = 0u; kw < " << KW << "u; kw++) {\n";
489 os << " var iw: i32 = i32(ow) * " << stride << " + i32(kw) - " << pad << ";\n";
490 os << " if (iw < 0 || iw >= " << W << ") { continue; }\n";
491 os << " r += x[((n_ * " << C << "u + c) * " << H << "u + u32(ih)) * " << W << "u + u32(iw)] * w[((f * "
492 << C << "u + c) * " << KH << "u + kh) * " << KW << "u + kw];\n";
493 os << " }\n";
494 os << " }\n";
495 os << " }\n";
496 }
497 for (int u : grp.epilogue) ctx.emit(u);
498 const std::string finalExpr =
499 grp.epilogue.empty() ? std::string("r") : ctx.vars[static_cast<size_t>(grp.outputNode)];
500 os << " o[i_] = " << finalExpr << ";\n";
501 os << "}\n";
502 out.pass1.clear();
503 out.pass2 = os.str();
504 out.groupsX2 = groupsFor(g.node(grp.outputNode).size);
505 out.inputCount = 2 + int(hasConvBias) + int(hasBias);
506 return;
507}

509void genPool(const Graph &g, const FusedGroup &grp, KernelSpec &out) {
510 const GraphNode &pn = g.node(grp.outputNode);
511 const GraphNode &X = g.node(pn.in0);
512 const int ksize = pn.i0, stride = pn.i1, pad = pn.i2;
513 const int C = X.dims[1], H = X.dims[2], W = X.dims[3];
514 const int OH = pn.dims[2], OW = pn.dims[3];
515 const bool maxPool = pn.type == OpType::MaxPool2d;
516 std::ostringstream os;
517 os << header(kLocalSize);
518 os << bufferDecl(0, "in_");
519 os << bufferDecl(1, "o");
520 os << pushConstant();
521 os << "@compute @workgroup_size(workgroupX, workgroupY)\nfn main(@builtin(global_invocation_id) globalId: "
522 "vec3<u32>, @builtin(local_invocation_id) localId: vec3<u32>, @builtin(workgroup_id) groupId: vec3<u32>) {\n";
523 os << " var i_: u32 = globalId.x;\n";
524 os << " if (i_ >= " << pn.size << "u) { return; }\n";
525 os << " var n_: u32 = i_ / (" << C << "u * " << OH << "u * " << OW << "u);\n";
526 os << " var rem: u32 = i_ % (" << C << "u * " << OH << "u * " << OW << "u);\n";
527 os << " var c: u32 = rem / (" << OH << "u * " << OW << "u);\n";
528 os << " var rem2: u32 = rem % (" << OH << "u * " << OW << "u);\n";
529 os << " var oh: u32 = rem2 / " << OW << "u;\n";
530 os << " var ow: u32 = rem2 % " << OW << "u;\n";
531 os << " var acc: f32 = " << (maxPool ? "-3.402823e38" : "0.0") << ";\n";
532 os << " var valid: i32 = 0;\n";
533 os << " for (var kh: i32 = 0; kh < " << ksize << "; kh++) {\n";
534 os << " var ih: i32 = i32(oh) * " << stride << " + kh - " << pad << ";\n";
535 os << " if (ih < 0 || ih >= " << H << ") { continue; }\n";
536 os << " for (var kw: i32 = 0; kw < " << ksize << "; kw++) {\n";
537 os << " var iw: i32 = i32(ow) * " << stride << " + kw - " << pad << ";\n";
538 os << " if (iw < 0 || iw >= " << W << ") { continue; }\n";
539 os << " var v: f32 = in_[((n_ * " << C << "u + c) * " << H << "u + u32(ih)) * " << W << "u + u32(iw)];\n";
540 if (maxPool) {
541 os << " acc = max(acc, v);\n";
542 } else {
543 os << " acc += v; valid++;\n";
544 }
545 os << " }\n";
546 os << " }\n";
547 if (!maxPool) os << " if (valid > 0) { acc /= f32(valid); }\n";
548 os << " o[i_] = acc;\n";
549 os << "}\n";
550 out.pass1.clear();
551 out.pass2 = os.str();
552 out.groupsX2 = groupsFor(pn.size);
553 out.inputCount = 1;
554 return;
555}

559void generateKernelWgslImpl(const Graph &graph, const FusedGroup &group, KernelSpec &out) {
560 using namespace wgsl_detail;
561 out = KernelSpec{};
562 switch (group.kind) {
563 case GroupKind::Elementwise: return genElementwise(graph, group, out);
564 case GroupKind::MatMul:
565 // naive variant first; the runtime autotunes between naive and tiled
566 return genMatMul(graph, group, false, out);
567 case GroupKind::Conv1d:
568 case GroupKind::Conv2d: return genConv(graph, group, out);
569 case GroupKind::MaxPool2d:
570 case GroupKind::AvgPool2d: return genPool(graph, group, out);
571 case GroupKind::Softmax: return genSoftmax(graph, group, out);
572 case GroupKind::LayerNorm: return genNorm(graph, group, false, out);
573 case GroupKind::RMSNorm: return genNorm(graph, group, true, out);
574 case GroupKind::Reduce:
575 case GroupKind::ArgMax: return genReduceOrArgmax(graph, group, group.kind == GroupKind::ArgMax, out);
576 case GroupKind::Embedding: return genEmbedding(graph, group, out);
577 case GroupKind::Concat: return genConcat(graph, group, out);
578 case GroupKind::Slice: return genSlice(graph, group, out);
579 case GroupKind::Permute: return genPermute(graph, group, out);
580 case GroupKind::Resize2d: return genResize2d(graph, group, out);
581 case GroupKind::Sdpa: return genSdpa(graph, group, out);
582 case GroupKind::Alias: return; // no kernel; pure buffer alias
583 }
584 throw eve::Exception("Tensor WGSL: unsupported kernel variant or binding count");
585}

587Result<KernelSpec> generateWgslKernel(const Graph &graph, const FusedGroup &group, KernelVariant variant) {
588 try {
589 KernelSpec out;

591 if (group.kind != GroupKind::MatMul)

600 } catch (const std::exception &error) {
601 return Result<KernelSpec>::failure(
602 Diagnostic::error(DiagnosticCode::Unsupported, error.what(), "tensor.kernel"));
603 }
604}

621// Specialize only our generated declarations and identifier tokens, never arbitrary user shaders.
622void specializeSource(std::string &source, const std::vector<int> &representatives, int inputCount) {
623 std::vector<std::string> names(static_cast<size_t>(inputCount));
624 std::map<std::string, std::string> aliases;
625 for (int i = 0; i < inputCount; ++i) {
626 const std::string prefix = "@group(0) @binding(" + std::to_string(i) + ") var<storage, read_write> ";
627 const auto begin = source.find(prefix);
628 if (begin == std::string::npos) throw eve::Exception("Tensor WGSL: missing input declaration");
629 const auto nameBegin = begin + prefix.size();
630 const auto nameEnd = source.find(':', nameBegin);
631 names[static_cast<size_t>(i)] = source.substr(nameBegin, nameEnd - nameBegin);
632 if (representatives[static_cast<size_t>(i)] == i) continue;
633 aliases.emplace(names[static_cast<size_t>(i)],
634 names[static_cast<size_t>(representatives[static_cast<size_t>(i)])]);
635 source.erase(begin, source.find('\n', nameEnd) + 1 - begin);
636 }
637 if (aliases.empty()) return;
638 std::string renamed;
639 for (size_t i = 0; i < source.size();) {
640 const unsigned char ch = static_cast<unsigned char>(source[i]);
641 if (std::isalpha(ch) || ch == '_') {
642 const auto begin = i++;
643 while (i < source.size() && (std::isalnum(static_cast<unsigned char>(source[i])) || source[i] == '_')) ++i;
644 const std::string word = source.substr(begin, i - begin);
645 const auto found = aliases.find(word);
646 renamed += found == aliases.end() ? word : found->second;
647 } else {
648 renamed += source[i++];
649 }
650 }
651 source = std::move(renamed);
652}
653} // namespace

655void specializeInputBindings(const Graph &graph, const FusedGroup &group, KernelSpec &spec) {
656 std::map<int, int> first;
657 for (int i = 0; i < spec.inputCount; ++i) {

659 (void)inserted;
660 spec.inputRepresentatives.push_back(it->second);
661 }
662 if (spec.twoPass) specializeSource(spec.pass1, spec.inputRepresentatives, spec.inputsReadPass1);
663 if (!spec.pass2.empty()) specializeSource(spec.pass2, spec.inputRepresentatives, spec.inputCount);

669void genEmbedding(const Graph &g, const FusedGroup &grp, KernelSpec &out) {
670 const GraphNode &en = g.node(grp.outputNode);
671 const GraphNode &T = g.node(en.in0);
672 const bool tQuant = q::isQuantDType(static_cast<DType>(T.dtype)) && !T.constBytes.empty();
673 const bool tInt = T.dtype != static_cast<int>(DType::Fp16);
674 const int vocab = T.dims[0], dim = T.dims[1];
675 std::ostringstream os;
676 os << header(kLocalSize);
677 if (tQuant)
678 os << bufferDeclUint(0, "table");
679 else
680 os << bufferDecl(0, "table");
681 os << bufferDecl(1, "idx");
682 if (tQuant && tInt) os << bufferDecl(2, "bs");
683 os << bufferDecl(tQuant ? 3 : 2, "o");
684 os << pushConstant();
685 if (tQuant) os << emitQuantizedBVal(static_cast<DType>(T.dtype), T.qGroup, "table");
686 os << "@compute @workgroup_size(workgroupX, workgroupY)\nfn main(@builtin(global_invocation_id) globalId: "
687 "vec3<u32>, @builtin(local_invocation_id) localId: vec3<u32>, @builtin(workgroup_id) groupId: vec3<u32>) {\n";
688 os << " var i_: u32 = globalId.x;\n";
689 os << " if (i_ >= " << en.size << "u) { return; }\n";
690 os << " var r: u32 = i_ / " << dim << "u;\n";
691 os << " var d: u32 = i_ % " << dim << "u;\n";
692 os << " var ii: i32 = i32(idx[r]);\n";
693 os << " ii = clamp(ii, 0, " << (vocab - 1) << ");\n";
694 os << " o[i_] = "
695 << (tQuant ? "bval(u32(ii) * " + std::to_string(dim) + "u + d)"
696 : "table[u32(ii) * " + std::to_string(dim) + "u + d]")
697 << ";\n";
698 os << "}\n";
699 out.pass1.clear();
700 out.pass2 = os.str();
701 out.groupsX2 = groupsFor(en.size);
702 out.inputCount = 2;
703 out.qDtype = tQuant ? T.dtype : 0;
704 out.qGroup = T.qGroup;
705 out.scalesBinding = tQuant && tInt ? 2 : -1;
706 out.outputBinding = tQuant ? 3 : -1;
707 return;
708}

710void genConcat(const Graph &g, const FusedGroup &grp, KernelSpec &out) {
711 const GraphNode &cn = g.node(grp.outputNode);
712 const int axis = cn.i0;
713 const int n = int(grp.inputs.size());
714 if (n < 2 || n > 4) throw eve::Exception("Tensor WGSL: unsupported kernel variant or binding count");
715 int starts[4] = {};
716 int axisTotal = 0;
717 for (int k = 0; k < n; ++k) {
718 starts[k] = axisTotal;
719 axisTotal += g.node(grp.inputs[static_cast<size_t>(k)]).dims[axis];
720 }
721 int inner = 1;
722 for (int k = axis + 1; k < cn.rank; ++k) inner *= cn.dims[k];
723 std::ostringstream os;
724 os << header(kLocalSize);
725 for (int k = 0; k < n; ++k) os << bufferDecl(k, ("a" + std::to_string(k)).c_str());
726 os << bufferDecl(n, "o");
727 os << pushConstant();
728 os << "@compute @workgroup_size(workgroupX, workgroupY)\nfn main(@builtin(global_invocation_id) globalId: "
729 "vec3<u32>, @builtin(local_invocation_id) localId: vec3<u32>, @builtin(workgroup_id) groupId: vec3<u32>) {\n";
730 os << " var i_: u32 = globalId.x;\n";
731 os << " if (i_ >= " << cn.size << "u) { return; }\n";
732 os << " var ax: u32 = (i_ / " << inner << "u) % " << axisTotal << "u;\n";
733 os << " var op: u32 = i_ / (" << axisTotal << "u * " << inner << "u);\n";
734 os << " var ip: u32 = i_ % " << inner << "u;\n";
735 os << " var v: f32 = 0.0;\n";
736 for (int k = 0; k < n; ++k) {
737 const int sz = g.node(grp.inputs[static_cast<size_t>(k)]).dims[axis];
738 const char *cond = k == 0 ? "if" : "else if";
739 os << " " << cond << " (ax >= " << starts[k] << "u && ax < " << starts[k] + sz << "u) {\n";
740 os << " v = a" << k << "[op * " << sz << "u * " << inner << "u + (ax - " << starts[k] << "u) * " << inner
741 << "u + ip];\n";
742 os << " }\n";
743 }
744 os << " o[i_] = v;\n";
745 os << "}\n";
746 out.pass1.clear();
747 out.pass2 = os.str();
748 out.groupsX2 = groupsFor(cn.size);
749 out.inputCount = n;
750 return;
751}

782void genPermute(const Graph &g, const FusedGroup &grp, KernelSpec &out) {
783 const GraphNode &pn = g.node(grp.outputNode);
784 const GraphNode &X = g.node(pn.in0);
785 const int rank = pn.rank;
786 int S[Tensor::kMaxRank] = {};
787 S[rank - 1] = 1;
788 for (int k = rank - 2; k >= 0; --k) S[k] = S[k + 1] * pn.dims[k + 1];
789 int inStride[Tensor::kMaxRank] = {};
790 inStride[rank - 1] = 1;
791 for (int k = rank - 2; k >= 0; --k) inStride[k] = inStride[k + 1] * X.dims[k + 1];
792 std::ostringstream os;
793 os << header(kLocalSize);
794 os << bufferDecl(0, "in_");
795 os << bufferDecl(1, "o");
796 os << pushConstant();
797 os << "@compute @workgroup_size(workgroupX, workgroupY)\nfn main(@builtin(global_invocation_id) globalId: "
798 "vec3<u32>, @builtin(local_invocation_id) localId: vec3<u32>, @builtin(workgroup_id) groupId: vec3<u32>) {\n";
799 os << " var i_: u32 = globalId.x;\n";
800 os << " if (i_ >= " << pn.size << "u) { return; }\n";
801 os << " var idx: u32 = 0u;\n";
802 for (int k = 0; k < rank; ++k) {
803 const int inAxis = pn.perm[k];
804 os << " idx += ((i_ / " << S[k] << "u) % " << pn.dims[k] << "u) * " << inStride[inAxis] << "u;\n";
805 }
806 os << " o[i_] = in_[idx];\n";
807 os << "}\n";
808 out.pass1.clear();
809 out.pass2 = os.str();
810 out.groupsX2 = groupsFor(pn.size);
811 out.inputCount = 1;
812 return;
813}

815void genResize2d(const Graph &g, const FusedGroup &grp, KernelSpec &out) {
816 const GraphNode &rn = g.node(grp.outputNode);
817 const GraphNode &X = g.node(rn.in0);
818 const int H = X.dims[2], W = X.dims[3];
819 const int OH = rn.dims[2], OW = rn.dims[3];
820 const bool nearest = rn.i0 == 0;
821 std::ostringstream os;
822 os << header(kLocalSize);
823 os << bufferDecl(0, "in_");
824 os << bufferDecl(1, "o");
825 os << pushConstant();
826 os << "@compute @workgroup_size(workgroupX, workgroupY)\nfn main(@builtin(global_invocation_id) globalId: "
827 "vec3<u32>, @builtin(local_invocation_id) localId: vec3<u32>, @builtin(workgroup_id) groupId: vec3<u32>) {\n";
828 os << " var i_: u32 = globalId.x;\n";
829 os << " if (i_ >= " << rn.size << "u) { return; }\n";
830 os << " var ow: u32 = i_ % " << OW << "u;\n";
831 os << " var rem: u32 = i_ / " << OW << "u;\n";
832 os << " var oh: u32 = rem % " << OH << "u;\n";
833 os << " var rem2: u32 = rem / " << OH << "u;\n";
