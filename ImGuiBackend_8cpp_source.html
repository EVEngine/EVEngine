276void ImGuiBackend::shutdown() {
277 if (!initialized_) return;
278 if (frameOpen_) {
279 if (GImGui && GImGui->WithinFrameScope) ImGui::EndFrame();
280 frameOpen_ = false;
281 }
282 if (gfx_) {
283 if (gfx_->getPresentOverlayUser() == this) gfx_->setPresentOverlay(nullptr, nullptr);
284 }
285#ifdef EVENGINE_WEBGPU
286 ImGui_ImplWGPU_Shutdown();
287 ImGui_ImplSDL2_Shutdown();
288 ImGui::DestroyContext();
289#else
290 auto *vkg = dynamic_cast<eve::graphics::vulkan::Graphics *>(gfx_);
291 if (vkg) {
292 vkDeviceWaitIdle(static_cast<VkDevice>(vkg->getDevice().instance));
293 }
294 // The ImGui backend data lives on the context this backend created. It may
295 // not be the currently active context (headless tests switch contexts), so
296 // explicitly select it before tearing down ImGui_Impl* state.
297 if (ctx_) {
298 ::ImGuiContext *prev = ::ImGui::GetCurrentContext();
299 ::ImGuiContext *mine = ctx_;
300 ::ImGui::SetCurrentContext(mine);
301 ::ImGui_ImplVulkan_Shutdown();
302 ::ImGui_ImplSDL2_Shutdown();
303 ::ImGui::DestroyContext(mine);
304 ctx_ = nullptr;
305 // Restore the previous current context unless it was the one we just
306 // destroyed (DestroyContext already reset it to null).
307 if (prev && prev != mine) ::ImGui::SetCurrentContext(prev);
308 else ::ImGui::SetCurrentContext(nullptr);
309 }
310 if (imguiDescriptorPool_ && vkg) {
311 vkDestroyDescriptorPool(static_cast<VkDevice>(vkg->getDevice().instance),
312 static_cast<VkDescriptorPool>(imguiDescriptorPool_), nullptr);
313 imguiDescriptorPool_ = nullptr;
314 }
315 if (imguiTexturePool_ && vkg) {
316 vkDestroyDescriptorPool(static_cast<VkDevice>(vkg->getDevice().instance),
317 static_cast<VkDescriptorPool>(imguiTexturePool_), nullptr);
318 imguiTexturePool_ = nullptr;
319 }
320 if (imguiTextureLayout_ && vkg) {
321 vkDestroyDescriptorSetLayout(static_cast<VkDevice>(vkg->getDevice().instance),
322 static_cast<VkDescriptorSetLayout>(imguiTextureLayout_),
323 nullptr);
324 imguiTextureLayout_ = nullptr;
325 }
326#endif
327 textures_.clear();
328 queuedTextureDraws_.clear();
329 gfx_ = nullptr;
330 window_ = nullptr;
331 fontsUploaded_ = false;
332 initialized_ = false;
333}

375float ImGuiBackend::computeInitialScale() const {
376#if defined(EVENGINE_ANDROID) || defined(EVENGINE_IOS)
377 float ddpi = 160.f;
378 if (SDL_GetDisplayDPI(0, &ddpi, nullptr, nullptr) != 0 || ddpi < 1.f) ddpi = 320.f;
379 float s = ddpi / 160.f;
380 return std::clamp(s, 1.75f, 3.25f);
381#elif defined(_WIN32)
382 // A DPI-aware SDL 2.0 window uses physical pixels for both its window and
383 // drawable size, so their ratio stays 1 even at 125%/150% Windows scale.
384 // Scale ImGui geometry explicitly to retain the OS-requested UI size.
385 return computeDpiScale();
386#else
387 // Desktop: ImGui's backend already applies the display density via
388 // io.DisplayFramebufferScale, so the logical (point-space) UI scale stays
389 // 1.0. Keeping it constant makes the UI match the OS UI size on
390 // high-resolution displays instead of growing with the pixel ratio.
391 (void)window_;
392 return 1.f;
393#endif
394}

396float ImGuiBackend::computeDpiScale() const {
397#if defined(_WIN32)
398 const int display = window_ ? SDL_GetWindowDisplayIndex(window_) : 0;
399 float ddpi = 96.f;
400 if (display >= 0 && SDL_GetDisplayDPI(display, &ddpi, nullptr, nullptr) == 0 && ddpi > 0.f)
401 return std::clamp(ddpi / 96.f, 1.f, 4.f);
402 return 1.f;
403#else
404 int logicalW = 0, logicalH = 0, pixelW = 0, pixelH = 0;
405 SDL_GetWindowSize(window_, &logicalW, &logicalH);
406#ifdef EVENGINE_WEBGPU
407 // No GL context on WebGPU; drawable size == window size for the canvas.
408 SDL_GetWindowSize(window_, &pixelW, &pixelH);
409#else
410 SDL_Vulkan_GetDrawableSize(window_, &pixelW, &pixelH);
411#endif
412 if (logicalW > 0 && pixelW > 0) {
413 float s = float(pixelW) / float(logicalW);
414 if (s > 0.f) return std::clamp(s, 1.f, 4.f);
415 }
416 return 1.f;
417#endif
418}

420void ImGuiBackend::loadFonts() {
421 ImGuiIO &io = ImGui::GetIO();
422 ImFontAtlas *atlas = io.Fonts;
423 if (!atlas) return;
424
425 atlas->Clear();
426
427 // Rasterize at the physical DPI resolution so glyphs stay crisp; the
428 // FontGlobalScale set in applyThemeToImGui cancels this so the logical
429 // text size stays constant regardless of display density.
430 const float sizePx = kBaseFontSizePx * dpiScale_;
431
432 ImFontConfig cfg{};
433 cfg.OversampleH = 3;
434 cfg.OversampleV = 1;
435 cfg.PixelSnapH = true;
436 // A small coverage boost gives Segoe UI's thin strokes enough contrast
437 // after the physical-DPI atlas is scaled back into logical coordinates.
438 cfg.RasterizerMultiply = 1.12f;
439 // Request CJK ranges so the merged CJK font below rasterizes Chinese,
440 // Japanese and Korean text (the atlas grows by a few MB — acceptable).
441 fontRanges_.clear();
442 ImFontGlyphRangesBuilder rangeBuilder;
443 rangeBuilder.AddRanges(atlas->GetGlyphRangesDefault());
444 rangeBuilder.AddRanges(atlas->GetGlyphRangesChineseFull());
445 rangeBuilder.BuildRanges(&fontRanges_);
446 cfg.GlyphRanges = fontRanges_.Data;
447
448 bool added = false;
449 std::string primaryPath;
450 for (const char *path : regularFontCandidates()) {
451 if (!fileExists(path)) continue;
452 if (atlas->AddFontFromFileTTF(path, sizePx, &cfg)) {
453 primaryPath = path;
454 added = true;
455 break;
456 }
457 }
458 if (!added) atlas->AddFontDefault(&cfg);
459
460 // The first available regular font is usually Latin-only (Segoe UI,
461 // DejaVu Sans, Helvetica), so requesting CJK ranges on it adds no Chinese
462 // glyphs to the atlas. Merge a CJK-capable system font into the same atlas
463 // (same MergeMode pattern as the icon font below) so Chinese/Japanese text
464 // renders instead of ImGui's '?' fallback glyph.
465 //
466 // Glyph coverage cannot be verified until the atlas is rasterized (both
467 // ImFont::Glyphs and the IndexLookup table are populated during Build), so
468 // pick by file identity: every cjkFontCandidates() entry is CJK-capable,
469 // and the first existing one is the best match for the platform. Whether
470 // the merge actually covered CJK is checked after upload in
471 // checkFontCoverage().
472 cjkRanges_.clear();
473 ImFontGlyphRangesBuilder cjkRangeBuilder;
474 cjkRangeBuilder.AddRanges(atlas->GetGlyphRangesChineseFull());
475 cjkRangeBuilder.BuildRanges(&cjkRanges_);
476 bool primaryIsCjk = false;
477 for (const char *path : cjkFontCandidates()) {
478 if (primaryPath == path) {
479 primaryIsCjk = true;
480 break;
481 }
482 }
483 if (!primaryIsCjk) {
484 ImFontConfig cjkCfg{};
485 // CJK glyphs are dense (ChineseFull is ~20k+ codepoints); the physical
486 // size already scales with DPI, so oversampling 1 keeps the atlas sane.
487 cjkCfg.OversampleH = 1;
488 cjkCfg.OversampleV = 1;
489 cjkCfg.MergeMode = true;
490 cjkCfg.GlyphRanges = cjkRanges_.Data;
491 for (const char *path : cjkFontCandidates()) {
492 if (!fileExists(path)) continue;
493 if (atlas->AddFontFromFileTTF(path, sizePx, &cjkCfg, cjkRanges_.Data)) break;
494 }
495 }
496
497 ImFontConfig iconCfg{};
498 iconCfg.OversampleH = 2;
499 iconCfg.OversampleV = 2;
500 iconCfg.MergeMode = true;
501 iconCfg.PixelSnapH = true;
502 static const ImWchar iconRanges[] = {0xF000, 0xF8FF, 0};
503 for (const std::string &path : iconFontCandidates()) {
504 if (!fileExists(path.c_str())) continue;
505 if (atlas->AddFontFromFileTTF(path.c_str(), sizePx, &iconCfg, iconRanges)) break;
506 }
507}

509void ImGuiBackend::checkFontCoverage() const {
510 // Only valid after the atlas has been built/uploaded (lookup tables exist).
511 ImFontAtlas *atlas = ImGui::GetIO().Fonts;
512 if (!atlas || atlas->Fonts.Size == 0) return;
513 const ImFont *font = atlas->Fonts[0];
514 if (font && font->FindGlyphNoFallback(kCjkProbeCodepoint) == nullptr)
515 fprintf(stderr,
516 "[ui] warning: no CJK-capable system font found; Chinese text will render as '?'\n");
517 if (font && font->FindGlyphNoFallback(kIconProbeCodepoint) == nullptr)
518 fprintf(stderr,
519 "[ui] warning: editor icon font not found; semantic icons will render as '?'\n");
520}

545uint64_t ImGuiBackend::registerTexture(graphics::Texture *tex) {
546 if (!initialized_ || !tex || !tex->gpuHandle) return 0;
547 RegisteredTexture reg;
548 reg.texture = tex;
549#ifdef EVENGINE_WEBGPU
550 auto *gt = static_cast<eve::graphics::webgpu::GpuTexture *>(tex->gpuHandle);
551 reg.imId = (ImTextureID)(intptr_t)gt->view.Get();
552 if (!reg.imId) return 0;

605 vkUpdateDescriptorSets(device, 1, &write, 0, nullptr);
606 reg.imId = set;
607#endif
608 reg.width = tex->getWidth();
609 reg.height = tex->getHeight();
610 const uint64_t key = nextTextureKey_++;
611 textures_[key] = reg;

675void ImGuiBackend::renderDrawData(void *commandBuffer) {
676 if (!initialized_ || !commandBuffer) return;
677 if (!GImGui || !GImGui->WithinFrameScope) return;
678 ImGui::Render();
679 frameOpen_ = false;
680#ifdef EVENGINE_WEBGPU
681 WGPURenderPassEncoder enc = *static_cast<WGPURenderPassEncoder *>(commandBuffer);
682 ImGui_ImplWGPU_RenderDrawData(ImGui::GetDrawData(), enc);
683#else
684 ImGui_ImplVulkan_RenderDrawData(ImGui::GetDrawData(),
685 static_cast<VkCommandBuffer>(commandBuffer));
686 auto *vkg = dynamic_cast<eve::graphics::vulkan::Graphics *>(gfx_);
687 if (vkg && !queuedTextureDraws_.empty()) {
688 std::vector<eve::graphics::vulkan::UiTextureDraw> draws;
689 draws.reserve(queuedTextureDraws_.size());
690 for (const QueuedTextureDraw &queued : queuedTextureDraws_) {
691 const auto found = textures_.find(queued.id);
692 if (found == textures_.end() || !found->second.texture) continue;
693 eve::graphics::vulkan::UiTextureDraw draw;
694 draw.texture = found->second.texture;
695 draw.x = queued.x;
696 draw.y = queued.y;
697 draw.w = queued.w;
698 draw.h = queued.h;
699 draw.u0 = queued.u0;
700 draw.v0 = queued.v0;
701 draw.u1 = queued.u1;
702 draw.v1 = queued.v1;
703 draw.tint = eve::graphics::Color(queued.r, queued.g, queued.b, queued.a);
704 draw.clipX = queued.clipX;
705 draw.clipY = queued.clipY;
706 draw.clipW = queued.clipW;
707 draw.clipH = queued.clipH;
708 draw.opaque = queued.opaque;
709 draws.push_back(draw);
710 }
711 vkg->drawUiTextureRects(commandBuffer, draws);
712 }
713#endif
714}
