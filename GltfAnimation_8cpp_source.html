70std::uint32_t integerComponent(const Accessor& a, std::uint32_t row, std::uint32_t column) {
71 require(a.componentType == 5121 || a.componentType == 5123, "JOINTS must use unsigned bytes or shorts");
72 auto offset = a.offset + row * a.stride + column * componentSize(a.componentType);
