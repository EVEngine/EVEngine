28Value::Object collisionPayload(TileLayerRef layer, const TileCollisionRect* rects, size_t count) {
29 Value::Array encoded;
30 encoded.reserve(count);
31 for (size_t index = 0; index < count; ++index) {
32 const TileCollisionRect& rect = rects[index];
33 encoded.push_back(Value::object({{"x", Value::number(double(rect.x))},
34 {"y", Value::number(double(rect.y))},
35 {"width", Value::number(double(rect.width))},
36 {"height", Value::number(double(rect.height))}}));
37 }
38 return {{"layer", Value::object({{"id", Value::integer(static_cast<std::int64_t>(layer.slot))},
39 {"generation", Value::integer(static_cast<std::int64_t>(layer.generation))}})},
40 {"rects", Value::array(std::move(encoded))}};
41}

50class ScriptTileCollisionSink final : public ITileCollisionSink {
51public:
52 explicit ScriptTileCollisionSink(ScriptProviderHandle handle) : handle_(std::move(handle)) {}
53
55 [[nodiscard]] const ScriptProviderHandle& handle() const noexcept { return handle_; }
56
57 void replaceTileCollision(TileLayerRef layer, const TileCollisionRect* rects, size_t count) override {
58 const ScriptArg args[] = {ScriptArg::fromTable(collisionPayload(layer, rects, count))};
59 const auto published = ScriptProviderRegistry::instance().call(handle_, "replaceTileCollision", args);
60 if (published.ok()) return;
61
62 // 接口是 `void`：失败没有返回值可放，所以**计数 + 上报**（每一代只写一行日志，避免
63 // 地图重建循环里把日志刷满；计数是每一次都记，测试与诊断要看得见真实次数）。
64 ++scriptTileCollisionSinkFailureCount();
65 if (reportedGeneration_ == handle_.generation) return;
66 reportedGeneration_ = handle_.generation;
67 recordLogEvent("warn", "script tile collision sink '" + handle_.name +
68 "' did not accept the geometry for layer " + std::to_string(layer.slot) + ": " +
69 published.status().describe());
70 }
71
72private:
74 ScriptProviderHandle handle_;
76 std::uint64_t reportedGeneration_ = 0;
77};

96EVENGINE_API_WORLD void declareScriptTileCollisionSinkSeam() {
97 ScriptTileCollisionSinks& owned = scriptTileCollisionSinks();
98 ScriptProviderRegistry::instance().declareCapability(
99 ITileCollisionSink::capabilityName,
100 [&owned](const ScriptProviderHandle& handle) -> Result<void> {
101 auto adapter = std::make_unique<ScriptTileCollisionSink>(handle);
102 // 先注册再交所有权：从这一刻起 listener 表持有裸指针，`owned.live` 是让它有效的唯一原因。
103 cap::addListener<ITileCollisionSink>(adapter.get());
104 owned.live.push_back(std::move(adapter));
105 return Result<void>::success();
106 },
107 [&owned](const ScriptProviderHandle& handle) -> Result<void> {
108 const auto match = std::find_if(owned.live.begin(), owned.live.end(),
109 [&handle](const std::unique_ptr<ScriptTileCollisionSink>& adapter) {
110 return adapter->handle() == handle;
111 });
112 if (match == owned.live.end()) {
113 // 幂等：后置条件是"这一代没有适配器"，模块退出时可能已经撤过了。
114 return Result<void>::success();
115 }
116 cap::removeListener<ITileCollisionSink>(match->get());
117 owned.retired.push_back(std::move(*match));
118 owned.live.erase(match);
119 return Result<void>::success();
120 });
121}

123EVENGINE_API_WORLD void withdrawScriptTileCollisionSinkSeam() {
124 ScriptTileCollisionSinks& owned = scriptTileCollisionSinks();
125 for (auto& adapter : owned.live) {
126 cap::removeListener<ITileCollisionSink>(adapter.get());
127 owned.retired.push_back(std::move(adapter));
128 }
129 owned.live.clear();
130
131 // 连声明一起撤：模块不在了还留着接缝，只会让脚本发布出一个永远不会被派发的适配器。
132 ScriptProviderRegistry::instance()
133 .removeCapability(ITileCollisionSink::capabilityName)
134 .ignore("a module may be destroyed without ever having declared the seam");
135}
