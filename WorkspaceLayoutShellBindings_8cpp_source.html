14bool mountWorkspaceShell(ui::UI* uiModule, editor::EditorWorkspace* workspace,
15 const std::string& hostName) {
16 if (!uiModule || !workspace) return false;
17 WorkspaceShellOptions options;
18 options.windowId = "root";
19 auto toolbar = ui::toolbar(
20 {ui::text(workspace->getTitle(), "shell-title"), ui::spacer("shell-fill"),
21 ui::badge(workspace->getMode(), "shell-mode")},
22 "shell-toolbar");
23 auto status = ui::statusBar(
24 {ui::text("panels=" + std::to_string(workspace->getPanelCount()), "shell-panels"),
25 ui::spacer("shell-status-fill"),
26 ui::text(workspace->getActivePanel().empty() ? "no active panel"
27 : ("active=" + workspace->getActivePanel()),
28 "shell-active")},
29 "shell-status");
30 options.toolbar = std::move(toolbar);
31 options.statusBar = std::move(status);
32 ui::WidgetDesc root = WorkspaceLayoutShell::buildWindow(*workspace, {}, std::move(options));
33 const auto handle = uiModule->mountAs(hostName.empty() ? "workspace-shell" : hostName,
34 std::move(root));
35 if (auto host = ui::UIHost::resolve(handle)) {
36 // Lock chrome: fixed IDE layout, not freely draggable panels.
37 auto meta = host->get().meta();
38 meta->hasPos = true;
39 meta->posX = 0.f;
40 meta->posY = 0.f;
41 meta->lockPos = true;
42 meta->hasSize = true;
43 meta->lockSize = true;
44 meta->percentW = 1.f;
45 meta->percentH = 1.f;
46 return true;
47 }
48 return false;
49}
