22bool sameFiles(const Files& left, const Files& right) {
23 if (left.size() != right.size()) return false;
24 for (const auto& [path, file] : left) {
25 const auto other = right.find(path);
26 if (other == right.end() || file->hash != other->second->hash) return false;
27 }
28 return true;
29}
30State& stateFor(const std::filesystem::path& root) {
31 const auto key = pathText(root);
32 if (!states.contains(key) && states.size() >= 4) {
33 const auto oldest = std::min_element(states.begin(), states.end(), [](const auto& left, const auto& right) {
34 return left.second.used < right.second.used;
35 });
36 states.erase(oldest);
37 }
38 auto& state = states[key];
39 state.used = ++sequence;
40 return state;
41}
42void refresh(const std::filesystem::path& root, State& state) {
43 Object::Ptr stats = new Object();
44 auto files = scan(root, state.snapshot.files, stats);
45 if (!state.snapshot.revision.empty() && sameFiles(files, state.snapshot.files)) {
46 state.snapshot.stats = stats;
47 return;
48 }
49 auto candidate = project(std::move(files), stats, pathText(root.filename()));
50 candidate.projectDir = pathText(root);
51 // Detect a changed read set before publication; this is not a filesystem
52 // transaction. Consumers must recheck revision before a later action.
53 Object::Ptr confirmation = new Object();
54 if (!sameFiles(candidate.files, scan(root, candidate.files, confirmation)))
55 throw std::runtime_error("Project changed during indexing; retry");
56 state.snapshot = std::move(candidate);
57}
58Object::Ptr query(const Snapshot& snapshot, std::string_view mode, Object::Ptr args) {
59 const int limit = args->optValue<int>("maxNodes", 64), depth = args->optValue<int>("maxDepth", 8);
60 const int bytes = args->optValue<int>("maxBytes", 32768);
61 if (limit < 1 || limit > 256 || depth < 0 || depth > 8 || bytes < 1024 || bytes > 65536)
62 return mcpFailureObject("invalid_query", "maxNodes 1..256, maxDepth 0..8, maxBytes 1024..65536");
63 const auto expected = args->optValue<std::string>("expectedRevision", "");
64 if (!expected.empty() && expected != snapshot.revision) {
65 auto failure = mcpFailureObject("stale_project_revision", "Refresh project evidence and use the new revision");
66 failure->set("revision", snapshot.revision);
67 return failure;
68 }
69 Object::Ptr out = new Object();
70 out->set("ok", true);
71 out->set("schema", mode == "eve_task_context" ? "eve.project-context" : "eve.project-index");
72 out->set("schemaVersion", snapshot.automatic ? 2 : 1);
73 out->set("projectId", snapshot.projectId);
74 out->set("projectDir", snapshot.projectDir);
75 out->set("revision", snapshot.revision);
76 if (snapshot.automatic) out->set("analyzerHash", std::string(eve::script::scriptStructureProviderHash()));
77 out->set("declarationsResolved", snapshot.diagnostics.empty());
78 out->set("coverage", "partial-project");
79 out->set("evidence",
80 snapshot.automatic ? "compiler-structure+engine-contracts" : "source-metadata+checked-declarations");
81 out->set("untrustedProjectContent", true);
82 out->set("includeCalls", args->optValue<bool>("includeCalls", false));
83 out->set("limitations",
84 "Source evidence, not live state or runtime causality; symbol extraction is not compilation. Structural "
85 "addresses may change; declared semantic IDs survive unique source relocation.");
86 Array::Ptr diagnostics = new Array();
87 for (const auto& item : snapshot.diagnostics) diagnostics->add(item);
88 out->set("diagnostics", diagnostics);
89 out->set("stats", snapshot.stats);
90 Array::Ptr nodes = new Array(), edges = new Array(), contracts = new Array(), tests = new Array();
91 std::set<std::string> selected;
92 bool truncated = false;
93 if (mode == "eve_project_index_refresh") {
94 for (const auto& [id, item] : snapshot.nodes)
95 if (item->optValue<bool>("semantic", false) ||
96 (item->has("evidence") && (item->getValue<std::string>("evidence") == "project-declaration" ||
97 item->getValue<std::string>("evidence") == "engine-build-contract"))) {
98 if (selected.size() < static_cast<size_t>(limit))
99 selected.insert(id);
100 else
101 truncated = true;
102 }
103 } else {
104 const auto seed = args->optValue<std::string>("seed", "");
105 if (mode == "eve_task_context" && (seed.empty() || args->optValue<std::string>("task", "").empty()))
106 return mcpFailureObject("invalid_query", "Project task context requires seed and task");
107 if (!seed.empty()) {
108 if (!snapshot.nodes.contains(seed)) return mcpFailureObject("unknown_seed");
109 std::map<std::string, std::vector<size_t>> adjacency;
110 for (size_t i = 0; i < snapshot.edges.size(); ++i) {
111 const auto& edge = snapshot.edges[i];
112 adjacency[edge->getValue<std::string>("from")].push_back(i);
113 adjacency[edge->getValue<std::string>("to")].push_back(i);
114 }
115 std::queue<std::pair<std::string, int>> pending;
116 selected.insert(seed);
117 pending.emplace(seed, 0);
118 while (!pending.empty()) {
119 const auto [current, level] = pending.front();
120 pending.pop();
121 if (snapshot.automatic && !args->optValue<bool>("includeCalls", false) &&
122 snapshot.nodes.at(current)->getValue<std::string>("kind") == "test")
123 continue;
124 for (const auto i : adjacency[current]) {
125 const auto& edge = snapshot.edges[i];
126 if (snapshot.automatic && !args->optValue<bool>("includeCalls", false) &&
127 snapshot.nodes.at(current)->getValue<std::string>("kind") == "engine_api" &&
128 edge->getValue<std::string>("relation") != "tested_by")
129 continue;
130 if (snapshot.automatic && !args->optValue<bool>("includeCalls", false) &&
131 edge->getValue<std::string>("relation") == "calls")
132 continue;
133 const auto from = edge->getValue<std::string>("from"), to = edge->getValue<std::string>("to");
134 const auto next = from == current ? to : from;
135 if (selected.contains(next)) continue;
136 if (level >= depth || selected.size() >= static_cast<size_t>(limit)) {
137 truncated = true;
138 continue;
139 }
140 selected.insert(next);
141 pending.emplace(next, level + 1);
142 }
143 }
144 } else {
145 const auto kind = args->optValue<std::string>("kind", ""), text = args->optValue<std::string>("query", "");
146 for (const auto& [id, item] : snapshot.nodes) {
147 if (!kind.empty() && item->getValue<std::string>("kind") != kind) continue;
148 if (!text.empty() && id.find(text) == std::string::npos &&
149 item->getValue<std::string>("label").find(text) == std::string::npos)
150 continue;
151 if (selected.size() >= static_cast<size_t>(limit)) {
152 truncated = true;
153 continue;
154 }
155 selected.insert(id);
156 }
157 }
158 }
159 for (const auto& id : selected) {
160 const auto& item = snapshot.nodes.at(id);
161 if (snapshot.automatic && mode == "eve_project_index_refresh") {
162 Object::Ptr summary = new Object();
163 for (const auto* field : {"id", "kind", "label", "status", "source", "evidence", "schemaContract"})
164 if (item->has(field)) summary->set(field, item->get(field));
165 nodes->add(summary);
166 if (item->getValue<std::string>("kind") == "test") tests->add(summary);
167 continue;
168 }
169 if (snapshot.automatic && item->has("contract")) {
170 Object::Ptr indexed = new Object();
171 for (const auto& field : *item)
172 if (field.first != "contract") indexed->set(field.first, field.second);
173 indexed->set("contractKey", id.substr(4));
174 nodes->add(indexed);
175 } else
176 nodes->add(item);
177 if (item->has("contract")) contracts->add(item->get("contract"));
178 if (item->getValue<std::string>("kind") == "test") tests->add(item);
179 }
180 for (const auto& edge : snapshot.edges)
181 if (!(snapshot.automatic &&
182 (mode == "eve_project_index_refresh" ||
183 (!args->optValue<bool>("includeCalls", false) && edge->getValue<std::string>("relation") == "calls"))) &&
184 selected.contains(edge->getValue<std::string>("from")) &&
185 selected.contains(edge->getValue<std::string>("to")))
186 edges->add(edge);
187 out->set("nodes", nodes);
188 out->set("edges", edges);
189 out->set("contracts", contracts);
190 out->set("tests", tests);
191 out->set("truncated", truncated);
192 if (mode == "eve_task_context") out->set("task", args->getValue<std::string>("task"));
193 if (mcpStringify(out).size() > static_cast<size_t>(bytes))
194 return mcpFailureObject("context_budget_exceeded",
195 "Reduce maxNodes/depth or increase maxBytes; no partial evidence returned");
196 return out;
197}

202std::string callMcpProjectIndexTool(std::string_view name, Object::Ptr args) {
203 Object::Ptr result;
204 std::string lastRevision;
205 try {
206 if (!isMcpProjectIndexTool(name) && name != "eve_task_context")
207 return textContentResult(mcpStringify(mcpFailureObject("unknown_tool")), true);
208 const auto root = std::filesystem::canonical(sourcePath(requiredString(args, "projectDir")));
209 if (!std::filesystem::is_directory(root) || std::filesystem::is_symlink(root / "game.semantic.json") ||
210 (std::filesystem::exists(root / "game.semantic.json") &&
211 !std::filesystem::is_regular_file(root / "game.semantic.json")))
212 throw std::runtime_error("Project requires a directory and an optional regular game.semantic.json");
213 auto& state = stateFor(root);
214 lastRevision = state.snapshot.revision;
215 refresh(root, state);
216 result = query(state.snapshot, name, args);
217 } catch (const std::exception& error) {
218 result = mcpFailureObject("project_index_failed", error.what());
219 if (!lastRevision.empty()) result->set("lastGoodRevision", lastRevision);
220 }
221 return textContentResult(mcpStringify(result), !result->getValue<bool>("ok"));
222}
