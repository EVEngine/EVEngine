54Result<Crowd::FlowBuildProgress> Crowd::advanceFlowBuild(FlowBuildPlan& plan, int maxWork) try {
55 if (maxWork < 0 || maxWork > 1000000)
56 return Result<FlowBuildProgress>::failure(
57 Diagnostic::error(DiagnosticCode::InvalidArgument, "Invalid navigation work budget", "flowBuild.maxWork"));
58 if (!plan.field_ || (plan.state_ != FlowBuildState::Pending && plan.state_ != FlowBuildState::Ready &&
59 plan.state_ != FlowBuildState::Unreachable))
60 return Result<FlowBuildProgress>::failure(
61 Diagnostic::error(DiagnosticCode::Conflict, "Navigation request is not advanceable", "flowBuild"));
62 if (plan.owner_ != impl_->owner)
63 return Result<FlowBuildProgress>::failure(
64 Diagnostic::error(DiagnosticCode::Conflict, "Foreign navigation request", "flowBuild"));
65 if (plan.revision_ != impl_->field.inputRevision()) {
66 auto cancelled = plan.cancel();
67 if (!cancelled) return Result<FlowBuildProgress>::failure(cancelled.status());
68 return Result<FlowBuildProgress>::success({FlowBuildState::Cancelled, 0, 0});
69 }
70 if (plan.state_ != FlowBuildState::Pending || maxWork == 0)
71 return Result<FlowBuildProgress>::success({plan.state_, 0, 0}, Status::success(StatusCode::NoOp));
72 const auto [pops, cells] = plan.field_->advanceBuildWorking(maxWork);
73 if (plan.field_->isBuilt()) {
74 bool reachable = false;
75 for (size_t index = 0; index + 1 < plan.field_->goals_.size(); index += 2)
76 reachable =
77 reachable || plan.field_->isReachable(plan.field_->goals_[index], plan.field_->goals_[index + 1]);
78 plan.state_ = reachable ? FlowBuildState::Ready : FlowBuildState::Unreachable;
79 }
80 return Result<FlowBuildProgress>::success({plan.state_, pops, cells});
81} catch (const std::exception& error) {
