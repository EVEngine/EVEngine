206std::vector<EditorDiagnostic> SpriteStackDocumentTarget::validate() const {
207 std::vector<EditorDiagnostic> d;
208 const auto& v = value_;
209 if (v.sourceKind != "primitive" && v.sourceKind != "model")
210 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument,
211 RuleId("editor.spritestack.source-kind"), DiagnosticSeverity::Error,
212 "SpriteStack source kind must be primitive or model"));
213 if (v.source.empty())
214 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument,
215 RuleId("editor.spritestack.source"), DiagnosticSeverity::Error, "SpriteStack source must not be empty"));
216 if (v.sourceKind == "primitive" && v.source != "box" && v.source != "sphere" && v.source != "cylinder" &&
217 v.source != "cone")
218 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument,
219 RuleId("editor.spritestack.primitive"), DiagnosticSeverity::Error, "Unknown SpriteStack primitive"));
220 const bool axisInvalid = v.bake.axis != "x" && v.bake.axis != "y" && v.bake.axis != "z";
221 if (v.bake.layerCount < 1 || v.bake.layerCount > 512 || v.bake.imageW < 1 || v.bake.imageW > 4096 ||
222 v.bake.imageH < 1 || v.bake.imageH > 4096 || axisInvalid || !std::isfinite(v.bake.thickness) ||
223 v.bake.thickness < 0 || !std::isfinite(v.bake.padding) || v.bake.padding < 0 || v.bake.padding >= .5f)
224 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument,
225 RuleId("editor.spritestack.bake-range"), DiagnosticSeverity::Error,
226 "SpriteStack bake settings are invalid"));
227 const std::uint64_t bytes = static_cast<std::uint64_t>(std::max(0, v.bake.layerCount)) *
228 std::max(0, v.bake.imageW) * std::max(0, v.bake.imageH) * 4ull;
229 if (bytes > 512ull * 1024 * 1024)
230 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::PreconditionViolation,
231 RuleId("editor.spritestack.memory"), DiagnosticSeverity::Error,
232 "SpriteStack output exceeds the 512 MiB decoded budget"));
233 else if (bytes > 128ull * 1024 * 1024)
234 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::PreconditionViolation,
235 RuleId("editor.spritestack.memory-warning"), DiagnosticSeverity::Warning,
236 "SpriteStack output exceeds 128 MiB decoded"));
237 if (!std::isfinite(v.displayWidth) || v.displayWidth <= 0 || !std::isfinite(v.displayHeight) ||
238 v.displayHeight <= 0 || !std::isfinite(v.layerOffset) || v.layerOffset <= 0 ||
239 !std::isfinite(v.shadowOpacity) || v.shadowOpacity < 0 || v.shadowOpacity > 1 ||
240 !std::isfinite(v.outlineWidth) || v.outlineWidth < 0)
241 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument,
242 RuleId("editor.spritestack.display"), DiagnosticSeverity::Error,
243 "SpriteStack presentation values are invalid"));
244 return d;
245}
246EditorResult<void> SpriteStackDocumentTarget::applyDomainOperation(const DomainOperation& op) {
