49double distanceSquared(const TargetLocation& a, const TargetLocation& b) noexcept {
50 return std::visit(
51 [](const auto& left, const auto& right) -> double {
52 if constexpr (std::is_same_v<std::decay_t<decltype(left)>, WorldPoint> &&
53 std::is_same_v<std::decay_t<decltype(right)>, WorldPoint>) {
54 const double dx = static_cast<double>(left.x()) - right.x();
55 const double dy = static_cast<double>(left.y()) - right.y();
56 const double dz = static_cast<double>(left.z()) - right.z();
57 return dx * dx + dy * dy + dz * dz;
58 } else if constexpr (std::is_same_v<std::decay_t<decltype(left)>, GridPoint> &&
59 std::is_same_v<std::decay_t<decltype(right)>, GridPoint>) {
60 const double dx = static_cast<double>(left.x()) - right.x();
61 const double dy = static_cast<double>(left.y()) - right.y();
62 const double dz = static_cast<double>(left.z()) - right.z();
63 return dx * dx + dy * dy + dz * dz;
64 } else {
65 return std::numeric_limits<double>::infinity();
66 }
67 },
68 a, b);
69}

98bool matches(const TargetingQuery& query, const TargetCandidate& candidate) {
99 if (!candidate.subject.isValid() || !sameSpace(candidate.location, query.spec.space)) return false;
100 if (query.spec.domain != TargetDomain::Any && candidate.domain != query.spec.domain) return false;
101 if (!containsAllTags(candidate.tags, query.spec.requiredTags, query.spec.excludedTags)) return false;
102 if (!inZone(candidate, query.spec.zone) || !contains(query.spec, candidate.location)) return false;
103
104 const double distance = std::sqrt(distanceSquared(query.originLocation, candidate.location));
105 return distance >= static_cast<double>(query.spec.minRange) && distance <= static_cast<double>(query.spec.maxRange);
106}

216Result<void> TargetingSpec::validate() const {
217 if ((isWorldSpace(space) && gridArea) || (isGridSpace(space) && worldArea))
218 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec area uses a different coordinate space");
219 if (worldArea && !worldArea->isValid())
220 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec world area must be valid");
221 if (gridArea && !gridArea->isValid())
222 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec grid area must be valid");
223 if (worldArea && worldArea->space() != space)
224 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec world area does not match its space");
225 if (gridArea && gridArea->space() != space)
226 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec grid area does not match its space");
227 if (minCount > maxCount)
228 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec minCount exceeds maxCount");
229 if (!std::isfinite(minRange) ||
230 (!(std::isfinite(maxRange) || maxRange == std::numeric_limits<float>::infinity())) || minRange < 0.f ||
231 maxRange < minRange)
232 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec range must be finite and ordered");
233 if (zone && !zone->isValid())
234 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec zone must be valid");
235 for (const auto& tag : requiredTags)
236 if (tag.empty())
237 return failure<void>(DiagnosticCode::InvalidArgument, "requiredTags cannot contain empty keys");
238 for (const auto& tag : excludedTags)
239 if (tag.empty())
240 return failure<void>(DiagnosticCode::InvalidArgument, "excludedTags cannot contain empty keys");
241 return Result<void>::success();
242}

335Result<TargetSet> TargetingResolver::resolve(const TargetingQuery& query) const {
336 auto valid = query.validate();
337 if (!valid) return failure<TargetSet>(valid.status());
338
339 auto* provider = cap::query<ISensingCandidateProvider>();
340 if (!provider) return unsupported<TargetSet>("Targeting requires an ISensingCandidateProvider");
341
342 auto candidateResult = provider->query(query);
343 if (!candidateResult) return failure<TargetSet>(candidateResult.status());
344 auto candidates = std::move(candidateResult).takeValue();
345
346 ILineOfSightQuery* los = nullptr;
347 if (query.spec.lineOfSight == LineOfSightMode::Required) {
348 los = cap::query<ILineOfSightQuery>();
349 if (!los) return unsupported<TargetSet>("Targeting line-of-sight was requested but no provider is registered");
350 }
351
352 TargetSet result;
353 for (const auto& candidate : candidates) {
354 if (!candidate.subject.isValid() ||
355 !std::visit([](const auto& value) { return value.isValid(); }, candidate.location))
356 return failure<TargetSet>(DiagnosticCode::InvariantViolation,
357 "candidate provider returned an invalid candidate");
358 if (!sameSpace(candidate.location, query.spec.space))
359 return failure<TargetSet>(DiagnosticCode::InvariantViolation, "candidate provider mixed coordinate spaces");
360 if (!matches(query, candidate)) continue;
361 if (los) {
362 auto visible = los->query(query.originLocation, candidate.location);
363 if (!visible) return failure<TargetSet>(visible.status());
364 if (!std::move(visible).takeValue().visible) continue;
365 }
366 auto added = result.addSubject(candidate.subject);
367 std::move(added).expect("TargetingResolver could not add a validated candidate");
368 }
369
370 if (result.subjects().size() < query.spec.minCount || result.subjects().size() > query.spec.maxCount)
371 return failure<TargetSet>(DiagnosticCode::PreconditionViolation,
372 "target candidate count violates TargetingSpec");
373 return Result<TargetSet>::success(std::move(result));
374}
