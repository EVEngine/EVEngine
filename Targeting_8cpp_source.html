50double distanceSquared(const TargetLocation& a, const TargetLocation& b) noexcept {
51 return std::visit(
52 [](const auto& left, const auto& right) -> double {
53 if constexpr (std::is_same_v<std::decay_t<decltype(left)>, WorldPoint> &&
54 std::is_same_v<std::decay_t<decltype(right)>, WorldPoint>) {
55 const double dx = static_cast<double>(left.x()) - right.x();
56 const double dy = static_cast<double>(left.y()) - right.y();
57 const double dz = static_cast<double>(left.z()) - right.z();
58 return dx * dx + dy * dy + dz * dz;
59 } else if constexpr (std::is_same_v<std::decay_t<decltype(left)>, GridPoint> &&
60 std::is_same_v<std::decay_t<decltype(right)>, GridPoint>) {
61 const double dx = static_cast<double>(left.x()) - right.x();
62 const double dy = static_cast<double>(left.y()) - right.y();
63 const double dz = static_cast<double>(left.z()) - right.z();
64 return dx * dx + dy * dy + dz * dz;
65 } else {
66 return std::numeric_limits<double>::infinity();
67 }
68 },
69 a, b);
70}

99bool matches(const TargetingQuery& query, const TargetCandidate& candidate) {
100 if (!candidate.subject.isValid() || !sameSpace(candidate.location, query.spec.space)) return false;
101 if (query.spec.domain != TargetDomain::Any && candidate.domain != query.spec.domain) return false;
102 if (!containsAllTags(candidate.tags, query.spec.requiredTags, query.spec.excludedTags)) return false;
103 if (!inZone(candidate, query.spec.zone) || !contains(query.spec, candidate.location)) return false;
104
105 const double distance = std::sqrt(distanceSquared(query.originLocation, candidate.location));
106 return distance >= static_cast<double>(query.spec.minRange) && distance <= static_cast<double>(query.spec.maxRange);
107}

199bool WorldArea::contains(WorldPoint point) const noexcept {
200 if (!valid_ || !point.isValid() || point.space() != space_) return false;
201 if (shape_ == Shape::Circle2D || shape_ == Shape::Sphere3D) {
202 const double dx = static_cast<double>(point.x()) - first_.x();
203 const double dy = static_cast<double>(point.y()) - first_.y();
204 const double dz = static_cast<double>(point.z()) - first_.z();
205 const double distance = dx * dx + dy * dy + dz * dz;
206 return distance <= static_cast<double>(radius_) * radius_;
207 }
208 if (shape_ == Shape::Cone2D) {
209 const float dx = point.x() - first_.x();
210 const float dy = point.y() - first_.y();
211 const float distSq = dx * dx + dy * dy;
212 if (distSq > radius_ * radius_) return false;
213 if (distSq == 0.f) return true;
214 const float facingLen = std::hypot(second_.x(), second_.y());
215 if (!(facingLen > 0.f)) return false;
216 const float invDist = 1.f / std::sqrt(distSq);
217 const float nx = dx * invDist;
218 const float ny = dy * invDist;
219 const float fx = second_.x() / facingLen;
220 const float fy = second_.y() / facingLen;
221 const float dot = std::clamp(nx * fx + ny * fy, -1.f, 1.f);
222 return std::acos(dot) <= halfAngle_;
223 }
224 return point.x() >= first_.x() && point.x() <= second_.x() && point.y() >= first_.y() && point.y() <= second_.y() &&
225 (shape_ == Shape::Box2D || (point.z() >= first_.z() && point.z() <= second_.z()));
226}

247Result<void> TargetingSpec::validate() const {
248 if ((isWorldSpace(space) && gridArea) || (isGridSpace(space) && worldArea))
249 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec area uses a different coordinate space");
250 if (worldArea && !worldArea->isValid())
251 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec world area must be valid");
252 if (gridArea && !gridArea->isValid())
253 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec grid area must be valid");
254 if (worldArea && worldArea->space() != space)
255 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec world area does not match its space");
256 if (gridArea && gridArea->space() != space)
257 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec grid area does not match its space");
258 if (minCount > maxCount)
259 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec minCount exceeds maxCount");
260 if (!std::isfinite(minRange) ||
261 (!(std::isfinite(maxRange) || maxRange == std::numeric_limits<float>::infinity())) || minRange < 0.f ||
262 maxRange < minRange)
263 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec range must be finite and ordered");
264 if (zone && !zone->isValid())
265 return failure<void>(DiagnosticCode::InvalidArgument, "TargetingSpec zone must be valid");
266 for (const auto& tag : requiredTags)
267 if (tag.empty())
268 return failure<void>(DiagnosticCode::InvalidArgument, "requiredTags cannot contain empty keys");
269 for (const auto& tag : excludedTags)
270 if (tag.empty())
271 return failure<void>(DiagnosticCode::InvalidArgument, "excludedTags cannot contain empty keys");
272 return Result<void>::success();
273}

366Result<std::vector<TargetCandidate>> SensingWorldCandidateProvider::query(const TargetingQuery& query) const {
367 auto valid = query.validate();
368 if (!valid) return failure<std::vector<TargetCandidate>>(valid.status());
369 if (world_ == nullptr)
370 return unsupported<std::vector<TargetCandidate>>("SensingWorldCandidateProvider has no bound SensingWorld");
371 if (query.spec.space != CoordinateSpace::World2D)
372 return unsupported<std::vector<TargetCandidate>>("SensingWorldCandidateProvider supports World2D only");
373 if (query.spec.gridArea)
374 return unsupported<std::vector<TargetCandidate>>("SensingWorldCandidateProvider does not support grid areas");
375 if (query.spec.domain != TargetDomain::Any && !relation_)
376 return unsupported<std::vector<TargetCandidate>>(
377 "TargetDomain filters require an injected FactionRelationFn on SensingWorldCandidateProvider");
378
379 const auto* originPoint = std::get_if<WorldPoint>(&query.originLocation);
380 if (originPoint == nullptr || !originPoint->isValid() || originPoint->space() != CoordinateSpace::World2D)
381 return failure<std::vector<TargetCandidate>>(DiagnosticCode::InvalidArgument,
382 "SensingWorldCandidateProvider origin must be a World2D point");
383
384 QuerySpec spec;
385 spec.minRange = query.spec.minRange;
386 spec.maxRange = query.spec.maxRange;
387 spec.requiredTags = query.spec.requiredTags;
388 spec.excludedTags = query.spec.excludedTags;
389 spec.minCount = 0;
390 spec.maxCount = std::numeric_limits<std::uint32_t>::max();
391 spec.countPolicy = CountPolicy::TruncateToMax;
392 spec.sortKey = SortKey::DistanceAscending;
393
394 auto queried = world_->query(QueryOrigin{originPoint->x(), originPoint->y(), query.origin.format()}, spec);
395 if (!queried) return failure<std::vector<TargetCandidate>>(queried.status());
396
397 std::string originFaction;
398 if (const auto found = world_->subjects().find(query.origin.format()); found != world_->subjects().end())
399 originFaction = found->second.faction;
400
401 std::vector<TargetCandidate> result;
402 result.reserve(queried.value().size());
403 for (const auto& ranked : queried.value().ranked()) {
404 auto persistent = PersistentId::parse(ranked.id);
405 if (!persistent)
406 return failure<std::vector<TargetCandidate>>(
407 DiagnosticCode::InvalidArgument,
408 "SensingWorld subject id must be a PersistentId UUID for Targeting adapters", "subject.id");
409 auto location = WorldPoint::world2D(ranked.x, ranked.y);
410 if (!location) return failure<std::vector<TargetCandidate>>(location.status());
411 if (query.spec.worldArea && !query.spec.worldArea->contains(location.value())) continue;
412
413 TargetDomain domain = TargetDomain::Neutral;
414 if (relation_) {
415 const auto subjectIt = world_->subjects().find(ranked.id);
416 const std::string_view candidateFaction =
417 subjectIt == world_->subjects().end() ? std::string_view{} : std::string_view{subjectIt->second.faction};
418 auto related = relation_(originFaction, candidateFaction);
419 if (!related) return failure<std::vector<TargetCandidate>>(related.status());
420 domain = std::move(related).takeValue();
421 }
422 if (query.spec.domain != TargetDomain::Any && domain != query.spec.domain) continue;
423
424 TargetCandidate candidate;
425 candidate.subject = SubjectRef::fromPersistentId(*persistent);
426 candidate.location = std::move(location).takeValue();
427 candidate.domain = domain;
428 const auto& subjectFacts = world_->subjects().at(ranked.id);
429 candidate.tags.assign(subjectFacts.tags.begin(), subjectFacts.tags.end());
430 candidate.zones.clear();
431 candidate.zones.reserve(subjectFacts.zones.size());
432 for (const auto& zoneText : subjectFacts.zones) {
433 auto logical = LogicalId::parse(zoneText);
434 if (!logical)
435 return failure<std::vector<TargetCandidate>>(
436 DiagnosticCode::InvalidArgument,
437 "SensingWorld subject zone must be a valid LogicalId", "subject.zones");
438 auto zone = ZoneRef::fromLogicalId(*logical);
439 if (!zone)
440 return failure<std::vector<TargetCandidate>>(
441 DiagnosticCode::InvalidArgument,
442 "SensingWorld subject zone must form a valid ZoneRef", "subject.zones");
443 candidate.zones.push_back(*zone);
444 }
445 if (query.spec.zone && !inZone(candidate, query.spec.zone)) continue;
446 result.push_back(std::move(candidate));
447 }
448 return Result<std::vector<TargetCandidate>>::success(std::move(result));
449}

451Result<TargetSet> TargetingResolver::resolve(const TargetingQuery& query) const {
452 auto valid = query.validate();
453 if (!valid) return failure<TargetSet>(valid.status());
454
455 auto* provider = cap::query<ISensingCandidateProvider>();
456 if (!provider) return unsupported<TargetSet>("Targeting requires an ISensingCandidateProvider");
457
458 auto candidateResult = provider->query(query);
459 if (!candidateResult) return failure<TargetSet>(candidateResult.status());
460 auto candidates = std::move(candidateResult).takeValue();
461
462 ILineOfSightQuery* los = nullptr;
463 if (query.spec.lineOfSight == LineOfSightMode::Required) {
464 los = cap::query<ILineOfSightQuery>();
465 if (!los) return unsupported<TargetSet>("Targeting line-of-sight was requested but no provider is registered");
466 }
467
468 struct Ranked {
469 TargetCandidate candidate;
470 double distance = 0.0;
471 };
472 std::vector<Ranked> accepted;
473 accepted.reserve(candidates.size());
474 for (const auto& candidate : candidates) {
475 if (!candidate.subject.isValid() ||
476 !std::visit([](const auto& value) { return value.isValid(); }, candidate.location))
477 return failure<TargetSet>(DiagnosticCode::InvariantViolation,
478 "candidate provider returned an invalid candidate");
479 if (!sameSpace(candidate.location, query.spec.space))
480 return failure<TargetSet>(DiagnosticCode::InvariantViolation, "candidate provider mixed coordinate spaces");
481 if (!matches(query, candidate)) continue;
482 if (los) {
483 auto visible = los->query(query.originLocation, candidate.location);
484 if (!visible) return failure<TargetSet>(visible.status());
485 if (!std::move(visible).takeValue().visible) continue;
486 }
487 accepted.push_back(Ranked{candidate, std::sqrt(distanceSquared(query.originLocation, candidate.location))});
488 }
489
490 if (query.spec.countPolicy == CountPolicy::TruncateToMax) {
491 std::sort(accepted.begin(), accepted.end(), [](const Ranked& a, const Ranked& b) {
492 if (a.distance != b.distance) return a.distance < b.distance;
493 return a.candidate.subject.format() < b.candidate.subject.format();
494 });
495 if (accepted.size() > static_cast<std::size_t>(query.spec.maxCount))
496 accepted.resize(static_cast<std::size_t>(query.spec.maxCount));
497 if (accepted.size() < static_cast<std::size_t>(query.spec.minCount))
498 return failure<TargetSet>(DiagnosticCode::PreconditionViolation,
499 "target candidate count is below TargetingSpec.minCount after truncation");
500 } else if (accepted.size() < static_cast<std::size_t>(query.spec.minCount) ||
501 accepted.size() > static_cast<std::size_t>(query.spec.maxCount)) {
502 return failure<TargetSet>(DiagnosticCode::PreconditionViolation,
503 "target candidate count violates TargetingSpec");
504 }
505
506 TargetSet result;
507 for (const auto& entry : accepted) {
508 auto added = result.addSubject(entry.candidate.subject);
509 std::move(added).expect("TargetingResolver could not add a validated candidate");
510 }
511 return Result<TargetSet>::success(std::move(result));
512}
