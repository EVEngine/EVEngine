41Result<void> GuardWindowState::enter(const action::ActionStateWindowBinding& binding,
42 const action::ActionTimelineEvent& event,
43 const action::ActionNotifyContext& context) {
44 if (!supports(binding.kind))
45 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Unsupported, "combat guard does not own this state-window kind", "kind"));
46 if (!resolver_) return Result<void>::failure(Diagnostic::error(DiagnosticCode::NotFound, "guard subject resolver is unavailable", "resolver"));
47 auto mode = parseMode(binding.resource);
48 if (!mode) return Result<void>::failure(mode.status());
49 std::optional<ecs::EntityHandle> handle;
50 if (binding.targetIndex) {
51 if (*binding.targetIndex >= context.targets.size())
52 return Result<void>::failure(Diagnostic::error(DiagnosticCode::NotFound, "guard target index is unavailable", "targetIndex"));
53 handle = context.targets[*binding.targetIndex];
54 } else {
55 handle = context.source;
56 }
57 if (!handle) return Result<void>::failure(Diagnostic::error(DiagnosticCode::NotFound, "guard window has no source or target", "subject"));
58 auto subject = resolver_(*handle);
59 if (!subject) return Result<void>::failure(subject.status());
60 if (!subject.value().isValid())
61 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "guard-window subject is nil", "subject"));
62
63 ActiveKey key{context.executionId, event.itemId.format()};
64 const auto found = active_.find(key);
65 if (found != active_.end()) {
66 const bool same = found->second.subject == subject.value() && found->second.mode == mode.value();
67 if (same) return Result<void>::success(Status::success(StatusCode::NoOp));
68 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Conflict, "guard-window key is already active with different data", "itemId"));
69 }
70 active_.emplace(std::move(key), ActiveGuard{binding, subject.value(), mode.value()});
71 return Result<void>::success(Status::success(StatusCode::Applied));
72}

74Result<void> GuardWindowState::exit(const action::ActionStateWindowBinding& binding,
75 const action::ActionTimelineEvent& event,
76 const action::ActionNotifyContext& context) {
77 if (!supports(binding.kind))
78 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Unsupported, "combat guard does not own this state-window kind", "kind"));
79 const ActiveKey key{context.executionId, event.itemId.format()};
80 const auto found = active_.find(key);
81 if (found == active_.end()) return Result<void>::success(Status::success(StatusCode::NoOp));
82 if (found->second.binding.kind != binding.kind)
83 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Conflict, "guard-window exit kind does not match active key", "kind"));
84 active_.erase(found);
85 return Result<void>::success(Status::success(StatusCode::Applied));
86}

88Result<GuardEvaluation> GuardWindowState::evaluate(SubjectRef defender, SubjectRef attacker) const {
89 if (!defender.isValid() || !attacker.isValid())
90 return Result<GuardEvaluation>::failure(
91 Diagnostic::error(DiagnosticCode::InvalidArgument, "guard subjects must be valid", "subject"));
92 GuardEvaluation evaluation;
93 evaluation.defender = defender;
94 evaluation.attacker = attacker;
95 bool hasBlock = false;
96 bool hasParry = false;
97 for (const auto& [key, window] : active_) {
98 (void)key;
99 if (window.subject != defender) continue;
100 if (window.mode == GuardMode::Parry) hasParry = true;
101 if (window.mode == GuardMode::Block) hasBlock = true;
102 }
103 if (hasParry) {
104 evaluation.result = GuardResult::Parried;
105 evaluation.mode = GuardMode::Parry;
106 } else if (hasBlock) {
107 evaluation.result = GuardResult::Blocked;
108 evaluation.mode = GuardMode::Block;
109 }
110 return Result<GuardEvaluation>::success(evaluation);
111}
