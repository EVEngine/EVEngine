15void put32(std::vector<std::uint8_t>& bytes, std::uint32_t value) {
16 for (unsigned shift = 0; shift != 32; shift += 8) bytes.push_back(static_cast<std::uint8_t>(value >> shift));
17}

19void putFloat(std::vector<std::uint8_t>& bytes, float value) { put32(bytes, std::bit_cast<std::uint32_t>(value)); }

21void putString(std::vector<std::uint8_t>& bytes, std::string_view value) {
22 put32(bytes, static_cast<std::uint32_t>(value.size()));
23 bytes.insert(bytes.end(), value.begin(), value.end());
24}

26Result<void> addAsset(PreparedAssetImport& output, PersistentId id, std::string type,
27 Value::Object definition, std::vector<std::uint8_t> blob,
28 std::string blobName, std::vector<std::string> tags) {
29 auto reference = detail::assetRef(id);
30 if (!reference) return Result<void>::failure(reference.status());
31 const std::string root = "assets/" + id.format() + "/";
32 const std::string definitionPath = root + "asset.json";
33 if (!blob.empty()) definition["blob"] = Value(root + blobName);
34 auto encoded = Value(std::move(definition)).toJson();
35 if (!encoded) return Result<void>::failure(encoded.status());
36 std::string text = std::move(encoded).takeValue();
37 output.manifest.assets.push_back({std::move(reference).takeValue(), std::move(type), SchemaVersion(1),
38 definitionPath,
39 detail::sha256(std::span<const std::uint8_t>(
40 reinterpret_cast<const std::uint8_t*>(text.data()), text.size())),
41 std::move(tags)});
42 output.entries.push_back({definitionPath, {text.begin(), text.end()}});
43 if (!blob.empty()) output.entries.push_back({root + blobName, std::move(blob)});
44 return Result<void>::success();
45}

49Result<PreparedAssetImport> prepareCanonicalTerrainImport(const ImportPackageIdentity& package,
50 const CanonicalTerrainInput& terrain,
51 std::string_view importer,
52 const AssetImportLimits& limits) {
53 if (terrain.width < 2 || terrain.height < 2 || !std::isfinite(terrain.spacingX) ||
54 !std::isfinite(terrain.spacingZ) || terrain.spacingX <= 0 || terrain.spacingZ <= 0 ||
55 terrain.heightsMeters.size() != std::uint64_t(terrain.width) * terrain.height)
56 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
57 "terrain dimensions, spacing or height count is invalid");
58 const std::uint64_t heightBytes = std::uint64_t(terrain.heightsMeters.size()) * sizeof(float) + 24;
59 if (heightBytes > limits.maximumDecodedBytes || terrain.layers.size() > limits.maximumAssets ||
60 terrain.scatterRules.size() > limits.maximumAssets || terrain.instances.size() > limits.maximumAssets)
61 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
62 "terrain import budget is exceeded");
63 for (const float height : terrain.heightsMeters)
64 if (!std::isfinite(height))
65 return detail::failure<PreparedAssetImport>(DiagnosticCode::ParseError,
66 "terrain height contains a non-finite value");
67 auto manifest = detail::baseManifest(package, importer);
68 if (!manifest) return Result<PreparedAssetImport>::failure(manifest.status());
69 PreparedAssetImport output;
70 output.manifest = std::move(manifest).takeValue();
71 output.manifest.provenance["sourceTransform"] = Value(terrain.sourceTransform);
72
73 const PersistentId terrainId = package.packageId.child("terrain:root");
74 const PersistentId materialId = package.packageId.child("terrain:material");
75 const PersistentId pcgId = package.packageId.child("terrain:pcg");
76 const PersistentId instancesId = package.packageId.child("terrain:instances");
77
78 std::vector<std::uint8_t> heights;
79 heights.insert(heights.end(), {'E', 'V', 'T', 'R', 'N', 0, 1, 0});
80 put32(heights, terrain.width); put32(heights, terrain.height);
81 putFloat(heights, terrain.spacingX); putFloat(heights, terrain.spacingZ);
82 for (float height : terrain.heightsMeters) putFloat(heights, height);
83 Value::Object terrainDefinition;
84 terrainDefinition["schema"] = Value("eve.terrain");
85 terrainDefinition["schemaVersion"] = Value(std::int64_t(1));
86 terrainDefinition["width"] = Value(static_cast<std::int64_t>(terrain.width));
87 terrainDefinition["height"] = Value(static_cast<std::int64_t>(terrain.height));
88 terrainDefinition["spacingX"] = Value(double(terrain.spacingX));
89 terrainDefinition["spacingZ"] = Value(double(terrain.spacingZ));
90 terrainDefinition["coordinateSystem"] = Value("right-handed-x-right-y-up-minus-z-forward");
91 terrainDefinition["heightUnit"] = Value("meter");
92 if (!terrain.layers.empty()) terrainDefinition["material"] = Value("asset://" + materialId.format());
93 if (!terrain.scatterRules.empty()) terrainDefinition["pcg"] = Value("asset://" + pcgId.format());
94 if (!terrain.instances.empty()) terrainDefinition["instances"] = Value("asset://" + instancesId.format());
95 auto addedTerrain = addAsset(output, terrainId, "eve.terrain", std::move(terrainDefinition),
96 std::move(heights), "heightfield.bin", {"terrain"});
97 if (!addedTerrain) return Result<PreparedAssetImport>::failure(addedTerrain.status());
98
99 if (!terrain.layers.empty()) {
100 Value::Array layers;
101 for (const auto& layer : terrain.layers) {
102 if (layer.name.empty() || !std::isfinite(layer.tileSizeMeters) || layer.tileSizeMeters <= 0 ||
103 (layer.normalConvention != "opengl" && layer.normalConvention != "directx"))
104 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
105 "terrain layer metadata is invalid", layer.name);
106 Value::Object value;
107 value["name"] = Value(layer.name); value["diffuseSource"] = Value(layer.diffuseSource);
108 value["normalSource"] = Value(layer.normalSource); value["weightSource"] = Value(layer.weightSource);
109 value["normalConvention"] = Value(layer.normalConvention);
110 value["tileSizeMeters"] = Value(double(layer.tileSizeMeters));
111 layers.emplace_back(std::move(value));
112 }
113 Value::Object definition;
114 definition["schema"] = Value("eve.terrain-material");
115 definition["schemaVersion"] = Value(std::int64_t(1));
116 definition["layers"] = Value(std::move(layers));
117 auto added = addAsset(output, materialId, "eve.terrain-material", std::move(definition), {}, {},
118 {"terrain", "material"});
119 if (!added) return Result<PreparedAssetImport>::failure(added.status());
120 }
121
122 if (!terrain.scatterRules.empty()) {
123 Value::Array rules;
124 std::set<std::string> ids;
125 for (const auto& rule : terrain.scatterRules) {
126 if (rule.id.empty() || rule.prototype.empty() || !ids.emplace(rule.id).second ||
127 !std::isfinite(rule.densityPerSquareMeter) || rule.densityPerSquareMeter < 0 ||
128 !std::isfinite(rule.minimumSlopeRadians) || !std::isfinite(rule.maximumSlopeRadians) ||
129 rule.minimumSlopeRadians > rule.maximumSlopeRadians)
130 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
131 "terrain scatter rule is invalid", rule.id);
132 Value::Object value;
133 value["id"] = Value(rule.id); value["prototype"] = Value(rule.prototype);
134 value["layer"] = Value(rule.layer); value["densityPerSquareMeter"] = Value(double(rule.densityPerSquareMeter));
135 value["minimumSlopeRadians"] = Value(double(rule.minimumSlopeRadians));
136 value["maximumSlopeRadians"] = Value(double(rule.maximumSlopeRadians));
137 value["seed"] = Value(static_cast<std::int64_t>(rule.seed & 0x7fffffffffffffffULL));
138 rules.emplace_back(std::move(value));
139 }
140 Value::Object definition;
141 definition["schema"] = Value("eve.pcg-graph"); definition["schemaVersion"] = Value(std::int64_t(1));
142 definition["algorithm"] = Value("terrain-layer-scatter-v1"); definition["rules"] = Value(std::move(rules));
143 auto added = addAsset(output, pcgId, "eve.pcg-graph", std::move(definition), {}, {}, {"terrain", "pcg"});
144 if (!added) return Result<PreparedAssetImport>::failure(added.status());
145 }
146
147 if (!terrain.instances.empty()) {
148 std::vector<std::uint8_t> instances;
149 instances.insert(instances.end(), {'E', 'V', 'I', 'N', 'S', 'T', 0, 1});
150 put32(instances, static_cast<std::uint32_t>(terrain.instances.size())); put32(instances, 0);
151 for (const auto& instance : terrain.instances) {
152 if (instance.prototype.empty() || instance.prototype.size() > limits.maximumStringBytes ||
153 !isValidUtf8(instance.prototype, Utf8NullPolicy::Reject))
154 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
155 "terrain instance prototype is invalid");
156 const float rotationLength = std::sqrt(
157 instance.rotation[0] * instance.rotation[0] +
158 instance.rotation[1] * instance.rotation[1] +
159 instance.rotation[2] * instance.rotation[2] +
160 instance.rotation[3] * instance.rotation[3]);
161 if (!std::isfinite(rotationLength) || rotationLength < 0.999f ||
162 rotationLength > 1.001f || instance.scale[0] == 0.f ||
163 instance.scale[1] == 0.f || instance.scale[2] == 0.f)
164 return detail::failure<PreparedAssetImport>(
165 DiagnosticCode::InvalidArgument,
166 "terrain instance rotation or scale is invalid", instance.prototype);
167 putString(instances, instance.prototype);
168 for (float value : instance.position) { if (!std::isfinite(value)) return detail::failure<PreparedAssetImport>(DiagnosticCode::ParseError, "instance position is non-finite"); putFloat(instances, value); }
169 for (float value : instance.rotation) { if (!std::isfinite(value)) return detail::failure<PreparedAssetImport>(DiagnosticCode::ParseError, "instance rotation is non-finite"); putFloat(instances, value); }
170 for (float value : instance.scale) { if (!std::isfinite(value)) return detail::failure<PreparedAssetImport>(DiagnosticCode::ParseError, "instance scale is non-finite"); putFloat(instances, value); }
171 }
172 if (instances.size() > limits.maximumDecodedBytes)
173 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
174 "terrain instance blob exceeds budget");
175 Value::Object definition;
176 definition["schema"] = Value("eve.instance-set"); definition["schemaVersion"] = Value(std::int64_t(1));
177 definition["count"] = Value(static_cast<std::int64_t>(terrain.instances.size()));
178 definition["partition"] = Value("single-cell");
179 auto added = addAsset(output, instancesId, "eve.instance-set", std::move(definition),
180 std::move(instances), "instances.bin", {"terrain", "instances"});
181 if (!added) return Result<PreparedAssetImport>::failure(added.status());
182 }
183
184 auto terrainRef = detail::assetRef(terrainId);
185 if (!terrainRef) return Result<PreparedAssetImport>::failure(terrainRef.status());
186 output.manifest.entrypoints.emplace("default", std::move(terrainRef).takeValue());
187 auto rootRef = detail::assetRef(terrainId);
188 if (!rootRef) return Result<PreparedAssetImport>::failure(rootRef.status());
189 if (!terrain.layers.empty()) {
190 auto target = detail::assetRef(materialId);
191 if (!target) return Result<PreparedAssetImport>::failure(target.status());
192 output.manifest.dependencies.push_back({rootRef.value(), std::move(target).takeValue(),
193 asset::EvaDependencyKind::RuntimeRequired, "material", {},
194 "eve.terrain-material/1"});
195 }
196 if (!terrain.scatterRules.empty()) {
197 auto target = detail::assetRef(pcgId);
198 if (!target) return Result<PreparedAssetImport>::failure(target.status());
199 output.manifest.dependencies.push_back({rootRef.value(), std::move(target).takeValue(),
200 asset::EvaDependencyKind::RuntimeOptional, "pcg", {},
201 "eve.pcg-graph/1",
202 {{"behavior", Value("omit-feature")},
203 {"observableCode", Value("terrain.pcg-unavailable")}}});
204 }
205 if (!terrain.instances.empty()) {
206 auto target = detail::assetRef(instancesId);
207 if (!target) return Result<PreparedAssetImport>::failure(target.status());
208 output.manifest.dependencies.push_back({std::move(rootRef).takeValue(), std::move(target).takeValue(),
209 asset::EvaDependencyKind::RuntimeOptional, "instances", {},
210 "eve.instance-set/1",
211 {{"behavior", Value("omit-feature")},
212 {"observableCode", Value("terrain.instances-unavailable")}}});
213 }
214 return Result<PreparedAssetImport>::success(std::move(output));
215}
