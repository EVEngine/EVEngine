23UV projectUv(Vec3 p, Vec3 normal) {
24 const float ax = std::abs(normal.x);
25 const float ay = std::abs(normal.y);
26 const float az = std::abs(normal.z);
27 if (ay >= ax && ay >= az) return {p.x, p.z};
28 if (ax >= az) return {p.z, p.y};
29 return {p.x, p.y};
30}

46Vec3 normalOf(Vec3 a, Vec3 b, Vec3 c) {
47 const Vec3 u{b.x - a.x, b.y - a.y, b.z - a.z};
48 const Vec3 v{c.x - a.x, c.y - a.y, c.z - a.z};
49 Vec3 n{u.y * v.z - u.z * v.y, u.z * v.x - u.x * v.z, u.x * v.y - u.y * v.x};
50 const float length = std::sqrt(n.x * n.x + n.y * n.y + n.z * n.z);
51 if (length > 1.0e-6f) {
52 n.x /= length;
53 n.y /= length;
54 n.z /= length;
55 }
56 return n;
57}

120void addCylinder(MeshBuild& mesh, Vec3 center, float radiusBottom, float radiusTop, float height, int sides,
121 const char* group) {
122 mesh.setActiveGroup(group);
123 sides = std::clamp(sides, 3, 64);
124 const float y0 = center.y - height * 0.5f;
125 const float y1 = center.y + height * 0.5f;
126 const float radius = (radiusBottom + radiusTop) * 0.5f;
127 for (int i = 0; i < sides; ++i) {
128 const float a0 = 2.0f * kPi * float(i) / float(sides);
129 const float a1 = 2.0f * kPi * float(i + 1) / float(sides);
130 const Vec3 p0{center.x + std::cos(a0) * radiusBottom, y0, center.z + std::sin(a0) * radiusBottom};
131 const Vec3 p1{center.x + std::cos(a1) * radiusBottom, y0, center.z + std::sin(a1) * radiusBottom};
132 const Vec3 p2{center.x + std::cos(a1) * radiusTop, y1, center.z + std::sin(a1) * radiusTop};
133 const Vec3 p3{center.x + std::cos(a0) * radiusTop, y1, center.z + std::sin(a0) * radiusTop};
134 const Vec3 normal = normalOf(p0, p2, p1);
135 const uint32_t side = static_cast<uint32_t>(mesh.getVertexCount());
136 addVertex(mesh, p0, normal, a0 * radius, y0);
137 addVertex(mesh, p1, normal, a1 * radius, y0);
138 addVertex(mesh, p2, normal, a1 * radius, y1);
139 addVertex(mesh, p3, normal, a0 * radius, y1);
140 mesh.addTriangle(side, side + 2, side + 1);
141 mesh.addTriangle(side, side + 3, side + 2);
142 const uint32_t bottom = static_cast<uint32_t>(mesh.getVertexCount());
143 addVertex(mesh, {center.x, y0, center.z}, {0, -1, 0}, center.x, center.z);
144 addVertex(mesh, p1, {0, -1, 0}, p1.x, p1.z);
145 addVertex(mesh, p0, {0, -1, 0}, p0.x, p0.z);
146 mesh.addTriangle(bottom, bottom + 2, bottom + 1);
147 const uint32_t top = static_cast<uint32_t>(mesh.getVertexCount());
148 addVertex(mesh, {center.x, y1, center.z}, {0, 1, 0}, center.x, center.z);
149 addVertex(mesh, p3, {0, 1, 0}, p3.x, p3.z);
150 addVertex(mesh, p2, {0, 1, 0}, p2.x, p2.z);
151 mesh.addTriangle(top, top + 2, top + 1);
152 }

155void addSphere(MeshBuild& mesh, Vec3 center, Vec3 radii, int rings, int sides, const char* group) {
156 mesh.setActiveGroup(group);
157 rings = std::clamp(rings, 2, 32);
158 sides = std::clamp(sides, 3, 64);
159 const uint32_t base = static_cast<uint32_t>(mesh.getVertexCount());
160 for (int ring = 0; ring <= rings; ++ring) {
161 const float v = float(ring) / float(rings);
162 const float phi = v * kPi;
163 for (int side = 0; side <= sides; ++side) {
164 const float u = float(side) / float(sides);
165 const float theta = u * 2.0f * kPi;
166 const Vec3 n{std::sin(phi) * std::cos(theta), std::cos(phi), std::sin(phi) * std::sin(theta)};
167 const float horizontalRadius = (radii.x + radii.z) * 0.5f;
168 addVertex(mesh, {center.x + n.x * radii.x, center.y + n.y * radii.y, center.z + n.z * radii.z}, n,
169 theta * horizontalRadius, phi * radii.y);
170 }
171 }
172 for (int ring = 0; ring < rings; ++ring) {
173 for (int side = 0; side < sides; ++side) {
174 const uint32_t a = base + uint32_t(ring * (sides + 1) + side);
175 const uint32_t b = a + uint32_t(sides + 1);
176 mesh.addTriangle(a, a + 1, b);
177 mesh.addTriangle(a + 1, b + 1, b);
178 }
179 }

182void addTorus(MeshBuild& mesh, Vec3 center, float majorRadius, float minorRadius, int majorSegments, int minorSegments,
183 const char* group) {
184 mesh.setActiveGroup(group);

186 minorSegments = std::clamp(minorSegments, 3, 32);
187 const uint32_t base = static_cast<uint32_t>(mesh.getVertexCount());
188 for (int major = 0; major <= majorSegments; ++major) {

190 const float a = 2.0f * kPi * u;
191 for (int minor = 0; minor <= minorSegments; ++minor) {

194 const Vec3 n{std::cos(a) * std::cos(b), std::sin(b), std::sin(a) * std::cos(b)};
195 const float radial = majorRadius + minorRadius * std::cos(b);
196 addVertex(mesh,

198 center.z + radial * std::sin(a)},

200 }
201 }
202 for (int major = 0; major < majorSegments; ++major) {
203 for (int minor = 0; minor < minorSegments; ++minor) {
