133void AudioSourceEditor::generateTonePcm() {
134 constexpr int kRate = 22050;
135 constexpr double kSeconds = 2.0;
136 const int samples = static_cast<int>(kRate * kSeconds);
137 pcm_.sampleRate = kRate;
138 pcm_.channels = 1;
139 pcm_.samples.resize(static_cast<std::size_t>(samples));
140 for (int i = 0; i < samples; ++i) {
141 const double t = static_cast<double>(i) / static_cast<double>(kRate);
142 pcm_.samples[static_cast<std::size_t>(i)] =
143 static_cast<float>(0.32 * std::sin(6.283185307179586 * 440.0 * t) +
144 0.12 * std::sin(6.283185307179586 * 660.0 * t));
145 }
146}

342audio_editing::EditorResult<void> AudioSourceEditor::attachLiveAudition() {
343#if !defined(EVE_AUDIO_EDITOR_LIVE)
344 return editorError(audio_editing::EditorStatus::Unsupported, "editor.audio.live-module",
345 "Audio module is not available for live audition");
346#else
347 auto* audioMod = eve::ModuleManager::getInstance<eve::audio::Audio>("Audio");
348 if (!audioMod)
349 return editorError(audio_editing::EditorStatus::Unsupported, "editor.audio.live-module",
350 "Audio module is not available for live audition");
351 std::vector<std::uint8_t> bytes(pcm_.samples.size() * 2);
352 for (std::size_t i = 0; i < pcm_.samples.size(); ++i) {
353 const float sample = std::clamp(pcm_.samples[i], -1.0f, 1.0f);
354 const auto quantized = static_cast<std::int16_t>(std::lround(sample * 32767.0f));
355 bytes[i * 2] = static_cast<std::uint8_t>(quantized & 0xff);
356 bytes[i * 2 + 1] = static_cast<std::uint8_t>((quantized >> 8) & 0xff);
357 }
358 const bool wasPlaying = isPlaying();
359 const double head = playhead();
360 auto live = std::make_unique<LiveAudition>();
361 try {
362 live->data = std::make_unique<eve::sound::SoundData>(std::move(bytes), pcm_.sampleRate, 16, 1);
363 transport_.unbind();
364 live->source = audioMod->newSource(live->data.get());
365 } catch (const eve::Exception& ex) {
366 auto restored = bindClockAudition();
367 if (!restored.ok()) restored.ignore("clock audition restored after live attach failure");
368 return editorError(audio_editing::EditorStatus::Failed, "editor.audio.live-source",
369 std::string("Live audition source could not be created: ") + ex.what());
370 }
371 live->backend = std::make_unique<audio_editing::AudioSourceTransportBackend>(live->source);
372 auto bound = transport_.bind(audio_editing::StableId(target_.targetId().value()), auditionRevision(),
373 live->backend.get());
374 if (!bound.ok()) {
375 auto restored = bindClockAudition();
376 if (!restored.ok()) restored.ignore("clock audition restored after live bind failure");
377 auto looped = syncLoop();
378 if (!looped.ok()) looped.ignore("loop restored after live bind failure");
379 return bound;
380 }
381 liveAudition_ = std::move(live);
382 liveBound_ = true;
383 auto looped = syncLoop();
384 if (!looped.ok()) return looped;
385 auto published = publishLive();
386 if (!published.ok()) return published;
387 auto sought = transport_.seek(auditionRevision(), head);
388 if (!sought.ok()) return sought;
389 if (wasPlaying) return play();
390 return eve::editing::applied<void>();
391#endif
392}
