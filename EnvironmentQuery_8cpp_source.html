52Result<CompiledTest> compileTest(const EnvironmentQueryDefinition& definition, const EnvironmentTest& test) {
53 auto slot = contextSlot(definition, test.context);
54 if (!slot) return Result<CompiledTest>::failure(slot.status());
55 if (test.id.empty() || test.metric > EnvironmentMetricKind::Custom ||
56 test.purpose > EnvironmentPurpose::FilterAndScore || test.filterOp > EnvironmentFilterOp::Any ||
57 test.aggregate > EnvironmentAggregate::Maximum ||
58 test.normalization > EnvironmentNormalization::ActivePopulation ||
59 test.preference > EnvironmentPreference::Reference || test.curve > EnvironmentCurve::SquareRoot ||
60 !std::isfinite(test.minimum) || !std::isfinite(test.maximum) || test.minimum > test.maximum ||
61 !std::isfinite(test.scoreMinimum) || !std::isfinite(test.scoreMaximum) ||
62 test.scoreMinimum >= test.scoreMaximum || !std::isfinite(test.scoreMaximum - test.scoreMinimum) ||
63 !std::isfinite(test.referenceValue) || !std::isfinite(test.weight) || test.weight < 0) {
64 return invalid<CompiledTest>("invalid metric, threshold or scoring configuration", test.id);
65 }
66 EnvironmentMetricContract contract{definition.space, EnvironmentCost::Low, true, false};
67 if (test.metric == EnvironmentMetricKind::Custom) {
68 if (!test.custom) return invalid<CompiledTest>("custom metric is missing", test.id);
69 contract = test.custom->contract();
70 if (contract.space != definition.space || contract.cost > EnvironmentCost::High)
71 return invalid<CompiledTest>("metric does not support query space or cost", test.id);
72 } else if (test.custom) {
73 return invalid<CompiledTest>("built-in metric cannot carry a custom implementation", test.id);
74 } else if (test.metric == EnvironmentMetricKind::LineOfSight) {
75 contract.cost = EnvironmentCost::High;
76 contract.requiresLineOfSight = true;
77 }
78 return Result<CompiledTest>::success(CompiledTest{test, slot.value(), contract});
79}

81void optimize(CompiledOption& option, EnvironmentRunMode mode) {
82 // Population-relative normalization is order-sensitive: retain authored order.
83 if (std::any_of(option.tests.begin(), option.tests.end(), [](const auto& test) {
84 return scores(test.definition) &&
85 test.definition.normalization == EnvironmentNormalization::ActivePopulation;
86 }))
87 return;
88 auto begin = option.tests.begin();
89 while (begin != option.tests.end()) {
90 if (!begin->contract.reorderable) {
91 ++begin;
92 continue;
93 }
94 auto end = std::find_if(begin, option.tests.end(), [](const auto& test) { return !test.contract.reorderable; });
95 std::stable_sort(begin, end, [](const auto& a, const auto& b) {
96 // Preserve authored score accumulation order, including FilterAndScore,
97 // so floating-point rounding cannot change a close ranking after optimization.
98 if (scores(a.definition) != scores(b.definition)) return !scores(a.definition);
99 if (scores(a.definition)) return false;
100 return a.contract.cost < b.contract.cost;
101 });
102 begin = end;
103 }
104 if (mode != EnvironmentRunMode::Best || std::any_of(option.tests.begin(), option.tests.end(),
105 [](const auto& test) { return !test.contract.reorderable; }))
106 return;
107 std::size_t best = noIndex;
108 for (std::size_t i = 0; i < option.tests.size(); ++i) {
109 if (option.tests[i].definition.purpose == EnvironmentPurpose::Filter &&
110 option.tests[i].contract.cost != EnvironmentCost::Low &&
111 (best == noIndex || option.tests[i].contract.cost > option.tests[best].contract.cost))
112 best = i;
113 }
114 if (best != noIndex) {
115 auto deferred = std::move(option.tests[best]);
116 option.tests.erase(option.tests.begin() + static_cast<std::ptrdiff_t>(best));
117 option.tests.push_back(std::move(deferred));
118 option.deferred = true;
119 }
120}

122Result<std::vector<EnvironmentItem>> generate(const CompiledOption& option, const EnvironmentRequest& request,
123 CoordinateSpace space) {
124 const auto& generator = option.definition.generator;
125 if (generator.kind == EnvironmentGeneratorKind::Input) {
126 if (request.input.size() > generator.maxItems)
127 return invalid<std::vector<EnvironmentItem>>("input exceeds candidate budget", option.definition.id);
128 return Result<std::vector<EnvironmentItem>>::success(request.input);
129 }
130 const auto& centers = request.contexts[option.generatorContext].locations;
131 const double sideValue = std::floor(2 * generator.radius / generator.spacing) + 1;
132 const double perCenter =
133 generator.kind == EnvironmentGeneratorKind::Ring ? double(generator.samples) : sideValue * sideValue;
134 if (!std::isfinite(perCenter) || perCenter * double(centers.size()) > generator.maxItems)
135 return invalid<std::vector<EnvironmentItem>>("generated candidate budget exceeded", option.definition.id);
136 std::vector<EnvironmentItem> items;
137 items.reserve(static_cast<std::size_t>(perCenter) * centers.size());
138 auto append = [&](double x, double y, double z) -> Result<void> {
139 auto point = space == CoordinateSpace::World2D ? WorldPoint::world2D(float(x), float(y))
140 : WorldPoint::world3D(float(x), float(y), float(z));
141 if (!point) return Result<void>::failure(point.status());
142 items.push_back({items.size(), TargetLocation{point.value()}, std::nullopt});
143 return Result<void>::success();
144 };
145 for (const auto& center : centers) {
146 const auto p = position(center);
147 if (generator.kind == EnvironmentGeneratorKind::Ring) {
148 for (std::uint32_t i = 0; i < generator.samples; ++i) {
149 const double angle = 6.2831853071795864769 * i / generator.samples;
150 auto added =
151 append(p.x + generator.radius * std::cos(angle), p.y + generator.radius * std::sin(angle), p.z);
152 if (!added) return Result<std::vector<EnvironmentItem>>::failure(added.status());
153 }
154 } else {
155 const auto side = static_cast<std::uint32_t>(sideValue);
156 for (std::uint32_t x = 0; x < side; ++x) {
157 for (std::uint32_t y = 0; y < side; ++y) {
158 auto added = append(p.x + (double(x) - double(side - 1) / 2) * generator.spacing,
159 p.y + (double(y) - double(side - 1) / 2) * generator.spacing, p.z);
160 if (!added) return Result<std::vector<EnvironmentItem>>::failure(added.status());
161 }
162 }
163 }
164 }
165 return Result<std::vector<EnvironmentItem>>::success(std::move(items));
166}

168struct Measurement {
169 double raw = 0;
170 bool passed = false;
171 std::size_t calls = 0;
172};
173Result<Measurement> measure(const CompiledTest& test, const EnvironmentItem& item,
174 std::span<const TargetLocation> contexts, const EnvironmentServices& services) {
175 Measurement result;
176 std::size_t passed = 0;
177 for (const auto& context : contexts) {
178 double raw = 0;
179 if (test.definition.metric == EnvironmentMetricKind::Custom) {
180 auto measured = test.definition.custom->measure(item, context, services);
181 if (!measured) return Result<Measurement>::failure(measured.status());
182 raw = measured.value();
183 } else if (test.definition.metric == EnvironmentMetricKind::LineOfSight) {
184 auto visible = services.lineOfSight->get().query(context, item.location);
185 if (!visible) return Result<Measurement>::failure(visible.status());
186 raw = visible.value().visible ? 1 : 0;
187 } else {
188 const auto a = position(item.location), b = position(context);
189 raw = std::hypot(a.x - b.x, a.y - b.y, a.z - b.z);
190 }
191 if (!std::isfinite(raw)) return invalid<Measurement>("metric produced a non-finite value", test.definition.id);
192 if (raw >= test.definition.minimum && raw <= test.definition.maximum) ++passed;
193 ++result.calls;
194 if (result.calls == 1)
195 result.raw = raw;
196 else if (test.definition.aggregate == EnvironmentAggregate::Minimum)
197 result.raw = std::min(result.raw, raw);
198 else if (test.definition.aggregate == EnvironmentAggregate::Maximum)
199 result.raw = std::max(result.raw, raw);
200 else
201 result.raw = std::lerp(result.raw, raw, 1.0 / double(result.calls));
202 }
203 result.passed = !filters(test.definition) ||
204 (test.definition.filterOp == EnvironmentFilterOp::All ? passed == contexts.size() : passed > 0);
205 return Result<Measurement>::success(result);
206}

208double utility(const EnvironmentTest& test, double raw, double lower, double upper) {
209 if (upper == lower) return 0; // No discrimination in a constant relative population.
210 const double value = std::clamp(raw, lower, upper);
211 double normalized = (value - lower) / (upper - lower);
212 if (test.preference == EnvironmentPreference::Lower) normalized = 1 - normalized;
213 if (test.preference == EnvironmentPreference::Reference) {
214 const double reference = std::clamp(test.referenceValue, lower, upper);
215 normalized = 1 - std::abs(value - reference) / std::max(reference - lower, upper - reference);
216 }
217 normalized = std::clamp(normalized, 0.0, 1.0);
218 if (test.curve == EnvironmentCurve::Square) normalized *= normalized;
219 if (test.curve == EnvironmentCurve::SquareRoot) normalized = std::sqrt(normalized);
220 return normalized;
221}

229Result<EnvironmentQueryPlan> EnvironmentQueryPlan::compile(const EnvironmentQueryDefinition& definition) {
230 if (definition.id.empty() || definition.revision == 0 || definition.options.empty() ||
231 definition.options.size() > 32 || definition.contexts.size() > 256 ||
232 definition.space > CoordinateSpace::Grid3D || definition.mode > EnvironmentRunMode::Best ||
233 definition.trace > EnvironmentTraceLevel::Detailed || definition.maxResults == 0 ||
234 definition.maxResults > itemLimit || definition.maxTraceItems > itemLimit) {
235 return invalid<EnvironmentQueryPlan>("invalid query definition or resource limits", "query");
236 }
237 std::set<std::string> names;
238 for (const auto& context : definition.contexts) {
239 if (context.empty() || !names.insert(context).second)
240 return invalid<EnvironmentQueryPlan>("empty or duplicate context", context);
241 }
242 auto plan = std::make_shared<EnvironmentPlanData>();
243 plan->definition = definition;
244 names.clear();
245 for (const auto& option : definition.options) {
246 const auto& generator = option.generator;
247 if (option.id.empty() || !names.insert(option.id).second || option.tests.size() > 64 ||
248 generator.kind > EnvironmentGeneratorKind::Grid || generator.maxItems == 0 ||
249 generator.maxItems > itemLimit || !std::isfinite(generator.radius) || generator.radius < 0 ||
250 !std::isfinite(generator.spacing) || generator.spacing <= 0 || generator.samples == 0 ||
251 generator.samples > itemLimit) {
252 return invalid<EnvironmentQueryPlan>("invalid option or generator", option.id);
253 }
254 CompiledOption compiled;
255 compiled.definition = option;
256 if (generator.kind != EnvironmentGeneratorKind::Input) {
257 if (definition.space != CoordinateSpace::World2D && definition.space != CoordinateSpace::World3D)
258 return invalid<EnvironmentQueryPlan>("ring/grid generators require world coordinates", option.id);
259 auto slot = contextSlot(definition, generator.context);
260 if (!slot) return Result<EnvironmentQueryPlan>::failure(slot.status());
261 compiled.generatorContext = slot.value();
262 }
263 std::set<std::string> testNames;
264 for (const auto& test : option.tests) {
265 if (!testNames.insert(test.id).second) return invalid<EnvironmentQueryPlan>("duplicate test id", test.id);
266 auto compiledTest = compileTest(definition, test);
267 if (!compiledTest) return Result<EnvironmentQueryPlan>::failure(compiledTest.status());
268 compiled.tests.push_back(std::move(compiledTest).takeValue());
269 }
270 if (definition.optimize) optimize(compiled, definition.mode);
271 plan->options.push_back(std::move(compiled));
272 }
273 return Result<EnvironmentQueryPlan>::success(EnvironmentQueryPlan{std::move(plan)});
274}

276Result<EnvironmentQueryInstance> EnvironmentQueryPlan::instantiate(EnvironmentRequest request) const {
277 if (!data_) return invalid<EnvironmentQueryInstance>("moved-from query plan", "plan");
278 if (request.input.size() > itemLimit || request.contexts.size() != data_->definition.contexts.size())
279 return invalid<EnvironmentQueryInstance>("input limit or context count mismatch", "request");
280 std::set<std::uint64_t> ids;
281 for (const auto& item : request.input) {
282 if (!validLocation(item.location, data_->definition.space) || !ids.insert(item.id).second ||
283 (item.subjectKey && item.subjectKey->empty()))
284 return invalid<EnvironmentQueryInstance>("invalid candidate location, identity or duplicate id", "input");
285 }
286 std::vector<EnvironmentContext> resolved;
287 resolved.reserve(request.contexts.size());
288 for (const auto& name : data_->definition.contexts) {
289 auto found = std::find_if(request.contexts.begin(), request.contexts.end(),
290 [&](const auto& c) { return c.name == name; });
291 if (found == request.contexts.end() || found->locations.empty() || found->locations.size() > 256)
292 return invalid<EnvironmentQueryInstance>("missing, empty or oversized context", name);
293 for (const auto& location : found->locations) {
294 if (!validLocation(location, data_->definition.space))
295 return invalid<EnvironmentQueryInstance>("context coordinate mismatch", name);
296 }
297 resolved.push_back(std::move(*found));
298 }
299 request.contexts = std::move(resolved);
300 return Result<EnvironmentQueryInstance>::success(EnvironmentQueryInstance{data_, std::move(request)});
301}

311 if (started_ || !plan_)

313 started_ = true;
314 trace_.state = "running";
315 auto result = [&]() -> Result<EnvironmentQueryResult> {
316 try {
317 return evaluate(services);

320 Diagnostic::error(DiagnosticCode::CallbackFailure, error.what(), "environment.metric", {}, "sensing"));
321 } catch (...) {
322 return Result<EnvironmentQueryResult>::failure(
323 Diagnostic::error(DiagnosticCode::CallbackFailure, "query provider threw an unknown exception",
324 "environment.metric", {}, "sensing"));
325 }
326 }();
327 trace_.state = result.ok() ? "completed" : "failed";
328 if (!result.ok()) trace_.diagnostics = result.diagnostics();
329 return result;

332Result<EnvironmentQueryResult> EnvironmentQueryInstance::evaluate(const EnvironmentServices& services) {
333 const auto& definition = plan_->definition;
334 for (const auto& option : plan_->options) {
335 // Missing required providers are errors, even when generation would produce no items.
336 for (const auto& test : option.tests) {
337 if (test.contract.requiresLineOfSight && !services.lineOfSight) {
338 return Result<EnvironmentQueryResult>::failure(
339 Diagnostic::error(DiagnosticCode::Unsupported, "query requires a line-of-sight provider",
340 test.definition.id, {}, "sensing"));
341 }
342 }
343 auto generated = generate(option, request_, definition.space);
344 if (!generated) return Result<EnvironmentQueryResult>::failure(generated.status());
345 auto items = std::move(generated).takeValue();
346 std::vector<std::size_t> active(items.size());
347 std::iota(active.begin(), active.end(), 0);
348 std::vector<double> totals(items.size(), 0);
349 std::vector<std::size_t> traceIndex(items.size(), noIndex);
350 if (definition.trace != EnvironmentTraceLevel::None)
351 trace_.stages.push_back({option.definition.id, "generate", 0, items.size(), 0, false});
352 if (definition.trace == EnvironmentTraceLevel::Detailed) {
353 for (std::size_t i = 0; i < items.size(); ++i) {
354 if (trace_.items.size() >= definition.maxTraceItems) {
355 trace_.truncated = true;
356 break;
357 }
358 traceIndex[i] = trace_.items.size();
359 trace_.items.push_back({option.definition.id, items[i].id, {}, {}});
360 }
361 }
362 auto rank = [&](std::size_t a, std::size_t b) {
363 if (totals[a] != totals[b]) return totals[a] > totals[b];
364 return items[a].id < items[b].id;
365 };
366 for (std::size_t ti = 0; ti < option.tests.size() && !active.empty(); ++ti) {
367 const auto& compiled = option.tests[ti];
368 const auto& test = compiled.definition;
369 const bool deferred = option.deferred && ti + 1 == option.tests.size();
370 if (deferred) std::sort(active.begin(), active.end(), rank);
371 EnvironmentStageTrace stage{option.definition.id, test.id, active.size(), 0, 0, deferred};
372 std::vector<std::pair<std::size_t, Measurement>> measured;
373 measured.reserve(active.size());
374 for (auto index : active) {
375 auto measurement =
376 measure(compiled, items[index], request_.contexts[compiled.context].locations, services);
377 if (!measurement) {
378 if (definition.trace != EnvironmentTraceLevel::None) trace_.stages.push_back(stage);
379 return Result<EnvironmentQueryResult>::failure(measurement.status());
380 }
381 const auto value = measurement.value();
382 stage.measured += value.calls;
383 if (value.passed) ++stage.output;
384 measured.emplace_back(index, value);
385 if (deferred && value.passed) break;
386 }
387 double lower = test.scoreMinimum, upper = test.scoreMaximum;
388 if (scores(test) && test.normalization == EnvironmentNormalization::ActivePopulation) {
389 lower = std::numeric_limits<double>::max();
390 upper = std::numeric_limits<double>::lowest();
391 for (const auto& [index, value] : measured) {
392 (void)index;
393 if (value.passed) {
394 lower = std::min(lower, value.raw);
395 upper = std::max(upper, value.raw);
396 }
397 }
398 if (stage.output == 0) lower = upper = 0;
399 if (!std::isfinite(upper - lower))
400 return invalid<EnvironmentQueryResult>("normalization span overflow", test.id);
401 }
402 active.clear();
403 for (const auto& [index, value] : measured) {
404 const double normalized = value.passed && scores(test) ? utility(test, value.raw, lower, upper) : 0;
405 const double weighted = normalized * test.weight;
406 if (value.passed) {
407 totals[index] += weighted;
408 if (!std::isfinite(totals[index]))
409 return invalid<EnvironmentQueryResult>("score accumulation overflow", test.id);
410 active.push_back(index);
411 }
412 if (traceIndex[index] != noIndex) {
413 auto& detail = trace_.items[traceIndex[index]];
414 detail.contributions.push_back(
415 {test.id, value.raw, lower, upper, normalized, weighted, value.passed});
416 if (!value.passed) detail.rejectedBy = test.id;
417 }
418 }
419 if (definition.trace != EnvironmentTraceLevel::None) trace_.stages.push_back(std::move(stage));
420 }
421 if (active.empty()) continue;
422 std::sort(active.begin(), active.end(), rank);
423 const auto count = definition.mode == EnvironmentRunMode::Best ? 1u : definition.maxResults;
424 if (active.size() > count) active.resize(count);
425 EnvironmentQueryResult output;
426 output.option = option.definition.id;
427 for (auto index : active) output.ranked.push_back({std::move(items[index]), totals[index]});
428 return Result<EnvironmentQueryResult>::success(std::move(output));
429 }
430 return Result<EnvironmentQueryResult>::success(EnvironmentQueryResult{});
431}

433Value EnvironmentTrace::toValue() const {
434 Value::Array stageValues, itemValues, failures;
435 for (const auto& diagnostic : diagnostics) {
436 failures.emplace_back(Value::Object{
437 {"code", std::int64_t(diagnostic.code())}, {"message", diagnostic.message()}, {"path", diagnostic.path()}});
438 }
439 for (const auto& stage : stages) {
440 stageValues.emplace_back(Value::Object{{"option", stage.option},
441 {"test", stage.test},
442 {"input", std::int64_t(stage.input)},
443 {"output", std::int64_t(stage.output)},
444 {"measurements", std::int64_t(stage.measured)},
445 {"deferred", stage.deferred}});
446 }
447 for (const auto& item : items) {
448 Value::Array contributions;
449 for (const auto& contribution : item.contributions) {
450 contributions.emplace_back(Value::Object{{"test", contribution.test},
451 {"raw", contribution.raw},
452 {"lower", contribution.lower},
453 {"upper", contribution.upper},
454 {"normalized", contribution.normalized},
455 {"weighted", contribution.weighted},
456 {"passed", contribution.passed}});
457 }
458 itemValues.emplace_back(Value::Object{{"option", item.option},
459 {"id", std::to_string(item.id)},
460 {"rejectedBy", item.rejectedBy},
461 {"contributions", std::move(contributions)}});
462 }
463 return Value::Object{{"schema", "eve.sensing.environmentTrace"},
464 {"version", 1},
465 {"query", query},
466 {"revision", std::to_string(revision)},
467 {"state", state},
468 {"truncated", truncated},
469 {"diagnostics", std::move(failures)},
470 {"stages", std::move(stageValues)},
471 {"items", std::move(itemValues)}};
472}
