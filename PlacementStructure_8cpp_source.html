21void PlacementSystem::dissolveEdgeCurveGroup(PlacementWorld *world, EdgeCurveGroupId groupId) {
22 if (!world || !groupId) return;
23 const auto group = world->edgeCurveGroups_.find(groupId.value);
24 if (group == world->edgeCurveGroups_.end()) return;
25 for (int memberId : group->second.instanceIds) {
26 const auto member = world->buildings_.find(memberId);
27 if (member != world->buildings_.end() && member->second.edgeCurveGroupId == groupId)
28 member->second.edgeCurveGroupId = {};
29 }
30 world->edgeCurveGroups_.erase(group);
31}

33bool PlacementSystem::removeBuildingUnchecked(PlacementWorld *world, int instanceId) {
34 if (!world || instanceId <= 0) return false;

36 if (it == world->buildings().end()) return false;
37

40 ev.action = "remove";
41 ev.worldId = world->getId();
42 ev.buildingId = it->second.buildingId;
43 ev.instanceId = instanceId;
44 ev.cellX = it->second.originCellX;
45 ev.cellY = it->second.originCellY;
46 ev.rotationDeg = it->second.rotationDeg;
47 ev.worldX = it->second.worldX;
48 ev.worldY = it->second.worldY;
49 ev.elevation = it->second.elevation;
50 ev.channel = it->second.channel;
51
52 dissolveEdgeCurveGroup(world, curveGroupId);
53 clearOccupancy(*world, instanceId);
54 assembly_detail::removeMembership(*world->assemblyState_, instanceId);
55 world->buildings().erase(it);

72eve::Result<StructuralRemovalReceipt> PlacementSystem::removeBuildingCascadeResult(PlacementWorld *world,
73 int instanceId) {
74 if (!world || instanceId <= 0) {
75 return eve::Result<StructuralRemovalReceipt>::failure(eve::Diagnostic::error(
76 eve::DiagnosticCode::InvalidArgument, "cascade removal requires a world and positive instance id",
77 std::to_string(instanceId), {}, "building.structure"));
78 }
79 if (!world->hasBuilding(instanceId)) {
80 return eve::Result<StructuralRemovalReceipt>::failure(
81 eve::Diagnostic::error(eve::DiagnosticCode::NotFound, "cascade removal source instance was not found",
82 std::to_string(instanceId), {}, "building.structure"));
83 }
84 std::vector<int> ordered;
85 std::unordered_set<int> visiting;
86 std::unordered_set<int> visited;
87 bool cycle = false;
88 std::function<void(int)> collect = [&](int supportId) {
89 if (visited.contains(supportId)) return;
90 if (!visiting.insert(supportId).second) {
91 cycle = true;
92 return;
93 }
94 for (const auto &[candidateId, placed] : world->buildings_) {
95 if (std::find(placed.supportInstanceIds.begin(), placed.supportInstanceIds.end(), supportId) !=
96 placed.supportInstanceIds.end())
97 collect(candidateId);
98 }
99 visiting.erase(supportId);
100 visited.insert(supportId);
101 ordered.push_back(supportId);
102 };
103 collect(instanceId);
104 if (cycle) {
105 return eve::Result<StructuralRemovalReceipt>::failure(
106 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "structural dependency cycle prevents removal",
107 "support_cycle", {}, "building.structure"));
108 }
109
110 StructuralRemovalReceipt receipt;
111 receipt.removed.reserve(ordered.size());
112 for (int id : ordered) receipt.removed.push_back(world->buildings_.at(id));
113 const bool publishEvents = world->publishEvents_;
114 world->publishEvents_ = false;
115 for (int id : ordered) removeBuildingUnchecked(world, id);
116 world->publishEvents_ = publishEvents;
117 if (publishEvents) {
118 for (const PlacedBuilding &removed : receipt.removed) {
119 BuildingChangeEvent event;
120 event.action = "remove";
121 event.worldId = world->getId();
122 event.buildingId = removed.buildingId;
123 event.instanceId = removed.instanceId;
124 event.cellX = removed.originCellX;
125 event.cellY = removed.originCellY;
126 event.level = removed.level;
127 event.rotationDeg = removed.rotationDeg;
128 event.worldX = removed.worldX;
129 event.worldY = removed.worldY;
130 event.elevation = removed.elevation;
131 event.channel = removed.channel;
132 emit(std::move(event));
133 }
134 }
135 return eve::Result<StructuralRemovalReceipt>::success(std::move(receipt));
136}

143eve::Result<StructuralLinkRebuildReceipt> PlacementSystem::rebuildStructuralLinksResult(PlacementWorld *world) {
144 if (!world) {
145 return eve::Result<StructuralLinkRebuildReceipt>::failure(eve::Diagnostic::error(
146 eve::DiagnosticCode::InvalidArgument, "structural rebuild requires a world", {}, {}, "building.structure"));
147 }
148 auto candidate = world->cloneState();
149 if (!candidate) {
150 return eve::Result<StructuralLinkRebuildReceipt>::failure(
151 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
152 "failed to clone structural rebuild candidate", {}, {}, "building.structure"));
153 }
154
155 StructuralLinkRebuildReceipt receipt;
156 for (int id : candidate->instanceOrder_) {
157 auto placedIt = candidate->buildings_.find(id);
158 if (placedIt == candidate->buildings_.end()) continue;
159 PlacedBuilding &placed = placedIt->second;
160 ++receipt.inspectedCount;
161 const BuildingDefinition *def = BuildingRegistry::find(placed.buildingId);
162 if (!def) {
163 return eve::Result<StructuralLinkRebuildReceipt>::failure(eve::Diagnostic::error(
164 eve::DiagnosticCode::NotFound, "structural rebuild encountered an unknown building definition",
165 std::to_string(id), {}, "building.structure"));
166 }
167
168 std::vector<int> rebuilt;
169 if (def->supportMode != "none") {
170 PlacementQuery query;
171 query.buildingId = placed.buildingId;
172 query.cellX = placed.originCellX;
173 query.cellY = placed.originCellY;
174 query.level = placed.level;
175 query.rotationDeg = placed.rotationDeg;
176 std::string reason;
177 if (!checkStructuralSupport(*candidate, *def, query, &reason)) {
178 return eve::Result<StructuralLinkRebuildReceipt>::failure(eve::Diagnostic::error(
179 eve::DiagnosticCode::Conflict, "structural rebuild found an invalid dependency",
180 std::to_string(id) + ":" + reason, {}, "building.structure"));
181 }
182 rebuilt = collectStructuralSupports(*candidate, *def, query);
183 }
184 if (rebuilt != placed.supportInstanceIds) {
185 placed.supportInstanceIds = std::move(rebuilt);
186 ++receipt.changedCount;
187 }
188 }
189 world->swapState(*candidate);
190 return eve::Result<StructuralLinkRebuildReceipt>::success(std::move(receipt));
191}
