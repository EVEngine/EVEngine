11McpToolFamilyCatalog::McpToolFamilyCatalog(const std::vector<McpToolFamily>& families) {
12 std::set<std::string_view> ids;
13 // All state belongs to this unexposed object until construction succeeds.
14 for (const auto& family : families) {
15 if (family.id.empty() || !family.schemas || !family.route.call || !ids.insert(family.id).second) {
16 throw std::invalid_argument("invalid or duplicate MCP family: " + std::string(family.id));
17 }
18 Poco::JSON::Array::Ptr tools;
19 try {
20 Poco::JSON::Parser parser;
21 tools = parser.parse("[" + std::string(family.schemas()) + "]").extract<Poco::JSON::Array::Ptr>();
22 } catch (const std::exception& error) {
23 throw std::invalid_argument("invalid MCP schemas in " + std::string(family.id) + ": " + error.what());
24 }
25 for (std::size_t i = 0; i < tools->size(); ++i) {
26 const auto tool = tools->getObject(static_cast<unsigned>(i));
27 if (!tool || !tool->has("name") || !tool->get("name").isString() || !tool->getObject("inputSchema")) {
28 throw std::invalid_argument("MCP schema needs a string name and object inputSchema: " +
29 std::string(family.id));
30 }
31 const auto name = tool->getValue<std::string>("name");
32 if (name.empty() || !routes_.emplace(name, family.route).second) {
33 throw std::invalid_argument("empty or duplicate MCP tool name: " + name);
34 }
35 if (!schemas_.empty()) schemas_ += ',';
36 schemas_ += mcpStringify(Poco::Dynamic::Var(tool));
37 }
38 }
39}
