23// Runtime binary name inside the *target* SDK's bin/. The packaged game runs on
24// the SDK's platform, which is not necessarily the host running `eve package`
25// (e.g. a Linux host can produce a Windows game folder for the win32 SDK).
26std::string targetRuntimeName(const std::string& sdkRoot) {
27 std::ifstream in(std::filesystem::path(sdkRoot) / "share" / "eve" / "TARGET_PLATFORM");
28 std::string plat;
29 std::getline(in, plat);
30 return plat == "win32" ? "eve.exe" : "eve";
31}

33bool copyFileIf(const path& src, const path& dst) {
34 error_code ec;
35 if (!exists(src, ec)) return false;
36 create_directories(dst.parent_path(), ec);
37 copy_file(src, dst, copy_options::overwrite_existing, ec);
38 return !ec;
39}

41// Best-effort copy of a DLL from any of the candidate source directories.
42bool copyFirst(const std::string& dll, const std::vector<path>& candidates, const path& dstDir) {
43 for (const auto& dir : candidates) {
44 path p = dir / dll;
45 if (copyFileIf(p, dstDir / dll)) return true;
46 }
47 return false;
48}

50// Locate the target SDK root: --sdk, EVENGINE_SDK, or next to the running binary.
51std::string resolveSdkRoot(const std::string& sdkArg) {
52 if (!sdkArg.empty() && is_directory(sdkArg)) return sdkArg;
53
54 const char* env = getenv("EVENGINE_SDK");
55 if (env && is_directory(env)) return env;
56
57 // Running from <sdk>/bin/<runtime> -> root is two directories up.
58 path exe;
59#ifdef EVENGINE_WINDOWS
60 wchar_t buf[MAX_PATH + 1] = {0};
61 if (GetModuleFileNameW(nullptr, buf, MAX_PATH) != 0) exe = path(buf);
62#else
63 char buf[4096] = {0};
64 const ssize_t n = readlink("/proc/self/exe", buf, sizeof(buf) - 1);
65 if (n > 0) exe = path(std::string(buf, static_cast<size_t>(n)));
66#endif
67 if (!exe.empty()) {
68 path root = exe.parent_path().parent_path();
69 if (exists(root / "share" / "eve" / "TARGET_PLATFORM")) return root.string();
70 }
71 return "";
72}

98int Cmdline::Package(std::string gamePath, std::string output, std::string sdk) {
99 error_code ec;
100 if (!is_directory(gamePath, ec)) {
101 cerr << rang::fg::red << "Not a directory: " << rang::fg::reset << gamePath << endl;
102 return 1;
103 }
104
105 std::string sdkRoot = resolveSdkRoot(sdk);
106 if (sdkRoot.empty()) {
107 cerr << rang::fg::red
108 << "Cannot locate the target SDK. Pass --sdk <dir>, set $EVENGINE_SDK, "
109 "or run from <sdk>/bin/<runtime>."
110 << rang::fg::reset << endl;
111 return 2;
112 }
113
114 // Game folder name, trailing separators stripped.
115 std::string gameName = gamePath;
116 while (!gameName.empty() && (gameName.back() == '/' || gameName.back() == '\\'))
117 gameName.pop_back();
118 if (gameName.find_first_of("/\\") != string::npos)
119 gameName = std::filesystem::path(gameName).filename().string();
120 if (gameName.empty()) gameName = "game";
121
122 path outDir = output.empty() ? path(gameName + "-package") : path(output);
123 create_directories(outDir, ec);
124
125 path sdkBin = path(sdkRoot) / "bin";
126 path sdkLib = path(sdkRoot) / "lib";
127 path sdkPlat = path(sdkRoot) / "platform";
128
129 const std::string runtimeName = targetRuntimeName(sdkRoot);
130
131 // 1. Runtime executable.
132 path runtimeSrc = sdkBin / runtimeName;
133 path runtimeDst = outDir / runtimeName;
134 if (!copyFileIf(runtimeSrc, runtimeDst)) {
135 cerr << rang::fg::red << "SDK missing " << runtimeSrc.string() << rang::fg::reset << endl;
136 return 3;
137 }
138
139 // 2. Windows runtime DLLs the packaged game needs (vulkan loader + VC CRT).
140 // win32 SDKs bundle them in bin/ so ANY host can produce a self-contained
141 // package; on a Windows host we also fall back to the local VS redist for
142 // older SDKs that predate bundling.
143 if (runtimeName == "eve.exe") {
144 std::vector<path> crtCandidates;
145 if (exists(sdkBin)) crtCandidates.push_back(sdkBin);
146 if (exists(sdkLib)) crtCandidates.push_back(sdkLib);
147#ifdef EVENGINE_WINDOWS
148 // VC redist (version-independent glob): both release CRT and debug CRT.
149 for (const auto& msroot : {"C:/Program Files/Microsoft Visual Studio"}) {
150 path base = path(msroot);
151 if (exists(base / "18")) base = base / "18";
152 for (const auto& entry : directory_iterator(base, ec)) {
153 if (!entry.is_directory()) continue;
154 path redist = entry.path() / "Community" / "VC" / "Redist" / "MSVC";
155 if (!exists(redist)) continue;
156 for (const auto& ver : directory_iterator(redist, ec)) {
157 for (const auto& crtDir : {"x64", "debug_nonredist/x64"}) {
158 path crt = ver.path() / crtDir;
159 if (!exists(crt)) continue;
160 for (const auto& sub : directory_iterator(crt, ec)) {
161 if (!sub.is_directory()) continue;
162 const std::string n = sub.path().filename().string();
163 if (n.find("Microsoft.VC") == 0 &&
164 (n.find(".CRT") != string::npos ||
165 n.find("DebugCRT") != string::npos))
166 crtCandidates.push_back(sub.path());
167 }
168 }
169 }
170 }
171 }
172 if (exists(path("C:/Windows/System32")))
173 crtCandidates.push_back(path("C:/Windows/System32"));
174#endif
175
176 // eve.exe links the VC runtime dynamically; bundle both release and
177 // debug sets so the package is self-contained regardless of build type.
178 for (const auto& dll : {"vulkan-1.dll", "msvcp140.dll", "msvcp140_1.dll",
179 "msvcp140_2.dll", "vcruntime140.dll",
180 "vcruntime140_1.dll", "concrt140.dll",
181 "msvcp140d.dll", "vcruntime140d.dll",
182 "vcruntime140_1d.dll", "ucrtbased.dll",
183 "concrt140d.dll"}) {
184 if (!copyFirst(dll, crtCandidates, outDir))
185 cerr << rang::fg::yellow
186 << "note: runtime DLL not found (may be system-installed): "
187 << dll << rang::fg::reset << endl;
188 }
189 }
190
191#ifdef EVENGINE_MACOSX
192 // 2b. macOS: bundle the runtime dylibs (Vulkan loader, zlib, ...) that the
193 // executable links from the SDK's lib/ directory, so the packaged game is
194 // self-contained next to the eve binary.
195 if (exists(sdkLib)) {
196 for (const auto& entry : directory_iterator(sdkLib, ec)) {
197 if (entry.is_regular_file() && entry.path().extension() == ".dylib")
198 copyFileIf(entry.path(), outDir / entry.path().filename());
199 }
200 ec.clear();
201 }
202#endif // EVENGINE_MACOSX
203
204 // 3. Platform packaging template (win32/linux README etc).
205 if (exists(sdkPlat)) {
206 for (const auto& entry : directory_iterator(sdkPlat, ec)) {
207 copy(entry.path(), outDir / entry.path().filename(),
208 copy_options::recursive | copy_options::overwrite_existing, ec);
209 ec.clear();
210 }
211 }
212
213 // 4. Compress the game directory into game.eve inside the package.
214 path archive = outDir / "game.eve";
215 if (!cmdline::createGameArchive(gamePath, archive)) {
216 cerr << rang::fg::red << "Failed to package game archive." << rang::fg::reset << endl;
217 return 4;
218 }
219
220 cout << rang::fg::green << "Built game package -> " << rang::fg::reset << outDir.string() << endl;
221 cout << " run: " << runtimeDst.string() << endl;
222 return 0;
223}
