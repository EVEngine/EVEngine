17 DiagnosticSeverity::Error, diagnostic.message()));
18 const EditorStatus status = captured.code() == eve::StatusCode::NotFound
19 ? EditorStatus::NotFound
20 : EditorStatus::Failed;
21 return EditorResult<void>::failure(eve::Status(status, std::move(diagnostics)));
22 }
23 auto runtimeFrame = std::move(captured).takeValue();
24 EditorProfilerFrame frame;

26 frame.cpuFrameMs = runtimeFrame.cpuFrameMs;
27 frame.gpuFrameMs = runtimeFrame.gpuFrameMs;
28 frame.gpuTimingAvailable = runtimeFrame.gpuTimingAvailable;
29 frame.zones.reserve(runtimeFrame.zones.size());
30 for (auto& zone : runtimeFrame.zones) {
31 frame.zones.push_back({std::move(zone.module), std::move(zone.name),
32 std::move(zone.thread), zone.selfMs, zone.totalMs, zone.count,
33 zone.depth});
34 }
35 return model.ingest(std::move(frame));
