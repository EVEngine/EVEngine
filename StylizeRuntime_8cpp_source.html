24 if (!graphics)

26 const auto diagnostics = document.validate();
27 for (const auto& d : diagnostics)
28 if (d.severity() == DiagnosticSeverity::Error)
29 return EditorResult<void>::failure(eve::Status(EditorStatus::Rejected, diagnostics));
30 std::vector<std::unique_ptr<stylize::StyleInstance>> instances;

32 try {
33 for (const auto& pass : document.passes()) {
34 if (!pass.enabled) continue;
35 auto instance = std::make_unique<stylize::StyleInstance>(pass.style);
36 instance->setPriority(pass.priority);
37 for (const auto& [name, value] : pass.overrides) instance->setFloat(name, static_cast<float>(value));
38 recipe->add(instance.get());
39 instances.push_back(std::move(instance));
40 }
41 if (!instances.empty()) recipe->compile(graphics);
42 } catch (const std::exception& exception) {

44 }
45 instances_ = std::move(instances);
