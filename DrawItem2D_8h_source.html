60inline void sortDrawItems2D(std::vector<DrawItem2D> &items) {
61 std::stable_sort(items.begin(), items.end(), [](const DrawItem2D &a, const DrawItem2D &b) {
62 const bool aOff = a.canvas != nullptr;
63 const bool bOff = b.canvas != nullptr;
64 if (aOff != bOff) return aOff && !bOff;
65 if (a.canvas != b.canvas) return a.canvas < b.canvas;
66 if (a.layer != b.layer) return a.layer < b.layer;
67 if (a.hasOrder && b.hasOrder && a.order != b.order) return a.order < b.order;
68 if (a.depthY != b.depthY) return a.depthY < b.depthY;
69 // Do not use pipeline/texture state as a tie-breaker: alpha compositing is
70 // order-dependent. stable_sort preserves submission order for equal visual keys,
71 // while the backend still batches adjacent compatible items.
72 return false;
73 });
74}
