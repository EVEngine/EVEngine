26bool TagStore::mutate(std::map<SubjectId, ValueSet>& valuesBySubject, Index* reverse, const SubjectId& subject,
27 const std::string& value, bool add, const char* addedAction, const char* removedAction) {
28 if (subject.empty() || value.empty()) return false;
29 auto found = valuesBySubject.find(subject);
30 if (add) {
31 auto& values = valuesBySubject[subject];
32 if (!values.insert(value).second) return false;
33 if (reverse) (*reverse)[value].insert(subject);
34 emit(addedAction, subject, value);
35 return true;
36 }
37 if (found == valuesBySubject.end() || found->second.erase(value) == 0) return false;
38 if (found->second.empty()) valuesBySubject.erase(found);
39 if (reverse) {
40 auto indexed = reverse->find(value);
41 if (indexed != reverse->end()) {
42 indexed->second.erase(subject);
43 if (indexed->second.empty()) reverse->erase(indexed);
44 }
45 }
46 emit(removedAction, subject, value);
47 return true;
48}

124bool TagStore::removeSubject(const SubjectId& subject) {
125 const auto tags = tagsOf(subject);
126 const auto capabilities = capabilitiesOf(subject);
127 for (const auto& tag : tags) removeTag(subject, tag);
128 for (const auto& capability : capabilities) removeCapability(subject, capability);
129 if (tags.empty() && capabilities.empty()) return false;
130 emit("subject_removed", subject, {});
131 return true;
