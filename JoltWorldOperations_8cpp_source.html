41eve::Result<void> WorldBackend::setContactRecycleDistance(float distance) {
42 if (!isValid()) return rejected<void>("World expired", eve::DiagnosticCode::StaleHandle);
43 if (!std::isfinite(distance) || distance < 0 || double(distance) * distance > FLT_MAX)
44 return rejected<void>("Invalid contact persistence distance", eve::DiagnosticCode::InvalidArgument);
45 auto settings = state->system.GetPhysicsSettings();
46 settings.mContactPointPreserveLambdaMaxDistSq = distance * distance;
47 settings.mUseBodyPairContactCache = distance > 0;
48 state->system.SetPhysicsSettings(settings);
49 for (const auto& [id, body] : state->bodies) body->api().InvalidateContactCache(body->id);
50 return success();
51}
