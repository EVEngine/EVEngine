49EditorResult<void> AudioSourceTarget::applyDomainOperation(const DomainOperation& operation) {
50 if (operation.target != TargetId(id_) || operation.type != "audio.source.property.set.v1")
51 return audioError<void>(EditorStatus::Rejected, "editor.audio.operation",
52 "Operation is not valid for this audio source");
53 const EditorValue* pathValue = field(operation.payload, "path");
54 const EditorValue* value = field(operation.payload, "value");
55 const auto* path = pathValue ? pathValue->getIf<std::string>() : nullptr;
56 if (!path || !value)
57 return audioError<void>(EditorStatus::Rejected, "editor.audio.payload",
58 "Audio source operation requires path and value");
59 auto descriptor = sourceSchema().find(PropertyPath(*path));
60 if (!descriptor)
61 return audioError<void>(EditorStatus::Unsupported, "editor.audio.property",
62 "Unknown audio source property: " + *path);
63 auto valid = validatePropertyValue(*descriptor, *value);
64 if (!valid.ok()) return valid;
65 values_[*path] = *value;
66 ++revision_;
67 dirty_.include(0, 0);
68 return eve::editing::applied<void>();
69}

103EditorResult<DomainOperation> AudioSourceTarget::makeSet(const SelectionSnapshot& selection,
104 const PropertyPath& path,
105 const EditorValue& value,
106 PropertySetMode mode) const {
107 if (!selectionMatches(selection))
108 return audioError<DomainOperation>(EditorStatus::Rejected, "editor.audio.selection",
109 "Selection does not belong to this audio source");
110 if (mode == PropertySetMode::Reset) return makeReset(selection, path);
111 if (mode != PropertySetMode::Absolute)
112 return audioError<DomainOperation>(EditorStatus::Unsupported, "editor.audio.property-mode",
113 "Audio source properties require absolute assignment");
114 auto descriptor = sourceSchema().find(path);
115 if (!descriptor)
116 return audioError<DomainOperation>(EditorStatus::Unsupported, "editor.audio.property",
117 "Unknown audio source property: " + path.value());
118 auto valid = validatePropertyValue(*descriptor, value);
119 if (!valid.ok()) return EditorResult<DomainOperation>::failure(valid.status());
120 auto payload = [&](const EditorValue& assigned) {
121 EditorValue::Object object;
122 object["path"] = path.value();
123 object["value"] = assigned;
124 return EditorValue(std::move(object));
125 };
126 DomainOperation operation;
127 operation.type = "audio.source.property.set.v1";
128 operation.target = TargetId(id_);
129 operation.payload = payload(value);
130 operation.inverse = payload(values_.at(path.value()));
131 operation.hasInverse = true;
132 operation.affectedProperties.push_back(path.value());
133 operation.mergeKey = "audio.source:" + id_ + ":" + path.value();
134 return eve::editing::applied<DomainOperation>(std::move(operation));
135}

146PropertySchema AudioSourceTarget::sourceSchema() {
147 PropertySchema result;
148 result.typeId = "audio.source";
149 auto asset = property("clip.asset", "editor.audio.clip", "clip", PropertyType::AssetRef, "");
150 asset.assetTypeFilters = {"audio", "sound"};
151 result.properties.push_back(std::move(asset));
152 auto mode = property("clip.mode", "editor.audio.mode", "clip", PropertyType::Enum, "static");
153 mode.enumItems = {"static", "stream"};
154 result.properties.push_back(std::move(mode));
155 for (const auto& [path, label, value] : {
156 std::tuple{"play.autoplay", "editor.audio.autoplay", false},
157 {"play.loop", "editor.audio.loop", false},
158 {"spatial.relative", "editor.audio.relative", false}})
159 result.properties.push_back(property(path, label, path[0] == 'p' ? "playback" : "spatial",
160 PropertyType::Bool, value));
161 auto numeric = [&](const char* path, const char* label, const char* category,
162 double value, double minimum, double maximum) {
163 auto descriptor = property(path, label, category, PropertyType::Float, value);
164 descriptor.numeric.minimum = minimum;
165 descriptor.numeric.maximum = maximum;
166 descriptor.numeric.step = 0.01;
167 result.properties.push_back(std::move(descriptor));
168 };
169 numeric("play.volume", "editor.audio.volume", "playback", 1.0, 0.0, 16.0);
170 numeric("play.pitch", "editor.audio.pitch", "playback", 1.0, 0.01, 16.0);

172 numeric("play.loop-end", "editor.audio.loop-end", "playback", 0.0, 0.0, 86400.0);
173 numeric("spatial.reference-distance", "editor.audio.reference-distance", "spatial", 1.0, 0.0, 1000000.0);
174 numeric("spatial.maximum-distance", "editor.audio.maximum-distance", "spatial", 10000.0, 0.0, 1000000.0);
175 for (const auto& [path, label, value] : {

177 {"spatial.velocity", "editor.audio.velocity", EditorValue::Array{0.0, 0.0, 0.0}},
178 {"spatial.direction", "editor.audio.direction", EditorValue::Array{0.0, 0.0, -1.0}}})
