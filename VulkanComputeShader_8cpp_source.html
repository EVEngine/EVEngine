83void VulkanComputeShader::flushDescriptors(vkb::Device &device) {
84 if (!descriptorsDirty_ && descriptorSet_) return;
85
86 if (!descriptorPool_) {
87 // maxSets sets x kMaxBindings storage-buffer descriptors per set: the
88 // pool must hold the deferred backlog from multi-dispatch Sequences.
89 vk::DescriptorPoolSize poolSize{vk::DescriptorType::eStorageBuffer,
90 uint32_t(kMaxBindings) * 64};
91 vk::DescriptorPoolCreateInfo poolInfo{};
92 poolInfo.flags = vk::DescriptorPoolCreateFlagBits::eFreeDescriptorSet;
93 poolInfo.maxSets = 64;
94 poolInfo.poolSizeCount = 1;
95 poolInfo.pPoolSizes = &poolSize;
96 descriptorPool_ = device->createDescriptorPool(poolInfo, device.allocation_callbacks);
97 }
98
99 if (descriptorSet_) {
100 if (activeSequenceCount_ > 0) {
101 // A previously recorded command buffer in the open Sequence may
102 // still reference this set. Defer the free until that Sequence
103 // submits and waits; the pool is sized for the deferred backlog.
104 pendingSets_.push_back(descriptorSet_);
105 } else {
106 device->freeDescriptorSets(descriptorPool_, descriptorSet_);
107 }
108 descriptorSet_ = vk::DescriptorSet{};
109 }
110
111 vk::DescriptorSetAllocateInfo alloc{};
112 alloc.descriptorPool = descriptorPool_;
113 alloc.descriptorSetCount = 1;
114 alloc.pSetLayouts = &setLayout_;
115 auto sets = device->allocateDescriptorSets(alloc);
116 descriptorSet_ = sets[0];
117
118 // Dummy 4-byte SSBO for unused bindings (validation requires all layout slots valid).
119 if (!dummyBuffer_) {
120 using buf = vk::BufferUsageFlagBits;
121 using pfb = vk::MemoryPropertyFlagBits;
122 vkb::GenericBuffer tmp(device, buf::eStorageBuffer | buf::eTransferDst, 4, pfb::eDeviceLocal);
123 dummyBuffer_ = tmp.buffer;
124 dummyMemory_ = tmp.memory;
125#if defined(VKB_ENABLE_VMA)
126 dummyVmaAllocation_ = tmp.vma_allocation;
127#endif
128 // Without detach(), tmp's destructor frees the handles, leaving the
129 // dummy SSBO dangling: descriptor writes reference a destroyed buffer
130 // (crashes MoltenVK) and ~VulkanComputeShader double-frees it.
131 tmp.detach();
132 }
133
134 std::array<vk::DescriptorBufferInfo, kMaxBindings> infos{};
135 std::array<vk::WriteDescriptorSet, kMaxBindings> writes{};
136 for (int i = 0; i < kMaxBindings; ++i) {
137 auto *vb = dynamic_cast<VulkanGpuBuffer *>(bindings_[size_t(i)]);
138 infos[size_t(i)].buffer = (vb && vb->buffer_) ? vb->buffer_ : dummyBuffer_;
139 infos[size_t(i)].offset = 0;
140 infos[size_t(i)].range = (vb && vb->buffer_) ? vb->size_ : vk::DeviceSize(4);
141 writes[size_t(i)].dstSet = descriptorSet_;
142 writes[size_t(i)].dstBinding = uint32_t(i);
143 writes[size_t(i)].dstArrayElement = 0;
144 writes[size_t(i)].descriptorCount = 1;
145 writes[size_t(i)].descriptorType = vk::DescriptorType::eStorageBuffer;
146 writes[size_t(i)].pBufferInfo = &infos[size_t(i)];
147 }
148 device->updateDescriptorSets(kMaxBindings, writes.data(), 0, nullptr);
149
150 descriptorsDirty_ = false;
151}
