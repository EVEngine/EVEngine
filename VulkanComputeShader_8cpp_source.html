84void VulkanComputeShader::flushDescriptors(vkb::Device &device) {
85 if (!descriptorsDirty_ && descriptorSet_) return;
86
87 if (descriptorSet_) {
88 if (activeSequenceCount_ > 0) {
89 // Keep each immutable binding snapshot alive until all sequences
90 // that may reference it have retired.
91 pendingSets_.push_back({descriptorPool_, descriptorSet_});
92 } else {
93 device->freeDescriptorSets(descriptorPool_, descriptorSet_);
94 --allocatedSets_;
95 }
96 descriptorSet_ = vk::DescriptorSet{};
97 }
98
99 constexpr uint32_t setsPerPool = 64;
100 if (!descriptorPool_ || allocatedSets_ == setsPerPool) {
101 // Long recording batches can retain more than one pool of snapshots.
102 // Grow before exhausting maxSets; some drivers report fragmentation
103 // while others allow allocations beyond the declared pool capacity.
104 vk::DescriptorPoolSize poolSize{vk::DescriptorType::eStorageBuffer, uint32_t(kMaxBindings) * setsPerPool};
105 vk::DescriptorPoolCreateInfo poolInfo{};
106 poolInfo.flags = vk::DescriptorPoolCreateFlagBits::eFreeDescriptorSet;
107 poolInfo.maxSets = setsPerPool;
108 poolInfo.poolSizeCount = 1;
109 poolInfo.pPoolSizes = &poolSize;
110 const auto nextPool = device->createDescriptorPool(poolInfo, device.allocation_callbacks);
111 try {
112 if (descriptorPool_) retiredPools_.push_back(descriptorPool_);
113 } catch (...) {
114 device->destroyDescriptorPool(nextPool, device.allocation_callbacks);
115 throw;
116 }
117 descriptorPool_ = nextPool;
118 allocatedSets_ = 0;
119 }
120
121 vk::DescriptorSetAllocateInfo alloc{};
122 alloc.descriptorPool = descriptorPool_;
123 alloc.descriptorSetCount = 1;
124 alloc.pSetLayouts = &setLayout_;
125 auto sets = device->allocateDescriptorSets(alloc);
126 descriptorSet_ = sets[0];
127 ++allocatedSets_;
128
129 // Dummy 4-byte SSBO for unused bindings (validation requires all layout slots valid).
130 if (!dummyBuffer_) {
131 using buf = vk::BufferUsageFlagBits;
132 using pfb = vk::MemoryPropertyFlagBits;
133 vkb::GenericBuffer tmp(device, buf::eStorageBuffer | buf::eTransferDst, 4, pfb::eDeviceLocal);
134 dummyBuffer_ = tmp.buffer;
135 dummyMemory_ = tmp.memory;
136#if defined(VKB_ENABLE_VMA)
137 dummyVmaAllocation_ = tmp.vma_allocation;
138#endif
139 // Without detach(), tmp's destructor frees the handles, leaving the
140 // dummy SSBO dangling: descriptor writes reference a destroyed buffer
141 // (crashes MoltenVK) and ~VulkanComputeShader double-frees it.
142 tmp.detach();
143 }
144
145 std::array<vk::DescriptorBufferInfo, kMaxBindings> infos{};
146 std::array<vk::WriteDescriptorSet, kMaxBindings> writes{};
147 for (int i = 0; i < kMaxBindings; ++i) {
148 auto *vb = dynamic_cast<VulkanGpuBuffer *>(bindings_[size_t(i)]);
149 infos[size_t(i)].buffer = (vb && vb->buffer_) ? vb->buffer_ : dummyBuffer_;
150 infos[size_t(i)].offset = 0;
151 infos[size_t(i)].range = (vb && vb->buffer_) ? vb->size_ : vk::DeviceSize(4);
152 writes[size_t(i)].dstSet = descriptorSet_;
153 writes[size_t(i)].dstBinding = uint32_t(i);
154 writes[size_t(i)].dstArrayElement = 0;
155 writes[size_t(i)].descriptorCount = 1;
156 writes[size_t(i)].descriptorType = vk::DescriptorType::eStorageBuffer;
157 writes[size_t(i)].pBufferInfo = &infos[size_t(i)];
158 }
159 device->updateDescriptorSets(kMaxBindings, writes.data(), 0, nullptr);
160
161 descriptorsDirty_ = false;
162}

170void VulkanComputeShader::releasePendingDescriptors(vkb::Device &device) {
171 if (activeSequenceCount_ > 0) return;
172 if (pendingSets_.empty()) return;
173 for (const auto &lease : pendingSets_) {
174 if (lease.pool != descriptorPool_) continue;
175 device->freeDescriptorSets(lease.pool, lease.set);
176 --allocatedSets_;
177 }
178 pendingSets_.clear();
179 // Every set in a retired pool is now idle. Destroying the whole pool also
180 // releases its descriptors without retaining a high-water allocation.
181 for (vk::DescriptorPool pool : retiredPools_) device->destroyDescriptorPool(pool, device.allocation_callbacks);
182 retiredPools_.clear();
183}
