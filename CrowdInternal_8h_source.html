247 void forEachNeighbor(float qx, float qy, float radius, Fn&& fn, StepReport* partitionReport = nullptr) const {
248 const double r2 = double(radius) * radius;
249 StepReport& counters = partitionReport == nullptr ? workReport : *partitionReport;
250 const auto baseSpatial = partitionReport == nullptr ? 0 : workReport.spatialVisits;
251 const auto baseCandidates = partitionReport == nullptr ? 0 : workReport.candidatesVisited;
252 spatialIndex.visit(
253 qx, qy, radius,
254 budgetActive ? activeBudget.maxSpatialVisits - baseSpatial : std::numeric_limits<std::int64_t>::max(),
255 counters.spatialVisits, counters.cellsVisited, [&](int j) {
256 if (budgetActive && counters.candidatesVisited >= activeBudget.maxNeighborVisits - baseCandidates)
257 throw WorkLimit{"Crowd neighbor visit budget exhausted"};
258 ++counters.candidatesVisited;
259 const double dx = double(xs[size_t(j)]) - qx, dy = double(ys[size_t(j)]) - qy;
260 if (dx * dx + dy * dy <= r2) fn(j);
261 });
262 }
