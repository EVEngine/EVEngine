90void FluidSurfaceRenderer::render(const std::vector<glm::vec3>& positions, float particleRadius) {
91 positions_ = positions;
92 params_.particleRadius = particleRadius;
93 if (preferGpu_ && !gpuOk_) ensureGpu();
94 if (gpuOk_) {
95 // GPU path implemented below; the CPU path is exercised by tests.
96 const int pixels = params_.width * params_.height;
97 seq_->begin();
98 uploadParticles();
99 const int groupsPix = groupsFor(pixels);
100 setCommonConstants(shClear_, params_.depthFalloff);
101 seq_->recordDispatch(shClear_, groupsPix);
102 setCommonConstants(shSplat_, params_.depthFalloff);
103 seq_->recordDispatch(shSplat_, groupsFor(int(positions_.size())));
104 bool inA = true;
105 for (int it = 0; it < params_.smoothIterations; ++it) {
106 shSmooth_->bindBuffer(1, inA ? bufDepthA_ : bufDepthB_);
107 shSmooth_->bindBuffer(2, inA ? bufDepthB_ : bufDepthA_);
108 setCommonConstants(shSmooth_, params_.depthFalloff);
109 seq_->recordDispatch(shSmooth_, groupsPix);
110 inA = !inA;
111 }
112 gpgpu::GpuBuffer* finalDepth = inA ? bufDepthA_ : bufDepthB_;
113 shNormal_->bindBuffer(1, finalDepth);
114 setCommonConstants(shNormal_, params_.depthFalloff);
115 seq_->recordDispatch(shNormal_, groupsPix);
116 shShade_->bindBuffer(1, finalDepth);
117 setCommonConstants(shShade_, params_.depthFalloff);
118 seq_->recordDispatch(shShade_, groupsPix);
119 seq_->recordDownload(finalDepth, stDepth_, uint64_t(pixels) * sizeof(uint32_t));
120 seq_->recordDownload(bufThick_, stThick_, uint64_t(pixels) * sizeof(uint32_t));
121 seq_->recordDownload(bufNormal_, stNormal_, uint64_t(pixels) * 4u * sizeof(float));
122 seq_->recordDownload(bufColor_, stColor_, uint64_t(pixels) * 4u * sizeof(float));
123 seq_->submit();
124
125 std::vector<uint32_t> key(size_t(pixels), 0u);
126 std::vector<uint32_t> thickU(size_t(pixels), 0u);
127 std::vector<glm::vec4> normal4(size_t(pixels), glm::vec4(0.f));
128 std::vector<glm::vec4> color4(size_t(pixels), glm::vec4(0.f));
129 stDepth_->downloadBytes(key.data(), uint64_t(key.size()) * sizeof(uint32_t));
130 stThick_->downloadBytes(thickU.data(), uint64_t(thickU.size()) * sizeof(uint32_t));
131 stNormal_->downloadBytes(normal4.data(), uint64_t(normal4.size()) * sizeof(glm::vec4));
132 stColor_->downloadBytes(color4.data(), uint64_t(color4.size()) * sizeof(glm::vec4));
133 const float nearZ = params_.nearZ;
134 const float farZ = params_.farZ;
135 for (int i = 0; i < pixels; ++i) {
136 depth_[size_t(i)] =
137 key[size_t(i)] == kEmptyKey ? 1e30f : nearZ + (float(key[size_t(i)]) / kKeyScale) * (farZ - nearZ);
138 thickness_[size_t(i)] = float(thickU[size_t(i)]) / 256.f;
139 normals_[size_t(i)] = glm::vec3(normal4[size_t(i)]);
140 color_[size_t(i) * 4u + 0] = uint8_t(std::clamp(color4[size_t(i)].x, 0.f, 1.f) * 255.f);
141 color_[size_t(i) * 4u + 1] = uint8_t(std::clamp(color4[size_t(i)].y, 0.f, 1.f) * 255.f);
142 color_[size_t(i) * 4u + 2] = uint8_t(std::clamp(color4[size_t(i)].z, 0.f, 1.f) * 255.f);
143 color_[size_t(i) * 4u + 3] = uint8_t(std::clamp(color4[size_t(i)].w, 0.f, 1.f) * 255.f);
144 }
145 return;
146 }
147 renderCpu();
148}

150bool FluidSurfaceRenderer::ensureGpu() {
151 if (gpuOk_) return true;
152 gpgpu_ = eve::gpgpu::Gpgpu::create();
153 if (!gpgpu_ || !gpgpu_->isAvailable()) return false;
154 const int pixels = params_.width * params_.height;
155 const int maxP = 65536;
156 try {
157 shClear_ = gpgpu_->newShader(kSsfClear);
158 shSplat_ = gpgpu_->newShader(kSsfSplat);
159 shSmooth_ = gpgpu_->newShader(kSsfSmooth);
160 shNormal_ = gpgpu_->newShader(kSsfNormal);
161 shShade_ = gpgpu_->newShader(kSsfShade);
162 bufParts_ = gpgpu_->newBuffer(maxP * 4 * int(sizeof(float)), "storage");
163 bufDepthA_ = gpgpu_->newBuffer(pixels * int(sizeof(uint32_t)), "storage");
164 bufDepthB_ = gpgpu_->newBuffer(pixels * int(sizeof(uint32_t)), "storage");
165 bufThick_ = gpgpu_->newBuffer(pixels * int(sizeof(uint32_t)), "storage");
166 bufNormal_ = gpgpu_->newBuffer(pixels * 4 * int(sizeof(float)), "storage");
167 bufColor_ = gpgpu_->newBuffer(pixels * 4 * int(sizeof(float)), "storage");
168 stDepth_ = gpgpu_->newBuffer(pixels * int(sizeof(uint32_t)), "staging");
169 stThick_ = gpgpu_->newBuffer(pixels * int(sizeof(uint32_t)), "staging");
170 stNormal_ = gpgpu_->newBuffer(pixels * 4 * int(sizeof(float)), "staging");
171 stColor_ = gpgpu_->newBuffer(pixels * 4 * int(sizeof(float)), "staging");
172 seq_ = gpgpu_->newSequence();
173 } catch (...) {
174 return false;
175 }
176 if (!seq_ || !seq_->isAvailable()) return false;
177 shClear_->bindBuffer(1, bufDepthA_);
178 shClear_->bindBuffer(2, bufDepthB_);
179 shClear_->bindBuffer(3, bufThick_);
180 shClear_->bindBuffer(4, bufNormal_);
181 shClear_->bindBuffer(5, bufColor_);
182 shSplat_->bindBuffer(0, bufParts_);
183 shSplat_->bindBuffer(1, bufDepthA_);
184 shSplat_->bindBuffer(3, bufThick_);
185 shSmooth_->bindBuffer(1, bufDepthA_);
186 shSmooth_->bindBuffer(2, bufDepthB_);
187 shNormal_->bindBuffer(4, bufNormal_);
188 shShade_->bindBuffer(3, bufThick_);
189 shShade_->bindBuffer(4, bufNormal_);
190 shShade_->bindBuffer(5, bufColor_);
191 gpuOk_ = true;
192 return true;
193}

208void FluidSurfaceRenderer::setCommonConstants(gpgpu::ComputeShader* shader, float falloff) {
209 if (!shader) return;
210 const glm::mat4 view = glm::lookAtRH(params_.eye, params_.target, params_.up);
211 const glm::mat4 proj =
212 glm::perspectiveRH(glm::radians(params_.fovYDeg), params_.aspect, params_.nearZ, params_.farZ);
213 const glm::mat4 vp = proj * view;
214 const float* vpPtr = &vp[0][0];
215 for (int i = 0; i < 16; ++i) shader->setFloat(kSsfPushVP0 + i, vpPtr[i]);
216 shader->setFloat(kSsfPushCount, float(std::min(positions_.size(), size_t(65536))));
217 shader->setFloat(kSsfPushNear, params_.nearZ);
218 shader->setFloat(kSsfPushFar, params_.farZ);
219 shader->setFloat(kSsfPushTanHalf, std::tan(glm::radians(params_.fovYDeg) * 0.5f));
220 shader->setFloat(kSsfPushAspect, params_.aspect);
221 shader->setFloat(kSsfPushW, float(params_.width));
222 shader->setFloat(kSsfPushH, float(params_.height));
223 shader->setFloat(kSsfPushRadius, params_.particleRadius);
224 shader->setFloat(kSsfPushMode, float(params_.mode));
225 shader->setFloat(kSsfPushThick, params_.thicknessScale);
226 shader->setFloat(kSsfPushFalloff, falloff);
227}

229void FluidSurfaceRenderer::renderCpu() {
230 const int W = params_.width;
231 const int H = params_.height;
232 const int pixels = W * H;
233 const float nearZ = params_.nearZ;
234 const float farZ = params_.farZ;
235 const float tanHalf = std::tan(glm::radians(params_.fovYDeg) * 0.5f);
236 const glm::mat4 view = glm::lookAtRH(params_.eye, params_.target, params_.up);
237 const glm::mat4 proj = glm::perspectiveRH(glm::radians(params_.fovYDeg), params_.aspect, nearZ, farZ);
238 const glm::mat4 vp = proj * view;
239
240 std::vector<float> depth(size_t(pixels), 1e30f);
241 std::vector<float> thick(size_t(pixels), 0.f);
242
243 // 1. Splat.
244 for (const glm::vec3& p : positions_) {
245 const glm::vec4 clip = vp * glm::vec4(p, 1.f);
246 if (clip.w <= 1e-4f) continue;
247 const glm::vec3 ndc = glm::vec3(clip) / clip.w;
248 if (glm::any(glm::lessThan(ndc, glm::vec3(-1.f))) || glm::any(glm::greaterThan(ndc, glm::vec3(1.f)))) continue;
249 const float sx = (ndc.x * 0.5f + 0.5f) * float(W);
250 const float sy = (0.5f - ndc.y * 0.5f) * float(H);
251 const float depthVal = clip.w; // -viewZ
252 const float radiusPx = (params_.particleRadius * (float(H) * 0.5f) / tanHalf) / std::max(depthVal, 1e-4f);
253 if (radiusPx < 0.5f) continue;
254 const int x0 = std::max(int(std::floor(sx - radiusPx)), 0);
255 const int x1 = std::min(int(std::ceil(sx + radiusPx)), W - 1);
256 const int y0 = std::max(int(std::floor(sy - radiusPx)), 0);
257 const int y1 = std::min(int(std::ceil(sy + radiusPx)), H - 1);
258 const float r2 = radiusPx * radiusPx;
259 for (int yy = y0; yy <= y1; ++yy) {
260 for (int xx = x0; xx <= x1; ++xx) {
261 const float ddx = float(xx) + 0.5f - sx;
262 const float ddy = float(yy) + 0.5f - sy;
263 const float q = (ddx * ddx + ddy * ddy) / r2;
264 if (q >= 1.f) continue;
265 const size_t idx = size_t(yy) * size_t(W) + size_t(xx);
266 if (depthVal < depth[idx]) depth[idx] = depthVal;
267 const float w = 1.f - glm::smoothstep(0.4f, 1.f, std::sqrt(q));
268 thick[idx] += w * 2.f * params_.particleRadius * params_.thicknessScale;
269 }
270 }
271 }
272
273 // 2. Bilateral smooth (ping-pong).
274 std::vector<float> depthB = depth;
275 for (int it = 0; it < params_.smoothIterations; ++it) {
276 const std::vector<float>& src = (it % 2 == 0) ? depth : depthB;
277 std::vector<float>& dst = (it % 2 == 0) ? depthB : depth;
278 for (int y = 0; y < H; ++y) {
279 for (int x = 0; x < W; ++x) {
280 const size_t idx = size_t(y) * size_t(W) + size_t(x);
281 if (src[idx] >= 1e29f) {
282 dst[idx] = src[idx];
283 continue;
284 }
285 float sum = src[idx];
286 float wsum = 1.f;
287 for (int oy = -2; oy <= 2; ++oy) {
288 for (int ox = -2; ox <= 2; ++ox) {
289 if (ox == 0 && oy == 0) continue;
290 const int xx = x + ox;
291 const int yy = y + oy;
292 if (xx < 0 || yy < 0 || xx >= W || yy >= H) continue;
293 const size_t nidx = size_t(yy) * size_t(W) + size_t(xx);
294 if (src[nidx] >= 1e29f) continue;
295 const float spatial = std::exp(-float(ox * ox + oy * oy) * 0.5f);
296 const float wDepth = std::exp(-std::fabs(src[nidx] - src[idx]) / params_.depthFalloff);
297 const float w = spatial * wDepth;
298 sum += src[nidx] * w;
299 wsum += w;
300 }
301 }
302 dst[idx] = sum / std::max(wsum, 1e-5f);
303 }
304 }
305 }
306 const std::vector<float>& smooth = (params_.smoothIterations % 2 == 0) ? depth : depthB;
307
308 // 3. Normals from depth gradients (view space).
309 for (int y = 0; y < H; ++y) {
310 for (int x = 0; x < W; ++x) {
311 const size_t idx = size_t(y) * size_t(W) + size_t(x);
312 if (smooth[idx] >= 1e29f) {
313 normals_[idx] = glm::vec3(0.f);
314 continue;
315 }
316 const auto viewPos = [&](int px, int py, float d) {
317 const float u = (float(px) + 0.5f) / float(W);
318 const float v = (float(py) + 0.5f) / float(H);
319 return glm::vec3((u * 2.f - 1.f) * params_.aspect * tanHalf * d, (v * 2.f - 1.f) * tanHalf * d, -d);
320 };
321 const auto sd = [&](int ox, int oy) {
322 const int xx = std::clamp(x + ox, 0, W - 1);
323 const int yy = std::clamp(y + oy, 0, H - 1);
324 const float d = smooth[size_t(yy) * size_t(W) + size_t(xx)];
325 return d < 1e29f ? d : smooth[idx];
326 };
327 const glm::vec3 dpx = (viewPos(x + 1, y, sd(1, 0)) - viewPos(x - 1, y, sd(-1, 0))) * 0.5f;
328 const glm::vec3 dpy = (viewPos(x, y + 1, sd(0, 1)) - viewPos(x, y - 1, sd(0, -1))) * 0.5f;
329 glm::vec3 n = glm::normalize(glm::cross(dpx, dpy));
330 if (n.z < 0.f) n = -n;
331 normals_[idx] = n;
332 }
333 }
334
335 // 4. Shade.
336 const glm::vec3 L(0.35f, 0.65f, 0.55f);
337 const glm::vec3 V(0.f, 0.f, 1.f);
338 for (int y = 0; y < H; ++y) {
339 for (int x = 0; x < W; ++x) {
340 const size_t idx = size_t(y) * size_t(W) + size_t(x);
341 if (smooth[idx] >= 1e29f) {
342 color_[idx * 4u + 0] = 0;
343 color_[idx * 4u + 1] = 0;
344 color_[idx * 4u + 2] = 0;
345 color_[idx * 4u + 3] = 0;
346 continue;
347 }
348 const glm::vec3 n = normals_[idx];
349 const float diff = std::max(glm::dot(n, L), 0.f);
350 glm::vec3 outC;
351 float alpha;
352 if (params_.mode == 1) {
353 const glm::vec3 base = glm::vec3(0.36f, 0.23f, 0.12f) * (0.45f + 0.55f * diff);
354 const float attenuation = std::exp(-thick[idx] * 1.8f);
355 const glm::vec3 hv = glm::normalize(L + V);
356 const float spec = std::pow(std::max(glm::dot(n, hv), 0.f), 8.f) * 0.12f;
357 outC = base * attenuation + glm::vec3(spec);
358 alpha = std::clamp(thick[idx] * 0.6f, 0.f, 1.f);
359 } else {
360 const glm::vec3 base = glm::vec3(0.05f, 0.32f, 0.72f) * (0.55f + 0.45f * diff);
361 const float fresnel = 0.04f + 0.96f * std::pow(1.f - std::max(glm::dot(n, V), 0.f), 5.f);
362 const glm::vec3 hv = glm::normalize(L + V);
363 const float spec = std::pow(std::max(glm::dot(n, hv), 0.f), 64.f) * 0.45f;
364 outC = base + glm::vec3(0.55f, 0.72f, 1.f) * fresnel * 0.75f + glm::vec3(spec);
365 alpha = std::clamp(thick[idx] * 0.35f, 0.f, 1.f);
366 }
367 color_[idx * 4u + 0] = uint8_t(std::clamp(outC.x, 0.f, 1.f) * 255.f);
368 color_[idx * 4u + 1] = uint8_t(std::clamp(outC.y, 0.f, 1.f) * 255.f);
369 color_[idx * 4u + 2] = uint8_t(std::clamp(outC.z, 0.f, 1.f) * 255.f);
370 color_[idx * 4u + 3] = uint8_t(std::clamp(alpha, 0.f, 1.f) * 255.f);
371 }
372 }
373 depth_ = std::move(depth);
374 thickness_ = std::move(thick);
375}
