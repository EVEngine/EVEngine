79Result<std::unique_ptr<OceanGpuSpectrum>> OceanGpuSpectrumBuilder::build() const {
80 using Output = Result<std::unique_ptr<OceanGpuSpectrum>>;
81 if (graphics_.resourceLifetime().expired())
82 return Output::failure(diagnostic(DiagnosticCode::StaleHandle, "Graphics provider retired"));
83 if (graphics_.getBackendName() != "vulkan" || !compute_.isAvailable())
84 return Output::failure(
85 diagnostic(DiagnosticCode::Unsupported, "Ocean FFT requires an available Vulkan compute provider"));
86 std::unique_ptr<OceanGpuSpectrum> result(new OceanGpuSpectrum);
87 auto& state = *result->impl_;
88 state.data = data_;
89 state.target = target_;
90 state.start = start_;
91 state.duration = duration_;
92 state.lifetime = graphics_.resourceLifetime();
93 auto retiring = graphics_.onResourcesRetiring([target = &state]() { target->release(); });
94 if (!retiring) return Output::failure(retiring.status());
95 state.retirement = std::move(retiring.value());
96 try {
97 auto evolve = gpgpu::createComputeShader(
98 std::vector<uint32_t>(ocean_evolve_spv, ocean_evolve_spv + ocean_evolve_spv_count));
99 if (!evolve) return Output::failure(evolve.status());
100 state.evolve = std::move(evolve.value());
101 auto fft =
102 gpgpu::createComputeShader(std::vector<uint32_t>(ocean_fft_spv, ocean_fft_spv + ocean_fft_spv_count));
103 if (!fft) return Output::failure(fft.status());
104 state.fft = std::move(fft.value());
105 auto pack =
106 gpgpu::createComputeShader(std::vector<uint32_t>(ocean_pack_spv, ocean_pack_spv + ocean_pack_spv_count));
107 if (!pack) return Output::failure(pack.status());
108 state.pack = std::move(pack.value());
109 state.sequence.reset(compute_.newSequence());
110 if (!state.sequence || !state.sequence->isAvailable())
111 return Output::failure(diagnostic(DiagnosticCode::Unsupported, "Ocean GPU command sequence unavailable"));
112 for (const auto& source : data_->bands) {
113 OceanGpuSpectrum::Impl::Band band;
114 const auto count = source.settings.resolution * source.settings.resolution;
115 band.coefficients.reset(
116 compute_.newBuffer(static_cast<int>(count * 8 * (target_ ? 2 : 1) * sizeof(float))));
117 band.ping.reset(compute_.newBuffer(static_cast<int>(count * 12 * sizeof(float))));
118 band.pong.reset(compute_.newBuffer(static_cast<int>(count * 12 * sizeof(float))));
119 band.packed.reset(compute_.newBuffer(static_cast<int>(count * 6 * sizeof(float))));
120 if (!band.coefficients || !band.ping || !band.pong || !band.packed)
121 return Output::failure(diagnostic(DiagnosticCode::Unsupported, "Ocean GPU allocation unavailable"));
122 state.bands.push_back(std::move(band));
123 auto& layout = state.atlasLayout;
124 layout.width = std::max(layout.width, source.settings.resolution);
125 layout.bands[layout.bandCount++] = {source.settings.resolution, layout.height, source.settings.length};
126 layout.height += source.settings.resolution * 3;
127 }
128 state.atlasLayout.seaLevel = data_->settings.seaLevel;
129 state.atlas.reset(
130 compute_.newBuffer(static_cast<int>(state.atlasLayout.width * state.atlasLayout.height * sizeof(float))));
131 if (!state.atlas)
132 return Output::failure(diagnostic(DiagnosticCode::Unsupported, "Ocean GPU atlas unavailable"));
133 } catch (const std::exception& error) {
134 return Output::failure(diagnostic(DiagnosticCode::Failed, error.what()));
135 }
136 return Output::success(std::move(result));
137}

143Result<OceanGpuFrameToken> OceanGpuFrameBuilder::submit() const {
144 using Output = Result<OceanGpuFrameToken>;
145 auto& state = *owner_.impl_;
146 if (state.lifetime.expired() || !state.sequence)
147 return Output::failure(diagnostic(DiagnosticCode::StaleHandle, "Ocean GPU provider retired"));
148 if (state.status == OceanGpuSpectrum::Impl::State::Failed)
149 return Output::failure(
150 diagnostic(DiagnosticCode::Failed, "Rebuild the ocean GPU projection after a backend failure"));
151 if (!std::isfinite(time_) || std::abs(time_) > 1e9)
152 return Output::failure(diagnostic(DiagnosticCode::InvalidArgument, "Invalid ocean GPU time"));
153 ++state.revision;
154 state.status = OceanGpuSpectrum::Impl::State::Failed;
155 // Rebase phase in double once per 16 simulation seconds, around the nearest
156 // anchor. A signed relative time in [-8, 8) halves the worst float phase
157 // error while retaining the same coefficient upload frequency.
158 const double anchor = std::floor((time_ + 8.0) / 16.0) * 16.0;
159 const double u = state.target ? std::clamp((time_ - state.start) / state.duration, 0.0, 1.0) : 0.0;
160 const double weight = u * u * u * (10.0 + u * (-15.0 + 6.0 * u));
161 const double rate = state.target ? 30.0 * u * u * (1.0 - u) * (1.0 - u) / state.duration : 0.0;
162 const double levelDelta = state.target ? state.target->settings.seaLevel - state.data->settings.seaLevel : 0.0;
163 try {
164 auto& sequence = *state.sequence;
165 sequence.begin();
166 for (std::size_t bandIndex = 0; bandIndex < state.bands.size(); ++bandIndex) {
167 const auto& source = state.data->bands[bandIndex];
168 auto& band = state.bands[bandIndex];
169 const unsigned n = source.settings.resolution, count = n * n;
170 const int groups = static_cast<int>((count * 6 + 63) / 64);
171 if (anchor != state.anchor) {
172 std::vector<float> coefficients(count * 8 * (state.target ? 2 : 1));
173 for (unsigned z = 0; z < n; ++z)
174 for (unsigned x = 0; x < n; ++x) {
175 const int ix = x < n / 2 ? int(x) : int(x) - int(n);
176 const int iz = z < n / 2 ? int(z) : int(z) - int(n);
177 const glm::dvec2 k = glm::dvec2(ix, iz) * (tau / source.settings.length);
178 const double omega = std::sqrt(state.data->settings.gravity * glm::length(k));
179 const double advection = glm::dot(k, state.data->settings.current);
180 const unsigned i = z * n + x, opposite = ((n - z) % n) * n + (n - x) % n;
181 // Match CPU phase reduction before multiplication, including very large times.
182 const auto wind = std::polar(1.0, std::remainder(-omega * anchor, tau));
183 const auto flow = std::polar(1.0, std::remainder(-advection * anchor, tau));
184 const auto a = source.coefficients[i] * wind * flow;
185 const auto b = std::conj(source.coefficients[opposite]) * std::conj(wind) * flow;
186 const std::array<double, 8> values{a.real(), a.imag(), b.real(), b.imag(),
187 k.x, k.y, omega, advection};
188 for (unsigned component = 0; component < 8; ++component)
189 coefficients[i * 8 + component] = static_cast<float>(values[component]);
190 if (state.target) {
191 const auto& target = state.target->bands[bandIndex];
192 const auto ta = target.coefficients[i] * wind * flow;
193 const auto tb = std::conj(target.coefficients[opposite]) * std::conj(wind) * flow;
194 const std::array<double, 8> next{ta.real(), ta.imag(), tb.real(), tb.imag(),
195 k.x, k.y, omega, advection};
196 for (unsigned component = 0; component < 8; ++component)
197 coefficients[(count + i) * 8 + component] = static_cast<float>(next[component]);
198 }
199 }
200 sequence.recordUpload(band.coefficients.get(), coefficients.data(),
201 coefficients.size() * sizeof(float));
202 }
203 state.evolve->bindBuffer(0, band.coefficients.get());
204 state.evolve->bindBuffer(1, band.ping.get());
205 state.evolve->setFloat(0, static_cast<float>(n));
206 state.evolve->setFloat(1, static_cast<float>(time_ - anchor));
207 state.evolve->setFloat(2, static_cast<float>(state.data->settings.choppiness));
208 state.evolve->setFloat(3, state.target ? 1.0F : 0.0F);
209 state.evolve->setFloat(4, static_cast<float>(weight));
210 state.evolve->setFloat(5, static_cast<float>(rate));
211 state.evolve->setFloat(6, state.target ? static_cast<float>(state.target->settings.choppiness) : 0.0F);
212 state.evolve->setFloat(7, bandIndex == 0 ? static_cast<float>(levelDelta * rate) : 0.0F);
213 sequence.recordDispatch(state.evolve.get(), groups);
214 auto* sourceBuffer = band.ping.get();
215 auto* destination = band.pong.get();
216 for (unsigned axis = 0; axis < 2; ++axis)
217 for (unsigned width = 2; width <= n; width *= 2) {
218 state.fft->bindBuffer(0, sourceBuffer);
219 state.fft->bindBuffer(1, destination);
220 state.fft->setFloat(0, static_cast<float>(n));
221 state.fft->setFloat(1, static_cast<float>(width));
222 state.fft->setFloat(2, static_cast<float>(axis));
223 sequence.recordDispatch(state.fft.get(), groups);
224 std::swap(sourceBuffer, destination);
225 }
226 state.pack->bindBuffer(0, sourceBuffer);
227 state.pack->bindBuffer(1, band.packed.get());
228 state.pack->bindBuffer(2, state.atlas.get());
229 state.pack->setFloat(0, static_cast<float>(count * 6));
230 state.pack->setFloat(1, static_cast<float>(n));
231 state.pack->setFloat(2, static_cast<float>(state.atlasLayout.width));
232 state.pack->setFloat(3, static_cast<float>(state.atlasLayout.bands[bandIndex].firstRow));
233 sequence.recordDispatch(state.pack.get(), groups);
234 }
235 sequence.submit();
236 if (sequence.getStatus() != gpgpu::SequenceStatus::Complete)
237 return Output::failure(diagnostic(DiagnosticCode::Failed, "Ocean GPU submission did not complete"));
238 } catch (const std::exception& error) {
239 state.anchor = std::numeric_limits<double>::quiet_NaN();
240 return Output::failure(diagnostic(DiagnosticCode::Failed, error.what()));
241 }
242 state.atlasLayout.seaLevel = state.data->settings.seaLevel + weight * levelDelta;
243 state.anchor = anchor;
244 state.status = OceanGpuSpectrum::Impl::State::Ready;
245 OceanGpuFrameToken token;
246 token.owner_ = state.owner;
247 token.revision_ = state.revision;
248 token.time_ = time_;
249 return Output::success(token);
250}

270Result<OceanSpectrumFrame> OceanGpuReadbackBuilder::build() const {
271 const auto& state = *owner_.impl_;
272 auto valid = owner_.band(token_, 0);
273 if (!valid) return Result<OceanSpectrumFrame>::failure(valid.status());
274 OceanSpectrumFrame result;
275 result.time_ = token_.time_;
276 result.seaLevel_ = state.atlasLayout.seaLevel;
277 result.current_ = state.data->settings.current;
278 try {
279 for (std::size_t b = 0; b < state.bands.size(); ++b) {
280 const auto& settings = state.data->bands[b].settings;
281 const auto count = settings.resolution * settings.resolution;
282 std::vector<float> values(count * 6);
283 state.bands[b].packed->readFloat32s(values.data(), static_cast<int>(values.size()));
284 if (!std::all_of(values.begin(), values.end(), [](float value) { return std::isfinite(value); }))
285 return Result<OceanSpectrumFrame>::failure(
286 diagnostic(DiagnosticCode::Failed, "Ocean GPU output contains nonfinite values"));
287 OceanSpectrumGrid grid;
288 grid.resolution = settings.resolution;
289 grid.length = settings.length;
290 grid.displacement.resize(count);
291 grid.velocity.resize(count);
292 for (unsigned i = 0; i < count; ++i) {
293 grid.displacement[i] = {values[i], values[count + i], values[2 * count + i]};
294 grid.velocity[i] = {values[3 * count + i], values[4 * count + i], values[5 * count + i]};
295 }
296 result.grids_.push_back(std::move(grid));
297 }
298 } catch (const std::exception& error) {
299 return Result<OceanSpectrumFrame>::failure(diagnostic(DiagnosticCode::Failed, error.what()));
300 }
301 return Result<OceanSpectrumFrame>::success(std::move(result));
302}
