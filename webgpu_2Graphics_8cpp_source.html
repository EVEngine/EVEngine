94void Graphics::initWithWindow(void *nativeWindow) {
95 sdlWindow = nativeWindow;
96 if (deviceInitDone) return;
97
98 createInstanceAndAdapter();
99 requestDevice();
100 if (!device) throw Exception("WebGPU: device request failed");
101
102 queue = device.GetQueue();
103
104 // ---- Surface ----
105 WGPUSurfaceDescriptor surfDesc{};
106 surfDesc.label = sv("eve_surface");
107#if defined(__EMSCRIPTEN__)
108 WGPUEmscriptenSurfaceSourceCanvasHTMLSelector canvasSel{};
109 canvasSel.chain.sType = WGPUSType_EmscriptenSurfaceSourceCanvasHTMLSelector;
110 canvasSel.selector = sv("#canvas");
111 surfDesc.nextInChain = &canvasSel.chain;
112 surface = instance.CreateSurface(reinterpret_cast<const wgpu::SurfaceDescriptor*>(&surfDesc));
113 surfaceFormat = WGPUTextureFormat_BGRA8Unorm;
114#else
115 SDL_SysWMinfo wminfo;
116 SDL_VERSION(&wminfo.version);
117 if (!SDL_GetWindowWMInfo(static_cast<SDL_Window *>(sdlWindow), &wminfo))
118 throw Exception("WebGPU: SDL_GetWindowWMInfo failed: %s", SDL_GetError());
119#if defined(_WIN32)
120 WGPUSurfaceSourceWindowsHWND winChain{};
121 winChain.chain.sType = WGPUSType_SurfaceSourceWindowsHWND;
122 winChain.hwnd = wminfo.info.win.window;
123 winChain.hinstance = GetModuleHandle(nullptr);
124 surfDesc.nextInChain = &winChain.chain;
125 surface = instance.CreateSurface(reinterpret_cast<const wgpu::SurfaceDescriptor*>(&surfDesc));
126#elif defined(__linux__)
127 if (wminfo.subsystem == SDL_SYSWM_X11) {
128 WGPUSurfaceSourceXlibWindow x11Chain{};
129 x11Chain.chain.sType = WGPUSType_SurfaceSourceXlibWindow;
130 x11Chain.display = wminfo.info.x11.display;
131 x11Chain.window = wminfo.info.x11.window;
132 surfDesc.nextInChain = &x11Chain.chain;
133 surface = instance.CreateSurface(reinterpret_cast<const wgpu::SurfaceDescriptor*>(&surfDesc));
134 } else if (wminfo.subsystem == SDL_SYSWM_WAYLAND) {
135 WGPUSurfaceSourceWaylandSurface wlChain{};
136 wlChain.chain.sType = WGPUSType_SurfaceSourceWaylandSurface;
137 wlChain.display = wminfo.info.wl.display;
138 wlChain.surface = wminfo.info.wl.surface;
139 surfDesc.nextInChain = &wlChain.chain;
140 surface = instance.CreateSurface(reinterpret_cast<const wgpu::SurfaceDescriptor*>(&surfDesc));
141 } else {
142 throw Exception("WebGPU: unsupported SDL window subsystem on Linux");
143 }
144#elif defined(__APPLE__)
145 throw Exception("WebGPU: native macOS surface (Metal layer) not yet supported; "
146 "use the Emscripten browser build or the Vulkan backend on macOS");
147#else
148 throw Exception("WebGPU: unsupported native platform for surface creation");
149#endif
150 if (!surface) throw Exception("WebGPU: surface creation failed");
151 // Preferred surface format comes from wgpuSurfaceGetCapabilities in the
152 // current ABI (Surface::GetPreferredFormat was removed).
153 WGPUSurfaceCapabilities caps{};
154 if (wgpuSurfaceGetCapabilities(surface.Get(), adapter.Get(), &caps) == WGPUStatus_Success &&
155 caps.formatCount > 0) {
156 surfaceFormat = caps.formats[0];
157 }
158 wgpuSurfaceCapabilitiesFreeMembers(&caps);
159#endif
160
161 swapchainConfigured = false;
162 // Bind group layouts / pipelines must exist before the default textures
163 // (texture bind groups reference the 2D / mesh3d layouts), and the shadow
164 // depth array must exist before mesh bind groups are built at flush time.
165 createPipelineResources();
166 createShadowResources();
167 createDefaultTextures();
168 initialized = true;
169}

171void Graphics::createInstanceAndAdapter() {
172 instance = wgpu::CreateInstance();
173 if (!instance) throw Exception("WebGPU: wgpuCreateInstance failed");
174
175 WGPURequestAdapterOptions opts{};
176 opts.compatibleSurface = surface.Get();
177 opts.powerPreference = WGPUPowerPreference_HighPerformance;
178
179 adapterReceived = false;
180 WGPURequestAdapterCallbackInfo cbInfo{};
181 cbInfo.nextInChain = nullptr;
182 cbInfo.mode = WGPUCallbackMode_AllowProcessEvents;
183 cbInfo.callback = [](WGPURequestAdapterStatus status, WGPUAdapter a, WGPUStringView msg,
184 void *userdata1, void * /*userdata2*/) {
185 auto *self = static_cast<Graphics *>(userdata1);
186 if (status == WGPURequestAdapterStatus_Success && a) {
187 self->adapter = wgpu::Adapter(a);
188 } else {
189 self->adapterError =
190 msg.data ? std::string(msg.data, msg.length) : "unknown adapter error";
191 }
192 self->adapterReceived.store(true);
193 };
194 cbInfo.userdata1 = this;
195 cbInfo.userdata2 = nullptr;
196
197 wgpuInstanceRequestAdapter(instance.Get(), &opts, cbInfo);
198 waitForAdapter();
199 if (!adapter) {
200 throw Exception("WebGPU: no adapter found (%s)", adapterError.c_str());
201 }
202}

204void Graphics::requestDevice() {
205 WGPUDeviceDescriptor devDesc{};
206 devDesc.label = sv("eve_device");
207
208 deviceReceived = false;
209 WGPURequestDeviceCallbackInfo cbInfo{};
210 cbInfo.nextInChain = nullptr;
211 cbInfo.mode = WGPUCallbackMode_AllowProcessEvents;
212 cbInfo.callback = [](WGPURequestDeviceStatus status, WGPUDevice d, WGPUStringView msg,
213 void *userdata1, void * /*userdata2*/) {
214 auto *self = static_cast<Graphics *>(userdata1);
215 if (status == WGPURequestDeviceStatus_Success && d) {
216 self->device = wgpu::Device(d);
217 // Default device limits support 8+ uniform/vertex/storage buffers.
218 self->deviceReceived.store(true);
219 self->deviceError.clear();
220 } else {
221 self->deviceError =
222 msg.data ? std::string(msg.data, msg.length) : "unknown device error";
223 self->deviceReceived.store(true);
224 }
225 };
226 cbInfo.userdata1 = this;
227 cbInfo.userdata2 = nullptr;
228
229 wgpuAdapterRequestDevice(adapter.Get(), &devDesc, cbInfo);
230 waitForDevice();
231 if (!device) {
232 throw Exception("WebGPU: device request failed (%s)", deviceError.c_str());
233 }

278void Graphics::configureSurface(int width, int height) {
279 if (!surface || !device || width <= 0 || height <= 0) return;
280 WGPUSurfaceConfiguration cfg{};
281 cfg.nextInChain = nullptr;
282 cfg.device = device.Get();
283 cfg.width = static_cast<uint32_t>(width);
284 cfg.height = static_cast<uint32_t>(height);
285 cfg.format = surfaceFormat;
286 cfg.usage = WGPUTextureUsage_RenderAttachment;
287 cfg.viewFormatCount = 0;
288 cfg.viewFormats = nullptr;
289 // emdawnwebgpu only accepts Fifo / Undefined present modes.
290 cfg.presentMode = WGPUPresentMode_Fifo;
291 // Auto (0) maps to an empty slot in emdawnwebgpu's JS CompositeAlphaMode
292 // table �?`alphaMode: undefined`, which some browsers reject. Use Opaque,
293 // which maps to the JS value 'opaque' used by the working replica.
294 cfg.alphaMode = WGPUCompositeAlphaMode_Opaque;
295 surface.Configure(reinterpret_cast<const wgpu::SurfaceConfiguration*>(&cfg));
296 swapchainConfigured = true;
297 // Recreate the offscreen targets at the new size.
298 int w = pixelW > 0 ? pixelW : width;
299 int h = pixelH > 0 ? pixelH : height;
300 if (w > 0 && h > 0) {
301 createSceneColorResources(w, h);
302 createShadowResources();
303 }
304}

734// Returns a RAII-held module: the caller must keep it alive across
735// CreateRenderPipeline. Returning a raw WGPUShaderModule would let the
736// temporary wgpu::ShaderModule destructor Release() the handle, dropping it

739 WGPUShaderSourceWGSL wgslDesc{};

743 desc.nextInChain = &wgslDesc.chain;
744 return dev.CreateShaderModule(reinterpret_cast<const wgpu::ShaderModuleDescriptor*>(&desc));
745}

748// static so its chain pointer stays valid). The returned descriptor is a value

751 static WGPUShaderSourceWGSL wd{};

753 wd.code = sv(code.c_str());

755 md.nextInChain = &wd.chain;

804wgpu::RenderPipeline make2DColorPipeline(wgpu::Device &dev, WGPUTextureFormat format,
805 BlendMode mode) {
806 WGPUVertexAttribute attrs[2] = {};
807 attrs[0].format = WGPUVertexFormat_Float32x2;
808 attrs[0].offset = 0;
809 attrs[0].shaderLocation = 0;
810 attrs[1].format = WGPUVertexFormat_Float32x4;
811 attrs[1].offset = 8;
812 attrs[1].shaderLocation = 1;
813 WGPUVertexBufferLayout vb{};
814 fillVertexLayout(vb, 24, attrs, 2);
815
816 WGPUColorTargetState target{};
817 target.format = format;
818 target.blend = nullptr;
819 // Zero-init yields WGPUColorWriteMask_None, which silently discards every
820 // fragment (the clear still shows because clearValue is unaffected by the
821 // mask). emdawnwebgpu forwards this explicit 0 to the browser, unlike the
822 // JS default of "all".
823 target.writeMask = WGPUColorWriteMask_All;
824 WGPUBlendState bs = alphaBlend();
825 if (mode == BlendMode::Additive)
826 bs = additiveBlend();
827 else if (mode == BlendMode::Opaque)
828 bs = noBlend();
829 if (mode != BlendMode::Opaque) target.blend = &bs;
830
831 // The solid-color shader declares no bindings, so it must use an empty
832 // pipeline layout. Reusing the textured 2D layout here would require a
833 // bind group for every solid draw (WebGPU validation fails the whole
834 // command buffer otherwise).
835 WGPUPipelineLayoutDescriptor pld{};
836 pld.label = sv("eve_2d_color_layout");
837 pld.bindGroupLayoutCount = 0;
838 pld.bindGroupLayouts = nullptr;
839 wgpu::PipelineLayout emptyLayout = dev.CreatePipelineLayout(reinterpret_cast<const wgpu::PipelineLayoutDescriptor*>(&pld));
840
841 WGPURenderPipelineDescriptor pd{};
842 pd.label = sv("eve_color2d");
843 // nullptr = auto (default) pipeline layout; the solid-color shader has no
844 // bindings so the derived layout matches.
845 pd.layout = nullptr;
846 wgpu::ShaderModule vertModule = makeWgslModule(dev, kColorVertWgsl);
847 wgpu::ShaderModule fragModule = makeWgslModule(dev, kColorFragWgsl);
848 pd.vertex.module = vertModule.Get();
849 pd.vertex.entryPoint = sv("vs_main");
850 pd.vertex.bufferCount = 1;
851 pd.vertex.buffers = &vb;
852 WGPUFragmentState fs{};
853 fs.module = fragModule.Get();
854 fs.entryPoint = sv("fs_main");
855 fs.targetCount = 1;
856 fs.targets = &target;
857 pd.fragment = &fs;
858 pd.primitive.topology = WGPUPrimitiveTopology_TriangleList;
859 pd.primitive.frontFace = WGPUFrontFace_CCW;
860 pd.primitive.cullMode = WGPUCullMode_None;
861 pd.primitive.stripIndexFormat = WGPUIndexFormat_Undefined;
862 pd.depthStencil = nullptr;
863 pd.multisample.count = 1;
864 // Zero-init would leave mask=0, which discards every fragment
865 // (sampleMask=0). The WebGPU default is 0xFFFFFFFF (all samples).
866 pd.multisample.mask = 0xFFFFFFFFu;
867 return dev.CreateRenderPipeline(reinterpret_cast<const wgpu::RenderPipelineDescriptor*>(&pd));
868}

1530 s.mipmap == MipmapMode::Nearest ? WGPUMipmapFilterMode_Nearest : WGPUMipmapFilterMode_Linear;
1531 d.lodMinClamp = s.minLod;
1532 d.lodMaxClamp = std::min(std::max(s.maxLod, 0.f), float(mipLevels));
1533 }
1534 float aniso = s.maxAnisotropy > 1.f ? s.maxAnisotropy : 1.f;
1535 d.maxAnisotropy = std::min(std::max(aniso, 1.f), std::max(maxSamplerAnisotropy, 1.f));

1633 static_cast<uint64_t>(w) * h * 4,
1634 reinterpret_cast<const wgpu::TexelCopyBufferLayout*>(&layout),
1635 reinterpret_cast<const wgpu::Extent3D*>(&extent));
1636 if (m + 1 < gt->mipLevels) {
1637 // Box-filter downsample into the CPU buffer for the next level.
1638 std::vector<uint8_t> next((w / 2) * (h / 2) * 4);
1639 for (int y = 0; y < h / 2; ++y) {
1640 for (int x = 0; x < w / 2; ++x) {
1641 for (int c = 0; c < 4; ++c) {
1642 uint32_t acc = 0;
1643 acc += rgba[((y * 2 + 0) * w + (x * 2 + 0)) * 4 + c];
1644 acc += rgba[((y * 2 + 0) * w + (x * 2 + 1)) * 4 + c];
1645 acc += rgba[((y * 2 + 1) * w + (x * 2 + 0)) * 4 + c];
1646 acc += rgba[((y * 2 + 1) * w + (x * 2 + 1)) * 4 + c];
1647 next[(y * (w / 2) + x) * 4 + c] = uint8_t(acc / 4);
1648 }
1649 }
1650 }
1651 w /= 2;
1652 h /= 2;
1653 rgba = next.data();
1654 layout.bytesPerRow = static_cast<uint32_t>(w * 4);
1655 layout.rowsPerImage = static_cast<uint32_t>(h);
1656 extent = {static_cast<uint32_t>(w), static_cast<uint32_t>(h), 1};
1657 }
1658 }

1778bool Graphics::releaseTexture(Texture *texture) {
1779 if (!texture || !texture->gpuHandle) return false;
1780 // Renderer-owned fallback textures must never be released by callers.
1781 if (texture->gpuHandle == whiteTexture || texture->gpuHandle == flatNormalTexture ||
1782 texture->gpuHandle == flatNormalTexture3D ||
1783 texture->gpuHandle == defaultEnvCubemap)
1784 return false;
1785
1786 auto *gpu = static_cast<GpuTexture *>(texture->gpuHandle);
1787 auto gpuIt = std::find_if(ownedGpuTextures.begin(), ownedGpuTextures.end(),
1788 [&](const std::unique_ptr<GpuTexture> &g) {
1789 return g.get() == gpu;
1790 });
1791 if (gpuIt == ownedGpuTextures.end()) return false;
1792
1793 auto texIt = std::find_if(ownedTextures.begin(), ownedTextures.end(),
1794 [&](const std::unique_ptr<Texture> &t) {
1795 return t.get() == texture;
1796 });
1797 if (texIt == ownedTextures.end()) return false;
1798
1799 // Path-cached textures must leave the hot-reload cache once released.
1800 for (auto it = texturesByPath.begin(); it != texturesByPath.end();) {
1801 if (it->second == texture)
1802 it = texturesByPath.erase(it);
1803 else
1804 ++it;
1805 }
1806
1807 texture->gpuHandle = nullptr;
1808 ownedGpuTextures.erase(gpuIt);
1809 // Transfer the CPU facade to the caller instead of destroying it.
1810 (void)texIt->release();
1811 ownedTextures.erase(texIt);
1812 return true;
1813}

1980 int vertexCount, const uint32_t *indices, int indexCount) {
1981 if (vertexCount <= 0 || !posXYZ) throw Exception("newMeshFromArrays: invalid vertex data");
1982
1983 std::vector<float> verts;
1984 verts.reserve(vertexCount * 8);
1985 for (int i = 0; i < vertexCount; ++i) {
1986 verts.push_back(posXYZ[i * 3 + 0]);
1987 verts.push_back(posXYZ[i * 3 + 1]);
1988 verts.push_back(posXYZ[i * 3 + 2]);
1989 if (nrmXYZ) {
1990 verts.push_back(nrmXYZ[i * 3 + 0]);
1991 verts.push_back(nrmXYZ[i * 3 + 1]);
1992 verts.push_back(nrmXYZ[i * 3 + 2]);
1993 } else {
1994 verts.push_back(0.f);
1995 verts.push_back(0.f);
1996 verts.push_back(1.f);
1997 }
1998 if (uvST) {
1999 verts.push_back(uvST[i * 2 + 0]);
2000 verts.push_back(uvST[i * 2 + 1]);
2001 } else {
2002 verts.push_back(0.f);
2003 verts.push_back(0.f);
2004 }
2005 }
2006
2007 auto gpu = std::make_unique<GpuMesh>();
2008 gpu->vertexCount = uint32_t(vertexCount);
2009 gpu->indexCount = indexCount > 0 ? uint32_t(indexCount) : 0;
2010 gpu->vertexStride = 32;
2011
2012 WGPUBufferDescriptor vbd{};
2013 vbd.label = sv("eve_mesh_vb");
2014 vbd.size = verts.size() * sizeof(float);
2015 vbd.usage = WGPUBufferUsage_CopyDst | WGPUBufferUsage_Vertex;
2016 vbd.mappedAtCreation = false;
2017 gpu->vertexBuffer = device.CreateBuffer(reinterpret_cast<const wgpu::BufferDescriptor*>(&vbd));
2018 queue.WriteBuffer(gpu->vertexBuffer, 0, verts.data(), vbd.size);
2019
2020 if (indexCount > 0) {
2021 // 16-bit index format halves index memory for meshes with <= 65535 verts.
2022 if (vertexCount <= 65535) {
2023 std::vector<uint16_t> idx16;
2024 idx16.reserve(indexCount);

2026 WGPUBufferDescriptor ibd{};
2027 ibd.label = sv("eve_mesh_ib");
2028 ibd.size = uint64_t(indexCount) * sizeof(uint16_t);

2030 ibd.mappedAtCreation = false;
2031 gpu->indexBuffer = device.CreateBuffer(reinterpret_cast<const wgpu::BufferDescriptor*>(&ibd));
2032 queue.WriteBuffer(gpu->indexBuffer, 0, idx16.data(), ibd.size);
2033 gpu->indexFormat = wgpu::IndexFormat::Uint16;
2034 } else {
2035 WGPUBufferDescriptor ibd{};
2036 ibd.label = sv("eve_mesh_ib");
2037 ibd.size = uint64_t(indexCount) * sizeof(uint32_t);
2038 ibd.usage = WGPUBufferUsage_CopyDst | WGPUBufferUsage_Index;
2039 ibd.mappedAtCreation = false;
2040 gpu->indexBuffer = device.CreateBuffer(reinterpret_cast<const wgpu::BufferDescriptor*>(&ibd));
2041 queue.WriteBuffer(gpu->indexBuffer, 0, indices, ibd.size);
2042 gpu->indexFormat = wgpu::IndexFormat::Uint32;
2043 }
2044 }

2048 mesh->gpuHandle = gpu.get();

2050 ownedGpuMeshes.push_back(std::move(gpu));
2051 ownedMeshes.push_back(std::unique_ptr<Mesh>(mesh));

2053}

2467 solidUploads.reserve(solidBatches.size());

2469 if (sb.batch.empty()) {
2470 solidUploads.push_back(SolidUpload{sb.blend, 0, 0});

2491 auto &arena = currentVertexArena();

2493 const uint64_t offset = arena.alloc(bytes);
2494 queue.WriteBuffer(arena.buffer, offset, data.data(), bytes);
2495 solidUploads.push_back(SolidUpload{sb.blend, offset, bytes});

2498 auto solidPipe = [&](BlendMode mode) -> wgpu::RenderPipeline {
2499 switch (mode) {
2500 case BlendMode::Additive:
2501 return offscreen ? offscreenColorAdditivePipeline : colorAdditivePipeline;
2502 case BlendMode::Opaque:

3698 apass.SetBindGroup(0, aoBg, 0, nullptr);

3700 apass.SetIndexBuffer(fullscreenQuadIb, wgpu::IndexFormat::Uint32, 0, 24);
3701 apass.DrawIndexed(6, 1, 0, 0, 0);
3702 apass.End();
3703 popValidationScope();

3705 }
3706 }
3707 }

3710 if (activeCanvas) {
3711 auto *oc = static_cast<OffscreenCanvas *>(activeCanvas);
3712 flush2DToCanvas(oc);
3713 }

3716 {

3749 queue.WriteBuffer(fullscreenQuadIb, 0, indices, sizeof(indices));

3754 sceneGpu.texture = sceneTex;
3755 sceneGpu.view = sceneView;
3756 sceneGpu.sampler = createLinearSampler(device);

3758 uint32_t offsets[1] = {0};

3760 pass.SetBindGroup(0, bg, 1, offsets);
3761 pass.SetVertexBuffer(0, fullscreenQuadVb, 0, 4 * 32);
3762 pass.SetIndexBuffer(fullscreenQuadIb, wgpu::IndexFormat::Uint32, 0, 24);
3763 pass.DrawIndexed(6, 1, 0, 0, 0);
3764 }

3766 if (!activeCanvas) {

3768 surfaceFormat);
3769 }
3770
3771 if (presentOverlayFn_) {
3772 WGPURenderPassEncoder cPass = pass.Get();
3773 presentOverlayFn_(presentOverlayUser_, &cPass);
3774 }
3775 pass.End();
3776 }

3781 // (draws, writes) only execute when ProcessEvents drives the work queue.

3790 // the engine frame is driven by emscripten_set_main_loop (rAF), so the
3791 // Squirrel main loop no longer blocks and no sleep is needed here.
3792#if !defined(__EMSCRIPTEN__)
3793 surface.Present();
3794#endif

3798 frameHad3DThisFrame = false;
3799 frameHad3D = false;
3800 sceneColorPassOpen = false;
3801 active3DCanvas = nullptr;
3802 gbufferPassPending = false;

3892bool copyTextureToCpu(wgpu::Instance &instance, wgpu::Device &device, wgpu::Queue &queue,
3893 wgpu::Texture src, int width, int height, std::vector<uint8_t> &outRgba) {
3894 if (!src) return false;
3895 uint64_t bytesPerRow = static_cast<uint64_t>(width * 4);
3896 bytesPerRow = (bytesPerRow + 255) / 256 * 256; // copy alignment 256
3897 uint64_t size = bytesPerRow * height;
3898
3899 WGPUBufferDescriptor bd{};
3900 bd.label = sv("eve_readback");
3901 bd.size = size;
3902 bd.usage = WGPUBufferUsage_CopyDst | WGPUBufferUsage_MapRead;
3903 bd.mappedAtCreation = false;
3904 wgpu::Buffer dst = device.CreateBuffer(reinterpret_cast<const wgpu::BufferDescriptor*>(&bd));
3905
3906 wgpu::CommandEncoder enc = device.CreateCommandEncoder();
3907 WGPUTexelCopyTextureInfo from{};
3908 from.texture = src.Get();
3909 from.mipLevel = 0;
3910 from.aspect = WGPUTextureAspect_All;
3911 from.origin = {0, 0, 0};
3912 WGPUTexelCopyBufferInfo to{};
3913 to.buffer = dst.Get();
3914 to.layout.offset = 0;
3915 to.layout.bytesPerRow = static_cast<uint32_t>(bytesPerRow);
3916 to.layout.rowsPerImage = static_cast<uint32_t>(height);
3917 WGPUExtent3D extent{static_cast<uint32_t>(width), static_cast<uint32_t>(height), 1};
3918 enc.CopyTextureToBuffer(reinterpret_cast<const wgpu::TexelCopyTextureInfo*>(&from),
3919 reinterpret_cast<const wgpu::TexelCopyBufferInfo*>(&to),
3920 reinterpret_cast<const wgpu::Extent3D*>(&extent));
3921 wgpu::CommandBuffer cmd = enc.Finish();
3922 queue.Submit(1, &cmd);
3923
3924 bool mapped = false;
3925 WGPUBufferMapCallbackInfo cbInfo{};
3926 cbInfo.mode = WGPUCallbackMode_AllowProcessEvents;
3927 cbInfo.callback = [](WGPUMapAsyncStatus status, WGPUStringView /*message*/, void *userdata1,
3928 void * /*userdata2*/) {
3929 bool *ok = static_cast<bool *>(userdata1);
3930 *ok = (status == WGPUMapAsyncStatus_Success);
3931 };
3932 cbInfo.userdata1 = &mapped;
3933 wgpuBufferMapAsync(dst.Get(), WGPUMapMode_Read, 0, size, cbInfo);
3934
3935 int guard = 0;
3936 while (!mapped && guard < 2000) {
3937#if defined(__EMSCRIPTEN__)
3938 emscripten_sleep(0);
3939#endif
3940 wgpuInstanceProcessEvents(instance.Get());
3941 ++guard;
3942 }
3943 if (!mapped) return false;
3944
3945 const uint8_t *data = static_cast<const uint8_t *>(dst.GetConstMappedRange(0, size));
3946 if (!data) return false;
3947 outRgba.resize(static_cast<size_t>(width) * height * 4);
3948 for (int y = 0; y < height; ++y)
3949 std::memcpy(outRgba.data() + size_t(y) * width * 4, data + size_t(y) * bytesPerRow,
3950 size_t(width) * 4);
3951 dst.Unmap();
3952 return true;
3953}

4008 if ((pendingReadback_ && !pendingReadback_->done) || !device) return false;

4014 if (sceneColorSlots.empty()) return false;

4017 src = sceneColorSlots[slot].color;
4018 w = sceneColorWidth;
4019 h = sceneColorHeight;

4022
4023 uint64_t bytesPerRow = static_cast<uint64_t>(w * 4);
