47ProjectileVector steer(ProjectileVector velocity, ProjectileVector desired, double maxRadians) {
48 const double speed = length(velocity);
49 if (speed <= 0.0) return velocity;
50 ProjectileVector current = normalized(velocity);
51 desired = normalized(desired);
52 const double dot = std::clamp(current.x * desired.x + current.y * desired.y + current.z * desired.z, -1.0, 1.0);
53 const double angle = std::acos(dot);
54 if (angle <= maxRadians || angle <= 1e-12) return {desired.x * speed, desired.y * speed, desired.z * speed};
55 const double alpha = maxRadians / angle;
56 ProjectileVector blended{current.x + (desired.x - current.x) * alpha, current.y + (desired.y - current.y) * alpha,
57 current.z + (desired.z - current.z) * alpha};
58 blended = normalized(blended);
59 return {blended.x * speed, blended.y * speed, blended.z * speed};
60}

62ProjectileVector reflect(ProjectileVector velocity, ProjectileVector normal) {
63 const double nLen = length(normal);
64 if (nLen <= 1e-12) return velocity;
65 normal = {normal.x / nLen, normal.y / nLen, normal.z / nLen};
66 const double into = velocity.x * normal.x + velocity.y * normal.y + velocity.z * normal.z;
67 return {velocity.x - 2.0 * into * normal.x, velocity.y - 2.0 * into * normal.y,
68 velocity.z - 2.0 * into * normal.z};
69}

76ProjectileVector rotateAroundAxis(ProjectileVector vector, ProjectileVector axis, double radians) {
77 axis = normalized(axis);
78 const double cosA = std::cos(radians);
79 const double sinA = std::sin(radians);
80 const auto axisCrossV = cross(axis, vector);
81 const double axisDotV = axis.x * vector.x + axis.y * vector.y + axis.z * vector.z;
82 return {vector.x * cosA + axisCrossV.x * sinA + axis.x * axisDotV * (1.0 - cosA),
83 vector.y * cosA + axisCrossV.y * sinA + axis.y * axisDotV * (1.0 - cosA),
84 vector.z * cosA + axisCrossV.z * sinA + axis.z * axisDotV * (1.0 - cosA)};
85}

119Result<std::vector<ProjectileVector>> volleyDirections(const ProjectileVector& base,
120 const CarrierVolleySpec& volley) {
121 auto valid = validateVolley(volley);
122 if (!valid) return Result<std::vector<ProjectileVector>>::failure(valid.status());
123 if (length(base) <= 1e-12)
124 return carrierValueError<std::vector<ProjectileVector>>(DiagnosticCode::InvalidArgument,
125 "Volley base direction must be non-zero",
126 "direction");
127
128 const auto forward = normalized(base);
129 ProjectileVector yawAxis{0.0, 1.0, 0.0};
130 if (std::fabs(forward.x * yawAxis.x + forward.y * yawAxis.y + forward.z * yawAxis.z) > 0.999)
131 yawAxis = lateralBasis(forward);
132
133 std::vector<ProjectileVector> directions;
134 directions.reserve(static_cast<std::size_t>(volley.count));
135 if (volley.count == 1) {
136 directions.push_back(forward);
137 return Result<std::vector<ProjectileVector>>::success(std::move(directions));
138 }
139 if (volley.pattern == CarrierVolleyPattern::Ring) {
140 for (int i = 0; i < volley.count; ++i) {
141 const double radians = (2.0 * kPi * static_cast<double>(i)) / static_cast<double>(volley.count);
142 directions.push_back(normalized(rotateAroundAxis(forward, yawAxis, radians)));
143 }
144 return Result<std::vector<ProjectileVector>>::success(std::move(directions));
145 }
146 const double total = volley.spreadDegrees * kPi / 180.0;
147 for (int i = 0; i < volley.count; ++i) {
148 const double t = static_cast<double>(i) / static_cast<double>(volley.count - 1);
149 const double radians = -0.5 * total + t * total;
150 directions.push_back(normalized(rotateAroundAxis(forward, yawAxis, radians)));
151 }
152 return Result<std::vector<ProjectileVector>>::success(std::move(directions));
153}

157Result<void> CarrierRecipe::validate() const {
158 if (!id.isValid()) return carrierError(DiagnosticCode::InvalidArgument, "Carrier recipe id must not be empty", "id");
159 if (lifetime <= Duration::zero())
160 return carrierError(DiagnosticCode::InvalidArgument, "Carrier lifetime must be positive", "lifetime");
161 if (!std::isfinite(speed) || speed <= 0.0)
162 return carrierError(DiagnosticCode::InvalidArgument, "Carrier speed must be finite and positive", "speed");
163 if (motionOps.empty())
164 return carrierError(DiagnosticCode::InvalidArgument, "Carrier recipe requires at least one motion op",
165 "motionOps");
166
167 bool hasIntegrate = false;
168 for (std::size_t i = 0; i < motionOps.size(); ++i) {
169 const auto& op = motionOps[i];
170 const auto path = "motionOps[" + std::to_string(i) + "]";
171 switch (op.kind) {
172 case CarrierMotionOpKind::SteerHoming:
173 if (!std::isfinite(op.maxTurnRateDegrees) || op.maxTurnRateDegrees <= 0.0)
174 return carrierError(DiagnosticCode::InvalidArgument,
175 "Homing turn rate must be finite and positive", path + ".maxTurnRateDegrees");
176 break;
177 case CarrierMotionOpKind::SteerAvoidBody:
178 if (!std::isfinite(op.avoidLookAhead) || op.avoidLookAhead <= 0.0)
179 return carrierError(DiagnosticCode::InvalidArgument,
180 "Body-avoid look-ahead must be finite and positive", path + ".avoidLookAhead");
181 if (!std::isfinite(op.avoidStrength) || op.avoidStrength < 0.0)
182 return carrierError(DiagnosticCode::InvalidArgument,
183 "Body-avoid strength must be finite and non-negative", path + ".avoidStrength");
184 if (!std::isfinite(op.avoidRadiusPadding) || op.avoidRadiusPadding < 0.0)
185 return carrierError(DiagnosticCode::InvalidArgument,
186 "Body-avoid radius padding must be finite and non-negative",
187 path + ".avoidRadiusPadding");
188 if (!std::isfinite(op.maxTurnRateDegrees) || op.maxTurnRateDegrees <= 0.0)
189 return carrierError(DiagnosticCode::InvalidArgument,
190 "Body-avoid turn rate must be finite and positive",
191 path + ".maxTurnRateDegrees");
192 break;
193 case CarrierMotionOpKind::ApplyGravity:
194 if (!std::isfinite(op.gravity) || op.gravity < 0.0)
195 return carrierError(DiagnosticCode::InvalidArgument,
196 "Gravity must be finite and non-negative", path + ".gravity");
197 break;
198 case CarrierMotionOpKind::Accelerate:
199 if (!std::isfinite(op.acceleration) || op.acceleration < 0.0)
200 return carrierError(DiagnosticCode::InvalidArgument,
201 "Acceleration must be finite and non-negative", path + ".acceleration");
202 break;
203 case CarrierMotionOpKind::CurveSway:
204 case CarrierMotionOpKind::CurveHelix:
205 if (!std::isfinite(op.curveAmplitude) || op.curveAmplitude < 0.0)
206 return carrierError(DiagnosticCode::InvalidArgument,
207 "Curve amplitude must be finite and non-negative", path + ".curveAmplitude");
208 if (!std::isfinite(op.curveFrequencyHz) || op.curveFrequencyHz <= 0.0)
209 return carrierError(DiagnosticCode::InvalidArgument,
210 "Curve frequency must be finite and positive", path + ".curveFrequencyHz");
211 break;
212 case CarrierMotionOpKind::IntegrateLinear:
213 hasIntegrate = true;
214 break;
215 }
216 }
217 if (!hasIntegrate)
218 return carrierError(DiagnosticCode::InvalidArgument,
219 "Carrier recipe must include IntegrateLinear so position advances", "motionOps");
220
221 if (triggers.empty())
222 return carrierError(DiagnosticCode::InvalidArgument, "Carrier recipe requires at least one trigger",
223 "triggers");
224
225 for (std::size_t i = 0; i < triggers.size(); ++i) {
226 const auto& trigger = triggers[i];
227 const auto path = "triggers[" + std::to_string(i) + "]";
228 switch (trigger.kind) {
229 case CarrierTriggerKind::OnFuse:
230 if (trigger.fuse <= Duration::zero())
231 return carrierError(DiagnosticCode::InvalidArgument, "Fuse trigger requires positive fuse",
232 path + ".fuse");
233 break;
234 case CarrierTriggerKind::OnInterval:
235 if (trigger.interval <= Duration::zero())
236 return carrierError(DiagnosticCode::InvalidArgument, "Interval trigger requires positive interval",
237 path + ".interval");
238 break;
239 case CarrierTriggerKind::OnProximity:
240 if (!std::isfinite(trigger.proximityRadius) || trigger.proximityRadius <= 0.0)
241 return carrierError(DiagnosticCode::InvalidArgument,
242 "Proximity trigger requires positive radius", path + ".proximityRadius");
243 break;
244 case CarrierTriggerKind::OnExpire:
245 case CarrierTriggerKind::OnHit:
246 break;
247 }
248 }
249
250 for (std::size_t i = 0; i < impacts.size(); ++i) {
251 const auto& impact = impacts[i];
252 const auto path = "impacts[" + std::to_string(i) + "]";
253 if (!std::isfinite(impact.damage) || impact.damage < 0.0)
254 return carrierError(DiagnosticCode::InvalidArgument, "Impact damage must be finite and non-negative",
255 path + ".damage");
256 if (!std::isfinite(impact.splashRadius) || impact.splashRadius < 0.0)
257 return carrierError(DiagnosticCode::InvalidArgument, "Splash radius must be finite and non-negative",
258 path + ".splashRadius");
259 if (!std::isfinite(impact.restitution) || impact.restitution < 0.0)
260 return carrierError(DiagnosticCode::InvalidArgument, "Bounce restitution must be finite and non-negative",
261 path + ".restitution");
262 if (impact.pierceCount < 0 || impact.bounceCount < 0)
263 return carrierError(DiagnosticCode::InvalidArgument, "Pierce/bounce budgets must be non-negative", path);
264 if (impact.kind == CarrierImpactKind::SpawnChild && !impact.childRecipeId.isValid())
265 return carrierError(DiagnosticCode::InvalidArgument, "SpawnChild requires childRecipeId",
266 path + ".childRecipeId");
267 if (impact.kind == CarrierImpactKind::Splash && impact.splashRadius <= 0.0)
268 return carrierError(DiagnosticCode::InvalidArgument, "Splash impact requires positive splashRadius",
269 path + ".splashRadius");
270 }
271 return Result<void>::success();
272}

274Result<CarrierRecipe> carrierRecipeFromProjectile(const ProjectileDefinition& definition, double damage) {
275 auto valid = definition.validate();
276 if (!valid) return Result<CarrierRecipe>::failure(valid.status());
277 if (!std::isfinite(damage) || damage < 0.0)
278 return carrierValueError<CarrierRecipe>(DiagnosticCode::InvalidArgument,
279 "Bridge damage must be finite and non-negative", "damage");
280
281 CarrierRecipe recipe;
282 recipe.id = definition.id;
283 recipe.lifetime = definition.lifetime;
284 recipe.speed = definition.speed;
285
286 switch (definition.mode) {
287 case ProjectileMode::Linear:
288 recipe.motionOps.push_back({CarrierMotionOpKind::IntegrateLinear});
289 break;
290 case ProjectileMode::Ballistic:
291 recipe.motionOps.push_back({CarrierMotionOpKind::ApplyGravity, definition.gravity});
292 recipe.motionOps.push_back({CarrierMotionOpKind::IntegrateLinear});
293 break;
294 case ProjectileMode::Homing:
295 recipe.motionOps.push_back(
296 {CarrierMotionOpKind::SteerHoming, 0.0, definition.maxTurnRateDegrees});
297 recipe.motionOps.push_back({CarrierMotionOpKind::IntegrateLinear});
298 break;
299 }
300
301 recipe.triggers.push_back({CarrierTriggerKind::OnExpire});
302 recipe.triggers.push_back({CarrierTriggerKind::OnHit});
303 recipe.impacts.push_back({CarrierImpactKind::EmitHit, CarrierTriggerKind::OnHit, damage});
304 recipe.impacts.push_back({CarrierImpactKind::Release, CarrierTriggerKind::OnHit});
305 recipe.impacts.push_back({CarrierImpactKind::Release, CarrierTriggerKind::OnExpire});
306
307 auto recipeValid = recipe.validate();
308 if (!recipeValid) return Result<CarrierRecipe>::failure(recipeValid.status());
309 return Result<CarrierRecipe>::success(std::move(recipe));
310}

324Result<CombatCarrierRuntime::LiveRecipe> CombatCarrierRuntime::prepareRecipe(const CarrierRecipe& recipe) {
325 auto valid = recipe.validate();
326 if (!valid) return Result<LiveRecipe>::failure(valid.status());
327
328 LiveRecipe live;
329 live.recipe = recipe;
330 for (const auto& op : recipe.motionOps) {
331 if (op.kind == CarrierMotionOpKind::SteerHoming) live.needsHoming = true;
332 if (op.kind == CarrierMotionOpKind::SteerAvoidBody) live.needsBodyAvoid = true;
333 }
334 for (const auto& trigger : recipe.triggers) {
335 if (trigger.kind == CarrierTriggerKind::OnHit) live.needsHit = true;
336 if (trigger.kind == CarrierTriggerKind::OnProximity) live.needsProximity = true;
337 }
338 for (const auto& impact : recipe.impacts) {
339 if (impact.kind == CarrierImpactKind::Pierce)
340 live.initialPierce = std::max(live.initialPierce, impact.pierceCount);
341 if (impact.kind == CarrierImpactKind::Bounce)
342 live.initialBounce = std::max(live.initialBounce, impact.bounceCount);
343 }
344 return Result<LiveRecipe>::success(std::move(live));
345}

396 for (std::uint32_t slot = 0; slot < slots_.size(); ++slot) {

435 const CarrierSpawnRequest& request,

457 }

512 if (live->needsBodyAvoid && bodies == nullptr)
513 return carrierValueError<CarrierFrame>(DiagnosticCode::Unsupported,
514 "SteerAvoidBody carrier update requires a body-defense provider",
515 "bodies");
516 candidates.push_back({*slot.state, false});
517 }
518
519 for (auto& candidate : candidates) {
520 const LiveRecipe* live = lookupLive(candidate.state.recipeId);
521 if (live == nullptr)
522 return carrierValueError<CarrierFrame>(DiagnosticCode::NotFound,
523 "Live carrier references an unregistered recipe", "recipeId");
524
525 const Duration previousAge = candidate.state.age;
526 CarrierMotion previous = candidate.state.motion;
527 CarrierMotion motion = previous;
528
529 std::optional<ProjectilePoint> targetPos;
530 if (live->needsHoming || live->needsProximity) {
531 if (!candidate.state.target.has_value())
532 return carrierValueError<CarrierFrame>(DiagnosticCode::InvalidArgument,
533 "Homing/proximity carrier lost its target handle", "target");
534 auto resolved = targets->position(*candidate.state.target);
535 if (!resolved) return Result<CarrierFrame>::failure(resolved.status());
536 targetPos = resolved.value();
537 }
538
539 const double dtSeconds = delta.seconds();
540 const double ageSeconds = candidate.state.age.seconds();
541 for (const auto& op : live->recipe.motionOps) {
542 switch (op.kind) {
543 case CarrierMotionOpKind::SteerHoming: {
544 const auto desired = directionTo(motion.position, *targetPos);
545 if (length(desired) <= 1e-12) break;
546 const double maxRadians = op.maxTurnRateDegrees * kPi / 180.0 * dtSeconds;
547 motion.velocity = steer(motion.velocity, desired, maxRadians);
548 break;
549 }
550 case CarrierMotionOpKind::SteerAvoidBody: {
551 ecs::EntityHandle targetHandle{};
552 if (candidate.state.target.has_value()) targetHandle = *candidate.state.target;
553 auto samples = bodies->sample(candidate.state.handle, targetHandle, motion);
554 if (!samples) return Result<CarrierFrame>::failure(samples.status());
555 ProjectileVector avoid{0.0, 0.0, 0.0};
556 bool any = false;
557 const auto fwd = length(motion.velocity) > 1e-12 ? normalized(motion.velocity)
558 : ProjectileVector{1.0, 0.0, 0.0};
559 for (const auto& body : samples.value()) {
560 if (!std::isfinite(body.radius) || body.radius < 0.0) {
561 return carrierValueError<CarrierFrame>(
562 DiagnosticCode::InvalidArgument,
563 "Body defense radius must be finite and non-negative", "bodies");
564 }
565 const auto toCenter = directionTo(motion.position, body.center);
566 const double dist = length(toCenter);
567 const double blockAt = body.radius + op.avoidRadiusPadding + op.avoidLookAhead;
568 if (dist <= 1e-12 || dist > blockAt) continue;
569 if (length(body.facing) > 1e-12 && body.frontConeDegrees < 180.0) {
570 const auto approach =
571 normalized(ProjectileVector{-toCenter.x, -toCenter.y, -toCenter.z});
572 const auto face = normalized(body.facing);
573 const double dot = std::clamp(
574 approach.x * face.x + approach.y * face.y + approach.z * face.z, -1.0, 1.0);
575 const double angleDeg = std::acos(dot) * 180.0 / kPi;
576 if (angleDeg > body.frontConeDegrees) continue;
577 }
578 auto lateral = cross(fwd, toCenter);
579 if (length(lateral) <= 1e-12) lateral = lateralBasis(fwd);
580 lateral = normalized(lateral);
581 if (targetPos.has_value()) {
582 const auto around = directionTo(body.center, *targetPos);
583 const auto left = lateral;
584 const auto right = ProjectileVector{-lateral.x, -lateral.y, -lateral.z};
585 const double leftScore = left.x * around.x + left.y * around.y + left.z * around.z;
586 const double rightScore = right.x * around.x + right.y * around.y + right.z * around.z;
587 if (rightScore > leftScore) lateral = right;
588 }
589 const double weight = op.avoidStrength * (1.0 - dist / blockAt);
590 avoid.x += lateral.x * weight;
591 avoid.y += lateral.y * weight;
592 avoid.z += lateral.z * weight;
593 any = true;
594 }
595 if (any && length(avoid) > 1e-12) {
596 const auto desired = ProjectileVector{fwd.x + avoid.x, fwd.y + avoid.y, fwd.z + avoid.z};
597 const double maxRadians = op.maxTurnRateDegrees * kPi / 180.0 * dtSeconds;
598 motion.velocity = steer(motion.velocity, desired, maxRadians);
599 }
600 break;
601 }
602 case CarrierMotionOpKind::ApplyGravity:
603 motion.velocity.y -= op.gravity * dtSeconds;
604 break;
605 case CarrierMotionOpKind::Accelerate: {
606 const double speed = length(motion.velocity);
607 if (speed <= 1e-12) break;
608 const auto dir = normalized(motion.velocity);
609 const double next = speed + op.acceleration * dtSeconds;
610 motion.velocity = {dir.x * next, dir.y * next, dir.z * next};
611 break;
612 }
613 case CarrierMotionOpKind::CurveSway: {
614 const double speed = length(motion.velocity);
615 if (speed <= 1e-12) break;
616 const auto fwd = normalized(motion.velocity);
617 const auto side = lateralBasis(fwd);
618 const double omega = 2.0 * kPi * op.curveFrequencyHz;
619 const double vLat = op.curveAmplitude * omega * std::cos(omega * ageSeconds);
620 motion.velocity = {fwd.x * speed + side.x * vLat, fwd.y * speed + side.y * vLat,
621 fwd.z * speed + side.z * vLat};
622 break;
623 }
624 case CarrierMotionOpKind::CurveHelix: {
625 const double speed = length(motion.velocity);
626 if (speed <= 1e-12) break;
627 const auto fwd = normalized(motion.velocity);
628 const auto side0 = lateralBasis(fwd);
629 const auto up = normalized(cross(side0, fwd));
630 const double omega = 2.0 * kPi * op.curveFrequencyHz;
631 const double phase = omega * ageSeconds;
632 const auto side = ProjectileVector{side0.x * std::cos(phase) + up.x * std::sin(phase),
633 side0.y * std::cos(phase) + up.y * std::sin(phase),
634 side0.z * std::cos(phase) + up.z * std::sin(phase)};
635 const double vLat = op.curveAmplitude * omega;
636 motion.velocity = {fwd.x * speed + side.x * vLat, fwd.y * speed + side.y * vLat,
637 fwd.z * speed + side.z * vLat};
638 break;
639 }
640 case CarrierMotionOpKind::IntegrateLinear:
641 motion.position.x += motion.velocity.x * dtSeconds;
642 motion.position.y += motion.velocity.y * dtSeconds;
643 motion.position.z += motion.velocity.z * dtSeconds;
644 break;
645 }
646 }
647
648 if (!finite(motion.position) || !finite(motion.velocity))
649 return carrierValueError<CarrierFrame>(DiagnosticCode::Failed,
650 "Carrier motion produced non-finite values", "motion");
651
652 auto nextAge = previousAge.tryAdd(delta);
653 if (!nextAge) return Result<CarrierFrame>::failure(nextAge.status());
654
655 candidate.state.motion = motion;
656 candidate.state.age = std::move(nextAge).takeValue();
657
658 std::vector<TriggerFire> fires;
659 for (const auto& trigger : live->recipe.triggers) {
660 switch (trigger.kind) {
661 case CarrierTriggerKind::OnExpire:
662 if (crossed(previousAge, candidate.state.age, candidate.state.lifetime) ||
663 candidate.state.age >= candidate.state.lifetime)
664 fires.push_back({CarrierTriggerKind::OnExpire, std::nullopt});
665 break;
666 case CarrierTriggerKind::OnFuse:
667 if (crossed(previousAge, candidate.state.age, trigger.fuse))
668 fires.push_back({CarrierTriggerKind::OnFuse, std::nullopt});
669 break;
670 case CarrierTriggerKind::OnInterval: {
671 const int before = intervalIndex(previousAge, trigger.interval);
672 const int after = intervalIndex(candidate.state.age, trigger.interval);
673 for (int i = before + 1; i <= after; ++i)
674 fires.push_back({CarrierTriggerKind::OnInterval, std::nullopt});
675 break;
676 }
677 case CarrierTriggerKind::OnHit: {
678 auto contacts = hits->query(candidate.state.handle, previous, motion);
679 if (!contacts) return Result<CarrierFrame>::failure(contacts.status());
680 for (auto& contact : contacts.value())
681 fires.push_back({CarrierTriggerKind::OnHit, std::move(contact)});
682 break;
683 }
684 case CarrierTriggerKind::OnProximity: {
685 const auto deltaPos = directionTo(motion.position, *targetPos);
686 const double dist = length(deltaPos);
687 if (dist <= trigger.proximityRadius)
688 fires.push_back({CarrierTriggerKind::OnProximity, std::nullopt});
689 break;
690 }
691 }
692 }
693
694 for (const auto& fire : fires) {
695 bool suppressRelease = false;
696 bool requestRelease = false;
697
698 for (const auto& impact : live->recipe.impacts) {
699 if (impact.on != fire.kind) continue;
700
701 CarrierEvent event;
702 event.carrier = candidate.state.handle;
703 event.recipeId = candidate.state.recipeId;
704 event.trigger = fire.kind;
705 event.impact = impact.kind;
706 event.position = candidate.state.motion.position;
707 event.source = candidate.state.source;
708 if (fire.contact.has_value()) {
709 event.target = fire.contact->target;
710 event.normal = fire.contact->normal;
711 event.position = fire.contact->point;
712 } else if (candidate.state.target.has_value()) {
713 event.target = *candidate.state.target;
714 }
715
716 switch (impact.kind) {
717 case CarrierImpactKind::EmitHit:
718 event.damage = impact.damage;
719 event.damageType = impact.damageType;
720 event.element = impact.element;
721 events.push_back(std::move(event));
722 break;
723 case CarrierImpactKind::Splash:
724 event.damage = impact.damage;
725 event.splashRadius = impact.splashRadius;
726 event.damageType = impact.damageType;
727 event.element = impact.element;
728 event.position = candidate.state.motion.position;
729 events.push_back(std::move(event));
730 break;
731 case CarrierImpactKind::Pierce:
732 if (candidate.state.pierceRemaining > 0) {
733 --candidate.state.pierceRemaining;
734 suppressRelease = true;
735 }
736 break;
737 case CarrierImpactKind::Bounce:
738 if (candidate.state.bounceRemaining > 0 && fire.contact.has_value()) {
739 --candidate.state.bounceRemaining;
740 auto reflected = reflect(candidate.state.motion.velocity, fire.contact->normal);
741 reflected.x *= impact.restitution;
742 reflected.y *= impact.restitution;
743 reflected.z *= impact.restitution;
744 candidate.state.motion.velocity = reflected;
745 suppressRelease = true;
746 }
747 break;
748 case CarrierImpactKind::Release:
749 requestRelease = true;
750 break;
751 case CarrierImpactKind::SpawnChild: {
752 CarrierChildSpawn child;
753 child.recipeId = impact.childRecipeId;
754 child.parent = candidate.state.handle;
755 child.request.position = candidate.state.motion.position;
756 child.request.direction = candidate.state.motion.velocity;
757 if (length(child.request.direction) <= 1e-12) child.request.direction = {1.0, 0.0, 0.0};
758 child.request.target = candidate.state.target;
759 child.request.source = candidate.state.source;
760 childSpawns.push_back(std::move(child));
761 break;
762 }
763 }
764 }
765
766 if (requestRelease && !suppressRelease) candidate.release = true;
767 }
768
769 if (candidate.state.age >= candidate.state.lifetime) {
770 bool hasExpireRelease = false;
771 for (const auto& impact : live->recipe.impacts) {
772 if (impact.kind == CarrierImpactKind::Release && impact.on == CarrierTriggerKind::OnExpire)
773 hasExpireRelease = true;
774 }
775 if (!hasExpireRelease &&
776 std::any_of(fires.begin(), fires.end(),
777 [](const TriggerFire& f) { return f.kind == CarrierTriggerKind::OnExpire; }))
778 candidate.release = true;
779 }
780 }
781
782 std::vector<Slot> staged = slots_;
783 std::size_t stagedActive = activeCount_;
784 for (const auto& candidate : candidates) {
785 auto& cell = staged[candidate.state.handle.slot];
786 if (cell.generation != candidate.state.handle.generation || !cell.state.has_value())
787 return carrierValueError<CarrierFrame>(DiagnosticCode::StaleHandle, "Carrier handle became stale",
788 "handle");
789 if (candidate.release) {
790 cell.state.reset();
791 --stagedActive;
792 } else {
793 cell.state = candidate.state;
794 }
795 }
796
797 std::vector<CarrierHandle> spawnedChildren;
798 spawnedChildren.reserve(childSpawns.size());
799 for (const auto& child : childSpawns) {
800 const LiveRecipe* childLive = lookupLive(child.recipeId);
801 if (childLive == nullptr)
802 return carrierValueError<CarrierFrame>(DiagnosticCode::NotFound,
803 "Child spawn references an unregistered recipe", "childRecipeId");
804 auto valid = validateSpawn(childLive->recipe, child.request);
805 if (!valid) return Result<CarrierFrame>::failure(valid.status());
806
807 bool placed = false;
808 for (std::uint32_t slot = 0; slot < staged.size(); ++slot) {
809 if (staged[slot].state.has_value()) continue;
810 Slot& cell = staged[slot];
811 ++cell.generation;
812 if (cell.generation == 0) ++cell.generation;
813 CarrierState state;
814 state.handle = {slot, cell.generation};
815 state.recipeId = childLive->recipe.id;
816 state.lifetime = childLive->recipe.lifetime;
817 state.age = Duration::zero();
818 state.pierceRemaining = childLive->initialPierce;
819 state.bounceRemaining = childLive->initialBounce;
820 state.target = child.request.target;
821 state.source = child.request.source;
822 const auto dir = normalized(child.request.direction);
823 state.motion.position = child.request.position;
824 state.motion.velocity = {dir.x * childLive->recipe.speed, dir.y * childLive->recipe.speed,
825 dir.z * childLive->recipe.speed};
826 cell.state = std::move(state);
827 ++stagedActive;
828 spawnedChildren.push_back(cell.state->handle);
829 placed = true;
830 break;
831 }
832 if (!placed)
833 return carrierValueError<CarrierFrame>(DiagnosticCode::Conflict,
834 "Carrier pool cannot fit deferred child spawns", "capacity");
835 }
836
837 slots_ = std::move(staged);
838 activeCount_ = stagedActive;
839
840 CarrierFrame frame;
841 for (const auto& candidate : candidates) {
842 frame.advanced.push_back(candidate.state.handle);
843 if (candidate.release) frame.released.push_back(candidate.state.handle);
844 }
845 for (const auto& child : spawnedChildren) frame.advanced.push_back(child);
846 frame.events = std::move(events);
847 frame.childSpawns = std::move(childSpawns);
848 return Result<CarrierFrame>::success(std::move(frame));
849}

885Result<void> CombatCarrierRuntime::restore(const CarrierRuntimeSnapshot& snapshot) {
886 if (snapshot.slots.empty() || snapshot.slots.size() > kMaximumCapacity)
887 return carrierError(DiagnosticCode::InvalidArgument, "Carrier snapshot capacity must be in 1..1048576",
888 "slots");
889
890 std::vector<Slot> staged;
891 staged.reserve(snapshot.slots.size());
892 std::size_t active = 0;
893 for (const auto& slot : snapshot.slots) {
894 if (slot.state.has_value()) {
895 if (slot.state->handle.slot >= snapshot.slots.size() || slot.state->handle.generation != slot.generation)
896 return carrierError(DiagnosticCode::InvalidArgument,
897 "Carrier snapshot state handle does not match slot metadata", "slots");
898 if (!findRecipe(slot.state->recipeId).has_value())
899 return carrierError(DiagnosticCode::NotFound,
900 "Carrier snapshot references an unregistered recipe", "recipeId");
901 ++active;
902 }
903 staged.push_back({slot.generation, slot.state});
904 }
905 slots_ = std::move(staged);
906 activeCount_ = active;
907 return Result<void>::success();
908}
