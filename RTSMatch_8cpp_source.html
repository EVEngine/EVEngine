41bool survives(Faction& faction, const Match::Rules& rules) {
42 if (rules.rule == VictoryRule::DestroyHeadquarters) {
43 auto buildings = ecs::View<Building, Building::Definition, Building::Faction, Building::Integrity,
44 Building::Construction>();
45 for (auto it = buildings.begin(); it != buildings.end(); ++it) {
46 auto [definition, owner, integrity, construction] = *it;
47 if (owns(owner->link, faction) && integrity->alive && integrity->state.health > 0.0 &&
48 construction->progress >= 1.0f && definition->id.name() == rules.archetype)
49 return true;
50 }
51 return false;
52 }

54 for (auto it = units.begin(); it != units.end(); ++it) {

85void settleTeams(Match& match) {
86 std::set<int> teams;
87 for (auto& entry : match.participants()->entries)
88 if (!entry.eliminated && entry.faction.resolve() != nullptr) teams.insert(entry.team);
89 if (teams.size() > 1) return;
90 match.state()->phase = MatchPhase::Finished;
91 match.state()->winningTeam = teams.empty() ? -1 : *teams.begin();
92 emit(match, teams.empty() ? "match_draw" : "match_won", nullptr, match.state()->winningTeam,
93 teams.empty() ? "no_survivors" : "last_team_standing");
94}

109Result<void> MatchSystem::start(Match& match) {
110 if (match.state()->phase != MatchPhase::Setup)
111 return failure<void>(DiagnosticCode::Conflict, "RTS match is not in setup", "match.phase");
112 if (match.participants()->entries.size() < 2)
113 return failure<void>(DiagnosticCode::PreconditionViolation, "RTS match requires at least two factions",
114 "participants");
115 for (const auto& entry : match.participants()->entries)
116 if (entry.faction.resolve() == nullptr)
117 return failure<void>(DiagnosticCode::StaleHandle, "RTS match contains a stale faction", "participants");
118 const auto& rules = *match.rules();
119 if (rules.rule == VictoryRule::DestroyHeadquarters && rules.archetype.empty())
120 return failure<void>(DiagnosticCode::InvalidArgument, "headquarters victory requires an archetype", "rules");
121 if (rules.rule == VictoryRule::ResourceTarget &&
122 (rules.archetype.empty() || rules.targetValue <= 0.0 || !std::isfinite(rules.targetValue)))
123 return failure<void>(DiagnosticCode::InvalidArgument, "resource victory requires a positive target", "rules");
124 match.state()->phase = MatchPhase::Running;
125 match.state()->winningTeam = -1;
126 emit(match, "match_started", nullptr, -1);
127 return Result<void>::success(Status::success(StatusCode::Applied));
128}

130Result<void> MatchSystem::surrender(Match& match, Faction& faction) {
131 if (match.state()->phase != MatchPhase::Running)
132 return failure<void>(DiagnosticCode::Conflict, "RTS match is not running", "match.phase");
133 auto* entry = participant(match, faction);
134 if (entry == nullptr || entry->eliminated)
135 return failure<void>(DiagnosticCode::NotFound, "RTS faction is not an active participant", "faction");
136 destroyFactionEntities(faction);

139 return Result<void>::success(Status::success(StatusCode::Applied));

142Result<std::size_t> MatchSystem::step(Match& match, const MatchResourceQuery& resources) {
143 if (match.state()->phase != MatchPhase::Running)
144 return Result<std::size_t>::success(0, Status::success(StatusCode::NoOp));
145 std::size_t eliminated = 0;
146 if (match.rules()->rule == VictoryRule::ResourceTarget) {
147 if (!resources)
148 return failure<std::size_t>(DiagnosticCode::PreconditionViolation,
149 "resource victory requires an authoritative economy query", "resources");
150 for (auto& entry : match.participants()->entries) {
151 if (entry.eliminated) continue;
152 auto* faction = dynamic_cast<Faction*>(entry.faction.resolve());
153 if (faction == nullptr) {
154 eliminate(match, entry, "stale_faction");
155 ++eliminated;
156 continue;
157 }
158 auto balance = resources(*faction, match.rules()->archetype);
159 if (!balance) return Result<std::size_t>::failure(balance.status());
160 if (balance.value() >= match.rules()->targetValue) {
161 match.state()->phase = MatchPhase::Finished;
162 match.state()->winningTeam = entry.team;
163 emit(match, "match_won", faction, entry.team, "resource_target");
164 return Result<std::size_t>::success(eliminated + 1, Status::success(StatusCode::Applied));
165 }
166 }
167 } else {
168 for (auto& entry : match.participants()->entries) {
169 if (entry.eliminated) continue;
170 auto* faction = dynamic_cast<Faction*>(entry.faction.resolve());
171 if (faction == nullptr || !survives(*faction, *match.rules())) {
172 eliminate(match, entry, faction == nullptr ? "stale_faction" :
173 match.rules()->rule == VictoryRule::DestroyHeadquarters ? "headquarters_destroyed" :
174 "annihilated");
175 ++eliminated;
176 }
177 }
178 }
179 settleTeams(match);
180 return Result<std::size_t>::success(eliminated,
181 Status::success(eliminated == 0 ? StatusCode::NoOp : StatusCode::Applied));
182}
