18editor::EditorResult<editor::AutomationOwnedTarget> makeTarget(const editor::TargetId& id,
19 const editor::EditorValue::Object& request) {
20 auto target = std::make_unique<Target>(id.value());
21 const auto snapshot = request.find("snapshot");
22 if (snapshot != request.end()) {
23 auto loaded = target->loadSnapshot(snapshot->second);
24 if (!loaded.ok()) {
25 return editor::EditorResult<editor::AutomationOwnedTarget>::failure(loaded.status());
26 }
27 }
28 editor::AutomationOwnedTarget owned;
29 owned.target = std::move(target);
30 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
31}

39editor::EditorResult<editor::AutomationOwnedTarget> FluidsAutomationTargetFactory::create(
