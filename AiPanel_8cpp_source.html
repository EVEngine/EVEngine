15AiPanel& AiPanel::instance() {
16 // Process-immortal singleton: McpServer's detached stdio reader and the
17 // other DevTools singletons may still touch AiPanel during teardown, so
18 // destroying it would let them lock a destroyed mutex (libc++ aborts).
19 // See devtools/Immortal.hpp for the shared contract.

23std::string AiPanel::nowStamp() {

97std::vector<AiLogEntry> AiPanel::recentLog(size_t max) const {
98 std::lock_guard<std::mutex> lock(mu_);
99 std::vector<AiLogEntry> out;
100 if (log_.empty() || max == 0) return out;
101 const size_t start = log_.size() > max ? log_.size() - max : 0;
102 out.assign(log_.begin() + static_cast<std::ptrdiff_t>(start), log_.end());
103 return out;
104}

106std::string AiPanel::formatLog(size_t max) const {
107 auto entries = recentLog(max);
108 std::ostringstream oss;
109 for (const auto& e : entries) {
110 oss << '[' << e.timestamp << "] " << e.kind << " | " << e.title;
111 if (!e.detail.empty()) oss << " — " << e.detail;
112 oss << '\n';
113 }
114 return oss.str();
115}
