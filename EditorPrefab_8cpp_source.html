54EditorResult<void> validatePrefab(const PrefabAssetSnapshot& prefab) {
55 if (prefab.asset.empty() || prefab.rootSourceId.empty() || prefab.objects.empty() || prefab.revision == 0)
56 return prefabError<void>(EditorStatus::Rejected, "editor.prefab.invalid-identity",
57 "Prefab requires asset, revision, root and objects");
58 std::set<ObjectId> ids;
59 for (const PrefabObjectRecord& object : prefab.objects)
60 if (object.sourceId.empty() || !ids.insert(object.sourceId).second)
61 return prefabError<void>(EditorStatus::Rejected, "editor.prefab.duplicate-object",
62 "Prefab object ids must be unique and non-empty");
63 for (const PrefabObjectRecord& object : prefab.objects)
64 if (object.nestedPrefab.empty() != (object.nestedRevision == 0))
65 return prefabError<void>(EditorStatus::Rejected, "editor.prefab.invalid-nested-pin",
66 "Nested prefab references require both asset and non-zero revision");
67 if (!ids.contains(prefab.rootSourceId))
68 return prefabError<void>(EditorStatus::Rejected, "editor.prefab.root-not-found",
69 "Prefab root does not exist");
70 for (const PrefabObjectRecord& object : prefab.objects) {
71 if (object.sourceId == prefab.rootSourceId && !object.parentSourceId.empty())
72 return prefabError<void>(EditorStatus::Rejected, "editor.prefab.root-has-parent",
73 "Prefab root must not have an internal parent");
74 if (object.sourceId != prefab.rootSourceId && !ids.contains(object.parentSourceId))
75 return prefabError<void>(EditorStatus::Rejected, "editor.prefab.parent-not-found",
76 "Prefab object references a missing parent");
77 std::set<ObjectId> ancestry;
78 const PrefabObjectRecord* cursor = &object;
79 while (!cursor->parentSourceId.empty()) {
80 if (!ancestry.insert(cursor->sourceId).second)
81 return prefabError<void>(EditorStatus::Rejected, "editor.prefab.hierarchy-cycle",
82 "Prefab hierarchy contains a cycle");
83 const auto parent = std::find_if(prefab.objects.begin(), prefab.objects.end(),
84 [&](const PrefabObjectRecord& candidate) {
85 return candidate.sourceId == cursor->parentSourceId;
86 });
87 if (parent == prefab.objects.end()) break;
88 cursor = &*parent;
89 }
90 if (cursor->sourceId != prefab.rootSourceId)
91 return prefabError<void>(EditorStatus::Rejected, "editor.prefab.disconnected-object",
92 "Every prefab object must descend from its root");
93 }
94 return eve::editing::applied<void>();
95}

99EditorResult<PrefabAssetSnapshot> ScenePrefabService::capture(const AssetGuid& asset,
100 const SceneTargetBase& scene,
101 const ObjectId& root) const {
102 if (asset.empty())
103 return prefabError<PrefabAssetSnapshot>(EditorStatus::Rejected, "editor.prefab.asset-required",
104 "Prefab asset identity is required");
105 auto rootObject = scene.sceneObject(root);
106 if (!rootObject.ok())
107 return prefabError<PrefabAssetSnapshot>(EditorStatus::NotFound, "editor.prefab.root-not-found",
108 "Prefab capture root does not exist");
109 PrefabAssetSnapshot prefab;
110 prefab.asset = asset;
111 prefab.rootSourceId = root;
112 std::vector<ObjectId> pending{root};
113 for (size_t index = 0; index < pending.size(); ++index) {
114 const ObjectId id = pending[index];
115 auto object = scene.sceneObject(id);
116 if (!object.ok())
117 return prefabError<PrefabAssetSnapshot>(EditorStatus::Conflict, "editor.prefab.capture-diverged",
118 "Scene hierarchy changed during prefab capture");
119 prefab.objects.push_back({id, id == root ? ObjectId{} : object.value().parent,
120 object.value().name, object.value().transform, {}, 0});
121 const auto children = scene.sceneChildren(id);
122 pending.insert(pending.end(), children.begin(), children.end());
123 }
124 return eve::editing::applied<PrefabAssetSnapshot>(std::move(prefab));
125}

127EditorResult<PrefabInstancePlan> ScenePrefabService::instantiate(const PrefabAssetSnapshot& prefab,
128 const std::string& instanceId,
129 const ObjectId& parent,
130 const SceneTargetBase& scene) const {
131 if (!validatePrefab(prefab).ok() || instanceId.empty())
132 return prefabError<PrefabInstancePlan>(EditorStatus::Rejected, "editor.prefab.invalid-instance",
133 "Prefab and non-empty instance id are required");
134 if (!parent.empty() && !scene.sceneObject(parent).ok())
135 return prefabError<PrefabInstancePlan>(EditorStatus::NotFound, "editor.prefab.scene-parent-not-found",
136 "Prefab instance parent does not exist");
137 PrefabInstancePlan plan;
138 plan.instanceId = instanceId;
139 plan.sourceAsset = prefab.asset;
140 plan.sourceRevision = prefab.revision;
141 for (const PrefabObjectRecord& object : prefab.objects) {
142 const ObjectId id = instanceObject(instanceId, object.sourceId);
143 if (scene.sceneObject(id).ok())
144 return prefabError<PrefabInstancePlan>(EditorStatus::Conflict, "editor.prefab.instance-id-conflict",
145 "Prefab instance object already exists: " + id.value());
146 CreateSceneObjectRequest request;
147 request.id = id;
148 request.parent = object.sourceId == prefab.rootSourceId
149 ? parent
150 : instanceObject(instanceId, object.parentSourceId);
151 request.name = object.name;
152 request.transform = object.transform;
153 auto operation = scene.makeCreate(request);
154 if (!operation.ok()) {
155 // Parents created earlier in this plan are intentionally not yet in
156 // the target, so construct the same validated operation value here.
157 DomainOperation create;
158 create.type = "scene.object.create.v1";
159 create.inverseType = "scene.object.delete.v1";
160 create.target = TargetId(scene.targetId());
161 EditorValue::Object payload{{"id", request.id.value()}, {"parent", request.parent.value()},
162 {"name", request.name}, {"transform", transformValue(request.transform)}};
163 create.payload = EditorValue(payload);
164 create.inverse = EditorValue(std::move(payload));
165 create.hasInverse = true;
166 create.affectedObjects.push_back({TargetId(scene.targetId()), request.id.value(), 0});
167 plan.operations.push_back(std::move(create));
168 } else {
169 plan.operations.push_back(std::move(operation).takeValue());
170 }
171 }
172 return eve::editing::applied<PrefabInstancePlan>(std::move(plan));
173}

180 "Prefab and instance id are required");

215 const std::string& instanceId,
216 const ObjectId& parent,
217 const SceneTargetBase& scene) const {

220 "Prefab and instance id are required");
221 PrefabAssetSnapshot result = prefab;
222 ++result.revision;
223 std::map<ObjectId, ObjectId> instanceToSource;
224 for (const PrefabObjectRecord& source : prefab.objects)
225 instanceToSource.emplace(instanceObject(instanceId, source.sourceId), source.sourceId);
226 for (PrefabObjectRecord& destination : result.objects) {
227 const ObjectId instance = instanceObject(instanceId, destination.sourceId);
228 auto object = scene.sceneObject(instance);
229 if (!object.ok())
230 return prefabError<PrefabAssetSnapshot>(EditorStatus::Conflict,
231 "editor.prefab.instance-structure-changed",
232 "Prefab instance object is missing: " + instance.value());
233 destination.name = object.value().name;
234 destination.transform = object.value().transform;
235 if (destination.sourceId == prefab.rootSourceId) {
236 if (object.value().parent != parent)
237 return prefabError<PrefabAssetSnapshot>(EditorStatus::Conflict,
238 "editor.prefab.root-reparented",
239 "Prefab root moved outside its instance parent");
240 destination.parentSourceId = ObjectId{};
241 } else {
242 const auto sourceParent = instanceToSource.find(object.value().parent);
243 if (sourceParent == instanceToSource.end())
244 return prefabError<PrefabAssetSnapshot>(EditorStatus::Conflict,
245 "editor.prefab.child-reparented-outside",
246 "Prefab child moved outside its instance");
247 destination.parentSourceId = sourceParent->second;
248 }
249 }

253EditorResult<std::vector<PrefabOverrideRecord>> ScenePrefabService::inspectOverrides(
254 const PrefabAssetSnapshot& prefab, const std::string& instanceId, const ObjectId& parent,
255 const SceneTargetBase& scene) const {
256 if (!validatePrefab(prefab).ok() || instanceId.empty())
257 return prefabError<std::vector<PrefabOverrideRecord>>(EditorStatus::Rejected,
258 "editor.prefab.invalid-instance", "Prefab and instance id are required");
259 std::vector<PrefabOverrideRecord> result;
260 for (const PrefabObjectRecord& source : prefab.objects) {
261 const ObjectId instance = instanceObject(instanceId, source.sourceId);
262 auto object = scene.sceneObject(instance);
263 if (!object.ok()) {
264 result.push_back({instance, source.sourceId, "object", "missing"});
265 continue;
266 }
267 const ObjectId expectedParent = source.sourceId == prefab.rootSourceId
268 ? parent : instanceObject(instanceId, source.parentSourceId);
269 if (object.value().parent != expectedParent)
270 result.push_back({instance, source.sourceId, "hierarchy.parent", "modified"});
271 if (object.value().name != source.name)
272 result.push_back({instance, source.sourceId, "object.name", "modified"});
273 if (object.value().transform != source.transform)
274 result.push_back({instance, source.sourceId, "transform", "modified"});
275 }
276 std::sort(result.begin(), result.end(), [](const auto& a, const auto& b) {
277 return a.instanceObject == b.instanceObject ? a.property < b.property
278 : a.instanceObject < b.instanceObject;
279 });
280 return eve::editing::applied<std::vector<PrefabOverrideRecord>>(std::move(result));
281}

283PrefabDependencyReport ScenePrefabService::inspectDependencies(
284 const PrefabAssetSnapshot& prefab, const PrefabResolver& resolver) const {
285 PrefabDependencyReport report;
286 const auto diagnostic = [&](DiagnosticCode code, const char* rule, DiagnosticSeverity severity,
287 std::string message) {
288 report.diagnostics.push_back(eve::editing::ruleDiagnostic(
289 code, RuleId(rule), severity, std::move(message)));
290 };
291 report.rootAsset = prefab.asset; report.rootRevision = prefab.revision;
292 if (!resolver || !validatePrefab(prefab).ok()) {
293 report.status = EditorStatus::Rejected;
294 diagnostic(DiagnosticCode::InvalidArgument, "editor.prefab.invalid-dependency-root",
295 DiagnosticSeverity::Error,
296 "Dependency inspection requires a valid prefab and resolver");
297 return report;
298 }
299 std::set<AssetGuid> visiting, visited;
300 std::function<bool(const PrefabAssetSnapshot&)> visit = [&](const PrefabAssetSnapshot& current) {
301 if (visiting.contains(current.asset)) {
302 diagnostic(DiagnosticCode::Conflict, "editor.prefab.dependency-cycle",
303 DiagnosticSeverity::Error,
304 "Nested prefab dependency cycle reaches: " + current.asset.value());
305 return false;
306 }
307 if (visited.contains(current.asset)) return true;
308 visiting.insert(current.asset);
309 bool valid = true;
310 for (const PrefabObjectRecord& object : current.objects) {
311 if (object.nestedPrefab.empty()) continue;
312 auto resolved = resolver(object.nestedPrefab);
313 if (!resolved.ok()) {
314 diagnostic(DiagnosticCode::NotFound, "editor.prefab.dependency-missing",
315 DiagnosticSeverity::Error,
316 "Nested prefab cannot be resolved: " + object.nestedPrefab.value());
317 valid = false; continue;
318 }
319 if (!validatePrefab(resolved.value()).ok()) {
320 diagnostic(DiagnosticCode::PreconditionViolation, "editor.prefab.dependency-invalid",
321 DiagnosticSeverity::Error,
322 "Nested prefab is invalid: " + object.nestedPrefab.value());
323 valid = false; continue;
324 }
325 if (resolved.value().revision != object.nestedRevision)
326 diagnostic(DiagnosticCode::Conflict, "editor.prefab.dependency-stale",
327 DiagnosticSeverity::Warning,
328 "Nested prefab revision pin is stale: " + object.nestedPrefab.value());
329 if (!visit(resolved.value())) valid = false;
330 }
331 visiting.erase(current.asset); visited.insert(current.asset);
332 if (current.asset != prefab.asset) report.dependencyOrder.push_back(current.asset);
333 return valid;
334 };
335 const bool valid = visit(prefab);
336 report.status = valid ? EditorStatus::Applied : EditorStatus::Rejected;
337 return report;
338}

340EditorResult<PrefabAssetSnapshot> ScenePrefabService::refreshNestedRevisions(
341 const PrefabAssetSnapshot& prefab, const PrefabResolver& resolver) const {
342 auto dependencies = inspectDependencies(prefab, resolver);
343 if (dependencies.status != EditorStatus::Applied)
344 return prefabError<PrefabAssetSnapshot>(dependencies.status, "editor.prefab.invalid-dependencies",
345 "Nested prefab dependencies are missing, invalid or cyclic");
346 PrefabAssetSnapshot result = prefab;
347 bool changed = false;
348 for (PrefabObjectRecord& object : result.objects) {
349 if (object.nestedPrefab.empty()) continue;
350 auto resolved = resolver(object.nestedPrefab);
351 if (!resolved.ok())
352 return prefabError<PrefabAssetSnapshot>(resolved.code(), "editor.prefab.dependency-missing",
353 "Nested prefab cannot be resolved");
354 if (object.nestedRevision != resolved.value().revision) {
355 object.nestedRevision = resolved.value().revision;
356 changed = true;
357 }
358 }
359 if (changed) ++result.revision;
360 return eve::editing::applied<PrefabAssetSnapshot>(std::move(result));
361}

377EditorResult<PrefabAssetSnapshot> ScenePrefabService::loadSnapshot(const EditorValue& snapshot) const {
378 const EditorValue* schemaValue = field(snapshot, "schemaVersion");
379 const EditorValue* assetValue = field(snapshot, "asset");
380 const EditorValue* revisionValue = field(snapshot, "revision");
381 const EditorValue* rootValue = field(snapshot, "rootSourceId");
382 const EditorValue* objectsValue = field(snapshot, "objects");
383 const auto* schema = schemaValue ? schemaValue->getIf<int64_t>() : nullptr;
384 const auto* asset = assetValue ? assetValue->getIf<std::string>() : nullptr;
385 const auto* revision = revisionValue ? revisionValue->getIf<int64_t>() : nullptr;
386 const auto* root = rootValue ? rootValue->getIf<std::string>() : nullptr;
387 const auto* objects = objectsValue ? objectsValue->getIf<EditorValue::Array>() : nullptr;
388 if (!schema || (*schema != 1 && *schema != 2) || !asset || !revision || *revision <= 0 || !root || !objects)
389 return prefabError<PrefabAssetSnapshot>(EditorStatus::Unsupported, "editor.prefab.invalid-snapshot",
390 "Prefab snapshot schema is invalid or unsupported");
391 PrefabAssetSnapshot prefab{AssetGuid(*asset), static_cast<Revision>(*revision), ObjectId(*root), {}};
392 for (const EditorValue& value : *objects) {
393 const EditorValue* idValue = field(value, "sourceId");
394 const EditorValue* parentValue = field(value, "parentSourceId");
395 const EditorValue* nameValue = field(value, "name");
396 const EditorValue* transform = field(value, "transform");
397 const EditorValue* nestedAssetValue = field(value, "nestedPrefab");
398 const EditorValue* nestedRevisionValue = field(value, "nestedRevision");
399 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
400 const auto* parentId = parentValue ? parentValue->getIf<std::string>() : nullptr;
401 const auto* name = nameValue ? nameValue->getIf<std::string>() : nullptr;
402 if (!id || !parentId || !name || !transform)
403 return prefabError<PrefabAssetSnapshot>(EditorStatus::Rejected, "editor.prefab.invalid-object",
404 "Prefab snapshot contains an invalid object");
405 auto parsed = parseTransform(*transform);
406 if (!parsed.ok())
407 return prefabError<PrefabAssetSnapshot>(EditorStatus::Rejected, "editor.prefab.invalid-transform",
408 "Prefab snapshot contains an invalid transform");
409 AssetGuid nestedAsset;
410 Revision nestedRevision = 0;
411 if (*schema == 2) {
412 const auto* nested = nestedAssetValue ? nestedAssetValue->getIf<std::string>() : nullptr;
413 const auto* pinned = nestedRevisionValue ? nestedRevisionValue->getIf<int64_t>() : nullptr;
414 if (!nested || !pinned || *pinned < 0)
415 return prefabError<PrefabAssetSnapshot>(EditorStatus::Rejected, "editor.prefab.invalid-nested-pin",
416 "Prefab snapshot contains an invalid nested revision pin");
417 nestedAsset = AssetGuid(*nested); nestedRevision = static_cast<Revision>(*pinned);
418 }
419 prefab.objects.push_back({ObjectId(*id), ObjectId(*parentId), *name, parsed.value(),
420 nestedAsset, nestedRevision});
421 }
422 auto valid = validatePrefab(prefab);
423 if (!valid.ok()) {
424 return EditorResult<PrefabAssetSnapshot>::failure(valid.status());
425 }
426 return eve::editing::applied<PrefabAssetSnapshot>(std::move(prefab));
427}
