18std::string text(const DamageNumberRequest& request, const DamageNumberStyle& style) {
19 if (request.kind == DamageNumberKind::Miss) return "MISS";
20 double value = request.amount;
21 std::string suffix;
22 if (style.compact && value >= 1000000.0) {
23 value /= 1000000.0;
24 suffix = "M";
25 } else if (style.compact && value >= 1000.0) {
26 value /= 1000.0;
27 suffix = "K";
28 }
29 std::ostringstream stream;
30 stream.imbue(std::locale::classic());
31 stream << (request.kind == DamageNumberKind::Heal ? "+" : "-") << std::fixed << std::setprecision(style.decimals)
32 << value << suffix;
33 if (request.kind == DamageNumberKind::Critical) stream << '!';
34 return stream.str();
35}

79Result<DamageNumberAdmission> DamageNumbers::spawn(const DamageNumberRequest& request) {
80 if (!validKind(request.kind) || !bounded(request.amount, 1.0e12) || request.amount < 0.0 ||
81 !bounded(request.x, 1.0e7) || !bounded(request.y, 1.0e7) || !bounded(request.velocityX, 10000.0) ||
82 request.group.size() > 128)
83 return Result<DamageNumberAdmission>::failure(invalid("Invalid popup request").status());
84 const auto style = impl_->styles[static_cast<std::size_t>(request.kind)];
85 if (!request.group.empty() && request.kind != DamageNumberKind::Miss) {
86 for (auto& p : impl_->items) {
87 const double dx = static_cast<double>(p.request.x) - request.x;
88 const double dy = static_cast<double>(p.request.y) - request.y;
89 if (p.request.kind == request.kind && p.request.group == request.group && p.style.mergeWindow > 0.0 &&
90 p.age < p.style.mergeWindow && dx * dx + dy * dy <= p.style.mergeRadius * p.style.mergeRadius &&
91 p.request.amount + request.amount <= 1.0e12) {
92 auto merged = p.request;
93 merged.amount += request.amount;
94 auto label = text(merged, p.style);
95 p.request = std::move(merged);
96 p.text = std::move(label);
97 return Result<DamageNumberAdmission>::success(DamageNumberAdmission::Merged);
98 }
99 }
100 }
101 // Explicit deterministic lanes reduce exact overlap without a per-frame pairwise solver.
102 Impl::Popup popup{request, style, text(request, style), 0.0,
103 static_cast<float>(static_cast<int>(impl_->sequence % 5) - 2) * 7.f};
104 auto admission = DamageNumberAdmission::Spawned;
105 if (impl_->items.size() == capacity()) {
106 impl_->items.erase(impl_->items.begin());
107 admission = DamageNumberAdmission::ReplacedOldest;
108 }
109 impl_->items.push_back(std::move(popup));
110 impl_->sequence = (impl_->sequence + 1) % 5;
111 return Result<DamageNumberAdmission>::success(admission);
112}
113Result<void> DamageNumbers::update(double seconds) {
114 if (!std::isfinite(seconds) || seconds < 0.0) return invalid("dt must be finite and nonnegative");
115 for (auto& p : impl_->items) p.age = std::min(static_cast<double>(p.style.lifetime), p.age + seconds);
116 std::erase_if(impl_->items, [](const auto& p) { return p.age >= p.style.lifetime; });
117 return Result<void>::success();

124DamageNumberSample DamageNumbers::Impl::sample(const Popup& p) {
125 const auto& s = p.style;
126 const float t = static_cast<float>(p.age);
127 const double fadeIn = s.fadeIn > 0.0 ? std::min(1.0, p.age / s.fadeIn) : 1.0;
128 const double fadeOut = s.fadeOut > 0.0 ? std::min(1.0, (s.lifetime - p.age) / s.fadeOut) : 1.0;
129 const float pulse = std::max(0.f, 1.f - t / 0.18f);
130 const float shake = std::sin(t * 110.f) * s.shake * std::max(0.f, 1.f - t / 0.24f);
131 return {p.text,
132 p.request.x + p.lane + p.request.velocityX * t + shake,
133 p.request.y + s.velocityY * t + 0.5f * s.gravity * t * t,
134 s.scale * (1.f + s.pop * pulse),
135 static_cast<float>(fadeIn * fadeOut),
136 s.r,
137 s.g,
138 s.b,
139 s.outline};
140}
