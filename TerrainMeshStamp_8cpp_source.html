30Result<int> TerrainMeshStampBuilder::bake(Heightmap& heights, Heightmap& coverage, double minX, double minZ,
31 double width, double depth, double feather) const {
32 if (&heights == &coverage || !validRaster(heights) || !validRaster(coverage) || heights.getWidth() < 2 ||
33 heights.getHeight() < 2 || heights.getWidth() != coverage.getWidth() ||
34 heights.getHeight() != coverage.getHeight() || heights.data().size() > 4U * 1024U * 1024U || indices_.empty() ||

39 const int columns = heights.getWidth(), rows = heights.getHeight();
40 const double dx = width / (columns - 1), dz = depth / (rows - 1);
41 if (dx <= 0 || dz <= 0 || minX + dx == minX || minZ + dz == minZ)
42 return invalid("terrain.meshStamp: grid spacing is not representable");
43 std::vector<float> candidate(heights.data().size(), 0.f), mask(candidate.size(), 0.f);
44 std::uint64_t work = 0;
45 int covered = 0;
46 struct Projection {
47 std::size_t a, b, c;
48 double determinant;
49 int x0, x1, z0, z1;
50 };
51 std::vector<Projection> projections;
52 projections.reserve(indices_.size() / 3);
53 for (std::size_t triangle = 0; triangle < indices_.size(); triangle += 3) {
54 const auto a = static_cast<std::size_t>(indices_[triangle]) * 3;
55 const auto b = static_cast<std::size_t>(indices_[triangle + 1]) * 3;
56 const auto c = static_cast<std::size_t>(indices_[triangle + 2]) * 3;
57 const double ax = positions_[a], az = positions_[a + 2];
58 const double bx = positions_[b], bz = positions_[b + 2];
59 const double cx = positions_[c], cz = positions_[c + 2];
60 const double determinant = (bz - cz) * (ax - cx) + (cx - bx) * (az - cz);
61 if (determinant == 0) continue;
62 const double left = std::max(minX, std::min({ax, bx, cx}));
63 const double right = std::min(minX + width, std::max({ax, bx, cx}));
64 const double bottom = std::max(minZ, std::min({az, bz, cz}));
65 const double top = std::min(minZ + depth, std::max({az, bz, cz}));
66 if (left > right || bottom > top) continue;
67 // Expand by one sample to avoid losing a shared edge through rounding.
68 const int x0 = static_cast<int>(std::clamp(std::floor((left - minX) / dx) - 1, 0.0, double(columns - 1)));
69 const int x1 = static_cast<int>(std::clamp(std::ceil((right - minX) / dx) + 1, 0.0, double(columns - 1)));
70 const int z0 = static_cast<int>(std::clamp(std::floor((bottom - minZ) / dz) - 1, 0.0, double(rows - 1)));
71 const int z1 = static_cast<int>(std::clamp(std::ceil((top - minZ) / dz) + 1, 0.0, double(rows - 1)));
72 work += std::uint64_t(x1 - x0 + 1) * std::uint64_t(z1 - z0 + 1);
73 if (work > 128U * 1024U * 1024U)
74 return invalid("terrain.meshStamp: projected triangle work exceeds 128M sample budget");
75 projections.push_back({a, b, c, determinant, x0, x1, z0, z1});
76 }
77 for (const auto& projection : projections) {
78 const auto [a, b, c, determinant, x0, x1, z0, z1] = projection;
79 const double ax = positions_[a], az = positions_[a + 2], bx = positions_[b], bz = positions_[b + 2],
80 cx = positions_[c], cz = positions_[c + 2];
81 for (int z = z0; z <= z1; ++z) {
82 for (int x = x0; x <= x1; ++x) {
83 const double px = minX + x * dx, pz = minZ + z * dz;
84 double wa = ((bz - cz) * (px - cx) + (cx - bx) * (pz - cz)) / determinant;
85 double wb = ((cz - az) * (px - cx) + (ax - cx) * (pz - cz)) / determinant;
86 double wc = 1.0 - wa - wb;
87 constexpr double tolerance = 1e-10;
88 if (wa < -tolerance || wb < -tolerance || wc < -tolerance) continue;
89 wa = std::max(0.0, wa);
90 wb = std::max(0.0, wb);
91 wc = std::max(0.0, wc);
92 const double y =
93 (wa * positions_[a + 1] + wb * positions_[b + 1] + wc * positions_[c + 1]) / (wa + wb + wc);
94 if (!isRepresentable(y)) return invalid("terrain.meshStamp: nonfinite projected height");
95 const auto index = static_cast<std::size_t>(z) * columns + x;
96 if (mask[index] == 0 || y > candidate[index]) candidate[index] = static_cast<float>(y);
97 if (mask[index] == 0) ++covered;
98 mask[index] = 1;
99 }
100 }
101 }
102 if (feather > 0) {
103 // Distance to uncovered samples or the rectangle edge in the bake coordinate system.
104 std::vector<double> distances(mask.size(), 0.0);
105 for (int z = 0; z < rows; ++z)
106 for (int x = 0; x < columns; ++x) {
107 const auto i = static_cast<std::size_t>(z) * columns + x;
108 if (mask[i] == 0) continue;
109 distances[i] = std::min({x * dx, (columns - 1 - x) * dx, z * dz, (rows - 1 - z) * dz, feather});
110 if (x > 0) distances[i] = std::min(distances[i], distances[i - 1] + dx);
111 if (z > 0) distances[i] = std::min(distances[i], distances[i - columns] + dz);
112 }
113 for (int z = rows - 1; z >= 0; --z)
114 for (int x = columns - 1; x >= 0; --x) {
115 const auto i = static_cast<std::size_t>(z) * columns + x;
116 if (x + 1 < columns) distances[i] = std::min(distances[i], distances[i + 1] + dx);
117 if (z + 1 < rows) distances[i] = std::min(distances[i], distances[i + columns] + dz);
118 const double t = std::clamp(distances[i] / feather, 0.0, 1.0);
119 mask[i] = static_cast<float>(t * t * (3.0 - 2.0 * t));
120 }
121 }
122 heights.data().swap(candidate);
123 coverage.data().swap(mask);
124 return Result<int>::success(covered);
125}
