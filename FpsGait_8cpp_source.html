14Result<void> FpsGaitProfile::validate() const {
15 if (!finitePositive(walkSpeed) || !finitePositive(sprintSpeed) || !finitePositive(crouchSpeed))
16 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
17 "fps gait speeds must be finite and positive", "speed"));
18 if (!finitePositive(standHeight) || !finitePositive(crouchHeight) || crouchHeight > standHeight)
19 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
20 "fps gait capsule heights must be positive with crouch <= stand",
21 "height"));
22 if (!finitePositive(standEyeHeight) || !finitePositive(crouchEyeHeight) || crouchEyeHeight > standEyeHeight)
23 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
24 "fps gait eye heights must be positive with crouch <= stand",
25 "eye"));
26 if (!finitePositive(capsuleRadius))
27 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
28 "fps gait capsule radius must be finite and positive", "radius"));
29 if (standEyeHeight > standHeight || crouchEyeHeight > crouchHeight)
30 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
31 "fps gait eye height must not exceed capsule height", "eye"));
32 return Result<void>::success();
33}

42Result<FpsGaitPose> FpsGait::evaluate(const FpsGaitInput& input) {
43 auto valid = profile_.validate();
44 if (!valid) return Result<FpsGaitPose>::failure(valid.status());
45
46 FpsGaitMode next = FpsGaitMode::Walk;
47 if (input.wantCrouch || (mode_ == FpsGaitMode::Crouch && !input.canStand))
48 next = FpsGaitMode::Crouch;
49 else if (input.wantSprint && input.grounded)
50 next = FpsGaitMode::Sprint;
51
52 mode_ = next;
53
54 FpsGaitPose pose;
55 pose.mode = mode_;
56 pose.capsuleRadius = profile_.capsuleRadius;
57 switch (mode_) {
58 case FpsGaitMode::Sprint:
59 pose.moveSpeed = profile_.sprintSpeed;
60 pose.capsuleHeight = profile_.standHeight;
61 pose.eyeHeight = profile_.standEyeHeight;
62 break;
63 case FpsGaitMode::Crouch:
64 pose.moveSpeed = profile_.crouchSpeed;
65 pose.capsuleHeight = profile_.crouchHeight;
66 pose.eyeHeight = profile_.crouchEyeHeight;
67 break;
68 case FpsGaitMode::Walk:
69 default:
70 pose.moveSpeed = profile_.walkSpeed;
71 pose.capsuleHeight = profile_.standHeight;
72 pose.eyeHeight = profile_.standEyeHeight;
73 break;
74 }
75 return Result<FpsGaitPose>::success(pose);
76}

78Result<FpsHeadBobOffset> FpsHeadBob::solve(const FpsHeadBobRequest& request) {
79 if (!std::isfinite(request.stridePhase) || !std::isfinite(request.moveSpeed) ||
80 !std::isfinite(request.referenceSpeed) || !std::isfinite(request.amplitude) ||
81 !std::isfinite(request.landDip))
82 return Result<FpsHeadBobOffset>::failure(Diagnostic::error(
83 DiagnosticCode::InvalidArgument, "fps head bob request must be finite", "request"));
84 if (request.referenceSpeed <= 0.0)
85 return Result<FpsHeadBobOffset>::failure(Diagnostic::error(
86 DiagnosticCode::InvalidArgument, "fps head bob reference speed must be positive", "referenceSpeed"));
87 if (request.amplitude < 0.0)
88 return Result<FpsHeadBobOffset>::failure(Diagnostic::error(
89 DiagnosticCode::InvalidArgument, "fps head bob amplitude must be non-negative", "amplitude"));
90
91 FpsHeadBobOffset offset;
92 offset.eyeY = request.landDip;
93 if (!request.enabled || !request.grounded) return Result<FpsHeadBobOffset>::success(offset);
94
95 const double speed = std::max(0.0, request.moveSpeed);
96 const double frac = std::min(1.0, speed / request.referenceSpeed);
97 if (frac <= 1e-4) return Result<FpsHeadBobOffset>::success(offset);
98
99 offset.eyeY += std::sin(request.stridePhase) * request.amplitude * frac;
100 return Result<FpsHeadBobOffset>::success(offset);
101}
