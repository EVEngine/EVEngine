45editing::Result<editing::CommandPlan> propertySet(editing::IEditableTarget& target,
46 const editing::CommandRequest& request) {
47 auto* properties = static_cast<editing::IPropertyProvider*>(
48 target.queryCapability(editing::CapabilityId("eve.editor.target.avatar-properties")));
49 if (!properties)
50 properties = static_cast<editing::IPropertyProvider*>(
51 target.queryCapability(editing::IPropertyProvider::editingCapabilityId()));
52 const auto* item = stringField(request.payload, "item");
53 const auto* type = stringField(request.payload, "type");
54 const auto* path = stringField(request.payload, "path");
55 const editing::Value* value = field(request.payload, "value");
56 if (!properties || !item || !type || !path || !value)
57 return error<editing::CommandPlan>(editing::Status::Rejected, "avatar.editing.property-payload",
58 "Avatar property requires a property target, item, type, path and value");
59 auto operation = properties->makeSet(selectionFor(target, *item, *type), editing::PropertyPath(*path), *value,
60 editing::PropertySetMode::Absolute);
61 if (!operation.ok())
62 return error<editing::CommandPlan>(operation.code(), "avatar.editing.property-operation",
63 "Avatar target rejected the property value");
64 editing::CommandPlan plan;
65 plan.operations.push_back(std::move(operation.value()));
66 plan.summary = editing::Value::Object{{"item", *item}, {"path", *path}};
67 return eve::editing::applied<editing::CommandPlan>(std::move(plan));
68}
