122void ensureDynamicRing(GpuMesh &gpu) {
123 if (gpu.dynamic) return;
124 gpu.dynamic = true;
125 gpu.dynamicWriteCount = 0;
126 gpu.cpuIndices.resize(gpu.indexCount);
127 if (gpu.indexCount > 0 && gpu.indices.buffer) {
128 void *ptr = gpu.indices.map();
129 if (gpu.indexType == vk::IndexType::eUint16) {
130 const auto *src = static_cast<const uint16_t *>(ptr);
131 for (uint32_t i = 0; i < gpu.indexCount; ++i)
132 gpu.cpuIndices[size_t(i)] = src[i];
133 } else {
134 std::memcpy(gpu.cpuIndices.data(), ptr,
135 size_t(gpu.indexCount) * sizeof(uint32_t));
136 }
137 gpu.indices.unmap();
138 }
139 gpu.indexType = vk::IndexType::eUint32;
140}

144void writeDynamicMesh(GpuMesh &gpu, const std::vector<MeshVertex> &verts, vkb::Device &device,
145 vkb::FrameSlot frame, const uint32_t *indices, int indexCount) {
146 const size_t slot = size_t(gpu.dynamicWriteCount % GpuMesh::kDynamicVertexCopies);
147 gpu.dynVertices[slot].allocate<MeshVertex>(frame, device, verts);
148 if (indices && indexCount > 0) {
149 gpu.cpuIndices.assign(indices, indices + indexCount);
150 gpu.indexCount = uint32_t(indexCount);
151 }
152 if (!gpu.cpuIndices.empty()) {
153 auto &ib = gpu.dynIndices[slot];
154 ib.allocate(frame, device, vk::BufferUsageFlagBits::eIndexBuffer,
155 vk::DeviceSize(gpu.cpuIndices.size()) * sizeof(uint32_t),
156 kHostVisibleCoherent);
157 ib.updateLocal(frame, gpu.cpuIndices.data(),
158 vk::DeviceSize(gpu.cpuIndices.size()) * sizeof(uint32_t));
159 }
160 ++gpu.dynamicWriteCount;
161}

196void assignMeshBounds(Mesh *mesh, const std::vector<MeshVertex> &verts) {
197 if (!mesh || verts.empty()) return;
198 glm::vec3 c(0.f);
199 for (const auto &v : verts) c += v.pos;
200 c /= float(verts.size());
201 mesh->boundsCx = c.x;
202 mesh->boundsCy = c.y;
203 mesh->boundsCz = c.z;
204 float r = 0.f;
205 for (const auto &v : verts) {
206 const glm::vec3 d = v.pos - c;
207 const float len = std::sqrt(d.x * d.x + d.y * d.y + d.z * d.z);
208 if (len > r) r = len;
209 }
210 // Same degenerate-mesh rule as Mesh::computeBounds: keep a tiny non-zero
211 // sphere so hasBounds() stays meaningful for culling.
212 mesh->boundsRadius = r > 0.f ? r : 1e-4f;
213}

295void appendBoxFilteredMip(std::vector<uint8_t> &out, const uint8_t *src, uint32_t srcW,
296 uint32_t srcH, uint32_t dstW, uint32_t dstH) {
297 const size_t base = out.size();
298 out.resize(base + size_t(rgba8MipBytes(dstW, dstH)));
299 uint8_t *dst = out.data() + base;
300 for (uint32_t y = 0; y < dstH; ++y) {
301 const uint32_t y0 = std::min(y * 2u, srcH - 1u);
302 const uint32_t y1 = std::min(y0 + 1u, srcH - 1u);
303 for (uint32_t x = 0; x < dstW; ++x) {
304 const uint32_t x0 = std::min(x * 2u, srcW - 1u);
305 const uint32_t x1 = std::min(x0 + 1u, srcW - 1u);
306 uint32_t acc[4] = {0, 0, 0, 0};
307 const uint32_t samples[4][2] = {{x0, y0}, {x1, y0}, {x0, y1}, {x1, y1}};
308 for (const auto &s : samples) {
309 const size_t i = (size_t(s[1]) * srcW + s[0]) * 4u;
310 acc[0] += src[i + 0];
311 acc[1] += src[i + 1];
312 acc[2] += src[i + 2];
313 acc[3] += src[i + 3];
314 }
315 const size_t o = (size_t(y) * dstW + x) * 4u;
316 dst[o + 0] = static_cast<uint8_t>((acc[0] + 2u) / 4u);
317 dst[o + 1] = static_cast<uint8_t>((acc[1] + 2u) / 4u);
318 dst[o + 2] = static_cast<uint8_t>((acc[2] + 2u) / 4u);
319 dst[o + 3] = static_cast<uint8_t>((acc[3] + 2u) / 4u);
320 }
321 }

325std::vector<uint8_t> buildMipChain2D(const uint8_t *rgba, uint32_t width, uint32_t height,
326 uint32_t mipLevels) {
327 std::vector<uint8_t> packed;
328 packed.reserve(size_t(width) * size_t(height) * 4u * 2u);
329 packed.insert(packed.end(), rgba, rgba + size_t(rgba8MipBytes(width, height)));
330
331 uint32_t srcW = width;
332 uint32_t srcH = height;
333 size_t srcOffset = 0;
334 for (uint32_t level = 1; level < mipLevels; ++level) {
335 const uint32_t dstW = std::max(srcW >> 1, 1u);
336 const uint32_t dstH = std::max(srcH >> 1, 1u);
337 const uint8_t *src = packed.data() + srcOffset;
338 srcOffset = packed.size();
339 appendBoxFilteredMip(packed, src, srcW, srcH, dstW, dstH);
340 srcW = dstW;
341 srcH = dstH;
342 }
343 return packed;
344}

350std::vector<uint8_t> buildMipChainCube(const uint8_t *rgbaFaces, uint32_t faceSize,
351 uint32_t mipLevels) {
352 const uint32_t faceBytes = rgba8MipBytes(faceSize, faceSize);
353 std::vector<std::vector<uint8_t>> faceChains(6);
354 for (uint32_t f = 0; f < 6; ++f) {
355 faceChains[f] = buildMipChain2D(rgbaFaces + size_t(f) * faceBytes, faceSize, faceSize,
356 mipLevels);
357 }
358
359 std::vector<uint8_t> packed;
360 uint32_t w = faceSize;
361 uint32_t h = faceSize;
362 size_t faceOffsets[6] = {0, 0, 0, 0, 0, 0};
363 for (uint32_t level = 0; level < mipLevels; ++level) {
364 const uint32_t levelBytes = rgba8MipBytes(w, h);
365 for (uint32_t f = 0; f < 6; ++f) {
366 const uint8_t *src = faceChains[f].data() + faceOffsets[f];
367 packed.insert(packed.end(), src, src + levelBytes);
368 faceOffsets[f] += levelBytes;
369 }
370 w = std::max(w >> 1, 1u);
371 h = std::max(h >> 1, 1u);
372 }
373 return packed;
374}
