123void ensureDynamicRing(GpuMesh &gpu) {
124 if (gpu.dynamic) return;
125 gpu.dynamic = true;
126 gpu.dynamicWriteCount = 0;
127 gpu.cpuIndices.resize(gpu.indexCount);
128 if (gpu.indexCount > 0 && gpu.indices.buffer) {
129 void *ptr = gpu.indices.map();
130 if (gpu.indexType == vk::IndexType::eUint16) {
131 const auto *src = static_cast<const uint16_t *>(ptr);
132 for (uint32_t i = 0; i < gpu.indexCount; ++i)
133 gpu.cpuIndices[size_t(i)] = src[i];
134 } else {
135 std::memcpy(gpu.cpuIndices.data(), ptr,
136 size_t(gpu.indexCount) * sizeof(uint32_t));
137 }
138 gpu.indices.unmap();
139 }
140 gpu.indexType = vk::IndexType::eUint32;
141}

145void writeDynamicMesh(GpuMesh &gpu, const std::vector<MeshVertex> &verts, vkb::Device &device,
146 vkb::FrameSlot frame, const uint32_t *indices, int indexCount) {
147 const size_t slot = size_t(gpu.dynamicWriteCount % GpuMesh::kDynamicVertexCopies);
148 gpu.vertexCount = uint32_t(verts.size());
149 gpu.dynVertices[slot].allocate<MeshVertex>(frame, device, verts);
150 if (indices && indexCount > 0) {
151 gpu.cpuIndices.assign(indices, indices + indexCount);
152 gpu.indexCount = uint32_t(indexCount);
153 }
154 if (!gpu.cpuIndices.empty()) {
155 auto &ib = gpu.dynIndices[slot];
156 ib.allocate(frame, device, vk::BufferUsageFlagBits::eIndexBuffer,
157 vk::DeviceSize(gpu.cpuIndices.size()) * sizeof(uint32_t),
158 kHostVisibleCoherent);
159 ib.updateLocal(frame, gpu.cpuIndices.data(),
160 vk::DeviceSize(gpu.cpuIndices.size()) * sizeof(uint32_t));
161 }
162 ++gpu.dynamicWriteCount;
163}

201void assignMeshBounds(Mesh *mesh, const std::vector<MeshVertex> &verts) {
202 if (!mesh || verts.empty()) return;
203 glm::vec3 c(0.f);
204 for (const auto &v : verts) c += v.pos;
205 c /= float(verts.size());
206 mesh->boundsCx = c.x;
207 mesh->boundsCy = c.y;
208 mesh->boundsCz = c.z;
209 float r = 0.f;
210 for (const auto &v : verts) {
211 const glm::vec3 d = v.pos - c;
212 const float len = std::sqrt(d.x * d.x + d.y * d.y + d.z * d.z);
213 if (len > r) r = len;
214 }
215 // Same degenerate-mesh rule as Mesh::computeBounds: keep a tiny non-zero
216 // sphere so hasBounds() stays meaningful for culling.
217 mesh->boundsRadius = r > 0.f ? r : 1e-4f;
218}

238 vk::ColorComponentFlagBits::eB | vk::ColorComponentFlagBits::eA;

314void appendBoxFilteredMip(std::vector<uint8_t> &out, const uint8_t *src, uint32_t srcW,
315 uint32_t srcH, uint32_t dstW, uint32_t dstH) {
316 const size_t base = out.size();
317 out.resize(base + size_t(rgba8MipBytes(dstW, dstH)));
318 uint8_t *dst = out.data() + base;
319 for (uint32_t y = 0; y < dstH; ++y) {
320 const uint32_t y0 = std::min(y * 2u, srcH - 1u);
321 const uint32_t y1 = std::min(y0 + 1u, srcH - 1u);
322 for (uint32_t x = 0; x < dstW; ++x) {
323 const uint32_t x0 = std::min(x * 2u, srcW - 1u);
324 const uint32_t x1 = std::min(x0 + 1u, srcW - 1u);
325 uint32_t acc[4] = {0, 0, 0, 0};
326 const uint32_t samples[4][2] = {{x0, y0}, {x1, y0}, {x0, y1}, {x1, y1}};
327 for (const auto &s : samples) {
328 const size_t i = (size_t(s[1]) * srcW + s[0]) * 4u;
329 acc[0] += src[i + 0];
330 acc[1] += src[i + 1];
331 acc[2] += src[i + 2];
332 acc[3] += src[i + 3];
333 }
334 const size_t o = (size_t(y) * dstW + x) * 4u;
335 dst[o + 0] = static_cast<uint8_t>((acc[0] + 2u) / 4u);
336 dst[o + 1] = static_cast<uint8_t>((acc[1] + 2u) / 4u);
337 dst[o + 2] = static_cast<uint8_t>((acc[2] + 2u) / 4u);
338 dst[o + 3] = static_cast<uint8_t>((acc[3] + 2u) / 4u);
339 }
340 }

344std::vector<uint8_t> buildMipChain2D(const uint8_t *rgba, uint32_t width, uint32_t height,
345 uint32_t mipLevels) {
346 std::vector<uint8_t> packed;
347 packed.reserve(size_t(width) * size_t(height) * 4u * 2u);
348 packed.insert(packed.end(), rgba, rgba + size_t(rgba8MipBytes(width, height)));
349
350 uint32_t srcW = width;
351 uint32_t srcH = height;
352 size_t srcOffset = 0;
353 for (uint32_t level = 1; level < mipLevels; ++level) {
354 const uint32_t dstW = std::max(srcW >> 1, 1u);
355 const uint32_t dstH = std::max(srcH >> 1, 1u);
356 const uint8_t *src = packed.data() + srcOffset;
357 srcOffset = packed.size();
358 appendBoxFilteredMip(packed, src, srcW, srcH, dstW, dstH);
359 srcW = dstW;
360 srcH = dstH;
361 }
362 return packed;
363}
