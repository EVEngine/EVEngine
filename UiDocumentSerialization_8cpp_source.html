72EditorResult<UiStyleValue> UiDocumentTarget::parseStyle(const EditorValue& value) {
73 UiStyleValue style;
74 double* outputs[]{&style.marginLeft, &style.marginTop, &style.marginRight, &style.marginBottom,
75 &style.paddingLeft, &style.paddingTop, &style.paddingRight, &style.paddingBottom,
76 &style.tintR, &style.tintG, &style.tintB, &style.tintA, &style.cornerRadius,
77 &style.gap, &style.flexGrow};
78 const char* keys[]{"marginLeft", "marginTop", "marginRight", "marginBottom", "paddingLeft",
79 "paddingTop", "paddingRight", "paddingBottom", "tintR", "tintG", "tintB",
80 "tintA", "cornerRadius", "gap", "flexGrow"};
81 for (std::size_t i = 0; i < std::size(outputs); ++i) {
82 const double* number = numberField(value, keys[i]);
83 if (!number || !std::isfinite(*number))
84 return serializationError<UiStyleValue>(EditorStatus::Rejected, "editor.ui.invalid-style",
85 "UI style numeric fields are missing or invalid");
86 *outputs[i] = *number;
87 }
88 const EditorValue* directionValue = field(value, "direction");
89 const EditorValue* alignValue = field(value, "align");
90 const EditorValue* justifyValue = field(value, "justify");
91 const auto* direction = directionValue ? directionValue->getIf<std::string>() : nullptr;
92 const auto* align = alignValue ? alignValue->getIf<std::string>() : nullptr;
93 const auto* justify = justifyValue ? justifyValue->getIf<std::string>() : nullptr;
94 if (!direction || !align || !justify)
95 return serializationError<UiStyleValue>(EditorStatus::Rejected, "editor.ui.invalid-style",
96 "UI flex style fields are missing");
97 style.direction = *direction; style.align = *align; style.justify = *justify;
98 const std::set<std::string> directions{"row", "column"};
99 const std::set<std::string> aligns{"start", "center", "end", "stretch"};
100 const std::set<std::string> justifies{"start", "center", "end", "space-between", "space-around"};
101 for (std::size_t i = 0; i < std::size(outputs); ++i)
102 if (*outputs[i] < 0.0 || (i >= 8 && i <= 11 && *outputs[i] > 1.0))
103 return serializationError<UiStyleValue>(EditorStatus::Rejected, "editor.ui.invalid-style-range",
104 "UI box values must be non-negative and tint normalized");
105 if (!directions.contains(style.direction) || !aligns.contains(style.align) ||
106 !justifies.contains(style.justify))
107 return serializationError<UiStyleValue>(EditorStatus::Rejected, "editor.ui.invalid-flex-style",
108 "UI flex direction, alignment or justification is invalid");
109 return eve::editing::applied<UiStyleValue>(std::move(style));
110}

234EditorResult<void> UiDocumentTarget::loadSnapshot(const EditorValue& snapshot) {
235 const EditorValue* versionValue = field(snapshot, "schemaVersion");
236 const EditorValue* widgetsValue = field(snapshot, "widgets");
237 const auto* version = versionValue ? versionValue->getIf<int64_t>() : nullptr;
238 const auto* widgets = widgetsValue ? widgetsValue->getIf<EditorValue::Array>() : nullptr;
239 if (!version || (*version != 1 && *version != 2) || !widgets)
240 return serializationError<void>(EditorStatus::Rejected, "editor.ui.snapshot-format",
241 "UI snapshot requires schemaVersion 1 or 2 and a widget array");
242 std::map<ObjectId, UiWidgetSnapshot> candidate;
243 for (const EditorValue& value : *widgets) {
244 auto parsed = parseWidget(value);
245 if (!parsed.ok())
246 return serializationError<void>(EditorStatus::Rejected, "editor.ui.snapshot-widget",
247 "UI snapshot contains an invalid widget");
248 const auto& widget = parsed.value();
249 if (!candidate.emplace(widget.id, widget).second)
250 return serializationError<void>(EditorStatus::Conflict, "editor.ui.snapshot-duplicate",
251 "UI snapshot contains duplicate widget ids");
252 }
253 for (const auto& [id, widget] : candidate) {
254 if ((!widget.parent.empty() && !candidate.contains(widget.parent)) ||
255 wouldCycle(id, widget.parent, candidate))
256 return serializationError<void>(EditorStatus::Rejected, "editor.ui.snapshot-hierarchy",
257 "UI snapshot hierarchy is missing a parent or contains a cycle");
258 }
259 widgets_ = std::move(candidate);
260 ++revision_;
261 dirty_.clear();
262 return eve::editing::applied<void>();
263}
