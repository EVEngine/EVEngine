67editing::Result<void> registerEditingCommands(editing::IEditingCommandRegistry& registry) {
68 return registry.registerPlannedCommand(
69 actionPropertySetCommandDescriptor(),
70 [](editing::IEditableTarget& target, const editing::CommandRequest& request) {
71 auto* properties = static_cast<editing::IPropertyProvider*>(
72 target.queryCapability(editor::ActionTimelineTarget::propertyCapabilityId()));
73 const editing::Value* pathValue = field(request.payload, "path");
74 const auto* path = pathValue ? pathValue->getIf<std::string>() : nullptr;
75 const editing::Value* value = field(request.payload, "value");
76 if (!properties || !path || !value)
77 return eve::editing::failed<editing::CommandPlan>(
78 editing::Status::Rejected, editing::RuleId("action.editing.property-payload"),
79 "Action property requires a property target, path and value");
80 auto operation = properties->makeSet(selectionFor(target), editing::PropertyPath(*path), *value,
81 editing::PropertySetMode::Absolute);
82 if (!operation.ok())
83 return eve::editing::failed<editing::CommandPlan>(operation.code(),
84 editing::RuleId("action.editing.property-operation"),
85 "Action timeline target rejected the property value");
86 editing::CommandPlan plan;
87 plan.operations.push_back(std::move(operation.value()));
88 plan.summary = editing::Value::Object{{"path", *path}};
89 return eve::editing::applied<editing::CommandPlan>(std::move(plan));
90 });
