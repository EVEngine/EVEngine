67Result<Value> WorldSnapshotAccess::captureJoints2D(const World& world) {
68 if (static_cast<std::size_t>(world.raw()->GetJointCount()) != world.joints_.size())
69 return Result<Value>::failure(Diagnostic::error(
70 DiagnosticCode::Unsupported, "Snapshot cannot capture unmanaged backend joints", "physics.world.snapshot"));
71 std::vector<Joint2D*> sorted(world.joints_.begin(), world.joints_.end());
72 std::sort(sorted.begin(), sorted.end(), [](auto* a, auto* b) { return a->getId() < b->getId(); });
73 Value::Array output;
74 for (const auto* wrapper : sorted) {
75 auto& raw = *wrapper->joint_;
76 Object data;
77 switch (wrapper->kind_) {
78 case Joint2DKind::Revolute: {
79 const auto& j = static_cast<const b2RevoluteJoint&>(raw);
80 anchors(data, j.GetLocalAnchorA(), j.GetLocalAnchorB());
81 data.emplace("referenceAngle", Value(j.GetReferenceAngle()));
82 limits(data, j);
83 motor(data, j.IsMotorEnabled(), j.GetMotorSpeed(), j.GetMaxMotorTorque());
84 break;
85 }
86 case Joint2DKind::Prismatic: {
87 const auto& j = static_cast<const b2PrismaticJoint&>(raw);
88 anchors(data, j.GetLocalAnchorA(), j.GetLocalAnchorB());
89 data.emplace("referenceAngle", Value(j.GetReferenceAngle()));
90 data.emplace("axisX", Value(j.GetLocalAxisA().x));
91 data.emplace("axisY", Value(j.GetLocalAxisA().y));
92 limits(data, j);
93 motor(data, j.IsMotorEnabled(), j.GetMotorSpeed(), j.GetMaxMotorForce());
94 break;
95 }
96 case Joint2DKind::Distance: {
97 const auto& j = static_cast<const b2DistanceJoint&>(raw);
98 anchors(data, j.GetLocalAnchorA(), j.GetLocalAnchorB());
99 data.emplace("length", Value(j.GetLength()));
100 spring(data, j.GetFrequency(), j.GetDampingRatio());
101 break;
102 }
103 case Joint2DKind::Wheel: {
104 const auto& j = static_cast<const b2WheelJoint&>(raw);
105 anchors(data, j.GetLocalAnchorA(), j.GetLocalAnchorB());
106 data.emplace("axisX", Value(j.GetLocalAxisA().x));
107 data.emplace("axisY", Value(j.GetLocalAxisA().y));
108 spring(data, j.GetSpringFrequencyHz(), j.GetSpringDampingRatio());
109 motor(data, j.IsMotorEnabled(), j.GetMotorSpeed(), j.GetMaxMotorTorque());
110 break;
111 }
112 case Joint2DKind::Weld: {
113 const auto& j = static_cast<const b2WeldJoint&>(raw);
114 anchors(data, j.GetLocalAnchorA(), j.GetLocalAnchorB());
115 data.emplace("referenceAngle", Value(j.GetReferenceAngle()));
116 spring(data, j.GetFrequency(), j.GetDampingRatio());
117 break;
118 }
119 case Joint2DKind::Pulley: {
120 const auto& j = static_cast<const b2PulleyJoint&>(raw);
121 anchors(data, {wrapper->pulleyLocalA_.x, wrapper->pulleyLocalA_.y},
122 {wrapper->pulleyLocalB_.x, wrapper->pulleyLocalB_.y});
123 data.emplace("groundAx", Value(j.GetGroundAnchorA().x));
124 data.emplace("groundAy", Value(j.GetGroundAnchorA().y));
125 data.emplace("groundBx", Value(j.GetGroundAnchorB().x));
126 data.emplace("groundBy", Value(j.GetGroundAnchorB().y));
127 data.emplace("lengthA", Value(j.GetLengthA()));
128 data.emplace("lengthB", Value(j.GetLengthB()));
129 data.emplace("ratio", Value(j.GetRatio()));
130 break;
131 }
132 case Joint2DKind::Motor: {
133 const auto& j = static_cast<const b2MotorJoint&>(raw);
134 data.emplace("offsetX", Value(j.GetLinearOffset().x));
135 data.emplace("offsetY", Value(j.GetLinearOffset().y));
136 data.emplace("angle", Value(j.GetAngularOffset()));
137 data.emplace("force", Value(j.GetMaxForce()));
138 data.emplace("torque", Value(j.GetMaxTorque()));
139 data.emplace("correction", Value(j.GetCorrectionFactor()));
140 break;
141 }
142 case Joint2DKind::Gear: {
143 const auto& j = static_cast<const b2GearJoint&>(raw);
144 int first = -1, second = -1;
145 for (const auto* source : sorted) {
146 if (source->handle_ == wrapper->firstSource_) first = source->getId();
147 if (source->handle_ == wrapper->secondSource_) second = source->getId();
148 }
149 if (first < 0 || second < 0) return malformed<Value>("Gear source is missing");
150 data.emplace("first", Value(std::to_string(first)));
151 data.emplace("second", Value(std::to_string(second)));
152 data.emplace("ratio", Value(j.GetRatio()));
153 Value::Array poses;
154 for (float value : wrapper->gearReferencePoses_) poses.emplace_back(value);
155 data.emplace("referencePoses", Value(std::move(poses)));
156 break;
157 }
158 }
159 output.emplace_back(Object{{"id", Value(std::to_string(wrapper->getId()))},
160 {"kind", Value(std::to_string(static_cast<int>(wrapper->kind_)))},
161 {"bodyA", Value(std::to_string(bodyId(*raw.GetBodyA())))},
162 {"bodyB", Value(std::to_string(bodyId(*raw.GetBodyB())))},
163 {"collideConnected", Value(raw.GetCollideConnected())},
164 {"data", Value(std::move(data))}});
165 }
166 return Result<Value>::success(Value(std::move(output)));
167}
