27Result<MeshBuild> compact(const MeshBuild& source, const RoadGeometrySourceSpan& span, std::size_t& used,
28 std::size_t cap) {
29 MeshBuild mesh;
30 std::unordered_map<std::uint32_t, int> vertices;
31 std::vector<float> colors;
32 const auto& indices = source.indices();
33 const std::size_t end = static_cast<std::size_t>(span.firstTriangle) + span.triangleCount;
34 for (std::size_t t = span.firstTriangle; t < end; ++t) {
35 if (cap - used < 3) return fail<MeshBuild>("prepared index budget exhausted");
36 used += 3;
37 mesh.setActiveGroup(source.getGroupName(source.getTriangleGroup(static_cast<int>(t))));
38 int out[3];
39 for (int j = 0; j < 3; ++j) {
40 auto v = indices[t * 3 + j];
41 auto found = vertices.find(v);
42 if (found != vertices.end()) {
43 out[j] = found->second;
44 continue;
45 }
46 const std::size_t cost = source.hasVertexColors() ? 12 : 8;
47 if (cost > cap - used) return fail<MeshBuild>("prepared vertex budget exhausted");
48 used += cost;
49 auto i = static_cast<std::size_t>(v);
50 int index = mesh.getVertexCount();
51 mesh.addVertex(source.positions()[i * 3], source.positions()[i * 3 + 1], source.positions()[i * 3 + 2],
52 source.normals()[i * 3], source.normals()[i * 3 + 1], source.normals()[i * 3 + 2],
53 source.uvs()[i * 2], source.uvs()[i * 2 + 1]);
54 vertices.emplace(v, index);
55 out[j] = index;
56 if (source.hasVertexColors())
57 colors.insert(colors.end(), source.colors().begin() + i * 4, source.colors().begin() + i * 4 + 4);
58 }
59 mesh.addTriangle(out[0], out[1], out[2]);
60 }
61 auto colored = mesh.setVertexColors(std::move(colors));
62 if (!colored.ok()) return Result<MeshBuild>::failure(colored.status());
63 return Result<MeshBuild>::success(std::move(mesh));
64}

84 const RoadTerrainPolicy& sharedPolicy,

93 auto sources = PreparedRoadGeometrySources::prepare(baked.value().mesh);
94 if (!sources.ok()) return Result<RoadTerrainPlan>::failure(sources.status());
95 std::unordered_map<std::uint32_t, RoadTerrainPolicy> policies;
96 for (const auto& edge : network.edges()) policies.emplace(edge.id, edge.style.terrainPolicy);
97 auto impl = std::make_unique<Impl>();
98 std::size_t used = 0;
99 for (const auto& span : sources.value().spans()) {
100 RoadTerrainPolicy policy = sharedPolicy;
101 if (span.kind == RoadGeometrySourceKind::Edge) {
102 auto edge = policies.find(span.id);
103 if (edge == policies.end()) return fail<RoadTerrainPlan>("source edge has no author policy");

106 if (!policy.enabled) continue;

109 if (!mesh.ok()) return Result<RoadTerrainPlan>::failure(mesh.status());
110 impl->items.push_back({std::move(mesh).takeValue(), policy});
111 }
112 return Result<RoadTerrainPlan>::success(RoadTerrainPlan(std::move(impl)));
113}
