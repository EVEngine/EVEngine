42Result<void> ActionNotifyRegistry::registerDescriptor(ActionNotifyDescriptor descriptor) {
43 if (!validType(descriptor.type)) return failure(DiagnosticCode::InvalidArgument, "Notify type is invalid", "type");
44 if (descriptor.displayName.empty())
45 return failure(DiagnosticCode::InvalidArgument, "Notify display name is empty", "displayName");
46 if (descriptor.category.empty())
47 return failure(DiagnosticCode::InvalidArgument, "Notify category is empty", "category");
48 for (const auto& field : descriptor.requiredPayloadFields)
49 if (field.empty())
50 return failure(DiagnosticCode::InvalidArgument, "Required payload field is empty", "requiredPayloadFields");
51 const std::string key = descriptor.type;
52 if (descriptors_.contains(key))
53 return failure(DiagnosticCode::AlreadyExists, "Notify descriptor is already registered", key);
54 descriptors_.emplace(key, std::move(descriptor));
55 return Result<void>::success(Status::success(StatusCode::Applied));
56}

92Result<void> ActionNotifyRegistry::validate(const ActionTimelineEvent& event) const {
93 auto found = descriptors_.find(event.type.format());
94 if (found == descriptors_.end())
95 return failure(DiagnosticCode::NotFound, "Timeline event notify type is not registered", event.type.format());
96 const bool instant = event.kind == ActionTimelineEventKind::Notify;
97 if ((found->second.shape == ActionNotifyShape::Instant) != instant)
98 return failure(DiagnosticCode::InvalidArgument, "Timeline event boundary does not match notify shape",
99 event.type.format());
100 for (const auto& field : found->second.requiredPayloadFields)
101 if (!event.payload.contains(field))
102 return failure(DiagnosticCode::InvalidArgument, "Timeline event is missing required payload field", field);
103 return Result<void>::success();
104}

106Result<void> ActionNotifyRegistry::dispatch(const ActionTimelineEvent& event, const ActionNotifyContext& context) {
107 auto valid = validate(event);
108 if (!valid) return valid;
109 const auto handler = handlers_.find(event.type.format());
110 if (handler == handlers_.end())
111 return failure(DiagnosticCode::NotFound, "No runtime handler is registered for notify type",
112 event.type.format());
113 return handler->second->handle(event, context);
114}
