110class ActionParameterCurveHandler final : public IActionNotifyHandler {
111public:
112 Result<void> handle(const ActionTimelineEvent& event, const ActionNotifyContext& context) override {
113 auto binding = ActionParameterCurveBinding::fromPayload(event.payload);
114 if (!binding) return Result<void>::failure(binding.status());
115 const ActiveKey key{context.executionId, event.itemId.format()};
116 if (event.kind == ActionTimelineEventKind::StateEnter)
117 return Result<void>::success(Status::success(StatusCode::NoOp));
118 if (event.kind != ActionTimelineEventKind::StateExit)

120 const auto found = active_.find(key);
121 if (found == active_.end())
122 return Result<void>::success(Status::success(StatusCode::NoOp));
123 ActionParameterSample sample{ActionParameterPhase::End, context.executionId, event.itemId,
124 found->second.target, found->second.operation, found->second.value,
125 context.preview};
126 auto applied = dispatch(sample);
127 if (!applied) return applied;
128 active_.erase(found);
129 return applied;
130 }

132 Result<void> update(const ActionActiveBlock& block, const ActionNotifyContext& context) override {
133 auto binding = ActionParameterCurveBinding::fromPayload(block.payload);
134 if (!binding) return Result<void>::failure(binding.status());
135 if (block.duration <= Duration::zero())
136 return failure(DiagnosticCode::InvalidArgument, "parameter curve block duration must be positive",
137 "duration");
138 const double progress = std::clamp(block.localTime.seconds() / block.duration.seconds(), 0.0, 1.0);
139 const double value = binding.value().sample(progress);
140 const ActiveKey key{context.executionId, block.itemId.format()};
141 const auto found = active_.find(key);
142 const auto phase = found == active_.end() ? ActionParameterPhase::Begin : ActionParameterPhase::Update;
143 ActionParameterSample sample{phase, context.executionId, block.itemId, binding.value().target,
144 binding.value().operation, value, context.preview};
145 auto applied = dispatch(sample);
146 if (!applied) return applied;
147 active_[key] = ActiveParameter{binding.value().target, binding.value().operation, value};
148 return applied;
149 }

226 if (!providers) return;

233Result<void> ActionNotifyRegistry::registerDescriptor(ActionNotifyDescriptor descriptor) {
234 if (!validType(descriptor.type)) return failure(DiagnosticCode::InvalidArgument, "Notify type is invalid", "type");
235 if (descriptor.displayName.empty())
236 return failure(DiagnosticCode::InvalidArgument, "Notify display name is empty", "displayName");
237 if (descriptor.category.empty())
238 return failure(DiagnosticCode::InvalidArgument, "Notify category is empty", "category");
239 for (const auto& field : descriptor.requiredPayloadFields)
240 if (field.empty())
241 return failure(DiagnosticCode::InvalidArgument, "Required payload field is empty", "requiredPayloadFields");
242 const std::string key = descriptor.type;
243 if (descriptors_.contains(key))
244 return failure(DiagnosticCode::AlreadyExists, "Notify descriptor is already registered", key);
245 descriptors_.emplace(key, std::move(descriptor));
246 return Result<void>::success(Status::success(StatusCode::Applied));
247}

257 return Result<void>::success(Status::success(StatusCode::Applied));

285Result<void> ActionNotifyRegistry::validate(const ActionTimelineEvent& event) const {
286 auto found = descriptors_.find(event.type.format());
287 if (found == descriptors_.end())
288 return failure(DiagnosticCode::NotFound, "Timeline event notify type is not registered", event.type.format());
289 const bool instant = event.kind == ActionTimelineEventKind::Notify;
290 if ((found->second.shape == ActionNotifyShape::Instant) != instant)
291 return failure(DiagnosticCode::InvalidArgument, "Timeline event boundary does not match notify shape",
292 event.type.format());
293 for (const auto& field : found->second.requiredPayloadFields)
294 if (!event.payload.contains(field))
295 return failure(DiagnosticCode::InvalidArgument, "Timeline event is missing required payload field", field);
296 if (event.type.format() == "gameplay:prefab-spawn") {
297 auto prefab = ActionPrefabSpawnBinding::fromPayload(event.payload);
298 if (!prefab) return Result<void>::failure(prefab.status());
299 }
300 if (event.type.format() == "presentation:audio" || event.type.format() == "presentation:audio-state") {
301 const auto shape = event.kind == ActionTimelineEventKind::Notify ? ActionAudioShape::Instant
302 : ActionAudioShape::State;
303 auto audio = ActionAudioBinding::fromPayload(event.payload, shape);
304 if (!audio) return Result<void>::failure(audio.status());
305 }
306 if (event.type.format() == "presentation:vfx" || event.type.format() == "presentation:vfx-state") {
307 const auto shape = event.kind == ActionTimelineEventKind::Notify ? ActionVfxShape::Instant
308 : ActionVfxShape::State;
309 auto vfx = ActionVfxBinding::fromPayload(event.payload, shape);
310 if (!vfx) return Result<void>::failure(vfx.status());
311 }
312 if (event.type.format() == "presentation:parameter-curve") {
313 auto curve = ActionParameterCurveBinding::fromPayload(event.payload);
314 if (!curve) return Result<void>::failure(curve.status());
315 }
316 if (event.type.format() == "presentation:camera") {
317 auto camera = ActionCameraCueBinding::fromPayload(event.payload);
318 if (!camera) return Result<void>::failure(camera.status());
319 }
320 if (event.type.format() == "combat:damage") {
321 auto damage = ActionDamageBinding::fromPayload(event.payload);
322 if (!damage) return Result<void>::failure(damage.status());
323 }
324 if (event.type.format() == "gameplay:event") {
325 auto gameplayEvent = ActionGameplayEventBinding::fromPayload(event.payload);
326 if (!gameplayEvent) return Result<void>::failure(gameplayEvent.status());
327 }
328 if (event.type.format() == "collision:ignore-window" || event.type.format() == "combat:hitbox-window" ||
329 event.type.format() == "combat:invulnerability-window" || event.type.format() == "input:combo-window") {
330 auto window = ActionStateWindowBinding::fromPayload(event.type.format(), event.payload);
331 if (!window) return Result<void>::failure(window.status());
332 }
333 return Result<void>::success();
334}

371Result<void> ActionNotifyRegistry::advanceHandlers(const ActionNotifyContext& context) {
372 std::set<IActionNotifyHandler*> advanced;
373 StatusCode outcome = StatusCode::NoOp;
374 for (const auto& [type, handler] : handlers_) {
375 (void)type;
376 if (!advanced.insert(handler.get()).second) continue;
377 auto result = handler->advance(context);
378 if (!result) return result;
379 if (result.status().code() == StatusCode::Applied) outcome = StatusCode::Applied;
380 }
381 return Result<void>::success(Status::success(outcome));
382}
