108 if (signedSpeed < -1e-3f) {

194float Mechanism3D::axisSpinSpeed() const {
195 if (!isValid() || !drive_) return 0.f;
196 Body3D *bodyA = world_ ? world_->findBodyById(drive_->getBodyAId()) : nullptr;
197 Body3D *bodyB = world_ ? world_->findBodyById(drive_->getBodyBId()) : nullptr;
198 if (!bodyA || !bodyB || !bodyA->isValid() || !bodyB->isValid()) return 0.f;
199 const float wx = bodyB->getAngularVelocityX() - bodyA->getAngularVelocityX();
200 const float wy = bodyB->getAngularVelocityY() - bodyA->getAngularVelocityY();
201 const float wz = bodyB->getAngularVelocityZ() - bodyA->getAngularVelocityZ();
202 return wx * axisX_ + wy * axisY_ + wz * axisZ_;
203}
