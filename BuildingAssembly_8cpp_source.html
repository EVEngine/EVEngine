18Result<EdgeCurveGroup> transformedCurve(const EdgeCurveGroup& source, AssemblyPose from, AssemblyPose to,
19 const std::map<int, int>& remap, EdgeCurveGroupId id) {
20 if (!source.surfaceSamples.empty())
21 return fail<EdgeCurveGroup>("Surface-attached curves must be recaptured on their provider");
22 auto out = source;
23 out.id = id;
24 out.level = source.level - from.level + to.level;
25 int q = (to.quarterTurns - from.quarterTurns + 4) % 4;
26 for (auto& p : out.controlPoints) {
27 float x = p.x - from.x, y = p.y - from.y;
28 for (int i = 0; i < q; ++i) {
29 float old = x;
30 x = y;
31 y = -old;
32 }
33 p.x = x + to.x;
34 p.y = y + to.y;
35 }
36 for (int& member : out.instanceIds) {
37 auto it = remap.find(member);
38 if (it == remap.end()) return fail<EdgeCurveGroup>("Capture requires complete curve membership");
39 member = it->second;
40 }
41 return Result<EdgeCurveGroup>::success(std::move(out));
42}

45 AssemblyPose origin) {
46 auto grid = checkGrid(w);
47 if (!grid.ok() || id.isNil() || members.empty())
48 return fail<BuildingBlueprint>("Capture requires a valid grid, asset identity and selection");
49 BuildingBlueprint b;
50 b.id = id;
51 b.cellSize = w.getCellSize();
52 b.floorHeight = w.getFloorHeight();
53 b.definitions = Value::Object{};
54 std::map<int, int> remap;
55 int key = 0;
56 for (int member : members)

70 local.worldY -= w.getOriginY();
71 local.garrison.clear();
72 local.garrisonRevision = Revision::zero();
73 // Arbitrary properties may contain foreign entity ids; only definition defaults transfer.
74 local.props.clear();

79 }
80 for (auto curveId : w.edgeCurveGroupIds()) {
81 const auto& curve = w.edgeCurveGroups_.at(curveId.value);
82 if (std::none_of(curve.instanceIds.begin(), curve.instanceIds.end(),
83 [&](int member) { return remap.contains(member); }))
84 continue;

86 if (!transformed.ok()) return fail<BuildingBlueprint>("Selection contains a partial or surface-attached curve");

91 }

112 AssemblyId id, AssemblyPose pose) {

135 p.edgeCurveGroupId = {};
136 remap.emplace(part.local.instanceId, p.instanceId);
137 a.members.emplace(part.id.value, p.instanceId);
138 a.nextPartKey = std::max(a.nextPartKey, part.id.value + 1);
139 candidate->instanceOrder_.push_back(p.instanceId);
140 candidate->buildings_.emplace(p.instanceId, std::move(p));
141 }
142 for (const auto& c : b.curves) {
143 auto curve = transformedCurve(c.group, {}, pose, remap, {candidate->nextEdgeCurveGroupId_++});
144 if (!curve.ok()) return fail<PreparedAssembly>("Invalid transformed curve");
145 const auto curveId = curve.value().id.value;
146 candidate->edgeCurveGroups_.emplace(curveId, std::move(curve).takeValue());
147 }
148 candidate->assemblyState_->records.emplace(id, std::move(a));
149 candidate->assemblyState_->handles[id] =
150 eve::RuntimeHandle<AssemblyHandleTag>(candidate->assemblyState_->nextSlot++, 1);

153Result<PreparedAssembly> AssemblyService::prepareMove(const PlacementWorld& w, AssemblyId id, AssemblyPose pose) {
154 auto it = w.assemblyState_->records.find(id);
155 if (it == w.assemblyState_->records.end())
156 return fail<PreparedAssembly>("House not found", DiagnosticCode::NotFound);
157 auto candidate = w.cloneState();
158 std::map<int, int> remap;

185Result<PreparedAssembly> AssemblyService::prepareRemodel(const PlacementWorld& w, AssemblyId id,
186 std::span<const PlacedBuilding> members) {
187 auto old = assembly(w, id);
188 if (!old.ok()) return Result<PreparedAssembly>::failure(old.status());
189 if (members.empty()) return prepareRemoval(w, id);
190 auto candidate = w.cloneState();
191 auto a = old.value();
192 std::map<int, uint32_t> keys;
193 uint32_t nextKey = a.nextPartKey;
194 for (const auto& [key, member] : a.members) {
195 keys[member] = key;
196 candidate->buildings_.erase(member);
197 }
198 std::erase_if(candidate->instanceOrder_, [&](int member) { return keys.contains(member); });
199 a.members.clear();
200 std::set<int> seen;
201 for (auto p : members) {
202 uint32_t key = 0;
203 if (p.instanceId) {
204 auto it = keys.find(p.instanceId);
205 if (it == keys.end() || !seen.insert(p.instanceId).second)
206 return fail<PreparedAssembly>("Remodel cannot steal or duplicate a member");
207 key = it->second;
208 } else {
209 if (candidate->instanceCounter_ == std::numeric_limits<int>::max() ||
210 nextKey >= uint32_t(std::numeric_limits<int>::max()))
211 return fail<PreparedAssembly>("Member identity capacity exhausted");
212 p.instanceId = PlacementSystem::nextInstanceId(*candidate);
213 key = nextKey++;
214 }
215 a.members.emplace(key, p.instanceId);
216 candidate->instanceOrder_.push_back(p.instanceId);
217 candidate->buildings_.emplace(p.instanceId, std::move(p));
218 }
219 // Curves remain authoritative only if all exact addresses survive; partial edits dissolve them.
220 std::erase_if(candidate->edgeCurveGroups_, [&](const auto& entry) {
221 for (int member : entry.second.instanceIds) {
222 auto current = candidate->buildings_.find(member);
223 auto previous = w.buildings_.find(member);
224 if (current == candidate->buildings_.end() || previous == w.buildings_.end() ||
225 current->second.edge != previous->second.edge ||
226 current->second.buildingId != previous->second.buildingId ||
227 current->second.level != previous->second.level)
228 return true;
229 }
230 return false;
231 });
232 a.nextPartKey = nextKey;
233 candidate->assemblyState_->records[id] = std::move(a);
234 return prepareState(w, std::move(candidate), id);
235}
236Result<AssemblyReceipt> AssemblyService::commit(PlacementWorld& w, PreparedAssembly token) {
237 if (!token.data_ || token.data_->epoch != w.assemblyState_->epoch)
238 return fail<AssemblyReceipt>("Expired prepared house operation", DiagnosticCode::Conflict);
239 auto& t = *token.data_;
240 auto current = snapshotWorld(w);
241 auto defs = checkDefinitions(t.definitions);
242 const bool currentOk = current.ok();
243 const bool definitionsOk = defs.ok();
244 if (!currentOk || current.value() != t.before || !definitionsOk || admissionStamp(w) != t.admission)
245 return fail<AssemblyReceipt>("World or definitions changed since preparation", DiagnosticCode::Conflict);
246 auto valid = validateFinal(*t.candidate);
247 if (!valid.ok()) return Result<AssemblyReceipt>::failure(valid.status());
248 auto after = snapshotWorld(*t.candidate);
249 if (!after.ok()) return fail<AssemblyReceipt>("Cannot capture commit receipt");
250 AssemblyReceipt receipt{t.id, std::move(t.before), std::move(after).takeValue()};
251 // Invalidate other prepared tokens after every commit; existing assembly handles are
252 // intentionally invalidated too, and callers resolve persistent identities again.
253 t.candidate->assemblyState_->epoch = nextEpoch();
254 w.swapState(*t.candidate);
255 if (w.publishEvents_) {
256 BuildingChangeEvent e;
257 e.action = "reset";
258 e.worldId = w.getId();
259 PlacementSystem::pushEvent(std::move(e));
260 }
261 return Result<AssemblyReceipt>::success(std::move(receipt));
262}
