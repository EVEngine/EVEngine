35EditorResult<void> AudioMixerTarget::applyDomainOperation(const DomainOperation& operation) {
36 if (operation.target != TargetId(id_))
37 return mixerError<void>(EditorStatus::Rejected, "editor.audio.mixer-target",
38 "Mixer operation targets another document");
39 auto parsed = parseBus(operation.payload);
40 if (!parsed.ok())
41 return mixerError<void>(EditorStatus::Rejected, "editor.audio.mixer-payload",
42 "Mixer operation contains an invalid bus");
43 if (operation.type == "audio.bus.create.v1") {
44 if (buses_.contains(parsed.value().id) ||
45 (!parsed.value().parent.empty() && !buses_.contains(parsed.value().parent)))
46 return mixerError<void>(EditorStatus::Conflict, "editor.audio.bus-create",
47 "Mixer bus already exists or its parent is missing");
48 buses_.emplace(parsed.value().id, parsed.value());
49 } else if (operation.type == "audio.bus.delete.v1") {
50 if (parsed.value().id == ObjectId("master") || !buses_.contains(parsed.value().id) ||
51 !children(parsed.value().id).empty())
52 return mixerError<void>(EditorStatus::Rejected, "editor.audio.bus-delete",
53 "Master, missing, or non-leaf mixer bus cannot be deleted");
54 buses_.erase(parsed.value().id);
55 } else if (operation.type == "audio.bus.replace.v1") {
56 if (!buses_.contains(parsed.value().id) || parsed.value().id == ObjectId("master") ||
57 !buses_.contains(parsed.value().parent) ||
58 wouldCycle(parsed.value().id, parsed.value().parent))
59 return mixerError<void>(EditorStatus::Rejected, "editor.audio.bus-replace",
60 "Mixer bus replacement produces an invalid hierarchy");
61 buses_[parsed.value().id] = parsed.value();
62 } else {
63 return mixerError<void>(EditorStatus::Unsupported, "editor.audio.mixer-operation",
64 "Unsupported mixer operation: " + operation.type);
65 }
66 ++revision_;
67 dirty_.include(0, 0);
68 return eve::editing::applied<void>();
69}

100EditorResult<DomainOperation> AudioMixerTarget::makeCreate(AudioBusSnapshot bus) const {
101 if (bus.id.empty() || bus.id == ObjectId("master") || bus.name.empty() ||
102 bus.volume < 0.0 || buses_.contains(bus.id) || !buses_.contains(bus.parent))
103 return mixerError<DomainOperation>(EditorStatus::Rejected, "editor.audio.bus-create",

132EditorResult<DomainOperation> AudioMixerTarget::makeReplace(AudioBusSnapshot changed) const {
133 const auto found = buses_.find(changed.id);
134 if (found == buses_.end() || changed.id == ObjectId("master") || changed.name.empty() ||
135 changed.volume < 0.0 || !buses_.contains(changed.parent) ||
136 wouldCycle(changed.id, changed.parent))
137 return mixerError<DomainOperation>(EditorStatus::Rejected, "editor.audio.bus-replace",
138 "Mixer bus settings or hierarchy are invalid");
139 DomainOperation operation;
140 operation.type = "audio.bus.replace.v1";
141 operation.inverseType = operation.type;
142 operation.target = TargetId(id_);
143 operation.payload = busValue(changed);
144 operation.inverse = busValue(found->second);
145 operation.hasInverse = true;
146 operation.affectedObjects.push_back({TargetId(id_), changed.id.value(), 0});
147 operation.affectedProperties = {"bus"};
148 operation.mergeKey = "audio.bus:" + changed.id.value();
149 return eve::editing::applied<DomainOperation>(std::move(operation));

152bool AudioMixerTarget::wouldCycle(const ObjectId& id, const ObjectId& parent) const {
153 ObjectId ancestor = parent;
154 while (!ancestor.empty()) {
155 if (ancestor == id) return true;
156 const auto found = buses_.find(ancestor);
157 if (found == buses_.end()) return false;
158 ancestor = found->second.parent;
159 }
160 return false;
161}

203EditorResult<void> AudioMixerTarget::loadSnapshot(const EditorValue& snapshot) {
204 const EditorValue* versionValue = field(snapshot, "schemaVersion");
205 const EditorValue* busesValue = field(snapshot, "buses");
206 const auto* version = versionValue ? versionValue->getIf<int64_t>() : nullptr;
207 const auto* buses = busesValue ? busesValue->getIf<EditorValue::Array>() : nullptr;
208 if (!version || *version != 1 || !buses)
209 return mixerError<void>(EditorStatus::Rejected, "editor.audio.mixer-snapshot",
210 "Mixer snapshot requires schemaVersion 1 and buses");
211 std::map<ObjectId, AudioBusSnapshot> candidate;
212 for (const EditorValue& value : *buses) {
213 auto parsed = parseBus(value);
214 if (!parsed.ok() || !candidate.emplace(parsed.value().id, parsed.value()).second)
215 return mixerError<void>(EditorStatus::Rejected, "editor.audio.mixer-snapshot-bus",
216 "Mixer snapshot contains an invalid or duplicate bus");
217 }
218 const auto master = candidate.find(ObjectId("master"));
219 if (master == candidate.end() || !master->second.parent.empty())
220 return mixerError<void>(EditorStatus::Rejected, "editor.audio.mixer-master",
221 "Mixer snapshot requires one root master bus");
222 for (const auto& [id, bus] : candidate) {
223 if (id == ObjectId("master")) continue;
224 if (!candidate.contains(bus.parent))
225 return mixerError<void>(EditorStatus::Rejected, "editor.audio.mixer-parent",
226 "Mixer snapshot contains a missing parent bus");
227 ObjectId ancestor = bus.parent;
228 while (ancestor != ObjectId("master")) {
229 if (ancestor == id)
230 return mixerError<void>(EditorStatus::Rejected, "editor.audio.mixer-cycle",
231 "Mixer snapshot contains a routing cycle");
232 const auto found = candidate.find(ancestor);
233 if (found == candidate.end() || found->second.parent.empty())
234 return mixerError<void>(EditorStatus::Rejected, "editor.audio.mixer-root",
235 "Every mixer bus must route to master");
236 ancestor = found->second.parent;
237 }
238 }
239 buses_ = std::move(candidate);
240 ++revision_;
241 dirty_.clear();
242 return eve::editing::applied<void>();
243}
