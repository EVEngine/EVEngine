37eve::Result<int> WorldInteraction::collectLoot(GameState &gameState, Tracker &tracker,
38 inventory::Bag &bag,
39 const WorldLootRequest &request) {
40 const bool hasItem = !request.itemId.empty() || request.itemQuantity != 0;
41 const bool hasAttribute = !request.attributeId.empty() || request.attributeAmount != 0.0;
42 const bool hasNotification = !request.notifyTopic.empty() || !request.notifyTarget.empty() ||
43 request.notifyAmount != 0;
44 if (!hasItem && !hasAttribute && !hasNotification)
45 return interactionFailure(eve::DiagnosticCode::InvalidArgument,
46 "world loot must contain at least one effect", "request", request);
47 if (hasItem && (request.itemId.empty() || request.itemQuantity <= 0))
48 return interactionFailure(eve::DiagnosticCode::InvalidArgument,
49 "item loot requires an id and positive quantity", "item", request);
50 if (hasAttribute && (request.attributeId.empty() || !std::isfinite(request.attributeAmount) ||
51 request.attributeAmount <= 0.0))
52 return interactionFailure(eve::DiagnosticCode::InvalidArgument,
53 "attribute loot requires an id and finite positive amount", "attribute", request);
54 if (hasAttribute &&
55 !std::isfinite(gameState.getVariable(request.attributeId) + request.attributeAmount))
56 return interactionFailure(eve::DiagnosticCode::InvalidArgument,
57 "attribute loot would produce a non-finite value", "attribute.amount", request);
58 if (hasNotification && (request.notifyTopic.empty() || request.notifyTarget.empty() ||
59 request.notifyAmount <= 0))
60 return interactionFailure(eve::DiagnosticCode::InvalidArgument,
61 "quest notification requires topic, target, and positive amount",
62 "notification", request);
63 if (!request.requiredQuestId.empty() && tracker.getState(request.requiredQuestId) != "active")
64 return interactionFailure(eve::DiagnosticCode::PreconditionViolation,
65 "required quest must be active before collecting world loot",
66 "requiredQuestId", request);
67
68 auto gameStateBefore = gameState.snapshotJson();
69 if (!gameStateBefore.ok()) return eve::Result<int>::failure(gameStateBefore.status());
70 std::string gameStateSnapshot = std::move(gameStateBefore).takeValue();
71 GameState candidateState;
72 auto candidateRestore = candidateState.restoreSnapshotJson(gameStateSnapshot);
73 if (!candidateRestore.ok()) return eve::Result<int>::failure(candidateRestore.status());
74 WorldState candidateWorld(candidateState);
75 if (candidateWorld.isObjectConsumed(request.mapId, request.objectId))
76 return interactionFailure(eve::DiagnosticCode::Conflict,
77 "world loot object was already consumed", "objectId", request);
78 auto candidateConsume = candidateWorld.consumeObject(request.mapId, request.objectId);
79 if (!candidateConsume.ok()) return eve::Result<int>::failure(candidateConsume.status());
80
81 std::optional<inventory::PreparedInventoryAdd> preparedInventory;
82 if (hasItem) {
83 auto prepared = inventory::InventorySystem::prepareAddBatch(
84 &bag, {{request.itemId, request.itemQuantity}});
85 if (!prepared.ok()) return eve::Result<int>::failure(prepared.status());
86 preparedInventory.emplace(std::move(prepared).takeValue());
87 }
88
89 TrackerRollback trackerBefore{tracker.entries, tracker.order, tracker.pending, tracker.polled};
90 if (hasAttribute) gameState.addVariable(request.attributeId, request.attributeAmount);
91 if (hasNotification)
92 tracker.notify(request.notifyTopic, request.notifyTarget, request.notifyAmount);
93 WorldState world(gameState);
94 auto consumed = world.consumeObject(request.mapId, request.objectId);
95 if (!consumed.ok() || consumed.code() != eve::StatusCode::Applied) {
96 auto restored = gameState.restoreSnapshotJson(gameStateSnapshot);
97 const bool restoredOk = restored.ok();
98 restoreTracker(tracker, std::move(trackerBefore));
99 if (!restoredOk)
100 return interactionFailure(eve::DiagnosticCode::InvariantViolation,
101 "world loot rollback could not restore GameState", "rollback", request);
102 return interactionFailure(eve::DiagnosticCode::Conflict,
103 "world loot object changed before commit", "objectId", request);
104 }
105
106 if (preparedInventory) {
107 auto committed = inventory::InventorySystem::commitAddBatch(std::move(*preparedInventory));
108 if (!committed.ok()) {
109 auto restored = gameState.restoreSnapshotJson(gameStateSnapshot);
110 const bool restoredOk = restored.ok();
111 restoreTracker(tracker, std::move(trackerBefore));
112 if (!restoredOk)
113 return interactionFailure(eve::DiagnosticCode::InvariantViolation,
114 "world loot rollback could not restore GameState", "rollback", request);
115 return eve::Result<int>::failure(committed.status());
116 }
117 }
118
119 return eve::Result<int>::success(static_cast<int>(hasItem) + static_cast<int>(hasAttribute) +
120 static_cast<int>(hasNotification) + 1,
121 eve::Status::success(eve::StatusCode::Applied));
122}
