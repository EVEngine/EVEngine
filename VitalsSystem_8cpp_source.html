46double VitalsSystem::takeDamage(RPGActor *actor, const std::string &resource, double amount,
47 const std::string &source) {
48 if (!actor || amount <= 0.0) return 0.0;
49 auto &current = actor->vitals()->current;
50 const double max = getMax(actor, resource);
51 const double cur = getCurrent(actor, resource);
52 if (cur <= 0.0) return 0.0; // 已死，不再受伤
53 const double applied = std::min(amount, cur);
54 current[resource] = cur - applied;
55 emit(actor, resource, "damage", applied, source, cur - applied, max);
56 if (cur - applied <= 0.0) {
57 emit(actor, resource, "death", applied, source, 0.0, max);
58 }
59 return applied;
60}

62double VitalsSystem::heal(RPGActor *actor, const std::string &resource, double amount) {
63 if (!actor || amount <= 0.0) return 0.0;
64 auto &current = actor->vitals()->current;
65 const double max = getMax(actor, resource);
66 const double cur = getCurrent(actor, resource);
67 const double applied = std::min(amount, max - cur);
68 if (applied <= 0.0) return 0.0;
69 current[resource] = cur + applied;
70 emit(actor, resource, "heal", applied, "", cur + applied, max);
71 return applied;
