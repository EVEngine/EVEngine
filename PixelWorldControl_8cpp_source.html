97eve::Result<PixelWorldStepReceipt> PixelWorldControlService::step(std::uint64_t worldId,
98 std::uint32_t count) {
99 PixelWorld* value = find(worldId);
100 if (!value) return missingWorld<PixelWorldStepReceipt>();
101 if (count == 0 || count > 1024)
102 return eve::Result<PixelWorldStepReceipt>::failure(eve::Diagnostic::error(
103 eve::DiagnosticCode::InvalidArgument, "step count must be in [1, 1024]", "count", {},
104 "pixelworld.control"));
105 PixelWorldStepReceipt receipt;
106 receipt.world = worldId;
107 receipt.steps = count;
108 receipt.firstTick = value->tickValue() + 1;
109 const auto start = std::chrono::steady_clock::now();
110 for (std::uint32_t index = 0; index < count; ++index) {
111 auto advanced = value->advanceImpl(eve::SimulationTick(value->tickValue() + 1), nullptr);
112 if (!advanced.ok()) return eve::Result<PixelWorldStepReceipt>::failure(advanced.status());
113 receipt.finalStep = advanced.value();
114 }
115 receipt.lastTick = value->tickValue();
116 receipt.elapsedMicroseconds = std::uint64_t(std::chrono::duration_cast<std::chrono::microseconds>(
117 std::chrono::steady_clock::now() - start)
118 .count());
119 return eve::Result<PixelWorldStepReceipt>::success(receipt);
120}

146eve::Result<std::vector<PixelWorldPerformanceSample>>
147PixelWorldControlService::performanceSamples(std::uint64_t worldId, std::uint32_t limit) const {
148 if (!find(worldId)) return missingWorld<std::vector<PixelWorldPerformanceSample>>();
149 if (limit == 0 || limit > 256)
150 return eve::Result<std::vector<PixelWorldPerformanceSample>>::failure(eve::Diagnostic::error(
151 eve::DiagnosticCode::InvalidArgument, "sample limit must be in [1, 256]", "limit", {},
152 "pixelworld.control"));
153 const auto found = impl().samples.find(worldId);
154 if (found == impl().samples.end())
155 return eve::Result<std::vector<PixelWorldPerformanceSample>>::success({});
156 const auto& samples = found->second;
157 const std::size_t begin = samples.size() > limit ? samples.size() - limit : 0;
158 std::vector<PixelWorldPerformanceSample> result(samples.begin() + std::ptrdiff_t(begin),
159 samples.end());
160 return eve::Result<std::vector<PixelWorldPerformanceSample>>::success(
161 std::move(result));
162}
