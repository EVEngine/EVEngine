28Result<SpriteAnimationClip> SpriteAnimationClip::decode(const Value& value) {
29 const auto* schema = field(value, "schema");
30 const auto* version = field(value, "schemaVersion");
31 const auto* loop = field(value, "loop");
32 const auto* keys = field(value, "frames");
33 const auto* frames = keys ? keys->getIf<Value::Array>() : nullptr;
34 SpriteAnimationClip out;
35 if (!schema || !schema->isString() || schema->asString() != "eve.sprite-animation" || !version ||
36 !version->isInt64() || version->asInt() != 1 || !loop || !loop->isBool() ||
37 !number(field(value, "duration"), out.duration_) || out.duration_ <= 0 || !frames || frames->empty() ||
38 frames->size() > 100000)
39 return bad<SpriteAnimationClip>("invalid sprite timeline");
40 out.loop_ = loop->asBool();
41 for (const auto& key : *frames) {
42 const auto* image = field(key, "image");
43 const auto* name = field(key, "name");
44 const auto* filter = field(key, "filter");
45 if (!image || !image->isString()) return bad<SpriteAnimationClip>("missing sprite image reference");
46 auto ref = AssetRef::parse(image->asString());
47 if (!ref) return Result<SpriteAnimationClip>::failure(ref.status());
48 SpriteAnimationFrame frame{std::move(ref).takeValue()};
49 if (!image || !image->isString() || !name || !name->isString() || !filter || !filter->isString() ||
50 (filter->asString() != "nearest" && filter->asString() != "linear") ||
51 !number(field(key, "time"), frame.time) || !number(field(key, "pixelsPerUnit"), frame.pixelsPerUnit) ||
52 !array(field(key, "rect"), frame.rect) || !array(field(key, "pivot"), frame.pivot) ||
53 !array(field(key, "imageSize"), frame.imageSize))
54 return bad<SpriteAnimationClip>("invalid sprite key fields");
55 frame.name = name->asString();
56 frame.nearest = filter->asString() == "nearest";
57 if (frame.time < 0 || frame.time >= out.duration_ ||
58 (out.frames_.empty() ? frame.time != 0 : frame.time <= out.frames_.back().time) ||
59 frame.pixelsPerUnit <= 0 || frame.rect[0] < 0 || frame.rect[1] < 0 || frame.rect[2] <= 0 ||
60 frame.rect[3] <= 0 || frame.imageSize[0] <= 0 || frame.imageSize[1] <= 0 ||
61 frame.rect[0] + frame.rect[2] > frame.imageSize[0] || frame.rect[1] + frame.rect[3] > frame.imageSize[1] ||
62 frame.pivot[0] < 0 || frame.pivot[0] > 1 || frame.pivot[1] < 0 || frame.pivot[1] > 1)
63 return bad<SpriteAnimationClip>("sprite key bounds or timing invalid");
64 out.frames_.push_back(std::move(frame));
65 }
66 return Result<SpriteAnimationClip>::success(std::move(out));
67}
