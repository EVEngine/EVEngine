36PropertyReadResult UiThemeCatalogTarget::read(const SelectionSnapshot& selection, const PropertyPath& path) const {
37 if (!matches(selection) || !schema(selection).find(path)) return {};
38 std::optional<EditorValue> common;
39 for (const auto& item : selection.items) {
40 const UiThemeAsset* asset = findTheme(ObjectId(item.item.value()));
41 EditorValue value = readThemeToken(asset->tokens, path);
42 if (!common)
43 common = value;
44 else if (*common != value)
45 return {PropertyReadState::Mixed, {}, {}};
46 }
47 return {PropertyReadState::Value, *common, {}};
48}

50EditorResult<DomainOperation> UiThemeCatalogTarget::makeSet(const SelectionSnapshot& selection,
51 const PropertyPath& path, const EditorValue& value,
52 PropertySetMode mode) const {
53 if (mode == PropertySetMode::Reset) return makeReset(selection, path);
54 auto descriptor = schema(selection).find(path);
55 if (!matches(selection) || !descriptor || mode != PropertySetMode::Absolute ||
56 !editing::validatePropertyValue(*descriptor, value).ok())
57 return fail<DomainOperation>(EditorStatus::Rejected, "editor.ui-theme.set",
58 "Theme property edit is invalid");
59 auto candidate = *this;
60 for (const auto& item : selection.items) {
61 UiThemeAsset* asset = candidate.mutableTheme(ObjectId(item.item.value()));
62 if (!asset)
63 return fail<DomainOperation>(EditorStatus::NotFound, "editor.ui-theme.missing",
64 "Theme asset does not exist");
65 auto assigned = assignThemeToken(asset->tokens, path, value);
66 if (!assigned.ok())
67 return fail<DomainOperation>(EditorStatus::Rejected, "editor.ui-theme.token",
68 "Theme token could not be assigned");
69 if (asset->id.value() != "dark" && asset->id.value() != "light")
70 asset->basePreset = UiThemeBasePreset::Custom;
71 }
72 auto diagnostics = candidate.validate();
73 if (std::any_of(diagnostics.begin(), diagnostics.end(),
74 [](const EditorDiagnostic& item) { return item.severity() == DiagnosticSeverity::Error; }))
75 return fail<DomainOperation>(EditorStatus::Rejected, "editor.ui-theme.invalid",
76 "Theme property edit is out of range");
77 return replacement(candidate.contentValue(), path.value());
78}
