248bool AnimClip::hasCompatibleSyncMarkers(const AnimClip* target) const {
249 if (!target || duration_ <= 1e-8f || target->duration_ <= 1e-8f) return false;
250 std::vector<const SyncMarker*> sourceCommon;
251 std::vector<const SyncMarker*> targetCommon;
252 for (const SyncMarker& marker : syncMarkers_) {
253 if (std::any_of(target->syncMarkers_.begin(), target->syncMarkers_.end(),
254 [&](const SyncMarker& other) { return other.name == marker.name; }))
255 sourceCommon.push_back(&marker);
256 }
257 for (const SyncMarker& marker : target->syncMarkers_) {
258 if (std::any_of(syncMarkers_.begin(), syncMarkers_.end(),
259 [&](const SyncMarker& other) { return other.name == marker.name; }))
260 targetCommon.push_back(&marker);
261 }
262 if (sourceCommon.size() < 2 || targetCommon.size() < 2) return false;
263 const size_t sourcePairCount = loop_ ? sourceCommon.size() : sourceCommon.size() - 1;
264 const size_t targetPairCount = target->loop_ ? targetCommon.size() : targetCommon.size() - 1;
265 for (size_t sourceIndex = 0; sourceIndex < sourcePairCount; ++sourceIndex) {
266 const std::string& previous = sourceCommon[sourceIndex]->name;
267 const std::string& next = sourceCommon[(sourceIndex + 1) % sourceCommon.size()]->name;
268 for (size_t targetIndex = 0; targetIndex < targetPairCount; ++targetIndex)
269 if (targetCommon[targetIndex]->name == previous &&
270 targetCommon[(targetIndex + 1) % targetCommon.size()]->name == next)
271 return true;
272 }
273 return false;
274}

276float AnimClip::mapSyncTimeTo(float time, const AnimClip* target) const {
277 if (!target || duration_ <= 1e-8f || target->duration_ <= 1e-8f) return 0.f;
278 const float normalizedFallback = wrapTime(time) / duration_ * target->duration_;
279 if (!hasCompatibleSyncMarkers(target)) return normalizedFallback;
280
281 std::vector<const SyncMarker*> sourceCommon;
282 std::vector<const SyncMarker*> targetCommon;
283 for (const SyncMarker& marker : syncMarkers_) {
284 if (std::any_of(target->syncMarkers_.begin(), target->syncMarkers_.end(),
285 [&](const SyncMarker& other) { return other.name == marker.name; }))
286 sourceCommon.push_back(&marker);
287 }
288 for (const SyncMarker& marker : target->syncMarkers_) {
289 if (std::any_of(syncMarkers_.begin(), syncMarkers_.end(),
290 [&](const SyncMarker& other) { return other.name == marker.name; }))
291 targetCommon.push_back(&marker);
292 }
293
294 const float sourceTime = wrapTime(time);
295 if (!loop_ && (sourceTime < sourceCommon.front()->t || sourceTime > sourceCommon.back()->t))
296 return normalizedFallback;
297 size_t sourcePrevious = sourceCommon.size() - 1;
298 for (size_t i = 0; i < sourceCommon.size(); ++i) {
299 if (sourceCommon[i]->t <= sourceTime)
300 sourcePrevious = i;
301 else
302 break;
303 }
304 const size_t sourceNext = (sourcePrevious + 1) % sourceCommon.size();
305 const float sourceStart = sourceCommon[sourcePrevious]->t;
306 float sourceEnd = sourceCommon[sourceNext]->t;
307 float adjustedSourceTime = sourceTime;
308 if (sourceNext == 0) {
309 sourceEnd += duration_;
310 if (adjustedSourceTime < sourceStart) adjustedSourceTime += duration_;
311 }
312 const float span = sourceEnd - sourceStart;
313 if (span <= 1e-8f) return normalizedFallback;
314 const float alpha = clampf((adjustedSourceTime - sourceStart) / span, 0.f, 1.f);
315
316 const std::string& previousName = sourceCommon[sourcePrevious]->name;
317 const std::string& nextName = sourceCommon[sourceNext]->name;
318 int bestTarget = -1;
319 float bestDistance = target->duration_ + 1.f;
320 const float sourcePhase = sourceStart / duration_;
321 const size_t targetPairCount = target->loop_ ? targetCommon.size() : targetCommon.size() - 1;
322 for (size_t i = 0; i < targetPairCount; ++i) {
323 if (targetCommon[i]->name != previousName || targetCommon[(i + 1) % targetCommon.size()]->name != nextName)
324 continue;
325 const float targetPhase = targetCommon[i]->t / target->duration_;
326 const float distance = std::abs(targetPhase - sourcePhase);
327 const float circularDistance = std::min(distance, 1.f - distance);
328 if (circularDistance < bestDistance) {
329 bestDistance = circularDistance;
330 bestTarget = static_cast<int>(i);
331 }
332 }
333 if (bestTarget < 0) return normalizedFallback;
334 const size_t targetPrevious = static_cast<size_t>(bestTarget);
335 const size_t targetNext = (targetPrevious + 1) % targetCommon.size();
336 const float targetStart = targetCommon[targetPrevious]->t;
337 float targetEnd = targetCommon[targetNext]->t;
338 if (targetNext == 0) targetEnd += target->duration_;
339 float mapped = targetStart + alpha * (targetEnd - targetStart);
340 if (target->getLoop()) mapped = std::fmod(mapped, target->duration_);
341 return target->wrapTime(mapped);
342}

344void AnimClip::collectEvents(float previousTime, float currentTime, bool loop, std::vector<std::string>& out) const {
345 if (events_.empty() || duration_ <= 0.f || currentTime == previousTime) return;
346 const float from = wrapTime(previousTime);
347 const float to = wrapTime(currentTime);
348 auto appendRange = [&](float lo, float hi, bool includeLo) {
349 for (const EventMarker& event : events_) {
350 if ((includeLo ? event.t >= lo : event.t > lo) && event.t <= hi) out.push_back(event.name);
351 }
352 };
353 if (loop && (currentTime - previousTime >= duration_ || to < from)) {
354 appendRange(from, duration_, false);
355 appendRange(0.f, to, true);
356 } else {
357 appendRange(from, to, false);
358 }
359}

464void AnimClip::sampleVec3(const std::vector<Vec3Key>& keys, float time, float& x, float& y, float& z, bool& ok) {
465 ok = false;
466 if (keys.empty()) return;
467 ok = true;
468 if (time <= keys.front().t || keys.size() == 1) {
469 x = keys.front().x;
470 y = keys.front().y;
471 z = keys.front().z;
472 return;
473 }
474 if (time >= keys.back().t) {
475 x = keys.back().x;
476 y = keys.back().y;
477 z = keys.back().z;
478 return;
479 }

574int AnimClip::compress(float positionError, float rotationErrorDegrees, float scaleError) {
575 if (positionError < 0.f || rotationErrorDegrees < 0.f || scaleError < 0.f)
576 throw Exception("AnimClip.compress: tolerances must be >= 0");
577 const float rotationError = rotationErrorDegrees * 0.01745329251994329577f;
578 int removed = 0;
579 for (BoneTrack& track : tracks_) {
580 auto reduceVec3 = [&](std::vector<Vec3Key>& keys, float tolerance) {
581 if (keys.size() <= 2) return;
582 std::vector<unsigned char> keep(keys.size(), 0);
583 keep.front() = keep.back() = 1;
584 std::function<void(size_t, size_t)> split = [&](size_t first, size_t last) {
585 if (last <= first + 1) return;
586 const float span = keys[last].t - keys[first].t;
587 float worst = -1.f;
588 size_t worstIndex = first;
589 for (size_t i = first + 1; i < last; ++i) {
590 const float alpha = span > 1e-8f ? (keys[i].t - keys[first].t) / span : 0.f;
591 const float dx = keys[i].x - lerpf(keys[first].x, keys[last].x, alpha);
592 const float dy = keys[i].y - lerpf(keys[first].y, keys[last].y, alpha);
593 const float dz = keys[i].z - lerpf(keys[first].z, keys[last].z, alpha);
594 const float error = std::sqrt(dx * dx + dy * dy + dz * dz);
595 if (error > worst) {
596 worst = error;
597 worstIndex = i;
598 }
599 }
600 if (worst > tolerance) {
601 keep[worstIndex] = 1;
602 split(first, worstIndex);
603 split(worstIndex, last);
604 }
605 };
606 split(0, keys.size() - 1);
607 std::vector<Vec3Key> reduced;
608 reduced.reserve(keys.size());
609 for (size_t i = 0; i < keys.size(); ++i)
610 if (keep[i]) reduced.push_back(keys[i]);
611 removed += static_cast<int>(keys.size() - reduced.size());
612 keys.swap(reduced);
613 };
614 auto reduceQuat = [&](std::vector<QuatKey>& keys) {
615 if (keys.size() <= 2) return;
616 std::vector<unsigned char> keep(keys.size(), 0);
617 keep.front() = keep.back() = 1;
618 std::function<void(size_t, size_t)> split = [&](size_t first, size_t last) {
619 if (last <= first + 1) return;
620 const float span = keys[last].t - keys[first].t;
621 float worst = -1.f;
622 size_t worstIndex = first;
623 for (size_t i = first + 1; i < last; ++i) {
624 const float alpha = span > 1e-8f ? (keys[i].t - keys[first].t) / span : 0.f;
625 float x, y, z, w;
626 slerpQuat(keys[first].x, keys[first].y, keys[first].z, keys[first].w, keys[last].x, keys[last].y,
627 keys[last].z, keys[last].w, alpha, x, y, z, w);
628 const float dot = std::abs(x * keys[i].x + y * keys[i].y + z * keys[i].z + w * keys[i].w);
629 const float error = 2.f * std::acos(clampf(dot, -1.f, 1.f));
630 if (error > worst) {
631 worst = error;
632 worstIndex = i;
633 }
634 }
635 if (worst > rotationError) {
636 keep[worstIndex] = 1;
637 split(first, worstIndex);
638 split(worstIndex, last);
639 }
640 };
641 split(0, keys.size() - 1);
642 std::vector<QuatKey> reduced;
643 reduced.reserve(keys.size());
644 for (size_t i = 0; i < keys.size(); ++i)
645 if (keep[i]) reduced.push_back(keys[i]);
646 removed += static_cast<int>(keys.size() - reduced.size());
647 keys.swap(reduced);
648 };
649 reduceVec3(track.positions, positionError);
650 reduceQuat(track.rotations);
651 reduceVec3(track.scales, scaleError);
652 }
653 return removed;
654}

763AnimClip* AnimClip::retargetWithProfile(const AnimSkeleton* sourceSkeleton, const AnimSkeleton* targetSkeleton,
764 AnimRetargetProfile* profile) const {
765 if (!sourceSkeleton || !targetSkeleton) throw Exception("AnimClip.retargetWithProfile: skeleton is null");
766 if (!profile) throw Exception("AnimClip.retargetWithProfile: profile is null");
767
768 auto out = std::make_unique<AnimClip>(name_ + "_retargeted");
769 out->duration_ = duration_;
770 out->loop_ = loop_;
771 out->sampleRate_ = sampleRate_;
772 out->events_ = events_;
773 out->syncMarkers_ = syncMarkers_;
774 out->tracks_.resize(static_cast<size_t>(targetSkeleton->getBoneCount()));
775
776 std::vector<int> sourceForTarget(static_cast<size_t>(targetSkeleton->getBoneCount()), -1);
777 profile->matchedBoneCount_ = 0;
778 profile->unmatchedTargetBones_.clear();
779 for (int targetBone = 0; targetBone < targetSkeleton->getBoneCount(); ++targetBone) {
780 const std::string targetName = targetSkeleton->getBoneName(targetBone);
781 int sourceBone = -1;
782 bool explicitlyMapped = false;
783 for (const auto& mapping : profile->mappings_) {
784 if (mapping.target == targetName) {
785 explicitlyMapped = true;
786 sourceBone = sourceSkeleton->findBone(mapping.source);
787 break;
788 }
789 }
790 if (!explicitlyMapped) sourceBone = sourceSkeleton->findBone(targetName);
791 if (!explicitlyMapped && sourceBone < 0 && profile->normalizedNameMatching_)
792 sourceBone = findNormalizedBone(sourceSkeleton, targetName);
793 if (sourceBone < 0) {
794 profile->unmatchedTargetBones_.push_back(targetName);
795 continue;
796 }
797 sourceForTarget[static_cast<size_t>(targetBone)] = sourceBone;
798 ++profile->matchedBoneCount_;
799 }
800
801 int targetRoot = profile->targetRoot_.empty() ? -1 : targetSkeleton->findBone(profile->targetRoot_);
802 int sourceRoot = profile->sourceRoot_.empty() ? -1 : sourceSkeleton->findBone(profile->sourceRoot_);
803 if ((!profile->targetRoot_.empty() && targetRoot < 0) || (!profile->sourceRoot_.empty() && sourceRoot < 0))
804 throw Exception("AnimClip.retargetWithProfile: configured root bone was not found");
805 if (targetRoot < 0 || sourceRoot < 0) {
806 for (int targetBone = 0; targetBone < targetSkeleton->getBoneCount(); ++targetBone) {
807 const int candidate = sourceForTarget[static_cast<size_t>(targetBone)];
808 if (candidate < 0 || candidate >= static_cast<int>(tracks_.size()) ||
809 tracks_[static_cast<size_t>(candidate)].positions.empty())
810 continue;
811 const std::string name = normalizedBoneName(targetSkeleton->getBoneName(targetBone));
812 if (name.find("hips") != std::string::npos || name.find("pelvis") != std::string::npos) {
813 targetRoot = targetBone;
814 sourceRoot = candidate;
815 break;
816 }
817 }
818 }
819 if (targetRoot < 0 || sourceRoot < 0) {
820 for (int targetBone = 0; targetBone < targetSkeleton->getBoneCount(); ++targetBone) {
821 if (targetSkeleton->getParent(targetBone) == -1 && sourceForTarget[static_cast<size_t>(targetBone)] >= 0) {
822 targetRoot = targetBone;
823 sourceRoot = sourceForTarget[static_cast<size_t>(targetBone)];
824 break;
825 }
826 }
827 }
828
829 AnimPose sourceBindPose, targetBindPose, sourcePose;
830 sourceSkeleton->applyBindPose(&sourceBindPose);
831 targetSkeleton->applyBindPose(&targetBindPose);
832 sourceBindPose.computeWorld(sourceSkeleton);
833 targetBindPose.computeWorld(targetSkeleton);
834 const float automaticScale = profile->autoRootScale_
835 ? retargetRootMeasure(targetSkeleton, targetBindPose, targetRoot) /
836 retargetRootMeasure(sourceSkeleton, sourceBindPose, sourceRoot)
837 : 1.f;
838
839 const int frameCount = duration_ <= 0.f ? 1 : std::max(1, static_cast<int>(std::ceil(duration_ * sampleRate_)));
840 std::vector<TransformTRS> targetWorld(static_cast<size_t>(targetSkeleton->getBoneCount()));
841 for (int frame = 0; frame <= frameCount; ++frame) {
842 if (duration_ <= 0.f && frame > 0) break;
843 const float time = duration_ <= 0.f ? 0.f : duration_ * static_cast<float>(frame) / frameCount;
844 sample(time, &sourcePose, sourceSkeleton);
845 sourcePose.computeWorld(sourceSkeleton);
846
847 for (int targetBone = 0; targetBone < targetSkeleton->getBoneCount(); ++targetBone) {
848 const int sourceBone = sourceForTarget[static_cast<size_t>(targetBone)];
849 TransformTRS local = targetSkeleton->bindLocal(targetBone);
850 if (sourceBone >= 0 && sourceBone < static_cast<int>(tracks_.size())) {
851 const BoneTrack& sourceTrack = tracks_[static_cast<size_t>(sourceBone)];
852 const TransformTRS& sourceLocal = sourcePose.local(sourceBone);
853 const TransformTRS& sourceBind = sourceSkeleton->bindLocal(sourceBone);
854 const TransformTRS& targetBind = targetSkeleton->bindLocal(targetBone);
855 const float sourceLength = std::sqrt(sourceBind.px * sourceBind.px + sourceBind.py * sourceBind.py +
856 sourceBind.pz * sourceBind.pz);
857 const float targetLength = std::sqrt(targetBind.px * targetBind.px + targetBind.py * targetBind.py +
858 targetBind.pz * targetBind.pz);
859 const float boneScale =
860 sourceLength > 1e-6f && targetLength > 1e-6f ? targetLength / sourceLength : 1.f;
861 if (!sourceTrack.positions.empty()) {
862 if (targetBone == targetRoot && sourceBone == sourceRoot) {
863 const float horizontal = automaticScale * profile->rootHorizontalScale_;
864 const float vertical = automaticScale * profile->rootVerticalScale_;
865 local.px = targetBind.px + (sourceLocal.px - sourceBind.px) * horizontal;
866 local.py = targetBind.py + (sourceLocal.py - sourceBind.py) * vertical;
867 local.pz = targetBind.pz + (sourceLocal.pz - sourceBind.pz) * horizontal;
868 } else {
869 local.px = targetBind.px + (sourceLocal.px - sourceBind.px) * boneScale;
870 local.py = targetBind.py + (sourceLocal.py - sourceBind.py) * boneScale;
871 local.pz = targetBind.pz + (sourceLocal.pz - sourceBind.pz) * boneScale;
872 }
873 out->tracks_[static_cast<size_t>(targetBone)].positions.push_back(
874 {time, local.px, local.py, local.pz});
875 }
876 if (!sourceTrack.rotations.empty()) {
877 RetargetQuat desired;
878 if (profile->skeletonSpaceRotation_) {
879 desired = retargetMul(rotationOf(targetBindPose.world(targetBone)),
880 retargetMul(retargetInverse(rotationOf(sourceBindPose.world(sourceBone))),
881 rotationOf(sourcePose.world(sourceBone))));
882 const int parent = targetSkeleton->getParent(targetBone);
883 if (parent >= 0)
884 desired = retargetMul(retargetInverse(rotationOf(targetWorld[static_cast<size_t>(parent)])),
885 desired);
886 } else {
887 desired =
888 retargetMul(rotationOf(targetBind),
889 retargetMul(retargetInverse(rotationOf(sourceBind)), rotationOf(sourceLocal)));
890 }
891 local.qx = desired.x;
892 local.qy = desired.y;
893 local.qz = desired.z;
894 local.qw = desired.w;
895 local.normalizeRotation();
896 out->tracks_[static_cast<size_t>(targetBone)].rotations.push_back(
897 {time, local.qx, local.qy, local.qz, local.qw});
898 }
899 if (!sourceTrack.scales.empty()) {
900 local.sx = targetBind.sx *
901 (std::abs(sourceBind.sx) > 1e-6f ? sourceLocal.sx / sourceBind.sx : sourceLocal.sx);
902 local.sy = targetBind.sy *
903 (std::abs(sourceBind.sy) > 1e-6f ? sourceLocal.sy / sourceBind.sy : sourceLocal.sy);
904 local.sz = targetBind.sz *
905 (std::abs(sourceBind.sz) > 1e-6f ? sourceLocal.sz / sourceBind.sz : sourceLocal.sz);
906 out->tracks_[static_cast<size_t>(targetBone)].scales.push_back(
907 {time, local.sx, local.sy, local.sz});
908 }
909 }
910 const int parent = targetSkeleton->getParent(targetBone);
911 if (parent < 0) {
912 targetWorld[static_cast<size_t>(targetBone)] = local;
913 } else {
914 const TransformTRS& parentWorld = targetWorld[static_cast<size_t>(parent)];
915 TransformTRS& world = targetWorld[static_cast<size_t>(targetBone)];
916 const RetargetQuat worldRotation = retargetMul(rotationOf(parentWorld), rotationOf(local));
917 world = local;
918 world.qx = worldRotation.x;
919 world.qy = worldRotation.y;
920 world.qz = worldRotation.z;
921 world.qw = worldRotation.w;
922 }
923 }
924 }
925 return out.release();
