34Result<MeshStaging> decodeMesh(std::span<const std::uint8_t> bytes,
35 const GraphicsAssetLoadLimits& limits) {
36 static constexpr std::uint8_t magic[] = {'E', 'V', 'M', 'E', 'S', 'H', 0, 1};
37 if (bytes.size() < 24 || !std::equal(std::begin(magic), std::end(magic), bytes.begin()))
38 return failure<MeshStaging>(DiagnosticCode::ParseError, "canonical mesh header is invalid");
39 const std::uint32_t vertexCount = little32(bytes, 8);
40 const std::uint32_t indexCount = little32(bytes, 12);
41 const std::uint32_t flags = little32(bytes, 16);
42 const std::uint32_t reserved = little32(bytes, 20);
43 if (vertexCount == 0 || indexCount == 0 || indexCount % 3 != 0 || flags > 3 || reserved != 0)
44 return failure<MeshStaging>(DiagnosticCode::ParseError, "canonical mesh metadata is invalid");
45 if (vertexCount > limits.maximumVertices || indexCount > limits.maximumIndices ||
46 vertexCount > static_cast<std::uint32_t>(std::numeric_limits<int>::max()) ||
47 indexCount > static_cast<std::uint32_t>(std::numeric_limits<int>::max()))
48 return failure<MeshStaging>(DiagnosticCode::InvalidArgument, "canonical mesh exceeds graphics limits");
49 const std::uint64_t floatsPerVertex = 3 + ((flags & 1u) ? 3 : 0) + ((flags & 2u) ? 2 : 0);
50 const std::uint64_t payloadBytes = std::uint64_t(vertexCount) * floatsPerVertex * 4 +
51 std::uint64_t(indexCount) * 4;
52 if (payloadBytes > limits.maximumDecodedBytes || payloadBytes > bytes.size() - 24 ||
53 payloadBytes != bytes.size() - 24)
54 return failure<MeshStaging>(DiagnosticCode::InvalidArgument,
55 "canonical mesh byte size is invalid or exceeds budget");
56
57 MeshStaging result;
58 result.positions.reserve(std::size_t(vertexCount) * 3);
59 if (flags & 1u) result.normals.reserve(std::size_t(vertexCount) * 3);
60 if (flags & 2u) result.texcoords.reserve(std::size_t(vertexCount) * 2);
61 std::size_t cursor = 24;
62 auto appendFloats = [&](std::vector<float>& output, std::uint32_t count) -> Result<void> {
63 for (std::uint32_t index = 0; index < count; ++index) {
64 const float value = littleFloat(bytes, cursor);
65 cursor += 4;
66 if (!std::isfinite(value))
67 return Result<void>::failure(Diagnostic::error(
68 DiagnosticCode::ParseError, "canonical mesh contains a non-finite value", {}, {},
69 "asset.graphics"));
70 output.push_back(value);
71 }
72 return Result<void>::success();
73 };
74 for (std::uint32_t vertex = 0; vertex < vertexCount; ++vertex) {
75 auto positions = appendFloats(result.positions, 3);
76 if (!positions) return Result<MeshStaging>::failure(positions.status());
77 if (flags & 1u) {
78 auto normals = appendFloats(result.normals, 3);
79 if (!normals) return Result<MeshStaging>::failure(normals.status());
80 }
81 if (flags & 2u) {
82 auto texcoords = appendFloats(result.texcoords, 2);
83 if (!texcoords) return Result<MeshStaging>::failure(texcoords.status());
84 }
85 }
86 result.indices.reserve(indexCount);
87 for (std::uint32_t index = 0; index < indexCount; ++index) {
88 const std::uint32_t value = little32(bytes, cursor);
89 cursor += 4;
90 if (value >= vertexCount)
91 return failure<MeshStaging>(DiagnosticCode::ParseError,
92 "canonical mesh index exceeds vertex count");
93 result.indices.push_back(value);
94 }
95 return Result<MeshStaging>::success(std::move(result));
96}

100Result<LoadedGraphicsMesh> EvpackGraphicsLoader::loadMesh(
101 const AssetRef& asset, const asset::EvpackCapabilities& capabilities,
102 const GraphicsAssetLoadLimits& limits) const {
103 auto payload = reader_.read(asset, "eve.mesh/1", capabilities, limits.maximumDecodedBytes);
104 if (!payload) return Result<LoadedGraphicsMesh>::failure(payload.status());
105 const asset::RuntimeAssetChunk* bulk = nullptr;
106 for (const auto& chunk : payload.value().chunks) {
107 if (chunk.kind != asset::EvpackChunkKind::Bulk) continue;
108 if (bulk)
109 return failure<LoadedGraphicsMesh>(DiagnosticCode::ParseError,
110 "eve.mesh/1 must contain exactly one bulk chunk");
111 bulk = &chunk;
112 }
113 if (!bulk)
114 return failure<LoadedGraphicsMesh>(DiagnosticCode::NotFound,
115 "eve.mesh/1 bulk chunk is missing");
116 auto staging = decodeMesh(bulk->bytes, limits);
117 if (!staging) return Result<LoadedGraphicsMesh>::failure(staging.status());
118 auto uploaded = factory_.uploadMesh(
119 staging.value().positions.data(),
120 staging.value().normals.empty() ? nullptr : staging.value().normals.data(),
121 staging.value().texcoords.empty() ? nullptr : staging.value().texcoords.data(),
122 static_cast<int>(staging.value().positions.size() / 3), staging.value().indices.data(),
123 static_cast<int>(staging.value().indices.size()));
124 if (!uploaded) return Result<LoadedGraphicsMesh>::failure(uploaded.status());
125 return Result<LoadedGraphicsMesh>::success(
126 {asset, uploaded.value(), std::move(payload).takeValue().variant});
127}
