29Result<OceanSpectrumSettings> decodeOceanSpectrumSettings(const Value& value) {
30 try {
31 keys(value, {"schema", "version", "seed", "significantWaveHeight", "peakPeriod", "peakEnhancement", "gravity",
32 "seaLevel", "choppiness", "windDirection", "current", "bands"});
33 const auto& schema = field(value, "schema");
34 const auto& version = field(value, "version");
35 if (!schema.isString() || schema.asString() != "eve.ocean.spectrum" || !version.isInt64() ||
36 version.asInt() != 1)
37 throw std::runtime_error("unsupported schema/version");
38 OceanSpectrumSettings out;
39 const auto& seed = field(value, "seed");
40 if (!seed.isString()) throw std::runtime_error("seed requires canonical decimal text");
41 const auto& text = seed.asString();
42 if (text.empty() || text.size() > 20 || (text.size() > 1 && text[0] == '0'))
43 throw std::runtime_error("noncanonical seed");
44 const auto parsed = std::from_chars(text.data(), text.data() + text.size(), out.seed);
45 if (parsed.ec != std::errc{} || parsed.ptr != text.data() + text.size())
46 throw std::runtime_error("seed outside uint64");
47 out.significantWaveHeight = number(field(value, "significantWaveHeight"));
48 out.peakPeriod = number(field(value, "peakPeriod"));
49 out.peakEnhancement = number(field(value, "peakEnhancement"));
50 out.gravity = number(field(value, "gravity"));
51 out.seaLevel = number(field(value, "seaLevel"));
52 out.choppiness = number(field(value, "choppiness"));
53 out.windDirection = vector(field(value, "windDirection"));
54 out.current = vector(field(value, "current"));
55 const auto& bands = field(value, "bands");
56 if (!bands.isArray() || bands.arraySize() == 0 || bands.arraySize() > 4) throw std::runtime_error("band count");
57 out.bands.clear();
58 for (std::size_t i = 0; i < bands.arraySize(); ++i) {
59 const auto& b = bands.at(i);
60 keys(b, {"resolution", "length", "minimumWavelength", "maximumWavelength"});
61 const auto& resolution = field(b, "resolution");
62 if (!resolution.isInt64() || resolution.asInt() < 0 ||
63 resolution.asInt() > std::numeric_limits<unsigned>::max())
64 throw std::runtime_error("resolution integer range");
65 out.bands.push_back({static_cast<unsigned>(resolution.asInt()), number(field(b, "length")),
66 number(field(b, "minimumWavelength")), number(field(b, "maximumWavelength"))});
67 }
