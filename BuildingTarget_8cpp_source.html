225 }

230 const auto* values = value.getIf<EditorValue::Array>();
231 if (!values || values->empty())
232 return buildingError<std::vector<BuildingInstanceSnapshot>>(
233 EditorStatus::Rejected, "editor.building.invalid-area-payload",
234 "Building area payload must contain at least one instance");
235 std::vector<BuildingInstanceSnapshot> result;

245 return buildingError<std::vector<BuildingInstanceSnapshot>>(
246 EditorStatus::Rejected, "editor.building.duplicate-area-instance",
247 "Building area payload contains duplicate instance ids");
248 result.push_back(std::move(parsed).takeValue());
249 }
250 return eve::editing::applied<std::vector<BuildingInstanceSnapshot>>(std::move(result));
251}

286EditorResult<EdgeCurvePayload> parseEdgeCurve(const EditorValue& value) {
287 const EditorValue* groupIdValue = field(value, "groupId");
288 const EditorValue* buildingIdValue = field(value, "buildingId");
289 const EditorValue* levelValue = field(value, "level");
290 const EditorValue* subdivisionsValue = field(value, "subdivisions");
291 const EditorValue* controlsValue = field(value, "controlPoints");
292 const EditorValue* instancesValue = field(value, "instances");
293 const auto* groupId = groupIdValue ? groupIdValue->getIf<int64_t>() : nullptr;
294 const auto* buildingId =
295 buildingIdValue ? buildingIdValue->getIf<std::string>() : nullptr;
296 const auto* level = levelValue ? levelValue->getIf<int64_t>() : nullptr;
297 const auto* subdivisions =
298 subdivisionsValue ? subdivisionsValue->getIf<int64_t>() : nullptr;
299 const auto* controls = controlsValue ? controlsValue->getIf<EditorValue::Array>() : nullptr;
300 if (!groupId || *groupId <= 0 || !buildingId || buildingId->empty() || !level ||
301 !subdivisions || *subdivisions < 2 || *subdivisions > 4096 || !controls ||
302 controls->size() != 8 || !instancesValue)
303 return buildingError<EdgeCurvePayload>(EditorStatus::Rejected,
304 "editor.building.invalid-edge-curve",
305 "Edge curve payload is incomplete");
306 auto instances = parseInstances(*instancesValue);
307 if (!instances.ok())
308 return buildingError<EdgeCurvePayload>(instances.code(),
309 "editor.building.invalid-edge-curve-instances",
310 "Edge curve instances are invalid");
311 EdgeCurvePayload result;
312 result.group.id = {static_cast<std::uint64_t>(*groupId)};
313 result.group.buildingId = *buildingId;
314 result.group.level = static_cast<int>(*level);
315 result.group.subdivisions = static_cast<int>(*subdivisions);
316 if (const EditorValue* provider = field(value, "surfaceProviderName"))
317 if (const auto* text = provider->getIf<std::string>())
318 result.group.surfaceProviderName = *text;
319 if (const EditorValue* surface = field(value, "surfaceId"))
320 if (const auto* text = surface->getIf<std::string>()) result.group.surfaceId = *text;
321 if (const EditorValue* revision = field(value, "surfaceRevision")) {
322 const auto* integer = revision->getIf<int64_t>();
323 if (!integer || *integer < 0)
324 return buildingError<EdgeCurvePayload>(
325 EditorStatus::Rejected, "editor.building.invalid-edge-curve-surface-revision",
326 "Edge curve surface revision is invalid");
327 result.group.surfaceRevision = static_cast<std::uint64_t>(*integer);
328 }
329 if (const EditorValue* sampleValue = field(value, "surfaceSamples")) {
330 const auto* samples = sampleValue->getIf<EditorValue::Array>();
331 if (!samples || samples->size() % 6 != 0)
332 return buildingError<EdgeCurvePayload>(
333 EditorStatus::Rejected, "editor.building.invalid-edge-curve-surface-samples",
334 "Edge curve surface samples must contain position and normal sextuples");
335 for (size_t index = 0; index < samples->size(); index += 6) {
336 std::array<float, 6> frame{};
337 for (size_t component = 0; component < 6; ++component) {
338 const auto* number = (*samples)[index + component].getIf<double>();
339 if (!number)
340 return buildingError<EdgeCurvePayload>(
341 EditorStatus::Rejected,
342 "editor.building.invalid-edge-curve-surface-samples",
343 "Edge curve surface samples must be numeric");
344 frame[component] = static_cast<float>(*number);
345 }
346 result.group.surfaceSamples.push_back(
347 {frame[0], frame[1], frame[2], frame[3], frame[4], frame[5]});
348 }
349 }
350 for (size_t index = 0; index < controls->size(); index += 2) {
351 const auto* x = (*controls)[index].getIf<double>();
352 const auto* y = (*controls)[index + 1].getIf<double>();
353 if (!x || !y)
354 return buildingError<EdgeCurvePayload>(EditorStatus::Rejected,
355 "editor.building.invalid-edge-curve-controls",
356 "Edge curve controls must be numbers");
357 result.group.controlPoints.push_back(
358 {static_cast<float>(*x), static_cast<float>(*y)});
359 }
360 result.instances = std::move(instances).takeValue();
361 for (const BuildingInstanceSnapshot& instance : result.instances)
362 result.group.instanceIds.push_back(instance.instanceId);
363 return eve::editing::applied<EdgeCurvePayload>(std::move(result));
364}

463EditorResult<DomainOperation> areaOperation(
464 const std::string& target, building::PlacementWorld* world,
465 const building::PlacementSystem::AreaPreview& preview) {
466 if (!world || preview.cells.empty() || preview.rejectedCount != 0 ||
467 preview.acceptedCount != static_cast<int>(preview.cells.size()))
468 return buildingError<DomainOperation>(EditorStatus::Rejected,
469 "editor.building.area-rejected",
470 "Area placement requires a fully accepted preview");
471 const building::BuildingDefinition* definition =
472 building::BuildingRegistry::find(preview.buildingId);
473 if (!definition || definition->placementKind == "edge")
474 return buildingError<DomainOperation>(EditorStatus::Rejected,
475 "editor.building.area-definition-invalid",
476 "Area placement requires a cell building definition");
477 DomainOperation operation;
478 operation.type = "building.area.set.v1";
479 operation.inverseType = "building.area.delete.v1";
480 operation.target = TargetId(target);
481 EditorValue::Array instances;
482 int nextId = 1;
483 for (const building::PlacementSystem::AreaCellPreview& cell : preview.cells) {
484 while (world->hasBuilding(nextId)) ++nextId;
485 BuildingInstanceSnapshot placed;
486 placed.instanceId = nextId++;
487 placed.buildingId = preview.buildingId;
488 placed.cellX = cell.cellX;
489 placed.cellY = cell.cellY;
490 placed.rotationDegrees = preview.rotationDeg;
491 placed.channel = definition->channel;
492 placed.tags = definition->tags;
493 float worldX = 0.f;
494 float worldY = 0.f;
495 world->cellToWorldPlane(cell.cellX, cell.cellY, worldX, worldY);
496 placed.worldX = worldX;
497 placed.worldY = worldY;

805 const std::string& buildingId,
806 const std::vector<BuildingEdgePathVertex>& vertices) const {
807 std::vector<building::CornerAddress> runtimeVertices;
808 runtimeVertices.reserve(vertices.size());
809 for (const BuildingEdgePathVertex& vertex : vertices)
810 runtimeVertices.push_back({vertex.x, vertex.y});
811 auto preview = building::PlacementSystem::previewEdgePath(world_, buildingId,
812 runtimeVertices);
813 if (!preview.ok())
814 return buildingError<DomainOperation>(EditorStatus::Rejected,
815 "editor.building.edge-path-rejected",
816 "Edge path preview was rejected");
817 const building::BuildingDefinition* definition =
818 building::BuildingRegistry::find(buildingId);
819 if (!definition)
820 return buildingError<DomainOperation>(EditorStatus::NotFound,
821 "editor.building.definition-not-found",
822 "Edge path definition was not found");

824 operation.type = "building.area.set.v1";
825 operation.inverseType = "building.area.delete.v1";
826 operation.target = TargetId(id_);
827 EditorValue::Array instances;
828 int nextId = 1;
829 for (const building::EdgeAddress& edge : preview.value().edges) {
830 while (world_->hasBuilding(nextId)) ++nextId;
831 BuildingInstanceSnapshot placed;
832 placed.instanceId = nextId++;
833 placed.buildingId = buildingId;
834 placed.placementKind = "edge";
835 placed.edgeX = edge.x;
836 placed.edgeY = edge.y;
837 placed.edgeAxis = edge.axis == building::EdgeAxis::Horizontal ? "horizontal" : "vertical";
838 placed.cellX = edge.x;
839 placed.cellY = edge.y;
840 placed.level = preview.value().level;
841 placed.rotationDegrees = edge.axis == building::EdgeAxis::Horizontal ? 0.0 : 90.0;
842 placed.channel = definition->channel;
843 placed.tags = definition->tags;
844 float ax = 0.f, ay = 0.f, bx = 0.f, by = 0.f;
845 world_->cellToWorldPlane(edge.x, edge.y, ax, ay);
846 world_->cellToWorldPlane(edge.x + (edge.axis == building::EdgeAxis::Horizontal ? 1 : 0),
847 edge.y + (edge.axis == building::EdgeAxis::Vertical ? 1 : 0),
848 bx, by);
849 placed.worldX = (ax + bx) * 0.5;
850 placed.worldY = (ay + by) * 0.5;
851 instances.push_back(instanceValue(placed));
852 operation.affectedObjects.push_back(
853 {TargetId(id_), std::to_string(placed.instanceId), 0});
854 }
855 operation.payload = instances;
856 operation.inverse = std::move(instances);
857 operation.hasInverse = true;
858 return eve::editing::applied<DomainOperation>(std::move(operation));

861EditorResult<DomainOperation> BuildingPlacementTarget::makeEdgeCubicBezier(
862 const std::string& buildingId,
863 const std::vector<BuildingEdgeCurvePoint>& controlPoints,
864 int subdivisions) const {
865 std::vector<building::PlacementSystem::EdgeCurvePoint> runtimeControls;
866 runtimeControls.reserve(controlPoints.size());
867 for (const BuildingEdgeCurvePoint& point : controlPoints)
868 runtimeControls.push_back(
869 {static_cast<float>(point.x), static_cast<float>(point.y)});
870 auto vertices =
871 building::PlacementSystem::sampleEdgeCubicBezier(runtimeControls, subdivisions);
872 if (!vertices.ok())
873 return buildingError<DomainOperation>(EditorStatus::Rejected,
874 "editor.building.edge-curve-rejected",
875 "Cubic edge curve sampling was rejected");
876 std::vector<BuildingEdgePathVertex> editorVertices;
877 editorVertices.reserve(vertices.value().size());
878 for (const building::CornerAddress& vertex : vertices.value())
879 editorVertices.push_back({vertex.x, vertex.y});
880 auto path = makeEdgePath(buildingId, editorVertices);
881 if (!path.ok()) return path;
882 auto instances = parseInstances(path.value().payload);
883 if (!instances.ok())
884 return buildingError<DomainOperation>(EditorStatus::Failed,
885 "editor.building.edge-curve-encoding-failed",
886 "Edge curve instances could not be encoded");
887 building::EdgeCurveGroup group;
888 group.id = world_->nextEdgeCurveGroupId();
889 group.buildingId = buildingId;
890 group.level = instances.value().front().level;
891 group.subdivisions = subdivisions;
892 for (const BuildingEdgeCurvePoint& point : controlPoints)
893 group.controlPoints.push_back(
894 {static_cast<float>(point.x), static_cast<float>(point.y)});
895 for (const BuildingInstanceSnapshot& instance : instances.value())
896 group.instanceIds.push_back(instance.instanceId);
897 DomainOperation operation = std::move(path).takeValue();
898 operation.type = "building.edge-curve.set.v1";
899 operation.inverseType = "building.edge-curve.delete.v1";
900 operation.payload = edgeCurveValue(group, instances.value());
901 operation.inverse = operation.payload;
902 return eve::editing::applied<DomainOperation>(std::move(operation));
903}

941BuildingEdgeCurvePreview BuildingPlacementTarget::previewEdgeCubicBezier(
942 const std::string& buildingId,
943 const std::vector<BuildingEdgeCurvePoint>& controlPoints, int subdivisions,
944 int replacingMemberInstanceId, const std::string& surfaceName) const {
945 BuildingEdgeCurvePreview result;
946 result.worldRevision = revision_;
947 result.controlPoints = controlPoints;
948 if (!world_) {
949 result.status = EditorStatus::Rejected;
950 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
951 eve::DiagnosticCode::PreconditionViolation,
952 RuleId("editor.building.world-required"), DiagnosticSeverity::Error,
953 "Placement world is unavailable"));
954 return result;
955 }
956 std::vector<building::PlacementSystem::EdgeCurvePoint> controls;
957 for (const BuildingEdgeCurvePoint& point : controlPoints)
958 controls.push_back({static_cast<float>(point.x), static_cast<float>(point.y)});
959 auto sampled = building::PlacementSystem::sampleEdgeCubicBezier(controls, subdivisions);
960 if (!sampled.ok()) {
961 result.status = EditorStatus::Rejected;
962 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
963 eve::DiagnosticCode::InvalidArgument,
964 RuleId("editor.building.edge-curve-invalid"), DiagnosticSeverity::Error,
965 "Cubic edge curve controls or subdivisions are invalid"));
966 return result;
967 }
968 for (const building::CornerAddress& vertex : sampled.value())
969 result.sampledVertices.push_back({vertex.x, vertex.y});
970 std::unique_ptr<building::PlacementWorld> previewOwner;
971 building::PlacementWorld* previewWorld = world_;
972 if (replacingMemberInstanceId > 0) {
973 auto replacingGroup = world_->edgeCurveGroupForInstance(replacingMemberInstanceId);
974 if (!replacingGroup.ok()) {
975 result.status = EditorStatus::NotFound;
976 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
977 eve::DiagnosticCode::NotFound,
978 RuleId("editor.building.edge-curve-group-not-found"),
979 DiagnosticSeverity::Error,
980 "Replacement preview instance does not belong to a curve group"));
981 return result;
982 }
983 previewOwner = world_->cloneState();
984 for (int instanceId : replacingGroup.value().instanceIds)
985 building::PlacementSystem::removeBuilding(previewOwner.get(), instanceId);
986 previewWorld = previewOwner.get();
987 }
988 auto preview = building::PlacementSystem::previewEdgeCubicBezier(
989 previewWorld, buildingId, controls, subdivisions);
990 if (!preview.ok()) {
991 result.status = EditorStatus::Rejected;
992 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
993 preview.error() ? preview.error()->code()
994 : eve::DiagnosticCode::PreconditionViolation,
995 RuleId("editor.building.edge-curve-rejected"), DiagnosticSeverity::Error,
996 preview.error() ? preview.error()->message()
997 : "Cubic edge curve preview was rejected"));
998 return result;
999 }
1000 for (const building::EdgeAddress& edge : preview.value().edges)
1001 result.edges.push_back(
1002 {edge.x, edge.y,
1003 edge.axis == building::EdgeAxis::Horizontal ? "horizontal" : "vertical"});
1004 if (!surfaceName.empty()) {
1005 auto surface = building::PlacementSystem::sampleEdgeCurveSurface(
1006 *world_, surfaceName, controls, subdivisions);
1007 if (!surface.ok()) {
1008 result.status = EditorStatus::Rejected;
1009 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
1010 surface.error() ? surface.error()->code()
1011 : eve::DiagnosticCode::PreconditionViolation,
1012 RuleId("editor.building.edge-curve-surface-rejected"),
1013 DiagnosticSeverity::Error,
1014 surface.error() ? surface.error()->message()
1015 : "Curve surface preview was rejected"));
1016 return result;
1017 }
1018 result.surfaceProviderName = surface.value().providerName;
1019 result.surfaceId = surface.value().surfaceId;
1020 result.surfaceRevision = surface.value().surfaceRevision;
1021 for (const building::EdgeCurveSurfaceSample& sample : surface.value().samples)
1022 result.surfaceSamples.push_back(
1023 {sample.worldX, sample.worldY, sample.worldZ, sample.normalX,
1024 sample.normalY, sample.normalZ});
1025 }
1026 result.status = EditorStatus::Applied;
1027 return result;
1028}
1029
1030editing::GizmoSnapshot BuildingPlacementTarget::edgeCubicBezierGizmo(
1031 const std::string& buildingId,
1032 const std::vector<BuildingEdgeCurvePoint>& controlPoints, int subdivisions,
1033 int level, int replacingMemberInstanceId, const std::string& surfaceName) const {
1034 editing::GizmoSnapshot gizmo;
1035 gizmo.target = id_;
1036 gizmo.targetRevision = revision_;
1037 const BuildingEdgeCurvePreview preview =
1038 previewEdgeCubicBezier(buildingId, controlPoints, subdivisions,
1039 replacingMemberInstanceId, surfaceName);
1040 gizmo.status = preview.status;
1041 gizmo.diagnostics = preview.diagnostics;
1042 if (!world_) return gizmo;
1043 const auto project = [&](double gx, double gy) {
1044 float ox = 0.f, oy = 0.f, xx = 0.f, xy = 0.f, yx = 0.f, yy = 0.f;
1045 world_->cellToWorldPlane(0, 0, ox, oy);
1046 world_->cellToWorldPlane(1, 0, xx, xy);
1047 world_->cellToWorldPlane(0, 1, yx, yy);
1048 const double px = ox + gx * (xx - ox) + gy * (yx - ox);
1049 const double py = oy + gx * (xy - oy) + gy * (yy - oy);
1050 const double elevation = static_cast<double>(level) * world_->getFloorHeight();
1051 if (!surfaceName.empty()) {
1052 auto hit = building::PlacementSystem::sampleSurface(
1053 *world_, surfaceName, static_cast<float>(px), static_cast<float>(py));
1054 if (hit.ok())
1055 return std::array<double, 3>{hit.value().worldX, hit.value().worldY,
1056 hit.value().worldZ};
1057 }
1058 return world_->getGridPlaneName() == "xz"
1059 ? std::array<double, 3>{px, elevation, py}
1060 : std::array<double, 3>{px, py, elevation};
1061 };
1062 const auto addLine = [&](const std::string& id, const std::array<double, 3>& a,
1063 const std::array<double, 3>& b,
1064 const std::array<double, 4>& color, bool dashed) {
1065 editing::GizmoPrimitive line;
1066 line.id = id;
1067 line.kind = "line";
1068 for (size_t axis = 0; axis < 3; ++axis) {
1069 line.position[axis] = (a[axis] + b[axis]) * 0.5;
1070 line.direction[axis] = b[axis] - a[axis];
1071 }
1072 line.length = std::sqrt(line.direction[0] * line.direction[0] +
1073 line.direction[1] * line.direction[1] +
1074 line.direction[2] * line.direction[2]);
1075 if (line.length > 0.0)
1076 for (double& component : line.direction) component /= line.length;
1077 line.color = color;
1078 line.dashed = dashed;
1079 gizmo.primitives.push_back(std::move(line));
1080 };
1081 const std::array<double, 4> accepted{0.2, 0.95, 0.35, 1.0};
1082 const std::array<double, 4> rejected{0.95, 0.2, 0.2, 1.0};
1083 const auto& pathColor = preview.status == EditorStatus::Applied ? accepted : rejected;
1084 for (size_t index = 0; index < controlPoints.size(); ++index) {
1085 editing::GizmoPrimitive handle;
1086 handle.id = "curve.control." + std::to_string(index);
1087 handle.kind = "sphere";
1088 handle.position = project(controlPoints[index].x, controlPoints[index].y);
1089 handle.radius = world_->getCellSize() * 0.15;
1090 handle.color = index == 0 || index + 1 == controlPoints.size()
1091 ? std::array<double, 4>{0.2, 0.65, 1.0, 1.0}
1092 : std::array<double, 4>{1.0, 0.7, 0.15, 1.0};
1093 gizmo.primitives.push_back(std::move(handle));
1094 if (index > 0)
1095 addLine("curve.tangent." + std::to_string(index - 1),
1096 project(controlPoints[index - 1].x, controlPoints[index - 1].y),
1097 project(controlPoints[index].x, controlPoints[index].y),
1098 {0.65, 0.65, 0.7, 0.8}, true);
1099 }
1100 if (!preview.surfaceSamples.empty()) {
1101 for (size_t index = 1; index < preview.surfaceSamples.size(); ++index) {
1102 const auto& a = preview.surfaceSamples[index - 1];
1103 const auto& b = preview.surfaceSamples[index];
1104 addLine("curve.path." + std::to_string(index - 1),
1105 {a.worldX, a.worldY, a.worldZ}, {b.worldX, b.worldY, b.worldZ},
1106 pathColor, false);
1107 }
1108 } else {
1109 for (size_t index = 1; index < preview.sampledVertices.size(); ++index)
1110 addLine("curve.path." + std::to_string(index - 1),
1111 project(preview.sampledVertices[index - 1].x,
1112 preview.sampledVertices[index - 1].y),
1113 project(preview.sampledVertices[index].x,
1114 preview.sampledVertices[index].y),
1115 pathColor, false);
1116 }
1117 return gizmo;
1118}

1176EditorResult<BuildingEdgeCurvePoint> BuildingPlacementTarget::curveLogicalPointFromWorld(
1177 double worldX, double worldY, double worldZ) const {
1178 if (!world_ || !std::isfinite(worldX) || !std::isfinite(worldY) ||
1179 !std::isfinite(worldZ))
1180 return buildingError<BuildingEdgeCurvePoint>(
1181 EditorStatus::Rejected, "editor.building.curve-world-point-invalid",
1182 "Curve drag requires a finite world-space point and an available world");
1183 float ox = 0.f, oy = 0.f, xx = 0.f, xy = 0.f, yx = 0.f, yy = 0.f;
1184 world_->cellToWorldPlane(0, 0, ox, oy);
1185 world_->cellToWorldPlane(1, 0, xx, xy);
1186 world_->cellToWorldPlane(0, 1, yx, yy);
1187 const double bxX = xx - ox;
1188 const double bxY = xy - oy;
1189 const double byX = yx - ox;
1190 const double byY = yy - oy;
1191 const double determinant = bxX * byY - bxY * byX;
1192 if (std::abs(determinant) < 1e-9)
1193 return buildingError<BuildingEdgeCurvePoint>(
1194 EditorStatus::Unsupported, "editor.building.curve-grid-basis-singular",
1195 "Curve drag requires an invertible affine grid basis");
1196 const double planeX = worldX;
1197 const double planeY = world_->getGridPlaneName() == "xz" ? worldZ : worldY;
1198 const double dx = planeX - ox;
1199 const double dy = planeY - oy;
1200 BuildingEdgeCurvePoint point;
1201 point.x = (dx * byY - dy * byX) / determinant;
1202 point.y = (bxX * dy - bxY * dx) / determinant;
1203 return eve::editing::applied<BuildingEdgeCurvePoint>(point);
1204}

1214EditorResult<void> BuildingPlacementTarget::applyDomainOperation(const DomainOperation& operation) {
1215 if (!world_)
1216 return buildingError<void>(EditorStatus::Rejected, "editor.building.world-required",
1217 "Placement world is unavailable");

1219 return buildingError<void>(EditorStatus::Rejected, "editor.building.target-mismatch",
1220 "Building operation targets another world");

1234 if (!buildingCandidate)

1282 auto* buildingCandidate = dynamic_cast<BuildingPlacementTarget*>(candidate.get());
1283 if (!buildingCandidate)

1289 id_, areaSet ? "building.instance.set.v4" : "building.instance.delete.v4",

1300 if (!parsed.ok())

1495EditorResult<BuildingEdgeCurveDragPreview> BuildingEdgeCurveDragSession::beginDrag(
1496 double rayOriginX, double rayOriginY, double rayOriginZ, double rayDirectionX,
1497 double rayDirectionY, double rayDirectionZ) {
1498 cancelDrag();
1499 if (!target_ || memberInstanceId_ <= 0 || controls_.size() != 4)
1500 return buildingError<BuildingEdgeCurveDragPreview>(
1501 EditorStatus::Rejected, "editor.building.curve-drag-state-invalid",
1502 "Curve drag requires an existing group member and four controls");
1503 const double directionLength = std::sqrt(rayDirectionX * rayDirectionX +
1504 rayDirectionY * rayDirectionY +
1505 rayDirectionZ * rayDirectionZ);
1506 if (!std::isfinite(directionLength) || directionLength < 1e-9)
1507 return buildingError<BuildingEdgeCurveDragPreview>(
1508 EditorStatus::Rejected, "editor.building.curve-drag-ray-invalid",
1509 "Curve drag requires a finite non-zero pointer ray");
1510 const std::array<double, 3> origin{rayOriginX, rayOriginY, rayOriginZ};
1511 const std::array<double, 3> direction{rayDirectionX / directionLength,
1512 rayDirectionY / directionLength,
1513 rayDirectionZ / directionLength};
1514 const editing::GizmoSnapshot gizmo = target_->edgeCubicBezierGizmo(
1515 buildingId_, controls_, subdivisions_, 0, memberInstanceId_, surfaceName_);
1516 double closest = std::numeric_limits<double>::infinity();
1517 for (const editing::GizmoPrimitive& primitive : gizmo.primitives) {
1518 if (primitive.kind != "sphere" || primitive.id.rfind("curve.control.", 0) != 0)
1519 continue;
1520 const std::array<double, 3> offset{origin[0] - primitive.position[0],
1521 origin[1] - primitive.position[1],
1522 origin[2] - primitive.position[2]};
1523 const double b = offset[0] * direction[0] + offset[1] * direction[1] +
1524 offset[2] * direction[2];
1525 const double c = offset[0] * offset[0] + offset[1] * offset[1] +
1526 offset[2] * offset[2] - primitive.radius * primitive.radius;
1527 const double discriminant = b * b - c;
1528 if (discriminant < 0.0) continue;
1529 const double nearHit = -b - std::sqrt(discriminant);
1530 const double farHit = -b + std::sqrt(discriminant);
1531 const double hit = nearHit >= 0.0 ? nearHit : farHit;
1532 if (hit < 0.0 || hit >= closest) continue;
1533 closest = hit;
1534 activeControlPointIndex_ = std::stoi(primitive.id.substr(14));
1535 dragPlanePoint_ = primitive.position;
1536 }
1537 if (activeControlPointIndex_ < 0)
1538 return buildingError<BuildingEdgeCurveDragPreview>(
1539 EditorStatus::NotFound, "editor.building.curve-control-not-hit",
1540 "Pointer ray did not hit a curve control handle");
1541 dragPlaneNormal_ = direction;
1542 baseRevision_ = target_->revision();
1543 dragging_ = true;
1544 lastPreviewValid_ = gizmo.status == EditorStatus::Applied;
1545 return previewCurrent();
1546}
