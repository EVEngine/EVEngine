52EditorResult<AnimationTransformKey> parseKey(const EditorValue& value) {
53 const auto* id = stringField(value, "id");
54 double time = 0, px = 0, py = 0, pz = 0, rx = 0, ry = 0, rz = 0, rw = 0, sx = 0, sy = 0, sz = 0;
55 if (!id || id->empty() || !readNumber(value, "time", time) || !readNumber(value, "px", px) ||
56 !readNumber(value, "py", py) || !readNumber(value, "pz", pz) || !readNumber(value, "rx", rx) ||
57 !readNumber(value, "ry", ry) || !readNumber(value, "rz", rz) || !readNumber(value, "rw", rw) ||
58 !readNumber(value, "sx", sx) || !readNumber(value, "sy", sy) || !readNumber(value, "sz", sz))
59 return clipError<AnimationTransformKey>(EditorStatus::Rejected, "editor.animation.invalid-key",
60 "Transform key requires a stable id, time and complete TRS");
61 const double values[]{time, px, py, pz, rx, ry, rz, rw, sx, sy, sz};
62 for (double component : values)
63 if (!std::isfinite(component))
64 return clipError<AnimationTransformKey>(EditorStatus::Rejected, "editor.animation.nonfinite-key",
65 "Transform key components must be finite");
66 const double length = std::sqrt(rx * rx + ry * ry + rz * rz + rw * rw);
67 if (time < 0.0 || sx <= 0.0 || sy <= 0.0 || sz <= 0.0 || length < 1e-8)
68 return clipError<AnimationTransformKey>(EditorStatus::Rejected, "editor.animation.invalid-key-range",
69 "Key time must be non-negative, scale positive and rotation non-zero");
70 return eve::editing::applied<AnimationTransformKey>({StableId(*id), time, px, py, pz,
71 rx / length, ry / length, rz / length, rw / length,
72 sx, sy, sz});
73}

81EditorResult<AnimationBoneTrack> parseTrack(const EditorValue& value) {
82 const auto* id = stringField(value, "id");
83 const auto* bone = stringField(value, "bone");
84 const EditorValue* keyEntry = field(value, "keys");
85 const auto* keys = keyEntry ? keyEntry->getIf<EditorValue::Array>() : nullptr;
86 if (!id || id->empty() || !bone || bone->empty() || !keys)
87 return clipError<AnimationBoneTrack>(EditorStatus::Rejected, "editor.animation.invalid-track",
88 "Bone track requires stable id, bone name and keys");
89 AnimationBoneTrack result{StableId(*id), *bone, {}};
90 std::set<StableId> ids;
91 for (const auto& entry : *keys) {
92 auto key = parseKey(entry);
93 if (!key.ok() || !ids.insert(key.value().id).second)
94 return clipError<AnimationBoneTrack>(EditorStatus::Rejected, "editor.animation.duplicate-key",
95 "Transform key ids must be unique inside a track");
96 result.keys.push_back(std::move(key).value());
97 }
98 std::sort(result.keys.begin(), result.keys.end(), [](const auto& a, const auto& b) {
99 return a.time == b.time ? a.id < b.id : a.time < b.time;
100 });
101 return eve::editing::applied<AnimationBoneTrack>(std::move(result));
102}

168EditorResult<void> AnimationClipDocumentTarget::applyDomainOperation(const DomainOperation& op) {
169 if (op.target != TargetId(id_))
170 return clipError<void>(EditorStatus::Rejected, "editor.animation.target-mismatch", "Operation targets another clip");
171 if (op.type == "animation.clip.settings.v1") {
172 double duration = 0, rate = 0;
173 const EditorValue* loopEntry = field(op.payload, "loop"); const auto* loop = loopEntry ? loopEntry->getIf<bool>() : nullptr;
174 if (!readNumber(op.payload, "duration", duration) || !readNumber(op.payload, "sampleRate", rate) ||
175 !loop || !std::isfinite(duration) || !std::isfinite(rate) || duration <= 0.0 || rate <= 0.0)
176 return clipError<void>(EditorStatus::Rejected, "editor.animation.invalid-settings", "Duration and sample rate must be positive");
177 duration_ = duration; sampleRate_ = rate; loop_ = *loop;
178 } else if (op.type == "animation.clip.track.set.v1") {
179 auto track = parseTrack(op.payload); if (!track.ok()) return clipError<void>(track.code(), "editor.animation.invalid-track", "Track payload is invalid");
180 for (const auto& key : track.value().keys)
181 if (key.time > duration_) return clipError<void>(EditorStatus::Rejected, "editor.animation.key-after-duration", "Track key exceeds clip duration");
182 for (const auto& [id, current] : tracks_)
183 if (id != track.value().id && current.bone == track.value().bone)
184 return clipError<void>(EditorStatus::Conflict, "editor.animation.duplicate-bone-track", "A bone may have only one track");
185 auto trackRecord = std::move(track).value(); tracks_[trackRecord.id] = std::move(trackRecord);
186 } else if (op.type == "animation.clip.track.delete.v1") {
187 auto track = parseTrack(op.payload); if (!track.ok() || !tracks_.erase(track.value().id))
188 return clipError<void>(EditorStatus::NotFound, "editor.animation.track-not-found", "Bone track was not found");
189 } else if (op.type == "animation.clip.event.set.v1") {
190 auto event = parseEvent(op.payload); if (!event.ok()) return clipError<void>(event.code(), "editor.animation.invalid-event", "Event payload is invalid");
191 if (event.value().time > duration_) return clipError<void>(EditorStatus::Rejected, "editor.animation.event-after-duration", "Event exceeds clip duration");
192 auto eventRecord = std::move(event).value(); events_[eventRecord.id] = std::move(eventRecord);
193 } else if (op.type == "animation.clip.event.delete.v1") {
194 auto event = parseEvent(op.payload); if (!event.ok() || !events_.erase(event.value().id))
195 return clipError<void>(EditorStatus::NotFound, "editor.animation.event-not-found", "Event was not found");
196 } else if (op.type == "animation.clip.mask.set.v1") {
197 const auto* bone = stringField(op.payload, "bone"); double weight = 0;
198 if (!bone || bone->empty() || !readNumber(op.payload, "weight", weight) || !std::isfinite(weight) ||
199 weight < 0.0 || weight > 1.0)
200 return clipError<void>(EditorStatus::Rejected, "editor.animation.invalid-mask", "Mask weight must be in [0, 1]");
201 mask_[*bone] = weight;
202 } else {
203 return clipError<void>(EditorStatus::Unsupported, "editor.animation.unsupported-operation", "Unsupported clip operation");
204 }
205 ++revision_; dirty_.include(0, 0); return eve::editing::applied<void>();
206}

281AnimationClipPreview AnimationClipDocumentTarget::preview(double time, const std::vector<std::string>& bones) const {
282 AnimationClipPreview result; result.documentRevision = revision_;
283 if (!std::isfinite(time)) { result.diagnostics.push_back(clipDiagnostic("editor.animation.invalid-preview-time", DiagnosticSeverity::Error, "Preview time must be finite")); return result; }
284 result.time = loop_ ? std::fmod(std::max(0.0, time), duration_) : std::clamp(time, 0.0, duration_);
285 result.diagnostics = validate(bones);
286 for (const auto& [id, track] : tracks_) { static_cast<void>(id); if (track.keys.empty()) continue;
287 auto upper = std::upper_bound(track.keys.begin(), track.keys.end(), result.time, [](double t, const auto& key) { return t < key.time; });
288 const auto& b = upper == track.keys.end() ? track.keys.back() : *upper;
289 const auto& a = upper == track.keys.begin() ? track.keys.front() : *(upper - 1);
290 const double amount = b.time > a.time ? (result.time - a.time) / (b.time - a.time) : 0.0;
291 AnimationSampledBone sample; sample.bone = track.bone;
292 sample.positionX = mix(a.positionX, b.positionX, amount); sample.positionY = mix(a.positionY, b.positionY, amount); sample.positionZ = mix(a.positionZ, b.positionZ, amount);
293 sample.rotationX = mix(a.rotationX, b.rotationX, amount); sample.rotationY = mix(a.rotationY, b.rotationY, amount); sample.rotationZ = mix(a.rotationZ, b.rotationZ, amount); sample.rotationW = mix(a.rotationW, b.rotationW, amount);
294 const double length = std::sqrt(sample.rotationX * sample.rotationX + sample.rotationY * sample.rotationY + sample.rotationZ * sample.rotationZ + sample.rotationW * sample.rotationW);
295 if (length > 1e-8) { sample.rotationX /= length; sample.rotationY /= length; sample.rotationZ /= length; sample.rotationW /= length; }
296 sample.scaleX = mix(a.scaleX, b.scaleX, amount); sample.scaleY = mix(a.scaleY, b.scaleY, amount); sample.scaleZ = mix(a.scaleZ, b.scaleZ, amount);
297 const auto mask = mask_.find(track.bone); sample.maskWeight = mask == mask_.end() ? 1.0 : mask->second; result.bones.push_back(std::move(sample));
298 }
299 result.status = std::any_of(result.diagnostics.begin(), result.diagnostics.end(), [](const auto& d) { return d.severity() == DiagnosticSeverity::Error; }) ? EditorStatus::Failed : EditorStatus::Applied;
300 return result;
301}

303AnimationRetargetPreview AnimationClipDocumentTarget::previewRetarget(const std::vector<std::string>& targetBones) const {
304 AnimationRetargetPreview result; result.documentRevision = revision_; std::set<std::string> used;
305 for (const auto& [id, track] : tracks_) { static_cast<void>(id); auto exact = std::find(targetBones.begin(), targetBones.end(), track.bone);
306 auto found = exact; if (found == targetBones.end()) found = std::find_if(targetBones.begin(), targetBones.end(), [&](const auto& target) { return normalized(target) == normalized(track.bone); });
307 if (found == targetBones.end() || used.contains(*found)) result.unmatchedSourceBones.push_back(track.bone); else { result.mapping[track.bone] = *found; used.insert(*found); }
308 }
309 for (const auto& target : targetBones) if (!used.contains(target)) result.unmatchedTargetBones.push_back(target);
310 if (!result.unmatchedSourceBones.empty()) result.diagnostics.push_back(clipDiagnostic("editor.animation.unmatched-source-bones", DiagnosticSeverity::Warning, "Some source tracks cannot be mapped"));
311 if (!result.unmatchedTargetBones.empty()) result.diagnostics.push_back(clipDiagnostic("editor.animation.unmatched-target-bones", DiagnosticSeverity::Info, "Some target bones remain at bind pose"));
312 result.status = EditorStatus::Applied; return result;

322EditorResult<void> AnimationClipDocumentTarget::loadSnapshot(const EditorValue& snapshot) {
323 const EditorValue* versionEntry = field(snapshot, "schemaVersion"); const auto* version = versionEntry ? versionEntry->getIf<int64_t>() : nullptr;
324 const EditorValue* settings = field(snapshot, "settings"); const EditorValue* tracksEntry = field(snapshot, "tracks"); const EditorValue* eventsEntry = field(snapshot, "events"); const EditorValue* masksEntry = field(snapshot, "masks");
325 const auto* tracks = tracksEntry ? tracksEntry->getIf<EditorValue::Array>() : nullptr; const auto* events = eventsEntry ? eventsEntry->getIf<EditorValue::Array>() : nullptr; const auto* masks = masksEntry ? masksEntry->getIf<EditorValue::Array>() : nullptr;
326 if (!version || *version != 1 || !settings || !tracks || !events || !masks) return clipError<void>(EditorStatus::Unsupported, "editor.animation.invalid-snapshot", "Clip snapshot schema is unsupported");
327 double duration = 0, rate = 0;
328 const EditorValue* loopEntry = field(*settings, "loop"); const auto* loop = loopEntry ? loopEntry->getIf<bool>() : nullptr;
329 if (!readNumber(*settings, "duration", duration) || !readNumber(*settings, "sampleRate", rate) || !loop)
330 return clipError<void>(EditorStatus::Rejected, "editor.animation.invalid-settings", "Clip snapshot settings are incomplete");
331 AnimationClipDocumentTarget candidate(id_); auto settingsOp = candidate.makeSetSettings(duration, rate, *loop);
