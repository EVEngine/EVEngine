60EditorResult<void> Renderable3DMaterialRuntimeSink::publish(
61 const MaterialDocumentTarget& candidate) {
62 if (!impl_->assets)
63 return runtimeError<void>(EditorStatus::Rejected, "editor.material.runtime-input",
64 "Material asset resolver is required");
65 auto* renderable = dynamic_cast<graphics::Renderable3D*>(ecs::try_get(impl_->handle));
66 if (!renderable)
67 return runtimeError<void>(EditorStatus::Conflict, "editor.material.runtime-stale",
68 "Renderable3D handle is missing or stale");
69 const auto diagnostics = candidate.validate();
70 for (const EditorDiagnostic& diagnostic : diagnostics) {
71 if (diagnostic.severity() == DiagnosticSeverity::Error)
72 return EditorResult<void>::failure(eve::Status(EditorStatus::Rejected, diagnostics));
73 }
74 EditorValue snapshot;
75 const auto* values = properties(candidate, snapshot);
76 if (!values)
77 return runtimeError<void>(EditorStatus::Failed, "editor.material.runtime-properties",
78 "Material properties are unavailable");
79 if (*value<std::string>(*values, "shading.model") != "pbr" ||
80 *value<std::string>(*values, "surface.mode") != "opaque" ||
81 *value<std::string>(*values, "surface.blend") != "alpha" ||
82 *value<bool>(*values, "surface.double-sided"))
83 return runtimeError<void>(EditorStatus::Unsupported,
84 "editor.material.runtime-legacy-surface",
85 "Legacy Renderable3D supports PBR opaque single-sided materials only");
86
87 auto albedo = resolveAsset<graphics::Texture>(
88 *value<std::string>(*values, "textures.albedo"),
89 [&](const std::string& asset) { return impl_->assets->resolveTexture(asset); });
90 auto normal = resolveAsset<graphics::Texture>(
91 *value<std::string>(*values, "textures.normal"),
92 [&](const std::string& asset) { return impl_->assets->resolveTexture(asset); });
93 auto height = resolveAsset<graphics::Texture>(
94 *value<std::string>(*values, "textures.height"),
95 [&](const std::string& asset) { return impl_->assets->resolveTexture(asset); });
96 auto shader = resolveAsset<graphics::Shader>(
97 *value<std::string>(*values, "textures.shader"),
98 [&](const std::string& asset) { return impl_->assets->resolveShader(asset); });
99 for (const auto* result : {static_cast<const EditorResult<graphics::Texture*>*>(&albedo),
100 static_cast<const EditorResult<graphics::Texture*>*>(&normal),
101 static_cast<const EditorResult<graphics::Texture*>*>(&height)}) {
102 if (!result->ok()) return EditorResult<void>::failure(result->status());
103 }
104 if (!shader.ok()) return EditorResult<void>::failure(shader.status());
105
106 const auto& tint = *value<EditorValue::Array>(*values, "shading.tint");
107 renderable->setTexture(albedo.value());
108 renderable->setNormalTexture(normal.value());
109 renderable->setHeightTexture(height.value());
110 renderable->setShader(shader.value());
111 renderable->setTint(static_cast<float>(number(tint[0])),
112 static_cast<float>(number(tint[1])),
113 static_cast<float>(number(tint[2])),
114 static_cast<float>(number(tint[3])));
115 renderable->setMetallic(static_cast<float>(*value<double>(*values, "shading.metallic")));
116 renderable->setRoughness(static_cast<float>(*value<double>(*values, "shading.roughness")));
117 renderable->setParallax(static_cast<float>(*value<double>(*values, "parallax.scale")));
118 renderable->setReceiveLight(*value<bool>(*values, "lighting.receive"));
119 renderable->setCastShadow(*value<bool>(*values, "shadow.cast"));
120 renderable->setReceiveShadow(*value<bool>(*values, "shadow.receive"));
121 return eve::editing::applied<void>();
122}
