10eve::Result<eve::SimulationStep> makeLegacyStep(float dt, eve::SimulationTick currentTick) {
11 if (!std::isfinite(dt) || dt < 0.f) {
12 return eve::Result<eve::SimulationStep>::failure(
13 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
14 "World3D update dt must be finite and non-negative", "physics.world3d.update.dt"));
15 }
16 const auto nextTick = currentTick.incremented();
17 if (!nextTick) {
18 return eve::Result<eve::SimulationStep>::failure(
19 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
20 "World3D simulation tick cannot be incremented", "physics.world3d.simulationTick"));
21 }
22 auto duration = eve::Duration::fromSeconds(static_cast<double>(std::min(dt, 0.05f)));
23 if (!duration) return eve::Result<eve::SimulationStep>::failure(duration.status());
24 return eve::Result<eve::SimulationStep>::success({*nextTick, std::move(duration).takeValue()});
25}

41 if (!isValid() || !simulation_ || couplingStepActive_) {

46 auto forceStep = validateForceStep(stepValue);
