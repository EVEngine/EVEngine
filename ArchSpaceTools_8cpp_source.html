14bool intersectFloor(const ArchSpaceViewportRay& ray, double elevation, archspace::Vec2& out) {
15 if (std::abs(ray.direction[1]) <= kEps) return false;
16 const double t = (elevation - ray.origin[1]) / ray.direction[1];
17 if (t < 0.0) return false;
18 out.x = ray.origin[0] + ray.direction[0] * t;
19 out.z = ray.origin[2] + ray.direction[2] * t;
20 return std::isfinite(out.x) && std::isfinite(out.z);
21}

23archspace::Vec2 snapOrtho(const archspace::Vec2& start, const archspace::Vec2& end) {
24 if (std::abs(end.x - start.x) >= std::abs(end.z - start.z)) return {end.x, start.z};
25 return {start.x, end.z};
26}

28bool projectOntoWall(const archspace::Node& wall, const archspace::Vec2& point, double& t, double& dist) {
29 const double dx = wall.end.x - wall.start.x;
30 const double dz = wall.end.z - wall.start.z;
31 const double len2 = dx * dx + dz * dz;
32 if (len2 <= kEps) return false;
33 t = std::clamp(((point.x - wall.start.x) * dx + (point.z - wall.start.z) * dz) / len2, 0.0, 1.0);
34 const double px = wall.start.x + dx * t;
35 const double pz = wall.start.z + dz * t;
36 dist = std::sqrt((point.x - px) * (point.x - px) + (point.z - pz) * (point.z - pz));
37 return true;
38}

40bool commitDomain(editing::IEditAuthority* authority, eve::archspace_editing::ArchSpaceDocumentTarget& target,
41 editing::DomainOperation operation, const ToolDescriptor& descriptor, std::uint64_t sequence,
42 std::string mergeKey, std::optional<editing::TransactionReceipt>& outReceipt) {
43 if (!authority) return false;
44 editing::TransactionSpec transaction;
45 transaction.id = editing::TransactionId(descriptor.id + "." + std::to_string(sequence));
46 transaction.label = descriptor.label;
47 transaction.origin = editing::ActionOrigin::User;
48 transaction.target = target.targetId();
49 transaction.baseRevision = target.revision();
50 transaction.mergeKey = std::move(mergeKey);
51 const std::array<editing::DomainOperation, 1> operations{std::move(operation)};
52 auto plan = authority->preflight(transaction, std::span<const editing::DomainOperation>(operations));
53 if (!plan.ok()) return false;
54 auto receipt = authority->commit(plan.value());
55 if (!receipt.ok()) return false;
56 outReceipt = std::move(receipt.value());
57 return true;
58}

193std::optional<ArchSpaceOpeningPlaceTool::Pick> ArchSpaceOpeningPlaceTool::pickWall(
194 const eve::archspace_editing::ArchSpaceDocumentTarget& target, const archspace::Vec2& point) const {
195 std::optional<Pick> best;
196 double bestDist = 1e9;
197 for (const auto& [id, node] : target.document().nodes()) {
198 if (node.kind != archspace::NodeKind::Wall) continue;
199 double t = 0.5;
200 double dist = 0.0;
201 if (!projectOntoWall(node, point, t, dist)) continue;
202 if (dist > std::max(0.75, node.thickness * 2.0)) continue;
203 if (dist >= bestDist) continue;
204 bestDist = dist;
205 best = Pick{
206 id, t, {node.start.x + (node.end.x - node.start.x) * t, node.start.z + (node.end.z - node.start.z) * t}};
207 }
208 return best;
209}
