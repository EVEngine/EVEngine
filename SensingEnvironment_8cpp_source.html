8Result<EnvironmentRequest> SensingWorld::environmentInput(const QueryOrigin& origin, const QuerySpec& spec) const {
9 auto collected = collectCandidates(origin, spec);
10 if (!collected) return Result<EnvironmentRequest>::failure(collected.status());
11 // The spatial index's traversal order is not an identity. Canonicalize once at admission.
12 auto candidates = std::move(collected).takeValue().candidates;
13 std::sort(candidates.begin(), candidates.end(), [](const auto& a, const auto& b) { return a.id < b.id; });
14 EnvironmentRequest request;
15 for (const auto& candidate : candidates) {
16 auto location = WorldPoint::world2D(candidate.x, candidate.y);
17 if (!location) return Result<EnvironmentRequest>::failure(location.status());
18 request.input.push_back({request.input.size(), TargetLocation{location.value()}, candidate.id});
19 }
20 return Result<EnvironmentRequest>::success(std::move(request));
21}

23Result<EnvironmentQueryResult> SensingWorld::executeEnvironmentQuery(const EnvironmentQueryPlan& plan,
24 EnvironmentRequest request,
25 const EnvironmentServices& services) {
26 auto prepared = plan.instantiate(std::move(request));
27 if (!prepared) {
28 EnvironmentTrace trace;
29 trace.state = "rejected";
30 trace.diagnostics = prepared.diagnostics();
31 auto encoded = trace.toValue().toJson();
32 if (!encoded) return Result<EnvironmentQueryResult>::failure(encoded.status());
33 lastEnvironmentQuery_ = std::move(encoded).takeValue();
34 return Result<EnvironmentQueryResult>::failure(prepared.status());
35 }
36 auto instance = std::move(prepared).takeValue();
37 auto result = instance.run(services);
38 const bool ok = result.ok();
39 auto document = instance.trace().toValue();
40 if (ok) document.set("result", result.value().toValue());
41 auto encoded = document.toJson();
42 if (!encoded) return Result<EnvironmentQueryResult>::failure(encoded.status());
43 lastEnvironmentQuery_ = std::move(encoded).takeValue();
44 return result;

47Result<Value> SensingWorld::queryPositions(float originX, float originY, float targetX, float targetY, float radius,
48 int samples, int limit, bool requireLineOfSight) {
49 if (samples <= 0 || limit <= 0)
50 return Result<Value>::failure(Diagnostic::error(
51 DiagnosticCode::InvalidArgument, "samples and limit must be positive", "queryPositions", {}, "sensing"));
52 auto origin = WorldPoint::world2D(originX, originY);
53 if (!origin) return Result<Value>::failure(origin.status());
54 auto target = WorldPoint::world2D(targetX, targetY);
55 if (!target) return Result<Value>::failure(target.status());
56 EnvironmentQueryDefinition definition;
57 definition.id = "sensing.positions.ring";
58 definition.contexts = {"self", "target"};
59 definition.maxResults = static_cast<std::uint32_t>(limit);
60 definition.mode = limit == 1 ? EnvironmentRunMode::Best : EnvironmentRunMode::All;
61 definition.trace = EnvironmentTraceLevel::Detailed;
62 EnvironmentOption option;
63 option.id = "ring";
64 option.generator.kind = EnvironmentGeneratorKind::Ring;
65 option.generator.context = "target";
66 option.generator.radius = radius;
67 option.generator.samples = static_cast<std::uint32_t>(samples);
68 EnvironmentTest distance;
69 distance.id = "travelDistance";
70 distance.context = "self";
71 distance.purpose = EnvironmentPurpose::Score;
72 // Fixed metric scale preserves ordering when the terminal LOS filter is deferred.
73 distance.scoreMaximum = std::hypot(double(originX) - targetX, double(originY) - targetY) + double(radius) + 1;
74 option.tests.push_back(distance);
75 if (requireLineOfSight) {
76 EnvironmentTest los;
77 los.id = "visibleFromTarget";
78 los.context = "target";
79 los.metric = EnvironmentMetricKind::LineOfSight;
80 los.purpose = EnvironmentPurpose::Filter;
81 los.minimum = los.maximum = 1;
82 option.tests.push_back(los);
83 }
84 definition.options.push_back(std::move(option));
85 auto plan = EnvironmentQueryPlan::compile(definition);
86 if (!plan) return Result<Value>::failure(plan.status());
87 EnvironmentRequest request;
88 request.contexts = {{"self", {TargetLocation{origin.value()}}}, {"target", {TargetLocation{target.value()}}}};
89 EnvironmentServices services;
90 if (requireLineOfSight) {
91 // Resolve once at the compatibility/script composition boundary, never per item.
92 if (const auto* los = cap::query<ILineOfSightQuery>()) services.lineOfSight = std::cref(*los);
93 }
94 auto result = executeEnvironmentQuery(plan.value(), std::move(request), services);
95 if (!result) return Result<Value>::failure(result.status());
96 return Result<Value>::success(result.value().toValue());
97}
