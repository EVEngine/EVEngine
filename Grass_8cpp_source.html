40float radicalInverse(uint32_t n, uint32_t base) {
41 const float inv = 1.f / float(base);
42 float f = inv;
43 float val = 0.f;
44 while (n > 0) {
45 val += float(n % base) * f;
46 n /= base;
47 f *= inv;
48 }
49 return val;
50}

78bool buildTriangles(const float *posXYZ, const float *nrmXYZ, int vertexCount,
79 const uint32_t *indices, int indexCount, float minSlopeDot,
80 std::vector<Triangle> &tris, std::vector<float> &cdf) {
81 tris.clear();
82 cdf.clear();
83 if (!posXYZ || !indices || vertexCount < 3 || indexCount < 3) return false;
84 if (indexCount % 3 != 0) return false;
85
86 float total = 0.f;
87 for (int t = 0; t + 2 < indexCount; t += 3) {
88 const uint32_t i0 = indices[t];
89 const uint32_t i1 = indices[t + 1];
90 const uint32_t i2 = indices[t + 2];
91 if (int(i0) >= vertexCount || int(i1) >= vertexCount || int(i2) >= vertexCount) continue;
92 const glm::vec3 a = readPos(posXYZ, int(i0));
93 const glm::vec3 b = readPos(posXYZ, int(i1));
94 const glm::vec3 c = readPos(posXYZ, int(i2));
95 glm::vec3 n = glm::cross(b - a, c - a);
96 const float twiceArea = glm::length(n);
97 if (twiceArea < 1e-10f) continue;
98 n /= twiceArea;
99 if (nrmXYZ) {
100 glm::vec3 ns = readNrm(nrmXYZ, int(i0)) + readNrm(nrmXYZ, int(i1)) + readNrm(nrmXYZ, int(i2));
101 if (glm::dot(ns, ns) > 1e-8f) n = glm::normalize(ns);
102 }
103 if (n.y < minSlopeDot) continue;
104 Triangle tri;
105 tri.i0 = i0;
106 tri.i1 = i1;
107 tri.i2 = i2;
108 tri.area = 0.5f * twiceArea;
109 tri.n = n;
110 total += tri.area;
111 tris.push_back(tri);
112 cdf.push_back(total);
113 }
114 return total > 1e-12f && !tris.empty();
115}

151bool tooClose(const glm::vec3 &p, float radius,
152 const std::vector<Point> &accepted,
153 const std::unordered_map<GridKey, std::vector<int>, GridKeyHash> &grid, float cell) {
154 const GridKey c = toKey(p, cell);
155 const float r2 = radius * radius;
156 for (int dz = -1; dz <= 1; ++dz) {
157 for (int dy = -1; dy <= 1; ++dy) {
158 for (int dx = -1; dx <= 1; ++dx) {
159 GridKey k{c.x + dx, c.y + dy, c.z + dz};
160 auto it = grid.find(k);
161 if (it == grid.end()) continue;
162 for (int idx : it->second) {
163 const glm::vec3 d = accepted[size_t(idx)].position - p;
164 if (glm::dot(d, d) < r2) return true;
165 }
166 }
167 }
168 }
169 return false;
170}

185void stampOver(std::vector<uint8_t> &rgba, int w, int h, int x, int y, float luma, float a) {
186 if (x < 0 || y < 0 || x >= w || y >= h) return;
187 a = clampf(a, 0.f, 1.f);
188 luma = clampf(luma, 0.f, 1.f);
189 if (a < 0.02f) return;
190 const size_t i = (size_t(y) * size_t(w) + size_t(x)) * 4u;
191 const float oa = float(rgba[i + 3]) / 255.f;
192 const float or_ = float(rgba[i + 0]) / 255.f;
193 const float og = float(rgba[i + 1]) / 255.f;
194 const float ob = float(rgba[i + 2]) / 255.f;
195 const float outA = a + oa * (1.f - a);
196 if (outA < 1e-4f) return;
197 const float nr = (luma * a + or_ * oa * (1.f - a)) / outA;
198 const float ng = (luma * a + og * oa * (1.f - a)) / outA;
199 const float nb = (luma * a + ob * oa * (1.f - a)) / outA;
200 rgba[i + 0] = uint8_t(std::round(clampf(nr, 0.f, 1.f) * 255.f));
201 rgba[i + 1] = uint8_t(std::round(clampf(ng, 0.f, 1.f) * 255.f));
202 rgba[i + 2] = uint8_t(std::round(clampf(nb, 0.f, 1.f) * 255.f));
203 rgba[i + 3] = uint8_t(std::round(clampf(outA, 0.f, 1.f) * 255.f));
204}

206void stampBlade(std::vector<uint8_t> &rgba, int atlasW, int atlasH, int ox, int frameW, int frameH,
207 float baseU, float height, float lean, float baseHalf, float tipHalf, float luma0,
208 float luma1) {
209 const float fw = float(frameW);
210 const float fh = float(frameH);
211 height = clampf(height, 0.12f, 1.f);
212 for (int y = 0; y < frameH; ++y) {
213 for (int x = 0; x < frameW; ++x) {
214 const float u = (float(x) + 0.5f) / fw;
215 const float v = 1.f - (float(y) + 0.5f) / fh;
216 if (v < -0.02f || v > height + 0.04f) continue;
217 const float t = clampf(v / height, 0.f, 1.f);
218 const float cx = baseU + lean * t * t;
219 const float half = baseHalf * (1.f - t * 0.82f) + tipHalf * t;
220 const float dist = std::abs(u - cx) * fw;
221 float cov = coverEdge(dist, half * fw);
222 cov *= 1.f - clampf((v - height) / 0.03f, 0.f, 1.f);
223 cov *= clampf((v + 0.02f) / 0.05f, 0.f, 1.f);
224 if (cov > 0.5f) cov = 1.f;
225 else if (cov < 0.25f) cov = 0.f;
226 if (cov <= 0.f) continue;
227 const float side = (u - cx) * fw; // +right
228 float luma = luma0 + (luma1 - luma0) * t;
229 if (side > 0.15f) luma *= 0.78f; // 1px-ish self-shadow on the right
230 stampOver(rgba, atlasW, atlasH, ox + x, y, luma, cov);
231 }
232 }

245void stampTuft(std::vector<uint8_t> &rgba, int atlasW, int atlasH, int ox, int frameW, int frameH,
246 float wind) {
247 // Tiny root pad only — a solid mound turns overlapping cards into a flat lime slab.
248 const float fw = float(frameW);
249 const float fh = float(frameH);
250 for (int y = 0; y < frameH; ++y) {
251 for (int x = 0; x < frameW; ++x) {
252 const float u = (float(x) + 0.5f) / fw;
253 const float v = 1.f - (float(y) + 0.5f) / fh;
254 if (v > 0.16f) continue;
255 const float cx = 0.50f + wind * 0.02f;
256 const float hw = 0.16f * (1.f - v / 0.16f);
257 float cov = coverEdge(std::abs(u - cx) * fw, hw * fw);
258 if (cov > 0.5f) cov = 1.f;
259 else if (cov < 0.25f) cov = 0.f;
260 if (cov > 0.f) stampOver(rgba, atlasW, atlasH, ox + x, y, 0.72f, cov);
261 }
262 }
263
264 // 7 separated blades with gaps so later cards show through. Outline then fill.
265 const BladeDesc blades[] = {
266 {0.50f, 0.98f, 0.02f, 0.070f, 0.018f, 0.86f, 1.00f}, {0.38f, 0.90f, -0.14f, 0.062f, 0.016f, 0.82f, 0.97f},
267 {0.62f, 0.92f, 0.16f, 0.062f, 0.016f, 0.82f, 0.97f}, {0.28f, 0.74f, -0.26f, 0.056f, 0.016f, 0.78f, 0.93f},
268 {0.72f, 0.76f, 0.28f, 0.056f, 0.016f, 0.78f, 0.93f}, {0.44f, 0.84f, -0.06f, 0.050f, 0.014f, 0.84f, 0.98f},
269 {0.56f, 0.86f, 0.08f, 0.050f, 0.014f, 0.84f, 0.98f},
270 };
271 const float outline = 1.15f / fw;
272 for (const BladeDesc &b : blades) {
273 const float lean = b.lean + wind * 0.32f;
274 const float u = b.u + wind * 0.03f;
275 stampBlade(rgba, atlasW, atlasH, ox, frameW, frameH, u, b.height, lean, b.baseHalf + outline,
276 b.tipHalf + outline * 0.5f, 0.42f, 0.55f);
277 stampBlade(rgba, atlasW, atlasH, ox, frameW, frameH, u, b.height, lean, b.baseHalf, b.tipHalf,
278 b.luma0, b.luma1);
279 }

365 if (frameCount < 1) frameCount = 1;

459void packSwayAtlasRGBA(const std::vector<std::string> &grassFiles,
460 const std::vector<std::string> &leafFiles, std::vector<uint8_t> &rgbaOut,
461 PackedAtlasInfo &info) {
462 if (grassFiles.empty()) throw eve::Exception("grass::packSwayAtlasRGBA: no grass atlas files");
463
464 struct Loaded {
465 std::vector<uint8_t> rgba;
466 int w = 0;
467 int h = 0;
468 };
469 auto loadOne = [](const std::string &path) {
470 Loaded img;
471 if (!loadSwayMaskPng(path, img.rgba, img.w, img.h))
472 throw eve::Exception("grass::packSwayAtlasRGBA: failed to load '%s'", path.c_str());
473 return img;
474 };
475
476 std::vector<Loaded> grass;
477 grass.reserve(grassFiles.size());
478 for (const auto &p : grassFiles) grass.push_back(loadOne(p));
479 std::vector<Loaded> leaf;
480 leaf.reserve(leafFiles.size());
481 for (const auto &p : leafFiles) leaf.push_back(loadOne(p));
482
483 const int tw = grass.front().w;
484 const int th = grass.front().h;

488 kind, img.w, img.h, tw, th);

491 for (const auto &img : leaf) checkSize(img, "leaf");

497 info.frames = 4;

526std::vector<Point> sampleHalton(const float *posXYZ, const float *nrmXYZ, int vertexCount,
527 const uint32_t *indices, int indexCount, int count, uint32_t seed,
528 float minSlopeDot) {
529 std::vector<Point> out;
530 if (count <= 0) return out;
531 std::vector<Triangle> tris;
532 std::vector<float> cdf;
533 if (!buildTriangles(posXYZ, nrmXYZ, vertexCount, indices, indexCount, minSlopeDot, tris, cdf))
534 return out;
535
536 out.reserve(size_t(count));
537 for (int i = 0; i < count; ++i) {
538 const uint32_t n = mixSeed(seed, uint32_t(i + 1));
539 const float uTri = wrap01(radicalInverse(n, 2));
540 const float u = wrap01(radicalInverse(n, 3));
541 const float v = wrap01(radicalInverse(n, 5));
542 const Triangle &tri = tris[size_t(pickTriangle(cdf, uTri))];
543 Point p;
544 sampleOnTriangle(posXYZ, tri, u, v, p.position);
545 p.normal = tri.n;
546 p.id = uint32_t(i);
547 p.scale = 0.85f + 0.3f * hash01(p.id + seed);
548 out.push_back(p);
549 }
550 return out;
551}

553std::vector<Point> samplePoisson(const float *posXYZ, const float *nrmXYZ, int vertexCount,
554 const uint32_t *indices, int indexCount,
555 const SampleParams &params) {
556 std::vector<Point> accepted;
557 if (params.maxPoints <= 0 || params.radius <= 1e-6f) return accepted;
558
559 std::vector<Triangle> tris;
560 std::vector<float> cdf;
561 if (!buildTriangles(posXYZ, nrmXYZ, vertexCount, indices, indexCount, params.minSlopeDot, tris,
562 cdf))
563 return accepted;
564
565 const float cell = params.radius;
566 std::unordered_map<GridKey, std::vector<int>, GridKeyHash> grid;
567 const int attempts = std::max(params.maxPoints * 24, params.maxPoints + 16);
568
569 for (int i = 0; i < attempts && int(accepted.size()) < params.maxPoints; ++i) {
570 const uint32_t n = mixSeed(params.seed, uint32_t(i + 1));
571 const float uTri = wrap01(radicalInverse(n, 2));
572 const float u = wrap01(radicalInverse(n, 3));
573 const float v = wrap01(radicalInverse(n, 5));
574 const Triangle &tri = tris[size_t(pickTriangle(cdf, uTri))];
575 glm::vec3 pos;
576 sampleOnTriangle(posXYZ, tri, u, v, pos);
577 if (tooClose(pos, params.radius, accepted, grid, cell)) continue;
578
579 Point p;
580 p.position = pos;
581 p.normal = tri.n;
582 p.id = uint32_t(accepted.size());
583 p.scale = 0.85f + 0.3f * hash01(p.id + params.seed);
584 const int idx = int(accepted.size());
585 accepted.push_back(p);
586 grid[toKey(pos, cell)].push_back(idx);
587 }
588 return accepted;
589}

591BillboardMesh buildBillboards(const std::vector<Point> &points, float width, float height,
592 bool alwaysDark) {
593 (void)width;
594 (void)height;
595 BillboardMesh mesh;
596 const size_t n = points.size();
597 mesh.posXYZ.reserve(n * 4 * 3);
598 mesh.nrmXYZ.reserve(n * 4 * 3);
599 mesh.uvST.reserve(n * 4 * 2);
600 mesh.indices.reserve(n * 6);
601
602 // Local corners: (0,0) bottom-left, (1,0) bottom-right, (1,1) top-right, (0,1) top-left.
603 // Root is the bottom-center of the rectangle, i.e. UV (0.5, 0).
604 const float cu[4] = {0.f, 1.f, 1.f, 0.f};
605 const float cv[4] = {0.f, 0.f, 1.f, 1.f};
606 const uint32_t corners[6] = {0, 1, 2, 0, 2, 3};
607
608 for (size_t i = 0; i < n; ++i) {
609 const Point &p = points[i];
610 const uint32_t base = uint32_t(i * 4);
611 for (int c = 0; c < 4; ++c) {
612 mesh.posXYZ.push_back(p.position.x);
613 mesh.posXYZ.push_back(p.position.y);
614 mesh.posXYZ.push_back(p.position.z);
615 mesh.nrmXYZ.push_back(float(p.id));
616 mesh.nrmXYZ.push_back(p.scale > 1e-3f ? p.scale : 1.f);
617 mesh.nrmXYZ.push_back(alwaysDark ? 1.f : 0.f);
618 mesh.uvST.push_back(cu[c]);
619 mesh.uvST.push_back(cv[c]);
620 }
621 for (uint32_t k : corners) mesh.indices.push_back(base + k);
622 }
623 return mesh;
624}

626void makePlane(float sizeX, float sizeZ, int segX, int segZ, std::vector<float> &posXYZ,
627 std::vector<float> &nrmXYZ, std::vector<uint32_t> &indices) {
628 segX = std::max(segX, 1);
629 segZ = std::max(segZ, 1);
630 sizeX = std::max(sizeX, 1e-3f);
631 sizeZ = std::max(sizeZ, 1e-3f);
632 const int nx = segX + 1;
633 const int nz = segZ + 1;
634 posXYZ.clear();
635 nrmXYZ.clear();
636 indices.clear();
637 posXYZ.reserve(size_t(nx * nz * 3));
638 nrmXYZ.reserve(size_t(nx * nz * 3));
639 indices.reserve(size_t(segX * segZ * 6));
640
641 for (int z = 0; z < nz; ++z) {
642 for (int x = 0; x < nx; ++x) {
643 const float px = (float(x) / float(segX) - 0.5f) * sizeX;
644 const float pz = (float(z) / float(segZ) - 0.5f) * sizeZ;
645 posXYZ.push_back(px);
646 posXYZ.push_back(0.f);
647 posXYZ.push_back(pz);
648 nrmXYZ.push_back(0.f);
649 nrmXYZ.push_back(1.f);
650 nrmXYZ.push_back(0.f);
651 }
652 }
653 for (int z = 0; z < segZ; ++z) {
654 for (int x = 0; x < segX; ++x) {
655 const uint32_t i0 = uint32_t(z * nx + x);
656 const uint32_t i1 = i0 + 1;
657 const uint32_t i2 = i0 + uint32_t(nx);
658 const uint32_t i3 = i2 + 1;
659 indices.push_back(i0);
660 indices.push_back(i2);
661 indices.push_back(i1);
662 indices.push_back(i1);
663 indices.push_back(i2);
664 indices.push_back(i3);
665 }
666 }
667}

685void GrassField::bake(const float *posXYZ, const float *nrmXYZ, int vertexCount,
686 const uint32_t *indices, int indexCount, const BakeParams &params) {
687 if (!gfx_) throw eve::Exception("GrassField::bake: null graphics");
688
689 grass::SampleParams denseP;
690 denseP.radius = params.denseRadius;
691 denseP.maxPoints = params.maxDense;
692 denseP.seed = params.seed;
693 denseP.minSlopeDot = params.minSlopeDot;
694
695 grass::SampleParams sparseP = denseP;
696 sparseP.radius = params.sparseRadius;
697 sparseP.maxPoints = params.maxSparse;
698 sparseP.seed = params.seed * 7477u + 13u;
699
700 const auto densePts =
701 grass::samplePoisson(posXYZ, nrmXYZ, vertexCount, indices, indexCount, denseP);
702 const auto sparsePts =
703 grass::samplePoisson(posXYZ, nrmXYZ, vertexCount, indices, indexCount, sparseP);
704 denseCount_ = int(densePts.size());
705 sparseCount_ = int(sparsePts.size());
706
707 const auto denseMesh = grass::buildBillboards(densePts, params.width, params.height, false);
708 const auto sparseMesh = grass::buildBillboards(sparsePts, params.width, params.height, true);
709
710 if (!shader_) shader_ = grass::createShader(gfx_);
711 grass::PackedAtlasInfo layout;
712 if (!params.grassAtlasFiles.empty()) {
713 atlas_ = grass::createSwayAtlasFromFiles(gfx_, params.grassAtlasFiles, params.leafAtlasFiles,
714 &layout);
715 } else {
716 if (!atlas_)
717 atlas_ = grass::createSwayAtlas(gfx_, params.atlasFrameW, params.atlasFrameH,
718 params.atlasFrames);
719 layout.frames = std::max(params.atlasFrames, 1);
720 layout.atlasCols = layout.frames;
721 layout.atlasRows = 1;
722 layout.grassVariants = 1;
723 layout.leafVariants = 1;
724 layout.leafRowOffset = 0;
725 }
726
727 grass::bindDefaults(shader_);
728 grass::bindAtlasLayout(shader_, layout);
729 shader_->sendFloat("grassWidth", params.width);
730 shader_->sendFloat("grassHeight", params.height);
731 shader_->sendFloat("frameDuration", frameDuration_);
732 shader_->sendFloat("time", time_);
733
734 denseMesh_ = nullptr;
735 sparseMesh_ = nullptr;
736 if (!denseMesh.indices.empty())
737 denseMesh_ = gfx_->newMeshFromArrays(denseMesh.posXYZ.data(), denseMesh.nrmXYZ.data(),
738 denseMesh.uvST.data(), int(denseMesh.posXYZ.size() / 3),
739 denseMesh.indices.data(), int(denseMesh.indices.size()));
740 if (!sparseMesh.indices.empty())
741 sparseMesh_ =
742 gfx_->newMeshFromArrays(sparseMesh.posXYZ.data(), sparseMesh.nrmXYZ.data(),
743 sparseMesh.uvST.data(), int(sparseMesh.posXYZ.size() / 3),
744 sparseMesh.indices.data(), int(sparseMesh.indices.size()));
745}
