26[[nodiscard]] bool hasDuplicateMemberships(const ContainerSnapshot& snapshot) {
27 std::unordered_set<MembershipId> ids;
28 std::unordered_set<std::int32_t> slots;
29 for (const auto& entry : snapshot.entries) {
30 if (!entry.membership.object.isValid() || !ids.insert(entry.membership.object).second) return true;
31 if (!entry.membership.slot.isValid() || !slots.insert(entry.membership.slot.value()).second) return true;
32 if (entry.object.id != entry.membership.object || entry.object.quantity == 0 ||
33 entry.membership.generation.isZero())
34 return true;
35 }
36 return false;
37}

39[[nodiscard]] Result<void> validateSnapshot(const ContainerSnapshot& snapshot, const IContainer& container) {
40 if (!snapshot.id.isValid() || snapshot.id != container.descriptor().id)
41 return failure(DiagnosticCode::InvariantViolation, "container snapshot identity does not match adapter");
42 if (hasDuplicateMemberships(snapshot))
43 return failure(DiagnosticCode::Conflict, "container snapshot contains duplicate or invalid membership");
44 const auto& descriptor = container.descriptor();
45 if (!descriptor.capacity.isUnlimited() && snapshot.entries.size() > descriptor.capacity.value())
46 return failure(DiagnosticCode::InvariantViolation, "container snapshot exceeds capacity");
47 if (!descriptor.capacity.isUnlimited()) {
48 for (const auto& entry : snapshot.entries) {
49 if (static_cast<std::size_t>(entry.membership.slot.value()) >= descriptor.capacity.value())
50 return failure(DiagnosticCode::InvariantViolation,
51 "container snapshot contains a slot outside capacity");
52 }
53 }
54 return Result<void>::success();
55}

57[[nodiscard]] Result<ContainerSnapshot> removeEntry(const ContainerSnapshot& original,
58 const ContainerDescriptor& descriptor, const MembershipId& object) {
59 ContainerSnapshot candidate = original;
60 const auto it = std::find_if(candidate.entries.begin(), candidate.entries.end(),
61 [&](const MembershipEntry& entry) { return entry.membership.object == object; });
62 if (it == candidate.entries.end())
63 return Result<ContainerSnapshot>::failure(
64 Diagnostic::error(DiagnosticCode::NotFound, "snapshot membership was not found"));
65 candidate.entries.erase(it);
66 if (descriptor.ordering != Ordering::ExplicitSlots) {
67 if (candidate.entries.size() > static_cast<std::size_t>(std::numeric_limits<std::int32_t>::max()))
68 return Result<ContainerSnapshot>::failure(
69 Diagnostic::error(DiagnosticCode::InvariantViolation, "container has too many indexed memberships"));
70 for (std::size_t index = 0; index < candidate.entries.size(); ++index)
71 candidate.entries[index].membership.slot = SlotIndex(static_cast<std::int32_t>(index));
72 }
73 return Result<ContainerSnapshot>::success(std::move(candidate));
74}

76[[nodiscard]] Result<SlotIndex> chooseDestinationSlot(const ContainerSnapshot& snapshot,
77 const ContainerDescriptor& descriptor,
78 std::optional<SlotIndex> requested) {
79 if (requested) {
80 if (!requested->isValid())
81 return Result<SlotIndex>::failure(
82 Diagnostic::error(DiagnosticCode::InvalidArgument, "destination slot is invalid"));
83 const auto value = static_cast<std::size_t>(requested->value());
84 if (descriptor.ordering == Ordering::ExplicitSlots) {
85 if (!descriptor.capacity.isUnlimited() && value >= descriptor.capacity.value())
86 return Result<SlotIndex>::failure(
87 Diagnostic::error(DiagnosticCode::Conflict, "destination slot exceeds container capacity"));
88 for (const auto& entry : snapshot.entries)
89 if (entry.membership.slot == *requested)
90 return Result<SlotIndex>::failure(
91 Diagnostic::error(DiagnosticCode::Conflict, "destination slot is occupied"));
92 return Result<SlotIndex>::success(*requested);
93 }
94 if (value > snapshot.entries.size())
95 return Result<SlotIndex>::failure(
96 Diagnostic::error(DiagnosticCode::InvalidArgument, "destination slot is out of range"));
97 return Result<SlotIndex>::success(*requested);
98 }
99
100 if (descriptor.ordering != Ordering::ExplicitSlots) {
101 if (snapshot.entries.size() > static_cast<std::size_t>(std::numeric_limits<std::int32_t>::max()))
102 return Result<SlotIndex>::failure(
103 Diagnostic::error(DiagnosticCode::InvariantViolation, "container has too many indexed memberships"));
104 return Result<SlotIndex>::success(SlotIndex(static_cast<std::int32_t>(snapshot.entries.size())));
105 }
106
107 const std::size_t limit =
108 descriptor.capacity.isUnlimited() ? snapshot.entries.size() + 1u : descriptor.capacity.value();
109 std::unordered_set<std::int32_t> occupied;
110 for (const auto& entry : snapshot.entries) occupied.insert(entry.membership.slot.value());
111 for (std::size_t index = 0; index < limit; ++index) {
112 if (index > static_cast<std::size_t>(std::numeric_limits<std::int32_t>::max())) break;
113 if (!occupied.contains(static_cast<std::int32_t>(index)))
114 return Result<SlotIndex>::success(SlotIndex(static_cast<std::int32_t>(index)));
115 }
116 return Result<SlotIndex>::failure(Diagnostic::error(DiagnosticCode::Conflict, "container has no available slot"));
117}

119[[nodiscard]] Result<ContainerSnapshot> insertEntry(ContainerSnapshot base, const ContainerDescriptor& descriptor,
120 const MembershipEntry& sourceEntry,
121 std::optional<SlotIndex> requested) {
122 auto destinationResult = chooseDestinationSlot(base, descriptor, requested);
123 if (!destinationResult) return Result<ContainerSnapshot>::failure(destinationResult.status());
124 const SlotIndex destination = destinationResult.value();
125 MembershipEntry moved = sourceEntry;
126 moved.membership.slot = destination;
127 if (descriptor.ordering == Ordering::ExplicitSlots) {
128 base.entries.push_back(std::move(moved));
129 } else {
130 base.entries.insert(base.entries.begin() + destination.value(), std::move(moved));
131 for (std::size_t index = 0; index < base.entries.size(); ++index)
132 base.entries[index].membership.slot = SlotIndex(static_cast<std::int32_t>(index));
133 }
134 return Result<ContainerSnapshot>::success(std::move(base));
135}

139Result<TransferReceipt> TransferService::transfer(const TransferRequest& request, TransferEventSink eventSink) {
140 if (request.source == nullptr || request.destination == nullptr || !request.object.isValid()) {
141 return Result<TransferReceipt>::failure(Diagnostic::error(
142 DiagnosticCode::InvalidArgument, "transfer requires two containers and a membership object"));
143 }
144
145 const bool sameContainer = request.source == request.destination;
146 auto sourceSnapshotResult = request.source->snapshot();
147 if (!sourceSnapshotResult) {
148 return Result<TransferReceipt>::failure(sourceSnapshotResult.status());
149 }
150 ContainerSnapshot sourceSnapshot = std::move(sourceSnapshotResult).takeValue();
151 ContainerSnapshot destinationSnapshot;
152 if (sameContainer) {
153 destinationSnapshot = sourceSnapshot;
154 } else {
155 auto destinationSnapshotResult = request.destination->snapshot();
156 if (!destinationSnapshotResult) {
157 return Result<TransferReceipt>::failure(destinationSnapshotResult.status());
158 }
159 destinationSnapshot = std::move(destinationSnapshotResult).takeValue();
160 }
161 auto sourceSnapshotValid = validateSnapshot(sourceSnapshot, *request.source);
162 if (!sourceSnapshotValid) {
163 return Result<TransferReceipt>::failure(sourceSnapshotValid.status());
164 }
165 auto destinationSnapshotValid = validateSnapshot(destinationSnapshot, *request.destination);
166 if (!destinationSnapshotValid) {
167 return Result<TransferReceipt>::failure(destinationSnapshotValid.status());
168 }
169
170 if (request.expectedSourceRevision && *request.expectedSourceRevision != sourceSnapshot.revision) {
171 return Result<TransferReceipt>::failure(
172 Diagnostic::error(DiagnosticCode::StaleHandle, "source container revision is stale"));
173 }
174 if (request.expectedDestinationRevision && *request.expectedDestinationRevision != destinationSnapshot.revision) {
175 return Result<TransferReceipt>::failure(
176 Diagnostic::error(DiagnosticCode::StaleHandle, "destination container revision is stale"));
177 }
178
179 const MembershipEntry* sourceEntry = findEntry(sourceSnapshot, request.object);
180 if (sourceEntry == nullptr) {
181 bool duplicate = false;
182 for (const auto& entry : sourceSnapshot.entries)
183 if (entry.membership.object == request.object) duplicate = true;
184 return Result<TransferReceipt>::failure(
185 Diagnostic::error(duplicate ? DiagnosticCode::Conflict : DiagnosticCode::NotFound,
186 duplicate ? "source contains duplicate membership" : "source membership was not found"));
187 }
188 if (request.sourceSlot && *request.sourceSlot != sourceEntry->membership.slot) {
189 return Result<TransferReceipt>::failure(
190 Diagnostic::error(DiagnosticCode::StaleHandle, "source slot no longer matches object"));
191 }
192
193 if (!sameContainer && request.source->descriptor().id == request.destination->descriptor().id) {
194 return Result<TransferReceipt>::failure(Diagnostic::error(
195 DiagnosticCode::Conflict, "different container adapters must not expose the same container identity"));
196 }
197 for (const auto& entry : destinationSnapshot.entries) {
198 if (entry.membership.object == request.object && !sameContainer)
199 return Result<TransferReceipt>::failure(
200 Diagnostic::error(DiagnosticCode::Conflict, "destination already contains this membership object"));
201 if (request.destinationSlot && entry.membership.slot == *request.destinationSlot &&
202 !(sameContainer && entry.membership.object == request.object)) {
203 return Result<TransferReceipt>::failure(
204 Diagnostic::error(DiagnosticCode::Conflict, "destination slot is occupied"));
205 }
206 }
207
208 auto accepted =
209 request.destination->validateInsert(sourceEntry->object, request.destinationSlot,
210 sameContainer ? std::optional<MembershipId>(request.object) : std::nullopt);
211 if (!accepted) {
212 return Result<TransferReceipt>::failure(accepted.status());
213 }
214
215 if (sameContainer && request.destinationSlot && sourceEntry->membership.slot == *request.destinationSlot) {
216 TransferReceipt receipt{sourceSnapshot.id,
217 destinationSnapshot.id,
218 request.object,
219 sourceEntry->membership.slot,
220 sourceEntry->membership.slot,
221 sourceSnapshot.revision,
222 destinationSnapshot.revision};
223 return Result<TransferReceipt>::success(std::move(receipt), Status::success(StatusCode::NoOp));
224 }
225
226 auto sourceRevision = sourceSnapshot.revision.incremented();
227 if (!sourceRevision)
228 return Result<TransferReceipt>::failure(
229 Diagnostic::error(DiagnosticCode::InvariantViolation, "source revision exhausted"));
230 auto sourceCandidateResult = removeEntry(sourceSnapshot, request.source->descriptor(), request.object);
231 if (!sourceCandidateResult) return Result<TransferReceipt>::failure(sourceCandidateResult.status());
232 ContainerSnapshot sourceCandidate = std::move(sourceCandidateResult).takeValue();
233 sourceCandidate.revision = *sourceRevision;
234
235 ContainerSnapshot destinationCandidate;
236 SlotIndex destinationSlot = SlotIndex::invalid();
237 if (sameContainer) {
238 auto candidateResult = insertEntry(std::move(sourceCandidate), request.source->descriptor(), *sourceEntry,
239 request.destinationSlot);
240 if (!candidateResult) {
241 return Result<TransferReceipt>::failure(candidateResult.status());
242 }
243 destinationCandidate = std::move(candidateResult).takeValue();
244 destinationCandidate.revision = *sourceRevision;
245 const auto* moved = findEntry(destinationCandidate, request.object);
246 if (moved == nullptr)
247 return Result<TransferReceipt>::failure(
248 Diagnostic::error(DiagnosticCode::InvariantViolation, "same-container candidate lost membership"));
249 destinationSlot = moved->membership.slot;
250 } else {
251 auto destinationRevision = destinationSnapshot.revision.incremented();
252 if (!destinationRevision)
253 return Result<TransferReceipt>::failure(
254 Diagnostic::error(DiagnosticCode::InvariantViolation, "destination revision exhausted"));
255 auto candidateResult =
256 insertEntry(destinationSnapshot, request.destination->descriptor(), *sourceEntry, request.destinationSlot);
257 if (!candidateResult) return Result<TransferReceipt>::failure(candidateResult.status());
258 destinationCandidate = std::move(candidateResult).takeValue();
259 destinationCandidate.revision = *destinationRevision;
260 const auto* moved = findEntry(destinationCandidate, request.object);
261 if (moved == nullptr)
262 return Result<TransferReceipt>::failure(
263 Diagnostic::error(DiagnosticCode::InvariantViolation, "destination candidate lost membership"));
264 destinationSlot = moved->membership.slot;
265 }
266
267 // For a same-container move, `sourceCandidate` was consumed while building
268 // the final candidate. The adapter must validate and commit that final
269 // layout, not the moved-from removal-only intermediate.
270 const auto& sourceCommitCandidate = sameContainer ? destinationCandidate : sourceCandidate;
271 auto sourcePreparedResult = request.source->prepare(sourceSnapshot, sourceCommitCandidate);
272 if (!sourcePreparedResult) return Result<TransferReceipt>::failure(sourcePreparedResult.status());
273 auto sourcePrepared = std::move(sourcePreparedResult).takeValue();
274
275 if (sameContainer) {
276 sourcePrepared->commit();
277 } else {
278 auto destinationPreparedResult = request.destination->prepare(destinationSnapshot, destinationCandidate);
279 if (!destinationPreparedResult) {
280 sourcePrepared->rollback();
281 return Result<TransferReceipt>::failure(destinationPreparedResult.status());
282 }
283 auto destinationPrepared = std::move(destinationPreparedResult).takeValue();
284 sourcePrepared->commit();
285 destinationPrepared->commit();
286 }
287
288 TransferReceipt receipt{sourceSnapshot.id,
289 destinationSnapshot.id,
290 request.object,
291 sourceEntry->membership.slot,
292 destinationSlot,
293 *sourceRevision,
294 sameContainer ? *sourceRevision : destinationCandidate.revision};
295 Status resultStatus = Status::success(StatusCode::Applied);
296 if (eventSink) {
297 try {
298 eventSink(TransferEvent{receipt.source, receipt.destination, receipt.object, receipt.sourceSlot,
299 receipt.destinationSlot, receipt.sourceRevision, receipt.destinationRevision});
300 } catch (const std::exception&) {
301 try {
302 std::vector<Diagnostic> diagnostics;
303 diagnostics.emplace_back(Diagnostic::warning(DiagnosticCode::CallbackFailure,
304 "transfer committed but its event callback threw"));
305 resultStatus = Status(StatusCode::Applied, std::move(diagnostics));
306 } catch (...) {
307 resultStatus = Status::success(StatusCode::Applied);
308 }
309 } catch (...) {
310 try {
311 std::vector<Diagnostic> diagnostics;
312 diagnostics.emplace_back(Diagnostic::warning(DiagnosticCode::CallbackFailure,
313 "transfer committed but its event callback threw"));
314 resultStatus = Status(StatusCode::Applied, std::move(diagnostics));
315 } catch (...) {
316 resultStatus = Status::success(StatusCode::Applied);
317 }
318 }
319 }
320 return Result<TransferReceipt>::success(std::move(receipt), std::move(resultStatus));
321}
