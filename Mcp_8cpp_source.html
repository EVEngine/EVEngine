48bool isEditorHostResource(const std::string& path) {
49 if (path == "mcp.nut") return true;
50 if (path.rfind("mcp/", 0) == 0 && endsWith(path, ".nut")) return true;
51 if (path.rfind("editors/", 0) != 0) return false;
52 return endsWith(path, ".vm.nut") || endsWith(path, ".editor.json");

83int Cmdline::McpHost(std::string path, int port) {
84#if defined(EVENGINE_ANDROID) || defined(EVENGINE_IOS) || defined(EVENGINE_WEBGPU)
85 (void)path;
86 (void)port;
87 std::cerr << "eve mcp: headless MCP host is desktop-only" << std::endl;
88 return 4;
89#else
90 try {
91 std::string gameDir = path;
92 {
93 std::error_code ec;
94 if (path.empty() || path == ".")
95 gameDir = std::filesystem::current_path(ec).string();
96 else
97 gameDir = std::filesystem::absolute(path, ec).string();
98 if (ec) gameDir = path;
99 }
100
101 if (!gameDir.empty() && gameDir != ".") {
102 std::error_code ec;
103 std::filesystem::current_path(gameDir, ec);
104 if (ec) {
105 std::cerr << "Cannot chdir to project path '" << gameDir << "': " << ec.message()
106 << std::endl;
107 return 2;
108 }
109 }
110
111 Runtime runtime(2048, ssq::Libs::ALL);
112 runtime.initialize();
113 runtime.vm().setPrintFunc(&hostPrintFunc, &hostPrintFunc);
114
115 // Mount the project for PhysFS so script file APIs / editors/ resolve.
116 // Optional for the headless host: keep running when the environment
117 // cannot initialize PhysFS (sandboxed/CI shells), scripts just lose
118 // the PhysFS-backed file API.
119 eve::filesystem::Filesystem* projectFs = nullptr;
120 try {
121 projectFs = eve::filesystem::Filesystem::create();
122 if (projectFs) {
123 std::error_code ec;
124 auto cwd = std::filesystem::current_path(ec);
125 if (!ec) {
126 try {
127 projectFs->setSource(cwd.string());
128 } catch (...) {
129 }
130 }
131 }
132 eve::filesystem::physfs::installScriptFileApi(runtime.vm());
133 } catch (const std::exception& e) {
134 std::cerr << "eve mcp: filesystem mount skipped (" << e.what() << ")" << std::endl;
135 }
136
137 // DevTools: attach the VM so eve_eval / eve_run_script / snapshot work;
138 // expose eve.dev. No statement-level local sampling (host is a service).
139 auto& dt = eve::dev::DevTool::instance();
140 dt.attach(runtime, /*sampleLocals=*/false);
141 dt.exposeScriptApi(runtime.vm());
142
143 // The headless host never runs load.nut, so registered modules are not
144 // instantiated automatically and their capability providers
145 // (IEditorHost, IRenderCapture, ISceneQuery, ...) are absent. Mirror
146 // load.nut's binding loop so the host exposes the full tool surface.
147 eve::ModuleManager::requireAll();
148
149 // Editor host: JSON Views <-> Squirrel ViewModel binding + window/render.
150 auto* host = eve::cap::query<eve::IEditorHost>();
151 if (host) {
152 host->start(runtime.vm(), gameDir, /*allowWindow=*/true);
153 host->exposeScriptApi(runtime.vm());
154 } else {
155 std::cerr << "eve mcp: editor host unavailable (ui module missing)"
156 << std::endl;
157 return 3;
158 }
159
160 // Unlike `eve run`, the MCP host does not execute load.nut. Start the
161 // same project watcher here and route editor/host scripts through the
162 // EditorHost while other assets keep using the shared asset reloaders.
163 auto* hot = eve::ModuleManager::getInstance<eve::filesystem::HotReload>("HotReload");
164 const int watchCount = projectFs && hot ? hot->watchTree(".") : 0;
165 host->setHotReloadWatchCount(watchCount);
166 if (std::filesystem::is_regular_file(std::filesystem::path(gameDir) / "mcp.nut")) {
167 const std::string initial = host->reloadResource("mcp.nut");
168 if (initial.rfind("error:", 0) == 0)
169 std::cerr << "MCP host script load failed: " << initial << std::endl;
170 }
171 std::cerr << "MCP hot-reload: watching " << watchCount << " path(s)" << std::endl;
172
173 auto& mcp = dt.mcp();
174 mcp.setGameRoot(gameDir);
175
176 if (port > 0) {
177 const int bound = dt.startMcp(static_cast<uint16_t>(port));
178 if (bound <= 0) {
179 std::cerr << "Failed to start MCP server on port " << port << std::endl;
180 return 3;
181 }
182 std::cerr << "MCP listening on 127.0.0.1:" << bound
183 << " (newline JSON-RPC; use tools/eve-mcp for Cursor stdio)" << std::endl;
184 } else {
185 if (!mcp.listenStdio()) {
186 std::cerr << "Failed to start stdio MCP transport" << std::endl;
187 return 3;
188 }
189 std::cerr << "MCP stdio transport ready (headless editor host; project=" << gameDir
190 << ")" << std::endl;
191 }
192
193 std::map<std::string, std::chrono::steady_clock::time_point> pendingReloads;
194 constexpr auto reloadDebounce = std::chrono::milliseconds(100);
195 while (!host->exitRequested()) {
196 dt.poll(); // drains MCP requests on the main thread
197 if (projectFs) {
198 while (true) {
199 const std::string kind = projectFs->pollWatch();
200 if (kind.empty()) break;
201 if (kind != "modified" && kind != "added" && kind != "movedTo") continue;
202 std::string changed = projectFs->getLastWatchPath();
203 changed = eve::filesystem::HotReload::normalizePath(std::move(changed));
204 if (!changed.empty()) pendingReloads[changed] = std::chrono::steady_clock::now();
205 }
206 }
207 const auto now = std::chrono::steady_clock::now();
208 for (auto it = pendingReloads.begin(); it != pendingReloads.end();) {
209 if (now - it->second < reloadDebounce) {
210 ++it;
211 continue;
212 }
213 const std::string changed = it->first;
214 if (isEditorHostResource(changed)) {
215 const std::string result = host->reloadResource(changed);
216 std::cerr << "MCP hot-reload: " << changed << ": " << result << std::endl;
217 } else if (hot && hot->tryReload(changed)) {
218 std::cerr << "MCP hot-reload asset: " << changed << std::endl;
219 }
220 it = pendingReloads.erase(it);
221 }
222 host->frame();
223 if (mcp.transport() == eve::dev::McpServer::Transport::Stdio &&
224 mcp.stdinClosed()) {
225 std::cerr << "MCP stdio closed; host exiting" << std::endl;
226 break;
227 }
228 std::this_thread::sleep_for(std::chrono::milliseconds(2));
229 }
230
231 host->stop();
232 dt.detach();
233 return 0;
234 } catch (const std::exception& e) {
235 std::cerr << "eve mcp failed: " << e.what() << std::endl;
236 return 3;
237 }
238#endif
239}
