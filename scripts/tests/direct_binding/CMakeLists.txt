cmake_minimum_required(VERSION 3.21)
project(direct_binding_test LANGUAGES C CXX)
get_filename_component(ENGINE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(DEPENDENCY_CHECKOUT "${ENGINE_ROOT}" CACHE PATH "Checkout with initialized third-party sources")
set(DEPS "${DEPENDENCY_CHECKOUT}/third-party")
set(DISABLE_DYNAMIC ON CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/squirrel" squirrel EXCLUDE_FROM_ALL)
set(SQUIRREL_INCLUDE_DIR "${DEPS}/squirrel/include")
set(SQUIRREL_LIBRARIES squirrel_static)
set(SQSTDLIB_LIBRARIESRARIES sqstdlib_static)
add_subdirectory("${DEPS}/simplesquirrel" simplesquirrel EXCLUDE_FROM_ALL)
target_link_libraries(simplesquirrel_static PUBLIC squirrel_static sqstdlib_static)
add_executable(direct_binding_test test.cpp "${ENGINE_ROOT}/src/engine/common/ScriptBinding.cpp")
target_compile_features(direct_binding_test PRIVATE cxx_std_20)
target_include_directories(direct_binding_test PRIVATE "${ENGINE_ROOT}/src/engine" "${DEPS}/simplesquirrel/include" "${DEPS}/squirrel/include")
target_link_libraries(direct_binding_test PRIVATE simplesquirrel_static)
enable_testing()
add_test(NAME direct_binding COMMAND direct_binding_test)
# NetStream now lives in the amalgamated NetHost.cpp. Compile its actual marked
# section without linking sockets, workers and the rest of the engine.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ENGINE_ROOT}/src/modules/network/NetHost.cpp")
file(READ "${ENGINE_ROOT}/src/modules/network/NetHost.cpp" NETWORK_SOURCE)
string(FIND "${NETWORK_SOURCE}" "// NetStream.cpp" STREAM_START)
string(FIND "${NETWORK_SOURCE}" "// NetWorker.cpp" STREAM_END)
if(STREAM_START LESS 0 OR STREAM_END LESS STREAM_START)
    message(FATAL_ERROR "Cannot locate the NetStream section in NetHost.cpp")
endif()
math(EXPR STREAM_LENGTH "${STREAM_END} - ${STREAM_START}")
string(SUBSTRING "${NETWORK_SOURCE}" ${STREAM_START} ${STREAM_LENGTH} STREAM_SOURCE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/NetStream.cpp" "#include \"network/NetStream.h\"\n#include <cstring>\n${STREAM_SOURCE}")
add_executable(network_binding_test network.cpp "${ENGINE_ROOT}/src/engine/common/ScriptBinding.cpp" "${CMAKE_CURRENT_BINARY_DIR}/NetStream.cpp")
target_compile_features(network_binding_test PRIVATE cxx_std_20)
target_include_directories(network_binding_test PRIVATE "${ENGINE_ROOT}/src/engine" "${ENGINE_ROOT}/src/modules" "${DEPS}/simplesquirrel/include" "${DEPS}/squirrel/include")
target_link_libraries(network_binding_test PRIVATE simplesquirrel_static)
add_test(NAME network_binding COMMAND network_binding_test)
