77bool raster(const Value* value, Heightmap& out, std::size_t& budget) {
78 int width = 0, height = 0;
79 if (!value || !dimensions(*value, width, height, budget)) return false;
80 const auto* samples = array(value->find("samples"));
81 if (!samples || samples->size() != static_cast<std::size_t>(width) * height) return false;
82 out = Heightmap(width, height);
83 for (std::size_t i = 0; i < samples->size(); ++i)
84 if (!number(&(*samples)[i], out.data()[i])) return false;
85 return true;
86}
87bool decodeTile(const Value& value, RoadTerrainTile& tile, std::size_t& budget, bool requireOriginY) {
88 if ((requireOriginY || value.find("originY")) && !number(value.find("originY"), tile.originY)) return false;
89 if (!text(value.find("id"), tile.id) || !number(value.find("originX"), tile.originX) ||
90 !number(value.find("originZ"), tile.originZ) || !number(value.find("sizeX"), tile.sizeX) ||
91 !number(value.find("sizeZ"), tile.sizeZ) || !number(value.find("heightScale"), tile.heightScale) ||
92 !raster(value.find("heights"), tile.heights, budget) || !raster(value.find("holes"), tile.holes, budget))
93 return false;
94 const auto *splat = array(value.find("splatLayers")), *details = array(value.find("details")),
95 *trees = array(value.find("trees"));
96 if (!splat || splat->size() > 64u || !details || details->size() > 256u || !trees || trees->size() > 1000000u ||
97 trees->size() > budget / 8u)
98 return false;
99 budget -= trees->size() * 8u;
100 for (const auto& encoded : *splat) {
101 Heightmap map;
102 if (!raster(&encoded, map, budget)) return false;
103 tile.splatLayers.push_back(std::move(map));
104 }
105 for (const auto& encoded : *details) {
106 RoadTerrainDetail d;
107 if (!text(encoded.find("resourceId"), d.resourceId) || !dimensions(encoded, d.width, d.height, budget))
108 return false;
109 const auto* samples = array(encoded.find("samples"));
110 if (!samples || samples->size() != static_cast<std::size_t>(d.width) * d.height) return false;
111 for (const auto& sample : *samples) {
112 int density = 0;
113 if (!integer(&sample, density, std::numeric_limits<int>::max())) return false;
114 d.density.push_back(density);
115 }
116 tile.details.push_back(std::move(d));
117 }
118 for (const auto& encoded : *trees) {
119 RoadTerrainTree t;
120 if (!text(encoded.find("id"), t.id) || !text(encoded.find("resourceId"), t.resourceId) ||
121 !number(encoded.find("x"), t.x) || !number(encoded.find("y"), t.y) || !number(encoded.find("z"), t.z) ||
122 !number(encoded.find("yaw"), t.yaw) || !number(encoded.find("width"), t.width) ||
123 !number(encoded.find("height"), t.height))
124 return false;
125 tile.trees.push_back(std::move(t));
126 }
127 return true;
128}

160Result<int> RoadTerrainWorkspace::restore(const Value& snapshot) {
161 const auto *schema = snapshot.find("schema"), *version = snapshot.find("version");
162 const auto* history = array(snapshot.find("history"));
163 int cursor = 0;
164 if (!schema || !schema->isString() || schema->asString() != "procgen.road.terrain" || !version ||
165 !version->isInt64() || (version->asInt() != 1 && version->asInt() != 2) || !history || history->empty() ||
166 history->size() > 4096u || !integer(snapshot.find("cursor"), cursor) ||
167 static_cast<std::size_t>(cursor) >= history->size())
168 return invalidSnapshot<int>();
169 auto candidate = std::make_unique<Impl>();
170 std::size_t budget = 16u * 1024u * 1024u;
171 for (const auto& encodedFrame : *history) {
172 const auto* tiles = array(&encodedFrame);
173 if (!tiles || tiles->empty() || tiles->size() > 256u) return invalidSnapshot<int>();
174 Impl::Frame frame;
175 for (const auto& encoded : *tiles) {
176 RoadTerrainTile tile;
177 if (!decodeTile(encoded, tile, budget, version->asInt() == 2)) return invalidSnapshot<int>();
178 frame.push_back(std::move(tile));
179 }
180 auto valid = detail::validateTerrainTiles(frame);
181 if (!valid.ok()) return Result<int>::failure(valid.status());
182 if (!candidate->history.empty()) {
183 const auto& baseline = candidate->history.front();
184 if (frame.size() != baseline.size()) return invalidSnapshot<int>();
185 for (std::size_t i = 0; i < frame.size(); ++i)
186 if (!sameTopology(frame[i], baseline[i])) return invalidSnapshot<int>();
187 }
188 candidate->history.push_back(std::move(frame));
189 }
190 candidate->cursor = static_cast<std::size_t>(cursor);
191 impl_.swap(candidate);
192 return Result<int>::success(static_cast<int>(impl_->history[impl_->cursor].size()));
193}
