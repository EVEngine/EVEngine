25bool commandMatches(ClimbingCommandRequirement requirement, ClimbingCommand command) {

46eve::Result<void> validateFullCapsulePath(physics::World3D& world, Vec3 start, const ClimbingCandidate& candidate,
47 const ClimbingActionDefinition& action,
48 const ClimbingProfileDefinition& profile, std::uint32_t& queryCount,
49 std::uint32_t queryBudget, bool& budgetExceeded,
50 std::uint32_t& moverIterations) {
51 physics::QueryFilter3D filter = profile.queryFilter;
52 filter.ignoredBodyId = candidate.ignoredBodyId;
53 Vec3 current = start;
54 const std::uint32_t remainingBudget = queryCount < queryBudget ? queryBudget - queryCount : 0;
55 const std::uint32_t segmentCount = std::min(profile.pathValidationSegments, remainingBudget);
56 if (segmentCount == 0) {
57 budgetExceeded = true;
58 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
59 "climbing.query_budget.exceeded", "candidate.path");
60 }
61 if (segmentCount < profile.pathValidationSegments) budgetExceeded = true;
62 for (std::uint32_t segment = 1; segment <= segmentCount; ++segment) {
63 const float t = static_cast<float>(segment) / static_cast<float>(segmentCount);
64 const Vec3 target = detail::trajectoryPoint(start, candidate, action, profile, t);
65 const Vec3 desired = subtract(target, current);
66 const float lowerY = current.y + profile.capsuleRadius;
67 const float collisionHeight = action.kind == ClimbingActionKind::Slide
68 ? profile.compactCapsuleHeight
69 : profile.capsuleHeight;
70 const float upperY = current.y + collisionHeight - profile.capsuleRadius;
71 auto moved = world.moveCapsuleOwned(current.x, lowerY, current.z, current.x, upperY, current.z,
72 profile.capsuleRadius, desired.x, desired.y, desired.z, filter);
73 ++queryCount;
74 if (!moved) return eve::Result<void>::failure(moved.status());
75 moverIterations += static_cast<std::uint32_t>(std::max(0, moved.value().iterations));
76 const Vec3 actual{moved.value().deltaX, moved.value().deltaY, moved.value().deltaZ};
77 if (length(subtract(desired, actual)) > profile.skin + 0.001f)
78 return failure<void>(eve::DiagnosticCode::PreconditionViolation, "climbing.candidate.path_blocked",
79 "candidate.path." + std::to_string(segment));
80 current = target;
81 }
82 return eve::Result<void>::success();
83}

106eve::Result<ClimbingRuntime::PreparedBegin> ClimbingRuntime::prepareBeginCandidate(
107 physics::World3D& world, const ClimbingPose& pose, eve::SimulationTick tick, ClimbingCandidate candidate) {
108 if (activePhase(phase_))
109 return failure<PreparedBegin>(eve::DiagnosticCode::Conflict, "a climbing execution is already active",
110 "runtime.phase");
111 if (nextExecutionId_ == 0 || nextExecutionId_ == std::numeric_limits<std::uint64_t>::max())
112 return failure<PreparedBegin>(eve::DiagnosticCode::PreconditionViolation,
113 "climbing execution id space is exhausted", "runtime.nextExecutionId");

123 std::uint32_t moverIterations = 0;

126 lastCounters_.queryCount = lastQueryCount_;

128 if (budgetExceeded) lastCounters_.budgetState = ClimbingQueryBudgetState::Exceeded;
129 telemetry_.updateLatestCounters(lastCounters_);
130 if (!path) return eve::Result<PreparedBegin>::failure(path.status());
131

133 prepared.executionId = ClimbingExecutionId(nextExecutionId_);
134 prepared.candidate = std::move(candidate);
135 prepared.action = *action;
136 prepared.startFeet = pose.feet;
137 prepared.tick = tick;

142eve::Result<ClimbingStart> ClimbingRuntime::commitBegin(PreparedBegin prepared) {
143 if (activePhase(phase_) || execution_)
144 return failure<ClimbingStart>(eve::DiagnosticCode::Conflict, "climbing state changed before commit",
145 "runtime.phase");
146 if (prepared.executionId.isZero() || prepared.executionId.value() != nextExecutionId_)
147 return failure<ClimbingStart>(eve::DiagnosticCode::Conflict, "prepared climbing execution is stale",
148 "executionId");
149 if (!prepared.conditionsSatisfied)
150 return failure<ClimbingStart>(eve::DiagnosticCode::Unsupported,
151 "required climbing conditions were not evaluated by a service authority",
152 "action.requiredConditionTags");
153 auto eventCapacity = requireEventCapacity(1, prepared.tick);
154 if (!eventCapacity) return eve::Result<ClimbingStart>::failure(eventCapacity.status());
155
156 Execution execution;
157 execution.executionId = prepared.executionId;
158 execution.definitionGeneration = prepared.candidate.definitionGeneration;
159 execution.candidate = prepared.candidate;
160 execution.action = std::move(prepared.action);
161 execution.startFeet = prepared.startFeet;
162 execution.currentFeet = prepared.startFeet;
163 execution.lastPlannedFeet = prepared.startFeet;
164 execution.duration = execution.action.duration;
165 execution.lastTick = prepared.tick;
166
167 ClimbingStart started{prepared.executionId, prepared.candidate};
168 execution_ = std::move(execution);
169 previousActionId_ = started.candidate.actionId;
170 if (debugCapture_ == ClimbingDebugCapture::Enabled) motionEvidence_.clear();
171 phase_ = ClimbingPhase::Requested;
172 terminalCode_.clear();
173 ++nextExecutionId_;
174 enqueueEvent({ClimbingEventKind::Started, started.candidate.actionId, prepared.tick, started.executionId});
175 return eve::Result<ClimbingStart>::success(std::move(started), eve::Status::success(eve::StatusCode::Applied));
176}

188eve::Result<ClimbingStart> ClimbingSelectionSystem::tryStart(ClimbingRuntime& runtime, physics::World3D& world,
189 const ClimbingPose& pose, ClimbingIntent& intent,
190 ClimbingCommand command, eve::SimulationTick tick,
191 eve::SimulationTick lastGroundedTick) {
192 if (!ClimbingInputSystem::peek(intent, command, tick))
193 return failure<ClimbingStart>(eve::DiagnosticCode::NotFound, "climbing.input.no_match", "intent.commands");
194
195 ClimbingPose effectivePose = pose;
196 if (!effectivePose.grounded && command != ClimbingCommand::Drop &&
197 ClimbingInputSystem::coyoteWindowState(tick, lastGroundedTick, runtime.profile_.coyoteTicks) ==
198 ClimbingCoyoteState::Eligible)
199 effectivePose.grounded = true;
200
201 auto prepared = runtime.prepareBegin(world, effectivePose, tick);
202 if (!prepared) return eve::Result<ClimbingStart>::failure(prepared.status());
203 if (!commandMatches(prepared.value().action.requiredCommand, command))
204 return failure<ClimbingStart>(eve::DiagnosticCode::NotFound,
205 "selected climbing action does not accept the buffered command",
206 "action.requiredCommand");
207
208 const ClimbingIntent originalIntent = intent;
209 auto consumed = ClimbingInputSystem::consume(intent, command, tick, prepared.value().executionId);
210 if (!consumed || !consumed.value()) {
211 intent = originalIntent;
212 if (!consumed) return eve::Result<ClimbingStart>::failure(consumed.status());
213 return failure<ClimbingStart>(eve::DiagnosticCode::Conflict,
214 "eligible climbing input changed before transaction commit", "intent.commands");
215 }
216
217 auto committed = runtime.commitBegin(std::move(prepared).takeValue());
218 if (!committed) {
219 intent = originalIntent;
220 return eve::Result<ClimbingStart>::failure(committed.status());
221 }
222 return committed;
223}
