33template <class P>
34std::vector<P> chain(const std::vector<P>& points) {
35 if (points.size() <= 1) return points;
36 auto turn = [](const P& a, const P& b, const P& c) {
37 return (double(b[0]) - a[0]) * (double(c[1]) - a[1]) - (double(b[1]) - a[1]) * (double(c[0]) - a[0]);
38 };
39 std::vector<P> hull;
40 for (const auto& p : points) {
41 while (hull.size() >= 2 && turn(hull[hull.size() - 2], hull.back(), p) <= 0) hull.pop_back();
42 hull.push_back(p);
43 }
44 const auto lower = hull.size();
45 for (auto it = points.rbegin() + 1; it != points.rend(); ++it) {
46 while (hull.size() > lower && turn(hull[hull.size() - 2], hull.back(), *it) <= 0) hull.pop_back();
47 hull.push_back(*it);
48 }
49 hull.pop_back();
50 return hull;
51}
