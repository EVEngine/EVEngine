32Result<CombatCameraView> CombatCameraFraming::solve(const CombatCameraFramingRequest& request) {
33 auto valid = request.validate();
34 if (!valid) return Result<CombatCameraView>::failure(valid.status());
35
36 CombatVector3 back = normalizedXZ({-request.playerFacing.x, 0.0, -request.playerFacing.z});
37 CombatVector3 look = {request.player.x, request.player.y + request.lookHeight, request.player.z};
38 if (request.lockTarget) {
39 look = {(request.player.x + request.lockTarget->x) * 0.5,
40 (request.player.y + request.lockTarget->y) * 0.5 + request.lookHeight,
41 (request.player.z + request.lockTarget->z) * 0.5};
42 CombatVector3 toTarget{request.lockTarget->x - request.player.x, 0.0, request.lockTarget->z - request.player.z};
43 if (lengthXZ(toTarget) > 1e-6) back = normalizedXZ({-toTarget.x, 0.0, -toTarget.z});
44 }
45 CombatCameraView view;
46 view.lookAt = look;
47 view.eye = {request.player.x + back.x * request.distance, request.player.y + request.height,
48 request.player.z + back.z * request.distance};
49 return Result<CombatCameraView>::success(view);
50}
