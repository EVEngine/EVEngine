21Result<RibbonMesh> buildRibbons(const StrandsDatas &strands, const RibbonParams &params) {
22 auto validated = strands.validate();
23 if (!validated.ok()) return Result<RibbonMesh>::failure(validated.status());
24
25 if (params.widthScale <= 0.f || !std::isfinite(params.widthScale)) {

27 DiagnosticCode::InvalidArgument, "buildRibbons: widthScale must be positive",
28 "hair.ribbon.widthScale"));
29 }
30

33 RibbonMesh mesh;
34 const size_t curveN = strands.curveCount();

36 mesh.nrmXYZ.reserve(curveN * 8u * 3u);
37 mesh.uvST.reserve(curveN * 8u * 2u);
38 mesh.indices.reserve(curveN * 7u * 6u);

40 for (size_t ci = 0; ci < curveN; ++ci) {
41 const auto pts = strands.curvePoints(ci);
42 if (pts.size() < 2) continue;
43
44 const uint32_t baseVert = uint32_t(mesh.posXYZ.size() / 3u);
45 for (size_t pi = 0; pi < pts.size(); ++pi) {
46 const StrandPoint &p = pts[pi];
47 glm::vec3 tangent(0.f, 1.f, 0.f);
48 if (pi + 1 < pts.size()) {
49 tangent = pts[pi + 1].position - p.position;
50 } else {
51 tangent = p.position - pts[pi - 1].position;
52 }
53 tangent = safeNormalize(tangent, glm::vec3(0.f, 1.f, 0.f));
54
55 glm::vec3 side = glm::cross(tangent, sideHint);
56 if (glm::dot(side, side) < 1e-10f) {
57 side = glm::cross(tangent, glm::vec3(0.f, 0.f, 1.f));
58 }
59 side = safeNormalize(side, glm::vec3(1.f, 0.f, 0.f));
60
61 const float halfW = std::max(p.radius * params.widthScale, params.minWidth);
62 const glm::vec3 left = p.position - side * halfW;
63 const glm::vec3 right = p.position + side * halfW;

65 auto pushVert = [&](const glm::vec3 &pos, float vSide) {

67 mesh.posXYZ.push_back(pos.y);
68 mesh.posXYZ.push_back(pos.z);

70 mesh.nrmXYZ.push_back(tangent.y);
71 mesh.nrmXYZ.push_back(tangent.z);
72 mesh.uvST.push_back(vSide);
73 mesh.uvST.push_back(p.u);
74 };
75 pushVert(left, 0.f);
76 pushVert(right, 1.f);
77 }
78
79 const uint32_t segs = uint32_t(pts.size() - 1u);
80 for (uint32_t s = 0; s < segs; ++s) {
81 const uint32_t i0 = baseVert + s * 2u;
82 const uint32_t i1 = i0 + 1u;
83 const uint32_t i2 = i0 + 2u;
84 const uint32_t i3 = i0 + 3u;
85 mesh.indices.push_back(i0);
86 mesh.indices.push_back(i2);
87 mesh.indices.push_back(i1);
88 mesh.indices.push_back(i1);
89 mesh.indices.push_back(i2);
90 mesh.indices.push_back(i3);
91 }
92 }

96 DiagnosticCode::Failed, "buildRibbons: produced empty mesh", "hair.ribbon"));
97 }
98 return Result<RibbonMesh>::success(std::move(mesh));
99}
