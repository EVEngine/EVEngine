121std::string pollJob(const std::string& token, uint64_t generation) {
122 auto it = jobs.find(token);
123 if (it == jobs.end() || it->second.generation != generation) return error("StaleJob");
124 auto& job = it->second;
125 if (job.cancelled->load()) return error("CancelledJob");
126 if (job.response.empty()) {
127 if (job.future.wait_for(std::chrono::seconds(0)) != std::future_status::ready) return pending(token);
128 try {
129 job.response = job.future.get().response;
130 } catch (const std::exception& e) {
131 job.response = error(std::string("WorkerFailed: ") + e.what());
132 }
133 }
134 return job.response;
135}
136bool validArguments(Object::Ptr args, const std::set<std::string>& allowed) {
137 if (!args) return true;
138 for (const auto& [key, value] : *args) {
139 if (!allowed.contains(key)) return false;
140 if (key == "buffers") {
141 if (value.type() != typeid(Array::Ptr)) return false;
142 } else if (key == "timeoutMs" || key == "x" || key == "y" || key == "channel" || key == "maxWidth" ||
143 key == "maxHeight" || key == "maxBytes") {
144 if (!value.isInteger() || value.convert<double>() < -2147483648.0 || value.convert<double>() > 2147483647.0)
145 return false;
146 } else if (key == "rangeMin" || key == "rangeMax" || key == "exposure") {
147 if (!value.isNumeric()) return false;
148 } else if (!value.isString())
149 return false;
150 }
151 return true;
152}
