199Shader* Graphics::prepareSceneColorResolveShader(Texture* scene) {
200 if (!scene) return nullptr;
201 AntiAliasing* aa = pipelineAntiAliasing();
202 const bool doAA = !renderControl_ || renderControl_->isEnabled("aa") || renderControl_->isEnabled("taa");
203 aa->setMode("fxaa");
204 if (doAA) {
205 const std::string requested =
206 renderControl_ ? renderControl_->getPostProcessQuality() : "high";
207 const std::string quality = requested == "ultra" ? "high" : requested;
208 if (aa->getQuality() != quality) aa->setQuality(quality);
209 } else {
210 if (aa->getMode() == "fxaa") {
211 aa->setFloat("edgeThreshold", 1.f);
212 aa->setFloat("edgeThresholdMin", 1.f);
213 aa->setFloat("subpix", 0.f);
214 }
215 }
216 aa->prepareSource(scene);
217 return aa->getShader();
218}

221 Texture* scene = getSceneColorTexture();
222 if (!scene) return;
223 // Queue a scene-color blit at this 2D overlay slot so present() owns

227 // the auto ACES composite — two fullscreen 3D quads that strobe.
228 drawTexturedRectShaderUV(scene, nullptr, x, y, w, h, 0.f, 0.f, 1.f, 1.f,
229 Color(r, g, b, a), false, BlendMode::Opaque);

1207Texture* Graphics::prepareFinalSceneTexture(Texture *scene, Texture *motion) {
1208 if (!scene) return nullptr;
1209 Texture *resolved = scene;
1210 AntiAliasing *aa = pipelineAntiAliasing();
1211 const bool temporal = renderControl_ && renderControl_->isEnabled("taa");
1212 const bool spatial = !renderControl_ || renderControl_->isEnabled("aa");
1213 const std::string requested = renderControl_ ? renderControl_->getPostProcessQuality() : "high";
1214 const std::string quality = requested == "ultra" ? "high" : requested;
1215 if (aa->getQuality() != quality) aa->setQuality(quality);
1216 if (temporal) {
1217 aa->setMode("taa");
1218 resolved = aa->resolveTemporal(this, scene, motion);
1219 } else if (spatial) {
1220 aa->setMode("fxaa");
1221 const int width = std::max(scene->getWidth(), 1);
1222 const int height = std::max(scene->getHeight(), 1);
1223 if (!spatialAAResolve_ || spatialAAResolveWidth_ != width ||
1224 spatialAAResolveHeight_ != height) {
1225 spatialAAResolve_ = newHDRCanvas(width, height);
1226 spatialAAResolveWidth_ = width;
1227 spatialAAResolveHeight_ = height;
1228 }
1229 aa->applyTo(this, scene, spatialAAResolve_);
1230 resolved = spatialAAResolve_->getTexture();
1231 }
1232 if (getSceneDofMaxBlur() > 0.f && renderControl_) {
1233 GBuffer *gb = renderControl_->getGBuffer();
1234 if (gb && gb->isValid()) {
1235 Texture *depth = getBackendName() == "webgpu" ? gb->getDepthTexture()
1236 : gb->getHwDepthTexture();
1237 if (depth) {
1238 resolved = pipelineDepthOfField()->apply(
1239 resolved, depth, getSceneDofFocusDistance(), getSceneDofMaxBlur(),
1240 getSceneDofFocusRange(), getSceneDofNearZ(), getSceneDofFarZ());
1241 }
1242 }
1243 }
1244 Texture *meterSource = resolved;
1245 resolved = pipelineBloom()->apply(resolved, getSceneBloomIntensity(),
1246 getSceneBloomThreshold());
1247 return pipelineExposure()->apply(resolved, getSceneExposure(), getSceneAutoExposure(),
1248 getSceneAutoExposureMinEV(), getSceneAutoExposureMaxEV(),
1249 meterSource);
1250}
