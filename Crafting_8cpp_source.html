21eve::Value encodeReservation(const CraftingRecipe& recipe, std::uint32_t batchSize) {
22 eve::Value::Array outputs;
23 outputs.reserve(recipe.outputs.size());
24 for (const auto& output : recipe.outputs) {
25 outputs.emplace_back(eve::Value::Object{
26 {"item", eve::Value(output.itemId)},
27 {"quantity", eve::Value(std::int64_t(output.quantity) * static_cast<std::int64_t>(batchSize))}});
28 }
29 return eve::Value(eve::Value::Object{{"schema", eve::Value("eve.rpg.crafting-reservation")},
30 {"version", eve::Value(std::int64_t(1))},
31 {"outputs", eve::Value(std::move(outputs))}});
32}

34eve::Result<eve::resource::CostSpec> scaleCost(
35 const eve::resource::CostSpec& source, std::uint32_t batchSize) {
36 const auto multiplier = static_cast<std::int64_t>(batchSize);
37 std::vector<eve::resource::ResourceCost> items;
38 items.reserve(source.items().size());
39 for (const auto& item : source.items()) {
40 if (item.amount.value() > std::numeric_limits<std::int64_t>::max() / multiplier)
41 return failure<eve::resource::CostSpec>(eve::DiagnosticCode::InvalidArgument,
42 "crafting batch ingredient quantity overflows", "batchSize");
43 auto scaled = eve::resource::ResourceCost::create(
44 item.resource.value(), item.amount.value() * multiplier);
45 if (!scaled) return eve::Result<eve::resource::CostSpec>::failure(scaled.status());
46 items.push_back(std::move(scaled).takeValue());
47 }
48 return eve::resource::CostSpec::create(std::move(items));
49}

51eve::Result<std::vector<inventory::InventoryItemGrant>> decodeOutputs(const eve::Value& reservation) {
52 const auto* object = reservation.getIf<eve::Value::Object>();
53 if (object == nullptr)
54 return failure<std::vector<inventory::InventoryItemGrant>>(
55 eve::DiagnosticCode::ParseError, "crafting reservation must be an object", "reservation");
56 const auto schema = object->find("schema");
57 const auto version = object->find("version");
58 const auto outputs = object->find("outputs");
59 if (schema == object->end() || !schema->second.isString() ||
60 schema->second.asString() != "eve.rpg.crafting-reservation" || version == object->end() ||
61 !version->second.isInt64() || version->second.asInt() != 1 || outputs == object->end() ||
62 !outputs->second.isArray())
63 return failure<std::vector<inventory::InventoryItemGrant>>(
64 eve::DiagnosticCode::ParseError, "crafting reservation schema is invalid", "reservation");
65 std::vector<inventory::InventoryItemGrant> result;
66 const auto* outputValues = outputs->second.getIf<eve::Value::Array>();
67 for (const auto& value : *outputValues) {
68 const auto* entry = value.getIf<eve::Value::Object>();
69 if (entry == nullptr)
70 return failure<std::vector<inventory::InventoryItemGrant>>(
71 eve::DiagnosticCode::ParseError, "crafting output must be an object", "reservation.outputs");
72 const auto item = entry->find("item");
73 const auto quantity = entry->find("quantity");
74 if (item == entry->end() || !item->second.isString() || item->second.asString().empty() ||
75 quantity == entry->end() || !quantity->second.isInt64() || quantity->second.asInt() <= 0 ||
76 quantity->second.asInt() > std::numeric_limits<int>::max())
77 return failure<std::vector<inventory::InventoryItemGrant>>(
78 eve::DiagnosticCode::ParseError, "crafting output fields are invalid", "reservation.outputs");
79 result.push_back({item->second.asString(), static_cast<int>(quantity->second.asInt())});
80 }
81 if (result.empty())
82 return failure<std::vector<inventory::InventoryItemGrant>>(
83 eve::DiagnosticCode::InvalidArgument, "crafting reservation has no outputs", "reservation.outputs");
84 return eve::Result<std::vector<inventory::InventoryItemGrant>>::success(std::move(result));
85}

95 auto snapshot = queue_.snapshot();

97 before_ = std::move(snapshot).takeValue();
98 staged_ = std::make_unique<production::WorkQueue>();
99 auto restored = staged_->restore(before_);
100 if (!restored) return restored;

102 if (!enqueued) return eve::Result<void>::failure(enqueued.status());

104 auto after = staged_->snapshot();
105 if (!after) return eve::Result<void>::failure(after.status());
106 after_ = std::move(after).takeValue();

112 if (!prepared_ || committed_)
113 return failure<void>(eve::DiagnosticCode::Conflict, "crafting production is not prepared", "production");
114 auto current = queue_.snapshot();
115 if (!current) return eve::Result<void>::failure(current.status());
116 if (current.value() != before_)
117 return failure<void>(eve::DiagnosticCode::StaleHandle,
118 "crafting production queue changed while staged", "production");
119 auto restored = queue_.restore(after_);
120 if (!restored) return restored;
121 committed_ = true;
122 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));

138 if (current.value() != after_)
139 return failure<void>(eve::DiagnosticCode::StaleHandle,
140 "crafting production queue changed before compensation", "production");
141 auto restored = queue_.restore(before_);
142 if (!restored) return restored;
143 staged_.reset();
144 prepared_ = false;
145 committed_ = false;
146 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
147 }

282eve::Result<CraftingReceipt> Crafting::begin(CraftingRequest request) {
283 if (request.production == nullptr || request.ingredients == nullptr)
284 return failure<CraftingReceipt>(eve::DiagnosticCode::InvalidArgument,
285 "crafting requires production and ingredient account ports", "request");
286 if (request.owner.empty() || request.recipe.id.empty() || request.recipe.process.empty())
287 return failure<CraftingReceipt>(eve::DiagnosticCode::InvalidArgument,
288 "crafting owner, recipe id and process are required", "request");
289 if (!request.recipe.ingredients.isValid() || request.recipe.outputs.empty() || request.batchSize == 0 ||
290 request.recipe.duration.nanoseconds() <= 0)
291 return failure<CraftingReceipt>(eve::DiagnosticCode::InvalidArgument,
292 "crafting recipe requires ingredients, outputs and positive duration",
293 "recipe");
294 std::int64_t totalOutput = 0;
295 for (const auto& output : request.recipe.outputs) {
296 const auto quantity = static_cast<std::int64_t>(output.quantity);
297 const auto batchSize = static_cast<std::int64_t>(request.batchSize);
298 if (output.itemId.empty() || quantity <= 0 ||
299 quantity > static_cast<std::int64_t>(std::numeric_limits<int>::max()) / batchSize)
300 return failure<CraftingReceipt>(eve::DiagnosticCode::InvalidArgument,
301 "crafting outputs require item ids and positive quantities",
302 "recipe.outputs");
303 const auto scaled = quantity * batchSize;
304 if (totalOutput > std::numeric_limits<int>::max() - scaled)
305 return failure<CraftingReceipt>(eve::DiagnosticCode::InvalidArgument,
306 "crafting batch output total exceeds inventory limits",
307 "recipe.outputs");
308 totalOutput += scaled;
309 }
310
311 production::ProductionRequest productionRequest;
312 productionRequest.owner = std::move(request.owner);
313 productionRequest.kind = request.recipe.process;
314 productionRequest.product = request.recipe.id;
315 productionRequest.duration = request.recipe.duration;
316 productionRequest.priority = request.priority;
317 productionRequest.definition = request.recipe.definition;
318 productionRequest.reservation = encodeReservation(request.recipe, request.batchSize);
319 productionRequest.batchSize = request.batchSize;
320 productionRequest.context = eve::Value(eve::Value::Object{{"recipe", eve::Value(request.recipe.id)},
321 {"process", eve::Value(request.recipe.process)}});
322 productionRequest.settlementRequired = true;
323
324 if (request.transactionId.empty()) request.transactionId = "rpg.crafting." + request.recipe.id;
325 auto batchIngredients = scaleCost(request.recipe.ingredients, request.batchSize);
326 if (!batchIngredients) return eve::Result<CraftingReceipt>::failure(batchIngredients.status());
327 transaction::TransactionContext context(std::move(request.transactionId));
328 ProductionParticipant participant(*request.production, std::move(productionRequest));
329 auto committed = transaction::AtomicResourcePayment::execute(
330 context, *request.ingredients, batchIngredients.value(), participant);
331 if (!committed) return eve::Result<CraftingReceipt>::failure(committed.status());
332 CraftingReceipt receipt{std::move(committed).takeValue(), participant.taskId()};
333 return eve::Result<CraftingReceipt>::success(std::move(receipt),
334 eve::Status::success(eve::StatusCode::Applied));
335}

337eve::Result<int> Crafting::settle(production::WorkQueue& queue, inventory::Bag& output,
338 std::string_view taskId) {
339 auto task = queue.find(taskId);
340 if (!task)
341 return failure<int>(eve::DiagnosticCode::NotFound, "crafting task was not found", "taskId");
342 const std::string settlementId = "rpg.crafting:" + std::string(taskId);
343 if (task->get().state == production::TaskState::Completed &&
344 task->get().settlement.settlementId == settlementId)
345 return eve::Result<int>::success(0, eve::Status::success(eve::StatusCode::NoOp));
346 if (task->get().state != production::TaskState::ReadyToSettle &&
347 task->get().state != production::TaskState::SettlementFailed)
348 return failure<int>(eve::DiagnosticCode::Conflict, "crafting task is not ready to settle", "taskId");
349 auto outputs = decodeOutputs(task->get().reservation);
350 if (!outputs) return eve::Result<int>::failure(outputs.status());
351 InventoryAddParticipant inventoryParticipant(output, outputs.value());
352 production::ProductionSettlementReceipt receipt;
353 receipt.settlementId = settlementId;
354 receipt.payload = eve::Value(eve::Value::Object{{"destination", eve::Value(output.getId())}});
355 SettlementParticipant settlementParticipant(queue, std::string(taskId), std::move(receipt));
356 transaction::TransactionContext context("rpg.crafting.settle:" + std::string(taskId));
357 std::array<transaction::ITransactionParticipant*, 2> participants{
358 &settlementParticipant, &inventoryParticipant};
359 transaction::Coordinator coordinator;
360 auto committed = coordinator.execute(context, std::span<transaction::ITransactionParticipant*>(participants));
361 if (!committed) {
362 const auto* diagnostic = committed.error();
363 auto failed = queue.failSettlement(taskId,
364 diagnostic == nullptr ? "crafting output rejected" : diagnostic->message());
365 const auto current = queue.find(taskId);
366 if (!failed && (!current || current->get().state != production::TaskState::SettlementFailed))
367 return eve::Result<int>::failure(failed.status());
368 return eve::Result<int>::failure(committed.status());
369 }
370 return eve::Result<int>::success(inventoryParticipant.added(),
371 eve::Status::success(eve::StatusCode::Applied));
372}
