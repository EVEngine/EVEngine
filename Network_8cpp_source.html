88bool Network::httpRequest(const std::string& method, const std::string& url,
89 const std::string& body, int timeoutMs, int& status,
90 std::string& responseBody) {
91 HttpRequest* req = newHttp(method, url);
92 if (!req) return false;
93 if (!body.empty()) req->setBodyString(body);
94 req->setTimeout(timeoutMs > 0 ? timeoutMs : 10000);
95 if (!req->submit()) return false;
96
97 const auto deadline = std::chrono::steady_clock::now() +
98 std::chrono::milliseconds(timeoutMs > 0 ? timeoutMs : 1);
99 while (std::chrono::steady_clock::now() < deadline) {
100 std::vector<NetCompletion> out;
101 drainForTest(out);
102 for (const auto& c : out) {
103 if (c.handle != req) continue;
104 if (c.type == NetEvType::HttpResp) {
105 status = c.status;
106 if (c.bytes) responseBody.assign(c.bytes->begin(), c.bytes->end());
107 return true;
108 }
109 if (c.type == NetEvType::Err) return false;
110 }
111 std::this_thread::sleep_for(std::chrono::milliseconds(10));
112 }
113 return false;
114}

230void Network::pollSockets() {
231 std::vector<TcpSocket*> tcpCopy;
232 std::vector<UdpSocket*> udpCopy;
233 {
234 std::lock_guard<std::mutex> lock(watchMu_);
235 tcpCopy = watchedTcp_;
236 udpCopy = watchedUdp_;
237 }
238 for (TcpSocket* sock : tcpCopy) {
239 if (!sock) continue;
240 if (sock->isListening() && sock->server()) {
241 try {
242 Poco::Net::StreamSocket ss = sock->server()->acceptConnection();
243 ss.setBlocking(false);
244 auto peer = std::make_unique<TcpSocket>(this);
245 auto stream = std::make_unique<Poco::Net::StreamSocket>(ss);
246 peer->setConnectedSocket(std::move(stream));
247 TcpSocket* raw = peer.get();
248 sock->pushAccepted(std::move(peer));
249 watchTcp(raw);
250 NetCompletion c;
251 c.type = NetEvType::Conn;
252 c.kind = NetKind::Tcp;
253 c.handle = raw;
254 c.reason = "ok";
255 c.peer = raw->getPeer();
256 post(std::move(c));
257 } catch (...) {
258 }
259 }
260 if (sock->isConnected() && sock->stream()) {
261 sock->flushSend();
262 try {
263 char buf[64 * 1024];
264 int n = sock->stream()->receiveBytes(buf, sizeof(buf));
265 if (n > 0) {
266 NetCompletion c;
267 c.type = NetEvType::Data;
268 c.kind = NetKind::Tcp;
269 c.handle = sock;
270 c.bytes = std::make_shared<std::vector<char>>(buf, buf + n);
271 c.peer = sock->getPeer();
272 post(std::move(c));
273 } else if (n == 0) {
274 NetCompletion c;
275 c.type = NetEvType::Err;
276 c.kind = NetKind::Tcp;
277 c.handle = sock;
278 c.reason = "closed";
279 post(std::move(c));
280 unwatchTcp(sock);
281 }
282 } catch (const Poco::Exception&) {
283 } catch (...) {
284 }
285 }
286 }
287 for (UdpSocket* sock : udpCopy) {
288 if (!sock || !sock->datagram()) continue;
289 for (int i = 0; i < 64; ++i) {
290 try {
291 char buf[64 * 1024];
292 Poco::Net::SocketAddress sender;
293 int n = sock->datagram()->receiveFrom(buf, sizeof(buf), sender);
294 if (n <= 0) break; // would-block
295 NetCompletion c;
296 c.type = NetEvType::Data;
297 c.kind = NetKind::Udp;
298 c.handle = sock;
299 c.bytes = std::make_shared<std::vector<char>>(buf, buf + n);
300 c.peer = sender.toString();
301 post(std::move(c));
302 } catch (const Poco::Exception&) {
303 break;
304 } catch (...) {
305 break;
306 }
307 }
308 }

311void Network::emitCompletion(const NetCompletion& c) {
312 auto* ev = eve::ModuleManager::getInstance<eve::event::Event>("Event");
313 if (!ev) return;
314
315 using eve::event::Variant;
316 using eve::event::Message;
317
318 std::vector<Variant> args;
319 switch (c.type) {
320 case NetEvType::Conn:
321 args.push_back(Variant::makePtr(c.handle));
322 args.push_back(Variant::makeString(c.reason.empty() ? "ok" : c.reason));
323 ev->push(new Message("netconn", args));
324 break;
325 case NetEvType::Data: {
326 Channel* ch = channelFor(static_cast<TcpSocket*>(c.handle));
327 if (ch && c.bytes) {
328 ch->feed(*c.bytes);
329 return;
330 }
331 eve::data::ByteData* bd = nullptr;
332 if (c.bytes && !c.bytes->empty())
333 bd = new eve::data::ByteData(c.bytes->data(), c.bytes->size());
334 args.push_back(Variant::makePtr(c.handle));
335 args.push_back(Variant::makePtr(bd));
336 args.push_back(Variant::makeString(c.peer));
337 ev->push(new Message("netdata", args));
338 break;
339 }
340 case NetEvType::Err: {
341 Channel* ch = channelFor(static_cast<TcpSocket*>(c.handle));
342 if (ch) {
343 args.push_back(Variant::makePtr(ch));
344 args.push_back(Variant::makeString(c.reason));
345 ev->push(new Message("chclose", args));
346 }
347 args.clear();
348 args.push_back(Variant::makePtr(c.handle));
349 args.push_back(Variant::makeString(c.reason));
350 ev->push(new Message("neterr", args));
351 break;
352 }
353 case NetEvType::HttpResp: {
354 eve::data::ByteData* bd = nullptr;
355 if (c.bytes && !c.bytes->empty())
356 bd = new eve::data::ByteData(c.bytes->data(), c.bytes->size());
357 args.push_back(Variant::makePtr(c.handle));
358 args.push_back(Variant::makeInt(c.status));
359 args.push_back(Variant::makePtr(bd));
360 ev->push(new Message("httpresp", args));
361 break;
362 }
363 case NetEvType::ChMsg: {
364 eve::data::ByteData* bd = nullptr;
365 if (c.bytes && !c.bytes->empty())
366 bd = new eve::data::ByteData(c.bytes->data(), c.bytes->size());
367 args.push_back(Variant::makePtr(c.handle));
368 args.push_back(Variant::makePtr(bd));
369 ev->push(new Message("chmsg", args));
370 break;
371 }
372 case NetEvType::ChClose:
373 args.push_back(Variant::makePtr(c.handle));
374 args.push_back(Variant::makeString(c.reason));
375 ev->push(new Message("chclose", args));
376 break;
377 }
378}

380void Network::pump() {
381 std::vector<NetCompletion> batch;
382 if (worker_) worker_->drain(batch);
383 const int64_t now = std::chrono::duration_cast<std::chrono::milliseconds>(
384 std::chrono::steady_clock::now().time_since_epoch())
385 .count();
386 for (auto& c : batch) {
387 if (c.type == NetEvType::Data && c.kind == NetKind::Udp) {
388 auto* sock = static_cast<UdpSocket*>(c.handle);
389 auto hit = udpHosts_.find(sock);
390 if (hit != udpHosts_.end()) {
391 if (c.bytes) hit->second->onDatagram(*c.bytes, c.peer);
392 continue;
393 }
394 auto lit = udpLinks_.find(sock);
395 if (lit != udpLinks_.end()) {
396 if (c.bytes) lit->second->onDatagram(*c.bytes, c.peer);
397 continue;
398 }
399 }
400 emitCompletion(c);
401 }
402 for (auto& kv : udpHosts_) kv.second->pump(now);
403 for (auto& kv : udpLinks_) kv.second->pump(now);
404}
