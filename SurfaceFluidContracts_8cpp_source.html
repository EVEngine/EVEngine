24eve::Result<void> validateTick(const eve::SimulationStep& step, eve::SimulationTick lastTick, const char* path) {
25 const double seconds = step.delta.seconds();
26 if (!std::isfinite(seconds) || seconds < 0.0)
27 return invalidInput(path, "Surface-fluid duration must be finite and non-negative");
28 if (!(step.tick > lastTick))
29 return eve::Result<void>::failure(eve::Diagnostic::error(
30 eve::DiagnosticCode::PreconditionViolation, "Surface-fluid simulation tick must increase strictly", path));
31 return applied();
32}

43eve::Result<void> FluidSimulationAdapter::step(const eve::SimulationStep& stepValue) {
44 if (!solver_) return staleSolver("fluids.surfaceSimulation.solver");
45 auto valid = validateTick(stepValue, lastTick_, "fluids.surfaceSimulation.step");
46 if (!valid) return valid;
47 solver_->step(static_cast<float>(stepValue.delta.seconds()));
48 lastTick_ = stepValue.tick;
49 return applied();

86eve::Result<SurfaceSample> FluidSurfaceConstraintAdapter::evaluate(const SurfaceLocation& location,
87 eve::Duration poseDelta) const {
88 auto validDuration = validateDuration(poseDelta, "fluids.surfaceConstraint.evaluate.dt");
89 if (!validDuration) return eve::Result<SurfaceSample>::failure(validDuration.status());
90 if (!binding_ || !binding_->isValid())
91 return eve::Result<SurfaceSample>::failure(eve::Diagnostic::error(
92 eve::DiagnosticCode::PreconditionViolation, "Cannot evaluate an invalid surface constraint",
93 "fluids.surfaceConstraint.evaluate"));
94 if (location.triangle >= static_cast<std::uint32_t>(binding_->triangleCount()))
95 return eve::Result<SurfaceSample>::failure(eve::Diagnostic::error(
96 eve::DiagnosticCode::NotFound, "Surface location triangle is outside the bound topology",
97 "fluids.surfaceConstraint.evaluate.triangle"));
98 return eve::Result<SurfaceSample>::success(binding_->evaluate(location, static_cast<float>(poseDelta.seconds())));
99}

144 if (!renderer_) return staleSolver("fluids.screenSpaceReconstruction.renderer");
145 if (!std::isfinite(particleRadius) || particleRadius <= 0.f)
