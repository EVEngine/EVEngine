50EditorResult<void> SceneTargetBase::applyDomainOperation(const DomainOperation& operation) {
51 if (operation.target != TargetId(id_))
52 return sceneError<void>(EditorStatus::Rejected, "editor.scene.target-mismatch",
53 "Scene operation targets another backend");
54 if (operation.type == "scene.object.create.v1") {
55 EditorResult<SceneObjectSnapshot> parsed = parseObject(operation.payload);
56 if (!parsed.ok()) return EditorResult<void>::failure(parsed.status());
57 const SceneObjectSnapshot& object = parsed.value();
58 if (object.id.empty() || objects_.contains(object.id))
59 return sceneError<void>(EditorStatus::Conflict, "editor.scene.object-exists",
60 "Scene object id is empty or already exists");
61 if (!object.parent.empty() && !objects_.contains(object.parent))
62 return sceneError<void>(EditorStatus::NotFound, "editor.scene.parent-not-found",
63 "Scene object parent does not exist");
64 objects_.emplace(object.id, object);
65 } else if (operation.type == "scene.object.delete.v1") {
66 EditorResult<SceneObjectSnapshot> parsed = parseObject(operation.payload);
67 if (!parsed.ok()) return EditorResult<void>::failure(parsed.status());
68 const ObjectId& id = parsed.value().id;
69 if (!objects_.contains(id))
70 return sceneError<void>(EditorStatus::NotFound, "editor.scene.object-not-found",
71 "Scene object does not exist");
72 for (const auto& [childId, child] : objects_) {
73 (void)childId;
74 if (child.parent == id)
75 return sceneError<void>(EditorStatus::Rejected, "editor.scene.object-has-children",
76 "Scene object with children cannot be deleted by this operation");
77 }
78 objects_.erase(id);
79 } else if (operation.type == "scene.transform.set.v1") {
80 const EditorValue* idValue = field(operation.payload, "id");
81 const EditorValue* transform = field(operation.payload, "transform");
82 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
83 if (!id || !transform)
84 return sceneError<void>(EditorStatus::Rejected, "editor.scene.transform-payload",
85 "Transform operation requires id and transform");
86 auto object = objects_.find(ObjectId(*id));
87 if (object == objects_.end())
88 return sceneError<void>(EditorStatus::NotFound, "editor.scene.object-not-found",
89 "Scene object does not exist");
90 EditorResult<SceneTransformValue> parsed = parseTransform(*transform);
91 if (!parsed.ok()) return EditorResult<void>::failure(parsed.status());
92 object->second.transform = parsed.value();
93 } else if (operation.type == "scene.transform.multi.set.v1") {
94 const auto* entries = operation.payload.getIf<EditorValue::Array>();
95 if (!entries)
96 return sceneError<void>(EditorStatus::Rejected, "editor.scene.multi-transform-payload",
97 "Multi-transform operation requires an array payload");
98 std::vector<std::pair<ObjectId, SceneTransformValue>> updates;
99 updates.reserve(entries->size());
100 for (const EditorValue& entry : *entries) {
101 const EditorValue* idValue = field(entry, "id");
102 const EditorValue* transform = field(entry, "transform");
103 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
104 if (!id || !transform || !objects_.contains(ObjectId(*id)))
105 return sceneError<void>(EditorStatus::NotFound, "editor.scene.multi-transform-object",
106 "Multi-transform entry references an invalid scene object");
107 auto parsed = parseTransform(*transform);
108 if (!parsed.ok())
109 return sceneError<void>(EditorStatus::Rejected, "editor.scene.multi-transform-value",
110 "Multi-transform entry contains an invalid transform");
111 updates.emplace_back(ObjectId(*id), parsed.value());
112 }
113 for (const auto& [id, transform] : updates) objects_.at(id).transform = transform;
114 } else if (operation.type == "scene.object.rename.v1") {
115 const EditorValue* idValue = field(operation.payload, "id");
116 const EditorValue* nameValue = field(operation.payload, "name");
117 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
118 const auto* name = nameValue ? nameValue->getIf<std::string>() : nullptr;
119 if (!id || !name || name->empty())
120 return sceneError<void>(EditorStatus::Rejected, "editor.scene.rename-payload",
121 "Rename operation requires id and a non-empty name");
122 auto object = objects_.find(ObjectId(*id));
123 if (object == objects_.end())
124 return sceneError<void>(EditorStatus::NotFound, "editor.scene.object-not-found",
125 "Scene object does not exist");
126 object->second.name = *name;
127 } else if (operation.type == "scene.object.reparent.v1") {
128 const EditorValue* idValue = field(operation.payload, "id");
129 const EditorValue* parentValue = field(operation.payload, "parent");
130 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
131 const auto* parent = parentValue ? parentValue->getIf<std::string>() : nullptr;
132 if (!id || !parent)
133 return sceneError<void>(EditorStatus::Rejected, "editor.scene.reparent-payload",
134 "Reparent operation requires id and parent");
135 auto object = objects_.find(ObjectId(*id));
136 if (object == objects_.end())
137 return sceneError<void>(EditorStatus::NotFound, "editor.scene.object-not-found",
138 "Scene object does not exist");
139 if (!parent->empty() && !objects_.contains(ObjectId(*parent)))
140 return sceneError<void>(EditorStatus::NotFound, "editor.scene.parent-not-found",
141 "Scene object parent does not exist");
142 ObjectId ancestor(*parent);
143 while (!ancestor.empty()) {
144 if (ancestor == object->first)
145 return sceneError<void>(EditorStatus::Rejected, "editor.scene.hierarchy-cycle",
146 "Reparenting would create a scene hierarchy cycle");
147 auto node = objects_.find(ancestor);
148 if (node == objects_.end()) break;
149 ancestor = node->second.parent;
150 }
151 object->second.parent = ObjectId(*parent);
152 } else {
153 return sceneError<void>(EditorStatus::Unsupported, "editor.scene.operation-unsupported",
154 "Scene target does not support operation: " + operation.type);
155 }
156 ++revision_;
157 dirty_.include(0, 0);
158 return eve::editing::applied<void>();
159}

192 if (request.id.empty())
193 return sceneError<DomainOperation>(EditorStatus::Rejected, "editor.scene.missing-object-id",
194 "Scene object id is required");
195 if (objects_.contains(request.id))
196 return sceneError<DomainOperation>(EditorStatus::Conflict, "editor.scene.object-exists",
197 "Scene object already exists: " + request.id.value());
198 if (!request.parent.empty() && !objects_.contains(request.parent))

202 DomainOperation operation;
203 operation.type = "scene.object.create.v1";
204 operation.inverseType = "scene.object.delete.v1";
205 operation.target = TargetId(id_);
206 operation.payload = objectValue(object);
207 operation.inverse = objectValue(object);
208 operation.hasInverse = true;
209 operation.affectedObjects.push_back({TargetId(id_), request.id.value(), 0});
210 return eve::editing::applied<DomainOperation>(std::move(operation));

258EditorResult<DomainOperation> SceneTargetBase::makeReparent(const ObjectId& id, const ObjectId& parent) const {
259 auto object = objects_.find(id);
260 if (object == objects_.end())
261 return sceneError<DomainOperation>(EditorStatus::NotFound, "editor.scene.object-not-found",
262 "Scene object does not exist: " + id.value());
263 if (!parent.empty() && !objects_.contains(parent))
264 return sceneError<DomainOperation>(EditorStatus::NotFound, "editor.scene.parent-not-found",

277 payload["id"] = id.value();

280 };
281 DomainOperation operation;
282 operation.type = "scene.object.reparent.v1";
283 operation.target = TargetId(id_);
284 operation.payload = value(parent);
285 operation.inverse = value(object->second.parent);
286 operation.hasInverse = true;
287 operation.affectedObjects.push_back({TargetId(id_), id.value(), 0});
288 operation.affectedProperties.push_back("parent");

349EditorResult<SceneTransformValue> SceneTargetBase::parseTransform(const EditorValue& value) {
350 const EditorValue* xValue = field(value, "x");
351 const EditorValue* yValue = field(value, "y");
352 const EditorValue* zValue = field(value, "z");
353 const auto* x = xValue ? xValue->getIf<double>() : nullptr;
354 const auto* y = yValue ? yValue->getIf<double>() : nullptr;
355 const auto* z = zValue ? zValue->getIf<double>() : nullptr;
356 if (!x || !y || !z)
357 return sceneError<SceneTransformValue>(EditorStatus::Rejected, "editor.scene.transform-value",
358 "Transform requires numeric x, y and z fields");
359 SceneTransformValue result;
360 result.x = *x;
361 result.y = *y;
362 result.z = *z;
363 auto optionalNumber = [&](const char* name, double fallback) {
364 const EditorValue* entry = field(value, name);
365 if (!entry) return fallback;
366 const auto* number = entry->getIf<double>();
367 return number ? *number : fallback;
368 };
369 result.rotationX = optionalNumber("rotationX", 0.0);
370 result.rotationY = optionalNumber("rotationY", 0.0);
371 result.rotationZ = optionalNumber("rotationZ", 0.0);
372 result.scaleX = optionalNumber("scaleX", 1.0);
373 result.scaleY = optionalNumber("scaleY", 1.0);
374 result.scaleZ = optionalNumber("scaleZ", 1.0);
375 return eve::editing::applied<SceneTransformValue>(result);
376}

484PropertyReadResult ScenePropertyProvider::read(const SelectionSnapshot& selection,
485 const PropertyPath& path) const {
486 if (!target_ || selection.items.empty()) return {};
487 auto component = [&](const SceneTransformValue& value) -> EditorValue {
488 if (path == PropertyPath("transform.position")) return EditorValue::Array{value.x, value.y, value.z};
489 if (path == PropertyPath("transform.rotation"))
490 return EditorValue::Array{value.rotationX, value.rotationY, value.rotationZ};
491 if (path == PropertyPath("transform.scale"))
492 return EditorValue::Array{value.scaleX, value.scaleY, value.scaleZ};
493 return {};
494 };
495 std::optional<EditorValue> common;
496 for (const SelectionItem& item : selection.items) {
497 if (item.target != TargetId(target_->targetId())) return {};
498 auto transform = target_->readTransform(ObjectId(item.item.value()));
499 if (!transform.ok()) return {PropertyReadState::Error, {}, transform.diagnostics()};
500 EditorValue value = component(transform.value());
501 if (value.type() == EditorValue::Type::Null) return {};
502 if (!common) common = value;
503 else if (*common != value) return {PropertyReadState::Mixed, {}, {}};
504 }
505 return common ? PropertyReadResult{PropertyReadState::Value, *common, {}} : PropertyReadResult{};
506}

510 const EditorValue& value,
511 PropertySetMode mode) const {
512 if (!target_ || mode != PropertySetMode::Absolute)
513 return sceneError<DomainOperation>(EditorStatus::Unsupported, "editor.scene.property-mode",
514 "Scene properties currently require absolute assignment");
515 const auto* tuple = value.getIf<EditorValue::Array>();
516 if (!tuple || tuple->size() != 3)
517 return sceneError<DomainOperation>(EditorStatus::Rejected, "editor.scene.property-value",
518 "Scene TRS property requires three numeric values");
519 const auto* a = (*tuple)[0].getIf<double>();
520 const auto* b = (*tuple)[1].getIf<double>();
521 const auto* c = (*tuple)[2].getIf<double>();
522 if (!a || !b || !c)
523 return sceneError<DomainOperation>(EditorStatus::Rejected, "editor.scene.property-value",
524 "Scene TRS property requires three numeric values");
525 EditorValue::Array payload;
526 EditorValue::Array inverse;
527 DomainOperation operation;
528 for (const SelectionItem& item : selection.items) {
529 if (item.target != TargetId(target_->targetId()))
530 return sceneError<DomainOperation>(EditorStatus::Rejected, "editor.scene.property-target",
531 "Selection contains objects from another target");
532 auto current = target_->readTransform(ObjectId(item.item.value()));
533 if (!current.ok())
534 return sceneError<DomainOperation>(EditorStatus::NotFound, "editor.scene.property-object",
535 "Selected scene object does not exist");
536 SceneTransformValue changed = current.value();
537 if (path == PropertyPath("transform.position")) {
538 changed.x = *a; changed.y = *b; changed.z = *c;
539 } else if (path == PropertyPath("transform.rotation")) {
540 changed.rotationX = *a; changed.rotationY = *b; changed.rotationZ = *c;
541 } else if (path == PropertyPath("transform.scale")) {
542 changed.scaleX = *a; changed.scaleY = *b; changed.scaleZ = *c;
543 } else {
544 return sceneError<DomainOperation>(EditorStatus::Unsupported, "editor.scene.property-path",
545 "Scene property path is unsupported: " + path.value());
546 }
547 EditorValue::Object nextEntry;
548 nextEntry["id"] = item.item.value();
549 nextEntry["transform"] = SceneTargetBase::transformValue(changed);
550 payload.emplace_back(std::move(nextEntry));
551 EditorValue::Object previousEntry;
552 previousEntry["id"] = item.item.value();
553 previousEntry["transform"] = SceneTargetBase::transformValue(current.value());
554 inverse.emplace_back(std::move(previousEntry));
555 operation.affectedObjects.push_back({TargetId(target_->targetId()), item.item.value(), 0});
556 }
557 operation.type = "scene.transform.multi.set.v1";
558 operation.target = TargetId(target_->targetId());
559 operation.payload = EditorValue(std::move(payload));
560 operation.inverse = EditorValue(std::move(inverse));
561 operation.hasInverse = true;
562 operation.affectedProperties.push_back(path.value());
563 operation.mergeKey = "scene.selection:" + path.value();
564 return eve::editing::applied<DomainOperation>(std::move(operation));
565}
