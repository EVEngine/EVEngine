18Result<int> integerParameter(const Value::Object& parameters, std::string_view key) {
19 const auto found = parameters.find(std::string(key));
20 if (found == parameters.end() || !found->second.isInt64())
21 return failure<int>(DiagnosticCode::InvalidArgument, "tactics move parameter must be an integer",
22 "request.parameters." + std::string(key));
23 const std::int64_t value = found->second.asInt();
24 if (value < std::numeric_limits<int>::min() || value > std::numeric_limits<int>::max())
25 return failure<int>(DiagnosticCode::InvalidArgument, "tactics move parameter is outside int range",
26 "request.parameters." + std::string(key));
27 return Result<int>::success(static_cast<int>(value));
28}

30Result<Revision> revisionParameter(const Value::Object& parameters) {
31 const auto found = parameters.find("expectedRevision");
32 const auto* text = found == parameters.end() ? nullptr : found->second.getIf<std::string>();
33 if (!text)
34 return failure<Revision>(DiagnosticCode::InvalidArgument,
35 "tactics move requires an expected revision decimal string",
36 "request.parameters.expectedRevision");
37 std::uint64_t value = 0;
38 const auto [end, error] = std::from_chars(text->data(), text->data() + text->size(), value);
39 if (error != std::errc{} || end != text->data() + text->size())
40 return failure<Revision>(DiagnosticCode::InvalidArgument, "tactics expected revision is invalid",
41 "request.parameters.expectedRevision");
42 return Result<Revision>::success(Revision(value));
43}

128Result<std::unique_ptr<action::IActionEffectOperation>> TacticsActionExecutor::prepare(
129 const action::ActionDefinition& definition, const action::ActionRequest& request, const sensing::TargetSet* targets,
130 SimulationTick tick) {
131 (void)targets;
132 const auto moveId = LogicalId::parse("tactics:move");
133 const auto faceId = LogicalId::parse("tactics:face");
134 const auto waitId = LogicalId::parse("tactics:wait");
135 if (!moveId || !faceId || !waitId || definition.id != request.actionId ||
136 (definition.id != *moveId && definition.id != *faceId && definition.id != *waitId))
137 return failure<std::unique_ptr<action::IActionEffectOperation>>(
138 DiagnosticCode::InvalidArgument, "tactics executor received an unsupported action definition",
139 "definition.id");
140 if (!request.source)
141 return failure<std::unique_ptr<action::IActionEffectOperation>>(
142 DiagnosticCode::InvalidArgument, "tactics action request requires a source unit", "request.source");
143 auto* unit = dynamic_cast<TacticalUnit*>(ecs::try_get(*request.source));
144 if (unit == nullptr)
145 return failure<std::unique_ptr<action::IActionEffectOperation>>(
146 DiagnosticCode::StaleHandle, "tactics action source is stale or has the wrong type", "request.source");
147 auto expectedRevision = revisionParameter(request.parameters);
148 if (!expectedRevision)
149 return Result<std::unique_ptr<action::IActionEffectOperation>>::failure(expectedRevision.status());
150 if (battle_.turn()->revision != expectedRevision.value())
151 return failure<std::unique_ptr<action::IActionEffectOperation>>(
152 DiagnosticCode::Conflict, "tactics action request was prepared against an older battle revision",
153 "request.parameters.expectedRevision");
154
155 if (definition.id == *moveId) {
156 auto x = integerParameter(request.parameters, "x");
157 auto y = integerParameter(request.parameters, "y");
158 auto layer = integerParameter(request.parameters, "layer");
159 if (!x) return Result<std::unique_ptr<action::IActionEffectOperation>>::failure(x.status());
160 if (!y) return Result<std::unique_ptr<action::IActionEffectOperation>>::failure(y.status());
161 if (!layer) return Result<std::unique_ptr<action::IActionEffectOperation>>::failure(layer.status());
162 const Cell destination{x.value(), y.value(), layer.value()};
163 auto preview = BattleSystem::previewMove(battle_, unit->identity()->subject, destination);
164 if (!preview) return Result<std::unique_ptr<action::IActionEffectOperation>>::failure(preview.status());
165 std::move(preview).takeValue();
166 return Result<std::unique_ptr<action::IActionEffectOperation>>::success(
167 std::make_unique<PreparedMoveOperation>(battle_, unit->identity()->subject, destination, tick),
168 Status::success(StatusCode::Applied));
169 }
170 if (definition.id == *faceId) {
171 auto facing = integerParameter(request.parameters, "facing");
172 if (!facing) return Result<std::unique_ptr<action::IActionEffectOperation>>::failure(facing.status());
173 auto preview = BattleSystem::previewFace(battle_, unit->identity()->subject, facing.value());
174 if (!preview) return Result<std::unique_ptr<action::IActionEffectOperation>>::failure(preview.status());
175 return Result<std::unique_ptr<action::IActionEffectOperation>>::success(
176 std::make_unique<PreparedFaceOperation>(battle_, unit->identity()->subject, facing.value(), tick),
177 Status::success(StatusCode::Applied));
178 }
179 auto preview = BattleSystem::previewWait(battle_, unit->identity()->subject);
180 if (!preview) return Result<std::unique_ptr<action::IActionEffectOperation>>::failure(preview.status());
181 return Result<std::unique_ptr<action::IActionEffectOperation>>::success(
182 std::make_unique<PreparedWaitOperation>(battle_, unit->identity()->subject, tick),
183 Status::success(StatusCode::Applied));
184}

207Result<action::ActionRequest> makeMoveRequest(ecs::EntityHandle unit, Cell destination, SimulationTick tick) {
208 auto owners = requestOwners(unit);

210 action::ActionRequest request = baseRequest(*LogicalId::parse("tactics:move"), unit, *owners.value().second, tick);
211 request.parameters.emplace("x", Value(destination.x));
212 request.parameters.emplace("y", Value(destination.y));
213 request.parameters.emplace("layer", Value(destination.layer));
