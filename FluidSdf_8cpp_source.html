12float triangleDistance(const glm::vec3& p, const glm::vec3& a, const glm::vec3& b, const glm::vec3& c) {
13 const glm::vec3 ab = b - a;
14 const glm::vec3 ac = c - a;
15 const glm::vec3 ap = p - a;
16 const float d1 = glm::dot(ab, ap);
17 const float d2 = glm::dot(ac, ap);
18 if (d1 <= 0.f && d2 <= 0.f) return glm::length(p - a);
19
20 const glm::vec3 bp = p - b;
21 const float d3 = glm::dot(ab, bp);
22 const float d4 = glm::dot(ac, bp);
23 if (d3 >= 0.f && d4 <= d3) return glm::length(p - b);
24
25 const float vc = d1 * d4 - d3 * d2;
26 if (vc <= 0.f && d1 >= 0.f && d3 <= 0.f) {
27 const float t = d1 / (d1 - d3);
28 return glm::length(p - (a + ab * t));
29 }
30
31 const glm::vec3 cp = p - c;
32 const float d5 = glm::dot(ab, cp);
33 const float d6 = glm::dot(ac, cp);
34 if (d6 >= 0.f && d5 <= d6) return glm::length(p - c);
35
36 const float vb = d5 * d2 - d1 * d6;
37 if (vb <= 0.f && d2 >= 0.f && d6 <= 0.f) {
38 const float t = d2 / (d2 - d6);
39 return glm::length(p - (a + ac * t));
40 }
41
42 const float va = d3 * d6 - d5 * d4;
43 if (va <= 0.f && d4 - d3 >= 0.f && d5 - d6 >= 0.f) {
44 const float t2 = (d4 - d3) / ((d4 - d3) + (d5 - d6));
45 return glm::length(p - (b + (c - b) * t2));
46 }
47
48 const float denom = 1.f / (va + vb + vc);
49 const float v = vb * denom;
50 const float w = vc * denom;
51 return glm::length(p - (a + ab * v + ac * w));
52}

55bool pointInsideMesh(const glm::vec3& p, const std::vector<glm::vec3>& pos, const std::vector<uint32_t>& idx,
56 int triCount) {
57 int hits = 0;
58 for (int tri = 0; tri < triCount; ++tri) {
59 const glm::vec3 a = pos[idx[uint32_t(tri) * 3 + 0]];
60 const glm::vec3 b = pos[idx[uint32_t(tri) * 3 + 1]];
61 const glm::vec3 c = pos[idx[uint32_t(tri) * 3 + 2]];
62 // Moller-Trumbore ray from p along +X.
63 const glm::vec3 e1 = b - a;
64 const glm::vec3 e2 = c - a;
65 const glm::vec3 h = glm::cross(glm::vec3(1.f, 0.f, 0.f), e2);
66 const float det = glm::dot(e1, h);
67 if (std::fabs(det) < 1e-12f) continue;
68 const float invDet = 1.f / det;
69 const glm::vec3 s = p - a;
70 const float u = invDet * glm::dot(s, h);
71 if (u < 0.f || u > 1.f) continue;
72 const glm::vec3 q = glm::cross(s, e1);
73 const float v = invDet * glm::dot(glm::vec3(1.f, 0.f, 0.f), q);
74 if (v < 0.f || u + v > 1.f) continue;
75 const float t = invDet * glm::dot(e2, q);
76 if (t > 1e-8f) ++hits;
77 }
78 return (hits & 1) != 0;
79}

91float MeshSdf::sample(const glm::vec3& p) const {
92 if (voxelCount() <= 0) return FLT_MAX;
93 const glm::vec3 f = (p - origin) / cellSize;
94 const glm::ivec3 i0 = glm::clamp(glm::ivec3(glm::floor(f)), glm::ivec3(0), dims - glm::ivec3(1));
95 const glm::ivec3 i1 = glm::min(i0 + glm::ivec3(1), dims - glm::ivec3(1));
96 const glm::vec3 t = glm::fract(f);
97
98 const float v000 = distances[index(i0.x, i0.y, i0.z)];
99 const float v100 = distances[index(i1.x, i0.y, i0.z)];
100 const float v010 = distances[index(i0.x, i1.y, i0.z)];
101 const float v110 = distances[index(i1.x, i1.y, i0.z)];
102 const float v001 = distances[index(i0.x, i0.y, i1.z)];
103 const float v101 = distances[index(i1.x, i0.y, i1.z)];
104 const float v011 = distances[index(i0.x, i1.y, i1.z)];
105 const float v111 = distances[index(i1.x, i1.y, i1.z)];
106
107 const float c00 = v000 + (v100 - v000) * t.x;
108 const float c10 = v010 + (v110 - v010) * t.x;
109 const float c01 = v001 + (v101 - v001) * t.x;
110 const float c11 = v011 + (v111 - v011) * t.x;
111 const float c0 = c00 + (c10 - c00) * t.y;
112 const float c1 = c01 + (c11 - c01) * t.y;
113 return c0 + (c1 - c0) * t.z;
114}

116glm::vec3 MeshSdf::gradient(const glm::vec3& p) const {
117 const float e = cellSize;
118 return glm::vec3(sample(p + glm::vec3(e, 0.f, 0.f)) - sample(p - glm::vec3(e, 0.f, 0.f)),
119 sample(p + glm::vec3(0.f, e, 0.f)) - sample(p - glm::vec3(0.f, e, 0.f)),

121 (2.f * e);

160MeshSdf MeshSdf::makeFromTriangles(const std::vector<glm::vec3>& positions, const std::vector<uint32_t>& indices,
161 const glm::ivec3& dims) {
162 const int triCount = int(indices.size()) / 3;
163 glm::vec3 minP(FLT_MAX);
164 glm::vec3 maxP(-FLT_MAX);
165 for (const glm::vec3& v : positions) {
166 minP = glm::min(minP, v);
167 maxP = glm::max(maxP, v);
168 }
169 const glm::vec3 extent = maxP - minP;
170 const glm::vec3 pad = extent * 0.25f + glm::vec3(1e-3f);
171 const glm::vec3 total = extent + pad * 2.f;
172 const float cell = std::max({total.x / float(dims.x), total.y / float(dims.y), total.z / float(dims.z)});
173
174 MeshSdf sdf;
175 sdf.dims = dims;
176 sdf.cellSize = cell;
177 sdf.origin = minP - pad;
178 sdf.distances.assign(size_t(sdf.voxelCount()), FLT_MAX);
179
180 // Sweep each triangle over its expanded AABB and keep the min distance.
181 for (int t = 0; t < triCount; ++t) {
182 const glm::vec3 a = positions[indices[uint32_t(t) * 3 + 0]];
183 const glm::vec3 b = positions[indices[uint32_t(t) * 3 + 1]];
184 const glm::vec3 c = positions[indices[uint32_t(t) * 3 + 2]];
185 const glm::vec3 tmin = glm::min(a, glm::min(b, c));
186 const glm::vec3 tmax = glm::max(a, glm::max(b, c));
187 const glm::ivec3 c0 = glm::clamp(glm::ivec3(glm::floor((tmin - sdf.origin) / cell)) - glm::ivec3(1),
188 glm::ivec3(0), dims - glm::ivec3(1));
189 const glm::ivec3 c1 = glm::clamp(glm::ivec3(glm::floor((tmax - sdf.origin) / cell)) + glm::ivec3(1),
190 glm::ivec3(0), dims - glm::ivec3(1));
191 for (int z = c0.z; z <= c1.z; ++z) {
192 for (int y = c0.y; y <= c1.y; ++y) {
193 for (int x = c0.x; x <= c1.x; ++x) {
194 const glm::vec3 p = sdf.origin + glm::vec3(float(x), float(y), float(z)) * cell;
195 const float d = triangleDistance(p, a, b, c);
196 float& slot = sdf.distances[size_t(sdf.index(x, y, z))];
197 slot = std::min(slot, d);
198 }
199 }
200 }
201 }
202
203 // Sign from even-odd raycast per voxel center.
204 for (int z = 0; z < dims.z; ++z) {
205 for (int y = 0; y < dims.y; ++y) {
206 for (int x = 0; x < dims.x; ++x) {
207 const glm::vec3 p = sdf.origin + glm::vec3(float(x), float(y), float(z)) * cell;
208 const size_t i = size_t(sdf.index(x, y, z));
209 if (sdf.distances[i] >= FLT_MAX) {
210 // No triangle in range: sign still matters far outside.
211 sdf.distances[i] = pointInsideMesh(p, positions, indices, triCount) ? -cell : cell;
212 } else if (pointInsideMesh(p, positions, indices, triCount)) {
213 sdf.distances[i] = -sdf.distances[i];
214 }
215 }
216 }
217 }
218 return sdf;
219}
