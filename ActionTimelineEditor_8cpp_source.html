37void sortTrack(action::ActionTrack& track) {
38 std::stable_sort(track.notifies.begin(), track.notifies.end(), [](const auto& left, const auto& right) {
39 return left.time == right.time ? left.id.format() < right.id.format() : left.time < right.time;
40 });
41 std::stable_sort(track.states.begin(), track.states.end(), [](const auto& left, const auto& right) {
42 return left.start == right.start ? left.id.format() < right.id.format() : left.start < right.start;

170EditorResult<DomainOperation> ActionTimelineTarget::replacement(const action::ActionTimeline& candidate,
171 std::string property) const {
172 auto beforeValue = timeline_.toValue();
173 auto afterValue = candidate.toValue();
174 if (!beforeValue || !afterValue)
175 return editorError<DomainOperation>(EditorStatus::Rejected, "editor.action.timeline.encode-failed",
176 "Could not encode the action timeline edit");
177 auto beforeJson = beforeValue.value().toJson();
178 auto afterJson = afterValue.value().toJson();
179 if (!beforeJson || !afterJson)
180 return editorError<DomainOperation>(EditorStatus::Rejected, "editor.action.timeline.encode-failed",
181 "Could not serialize the action timeline edit");
182 DomainOperation operation;
183 operation.type = std::string(kReplaceOperation);
184 operation.inverseType = std::string(kReplaceOperation);
185 operation.target = TargetId(targetId_);
186 operation.payload = replacementPayload(afterJson.value());
187 operation.inverse = replacementPayload(beforeJson.value());
188 operation.hasInverse = true;
189 operation.affectedObjects.push_back({operation.target, timeline_.actionId.format(), revision_});
190 if (!property.empty()) operation.affectedProperties.push_back(std::move(property));
191 return eve::editing::applied<DomainOperation>(std::move(operation));
192}

225EditorResult<DomainOperation> ActionTimelineTarget::makeSet(const SelectionSnapshot& selection,
226 const PropertyPath& path, const EditorValue& value,
227 PropertySetMode mode) const {
228 if (mode == PropertySetMode::Reset) return makeReset(selection, path);
229 const auto descriptor = schema(selection).find(path);
230 if (!matches(selection) || !descriptor || mode != PropertySetMode::Absolute ||
231 !validatePropertyValue(*descriptor, value).ok())
232 return editorError<DomainOperation>(EditorStatus::Rejected, "editor.action.timeline.property-invalid",
233 "Action timeline property assignment is invalid");
234 action::ActionTimeline candidate = timeline_;
235 if (path == PropertyPath("timeline.actionId")) {
236 const auto* text = value.getIf<std::string>();
237 auto parsed = text ? LogicalId::parse(*text) : std::optional<LogicalId>{};
238 if (!parsed)
239 return editorError<DomainOperation>(EditorStatus::Rejected, "editor.action.timeline.action-id",
240 "Action id must be a namespace:name logical id");
241 candidate.actionId = std::move(*parsed);
242 } else if (path == PropertyPath("timeline.durationSeconds")) {
243 const auto* seconds = value.getIf<double>();
244 if (!seconds)
245 return editorError<DomainOperation>(EditorStatus::Rejected, "editor.action.timeline.duration",
246 "Timeline duration must be a number of seconds");
247 auto duration = Duration::fromSeconds(*seconds);
248 if (!duration)
249 return editorError<DomainOperation>(EditorStatus::Rejected, "editor.action.timeline.duration",
250 commonMessage(duration.status(), "Timeline duration is invalid"));
251 candidate.duration = std::move(duration).takeValue();
252 } else {
253 const auto* uri = value.getIf<std::string>();
254 if (!uri)
255 return editorError<DomainOperation>(EditorStatus::Rejected, "editor.action.timeline.animation-uri",
256 "Animation URI must be a string");
257 candidate.animationUri = *uri;
258 }
259 auto valid = candidate.validate();
260 if (!valid)
261 return editorError<DomainOperation>(EditorStatus::Rejected, "editor.action.timeline.asset-invalid",
262 commonMessage(valid.status(), "Action timeline property edit is invalid"));
263 return replacement(candidate, path.value());
264}

336EditorResult<void> ActionTimelineEditor::commit(action::ActionTimeline candidate, std::string label,
337 std::string mergeKey) {
338 auto valid = candidate.validate();
339 if (!valid)
340 return rejected("editor.action.timeline.candidate-invalid",
341 commonMessage(valid.status(), "Action timeline edit is invalid"));
342 auto beforeValue = target_.timeline().toValue();
343 auto afterValue = candidate.toValue();
344 if (!beforeValue || !afterValue)
345 return rejected("editor.action.timeline.encode-failed", "Could not encode the action timeline edit");
346 auto beforeJson = beforeValue.value().toJson();
347 auto afterJson = afterValue.value().toJson();
348 if (!beforeJson || !afterJson)
349 return rejected("editor.action.timeline.encode-failed", "Could not serialize the action timeline edit");
350 if (beforeJson.value() == afterJson.value()) {
351 return eve::editing::noOp();
352 }
353
354 const std::string operationMergeKey = mergeKey;
355 TransactionSpec transaction;
356 transaction.id = TransactionId("action.timeline.tx." + std::to_string(++transactionSequence_));
357 transaction.label = std::move(label);
358 transaction.target = TargetId(target_.targetId());
359 transaction.baseRevision = target_.revision();
360 transaction.mergeKey = std::move(mergeKey);
361 auto begun = transactions_.begin(std::move(transaction));
362 if (!begun.ok())
363 return editorError(begun.code(), "editor.action.timeline.begin-failed", "Could not begin action edit");
364
365 DomainOperation operation;
366 operation.type = std::string(kReplaceOperation);
367 operation.inverseType = std::string(kReplaceOperation);
368 operation.target = TargetId(target_.targetId());
369 operation.payload = replacementPayload(afterJson.value());
370 operation.inverse = replacementPayload(beforeJson.value());
371 operation.hasInverse = true;
372 operation.affectedObjects.push_back({operation.target, target_.timeline().actionId.format(), target_.revision()});
373 operation.affectedProperties.push_back("timeline");
374 operation.mergeKey = operationMergeKey;
375 auto appended = transactions_.append(std::move(operation));
376 if (!appended.ok()) {
377 auto discarded = transactions_.discard();
378 (void)discarded;
379 return appended;
380 }
381 auto committed = transactions_.commit();
382 if (!committed.ok())
383 return editorError(committed.code(), "editor.action.timeline.commit-failed", "Action edit was rejected");
384 return eve::editing::applied<void>();
385}

413EditorResult<void> ActionTimelineEditor::moveItem(const LogicalId& itemId, Duration delta) {
414 action::ActionTimeline candidate = target_.timeline();
415 for (auto& track : candidate.tracks) {
416 for (auto& notify : track.notifies) {
417 if (notify.id != itemId) continue;
418 if (track.locked) return rejected("editor.action.timeline.track-locked", "Action track is locked");
419 auto moved = notify.time.tryAdd(delta);
420 if (!moved) return rejected("editor.action.timeline.time-overflow", "Notify time overflowed");
421 notify.time = std::move(moved).takeValue();
422 sortTrack(track);
423 return commit(std::move(candidate), "Move action notify", "action.timeline.item.move");
424 }
425 for (auto& state : track.states) {
426 if (state.id != itemId) continue;
427 if (track.locked) return rejected("editor.action.timeline.track-locked", "Action track is locked");
428 auto start = state.start.tryAdd(delta);
429 auto end = state.end.tryAdd(delta);
430 if (!start || !end) return rejected("editor.action.timeline.time-overflow", "Notify-state time overflowed");
431 state.start = std::move(start).takeValue();
432 state.end = std::move(end).takeValue();
433 sortTrack(track);
434 return commit(std::move(candidate), "Move action notify state", "action.timeline.item.move");
435 }
436 }
437 return rejected("editor.action.timeline.item-not-found", "Timeline item was not found");
438}

514 const auto notify = std::find_if(track.notifies.begin(), track.notifies.end(),

516 if (notify != track.notifies.end()) {
517 if (track.locked) return rejected("editor.action.timeline.track-locked", "Action track is locked");
518 track.notifies.erase(notify);
519 selection_.erase(itemId.format());
520 return commit(std::move(candidate), "Delete action notify", "action.timeline.item.delete");
521 }
522 const auto state = std::find_if(track.states.begin(), track.states.end(),
523 [&](const auto& value) { return value.id == itemId; });
524 if (state != track.states.end()) {
525 if (track.locked) return rejected("editor.action.timeline.track-locked", "Action track is locked");
526 track.states.erase(state);
527 selection_.erase(itemId.format());
528 return commit(std::move(candidate), "Delete action notify state", "action.timeline.item.delete");
529 }
530 }
531 return rejected("editor.action.timeline.item-not-found", "Timeline item was not found");

599EditorResult<std::size_t> ActionTimelineEditor::paste(Duration offset) {
600 if (clipboard_.empty())
601 return eve::editing::failed<std::size_t>(EditorStatus::Rejected,
602 RuleId("editor.action.timeline.clipboard-empty"),
603 "Action timeline clipboard is empty");
604 action::ActionTimeline candidate = target_.timeline();
605 std::set<std::string> pastedSelection;
606 for (const auto& item : clipboard_) {
607 auto* track = findTrack(candidate, item.trackId);
608 if (!track)
609 return eve::editing::failed<std::size_t>(EditorStatus::Conflict,
610 RuleId("editor.action.timeline.track-not-found"),
611 "Clipboard track no longer exists");
612 if (track->locked)
613 return eve::editing::failed<std::size_t>(
614 EditorStatus::Rejected, RuleId("editor.action.timeline.track-locked"), "Clipboard track is locked");
615 if (item.state) {
616 auto copy = item.notifyState;
617 copy.id = copiedId(copy.id);
618 auto start = copy.start.tryAdd(offset);
619 auto end = copy.end.tryAdd(offset);
620 if (!start || !end)
621 return eve::editing::failed<std::size_t>(EditorStatus::Rejected,
622 RuleId("editor.action.timeline.time-overflow"),
623 "Pasted notify-state time overflowed");
624 copy.start = std::move(start).takeValue();
625 copy.end = std::move(end).takeValue();
626 pastedSelection.insert(copy.id.format());
627 track->states.push_back(std::move(copy));
628 } else {
629 auto copy = item.notify;
630 copy.id = copiedId(copy.id);
631 auto time = copy.time.tryAdd(offset);
632 if (!time)
633 return eve::editing::failed<std::size_t>(EditorStatus::Rejected,
634 RuleId("editor.action.timeline.time-overflow"),
635 "Pasted notify time overflowed");
636 copy.time = std::move(time).takeValue();
637 pastedSelection.insert(copy.id.format());
638 track->notifies.push_back(std::move(copy));
639 }
640 sortTrack(*track);
641 }
642 const std::size_t count = pastedSelection.size();
643 auto committed = commit(std::move(candidate), "Paste action timeline items", "action.timeline.item.paste");
644 if (!committed.ok())
645 return eve::editing::failed<std::size_t>(committed.code(), RuleId("editor.action.timeline.paste-failed"),
646 "Could not paste action timeline items");
647 selection_ = std::move(pastedSelection);
648 return eve::editing::applied<std::size_t>(count);
649}

651EditorResult<void> ActionTimelineEditor::deleteSelection() {
652 if (selection_.empty()) return rejected("editor.action.timeline.selection-empty", "No timeline items are selected");
653 action::ActionTimeline candidate = target_.timeline();
654 for (auto& track : candidate.tracks) {
655 const bool touchesLocked =
656 track.locked && (std::any_of(track.notifies.begin(), track.notifies.end(),
657 [&](const auto& item) { return selection_.contains(item.id.format()); }) ||
658 std::any_of(track.states.begin(), track.states.end(),
659 [&](const auto& item) { return selection_.contains(item.id.format()); }));
660 if (touchesLocked)
661 return rejected("editor.action.timeline.track-locked", "Selection contains an item on a locked track");
662 std::erase_if(track.notifies, [&](const auto& item) { return selection_.contains(item.id.format()); });
663 std::erase_if(track.states, [&](const auto& item) { return selection_.contains(item.id.format()); });
664 }
665 auto result = commit(std::move(candidate), "Delete action timeline selection", "action.timeline.item.delete");
666 if (result.ok()) selection_.clear();
667 return result;
668}

683EditorResult<std::size_t> ActionTimelineEditor::update(Duration delta) {
684 auto plan = planPreview(delta);
685 if (!plan.ok())
686 return eve::editing::failed<std::size_t>(plan.code(), RuleId("editor.action.timeline.preview-plan"),
687 "Could not prepare preview advance");
688 if (plan.code() == EditorStatus::NoOp) {
689 previewEvents_.clear();
690 return EditorResult<std::size_t>::success(0, Status::success(EditorStatus::NoOp));
691 }
692 return applyPreviewPlan(std::move(plan).takeValue());

694
695EditorResult<ActionTimelinePreviewPlan> ActionTimelineEditor::planPreview(Duration delta) const {
696 if (!playing_) {
697 ActionTimelinePreviewPlan plan;
698 plan.timelineRevision = target_.revision();
699 plan.previous = previewTime_;
700 plan.current = previewTime_;
701 return EditorResult<ActionTimelinePreviewPlan>::success(
702 std::move(plan), Status::success(EditorStatus::NoOp));
703 }
704 if (delta < Duration::zero())
705 return eve::editing::failed<ActionTimelinePreviewPlan>(EditorStatus::Rejected,
706 RuleId("editor.action.timeline.preview-delta"),
707 "Preview delta must be non-negative");
708 auto next = previewTime_.tryAdd(delta);
709 if (!next)
710 return eve::editing::failed<ActionTimelinePreviewPlan>(
711 EditorStatus::Rejected, RuleId("editor.action.timeline.preview-overflow"), "Preview time overflowed");
712 Duration current = std::move(next).takeValue();
713 if (current > target_.timeline().duration) current = target_.timeline().duration;
714 auto sampled = target_.timeline().sample(previewTime_, current, !previewStarted_);
715 if (!sampled)
716 return eve::editing::failed<ActionTimelinePreviewPlan>(
717 EditorStatus::Rejected, RuleId("editor.action.timeline.preview-invalid"),
718 commonMessage(sampled.status(), "Could not sample action timeline"));
719 ActionTimelinePreviewPlan plan;
720 plan.timelineRevision = target_.revision();
721 plan.previous = previewTime_;
722 plan.current = current;
723 plan.includeStart = !previewStarted_;
724 plan.reachesEnd = current == target_.timeline().duration;
725 plan.events = std::move(sampled).takeValue();
726 return eve::editing::applied<ActionTimelinePreviewPlan>(std::move(plan));
727}
728
729EditorResult<std::size_t> ActionTimelineEditor::applyPreviewPlan(ActionTimelinePreviewPlan plan) {
730 if (!playing_ || plan.timelineRevision != target_.revision() || plan.previous != previewTime_ ||
731 plan.includeStart == previewStarted_)
732 return eve::editing::failed<std::size_t>(EditorStatus::Conflict,
733 RuleId("editor.action.timeline.preview-plan-stale"),
734 "Prepared preview advance no longer matches transport state");
735 auto expected = target_.timeline().sample(plan.previous, plan.current, plan.includeStart);
736 if (!expected || expected.value() != plan.events ||
737 plan.reachesEnd != (plan.current == target_.timeline().duration))
738 return eve::editing::failed<std::size_t>(EditorStatus::Rejected,
739 RuleId("editor.action.timeline.preview-plan-invalid"),
740 "Prepared preview advance does not match the authoritative timeline");
741 previewEvents_ = std::move(plan.events);
742 previewTime_ = plan.current;
743 previewStarted_ = true;
744 if (plan.reachesEnd) playing_ = false;
745 return eve::editing::applied<std::size_t>(previewEvents_.size());
746}
