40 Status(EditorStatus::Rejected, diagnostics));

43 "VirtualGeometry build requires a source asset");

48 const bool normalsInvalid = !mesh.normals.empty() && mesh.normals.size() != mesh.positions.size();
49 if (mesh.positions.empty() || mesh.positions.size() % 3 != 0 || mesh.indices.empty() ||
50 mesh.indices.size() % 3 != 0 || mesh.positions.size() / 3 > 50000000 || mesh.indices.size() > 150000000 ||
51 normalsInvalid)
52 return fail<VirtualGeometryBuildArtifact>(EditorStatus::Rejected, "editor.virtualgeometry.mesh",
53 "VirtualGeometry mesh shape or budget is invalid");
54 const std::size_t vertices = mesh.positions.size() / 3;
55 for (float value : mesh.positions)
56 if (!std::isfinite(value))
57 return fail<VirtualGeometryBuildArtifact>(EditorStatus::Rejected, "editor.virtualgeometry.position",
58 "VirtualGeometry positions must be finite");
59 for (std::uint32_t index : mesh.indices)
60 if (index >= vertices)
61 return fail<VirtualGeometryBuildArtifact>(EditorStatus::Rejected, "editor.virtualgeometry.index",
62 "VirtualGeometry index is out of range");
63 auto candidate = std::make_unique<virtualgeometry::VirtualGeometryAsset>();
64 virtualgeometry::VirtualGeometryBuilder builder;
65 virtualgeometry::VirtualGeometryBuilder::MeshInput input;
66 input.vertexCount = static_cast<int>(vertices);
67 input.positions = mesh.positions.data();
68 input.normals = mesh.normals.empty() ? nullptr : mesh.normals.data();
69 input.indices = mesh.indices.data();
70 input.indexCount = static_cast<int>(mesh.indices.size());
71 if (!builder.build(input, document.value().builder, *candidate))
72 return fail<VirtualGeometryBuildArtifact>(EditorStatus::Failed, "editor.virtualgeometry.build",
73 "VirtualGeometry builder rejected the candidate mesh");

75 artifact.sourceRevision = document.revision();
76 artifact.vertices = candidate->vertexCount;
77 artifact.sourceTriangles = mesh.indices.size() / 3;
78 artifact.storedTriangles = candidate->triangles.size() / 3;
79 artifact.clusters = static_cast<int>(candidate->clusters.size());
80 artifact.checksum = checksum(*candidate);
81 for (const auto& cluster : candidate->clusters) {
82 artifact.maxLod = std::max(artifact.maxLod, static_cast<int>(cluster.lodLevel));
83 if (cluster.parent == 0xFFFFFFFFu) ++artifact.roots;
84 if (cluster.childCount > 4 || cluster.triStart + cluster.triCount > candidate->triangles.size() / 3)
85 return fail<VirtualGeometryBuildArtifact>(EditorStatus::Failed, "editor.virtualgeometry.dag",
86 "VirtualGeometry builder produced an invalid cluster range");
87 for (std::uint32_t i = 0; i < cluster.childCount; ++i)
88 if (cluster.children[i] >= candidate->clusters.size())
89 return fail<VirtualGeometryBuildArtifact>(
90 EditorStatus::Failed, "editor.virtualgeometry.dag",
91 "VirtualGeometry builder produced an invalid child reference");
92 }

107 }
