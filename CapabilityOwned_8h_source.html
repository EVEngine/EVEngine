134 [[nodiscard]] static Result<OwnedProviderHandle> provide(std::string id, std::shared_ptr<I> provider) {
135 if (id.empty() || !provider) {
136 return Result<OwnedProviderHandle>::failure(Diagnostic::error(
137 DiagnosticCode::InvalidArgument, "owned capability requires id and implementation", "id"));
138 }
139 std::scoped_lock lock(detail::ownedMutex());
140 auto& slots = detail::ownedSlots(I::capabilityName);
141 if (slots.contains(id)) {
142 return Result<OwnedProviderHandle>::failure(Diagnostic::error(
143 DiagnosticCode::Conflict, "owned capability id is already published", "id"));
144 }
145 auto& generations = detail::ownedGenerations(I::capabilityName);
146 OwnedProviderHandle handle{id, ++generations[id]};
147 auto [it, inserted] =
148 slots.emplace(std::move(id), detail::OwnedSlot{handle, std::make_unique<detail::OwnedEntry<I>>(
149 std::move(provider))});
150 (void)inserted;
151 // Also expose the current generation through the borrowed single-provider slot so
152 // existing `query` / `ProviderRef` consumers see the same instance.
153 detail::provideRaw(I::capabilityName, it->second.entry->raw());
154 return Result<OwnedProviderHandle>::success(std::move(handle), Status::success(StatusCode::Applied));
155 }
156
161 [[nodiscard]] static Result<OwnedProviderLease<I>> acquire(const OwnedProviderHandle& handle) {
162 std::scoped_lock lock(detail::ownedMutex());
163 auto& slots = detail::ownedSlots(I::capabilityName);
164 const auto found = slots.find(handle.id);
165 if (found == slots.end()) {
166 const auto& generations = detail::ownedGenerations(I::capabilityName);
167 const bool seen = generations.contains(handle.id);
168 return Result<OwnedProviderLease<I>>::failure(Diagnostic::error(
169 seen ? DiagnosticCode::Conflict : DiagnosticCode::Unsupported,
170 seen ? "owned capability handle is stale" : "owned capability provider is not available", "handle"));
171 }
172 if (found->second.handle != handle) {
173 return Result<OwnedProviderLease<I>>::failure(
174 Diagnostic::error(DiagnosticCode::Conflict, "owned capability handle is stale", "handle"));
175 }
176 auto typed = std::static_pointer_cast<I>(found->second.entry->shared());
177 return Result<OwnedProviderLease<I>>::success(OwnedProviderLease<I>(handle, std::move(typed)),
178 Status::success(StatusCode::Applied));
179 }
180
184 [[nodiscard]] static Result<OwnedProviderLease<I>> acquire(std::string_view id) {
185 std::scoped_lock lock(detail::ownedMutex());
186 auto& slots = detail::ownedSlots(I::capabilityName);
187 const auto found = slots.find(std::string(id));
188 if (found == slots.end()) {
189 return Result<OwnedProviderLease<I>>::failure(Diagnostic::error(
190 DiagnosticCode::Unsupported, "owned capability provider is not available", "id"));
191 }
192 auto typed = std::static_pointer_cast<I>(found->second.entry->shared());
193 return Result<OwnedProviderLease<I>>::success(
194 OwnedProviderLease<I>(found->second.handle, std::move(typed)), Status::success(StatusCode::Applied));
195 }

200 [[nodiscard]] static Result<void> unload(const OwnedProviderHandle& handle) {
201 std::shared_ptr<void> keepAlive;
202 {
203 std::scoped_lock lock(detail::ownedMutex());
204 auto& slots = detail::ownedSlots(I::capabilityName);
205 const auto found = slots.find(handle.id);
206 if (found == slots.end() || found->second.handle != handle) {
207 return Result<void>::failure(
208 Diagnostic::error(DiagnosticCode::Conflict, "owned capability handle is stale", "handle"));
209 }
210 // Use auto (not void*) so the local is not linted as a public pointer API.
211 const auto borrowed = found->second.entry->raw();
212 keepAlive = found->second.entry->shared();
213 slots.erase(found);
214 detail::revokeRaw(I::capabilityName, borrowed);
215 }
216 (void)keepAlive;
217 return Result<void>::success(Status::success(StatusCode::Applied));
218 }
