77avatar_editing::EditorResult<void> AvatarDocumentEditor::configureWorkspace(
78 editor::EditorWorkspace& workspace) const {
79 editor::EditorWorkspace candidate = workspace;
80 struct Panel {
81 const char* id;
82 const char* title;
83 const char* region;
84 const char* context;
85 int order;
86 };
87 constexpr Panel panels[] = {
88 {"avatar.layers", "Layers", "left", "list", 100},
89 {"avatar.preview", "Avatar Preview", "center", "preview", 100},
90 {"avatar.inspector", "Avatar Inspector", "right", "inspector", 100},
91 {"avatar.parameters", "Parameters", "bottom", "parameters", 100},
92 {"avatar.expressions", "Expressions", "right", "expressions", 200},
93 };
94 for (const auto& panel : panels) {
95 if (!candidate.registerPanel(panel.id, panel.title, panel.region, panel.order) ||
96 !candidate.setPanelCapability(panel.id, "avatar.document") ||
97 !candidate.setPanelContext(panel.id, panel.context))
98 return editorError(avatar_editing::EditorStatus::Rejected, "editor.avatar.workspace-conflict",
99 "Could not install the avatar workspace composition");
100 }
101 if (!candidate.activatePanel("avatar.preview"))
102 return editorError(avatar_editing::EditorStatus::Rejected, "editor.avatar.workspace-activate",
103 "Could not activate the avatar preview panel");
104 workspace = std::move(candidate);
105 return eve::editing::applied<void>();
106}

121avatar_editing::EditorResult<void> AvatarDocumentEditor::commit(
122 avatar_editing::EditorResult<avatar_editing::DomainOperation> operation, std::string label) {
123 if (!operation.ok())
124 return avatar_editing::EditorResult<void>::failure(operation.status());
125 editor::TransactionSpec spec;
126 spec.id = editor::TransactionId("avatar.document.tx." + std::to_string(++txSequence_));
127 spec.label = std::move(label);
128 spec.target = editor::TargetId(target_.targetId());
129 spec.baseRevision = target_.revision();
130 auto begun = transactions_.begin(std::move(spec));
131 if (!begun.ok())
132 return editorError(begun.code(), "editor.avatar.begin", "Could not begin the avatar transaction");
133 auto appended = transactions_.append(std::move(operation).takeValue());
134 if (!appended.ok()) {
135 auto discarded = transactions_.discard();
136 if (!discarded.ok()) discarded.ignore("pending avatar transaction already inactive");
137 return avatar_editing::EditorResult<void>::failure(appended.status());
138 }
139 auto committed = transactions_.commit();
140 if (!committed.ok())
141 return avatar_editing::EditorResult<void>::failure(committed.status());
142 return refreshPreview();
143}

145avatar_editing::EditorResult<void> AvatarDocumentEditor::refreshPreview() {
146 std::vector<PreviewRect> next;
147 auto layers = target_.layers();
148 std::sort(layers.begin(), layers.end(), [](const auto& a, const auto& b) {
149 return a.zIndex == b.zIndex ? a.id.value() < b.id.value() : a.zIndex < b.zIndex;
150 });
151 const float smile = smileAmount();
152 for (const auto& layer : layers) {
153 if (!layer.visible) continue;
154 PreviewRect rect;
155 rect.id = layer.id.value();
156 rect.name = layer.name;
157 rect.x = layer.offset[0];
158 rect.y = layer.offset[1] + (layer.name == "eyes" ? smile * 10.0f : 0.0f);
159 rect.w = layer.size[0];
160 rect.h = layer.size[1];
161 rect.r = layer.color[0];
162 rect.g = layer.color[1];
163 rect.b = layer.color[2];
164 rect.a = layer.color[3];
165 rect.selected = layer.id.value() == selectedId_;
166 next.push_back(std::move(rect));
167 }
168 preview_ = std::move(next);
169 previewRevision_ = target_.revision();
170 return eve::editing::applied<void>();
171}
