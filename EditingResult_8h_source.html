72[[nodiscard]] constexpr eve::DiagnosticCode diagnosticCodeForStatus(Status status) noexcept {
73 switch (status) {
74 case Status::Rejected: return eve::DiagnosticCode::PreconditionViolation;
75 case Status::Conflict: return eve::DiagnosticCode::Conflict;
76 case Status::NotFound: return eve::DiagnosticCode::NotFound;
77 case Status::Unsupported: return eve::DiagnosticCode::Unsupported;
78 case Status::Cancelled: return eve::DiagnosticCode::Cancelled;
79 case Status::Failed: return eve::DiagnosticCode::Failed;
80 case Status::Ok:
81 case Status::Applied:
82 case Status::NoOp:
83 case Status::Pending: return eve::DiagnosticCode::None;
84 }
85 return eve::DiagnosticCode::Failed;
