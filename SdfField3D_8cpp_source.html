17Result<void> validateSdfField3DBlob(const SdfField3DBlob& blob) {
18 if (blob.kind != SdfField3DSampleKind::VertexGrid)
19 return Result<void>::failure(invalidBlob("Unsupported SDF sample kind"));
20 if (blob.width < 2 || blob.height < 2 || blob.depth < 2 || blob.width > 512 || blob.height > 512 ||
21 blob.depth > 512)
22 return Result<void>::failure(invalidBlob("SDF grid dimensions must be in [2,512]"));
23 if (!finiteScalar(blob.cellSize) || blob.cellSize <= 0.f || !finiteScalar(blob.originX) ||
24 !finiteScalar(blob.originY) || !finiteScalar(blob.originZ) || !finiteScalar(blob.outsideDistance))
25 return Result<void>::failure(invalidBlob("SDF grid metadata must be finite with positive cellSize"));
26 const uint64_t samples = uint64_t(blob.width) * uint64_t(blob.height) * uint64_t(blob.depth);
27 if (samples > 4000000ull || samples != blob.distances.size())
28 return Result<void>::failure(invalidBlob("SDF sample count mismatch or exceeds 4M budget"));
29 for (float distance : blob.distances)
30 if (!finiteScalar(distance) || std::abs(distance) > 10000.f)
31 return Result<void>::failure(invalidBlob("SDF samples must be finite within ±10000"));
32 return Result<void>::success();
33}

35Result<Value> encodeSdfField3DBlob(const SdfField3DBlob& blob) {
36 auto valid = validateSdfField3DBlob(blob);
37 if (!valid) return Result<Value>::failure(valid.status());
38 Value::Array distances;
39 distances.reserve(blob.distances.size());
40 for (float distance : blob.distances) distances.push_back(double(distance));
41 return Result<Value>::success(
42 Value::Object{{"schemaId", std::string(SdfField3DBlob::schemaId)},
43 {"schemaVersion", std::int64_t(SdfField3DBlob::schemaVersion)},
44 {"kind", std::int64_t(blob.kind)},
45 {"width", std::int64_t(blob.width)},
46 {"height", std::int64_t(blob.height)},
47 {"depth", std::int64_t(blob.depth)},
48 {"cellSize", double(blob.cellSize)},
49 {"origin", Value::Array{double(blob.originX), double(blob.originY), double(blob.originZ)}},
50 {"outsideDistance", double(blob.outsideDistance)},
51 {"distances", Value(std::move(distances))}});

54Result<SdfField3DBlob> decodeSdfField3DBlob(const Value& value) {
55 const auto* object = value.getIf<Value::Object>();
56 if (!object) return Result<SdfField3DBlob>::failure(invalidBlob("SDF blob must be an object"));
57 for (const char* key : {"schemaId", "schemaVersion", "kind", "width", "height", "depth", "cellSize", "origin",
58 "outsideDistance", "distances"})
59 if (!object->contains(key))
60 return Result<SdfField3DBlob>::failure(invalidBlob("SDF blob missing required field"));
61 const auto* schema = object->at("schemaId").getIf<std::string>();
62 const auto* version = object->at("schemaVersion").getIf<std::int64_t>();
63 const auto* kind = object->at("kind").getIf<std::int64_t>();
64 const auto* width = object->at("width").getIf<std::int64_t>();
65 const auto* height = object->at("height").getIf<std::int64_t>();
66 const auto* depth = object->at("depth").getIf<std::int64_t>();
67 const auto* origin = object->at("origin").getIf<Value::Array>();
68 const auto* samples = object->at("distances").getIf<Value::Array>();
69 auto asFloat = [](const Value& entry) -> float {
70 if (auto real = entry.getIf<double>()) return float(*real);
71 if (auto integer = entry.getIf<std::int64_t>()) return float(*integer);
72 return std::numeric_limits<float>::quiet_NaN();
73 };
74 if (!schema || *schema != SdfField3DBlob::schemaId || !version || *version != SdfField3DBlob::schemaVersion ||
75 !kind || !width || !height || !depth || !origin || origin->size() != 3 || !samples)
76 return Result<SdfField3DBlob>::failure(invalidBlob("SDF blob schema or dimensions are invalid"));
77 SdfField3DBlob blob;
78 blob.kind = static_cast<SdfField3DSampleKind>(*kind);
79 blob.width = int(*width);
80 blob.height = int(*height);
81 blob.depth = int(*depth);
82 blob.cellSize = asFloat(object->at("cellSize"));
83 blob.originX = asFloat((*origin)[0]);
84 blob.originY = asFloat((*origin)[1]);
85 blob.originZ = asFloat((*origin)[2]);
86 blob.outsideDistance = asFloat(object->at("outsideDistance"));
87 blob.distances.reserve(samples->size());
88 for (const auto& sample : *samples) blob.distances.push_back(asFloat(sample));
89 auto valid = validateSdfField3DBlob(blob);
90 if (!valid) return Result<SdfField3DBlob>::failure(valid.status());
91 return Result<SdfField3DBlob>::success(std::move(blob));
92}
