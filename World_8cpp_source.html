39// Signed distance from a world point to a shape (meters). Negative when inside.
40// `normal` points from the shape toward the point (world space, unit length).
41bool shapePointDistance(const b2Shape *shape, const b2Transform &xf, const b2Vec2 &p,
42 float &dist, b2Vec2 &normal) {
43 dist = 0.f;
44 normal = b2Vec2(0.f, 1.f);
45 if (!shape) return false;
46
47 if (shape->GetType() == b2Shape::e_circle) {
48 const auto *circle = static_cast<const b2CircleShape *>(shape);
49 const b2Vec2 center = b2Mul(xf, circle->m_p);
50 const b2Vec2 d = p - center;
51 const float r = circle->m_radius;
52 dist = b2Distance(p, center) - r;
53 const float len = d.Length();
54 if (len > 1e-6f) {
55 normal = (1.f / len) * d;
56 } else if (r > 0.f) {
57 normal = b2Vec2(0.f, 1.f);
58 }
59 return true;
60 }
61
62 if (shape->GetType() == b2Shape::e_polygon) {
63 const auto *poly = static_cast<const b2PolygonShape *>(shape);
64 const b2Vec2 local = b2MulT(xf, p);
65 const int n = poly->m_count;
66 if (n < 3) return false;
67
68 // Signed distance to each edge: positive outside for Box2D CCW polygons
69 // (m_normals point outward). Inside when every side <= 0.
70 float minSide = std::numeric_limits<float>::max();
71 float bestEdge = std::numeric_limits<float>::max();
72 b2Vec2 bestN(0.f, 1.f);
73 b2Vec2 bestQ;
74 bool inside = true;
75 for (int i = 0; i < n; ++i) {
76 const b2Vec2 &a = poly->m_vertices[i];
77 const b2Vec2 &b = poly->m_vertices[(i + 1) % n];
78 const float side = b2Dot(local - a, poly->m_normals[i]);
79 if (side > 0.f) inside = false;
80 if (side < minSide) {
81 minSide = side;
82 bestN = poly->m_normals[i];
83 }
84 // Closest point on the edge segment.
85 const b2Vec2 ab = b - a;
86 const float len2 = b2Dot(ab, ab);
87 float t = 0.f;
88 if (len2 > 1e-12f) t = b2Clamp(b2Dot(local - a, ab) / len2, 0.f, 1.f);
89 const b2Vec2 q = a + t * ab;
90 const float d2 = b2DistanceSquared(local, q);
91 if (d2 < bestEdge) {
92 bestEdge = d2;
93 bestQ = q;
94 }
95 }
96 if (inside) {
97 dist = minSide; // negative penetration
98 normal = b2Mul(xf.q, bestN);
99 } else {
100 dist = std::sqrt(bestEdge);
101 const b2Vec2 delta = local - bestQ;
102 if (delta.LengthSquared() > 1e-8f) {
103 normal = b2Mul(xf.q, (1.f / delta.Length()) * delta);
104 } else {
105 normal = b2Mul(xf.q, bestN);
106 }
107 }
108 return true;
109 }
110
111 if (shape->GetType() == b2Shape::e_edge) {
112 const auto *edge = static_cast<const b2EdgeShape *>(shape);
113 const b2Vec2 a = b2Mul(xf, edge->m_vertex1);
114 const b2Vec2 b = b2Mul(xf, edge->m_vertex2);
115 const b2Vec2 ab = b - a;
116 const float len2 = b2Dot(ab, ab);
117 float t = 0.f;
118 if (len2 > 1e-12f) t = b2Clamp(b2Dot(p - a, ab) / len2, 0.f, 1.f);
119 const b2Vec2 q = a + t * ab;
120 const b2Vec2 delta = p - q;
121 dist = delta.Length();
122 if (dist > 1e-6f) {
123 normal = (1.f / dist) * delta;
124 } else if (len2 > 1e-12f) {
125 normal = (1.f / std::sqrt(len2)) * b2Vec2(-ab.y, ab.x);
126 }
127 return true;
128 }
129
130 if (shape->GetType() == b2Shape::e_chain) {
131 const auto *chain = static_cast<const b2ChainShape *>(shape);
132 float best = std::numeric_limits<float>::max();
133 b2Vec2 bestN(0.f, 1.f);
134 for (int i = 0; i + 1 < chain->m_count; ++i) {
135 const b2Vec2 a = b2Mul(xf, chain->m_vertices[i]);
136 const b2Vec2 b = b2Mul(xf, chain->m_vertices[i + 1]);
137 const b2Vec2 ab = b - a;
138 const float len2 = b2Dot(ab, ab);
139 float t = 0.f;
140 if (len2 > 1e-12f) t = b2Clamp(b2Dot(p - a, ab) / len2, 0.f, 1.f);
141 const b2Vec2 q = a + t * ab;
142 const b2Vec2 delta = p - q;
143 const float d = delta.Length();
144 if (d < best) {
145 best = d;
146 bestN = d > 1e-6f
147 ? (1.f / d) * delta
148 : (len2 > 1e-12f ? (1.f / std::sqrt(len2)) * b2Vec2(-ab.y, ab.x)
149 : b2Vec2(0.f, 1.f));
150 }
151 }
152 if (best < std::numeric_limits<float>::max()) {
153 dist = best;
154 normal = bestN;
155 return true;
156 }
157 }
158 return false;
159}

177eve::Result<eve::SimulationStep> makeLegacyStep(float dt, eve::SimulationTick currentTick) {
178 if (!std::isfinite(dt) || dt < 0.f) {
179 return eve::Result<eve::SimulationStep>::failure(
180 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
181 "World update dt must be finite and non-negative", "physics.world.update.dt"));
182 }
183 const float normalized = std::min(dt, 0.05f);
184 const auto nextTick = currentTick.incremented();
185 if (!nextTick) {
186 return eve::Result<eve::SimulationStep>::failure(
187 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
188 "World simulation tick cannot be incremented", "physics.world.simulationTick"));
189 }
190 auto duration = eve::Duration::fromSeconds(static_cast<double>(normalized));
191 if (!duration) return eve::Result<eve::SimulationStep>::failure(duration.status());
192 return eve::Result<eve::SimulationStep>::success({*nextTick, std::move(duration).takeValue()});
193}

235void World::destroy() {
236 if (destroyed_) return;
237 destroyed_ = true;
238
239 // Copy sets 鈥?Body/Fixture destructors erase from them.
240 std::vector<Body *> bodies(bodies_.begin(), bodies_.end());
241 for (Body *b : bodies) {
242 if (b) {
243 b->invalidate();
244 // Script may still hold Body*; leave object but null raw pointer.
245 // If World is script-owned and Body is also script-owned, Body::~Body
246 // will see null body_ and skip DestroyBody.
247 }
248 }
249 bodies_.clear();
250
251 std::vector<Fixture *> fixtures(fixtures_.begin(), fixtures_.end());
252 for (Fixture *f : fixtures) {
253 if (f) f->invalidate();
254 }
255 fixtures_.clear();
256 clearContactEvents();
257
258 simulation_.reset();
259
260 if (world_) {
261 world_->SetContactListener(nullptr);
262 delete world_;
263 world_ = nullptr;
264 }
265 delete relay_;
266 relay_ = nullptr;
267 runtimeHandle_ = PhysicsWorldHandle::invalid();
268}

270bool World::pointProbe(float x, float y, float radius, ClothContact *out) const {
271 if (out) *out = ClothContact{};
272 if (!isValid() || radius <= 0.f) return false;
273
274 const float rM = toMeters(radius);

277 struct Probe : b2QueryCallback {
278 const World *world = nullptr;
279 ClothContact *best = nullptr;
280 b2Vec2 center;
281 float radiusM = 0.f;
282
283 bool ReportFixture(b2Fixture *fixture) override {
284 if (!fixture || fixture->IsSensor()) return true;
285 const b2Shape *shape = fixture->GetShape();
286 if (!shape) return true;
287 float dist;
288 b2Vec2 normal;
289 if (!shapePointDistance(shape, fixture->GetBody()->GetTransform(), center, dist,
290 normal)) {
291 return true;
292 }
293 const float depth = radiusM - dist;
294 if (depth > 0.f && depth > best->depth) {
295 best->hit = true;
296 best->depth = world->toPixels(depth);
297 best->nx = normal.x;
298 best->ny = normal.y;
299 best->body = static_cast<Body *>(fixture->GetBody()->GetUserData());
300 }
301 return true;
302 }
303 } probe;
304 probe.world = this;
305 probe.best = out;
306 probe.center = centerM;
307 probe.radiusM = rM;
308
309 b2AABB aabb;
310 aabb.lowerBound = centerM - b2Vec2(rM, rM);
311 aabb.upperBound = centerM + b2Vec2(rM, rM);
312 world_->QueryAABB(&probe, aabb);
313 return out->hit;
314}

429void World::forgetFixture(Fixture *fixture) {
430 if (!fixture) return;
431 const int bodyId = fixture->getBodyId();
432 const std::string tag = fixture->getTag();
433 auto touches = [&](const ContactEvent &e) {
434 return (e.bodyAId == bodyId && e.fixtureATag == tag) ||
435 (e.bodyBId == bodyId && e.fixtureBTag == tag);
436 };
437 beginContacts_.erase(std::remove_if(beginContacts_.begin(), beginContacts_.end(), touches),
438 beginContacts_.end());
439 endContacts_.erase(std::remove_if(endContacts_.begin(), endContacts_.end(), touches),
440 endContacts_.end());
441 impacts_.erase(std::remove_if(impacts_.begin(), impacts_.end(), touches), impacts_.end());
442 b2Fixture *raw = fixture->raw();
443 for (auto it = preSolve_.begin(); it != preSolve_.end();) {
444 b2Contact *contact = it->first;
445 if (contact && (contact->GetFixtureA() == raw || contact->GetFixtureB() == raw))
446 it = preSolve_.erase(it);
447 else
448 ++it;
449 }
450 fixtures_.erase(fixture);
451}

486void World::onPreSolve(b2Contact *contact, const b2Manifold * /*oldManifold*/) {
487 if (!contact || !world_) return;
488 b2WorldManifold manifold;
489 contact->GetWorldManifold(&manifold);
490 const b2Manifold *local = contact->GetManifold();
491 if (!local || local->pointCount <= 0) return;
492
493 b2Body *a = contact->GetFixtureA()->GetBody();
494 b2Body *b = contact->GetFixtureB()->GetBody();
495 if (!a || !b) return;
496 const b2Vec2 point = manifold.points[0];
497 const b2Vec2 va = a->GetLinearVelocityFromWorldPoint(point);
498 const b2Vec2 vb = b->GetLinearVelocityFromWorldPoint(point);
499
500 PreSolveData data;
501 data.pointX = toPixels(point.x);
502 data.pointY = toPixels(point.y);
503 data.normalX = manifold.normal.x;
504 data.normalY = manifold.normal.y;
505 data.relativeNormalSpeed = toPixels(std::max(0.f, b2Dot(va - vb, manifold.normal)));
506 preSolve_[contact] = data;
507}

509void World::onPostSolve(b2Contact *contact, const b2ContactImpulse *impulse) {
510 if (!contact || !impulse) return;
511 auto found = preSolve_.find(contact);
512 if (found == preSolve_.end()) return;
513
514 ImpactEvent out;
515 static_cast<ContactEvent &>(out) = contactEventFrom(contact);
516 out.pointX = found->second.pointX;
517 out.pointY = found->second.pointY;
518 out.normalX = found->second.normalX;
519 out.normalY = found->second.normalY;
520 out.relativeNormalSpeed = found->second.relativeNormalSpeed;
521 const int count = contact->GetManifold() ? contact->GetManifold()->pointCount : 0;
522 for (int i = 0; i < count; ++i) {
523 out.normalImpulse += toPixels(impulse->normalImpulses[i]);
524 out.tangentImpulse += toPixels(std::fabs(impulse->tangentImpulses[i]));
525 }
526 if (out.normalImpulse > 0.f) impacts_.push_back(std::move(out));
527}
