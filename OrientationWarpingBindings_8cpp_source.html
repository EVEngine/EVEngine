19std::vector<int> copyBoneArray(ssq::Array values, const char* label, int maximum) {
20 if (values.size() > static_cast<std::size_t>(maximum))
21 throw std::runtime_error(std::string(label) + " exceeds the supported bone count");
22 std::vector<int> bones;
