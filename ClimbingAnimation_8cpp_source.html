37eve::Result<void> beginClimbingAnimation(animation::AnimPlayer& player, animation::AnimClip& clip,
38 int rootMotionBone) {
39 animation::AnimSkeleton* skeleton = player.getSkeleton();
40 if (!skeleton || rootMotionBone < 0 || rootMotionBone >= skeleton->getBoneCount())
41 return failure<void>(eve::DiagnosticCode::InvalidArgument, "root-motion bone is invalid", "rootMotionBone");
42 if (!std::isfinite(clip.getDuration()) || clip.getDuration() <= 0.f)
43 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
44 "climbing animation clip must have a positive finite duration", "clip.duration");
45 player.setRootMotionBone(rootMotionBone);
46 player.setLoop(false);
47 player.play(&clip);
48 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
49}

51eve::Result<void> beginClimbingAnimation(animation::AnimPlayer& player, animation::AnimClip& clip,
52 const ClimbingAnimationBinding& binding) {
53 if (!binding.clipId.empty() && clip.getName() != binding.clipId)
54 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
55 "animation clip does not match the climbing binding", "binding.clipId");
56 animation::AnimSkeleton* skeleton = player.getSkeleton();
57 if (!skeleton)
58 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
59 "animation player has no skeleton", "player.skeleton");
60 const int rootMotionBone = skeleton->findBone(binding.rootBone);
61 if (rootMotionBone < 0)
62 return failure<void>(eve::DiagnosticCode::NotFound,
63 "climbing root-motion bone does not exist", "binding.rootBone");
64 return beginClimbingAnimation(player, clip, rootMotionBone);
65}

67eve::Result<ClimbingAnimationFrame> advanceClimbingAnimation(animation::AnimPlayer& player,
68 const eve::SimulationStep& step, Vec3 facing,
69 Vec3 pelvisOffset) {
70 if (!finite(facing) || !finite(pelvisOffset))
71 return failure<ClimbingAnimationFrame>(eve::DiagnosticCode::InvalidArgument,
72 "facing and pelvis offset must be finite", "animationInput");
73 auto advanced = player.advance(step);
74 if (!advanced) return eve::Result<ClimbingAnimationFrame>::failure(advanced.status());
75
76 ClimbingAnimationFrame frame;
77 frame.motion.rootTranslation = {player.getRootMotionX(), player.getRootMotionY(), player.getRootMotionZ()};
78 frame.motion.facing = facing;
79 frame.motion.pelvisOffset = pelvisOffset;
80 const float qx = player.getRootMotionRotationX();
81 const float qy = player.getRootMotionRotationY();
82 const float qz = player.getRootMotionRotationZ();
83 const float qw = player.getRootMotionRotationW();
84 frame.motion.rootYawRadians = std::atan2(2.f * (qw * qy + qx * qz), 1.f - 2.f * (qy * qy + qz * qz));
85 frame.motion.hasRootMotion = player.isPlaying();
86 if (!finite(frame.motion.rootTranslation) || !std::isfinite(frame.motion.rootYawRadians))
87 return failure<ClimbingAnimationFrame>(eve::DiagnosticCode::InvariantViolation,
88 "animation player produced non-finite root motion", "rootMotion");
89
90 for (int index = 0; index < player.getEventCount(); ++index) {
91 const auto parsed = parseNotify(player.getEventName(index));
92 if (parsed) {
93 frame.motion.notifies.push_back(*parsed);
94 ++frame.recognizedNotifyCount;
95 } else {
96 ++frame.ignoredNotifyCount;
97 }
98 }
99 return eve::Result<ClimbingAnimationFrame>::success(std::move(frame),
100 eve::Status::success(eve::StatusCode::Applied));
101}

103eve::Result<ClimbingGraphProvider> driveClimbingGraph(animation::AnimGraph& graph, int oneShotNode,
104 ClimbingExecutionId executionId, ClimbingGraphState& state) {
105 if (executionId.isZero())
106 return failure<ClimbingGraphProvider>(eve::DiagnosticCode::InvalidArgument,
107 "graph projection requires a committed execution id", "executionId");
108 if (oneShotNode < 0 || oneShotNode >= graph.getNodeCount())
109 return failure<ClimbingGraphProvider>(eve::DiagnosticCode::InvalidArgument,
110 "climbing graph one-shot node is invalid", "oneShotNode");
111 if (state.activeExecutionId == executionId && state.provider == ClimbingGraphProvider::AnimGraph)
112 return eve::Result<ClimbingGraphProvider>::success(ClimbingGraphProvider::AnimGraph,
113 eve::Status::success(eve::StatusCode::NoOp));
114 try {
115 graph.trigger(oneShotNode);
116 } catch (const eve::Exception& exception) {
117 return failure<ClimbingGraphProvider>(eve::DiagnosticCode::PreconditionViolation, exception.what(),
118 "oneShotNode");
119 }
120 state.activeExecutionId = executionId;
121 state.provider = ClimbingGraphProvider::AnimGraph;
122 return eve::Result<ClimbingGraphProvider>::success(state.provider,
123 eve::Status::success(eve::StatusCode::Applied));
124}
