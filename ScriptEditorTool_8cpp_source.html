20void invoke(const ssq::Object &callback, const Args &...args) {
21 if (callback.isEmpty()) return;
22 ssq::Function function = callback.toFunction();
23 if (function.isEmpty()) return;
24 HSQUIRRELVM vm = function.getHandle();
25 const SQInteger top = sq_gettop(vm);
26 sq_pushobject(vm, function.getRaw());
27 sq_pushroottable(vm);
28 (ssq::detail::pushValue(vm, args), ...);
29 sq_call(vm, sizeof...(Args) + 1, SQFalse, SQTrue);
30 sq_settop(vm, top);
31}

33template <typename... Args>
34int invokeFlags(const ssq::Object &callback, const Args &...args) {
35 if (callback.isEmpty()) return 0;
36 ssq::Function function = callback.toFunction();
37 if (function.isEmpty()) return 0;
38 HSQUIRRELVM vm = function.getHandle();
39 const SQInteger top = sq_gettop(vm);
40 sq_pushobject(vm, function.getRaw());
41 sq_pushroottable(vm);
42 (ssq::detail::pushValue(vm, args), ...);
43 int flags = 0;
44 if (SQ_SUCCEEDED(sq_call(vm, sizeof...(Args) + 1, SQTrue, SQTrue))) {
45 SQInteger result = 0;
46 if (SQ_SUCCEEDED(sq_getinteger(vm, -1, &result))) flags = static_cast<int>(result);
47 }
48 sq_settop(vm, top);
49 return flags;
50}
