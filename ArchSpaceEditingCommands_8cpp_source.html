13const editing::Value* field(const editing::Value& value, const char* key) {
14 const auto* object = value.getIf<editing::Value::Object>();
15 if (!object) return nullptr;
16 const auto found = object->find(key);
17 return found == object->end() ? nullptr : &found->second;
18}

20const std::string* stringField(const editing::Value& value, const char* key) {
21 const editing::Value* entry = field(value, key);
22 return entry ? entry->getIf<std::string>() : nullptr;
23}

25bool readNumber(const editing::Value& value, const char* key, double& output, bool required) {
26 const editing::Value* entry = field(value, key);
27 if (!entry) return !required;
28 if (const auto* real = entry->getIf<double>()) {
29 output = *real;
30 return std::isfinite(output);
31 }
32 if (const auto* integer = entry->getIf<std::int64_t>()) {
33 output = static_cast<double>(*integer);
34 return true;
35 }
36 return false;
37}

59editing::Result<editing::CommandPlan> planFrom(editing::Result<editing::DomainOperation> operation,
60 editing::Value summary) {
61 if (!operation.ok())
62 return error<editing::CommandPlan>(operation.code(), "archspace.editing.operation",
63 "ArchSpace target rejected the planned edit");
64 editing::CommandPlan plan;
65 plan.operations.push_back(std::move(operation).value());
66 plan.summary = std::move(summary);
67 return eve::editing::applied<editing::CommandPlan>(std::move(plan));

86 editing::Value::Object{{"levelId", *level}});
