23Module* ModuleManager::find(const char* name, creator_t factory) {
24 EV_PARAM_CHECK(name != nullptr, "module name must not be null");
25 auto& state = *inst().state_;
26 auto current = state.registered_modules.find(name);
27 if (current != state.registered_modules.end() && current->second.creator == factory)
28 return current->second.instance;
29 auto previous = state.plugin_registration_previous_.find(name);
30 if (previous != state.plugin_registration_previous_.end() && previous->second.creator == factory)
31 return previous->second.instance;
32 for (auto& retired : state.retired_modules_)
33 if (retired.first == name && retired.second.creator == factory) return retired.second.instance;
34 return nullptr;
35}

37void ModuleManager::insert(const char* name, Module* instance, creator_t factory) {
38 EV_PARAM_CHECK(name != nullptr, "module name must not be null");
39 EV_PARAM_CHECK(instance != nullptr, "module instance must not be null");
40 auto& state = *inst().state_;
41 auto current = state.registered_modules.find(name);
42 if (current != state.registered_modules.end() && current->second.creator == factory) {
43 current->second.instance = instance;
44 return;
45 }
46 auto previous = state.plugin_registration_previous_.find(name);
47 if (previous != state.plugin_registration_previous_.end() && previous->second.creator == factory) {
48 previous->second.instance = instance;
49 return;

51 for (auto& retired : state.retired_modules_) {
52 if (retired.first == name && retired.second.creator == factory) {
53 retired.second.instance = instance;

84void ModuleManager::register_module(const char* name, creator_t c, exposer_t e) {
85 EV_PARAM_CHECK(name != nullptr, "module name must not be null");
86 const bool hasCreator = c != nullptr; // function pointers can't be printed by zeroerr
87 EV_PARAM_CHECK(hasCreator, "module creator must not be null");
88 auto p = inst().state_->registered_modules.find(name);
89 if (inst().state_->plugin_registration_active_ && p != inst().state_->registered_modules.end()) {
90 auto& state = *inst().state_;
91 if (state.plugin_registration_owner_.empty() || p->second.owner != state.plugin_registration_owner_ ||
92 std::find(state.plugin_registration_added_.begin(), state.plugin_registration_added_.end(), name) !=
93 state.plugin_registration_added_.end()) {
94 if (state.plugin_registration_error_.empty())
95 state.plugin_registration_error_ = std::string("module name already registered: ") + name;
96 return;
97 }
98 state.plugin_registration_previous_.emplace(name, p->second);
99 state.plugin_registration_added_.emplace_back(name);
100 p->second = {c, e, nullptr, false, state.plugin_registration_owner_};
101 return;
102 }
103 if (p == inst().state_->registered_modules.end()) {
104 inst().state_->registered_modules.insert(std::make_pair(
105 std::string(name), State::ModuleInfo{c, e, nullptr, false, inst().state_->plugin_registration_owner_}));
106 if (inst().state_->plugin_registration_active_) inst().state_->plugin_registration_added_.emplace_back(name);
107 } else {
108 p->second = {c, e, nullptr, false, {}};
109 }
110}

127eve::Result<void> ModuleManager::finishPluginRegistration(bool commit) {
128 auto& manager = inst();
129 if (!manager.state_->plugin_registration_active_)
130 return eve::Result<void>::failure(eve::Diagnostic::error(
131 eve::DiagnosticCode::Conflict, "no plugin registration transaction is active", {}, {}, "module"));
132
133 std::string error = manager.state_->plugin_registration_error_;
134 if (commit && error.empty() && !manager.state_->plugin_registration_owner_.empty()) {
135 // Reserve retirement slots before publishing script bindings.
136 manager.state_->retired_modules_.reserve(manager.state_->retired_modules_.size() +
137 manager.state_->registered_modules.size());
138 auto published = publishPluginBindings();
139 if (!published.ok()) error = published.status().describe();
140 }
141 if (!commit || !error.empty()) {
142 for (const auto& name : manager.state_->plugin_registration_added_) {
143 auto it = manager.state_->registered_modules.find(name);
144 if (it == manager.state_->registered_modules.end()) continue;
145 delete it->second.instance;
146 manager.state_->registered_modules.erase(it);
147 }
148 for (auto& previous : manager.state_->plugin_registration_previous_)
149 manager.state_->registered_modules.emplace(previous.first, std::move(previous.second));
150 } else if (!manager.state_->plugin_registration_owner_.empty()) {
151 for (auto& previous : manager.state_->plugin_registration_previous_)
152 manager.state_->retired_modules_.emplace_back(previous.first, std::move(previous.second));
153 // A module removed by the new binary disappears from the current registry,
154 // while its captured constructors and live instances keep their generation.
155 auto& modules = manager.state_->registered_modules;
156 for (auto it = modules.begin(); it != modules.end();) {
157 if (it->second.owner == manager.state_->plugin_registration_owner_ &&
158 std::find(manager.state_->plugin_registration_added_.begin(),
159 manager.state_->plugin_registration_added_.end(),
160 it->first) == manager.state_->plugin_registration_added_.end()) {
161 manager.state_->retired_modules_.emplace_back(it->first, std::move(it->second));
162 it = modules.erase(it);
163 } else {
164 ++it;
165 }
166 }
167 }
168 manager.state_->plugin_registration_active_ = false;
169 manager.state_->plugin_registration_added_.clear();
170 manager.state_->plugin_registration_error_.clear();
171 manager.state_->plugin_registration_previous_.clear();
172 manager.state_->plugin_registration_owner_.clear();
173 if (!error.empty())
174 return eve::Result<void>::failure(
175 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, error, {}, {}, "module"));
176 return eve::Result<void>::success(eve::Status::success(commit ? eve::StatusCode::Applied : eve::StatusCode::NoOp));
177}
