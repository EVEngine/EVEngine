206int VoxelWorld::unloadChunksOutside(int centerX, int centerY, int centerZ, int radiusChunks) {
207 if (radiusChunks < 0) return 0;
208 const int64_t r2 = int64_t(radiusChunks) * int64_t(radiusChunks);
209 std::vector<uint64_t> evict;
210 evict.reserve(chunks_.size() / 4);
211 for (auto &kv : chunks_) {
212 int cx, cy, cz;
213 unpackKey(kv.first, cx, cy, cz);
214 const int64_t dx = int64_t(cx) - centerX;
215 const int64_t dy = int64_t(cy) - centerY;
216 const int64_t dz = int64_t(cz) - centerZ;
217 if (dx * dx + dy * dy + dz * dz > r2) evict.push_back(kv.first);
218 }
219 for (uint64_t k : evict) chunks_.erase(k);
220 if (!evict.empty()) {
221 // Batch pointers may dangle after eviction; force re-selection.
222 visible_.clear();
223 visibleChunkKeys_.clear();
224 ++revision_;
225 }
226 return int(evict.size());
227}

229StreamStats VoxelWorld::streamAround(int centerX, int centerY, int centerZ, int radiusChunks,
230 const std::function<void(Chunk &, int, int, int)> &generator) {
231 StreamStats stats;
232 if (radiusChunks < 0) return stats;
233
234 if (!generator && terrainAssetEnabled_) {
235 const int assetChunk = terrainAsset_.asset().getChunkSize();
236 const int worldRadius = (radiusChunks + 1) * kChunkSize;
237 const int assetRadius = assetChunk > 0
238 ? int(std::ceil(float(worldRadius) * 1.41421356f / float(assetChunk))) + 1 : 0;
239 terrainAsset_.streamAround(centerX * kChunkSize + kChunkSize / 2,
240 centerZ * kChunkSize + kChunkSize / 2, assetRadius);
241 }
242
243 // Evict first so far-away dirty chunks are not remeshed below.
244 stats.evicted = unloadChunksOutside(centerX, centerY, centerZ, radiusChunks);
245
246 const int64_t r2 = int64_t(radiusChunks) * int64_t(radiusChunks);
247 for (int dz = -radiusChunks; dz <= radiusChunks; ++dz)
248 for (int dy = -radiusChunks; dy <= radiusChunks; ++dy)
249 for (int dx = -radiusChunks; dx <= radiusChunks; ++dx) {
250 const int64_t d2 = int64_t(dx) * dx + int64_t(dy) * dy + int64_t(dz) * dz;
251 if (d2 > r2) continue;
252 const int nx = centerX + dx;
253 const int ny = centerY + dy;
254 const int nz = centerZ + dz;
255 if (hasChunk(nx, ny, nz)) continue;
256 Chunk *c = getOrCreateChunk(nx, ny, nz);
257 if (generator)
258 generator(*c, nx, ny, nz);
259 else if (terrainEnabled_ || terrainAssetEnabled_) {
260 // Heightmap terrain via procgen::TerrainSampler: one column
261 // per (x, z); sampling world coords keeps chunk seams flush.
262 const int wy0 = ny * kChunkSize;
263 for (int lz = 0; lz < kChunkSize; ++lz)
264 for (int lx = 0; lx < kChunkSize; ++lx) {
265 int h = 0;
266 uint8_t surface = terrainTop_;
267 const int wx = nx * kChunkSize + lx;
268 const int wz = nz * kChunkSize + lz;
269 if (terrainAssetEnabled_) {
270 h = terrainHeightAt(wx, wz);
271 surface = terrainSurfaceAt(wx, wz);
272 } else {
273 sampleTerrainColumn(terrainSampler_.get(), terrainBase_, terrainAmplitude_,
274 terrainTop_, terrainSand_, sandLevel_, wx, wz, h, surface);
275 }
276 for (int ly = 0; ly < kChunkSize; ++ly) {
277 const int wy = wy0 + ly;
278 if (wy <= h - 4)
279 c->set(lx, ly, lz, terrainStone_);
280 else if (wy <= h - 1)
281 c->set(lx, ly, lz, terrainSub_);
282 else if (wy == h)
283 c->set(lx, ly, lz, surface);
284 }
285 }
286 }
287 ++stats.created;
288 }
289
290 if (stats.created > 0) {
291 ++revision_;
292 remeshDirty();
293 }
294 return stats;
295}

299void putU32(std::vector<uint8_t> &out, uint32_t v) {
300 out.push_back(uint8_t(v));
301 out.push_back(uint8_t(v >> 8));
302 out.push_back(uint8_t(v >> 16));
303 out.push_back(uint8_t(v >> 24));

308bool getU32(const uint8_t *&p, const uint8_t *end, uint32_t &out) {
309 if (end - p < 4) return false;
310 out = uint32_t(p[0]) | (uint32_t(p[1]) << 8) | (uint32_t(p[2]) << 16) |
311 (uint32_t(p[3]) << 24);
312 p += 4;
313 return true;
314}
315
316bool getI32(const uint8_t *&p, const uint8_t *end, int32_t &out) {
317 uint32_t v;
318 if (!getU32(p, end, v)) return false;
319 out = int32_t(v);
320 return true;
321}

325void VoxelWorld::serializeWorld(std::vector<uint8_t> &out) const {
326 out.clear();
327 const char magic[4] = {'E', 'V', 'V', 'X'};
328 out.insert(out.end(), magic, magic + 4);
329 out.push_back(1); // version
330 putU32(out, uint32_t(chunks_.size()));
331
332 // Deterministic output: sort chunk keys so saves are byte-stable.
333 std::vector<uint64_t> keys;
334 keys.reserve(chunks_.size());
335 for (auto &kv : chunks_) keys.push_back(kv.first);
336 std::sort(keys.begin(), keys.end());
337 for (uint64_t key : keys) {
338 const auto &chunk = chunks_.at(key);
339 int cx, cy, cz;
340 unpackKey(key, cx, cy, cz);
341 putI32(out, int32_t(cx));
342 putI32(out, int32_t(cy));
343 putI32(out, int32_t(cz));
344 const uint8_t *raw = chunk->rawVoxels();
345 out.insert(out.end(), raw, raw + kChunkSize * kChunkSize * kChunkSize);
346 }
347}

349bool VoxelWorld::deserializeWorld(const uint8_t *data, size_t size) {
350 const uint8_t *p = data;
351 const uint8_t *end = data + size;
352 if (size < 9 || std::memcmp(p, "EVVX", 4) != 0 || p[4] != 1) return false;
353 p += 5;
354 uint32_t count = 0;
355 if (!getU32(p, end, count)) return false;
356 const size_t voxelBytes = size_t(kChunkSize) * kChunkSize * kChunkSize;
357 if (uint64_t(count) > (uint64_t(end - p)) / (12 + voxelBytes)) return false;
358
359 clear();
360 for (uint32_t i = 0; i < count; ++i) {
361 int32_t cx = 0, cy = 0, cz = 0;
362 if (!getI32(p, end, cx) || !getI32(p, end, cy) || !getI32(p, end, cz)) return false;
363 if (size_t(end - p) < voxelBytes) return false;
364 Chunk *c = getOrCreateChunk(cx, cy, cz);
365 c->setVoxelData(p);
366 p += voxelBytes;
367 }
368 ++revision_;
369 return true;
370}

383int VoxelWorld::remeshDirty(int maxThreads) {
384 std::vector<Chunk *> dirty;
385 dirty.reserve(chunks_.size());
386 for (auto &kv : chunks_) {
387 if (kv.second->isDirty()) dirty.push_back(kv.second.get());
388 }
389 const int count = int(dirty.size());
390 if (count == 0) return 0;
391
392 const auto remeshOne = [this](Chunk *c) {
393 c->remesh(types_, &VoxelWorld::chunkNeighborSampler, this);
394 };
395
396 int workers = maxThreads;
397 if (workers <= 0) {
398 workers = static_cast<int>(std::thread::hardware_concurrency());
399 if (workers < 1) workers = 1;
400 if (workers > kMaxRemeshWorkers) workers = kMaxRemeshWorkers;
401 }
402
403 if (workers <= 1 || count <= 1) {
404 for (Chunk *c : dirty) remeshOne(c);
405 return count;
406 }
407
408 // Parallel remesh through the engine JobSystem: each child task remeshes
409 // its own slice of distinct chunks. The sampler only reads the chunk map
410 // (no concurrent mutation), and each chunk is touched by exactly one task,
411 // so this is safe. wait() on the loop joins every slice.
412 auto *jobs = thread::Thread::create()->getJobSystem();
413 thread::Job *loop = nullptr;
414 try {
415 loop = jobs->parallelFor(0, count,
416 [this, &dirty, &remeshOne](int first, int last) {
417 for (int k = first; k < last; ++k) remeshOne(dirty[size_t(k)]);
418 },
419 (count + workers - 1) / workers);
420 } catch (...) {
421 // Job allocation failed (resource limits): finish everything serially.
422 // Remesh is idempotent, so any chunks already handled are done twice.
423 for (Chunk *c : dirty) remeshOne(c);
424 return count;
425 }
426 loop->wait();
427 delete loop;
428 return count;
429}

431void VoxelWorld::selectVisible(const float *viewProj16, float eyeX, float eyeY, float eyeZ,
432 float viewRange, bool faceCull) {
433 visible_.clear();
434 visibleChunkKeys_.clear();
435 if (!viewProj16) return;
436
437 const Frustum frustum = Frustum::fromViewProjColumnMajor(viewProj16);
438 const float rangeSq = viewRange > 0.f ? viewRange * viewRange : 0.f;
439
440 for (auto &kv : chunks_) {
441 Chunk *chunk = kv.second.get();
442
443 float minX, minY, minZ, maxX, maxY, maxZ;
444 chunk->worldAABB(minX, minY, minZ, maxX, maxY, maxZ);
445 const float cx = (minX + maxX) * 0.5f;
446 const float cy = (minY + maxY) * 0.5f;
447 const float cz = (minZ + maxZ) * 0.5f;
448
449 if (viewRange > 0.f) {
450 const float dx = cx - eyeX;
451 const float dy = cy - eyeY;
452 const float dz = cz - eyeZ;
453 if (dx * dx + dy * dy + dz * dz > rangeSq) continue;
454 }
455
456 if (!frustum.intersectsAABB(minX, minY, minZ, maxX, maxY, maxZ)) continue;
457
458 // Mesh only after range/frustum culling so edits far outside the view
459 // are not remeshed every frame.
460 chunk->ensureMeshed(types_, &VoxelWorld::chunkNeighborSampler, this);
461
462 visibleChunkKeys_.push_back(kv.first);
463
464 for (int i = 0; i < faceDirCount(); ++i) {
465 const FaceDir dir = FaceDir(i);
466 const int count = chunk->faceRectCount(dir);
467 if (count <= 0) continue;
468
469 if (faceCull) {
470 // A perspective camera has a different view vector at every surface point.
471 // Reject a direction only when every possible face plane in the chunk faces
472 // away from the eye. Testing against the chunk centre is not conservative:
473 // when the eye crosses that centre plane it can discard faces on the far half
474 // of the chunk that are still front-facing and visible (notably cave walls).
475 bool allBackFacing = false;
476 switch (dir) {
477 case FaceDir::PosX: allBackFacing = eyeX <= minX; break;
478 case FaceDir::NegX: allBackFacing = eyeX >= maxX; break;
479 case FaceDir::PosY: allBackFacing = eyeY <= minY; break;
480 case FaceDir::NegY: allBackFacing = eyeY >= maxY; break;
481 case FaceDir::PosZ: allBackFacing = eyeZ <= minZ; break;
482 case FaceDir::NegZ: allBackFacing = eyeZ >= maxZ; break;
483 case FaceDir::Count: break;
484 }
485 if (allBackFacing) continue;
486 }
487
488 DrawBatch batch;
489 batch.chunk = chunk;
490 batch.dir = dir;
491 batch.packed = chunk->facePackedData(dir);
492 batch.ao = chunk->faceAOPackedData(dir);
493 batch.count = count;
494 visible_.push_back(batch);
495 }
496 }

554void VoxelWorld::markNeighborChunksDirty(int cx, int cy, int cz, int lx, int ly, int lz) {
555 auto mark = [this](int nx, int ny, int nz) {
556 if (Chunk *n = getChunk(nx, ny, nz)) n->markDirty();
557 };
558 if (lx == 0) mark(cx - 1, cy, cz);
559 if (lx == kChunkSize - 1) mark(cx + 1, cy, cz);
560 if (ly == 0) mark(cx, cy - 1, cz);

575bool VoxelWorld::raycast(float ox, float oy, float oz, float dx, float dy, float dz,
576 float maxDist, int &hitX, int &hitY, int &hitZ, int &prevX,
577 int &prevY, int &prevZ, int &faceX, int &faceY, int &faceZ) const {
578 const float lenSq = dx * dx + dy * dy + dz * dz;
579 if (lenSq <= 1e-12f || maxDist <= 0.f) return false;
580 const float invLen = 1.f / std::sqrt(lenSq);
581 const float rx = dx * invLen;
582 const float ry = dy * invLen;
583 const float rz = dz * invLen;
584
585 int ix = int(std::floor(ox));
586 int iy = int(std::floor(oy));
587 int iz = int(std::floor(oz));
588
589 if (getVoxel(ix, iy, iz) != 0) {
590 hitX = prevX = ix;
591 hitY = prevY = iy;
592 hitZ = prevZ = iz;
593 faceX = faceY = faceZ = 0;
594 return true;
595 }
596
597 const float inf = std::numeric_limits<float>::infinity();
598 const int stepX = rx > 0.f ? 1 : -1;
599 const int stepY = ry > 0.f ? 1 : -1;
600 const int stepZ = rz > 0.f ? 1 : -1;
601 const float absInvX = rx != 0.f ? std::fabs(1.f / rx) : inf;
602 const float absInvY = ry != 0.f ? std::fabs(1.f / ry) : inf;
603 const float absInvZ = rz != 0.f ? std::fabs(1.f / rz) : inf;
604 float tMaxX = rx != 0.f ? (rx > 0.f ? float(ix + 1) - ox : ox - float(ix)) * absInvX : inf;
605 float tMaxY = ry != 0.f ? (ry > 0.f ? float(iy + 1) - oy : oy - float(iy)) * absInvY : inf;
606 float tMaxZ = rz != 0.f ? (rz > 0.f ? float(iz + 1) - oz : oz - float(iz)) * absInvZ : inf;
607
608 prevX = ix;
609 prevY = iy;
610 prevZ = iz;
611 faceX = faceY = faceZ = 0;
612
613 constexpr int kMaxSteps = 4096;
614 for (int iter = 0; iter < kMaxSteps; ++iter) {
615 float t;
616 if (tMaxX <= tMaxY && tMaxX <= tMaxZ) {
617 ix += stepX;
618 t = tMaxX;
619 tMaxX += absInvX;
620 faceX = -stepX;
621 faceY = 0;
622 faceZ = 0;
623 } else if (tMaxY <= tMaxZ) {
624 iy += stepY;
625 t = tMaxY;
626 tMaxY += absInvY;
627 faceX = 0;
628 faceY = -stepY;
629 faceZ = 0;
630 } else {
631 iz += stepZ;
632 t = tMaxZ;
633 tMaxZ += absInvZ;
634 faceX = 0;
635 faceY = 0;
636 faceZ = -stepZ;
637 }
638 if (t > maxDist) return false;
639 if (getVoxel(ix, iy, iz) != 0) {
640 hitX = ix;
641 hitY = iy;
642 hitZ = iz;
643 return true;
644 }
645 prevX = ix;
646 prevY = iy;
647 prevZ = iz;
648 }
649 return false;
650}
