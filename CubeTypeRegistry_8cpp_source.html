18// 把某体素在面 dir 上应显示的纹理，从“基础类型 faceTex + orientation”旋转出来。
19// faceTex_variant[dir] = faceTex_base[rotateFaceY(dir, -orientation)]。
20void rotateFaceTex(const uint8_t src[6], int orientation, uint8_t dst[6]) {
21 for (int d = 0; d < 6; ++d) {
22 const FaceDir srcDir = rotateFaceY(FaceDir(d), (4 - (orientation & 3)) & 3);
23 dst[d] = src[int(srcDir)];
24 }
25}

27void readFaceTex(Value o, uint8_t out[6]) {
28 std::memset(out, 0, sizeof(uint8_t) * 6);
29 const Value arr = o.get("faceTex");
30 const size_t n = arr.size() < 6 ? arr.size() : 6;
31 for (size_t i = 0; i < n; ++i) out[i] = uint8_t(arr.at(i).asInt(0) & 0xFF);
32}

49uint8_t CubeTypeRegistry::add(const CubeType &type) {
50 if (type.name.empty()) return 0;
51 auto it = byName_.find(type.name);
52 if (it != byName_.end()) return it->second;
53
54 const int variants = type.directional ? 4 : 1;
55 // 上限：id 为 uint8_t，0 保留给空气，最多 255 个类型槽位。
56 if (int(types_.size()) + variants > 256) return 0;
57
58 const uint8_t baseId = uint8_t(types_.size());
59 for (int o = 0; o < variants; ++o) {
60 CubeType v = type;
61 v.id = uint8_t(baseId + o);
62 if (type.directional && o > 0) {
63 rotateFaceTex(type.faceTex, o, v.faceTex);
64 }
65 types_.push_back(v);
66 }
67 byName_[type.name] = baseId;
68 return baseId;
69}

71int CubeTypeRegistry::loadFromJson(const std::string &json, std::string *error) {
72 std::string err;
73 const eve::json::Document doc = eve::json::Document::parse(json, &err);
74 if (!doc.valid()) {
75 if (error) *error = err.empty() ? "invalid json" : err;
76 return 0;
77 }
78
79 const Value root = doc.root();
80 int n = 0;
81 if (root.isArray()) {
82 for (size_t i = 0; i < root.size(); ++i) {
83 CubeType t = parseCubeType(root.at(i));
84 if (t.name.empty()) continue;
85 add(t);
86 ++n;
87 }
88 } else if (root.isObject()) {
89 CubeType t = parseCubeType(root);
90 if (!t.name.empty()) {
91 add(t);
92 ++n;
93 }
94 }
95 return n;
96}
