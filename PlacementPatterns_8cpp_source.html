20eve::Result<PlacementSystem::EdgePathPreview> PlacementSystem::previewEdgePath(
21 PlacementWorld *world, const std::string &buildingId, const std::vector<CornerAddress> &vertices) {
22 if (!world || vertices.size() < 2) {
23 return eve::Result<EdgePathPreview>::failure(eve::Diagnostic::error(
24 eve::DiagnosticCode::InvalidArgument, "edge path preview requires a world and at least two vertices",
25 buildingId, {}, "building.edge-path"));
26 }
27 const BuildingDefinition *definition = BuildingRegistry::find(buildingId);
28 if (!definition || definition->placementKind != "edge") {
29 return eve::Result<EdgePathPreview>::failure(eve::Diagnostic::error(
30 eve::DiagnosticCode::InvalidArgument, "edge path preview requires an edge building definition", buildingId,
31 {}, "building.edge-path"));
32 }
33 EdgePathPreview preview;
34 preview.buildingId = buildingId;
35 preview.level = world->getActiveLevel();
36 std::unordered_set<std::string> uniqueEdges;
37 for (size_t i = 1; i < vertices.size(); ++i) {
38 const CornerAddress start = vertices[i - 1];
39 const CornerAddress end = vertices[i];
40 const bool horizontal = start.y == end.y;
41 const bool vertical = start.x == end.x;
42 if (horizontal == vertical) {
43 return eve::Result<EdgePathPreview>::failure(eve::Diagnostic::error(
44 eve::DiagnosticCode::InvalidArgument, "each edge path segment must be non-empty and axis-aligned",
45 std::to_string(i - 1), {}, "building.edge-path"));
46 }
47 const int length = horizontal ? std::abs(end.x - start.x) : std::abs(end.y - start.y);
48 for (int step = 0; step < length; ++step) {
49 EdgeAddress edge{horizontal ? std::min(start.x, end.x) + step : start.x,
50 vertical ? std::min(start.y, end.y) + step : start.y,
51 horizontal ? EdgeAxis::Horizontal : EdgeAxis::Vertical};
52 const std::string key =
53 std::to_string(edge.x) + ":" + std::to_string(edge.y) + ":" + (horizontal ? "h" : "v");
54 if (!uniqueEdges.insert(key).second) {
55 return eve::Result<EdgePathPreview>::failure(eve::Diagnostic::error(
56 eve::DiagnosticCode::Conflict, "edge path contains the same canonical edge more than once",
57 "edge_path_duplicate", {}, "building.edge-path"));
58 }
59 const char *direction = horizontal ? "north" : "west";
60 std::string reason;
61 if (!canPlaceEdge(world, buildingId, edge.x, edge.y, direction, 0, &reason)) {
62 return eve::Result<EdgePathPreview>::failure(eve::Diagnostic::error(
63 eve::DiagnosticCode::Conflict, "edge path preflight rejected without mutation", reason, {},
64 "building.edge-path"));
65 }
66 preview.edges.push_back(edge);
67 }
68 }
69 return eve::Result<EdgePathPreview>::success(std::move(preview));
70}

72eve::Result<std::vector<CornerAddress>> PlacementSystem::sampleEdgeCubicBezier(
73 const std::vector<EdgeCurvePoint> &controlPoints, int subdivisions) {
74 constexpr int kMaxSubdivisions = 4096;
75 constexpr size_t kMaxRasterEdges = 65536;
76 if (controlPoints.size() != 4 || subdivisions < 2 || subdivisions > kMaxSubdivisions) {
77 return eve::Result<std::vector<CornerAddress>>::failure(eve::Diagnostic::error(
78 eve::DiagnosticCode::InvalidArgument, "cubic edge curve requires four controls and 2..4096 subdivisions",
79 {}, {}, "building.edge-curve"));
80 }
81 for (const EdgeCurvePoint &point : controlPoints) {
82 if (!std::isfinite(point.x) || !std::isfinite(point.y)) {
83 return eve::Result<std::vector<CornerAddress>>::failure(
84 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "edge curve controls must be finite", {},
85 {}, "building.edge-curve"));
86 }
87 }
88
89 auto roundedPoint = [&](int sample) -> eve::Result<CornerAddress> {
90 const double t = static_cast<double>(sample) / static_cast<double>(subdivisions);
91 const double u = 1.0 - t;
92 const double x = u * u * u * controlPoints[0].x + 3.0 * u * u * t * controlPoints[1].x +
93 3.0 * u * t * t * controlPoints[2].x + t * t * t * controlPoints[3].x;
94 const double y = u * u * u * controlPoints[0].y + 3.0 * u * u * t * controlPoints[1].y +
95 3.0 * u * t * t * controlPoints[2].y + t * t * t * controlPoints[3].y;
96 constexpr double kIntMin = static_cast<double>(std::numeric_limits<int>::min()) + 1.0;
97 constexpr double kIntMax = static_cast<double>(std::numeric_limits<int>::max()) - 1.0;
98 if (x < kIntMin || x > kIntMax || y < kIntMin || y > kIntMax) {
99 return eve::Result<CornerAddress>::failure(
100 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "edge curve exceeds grid coordinate range",
101 {}, {}, "building.edge-curve"));
102 }
103 return eve::Result<CornerAddress>::success(
104 {static_cast<int>(std::lround(x)), static_cast<int>(std::lround(y))});
105 };
106
107 auto first = roundedPoint(0);
108 if (!first.ok()) return eve::Result<std::vector<CornerAddress>>::failure(first.status());
109 std::vector<CornerAddress> vertices;
110 vertices.push_back(std::move(first).takeValue());
111 for (int sample = 1; sample <= subdivisions; ++sample) {
112 auto targetResult = roundedPoint(sample);
113 if (!targetResult.ok()) return eve::Result<std::vector<CornerAddress>>::failure(targetResult.status());
114 const CornerAddress target = std::move(targetResult).takeValue();
115 CornerAddress cursor = vertices.back();
116 while (cursor != target) {
117 const int remainingX = target.x - cursor.x;
118 const int remainingY = target.y - cursor.y;
119 if (remainingX != 0 && (remainingY == 0 || std::abs(remainingX) >= std::abs(remainingY)))
120 cursor.x += remainingX > 0 ? 1 : -1;
121 else
122 cursor.y += remainingY > 0 ? 1 : -1;
123 vertices.push_back(cursor);
124 if (vertices.size() - 1 > kMaxRasterEdges) {
125 return eve::Result<std::vector<CornerAddress>>::failure(eve::Diagnostic::error(
126 eve::DiagnosticCode::InvalidArgument, "edge curve rasterization exceeds 65536 unit edges", {}, {},
127 "building.edge-curve"));
128 }
129 }
130 }
131 if (vertices.size() < 2) {
132 return eve::Result<std::vector<CornerAddress>>::failure(eve::Diagnostic::error(
133 eve::DiagnosticCode::InvalidArgument, "edge curve rasterization must cover at least one unit edge", {}, {},
134 "building.edge-curve"));
135 }
136 return eve::Result<std::vector<CornerAddress>>::success(std::move(vertices));
137}

155eve::Result<PlacementSystem::EdgeCurveSurface> PlacementSystem::sampleEdgeCurveSurface(
156 const PlacementWorld &world, const std::string &surfaceName, const std::vector<EdgeCurvePoint> &controlPoints,
157 int subdivisions) {
158 auto validated = sampleEdgeCubicBezier(controlPoints, subdivisions);
159 if (!validated.ok()) return eve::Result<EdgeCurveSurface>::failure(validated.status());
160 if (surfaceName.empty())
161 return eve::Result<EdgeCurveSurface>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
162 "curve surface provider name is required",
163 {}, {}, "building.edge-curve-surface"));
164 float ox = 0.f, oy = 0.f, xx = 0.f, xy = 0.f, yx = 0.f, yy = 0.f;
165 world.cellToWorldPlane(0, 0, ox, oy);
166 world.cellToWorldPlane(1, 0, xx, xy);
167 world.cellToWorldPlane(0, 1, yx, yy);
168 xx -= ox;
169 xy -= oy;
170 yx -= ox;
171 yy -= oy;
172 EdgeCurveSurface result;
173 result.providerName = surfaceName;
174 result.samples.reserve(static_cast<size_t>(subdivisions + 1));
175 for (int index = 0; index <= subdivisions; ++index) {
176 const double t = static_cast<double>(index) / subdivisions;
177 const double u = 1.0 - t;
178 const double gx = u * u * u * controlPoints[0].x + 3.0 * u * u * t * controlPoints[1].x +
179 3.0 * u * t * t * controlPoints[2].x + t * t * t * controlPoints[3].x;
180 const double gy = u * u * u * controlPoints[0].y + 3.0 * u * u * t * controlPoints[1].y +
181 3.0 * u * t * t * controlPoints[2].y + t * t * t * controlPoints[3].y;
182 auto sampled = sampleSurface(world, surfaceName, ox + static_cast<float>(gx) * xx + static_cast<float>(gy) * yx,
183 oy + static_cast<float>(gx) * xy + static_cast<float>(gy) * yy);
184 if (!sampled.ok()) return eve::Result<EdgeCurveSurface>::failure(sampled.status());
185 const PlacementHit &hit = sampled.value();
186 if (index == 0) {
187 result.surfaceId = hit.surfaceId;
188 result.surfaceRevision = hit.surfaceRevision;
189 } else if (hit.surfaceId != result.surfaceId || hit.surfaceRevision != result.surfaceRevision) {
190 return eve::Result<EdgeCurveSurface>::failure(eve::Diagnostic::error(
191 eve::DiagnosticCode::Conflict, "curve crosses a surface identity or revision boundary", hit.surfaceId,
192 {}, "building.edge-curve-surface"));
193 }
194 result.samples.push_back({hit.worldX, hit.worldY, hit.worldZ, hit.normalX, hit.normalY, hit.normalZ});
195 }
196 return eve::Result<EdgeCurveSurface>::success(std::move(result));
197}

231eve::Result<PlacementSystem::EdgePathPlacement> PlacementSystem::commitEdgePath(
232 PlacementWorld *world, EdgePathPreview preview, const std::vector<EdgeCurvePoint> *curveControls,
233 int curveSubdivisions, const EdgeCurveSurface *curveSurface) {
234 const BuildingDefinition *def = BuildingRegistry::find(preview.buildingId);
235 const std::vector<EdgeAddress> &edges = preview.edges;
236 const EdgeCurveGroupId curveGroupId =
237 curveControls ? EdgeCurveGroupId{world->nextEdgeCurveGroupId_++} : EdgeCurveGroupId{};
238
239 EdgePathPlacement result;
240 result.instanceIds.reserve(edges.size());
241 std::vector<BuildingChangeEvent> events;
242 events.reserve(edges.size());
243 for (const EdgeAddress &edge : edges) {
244 PlacedBuilding placed;
245 placed.instanceId = nextInstanceId(*world);
246 placed.buildingId = preview.buildingId;
247 placed.placementKind = "edge";
248 placed.edge = edge;
249 placed.edgeCurveGroupId = curveGroupId;
250 placed.originCellX = edge.x;
251 placed.originCellY = edge.y;
252 placed.level = preview.level;
253 placed.channel = def->channel;
254 placed.tags = def->tags;
255 float ax = 0.f, ay = 0.f, bx = 0.f, by = 0.f;
256 world->cellToWorldPlane(edge.x, edge.y, ax, ay);
257 world->cellToWorldPlane(edge.x + (edge.axis == EdgeAxis::Horizontal ? 1 : 0),
258 edge.y + (edge.axis == EdgeAxis::Vertical ? 1 : 0), bx, by);
259 placed.worldX = (ax + bx) * 0.5f;
260 placed.worldY = (ay + by) * 0.5f;
261 placed.rotationDeg = edge.axis == EdgeAxis::Horizontal ? 0.f : 90.f;
262
263 auto &channels = world->edgeChannels(edge.axis, placed.level);
264 auto &occupancy = channels[def->channel];
265 const size_t required = edge.axis == EdgeAxis::Horizontal ? size_t(world->width_) * size_t(world->height_ + 1)
266 : size_t(world->width_ + 1) * size_t(world->height_);
267 if (occupancy.size() != required) occupancy.assign(required, 0);
268 const size_t index = edge.axis == EdgeAxis::Horizontal
269 ? size_t(edge.y) * size_t(world->width_) + size_t(edge.x)
270 : size_t(edge.y) * size_t(world->width_ + 1) + size_t(edge.x);
271 occupancy[index] = placed.instanceId;
272 world->buildings()[placed.instanceId] = placed;
273 world->instanceOrder_.push_back(placed.instanceId);
274 result.instanceIds.push_back(placed.instanceId);
275
276 BuildingChangeEvent event;
277 event.action = "place";
278 event.worldId = world->getId();
279 event.buildingId = preview.buildingId;
280 event.instanceId = placed.instanceId;
281 event.cellX = edge.x;
282 event.cellY = edge.y;
283 event.level = placed.level;
284 event.rotationDeg = placed.rotationDeg;
285 event.worldX = placed.worldX;
286 event.worldY = placed.worldY;
287 event.channel = placed.channel;
288 events.push_back(std::move(event));
289 }
290 if (curveControls) {
291 EdgeCurveGroup group;
292 group.id = curveGroupId;
293 group.buildingId = preview.buildingId;
294 group.level = preview.level;
295 group.controlPoints = *curveControls;
296 group.subdivisions = curveSubdivisions;
297 group.instanceIds = result.instanceIds;
298 if (curveSurface) {
299 group.surfaceProviderName = curveSurface->providerName;
300 group.surfaceId = curveSurface->surfaceId;
301 group.surfaceRevision = curveSurface->surfaceRevision;
302 group.surfaceSamples = curveSurface->samples;
303 }
304 world->edgeCurveGroups_.emplace(group.id.value, std::move(group));
305 }
306 if (world->publishEvents_)
307 for (BuildingChangeEvent &event : events) emit(std::move(event));
308 return eve::Result<EdgePathPlacement>::success(std::move(result));
309}

311eve::Result<PlacementSystem::AreaPreview> PlacementSystem::previewArea(PlacementWorld *world,
312 const std::string &buildingId,
313 const std::vector<std::pair<int, int>> &anchors,
314 float rotationDeg) {
315 if (!world || anchors.empty()) {
316 return eve::Result<AreaPreview>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
317 "area preview requires a world and anchors",
318 buildingId, {}, "building.area"));
319 }
320 const BuildingDefinition *def = BuildingRegistry::find(buildingId);
321 if (!def || def->placementKind == "edge") {
322 return eve::Result<AreaPreview>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
323 "area tools require a cell building definition",
324 buildingId, {}, "building.area"));
325 }
326 AreaPreview preview;
327 preview.buildingId = buildingId;
328 preview.level = world->getActiveLevel();
329 preview.rotationDeg = normalizeRotation(buildingId, rotationDeg);
330 std::unordered_set<uint64_t> reserved;
331 for (const auto &[x, y] : anchors) {
332 AreaCellPreview cell{x, y, false, {}};
333 if (!canPlace(world, buildingId, x, y, preview.rotationDeg, 0, &cell.reason)) {
334 ++preview.rejectedCount;
335 preview.cells.push_back(std::move(cell));
336 continue;
337 }
338 bool overlaps = false;
339 std::vector<uint64_t> footprint;
340 foreachFootprintCell(*def, x, y, preview.rotationDeg, [&](int cx, int cy) {
341 const uint64_t key = (uint64_t(uint32_t(cx)) << 32u) | uint32_t(cy);
342 footprint.push_back(key);
343 if (reserved.count(key) != 0) overlaps = true;
344 return true;
345 });
346 if (overlaps) {
347 cell.reason = "area_candidate_overlap";
348 ++preview.rejectedCount;
349 } else {
350 cell.accepted = true;
351 ++preview.acceptedCount;
352 reserved.insert(footprint.begin(), footprint.end());
353 }
354 preview.cells.push_back(std::move(cell));
355 }
356 return eve::Result<AreaPreview>::success(std::move(preview));
357}

405 });
406 return previewArea(world, buildingId, anchors, rotationDeg);

409eve::Result<PlacementSystem::PatternPreview> PlacementSystem::previewPattern(PlacementWorld *world,
410 const std::string &buildingId,
411 const PatternRequest &request) {
412 const auto failure = [&](std::string message) {
413 return eve::Result<PatternPreview>::failure(
414 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, message, buildingId, {}, "building.pattern"));
415 };
416 const auto corners = [&]() -> eve::Result<std::vector<CornerAddress>> {
417 std::vector<CornerAddress> result;
418 result.reserve(request.points.size());
419 for (const EdgeCurvePoint &point : request.points) {
420 if (!std::isfinite(point.x) || !std::isfinite(point.y) ||
421 std::abs(point.x - std::round(point.x)) > 0.0001f || std::abs(point.y - std::round(point.y)) > 0.0001f)
422 return eve::Result<std::vector<CornerAddress>>::failure(
423 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
424 "discrete pattern points must be finite integer grid vertices", buildingId,
425 {}, "building.pattern"));
426 result.push_back({static_cast<int>(std::lround(point.x)), static_cast<int>(std::lround(point.y))});
427 }
428 return eve::Result<std::vector<CornerAddress>>::success(std::move(result));
429 };
430
431 PatternPreview preview;
432 preview.request = request;
433 switch (request.kind) {
434 case PatternKind::EdgeLine:
435 case PatternKind::EdgePath: {
436 if ((request.kind == PatternKind::EdgeLine && request.points.size() != 2) ||
437 (request.kind == PatternKind::EdgePath && request.points.size() < 2))
438 return failure(request.kind == PatternKind::EdgeLine
439 ? "edge-line pattern requires exactly two points"
440 : "edge-path pattern requires at least two points");
441 if (!request.surfaceName.empty())
442 return failure("custom surfaces are currently supported only by cubic curves");
443 auto vertices = corners();
444 if (!vertices.ok()) return eve::Result<PatternPreview>::failure(vertices.status());
445 auto edge = previewEdgePath(world, buildingId, vertices.value());
446 if (!edge.ok()) return eve::Result<PatternPreview>::failure(edge.status());
447 preview.edge = std::move(edge).takeValue();
448 break;
449 }
450 case PatternKind::EdgeCubicBezier: {
451 if (request.points.size() != 4) return failure("cubic edge pattern requires exactly four control points");
452 auto edge = previewEdgeCubicBezier(world, buildingId, request.points, request.subdivisions);
453 if (!edge.ok()) return eve::Result<PatternPreview>::failure(edge.status());
454 if (!request.surfaceName.empty()) {
455 auto surface =
456 sampleEdgeCurveSurface(*world, request.surfaceName, request.points, request.subdivisions);
457 if (!surface.ok()) return eve::Result<PatternPreview>::failure(surface.status());
458 }
459 preview.edge = std::move(edge).takeValue();
460 break;
461 }
462 case PatternKind::RectangleFill:
463 case PatternKind::RectangleOutline: {
464 if (request.points.size() != 2) return failure("rectangle pattern requires exactly two corner points");
465 if (!request.surfaceName.empty()) return failure("area patterns do not accept a custom surface name");
466 auto vertices = corners();
467 if (!vertices.ok()) return eve::Result<PatternPreview>::failure(vertices.status());
468 const CornerAddress &a = vertices.value()[0];
469 const CornerAddress &b = vertices.value()[1];
470 auto area = request.kind == PatternKind::RectangleFill
471 ? previewRectangle(world, buildingId, a.x, a.y, b.x, b.y, request.rotationDeg)
472 : previewRectangleOutline(world, buildingId, a.x, a.y, b.x, b.y, request.rotationDeg);
473 if (!area.ok()) return eve::Result<PatternPreview>::failure(area.status());
474 preview.area = std::move(area).takeValue();
475 break;
476 }
477 case PatternKind::CircleBrush: {
478 if (request.points.size() != 1) return failure("circle-brush pattern requires exactly one center point");
479 if (!request.surfaceName.empty()) return failure("area patterns do not accept a custom surface name");
480 auto vertices = corners();
481 if (!vertices.ok()) return eve::Result<PatternPreview>::failure(vertices.status());
482 auto area = previewBrush(world, buildingId, vertices.value()[0].x, vertices.value()[0].y, request.radius,
483 request.rotationDeg);
484 if (!area.ok()) return eve::Result<PatternPreview>::failure(area.status());
485 preview.area = std::move(area).takeValue();
486 break;
487 }
488 }
489 return eve::Result<PatternPreview>::success(std::move(preview));
490}

492eve::Result<PlacementSystem::PatternPlacement> PlacementSystem::placePattern(PlacementWorld *world,
493 const std::string &buildingId,
494 const PatternRequest &request) {
495 auto expanded = previewPattern(world, buildingId, request);
496 if (!expanded.ok()) return eve::Result<PatternPlacement>::failure(expanded.status());
497 PatternPlacement result;
498 result.preview = std::move(expanded).takeValue();
499 if (request.kind == PatternKind::EdgeLine || request.kind == PatternKind::EdgePath) {
500 auto placed = commitEdgePath(world, result.preview.edge);
501 if (!placed.ok()) return eve::Result<PatternPlacement>::failure(placed.status());
502 result.instanceIds = std::move(placed).takeValue().instanceIds;
503 } else if (request.kind == PatternKind::EdgeCubicBezier) {
504 auto placed = [&]() -> eve::Result<EdgePathPlacement> {
505 if (request.surfaceName.empty())
506 return commitEdgePath(world, result.preview.edge, &result.preview.request.points, request.subdivisions);
507 auto surface = sampleEdgeCurveSurface(*world, request.surfaceName, result.preview.request.points,
508 request.subdivisions);
509 if (!surface.ok()) return eve::Result<EdgePathPlacement>::failure(surface.status());
510 const EdgeCurveSurface frames = std::move(surface).takeValue();
511 return commitEdgePath(world, result.preview.edge, &result.preview.request.points, request.subdivisions,
512 &frames);
513 }();
514 if (!placed.ok()) return eve::Result<PatternPlacement>::failure(placed.status());
515 result.instanceIds = std::move(placed).takeValue().instanceIds;
516 } else {
517 auto placed = commitArea(world, result.preview.area);
518 if (!placed.ok()) return eve::Result<PatternPlacement>::failure(placed.status());
519 result.instanceIds = std::move(placed).takeValue().instanceIds;
520 }
521 return eve::Result<PatternPlacement>::success(std::move(result));
522}

524eve::Result<PlacementSystem::AreaPlacement> PlacementSystem::commitArea(PlacementWorld *world, AreaPreview preview) {
525 if (!world || preview.rejectedCount != 0 || preview.cells.empty()) {
526 return eve::Result<AreaPlacement>::failure(
527 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "area commit requires a fully accepted preview",
528 std::to_string(preview.rejectedCount), {}, "building.area"));
529 }
530 const BuildingDefinition *def = BuildingRegistry::find(preview.buildingId);
531 if (!def) {
532 return eve::Result<AreaPlacement>::failure(eve::Diagnostic::error(
533 eve::DiagnosticCode::NotFound, "area definition was not found", preview.buildingId, {}, "building.area"));
534 }
535 AreaPlacement placement;
536 placement.preview = std::move(preview);
537 std::vector<BuildingChangeEvent> events;
538 for (const AreaCellPreview &cell : placement.preview.cells) {
539 PlacedBuilding pb;
540 pb.instanceId = nextInstanceId(*world);
541 pb.buildingId = placement.preview.buildingId;
542 pb.originCellX = cell.cellX;
543 pb.originCellY = cell.cellY;
544 pb.level = placement.preview.level;
545 pb.rotationDeg = placement.preview.rotationDeg;
546 pb.channel = def->channel;
547 pb.tags = def->tags;
548 world->cellToWorldPlane(cell.cellX, cell.cellY, pb.worldX, pb.worldY);
549 PlacementQuery supportQuery;
550 supportQuery.cellX = pb.originCellX;
551 supportQuery.cellY = pb.originCellY;
552 supportQuery.level = pb.level;
553 supportQuery.rotationDeg = pb.rotationDeg;
554 pb.supportInstanceIds = collectStructuralSupports(*world, *def, supportQuery);
555 writeOccupancy(*world, *def, pb, pb.instanceId);
556 world->buildings()[pb.instanceId] = pb;
557 world->instanceOrder_.push_back(pb.instanceId);
558 placement.instanceIds.push_back(pb.instanceId);
559 BuildingChangeEvent event;
560 event.action = "place";
561 event.worldId = world->getId();
562 event.buildingId = pb.buildingId;
563 event.instanceId = pb.instanceId;
564 event.cellX = pb.originCellX;
565 event.cellY = pb.originCellY;
566 event.rotationDeg = pb.rotationDeg;
567 event.worldX = pb.worldX;
568 event.worldY = pb.worldY;
569 event.channel = pb.channel;
570 events.push_back(std::move(event));
571 }
572 if (world->publishEvents_)
573 for (BuildingChangeEvent &event : events) emit(std::move(event));
574 return eve::Result<AreaPlacement>::success(std::move(placement));
575}
