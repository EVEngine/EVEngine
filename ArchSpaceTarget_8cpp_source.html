22bool readNumber(const EditorValue* value, double& out) {
23 if (!value) return false;
24 if (const auto* real = value->getIf<double>()) {
25 out = *real;
26 return std::isfinite(out);
27 }
28 if (const auto* integer = value->getIf<std::int64_t>()) {
29 out = static_cast<double>(*integer);
30 return true;
31 }
32 return false;

80EditorResult<archspace::Node> parseNode(const EditorValue& value) {
81 archspace::Node node;
82 const auto* id = field(value, "id");
83 const auto* kind = field(value, "kind");
84 const auto* parentId = field(value, "parentId");
85 const auto* name = field(value, "name");
86 const auto* ids = id ? id->getIf<std::string>() : nullptr;
87 const auto* kinds = kind ? kind->getIf<std::string>() : nullptr;
88 const auto* parents = parentId ? parentId->getIf<std::string>() : nullptr;
89 const auto* names = name ? name->getIf<std::string>() : nullptr;
90 if (!ids || ids->empty() || !kinds || !parents || !names)
91 return fail<archspace::Node>(EditorStatus::Rejected, "editor.archspace.node",
92 "ArchSpace node identity is invalid");
93 auto parsedKind = archspace::parseNodeKind(*kinds);
94 if (!parsedKind.ok())
95 return fail<archspace::Node>(EditorStatus::Rejected, "editor.archspace.kind", "ArchSpace node kind is unknown");
96 node.kind = parsedKind.value();
97 node.id = *ids;
98 node.parentId = *parents;
99 node.name = *names;
100 if (const auto* children = field(value, "children")) {
101 const auto* array = children->getIf<EditorValue::Array>();
102 if (!array)
103 return fail<archspace::Node>(EditorStatus::Rejected, "editor.archspace.children",
104 "ArchSpace children must be an array");
105 for (const auto& child : *array) {
106 const auto* text = child.getIf<std::string>();
107 if (!text)
108 return fail<archspace::Node>(EditorStatus::Rejected, "editor.archspace.children",
109 "ArchSpace child id must be a string");
110 node.children.push_back(*text);
111 }
112 }
113 readNumber(field(value, "elevation"), node.elevation);
114 readNumber(field(value, "height"), node.height);
115 if (const auto* start = field(value, "start"); start && !parseVec2(*start, node.start))
116 return fail<archspace::Node>(EditorStatus::Rejected, "editor.archspace.wall",
117 "ArchSpace wall start is invalid");
118 if (const auto* end = field(value, "end"); end && !parseVec2(*end, node.end))
119 return fail<archspace::Node>(EditorStatus::Rejected, "editor.archspace.wall", "ArchSpace wall end is invalid");
120 readNumber(field(value, "thickness"), node.thickness);
121 if (const auto* polygon = field(value, "polygon")) {
122 const auto* array = polygon->getIf<EditorValue::Array>();
123 if (!array)
124 return fail<archspace::Node>(EditorStatus::Rejected, "editor.archspace.polygon",
125 "ArchSpace polygon must be an array");
126 for (const auto& point : *array) {
127 archspace::Vec2 v;
128 if (!parseVec2(point, v))
129 return fail<archspace::Node>(EditorStatus::Rejected, "editor.archspace.polygon",
130 "ArchSpace polygon point is invalid");
131 node.polygon.push_back(v);
132 }
133 }
134 readNumber(field(value, "slabThickness"), node.slabThickness);
135 if (const auto* position = field(value, "position"); position && !parseVec3(*position, node.position))
136 return fail<archspace::Node>(EditorStatus::Rejected, "editor.archspace.item",
137 "ArchSpace item position is invalid");
138 readNumber(field(value, "yawDegrees"), node.yawDegrees);
139 if (const auto* catalog = field(value, "catalogId")) {
140 const auto* text = catalog->getIf<std::string>();
141 if (!text)
142 return fail<archspace::Node>(EditorStatus::Rejected, "editor.archspace.item",
143 "ArchSpace catalog id must be a string");
144 node.catalogId = *text;
145 }
146 if (const auto* opening = field(value, "openingKind")) {
147 const auto* text = opening->getIf<std::string>();
148 if (!text)
149 return fail<archspace::Node>(EditorStatus::Rejected, "editor.archspace.opening",
150 "ArchSpace opening kind must be a string");
151 auto kind = archspace::parseOpeningKind(*text);
152 if (!kind.ok())
153 return fail<archspace::Node>(EditorStatus::Rejected, "editor.archspace.opening",
154 "ArchSpace opening kind is invalid");
155 node.openingKind = kind.value();
156 }
157 readNumber(field(value, "t"), node.t);
158 readNumber(field(value, "width"), node.width);
159 readNumber(field(value, "openingHeight"), node.openingHeight);
160 readNumber(field(value, "sill"), node.sill);
161 return eve::editing::applied<archspace::Node>(std::move(node));
162}

228PropertySchema ArchSpaceDocumentTarget::schema(const SelectionSnapshot& selection) const {
229 PropertySchema schema;
230 const archspace::Node* node = selectedNode(selection);
231 if (!node) return schema;
232 schema.typeId = std::string("archspace.") + archspace::nodeKindName(node->kind);
233 schema.properties.push_back(prop("name", PropertyType::String, node->name, "common"));
234 switch (node->kind) {
235 case archspace::NodeKind::Level:
236 schema.properties.push_back(prop("elevation", PropertyType::Float, node->elevation, "level"));
237 schema.properties.push_back(prop("height", PropertyType::Float, node->height, "level", 0.1, 1000));
238 break;
239 case archspace::NodeKind::Wall:
240 schema.properties.push_back(prop("thickness", PropertyType::Float, node->thickness, "wall", 0.01, 10));
241 schema.properties.push_back(prop("height", PropertyType::Float, node->height, "wall", 0.1, 1000));
242 break;
243 case archspace::NodeKind::Item:
244 schema.properties.push_back(prop("catalogId", PropertyType::String, node->catalogId, "item"));
245 schema.properties.push_back(prop("yawDegrees", PropertyType::Float, node->yawDegrees, "item"));
246 break;
247 case archspace::NodeKind::Opening:
248 schema.properties.push_back(prop("t", PropertyType::Float, node->t, "opening", 0, 1));
249 schema.properties.push_back(prop("width", PropertyType::Float, node->width, "opening", 0.01, 20));
250 schema.properties.push_back(
251 prop("openingHeight", PropertyType::Float, node->openingHeight, "opening", 0.01, 20));
252 schema.properties.push_back(prop("sill", PropertyType::Float, node->sill, "opening", 0, 20));
253 break;
254 default: break;
255 }
256 return schema;
257}

297EditorResult<void> ArchSpaceDocumentTarget::installContent(const EditorValue& content) {
298 const auto* rootIdValue = field(content, "rootId");
299 const auto* nodesValue = field(content, "nodes");
300 const auto* rootId = rootIdValue ? rootIdValue->getIf<std::string>() : nullptr;
301 const auto* nodes = nodesValue ? nodesValue->getIf<EditorValue::Array>() : nullptr;
302 if (!rootId || !nodes || !content.isWithinLimits(12, 250000, 32 * 1024 * 1024))
303 return fail<void>(EditorStatus::Rejected, "editor.archspace.payload",
304 "ArchSpace document payload is invalid or exceeds limits");
305 if (nodes->size() > 100000)
306 return fail<void>(EditorStatus::Rejected, "editor.archspace.budget", "ArchSpace document exceeds 100000 nodes");
307
308 archspace::Document candidate;
309 if (rootId->empty() && nodes->empty()) {
310 document_ = std::move(candidate);
311 return eve::editing::applied<void>();
312 }
313 // Insert sites first, then remaining nodes in dependency-friendly passes.
314 std::vector<archspace::Node> parsed;
315 parsed.reserve(nodes->size());
316 for (const auto& value : *nodes) {
317 auto node = parseNode(value);
318 if (!node.ok())
319 return fail<void>(EditorStatus::Rejected, "editor.archspace.node",
320 "ArchSpace document contains an invalid node");
321 parsed.push_back(std::move(node.value()));
322 }
323 std::sort(parsed.begin(), parsed.end(), [](const archspace::Node& a, const archspace::Node& b) {
324 return static_cast<int>(a.kind) < static_cast<int>(b.kind);
325 });
326 for (auto& node : parsed) {
327 node.children.clear(); // re-linked by insert()
328 auto inserted = candidate.insert(std::move(node));
329 if (!inserted.ok())
330 return fail<void>(EditorStatus::Rejected, "editor.archspace.insert",
331 "ArchSpace document failed hierarchical insert");
332 }
333 if (candidate.rootId() != *rootId)
334 return fail<void>(EditorStatus::Rejected, "editor.archspace.root",
335 "ArchSpace root id does not match document content");
336 if (!candidate.diagnostics().empty())
337 return fail<void>(EditorStatus::Rejected, "editor.archspace.invalid", "ArchSpace document validation failed");
338 document_ = std::move(candidate);
339 return eve::editing::applied<void>();
340}

357EditorResult<DomainOperation> ArchSpaceDocumentTarget::makeSet(const SelectionSnapshot& selection,
358 const PropertyPath& path, const EditorValue& value,
359 PropertySetMode mode) const {
360 if (mode == PropertySetMode::Reset) return makeReset(selection, path);
361 const auto descriptor = schema(selection).find(path);
362 const archspace::Node* node = selectedNode(selection);
363 if (!descriptor || !node || mode != PropertySetMode::Absolute)
364 return fail<DomainOperation>(EditorStatus::Rejected, "editor.archspace.set",
365 "ArchSpace property requires a matching absolute edit");
366 auto checked = validatePropertyValue(*descriptor, value);
367 if (!checked.ok())
368 return fail<DomainOperation>(EditorStatus::Rejected, "editor.archspace.value",
369 "ArchSpace property value is invalid");
370 archspace::Document candidate = document_;
371 archspace::Node* editable = candidate.findMutable(node->id);
372 if (!editable)
373 return fail<DomainOperation>(EditorStatus::NotFound, "editor.archspace.node", "ArchSpace node is missing");
374 if (path == PropertyPath("name")) {
375 const auto* text = value.getIf<std::string>();
376 if (!text) return fail<DomainOperation>(EditorStatus::Rejected, "editor.archspace.name", "Name must be string");
377 editable->name = *text;
378 } else if (path == PropertyPath("catalogId")) {
379 const auto* text = value.getIf<std::string>();
380 if (!text)
381 return fail<DomainOperation>(EditorStatus::Rejected, "editor.archspace.catalog",
382 "Catalog id must be string");
383 editable->catalogId = *text;
384 } else {
385 double number = 0.0;
386 if (!readNumber(&value, number))
387 return fail<DomainOperation>(EditorStatus::Rejected, "editor.archspace.number",
388 "ArchSpace property requires a number");
389 if (path == PropertyPath("elevation"))
390 editable->elevation = number;
391 else if (path == PropertyPath("height"))
392 editable->height = number;
393 else if (path == PropertyPath("thickness"))
394 editable->thickness = number;
395 else if (path == PropertyPath("yawDegrees"))
396 editable->yawDegrees = number;
397 else if (path == PropertyPath("t"))
398 editable->t = number;
399 else if (path == PropertyPath("width"))
400 editable->width = number;
401 else if (path == PropertyPath("openingHeight"))
402 editable->openingHeight = number;
403 else if (path == PropertyPath("sill"))
404 editable->sill = number;
405 else
406 return fail<DomainOperation>(EditorStatus::Unsupported, "editor.archspace.property",
407 "Unsupported ArchSpace property");
408 }
409 auto replaced = candidate.replace(*editable);
410 if (!replaced.ok())
411 return fail<DomainOperation>(EditorStatus::Rejected, "editor.archspace.invalid",
412 "ArchSpace property update failed validation");
413 ArchSpaceDocumentTarget staged(id_);
414 staged.document_ = std::move(candidate);
415 if (hasErrors(staged.validate()))
416 return fail<DomainOperation>(EditorStatus::Rejected, "editor.archspace.invalid",
417 "ArchSpace property update failed validation");
418 return replacement(staged.contentValue(), path.value());
419}

512EditorResult<DomainOperation> ArchSpaceDocumentTarget::makeDeleteNode(const ObjectId& id) const {
513 archspace::Document candidate = document_;
514 auto erased = candidate.eraseCascade(id.value());
515 if (!erased.ok())
516 return fail<DomainOperation>(EditorStatus::NotFound, "editor.archspace.delete", "ArchSpace node was not found");
517 ArchSpaceDocumentTarget staged(id_);
518 staged.document_ = std::move(candidate);
519 return replacement(staged.contentValue());
520}

522EditorResult<void> ArchSpaceDocumentTarget::applyDomainOperation(const DomainOperation& operation) {
523 if (operation.target != TargetId(id_) || operation.type != "archspace.document.replace.v1")
524 return fail<void>(EditorStatus::Rejected, "editor.archspace.operation", "ArchSpace operation mismatch");
525 ArchSpaceDocumentTarget candidate(id_);
526 auto installed = candidate.installContent(operation.payload);
527 if (!installed.ok()) return installed;
528 if (hasErrors(candidate.validate()))
529 return fail<void>(EditorStatus::Rejected, "editor.archspace.invalid", "ArchSpace document validation failed");
530 document_ = std::move(candidate.document_);
531 ++revision_;
532 dirty_.include(0, 0);
533 return eve::editing::applied<void>();
534}

548EditorResult<void> ArchSpaceDocumentTarget::loadSnapshot(const EditorValue& snapshot) {
549 const auto* schemaIdValue = field(snapshot, "schemaId");
550 const auto* versionValue = field(snapshot, "schemaVersion");
551 const auto* content = field(snapshot, "content");
552 const auto* schemaId = schemaIdValue ? schemaIdValue->getIf<std::string>() : nullptr;
553 const auto* version = versionValue ? versionValue->getIf<std::int64_t>() : nullptr;
554 if (!schemaId || *schemaId != "eve.archspace.document" || !version || *version != 1 || !content)
555 return fail<void>(EditorStatus::Unsupported, "editor.archspace.snapshot", "Unsupported ArchSpace snapshot");
556 DomainOperation op;
557 op.target = TargetId(id_);
558 op.type = "archspace.document.replace.v1";
559 op.payload = *content;
560 auto result = applyDomainOperation(op);
561 if (result.ok()) dirty_.clear();
562 return result;
563}

565EditorResult<EditorGizmoSnapshot> ArchSpaceDocumentTarget::gizmo() const {
566 EditorGizmoSnapshot snapshot;
567 snapshot.status = EditorStatus::Applied;
568 snapshot.target = id_;
569 snapshot.targetRevision = revision_;
570 for (const auto& [id, node] : document_.nodes()) {
571 (void)id;
572 if (node.kind == archspace::NodeKind::Wall) {
573 EditorGizmoPrimitive primitive;
574 primitive.id = node.id;
575 primitive.kind = "segment";
576 primitive.position = {node.start.x,
577 document_.find(node.parentId) ? document_.find(node.parentId)->elevation : 0.0,
578 node.start.z};
579 primitive.direction = {node.end.x - node.start.x, 0.0, node.end.z - node.start.z};
580 primitive.length = std::sqrt(primitive.direction[0] * primitive.direction[0] +
581 primitive.direction[2] * primitive.direction[2]);
582 primitive.color = {0.2, 0.7, 1.0, 0.9};
583 snapshot.primitives.push_back(std::move(primitive));
584 } else if (node.kind == archspace::NodeKind::Zone && !node.polygon.empty()) {
585 double minX = node.polygon.front().x, maxX = minX, minZ = node.polygon.front().z, maxZ = minZ;
586 for (const auto& p : node.polygon) {
587 minX = std::min(minX, p.x);
588 maxX = std::max(maxX, p.x);
589 minZ = std::min(minZ, p.z);
590 maxZ = std::max(maxZ, p.z);
591 }
592 EditorGizmoPrimitive primitive;
593 primitive.id = node.id;
594 primitive.kind = "area";
595 primitive.dashed = true;
596 primitive.position = {(minX + maxX) * 0.5, 0.02, (minZ + maxZ) * 0.5};
597 primitive.size = {maxX - minX, 0.01, maxZ - minZ};
598 primitive.color = {0.8, 0.2, 1.0, 0.25};
599 snapshot.primitives.push_back(std::move(primitive));
600 } else if (node.kind == archspace::NodeKind::Item) {
601 EditorGizmoPrimitive primitive;
602 primitive.id = node.id;
603 primitive.kind = "point";
604 primitive.position = {node.position.x, node.position.y, node.position.z};
605 primitive.radius = 0.15;
606 primitive.color = {1.0, 0.8, 0.2, 0.9};
607 primitive.text = node.catalogId;
608 snapshot.primitives.push_back(std::move(primitive));
609 } else if (node.kind == archspace::NodeKind::Opening) {
610 const archspace::Node* wall = document_.find(node.parentId);
611 if (!wall) continue;
612 EditorGizmoPrimitive primitive;
613 primitive.id = node.id;
614 primitive.kind = "point";
615 primitive.position = {wall->start.x + (wall->end.x - wall->start.x) * node.t,
616 node.sill + node.openingHeight * 0.5,
617 wall->start.z + (wall->end.z - wall->start.z) * node.t};
618 primitive.radius = 0.12;
619 primitive.color = {0.2, 1.0, 0.4, 0.9};
620 primitive.text = archspace::openingKindName(node.openingKind);
621 snapshot.primitives.push_back(std::move(primitive));
622 }
623 }
624 return eve::editing::applied<EditorGizmoSnapshot>(std::move(snapshot));
625}
