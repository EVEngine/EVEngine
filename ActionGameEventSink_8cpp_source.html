35Result<void> ActionGameEventSink::emit(const ActionGameplayEventBinding& binding,
36 const ActionNotifyContext& context) {
37 if (!log_) return failure(DiagnosticCode::NotFound, "game event log is unavailable", "log");
38 if (!resolver_) return failure(DiagnosticCode::NotFound, "action subject resolver is unavailable", "resolver");
39
40 std::optional<SubjectRef> source;
41 if (context.source) {
42 auto resolved = resolver_(*context.source);
43 if (!resolved) return Result<void>::failure(resolved.status());
44 source = resolved.value();
45 }
46 SubjectRef subject = SubjectRef::nil();
47 if (binding.targetIndex) {
48 if (*binding.targetIndex >= context.targets.size())
49 return failure(DiagnosticCode::NotFound, "gameplay event target index is unavailable", "targetIndex");
50 auto resolved = resolver_(context.targets[*binding.targetIndex]);
51 if (!resolved) return Result<void>::failure(resolved.status());
52 subject = resolved.value();
53 } else if (source) {
54 subject = *source;
55 } else {
56 return failure(DiagnosticCode::NotFound, "gameplay event has no source or target subject", "subject");
57 }
58
59 auto payload = binding.data.toJson();
60 if (!payload) return Result<void>::failure(payload.status());
61 auto schema = LogicalId::parse("action:gameplay-event-v1");
62 if (!schema) return failure(DiagnosticCode::Failed, "built-in gameplay event schema is invalid", "schemaId");
63
64 eve::game_event::GameEvent envelope;
65 envelope.type = binding.tag;
66 envelope.source = source ? source->format() : std::string{};
67 envelope.subject = subject.format();
68 envelope.schemaId = *schema;
69 envelope.schemaVersion = SchemaVersion(1);
70 envelope.tick = context.tick;
71 envelope.payload = std::move(payload).takeValue();
72 auto appended = log_->append(std::move(envelope));
73 if (!appended) return Result<void>::failure(appended.status());
74 lastSequence_ = appended.value();
75 return Result<void>::success(Status::success(StatusCode::Applied));
76}
