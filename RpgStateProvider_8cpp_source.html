37bool captureRpgState(StateValue& out) {
38 StateValue actors = StateValue::array();
39 for (RPGActor* actor : RPGActor::liveActors()) {
40 StateValue entry = StateValue::object();
41
42 StateValue skills = StateValue::object();
43 for (const auto& kv : actor->skills()->known) {
44 skills.set(kv.first, StateValue::number(kv.second.cooldownRemaining));
45 }
46 entry.set("skills", std::move(skills));
47
48 StateValue casting = StateValue::object();
49 const CastingState& c = actor->skills()->casting;
50 casting.set("active", StateValue::boolean(c.active));
51 casting.set("skillId", StateValue::string(c.skillId));
52 casting.set("remaining", StateValue::number(c.remaining));
53 casting.set("totalCastTime", StateValue::number(c.totalCastTime));
54 entry.set("casting", std::move(casting));
55
56 actors.pushBack(std::move(entry));
57 }
58 out = std::move(actors);
59 return true;
60}

62bool restoreRpgState(const StateValue& in, std::string* err) {
63 if (!in.isArray()) {
64 if (err) *err = "rpg: expected array of actors";
65 return false;
66 }
67 const std::vector<RPGActor*>& actors = RPGActor::liveActors();
68 const size_t n = std::min(in.arraySize(), actors.size());
69 for (size_t i = 0; i < n; ++i) {
70 const StateValue& entry = in.at(i);
71 if (!entry.isObject()) {
72 if (err) *err = "rpg: actor state is not an object";
73 return false;
74 }
75 RPGActor* actor = actors[i];
76
77 if (const StateValue* skills = entry.find("skills"); skills && skills->isObject()) {
78 for (const auto& key : skills->keys()) {
79 const StateValue* v = skills->find(key);
80 if (!v || !(v->isInt() || v->isFloat())) continue;
81 if (SkillSystem::knows(actor, key)) {
82 SkillSystem::setCooldownRemaining(actor, key, static_cast<float>(toNumber(*v)));
83 }
84 }
85 }
86
87 CastingState& casting = actor->skills()->casting;
88 casting = CastingState();
89 if (const StateValue* c = entry.find("casting"); c && c->isObject()) {
90 const StateValue* active = c->find("active");
91 if (active && active->isBool() && active->asBool()) {
92 const StateValue* skillId = c->find("skillId");
93 const std::string id = skillId && skillId->isString() ? skillId->asString() : "";
94 if (!id.empty() && SkillSystem::knows(actor, id)) {
95 casting.active = true;
96 casting.skillId = id;
97 casting.remaining = 0.f;
98 casting.totalCastTime = 0.f;
99 if (const StateValue* r = c->find("remaining"); r && (r->isInt() || r->isFloat()))
100 casting.remaining = static_cast<float>(toNumber(*r));
101 if (const StateValue* t = c->find("totalCastTime"); t && (t->isInt() || t->isFloat()))
102 casting.totalCastTime = static_cast<float>(toNumber(*t));
103 casting.target = nullptr; // pointers are not serializable
104 }
105 }
106 }
107 }
108 return true;
109}
