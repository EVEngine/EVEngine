38eve::Result<action::ActionDefinition> SkillActionAdapter::makeDefinition(const SkillDefinition& skill) {
39 auto id = skillActionId(skill);
40 if (!id) return eve::Result<action::ActionDefinition>::failure(id.status());
41 auto windup = seconds(skill.castTime, "skill.castTime");
42 if (!windup) return eve::Result<action::ActionDefinition>::failure(windup.status());
43 auto cooldown = seconds(skill.cooldown, "skill.cooldown");
44 if (!cooldown) return eve::Result<action::ActionDefinition>::failure(cooldown.status());
45
46 action::ActionDefinition definition;
47 definition.id = std::move(id).takeValue();
48 definition.timing.windup = std::move(windup).takeValue();
49 definition.condition = skill.castCondition;
50 definition.targetingMode =
51 skill.targetType == "self" ? action::TargetingMode::None : action::TargetingMode::Explicit;
52 definition.cost = skill.cost;
53 definition.effectIds = skill.grantedEffects;
54 // A skill is an Active operation even when it currently has no effect id;
55 // the injected executor may emit a settlement/event for that operation.
56 definition.activeExecutionRequired = true;
57 definition.metadata.emplace("cooldown", eve::Value(cooldown.value().seconds()));
58 definition.metadata.emplace("targetType", eve::Value(skill.targetType));
59 definition.metadata.emplace("skillId", eve::Value(skill.id));
60
61 auto valid = definition.validate();
62 if (!valid) return eve::Result<action::ActionDefinition>::failure(valid.status());
63 return eve::Result<action::ActionDefinition>::success(std::move(definition));
64}

66eve::Result<action::ActionRequest> SkillActionAdapter::makeRequest(const SkillDefinition& skill,
67 std::optional<ecs::EntityHandle> source,
68 std::optional<ecs::EntityHandle> target,
69 eve::SimulationTick requestedTick) {
70 auto definition = makeDefinition(skill);
71 if (!definition) return eve::Result<action::ActionRequest>::failure(definition.status());
72 auto checkedDefinition = std::move(definition).takeValue();
73
74 if (checkedDefinition.targetingMode == action::TargetingMode::None && target)
75 return failure<action::ActionRequest>(eve::DiagnosticCode::InvalidArgument,
76 "self skill request cannot carry an explicit target", "target");
77 if (checkedDefinition.targetingMode == action::TargetingMode::Explicit && !target)
78 return failure<action::ActionRequest>(eve::DiagnosticCode::PreconditionViolation,
79 "non-self skill request requires an explicit target", "target");
80
81 action::ActionRequest request;
82 request.actionId = checkedDefinition.id;
83 request.source = std::move(source);
84 request.requestedTick = requestedTick;
85 if (target) request.targetEntities.push_back(*target);
86 auto valid = request.validate(checkedDefinition);
87 if (!valid) return eve::Result<action::ActionRequest>::failure(valid.status());
88 return eve::Result<action::ActionRequest>::success(std::move(request));
89}
