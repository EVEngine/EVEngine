23std::unique_ptr<IEditCommand> AddScalarFieldOperation::createCommand(
24 IEditableTarget *target, const BrushSampleBuffer &samples, float strength) const {
25 auto *field = target ? target->query<IScalarFieldTarget>() : nullptr;
26 if (!field) return nullptr;
27 auto command = std::make_unique<ScalarFieldEditCommand>("add scalar field", target);
28 for (int i = 0; i < samples.size(); ++i) {
29 const BrushPoint &point = samples.point(i);
30 if (point.weight > 0.f && field->containsCell(point.x, point.y)) {
31 command->record(point.x, point.y,
32 field->readScalar(point.x, point.y) + strength * point.weight);
33 }
34 }
35 return command;
36}

52ToolResponse FieldBrushTool::pointerEvent(EditorContext &context, const EditorPointerEvent &event) {
53 cursorX_ = event.x;
54 cursorY_ = event.y;
55 if (event.phase == EditorPointerEvent::Phase::Down) {
56 if (!kernel_ || !operation_ || !context.target() ||
57 !context.transactions().begin(descriptor_.label)) return ToolResponse::ignored();
58 stroking_ = true;
59 stamp(context, event.x, event.y);
60 return ToolResponse::capture();
61 }
62 if (!stroking_) return ToolResponse::ignored();
63 if (event.phase == EditorPointerEvent::Phase::Move) {
64 stamp(context, event.x, event.y);
65 return ToolResponse::consumed();
66 }
67 if (event.phase == EditorPointerEvent::Phase::Up) {
68 context.transactions().commit();
69 stroking_ = false;
70 return ToolResponse::release();
71 }
72 cancel(context);
73 return ToolResponse::release();
74}
