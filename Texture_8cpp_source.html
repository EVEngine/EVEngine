64void Texture::drawOcclusion(Graphics* gfx, const glm::mat4& matrix) const {
65 if (!gfx || !getCastOcclusion()) return;
66 // Treat matrix as 2D affine: translation in column 3, scale from basis lengths.
67 const float x = matrix[3][0];
68 const float y = matrix[3][1];
69 const float w = std::sqrt(matrix[0][0] * matrix[0][0] + matrix[0][1] * matrix[0][1]);
70 const float h = std::sqrt(matrix[1][0] * matrix[1][0] + matrix[1][1] * matrix[1][1]);
71 if (w <= 0.f || h <= 0.f) return;
72 gfx->drawOcclusionTexture(const_cast<Texture*>(this), x, y, w, h);
73}
