24Result<LoadedPointGraph> EvpackPointGraphLoader::load(
25 const AssetRef& graphRef, const asset::EvpackCapabilities& capabilities,
26 const procgen::SpatialData& terrain, std::uint64_t maximumDecodedBytes) const {
27 auto payload = reader_.read(graphRef, "eve.pcg-graph/1", capabilities, maximumDecodedBytes);
28 if (!payload) return Result<LoadedPointGraph>::failure(payload.status());
29 const asset::RuntimeAssetChunk* definition = nullptr;
30 const asset::RuntimeAssetChunk* plan = nullptr;
31 for (const auto& chunk : payload.value().chunks) {
32 if (chunk.kind == asset::EvpackChunkKind::Definition) {
33 if (definition) return failure<LoadedPointGraph>(DiagnosticCode::Conflict,
34 "PCG asset has duplicate definitions");
35 definition = &chunk;
36 } else if (chunk.kind == asset::EvpackChunkKind::Bulk) {
37 if (plan) return failure<LoadedPointGraph>(DiagnosticCode::Conflict,
38 "PCG asset has duplicate execution plans");
39 plan = &chunk;
40 }
41 }
42 if (!definition || !plan || plan->chunkId != 1)
43 return failure<LoadedPointGraph>(DiagnosticCode::ParseError,
44 "PCG definition or execution plan is missing");
45 asset::RuntimeDefinitionLimits definitionLimits;
46 definitionLimits.maximumBytes = maximumDecodedBytes;
47 auto metadata = asset::decodeRuntimeDefinition(definition->bytes, definitionLimits);
48 if (!metadata) return Result<LoadedPointGraph>::failure(metadata.status());
49 const auto* root = metadata.value().getIf<Value::Object>();
50 const Value* schema = root ? field(*root, "schema") : nullptr;
51 const Value* version = root ? field(*root, "schemaVersion") : nullptr;
52 const Value* planReference = root ? field(*root, "executionPlan") : nullptr;
53 const Value* format = root ? field(*root, "executionPlanFormat") : nullptr;
54 const Value* output = root ? field(*root, "outputNode") : nullptr;
55 const Value* slotsValue = root ? field(*root, "spatialSlots") : nullptr;
56 const auto* slots = slotsValue ? slotsValue->getIf<Value::Array>() : nullptr;
57 if (!schema || !schema->isString() || schema->asString() != "eve.pcg-graph" ||
58 !version || !version->isInt64() || version->asInt() != 1 || !planReference ||
59 !planReference->isString() || planReference->asString() != "chunk:1" || !format ||
60 !format->isString() || format->asString() != "EVPCG_POINT_GRAPH/1" ||
61 !output || !output->isString() || output->asString().empty() || !slots || slots->empty())
62 return failure<LoadedPointGraph>(DiagnosticCode::ParseError,
63 "PCG runtime metadata is invalid");
64 auto graph = std::make_unique<procgen::PointGraph>();
65 const std::string planText(reinterpret_cast<const char*>(plan->bytes.data()), plan->bytes.size());
66 if (!graph->deserializeDefinition(planText))
67 return failure<LoadedPointGraph>(DiagnosticCode::ParseError,
68 "PCG execution plan is invalid: " + graph->getError());
69 if (!graph->hasNode(output->asString()))
70 return failure<LoadedPointGraph>(DiagnosticCode::ParseError,
71 "PCG output node does not exist", "$.outputNode");
72 procgen::SpatialData terrainCopy = terrain;
73 for (std::size_t i = 0; i < slots->size(); ++i) {
74 const auto* slot = (*slots)[i].getIf<Value::Object>();
75 const Value* node = slot ? field(*slot, "node") : nullptr;
76 const Value* role = slot ? field(*slot, "role") : nullptr;
77 if (!node || !node->isString() || !role || !role->isString() ||
78 role->asString() != "terrain" ||
79 !graph->setNodeSpatial(node->asString(), &terrainCopy))
80 return failure<LoadedPointGraph>(DiagnosticCode::ParseError,
81 "PCG spatial slot is invalid",
82 "$.spatialSlots[" + std::to_string(i) + "]");
83 }
84 if (!graph->validate())
85 return failure<LoadedPointGraph>(DiagnosticCode::ParseError,
86 "bound PCG graph is invalid: " + graph->getError());
87 return Result<LoadedPointGraph>::success(
88 {graphRef, std::move(graph), output->asString(), std::move(payload).takeValue().variant});
89}
