56bool decodeComponentId(std::string_view text, std::string& nodeId, std::string& name, std::uint64_t& version) {
57 const std::size_t nodeEnd = text.find(kNodeSeparator);
58 if (nodeEnd == std::string_view::npos || nodeEnd == 0) return false;
59 const std::size_t versionPos = text.rfind(kVersionSeparator);
60 if (versionPos == std::string_view::npos || versionPos <= nodeEnd + 1 || versionPos + 1 >= text.size())
61 return false;
62 nodeId = std::string(text.substr(0, nodeEnd));
63 name = std::string(text.substr(nodeEnd + 1, versionPos - nodeEnd - 1));
64 if (name.empty()) return false;
65 std::uint64_t parsed = 0;
66 for (const char digit : text.substr(versionPos + 1)) {
67 if (digit < '0' || digit > '9') return false;
68 parsed = parsed * 10u + static_cast<std::uint64_t>(digit - '0');
69 }
70 if (parsed == 0) return false;
71 version = parsed;
72 return true;
73}

98bool readPositiveNumber(const EditorValue& value, std::uint64_t& out) {
99 if (const auto* integer = value.getIf<std::int64_t>()) {
100 if (*integer <= 0) return false;

161Result<void> ScriptComponentOperationTarget::applyDomainOperation(const DomainOperation& operation) {
162 if (!bound_)
163 return editRefusal<void>(EditorStatus::Rejected, "editor.scene.script-component.unbound",
164 "script component operation target is not bound to a component");
165 if (std::string_view(operation.type) != setOperationType() || operation.target != target_)
166 return editRefusal<void>(EditorStatus::Rejected, "editor.scene.script-component.operation",
167 "operation does not write a field of this script component target");
168
169 const EditorValue* node = payloadField(operation.payload, "node");
170 const EditorValue* name = payloadField(operation.payload, "name");
171 const EditorValue* version = payloadField(operation.payload, "version");
172 const EditorValue* field = payloadField(operation.payload, "field");
173 const EditorValue* value = payloadField(operation.payload, "value");
174 const auto* nodeText = node != nullptr ? node->getIf<std::string>() : nullptr;
175 const auto* nameText = name != nullptr ? name->getIf<std::string>() : nullptr;
176 const auto* fieldText = field != nullptr ? field->getIf<std::string>() : nullptr;
177 std::uint64_t payloadVersion = 0;
178 if (nodeText == nullptr || nameText == nullptr || fieldText == nullptr || value == nullptr || version == nullptr ||
179 !readPositiveNumber(*version, payloadVersion))
180 return editRefusal<void>(EditorStatus::Rejected, "editor.scene.script-component.payload",
181 "script component operation payload is malformed");
182
183 // 载荷自带身份：绑定被后来的选中项改写过的过期目标只会被拒绝，不会写到别的组件上。
184 if (*nodeText != nodeId_ || *nameText != name_ || payloadVersion != version_.value())
185 return editRefusal<void>(EditorStatus::Conflict, "editor.scene.script-component.stale-target",
186 "operation belongs to another script component than the bound one");
187 return writeField(*fieldText, eve::editing::toPresentationValue(*value));
188}

190std::unique_ptr<scene_editing::IDomainOperationTarget> ScriptComponentOperationTarget::cloneDomainState() const {

194 // 而不是绑定当时那份可能已经过期的拷贝。数据已经不在了就留着绑定时的拷贝，发布时会被拒绝。

200Result<void> ScriptComponentOperationTarget::commitDomainState(
201 std::unique_ptr<scene_editing::IDomainOperationTarget> candidate) {
202 auto* typed = dynamic_cast<ScriptComponentOperationTarget*>(candidate.get());
203 if (typed == nullptr || !typed->staging_ || !bound_ || !typed->bound_ || typed->target_ != target_ ||
204 typed->entityId_ != entityId_ || typed->name_ != name_ || typed->version_ != version_)
205 return editRefusal<void>(EditorStatus::Conflict, "editor.scene.script-component.candidate",
206 "staged candidate belongs to another script component");
207 if (eve::findScriptComponentValues(ScriptComponentStoreKey{entityId_, name_, version_}) == nullptr)
208 return editRefusal<void>(EditorStatus::Conflict, "editor.scene.script-component.missing-data",
209 "script component data is gone, so the edit cannot be published");
210 // 换代（同名同版本重新登记）之后这份数据属于旧 schema：宁可拒绝，也不按新 schema 发布。
211 const auto declared = eve::scriptComponentDeclaredSchema(name_, version_);
212 if (!declared.ok() || declared.value().generation != generation_)
213 return editRefusal<void>(EditorStatus::Conflict, "editor.scene.script-component.stale-generation",
214 "script component was re-registered since the reference, so the edit is refused");
215
216 // 整份发布：候选是按权威状态克隆的，事务两侧又都有版本号校验，所以这一次
217 // `setScriptComponentValues` 就是"原子发布"——不存在写了一半的字段表。
218 const auto published = eve::setScriptComponentValues(ScriptComponentStoreKey{entityId_, name_, version_},
219 Value::object(typed->fields_));
220 if (!published.ok())
221 return editRefusal<void>(EditorStatus::Failed, "editor.scene.script-component.publish",
222 "engine-side store refused the staged script component candidate");
223 fields_ = typed->fields_;
224 return eve::editing::applied<void>();
225}

279 std::string name;

311PropertySchema ScriptComponentPayloadProvider::schemaOf(const ScriptComponentInstanceState& state) const {
312 PropertySchema out;
313 out.typeId = payloadType();
314
315 const auto declared = eve::scriptComponentDeclaredSchema(state.name, state.version);
316 // 声明 schema 拿不到（例如登记已经丢了）时，退回当前值那张表：工具至少看得见"有组件、有这些
317 // 数据"，而不是一个空面板；这些字段全部只读（没有声明就没有类型可校验）。
318 std::vector<std::string> fallback;
319 const std::vector<std::string>* fields = nullptr;
320 if (declared.ok() && !declared.value().fields.empty()) {
321 fields = &declared.value().fields;
322 } else {
323 fallback.reserve(state.fields.size());
324 for (const auto& [field, value] : state.fields) fallback.push_back(field);
325 fields = &fallback;
326 }
327
328 for (std::size_t index = 0; index < fields->size(); ++index) {
329 const std::string& field = (*fields)[index];
330 const Value* declaredDefault =
331 declared.ok() && index < declared.value().defaults.size() ? &declared.value().defaults[index] : nullptr;
332 const auto current = state.fields.find(field);
333 const bool hasCurrent = current != state.fields.end();
334
335 // 可写 = 脚本里给了非 null 的默认值：那才是类型与默认值的来源。声明成 null 的字段只读，
336 // 但当前值照样显示（类型从当前值取）。
337 const bool writable = declaredDefault != nullptr && declaredDefault->type() != Value::Type::Null;
338 const Value* typeSource = writable ? declaredDefault : (hasCurrent ? &current->second : nullptr);
339
340 PropertyDescriptor descriptor;
341 descriptor.path = PropertyPath(field);
342 descriptor.displayNameKey = field;
343 descriptor.category = state.name;
344 descriptor.type = typeSource != nullptr ? propertyTypeOf(*typeSource) : PropertyType::ReadOnlyText;
345 descriptor.defaultValue = eve::editing::toEditingValue(
346 declaredDefault != nullptr ? *declaredDefault : (hasCurrent ? current->second : Value::null()));
347 if (!writable) descriptor.flags = PropertyFlag::ReadOnly;
348 out.properties.push_back(std::move(descriptor));
349 }
350 return out;
351}

370Result<DomainOperation> ScriptComponentPayloadProvider::planWrite(const Resolved& target, const PropertyPath& path,
371 const EditorValue& value) const {
372 const auto current = target.state.fields.find(path.value());
373 const EditorValue previous =
374 current != target.state.fields.end() ? eve::editing::toEditingValue(current->second) : EditorValue{};
375
376 DomainOperation operation;
377 operation.type = std::string(ScriptComponentOperationTarget::setOperationType());
378 operation.inverseType = operation.type;
379 operation.target = target.target;
380 // 载荷里带着组件身份：过期的操作目标据此拒绝写错组件；撤销用的反向值也一起进去。
381 operation.payload = EditorValue::Object{
382 {"node", EditorValue(target.nodeId)},
383 {"name", EditorValue(target.state.name)},
384 {"version", EditorValue(static_cast<std::int64_t>(target.state.version.value()))},
385 {"field", EditorValue(path.value())},
386 {"value", value},
387 };
388 operation.inverse = EditorValue::Object{
389 {"node", EditorValue(target.nodeId)},
390 {"name", EditorValue(target.state.name)},
391 {"version", EditorValue(static_cast<std::int64_t>(target.state.version.value()))},
392 {"field", EditorValue(path.value())},
393 {"value", previous},
394 };
395 operation.hasInverse = true;
396 operation.affectedObjects.push_back({target.target, target.nodeId, target.state.generation.value()});
397 operation.affectedProperties.push_back(path.value());
398 operation.mergeKey = "script-component:" + target.nodeId + ":" + target.state.name + "@" +
399 std::to_string(target.state.version.value()) + ":" + path.value();
400 return eve::editing::applied<DomainOperation>(std::move(operation));
401}

403Result<DomainOperation> ScriptComponentPayloadProvider::makeSet(const SelectionSnapshot& selection,
404 const PropertyPath& path, const EditorValue& value,
405 PropertySetMode mode) const {
406 if (mode == PropertySetMode::Reset) return makeReset(selection, path);
407 if (mode != PropertySetMode::Absolute)
408 return editRefusal<DomainOperation>(EditorStatus::Rejected, "editor.scene.script-component.mode",
409 "script component fields accept absolute edits only");
410
411 auto resolved = resolve(selection);
412 if (!resolved.ok()) return Result<DomainOperation>::failure(resolved.status());
413 const Resolved& target = resolved.value();
414
415 const auto descriptor = schemaOf(target.state).find(path);
416 if (!descriptor)
417 return editRefusal<DomainOperation>(EditorStatus::NotFound, "editor.scene.script-component.field",
418 "script component has no declared field: " + path.value());
419 if (hasPropertyFlag(descriptor->flags, PropertyFlag::ReadOnly))
420 return editRefusal<DomainOperation>(
421 EditorStatus::Rejected, "editor.scene.script-component.untyped",
422 "field has no declared non-null default, so its type is unknown and it stays read-only: " + path.value());
423 const auto validation = validatePropertyValue(*descriptor, value);
424 if (!validation.ok()) return Result<DomainOperation>::failure(validation.status());
425 return planWrite(target, path, value);

444eve::Result<ScriptComponentPayloadProvider::Resolved> ScriptComponentPayloadProvider::resolve(
445 const SelectionSnapshot& selection) const {
446 const auto failure = [](eve::DiagnosticCode code, std::string message, std::string path) {
447 return eve::Result<Resolved>::failure(
448 eve::Diagnostic::error(code, std::move(message), std::move(path), {}, "scene.script-component"));
449 };
450
451 if (selection.items.size() != 1)
452 return failure(eve::DiagnosticCode::InvalidArgument,
453 "script component properties are edited one component at a time", "selection");
454 const scene_editing::SelectionItem& item = selection.items.front();
455 if (item.type != payloadType())
456 return failure(eve::DiagnosticCode::InvalidArgument,
457 "selection does not belong to the script component provider", item.type);
458
459 std::string nodeId;
460 std::string name;
461 std::uint64_t version = 0;
462 if (!decodeComponentId(item.item.value(), nodeId, name, version))
463 return failure(eve::DiagnosticCode::InvalidArgument, "script component reference is malformed",
464 item.item.value());
465
466 scene::Scene* sceneModule = scene::Scene::create();
467 if (sceneModule == nullptr)
468 return failure(eve::DiagnosticCode::Failed, "scene module is not available", item.target.value());
469
470 for (const ScriptComponentInstanceState& state :
471 sceneModule->scriptComponentStatesAt(hostNameOf(item.target), nodeId)) {
472 if (state.name != name || state.version.value() != version) continue;
473 return eve::Result<Resolved>::success(Resolved{item.target, nodeId, state});
474 }
475 return failure(eve::DiagnosticCode::NotFound, "script component is no longer on this node", item.item.value());
476}
