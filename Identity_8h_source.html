80 [[nodiscard]] static std::optional<Id128> parse(std::string_view text) noexcept {
81 if (text.size() != 36 || text[8] != '-' || text[13] != '-' || text[18] != '-' || text[23] != '-') {
82 return std::nullopt;
83 }
84
85 Bytes bytes{};
86 std::size_t byteIndex = 0;
87 for (std::size_t i = 0; i < text.size();) {
88 if (text[i] == '-') {
89 ++i;
90 continue;
91 }
92
93 if (i + 1 >= text.size()) return std::nullopt;
94 const auto high = hexValue(text[i]);
95 const auto low = hexValue(text[i + 1]);
96 if (!high || !low || byteIndex >= bytes.size()) {
97 return std::nullopt;
98 }
99 bytes[byteIndex++] = static_cast<std::uint8_t>((*high << 4u) | *low);
100 i += 2;
101 }
102 if (byteIndex != bytes.size()) return std::nullopt;
103 return Id128(bytes);
104 }

112 if (bytes.size() != 16) return std::nullopt;
113 Bytes result{};
114 for (std::size_t i = 0; i < result.size(); ++i) result[i] = bytes[i];
115 return Id128(result);

122 [[nodiscard]] std::string format() const {
123 static constexpr char digits[] = "0123456789abcdef";
124 std::string result;
125 result.reserve(36);
126 for (std::size_t i = 0; i < bytes_.size(); ++i) {
127 if (i == 4 || i == 6 || i == 8 || i == 10) result.push_back('-');
128 result.push_back(digits[(bytes_[i] >> 4u) & 0x0fu]);
129 result.push_back(digits[bytes_[i] & 0x0fu]);

151 [[nodiscard]] Id128 child(std::string_view role) const noexcept {
152 if (isNil() || role.empty()) return Id128::nil();
153
154 Bytes result = bytes_;
155 std::uint64_t hash = 14695981039346656037ull;
156 for (const unsigned char byte : role) {
157 hash ^= byte;
158 hash *= 1099511628211ull;
159 }
160 for (std::size_t i = 0; i < sizeof(hash); ++i) {
161 result[8 + i] ^= static_cast<std::uint8_t>((hash >> (i * 8u)) & 0xffu);
162 }
163 // Preserve the UUID variant while retaining the parent's namespace.
164 result[8] = static_cast<std::uint8_t>((result[8] & 0x3fu) | 0x80u);
