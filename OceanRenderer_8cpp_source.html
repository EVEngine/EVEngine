38Result<std::unique_ptr<OceanRenderer>> OceanRendererBuilder::build() const {
39 const auto lifetime = graphics_.resourceLifetime();
40 if (lifetime.expired())
41 return Result<std::unique_ptr<OceanRenderer>>::failure(
42 failure(DiagnosticCode::StaleHandle, "Graphics provider has retired"));
43 if (graphics_.getBackendName() != "vulkan" && graphics_.getBackendName() != "webgpu")
44 return Result<std::unique_ptr<OceanRenderer>>::failure(
45 failure(DiagnosticCode::Unsupported, "Ocean renderer requires Vulkan or WebGPU"));
46 if (backend != OceanRenderBackend::CpuReference && backend != OceanRenderBackend::GpuResident)
47 return Result<std::unique_ptr<OceanRenderer>>::failure(
48 failure(DiagnosticCode::InvalidArgument, "Invalid ocean render backend"));
49 if (backend == OceanRenderBackend::GpuResident && graphics_.getBackendName() != "vulkan")
50 return Result<std::unique_ptr<OceanRenderer>>::failure(
51 failure(DiagnosticCode::Unsupported, "Resident ocean rendering requires Vulkan"));
52 OceanMeshBuilder builder;
53 builder.settings = mesh;
54 auto topology = builder.build();
55 if (!topology) return Result<std::unique_ptr<OceanRenderer>>::failure(topology.status());
56 std::unique_ptr<OceanRenderer> renderer(new OceanRenderer);
57 auto& state = *renderer->impl_;
58 state.graphics = &graphics_;
59 state.lifetime = lifetime;
60 state.topology = std::move(topology.value());
61 state.backend = backend;
62 try {
63 if (graphics_.getBackendName() == "webgpu")
64 state.shader = graphics_.newMeshShaderFromWgsl(shaders::oceanVertex, shaders::oceanFragment);
65 else
66 state.shader = graphics_.newMeshShaderFromSpv(
67 backend == OceanRenderBackend::GpuResident
68 ? std::vector<uint32_t>(ocean_gpu_vert_spv, ocean_gpu_vert_spv + ocean_gpu_vert_spv_count)
69 : std::vector<uint32_t>(ocean_vert_spv, ocean_vert_spv + ocean_vert_spv_count),
70 std::vector<uint32_t>(ocean_frag_spv, ocean_frag_spv + ocean_frag_spv_count));
71 if (!state.shader)
72 return Result<std::unique_ptr<OceanRenderer>>::failure(
73 failure(DiagnosticCode::Unsupported, "Ocean mesh shader unavailable"));
74 auto configured = graphics_.configureMeshShaderSurface(*state.shader, graphics::BlendMode::Opaque, true, true);
75 if (!configured) return Result<std::unique_ptr<OceanRenderer>>::failure(configured.status());
76 if (backend == OceanRenderBackend::GpuResident) {
77 state.shader->declareMatrix("oceanBands");
78 state.shader->declareMatrix("oceanFrame");
79 }
80 const auto count = state.topology.parameterOffsets.size();
81 std::vector<float> positions(count * 3, 0.0F), normals(count * 3, 0.0F), uv(count * 2, 0.0F);
82 for (std::size_t i = 0; i < count; ++i) {
83 positions[i * 3] = static_cast<float>(state.topology.parameterOffsets[i].x);
84 positions[i * 3 + 2] = static_cast<float>(state.topology.parameterOffsets[i].y);
85 normals[i * 3 + 1] = 1.0F;
86 }
87 state.mesh =
88 graphics_.newMeshFromArrays(positions.data(), normals.data(), uv.data(), static_cast<int>(count),
89 state.topology.indices.data(), static_cast<int>(state.topology.indices.size()));
90 if (!state.mesh)
91 return Result<std::unique_ptr<OceanRenderer>>::failure(
92 failure(DiagnosticCode::Unsupported, "Ocean mesh allocation unavailable"));
93 } catch (const std::exception& error) {
94 return Result<std::unique_ptr<OceanRenderer>>::failure(failure(DiagnosticCode::Unsupported, error.what()));
95 }
96 return Result<std::unique_ptr<OceanRenderer>>::success(std::move(renderer));
97}
98Result<void> OceanRenderer::publish(const OceanMeshFrame& frame) {
99 auto& state = *impl_;
100 if (state.backend != OceanRenderBackend::CpuReference)
101 return Result<void>::failure(failure(DiagnosticCode::Conflict, "Renderer expects a resident GPU projection"));
102 if (state.lifetime.expired())
103 return Result<void>::failure(failure(DiagnosticCode::StaleHandle, "Graphics retired"));
104 const auto count = state.topology.parameterOffsets.size();
105 auto finite = [](float value) { return std::isfinite(value); };
106 if (frame.positions.size() != count * 3 || frame.normals.size() != count * 3 || frame.surface.size() != count * 2 ||
107 !std::isfinite(frame.time) || !std::isfinite(frame.origin.x) || !std::isfinite(frame.origin.y) ||
108 !std::isfinite(frame.origin.z) || !std::all_of(frame.positions.begin(), frame.positions.end(), finite) ||
109 !std::all_of(frame.normals.begin(), frame.normals.end(), finite) ||
110 !std::all_of(frame.surface.begin(), frame.surface.end(), finite))
111 return Result<void>::failure(failure(DiagnosticCode::InvalidArgument, "Invalid ocean vertex frame"));
112 for (std::size_t i = 0; i < count; ++i) {
113 const glm::vec3 normal(frame.normals[i * 3], frame.normals[i * 3 + 1], frame.normals[i * 3 + 2]);
114 const float length = glm::length(normal);
115 if (length < 0.99F || length > 1.01F || frame.surface[i * 2] < 0.0F || frame.surface[i * 2] > 1.0F)
116 return Result<void>::failure(failure(DiagnosticCode::InvalidArgument, "Ocean normals or foam invalid"));
117 }
118 if (!state.graphics->updateMeshVertices(state.mesh, frame.positions.data(), frame.normals.data(),
119 frame.surface.data(), static_cast<int>(count), nullptr, 0))
120 return Result<void>::failure(failure(DiagnosticCode::Unsupported, "Ocean mesh upload rejected"));
121 state.origin = frame.origin;
122 state.published = true;
123 return Result<void>::success();
124}
125Result<void> OceanRenderer::draw() {
126 auto& state = *impl_;
127 if (state.lifetime.expired())
128 return Result<void>::failure(failure(DiagnosticCode::StaleHandle, "Graphics retired"));
129 if (!state.published) return Result<void>::failure(failure(DiagnosticCode::Conflict, "Ocean frame not published"));
130 if (state.hullPublished && state.origin != state.hullOrigin)
131 return Result<void>::failure(failure(DiagnosticCode::Conflict, "Hull mask and ocean render origins differ"));
132 state.graphics->setMesh3DNormalTexture(state.hullPublished ? state.hullMask[state.hullFront] : nullptr);
133 auto* atlas = state.backend == OceanRenderBackend::GpuResident ? state.atlas[state.front] : nullptr;
134 if (state.backend == OceanRenderBackend::GpuResident)
135 state.graphics->setMesh3DHeightTexture(state.wake[state.front]);
136 state.graphics->drawMeshShader(state.mesh, glm::mat4(1.0F), atlas, glm::vec4(1.0F), state.shader);
137 if (state.backend == OceanRenderBackend::GpuResident) state.graphics->setMesh3DHeightTexture(nullptr);
138 state.graphics->setMesh3DNormalTexture(nullptr);
139 return Result<void>::success();
140}
