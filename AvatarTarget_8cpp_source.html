27 const auto* a = v ? v->getIf<EditorValue::Array>() : nullptr;
28 if (!a || static_cast<int>(a->size()) != n) return false;
29 for (int i = 0; i < n; ++i) {

55EditorResult<AvatarLayerValue> parseLayer(const EditorValue& v) {
56 AvatarLayerValue out;
57 const auto * id = field(v, "id"), *name = field(v, "name"), *texture = field(v, "texture"), *z = field(v, "z"),
58 *visible = field(v, "visible");
59 const auto* ids = id ? id->getIf<std::string>() : nullptr;
60 const auto* names = name ? name->getIf<std::string>() : nullptr;
61 const auto* textures = texture ? texture->getIf<std::string>() : nullptr;
62 const auto* zs = z ? z->getIf<int64_t>() : nullptr;
63 const auto* vis = visible ? visible->getIf<bool>() : nullptr;
64 if (!ids || ids->empty() || !names || !textures || !zs || !vis ||
65 !readFloats(field(v, "offset"), out.offset.data(), 2) || !readFloats(field(v, "size"), out.size.data(), 2) ||
66 !readFloats(field(v, "color"), out.color.data(), 4))
67 return fail<AvatarLayerValue>(EditorStatus::Rejected, "editor.avatar.layer", "Avatar layer is invalid");
68 out.id = ObjectId(*ids);
69 out.name = *names;
70 out.textureAsset = *textures;
71 out.zIndex = static_cast<int>(*zs);
72 out.visible = *vis;
73 return eve::editing::applied<AvatarLayerValue>(std::move(out));
74}

151bool AvatarDocumentTarget::matches(const SelectionSnapshot& s) const {
152 if (s.items.empty()) return false;
153 const std::string type = s.items.front().type;
154 for (const auto& i : s.items) {
155 if (i.target != TargetId(id_) || i.type != type) return false;
156 if (type == "avatar.layer" &&
157 std::none_of(layers_.begin(), layers_.end(), [&](const auto& v) { return v.id.value() == i.item.value(); }))
158 return false;
159 if (type == "avatar.parameter" && std::none_of(parameters_.begin(), parameters_.end(),
160 [&](const auto& v) { return v.id.value() == i.item.value(); }))
161 return false;
162 }
163 return type == "avatar.layer" || type == "avatar.parameter";

192PropertyReadResult AvatarDocumentTarget::read(const SelectionSnapshot& s, const PropertyPath& p) const {
193 if (!matches(s) || !schema(s).find(p)) return {};
194 std::optional<EditorValue> common;
195 for (const auto& i : s.items) {
196 EditorValue value;
197 if (i.type == "avatar.layer") {
198 const auto& v = *std::find_if(layers_.begin(), layers_.end(),
199 [&](const auto& x) { return x.id.value() == i.item.value(); });
200 if (p == PropertyPath("layer.texture"))
201 value = v.textureAsset;
202 else if (p == PropertyPath("layer.z"))
203 value = int64_t{v.zIndex};
204 else if (p == PropertyPath("layer.visible"))
205 value = v.visible;
206 else if (p == PropertyPath("layer.offset"))
207 value = floats(v.offset.data(), 2);
208 else if (p == PropertyPath("layer.size"))
209 value = floats(v.size.data(), 2);
210 else
211 value = floats(v.color.data(), 4);
212 } else {
213 const auto& v = *std::find_if(parameters_.begin(), parameters_.end(),
214 [&](const auto& x) { return x.id.value() == i.item.value(); });
215 if (p == PropertyPath("parameter.default"))
216 value = v.defaultValue;
217 else if (p == PropertyPath("parameter.minimum"))
218 value = v.minimum;
219 else if (p == PropertyPath("parameter.maximum"))
220 value = v.maximum;
221 else
222 value = v.value;
223 }
224 if (!common)
225 common = value;
226 else if (*common != value)
227 return {PropertyReadState::Mixed, {}, {}};
228 }
229 return {PropertyReadState::Value, *common, {}};
230}

254EditorResult<DomainOperation> AvatarDocumentTarget::makeSet(const SelectionSnapshot& s, const PropertyPath& p,
255 const EditorValue& value, PropertySetMode mode) const {
256 if (mode == PropertySetMode::Reset) return makeReset(s, p);
257 auto d = schema(s).find(p);
258 if (!matches(s) || !d || mode != PropertySetMode::Absolute || !validatePropertyValue(*d, value).ok())
259 return fail<DomainOperation>(EditorStatus::Rejected, "editor.avatar.set", "Avatar property edit is invalid");
260 auto candidate = *this;
261 for (const auto& i : s.items)
262 if (i.type == "avatar.layer") {
263 auto& v = *std::find_if(candidate.layers_.begin(), candidate.layers_.end(),
264 [&](const auto& x) { return x.id.value() == i.item.value(); });
265 if (p == PropertyPath("layer.texture"))
266 v.textureAsset = *value.getIf<std::string>();
267 else if (p == PropertyPath("layer.z"))
268 v.zIndex = static_cast<int>(*value.getIf<int64_t>());
269 else if (p == PropertyPath("layer.visible"))
270 v.visible = *value.getIf<bool>();
271 else if (p == PropertyPath("layer.offset"))
272 readFloats(&value, v.offset.data(), 2);
273 else if (p == PropertyPath("layer.size"))
274 readFloats(&value, v.size.data(), 2);
275 else
276 readFloats(&value, v.color.data(), 4);
277 } else {
278 auto& v = *std::find_if(candidate.parameters_.begin(), candidate.parameters_.end(),
279 [&](const auto& x) { return x.id.value() == i.item.value(); });
280 const float n = static_cast<float>(*value.getIf<double>());
281 if (p == PropertyPath("parameter.default"))
282 v.defaultValue = n;
283 else if (p == PropertyPath("parameter.minimum"))
284 v.minimum = n;
285 else if (p == PropertyPath("parameter.maximum"))
286 v.maximum = n;
287 else
288 v.value = n;
289 }
290 if (errors(candidate.validate()))
291 return fail<DomainOperation>(EditorStatus::Rejected, "editor.avatar.invalid",
292 "Avatar edit produces an invalid asset");
293 return replacement(candidate.contentValue(), p.value());
294}

377std::vector<EditorDiagnostic> AvatarDocumentTarget::validate() const {
378 std::vector<EditorDiagnostic> d;
379 if (kind_ != "image" && kind_ != "live2d" && kind_ != "vroid")
380 d.push_back(diagnostic("editor.avatar.kind", "Avatar kind must be image, live2d or vroid"));
381 if (kind_ != "image" && sourceAsset_.empty())
382 d.push_back(diagnostic("editor.avatar.source", "Live2D/VRoid Avatar requires a source asset"));
383 if (kind_ != "image" && !layers_.empty())
384 d.push_back(diagnostic("editor.avatar.backend-layers",
385 "Image layers are supported only by image Avatar assets"));
386 if (layers_.size() > 512 || parameters_.size() > 2048 || expressions_.size() > 1024)
387 d.push_back(diagnostic("editor.avatar.budget", "Avatar asset exceeds layer, parameter or expression budgets"));
388 std::set<std::string> ids, names, channels;
389 for (const auto& v : layers_) {
390 if (v.id.empty() || v.name.empty() || !ids.insert(v.id.value()).second || !names.insert(v.name).second)
391 d.push_back(diagnostic("editor.avatar.layer-identity", "Avatar layer IDs and names must be unique"));
392 channels.insert(v.name);
393 const bool vectorsFinite =
394 std::all_of(v.offset.begin(), v.offset.end(), [](float n) { return std::isfinite(n); }) &&
395 std::all_of(v.size.begin(), v.size.end(), [](float n) { return std::isfinite(n); });
396 if (v.textureAsset.empty() || v.zIndex < -100000 || v.zIndex > 100000 || !vectorsFinite || v.size[0] <= 0 ||
397 v.size[1] <= 0 ||
398 !std::all_of(v.color.begin(), v.color.end(), [](float n) { return std::isfinite(n) && n >= 0 && n <= 1; }))
399 d.push_back(diagnostic("editor.avatar.layer-range", "Avatar layer values are invalid"));
400 }
401 for (const auto& v : parameters_) {
402 if (v.id.empty() || v.name.empty() || !ids.insert(v.id.value()).second || !names.insert(v.name).second)
403 d.push_back(diagnostic("editor.avatar.parameter-identity",
404 "Avatar parameter IDs and names must be unique"));
405 channels.insert(v.name);
406 if (!std::isfinite(v.minimum) || !std::isfinite(v.maximum) || !std::isfinite(v.defaultValue) ||
407 !std::isfinite(v.value) || v.minimum > v.maximum || v.defaultValue < v.minimum ||
408 v.defaultValue > v.maximum || v.value < v.minimum || v.value > v.maximum)
409 d.push_back(diagnostic("editor.avatar.parameter-range", "Avatar parameter range/default/value is invalid"));
410 }
411 for (const auto& e : expressions_) {
412 if (e.id.empty() || e.name.empty() || !ids.insert(e.id.value()).second || !names.insert(e.name).second)
413 d.push_back(diagnostic("editor.avatar.expression-identity",
414 "Avatar expression IDs and names must be unique"));
415 if (e.channels.size() > 2048)
416 d.push_back(diagnostic("editor.avatar.expression-budget", "Avatar expression exceeds 2048 channels"));
417 for (const auto& [name, value] : e.channels)
418 if (!channels.contains(name) || !std::isfinite(value))
419 d.push_back(diagnostic("editor.avatar.expression-channel",
420 "Avatar expression references an unknown or invalid channel"));
421 }
422 return d;
423}
424EditorResult<void> AvatarDocumentTarget::applyDomainOperation(const DomainOperation& op) {
425 if (op.target != TargetId(id_) || op.type != "avatar.document.replace.v1" ||
426 !op.payload.isWithinLimits(8, 100000, 8 * 1024 * 1024))
427 return fail<void>(EditorStatus::Rejected, "editor.avatar.operation", "Avatar operation is invalid");
428 const auto *kind = field(op.payload, "kind"), *source = field(op.payload, "source"),
429 *layers = field(op.payload, "layers"), *parameters = field(op.payload, "parameters"),
430 *expressions = field(op.payload, "expressions");
431 const auto* k = kind ? kind->getIf<std::string>() : nullptr;
432 const auto* s = source ? source->getIf<std::string>() : nullptr;
433 const auto* la = layers ? layers->getIf<EditorValue::Array>() : nullptr;
434 const auto* pa = parameters ? parameters->getIf<EditorValue::Array>() : nullptr;
435 const auto* ea = expressions ? expressions->getIf<EditorValue::Array>() : nullptr;
436 if (!k || !s || !la || !pa || !ea || la->size() > 512 || pa->size() > 2048 || ea->size() > 1024)
