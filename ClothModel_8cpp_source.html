58eve::Result<ClothModel> ClothModel::fromTriangles(std::span<const float> positions,
59 std::span<const int> triangleIndices,
60 std::span<const float> inverseMasses) {
61 if (positions.empty() || positions.size() % 3 != 0)
62 return invalidModel("cloth positions must be a non-empty packed XYZ array");
63 if (triangleIndices.empty() || triangleIndices.size() % 3 != 0)
64 return invalidModel("cloth indices must contain complete triangles");
65
66 const size_t particleCount = positions.size() / 3;
67 if (!inverseMasses.empty() && inverseMasses.size() != particleCount)
68 return invalidModel("cloth inverse-mass count must match the particle count");
69
70 ClothModel candidate;
71 candidate.particles_.reserve(particleCount);
72 for (size_t i = 0; i < particleCount; ++i) {
73 const float inverseMass = inverseMasses.empty() ? 1.f : inverseMasses[i];
74 if (!std::isfinite(positions[i * 3]) || !std::isfinite(positions[i * 3 + 1]) ||
75 !std::isfinite(positions[i * 3 + 2]) || !std::isfinite(inverseMass) || inverseMass < 0.f)
76 return invalidModel("cloth particles require finite positions and non-negative inverse masses");
77 candidate.particles_.push_back({positions[i * 3], positions[i * 3 + 1], positions[i * 3 + 2], inverseMass});
78 }
79
80 std::map<int64_t, std::vector<int>> edgeTriangles;
81 std::set<int64_t> structuralEdges;
82 for (size_t i = 0; i < triangleIndices.size(); i += 3) {
83 const int a = triangleIndices[i];
84 const int b = triangleIndices[i + 1];
85 const int c = triangleIndices[i + 2];
86 if (a < 0 || b < 0 || c < 0 || static_cast<size_t>(a) >= particleCount ||
87 static_cast<size_t>(b) >= particleCount || static_cast<size_t>(c) >= particleCount)
88 return invalidModel("cloth triangle index is out of range");
89 if (a == b || b == c || c == a) return invalidModel("cloth triangles must reference three distinct particles");
90
91 const auto& pa = candidate.particles_[static_cast<size_t>(a)];
92 const auto& pb = candidate.particles_[static_cast<size_t>(b)];
93 const auto& pc = candidate.particles_[static_cast<size_t>(c)];
94 const float abx = pb.x - pa.x, aby = pb.y - pa.y, abz = pb.z - pa.z;
95 const float acx = pc.x - pa.x, acy = pc.y - pa.y, acz = pc.z - pa.z;
96 const float nx = aby * acz - abz * acy;
97 const float ny = abz * acx - abx * acz;
98 const float nz = abx * acy - aby * acx;
99 if (nx * nx + ny * ny + nz * nz <= 1e-12f) return invalidModel("cloth triangles must have non-zero rest area");
100
101 const int triangle = static_cast<int>(candidate.triangles_.size());
102 candidate.triangles_.push_back({{a, b, c}});
103 const int edges[3][2] = {{a, b}, {b, c}, {c, a}};
104 for (const auto& edge : edges) {
105 const int64_t key = edgeKey(edge[0], edge[1]);
106 edgeTriangles[key].push_back(triangle);
107 structuralEdges.insert(key);
108 }
109 }
110
111 auto addDistance = [&](int a, int b, ClothConstraintKind kind) {
112 const auto& pa = candidate.particles_[static_cast<size_t>(a)];
113 const auto& pb = candidate.particles_[static_cast<size_t>(b)];
114 const float dx = pb.x - pa.x, dy = pb.y - pa.y, dz = pb.z - pa.z;
115 candidate.distanceConstraints_.push_back({a, b, std::sqrt(dx * dx + dy * dy + dz * dz), kind});
116 };
117 for (int64_t key : structuralEdges) {
118 addDistance(static_cast<int>(key >> 32), static_cast<int>(static_cast<uint32_t>(key)),
119 ClothConstraintKind::Structural);
120 }
121
122 std::set<int64_t> bendEdges;
123 for (const auto& entry : edgeTriangles) {
124 if (entry.second.size() > 2)
125 return invalidModel("cloth topology must be manifold: an edge has more than two triangles");
126 if (entry.second.size() != 2) continue;
127 const auto& first = candidate.triangles_[static_cast<size_t>(entry.second[0])];
128 const auto& second = candidate.triangles_[static_cast<size_t>(entry.second[1])];
129 const int edgeA = static_cast<int>(entry.first >> 32);
130 const int edgeB = static_cast<int>(static_cast<uint32_t>(entry.first));
131 int oppositeA = -1;
132 int oppositeB = -1;
133 for (int vertex : first.vertices)
134 if (vertex != edgeA && vertex != edgeB) oppositeA = vertex;
135 for (int vertex : second.vertices)
136 if (vertex != edgeA && vertex != edgeB) oppositeB = vertex;
137 if (oppositeA < 0 || oppositeB < 0 || oppositeA == oppositeB)
138 return invalidModel("cloth adjacent triangles have invalid topology");
139 candidate.foldConstraints_.push_back({edgeA, edgeB, oppositeA, oppositeB});
140 if (bendEdges.insert(edgeKey(oppositeA, oppositeB)).second)
141 addDistance(oppositeA, oppositeB, ClothConstraintKind::Bend);
142 }
143
144 candidate.closed_ = std::all_of(edgeTriangles.begin(), edgeTriangles.end(),
145 [](const auto& entry) { return entry.second.size() == 2; });
146 if (candidate.closed_) {
147 double signedVolume = 0.0;
148 for (const ClothModelTriangle& triangle : candidate.triangles_) {
149 const auto& a = candidate.particles_[static_cast<size_t>(triangle.vertices[0])];
150 const auto& b = candidate.particles_[static_cast<size_t>(triangle.vertices[1])];
151 const auto& c = candidate.particles_[static_cast<size_t>(triangle.vertices[2])];
152 signedVolume += (double(a.x) * (double(b.y) * c.z - double(b.z) * c.y) +
153 double(a.y) * (double(b.z) * c.x - double(b.x) * c.z) +
154 double(a.z) * (double(b.x) * c.y - double(b.y) * c.x)) /
155 6.0;
156 }
157 if (std::abs(signedVolume) <= 1e-9) return invalidModel("closed cloth topology must have non-zero rest volume");
158 candidate.restVolume_ = static_cast<float>(signedVolume);
159 }
160
161 struct Neighbor {
162 int particle = 0;
163 float distance = 0.f;
164 };
165 std::vector<std::vector<Neighbor>> adjacency(particleCount);
166 for (int64_t key : structuralEdges) {
167 const int a = static_cast<int>(key >> 32);
168 const int b = static_cast<int>(static_cast<uint32_t>(key));
169 const auto& pa = candidate.particles_[static_cast<size_t>(a)];
170 const auto& pb = candidate.particles_[static_cast<size_t>(b)];
171 const float dx = pb.x - pa.x, dy = pb.y - pa.y, dz = pb.z - pa.z;
172 const float distance = std::sqrt(dx * dx + dy * dy + dz * dz);
173 adjacency[static_cast<size_t>(a)].push_back({b, distance});
174 adjacency[static_cast<size_t>(b)].push_back({a, distance});
175 }
176 using QueueEntry = std::tuple<float, int, int>;
177 std::priority_queue<QueueEntry, std::vector<QueueEntry>, std::greater<>> queue;
178 std::vector<float> tetherDistance(particleCount, std::numeric_limits<float>::infinity());
179 std::vector<int> tetherAnchor(particleCount, -1);
180 for (size_t particle = 0; particle < particleCount; ++particle) {
181 if (candidate.particles_[particle].inverseMass != 0.f) continue;
182 tetherDistance[particle] = 0.f;
183 tetherAnchor[particle] = static_cast<int>(particle);
184 queue.emplace(0.f, static_cast<int>(particle), static_cast<int>(particle));
185 }
186 while (!queue.empty()) {
187 const auto [distance, anchor, particle] = queue.top();
188 queue.pop();
189 if (distance > tetherDistance[static_cast<size_t>(particle)] + 1e-6f ||
190 anchor != tetherAnchor[static_cast<size_t>(particle)])
191 continue;
192 for (const Neighbor& neighbor : adjacency[static_cast<size_t>(particle)]) {
193 const float nextDistance = distance + neighbor.distance;
194 const size_t next = static_cast<size_t>(neighbor.particle);
195 if (nextDistance + 1e-6f < tetherDistance[next] ||
196 (std::abs(nextDistance - tetherDistance[next]) <= 1e-6f &&
197 (tetherAnchor[next] < 0 || anchor < tetherAnchor[next]))) {
198 tetherDistance[next] = nextDistance;
199 tetherAnchor[next] = anchor;
200 queue.emplace(nextDistance, anchor, neighbor.particle);
201 }
202 }
203 }
204 for (size_t particle = 0; particle < particleCount; ++particle)
205 if (candidate.particles_[particle].inverseMass > 0.f && tetherAnchor[particle] >= 0)
206 candidate.tetherConstraints_.push_back(
207 {static_cast<int>(particle), tetherAnchor[particle], tetherDistance[particle]});
208
209 return eve::Result<ClothModel>::success(std::move(candidate));
210}

212eve::Result<ClothModel> ClothModel::grid(int cols, int rows, float spacing, float originX, float originY,
213 float originZ) {
214 if (cols < 2 || rows < 2) return invalidModel("cloth grid dimensions must be at least two");
215 if (!std::isfinite(spacing) || spacing <= 0.f)
216 return invalidModel("cloth grid spacing must be finite and positive");
217 if (!std::isfinite(originX) || !std::isfinite(originY) || !std::isfinite(originZ))
218 return invalidModel("cloth grid origin must be finite");
219 if (static_cast<size_t>(cols) > static_cast<size_t>(std::numeric_limits<int>::max()) / static_cast<size_t>(rows))
220 return invalidModel("cloth grid particle count exceeds the supported index range");
221
222 std::vector<float> positions;
223 std::vector<float> inverseMasses;
224 std::vector<int> indices;
225 positions.reserve(static_cast<size_t>(cols * rows * 3));
226 inverseMasses.reserve(static_cast<size_t>(cols * rows));
227 for (int row = 0; row < rows; ++row) {
228 for (int col = 0; col < cols; ++col) {
229 positions.push_back(originX + static_cast<float>(col) * spacing);
230 positions.push_back(originY);
231 positions.push_back(originZ + static_cast<float>(row) * spacing);
232 inverseMasses.push_back(row == 0 ? 0.f : 1.f);
233 }
234 }
235 for (int row = 0; row + 1 < rows; ++row) {
236 for (int col = 0; col + 1 < cols; ++col) {
237 const int a = row * cols + col;
238 const int b = a + 1;
239 const int d = (row + 1) * cols + col;
240 const int c = d + 1;
241 indices.insert(indices.end(), {a, d, b, d, c, b});
242 }
243 }
244 auto result = fromTriangles(positions, indices, inverseMasses);
245 if (!result) return result;
246 ClothModel model = std::move(result).takeValue();
247
248 // Preserve the established grid behavior and solver order exactly: PBD
249 // projection order affects finite-iteration results. Arbitrary meshes use
250 // the canonical edge order produced by fromTriangles above.
251 model.distanceConstraints_.clear();
252 auto addGridDistance = [&](int a, int b, ClothConstraintKind kind) {
253 const auto& pa = model.particles_[static_cast<size_t>(a)];
254 const auto& pb = model.particles_[static_cast<size_t>(b)];
255 const float dx = pb.x - pa.x, dy = pb.y - pa.y, dz = pb.z - pa.z;
256 model.distanceConstraints_.push_back({a, b, std::sqrt(dx * dx + dy * dy + dz * dz), kind});
257 };
258 for (int row = 0; row < rows; ++row) {
259 for (int col = 0; col < cols; ++col) {
260 const int particle = row * cols + col;
261 if (col + 1 < cols) addGridDistance(particle, particle + 1, ClothConstraintKind::Structural);
262 if (row + 1 < rows) addGridDistance(particle, particle + cols, ClothConstraintKind::Structural);
263 if (col + 1 < cols && row + 1 < rows)
264 addGridDistance(particle, particle + cols + 1, ClothConstraintKind::Shear);
265 if (col > 0 && row + 1 < rows) addGridDistance(particle, particle + cols - 1, ClothConstraintKind::Shear);
266 if (col + 2 < cols) addGridDistance(particle, particle + 2, ClothConstraintKind::Bend);
267 if (row + 2 < rows) addGridDistance(particle, particle + cols * 2, ClothConstraintKind::Bend);
268 }
269 }
270 model.gridCols_ = cols;
271 model.gridRows_ = rows;
272 model.gridSpacing_ = spacing;
273 return eve::Result<ClothModel>::success(std::move(model));
274}
