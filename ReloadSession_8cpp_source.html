15bool ReloadSession::begin(HSQUIRRELVM vm, std::string* err) {
16 if (active_) {
17 if (err) *err = "reload session already active";
18 return false;
19 }
20 persistentRoots_ = Snapshot::instance().roots();
21 transientRoots_ = Snapshot::instance().transientRoots();
22 if (!Snapshot::instance().captureState(vm, buffer_, err)) {
23 buffer_ = StateValue::null();
24 persistentRoots_.clear();
25 transientRoots_.clear();
26 return false;
27 }
28 // Values the capture could not represent (entity instances, closures, ...) are left
29 // out, not fatal: refusing the session here used to make "keep a list of entities in a
30 // global" block hot reload entirely. Say so once; the same list stays in the captured
31 // state and in eve_snapshot.json for a tool to read.
32 if (const StateValue* skipped = buffer_.find("skipped");
33 skipped && skipped->isArray() && skipped->arraySize() > 0) {
34 std::string names;
35 for (std::size_t i = 0; i < skipped->arraySize() && i < 3; ++i) {
36 if (!names.empty()) names += ", ";
37 names += skipped->at(i).asString();
38 }
39 if (skipped->arraySize() > 3) names += ", ...";
40 std::fprintf(stderr, "EVEngine: reload state skips %zu unserializable value(s): %s\n", skipped->arraySize(),
41 names.c_str());
42 }
43 // Best-effort manual rollback point (same file as the F6/F7 snapshot).
44 std::string saveErr;
45 Snapshot::instance().saveFile(vm, "eve_snapshot.json", &saveErr);
46 resetMcpFrameCapture();
47 active_ = true;
48 return true;
49}

51bool ReloadSession::commit(HSQUIRRELVM vm, std::string* err) {
52 if (!active_) {
53 if (err) *err = "no active reload session";
54 return false;
55 }
56 StateValue merged = buffer_;
57
58 // Merge freshly-defined roots under the captured values: captured values
59 // win, fields the reloaded script added are kept.
60 StateValue current;
61 std::string captureErr;
62 if (Snapshot::instance().captureState(vm, current, &captureErr)) {
63 StateValue* mergedRoots = merged.find("roots");
64 const StateValue* currentRoots = current.find("roots");
65 if (mergedRoots && mergedRoots->isObject() && currentRoots && currentRoots->isObject()) {
66 for (const auto& name : mergedRoots->keys()) {
67 StateValue* oldRoot = mergedRoots->find(name);
68 const StateValue* newRoot = currentRoots->find(name);
69 if (oldRoot && oldRoot->isObject() && newRoot && newRoot->isObject()) {
70 oldRoot->mergeDefaults(*newRoot);
71 }
72 }
73 }
74 }
75
76 const bool ok = restore(vm, merged, err);
77 // Keep the original buffer and the session open when restoration fails.
78 // The script orchestrator can then restore old root bindings and call
79 // abort(), instead of being stranded in a partially restored state.
80 if (ok) {
81 buffer_ = StateValue::null();
82 persistentRoots_.clear();
83 transientRoots_.clear();
84 active_ = false;
85 }
86 return ok;
87}
