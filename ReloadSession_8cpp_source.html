32bool ReloadSession::commit(HSQUIRRELVM vm, std::string* err) {
33 if (!active_) {
34 if (err) *err = "no active reload session";
35 return false;
36 }
37 StateValue merged = buffer_;
38
39 // Merge freshly-defined roots under the captured values: captured values
40 // win, fields the reloaded script added are kept.
41 StateValue current;
42 std::string captureErr;
43 if (Snapshot::instance().captureState(vm, current, &captureErr)) {
44 StateValue* mergedRoots = merged.find("roots");
45 const StateValue* currentRoots = current.find("roots");
46 if (mergedRoots && mergedRoots->isObject() && currentRoots && currentRoots->isObject()) {
47 for (const auto& name : mergedRoots->keys()) {
48 StateValue* oldRoot = mergedRoots->find(name);
49 const StateValue* newRoot = currentRoots->find(name);
50 if (oldRoot && oldRoot->isObject() && newRoot && newRoot->isObject()) {
51 oldRoot->mergeDefaults(*newRoot);
52 }
53 }
54 }
55 }
56
57 const bool ok = restore(vm, merged, err);
58 // Keep the original buffer and the session open when restoration fails.
59 // The script orchestrator can then restore old root bindings and call
60 // abort(), instead of being stranded in a partially restored state.
61 if (ok) {
62 buffer_ = StateValue::null();
63 persistentRoots_.clear();
64 transientRoots_.clear();
65 active_ = false;
66 }
