83void EditorShell::open(UIHostHandle inspector, UIHostHandle database, UIHostHandle scene) {
84 inspector_ = inspector;
85 database_ = database;
86 scene_ = scene;
87 auto host = UIHost::resolve(host_);
88 if (!host) {
89 host_ = UIHost::createHost(kShellHostName);
90 host = UIHost::resolve(host_);
91 }
92 if (!host) return;
93 UIHost &shell = host->get();
94 auto meta = shell.meta();
95 meta->overlay = true; // no title bar
96 meta->hasPos = true;
97 meta->lockPos = true;
98 meta->posX = 0.f;
99 meta->posY = 0.f;
100 meta->hasSize = true;
101 meta->sizeX = 0.f; // width = full display (percentW below)
102 meta->sizeY = toolbarHeight_;
103 meta->lockSize = true;
104 meta->percentW = 1.f;
105 meta->visible = true;
106 meta->overlayBgAlpha = 1.f;
107 meta->overlayFlush = true;
108 shell.setTree(build());
109 relayout();
110 setHostVisible(inspector_, true);
111 setHostVisible(database_, true);
112 setHostVisible(scene_, true);
113}

127bool EditorShell::selectPanel(const std::string &name) {
128 auto host = UIHost::resolve(host_);
129 if (!host || !host->get().meta()->visible) return false;
130 if (name == "inspector") {
131 setHostVisible(inspector_, true);
132 setHostVisible(database_, false);
133 setHostVisible(scene_, false);
134 return true;
135 }
136 if (name == "database") {
137 setHostVisible(database_, true);
138 setHostVisible(inspector_, false);
139 setHostVisible(scene_, false);
140 return true;
141 }
142 if (name == "scene") {
143 setHostVisible(scene_, true);
144 setHostVisible(inspector_, false);
145 setHostVisible(database_, false);
146 return true;
147 }
148 if (name.empty()) {
149 setHostVisible(inspector_, false);
150 setHostVisible(database_, false);
151 setHostVisible(scene_, false);
152 return true;
153 }
154 return false;
155}

157bool EditorShell::togglePanel(const std::string &name) {
158 auto host = UIHost::resolve(host_);
159 if (!host || !host->get().meta()->visible) return false;
160 UIHostHandle handle{};
161 if (name == "inspector") handle = inspector_;
162 if (name == "database") handle = database_;
163 if (name == "scene") handle = scene_;
164 auto panel = UIHost::resolve(handle);
165 if (!panel) return false;
166 UIHost &panelHost = panel->get();
167 panelHost.setVisible(!panelHost.meta()->visible);
168 return true;
169}
