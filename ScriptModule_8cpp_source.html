19std::string normalizedPath(std::string value, bool& escaped) {
20 std::replace(value.begin(), value.end(), '\\', '/');
21 std::vector<std::string> parts;
22 size_t begin = 0;
23 while (begin <= value.size()) {
24 const size_t end = value.find('/', begin);
25 const std::string part = value.substr(begin, end - begin);
26 if (part.empty() || part == ".") {
27 } else if (part == "..") {
28 if (parts.empty()) {
29 escaped = true;
30 return {};
31 }
32 parts.pop_back();
33 } else {
34 parts.push_back(part);
35 }
36 if (end == std::string::npos) break;
37 begin = end + 1;
38 }
39 std::ostringstream out;
40 for (size_t i = 0; i < parts.size(); ++i) {
41 if (i) out << '/';
42 out << parts[i];
43 }
44 return out.str();
45}

58class GameFileProvider final : public IScriptModuleProvider {
59public:
60 ScriptModuleStatus resolve(const ScriptModuleRequest& request, std::string& canonicalUri,
61 std::string& error) override {
62 if (!ScriptModuleResolver::canonicalize(request, canonicalUri, error)) return ScriptModuleStatus::Error;
63 return canonicalUri.rfind("game:/", 0) == 0 ? ScriptModuleStatus::Found : ScriptModuleStatus::NotHandled;
64 }
65
66 ScriptModuleStatus load(std::string_view canonicalUri, ScriptModuleSource& source, std::string& error) override {
67 if (canonicalUri.rfind("game:/", 0) != 0) return ScriptModuleStatus::NotHandled;
68 const std::string path(canonicalUri.substr(6));
69 std::vector<uint8_t> bytes;
70 if (auto* filesystem = eve::cap::query<eve::service::IFileSystem>();
71 filesystem && filesystem->readFile(path, bytes)) {
72 source.utf8Source.assign(reinterpret_cast<const char*>(bytes.data()), bytes.size());
73 } else {
74 std::ifstream input(path, std::ios::binary);
75 if (!input) {
76 error = "script module not found: " + std::string(canonicalUri);
77 return ScriptModuleStatus::Error;
78 }
79 source.utf8Source.assign(std::istreambuf_iterator<char>(input), {});
80 }
81 source.canonicalUri = std::string(canonicalUri);
82 source.debugOrigin = path;
83 source.contentHash = hashSource(source.utf8Source);
84 return ScriptModuleStatus::Found;
85 }

181 const std::string canonical = resolve({std::string(importer), std::string(specifier)});
182 if (const auto found = modules.find(canonical); found != modules.end()) {

185 return canonical;

221 if (module.state == Module::State::Ready) return;

258 std::vector<std::string> reloadAffected(const std::string& canonical) {
259 if (modules.find(canonical) == modules.end()) return {};
260
261 std::unordered_set<std::string> affected{canonical};
262 bool changed = true;
263 while (changed) {
264 changed = false;
265 for (const auto& [importer, edges] : dependencies) {
266 if (affected.count(importer) || modules.find(importer) == modules.end()) continue;
267 if (std::any_of(edges.begin(), edges.end(), [&](const auto& edge) { return affected.count(edge); })) {
268 affected.insert(importer);
269 changed = true;
270 }
271 }
272 }
273
274 std::vector<std::string> order(affected.begin(), affected.end());
275 std::sort(order.begin(), order.end());
276 const auto previousEdges = dependencies;
277 std::unordered_set<std::string> previousKeys;
278 for (const auto& [uri, _] : modules) previousKeys.insert(uri);
279
280 std::unordered_map<std::string, Module> previous;
281 for (const std::string& uri : order) {
282 auto found = modules.find(uri);
283 if (found == modules.end()) continue;
284 previous.emplace(uri, std::move(found->second));
285 modules.erase(found);
286 dependencies.erase(uri);
287 }
288
289 try {
290 for (const std::string& uri : order) ensureCompiled("game:/__hot_reload__.nut", uri);
291 for (const std::string& uri : order) instantiate(uri);
292 } catch (...) {
293 for (auto it = modules.begin(); it != modules.end();) {
294 if (affected.count(it->first) || !previousKeys.count(it->first)) {
295 release(it->second);
296 it = modules.erase(it);
297 } else {
298 ++it;
299 }
300 }
301 for (auto& [uri, module] : previous) modules.emplace(uri, std::move(module));
302 dependencies = previousEdges;
303 throw;
304 }
305
306 for (auto& [_, module] : previous) release(module);
307 return order;
308 }

352void ScriptModuleResolver::prepareDependencies(std::string_view importerUri) {
353 enum class Visit { Visiting, Ready };
354 std::unordered_map<std::string, Visit> visits;
355 std::vector<std::string> stack;
356 const auto verify = [&](const auto& self, const std::string& uri) -> void {
357 if (const auto seen = visits.find(uri); seen != visits.end()) {
358 if (seen->second == Visit::Ready) return;
359 std::string cycle;
360 const auto begin = std::find(stack.begin(), stack.end(), uri);
361 for (auto it = begin; it != stack.end(); ++it) {
362 if (!cycle.empty()) cycle += " -> ";
363 cycle += *it;
364 }
365 if (!cycle.empty()) cycle += " -> ";
366 throw std::runtime_error("cyclic script import: " + cycle + uri);
367 }
368
369 visits.emplace(uri, Visit::Visiting);
370 stack.push_back(uri);
371 const auto edges = impl_->dependencies.find(uri);
372 if (edges != impl_->dependencies.end()) {
373 for (const std::string& dependency : edges->second) {
374 if (impl_->modules.find(dependency) == impl_->modules.end())
375 throw std::runtime_error("script dependency was not compiled: " + dependency);
376 self(self, dependency);
377 }
378 }
379 stack.pop_back();
380 visits[uri] = Visit::Ready;
381 };
382
383 const std::string root(importerUri);
384 const auto edges = impl_->dependencies.find(root);
385 if (edges == impl_->dependencies.end()) return;
386 for (const std::string& dependency : edges->second) verify(verify, dependency);
387}

442bool ScriptModuleResolver::canonicalize(const ScriptModuleRequest& request, std::string& output, std::string& error) {
443 std::string scheme = "game:";
444 std::string path;
445 const size_t schemeEnd = request.specifier.find(":/");
446 if (schemeEnd != std::string::npos) {
447 scheme = request.specifier.substr(0, schemeEnd + 1);
448 path = request.specifier.substr(schemeEnd + 2);
449 } else if (request.specifier.rfind("./", 0) == 0 || request.specifier.rfind("../", 0) == 0) {
450 std::string importer = request.importerUri;
451 const size_t importerScheme = importer.find(":/");
452 if (importerScheme != std::string::npos) {
453 scheme = importer.substr(0, importerScheme + 1);
454 importer = importer.substr(importerScheme + 2);
455 }
456 const size_t slash = importer.find_last_of("/\\");
457 path = (slash == std::string::npos ? std::string() : importer.substr(0, slash + 1)) + request.specifier;
458 } else {
459 path = request.specifier;
460 }
461 bool escaped = false;
462 path = normalizedPath(path, escaped);
463 if (escaped || path.empty()) {
464 error = escaped ? "script import escapes its module root: " + request.specifier : "empty script module path";
465 return false;
466 }
467 output = scheme + "/" + path;
468 return true;
469}
