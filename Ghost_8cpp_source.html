150void Ghost::setEdge(PlacementWorld *world, int cellX, int cellY, const std::string &direction) {
151 if (!world) return;
152 auto result = PlacementSystem::canonicalEdge(cellX, cellY, direction);
153 if (!result.ok()) {
154 placementKind_ = "edge";
155 valid_ = false;
156 reason_ = "invalid_edge_direction";
157 return;
158 }
159 placementKind_ = "edge";
160 edge_ = std::move(result).takeValue();
161 cellX_ = edge_.x;
162 cellY_ = edge_.y;
163 float ax = 0.f, ay = 0.f, bx = 0.f, by = 0.f;
164 world->cellToWorldPlane(edge_.x, edge_.y, ax, ay);
165 world->cellToWorldPlane(edge_.x + (edge_.axis == EdgeAxis::Horizontal ? 1 : 0),
166 edge_.y + (edge_.axis == EdgeAxis::Vertical ? 1 : 0), bx, by);
167 worldX_ = (ax + bx) * 0.5f;
168 worldY_ = (ay + by) * 0.5f;
169 rotationDeg_ = edge_.axis == EdgeAxis::Horizontal ? 0.f : 90.f;
170 surfaceId_.clear();
171 surfaceRevision_ = 0;
172 valid_ = false;
173 reason_.clear();
174}

251bool Ghost::validate(PlacementWorld *world) {
252 if (!world) {
253 valid_ = false;
254 reason_ = "no_world";
255 return false;
256 }
257 const BuildingDefinition *definition = BuildingRegistry::find(buildingId_);
258 if (definition && definition->placementKind == "edge") {
259 if (placementKind_ != "edge") {
260 valid_ = false;
261 reason_ = "edge_requires_edge_address";
262 return false;
263 }
264 const char *direction = edge_.axis == EdgeAxis::Horizontal ? "north" : "west";
265 valid_ = PlacementSystem::canPlaceEdge(world, buildingId_, edge_.x, edge_.y, direction, 0,
266 &reason_);
267 return valid_;
268 }
269 if (definition && definition->placementKind == "corner") {
270 if (placementKind_ != "corner") {
271 valid_ = false;
272 reason_ = "corner_requires_corner_address";
273 return false;
274 }
275 valid_ = PlacementSystem::canPlaceCorner(world, buildingId_, corner_.x, corner_.y, 0,
276 &reason_);
277 return valid_;
278 }
279 if (definition && definition->placementKind == "free") {
280 if (placementKind_ != "free") {
281 valid_ = false;
282 reason_ = "free_requires_free_address";
283 return false;
284 }
285 valid_ = PlacementSystem::canPlaceFree(
286 world, buildingId_, worldX_, worldY_, 0, &reason_,
287 std::numeric_limits<int>::min(), rotationDeg_);
288 if (valid_ && surfaceMaxSlopeDegrees_ > definition->maxSurfaceSlopeDegrees) {
289 valid_ = false;
290 reason_ = "surface_slope";
291 }
292 if (valid_ && definition->maxSurfaceHeightDelta >= 0.f &&
293 surfaceHeightDelta_ > definition->maxSurfaceHeightDelta) {
294 valid_ = false;
295 reason_ = "surface_height_delta";
296 }
297 return valid_;
298 }
299 if (surfacePatchStale_) {
300 setFromSurface(world, surfaceProviderName_, surfaceInputX_, surfaceInputY_);
301 if (surfaceId_.empty()) return false;
302 }
303 rotationDeg_ = PlacementSystem::normalizeRotation(buildingId_, rotationDeg_);
304 PlacementQuery query;
305 query.buildingId = buildingId_;
306 query.cellX = cellX_;
307 query.cellY = cellY_;
308 query.worldX = worldX_;
309 query.worldY = worldY_;
310 query.elevation = elevation_;
311 query.rotationDeg = rotationDeg_;
312 query.surfaceId = surfaceId_;
313 query.surfaceRevision = surfaceRevision_;
314 query.surfaceNormalX = surfaceNormalX_;
315 query.surfaceNormalY = surfaceNormalY_;
316 query.surfaceNormalZ = surfaceNormalZ_;
317 query.surfaceSampleCount = surfaceSampleCount_;
318 query.surfaceMaxSlopeDegrees = surfaceMaxSlopeDegrees_;
319 query.surfaceHeightDelta = surfaceHeightDelta_;
320 std::string reason;
321 valid_ = PlacementSystem::canPlaceQuery(world, query, &reason);
322 reason_ = reason;
323 return valid_;
324}
