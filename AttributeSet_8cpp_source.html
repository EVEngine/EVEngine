30eve::Result<ParsedAttributeOperation> parseAttributeOperation(std::string_view operation, double value) {
31 ParsedAttributeOperation parsed;
32 if (operation == "add") {
33 parsed.operation = AttributeOperation::Add;
34 } else if (operation == "additive_percent" || operation == "mulAdd") {
35 parsed.operation = AttributeOperation::AdditivePercent;
36 } else if (operation == "multiplicative_percent" || operation == "mulMul") {
37 parsed.operation = AttributeOperation::MultiplicativePercent;
38 } else if (operation == "multiply") {
39 parsed.operation = AttributeOperation::MultiplicativePercent;
40 value -= 1.0;
41 } else if (operation == "override") {
42 parsed.operation = AttributeOperation::Override;
43 } else if (operation == "clamp_min" || operation == "clampMin" || operation == "min") {
44 parsed.operation = AttributeOperation::ClampMin;
45 } else if (operation == "clamp_max" || operation == "clampMax" || operation == "max") {
46 parsed.operation = AttributeOperation::ClampMax;
47 } else {
48 return eve::Result<ParsedAttributeOperation>::failure(
49 invalidArgument("unknown attribute operation", std::string(operation)));
50 }
51 parsed.value = value;
52 return eve::Result<ParsedAttributeOperation>::success(std::move(parsed));

100double computeAttributeValue(const AttributeValue& attribute, const AttributeOperationRegistry* customOperations) {
101 double additive = 0.0;
102 double additivePercent = 0.0;
103 double multiplicativePercent = 1.0;
104 std::vector<const AttributeModifier*> ordered;
105 ordered.reserve(attribute.modifiers.size());
106
107 for (const auto& modifier : attribute.modifiers) {
108 switch (modifier.operation) {
109 case AttributeOperation::Add: additive += modifier.value; break;
110 case AttributeOperation::AdditivePercent: additivePercent += modifier.value; break;
111 case AttributeOperation::MultiplicativePercent: multiplicativePercent *= 1.0 + modifier.value; break;
112 case AttributeOperation::Override:
113 case AttributeOperation::ClampMin:
114 case AttributeOperation::ClampMax:
115 case AttributeOperation::Custom: ordered.push_back(&modifier); break;
116 }
117 }
118
119 double result = (attribute.base + additive) * (1.0 + additivePercent);
120 result *= multiplicativePercent;
121
122 std::stable_sort(ordered.begin(), ordered.end(), [](const auto* left, const auto* right) {
123 if (left->priority != right->priority) return left->priority < right->priority;
124 return left->sequence < right->sequence;
125 });
126
127 for (const auto* modifier : ordered) {
128 switch (modifier->operation) {
129 case AttributeOperation::Override: result = modifier->value; break;
130 case AttributeOperation::ClampMin: result = std::max(result, modifier->value); break;
131 case AttributeOperation::ClampMax: result = std::min(result, modifier->value); break;
132 case AttributeOperation::Custom:
133 if (customOperations && customOperations->has(modifier->policyId))
134 result = customOperations->apply(modifier->policyId, result, modifier->value);
135 break;
136 case AttributeOperation::Add:
137 case AttributeOperation::AdditivePercent:
138 case AttributeOperation::MultiplicativePercent: break;
139 }
140 }
141 return result;
142}

233eve::Result<ModifierId> AttributeSet::addModifier(AttributeModifier modifier) {
234 if (modifier.attribute.empty())
235 return eve::Result<ModifierId>::failure(
236 invalidArgument("attribute modifier requires an attribute id", "attribute"));
237 if (!isValidOperation(modifier))
238 return eve::Result<ModifierId>::failure(
239 invalidArgument("attribute modifier has an invalid operation", "operation"));
240 if (!std::isfinite(modifier.value))
241 return eve::Result<ModifierId>::failure(invalidArgument("attribute modifier value must be finite", "value"));
242
243 if (modifier.id.empty()) {
244 do {
245 modifier.id = "attribute:modifier:";
246 if (!subject_.empty()) modifier.id += subject_ + ":";
247 modifier.id += std::to_string(nextModifierId_++);
248 } while ([&] {
249 for (const auto& [attribute, state] : values_)
250 for (const auto& existing : state.modifiers)
251 if (existing.id == modifier.id) return true;
252 return false;
253 }());
254 }
255 const ModifierId modifierId = modifier.id;
256 const AttributeId attributeId = modifier.attribute;
257 removeModifier(modifierId).ignore("replace existing attribute modifier");
258 modifier.sequence = nextSequence_++;
259 auto& state = values_[attributeId];
260 state.modifiers.push_back(std::move(modifier));
261 state.dirty = true;
262 orderDirty_ = true;
263 return eve::Result<ModifierId>::success(modifierId, eve::Status::success(eve::StatusCode::Applied));
264}

276eve::Result<void> AttributeSet::removeModifier(const ModifierId& id) {
277 if (id.empty()) return eve::Result<void>::failure(invalidArgument("modifier id must not be empty", "id"));
278 for (auto& [attribute, state] : values_) {

280 [&](const auto& modifier) { return modifier.id == id; });

315eve::Result<int> AttributeSet::removeBySource(const SourceId& source, const AttributeId& attribute) {
316 if (source.empty()) return eve::Result<int>::failure(invalidArgument("source id must not be empty", "source"));
317 int removed = 0;
318 for (auto& [attributeId, state] : values_) {
319 if (!attribute.empty() && attributeId != attribute) continue;
320 const auto oldSize = state.modifiers.size();
321 state.modifiers.erase(std::remove_if(state.modifiers.begin(), state.modifiers.end(),
322 [&](const auto& modifier) { return modifier.source == source; }),
323 state.modifiers.end());
324 if (state.modifiers.size() != oldSize) {
325 state.dirty = true;
326 removed += static_cast<int>(oldSize - state.modifiers.size());
327 }
328 }
329 if (removed != 0) orderDirty_ = true;
330 const auto code = removed == 0 ? eve::StatusCode::NoOp : eve::StatusCode::Applied;
331 return eve::Result<int>::success(removed, eve::Status::success(code));
332}
