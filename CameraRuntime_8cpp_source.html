15EditorResult<void> CameraDocumentRuntime::publish(const CameraDocumentTarget& document,
16 graphics::Camera3D* camera) {
17 const auto diagnostics = document.validate();
18 for (const auto& diagnostic : diagnostics)
19 if (diagnostic.severity() == DiagnosticSeverity::Error)
20 return EditorResult<void>::failure(eve::Status(EditorStatus::Rejected, diagnostics));
21 auto candidate = std::make_unique<camera::CameraController>();
22 candidate->setCamera(camera);
23 for (const auto& rig : document.rigs()) {
24 candidate->setMode(rig.mode);
25 candidate->setTarget(rig.target.x, rig.target.y, rig.target.z);
26 candidate->setOffset(rig.offset.x, rig.offset.y, rig.offset.z);
27 candidate->setLookAhead(rig.lookAhead.x, rig.lookAhead.y, rig.lookAhead.z);
28 candidate->setComposition(rig.composition.x, rig.composition.y);
29 candidate->setRadius(rig.radius);
30 candidate->setAzimuth(rig.azimuth);
31 candidate->setElevation(rig.elevation);
32 candidate->setYaw(rig.yaw);
33 candidate->setPitch(rig.pitch);
34 candidate->setFov(rig.fov);
35 candidate->setSmooth(rig.smooth);
36 candidate->setMaxSpeed(rig.maxSpeed);
37 if (!candidate->addRig(rig.id.value(), rig.mode, rig.priority) ||
38 !candidate->saveRigState(rig.id.value()) ||
39 !candidate->setRigEnabled(rig.id.value(), rig.enabled))
40 return fail<void>(EditorStatus::Failed, "editor.camera.runtime-rig",
41 "Camera runtime rejected a validated rig");
42 }
43 for (const auto& key : document.timeline()) {
44 bool accepted = false;
45 if (key.kind == "cut") accepted = candidate->addTimelineCut(key.time, key.rig.value(), key.blend);
46 else if (key.kind == "float") accepted = candidate->addTimelineFloat(key.time, key.property, key.value);
47 else if (key.kind == "event") accepted = candidate->addTimelineEvent(key.time, key.name, key.data);
48 if (!accepted)
49 return fail<void>(EditorStatus::Failed, "editor.camera.runtime-key",
50 "Camera runtime rejected a validated timeline key");
51 }
52 controller_ = std::move(candidate);
53 revision_ = document.revision();
54 return eve::editing::applied<void>(diagnostics);
55}
