78Result<std::uint64_t> ClimbingControl::refreshRevision() const {
79 if (!runtime_ || !world_ || !pose_)
80 return failure<std::uint64_t>(DiagnosticCode::StaleHandle,
81 "climbing control collaborators are invalid", "instance");
82 auto snapshot = runtime_->snapshotJson();
83 if (!snapshot) return Result<std::uint64_t>::failure(snapshot.status());
84 if (snapshot.value() != stateFingerprint_) {
85 stateFingerprint_ = snapshot.value();
86 ++revision_;
87 }
88 return Result<std::uint64_t>::success(revision_);

91Result<GameplayObservation> ClimbingControl::observeGameplay(const GameplaySession& session,
92 SubjectRef instance) const {
93 if (instance != instance_ || !instance_.isValid())
94 return failure<GameplayObservation>(DiagnosticCode::NotFound,
95 "climbing gameplay instance was not found", "instance");
96 if (!controls(session))
97 return failure<GameplayObservation>(DiagnosticCode::PreconditionViolation,
98 "session does not control this character", "instance");
99 auto revision = refreshRevision();
100 if (!revision) return Result<GameplayObservation>::failure(revision.status());
101 auto snapshot = runtime_->snapshot();
102 if (!snapshot) return Result<GameplayObservation>::failure(snapshot.status());
103 GameplayObservation observation;
104 observation.domain = id("gameplay:climbing");
105 observation.instance = instance_;
106 observation.tick = tick_;
107 observation.revision = revision.value();
108 observation.state = Value(Value::Object{
109 {"phase", Value(phaseName(runtime_->phase()))},
110 {"runtime", std::move(snapshot).takeValue()},
111 });
112 return Result<GameplayObservation>::success(std::move(observation));
113}

115Result<std::vector<GameplayActionDescriptor>> ClimbingControl::availableGameplayActions(
116 const GameplaySession& session, SubjectRef instance, SubjectRef subject) const {
117 auto observed = observeGameplay(session, instance);
118 if (!observed) return Result<std::vector<GameplayActionDescriptor>>::failure(observed.status());
119 std::move(observed).takeValue();
120 if (subject != character_)
121 return failure<std::vector<GameplayActionDescriptor>>(
122 DiagnosticCode::PreconditionViolation, "climbing action subject must be the character", "subject");
123 const Value emptySchema(Value::Object{});
124 std::vector<GameplayActionDescriptor> actions;
125 const auto phase = runtime_->phase();
126 if (terminal(phase)) actions.push_back({id("climbing:begin-best"), emptySchema});
127 if (phase == ClimbingPhase::Hanging) {
128 actions.push_back({id("climbing:climb-up"), emptySchema});
129 actions.push_back({id("climbing:drop"), emptySchema});
130 }
131 if (!terminal(phase)) actions.push_back({id("climbing:cancel"), emptySchema});
132 return Result<std::vector<GameplayActionDescriptor>>::success(std::move(actions));
133}

135Result<void> ClimbingControl::captureEvents(std::string_view commandId) {
136 auto drained = runtime_->drainEvents();
137 if (!drained) return Result<void>::failure(drained.status());
138 for (auto& source : std::move(drained).takeValue()) {
139 GameplayEvent event;
140 event.sequence = nextEventSequence_++;
141 event.tick = source.tick;
142 event.type = "climbing." + eventName(source.kind);
143 event.subject = character_;
144 event.causationCommandId = std::string(commandId);
145 event.correlationId = std::string(commandId);
146 Value::Array metadata;
147 for (auto& item : source.metadata) metadata.emplace_back(std::move(item));
148 event.payload = Value(Value::Object{
149 {"actionId", Value(std::move(source.actionId))},
150 {"executionId", Value(static_cast<std::int64_t>(source.executionId.value()))},
151 {"metadata", Value(std::move(metadata))},
152 });
153 events_.push_back(std::move(event));
154 }
155 return Result<void>::success();
156}

158Result<GameplayCommandReceipt> ClimbingControl::submitGameplay(const GameplaySession& session,
159 SubjectRef instance,
160 const GameplayCommand& command) {
161 if (instance != instance_)
162 return failure<GameplayCommandReceipt>(DiagnosticCode::NotFound,
163 "climbing gameplay instance was not found", "instance");
164 if (!controls(session) || command.subject != character_)
165 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation,
166 "session does not control this character", "command.subject");
167 if (command.id.empty())
168 return failure<GameplayCommandReceipt>(DiagnosticCode::InvalidArgument,
169 "command id must not be empty", "command.id");
170 auto revision = refreshRevision();
171 if (!revision) return Result<GameplayCommandReceipt>::failure(revision.status());
172 if (command.observedTick != tick_ || command.expectedRevision != revision.value())
173 return failure<GameplayCommandReceipt>(DiagnosticCode::Conflict,
174 "climbing command was based on a stale observation",
175 "command.expectedRevision");
176
177 std::string operation;
178 if (command.action == id("climbing:begin-best")) {
179 auto begun = runtime_->tryBegin(*world_, *pose_, tick_);
180 if (!begun) return Result<GameplayCommandReceipt>::failure(begun.status());
181 std::move(begun).takeValue();
182 operation = "begin-best";
183 } else if (command.action == id("climbing:climb-up")) {
184 auto climbed = runtime_->climbUp(tick_);
185 if (!climbed) return Result<GameplayCommandReceipt>::failure(climbed.status());
186 operation = "climb-up";
187 } else if (command.action == id("climbing:drop")) {
188 auto dropped = runtime_->drop(tick_);
189 if (!dropped) return Result<GameplayCommandReceipt>::failure(dropped.status());
190 operation = "drop";
191 } else if (command.action == id("climbing:cancel")) {
192 auto cancelled = runtime_->cancel(ClimbingCancelReason::PlayerRequest, tick_);
193 if (!cancelled) return Result<GameplayCommandReceipt>::failure(cancelled.status());
194 operation = "cancel";
195 } else {
196 return failure<GameplayCommandReceipt>(DiagnosticCode::Unsupported,
197 "unsupported climbing action", "command.action");
198 }
199 auto captured = captureEvents(command.id);
200 if (!captured) return Result<GameplayCommandReceipt>::failure(captured.status());
201 stateFingerprint_.clear();
202 auto resultingRevision = refreshRevision();
203 if (!resultingRevision) return Result<GameplayCommandReceipt>::failure(resultingRevision.status());
204 GameplayCommandReceipt receipt;
205 receipt.commandId = command.id;
206 receipt.executionId = "climbing-" + std::to_string(runtime_->executionId().value());
207 receipt.acceptedTick = tick_;
208 receipt.resultingRevision = resultingRevision.value();
209 receipt.details = Value(Value::Object{{"operation", Value(operation)},
210 {"phase", Value(phaseName(runtime_->phase()))}});
211 return Result<GameplayCommandReceipt>::success(std::move(receipt),
212 Status::success(StatusCode::Applied));
213}

215Result<GameplayObservation> ClimbingControl::advanceGameplay(const GameplaySession& session,
216 SubjectRef instance,
217 const SimulationStep& step) {
218 if (step.tick <= tick_)
219 return failure<GameplayObservation>(DiagnosticCode::Conflict,
220 "climbing simulation tick must increase", "step.tick");
221 auto observed = observeGameplay(session, instance);
222 if (!observed) return Result<GameplayObservation>::failure(observed.status());
223 std::move(observed).takeValue();
224 if (!terminal(runtime_->phase())) {
225 auto advanced = runtime_->advance(*world_, step);
226 if (!advanced) return Result<GameplayObservation>::failure(advanced.status());
227 std::move(advanced).takeValue();
228 auto captured = captureEvents("advance");
229 if (!captured) return Result<GameplayObservation>::failure(captured.status());
230 }
231 tick_ = step.tick;
232 stateFingerprint_.clear();
233 return observeGameplay(session, instance);
234}

236Result<std::vector<GameplayEvent>> ClimbingControl::gameplayEvents(const GameplaySession& session,
237 SubjectRef instance,
238 std::uint64_t afterSequence) const {
239 auto observed = observeGameplay(session, instance);
240 if (!observed) return Result<std::vector<GameplayEvent>>::failure(observed.status());
241 std::move(observed).takeValue();
242 std::vector<GameplayEvent> result;
243 for (const auto& event : events_)
244 if (event.sequence > afterSequence) result.push_back(event);
245 const bool empty = result.empty();
246 return Result<std::vector<GameplayEvent>>::success(
247 std::move(result), Status::success(empty ? StatusCode::NoOp : StatusCode::Applied));
248}
