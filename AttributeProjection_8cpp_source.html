36Result<void> AttributeProjection::bindOwner(ecs::EntityHandle owner) {
37 if (owner.table == nullptr || ecs::try_get(owner) == nullptr)
38 return failureVoid(DiagnosticCode::StaleHandle, "attribute projection owner is not a live ECS generation",

122Result<AttributeProjectionSnapshot> AttributeProjection::snapshot(std::span<const std::string_view> attributes) const {
123 for (const auto name : attributes) {
124 if (name.empty())
125 return failure<AttributeProjectionSnapshot>(DiagnosticCode::InvalidArgument,
126 "snapshot attribute name is empty", "attributes");
127 }
128 AttributeProjectionSnapshot result;
129 result.owner = owner_;
130 result.capturedRevision = revision_;
131 result.bases.reserve(attributes.size());
132 for (const auto name : attributes)
133 result.bases.push_back({std::string(name), values_.getBase(std::string(name), 0.0)});
134
135 for (int index = 0; index < values_.modifierCount(); ++index) {
136 const auto* modifier = values_.modifierAt(index);
137 if (modifier != nullptr && contains(attributes, modifier->attribute)) result.modifiers.push_back(*modifier);
138 }
139 return Result<AttributeProjectionSnapshot>::success(std::move(result));
140}

142Result<void> AttributeProjection::restore(const AttributeProjectionSnapshot& snapshotValue, Revision expectedRevision) {
143 if (!ownsSameLiveEntity(snapshotValue.owner))
144 return failureVoid(DiagnosticCode::StaleHandle, "attribute snapshot owner is stale or does not match", "owner");
145 if (revision_ != expectedRevision)
146 return failureVoid(DiagnosticCode::Conflict, "attribute snapshot revision does not match current state",
147 "revision");
148
149 std::vector<std::string_view> names;
150 names.reserve(snapshotValue.bases.size());
151 for (const auto& base : snapshotValue.bases) {
152 if (base.attribute.empty() || !std::isfinite(base.base))
153 return failureVoid(DiagnosticCode::InvalidArgument, "attribute snapshot contains invalid base data",
154 "bases");
155 names.push_back(base.attribute);
156 }
157
158 AttributeSet candidate = values_;
159 std::vector<ModifierId> toRemove;
160 for (int index = 0; index < candidate.modifierCount(); ++index) {
161 const auto* modifier = candidate.modifierAt(index);
162 if (modifier != nullptr && contains(names, modifier->attribute)) toRemove.push_back(modifier->id);
163 }
164 for (const auto& id : toRemove) {
165 auto removed = candidate.removeModifier(id);
166 if (!removed)
167 return failureVoid(DiagnosticCode::InvariantViolation,
168 "attribute snapshot could not remove candidate modifier", "modifiers");
169 }
170 for (const auto& base : snapshotValue.bases) candidate.setBase(base.attribute, base.base);
171
172 auto modifiers = snapshotValue.modifiers;
173 std::stable_sort(modifiers.begin(), modifiers.end(),
174 [](const auto& left, const auto& right) { return left.sequence < right.sequence; });
175 for (const auto& modifier : modifiers) {
176 if (!contains(names, modifier.attribute))
177 return failureVoid(DiagnosticCode::InvalidArgument, "attribute snapshot modifier is outside its allow-list",
178 "modifiers");
179 auto added = candidate.addModifier(modifier);
180 if (!added)
181 return failureVoid(DiagnosticCode::InvalidArgument, "attribute snapshot contains an invalid modifier",
182 "modifiers");
183 }
184
185 auto next = advanceRevision();
186 if (!next) return next;
187 values_ = std::move(candidate);
188 initialized_ = true;
189 return next;
190}
