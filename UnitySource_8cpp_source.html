10Result<std::string> metaScalar(std::span<const std::uint8_t> bytes, std::string_view key, std::string_view path) {
11 if (std::find(bytes.begin(), bytes.end(), 0) != bytes.end())
12 return failure<std::string>(DiagnosticCode::ParseError, "binary Unity metadata", std::string(path));
13 const std::string_view text(reinterpret_cast<const char*>(bytes.data()), bytes.size());
14 if (!isValidUtf8(text, Utf8NullPolicy::Reject))
15 return failure<std::string>(DiagnosticCode::ParseError, "Unity metadata is not UTF-8", std::string(path));
16 std::string value;
17 bool found = false;
18 for (std::size_t at = 0; at < text.size();) {
19 auto end = text.find('\n', at);
20 if (end == std::string_view::npos) end = text.size();
21 auto line = text.substr(at, end - at);
22 if (at == 0 && line.starts_with("\xef\xbb\xbf")) line.remove_prefix(3);
23 if (line.starts_with(key) && line.size() > key.size() && line[key.size()] == ':') {
24 if (found)
25 return failure<std::string>(DiagnosticCode::Conflict,
26 "duplicate Unity metadata key: " + std::string(key), std::string(path));
27 found = true;
28 line.remove_prefix(key.size() + 1);
29 const auto first = line.find_first_not_of(" \t\r");
30 if (first != std::string_view::npos)
31 value = std::string(line.substr(first, line.find_last_not_of(" \t\r") - first + 1));
32 }
33 at = end + 1;
34 }
35 return Result<std::string>::success(std::move(value));
36}

44UnitySourceKind classify(std::string_view path) {
45 const auto dot = path.find_last_of('.');
46 const auto ext = dot == std::string_view::npos ? std::string{} : foldAscii(std::string(path.substr(dot)));
47 if (ext == ".prefab") return UnitySourceKind::Prefab;
48 if (ext == ".unity") return UnitySourceKind::Scene;
49 if (ext == ".fbx" || ext == ".obj" || ext == ".gltf" || ext == ".glb" || ext == ".blend" || ext == ".dae")
50 return UnitySourceKind::Model;
51 if (ext == ".mat") return UnitySourceKind::Material;
52 if (ext == ".png" || ext == ".jpg" || ext == ".jpeg" || ext == ".tga" || ext == ".psd" || ext == ".exr" ||
53 ext == ".tif" || ext == ".tiff" || ext == ".bmp")
54 return UnitySourceKind::Image;
55 if (ext == ".anim" || ext == ".controller" || ext == ".overridecontroller") return UnitySourceKind::Animation;
56 if (ext == ".wav" || ext == ".ogg" || ext == ".mp3" || ext == ".aiff" || ext == ".flac")
57 return UnitySourceKind::Audio;
58 if (ext == ".ttf" || ext == ".otf") return UnitySourceKind::Font;
59 if (ext == ".shader" || ext == ".shadergraph" || ext == ".shadersubgraph" || ext == ".compute")
60 return UnitySourceKind::Shader;
61 if (ext == ".cs" || ext == ".dll" || ext == ".asmdef") return UnitySourceKind::Script;
62 if (ext == ".asset" || ext == ".terrainlayer") return UnitySourceKind::Data;
63 return UnitySourceKind::Other;
64}

71Result<std::vector<UnitySourceReference>> references(std::span<const std::uint8_t> bytes, const std::string& path,
72 const AssetImportLimits& limits) {
73 std::vector<UnitySourceReference> out;
74 if (bytes.empty() || std::find(bytes.begin(), bytes.end(), 0) != bytes.end())
75 return Result<std::vector<UnitySourceReference>>::success(std::move(out));
76 const std::string_view text(reinterpret_cast<const char*>(bytes.data()), bytes.size());
77 const std::regex guidField(R"(\bguid:\s*([^,}\s]+))");
78 const std::regex idField(R"(\bfileID:\s*([^,}\s]+))");
79 for (std::size_t at = 0; at < text.size();) {
80 const auto begin = text.find('{', at);
81 if (begin == std::string_view::npos) break;
82 const auto end = text.find_first_of("{}\r\n", begin + 1);
83 if (end == std::string_view::npos) break;
84 at = text[end] == '{' ? end : end + 1;
85 if (text[end] != '}') continue;
86 if (end - begin > limits.maximumStringBytes)
87 return failure<std::vector<UnitySourceReference>>(
88 DiagnosticCode::InvalidArgument, "Unity serialized flow record exceeds string budget", path);
89 const std::string ref(text.substr(begin, end - begin + 1));
90 std::smatch guid, id;
91 if (!std::regex_search(ref, guid, guidField)) continue;
92 if (!validGuid(guid[1].str()) || !std::regex_search(ref, id, idField))
93 return failure<std::vector<UnitySourceReference>>(DiagnosticCode::ParseError,
94 "malformed Unity object reference", path);
95 const auto idText = id[1].str();
96 std::int64_t fileId = 0;
97 auto parsed = std::from_chars(idText.data(), idText.data() + idText.size(), fileId);
98 if (parsed.ec != std::errc{} || parsed.ptr != idText.data() + idText.size())
99 return failure<std::vector<UnitySourceReference>>(
100 DiagnosticCode::ParseError, "Unity reference fileID exceeds signed 64-bit range", path);
101 if (out.size() >= limits.maximumAssets)
102 return failure<std::vector<UnitySourceReference>>(DiagnosticCode::InvalidArgument,
103 "Unity reference count exceeds budget", path);
104 out.push_back({foldAscii(guid[1].str()), fileId});
105 }
106 std::sort(out.begin(), out.end(),
107 [](const auto& a, const auto& b) { return std::tie(a.guid, a.fileId) < std::tie(b.guid, b.fileId); });
108 out.erase(std::unique(out.begin(), out.end(),
109 [](const auto& a, const auto& b) { return a.guid == b.guid && a.fileId == b.fileId; }),
110 out.end());
111 return Result<std::vector<UnitySourceReference>>::success(std::move(out));
112}

115Result<UnitySourceIndex> indexUnitySources(const UnitySourceFiles& files, const AssetImportLimits& limits) {
116 if (files.empty() || files.size() > std::uint64_t(limits.maximumAssets) * 2)
117 return failure<UnitySourceIndex>(DiagnosticCode::InvalidArgument,
118 "Unity source count exceeds budget or is empty");
119 std::uint64_t total = 0;
120 std::set<std::string> paths;
121 for (const auto& [path, bytes] : files) {
122 if (!validPath(path, limits) || bytes.size() > limits.maximumSourceBytes ||
123 total > limits.maximumSourceBytes - bytes.size())
124 return failure<UnitySourceIndex>(DiagnosticCode::InvalidArgument, "Unity path or source budget is invalid",
125 path);
126 total += bytes.size();
127 if (!paths.insert(foldAscii(path)).second)
128 return failure<UnitySourceIndex>(DiagnosticCode::Conflict, "Unity source paths differ only by case", path);
129 }
130 UnitySourceIndex index;
131 std::map<std::string, std::string> guidPaths;
132 std::map<std::string, std::size_t> assetPaths;
133 for (const auto& [path, bytes] : files) {
134 if (path.ends_with(".meta")) continue;
135 assetPaths.emplace(path, index.assets.size());
136 index.assets.push_back({path, {}, classify(path), {}, {}});
137 }
138 for (const auto& [path, bytes] : files) {
139 if (!path.ends_with(".meta")) continue;
140 const std::string sourcePath = path.substr(0, path.size() - 5);
141 auto guid = metaScalar(bytes, "guid", path);
142 if (!guid) return Result<UnitySourceIndex>::failure(guid.status());
143 if (!validGuid(guid.value()) || builtin(foldAscii(guid.value())))
144 return failure<UnitySourceIndex>(DiagnosticCode::ParseError, "Unity .meta needs a non-reserved GUID", path);
145 const auto normalizedGuid = foldAscii(guid.value());
146 if (!guidPaths.emplace(normalizedGuid, sourcePath).second)
147 return failure<UnitySourceIndex>(DiagnosticCode::Conflict, "duplicate Unity GUID", path);
148 auto folder = metaScalar(bytes, "folderAsset", path);
149 if (!folder) return Result<UnitySourceIndex>::failure(folder.status());
150 const auto found = assetPaths.find(sourcePath);
151 if (folder.value() == "yes") {
152 if (found != assetPaths.end())
153 return failure<UnitySourceIndex>(DiagnosticCode::Conflict, "Unity folder also contains a file payload",
154 path);
155 index.assets.push_back({sourcePath, normalizedGuid, UnitySourceKind::Folder, {}, {}});
156 continue;
157 }
158 if (found == assetPaths.end())
159 return failure<UnitySourceIndex>(DiagnosticCode::NotFound, "Unity metadata has no source payload", path);
160 auto& source = index.assets[found->second];
161 source.guid = normalizedGuid;
162 const std::string_view meta(reinterpret_cast<const char*>(bytes.data()), bytes.size());
163 for (std::size_t at = 0; at < meta.size();) {
164 auto end = meta.find('\n', at);
165 if (end == std::string_view::npos) end = meta.size();
166 auto line = meta.substr(at, end - at);
167 if (line.ends_with('\r')) line.remove_suffix(1);
168 if (line.ends_with("Importer:") && line.find_first_of(" \t") == std::string_view::npos)
169 source.importer = std::string(line.substr(0, line.size() - 1));
170 at = end + 1;
171 }
172 auto refs = references(bytes, path, limits);
173 if (!refs) return Result<UnitySourceIndex>::failure(refs.status());
174 source.references = std::move(refs).takeValue();
175 }
176 if (index.assets.size() > limits.maximumAssets)
177 return failure<UnitySourceIndex>(DiagnosticCode::InvalidArgument, "Unity asset count exceeds budget");
178 std::map<std::string, UnitySourceKind> sourcePaths;
179 for (const auto& entry : index.assets) {
180 if (!sourcePaths.emplace(foldAscii(entry.path), entry.kind).second)
181 return failure<UnitySourceIndex>(DiagnosticCode::Conflict, "Unity source identities share a path",
182 entry.path);
183 }
184 for (const auto& entry : index.assets) {
185 for (auto slash = entry.path.find('/'); slash != std::string::npos; slash = entry.path.find('/', slash + 1)) {
186 const auto parent = sourcePaths.find(foldAscii(entry.path.substr(0, slash)));
187 if (parent != sourcePaths.end() && parent->second != UnitySourceKind::Folder)
188 return failure<UnitySourceIndex>(DiagnosticCode::Conflict,
189 "Unity source file is also used as a directory", entry.path);
190 }
191 }
192 std::sort(index.assets.begin(), index.assets.end(), [](const auto& a, const auto& b) { return a.path < b.path; });
193 for (auto& entry : index.assets) {
194 if (entry.kind == UnitySourceKind::Folder) continue;
195 if (entry.guid.empty())
196 index.findings.push_back({entry.path, "identity.meta", ImportDisposition::Unsupported,
197 "source has no .meta GUID; stable Unity reimport identity is unavailable"});
198 if (entry.kind == UnitySourceKind::Prefab || entry.kind == UnitySourceKind::Scene ||
199 entry.kind == UnitySourceKind::Material || entry.kind == UnitySourceKind::Data ||
200 entry.kind == UnitySourceKind::Animation) {
201 auto refs = references(files.at(entry.path), entry.path, limits);
202 if (!refs) return Result<UnitySourceIndex>::failure(refs.status());
203 for (auto& ref : refs.value()) entry.references.push_back(std::move(ref));
204 }
205 for (const auto& ref : entry.references)
206 if (ref.fileId != 0 && !builtin(ref.guid) && !guidPaths.contains(ref.guid))
207 index.findings.push_back({entry.path, "dependency:" + ref.guid + ":" + std::to_string(ref.fileId),
208 ImportDisposition::Unsupported, "referenced Unity asset was not supplied"});
209 }
210 return Result<UnitySourceIndex>::success(std::move(index));
211}
