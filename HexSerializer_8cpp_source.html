82 if (offset > size_) {
83 ok_ = false;

86 offset_ = offset;
87 return true;

93 [[nodiscard]] std::uint32_t u32() noexcept {
94 if (remaining() < 4) {
95 ok_ = false;
96 return 0u;
97 }
98 std::uint32_t value = 0;
99 for (std::size_t index = 0; index < 4; ++index)
100 value |= static_cast<std::uint32_t>(data_[offset_ + index]) << (8u * static_cast<unsigned>(index));
101 offset_ += 4;
102 return value;
103 }

113 offset_ += 8;

115 }

132Result<void> saveHexMap(const HexMap& map, const std::vector<HexUnitState>& units, std::vector<std::uint8_t>& out) {
133 if (map.empty()) return Result<void>::failure(invalidArgument("cannot serialize an empty hex map"));
134
135 const std::int32_t cellCount = map.cellCount();
136 const std::size_t payload =
137 kHeaderSize + static_cast<std::size_t>(cellCount) * kCellRecordSize + 4 + units.size() * kUnitRecordSize;
138 out.clear();
139 out.reserve(payload);
140
141 for (const std::uint8_t byte : kHexSaveMagic) out.push_back(byte);
142 pushU32(out, kHexSaveVersion);
143 pushU32(out, static_cast<std::uint32_t>(map.cellCountX()));
144 pushU32(out, static_cast<std::uint32_t>(map.cellCountZ()));
145 pushU32(out, map.seed());
146 pushU64(out, map.revision());
147
148 for (std::int32_t index = 0; index < cellCount; ++index) {
149 // `cellAt` is non-null for every index below `cellCount()`, the loop bound.
150 const HexCellData* cell = map.cellAt(index);
151 pushU32(out, cell->values.raw());
152 pushU32(out, cell->flags.raw());
153 }
154
155 pushU32(out, static_cast<std::uint32_t>(units.size()));
156 for (const HexUnitState& unit : units) {
157 pushI32(out, unit.locationIndex);
158 pushF32(out, unit.orientation);
159 }
160
161 return Result<void>::success();
162}

166Result<void> loadHexMap(const std::vector<std::uint8_t>& bytes, HexMap& map, std::vector<HexUnitState>& units) {
167 if (bytes.size() < kHeaderSize) return Result<void>::failure(invalidArgument("hex save payload is truncated"));
168
169 for (std::size_t index = 0; index < kMagicSize; ++index) {
170 if (bytes[index] != kHexSaveMagic[index])
171 return Result<void>::failure(invalidArgument("hex save payload has an unknown magic prefix"));
172 }
173
174 Reader reader(bytes.data(), bytes.size());
175 // The payload size check above already covers the magic, so this can only fail
176 // if that check is ever relaxed.
177 if (!reader.skip(kMagicSize)) return Result<void>::failure(invalidArgument("hex save payload is truncated"));
178
179 const std::uint32_t version = reader.u32();
180 if (version != kHexSaveVersion) {
181 return Result<void>::failure(invalidArgument("hex save version " + std::to_string(version) +
182 " is not supported (expected " + std::to_string(kHexSaveVersion) +
183 ")"));
184 }
185 const std::uint32_t cellCountX = reader.u32();
186 const std::uint32_t cellCountZ = reader.u32();
187 const std::uint32_t seed = reader.u32();
188 // Diagnostic only: the restored map's revision is whatever `reset` produces.
189 const std::uint64_t savedRevision = reader.u64();
190 (void)savedRevision;
191
192 if (cellCountX == 0 || cellCountZ == 0)
193 return Result<void>::failure(invalidArgument("hex save grid dimensions must be positive"));
194 if (cellCountX > kMaxGridDimension || cellCountZ > kMaxGridDimension)
195 return Result<void>::failure(invalidArgument("hex save grid dimensions exceed the 512 cell limit"));
196 if (cellCountX % static_cast<std::uint32_t>(HexMetrics::kChunkSizeX) != 0u ||
197 cellCountZ % static_cast<std::uint32_t>(HexMetrics::kChunkSizeZ) != 0u)
198 return Result<void>::failure(
199 invalidArgument("hex save grid dimensions must be a multiple of the 5x5 chunk size"));
200
201 const std::size_t cellCount = static_cast<std::size_t>(cellCountX) * static_cast<std::size_t>(cellCountZ);
202 const std::size_t cellsEnd = kHeaderSize + cellCount * kCellRecordSize;
203 const std::size_t unitsStart = cellsEnd + 4;
204 if (bytes.size() < unitsStart)
205 return Result<void>::failure(invalidArgument("hex save payload is truncated before the unit records"));
206
207 if (!reader.seek(cellsEnd)) return Result<void>::failure(invalidArgument("hex save payload is truncated"));
208 const std::uint32_t unitCount = reader.u32();
209 if (static_cast<std::size_t>(unitCount) > (bytes.size() - unitsStart) / kUnitRecordSize)
210 return Result<void>::failure(invalidArgument("hex save unit count exceeds the payload size"));
211 if (unitsStart + static_cast<std::size_t>(unitCount) * kUnitRecordSize != bytes.size())
212 return Result<void>::failure(invalidArgument("hex save payload size does not match its contents"));
213
214 // Decode into locals first: a rejected payload must leave both outputs alone.
215 std::vector<HexCellData> cells(cellCount);
216 if (!reader.seek(kHeaderSize)) return Result<void>::failure(invalidArgument("hex save payload is truncated"));
217 for (std::size_t index = 0; index < cellCount; ++index) {
218 cells[index].values = HexValues{reader.u32()};
219 cells[index].flags = HexFlags{reader.u32()};
220 }
221
222 std::vector<HexUnitState> restored(unitCount);
223 if (!reader.seek(unitsStart)) return Result<void>::failure(invalidArgument("hex save payload is truncated"));
224 for (std::uint32_t index = 0; index < unitCount; ++index) {
225 const std::int32_t locationIndex = reader.i32();
226 const float orientation = reader.f32();
227 if (locationIndex < 0 || static_cast<std::size_t>(locationIndex) >= cellCount)
228 return Result<void>::failure(invalidArgument("hex save unit cell is outside the grid"));
229 restored[index].locationIndex = locationIndex;
230 restored[index].orientation = orientation;
231 }
232 if (!reader.ok()) return Result<void>::failure(invalidArgument("hex save payload is truncated"));
233
234 // The unit list must be legal *before* the grid is adopted, because the unit
235 // registry consumes it afterwards and would otherwise reject the whole list
236 // against a grid the caller had already replaced. `canHoldUnit` is the same
237 // terrain rule the registry applies through `isValidDestination`.
238 for (std::size_t index = 0; index < restored.size(); ++index) {
239 const auto cell = static_cast<std::size_t>(restored[index].locationIndex);
240 if (!canHoldUnit(cells[cell]))
241 return Result<void>::failure(invalidArgument("hex save unit stands on an unusable cell"));
242 for (std::size_t other = 0; other < index; ++other) {
243 if (restored[other].locationIndex == restored[index].locationIndex)
244 return Result<void>::failure(invalidArgument("hex save units occupy the same cell"));
245 }
246 }
247
248 // Everything decoded; only now are the outputs rebuilt.
249 auto resized = map.reset(static_cast<std::int32_t>(cellCountX), static_cast<std::int32_t>(cellCountZ), seed);
250 if (!resized.ok()) return Result<void>::failure(resized.status());
251 for (std::size_t index = 0; index < cellCount; ++index) {
252 map.setCellState(map.coordinatesAt(static_cast<std::int32_t>(index)), cells[index].values, cells[index].flags)
253 .ignore("every index below the freshly reset cell count is inside the grid");
254 }
255 map.markAllChunksDirty();
256 units = std::move(restored);
257 return Result<void>::success();
258}
