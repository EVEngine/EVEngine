32 std::vector<float> pos;
33 std::vector<float> nrm;
34 std::vector<float> uv;

36};

39void buildHeightmapArrays(const eve::procgen::Heightmap &hm, float cell, float hScale,
40 HeightmapArrays &out) {
41 out.pos.clear();
42 out.nrm.clear();
43 out.uv.clear();
44 out.idx.clear();
45 const int w = hm.getWidth();
46 const int h = hm.getHeight();
47 if (w < 2 || h < 2) return;
48 const float uw = float(w - 1);
49 const float uh = float(h - 1);
50
51 auto addTri = [&](float ax, float az, float ay, float bx, float bz, float by, float cx,
52 float cz, float cy) {
53 const float p0x = ax * cell, p0y = ay * hScale, p0z = az * cell;
54 const float p1x = bx * cell, p1y = by * hScale, p1z = bz * cell;
55 const float p2x = cx * cell, p2y = cy * hScale, p2z = cz * cell;
56 const float e1x = p1x - p0x, e1y = p1y - p0y, e1z = p1z - p0z;
57 const float e2x = p2x - p0x, e2y = p2y - p0y, e2z = p2z - p0z;
58 float nx = e1y * e2z - e1z * e2y;
59 float ny = e1z * e2x - e1x * e2z;
60 float nz = e1x * e2y - e1y * e2x;
61 const float len = std::sqrt(nx * nx + ny * ny + nz * nz);
62 if (len > 1e-8f) {
63 nx /= len;
64 ny /= len;
65 nz /= len;
66 }
67 const uint32_t base = uint32_t(out.pos.size() / 3);
68 out.pos.insert(out.pos.end(), {p0x, p0y, p0z, p1x, p1y, p1z, p2x, p2y, p2z});
69 out.nrm.insert(out.nrm.end(), {nx, ny, nz, nx, ny, nz, nx, ny, nz});
70 out.uv.insert(out.uv.end(),
71 {ax / uw, az / uh, bx / uw, bz / uh, cx / uw, cz / uh});
72 out.idx.insert(out.idx.end(), {base, base + 1, base + 2});
73 };
74
75 for (int y = 0; y < h - 1; ++y) {
76 for (int x = 0; x < w - 1; ++x) {
77 const float h00 = hm.height(x, y);
78 const float h10 = hm.height(x + 1, y);
79 const float h01 = hm.height(x, y + 1);
80 const float h11 = hm.height(x + 1, y + 1);
81 // Grid is XZ; y in the function is the XZ "row" axis.
82 addTri(float(x), float(y), h00, float(x + 1), float(y), h10, float(x), float(y + 1), h01);
83 addTri(float(x + 1), float(y), h10, float(x + 1), float(y + 1), h11, float(x),
84 float(y + 1), h01);
85 }
86 }
87}

110graphics::Mesh *Editor::newHeightmapMesh(procgen::Heightmap *hm, float cellSize,
111 float heightScale) {
112 auto *gfx = eve::ModuleManager::getInstance<graphics::Graphics>("Graphics");
113 if (!gfx || !hm) return nullptr;
114 HeightmapArrays a;
115 buildHeightmapArrays(*hm, cellSize, heightScale, a);
116 if (a.idx.empty()) return nullptr;
117 return gfx->newMeshFromArrays(a.pos.data(), a.nrm.data(), a.uv.data(),
118 int(a.pos.size() / 3), a.idx.data(), int(a.idx.size()));
119}
