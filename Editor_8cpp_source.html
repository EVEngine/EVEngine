66bool squirrelToEditorValue(HSQUIRRELVM vm, SQInteger index, EditorValue& out, size_t depth = 0) {
67 if (!vm || depth > 32) return false;
68 const SQInteger absolute = index > 0 ? index : sq_gettop(vm) + index + 1;
69 switch (sq_gettype(vm, absolute)) {
70 case OT_NULL: out = EditorValue{}; return true;
71 case OT_BOOL: {
72 SQBool value = SQFalse;
73 if (SQ_FAILED(sq_getbool(vm, absolute, &value))) return false;
74 out = EditorValue(value != SQFalse);
75 return true;
76 }
77 case OT_INTEGER: {
78 SQInteger value = 0;
79 if (SQ_FAILED(sq_getinteger(vm, absolute, &value))) return false;
80 out = EditorValue(static_cast<int64_t>(value));
81 return true;
82 }
83 case OT_FLOAT: {
84 SQFloat value = 0;
85 if (SQ_FAILED(sq_getfloat(vm, absolute, &value))) return false;
86 out = EditorValue(static_cast<double>(value));
87 return true;
88 }
89 case OT_STRING: {
90 const SQChar* value = nullptr;
91 if (SQ_FAILED(sq_getstring(vm, absolute, &value))) return false;
92 out = EditorValue(value ? value : "");
93 return true;
94 }
95 case OT_ARRAY: {
96 EditorValue::Array values;
97 const SQInteger count = sq_getsize(vm, absolute);
98 values.reserve(static_cast<size_t>(count));
99 for (SQInteger i = 0; i < count; ++i) {
100 sq_pushinteger(vm, i);
101 if (SQ_FAILED(sq_get(vm, absolute))) return false;
102 EditorValue value;
103 const bool ok = squirrelToEditorValue(vm, -1, value, depth + 1);
104 sq_pop(vm, 1);
105 if (!ok) return false;
106 values.push_back(std::move(value));
107 }
108 out = EditorValue(std::move(values));
109 return true;
110 }
111 case OT_TABLE: {
112 EditorValue::Object values;
113 sq_pushnull(vm);
114 while (SQ_SUCCEEDED(sq_next(vm, absolute))) {
115 const SQChar* key = nullptr;
116 const bool keyOk = sq_gettype(vm, -2) == OT_STRING && SQ_SUCCEEDED(sq_getstring(vm, -2, &key)) && key;
117 const std::string stableKey = keyOk ? key : "";
118 EditorValue value;
119 const bool valueOk = keyOk && squirrelToEditorValue(vm, -1, value, depth + 1);
120 sq_pop(vm, 2);
121 if (!valueOk) {
122 sq_pop(vm, 1);
123 return false;
124 }
125 values[stableKey] = std::move(value);
126 }
127 sq_pop(vm, 1);
128 out = EditorValue(std::move(values));
129 return true;
130 }
131 default: return false;
132 }
133}

145void pushEditorValue(HSQUIRRELVM vm, const EditorValue& value) {
146 std::visit(
147 [&](const auto& current) {
148 using T = std::decay_t<decltype(current)>;
149 if constexpr (std::is_same_v<T, std::monostate>) {
150 sq_pushnull(vm);
151 } else if constexpr (std::is_same_v<T, bool>) {
152 sq_pushbool(vm, current ? SQTrue : SQFalse);
153 } else if constexpr (std::is_same_v<T, int64_t>) {
154 sq_pushinteger(vm, static_cast<SQInteger>(current));
155 } else if constexpr (std::is_same_v<T, double>) {
156 sq_pushfloat(vm, static_cast<SQFloat>(current));
157 } else if constexpr (std::is_same_v<T, std::string>) {
158 sq_pushstring(vm, current.c_str(), static_cast<SQInteger>(current.size()));
159 } else if constexpr (std::is_same_v<T, EditorValue::Array>) {
160 sq_newarray(vm, 0);
161 for (const EditorValue& entry : current) {
162 pushEditorValue(vm, entry);
163 sq_arrayappend(vm, -2);
164 }
165 } else if constexpr (std::is_same_v<T, EditorValue::Object>) {
166 sq_newtable(vm);
167 for (const auto& [key, entry] : current) {
168 sq_pushstring(vm, key.c_str(), static_cast<SQInteger>(key.size()));
169 pushEditorValue(vm, entry);
170 sq_newslot(vm, -3, SQFalse);
171 }
172 }
173 },
174 value.storage());
175}

213bool invokeScriptCommand(const ssq::Object& callback, const EditorValue& payload) {
214 const SQObjectType type = callback.getRaw()._type;
215 if (type != OT_CLOSURE && type != OT_NATIVECLOSURE) return false;
216 HSQUIRRELVM vm = callback.getHandle();
217 SQInteger top = sq_gettop(vm);
218 sq_pushobject(vm, callback.getRaw());
219 sq_pushroottable(vm);
220 pushEditorValue(vm, payload);
221 bool accepted = false;
222 if (SQ_SUCCEEDED(sq_call(vm, 2, SQTrue, SQTrue))) {
223 SQBool result = SQFalse;
224 accepted = SQ_SUCCEEDED(sq_getbool(vm, -1, &result)) && result != SQFalse;
225 }
226 sq_settop(vm, top);
227 return accepted;
228}

230bool registerScriptCommand(Editor* editor, const std::string& id, const std::string& displayName,
231 const std::string& category, ssq::Object callback) {
232 if (!editor || id.empty()) return false;
233 const SQObjectType callbackType = callback.getRaw()._type;
234 if (callbackType != OT_CLOSURE && callbackType != OT_NATIVECLOSURE) return false;
235
236 CommandDescriptor descriptor;
237 descriptor.id = CommandId(id);
238 descriptor.ownerModule = "script:" + id;
239 descriptor.displayName = displayName;
240 descriptor.category = category;
241 descriptor.automationAllowed = false;
242 return editor->commandService()
243 .registerPlannedCommand(
244 descriptor,
245 [id](const CommandRequest& request) {
246 CommandPlan plan;
247 plan.summary = EditorValue::Object{{"command", EditorValue(id)}, {"payload", request.payload}};
248 DomainOperation operation;
249 operation.type = id;
250 operation.payload = request.payload;
251 plan.operations.push_back(std::move(operation));
252 return eve::editing::applied<CommandPlan>(std::move(plan));
253 },
254 [callback = std::move(callback)](const CommandRequest& request, const CommandPlan& plan) {
255 if (!invokeScriptCommand(callback, request.payload))
256 return eve::editing::failed<TransactionReceipt>(
257 EditorStatus::Rejected, RuleId("editor.script.command-rejected"),
258 "Script command callback rejected the planned payload");
259 TransactionReceipt receipt;
260 receipt.id = TransactionId(plan.id.value());
261 receipt.state = TransactionState::Committed;
262 receipt.beforeRevision = plan.baseRevision;
263 receipt.afterRevision = plan.baseRevision + 1;
264 receipt.authorityReceipt = "script:local";
265 return eve::editing::applied<TransactionReceipt>(std::move(receipt));
266 },
267 true)
268 .ok();
269}
