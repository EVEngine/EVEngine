16bool compareValues(const BlackboardValue& lhs, CompareOp op, const BlackboardValue& rhs) {
17 if (lhs.index() != rhs.index()) return op == CompareOp::NotEqual;
18 if (op == CompareOp::Equal) return lhs == rhs;
19 if (op == CompareOp::NotEqual) return lhs != rhs;
20 return std::visit(
21 [op](const auto& a, const auto& b) {
22 using A = std::decay_t<decltype(a)>;
23 using B = std::decay_t<decltype(b)>;
24 if constexpr (!std::is_same_v<A, B> || std::is_same_v<A, bool>)
25 return false;
26 else {
27 switch (op) {
28 case CompareOp::Less: return a < b;
29 case CompareOp::LessEqual: return a <= b;
30 case CompareOp::Greater: return a > b;
31 case CompareOp::GreaterEqual: return a >= b;
32 default: return false;
33 }
34 }
35 },
36 lhs, rhs);
37}

39BlackboardType valueType(const BlackboardValue& value) {
40 if (std::holds_alternative<bool>(value)) return BlackboardType::Boolean;
41 if (std::holds_alternative<std::int64_t>(value)) return BlackboardType::Integer;
42 if (std::holds_alternative<double>(value)) return BlackboardType::Number;
43 return BlackboardType::String;
44}

53Result<void> NpcAiWorld::validate(const BehaviorDefinition& definition) {
54 if (definition.schemaVersion != 1)
55 return failure<void>(DiagnosticCode::Unsupported, "unsupported NPC behavior schema version");
56 if (definition.id.empty() || definition.initialState.empty())
57 return failure<void>(DiagnosticCode::InvalidArgument, "behavior id and initial state are required");
58 std::set<std::string> ids;
59 std::set<std::string> schemaKeys;
60 for (const auto& key : definition.blackboardSchema) {
61 if (key.key.empty() || !schemaKeys.insert(key.key).second)
62 return failure<void>(DiagnosticCode::InvalidArgument,
63 "blackboard schema keys must be non-empty and unique");
64 if (key.defaultValue && valueType(*key.defaultValue) != key.type)
65 return failure<void>(DiagnosticCode::InvalidArgument,
66 "blackboard default value does not match its declared type");
67 if (key.required && !key.defaultValue)
68 return failure<void>(DiagnosticCode::InvalidArgument,
69 "required blackboard keys need a default for atomic agent creation");
70 }
71 for (const auto& state : definition.states) {
72 if (state.id.empty() || !ids.insert(state.id).second)
73 return failure<void>(DiagnosticCode::InvalidArgument, "state ids must be non-empty and unique");
74 std::set<std::string> taskIds;
75 for (const auto& task : state.tasks)
76 if (task.id.empty() || task.type.empty() || !taskIds.insert(task.id).second)
77 return failure<void>(DiagnosticCode::InvalidArgument,
78 "task ids must be unique per state and task types are required");
79 }
80 if (!ids.contains(definition.initialState))
81 return failure<void>(DiagnosticCode::NotFound, "initial state does not exist");
82 for (const auto& state : definition.states) {
83 if (state.parent && (!ids.contains(*state.parent) || *state.parent == state.id))
84 return failure<void>(DiagnosticCode::InvalidArgument, "state parent is missing or self-referential");
85 for (const auto& transition : state.transitions) {
86 if (!ids.contains(transition.targetState))
87 return failure<void>(DiagnosticCode::NotFound, "transition target does not exist");
88 if (transition.targetState == state.id && transition.signal.empty())
89 return failure<void>(DiagnosticCode::InvalidArgument,
90 "an unconditional self-transition would exhaust the transition budget");
91 for (const auto& predicate : transition.conditions)
92 if (!validPredicate(predicate))
93 return failure<void>(DiagnosticCode::InvalidArgument, "invalid transition predicate");
94 }
95 for (const auto& predicate : state.enterConditions)
96 if (!validPredicate(predicate))
97 return failure<void>(DiagnosticCode::InvalidArgument, "invalid enter predicate");
98 }
99 for (const auto& origin : definition.states) {
100 std::set<std::string> chain;
101 const StateDefinition* current = &origin;
102 while (current->parent) {
103 if (!chain.insert(current->id).second)
104 return failure<void>(DiagnosticCode::InvalidArgument, "state parent cycle detected");
105 auto it = std::find_if(definition.states.begin(), definition.states.end(),
106 [&](const auto& s) { return s.id == *current->parent; });
107 current = &*it;
108 }
109 }
110 return Result<void>::success();
111}

131Result<AgentHandle> NpcAiWorld::createAgent(std::string_view behaviorId) {
132 auto behavior = behaviors_.find(std::string(behaviorId));
133 if (behavior == behaviors_.end())
134 return failure<AgentHandle>(DiagnosticCode::NotFound, "behavior definition was not registered");
135 std::uint32_t index;
136 if (freeSlots_.empty()) {
137 index = static_cast<std::uint32_t>(slots_.size());
138 slots_.push_back({});
139 } else {
140 index = freeSlots_.back();
141 freeSlots_.pop_back();
142 }
143 Agent candidate;
144 candidate.behaviorId = behavior->first;
145 candidate.activeState = behavior->second.initialState;
146 for (const auto& key : behavior->second.blackboardSchema)
147 if (key.defaultValue) candidate.blackboard.emplace(key.key, *key.defaultValue);
148 for (const auto state : activePath(behavior->second, candidate.activeState)) {
149 if (!predicatesPass(state.get().enterConditions, candidate)) {
150 freeSlots_.push_back(index);
151 return failure<AgentHandle>(DiagnosticCode::PreconditionViolation,
152 "initial state path enter conditions are false");
153 }
154 }
155 slots_[index].agent = std::move(candidate);
156 const AgentHandle handle(index, slots_[index].generation);
157 pushTrace({0, handle, TraceKind::AgentCreated, behavior->first, {}});
158 return Result<AgentHandle>::success(handle);
159}

178Result<void> NpcAiWorld::destroyAgent(AgentHandle handle) {
179 auto resolved = resolve(handle);
180 if (!resolved.ok()) return Result<void>::failure(resolved.status());
181 Agent& agent = resolved.value().get();
182 const auto& behavior = behaviors_.at(agent.behaviorId);
183 auto path = activePath(behavior, agent.activeState);
184 for (auto state = path.rbegin(); state != path.rend(); ++state)
185 stopStateTasks(handle, agent, state->get(), StopReason::AgentDestroyed);
186 pushTrace({agent.lastTick, handle, TraceKind::AgentDestroyed, agent.behaviorId, {}});
187 auto& slot = slots_[handle.index()];
188 slot.agent.reset();
189 auto next = AgentHandle::nextGeneration(slot.generation);
190 if (next) {
191 slot.generation = *next;
192 freeSlots_.push_back(handle.index());
193 }
194 return Result<void>::success(Status::success(StatusCode::Applied));
195}

225Result<void> NpcAiWorld::remember(AgentHandle handle, PerceptionMemory memory) {
226 if (memory.subject.empty() || memory.sense.empty() || !std::isfinite(memory.confidence) ||
227 memory.confidence < 0.0 || memory.confidence > 1.0 || memory.forgetAfterTicks == 0)
228 return failure<void>(DiagnosticCode::InvalidArgument, "perception memory fields are invalid");
229 auto resolved = resolve(handle);
230 if (!resolved.ok()) return Result<void>::failure(resolved.status());
231 auto& agent = resolved.value().get();
232 auto found = std::find_if(agent.perception.begin(), agent.perception.end(), [&](const auto& item) {
233 return item.subject == memory.subject && item.sense == memory.sense;
234 });
235 if (found != agent.perception.end()) {
236 *found = memory;
237 } else {
238 if (config_.maxMemoriesPerAgent == 0)
239 return failure<void>(DiagnosticCode::Unsupported, "perception memory is disabled by world configuration");
240 if (agent.perception.size() == config_.maxMemoriesPerAgent) {
241 auto oldest =
242 std::min_element(agent.perception.begin(), agent.perception.end(), [](const auto& a, const auto& b) {
243 return std::pair{a.observedTick, a.subject + "\n" + a.sense} <
244 std::pair{b.observedTick, b.subject + "\n" + b.sense};
245 });
246 pushTrace({memory.observedTick, handle, TraceKind::PerceptionForgotten, oldest->subject, oldest->sense});
247 *oldest = memory;
248 } else {
249 agent.perception.push_back(memory);
250 }
251 }
252 std::sort(agent.perception.begin(), agent.perception.end(), [](const auto& a, const auto& b) {
253 return std::pair{a.subject, a.sense} < std::pair{b.subject, b.sense};
254 });
255 agent.signals.push_back("perception." + memory.sense);
256 pushTrace({memory.observedTick, handle, TraceKind::PerceptionUpdated, memory.subject, memory.sense});
257 return Result<void>::success(Status::success(StatusCode::Applied));
258}

275bool NpcAiWorld::predicatesPass(std::span<const BlackboardPredicate> predicates, const Agent& agent) const {
276 for (const auto& predicate : predicates) {
277 const auto found = agent.blackboard.find(predicate.key);
278 if (predicate.op == CompareOp::Exists) {
279 if (found == agent.blackboard.end()) return false;
280 continue;
281 }
282 if (found == agent.blackboard.end() || !predicate.value ||
283 !compareValues(found->second, predicate.op, *predicate.value))
284 return false;
285 }
286 return true;
287}

297std::vector<std::reference_wrapper<const StateDefinition>> NpcAiWorld::activePath(const BehaviorDefinition& behavior,
298 std::string_view leaf) const {
299 std::vector<std::reference_wrapper<const StateDefinition>> path;
300 auto current = findState(behavior, leaf);
301 while (current) {
302 path.push_back(*current);
303 current = current->get().parent ? findState(behavior, *current->get().parent) : std::nullopt;
304 }
305 std::reverse(path.begin(), path.end());
306 return path;
307}

328 auto runtime = agent.taskRuntime.find(key);

338Result<TickReport> NpcAiWorld::tick(const TickContext& context) {
339 if (!std::isfinite(context.deltaSeconds) || context.deltaSeconds < 0.0 || context.maxTransitionsPerAgent == 0)
340 return failure<TickReport>(DiagnosticCode::InvalidArgument, "tick dt and transition budget are invalid");
341 TickReport report;
342 if (slots_.empty() || context.maxAgents == 0) return Result<TickReport>::success(report);
343 std::uint32_t visited = 0;
344 while (visited < slots_.size() && report.agentsUpdated < context.maxAgents) {
345 const std::uint32_t index = (scheduleCursor_ + visited) % static_cast<std::uint32_t>(slots_.size());
346 ++visited;
347 auto& slot = slots_[index];
348 if (!slot.agent) continue;
349 AgentHandle handle(index, slot.generation);
350 Agent& agent = *slot.agent;
351 auto& behavior = behaviors_.at(agent.behaviorId);
352 auto path = activePath(behavior, agent.activeState);
353 if (path.empty())
354 return failure<TickReport>(DiagnosticCode::InvariantViolation, "agent active state is missing");
355
356 for (auto memory = agent.perception.begin(); memory != agent.perception.end();) {
357 const bool expired = context.simulationTick >= memory->observedTick &&
358 context.simulationTick - memory->observedTick >= memory->forgetAfterTicks;
359 if (!expired) {
360 ++memory;
361 continue;
362 }
363 const std::string signalName = "perception.forgotten." + memory->sense;
364 agent.signals.push_back(signalName);
365 pushTrace({context.simulationTick, handle, TraceKind::PerceptionForgotten, memory->subject, memory->sense});
366 memory = agent.perception.erase(memory);
367 }
368
369 std::uint32_t transitions = 0;
370 while (transitions < context.maxTransitionsPerAgent) {
371 const Transition* selected = nullptr;
372 std::vector<std::reference_wrapper<const StateDefinition>> selectedPath;
373 for (auto state = path.rbegin(); state != path.rend(); ++state) {
374 for (const auto& candidate : state->get().transitions) {
375 const bool signalMatches =
376 candidate.signal.empty() ||
377 std::find(agent.signals.begin(), agent.signals.end(), candidate.signal) != agent.signals.end();
378 if (!signalMatches || !predicatesPass(candidate.conditions, agent) ||
379 (selected && candidate.priority <= selected->priority))
380 continue;
381 auto candidatePath = activePath(behavior, candidate.targetState);
382 const bool canEnter = std::all_of(
383 candidatePath.begin(), candidatePath.end(),
384 [&](const auto target) { return predicatesPass(target.get().enterConditions, agent); });
385 if (canEnter) {
386 selected = &candidate;
387 selectedPath = std::move(candidatePath);
388 }
389 }
390 }
391 if (!selected) break;
392
393 std::size_t commonPrefix = 0;
394 while (commonPrefix < path.size() && commonPrefix < selectedPath.size() &&
395 path[commonPrefix].get().id == selectedPath[commonPrefix].get().id)
396 ++commonPrefix;
397 for (std::size_t exiting = path.size(); exiting > commonPrefix; --exiting)
398 stopStateTasks(handle, agent, path[exiting - 1].get(), StopReason::Transition);
399
400 const std::string previousState = agent.activeState;
401 agent.activeState = selected->targetState;
402 path = std::move(selectedPath);
403 ++transitions;
404 ++report.transitionsApplied;
405 pushTrace({context.simulationTick, handle, TraceKind::TransitionApplied, previousState, agent.activeState});
406 if (!selected->signal.empty())
407 agent.signals.erase(std::remove(agent.signals.begin(), agent.signals.end(), selected->signal),
408 agent.signals.end());
409 }
410 if (transitions == context.maxTransitionsPerAgent) ++report.transitionBudgetsExhausted;
411 agent.signals.clear();
412 agent.lastTick = context.simulationTick;
413
414 for (const auto state : path)
415 for (const auto& task : state.get().tasks)
416 if (!taskServices_.contains(task.type))
417 return failure<TickReport>(DiagnosticCode::NotFound,
418 "task service is not registered: " + task.type);
419
420 struct StartedTask {
421 std::reference_wrapper<const StateDefinition> state;
422 std::reference_wrapper<const TaskSpec> task;
423 std::string runtimeKey;
424 };
425 std::vector<StartedTask> startedThisTick;
426 for (const auto state : path) {
427 for (const auto& task : state.get().tasks) {
428 auto& service = taskServices_.at(task.type);
429 const auto runtimeKey = taskRuntimeKey(state.get().id, task.id);
430 auto& runtime = agent.taskRuntime[runtimeKey];
431 if (runtime.completed) continue;
432 TaskContext taskContext{handle, state.get().id, context.simulationTick, context.deltaSeconds,
433 agent.blackboard, agent.perception};
434 if (!runtime.started) {
435 auto started = service->start(taskContext, task, runtime.memoryJson);
436 if (!started.ok()) {
437 agent.taskRuntime.erase(runtimeKey);
438 for (auto rollback = startedThisTick.rbegin(); rollback != startedThisTick.rend(); ++rollback) {
439 auto active = agent.taskRuntime.find(rollback->runtimeKey);
440 if (active == agent.taskRuntime.end()) continue;
441 auto& rollbackService = taskServices_.at(rollback->task.get().type);
442 const TaskContext rollbackContext{handle,
443 rollback->state.get().id,
444 context.simulationTick,
445 context.deltaSeconds,
446 agent.blackboard,
447 agent.perception};
448 rollbackService->stop(rollbackContext, rollback->task.get(), StopReason::StartupRollback,
449 active->second.memoryJson);
450 agent.taskRuntime.erase(active);
451 }
452 return Result<TickReport>::failure(started.status());
453 }
454 runtime.started = true;
455 startedThisTick.push_back({state, std::cref(task), runtimeKey});
456 pushTrace({context.simulationTick, handle, TraceKind::TaskStarted, state.get().id, task.id});
457 }
458 }
459 }
460
461 for (const auto state : path) {
462 for (const auto& task : state.get().tasks) {
463 auto& service = taskServices_.at(task.type);
464 auto& runtime = agent.taskRuntime.at(taskRuntimeKey(state.get().id, task.id));
465 if (runtime.completed) continue;
466 TaskContext taskContext{handle, state.get().id, context.simulationTick, context.deltaSeconds,
467 agent.blackboard, agent.perception};
468 auto taskResult = service->tick(taskContext, task, runtime.memoryJson);
469 if (!taskResult.ok()) return Result<TickReport>::failure(taskResult.status());
470 ++report.tasksTicked;
471 if (taskResult.value() != TaskStatus::Running) {
472 service->stop(taskContext, task, StopReason::Completed, runtime.memoryJson);
473 runtime.started = false;
474 runtime.completed = true;
475 const bool succeeded = taskResult.value() == TaskStatus::Succeeded;
476 agent.signals.push_back(std::string("task.") + (succeeded ? "succeeded." : "failed.") + task.id);
477 pushTrace({context.simulationTick, handle,
478 succeeded ? TraceKind::TaskCompleted : TraceKind::TaskFailed, state.get().id, task.id});
479 }
480 }
481 }
482 ++report.agentsUpdated;
483 }
484 scheduleCursor_ = (scheduleCursor_ + visited) % static_cast<std::uint32_t>(slots_.size());
485 for (const auto& slot : slots_)
486 if (slot.agent) ++report.agentsDeferred;
487 report.agentsDeferred -= report.agentsUpdated;
488 return Result<TickReport>::success(report);
489}

502Result<AgentArchive> NpcAiWorld::archive(AgentHandle handle) const {
503 auto resolved = resolve(handle);
504 if (!resolved.ok()) return Result<AgentArchive>::failure(resolved.status());
505 const Agent& agent = resolved.value().get();
506 AgentArchive result;
507 result.behaviorId = agent.behaviorId;
508 result.activeState = agent.activeState;
509 result.blackboard = agent.blackboard;
510 result.pendingSignals = agent.signals;
511 result.perception = agent.perception;
512 result.lastTick = agent.lastTick;
513 const auto& behavior = behaviors_.at(agent.behaviorId);
514 for (const auto state : activePath(behavior, agent.activeState)) {
515 for (const auto& task : state.get().tasks) {
516 const auto runtime = agent.taskRuntime.find(taskRuntimeKey(state.get().id, task.id));
517 if (runtime == agent.taskRuntime.end()) continue;
518 result.tasks.push_back({state.get().id, task.id, runtime->second.completed, runtime->second.memoryJson});
519 }
520 }
521 return Result<AgentArchive>::success(std::move(result));
522}

524Result<AgentHandle> NpcAiWorld::restoreAgent(const AgentArchive& archive) {
525 if (archive.schemaId != AgentArchive::SchemaId || archive.schemaVersion != AgentArchive::SchemaVersion)
526 return failure<AgentHandle>(DiagnosticCode::Unsupported, "unsupported NPC agent archive schema");
527 const auto behaviorIt = behaviors_.find(archive.behaviorId);
528 if (behaviorIt == behaviors_.end())
529 return failure<AgentHandle>(DiagnosticCode::NotFound, "archive behavior definition was not registered");
530 const auto& behavior = behaviorIt->second;
531 if (!findState(behavior, archive.activeState))
532 return failure<AgentHandle>(DiagnosticCode::NotFound, "archive active state does not exist");
533 if (archive.perception.size() > config_.maxMemoriesPerAgent)
534 return failure<AgentHandle>(DiagnosticCode::PreconditionViolation,
535 "archive exceeds the perception memory budget");
536
537 Agent candidate;
538 candidate.behaviorId = archive.behaviorId;
539 candidate.activeState = archive.activeState;
540 candidate.blackboard = archive.blackboard;
541 candidate.signals = archive.pendingSignals;
542 candidate.perception = archive.perception;
543 candidate.lastTick = archive.lastTick;
544 for (const auto& signalName : candidate.signals)
545 if (signalName.empty())
546 return failure<AgentHandle>(DiagnosticCode::InvalidArgument, "archive contains an empty signal");
547 std::set<std::pair<std::string, std::string>> memoryKeys;
548 for (const auto& memory : candidate.perception) {
549 if (memory.subject.empty() || memory.sense.empty() || !std::isfinite(memory.confidence) ||
550 memory.confidence < 0.0 || memory.confidence > 1.0 || memory.forgetAfterTicks == 0)
551 return failure<AgentHandle>(DiagnosticCode::InvalidArgument, "archive contains invalid perception memory");
552 if (!memoryKeys.emplace(memory.subject, memory.sense).second)
553 return failure<AgentHandle>(DiagnosticCode::InvalidArgument,
554 "archive contains duplicate perception memory");
555 }
556
557 if (!behavior.blackboardSchema.empty()) {
558 if (candidate.blackboard.size() > behavior.blackboardSchema.size())
559 return failure<AgentHandle>(DiagnosticCode::InvalidArgument, "archive contains undeclared blackboard keys");
560 for (const auto& [key, value] : candidate.blackboard) {
561 const auto spec = std::find_if(behavior.blackboardSchema.begin(), behavior.blackboardSchema.end(),
562 [&](const auto& item) { return item.key == key; });
563 if (spec == behavior.blackboardSchema.end() || spec->type != valueType(value))
564 return failure<AgentHandle>(DiagnosticCode::InvalidArgument,
565 "archive blackboard does not match the behavior schema");
566 }
567 for (const auto& spec : behavior.blackboardSchema)
568 if (spec.required && !candidate.blackboard.contains(spec.key))
569 return failure<AgentHandle>(DiagnosticCode::InvalidArgument,
570 "archive is missing a required blackboard key");
571 }
572
573 const auto path = activePath(behavior, candidate.activeState);
574 if (!std::all_of(path.begin(), path.end(),
575 [&](const auto state) { return predicatesPass(state.get().enterConditions, candidate); }))
576 return failure<AgentHandle>(DiagnosticCode::PreconditionViolation,
577 "archive active path enter conditions are false");
578 std::set<std::string> validTasks;
579 for (const auto state : path)
580 for (const auto& task : state.get().tasks) validTasks.insert(taskRuntimeKey(state.get().id, task.id));
581 std::set<std::string> archivedTasks;
582 for (const auto& task : archive.tasks) {
583 const auto key = taskRuntimeKey(task.stateId, task.taskId);
584 if (!validTasks.contains(key) || !archivedTasks.insert(key).second)
585 return failure<AgentHandle>(DiagnosticCode::InvalidArgument,
586 "archive task is duplicated or outside the active path");
587 candidate.taskRuntime.emplace(key, TaskRuntime{false, task.completed, task.memoryJson});
588 }
589 std::sort(candidate.perception.begin(), candidate.perception.end(), [](const auto& a, const auto& b) {
590 return std::pair{a.subject, a.sense} < std::pair{b.subject, b.sense};
591 });
592
593 std::uint32_t index;
594 if (freeSlots_.empty()) {
595 index = static_cast<std::uint32_t>(slots_.size());
596 slots_.push_back({});
597 } else {
598 index = freeSlots_.back();
599 freeSlots_.pop_back();
600 }
601 slots_[index].agent = std::move(candidate);
602 const AgentHandle handle(index, slots_[index].generation);
603 pushTrace({archive.lastTick, handle, TraceKind::AgentRestored, archive.behaviorId, archive.activeState});
604 return Result<AgentHandle>::success(handle, Status::success(StatusCode::Applied));
605}

607bool NpcAiWorld::isStale(AgentHandle handle) const noexcept {
608 if (!handle.isValid() || handle.index() >= slots_.size()) return true;
609 const auto& slot = slots_[handle.index()];
610 return !slot.agent || slot.generation != handle.generation();
611}
