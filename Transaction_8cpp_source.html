18std::string quote(const std::string& value) {
19 std::ostringstream out;
20 out << '"';
21 for (unsigned char c : value) {
22 switch (c) {
23 case '"': out << "\\\""; break;
24 case '\\': out << "\\\\"; break;
25 case '\b': out << "\\b"; break;
26 case '\f': out << "\\f"; break;
27 case '\n': out << "\\n"; break;
28 case '\r': out << "\\r"; break;
29 case '\t': out << "\\t"; break;
30 default:
31 if (c < 0x20)
32 out << "\\u" << std::hex << std::setw(4) << std::setfill('0') << int(c) << std::dec;
33 else
34 out << static_cast<char>(c);
35 }
36 }
37 return out.str() + '"';
38}

40std::string canonicalJson(const eve::json::Value& value) {
41 if (value.isNull()) return "null";
42 if (value.isBool()) return value.asBool() ? "true" : "false";
43 if (value.isNumber()) return value.asString();
44 if (value.isString()) return quote(value.asString());
45 if (value.isArray()) {
46 std::string out = "[";
47 for (size_t i = 0; i < value.size(); ++i) {
48 if (i) out += ',';
49 out += canonicalJson(value.at(i));
50 }
51 return out + ']';
52 }
53 if (value.isObject()) {
54 auto keys = value.keys();
55 std::sort(keys.begin(), keys.end());
56 std::string out = "{";
57 for (size_t i = 0; i < keys.size(); ++i) {
58 if (i) out += ',';
59 out += quote(keys[i]) + ':' + canonicalJson(value.get(keys[i].c_str()));
60 }
61 return out + '}';
62 }
63 return {};
64}

187eve::Result<TransactionReceipt> Coordinator::execute(const TransactionContext& context,
188 std::span<ITransactionParticipant*> participants) const {
189 if (context.transactionId().empty())
190 return coordinatorFailure(eve::StatusCode::Rejected,
191 {eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
192 "transaction id must not be empty", "transactionId")});
193 if (participants.empty())
194 return coordinatorFailure(
195 eve::StatusCode::Rejected,
196 {eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
197 "transaction requires at least one participant", "participants")});
198 for (size_t i = 0; i < participants.size(); ++i) {
199 if (!participants[i])
200 return coordinatorFailure(
201 eve::StatusCode::Rejected,
202 {eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "participant must not be null",
203 "participants[" + std::to_string(i) + "]")});
204 for (size_t previous = 0; previous < i; ++previous)
205 if (participants[previous] == participants[i])
206 return coordinatorFailure(eve::StatusCode::Rejected,
207 {eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
208 "a participant may occur only once in a transaction",
209 "participants[" + std::to_string(i) + "]")});
210 }
211
212 TransactionReceipt receipt;
213 receipt.identity = context.identity();
214 receipt.transactionId = context.transactionId();
215 receipt.correlationId = context.correlationId();
216 receipt.causationId = context.causationId();
217 receipt.participantCount = participants.size();
218
219 std::vector<size_t> prepared;
220 prepared.reserve(participants.size());
221 for (size_t i = 0; i < participants.size(); ++i) {
222 auto result = invokeParticipant([&] { return participants[i]->prepare(context); }, "prepare", *participants[i]);
223 if (!result) {
224 std::vector<eve::Diagnostic> diagnostics;
225 appendFailure(diagnostics, result.status(), "prepare", i, *participants[i]);
226 bool cleanupFailed = false;
227 for (auto it = prepared.rbegin(); it != prepared.rend(); ++it) {
228 auto rollback = invokeParticipant([&] { return participants[*it]->rollback(context); }, "rollback",
229 *participants[*it]);
230 if (!rollback) {
231 cleanupFailed = true;
232 appendFailure(diagnostics, rollback.status(), "rollback", *it, *participants[*it]);
233 }
234 }
235 return coordinatorFailure(cleanupFailed ? eve::StatusCode::Failed : result.code(), std::move(diagnostics));
236 }
237 prepared.push_back(i);
238 }
239 receipt.preparedCount = prepared.size();
240
241 std::vector<size_t> committed;
242 committed.reserve(prepared.size());
243 for (size_t i : prepared) {
244 auto result = invokeParticipant([&] { return participants[i]->commit(context); }, "commit", *participants[i]);
245 if (!result) {
246 std::vector<eve::Diagnostic> diagnostics;
247 appendFailure(diagnostics, result.status(), "commit", i, *participants[i]);
248 bool cleanupFailed = false;
249
250 // A failed commit is contractually still uncommitted. Roll it and
251 // every later prepared participant back before compensating effects
252 // that were already made observable by earlier commits.
253 for (auto it = prepared.rbegin(); it != prepared.rend(); ++it) {
254 if (std::find(committed.begin(), committed.end(), *it) != committed.end()) continue;
255 auto rollback = invokeParticipant([&] { return participants[*it]->rollback(context); }, "rollback",
256 *participants[*it]);
257 if (!rollback) {
258 cleanupFailed = true;
259 appendFailure(diagnostics, rollback.status(), "rollback", *it, *participants[*it]);
260 }
261 }
262 for (auto it = committed.rbegin(); it != committed.rend(); ++it) {
263 auto compensation = invokeParticipant([&] { return participants[*it]->compensate(context); },
264 "compensation", *participants[*it]);
265 if (!compensation) {
266 cleanupFailed = true;
267 appendFailure(diagnostics, compensation.status(), "compensation", *it, *participants[*it]);
268 }
269 }
270 return coordinatorFailure(cleanupFailed ? eve::StatusCode::Failed : result.code(), std::move(diagnostics));
271 }
272 committed.push_back(i);
273 }
274
275 receipt.committedCount = committed.size();
276 receipt.state = CoordinatorState::Committed;
277 return eve::Result<TransactionReceipt>::success(std::move(receipt), eve::Status::success(eve::StatusCode::Applied));
278}

280eve::Result<TransactionReceipt> Coordinator::compensate(const TransactionContext& context,
281 std::span<ITransactionParticipant*> participants) const {
282 if (context.transactionId().empty())
283 return coordinatorFailure(eve::StatusCode::Rejected,
284 {eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
285 "transaction id must not be empty", "transactionId")});
286 if (participants.empty())
287 return coordinatorFailure(
288 eve::StatusCode::Rejected,
289 {eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
290 "compensation requires at least one participant", "participants")});
291 for (size_t i = 0; i < participants.size(); ++i) {
292 if (!participants[i])
293 return coordinatorFailure(
294 eve::StatusCode::Rejected,
295 {eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "participant must not be null",
296 "participants[" + std::to_string(i) + "]")});
297 for (size_t previous = 0; previous < i; ++previous)
298 if (participants[previous] == participants[i])
299 return coordinatorFailure(eve::StatusCode::Rejected,
300 {eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
301 "a participant may occur only once in compensation",
302 "participants[" + std::to_string(i) + "]")});
303 }
304
305 TransactionReceipt receipt;
306 receipt.identity = context.identity();
307 receipt.transactionId = context.transactionId();
308 receipt.correlationId = context.correlationId();
309 receipt.causationId = context.causationId();
310 receipt.participantCount = participants.size();
311 std::vector<eve::Diagnostic> diagnostics;
312 for (size_t offset = 0; offset < participants.size(); ++offset) {
313 const size_t index = participants.size() - 1 - offset;
314 auto result = invokeParticipant([&] { return participants[index]->compensate(context); }, "compensation",
315 *participants[index]);
316 if (!result) {
317 appendFailure(diagnostics, result.status(), "compensation", index, *participants[index]);
318 continue;
319 }
320 ++receipt.compensatedCount;
321 }
322 if (!diagnostics.empty()) {
323 receipt.state = CoordinatorState::CompensationFailed;
324 return coordinatorFailure(eve::StatusCode::Failed, std::move(diagnostics));
325 }
326 receipt.state = CoordinatorState::Compensated;
327 return eve::Result<TransactionReceipt>::success(std::move(receipt), eve::Status::success(eve::StatusCode::Applied));
328}

366eve::Result<eve::OperationId> Plan::stage(const std::string& kind, const std::string& target,
367 const std::string& payloadJson, eve::OperationId operationId) {
368 const auto failure = [](eve::DiagnosticCode code, std::string message) {
369 return eve::Result<eve::OperationId>::failure(eve::Diagnostic::error(code, std::move(message)));
370 };
371 error_.clear();
372 if (state_ != State::Open) return failure(eve::DiagnosticCode::Conflict, "transaction plan is frozen");
373 if (kind.empty()) return failure(eve::DiagnosticCode::InvalidArgument, "operation kind must not be empty");
374 auto document = eve::json::Document::parse(payloadJson);
375 if (!document.valid()) return failure(eve::DiagnosticCode::ParseError, "payload must be valid JSON");
376 if (nextOperation_ == std::numeric_limits<uint64_t>::max())
377 return failure(eve::DiagnosticCode::Failed, "operation sequence exhausted");
378
379 if (operationId.isNil()) {
380 if (!operationIdGenerator_)
381 return failure(eve::DiagnosticCode::Unsupported,
382 "canonical operation identity requires an injected UUID entropy source");
383 const auto generated = operationIdGenerator_->generate();
384 if (!generated) return failure(eve::DiagnosticCode::Failed, "canonical operation identity generation failed");
385 operationId = eve::OperationId::fromUuid(*generated);
386 }
387 if (findOperation(operationId))
388 return failure(eve::DiagnosticCode::Conflict, "operation identity already exists in this plan");
389
390 Operation operation;
391 operation.identity = operationId;
392 operation.id = operationId.format();
393 operation.kind = kind;
394 operation.target = target;
395 operation.payload = canonicalJson(document.root());
396 operations_.push_back(std::move(operation));
397 ++nextOperation_;
398 emit("staged", operations_.back().id);
399 return eve::Result<eve::OperationId>::success(operationId, eve::Status::success(eve::StatusCode::Applied));
400}

462eve::Result<void> Plan::validate() {
463 error_.clear();
464 if (state_ != State::Open) return planFailure(eve::DiagnosticCode::Conflict, "transaction is not open");
465 if (operations_.empty()) {
466 error_ = "transaction has no operations";
467 emit("validation_failed", {}, error_);
468 return planFailure(eve::DiagnosticCode::InvalidArgument, error_);
469 }
470 for (const auto& operation : operations_) {
471 if (!operation.checked || !operation.valid) {
472 error_ = operation.checked ? operation.error : "operation was not validated: " + operation.id;
473 emit("validation_failed", operation.id, error_);
474 return planFailure(eve::DiagnosticCode::InvalidArgument, error_);
475 }
476 }
477 state_ = State::Validated;
478 emit("validated");
479 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));

529std::string Plan::snapshotJson() const {
530 std::ostringstream out;
531 out << "{\"id\":" << quote(id_);
532 if (!identity_.isNil()) out << ",\"identity\":" << quote(identity_.format());
533 out << ",\"state\":" << quote(stateName(state_)) << ",\"correlation\":" << quote(correlation_)
534 << ",\"causation\":" << quote(causation_) << ",\"error\":" << quote(error_)
535 << ",\"nextOperation\":" << quote(std::to_string(nextOperation_))
536 << ",\"nextEvent\":" << quote(std::to_string(nextEvent_)) << ",\"operations\":[";
537 for (size_t i = 0; i < operations_.size(); ++i) {
538 if (i) out << ',';
539 const auto& op = operations_[i];
540 out << "{\"id\":" << quote(op.id);
541 if (!op.identity.isNil()) out << ",\"identity\":" << quote(op.identity.format());

603eve::Result<void> Ledger::restore(std::string_view json) {
604 const auto failure = [](eve::DiagnosticCode code, std::string message) {
605 return eve::Result<void>::failure(eve::Diagnostic::error(code, std::move(message), "transaction.ledger"));
606 };
607 auto document = eve::json::Document::parse(std::string(json));
608 auto root = document.root();
609 uint64_t restoredNext = 0;
610 if (!document.valid() || !root.isObject() || root.getInt("version") != 1 ||
611 !parseU64(root.get("nextTransaction"), restoredNext) || restoredNext == 0 || !root.get("plans").isArray()) {
612 return failure(eve::DiagnosticCode::ParseError, "invalid transaction ledger snapshot");
613 }
614 std::vector<std::unique_ptr<Plan>> restored;
615 uint64_t previousNumeric = 0;
616 std::vector<std::string> restoredPlanIds;
617 auto plans = root.get("plans");
618 for (size_t i = 0; i < plans.size(); ++i) {
619 auto value = plans.at(i);
620 State state = State::Open;
621 uint64_t nextOperation = 0, nextEvent = 0;
622 if (!value.isObject() || !parseState(value.getString("state"), state) ||
623 !parseU64(value.get("nextOperation"), nextOperation) || nextOperation == 0 ||
624 !parseU64(value.get("nextEvent"), nextEvent) || nextEvent == 0 || !value.get("operations").isArray() ||
625 !value.get("events").isArray()) {
626 return failure(eve::DiagnosticCode::ParseError, "invalid transaction plan at index " + std::to_string(i));
627 }
628 const std::string id = value.getString("id");
629 const std::string identityText = value.getString("identity");
630 eve::TransactionId identity;
631 bool canonicalIdentity = false;
632 if (!identityText.empty()) {
633 const auto parsed = eve::TransactionId::parse(identityText);
634 if (!parsed || parsed->isNil() || id != parsed->format()) {
635 return failure(eve::DiagnosticCode::ParseError,
636 "invalid canonical transaction id at index " + std::to_string(i));
637 }
638 identity = *parsed;
639 canonicalIdentity = true;
640 } else if (const auto parsed = eve::TransactionId::parse(id); parsed && !parsed->isNil()) {
641 identity = *parsed;
642 canonicalIdentity = true;
643 }
644
645 uint64_t numeric = 0;
646 if (!canonicalIdentity) {
647 const std::string prefix = "transaction-";
648 try {
649 numeric = std::stoull(id.substr(prefix.size()));
650 } catch (...) {
651 numeric = 0;
652 }
653 if (!id.starts_with(prefix) || numeric <= previousNumeric || numeric >= restoredNext) {
654 return failure(eve::DiagnosticCode::ParseError, "invalid transaction id at index " + std::to_string(i));
655 }
656 }
657 if (id.empty() || std::find(restoredPlanIds.begin(), restoredPlanIds.end(), id) != restoredPlanIds.end()) {
658 return failure(eve::DiagnosticCode::Conflict, "duplicate transaction id at index " + std::to_string(i));
659 }
660 restoredPlanIds.push_back(id);
661 auto plan = std::unique_ptr<Plan>(new Plan(id, value.getString("correlation"), value.getString("causation"),
662 identity, transactionEntropy_, transactionClock_));
663 plan->state_ = state;
664 plan->error_ = value.getString("error");
665 plan->operations_.clear();
666 plan->events_.clear();
667 auto operations = value.get("operations");
668 std::vector<std::string> restoredOperationIds;
669 for (size_t j = 0; j < operations.size(); ++j) {
670 auto opValue = operations.at(j);
671 if (!opValue.isObject() || opValue.getString("id").empty() || opValue.getString("kind").empty() ||
672 !opValue.get("payload") || !opValue.get("checked").isBool() || !opValue.get("valid").isBool()) {
673 return failure(eve::DiagnosticCode::ParseError, "invalid operation in plan " + id);
674 }
675 Operation op;
676 op.id = opValue.getString("id");
677 const std::string operationIdentityText = opValue.getString("identity");
678 if (!operationIdentityText.empty()) {
679 const auto parsed = eve::OperationId::parse(operationIdentityText);
680 if (!parsed || parsed->isNil() || op.id != parsed->format()) {
681 return failure(eve::DiagnosticCode::ParseError, "invalid canonical operation id in plan " + id);
682 }
683 op.identity = *parsed;
684 } else if (const auto parsed = eve::OperationId::parse(op.id); parsed && !parsed->isNil()) {
685 op.identity = *parsed;
686 }
687 if (std::find(restoredOperationIds.begin(), restoredOperationIds.end(), op.id) !=
688 restoredOperationIds.end()) {
689 return failure(eve::DiagnosticCode::Conflict, "duplicate operation id in plan " + id);
690 }
691 restoredOperationIds.push_back(op.id);
692 op.kind = opValue.getString("kind");
693 op.target = opValue.getString("target");
694 op.payload = canonicalJson(opValue.get("payload"));
695 op.checked = opValue.getBool("checked");
696 op.valid = opValue.getBool("valid");
697 op.error = opValue.getString("error");
698 plan->operations_.push_back(std::move(op));
699 }
700 uint64_t previousEvent = 0;
701 auto events = value.get("events");
702 for (size_t j = 0; j < events.size(); ++j) {
703 auto eventValue = events.at(j);
704 Event event;
705 if (!eventValue.isObject() || !parseU64(eventValue.get("sequence"), event.sequence) ||
706 event.sequence <= previousEvent || eventValue.getString("transactionId") != id ||
707 eventValue.getString("type").empty()) {
708 return failure(eve::DiagnosticCode::ParseError, "invalid event in plan " + id);
709 }
710 event.transactionId = id;
711 event.operationId = eventValue.getString("operationId");
712 event.type = eventValue.getString("type");
713 event.detail = eventValue.getString("detail");
714 event.transactionIdentity = identity;
715 if (const auto parsed = eve::OperationId::parse(event.operationId)) event.operationIdentity = *parsed;
716 previousEvent = event.sequence;
717 plan->events_.push_back(std::move(event));
718 }
719 if (!plan->events_.empty() && plan->events_.back().sequence >= nextEvent) {
720 return failure(eve::DiagnosticCode::ParseError, "invalid event allocator in plan " + id);
721 }
722 plan->nextOperation_ = nextOperation;
723 plan->nextEvent_ = nextEvent;
724 if (!canonicalIdentity) previousNumeric = numeric;
725 restored.push_back(std::move(plan));
726 }
727 plans_ = std::move(restored);
728 nextTransaction_ = restoredNext;
729 return eve::Result<void>::success();
730}

739eve::Result<void> Ledger::restoreSnapshot(const eve::SnapshotEnvelope& source,
740 const eve::SnapshotHashProvider& hashProvider) {
741 if (source.type != "transaction.ledger" || source.schema != transactionSchema())
742 return snapshotFailure<void>(eve::DiagnosticCode::InvalidArgument,
743 "snapshot does not belong to transaction::Ledger");
744 if (!instanceId_.isNil() && source.instanceId != instanceId_)
745 return snapshotFailure<void>(eve::DiagnosticCode::Conflict,
746 "snapshot instanceId does not match transaction::Ledger");
747
748 auto migrated = transactionMigrations().migrate(source, eve::SchemaVersion(1), hashProvider);
749 if (!migrated.ok()) return eve::Result<void>::failure(migrated.status());
750 const auto& candidateEnvelope = migrated.value();
751 auto metadata = eve::validateSnapshotPayloadMetadata(candidateEnvelope.payload, candidateEnvelope.revision,
752 candidateEnvelope.tick);
753 if (!metadata.ok()) return eve::Result<void>::failure(metadata.status());
754 auto payload = candidateEnvelope.payload.toJson();
755 if (!payload.ok()) return eve::Result<void>::failure(payload.status());
756
757 Ledger candidate(instanceId_, transactionEntropy_, transactionClock_);
758 auto restored = candidate.restore(std::move(payload).takeValue());
759 if (!restored.ok()) return eve::Result<void>::failure(restored.status());
760 candidate.instanceId_ = candidateEnvelope.instanceId;
761 candidate.revision_ = candidateEnvelope.revision;
762 candidate.tick_ = candidateEnvelope.tick;
763 *this = std::move(candidate);
764 return eve::Result<void>::success();
765}
