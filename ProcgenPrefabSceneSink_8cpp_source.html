24 auto matrix = glm::translate(glm::mat4(1.f), glm::vec3(value.x, value.y, value.z));

84 Batches candidates;
85 std::size_t parts = 0, instances = 0;

88 instances += batch.source.size();

101 if (instances > 65536 || snapshot.instances.size() > 65536 - instances)
102 return Output::failure(
103 Diagnostic::error(DiagnosticCode::PreconditionViolation, "prefab instance budget exceeded"));
104 instances += snapshot.instances.size();

106 batch.revision = snapshot.targetRevision;
107 batch.source = snapshot.instances;
108 std::unordered_set<std::string> ids;
109 std::unordered_set<uint64_t> pointIds;
110 for (const auto& value : snapshot.instances) {
111 if (value.id.empty() || !value.sourcePointId || !ids.insert(value.id).second ||
112 !pointIds.insert(value.sourcePointId).second)
113 return Output::failure(
114 Diagnostic::error(DiagnosticCode::Conflict, "prefab identities must be unique"));
115 for (float scalar : {value.x, value.y, value.z, value.yaw, value.pitch, value.roll, value.scaleX,
116 value.scaleY, value.scaleZ})
117 if (!std::isfinite(scalar))
118 return Output::failure(
119 Diagnostic::error(DiagnosticCode::InvalidArgument, "nonfinite prefab pose"));
120 if (!value.assetBinding) continue;
121 const auto& binding = *value.assetBinding;
122 if (binding.buildId.isNil() || value.asset != binding.asset.format())
123 return Output::failure(
124 Diagnostic::error(DiagnosticCode::InvalidArgument, "invalid typed prefab binding"));
125 const auto prefab = assets.find({binding.asset.id(), binding.buildId});
126 if (prefab == assets.end()) {
127 const auto firstVersion = assets.lower_bound({binding.asset.id(), PersistentId{}});
128 const auto code = firstVersion == assets.end() || firstVersion->first.first != binding.asset.id()
129 ? DiagnosticCode::NotFound

132 Diagnostic::error(code, "prefab asset/build version is not admitted", value.asset));
133 }

138 parts += count;
139 batch.parts += count;
140 auto preparation =
141 EvpackStaticPrefab::prepareScene(prefab->second.prefab, graphics, value.assetBinding->rootPose);
142 if (!preparation.ok()) return Output::failure(preparation.status());
143 auto instance =
144 StaticPrefabRenderInstance::create(std::move(preparation).takeValue(), instanceMatrix(value));
145 if (!instance.ok()) return Output::failure(instance.status());
146 visibility(*instance.value(), false);
147 batch.renderers.push_back(std::move(instance).takeValue());
148 }
149 }
150 std::size_t added = 0;
151 for (const auto& [id, batch] : candidates)
152 if (!active.contains(id)) ++added;
153 if (active.size() + added > 64)
154 return Output::failure(
155 Diagnostic::error(DiagnosticCode::PreconditionViolation, "active prefab batch budget exceeded"));
156 return Output::success(std::move(candidates));
157 }
158 void commit(Batches& candidates) {
159 while (!candidates.empty()) {
160 auto node = candidates.extract(candidates.begin());
161 const auto old = active.find(node.key());
162 if (old != active.end()) {
163 visibility(old->second, false);
164 retired.push_back(active.extract(old));
165 }
166 visibility(node.mapped(), true);
167 auto inserted = active.insert(std::move(node));
168 EV_ASSERT(inserted.inserted, "prepared prefab batch key must be absent at commit");
169 }

224 if (const auto found = impl_->active.find(id); found != impl_->active.end())

226 for (const auto& value : target)
