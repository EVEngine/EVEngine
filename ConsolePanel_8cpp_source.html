47// One console line has to survive a full debug dump: the previous fixed 1 KiB
48// buffer silently cut long prints, and truncated output is exactly what an
49// unattended agent cannot afford to miss. Over-long lines are now truncated
50// explicitly, and the marker says how much was dropped.
51constexpr std::size_t kMaxCapturedLine = 64 * 1024;
52
53std::string formatCaptured(const SQChar* s, va_list args) {
54 const char* format = s ? s : "";
55 va_list probe;
56 va_copy(probe, args);
57 const int needed = std::vsnprintf(nullptr, 0, format, probe);
58 va_end(probe);
59 if (needed < 0) return std::string(format);
60
61 const std::size_t size = static_cast<std::size_t>(needed);
62 const std::size_t keep = std::min(size, kMaxCapturedLine);
63 std::vector<char> buffer(keep + 1, '\0');
64 std::vsnprintf(buffer.data(), buffer.size(), format, args);
65 std::string text(buffer.data(), keep);
66 if (size > keep) {
67 text += "\n[console: truncated " + std::to_string(size - keep) + " of " + std::to_string(size) + " bytes]";
68 }
69 return text;
70}

188void ConsolePanel::appendLocked(std::string level, std::string text) {
189 ConsoleLine line;
190 line.seq = nextSeq_++;
191 line.timestamp = nowStamp();
192 line.level = level;
193 line.text = text;
194 log_.push_back(std::move(line));
195 while (log_.size() > maxEntries_) {
196 droppedThrough_ = log_.front().seq;
197 log_.pop_front();
198 }
199 // Errors also go to the persistent crash/error log: after a crash the MCP
200 // session is gone, and eve.log is the only evidence of what went wrong.
201 if (level == "error") eve::recordLogEvent("error", text);
202}

204void ConsolePanel::addEngineLine(const std::string& text) {
205 if (text.empty()) return;
206 std::lock_guard<std::mutex> lock(mu_);
207 // capturePrint/captureError forward to stderr, so the descriptor-level
208 // capture sees the same text a moment later. Drop only that echo: a genuine
209 // repeat emitted by the engine itself keeps its own entry.
210 if (!log_.empty()) {
211 const ConsoleLine& previous = log_.back();
212 if (previous.text == text && (previous.level == "print" || previous.level == "error")) return;
213 }
214 appendLocked("engine", text);
215}

252ConsoleSlice ConsolePanel::read(std::uint64_t afterSeq, size_t max, const std::string& level) const {
253 std::lock_guard<std::mutex> lock(mu_);
254
255 ConsoleSlice slice;
256 slice.nextSeq = nextSeq_;
257 slice.droppedThrough = droppedThrough_;
258 slice.firstSeq = log_.empty() ? nextSeq_ : log_.front().seq;
259
260 const size_t want = max == 0 ? 64 : max;
261 if (afterSeq == 0) {
262 // Tail read: newest matching lines, returned oldest first.
263 slice.truncated = droppedThrough_ > 0;
264 for (auto it = log_.rbegin(); it != log_.rend() && slice.lines.size() < want; ++it) {
265 if (!level.empty() && it->level != level) continue;
266 slice.lines.push_back(*it);
267 }
268 std::reverse(slice.lines.begin(), slice.lines.end());
269 } else {
270 slice.truncated = afterSeq < droppedThrough_;
271 for (const auto& line : log_) {
272 if (line.seq <= afterSeq) continue;
273 if (!level.empty() && line.level != level) continue;
274 if (slice.lines.size() >= want) break;
275 slice.lines.push_back(line);
276 }
277 }
278 // The resumable cursor is the last line actually covered by this read. It is
279 // deliberately not `nextSeq`: that is the seq the *next* appended line will
280 // receive, and paging from it would skip exactly one line.
281 slice.cursor = slice.lines.empty() ? afterSeq : slice.lines.back().seq;
282 return slice;
283}

303void ConsolePanel::attach(HSQUIRRELVM vm) {
304 if (!vm) return;
305 detach();
306 vm_ = vm;
307 // Snapshot existing handlers once (they may be the stdlib defaults).
308 if (!g_prevPrint) g_prevPrint = sq_getprintfunc(vm);
309 if (!g_prevError) g_prevError = sq_geterrorfunc(vm);
310 sq_setprintfunc(vm, capturePrint, captureError);
311 addLog("info", "console attached to VM");
312 // Engine-side diagnostics (std::cerr and fprintf(stderr)) are the other half
313 // of a scripted game's output; mirror them into the same ordered stream.
314 const StderrCaptureStatus capture =
315 startStderrCapture([](const std::string& line) { ConsolePanel::instance().addEngineLine(line); });
316 // A handler-less capture must be observable, not silent: an agent that sees
317 // no `engine` lines has to know the coverage is missing rather than assume
318 // the engine printed nothing.
319 if (capture != StderrCaptureStatus::Active) {
320 addLog("warn", "engine stderr capture unavailable; only script print/error reaches this console");
321 }
322}

324void ConsolePanel::detach() {

332std::string ConsolePanel::eval(const std::string& expression) {
333 if (expression.empty()) return "error: empty expression";
334 HSQUIRRELVM vm = vm_;
335 if (!vm) return "error: no VM attached";
336 addLog("cmd", expression);
337
338 const SQInteger top = sq_gettop(vm);
339 // Compile `return (expr);` so the evaluation result lands on the stack.
340 const std::string source = "return (" + expression + ");";
341 if (SQ_FAILED(eve::script::ScriptCompiler::compileBuffer(
342 vm, source.c_str(), static_cast<SQInteger>(source.size()), _SC("console_repl.nut"), SQTrue))) {
343 sq_settop(vm, top);
344 const eve::script::ScriptErrorContext ctx = eve::script::captureCompileError(vm);
345 const std::string err = "error: " +
346 (ctx.empty() ? std::string("compile failed")
347 : eve::script::formatScriptError(ctx));
348 addLog("error", err);
349 return err;
350 }
351 sq_pushroottable(vm);
352 if (SQ_FAILED(sq_call(vm, 1, SQTrue, SQTrue))) {
353 sq_settop(vm, top);
354 eve::script::ScriptErrorContext ctx = eve::script::takeLastScriptError(vm);
355 const std::string err = "error: " +
356 (ctx.empty() ? std::string("runtime failed")
357 : eve::script::formatScriptError(ctx));
358 addLog("error", err);
359 return err;
360 }
361 std::string result = formatValue(vm, -1);
362 sq_settop(vm, top);
363 addLog("result", result);
364 return result;
365}
