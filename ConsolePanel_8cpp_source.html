42// Squirrel print callback (varargs, printf-style). Captured into the console
43// log, then forwarded to the previous handler so stdout/stderr behavior stays.
44void capturePrint(HSQUIRRELVM v, const SQChar* s, ...) {
45 va_list args;
46 va_start(args, s);
47 char buf[1024];
48 vsnprintf(buf, sizeof(buf), s ? s : "", args);
49 va_end(args);
50 ConsolePanel::instance().addLog("print", buf);
51 forwardPrint(v, buf);
52}

85std::string formatValue(HSQUIRRELVM vm, SQInteger idx) {
86 switch (sq_gettype(vm, idx)) {
87 case OT_NULL: return "null";
88 case OT_BOOL: {
89 SQBool b = SQFalse;
90 sq_getbool(vm, idx, &b);
91 return b ? "true" : "false";
92 }
93 case OT_INTEGER: {
94 SQInteger v = 0;
95 sq_getinteger(vm, idx, &v);
96 return std::to_string(static_cast<long long>(v));
97 }
98 case OT_FLOAT: {
99 SQFloat v = 0;
100 sq_getfloat(vm, idx, &v);
101 std::ostringstream oss;
102 oss << static_cast<double>(v);
103 return oss.str();
104 }
105 case OT_STRING: {
106 const SQChar* s = nullptr;
107 sq_getstring(vm, idx, &s);
108 return std::string("\"") + (s ? s : "") + "\"";
109 }
110 case OT_TABLE: return "<table>";
111 case OT_ARRAY: return "<array>";
112 case OT_CLOSURE: return "<closure>";
113 case OT_NATIVECLOSURE: return "<native>";
114 case OT_USERDATA: return "<userdata>";
115 case OT_INSTANCE: return "<instance>";
116 case OT_CLASS: return "<class>";
117 case OT_THREAD: return "<thread>";
118 default: return "<" + typeName(vm, idx) + ">";
119 }

182std::vector<ConsoleLine> ConsolePanel::recent(size_t max) const {
183 std::lock_guard<std::mutex> lock(mu_);
184 std::vector<ConsoleLine> out;
185 if (log_.empty() || max == 0) return out;

200void ConsolePanel::attach(HSQUIRRELVM vm) {
201 if (!vm) return;
202 detach();
203 vm_ = vm;
204 // Snapshot existing handlers once (they may be the stdlib defaults).
205 if (!g_prevPrint) g_prevPrint = sq_getprintfunc(vm);
206 if (!g_prevError) g_prevError = sq_geterrorfunc(vm);
207 sq_setprintfunc(vm, capturePrint, captureError);
208 addLog("info", "console attached to VM");
209}

218std::string ConsolePanel::eval(const std::string& expression) {
219 if (expression.empty()) return "error: empty expression";
220 HSQUIRRELVM vm = vm_;
221 if (!vm) return "error: no VM attached";
222 addLog("cmd", expression);
223
224 const SQInteger top = sq_gettop(vm);
225 // Compile `return (expr);` so the evaluation result lands on the stack.
226 const std::string source = "return (" + expression + ");";
227 if (SQ_FAILED(eve::script::ScriptCompiler::compileBuffer(
228 vm, source.c_str(), static_cast<SQInteger>(source.size()), _SC("console_repl.nut"), SQTrue))) {
229 sq_settop(vm, top);
230 const eve::script::ScriptErrorContext ctx = eve::script::captureCompileError(vm);
231 const std::string err = "error: " +
232 (ctx.empty() ? std::string("compile failed")
233 : eve::script::formatScriptError(ctx));
234 addLog("error", err);
235 return err;
236 }
237 sq_pushroottable(vm);
238 if (SQ_FAILED(sq_call(vm, 1, SQTrue, SQTrue))) {
239 sq_settop(vm, top);
240 eve::script::ScriptErrorContext ctx = eve::script::takeLastScriptError(vm);
241 const std::string err = "error: " +
242 (ctx.empty() ? std::string("runtime failed")
243 : eve::script::formatScriptError(ctx));
244 addLog("error", err);
245 return err;
246 }
247 std::string result = formatValue(vm, -1);
248 sq_settop(vm, top);
249 addLog("result", result);
250 return result;
251}
