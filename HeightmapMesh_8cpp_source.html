22void buildSmooth(const procgen::Heightmap& heightmap, float cellSize, float heightScale,
23 HeightmapArrays& output) {
24 const int width = heightmap.getWidth();
25 const int height = heightmap.getHeight();
26 if (width < 2 || height < 2 || cellSize == 0.0F) return;
27 output.positions.reserve(static_cast<std::size_t>(width * height * 3));
28 output.normals.reserve(static_cast<std::size_t>(width * height * 3));
29 output.texcoords.reserve(static_cast<std::size_t>(width * height * 2));
30 output.indices.reserve(static_cast<std::size_t>((width - 1) * (height - 1) * 6));
31 auto sample = [&](int x, int y) {
32 return heightmap.height(std::clamp(x, 0, width - 1), std::clamp(y, 0, height - 1));
33 };
34 for (int y = 0; y < height; ++y) {
35 for (int x = 0; x < width; ++x) {
36 output.positions.insert(output.positions.end(),
37 {static_cast<float>(x) * cellSize,
38 heightmap.height(x, y) * heightScale,
39 static_cast<float>(y) * cellSize});
40 float nx = -(sample(x + 1, y) - sample(x - 1, y)) * 0.5F * heightScale / cellSize;
41 float ny = 1.0F;
42 float nz = -(sample(x, y + 1) - sample(x, y - 1)) * 0.5F * heightScale / cellSize;
43 const float length = std::sqrt(nx * nx + ny * ny + nz * nz);
44 if (length > 1.0e-8F) {
45 nx /= length;
46 ny /= length;
47 nz /= length;
48 }
49 output.normals.insert(output.normals.end(), {nx, ny, nz});
50 output.texcoords.insert(output.texcoords.end(),
51 {static_cast<float>(x) / static_cast<float>(width - 1),
52 static_cast<float>(y) / static_cast<float>(height - 1)});
53 }
54 }
55 for (int y = 0; y < height - 1; ++y) {
56 for (int x = 0; x < width - 1; ++x) {
57 const std::uint32_t i00 = static_cast<std::uint32_t>(y * width + x);
58 const std::uint32_t i10 = i00 + 1U;
59 const std::uint32_t i01 = i00 + static_cast<std::uint32_t>(width);
60 const std::uint32_t i11 = i01 + 1U;
61 output.indices.insert(output.indices.end(), {i00, i01, i10, i10, i01, i11});
62 }
63 }
64}

66void addFlatTriangle(HeightmapArrays& output, float cellSize, float heightScale,
67 float textureWidth, float textureHeight, float ax, float ay, float ah,
68 float bx, float by, float bh, float cx, float cy, float ch) {
69 const float p0x = ax * cellSize, p0y = ah * heightScale, p0z = ay * cellSize;
70 const float p1x = bx * cellSize, p1y = bh * heightScale, p1z = by * cellSize;
71 const float p2x = cx * cellSize, p2y = ch * heightScale, p2z = cy * cellSize;
72 float nx = (p1y - p0y) * (p2z - p0z) - (p1z - p0z) * (p2y - p0y);
73 float ny = (p1z - p0z) * (p2x - p0x) - (p1x - p0x) * (p2z - p0z);
74 float nz = (p1x - p0x) * (p2y - p0y) - (p1y - p0y) * (p2x - p0x);
75 const float length = std::sqrt(nx * nx + ny * ny + nz * nz);
76 if (length > 1.0e-8F) {
77 nx /= length;
78 ny /= length;
79 nz /= length;
80 }
81 const std::uint32_t base = static_cast<std::uint32_t>(output.positions.size() / 3U);
82 output.positions.insert(output.positions.end(), {p0x, p0y, p0z, p1x, p1y, p1z, p2x, p2y, p2z});
83 output.normals.insert(output.normals.end(), {nx, ny, nz, nx, ny, nz, nx, ny, nz});
84 output.texcoords.insert(output.texcoords.end(),
85 {ax / textureWidth, ay / textureHeight, bx / textureWidth,
86 by / textureHeight, cx / textureWidth, cy / textureHeight});
87 output.indices.insert(output.indices.end(), {base, base + 1U, base + 2U});
88}

90void buildFlat(const procgen::Heightmap& heightmap, float cellSize, float heightScale,
91 HeightmapArrays& output) {
92 const int width = heightmap.getWidth();
93 const int height = heightmap.getHeight();
94 if (width < 2 || height < 2) return;
95 const float textureWidth = static_cast<float>(width - 1);
96 const float textureHeight = static_cast<float>(height - 1);
97 for (int y = 0; y < height - 1; ++y) {
98 for (int x = 0; x < width - 1; ++x) {
99 const float h00 = heightmap.height(x, y);
100 const float h10 = heightmap.height(x + 1, y);
101 const float h01 = heightmap.height(x, y + 1);
102 const float h11 = heightmap.height(x + 1, y + 1);
103 addFlatTriangle(output, cellSize, heightScale, textureWidth, textureHeight,
104 static_cast<float>(x), static_cast<float>(y), h00,
105 static_cast<float>(x), static_cast<float>(y + 1), h01,
106 static_cast<float>(x + 1), static_cast<float>(y), h10);
107 addFlatTriangle(output, cellSize, heightScale, textureWidth, textureHeight,
108 static_cast<float>(x + 1), static_cast<float>(y), h10,
109 static_cast<float>(x), static_cast<float>(y + 1), h01,
110 static_cast<float>(x + 1), static_cast<float>(y + 1), h11);
111 }
112 }
