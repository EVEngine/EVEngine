30inline std::string mcpJsonEscape(const std::string& value) {
31 std::string out;
32 out.reserve(value.size() + 8);
33 for (char c : value) {
34 switch (c) {
35 case '"': out += "\\\""; break;
36 case '\\': out += "\\\\"; break;
37 case '\n': out += "\\n"; break;
38 case '\r': out += "\\r"; break;
39 case '\t': out += "\\t"; break;
40 default:
41 if (static_cast<unsigned char>(c) < 0x20) {
42 char buf[8];
43 std::snprintf(buf, sizeof(buf), "\\u%04x", static_cast<unsigned>(c));
44 out += buf;
45 } else {
46 out += c;
47 }
48 break;
49 }
50 }
51 return out;
52}

59inline std::string textContentResult(const std::string& text, bool isError = false) {
60 Poco::JSON::Object::Ptr result = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
61 Poco::JSON::Array::Ptr content = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
62 Poco::JSON::Object::Ptr item = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
63 item->set("type", "text");
64 item->set("text", text);
65 content->add(item);
66 result->set("content", content);
67 if (isError) result->set("isError", true);
68 return mcpStringify(Poco::Dynamic::Var(result));
69}
