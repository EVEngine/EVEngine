152Result<void> validatePack(const ActionGuidePack& pack) {
153 if (pack.revision == 0) return invalidVoid("revision must be positive", "revision");
154 if (pack.subject.domain.empty()) return invalidVoid("subject.domain is required", "subject.domain");
155 if (pack.skeletonRef.empty()) return invalidVoid("skeletonRef is required", "skeletonRef");
156 if (pack.clipRef.empty()) return invalidVoid("clipRef is required", "clipRef");
157 if (pack.canvas.width <= 0 || pack.canvas.height <= 0) return invalidVoid("canvas size must be positive", "canvas");
158 if (!(pack.canvas.pixelsPerUnit > 0.0) || !std::isfinite(pack.canvas.pixelsPerUnit))
159 return invalidVoid("canvas.pixelsPerUnit must be finite and > 0", "canvas.pixelsPerUnit");
160 for (double channel : pack.canvas.backgroundRgba) {
161 if (!std::isfinite(channel) || channel < 0.0 || channel > 1.0)
162 return invalidVoid("canvas.backgroundRgba channels must be in [0,1]", "canvas.backgroundRgba");
163 }
164 if (!(pack.camera.nearPlane > 0.0) || !(pack.camera.farPlane > pack.camera.nearPlane) ||
165 !std::isfinite(pack.camera.nearPlane) || !std::isfinite(pack.camera.farPlane))
166 return invalidVoid("camera near/far must be finite with far > near > 0", "camera");
167 if (pack.camera.projection == ActionGuideProjection::Orthographic) {
168 if (!(pack.camera.orthoSize > 0.0) || !std::isfinite(pack.camera.orthoSize))
169 return invalidVoid("camera.orthoSize must be finite and > 0", "camera.orthoSize");
170 } else if (!(pack.camera.fovYDegrees > 0.0) || !(pack.camera.fovYDegrees < 180.0) ||
171 !std::isfinite(pack.camera.fovYDegrees)) {
172 return invalidVoid("camera.fovYDegrees must be in (0,180)", "camera.fovYDegrees");
173 }
174 const auto toEye =
175 std::array{pack.camera.target[0] - pack.camera.eye[0], pack.camera.target[1] - pack.camera.eye[1],
176 pack.camera.target[2] - pack.camera.eye[2]};
177 const double eyeLen2 = toEye[0] * toEye[0] + toEye[1] * toEye[1] + toEye[2] * toEye[2];
178 if (!(eyeLen2 > 1e-12)) return invalidVoid("camera eye and target must differ", "camera");
179 if (pack.shots.empty()) return invalidVoid("shots must be non-empty", "shots");
180 std::set<std::string, std::less<>> shotIds;
181 for (std::size_t i = 0; i < pack.shots.size(); ++i) {
182 const auto& shot = pack.shots[i];
183 const std::string path = "shots[" + std::to_string(i) + "]";
184 if (shot.shotId.empty()) return invalidVoid("shotId is required", path + ".shotId");
185 if (!shotIds.insert(shot.shotId).second) return invalidVoid("duplicate shotId", path + ".shotId");
186 if (!std::isfinite(shot.timeSeconds) || shot.timeSeconds < 0.0)
187 return invalidVoid("timeSeconds must be finite and >= 0", path + ".timeSeconds");
188 if (shot.sampleIndex && *shot.sampleIndex < 0)
189 return invalidVoid("sampleIndex must be >= 0", path + ".sampleIndex");
190 if (shot.view.empty()) return invalidVoid("view is required", path + ".view");
191 if (shot.aovs.empty()) return invalidVoid("aovs must be non-empty", path + ".aovs");
192 for (std::size_t a = 0; a < shot.aovs.size(); ++a) {
193 if (!allowedAov(shot.aovs[a]))
194 return invalidVoid("unsupported aov", path + ".aovs[" + std::to_string(a) + "]");
195 }
196 }
197 if (pack.actionBinding.abilityId.empty())
198 return invalidVoid("actionBinding.abilityId is required", "actionBinding.abilityId");
199 if (pack.actionBinding.clipName.empty())
200 return invalidVoid("actionBinding.clipName is required", "actionBinding.clipName");
201 if (!(pack.actionBinding.fps > 0.0) || !std::isfinite(pack.actionBinding.fps))
202 return invalidVoid("actionBinding.fps must be finite and > 0", "actionBinding.fps");
203 for (std::size_t i = 0; i < pack.actionBinding.phaseMarks.size(); ++i) {
204 const auto& mark = pack.actionBinding.phaseMarks[i];
205 const std::string path = "actionBinding.phaseMarks[" + std::to_string(i) + "]";
206 if (mark.name.empty()) return invalidVoid("phase mark name is required", path + ".name");
207 if (!std::isfinite(mark.timeSeconds) || mark.timeSeconds < 0.0)
208 return invalidVoid("phase mark timeSeconds must be finite and >= 0", path + ".timeSeconds");
209 }
210 if (!std::isfinite(pack.qc.minSilhouetteIoU) || pack.qc.minSilhouetteIoU < 0.0 || pack.qc.minSilhouetteIoU > 1.0)
211 return invalidVoid("qc.minSilhouetteIoU must be in [0,1]", "qc.minSilhouetteIoU");
212 if (!std::isfinite(pack.qc.maxJointErrorPx) || pack.qc.maxJointErrorPx < 0.0)
213 return invalidVoid("qc.maxJointErrorPx must be finite and >= 0", "qc.maxJointErrorPx");
214 if (pack.qc.maxRetries < 0) return invalidVoid("qc.maxRetries must be >= 0", "qc.maxRetries");
215 return Result<void>::success();
216}

394Mat4d mul(const Mat4d& a, const Mat4d& b) {
395 Mat4d out{};
396 for (int column = 0; column < 4; ++column) {
397 for (int row = 0; row < 4; ++row) {
398 out.m[column * 4 + row] = a.m[0 * 4 + row] * b.m[column * 4 + 0] + a.m[1 * 4 + row] * b.m[column * 4 + 1] +
399 a.m[2 * 4 + row] * b.m[column * 4 + 2] + a.m[3 * 4 + row] * b.m[column * 4 + 3];
400 }
401 }
402 return out;
403}

412Result<Mat4d> lookAt(const ActionGuideCamera& camera) {
413 Vec3d eye{camera.eye[0], camera.eye[1], camera.eye[2]};
414 Vec3d target{camera.target[0], camera.target[1], camera.target[2]};
415 Vec3d up{camera.up[0], camera.up[1], camera.up[2]};
416 auto forward = normalize(sub(target, eye), "camera");
417 if (!forward) return Result<Mat4d>::failure(forward.status());
418 auto right = normalize(cross(forward.value(), up), "camera.up");
419 if (!right) return Result<Mat4d>::failure(right.status());
420 auto trueUp = cross(right.value(), forward.value());
421 Mat4d view{};
422 view.m[0] = right.value().x;
423 view.m[4] = right.value().y;
424 view.m[8] = right.value().z;
425 view.m[1] = trueUp.x;
426 view.m[5] = trueUp.y;
427 view.m[9] = trueUp.z;
428 view.m[2] = -forward.value().x;
429 view.m[6] = -forward.value().y;
430 view.m[10] = -forward.value().z;
431 view.m[15] = 1.0;
432 view.m[12] = -dot(right.value(), eye);
433 view.m[13] = -dot(trueUp, eye);
434 view.m[14] = dot(forward.value(), eye);
435 return Result<Mat4d>::success(view);
436}

438Result<Mat4d> projectionMatrix(const ActionGuidePack& pack) {
439 const double aspect = static_cast<double>(pack.canvas.width) / static_cast<double>(pack.canvas.height);
440 Mat4d proj{};
441 if (pack.camera.projection == ActionGuideProjection::Orthographic) {
442 const double halfH = pack.camera.orthoSize;
443 const double halfW = halfH * aspect;
444 const double nearP = pack.camera.nearPlane;
445 const double farP = pack.camera.farPlane;
446 proj.m[0] = 1.0 / halfW;
447 proj.m[5] = 1.0 / halfH;
448 proj.m[10] = -2.0 / (farP - nearP);
449 proj.m[14] = -(farP + nearP) / (farP - nearP);
450 proj.m[15] = 1.0;
451 return Result<Mat4d>::success(proj);
452 }
453 const double f = 1.0 / std::tan((pack.camera.fovYDegrees * 0.5) * 3.14159265358979323846 / 180.0);
454 const double nearP = pack.camera.nearPlane;
455 const double farP = pack.camera.farPlane;
456 proj.m[0] = f / aspect;
457 proj.m[5] = f;
458 proj.m[10] = (farP + nearP) / (nearP - farP);
459 proj.m[11] = -1.0;
460 proj.m[14] = (2.0 * farP * nearP) / (nearP - farP);
461 return Result<Mat4d>::success(proj);
462}

524Result<ActionGuidePack> decodeActionGuidePack(const Value& value) {
525 auto unknown = rejectUnknown(value,
526 {"schema", "schemaVersion", "revision", "subject", "skeletonRef", "clipRef",
527 "rootMode", "canvas", "camera", "shots", "actionBinding", "qc"},
528 "$");
529 if (!unknown) return Result<ActionGuidePack>::failure(unknown.status());
530 auto schema = text(value, "schema", "schema");
531 auto version = integer(value, "schemaVersion", "schemaVersion");
532 auto revision = integer(value, "revision", "revision");
533 auto skeletonRef = text(value, "skeletonRef", "skeletonRef");
534 auto clipRef = text(value, "clipRef", "clipRef");
535 auto rootModeText = text(value, "rootMode", "rootMode");
536 if (!schema) return Result<ActionGuidePack>::failure(schema.status());
537 if (!version) return Result<ActionGuidePack>::failure(version.status());
538 if (!revision) return Result<ActionGuidePack>::failure(revision.status());
539 if (!skeletonRef) return Result<ActionGuidePack>::failure(skeletonRef.status());
540 if (!clipRef) return Result<ActionGuidePack>::failure(clipRef.status());
541 if (!rootModeText) return Result<ActionGuidePack>::failure(rootModeText.status());
542 if (schema.value() != kActionGuidePackSchemaId)
543 return invalid<ActionGuidePack>("unexpected guide-pack schema", "schema");
544 if (version.value() != static_cast<std::int64_t>(kActionGuidePackSchemaVersion))
545 return invalid<ActionGuidePack>("unsupported guide-pack schema version", "schemaVersion");
546 if (revision.value() <= 0) return invalid<ActionGuidePack>("revision must be positive", "revision");
547
548 const auto* subjectValue = field(value, "subject");
549 const auto* canvasValue = field(value, "canvas");
550 const auto* cameraValue = field(value, "camera");
551 const auto* shotsValue = field(value, "shots");
552 const auto* bindingValue = field(value, "actionBinding");
553 const auto* qcValue = field(value, "qc");
554 if (!subjectValue) return invalid<ActionGuidePack>("subject is required", "subject");
555 if (!canvasValue) return invalid<ActionGuidePack>("canvas is required", "canvas");
556 if (!cameraValue) return invalid<ActionGuidePack>("camera is required", "camera");
557 if (!shotsValue || !shotsValue->isArray()) return invalid<ActionGuidePack>("shots must be an array", "shots");
558 if (!bindingValue) return invalid<ActionGuidePack>("actionBinding is required", "actionBinding");
559 if (!qcValue) return invalid<ActionGuidePack>("qc is required", "qc");
560
561 auto subject = decodeSubject(*subjectValue);
562 if (!subject) return Result<ActionGuidePack>::failure(subject.status());
563 auto canvas = decodeCanvas(*canvasValue);
564 if (!canvas) return Result<ActionGuidePack>::failure(canvas.status());
565 auto camera = decodeCamera(*cameraValue);
566 if (!camera) return Result<ActionGuidePack>::failure(camera.status());
567 auto binding = decodeBinding(*bindingValue);
568 if (!binding) return Result<ActionGuidePack>::failure(binding.status());
569 auto qc = decodeQc(*qcValue);
570 if (!qc) return Result<ActionGuidePack>::failure(qc.status());
571 auto rootMode = parseRootMode(rootModeText.value());
572 if (!rootMode) return Result<ActionGuidePack>::failure(rootMode.status());
573
574 ActionGuidePack pack;
575 pack.revision = static_cast<std::uint64_t>(revision.value());
576 pack.subject = std::move(subject).takeValue();
577 pack.skeletonRef = std::move(skeletonRef).takeValue();
578 pack.clipRef = std::move(clipRef).takeValue();
579 pack.rootMode = rootMode.value();
580 pack.canvas = std::move(canvas).takeValue();
581 pack.camera = std::move(camera).takeValue();
582 pack.actionBinding = std::move(binding).takeValue();
583 pack.qc = std::move(qc).takeValue();
584 for (std::size_t i = 0; i < shotsValue->getIf<Value::Array>()->size(); ++i) {
585 auto shot = decodeShot(shotsValue->getIf<Value::Array>()->at(i), "shots[" + std::to_string(i) + "]");
586 if (!shot) return Result<ActionGuidePack>::failure(shot.status());
587 pack.shots.push_back(std::move(shot).takeValue());
588 }
589
590 auto valid = validatePack(pack);
591 if (!valid) return Result<ActionGuidePack>::failure(valid.status());
592 return Result<ActionGuidePack>::success(std::move(pack));
593}

595Result<std::vector<ActionGuideLandmark>> projectActionGuideLandmarks(const ActionGuidePack& pack,
596 std::span<const ActionGuideBoneWorld> bones) {
597 auto valid = validatePack(pack);
598 if (!valid) return Result<std::vector<ActionGuideLandmark>>::failure(valid.status());
599 auto view = lookAt(pack.camera);
600 if (!view) return Result<std::vector<ActionGuideLandmark>>::failure(view.status());
601 auto proj = projectionMatrix(pack);
602 if (!proj) return Result<std::vector<ActionGuideLandmark>>::failure(proj.status());
603 const Mat4d viewProj = mul(proj.value(), view.value());
604
605 std::vector<ActionGuideLandmark> landmarks;
606 landmarks.reserve(bones.size());
607 for (std::size_t i = 0; i < bones.size(); ++i) {
608 const auto& bone = bones[i];
609 if (bone.boneId.empty())
610 return invalid<std::vector<ActionGuideLandmark>>("boneId is required",
611 "bones[" + std::to_string(i) + "].boneId");
612 if (!std::isfinite(bone.position[0]) || !std::isfinite(bone.position[1]) || !std::isfinite(bone.position[2]))
613 return invalid<std::vector<ActionGuideLandmark>>("bone position must be finite",
614 "bones[" + std::to_string(i) + "].position");
615 const auto clip = transform(viewProj, bone.position[0], bone.position[1], bone.position[2], 1.0);
616 if (!(std::abs(clip[3]) > 1e-12) || !std::isfinite(clip[3]))
617 return invalid<std::vector<ActionGuideLandmark>>("projected bone has non-finite w",
618 "bones[" + std::to_string(i) + "]");
619 const double invW = 1.0 / clip[3];
620 const double ndcX = clip[0] * invW;
621 const double ndcY = clip[1] * invW;
622 if (!std::isfinite(ndcX) || !std::isfinite(ndcY))
623 return invalid<std::vector<ActionGuideLandmark>>("projected bone has non-finite ndc",
624 "bones[" + std::to_string(i) + "]");
625 ActionGuideLandmark landmark;
626 landmark.boneId = bone.boneId;
627 landmark.name = bone.name.empty() ? bone.boneId : bone.name;
628 landmark.x = (ndcX * 0.5 + 0.5) * static_cast<double>(pack.canvas.width);
629 // Top-left origin: NDC +Y maps to smaller pixel Y.
630 landmark.y = (1.0 - (ndcY * 0.5 + 0.5)) * static_cast<double>(pack.canvas.height);
631 const auto viewPos = transform(view.value(), bone.position[0], bone.position[1], bone.position[2], 1.0);
632 landmark.depth = -viewPos[2];
633 landmark.parentId = bone.parentId;
634 landmarks.push_back(std::move(landmark));
635 }
636 return Result<std::vector<ActionGuideLandmark>>::success(std::move(landmarks));
637}
