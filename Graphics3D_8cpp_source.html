61void Graphics::begin3DFrame() {
62 ensureFileTexturesReady();
63 ASSERT(initialized);
64 if (!initialized) throw Exception("begin3DFrame: graphics not initialized");
65 if (isCanvasActive()) throw Exception("begin3DFrame: cannot start 3D while a Canvas is active");
66 createMesh3DPipeline();
67 if (swapchainPassOpen) {
68 // Previous eve_render opened 3D then threw before present().
69 try {
70 flushToSwapchain();
71 } catch (...) {
72 abortOpen3DFrame();
73 }
74 }
75 // Soft-fail like flushToSwapchain: on Android/iOS the surface may still be
76 // settling after orientation change; throwing would abort the whole script.
77 if (!beginPresentCommandBuffer())
78 return;
79 decalLayerFresh = false;
80 recordDeferredFrameGraph();
81 recordDecalPass();
82 vgAnyThisFrame_ = false;
83 currentFrameArena().reset();
84 ensureGpuDrivenCullResources(gbufferWidth > 0 ? gbufferWidth : int(swapchain.extent.width),
85 gbufferHeight > 0 ? gbufferHeight : int(swapchain.extent.height));
86
87 // 3D pass clears with backgroundColor (setBackgroundColor), not a stale 2D clear.
88 clearColor = backgroundColor;
89 createSceneColorResources(int(swapchain.extent.width), int(swapchain.extent.height));
90 // GPU-driven cull defers the scene color pass so the compute section
91 // (HZB build + cull + emit + VG cull) can run before the opaque draws;
92 // gpuDrivenOpenScenePass() opens it when the renderer reaches the forward
93 // section.
94 gpuDrivenScenePassPending_ = gpuDrivenEnabled_ && gpuDrivenCullReady_;
95 if (!gpuDrivenScenePassPending_) {
96 if (beginSceneColorRenderPass()) {
97 // Rebuild scene-pass pipelines to match the active scene pass (MSAA).
98 // Must happen AFTER the pass opens: if the MSAA scene pass is unavailable
99 // we fall back to the swapchain (e1) render pass below, and rasterizing
100 // with an Nx pipeline into an e1 pass is UB that can hang the GPU (TDR).
101 ensureScenePassPipelines(activeScenePass(), activeSceneSamples());
102 } else {
103 // Scene pass unavailable — draw 3D straight into the swapchain.
104 ensureScenePassPipelines(renderpass, vk::SampleCountFlagBits::e1);
105 beginSwapchainColorPass();
106 }
107 }
108
109 auto &cb = currentPresentCb();
110 setViewportAndScissor(cb, swapchain.extent.width, swapchain.extent.height);
111
112 {
113 auto &fslots = currentMesh3dFrameSlots();
114 fslots.lastDrawCount = fslots.drawIndex;
115 fslots.drawIndex = 0;
116 ensureMesh3dRing(fslots);
117 auto &cslots = currentMesh3dClusteredFrameSlots();
118 cslots.lastDrawCount = cslots.drawIndex;
119 cslots.drawIndex = 0;
120 ensureMesh3dClusteredRing(cslots);
121 }
122 lastMesh3dPipeline = nullptr;
123 lastMesh3dClusteredPipeline = nullptr;
124 currentVoxelInstanceFrame().drawIndex = 0;
125 swapchainPassOpen = !gpuDrivenScenePassPending_;
126 frameHad3D = true;
127 hasPendingClear = false;
128}

130void Graphics::drawPrimitiveScene(const PrimitiveSceneCanvas3D &canvas) {
131 ASSERT(initialized);
132 if (!initialized) throw Exception("drawPrimitiveScene: graphics not initialized");
133 const bool offscreen = offscreen3DPassOpen;
134 if (!offscreen && gpuDrivenScenePassPending_) gpuDrivenOpenScenePass();
135 if (!offscreen && !swapchainPassOpen) throw Exception("drawPrimitiveScene: call begin3DFrame first");
136
137 const ResolvedPrimitiveTriangles resolved = resolvePrimitiveStrokes3D(canvas);
138 if (resolved.vertices.empty()) return;
139
140 struct DrawGroup {
141 ScenePrimitivePaint paint;
142 std::size_t sequence = 0;
143 float averageDepth = 0.f;
144 std::vector<Primitive3DVertex> vertices;
145 };
146 std::vector<DrawGroup> groups;
147 groups.reserve(resolved.batches3D.size());
148 for (const ResolvedPrimitiveBatch3D &batch : resolved.batches3D) {
149 DrawGroup group;
150 group.paint = batch.paint;
151 group.sequence = batch.sequence;
152 group.averageDepth = batch.averageDepth;
153 group.vertices.reserve(batch.vertexCount);
154 for (std::size_t vertex = batch.firstVertex; vertex < batch.firstVertex + batch.vertexCount; ++vertex)
155 group.vertices.push_back({resolved.vertices[vertex].clipPosition, resolved.vertices[vertex].color});
156 groups.push_back(std::move(group));
157 }
158 std::stable_sort(groups.begin(), groups.end(), [&](const DrawGroup &a, const DrawGroup &b) {
159 const ScenePrimitivePaint &paintA = a.paint;
160 const ScenePrimitivePaint &paintB = b.paint;
161 const bool ignoreA = paintA.depth == PrimitiveDepthMode::Ignore;
162 const bool ignoreB = paintB.depth == PrimitiveDepthMode::Ignore;
163 if (ignoreA != ignoreB) return !ignoreA;
164 if (paintA.layer != paintB.layer) return paintA.layer < paintB.layer;
165 const bool transparentA = paintA.blend != BlendMode::Opaque || paintA.color.a < 1.f;
166 const bool transparentB = paintB.blend != BlendMode::Opaque || paintB.color.a < 1.f;
167 if (transparentA != transparentB) return !transparentA;
168 if (transparentA && a.averageDepth != b.averageDepth) return a.averageDepth > b.averageDepth;
169 return a.sequence < b.sequence;
170 });
171
172 auto &buffers = offscreen ? offscreenPrimitive3DBufs : currentFrame2DBuffers().primitive3DBufs;
173 if (buffers.size() < groups.size()) buffers.resize(groups.size());
174 const vk::CommandBuffer cb = offscreen ? offscreen3DCB : currentPresentCb();
175 const uint32_t targetWidth =
176 offscreen ? static_cast<uint32_t>(offscreen3DCanvas->getWidth()) : swapchain.extent.width;
177 const uint32_t targetHeight =
178 offscreen ? static_cast<uint32_t>(offscreen3DCanvas->getHeight()) : swapchain.extent.height;
179 auto &pipelines = offscreen
180 ? (offscreen3DHDRActive ? hdrOffscreenPrimitive3DPipelines : offscreenPrimitive3DPipelines)
181 : primitive3DPipelines;
182 setViewportAndScissor(cb, targetWidth, targetHeight);
183 for (std::size_t groupIndex = 0; groupIndex < groups.size(); ++groupIndex) {
184 const DrawGroup &group = groups[groupIndex];
185 const ScenePrimitivePaint &paint = group.paint;
186 const std::size_t pipelineIndex = primitive3DPipelineIndex(paint.depth, paint.blend, paint.cull);
187 const vk::Pipeline primitivePipeline = pipelines[pipelineIndex];
188 if (!primitivePipeline) throw Exception("drawPrimitiveScene: primitive pipeline unavailable");
189 buffers[groupIndex].allocate<Primitive3DVertex>(frameToken(), device, group.vertices);
190 const vk::DeviceSize offset = 0;
191 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, primitivePipeline);
192 cb.bindVertexBuffers(0, 1, buffers[groupIndex], &offset);
193 cb.draw(static_cast<uint32_t>(group.vertices.size()), 1, 0, 0);
194 }
195}

197void Graphics::ensureOffscreen3DResources() {
198 auto &dev = getDevice();
199 auto createResources = [&](vk::Format colorFormat, vkb::BuiltRenderPass &renderPass, vk::Pipeline &meshPipeline,
200 std::array<vk::Pipeline, kMesh3DPipelineVariants> &surfacePipelines,
201 std::array<vk::Pipeline, kPrimitive3DPipelineVariants> &primitivePipelines) {
202 if (renderPass) return;
203 renderPass =
204 dev.createRenderPass()
205 .addSampledColorAttachment(colorFormat)
206 .addDepthAttachment(depthFormat, vk::AttachmentLoadOp::eClear,
207 vk::AttachmentStoreOp::eDontCare)
208 .addSubpass(vkb::SubpassBuilder()
209 .addAttachmentRef(0, vk::ImageLayout::eColorAttachmentOptimal)
210 .setDepthStencilAttachment(
211 1, vk::ImageLayout::eDepthStencilAttachmentOptimal))
212 .addExternalShaderReadDependencies()
213 .build();
214 meshPipeline = createMesh3DStylePipeline(
215 embeddedSpirv(mesh3d_vert_spv), embeddedSpirv(mesh3d_frag_spv), mesh3dPipelineLayout,
216 renderPass, vk::SampleCountFlagBits::e1);
217 for (int blendValue = 0; blendValue < 5; ++blendValue) {
218 const auto blend = BlendMode(blendValue);
219 for (int depthValue = 0; depthValue < 2; ++depthValue) {
220 for (int doubleValue = 0; doubleValue < 2; ++doubleValue) {
221 const size_t index =
222 mesh3dPipelineIndex(blend, depthValue != 0, doubleValue != 0);
223 surfacePipelines[index] = createMesh3DStylePipeline(
224 embeddedSpirv(mesh3d_vert_spv), embeddedSpirv(mesh3d_frag_spv),
225 mesh3dPipelineLayout, renderPass, vk::SampleCountFlagBits::e1,
226 blend, depthValue != 0, doubleValue != 0);
227 }
228 }
229 }
230 buildPrimitive3DPipelines(renderPass, vk::SampleCountFlagBits::e1, primitivePipelines);
231 };
232 createResources(vk::Format::eR8G8B8A8Unorm, offscreen3DRenderPass, offscreen3DMeshPipeline,
233 offscreen3DSurfacePipelines, offscreenPrimitive3DPipelines);
234 createResources(vk::Format::eR16G16B16A16Sfloat, hdrOffscreen3DRenderPass, hdrOffscreen3DMeshPipeline,
235 hdrOffscreen3DSurfacePipelines, hdrOffscreenPrimitive3DPipelines);
236 if (!offscreen3DPool) {
237 vk::CommandPoolCreateInfo poolInfo{};
238 poolInfo.flags = vk::CommandPoolCreateFlagBits::eTransient;
239 offscreen3DPool = dev->createCommandPool(poolInfo);
240 vk::CommandBufferAllocateInfo allocInfo{};
241 allocInfo.commandPool = offscreen3DPool;
242 allocInfo.level = vk::CommandBufferLevel::ePrimary;
243 allocInfo.commandBufferCount = 1;
244 auto bufs = dev->allocateCommandBuffers(allocInfo);
245 offscreen3DCB = bufs[0];
246 vk::FenceCreateInfo fenceInfo{};
247 fenceInfo.flags = vk::FenceCreateFlagBits::eSignaled; // first wait passes immediately
248 offscreen3DFence = dev->createFence(fenceInfo);
249 const auto &limits = dev.physical_device.properties.limits;
250 if (limits.timestampComputeAndGraphics && limits.timestampPeriod > 0.f) {
251 vk::QueryPoolCreateInfo queryInfo{};
252 queryInfo.queryType = vk::QueryType::eTimestamp;
253 queryInfo.queryCount = 2;
254 offscreen3DTimestampQueryPool = dev->createQueryPool(queryInfo);
255 offscreen3DTimestampPeriodNs = limits.timestampPeriod;
256 }
257 }

376void Graphics::end3DFrameToCanvas() {
377 if (!offscreen3DPassOpen || !offscreen3DCB) return;
378 offscreen3DCB.endRenderPass();
379 if (offscreen3DCanvas) {
380 offscreen3DCanvas->colorImage().setLayout(offscreen3DCB,
381 vk::ImageLayout::eShaderReadOnlyOptimal);
382 }
383 if (offscreen3DTimestampQueryPool) {
384 offscreen3DCB.writeTimestamp(vk::PipelineStageFlagBits::eBottomOfPipe,
385 offscreen3DTimestampQueryPool, 1);
386 }
387 offscreen3DCB.end();
388
389 // Submit the offscreen render directly to the graphics queue (no present),
390 // then wait for it so it is fully drained before any subsequent render or
391 // teardown.
392 vk::SubmitInfo submitInfo{};
393 submitInfo.commandBufferCount = 1;
394 submitInfo.pCommandBuffers = &offscreen3DCB;
395 if (offscreen3DFence) (void)device->resetFences(1, &offscreen3DFence);
396 (void)device.getQueue(vkb::QueueType::graphics).submit(1, &submitInfo, offscreen3DFence);
397 if (offscreen3DFence)
398 (void)device->waitForFences(1, &offscreen3DFence, VK_TRUE, UINT64_MAX);
399 if (offscreen3DTimestampQueryPool && offscreen3DTimestampPeriodNs > 0.f) {
400 std::array<uint64_t, 2> ticks{};
401 const vk::Result queryResult = device->getQueryPoolResults(
402 offscreen3DTimestampQueryPool, 0, uint32_t(ticks.size()), sizeof(ticks), ticks.data(),
403 sizeof(uint64_t), vk::QueryResultFlagBits::e64 | vk::QueryResultFlagBits::eWait);
404 if (queryResult == vk::Result::eSuccess && ticks[1] >= ticks[0]) {
405 const double nanoseconds =
406 double(ticks[1] - ticks[0]) * double(offscreen3DTimestampPeriodNs);
407 lastOffscreen3DGpuDurationMs = float(nanoseconds * 1.0e-6);
408 }
409 }
410
411 offscreen3DPassOpen = false;
412 offscreen3DHDRActive = false;
413 offscreen3DCanvas = nullptr;
414}

477vk::Pipeline Graphics::buildVoxelRectPipeline(const vkb::BuiltRenderPass &rp,
478 vk::SampleCountFlagBits samples) {
479 auto vert = embeddedSpirv(voxel_rect_vert_spv);
480 auto frag = embeddedSpirv(voxel_rect_frag_spv);
481 ShaderModulePair modules(device, vert, frag);
482
483 vk::VertexInputBindingDescription bindings[3]{};
484 bindings[0].binding = 0;
485 bindings[0].stride = sizeof(glm::vec2);
486 bindings[0].inputRate = vk::VertexInputRate::eVertex;
487 bindings[1].binding = 1;
488 bindings[1].stride = sizeof(uint32_t);
489 bindings[1].inputRate = vk::VertexInputRate::eInstance;
490 bindings[2].binding = 2;
491 bindings[2].stride = sizeof(uint32_t);
492 bindings[2].inputRate = vk::VertexInputRate::eInstance;
493
494 vk::VertexInputAttributeDescription attrs[3]{};
495 attrs[0].location = 0;
496 attrs[0].binding = 0;
497 attrs[0].format = vk::Format::eR32G32Sfloat;
498 attrs[0].offset = 0;
499 attrs[1].location = 1;
500 attrs[1].binding = 1;
501 attrs[1].format = vk::Format::eR32Uint;
502 attrs[1].offset = 0;
503 attrs[2].location = 2;
504 attrs[2].binding = 2;
505 attrs[2].format = vk::Format::eR32Uint;
506 attrs[2].offset = 0;
507
508 vk::PipelineVertexInputStateCreateInfo vi{};
509 vi.vertexBindingDescriptionCount = 3;
510 vi.pVertexBindingDescriptions = bindings;
511 vi.vertexAttributeDescriptionCount = 3;
512 vi.pVertexAttributeDescriptions = attrs;
513
514 // Unit-quad indices 0-2-1 / 0-3-2 are object-space CCW for outward faces.
515 // perspectiveVulkanRH_ZO flips clip Y, so those faces are CCW in the
516 // framebuffer — CounterClockwise frontFace keeps them. Clockwise + Back
517 // (the mesh-pipeline convention) culls every submitted face after the Y
518 // flip; CPU 6-dir cull already dropped the true back faces, so the frame
519 // collapses to a 1px silhouette.
520 return device.createPipeline()
521 .useClassicPipeline(modules.vert, modules.frag)

570void Graphics::drawVoxelFaceInstances(const uint32_t *packed, int count, float originX,
571 float originY, float originZ, const std::string &faceDir,
572 Texture *atlas, int tilesPerRow, const uint32_t *ao) {
573 ASSERT(initialized);
574 if (!initialized) throw Exception("drawVoxelFaceInstances: graphics not initialized");
575 if (!swapchainPassOpen) throw Exception("drawVoxelFaceInstances: call begin3DFrame first");
576 if (!voxelRectPipeline) createVoxelRectPipeline();
577 if (!packed || count <= 0) return;
578
579 int face = -1;
580 if (faceDir == "posX" || faceDir == "+x")
581 face = 0;
582 else if (faceDir == "negX" || faceDir == "-x")
583 face = 1;
584 else if (faceDir == "posY" || faceDir == "+y")
585 face = 2;
586 else if (faceDir == "negY" || faceDir == "-y")
587 face = 3;
588 else if (faceDir == "posZ" || faceDir == "+z")
589 face = 4;
590 else if (faceDir == "negZ" || faceDir == "-z")
591 face = 5;
592 else
593 throw Exception("drawVoxelFaceInstances: unknown faceDir '%s'", faceDir.c_str());
594
595 Texture *tex = atlas ? atlas : whiteTexture;
596 if (!tex || !tex->gpuHandle) throw Exception("drawVoxelFaceInstances: missing texture");
597 auto *gpuTex = static_cast<GpuTexture *>(tex->gpuHandle);
598
599 ensureVoxelUnitQuad();
600
601 const vk::DeviceSize bytes = vk::DeviceSize(count) * sizeof(uint32_t);
602 auto &vframe = currentVoxelInstanceFrame();
603 const size_t slotIndex = vframe.drawIndex++;
604 while (vframe.slots.size() <= slotIndex) vframe.slots.emplace_back();
605 auto &slot = vframe.slots[slotIndex];
606 if (slot.capacityBytes < size_t(bytes) || !slot.buffer.buffer) {
607 const size_t alloc = std::max(size_t(bytes), slot.capacityBytes ? slot.capacityBytes * 2 : size_t(bytes));
608 slot.buffer.allocate(frameToken(), device, vk::BufferUsageFlagBits::eVertexBuffer, vk::DeviceSize(alloc),
609 vk::MemoryPropertyFlagBits::eHostVisible |
610 vk::MemoryPropertyFlagBits::eHostCoherent);
611 slot.capacityBytes = alloc;
612 }
613 slot.buffer.updateLocal(frameToken(), packed, size_t(bytes));
614
615 // Ambient-occlusion words (null → full bright). Kept parallel to packed.
616 const uint32_t defaultAO = 0xFFu; // all four corners AO=3
617 std::vector<uint32_t> aoDefaults;
618 if (!ao) {
619 aoDefaults.assign(size_t(count), defaultAO);
620 ao = aoDefaults.data();
621 }
622 const vk::DeviceSize aoBytes = vk::DeviceSize(count) * sizeof(uint32_t);
623 if (slot.aoCapacityBytes < size_t(aoBytes) || !slot.aoBuffer.buffer) {
624 const size_t alloc = std::max(size_t(aoBytes),
625 slot.aoCapacityBytes ? slot.aoCapacityBytes * 2
626 : size_t(aoBytes));
627 slot.aoBuffer.allocate(frameToken(), device, vk::BufferUsageFlagBits::eVertexBuffer,
628 vk::DeviceSize(alloc),
629 vk::MemoryPropertyFlagBits::eHostVisible |
630 vk::MemoryPropertyFlagBits::eHostCoherent);
631 slot.aoCapacityBytes = alloc;
632 }
633 slot.aoBuffer.updateLocal(frameToken(), ao, size_t(aoBytes));
634
635 VoxelRectPC pc{};
636 pc.viewProj = mesh3dFrameUbo.mvp;
637 pc.chunkOrigin = glm::vec4(originX, originY, originZ, float(face));
638 pc.atlasInfo = glm::vec4(float(std::max(1, tilesPerRow)), 0.f, 0.f, 0.f);
639 pc.tint = mesh3dFrameUbo.tint;
640
641 auto &cb = currentPresentCb();
642 vk::DescriptorSet set = voxelRectSetFor(gpuTex);
643 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, voxelRectPipeline);
644 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, voxelRectPipelineLayout, 0, 1, &set, 0,
645 nullptr);
646 cb.pushConstants(voxelRectPipelineLayout, vk::ShaderStageFlagBits::eVertex, 0, sizeof(VoxelRectPC),
647 &pc);
648
649 vk::Buffer vbufs[3] = {voxelUnitQuadVerts.buffer, slot.buffer.buffer, slot.aoBuffer.buffer};
650 vk::DeviceSize offsets[3] = {0, 0, 0};
651 cb.bindVertexBuffers(0, 3, vbufs, offsets);
652 cb.bindIndexBuffer(voxelUnitQuadIndices.buffer, 0, vk::IndexType::eUint32);
653 cb.drawIndexed(6, uint32_t(count), 0, 0, 0);
654}

764bool Graphics::prepareSkinPass(Mesh *mesh, Texture *albedo, const glm::mat4 &mvp,
765 const glm::mat4 &model, const glm::vec4 &clip,
766 vk::DescriptorSet &set, uint32_t &uboOffset) {
767 if (!mesh || !mesh->hasGpuSkinning()) return false;
768 auto &fslots = currentMesh3dFrameSlots();
769 // A 3D frame can inherit an already-open present command buffer (for
770 // example, when script-side 2D clear work precedes render3D). In that
771 // path begin3DFrame may not have initialized this slot yet. Allocate the
772 // empty slot lazily before the first skinned draw; no recorded command can
773 // reference it while capacity is zero.
774 if (!fslots.uboRing.buffer || fslots.capacity == 0) ensureMesh3dRing(fslots);
775 if (fslots.drawIndex >= fslots.capacity) {
776 std::fprintf(stderr, "[vulkan] skin pass UBO ring exhausted (%zu draws); draw skipped\n",
777 fslots.capacity);
778 return false;
779 }
780 SkinPassUBO ubo;
781 ubo.mvp = mvp;
782 ubo.model = model;
783 ubo.clip = clip;
784 const int paletteCount = std::min(mesh->getSkinPaletteCount(), Mesh::kMaxSkinBones);
785 ubo.skinInfo.x = static_cast<float>(paletteCount);
786 const auto &palette = mesh->skinPalette();
787 for (int i = 0; i < paletteCount; ++i) {
788 const float *matrix = palette.data() + static_cast<size_t>(i) * 16u;
789 for (int column = 0; column < 4; ++column)
790 for (int row = 0; row < 4; ++row)
791 ubo.skinBones[i][column][row] = matrix[column * 4 + row];
792 }
793 const size_t slot = fslots.drawIndex++;
794 ensureMesh3dStrides();
795 uboOffset = uint32_t(slot) * mesh3dUboStride;
796 updateRingLocal(fslots.uboRing, uboOffset, &ubo, sizeof(ubo));
797 Texture *texture = albedo ? albedo : whiteTexture;
798 if (!texture || !texture->gpuHandle) return false;
799 set = skinPassSetFor(static_cast<GpuTexture *>(texture->gpuHandle), fslots);
800 return bool(set);
801}

865void Graphics::drawMeshGBuffer(Mesh *mesh, const glm::mat4 &mvp, const glm::mat4 &model, float nearZ,
866 float farZ, Texture *albedo, float tintR, float tintG, float tintB,
867 float motionX, float motionY, float roughness, float metallic) {
868 if (!gbufferPassActive) throw Exception("drawMeshGBuffer: call beginGBufferPass first");
869 if (!mesh || !mesh->gpuHandle) throw Exception("drawMeshGBuffer: null mesh");
870 GBufferDraw d{};
871 d.mesh = mesh;
872 d.albedo = albedo;
873 d.push.mvp = mvp;
874 d.push.modelR0 = glm::vec4(model[0][0], model[1][0], model[2][0], model[3][0]);
875 d.push.modelR1 = glm::vec4(model[0][1], model[1][1], model[2][1], model[3][1]);
876 d.push.modelR2 = glm::vec4(model[0][2], model[1][2], model[2][2], model[3][2]);
877 auto u6 = [](float x) -> uint32_t {
878 return uint32_t(std::lround(std::clamp(x, 0.f, 1.f) * 63.f));
879 };
880 // A float exactly represents every integer through 24 bits. Numeric packing avoids
881 // NaN canonicalization and subnormal flush-to-zero corrupting bit-cast payloads.
882 const uint32_t rough3 = uint32_t(std::lround(std::clamp(roughness, 0.f, 1.f) * 7.f));
883 const uint32_t metal3 = uint32_t(std::lround(std::clamp(metallic, 0.f, 1.f) * 7.f));
884 const uint32_t packedTint = u6(tintR) | (u6(tintG) << 6) | (u6(tintB) << 12) |
885 (rough3 << 18) | (metal3 << 21);
886 auto motion12 = [](float value) {
887 return uint32_t(std::lround(std::clamp(value, -1.f, 1.f) * 2047.f)) + 2047u;
888 };
889 const uint32_t packedMotion = motion12(motionX) | (motion12(motionY) << 12);
890 d.push.clip = glm::vec4(nearZ, farZ, float(packedTint), float(packedMotion));
891 prepareSkinPass(mesh, albedo, mvp, model, d.push.clip, d.skinSet, d.skinUboOffset);
892 gbufferPassDraws.push_back(d);
893}

1290void Graphics::buildDeferredFrameGraphs() {
1291 // One FrameGraph per in-flight slot imports the engine-owned targets and
1292 // owns the deferred passes: the 3 CSM cascades (per-layer views of the
1293 // shadow array) + the G-buffer fill share one dependency-free layer, so the
1294 // JobSystem executor records all four command buffers concurrently (see
1295 // recordDeferredFrameGraph). The engine keeps image ownership so
1296 // renderEntityIdMask / readGBufferToImageData and the postFX wrappers are
1297 // unaffected. Each graph is only used on its slot's frames, so its command
1298 // buffer is reused two frames later — by then the present slot fence
1299 // guarantees the previous graph submit completed (same queue, submitted
1300 // before the present command buffer).
1301 const vk::Format depthFmt = vk::Format::eD32Sfloat;
1302 const vk::Format colorFmt = pickGBufferColorFormat(device);
1303 const uint32_t mapSize = uint32_t(ShadowConfig::kMapSize);
1304 const uint32_t shadowLayers = uint32_t(ShadowConfig::kCascades);
1305 const uint32_t w = gbufferWidth > 0 ? uint32_t(gbufferWidth) : 1u;
1306 const uint32_t h = gbufferHeight > 0 ? uint32_t(gbufferHeight) : 1u;
1307
1308 for (size_t i = 0; i < deferredFrameGraphs_.size(); ++i) {
1309 auto graph = std::make_unique<vkb::FrameGraph>(&device, 1);
1310
1311 vkb::TextureDesc shadowDesc;
1312 shadowDesc.format = depthFmt;
1313 shadowDesc.extent = vk::Extent3D{mapSize, mapSize, 1};
1314 shadowDesc.arrayLayers = shadowLayers;
1315 shadowDesc.aspect = vk::ImageAspectFlagBits::eDepth;
1316 shadowDesc.usage = vk::ImageUsageFlagBits::eSampled |
1317 vk::ImageUsageFlagBits::eDepthStencilAttachment;
1318 shadowDesc.afterLayout = vk::ImageLayout::eDepthStencilReadOnlyOptimal;
1319 vk::ClearValue shadowClear{};
1320 shadowClear.depthStencil = vk::ClearDepthStencilValue{1.0f, 0};
1321 const bool haveShadowSlot =
1322 i < shadowMaps.size() && shadowMaps[i].image.layerCount() >= shadowLayers;
1323 if (haveShadowSlot) {
1324 const vk::Image shadowImage = shadowMaps[i].image.image();
1325 for (uint32_t c = 0; c < shadowLayers; ++c) {
1326 auto shadowH = graph->importTexture("shadowCascade" + std::to_string(c),
1327 shadowImage, shadowMaps[i].image.layerView(c),
1328 shadowDesc);
1329 graph->addPass("shadow" + std::to_string(c))
1330 .depthAttachment(shadowH, vkb::AttachmentOp::clear(shadowClear))
1331 .record([this, c](vkb::FrameGraphPassContext &ctx) {
1332 recordShadowCascadePass(ctx, int(c));
1333 });
1334 }
1335 }
1336
1337 if (i < gbufferSlots.size() && gbufferWidth > 0 && gbufferHeight > 0) {
1338 auto &slot = gbufferSlots[i];
1339 vkb::TextureDesc colorDesc;
1340 colorDesc.format = colorFmt;
1341 colorDesc.extent = vk::Extent3D{w, h, 1};
1342 colorDesc.usage = vk::ImageUsageFlagBits::eSampled |
1343 vk::ImageUsageFlagBits::eColorAttachment |
1344 vk::ImageUsageFlagBits::eTransferSrc;
1345 colorDesc.afterLayout = vk::ImageLayout::eShaderReadOnlyOptimal;
1346 auto normalH = graph->importTexture("gbNormal", slot.normal.image(),
1347 slot.normal.imageView(), colorDesc);
1348 auto depthColorH = graph->importTexture("gbDepthColor", slot.depthColor.image(),
1349 slot.depthColor.imageView(), colorDesc);
1350 auto albedoH = graph->importTexture("gbAlbedo", slot.albedo.image(),
1351 slot.albedo.imageView(), colorDesc);
1352
1353 vkb::TextureDesc depthDesc;
1354 depthDesc.format = depthFmt;
1355 depthDesc.extent = vk::Extent3D{w, h, 1};
1356 depthDesc.aspect = vk::ImageAspectFlagBits::eDepth;
1357 depthDesc.usage = vk::ImageUsageFlagBits::eSampled |
1358 vk::ImageUsageFlagBits::eDepthStencilAttachment;
1359 depthDesc.afterLayout = vk::ImageLayout::eShaderReadOnlyOptimal;
1360 auto depthH = graph->importTexture("gbHwDepth", slot.depth.image(),
1361 slot.depth.imageView(), depthDesc);
1362
1363 std::array<vk::ClearValue, 4> clears{};
1364 clears[0].color = vk::ClearColorValue(std::array<float, 4>{0, 0, 0, 0});
1365 clears[1].color = vk::ClearColorValue(std::array<float, 4>{1, 1, 1, 1});
1366 clears[2].color = vk::ClearColorValue(std::array<float, 4>{0, 0, 0, 0});
1367 clears[3].depthStencil = vk::ClearDepthStencilValue{1.0f, 0};
1368 graph->addPass("gbuffer")
1369 .colorAttachment(normalH, vkb::AttachmentOp::clear(clears[0]))
1370 .colorAttachment(depthColorH, vkb::AttachmentOp::clear(clears[1]))
1371 .colorAttachment(albedoH, vkb::AttachmentOp::clear(clears[2]))
1372 .depthAttachment(depthH, vkb::AttachmentOp::clear(clears[3]))
1373 .record([this](vkb::FrameGraphPassContext &ctx) { recordGBufferPassDraws(ctx); });
1374 }
1375 graph->compile();
1376 deferredFrameGraphs_[i] = std::move(graph);
1377 }
1378}

1380void Graphics::recordShadowCascadePass(vkb::FrameGraphPassContext &ctx, int cascade) {
1381 // Runs inside the FrameGraph's "shadow<cascade>" render-pass instance
1382 // (already begun with a depth clear); only draw commands go here. The pass
1383 // may be recorded on a JobSystem worker, so everything below must be
1384 // read-only: shadowCascadeDraws was captured by endShadowPass on the main
1385 // thread before the graph records.
1386 auto &cb = ctx.commandBuffer();
1387 const vk::Extent2D extent = ctx.extent();
1388 const uint32_t size = extent.width ? extent.width : uint32_t(ShadowConfig::kMapSize);
1389 setViewportAndScissor(cb, size, size);
1390 vk::Pipeline boundPipeline{};
1391 for (const auto &d : shadowCascadeDraws[cascade]) {
1392 if (!d.mesh || !d.mesh->gpuHandle) continue;
1393 const bool wantAlpha = d.alphaTest && shadowAlphaPipeline;
1394 const bool skinned = d.skinSet && d.mesh->hasGpuSkinning();
1395 vk::Pipeline wanted = skinned ? (wantAlpha ? shadowSkinAlphaPipeline : shadowSkinPipeline)
1396 : (wantAlpha ? shadowAlphaPipeline : shadowPipeline);
1397 if (wanted != boundPipeline) {
1398 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, wanted);
1399 boundPipeline = wanted;
1400 }
1401 auto *gpuMesh = static_cast<GpuMesh *>(d.mesh->gpuHandle);
1402 if (skinned) {
1403 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, skinPassPipelineLayout, 0, 1,
1404 &d.skinSet, 1, &d.skinUboOffset);
1405 } else if (wantAlpha) {
1406 Texture *alb = d.albedo ? d.albedo : whiteTexture;
1407 if (alb && alb->gpuHandle && texSetLayout) {
1408 auto *gpuTex = static_cast<GpuTexture *>(alb->gpuHandle);
1409 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics,
1410 shadowAlphaPipelineLayout, 0, 1,
1411 gpuTex->descriptorSet.ptr(), 0, nullptr);
1412 }
1413 }
1414 if (!skinned)
1415 cb.pushConstants(wantAlpha ? shadowAlphaPipelineLayout : shadowPipelineLayout,
1416 vk::ShaderStageFlagBits::eVertex, 0, sizeof(glm::mat4), &d.mvp);
1417 drawIndexedMesh(cb, *gpuMesh);
1418 }
1419}

1423 // begun with the planned clear values); only draw commands go here. The

1429 const uint32_t h = extent.height ? extent.height : uint32_t(gbufferHeight);

1437 : (wantAlpha ? gbufferAlphaPipeline : gbufferPipeline);
1438 if (wanted != boundPipeline) {
1439 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, wanted);
1440 boundPipeline = wanted;
1441 }
1442 auto *gpuMesh = static_cast<GpuMesh *>(d.mesh->gpuHandle);
1443 Texture *alb = d.albedo ? d.albedo : whiteTexture;
1444 if (skinned) {
1445 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, skinPassPipelineLayout, 0, 1,

1447 } else if (alb && alb->gpuHandle && texSetLayout) {

1449 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, gbufferPipelineLayout, 0, 1,

1452 if (!skinned)
1453 cb.pushConstants(gbufferPipelineLayout,
1454 vk::ShaderStageFlagBits::eVertex | vk::ShaderStageFlagBits::eFragment,
1455 0, sizeof(GBufferPush), &d.push);

1461 const size_t slot = currentFrameSlot();
