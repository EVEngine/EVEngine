61void Graphics::begin3DFrame() {
62 ensureFileTexturesReady();
63 ASSERT(initialized);
64 if (!initialized) throw Exception("begin3DFrame: graphics not initialized");
65 if (isCanvasActive()) throw Exception("begin3DFrame: cannot start 3D while a Canvas is active");
66 createMesh3DPipeline();
67 if (swapchainPassOpen) {
68 // Previous eve_render opened 3D then threw before present().
69 try {
70 flushToSwapchain();
71 } catch (...) {
72 abortOpen3DFrame();
73 }
74 }
75 // Soft-fail like flushToSwapchain: on Android/iOS the surface may still be
76 // settling after orientation change; throwing would abort the whole script.
77 if (!beginPresentCommandBuffer())
78 return;
79 currentFrame2DBuffers().primitive3DDrawIndex = 0;
80 decalLayerFresh = false;
81 recordDeferredFrameGraph();
82 recordDecalPass();
83 vgAnyThisFrame_ = false;
84 currentFrameArena().reset();
85 ensureGpuDrivenCullResources(gbufferWidth > 0 ? gbufferWidth : int(swapchain.extent.width),
86 gbufferHeight > 0 ? gbufferHeight : int(swapchain.extent.height));
87
88 // 3D pass clears with backgroundColor (setBackgroundColor), not a stale 2D clear.
89 clearColor = backgroundColor;
90 createSceneColorResources(int(swapchain.extent.width), int(swapchain.extent.height));
91 // GPU-driven cull defers the scene color pass so the compute section
92 // (HZB build + cull + emit + VG cull) can run before the opaque draws;
93 // gpuDrivenOpenScenePass() opens it when the renderer reaches the forward
94 // section.
95 gpuDrivenScenePassPending_ = gpuDrivenEnabled_ && gpuDrivenCullReady_;
96 if (!gpuDrivenScenePassPending_) {
97 if (beginSceneColorRenderPass()) {
98 // Rebuild scene-pass pipelines to match the active scene pass (MSAA).
99 // Must happen AFTER the pass opens: if the MSAA scene pass is unavailable
100 // we fall back to the swapchain (e1) render pass below, and rasterizing
101 // with an Nx pipeline into an e1 pass is UB that can hang the GPU (TDR).
102 ensureScenePassPipelines(activeScenePass(), activeSceneSamples());
103 } else {
104 // Scene pass unavailable — draw 3D straight into the swapchain.
105 ensureScenePassPipelines(renderpass, vk::SampleCountFlagBits::e1);
106 beginSwapchainColorPass();
107 }
108 }
109
110 auto &cb = currentPresentCb();
111 setViewportAndScissor(cb, swapchain.extent.width, swapchain.extent.height);
112
113 {
114 auto &fslots = currentMesh3dFrameSlots();
115 fslots.lastDrawCount = fslots.drawIndex;
116 fslots.drawIndex = 0;
117 fslots.activePalette = 0;
118 ensureMesh3dRing(fslots);
119 auto &cslots = currentMesh3dClusteredFrameSlots();
120 cslots.lastDrawCount = cslots.drawIndex;
121 cslots.drawIndex = 0;
122 ensureMesh3dClusteredRing(cslots);
123 }
124 lastMesh3dPipeline = nullptr;
125 lastMesh3dClusteredPipeline = nullptr;
126 currentVoxelInstanceFrame().drawIndex = 0;
127 swapchainPassOpen = !gpuDrivenScenePassPending_;
128 frameHad3D = true;
129 hasPendingClear = false;
130}

133void Graphics::ensureOffscreen3DResources() {
134 auto &dev = getDevice();
135 auto createResources = [&](vk::Format colorFormat, vkb::BuiltRenderPass &renderPass, vk::Pipeline &meshPipeline,
136 std::array<vk::Pipeline, kMesh3DPipelineVariants> &surfacePipelines,
137 std::array<vk::Pipeline, kPrimitive3DPipelineVariants> &primitivePipelines) {
138 if (renderPass) return;
139 renderPass =
140 dev.createRenderPass()
141 .addSampledColorAttachment(colorFormat)
142 .addDepthAttachment(depthFormat, vk::AttachmentLoadOp::eClear,
143 vk::AttachmentStoreOp::eDontCare)
144 .addSubpass(vkb::SubpassBuilder()
145 .addAttachmentRef(0, vk::ImageLayout::eColorAttachmentOptimal)
146 .setDepthStencilAttachment(
147 1, vk::ImageLayout::eDepthStencilAttachmentOptimal))
148 .addExternalShaderReadDependencies()
149 .build();
150 meshPipeline = createMesh3DStylePipeline(
151 embeddedSpirv(mesh3d_vert_spv), embeddedSpirv(mesh3d_frag_spv), mesh3dPipelineLayout,
152 renderPass, vk::SampleCountFlagBits::e1);
153 for (int blendValue = 0; blendValue < 5; ++blendValue) {
154 const auto blend = BlendMode(blendValue);
155 for (int depthValue = 0; depthValue < 2; ++depthValue) {
156 for (int doubleValue = 0; doubleValue < 2; ++doubleValue) {
157 const size_t index =
158 mesh3dPipelineIndex(blend, depthValue != 0, doubleValue != 0);
159 surfacePipelines[index] = createMesh3DStylePipeline(
160 embeddedSpirv(mesh3d_vert_spv), embeddedSpirv(mesh3d_frag_spv),
161 mesh3dPipelineLayout, renderPass, vk::SampleCountFlagBits::e1,
162 blend, depthValue != 0, doubleValue != 0);
163 }
164 }
165 }
166 buildPrimitive3DPipelines(renderPass, vk::SampleCountFlagBits::e1, primitivePipelines);
167 };
168 createResources(vk::Format::eR8G8B8A8Unorm, offscreen3DRenderPass, offscreen3DMeshPipeline,
169 offscreen3DSurfacePipelines, offscreenPrimitive3DPipelines);
170 createResources(vk::Format::eR16G16B16A16Sfloat, hdrOffscreen3DRenderPass, hdrOffscreen3DMeshPipeline,
171 hdrOffscreen3DSurfacePipelines, hdrOffscreenPrimitive3DPipelines);
172 if (!offscreen3DPool) {
173 vk::CommandPoolCreateInfo poolInfo{};
174 poolInfo.flags = vk::CommandPoolCreateFlagBits::eTransient | vk::CommandPoolCreateFlagBits::eResetCommandBuffer;
175 offscreen3DPool = dev->createCommandPool(poolInfo);
176 vk::CommandBufferAllocateInfo allocInfo{};
177 allocInfo.commandPool = offscreen3DPool;
178 allocInfo.level = vk::CommandBufferLevel::ePrimary;
179 allocInfo.commandBufferCount = 1;
180 auto bufs = dev->allocateCommandBuffers(allocInfo);
181 offscreen3DCB = bufs[0];
182 vk::FenceCreateInfo fenceInfo{};
183 fenceInfo.flags = vk::FenceCreateFlagBits::eSignaled; // first wait passes immediately
184 offscreen3DFence = dev->createFence(fenceInfo);
185 const auto &limits = dev.physical_device.properties.limits;
186 if (limits.timestampComputeAndGraphics && limits.timestampPeriod > 0.f) {
187 vk::QueryPoolCreateInfo queryInfo{};
188 queryInfo.queryType = vk::QueryType::eTimestamp;
189 queryInfo.queryCount = 2;
190 offscreen3DTimestampQueryPool = dev->createQueryPool(queryInfo);
191 offscreen3DTimestampPeriodNs = limits.timestampPeriod;
192 }
193 }
194}

249 ensureFileTexturesReady();

251 if (!initialized) throw Exception("begin3DFrameToCanvas: graphics not initialized");
252 if (!canvas) throw Exception("begin3DFrameToCanvas: null canvas");
253 createMesh3DPipeline();
254 auto *oc = dynamic_cast<OffscreenCanvas *>(canvas);
255 if (!oc) throw Exception("begin3DFrameToCanvas: not an offscreen canvas");
256 if (offscreen3DPassOpen) throw Exception("begin3DFrameToCanvas: already open");
257 if (swapchainPassOpen) {
258 try {
259 flushToSwapchain();
260 } catch (...) {
261 abortOpen3DFrame();
262 }
263 }
264 clearColor = backgroundColor;
265 ensureOffscreen3DResources();
266 oc->ensure3D();
267 offscreen3DCanvas = oc;
268 offscreen3DHDRActive = oc->isHDR();
269
270 if (offscreen3DFence) (void)device->waitForFences(1, &offscreen3DFence, VK_TRUE, UINT64_MAX);
271 offscreenPrimitive3DDrawIndex = 0;
272 vk::CommandBufferBeginInfo beginInfo{};
273 offscreen3DCB.begin(beginInfo);
274 lastOffscreen3DGpuDurationMs = 0.f;
275 if (offscreen3DTimestampQueryPool) {
276 offscreen3DCB.resetQueryPool(offscreen3DTimestampQueryPool, 0, 2);
277 offscreen3DCB.writeTimestamp(vk::PipelineStageFlagBits::eTopOfPipe,
278 offscreen3DTimestampQueryPool, 0);
279 }
280 std::array<vk::ClearValue, 2> clears{};
281 clears[0].color =
282 vk::ClearColorValue(std::array<float, 4>{clearColor.r, clearColor.g, clearColor.b, 1.f});
283 clears[1].depthStencil = vk::ClearDepthStencilValue{1.0f, 0};
284 vk::RenderPassBeginInfo rpBegin{};
285 rpBegin.renderPass = getOffscreen3DRenderPass(offscreen3DHDRActive);
286 rpBegin.framebuffer = oc->framebuffer3D();
287 rpBegin.renderArea =
288 vk::Rect2D{{0, 0}, {uint32_t(oc->getWidth()), uint32_t(oc->getHeight())}};
289 rpBegin.clearValueCount = uint32_t(clears.size());
290 rpBegin.pClearValues = clears.data();
291 oc->colorImage().beginColorAttachment();
292 oc->depthImage().beginDepthAttachment();
293 offscreen3DCB.beginRenderPass(rpBegin, vk::SubpassContents::eInline);
294 setViewportAndScissor(offscreen3DCB, oc->getWidth(), oc->getHeight());
295
296 {
297 auto &fslots = currentMesh3dFrameSlots();
298 fslots.lastDrawCount = fslots.drawIndex;
299 fslots.drawIndex = 0;
300 fslots.activePalette = 0;
301 ensureMesh3dRing(fslots);
302 auto &cslots = currentMesh3dClusteredFrameSlots();
303 cslots.lastDrawCount = cslots.drawIndex;
304 cslots.drawIndex = 0;
305 ensureMesh3dClusteredRing(cslots);
306 }
307 lastMesh3dPipeline = nullptr;
308 lastMesh3dClusteredPipeline = nullptr;
309 offscreen3DPassOpen = true;
310 frameHad3D = true;
311 hasPendingClear = false;
312}

314void Graphics::end3DFrameToCanvas() {
315 if (!offscreen3DPassOpen || !offscreen3DCB) return;
316 offscreen3DCB.endRenderPass();
317 if (offscreen3DCanvas) {
318 // The sampled render pass already performed its final-layout transition.
319 offscreen3DCanvas->colorImage().endSampledLayout();
320 }
321 if (offscreen3DTimestampQueryPool) {
322 offscreen3DCB.writeTimestamp(vk::PipelineStageFlagBits::eBottomOfPipe,
323 offscreen3DTimestampQueryPool, 1);
324 }
325 offscreen3DCB.end();
326
327 // Submit the offscreen render directly to the graphics queue (no present),
328 // then wait for it so it is fully drained before any subsequent render or
329 // teardown.
330 vk::SubmitInfo submitInfo{};
331 submitInfo.commandBufferCount = 1;
332 submitInfo.pCommandBuffers = &offscreen3DCB;
333 if (offscreen3DFence) (void)device->resetFences(1, &offscreen3DFence);
334 (void)device.getQueue(vkb::QueueType::graphics).submit(1, &submitInfo, offscreen3DFence);
335 if (offscreen3DFence)
336 (void)device->waitForFences(1, &offscreen3DFence, VK_TRUE, UINT64_MAX);
337 if (offscreen3DTimestampQueryPool && offscreen3DTimestampPeriodNs > 0.f) {
338 std::array<uint64_t, 2> ticks{};
339 const vk::Result queryResult = device->getQueryPoolResults(
340 offscreen3DTimestampQueryPool, 0, uint32_t(ticks.size()), sizeof(ticks), ticks.data(),
341 sizeof(uint64_t), vk::QueryResultFlagBits::e64 | vk::QueryResultFlagBits::eWait);
342 if (queryResult == vk::Result::eSuccess && ticks[1] >= ticks[0]) {
343 const double nanoseconds =
344 double(ticks[1] - ticks[0]) * double(offscreen3DTimestampPeriodNs);
345 lastOffscreen3DGpuDurationMs = float(nanoseconds * 1.0e-6);
346 }
347 }
348
349 offscreen3DPassOpen = false;
350 offscreen3DHDRActive = false;
351 offscreen3DCanvas = nullptr;
352}

415vk::Pipeline Graphics::buildVoxelRectPipeline(const vkb::BuiltRenderPass &rp,
416 vk::SampleCountFlagBits samples) {
417 auto vert = embeddedSpirv(voxel_rect_vert_spv);
418 auto frag = embeddedSpirv(voxel_rect_frag_spv);
419 ShaderModulePair modules(device, vert, frag);
420
421 vk::VertexInputBindingDescription bindings[3]{};
422 bindings[0].binding = 0;
423 bindings[0].stride = sizeof(glm::vec2);
424 bindings[0].inputRate = vk::VertexInputRate::eVertex;
425 bindings[1].binding = 1;
426 bindings[1].stride = sizeof(uint32_t);
427 bindings[1].inputRate = vk::VertexInputRate::eInstance;
428 bindings[2].binding = 2;
429 bindings[2].stride = sizeof(uint32_t);
430 bindings[2].inputRate = vk::VertexInputRate::eInstance;
431
432 vk::VertexInputAttributeDescription attrs[3]{};
433 attrs[0].location = 0;
434 attrs[0].binding = 0;
435 attrs[0].format = vk::Format::eR32G32Sfloat;
436 attrs[0].offset = 0;
437 attrs[1].location = 1;
438 attrs[1].binding = 1;
439 attrs[1].format = vk::Format::eR32Uint;
440 attrs[1].offset = 0;
441 attrs[2].location = 2;
442 attrs[2].binding = 2;
443 attrs[2].format = vk::Format::eR32Uint;
444 attrs[2].offset = 0;
445
446 vk::PipelineVertexInputStateCreateInfo vi{};
447 vi.vertexBindingDescriptionCount = 3;
448 vi.pVertexBindingDescriptions = bindings;
449 vi.vertexAttributeDescriptionCount = 3;
450 vi.pVertexAttributeDescriptions = attrs;
451
452 // Unit-quad indices 0-2-1 / 0-3-2 are object-space CCW for outward faces.
453 // perspectiveVulkanRH_ZO flips clip Y, so those faces are CCW in the
454 // framebuffer — CounterClockwise frontFace keeps them. Clockwise + Back
455 // (the mesh-pipeline convention) culls every submitted face after the Y
456 // flip; CPU 6-dir cull already dropped the true back faces, so the frame
457 // collapses to a 1px silhouette.
458 return device.createPipeline()
459 .useClassicPipeline(modules.vert, modules.frag)
460 .setPipelineLayout(voxelRectPipelineLayout)
461 .setVertexInputState(vi)
462 .setDynamicStatesViewportScissor()
463 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f, vk::CullModeFlagBits::eBack,
464 vk::FrontFace::eCounterClockwise)
465 .setMultisampler(false, samples)
466 .setDepthStencil(true, true, vk::CompareOp::eLess)
467 .setColorAttachmentCount(1)
468 .build(rp);
469}

508void Graphics::drawVoxelFaceInstances(const uint32_t *packed, int count, float originX,
509 float originY, float originZ, const std::string &faceDir,
510 Texture *atlas, int tilesPerRow, const uint32_t *ao) {

515 if (!packed || count <= 0) return;

517 int face = -1;
518 if (faceDir == "posX" || faceDir == "+x")
519 face = 0;

521 face = 1;
522 else if (faceDir == "posY" || faceDir == "+y")
523 face = 2;
524 else if (faceDir == "negY" || faceDir == "-y")
525 face = 3;
526 else if (faceDir == "posZ" || faceDir == "+z")
527 face = 4;
528 else if (faceDir == "negZ" || faceDir == "-z")
529 face = 5;
530 else
531 throw Exception("drawVoxelFaceInstances: unknown faceDir '%s'", faceDir.c_str());
532
533 Texture *tex = atlas ? atlas : whiteTexture;
534 if (!tex || !tex->gpuHandle) throw Exception("drawVoxelFaceInstances: missing texture");
535 auto *gpuTex = static_cast<GpuTexture *>(tex->gpuHandle);
536
537 ensureVoxelUnitQuad();
538
539 const vk::DeviceSize bytes = vk::DeviceSize(count) * sizeof(uint32_t);
540 auto &vframe = currentVoxelInstanceFrame();
541 const size_t slotIndex = vframe.drawIndex++;
542 while (vframe.slots.size() <= slotIndex) vframe.slots.emplace_back();
543 auto &slot = vframe.slots[slotIndex];
544 if (slot.capacityBytes < size_t(bytes) || !slot.buffer.buffer) {
545 const size_t alloc = std::max(size_t(bytes), slot.capacityBytes ? slot.capacityBytes * 2 : size_t(bytes));
546 slot.buffer.allocate(frameToken(), device, vk::BufferUsageFlagBits::eVertexBuffer, vk::DeviceSize(alloc),
547 vk::MemoryPropertyFlagBits::eHostVisible |
548 vk::MemoryPropertyFlagBits::eHostCoherent);
549 slot.capacityBytes = alloc;
550 }
551 slot.buffer.updateLocal(frameToken(), packed, size_t(bytes));
552
553 // Ambient-occlusion words (null → full bright). Kept parallel to packed.
554 const uint32_t defaultAO = 0xFFu; // all four corners AO=3
555 std::vector<uint32_t> aoDefaults;
556 if (!ao) {
557 aoDefaults.assign(size_t(count), defaultAO);
558 ao = aoDefaults.data();
559 }
560 const vk::DeviceSize aoBytes = vk::DeviceSize(count) * sizeof(uint32_t);
561 if (slot.aoCapacityBytes < size_t(aoBytes) || !slot.aoBuffer.buffer) {
562 const size_t alloc = std::max(size_t(aoBytes),
563 slot.aoCapacityBytes ? slot.aoCapacityBytes * 2
564 : size_t(aoBytes));
565 slot.aoBuffer.allocate(frameToken(), device, vk::BufferUsageFlagBits::eVertexBuffer,
566 vk::DeviceSize(alloc),
567 vk::MemoryPropertyFlagBits::eHostVisible |
568 vk::MemoryPropertyFlagBits::eHostCoherent);
569 slot.aoCapacityBytes = alloc;
570 }
571 slot.aoBuffer.updateLocal(frameToken(), ao, size_t(aoBytes));
572
573 VoxelRectPC pc{};
574 pc.viewProj = mesh3dFrameUbo.mvp;
575 pc.chunkOrigin = glm::vec4(originX, originY, originZ, float(face));
576 pc.atlasInfo = glm::vec4(float(std::max(1, tilesPerRow)), 0.f, 0.f, 0.f);
577 pc.tint = mesh3dFrameUbo.tint;
578
579 auto &cb = currentPresentCb();
580 vk::DescriptorSet set = voxelRectSetFor(gpuTex);
581 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, voxelRectPipeline);
582 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, voxelRectPipelineLayout, 0, 1, &set, 0,
583 nullptr);
584 cb.pushConstants(voxelRectPipelineLayout, vk::ShaderStageFlagBits::eVertex, 0, sizeof(VoxelRectPC),
585 &pc);
586
587 vk::Buffer vbufs[3] = {voxelUnitQuadVerts.buffer, slot.buffer.buffer, slot.aoBuffer.buffer};
588 vk::DeviceSize offsets[3] = {0, 0, 0};
589 cb.bindVertexBuffers(0, 3, vbufs, offsets);
590 cb.bindIndexBuffer(voxelUnitQuadIndices.buffer, 0, vk::IndexType::eUint32);
591 cb.drawIndexed(6, uint32_t(count), 0, 0, 0);
592}

744void Graphics::drawMeshGBuffer(Mesh *mesh, const glm::mat4 &mvp, const glm::mat4 &model, float nearZ,
745 float farZ, Texture *albedo, float tintR, float tintG, float tintB,
746 float motionX, float motionY, float roughness, float metallic) {
747 if (!gbufferPassActive) throw Exception("drawMeshGBuffer: call beginGBufferPass first");
748 if (!mesh || !mesh->gpuHandle) throw Exception("drawMeshGBuffer: null mesh");
749 GBufferDraw d{};
750 d.mesh = mesh;
751 d.albedo = albedo;
752 d.push.mvp = mvp;
753 d.push.modelR0 = glm::vec4(model[0][0], model[1][0], model[2][0], model[3][0]);
754 d.push.modelR1 = glm::vec4(model[0][1], model[1][1], model[2][1], model[3][1]);
755 d.push.modelR2 = glm::vec4(model[0][2], model[1][2], model[2][2], model[3][2]);
756 auto u6 = [](float x) -> uint32_t {
757 return uint32_t(std::lround(std::clamp(x, 0.f, 1.f) * 63.f));
758 };
759 // A float exactly represents every integer through 24 bits. Numeric packing avoids
760 // NaN canonicalization and subnormal flush-to-zero corrupting bit-cast payloads.
761 const uint32_t rough3 = uint32_t(std::lround(std::clamp(roughness, 0.f, 1.f) * 7.f));
762 const uint32_t metal3 = uint32_t(std::lround(std::clamp(metallic, 0.f, 1.f) * 7.f));
763 const uint32_t packedTint = u6(tintR) | (u6(tintG) << 6) | (u6(tintB) << 12) |
764 (rough3 << 18) | (metal3 << 21);
765 auto motion12 = [](float value) {
766 return uint32_t(std::lround(std::clamp(value, -1.f, 1.f) * 2047.f)) + 2047u;
767 };
768 const uint32_t packedMotion = motion12(motionX) | (motion12(motionY) << 12);
769 d.push.clip = glm::vec4(nearZ, farZ, float(packedTint), float(packedMotion));
770 prepareSkinPass(mesh, albedo, mvp, model, d.push.clip, d.skinSet, d.skinUboOffset);
771 gbufferPassDraws.push_back(d);
772}

905void Graphics::recordDecalPassInto(vk::CommandBuffer cb, DecalSlot &slot, GBufferSlot &gslot) {
906 const uint32_t w = uint32_t(decalWidth);
907 const uint32_t h = uint32_t(decalHeight);
908 if (w == 0 || h == 0) {
909 decalPassDraws.clear();
910 return;
911 }
912
913 DecalCameraUBO cam{};
914 cam.viewProj = decalViewProj;
915 cam.invViewProj = glm::inverse(decalViewProj);
916 cam.nearFarTexel =
917 glm::vec4(decalNear, decalFar, 1.f / float(w), 1.f / float(h));
918 updateRingLocal(slot.cameraUbo, 0, &cam, sizeof(cam));
919
920 slot.albedo.beginColorAttachment();
921 slot.normal.beginColorAttachment();
922 slot.params.beginColorAttachment();
923 std::array<vk::ClearValue, 3> clears{};
924 clears[0].color = vk::ClearColorValue(std::array<float, 4>{0, 0, 0, 0});
925 clears[1].color = vk::ClearColorValue(std::array<float, 4>{0, 0, 0, 0});
926 clears[2].color = vk::ClearColorValue(std::array<float, 4>{0, 0, 0, 0});
927 vk::RenderPassBeginInfo rpBegin{};
928 rpBegin.renderPass = decalRenderPass;
929 rpBegin.framebuffer = slot.framebuffer;
930 rpBegin.renderArea = vk::Rect2D{{0, 0}, {w, h}};
931 rpBegin.clearValueCount = uint32_t(clears.size());
932 rpBegin.pClearValues = clears.data();
933 cb.beginRenderPass(rpBegin, vk::SubpassContents::eInline);
934 setViewportAndScissor(cb, w, h);
935 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, decalPipeline);
936 decalLayerFresh = true;
937
938 for (const auto &d : decalPassDraws) {
939 GpuTexture *gpuAlb = d.albedo && d.albedo->gpuHandle
940 ? static_cast<GpuTexture *>(d.albedo->gpuHandle)
941 : static_cast<GpuTexture *>(decalFlatAlbedo->gpuHandle);
942 GpuTexture *gpuNrm = d.normal && d.normal->gpuHandle
943 ? static_cast<GpuTexture *>(d.normal->gpuHandle)
944 : static_cast<GpuTexture *>(decalFlatNormal->gpuHandle);
945 GpuTexture *gpuPrm = d.params && d.params->gpuHandle
946 ? static_cast<GpuTexture *>(d.params->gpuHandle)
947 : static_cast<GpuTexture *>(decalFlatParams->gpuHandle);
948 DecalInstanceData inst{};
949 inst.model = d.model;
950 inst.uvRect = glm::vec4(d.uvRect[0], d.uvRect[1], d.uvRect[2], d.uvRect[3]);
951 inst.fadeParams =
952 glm::vec4(d.fade, d.normalStrength, d.roughnessStrength, d.metalStrength);
953 inst.extraParams = glm::vec4(d.emissiveStrength, float(d.blendMode), 0.f, 0.f);
954 const vkb::BoundSet set = decalSetFor(slot, gpuAlb, gpuNrm, gpuPrm, &gslot.depthGpu,
955 &gslot.normalGpu);
956 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, decalPipelineLayout, 0, 1,
957 set.ptr(), 0, nullptr);
958 cb.pushConstants(decalPipelineLayout,
959 vk::ShaderStageFlagBits::eVertex |
960 vk::ShaderStageFlagBits::eFragment,
961 0, sizeof(inst), &inst);
962 cb.draw(3, 1, 0, 0);
963 }
964 decalPassDraws.clear();
965 cb.endRenderPass();
966 slot.albedo.endSampledLayout();
967 slot.normal.endSampledLayout();
968 slot.params.endSampledLayout();
969}
