59void Graphics::begin3DFrame() {
60 ASSERT(initialized);
61 if (!initialized) throw Exception("begin3DFrame: graphics not initialized");
62 if (isCanvasActive()) throw Exception("begin3DFrame: cannot start 3D while a Canvas is active");
63 if (swapchainPassOpen) {
64 // Previous eve_render opened 3D then threw before present().
65 try {
66 flushToSwapchain();
67 } catch (...) {
68 abortOpen3DFrame();
69 }
70 }
71 // Soft-fail like flushToSwapchain: on Android/iOS the surface may still be
72 // settling after orientation change; throwing would abort the whole script.
73 if (!beginPresentCommandBuffer())
74 return;
75 recordDeferredFrameGraph();
76 vgAnyThisFrame_ = false;
77 currentFrameArena().reset();
78 ensureGpuDrivenCullResources(gbufferWidth > 0 ? gbufferWidth : int(swapchain.extent.width),
79 gbufferHeight > 0 ? gbufferHeight : int(swapchain.extent.height));
80
81 // 3D pass clears with backgroundColor (setBackgroundColor), not a stale 2D clear.
82 clearColor = backgroundColor;
83 createSceneColorResources(int(swapchain.extent.width), int(swapchain.extent.height));
84 // GPU-driven cull defers the scene color pass so the compute section
85 // (HZB build + cull + emit + VG cull) can run before the opaque draws;
86 // gpuDrivenOpenScenePass() opens it when the renderer reaches the forward
87 // section.
88 gpuDrivenScenePassPending_ = gpuDrivenEnabled_ && gpuDrivenCullReady_;
89 if (!gpuDrivenScenePassPending_) {
90 if (beginSceneColorRenderPass()) {
91 // Rebuild scene-pass pipelines to match the active scene pass (MSAA).
92 // Must happen AFTER the pass opens: if the MSAA scene pass is unavailable
93 // we fall back to the swapchain (e1) render pass below, and rasterizing
94 // with an Nx pipeline into an e1 pass is UB that can hang the GPU (TDR).
95 ensureScenePassPipelines(activeScenePass(), activeSceneSamples());
96 } else {
97 // Scene pass unavailable — draw 3D straight into the swapchain.
98 ensureScenePassPipelines(renderpass, vk::SampleCountFlagBits::e1);
99 beginSwapchainColorPass();
100 }
101 }
102
103 auto &cb = currentPresentCb();
104 setViewportAndScissor(cb, swapchain.extent.width, swapchain.extent.height);
105
106 {
107 auto &fslots = currentMesh3dFrameSlots();
108 fslots.lastDrawCount = fslots.drawIndex;
109 fslots.drawIndex = 0;
110 ensureMesh3dRing(fslots);
111 auto &cslots = currentMesh3dClusteredFrameSlots();
112 cslots.lastDrawCount = cslots.drawIndex;
113 cslots.drawIndex = 0;
114 ensureMesh3dClusteredRing(cslots);
115 }
116 lastMesh3dPipeline = nullptr;
117 lastMesh3dClusteredPipeline = nullptr;
118 currentVoxelInstanceFrame().drawIndex = 0;
119 swapchainPassOpen = !gpuDrivenScenePassPending_;
120 frameHad3D = true;
121 hasPendingClear = false;
122}

124void Graphics::ensureOffscreen3DResources() {
125 auto &dev = getDevice();
126 if (!offscreen3DRenderPass) {
127 offscreen3DRenderPass =
128 dev.createRenderPass()
129 .addSampledColorAttachment(vk::Format::eR8G8B8A8Unorm)
130 .addDepthAttachment(depthFormat, vk::AttachmentLoadOp::eClear,
131 vk::AttachmentStoreOp::eDontCare)
132 .addSubpass(vkb::SubpassBuilder()
133 .addAttachmentRef(0, vk::ImageLayout::eColorAttachmentOptimal)
134 .setDepthStencilAttachment(
135 1, vk::ImageLayout::eDepthStencilAttachmentOptimal))
136 .addExternalShaderReadDependencies()
137 .build();
138 offscreen3DMeshPipeline = createMesh3DStylePipeline(
139 embeddedSpirv(mesh3d_vert_spv), embeddedSpirv(mesh3d_frag_spv), mesh3dPipelineLayout,
140 offscreen3DRenderPass, vk::SampleCountFlagBits::e1);
141 }
142 if (!offscreen3DPool) {
143 vk::CommandPoolCreateInfo poolInfo{};
144 poolInfo.flags = vk::CommandPoolCreateFlagBits::eTransient;
145 offscreen3DPool = dev->createCommandPool(poolInfo);
146 vk::CommandBufferAllocateInfo allocInfo{};
147 allocInfo.commandPool = offscreen3DPool;
148 allocInfo.level = vk::CommandBufferLevel::ePrimary;
149 allocInfo.commandBufferCount = 1;
150 auto bufs = dev->allocateCommandBuffers(allocInfo);
151 offscreen3DCB = bufs[0];
152 vk::FenceCreateInfo fenceInfo{};
153 fenceInfo.flags = vk::FenceCreateFlagBits::eSignaled; // first wait passes immediately
154 offscreen3DFence = dev->createFence(fenceInfo);
155 }
156}

179void Graphics::begin3DFrameToCanvas(Canvas *canvas) {
180 ASSERT(initialized);
181 if (!initialized) throw Exception("begin3DFrameToCanvas: graphics not initialized");
182 if (!canvas) throw Exception("begin3DFrameToCanvas: null canvas");
183 auto *oc = dynamic_cast<OffscreenCanvas *>(canvas);
184 if (!oc) throw Exception("begin3DFrameToCanvas: not an offscreen canvas");
185 if (offscreen3DPassOpen) throw Exception("begin3DFrameToCanvas: already open");
186 if (swapchainPassOpen) {
187 try {
188 flushToSwapchain();
189 } catch (...) {
190 abortOpen3DFrame();
191 }
192 }
193 clearColor = backgroundColor;
194 ensureOffscreen3DResources();
195 oc->ensure3D();
196 offscreen3DCanvas = oc;
197
198 if (offscreen3DFence) (void)device->waitForFences(1, &offscreen3DFence, VK_TRUE, UINT64_MAX);
199 vk::CommandBufferBeginInfo beginInfo{};
200 offscreen3DCB.begin(beginInfo);
201 std::array<vk::ClearValue, 2> clears{};
202 clears[0].color =
203 vk::ClearColorValue(std::array<float, 4>{clearColor.r, clearColor.g, clearColor.b, 1.f});
204 clears[1].depthStencil = vk::ClearDepthStencilValue{1.0f, 0};
205 vk::RenderPassBeginInfo rpBegin{};
206 rpBegin.renderPass = offscreen3DRenderPass;
207 rpBegin.framebuffer = oc->framebuffer3D();
208 rpBegin.renderArea =
209 vk::Rect2D{{0, 0}, {uint32_t(oc->getWidth()), uint32_t(oc->getHeight())}};
210 rpBegin.clearValueCount = uint32_t(clears.size());
211 rpBegin.pClearValues = clears.data();
212 oc->colorImage().beginColorAttachment();
213 oc->depthImage().beginDepthAttachment();
214 offscreen3DCB.beginRenderPass(rpBegin, vk::SubpassContents::eInline);
215 setViewportAndScissor(offscreen3DCB, oc->getWidth(), oc->getHeight());
216
217 {
218 auto &fslots = currentMesh3dFrameSlots();
219 fslots.lastDrawCount = fslots.drawIndex;
220 fslots.drawIndex = 0;
221 ensureMesh3dRing(fslots);
222 auto &cslots = currentMesh3dClusteredFrameSlots();
223 cslots.lastDrawCount = cslots.drawIndex;
224 cslots.drawIndex = 0;
225 ensureMesh3dClusteredRing(cslots);
226 }
227 lastMesh3dPipeline = nullptr;
228 lastMesh3dClusteredPipeline = nullptr;
229 offscreen3DPassOpen = true;
230 frameHad3D = true;
231 hasPendingClear = false;
232}

234void Graphics::end3DFrameToCanvas() {
235 if (!offscreen3DPassOpen || !offscreen3DCB) return;
236 offscreen3DCB.endRenderPass();
237 if (offscreen3DCanvas) {
238 offscreen3DCanvas->colorImage().setLayout(offscreen3DCB,
239 vk::ImageLayout::eShaderReadOnlyOptimal);
240 }
241 offscreen3DCB.end();
242
243 // Submit the offscreen render directly to the graphics queue (no present),
244 // then wait for it so it is fully drained before any subsequent render or
245 // teardown.
246 vk::SubmitInfo submitInfo{};
247 submitInfo.commandBufferCount = 1;
248 submitInfo.pCommandBuffers = &offscreen3DCB;
249 if (offscreen3DFence) (void)device->resetFences(1, &offscreen3DFence);
250 (void)device.getQueue(vkb::QueueType::graphics).submit(1, &submitInfo, offscreen3DFence);
251 if (offscreen3DFence)
252 (void)device->waitForFences(1, &offscreen3DFence, VK_TRUE, UINT64_MAX);
253
254 offscreen3DPassOpen = false;
255 offscreen3DCanvas = nullptr;
256}

319vk::Pipeline Graphics::buildVoxelRectPipeline(const vkb::BuiltRenderPass &rp,
320 vk::SampleCountFlagBits samples) {
321 auto vert = embeddedSpirv(voxel_rect_vert_spv);
322 auto frag = embeddedSpirv(voxel_rect_frag_spv);
323 ShaderModulePair modules(device, vert, frag);
324
325 vk::VertexInputBindingDescription bindings[3]{};
326 bindings[0].binding = 0;
327 bindings[0].stride = sizeof(glm::vec2);
328 bindings[0].inputRate = vk::VertexInputRate::eVertex;
329 bindings[1].binding = 1;
330 bindings[1].stride = sizeof(uint32_t);
331 bindings[1].inputRate = vk::VertexInputRate::eInstance;
332 bindings[2].binding = 2;
333 bindings[2].stride = sizeof(uint32_t);
334 bindings[2].inputRate = vk::VertexInputRate::eInstance;
335
336 vk::VertexInputAttributeDescription attrs[3]{};
337 attrs[0].location = 0;
338 attrs[0].binding = 0;
339 attrs[0].format = vk::Format::eR32G32Sfloat;
340 attrs[0].offset = 0;
341 attrs[1].location = 1;
342 attrs[1].binding = 1;
343 attrs[1].format = vk::Format::eR32Uint;
344 attrs[1].offset = 0;
345 attrs[2].location = 2;
346 attrs[2].binding = 2;
347 attrs[2].format = vk::Format::eR32Uint;
348 attrs[2].offset = 0;
349
350 vk::PipelineVertexInputStateCreateInfo vi{};
351 vi.vertexBindingDescriptionCount = 3;
352 vi.pVertexBindingDescriptions = bindings;
353 vi.vertexAttributeDescriptionCount = 3;
354 vi.pVertexAttributeDescriptions = attrs;
355
356 // Unit-quad indices 0-2-1 / 0-3-2 are object-space CCW for outward faces.
357 // perspectiveVulkanRH_ZO flips clip Y, so those faces are CCW in the
358 // framebuffer — CounterClockwise frontFace keeps them. Clockwise + Back
359 // (the mesh-pipeline convention) culls every submitted face after the Y
360 // flip; CPU 6-dir cull already dropped the true back faces, so the frame
361 // collapses to a 1px silhouette.
362 return device.createPipeline()
363 .useClassicPipeline(modules.vert, modules.frag)
364 .setPipelineLayout(voxelRectPipelineLayout)
365 .setVertexInputState(vi)
366 .setDynamicStatesViewportScissor()
367 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f, vk::CullModeFlagBits::eBack,
368 vk::FrontFace::eCounterClockwise)
369 .setMultisampler(false, samples)
370 .setDepthStencil(true, true, vk::CompareOp::eLess)
371 .setColorAttachmentCount(1)
372 .build(rp);
373}

412void Graphics::drawVoxelFaceInstances(const uint32_t *packed, int count, float originX,
413 float originY, float originZ, const std::string &faceDir,
414 Texture *atlas, int tilesPerRow, const uint32_t *ao) {
415 ASSERT(initialized);
416 if (!initialized) throw Exception("drawVoxelFaceInstances: graphics not initialized");
417 if (!swapchainPassOpen) throw Exception("drawVoxelFaceInstances: call begin3DFrame first");
418 if (!voxelRectPipeline) createVoxelRectPipeline();
419 if (!packed || count <= 0) return;
420
421 int face = -1;
422 if (faceDir == "posX" || faceDir == "+x")
423 face = 0;
424 else if (faceDir == "negX" || faceDir == "-x")
425 face = 1;
426 else if (faceDir == "posY" || faceDir == "+y")
427 face = 2;
428 else if (faceDir == "negY" || faceDir == "-y")
429 face = 3;
430 else if (faceDir == "posZ" || faceDir == "+z")
431 face = 4;
432 else if (faceDir == "negZ" || faceDir == "-z")
433 face = 5;
434 else
435 throw Exception("drawVoxelFaceInstances: unknown faceDir '%s'", faceDir.c_str());
436
437 Texture *tex = atlas ? atlas : whiteTexture;
438 if (!tex || !tex->gpuHandle) throw Exception("drawVoxelFaceInstances: missing texture");
439 auto *gpuTex = static_cast<GpuTexture *>(tex->gpuHandle);
440
441 ensureVoxelUnitQuad();
442
443 const vk::DeviceSize bytes = vk::DeviceSize(count) * sizeof(uint32_t);
444 auto &vframe = currentVoxelInstanceFrame();
445 const size_t slotIndex = vframe.drawIndex++;
446 while (vframe.slots.size() <= slotIndex) vframe.slots.emplace_back();
447 auto &slot = vframe.slots[slotIndex];
448 if (slot.capacityBytes < size_t(bytes) || !slot.buffer.buffer) {
449 const size_t alloc = std::max(size_t(bytes), slot.capacityBytes ? slot.capacityBytes * 2 : size_t(bytes));
450 slot.buffer.allocate(frameToken(), device, vk::BufferUsageFlagBits::eVertexBuffer, vk::DeviceSize(alloc),
451 vk::MemoryPropertyFlagBits::eHostVisible |
452 vk::MemoryPropertyFlagBits::eHostCoherent);
453 slot.capacityBytes = alloc;
454 }
455 slot.buffer.updateLocal(frameToken(), packed, size_t(bytes));
456
457 // Ambient-occlusion words (null → full bright). Kept parallel to packed.
458 const uint32_t defaultAO = 0xFFu; // all four corners AO=3
459 std::vector<uint32_t> aoDefaults;
460 if (!ao) {
461 aoDefaults.assign(size_t(count), defaultAO);
462 ao = aoDefaults.data();
463 }
464 const vk::DeviceSize aoBytes = vk::DeviceSize(count) * sizeof(uint32_t);
465 if (slot.aoCapacityBytes < size_t(aoBytes) || !slot.aoBuffer.buffer) {
466 const size_t alloc = std::max(size_t(aoBytes),
467 slot.aoCapacityBytes ? slot.aoCapacityBytes * 2
468 : size_t(aoBytes));
469 slot.aoBuffer.allocate(frameToken(), device, vk::BufferUsageFlagBits::eVertexBuffer,
470 vk::DeviceSize(alloc),
471 vk::MemoryPropertyFlagBits::eHostVisible |
472 vk::MemoryPropertyFlagBits::eHostCoherent);
473 slot.aoCapacityBytes = alloc;
474 }
475 slot.aoBuffer.updateLocal(frameToken(), ao, size_t(aoBytes));
476
477 VoxelRectPC pc{};
478 pc.viewProj = mesh3dFrameUbo.mvp;
479 pc.chunkOrigin = glm::vec4(originX, originY, originZ, float(face));
480 pc.atlasInfo = glm::vec4(float(std::max(1, tilesPerRow)), 0.f, 0.f, 0.f);
481 pc.tint = mesh3dFrameUbo.tint;
482
483 auto &cb = currentPresentCb();
484 vk::DescriptorSet set = voxelRectSetFor(gpuTex);
485 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, voxelRectPipeline);
486 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, voxelRectPipelineLayout, 0, 1, &set, 0,
487 nullptr);
488 cb.pushConstants(voxelRectPipelineLayout, vk::ShaderStageFlagBits::eVertex, 0, sizeof(VoxelRectPC),
489 &pc);
490
491 vk::Buffer vbufs[3] = {voxelUnitQuadVerts.buffer, slot.buffer.buffer, slot.aoBuffer.buffer};
492 vk::DeviceSize offsets[3] = {0, 0, 0};
493 cb.bindVertexBuffers(0, 3, vbufs, offsets);
494 cb.bindIndexBuffer(voxelUnitQuadIndices.buffer, 0, vk::IndexType::eUint32);
495 cb.drawIndexed(6, uint32_t(count), 0, 0, 0);
496}

783void Graphics::buildDeferredFrameGraphs() {
784 // One FrameGraph per in-flight slot imports the engine-owned targets and
785 // owns the deferred passes: the 3 CSM cascades (per-layer views of the
786 // shadow array) + the G-buffer fill share one dependency-free layer, so the
787 // JobSystem executor records all four command buffers concurrently (see
788 // recordDeferredFrameGraph). The engine keeps image ownership so
789 // renderEntityIdMask / readGBufferToImageData and the postFX wrappers are
790 // unaffected. Each graph is only used on its slot's frames, so its command
791 // buffer is reused two frames later — by then the present slot fence
792 // guarantees the previous graph submit completed (same queue, submitted
793 // before the present command buffer).
794 const vk::Format depthFmt = vk::Format::eD32Sfloat;
795 const vk::Format colorFmt = pickGBufferColorFormat(device);
796 const uint32_t mapSize = uint32_t(ShadowConfig::kMapSize);
797 const uint32_t shadowLayers = uint32_t(ShadowConfig::kCascades);
798 const uint32_t w = gbufferWidth > 0 ? uint32_t(gbufferWidth) : 1u;
799 const uint32_t h = gbufferHeight > 0 ? uint32_t(gbufferHeight) : 1u;
800
801 for (size_t i = 0; i < deferredFrameGraphs_.size(); ++i) {
802 auto graph = std::make_unique<vkb::FrameGraph>(&device, 1);
803
804 vkb::TextureDesc shadowDesc;
805 shadowDesc.format = depthFmt;
806 shadowDesc.extent = vk::Extent3D{mapSize, mapSize, 1};
807 shadowDesc.arrayLayers = shadowLayers;
808 shadowDesc.aspect = vk::ImageAspectFlagBits::eDepth;
809 shadowDesc.usage = vk::ImageUsageFlagBits::eSampled |
810 vk::ImageUsageFlagBits::eDepthStencilAttachment;
811 shadowDesc.afterLayout = vk::ImageLayout::eDepthStencilReadOnlyOptimal;
812 vk::ClearValue shadowClear{};
813 shadowClear.depthStencil = vk::ClearDepthStencilValue{1.0f, 0};
814 const bool haveShadowSlot =
815 i < shadowMaps.size() && shadowMaps[i].image.layerCount() >= shadowLayers;
816 if (haveShadowSlot) {
817 const vk::Image shadowImage = shadowMaps[i].image.image();
818 for (uint32_t c = 0; c < shadowLayers; ++c) {
819 auto shadowH = graph->importTexture("shadowCascade" + std::to_string(c),
820 shadowImage, shadowMaps[i].image.layerView(c),
821 shadowDesc);
822 graph->addPass("shadow" + std::to_string(c))
823 .depthAttachment(shadowH, vkb::AttachmentOp::clear(shadowClear))
824 .record([this, c](vkb::FrameGraphPassContext &ctx) {
825 recordShadowCascadePass(ctx, int(c));
826 });
827 }
828 }
829
830 if (i < gbufferSlots.size() && gbufferWidth > 0 && gbufferHeight > 0) {
831 auto &slot = gbufferSlots[i];
832 vkb::TextureDesc colorDesc;
833 colorDesc.format = colorFmt;
834 colorDesc.extent = vk::Extent3D{w, h, 1};
835 colorDesc.usage = vk::ImageUsageFlagBits::eSampled |
836 vk::ImageUsageFlagBits::eColorAttachment |
837 vk::ImageUsageFlagBits::eTransferSrc;
838 colorDesc.afterLayout = vk::ImageLayout::eShaderReadOnlyOptimal;
839 auto normalH = graph->importTexture("gbNormal", slot.normal.image(),
840 slot.normal.imageView(), colorDesc);
841 auto depthColorH = graph->importTexture("gbDepthColor", slot.depthColor.image(),
842 slot.depthColor.imageView(), colorDesc);
843 auto albedoH = graph->importTexture("gbAlbedo", slot.albedo.image(),
844 slot.albedo.imageView(), colorDesc);
845
846 vkb::TextureDesc depthDesc;
847 depthDesc.format = depthFmt;
848 depthDesc.extent = vk::Extent3D{w, h, 1};
849 depthDesc.aspect = vk::ImageAspectFlagBits::eDepth;
850 depthDesc.usage = vk::ImageUsageFlagBits::eSampled |
851 vk::ImageUsageFlagBits::eDepthStencilAttachment;
852 depthDesc.afterLayout = vk::ImageLayout::eDepthStencilReadOnlyOptimal;
853 auto depthH = graph->importTexture("gbHwDepth", slot.depth.image(),
854 slot.depth.imageView(), depthDesc);
855
856 std::array<vk::ClearValue, 4> clears{};
857 clears[0].color = vk::ClearColorValue(std::array<float, 4>{0, 0, 0, 0});
858 clears[1].color = vk::ClearColorValue(std::array<float, 4>{1, 1, 1, 1});
859 clears[2].color = vk::ClearColorValue(std::array<float, 4>{0, 0, 0, 0});
860 clears[3].depthStencil = vk::ClearDepthStencilValue{1.0f, 0};
861 graph->addPass("gbuffer")
862 .colorAttachment(normalH, vkb::AttachmentOp::clear(clears[0]))
863 .colorAttachment(depthColorH, vkb::AttachmentOp::clear(clears[1]))
864 .colorAttachment(albedoH, vkb::AttachmentOp::clear(clears[2]))
865 .depthAttachment(depthH, vkb::AttachmentOp::clear(clears[3]))
866 .record([this](vkb::FrameGraphPassContext &ctx) { recordGBufferPassDraws(ctx); });
867 }
868 graph->compile();
869 deferredFrameGraphs_[i] = std::move(graph);
870 }

873void Graphics::recordShadowCascadePass(vkb::FrameGraphPassContext &ctx, int cascade) {
874 // Runs inside the FrameGraph's "shadow<cascade>" render-pass instance
875 // (already begun with a depth clear); only draw commands go here. The pass
876 // may be recorded on a JobSystem worker, so everything below must be
877 // read-only: shadowCascadeDraws was captured by endShadowPass on the main
878 // thread before the graph records.
879 auto &cb = ctx.commandBuffer();
880 const vk::Extent2D extent = ctx.extent();
881 const uint32_t size = extent.width ? extent.width : uint32_t(ShadowConfig::kMapSize);
882 setViewportAndScissor(cb, size, size);
883 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, shadowPipeline);
884 bool alphaBound = false;
885 for (const auto &d : shadowCascadeDraws[cascade]) {
886 if (!d.mesh || !d.mesh->gpuHandle) continue;
887 const bool wantAlpha = d.alphaTest && shadowAlphaPipeline;
888 if (wantAlpha != alphaBound) {
889 cb.bindPipeline(vk::PipelineBindPoint::eGraphics,
890 wantAlpha ? shadowAlphaPipeline : shadowPipeline);
891 alphaBound = wantAlpha;
892 }
893 auto *gpuMesh = static_cast<GpuMesh *>(d.mesh->gpuHandle);
894 if (wantAlpha) {
895 Texture *alb = d.albedo ? d.albedo : whiteTexture;
896 if (alb && alb->gpuHandle && texSetLayout) {
897 auto *gpuTex = static_cast<GpuTexture *>(alb->gpuHandle);
898 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics,
899 shadowAlphaPipelineLayout, 0, 1,
900 gpuTex->descriptorSet.ptr(), 0, nullptr);
901 }
902 }
903 cb.pushConstants(wantAlpha ? shadowAlphaPipelineLayout : shadowPipelineLayout,
904 vk::ShaderStageFlagBits::eVertex, 0, sizeof(glm::mat4), &d.mvp);
905 drawIndexedMesh(cb, *gpuMesh);
906 }

943void Graphics::recordDeferredFrameGraph() {
944 const size_t slot = currentFrameSlot();
945 if (deferredGraphRecorded_ && deferredGraphRecordedSlot_ == slot) {
946 // render3D can be called several times per script frame (e.g. the
947 // render tests call it 3x before present). The deferred graph's command
948 // buffers must only be recorded once per slot per frame — re-recording
949 // them while the previous submit is still in flight would reset
950 // in-use command buffers (UB, GPU hang).
951 return;
952 }
953 auto *graph = currentDeferredFrameGraph();
954 if (!graph && (!shadowMaps.empty() || !gbufferSlots.empty())) {
955 // Shadows can be enabled without a G-buffer pass (or vice versa); build
956 // the deferred graphs on demand from whatever targets exist today.
957 buildDeferredFrameGraphs();
958 graph = currentDeferredFrameGraph();
959 }
960 if (!graph || !gbufferPipeline || !gbufferRenderPass || !shadowPipeline) {
961 dropPendingOffscreenPasses();
962 return;
963 }
964 // Record the declarative deferred passes (3 CSM cascades + G-buffer, one
965 // independent layer) with the JobSystem executor — the four command
966 // buffers are recorded concurrently on workers — then submit them on the
967 // graphics queue before the swapchain pass begins. Layout transitions and
968 // the render-pass instances are planned by the FrameGraph. Same-queue
969 // submission order plus the present slot fence (waited in Present::begin)
970 // keep this slot's graph command buffers safe to reuse two frames later.
971 auto *jobs = thread::Thread::create()->getJobSystem();
972 jobs->beginFrame(); // idempotent wait; recycles the per-frame arena
973 // Re-plan every frame (cheap; device objects are cached) so the graph is
974 // in the compiled phase for this record cycle — vkb::FrameGraph enforces
975 // build -> compile -> record -> submit and record() exactly once per
976 // compile.
977 graph->compile();
978 // Parallel executor: each pass owns a dedicated command pool (one pool per
979 // frame slot per pass), so the workers never share a pool while recording
980 // concurrently — the Vulkan external-synchronization rule for command
981 // pools is satisfied structurally.
982 recordFrameGraphWithJobSystem(*graph, jobs);
983 graph->submit();
984 jobs->endFrame();
985 for (auto &d : shadowCascadeDraws) d.clear();
986 gbufferPassDraws.clear();
987 shadowPendingMask = 0;
988 gbufferPending = false;
989 deferredGraphRecorded_ = true;
990 deferredGraphRecordedSlot_ = slot;
991}
