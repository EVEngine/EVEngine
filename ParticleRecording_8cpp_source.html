22void recordParticleEvent(const ParticleEmitter::Config& config, ParticleEmitter::Sim& sim, const Particle& particle,
23 ParticleRecordEventKind kind) {
24 if (sim.recording.state() != ParticleRecordState::Recording) return;
25 if (config.gpuSimulation || config.sortMode != "none" ||
26 (config.renderMode != "billboard" && config.renderMode != "axis" && config.renderMode != "stretched")) {
27 sim.recording.invalidate();
28 return;
29 }
30 auto appearance = evaluateParticleAppearance(config, particle);
31 if (!appearance) {
32 sim.recording.invalidate();
33 return;
34 }
35 auto appended = sim.recording.append({sim.recordingSeconds, particle.birthSerial, kind, appearance.value()});
36 if (!appended) sim.recording.invalidate();
37}
38Result<std::vector<ParticleRecordEvent>> finishParticleRecording(ParticleEmitter& emitter) {
39 auto sim = emitter.sim();
40 if (emitter.gpuSim()->residentActive) sim->recording.invalidate();
41 for (int i = 0; i < sim->alive; ++i)
42 recordParticleEvent(*emitter.config(), *sim, sim->particles[std::size_t(i)], ParticleRecordEventKind::Death);
43 return sim->recording.finish();
44}
