57eve::Result<void> CardAttributeAdapter::ensure(CardData& card) {
58 auto bound = card.attributes()->values.bindOwner(ecs::handle_of(&card));
59 if (!bound) return failureVoid(bound.code(), "card attribute owner could not be bound", "owner");
60 if (!card.attributes()->values.initialized()) {
61 const std::array<eve::attributes::AttributeSnapshotBase, 2> bases = {
62 eve::attributes::AttributeSnapshotBase{std::string(attackAttribute),
63 static_cast<double>(card.stats()->attack)},
64 eve::attributes::AttributeSnapshotBase{std::string(healthAttribute),
65 static_cast<double>(card.stats()->health)}};
66 auto initialized = card.attributes()->values.initialize(bases);
67 if (!initialized)
68 return failureVoid(initialized.code(), "card attribute state could not be initialized", "attributes");
69 }
70 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
71}

73eve::Result<void> CardAttributeAdapter::project(CardData& card) {
74 auto ready = ensure(card);
75 if (!ready) return failureVoid(ready.code(), "card attributes are not available", "attributes");
76
77 auto attack = card.attributes()->values.getFinal(attackAttribute);
78 if (!attack) return failureVoid(attack.code(), "card attack attribute could not be read", "attack");
79 auto health = card.attributes()->values.getFinal(healthAttribute);
80 if (!health) return failureVoid(health.code(), "card health attribute could not be read", "health");
81 const double values[] = {attack.value(), health.value()};
82 for (const double value : values) {
83 if (!std::isfinite(value) || value < static_cast<double>(std::numeric_limits<int>::min()) ||
84 value > static_cast<double>(std::numeric_limits<int>::max()))
85 return failureVoid(eve::DiagnosticCode::InvariantViolation,
86 "card attribute cannot be represented by the legacy integer projection", "stats");
87 }
88 card.stats()->attack = static_cast<int>(std::llround(attack.value()));
89 card.stats()->health = static_cast<int>(std::llround(health.value()));
90 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
91}

93eve::Result<double> CardAttributeAdapter::read(CardData& card, std::string_view attribute) {
94 if (!selected(attribute))
95 return failure<double>(eve::DiagnosticCode::Unsupported,
96 "card attribute is outside the selective combat projection", "attribute");
97 auto projected = project(card);
98 if (!projected) return failure<double>(projected.code(), "card compatibility projection failed", "stats");
99 auto value = card.attributes()->values.getFinal(attribute);
100 if (!value) return failure<double>(value.code(), "card attribute read failed", "attribute");
101 return eve::Result<double>::success(value.value());
102}
