49Result<resource::CostSpec> AtomicBatch::parseCost(const std::string& json) {
50 auto value = Value::fromJson(json);
51 if (!value) return Result<resource::CostSpec>::failure(value.status());
52 const auto* object = value.value().getIf<Value::Object>();
53 if (!object)
54 return Result<resource::CostSpec>::failure(
55 Diagnostic::error(DiagnosticCode::InvalidArgument, "cost must be an object", "cost"));
56 std::vector<resource::ResourceCost> items;
57 for (const auto& [name, amount] : *object) {
58 const auto* integer = amount.getIf<std::int64_t>();
59 if (!integer)
60 return Result<resource::CostSpec>::failure(
61 Diagnostic::error(DiagnosticCode::InvalidArgument, "cost must be integral", "cost"));
62 auto item = resource::ResourceCost::create(name, *integer);
63 if (!item) return Result<resource::CostSpec>::failure(item.status());
64 items.push_back(std::move(item).takeValue());
65 }
66 return resource::CostSpec::create(std::move(items));
67}

75Result<void> AtomicBatch::protect(resource::ResourceCost floor) {
76 if (impl_->consumed || impl_->executing)
77 return reject(Status::failure(Diagnostic::error(DiagnosticCode::Conflict, "batch is consumed", "batch")));
78 if (impl_->failure) return Result<void>::failure(*impl_->failure);
79 if (!floor.resource.isValid() || floor.amount.isZero())
80 return reject(
81 Status::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "invalid resource floor", "batch")));
82 const auto name = std::string(floor.resource.value());

84 if (found == impl_->floors.end() || found->second.amount.value() < floor.amount.value())
85 impl_->floors.insert_or_assign(name, std::move(floor));
86 return Result<void>::success();

144Result<TransactionReceipt> AtomicBatch::commit() {
145 if (impl_->consumed || impl_->executing)
146 return Result<TransactionReceipt>::failure(
147 Diagnostic::error(DiagnosticCode::Conflict, "batch is consumed", "batch"));
148 // Keep condition queries available through prepare, then release all borrowed roots.
149 impl_->executing = true;
150 auto result = [&]() -> Result<TransactionReceipt> {
151 if (impl_->failure) return Result<TransactionReceipt>::failure(*impl_->failure);
152 for (const auto& guard : impl_->guards) {
153 auto valid = guard();
154 if (!valid) return Result<TransactionReceipt>::failure(valid.status());
155 }
156 if (impl_->effects.empty() && !impl_->cost)
157 return Result<TransactionReceipt>::failure(
158 Diagnostic::error(DiagnosticCode::InvalidArgument, "batch has no effects", "batch"));
159 if (!impl_->floors.empty()) {
160 if (!impl_->account)
161 return Result<TransactionReceipt>::failure(
162 Diagnostic::error(DiagnosticCode::InvalidArgument, "protected batch has no payer", "batch"));
163 auto items = impl_->cost ? impl_->cost->items() : std::vector<resource::ResourceCost>{};
164 for (const auto& [name, floor] : impl_->floors) items.push_back(floor);
165 auto cost = resource::CostSpec::create(std::move(items));
166 if (!cost) return Result<TransactionReceipt>::failure(cost.status());
167 auto affordable = impl_->account->canAfford(cost.value());
168 if (!affordable) return Result<TransactionReceipt>::failure(affordable.status());
169 if (!affordable.value().isAffordable())
170 return Result<TransactionReceipt>::failure(
171 Diagnostic::error(DiagnosticCode::PreconditionViolation,
172 "merged payment would consume protected resources", "batch"));
173 }
174 std::vector<ITransactionParticipant*> effects;
175 for (auto& effect : impl_->effects) effects.push_back(effect.get());
176 TransactionContext context("script.atomic");
177 if (impl_->cost) {
178 if (!impl_->account)
179 return Result<TransactionReceipt>::failure(
180 Diagnostic::error(DiagnosticCode::InvalidArgument, "paid batch has no payer", "batch.account"));
181 return AtomicResourcePayment::execute(context, *impl_->account, *impl_->cost, effects);
182 }
183 return AtomicResourcePayment::execute(context, effects);
184 }();
185 impl_->consumed = true;
186 impl_->executing = false;
187 impl_->query = {};
188 impl_->effects.clear();
189 impl_->guards.clear();
190 impl_->account = nullptr;
191 impl_->accountOwner.reset();
192 return result;
193}
