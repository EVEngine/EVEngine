25 defaultMaterialId.empty() || sources_.size() >= 4096)

27 animation::AnimSkinStreamData data;
28 data.vertexCount = skin.getVertexCount();
29 data.bindPositions.reserve(static_cast<std::size_t>(data.vertexCount) * 3U);
30 for (int vertex = 0; vertex < data.vertexCount; ++vertex) {
31 data.bindPositions.insert(data.bindPositions.end(), {skin.getBindPositionX(vertex),
32 skin.getBindPositionY(vertex),
33 skin.getBindPositionZ(vertex)});
34 for (int influence = 0; influence < animation::AnimSkin::kMaxInfluences; ++influence) {
35 data.vertexSkinJoints.push_back(skin.getVertexSkinJoint(vertex, influence));
36 data.vertexWeights.push_back(skin.getVertexWeight(vertex, influence));
37 }
38 }
39 data.skeletonBones.reserve(static_cast<std::size_t>(skin.getBoneCount()));
40 data.boneNames.reserve(static_cast<std::size_t>(skin.getBoneCount()));
41 data.inverseBindMatrices.reserve(static_cast<std::size_t>(skin.getBoneCount()));
42 for (int joint = 0; joint < skin.getBoneCount(); ++joint) {

44 data.boneNames.push_back(skin.getSkinBoneName(joint));
45 std::array<float, 16> bind{};

47 bind[static_cast<std::size_t>(element)] = skin.getInverseBindElement(joint, element);
48 data.inverseBindMatrices.push_back(bind);
49 }
50 sources_.push_back(Source{mesh, transform, defaultMaterialId, std::move(data)});

54Result<void> combinePcgSkinnedMeshesInto(PcgSkinnedMeshResult& output,
55 const PcgSkinnedMeshPlan& plan) {
56 if (plan.sources_.empty()) return invalid<void>("Pcg skinned combine plan is empty");
57 procgen::PcgMeshCombinePlan meshPlan;
58 for (const auto& source : plan.sources_) {
59 auto appended = meshPlan.appendSource(source.mesh, source.transform, source.defaultMaterialId);
60 if (!appended.ok()) return Result<void>::failure(appended.status());
61 }
62 procgen::MeshBuild combinedMesh;
63 auto combined = procgen::combinePcgStaticMeshesInto(combinedMesh, meshPlan);
64 if (!combined.ok()) return Result<void>::failure(combined.status());
65
66 animation::AnimSkinStreamData streams;
67 streams.vertexCount = combinedMesh.getVertexCount();
68 streams.bindPositions = combinedMesh.positions();
69 streams.bindNormals = combinedMesh.normals();
70 streams.vertexSkinJoints.reserve(static_cast<std::size_t>(streams.vertexCount) * 4U);
71 streams.vertexWeights.reserve(static_cast<std::size_t>(streams.vertexCount) * 4U);
72
73 bool seeded = false;
74 for (const auto& source : plan.sources_) {
75 const auto& skin = source.skin;
76 std::vector<int> remap(skin.skeletonBones.size(), -1);
77 if (!seeded) {
78 streams.skeletonBones = skin.skeletonBones;
79 streams.boneNames = skin.boneNames;
80 streams.inverseBindMatrices = skin.inverseBindMatrices;
81 seeded = true;
82 }
83 for (std::size_t joint = 0; joint < skin.skeletonBones.size(); ++joint) {
84 int firstIdentity = -1;
85 for (std::size_t used = 0; used < streams.skeletonBones.size(); ++used)
86 if (streams.skeletonBones[used] == skin.skeletonBones[joint]) {
87 firstIdentity = static_cast<int>(used);
88 break;
89 }
90 if (firstIdentity < 0 || !sameBind(skin.inverseBindMatrices[joint],
91 streams.inverseBindMatrices[static_cast<std::size_t>(firstIdentity)])) {
92 firstIdentity = static_cast<int>(streams.skeletonBones.size());
93 streams.skeletonBones.push_back(skin.skeletonBones[joint]);
94 streams.boneNames.push_back(skin.boneNames[joint]);
95 streams.inverseBindMatrices.push_back(skin.inverseBindMatrices[joint]);
96 }
97 remap[joint] = firstIdentity;
98 }
99 for (std::size_t influence = 0; influence < skin.vertexWeights.size(); ++influence) {
100 const float weight = skin.vertexWeights[influence];
101 const int sourceJoint = skin.vertexSkinJoints[influence];
102 if (weight > 0.F && (sourceJoint < 0 || static_cast<std::size_t>(sourceJoint) >= remap.size()))
103 return invalid<void>("Pcg skinned source influence references an invalid joint");
104 streams.vertexSkinJoints.push_back(weight > 0.F ? remap[static_cast<std::size_t>(sourceJoint)] : -1);
105 streams.vertexWeights.push_back(weight);
106 }
107 }
108 auto skin = animation::AnimSkin::fromStreams(std::move(streams));
109 if (!skin.ok()) return Result<void>::failure(skin.status());
110 PcgSkinnedMeshResult candidate;
111 candidate.mesh_ = std::move(combinedMesh);
112 candidate.skin_ = std::move(skin).takeValue();
113 output = std::move(candidate);
114 return Result<void>::success();
115}
