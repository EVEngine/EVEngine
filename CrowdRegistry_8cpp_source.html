10std::uint64_t allocateOwner() {
11 static std::atomic<std::uint64_t> next{1};
12 auto value = next.load(std::memory_order_relaxed);
13 for (;;) {
14 if (value >= static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max()))
15 throw eve::Exception("Crowd owner identity exhausted");
16 if (next.compare_exchange_weak(value, value + 1, std::memory_order_relaxed)) return value;
17 }
18}
