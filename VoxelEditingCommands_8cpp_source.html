23bool readInt(const editing::Value& value, const char* key, int& output, bool required) {
24 const editing::Value* entry = field(value, key);
25 if (!entry) return !required;
26 if (const auto* integer = entry->getIf<int64_t>()) {
27 output = static_cast<int>(*integer);
28 return true;
29 }
30 if (const auto* real = entry->getIf<double>()) {
31 output = static_cast<int>(*real);
32 return true;
33 }
34 return false;

92editing::Result<void> registerEditingCommands(editing::IEditingCommandRegistry& registry) {
93 auto registered = addCommand(
94 registry, "voxel.model.create.v1", "Create voxel model",
95 [](editing::IEditableTarget& target, const editing::CommandRequest& request) {
96 auto* catalog = asCatalog(target);
97 const std::string* id = stringField(request.payload, "id");
98 if (!catalog || !id || id->empty())
99 return error<editing::CommandPlan>(editing::Status::Rejected, "voxel.editing.model-payload",
100 "Voxel model create requires id");
101 const std::string* name = stringField(request.payload, "name");
102 VoxelModelValue model;
103 model.id = editing::ObjectId(*id);
104 model.name = name && !name->empty() ? *name : *id;
105 if (!readInt(request.payload, "sizeX", model.sizeX, false) ||
106 !readInt(request.payload, "sizeY", model.sizeY, false) ||
107 !readInt(request.payload, "sizeZ", model.sizeZ, false))
108 return error<editing::CommandPlan>(editing::Status::Rejected, "voxel.editing.size",
109 "Voxel model size must be integers");
110 return planFrom(catalog->makeCreateModel(model), editing::Value::Object{{"id", *id}});
111 });
112 if (!registered.ok()) return registered;
113
114 registered = addCommand(
115 registry, "voxel.model.delete.v1", "Delete voxel model",

153 return error<editing::CommandPlan>(editing::Status::Rejected, "voxel.editing.property-payload",
154 "Voxel property requires item, path and value");

156 editing::PropertySetMode::Absolute),
157 editing::Value::Object{{"item", *item}, {"path", *path}});
158 });
