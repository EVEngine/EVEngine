12const editing::Value* field(const editing::Value& value, const char* key) {
13 const auto* object = value.getIf<editing::Value::Object>();

24bool readInt(const editing::Value& value, const char* key, int& output, bool required) {
25 const editing::Value* entry = field(value, key);
26 if (!entry) return !required;
27 if (const auto* integer = entry->getIf<int64_t>()) {
28 output = static_cast<int>(*integer);
29 return true;
30 }
31 if (const auto* real = entry->getIf<double>()) {
32 output = static_cast<int>(*real);
33 return true;
34 }
35 return false;
36}

38bool readBool(const editing::Value& value, const char* key, bool& output, bool required) {

157editing::Result<void> registerEditingCommands(editing::IEditingCommandRegistry& registry) {
158 auto descriptors = voxelEditingCommandDescriptors();
159 auto registered = registry.registerPlannedCommand(
160 descriptors[0], [](editing::IEditableTarget& target, const editing::CommandRequest& request) {
161 auto* catalog = asCatalog(target);
162 const std::string* id = stringField(request.payload, "id");
163 if (!catalog || !id || id->empty())
164 return eve::editing::failed<editing::CommandPlan>(editing::Status::Rejected,
165 editing::RuleId("voxel.editing.model-payload"),
166 "Voxel model create requires id");
167 const std::string* name = stringField(request.payload, "name");
168 VoxelModelValue model;
169 model.id = editing::ObjectId(*id);
170 model.name = name && !name->empty() ? *name : *id;
171 if (!readInt(request.payload, "sizeX", model.sizeX, false) ||
172 !readInt(request.payload, "sizeY", model.sizeY, false) ||
173 !readInt(request.payload, "sizeZ", model.sizeZ, false))
174 return eve::editing::failed<editing::CommandPlan>(editing::Status::Rejected,
175 editing::RuleId("voxel.editing.size"),
176 "Voxel model size must be integers");
177 return planFrom(catalog->makeCreateModel(model), editing::Value::Object{{"id", *id}});
178 });
179 if (!registered.ok()) return registered;
180
181 registered = registry.registerPlannedCommand(
182 descriptors[1], [](editing::IEditableTarget& target, const editing::CommandRequest& request) {
183 auto* catalog = asCatalog(target);
184 const std::string* id = stringField(request.payload, "id");
185 if (!catalog || !id || id->empty())
186 return eve::editing::failed<editing::CommandPlan>(editing::Status::Rejected,
187 editing::RuleId("voxel.editing.model-payload"),
188 "Voxel model delete requires id");
189 return planFrom(catalog->makeDeleteModel(editing::ObjectId(*id)), editing::Value::Object{{"id", *id}});
190 });
191 if (!registered.ok()) return registered;
192
193 registered = registry.registerPlannedCommand(
194 descriptors[2], [](editing::IEditableTarget& target, const editing::CommandRequest& request) {
195 auto* catalog = asCatalog(target);
196 const std::string* item = stringField(request.payload, "item");
197 int x = 0, y = 0, z = 0;
198 bool occupied = true;
199 if (!catalog || !item || item->empty() || !readInt(request.payload, "x", x, true) ||
200 !readInt(request.payload, "y", y, true) || !readInt(request.payload, "z", z, true) ||
201 !readBool(request.payload, "occupied", occupied, true))
202 return eve::editing::failed<editing::CommandPlan>(
203 editing::Status::Rejected, editing::RuleId("voxel.editing.voxel-payload"),
204 "Voxel occupancy requires item, x, y, z and occupied");
205 return planFrom(catalog->makeSetVoxel(editing::ObjectId(*item), x, y, z, occupied),
206 editing::Value::Object{{"item", *item}});
207 });
208 if (!registered.ok()) return registered;
209
210 return registry.registerPlannedCommand(
211 descriptors[3], [](editing::IEditableTarget& target, const editing::CommandRequest& request) {
212 auto* catalog = asCatalog(target);
213 auto* properties = static_cast<editing::IPropertyProvider*>(
214 target.queryCapability(VoxelCatalogTarget::propertyCapabilityId()));
215 const std::string* item = stringField(request.payload, "item");
216 const std::string* path = stringField(request.payload, "path");
217 const editing::Value* value = field(request.payload, "value");
218 if (!catalog || !properties || !item || item->empty() || !path || !value)
219 return eve::editing::failed<editing::CommandPlan>(editing::Status::Rejected,
220 editing::RuleId("voxel.editing.property-payload"),
221 "Voxel property requires item, path and value");
222 return planFrom(properties->makeSet(selectionFor(*catalog, *item), editing::PropertyPath(*path), *value,
223 editing::PropertySetMode::Absolute),
224 editing::Value::Object{{"item", *item}, {"path", *path}});
225 });
226}
