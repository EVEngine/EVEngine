11EditorResult<ui::NodeType> nodeType(const std::string& type) {
12 static const std::map<std::string, ui::NodeType> types{
13 {"window", ui::NodeType::Window}, {"text", ui::NodeType::Text},
14 {"button", ui::NodeType::Button}, {"group", ui::NodeType::Group},
15 {"checkbox", ui::NodeType::Checkbox}, {"slider", ui::NodeType::Slider},
16 {"input", ui::NodeType::InputText}, {"flex", ui::NodeType::Flex},
17 {"image", ui::NodeType::Image}, {"image-button", ui::NodeType::ImageButton},
18 {"viewport", ui::NodeType::Viewport}, {"card", ui::NodeType::Card},
19 {"toolbar", ui::NodeType::Toolbar}, {"sidebar", ui::NodeType::Sidebar},
20 {"split-pane", ui::NodeType::SplitPane}, {"nine-patch", ui::NodeType::NinePatchPanel}};
21 const auto found = types.find(type);
22 if (found == types.end())
23 return eve::editing::failed<ui::NodeType>(EditorStatus::Unsupported,
24 RuleId("editor.ui.unsupported-runtime-widget"),
25 "Runtime UI does not support widget type: " + type);
26 return eve::editing::applied<ui::NodeType>(found->second);
27}
