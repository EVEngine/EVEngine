21Result<void> stageLayout(spirv_cross::Compiler& compiler, bool vertex, bool mesh) {
22 for (const auto capability : compiler.get_declared_capabilities())
23 if (capability != spv::CapabilityShader && capability != spv::CapabilityMatrix)
24 return failure(DiagnosticCode::Unsupported,
25 "Shader requires device features outside the baseline contract");
26 if (!compiler.get_declared_extensions().empty())
27 return failure(DiagnosticCode::Unsupported, "Shader extensions are outside the baseline contract");
28 const auto resources = compiler.get_shader_resources();
29 if (!resources.storage_buffers.empty() || !resources.storage_images.empty() || !resources.subpass_inputs.empty() ||
30 !resources.separate_images.empty() || !resources.separate_samplers.empty() ||
31 !resources.atomic_counters.empty() || !resources.acceleration_structures.empty() ||
32 !compiler.get_specialization_constants().empty())
33 return failure(DiagnosticCode::Unsupported, "Shader uses unsupported resources or specialization constants");
34 if (resources.sampled_images.size() > 1 || (vertex && !resources.sampled_images.empty()))
35 return failure(DiagnosticCode::Unsupported, "Only one fragment albedo sampler is supported");
36 for (const auto& sampler : resources.sampled_images) {
37 const auto& type = compiler.get_type(sampler.type_id);
38 if (compiler.get_decoration(sampler.id, spv::DecorationDescriptorSet) != 0 ||
39 compiler.get_decoration(sampler.id, spv::DecorationBinding) != (mesh ? 1u : 0u) || !type.array.empty() ||
40 type.image.dim != spv::Dim2D || type.image.arrayed || type.image.ms || type.image.depth ||
41 compiler.get_type(type.image.type).basetype != spirv_cross::SPIRType::Float)
42 return failure(DiagnosticCode::Unsupported, "Albedo must use the engine's single float sampler2D slot");
43 }
44 if (resources.uniform_buffers.size() > 1 || (!mesh && !resources.uniform_buffers.empty()))
45 return failure(DiagnosticCode::Unsupported, "Shader uses unsupported uniform buffers");
46 for (const auto& buffer : resources.uniform_buffers) {
47 const auto& type = compiler.get_type(buffer.base_type_id);
48 if (compiler.get_decoration(buffer.id, spv::DecorationDescriptorSet) != 0 ||
49 compiler.get_decoration(buffer.id, spv::DecorationBinding) != 0 ||
50 !compiler.get_type(buffer.type_id).array.empty() || type.member_types.empty() ||
51 type.member_types.size() > 6)
52 return failure(DiagnosticCode::Unsupported, "Mesh Frame buffer does not match the supported prefix");
53 for (unsigned i = 0; i < type.member_types.size(); ++i) {
54 const auto& member = compiler.get_type(type.member_types[i]);
55 const auto offset = i < 2 ? i * 64 : 128 + (i - 2) * 16;
56 if (compiler.type_struct_member_offset(type, i) != offset)
57 return failure(DiagnosticCode::Unsupported, "Mesh Frame member offset mismatch");
58 if (i < 2) {
59 if (member.basetype != spirv_cross::SPIRType::Float || member.width != 32 || member.columns != 4 ||
60 member.vecsize != 4 || !member.array.empty() ||
61 compiler.type_struct_member_matrix_stride(type, i) != 16 ||
62 compiler.has_member_decoration(type.self, i, spv::DecorationRowMajor))
63 return failure(DiagnosticCode::Unsupported, "Mesh Frame matrices require column-major mat4");
64 } else if (!floatType(member, 4))
65 return failure(DiagnosticCode::Unsupported, "Mesh Frame requires vec4 fields");
66 }
67 }
68 if (resources.push_constant_buffers.size() > 1)
69 return failure(DiagnosticCode::Unsupported, "Multiple push-constant blocks are unsupported");
70 for (const auto& buffer : resources.push_constant_buffers) {
71 const auto& type = compiler.get_type(buffer.base_type_id);
72 if (type.member_types.size() != 1 || compiler.type_struct_member_offset(type, 0) != 0)
73 return failure(DiagnosticCode::Unsupported, "Push constants require one float data[32] member");
74 const auto& member = compiler.get_type(type.member_types[0]);
75 if (member.basetype != spirv_cross::SPIRType::Float || member.width != 32 || member.vecsize != 1 ||
76 member.columns != 1 || member.array.size() != 1 || !member.array_size_literal[0] || member.array[0] != 32 ||
77 compiler.type_struct_member_array_stride(type, 0) != 4)
78 return failure(DiagnosticCode::Unsupported, "Push constants require tightly packed float data[32]");
79 }
80 if (vertex) {
81 for (const auto& input : resources.stage_inputs) {
82 const auto location = compiler.get_decoration(input.id, spv::DecorationLocation);
83 const unsigned sizes[] = {mesh ? 3u : 2u, mesh ? 3u : 4u, 2u};
84 if (location > 2 || compiler.get_decoration(input.id, spv::DecorationComponent) != 0 ||
85 !floatType(compiler.get_type(input.type_id), sizes[location]))
86 return failure(DiagnosticCode::Unsupported, "Vertex attributes do not match the engine layout");
87 }
88 } else {
89 if (resources.stage_outputs.size() != 1 ||
90 compiler.get_decoration(resources.stage_outputs[0].id, spv::DecorationLocation) != 0 ||
91 compiler.get_decoration(resources.stage_outputs[0].id, spv::DecorationIndex) != 0 ||
92 compiler.get_decoration(resources.stage_outputs[0].id, spv::DecorationComponent) != 0 ||
93 !floatType(compiler.get_type(resources.stage_outputs[0].type_id), 4))
94 return failure(DiagnosticCode::Unsupported, "Fragment output must be location zero vec4");
95 }
96 return Result<void>::success();
97}

101#ifndef EVENGINE_SPIRV_ASSET_VALIDATION
102 (void)shader;
103 return failure(DiagnosticCode::Unsupported, "Shader asset upload requires SPIRV-Tools and SPIRV-Cross");
104#else
105 try {
106 spvtools::SpirvTools validator(SPV_ENV_VULKAN_1_2);
107 std::string diagnostics;
108 validator.SetMessageConsumer([&](spv_message_level_t, const char*, const spv_position_t&, const char* text) {
109 if (diagnostics.size() < 4096) diagnostics += text;
110 });
111 if (!validator.Validate(shader.vertex) || !validator.Validate(shader.fragment))
112 return failure(DiagnosticCode::ParseError, "SPIR-V validation failed: " + diagnostics);
113 spirv_cross::Compiler vertex(shader.vertex), fragment(shader.fragment);
114 vertex.set_entry_point("main", spv::ExecutionModelVertex);
115 fragment.set_entry_point("main", spv::ExecutionModelFragment);
116 const bool mesh = shader.interface == asset::ShaderAssetInterface::Mesh3D;
117 auto v = stageLayout(vertex, true, mesh);
118 if (!v) return v;
119 auto f = stageLayout(fragment, false, mesh);
120 if (!f) return f;
121 const auto outputs = vertex.get_shader_resources().stage_outputs;
122 for (const auto& input : fragment.get_shader_resources().stage_inputs) {
123 const auto location = fragment.get_decoration(input.id, spv::DecorationLocation);
124 bool matched = false;
125 const auto& type = fragment.get_type(input.type_id);
126 if (!floatType(type, type.vecsize) || fragment.get_decoration(input.id, spv::DecorationComponent) != 0)
127 return failure(DiagnosticCode::Unsupported, "Fragment varyings must be float vectors");
128 for (const auto& output : outputs)
129 if (vertex.get_decoration(output.id, spv::DecorationLocation) == location &&
130 vertex.get_decoration(output.id, spv::DecorationComponent) == 0 &&
131 floatType(vertex.get_type(output.type_id), type.vecsize))
132 matched = true;
133 if (!matched) return failure(DiagnosticCode::Unsupported, "Vertex/fragment varying mismatch");
134 }
135 return Result<void>::success();
136 } catch (const std::exception& error) {
137 return failure(DiagnosticCode::ParseError, error.what());
