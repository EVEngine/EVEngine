25double detailColorNoise(double u, double v, float spread, int32_t seed) {
26 if (spread == 0) return 0.5;
27 const double frequency = 1.0 + 31.0 * double(spread);
28 const double x = u * frequency, z = v * frequency;
29 const int64_t x0 = int64_t(std::floor(x)), z0 = int64_t(std::floor(z));
30 const double tx = smooth(x - x0), tz = smooth(z - z0);
31 const double a = std::lerp(lattice(x0, z0, seed), lattice(x0 + 1, z0, seed), tx);
32 const double b = std::lerp(lattice(x0, z0 + 1, seed), lattice(x0 + 1, z0 + 1, seed), tx);
33 return std::lerp(a, b, tz);
34}
