69Result<void> Mesh::validateTangentFrame(std::span<const float> tangents, std::span<const float> bitangents) const {
70 auto invalid = [] {
71 return Result<void>::failure(
72 Diagnostic::error(DiagnosticCode::InvalidArgument,
73 "tangent frame must contain finite independent XYZ vectors for every mesh vertex"));
74 };
75 if (tangents.empty() && bitangents.empty()) {
76 return Result<void>::success();
77 }
78 if (gpuVertexCount <= 0 || tangents.size() != size_t(gpuVertexCount) * 3 || bitangents.size() != tangents.size())
79 return invalid();
80 for (size_t i = 0; i < tangents.size(); i += 3) {
81 double tt = 0, bb = 0, tb = 0;
82 for (unsigned c = 0; c < 3; ++c) {
83 const double t = tangents[i + c], b = bitangents[i + c];
84 if (!std::isfinite(t) || !std::isfinite(b)) return invalid();
85 tt += t * t;
86 bb += b * b;
87 tb += t * b;
88 }
89 if (tt <= 1e-20 || bb <= 1e-20 || tt * bb - tb * tb <= tt * bb * 1e-12) return invalid();
90 }
91 return Result<void>::success();
92}
