37eve::Result<CompareOperator> parseOperator(const eve::Value& value) {
38 const auto* text = value.getIf<std::string>();

44 {"eq", CompareOperator::Equal}, {"ne", CompareOperator::NotEqual}, {"lt", CompareOperator::Less},
45 {"le", CompareOperator::LessEqual}, {"gt", CompareOperator::Greater}, {"ge", CompareOperator::GreaterEqual},

87 const auto* value = field(object, "children");
88 const auto* array = value == nullptr ? nullptr : value->getIf<eve::Value::Array>();
89 if (array == nullptr)

91 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "condition children must be an array",
92 "children", {}, "decision.condition_codec"));
93 std::vector<Condition> result;
94 result.reserve(array->size());
95 for (const auto& child : *array) {
96 auto decoded = decodeNode(child);
97 if (!decoded) return eve::Result<std::vector<Condition>>::failure(decoded.status());
98 result.push_back(std::move(decoded).takeValue());
99 }
100 return eve::Result<std::vector<Condition>>::success(std::move(result));

103eve::Result<Condition> decodeLegacyLeaf(std::string_view kind, const eve::Value::Object& object) {
104 if (kind == "has_tag" || kind == "has_attribute" || kind == "has_resource" || kind == "authority_check") {
105 if (!exactFields(object, {"kind", "key"}))
106 return eve::Result<Condition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
107 "leaf condition contains an unknown field",
108 {}, {}, "decision.condition_codec"));
109 auto key = requiredString(object, "key");
110 if (!key) return eve::Result<Condition>::failure(key.status());
111 auto name = std::move(key).takeValue();
112 if (kind == "has_tag") return eve::Result<Condition>::success(tagPresent(std::move(name)));
113 if (kind == "has_attribute") return eve::Result<Condition>::success(attributeExists(std::move(name)));
114 if (kind == "has_resource") return eve::Result<Condition>::success(resourceExists(std::move(name)));
115 return eve::Result<Condition>::success(authorityAllow(std::move(name)));
116 }
117 if (kind == "state_equals") {
118 if (!exactFields(object, {"kind", "key", "expected"}))
119 return eve::Result<Condition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
120 "state_equals contains an unknown field", {},
121 {}, "decision.condition_codec"));
122 auto key = requiredString(object, "key");
123 if (!key) return eve::Result<Condition>::failure(key.status());
124 const auto* expected = field(object, "expected");
125 if (expected == nullptr)
126 return eve::Result<Condition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
127 "state_equals requires expected", "expected",
128 {}, "decision.condition_codec"));
129 return eve::Result<Condition>::success(stateEquals(std::move(key).takeValue(), *expected));
130 }
131 return eve::Result<Condition>::failure(eve::Diagnostic::error(
132 eve::DiagnosticCode::InvalidArgument, "unknown condition kind", "kind", {}, "decision.condition_codec"));
133}

135eve::Result<Condition> decodeNode(const eve::Value& value) {
136 const auto* object = value.getIf<eve::Value::Object>();
137 if (object == nullptr)
138 return eve::Result<Condition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
139 "condition node must be an object", {}, {},
140 "decision.condition_codec"));
141 const auto* kindValue = field(*object, "kind");
142 const auto* kindText = kindValue == nullptr ? nullptr : kindValue->getIf<std::string>();
143 if (kindText == nullptr || kindText->empty())
144 return eve::Result<Condition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
145 "condition node requires kind", "kind", {},
146 "decision.condition_codec"));
147 const std::string& kind = *kindText;
148 if (kind == "all" || kind == "any" || kind == "not") {
149 if (!exactFields(*object, {"kind", "children"}))
150 return eve::Result<Condition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
151 "logical condition contains an unknown field",
152 {}, {}, "decision.condition_codec"));
153 auto children = decodeChildren(*object);
154 if (!children) return eve::Result<Condition>::failure(children.status());
155 auto values = std::move(children).takeValue();
156 if (kind == "not" && values.size() != 1)
157 return eve::Result<Condition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
158 "not condition requires one child",
159 "children", {}, "decision.condition_codec"));
160 if (kind == "all") return eve::Result<Condition>::success(Condition::all(std::move(values)));
161 if (kind == "any") return eve::Result<Condition>::success(Condition::any(std::move(values)));
162 return eve::Result<Condition>::success(Condition::not_(std::move(values.front())));
163 }
164 if (kind == "compare") {
165 if (!exactFields(*object, {"kind", "key", "operator", "expected"}) &&
166 !exactFields(*object, {"kind", "key", "operator"}))
167 return eve::Result<Condition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
168 "compare condition contains an unknown field",
169 {}, {}, "decision.condition_codec"));
170 auto key = requiredString(*object, "key");
171 if (!key) return eve::Result<Condition>::failure(key.status());
172 const auto* opValue = field(*object, "operator");
173 if (opValue == nullptr)
174 return eve::Result<Condition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
175 "compare requires operator", "operator", {},
176 "decision.condition_codec"));
177 auto op = parseOperator(*opValue);
178 if (!op) return eve::Result<Condition>::failure(op.status());
179 eve::Value expected;
180 const auto* expectedField = field(*object, "expected");
181 if (op.value() == CompareOperator::Exists) {
182 if (expectedField != nullptr) expected = *expectedField;
183 } else {
184 if (expectedField == nullptr)
185 return eve::Result<Condition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
186 "compare requires expected", "expected",
187 {}, "decision.condition_codec"));
188 expected = *expectedField;
189 }
190 return eve::Result<Condition>::success(
191 Condition::compare(std::move(key).takeValue(), op.value(), std::move(expected)));
192 }
193 if (kind == "policy_call") {
194 if (!exactFields(*object, {"kind", "key", "arguments", "scriptDeclaration", "script"}))
195 return eve::Result<Condition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
196 "policy_call contains an unknown field", {},
197 {}, "decision.condition_codec"));
198 auto key = requiredString(*object, "key");
199 if (!key) return eve::Result<Condition>::failure(key.status());
200 eve::Value arguments = eve::Value::Object{};
201 if (const auto* args = field(*object, "arguments")) arguments = *args;
202 std::optional<ScriptConditionDeclaration> declaration;
203 const eve::Value* declValue = field(*object, "scriptDeclaration");
204 if (declValue == nullptr) declValue = field(*object, "script");
205 if (declValue != nullptr) {
206 const auto* declObject = declValue->getIf<eve::Value::Object>();
207 if (declObject == nullptr)
208 return eve::Result<Condition>::failure(
209 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "scriptDeclaration must be an object",
210 "scriptDeclaration", {}, "decision.condition_codec"));
211 ScriptConditionDeclaration parsed;
212 auto name = requiredString(*declObject, "name");
213 if (!name) return eve::Result<Condition>::failure(name.status());
214 parsed.name = std::move(name).takeValue();
215 if (const auto* deps = field(*declObject, "dependencies")) {
216 const auto* array = deps->getIf<eve::Value::Array>();
217 if (array == nullptr)
218 return eve::Result<Condition>::failure(
219 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "dependencies must be an array",
220 "scriptDeclaration.dependencies", {}, "decision.condition_codec"));
221 for (const auto& item : *array) {
222 const auto* text = item.getIf<std::string>();
223 if (text == nullptr || text->empty())
224 return eve::Result<Condition>::failure(eve::Diagnostic::error(
225 eve::DiagnosticCode::InvalidArgument, "dependency must be a non-empty string",
226 "scriptDeclaration.dependencies", {}, "decision.condition_codec"));
227 parsed.dependencies.push_back(*text);
228 }
229 }
230 if (const auto* det = field(*declObject, "determinism")) {
231 if (det->isString()) {
232 auto level = parseDeterminism(*det);
233 if (!level) return eve::Result<Condition>::failure(level.status());
234 parsed.determinism = level.value();
235 } else if (det->isInt64()) {
236 const auto raw = det->asInt();
237 if (raw < 0 || raw > static_cast<std::int64_t>(DeterminismLevel::ExplicitlyNondeterministic))
238 return eve::Result<Condition>::failure(eve::Diagnostic::error(
239 eve::DiagnosticCode::InvalidArgument, "unknown condition determinism level",
240 "scriptDeclaration.determinism", {}, "decision.condition_codec"));
241 parsed.determinism = static_cast<DeterminismLevel>(raw);
242 } else {
243 return eve::Result<Condition>::failure(eve::Diagnostic::error(
244 eve::DiagnosticCode::InvalidArgument, "determinism must be a string or integer",
245 "scriptDeclaration.determinism", {}, "decision.condition_codec"));
246 }
247 }
248 declaration = std::move(parsed);
249 }
250 return eve::Result<Condition>::success(
251 Condition::policyCall(std::move(key).takeValue(), std::move(arguments), std::move(declaration)));
252 }
253 return decodeLegacyLeaf(kind, *object);
254}

258eve::Result<eve::Value> encodeCondition(const Condition& condition) {
259 if (!condition.isValid())
260 return eve::Result<eve::Value>::failure(eve::Diagnostic::error(
261 eve::DiagnosticCode::InvalidArgument, "condition is invalid", {}, {}, "decision.condition_codec"));
262 eve::Value::Object object{{"kind", eve::Value(conditionKindName(condition.kind()))}};
263 if (condition.kind() == ConditionKind::All || condition.kind() == ConditionKind::Any ||
264 condition.kind() == ConditionKind::Not) {
265 eve::Value::Array children;
266 for (const auto& child : condition.children()) {
267 auto encoded = encodeCondition(child);
268 if (!encoded) return eve::Result<eve::Value>::failure(encoded.status());
269 children.push_back(std::move(encoded).takeValue());
270 }
271 object.emplace("children", eve::Value(std::move(children)));
272 } else if (condition.kind() == ConditionKind::Compare) {
273 object.emplace("key", eve::Value(condition.key()));
274 object.emplace("operator", eve::Value(compareOperatorName(condition.compareOperator())));
275 if (condition.compareOperator() != CompareOperator::Exists || !condition.expected().isNull())
276 object.emplace("expected", condition.expected());
277 } else if (condition.kind() == ConditionKind::PolicyCall) {
278 object.emplace("key", eve::Value(condition.key()));
279 object.emplace("arguments", condition.arguments());
280 if (condition.scriptDeclaration()) {
281 const auto& declaration = *condition.scriptDeclaration();
282 eve::Value::Array dependencies;
283 for (const auto& dependency : declaration.dependencies) dependencies.emplace_back(dependency);
284 object.emplace(
285 "scriptDeclaration",
286 eve::Value(eve::Value::Object{{"name", eve::Value(declaration.name)},
287 {"dependencies", eve::Value(std::move(dependencies))},
288 {"determinism", eve::Value(determinismName(declaration.determinism))}}));
289 }
290 } else {
291 return eve::Result<eve::Value>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
292 "unsupported condition kind", "kind", {},
293 "decision.condition_codec"));
294 }
295 return eve::Result<eve::Value>::success(eve::Value(std::move(object)));
296}
