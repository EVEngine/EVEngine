20template <class T>
21ResultRef<T> argument(const ssq::Object& object) {
22 if (object.getType() == ssq::Type::INSTANCE && object.getTypeTag() == script::detail::squirrelTypeHash<T*>()) {
23 try {
24 if (auto* value = object.to<T*>()) return ResultRef<T>::success(std::ref(*value));
25 } catch (const ssq::TypeException&) { /* Invalid native instances fail before target mutation. */
26 }
27 }
28 return ResultRef<T>::failure(invalid("null or wrong native player/target type"));
29}
30bool isVmOwnedPlayer(HSQUIRRELVM vm, const ssq::Object& object) {
31 sq_pushobject(vm, object.getRaw());
32 const bool owned = sq_getreleasehook(vm, -1) != nullptr;
33 sq_pop(vm, 1);
34 return owned;
35}

114Result<std::size_t> drawParticles(ScriptParticlePresentation& self, const ssq::Object& graphicsObject,
115 const ssq::Object& materialObjects) {
116 auto graphics = argument<graphics::Graphics>(graphicsObject);
117 if (!graphics) return Result<std::size_t>::failure(graphics.status());
118 if (materialObjects.getType() != ssq::Type::ARRAY)
119 return Result<std::size_t>::failure(invalid("expected an array of live material sprites"));
120 struct Clear {
121 ScriptParticlePresentation& self;
122 ~Clear() {
123 self.bindings.clear();
124 self.templates.clear();
125 }
126 } clear{self};
127 auto materials = materialObjects.toArray();
128 if (materials.size() != self.identities.size())
129 return Result<std::size_t>::failure(invalid("material count must match prepared emitter identities"));
130 for (std::size_t i = 0; i < self.identities.size(); ++i) {
131 auto target = argument<graphics::Renderable2D>(materials.get<ssq::Object>(i));
132 if (!target) return Result<std::size_t>::failure(target.status());
133 auto sprite = target.value().get().sprite();
134 graphics::DrawItem2D material;
135 material.texture = sprite->texture;
136 material.normal = sprite->normalTexture;
137 material.shader = sprite->shader;
138 material.canvas = sprite->canvas;
139 material.camera = sprite->camera;
140 material.layer = sprite->layer;
141 material.blend = sprite->blend;
142 material.receiveLight = sprite->receiveLight;
143 material.litPath = sprite->receiveLight && sprite->normalTexture && sprite->texture && !sprite->shader;
144 self.templates.push_back(material);
145 self.bindings.push_back({self.identities[i], self.templates.back()});
146 }
147 return self.presentation->draw(self.player(), graphics.value().get(), self.bindings);
148}

162ssq::Table prepareParticles(HSQUIRRELVM vm, const ssq::Object& playerObject, const ssq::Object& identitiesObject) {
163 auto player = argument<BakedParticlePlayback>(playerObject);
164 if (!player) return script::projectStatusResult(vm, player.status());
165 if (!isVmOwnedPlayer(vm, playerObject))
166 return script::projectStatusResult(
167 vm, Status::failure(invalid("presentation requires a VM-owned player; use prepareInstance")));
168 if (identitiesObject.getType() != ssq::Type::ARRAY)
169 return script::projectStatusResult(vm, Status::failure(invalid("expected emitter UUID array")));
170 try {
171 auto identities = identitiesObject.toArray();
172 if (identities.size() > 65536)
173 return script::projectStatusResult(vm, Status::failure(invalid("emitter identity budget exceeded")));
174 auto result = std::make_unique<ScriptParticlePresentation>();
175 result->identities.reserve(identities.size());
176 for (std::size_t i = 0; i < identities.size(); ++i) {
177 auto value = identities.get<ssq::Object>(i);
178 if (value.getType() != ssq::Type::STRING)
179 return script::projectStatusResult(vm, Status::failure(invalid("emitter UUID must be a string")));
180 const auto id = PersistentId::parse(value.to<std::string>());
181 if (!id || id->isNil() || (!result->identities.empty() && !(result->identities.back() < *id)))
182 return script::projectStatusResult(
183 vm, Status::failure(invalid("emitter UUIDs must be sorted, unique and non-nil")));
184 result->identities.push_back(*id);
185 }
186 auto prepared = ParticleBakePresentation::prepare(player.value().get().track());
187 if (!prepared) return script::projectStatusResult(vm, prepared.status());
188 result->playerOwner = playerObject;
189 result->presentation = std::move(prepared).takeValue();
190 result->templates.reserve(result->identities.size());
191 result->bindings.reserve(result->identities.size());
192 auto object = script::makeOwnedSquirrelInstance<ScriptParticlePresentation>(vm, std::move(result));
193 if (!object) return script::projectStatusResult(vm, object.status());
194 return script::projectStatusResult(vm, Status::success(), std::move(object).takeValue());
195 } catch (const std::bad_alloc&) {
196 return script::projectStatusResult(
197 vm, Status::failure(
198 Diagnostic::error(DiagnosticCode::Failed, "particle script presentation allocation failed")));
199 }
200}
