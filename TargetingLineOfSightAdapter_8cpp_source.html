20eve::Result<void> TargetingLineOfSightAdapter::addWorld(World3D* world) {
21 if (!world)
22 return failure<void>(eve::DiagnosticCode::InvalidArgument, "Physics LOS cannot register a null World3D");
23 if (!world->isValid())
24 return failure<void>(eve::DiagnosticCode::NotFound, "Physics LOS cannot register an invalid World3D");
25
26 for (auto it = worlds_.begin(); it != worlds_.end();) {
27 auto lifetime = it->lifetime.lock();
28 if (!lifetime) {
29 it = worlds_.erase(it);
30 continue;
31 }
32 if (it->world == world) return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
33 return failure<void>(eve::DiagnosticCode::Conflict, "Physics LOS already has a different active World3D");
34 }
35 worlds_.push_back({world, world->queryLifetime_});
36 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
37}

47eve::Result<sensing::LineOfSightResult> TargetingLineOfSightAdapter::query(const sensing::TargetLocation& from,
48 const sensing::TargetLocation& to) const {
49 const auto* fromPoint = std::get_if<sensing::WorldPoint>(&from);
50 const auto* toPoint = std::get_if<sensing::WorldPoint>(&to);
51 if (!fromPoint || !toPoint || !fromPoint->isValid() || !toPoint->isValid() ||
52 fromPoint->space() != sensing::CoordinateSpace::World3D ||
53 toPoint->space() != sensing::CoordinateSpace::World3D)
54 return failure<sensing::LineOfSightResult>(
55 eve::DiagnosticCode::Unsupported, "Physics LOS supports only two valid World3D points in the same space");
56
57 if (!std::isfinite(fromPoint->x()) || !std::isfinite(fromPoint->y()) || !std::isfinite(fromPoint->z()) ||
58 !std::isfinite(toPoint->x()) || !std::isfinite(toPoint->y()) || !std::isfinite(toPoint->z()))
59 return failure<sensing::LineOfSightResult>(eve::DiagnosticCode::InvalidArgument,
60 "Physics LOS points must be finite");
61
62 World3D* activeWorld = nullptr;
63 std::shared_ptr<const void> activeLifetime;
64 for (const RegisteredWorld& entry : worlds_) {
65 auto lifetime = entry.lifetime.lock();
66 World3D* world = entry.world;
67 if (!lifetime || !world || !world->isValid()) continue;
68 if (activeWorld)
69 return failure<sensing::LineOfSightResult>(eve::DiagnosticCode::Conflict,
70 "Physics LOS has more than one valid active World3D");
71 activeWorld = world;
72 activeLifetime = std::move(lifetime);
73 }
74 if (!activeWorld)
75 return failure<sensing::LineOfSightResult>(eve::DiagnosticCode::Unsupported,
76 "Physics LOS has no valid active World3D");
77
78 activeWorld->rayCast(fromPoint->x(), fromPoint->y(), fromPoint->z(), toPoint->x(), toPoint->y(), toPoint->z());
79 if (activeWorld->hasRayHit())
80 return eve::Result<sensing::LineOfSightResult>::success(sensing::LineOfSightResult{false, std::nullopt});
81 return eve::Result<sensing::LineOfSightResult>::success(sensing::LineOfSightResult{true, std::nullopt});
82}
