17 const auto invalid = [](const char* message) {
18 return Result<std::vector<ShaderImageRegion>>::failure(
19 Diagnostic::error(DiagnosticCode::InvalidArgument, message, "image"));
20 };
21 if (!image.width || !image.height || image.width > 32768 || image.height > 32768 || !image.layers ||
22 image.layers > 2048 || !image.mipLevels ||
23 image.mipLevels > static_cast<uint32_t>(std::bit_width(std::max(image.width, image.height))) ||
24 image.binding > 31)
25 return invalid("Invalid shader image dimensions, mip count or binding");
26 switch (image.dimension) {
27 case ShaderImageDimension::Image2D:
28 if (image.layers != 1) return invalid("A 2D image must have one layer");
29 break;
30 case ShaderImageDimension::Array2D: break;
31 case ShaderImageDimension::Cube:
32 if (image.layers != 6 || image.width != image.height) return invalid("A cube image needs six square faces");
33 break;
34 default: return invalid("Unknown image dimension");
35 }
36 std::size_t block = 1, stride = 0;
37 switch (image.format) {
38 case ShaderImageFormat::R8: stride = 1; break;
39 case ShaderImageFormat::RG8:
40 case ShaderImageFormat::R16: stride = 2; break;
41 case ShaderImageFormat::RGBA8:
42 case ShaderImageFormat::BGRA8:
43 case ShaderImageFormat::RGBA8Srgb:
44 case ShaderImageFormat::BGRA8Srgb: stride = 4; break;
45 case ShaderImageFormat::BC1:
46 case ShaderImageFormat::BC1Srgb:
47 block = 4;
48 stride = 8;
49 break;
50 case ShaderImageFormat::BC3:
51 case ShaderImageFormat::BC7:
52 case ShaderImageFormat::BC7Srgb:
53 case ShaderImageFormat::BC3Srgb:
54 block = 4;
55 stride = 16;
56 break;
57 default: return invalid("Unknown image format");
58 }
59 if (!std::isfinite(image.sampler.maxAnisotropy) || image.sampler.maxAnisotropy < 1 ||
60 !std::isfinite(image.sampler.lodBias) || !std::isfinite(image.sampler.minLod) ||
61 !std::isfinite(image.sampler.maxLod) || image.sampler.minLod < 0 || image.sampler.maxLod < image.sampler.minLod)
62 return invalid("Invalid shader image sampler");
63 std::vector<ShaderImageRegion> result;
64 std::size_t offset = 0;
65 for (std::uint32_t layer = 0; layer < image.layers; ++layer) {
66 for (std::uint32_t mip = 0; mip < image.mipLevels; ++mip) {
67 const auto width = std::max(1u, image.width >> mip);
68 const auto height = std::max(1u, image.height >> mip);
69 const auto size = ((width + block - 1) / block) * ((height + block - 1) / block) * stride;
70 if (size > 1024ull * 1024 * 1024 || offset > 1024ull * 1024 * 1024 - size)
71 return invalid("Shader image exceeds 1 GiB upload limit");
72 result.push_back({layer, mip, width, height, offset, size});
73 offset += size;
74 }
75 }
76 if (image.bytes.size() != offset) return invalid("Shader image payload size does not match its layout");
77 return Result<std::vector<ShaderImageRegion>>::success(std::move(result));
78}
