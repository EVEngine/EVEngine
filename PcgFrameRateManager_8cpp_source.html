33 interval_ = interval;
34 min_ = minimum;
35 max_ = maximum;
36 quality_ = current;

44Result<int> PcgFrameRateManager::update(float dt, float scale) {
45 if (!std::isfinite(dt) || dt <= 0 || !std::isfinite(scale) || scale < 0)
46 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
47 "dt must be positive and timeScale non-negative",
48 "procgen.frameRate.update"));
49 changed_ = false;
50 fpsLeft_ -= dt;
51 accum_ += scale / dt;
52 ++frames_;
53 if (fpsLeft_ <= 0) {
54 fps_ = accum_ / float(frames_);
55 fpsLeft_ = .5f;
56 accum_ = 0;
57 frames_ = 0;
58 }
59 if (automatic_) {
60 left_ -= dt;
61 if (left_ < 0) {
62 int old = quality_;
63 if (fps_ + 10 >= target_ && quality_ < max_)
64 ++quality_;
65 else if (fps_ - 10 <= target_ && quality_ > min_)
66 --quality_;
67 changed_ = old != quality_;
68 left_ = interval_;
69 }
70 }
71 return Result<int>::success(quality_);
72}

159Result<int> combinePcgStaticMeshesInto(MeshBuild& output, const PcgMeshCombinePlan& plan) {
160 if (plan.sources_.empty())
161 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
162 "Pcg static mesh combine plan is empty", "procgen.pcgMeshLod"));
163 std::uint64_t vertexCount = 0, indexCount = 0;
164 bool anyColors = false;
165 for (const auto& source : plan.sources_) {
166 vertexCount += static_cast<std::uint64_t>(source.mesh.getVertexCount());
167 indexCount += static_cast<std::uint64_t>(source.mesh.getIndexCount());
168 anyColors = anyColors || source.mesh.hasVertexColors();
169 }
170 if (vertexCount > 8U * 1024U * 1024U || indexCount > 48U * 1024U * 1024U ||
171 vertexCount > static_cast<std::uint64_t>(std::numeric_limits<int>::max()) ||
172 indexCount > static_cast<std::uint64_t>(std::numeric_limits<int>::max()))
173 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
174 "Pcg static mesh combine output exceeds native mesh budgets",
175 "procgen.pcgMeshLod"));
176 MeshBuild candidate;
177 candidate.reserve(static_cast<int>(vertexCount), static_cast<int>(indexCount));
178 std::vector<float> colors;
179 if (anyColors) colors.reserve(static_cast<std::size_t>(vertexCount) * 4U);
180 struct Bucket {
181 std::string material;
182 std::vector<std::uint32_t> indices;
183 };
184 std::vector<Bucket> buckets;
185 std::unordered_map<std::string, std::size_t> materialMap;
186 std::uint32_t vertexOffset = 0;
187 for (const auto& source : plan.sources_) {
188 const auto& mesh = source.mesh;
189 const auto& m = source.transform.matrix_;
190 for (int vertex = 0; vertex < mesh.getVertexCount(); ++vertex) {
191 const float x = mesh.getPositionX(vertex), y = mesh.getPositionY(vertex), z = mesh.getPositionZ(vertex);
192 const float nx = mesh.getNormalX(vertex), ny = mesh.getNormalY(vertex), nz = mesh.getNormalZ(vertex);
193 const float px = m[0] * x + m[1] * y + m[2] * z + m[3], py = m[4] * x + m[5] * y + m[6] * z + m[7];
194 const float pz = m[8] * x + m[9] * y + m[10] * z + m[11];
195 const float tx = m[0] * nx + m[1] * ny + m[2] * nz, ty = m[4] * nx + m[5] * ny + m[6] * nz;
196 const float tz = m[8] * nx + m[9] * ny + m[10] * nz;
197 if (!std::isfinite(px) || !std::isfinite(py) || !std::isfinite(pz) || !std::isfinite(tx) ||
198 !std::isfinite(ty) || !std::isfinite(tz))
199 return Result<int>::failure(Diagnostic::error(
200 DiagnosticCode::InvalidArgument, "Pcg static mesh transform overflowed", "procgen.pcgMeshLod"));
201 candidate.addVertex(px, py, pz, tx, ty, tz, mesh.getUvU(vertex), mesh.getUvV(vertex));
202 if (anyColors)
203 for (int component = 0; component < 4; ++component)
204 colors.push_back(mesh.hasVertexColors() ? mesh.getColor(vertex, component) : 1.F);
205 }
206 for (int triangle = 0; triangle < mesh.getIndexCount() / 3; ++triangle) {
207 const int group = mesh.getTriangleGroup(triangle);
208 std::string material = group >= 0 ? mesh.getGroupName(group) : source.defaultMaterialId;
209 if (material.empty())
210 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
211 "Pcg static mesh triangle has no material identity",
212 "procgen.pcgMeshLod"));
213 auto [found, inserted] = materialMap.emplace(material, buckets.size());
214 if (inserted) buckets.push_back(Bucket{material, {}});
215 auto& indices = buckets[found->second].indices;
216 for (int corner = 0; corner < 3; ++corner)
217 indices.push_back(vertexOffset + static_cast<std::uint32_t>(mesh.getIndex(triangle * 3 + corner)));
218 }
219 vertexOffset += static_cast<std::uint32_t>(mesh.getVertexCount());
220 }
221 for (auto& bucket : buckets) {
222 candidate.setActiveGroup(bucket.material);
223 for (std::size_t i = 0; i < bucket.indices.size(); i += 3)
224 candidate.addTriangle(bucket.indices[i], bucket.indices[i + 1], bucket.indices[i + 2]);
225 }
226 if (anyColors) {
227 auto set = candidate.setVertexColors(std::move(colors));
228 if (!set.ok()) return Result<int>::failure(set.status());
229 }
230 candidate.setMeta("kind", "pcg.mesh.combined");
231 candidate.setMeta("sourceCount", std::to_string(plan.sources_.size()));
232 output = std::move(candidate);
233 return Result<int>::success(static_cast<int>(indexCount / 3U));
234}

443Result<std::uint64_t> PcgTaskQueue::add(double wait) {

445 return Result<std::uint64_t>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
446 "task wait time must be finite and non-negative", {},
447 {}, "procgen.taskQueue"));
448 const auto id = nextId_++;
449 tasks_.push_back({id, wait, wait});
450 if (status_ == PcgTaskQueueStatus::Idle) status_ = PcgTaskQueueStatus::Waiting;
451 return Result<std::uint64_t>::success(id);
452}
453Result<PcgTaskQueueStatus> PcgTaskQueue::tick(double dt) {
454 if (!std::isfinite(dt) || dt < 0)
455 return Result<PcgTaskQueueStatus>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
456 "delta time must be finite and non-negative", {},
457 {}, "procgen.taskQueue"));
458 if (status_ == PcgTaskQueueStatus::Ready)
459 return Result<PcgTaskQueueStatus>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
460 "ready task must be resolved before ticking", {},
461 {}, "procgen.taskQueue"));
462 if (tasks_.empty()) {
463 status_ = PcgTaskQueueStatus::Idle;
464 return Result<PcgTaskQueueStatus>::success(status_);
465 }
466 cursor_ %= tasks_.size();
467 auto& task = tasks_[cursor_];
468 task.remaining -= dt;
469 if (task.remaining <= 1e-12) {
470 readyId_ = task.id;
471 status_ = PcgTaskQueueStatus::Ready;
472 } else
473 status_ = PcgTaskQueueStatus::Waiting;
474 return Result<PcgTaskQueueStatus>::success(status_);
475}
476Result<PcgTaskQueueStatus> PcgTaskQueue::resolveReady(bool finished) {
477 if (status_ != PcgTaskQueueStatus::Ready || tasks_.empty())
478 return Result<PcgTaskQueueStatus>::failure(
479 Diagnostic::error(DiagnosticCode::InvalidArgument, "no task is ready", {}, {}, "procgen.taskQueue"));
480 if (finished)
481 tasks_.erase(tasks_.begin() + static_cast<std::ptrdiff_t>(cursor_));
482 else {
483 tasks_[cursor_].remaining = tasks_[cursor_].wait;
484 ++cursor_;
485 }
486 readyId_ = 0;
487 if (tasks_.empty()) {
488 cursor_ = 0;
489 status_ = PcgTaskQueueStatus::Idle;
490 } else {
491 cursor_ %= tasks_.size();
492 status_ = PcgTaskQueueStatus::Waiting;
493 }
494 return Result<PcgTaskQueueStatus>::success(status_);
495}
496PcgTaskQueueStatus PcgTaskQueue::cancelAll() noexcept {
497 tasks_.clear();
498 cursor_ = 0;
499 readyId_ = 0;
500 return status_ = PcgTaskQueueStatus::Idle;
501}
