55bool PcgInteriorWeatherVolume::contains(float x, float y, float z) const noexcept {
56 if (!finite3(x, y, z)) return false;
57 const float dx = x - centerX_, dy = y - centerY_, dz = z - centerZ_;
58 if (shape_ == PcgInteriorWeatherShape::Sphere)
59 return dx * dx + dy * dy + dz * dz <= extentX_ * extentX_;
60 return std::fabs(dx) <= extentX_ && std::fabs(dy) <= extentY_ &&
61 std::fabs(dz) <= extentZ_;
62}
