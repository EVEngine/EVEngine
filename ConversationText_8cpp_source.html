33std::string renderToken(std::string token, const StateValue& bindings, const StateValue& locals) {
34 std::string operation;
35 if (const size_t separator = token.find('|'); separator != std::string::npos) {
36 operation = token.substr(separator + 1);
37 token.resize(separator);
38 }
39 std::string fallback;
40 if (const size_t separator = token.find("??"); separator != std::string::npos) {
41 fallback = token.substr(separator + 2);
42 token.resize(separator);
43 }
44 const StateValue* value = locals.get(token);
45 if (!value) value = bindings.get(token);
46 std::string rendered = scalarText(value);
47 if (rendered.empty()) rendered = std::move(fallback);
48 return transform(std::move(rendered), operation);
49}

58std::string ConversationTextRenderer::render(const std::string& text, const StateValue& bindings,
59 const StateValue& locals, const Evaluator& evaluate) const {
60 std::string output;
61 output.reserve(text.size());
62 for (size_t cursor = 0; cursor < text.size();) {
63 if (text[cursor] != '{') {
64 output.push_back(text[cursor++]);
65 continue;
66 }
67 const size_t close = text.find('}', cursor + 1);
68 if (close == std::string::npos) {
69 output.append(text, cursor, std::string::npos);
70 break;
71 }
72 const std::string value = renderToken(text.substr(cursor + 1, close - cursor - 1), bindings, locals);
73 if (value.empty())
74 output.append(text, cursor, close - cursor + 1);
75 else
76 output += value;
77 cursor = close + 1;
78 }
79 for (const auto& rule : rules_) {
80 if (!rule.expression.empty() && (!evaluate || !evaluate(rule.expression))) continue;
81 replaceAll(output, rule.find, rule.replacement);
82 output = rule.prefix + output + rule.suffix;
83 }
84 return output;
85}
