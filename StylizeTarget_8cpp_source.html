31EditorResult<StylizePassValue> parsePass(const EditorValue& v) {
32 const auto *idv = field(v, "id"), *stylev = field(v, "style"), *enabledv = field(v, "enabled"),
33 *priorityv = field(v, "priority"), *overridesv = field(v, "overrides");
34 const auto* id = idv ? idv->getIf<std::string>() : nullptr;
35 const auto* style = stylev ? stylev->getIf<std::string>() : nullptr;
36 const auto* enabled = enabledv ? enabledv->getIf<bool>() : nullptr;
37 const auto* priority = priorityv ? priorityv->getIf<int64_t>() : nullptr;
38 const auto* overrides = overridesv ? overridesv->getIf<EditorValue::Object>() : nullptr;
39 if (!id || id->empty() || !style || !stylize::isKnownStyle(*style) || !enabled || !priority ||
40 *priority < -100000 || *priority > 100000 || !overrides)
41 return fail<StylizePassValue>(EditorStatus::Rejected, "editor.stylize.pass", "Invalid stylize pass fields");
42 StylizePassValue p{ObjectId(*id), *style, *enabled, static_cast<int>(*priority), {}};
43 for (const auto& [name, value] : *overrides) {
44 const auto* n = value.getIf<double>();
45 const auto* d = stylize::findStyleParameter(*style, name);
46 if (!n || !std::isfinite(*n) || !d || *n < d->minValue || *n > d->maxValue)
47 return fail<StylizePassValue>(EditorStatus::Rejected, "editor.stylize.parameter",
48 "Invalid stylize parameter override");
49 p.overrides[name] = *n;
50 }
51 return eve::editing::applied<StylizePassValue>(std::move(p));
52}

110PropertyReadResult StylizeRecipeTarget::read(const SelectionSnapshot& s, const PropertyPath& path) const {
111 if (!matches(s) || !schema(s).find(path)) return {};
112 std::optional<EditorValue> common;
113 for (const auto& i : s.items) {
114 const auto& p = passes_.at(ObjectId(i.item.value()));
115 EditorValue value;
116 if (path == PropertyPath("pass.enabled"))
117 value = p.enabled;
118 else if (path == PropertyPath("pass.priority"))
119 value = int64_t{p.priority};
120 else {
121 const std::string name = path.value().substr(std::string("parameter.").size());
122 auto found = p.overrides.find(name);
123 const auto* d = stylize::findStyleParameter(p.style, name);
124 value = found == p.overrides.end() ? EditorValue(static_cast<double>(d->defaultValue))
125 : EditorValue(found->second);
126 }
127 if (!common)
128 common = value;
129 else if (*common != value)
130 return {PropertyReadState::Mixed, {}, {}};
131 }
132 return {PropertyReadState::Value, *common, {}};
133}

152EditorResult<DomainOperation> StylizeRecipeTarget::makeSet(const SelectionSnapshot& s, const PropertyPath& path,
153 const EditorValue& value, PropertySetMode mode) const {
154 if (mode == PropertySetMode::Reset) return makeReset(s, path);
155 auto d = schema(s).find(path);
156 if (!matches(s) || !d || mode != PropertySetMode::Absolute)
157 return fail<DomainOperation>(EditorStatus::Rejected, "editor.stylize.set",
158 "Stylize property requires matching absolute selection");
159 auto checked = validatePropertyValue(*d, value);
160 if (!checked.ok())
161 return fail<DomainOperation>(EditorStatus::Rejected, "editor.stylize.value",
162 "Stylize property is outside its metadata range");
163 auto candidate = *this;
164 for (const auto& i : s.items) {
165 auto& p = candidate.passes_[ObjectId(i.item.value())];
166 if (path == PropertyPath("pass.enabled"))
167 p.enabled = *value.getIf<bool>();
168 else if (path == PropertyPath("pass.priority"))
169 p.priority = static_cast<int>(*value.getIf<int64_t>());
170 else
171 p.overrides[path.value().substr(10)] = *value.getIf<double>();
172 }
173 if (hasError(candidate.validate()))
174 return fail<DomainOperation>(EditorStatus::Rejected, "editor.stylize.invalid",
175 "Stylize edit produces an invalid recipe");
176 return replacement(candidate.contentValue(), path.value());
177}
178EditorResult<DomainOperation> StylizeRecipeTarget::makeReset(const SelectionSnapshot& s,
179 const PropertyPath& path) const {
180 auto d = schema(s).find(path);
181 if (!d)
182 return fail<DomainOperation>(EditorStatus::Unsupported, "editor.stylize.property", "Unknown stylize property");
183 if (path.value().starts_with("parameter.")) {
184 auto candidate = *this;
185 for (const auto& i : s.items)
186 candidate.passes_[ObjectId(i.item.value())].overrides.erase(path.value().substr(10));
187 return replacement(candidate.contentValue(), path.value());
188 }
189 return makeSet(s, path, d->defaultValue, PropertySetMode::Absolute);

222EditorResult<void> StylizeRecipeTarget::applyDomainOperation(const DomainOperation& op) {
223 if (op.target != TargetId(id_) || op.type != "stylize.recipe.replace.v1")
224 return fail<void>(EditorStatus::Rejected, "editor.stylize.operation", "Stylize operation mismatch");
225 const auto* p = field(op.payload, "passes");
226 const auto* array = p ? p->getIf<EditorValue::Array>() : nullptr;
227 if (!array || array->size() > 64)
228 return fail<void>(EditorStatus::Rejected, "editor.stylize.payload",
229 "Stylize recipe payload is invalid or exceeds 64 passes");
230 StylizeRecipeTarget candidate(id_);
231 for (const auto& entry : *array) {
232 auto parsed = parsePass(entry);
233 if (!parsed.ok() || candidate.passes_.contains(parsed.value().id))
234 return fail<void>(EditorStatus::Rejected, "editor.stylize.pass",
235 "Stylize recipe contains invalid or duplicate passes");
236 candidate.order_.push_back(parsed.value().id);
237 candidate.passes_.emplace(parsed.value().id, parsed.value());
238 }
239 if (hasError(candidate.validate()))
240 return fail<void>(EditorStatus::Rejected, "editor.stylize.invalid",
241 "Stylize recipe violates stage or parameter rules");
242 passes_ = std::move(candidate.passes_);
243 order_ = std::move(candidate.order_);
244 ++revision_;
245 dirty_.include(0, 0);
246 return eve::editing::applied<void>();
247}

263std::vector<EditorDiagnostic> StylizeRecipeTarget::validate() const {
264 std::vector<EditorDiagnostic> d;
265 std::string stage;
266 for (const auto& id : order_) {
267 const auto& p = passes_.at(id);
268 const auto* definition = stylize::findStyleDefinition(p.style);
269 if (!definition) {
270 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument,
271 RuleId("editor.stylize.unknown-style"), DiagnosticSeverity::Error, "Unknown style: " + p.style));
272 continue;
273 }
274 if (!p.enabled) continue;
275 const std::string current = graphics::postEffectStageName(definition->stage);
276 if (stage.empty())
277 stage = current;
278 else if (stage != current)
279 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::PreconditionViolation,
280 RuleId("editor.stylize.mixed-stage"), DiagnosticSeverity::Error,
281 "Enabled recipe passes must share one pipeline stage"));
282 if (!definition->post)
283 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::PreconditionViolation,
284 RuleId("editor.stylize.mesh-in-recipe"), DiagnosticSeverity::Error,
285 "Post recipe cannot contain a mesh-only style: " + p.style));
286 }
287 if (order_.empty())
288 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::PreconditionViolation,
289 RuleId("editor.stylize.empty"), DiagnosticSeverity::Warning, "Stylize recipe has no passes"));
290 return d;
