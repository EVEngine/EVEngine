25Result<std::uint64_t> readUint64(const Value& value, std::string_view field) {
26 const auto* text = value.getIf<std::string>();
27 if (!text || text->empty())
28 return failure<std::uint64_t>(DiagnosticCode::ParseError, "snapshot integer must be a decimal string",
29 std::string(field));
30 std::uint64_t output = 0;
31 const auto [end, error] = std::from_chars(text->data(), text->data() + text->size(), output);
32 if (error != std::errc{} || end != text->data() + text->size())
33 return failure<std::uint64_t>(DiagnosticCode::ParseError, "snapshot integer is out of range",
34 std::string(field));
35 return Result<std::uint64_t>::success(output);
36}

66Result<void> verifySnapshotEnvelope(const SnapshotEnvelope& snapshot, const SnapshotHashProvider& hashProvider) {
67 auto valid = validateHeader(snapshot);
68 if (!valid.ok()) return valid;
69 if (!hashProvider)
70 return failure<void>(DiagnosticCode::Unsupported, "snapshot hash provider is required", "contentHash");
71
72 auto input = snapshotHashInput(snapshot);
73 if (!input.ok()) return Result<void>::failure(input.status());
74 auto computed = hashProvider(input.value());
75 if (!computed.ok()) return Result<void>::failure(computed.status());
76 if (std::move(computed).takeValue() != snapshot.contentHash)
77 return failure<void>(DiagnosticCode::HashMismatch, "snapshot content hash does not match", "contentHash");
78 return Result<void>::success();
79}

81Result<void> validateSnapshotPayloadMetadata(const Value& payload, Revision revision, SimulationTick tick) {
82 const auto* object = payload.getIf<Value::Object>();
83 if (!object) return Result<void>::success();
84
85 const auto validate = [&](std::string_view field, std::uint64_t expected) -> Result<void> {
86 const auto found = object->find(std::string(field));
87 if (found == object->end()) return Result<void>::success();
88 auto parsed = readUint64(found->second, field);
89 if (!parsed.ok()) return Result<void>::failure(parsed.status());
90 if (parsed.value() != expected)
91 return failure<void>(DiagnosticCode::Conflict,
92 "snapshot payload " + std::string(field) + " disagrees with the envelope",
93 "payload." + std::string(field));
94 return Result<void>::success();
95 };
96
97 auto revisionResult = validate("revision", revision.value());
98 if (!revisionResult.ok()) return revisionResult;
99 auto tickResult = validate("tick", tick.value());
100 if (!tickResult.ok()) return tickResult;
101 return Result<void>::success();
102}

137Result<SnapshotEnvelope> parseSnapshotEnvelopeValue(const Value& value, const SnapshotHashProvider& hashProvider) {
138 const auto* object = value.getIf<Value::Object>();
139 if (!object || !hasOnlyEnvelopeFields(*object))
140 return failure<SnapshotEnvelope>(DiagnosticCode::ParseError,
141 "snapshot envelope must be an object with the exact public fields");
142
143 const auto type = value.getIf<Value::Object>()->find("type");
144 const auto schemaText = object->find("schema");
145 const auto schemaVersion = object->find("schemaVersion");
146 const auto instanceId = object->find("instanceId");
147 const auto revision = object->find("revision");
148 const auto tick = object->find("tick");
149 const auto contentHash = object->find("contentHash");
150 const auto payload = object->find("payload");
151 if (type == object->end() || schemaText == object->end() || schemaVersion == object->end() ||
152 instanceId == object->end() || revision == object->end() || tick == object->end() ||
153 contentHash == object->end() || payload == object->end() || !type->second.isString() ||
154 !schemaText->second.isString() || !instanceId->second.isString() || !contentHash->second.isString())
155 return failure<SnapshotEnvelope>(DiagnosticCode::ParseError, "snapshot envelope has invalid fields");
156
157 const auto schema = LogicalId::parse(*schemaText->second.getIf<std::string>());
158 const auto persistentId = PersistentId::parse(*instanceId->second.getIf<std::string>());
159 const auto contentId = ContentId::parse(*contentHash->second.getIf<std::string>());
160 if (!schema || !persistentId || !contentId)
161 return failure<SnapshotEnvelope>(DiagnosticCode::ParseError, "snapshot envelope has invalid identity fields");
162
163 auto version = readUint64(schemaVersion->second, "schemaVersion");
164 if (!version.ok()) return Result<SnapshotEnvelope>::failure(version.status());
165 auto revisionValue = readUint64(revision->second, "revision");
166 if (!revisionValue.ok()) return Result<SnapshotEnvelope>::failure(revisionValue.status());
167 auto tickValue = readUint64(tick->second, "tick");
168 if (!tickValue.ok()) return Result<SnapshotEnvelope>::failure(tickValue.status());
169
170 SnapshotEnvelope snapshot{*type->second.getIf<std::string>(),
171 *schema,
172 SchemaVersion(std::move(version).takeValue()),
173 *persistentId,
174 Revision(std::move(revisionValue).takeValue()),
175 SimulationTick(std::move(tickValue).takeValue()),
176 *contentId,
177 payload->second};
178 auto verified = verifySnapshotEnvelope(snapshot, hashProvider);
179 if (!verified.ok()) return Result<SnapshotEnvelope>::failure(verified.status());
180 return Result<SnapshotEnvelope>::success(std::move(snapshot));
181}

204Result<SnapshotEnvelope> SnapshotMigrationChain::migrate(SnapshotEnvelope snapshot, SchemaVersion targetVersion,
205 const SnapshotHashProvider& hashProvider) const {
206 auto verified = verifySnapshotEnvelope(snapshot, hashProvider);
207 if (!verified.ok()) return Result<SnapshotEnvelope>::failure(verified.status());
208 if (snapshot.schemaVersion.value() > targetVersion.value())
209 return failure<SnapshotEnvelope>(DiagnosticCode::UnknownVersion,
210 "snapshot schema version is newer than the supported target", "schemaVersion");
211 if (snapshot.schemaVersion == targetVersion) return Result<SnapshotEnvelope>::success(std::move(snapshot));
212
213 while (snapshot.schemaVersion.value() < targetVersion.value()) {
214 const Key key{snapshot.schema.format(), snapshot.schemaVersion.value()};
215 const auto step = steps_.find(key);
216 if (step == steps_.end())
217 return failure<SnapshotEnvelope>(DiagnosticCode::Unsupported,
218 "no snapshot migration step reaches the requested version",
219 "schemaVersion");
220 if (step->second.to.value() > targetVersion.value())
221 return failure<SnapshotEnvelope>(DiagnosticCode::Unsupported,
222 "snapshot migration step overshoots the requested version",
223 "schemaVersion");
224 auto payload = step->second.migration(snapshot.payload);
225 if (!payload.ok()) return Result<SnapshotEnvelope>::failure(payload.status());
226 snapshot.payload = std::move(payload).takeValue();
227 snapshot.schemaVersion = step->second.to;
228 }
229
230 return makeSnapshotEnvelope(snapshot.type, snapshot.schema, snapshot.schemaVersion, snapshot.instanceId,
231 snapshot.revision, snapshot.tick, std::move(snapshot.payload), hashProvider);
232}
