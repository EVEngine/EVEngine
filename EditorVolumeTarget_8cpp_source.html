13bool IntVolumeEditCommand::record(int x, int y, int z, int after) {
14 auto* volume = target_ ? target_->query<IIntVolumeTarget>() : nullptr;
15 if (!volume) return false;
16 for (auto& change : changes_) {
17 if (change.x == x && change.y == y && change.z == z) {
18 change.after = after;
19 return true;
20 }
21 }
22 const int before = volume->readInt3(x, y, z);
23 if (before == after) return false;
24 changes_.push_back({x, y, z, before, after});
25 dirty_.include(x, y, z);
26 return true;
27}
28
29bool IntVolumeEditCommand::apply() {
30 auto* volume = target_ ? target_->query<IIntVolumeTarget>() : nullptr;
31 if (!volume || changes_.empty()) return false;
32 for (std::size_t i = 0; i < changes_.size(); ++i) {
33 if (volume->writeInt3(changes_[i].x, changes_[i].y, changes_[i].z, changes_[i].after) ==
34 editing::FieldWriteStatus::Rejected) {
35 while (i > 0) {
36 --i;
37 (void)volume->writeInt3(changes_[i].x, changes_[i].y, changes_[i].z, changes_[i].before);
38 }
39 return false;
40 }
41 }
42 return true;
43}

45void IntVolumeEditCommand::revert() {
46 auto* volume = target_ ? target_->query<IIntVolumeTarget>() : nullptr;
47 if (!volume) return;
48 for (auto it = changes_.rbegin(); it != changes_.rend(); ++it)
49 (void)volume->writeInt3(it->x, it->y, it->z, it->before);
50}

56bool IntVolumeEditCommand::mergeWith(const IEditCommand& laterBase) {
57 const auto* later = dynamic_cast<const IntVolumeEditCommand*>(&laterBase);
58 if (!later || later->target_ != target_) return false;
59 for (const auto& incoming : later->changes_) {
60 bool merged = false;
61 for (auto& current : changes_) {
62 if (current.x == incoming.x && current.y == incoming.y && current.z == incoming.z) {
63 current.after = incoming.after;
64 merged = true;
65 break;
66 }
67 }
68 if (!merged) changes_.push_back(incoming);
69 }
70 dirty_.include(later->dirty_);
71 return true;
72}
