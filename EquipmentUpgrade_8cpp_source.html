44eve::Result<DecodedUpgradeReservation> decodeReservation(const eve::Value& reservation) {
45 const auto* object = reservation.getIf<eve::Value::Object>();
46 if (object == nullptr)
47 return failure<DecodedUpgradeReservation>(eve::DiagnosticCode::ParseError,
48 "upgrade reservation must be an object", "reservation");
49 const auto schema = object->find("schema");
50 const auto version = object->find("version");
51 if (schema == object->end() || !schema->second.isString() ||
52 schema->second.asString() != "eve.rpg.upgrade-reservation" || version == object->end() ||
53 !version->second.isInt64() || version->second.asInt() != 1)
54 return failure<DecodedUpgradeReservation>(eve::DiagnosticCode::ParseError,
55 "upgrade reservation schema is invalid", "reservation");
56 const auto itemId = object->find("itemId");
57 const auto fromLevel = object->find("fromLevel");
58 const auto toLevel = object->find("toLevel");
59 const auto bagSlot = object->find("bagSlot");
60 const auto instanceId = object->find("instanceId");
61 if (itemId == object->end() || !itemId->second.isString() || itemId->second.asString().empty() ||
62 fromLevel == object->end() || !fromLevel->second.isInt64() || toLevel == object->end() ||
63 !toLevel->second.isInt64() || bagSlot == object->end() || !bagSlot->second.isInt64() ||
64 instanceId == object->end() || !instanceId->second.isInt64())
65 return failure<DecodedUpgradeReservation>(eve::DiagnosticCode::ParseError,
66 "upgrade reservation fields are invalid", "reservation");
67 DecodedUpgradeReservation decoded;
68 decoded.itemId = itemId->second.asString();
69 decoded.fromLevel = static_cast<int>(fromLevel->second.asInt());
70 decoded.toLevel = static_cast<int>(toLevel->second.asInt());
71 decoded.bagSlot = static_cast<int>(bagSlot->second.asInt());
72 decoded.instanceId = static_cast<int>(instanceId->second.asInt());
73 if (decoded.fromLevel < 0 || decoded.toLevel <= decoded.fromLevel || decoded.bagSlot < 0 || decoded.instanceId <= 0)
74 return failure<DecodedUpgradeReservation>(eve::DiagnosticCode::InvalidArgument,
75 "upgrade reservation levels or identity are invalid", "reservation");
76 return eve::Result<DecodedUpgradeReservation>::success(std::move(decoded));
77}

289eve::Result<void> validateTargetStack(const inventory::Bag& bag, int bagSlot, int expectedInstanceId,
290 const EquipmentUpgradeRecipe& recipe) {
291 if (bagSlot < 0 || bagSlot >= bag.getSlotCount() || bag.isSlotEmpty(bagSlot))
292 return failure<void>(eve::DiagnosticCode::NotFound, "upgrade requires a bag slot with the equipment",
293 "bagSlot");
294 if (expectedInstanceId <= 0 || bag.getSlotInstanceId(bagSlot) != expectedInstanceId)
295 return failure<void>(eve::DiagnosticCode::Conflict, "upgrade expectedInstanceId does not match the slot",
296 "expectedInstanceId");
297 if (bag.getSlotItemId(bagSlot) != recipe.itemId)
298 return failure<void>(eve::DiagnosticCode::Conflict, "upgrade item id does not match the recipe", "itemId");
299 if (bag.getSlotQuantity(bagSlot) != 1)
300 return failure<void>(eve::DiagnosticCode::InvalidArgument, "upgrade target must be a single-item stack",
301 "bagSlot");
302 const auto* definition = inventory::ItemRegistry::find(recipe.itemId);
303 if (definition == nullptr)
304 return failure<void>(eve::DiagnosticCode::NotFound, "upgrade item definition is missing", "itemId");
305 if (definition->maxStack != 1)
306 return failure<void>(eve::DiagnosticCode::InvalidArgument,
307 "upgrade targets must use maxStack=1 equipment items", "itemId");
308 auto level = EquipmentUpgrade::parseEnhanceLevel(bag.getSlotProp(bagSlot, kEquipmentEnhanceProp, "0"));
309 if (!level) return eve::Result<void>::failure(level.status());
310 if (level.value() != recipe.fromLevel)
311 return failure<void>(eve::DiagnosticCode::Conflict, "upgrade fromLevel does not match the instance",
312 "fromLevel");
313 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
314}

318eve::Result<int> EquipmentUpgrade::parseEnhanceLevel(std::string_view text) {
319 if (text.empty()) return eve::Result<int>::success(0);
320 for (char ch : text) {
321 if (!std::isdigit(static_cast<unsigned char>(ch)))
322 return failure<int>(eve::DiagnosticCode::ParseError, "enhance prop must be a non-negative integer",
323 kEquipmentEnhanceProp);
324 }
325 int value = 0;
326 const auto* begin = text.data();
327 const auto* end = text.data() + text.size();
328 const auto parsed = std::from_chars(begin, end, value);
329 if (parsed.ec != std::errc{} || parsed.ptr != end || value < 0)
330 return failure<int>(eve::DiagnosticCode::ParseError, "enhance prop must be a non-negative integer",
331 kEquipmentEnhanceProp);
332 return eve::Result<int>::success(value);
333}

380eve::Result<int> EquipmentUpgrade::settle(production::WorkQueue& queue, inventory::Bag& bag, std::string_view taskId) {
381 auto task = queue.find(taskId);
382 if (!task) return failure<int>(eve::DiagnosticCode::NotFound, "upgrade task was not found", "taskId");
383 const std::string settlementId = "rpg.upgrade:" + std::string(taskId);
384 if (task->get().state == production::TaskState::Completed && task->get().settlement.settlementId == settlementId) {
385 auto decoded = decodeReservation(task->get().reservation);
386 if (!decoded) return eve::Result<int>::failure(decoded.status());
387 return eve::Result<int>::success(decoded.value().toLevel, eve::Status::success(eve::StatusCode::NoOp));
388 }
389 if (task->get().state != production::TaskState::ReadyToSettle &&
390 task->get().state != production::TaskState::SettlementFailed)
391 return failure<int>(eve::DiagnosticCode::Conflict, "upgrade task is not ready to settle", "taskId");
392 auto decoded = decodeReservation(task->get().reservation);
393 if (!decoded) return eve::Result<int>::failure(decoded.status());
394
395 EnhanceParticipant enhanceParticipant(bag, decoded.value());
396 production::ProductionSettlementReceipt receipt;
397 receipt.settlementId = settlementId;
398 receipt.payload = eve::Value(eve::Value::Object{
399 {"destination", eve::Value(bag.getId())},
400 {"instanceId", eve::Value(std::int64_t(decoded.value().instanceId))},
401 {"toLevel", eve::Value(std::int64_t(decoded.value().toLevel))},
402 });
403 SettlementParticipant settlementParticipant(queue, std::string(taskId), std::move(receipt));
404 transaction::TransactionContext context("rpg.upgrade.settle:" + std::string(taskId));
405 std::array<transaction::ITransactionParticipant*, 2> participants{&settlementParticipant, &enhanceParticipant};
406 transaction::Coordinator coordinator;
407 auto committed = coordinator.execute(context, std::span<transaction::ITransactionParticipant*>(participants));
408 if (!committed) {
409 const auto* diagnostic = committed.error();
410 auto failed =
411 queue.failSettlement(taskId, diagnostic == nullptr ? "upgrade enhance rejected" : diagnostic->message());
412 const auto current = queue.find(taskId);
413 if (!failed && (!current || current->get().state != production::TaskState::SettlementFailed))
414 return eve::Result<int>::failure(failed.status());
415 return eve::Result<int>::failure(committed.status());
416 }
417 return eve::Result<int>::success(enhanceParticipant.appliedLevel(), eve::Status::success(eve::StatusCode::Applied));
418}
