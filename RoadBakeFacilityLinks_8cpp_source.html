63 const RoadContinuousFacility& facility) {
64 if ((edge.from == nodeId && facility.start > 1e-5f) || (edge.to == nodeId && facility.end < 1.f - 1e-5f))

167 const auto& b = second.value();

177 const V3 position = p0 * (u * u * u) + control0 * (3.f * u * u * t) + control1 * (3.f * u * t * t) +
178 p1 * (t * t * t);
179 const V3 forward = normalize((control0 - p0) * (u * u) + (control1 - control0) * (2.f * u * t) +
180 (p1 - control1) * (t * t));
181 const V3 upHint = normalize(
182 V3{std::lerp(a.upX, b.upX, t), std::lerp(a.upY, b.upY, t), std::lerp(a.upZ, b.upZ, t)});

184 SplineFrameSample frame;

187 frame.sample.z = position.z;
188 frame.sample.normalizedDistance = t;
189 frame.sideX = side.x;
190 frame.sideY = side.y;
191 frame.sideZ = side.z;
192 frame.upX = up.x;
193 frame.upY = up.y;
194 frame.upZ = up.z;
195 frame.forwardX = forward.x;
196 frame.forwardY = forward.y;
197 frame.forwardZ = forward.z;
198 frames.push_back(frame);
199 }
200 RoadProfile profile;
201 profile.points = facility.profile;
202 profile.closed = true;
203 profile.explicitMaterials = true;
204 if (edge.from == node.id) reverseSectionGeometry(profile.points, true);
205 if (!facility.prefab.vertices.empty()) {

214 const float rawFirst =

226 frames[i].sample.y - frames[i - 1].sample.y,
227 frames[i].sample.z - frames[i - 1].sample.z});

229 bakeFacilityPrefab(mesh, prefab, frames, sourceLow, sourceHigh,

231 if (!deformed.ok()) return deformed;

233 }
234 auto loft = loftProfileClean(mesh, frames, profile, span, facility.uvMeters, RoadMaterial::Curb,

237 }
