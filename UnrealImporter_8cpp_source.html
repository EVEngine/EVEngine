59bool safeProjectPath(std::string_view path, const AssetImportLimits& limits) {
60 if (path.empty() || path.size() > limits.maximumStringBytes || path.front() == '/' ||
61 path.find('\\') != std::string_view::npos || path.find(':') != std::string_view::npos) return false;
62 std::size_t start = 0;
63 for (std::size_t index = 0; index <= path.size(); ++index) {
64 if (index != path.size() && path[index] != '/') continue;
65 const auto segment = path.substr(start, index - start);
66 if (segment.empty() || segment == "." || segment == "..") return false;
67 start = index + 1;
68 }
69 return true;
70}

72std::array<float, 4> unrealQuaternionToCanonical(const std::array<float, 4>& source) {
73 const float x = source[0], y = source[1], z = source[2], w = source[3];
74 const float r[3][3] = {
75 {1 - 2*y*y - 2*z*z, 2*x*y - 2*z*w, 2*x*z + 2*y*w},
76 {2*x*y + 2*z*w, 1 - 2*x*x - 2*z*z, 2*y*z - 2*x*w},
77 {2*x*z - 2*y*w, 2*y*z + 2*x*w, 1 - 2*x*x - 2*y*y}
78 };
79 // C maps UE (X forward,Y right,Z up) to canonical (X right,Y up,-Z forward).
80 const float c[3][3] = {{0,1,0},{0,0,1},{-1,0,0}};
81 float cr[3][3]{};
82 float out[3][3]{};
83 for (int row = 0; row < 3; ++row) for (int col = 0; col < 3; ++col)
84 for (int k = 0; k < 3; ++k) cr[row][col] += c[row][k] * r[k][col];
85 for (int row = 0; row < 3; ++row) for (int col = 0; col < 3; ++col)
86 for (int k = 0; k < 3; ++k) out[row][col] += cr[row][k] * c[col][k];
87 std::array<float, 4> q{};
88 const float trace = out[0][0] + out[1][1] + out[2][2];
89 if (trace > 0) {
90 const float s = std::sqrt(trace + 1.0f) * 2;
91 q = {(out[2][1] - out[1][2]) / s, (out[0][2] - out[2][0]) / s,
92 (out[1][0] - out[0][1]) / s, 0.25f * s};
93 } else if (out[0][0] > out[1][1] && out[0][0] > out[2][2]) {
94 const float s = std::sqrt(1 + out[0][0] - out[1][1] - out[2][2]) * 2;
95 q = {0.25f*s, (out[0][1]+out[1][0])/s, (out[0][2]+out[2][0])/s,
96 (out[2][1]-out[1][2])/s};
97 } else if (out[1][1] > out[2][2]) {
98 const float s = std::sqrt(1 + out[1][1] - out[0][0] - out[2][2]) * 2;
99 q = {(out[0][1]+out[1][0])/s, 0.25f*s, (out[1][2]+out[2][1])/s,
100 (out[0][2]-out[2][0])/s};
101 } else {
102 const float s = std::sqrt(1 + out[2][2] - out[0][0] - out[1][1]) * 2;
103 q = {(out[0][2]+out[2][0])/s, (out[1][2]+out[2][1])/s, 0.25f*s,
104 (out[1][0]-out[0][1])/s};
105 }
106 return q;
107}
