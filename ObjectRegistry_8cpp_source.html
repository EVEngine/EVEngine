58eve::Result<ObjectHandle> ObjectRegistry::registerObject(const std::string& className, const ssq::Object& object,
59 const std::string& label) {
60 std::uint32_t allocatedIndex = ObjectHandle::invalidIndex;
61 bool appendedSlot = false;
62 try {
63 if (object.getType() != ssq::Type::INSTANCE) {
64 return registryFailure<ObjectHandle>(eve::DiagnosticCode::InvalidArgument,
65 "UI object registry accepts only script instances");
66 }
67
68 std::string group = className;
69 if (group.empty()) {
70 Runtime* runtime = ModuleManager::runtime();
71 if (!runtime) {
72 return registryFailure<ObjectHandle>(
73 eve::DiagnosticCode::Failed, "UI object registry requires an active Runtime to derive the class");
74 }
75 group = runtime->classNameOf(object);
76 if (group.empty()) {
77 return registryFailure<ObjectHandle>(eve::DiagnosticCode::InvalidArgument,
78 "UI object registry could not derive a script class name");
79 }
80 }
81
82 if (!freeSlots_.empty()) {
83 allocatedIndex = freeSlots_.back();
84 } else {
85 if (slots_.size() >= ObjectHandle::invalidIndex) {
86 return registryFailure<ObjectHandle>(eve::DiagnosticCode::Failed,
87 "UI object registry exhausted its slot index space");
88 }
89 allocatedIndex = static_cast<std::uint32_t>(slots_.size());
90 slots_.emplace_back();
91 appendedSlot = true;
92 }
93
94 Slot& target = slots_[allocatedIndex];
95 if (target.retired || target.entry.has_value()) {
96 if (appendedSlot) slots_.pop_back();
97 return registryFailure<ObjectHandle>(eve::DiagnosticCode::InvariantViolation,
98 "UI object registry selected an occupied or retired slot");
99 }
100
101 const ObjectHandle handle(allocatedIndex, target.generation);
102 auto ownedObject = eve::script::ownSquirrelObject(eve::script::borrowSquirrelObject(object));
103 if (!ownedObject) return eve::Result<ObjectHandle>::failure(ownedObject.status());
104 ObjectEntry candidate;
105 candidate.handle = handle;
106 candidate.className = group;
107 candidate.object = std::move(ownedObject).takeValue().release();
108 candidate.label = label.empty() ? group + " #" + std::to_string(count(group) + 1) : label;
109 target.entry.emplace(std::move(candidate));
110
111 bool insertedClass = false;
112 try {
113 auto [classIt, inserted] = byClass_.try_emplace(group);
114 insertedClass = inserted;
115 classIt->second.push_back(handle);
116 } catch (...) {
117 if (insertedClass) byClass_.erase(group);
118 target.entry.reset();
119 throw;
120 }
121
122 // The free-list entry is consumed only after all potentially-throwing
123 // registration work has completed.
124 if (!freeSlots_.empty() && freeSlots_.back() == allocatedIndex) freeSlots_.pop_back();
125 return eve::Result<ObjectHandle>::success(handle);
126 } catch (const std::exception& error) {
127 if (appendedSlot && !slots_.empty() && allocatedIndex + 1 == slots_.size() &&
128 !slots_.back().entry.has_value()) {
129 slots_.pop_back();
130 }
131 return registryFailure<ObjectHandle>(eve::DiagnosticCode::Failed,
132 std::string("UI object registration failed: ") + error.what());
133 } catch (...) {
134 if (appendedSlot && !slots_.empty() && allocatedIndex + 1 == slots_.size() &&
135 !slots_.back().entry.has_value()) {
136 slots_.pop_back();
137 }
138 return registryFailure<ObjectHandle>(eve::DiagnosticCode::Failed, "UI object registration failed");
139 }
140}

142eve::Result<void> ObjectRegistry::unregister(ObjectHandle handle) {
143 if (!handle.isValid()) {
144 return registryFailure<void>(eve::DiagnosticCode::InvalidArgument,
145 "cannot unregister an invalid UI object handle");
146 }
147
148 Slot* target = slot(handle);
149 if (!target) {
150 return registryFailure<void>(eve::DiagnosticCode::NotFound, "UI object handle index is outside the registry");
151 }
152 if (!target->entry.has_value() || target->generation != handle.generation()) {
153 return registryFailure<void>(eve::DiagnosticCode::StaleHandle, "UI object handle is stale");
154 }
155
156 auto classIt = byClass_.find(target->entry->className);
157 if (classIt == byClass_.end()) {
158 return registryFailure<void>(eve::DiagnosticCode::InvariantViolation, "UI object class index is missing");
159 }
160 auto& classEntries = classIt->second;
161 const auto handleIt = std::find(classEntries.begin(), classEntries.end(), handle);
162 if (handleIt == classEntries.end()) {
163 return registryFailure<void>(eve::DiagnosticCode::InvariantViolation,
164 "UI object class index does not contain its handle");
165 }
166
167 const auto next = ObjectHandle::nextGeneration(target->generation);
168 if (next) {
169 try {
170 // Reserve the reusable-slot bookkeeping before making the live
171 // entry unreachable. The remaining operations are noexcept.
172 freeSlots_.push_back(handle.index());
173 } catch (const std::exception& error) {
174 return registryFailure<void>(eve::DiagnosticCode::Failed,
175 std::string("UI object slot release failed: ") + error.what());
176 } catch (...) {
177 return registryFailure<void>(eve::DiagnosticCode::Failed, "UI object slot release failed");
178 }
179 }
180
181 classEntries.erase(handleIt);
182 if (classEntries.empty()) byClass_.erase(classIt);
183 target->entry.reset();
184 if (next) {
185 target->generation = *next;
186 } else {
187 target->retired = true;
188 }
189 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
190}

192void ObjectRegistry::clear(const std::string& className) {
193 const auto found = byClass_.find(className);
194 if (found == byClass_.end()) return;
195
196 const std::vector<ObjectHandle> handles = found->second;
197 for (const ObjectHandle handle : handles) {
198 unregister(handle).ignore("clear UI object class");
199 }

202void ObjectRegistry::clearAll() {
203 std::vector<ObjectHandle> handles;
204 for (const auto& [className, classHandles] : byClass_) {
205 (void)className;
206 handles.insert(handles.end(), classHandles.begin(), classHandles.end());
207 }
208 for (const ObjectHandle handle : handles) {
209 unregister(handle).ignore("clear all UI objects");
