48bool ObjectRegistry::unregister(uint64_t id) {
49 for (auto& pair : byClass_) {
50 auto& entries = pair.second;
51 const auto it = std::find_if(entries.begin(), entries.end(),
52 [&](const ObjectEntry& e) { return e.id == id; });
53 if (it == entries.end()) continue;
54 entries.erase(it);
55 if (entries.empty()) byClass_.erase(pair.first);
56 return true;
57 }
58 return false;
59}
