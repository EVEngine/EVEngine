12std::optional<eve::Value> skillValue(const RPGActor* actor, const SkillDefinition& definition, std::string_view key) {
13 if (!actor) return std::nullopt;
14 auto* mutableActor = const_cast<RPGActor*>(actor);
15 const auto* runtime = [&]() -> const SkillRuntime* {
16 const auto it = mutableActor->skills()->known.find(definition.id);
17 return it == mutableActor->skills()->known.end() ? nullptr : &it->second;
18 }();
19 if (key == "skill.id") return eve::Value(definition.id);
20 if (key == "skill.targetType") return eve::Value(definition.targetType);
21 if (key == "skill.known") return eve::Value(runtime != nullptr);
22 if (key == "skill.casting") return eve::Value(mutableActor->skills()->casting.active);
23 if (key == "skill.cooldown") return eve::Value(runtime ? runtime->cooldownRemaining : 0.0);
24 if (key == "skill.phase") return eve::Value(mutableActor->skills()->casting.active ? "casting" : "idle");
25 return std::nullopt;
26}
