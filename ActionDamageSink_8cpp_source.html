47Result<void> CombatActionDamageSink::apply(const action::ActionDamageBinding& binding,
48 const action::ActionNotifyContext& context) {
49 if (!resolver_) return failure(DiagnosticCode::NotFound, "combat-state resolver is unavailable", "resolver");
50 if (binding.targetIndex >= context.targets.size())
51 return failure(DiagnosticCode::NotFound, "damage target index is unavailable", "targetIndex");
52 auto target = resolver_(context.targets[binding.targetIndex]);
53 if (!target) return failure(DiagnosticCode::NotFound, "damage target has no combat state", "target");
54 if (windowState_ && windowState_->isInvulnerable(target->get().subject)) {
55 lastOutcome_.reset();
56 return Result<void>::success(Status::success(StatusCode::NoOp));
57 }
58
59 SubjectRef source = SubjectRef::nil();
60 if (context.source) {
61 auto resolvedSource = resolver_(*context.source);
62 if (resolvedSource) source = resolvedSource->get().subject;
63 }
64 DamageRequest request;
65 request.source = source;
66 request.target = target->get().subject;
67 request.actionExecution = context.executionId;
68 request.damageType = binding.damageType;
69 request.healthDamage = binding.amount;
70 request.poiseDamage = binding.poiseAmount;
71 request.knockback = {binding.knockback.x, binding.knockback.y, binding.knockback.z};
72 auto outcome = runtime_.apply(target->get(), request);
73 if (!outcome) return Result<void>::failure(outcome.status());
74 lastOutcome_ = std::move(outcome).takeValue();
75 return Result<void>::success(Status::success(StatusCode::Applied));
76}
