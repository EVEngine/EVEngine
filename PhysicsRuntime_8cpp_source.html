26EditorResult<physics::Fixture*> PhysicsColliderRuntimeBuilder::build2D(const PhysicsColliderTarget& target,
27 physics::Body* body) const {
28 if (!body || target.describe().type != "physics-collider-2d")
29 return bridgeError<physics::Fixture*>(EditorStatus::Rejected, "editor.physics.runtime-2d-input",
30 "A 2D collider target and body are required");
31 const EditorValue snapshot = target.snapshotValue();
32 const auto* root = snapshot.getIf<EditorValue::Object>();
33 const auto* values = root ? root->at("properties").getIf<EditorValue::Object>() : nullptr;
34 if (!values)
35 return bridgeError<physics::Fixture*>(EditorStatus::Failed, "editor.physics.runtime-properties",
36 "Collider properties are unavailable");
37 const std::string kind = *propertyValue<std::string>(*values, "shape.kind");
38 const auto* size = propertyValue<EditorValue::Array>(*values, "shape.size");
39 const auto* offset = propertyValue<EditorValue::Array>(*values, "shape.offset");
40 const float radius = static_cast<float>(*propertyValue<double>(*values, "shape.radius"));
41 const float density = static_cast<float>(*propertyValue<double>(*values, "material.density"));
42 const float friction = static_cast<float>(*propertyValue<double>(*values, "material.friction"));
43 const float restitution = static_cast<float>(*propertyValue<double>(*values, "material.restitution"));
44 physics::Fixture* fixture = nullptr;
45 if (kind == "box")
46 fixture = body->newRectangleFixtureAt(
47 static_cast<float>(*(*size)[0].getIf<double>()), static_cast<float>(*(*size)[1].getIf<double>()),
48 static_cast<float>(*(*offset)[0].getIf<double>()), static_cast<float>(*(*offset)[1].getIf<double>()),
49 density, friction, restitution);
50 else if (kind == "circle")
51 fixture = body->newCircleFixture(radius, density, friction, restitution);
52 else
53 return bridgeError<physics::Fixture*>(EditorStatus::Unsupported, "editor.physics.runtime-2d-asset-shape",
54 "Polygon and chain colliders require an asset resolver");
55 if (!fixture)
56 return bridgeError<physics::Fixture*>(EditorStatus::Failed, "editor.physics.runtime-create",
57 "Physics backend failed to create the fixture");
58 fixture->setSensor(*propertyValue<bool>(*values, "shape.sensor"));
59 fixture->setCategoryBits(static_cast<int>(*propertyValue<int64_t>(*values, "collision.category")));
60 fixture->setMaskBits(static_cast<int>(*propertyValue<int64_t>(*values, "collision.mask")));
61 return eve::editing::applied<physics::Fixture*>(fixture);

64EditorResult<physics::Fixture*> PhysicsColliderRuntimeBuilder::build2D(
65 const PhysicsColliderTarget& target, physics::Body* body, const IPhysicsColliderAssetResolver& assets) const {
66 if (!body || target.describe().type != "physics-collider-2d")
67 return bridgeError<physics::Fixture*>(EditorStatus::Rejected, "editor.physics.runtime-2d-input",
68 "A 2D collider target and body are required");
69 const EditorValue snapshot = target.snapshotValue();
70 const auto* root = snapshot.getIf<EditorValue::Object>();
71 const auto* values = root ? root->at("properties").getIf<EditorValue::Object>() : nullptr;
72 if (!values)
73 return bridgeError<physics::Fixture*>(EditorStatus::Failed, "editor.physics.runtime-properties",
74 "Collider properties are unavailable");
75 const std::string kind = *propertyValue<std::string>(*values, "shape.kind");
76 if (kind == "box" || kind == "circle") return build2D(target, body);
77 const std::string asset = *propertyValue<std::string>(*values, "shape.asset");
78 auto geometry = assets.resolve(asset, kind);
79 if (!geometry.ok()) return EditorResult<physics::Fixture*>::failure(geometry.status());
80 std::vector<float> vertices = geometry.value().vertices;
81 const auto* offset = propertyValue<EditorValue::Array>(*values, "shape.offset");
82 const float offsetX = static_cast<float>(*(*offset)[0].getIf<double>());
83 const float offsetY = static_cast<float>(*(*offset)[1].getIf<double>());
84 for (size_t index = 0; index < vertices.size(); index += 2) {
85 vertices[index] += offsetX;
86 vertices[index + 1] += offsetY;
87 }
88 physics::Fixture* fixture = nullptr;
89 try {
90 const float density = static_cast<float>(*propertyValue<double>(*values, "material.density"));
91 const float friction = static_cast<float>(*propertyValue<double>(*values, "material.friction"));
92 const float restitution = static_cast<float>(*propertyValue<double>(*values, "material.restitution"));
93 if (kind == "polygon")
94 fixture = body->newPolygonFixture(vertices, density, friction, restitution);
95 else if (kind == "chain")
96 fixture = body->newChainFixture(vertices, geometry.value().loop, friction, restitution);
97 } catch (const std::exception& exception) {
98 return bridgeError<physics::Fixture*>(EditorStatus::Rejected, "editor.physics.runtime-2d-asset-shape",
99 exception.what());
100 }
101 if (!fixture)
102 return bridgeError<physics::Fixture*>(EditorStatus::Failed, "editor.physics.runtime-create",
103 "Physics backend failed to create the 2D asset fixture");
104 fixture->setSensor(*propertyValue<bool>(*values, "shape.sensor"));
105 fixture->setCategoryBits(static_cast<int>(*propertyValue<int64_t>(*values, "collision.category")));
106 fixture->setMaskBits(static_cast<int>(*propertyValue<int64_t>(*values, "collision.mask")));
107 return eve::editing::applied<physics::Fixture*>(fixture);
108}

203EditorResult<void> PhysicsCollider3DRuntimeSink::publish(const PhysicsColliderTarget& candidate) {
204 if (!body_ || !body_->isValid() || candidate.describe().type != "physics-collider-3d")
205 return bridgeError<void>(EditorStatus::Rejected, "editor.physics.live-3d-input",
206 "A live Body3D and 3D collider candidate are required");
207 if (current_ && current_->isValid() && current_->getBody() != body_)
208 return bridgeError<void>(EditorStatus::Conflict, "editor.physics.live-shape-owner",
209 "Current collider shape belongs to another body");
210 const auto diagnostics = candidate.validate();
211 for (const EditorDiagnostic& diagnostic : diagnostics) {
212 if (diagnostic.severity() == DiagnosticSeverity::Error)
213 return EditorResult<void>::failure(eve::Status(EditorStatus::Rejected, diagnostics));
214 }
215 const EditorValue snapshot = candidate.snapshotValue();
216 const auto* root = snapshot.getIf<EditorValue::Object>();
217 const auto* values = root ? root->at("properties").getIf<EditorValue::Object>() : nullptr;
218 if (!values)
219 return bridgeError<void>(EditorStatus::Failed, "editor.physics.live-properties",
220 "Collider properties are unavailable");
221 const std::string kind = *propertyValue<std::string>(*values, "shape.kind");
222 const std::string desiredBodyType = *propertyValue<std::string>(*values, "body.type");
223 const std::string previousBodyType = body_->getType();
224 const bool currentRequiresStatic = current_ && current_->isValid() &&
225 (current_->getKind() == "triangleMesh" || current_->getKind() == "heightField");
226 if (currentRequiresStatic && desiredBodyType != "static" && desiredBodyType != previousBodyType)
227 return bridgeError<void>(EditorStatus::Unsupported, "editor.physics.live-complex-type-transition",
228 "Swap the complex collider to a primitive before changing its body type");
229 try {
230 if (previousBodyType != desiredBodyType) body_->setType(desiredBodyType);
231 } catch (const std::exception& exception) {
232 return bridgeError<void>(EditorStatus::Rejected, "editor.physics.live-body-type", exception.what());
233 }
234
235 EditorResult<physics::Shape3D*> built = assets_
236 ? PhysicsColliderRuntimeBuilder().build3D(candidate, body_, *assets_)
237 : PhysicsColliderRuntimeBuilder().build3D(candidate, body_);
238 if (!built.ok()) {
239 if (previousBodyType != desiredBodyType) {
240 try {
241 body_->setType(previousBodyType);
242 } catch (...) {
243 auto diagnostics = built.diagnostics();
244 diagnostics.push_back(eve::editing::ruleDiagnostic(
245 eve::DiagnosticCode::Failed, RuleId("editor.physics.live-rollback-body-type"),
246 DiagnosticSeverity::Error,
247 "Collider build failed and the previous body type could not be restored"));
248 return EditorResult<void>::failure(eve::Status(EditorStatus::Failed, std::move(diagnostics)));
249 }
250 }
251 return EditorResult<void>::failure(built.status());
252 }
253
254 physics::Shape3D* replacement = built.value();
255 if (current_ && current_->isValid()) current_->destroy();
256 current_ = replacement;
257 return eve::editing::applied<void>();
258}
