20const HouseComponent *pick(const std::vector<std::reference_wrapper<const HouseComponent>> &choices,
21 std::mt19937 &rng) {
22 if (choices.empty()) return nullptr;
23 int total = 0;
24 for (const auto &choice : choices) total += choice.get().weight;
25 std::uniform_int_distribution<int> roll(1, total);
26 int target = roll(rng);
27 for (const auto &choice : choices) {
28 target -= choice.get().weight;
29 if (target <= 0) return &choice.get();
30 }
31 return &choices.back().get();
32}

43 const std::vector<std::reference_wrapper<const HouseComponent>> &choices, bool wantWindow) {
44 bool hasWindow = false, hasSolid = false;
45 for (const auto &choice : choices) {

47 hasWindow = true;

83std::vector<uint8_t> footprintMask(const std::string &shape, int width, int depth, int inset) {
84 std::vector<uint8_t> mask(size_t(width * depth), 0);
85 const int minX = inset, minY = inset, maxX = width - 1 - inset, maxY = depth - 1 - inset;
86 if (minX > maxX || minY > maxY) return mask;
87 const int spanX = maxX - minX + 1, spanY = maxY - minY + 1;
88 const int cutX = minX + std::max(1, spanX / 2) - 1;
89 const int cutY = minY + std::max(1, spanY / 2) - 1;
90 const int stemInset = spanX >= 5 ? 1 : 0;
91 for (int y = minY; y <= maxY; ++y) for (int x = minX; x <= maxX; ++x) {
92 bool active = true;
93 if (shape == "l_shape") active = !(x > cutX && y > cutY);
94 else if (shape == "t_shape") active = y <= cutY || (x >= minX + stemInset && x <= maxX - stemInset);
95 if (active) mask[size_t(y * width + x)] = 1;
96 }
97 return mask;
98}

100bool active(const std::vector<uint8_t> &mask, int width, int depth, int x, int y) {
101 return x >= 0 && y >= 0 && x < width && y < depth && mask[size_t(y * width + x)] != 0;
102}

104SocketDirection rotate(SocketDirection direction, int degrees) {
105 if (direction == SocketDirection::Up || direction == SocketDirection::Down) return direction;
106 int side = direction == SocketDirection::North ? 0 : direction == SocketDirection::East ? 1 :
107 direction == SocketDirection::South ? 2 : 3;
108 side = (side + degrees / 90) % 4;
109 return side == 0 ? SocketDirection::North : side == 1 ? SocketDirection::East :
110 side == 2 ? SocketDirection::South : SocketDirection::West;
111}

113std::vector<std::reference_wrapper<const HouseComponent>> compatibleOnFace(
114 const std::vector<std::reference_wrapper<const HouseComponent>> &choices, SocketDirection face, int rotation) {
115 std::vector<std::reference_wrapper<const HouseComponent>> out;
116 for (const auto &choice : choices) {
117 const auto &c = choice.get();
118 // Components without sockets are intentionally wildcard-compatible. This preserves
119 // compatibility with small legacy kits while new player kits opt into strict sockets.
120 if (c.sockets.empty()) {
121 out.push_back(choice);
122 continue;
123 }
124 for (const auto &socket : c.sockets) {
125 if (rotate(socket.direction, rotation) == face && !socket.type.empty()) {
126 out.push_back(choice);
127 break;
128 }
129 }
130 }
131 return out;
132}

136eve::Result<void> HouseGenerator::generate(const HouseRequest &r, HouseLayout &out) const {
137 HouseLayout generated;
138 if (r.width < 3 || r.depth < 3 || r.floors < 1 || r.maxAttempts < 1) {
139 return failure<void>(eve::DiagnosticCode::InvalidArgument,
140 "house needs a 3x3 plot, at least one floor and one attempt");
141 }
142 if (!oneOf(r.footprint, {"auto", "rectangle", "l_shape", "t_shape"}) ||
143 !oneOf(r.roof, {"auto", "gable", "flat", "shed"}) ||
144 !oneOf(r.entrance, {"auto", "north", "east", "south", "west"})) {
145 return failure<void>(eve::DiagnosticCode::Unsupported, "unsupported footprint, roof or entrance mode");
146 }
147 const auto foundation = library_.byCategory("foundation", r.style);
148 const auto floor = library_.byCategory("floor", r.style);
149 const auto wall = library_.byCategory("wall", r.style);
150 const auto door = library_.byCategory("door", r.style);
151 const auto roof = library_.byCategory("roof", r.style);
152 if (!has(foundation) || !has(floor) || !has(wall) || !has(door) || !has(roof)) {
153 return failure<void>(eve::DiagnosticCode::NotFound,
154 "library needs foundation, floor, wall, door and roof categories");
155 }
156
157 std::mt19937 rng(r.seed);
158 generated.seed = r.seed;
159 generated.moduleSize = r.moduleSize;
160 generated.floorHeight = r.floorHeight;
161 static constexpr const char *shapes[] = {"rectangle", "l_shape", "t_shape"};
162 static constexpr const char *roofs[] = {"gable", "flat", "shed"};
163 static constexpr SocketDirection sides[] = {SocketDirection::North, SocketDirection::East,
164 SocketDirection::South, SocketDirection::West};
165 generated.footprintStyle = r.footprint == "auto" ? shapes[rng() % 3] : r.footprint;
166 generated.roofStyle = r.roof == "auto" ? roofs[rng() % 3] : r.roof;
167 const SocketDirection entranceDirection =
168 r.entrance == "auto" ? sides[rng() % 4] : directionFromName(r.entrance);
169 generated.entranceSide = directionName(entranceDirection);
170
171 // Grammar pass: construct a connected footprint mask, then emit one module per exposed face.
172 // Upper masks are monotonically inset, preserving a direct vertical support chain.
173 int inset = 0;
174 std::vector<uint8_t> previousMask;
175 for (int z = 0; z < r.floors; ++z) {
176 // Upper floors may step inward, but can never expand again above an inset floor. This
177 // gives every floor cell a direct support chain to the foundation.
178 if (z > 0 && inset == 0 && r.width > 4 && r.depth > 4 && (rng() & 3u) == 0u)
179 inset = 1;
180 const auto mask = footprintMask(generated.footprintStyle, r.width, r.depth, inset);
181 // Any lower-floor cell not covered by this floor becomes a roof terrace/canopy at the
182 // current level. Thus every floor cell is covered by either another floor or a roof.
183 if (!previousMask.empty()) {
184 for (int y = 0; y < r.depth; ++y) for (int x = 0; x < r.width; ++x) {
185 if (active(previousMask, r.width, r.depth, x, y) &&
186 !active(mask, r.width, r.depth, x, y))
187 generated.instances.push_back({pick(roof, rng)->id, x, y, z, 0});
188 }
189 }
190 int minX = r.width, minY = r.depth, maxX = -1, maxY = -1;
191 for (int y = 0; y < r.depth; ++y) for (int x = 0; x < r.width; ++x) if (active(mask, r.width, r.depth, x, y)) {
192 minX = std::min(minX, x); minY = std::min(minY, y);
193 maxX = std::max(maxX, x); maxY = std::max(maxY, y);
194 if (z == 0) generated.instances.push_back({pick(foundation, rng)->id, x, y, z, 0});
195 generated.instances.push_back({pick(floor, rng)->id, x, y, z, 0});
196 }
197 if (maxX < minX || maxY < minY)
198 return failure<void>(eve::DiagnosticCode::Failed, "footprint collapsed after inset");
199 generated.rooms.push_back({z == 0 ? "living" : "upper", minX, minY, maxX - minX + 1, maxY - minY + 1});
200
201 using Face = std::tuple<int, int, SocketDirection>;
202 std::vector<Face> faces;
203 std::vector<Face> entranceCandidates;
204 std::unordered_map<int, int> facesPerCell;
205 for (int y = 0; y < r.depth; ++y) for (int x = 0; x < r.width; ++x) {
206 if (!active(mask, r.width, r.depth, x, y)) continue;
207 const SocketDirection directions[] = {SocketDirection::North, SocketDirection::East,
208 SocketDirection::South, SocketDirection::West};
209 const int dx[] = {0, 1, 0, -1};
210 const int dy[] = {-1, 0, 1, 0};
211 for (int side = 0; side < 4; ++side) if (!active(mask, r.width, r.depth, x + dx[side], y + dy[side])) {
212 Face face{x, y, directions[side]};
213 faces.push_back(face);
214 ++facesPerCell[y * r.width + x];
215 if (z == 0 && directions[side] == entranceDirection) entranceCandidates.push_back(face);
216 }
217 }
218 std::sort(entranceCandidates.begin(), entranceCandidates.end(), [](const Face &a, const Face &b) {
219 return std::tie(std::get<1>(a), std::get<0>(a)) < std::tie(std::get<1>(b), std::get<0>(b));
220 });
221 Face entranceFace{-1, -1, entranceDirection};
222 if (z == 0 && !entranceCandidates.empty()) entranceFace = entranceCandidates[entranceCandidates.size() / 2];
223 for (const auto &faceData : faces) {
224 const auto [x, y, face] = faceData;
225 const bool isEntrance = z == 0 && faceData == entranceFace;
226 const int rotation = directionRotation(face);
227 auto choices = compatibleOnFace(isEntrance ? door : wall, face, rotation);
228 if (!isEntrance) {
229 const bool corner = facesPerCell[y * r.width + x] > 1;
230 const bool besideEntrance = z == 0 &&
231 std::abs(x - std::get<0>(entranceFace)) + std::abs(y - std::get<1>(entranceFace)) <= 1;
232 const int facadeAxis = (face == SocketDirection::North || face == SocketDirection::South) ? x : y;
233 const bool rhythmicWindow = ((facadeAxis + int(r.seed & 1u)) & 1) == 0;
234 choices = facadeVariant(choices, !corner && !besideEntrance && rhythmicWindow);
235 }
236 const auto *selected = pick(choices, rng);
237 if (!selected) {
238 return failure<void>(
239 eve::DiagnosticCode::NotFound,
240 "no socket-compatible " + std::string(isEntrance ? "door" : "wall") + " component");
241 }
242 generated.instances.push_back({selected->id, x, y, z, rotation});
243 }
244 if (z + 1 == r.floors) {
245 for (int y = 0; y < r.depth; ++y) for (int x = 0; x < r.width; ++x)
246 if (active(mask, r.width, r.depth, x, y))
247 generated.instances.push_back({pick(roof, rng)->id, x, y, z + 1, 0});
248 }
249 previousMask = mask;
250 }
251 if (!r.requiredRooms.empty() && r.requiredRooms.front() != "living")
252 generated.diagnostics.push_back(
253 "requested room types beyond living/upper are approximated by the base grammar");
254 auto validated = generated.validate(library_);
255 if (!validated.ok()) return validated;
256 out = std::move(generated);
257 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
258}
