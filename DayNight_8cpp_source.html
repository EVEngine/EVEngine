29// Convert an elevation/azimuth to a unit direction pointing at the sun.
30// azimuth measured clockwise from +Z, elevation above the horizon.
31inline void sunDirection(float elevDeg, float azimDeg, float &dx, float &dy, float &dz) {
32 const float el = deg2rad(elevDeg);
33 const float az = deg2rad(azimDeg);
34 const float he = std::cos(el);
35 dx = he * std::sin(az);
36 dy = std::sin(el);
37 dz = he * std::cos(az);
38}

40// Tiny deterministic hash for stars (no <random> dependency).
41inline uint32_t hash13(uint32_t x) {
42 x ^= x >> 16;
43 x *= 0x7feb352du;
44 x ^= x >> 15;
45 x *= 0x846ca68bu;
46 x ^= x >> 16;
47 return x;
48}
49inline float hashUnit(uint32_t x) { return float(hash13(x) % 10000u) / 9999.f; }

51// Fill one cubemap face's RGBA. `face` in {0..5} order +X,-X,+Y,-Y,+Z,-Z.
52// dirAt(x,y) writes the world direction (unnormalized ok) for a pixel.
53void fillSkyFace(std::vector<uint8_t> &px, int size, int face,
54 const float sunDir[3], float sunEnergy, float nightAmount,
55 void (*dirAt)(int face, int size, int x, int y, float out[3])) {
56 const int n = size;
57 for (int y = 0; y < n; ++y) {
58 for (int x = 0; x < n; ++x) {
59 float d[3];
60 dirAt(face, n, x, y, d);
61 const float len = std::sqrt(d[0] * d[0] + d[1] * d[1] + d[2] * d[2]);
62 if (len < 1e-6f) { d[0] = 0.f; d[1] = 1.f; d[2] = 0.f; }
63 else { d[0] /= len; d[1] /= len; d[2] /= len; }
64
65 const float up = d[1]; // -1 horizon .. +1 zenith
66 // Sky gradient: blue zenith, pale horizon.
67 float r, g, b;
68 if (nightAmount > 0.5f) {
69 // Deep night navy, slightly lifted at horizon.
70 const float t = 0.5f + 0.5f * up;
71 r = 0.015f + 0.03f * t;
72 g = 0.025f + 0.045f * t;
73 b = 0.07f + 0.10f * t;
74 } else {
75 const float t = 0.5f + 0.5f * up;
76 r = 0.35f + 0.30f * t;
77 g = 0.55f + 0.25f * t;
78 b = 0.75f + 0.20f * t;
79 }
80 // Blend the two regimes by night amount.
81 if (nightAmount > 0.f && nightAmount < 1.f) {
82 float nr = 0.015f + 0.03f * (0.5f + 0.5f * up);
83 float ng = 0.025f + 0.045f * (0.5f + 0.5f * up);
84 float nb = 0.07f + 0.10f * (0.5f + 0.5f * up);
85 r = r * (1.f - nightAmount) + nr * nightAmount;
86 g = g * (1.f - nightAmount) + ng * nightAmount;
87 b = b * (1.f - nightAmount) + nb * nightAmount;
88 }
89
90 // Sun disc: a tight highlight around the sun direction.
91 const float dot = d[0] * sunDir[0] + d[1] * sunDir[1] + d[2] * sunDir[2];
92 const float disc = std::pow(std::max(0.f, dot), 400.f) * sunEnergy;
93 r += disc * 1.0f;
94 g += disc * 0.95f;
95 b += disc * 0.85f;
96 // Broad glow near the sun.
97 const float glow = std::pow(std::max(0.f, dot), 8.f) * sunEnergy * 0.25f;
98 r += glow * 1.0f; g += glow * 0.9f; b += glow * 0.7f;
99
100 // Stars (only at night, only in the sky hemisphere, avoid the sun).
101 float star = 0.f;
102 if (nightAmount > 0.5f && up > 0.05f && dot < 0.98f) {
103 uint32_t h = hash13(uint32_t((x * 73856093) ^ (y * 19349663) ^ (face * 83492791)));
104 if (h % 40u == 0u) {
105 const float tw = 0.6f + 0.4f * hashUnit(h + 1u);
106 star = nightAmount * tw;
107 }
108 }
109 r += star * 0.9f; g += star * 0.95f; b += star * 1.0f;
110
111 const int i = (y * n + x) * 4;
112 px[i + 0] = uint8_t(std::min(255.f, r * 255.f));
113 px[i + 1] = uint8_t(std::min(255.f, g * 255.f));
114 px[i + 2] = uint8_t(std::min(255.f, b * 255.f));
115 px[i + 3] = 255;
116 }
117 }
118}

120// Standard cubemap direction mapping for each face (u,v in [0,size]).
121void cubeDir(int face, int size, int x, int y, float out[3]) {
122 const float u = (2.f * (float(x) + 0.5f) / float(size)) - 1.f; // -1..1
123 const float v = (2.f * (float(y) + 0.5f) / float(size)) - 1.f; // -1..1
124 switch (face) {
125 case 0: out[0] = 1.f; out[1] = -v; out[2] = -u; break; // +X
126 case 1: out[0] = -1.f; out[1] = -v; out[2] = u; break; // -X
127 case 2: out[0] = u; out[1] = 1.f; out[2] = v; break; // +Y
128 case 3: out[0] = u; out[1] = -1.f; out[2] = -v; break; // -Y
129 case 4: out[0] = u; out[1] = -v; out[2] = 1.f; break; // +Z
130 default: out[0] = -u; out[1] = -v; out[2] = -1.f; break; // -Z
131 }

315void DayNight::update(float dt, graphics::Graphics *gfx) {
316 if (!impl_->built) init(gfx);
317 if (!impl_->gfx) return;
318
319 if (!impl_->paused) {
320 impl_->timeOfDay += dt * impl_->speed;
321 impl_->timeOfDay = std::fmod(impl_->timeOfDay, 24.f);
322 if (impl_->timeOfDay < 0.f) impl_->timeOfDay += 24.f;
323 }
324 const float hours = impl_->timeOfDay;
325
326 // Solar elevation: sine curve peaking at noon (hours=12).
327 const float frac = (hours - 6.f) / 12.f; // -1 at 6h, 0 at 12h, +1 at 18h
328 const float elevDeg = kMaxElevationDeg * std::sin(kPi * frac);
329 const float azimDeg = (hours / 24.f) * 360.f; // full rotation per day
330 impl_->elevDeg = elevDeg;
331 impl_->azimDeg = azimDeg;
332
333 // Sun energy: ramps up a few degrees above the horizon.
334 impl_->sunEnergy = std::clamp((elevDeg + 6.f) / 14.f, 0.f, 1.f);
335 const float nightAmount = std::clamp((-elevDeg) / 12.f, 0.f, 1.f);
336
337 sunDirection(elevDeg, azimDeg, impl_->sunDir[0], impl_->sunDir[1], impl_->sunDir[2]);
338
339 // Push the directional sun (replaces the legacy directional when no other
340 // dir Light3D is active; we keep moon as a Light3D instead so it can have
341 // different color/intensity than the sun slot).
342 gfx->setDirectionalLight(impl_->sunDir[0], impl_->sunDir[1], impl_->sunDir[2],
343 1.0f * impl_->sunEnergy, 0.9f * impl_->sunEnergy,
344 0.8f * impl_->sunEnergy);
345
346 // Background matches the sky at the horizon for the clear color.
347 const float skyR = getSkyR(), skyG = getSkyG(), skyB = getSkyB();
348 gfx->setBackgroundColorRGBA(skyR, skyG, skyB, 1.f);
349
350 // --- procedural skybox (IBL env), regenerated per sun bucket ---
351 if (impl_->skyboxEnabled) {
352 const int bucket = int(elevDeg) + int(azimDeg / 4.f) * 1000;
353 if (bucket != impl_->lastSkyBucket) {
354 impl_->lastSkyBucket = bucket;
355 std::vector<uint8_t> faces(
356 size_t(kSkyCubeSize) * size_t(kSkyCubeSize) * 4 * 6);
357 for (int f = 0; f < 6; ++f) {
358 std::vector<uint8_t> face(
359 size_t(kSkyCubeSize) * size_t(kSkyCubeSize) * 4);
360 fillSkyFace(face, int(kSkyCubeSize), f, impl_->sunDir,
361 impl_->sunEnergy, nightAmount, cubeDir);
362 std::memcpy(faces.data() + size_t(f) * face.size(), face.data(), face.size());
363 }
364 // Replace the previous env cube; Graphics owns old textures.
365 impl_->skyCube = gfx->newCubemap(int(kSkyCubeSize), faces.data());
366 gfx->setMesh3DEnv(impl_->skyCube, 0.5f + 0.5f * impl_->sunEnergy);
367 }
368 }
369
370 // --- night light systems (only meaningful below the horizon) ---
371 const bool night = elevDeg < 0.f;
372
373 // Moonlight: a directional light at the opposite-ish angle of the sun.
374 if (impl_->moonLight) {
375 const bool on = night && impl_->nightLight[0];
376 impl_->moonLight->setEnabled(on);
377 if (on) {
378 impl_->moonLight->setDirection(-impl_->sunDir[0], -impl_->sunDir[1],
379 -impl_->sunDir[2]);
380 }
381 }
382
383 // Fire.
384 if (impl_->fireLight) {
385 impl_->fireLight->setEnabled(night && impl_->nightLight[2]);
386 }
387
388 // Fireflies: gentle sinusoidal drift.
389 if (impl_->flyLights.size() == impl_->flies.size()) {
390 for (size_t i = 0; i < impl_->flies.size(); ++i) {
391 graphics::Light3D *l = impl_->flyLights[i];
392 const Impl::Fly &f = impl_->flies[i];
393 const bool on = night && impl_->nightLight[3];
394 l->setEnabled(on);
395 if (on) {
396 const float t = impl_->timeOfDay + f.seed;
397 l->setPosition(f.x + std::sin(t * 0.9f) * 0.6f,
398 f.y + std::sin(t * 1.3f + 1.7f) * 0.4f,
399 f.z + std::cos(t * 0.8f) * 0.6f);
400 }
401 }
402 }
403}
