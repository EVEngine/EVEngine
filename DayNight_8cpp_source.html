30inline float smoothstep(float e0, float e1, float x) {
31 if (std::fabs(e1 - e0) < 1e-7f) return x < e0 ? 0.f : 1.f;
32 const float t = std::clamp((x - e0) / (e1 - e0), 0.f, 1.f);
33 return t * t * (3.f - 2.f * t);

44// Compact single-scattering atmosphere approximation. The wavelength-dependent
45// coefficients preserve the important physical relationships (blue Rayleigh sky,
46// neutral Mie haze, warm attenuated low sun) while remaining cheap enough to build
47// the IBL cubemap on the CPU. It is deliberately isolated so a future GPU LUT
48// implementation can consume the same public atmosphere parameters.
49Vec3 atmosphereRadiance(Vec3 view, Vec3 sun, float turbidity, float mieStrength) {
50 const float mu = std::clamp(view.x * sun.x + view.y * sun.y + view.z * sun.z, -1.f, 1.f);
51 const float horizonMass = 1.f / std::max(0.08f, view.y + 0.14f);
52 const float sunMass = 1.f / std::max(0.06f, sun.y + 0.12f);
53 const Vec3 betaR{0.055f, 0.130f, 0.285f};
54 const Vec3 betaM = scale(Vec3{0.18f, 0.17f, 0.15f}, mieStrength * (0.35f + turbidity * 0.09f));
55 const Vec3 extinction = add(betaR, betaM);
56 const Vec3 viewT{std::exp(-extinction.x * horizonMass),
57 std::exp(-extinction.y * horizonMass),
58 std::exp(-extinction.z * horizonMass)};
59 const Vec3 sunT{std::exp(-extinction.x * sunMass),
60 std::exp(-extinction.y * sunMass),
61 std::exp(-extinction.z * sunMass)};
62 const float rayleighPhase = 3.f / (16.f * kPi) * (1.f + mu * mu);
63 const float g = std::clamp(0.72f + turbidity * 0.008f, 0.72f, 0.82f);
64 const float gg = g * g;
65 const float miePhase = (1.f - gg) /
66 (4.f * kPi * std::pow(std::max(0.015f, 1.f + gg - 2.f * g * mu), 1.5f));
67 const Vec3 scatter = add(scale(betaR, rayleighPhase * 13.f),
68 scale(betaM, miePhase * 2.2f));
69 Vec3 sky = mul(mul(scatter, sunT), Vec3{1.f - viewT.x, 1.f - viewT.y, 1.f - viewT.z});
70 // Multiple-scattering floor prevents a black antisolar horizon and approximates
71 // light returned by the ground/atmosphere without an expensive integral.
72 sky = add(sky, scale(mul(sunT, Vec3{0.18f, 0.22f, 0.30f}),
73 0.12f + 0.18f * (1.f - std::max(view.y, 0.f))));
74 return sky;
75}

77Vec3 toneMapSky(Vec3 c, float exposure) {
78 c = scale(c, std::max(exposure, 0.01f));
79 // ACES fitted curve, followed by display gamma. Keeps a smooth solar halo in
80 // the current RGBA8 backend rather than clipping radiance before conversion.
81 auto channel = [](float x) {
82 x = std::clamp((x * (2.51f * x + 0.03f)) /
83 (x * (2.43f * x + 0.59f) + 0.14f), 0.f, 1.f);
84 return std::pow(x, 1.f / 2.2f);
85 };
86 return {channel(c.x), channel(c.y), channel(c.z)};
87}

89Vec3 attenuatedSunColor(float sunElevation, float turbidity, float mieStrength) {
90 const float mass = 1.f / std::max(0.06f, sunElevation + 0.12f);
91 const Vec3 extinction = add(Vec3{0.055f, 0.130f, 0.285f},
92 scale(Vec3{0.18f, 0.17f, 0.15f}, mieStrength * (0.35f + turbidity * 0.09f)));
93 Vec3 c{std::exp(-extinction.x * mass), std::exp(-extinction.y * mass),
94 std::exp(-extinction.z * mass)};
95 const float maxChannel = std::max({c.x, c.y, c.z, 1e-5f});
96 return scale(c, 1.f / maxChannel);
97}

99// Convert an elevation/azimuth to a unit direction pointing at the sun.
100// azimuth measured clockwise from +Z, elevation above the horizon.
101inline void sunDirection(float elevDeg, float azimDeg, float &dx, float &dy, float &dz) {
102 const float el = deg2rad(elevDeg);
103 const float az = deg2rad(azimDeg);
104 const float he = std::cos(el);
105 dx = he * std::sin(az);
106 dy = std::sin(el);
107 dz = he * std::cos(az);
108}

111inline uint32_t hash13(uint32_t x) {
112 x ^= x >> 16;
113 x *= 0x7feb352du;
114 x ^= x >> 15;
115 x *= 0x846ca68bu;
116 x ^= x >> 16;
117 return x;

121// Fill one cubemap face's RGBA. `face` in {0..5} order +X,-X,+Y,-Y,+Z,-Z.
122// dirAt(x,y) writes the world direction (unnormalized ok) for a pixel.
123void fillSkyFace(std::vector<uint8_t> &px, int size, int face,
124 const float sunDir[3], float sunEnergy, float nightAmount,
125 float turbidity, float mieStrength, float exposure, float cloudiness,
126 void (*dirAt)(int face, int size, int x, int y, float out[3])) {
127 const int n = size;
128 for (int y = 0; y < n; ++y) {
129 for (int x = 0; x < n; ++x) {
130 float d[3];
131 dirAt(face, n, x, y, d);
132 const float len = std::sqrt(d[0] * d[0] + d[1] * d[1] + d[2] * d[2]);
133 if (len < 1e-6f) { d[0] = 0.f; d[1] = 1.f; d[2] = 0.f; }
134 else { d[0] /= len; d[1] /= len; d[2] /= len; }
135
136 const float up = d[1];
137 const Vec3 view{d[0], std::max(d[1], 0.002f), d[2]};
138 const Vec3 sun{sunDir[0], sunDir[1], sunDir[2]};
139 Vec3 day = atmosphereRadiance(view, sun, turbidity, mieStrength);
140 Vec3 night{0.006f + 0.012f * std::max(up, 0.f),
141 0.010f + 0.020f * std::max(up, 0.f),
142 0.035f + 0.070f * std::max(up, 0.f)};
143 Vec3 color = add(scale(day, (1.f - nightAmount) * sunEnergy),
144 scale(night, nightAmount));
145
146 // Sun disc: a tight highlight around the sun direction.
147 const float dot = d[0] * sunDir[0] + d[1] * sunDir[1] + d[2] * sunDir[2];
148 // Physical solar angular radius is about 0.27 degrees. Slightly enlarge it
149 // to remain stable in a 128px cubemap; the Mie term supplies the broad halo.
150 const float disc = smoothstep(std::cos(deg2rad(0.65f)),
151 std::cos(deg2rad(0.35f)), dot) * sunEnergy;
152 const float sunMass = 1.f / std::max(0.06f, sun.y + 0.12f);
153 const Vec3 sunColor{std::exp(-0.16f * sunMass), std::exp(-0.28f * sunMass),
154 std::exp(-0.58f * sunMass)};
155 color = add(color, scale(sunColor, disc * 12.f));
156
157 // Stars (only at night, only in the sky hemisphere, avoid the sun).
158 float star = 0.f;
159 if (nightAmount > 0.5f && up > 0.05f && dot < 0.98f) {
160 uint32_t h = hash13(uint32_t((x * 73856093) ^ (y * 19349663) ^ (face * 83492791)));
161 if (h % 40u == 0u) {
162 const float tw = 0.6f + 0.4f * hashUnit(h + 1u);
163 star = nightAmount * tw;
164 }
165 }
166 color = add(color, scale(Vec3{0.9f, 0.95f, 1.f}, star * (1.f - cloudiness)));
167 const Vec3 overcast{0.075f + 0.035f * std::max(up, 0.f),
168 0.090f + 0.045f * std::max(up, 0.f),
169 0.125f + 0.060f * std::max(up, 0.f)};
170 color = add(scale(color, 1.f - cloudiness * 0.82f),
171 scale(overcast, cloudiness));
172 color = toneMapSky(color, exposure);
173
174 const int i = (y * n + x) * 4;
175 px[i + 0] = uint8_t(std::clamp(color.x * 255.f, 0.f, 255.f));
176 px[i + 1] = uint8_t(std::clamp(color.y * 255.f, 0.f, 255.f));
177 px[i + 2] = uint8_t(std::clamp(color.z * 255.f, 0.f, 255.f));
178 px[i + 3] = 255;
179 }
180 }

183// Standard cubemap direction mapping for each face (u,v in [0,size]).
184void cubeDir(int face, int size, int x, int y, float out[3]) {
185 const float u = (2.f * (float(x) + 0.5f) / float(size)) - 1.f; // -1..1
186 const float v = (2.f * (float(y) + 0.5f) / float(size)) - 1.f; // -1..1
187 switch (face) {
188 case 0: out[0] = 1.f; out[1] = -v; out[2] = -u; break; // +X
189 case 1: out[0] = -1.f; out[1] = -v; out[2] = u; break; // -X
190 case 2: out[0] = u; out[1] = 1.f; out[2] = v; break; // +Y
191 case 3: out[0] = u; out[1] = -1.f; out[2] = -v; break; // -Y
192 case 4: out[0] = u; out[1] = -v; out[2] = 1.f; break; // +Z
193 default: out[0] = -u; out[1] = -v; out[2] = -1.f; break; // -Z
194 }

362void DayNight::applyReflectionProbeSky(graphics::ReflectionProbeCapture *probe) const {
363 if (!probe) return;
364 static constexpr Vec3 directions[6] = {
365 {1.f, 0.f, 0.f}, {-1.f, 0.f, 0.f}, {0.f, 1.f, 0.f},
366 {0.f, -1.f, 0.f}, {0.f, 0.f, 1.f}, {0.f, 0.f, -1.f},
367 };
368 const Vec3 sun{impl_->sunDir[0], impl_->sunDir[1], impl_->sunDir[2]};
369 const float nightAmount = std::clamp((-impl_->elevDeg) / 12.f, 0.f, 1.f);
370 for (int face = 0; face < 6; ++face) {
371 const Vec3 direction = directions[face];
372 const float up = direction.y;
373 const Vec3 view{direction.x, std::max(direction.y, 0.002f), direction.z};
374 const Vec3 day = atmosphereRadiance(view, sun, impl_->turbidity, impl_->mieStrength);
375 const Vec3 night{0.006f + 0.012f * std::max(up, 0.f),
376 0.010f + 0.020f * std::max(up, 0.f),
377 0.035f + 0.070f * std::max(up, 0.f)};
378 Vec3 color = add(scale(day, (1.f - nightAmount) * impl_->sunEnergy),
379 scale(night, nightAmount));
380 const Vec3 overcast{0.075f + 0.035f * std::max(up, 0.f),
381 0.090f + 0.045f * std::max(up, 0.f),
382 0.125f + 0.060f * std::max(up, 0.f)};
383 color = add(scale(color, 1.f - impl_->weatherCloudiness * 0.82f),
384 scale(overcast, impl_->weatherCloudiness));
385 color = add(color, scale(Vec3{0.75f, 0.90f, 1.20f}, impl_->weatherFlash));
386 color = scale(color, impl_->skyExposure);
387 probe->setSkyFaceColor(face, color.x, color.y, color.z);
388 probe->setSkyFaceTexture(face, impl_->skyFaces[static_cast<size_t>(face)]);
389 const float linearLuminance =
390 color.x * 0.2126f + color.y * 0.7152f + color.z * 0.0722f;
391 const float encodedLuminance =
392 impl_->skyFaceCenterLuminance[static_cast<size_t>(face)];
393 probe->setSkyFaceTextureScale(
394 face, encodedLuminance > 1e-4f ? linearLuminance / encodedLuminance : 1.f);
395 }
396 probe->setEnvironmentLighting(impl_->skyCube, 0.5f + 0.5f * impl_->sunEnergy);
397}

437void DayNight::addFirefly(float x, float y, float z) {
438 if (int(impl_->flies.size()) >= kMaxFireflies) return;
439 Impl::Fly f;
440 f.x = x; f.y = y; f.z = z;
441 f.seed = float(impl_->flies.size()) * 1.7f;
442 impl_->flies.push_back(f);

492void DayNight::update(float dt, graphics::Graphics *gfx) {
493 if (!impl_->built) init(gfx);
494 if (!impl_->gfx) return;
495
496 if (!impl_->paused) {
497 impl_->timeOfDay += dt * impl_->speed;
498 impl_->timeOfDay = std::fmod(impl_->timeOfDay, 24.f);
499 if (impl_->timeOfDay < 0.f) impl_->timeOfDay += 24.f;
500 }
501 const float hours = impl_->timeOfDay;
502
503 // Solar elevation: sine curve peaking at noon (hours=12).
504 const float frac = (hours - 6.f) / 12.f; // -1 at 6h, 0 at 12h, +1 at 18h
505 const float elevDeg = kMaxElevationDeg * std::sin(kPi * frac);
506 const float azimDeg = (hours / 24.f) * 360.f; // full rotation per day
507 impl_->elevDeg = elevDeg;
508 impl_->azimDeg = azimDeg;
509
510 // Sun energy: ramps up a few degrees above the horizon.
511 impl_->sunEnergy = std::clamp((elevDeg + 6.f) / 14.f, 0.f, 1.f);
512 const float nightAmount = std::clamp((-elevDeg) / 12.f, 0.f, 1.f);
513
514 sunDirection(elevDeg, azimDeg, impl_->sunDir[0], impl_->sunDir[1], impl_->sunDir[2]);
515
516 // Push the directional sun (replaces the legacy directional when no other
517 // dir Light3D is active; we keep moon as a Light3D instead so it can have
518 // different color/intensity than the sun slot).
519 const Vec3 directSun = attenuatedSunColor(impl_->sunDir[1], impl_->turbidity,
520 impl_->mieStrength);
521 const float weatherSun = 1.f - 0.82f * impl_->weatherCloudiness;
522 const Vec3 flashLight{impl_->weatherFlash * 0.75f, impl_->weatherFlash * 0.90f,
523 impl_->weatherFlash * 1.20f};
524 gfx->setDirectionalLight(impl_->sunDir[0], impl_->sunDir[1], impl_->sunDir[2],
525 directSun.x * impl_->sunEnergy * weatherSun + flashLight.x,
526 directSun.y * impl_->sunEnergy * weatherSun + flashLight.y,
527 directSun.z * impl_->sunEnergy * weatherSun + flashLight.z);
528
529 // Background matches the sky at the horizon for the clear color.
530 const float skyR = getSkyR(), skyG = getSkyG(), skyB = getSkyB();
531 gfx->setBackgroundColorRGBA(skyR, skyG, skyB, 1.f);
532
533 // --- procedural skybox (IBL env), regenerated per sun bucket ---
534 if (impl_->skyboxEnabled) {
535 const int bucket = int(elevDeg) + int(azimDeg / 4.f) * 1000 +
536 int(impl_->weatherCloudiness * 10.f) * 100000;
537 if (bucket != impl_->lastSkyBucket) {
538 impl_->lastSkyBucket = bucket;
539 std::vector<uint8_t> faces(
540 size_t(kSkyCubeSize) * size_t(kSkyCubeSize) * 4 * 6);
541 for (int f = 0; f < 6; ++f) {
542 std::vector<uint8_t> face(
543 size_t(kSkyCubeSize) * size_t(kSkyCubeSize) * 4);
544 fillSkyFace(face, int(kSkyCubeSize), f, impl_->sunDir,
545 impl_->sunEnergy, nightAmount, impl_->turbidity,
546 impl_->mieStrength, impl_->skyExposure,
547 impl_->weatherCloudiness, cubeDir);
548 impl_->skyFaces[static_cast<size_t>(f)] =
549 gfx->newTexture(int(kSkyCubeSize), int(kSkyCubeSize), face.data());
550 const size_t center =
551 (size_t(kSkyCubeSize / 2) * size_t(kSkyCubeSize) + size_t(kSkyCubeSize / 2)) *
552 4u;
553 impl_->skyFaceCenterLuminance[static_cast<size_t>(f)] =
554 (float(face[center]) * 0.2126f + float(face[center + 1]) * 0.7152f +
555 float(face[center + 2]) * 0.0722f) /
556 255.f;
557 std::memcpy(faces.data() + size_t(f) * face.size(), face.data(), face.size());
558 }
559 // Replace the previous env cube; Graphics owns old textures.
560 impl_->skyCube = gfx->newCubemap(int(kSkyCubeSize), faces.data(),
561 graphics::TextureCreateInfo::withMipmaps(true));
562 gfx->setMesh3DEnv(impl_->skyCube, 0.5f + 0.5f * impl_->sunEnergy);
563 }
564 }
565
566 // --- night light systems (only meaningful below the horizon) ---
567 const bool night = elevDeg < 0.f;
568
569 // Moonlight: a directional light at the opposite-ish angle of the sun.
570 if (impl_->moonLight) {
571 const bool on = night && impl_->nightLight[0];
572 impl_->moonLight->setEnabled(on);
573 if (on) {
574 impl_->moonLight->setDirection(-impl_->sunDir[0], -impl_->sunDir[1],
575 -impl_->sunDir[2]);
576 }
577 }
578
579 // Fire.
580 if (impl_->fireLight) {
581 impl_->fireLight->setEnabled(night && impl_->nightLight[2]);
582 }
583
584 // Fireflies: gentle sinusoidal drift.
585 if (impl_->flyLights.size() == impl_->flies.size()) {
586 for (size_t i = 0; i < impl_->flies.size(); ++i) {
587 graphics::Light3D *l = impl_->flyLights[i];
588 const Impl::Fly &f = impl_->flies[i];
589 const bool on = night && impl_->nightLight[3];
590 l->setEnabled(on);
591 if (on) {
592 const float t = impl_->timeOfDay + f.seed;
593 l->setPosition(f.x + std::sin(t * 0.9f) * 0.6f,
594 f.y + std::sin(t * 1.3f + 1.7f) * 0.4f,
595 f.z + std::cos(t * 0.8f) * 0.6f);
596 }
597 }
598 }
