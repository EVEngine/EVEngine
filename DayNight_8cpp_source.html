44// Compact single-scattering atmosphere approximation. The wavelength-dependent
45// coefficients preserve the important physical relationships (blue Rayleigh sky,
46// neutral Mie haze, warm attenuated low sun) while remaining cheap enough to build
47// the IBL cubemap on the CPU. It is deliberately isolated so a future GPU LUT
48// implementation can consume the same public atmosphere parameters.
49Vec3 atmosphereRadiance(Vec3 view, Vec3 sun, float turbidity, float mieStrength) {
50 const float mu = std::clamp(view.x * sun.x + view.y * sun.y + view.z * sun.z, -1.f, 1.f);
51 const float horizonMass = 1.f / std::max(0.08f, view.y + 0.14f);
52 const float sunMass = 1.f / std::max(0.06f, sun.y + 0.12f);
53 const Vec3 betaR{0.055f, 0.130f, 0.285f};
54 const Vec3 betaM = scale(Vec3{0.18f, 0.17f, 0.15f}, mieStrength * (0.35f + turbidity * 0.09f));
55 const Vec3 extinction = add(betaR, betaM);
56 const Vec3 viewT{std::exp(-extinction.x * horizonMass),
57 std::exp(-extinction.y * horizonMass),
58 std::exp(-extinction.z * horizonMass)};
59 const Vec3 sunT{std::exp(-extinction.x * sunMass),
60 std::exp(-extinction.y * sunMass),
61 std::exp(-extinction.z * sunMass)};
62 const float rayleighPhase = 3.f / (16.f * kPi) * (1.f + mu * mu);
63 const float g = std::clamp(0.72f + turbidity * 0.008f, 0.72f, 0.82f);
64 const float gg = g * g;
65 const float miePhase = (1.f - gg) /
66 (4.f * kPi * std::pow(std::max(0.015f, 1.f + gg - 2.f * g * mu), 1.5f));
67 const Vec3 scatter = add(scale(betaR, rayleighPhase * 13.f),
68 scale(betaM, miePhase * 2.2f));
69 Vec3 sky = mul(mul(scatter, sunT), Vec3{1.f - viewT.x, 1.f - viewT.y, 1.f - viewT.z});
70 // Multiple-scattering floor prevents a black antisolar horizon and approximates
71 // light returned by the ground/atmosphere without an expensive integral.
72 sky = add(sky, scale(mul(sunT, Vec3{0.18f, 0.22f, 0.30f}),
73 0.12f + 0.18f * (1.f - std::max(view.y, 0.f))));
74 return sky;
75}

77Vec3 toneMapSky(Vec3 c, float exposure) {
78 c = scale(c, std::max(exposure, 0.01f));
79 // ACES fitted curve, followed by display gamma. Keeps a smooth solar halo in
80 // the current RGBA8 backend rather than clipping radiance before conversion.
81 auto channel = [](float x) {
82 x = std::clamp((x * (2.51f * x + 0.03f)) /
83 (x * (2.43f * x + 0.59f) + 0.14f), 0.f, 1.f);
84 return std::pow(x, 1.f / 2.2f);
85 };
86 return {channel(c.x), channel(c.y), channel(c.z)};
87}

139// Fill one cubemap face's RGBA. `face` in {0..5} order +X,-X,+Y,-Y,+Z,-Z.
140// dirAt(x,y) writes the world direction (unnormalized ok) for a pixel.
141void fillSkyFace(std::vector<uint8_t> &px, int size, int face,
142 const float sunDir[3], const Vec3 &directSunColor, float sunEnergy,
143 float nightAmount,
144 float turbidity, float mieStrength, float exposure, float cloudiness,
145 float rotationDegrees, const Vec3 &tint,
146 void (*dirAt)(int face, int size, int x, int y, float out[3])) {
147 const int n = size;
148 for (int y = 0; y < n; ++y) {
149 for (int x = 0; x < n; ++x) {
150 float d[3];
151 dirAt(face, n, x, y, d);
152 const float len = std::sqrt(d[0] * d[0] + d[1] * d[1] + d[2] * d[2]);
153 if (len < 1e-6f) { d[0] = 0.f; d[1] = 1.f; d[2] = 0.f; }
154 else { d[0] /= len; d[1] /= len; d[2] /= len; }
155 if (rotationDegrees != 0.f) {
156 const float angle = deg2rad(rotationDegrees);
157 const float x = d[0] * std::cos(angle) + d[2] * std::sin(angle);
158 d[2] = -d[0] * std::sin(angle) + d[2] * std::cos(angle);
159 d[0] = x;
160 }
161
162 const float up = d[1];
163 const Vec3 view{d[0], std::max(d[1], 0.002f), d[2]};
164 const Vec3 sun{sunDir[0], sunDir[1], sunDir[2]};
165 Vec3 day = atmosphereRadiance(view, sun, turbidity, mieStrength);
166 Vec3 night{0.006f + 0.012f * std::max(up, 0.f),
167 0.010f + 0.020f * std::max(up, 0.f),
168 0.035f + 0.070f * std::max(up, 0.f)};
169 Vec3 color = add(scale(day, (1.f - nightAmount) * sunEnergy),
170 scale(night, nightAmount));
171
172 // Sun disc: a tight highlight around the sun direction.
173 const float dot = d[0] * sunDir[0] + d[1] * sunDir[1] + d[2] * sunDir[2];
174 // Physical solar angular radius is about 0.27 degrees. Slightly enlarge it
175 // to remain stable in a 128px cubemap; the Mie term supplies the broad halo.
176 const float disc = smoothstep(std::cos(deg2rad(0.65f)),
177 std::cos(deg2rad(0.35f)), dot) * sunEnergy;
178 const float sunMass = 1.f / std::max(0.06f, sun.y + 0.12f);
179 const Vec3 atmosphericSun{std::exp(-0.16f * sunMass),
180 std::exp(-0.28f * sunMass),
181 std::exp(-0.58f * sunMass)};
182 const Vec3 sunColor{atmosphericSun.x * directSunColor.x,
183 atmosphericSun.y * directSunColor.y,
184 atmosphericSun.z * directSunColor.z};
185 color = add(color, scale(sunColor, disc * 12.f));
186
187 // Stars (only at night, only in the sky hemisphere, avoid the sun).
188 float star = 0.f;
189 if (nightAmount > 0.5f && up > 0.05f && dot < 0.98f) {
190 uint32_t h = hash13(uint32_t((x * 73856093) ^ (y * 19349663) ^ (face * 83492791)));
191 if (h % 40u == 0u) {
192 const float tw = 0.6f + 0.4f * hashUnit(h + 1u);
193 star = nightAmount * tw;
194 }
195 }
196 color = add(color, scale(Vec3{0.9f, 0.95f, 1.f}, star * (1.f - cloudiness)));
197 const Vec3 overcast{0.075f + 0.035f * std::max(up, 0.f),
198 0.090f + 0.045f * std::max(up, 0.f),
199 0.125f + 0.060f * std::max(up, 0.f)};
200 color = add(scale(color, 1.f - cloudiness * 0.82f),
201 scale(overcast, cloudiness));
202 color = {color.x * tint.x, color.y * tint.y, color.z * tint.z};
203 color = toneMapSky(color, exposure);
204
205 const int i = (y * n + x) * 4;
206 px[i + 0] = uint8_t(std::clamp(color.x * 255.f, 0.f, 255.f));
207 px[i + 1] = uint8_t(std::clamp(color.y * 255.f, 0.f, 255.f));
208 px[i + 2] = uint8_t(std::clamp(color.z * 255.f, 0.f, 255.f));
209 px[i + 3] = 255;
210 }
211 }

214// Standard cubemap direction mapping for each face (u,v in [0,size]).
215void cubeDir(int face, int size, int x, int y, float out[3]) {
216 const float u = (2.f * (float(x) + 0.5f) / float(size)) - 1.f; // -1..1
217 const float v = (2.f * (float(y) + 0.5f) / float(size)) - 1.f; // -1..1
218 switch (face) {
219 case 0: out[0] = 1.f; out[1] = -v; out[2] = -u; break; // +X
220 case 1: out[0] = -1.f; out[1] = -v; out[2] = u; break; // -X
221 case 2: out[0] = u; out[1] = 1.f; out[2] = v; break; // +Y
222 case 3: out[0] = u; out[1] = -1.f; out[2] = -v; break; // -Y
223 case 4: out[0] = u; out[1] = -v; out[2] = 1.f; break; // +Z
224 default: out[0] = -u; out[1] = -v; out[2] = -1.f; break; // -Z
225 }

459// Sky / ambient colors are functions of the sun energy and night amount.
460float DayNight::getSkyR() const {
461 const float angle = deg2rad(impl_->pcgSkybox.enabled ? impl_->pcgSkybox.rotationDegrees : 0.f);
462 const Vec3 c = toneMapSky(atmosphereRadiance({std::sin(angle), 0.04f, std::cos(angle)},
463 {impl_->sunDir[0], impl_->sunDir[1], impl_->sunDir[2]}, impl_->turbidity,
464 impl_->mieStrength), impl_->pcgSkybox.enabled ? impl_->pcgSkybox.exposure : impl_->skyExposure);
465 const float clear = c.x * impl_->sunEnergy + 0.012f * (1.f - impl_->sunEnergy);
466 const float result = clear * (1.f - impl_->weatherCloudiness * 0.72f) +
467 impl_->weatherCloudiness * 0.075f + impl_->weatherFlash * 0.38f;
468 return result * (impl_->pcgSkybox.enabled ? impl_->pcgSkybox.tintRed : 1.f);
469}
470float DayNight::getSkyG() const {
471 const float angle = deg2rad(impl_->pcgSkybox.enabled ? impl_->pcgSkybox.rotationDegrees : 0.f);
472 const Vec3 c = toneMapSky(atmosphereRadiance({std::sin(angle), 0.04f, std::cos(angle)},
473 {impl_->sunDir[0], impl_->sunDir[1], impl_->sunDir[2]}, impl_->turbidity,
474 impl_->mieStrength), impl_->pcgSkybox.enabled ? impl_->pcgSkybox.exposure : impl_->skyExposure);
475 const float clear = c.y * impl_->sunEnergy + 0.020f * (1.f - impl_->sunEnergy);
476 const float result = clear * (1.f - impl_->weatherCloudiness * 0.72f) +
477 impl_->weatherCloudiness * 0.090f + impl_->weatherFlash * 0.48f;
478 return result * (impl_->pcgSkybox.enabled ? impl_->pcgSkybox.tintGreen : 1.f);
479}
480float DayNight::getSkyB() const {
481 const float angle = deg2rad(impl_->pcgSkybox.enabled ? impl_->pcgSkybox.rotationDegrees : 0.f);
482 const Vec3 c = toneMapSky(atmosphereRadiance({std::sin(angle), 0.04f, std::cos(angle)},
483 {impl_->sunDir[0], impl_->sunDir[1], impl_->sunDir[2]}, impl_->turbidity,
484 impl_->mieStrength), impl_->pcgSkybox.enabled ? impl_->pcgSkybox.exposure : impl_->skyExposure);
485 const float clear = c.z * impl_->sunEnergy + 0.060f * (1.f - impl_->sunEnergy);
486 const float result = clear * (1.f - impl_->weatherCloudiness * 0.62f) +
487 impl_->weatherCloudiness * 0.125f + impl_->weatherFlash * 0.68f;
488 return result * (impl_->pcgSkybox.enabled ? impl_->pcgSkybox.tintBlue : 1.f);

490float DayNight::getAmbientBrightness() const {
491 const float night = impl_->nightLight[1] ? 1.0f : 0.6f; // starlight boost
492 return 0.05f * night + impl_->sunEnergy * 0.5f *
493 (1.f - impl_->weatherCloudiness * 0.68f) + impl_->weatherFlash * 0.45f;
494}

533void DayNight::applyAtmosphere(graphics::Volumetric *fog) const {
534 if (!fog) return;
535 fog->setLightDirection(getSunDirX(), getSunDirY(), getSunDirZ());
536 fog->setFogColor(getSkyR(), getSkyG(), getSkyB());
537 fog->setCloudLightColor(getSunR() + impl_->weatherFlash * 0.5f,
538 getSunG() + impl_->weatherFlash * 0.65f,
539 getSunB() + impl_->weatherFlash * 0.9f);
540 fog->setIntensity(std::max(0.08f, impl_->sunEnergy + impl_->weatherFlash));
541 fog->setTime(impl_->timeOfDay * 18.f);
542 const PcgFogState &pcg = impl_->pcgFog;
543 const bool hasWeatherOffset = pcg.additionalLinearDistance != 0.f ||
544 pcg.additionalExponentialDensity != 0.f;
545 const bool pcgFogActive = pcg.overrideDensityVolume || pcg.overrideFog ||
546 hasWeatherOffset;
547 if (impl_->pcgFogTarget != fog) {
548 impl_->pcgFogTarget = fog;
549 impl_->pcgFogWasActive = false;
550 }
551 if (pcgFogActive && !impl_->pcgFogWasActive) {
552 impl_->pcgFogBaseMode = fog->getMode();
553 impl_->pcgFogBaseQuality = fog->getQuality();
554 impl_->pcgFogBaseDensity = fog->getFloat("density");
555 impl_->pcgFogBaseStart = fog->getFloat("fogStart");
556 impl_->pcgFogBaseEnd = fog->getFloat("fogEnd");
557 } else if (!pcgFogActive && impl_->pcgFogWasActive) {
558 fog->setMode(impl_->pcgFogBaseMode);
559 fog->setQuality(impl_->pcgFogBaseQuality);
560 fog->setDensity(impl_->pcgFogBaseDensity);
561 fog->setFogStart(impl_->pcgFogBaseStart);
562 fog->setFogEnd(impl_->pcgFogBaseEnd);
563 }
564 impl_->pcgFogWasActive = pcgFogActive;
565 if (pcg.overrideDensityVolume) {
566 static constexpr float hazeDensity[] = {0.0025f, 0.005f, 0.01f, 0.02f, 0.04f};
567 fog->setMode("fog");
568 fog->setQuality(pcg.densityVolumeTiling < 2 ? "low" :
569 pcg.densityVolumeTiling < 4 ? "medium" : "high");
570 fog->setFogColor(pcg.densityAlbedoRed, pcg.densityAlbedoGreen,
571 pcg.densityAlbedoBlue);
572 fog->setDensity(hazeDensity[pcg.densityVolumeEffect] *
573 pcg.globalDensityMultiplier);
574 fog->setFogStart(0.f);
575 fog->setFogEnd(pcg.densityVolumeDistance);
576 } else if (pcg.overrideFog || hasWeatherOffset) {
577 fog->setMode("fog");
578 fog->setFogColor(pcg.red, pcg.green, pcg.blue);
579 const float modeScale = pcg.mode == 2 ? 1.5f : 1.f;
580 fog->setDensity((pcg.density + pcg.additionalExponentialDensity) *
581 pcg.globalDensityMultiplier * modeScale);
582 const float start = std::max(0.f, pcg.startDistance + pcg.additionalLinearDistance);
583 fog->setFogStart(start);
584 fog->setFogEnd(std::max(start + 1.f,
585 pcg.endDistance + pcg.additionalLinearDistance));
586 }

589void DayNight::applyReflectionProbeSky(graphics::ReflectionProbeCapture *probe) const {
590 if (!probe) return;
591 static constexpr Vec3 directions[6] = {
592 {1.f, 0.f, 0.f}, {-1.f, 0.f, 0.f}, {0.f, 1.f, 0.f},
593 {0.f, -1.f, 0.f}, {0.f, 0.f, 1.f}, {0.f, 0.f, -1.f},
594 };
595 const Vec3 sun{impl_->sunDir[0], impl_->sunDir[1], impl_->sunDir[2]};
596 const float nightAmount = std::clamp((-impl_->elevDeg) / 12.f, 0.f, 1.f);
597 const float skyRotation = deg2rad(
598 impl_->pcgSkybox.enabled ? impl_->pcgSkybox.rotationDegrees : 0.f);
599 const Vec3 skyTint = impl_->pcgSkybox.enabled
600 ? Vec3{impl_->pcgSkybox.tintRed, impl_->pcgSkybox.tintGreen,
601 impl_->pcgSkybox.tintBlue}
602 : Vec3{1.f, 1.f, 1.f};
603 for (int face = 0; face < 6; ++face) {
604 Vec3 direction = directions[face];
605 const float rotatedX = direction.x * std::cos(skyRotation) +
606 direction.z * std::sin(skyRotation);
607 direction.z = -direction.x * std::sin(skyRotation) +
608 direction.z * std::cos(skyRotation);
609 direction.x = rotatedX;
610 const float up = direction.y;
611 const Vec3 view{direction.x, std::max(direction.y, 0.002f), direction.z};
612 const Vec3 day = atmosphereRadiance(view, sun, impl_->turbidity, impl_->mieStrength);
613 const Vec3 night{0.006f + 0.012f * std::max(up, 0.f),
614 0.010f + 0.020f * std::max(up, 0.f),
615 0.035f + 0.070f * std::max(up, 0.f)};
616 Vec3 color = add(scale(day, (1.f - nightAmount) * impl_->sunEnergy),
617 scale(night, nightAmount));
618 const Vec3 overcast{0.075f + 0.035f * std::max(up, 0.f),
619 0.090f + 0.045f * std::max(up, 0.f),
620 0.125f + 0.060f * std::max(up, 0.f)};
621 color = add(scale(color, 1.f - impl_->weatherCloudiness * 0.82f),
622 scale(overcast, impl_->weatherCloudiness));
623 color = add(color, scale(Vec3{0.75f, 0.90f, 1.20f}, impl_->weatherFlash));
624 color = scale(color, impl_->pcgSkybox.enabled ? impl_->pcgSkybox.exposure
625 : impl_->skyExposure);
626 color = {color.x * skyTint.x, color.y * skyTint.y, color.z * skyTint.z};
627 probe->setSkyFaceColor(face, color.x, color.y, color.z);
628 probe->setSkyFaceTexture(face, impl_->skyFaces[static_cast<size_t>(face)]);
629 const float linearLuminance =
630 color.x * 0.2126f + color.y * 0.7152f + color.z * 0.0722f;
631 const float encodedLuminance =
632 impl_->skyFaceCenterLuminance[static_cast<size_t>(face)];
633 probe->setSkyFaceTextureScale(
634 face, encodedLuminance > 1e-4f ? linearLuminance / encodedLuminance : 1.f);
635 }
636 probe->setEnvironmentLighting(impl_->skyCube, 0.5f + 0.5f * impl_->sunEnergy);
637}

732void DayNight::update(float dt, graphics::Graphics *gfx) {
733 if (!impl_->built) init(gfx);
734 if (!impl_->gfx) return;
735
736 if (!impl_->paused) {
737 impl_->timeOfDay += dt * impl_->speed;
738 impl_->timeOfDay = std::fmod(impl_->timeOfDay, 24.f);
739 if (impl_->timeOfDay < 0.f) impl_->timeOfDay += 24.f;
740 }
741 const float hours = impl_->timeOfDay;
742
743 // Solar elevation: sine curve peaking at noon (hours=12).
744 const float frac = (hours - 6.f) / 12.f; // -1 at 6h, 0 at 12h, +1 at 18h
745 float elevDeg = kMaxElevationDeg * std::sin(kPi * frac);
746 float azimDeg = (hours / 24.f) * 360.f; // full rotation per day
747 if (impl_->manualSun.enabled) {
748 const float pitch = deg2rad(impl_->manualSun.pitchDegrees);
749 const float rotation = deg2rad(impl_->manualSun.rotationDegrees);
750 impl_->sunDir[0] = -std::cos(pitch) * std::sin(rotation);
751 impl_->sunDir[1] = std::sin(pitch);
752 impl_->sunDir[2] = -std::cos(pitch) * std::cos(rotation);
753 elevDeg = std::asin(std::clamp(impl_->sunDir[1], -1.f, 1.f)) * 180.f / kPi;
754 azimDeg = std::atan2(impl_->sunDir[0], impl_->sunDir[2]) * 180.f / kPi;
755 if (azimDeg < 0.f) azimDeg += 360.f;
756 } else {
757 sunDirection(elevDeg, azimDeg, impl_->sunDir[0], impl_->sunDir[1],
758 impl_->sunDir[2]);
759 }
760 impl_->elevDeg = elevDeg;
761 impl_->azimDeg = azimDeg;
762
763 // Sun energy: ramps up a few degrees above the horizon.
764 impl_->sunEnergy = impl_->manualSun.enabled
765 ? impl_->manualSun.intensity
766 : std::clamp((elevDeg + 6.f) / 14.f, 0.f, 1.f);
767 const float nightAmount = std::clamp((-elevDeg) / 12.f, 0.f, 1.f);
768
769 // Push the directional sun (replaces the legacy directional when no other
770 // dir Light3D is active; we keep moon as a Light3D instead so it can have
771 // different color/intensity than the sun slot).
772 const Vec3 directSun = impl_->manualSun.enabled
773 ? manualSunColor(impl_->manualSun)
774 : attenuatedSunColor(impl_->sunDir[1], impl_->turbidity,
775 impl_->mieStrength);
776 const float weatherSun = 1.f - 0.82f * impl_->weatherCloudiness;
777 const Vec3 flashLight{impl_->weatherFlash * 0.75f, impl_->weatherFlash * 0.90f,
778 impl_->weatherFlash * 1.20f};
779 const float globalLight = impl_->pcgAmbient.active
780 ? impl_->pcgAmbient.globalLightMultiplier
781 : 1.f;
782 gfx->setDirectionalLight(impl_->sunDir[0], impl_->sunDir[1], impl_->sunDir[2],
783 directSun.x * impl_->sunEnergy * weatherSun * globalLight + flashLight.x,
784 directSun.y * impl_->sunEnergy * weatherSun * globalLight + flashLight.y,
785 directSun.z * impl_->sunEnergy * weatherSun * globalLight + flashLight.z);
786
787 // Background matches the sky at the horizon for the clear color.
788 const float skyR = getSkyR(), skyG = getSkyG(), skyB = getSkyB();
789 gfx->setBackgroundColorRGBA(skyR, skyG, skyB, 1.f);
790
791 // --- procedural skybox (IBL env), regenerated per sun bucket ---
792 if (impl_->skyboxEnabled) {
793 const int bucket = int(elevDeg) + int(azimDeg / 4.f) * 1000 +
794 int(impl_->weatherCloudiness * 10.f) * 100000;
795 if (bucket != impl_->lastSkyBucket) {
796 impl_->lastSkyBucket = bucket;
797 std::vector<uint8_t> faces(
798 size_t(kSkyCubeSize) * size_t(kSkyCubeSize) * 4 * 6);
799 for (int f = 0; f < 6; ++f) {
800 std::vector<uint8_t> face(
801 size_t(kSkyCubeSize) * size_t(kSkyCubeSize) * 4);
802 const Vec3 tint = impl_->pcgSkybox.enabled
803 ? Vec3{impl_->pcgSkybox.tintRed, impl_->pcgSkybox.tintGreen,
804 impl_->pcgSkybox.tintBlue}
805 : Vec3{1.f, 1.f, 1.f};
806 fillSkyFace(face, int(kSkyCubeSize), f, impl_->sunDir, directSun,
807 impl_->sunEnergy, nightAmount, impl_->turbidity,
808 impl_->mieStrength,
809 impl_->pcgSkybox.enabled ? impl_->pcgSkybox.exposure : impl_->skyExposure,
810 impl_->weatherCloudiness,
811 impl_->pcgSkybox.enabled ? impl_->pcgSkybox.rotationDegrees : 0.f,
812 tint, cubeDir);
813 impl_->skyFaces[static_cast<size_t>(f)] =
814 gfx->newTexture(int(kSkyCubeSize), int(kSkyCubeSize), face.data());
815 const size_t center =
816 (size_t(kSkyCubeSize / 2) * size_t(kSkyCubeSize) + size_t(kSkyCubeSize / 2)) *
817 4u;
818 impl_->skyFaceCenterLuminance[static_cast<size_t>(f)] =
819 (float(face[center]) * 0.2126f + float(face[center + 1]) * 0.7152f +
820 float(face[center + 2]) * 0.0722f) /
821 255.f;
822 std::memcpy(faces.data() + size_t(f) * face.size(), face.data(), face.size());
823 }
824 // Replace the previous env cube; Graphics owns old textures.
825 impl_->skyCube = gfx->newCubemap(int(kSkyCubeSize), faces.data(),
826 graphics::TextureCreateInfo::withMipmaps(true));
827 gfx->setMesh3DEnv(impl_->skyCube, 0.5f + 0.5f * impl_->sunEnergy);
828 }
829 }
830
831 // --- night light systems (only meaningful below the horizon) ---
832 const bool night = elevDeg < 0.f;
833
834 // Moonlight: a directional light at the opposite-ish angle of the sun.
835 if (impl_->moonLight) {
836 const bool on = night && impl_->nightLight[0];
837 impl_->moonLight->setEnabled(on);
838 if (on) {
839 impl_->moonLight->setDirection(-impl_->sunDir[0], -impl_->sunDir[1],
840 -impl_->sunDir[2]);
841 }
842 }
843
844 // Fire.
845 if (impl_->fireLight) {
846 impl_->fireLight->setEnabled(night && impl_->nightLight[2]);
847 }
848
849 // Fireflies: gentle sinusoidal drift.
850 if (impl_->flyLights.size() == impl_->flies.size()) {
851 for (size_t i = 0; i < impl_->flies.size(); ++i) {
852 graphics::Light3D *l = impl_->flyLights[i];
853 const Impl::Fly &f = impl_->flies[i];
854 const bool on = night && impl_->nightLight[3];
855 l->setEnabled(on);
856 if (on) {
857 const float t = impl_->timeOfDay + f.seed;
858 l->setPosition(f.x + std::sin(t * 0.9f) * 0.6f,
859 f.y + std::sin(t * 1.3f + 1.7f) * 0.4f,
860 f.z + std::cos(t * 0.8f) * 0.6f);
861 }
862 }
863 }
