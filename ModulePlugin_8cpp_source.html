19ssq::Table cloneTable(const ssq::Table& source) {
20 auto vm = source.getHandle();
21 sq_pushobject(vm, source.getRaw());
22 if (SQ_FAILED(sq_clone(vm, -1))) throw std::runtime_error("cannot stage plugin namespace");
23 ssq::Object object(vm);
24 sq_getstackobj(vm, -1, &object.getRaw());
25 sq_addref(vm, &object.getRaw());
26 sq_pop(vm, 2);
27 return ssq::Table(object);

30using Slots = std::unordered_map<std::string, ssq::Object>;

32Slots slots(const ssq::Table& table) {
33 auto vm = table.getHandle();
34 Slots result;
35 sq_pushobject(vm, table.getRaw());
36 sq_pushnull(vm);
37 while (SQ_SUCCEEDED(sq_next(vm, -2))) {
38 const SQChar* key = nullptr;
39 if (SQ_SUCCEEDED(sq_getstring(vm, -2, &key))) {
40 ssq::Object value(vm);
41 sq_getstackobj(vm, -1, &value.getRaw());
42 sq_addref(vm, &value.getRaw());
43 result.emplace(key, std::move(value));
44 }
45 sq_pop(vm, 2);
46 }
47 sq_pop(vm, 2);
48 return result;
49}

59bool sameObject(const ssq::Object& a, const ssq::Object& b) {
60 // Identity comparison must not invoke _cmp or compare incompatible types:
61 // sq_cmp leaves its result undefined when the VM rejects a comparison.
62 return a.getRaw()._type == b.getRaw()._type && a.getRaw()._unVal.raw == b.getRaw()._unVal.raw;
63}

65// SimpleSquirrel's native type map stores unrooted class handles. Keep every
66// generation (including a partially bound candidate) rooted in the VM registry,
67// so those handles and their native closure destructors stay valid until VM close.
68void retainBindings(const ssq::Table& table) {
69 auto vm = table.getHandle();
70 sq_pushregistrytable(vm);
71 sq_pushstring(vm, "eve.plugin.generations", -1);
72 if (SQ_FAILED(sq_rawget(vm, -2))) {
73 sq_pushstring(vm, "eve.plugin.generations", -1);
74 sq_newarray(vm, 0);
75 sq_newslot(vm, -3, SQFalse);
76 sq_pushstring(vm, "eve.plugin.generations", -1);
77 sq_rawget(vm, -2);
78 }
79 sq_pushobject(vm, table.getRaw());
80 sq_arrayappend(vm, -2);
81 sq_pop(vm, 2);
82}

98Result<void> ModuleManager::publishPluginBindings() {
99 auto& state = *inst().state_;
100 auto* machine = vm();
101 if (!machine)
102 return Result<void>::failure(Diagnostic::error(
103 DiagnosticCode::Conflict, "plugin loading requires an attached script VM", {}, {}, "module"));
104 auto handle = machine->getHandle();
105 StackRestore stack{handle, sq_gettop(handle)};
106 Slots before;
107 try {
108 auto live = ssq::Table(machine->find("eve"));
109 auto previous = cloneTable(live);
110 auto candidate = cloneTable(live);
111 before = slots(previous);
112 retainBindings(previous);
113 retainBindings(candidate);
114 const auto& owner = state.plugin_registration_owner_;
115 const auto found = state.plugin_binding_keys_.find(owner);
116 const std::vector<std::string> oldKeys =
117 found == state.plugin_binding_keys_.end() ? std::vector<std::string>{} : found->second;
118 for (const auto& key : oldKeys) eraseSlot(candidate, key);
119 for (const auto& name : state.plugin_registration_added_) {
120 auto& info = state.registered_modules.at(name);
121 if (info.exposer) info.exposer(candidate);
122 info.exposed = true;
123 }
124 const auto after = slots(candidate);
125 std::vector<std::string> keys;
126 for (const auto& [key, value] : after) {
127 const auto existing = before.find(key);
128 const bool owned = std::find(oldKeys.begin(), oldKeys.end(), key) != oldKeys.end();
129 if (existing != before.end() && !owned) {
130 if (!sameObject(existing->second, value))
131 throw std::runtime_error("plugin binding conflicts with existing eve slot: " + key);
132 } else {
133 keys.push_back(key);
134 }
135 }
136 // Publication only mutates eve's slots; existing references to the eve
137 // table remain valid. No script or plugin callback runs during publication.
138 state.plugin_binding_keys_[owner] = keys;
139 for (const auto& key : oldKeys) eraseSlot(live, key);
140 for (const auto& key : keys) live.set(key.c_str(), after.at(key));
141 return Result<void>::success(Status::success(StatusCode::Applied));
142 } catch (const std::exception& error) {
143 restoreTypeMap(*machine, before);
144 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Conflict, error.what(), {}, {}, "module"));
145 } catch (...) {
146 restoreTypeMap(*machine, before);
147 return Result<void>::failure(
148 Diagnostic::error(DiagnosticCode::Conflict, "plugin script binding threw", {}, {}, "module"));
149 }
150}
