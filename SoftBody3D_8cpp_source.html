19float tetraVolume(const V3& a, const V3& b, const V3& c, const V3& d) {
20 return std::fabs(glm::dot(b - a, glm::cross(c - a, d - a))) / 6.f;
21}

23glm::quat bestFitRotation(const glm::mat3& covariance) {
24 glm::quat rotation(1.f, 0.f, 0.f, 0.f);
25 for (int i = 0; i < 8; ++i) {
26 const glm::mat3 basis = glm::mat3_cast(rotation);
27 const V3 omega = (glm::cross(basis[0], covariance[0]) + glm::cross(basis[1], covariance[1]) +
28 glm::cross(basis[2], covariance[2])) /
29 (std::fabs(glm::dot(basis[0], covariance[0]) + glm::dot(basis[1], covariance[1]) +
30 glm::dot(basis[2], covariance[2])) +
31 1e-8f);
32 const float magnitude = glm::length(omega);
33 if (magnitude < 1e-6f) break;
34 rotation = glm::normalize(glm::angleAxis(magnitude, omega / magnitude) * rotation);
35 }
36 return rotation;
37}

74SoftBody3D::SoftBody3D(int cols, int rows, int layers, float spacing, float originX, float originY, float originZ)
75 : cols_(cols), rows_(rows), layers_(layers), spacing_(spacing) {
76 particles_.reserve(static_cast<size_t>(cols_) * rows_ * layers_);
77 for (int z = 0; z < layers_; ++z) {
78 for (int y = 0; y < rows_; ++y) {
79 for (int x = 0; x < cols_; ++x) {
80 const float px = originX + float(x) * spacing_;
81 const float py = originY + float(y) * spacing_;
82 const float pz = originZ + float(z) * spacing_;
83 particles_.push_back({px, py, pz, px, py, pz, px, py, pz, false});
84 }
85 }
86 }
87 clusters_.reserve(static_cast<size_t>(cols_ - 1) * (rows_ - 1) * (layers_ - 1));
88 for (int z = 0; z + 1 < layers_; ++z) {
89 for (int y = 0; y + 1 < rows_; ++y) {
90 for (int x = 0; x + 1 < cols_; ++x) {
91 Cluster cluster;
92 cluster.indices[0] = indexOf(x, y, z);
93 cluster.indices[1] = indexOf(x + 1, y, z);
94 cluster.indices[2] = indexOf(x, y + 1, z);
95 cluster.indices[3] = indexOf(x + 1, y + 1, z);
96 cluster.indices[4] = indexOf(x, y, z + 1);
97 cluster.indices[5] = indexOf(x + 1, y, z + 1);
98 cluster.indices[6] = indexOf(x, y + 1, z + 1);
99 cluster.indices[7] = indexOf(x + 1, y + 1, z + 1);
100 clusters_.push_back(cluster);
101 }
102 }
103 }
104 particleRadius_ = spacing_ * 0.35f;
105}

177int SoftBody3D::grabAt(float x, float y, float z, float radius) {
178 float best = radius * radius;
179 int found = -1;
180 for (int i = 0; i < getParticleCount(); ++i) {
181 const Particle& p = particles_[static_cast<size_t>(i)];
182 const float dx = p.x - x, dy = p.y - y, dz = p.z - z;
183 const float distanceSquared = dx * dx + dy * dy + dz * dz;
184 if (!p.pinned && distanceSquared <= best) {
185 best = distanceSquared;
186 found = i;
187 }
188 }
189 if (found >= 0) {
190 grabIndex_ = found;
191 moveGrab(x, y, z);
192 }
193 return found;
194}

258void SoftBody3D::integrate(float dt) {
259 const float velocityScale = std::max(0.f, 1.f - damping_);
260 for (int i = 0; i < getParticleCount(); ++i) {
261 Particle& p = particles_[static_cast<size_t>(i)];
262 if (p.pinned || i == grabIndex_) continue;
263 float ax = gravityX_ + forceX_ / particleMass_;
264 float ay = gravityY_ + forceY_ / particleMass_;
265 float az = gravityZ_ + forceZ_ / particleMass_;
266 if (hasInteraction_) {
267 const V3 delta = V3(interactX_ - p.x, interactY_ - p.y, interactZ_ - p.z);
268 const float distance = glm::length(delta);
269 if (distance > 1e-6f && distance < interactRadius_) {
270 const float falloff = 1.f - distance / interactRadius_;
271 const V3 field = delta * (interactStrength_ * falloff / distance);
272 ax += field.x;
273 ay += field.y;
274 az += field.z;
275 }
276 }
277 const float vx = (p.x - p.px) * velocityScale;
278 const float vy = (p.y - p.py) * velocityScale;
279 const float vz = (p.z - p.pz) * velocityScale;
280 p.px = p.x;
281 p.py = p.y;
282 p.pz = p.z;
283 p.x += vx + ax * dt * dt;
284 p.y += vy + ay * dt * dt;
285 p.z += vz + az * dt * dt;
286 }

289void SoftBody3D::solveShapeMatching(float dt) {
290 const float stiffness = 1.f - std::pow(1.f - deformationResistance_, 1.f / float(iterations_));
291 for (Cluster& cluster : clusters_) {
292 V3 currentCom(0.f), restCom(0.f);
293 for (int i = 0; i < 8; ++i) {
294 const Particle& p = particles_[static_cast<size_t>(cluster.indices[i])];
295 currentCom += V3(p.x, p.y, p.z);
296 restCom += V3(p.rx + cluster.plastic[i][0], p.ry + cluster.plastic[i][1], p.rz + cluster.plastic[i][2]);
297 }
298 currentCom /= 8.f;
299 restCom /= 8.f;
300 glm::mat3 covariance(0.f);
301 for (int i = 0; i < 8; ++i) {
302 const Particle& p = particles_[static_cast<size_t>(cluster.indices[i])];
303 const V3 current = V3(p.x, p.y, p.z) - currentCom;
304 const V3 rest =
305 V3(p.rx + cluster.plastic[i][0], p.ry + cluster.plastic[i][1], p.rz + cluster.plastic[i][2]) - restCom;
306 covariance += glm::outerProduct(current, rest);
307 }
308 const glm::mat3 rotation = glm::mat3_cast(bestFitRotation(covariance));
309 float deformation = 0.f;
310 V3 restCurrent[8];
311 for (int i = 0; i < 8; ++i) {
312 Particle& p = particles_[static_cast<size_t>(cluster.indices[i])];
313 restCurrent[i] = glm::transpose(rotation) * (V3(p.x, p.y, p.z) - currentCom);
314 const V3 rest =
315 V3(p.rx + cluster.plastic[i][0], p.ry + cluster.plastic[i][1], p.rz + cluster.plastic[i][2]) - restCom;
316 deformation += glm::length(restCurrent[i] - rest) / spacing_;
317 if (!p.pinned && cluster.indices[i] != grabIndex_) {
318 const V3 goal = currentCom + rotation * rest;
319 p.x += (goal.x - p.x) * stiffness;
320 p.y += (goal.y - p.y) * stiffness;
321 p.z += (goal.z - p.z) * stiffness;
322 }
323 }
324 deformation /= 8.f;
325 const float creep = deformation > plasticYield_ ? plasticCreep_ * dt : 0.f;
326 const float recovery = plasticRecovery_ * dt;
327 if (maxDeformation_ > 0.f && (creep > 0.f || recovery > 0.f)) {
328 for (int i = 0; i < 8; ++i) {
329 const Particle& p = particles_[static_cast<size_t>(cluster.indices[i])];
330 const V3 original = V3(p.rx, p.ry, p.rz) - restCom;
331 V3 offset(cluster.plastic[i][0], cluster.plastic[i][1], cluster.plastic[i][2]);
332 offset += (restCurrent[i] - original - offset) * std::min(1.f, creep);
333 offset += (V3(0.f) - offset) * std::min(1.f, recovery);
334 const float limit = maxDeformation_ * spacing_;
335 const float length = glm::length(offset);
336 if (length > limit && length > 1e-6f) offset *= limit / length;
337 cluster.plastic[i][0] = offset.x;
338 cluster.plastic[i][1] = offset.y;
339 cluster.plastic[i][2] = offset.z;
340 }
341 }
342 }

345void SoftBody3D::solveSelfCollision() {
346 if (!selfCollision_ || particleRadius_ <= 0.f) return;
347 const float minimum = particleRadius_ * 2.f;
348 const float minimumSquared = minimum * minimum;
349 for (int i = 0; i < getParticleCount(); ++i) {
350 for (int j = i + 1; j < getParticleCount(); ++j) {
351 Particle& a = particles_[static_cast<size_t>(i)];
352 Particle& b = particles_[static_cast<size_t>(j)];
353 V3 delta(b.x - a.x, b.y - a.y, b.z - a.z);
354 const float distanceSquared = glm::dot(delta, delta);
355 if (distanceSquared >= minimumSquared || distanceSquared < 1e-12f) continue;
356 const float distance = std::sqrt(distanceSquared);
357 const V3 correction = delta * ((minimum - distance) / distance * 0.5f);
358 if (!a.pinned && i != grabIndex_) {
359 a.x -= correction.x;
360 a.y -= correction.y;
361 a.z -= correction.z;
362 }
363 if (!b.pinned && j != grabIndex_) {
364 b.x += correction.x;
365 b.y += correction.y;
366 b.z += correction.z;
367 }
368 }
369 }

372void SoftBody3D::collideWorld(float dt) {
373 if (!collisionWorld_ || !collisionWorld_->softBodyCollisionAvailable() || particleRadius_ <= 0.f) return;
374 softbody::SoftBodyContact contact;
375 const float inverseDt = dt > 1e-6f ? 1.f / dt : 0.f;
376 for (Particle& p : particles_) {
377 if (p.pinned ||
378 collisionWorld_->probeSoftBodyParticle(p.x, p.y, p.z, particleRadius_, contact) !=
379 softbody::SoftBodyContactState::Hit ||
380 !contact.hit)
381 continue;
382 const V3 velocity = V3(p.x - p.px, p.y - p.py, p.z - p.pz) * inverseDt;
383 p.x += contact.nx * contact.depth;
384 p.y += contact.ny * contact.depth;
385 p.z += contact.nz * contact.depth;
386 if (contact.dynamicBody) {
387 const float normalVelocity = glm::dot(velocity, V3(contact.nx, contact.ny, contact.nz));
388 if (normalVelocity < 0.f) {
389 const float impulse = std::min(-normalVelocity * particleMass_, particleMass_ * 8.f);
390 collisionWorld_->applySoftBodyImpulse(contact.bodyId, -contact.nx * impulse, -contact.ny * impulse,
391 -contact.nz * impulse);
392 }
393 }
394 }

414void SoftBody3D::updateSubsteps(float dt, int substeps) {
415 if (destroyed_ || substeps < 1) return;
416 const float h = dt / float(substeps);
417 for (int substep = 0; substep < substeps; ++substep) {
418 integrate(h);
419 for (int iteration = 0; iteration < iterations_; ++iteration) {
420 solveShapeMatching(h);
421 solveSelfCollision();
422 collideWorld(h);
423 collideBounds();
424 }
425 if (grabIndex_ >= 0) moveGrab(grabX_, grabY_, grabZ_);
426 }
427 forceX_ = forceY_ = forceZ_ = 0.f;
428 hasInteraction_ = false;

456float SoftBody3D::getVolumeRatio() const {
457 if (clusters_.empty()) return 1.f;
458 float volume = 0.f;
459 for (const Cluster& cluster : clusters_) {
460 V3 p[8];
461 for (int i = 0; i < 8; ++i) {
462 const Particle& particle = particles_[static_cast<size_t>(cluster.indices[i])];
463 p[i] = V3(particle.x, particle.y, particle.z);
464 }
465 volume += tetraVolume(p[0], p[1], p[3], p[7]) + tetraVolume(p[0], p[3], p[2], p[7]) +
466 tetraVolume(p[0], p[2], p[6], p[7]) + tetraVolume(p[0], p[6], p[4], p[7]) +
467 tetraVolume(p[0], p[4], p[5], p[7]) + tetraVolume(p[0], p[5], p[1], p[7]);
468 }
469 const float rest = float(clusters_.size()) * spacing_ * spacing_ * spacing_;
470 return rest > 0.f ? volume / rest : 1.f;
471}

479eve::Result<void> SoftBodyModel::validate() const {
480 if (sourceMesh.empty())
481 return eve::Result<void>::failure(
482 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "sourceMesh must not be empty", "sourceMesh"));
483 if (particles.empty())
484 return eve::Result<void>::failure(
485 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "model must contain particles", "particles"));
486 for (std::size_t index = 0; index < particles.size(); ++index) {
487 const auto& particle = particles[index];
