22void appendBox(std::vector<float>& positions, std::vector<float>& normals, std::vector<float>& uv,
23 std::vector<std::uint32_t>& indices, const glm::mat4& model, float hx, float hy, float hz) {
24 struct Face {
25 glm::vec3 n;
26 glm::vec3 a, b, c, d;
27 };
28 const Face faces[] = {
29 {{0, 0, 1}, {-hx, -hy, hz}, {hx, -hy, hz}, {hx, hy, hz}, {-hx, hy, hz}},
30 {{0, 0, -1}, {hx, -hy, -hz}, {-hx, -hy, -hz}, {-hx, hy, -hz}, {hx, hy, -hz}},
31 {{0, 1, 0}, {-hx, hy, hz}, {hx, hy, hz}, {hx, hy, -hz}, {-hx, hy, -hz}},
32 {{0, -1, 0}, {-hx, -hy, -hz}, {hx, -hy, -hz}, {hx, -hy, hz}, {-hx, -hy, hz}},
33 {{1, 0, 0}, {hx, -hy, hz}, {hx, -hy, -hz}, {hx, hy, -hz}, {hx, hy, hz}},
34 {{-1, 0, 0}, {-hx, -hy, -hz}, {-hx, -hy, hz}, {-hx, hy, hz}, {-hx, hy, -hz}},
35 };
36 const float tex[8] = {0.f, 0.f, 1.f, 0.f, 1.f, 1.f, 0.f, 1.f};
37 for (const auto& face : faces) {
38 const std::uint32_t base = static_cast<std::uint32_t>(positions.size() / 3);
39 const glm::vec3 corners[4] = {face.a, face.b, face.c, face.d};
40 const glm::vec3 worldN = glm::normalize(glm::mat3(model) * face.n);
41 for (int i = 0; i < 4; ++i) {
42 const glm::vec4 world = model * glm::vec4(corners[i], 1.f);
43 positions.insert(positions.end(), {world.x, world.y, world.z});
44 normals.insert(normals.end(), {worldN.x, worldN.y, worldN.z});
45 uv.insert(uv.end(), {tex[i * 2], tex[i * 2 + 1]});
46 }
47 indices.insert(indices.end(), {base, base + 1, base + 2, base, base + 2, base + 3});
48 }
49}

51glm::mat4 boneModel(const BonePresentation& bone) {
52 const glm::quat rotation(bone.rotW, bone.rotX, bone.rotY, bone.rotZ);
53 glm::mat4 model = glm::translate(glm::mat4(1.f), glm::vec3(bone.worldX, bone.worldY, bone.worldZ));
54 model *= glm::mat4_cast(glm::normalize(rotation));
55 return model;
56}

154eve::Result<void> GeometryCollectionRenderer::rebuildSleepBatch(graphics::Graphics& graphics) {
155 if (!instance_)
156 return eve::Result<void>::failure(eve::Diagnostic::error(
157 eve::DiagnosticCode::InvalidArgument, "geometry-collection renderer has no instance", "instance"));
158 if (meshOwner_ != &graphics) {
159 releaseOwnedMeshes(meshOwner_);
160 meshOwner_ = &graphics;
161 }
162 invalidateSleepBatch(&graphics);
163 std::vector<float> positions;
164 std::vector<float> normals;
165 std::vector<float> uv;
166 std::vector<std::uint32_t> indices;
167 int sleepCount = 0;
168 for (int i = 0; i < instance_->boneCount(); ++i) {
169 auto presentation = instance_->bonePresentation(i);
170 if (!presentation) return eve::Result<void>::failure(presentation.status());
171 if (presentation.value().state != BoneRuntimeState::Sleeping) continue;
172 appendBox(positions, normals, uv, indices, boneModel(presentation.value()),
173 presentation.value().halfExtentX, presentation.value().halfExtentY,
174 presentation.value().halfExtentZ);
175 ++sleepCount;
176 }
177 sleepBatchBoneCount_ = sleepCount;
178 sleepBatchRevision_ = instance_->sleepBatchRevision();
179 if (sleepCount == 0) return eve::Result<void>::success();
180 sleepBatch_ = graphics.newMeshFromArrays(positions.data(), normals.data(), uv.data(),
181 static_cast<int>(positions.size() / 3), indices.data(),
182 static_cast<int>(indices.size()));
183 if (!sleepBatch_)
184 return eve::Result<void>::failure(eve::Diagnostic::error(
185 eve::DiagnosticCode::Failed, "geometry-collection sleep batch mesh allocation failed", "mesh"));
186 return eve::Result<void>::success();
187}

189eve::Result<void> GeometryCollectionRenderer::draw(graphics::Graphics* graphics) {
190 lastActiveDrawCount_ = 0;
191 lastSleepBatchCount_ = 0;
192 if (!graphics)
193 return eve::Result<void>::failure(eve::Diagnostic::error(
194 eve::DiagnosticCode::InvalidArgument, "geometry-collection draw requires Graphics", "graphics"));
195 if (!instance_)
196 return eve::Result<void>::failure(eve::Diagnostic::error(
197 eve::DiagnosticCode::InvalidArgument, "geometry-collection renderer has no instance", "instance"));
198 if (!instance_->hasLiveWorld())
199 return eve::Result<void>::failure(eve::Diagnostic::error(
200 eve::DiagnosticCode::StaleHandle, "geometry-collection world is gone", "world"));
201
202 if (auto geometry = instance_->fragmentGeometry()) {
203 if (preparedGeometry_ != geometry || meshOwner_ != graphics)
204 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
205 "prepareInstance is required before mesh draws",
206 "geometry"));
207 poseScratch_.clear();
208 for (int i = 0; i < instance_->boneCount(); ++i) {
209 auto pose = instance_->bonePresentation(i);
210 if (!pose) return eve::Result<void>::failure(pose.status());
211 poseScratch_.push_back(pose.value());
212 ++lastActiveDrawCount_;
213 }
214 auto updated = updateGeometry(*graphics);
215 if (!updated) return updated;
216 return drawGeometry(*graphics);
217 }
218 ensureUnitBox(*graphics);
219 if (!unitBox_)
220 return eve::Result<void>::failure(eve::Diagnostic::error(
221 eve::DiagnosticCode::Failed, "geometry-collection unit box mesh allocation failed", "mesh"));
222
223 if (sleepBatchRevision_ != instance_->sleepBatchRevision() ||
224 (sleepBatch_ == nullptr && instance_->sleepBatchRevision() != 0)) {
225 auto rebuilt = rebuildSleepBatch(*graphics);
226 if (!rebuilt) return rebuilt;
227 }
228
229 for (int i = 0; i < instance_->boneCount(); ++i) {
230 auto presentation = instance_->bonePresentation(i);
231 if (!presentation) return eve::Result<void>::failure(presentation.status());
232 const auto& bone = presentation.value();
233 if (bone.state == BoneRuntimeState::Sleeping) continue;
234 glm::mat4 model = boneModel(bone);
235 model = glm::scale(model, glm::vec3(bone.halfExtentX, bone.halfExtentY, bone.halfExtentZ));
236 const bool detached = bone.state == BoneRuntimeState::Detached;
237 const graphics::Color tint(detached ? interiorR_ : exteriorR_, detached ? interiorG_ : exteriorG_,
238 detached ? interiorB_ : exteriorB_, detached ? interiorA_ : exteriorA_);
239 graphics->drawMesh(unitBox_, model, nullptr, tint);
240 ++lastActiveDrawCount_;
241 }
242
243 if (sleepBatch_) {
244 graphics->drawMesh(sleepBatch_, glm::mat4(1.f), nullptr,
245 graphics::Color(sleepR_, sleepG_, sleepB_, sleepA_));
246 lastSleepBatchCount_ = sleepBatchBoneCount_;
247 }
248 return eve::Result<void>::success();
249}

251eve::Result<void> GeometryCollectionRenderer::drawAnimation(const CollapseAnimation& animation,
252 graphics::Graphics* graphics, float seconds, float offsetX,
253 float offsetY, float offsetZ) {
254 if (!graphics || animation.boneCount() == 0 || !std::isfinite(seconds) || !std::isfinite(offsetX) ||
255 !std::isfinite(offsetY) || !std::isfinite(offsetZ))
256 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
257 "invalid animation draw arguments", "animation"));
258 const auto geometry = animation.fragmentGeometry();
259 if (geometry && (preparedGeometry_ != geometry || meshOwner_ != graphics))
260 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
261 "prepareAnimation is required before mesh playback",
262 "geometry"));
263 if (!geometry) ensureUnitBox(*graphics);
264 if (!geometry && !unitBox_)
265 return eve::Result<void>::failure(
266 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "box allocation failed", "mesh"));
267 if (geometry) poseScratch_.clear();
268 for (int i = 0; i < animation.boneCount(); ++i) {
269 auto sampled = animation.sample(seconds, i);
270 if (!sampled) return eve::Result<void>::failure(sampled.status());
271 auto bone = sampled.value();
272 bone.worldX += offsetX;
273 bone.worldY += offsetY;
274 bone.worldZ += offsetZ;
275 if (geometry) {
276 poseScratch_.push_back(bone);
277 continue;
278 }
279 auto model = glm::scale(boneModel(bone), glm::vec3(bone.halfExtentX, bone.halfExtentY, bone.halfExtentZ));
280 graphics->drawMesh(unitBox_, model, nullptr, graphics::Color(exteriorR_, exteriorG_, exteriorB_, exteriorA_));
281 }
282 if (geometry) {
283 auto updated = updateGeometry(*graphics);
284 if (!updated) return updated;
285 return drawGeometry(*graphics);
286 }
287 return eve::Result<void>::success();
288}
