35eve::Result<double> mergeDuration(const EffectInstance& current, const EffectDefinition& definition) {
36 switch (definition.policy.duration) {
37 case DurationPolicy::Keep: return eve::Result<double>::success(current.remaining);
38 case DurationPolicy::Replace: return eve::Result<double>::success(normalisedRemaining(definition.duration));
39 case DurationPolicy::Extend:
40 if (definition.duration <= 0.0 || current.remaining < 0.0)
41 return eve::Result<double>::success(current.remaining);
42 if (current.remaining > std::numeric_limits<double>::max() - definition.duration)
43 return eve::Result<double>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
44 "effect duration extension overflows"));
45 return eve::Result<double>::success(current.remaining + definition.duration);
46 }
47 return eve::Result<double>::failure(
48 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "unknown effect duration policy"));
49}

51eve::Result<double> mergeMagnitude(const EffectInstance& current, const EffectDefinition& definition) {
52 switch (definition.policy.magnitude) {
53 case MagnitudePolicy::Keep: return eve::Result<double>::success(current.magnitude);
54 case MagnitudePolicy::Replace: return eve::Result<double>::success(definition.magnitude);
55 case MagnitudePolicy::Add: {
56 const double value = current.magnitude + definition.magnitude;
57 if (!std::isfinite(value))
58 return eve::Result<double>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
59 "effect magnitude addition overflows"));
60 return eve::Result<double>::success(value);
61 }
62 case MagnitudePolicy::Max:
63 return eve::Result<double>::success(std::max(current.magnitude, definition.magnitude));
64 }
65 return eve::Result<double>::failure(
66 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "unknown effect magnitude policy"));
67}

69bool stackWouldOverflow(const EffectInstance& current, const EffectDefinition& definition) {
70 if (definition.policy.maxStacks == 0) return false;
71 switch (definition.policy.stackCount) {
72 case StackCountPolicy::Keep: return current.stackCount > definition.policy.maxStacks;
73 case StackCountPolicy::Increment:
74 return definition.stackCount > definition.policy.maxStacks - current.stackCount;
75 case StackCountPolicy::Set: return definition.stackCount > definition.policy.maxStacks;
76 }
77 return true;
78}

80eve::Result<std::uint32_t> mergeStackCount(const EffectInstance& current, const EffectDefinition& definition) {
81 std::uint32_t value = current.stackCount;
82 switch (definition.policy.stackCount) {
83 case StackCountPolicy::Keep: break;
84 case StackCountPolicy::Increment:
85 if (definition.stackCount > std::numeric_limits<std::uint32_t>::max() - current.stackCount)
86 return eve::Result<std::uint32_t>::failure(
87 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect stack count overflows"));
88 value = current.stackCount + definition.stackCount;
89 break;
90 case StackCountPolicy::Set: value = definition.stackCount; break;
91 }
92 if (definition.policy.maxStacks != 0 && value > definition.policy.maxStacks) {
93 if (definition.policy.overflow == OverflowPolicy::Clamp)
94 value = definition.policy.maxStacks;
95 else
96 return eve::Result<std::uint32_t>::failure(
97 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "effect stack count exceeds its maximum"));
98 }
99 return eve::Result<std::uint32_t>::success(value);
100}

194eve::Result<std::string> EffectContainer::apply(const EffectDefinition& definition, const std::string& subject,
195 const std::string& source) {
196 auto validation = definition.validate();
197 if (!validation) return eve::Result<std::string>::failure(validation.status());
198 if (subject.empty()) return rejected("effect subject must not be empty");
199
200 // Resolve the new instance's initial stack before any Replace/overflow
201 // mutation. A rejected application must leave both instances and events
202 // unchanged.
203 std::uint32_t initialStacks = definition.stackCount;
204 if (definition.policy.maxStacks != 0 && initialStacks > definition.policy.maxStacks) {
205 if (definition.policy.overflow == OverflowPolicy::Reject)
206 return conflict("effect stack count exceeds its maximum");
207 initialStacks = definition.policy.maxStacks;
208 }
209
210 const std::string key = effectiveKey(definition);
211 auto matches = [&subject, &key](const auto& effect) {
212 return effect->subject == subject && effect->stackKey == key;
213 };
214 auto existing = std::find_if(effects_.begin(), effects_.end(), matches);
215
216 if (existing != effects_.end() && definition.policy.stackMode == StackMode::Reuse) {
217 EffectInstance& current = **existing;
218 auto duration = mergeDuration(current, definition);
219 if (!duration) return eve::Result<std::string>::failure(duration.status());
220 auto magnitude = mergeMagnitude(current, definition);
221 if (!magnitude) return eve::Result<std::string>::failure(magnitude.status());
222 if (stackWouldOverflow(current, definition) && definition.policy.overflow == OverflowPolicy::ReplaceOldest) {
223 emit(EffectEventKind::Removed, current, "overflow");
224 effects_.erase(existing);
225 existing = effects_.end();
226 } else {
227 auto stacks = mergeStackCount(current, definition);
228 if (!stacks) return eve::Result<std::string>::failure(stacks.status());
229 current.priority = definition.priority;
230 current.duration = definition.duration;
231 current.period = definition.period;
232 current.periodElapsed = 0.0;
233 current.remaining = std::move(duration).takeValue();
234 current.magnitude = std::move(magnitude).takeValue();
235 current.stackCount = std::move(stacks).takeValue();
236 emit(EffectEventKind::Refreshed, current);
237 return eve::Result<std::string>::success(current.id, eve::Status::success(eve::StatusCode::Applied));
238 }
239 }
240
241 if (existing != effects_.end() && definition.policy.stackMode == StackMode::Accumulate) {
242 EffectInstance& current = **existing;
243 auto duration = mergeDuration(current, definition);
244 if (!duration) return eve::Result<std::string>::failure(duration.status());
245 auto magnitude = mergeMagnitude(current, definition);
246 if (!magnitude) return eve::Result<std::string>::failure(magnitude.status());
247 if (stackWouldOverflow(current, definition) && definition.policy.overflow == OverflowPolicy::ReplaceOldest) {
248 emit(EffectEventKind::Removed, current, "overflow");
249 effects_.erase(existing);
250 existing = effects_.end();
251 } else {
252 auto stacks = mergeStackCount(current, definition);
253 if (!stacks) {
254 if (stacks.code() == eve::StatusCode::Conflict)
255 return conflict("effect stack count exceeds its maximum");
256 return eve::Result<std::string>::failure(stacks.status());
257 }
258 current.priority = definition.priority;
259 current.duration = definition.duration;
260 current.period = definition.period;
261 current.periodElapsed = 0.0;
262 current.remaining = std::move(duration).takeValue();
263 current.magnitude = std::move(magnitude).takeValue();
264 current.stackCount = std::move(stacks).takeValue();
265 emit(EffectEventKind::Stacked, current);
266 return eve::Result<std::string>::success(current.id, eve::Status::success(eve::StatusCode::Applied));
267 }
268 }
269
270 if (definition.policy.stackMode == StackMode::Replace) {
271 for (auto it = effects_.begin(); it != effects_.end();) {
272 if (matches(*it)) {
273 emit(EffectEventKind::Removed, **it, "replaced");
274 it = effects_.erase(it);
275 } else {
276 ++it;
277 }
278 }
279 }
280
281 auto effect = std::make_unique<EffectInstance>();
282 if (effectIdGenerator_) {
283 const auto generated = effectIdGenerator_->generate();
284 if (!generated)
285 return eve::Result<std::string>::failure(
286 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "canonical effect identity generation failed"));
287 effect->identity = eve::EffectId::fromUuid(*generated);
288 if (find(effect->identity) != nullptr)
289 return eve::Result<std::string>::failure(
290 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "canonical effect identity already exists"));
291 effect->id = effect->identity.format();
292 } else {
293 std::ostringstream id;
294 id << "effect-" << std::setw(16) << std::setfill('0') << nextId_++;
295 effect->id = id.str();
296 }
297 effect->subject = subject;
298 effect->type = definition.id;
299 effect->source = source;
300 effect->stackKey = key;
301 effect->priority = definition.priority;
302 effect->duration = std::max(0.0, definition.duration);
303 effect->remaining = normalisedRemaining(definition.duration);
304 effect->period = definition.period;
305 effect->periodElapsed = 0.0;
306 effect->magnitude = definition.magnitude;
307 effect->stackCount = initialStacks;
308 effect->payload = definition.payload;
309 effect->tags = definition.tags;
310 std::sort(effect->tags.begin(), effect->tags.end());
311 effect->tags.erase(std::unique(effect->tags.begin(), effect->tags.end()), effect->tags.end());
312
313 const std::string result = effect->id;
314 effects_.push_back(std::move(effect));
315 emit(EffectEventKind::Applied, *effects_.back());
316 return eve::Result<std::string>::success(result, eve::Status::success(eve::StatusCode::Applied));
317}

391eve::Result<EffectUpdateSummary> EffectContainer::advanceInstances(double dtSeconds, eve::SimulationTick tick) {
392 if (!std::isfinite(dtSeconds) || dtSeconds < 0.0) return invalidDelta();
393 EffectUpdateSummary summary;
394 summary.elapsedSeconds = dtSeconds;
395 if (dtSeconds == 0.0)
396 return eve::Result<EffectUpdateSummary>::success(summary, eve::Status::success(eve::StatusCode::NoOp));
397
398 for (auto it = effects_.begin(); it != effects_.end();) {
399 EffectInstance& effect = **it;
400 const double activeDelta = effect.remaining < 0.0 ? dtSeconds : std::min(dtSeconds, effect.remaining);
401 if (effect.period > 0.0 && activeDelta > 0.0) {
402 if (!std::isfinite(effect.periodElapsed) ||
403 effect.periodElapsed > std::numeric_limits<double>::max() - activeDelta)
404 return eve::Result<EffectUpdateSummary>::failure(eve::Diagnostic::error(
405 eve::DiagnosticCode::InvalidArgument, "effect periodic accumulator overflow"));
406 effect.periodElapsed += activeDelta;
407 while (effect.periodElapsed >= effect.period) {
408 effect.periodElapsed -= effect.period;
409 EffectPeriodicTick periodic;
410 periodic.effectId = effect.id;
411 periodic.subject = effect.subject;
412 periodic.source = effect.source;
413 periodic.magnitude = effect.magnitude;
414 periodic.stackCount = effect.stackCount;
415 periodic.tags = effect.tags;
416 periodic.effectIdentity = effect.identity;
417 periodic.tick = tick;
418 summary.periodicTicks.push_back(std::move(periodic));
419 emit(EffectEventKind::Periodic, effect, "periodic");
420 }
421 }
422 if (effect.remaining < 0.0) {
423 ++it;
424 continue;
425 }
426 effect.remaining = std::max(0.0, effect.remaining - dtSeconds);
427 if (effect.remaining == 0.0) {
428 emit(EffectEventKind::Expired, effect);
429 ++summary.expired;
430 it = effects_.erase(it);
431 } else {
432 ++it;
433 }
434 }
435 return eve::Result<EffectUpdateSummary>::success(summary, eve::Status::success(eve::StatusCode::Applied));
436}

438eve::Result<EffectUpdateSummary> EffectContainer::update(double dtSeconds) {
439 auto duration = eve::Duration::fromSeconds(dtSeconds);
440 if (!duration) return eve::Result<EffectUpdateSummary>::failure(duration.status());
441
442 auto nextTick = hasLastTick_ ? lastTick_.incremented() : std::optional<eve::SimulationTick>(eve::SimulationTick(1));
443 if (!nextTick)
444 return eve::Result<EffectUpdateSummary>::failure(
445 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation, "effect simulation tick overflow"));
446
447 return advance({*nextTick, std::move(duration).takeValue()});
448}

450eve::Result<EffectUpdateSummary> EffectContainer::advance(const eve::SimulationStep& step) {
451 if (step.delta.nanoseconds() < 0)
452 return eve::Result<EffectUpdateSummary>::failure(eve::Diagnostic::error(
453 eve::DiagnosticCode::InvalidArgument, "effect simulation delta must be non-negative"));
454 if (hasLastTick_ && step.tick <= lastTick_)
455 return eve::Result<EffectUpdateSummary>::failure(
456 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "effect simulation tick must advance monotonically"));
457
458 // Stage the lifecycle mutation so a failure does not publish partial state.
459 auto candidate = clone();
460 candidate.lastTick_ = step.tick;
461 candidate.hasLastTick_ = true;
462 auto result = candidate.advanceInstances(step.delta.seconds(), step.tick);
463 if (!result) return eve::Result<EffectUpdateSummary>::failure(result.status());
464 auto status = result.status();
465 EffectUpdateSummary summary = std::move(result).takeValue();
466 summary.tick = step.tick;
467 candidate.lastTick_ = step.tick;
468 candidate.hasLastTick_ = true;
469 *this = std::move(candidate);
470 return eve::Result<EffectUpdateSummary>::success(std::move(summary), std::move(status));
471}
