156fn remainsVisibleAgainstDepth(center: vec3f, radius: f32) -> bool {
157 if (params.clipNearFar.w < 0.5) { return true; }
158 let clip = params.viewProj * vec4f(center, 1.0);
159 if (clip.w <= params.clipNearFar.x) { return true; }
160 let ndc = clip.xyz / clip.w;
161 if (abs(ndc.x) > 1.0 || abs(ndc.y) > 1.0 || ndc.z > 1.0) { return true; }
162
163 let toEye = normalize(params.cameraPos.xyz - center);
164 let frontClip = params.viewProj * vec4f(center + toEye * radius, 1.0);
165 let frontZ = frontClip.z / max(frontClip.w, 1e-6);
166 let viewDepth = max(length(center - params.cameraPos.xyz), 1e-4);
167 let radiusPx = radius * params.clipNearFar.z / viewDepth;
168 let diameterPx = max(radiusPx * 2.0, 1.0);
169 let mip = min(u32(ceil(log2(diameterPx))), u32(params.hzbInfo.x));
170 let width = max(u32(params.hzbInfo.z) >> mip, 1u);
171 let height = max(u32(params.hzbInfo.w) >> mip, 1u);
172 let uv = vec2f(ndc.x * 0.5 + 0.5, 0.5 - ndc.y * 0.5);
173 let p0 = clamp(vec2i(uv * vec2f(f32(width), f32(height))) - vec2i(1),
174 vec2i(0), vec2i(i32(width) - 1, i32(height) - 1));
175 let p1 = min(p0 + vec2i(1), vec2i(i32(width) - 1, i32(height) - 1));
176 let base = 16u + hzb[mip];
177 let a = bitcast<f32>(hzb[base + u32(p0.y) * width + u32(p0.x)]);
178 let b = bitcast<f32>(hzb[base + u32(p0.y) * width + u32(p1.x)]);
179 let c = bitcast<f32>(hzb[base + u32(p1.y) * width + u32(p0.x)]);
180 let d = bitcast<f32>(hzb[base + u32(p1.y) * width + u32(p1.x)]);
181 let blocker = min(min(a, b), min(c, d));
182 // All conservative samples must contain geometry closer than the sphere.
183 // The epsilon prevents the object's own previous-frame depth from culling it.
184 return blocker >= 0.9999 || frontZ <= blocker + 0.002;
185}

191 let input = inputs[index];

193 let scale = max(length(input.model[0].xyz),

202 atomicAdd(&visCommands[input.bucket].instanceCount, 1u);

204}
205)wgsl";

330GpuResidentSubmitStatus Graphics::gpuDrivenSubmitResident(const GpuResidentInstanceBatch &batch) {
331 if (!gpuDrivenEnabled_ || !device || gpuDrivenComputePending_ || gpuDrivenDrawPending_)
332 return GpuResidentSubmitStatus::ResourceUnavailable;
333 if (batch.buffer.backend != GpuResidentBackend::WebGpu) return GpuResidentSubmitStatus::BackendMismatch;
334 if (batch.buffer.nativeHandle == 0 || batch.buffer.offsetBytes % kGpuResidentStorageOffsetAlignment != 0 ||
335 batch.buffer.strideBytes != sizeof(GpuInstance) || !batch.buckets || batch.bucketCount == 0 ||
336 batch.instanceCount == 0)
337 return GpuResidentSubmitStatus::InvalidArgument;
338 const uint64_t required = batch.buffer.offsetBytes + uint64_t(batch.instanceCount) * sizeof(GpuInstance);
339 if (required < batch.buffer.offsetBytes || required > batch.buffer.sizeBytes)
340 return GpuResidentSubmitStatus::InvalidArgument;
341
342 std::vector<GpuIndirectCommand> commands(batch.bucketCount);
343 std::vector<GpuDrivenBucket> buckets(batch.bucketCount);
344 uint64_t coveredInstances = 0;
345 for (uint32_t i = 0; i < batch.bucketCount; ++i) {
346 const GpuResidentInstanceBucket &bucket = batch.buckets[i];
347 const uint64_t end = uint64_t(bucket.firstInstance) + bucket.instanceCount;
348 if (bucket.instanceCount == 0 || bucket.firstInstance != coveredInstances || end > batch.instanceCount ||
349 bucket.meshId >= gpuDrivenMeshes_.size() || bucket.materialId >= gpuDrivenMaterials_.size())
350 return GpuResidentSubmitStatus::InvalidArgument;
351 Mesh *mesh = gpuDrivenMeshes_[bucket.meshId];
352 Material *material = gpuDrivenMaterials_[bucket.materialId];
353 if (!mesh || !mesh->gpuHandle || !gpuDrivenMaterialUsable(material))
354 return GpuResidentSubmitStatus::InvalidArgument;
355 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
356 commands[i] = {gpu->indexCount, bucket.instanceCount, 0, 0, bucket.firstInstance};
357 buckets[i] = {mesh, material, bucket.firstInstance, bucket.instanceCount};
358 coveredInstances = end;
359 }
360 if (coveredInstances != batch.instanceCount) return GpuResidentSubmitStatus::InvalidArgument;
361
362 ensureGpuDrivenResources(batch.instanceCount, batch.bucketCount);
363 if (!gpuDrivenResidentRenderPipeline_ || !gpuDrivenIndirectBuffer_)
364 return GpuResidentSubmitStatus::ResourceUnavailable;
365 queue.WriteBuffer(gpuDrivenIndirectBuffer_, 0, commands.data(), commands.size() * sizeof(GpuIndirectCommand));
366 WGPUBuffer rawBuffer{};
367 static_assert(sizeof(rawBuffer) <= sizeof(batch.buffer.nativeHandle));
368 std::memcpy(&rawBuffer, &batch.buffer.nativeHandle, sizeof(rawBuffer));
369 if (!rawBuffer) return GpuResidentSubmitStatus::ResourceUnavailable;
370 gpuDrivenBuckets_ = std::move(buckets);
371 gpuDrivenResidentBuffer_ = rawBuffer;
372 gpuDrivenResidentOffset_ = batch.buffer.offsetBytes;
373 gpuDrivenResidentSize_ = uint64_t(batch.instanceCount) * sizeof(GpuInstance);
374 gpuDrivenResidentDrawPending_ = true;
375 gpuDrivenDrawPending_ = true;
376 frameHad3DThisFrame = true;
377 frameHad3D = true;
378 return GpuResidentSubmitStatus::Submitted;
379}

715void Graphics::gpuDrivenCullEmit(const glm::mat4 &viewProj, const glm::vec3 &eye, float fovYDeg,
716 float nearZ, float farZ) {
717 if (sceneColorWidth > 0 && sceneColorHeight > 0)
718 createSceneColorResources(sceneColorWidth, sceneColorHeight);
719 if (gbufferSlots.empty() && sceneColorWidth > 0 && sceneColorHeight > 0)
720 createGbufferResources(sceneColorWidth, sceneColorHeight);
721 CullParams params{};
722 params.viewProj = viewProj;
723 params.planes[0] = glm::row(viewProj, 3) + glm::row(viewProj, 0);
724 params.planes[1] = glm::row(viewProj, 3) - glm::row(viewProj, 0);
725 params.planes[2] = glm::row(viewProj, 3) + glm::row(viewProj, 1);
726 params.planes[3] = glm::row(viewProj, 3) - glm::row(viewProj, 1);
727 params.planes[4] = glm::row(viewProj, 2);
728 params.planes[5] = glm::row(viewProj, 3) - glm::row(viewProj, 2);
729 for (glm::vec4 &plane : params.planes) {
730 const float length = glm::length(glm::vec3(plane));
731 if (length > 1e-6f) plane /= length;
732 }
733 params.cameraPos = glm::vec4(eye, 1.f);
734 params.screen = glm::vec4(float(gbufferWidth), float(gbufferHeight),
735 gbufferWidth > 0 ? 1.f / float(gbufferWidth) : 0.f,
736 gbufferHeight > 0 ? 1.f / float(gbufferHeight) : 0.f);
737 const float projScaleY = float(gbufferHeight) /
738 (2.f * std::tan(glm::radians(fovYDeg) * 0.5f));
739 params.clipNearFar = glm::vec4(nearZ, farZ, projScaleY,
740 gbufferDepthValid_ ? 1.f : 0.f);
741 params.hzbInfo = glm::vec4(float(gpuDrivenHzbOffsets_.size() - 1u), 0.f,
742 float(gpuDrivenHzbWidth_), float(gpuDrivenHzbHeight_));
743 uint32_t mipWidth = gpuDrivenHzbWidth_;
744 uint32_t mipHeight = gpuDrivenHzbHeight_;
745 uint32_t previousWidth = mipWidth;
746 uint32_t previousHeight = mipHeight;
747 for (uint32_t mip = 0; mip < gpuDrivenHzbOffsets_.size(); ++mip) {
748 HzbBuildParams build{};
749 build.info = glm::uvec4(mip, mipWidth, mipHeight,
750 mip == 0 ? 0u : gpuDrivenHzbOffsets_[mip - 1]);
751 build.source = glm::uvec4(previousWidth, previousHeight, 0u, 0u);
752 queue.WriteBuffer(gpuDrivenHzbParamsBuffer_, uint64_t(mip) * 256u, &build,
753 sizeof(build));
754 previousWidth = mipWidth;
755 previousHeight = mipHeight;
756 mipWidth = std::max(mipWidth >> 1u, 1u);
757 mipHeight = std::max(mipHeight >> 1u, 1u);
758 }
759
760 using BucketKey = std::pair<uint32_t, uint32_t>;
761 std::map<BucketKey, std::vector<const GpuInstance *>> grouped;
762 for (const GpuInstance &instance : gpuDrivenPending_)
763 grouped[{instance.meshId, instance.materialId}].push_back(&instance);
764
765 std::vector<CullInput> inputs;
766 std::vector<GpuIndirectCommand> commands;
767 std::vector<VisIndirectCommand> visCommands;
768 inputs.reserve(gpuDrivenPending_.size());
769 commands.reserve(grouped.size());
770 visCommands.reserve(grouped.size());
771 uint32_t outputBase = 0;
772 for (const auto &[key, bucketInstances] : grouped) {
773 Mesh *mesh = gpuDrivenMeshes_[key.first];
774 Material *material = gpuDrivenMaterials_[key.second];
775 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
776 const uint32_t bucketIndex = static_cast<uint32_t>(gpuDrivenBuckets_.size());
777 gpuDrivenBuckets_.push_back(
778 {mesh, material, outputBase, static_cast<uint32_t>(bucketInstances.size())});
779 GpuIndirectCommand command{};
780 command.indexCount = gpu->indexCount;
781 // Keep firstInstance zero: it is optional in WebGPU and requires the
782 // IndirectFirstInstance feature on adapters that expose it. The
783 // bucket's compacted model range is selected through the storage
784 // binding offset when drawing.
785 command.firstInstance = 0;
786 commands.push_back(command);
787 VisIndirectCommand visCommand{};
788 visCommand.vertexCount = gpu->indexCount;
789 visCommand.firstInstance = 0;
790 visCommands.push_back(visCommand);
791 for (const GpuInstance *instance : bucketInstances) {
792 CullInput input{};
793 input.model = instance->model;
794 input.bounds = mesh->hasBounds()
795 ? glm::vec4(mesh->boundsCx, mesh->boundsCy, mesh->boundsCz,
796 mesh->boundsRadius)
797 : glm::vec4(0.f, 0.f, 0.f, 1e20f);
798 input.bucket = bucketIndex;
799 input.outputBase = outputBase;
800 inputs.push_back(input);
801
802 const glm::vec3 center = glm::vec3(
803 input.model * glm::vec4(glm::vec3(input.bounds), 1.f));
804 const float scale = std::max({glm::length(glm::vec3(input.model[0])),
805 glm::length(glm::vec3(input.model[1])),
806 glm::length(glm::vec3(input.model[2]))});
807 bool visible = true;
808 for (const glm::vec4 &plane : params.planes) {
809 if (glm::dot(glm::vec3(plane), center) + plane.w < -input.bounds.w * scale) {
810 visible = false;
811 break;
812 }
813 }
814 if (visible) gpuDrivenVisible_.push_back(*instance);
815 }
816 // Storage-buffer binding offsets are aligned to WebGPU's common
817 // 256-byte limit (four mat4 values), so every bucket can bind its
818 // compacted range without relying on indirect firstInstance.
819 outputBase +=
820 (static_cast<uint32_t>(bucketInstances.size()) + 3u) & ~uint32_t(3u);
821 }
822
823 ensureGpuDrivenResources(outputBase,
824 static_cast<uint32_t>(commands.size()));
825 params.counts.x = static_cast<uint32_t>(inputs.size());
826 queue.WriteBuffer(gpuDrivenParamsBuffer_, 0, &params, sizeof(params));
827 queue.WriteBuffer(gpuDrivenInputBuffer_, 0, inputs.data(), inputs.size() * sizeof(CullInput));
828 queue.WriteBuffer(gpuDrivenIndirectBuffer_, 0, commands.data(),
829 commands.size() * sizeof(GpuIndirectCommand));
830 queue.WriteBuffer(gpuDrivenVisIndirectBuffer_, 0, visCommands.data(),
831 visCommands.size() * sizeof(VisIndirectCommand));
832 gpuDrivenDispatchCount_ = static_cast<uint32_t>(inputs.size());
833 gpuDrivenLastBucketCount_ = static_cast<uint32_t>(commands.size());
834 gpuDrivenComputePending_ = !inputs.empty() && !commands.empty();
835}
