131eve::Result<ObservationState> parseObservation(const eve::Value::Object& object,
132 const eve::SnapshotEnvelope& snapshot) {
133 auto tick = readUint64(object, "tick");
134 if (!tick) return eve::Result<ObservationState>::failure(tick.status());
135 auto revision = readUint64(object, "revision");
136 if (!revision) return eve::Result<ObservationState>::failure(revision.status());
137 auto stepCount = readUint64(object, "stepCount");
138 if (!stepCount) return eve::Result<ObservationState>::failure(stepCount.status());
139 auto duration = readInt64(object, "simulatedDurationNs");
140 if (!duration) return eve::Result<ObservationState>::failure(duration.status());
141 auto lastDelta = readFloat(object, "lastDeltaSeconds");
142 if (!lastDelta) return eve::Result<ObservationState>::failure(lastDelta.status());
143 if (duration.value() < 0 || revision.value() != stepCount.value() || tick.value() != snapshot.tick.value() ||
144 stepCount.value() != snapshot.revision.value()) {
145 return failure<ObservationState>(eve::DiagnosticCode::Conflict,
146 "physics snapshot payload progress disagrees with its envelope",
147 "payload.observation");
148 }
149
150 ObservationState result;
151 result.value.stepCount = stepCount.value();
152 result.value.lastTick = eve::SimulationTick(tick.value());
153 result.value.simulatedDuration = eve::Duration::fromNanoseconds(duration.value());
154 result.value.simulatedSeconds = result.value.simulatedDuration.seconds();
155 result.value.lastDeltaSeconds = lastDelta.value();
156 auto valid = detail::validateSimulationObservation(result.value, "physics.snapshot.observation");
157 if (!valid) return eve::Result<ObservationState>::failure(valid.status());
158 return eve::Result<ObservationState>::success(std::move(result));
159}

232 const auto* object = value.getIf<eve::Value::Object>();
233 if (!object || !hasExactFields(*object, {"active", "awake", "bullet", "fixedRotation", "id", "type", "x", "y", "z",

236 "3D physics snapshot body has unknown or missing fields", "payload.bodies");
237 Body3DState state;
238 auto id = readUint64(*object, "id");
239 if (!id || id.value() > static_cast<std::uint64_t>(std::numeric_limits<int>::max()))
240 return failure<Body3DState>(eve::DiagnosticCode::ParseError, "3D physics snapshot body id is invalid",
241 "payload.bodies.id");
242 state.id = static_cast<int>(id.value());
243 auto type = readString(*object, "type");
244 if (!type || !validBodyType(type.value()))
245 return failure<Body3DState>(eve::DiagnosticCode::InvalidArgument, "3D physics snapshot body type is invalid",
246 "payload.bodies.type");
247 state.type = std::move(type).takeValue();
248#define EV_READ_BODY3D_FLOAT(name) \
249 do { \
250 auto value = readFloat(*object, #name); \
251 if (!value) return eve::Result<Body3DState>::failure(value.status()); \
252 state.name = value.value(); \
253 } while (false)
254 EV_READ_BODY3D_FLOAT(x);
255 EV_READ_BODY3D_FLOAT(y);
256 EV_READ_BODY3D_FLOAT(z);
257 EV_READ_BODY3D_FLOAT(qx);
258 EV_READ_BODY3D_FLOAT(qy);
259 EV_READ_BODY3D_FLOAT(qz);
260 EV_READ_BODY3D_FLOAT(qw);
261 EV_READ_BODY3D_FLOAT(vx);
262 EV_READ_BODY3D_FLOAT(vy);
263 EV_READ_BODY3D_FLOAT(vz);
264 EV_READ_BODY3D_FLOAT(wx);
265 EV_READ_BODY3D_FLOAT(wy);
266 EV_READ_BODY3D_FLOAT(wz);
267#undef EV_READ_BODY3D_FLOAT
268 const double quaternionLengthSquared =
269 static_cast<double>(state.qx) * state.qx + static_cast<double>(state.qy) * state.qy +
270 static_cast<double>(state.qz) * state.qz + static_cast<double>(state.qw) * state.qw;
271 if (!(quaternionLengthSquared > 1e-16))
272 return failure<Body3DState>(eve::DiagnosticCode::InvalidArgument,
273 "3D physics snapshot rotation must be non-zero", "payload.bodies.rotation");
274#define EV_READ_BODY3D_BOOL(name) \

276 auto value = readBool(*object, #name); \
277 if (!value) return eve::Result<Body3DState>::failure(value.status()); \
278 state.name = value.value(); \
279 } while (false)
280 EV_READ_BODY3D_BOOL(active);
281 EV_READ_BODY3D_BOOL(bullet);
282 EV_READ_BODY3D_BOOL(awake);
283 EV_READ_BODY3D_BOOL(fixedRotation);
284#undef EV_READ_BODY3D_BOOL
285 return eve::Result<Body3DState>::success(std::move(state));

339eve::Value make2DPayload(const World& world, const std::vector<Body*>& bodies) {
340 const auto observation = world.simulationObservation();
341 eve::Value::Object object = observationFields(observation, world.simulationTick());
342 object.emplace("gravityX", eve::Value(world.getGravityX()));
343 object.emplace("gravityY", eve::Value(world.getGravityY()));
344 object.emplace("meter", eve::Value(world.getMeter()));
345 std::vector<Body*> sortedBodies = bodies;
346 std::sort(sortedBodies.begin(), sortedBodies.end(),
347 [](const Body* left, const Body* right) { return left->getId() < right->getId(); });
348 eve::Value::Array values;
349 values.reserve(sortedBodies.size());
350 for (const Body* body : sortedBodies)
351 if (body && body->isValid()) values.push_back(bodyValue(*body));
352 object.emplace("bodies", eve::Value(std::move(values)));
353 return eve::Value(std::move(object));
354}

432eve::Result<void> World::restore(const eve::SnapshotEnvelope& snapshotValue,
433 const eve::SnapshotHashProvider& hashProvider) {
434 if (!isValid() || !simulation_)
435 return eve::Result<void>::failure(eve::Diagnostic::error(
436 eve::DiagnosticCode::PreconditionViolation, "Cannot restore a destroyed or uninitialized physics world",
437 "physics.world.restore"));
438 auto verified = eve::verifySnapshotEnvelope(snapshotValue, hashProvider);
439 if (!verified) return verified;
440 auto envelope = checkEnvelope(snapshotValue, kWorld2DType, kWorld2DSchema);
441 if (!envelope) return envelope;
442 const auto* object = snapshotValue.payload.getIf<eve::Value::Object>();
443 if (!object || !hasExactFields(*object, {"bodies", "gravityX", "gravityY", "meter", "tick", "revision", "stepCount",
444 "simulatedDurationNs", "lastDeltaSeconds"}))
445 return eve::Result<void>::failure(eve::Diagnostic::error(
446 eve::DiagnosticCode::ParseError, "2D physics snapshot payload has unknown or missing fields", "payload"));
447 auto observation = parseObservation(*object, snapshotValue);
448 if (!observation) return eve::Result<void>::failure(observation.status());
449 auto bodies = parseBodies2D(*object);
