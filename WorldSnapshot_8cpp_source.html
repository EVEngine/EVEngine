35eve::Result<eve::SnapshotEnvelope> World::snapshot(const eve::SnapshotHashProvider& hashProvider) const {
36 if (pendingForceStep_)
37 return Result<SnapshotEnvelope>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
38 "Complete the submitted force step before snapshot",
39 "physics.forces.snapshot"));
40 if (!isValid() || !simulation_)
41 return eve::Result<eve::SnapshotEnvelope>::failure(eve::Diagnostic::error(
42 eve::DiagnosticCode::PreconditionViolation, "Cannot snapshot a destroyed or uninitialized physics world",
43 "physics.world.snapshot"));
44 if (raw()->IsLocked() || simulationStepActive_)
45 return Result<SnapshotEnvelope>::failure(
46 Diagnostic::error(DiagnosticCode::PreconditionViolation, "Cannot snapshot while stepping"));
47 auto joints = WorldSnapshotAccess::captureJoints2D(*this);
48 if (!joints) return Result<SnapshotEnvelope>::failure(joints.status());
49 auto schema = snapshotSchema(kWorld2DSchema);
50 if (!schema) return eve::Result<eve::SnapshotEnvelope>::failure(schema.status());
51 const auto observation = simulationObservation();
52 std::vector<Body*> bodies(bodies_.begin(), bodies_.end());
53 auto payload = make2DPayload(*this, bodies);
54 payload.set("joints", std::move(joints).takeValue());
55 return eve::makeSnapshotEnvelope(
56 std::string(kWorld2DType), std::move(schema).takeValue(), eve::SchemaVersion(kSnapshot2DVersion), instanceId_,
57 eve::Revision(observation.stepCount), simulationTick(), std::move(payload), hashProvider);

60eve::Result<void> World::restore(const eve::SnapshotEnvelope& snapshotValue,
61 const eve::SnapshotHashProvider& hashProvider) {
62 if (!isValid() || !simulation_)
63 return eve::Result<void>::failure(eve::Diagnostic::error(
64 eve::DiagnosticCode::PreconditionViolation, "Cannot restore a destroyed or uninitialized physics world",
65 "physics.world.restore"));
66 auto verified = eve::verifySnapshotEnvelope(snapshotValue, hashProvider);
67 if (!verified) return verified;
68 auto envelope = checkEnvelope(snapshotValue, kWorld2DType, kWorld2DSchema);
69 if (!envelope) return envelope;
70 if (raw()->IsLocked() || simulationStepActive_)
71 return Result<void>::failure(
72 Diagnostic::error(DiagnosticCode::PreconditionViolation, "Cannot restore while stepping"));
73 if (snapshotValue.instanceId.isNil() || snapshotValue.instanceId != instanceId_)
74 return Result<void>::failure(Diagnostic::error(
75 DiagnosticCode::Conflict, "physics snapshot belongs to a different world instance", "snapshot.instanceId"));
76 eve::Value migratedPayload = snapshotValue.payload;
77 if (snapshotValue.schemaVersion.value() == 2) {
78 auto migration = migratePlanarV2(migratedPayload);
79 if (!migration) return migration;
80 }
81 const auto* object = migratedPayload.getIf<eve::Value::Object>();
82 if (!object || !hasExactFields(*object, {"bodies", "joints", "allowSleep", "gravityX", "gravityY", "meter", "tick",
83 "revision", "stepCount", "simulatedDurationNs", "lastDeltaSeconds"}))
84 return eve::Result<void>::failure(eve::Diagnostic::error(
85 eve::DiagnosticCode::ParseError, "2D physics snapshot payload has unknown or missing fields", "payload"));
86 auto observation = parseObservation(*object, snapshotValue);
87 if (!observation) return eve::Result<void>::failure(observation.status());
88 auto bodies = parseBodies2D(*object);
89 if (!bodies) return eve::Result<void>::failure(bodies.status());
90 auto gravityX = readFloat(*object, "gravityX");
91 if (!gravityX) return eve::Result<void>::failure(gravityX.status());
92 auto gravityY = readFloat(*object, "gravityY");
93 if (!gravityY) return eve::Result<void>::failure(gravityY.status());
94 auto meter = readFloat(*object, "meter");
95 if (!meter) return eve::Result<void>::failure(meter.status());
96 if (meter.value() <= 0.f)
97 return eve::Result<void>::failure(eve::Diagnostic::error(
98 eve::DiagnosticCode::InvalidArgument, "physics snapshot meter must be positive", "payload.meter"));
99
100 std::set<int> snapshotIds;
101 auto matched = matchBodyIds(bodies.value(), snapshotIds);
102 if (!matched) return matched;
103 auto handleRefresh = prepareRuntimeHandleRefresh(bodies.value().size());
104 if (!handleRefresh) return handleRefresh;
105 auto prepared = prepareWorld2D(bodies.value(), gravityX.value(), gravityY.value(), meter.value(),
106 observation.value().value, instanceId_);
107 if (!prepared) return eve::Result<void>::failure(prepared.status());
108 const auto* jointValues = field(*object, "joints")->getIf<Value::Array>();
109 if (!jointValues)
110 return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError, "Planar joints must be an array"));
111 auto sleeping = readBool(*object, "allowSleep");
112 if (!sleeping) return Result<void>::failure(sleeping.status());
113 prepared.value()->raw()->SetAllowSleeping(sleeping.value());
114 auto jointRestore = WorldSnapshotAccess::restoreJoints2D(*prepared.value(), *jointValues, nextJointHandleIndex_);
115 if (!jointRestore) return jointRestore;
116 WorldSnapshotAccess::adopt(*this, *prepared.value());
117 refreshRuntimeHandlesAfterRestore();
118 clearContactEvents();
119 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
120}

149eve::Result<void> World3D::restore(const eve::SnapshotEnvelope& snapshotValue,
150 const eve::SnapshotHashProvider& hashProvider) {
151 if (rigidBackendKind_ != RigidBackend3D::Box3D) return restorePortable(snapshotValue, hashProvider);
152 if (couplingStepActive_)
153 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
154 "Cannot restore while stepping", "physics.world3d.restore"));
155 if (!isValid() || !simulation_)
156 return eve::Result<void>::failure(eve::Diagnostic::error(
157 eve::DiagnosticCode::PreconditionViolation, "Cannot restore a destroyed or uninitialized 3D physics world",
158 "physics.world3d.restore"));
159 auto verified = eve::verifySnapshotEnvelope(snapshotValue, hashProvider);
160 if (!verified) return verified;
161 auto envelope = checkEnvelope(snapshotValue, kWorld3DType, kWorld3DSchema);
162 if (!envelope) return envelope;
163 const bool legacyV1 = snapshotValue.schemaVersion.value() == 1;
164 if (!legacyV1 && (snapshotValue.instanceId.isNil() || snapshotValue.instanceId != instanceId_))
165 return eve::Result<void>::failure(
166 eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
167 "3D physics snapshot belongs to a different world instance", "snapshot.instanceId"));
168 eve::Value migratedPayload = snapshotValue.payload;
169 if (legacyV1) {
170 auto* migrated = migratedPayload.getIf<eve::Value::Object>();
171 if (!migrated)
172 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::ParseError,
173 "v1 3D physics payload is malformed", "payload"));
174 auto legacyShapes = shapesTopologyValue(std::vector<Shape3D*>(shapes_.begin(), shapes_.end()));
175 for (auto& shape : *legacyShapes.getIf<Value::Array>()) {
176 shape.getIf<Value::Object>()->erase("localTransform");
177 shape.getIf<Value::Object>()->erase("contact");
178 }
179 migrated->emplace("shapes", std::move(legacyShapes));
180 auto legacyJoints = jointsTopologyValue(std::vector<Joint3D*>(joints_.begin(), joints_.end()));
181 for (auto& joint : *legacyJoints.getIf<Value::Array>()) joint.getIf<Value::Object>()->erase("parameters");
182 migrated->emplace("joints", std::move(legacyJoints));
183 const eve::Value* bodyValues = field(*migrated, "bodies");
184 if (!bodyValues || bodyValues->arraySize() != bodies_.size())
185 return eve::Result<void>::failure(
186 eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
187 "v1 snapshot requires topology-compatible live bodies", "payload.bodies"));
188 }
189 if (snapshotValue.schemaVersion.value() < 3) {
190 auto* legacy = migratedPayload.getIf<Value::Object>();
191 if (!legacy || legacy->contains("couplings"))
192 return Result<void>::failure(
193 Diagnostic::error(DiagnosticCode::ParseError, "Legacy snapshot has unexpected transmission fields"));
194 if (legacy->contains("allowSleep"))
195 return Result<void>::failure(
196 Diagnostic::error(DiagnosticCode::ParseError, "Legacy world has unexpected dynamics"));
197 legacy->emplace("allowSleep", Value(false));
198 auto* bodyValues = migratedPayload.find("bodies");
199 auto* bodyArray = bodyValues ? bodyValues->getIf<Value::Array>() : nullptr;
200 if (!bodyArray)
201 return Result<void>::failure(
202 Diagnostic::error(DiagnosticCode::ParseError, "Legacy bodies must be an array"));
203 const auto defaults = b3DefaultBodyDef();
204 for (auto& body : *bodyArray) {
205 auto* bodyObject = body.getIf<Value::Object>();
206 if (!bodyObject || bodyObject->contains("dynamics"))
207 return Result<void>::failure(
208 Diagnostic::error(DiagnosticCode::ParseError, "Legacy body has unexpected dynamics"));
209 bodyObject->emplace("dynamics", Value(Value::Object{{"linearDamping", Value(defaults.linearDamping)},
210 {"angularDamping", Value(defaults.angularDamping)},
211 {"gravityScale", Value(defaults.gravityScale)},
212 {"sleepThreshold", Value(defaults.sleepThreshold)},
213 {"allowSleep", Value(defaults.enableSleep)}}));
214 }
215 legacy->emplace("couplings", Value(Value::Array{}));
216 }
217 const auto* object = migratedPayload.getIf<eve::Value::Object>();
218 if (!object || !hasExactFields(
219 *object, {"bodies", "couplings", "allowSleep", "gravityX", "gravityY", "gravityZ", "joints",
220 "shapes", "tick", "revision", "stepCount", "simulatedDurationNs", "lastDeltaSeconds"}))
221 return eve::Result<void>::failure(eve::Diagnostic::error(
222 eve::DiagnosticCode::ParseError, "3D physics snapshot payload has unknown or missing fields", "payload"));
223 auto observation = parseObservation(*object, snapshotValue);
224 if (!observation) return eve::Result<void>::failure(observation.status());
225 auto bodies = parseBodies3D(*object);
226 if (!bodies) return eve::Result<void>::failure(bodies.status());
227 auto gravityX = readFloat(*object, "gravityX");
228 if (!gravityX) return eve::Result<void>::failure(gravityX.status());
229 auto gravityY = readFloat(*object, "gravityY");
230 if (!gravityY) return eve::Result<void>::failure(gravityY.status());
231 auto gravityZ = readFloat(*object, "gravityZ");
232 if (!gravityZ) return eve::Result<void>::failure(gravityZ.status());
233 const eve::Value* snapshotShapes = field(*object, "shapes");
234 const eve::Value* snapshotJoints = field(*object, "joints");
235 if (!snapshotShapes || !snapshotShapes->isArray() || !snapshotJoints || !snapshotJoints->isArray())
236 return eve::Result<void>::failure(eve::Diagnostic::error(
237 eve::DiagnosticCode::ParseError, "3D physics snapshot topology fields must be arrays", "payload.shapes"));
238 std::set<int> snapshotIds;
239 auto matched = matchBodyIds(bodies.value(), snapshotIds);
240 if (!matched) return matched;
241 auto handleRefresh =
242 prepareRuntimeHandleRefresh(bodies.value().size(), snapshotShapes->arraySize(), snapshotJoints->arraySize());
243 if (!handleRefresh) return handleRefresh;
244 auto allowSleep = readBool(*object, "allowSleep");
245 if (!allowSleep) return Result<void>::failure(allowSleep.status());
246 auto prepared = prepareWorld3D(bodies.value(), *snapshotShapes->getIf<eve::Value::Array>(),
247 *snapshotJoints->getIf<eve::Value::Array>(), gravityX.value(), gravityY.value(),
248 gravityZ.value(), observation.value().value, instanceId_,
249 snapshotValue.schemaVersion.value() < 3, allowSleep.value());
250 if (!prepared) return eve::Result<void>::failure(prepared.status());
251 auto preparedLinks = WorldSnapshotAccess::prepareRuntimeLinks(*this, *prepared.value());
252 if (!preparedLinks) return preparedLinks;
253 const auto* couplings = field(*object, "couplings")->getIf<Value::Array>();
254 if (!couplings)
255 return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError, "Couplings must be an array"));
256 auto preparedCouplings = WorldSnapshotAccess::restoreCouplings3D(*prepared.value(), *couplings);
257 if (!preparedCouplings) return preparedCouplings;
258 auto restoredObservation = simulation_->restoreObservation(observation.value().value);
259 if (!restoredObservation) return restoredObservation;
260 WorldSnapshotAccess::adopt(*this, *prepared.value());
261 ++queryEpoch_;
262 clearContactEvents();
263 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
264}

294void World3D::refreshRuntimeHandlesAfterRestore() {
295 ++queryEpoch_;
296 shapeRecords_.clear();
297 shapeHandles_.clear();
298 shapeRawHandles_.clear();
299 jointHandles_.clear();
300 for (Body3D* body : bodies_)
301 if (body && body->isValid()) body->runtimeHandle_ = nextBodyRuntimeHandle();
302 for (Shape3D* shape : shapes_) {
303 if (!shape || !shape->isValid()) continue;
304 shape->runtimeHandle_ = nextShapeRuntimeHandle();
305 indexShapeHandle(shape);
306 }
307 for (Joint3D* joint : joints_) {
308 if (!joint || !joint->isValid()) continue;
309 joint->runtimeHandle_ = nextJointRuntimeHandle();
310 jointHandles_[joint->runtimeHandle_] = joint;
311 }
312}
