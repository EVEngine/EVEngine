48eve::Result<void> SoftBodyModelDefinition::validate() const {
49 if (sourceMesh.empty()) return invalid<void>("sourceMesh must not be empty", "sourceMesh");
50 if (surfaceName(surfaceSampling).empty())
51 return invalid<void>("unsupported surface sampling mode", "surfaceSampling");
52 if (volumeName(volumeSampling).empty())
53 return invalid<void>("unsupported volume sampling mode", "volumeSampling");
54 if (surfaceResolution < 2 || surfaceResolution > 128 || volumeResolution < 2 || volumeResolution > 128 ||
55 shapeResolution < 2 || shapeResolution > 128)
56 return invalid<void>("sampling resolutions must be in [2, 128]", "resolution");
57 if (!std::isfinite(maxAnisotropy) || maxAnisotropy < 1.f || maxAnisotropy > 5.f)
58 return invalid<void>("maxAnisotropy must be in [1, 5]", "maxAnisotropy");
59 if (!std::isfinite(smoothing) || smoothing < 0.f || smoothing > 1.f)
60 return invalid<void>("smoothing must be in [0, 1]", "smoothing");
61 if (surfaceSampling == SoftBodySurfaceSampling::None && volumeSampling == SoftBodyVolumeSampling::None)
62 return invalid<void>("at least one sampling mode must be enabled", "sampling");
63 return eve::Result<void>::success();
