50Result<RoadEdgeSplitResult> RoadNetwork::splitEdgeAtSplineParameter(std::uint32_t edgeId, float parameter,
51 float junctionRadius, std::uint32_t nodeId,
52 std::uint32_t secondEdgeId) {
53 const int idx = findEdgeIndex(edgeId);
54 if (idx < 0)
55 return Result<RoadEdgeSplitResult>::failure(
56 Diagnostic::error(DiagnosticCode::NotFound, "unknown edge", "edgeId"));
57 if (!std::isfinite(parameter) || parameter <= 0.f || parameter >= 1.f)
58 return Result<RoadEdgeSplitResult>::failure(Diagnostic::error(
59 DiagnosticCode::InvalidArgument, "spline split parameter must be inside (0,1)", "parameter"));
60
61 const auto& original = edges_[static_cast<std::size_t>(idx)];
62 auto path = makeCenterlineSpline(original);
63 if (!path.ok()) return Result<RoadEdgeSplitResult>::failure(path.status());
64 auto evaluated = path.value().evaluateResult(parameter);
65 if (!evaluated.ok()) return Result<RoadEdgeSplitResult>::failure(evaluated.status());
66
67 const int segmentCount = path.value().segmentCount();
68 const float scaled = parameter * static_cast<float>(segmentCount);
69 const int nearest = static_cast<int>(std::round(scaled));
70 auto candidate = *this;
71 std::size_t splitIndex = 0;
72 if (nearest > 0 && nearest < segmentCount && std::fabs(scaled - static_cast<float>(nearest)) <= 1e-5f) {
73 splitIndex = static_cast<std::size_t>(nearest);
74 } else {
75 const int segment = std::clamp(static_cast<int>(std::floor(scaled)), 0, segmentCount - 1);
76 auto points = original.controlPoints;
77 splitIndex = static_cast<std::size_t>(segment + 1);
78 points.insert(points.begin() + static_cast<std::ptrdiff_t>(splitIndex),
79 RoadControlPoint{evaluated.value().x, evaluated.value().y, evaluated.value().z,
80 evaluated.value().scaleX, evaluated.value().rollDegrees,
81 original.controlPoints[static_cast<std::size_t>(segment)].splineStrength,
82 original.controlPoints[static_cast<std::size_t>(segment)].pathControl,
83 original.controlPoints[static_cast<std::size_t>(segment)].alignStraightStart,
84 original.controlPoints[static_cast<std::size_t>(segment)].alignStraightEnd,
85 original.controlPoints[static_cast<std::size_t>(segment)].circularFromEnd,
86 -evaluated.value().rollPivotSide});
87 auto changed = candidate.setEdgeControlPoints(edgeId, std::move(points));
88 if (!changed.ok()) return Result<RoadEdgeSplitResult>::failure(changed.status());
89 }
90 auto split = candidate.splitEdge(edgeId, splitIndex, junctionRadius, nodeId, secondEdgeId);
91 if (!split.ok()) return split;
92 const auto result = split.value();
93 candidate.revision_ = revision_ + 1;
94 *this = std::move(candidate);
95 return Result<RoadEdgeSplitResult>::success(result);
96}

98Result<RoadEdgeSplitResult> RoadNetwork::splitEdge(std::uint32_t edgeId, std::size_t controlPointIndex,
99 float junctionRadius, std::uint32_t nodeId,
100 std::uint32_t secondEdgeId) {
101 const int idx = findEdgeIndex(edgeId);
102 if (idx < 0)
103 return Result<RoadEdgeSplitResult>::failure(
104 Diagnostic::error(DiagnosticCode::NotFound, "unknown edge", "edgeId"));
105 const RoadEdge original = edges_[static_cast<std::size_t>(idx)];
106 if (controlPointIndex == 0 || controlPointIndex + 1 >= original.controlPoints.size())
107 return Result<RoadEdgeSplitResult>::failure(Diagnostic::error(
108 DiagnosticCode::InvalidArgument, "edge split requires an interior control point", "controlPointIndex"));
109 if (!std::isfinite(junctionRadius) || junctionRadius <= 0.f)
110 return Result<RoadEdgeSplitResult>::failure(
111 Diagnostic::error(DiagnosticCode::InvalidArgument, "edge split radius must be positive", "junctionRadius"));
112
113 auto candidate = *this;
114 const auto splitPoint = original.controlPoints[controlPointIndex];
115 std::uint32_t insertedNodeId = nodeId;
116 if (insertedNodeId == 0) {
117 auto node = candidate.addNode(splitPoint.x, splitPoint.y, splitPoint.z, junctionRadius);
118 if (!node.ok()) return Result<RoadEdgeSplitResult>::failure(node.status());
119 insertedNodeId = node.value();
120 } else {
121 auto node = candidate.restoreNode({insertedNodeId, splitPoint.x, splitPoint.y, splitPoint.z, junctionRadius});
122 if (!node.ok()) return Result<RoadEdgeSplitResult>::failure(node.status());
123 }
124
125 auto styles = detail::splitRoadAuthoring(original, controlPointIndex);
126 if (!styles.ok()) return Result<RoadEdgeSplitResult>::failure(styles.status());
127 auto& first = candidate.edges_[static_cast<std::size_t>(idx)];
128 first.style = styles.value().first;
129 first.to = insertedNodeId;
130 first.controlPoints.resize(controlPointIndex + 1u);
131 first.controlPoints.back() = splitPoint;
132 std::vector<RoadControlPoint> continuation(
133 original.controlPoints.begin() + static_cast<std::ptrdiff_t>(controlPointIndex), original.controlPoints.end());
134 std::uint32_t continuationEdgeId = secondEdgeId;
135 if (continuationEdgeId == 0) {
136 continuationEdgeId = candidate.nextEdgeId_;
137 auto second = candidate.restoreEdge({continuationEdgeId, insertedNodeId, original.to, std::move(continuation),
138 original.lanesForward, original.lanesBackward, styles.value().second});
139 if (!second.ok()) return Result<RoadEdgeSplitResult>::failure(second.status());
140 } else {
141 auto second = candidate.restoreEdge({continuationEdgeId, insertedNodeId, original.to, std::move(continuation),
142 original.lanesForward, original.lanesBackward, styles.value().second});
143 if (!second.ok()) return Result<RoadEdgeSplitResult>::failure(second.status());
144 }
145
146 for (auto& node : candidate.nodes_)
147 if (node.id == original.to) {
148 for (auto& port : node.connection.ports)
149 if (port.edgeId == edgeId) port.edgeId = continuationEdgeId;
150 for (auto& route : node.connection.facilityRoutes) {
151 if (route.firstEdge == edgeId) route.firstEdge = continuationEdgeId;
152 if (route.secondEdge == edgeId) route.secondEdge = continuationEdgeId;
153 }
154 }
155
156 // The original id remains attached to the `from` half. References whose
157 // port lived at the old `to` endpoint migrate to the continuation edge.
158 for (auto& link : candidate.laneLinks_) {
159 if (link.inEdge == edgeId && link.inDirection == RoadLaneDirection::Forward) link.inEdge = continuationEdgeId;
160 if (link.outEdge == edgeId && link.outDirection == RoadLaneDirection::Backward)
161 link.outEdge = continuationEdgeId;
162 }
163 for (auto& link : candidate.blockedLaneLinks_) {
164 if (link.inEdge == edgeId && link.inDirection == RoadLaneDirection::Forward) link.inEdge = continuationEdgeId;
165 if (link.outEdge == edgeId && link.outDirection == RoadLaneDirection::Backward)
166 link.outEdge = continuationEdgeId;
167 }
168 candidate.normalizeLaneLinkOrder();
169 for (int lane = 0; lane < original.lanesForward; ++lane) {
170 auto linked = candidate.addLaneLink(
171 {edgeId, lane, continuationEdgeId, lane, RoadLaneDirection::Forward, RoadLaneDirection::Forward});
172 if (!linked.ok()) return Result<RoadEdgeSplitResult>::failure(linked.status());
173 }
174 for (int lane = 0; lane < original.lanesBackward; ++lane) {
175 auto linked = candidate.addLaneLink(
176 {continuationEdgeId, lane, edgeId, lane, RoadLaneDirection::Backward, RoadLaneDirection::Backward});
177 if (!linked.ok()) return Result<RoadEdgeSplitResult>::failure(linked.status());
178 }
179 auto valid = candidate.validate();
180 if (!valid.ok()) return Result<RoadEdgeSplitResult>::failure(valid.status());
181 candidate.revision_ = revision_ + 1;
182 const RoadEdgeSplitResult result{insertedNodeId, edgeId, continuationEdgeId};
183 *this = std::move(candidate);
184 return Result<RoadEdgeSplitResult>::success(result);
185}

192Result<RoadEdgeSplitResult> RoadNetwork::splitEdgeAtPosition(std::uint32_t edgeId, RoadControlPoint position,
193 float maxDistance, float junctionRadius,
194 std::uint32_t nodeId, std::uint32_t secondEdgeId) {
195 const int idx = findEdgeIndex(edgeId);
196 if (idx < 0)
197 return Result<RoadEdgeSplitResult>::failure(
198 Diagnostic::error(DiagnosticCode::NotFound, "unknown edge", "edgeId"));
199 if (!finite3(position.x, position.y, position.z) || !std::isfinite(maxDistance) || maxDistance < 0.f)
200 return Result<RoadEdgeSplitResult>::failure(Diagnostic::error(
201 DiagnosticCode::InvalidArgument, "split position and snap distance must be finite", "position"));
202
203 const auto& edge = edges_[static_cast<std::size_t>(idx)];
204 auto path = makeCenterlineSpline(edge);
205 if (!path.ok()) return Result<RoadEdgeSplitResult>::failure(path.status());
206 const int intervals = std::min(4096, std::max(32, path.value().segmentCount() * 32));
207 float bestDistanceSquared = std::numeric_limits<float>::max();
208 float bestParameter = 0.f;
209 auto previous = path.value().evaluateResult(0.f);
210 if (!previous.ok()) return Result<RoadEdgeSplitResult>::failure(previous.status());
211 for (int i = 0; i < intervals; ++i) {
212 const float nextParameter = static_cast<float>(i + 1) / static_cast<float>(intervals);
213 auto next = path.value().evaluateResult(nextParameter);
214 if (!next.ok()) return Result<RoadEdgeSplitResult>::failure(next.status());
215 const RoadControlPoint a{previous.value().x, previous.value().y, previous.value().z};
216 const RoadControlPoint b{next.value().x, next.value().y, next.value().z};
217 const float dx = b.x - a.x, dy = b.y - a.y, dz = b.z - a.z;
218 const float lengthSquared = dx * dx + dy * dy + dz * dz;
219 if (lengthSquared > 1e-8f) {
220 const float local = std::clamp(
221 ((position.x - a.x) * dx + (position.y - a.y) * dy + (position.z - a.z) * dz) / lengthSquared, 0.f,
222 1.f);
223 const RoadControlPoint projected{a.x + dx * local, a.y + dy * local, a.z + dz * local};
224 const float px = position.x - projected.x, py = position.y - projected.y, pz = position.z - projected.z;
225 const float distanceSquared = px * px + py * py + pz * pz;
226 if (distanceSquared < bestDistanceSquared) {
227 bestDistanceSquared = distanceSquared;
228 const float previousParameter = static_cast<float>(i) / static_cast<float>(intervals);
229 bestParameter = previousParameter + (nextParameter - previousParameter) * local;
230 }
231 }
232 previous = std::move(next);
233 }
234 // The chord fraction is not a spline parameter fraction, even on a straight
235 // Catmull span. Refine on the actual curve before applying the snap limit.
236 float lower = std::max(0.f, bestParameter - 1.f / static_cast<float>(intervals));
237 float upper = std::min(1.f, bestParameter + 1.f / static_cast<float>(intervals));
238 auto squaredDistance = [&](float parameter) -> Result<double> {
239 auto point = path.value().evaluateResult(parameter);
240 if (!point.ok()) return Result<double>::failure(point.status());
241 const double dx = static_cast<double>(point.value().x) - position.x;
242 const double dy = static_cast<double>(point.value().y) - position.y;
243 const double dz = static_cast<double>(point.value().z) - position.z;
244 return Result<double>::success(dx * dx + dy * dy + dz * dz);
245 };
246 for (int iteration = 0; iteration < 40; ++iteration) {
247 const float left = lower + (upper - lower) / 3.f;
248 const float right = upper - (upper - lower) / 3.f;
249 if (left == right) break;
250 auto leftDistance = squaredDistance(left);
251 if (!leftDistance.ok()) return Result<RoadEdgeSplitResult>::failure(leftDistance.status());
252 auto rightDistance = squaredDistance(right);
253 if (!rightDistance.ok()) return Result<RoadEdgeSplitResult>::failure(rightDistance.status());
254 if (leftDistance.value() <= rightDistance.value())
255 upper = right;
256 else
257 lower = left;
258 }
259 bestParameter = lower + (upper - lower) * 0.5f;
260 auto actualDistance = squaredDistance(bestParameter);
261 if (!actualDistance.ok()) return Result<RoadEdgeSplitResult>::failure(actualDistance.status());
262 if (actualDistance.value() > static_cast<double>(maxDistance) * maxDistance)
263 return Result<RoadEdgeSplitResult>::failure(Diagnostic::error(
264 DiagnosticCode::PreconditionViolation, "position is outside the road snap distance", "maxDistance"));
265
266 constexpr float endpointTolerance = 1e-4f;
267 if (bestParameter <= endpointTolerance || bestParameter >= 1.f - endpointTolerance)
268 return Result<RoadEdgeSplitResult>::failure(Diagnostic::error(
269 DiagnosticCode::PreconditionViolation, "position resolves to an edge endpoint", "position"));
270 return splitEdgeAtSplineParameter(edgeId, bestParameter, junctionRadius, nodeId, secondEdgeId);
271}
