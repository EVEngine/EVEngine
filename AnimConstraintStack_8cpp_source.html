40void AnimConstraintStack::apply(AnimPose* pose) const {
41 if (!pose) throw Exception("AnimConstraintStack.apply: pose is null");
42 if (!skeleton_) throw Exception("AnimConstraintStack.apply: skeleton is null");
43 pose->computeWorld(skeleton_);
44 for (const Constraint& constraint : constraints_) {
45 if (constraint.type == Type::Aim) {
46 pose->aimBone(skeleton_, constraint.a, constraint.x, constraint.y, constraint.z, constraint.weight);
47 } else if (constraint.type == Type::TwoBone) {
48 pose->solveTwoBoneIK(skeleton_, constraint.a, constraint.b, constraint.c, constraint.x, constraint.y,
49 constraint.z, constraint.weight);
50 } else {
51 pose->computeWorld(skeleton_);
52 const TransformTRS& foot = pose->world(constraint.c);
53 const float targetY = constraint.y + constraint.offset;
54 pose->solveTwoBoneIK(skeleton_, constraint.a, constraint.b, constraint.c, foot.px, targetY, foot.pz,
55 constraint.weight);
56 pose->computeWorld(skeleton_);
57 const TransformTRS& planted = pose->world(constraint.c);
58 pose->aimBone(skeleton_, constraint.c, planted.px + constraint.nx, planted.py + constraint.ny,
59 planted.pz + constraint.nz, constraint.weight);
