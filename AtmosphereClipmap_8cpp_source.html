10void AtmosphereClipmap::configure(int levelCount, int resolution, float firstExtent) {

39int AtmosphereClipmap::levelForPosition(const glm::vec3 &worldPosition) const {
40 for (std::size_t i = 0; i < levels_.size(); ++i) {
41 const AtmosphereClipmapLevel &level = levels_[i];
42 const glm::vec3 d = glm::abs(worldPosition - level.center);
43 if (d.x <= level.extent && d.y <= level.extent && d.z <= level.extent) return int(i);
44 }
45 return -1;

48FogFroxel AtmosphereClipmap::sample(const glm::vec3 &worldPosition) const {
49 const int index = levelForPosition(worldPosition);
50 if (index < 0) return {};
51 const AtmosphereClipmapLevel &level = levels_[std::size_t(index)];
52 const glm::vec3 minCorner = level.center - glm::vec3(level.extent);
53 const glm::vec3 uvw = (worldPosition - minCorner) / (level.extent * 2.f);
54 return level.volume.sampleMedia(uvw.x, uvw.y, uvw.z);
55}
