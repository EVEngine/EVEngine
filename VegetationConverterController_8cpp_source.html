42EditorResult<void> VegetationConverterController::prepare() {
43 if (!request_)
44 return fail<void>(EditorStatus::Rejected, "editor.vegetation-converter.no-request",
45 "Set a vegetation conversion request before preparing it");
46 auto prepared = preparer_.prepare(*request_);
47 if (!prepared.ok()) {
48 candidate_.reset();
49 state_.preparedRevision = 0;
50 state_.assetCount = 0;
51 state_.findingCount = 0;
52 state_.diagnostics = project(prepared.diagnostics());
53 state_.phase = VegetationConverterState::Phase::Error;
54 return EditorResult<void>::failure(prepared.status());
55 }
56 candidate_ = std::move(prepared).takeValue();
57 state_.preparedRevision = state_.requestRevision;
58 state_.assetCount = candidate_->manifest.assets.size();
59 state_.findingCount = candidate_->findings.size();
60 state_.diagnostics.clear();
61 state_.phase = VegetationConverterState::Phase::Prepared;
62 return editing::applied<void>();
63}

65EditorResult<std::uint64_t> VegetationConverterController::publish() {
66 if (!candidate_ || state_.preparedRevision != state_.requestRevision)
67 return fail<std::uint64_t>(EditorStatus::Conflict, "editor.vegetation-converter.stale-candidate",
68 "Prepare the current vegetation request before publishing it");
69 auto published = publisher_.publish(*candidate_, expectedPublicationGeneration_);
70 if (!published.ok()) {
71 state_.diagnostics = published.diagnostics();
72 state_.phase = VegetationConverterState::Phase::Error;
73 return published;
74 }
75 expectedPublicationGeneration_ = published.value();
76 state_.publicationGeneration = published.value();
77 state_.diagnostics.clear();
78 state_.phase = VegetationConverterState::Phase::Published;
79 return published;
80}
