474void MotionDatabase::extractFeature(AnimClip* clip, float time, float dtSample, std::vector<float>& out, float& rootX,
475 float& rootZ, float& rootYaw, float& velX, float& velZ) const {
476 if (schema_) {
477 extractSchemaFeature(clip, time, out, rootX, rootZ, rootYaw, velX, velZ);
478 return;
479 }
480 if (locomotionFeatures_) {
481 extractLocomotionFeature(clip, time, out, rootX, rootZ, rootYaw, velX, velZ);
482 return;
483 }
484 out.assign(static_cast<size_t>(featureSize_), 0.f);
485 // Trajectory horizons need only the root's ancestor chain; pose features
486 // need only their own chains. Sampling every unrelated finger/face bone at
487 // all six horizons makes full-corpus baking unnecessarily expensive.
488 auto worldBone = [&](auto&& self, int bone, float sampleTime) -> TransformTRS {
489 const auto local = clip->sampleBone(bone, sampleTime, skeleton_->bindLocal(bone));
490 const int parent = skeleton_->getParent(bone);
491 return parent < 0 ? local : detail::mulTRS(self(self, parent, sampleTime), local);
492 };
493 auto worldAt = [&](int bone, float sampleTime, bool wrap = true) {
494 const float t = clampf(wrap ? clip->wrapTime(sampleTime) : sampleTime, 0.f, clip->getDuration());
495 return worldBone(worldBone, bone, t);
496 };
497 const int root = rootBone_;
498 const auto current = worldAt(root, time);
499 rootX = current.px;
500 rootZ = current.pz;
501 rootYaw = yawFromQuat(current.qx, current.qy, current.qz, current.qw);
502
503 // Sampling a looping pose wraps its root back to the origin. Trajectories
504 // must instead accumulate the displacement of every completed cycle.
505 float cycleX = 0.f, cycleZ = 0.f;
506 const float duration = clip->getDuration();
507 if (clip->getLoop() && duration > 1e-8f) {
508 const auto start = worldAt(root, 0.f, false);
509 const auto end = worldAt(root, duration, false);
510 cycleX = end.px - start.px;
511 cycleZ = end.pz - start.pz;
512 }
513 auto cyclesAt = [&](float t) { return clip->getLoop() && duration > 1e-8f ? std::floor(t / duration) : 0.f; };
514
515 // Velocity from nearby sample.
516 const float t1 = time + std::max(dtSample, 1e-3f);
517 const auto next = worldAt(root, t1);
518 const float nX = next.px + cyclesAt(t1) * cycleX;
519 const float nZ = next.pz + cyclesAt(t1) * cycleZ;
520 const float dtt = std::max(dtSample, 1e-3f);
521 velX = (nX - rootX) / dtt;
522 velZ = (nZ - rootZ) / dtt;
523
524 float cs = std::cos(rootYaw);
525 float sn = std::sin(rootYaw);
526 // Character-space: rotate world xz by -yaw
527 auto toLocal = [&](float wx, float wz, float& lx, float& lz) {
528 const float dx = wx - rootX;
529 const float dz = wz - rootZ;
530 lx = dx * cs - dz * sn;
531 lz = dx * sn + dz * cs;
532 };
533
534 out[0] = velX * cs - velZ * sn;
535 out[1] = velX * sn + velZ * cs;
536
537 const float horizons[3] = {0.33f, 0.66f, 1.0f};
538 for (int h = 0; h < 3; ++h) {
539 const auto fut = worldAt(root, time + horizons[h]);
540 float lx, lz;
541 const float cycles = cyclesAt(time + horizons[h]);
542 toLocal(fut.px + cycles * cycleX, fut.pz + cycles * cycleZ, lx, lz);
543 out[2 + h * 2] = lx;
544 out[2 + h * 2 + 1] = lz;
545 }
546
547 {
548 const auto fut = worldAt(root, time + 1.0f);
549 const float fyaw = yawFromQuat(fut.qx, fut.qy, fut.qz, fut.qw);
550 float fx, fz;
551 yawToForward(fyaw - rootYaw, fx, fz);
552 out[8] = fx;
553 out[9] = fz;
554 }
555
556 int base = 10;
557 for (int bone : featureBones_) {
558 float lx, lz;
559 const auto feature = worldAt(bone, time);
560 const float wy = feature.py;
561 toLocal(feature.px, feature.pz, lx, lz);
562 out[static_cast<size_t>(base)] = lx;
563 out[static_cast<size_t>(base + 1)] = wy;
564 out[static_cast<size_t>(base + 2)] = lz;
565 base += 3;
566 }

569void MotionDatabase::bake() {
570 if (schema_)
571 for (const auto& channel : schema_->layout.channels) {
572 if (channel.kind != MotionFeatureKind::Curve) continue;
573 if (!schema_->curves) throw Exception("MotionDatabase: scalar curve data must be configured before bake");
574 for (const auto* clip : clips_)
575 if (!schema_->curveSources.contains(clip))
576 throw Exception("MotionDatabase: added clip has no scalar curve source");
577 }
578 if (clips_.empty()) throw Exception("MotionDatabase.bake: no clips");
579 if (featureBones_.empty() && !schema_) {
580 // Default: all bones except root.
581 for (int i = 1; i < skeleton_->getBoneCount(); ++i) addFeatureBone(i);
582 if (featureBones_.empty() && skeleton_->getBoneCount() > 0) addFeatureBone(0);
583 }
584 computeFeatureSize();
585 baked_ = false;
586 std::vector<std::vector<Frame>> clipFrames(clips_.size());
587
588 detail::parallelAnimationItems(clips_.size(), clips_.size() < 16 ? 1 : 0, [&](std::size_t index) {
589 const int ci = static_cast<int>(index);
590 auto& output = clipFrames[index];
591 AnimClip* clip = clips_[static_cast<size_t>(ci)];
592 const float rate = schema_ ? static_cast<float>(schema_->layout.sampleRate)
593 : (clip->getSampleRate() > 0.f ? clip->getSampleRate() : 30.f);
594 const float dt = 1.f / rate;
595 const float dur = clip->getDuration();
596 if (dur <= 0.f) {
597 Frame f;
598 f.clipIndex = ci;
599 f.time = 0.f;
600 extractFeature(clip, 0.f, dt, f.feature, f.rootX, f.rootZ, f.rootYaw, f.velX, f.velZ);
601 if (schema_) f.trajectorySpeed = schemaTrajectorySpeed(f.feature);
602 output.push_back(std::move(f));
603 return;
604 }
605 const int schemaLast = schema_ ? static_cast<int>(std::floor(dur * rate)) : 0;
606 float t = 0.f;
607 for (int sample = 0;; ++sample, t = schema_ ? sample / rate : t + dt) {
608 if (schema_ ? (sample > schemaLast || (clip->getLoop() && t >= dur - 1e-5f)) : t >= dur - 1e-5f) break;
609 Frame f;
610 f.clipIndex = ci;
611 f.time = t;
612 extractFeature(clip, t, dt, f.feature, f.rootX, f.rootZ, f.rootYaw, f.velX, f.velZ);
613 if (schema_) f.trajectorySpeed = schemaTrajectorySpeed(f.feature);
614 output.push_back(std::move(f));
615 }
616 });
617 // Preserve frame IDs and floating-point reduction order exactly.
618 std::size_t total = 0;
619 for (const auto& output : clipFrames) total += output.size();
620 frames_.clear();
621 frames_.reserve(total);
622 clipFrameOffsets_.clear();
623 clipFrameOffsets_.reserve(clipFrames.size() + 1);
624 for (auto& output : clipFrames) {
625 clipFrameOffsets_.push_back(static_cast<int>(frames_.size()));
626 for (auto& frame : output) frames_.push_back(std::move(frame));
627 }
628 clipFrameOffsets_.push_back(static_cast<int>(frames_.size()));
629 if (schema_) {
630 normalizeSchemaFeatures();
631 baked_ = true;
632 return;
633 }
634 if (locomotionFeatures_) {
635 normalizeLocomotionFeatures();
636 baked_ = true;
637 return;
638 }
639 featureMean_.assign(static_cast<size_t>(featureSize_), 0.f);
640 featureInvStd_.assign(static_cast<size_t>(featureSize_), 0.f);
641 for (const Frame& frame : frames_)
642 for (int i = 0; i < featureSize_; ++i)
643 featureMean_[static_cast<size_t>(i)] += frame.feature[static_cast<size_t>(i)];
644 const float invCount = 1.f / static_cast<float>(frames_.size());
645 for (float& mean : featureMean_) mean *= invCount;
646 for (const Frame& frame : frames_)
647 for (int i = 0; i < featureSize_; ++i) {
648 const float d = frame.feature[static_cast<size_t>(i)] - featureMean_[static_cast<size_t>(i)];
649 featureInvStd_[static_cast<size_t>(i)] += d * d;
650 }
651 for (float& invStd : featureInvStd_) invStd = 1.f / std::sqrt(invStd * invCount + 1e-6f);
652 for (Frame& frame : frames_) normalizeFeature(frame.feature);
653 baked_ = true;
654}

707void MotionDatabase::extractLocomotionFeature(AnimClip* clip, float time, std::vector<float>& out, float& rootX,
708 float& rootZ, float& rootYaw, float& velX, float& velZ) const {
709 out.assign(detail::locomotionFeatureCount, 0.f);
710 const float duration = clip->getDuration();
711 const float delta = 1.f / 60.f;
712 auto world = [&](auto&& self, int bone, float t) -> TransformTRS {
713 const auto local = clip->sampleBone(bone, t, skeleton_->bindLocal(bone));
714 const int parent = skeleton_->getParent(bone);
715 return parent < 0 ? local : detail::mulTRS(self(self, parent, t), local);
716 };
717 auto poseAt = [&](int bone, float t) { return world(world, bone, clampf(clip->wrapTime(t), 0.f, duration)); };
718 // Accumulate complete root transforms at loop seams; turning cycles need
719 // rotation composition, not repeated addition of an unrotated displacement.
720 const auto start = world(world, rootBone_, 0.f), end = world(world, rootBone_, duration);
721 const float startYaw = detail::featureYaw(start);
722 const float cycleYaw = std::remainder(detail::featureYaw(end) - startYaw, 6.28318530718f);
723 auto rootAt = [&](float t) {
724 auto root = poseAt(rootBone_, t);
725 if (clip->getLoop() && duration > 1e-6f) {
726 const int cycles = static_cast<int>(std::floor(t / duration));
727 float x = root.px - start.px, z = root.pz - start.pz;
728 const float dx = end.px - start.px, dz = end.pz - start.pz;
729 const float cs = std::cos(cycleYaw), sn = std::sin(cycleYaw);
730 for (int i = 0; i < std::abs(cycles); ++i) {
731 if (cycles > 0) {
732 const float nextX = cs * x + sn * z + dx;
733 z = -sn * x + cs * z + dz;
734 x = nextX;
735 } else {
736 x -= dx;
737 z -= dz;
738 const float nextX = cs * x - sn * z;
739 z = sn * x + cs * z;
740 x = nextX;
741 }
742 }
743 root.px = start.px + x;
744 root.pz = start.pz + z;
745 root.py += cycles * (end.py - start.py);
746 const float angle = detail::featureYaw(root) + cycles * cycleYaw;
747 root.qx = root.qz = 0.f;
748 root.qy = std::sin(angle * .5f);
749 root.qw = std::cos(angle * .5f);
750 } else if (duration > 1e-6f && (t < 0.f || t > duration)) {
751 // Root trajectory extrapolation preserves velocity outside short
752 // starts/stops, instead of inventing a stationary future endpoint.
753 const float dt = std::min(delta, duration);
754 const auto a = world(world, rootBone_, t < 0.f ? 0.f : duration - dt);
755 const auto b = world(world, rootBone_, t < 0.f ? dt : duration);
756 const float extension = (t < 0.f ? t : t - duration) / dt;
757 root.px += (b.px - a.px) * extension;
758 root.py += (b.py - a.py) * extension;
759 root.pz += (b.pz - a.pz) * extension;
760 const float angle =
761 detail::featureYaw(root) +
762 std::remainder(detail::featureYaw(b) - detail::featureYaw(a), 6.28318530718f) * extension;
763 root.qx = root.qz = 0.f;
764 root.qy = std::sin(angle * .5f);
765 root.qw = std::cos(angle * .5f);
766 }
767 return root;
768 };
769 struct Sample {
770 float x, y, z, vx, vy, vz, yaw;
771 };
772 std::array<Sample, 5> samples{};
773 const auto current = rootAt(time);
774 rootX = current.px;
775 rootZ = current.pz;
776 rootYaw = detail::featureYaw(current);
777 for (int i = 0; i < 5; ++i) {
778 const float t = time + detail::locomotionHorizons[i];
779 const auto r = rootAt(t), previous = rootAt(t - delta);
780 samples[i] = {r.px - current.px,
781 r.py - current.py,
782 r.pz - current.pz,
783 (r.px - previous.px) / delta,
784 (r.py - previous.py) / delta,
785 (r.pz - previous.pz) / delta,
786 detail::featureYaw(r)};
787 }
788 velX = samples[1].vx;
789 velZ = samples[1].vz;
790 detail::encodeLocomotionTrajectory(std::span(out), samples, rootYaw);
791 std::array<TransformTRS, 4> now{}, past{};
792 now[0] = poseAt(rootBone_, time);
793 past[0] = poseAt(rootBone_, time - delta);
794 for (int i = 0; i < 3; ++i) {
795 now[i + 1] = poseAt(featureBones_[i], time);
796 past[i + 1] = poseAt(featureBones_[i], time - delta);
797 }
798 detail::encodeLocomotionPose(out, now, past, delta);
799}

801void MotionDatabase::normalizeLocomotionFeatures() {
802 featureMean_.assign(detail::locomotionFeatureCount, 0.f);
803 featureInvStd_.assign(detail::locomotionFeatureCount, 1.f);
804 // Separate channels use vector mean deviation (isotropic XZ/XYZ); the two
805 // feet share a centroid and scale, as the authored FeetVelZ group specifies.
806 const std::array<std::array<int, 3>, 12> channels{{{0, 2, 1},
807 {2, 2, 1},
808 {4, 2, 1},
809 {6, 2, 1},
810 {8, 2, 1},
811 {10, 2, 1},
812 {12, 2, 1},
813 {14, 2, 1},
814 {16, 3, 1},
815 {19, 3, 1},
816 {22, 3, 2},
817 {28, 2, 1}}};
818 for (const auto& channel : channels) {
819 const int base = channel[0], dimensions = channel[1], copies = channel[2];
820 std::array<double, 3> mean{};
821 const double count = static_cast<double>(frames_.size()) * copies;
822 for (const auto& frame : frames_)
823 for (int copy = 0; copy < copies; ++copy)
824 for (int axis = 0; axis < dimensions; ++axis)
825 mean[axis] += frame.feature[base + copy * dimensions + axis];
826 for (double& value : mean) value /= count;
827 double deviation = 0;
828 for (const auto& frame : frames_)
829 for (int copy = 0; copy < copies; ++copy) {
830 double square = 0;
831 for (int axis = 0; axis < dimensions; ++axis) {
832 const double d = frame.feature[base + copy * dimensions + axis] - mean[axis];
833 square += d * d;
834 }
835 deviation += std::sqrt(square);
836 }
837 deviation /= count;
838 const bool unitVector = base == 4 || base == 8 || base == 14 || base == 16 || base == 28;
839 const double minimum = unitVector ? 0.1 : 0.001; // UE 0.1 cm, converted to metres.
840 const float inverse = static_cast<float>(deviation > minimum ? 1.0 / deviation : (unitVector ? 1.0 : 100.0));
841 for (int copy = 0; copy < copies; ++copy)
842 for (int axis = 0; axis < dimensions; ++axis) {
843 const int i = base + copy * dimensions + axis;
844 featureMean_[i] = static_cast<float>(mean[axis]);
845 featureInvStd_[i] = inverse;
846 }
847 }
848 for (auto& frame : frames_) normalizeFeature(frame.feature);
849}

858 if (!schema_ || feature.size() != static_cast<std::size_t>(schema_->dimension)) return 0.f;
859 float total = 0.f;

866 float squared = 0.f;
867 for (int component = 0; component < count; ++component) {
868 const float value = feature[schema_->offsets[i] + static_cast<std::size_t>(component)];
869 squared += value * value;
870 }
871 total += std::sqrt(squared);
872 }
873 return total;

983TransformTRS inverseRigid(TransformTRS value) {
984 value.qx = -value.qx;
985 value.qy = -value.qy;
986 value.qz = -value.qz;
987 const auto p = detail::schemaRotate({-value.px, -value.py, -value.pz}, value);
988 value.px = p[0];
989 value.py = p[1];
990 value.pz = p[2];
991 value.sx = value.sy = value.sz = 1.f;
992 return value;
993}
994TransformTRS rigidPower(TransformTRS delta, float count) {
995 if (count < 0.f) {
996 delta = inverseRigid(delta);
997 count = -count;
998 }
999 auto whole = static_cast<unsigned>(std::floor(count));
1000 const float fraction = count - whole;
1001 TransformTRS result, power = delta;
1002 while (whole) {
1003 if (whole & 1) result = detail::mulTRS(result, power);
1004 power = detail::mulTRS(power, power);
1005 whole >>= 1;
1006 }
1007 TransformTRS partial;
1008 partial.px = delta.px * fraction;
1009 partial.py = delta.py * fraction;
1010 partial.pz = delta.pz * fraction;
1011 slerpQuat(0.f, 0.f, 0.f, 1.f, delta.qx, delta.qy, delta.qz, delta.qw, fraction, partial.qx, partial.qy, partial.qz,
1012 partial.qw);
1013 return detail::mulTRS(result, partial);
1014}

1016struct MotionDatabase::SchemaSampler {
1017 const AnimSkeleton& skeleton;
1018 const AnimClip& clip;
1019 int rootBone;
1020 TransformTRS raw(int bone, float time) const {
1021 const auto local = clip.sampleBone(bone, time, skeleton.bindLocal(bone));
1022 const int parent = skeleton.getParent(bone);
1023 return parent < 0 ? local : detail::mulTRS(raw(parent, time), local);
1024 }
1025 TransformTRS root(float time) const {
1026 const float duration = clip.getDuration();
1027 if (duration <= 0.f) return raw(rootBone, 0.f);
1028 if (clip.getLoop()) {
1029 const float cycles = std::floor(time / duration);
1030 const auto start = raw(rootBone, 0.f), end = raw(rootBone, duration);
1031 const auto delta = detail::mulTRS(inverseRigid(start), end);
1032 const auto local = detail::mulTRS(inverseRigid(start), raw(rootBone, clip.wrapTime(time)));
1033 return detail::mulTRS(detail::mulTRS(start, rigidPower(delta, cycles)), local);
1034 }
1035 if (time >= 0.f && time <= duration) return raw(rootBone, time);
1036 const float step = std::min(1.f / 30.f, duration);
1037 const auto a = raw(rootBone, time < 0.f ? 0.f : duration - step);
1038 const auto b = raw(rootBone, time < 0.f ? step : duration);
1039 const auto delta = detail::mulTRS(inverseRigid(a), b);
1040 return detail::mulTRS(time < 0.f ? a : b, rigidPower(delta, (time < 0.f ? time : time - duration) / step));
1041 }
1042 TransformTRS bone(int index, float time) const {
1043 const float sample = clip.getLoop() ? clip.wrapTime(time) : clampf(time, 0.f, clip.getDuration());
1044 return detail::mulTRS(root(time), detail::mulTRS(inverseRigid(raw(rootBone, sample)), raw(index, sample)));
1045 }
1046};

1048void MotionDatabase::extractSchemaFeature(AnimClip* clip, float time, std::vector<float>& out, float& rootX,
1049 float& rootZ, float& rootYaw, float& velX, float& velZ) const {
1050 constexpr float dt = 1.f / 60.f;
1051 const SchemaSampler sampler{*skeleton_, *clip, rootBone_};
1052 const auto root = sampler.root(time), previousRoot = sampler.root(time - dt);
1053 rootX = root.px;
1054 rootZ = root.pz;
1055 rootYaw =
1056 std::atan2(2.f * (root.qw * root.qy + root.qx * root.qz), 1.f - 2.f * (root.qx * root.qx + root.qy * root.qy));
1057 velX = (root.px - previousRoot.px) / dt;
1058 velZ = (root.pz - previousRoot.pz) / dt;
1059 out.assign(schema_->dimension, 0.f);
1060 for (std::size_t i = 0; i < schema_->layout.channels.size(); ++i) {
1061 const auto& c = schema_->layout.channels[i];
1062 std::array<float, 3> value{};
1063 if (c.kind == MotionFeatureKind::Curve) {
1064 auto sampled =
1065 schema_->curves->sample(schema_->curveSources.at(clip), c.curve, time + c.sampleTime, clip->getLoop());
1066 if (!sampled.ok()) throw std::runtime_error("configured motion curve cannot be sampled");
1067 value[0] = sampled.value().value_or(0.f);
1068 } else if (c.source == MotionFeatureSource::Trajectory) {
1069 const auto sample = sampler.root(time + c.sampleTime);
1070 if (c.kind == MotionFeatureKind::Position)
1071 value = detail::schemaDifference(sample, root, root);
1072 else if (c.kind == MotionFeatureKind::Heading)
1073 value = detail::schemaHeading(sample, root, c.headingAxis);
1074 else {
1075 value = detail::schemaDifference(sample, sampler.root(time + c.sampleTime - dt), root);
1076 for (float& v : value) v /= dt;
1077 }
1078 } else {
1079 const auto bone = sampler.bone(c.bone, time), origin = sampler.bone(c.origin, time);
1080 if (c.kind == MotionFeatureKind::Position)
1081 value = detail::schemaDifference(bone, origin, root);
1082 else if (c.kind == MotionFeatureKind::Heading)
1083 value = detail::schemaHeading(bone, root, c.headingAxis);
1084 else if (c.characterSpaceVelocity) {
1085 value = detail::schemaDifference(bone, origin, root);
1086 const auto past = detail::schemaDifference(sampler.bone(c.bone, time - dt),
1087 sampler.bone(c.origin, time - dt), previousRoot);
1088 for (int axis = 0; axis < 3; ++axis) value[axis] = (value[axis] - past[axis]) / dt;
1089 } else {
1090 value = detail::schemaDifference(bone, sampler.bone(c.bone, time - dt), root);
1091 for (float& v : value) v /= dt;
1092 }
1093 }
1094 detail::schemaEncode(std::span(out).subspan(schema_->offsets[i]), c, value,
1095 schema_->layout.normalizationLengthScale);
1096 }
