21TransformTRS sanitize(TransformTRS delta) {
22 if (!std::isfinite(delta.px)) delta.px = 0.f;
23 if (!std::isfinite(delta.py)) delta.py = 0.f;
24 if (!std::isfinite(delta.pz)) delta.pz = 0.f;
25 if (!std::isfinite(delta.qx) || !std::isfinite(delta.qy) || !std::isfinite(delta.qz) ||
26 !std::isfinite(delta.qw)) {
27 delta.qx = delta.qy = delta.qz = 0.f;
28 delta.qw = 1.f;
29 } else {
30 delta.normalizeRotation();
31 }
32 return delta;
33}

35TransformTRS rotatePlanarByYaw(TransformTRS delta, float yaw) {
36 const float c = std::cos(yaw);
37 const float s = std::sin(yaw);
38 const float x = delta.px;
39 const float z = delta.pz;
40 // Ry(yaw) * (x,0,z) with forward (sin(yaw), 0, cos(yaw)) for +Z local.
41 delta.px = x * c + z * s;
42 delta.pz = -x * s + z * c;
43 return delta;
44}

48eve::Result<void> validateRootMotionPolicy(const RootMotionPolicy& policy) {
49 if (!std::isfinite(policy.characterYaw))
50 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
51 "root motion policy characterYaw must be finite"));
52 if (!isKnownApplySpace(policy.applySpace))
53 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
54 "root motion policy applySpace is unknown"));
55 if (!isKnownLockMask(policy.lockAxes))
56 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
57 "root motion policy lockAxes has unknown bits"));
58 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
59}

61TransformTRS applyRootMotionPolicy(const TransformTRS& rawDelta, const RootMotionPolicy& policy) {
62 TransformTRS delta = sanitize(rawDelta);
63 if (policy.lockRotation) {
64 delta.qx = delta.qy = delta.qz = 0.f;
65 delta.qw = 1.f;
66 }
67 if (policy.applySpace == RootMotionApplySpace::CharacterFacing && std::isfinite(policy.characterYaw))
68 delta = rotatePlanarByYaw(delta, policy.characterYaw);
69 if (hasLock(policy.lockAxes, RootMotionLockAxes::X)) delta.px = 0.f;
70 if (hasLock(policy.lockAxes, RootMotionLockAxes::Y)) delta.py = 0.f;
71 if (hasLock(policy.lockAxes, RootMotionLockAxes::Z)) delta.pz = 0.f;
72 return delta;
73}

75eve::Result<void> bakeRootMotionIntoPose(AnimPose& pose, int boneIndex, const TransformTRS& referenceRoot,
76 const RootMotionPolicy& policy) {
77 if (boneIndex < 0 || boneIndex >= pose.getBoneCount())
78 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
79 "bake root motion bone index out of range"));
80 if (!policy.bakeTranslationIntoPose && !policy.bakeRotationIntoPose)
81 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
82
83 TransformTRS& root = pose.local(boneIndex);
84 bool changed = false;
85 if (policy.bakeTranslationIntoPose) {
86 // Locked pose-local axes stay animated (e.g. vertical bob); unlocked axes
87 // plant on the stable reference so the capsule owns locomotion.
88 if (!hasLock(policy.lockAxes, RootMotionLockAxes::X) && root.px != referenceRoot.px) {
89 root.px = referenceRoot.px;
90 changed = true;
91 }
92 if (!hasLock(policy.lockAxes, RootMotionLockAxes::Y) && root.py != referenceRoot.py) {
93 root.py = referenceRoot.py;
94 changed = true;
95 }
96 if (!hasLock(policy.lockAxes, RootMotionLockAxes::Z) && root.pz != referenceRoot.pz) {
97 root.pz = referenceRoot.pz;
98 changed = true;
99 }
100 }
101 if (policy.bakeRotationIntoPose && !policy.lockRotation) {
102 root.qx = referenceRoot.qx;
103 root.qy = referenceRoot.qy;
104 root.qz = referenceRoot.qz;
105 root.qw = referenceRoot.qw;
106 root.normalizeRotation();
107 changed = true;
108 }
109 return eve::Result<void>::success(
110 eve::Status::success(changed ? eve::StatusCode::Applied : eve::StatusCode::NoOp));
111}

113eve::Result<RootMotionLockAxes> parseRootMotionLockAxes(std::string_view name) {
114 if (name == "none" || name.empty())
115 return eve::Result<RootMotionLockAxes>::success(RootMotionLockAxes::None);
116 if (name == "x") return eve::Result<RootMotionLockAxes>::success(RootMotionLockAxes::X);
117 if (name == "y" || name == "vertical")
118 return eve::Result<RootMotionLockAxes>::success(RootMotionLockAxes::VerticalY);
119 if (name == "z") return eve::Result<RootMotionLockAxes>::success(RootMotionLockAxes::Z);
120 if (name == "xz" || name == "horizontal")
121 return eve::Result<RootMotionLockAxes>::success(RootMotionLockAxes::HorizontalXZ);
122 if (name == "xy")
123 return eve::Result<RootMotionLockAxes>::success(RootMotionLockAxes::X | RootMotionLockAxes::Y);
124 if (name == "yz")
125 return eve::Result<RootMotionLockAxes>::success(RootMotionLockAxes::Y | RootMotionLockAxes::Z);
126 if (name == "xyz" || name == "all")
127 return eve::Result<RootMotionLockAxes>::success(RootMotionLockAxes::All);
128 return eve::Result<RootMotionLockAxes>::failure(
129 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
130 "unknown root motion lockAxes: " + std::string(name)));
131}
