53Result<TerrainSpawnPlan> TerrainBiomePreset::compile(bool biomeScope) const {
54 TerrainSpawnPlan result;
55 for (const auto& entry : entries_) {
56 if (!(biomeScope ? entry.activeInBiome : entry.activeInStamper)) continue;
57 for (const auto& variant : entry.plan.rules()) {
58 auto added = std::visit([&](const auto& rule) -> Result<int> {
59 using Rule = std::decay_t<decltype(rule)>;
60 const auto id = namespacedRuleId(entry.entryId, rule.ruleId);
61 auto insertion = [&]() -> Result<int> {
62 if constexpr (std::is_same_v<Rule, TerrainSplatSpawnRule>)
63 return result.addSplat(id, rule.paint, rule.targetLayer, rule.operation);
64 else if constexpr (std::is_same_v<Rule, TerrainDetailSpawnRule>)
65 return result.addDetail(id, rule.fitness, rule.settings, rule.operation, rule.mode, rule.seed);
66 else if constexpr (std::is_same_v<Rule, TerrainTreeSpawnRule>)
67 return result.addTrees(id, rule.fitness, rule.settings, rule.operation, rule.mode);
68 else if constexpr (std::is_same_v<Rule, TerrainObjectSpawnRule>)
69 return result.addObjects(id, rule.fitness, rule.settings, rule.operation, rule.mode);
70 else if constexpr (std::is_same_v<Rule, TerrainProbeSpawnRule>)
71 return result.addProbes(id, rule.fitness, rule.settings, rule.operation, rule.mode);
72 else
73 return result.addModifierStamp(id, rule.stamp, rule.operation, rule.localMask,
74 rule.globalMask);
75 }();
76 if (insertion.ok() && !rule.enabled) {
77 auto disabled = result.setEnabled(id, false);
78 if (!disabled.ok()) return Result<int>::failure(disabled.status());
79 }
80 return insertion;
81 }, variant);
82 if (!added.ok()) return Result<TerrainSpawnPlan>::failure(added.status());
83 }
84 }
85 if (result.rules().empty()) return invalidPreset<TerrainSpawnPlan>("terrain.biomePreset: no active spawners");
86 return Result<TerrainSpawnPlan>::success(std::move(result));
87}
