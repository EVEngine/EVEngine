31 if (field->writeInt(changes_[i].x, changes_[i].y, changes_[i].after) == FieldWriteStatus::Rejected) {

157Result<void> EditConstraintPipeline::evaluateChecked(EditorContext &context, IEditCommand &command) {
158 diagnostics_.clear();
159 structuredDiagnostics_.clear();
160 rejected_ = false;
161 auto record = [&](DiagnosticSeverity severity, const char *rule, std::string message) {
162 if (message.empty() && severity != DiagnosticSeverity::Error) return;
163 if (message.empty()) message = "Editor command was rejected by a constraint";
164 structuredDiagnostics_.push_back(editing::ruleDiagnostic(severity == DiagnosticSeverity::Error
165 ? eve::DiagnosticCode::PreconditionViolation
166 : eve::DiagnosticCode::None,
167 RuleId(rule), severity, message));
168 diagnostics_.push_back(std::move(message));
169 };
170 for (auto *constraint : constraints_) {
171 ConstraintResult result = constraint->evaluate(context, command);
172 if (result.disposition == ConstraintDisposition::Reject) {
173 rejected_ = true;
174 record(DiagnosticSeverity::Error, "editor.command.constraint-rejected", std::move(result.message));
175 return Result<void>::failure(eve::Status(EditorStatus::Rejected, structuredDiagnostics_));
176 }
177 if (result.disposition == ConstraintDisposition::Warning)
178 record(DiagnosticSeverity::Warning, "editor.command.constraint-warning", std::move(result.message));
179 else if (!result.message.empty())
180 record(DiagnosticSeverity::Info, "editor.command.constraint-info", std::move(result.message));
181 }
182 return eve::editing::applied<void>(structuredDiagnostics_);
183}
184const std::string &EditConstraintPipeline::diagnostic(int index) const {
185 if (index < 0 || index >= diagnosticCount()) throw Exception("EditConstraintPipeline::diagnostic: bad index");
186 return diagnostics_[static_cast<size_t>(index)];
187}
