42float combine(float a, float b, uint64_t policyA, uint64_t policyB, int shift, bool friction) {
43 switch (std::max((policyA >> shift) & 7, (policyB >> shift) & 7)) {
44 case 1: return .5f * (a + b);
45 case 2: return std::min(a, b);
46 case 3: return a * b;
47 case 4: return std::max(a, b);
48 default: return friction ? std::sqrt(a * b) : std::max(a, b);
49 }
50}
51void configure(const JPH::Body& a, const JPH::Body& b, const JPH::ContactManifold& m, JPH::ContactSettings& settings) {
52 auto* sa = child(a, m.mSubShapeID1);
53 auto* sb = child(b, m.mSubShapeID2);
54 if (!sa || !sb) return;
55 const auto ma = contactMaterial(a, m.mSubShapeID1, *sa);
56 const auto mb = contactMaterial(b, m.mSubShapeID2, *sb);
57 auto velocity = [](const JPH::Body& body, const ShapeState& shape, const SurfaceMaterial3D& material) {
58 const auto& g = shape.geometry;
59 return body.GetRotation() *
60 (JPH::Quat(g.localQx, g.localQy, g.localQz, g.localQw).Normalized() *
61 JPH::Vec3(material.tangentVelocity.x, material.tangentVelocity.y, material.tangentVelocity.z));
62 };
63 settings.mRelativeLinearSurfaceVelocity = velocity(b, *sb, mb) - velocity(a, *sa, ma);
64 const auto point = m.GetWorldSpaceContactPointOn1(0);
65 const auto normal = m.mWorldSpaceNormal;
66 settings.mIsSensor = sa->sensor || sb->sensor || !sa->owner || !sb->owner ||
67 !WorldBackend::isContactAllowed(*sa->owner, *sb->owner, point.GetX(), point.GetY(),
68 point.GetZ(), normal.GetX(), normal.GetY(), normal.GetZ());
69 settings.mCombinedFriction = combine(ma.friction, mb.friction, ma.userMaterialId, mb.userMaterialId, 32, true);
70 settings.mCombinedRestitution =
71 combine(ma.restitution, mb.restitution, ma.userMaterialId, mb.userMaterialId, 35, false);
72 const auto radius = [](const ShapeState& shape) {
73 if (shape.geometry.kind == ShapeKind3D::Sphere) return shape.geometry.a;
74 if (shape.geometry.kind == ShapeKind3D::Capsule) return shape.geometry.b;
75 if (shape.geometry.kind == ShapeKind3D::Box || shape.geometry.kind == ShapeKind3D::ConvexHull)
76 return .25f * shape.native->GetInnerRadius();
77 return 0.f;
78 };
79 settings.mCombinedRollingResistance =
80 std::max(ma.rollingResistance, mb.rollingResistance) * std::max(radius(*sa), radius(*sb));
81}

96void Listener::OnStep(const JPH::PhysicsStepListenerContext&) {
97 ++substep;
98 // Jolt owns all body/constraint locks here. Never call the locking BodyInterface.
99 for (const auto& weak : state->joints)
100 if (auto joint = weak.lock(); joint && joint->alive && joint->native &&
101 (joint->kind == JointKind3D::Motor || joint->kind == JointKind3D::Spherical)) {
102 auto& constraint = joint->velocity ? *joint->velocity : static_cast<JPH::SixDOFConstraint&>(*joint->native);
103 const auto frame =
104 constraint.GetBody1()->GetCenterOfMassTransform() * constraint.GetConstraintToBody1Matrix();
105 constraint.SetTargetAngularVelocityCS(frame.Multiply3x3Transposed(joint->worldAngularVelocity));
106 if (joint->kind == JointKind3D::Motor)
107 constraint.SetTargetVelocityCS(frame.Multiply3x3Transposed(joint->worldLinearVelocity));
108 }

110JPH::ValidateResult Listener::OnContactValidate(const JPH::Body& a, const JPH::Body& b, JPH::RVec3Arg,
111 const JPH::CollideShapeResult& result) {
112 if (state->suppressedPairs.contains(pairKey(a.GetID(), b.GetID())))
113 return JPH::ValidateResult::RejectAllContactsForThisBodyPair;
114 const auto* sa = child(a, result.mSubShapeID1);
115 const auto* sb = child(b, result.mSubShapeID2);
116 if (!sa || !sb || !sa->owner || !sb->owner || !WorldBackend::isPairAllowed(*sa->owner, *sb->owner))
117 return JPH::ValidateResult::RejectContact;
118 const auto& fa = sa->filter;
119 const auto& fb = sb->filter;
120 bool accept = fa.groupIndex && fa.groupIndex == fb.groupIndex
121 ? fa.groupIndex > 0
122 : (fa.maskBits & fb.categoryBits) && (fb.maskBits & fa.categoryBits);
123 return accept ? JPH::ValidateResult::AcceptContact : JPH::ValidateResult::RejectContact;
124}
125void Listener::OnContactAdded(const JPH::Body& a, const JPH::Body& b, const JPH::ContactManifold& m,
126 JPH::ContactSettings& settings) {
127 configure(a, b, m, settings);
128 auto* sa = child(a, m.mSubShapeID1);
129 auto* sb = child(b, m.mSubShapeID2);
130 if (!sa || !sb || !sa->owner || !sb->owner) return;
131 auto ba = sa->body.lock();
132 auto bb = sb->body.lock();
133 if (!ba || !bb || !ba->owner || !bb->owner || (settings.mIsSensor && !sa->sensor && !sb->sensor)) return;
134 ContactPair pair{ba->owner->getId(), sa->owner->getId(), sa->owner->getTag(), bb->owner->getId(),
135 sb->owner->getId(), sb->owner->getTag(), sa->sensor, sb->sensor};
136 contacts.insert_or_assign(JPH::SubShapeIDPair(a.GetID(), m.mSubShapeID1, b.GetID(), m.mSubShapeID2), pair);
137 begins.push_back(pair);
138}
139void Listener::OnContactPersisted(const JPH::Body& a, const JPH::Body& b, const JPH::ContactManifold& m,
140 JPH::ContactSettings& settings) {
141 const JPH::SubShapeIDPair key(a.GetID(), m.mSubShapeID1, b.GetID(), m.mSubShapeID2);
142 configure(a, b, m, settings);
143 const auto* sa = child(a, m.mSubShapeID1);
144 const auto* sb = child(b, m.mSubShapeID2);
145 if (settings.mIsSensor && sa && sb && !sa->sensor && !sb->sensor) {
146 OnContactRemoved(key);
147 } else if (!contacts.contains(key)) {
148 OnContactAdded(a, b, m, settings);
149 }
150}
151void Listener::OnContactRemoved(const JPH::SubShapeIDPair& key) {
152 auto i = contacts.find(key);
153 if (i != contacts.end()) {
154 ends.push_back(i->second);
155 contacts.erase(i);
156 }
