115eve::Result<std::vector<eve::WorldSphereContact>> World3D::sweepSphereOwned(double x, double y, double z, double endX,
116 double endY, double endZ,
117 eve::WorldSphereSweepSettings settings) {
118 using Output = eve::Result<std::vector<eve::WorldSphereContact>>;
119 const double values[] = {x, y, z, endX, endY, endZ, endX - x, endY - y, endZ - z};
120 for (double value : values)
121 if (!std::isfinite(value) || std::abs(value) > FLT_MAX)
122 return Output::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
123 "Sweep coordinates exceed the finite physics float domain",
124 "motion"));
125 if (!isValid() || !std::isfinite(settings.radius) || settings.radius <= 0 || settings.ignoredBodyId < -1)
126 return Output::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
127 "Sphere sweep requires a live world and valid radius/filter",
128 "worldSweep"));
129 world3d_detail::ScopedQueryState3D scoped(*this,
130 {settings.categoryBits, settings.maskBits, settings.ignoredBodyId, -1});
131 auto cast = rigidBackend_->castSphereAll(static_cast<float>(x), static_cast<float>(y), static_cast<float>(z),
132 settings.radius, static_cast<float>(endX - x),
133 static_cast<float>(endY - y), static_cast<float>(endZ - z), 257);
134 if (!cast) return Output::failure(cast.status());
135 if (cast.value() > 256)
136 return Output::failure(
137 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "Sphere sweep contact budget exceeded", "contacts"));
138 std::vector<eve::WorldSphereContact> output;
139 for (int i = 0; i < cast.value(); ++i) {
140 // Sweeps report new crossings only. Backends may return an initial overlap with
141 // fraction zero and no usable normal; it is not a new impact after a bounce.
142 if (getShapeCastResultFraction(i) == 0.0f) continue;
143 const int bodyId = getShapeCastResultBodyId(i);
144 auto* body = findBodyById(bodyId);
145 if (!body)
146 return Output::failure(
147 eve::Diagnostic::error(eve::DiagnosticCode::NotFound, "Sweep returned a stale body", "body"));
148 eve::WorldSphereContact contact;
149 contact.identity = {runtimeHandle().packed(), body->runtimeHandle().packed()};
150 contact.bodyId = bodyId;
151 contact.shapeId = getShapeCastResultShapeId(i);
152 contact.fraction = getShapeCastResultFraction(i);
153 contact.x = x + (endX - x) * contact.fraction;
154 contact.y = y + (endY - y) * contact.fraction;
155 contact.z = z + (endZ - z) * contact.fraction;
156 contact.nx = getShapeCastResultNormalX(i);
157 contact.ny = getShapeCastResultNormalY(i);
158 contact.nz = getShapeCastResultNormalZ(i);
159 output.push_back(contact);
160 }
161 return Output::success(std::move(output));
162}
