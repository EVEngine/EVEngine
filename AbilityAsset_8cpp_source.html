204 if (!decoded) return Result<std::array<std::int32_t, 3>>::failure(decoded.status());

209 component < static_cast<double>(std::numeric_limits<std::int32_t>::min()) ||

217Result<sensing::TargetingSpec> decodeTargetingSpec(const Value& value) {
218 sensing::TargetingSpec result;
219 auto space = text(value, "space", "action.targetingSpec.space");
220 if (!space) return Result<sensing::TargetingSpec>::failure(space.status());
221 auto domain = integer(value, "domain", "action.targetingSpec.domain");
222 if (!domain) return Result<sensing::TargetingSpec>::failure(domain.status());
223 auto minCount = integer(value, "minCount", "action.targetingSpec.minCount");
224 if (!minCount) return Result<sensing::TargetingSpec>::failure(minCount.status());
225 auto maxCount = integer(value, "maxCount", "action.targetingSpec.maxCount");
226 if (!maxCount) return Result<sensing::TargetingSpec>::failure(maxCount.status());
227 auto minRange = number(value, "minRange", "action.targetingSpec.minRange");
228 if (!minRange) return Result<sensing::TargetingSpec>::failure(minRange.status());
229 auto required = decodeStrings(value, "requiredTags", "action.targetingSpec.requiredTags");
230 if (!required) return Result<sensing::TargetingSpec>::failure(required.status());
231 auto excluded = decodeStrings(value, "excludedTags", "action.targetingSpec.excludedTags");
232 if (!excluded) return Result<sensing::TargetingSpec>::failure(excluded.status());
233 auto lineOfSight = boolean(value, "lineOfSight", "action.targetingSpec.lineOfSight");
234 if (!lineOfSight) return Result<sensing::TargetingSpec>::failure(lineOfSight.status());
235 const std::pair<std::string_view, sensing::CoordinateSpace> spaces[] = {
236 {"world2d", sensing::CoordinateSpace::World2D}, {"world3d", sensing::CoordinateSpace::World3D},
237 {"grid2d", sensing::CoordinateSpace::Grid2D}, {"grid3d", sensing::CoordinateSpace::Grid3D}};
238 bool spaceFound = false;
239 for (const auto& [name, parsed] : spaces)
240 if (space.value() == name) { result.space = parsed; spaceFound = true; break; }
241 if (!spaceFound || domain.value() < 0 ||
242 domain.value() > static_cast<std::int64_t>(sensing::TargetDomain::Neutral) ||
243 minCount.value() < 0 || maxCount.value() < 0 ||
244 minCount.value() > std::numeric_limits<std::uint32_t>::max() ||
245 maxCount.value() > std::numeric_limits<std::uint32_t>::max())
246 return invalid<sensing::TargetingSpec>("targeting specification enum or count is invalid",
247 "action.targetingSpec");
248 result.domain = static_cast<sensing::TargetDomain>(domain.value());
249 result.minCount = static_cast<std::uint32_t>(minCount.value());
250 result.maxCount = static_cast<std::uint32_t>(maxCount.value());
251 result.minRange = static_cast<float>(minRange.value());
252 const auto* maxRange = field(value, "maxRange");
253 if (!maxRange) return invalid<sensing::TargetingSpec>("maximum range is missing", "action.targetingSpec.maxRange");
254 if (maxRange->isNull()) result.maxRange = std::numeric_limits<float>::infinity();
255 else if (maxRange->isDouble() || maxRange->isInt64())
256 result.maxRange = static_cast<float>(maxRange->isDouble() ? maxRange->asDouble() : maxRange->asInt());
257 else
258 return invalid<sensing::TargetingSpec>("maximum range must be numeric or null",
259 "action.targetingSpec.maxRange");
260 result.requiredTags = std::move(required).takeValue();
261 result.excludedTags = std::move(excluded).takeValue();
262 result.lineOfSight = lineOfSight.value() ? sensing::LineOfSightMode::Required
263 : sensing::LineOfSightMode::NotRequired;
264 if (const auto* zone = field(value, "zone")) {
265 if (!zone->isString()) return invalid<sensing::TargetingSpec>("zone must be text", "action.targetingSpec.zone");
266 auto id = LogicalId::parse(zone->asString());
267 if (!id) return invalid<sensing::TargetingSpec>("zone id is invalid", "action.targetingSpec.zone");
268 result.zone = sensing::ZoneRef::fromLogicalId(*id);
269 }
270 if (const auto* area = field(value, "worldArea")) {
271 auto shape = integer(*area, "shape", "action.targetingSpec.worldArea.shape");
272 if (!shape) return Result<sensing::TargetingSpec>::failure(shape.status());
273 auto first = vector3(*area, "first", "action.targetingSpec.worldArea.first");
274 if (!first) return Result<sensing::TargetingSpec>::failure(first.status());
275 auto second = vector3(*area, "second", "action.targetingSpec.worldArea.second");
276 if (!second) return Result<sensing::TargetingSpec>::failure(second.status());
277 auto radius = number(*area, "radius", "action.targetingSpec.worldArea.radius");
278 if (!radius) return Result<sensing::TargetingSpec>::failure(radius.status());
279 if (shape.value() < 0 || shape.value() > 3)
280 return invalid<sensing::TargetingSpec>("world area is invalid", "action.targetingSpec.worldArea");
281 const auto makePoint = [&](const std::array<double, 3>& point) {
282 return result.space == sensing::CoordinateSpace::World2D
283 ? sensing::WorldPoint::world2D(static_cast<float>(point[0]), static_cast<float>(point[1]))
284 : sensing::WorldPoint::world3D(static_cast<float>(point[0]), static_cast<float>(point[1]),
285 static_cast<float>(point[2]));
286 };
287 auto a = makePoint(first.value());
288 if (!a) return Result<sensing::TargetingSpec>::failure(a.status());
289 auto b = makePoint(second.value());
290 if (!b) return Result<sensing::TargetingSpec>::failure(b.status());
291 Result<sensing::WorldArea> decoded =
292 shape.value() == 0
293 ? sensing::WorldArea::circle2D(a.value(), static_cast<float>(radius.value()))
294 : shape.value() == 1
295 ? sensing::WorldArea::box2D(a.value(), b.value())
296 : shape.value() == 2
297 ? sensing::WorldArea::sphere3D(a.value(), static_cast<float>(radius.value()))
298 : sensing::WorldArea::box3D(a.value(), b.value());
299 if (!decoded) return Result<sensing::TargetingSpec>::failure(decoded.status());
300 result.worldArea = std::move(decoded).takeValue();
301 }
302 if (const auto* area = field(value, "gridArea")) {
303 auto shape = integer(*area, "shape", "action.targetingSpec.gridArea.shape");
304 if (!shape) return Result<sensing::TargetingSpec>::failure(shape.status());
305 auto minimum = gridVector3(*area, "minimum", "action.targetingSpec.gridArea.minimum");
306 if (!minimum) return Result<sensing::TargetingSpec>::failure(minimum.status());
307 auto maximum = gridVector3(*area, "maximum", "action.targetingSpec.gridArea.maximum");
308 if (!maximum) return Result<sensing::TargetingSpec>::failure(maximum.status());
309 if (shape.value() < 0 || shape.value() > 1)
310 return invalid<sensing::TargetingSpec>("grid area is invalid", "action.targetingSpec.gridArea");
311 const auto makePoint = [&](const std::array<std::int32_t, 3>& point) {
312 return result.space == sensing::CoordinateSpace::Grid2D
313 ? sensing::GridPoint::grid2D(point[0], point[1])
314 : sensing::GridPoint::grid3D(point[0], point[1], point[2]);
315 };
316 const auto a = makePoint(minimum.value());
317 const auto b = makePoint(maximum.value());
318 auto decoded = shape.value() == 0 ? sensing::GridArea::box2D(a, b) : sensing::GridArea::box3D(a, b);
319 if (!decoded) return Result<sensing::TargetingSpec>::failure(decoded.status());
320 result.gridArea = std::move(decoded).takeValue();
321 }
322 auto valid = result.validate();
323 if (!valid) return Result<sensing::TargetingSpec>::failure(valid.status());
324 return Result<sensing::TargetingSpec>::success(std::move(result));
325}

341Result<Value> encodeAbilityAsset(const AbilityDefinition& definition) {
342 auto valid = definition.validate();
343 if (!valid) return Result<Value>::failure(valid.status());
344 Value::Array costs;
345 if (definition.action.cost)
346 for (const auto& item : definition.action.cost->items())
347 costs.emplace_back(Value::Object{{"resource", item.resource.value()}, {"amount", item.amount.value()}});
348 Value::Array triggers;
349 for (const auto& trigger : definition.triggers)
350 triggers.emplace_back(Value::Object{
351 {"gameplayTag", trigger.gameplayTag},
352 {"match", trigger.match == tags::GameplayTagMatch::Exact ? "exact" : "include_descendants"}});
353 Value timeline;
354 if (definition.action.timeline) {
355 auto encoded = definition.action.timeline->toValue();
356 if (!encoded) return Result<Value>::failure(encoded.status());
357 timeline = std::move(encoded).takeValue();
358 }
359 Value::Object action{
360 {"id", definition.action.id.format()},
361 {"timing", Value::Object{{"windupNs", definition.action.timing.windup.nanoseconds()},
362 {"activeNs", definition.action.timing.active.nanoseconds()},
363 {"recoverNs", definition.action.timing.recover.nanoseconds()}}},
364 {"condition", encodeAbilityCondition(definition.action.condition)},
365 {"targetingMode", std::string(targetingModeName(definition.action.targetingMode))},
366 {"targetingSpec",
367 definition.action.targetingSpec ? encodeTargetingSpec(*definition.action.targetingSpec) : Value()},
368 {"cost", definition.action.cost ? Value(std::move(costs)) : Value()},
369 {"effectIds", strings(definition.action.effectIds)},
370 {"activeExecutionRequired", definition.action.activeExecutionRequired},
371 {"metadata", definition.action.metadata},
372 {"timeline", std::move(timeline)}};
373 Value::Object root{{"schema", std::string(kAbilityAssetSchemaId)},
374 {"schemaVersion", static_cast<std::int64_t>(kAbilityAssetSchemaVersion)},
375 {"id", definition.id.format()},
376 {"action", std::move(action)},
377 {"cooldownNs", definition.cooldown.nanoseconds()},
378 {"instancing", std::string(instancingName(definition.instancing))},
379 {"activationGroup", std::string(groupName(definition.activationGroup))},
380 {"triggers", std::move(triggers)}};
381 return Result<Value>::success(Value(std::move(root)));
382}

384Result<AbilityDefinition> decodeAbilityAsset(const Value& value) {
385 auto rootFields = rejectUnknown(value, {"schema", "schemaVersion", "id", "action", "cooldownNs", "cooldownMs",
386 "instancing", "activationGroup", "triggers"}, "ability");
387 if (!rootFields) return Result<AbilityDefinition>::failure(rootFields.status());
388 auto schema = text(value, "schema", "schema");
389 if (!schema) return Result<AbilityDefinition>::failure(schema.status());
390 auto versionRaw = integer(value, "schemaVersion", "schemaVersion");
391 if (!versionRaw) return Result<AbilityDefinition>::failure(versionRaw.status());
392 if (schema.value() != kAbilityAssetSchemaId)
393 return invalid<AbilityDefinition>("unexpected ability asset schema", "schema");
394 if (versionRaw.value() < 1 || versionRaw.value() > static_cast<std::int64_t>(kAbilityAssetSchemaVersion))
395 return invalid<AbilityDefinition>("unsupported ability asset schema version", "schemaVersion");
396 const auto version = static_cast<std::uint64_t>(versionRaw.value());
397 auto idText = text(value, "id", "id");
398 if (!idText) return Result<AbilityDefinition>::failure(idText.status());
399 auto cooldown = durationField(value, "cooldownNs", "cooldownMs", version, "cooldown");
400 if (!cooldown) return Result<AbilityDefinition>::failure(cooldown.status());
401 auto instancingText = text(value, "instancing", "instancing");
402 if (!instancingText) return Result<AbilityDefinition>::failure(instancingText.status());
403 auto groupText = text(value, "activationGroup", "activationGroup");
404 if (!groupText) return Result<AbilityDefinition>::failure(groupText.status());
405 const auto* actionValue = field(value, "action");
406 const auto* triggerValue = field(value, "triggers");
407 if (!actionValue || !triggerValue || !triggerValue->isArray())
408 return invalid<AbilityDefinition>("ability asset is incomplete", "ability");
409 auto id = LogicalId::parse(idText.value());
410 auto instancing = parseInstancing(instancingText.value());
411 if (!instancing) return Result<AbilityDefinition>::failure(instancing.status());
412 auto group = parseGroup(groupText.value());
413 if (!group) return Result<AbilityDefinition>::failure(group.status());
414 if (!id || !actionValue->isObject())
415 return invalid<AbilityDefinition>("ability identity or policy is invalid", "ability");
416
417 AbilityDefinition result;
418 result.id = *id;
419 result.cooldown = cooldown.value();
420 result.instancing = instancing.value();
421 result.activationGroup = group.value();
422 auto actionIdText = text(*actionValue, "id", "action.id");
423 if (!actionIdText) return Result<AbilityDefinition>::failure(actionIdText.status());
424 const auto* timing = field(*actionValue, "timing");
425 auto modeText = text(*actionValue, "targetingMode", "action.targetingMode");
426 if (!modeText) return Result<AbilityDefinition>::failure(modeText.status());
427 auto activeRequired = boolean(*actionValue, "activeExecutionRequired", "action.activeExecutionRequired");
428 if (!activeRequired) return Result<AbilityDefinition>::failure(activeRequired.status());
429 auto effects = decodeStrings(*actionValue, "effectIds", "action.effectIds");
430 if (!effects) return Result<AbilityDefinition>::failure(effects.status());
431 const auto* metadata = field(*actionValue, "metadata");
432 const auto* condition = field(*actionValue, "condition");
433 if (!timing || !timing->isObject() || !metadata || !metadata->isObject() || !condition)
434 return invalid<AbilityDefinition>("action definition is incomplete", "action");
435 auto actionId = LogicalId::parse(actionIdText.value());
436 auto windup = durationField(*timing, "windupNs", "windupMs", version, "action.timing.windup");
437 if (!windup) return Result<AbilityDefinition>::failure(windup.status());
438 auto active = durationField(*timing, "activeNs", "activeMs", version, "action.timing.active");
439 if (!active) return Result<AbilityDefinition>::failure(active.status());
440 auto recover = durationField(*timing, "recoverNs", "recoverMs", version, "action.timing.recover");
441 if (!recover) return Result<AbilityDefinition>::failure(recover.status());
442 auto mode = parseTargetingMode(modeText.value());
443 if (!mode) return Result<AbilityDefinition>::failure(mode.status());
444 auto decodedCondition = decodeAbilityCondition(*condition, "action.condition");
445 if (!decodedCondition) return Result<AbilityDefinition>::failure(decodedCondition.status());
446 if (!actionId) return invalid<AbilityDefinition>("action identity is invalid", "action.id");
447 result.action.id = *actionId;
448 result.action.timing = {windup.value(), active.value(), recover.value()};
449 result.action.condition = std::move(decodedCondition).takeValue();
450 result.action.targetingMode = mode.value();
451 result.action.activeExecutionRequired = activeRequired.value();
452 result.action.effectIds = std::move(effects).takeValue();
453 result.action.metadata = *metadata->getIf<Value::Object>();
454 if (const auto* spec = field(*actionValue, "targetingSpec"); spec && !spec->isNull()) {
455 auto decoded = decodeTargetingSpec(*spec);
456 if (!decoded) return Result<AbilityDefinition>::failure(decoded.status());
457 result.action.targetingSpec = std::move(decoded).takeValue();
458 }
459 if (const auto* cost = field(*actionValue, "cost"); cost && !cost->isNull()) {
460 if (!cost->isArray()) return invalid<AbilityDefinition>("action cost must be an array or null", "action.cost");
461 std::vector<resource::ResourceCost> items;
462 const auto& array = *cost->getIf<Value::Array>();
463 for (std::size_t index = 0; index < array.size(); ++index) {
464 auto resourceName = text(array[index], "resource", "action.cost[" + std::to_string(index) + "].resource");
465 if (!resourceName) return Result<AbilityDefinition>::failure(resourceName.status());
466 auto amount = integer(array[index], "amount", "action.cost[" + std::to_string(index) + "].amount");
467 if (!amount) return Result<AbilityDefinition>::failure(amount.status());
468 auto item = resource::ResourceCost::create(resourceName.value(), amount.value());
469 if (!item) return Result<AbilityDefinition>::failure(item.status());
470 items.push_back(std::move(item).takeValue());
471 }
472 auto costSpec = resource::CostSpec::create(std::move(items));
473 if (!costSpec) return Result<AbilityDefinition>::failure(costSpec.status());
474 result.action.cost = std::move(costSpec).takeValue();
475 }
476 if (const auto* timeline = field(*actionValue, "timeline"); timeline && !timeline->isNull()) {
477 auto decoded = ActionTimeline::fromValue(*timeline);
478 if (!decoded) return Result<AbilityDefinition>::failure(decoded.status());
479 result.action.timeline = std::move(decoded).takeValue();
480 }
481 for (std::size_t index = 0; index < triggerValue->getIf<Value::Array>()->size(); ++index) {
482 const auto& item = triggerValue->getIf<Value::Array>()->at(index);
483 auto tag = text(item, "gameplayTag", "triggers[" + std::to_string(index) + "].gameplayTag");
484 if (!tag) return Result<AbilityDefinition>::failure(tag.status());
485 auto match = text(item, "match", "triggers[" + std::to_string(index) + "].match");
486 if (!match) return Result<AbilityDefinition>::failure(match.status());
487 if (match.value() != "exact" && match.value() != "include_descendants")
488 return invalid<AbilityDefinition>("ability trigger is invalid", "triggers[" + std::to_string(index) + "]");
489 result.triggers.push_back({tag.value(), match.value() == "exact" ? tags::GameplayTagMatch::Exact
490 : tags::GameplayTagMatch::IncludeDescendants});
491 }
492 auto valid = result.validate();
493 if (!valid) return Result<AbilityDefinition>::failure(valid.status());
494 return Result<AbilityDefinition>::success(std::move(result));
495}
