21bool WorldBackend::do_sphereCast(float x1, float y1, float z1, float x2, float y2, float z2, float radius,
22 uint64_t maskBits, int ignoredBodyId, CameraSphereHit3D *out) const {

26 struct Collector {
27 CameraSphereHit3D *out = nullptr;
28 int ignoredBodyId = -1;
29 static float callback(b3ShapeId shapeId, b3Pos point, b3Vec3 normal, float fraction, uint64_t, int, int,
30 void *context) {
31 auto *self = static_cast<Collector *>(context);
32 Body3D *body = bodyFromShape(shapeId);
33 if (!body || body->getId() == self->ignoredBodyId) return -1.f;
34 if (!self->out->hit || fraction < self->out->fraction) {
35 self->out->hit = true;
36 self->out->bodyId = body->getId();
37 self->out->fraction = fraction;
38 self->out->x = static_cast<float>(point.x);
39 self->out->y = static_cast<float>(point.y);
40 self->out->z = static_cast<float>(point.z);
41 self->out->nx = normal.x;
42 self->out->ny = normal.y;
43 self->out->nz = normal.z;
44 }
45 return fraction;
46 }
47 } collector{out, ignoredBodyId};
48
49 const b3Vec3 point{0.f, 0.f, 0.f};
50 const b3ShapeProxy proxy{&point, 1, radius};
51 b3QueryFilter filter = b3DefaultQueryFilter();
52 filter.maskBits = maskBits;
53 b3World_CastShape(worldId_, b3Pos{x1, y1, z1}, &proxy, b3Vec3{x2 - x1, y2 - y1, z2 - z1}, filter,
54 &Collector::callback, &collector);

178 struct RayCollector {

180 int capacity = 0;

182 static bool better(const RayResult &a, const RayResult &b) {
183 return a.fraction < b.fraction || (a.fraction == b.fraction && a.shapeId < b.shapeId);
184 }
185
186 float clipFraction() const {
187 if (static_cast<int>(world->owner_->rayResults_.size()) < capacity) return 1.f;
188 const auto worst =
189 std::max_element(world->owner_->rayResults_.begin(), world->owner_->rayResults_.end(), better);
190 return worst->fraction;
191 }

193 static float callback(b3ShapeId shapeId, b3Pos point, b3Vec3 normal, float fraction, uint64_t materialId,
194 int triangleIndex, int, void *context) {

196 if (self->world->do_shouldIgnoreQueryShape(shapeId)) return -1.f;
197 Body3D *body = bodyFromShape(shapeId);
198 Shape3D *shape = shapeFromShape(shapeId);

200

202 shape->getId(),
203 shape->getTag(),
204 static_cast<int>(static_cast<uint32_t>(materialId)),
205 sourceTriangleIndex(shapeId, triangleIndex),
206 static_cast<float>(point.x),
207 static_cast<float>(point.y),
208 static_cast<float>(point.z),
209 normal.x,
210 normal.y,
211 normal.z,
212 fraction};
213 auto &results = self->world->owner_->rayResults_;
214 if (static_cast<int>(results.size()) < self->capacity) {
215 results.push_back(candidate);
216 } else {
217 auto worst = std::max_element(results.begin(), results.end(), better);
218 if (better(candidate, *worst)) *worst = candidate;
219 }
220 return self->clipFraction();
221 }
222 } collector{this, maxHits};
223 owner_->rayResults_.reserve(static_cast<size_t>(maxHits));
224 b3World_CastRay(worldId_, origin, translation, filter, &RayCollector::callback, &collector);
225 std::sort(owner_->rayResults_.begin(), owner_->rayResults_.end(), RayCollector::better);
226 if (!owner_->rayResults_.empty()) {

229 owner_->rayHitShapeId_ = hit.shapeId;
230 owner_->rayHitShapeTag_ = hit.shapeTag;

237 owner_->rayHitNormalY_ = hit.normalY;

240 }
241 return static_cast<int>(owner_->rayResults_.size());
242}

468int WorldBackend::do_castProxyAll(b3Pos origin, const b3ShapeProxy &proxy, b3Vec3 translation, int maxHits) {
469 owner_->shapeCastBodyId_ = -1;
470 owner_->shapeCastShapeId_ = -1;
471 owner_->shapeCastShapeTag_ = 0;
472 owner_->shapeCastMaterialId_ = 0;
473 owner_->shapeCastTriangleIndex_ = -1;
474 owner_->shapeCastX_ = owner_->shapeCastY_ = owner_->shapeCastZ_ = 0.f;
475 owner_->shapeCastNormalX_ = owner_->shapeCastNormalY_ = owner_->shapeCastNormalZ_ = 0.f;
476 owner_->shapeCastFraction_ = 0.f;
477 owner_->shapeCastResults_.clear();
478 constexpr int maxShapeCastResults = 4096;
479 if (maxHits < 1 || maxHits > maxShapeCastResults)
480 throw eve::Exception("World3D Shape Cast: maxHits must be in [1, 4096]");
481 if (!isValid()) return 0;
482
483 struct CastCollector {
484 WorldBackend *world = nullptr;
485 int capacity = 0;
486
487 static bool better(const ShapeCastResult &a, const ShapeCastResult &b) {
488 return a.fraction < b.fraction || (a.fraction == b.fraction && a.shapeId < b.shapeId);
489 }
490
491 float clipFraction() const {
492 if (static_cast<int>(world->owner_->shapeCastResults_.size()) < capacity) return 1.f;
493 const auto worst = std::max_element(world->owner_->shapeCastResults_.begin(),
494 world->owner_->shapeCastResults_.end(), better);
495 return worst->fraction;
496 }
497
498 static float callback(b3ShapeId shapeId, b3Pos point, b3Vec3 normal, float fraction, uint64_t materialId,
499 int triangleIndex, int, void *context) {
500 auto *self = static_cast<CastCollector *>(context);
501 if (self->world->do_shouldIgnoreQueryShape(shapeId)) return -1.f;
502 Body3D *body = bodyFromShape(shapeId);
503 Shape3D *shape = shapeFromShape(shapeId);
504 if (!body || !shape) return -1.f;
505
506 ShapeCastResult candidate{body->getId(),
507 shape->getId(),
508 shape->getTag(),
509 static_cast<int>(static_cast<uint32_t>(materialId)),
510 sourceTriangleIndex(shapeId, triangleIndex),
511 static_cast<float>(point.x),
512 static_cast<float>(point.y),
513 static_cast<float>(point.z),
514 normal.x,
515 normal.y,
516 normal.z,
517 fraction};
518 auto &results = self->world->owner_->shapeCastResults_;
519 if (static_cast<int>(results.size()) < self->capacity) {
520 results.push_back(candidate);
521 } else {
522 auto worst = std::max_element(results.begin(), results.end(), better);
523 if (better(candidate, *worst)) *worst = candidate;
524 }
525 return self->clipFraction();
526 }
527 } collector{this, maxHits};
528 owner_->shapeCastResults_.reserve(static_cast<size_t>(maxHits));
529 b3QueryFilter filter = do_makeQueryFilter();
530 b3World_CastShape(worldId_, origin, &proxy, translation, filter, &CastCollector::callback, &collector);
531 std::sort(owner_->shapeCastResults_.begin(), owner_->shapeCastResults_.end(), CastCollector::better);
532 if (!owner_->shapeCastResults_.empty()) {
533 const ShapeCastResult &hit = owner_->shapeCastResults_.front();
534 owner_->shapeCastBodyId_ = hit.bodyId;
535 owner_->shapeCastShapeId_ = hit.shapeId;
536 owner_->shapeCastShapeTag_ = hit.shapeTag;
537 owner_->shapeCastMaterialId_ = hit.materialId;
538 owner_->shapeCastTriangleIndex_ = hit.triangleIndex;
539 owner_->shapeCastX_ = hit.x;
540 owner_->shapeCastY_ = hit.y;
541 owner_->shapeCastZ_ = hit.z;
542 owner_->shapeCastNormalX_ = hit.normalX;
543 owner_->shapeCastNormalY_ = hit.normalY;
544 owner_->shapeCastNormalZ_ = hit.normalZ;
545 owner_->shapeCastFraction_ = hit.fraction;
546 }
547 return static_cast<int>(owner_->shapeCastResults_.size());
548}

603int WorldBackend::do_closestPoint(float x, float y, float z, float maxDistance) {
604 owner_->closestBodyId_ = -1;
605 owner_->closestShapeId_ = -1;
606 owner_->closestShapeTag_ = 0;
607 owner_->closestX_ = owner_->closestY_ = owner_->closestZ_ = 0.f;
608 owner_->closestNormalX_ = owner_->closestNormalY_ = owner_->closestNormalZ_ = 0.f;
609 owner_->closestDistance_ = 0.f;
610 if (!std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z) || !(maxDistance >= 0.f) ||
611 !std::isfinite(maxDistance))
612 throw eve::Exception("World3D.closestPoint: target must be finite; maxDistance finite and >= 0");
613 constexpr double floatLimit = static_cast<double>(FLT_MAX);
614 const double distance = static_cast<double>(maxDistance);
615 if (distance * distance > floatLimit || std::fabs(static_cast<double>(x)) + distance > floatLimit ||
616 std::fabs(static_cast<double>(y)) + distance > floatLimit ||
617 std::fabs(static_cast<double>(z)) + distance > floatLimit)
618 throw eve::Exception("World3D.closestPoint: search bounds are too large");
619 if (!isValid()) return -1;
620
621 struct ClosestCollector {
622 WorldBackend *world = nullptr;
623 b3Vec3 target = b3Vec3_zero;
624 float bestDistanceSquared = 0.f;
625 int bodyId = -1;
626 int stableShapeId = -1;
627 int shapeTag = 0;
628 b3Vec3 point = b3Vec3_zero;
629
630 static bool callback(b3ShapeId shapeId, void *context) {
631 auto *self = static_cast<ClosestCollector *>(context);
632 if (self->world->do_shouldIgnoreQueryShape(shapeId)) return true;
633 Body3D *body = bodyFromShape(shapeId);
634 Shape3D *shape = shapeFromShape(shapeId);
635 if (!body || !shape) return true;
636 const b3Vec3 point = b3Shape_GetClosestPoint(shapeId, self->target);
637 const b3Vec3 delta = self->target - point;
638 const float distanceSquared = b3LengthSquared(delta);
639 const int bodyId = body->getId();
640 const int stableShapeId = shape->getId();
641 constexpr float tieTolerance = 1e-7f;
642 if (distanceSquared < self->bestDistanceSquared - tieTolerance ||
643 (std::fabs(distanceSquared - self->bestDistanceSquared) <= tieTolerance &&
644 (self->bodyId < 0 || bodyId < self->bodyId ||
645 (bodyId == self->bodyId && stableShapeId < self->stableShapeId)))) {
646 self->bestDistanceSquared = distanceSquared;
647 self->bodyId = bodyId;
648 self->stableShapeId = stableShapeId;
649 self->shapeTag = shape->getTag();
650 self->point = point;
651 }
652 return true;
653 }
654 } collector;
655 collector.world = this;
656 collector.target = b3Vec3{x, y, z};
657 collector.bestDistanceSquared = maxDistance * maxDistance;
658
659 const b3Vec3 extent{maxDistance, maxDistance, maxDistance};
660 const b3AABB bounds{collector.target - extent, collector.target + extent};
661 const b3QueryFilter filter = do_makeQueryFilter();
662 b3World_OverlapAABB(worldId_, bounds, filter, &ClosestCollector::callback, &collector);
663 if (collector.bodyId < 0) return -1;
664
665 owner_->closestBodyId_ = collector.bodyId;
666 owner_->closestShapeId_ = collector.stableShapeId;
667 owner_->closestShapeTag_ = collector.shapeTag;
668 owner_->closestX_ = collector.point.x;
669 owner_->closestY_ = collector.point.y;
670 owner_->closestZ_ = collector.point.z;
671 owner_->closestDistance_ = std::sqrt(std::max(0.f, collector.bestDistanceSquared));
672 if (owner_->closestDistance_ > 1e-6f) {
673 const float inverseDistance = 1.f / owner_->closestDistance_;
674 owner_->closestNormalX_ = (x - owner_->closestX_) * inverseDistance;
675 owner_->closestNormalY_ = (y - owner_->closestY_) * inverseDistance;
676 owner_->closestNormalZ_ = (z - owner_->closestZ_) * inverseDistance;
677 }
678 return owner_->closestBodyId_;
679}
