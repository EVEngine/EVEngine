41Result<ActionGuideVerifyRequest> loadActionGuideVerifyRequest(const ActionGuidePack& pack,
42 const std::filesystem::path& guideArtifactDir,
43 const std::filesystem::path& candidateDir) {
44 if (pack.shots.empty()) return fail<ActionGuideVerifyRequest>(DiagnosticCode::InvalidArgument, "pack has no shots");
45 if (!std::filesystem::is_directory(guideArtifactDir))
46 return fail<ActionGuideVerifyRequest>(DiagnosticCode::NotFound, "guideArtifactDir is missing",
47 guideArtifactDir.generic_string());
48 if (!std::filesystem::is_directory(candidateDir))
49 return fail<ActionGuideVerifyRequest>(DiagnosticCode::NotFound, "candidateDir is missing",
50 candidateDir.generic_string());
51
52 ActionGuideVerifyRequest request;
53 request.pack = pack;
54 request.shots.reserve(pack.shots.size());
55 try {
56 for (size_t i = 0; i < pack.shots.size(); ++i) {
57 const auto folder = padFolder(i + 1);
58 const auto guidePng = guideArtifactDir / "shots" / folder / "silhouette.png";
59 const auto candidateP = candidateDir / folder / "frame.png";
60 if (!std::filesystem::is_regular_file(guidePng))
61 return fail<ActionGuideVerifyRequest>(DiagnosticCode::NotFound, "missing guide silhouette.png",
62 guidePng.generic_string());
63 if (!std::filesystem::is_regular_file(candidateP))
64 return fail<ActionGuideVerifyRequest>(DiagnosticCode::NotFound, "missing candidate frame.png",
65 candidateP.generic_string());
66
67 const auto guideBytes = readAllBytes(guidePng);
68 auto guideFrame = decodeActionGuideCandidatePng(guideBytes);
69 if (!guideFrame) return Result<ActionGuideVerifyRequest>::failure(guideFrame.status());
70 const auto candBytes = readAllBytes(candidateP);
71 auto candFrame = decodeActionGuideCandidatePng(candBytes);
72 if (!candFrame) return Result<ActionGuideVerifyRequest>::failure(candFrame.status());
73
74 ActionGuideVerifyShotInput shot;
75 shot.shotId = pack.shots[i].shotId;
76 shot.width = guideFrame.value().width;
77 shot.height = guideFrame.value().height;
78 shot.guideSilhouetteRgba8 = guideFrame.value().rgba8;
79 shot.candidateRgba8 = candFrame.value().rgba8;
80 if (shot.width != candFrame.value().width || shot.height != candFrame.value().height)
81 return fail<ActionGuideVerifyRequest>(DiagnosticCode::InvalidArgument,
82 "guide and candidate sizes differ", folder);
83 request.shots.push_back(std::move(shot));
84 }
85 } catch (const std::exception& e) {
86 return fail<ActionGuideVerifyRequest>(DiagnosticCode::Failed, std::string("verify load failed: ") + e.what());
87 }
88 return Result<ActionGuideVerifyRequest>::success(std::move(request));
89}

91Result<Value> writeActionGuideVerifyEvidence(const ActionGuideVerifyReport& report,
92 const std::filesystem::path& outputRoot) {
93 try {
94 std::filesystem::create_directories(outputRoot);
95 Result<std::string> json = [&]() -> Result<std::string> {
96 if (!report.evidence.isNull()) {
97 auto encodedEvidence = report.evidence.toJson();
98 if (encodedEvidence) return encodedEvidence;
99 encodedEvidence.ignore();
100 }
101 auto encoded = encodeActionGuideVerifyReport(report);
102 if (!encoded) return Result<std::string>::failure(encoded.status());
103 return encoded.value().toJson();
104 }();
105 if (!json) return Result<Value>::failure(json.status());
106 writeBytes(outputRoot / "verify.evidence.json", json.value().data(), json.value().size());
107 return Result<Value>::success(
108 Value::object({{"evidencePath", Value((outputRoot / "verify.evidence.json").generic_string())},
109 {"passed", Value(report.passed)},
110 {"sourceGuideRevision", Value(static_cast<std::int64_t>(report.sourceGuideRevision))}}));
111 } catch (const std::exception& e) {
112 return fail<Value>(DiagnosticCode::Failed, std::string("verify evidence write failed: ") + e.what());
113 }
114}
