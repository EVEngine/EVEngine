33Result<Document> parseDocument(const GltfImportRequest& request) {
34 if (request.documentBytes.empty() || request.documentBytes.size() > request.limits.maximumSourceBytes)
35 return detail::failure<Document>(DiagnosticCode::InvalidArgument, "glTF document size is outside limits");
36 std::string json;
37 std::vector<std::uint8_t> binary;
38 const auto bytes = std::span<const std::uint8_t>(request.documentBytes);
39 if (bytes.size() >= 12 && little32(bytes, 0) == 0x46546c67) {
40 if (little32(bytes, 4) != 2 || little32(bytes, 8) != bytes.size() || bytes.size() < 20)
41 return detail::failure<Document>(DiagnosticCode::ParseError, "GLB header is invalid");
42 std::size_t cursor = 12;
43 const std::uint32_t jsonSize = little32(bytes, cursor);
44 const std::uint32_t jsonType = little32(bytes, cursor + 4);
45 cursor += 8;
46 if (jsonType != 0x4e4f534a || jsonSize > bytes.size() - cursor)
47 return detail::failure<Document>(DiagnosticCode::ParseError, "GLB JSON chunk is invalid");
48 json.assign(reinterpret_cast<const char*>(bytes.data() + cursor), jsonSize);
49 while (!json.empty() && (json.back() == ' ' || json.back() == '\0')) json.pop_back();
50 cursor += jsonSize;
51 if (cursor < bytes.size()) {
52 if (bytes.size() - cursor < 8)
53 return detail::failure<Document>(DiagnosticCode::ParseError, "GLB trailing chunk is truncated");
54 const std::uint32_t binarySize = little32(bytes, cursor);
55 const std::uint32_t binaryType = little32(bytes, cursor + 4);
56 cursor += 8;
57 if (binaryType != 0x004e4942 || binarySize > bytes.size() - cursor || cursor + binarySize != bytes.size())
58 return detail::failure<Document>(DiagnosticCode::ParseError, "GLB BIN chunk is invalid");
59 binary.assign(bytes.begin() + cursor, bytes.end());
60 }
61 } else {
62 json.assign(request.documentBytes.begin(), request.documentBytes.end());
63 }
64 auto parsed = Value::fromJson(json);
65 if (!parsed) return Result<Document>::failure(parsed.status());
66 return Result<Document>::success({std::move(parsed).takeValue(), std::move(binary)});
67}

69bool safeUri(std::string_view uri, const AssetImportLimits& limits) {
70 if (uri.empty() || uri.size() > limits.maximumStringBytes || uri.front() == '/' ||
71 uri.find('\\') != std::string_view::npos || uri.starts_with("data:") || uri.find(':') != std::string_view::npos)
72 return false;
73 std::size_t start = 0;
74 for (std::size_t index = 0; index <= uri.size(); ++index) {
75 if (index != uri.size() && uri[index] != '/') continue;
76 const auto segment = uri.substr(start, index - start);
77 if (segment.empty() || segment == "." || segment == "..") return false;
78 start = index + 1;
79 }
80 return true;
81}

83Result<std::vector<std::span<const std::uint8_t>>> resolveBuffers(const Value::Object& root, const Document& document,
84 const GltfImportRequest& request) {
85 const Value* buffersValue = member(root, "buffers");
86 const auto* buffers = buffersValue ? buffersValue->getIf<Value::Array>() : nullptr;
87 if (!buffers || buffers->empty())
88 return detail::failure<std::vector<std::span<const std::uint8_t>>>(DiagnosticCode::ParseError,
89 "glTF buffers are required", "$.buffers");
90 std::vector<std::span<const std::uint8_t>> result;
91 result.reserve(buffers->size());
92 std::uint64_t totalBytes = 0;
93 for (std::size_t index = 0; index < buffers->size(); ++index) {
94 const auto* object = (*buffers)[index].getIf<Value::Object>();
95 if (!object)
96 return detail::failure<std::vector<std::span<const std::uint8_t>>>(DiagnosticCode::ParseError,
97 "glTF buffer must be an object");
98 auto declared =
99 unsignedValue(member(*object, "byteLength"), "$.buffers[" + std::to_string(index) + "].byteLength");
100 if (!declared) return Result<std::vector<std::span<const std::uint8_t>>>::failure(declared.status());
101 std::span<const std::uint8_t> data;
102 const Value* uriValue = member(*object, "uri");
103 if (uriValue) {
104 if (!uriValue->isString() || !safeUri(uriValue->asString(), request.limits))
105 return detail::failure<std::vector<std::span<const std::uint8_t>>>(
106 DiagnosticCode::Unsupported, "external glTF buffer URI is unsafe or unsupported");
107 const auto found = request.externalResources.find(uriValue->asString());
108 if (found == request.externalResources.end())
109 return detail::failure<std::vector<std::span<const std::uint8_t>>>(
110 DiagnosticCode::NotFound, "external glTF buffer was not supplied", uriValue->asString());
111 data = found->second;
112 } else {
113 if (index != 0 || document.binaryChunk.empty())
114 return detail::failure<std::vector<std::span<const std::uint8_t>>>(
115 DiagnosticCode::ParseError, "buffer without URI requires the first GLB BIN chunk");
116 data = document.binaryChunk;
117 }
118 if (declared.value() > data.size())
119 return detail::failure<std::vector<std::span<const std::uint8_t>>>(
120 DiagnosticCode::ParseError, "glTF buffer is shorter than byteLength");
121 if (declared.value() > request.limits.maximumSourceBytes ||
122 totalBytes > request.limits.maximumSourceBytes - declared.value())
123 return detail::failure<std::vector<std::span<const std::uint8_t>>>(DiagnosticCode::InvalidArgument,
124 "glTF buffer budget is exceeded");
125 totalBytes += declared.value();
126 result.push_back(data.first(static_cast<std::size_t>(declared.value())));
127 }
128 return Result<std::vector<std::span<const std::uint8_t>>>::success(std::move(result));
129}

161 std::uint32_t components = 0;

163 components = 1;
164 else if (typeValue->asString() == "VEC2")
165 components = 2;
166 else if (typeValue->asString() == "VEC3")
167 components = 3;
168 else if (typeValue->asString() == "VEC4")
169 components = 4;
170 else if (typeValue->asString() == "MAT4")
171 components = 16;

174 const auto* view = (*views)[viewIndex.value()].getIf<Value::Object>();
175 if (!view) return detail::failure<Accessor>(DiagnosticCode::ParseError, "glTF bufferView is malformed");
176 auto bufferIndex = unsignedValue(member(*view, "buffer"), "bufferView.buffer");
177 auto viewOffset = unsignedValue(member(*view, "byteOffset"), "bufferView.byteOffset", false, 0);

183 const std::uint64_t stride = byteStride.value() == 0 ? elementSize : byteStride.value();
184 if (elementSize == 0 || stride < elementSize || stride > 252 ||
185 viewOffset.value() > buffers[bufferIndex.value()].size() ||

189 const std::uint64_t required = (count.value() - 1) * stride + elementSize;
190 if (required > viewLength.value() - accessorOffset.value())
191 return detail::failure<Accessor>(DiagnosticCode::ParseError, "glTF accessor range is truncated");
192 const auto* normalized = member(*object, "normalized");
193 if (normalized && !normalized->isBool())
194 return detail::failure<Accessor>(DiagnosticCode::ParseError, "glTF normalized flag must be boolean");
195 return Result<Accessor>::success(
196 {buffers[bufferIndex.value()], static_cast<std::size_t>(viewOffset.value() + accessorOffset.value()),
197 static_cast<std::size_t>(stride), static_cast<std::uint32_t>(count.value()),
198 static_cast<std::uint32_t>(component.value()), components, normalized && normalized->asBool()});
199}
