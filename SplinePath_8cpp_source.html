160 if (chunk < 0 || chunk >= chunkCount())
161 return splineFailure<SplinePath>(DiagnosticCode::NotFound, "spline chunk index is out of range", "chunk.index");
162 SplinePath result;
163 result.kind_ = kind_;

195SplineSample SplinePath::evaluateUnchecked(float t) const {
196 t = std::clamp(t, 0.f, 1.f);
197 const int segments = segmentCount();
198 const float scaled = t * static_cast<float>(segments);
199 const int segment = std::min(static_cast<int>(scaled), segments - 1);
200 const float u = segment == segments - 1 && t == 1.f ? 1.f : scaled - static_cast<float>(segment);
201 return evaluateSegmentUnchecked(segment, u, t);

204SplineSample SplinePath::evaluateSegmentUnchecked(int segment, float u, float normalizedDistance) const {
205 const auto endpoints = segmentEndpoints();
206 const int aIndex = endpoints[static_cast<std::size_t>(segment)].first;
207 const int bIndex = endpoints[static_cast<std::size_t>(segment)].second;
208 const auto& a = points_[static_cast<std::size_t>(aIndex)];
209 const auto& b = points_[static_cast<std::size_t>(bIndex)];
210 int chunkIndex = 0;
211 for (int i = 1; i <= aIndex; ++i)
212 if (points_[static_cast<std::size_t>(i)].breakBefore) ++chunkIndex;
213 V3 value{}, tangent{};
214 if (kind_ == "linear") {
215 value = add(position(a), mul(sub(position(b), position(a)), u));
216 tangent = sub(position(b), position(a));
217 } else if (kind_ == "quadraticBezier") {
218 const V3 p0 = position(a), p2 = position(b);
219 const V3 outgoing = add(p0, {a.outX, a.outY, a.outZ});
220 const V3 incoming = add(p2, {b.inX, b.inY, b.inZ});
221 const V3 p1 = mul(add(outgoing, incoming), 0.5f);
222 const float v = 1.f - u;
223 value = add(add(mul(p0, v * v), mul(p1, 2.f * v * u)), mul(p2, u * u));
224 tangent = add(mul(sub(p1, p0), 2.f * v), mul(sub(p2, p1), 2.f * u));
225 } else if (kind_ == "bezier") {
226 const V3 p0 = position(a), p1 = add(p0, {a.outX, a.outY, a.outZ});
227 const V3 p3 = position(b), p2 = add(p3, {b.inX, b.inY, b.inZ});
228 const float v = 1.f - u;
229 value =
230 add(add(mul(p0, v * v * v), mul(p1, 3.f * v * v * u)), add(mul(p2, 3.f * v * u * u), mul(p3, u * u * u)));
231 tangent = add(add(mul(sub(p1, p0), 3.f * v * v), mul(sub(p2, p1), 6.f * v * u)), mul(sub(p3, p2), 3.f * u * u));
232 } else {
233 int previous = aIndex;
234 if (aIndex > 0 && !points_[static_cast<std::size_t>(aIndex)].breakBefore)
235 previous = aIndex - 1;
236 else if (closed_ && aIndex == 0)
237 previous = pointCount() - 1;
238 int next = bIndex;
239 if (bIndex + 1 < pointCount() && !points_[static_cast<std::size_t>(bIndex + 1)].breakBefore)
240 next = bIndex + 1;
241 else if (closed_ && bIndex == pointCount() - 1)
242 next = 0;
243 const V3 p0 = position(points_[static_cast<std::size_t>(previous)]);
244 const V3 p1 = position(a), p2 = position(b);
245 const V3 p3 = position(points_[static_cast<std::size_t>(next)]);
246 const float u2 = u * u, u3 = u2 * u;
247 value = mul(add(add(mul(p1, 2.f), mul(sub(p2, p0), u)),
248 add(mul(add(add(mul(p0, 2.f), mul(p1, -5.f)), add(mul(p2, 4.f), mul(p3, -1.f))), u2),
249 mul(add(add(mul(p0, -1.f), mul(p1, 3.f)), add(mul(p2, -3.f), p3)), u3))),
250 0.5f);
251 tangent = mul(
252 add(sub(p2, p0), add(mul(add(add(mul(p0, 2.f), mul(p1, -5.f)), add(mul(p2, 4.f), mul(p3, -1.f))), 2.f * u),
253 mul(add(add(mul(p0, -1.f), mul(p1, 3.f)), add(mul(p2, -3.f), p3)), 3.f * u2))),
254 0.5f);
255 }
256 const float tangentLength = length(tangent);
257 if (tangentLength > 1e-7f)
258 tangent = mul(tangent, 1.f / tangentLength);
259 else
260 tangent = {1.f, 0.f, 0.f};
261 return {value.x,
262 value.y,
263 value.z,
264 tangent.x,
265 tangent.y,
266 tangent.z,
267 normalizedDistance,
268 std::lerp(a.rollDegrees, b.rollDegrees, u),
269 std::lerp(a.scaleX, b.scaleX, u),
270 std::lerp(a.scaleY, b.scaleY, u),
271 chunkIndex,
272 std::lerp(a.pitchDegrees, b.pitchDegrees, u),
273 std::lerp(a.yawDegrees, b.yawDegrees, u)};
274}

284Result<void> SplinePath::ensureArcTable(int samplesPerSegment) const {
285 auto valid = validateReady();
286 if (!valid.ok()) return valid;
287 if (samplesPerSegment < 2 || samplesPerSegment > 1024)
288 return splineFailure<void>(DiagnosticCode::InvalidArgument, "arc samples per segment must be in [2, 1024]",
289 "samplesPerSegment");
290 if (arcSamplesPerSegment_ == samplesPerSegment && !arcTable_.empty()) return Result<void>::success();
291 arcTable_.clear();
292 arcTable_.reserve(static_cast<std::size_t>(segmentCount() * samplesPerSegment + chunkCount()));
293 float distance = 0.f;
294 const auto endpoints = segmentEndpoints();
295 for (int segment = 0; segment < segmentCount(); ++segment) {
296 const bool startsChunk = segment == 0 || endpoints[static_cast<std::size_t>(segment - 1)].second !=
297 endpoints[static_cast<std::size_t>(segment)].first;
298 auto previous =
299 evaluateSegmentUnchecked(segment, 0.f, static_cast<float>(segment) / static_cast<float>(segmentCount()));
300 if (startsChunk) arcTable_.push_back({segment, 0.f, distance});
301 for (int sample = 1; sample <= samplesPerSegment; ++sample) {
302 const float u = static_cast<float>(sample) / static_cast<float>(samplesPerSegment);
303 const auto current = evaluateSegmentUnchecked(
304 segment, u, (static_cast<float>(segment) + u) / static_cast<float>(segmentCount()));
305 distance += length({current.x - previous.x, current.y - previous.y, current.z - previous.z});
306 arcTable_.push_back({segment, u, distance});
307 previous = current;
308 }
309 }
310 arcSamplesPerSegment_ = samplesPerSegment;
311 return Result<void>::success();
312}

320Result<SplineSample> SplinePath::evaluateDistanceResult(float distance, int samplesPerSegment) const {
321 if (!std::isfinite(distance))
322 return splineFailure<SplineSample>(DiagnosticCode::InvalidArgument, "spline distance must be finite",
323 "distance");
324 auto ready = ensureArcTable(samplesPerSegment);
325 if (!ready.ok()) return Result<SplineSample>::failure(ready.status());
326 const float total = arcTable_.back().distance;
327 distance = std::clamp(distance, 0.f, total);
328 const auto upper = std::lower_bound(arcTable_.begin(), arcTable_.end(), distance,
329 [](const ArcEntry& entry, float value) { return entry.distance < value; });
330 int segment = 0;
331 float u = 0.f;
332 if (upper == arcTable_.begin()) {
333 segment = upper->segment;
334 u = upper->u;
335 } else if (upper == arcTable_.end()) {
336 segment = arcTable_.back().segment;
337 u = arcTable_.back().u;
338 } else {
339 const auto& before = *(upper - 1);
340 const float span = upper->distance - before.distance;
341 if (before.segment != upper->segment || span <= 1e-7f) {
342 segment = upper->segment;
343 u = upper->u;
344 } else {
345 segment = upper->segment;
346 u = std::lerp(before.u, upper->u, (distance - before.distance) / span);
347 }
348 }
349 auto sample =
350 evaluateSegmentUnchecked(segment, u, (static_cast<float>(segment) + u) / static_cast<float>(segmentCount()));
351 sample.normalizedDistance = total > 1e-7f ? distance / total : 0.f;
352 return Result<SplineSample>::success(sample);
353}

355Result<SplineSample> SplinePath::closestPointResult(float x, float y, float z, int samplesPerSegment) const {
356 if (!std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z))
357 return splineFailure<SplineSample>(DiagnosticCode::InvalidArgument, "closest point query must be finite",
358 "point");
359 auto ready = ensureArcTable(samplesPerSegment);
360 if (!ready.ok()) return Result<SplineSample>::failure(ready.status());
361 float bestSquared = 0.f;
362 SplineSample best;
363 bool assigned = false;
364 for (const auto& entry : arcTable_) {
365 auto sample = evaluateSegmentUnchecked(
366 entry.segment, entry.u, (static_cast<float>(entry.segment) + entry.u) / static_cast<float>(segmentCount()));
367 const float dx = sample.x - x, dy = sample.y - y, dz = sample.z - z;
368 const float squared = dx * dx + dy * dy + dz * dz;
369 if (!assigned || squared < bestSquared) {
370 assigned = true;
371 bestSquared = squared;
372 best = sample;
373 best.normalizedDistance =
374 arcTable_.back().distance > 1e-7f ? entry.distance / arcTable_.back().distance : 0.f;
375 }
376 }
377 return Result<SplineSample>::success(best);
378}

380Result<std::vector<SplineFrameSample>> SplinePath::sampleFramesResult(int sampleCount, bool uniformByDistance,
381 float rollDegrees, int samplesPerSegment) const {
382 if (sampleCount < 1 || sampleCount > 4096)
383 return splineFailure<std::vector<SplineFrameSample>>(DiagnosticCode::InvalidArgument,
384 "frame sampleCount must be in [1, 4096]", "sampleCount");
385 if (!std::isfinite(rollDegrees))
386 return splineFailure<std::vector<SplineFrameSample>>(DiagnosticCode::InvalidArgument,
387 "frame roll must be finite", "rollDegrees");
388 float totalLength = 0.f;
389 if (uniformByDistance) {
390 auto measured = lengthResult(samplesPerSegment);
391 if (!measured.ok()) return Result<std::vector<SplineFrameSample>>::failure(measured.status());
392 totalLength = measured.value();
393 } else {
394 auto ready = validateReady();
395 if (!ready.ok()) return Result<std::vector<SplineFrameSample>>::failure(ready.status());
396 }
397
398 std::vector<SplineFrameSample> frames;
399 frames.reserve(static_cast<std::size_t>(sampleCount + 1));
400 V3 previousTangent{}, previousSide{};
401 for (int index = 0; index <= sampleCount; ++index) {
402 const float ratio = static_cast<float>(index) / static_cast<float>(sampleCount);
403 auto sampled =
404 uniformByDistance ? evaluateDistanceResult(totalLength * ratio, samplesPerSegment) : evaluateResult(ratio);
405 if (!sampled.ok()) return Result<std::vector<SplineFrameSample>>::failure(sampled.status());
406 const auto sample = sampled.value();
407 const V3 tangent = normalized({sample.tangentX, sample.tangentY, sample.tangentZ});
408 V3 side;
409 if (index == 0 || sample.chunkIndex != frames.back().sample.chunkIndex) {
410 const V3 reference = std::abs(tangent.y) < 0.9f ? V3{0.f, 1.f, 0.f} : V3{1.f, 0.f, 0.f};
411 side = normalized(cross(reference, tangent));
412 } else {
413 const V3 axis = cross(previousTangent, tangent);
414 const float axisSize = length(axis);
415 if (axisSize > 1e-7f) {
416 const float angle = std::atan2(axisSize, std::clamp(dot(previousTangent, tangent), -1.f, 1.f));
417 side = rotateAround(previousSide, mul(axis, 1.f / axisSize), angle);
418 } else if (dot(previousTangent, tangent) < 0.f) {
419 side = mul(previousSide, -1.f);
420 } else {
421 side = previousSide;
422 }
423 side = normalized(sub(side, mul(tangent, dot(side, tangent))));
424 }
425 V3 up = normalized(cross(tangent, side));
426 side = normalized(cross(up, tangent));
427 frames.push_back({sample, side.x, side.y, side.z, up.x, up.y, up.z, tangent.x, tangent.y, tangent.z});
428 previousTangent = tangent;
429 previousSide = side;
430 }
431
432 if (closed_ && frames.size() > 1u) {
433 const V3 firstSide{frames.front().sideX, frames.front().sideY, frames.front().sideZ};
434 const V3 lastSide{frames.back().sideX, frames.back().sideY, frames.back().sideZ};
435 const V3 tangent{frames.front().sample.tangentX, frames.front().sample.tangentY,
436 frames.front().sample.tangentZ};
437 const float correction =
438 std::atan2(dot(tangent, cross(lastSide, firstSide)), std::clamp(dot(lastSide, firstSide), -1.f, 1.f));
439 for (int index = 1; index <= sampleCount; ++index) {
440 auto& frame = frames[static_cast<std::size_t>(index)];
441 const float ratio = static_cast<float>(index) / static_cast<float>(sampleCount);
442 const V3 currentTangent{frame.sample.tangentX, frame.sample.tangentY, frame.sample.tangentZ};
443 V3 side = rotateAround({frame.sideX, frame.sideY, frame.sideZ}, currentTangent, correction * ratio);
444 V3 up = normalized(cross(currentTangent, side));
445 side = normalized(cross(up, currentTangent));
446 frame.sideX = side.x;
447 frame.sideY = side.y;
448 frame.sideZ = side.z;
449 frame.upX = up.x;
450 frame.upY = up.y;
451 frame.upZ = up.z;
452 }
453 frames.back() = frames.front();
454 frames.back().sample.normalizedDistance = 1.f;
455 }
456
457 for (auto& frame : frames) {
458 V3 side{frame.sideX, frame.sideY, frame.sideZ};
459 V3 up{frame.upX, frame.upY, frame.upZ};
460 V3 forward{frame.sample.tangentX, frame.sample.tangentY, frame.sample.tangentZ};
461 const float yaw = frame.sample.yawDegrees * 0.01745329251994329577f;
462 if (std::abs(yaw) > 1e-7f) {
463 forward = rotateAround(forward, up, yaw);
464 side = rotateAround(side, up, yaw);
465 }
466 const float pitch = frame.sample.pitchDegrees * 0.01745329251994329577f;
467 if (std::abs(pitch) > 1e-7f) {
468 forward = rotateAround(forward, side, pitch);
469 up = rotateAround(up, side, pitch);
470 }
471 const float roll = (rollDegrees + frame.sample.rollDegrees) * 0.01745329251994329577f;
472 if (std::abs(roll) > 1e-7f) {
473 const float c = std::cos(roll), s = std::sin(roll);
474 const V3 rolledSide = add(mul(side, c), mul(up, s));
475 up = add(mul(up, c), mul(side, -s));
476 side = rolledSide;
477 }
478 frame.sideX = side.x;
479 frame.sideY = side.y;
480 frame.sideZ = side.z;
481 frame.upX = up.x;
482 frame.upY = up.y;
483 frame.upZ = up.z;
484 frame.forwardX = forward.x;
485 frame.forwardY = forward.y;
486 frame.forwardZ = forward.z;
487 }
488 return Result<std::vector<SplineFrameSample>>::success(std::move(frames));
489}

533Result<SplineSample> SplinePath::travelResult(float distance, std::string_view wrapMode, int samplesPerSegment) const {
534 if (!std::isfinite(distance))
535 return splineFailure<SplineSample>(DiagnosticCode::InvalidArgument, "spline travel distance must be finite",
536 "distance");
537 if (wrapMode != "clamp" && wrapMode != "loop" && wrapMode != "pingPong")
538 return splineFailure<SplineSample>(DiagnosticCode::InvalidArgument,
539 "spline travel wrap mode must be clamp, loop, or pingPong", "wrapMode");
540 auto measured = lengthResult(samplesPerSegment);
541 if (!measured.ok()) return Result<SplineSample>::failure(measured.status());
542 const float total = measured.value();
543 float resolved = distance;
544 if (wrapMode == "loop" && total > 1e-7f) {
545 resolved = std::fmod(distance, total);
546 if (resolved < 0.f) resolved += total;
547 } else if (wrapMode == "pingPong" && total > 1e-7f) {
548 const float period = total * 2.f;
549 resolved = std::fmod(distance, period);
550 if (resolved < 0.f) resolved += period;
551 if (resolved > total) resolved = period - resolved;
552 }
553 return evaluateDistanceResult(resolved, samplesPerSegment);
554}

556Result<SplineFrameSample> SplinePath::travelFrameResult(float distance, std::string_view wrapMode,
557 int samplesPerSegment) const {
558 auto sample = travelResult(distance, wrapMode, samplesPerSegment);
559 if (!sample.ok()) return Result<SplineFrameSample>::failure(sample.status());
560 const int frameIntervals = std::clamp(segmentCount() * samplesPerSegment, 1, 4096);
561 auto frames = sampleFramesResult(frameIntervals, true, 0.f, samplesPerSegment);
562 if (!frames.ok()) return Result<SplineFrameSample>::failure(frames.status());
563 const float location = sample.value().normalizedDistance * static_cast<float>(frameIntervals);
564 const int lowerIndex = std::clamp(static_cast<int>(std::floor(location)), 0, frameIntervals);
565 const int upperIndex = std::min(lowerIndex + 1, frameIntervals);
566 const float blend = location - static_cast<float>(lowerIndex);
567 const auto& lower = frames.value()[static_cast<std::size_t>(lowerIndex)];
568 const auto& upper = frames.value()[static_cast<std::size_t>(upperIndex)];
569 V3 forward =
570 normalized({std::lerp(lower.forwardX, upper.forwardX, blend), std::lerp(lower.forwardY, upper.forwardY, blend),
571 std::lerp(lower.forwardZ, upper.forwardZ, blend)});
572 V3 side = normalized({std::lerp(lower.sideX, upper.sideX, blend), std::lerp(lower.sideY, upper.sideY, blend),
573 std::lerp(lower.sideZ, upper.sideZ, blend)});
574 side = normalized(sub(side, mul(forward, dot(side, forward))));
575 const V3 up = normalized(cross(forward, side));
576 side = normalized(cross(up, forward));
577 return Result<SplineFrameSample>::success(
578 {sample.value(), side.x, side.y, side.z, up.x, up.y, up.z, forward.x, forward.y, forward.z});
579}

583 if (instanceCount < 1 || instanceCount > 100000)

594Result<SplinePolyline> SplinePath::polylineResult(int sampleCount, bool uniformByDistance,
595 int samplesPerSegment) const {
596 if (sampleCount < 2 || sampleCount > 65536)
597 return splineFailure<SplinePolyline>(DiagnosticCode::InvalidArgument,
598 "spline polyline sampleCount must be in [2, 65536]", "sampleCount");
599 if (sampleCount < chunkCount() * 2)
600 return splineFailure<SplinePolyline>(DiagnosticCode::InvalidArgument,
601 "spline polyline requires at least two samples per chunk", "sampleCount");
602 std::vector<std::vector<SplineSample>> chunks;
603 chunks.reserve(static_cast<std::size_t>(chunkCount()));
604 int remaining = sampleCount;
605 for (int chunk = 0; chunk < chunkCount(); ++chunk) {
606 auto path = chunkPathResult(chunk);
607 if (!path.ok()) return Result<SplinePolyline>::failure(path.status());
608 const int chunksLeft = chunkCount() - chunk;
609 const int maximum = remaining - (chunksLeft - 1) * 2;
610 const int proposed = sampleCount * path.value().segmentCount() / segmentCount();
611 const int chunkSamples = chunksLeft == 1 ? remaining : std::clamp(proposed, 2, maximum);
612 remaining -= chunkSamples;
613 auto measured = uniformByDistance ? path.value().lengthResult(samplesPerSegment) : Result<float>::success(0.f);
614 if (!measured.ok()) return Result<SplinePolyline>::failure(measured.status());
615 std::vector<SplineSample> points;
616 points.reserve(static_cast<std::size_t>(chunkSamples));
617 const int denominator = path.value().isClosed() ? chunkSamples : chunkSamples - 1;
618 for (int index = 0; index < chunkSamples; ++index) {
619 const float ratio = static_cast<float>(index) / static_cast<float>(denominator);
620 auto sample = uniformByDistance
621 ? path.value().evaluateDistanceResult(measured.value() * ratio, samplesPerSegment)
622 : path.value().evaluateResult(ratio);
623 if (!sample.ok()) return Result<SplinePolyline>::failure(sample.status());
624 points.push_back(sample.value());
625 }
626 chunks.push_back(std::move(points));
627 }
628 return Result<SplinePolyline>::success(SplinePolyline(std::move(chunks), closed_));
629}

631Result<void> SplinePath::applyShapePresetResult(std::string_view preset, int pointCount, float radius, float height,
632 float turns) {
633 if (preset != "line" && preset != "circle" && preset != "arc" && preset != "spiral" && preset != "wave")
634 return splineFailure<void>(DiagnosticCode::InvalidArgument,
635 "spline preset must be line, circle, arc, spiral, or wave", "preset");
636 if (pointCount < 2 || pointCount > 4096 || (preset == "circle" && pointCount < 3))
637 return splineFailure<void>(DiagnosticCode::InvalidArgument, "spline preset pointCount is invalid",
638 "pointCount");
639 if (!std::isfinite(radius) || !std::isfinite(height) || !std::isfinite(turns) || radius <= 0.f || turns <= 0.f)
640 return splineFailure<void>(DiagnosticCode::InvalidArgument,
641 "spline preset requires finite positive radius and turns", "preset.parameters");
642 constexpr float tau = 6.28318530717958647692f;
643 std::vector<SplinePoint> candidate;
644 candidate.reserve(static_cast<std::size_t>(pointCount));
645 for (int index = 0; index < pointCount; ++index) {
646 const float t = pointCount > 1 ? static_cast<float>(index) / static_cast<float>(pointCount - 1) : 0.f;
647 SplinePoint point;
648 if (preset == "line") {
649 point = {-radius + 2.f * radius * t, height * t, 0.f};
650 } else if (preset == "circle") {
651 const float angle = tau * static_cast<float>(index) / static_cast<float>(pointCount);
652 point = {std::cos(angle) * radius, 0.f, std::sin(angle) * radius};
653 } else if (preset == "arc") {
654 const float angle = (-0.25f + 0.5f * t) * tau * turns;
655 point = {std::cos(angle) * radius, height * t, std::sin(angle) * radius};
656 } else if (preset == "spiral") {
657 const float angle = tau * turns * t;
658 const float ring = radius * t;
659 point = {std::cos(angle) * ring, height * t, std::sin(angle) * ring};
660 } else {
661 point = {-radius + 2.f * radius * t, std::sin(tau * turns * t) * height, 0.f};
662 }
663 candidate.push_back(point);
664 }
665 points_ = std::move(candidate);
666 kind_ = preset == "line" ? "linear" : "catmullRom";
667 closed_ = preset == "circle";
668 invalidate();
669 return Result<void>::success();
670}
