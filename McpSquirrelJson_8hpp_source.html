53inline std::string sqValueToJson(HSQUIRRELVM vm, SQInteger idx, int depth) {
54 if (idx < 0) idx = sq_gettop(vm) + idx + 1; // normalize relative -> absolute
55 if (depth > kSqJsonMaxDepth) return "null";
56 switch (sq_gettype(vm, idx)) {
57 case OT_NULL: return "null";
58 case OT_BOOL: {
59 SQBool value = SQFalse;
60 sq_getbool(vm, idx, &value);
61 return value ? "true" : "false";
62 }
63 case OT_INTEGER: {
64 SQInteger value = 0;
65 sq_getinteger(vm, idx, &value);
66 return std::to_string(static_cast<long long>(value));
67 }
68 case OT_FLOAT: {
69 SQFloat value = 0;
70 sq_getfloat(vm, idx, &value);
71 char buffer[64];
72 std::snprintf(buffer, sizeof(buffer), "%g", static_cast<double>(value));
73 return buffer;
74 }
75 case OT_STRING: {
76 const SQChar* value = nullptr;
77 sq_getstring(vm, idx, &value);
78 return std::string("\"") + mcpJsonEscape(value ? value : "") + "\"";
79 }
80 case OT_ARRAY: {
81 std::string out = "[";
82 bool first = true;
83 sq_pushnull(vm);
84 while (SQ_SUCCEEDED(sq_next(vm, idx))) {
85 if (!first) out += ",";
86 first = false;
87 out += sqValueToJson(vm, -1, depth + 1);
88 sq_pop(vm, 2);
89 }
90 sq_pop(vm, 1);
91 out += "]";
92 return out;
93 }
94 case OT_TABLE: {
95 std::string out = "{";
96 bool first = true;
97 sq_pushnull(vm);
98 while (SQ_SUCCEEDED(sq_next(vm, idx))) {
99 if (!first) out += ",";
100 first = false;
101 out += sqKeyToJson(vm, -2);
102 out += ":";
103 out += sqValueToJson(vm, -1, depth + 1);
104 sq_pop(vm, 2);
105 }
106 sq_pop(vm, 1);
107 out += "}";
108 return out;
109 }
110 default: return "\"<unserializable>\"";
111 }
112}
