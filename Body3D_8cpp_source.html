66b3Quat normalizedQuaternion(float qx, float qy, float qz, float qw, const char *operation) {
67 requireFinite(qx, operation, "qx");
68 requireFinite(qy, operation, "qy");
69 requireFinite(qz, operation, "qz");
70 requireFinite(qw, operation, "qw");
71 const float lengthSquared = qx * qx + qy * qy + qz * qz + qw * qw;
72 if (lengthSquared <= 1e-16f)
73 throw eve::Exception("%s: quaternion length must be > 0", operation);
74 const float inverseLength = 1.f / std::sqrt(lengthSquared);
75 return b3Quat{{qx * inverseLength, qy * inverseLength, qz * inverseLength},
76 qw * inverseLength};
77}

89b3HullData *createCheckedHull(const std::vector<float> &vertices, int maxVertices,
90 const char *operation) {
91 if (vertices.size() < 12 || vertices.size() % 3 != 0)
92 throw eve::Exception("%s: vertices must contain at least four packed XYZ points",
93 operation);
94 if (vertices.size() / 3 > 100000)
95 throw eve::Exception("%s: source point count must be <= 100000", operation);
96 if (maxVertices < 4 || maxVertices > 254)
97 throw eve::Exception("%s: maxVertices must be in [4, 254]", operation);
98 std::vector<b3Vec3> points;
99 points.reserve(vertices.size() / 3);
100 for (size_t i = 0; i < vertices.size(); i += 3) {
101 if (!std::isfinite(vertices[i]) || !std::isfinite(vertices[i + 1]) ||
102 !std::isfinite(vertices[i + 2]))
103 throw eve::Exception("%s: all vertex components must be finite", operation);
104 points.push_back({vertices[i], vertices[i + 1], vertices[i + 2]});
105 }
106 b3HullData *hull = b3CreateHull(points.data(), static_cast<int>(points.size()), maxVertices);
107 if (!hull)
108 throw eve::Exception("%s: points do not form a valid three-dimensional convex hull",
109 operation);
110 return hull;
111}

113b3MeshData *createCheckedMesh(const std::vector<float> &vertices,
114 const std::vector<int32_t> &indices, bool weldVertices,
115 float weldTolerance, bool identifyEdges, bool useMedianSplit,
116 const char *operation) {
117 if (vertices.size() < 9 || vertices.size() % 3 != 0)
118 throw eve::Exception("%s: vertices must contain at least three packed XYZ points",
119 operation);
120 if (indices.size() < 3 || indices.size() % 3 != 0)
121 throw eve::Exception("%s: indices must contain complete triangles", operation);
122 if (vertices.size() / 3 > 1000000 || indices.size() / 3 > 2000000)
123 throw eve::Exception("%s: mesh exceeds 1000000 vertices or 2000000 triangles",
124 operation);
125 if (!std::isfinite(weldTolerance) || weldTolerance < 0.f)
126 throw eve::Exception("%s: weldTolerance must be finite and >= 0", operation);
127 std::vector<b3Vec3> points;
128 points.reserve(vertices.size() / 3);
129 for (size_t i = 0; i < vertices.size(); i += 3) {
130 if (!std::isfinite(vertices[i]) || !std::isfinite(vertices[i + 1]) ||
131 !std::isfinite(vertices[i + 2]))
132 throw eve::Exception("%s: all vertex components must be finite", operation);
133 points.push_back({vertices[i], vertices[i + 1], vertices[i + 2]});
134 }
135 for (int32_t index : indices) {
136 if (index < 0 || static_cast<size_t>(index) >= points.size())
137 throw eve::Exception("%s: triangle index is outside the vertex array", operation);
138 }
139 std::vector<int32_t> mutableIndices = indices;
140 b3MeshDef def{};
141 def.vertices = points.data();
142 def.indices = mutableIndices.data();
143 def.vertexCount = static_cast<int>(points.size());
144 def.triangleCount = static_cast<int>(indices.size() / 3);
145 def.weldVertices = weldVertices;
146 def.weldTolerance = weldTolerance;
147 def.identifyEdges = identifyEdges;
148 def.useMedianSplit = useMedianSplit;
149 b3MeshData *mesh = b3CreateMesh(&def, nullptr, 0);
150 if (!mesh || mesh->triangleCount != def.triangleCount) {
151 if (mesh) b3DestroyMesh(mesh);
152 throw eve::Exception("%s: mesh contains degenerate or zero-area triangles", operation);
153 }
154 return mesh;
155}

157b3HeightFieldData *createCheckedHeightField(int countX, int countZ, float cellSizeX,
158 float cellSizeZ,
159 const std::vector<float> &heights,
160 float globalMin, float globalMax,
161 bool clockwiseWinding, const char *operation) {
162 if (countX < 2 || countZ < 2)
163 throw eve::Exception("%s: countX and countZ must be >= 2", operation);
164 const size_t sampleCount = static_cast<size_t>(countX) * static_cast<size_t>(countZ);
165 if (sampleCount > 16000000)
166 throw eve::Exception("%s: sample count must be <= 16000000", operation);
167 if (heights.size() != sampleCount)
168 throw eve::Exception("%s: heights size must equal countX * countZ", operation);
169 if (!(cellSizeX > 0.f) || !(cellSizeZ > 0.f) || !std::isfinite(cellSizeX) ||
170 !std::isfinite(cellSizeZ))
171 throw eve::Exception("%s: cell sizes must be finite and > 0", operation);
172 if (!std::isfinite(globalMin) || !std::isfinite(globalMax) || globalMin > globalMax)
173 throw eve::Exception("%s: global height range must be finite and ordered", operation);
174 for (float height : heights) {
175 if (!std::isfinite(height) || height < globalMin || height > globalMax)
176 throw eve::Exception("%s: every height must be finite and inside global range",
177 operation);
178 }
179 std::vector<float> mutableHeights = heights;
180 b3HeightFieldDef def{};
181 def.heights = mutableHeights.data();
182 def.scale = {cellSizeX, 1.f, cellSizeZ};
183 def.countX = countX;
184 def.countZ = countZ;
185 def.globalMinimumHeight = globalMin;
186 def.globalMaximumHeight = globalMax;
187 def.clockwiseWinding = clockwiseWinding;
188 return b3CreateHeightField(&def);
189}

196Body3D::~Body3D() {

200 if (joint && (joint->bodyA_ == this || joint->bodyB_ == this)) {
201 world_->forgetJoint(joint);
202 joint->invalidate();
203 }
204 }
205 // Invalidate shapes that still reference this body.
206 std::vector<Shape3D *> shapes(world_->shapes_.begin(), world_->shapes_.end());
207 for (Shape3D *s : shapes) {
208 if (s && s->getBody() == this) {
209 world_->forgetShape(s);
210 s->invalidate();
211 }
212 }
213 b3Body_SetUserData(bodyId_, nullptr);
214 b3DestroyBody(bodyId_);
215 world_->forgetBody(this);
216 }
217 bodyId_ = {};
218 world_ = nullptr;
219 runtimeHandle_ = PhysicsBodyHandle::invalid();
220}

280void Body3D::setRotation(float qx, float qy, float qz, float qw) {
281 if (!isValid()) return;
282 b3Quat q;
283 q.v = b3Vec3{qx, qy, qz};
284 q.s = qw;
285 float len = std::sqrt(qx * qx + qy * qy + qz * qz + qw * qw);
286 if (len > 1e-8f) {
287 q.v.x /= len;
288 q.v.y /= len;
289 q.v.z /= len;
290 q.s /= len;
291 } else {
292 q = b3Quat_identity;
293 }
294 b3Body_SetTransform(bodyId_, b3Body_GetPosition(bodyId_), q);
295}

594void Body3D::setMassProperties(float mass, float centerX, float centerY, float centerZ,
595 float inertiaXX, float inertiaYY, float inertiaZZ,
596 float inertiaXY, float inertiaXZ, float inertiaYZ) {
597 if (!isValid()) return;
598 constexpr const char *operation = "Body3D.setMassProperties";
599 if (b3Body_GetType(bodyId_) != b3_dynamicBody)
600 throw eve::Exception("%s: body must be dynamic", operation);
601 requireFinite(mass, operation, "mass");
602 requireFinite(centerX, operation, "centerX");
603 requireFinite(centerY, operation, "centerY");
604 requireFinite(centerZ, operation, "centerZ");
605 requireFinite(inertiaXX, operation, "inertiaXX");
606 requireFinite(inertiaYY, operation, "inertiaYY");
607 requireFinite(inertiaZZ, operation, "inertiaZZ");
608 requireFinite(inertiaXY, operation, "inertiaXY");
609 requireFinite(inertiaXZ, operation, "inertiaXZ");
610 requireFinite(inertiaYZ, operation, "inertiaYZ");
611 if (!(mass > 0.f)) throw eve::Exception("%s: mass must be > 0", operation);
612
613 // Sylvester's criterion for a symmetric positive-definite 3x3 tensor.
614 const double xx = inertiaXX, yy = inertiaYY, zz = inertiaZZ;
615 const double xy = inertiaXY, xz = inertiaXZ, yz = inertiaYZ;
616 const double minor2 = xx * yy - xy * xy;
617 const double determinant =
618 xx * (yy * zz - yz * yz) - xy * (xy * zz - yz * xz) +
619 xz * (xy * yz - yy * xz);
620 if (!(xx > 0.0 && minor2 > 0.0 && determinant > 0.0) ||
621 !std::isfinite(minor2) || !std::isfinite(determinant))
622 throw eve::Exception("%s: inertia tensor must be positive definite", operation);
623
624 b3MassData data{};
625 data.mass = mass;
626 data.center = {centerX, centerY, centerZ};
627 data.inertia = {{inertiaXX, inertiaXY, inertiaXZ},
628 {inertiaXY, inertiaYY, inertiaYZ},
629 {inertiaXZ, inertiaYZ, inertiaZZ}};
630 b3Body_SetMassData(bodyId_, data);
631}
