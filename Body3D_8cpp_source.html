32Body3D::~Body3D() {
33 if (isValid() && world_ && world_->isValid()) {
34 std::vector<Joint3D *> joints(world_->joints_.begin(), world_->joints_.end());
35 for (Joint3D *joint : joints) {
36 if (joint && (joint->bodyA_ == this || joint->bodyB_ == this)) {
37 world_->forgetJoint(joint);
38 joint->invalidate();
39 }
40 }
41 // Invalidate shapes that still reference this body.
42 std::vector<Shape3D *> shapes(world_->shapes_.begin(), world_->shapes_.end());
43 for (Shape3D *s : shapes) {
44 if (s && s->getBody() == this) {
45 world_->forgetShape(s);
46 s->invalidate();
47 }
48 }
49 world_->forgetBody(this);
50 }
51 if (backend_) backend_->destroy();
52 if (backend_) backend_->invalidate();
53 world_ = nullptr;
54 runtimeHandle_ = PhysicsBodyHandle::invalid();
55}

65void Body3D::destroy() {
66 if (!isValid() || !world_ || !world_->isValid()) {
67 if (backend_) backend_->destroy();
68 invalidate();
69 return;
70 }
71 std::vector<Joint3D *> joints(world_->joints_.begin(), world_->joints_.end());
72 for (Joint3D *joint : joints) {
73 if (joint && (joint->bodyA_ == this || joint->bodyB_ == this)) {
74 world_->forgetJoint(joint);
75 joint->invalidate();
76 }
77 }
78 std::vector<Shape3D *> shapes(world_->shapes_.begin(), world_->shapes_.end());
79 for (Shape3D *s : shapes) {
80 if (s && s->getBody() == this) {
81 world_->forgetShape(s);
