24std::string writeTextFile(const std::string& path, const std::string& text) {
25 const std::filesystem::path p(path);
26 std::error_code ec;
27 std::filesystem::create_directories(p.parent_path(), ec);
28 std::ofstream out(p, std::ios::binary);
29 if (!out.good()) return "cannot open output file: " + path;
30 out.write(text.data(), static_cast<std::streamsize>(text.size()));
31 out.flush();
32 return out.good() ? std::string() : "failed writing: " + path;
33}

115InspectCapture SceneInspect::capture(const std::string& outDir, const std::string& tag,
116 const std::vector<std::string>& buffers) {
117 InspectCapture result;
118 auto* cap = captureIface();
119 if (!cap) {
120 result.error = "Graphics module not available";
121 return result;
122 }
123
124 bool wantColor = buffers.empty();
125 bool wantDepth = false, wantNormal = false, wantId = false, wantShadow = false;
126 for (const auto& b : buffers) {
127 if (b == "color") wantColor = true;
128 else if (b == "depth") wantDepth = true;
129 else if (b == "normal") wantNormal = true;
130 else if (b == "id") wantId = true;
131 else if (b == "shadow") wantShadow = true;
132 }
133 if (!wantColor) wantColor = true;
134
135 ensureCamera();
136 cap->setReadbackEnabled(true);
137
138 std::string dir = outDir;
139 if (dir.empty()) {
140 dir = cacheDir();
141 if (dir.empty()) dir = defaultCacheDir();
142 }
143 for (char& c : dir)
144 if (c == '\\') c = '/';
145
146 const std::string safeTag = tag.empty() ? "frame" : tag;
147 const std::string pngPath = dir + "/" + safeTag + ".png";
148 const std::string jsonPath = dir + "/" + safeTag + ".json";
149 const std::string depthPngPath = dir + "/" + safeTag + "_depth.png";
150 const std::string normalPngPath = dir + "/" + safeTag + "_normal.png";
151 const std::string idPngPath = dir + "/" + safeTag + "_id.png";
152 const std::string idJsonPath = dir + "/" + safeTag + "_id.json";
153
154 auto captured = cap->savePng(pngPath);
155 if (!captured) {
156 const auto* diag = captured.status().primaryDiagnostic();
157 result.error = diag ? diag->message() : captured.status().describe();
158 return result;
159 }
160 {
161 const eve::CaptureImageInfo info = std::move(captured).value();
162 result.width = info.width;
163 result.height = info.height;
164 }
165
166 const std::string geoJson = visibleEntitiesJson();
167 if (const std::string werr = writeTextFile(jsonPath, geoJson); !werr.empty()) {
168 result.error = werr;
169 return result;
170 }
171 result.entityCount = 0;
172 const auto pos = geoJson.find("\"count\":");
173 if (pos != std::string::npos) result.entityCount = std::atoi(geoJson.c_str() + pos + 9);
174
175 if (wantDepth) {
176 if (cap->gbufferPng("depth", depthPngPath))
177 result.depthPngPath = depthPngPath;
178 else
179 result.unsupported.push_back("depth");
180 }
181 if (wantNormal) {
182 if (cap->gbufferPng("normal", normalPngPath))
183 result.normalPngPath = normalPngPath;
184 else
185 result.unsupported.push_back("normal");
186 }
187 if (wantShadow) result.unsupported.push_back("shadow");
188
189 if (wantId) {
190 int iw = 0, ih = 0;
191 bool ok = false;
192 const std::string idJson = cap->entityIdMaskJson(&iw, &ih, &ok);
193 if (!ok) {
194 result.unsupported.push_back("id");
195 } else {
196 if (cap->entityIdMaskPng(idPngPath)) result.idPngPath = idPngPath;
197 if (writeTextFile(idJsonPath, idJson).empty()) result.idJsonPath = idJsonPath;
198 }
199 }
200
201 result.ok = true;
202 result.pngPath = pngPath;
203 result.jsonPath = jsonPath;
204 lastPngPath_ = pngPath;
205 lastJsonPath_ = jsonPath;
206 return result;
207}
