24std::string writeTextFile(const std::string& path, const std::string& text) {
25 const std::filesystem::path p(path);
26 std::error_code ec;
27 std::filesystem::create_directories(p.parent_path(), ec);
28 std::ofstream out(p, std::ios::binary);
29 if (!out.good()) return "cannot open output file: " + path;
30 out.write(text.data(), static_cast<std::streamsize>(text.size()));
31 out.flush();
32 return out.good() ? std::string() : "failed writing: " + path;
33}

115InspectCapture SceneInspect::capture(const std::string& outDir, const std::string& tag,
116 const std::vector<std::string>& buffers) {
117 InspectCapture result;
118 auto* cap = captureIface();
119 if (!cap) {
120 result.error = "Graphics module not available";
121 return result;
122 }
123
124 bool wantColor = buffers.empty();
125 bool wantDepth = false, wantNormal = false, wantId = false, wantShadow = false;
126 for (const auto& b : buffers) {
127 if (b == "color") wantColor = true;
128 else if (b == "depth") wantDepth = true;
129 else if (b == "normal") wantNormal = true;
130 else if (b == "id") wantId = true;
131 else if (b == "shadow") wantShadow = true;
132 }
133 if (!wantColor) wantColor = true;
134
135 ensureCamera();
136 cap->setReadbackEnabled(true);
137
138 std::string dir = outDir;
139 if (dir.empty()) {
140 dir = cacheDir();
141 if (dir.empty()) dir = defaultCacheDir();
142 }
143 for (char& c : dir)
144 if (c == '\\') c = '/';
145
146 const std::string safeTag = tag.empty() ? "frame" : tag;
147 const std::string pngPath = dir + "/" + safeTag + ".png";
148 const std::string jsonPath = dir + "/" + safeTag + ".json";
149 const std::string depthPngPath = dir + "/" + safeTag + "_depth.png";
150 const std::string normalPngPath = dir + "/" + safeTag + "_normal.png";
151 const std::string idPngPath = dir + "/" + safeTag + "_id.png";
152 const std::string idJsonPath = dir + "/" + safeTag + "_id.json";
153
154 std::string err;
155 if (!cap->savePng(pngPath, &result.width, &result.height, &err)) {
156 result.error = err;
157 return result;
158 }
159
160 const std::string geoJson = visibleEntitiesJson();
161 if (const std::string werr = writeTextFile(jsonPath, geoJson); !werr.empty()) {
162 result.error = werr;
163 return result;
164 }
165 result.entityCount = 0;
166 const auto pos = geoJson.find("\"count\":");
167 if (pos != std::string::npos) result.entityCount = std::atoi(geoJson.c_str() + pos + 9);
168
169 if (wantDepth) {
170 std::string berr;
171 if (cap->gbufferPng("depth", depthPngPath, &berr))
172 result.depthPngPath = depthPngPath;
173 else
174 result.unsupported.push_back("depth");
175 }
176 if (wantNormal) {
177 std::string berr;
178 if (cap->gbufferPng("normal", normalPngPath, &berr))
179 result.normalPngPath = normalPngPath;
180 else
181 result.unsupported.push_back("normal");
182 }
183 if (wantShadow) result.unsupported.push_back("shadow");
184
185 if (wantId) {
186 int iw = 0, ih = 0;
187 bool ok = false;
188 const std::string idJson = cap->entityIdMaskJson(&iw, &ih, &ok);
189 if (!ok) {
190 result.unsupported.push_back("id");
191 } else {
192 std::string perr;
193 if (cap->entityIdMaskPng(idPngPath, &perr)) result.idPngPath = idPngPath;
194 if (writeTextFile(idJsonPath, idJson).empty()) result.idJsonPath = idJsonPath;
195 }
196 }
197
198 result.ok = true;
199 result.pngPath = pngPath;
200 result.jsonPath = jsonPath;
201 lastPngPath_ = pngPath;
202 lastJsonPath_ = jsonPath;
203 return result;
204}
