52 static std::vector<PhysicsPlacementCollider> selectedColliders(const PhysicsPlacementObject& object,
53 PhysicsPlacementColliderPolicy policy) {
54 std::vector<PhysicsPlacementCollider> values = object.colliders;
55 if (values.empty()) {
56 PhysicsPlacementCollider legacy;
57 legacy.shape = object.shape;
58 legacy.halfExtentX = object.halfExtentX;
59 legacy.halfExtentY = object.halfExtentY;
60 legacy.halfExtentZ = object.halfExtentZ;
61 legacy.convexVertices = object.convexVertices;
62 legacy.convexMaxVertices = object.convexMaxVertices;
63 values.push_back(std::move(legacy));
64 }
65 const bool hasExisting = std::any_of(values.begin(), values.end(), [](const auto& value) {
66 return value.source == PhysicsPlacementColliderSource::Existing;
67 });
68 values.erase(std::remove_if(values.begin(), values.end(),
69 [&](const auto& value) {
70 switch (policy) {
71 case PhysicsPlacementColliderPolicy::ExistingOnly:
72 return value.source != PhysicsPlacementColliderSource::Existing;
73 case PhysicsPlacementColliderPolicy::GeneratedOnly:
74 return value.source != PhysicsPlacementColliderSource::Generated;
75 case PhysicsPlacementColliderPolicy::GeneratedWhenMissing:
76 return hasExisting
77 ? value.source != PhysicsPlacementColliderSource::Existing
78 : value.source != PhysicsPlacementColliderSource::Generated;
79 case PhysicsPlacementColliderPolicy::ExistingAndGenerated: return false;
80 }
81 return true;
82 }),
83 values.end());
84 return values;
85 }

87 static std::unique_ptr<physics::Shape3D> createShape(physics::Body3D& body,
88 const PhysicsPlacementCollider& collider,
89 const glm::dvec3& scale) {
90 const float width = static_cast<float>(2.0 * collider.halfExtentX * std::abs(scale.x));
91 const float height = static_cast<float>(2.0 * collider.halfExtentY * std::abs(scale.y));
92 const float depth = static_cast<float>(2.0 * collider.halfExtentZ * std::abs(scale.z));
93 physics::Shape3D* raw = nullptr;
94 if (collider.shape == PhysicsPlacementShape::Sphere)
95 raw = body.newSphereShape(std::max({width, height, depth}) * 0.5f, 1.0f, 0.5f, 0.0f);
96 else if (collider.shape == PhysicsPlacementShape::Capsule)
97 raw = body.newCapsuleShape(height, std::max(width, depth) * 0.5f, 1.0f, 0.5f, 0.0f);
98 else if (collider.shape == PhysicsPlacementShape::ConvexHull) {
99 auto vertices = collider.convexVertices;
100 for (std::size_t index = 0; index < vertices.size(); index += 3) {
101 vertices[index] *= static_cast<float>(scale.x);
102 vertices[index + 1] *= static_cast<float>(scale.y);
103 vertices[index + 2] *= static_cast<float>(scale.z);
104 }
105 raw = body.newConvexHullShape(vertices, collider.convexMaxVertices, 1.0f, 0.5f, 0.0f);
106 } else
107 raw = body.newBoxShape(width, height, depth, 1.0f, 0.5f, 0.0f);
108 std::unique_ptr<physics::Shape3D> shape(raw);
109 if (shape) {
110 const auto rotation = glm::quat(glm::vec3(static_cast<float>(collider.localRotationX),
111 static_cast<float>(collider.localRotationY),
112 static_cast<float>(collider.localRotationZ)));
113 shape->setLocalTransform(
114 static_cast<float>(collider.localX * scale.x), static_cast<float>(collider.localY * scale.y),
115 static_cast<float>(collider.localZ * scale.z), rotation.x, rotation.y, rotation.z, rotation.w);
116 }
117 return shape;
118 }

122 editing::Result<void> build() {
123 try {
124 if (request.settings.mode == PhysicsPlacementMode::Fall)
125 world.setGravity(static_cast<float>(request.settings.gravityX),
126 static_cast<float>(request.settings.gravityY),
127 static_cast<float>(request.settings.gravityZ));
128 world.setContinuousCollisionEnabled(true);
129 world.setMaximumLinearSpeed(static_cast<float>(request.settings.maximumLinearSpeed));
130 world.setContactTuning(
131 static_cast<float>(request.settings.softCollision ? request.settings.contactHertz
132 : std::max(120.0, request.settings.contactHertz)),
133 static_cast<float>(request.settings.softCollision ? request.settings.contactDampingRatio : 1.0),
134 static_cast<float>(request.settings.softCollision ? request.settings.maximumPushOutSpeed
135 : std::max(request.settings.maximumLinearSpeed,
136 request.settings.maximumPushOutSpeed)));
137 glm::dvec3 centroid(0.0);
138 std::size_t selectedCount = 0;
139 for (const auto& object : request.objects) {
140 const auto& transform = object.transform;
141 auto* rawBody = world.newBody(object.selected ? "dynamic" : "static", static_cast<float>(transform.x),
142 static_cast<float>(transform.y), static_cast<float>(transform.z));
143 if (!rawBody)
144 return placementError<void>(editing::Status::Failed, "scene.physics-placement.body-create",
145 "Box3D could not create a placement body");
146 RuntimeObject runtime;
147 runtime.admitted = object;
148 runtime.appliedScale = {transform.scaleX, transform.scaleY, transform.scaleZ};
149 runtime.body.reset(rawBody);
150 const glm::quat rotation = rotationOf(transform);
151 runtime.body->setRotation(rotation.x, rotation.y, rotation.z, rotation.w);
152 for (const auto& collider : selectedColliders(object, request.settings.colliderPolicy)) {
153 auto shape = createShape(*runtime.body, collider, runtime.appliedScale);
154 if (shape) runtime.shapes.push_back(std::move(shape));
155 }
156 if (runtime.shapes.empty())
157 return placementError<void>(editing::Status::Failed, "scene.physics-placement.shape-create",
158 "Collider policy left the placement object without a usable collider");
159 if (object.selected) {
160 const bool dropping = request.settings.mode == PhysicsPlacementMode::Fall;
161 runtime.body->setGravityScale(dropping ? 1.0f : 0.0f);
162 runtime.body->setLinearDamping(static_cast<float>(dropping ? 0.6 : request.settings.linearDamping));
163 runtime.body->setAngularDamping(
164 static_cast<float>(dropping ? 0.8 : request.settings.angularDamping));
165 runtime.body->setMotionLocks(request.settings.freezePositionX, request.settings.freezePositionY,
166 request.settings.freezePositionZ, request.settings.freezeRotationX,
167 request.settings.freezeRotationY, request.settings.freezeRotationZ);
168 centroid += glm::dvec3(transform.x, transform.y, transform.z);
169 if ((request.primaryObject.empty() && selectedCount == 0) ||
170 request.primaryObject == object.object) {
171 initialHandleRotation = handleRotation = rotation;
172 initialHandleScale = handleScale = runtime.appliedScale;
173 }
174 ++selectedCount;
175 }
176 objects.push_back(std::move(runtime));
177 }
178 centroid /= static_cast<double>(selectedCount);
179 initialHandle = handle = centroid;
180 if (request.settings.preserveSelectionLayout) {
181 for (std::size_t left = 0; left < objects.size(); ++left) {
182 if (!objects[left].admitted.selected) continue;
183 for (std::size_t right = left + 1; right < objects.size(); ++right) {
184 if (objects[right].admitted.selected)
185 world.setBodyPairCollisionEnabled(objects[left].body.get(), objects[right].body.get(),
186 false);
187 }
188 }
189 }
190 return editing::applied<void>();
191 } catch (const std::exception& error) {
192 return placementError<void>(editing::Status::Failed, "scene.physics-placement.build",
193 std::string("Could not build placement preview: ") + error.what());
194 }
195 }
196
197 PhysicsPlacementRequest request;
198 physics::World3D world;
199 std::vector<RuntimeObject> objects;
200 glm::dvec3 initialHandle{0.0};
201 glm::dvec3 handle{0.0};
202 glm::quat initialHandleRotation{1.0f, 0.0f, 0.0f, 0.0f};
203 glm::quat handleRotation{1.0f, 0.0f, 0.0f, 0.0f};
204 glm::dvec3 initialHandleScale{1.0};
205 glm::dvec3 handleScale{1.0};
206 bool surfaceAligned = false;
207 glm::vec3 surfaceNormal{0.0f, 1.0f, 0.0f};
208 std::uint64_t tick = 0;
209};

214editing::Result<std::unique_ptr<ScenePhysicsPlacementBackend>> ScenePhysicsPlacementBackend::create(
215 PhysicsPlacementRequest request) {
216 const auto admittedByPolicy = [&](const PhysicsPlacementObject& object) {
217 if (request.admission.excludeDisabled && !object.enabled) return false;
218 if (request.admission.excludeLocked && object.locked) return false;
219 if ((object.layerBits & request.admission.includedLayerBits) == 0 ||
220 (object.layerBits & request.admission.excludedLayerBits) != 0)
221 return false;
222 for (const auto& required : request.admission.requiredTags)
223 if (std::find(object.tags.begin(), object.tags.end(), required) == object.tags.end()) return false;
224 if (request.admission.useBounds) {
225 const auto& value = object.transform;
226 if (value.x < request.admission.minimumX || value.y < request.admission.minimumY ||
227 value.z < request.admission.minimumZ || value.x > request.admission.maximumX ||
228 value.y > request.admission.maximumY || value.z > request.admission.maximumZ)
229 return false;
230 }
231 return true;
232 };
233 for (const auto& object : request.objects)
234 if (object.selected && !admittedByPolicy(object))
235 return placementError<std::unique_ptr<ScenePhysicsPlacementBackend>>(
236 editing::Status::Rejected, "scene.physics-placement.selection-filtered",
237 "Admission policy must not exclude a selected placement object");
238 request.objects.erase(std::remove_if(request.objects.begin(), request.objects.end(),
239 [&](const auto& object) { return !admittedByPolicy(object); }),
240 request.objects.end());
241 if (request.objects.empty())
242 return placementError<std::unique_ptr<ScenePhysicsPlacementBackend>>(
243 editing::Status::Rejected, "scene.physics-placement.empty", "Placement requires at least one object");
244 if (request.settings.subStepCount <= 0 || request.settings.subStepCount > 64 ||
245 !finite(request.settings.maximumLinearSpeed) || request.settings.maximumLinearSpeed <= 0.0 ||
246 !finite(request.settings.gravityX) || !finite(request.settings.gravityY) ||
247 !finite(request.settings.gravityZ) || !finite(request.settings.linearDamping) ||
248 request.settings.linearDamping < 0.0 || !finite(request.settings.angularDamping) ||
249 request.settings.angularDamping < 0.0 || !finite(request.settings.contactHertz) ||
250 request.settings.contactHertz <= 0.0 || !finite(request.settings.contactDampingRatio) ||
251 request.settings.contactDampingRatio < 0.0 || !finite(request.settings.maximumPushOutSpeed) ||
252 request.settings.maximumPushOutSpeed < 0.0 || !finite(request.settings.maximumAngularSpeed) ||
253 request.settings.maximumAngularSpeed <= 0.0 || !finite(request.settings.minimumBoundsSpeedFactor) ||
254 request.settings.minimumBoundsSpeedFactor <= 0.0 || !finite(request.settings.maximumBoundsSpeedFactor) ||
255 request.settings.maximumBoundsSpeedFactor < request.settings.minimumBoundsSpeedFactor ||
256 !finite(request.settings.teleportDistance) || request.settings.teleportDistance < 0.0)
257 return placementError<std::unique_ptr<ScenePhysicsPlacementBackend>>(
258 editing::Status::Rejected, "scene.physics-placement.settings",
259 "Placement settings must be finite and use a substep count in [1, 64]");
260 std::set<editing::ObjectId> ids;
261 std::size_t selectedCount = 0;
262 for (const auto& object : request.objects) {
263 if (object.shape == PhysicsPlacementShape::Auto && object.colliders.empty())
264 return placementError<std::unique_ptr<ScenePhysicsPlacementBackend>>(
265 editing::Status::Rejected, "scene.physics-placement.unresolved-shape",
266 "Automatic placement shapes must be resolved by a scene editor session");
267 const bool validHull = object.shape != PhysicsPlacementShape::ConvexHull || !object.colliders.empty() ||
268 (object.convexVertices.size() >= 12 && object.convexVertices.size() % 3 == 0 &&
269 object.convexMaxVertices >= 4 && object.convexMaxVertices <= 254 &&
270 std::all_of(object.convexVertices.begin(), object.convexVertices.end(),
271 [](float value) { return std::isfinite(value); }));
272 const auto colliders = Impl::selectedColliders(object, request.settings.colliderPolicy);
273 const bool validColliders =
274 !colliders.empty() && std::all_of(colliders.begin(), colliders.end(), [](const auto& collider) {
275 const bool validGeometry = finite(collider.halfExtentX) && finite(collider.halfExtentY) &&
276 finite(collider.halfExtentZ) && collider.halfExtentX > 0.0 &&
277 collider.halfExtentY > 0.0 && collider.halfExtentZ > 0.0;
278 const bool validPose = finite(collider.localX) && finite(collider.localY) && finite(collider.localZ) &&
279 finite(collider.localRotationX) && finite(collider.localRotationY) &&
280 finite(collider.localRotationZ);
281 const bool hull = collider.shape != PhysicsPlacementShape::ConvexHull ||
282 (collider.convexVertices.size() >= 12 && collider.convexVertices.size() % 3 == 0 &&
283 collider.convexMaxVertices >= 4 && collider.convexMaxVertices <= 254 &&
284 std::all_of(collider.convexVertices.begin(), collider.convexVertices.end(),
285 [](float value) { return std::isfinite(value); }));
286 return collider.shape != PhysicsPlacementShape::Auto && validGeometry && validPose && hull;
287 });
288 if (object.object.empty() || !ids.insert(object.object).second || !finiteTransform(object.transform) ||
289 !finite(object.halfExtentX) || !finite(object.halfExtentY) || !finite(object.halfExtentZ) ||
290 object.halfExtentX <= 0.0 || object.halfExtentY <= 0.0 || object.halfExtentZ <= 0.0 ||
291 object.transform.scaleX == 0.0 || object.transform.scaleY == 0.0 || object.transform.scaleZ == 0.0 ||
292 !validHull || !validColliders)
293 return placementError<std::unique_ptr<ScenePhysicsPlacementBackend>>(
294 editing::Status::Rejected, "scene.physics-placement.object",
295 "Placement objects require unique ids, finite transforms, and positive nonzero bounds");
296 if (object.selected) ++selectedCount;
297 }
298 if (selectedCount == 0)
299 return placementError<std::unique_ptr<ScenePhysicsPlacementBackend>>(
300 editing::Status::Rejected, "scene.physics-placement.selection",
301 "Placement requires at least one selected object");
302 if (!request.primaryObject.empty()) {
303 const auto primary = std::find_if(request.objects.begin(), request.objects.end(), [&](const auto& object) {
304 return object.object == request.primaryObject && object.selected;
305 });
306 if (primary == request.objects.end())
307 return placementError<std::unique_ptr<ScenePhysicsPlacementBackend>>(
308 editing::Status::Rejected, "scene.physics-placement.primary",
309 "Placement primary object must belong to the selected objects");
310 }
311 auto impl = std::make_unique<Impl>(std::move(request));
312 auto built = impl->build();
313 if (!built.ok()) return editing::Result<std::unique_ptr<ScenePhysicsPlacementBackend>>::failure(built.status());
314 return editing::applied<std::unique_ptr<ScenePhysicsPlacementBackend>>(
315 std::unique_ptr<ScenePhysicsPlacementBackend>(new ScenePhysicsPlacementBackend(std::move(impl))));
316}

355editing::Result<void> ScenePhysicsPlacementBackend::alignHandleToSurface(double fromX, double fromY, double fromZ,
356 double toX, double toY, double toZ,
357 double offset) {
358 if (!finite(fromX) || !finite(fromY) || !finite(fromZ) || !finite(toX) || !finite(toY) || !finite(toZ) ||
359 !finite(offset) || (fromX == toX && fromY == toY && fromZ == toZ))
360 return placementError<void>(editing::Status::Rejected, "scene.physics-placement.surface-ray",
361 "Surface alignment requires a finite, nonzero ray and offset");
362 try {
363 impl_->world.rayCastAll(static_cast<float>(fromX), static_cast<float>(fromY), static_cast<float>(fromZ),
364 static_cast<float>(toX), static_cast<float>(toY), static_cast<float>(toZ), 256);
365 std::set<int> selectedBodies;
366 for (const auto& object : impl_->objects)
367 if (object.admitted.selected) selectedBodies.insert(object.body->getId());
368 int hit = -1;
369 for (int index = 0; index < impl_->world.getRayResultCount(); ++index) {
370 if (!selectedBodies.contains(impl_->world.getRayResultBodyId(index))) {
371 hit = index;
372 break;
373 }
374 }
375 if (hit < 0)
376 return placementError<void>(editing::Status::NotFound, "scene.physics-placement.surface-miss",
377 "Surface alignment ray did not hit an admitted static object");
378 glm::vec3 normal(impl_->world.getRayResultNormalX(hit), impl_->world.getRayResultNormalY(hit),
379 impl_->world.getRayResultNormalZ(hit));
380 if (glm::dot(normal, normal) < 1.0e-8f)
381 return placementError<void>(editing::Status::Failed, "scene.physics-placement.surface-normal",
382 "Surface alignment hit did not provide a usable normal");
383 normal = glm::normalize(normal);
384 const glm::vec3 initialUp = impl_->initialHandleRotation * glm::vec3(0.0f, 1.0f, 0.0f);
385 const glm::quat delta = glm::rotation(glm::normalize(initialUp), normal);
386 impl_->handleRotation = glm::normalize(delta * impl_->initialHandleRotation);
387 float minimumProjection = 0.0f;
388 bool hasProjection = false;
389 for (const auto& object : impl_->objects) {
390 if (!object.admitted.selected) continue;
391 const auto& transform = object.admitted.transform;
392 const glm::vec3 groupOffset = delta * glm::vec3(static_cast<float>(transform.x - impl_->initialHandle.x),
393 static_cast<float>(transform.y - impl_->initialHandle.y),
394 static_cast<float>(transform.z - impl_->initialHandle.z));
395 const glm::quat objectRotation = glm::normalize(delta * rotationOf(transform));
396 const glm::vec3 half(static_cast<float>(object.admitted.halfExtentX * std::abs(transform.scaleX)),
397 static_cast<float>(object.admitted.halfExtentY * std::abs(transform.scaleY)),
398 static_cast<float>(object.admitted.halfExtentZ * std::abs(transform.scaleZ)));
399 for (int mask = 0; mask < 8; ++mask) {
400 const glm::vec3 corner((mask & 1) ? half.x : -half.x, (mask & 2) ? half.y : -half.y,
401 (mask & 4) ? half.z : -half.z);
402 const float projection = glm::dot(groupOffset + objectRotation * corner, normal);
403 if (!hasProjection || projection < minimumProjection) minimumProjection = projection;
404 hasProjection = true;
405 }
406 }
407 const glm::vec3 hitPoint(impl_->world.getRayResultX(hit), impl_->world.getRayResultY(hit),
408 impl_->world.getRayResultZ(hit));
409 const glm::vec3 target = hitPoint + normal * (static_cast<float>(offset) - minimumProjection);
410 impl_->handle = glm::dvec3(target);
411 impl_->surfaceAligned = true;
412 impl_->surfaceNormal = normal;
413 return editing::applied<void>();
414 } catch (const std::exception& error) {
415 return placementError<void>(editing::Status::Failed, "scene.physics-placement.surface-align",
416 std::string("Surface alignment failed: ") + error.what());
417 }

420editing::Result<void> ScenePhysicsPlacementBackend::step(std::uint64_t tick, double fixedDelta) {
421 if (tick <= impl_->tick || !finite(fixedDelta) || fixedDelta <= 0.0 || fixedDelta > 1.0)
422 return placementError<void>(editing::Status::Rejected, "scene.physics-placement.step",
423 "Placement ticks must increase and delta must be within (0, 1]");
424 try {
425 if (impl_->request.settings.mode != PhysicsPlacementMode::Fall) {
426 const glm::quat delta = glm::normalize(impl_->handleRotation * glm::inverse(impl_->initialHandleRotation));
427 const glm::dvec3 scaleRatio = impl_->handleScale / impl_->initialHandleScale;
428 double requiredLinearSpeed = impl_->request.settings.maximumLinearSpeed;
429 for (auto& object : impl_->objects) {
430 if (!object.admitted.selected) continue;
431 const auto& transform = object.admitted.transform;
432 const glm::dvec3 desiredScale(transform.scaleX * scaleRatio.x, transform.scaleY * scaleRatio.y,
433 transform.scaleZ * scaleRatio.z);
434 if (desiredScale != object.appliedScale) {
435 const auto colliders =
436 Impl::selectedColliders(object.admitted, impl_->request.settings.colliderPolicy);
437 for (std::size_t index = 0; index < colliders.size(); ++index) {
438 const auto& collider = colliders[index];
439 const float width = static_cast<float>(2.0 * collider.halfExtentX * std::abs(desiredScale.x));
440 const float height = static_cast<float>(2.0 * collider.halfExtentY * std::abs(desiredScale.y));
441 const float depth = static_cast<float>(2.0 * collider.halfExtentZ * std::abs(desiredScale.z));
442 if (collider.shape == PhysicsPlacementShape::Sphere)
443 object.shapes[index]->setSphereRadius(std::max({width, height, depth}) * 0.5f);
444 else if (collider.shape == PhysicsPlacementShape::Capsule)
445 object.shapes[index]->setCapsuleSize(height, std::max(width, depth) * 0.5f);
446 else if (collider.shape == PhysicsPlacementShape::ConvexHull) {
447 auto vertices = collider.convexVertices;
448 for (std::size_t vertex = 0; vertex < vertices.size(); vertex += 3) {
449 vertices[vertex] *= static_cast<float>(desiredScale.x);
450 vertices[vertex + 1] *= static_cast<float>(desiredScale.y);
451 vertices[vertex + 2] *= static_cast<float>(desiredScale.z);
452 }
453 object.shapes[index]->setConvexHullVertices(vertices, collider.convexMaxVertices);
454 } else
455 object.shapes[index]->setBoxSize(width, height, depth);
456 const auto localRotation = glm::quat(glm::vec3(static_cast<float>(collider.localRotationX),
457 static_cast<float>(collider.localRotationY),
458 static_cast<float>(collider.localRotationZ)));
459 object.shapes[index]->setLocalTransform(static_cast<float>(collider.localX * desiredScale.x),
460 static_cast<float>(collider.localY * desiredScale.y),
461 static_cast<float>(collider.localZ * desiredScale.z),
462 localRotation.x, localRotation.y, localRotation.z,
463 localRotation.w);
464 }
465 object.appliedScale = desiredScale;
466 }
467 const glm::dvec3 unrotatedOffset((transform.x - impl_->initialHandle.x) * scaleRatio.x,
468 (transform.y - impl_->initialHandle.y) * scaleRatio.y,
469 (transform.z - impl_->initialHandle.z) * scaleRatio.z);
470 const glm::vec3 offset = delta * glm::vec3(unrotatedOffset);
471 const glm::quat rotation = glm::normalize(delta * rotationOf(transform));
472 glm::dvec3 target(static_cast<double>(offset.x) + impl_->handle.x,
473 static_cast<double>(offset.y) + impl_->handle.y,
474 static_cast<double>(offset.z) + impl_->handle.z);
475 glm::quat targetRotation = rotation;
476 if (impl_->request.settings.mode == PhysicsPlacementMode::Rotate)
477 target = glm::dvec3(offset) + impl_->initialHandle;
478 if (impl_->request.settings.mode == PhysicsPlacementMode::Point) {
479 target = {transform.x, transform.y, transform.z};
480 const glm::vec3 direction = glm::vec3(impl_->handle - target);
481 if (glm::dot(direction, direction) > 1.0e-8f) {
482 const glm::vec3 forward = glm::normalize(direction);
483 const glm::vec3 up = std::abs(glm::dot(forward, glm::vec3(0.0f, 1.0f, 0.0f))) > 0.99f
484 ? glm::vec3(0.0f, 0.0f, 1.0f)
485 : glm::vec3(0.0f, 1.0f, 0.0f);
486 targetRotation = glm::quatLookAt(forward, up);
487 }
488 }
489 const glm::dvec3 current(object.body->getX(), object.body->getY(), object.body->getZ());
490 const double bound = std::max({object.admitted.halfExtentX * 2.0, object.admitted.halfExtentY * 2.0,
491 object.admitted.halfExtentZ * 2.0});
492 const double boundFactor = impl_->request.settings.scaleSpeedByBounds
493 ? std::clamp(bound, impl_->request.settings.minimumBoundsSpeedFactor,
494 impl_->request.settings.maximumBoundsSpeedFactor)
495 : 1.0;
496 requiredLinearSpeed =
497 std::max(requiredLinearSpeed, glm::length(target - current) / fixedDelta * 1.1 * boundFactor);
498 if (impl_->request.settings.teleportDistance > 0.0 &&
499 glm::length(target - current) > impl_->request.settings.teleportDistance) {
500 object.body->setPosition(static_cast<float>(target.x), static_cast<float>(target.y),
501 static_cast<float>(target.z));
502 object.body->setRotation(targetRotation.x, targetRotation.y, targetRotation.z, targetRotation.w);
503 object.body->setLinearVelocity(0.0f, 0.0f, 0.0f);
504 object.body->setAngularVelocity(0.0f, 0.0f, 0.0f);
505 continue;
506 }
507 object.body->setTargetTransform(static_cast<float>(target.x), static_cast<float>(target.y),
508 static_cast<float>(target.z), targetRotation.x, targetRotation.y,
509 targetRotation.z, targetRotation.w, static_cast<float>(fixedDelta));
510 }
511 impl_->world.setMaximumLinearSpeed(static_cast<float>(requiredLinearSpeed));
512 }
513 auto duration = eve::Duration::fromSeconds(fixedDelta);
514 if (!duration.ok()) return editing::Result<void>::failure(duration.status());
515 physics::SimulationSettings settings;
516 settings.subStepCount = impl_->request.settings.subStepCount;
517 auto stepped = impl_->world.step({eve::SimulationTick{tick}, duration.value()}, settings);
518 if (!stepped.ok()) return editing::Result<void>::failure(stepped.status());
519 for (auto& object : impl_->objects) {
520 if (!object.admitted.selected) continue;
521 glm::vec3 angular(object.body->getAngularVelocityX(), object.body->getAngularVelocityY(),
522 object.body->getAngularVelocityZ());
523 const float length = glm::length(angular);
524 if (length > impl_->request.settings.maximumAngularSpeed) {
525 angular *= static_cast<float>(impl_->request.settings.maximumAngularSpeed) / length;
526 object.body->setAngularVelocity(angular.x, angular.y, angular.z);
527 }
528 }
529 impl_->tick = tick;
530 return editing::applied<void>();
531 } catch (const std::exception& error) {
532 return placementError<void>(editing::Status::Failed, "scene.physics-placement.advance",
533 std::string("Placement preview failed to advance: ") + error.what());
534 }
535}

556editing::Result<PhysicsPlacementFrame> ScenePhysicsPlacementBackend::placementFrame() const {
557 PhysicsPlacementFrame frame;
558 frame.tick = impl_->tick;
559 frame.colliding = impl_->world.getContactCount() > 0;
560 frame.settled = frame.colliding && frame.tick > 2;
561 frame.surfaceAligned = impl_->surfaceAligned;
562 frame.surfaceNormalX = impl_->surfaceNormal.x;
563 frame.surfaceNormalY = impl_->surfaceNormal.y;
564 frame.surfaceNormalZ = impl_->surfaceNormal.z;
565 frame.objects.reserve(impl_->objects.size());
566 for (const auto& object : impl_->objects) {
567 PhysicsPlacementObject captured = object.admitted;
568 captured.transform.x = object.body->getX();
569 captured.transform.y = object.body->getY();
570 captured.transform.z = object.body->getZ();
571 const glm::quat rotation(object.body->getRotW(), object.body->getRotX(), object.body->getRotY(),
572 object.body->getRotZ());
573 const glm::vec3 euler = glm::eulerAngles(rotation);
574 captured.transform.rotationX = euler.x;
575 captured.transform.rotationY = euler.y;
576 captured.transform.rotationZ = euler.z;
577 captured.transform.scaleX = object.appliedScale.x;
578 captured.transform.scaleY = object.appliedScale.y;
579 captured.transform.scaleZ = object.appliedScale.z;
580 if (captured.selected && (std::abs(object.body->getLinearVelocityX()) > 0.05f ||
581 std::abs(object.body->getLinearVelocityY()) > 0.05f ||
582 std::abs(object.body->getLinearVelocityZ()) > 0.05f ||
583 std::abs(object.body->getAngularVelocityX()) > 0.05f ||
584 std::abs(object.body->getAngularVelocityY()) > 0.05f ||
585 std::abs(object.body->getAngularVelocityZ()) > 0.05f))
586 frame.settled = false;
587 frame.objects.push_back(std::move(captured));
588 }
589 return editing::applied<PhysicsPlacementFrame>(std::move(frame));
590}
