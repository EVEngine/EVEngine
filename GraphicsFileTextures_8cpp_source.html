18ResultRef<Texture> Graphics::loadTexture(const std::string& filename) {
19 try {
20 auto requested = requestFileTexture(filename);
21 if (!requested.ok()) return ResultRef<Texture>::failure(requested.status());
22 auto* texture = &requested.value().get();
23 if (texture->hasDeferredFilePixels()) ensureFileTexturesReady();

27 return ResultRef<Texture>::success(std::ref(*texture));
28 } catch (const std::exception& error) {

30 Diagnostic::error(DiagnosticCode::Failed, error.what(), {}, {}, "graphics.texture.load"));
31 }

33void Graphics::ensureFileTexturesReady() {
34 if (deferredFileTextures_.empty() || realizingFileTextures_) return;
35 realizingFileTextures_ = true;
36 StartupStage stage("graphics: realize file textures");
37 struct Guard {
38 Graphics* g;
39 ~Guard() { g->realizingFileTextures_ = false; }
40 } guard{this};
41
42 std::vector<DeferredFileTexture> pending = std::move(deferredFileTextures_);
43 deferredFileTextures_.clear();
44
45 auto restoreUnrealized = [&]() {
46 for (const auto& item : pending) {
47 if (item.texture && item.texture->hasDeferredFilePixels()) deferredFileTextures_.push_back(item);
48 }
49 };
50
51 auto& resources = eve::ResourceManager::getInstance();
52 for (const auto& item : pending) {
53 auto waited = resources.waitFor(item.key);
54 if (!waited.ok()) {
55 restoreUnrealized();
56 throw eve::Exception("%s", waited.status().describe().c_str());
57 }
58 }
59
60 try {
61 for (const auto& item : pending) {
62 auto waited = resources.waitFor(item.key);
63 if (!waited.ok()) throw eve::Exception("%s", waited.status().describe().c_str());
64 auto* data = dynamic_cast<image::ImageData*>(&waited.value().get());
65 if (!data || !uploadDeferredFileTexture(item.texture, data))
66 throw eve::Exception("newTextureFromFile: GPU upload failed '%s'", item.key.c_str());
67 if (item.texture) item.texture->clearDeferredFilePixels();
68 }
69 } catch (...) {
70 restoreUnrealized();
71 throw;
72 }
73}
