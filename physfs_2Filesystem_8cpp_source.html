49 for (size_t i = 0; i < input.size(); ++i)
50 if (input[i] == '/' || input[i] == '\\') return i;

59std::string normalize(const std::string &input) {
60 std::stringstream out;
61
62 bool seenSep = false, isSep = false;
63 for (size_t i = 0; i < input.size(); ++i) {
64 isSep = (input[i] == EVENGINE_PATH_SEPARATOR[0]);
65 if (!isSep || !seenSep) out << input[i];
66 seenSep = isSep;

125bool Filesystem::setIdentity(std::string ident, bool appendToPath) {
126 if (!PHYSFS_isInit()) return false;
127
128 std::string old_save_path = save_path_full;
129
130 // Store the save directory.
131 save_identity = std::string(ident);
132
133 // Generate the relative path to the game save folder.
134 save_path_relative = std::string(EVENGINE_APPDATA_PREFIX EVENGINE_APPDATA_FOLDER EVENGINE_PATH_SEPARATOR) + save_identity;
135
136 // Generate the full path to the game save folder.
137 save_path_full = std::string(getAppdataDirectory()) + std::string(EVENGINE_PATH_SEPARATOR);
138 if (fused)
139 save_path_full += std::string(EVENGINE_APPDATA_PREFIX) + save_identity;
140 else
141 save_path_full += save_path_relative;
142
143 save_path_full = normalize(save_path_full);
144
145#ifdef EVENGINE_ANDROID
146 if (save_identity == "") save_identity = "unnamed";
147
148 std::string storage_path;
149 if (isAndroidSaveExternal())
150 storage_path = SDL_AndroidGetExternalStoragePath();
151 else
152 storage_path = SDL_AndroidGetInternalStoragePath();
153
154 std::string save_directory = storage_path + "/save";
155
156 save_path_full = storage_path + std::string("/save/") + save_identity;
157
158 if (!android::directoryExists(save_path_full.c_str()) && !android::mkdir(save_path_full.c_str()))
159 SDL_Log("Error: Could not create save directory %s!", save_path_full.c_str());
160#endif
161
162 // We now have something like:
163 // save_identity: game
164 // save_path_relative: ./LOVE/game
165 // save_path_full: C:\Documents and Settings\user\Application Data/LOVE/game
166
167 // We don't want old read-only save paths to accumulate when we set a new
168 // identity.
169 if (!old_save_path.empty()) PHYSFS_unmount(old_save_path.c_str());
170
171 // Try to add the save directory to the search path.
172 // (No error on fail, it means that the path doesn't exist).
173 PHYSFS_mount(save_path_full.c_str(), nullptr, appendToPath);
174
175 // HACK: This forces setupWriteDirectory to be called the next time a file
176 // is opened for writing - otherwise it won't be called at all if it was
177 // already called at least once before.
178 PHYSFS_setWriteDir(nullptr);
179
180 return true;
181}

185bool Filesystem::setSource(std::string source) {
186 if (!PHYSFS_isInit()) return false;
187
188 // Check whether directory is already set.
189 if (!game_source.empty()) return false;
190
191 std::string new_search_path = source;
192
193#ifdef EVENGINE_ANDROID
194 if (!android::createStorageDirectories()) SDL_Log("Error creating storage directories!");
195
196 new_search_path = "";
197
198 PHYSFS_Io *gameLoveIO;
199 bool hasFusedGame = android::checkFusedGame((void **)&gameLoveIO);
200
201 if (hasFusedGame) {
202 if (gameLoveIO) {
203 // Actually we should just be able to mount gameLoveIO, but that's experimental.
204 gameLoveIO->destroy(gameLoveIO);
205 goto oldschool;
206 } else {
207 if (!android::initializeVirtualArchive()) {
208 SDL_Log("Unable to mount AAsset: %s", PHYSFS_getErrorByCode(PHYSFS_getLastErrorCode()));
209 return false;
210 }
211 }
212 } else {
213 oldschool:
214 new_search_path = android::getSelectedGameFile();
215
216 // try mounting first, if that fails, load to memory and mount
217 if (!PHYSFS_mount(new_search_path.c_str(), nullptr, 1)) {
218 // PHYSFS cannot yet mount a zip file inside an .apk
219 SDL_Log("Mounting %s did not work. Loading to memory.", new_search_path.c_str());
220 char * game_archive_ptr = NULL;
221 size_t game_archive_size = 0;
222 if (!android::loadGameArchiveToMemory(new_search_path.c_str(), &game_archive_ptr,
223 &game_archive_size)) {
224 SDL_Log("Failure memory loading archive %s", new_search_path.c_str());
225 return false;
226 }
227 if (!PHYSFS_mountMemory(game_archive_ptr, game_archive_size, android::freeGameArchiveMemory,
228 "archive.zip", "/", 0)) {
229 SDL_Log("Failure mounting in-memory archive.");
230 android::freeGameArchiveMemory(game_archive_ptr);
231 return false;
232 }
233 }
234 }
235#else
236 // Add the directory.
237 if (!PHYSFS_mount(new_search_path.c_str(), nullptr, 1)) return false;
238#endif
239
240 // Save the game source.
241 game_source = new_search_path;
242
243 return true;
244}

248bool Filesystem::setSourceFromMemory(const void* data, size_t size) {
249 if (!PHYSFS_isInit() || !data || size == 0) return false;
250 // Only one source per process.
251 if (!game_source.empty()) return false;
252
253 // PhysFS keeps a pointer to the buffer for the lifetime of the mount, so we
254 // must own a copy that outlives it. Freed in the destructor.
255 void* owned = malloc(size);
256 if (!owned) return false;
257 memcpy(owned, data, size);
258
259 // Mount at "/" so dofile("config.nut") / "main.nut" resolve via PhysFS.
260 if (!PHYSFS_mountMemory(owned, size, nullptr, "game.eve", "/", 1)) {
261 const auto err = PHYSFS_getErrorByCode(PHYSFS_getLastErrorCode());
262 std::cerr << "PhysFS mountMemory failed: " << (err ? err : "unknown error") << std::endl;
263 free(owned);
264 return false;
265 }
266
267 memoryArchive_ = owned;
268 game_source = "game.eve";
269 return true;
270}

272bool Filesystem::setupWriteDirectory() {
273 if (!PHYSFS_isInit()) return false;
274
275 // These must have one be set.
276 if (save_identity.empty() || save_path_full.empty() || save_path_relative.empty()) return false;
277
278 // We need to make sure the write directory is created. To do that, we also
279 // need to make sure all its parent directories are also created.
280 std::string temp_writedir = getDriveRoot(save_path_full);
281 std::string temp_createdir = skipDriveRoot(save_path_full);
282
283 // On some sandboxed platforms, physfs will break when its write directory
284 // is the root of the drive and it tries to create a folder (even if the
285 // folder's path is in a writable location.) If the user's home folder is
286 // in the save path, we'll try starting from there instead.
287 if (save_path_full.find(getUserDirectory()) == 0) {
288 temp_writedir = getUserDirectory();
289 temp_createdir = save_path_full.substr(getUserDirectory().length());
290
291 // Strip leading '/' characters from the path we want to create.
292 size_t startpos = temp_createdir.find_first_not_of('/');
293 if (startpos != std::string::npos) temp_createdir = temp_createdir.substr(startpos);
294 }
295
296 // Set either '/' or the user's home as a writable directory.
297 // (We must create the save folder before mounting it).
298 if (!PHYSFS_setWriteDir(temp_writedir.c_str())) return false;
299
300 // Create the save folder. (We're now "at" either '/' or the user's home).
301 if (!createDirectory(temp_createdir.c_str())) {
302 // Clear the write directory in case of error.
303 PHYSFS_setWriteDir(nullptr);
304 return false;
305 }
306
307 // Set the final write directory.
308 if (!PHYSFS_setWriteDir(save_path_full.c_str())) return false;
309
310 // Add the directory. (Will not be readded if already present).
311 if (!PHYSFS_mount(save_path_full.c_str(), nullptr, 0)) {
312 PHYSFS_setWriteDir(nullptr); // Clear the write directory in case of error.
313 return false;
314 }
315
316 return true;
317}

319bool Filesystem::mount(std::string archive, std::string mountpoint, bool appendToPath) {
320 if (!PHYSFS_isInit() || archive.empty()) return false;
321
322 std::string realPath;
323 std::string sourceBase = getSourceBaseDirectory();
324
325 // Check whether the given archive path is in the list of allowed full paths.
326 auto it = std::find(allowedMountPaths.begin(), allowedMountPaths.end(), archive);
327
328 if (it != allowedMountPaths.end())
329 realPath = *it;
330 else if (isFused() && sourceBase.compare(archive) == 0) {
331 // Special case: if the game is fused and the archive is the source's
332 // base directory, mount it even though it's outside of the save dir.
333 realPath = sourceBase;
334 } else {
335 // Not allowed for safety reasons.
336 if (archive.size() == 0 || archive.find("..") != string::npos || archive == "/") return false;
337
338 const char* realDir = PHYSFS_getRealDir(archive.c_str());
339 if (!realDir) return false;
340
341 realPath = realDir;
342
343 // Always disallow mounting of files inside the game source, since it
344 // won't work anyway if the game source is a zipped .love file.
345 if (realPath.find(game_source) == 0) return false;
346
347 realPath += EVENGINE_PATH_SEPARATOR;
348 realPath += archive;
349 }
350
351 if (realPath.length() == 0) return false;
352
353 return PHYSFS_mount(realPath.c_str(), mountpoint.c_str(), appendToPath) != 0;
354}

367bool Filesystem::mountRealDirectory(std::string realDir, std::string mountpoint, bool appendToPath) {
368 if (!PHYSFS_isInit() || realDir.empty()) return false;
369 if (!isRealDirectory(realDir)) return false;
370 if (mountpoint.empty()) mountpoint = "/";
371
372 // Track the dir so unmountRealDirectory() / the destructor can clean it up,
373 // even though PhysFS itself needs the OS path string to unmount.
374 std::lock_guard<std::mutex> lock(mountMu_);
375 if (std::find(mountedRealDirs_.begin(), mountedRealDirs_.end(), realDir) == mountedRealDirs_.end()) {
376 if (!PHYSFS_mount(realDir.c_str(), mountpoint.c_str(), appendToPath ? 1 : 0)) return false;
377 mountedRealDirs_.push_back(realDir);
378 }
379 return true;
380}

391bool Filesystem::unmount(std::string archive) {
392 if (!PHYSFS_isInit() || archive.empty()) return false;
393
394 auto datait = mountedData.find(archive);
395
396 if (datait != mountedData.end() && PHYSFS_unmount(archive.c_str()) != 0) {
397 mountedData.erase(datait);
398 return true;
399 }
400
401 std::string realPath;
402 std::string sourceBase = getSourceBaseDirectory();
403
404 // Check whether the given archive path is in the list of allowed full paths.
405 auto it = std::find(allowedMountPaths.begin(), allowedMountPaths.end(), archive);
406
407 if (it != allowedMountPaths.end())
408 realPath = *it;
409 else if (isFused() && sourceBase.compare(archive) == 0) {
410 // Special case: if the game is fused and the archive is the source's
411 // base directory, unmount it even though it's outside of the save dir.
412 realPath = sourceBase;
413 } else {
414 // Not allowed for safety reasons.
415 if (archive.size() == 0 || archive.find("..") != string::npos || archive == "/") return false;
416
417 const char* realDir = PHYSFS_getRealDir(archive.c_str());
418 if (!realDir) return false;
419
420 realPath = realDir;
421 realPath += EVENGINE_PATH_SEPARATOR;
422 realPath += archive;
423 }
424
425 const char* mountPoint = PHYSFS_getMountPoint(realPath.c_str());
426 if (!mountPoint) return false;
427
428 return PHYSFS_unmount(realPath.c_str()) != 0;

520std::string Filesystem::getSourceBaseDirectory() const {
521 size_t source_len = game_source.length();
522
523 if (source_len == 0) return "";
524
525 // FIXME: This doesn't take into account parent and current directory
526 // symbols (i.e. '..' and '.')
527#ifdef EVENGINE_WINDOWS
528 // In windows, delimiters can be either '/' or '\'.
529 size_t base_end_pos = game_source.find_last_of("/\\", source_len - 2);
530#else
531 size_t base_end_pos = game_source.find_last_of('/', source_len - 2);
532#endif
533
534 if (base_end_pos == std::string::npos) return "";
535
536 // If the source is in the unix root (aka '/'), we want to keep the '/'.
537 if (base_end_pos == 0) base_end_pos = 1;
538
539 return game_source.substr(0, base_end_pos);
540}

552bool Filesystem::getInfo(std::string filepath, Info &info) const {
553 if (!PHYSFS_isInit()) return false;
554
555 PHYSFS_Stat stat = {};
556 if (!PHYSFS_stat(filepath.c_str(), &stat)) return false;
557
558 info.size = (int64_t)stat.filesize;
559 info.modtime = (int64_t)stat.modtime;
560
561 if (stat.filetype == PHYSFS_FILETYPE_REGULAR)
562 info.type = "file";
563 else if (stat.filetype == PHYSFS_FILETYPE_DIRECTORY)
564 info.type = "directory";
565 else if (stat.filetype == PHYSFS_FILETYPE_SYMLINK)
566 info.type = "symlink";
567 else
568 info.type = "other";
569
570 return true;
571}

657bool looksAbsolute(const std::string &path) {
658 if (path.empty()) return false;
659 if (path[0] == '/' || path[0] == '\\') return true;

666std::string joinPath(const std::string &a, const std::string &b) {
667 if (a.empty()) return b;
668 if (b.empty()) return a;
669 char last = a.back();
670 if (last == '/' || last == '\\') return a + b;
671 return a + "/" + b;
672}

674std::string parentDir(const std::string &path) {
675 auto pos = path.find_last_of("/\\");
676 if (pos == std::string::npos) return ".";
677 if (pos == 0) return "/";

679}

681std::string baseName(const std::string &path) {

687bool hasParentSegment(const std::string &path) {
688 if (path == "..") return true;
689 if (path.size() >= 3 && (path.compare(0, 3, "../") == 0 || path.compare(0, 3, "..\\") == 0))
690 return true;
691 return path.find("/..") != std::string::npos || path.find("\\..") != std::string::npos;
692}

711bool Filesystem::resolveWatchTarget(const std::string &path, std::string &realDir,
712 std::string &filterName, std::string &reportPath) {
713 reportPath = path;
714 filterName.clear();
715 realDir.clear();
716 if (path.empty()) return false;
717
718 // "." / "" are not reliable PhysFS paths — watch the real working directory.
719 if (path == "." || path == "./") {
720 reportPath = ".";
721 realDir = getWorkingDirectory();
722 filterName.clear();
723 return isRealDirectory(realDir);
724 }
725
726 if (looksAbsolute(path)) {
727 if (isRealDirectory(path)) {
728 realDir = path;
729 return true;
730 }
731 realDir = parentDir(path);
732 filterName = baseName(path);
733 return isRealDirectory(realDir);
734 }
735
736 // PhysFS rejects `..`. Resolve parent-relative paths against the real cwd
737 // so `fs.watch("..")` / `hot.watchTree("..")` can see a parent workflow.
738 if (hasParentSegment(path)) {
739 std::error_code ec;
740 auto candidate = std::filesystem::weakly_canonical(
741 pathFromUtf8(getWorkingDirectory()) / pathFromUtf8(path), ec);
742 if (!ec) {
743 const std::string full = pathToUtf8(candidate);
744 if (isRealDirectory(full)) {
745 realDir = full;
746 filterName.clear();
747 return true;
748 }
749 realDir = parentDir(full);
750 filterName = baseName(full);
751 return isRealDirectory(realDir);
752 }
753 }
754
755 Info info{};
756 if (getInfo(path, info)) {
757 std::string root;
758 try {
759 root = getRealDirectory(path);
760 } catch (...) {
761 return false;
762 }
763 std::string full = joinPath(root, path);
764 if (info.type == "directory") {
765 realDir = full;
766 filterName.clear();
767 return isRealDirectory(realDir);
768 }
769 realDir = parentDir(full);
770 filterName = baseName(full);
771 return isRealDirectory(realDir);
772 }
773
774 std::string parent = parentDir(path);
775 if (parent == ".") parent.clear();
776 if (!parent.empty()) {
777 Info pinfo{};
778 if (getInfo(parent, pinfo) && pinfo.type == "directory") {
779 std::string root;
780 try {
781 root = getRealDirectory(parent);
782 } catch (...) {
783 return false;
784 }
785 realDir = joinPath(root, parent);
786 filterName = baseName(path);
787 return isRealDirectory(realDir);
788 }
789 }
790
791 // Virtual path not in VFS yet (e.g. new file): watch under cwd, not save dir.
792 {
793 std::string workingDir = getWorkingDirectory();
794 if (!workingDir.empty() && isRealDirectory(workingDir)) {
795 if (parent.empty()) {
796 realDir = workingDir;
797 filterName = baseName(path);
798 return true;
799 }
800 }
801 }
802
803 std::string save = getSaveDirectory();
804 if (!save.empty() && isRealDirectory(save)) {
805 realDir = save;
806 filterName = baseName(path);
807 return true;
808 }
809 return false;
810}
