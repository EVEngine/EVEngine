45eve::Result<VehicleEffectUpdate> VehicleEffectExecutor::settle(VehicleEffectTarget& target,
46 effects::EffectUpdateSummary lifecycle) const {
47 auto valid = validate(target);
48 if (!valid) return eve::Result<VehicleEffectUpdate>::failure(valid.status());
49 VehicleEffectUpdate result;
50 result.lifecycle = std::move(lifecycle);
51 for (const auto& tick : result.lifecycle.periodicTicks) {
52 if (tick.stackCount == 0 || tick.magnitude < 0.0 || !std::isfinite(tick.magnitude))
53 return eve::Result<VehicleEffectUpdate>::failure(eve::Diagnostic::error(
54 eve::DiagnosticCode::InvalidArgument, "vehicle periodic effect value is invalid", "periodic"));
55 const double amount = tick.magnitude * static_cast<double>(tick.stackCount);
56 if (!std::isfinite(amount))
57 return eve::Result<VehicleEffectUpdate>::failure(eve::Diagnostic::error(
58 eve::DiagnosticCode::InvalidArgument, "vehicle periodic settlement overflows", "periodic"));
59 const auto has = [&tick](const char* tag) {
60 return std::find(tick.tags.begin(), tick.tags.end(), tag) != tick.tags.end();
61 };
62 if (has("vehicle:repair")) {
63 target.hull = std::min(target.maxHull, target.hull + amount);
64 } else if (has("vehicle:disable")) {
65 target.disabled = true;
66 } else {
67 const double damage = amount * zoneMultiplier(target, tick.tags);
68 if (!std::isfinite(damage))
69 return eve::Result<VehicleEffectUpdate>::failure(eve::Diagnostic::error(
70 eve::DiagnosticCode::InvalidArgument, "vehicle armor settlement overflows", "periodic"));
71 target.hull = std::max(0.0, target.hull - damage);
72 if (damage >= target.maxHull * 0.5) {
73 ++target.criticalHits;
74 ++result.criticalHits;
75 }
76 }
77 ++result.settled;
78 }
79 return eve::Result<VehicleEffectUpdate>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
80}

124eve::Result<VehicleEffectUpdate> VehicleEffectAdapter::advance(const eve::SimulationStep& step) {
125 auto candidate = container_.clone();
126 auto lifecycle = effects::EffectExecutor{}.advance(candidate, step);
127 if (!lifecycle) return eve::Result<VehicleEffectUpdate>::failure(lifecycle.status());
128 auto targetCandidate = target_;
129 auto settled = executor_.settle(targetCandidate, std::move(lifecycle).takeValue());
130 if (!settled) return settled;
131 auto result = std::move(settled).takeValue();
132 target_ = std::move(targetCandidate);
133 container_ = std::move(candidate);
134 return eve::Result<VehicleEffectUpdate>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
135}
