64std::vector<eve::graphics::GpuResidentInstanceBucket> scriptResidentBuckets(ssq::Array buckets) {
65 std::vector<eve::graphics::GpuResidentInstanceBucket> result;
66 result.reserve(buckets.size());
67 for (std::size_t i = 0; i < buckets.size(); ++i) {
68 ssq::Table bucket = buckets.get<ssq::Table>(i);
69 const int first = bucket.get<int>("firstInstance");
70 const int count = bucket.get<int>("instanceCount");
71 const int mesh = bucket.get<int>("meshId");
72 const int material = bucket.get<int>("materialId");
73 if (first < 0 || count <= 0 || mesh < 0 || material < 0)
74 throw Exception("Gpgpu.submitResidentInstances: bucket %d has invalid fields", static_cast<int>(i));
75 result.push_back({static_cast<uint32_t>(first), static_cast<uint32_t>(count), static_cast<uint32_t>(mesh),
76 static_cast<uint32_t>(material)});
77 }
78 return result;
79}

81std::string submitResidentInstances(GpuBuffer *buffer, ssq::Array buckets, int instanceCount, int offsetBytes) {
82 if (!buffer || instanceCount <= 0 || offsetBytes < 0 ||
83 offsetBytes % static_cast<int>(eve::kGpuResidentStorageOffsetAlignment) != 0)
84 return "invalid_argument";
85 auto *gfx = currentGraphics();
86 if (!gfx) return "resource_unavailable";
87 const auto nativeBuckets = scriptResidentBuckets(buckets);
88 eve::graphics::GpuResidentInstanceBatch batch;
89 batch.buffer = buffer->residentView();
90 batch.buffer.offsetBytes = static_cast<uint64_t>(offsetBytes);
91 batch.buffer.strideBytes = sizeof(eve::graphics::GpuInstance);
92 batch.buckets = nativeBuckets.data();
93 batch.bucketCount = static_cast<uint32_t>(nativeBuckets.size());
94 batch.instanceCount = static_cast<uint32_t>(instanceCount);
95 return residentSubmitStatusName(gfx->gpuDrivenSubmitResident(batch));
96}

98void writeGpuDrivenInstance(GpuBuffer *buffer, int instanceIndex, ssq::Array model, int meshId, int materialId,
99 int flags, int lodGroupId) {
100 if (!buffer || instanceIndex < 0 || model.size() != 16 || meshId < 0 || materialId < 0 || flags < 0 ||
101 lodGroupId < -1)
102 throw Exception("Gpgpu.writeGpuDrivenInstance: invalid argument");
103 eve::graphics::GpuInstance instance;
104 float *matrix = &instance.model[0][0];
105 for (std::size_t i = 0; i < 16; ++i) matrix[i] = model.get<float>(i);
106 instance.meshId = static_cast<uint32_t>(meshId);
107 instance.materialId = static_cast<uint32_t>(materialId);
108 instance.flags = static_cast<uint32_t>(flags);
109 instance.lodGroupId = lodGroupId < 0 ? eve::graphics::kInvalidGpuDrivenSlot : static_cast<uint32_t>(lodGroupId);
110 const uint64_t offset = static_cast<uint64_t>(instanceIndex) * sizeof(instance);
111 if (offset + sizeof(instance) > static_cast<uint64_t>(buffer->getSize()))
112 throw Exception("Gpgpu.writeGpuDrivenInstance: instance exceeds buffer capacity");
113 buffer->uploadBytes(&instance, sizeof(instance), offset);
114}

207ComputeShader *Gpgpu::newShader(const std::string &source) {
208#ifdef EVENGINE_WEBGPU
209 if (currentGraphicsBackend() != "webgpu")
210 throw Exception("Gpgpu.newShader: requires webgpu Graphics backend");

216 return vulkanNewShaderFromSpirv(compileComputeGlsl(source));

220ComputeShader *Gpgpu::newShaderFromBytecode(const std::string &path) {
221#ifdef EVENGINE_WEBGPU
222 if (currentGraphicsBackend() != "webgpu")
223 throw Exception("Gpgpu.newShaderFromBytecode: requires webgpu Graphics backend");
224 // Load WGSL text from the given path.
225 auto *fs = eve::filesystem::Filesystem::create();
226 if (!fs) throw Exception("Gpgpu.newShaderFromBytecode: no filesystem");
227 std::unique_ptr<eve::filesystem::FileData> file(fs->read(path));
228 if (!file) throw Exception("Gpgpu.newShaderFromBytecode: cannot open '%s'", path.c_str());
229 std::string src(reinterpret_cast<const char *>(file->getData()), file->getSize());
230 return webgpuNewShaderFromWgsl(src);
231#else
232 if (currentGraphicsBackend() != "vulkan")
233 throw Exception("Gpgpu.newShaderFromBytecode: requires vulkan Graphics backend");
234 return vulkanNewShaderFromSpirv(loadSpirvFile(path));
235#endif
236}
