27void sampleColor(const ParticleEmitter::Config &cfg, float t, Color &out) {
28 if (!cfg.colorGradient.empty()) {
29 float r, g, b, a;
30 cfg.colorGradient.sample(t, r, g, b, a);
31 out = Color(r, g, b, a);
32 return;
33 }
34 t = t < 0.f ? 0.f : (t > 1.f ? 1.f : t);
35 out.r = cfg.colorStart.r + (cfg.colorEnd.r - cfg.colorStart.r) * t;
36 out.g = cfg.colorStart.g + (cfg.colorEnd.g - cfg.colorStart.g) * t;
37 out.b = cfg.colorStart.b + (cfg.colorEnd.b - cfg.colorStart.b) * t;
38 out.a = cfg.colorStart.a + (cfg.colorEnd.a - cfg.colorStart.a) * t;
39}

41void sampleScale(const ParticleEmitter::Config &cfg, float t, float &scale) {
42 if (!cfg.sizeCurve.empty()) {
43 scale = cfg.sizeCurve.sample(t, 1.f);
44 return;

46 t = t < 0.f ? 0.f : (t > 1.f ? 1.f : t);
47 scale = cfg.sizeStart + (cfg.sizeEnd - cfg.sizeStart) * t;

50void flipbookUV(const ParticleEmitter::Config &cfg, float frame, float &u0, float &v0, float &u1,
51 float &v1) {
52 const int total = cfg.hframes * cfg.vframes;
53 if (total <= 1 || cfg.hframes <= 0 || cfg.vframes <= 0) {
54 u0 = 0.f;
55 v0 = 0.f;
56 u1 = 1.f;
57 v1 = 1.f;
58 return;
59 }
60 int fi = int(std::floor(frame));
61 fi = ((fi % total) + total) % total;
62 const int col = fi % cfg.hframes;
63 const int row = fi / cfg.hframes;
64 const float iw = 1.f / float(cfg.hframes);
65 const float ih = 1.f / float(cfg.vframes);
66 u0 = float(col) * iw;
67 v0 = float(row) * ih;
68 u1 = u0 + iw;
69 v1 = v0 + ih;
70}

72void appendParticleItem(const ParticleEmitter::Config &cfg, const ParticleEmitter::Draw &draw,
73 const Particle &p, int order, std::vector<graphics::DrawItem2D> &out) {
74 const float t = p.lifetime > 0.f ? 1.f - (p.life / p.lifetime) : 1.f;
75 Color c;
76 sampleColor(cfg, t, c);
77 float scale;
78 sampleScale(cfg, t, scale);
79 scale *= p.size > 0.f ? p.size : 1.f;
80 const float w = cfg.particleW * scale;
81 const float h = cfg.particleH * scale;
82
83 graphics::DrawItem2D item;
84 item.x = p.x - w * 0.5f;
85 item.y = p.y - h * 0.5f;
86 item.w = w;
87 item.h = h;
88 if (cfg.renderMode == "stretched") {
89 // Elongate along the velocity direction (comet / streak style).
90 const float speed = std::sqrt(p.vx * p.vx + p.vy * p.vy);
91 const float len = std::max(w, speed * cfg.stretchFactor);
92 item.w = len;
93 item.rotation = std::atan2(p.vy, p.vx) * kRad2Deg;
94 } else if (cfg.renderMode == "axis") {
95 item.rotation = cfg.renderAxisDegrees;
96 } else {
97 item.rotation = p.rot * kRad2Deg;
98 if (!cfg.rotationCurve.empty()) item.rotation += cfg.rotationCurve.sample(t, 0.f);
99 }
100 item.order = order;
101 item.hasOrder = true;
102 item.color = c;
103 item.layer = draw.layer;
104 item.blend = draw.blend;
105 item.texture = draw.texture;
106 item.normal = cfg.materialMode == "lit" ? draw.normalTexture : nullptr;
107 item.shader = draw.shader;
108 item.canvas = draw.canvas;
109 item.camera = draw.camera;
110 item.receiveLight = cfg.materialMode == "lit";
111 item.litPath = item.receiveLight && item.texture != nullptr && item.normal != nullptr && item.shader == nullptr;
112 item.sceneColorDistortion = cfg.materialMode == "distortion";
113 item.distortionStrength = cfg.distortionStrength;
114 if (draw.texture && (cfg.hframes > 1 || cfg.vframes > 1)) {
115 flipbookUV(cfg, p.frame, item.u0, item.v0, item.u1, item.v1);
116 item.hasUV = true;
117 }
118 out.push_back(item);
119}

121void appendEmitterItems(const ParticleEmitter::Config& cfg, const ParticleEmitter::Sim& sim,
122 const ParticleEmitter::Draw& draw, int& order, std::vector<graphics::DrawItem2D>& out) {
123 if (cfg.sortMode == "none") {
124 for (int i = 0; i < sim.alive; ++i) appendParticleItem(cfg, draw, sim.particles[std::size_t(i)], order++, out);
125 return;
126 }
127
128 std::vector<int> indices(std::size_t(sim.alive));
129 for (int i = 0; i < sim.alive; ++i) indices[std::size_t(i)] = i;
130 auto age = [&](int index) {
131 const Particle& p = sim.particles[std::size_t(index)];
132 return p.lifetime > 0.f ? 1.f - p.life / p.lifetime : 1.f;
133 };
134 if (cfg.sortMode == "oldest") {
135 std::stable_sort(indices.begin(), indices.end(), [&](int a, int b) { return age(a) > age(b); });
136 } else if (cfg.sortMode == "youngest") {
137 std::stable_sort(indices.begin(), indices.end(), [&](int a, int b) { return age(a) < age(b); });
138 } else if (cfg.sortMode == "distance" && draw.camera) {
139 const float cameraX = draw.camera->data()->x;
140 const float cameraY = draw.camera->data()->y;
141 auto distanceSquared = [&](int index) {
142 const Particle& p = sim.particles[std::size_t(index)];
143 const float dx = p.x - cameraX;
144 const float dy = p.y - cameraY;
145 return dx * dx + dy * dy;
146 };
147 std::stable_sort(indices.begin(), indices.end(),
148 [&](int a, int b) { return distanceSquared(a) > distanceSquared(b); });
149 }
150 for (int index : indices) appendParticleItem(cfg, draw, sim.particles[std::size_t(index)], order++, out);
151}

153int appendRibbonItems(const ParticleEmitter::Config& cfg, const ParticleEmitter::Sim& sim,
154 const ParticleEmitter::Draw& draw, int& order, std::vector<graphics::DrawItem2D>& out) {
155 int segments = 0;
156 for (int i = 1; i < sim.alive; ++i) {
157 const Particle& previous = sim.particles[std::size_t(i - 1)];
158 const Particle& current = sim.particles[std::size_t(i)];
159 const float dx = current.x - previous.x;
160 const float dy = current.y - previous.y;
161 const float length = std::sqrt(dx * dx + dy * dy);
162 if (length < cfg.ribbonMinSegmentLength) continue;
163
164 appendParticleItem(cfg, draw, current, order++, out);
165 auto& item = out.back();
166 const float h = item.h * cfg.ribbonWidth;
167 item.x = (previous.x + current.x - length) * 0.5f;
168 item.y = (previous.y + current.y - h) * 0.5f;
169 item.w = length;
170 item.h = h;
171 item.rotation = std::atan2(dy, dx) * kRad2Deg;
172 ++segments;
173 }
174 return segments;
175}

178bool emitterOffscreen(const ParticleEmitter::Config &cfg, const ParticleEmitter::Draw &draw) {
179 auto *cam = draw.camera;
180 if (!cam) return false;
181 if (cfg.cullDistance > 0.f) {
182 const float dx = cfg.x - cam->data()->x;
183 const float dy = cfg.y - cam->data()->y;
184 if (dx * dx + dy * dy > cfg.cullDistance * cfg.cullDistance) return true;
185 }
186 auto *gfx = eve::ModuleManager::getInstance<eve::graphics::Graphics>("Graphics");
187 if (!gfx) return false;
188 const float viewW = draw.canvas ? float(draw.canvas->getWidth()) : float(gfx->getWidth());
189 const float viewH = draw.canvas ? float(draw.canvas->getHeight()) : float(gfx->getHeight());
190 if (viewW <= 0.f || viewH <= 0.f) return false;
191 const float z = cam->data()->zoom > 0.f ? cam->data()->zoom : 1e-4f;
192 const float sx = (cfg.x - cam->data()->x) * z + viewW * 0.5f;
193 const float sy = (cfg.y - cam->data()->y) * z + viewH * 0.5f;
194 const float maxHalf =
195 std::max(cfg.particleW, cfg.particleH) *
196 std::max(std::abs(cfg.sizeStart), std::abs(cfg.sizeEnd)) * 0.5f +
197 1.f;
198 const float margin =
199 cfg.speedMax * cfg.lifeMax + std::max(cfg.areaX, cfg.areaY) + maxHalf;
200 return sx < -margin || sx > viewW + margin || sy < -margin || sy > viewH + margin;
201}

246eve::Result<bool> advanceResidentGpu(graphics::Graphics *gfx, ParticleEmitter *emitter, const eve::SimulationStep &step,
247 bool legacyControls) {
248 auto cfg = emitter->config();
249 auto sim = emitter->sim();
250 auto gpu = emitter->gpuSim();
251 if (!gfx || !gfx->canSubmitGpuParticles()) return eve::Result<bool>::success(false);
252
253 if (gpu->residentHandle == graphics::kInvalidGpuParticleHandle) {
254 gpu->residentHandle = gfx->createGpuParticleEmitter(std::uint32_t(sim->particles.size()));
255 if (gpu->residentHandle == graphics::kInvalidGpuParticleHandle) return eve::Result<bool>::success(false);
256 }
257
258 const bool hadLastPosition = sim->hasLastPos;
259 const float previousX = sim->lastX;
260 const float previousY = sim->lastY;
261 const int capacity = int(sim->particles.size());
262 const int conservativeRoom = std::max(0, capacity - gpu->estimatedAlive - sim->alive);
263 if (sim->spawnQuota < 0)
264 sim->spawnQuota = conservativeRoom;
265 else
266 sim->spawnQuota = std::min(sim->spawnQuota, conservativeRoom);
267
268 float simulatedDt = 0.f;
269 if (legacyControls) {
270 simulatedDt = advanceEmitterSim(*cfg, *sim, static_cast<float>(step.delta.seconds()));
271 } else {
272 auto simulation = advanceEmitterSim(*cfg, *sim, step);
273 if (!simulation) return eve::Result<bool>::failure(simulation.status());
274 simulatedDt = static_cast<float>(step.delta.seconds());
275 }
276 gpu->timeline += simulatedDt;
277 auto minHeap = std::greater<float>{};
278 while (!gpu->deathTimes.empty() && gpu->deathTimes.front() <= float(gpu->timeline) + 1e-6f) {
279 std::pop_heap(gpu->deathTimes.begin(), gpu->deathTimes.end(), minHeap);
280 gpu->deathTimes.pop_back();
281 }
282
283 const int room = std::max(0, capacity - int(gpu->deathTimes.size()));
284 const int accepted = std::min(sim->alive, room);
285 if (sim->alive > accepted) sim->droppedSpawnsThisFrame += sim->alive - accepted;
286 std::vector<graphics::GpuParticleSpawn> spawns;
287 spawns.reserve(std::size_t(accepted));
288 for (int i = 0; i < accepted; ++i) {
289 const Particle& particle = sim->particles[std::size_t(i)];
290 if (particle.life <= 0.f) continue;
291 spawns.push_back(gpuSpawn(particle));
292 gpu->deathTimes.push_back(float(gpu->timeline) + particle.life);
293 std::push_heap(gpu->deathTimes.begin(), gpu->deathTimes.end(), minHeap);
294 }
295
296 graphics::GpuParticleUpdate update;
297 update.dt = simulatedDt;
298 update.emitterX = cfg->x;
299 update.emitterY = cfg->y;
300 update.gravityX = cfg->gravityX;
301 update.gravityY = cfg->gravityY;
302 update.damping = cfg->damping;
303 update.velocityLimit = cfg->limitVelocity;
304 update.noiseStrength = cfg->noiseStrength;
305 update.noiseFrequency = cfg->noiseFrequency;
306 update.noiseSpeed = cfg->noiseSpeed;
307 update.time = sim->emitterAge;
308 update.frameRate = cfg->frameRate;
309 if (cfg->simSpace == "local" && hadLastPosition) {
310 update.localOffsetX = cfg->x - previousX;
311 update.localOffsetY = cfg->y - previousY;
312 }
313
314 if (!gfx->updateGpuParticleEmitter(gpu->residentHandle, update, spawns.data(), std::uint32_t(spawns.size()))) {
315 // Before activation the CPU state remains authoritative and already
316 // advanced. Once resident, a transient submit failure freezes state
317 // rather than losing it through an implicit CPU migration.
318 if (!gpu->residentActive) deactivateResidentGpu(gfx, *gpu);
319 gpu->failed = true;
320 return eve::Result<bool>::success(true);
321 }
322
323 for (int i = 0; i < sim->alive; ++i) sim->particles[std::size_t(i)].life = 0.f;
324 sim->alive = 0;
325 gpu->residentActive = true;
326 gpu->estimatedAlive = int(gpu->deathTimes.size());
327 return eve::Result<bool>::success(true);
328}

332eve::Result<void> runParticleSimulation(const eve::SimulationStep &step, bool legacyControls) {
333 if (step.delta.nanoseconds() < 0)
334 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
335 "particle simulation delta must be non-negative"));
336 const float dt = static_cast<float>(step.delta.seconds());
337 if (!std::isfinite(dt))
338 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
339 "particle simulation delta is outside float range"));
340
341 // Build the complete frame record off to the side. Checked failures must
342 // not publish counters that describe only the prefix of a frame; the
343 // previous published record remains the observable snapshot until the
344 // simulation reaches its commit point below.
345 ParticleFrameStats nextStats;
346 const uint64_t nextFrame = particleFrameStats().frameIndex + 1;
347 nextStats.frameIndex = nextFrame;
348 nextStats.simulationTick = step.tick;
349 auto &stats = nextStats;
350 const auto begin = std::chrono::steady_clock::now();
351
352 if (ecs::current()->getManager<ParticleEmitter>() == nullptr) {
353 mutableParticleFrameStats() = nextStats;
354 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
355 }
356
357 std::vector<ParticleEmitter *> allEmitters;
358 std::vector<ParticleEmitter *> emitters;
359 {
360 auto view = ecs::View<ParticleEmitter, ParticleEmitter::Config>();
361 for (auto it = view.begin(); it != view.end(); ++it) {
362 auto [cfg] = *it;
363 if (!cfg->entity) continue;
364 allEmitters.push_back(cfg->entity);
365 auto sim = cfg->entity->sim();
366 if (!legacyControls && sim->hasSimulationTick && step.tick <= sim->simulationTick)
367 return eve::Result<void>::failure(
368 eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
369 "particle simulation tick must advance monotonically for every emitter"));
370 if (sim->active || liveParticleCount(cfg->entity) > 0) emitters.push_back(cfg->entity);
371 }
372 }
373 std::stable_sort(emitters.begin(), emitters.end(), [](ParticleEmitter *a, ParticleEmitter *b) {
374 return a->config()->priority > b->config()->priority;
375 });
376
377 const auto &budget = particleBudgetConfig();
378 stats.emittersTotal = int(emitters.size());
379 for (auto *emitter : allEmitters) {
380 auto sim = emitter->sim();
381 stats.particlesBefore += liveParticleCount(emitter);
382 sim->spawnedThisFrame = 0;
383 sim->droppedSpawnsThisFrame = 0;
384 sim->spawnQuota = budget.maxParticles > 0 ? 0 : -1;
385 }
386 int reservedAlive = 0;
387 for (auto* emitter : emitters) reservedAlive += liveParticleCount(emitter);
388
389 auto* gfx = eve::ModuleManager::getInstance<eve::graphics::Graphics>("Graphics");
390
391 int processedAlive = 0;
392 for (auto *emitter : emitters) {
393 auto cfg = emitter->config();
394 auto sim = emitter->sim();
395 auto draw = emitter->draw();
396 const int aliveBefore = liveParticleCount(emitter);
397 reservedAlive -= aliveBefore;
398
399 if (budget.qualityLevel < cfg->minimumQuality) {
400 ++stats.emittersQualitySkipped;
401 processedAlive += aliveBefore;
402 continue;
403 }
404
405 const bool offscreen = emitterOffscreen(*cfg, *draw);
406 const bool culled = cfg->cullingMode == "pause" ? offscreen
407 : cfg->cullingMode == "always" ? false
408 : (aliveBefore <= 0 && offscreen);
409 if (culled) {
410 ++stats.emittersCulled;
411 processedAlive += aliveBefore;
412 continue;
413 }
414
415 if (budget.maxSimulatedEmitters > 0 &&
416 stats.emittersSimulated >= budget.maxSimulatedEmitters) {
417 ++stats.emittersBudgetSkipped;
418 processedAlive += aliveBefore;
419 continue;
420 }
421
422 int spawnQuota = cfg->maxSpawnPerFrame > 0 ? cfg->maxSpawnPerFrame : -1;
423 if (budget.maxParticles > 0) {
424 const int existingTotal = processedAlive + aliveBefore + reservedAlive;
425 const int globalRoom = std::max(0, budget.maxParticles - existingTotal);
426 spawnQuota = spawnQuota < 0 ? globalRoom : std::min(spawnQuota, globalRoom);
427 }
428 sim->spawnQuota = spawnQuota;
429
430 auto attach = emitter->attach();
431 auto skinSrc = emitter->skinSource();
432 auto gpuSim = emitter->gpuSim();
433 syncEmitterSources(*cfg, *sim, *attach, *skinSrc);
434 const bool gpuEligible = emitter->isGpuFeatureSetSupported();
435 if (gpuSim->residentActive && (!cfg->gpuSimulation || !gpuEligible)) deactivateResidentGpu(gfx, *gpuSim);
436 const bool wantsResident = cfg->gpuSimulation && gpuEligible;
437 bool handled = false;
438 if (wantsResident) {
439 auto gpuAdvance = advanceResidentGpu(gfx, emitter, step, legacyControls);
440 if (!gpuAdvance) return eve::Result<void>::failure(gpuAdvance.status());
441 handled = std::move(gpuAdvance).takeValue();
442 }
443 if (!handled) {
444 if (legacyControls)
445 advanceEmitterSim(*cfg, *sim, dt);
446 else {
447 auto cpuAdvance = advanceEmitterSim(*cfg, *sim, step);
448 if (!cpuAdvance) return eve::Result<void>::failure(cpuAdvance.status());
449 }
450 }
451 ++stats.emittersSimulated;
452 processedAlive += liveParticleCount(emitter);
453 if (budget.maxParticles > 0 || cfg->maxSpawnPerFrame > 0) sim->spawnQuota = 0;
454 }
455
456 stats.particlesAfter = 0;
457 stats.particlesSpawned = 0;
458 stats.droppedSpawns = 0;
459 for (auto *emitter : allEmitters) {
460 auto sim = emitter->sim();
461 auto gpu = emitter->gpuSim();
462 stats.particlesAfter += liveParticleCount(emitter);
463 stats.particlesSpawned += sim->spawnedThisFrame;
464 stats.droppedSpawns += sim->droppedSpawnsThisFrame;
465 if (gpu->residentActive) {
466 ++stats.gpuResidentEmitters;
467 stats.gpuResidentParticles += gpu->estimatedAlive;
468 }
469 sim->spawnQuota = -1;
470 }
471 stats.particlesKilled =
472 std::max(0, stats.particlesBefore + stats.particlesSpawned - stats.particlesAfter);
473 stats.simulationMs =
474 std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - begin)
475 .count();
476
477 if (!legacyControls) {
478 for (auto *emitter : allEmitters) {
479 auto sim = emitter->sim();
480 sim->simulationTick = step.tick;
481 sim->hasSimulationTick = true;
482 }
483 }
484 mutableParticleFrameStats() = nextStats;
485 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
486}

503int appendOneEmitter(graphics::Graphics& gfx, ParticleEmitter& emitter,
504 std::vector<graphics::DrawItem2D>& items, int& order,
505 bool respectSceneCulling, bool& usedCanvas, bool& culled) {
506 auto cfg = emitter.config(); auto sim = emitter.sim(); auto draw = emitter.draw();
507 auto gpu = emitter.gpuSim(); const int alive = liveParticleCount(&emitter);
508 if (!draw->visible || alive <= 0) return 0;
509 const auto& budget = particleBudgetConfig();
510 if (respectSceneCulling &&
511 (budget.qualityLevel < cfg->minimumQuality || emitterOffscreen(*cfg, *draw))) {
512 culled = true; return 0;
513 }
514 if (gpu->residentActive) {
515 graphics::GpuParticleDraw gpuDraw;
516 gpuDraw.texture = draw->texture; gpuDraw.sceneDepth = gfx.getSceneLinearDepthTexture();

554void ParticleRenderSystem::render(graphics::Graphics *gfx) {
555 auto &stats = mutableParticleFrameStats();
556 stats.renderedParticles = 0;
557 stats.renderCulledEmitters = 0;
558 stats.renderMs = 0.0;
559 const auto begin = std::chrono::steady_clock::now();
560 if (!gfx) return;
561 if (ecs::current()->getManager<ParticleEmitter>() == nullptr) return;
562
563 std::vector<graphics::DrawItem2D> items;
564 auto view = ecs::View<ParticleEmitter, ParticleEmitter::Config, ParticleEmitter::Sim,
565 ParticleEmitter::Draw>();
566 bool anyCanvas = false;
567 int order = 0;
568 for (auto it = view.begin(); it != view.end(); ++it) {
569 auto [cfg, sim, draw] = *it; (void)sim; (void)draw;
570 auto* emitter = cfg->entity;
571 if (!emitter) continue;
572 bool culled=false;
573 stats.renderedParticles += appendOneEmitter(*gfx,*emitter,items,order,true,anyCanvas,culled);
574 if(culled) ++stats.renderCulledEmitters;
575 }
576
577 // Unified 2D sprite path: rotation / flipbook UV / blend / layer sorting
578 // and camera handling all come from RenderSystem::drawItems.
579 if (!items.empty()) graphics::RenderSystem::drawItems(*gfx, items, false);
580 if (anyCanvas) gfx->setCanvas();
581 stats.renderMs =
582 std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - begin)
583 .count();
584}

586void ParticleLightSystem::update() {
587 if (ecs::current()->getManager<ParticleEmitter>() == nullptr) return;
588
589 // Pass 1: collect emitters. Creating Light2D entities inside a deferred
590 // View would stage them and invalidate stored raw pointers on publish.
591 std::vector<ParticleEmitter *> emitters;
592 {
593 auto view = ecs::View<ParticleEmitter, ParticleEmitter::Config>();
594 for (auto it = view.begin(); it != view.end(); ++it) {
595 auto [cfg] = *it;
596 if (cfg->entity) emitters.push_back(cfg->entity);
597 }
598 }
599
600 // Pass 2: create/sync lights with no View active (stable entity pointers).
601 for (auto *em : emitters) {
602 auto cfg = em->config();
603 auto sim = em->sim();
604 auto draw = em->draw();
605 auto lights = em->lights();
606 if (!cfg->lights.enabled) {
607 for (auto *l : lights->pool)
608 if (l) l->setEnabled(false);
609 continue;
610 }
611 const int maxL = cfg->lights.max > 0 ? (cfg->lights.max > 8 ? 8 : cfg->lights.max) : 0;
612 while (int(lights->pool.size()) < maxL)
613 lights->pool.push_back(graphics::Light2D::createLight());
614 const int n = sim->alive < maxL ? sim->alive : maxL;
615 for (int i = 0; i < maxL; ++i) {
616 graphics::Light2D *l = lights->pool[size_t(i)];
617 if (i < n) {
618 const Particle &p = sim->particles[size_t(i)];
619 l->setPosition(p.x, p.y);
620 l->setRadius(cfg->lights.radius);
621 l->setColor(cfg->lights.r, cfg->lights.g, cfg->lights.b, cfg->lights.intensity);
622 l->setCanvas(draw->canvas);
623 l->setEnabled(true);
624 } else {
625 l->setEnabled(false);
626 }
627 }
628 }
629}

631int ParticleConfigSystem::poll() {
632 if (ecs::current()->getManager<ParticleEmitter>() == nullptr) return 0;
633
634 // Watch events are drained by load.nut / HotReload. This polling path is a
635 // deliberate, explicit file-state observation for hosts that do not
636 // deliver a watch event; its result is recorded on each Resource.
637 int reloaded = 0;
638 auto view =
639 ecs::View<ParticleEmitter, ParticleEmitter::Config, ParticleEmitter::Resource>();
640 for (auto it = view.begin(); it != view.end(); ++it) {
641 auto [cfg, res] = *it;
642 if (!cfg->entity) continue;
643 if (res->path.empty()) {
644 res->lastReloadObservation = ParticleEmitter::Resource::ReloadObservation::Unbound;
645 continue;
646 }
647 if (!res->autoReload) {
648 res->lastReloadObservation = ParticleEmitter::Resource::ReloadObservation::AutoReloadDisabled;
649 continue;
650 }
651
652 const int64_t mt = fileModtime(res->path);
653 if (mt < 0) {
654 res->lastReloadObservation = ParticleEmitter::Resource::ReloadObservation::MtimeUnavailable;
655 continue;
656 }
657 if (mt == res->modtime) {
658 res->lastReloadObservation = ParticleEmitter::Resource::ReloadObservation::MtimePollingUnchanged;
659 continue;
660 }
661 if (reloadConfigFile(cfg->entity, nullptr)) {
662 res->lastReloadObservation = ParticleEmitter::Resource::ReloadObservation::MtimePollingReloaded;
663 ++reloaded;
664 } else {
665 res->lastReloadObservation = ParticleEmitter::Resource::ReloadObservation::MtimePollingReloadFailed;
666 }
667 }
668 return reloaded;
669}
