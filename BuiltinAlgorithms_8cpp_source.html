82bool genMazeBacktrack(const Params &params, Grid2D &out, std::string &error) {
83 const int w = params.getWidth();
84 const int h = params.getHeight();
85 // MazeDig works best on odd dimensions.
86 const int mw = w < 5 ? 5 : (w % 2 == 0 ? w - 1 : w);
87 const int mh = h < 5 ? 5 : (h % 2 == 0 ? h - 1 : h);
88
89 auto matrix = dtlutil::makeMatrix(mw, mh, 0);
90 // empty(floor)=1, wall=0
91 dtl::shape::MazeDig<std::uint_fast8_t> shape(1, 0);
92 if (!shape.drawSEED(matrix, params.getSeed())) {
93 error = "maze.backtrack: DTL MazeDig draw failed";
94 return false;
95 }
96
97 out.resize(w, h);
98 out.fill(Semantic::Wall);
99 for (int y = 0; y < mh && y < h; ++y) {
100 for (int x = 0; x < mw && x < w; ++x) {
101 out.setCell(x, y, matrix[size_t(y)][size_t(x)] == 1 ? Semantic::Floor : Semantic::Wall);
102 }
103 }
104 out.setMeta("algorithm", "maze.backtrack");
105 dtlutil::placeSpawnAndStairs(out, params.getSeed());
106 return true;
107}

120
121 auto matrix = dtlutil::makeMatrix(w, h, 0);
122 dtl::shape::PerlinIsland<std::uint_fast8_t> shape(double(std::max(0.1f, frequency)),
123 size_t(std::max(1, octaves)),
124 std::uint_fast8_t(std::max(1, maxHeight)), 0);
125 if (!shape.drawSEED(matrix, params.getSeed())) {
126 error = "noise.terrain: DTL PerlinIsland draw failed";
127 return false;
128 }

130 const float waterMax = params.getFloat("waterMax", 0.25f);
131 const float sandMax = params.getFloat("sandMax", 0.35f);
132 const float grassMax = params.getFloat("grassMax", 0.65f);
133 const float dirtMax = params.getFloat("dirtMax", 0.80f);
134 const float stoneMax = params.getFloat("stoneMax", 0.92f);
135
136 dtlutil::copyMatrixToGrid(matrix, out, [&](std::uint_fast8_t v) -> uint32_t {
137 const float t = maxHeight <= 0 ? 0.f : float(v) / float(maxHeight);
138 if (t < waterMax) return Semantic::Water;
139 if (t < sandMax) return Semantic::Sand;
140 if (t < grassMax) return Semantic::Grass;
141 if (t < dirtMax) return Semantic::Dirt;
142 if (t < stoneMax) return Semantic::Stone;
143 return Semantic::Snow;
144 });
145 out.setMeta("algorithm", "noise.terrain");
146 dtlutil::placeSpawnAndStairs(out, params.getSeed());
147 return true;

151bool genCaveDrunkard(const Params &params, Grid2D &out, std::string &error) {
152 const int w = params.getWidth();
153 const int h = params.getHeight();
154 if (w < 4 || h < 4) {
155 error = "cave.drunkard: size must be at least 4x4";
156 return false;
157 }
158 const float floorPct = std::clamp(params.getFloat("floorPct", 0.45f), 0.05f, 0.95f);
159 const int target = int(float(w * h) * floorPct);
160
161 out.resize(w, h);
162 out.fill(Semantic::Wall);
163
164 std::mt19937 rng(params.getSeed());
165 std::uniform_int_distribution<int> dirDist(0, 3);
166 int x = w / 2;
167 int y = h / 2;
168 int carved = 0;
169 int guard = w * h * 20;
170 while (carved < target && guard-- > 0) {
171 if (out.getCell(x, y) != Semantic::Floor) {
172 out.setCell(x, y, Semantic::Floor);
173 ++carved;
174 }
175 switch (dirDist(rng)) {
176 case 0: if (x > 1) --x; break;
177 case 1: if (x < w - 2) ++x; break;
178 case 2: if (y > 1) --y; break;
179 default: if (y < h - 2) ++y; break;
180 }
181 }
182 // Outer wall frame
183 for (int i = 0; i < w; ++i) {
184 out.setCell(i, 0, Semantic::Wall);
185 out.setCell(i, h - 1, Semantic::Wall);
186 }
187 for (int j = 0; j < h; ++j) {
188 out.setCell(0, j, Semantic::Wall);
189 out.setCell(w - 1, j, Semantic::Wall);
190 }
191 out.setMeta("algorithm", "cave.drunkard");
192 dtlutil::placeSpawnAndStairs(out, params.getSeed());
193 return true;
194}
