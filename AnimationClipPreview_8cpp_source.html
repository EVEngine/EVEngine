12 const animation_editing::AnimationClipDocumentTarget& document, double time,
13 const std::vector<AnimationPreviewBone>& skeleton, const std::string& selectedBone,
14 const editing::TaskContext& context) const override {
15 std::vector<std::string> names;
16 std::map<std::string, std::string> parents;
17 for (const auto& bone : skeleton) {
18 names.push_back(bone.name);
19 parents.emplace(bone.name, bone.parent);
20 }
21 auto sampled = document.preview(time, names);
22 if (sampled.status != editing::Status::Applied)
23 return editing::failed<AnimationPreviewPublication>(
24 sampled.status, editing::RuleId("editor.animation.preview"), "Could not sample animation pose");
25 std::vector<animation_editing::SkeletonOverlayBone> bones;
26 bones.reserve(sampled.bones.size());
27 for (const auto& sample : sampled.bones) {
28 if (context.isCancellationRequested())
29 return editing::failed<AnimationPreviewPublication>(
30 editing::Status::Cancelled, editing::RuleId("editor.animation.preview-cancelled"),
31 "Pose request cancelled");
32 animation_editing::SkeletonOverlayBone bone;
33 bone.id = animation_editing::StableId(sample.bone);
34 bone.parent = animation_editing::StableId(parents.at(sample.bone));
35 bone.name = sample.bone;
36 bone.position = {sample.positionX, sample.positionY, sample.positionZ};
37 bone.rotation = {sample.rotationX, sample.rotationY, sample.rotationZ, sample.rotationW};
38 bone.selected = sample.bone == selectedBone;
39 bone.maskWeight = sample.maskWeight;
40 bones.push_back(std::move(bone));
41 }
42 animation_editing::SkeletonOverlayOptions options;
43 options.showAxes = true;
44 options.axesForSelectedOnly = true;
45 options.showConstraints = false;
46 auto overlay = animation_editing::SkeletonOverlayBuilder{}.build(document.targetId().value(),
47 document.revision(), bones, options);
48 if (overlay.status != editing::Status::Applied)
49 return editing::failed<AnimationPreviewPublication>(
50 overlay.status, editing::RuleId("editor.animation.overlay"), "Could not build skeleton overlay");
51 return editing::applied(AnimationPreviewPublication{std::move(sampled), std::move(overlay)});
52 }

100animation_editing::Result<void> AnimationClipEditor::pollPreview() {
101 auto result = previewSession_->async.poll();
102 if (!result) return previewState().phase == editor::PreviewPhase::Pending ? editing::pending() : editing::noOp();
103 if (!result->ok()) return editing::Result<void>::failure(result->status());
104 auto publication = std::move(*result).takeValue();
105 auto& pose = publication.result.pose;
106 auto& overlay = publication.result.overlay;
107 if (pose.documentRevision != revision() || publication.requestedTime != playhead_ ||
108 publication.selected != selectedBone_ || pose.status != editing::Status::Applied ||
109 overlay.status != editing::Status::Applied || overlay.targetRevision != revision() ||
110 overlay.target != target_.targetId().value()) {
111 previewSession_->async.rejectPublication();
112 return editing::failed<void>(editing::Status::Conflict, editing::RuleId("editor.animation.stale-preview"),
113 "Pose inputs changed before publication");
114 }
115 preview_ = std::move(pose);
116 overlay_ = std::move(overlay);
117 return editing::applied<void>();
