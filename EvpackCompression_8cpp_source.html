18Result<std::vector<std::uint8_t>> compressEvpackChunk(
19 EvpackCodec codec, std::span<const std::uint8_t> decoded) {
20 if (codec == EvpackCodec::None)
21 return Result<std::vector<std::uint8_t>>::success(
22 std::vector<std::uint8_t>(decoded.begin(), decoded.end()));
23 if (codec != EvpackCodec::Zstd)
24 return codecFailure<std::vector<std::uint8_t>>(DiagnosticCode::Unsupported,
25 "unknown runtime chunk codec");
26 if (decoded.size() > ZSTD_MAX_INPUT_SIZE)
27 return codecFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
28 "chunk exceeds the Zstd input limit");
29 const std::size_t bound = ZSTD_compressBound(decoded.size());
30 std::vector<std::uint8_t> stored(bound);
31 const std::size_t size = ZSTD_compress(stored.data(), stored.size(), decoded.data(), decoded.size(), 3);
32 if (ZSTD_isError(size))
33 return codecFailure<std::vector<std::uint8_t>>(
34 DiagnosticCode::Failed, std::string("Zstd compression failed: ") + ZSTD_getErrorName(size));
35 stored.resize(size);
36 return Result<std::vector<std::uint8_t>>::success(std::move(stored));
37}

39Result<std::vector<std::uint8_t>> decompressEvpackChunk(
40 EvpackCodec codec, std::span<const std::uint8_t> stored, std::uint64_t decodedSize,
41 std::uint64_t maximumDecodedBytes) {
42 if (decodedSize > maximumDecodedBytes || decodedSize > std::numeric_limits<std::size_t>::max())
43 return codecFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
44 "decoded chunk exceeds its resource budget");
45 if (codec == EvpackCodec::None) {
46 if (stored.size() != decodedSize)
47 return codecFailure<std::vector<std::uint8_t>>(DiagnosticCode::ParseError,
48 "uncompressed chunk size does not match the TOC");
49 return Result<std::vector<std::uint8_t>>::success(
50 std::vector<std::uint8_t>(stored.begin(), stored.end()));
51 }
52 if (codec != EvpackCodec::Zstd)
53 return codecFailure<std::vector<std::uint8_t>>(DiagnosticCode::Unsupported,
54 "unknown runtime chunk codec");
55 std::vector<std::uint8_t> decoded(static_cast<std::size_t>(decodedSize));
56 const std::size_t size = ZSTD_decompress(decoded.data(), decoded.size(), stored.data(), stored.size());
57 if (ZSTD_isError(size))
58 return codecFailure<std::vector<std::uint8_t>>(
59 DiagnosticCode::ParseError, std::string("Zstd decompression failed: ") + ZSTD_getErrorName(size));
60 if (size != decoded.size())
61 return codecFailure<std::vector<std::uint8_t>>(DiagnosticCode::ParseError,
62 "decoded chunk size does not match the TOC");
63 return Result<std::vector<std::uint8_t>>::success(std::move(decoded));
64}
