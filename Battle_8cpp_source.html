80RPGActor *Battle::lowestHealthTarget(int side, bool sameSide) const {
81 RPGActor *selected = nullptr;
82 double selectedRatio = 0.0;
83 for (const auto &participant : participants_) {
84 if (isDead(participant) || ((participant.side == side) != sameSide)) continue;
85 const double maximum = VitalsSystem::getMax(participant.actor, "hp");
86 if (maximum <= 0.0) continue;
87 const double ratio = VitalsSystem::getCurrent(participant.actor, "hp") / maximum;
88 if (!selected || ratio < selectedRatio) {
89 selected = participant.actor;
90 selectedRatio = ratio;
91 }
92 }

96eve::Result<void> Battle::setActionChecked(RPGActor *actor, const std::string &skillId,
97 RPGActor *target) {
98 const auto reject = [](eve::DiagnosticCode code, std::string message, std::string path) {
99 return eve::Result<void>::failure(
100 eve::Diagnostic::error(code, std::move(message), std::move(path), {}, "rpg.battle.action"));
101 };
102 if (!actor) return reject(eve::DiagnosticCode::InvalidArgument, "actor must not be null", "actor");
103 if (finished_ || started_)
104 return reject(eve::DiagnosticCode::Conflict,
105 "actions can only be queued before a non-finished round", "battle");
106 const auto participant = std::find_if(participants_.begin(), participants_.end(),
107 [actor](const Participant &value) {
108 return value.actor == actor;
109 });
110 if (participant == participants_.end())
111 return reject(eve::DiagnosticCode::NotFound, "actor is not a battle participant", "actor");
112 if (isDead(*participant))
113 return reject(eve::DiagnosticCode::Conflict, "dead actors cannot queue actions", "actor");
114 if (std::any_of(roundActions_.begin(), roundActions_.end(), [actor](const PendingAction &value) {
115 return value.actor == actor;
116 }))
117 return reject(eve::DiagnosticCode::Conflict,
118 "actor already has an action queued for this round", "actor");
119 if (!skillId.empty() && !actor->knowsSkill(skillId))
120 return reject(eve::DiagnosticCode::NotFound, "actor does not know the requested skill", "skillId");
121 if (target) {
122 const auto targetParticipant = std::find_if(
123 participants_.begin(), participants_.end(), [target](const Participant &value) {
124 return value.actor == target;
125 });
126 if (targetParticipant == participants_.end())
127 return reject(eve::DiagnosticCode::NotFound,
128 "target is not a battle participant", "target");
129 if (isDead(*targetParticipant))
130 return reject(eve::DiagnosticCode::Conflict, "target is not alive", "target");
131 const std::string targetType = skillId.empty() ? "enemySingle"
132 : SkillSystem::getTargetType(actor, skillId);
133 if (targetType == "self" && target != actor)
134 return reject(eve::DiagnosticCode::InvalidArgument,
135 "self-target skill must target its caster", "target");
136 if (targetType == "allySingle" && targetParticipant->side != participant->side)
137 return reject(eve::DiagnosticCode::InvalidArgument,
138 "ally-target action must target the caster side", "target");
139 if (targetType != "self" && targetType != "allySingle" &&
140 targetParticipant->side == participant->side)
141 return reject(eve::DiagnosticCode::InvalidArgument,
142 "enemy-target action must target another side", "target");
143 }
144 roundActions_.push_back(PendingAction{actor, skillId, target, 0.0});
145 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
146}

148eve::Result<void> Battle::setActionByPolicyChecked(RPGActor *actor,
149 const std::string &skillId,
150 BattleTargetPolicy policy) {
151 if (policy == BattleTargetPolicy::Auto) return setActionChecked(actor, skillId, nullptr);
152 const int side = sideOf(actor);
153 RPGActor *target = nullptr;
154 if (policy == BattleTargetPolicy::Self) target = actor;
155 else if (policy == BattleTargetPolicy::LowestHealthAlly)
156 target = lowestHealthTarget(side, true);
157 else if (policy == BattleTargetPolicy::LowestHealthEnemy)
158 target = lowestHealthTarget(side, false);

160 return eve::Result<void>::failure(eve::Diagnostic::error(
161 eve::DiagnosticCode::NotFound, "target policy found no living participant", "policy", {},
162 "rpg.battle.action-policy"));
163 return setActionChecked(actor, skillId, target);
164}

166void Battle::autoEnemyActions() {
167 if (finished_) return;
168 for (const auto &p : participants_) {
169 if (p.side == playerSide_ || isDead(p)) continue;
170 bool already = false;
171 for (const auto &ra : roundActions_) {
172 if (ra.actor == p.actor) {
173 already = true;
174 break;
175 }
176 }
177 if (already) continue;
178 // 随机已学技能，否则普攻（skillId 空 = 普攻）
179 std::vector<std::string> known;
180 for (const auto &[id, unused] : p.actor->skills()->known) {
181 (void)unused;
182 known.push_back(id);
183 }
184 std::string skillId;
185 if (!known.empty()) {
186 skillId = known[size_t((seedCounter_++) % known.size())];
187 }
188 roundActions_.push_back(PendingAction{p.actor, skillId, randomOpponent(p.side), 0.0});
189 }

192void Battle::startRound() {
193 if (finished_) return;
194 ++turn_;
195 // 计算先攻：speed 属性 + 技能攻击速度修正。
196 for (auto &pa : roundActions_) {
197 double speed = AttributeSystem::getFinal(pa.actor, "speed");
198 if (speed <= 0.0) speed = AttributeSystem::getFinal(pa.actor, "agi");
199 if (speed <= 0.0) speed = 0.0;
200 pa.initiative = speed + TraitSystem::getAttackSpeed(pa.actor);
201 }
202 std::stable_sort(roundActions_.begin(), roundActions_.end(),
203 [](const PendingAction &x, const PendingAction &y) {
204 return x.initiative > y.initiative;
205 });
206 queue_ = std::move(roundActions_);
207 roundActions_.clear();
208 started_ = true;
209 events_.push_back(BattleEvent{"roundStart", "", nullptr, nullptr, 0.0, false});
210}

212bool Battle::executeNextAction() {
213 if (!started_ || finished_) return false;
214 if (queue_.empty()) {
215 started_ = false;
216 return false;
217 }
218 PendingAction pa = queue_.front();
219 queue_.erase(queue_.begin());
220
221 // 已死则跳过
222 if (VitalsSystem::isDead(pa.actor, "hp")) {
223 return true;
224 }
225
226 // 解析目标：未指定则按 targetType / 默认对手。
227 RPGActor *target = pa.target;
228 if (!target) {
229 if (pa.skillId.empty()) {
230 target = randomOpponent(sideOf(pa.actor));
231 } else if (SkillSystem::getTargetType(pa.actor, pa.skillId) == "self") {
232 target = pa.actor;
233 } else if (SkillSystem::getTargetType(pa.actor, pa.skillId) == "allySingle") {
234 target = lowestHealthTarget(sideOf(pa.actor), true);
235 } else {
236 target = randomOpponent(sideOf(pa.actor));
237 }
238 }
239 if (!target) return true;
240
241 execute(pa, seedCounter_);
242 // 每步检查结束：仅剩一方 → 胜；双方全灭 → 平；否则继续。
243 const int w = computeWinnerSide();
244 if (w != -1) {
245 finished_ = true;
246 winner_ = w;
247 events_.push_back(BattleEvent{winner_ == playerSide_ ? "victory" : "defeat", "", nullptr,
248 nullptr, 0.0, false});
249 } else {
250 bool anyAlive = false;
251 for (const auto &p : participants_) {
252 if (!isDead(p)) {
253 anyAlive = true;
254 break;
255 }
256 }
257 if (!anyAlive) {
258 finished_ = true;
259 winner_ = -1;
260 events_.push_back(BattleEvent{"defeat", "", nullptr, nullptr, 0.0, false});
261 }
262 }
263 return true;
264}

266void Battle::execute(PendingAction &pa, unsigned &seedCounter) {
267 const std::string skillId = pa.skillId;
268 const std::string targetType = skillId.empty() ? std::string("enemySingle")
269 : SkillSystem::getTargetType(pa.actor, skillId);
270 RPGActor *target = pa.target;
271 if (!target) {
272 if (targetType == "self") {
273 target = pa.actor;
274 } else if (targetType == "allySingle") {
275 target = lowestHealthTarget(sideOf(pa.actor), true);
276 } else {
277 target = randomOpponent(sideOf(pa.actor));
278 }
279 }
280 if (!target) return;
281
282 events_.push_back(BattleEvent{"actionStart", skillId, pa.actor, target, 0.0, false});
283
284 // 目标已死：换一个存活对手
285 if (targetType != "self" && VitalsSystem::isDead(target, "hp")) {
286 target = targetType == "allySingle" ? lowestHealthTarget(sideOf(pa.actor), true)
287 : randomOpponent(sideOf(pa.actor));
288 if (!target) return;
289 }
290
291 const SkillDamageSpec *spec = skillId.empty() ? nullptr : BattleSystem::findSkillDamage(skillId);
292 if (!spec && !skillId.empty()) {
293 // 无伤害规格：走技能授予效果（治疗/状态等，无公式伤害）。
294 SkillSystem::beginCast(pa.actor, skillId, target);
295 return;
296 }
297 if (!spec) {
298 // 普攻默认规格
299 static const SkillDamageSpec kDefault{};
300 spec = &kDefault;
301 }
302
303 DamageResult result = BattleSystem::resolveHit(pa.actor, target, *spec, seedCounter++);
304 if (!result.hit) {
305 events_.push_back(BattleEvent{"miss", skillId, pa.actor, target, 0.0, false});
306 return;
307 }
308
309 // 目标资源名任意：damageType 即资源名，"XHeal" 后缀表示治疗。
310 std::string resource = spec->damageType;
311 bool heal = false;
312 if (resource.size() >= 4 && resource.compare(resource.size() - 4, 4, "Heal") == 0) {
313 resource = resource.substr(0, resource.size() - 4);
314 heal = true;
315 }
316 if (heal) {
317 double healed = VitalsSystem::heal(target, resource, result.amount);
318 events_.push_back(BattleEvent{"heal", skillId, pa.actor, target, healed, false});
319 } else {
320 double dealt = VitalsSystem::takeDamage(target, resource, result.amount,
321 skillId.empty() ? "attack" : skillId);
322 events_.push_back(BattleEvent{"damage", skillId, pa.actor, target, dealt, result.crit});
323 }
