20EditorResult<void> EditorCommandService::checkExecutionPolicy(const CommandDescriptor& descriptor,
21 const CommandId& id, CommandSource source,
22 const EditorValue& payload,
23 const HostProfile& profile) const {
24 if (!profile.allowsCommand(id))
25 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.command.profile-denied"),
26 "Command is unavailable in this host: " + id.value());
27 if (!profile.hasFeatures(descriptor.requiredFeatures))
28 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.command.feature-denied"),
29 "Command requires features unavailable in this host: " + id.value());
30 if (source == CommandSource::Automation && !descriptor.automationAllowed)
31 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.command.automation-denied"),
32 "Command is not available to automation: " + id.value());
33 if (!payload.isWithinLimits(kMaxPayloadDepth, kMaxPayloadElements, profile.maxPayloadBytes()))
34 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.command.payload-limit"),
35 "Command payload exceeds host limits: " + id.value());
36 return eve::editing::applied<void>();
37}

44EditorResult<EditorValue> EditorCommandService::registerCommand(CommandDescriptor descriptor,
45 EditorCommandHandler handler, bool replace) {
46 if (!descriptor.id || descriptor.ownerModule.empty() || !handler)
47 return error(EditorStatus::Rejected, "editor.command.invalid-registration",
48 "Command id, owner module and handler are required");
49
50 auto it = std::find_if(commands_.begin(), commands_.end(),
51 [&](const Registration& entry) { return entry.descriptor.id == descriptor.id; });
52 if (it != commands_.end()) {
53 if (!replace || it->descriptor.ownerModule != descriptor.ownerModule)
54 return error(EditorStatus::Rejected, "editor.command.duplicate",
55 "Command is already registered: " + descriptor.id.value());
56 *it = {std::move(descriptor), std::move(handler), {}, {}, nextRegistrationGeneration_++};
57 } else {
58 commands_.push_back({std::move(descriptor), std::move(handler), {}, {}, nextRegistrationGeneration_++});
59 }
60 ++revision_;
61 return eve::editing::applied<EditorValue>(EditorValue{});
62}

133EditorResult<EditorValue> EditorCommandService::execute(const CommandId& id, const CommandContext& context,
134 const EditorValue& payload) const {
135 auto it = std::find_if(commands_.begin(), commands_.end(),
136 [&](const Registration& entry) { return entry.descriptor.id == id; });
137 if (it == commands_.end())
138 return error(EditorStatus::NotFound, "editor.command.not-found", "Command is not registered: " + id.value());
139 if (!context.profile)
140 return error(EditorStatus::Failed, "editor.command.missing-profile", "Command context has no host profile");
141 if (auto policy = checkExecutionPolicy(it->descriptor, id, context.source, payload, *context.profile);
142 !policy.ok())
143 return denyPolicy<EditorValue>(std::move(policy));
144
145 if (!it->handler)
146 return error(EditorStatus::Unsupported, "editor.command.requires-plan",
147 "Command must be planned before it can be executed: " + id.value());
148 try {
149 return it->handler(context, payload);
150 } catch (const std::exception& exception) {
151 return error(EditorStatus::Failed, "editor.command.handler-exception",
152 "Command handler failed: " + std::string(exception.what()));
153 } catch (...) {
154 return error(EditorStatus::Failed, "editor.command.handler-exception",
155 "Command handler failed with an unknown exception");
156 }
157}

159EditorResult<CommandPlan> EditorCommandService::plan(const CommandRequest& request, const HostProfile& profile) const {
160 auto failure = [](EditorStatus status, const char* rule, std::string message) {
161 return eve::editing::failed<CommandPlan>(status, RuleId(rule), std::move(message));
162 };
163 auto it = std::find_if(commands_.begin(), commands_.end(),
164 [&](const Registration& entry) { return entry.descriptor.id == request.id; });
165 if (it == commands_.end())
166 return failure(EditorStatus::NotFound, "editor.command.not-found",
167 "Command is not registered: " + request.id.value());
168 if (auto policy = checkExecutionPolicy(it->descriptor, request.id, request.source, request.payload, profile);
169 !policy.ok())
170 return denyPolicy<CommandPlan>(std::move(policy));
171 if (request.expectedRevision && *request.expectedRevision != request.context.targetRevision)
172 return failure(EditorStatus::Conflict, "editor.command.revision-conflict",
173 "Expected revision does not match the captured context");
174 if (!it->planner)
175 return failure(EditorStatus::Unsupported, "editor.command.planning-unsupported",
176 "Command does not provide a planning handler");
177 try {
178 EditorResult<CommandPlan> result = it->planner(request);
179 if (result.ok()) {
180 result.value().command = request.id;
181 result.value().target = request.context.target;
182 result.value().baseRevision = request.context.targetRevision;
183 if (result.value().targetGeneration == 0)
184 result.value().targetGeneration = request.context.targetGeneration;
185 result.value().commandGeneration = it->generation;
186 result.value().plannedPayload = request.payload;
187 result.value().plannedSource = request.source;
188 if (result.value().id.empty())
189 result.value().id = PlanId(request.id.value() + ".plan." + std::to_string(++planSequence_));
190 }
191 return result;
192 } catch (const std::exception& exception) {
193 return failure(EditorStatus::Failed, "editor.command.planner-exception", exception.what());
194 } catch (...) {
195 return failure(EditorStatus::Failed, "editor.command.planner-exception",
196 "Command planner failed with an unknown exception");
197 }

200EditorResult<TransactionReceipt> EditorCommandService::executePlan(const CommandRequest& request,
201 const CommandPlan& plan,
202 const HostProfile& profile) const {
203 auto failure = [](EditorStatus status, const char* rule, std::string message) {
204 return eve::editing::failed<TransactionReceipt>(status, RuleId(rule), std::move(message));
205 };
206 if (plan.id.empty() || plan.command != request.id || plan.target != request.context.target)
207 return failure(EditorStatus::Rejected, "editor.command.invalid-plan",
208 "Plan identity or target does not match the request");
209 if (plan.baseRevision != request.context.targetRevision)
210 return failure(EditorStatus::Conflict, "editor.command.stale-plan",
211 "Plan was produced for another target revision");
212 if (plan.targetGeneration != request.context.targetGeneration)
213 return failure(EditorStatus::Conflict, "editor.command.stale-target-generation",
214 "Plan was produced for another target lifetime");
215 if (request.expectedRevision && *request.expectedRevision != request.context.targetRevision)
216 return failure(EditorStatus::Conflict, "editor.command.revision-conflict",
217 "Expected revision does not match the captured context");
218 auto it = std::find_if(commands_.begin(), commands_.end(),
219 [&](const Registration& entry) { return entry.descriptor.id == request.id; });
220 if (it == commands_.end())
221 return failure(EditorStatus::NotFound, "editor.command.not-found",
222 "Command is not registered: " + request.id.value());
223 if (plan.commandGeneration != it->generation)
224 return failure(EditorStatus::Conflict, "editor.command.stale-registration",
225 "Command registration changed after planning");
226 if (auto policy = checkExecutionPolicy(it->descriptor, request.id, request.source, request.payload, profile);
227 !policy.ok())
228 return denyPolicy<TransactionReceipt>(std::move(policy));
229 if (request.payload != plan.plannedPayload || request.source != plan.plannedSource)
230 return failure(EditorStatus::Rejected, "editor.command.plan-input-mismatch",
231 "Plan payload or source differs from the validated request");
232 if (!it->planExecutor)
233 return failure(EditorStatus::Unsupported, "editor.command.execution-unsupported",
234 "Command does not provide a plan executor");
235 try {
236 return it->planExecutor(request, plan);
237 } catch (const std::exception& exception) {
238 return failure(EditorStatus::Failed, "editor.command.executor-exception", exception.what());
239 } catch (...) {
240 return failure(EditorStatus::Failed, "editor.command.executor-exception",
241 "Command executor failed with an unknown exception");
242 }
243}
