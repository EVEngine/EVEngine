71 std::string tag;

74 if (!magic) {
75 int version = 0;
76 if (tag != "EVA" || !(line >> version) || version != 1 || !complete(line))

78 "legacy animation must begin with EVA 1", path);
79 magic = true;
80 } else if (tag == "skeleton") {
81 if (!(line >> declaredBones) || declaredBones == 0 ||
82 declaredBones > request.limits.maximumAssets || !complete(line))

85 } else if (tag == "bone") {
86 std::uint32_t index = 0;

88 if (!(line >> index >> bone.parent) || index != result.bones.size())
89 return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid bone index", path);
90 std::getline(line, bone.name); bone.name = trim(std::move(bone.name));
91 if (bone.name.empty() || bone.name.size() > request.limits.maximumStringBytes ||
92 bone.parent >= static_cast<std::int32_t>(index) || bone.parent < -1)
93 return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid bone hierarchy", path);
94 result.bones.push_back(std::move(bone));
95 } else if (tag == "bind") {
96 std::uint32_t index = 0;
97 if (!(line >> index) || index >= result.bones.size())
98 return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid bind index", path);
99 auto& bone = result.bones[index];
100 if (!(line >> bone.position[0] >> bone.position[1] >> bone.position[2] >>
101 bone.rotation[0] >> bone.rotation[1] >> bone.rotation[2] >> bone.rotation[3] >>
102 bone.scale[0] >> bone.scale[1] >> bone.scale[2]) || !complete(line))
103 return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid bind transform", path);
104 bone.hasBind = true;
105 } else if (tag == "clip") {
106 std::getline(line, result.clipName); result.clipName = trim(std::move(result.clipName));
107 if (result.clipName.empty() || result.clipName.size() > request.limits.maximumStringBytes)
108 return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid clip name", path);
109 } else if (tag == "duration") {
110 if (!(line >> result.duration) || !complete(line) || !finite(result.duration) || result.duration < 0)
111 return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid clip duration", path);
112 } else if (tag == "rate") {
113 if (!(line >> result.rate) || !complete(line) || !finite(result.rate) || result.rate <= 0)
114 return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid sample rate", path);
115 } else if (tag == "loop") {
116 int value = 0;
117 if (!(line >> value) || (value != 0 && value != 1) || !complete(line))
118 return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid loop flag", path);
119 result.loop = value != 0;
120 } else if (tag == "sync") {
121 Marker marker;
122 if (!(line >> marker.time) || !finite(marker.time))
123 return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid sync marker", path);
124 std::getline(line, marker.name); marker.name = trim(std::move(marker.name));
125 if (marker.name.empty()) return detail::failure<Fixture>(DiagnosticCode::ParseError,
126 "empty sync marker", path);
127 result.markers.push_back(std::move(marker));
128 if (result.markers.size() > request.limits.maximumAssets)
129 return detail::failure<Fixture>(DiagnosticCode::InvalidArgument, "too many sync markers", path);
130 } else if (tag == "track") {
131 Track value;
132 if (!(line >> value.bone >> value.expectedPosition >> value.expectedRotation >> value.expectedScale) ||
133 !complete(line) || value.bone >= result.bones.size())
134 return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid track header", path);
135 result.tracks.push_back(std::move(value)); track = &result.tracks.back();
136 if (result.tracks.size() > request.limits.maximumAssets)
137 return detail::failure<Fixture>(DiagnosticCode::InvalidArgument, "too many animation tracks", path);
138 } else if (tag == "p" || tag == "s") {
139 VecKey key;
140 if (!track || !(line >> key.time >> key.value[0] >> key.value[1] >> key.value[2]) ||
141 !complete(line) || !finite(key.time) || !finite(key.value[0]) || !finite(key.value[1]) || !finite(key.value[2]))
142 return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid vector key", path);
143 (tag == "p" ? track->position : track->scale).push_back(key);
144 } else if (tag == "r") {
145 RotKey key;
146 if (!track || !(line >> key.time >> key.value[0] >> key.value[1] >> key.value[2] >> key.value[3]) ||
147 !complete(line) || !finite(key.time) || !finite(key.value[0]) || !finite(key.value[1]) ||
148 !finite(key.value[2]) || !finite(key.value[3]))
149 return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid rotation key", path);
150 track->rotation.push_back(key);
151 } else if (tag == "end") {
152 if (!complete(line)) return detail::failure<Fixture>(DiagnosticCode::ParseError, "invalid end", path);
153 ended = true; break;
154 } else {
155 return detail::failure<Fixture>(DiagnosticCode::ParseError, "unknown animation fixture record", path);
156 }
157 }
158 if (!magic || !ended || result.bones.size() != declaredBones || result.clipName.empty() || result.rate <= 0)
159 return detail::failure<Fixture>(DiagnosticCode::ParseError, "animation fixture is incomplete");
160 for (const auto& bone : result.bones)
161 if (!bone.hasBind) return detail::failure<Fixture>(DiagnosticCode::ParseError, "bone bind is missing", bone.name);
162 for (const auto& value : result.tracks)
163 if (value.position.size() != value.expectedPosition || value.rotation.size() != value.expectedRotation ||
164 value.scale.size() != value.expectedScale)
165 return detail::failure<Fixture>(DiagnosticCode::ParseError, "track key count does not match declaration");
166 std::set<std::uint32_t> trackBones;
167 std::uint64_t decodedBytes = 0;
168 const auto validTimes = [&](const auto& keys) {
169 float previous = -1.0f;
170 for (const auto& key : keys) {
171 if (key.time < 0 || key.time > result.duration || key.time < previous) return false;
172 previous = key.time;
173 }
174 return true;
175 };
176 for (const auto& value : result.tracks) {
177 if (!trackBones.emplace(value.bone).second || !validTimes(value.position) ||
178 !validTimes(value.rotation) || !validTimes(value.scale))
179 return detail::failure<Fixture>(DiagnosticCode::Conflict,
180 "tracks must be unique and key times canonical");
181 const std::uint64_t keys = value.position.size() + value.rotation.size() + value.scale.size();
182 if (keys > request.limits.maximumDecodedBytes / 24 ||
183 decodedBytes > request.limits.maximumDecodedBytes - keys * 24)
184 return detail::failure<Fixture>(DiagnosticCode::InvalidArgument,
185 "animation key data exceeds decoded budget");
186 decodedBytes += keys * 24;
187 }
188 return Result<Fixture>::success(std::move(result));
189}
