13T* borrowed(const ssq::Object& object) {
14 if (object.getType() != ssq::Type::INSTANCE || object.getTypeTag() != eve::script::detail::squirrelTypeHash<T*>())
15 return nullptr;
16 try {
17 return object.to<T*>();

48ResultRef<Coupling3D> resolve(Coupling3D* coupling) {
49 if (coupling && coupling->isValid()) return ResultRef<Coupling3D>::success(std::ref(*coupling));
50 return ResultRef<Coupling3D>::failure(
51 Diagnostic::error(DiagnosticCode::StaleHandle, "Transmission is absent or destroyed"));

53ResultRef<Coupling3D> resolve(CouplingObserver* observer) {
54 if (observer) {
55 const auto worldGuard = observer->worldLifetime.lock();
56 const auto instanceGuard = observer->instanceLifetime.lock();
57 if (worldGuard && instanceGuard) return observer->owner->resolveCoupling3D(observer->id);
58 }
59 return ResultRef<Coupling3D>::failure(
60 Diagnostic::error(DiagnosticCode::StaleHandle, "Transmission reference is stale"));
61}
62int identity(Coupling3D* coupling) { return coupling ? coupling->getId() : 0; }
63int identity(CouplingObserver* observer) { return observer ? observer->id : 0; }
64Result<void> remove(Coupling3D* coupling) {
65 return coupling ? coupling->destroy()
66 : Result<void>::failure(Diagnostic::error(DiagnosticCode::StaleHandle, "Transmission is absent"));
67}
68Result<void> remove(CouplingObserver* observer) {
69 auto target = resolve(observer);

71}
72template <class T>
73void exposeOperations(ssq::Class& coupling, HSQUIRRELVM vm) {
74 coupling.addFunc("getId", [vm](T* self) { return identity(receiver(vm, self)); });
75 coupling.addFunc("isValid", [vm](T* self) { return resolve(receiver(vm, self)).ok(); });
76 coupling.addFunc("setEnabled", [vm](T* self, bool enabled) {
77 auto target = resolve(receiver(vm, self));
78 return target ? script::projectResult(vm, target.value().get().setEnabled(enabled))
79 : script::projectStatusResult(vm, target.status());
80 });
81 coupling.addFunc("destroy", [vm](T* self) { return script::projectResult(vm, remove(receiver(vm, self))); });
82 coupling.addFunc("sample", [vm](T* self) {
83 auto target = resolve(receiver(vm, self));
84 if (!target) return script::projectStatusResult(vm, target.status());
