32Result<void> Graphics::replaceShaderFromSpv(Shader &shader,
33 const std::vector<uint32_t> &vertSpv,
34 const std::vector<uint32_t> &fragSpv) {
35 auto shaderIt = std::find_if(ownedShaders.begin(), ownedShaders.end(),
36 [&](const std::unique_ptr<Shader> &owned) {
37 return owned.get() == &shader;
38 });
39 if (shaderIt == ownedShaders.end() || !shader.gpuHandle)
40 return reloadFailure(DiagnosticCode::StaleHandle,
41 "shader is not a live resource owned by this Graphics instance",
42 "shader");
43 if (fragSpv.empty())
44 return reloadFailure(DiagnosticCode::InvalidArgument,
45 "fragment SPIR-V must not be empty", "fragSpv");
46 if (fragSpv.front() != 0x07230203 ||
47 (!vertSpv.empty() && vertSpv.front() != 0x07230203))
48 return reloadFailure(DiagnosticCode::ParseError, "SPIR-V magic mismatch", "source");
49
50 auto *current = static_cast<GpuShader *>(shader.gpuHandle);
51 auto gpuIt = std::find_if(ownedGpuShaders.begin(), ownedGpuShaders.end(),
52 [&](const std::unique_ptr<GpuShader> &owned) {
53 return owned.get() == current;
54 });
55 if (gpuIt == ownedGpuShaders.end())
56 return reloadFailure(DiagnosticCode::StaleHandle,
57 "shader GPU resource is no longer owned by this Graphics instance",
58 "shader.gpuHandle");
59
60 std::vector<uint32_t> vert = vertSpv;
61 if (vert.empty()) {
62 if (!current->isMesh3D)
63 vert.assign(textured_vert_spv, textured_vert_spv + textured_vert_spv_count);
64 else if (current->isHair3D)
65 vert.assign(mesh3d_hair_vert_spv,
66 mesh3d_hair_vert_spv + mesh3d_hair_vert_spv_count);
67 else
68 vert.assign(mesh3d_vert_spv, mesh3d_vert_spv + mesh3d_vert_spv_count);
69 }
70
71 GpuShader candidate;
72 candidate.isMesh3D = current->isMesh3D;
73 candidate.isHair3D = current->isHair3D;
74 candidate.pipelineLayout = current->pipelineLayout;
75 candidate.owner = &shader;
76 try {
77 if (!candidate.isMesh3D) {
78 candidate.swapchainPipeline =
79 createTexturedStylePipeline(vert, fragSpv, renderpass, candidate.pipelineLayout);
80 if (offscreenRenderPass)
81 candidate.offscreenPipeline = createTexturedStylePipeline(
82 vert, fragSpv, offscreenRenderPass, candidate.pipelineLayout);
83 } else if (candidate.isHair3D) {
84 candidate.mesh3dPipeline = createMesh3DHairPipeline(
85 vert, fragSpv, candidate.pipelineLayout, activeScenePass(), activeSceneSamples());
86 } else {
87 candidate.mesh3dPipeline = createMesh3DStylePipeline(
88 vert, fragSpv, candidate.pipelineLayout, activeScenePass(), activeSceneSamples());
89 candidate.mesh3dXrayPipeline = createMesh3DXrayPipeline(
90 vert, fragSpv, candidate.pipelineLayout, activeScenePass(), activeSceneSamples());
91 }
92 } catch (const std::exception &error) {
93 destroyCandidate(device, candidate);
94 return reloadFailure(DiagnosticCode::Failed,
95 std::string("failed to prepare replacement pipeline: ") + error.what(),
96 "pipeline");
97 } catch (...) {
98 destroyCandidate(device, candidate);
99 return reloadFailure(DiagnosticCode::Failed,
100 "failed to prepare replacement pipeline", "pipeline");
101 }
102
103 waitForSharedGpuResources();
104 if (current->swapchainPipeline) device->destroyPipeline(current->swapchainPipeline);
105 if (current->offscreenPipeline) device->destroyPipeline(current->offscreenPipeline);
106 if (current->mesh3dPipeline) device->destroyPipeline(current->mesh3dPipeline);
107 if (current->mesh3dXrayPipeline) device->destroyPipeline(current->mesh3dXrayPipeline);
108 current->swapchainPipeline = candidate.swapchainPipeline;
109 current->offscreenPipeline = candidate.offscreenPipeline;
110 current->mesh3dPipeline = candidate.mesh3dPipeline;
111 current->mesh3dXrayPipeline = candidate.mesh3dXrayPipeline;
112 shader.setSpirv(std::move(vert), fragSpv);
113 return Result<void>::success();
114}

123Result<void> Graphics::replaceShaderFromGlsl(Shader &shader, const std::string &vertGlsl,
124 const std::string &fragGlsl) {
125#if defined(_WIN32)
126 (void)shader;
127 (void)vertGlsl;
128 (void)fragGlsl;
129 return reloadFailure(DiagnosticCode::Unsupported,
130 "runtime GLSL replacement is unavailable on Windows Vulkan; compile "
131 "SPIR-V before publication",
132 "source");
133#else
134 if (fragGlsl.empty())
135 return reloadFailure(DiagnosticCode::InvalidArgument,
136 "fragment GLSL must not be empty", "fragGlsl");
137 auto shaderIt = std::find_if(ownedShaders.begin(), ownedShaders.end(),
138 [&](const std::unique_ptr<Shader> &owned) {
139 return owned.get() == &shader;
140 });
141 if (shaderIt == ownedShaders.end() || !shader.gpuHandle)
142 return reloadFailure(DiagnosticCode::StaleHandle,
143 "shader is not a live resource owned by this Graphics instance",
144 "shader");
145 auto *current = static_cast<GpuShader *>(shader.gpuHandle);
146 if (current->isHair3D)
147 return reloadFailure(DiagnosticCode::Unsupported,
148 "runtime GLSL replacement does not yet provide the hair vertex "
149 "contract; publish SPIR-V stages instead",
150 "shader.kind");
151
152 Shader *candidate = nullptr;
153 try {
154 candidate = shader.getKind() == Shader::Kind::eMesh3D
155 ? newMeshShader(vertGlsl, fragGlsl)
156 : newShader(vertGlsl, fragGlsl);
157 } catch (const Exception &error) {
158 return reloadFailure(DiagnosticCode::ParseError, error.what(), "source");
159 } catch (const std::exception &error) {
160 return reloadFailure(DiagnosticCode::Failed, error.what(), "compiler");
161 } catch (...) {
162 return reloadFailure(DiagnosticCode::Failed, "GLSL compilation failed", "compiler");
163 }
164
165 auto replaced = replaceShaderFromSpv(shader, candidate->vertexSpirv(),
166 candidate->fragmentSpirv());
167 const bool released = releaseShader(candidate);
168 if (released) delete candidate;
169 if (!released && replaced.ok())
170 return reloadFailure(DiagnosticCode::InvariantViolation,
171 "temporary compiled shader could not be released", "candidate");
172 return replaced;
173#endif
174}
