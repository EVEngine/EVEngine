59Result<ActionGuideCandidateMeta> decodeActionGuideCandidateMeta(const Value& value) {
60 auto unknown = rejectUnknown(
61 value, {"schema", "schemaVersion", "sourceGuideRevision", "shotId", "model", "seed", "promptHash"},
62 "candidate");
63 if (!unknown) return Result<ActionGuideCandidateMeta>::failure(unknown.status());
64 auto schema = text(value, "schema", "schema");
65 if (!schema) return Result<ActionGuideCandidateMeta>::failure(schema.status());
66 auto version = integer(value, "schemaVersion", "schemaVersion");
67 if (!version) return Result<ActionGuideCandidateMeta>::failure(version.status());
68 if (schema.value() != kActionGuideCandidateMetaSchemaId)
69 return invalid<ActionGuideCandidateMeta>("unexpected candidate schema", "schema");
70 if (version.value() != static_cast<std::int64_t>(kActionGuideCandidateMetaSchemaVersion))
71 return invalid<ActionGuideCandidateMeta>("unsupported candidate schema version", "schemaVersion");
72 auto revision = integer(value, "sourceGuideRevision", "sourceGuideRevision");
73 if (!revision) return Result<ActionGuideCandidateMeta>::failure(revision.status());
74 if (revision.value() < 0)
75 return invalid<ActionGuideCandidateMeta>("sourceGuideRevision must be >= 0", "sourceGuideRevision");
76 auto model = text(value, "model", "model");
77 if (!model) return Result<ActionGuideCandidateMeta>::failure(model.status());
78 if (model.value().empty()) return invalid<ActionGuideCandidateMeta>("model must be non-empty", "model");
79 auto seed = integer(value, "seed", "seed");
80 if (!seed) return Result<ActionGuideCandidateMeta>::failure(seed.status());
81 auto promptHash = text(value, "promptHash", "promptHash");
82 if (!promptHash) return Result<ActionGuideCandidateMeta>::failure(promptHash.status());
83 if (promptHash.value().empty())
84 return invalid<ActionGuideCandidateMeta>("promptHash must be non-empty", "promptHash");
85
86 ActionGuideCandidateMeta meta;
87 meta.sourceGuideRevision = static_cast<std::uint64_t>(revision.value());
88 meta.model = std::move(model).takeValue();
89 meta.seed = seed.value();
90 meta.promptHash = std::move(promptHash).takeValue();
91 if (field(value, "shotId")) {
92 auto shotId = text(value, "shotId", "shotId");
93 if (!shotId) return Result<ActionGuideCandidateMeta>::failure(shotId.status());
94 meta.shotId = std::move(shotId).takeValue();
95 }
96 return Result<ActionGuideCandidateMeta>::success(std::move(meta));
97}

99Result<Value> encodeActionGuideCandidateMeta(const ActionGuideCandidateMeta& meta) {
100 if (meta.model.empty()) return invalid<Value>("model must be non-empty", "model");
101 if (meta.promptHash.empty()) return invalid<Value>("promptHash must be non-empty", "promptHash");
102 Value::Object root{{"schema", std::string(kActionGuideCandidateMetaSchemaId)},
103 {"schemaVersion", static_cast<std::int64_t>(kActionGuideCandidateMetaSchemaVersion)},
104 {"sourceGuideRevision", static_cast<std::int64_t>(meta.sourceGuideRevision)},
105 {"model", meta.model},

107 {"promptHash", meta.promptHash}};
108 if (!meta.shotId.empty()) root.emplace("shotId", meta.shotId);
109 return Result<Value>::success(Value(std::move(root)));
110}

112Result<void> validateActionGuideCandidates(const ActionGuideIngestRequest& request) {
113 if (request.pack.shots.empty()) return invalidVoid("guide pack has no shots", "pack.shots");
114 if (request.frames.size() != request.pack.shots.size())
115 return invalidVoid("candidate frame count must equal pack.shots size", "frames");
116 const auto canvasW = static_cast<std::uint32_t>(request.pack.canvas.width);
117 const auto canvasH = static_cast<std::uint32_t>(request.pack.canvas.height);
118 if (!canvasW || !canvasH) return invalidVoid("pack canvas is empty", "pack.canvas");
119
120 for (size_t i = 0; i < request.frames.size(); ++i) {
121 const auto& frame = request.frames[i];
122 const auto path = "frames[" + std::to_string(i) + "]";
123 if (frame.meta.sourceGuideRevision != request.pack.revision)
124 return conflictVoid("candidate sourceGuideRevision does not match pack.revision",
125 path + ".sourceGuideRevision");
126 if (!frame.meta.shotId.empty() && frame.meta.shotId != request.pack.shots[i].shotId)
127 return invalidVoid("candidate shotId does not match pack.shots order", path + ".shotId");
128 if (frame.width != canvasW || frame.height != canvasH)
129 return invalidVoid("candidate frame size must match pack.canvas", path + ".size");
130 if (!frame.rgba8 || frame.rgba8->size() != size_t(frame.width) * frame.height * 4)
131 return invalidVoid("candidate RGBA buffer size mismatch", path + ".rgba8");
132
133 bool hasOpaque = false;
134 for (size_t p = 0; p + 3 < frame.rgba8->size(); p += 4) {
135 if ((*frame.rgba8)[p + 3] > 8) hasOpaque = true;
136 }
137 if (!hasOpaque) return invalidVoid("candidate frame has no opaque content", path);
138
139 if (request.requireTransparentEdge && request.pack.canvas.transparent) {
140 // Design contract: transparent canvas candidates keep a clear 1px border so
141 // Sprite sequence import does not bake solid edges into the atlas.
142 auto alphaAt = [&](std::uint32_t x, std::uint32_t y) -> std::uint8_t {
143 return (*frame.rgba8)[(static_cast<size_t>(y) * frame.width + x) * 4 + 3];
144 };
145 bool edgeClear = true;
146 for (std::uint32_t x = 0; x < frame.width && edgeClear; ++x) {
147 if (alphaAt(x, 0) >= 250 || alphaAt(x, frame.height - 1) >= 250) edgeClear = false;
148 }
149 for (std::uint32_t y = 0; y < frame.height && edgeClear; ++y) {
150 if (alphaAt(0, y) >= 250 || alphaAt(frame.width - 1, y) >= 250) edgeClear = false;
151 }
152 if (!edgeClear)
153 return invalidVoid("transparent canvas requires a fully clear 1px border", path + ".alphaEdge");
154 }
155 }
156 return Result<void>::success();
157}

159Result<ActionGuideIngestDraft> buildActionGuideIngestDraft(const ActionGuideIngestRequest& request) {
160 auto valid = validateActionGuideCandidates(request);
161 if (!valid) return Result<ActionGuideIngestDraft>::failure(valid.status());
162
163 ActionGuideIngestDraft draft;
164 draft.sourceGuideRevision = request.pack.revision;
165 draft.clipName = request.outputClipName.empty() ? request.pack.actionBinding.clipName : request.outputClipName;
166 if (draft.clipName.empty()) draft.clipName = "guide.clip";
167 draft.fps = request.pack.actionBinding.fps > 0.0 ? request.pack.actionBinding.fps : 12.0;
168 draft.loop = request.pack.actionBinding.loop;
169 draft.frameCount = static_cast<std::int64_t>(request.frames.size());
170 draft.sequenceFirst = 0;
171 draft.sequenceLast = draft.frameCount - 1;
172 draft.sequencePattern = "frames/frame_{n}.png";
173
174 const double frameDt = 1.0 / draft.fps;
175 double durationSeconds =
176 draft.frameCount > 0 ? (static_cast<double>(draft.frameCount - 1) * frameDt + frameDt) : frameDt;
177 if (!request.pack.shots.empty()) {
178 const double lastShot = request.pack.shots.back().timeSeconds;
179 if (std::isfinite(lastShot) && lastShot + frameDt > durationSeconds) durationSeconds = lastShot + frameDt;
180 }
181 double activeStart = 0.0;
182 double recoverStart = durationSeconds;
183 for (const auto& mark : request.pack.actionBinding.phaseMarks) {
184 if (mark.name == "active.start") activeStart = mark.timeSeconds;
185 if (mark.name == "recover.start") recoverStart = mark.timeSeconds;
186 }
187 if (!(activeStart >= 0.0 && recoverStart >= activeStart && durationSeconds >= recoverStart)) {
188 activeStart = 0.0;
189 recoverStart = durationSeconds * 0.5;
190 }
191 const std::int64_t windupNs = secondsToNs(activeStart);
192 const std::int64_t activeNs = secondsToNs(recoverStart - activeStart);
193 const std::int64_t recoverNs = secondsToNs(durationSeconds - recoverStart);
194
195 Value::Array frames;
196 Value::Array attribution;
197 for (size_t i = 0; i < request.frames.size(); ++i) {
198 const auto file = "frames/frame_" + std::to_string(i) + ".png";
199 frames.emplace_back(file);
200 Value::Object attr{
201 {"index", static_cast<std::int64_t>(i)},
202 {"shotId", request.pack.shots[i].shotId},
203 {"model", request.frames[i].meta.model},
204 {"seed", request.frames[i].meta.seed},
205 {"promptHash", request.frames[i].meta.promptHash},
206 {"sourceGuideRevision", static_cast<std::int64_t>(request.frames[i].meta.sourceGuideRevision)},
207 {"file", file}};
208 attribution.emplace_back(std::move(attr));
209 }
210
211 Value::Array phaseMarks;
212 for (const auto& mark : request.pack.actionBinding.phaseMarks)
213 phaseMarks.emplace_back(Value::Object{{"name", mark.name}, {"timeSeconds", mark.timeSeconds}});
214
215 draft.clipDescriptor =
216 Value::object({{"schema", std::string(kActionGuideClipRecipeSchemaId)},
217 {"schemaVersion", static_cast<std::int64_t>(kActionGuideClipRecipeSchemaVersion)},
218 {"clipName", draft.clipName},
219 {"fps", draft.fps},
220 {"loop", draft.loop},
221 {"sequencePattern", draft.sequencePattern},
222 {"sequenceFirst", draft.sequenceFirst},
223 {"sequenceLast", draft.sequenceLast},
224 {"frames", Value(std::move(frames))}});
225
226 const auto abilityId = request.pack.actionBinding.abilityId.empty() ? ("ability:" + draft.clipName)
227 : request.pack.actionBinding.abilityId;
228 draft.abilityDraft = Value::object(
229 {{"schema", std::string(kActionGuideAbilityDraftSchemaId)},
230 {"schemaVersion", static_cast<std::int64_t>(kActionGuideAbilityDraftSchemaVersion)},
231 {"sourceGuideRevision", static_cast<std::int64_t>(draft.sourceGuideRevision)},
232 {"abilityId", abilityId},
233 {"clipName", draft.clipName},
234 {"fps", draft.fps},
235 {"loop", draft.loop},
236 {"sequencePattern", draft.sequencePattern},
237 {"sequenceFirst", draft.sequenceFirst},
238 {"sequenceLast", draft.sequenceLast},
239 {"timing", Value::object({{"windupNs", windupNs}, {"activeNs", activeNs}, {"recoverNs", recoverNs}})},
240 {"phaseMarks", Value(std::move(phaseMarks))},
241 {"commitAbility", false}});
242
243 draft.attribution = Value::object({{"schema", std::string("eve.action.guide-ingest-attribution")},
244 {"schemaVersion", static_cast<std::int64_t>(1)},
245 {"frames", Value(std::move(attribution))}});
246 return Result<ActionGuideIngestDraft>::success(std::move(draft));
247}
