149bool decodeLayerData(Poco::JSON::Object::Ptr layerObj, size_t expectedCount,
150 std::vector<uint32_t> &out, std::string *error) {
151 if (!layerObj || !layerObj->has("data")) {
152 if (error) *error = "missing data";
153 return false;
154 }
155
156 try {
157 if (layerObj->isArray("data")) {
158 auto arr = layerObj->getArray("data");
159 if (!arr) {
160 if (error) *error = "invalid data array";
161 return false;
162 }
163 out.resize(arr->size());
164 for (size_t i = 0; i < arr->size(); ++i)
165 out[i] = uint32_t(asInt(arr->get(static_cast<unsigned int>(i)), 0));
166 if (expectedCount > 0 && out.size() != expectedCount) {
167 if (error) *error = "gid count mismatch";
168 return false;
169 }
170 return true;
171 }
172 } catch (...) {
173 }
174
175 std::string encoding = layerObj->has("encoding") ? asString(layerObj->get("encoding")) : "";
176 std::string compression =
177 layerObj->has("compression") ? asString(layerObj->get("compression")) : "";
178 if (encoding != "base64") {
179 if (error) *error = "unsupported encoding";
180 return false;
181 }
182 if (compression == "zstd") {
183 if (error) *error = "zstd not supported; export as zlib";
184 return false;
185 }
186
187 const std::string b64 = asString(layerObj->get("data"));
188 size_t decodedLen = 0;
189 std::unique_ptr<char[]> decoded(eve::data::decode("base64", b64.data(), b64.size(), decodedLen));
190 if (!decoded) {
191 if (error) *error = "base64 decode failed";
192 return false;
193 }
194
195 const char *bytes = decoded.get();
196 size_t nbytes = decodedLen;
197 std::unique_ptr<char[]> inflated;
198 if (!compression.empty()) {
199 if (compression != "zlib" && compression != "gzip") {
200 if (error) *error = "unsupported compression";
201 return false;
202 }
203 size_t rawsize = expectedCount * 4;
204 try {
205 inflated.reset(eve::data::decompress(compression, bytes, nbytes, rawsize));
206 } catch (...) {
207 if (error) *error = "decompress failed";
208 return false;
209 }
210 if (!inflated) {
211 if (error) *error = "decompress failed";
212 return false;
213 }
214 bytes = inflated.get();
215 nbytes = rawsize;
216 }
217
218 if (expectedCount > 0 && nbytes != expectedCount * 4) {
219 if (error) *error = "gid byte length mismatch";
220 return false;
221 }
222 const size_t count = nbytes / 4;
223 out.resize(count);
224 for (size_t i = 0; i < count; ++i) {
225 const unsigned char *p = reinterpret_cast<const unsigned char *>(bytes) + i * 4;
226 out[i] = uint32_t(p[0]) | (uint32_t(p[1]) << 8) | (uint32_t(p[2]) << 16) |
227 (uint32_t(p[3]) << 24);
228 }
229 return true;
230}

272bool parseOrientation(Poco::JSON::Object::Ptr root, TileLayer::Config *cfg, std::string *error) {
273 if (!root || !cfg) return false;
274 if (!root->has("orientation")) return true;
275 const std::string o = asString(root->get("orientation"));
276 if (o.empty() || o == "orthogonal")
277 cfg->orientation = MapOrientation::Orthogonal;
278 else if (o == "isometric")
279 cfg->orientation = MapOrientation::Isometric;
280 else if (o == "staggered")
281 cfg->orientation = MapOrientation::Staggered;
282 else if (o == "hexagonal")
283 cfg->orientation = MapOrientation::Hexagonal;
284 else {
285 if (error) *error = "unknown orientation: " + o;
286 return false;
287 }
288 if (root->has("staggeraxis")) {
289 const std::string a = asString(root->get("staggeraxis"));
290 cfg->staggerAxis = (a == "x") ? StaggerAxis::X : StaggerAxis::Y;
291 }
292 if (root->has("staggerindex")) {
293 const std::string i = asString(root->get("staggerindex"));
294 cfg->staggerIndex = (i == "even") ? StaggerIndex::Even : StaggerIndex::Odd;
295 }
296 if (root->has("hexsidelength"))
297 cfg->hexSideLength = asFloat(root->get("hexsidelength"), 0.f);
298 return true;

302 if (!layer || !root) return false;
303
304 int mapW = layer->getMapWidth();
305 int mapH = layer->getMapHeight();
306 float tileW = layer->getTileWidth();
307 float tileH = layer->getTileHeight();
308
309 if (root->has("width")) mapW = asInt(root->get("width"), mapW);
310 if (root->has("height")) mapH = asInt(root->get("height"), mapH);
311 if (root->has("tilewidth")) tileW = asFloat(root->get("tilewidth"), tileW);
312 else if (root->has("tileWidth")) tileW = asFloat(root->get("tileWidth"), tileW);
313 if (root->has("tileheight")) tileH = asFloat(root->get("tileheight"), tileH);
314 else if (root->has("tileHeight")) tileH = asFloat(root->get("tileHeight"), tileH);
315
316 if (mapW != layer->getMapWidth() || mapH != layer->getMapHeight()) layer->resize(mapW, mapH);
317 layer->setTileSize(tileW, tileH);
318
319 if (!parseOrientation(root, &(*layer->config()), error)) return false;
320
321 float ox = layer->getX(), oy = layer->getY();
322 if (readVec2(root, "origin", ox, oy))
323 layer->setOrigin(ox, oy);
324 else {
325 if (root->has("x")) ox = asFloat(root->get("x"), ox);
326 if (root->has("y")) oy = asFloat(root->get("y"), oy);
327 if (root->has("x") || root->has("y")) layer->setOrigin(ox, oy);
328 }
329
330 if (root->has("layer"))
331 layer->setLayer(asInt(root->get("layer"), layer->getLayer()));
332 if (root->has("visible"))

334 if (root->has("autoReload"))

336
337 float r = 1, g = 1, b = 1, a = 1;
338 if (readVec4(root, "tint", r, g, b, a)) layer->setTint(r, g, b, a);
339
340 if (root->has("tileset") && !root->isArray("tileset")) {
341 try {
342 auto o = root->getObject("tileset");
343 if (o && !o->has("source")) applyTileset(layer, readTilesetObject(o));
344 } catch (...) {
345 }
346 } else if (root->has("tilesets")) {
347 try {
348 auto arr = root->getArray("tilesets");
349 if (arr) {
350 for (size_t i = 0; i < arr->size(); ++i) {
351 try {
352 auto o = arr->getObject(static_cast<unsigned int>(i));
353 if (o && o->has("source")) continue;
354 applyTileset(layer, readTilesetObject(o));
355 break;
356 } catch (...) {
357 }
358 }
359 }
360 } catch (...) {
361 }
362 } else if (root->has("image") || root->has("texture")) {
363 applyTileset(layer, readTilesetObject(root));
364 }
365 return true;
366}

368bool applyFlatLayerData(TileLayer *layer, Poco::JSON::Object::Ptr root, std::string *error) {
369 const int mapW = layer->getMapWidth();

371 const size_t need = size_t(std::max(0, mapW) * std::max(0, mapH));
372 if (need == 0) {

374 return false;

376 std::vector<uint32_t> data;

379 gids.assign(need, 0u);
380 const size_t n = std::min(need, data.size());
381 for (size_t i = 0; i < n; ++i) gids[i] = data[i];
382 return true;

474std::vector<TileLayer *> loadMapObject(Poco::JSON::Object::Ptr root, const std::string &path,
475 eve::filesystem::Filesystem *fs,
476 std::vector<MapObject> *objects, std::string *error) {
477 std::vector<TileLayer *> out;
478 if (!root) {
479 if (error) *error = "config root must be object";
480 return out;
481 }
482
483 int mapW = 10, mapH = 10;
484 float tileW = 32.f, tileH = 32.f;
485 if (root->has("width")) mapW = asInt(root->get("width"), mapW);
486 if (root->has("height")) mapH = asInt(root->get("height"), mapH);
487 if (root->has("tilewidth")) tileW = asFloat(root->get("tilewidth"), tileW);
488 else if (root->has("tileWidth")) tileW = asFloat(root->get("tileWidth"), tileW);
489 if (root->has("tileheight")) tileH = asFloat(root->get("tileheight"), tileH);
490 else if (root->has("tileHeight")) tileH = asFloat(root->get("tileHeight"), tileH);
491
492 // Validate orientation early on a temp config.
493 TileLayer::Config orientCheck;
494 if (!parseOrientation(root, &orientCheck, error)) return out;
495
496 TilesetInfo defaultTs = readDefaultTileset(root);
497 if (objects) objects->clear();
498
499 auto bindResource = [&](TileLayer *layer) {
500 auto res = layer->resource();
501 res->path = path;
502 if (!path.empty()) {
503 res->modtime = fileModtime(path);
504 if (fs) {
505 fs->watch(path);
506 if (auto *hot =
507 eve::ModuleManager::getInstance<eve::filesystem::HotReload>("HotReload"))
508 hot->bind(path, "tilemap");
509 }
510 }
511 if (root->has("autoReload"))
512 res->autoReload = asBool(root->get("autoReload"), true);
513 };
514
515 if (root->has("layers")) {
516 try {
517 auto arr = root->getArray("layers");
518 if (arr) {
519 int sort = 0;
520 for (size_t i = 0; i < arr->size(); ++i) {
521 Poco::JSON::Object::Ptr lo;
522 try {
523 lo = arr->getObject(static_cast<unsigned int>(i));
524 } catch (...) {
525 continue;
526 }
527 if (objects && isObjectGroup(lo)) {
528 parseObjectGroup(lo, *objects);
529 continue;
530 }
531 if (!isTileLayerObject(lo)) continue;
532 TileLayer *layer = TileLayer::createLayer(mapW, mapH, tileW, tileH);
533 if (!applyMapGlobals(layer, root, error)) {
534 abandonLayers(out);
535 layer->clear();
536 layer->setVisible(false);
537 return {};
538 }
539 applyTileset(layer, defaultTs);
540 if (!applyOneLayerObject(layer, lo, mapW, mapH, error)) {
541 abandonLayers(out);
542 layer->clear();
543 layer->setVisible(false);
544 return {};
545 }
546 if (!lo->has("layer") && !lo->has("id")) layer->setLayer(sort);
547 ++sort;
548 bindResource(layer);
549 out.push_back(layer);
550 }
551 }
552 } catch (...) {
553 }
554 if (!out.empty()) return out;
555 }
556
557 TileLayer *layer = TileLayer::createLayer(mapW, mapH, tileW, tileH);
558 if (!applyMapGlobals(layer, root, error)) {
559 layer->clear();
560 layer->setVisible(false);
561 return {};
562 }
563 applyTileset(layer, defaultTs);
564 if (root->has("data")) {
565 if (!applyFlatLayerData(layer, root, error)) {
566 layer->clear();
567 layer->setVisible(false);
568 return {};
569 }
570 }
571 bindResource(layer);
572 out.push_back(layer);
573 return out;
574}

578bool applyConfigDocument(TileLayer *layer, data::JsonDocument *doc) {
579 if (!layer || !doc || !doc->isObject()) return false;
580 auto root = doc->object();
581 if (!root) return false;

588 auto arr = root->getArray("layers");

590 for (size_t i = 0; i < arr->size(); ++i) {
591 Poco::JSON::Object::Ptr lo;
592 try {
593 lo = arr->getObject(static_cast<unsigned int>(i));
594 } catch (...) {

611 auto *dm = eve::data::DataModule::create();
612 std::string err;
613 std::unique_ptr<data::JsonDocument> doc(dm->decodeJson(json, &err));
614 if (!doc) {
615 if (error) *error = err.empty() ? "invalid json" : err;

617 }
618 if (!doc->isObject()) {

623 if (!root) {

628 if (root->has("layers")) {
629 try {
630 auto arr = root->getArray("layers");
631 if (arr) {
632 for (size_t i = 0; i < arr->size(); ++i) {
633 Poco::JSON::Object::Ptr lo;

642 }
643 }
644 } catch (...) {
645 }
646 }
647 if (root->has("data")) return applyFlatLayerData(layer, root, error);

651bool loadConfigFile(TileLayer *layer, const std::string &path, std::string *error) {
652 if (!layer || path.empty()) {
653 if (error) *error = "empty path";
654 return false;
655 }
656 auto *fs = eve::ModuleManager::getInstance<eve::filesystem::Filesystem>("Filesystem");
657 if (!fs) fs = eve::filesystem::Filesystem::create();
658
659 std::unique_ptr<eve::filesystem::FileData> data;
660 try {
661 data.reset(fs->read(path));
662 } catch (...) {
663 if (error) *error = "read failed: " + path;
664 return false;
665 }
666 if (!data || data->getSize() == 0) {
667 if (error) *error = "empty file: " + path;
668 return false;
669 }
670
671 std::string text(static_cast<const char *>(data->getData()), data->getSize());
672 if (!applyConfigText(layer, text, error)) return false;
673
674 auto res = layer->resource();
675 res->path = path;
676 res->modtime = fileModtime(path);
677 fs->watch(path);
678 if (auto *hot = eve::ModuleManager::getInstance<eve::filesystem::HotReload>("HotReload"))
679 hot->bind(path, "tilemap");
680 return true;
681}
