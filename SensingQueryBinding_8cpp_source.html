21Result<Value> query(SensingWorld& world, const std::string& json) {
22 auto request = decodeSensingQuery(json);
23 if (!request) return Result<Value>::failure(request.status());
24 const auto injected = services();
25 auto result = world.query(request.value().origin, request.value().spec, injected.lineOfSight);
26 if (!result) return Result<Value>::failure(result.status());
27 EnvironmentQueryResult output;
28 output.option = "subjects";
29 for (const auto& candidate : result.value().ranked()) {
30 auto point = candidate.space == CoordinateSpace::World3D
31 ? WorldPoint::world3D(candidate.x, candidate.y, candidate.z)
32 : WorldPoint::world2D(candidate.x, candidate.y);
33 if (!point) return Result<Value>::failure(point.status());
34 output.ranked.push_back({{output.ranked.size(), TargetLocation{point.value()}, candidate.id}, candidate.score});
35 }
36 return Result<Value>::success(output.toValue());
37}
