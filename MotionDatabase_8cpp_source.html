78void MotionDatabase::extractFeature(AnimClip* clip, float time, float dtSample, std::vector<float>& out, float& rootX,
79 float& rootZ, float& rootYaw, float& velX, float& velZ) const {
80 out.assign(static_cast<size_t>(featureSize_), 0.f);
81 clip->sample(time, &scratchPose_, skeleton_);
82 scratchPose_.computeWorld(skeleton_);
83
84 const int root = rootBone_;
85 rootX = scratchPose_.getWorldPositionX(root);
86 rootZ = scratchPose_.getWorldPositionZ(root);
87 rootYaw = yawFromQuat(scratchPose_.getWorldRotationX(root), scratchPose_.getWorldRotationY(root),
88 scratchPose_.getWorldRotationZ(root), scratchPose_.getWorldRotationW(root));
89
90 // Velocity from nearby sample.
91 const float t1 = time + std::max(dtSample, 1e-3f);
92 AnimPose next;
93 clip->sample(t1, &next, skeleton_);
94 next.computeWorld(skeleton_);
95 const float nX = next.getWorldPositionX(root);
96 const float nZ = next.getWorldPositionZ(root);
97 const float dtt = std::max(dtSample, 1e-3f);
98 velX = (nX - rootX) / dtt;
99 velZ = (nZ - rootZ) / dtt;
100
101 float cs = std::cos(rootYaw);
102 float sn = std::sin(rootYaw);
103 // Character-space: rotate world xz by -yaw
104 auto toLocal = [&](float wx, float wz, float& lx, float& lz) {
105 const float dx = wx - rootX;
106 const float dz = wz - rootZ;
107 lx = dx * cs + dz * sn;
108 lz = -dx * sn + dz * cs;
109 };
110
111 out[0] = velX * cs + velZ * sn;
112 out[1] = -velX * sn + velZ * cs;
113
114 const float horizons[3] = {0.33f, 0.66f, 1.0f};
115 for (int h = 0; h < 3; ++h) {
116 AnimPose fut;
117 clip->sample(time + horizons[h], &fut, skeleton_);
118 fut.computeWorld(skeleton_);
119 float lx, lz;
120 toLocal(fut.getWorldPositionX(root), fut.getWorldPositionZ(root), lx, lz);
121 out[2 + h * 2] = lx;
122 out[2 + h * 2 + 1] = lz;
123 }
124
125 {
126 AnimPose fut;
127 clip->sample(time + 1.0f, &fut, skeleton_);
128 fut.computeWorld(skeleton_);
129 const float fyaw = yawFromQuat(fut.getWorldRotationX(root), fut.getWorldRotationY(root),
130 fut.getWorldRotationZ(root), fut.getWorldRotationW(root));
131 float fx, fz;
132 yawToForward(fyaw - rootYaw, fx, fz);
133 out[8] = fx;
134 out[9] = fz;
135 }
136
137 int base = 10;
138 for (int bone : featureBones_) {
139 float lx, lz;
140 const float wy = scratchPose_.getWorldPositionY(bone);
141 toLocal(scratchPose_.getWorldPositionX(bone), scratchPose_.getWorldPositionZ(bone), lx, lz);
142 out[static_cast<size_t>(base)] = lx;
143 out[static_cast<size_t>(base + 1)] = wy;
144 out[static_cast<size_t>(base + 2)] = lz;
145 base += 3;
146 }
147}

149void MotionDatabase::bake() {
150 if (clips_.empty()) throw Exception("MotionDatabase.bake: no clips");
151 if (featureBones_.empty()) {
152 // Default: all bones except root.
153 for (int i = 1; i < skeleton_->getBoneCount(); ++i) addFeatureBone(i);
154 if (featureBones_.empty() && skeleton_->getBoneCount() > 0) addFeatureBone(0);
155 }
156 computeFeatureSize();
157 frames_.clear();
158
159 for (int ci = 0; ci < getClipCount(); ++ci) {
160 AnimClip* clip = clips_[static_cast<size_t>(ci)];
161 const float rate = clip->getSampleRate() > 0.f ? clip->getSampleRate() : 30.f;
162 const float dt = 1.f / rate;
163 const float dur = clip->getDuration();
164 if (dur <= 0.f) {
165 Frame f;
166 f.clipIndex = ci;
167 f.time = 0.f;
168 extractFeature(clip, 0.f, dt, f.feature, f.rootX, f.rootZ, f.rootYaw, f.velX, f.velZ);
169 frames_.push_back(std::move(f));
170 continue;
171 }
172 for (float t = 0.f; t < dur - 1e-5f; t += dt) {
173 Frame f;
174 f.clipIndex = ci;
175 f.time = t;
176 extractFeature(clip, t, dt, f.feature, f.rootX, f.rootZ, f.rootYaw, f.velX, f.velZ);
177 frames_.push_back(std::move(f));
178 }
179 }
180 featureMean_.assign(static_cast<size_t>(featureSize_), 0.f);
181 featureInvStd_.assign(static_cast<size_t>(featureSize_), 0.f);
182 for (const Frame& frame : frames_)
183 for (int i = 0; i < featureSize_; ++i)
184 featureMean_[static_cast<size_t>(i)] += frame.feature[static_cast<size_t>(i)];
185 const float invCount = 1.f / static_cast<float>(frames_.size());
186 for (float& mean : featureMean_) mean *= invCount;
187 for (const Frame& frame : frames_)
188 for (int i = 0; i < featureSize_; ++i) {
189 const float d = frame.feature[static_cast<size_t>(i)] - featureMean_[static_cast<size_t>(i)];
190 featureInvStd_[static_cast<size_t>(i)] += d * d;
191 }
192 for (float& invStd : featureInvStd_) invStd = 1.f / std::sqrt(invStd * invCount + 1e-6f);
193 for (Frame& frame : frames_) normalizeFeature(frame.feature);
194 baked_ = true;
195}

197void MotionDatabase::normalizeFeature(std::vector<float>& feature) const {
198 if (static_cast<int>(feature.size()) != featureSize_ || static_cast<int>(featureMean_.size()) != featureSize_)
199 return;
200 for (int i = 0; i < featureSize_; ++i)
201 feature[static_cast<size_t>(i)] = (feature[static_cast<size_t>(i)] - featureMean_[static_cast<size_t>(i)]) *
202 featureInvStd_[static_cast<size_t>(i)];
203}
