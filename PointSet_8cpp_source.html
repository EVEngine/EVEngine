306Result<void> PointSet::assignPointIds(std::uint64_t namespaceId) {
307 if (namespaceId == 0)
308 return Result<void>::failure(Diagnostic::error(
309 DiagnosticCode::InvalidArgument, "point id namespace must be non-zero", "namespaceId"));
310 std::unordered_set<std::uint64_t> occupied;
311 occupied.reserve(points_.size());
312 for (const auto& point : points_)
313 if (point.id != 0 && !occupied.insert(point.id).second)
314 return Result<void>::failure(
315 Diagnostic::error(DiagnosticCode::Conflict, "point set contains duplicate ids", "points"));
316 PointSet staged = *this;
317 for (std::size_t row = 0; row < staged.points_.size(); ++row) {
318 if (staged.points_[row].id != 0) continue;
319 std::uint64_t candidate = derivePointId(namespaceId, row);
320 while (!occupied.insert(candidate).second) candidate = derivePointId(candidate, row + 1);
321 staged.points_[row].id = candidate;
322 }
323 *this = std::move(staged);
324 return Result<void>::success();
325}

464PointSet sampleGridPoints(int width, int depth, float spacing, uint32_t seed, float jitter) {
465 PointSet output;
466 if (width <= 0 || depth <= 0 || spacing <= 0.f) return output;
467 jitter = std::clamp(jitter, 0.f, 1.f);
468 output.reserve(size_t(width) * size_t(depth));
469 const float extent = spacing * jitter * 0.5f;
470 for (int z = 0; z < depth; ++z) {
471 for (int x = 0; x < width; ++x) {
472 const uint32_t pointSeed = mix32(seed ^ uint32_t(z * width + x));
473 ProcgenPoint point;
474 point.x = float(x) * spacing + (unitFloat(pointSeed) * 2.f - 1.f) * extent;
475 point.z = float(z) * spacing + (unitFloat(pointSeed ^ 0xa511e9b3u) * 2.f - 1.f) * extent;
476 point.seed = pointSeed ? pointSeed : 1u;
477 point.id = derivePointId((std::uint64_t(seed) << 32u) | 0x47524944u,
478 std::uint64_t(z) * std::uint64_t(width) + std::uint64_t(x));
479 const int appended = output.appendPoint(std::move(point));
480 (void)appended;
481 }
482 }
483 return output;
484}

486PointSet poissonDiskPoints(int width, int depth, float radius, uint32_t seed, int maxPoints) {
487 PointSet output;
488 if (width <= 0 || depth <= 0 || radius <= 0.f) return output;
489 maxPoints = std::max(0, maxPoints);
490
491 // Bridson's algorithm: candidate annulus [r, 2r], grid cell side r / sqrt(2).
492 const float cell = radius * 0.70710678118f;
493 const int gridW = std::max(1, int(std::ceil(float(width) / cell)));
494 const int gridH = std::max(1, int(std::ceil(float(depth) / cell)));
495 std::vector<int> grid(size_t(gridW) * size_t(gridH), -1);
496
497 std::vector<float> xs, ys;
498 std::mt19937 rng(seed);
499 std::uniform_real_distribution<float> unit(0.f, 1.f);
500
501 const auto cellIndex = [&](float x, float y) -> int {
502 int gx = int(x / cell);
503 int gy = int(y / cell);
504 gx = std::max(0, std::min(gx, gridW - 1));
505 gy = std::max(0, std::min(gy, gridH - 1));
506 return gy * gridW + gx;
507 };
508 const auto inBounds = [&](float x, float y) {
509 return x >= 0.f && x <= float(width) && y >= 0.f && y <= float(depth);
510 };
511 const auto tooClose = [&](float x, float y, float r2) {
512 const int gx = std::max(0, std::min(int(x / cell), gridW - 1));
513 const int gy = std::max(0, std::min(int(y / cell), gridH - 1));
514 for (int oy = -2; oy <= 2; ++oy) {
515 for (int ox = -2; ox <= 2; ++ox) {
516 const int nx = gx + ox, ny = gy + oy;
517 if (nx < 0 || nx >= gridW || ny < 0 || ny >= gridH) continue;
518 const int idx = grid[ny * gridW + nx];
519 if (idx < 0) continue;
520 const float dx = x - xs[size_t(idx)], dy = y - ys[size_t(idx)];
521 if (dx * dx + dy * dy < r2) return true;
522 }
523 }
524 return false;
525 };
526
527 // Seed with one random interior point.
528 float fx = unit(rng) * float(width);
529 float fy = unit(rng) * float(depth);
530 xs.push_back(fx);
531 ys.push_back(fy);
532 grid[cellIndex(fx, fy)] = 0;
533 std::vector<int> active{0};
534
535 const float r2 = radius * radius;
536 constexpr int kTries = 30;
537 while (!active.empty() && int(xs.size()) < maxPoints) {
538 const int pick = int(unit(rng) * float(active.size()));
539 const int base = active[size_t(std::min(pick, int(active.size()) - 1))];
540 bool found = false;
541 for (int k = 0; k < kTries; ++k) {
542 const float theta = unit(rng) * 6.28318530718f;
543 const float dist = radius * (1.f + unit(rng));
544 const float nx = xs[size_t(base)] + std::cos(theta) * dist;
545 const float ny = ys[size_t(base)] + std::sin(theta) * dist;

777 const float pitch = pitchDegrees * degreesToRadians;
778 const float yaw = yawDegrees * degreesToRadians;
779 const float roll = rollDegrees * degreesToRadians;
780 const float cp = std::cos(pitch), sp = std::sin(pitch);
781 const float cy = std::cos(yaw), sy = std::sin(yaw);
782 const float cr = std::cos(roll), sr = std::sin(roll);
783 const float rolledX = x * cr - y * sr;
784 const float rolledY = x * sr + y * cr;
785 const float pitchedY = rolledY * cp - z * sp;
786 const float pitchedZ = rolledY * sp + z * cp;
787 x = rolledX * cy - pitchedZ * sy;
788 y = pitchedY;
789 z = rolledX * sy + pitchedZ * cy;
