21bool squirrelToState(HSQUIRRELVM vm, SQInteger index, StateValue& out) {
22 switch (sq_gettype(vm, index)) {
23 case OT_NULL: out = StateValue::null(); return true;
24 case OT_INTEGER: {
25 SQInteger value = 0;
26 if (SQ_FAILED(sq_getinteger(vm, index, &value))) return false;
27 out = StateValue::integer(value);
28 return true;
29 }
30 case OT_FLOAT: {
31 SQFloat value = 0;
32 if (SQ_FAILED(sq_getfloat(vm, index, &value))) return false;
33 out = StateValue::number(value);
34 return true;
35 }
36 case OT_BOOL: {
37 SQBool value = SQFalse;
38 if (SQ_FAILED(sq_getbool(vm, index, &value))) return false;
39 out = StateValue::boolean(value != 0);
40 return true;
41 }
42 case OT_STRING: {
43 const SQChar* value = nullptr;
44 if (SQ_FAILED(sq_getstring(vm, index, &value))) return false;
45 out = StateValue::string(value ? value : "");
46 return true;
47 }
48 case OT_TABLE: {
49 out = StateValue::object();
50 const SQInteger absolute = index > 0 ? index : sq_gettop(vm) + index + 1;
51 sq_pushnull(vm);
52 while (SQ_SUCCEEDED(sq_next(vm, absolute))) {
53 const SQChar* key = nullptr;
54 StateValue value;
55 const bool ok = sq_gettype(vm, -2) == OT_STRING && SQ_SUCCEEDED(sq_getstring(vm, -2, &key)) && key &&
56 squirrelToState(vm, -1, value);
57 sq_pop(vm, 2);
58 if (!ok) {
59 sq_pop(vm, 1);
60 return false;
61 }
62 out.set(key, std::move(value));
63 }
64 sq_pop(vm, 1);
65 return true;
66 }
67 default: return false;
68 }

71void pushState(HSQUIRRELVM vm, const StateValue& value) {
72 switch (value.kind()) {
73 case StateValue::Kind::Null: sq_pushnull(vm); break;
74 case StateValue::Kind::Int: sq_pushinteger(vm, value.asInt()); break;
75 case StateValue::Kind::Float: sq_pushfloat(vm, static_cast<SQFloat>(value.asDouble())); break;
76 case StateValue::Kind::Bool: sq_pushbool(vm, value.asBool() ? SQTrue : SQFalse); break;
77 case StateValue::Kind::String: sq_pushstring(vm, value.asString().c_str(), value.asString().size()); break;
78 case StateValue::Kind::Array:
79 sq_newarray(vm, 0);
80 for (size_t i = 0; i < value.arraySize(); ++i) {
81 pushState(vm, value.at(i));
82 sq_arrayappend(vm, -2);
83 }
84 break;
85 case StateValue::Kind::Object:
86 sq_newtable(vm);
87 for (const auto& key : value.keys()) {
88 sq_pushstring(vm, key.c_str(), key.size());
89 pushState(vm, *value.find(key));
90 sq_newslot(vm, -3, SQFalse);
91 }
92 break;
93 }

119 [[nodiscard]] std::string_view name() const noexcept override { return "dialogue.choice"; }
120 [[nodiscard]] eve::Result<void> prepare(const eve::transaction::TransactionContext& context) override {
121 if (context.transactionId().empty())
122 return dialogueFailure(eve::DiagnosticCode::InvalidArgument,
123 "dialogue choice transaction requires a transaction id", "transactionId");
124 if (phase_ != Phase::Idle)
125 return dialogueFailure(eve::DiagnosticCode::Conflict, "dialogue choice is not idle",
126 "transaction.lifecycle");
127 phase_ = Phase::Prepared;
128 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
129 }
130 [[nodiscard]] eve::Result<void> commit(const eve::transaction::TransactionContext&) override {
131 if (phase_ != Phase::Prepared)
132 return dialogueFailure(eve::DiagnosticCode::Conflict, "dialogue choice has no prepared stage",
133 "transaction.lifecycle");
134 auto selected = runner_.selectRouteForTransaction(routeId_);
135 if (!selected.ok()) {
136 std::string error = selected.status().describe();
137 std::string restoreError;
138 if (!runner_.restoreState(before_, &restoreError) && error.empty()) error = restoreError;
139 return dialogueFailure(eve::DiagnosticCode::Failed,
140 error.empty() ? "dialogue choice selection failed" : error, "route");
141 }
142 phase_ = Phase::Committed;
143 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
144 }
145 [[nodiscard]] eve::Result<void> rollback(const eve::transaction::TransactionContext&) override {
146 if (phase_ != Phase::Prepared)
147 return dialogueFailure(eve::DiagnosticCode::Conflict, "dialogue choice has no prepared stage to roll back",
148 "transaction.lifecycle");
149 phase_ = Phase::RolledBack;
150 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
151 }
152 [[nodiscard]] eve::Result<void> compensate(const eve::transaction::TransactionContext&) override {
153 if (phase_ != Phase::Committed)
154 return dialogueFailure(eve::DiagnosticCode::Conflict, "dialogue choice is not committed",
155 "transaction.lifecycle");
156 std::string error;
157 if (!runner_.restoreState(before_, &error))
158 return dialogueFailure(eve::DiagnosticCode::Failed,
159 error.empty() ? "dialogue choice compensation failed" : error, "route");
160 phase_ = Phase::RolledBack;
161 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
162 }

164private:
165 enum class Phase : std::uint8_t { Idle, Prepared, Committed, RolledBack };
166 ConversationRunner& runner_;
167 std::string routeId_;
168 StateValue before_;
169 Phase phase_ = Phase::Idle;

262int DialogueFlow::reloadFromDnut(const std::string& source, const std::string& path) {
263 const size_t hash = std::hash<std::string>{}(source);
264 if (const auto cached = sourceHashes_.find(path); cached != sourceHashes_.end() && cached->second == hash) {
265 lastLoadChanged_ = false;
266 failureMessage_.clear();
267 return static_cast<int>(sourceAssets_[path].size());
268 }
269 std::vector<ConversationAsset> compiled;
270 std::vector<ConversationDiagnostic> candidateDiagnostics;
271 if (!compileDnutConversations(source, path, compiled, candidateDiagnostics)) {
272 diagnostics_ = std::move(candidateDiagnostics);
273 failureMessage_ = diagnostics_.empty() ? "conversation compilation failed" : diagnostics_.front().message;
274 lastLoadChanged_ = false;
275 return 0;
276 }
277
278 std::vector<ConversationAsset> candidate = assets_;
279 if (const auto old = sourceAssets_.find(path); old != sourceAssets_.end()) {
280 candidate.erase(std::remove_if(candidate.begin(), candidate.end(),
281 [&](const auto& asset) {
282 return std::find(old->second.begin(), old->second.end(), asset.id) !=
283 old->second.end();
284 }),
285 candidate.end());
286 }
287 std::vector<std::string> compiledIds;
288 for (auto& asset : compiled) {
289 compiledIds.push_back(asset.id);
290 auto existing =
291 std::find_if(candidate.begin(), candidate.end(), [&](const auto& old) { return old.id == asset.id; });
292 if (existing == candidate.end())
293 candidate.push_back(std::move(asset));
294 else
295 *existing = std::move(asset);
296 }
297 if (!lintConversationWorkspace(candidate, path, candidateDiagnostics)) {
298 diagnostics_ = std::move(candidateDiagnostics);
299 failureMessage_ = diagnostics_.empty() ? "conversation workspace lint failed" : diagnostics_.front().message;
300 lastLoadChanged_ = false;
301 return 0;
302 }
303
304 StateValue activeState;
305 const bool hadActive = runner_.isActive();
306 if (hadActive) runner_.captureState(activeState);
307 std::vector<ConversationAsset> previous = assets_;
308 runner_.stop();
309 assets_ = std::move(candidate);
310 if (hadActive) {
311 StateValue migrated = activeState;
312 std::string restoreError;
313 if (!migrations_.migrate(
314 migrated, [this](const std::string& id) { return find(id); }, &restoreError) ||
315 !runner_.restoreState(migrated, &restoreError)) {
316 assets_ = std::move(previous);
317 runner_.restoreState(activeState, nullptr);
318 failureMessage_ = "conversation hot reload rolled back: " + restoreError;
319 lastLoadChanged_ = false;
320 return 0;
321 }
322 }
323 diagnostics_ = std::move(candidateDiagnostics);
324 sourceHashes_[path] = hash;
325 sourceAssets_[path] = std::move(compiledIds);
326 lastLoadChanged_ = true;
327 failureMessage_.clear();
328 return static_cast<int>(compiled.size());
329}

573 const auto* node = runner_.currentNode();
574 if (!node) return runner_.selectRouteForTransaction(routeId);
575 const ConversationRoute* route = nullptr;

577 if (candidate.first == routeId) {

586 "dialogue choice state mutations require a StatePatch-compatible provider",
587 "route.stateMutations");

592 return dialogueFailure(eve::DiagnosticCode::Failed,
593 "dialogue choice could not capture its pre-transaction state", "route");

596 std::unique_ptr<DialogueStateMutationParticipant> state;
597 std::vector<eve::transaction::ITransactionParticipant*> effects{&selection};
598 if (!route->stateMutations.empty()) {
599 state = std::make_unique<DialogueStateMutationParticipant>(*stateMutationProvider_, route->stateMutations);
600 effects.push_back(state.get());
601 }
602 auto committed = paymentAdapter_.execute(nextTransactionId("choice"), route->payment, effects);
603 if (!committed) return eve::Result<void>::failure(committed.status());
604 (void)std::move(committed).takeValue();

612CommandResponse DialogueFlow::dispatchCommand(const CommandRequest& request) {
613 const bool transactional = !request.payment.empty() || !request.stateMutations.empty();
614 CommandRequestHandler* legacy =
615 request.kind == CommandRequestKind::Operation ? &operationRequestHandler_ : &gameplayActionHandler_;
616 if (!transactional && *legacy) return (*legacy)(request);
617
618 if (!transactional && !commandParticipantFactory_) {
619 CommandResponse response;
620 response.status = CommandResponse::Status::Failed;
621 response.error = request.kind == CommandRequestKind::Operation
622 ? "dialogue operation handler is not configured"
623 : "dialogue gameplay-action handler is not configured";
624 return response;
625 }
626
627 if (!request.payment.empty() && !commandParticipantFactory_ && request.stateMutations.empty()) {
628 CommandResponse response;
629 response.status = CommandResponse::Status::Failed;
630 response.error = "payment-bearing dialogue command requires an Action/operation participant";
631 return response;
632 }
633
634 std::vector<std::unique_ptr<eve::transaction::ITransactionParticipant>> owned;
635 std::vector<eve::transaction::ITransactionParticipant*> effects;
636 if (commandParticipantFactory_) {
637 auto participant = commandParticipantFactory_(request);
638 if (!participant) {
639 CommandResponse response;
640 response.status = CommandResponse::Status::Failed;
641 response.error = participant.status().describe();
642 return response;
643 }
644 auto action = std::move(participant).takeValue();
645 if (!action) {
646 CommandResponse response;
647 response.status = CommandResponse::Status::Failed;
648 response.error = "dialogue command participant factory returned null";
649 return response;
650 }
651 effects.push_back(action.get());
652 owned.push_back(std::move(action));
653 }
654 if (!request.stateMutations.empty()) {
655 if (!stateMutationProvider_) {
656 CommandResponse response;
657 response.status = CommandResponse::Status::Failed;
658 response.error = "dialogue command state mutations require a StatePatch-compatible provider";
659 return response;
660 }
661 auto state =
662 std::make_unique<DialogueStateMutationParticipant>(*stateMutationProvider_, request.stateMutations);
663 effects.push_back(state.get());
664 owned.push_back(std::move(state));
665 }
666 auto committed = paymentAdapter_.execute(nextTransactionId("command"), request.payment, effects);
667 if (!committed) {
668 CommandResponse response;
669 response.status = CommandResponse::Status::Failed;
670 response.error = committed.status().describe();
671 return response;
672 }
673 (void)std::move(committed).takeValue();
674 CommandResponse response;
675 response.status = CommandResponse::Status::Completed;
676 return response;
677}

748StateValue DialogueFlow::evaluate(const std::string& expression, const StateValue& bindings, const StateValue& locals) {
749 if (expression == "else") return StateValue::boolean(true);
750 if (!vm_ || !hasEvaluator_) return StateValue::boolean(false);
751 const SQInteger top = sq_gettop(vm_);
752 sq_pushobject(vm_, evaluator_);
753 sq_newtable(vm_);
754 sq_pushstring(vm_, "expression", -1);
755 sq_pushstring(vm_, expression.c_str(), expression.size());
756 sq_newslot(vm_, -3, SQFalse);
757 sq_pushstring(vm_, "bindings", -1);
758 pushState(vm_, bindings);
759 sq_newslot(vm_, -3, SQFalse);
760 sq_pushstring(vm_, "locals", -1);
761 pushState(vm_, locals);
762 sq_newslot(vm_, -3, SQFalse);
763 if (SQ_FAILED(sq_call(vm_, 1, SQTrue, SQTrue))) {
764 sq_settop(vm_, top);
765 return StateValue::boolean(false);
766 }
767 StateValue result;
768 if (!squirrelToState(vm_, -1, result)) result = StateValue::boolean(false);
769 sq_settop(vm_, top);
770 return result;
