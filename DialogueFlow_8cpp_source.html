31bool buildPoolWorkspace(const std::unordered_map<std::string, DataValue>& sources, DataValue& root,
32 std::string& error) {
33 DataValue::Object combined;
34 for (const auto& [sourceId, sourceRoot] : sources) {
35 const DataValue* pools = sourceRoot.find("pools");
36 if (!pools || !pools->isObject()) continue;
37 for (const auto& poolId : pools->keys()) {
38 if (combined.contains(poolId)) {
39 error = "pool '" + poolId + "' has multiple source owners (including '" + sourceId + "')";
40 return false;
41 }
42 combined.emplace(poolId, *pools->find(poolId));
43 }
44 }
45 root = DataValue::object({{"pools", DataValue::object(std::move(combined))}});
46 return true;
47}

49bool squirrelToState(HSQUIRRELVM vm, SQInteger index, StateValue& out) {
50 switch (sq_gettype(vm, index)) {
51 case OT_NULL: out = StateValue::null(); return true;
52 case OT_INTEGER: {
53 SQInteger value = 0;
54 if (SQ_FAILED(sq_getinteger(vm, index, &value))) return false;
55 out = StateValue::integer(value);
56 return true;
57 }
58 case OT_FLOAT: {
59 SQFloat value = 0;
60 if (SQ_FAILED(sq_getfloat(vm, index, &value))) return false;
61 out = StateValue::number(value);
62 return true;
63 }
64 case OT_BOOL: {
65 SQBool value = SQFalse;
66 if (SQ_FAILED(sq_getbool(vm, index, &value))) return false;
67 out = StateValue::boolean(value != 0);
68 return true;
69 }
70 case OT_STRING: {
71 const SQChar* value = nullptr;
72 if (SQ_FAILED(sq_getstring(vm, index, &value))) return false;
73 out = StateValue::string(value ? value : "");
74 return true;
75 }
76 case OT_TABLE: {
77 out = StateValue::object();
78 const SQInteger absolute = index > 0 ? index : sq_gettop(vm) + index + 1;
79 sq_pushnull(vm);
80 while (SQ_SUCCEEDED(sq_next(vm, absolute))) {
81 const SQChar* key = nullptr;
82 StateValue value;
83 const bool ok = sq_gettype(vm, -2) == OT_STRING && SQ_SUCCEEDED(sq_getstring(vm, -2, &key)) && key &&
84 squirrelToState(vm, -1, value);
85 sq_pop(vm, 2);
86 if (!ok) {
87 sq_pop(vm, 1);
88 return false;
89 }
90 out.set(key, std::move(value));
91 }
92 sq_pop(vm, 1);
93 return true;
94 }
95 default: return false;
96 }

99void pushState(HSQUIRRELVM vm, const StateValue& value) {
100 switch (value.kind()) {
101 case StateValue::Kind::Null: sq_pushnull(vm); break;
102 case StateValue::Kind::Int: sq_pushinteger(vm, value.asInt()); break;
103 case StateValue::Kind::Float: sq_pushfloat(vm, static_cast<SQFloat>(value.asDouble())); break;
104 case StateValue::Kind::Bool: sq_pushbool(vm, value.asBool() ? SQTrue : SQFalse); break;
105 case StateValue::Kind::String: sq_pushstring(vm, value.asString().c_str(), value.asString().size()); break;
106 case StateValue::Kind::Array:
107 sq_newarray(vm, 0);
108 for (size_t i = 0; i < value.arraySize(); ++i) {
109 pushState(vm, value.at(i));
110 sq_arrayappend(vm, -2);
111 }
112 break;
113 case StateValue::Kind::Object:
114 sq_newtable(vm);
115 for (const auto& key : value.keys()) {
116 sq_pushstring(vm, key.c_str(), key.size());
117 pushState(vm, *value.find(key));
118 sq_newslot(vm, -3, SQFalse);
119 }
120 break;
121 }

142class DialogueSelectionParticipant final : public eve::transaction::ITransactionParticipant {
143public:
144 DialogueSelectionParticipant(ConversationRunner& runner, std::string routeId, StateValue before)
145 : runner_(runner), routeId_(std::move(routeId)), before_(std::move(before)) {}
146
147 [[nodiscard]] std::string_view name() const noexcept override { return "dialogue.choice"; }
148 [[nodiscard]] eve::Result<void> prepare(const eve::transaction::TransactionContext& context) override {
149 if (context.transactionId().empty())
150 return dialogueFailure(eve::DiagnosticCode::InvalidArgument,
151 "dialogue choice transaction requires a transaction id", "transactionId");
152 if (phase_ != Phase::Idle)
153 return dialogueFailure(eve::DiagnosticCode::Conflict, "dialogue choice is not idle",
154 "transaction.lifecycle");
155 phase_ = Phase::Prepared;
156 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
157 }
158 [[nodiscard]] eve::Result<void> commit(const eve::transaction::TransactionContext&) override {
159 if (phase_ != Phase::Prepared)
160 return dialogueFailure(eve::DiagnosticCode::Conflict, "dialogue choice has no prepared stage",
161 "transaction.lifecycle");
162 auto selected = runner_.selectRouteForTransaction(routeId_);
163 if (!selected.ok()) {
164 std::string error = selected.status().describe();
165 auto restored = runner_.restoreStateChecked(before_);
166 if (!restored && error.empty()) error = restored.status().describe();
167 return dialogueFailure(eve::DiagnosticCode::Failed,
168 error.empty() ? "dialogue choice selection failed" : error, "route");
169 }
170 phase_ = Phase::Committed;
171 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
172 }
173 [[nodiscard]] eve::Result<void> rollback(const eve::transaction::TransactionContext&) override {
174 if (phase_ != Phase::Prepared)
175 return dialogueFailure(eve::DiagnosticCode::Conflict, "dialogue choice has no prepared stage to roll back",
176 "transaction.lifecycle");
177 phase_ = Phase::RolledBack;
178 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
179 }
180 [[nodiscard]] eve::Result<void> compensate(const eve::transaction::TransactionContext&) override {
181 if (phase_ != Phase::Committed)
182 return dialogueFailure(eve::DiagnosticCode::Conflict, "dialogue choice is not committed",
183 "transaction.lifecycle");
184 auto restored = runner_.restoreStateChecked(before_);
185 if (!restored)
186 return dialogueFailure(eve::DiagnosticCode::Failed,
187 restored.status().describe(), "route");
188 phase_ = Phase::RolledBack;
189 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
190 }
191
192private:
193 enum class Phase : std::uint8_t { Idle, Prepared, Committed, RolledBack };
194 ConversationRunner& runner_;
195 std::string routeId_;
196 StateValue before_;
197 Phase phase_ = Phase::Idle;
198};

301 if (const auto it = sourceTexts_.find(path); it != sourceTexts_.end() && it->second == source) {
302 lastLoadChanged_ = false;
303 failureMessage_.clear();
304 return static_cast<int>(sourceAssets_[path].size());
305 }
306 diagnostics_.clear();
307 auto compiledDocument = compileDnutDocument(source, path, diagnostics_);
308 if (!compiledDocument) {
309 failureMessage_ = diagnostics_.empty() ? "conversation compilation failed" : diagnostics_.front().message;
310 return 0;
311 }
312 DnutDocument document = std::move(compiledDocument).takeValue();
313 std::vector<ConversationAsset> compiled = std::move(document.conversations);
314 for (const auto& asset : compiled) {
315 const auto owner = assetSources_.find(asset.id);
316 if (owner != assetSources_.end() && owner->second != path) {
317 diagnostics_.push_back({ConversationDiagnostic::Severity::Error, path, 0,
318 "conversation '" + asset.id + "' is already owned by '" + owner->second + "'"});
319 failureMessage_ = diagnostics_.back().message;
320 return 0;
321 }
322 }
323 if (runner_.isActive()) {
324 failureMessage_ = "cannot load a dnut source while a conversation is active; use transactional reload";
325 return 0;
326 }
327 auto candidatePoolSources = sourcePools_;
328 candidatePoolSources[path] = std::move(document.poolRoot);
329 DataValue poolWorkspace;
330 if (!buildPoolWorkspace(candidatePoolSources, poolWorkspace, failureMessage_)) return 0;
331 Dialogue* dialogue = Dialogue::create();
332 if (!dialogue || (dialogue->replacePoolsFromData(poolWorkspace) == 0 && !dialogue->getLastPoolsError().empty())) {
333 failureMessage_ = dialogue ? dialogue->getLastPoolsError() : "dialogue pool runtime is unavailable";
334 return 0;
335 }
336 runner_.stop();
337 if (const auto old = sourceAssets_.find(path); old != sourceAssets_.end()) {
338 assets_.erase(std::remove_if(assets_.begin(), assets_.end(),
339 [&](const auto& asset) {
340 return std::find(old->second.begin(), old->second.end(), asset.id) !=
341 old->second.end();
342 }),
343 assets_.end());
344 }
345 std::vector<std::string> compiledIds;
346 for (auto& asset : compiled) {
347 compiledIds.push_back(asset.id);

349 if (it == assets_.end())
350 assets_.push_back(std::move(asset));
351 else
352 *it = std::move(asset);
353 }
354 sourceTexts_[path] = source;
355 sourcePools_ = std::move(candidatePoolSources);
356 sourceAssets_[path] = std::move(compiledIds);
357 for (const auto& id : sourceAssets_[path]) assetSources_[id] = path;
358 lastLoadChanged_ = true;
359 failureMessage_.clear();
360 return static_cast<int>(compiled.size());
361}

363int DialogueFlow::reloadDnutImpl(const std::string& source, const std::string& path) {
364 if (const auto cached = sourceTexts_.find(path); cached != sourceTexts_.end() && cached->second == source) {
365 lastLoadChanged_ = false;
366 failureMessage_.clear();
367 return static_cast<int>(sourceAssets_[path].size());
368 }
369 std::vector<ConversationDiagnostic> candidateDiagnostics;
370 auto compiledDocument = compileDnutDocument(source, path, candidateDiagnostics);
371 if (!compiledDocument) {
372 diagnostics_ = std::move(candidateDiagnostics);
373 failureMessage_ = diagnostics_.empty() ? "conversation compilation failed" : diagnostics_.front().message;
374 lastLoadChanged_ = false;
375 return 0;
376 }
377 DnutDocument document = std::move(compiledDocument).takeValue();
378 std::vector<ConversationAsset> compiled = std::move(document.conversations);
379 for (const auto& asset : compiled) {
380 const auto owner = assetSources_.find(asset.id);
381 if (owner != assetSources_.end() && owner->second != path) {
382 candidateDiagnostics.push_back({ConversationDiagnostic::Severity::Error, path, 0,
383 "conversation '" + asset.id + "' is already owned by '" +
384 owner->second + "'"});
385 diagnostics_ = std::move(candidateDiagnostics);
386 failureMessage_ = diagnostics_.back().message;
387 lastLoadChanged_ = false;
388 return 0;
389 }
390 }
391
392 std::vector<ConversationAsset> candidate = assets_;
393 if (const auto old = sourceAssets_.find(path); old != sourceAssets_.end()) {
394 candidate.erase(std::remove_if(candidate.begin(), candidate.end(),
395 [&](const auto& asset) {
396 return std::find(old->second.begin(), old->second.end(), asset.id) !=
397 old->second.end();
398 }),
399 candidate.end());
400 }
401 std::vector<std::string> compiledIds;
402 for (auto& asset : compiled) {
403 compiledIds.push_back(asset.id);
404 auto existing =
405 std::find_if(candidate.begin(), candidate.end(), [&](const auto& old) { return old.id == asset.id; });
406 if (existing == candidate.end())
407 candidate.push_back(std::move(asset));
408 else
409 *existing = std::move(asset);
410 }
411 if (!lintConversationWorkspace(candidate, path, candidateDiagnostics)) {
412 diagnostics_ = std::move(candidateDiagnostics);
413 failureMessage_ = diagnostics_.empty() ? "conversation workspace lint failed" : diagnostics_.front().message;
414 lastLoadChanged_ = false;
415 return 0;
416 }
417
418 StateValue activeState;
419 const bool hadActive = runner_.isActive();
420 if (hadActive) {
421 auto captured = runner_.captureStateChecked();
422 if (!captured) {
423 failureMessage_ = captured.status().describe();
424 return 0;
425 }
426 activeState = std::move(captured).takeValue();
427 }
428 std::vector<ConversationAsset> previous = assets_;
429 auto candidatePoolSources = sourcePools_;
430 candidatePoolSources[path] = std::move(document.poolRoot);
431 DataValue candidatePoolWorkspace;
432 if (!buildPoolWorkspace(candidatePoolSources, candidatePoolWorkspace, failureMessage_)) return 0;
433 Dialogue* dialogue = Dialogue::create();
434 if (!dialogue ||
435 (dialogue->replacePoolsFromData(candidatePoolWorkspace) == 0 && !dialogue->getLastPoolsError().empty())) {
436 failureMessage_ = dialogue ? dialogue->getLastPoolsError() : "dialogue pool runtime is unavailable";
437 return 0;
438 }
439 runner_.stop();
440 assets_ = std::move(candidate);
441 if (hadActive) {
442 StateValue migrated = activeState;
443 std::string restoreError;
444 auto migratedResult = migrations_.migrate(
445 migrated, [this](const std::string& id) { return find(id); });
446 bool restored = migratedResult.ok();
447 if (restored) {
448 migrated = std::move(migratedResult).takeValue();
449 auto restoreResult = runner_.restoreStateChecked(migrated);
450 restored = restoreResult.ok();
451 if (!restored) restoreError = restoreResult.status().describe();
452 } else {
453 restoreError = migratedResult.status().describe();
454 }
455 if (!restored) {
456 assets_ = std::move(previous);
457 runner_.restoreStateChecked(activeState).ignore("restore prior validated state after reload rollback");
458 DataValue previousPools;
459 std::string ignored;
460 if (dialogue && buildPoolWorkspace(sourcePools_, previousPools, ignored))
461 dialogue->replacePoolsFromData(previousPools);
462 failureMessage_ = "conversation hot reload rolled back: " + restoreError;
463 lastLoadChanged_ = false;
464 return 0;
465 }
466 }
467 diagnostics_ = std::move(candidateDiagnostics);
468 if (const auto old = sourceAssets_.find(path); old != sourceAssets_.end())
469 for (const auto& id : old->second) assetSources_.erase(id);
470 sourceTexts_[path] = source;
471 sourcePools_ = std::move(candidatePoolSources);
472 sourceAssets_[path] = std::move(compiledIds);
473 for (const auto& id : sourceAssets_[path]) assetSources_[id] = path;
474 lastLoadChanged_ = true;
475 failureMessage_.clear();
476 return static_cast<int>(compiled.size());
477}

512eve::Result<void> DialogueFlow::lintAllChecked() {
513 diagnostics_.clear();
514 bool valid = lintConversationWorkspace(assets_, "<dialogue-workspace>", diagnostics_).ok();
515 for (const auto& asset : assets_) {
516 for (const auto& node : asset.nodes) {
517 const std::string assetPath = asset.id + "/" + node.id;
518 if (node.kind == ConversationAsset::Node::Kind::Line) {
519 if (!node.text.empty() && node.i18nKey.empty())
520 diagnostics_.push_back({ConversationDiagnostic::Severity::Warning, "<dialogue-workspace>",
521 node.sourceLine, "line has display text but no localization key",
522 "MissingLocalizationReference", node.sourceColumn, assetPath});
523 if (!node.speaker.empty() && node.voice.empty())
524 diagnostics_.push_back({ConversationDiagnostic::Severity::Warning, "<dialogue-workspace>",
525 node.sourceLine, "spoken line has no voice reference",
526 "MissingVoiceReference", node.sourceColumn, assetPath});
527 }
528 if (node.kind == ConversationAsset::Node::Kind::Choice) {
529 for (const auto& route : node.routes)
530 if (!route.text.empty() && route.i18nKey.empty())
531 diagnostics_.push_back({ConversationDiagnostic::Severity::Warning, "<dialogue-workspace>",
532 route.sourceLine, "choice route has display text but no localization key",
533 "MissingLocalizationReference", route.sourceColumn,
534 assetPath + "/" + route.id});
535 }
536 if (node.kind != ConversationAsset::Node::Kind::Command) continue;
537 const bool transactional = !node.payment.empty() || !node.stateMutations.empty();
538 const bool legacyHandler = node.commandKind == CommandRequestKind::Operation
539 ? static_cast<bool>(operationRequestHandler_)
540 : static_cast<bool>(gameplayActionHandler_);
541 bool handled = false;
542 if (!transactional)
543 handled = legacyHandler || manualCommandMode_ || static_cast<bool>(commandParticipantFactory_);
544 else
545 handled = (commandParticipantFactory_ || !node.stateMutations.empty()) &&
546 (node.stateMutations.empty() || stateMutationProvider_) &&
547 (node.payment.empty() || commandParticipantFactory_ || !node.stateMutations.empty());
548 if (!handled) {
549 diagnostics_.push_back({ConversationDiagnostic::Severity::Error, "<dialogue-workspace>",

850eve::Result<void> DialogueFlow::select(const std::string& routeId) {
851 const auto* node = runner_.currentNode();
852 if (!node) return runner_.selectRouteForTransaction(routeId);
853 const ConversationRoute* route = nullptr;
854 for (const auto& candidate : node->routes) {
855 if (candidate.first == routeId) {
856 route = &candidate;
857 break;
858 }
859 }
860 if (!route || (route->payment.empty() && route->stateMutations.empty()))
861 return runner_.selectRouteForTransaction(routeId);
862 if (!stateMutationProvider_ && !route->stateMutations.empty()) {
863 return dialogueFailure(eve::DiagnosticCode::Unsupported,
864 "dialogue choice state mutations require a StatePatch-compatible provider",
865 "route.stateMutations");
866 }
867
868 auto captured = runner_.captureStateChecked();
869 if (!captured) return eve::Result<void>::failure(captured.status());
870 StateValue before = std::move(captured).takeValue();
871 DialogueSelectionParticipant selection(runner_, routeId, std::move(before));
872 std::unique_ptr<DialogueStateMutationParticipant> state;
873 std::vector<eve::transaction::ITransactionParticipant*> effects{&selection};
874 if (!route->stateMutations.empty()) {
875 state = std::make_unique<DialogueStateMutationParticipant>(*stateMutationProvider_, route->stateMutations);
876 effects.push_back(state.get());
877 }
878 auto committed = paymentAdapter_.execute(nextTransactionId("choice"), route->payment, effects);
879 if (!committed) return eve::Result<void>::failure(committed.status());
880 (void)std::move(committed).takeValue();
881 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
882}

888CommandResponse DialogueFlow::dispatchCommand(const CommandRequest& request) {
889 const bool transactional = !request.payment.empty() || !request.stateMutations.empty();
890 CommandRequestHandler* legacy =
891 request.kind == CommandRequestKind::Operation ? &operationRequestHandler_ : &gameplayActionHandler_;
892 if (!transactional && *legacy) return (*legacy)(request);
893
894 if (!transactional && manualCommandMode_) {
895 CommandResponse response;
896 response.status = CommandResponse::Status::Blocked;
897 return response;
898 }
899
900 if (!transactional && !commandParticipantFactory_) {
901 CommandResponse response;
902 response.status = CommandResponse::Status::Failed;
903 response.error = request.kind == CommandRequestKind::Operation
904 ? "dialogue operation handler is not configured"
905 : "dialogue gameplay-action handler is not configured";
906 return response;
907 }
908
909 if (!request.payment.empty() && !commandParticipantFactory_ && request.stateMutations.empty()) {
910 CommandResponse response;
911 response.status = CommandResponse::Status::Failed;
912 response.error = "payment-bearing dialogue command requires an Action/operation participant";
913 return response;
914 }
915
916 std::vector<std::unique_ptr<eve::transaction::ITransactionParticipant>> owned;
917 std::vector<eve::transaction::ITransactionParticipant*> effects;
918 if (commandParticipantFactory_) {
919 auto participant = commandParticipantFactory_(request);
920 if (!participant) {
921 CommandResponse response;
922 response.status = CommandResponse::Status::Failed;
923 response.error = participant.status().describe();
924 return response;
925 }
926 auto action = std::move(participant).takeValue();
927 if (!action) {
928 CommandResponse response;
929 response.status = CommandResponse::Status::Failed;
930 response.error = "dialogue command participant factory returned null";
931 return response;
932 }
933 effects.push_back(action.get());
934 owned.push_back(std::move(action));
935 }
936 if (!request.stateMutations.empty()) {
937 if (!stateMutationProvider_) {
938 CommandResponse response;
939 response.status = CommandResponse::Status::Failed;
940 response.error = "dialogue command state mutations require a StatePatch-compatible provider";
941 return response;
942 }
943 auto state =
944 std::make_unique<DialogueStateMutationParticipant>(*stateMutationProvider_, request.stateMutations);
945 effects.push_back(state.get());
946 owned.push_back(std::move(state));
947 }
948 auto committed = paymentAdapter_.execute(nextTransactionId("command"), request.payment, effects);
949 if (!committed) {
950 CommandResponse response;
951 response.status = CommandResponse::Status::Failed;
952 response.error = committed.status().describe();
953 return response;
954 }
955 (void)std::move(committed).takeValue();
956 CommandResponse response;
957 response.status = CommandResponse::Status::Completed;
958 return response;
959}

1038eve::Result<StateValue> DialogueFlow::evaluate(const std::string& expression, const StateValue& bindings,
1039 const StateValue& locals) {
1040 if (expression == "else") return eve::Result<StateValue>::success(StateValue::boolean(true));
1041 if (!vm_ || !hasEvaluator_)
1042 return eve::Result<StateValue>::failure(eve::Diagnostic::error(
1043 eve::DiagnosticCode::Unsupported, "dialogue expression evaluator is not configured", "expression", {},
1044 "dialogue.expression"));
1045 const SQInteger top = sq_gettop(vm_);
1046 sq_pushobject(vm_, evaluator_);
1047 sq_pushroottable(vm_);
1048 sq_newtable(vm_);
1049 sq_pushstring(vm_, "expression", -1);
1050 sq_pushstring(vm_, expression.c_str(), expression.size());
1051 sq_newslot(vm_, -3, SQFalse);
1052 sq_pushstring(vm_, "bindings", -1);
1053 pushState(vm_, bindings);
1054 sq_newslot(vm_, -3, SQFalse);
1055 sq_pushstring(vm_, "locals", -1);
1056 pushState(vm_, locals);
1057 sq_newslot(vm_, -3, SQFalse);
1058 if (SQ_FAILED(sq_call(vm_, 2, SQTrue, SQTrue))) {
1059 sq_settop(vm_, top);
1060 return eve::Result<StateValue>::failure(eve::Diagnostic::error(
1061 eve::DiagnosticCode::Failed, "dialogue expression evaluator raised an exception", "expression", {},
1062 "dialogue.expression"));
1063 }
1064 StateValue result;
1065 if (!squirrelToState(vm_, -1, result)) {
1066 sq_settop(vm_, top);
1067 return eve::Result<StateValue>::failure(eve::Diagnostic::error(
1068 eve::DiagnosticCode::InvalidArgument, "dialogue expression returned an unsupported value", "expression",
1069 {}, "dialogue.expression"));
1070 }
1071 sq_settop(vm_, top);
1072 return eve::Result<StateValue>::success(std::move(result));
1073}
