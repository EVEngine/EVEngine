25Result<void> ControlRigEditor::initialize(const animation::AnimSkeleton& source) {
26 auto candidate = std::make_unique<Impl>(impl_->document.targetId().value());
27 candidate->skeleton = std::make_unique<animation::AnimSkeleton>();
28 animation::RigDefinition definition;
29 for (int i = 0; i < source.getBoneCount(); ++i) {
30 const auto name = source.getBoneName(i);
31 const auto& t = source.bindLocal(i);
32 const int parent = source.getParent(i);
33 const int bone = candidate->skeleton->addBone(name, parent);
34 candidate->skeleton->setBindPosition(bone, t.px, t.py, t.pz);
35 candidate->skeleton->setBindRotation(bone, t.qx, t.qy, t.qz, t.qw);
36 candidate->skeleton->setBindScale(bone, t.sx, t.sy, t.sz);
37 animation::RigControl control;
38 control.id = name;
39 control.parent = parent < 0 ? "" : source.getBoneName(parent);
40 control.initial = t;
41 control.limits.reference = t;
42 definition.controls.push_back(std::move(control));
43 animation::RigNode node;
44 node.id = "fk:" + name;
45 node.control = name;
46 node.bone = name;
47 if (parent >= 0) node.after.push_back("fk:" + source.getBoneName(parent));
48 definition.nodes.push_back(std::move(node));
49 }
50 auto op = candidate->document.replace(definition);
51 if (!op.ok()) return Result<void>::failure(op.status());
52 auto applied = candidate->document.applyDomainOperation(op.value());
53 if (!applied.ok()) return applied;
54 candidate->skeleton->applyBindPose(&candidate->input);
55 auto rebuilt = candidate->rebuild();
56 if (!rebuilt.ok()) return rebuilt;
57 if (!impl_->hostLifetime.expired()) return failure<void>("disable automation before reinitializing skeleton");
58 impl_ = std::move(candidate);
59 return Result<void>::success();
60}

62Result<void> ControlRigEditor::replace(const animation::RigDefinition& definition) {
63 if (!impl_->skeleton) return failure<void>("initialize a skeleton first");
64 auto compiled = animation::ControlRig::compile(definition, *impl_->skeleton);
65 if (!compiled.ok()) return Result<void>::failure(compiled.status());
66 animation::AnimPose preview;
67 const double time = std::min(impl_->playhead, definition.duration);
68 auto evaluated = compiled.value()->evaluate(time, impl_->input, preview);
69 if (!evaluated.ok()) return evaluated;
70 auto op = impl_->document.replace(definition);
71 if (!op.ok()) return Result<void>::failure(op.status());
72 editor::TransactionSpec spec;
73 spec.id = editor::TransactionId("control-rig:" + std::to_string(++impl_->sequence));
74 spec.target = impl_->document.targetId();
75 spec.baseRevision = impl_->document.revision();
76 spec.label = "Edit control rig";
77 auto begun = impl_->transactions.begin(std::move(spec));
78 if (!begun.ok()) return Result<void>::failure(begun.status());
79 auto appended = impl_->transactions.append(std::move(op).takeValue());
80 if (!appended.ok()) {
81 auto discarded = impl_->transactions.discard();
82 if (!discarded.ok()) return discarded;
83 return appended;
84 }
85 auto committed = impl_->transactions.commit();
86 if (!committed.ok()) {
87 auto discarded = impl_->transactions.discard();
88 if (!discarded.ok()) return discarded;
89 return Result<void>::failure(committed.status());
90 }
91 impl_->rig = std::move(compiled).takeValue();
92 impl_->pose = std::move(preview);
93 impl_->playhead = time;
94 return Result<void>::success();
95}

110Result<void> ControlRigEditor::setControl(std::string_view id, const animation::TransformTRS& transform, bool key) {
111 auto definition = impl_->document.definition();
112 for (auto& control : definition.controls)
113 if (control.id == id) {
114 if (!key) {
115 if (!control.keys.empty()) return failure<void>("animated control requires a key edit");
116 control.initial = transform;
117 } else {
118 auto found = std::lower_bound(control.keys.begin(), control.keys.end(), impl_->playhead,
119 [](const animation::RigKey& k, double t) { return k.time < t; });
120 if (found != control.keys.end() && found->time == impl_->playhead)
121 found->transform = transform;
