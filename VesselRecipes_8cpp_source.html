19 auto invalid = [] {
20 return eve::Result<MeshBuild>::failure(eve::Diagnostic::error(
21 eve::DiagnosticCode::InvalidArgument,
22 "VesselParametersInvalid: Unknown shape/material or invalid curve, thickness, height or resolution"));
23 };
24 const auto name = params.getString("shape", "planter");
25 const Shape* shape = nullptr;
26 for (const auto& candidate : shapes)
27 if (name == candidate.name) shape = &candidate;
28 if (!shape) return invalid();
29 const float missing = std::numeric_limits<float>::quiet_NaN();
30 auto read = [&](const char* key, float fallback) {
31 return params.getFloat(key, params.has(key) ? missing : fallback);
32 };
33 const float requestedHeight = read("vesselHeight", 0.f);
34 const float height = requestedHeight == 0.f ? shape->height : requestedHeight;
35 const float thickness = read("wallThickness", 0.045f);
36 const float bottom = read("bottomThickness", 0.05f);
37 const int samples = params.getInt("curveSegments", params.has("curveSegments") ? -1 : 48);
38 const int radial = params.getInt("radialSegments", params.has("radialSegments") ? -1 : 96);
39 auto material = params.getString("material", "auto");
40 if (material == "auto") material = name == "planter" ? "terracotta" : "porcelain";
41 if (material != "terracotta" && material != "glazed_ceramic" && material != "porcelain" && material != "celadon" &&
42 material != "stoneware")
43 return invalid();
44 if (!std::isfinite(height) || !std::isfinite(thickness) || !std::isfinite(bottom) || height <= 0.f ||
45 height > 1000.f || thickness < 0.0001f || bottom < 0.0001f || bottom >= height || samples < 4 ||
46 samples > 256 || radial < 3 || radial > 512)
47 return invalid();
48 std::array<float, 4> r{};
49 const char* keys[] = {"radius0", "radius1", "radius2", "radius3"};
50 for (size_t i = 0; i < r.size(); ++i) {
51 r[i] = read(keys[i], 0.f);
52 if (r[i] == 0.f) r[i] = shape->radii[i];
53 if (!std::isfinite(r[i]) || r[i] <= thickness + 0.0001f || r[i] > 1000.f) return invalid();
54 }
55 auto radius = [&](float t) {
56 const float s = 1.f - t;
57 return s * s * s * r[0] + 3.f * s * s * t * r[1] + 3.f * s * t * t * r[2] + t * t * t * r[3];
58 };
59 std::vector<VesselSectionPoint> section{{0.f, 0.f}};
60 for (int i = 0; i <= samples; ++i) {
61 const float t = float(i) / samples;
62 section.push_back({radius(t), height * t});
63 }
64 // A semicircular lip joins the outer and inner surfaces without a zero-width top.
65 for (int i = 1; i <= 8; ++i) {
66 const float angle = 3.141592653589793f * float(i) / 8.f;
67 section.push_back({r[3] - thickness * 0.5f + thickness * 0.5f * std::cos(angle),
68 height + thickness * 0.5f * std::sin(angle)});
69 }
70 for (int i = samples - 1; i >= 0; --i) {
71 const float t = float(i) / samples;
72 section.push_back({radius(t) - thickness, bottom + (height - bottom) * t});
73 }
74 section.push_back({0.f, bottom});
75 auto result = revolveVesselSection(section, radial);
76 if (!result.ok()) return result;
77 result.value().setMeta("shape", name);
78 result.value().setMeta("materialRecipe", "pbr." + material);
79 return result;
80}
