11// simulation elapsed time is supplied by the owner, never wall-clock time.
12struct PoseInertia {
13 using Channels = std::array<float, 9>;
14 std::vector<Channels> offsets, velocities;

19 static Channels difference(const TransformTRS& source, const TransformTRS& target) {
20 auto a = source, b = target;
21 a.normalizeRotation();
22 b.normalizeRotation();
23 // log(source * inverse(target)), with hemisphere-independent shortest arc.
24 float x = -a.qw * b.qx + a.qx * b.qw - a.qy * b.qz + a.qz * b.qy;
25 float y = -a.qw * b.qy + a.qx * b.qz + a.qy * b.qw - a.qz * b.qx;
26 float z = -a.qw * b.qz - a.qx * b.qy + a.qy * b.qx + a.qz * b.qw;
27 float w = a.qw * b.qw + a.qx * b.qx + a.qy * b.qy + a.qz * b.qz;
28 if (w < 0.f) {
29 x = -x;
30 y = -y;
31 z = -z;
32 w = -w;
33 }
34 float length = std::sqrt(x * x + y * y + z * z);
35 float factor = length > 1e-8f ? 2.f * std::atan2(length, w) / length : 2.f;
36 return {a.px - b.px, a.py - b.py, a.pz - b.pz, x * factor, y * factor,
37 z * factor, a.sx - b.sx, a.sy - b.sy, a.sz - b.sz};
38 }

40 void begin(const AnimPose& source, const AnimPose& target, const AnimPose& previousTarget, float seconds) {
41 duration = seconds;
42 offsets.resize(source.getBoneCount());
43 velocities.resize(source.getBoneCount());
44 bool history = historyInterval > 1e-6f && previousOutput.getBoneCount() == source.getBoneCount();
45 for (int bone = 0; bone < source.getBoneCount(); ++bone) {
46 offsets[bone] = difference(source.local(bone), target.local(bone));
47 velocities[bone].fill(0.f);
48 if (!history) continue;
49 auto previous = difference(previousOutput.local(bone), previousTarget.local(bone));
50 // At the pi branch cut, use the equivalent log closest to the current offset.
51 float dot = 0.f, length = 0.f;
52 for (int c = 3; c < 6; ++c) {
53 dot += previous[c] * offsets[bone][c];
54 length += previous[c] * previous[c];
55 }
56 length = std::sqrt(length);
57 if (dot < 0.f && length > 1.57079632679f) {
58 float factor = (length - 6.28318530718f) / length;
59 for (int c = 3; c < 6; ++c) previous[c] *= factor;
60 }
61 for (int c = 0; c < 9; ++c) velocities[bone][c] = (offsets[bone][c] - previous[c]) / historyInterval;
62 }
63 }

71 void apply(const AnimPose& target, float elapsed, AnimPose& output, std::span<const float> factors = {}) const {
72 output.copyFrom(&target);
73 if (duration <= 0.f || elapsed >= duration) return;
74 for (int bone = 0; bone < target.getBoneCount(); ++bone) {
75 const float boneDuration = duration * (factors.empty() ? 1.f : factors[bone]);
76 if (boneDuration <= 0.f || elapsed >= boneDuration) continue;
77 float u = std::clamp(elapsed / boneDuration, 0.f, 1.f), u2 = u * u, u3 = u2 * u, u4 = u3 * u, u5 = u4 * u;
78 float position = 1.f - 10.f * u3 + 15.f * u4 - 6.f * u5;
79 float velocity = boneDuration * (u - 6.f * u3 + 8.f * u4 - 3.f * u5);
80 Channels value{};
81 for (int c = 0; c < 9; ++c) value[c] = offsets[bone][c] * position + velocities[bone][c] * velocity;
82 auto& result = output.local(bone);
83 result.px += value[0];
84 result.py += value[1];
85 result.pz += value[2];
86 result.sx += value[6];
87 result.sy += value[7];
88 result.sz += value[8];
89 float angle = std::sqrt(value[3] * value[3] + value[4] * value[4] + value[5] * value[5]);
90 float factor = angle > 1e-8f ? std::sin(angle * .5f) / angle : .5f;
91 float x = value[3] * factor, y = value[4] * factor, z = value[5] * factor, w = std::cos(angle * .5f);
92 auto q = result;
93 result.qx = w * q.qx + x * q.qw + y * q.qz - z * q.qy;
94 result.qy = w * q.qy - x * q.qz + y * q.qw + z * q.qx;
95 result.qz = w * q.qz + x * q.qy - y * q.qx + z * q.qw;
96 result.qw = w * q.qw - x * q.qx - y * q.qy - z * q.qz;
97 result.normalizeRotation();
98 }
99 }
