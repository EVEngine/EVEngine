31PointGraphNodeMetric makeMetric(const std::string& id, const PointSet& points, float milliseconds, bool cacheHit,
32 const std::string& backend = "cpu") {
33 PointGraphNodeMetric metric;
34 metric.id = id;
35 metric.outputCount = points.getCount();
36 metric.milliseconds = milliseconds;
37 metric.cacheHit = cacheHit;
38 metric.backend = backend;
39 if (points.empty()) return metric;
40 const auto& first = points.points().front();
41 metric.minX = metric.maxX = first.x;
42 metric.minY = metric.maxY = first.y;
43 metric.minZ = metric.maxZ = first.z;
44 double densitySum = 0.0;
45 for (const auto& point : points.points()) {
46 metric.minX = std::min(metric.minX, point.x);
47 metric.minY = std::min(metric.minY, point.y);
48 metric.minZ = std::min(metric.minZ, point.z);
49 metric.maxX = std::max(metric.maxX, point.x);
50 metric.maxY = std::max(metric.maxY, point.y);
51 metric.maxZ = std::max(metric.maxZ, point.z);
52 densitySum += point.density;
53 }
54 metric.averageDensity = float(densitySum / double(points.getCount()));
55 return metric;
56}

58uint64_t spatialSampleUpperBound(const SpatialData& spatial, float spacing) {
59 if (!spatial.hasBounds() || spacing <= 0.f) return 0;
60 const auto axisCount = [spacing](float minimum, float maximum) {
61 if (maximum <= minimum) return uint64_t(1);
62 return uint64_t(std::floor(double(maximum - minimum) / double(spacing) + 0.001)) + 1;
63 };
64 const uint64_t x = axisCount(spatial.getMinX(), spatial.getMaxX());
65 const uint64_t y =
66 spatial.getKind() == "surface.heightfield" ? uint64_t(1) : axisCount(spatial.getMinY(), spatial.getMaxY());
67 const uint64_t z = axisCount(spatial.getMinZ(), spatial.getMaxZ());
68 const uint64_t maximum = std::numeric_limits<uint64_t>::max();
69 if (x > maximum / y) return maximum;
70 const uint64_t xy = x * y;
71 return xy > maximum / z ? maximum : xy * z;
72}

385Result<OptionalRef<const PointSet>> PointGraph::evaluateTransformSegment(const std::string& id,
386 std::unordered_map<std::string, int>& states) {
387 if (!executionPlan_) return Result<OptionalRef<const PointSet>>::success(std::nullopt);
388 const auto segmentEntry = executionPlan_->segmentByOutput.find(id);
389 if (segmentEntry == executionPlan_->segmentByOutput.end())
390 return Result<OptionalRef<const PointSet>>::success(std::nullopt);
391 const ExecutionSegment& planned = executionPlan_->segments[segmentEntry->second];
392 if (!planned.gpuTransformChain || planned.nodes.size() < 2)
393 return Result<OptionalRef<const PointSet>>::success(std::nullopt);
394 const std::vector<std::string>& chain = planned.nodes;
395 if (std::any_of(chain.begin(), chain.end(),
396 [&](const std::string& nodeId) { return nodes_.at(nodeId).cacheValid; }))
397 return Result<OptionalRef<const PointSet>>::success(std::nullopt);
398 const std::string& cursor = nodes_.at(chain.front()).inputs[0];
399
400 auto evaluated = evaluate(cursor, states);
401 if (!evaluated.ok()) return Result<OptionalRef<const PointSet>>::failure(evaluated.status());
402 const PointSet* input = &evaluated.value().get();
403 const bool gpuEligible =
404 !input->empty() &&
405 (computePolicy_ == "gpu" || (computePolicy_ == "auto" && input->getCount() >= computeMinimumPoints_));
406 if (!gpuEligible) return Result<OptionalRef<const PointSet>>::success(std::nullopt);
407 if (executionNodeBudget_ > 0 && evaluatedNodes_ + int(chain.size()) > executionNodeBudget_) {
408 lastCancelled_ = true;
409 return nodeFailure<OptionalRef<const PointSet>>(DiagnosticCode::Cancelled, id,
410 "execution node budget exceeded at node: " + id);
411 }
412
413 std::vector<PointCompute::Transform> transforms;
414 transforms.reserve(chain.size());
415 for (const std::string& nodeId : chain) {
416 const Node& transformNode = nodes_.at(nodeId);
417 transforms.push_back({floatValue(transformNode, "x", 0.f), floatValue(transformNode, "y", 0.f),
418 floatValue(transformNode, "z", 0.f), floatValue(transformNode, "yaw", 0.f),
419 floatValue(transformNode, "scaleX", 1.f), floatValue(transformNode, "scaleY", 1.f),
420 floatValue(transformNode, "scaleZ", 1.f)});
421 }
422
423 PointSet output;
424 const uint64_t started = nowNanoseconds();
425 if (!pointCompute_->transformChain(*input, output, transforms)) {
426 computeFallbackReason_ = pointCompute_->getError();
427 return Result<OptionalRef<const PointSet>>::success(std::nullopt);
428 }
429
430 evaluatedNodes_ += int(chain.size());
431 for (const std::string& nodeId : chain) {
432 Node& transformNode = nodes_.at(nodeId);
433 transformNode.deferredTransformValid = true;
434 states[nodeId] = 2;
435 }
436 Node& finalNode = nodes_.at(id);
437 finalNode.cache = std::move(output);
438 finalNode.cacheValid = true;
439 finalNode.deferredTransformValid = false;
440#ifdef EVENGINE_WEBGPU
441 const std::string backend = "webgpu";
442#else
443 const std::string backend = "vulkan";
444#endif
445 const float elapsed = float(nowNanoseconds() - started) * 0.000001f;
446 finalNode.cacheMetric = makeMetric(id, finalNode.cache, elapsed, false, backend);
447 metrics_.push_back(finalNode.cacheMetric);
448 return Result<OptionalRef<const PointSet>>::success(std::cref(finalNode.cache));
449}

451Result<void> PointGraph::validateNode(const std::string& id, std::unordered_map<std::string, int>& states) const {
452 if (states[id] == 2) return Result<void>::success();
453 if (states[id] == 1) {
454 return nodeFailure<void>(DiagnosticCode::Conflict, id, "cycle at node: " + id);
455 }
456 const auto found = nodes_.find(id);
457 if (found == nodes_.end() || getOperationInputCount(found->second.operation) < 0) {
458 return nodeFailure<void>(DiagnosticCode::InvalidArgument, id, "invalid node: " + id);
459 }
460 states[id] = 1;
461 for (const auto& input : found->second.inputs) {
462 if (!input.empty()) {
463 auto result = validateNode(input, states);
464 if (!result.ok()) return result;
465 }
466 }
467 const auto& node = found->second;
468 if (node.operation == "input" && !node.hasPoints)
