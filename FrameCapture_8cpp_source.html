77void FrameCaptureStore::collect() {
78 std::erase_if(impl_->leases, [](const auto& item) { return item.first.expired(); });
79 const auto now = impl_->clock();
80 for (auto& [key, e] : impl_->entries) {
81 (void)key;
82 if (!terminal(e.snapshot.phase) && now >= e.deadline) {
83 e.snapshot.phase = FrameCapturePhase::Expired;
84 e.snapshot.reason = "CaptureExpired";
85 e.snapshot.buffers.clear();
86 e.cpuBytes = 0;
87 }
88 }
89 // Bound diagnostic tombstones without invalidating live captures or GPU reservations.
90 while (impl_->entries.size() > 64) {
91 auto it = std::find_if(impl_->entries.begin(), impl_->entries.end(), [](const auto& item) {
92 return terminal(item.second.snapshot.phase) && item.second.reserved == 0;
93 });
94 if (it == impl_->entries.end()) break;
95 impl_->entries.erase(it);
96 }
97}
98Result<std::string> FrameCaptureStore::request(FrameCaptureRequest request) {
99 collect();
100 if (request.attachments.empty() || request.attachments.size() > names.size() || request.timeout.count() < 100 ||
101 request.timeout.count() > 30000 || request.requestKey.size() > 128)
102 return failure<std::string>("InvalidCaptureRequest");
103 std::set<FrameAttachment> unique;
104 for (auto a : request.attachments) {
105 if (frameAttachmentName(a) == "unknown" || !unique.insert(a).second)
106 return failure<std::string>("InvalidOrDuplicateAttachment");
107 if (std::find(impl_->caps.attachments.begin(), impl_->caps.attachments.end(), a) ==
108 impl_->caps.attachments.end())
109 return failure<std::string>("UnsupportedAttachment: " + std::string(frameAttachmentName(a)),
110 DiagnosticCode::Unsupported);
111 }
112 size_t live = 0;
113 for (const auto& [token, e] : impl_->entries) {
114 if (e.snapshot.generation == impl_->caps.generation && !request.requestKey.empty() &&
115 request.requestKey == e.request.requestKey) {
116 if (request.attachments != e.request.attachments || request.timeout != e.request.timeout ||
117 request.reportUnavailable != e.request.reportUnavailable)
118 return failure<std::string>("RequestConflict");
119 return Result<std::string>::success(token);
120 }
121 if (!terminal(e.snapshot.phase)) {
122 ++live;
123 if (e.snapshot.phase != FrameCapturePhase::Ready) return failure<std::string>("CaptureBusy");
124 }
125 }
126 if (live >= 2 || impl_->used() >= impl_->caps.maxBytes) return failure<std::string>("BudgetExceeded");
127 Impl::Entry e;
128 e.request = std::move(request);
129 e.snapshot.token = std::to_string(impl_->caps.generation) + ":" + std::to_string(impl_->nextToken++);
130 e.snapshot.generation = impl_->caps.generation;
131 e.snapshot.backend = impl_->caps.backend;
132 e.deadline = impl_->clock() + e.request.timeout;
133 auto token = e.snapshot.token;
134 impl_->entries.emplace(token, std::move(e));
135 return Result<std::string>::success(std::move(token));
136}
137Result<FrameCaptureSnapshot> FrameCaptureStore::snapshot(std::string_view token) {

144Result<FrameCapturePhase> FrameCaptureStore::release(std::string_view token) {
145 collect();
146 auto it = impl_->entries.find(token);
147 if (it == impl_->entries.end() || it->second.snapshot.generation != impl_->caps.generation)
148 return failure<FrameCapturePhase>("StaleCapture", DiagnosticCode::NotFound);
149 auto& e = it->second;
150 if (e.snapshot.phase == FrameCapturePhase::Released)
151 return Result<FrameCapturePhase>::success(FrameCapturePhase::Released);
152 e.snapshot.phase =
153 e.snapshot.phase == FrameCapturePhase::Ready ? FrameCapturePhase::Released : FrameCapturePhase::Cancelled;
154 e.snapshot.buffers.clear();
155 e.cpuBytes = 0;
156 return Result<FrameCapturePhase>::success(e.snapshot.phase);
157}
158void FrameCaptureStore::invalidate() {
159 for (auto& [key, entry] : impl_->entries) {

165 impl_->caps.generation = nextGeneration.fetch_add(1);

173Result<void> FrameCaptureStore::begin(std::string_view token, uint64_t frameId, size_t bytes) {
174 collect();
175 auto it = impl_->entries.find(token);
176 if (it == impl_->entries.end() || it->second.snapshot.phase != FrameCapturePhase::Queued)
177 return failure<void>("InvalidCaptureTransition");
178 if (bytes > impl_->caps.maxBytes / 2 || impl_->used() > impl_->caps.maxBytes - bytes * 2)
179 return failure<void>("BudgetExceeded");
180 auto& e = it->second;
181 e.reserved = bytes * 2; // Reserve staging and peak CPU publication together.
182 e.snapshot.frameId = frameId;
183 e.snapshot.phase = FrameCapturePhase::Recording;
184 return Result<void>::success();
185}

193Result<void> FrameCaptureStore::complete(std::string_view token, FrameCaptureSnapshot snapshot) {
194 collect();
195 auto it = impl_->entries.find(token);
196 if (it == impl_->entries.end() || it->second.snapshot.phase != FrameCapturePhase::ReadbackPending)
197 return failure<void>("InvalidCaptureTransition");
198 auto& e = it->second;
199 std::set<FrameAttachment> seen;
200 std::set<const void*> storage;
201 size_t bytes = 0;
202 for (const auto& b : snapshot.buffers) {
203 if (!seen.insert(b.attachment).second) return failure<void>("DuplicateCaptureBuffer");
204 if (!b.bytes) {
205 if (!e.request.reportUnavailable || b.unavailableReason.empty())
206 return failure<void>("AttachmentUnavailable");
207 continue;
208 }
209 if (!b.width || !b.height || uint64_t(b.width) * b.height * 4 != b.bytes->size())
210 return failure<void>("InvalidCaptureStorage");
211 if (storage.insert(b.bytes.get()).second) bytes += b.bytes->size();
212 }
213 if (seen != std::set<FrameAttachment>(e.request.attachments.begin(), e.request.attachments.end()) ||
214 bytes > e.reserved / 2)
215 return failure<void>("IncompleteCapture");
216 snapshot.token = e.snapshot.token;
217 snapshot.generation = e.snapshot.generation;
218 snapshot.backend = e.snapshot.backend;
219 snapshot.frameId = e.snapshot.frameId;
220 snapshot.phase = FrameCapturePhase::Ready;
221 storage.clear();
222 for (const auto& b : snapshot.buffers)
223 if (b.bytes && storage.insert(b.bytes.get()).second) impl_->leases.emplace_back(b.bytes, b.bytes->size());
224 e.snapshot = std::move(snapshot);
225 e.cpuBytes = bytes;
226 e.reserved -= bytes; // Remaining reservation is retired with GPU staging.
227 e.deadline = impl_->clock() + std::chrono::seconds(60);
228 return Result<void>::success();
229}
230Result<void> FrameCaptureStore::fail(std::string_view token, std::string reason) {
231 auto it = impl_->entries.find(token);
232 if (it == impl_->entries.end()) return failure<void>("StaleCapture", DiagnosticCode::NotFound);
233 if (!terminal(it->second.snapshot.phase)) {
234 it->second.snapshot.phase = FrameCapturePhase::Failed;
235 it->second.snapshot.reason = std::move(reason);
236 it->second.snapshot.buffers.clear();
237 it->second.cpuBytes = 0;
238 }
239 return Result<void>::success();
240}
241void FrameCaptureStore::retire(std::string_view token) {
242 auto it = impl_->entries.find(token);
243 if (it != impl_->entries.end()) it->second.reserved = 0;
244}
