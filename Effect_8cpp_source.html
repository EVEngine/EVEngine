44EffectDefinition parseEffectObject(Value o) {
45 EffectDefinition def;
46 if (!o.isObject()) return def;
47 def.id = o.getString("id");
48 def.durationPolicy = o.getString("durationPolicy", "instant");
49 def.duration = o.getFloat("duration", 0.f);
50 def.period = o.getFloat("period", 0.f);
51 def.stackPolicy = o.getString("stackPolicy", "none");
52 def.maxStacks = o.getInt("maxStacks", 1);
53 def.tags = o.getStringArray("tags");
54 def.extra = o.getStringMap("extra");
55
56 const Value mods = o.get("modifiers");
57 for (size_t i = 0; i < mods.size(); ++i) {
58 const Value mo = mods.at(i);
59 EffectModifierSpec spec;
60 spec.attribute = mo.getString("attribute");
61 spec.op = mo.getString("op", "add");
62 spec.value = mo.getDouble("value", 0.0);
63 spec.priority = mo.getInt("priority", 0);
64 if (!spec.attribute.empty()) def.modifiers.push_back(std::move(spec));
65 }
66 return def;
67}

71int EffectRegistry::loadFromJson(const std::string &json, std::string *error) {
72 std::string err;
73 const eve::json::Document doc = eve::json::Document::parse(json, &err);
74 if (!doc.valid()) {
75 if (error) *error = err.empty() ? "invalid json" : err;
76 return 0;
77 }
78
79 const Value root = doc.root();
80 int n = 0;
81 if (root.isArray()) {
82 for (size_t i = 0; i < root.size(); ++i) {
83 EffectDefinition def = parseEffectObject(root.at(i));
84 if (def.id.empty()) continue;
85 registerEffect(def);
86 ++n;
87 }
88 } else if (root.isObject()) {
89 EffectDefinition def = parseEffectObject(root);
90 if (!def.id.empty()) {
91 registerEffect(def);
92 ++n;
93 }
94 }
95 return n;
96}
