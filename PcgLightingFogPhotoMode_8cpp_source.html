48Result<void> PcgLightingFogPhotoMode::applyPhotoModeField(
49 const PhotoModeAssignment &assignment) {
50 const auto fail = [&](DiagnosticCode code, const char *message) {
51 return Result<void>::failure(Diagnostic::error(
52 code, message, assignment.field, {}, "daynight.pcgLightingFog"));
53 };
54 if (!target_)
55 return fail(DiagnosticCode::Unsupported, "day-night target is unavailable");
56 if (!isFogField(assignment.field))
57 return fail(DiagnosticCode::InvalidArgument, "unsupported Pcg fog field");
58
59 PcgFogState candidate = state_;
60 if (assignment.field == "m_fogOverride" ||
61 assignment.field == "m_overrideDensityVolume") {
62 const auto *value = std::get_if<bool>(&assignment.value);
63 if (!value)
64 return fail(DiagnosticCode::InvalidArgument, "fog toggle requires a bool");
65 if (assignment.field == "m_fogOverride") candidate.overrideFog = *value;
66 else candidate.overrideDensityVolume = *value;
67 } else if (assignment.field == "m_fogMode" ||
68 assignment.field == "m_densityVolumeEffectType" ||
69 assignment.field == "m_densityVolumeTilingResolution") {
70 const auto *value = std::get_if<std::int64_t>(&assignment.value);
71 if (!value)
72 return fail(DiagnosticCode::InvalidArgument, "fog enum requires an integer");
73 if (assignment.field == "m_fogMode") candidate.mode = static_cast<int>(*value);
74 else if (assignment.field == "m_densityVolumeEffectType")
75 candidate.densityVolumeEffect = static_cast<int>(*value);
76 else candidate.densityVolumeTiling = static_cast<int>(*value);
77 } else if (assignment.field == "m_fogColor" ||
78 assignment.field == "m_densityVolumeAlbedoColor") {
79 const auto *value = std::get_if<PhotoModeColor>(&assignment.value);
80 if (!value || !std::isfinite(value->r) || !std::isfinite(value->g) ||
81 !std::isfinite(value->b) || value->r < 0.f || value->g < 0.f ||
82 value->b < 0.f)
83 return fail(DiagnosticCode::InvalidArgument,
84 "fog color requires finite non-negative RGB channels");
85 if (assignment.field == "m_fogColor") {
86 candidate.red = value->r;
87 candidate.green = value->g;
88 candidate.blue = value->b;
89 } else {
90 candidate.densityAlbedoRed = value->r;
91 candidate.densityAlbedoGreen = value->g;
92 candidate.densityAlbedoBlue = value->b;
93 }
94 } else {
95 const auto *value = std::get_if<float>(&assignment.value);
96 if (!value || !std::isfinite(*value))
97 return fail(DiagnosticCode::InvalidArgument,
98 "fog setting requires a finite float");
99 if (assignment.field == "m_pcgAdditionalLinearFog")
100 candidate.additionalLinearDistance = *value;
101 else if (assignment.field == "m_pcgAdditionalExponentialFog")
102 candidate.additionalExponentialDensity = *value;
103 else if (assignment.field == "m_fogDensity") candidate.density = *value;
104 else if (assignment.field == "m_fogStart") candidate.startDistance = *value;
105 else if (assignment.field == "m_fogEnd") candidate.endDistance = *value;
106 else if (assignment.field == "m_globalFogDensityMultiplier")
107 candidate.globalDensityMultiplier = *value;
108 else if (assignment.field == "m_densityVolumeFogDistance")
109 candidate.densityVolumeDistance = *value;
110 }
111
112 auto result = target_->setPcgFog(candidate);
113 if (!result.ok()) return result;
114 state_ = candidate;
115 return Result<void>::success();
116}
