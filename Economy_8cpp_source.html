43bool Economy::registerResourceType(const std::string& id, const std::string& category, int stockMax,
44 const std::string& depletion) {
45 ResourceTypeDef def;
46 def.id = id;
47 def.category = category;
48 def.stockMax = stockMax;
49 def.displayName = id;
50 if (depletion == "renewable")
51 def.depletion = DepletionModel::Renewable;
52 else if (depletion == "infinite")
53 def.depletion = DepletionModel::Infinite;
54 else if (depletion == "growing")
55 def.depletion = DepletionModel::Growing;
56 else
57 def.depletion = DepletionModel::Finite;
58 return ResourceTypeRegistry::registerType(def);
59}
