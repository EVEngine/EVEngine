59// Hydrostatic reconstruction (Audusse et al., 2004), including the pressure
60// correction on the receiving cell. A constant free surface remains at rest.
61[[nodiscard]] glm::vec3 bedFlux(glm::vec3 a, glm::vec3 b, float ba, float bb, float gravity, float minDepth, bool axisZ,
62 bool receiveA) noexcept {
63 const float top = std::max(ba, bb);
64 const float ha = std::max(0.f, a.x + ba - top);
65 const float hb = std::max(0.f, b.x + bb - top);
66 const glm::vec3 qa{ha, a.x > minDepth ? a.y * ha / a.x : 0.f, a.x > minDepth ? a.z * ha / a.x : 0.f};
67 const glm::vec3 qb{hb, b.x > minDepth ? b.y * hb / b.x : 0.f, b.x > minDepth ? b.z * hb / b.x : 0.f};
68 const int axis = axisZ ? 2 : 1;
69 const float va = ha > minDepth ? std::abs(qa[axis] / ha) : 0.f;
70 const float vb = hb > minDepth ? std::abs(qb[axis] / hb) : 0.f;
71 const float alpha = std::max(va + std::sqrt(gravity * ha), vb + std::sqrt(gravity * hb));
72 glm::vec3 f =
73 0.5f * ((axisZ ? fluxZ(qa.x, qa.y, qa.z, gravity, minDepth) : fluxX(qa.x, qa.y, qa.z, gravity, minDepth)) +
74 (axisZ ? fluxZ(qb.x, qb.y, qb.z, gravity, minDepth) : fluxX(qb.x, qb.y, qb.z, gravity, minDepth))) -
75 0.5f * alpha * (qb - qa);
76 const float original = receiveA ? a.x : b.x;
77 const float reconstructed = receiveA ? ha : hb;
78 f[axis] += 0.5f * gravity * (original * original - reconstructed * reconstructed);
79 return f;
80}

82[[nodiscard]] float colliderFootprintDistance(const ShallowWaterCollider& c, float worldX, float worldZ) noexcept {
83 const float dx = worldX - c.center.x;
84 const float dz = worldZ - c.center.z;
85 if (c.shape == ShallowWaterColliderShape::Box) {
86 const float ox = std::abs(dx) - c.halfExtent.x;
87 const float oz = std::abs(dz) - c.halfExtent.z;
88 const float outside = std::sqrt(std::max(ox, 0.f) * std::max(ox, 0.f) + std::max(oz, 0.f) * std::max(oz, 0.f));
89 return outside + std::min(std::max(ox, oz), 0.f);
90 }
91 return std::sqrt(dx * dx + dz * dz) - c.radius;
92}

94[[nodiscard]] float colliderImmersion(const ShallowWaterCollider& c, float surfaceY) noexcept {
95 if (c.shape == ShallowWaterColliderShape::Sphere) {
96 return std::clamp(surfaceY - (c.center.y - c.radius), 0.f, 2.f * c.radius);

98 if (c.shape == ShallowWaterColliderShape::Capsule) {
99 const float top = c.center.y + c.halfExtent.y + c.radius;
100 const float bottom = c.center.y - c.halfExtent.y - c.radius;
101 return std::clamp(surfaceY - bottom, 0.f, top - bottom);
102 }
103 return std::clamp(surfaceY - (c.center.y - c.halfExtent.y), 0.f, 2.f * c.halfExtent.y);
104}

259Result<void> ShallowWaterField::disturbAt(float worldX, float worldZ, float radius, float heightDelta,
260 float radialSpeed) {
261 if (!std::isfinite(worldX) || !std::isfinite(worldZ) || !std::isfinite(radius) || !std::isfinite(heightDelta) ||
262 !std::isfinite(radialSpeed) || radius <= 0.f || radius > 50.f || std::abs(heightDelta) > maxDepth_ ||
263 std::abs(radialSpeed) > 50.f)
264 return fail("invalid disturbAt arguments");
265 stateOnGpu_ = false;
266 const float r2 = radius * radius;
267 for (int z = 0; z < resZ_; ++z) {
268 for (int x = 0; x < resX_; ++x) {
269 const float wx = origin_.x + (float(x) + 0.5f) * cellSize_;
270 const float wz = origin_.z + (float(z) + 0.5f) * cellSize_;
271 const float dx = wx - worldX;
272 const float dz = wz - worldZ;
273 const float d2 = dx * dx + dz * dz;
274 if (d2 > r2) continue;
275 const float w = 1.f - std::sqrt(d2) / radius;
276 const int i = index(x, z);
277 heights_[size_t(i)] = std::clamp(heights_[size_t(i)] + heightDelta * w, 0.f, maxDepth_);
278 if (std::abs(radialSpeed) > 1e-6f && d2 > 1e-8f) {
279 const float inv = 1.f / std::sqrt(d2);
280 const float h = std::max(heights_[size_t(i)], minDepth_);
281 momentumX_[size_t(i)] += dx * inv * radialSpeed * w * h;
282 momentumZ_[size_t(i)] += dz * inv * radialSpeed * w * h;
283 }
284 }
285 }
286 return Result<void>::success();
287}

294float ShallowWaterField::surfaceHeightAt(float worldX, float worldZ) const noexcept {
295 float fx = 0.f, fz = 0.f;
296 worldToCell(worldX, worldZ, fx, fz);
297 const int x0 = int(std::floor(fx));
298 const int z0 = int(std::floor(fz));
299 const float tx = fx - float(x0);
300 const float tz = fz - float(z0);
301 auto sample = [&](int x, int z) -> float {
302 if (!inside(x, z)) return origin_.y;
303 const int i = index(x, z);
304 return beds_[size_t(i)] + heights_[size_t(i)];
305 };
306 const float s00 = sample(x0, z0);
307 const float s10 = sample(x0 + 1, z0);
308 const float s01 = sample(x0, z0 + 1);
309 const float s11 = sample(x0 + 1, z0 + 1);
310 const float a = s00 * (1.f - tx) + s10 * tx;
311 const float b = s01 * (1.f - tx) + s11 * tx;
312 return a * (1.f - tz) + b * tz;
313}

332 float fx = 0.f, fz = 0.f;

349Result<void> ShallowWaterField::buildSurfaceMesh(std::vector<float>& positionsXYZ, std::vector<float>& normalsXYZ,
350 std::vector<float>& uvsST, std::vector<uint32_t>& indices,
351 float visualHeightGain, std::vector<float>* colorsRGBA) const {
352 if (!(visualHeightGain > 0.f) || visualHeightGain > 32.f || !std::isfinite(visualHeightGain))
353 return fail("visualHeightGain must be in (0, 32]");
354
355 const int vx = resX_;
356 const int vz = resZ_;
357 positionsXYZ.resize(size_t(vx * vz) * 3u);
358 normalsXYZ.resize(size_t(vx * vz) * 3u);
359 uvsST.resize(size_t(vx * vz) * 2u);
360 if (colorsRGBA) colorsRGBA->assign(size_t(vx * vz) * 4u, 1.f);
361 indices.clear();
362 indices.reserve(size_t((vx - 1) * (vz - 1) * 6));
363
364 // Rest depth ≈ mean filled depth; used only for display exaggeration / vertex tint.
365 float restDepth = 0.f;
366 for (float h : heights_) restDepth += h;
367 restDepth /= float(std::max(cellCount(), 1));
368
369 for (int z = 0; z < vz; ++z) {
370 for (int x = 0; x < vx; ++x) {
371 const int i = index(x, z);
372 const size_t p = size_t(i) * 3u;
373 const size_t u = size_t(i) * 2u;
374 const float h = heights_[size_t(i)];
375 const float reference = restDepths_[size_t(i)];
376 const float y = beds_[size_t(i)] + std::max(0.f, reference + (h - reference) * visualHeightGain);
377 positionsXYZ[p] = origin_.x + (float(x) + 0.5f) * cellSize_;
378 positionsXYZ[p + 1] = y;
379 positionsXYZ[p + 2] = origin_.z + (float(z) + 0.5f) * cellSize_;
380 uvsST[u] = float(x) / float(std::max(vx - 1, 1));
381 uvsST[u + 1] = float(z) / float(std::max(vz - 1, 1));
382
383 if (colorsRGBA) {
384 // Match copyFieldToImage crest-heat packing so vertex colours carry depth/velocity.
385 constexpr float kVelScale = 1.5f;
386 const float depthContrast =
387 std::max(0.12f * std::max(restDepth, 1e-3f), 1e-3f);
388 const float invH = 1.f / std::max(h, minDepth_);
389 const float vxCell = momentumX_[size_t(i)] * invH;
390 const float vzCell = momentumZ_[size_t(i)] * invH;
391 const float speed = std::sqrt(vxCell * vxCell + vzCell * vzCell);
392 const size_t c = size_t(i) * 4u;
393 (*colorsRGBA)[c] = std::clamp((h - restDepth) / depthContrast, 0.f, 1.f);
394 (*colorsRGBA)[c + 1] = 0.5f + 0.5f * std::clamp(vxCell / kVelScale, -1.f, 1.f);
395 (*colorsRGBA)[c + 2] = 0.5f + 0.5f * std::clamp(vzCell / kVelScale, -1.f, 1.f);
396 (*colorsRGBA)[c + 3] = std::clamp(speed / kVelScale, 0.f, 1.f);
397 }
398 }
399 }
400 for (int z = 0; z < vz; ++z) {
401 for (int x = 0; x < vx; ++x) {
402 const auto pos = [&](int cx, int cz) -> glm::vec3 {
403 cx = std::clamp(cx, 0, vx - 1);
404 cz = std::clamp(cz, 0, vz - 1);
405 const size_t p = size_t(index(cx, cz)) * 3u;
406 return {positionsXYZ[p], positionsXYZ[p + 1], positionsXYZ[p + 2]};
407 };
408 const glm::vec3 dx = pos(x + 1, z) - pos(x - 1, z);
409 const glm::vec3 dz = pos(x, z + 1) - pos(x, z - 1);
410 glm::vec3 n = glm::normalize(glm::cross(dz, dx));
411 if (!std::isfinite(n.x) || !std::isfinite(n.y) || !std::isfinite(n.z)) n = {0.f, 1.f, 0.f};
412 const size_t p = size_t(index(x, z)) * 3u;
413 normalsXYZ[p] = n.x;
414 normalsXYZ[p + 1] = n.y;
415 normalsXYZ[p + 2] = n.z;
416 }
417 }
418 for (int z = 0; z < vz - 1; ++z) {
419 for (int x = 0; x < vx - 1; ++x) {
420 const uint32_t i00 = uint32_t(index(x, z));
421 const uint32_t i10 = uint32_t(index(x + 1, z));

561void ShallowWaterField::rasterizeCollidersCpu(float dt) {
562 std::fill(srcH_.begin(), srcH_.end(), 0.f);
563 std::fill(srcHu_.begin(), srcHu_.end(), 0.f);
564 std::fill(srcHv_.begin(), srcHv_.end(), 0.f);
565 if (colliders_.empty() || !(dt > 0.f)) return;
566
567 const float coverRadius = cellSize_ * 0.75f;
568 const float maxSrcH = maxDepth_ * 0.15f;
569 for (int z = 0; z < resZ_; ++z) {
570 for (int x = 0; x < resX_; ++x) {
571 const int i = index(x, z);
572 const float wx = origin_.x + (float(x) + 0.5f) * cellSize_;
573 const float wz = origin_.z + (float(z) + 0.5f) * cellSize_;
574 if (heights_[size_t(i)] <= minDepth_) continue;
575 const float surfaceY = beds_[size_t(i)] + heights_[size_t(i)];
576 for (const auto& c : colliders_) {
577 const float dist = colliderFootprintDistance(c, wx, wz);
578 if (dist >= coverRadius) continue;
579 const float immersion = colliderImmersion(c, surfaceY);
580 if (immersion <= 1e-4f) continue;
581 const float cover = std::clamp(1.f - dist / coverRadius, 0.f, 1.f);
582 // Rates × substep dt — absolute sources stacked with CFL substeps
583 // previously grew into multi-meter height spikes.
584 const float push = immersion * cover * c.displacementScale * dt;
585 float dx = wx - c.center.x;
586 float dz = wz - c.center.z;
587 const float olen = std::sqrt(dx * dx + dz * dz);
588 if (olen > 1e-5f) {
589 dx /= olen;
590 dz /= olen;
591 } else {
592 dx = 0.f;
593 dz = 1.f;
594 }
595 const float radial = push * 1.f;
596 const float h = std::max(heights_[size_t(i)], minDepth_);
597 srcH_[size_t(i)] -= push * 0.35f;
598 srcHu_[size_t(i)] += (dx * radial + c.velocity.x * c.dragScale * dt) * h;
599 srcHv_[size_t(i)] += (dz * radial + c.velocity.z * c.dragScale * dt) * h;
600 srcH_[size_t(i)] += std::max(0.f, -c.velocity.y) * cover * 0.15f * c.displacementScale * dt;
601 }
602 srcH_[size_t(i)] = std::clamp(srcH_[size_t(i)], -maxSrcH, maxSrcH);
603 }
604 }

607void ShallowWaterField::integrateCpu(float dt) {
608 auto sample = [&](int x, int z, float& h, float& hu, float& hv) {
609 x = std::clamp(x, 0, resX_ - 1);
610 z = std::clamp(z, 0, resZ_ - 1);
611 const int i = index(x, z);
612 h = std::max(heights_[size_t(i)] + srcH_[size_t(i)], 0.f);
613 hu = momentumX_[size_t(i)] + srcHu_[size_t(i)];
614 hv = momentumZ_[size_t(i)] + srcHv_[size_t(i)];
615 if (!std::isfinite(h)) h = 0.f;
616 if (!std::isfinite(hu)) hu = 0.f;
617 if (!std::isfinite(hv)) hv = 0.f;
618 };
619 auto sampleBed = [&](int x, int z) {
620 x = std::clamp(x, 0, resX_ - 1);
621 z = std::clamp(z, 0, resZ_ - 1);
622 return beds_[size_t(index(x, z))];
623 };
624
625 for (int z = 0; z < resZ_; ++z) {
626 for (int x = 0; x < resX_; ++x) {
627 float h = 0, hu = 0, hv = 0;
628 float hL = 0, huL = 0, hvL = 0;
629 float hR = 0, huR = 0, hvR = 0;
630 float hD = 0, huD = 0, hvD = 0;
631 float hU = 0, huU = 0, hvU = 0;
632 sample(x, z, h, hu, hv);
633 sample(x - 1, z, hL, huL, hvL);
634 sample(x + 1, z, hR, huR, hvR);
635 sample(x, z - 1, hD, huD, hvD);
636 sample(x, z + 1, hU, huU, hvU);
637
638 const glm::vec3 q{h, hu, hv};
639 const glm::vec3 qL{hL, huL, hvL};
640 const glm::vec3 qR{hR, huR, hvR};
641 const glm::vec3 qD{hD, huD, hvD};
642 const glm::vec3 qU{hU, huU, hvU};
643
644 const float bed = sampleBed(x, z);
645 const glm::vec3 fR = bedFlux(q, qR, bed, sampleBed(x + 1, z), gravity_, minDepth_, false, true);
646 const glm::vec3 fL = bedFlux(qL, q, sampleBed(x - 1, z), bed, gravity_, minDepth_, false, false);
647 const glm::vec3 fU = bedFlux(q, qU, bed, sampleBed(x, z + 1), gravity_, minDepth_, true, true);
648 const glm::vec3 fD = bedFlux(qD, q, sampleBed(x, z - 1), bed, gravity_, minDepth_, true, false);
649 glm::vec3 qn = q - (dt / cellSize_) * (fR - fL + fU - fD);
650 // dt-scaled Laplacian viscosity (was previously unscaled and blew up).
651 qn.y += viscosity_ * dt * (huL + huR + huD + huU - 4.f * hu);
652 qn.z += viscosity_ * dt * (hvL + hvR + hvD + hvU - 4.f * hv);
653 qn.y *= std::max(0.f, 1.f - friction_ * dt);
654 qn.z *= std::max(0.f, 1.f - friction_ * dt);
655 qn.x = std::clamp(qn.x, 0.f, maxDepth_);
656 if (!std::isfinite(qn.x)) qn.x = 0.f;
657 if (!std::isfinite(qn.y)) qn.y = 0.f;
658 if (!std::isfinite(qn.z)) qn.z = 0.f;
659 if (qn.x <= minDepth_) {
660 qn.y = 0.f;
661 qn.z = 0.f;
662 } else {
663 // Cap horizontal speed so a single splash cannot break CFL next step.
664 const float speed = std::sqrt(qn.y * qn.y + qn.z * qn.z) / std::max(qn.x, minDepth_);
665 const float maxSpeed = 8.f;
666 if (speed > maxSpeed) {
667 const float s = maxSpeed / speed;
668 qn.y *= s;
669 qn.z *= s;
670 }
671 }
672 if (x == 0 || x == resX_ - 1) qn.y = 0.f;
673 if (z == 0 || z == resZ_ - 1) qn.z = 0.f;
674
675 const int i = index(x, z);
676 scratchH_[size_t(i)] = qn.x;
677 scratchHu_[size_t(i)] = qn.y;
678 scratchHv_[size_t(i)] = qn.z;
679 }
680 }
681 heights_.swap(scratchH_);
682 momentumX_.swap(scratchHu_);
683 momentumZ_.swap(scratchHv_);
684}

691Result<void> ShallowWaterField::step(float dt) {
692 if (!std::isfinite(dt) || dt < 0.f) return fail("dt must be finite and >= 0");
693 if (dt == 0.f) return Result<void>::success();
694 // Long first frames after engine startup commonly exceed 1s; advance in
695 // capped chunks instead of rejecting the call.
696 float remaining = std::min(dt, 5.f);
697 while (remaining > 1e-6f) {
698 const float chunk = std::min(remaining, 1.f / 20.f); // <= 50 ms chunks
699 float maxH = minDepth_;
700 float maxVel = 0.f;
701 for (int i = 0; i < cellCount(); ++i) {
702 maxH = std::max(maxH, heights_[size_t(i)]);
703 const float h = std::max(heights_[size_t(i)], minDepth_);
704 const float u = momentumX_[size_t(i)] / h;
705 const float v = momentumZ_[size_t(i)] / h;
706 maxVel = std::max(maxVel, std::sqrt(u * u + v * v));
707 }
708 const float waveSpeed = std::sqrt(gravity_ * maxH) + maxVel + 1.f;
709 const float safeDt = 0.5f * cfl_ * cellSize_ / std::max(waveSpeed, 1e-3f);
710 int substeps = std::clamp(int(std::ceil(chunk / safeDt)), 1, maxSubsteps_);
711 const float subDt = chunk / float(substeps);
712
713 bool advanced = false;
714 if (preferGpu_) {
715 if (ensureGpu()) {
716 auto gpu = stepGpu(chunk);
717 if (gpu) {
718 advanced = true;
719 } else {
720 backendFallback_ = true;
721 gpuOk_ = false;
722 }
723 } else {
724 backendFallback_ = true;
725 }
726 }
727 if (!advanced) {
728 for (int i = 0; i < substeps; ++i) stepCpu(subDt);
729 stateOnGpu_ = false;
730 }
731 remaining -= chunk;
732 }
733 return Result<void>::success();
734}

829Result<void> ShallowWaterField::stepGpu(float dt) {
830 float maxH = minDepth_, maxVel = 0.f;
831 for (size_t i = 0; i < heights_.size(); ++i) {
832 maxH = std::max(maxH, heights_[i]);
833 const float h = std::max(heights_[i], minDepth_);
834 maxVel = std::max(maxVel, std::hypot(momentumX_[i], momentumZ_[i]) / h);
835 }
836 const float waveSpeed = std::sqrt(gravity_ * maxH) + maxVel + 1.f;
837 const float safeDt = 0.5f * cfl_ * cellSize_ / std::max(waveSpeed, 1e-3f);
838 const int substeps = std::clamp(int(std::ceil(dt / safeDt)), 1, maxSubsteps_);
839 const float subDt = dt / float(substeps);
840 const int groupsX = (resX_ + 7) / 8;
841 const int groupsZ = (resZ_ + 7) / 8;
842
843 try {
844 seq_->begin();
845 if (!stateOnGpu_) uploadState();
846 uploadColliders();
847 for (int s = 0; s < substeps; ++s) {
848 auto* src = pingA_ ? bufStateA_ : bufStateB_;
849 auto* dst = pingA_ ? bufStateB_ : bufStateA_;
850 shClear_->bindBuffer(3, bufSources_);
851 setPushConstants(shClear_, subDt);
852 seq_->recordDispatch(shClear_, groupsX, groupsZ);
853
854 shRaster_->bindBuffer(0, src);
855 shRaster_->bindBuffer(2, bufBed_);
856 shRaster_->bindBuffer(3, bufSources_);
857 shRaster_->bindBuffer(4, bufColliders_);
858 setPushConstants(shRaster_, subDt);
859 seq_->recordDispatch(shRaster_, groupsX, groupsZ);
860
861 shIntegrate_->bindBuffer(0, src);
862 shIntegrate_->bindBuffer(1, dst);
863 shIntegrate_->bindBuffer(2, bufBed_);
864 shIntegrate_->bindBuffer(3, bufSources_);
865 setPushConstants(shIntegrate_, subDt);
866 seq_->recordDispatch(shIntegrate_, groupsX, groupsZ);
867 pingA_ = !pingA_;
868 }
869 downloadState();
870 seq_->submit();
871 std::vector<float> packed(size_t(cellCount()) * 4u, 0.f);
872 stageState_->downloadBytes(packed.data(), uint64_t(packed.size()) * sizeof(float));
873 for (int i = 0; i < cellCount(); ++i) {
874 const size_t b = size_t(i) * 4u;
875 heights_[size_t(i)] = packed[b];
876 momentumX_[size_t(i)] = packed[b + 1];
877 momentumZ_[size_t(i)] = packed[b + 2];
878 }
879 stateOnGpu_ = true;
880 return Result<void>::success();
881 } catch (const std::exception& error) {
882 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Failed,
883 std::string("Shallow-water GPU step failed: ") + error.what(),
884 "fluids.shallowWater.gpu"));
885 } catch (...) {
886 return Result<void>::failure(
887 Diagnostic::error(DiagnosticCode::Failed, "Shallow-water GPU step failed", "fluids.shallowWater.gpu"));
888 }
889}
