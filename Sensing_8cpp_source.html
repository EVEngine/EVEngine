60bool csvMatches(std::string_view text, const std::set<std::string>& values) {
61 const auto containsToken = [text](std::string_view value) {
62 auto remaining = text;
63 while (!remaining.empty()) {
64 const auto comma = remaining.find(',');
65 if (remaining.substr(0, comma) == value) return true;
66 if (comma == std::string_view::npos) break;
67 remaining.remove_prefix(comma + 1);
68 }
69 return false;
70 };
71 for (const auto& value : values)
72 if (!containsToken(value)) return false;
73 while (!text.empty()) {
74 const auto comma = text.find(',');
75 const auto token = text.substr(0, comma);
76 if (!token.empty() &&
77 std::none_of(values.begin(), values.end(), [token](const std::string& value) { return value == token; }))
78 return false;
79 if (comma == std::string_view::npos) break;
80 text.remove_prefix(comma + 1);
81 }
82 return true;
83}

148bool insideCone(float px, float py, const QueryCone& cone) {
149 const float dx = px - cone.x;
150 const float dy = py - cone.y;
151 const float distSq = dx * dx + dy * dy;
152 if (distSq > cone.range * cone.range) return false;
153 if (distSq == 0.f) return true;
154 const float facingLen = std::hypot(cone.dirX, cone.dirY);
155 if (!(facingLen > 0.f)) return false;
156 const float invDist = 1.f / std::sqrt(distSq);
157 const float nx = dx * invDist;
158 const float ny = dy * invDist;
159 const float fx = cone.dirX / facingLen;
160 const float fy = cone.dirY / facingLen;
161 const float dot = std::clamp(nx * fx + ny * fy, -1.f, 1.f);
162 return std::acos(dot) <= cone.halfAngle;
163}

171bool insideShape(const Subject& subject, const QueryShape& shape, QueryAxes axes) {
172 const auto [px, py] = project(subject.x, subject.y, subject.z, axes);
173 return std::visit(
174 [&](const auto& value) -> bool {
175 using T = std::decay_t<decltype(value)>;
176 if constexpr (std::is_same_v<T, std::monostate>) {
177 return true;
178 } else if constexpr (std::is_same_v<T, QueryCircle>) {
179 const float dx = px - value.x;
180 const float dy = py - value.y;
181 return dx * dx + dy * dy <= value.radius * value.radius;
182 } else if constexpr (std::is_same_v<T, QueryBox>) {
183 return px >= value.minX && px <= value.maxX && py >= value.minY && py <= value.maxY;
184 } else if constexpr (std::is_same_v<T, QuerySphere>) {
185 return std::hypot(double(subject.x) - value.x, double(subject.y) - value.y,
186 double(subject.z) - value.z) <= value.radius;
187 } else {
188 return insideCone(px, py, value);
189 }
190 },
191 shape);
192}

241eve::Result<void> QuerySpec::validate() const {
242 if (axes > QueryAxes::XYZ || (axes == QueryAxes::XYZ && !std::holds_alternative<std::monostate>(shape) &&
243 !std::holds_alternative<QuerySphere>(shape)))
244 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
245 "XYZ queries require a sphere or range-only shape", "query.axes",
246 {}, "sensing"));
247 if (!std::isfinite(minRange) || minRange < 0.f) {
248 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
249 "QuerySpec.minRange must be finite and >= 0",
250 "query.minRange", {}, "sensing"));
251 }
252 if (!(std::isfinite(maxRange) || maxRange == std::numeric_limits<float>::infinity()) || maxRange < minRange) {
253 return eve::Result<void>::failure(eve::Diagnostic::error(
254 eve::DiagnosticCode::InvalidArgument, "QuerySpec.maxRange must be >= minRange (finite or +inf)",
255 "query.maxRange", {}, "sensing"));

258 return eve::Result<void>::failure(eve::Diagnostic::error(
259 eve::DiagnosticCode::InvalidArgument, "QuerySpec.minCount exceeds maxCount", "query.count", {}, "sensing"));
260 }
261 for (const auto& tag : requiredTags) {
262 if (tag.empty())
263 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
264 "requiredTags cannot contain empty keys",
265 "query.requiredTags", {}, "sensing"));
266 }
267 for (const auto& tag : excludedTags) {
268 if (tag.empty())
269 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
270 "excludedTags cannot contain empty keys",
271 "query.excludedTags", {}, "sensing"));
272 }
273 for (const auto& faction : includeFactions) {
274 if (faction.empty())
275 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
276 "includeFactions cannot contain empty keys",
277 "query.includeFactions", {}, "sensing"));
278 }
279 for (const auto& faction : excludeFactions) {
280 if (faction.empty())
281 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
282 "excludeFactions cannot contain empty keys",
283 "query.excludeFactions", {}, "sensing"));
284 }

286 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
287 "visibleTo cannot be empty when set",
288 "query.visibleTo", {}, "sensing"));

290 const auto shapeOk = std::visit(
291 [](const auto& value) -> eve::Result<void> {
292 using T = std::decay_t<decltype(value)>;

304 } else if constexpr (std::is_same_v<T, QueryBox>) {

306 value.minX > value.maxX || value.minY > value.maxY) {
307 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
308 "QueryBox requires finite ordered bounds",
309 "query.shape", {}, "sensing"));
310 }

315 return eve::Result<void>::failure(eve::Diagnostic::error(
316 eve::DiagnosticCode::InvalidArgument,

318 "query.shape", {}, "sensing"));
319 }
320 }
321 return eve::Result<void>::success();
322 },
323 shape);
324 if (!shapeOk) return eve::Result<void>::failure(shapeOk.status());
325 return eve::Result<void>::success();
326}

334 std::string_view t, std::string_view v) {

409Result<void> SensingWorld::rebuildIndex(SpatialIndexKind kind, float cellSize) {
410 if (kind > SpatialIndexKind::XYZ || (kind != SpatialIndexKind::None && (!finite(cellSize) || cellSize <= 0)))
411 return Result<void>::failure(Diagnostic::error(
412 DiagnosticCode::InvalidArgument, "invalid spatial index kind or cell size", "spatial", {}, "sensing"));
413 for (const auto& [id, subject] : subjects_) {
414 if (kind != SpatialIndexKind::None && !indexable(subject.x, subject.y, subject.z, cellSize))
415 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
416 "subject exceeds index cell range", id, {}, "sensing"));
417 }
418 // Build in a separate owner; allocation failure cannot replace the current derived index.
419 SensingWorld next;
420 next.subjects_ = subjects_;
421 next.indexKind_ = kind;
422 if (kind == SpatialIndexKind::XYZ)
423 next.spatialIndex3D_ = std::make_unique<spatial::SpatialHash3D>(cellSize);
424 else if (kind != SpatialIndexKind::None)
425 next.spatialIndex_ = std::make_unique<spatial::SpatialHash2D>(cellSize);
426 next.rebuildSpatialIndex();
427 for (const auto& [id, sid] : next.subjectSpatialIds_) next.spatialIdSubjects_.at(sid) = &subjects_.at(id);
428 spatialCellQueryCache.invalidate(this);
429 spatialIndex_ = std::move(next.spatialIndex_);
430 spatialIndex3D_ = std::move(next.spatialIndex3D_);
431 indexKind_ = kind;
432 subjectSpatialIds_ = std::move(next.subjectSpatialIds_);
433 spatialIdSubjects_ = std::move(next.spatialIdSubjects_);
434 nextSpatialId_ = next.nextSpatialId_;
435 return Result<void>::success(Status::success(StatusCode::Applied));
436}

491bool SensingWorld::trySpatialBroadphase(const QueryOrigin& origin, const QuerySpec& spec,
492 std::vector<const Subject*>& out) const {
493 if (!spatialIndexEnabled()) return false;
494 // A bounded 3D sphere can be conservatively projected into any planar index.
495 const auto* sphere = std::get_if<QuerySphere>(&spec.shape);
496 const bool range3D = spec.axes == QueryAxes::XYZ && std::isfinite(spec.maxRange);
497 if (sphere || range3D) {
498 const float x = sphere ? sphere->x : origin.x;
499 const float y = sphere ? sphere->y : origin.y;
500 const float z = sphere ? sphere->z : origin.z;
501 const float r = sphere ? sphere->radius : spec.maxRange;
502 const float cellSize = spatialIndex3D_ ? spatialIndex3D_->getCellSize() : spatialIndex_->getCellSize();
503 if (!boundedQuery(x, y, z, r, cellSize, spatialIndex3D_ ? 3 : 2)) return false;
504 if (spatialIndex3D_)
505 spatialIndex3D_->querySphere(x, y, z, r);
506 else {
507 const auto axes = indexKind_ == SpatialIndexKind::XZ ? QueryAxes::XZ
508 : indexKind_ == SpatialIndexKind::YZ ? QueryAxes::YZ
509 : QueryAxes::XY;
510 const auto [u, v] = project(x, y, z, axes);
511 spatialIndex_->queryCircle(u, v, r);
512 }
513 const int count = spatialIndex3D_ ? spatialIndex3D_->getResultCount() : spatialIndex_->getResultCount();
514 for (int i = 0; i < count; ++i) {
515 const int id = spatialIndex3D_ ? spatialIndex3D_->getResultId(i) : spatialIndex_->getResultId(i);
516 out.push_back(spatialIdSubjects_.at(id));
517 }
518 return true;
519 }
520 // Unbounded height cannot be culled by a different plane or a finite 3D volume.
521 const auto axes = indexKind_ == SpatialIndexKind::XZ ? QueryAxes::XZ
522 : indexKind_ == SpatialIndexKind::YZ ? QueryAxes::YZ
523 : QueryAxes::XY;
524 if (!spatialIndex_ || spec.axes != axes) return false;
525 const auto [originU, originV] = project(origin.x, origin.y, origin.z, axes);
526 const bool bounded = std::visit(
527 [&](const auto& shape) {
528 using T = std::decay_t<decltype(shape)>;
529 const float size = spatialIndex_->getCellSize();
530 if constexpr (std::is_same_v<T, std::monostate>)
531 return boundedQuery(originU, originV, 0, spec.maxRange, size, 2);
532 else if constexpr (std::is_same_v<T, QueryBox>)
533 return boundedQuery(shape.minX, shape.minY, 0,
534 std::max(shape.maxX - shape.minX, shape.maxY - shape.minY), size, 2);
535 else if constexpr (std::is_same_v<T, QueryCone>)
536 return boundedQuery(shape.x, shape.y, 0, shape.range, size, 2);
537 else
538 return boundedQuery(shape.x, shape.y, 0, shape.radius, size, 2);
539 },
540 spec.shape);
541 if (!bounded) return false;
542 if (const auto* circle = std::get_if<QueryCircle>(&spec.shape); circle && axes == QueryAxes::XY) {
543 spatialCellQueryCache.select(this, spatialIndex_.get());
544 auto& cache = spatialCellQueryCache;
545 const float cellSize = spatialIndex_->getCellSize();
546 const std::array<int, 4> cells{static_cast<int>(std::floor((circle->x - circle->radius) / cellSize)),
547 static_cast<int>(std::floor((circle->y - circle->radius) / cellSize)),
548 static_cast<int>(std::floor((circle->x + circle->radius) / cellSize)),
549 static_cast<int>(std::floor((circle->y + circle->radius) / cellSize))};
550 constexpr std::size_t maxCachedCells = 256;
551 constexpr std::size_t maxCachedIdsPerCellRange = 512;
552 constexpr std::size_t maxCachedIds = 32768;
553 std::vector<const Subject*> uncached;
554 const std::vector<const Subject*>* subjects = nullptr;
555 const auto cached = cache.candidates.find(cells);
556 if (cached != cache.candidates.end()) {
557 subjects = &cached->second;
558 } else {
559 spatialIndex_->queryRect(cells[0] * cellSize, cells[1] * cellSize,
560 static_cast<float>((static_cast<double>(cells[2]) + 1.0) * cellSize),
561 static_cast<float>((static_cast<double>(cells[3]) + 1.0) * cellSize));
562 uncached.reserve(static_cast<std::size_t>(spatialIndex_->getResultCount()));
563 for (int i = 0; i < spatialIndex_->getResultCount(); ++i) {
564 const auto found = spatialIdSubjects_.find(spatialIndex_->getResultId(i));
565 if (found != spatialIdSubjects_.end()) uncached.push_back(found->second);
566 }
567 if (uncached.size() <= maxCachedIdsPerCellRange) {
568 if (cache.candidates.size() == maxCachedCells ||
569 cache.candidateCount + uncached.size() > maxCachedIds) {
570 cache.candidates.clear();
571 cache.candidateCount = 0;
572 }
573 cache.candidateCount += uncached.size();
574 subjects = &cache.candidates.emplace(cells, std::move(uncached)).first->second;
575 } else {
576 subjects = &uncached;
577 }
578 }
579 out.clear();
580 out.reserve(subjects->size());
581 for (const Subject* subject : *subjects) {
582 if (eve::spatial::AABB2{subject->x, subject->y, subject->x, subject->y}.intersectsCircle(
583 circle->x, circle->y, circle->radius))
584 out.push_back(subject);
585 }
586 return true;
587 }
588 const bool ran = std::visit(
589 [&](const auto& value) -> bool {
590 using T = std::decay_t<decltype(value)>;
591 if constexpr (std::is_same_v<T, std::monostate>) {
592 if (!(std::isfinite(spec.maxRange))) return false;
593 spatialIndex_->queryCircle(originU, originV, spec.maxRange);
594 return true;
595 } else if constexpr (std::is_same_v<T, QueryCircle>) {
596 spatialIndex_->queryCircle(value.x, value.y, value.radius);
597 return true;
598 } else if constexpr (std::is_same_v<T, QueryBox>) {
599 spatialIndex_->queryRect(value.minX, value.minY, value.maxX, value.maxY);
600 return true;
601 } else if constexpr (std::is_same_v<T, QuerySphere>) {
602 return false;
603 } else {
604 spatialIndex_->queryCircle(value.x, value.y, value.range);
605 return true;
606 }
607 },
608 spec.shape);
609 if (!ran) return false;
610 out.clear();
611 out.reserve(static_cast<std::size_t>(spatialIndex_->getResultCount()));
612 for (int i = 0; i < spatialIndex_->getResultCount(); ++i) {
613 const int sid = spatialIndex_->getResultId(i);
614 const auto map = spatialIdSubjects_.find(sid);
615 if (map == spatialIdSubjects_.end()) continue;
616 out.push_back(map->second);
617 }
618 return true;
619}

621bool SensingWorld::accepts(const Subject& subject, const QuerySpec& spec) const {
622 if (!spec.includeFactions.empty()) {
623 bool included = false;
624 for (const auto& faction : spec.includeFactions)
625 if (subject.faction == faction) {
626 included = true;
627 break;
628 }
629 if (!included) return false;
630 }
631 for (const auto& faction : spec.excludeFactions)
632 if (subject.faction == faction) return false;
633 for (const auto& tag : spec.requiredTags)
634 if (!subject.tags.count(tag)) return false;
635 for (const auto& tag : spec.excludedTags)
636 if (subject.tags.count(tag)) return false;
637 if (spec.visibleTo && !subject.visibleTo.count(*spec.visibleTo)) return false;
638 return insideShape(subject, spec.shape, spec.axes);
639}

650eve::Result<CandidateCollection> SensingWorld::collectCandidates(const QueryOrigin& origin,
651 const QuerySpec& spec) const {
652 auto valid = spec.validate();
653 if (!valid) return eve::Result<CandidateCollection>::failure(valid.status());
654 if (!finite(origin.x) || !finite(origin.y) || !finite(origin.z) ||
655 (origin.space != CoordinateSpace::World2D && origin.space != CoordinateSpace::World3D) ||
656 (origin.space == CoordinateSpace::World2D &&
657 (origin.z != 0 || spec.axes != QueryAxes::XY || std::holds_alternative<QuerySphere>(spec.shape)))) {
658 return eve::Result<CandidateCollection>::failure(
659 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "QueryOrigin coordinates must be finite",
660 "query.origin", {}, "sensing"));
661 }
662
663 std::vector<const Subject*> candidates;
664 const bool usedSpatial = trySpatialBroadphase(origin, spec, candidates);
665 if (!usedSpatial) {
666 candidates.clear();
667 candidates.reserve(subjects_.size());
668 for (const auto& [id, subject] : subjects_) {
669 (void)id;
670 candidates.push_back(&subject);
671 }
672 }
673
674 std::vector<RankedCandidate> ranked;
675 ranked.reserve(candidates.size());
676 for (const Subject* subjectPtr : candidates) {
677 const Subject& subject = *subjectPtr;
678 if (origin.subjectId && subject.id == *origin.subjectId) continue;
679 if (!accepts(subject, spec)) continue;
680 const auto [sx, sy] = project(subject.x, subject.y, subject.z, spec.axes);
681 const auto [ox, oy] = project(origin.x, origin.y, origin.z, spec.axes);
682 const float distance = spec.axes == QueryAxes::XYZ
683 ? std::hypot(subject.x - origin.x, subject.y - origin.y, subject.z - origin.z)
684 : std::hypot(sx - ox, sy - oy);
685 if (distance < spec.minRange || distance > spec.maxRange) continue;
686 RankedCandidate candidate;
687 candidate.id = subject.id;
688 candidate.x = subject.x;
689 candidate.y = subject.y;
690 candidate.z = subject.z;
691 candidate.space = origin.space;
692 candidate.distance = distance;
693 candidate.score = -distance;
694 candidate.scoreReason = "distance";
695 ranked.push_back(std::move(candidate));
696 }
697
698 return eve::Result<CandidateCollection>::success(
699 CandidateCollection{std::move(ranked), usedSpatial, static_cast<std::uint32_t>(candidates.size())});
700}

706Result<CandidateQueryResult> SensingWorld::query(const QueryOrigin& origin, const QuerySpec& spec,
707 OptionalRef<const ILineOfSightQuery> lineOfSight) {
708 if (spec.requireLineOfSight && !lineOfSight)
709 return Result<CandidateQueryResult>::failure(Diagnostic::error(DiagnosticCode::Unsupported,
710 "query requires a line-of-sight provider",
711 "query.lineOfSight", {}, "sensing"));
712 auto collected = collectCandidates(origin, spec);
713 if (!collected) return eve::Result<CandidateQueryResult>::failure(collected.status());
714 auto collection = std::move(collected).takeValue();
715 auto ranked = std::move(collection.candidates);
716 if (spec.requireLineOfSight) {
717 auto from = origin.space == CoordinateSpace::World3D ? WorldPoint::world3D(origin.x, origin.y, origin.z)
718 : WorldPoint::world2D(origin.x, origin.y);
719 if (!from) return Result<CandidateQueryResult>::failure(from.status());
720 std::vector<RankedCandidate> visible;
721 for (auto& candidate : ranked) {
722 auto to = candidate.space == CoordinateSpace::World3D
723 ? WorldPoint::world3D(candidate.x, candidate.y, candidate.z)
724 : WorldPoint::world2D(candidate.x, candidate.y);
725 if (!to) return Result<CandidateQueryResult>::failure(to.status());
726 try {
727 auto sight = lineOfSight->get().query(TargetLocation{from.value()}, TargetLocation{to.value()});
728 if (!sight) return Result<CandidateQueryResult>::failure(sight.status());
729 if (sight.value().visible) visible.push_back(std::move(candidate));
730 } catch (const std::exception& error) {
731 return Result<CandidateQueryResult>::failure(
732 Diagnostic::error(DiagnosticCode::Failed, error.what(), "query.lineOfSight", {}, "sensing"));
733 }
734 }
735 ranked = std::move(visible);
736 }
737 if (spec.sortKey == SortKey::DistanceAscending) {
738 std::sort(ranked.begin(), ranked.end(), [](const RankedCandidate& a, const RankedCandidate& b) {
739 if (a.distance != b.distance) return a.distance < b.distance;
740 return a.id < b.id;
741 });
742 } else {
743 std::sort(ranked.begin(), ranked.end(),
744 [](const RankedCandidate& a, const RankedCandidate& b) { return a.id < b.id; });
745 }
746
747 if (spec.countPolicy == CountPolicy::TruncateToMax) {
748 if (ranked.size() > static_cast<std::size_t>(spec.maxCount)) ranked.resize(static_cast<std::size_t>(spec.maxCount));
749 if (ranked.size() < static_cast<std::size_t>(spec.minCount)) {
750 return eve::Result<CandidateQueryResult>::failure(eve::Diagnostic::error(
751 eve::DiagnosticCode::PreconditionViolation,
752 "candidate count is below QuerySpec.minCount after truncation", "query.count", {}, "sensing"));
753 }
754 } else if (ranked.size() < static_cast<std::size_t>(spec.minCount) ||
755 ranked.size() > static_cast<std::size_t>(spec.maxCount)) {
756 return eve::Result<CandidateQueryResult>::failure(
757 eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
758 "candidate count violates QuerySpec count policy", "query.count", {}, "sensing"));
759 }
760
761 lastQuery_.usedSpatial = collection.usedSpatial;
762 lastQuery_.scanned = collection.scanned;
763 lastQuery_.originX = origin.x;
764 lastQuery_.originY = origin.y;
765 lastQuery_.originZ = origin.z;
766 lastQuery_.space = origin.space;
767 lastQuery_.shapeKind = std::visit(
768 [](const auto& value) -> std::string {
769 using T = std::decay_t<decltype(value)>;
770 if constexpr (std::is_same_v<T, QueryCircle>) return "circle";
771 if constexpr (std::is_same_v<T, QueryBox>) return "box";
772 if constexpr (std::is_same_v<T, QueryCone>) return "cone";
773 if constexpr (std::is_same_v<T, QuerySphere>) return "sphere";
774 return "none";
775 },
776 spec.shape);
777
778 publishResults(ranked);
779 return eve::Result<CandidateQueryResult>::success(CandidateQueryResult{std::move(ranked)});
780}

865eve::Result<void> SensingWorld::restoreJson(const std::string& j) {
866 std::string err;
867 auto d = eve::json::Document::parse(j, &err);
868 if (!d.valid() || !d.root().isObject() || d.root().getString("schema") != "eve.sensing.world" ||
869 (d.root().getInt("version") != 1 && d.root().getInt("version") != 2) || !d.root().get("subjects").isArray()) {
870 return eve::Result<void>::failure(eve::Diagnostic::error(
871 eve::DiagnosticCode::ParseError, err.empty() ? "invalid snapshot" : err, {}, {}, "sensing.restore"));
872 }
873 SensingWorld next;
874 auto a = d.root().get("subjects");
875 for (size_t n = 0; n < a.size(); ++n) {
876 auto v = a.at(n);
877 if (!v.isObject()) {
878 return eve::Result<void>::failure(eve::Diagnostic::error(
879 eve::DiagnosticCode::InvalidArgument, "invalid snapshot subject", {}, {}, "sensing.restore"));
880 }
881 if (!v.get("x").isNumber() || !v.get("y").isNumber() ||
882 (d.root().getInt("version") == 2 && !v.get("z").isNumber()) || next.subjects_.contains(v.getString("id")))
883 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
884 "invalid or duplicate snapshot position", "subjects", {},
885 "sensing.restore"));
886 auto subject = next.upsertPosition(v.getString("id"), float(v.getDouble("x")), float(v.getDouble("y")),
887 d.root().getInt("version") == 1 ? 0.f : float(v.getDouble("z")),
888 v.getString("faction"), v.getString("tags"), v.getString("visibleTo"));
889 if (!subject.ok())
890 return eve::Result<void>::failure(eve::Diagnostic::error(
891 eve::DiagnosticCode::InvalidArgument, "invalid snapshot subject", {}, {}, "sensing.restore"));
892 if (v.has("zones")) {
893 auto zones = next.setZones(v.getString("id"), v.getString("zones"));
894 if (!zones.ok())
895 return eve::Result<void>::failure(eve::Diagnostic::error(
896 eve::DiagnosticCode::InvalidArgument, "invalid snapshot subject zones", {}, {},
897 "sensing.restore"));
898 }
899 }
900 const float size = spatialIndex3D_ ? spatialIndex3D_->getCellSize()
901 : spatialIndex_ ? spatialIndex_->getCellSize()
902 : 1.f;
903 auto rebuilt = next.rebuildIndex(indexKind_, size);
904 if (!rebuilt) return Result<void>::failure(rebuilt.status());
905 spatialCellQueryCache.invalidate(this);
906 *this = std::move(next);
907 results_.clear();
908 lastQuery_ = {};
909 lastEnvironmentQuery_ = "null";
910 return eve::Result<void>::success();
911}
