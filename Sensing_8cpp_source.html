102eve::Result<int> SensingWorld::circle(float x, float y, float radius, std::string_view r, std::string_view e,
103 std::string_view i, std::string_view ef, std::string_view v, int limit) {
104 results_.clear();
105 if (!finite(x) || !finite(y) || !finite(radius) || radius < 0 || limit < 0) {
106 return sensingFailure<int>(eve::DiagnosticCode::InvalidArgument,
107 "circle query requires finite coordinates, non-negative radius and limit", "query");
108 }
109 auto R = csv(r), E = csv(e), I = csv(i), F = csv(ef);
110 for (auto& [id, s] : subjects_)
111 if ((s.x - x) * (s.x - x) + (s.y - y) * (s.y - y) <= radius * radius && accepts(s, R, E, I, F, v))
112 results_.push_back({id, s.x, s.y, 0});
113 return eve::Result<int>::success(finish(x, y, limit));

115eve::Result<int> SensingWorld::box(float a, float b, float c, float d, std::string_view r, std::string_view e,
116 std::string_view i, std::string_view ef, std::string_view v, int limit) {
117 results_.clear();

119 return sensingFailure<int>(eve::DiagnosticCode::InvalidArgument,
120 "box query requires finite ordered bounds and a non-negative limit", "query");
121 }
122 auto R = csv(r), E = csv(e), I = csv(i), F = csv(ef);
123 float x = (a + c) / 2, y = (b + d) / 2;
124 for (auto& [id, s] : subjects_)

126 results_.push_back({id, s.x, s.y, 0});
127 return eve::Result<int>::success(finish(x, y, limit));

146eve::Result<void> SensingWorld::restoreJson(const std::string& j) {
147 std::string err;
148 auto d = eve::json::Document::parse(j, &err);
149 if (!d.valid() || !d.root().isObject() || d.root().getString("schema") != "eve.sensing.world" ||
150 d.root().getInt("version") != 1 || !d.root().get("subjects").isArray()) {
151 return eve::Result<void>::failure(eve::Diagnostic::error(
152 eve::DiagnosticCode::ParseError, err.empty() ? "invalid snapshot" : err, {}, {}, "sensing.restore"));
153 }
154 SensingWorld next;
155 auto a = d.root().get("subjects");
156 for (size_t n = 0; n < a.size(); ++n) {
157 auto v = a.at(n);
158 if (!v.isObject()) {
159 return eve::Result<void>::failure(eve::Diagnostic::error(
160 eve::DiagnosticCode::InvalidArgument, "invalid snapshot subject", {}, {}, "sensing.restore"));
161 }
162 auto subject = next.upsert(v.getString("id"), float(v.getDouble("x")), float(v.getDouble("y")),
163 v.getString("faction"), v.getString("tags"), v.getString("visibleTo"));
164 if (!subject.ok())
165 return eve::Result<void>::failure(eve::Diagnostic::error(
166 eve::DiagnosticCode::InvalidArgument, "invalid snapshot subject", {}, {}, "sensing.restore"));
167 }
168 subjects_ = std::move(next.subjects_);
169 results_.clear();
170 return eve::Result<void>::success();
171}
