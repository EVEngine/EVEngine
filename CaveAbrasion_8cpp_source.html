16CaveAbrasionSample sampleCaveFloodAbrasion(const CaveAbrasionInput& input) {
17 CaveAbrasionSample result;
18 if (input.sedimentLoad <= 0.f) return result;
19
20 // angle is zero at the crown and pi at the floor. Suspended impacts and
21 // saltating grains rapidly lose erosive power above the near-bed zone.
22 result.floorMask = std::pow(std::max(0.f, -std::cos(input.angle)), 1.65f);
23 result.toolAvailability = smoothstep(0.025f, 0.28f, input.sedimentLoad);
24 result.coverProtection = std::clamp(input.sedimentLoad * input.sedimentLoad * 0.68f, 0.f, 0.68f);
25 const float toolsAndCover = result.toolAvailability * (1.f - result.coverProtection);
26
27 const float seedPhase = float(input.seed % 2039u) * 0.003081503f;
28 // Low-frequency coherent cells represent persistent recirculation and
29 // impact patches. They are enhanced at bends but remain present in straight floods.
30 const float cell = 0.5f + 0.5f * std::sin(input.along * 12.5f + seedPhase +
31 std::sin(input.along * 3.1f + seedPhase * 1.7f) * 1.15f);
32 const float lateralCell = 0.5f + 0.5f * std::cos((input.angle - 3.1415926535f) * 3.2f +
33 std::sin(input.along * 5.3f + seedPhase * 0.7f) * 0.8f);
34 const float bendSide = std::max(0.f, std::cos(input.angle - input.outerBankAngle));
35 result.vortexMask = std::clamp(cell * lateralCell * (0.55f + input.bendStrength * bendSide * 0.75f), 0.f, 1.f);
36 const float fluting =
37 0.35f + 0.65f * smoothstep(-0.45f, 0.65f, std::sin(input.along * 25.f + input.angle * 2.f + seedPhase * 2.3f));
38 const float streamPower = std::pow(std::clamp(input.hydraulicIntensity, 0.f, 1.7f), 1.25f);
39 result.erosion = std::clamp(
40 result.floorMask * toolsAndCover * streamPower * (0.38f + 0.62f * result.vortexMask) * fluting, 0.f, 1.f);
41 return result;
42}
