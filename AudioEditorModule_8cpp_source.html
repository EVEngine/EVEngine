22class AudioEditorModule::TargetFactory final : public editor::IEditorAutomationTargetFactory {
23public:
24 std::vector<std::string_view> types() const override { return {"audio-source", "audio-mixer", "audio-effects"}; }

26 editor::Result<editor::AutomationOwnedTarget> create(const editor::TargetId& target, std::string_view type,
27 const editor::EditorValue::Object&) override {
28 editor::AutomationOwnedTarget owned;
29 if (type == "audio-source") {
30 owned.target = std::make_unique<audio_editing::AudioSourceTarget>(target.value());
31 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
32 }
33 if (type == "audio-mixer") {
34 owned.target = std::make_unique<audio_editing::AudioMixerTarget>(target.value());
35 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
36 }
37 owned.target = std::make_unique<audio_editing::AudioEffectChainTarget>(target.value());
38 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
39 }
40};
