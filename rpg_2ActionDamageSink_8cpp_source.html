37Result<void> RPGActionDamageSink::apply(const action::ActionDamageBinding& binding,
38 const action::ActionNotifyContext& context) {
39 if (context.preview || context.scrubbing) return Result<void>::success(Status::success(StatusCode::NoOp));
40 if (binding.poiseAmount != 0.0 || binding.knockback.x != 0.0 || binding.knockback.y != 0.0 ||
41 binding.knockback.z != 0.0)
42 return failure(DiagnosticCode::Unsupported, "RPG Vitals damage does not own poise or knockback");
43 if (binding.targetIndex >= context.targets.size() || !impl_->resolver)
44 return failure(DiagnosticCode::NotFound, "RPG damage target or resolver is unavailable");
45 const auto targetHandle = context.targets[binding.targetIndex];
46 if (!ecs::try_get(targetHandle)) return failure(DiagnosticCode::StaleHandle, "Action target was destroyed");
47 const auto mappedTarget = impl_->resolver(targetHandle);
48 if (!mappedTarget || !mappedTarget->subject.isValid())
49 return failure(DiagnosticCode::NotFound, "Action target has no RPG identity mapping");
50 std::optional<RPGDamageTarget> mappedSource;
51 if (context.source) {
52 if (!ecs::try_get(*context.source)) return failure(DiagnosticCode::StaleHandle, "Action source was destroyed");
53 mappedSource = impl_->resolver(*context.source);
54 if (!mappedSource || !mappedSource->subject.isValid())
55 return failure(DiagnosticCode::NotFound, "Action source has no RPG identity mapping");
56 }
57 // Resolve only after all host callbacks have returned; retain no borrowed ECS addresses.
58 auto* target = resolve(mappedTarget->actor);
59 auto* source = mappedSource ? resolve(mappedSource->actor) : nullptr;
60 if (!target || (mappedSource && !source))
61 return failure(DiagnosticCode::StaleHandle, "Mapped RPG actor was destroyed");
62
63 settlement::SettlementRequest request;
64 request.source = mappedSource ? mappedSource->subject : SubjectRef::nil();
65 request.target = mappedTarget->subject;
66 request.kind = "damage";
67 request.resource = "hp";
68 request.magnitude = binding.amount;
69 request.tags = {binding.damageType};
70 request.tick = context.tick;
71 RPGSettlementAdapter adapter(*target, request.target, source, request.source);
72 const double before = VitalsSystem::getCurrent(target, "hp");
73 impl_->receipts.reserve(impl_->receipts.size() + 1);
74 auto settled = impl_->pipeline.settle(request, adapter);
75 if (!settled) return Result<void>::failure(settled.status());
76 const double applied = settled.value().applied;
77 impl_->receipts.push_back({request.source, request.target, context.executionId, context.tick, applied,
78 before > 0.0 && VitalsSystem::getCurrent(target, "hp") == 0.0});
79 return Result<void>::success(Status::success(StatusCode::Applied));
80}
