27std::vector<std::uint8_t> lzw(std::span<const std::uint8_t> source, std::size_t expected) {
28 std::array<std::uint16_t, 4096> prefix{};
29 std::array<std::uint8_t, 4096> suffix{}, stack{};
30 std::vector<std::uint8_t> out;
31 out.reserve(expected);
32 std::size_t bit = 0;
33 unsigned width = 9, next = 258, previous = 4096;
34 std::uint8_t first = 0;
35 bool cleared = false;
36 while (true) {
37 if (bit > source.size() * 8 || width > source.size() * 8 - bit) throw Invalid{};
38 unsigned code = 0;
39 for (unsigned i = 0; i < width; ++i, ++bit) code = (code << 1) | ((source[bit / 8] >> (7 - bit % 8)) & 1);
40 if (code == 256) {
41 width = 9;
42 next = 258;
43 previous = 4096;
44 cleared = true;
45 continue;
46 }
47 if (!cleared) throw Invalid{};
48 if (code == 257) {
49 if (out.size() != expected) throw Invalid{};
50 return out;
51 }
52 const unsigned original = code;
53 std::size_t length = 0;
54 if (code == next && previous != 4096) {
55 stack[length++] = first;
56 code = previous;
57 } else if (code >= next)
58 throw Invalid{};
59 while (code >= 258) {
60 if (code >= next || length >= stack.size()) throw Invalid{};
61 stack[length++] = suffix[code];
62 code = prefix[code];
63 }
64 if (code >= 256 || length >= stack.size()) throw Invalid{};
65 first = std::uint8_t(code);
66 stack[length++] = first;
67 if (length > expected - out.size()) throw Invalid{};
68 while (length) out.push_back(stack[--length]);
69 if (previous != 4096 && next < 4096) {
70 prefix[next] = std::uint16_t(previous);
71 suffix[next] = first;
72 ++next;
73 // TIFF uses early-change code widths, unlike GIF's late-change LZW.
74 if (width < 12 && next == (1u << width) - 1) ++width;
75 }
76 previous = original;
77 }
78}
