18Result<MeshBuild> projectMeshUvResult(const MeshBuild& input, std::string_view mode, float scale, float offsetU,
19 float offsetV) {
20 if (input.empty() || input.positions().size() % 3u != 0u || input.normals().size() != input.positions().size())
21 return fail<MeshBuild>(DiagnosticCode::InvalidArgument, "UV projection requires a valid non-empty mesh", "mesh");
22 if ((mode != "planar" && mode != "box" && mode != "spherical") || !std::isfinite(scale) || scale <= 0.f ||
23 !std::isfinite(offsetU) || !std::isfinite(offsetV))
24 return fail<MeshBuild>(DiagnosticCode::InvalidArgument, "invalid UV projection parameters", "projection");
25 MeshBuild output = input;
26 output.uvs().assign(static_cast<std::size_t>(output.getVertexCount()) * 2u, 0.f);
27 constexpr float pi = 3.14159265358979323846f;
28 for (int i = 0; i < output.getVertexCount(); ++i) {
29 const std::size_t p = static_cast<std::size_t>(i) * 3u, uv = static_cast<std::size_t>(i) * 2u;
30 const float x = output.positions()[p], y = output.positions()[p + 1u], z = output.positions()[p + 2u];
31 float u = x, v = z;
32 if (mode == "spherical") {
33 const float length = std::sqrt(x * x + y * y + z * z);
34 if (length > 1e-7f) { u = std::atan2(z, x) / (2.f * pi) + 0.5f; v = std::asin(std::clamp(y / length, -1.f, 1.f)) / pi + 0.5f; }
35 } else if (mode == "box") {
36 const float nx = std::abs(output.normals()[p]), ny = std::abs(output.normals()[p + 1u]), nz = std::abs(output.normals()[p + 2u]);
37 if (nx >= ny && nx >= nz) { u = z; v = y; }
38 else if (ny >= nz) { u = x; v = z; }
39 else { u = x; v = y; }
40 }
41 output.uvs()[uv] = u * scale + offsetU;
42 output.uvs()[uv + 1u] = v * scale + offsetV;
43 }
44 output.setMeta("uvProjection", std::string(mode));
45 return Result<MeshBuild>::success(std::move(output));
46}
