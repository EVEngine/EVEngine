100Result<Problem> decodeProblem(const Value& value) {
101 try {
102 const auto& o = object(value, {"schema", "version", "parameters", "targets", "trainingSeeds", "validationSeeds",
103 "searchSeed", "candidates", "steps", "dt", "changeWeight"});
104 if (text(required(o, "schema")) != "evengine.balance.problem" || u32(required(o, "version")) != 1)
105 throw ParseError("Unsupported balance schema/version");
106 Problem p;
107 for (const auto& item : array(required(o, "parameters"), 64)) {
108 const auto& x = object(item, {"name", "initial", "min", "max"});
109 p.parameters.push_back({text(required(x, "name")), number(required(x, "initial")),
110 number(required(x, "min")), number(required(x, "max"))});
111 }
112 for (const auto& item : array(required(o, "targets"), 128)) {
113 const auto& x = object(item, {"name", "min", "max", "scale", "weight"});
114 Target t{text(required(x, "name")), number(required(x, "min")), number(required(x, "max")),
115 number(required(x, "scale")), 1};
116 if (auto it = x.find("weight"); it != x.end()) t.weight = number(it->second);
117 p.targets.push_back(std::move(t));
118 }
119 if (auto it = o.find("trainingSeeds"); it != o.end()) {
120 p.trainingSeeds.clear();
121 for (const auto& s : array(it->second, 128)) p.trainingSeeds.push_back(integer(s));
122 }
123 if (auto it = o.find("validationSeeds"); it != o.end()) {
124 p.validationSeeds.clear();
125 for (const auto& s : array(it->second, 128)) p.validationSeeds.push_back(integer(s));
126 }
127 if (auto it = o.find("searchSeed"); it != o.end()) p.searchSeed = integer(it->second);
128 if (auto it = o.find("candidates"); it != o.end()) p.candidates = u32(it->second);
129 if (auto it = o.find("steps"); it != o.end()) p.steps = u32(it->second);
130 if (auto it = o.find("dt"); it != o.end()) p.dt = number(it->second);
131 if (auto it = o.find("changeWeight"); it != o.end()) p.changeWeight = number(it->second);
132 return Result<Problem>::success(std::move(p));
133 } catch (const ParseError& e) {
134 return Result<Problem>::failure(
135 Diagnostic::error(DiagnosticCode::ParseError, e.what(), {}, {}, "agent.balance"));
136 }
137}
