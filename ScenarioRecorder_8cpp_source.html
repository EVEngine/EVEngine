93void ScenarioRecorder::onEventConsumed(const eve::platform_event::Message& msg) {
94 if (!recording_ || !vm_) return;
95 ScenarioEvent ev;
96 ev.name = msg.name;
97 for (const auto& a : msg.args) {
98 switch (a.type) {
99 case eve::platform_event::Variant::Type::String: ev.args.push_back(a.s); break;
100 case eve::platform_event::Variant::Type::Int:
101 ev.args.push_back(std::to_string(static_cast<long long>(a.i)));
102 break;
103 case eve::platform_event::Variant::Type::Nil: ev.args.push_back({}); break;
104 default:
105 // Ptr payloads (borrowed/owned native pointers) are not serializable; drop.
106 break;
107 }
108 }
109 if (frames_.empty() || frames_.back().frame != currentFrame_)
110 frames_.push_back(ScenarioFrame{currentFrame_});
111 frames_.back().events.push_back(std::move(ev));
112}

136bool ScenarioRecorder::stageFrame() {
137 if (!replay_ || !eventModule_) return false;
138 if (replayIndex_ >= static_cast<int>(frames_.size())) {
139 replay_ = false;
140 return false;
141 }
142 const ScenarioFrame& frame = frames_[replayIndex_++];
143 for (const auto& ev : frame.events) {
144 std::vector<eve::platform_event::Variant> args;
145 for (const auto& a : ev.args) args.push_back(eve::platform_event::Variant::makeString(a));
146 eventModule_->push(new eve::platform_event::Message(ev.name, args));
147 }
148 return true;
149}

151bool ScenarioRecorder::save(const std::string& baseline,
152 const std::vector<ScenarioFrame>& frames,
153 const std::string& errorReport, const std::string& errorSite,
154 const std::string& path, std::string* err) {
155 try {
156 Poco::JSON::Object root;
157 root.set("version", "1");
158 root.set("baseline", baseline);
159 Poco::JSON::Array framesArr;
160 for (const auto& f : frames) {
161 Poco::JSON::Object fo;
162 fo.set("frame", f.frame);
163 Poco::JSON::Array evs;
164 for (const auto& e : f.events) {
165 Poco::JSON::Object eo;
166 eo.set("name", e.name);
167 Poco::JSON::Array args;
168 for (const auto& a : e.args)
169 args.add(a);
170 eo.set("args", args);
171 evs.add(eo);
172 }
173 fo.set("events", evs);
174 framesArr.add(fo);
175 }
176 root.set("frames", framesArr);
177 root.set("errorReport", errorReport);
178 root.set("errorSite", errorSite);
179 std::ostringstream oss;
180 root.stringify(oss);
181 std::ofstream ofs(path, std::ios::binary);
182 if (!ofs) {
183 if (err) *err = "cannot write " + path;
184 return false;
185 }
186 ofs << oss.str();
187 return static_cast<bool>(ofs);
188 } catch (const std::exception& e) {
189 if (err) *err = e.what();
190 return false;
191 }
192}

194bool ScenarioRecorder::load(const std::string& path, std::string* baseline,
195 std::vector<ScenarioFrame>* frames, std::string* errorReport,
196 std::string* errorSite, std::string* err) {
197 try {
198 std::ifstream ifs(path, std::ios::binary);
199 if (!ifs) {
200 if (err) *err = "cannot read " + path;
201 return false;
202 }
203 std::ostringstream oss;
204 oss << ifs.rdbuf();
205 Poco::JSON::Parser parser;
206 Poco::Dynamic::Var result = parser.parse(oss.str());
207 Poco::JSON::Object::Ptr root = result.extract<Poco::JSON::Object::Ptr>();
208 if (baseline) *baseline = root->getValue<std::string>("baseline");
209 if (errorReport) *errorReport = root->optValue<std::string>("errorReport", "");
210 if (errorSite) *errorSite = root->optValue<std::string>("errorSite", "");
211 if (frames) {
212 frames->clear();
213 Poco::JSON::Array::Ptr fa = root->getArray("frames");
214 if (fa) {
215 for (size_t i = 0; i < fa->size(); ++i) {
216 Poco::JSON::Object::Ptr fo = fa->getObject(static_cast<unsigned int>(i));
217 ScenarioFrame f;
218 f.frame = fo->optValue<int>("frame", static_cast<int>(i));
219 Poco::JSON::Array::Ptr evs = fo->getArray("events");
220 if (evs) {
221 for (size_t j = 0; j < evs->size(); ++j) {
222 Poco::JSON::Object::Ptr eo =
223 evs->getObject(static_cast<unsigned int>(j));
224 ScenarioEvent e;
225 e.name = eo->getValue<std::string>("name");
226 Poco::JSON::Array::Ptr args = eo->getArray("args");
227 if (args) {
228 for (size_t k = 0; k < args->size(); ++k)
229 e.args.push_back(args->getElement<std::string>(
230 static_cast<unsigned int>(k)));
231 }
232 f.events.push_back(std::move(e));
233 }
234 }
235 frames->push_back(std::move(f));
236 }
237 }
238 }
239 return true;
240 } catch (const std::exception& e) {
241 if (err) *err = e.what();
242 return false;
243 }
244}
