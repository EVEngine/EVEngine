27void quatFromAxisAngle(float ax, float ay, float az, float angle, float& qx, float& qy, float& qz, float& qw) {
28 const float half = 0.5f * angle;
29 const float s = std::sin(half);
30 qx = ax * s;
31 qy = ay * s;
32 qz = az * s;
33 qw = std::cos(half);

46void stepTowardZero(float dt, float omega, float zeta, float& y, float& yd) {
47 if (dt <= 0.f || omega <= 0.f) return;
48 const float kp = omega * omega;
49 const float kd = 2.f * zeta * omega;
50 yd += dt * (-kp * y - kd * yd);
51 y += dt * yd;
52}

65void PhysicalBalancePose::ensureBones() {
66 const int n = skeleton_->getBoneCount();
67 if (pose_.getBoneCount() != n) pose_.resize(n);
68 if (target_.getBoneCount() != n) target_.resize(n);
69 if (static_cast<int>(masses_.size()) != n) {
70 masses_.assign(static_cast<size_t>(n), 1.f);
71 recoils_.assign(static_cast<size_t>(n), Recoil{});
72 }
73 if (n <= 0) {
74 supportBone_ = 0;
75 balanceBone_ = 0;
76 return;
77 }
78 if (!boneInRange(supportBone_)) supportBone_ = 0;
79 if (!boneInRange(balanceBone_)) balanceBone_ = n > 1 ? 1 : 0;
80}

230eve::Result<void> PhysicalBalancePose::applyImpulse(int boneIndex, float impulseX, float impulseY, float impulseZ,
231 float pointX, float pointY, float pointZ) {
232 if (!boneInRange(boneIndex))
233 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
234 "physical balance impulse bone is out of range"));
235 if (!std::isfinite(impulseX) || !std::isfinite(impulseY) || !std::isfinite(impulseZ) || !std::isfinite(pointX) ||
236 !std::isfinite(pointY) || !std::isfinite(pointZ))
237 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
238 "physical balance impulse values must be finite"));
239 const float mag2 = impulseX * impulseX + impulseY * impulseY + impulseZ * impulseZ;
240 if (mag2 < kImpulseEps) return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
241
242 pose_.computeWorld(skeleton_);
243 const TransformTRS& support = pose_.world(supportBone_);
244 const float rx = pointX - support.px;
245 const float ry = pointY - support.py;
246 const float rz = pointZ - support.pz;
247 pendingLeanTx_ += ry * impulseZ - rz * impulseY;
248 pendingLeanTz_ += rx * impulseY - ry * impulseX;
249
250 const TransformTRS& bone = pose_.world(boneIndex);
251 Recoil& rec = recoils_[static_cast<size_t>(boneIndex)];
252 const float brx = pointX - bone.px;
253 const float bry = pointY - bone.py;
254 const float brz = pointZ - bone.pz;
255 rec.pendingTx += bry * impulseZ - brz * impulseY;
256 rec.pendingTy += brz * impulseX - brx * impulseZ;
257 rec.pendingTz += brx * impulseY - bry * impulseX;
258 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
259}

261void PhysicalBalancePose::writeOverlayPose() {
262 pose_.copyFrom(&target_);
263 const int n = pose_.getBoneCount();
264 for (int i = 0; i < n; ++i) {
265 const Recoil& rec = recoils_[static_cast<size_t>(i)];
266 if (std::fabs(rec.x) < 1e-8f && std::fabs(rec.y) < 1e-8f && std::fabs(rec.z) < 1e-8f) continue;
267 float qx = 0.f, qy = 0.f, qz = 0.f, qw = 1.f;
268 float rx = 0.f, ry = 0.f, rz = 0.f, rw = 1.f;
269 float tx = 0.f, ty = 0.f, tz = 0.f, tw = 1.f;
270 quatFromAxisAngle(1.f, 0.f, 0.f, rec.x, qx, qy, qz, qw);
271 quatFromAxisAngle(0.f, 1.f, 0.f, rec.y, rx, ry, rz, rw);
272 quatMul(rx, ry, rz, rw, qx, qy, qz, qw, tx, ty, tz, tw);
273 quatFromAxisAngle(0.f, 0.f, 1.f, rec.z, qx, qy, qz, qw);
274 float ox = 0.f, oy = 0.f, oz = 0.f, ow = 1.f;
275 quatMul(qx, qy, qz, qw, tx, ty, tz, tw, ox, oy, oz, ow);
276 applyParentRotation(pose_.local(i), ox, oy, oz, ow);
277 }
278 if (boneInRange(balanceBone_) && (std::fabs(leanX_) > 1e-8f || std::fabs(leanZ_) > 1e-8f)) {
279 float qx = 0.f, qy = 0.f, qz = 0.f, qw = 1.f;
280 float rx = 0.f, ry = 0.f, rz = 0.f, rw = 1.f;
281 quatFromAxisAngle(1.f, 0.f, 0.f, leanX_, qx, qy, qz, qw);
282 quatFromAxisAngle(0.f, 0.f, 1.f, leanZ_, rx, ry, rz, rw);
283 float ox = 0.f, oy = 0.f, oz = 0.f, ow = 1.f;
284 quatMul(qx, qy, qz, qw, rx, ry, rz, rw, ox, oy, oz, ow);
285 applyParentRotation(pose_.local(balanceBone_), ox, oy, oz, ow);
286 }
287 refreshDiagnostics();
288}

290void PhysicalBalancePose::refreshDiagnostics() {
291 pose_.computeWorld(skeleton_);
292 const int n = pose_.getBoneCount();
293 float massSum = 0.f;
294 comX_ = comY_ = comZ_ = 0.f;
295 for (int i = 0; i < n; ++i) {
296 const float mass = masses_[static_cast<size_t>(i)];
297 if (mass <= 0.f) continue;
298 const TransformTRS& world = pose_.world(i);
299 comX_ += world.px * mass;
300 comY_ += world.py * mass;
301 comZ_ += world.pz * mass;
302 massSum += mass;
303 }
304 if (massSum > 1e-8f) {
305 const float inv = 1.f / massSum;
306 comX_ *= inv;
307 comY_ *= inv;
308 comZ_ *= inv;
309 }
310 if (boneInRange(supportBone_)) {
311 const TransformTRS& support = pose_.world(supportBone_);
312 supportX_ = support.px;
313 supportY_ = support.py;
314 supportZ_ = support.pz;
315 }

318void PhysicalBalancePose::updateUnchecked(float dt) {
319 if (dt <= 0.f || !hasTarget_) return;
320 ensureBones();
321
322 leanVelX_ += pendingLeanTx_ / inertia_;
323 leanVelZ_ += pendingLeanTz_ / inertia_;
324 pendingLeanTx_ = pendingLeanTz_ = 0.f;
325
326 const float omega = kTwoPi * recoveryHz_;
327 const float height = std::max(pendulumHeight_, 1e-4f);
328 const float grav = gravity_ / height;
329 const float kp = std::max(omega * omega - grav, 1e-4f);
330 const float kd = 2.f * recoveryZeta_ * omega;
331 leanVelX_ += dt * (-kp * leanX_ - kd * leanVelX_);
332 leanVelZ_ += dt * (-kp * leanZ_ - kd * leanVelZ_);
333 leanX_ += dt * leanVelX_;
334 leanZ_ += dt * leanVelZ_;
335 const float clampedX = clampf(leanX_, -maxLean_, maxLean_);
336 const float clampedZ = clampf(leanZ_, -maxLean_, maxLean_);
337 if (clampedX != leanX_) leanVelX_ = 0.f;
338 if (clampedZ != leanZ_) leanVelZ_ = 0.f;
339 leanX_ = clampedX;
340 leanZ_ = clampedZ;
341
342 const float recoilOmega = kTwoPi * recoilHz_;
343 for (Recoil& rec : recoils_) {
344 rec.vx += rec.pendingTx / recoilInertia_;
345 rec.vy += rec.pendingTy / recoilInertia_;
346 rec.vz += rec.pendingTz / recoilInertia_;
347 rec.pendingTx = rec.pendingTy = rec.pendingTz = 0.f;
348 stepTowardZero(dt, recoilOmega, recoilZeta_, rec.x, rec.vx);
349 stepTowardZero(dt, recoilOmega, recoilZeta_, rec.y, rec.vy);
350 stepTowardZero(dt, recoilOmega, recoilZeta_, rec.z, rec.vz);
351 rec.x = clampf(rec.x, -maxLean_, maxLean_);
352 rec.y = clampf(rec.y, -maxLean_, maxLean_);
353 rec.z = clampf(rec.z, -maxLean_, maxLean_);
354 }
355
356 writeOverlayPose();
357}
