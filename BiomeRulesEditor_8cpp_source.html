74biome_editing::Result<void> BiomeRulesEditor::configureWorkspace(editor::EditorWorkspace& workspace) const {
75 editor::EditorWorkspace candidate = workspace;
76 struct Panel {
77 const char* id;
78 const char* title;
79 const char* region;
80 const char* context;
81 int order;
82 };
83 constexpr Panel panels[] = {
84 {"biome.layers", "Layers", "left", "list", 100},
85 {"biome.preview", "World Preview", "center", "preview", 100},
86 {"biome.inspector", "Biome Inspector", "right", "inspector", 100},
87 {"biome.assets", "Layer Assets", "bottom", "assets", 100},
88 };
89 for (const auto& panel : panels) {
90 if (!candidate.registerPanel(panel.id, panel.title, panel.region, panel.order) ||
91 !candidate.setPanelCapability(panel.id, "biome.rules") ||
92 !candidate.setPanelContext(panel.id, panel.context))
93 return editorError(biome_editing::EditorStatus::Rejected, "editor.biome.workspace-conflict",
94 "Could not install the biome workspace composition");
95 }
96 if (!candidate.activatePanel("biome.preview"))
97 return editorError(biome_editing::EditorStatus::Rejected, "editor.biome.workspace-activate",
98 "Could not activate the biome preview panel");
99 workspace = std::move(candidate);
100 return eve::editing::applied<void>();

116biome_editing::Result<void> BiomeRulesEditor::commit(biome_editing::Result<biome_editing::DomainOperation> operation,
117 std::string label) {
118 if (!operation.ok()) return biome_editing::Result<void>::failure(operation.status());
119 editor::TransactionSpec spec;
120 spec.id = editor::TransactionId("biome.rules.tx." + std::to_string(++txSequence_));
121 spec.label = std::move(label);
122 spec.target = editor::TargetId(target_.targetId());
123 spec.baseRevision = target_.revision();
124 auto begun = transactions_.begin(std::move(spec));
125 if (!begun.ok()) return editorError(begun.code(), "editor.biome.begin", "Could not begin the biome transaction");
126 auto appended = transactions_.append(std::move(operation).takeValue());
127 if (!appended.ok()) {
128 auto discarded = transactions_.discard();
129 if (!discarded.ok()) discarded.ignore("pending biome transaction already inactive");
130 return biome_editing::Result<void>::failure(appended.status());
131 }
132 auto committed = transactions_.commit();
133 if (!committed.ok()) return biome_editing::Result<void>::failure(committed.status());
134 return refreshPreview();
135}

137biome_editing::Result<void> BiomeRulesEditor::refreshPreview() {
138 auto published = runtime_.publish(target_, resolver_);
139 if (!published.ok()) return biome_editing::Result<void>::failure(published.status());
140 auto generated = runtime_.preview(&forestDomain_, spacing_, seed_, 0.0f, runtime_.revision());
141 if (!generated.ok()) return biome_editing::Result<void>::failure(generated.status());
142 std::vector<PreviewPoint> next;
143 const auto& points = generated.value();
144 if (points) {
145 next.reserve(static_cast<std::size_t>(points->getCount()));
146 for (int i = 0; i < points->getCount(); ++i) {
147 PreviewPoint point;
148 point.x = points->getX(i);
149 point.z = points->getZ(i);
150 point.asset = points->getStringAttribute(i, "asset", {});
151 next.push_back(std::move(point));
152 }
153 }
154 points_ = std::move(next);
155 previewRevision_ = target_.revision();
156 return eve::editing::applied<void>();
157}
