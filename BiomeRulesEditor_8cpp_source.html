72biome_editing::EditorResult<void> BiomeRulesEditor::configureWorkspace(editor::EditorWorkspace& workspace) const {
73 editor::EditorWorkspace candidate = workspace;
74 struct Panel {
75 const char* id;
76 const char* title;
77 const char* region;
78 const char* context;
79 int order;
80 };
81 constexpr Panel panels[] = {
82 {"biome.layers", "Layers", "left", "list", 100},
83 {"biome.preview", "World Preview", "center", "preview", 100},
84 {"biome.inspector", "Biome Inspector", "right", "inspector", 100},
85 {"biome.assets", "Layer Assets", "bottom", "assets", 100},
86 };
87 for (const auto& panel : panels) {
88 if (!candidate.registerPanel(panel.id, panel.title, panel.region, panel.order) ||
89 !candidate.setPanelCapability(panel.id, "biome.rules") ||
90 !candidate.setPanelContext(panel.id, panel.context))
91 return editorError(biome_editing::EditorStatus::Rejected, "editor.biome.workspace-conflict",
92 "Could not install the biome workspace composition");
93 }
94 if (!candidate.activatePanel("biome.preview"))
95 return editorError(biome_editing::EditorStatus::Rejected, "editor.biome.workspace-activate",
96 "Could not activate the biome preview panel");
97 workspace = std::move(candidate);
98 return eve::editing::applied<void>();

114biome_editing::EditorResult<void> BiomeRulesEditor::commit(
115 biome_editing::EditorResult<biome_editing::DomainOperation> operation, std::string label) {
116 if (!operation.ok())
117 return biome_editing::EditorResult<void>::failure(operation.status());
118 editor::TransactionSpec spec;
119 spec.id = editor::TransactionId("biome.rules.tx." + std::to_string(++txSequence_));
120 spec.label = std::move(label);
121 spec.target = editor::TargetId(target_.targetId());
122 spec.baseRevision = target_.revision();
123 auto begun = transactions_.begin(std::move(spec));
124 if (!begun.ok())
125 return editorError(begun.code(), "editor.biome.begin", "Could not begin the biome transaction");
126 auto appended = transactions_.append(std::move(operation).takeValue());
127 if (!appended.ok()) {
128 auto discarded = transactions_.discard();
129 if (!discarded.ok()) discarded.ignore("pending biome transaction already inactive");
130 return biome_editing::EditorResult<void>::failure(appended.status());
131 }
132 auto committed = transactions_.commit();
133 if (!committed.ok())
134 return biome_editing::EditorResult<void>::failure(committed.status());
135 return refreshPreview();
136}

138biome_editing::EditorResult<void> BiomeRulesEditor::refreshPreview() {
139 auto published = runtime_.publish(target_, resolver_);
140 if (!published.ok())
141 return biome_editing::EditorResult<void>::failure(published.status());
142 auto generated = runtime_.preview(&forestDomain_, spacing_, seed_, 0.0f, runtime_.revision());
143 if (!generated.ok())
144 return biome_editing::EditorResult<void>::failure(generated.status());
145 std::vector<PreviewPoint> next;
146 const auto& points = generated.value();
147 if (points) {
148 next.reserve(static_cast<std::size_t>(points->getCount()));
149 for (int i = 0; i < points->getCount(); ++i) {
150 PreviewPoint point;
151 point.x = points->getX(i);
152 point.z = points->getZ(i);
153 point.asset = points->getStringAttribute(i, "asset", {});
154 next.push_back(std::move(point));
155 }
156 }
157 points_ = std::move(next);
158 previewRevision_ = target_.revision();
159 return eve::editing::applied<void>();
160}
