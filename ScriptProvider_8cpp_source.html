83Result<void> pushProviderMethod(HSQUIRRELVM vm, HSQOBJECT instance, const std::string& method) {
84 const SQInteger top = sq_gettop(vm);
85 sq_pushobject(vm, instance);
86 sq_pushstring(vm, method.c_str(), -1);
87 if (SQ_FAILED(sq_get(vm, -2))) {
88 sq_settop(vm, top);
89 return Result<void>::failure(
90 providerDiagnostic(DiagnosticCode::Unsupported, "script provider has no method '" + method + "'", method));
91 }
92 const SQObjectType type = sq_gettype(vm, -1);
93 if (type != OT_CLOSURE && type != OT_NATIVECLOSURE) {
94 sq_settop(vm, top);
95 return Result<void>::failure(providerDiagnostic(
96 DiagnosticCode::Unsupported, "script provider member '" + method + "' is not a method", method));
97 }
98 return Result<void>::success();
99}

109Result<void> pushScriptArg(HSQUIRRELVM vm, const ScriptArg& arg) {
110 switch (arg.kind) {
111 case ScriptArg::Kind::Integer: sq_pushinteger(vm, SQInteger(arg.integer)); break;
112 case ScriptArg::Kind::Number: sq_pushfloat(vm, SQFloat(arg.number)); break;
113 case ScriptArg::Kind::Boolean: sq_pushbool(vm, arg.boolean ? SQTrue : SQFalse); break;
114 case ScriptArg::Kind::Text: sq_pushstring(vm, arg.text.c_str(), SQInteger(arg.text.size())); break;
115 case ScriptArg::Kind::Table: {
116 auto pushed = script::pushValue(vm, Value::object(arg.table));
117 if (!pushed.ok()) return pushed;
118 break;
119 }
120 }
121 return Result<void>::success();
122}

125Result<double> readNumericResult(HSQUIRRELVM vm, const std::string& methodName) {
126 const SQObjectType resultType = sq_gettype(vm, -1);
127 if (resultType != OT_FLOAT && resultType != OT_INTEGER) {
128 return Result<double>::failure(

171Result<std::vector<double>> readNumericArray(HSQUIRRELVM vm, const std::string& methodName) {
172 if (sq_gettype(vm, -1) != OT_ARRAY) {
173 return Result<std::vector<double>>::failure(
174 providerDiagnostic(DiagnosticCode::InvalidArgument,
175 "script provider method '" + methodName + "' did not return an array", methodName));
176 }
177 const SQInteger count = sq_getsize(vm, -1);
178 if (count < 0 || count > kMaxResultElements) {
179 return Result<std::vector<double>>::failure(
180 providerDiagnostic(DiagnosticCode::InvalidArgument,
181 "script provider method '" + methodName + "' returned " + std::to_string(count) +
182 " elements; the limit is " + std::to_string(kMaxResultElements),
183 methodName));
184 }
185
186 std::vector<double> out;
187 out.reserve(std::size_t(count));
188 const SQInteger top = sq_gettop(vm);
189 for (SQInteger i = 0; i < count; ++i) {
190 sq_pushinteger(vm, i);
191 if (SQ_FAILED(sq_get(vm, -2))) break;
192 auto element = readNumericElement(vm, -1, methodName);
193 sq_settop(vm, top); // drop the element again
194 if (!element) return Result<std::vector<double>>::failure(element.status());
195 out.push_back(element.value());
196 }
197 return Result<std::vector<double>>::success(std::move(out));
198}

234template <class Reader>
235auto invokeProvider(const ssq::Object& instance, std::string_view method, std::span<const ScriptArg> args,
236 Reader&& readResult) -> std::invoke_result_t<Reader, HSQUIRRELVM, const std::string&> {
237 using Outcome = std::invoke_result_t<Reader, HSQUIRRELVM, const std::string&>;
238
239 HSQUIRRELVM vm = instance.getHandle();
240 const std::string methodName(method);
241 if (!vm) {
242 return Outcome::failure(
243 providerDiagnostic(DiagnosticCode::Failed, "script provider has no live VM", methodName));
244 }
245
246 const HSQOBJECT raw = instance.getRaw();
247 auto pushed = pushProviderMethod(vm, raw, methodName);
248 if (!pushed) return Outcome::failure(pushed.status());
249
250 const SQInteger top = sq_gettop(vm);
251 sq_pushobject(vm, raw); // `this`
252 for (const ScriptArg& arg : args) {
253 auto pushed = pushScriptArg(vm, arg);
254 if (!pushed) {
255 sq_settop(vm, top);
256 return Outcome::failure(pushed.status());
257 }
258 }
259
260 const SQInteger argCount = SQInteger(args.size()) + 1;
261 if (SQ_FAILED(sq_call(vm, argCount, SQTrue, SQFalse))) {
262 const std::string detail = lastScriptErrorText(vm);
263 sq_settop(vm, top);
264 return Outcome::failure(providerDiagnostic(
265 DiagnosticCode::Failed, "script provider method '" + methodName + "' raised: " + detail, methodName));
266 }
267
268 auto outcome = std::invoke(readResult, vm, methodName);
269 sq_settop(vm, top);
270 return outcome;
271}

325Result<ScriptProviderHandle> ScriptProviderRegistry::publish(std::string name, const ssq::Object& instance) {
326 if (name.empty()) {
327 return Result<ScriptProviderHandle>::failure(
328 providerDiagnostic(DiagnosticCode::InvalidArgument, "script provider name must not be empty", {}));
329 }
330 if (instance.isEmpty() || instance.getRaw()._type != OT_INSTANCE) {
331 return Result<ScriptProviderHandle>::failure(
332 providerDiagnostic(DiagnosticCode::InvalidArgument, "script provider must be a live class instance", name));
333 }
334
335 // Superseding a capability-bound provider must also remove its adapter; leaving
336 // it installed would keep routing native consumers into a dead script object.
337 auto previous = impl_->entries.find(name);
338 if (previous != impl_->entries.end() && previous->second.unbinder) {
339 auto retired = retireEntry(previous->second, name);
340 if (!retired) return Result<ScriptProviderHandle>::failure(retired.status());
341 }
342
343 // Re-lookup: running the unbinder above may have rehashed the map.
344 ProviderEntry& entry = impl_->entries[name];
345 entry.instance = script::OwnedSquirrelObject(instance);
346 entry.capability.clear();
347 entry.unbinder = {};
348 const std::uint64_t generation = ++entry.generation;
349 return Result<ScriptProviderHandle>::success(ScriptProviderHandle{std::move(name), generation});
350}

352Result<void> ScriptProviderRegistry::unload(const ScriptProviderHandle& handle) {
353 auto found = impl_->entries.find(handle.name);
354 if (found == impl_->entries.end() || !found->second.instance.isBound() ||
355 found->second.generation != handle.generation) {
356 return Result<void>::failure(providerDiagnostic(
357 DiagnosticCode::StaleHandle, "script provider generation is no longer published", handle.name));
358 }
359 // Keep the slot so the retired generation can never be resolved again, and so a
360 // later publish cannot hand an old handle a new instance.
361 return retireEntry(found->second, handle.name);
362}

425Result<ScriptProviderHandle> ScriptProviderRegistry::publishForCapability(std::string_view capability, std::string name,
426 const ssq::Object& instance) {
427 auto seam = impl_->capabilities.find(std::string(capability));
428 if (seam == impl_->capabilities.end()) {
429 return Result<ScriptProviderHandle>::failure(providerDiagnostic(
430 DiagnosticCode::NotFound,
431 "no script binding seam is declared for capability '" + std::string(capability) + "'", name));
432 }
433
434 // Copy the seam before publishing: the binder is user code and may redeclare
435 // capabilities, which can rehash the map and invalidate `seam`.
436 ScriptCapabilityBinder binder = seam->second.binder;
437 ScriptCapabilityUnbinder unbinder = seam->second.unbinder;
438
439 auto published = publish(std::move(name), instance);
440 if (!published) return Result<ScriptProviderHandle>::failure(published.status());
441 const ScriptProviderHandle handle = published.value();
442
443 // Record the seam on the entry *before* running the binder. A binder that fails
444 // halfway has usually already installed part of its adapter, and the rollback
445 // below removes it through this unbinder -- so the entry has to hold it by then
446 // (`publish` clears both fields). Measured before the fix: the rollback ran with
447 // an empty unbinder, so nothing was undone (plan doc §9.50).
448 {
449 auto pending = impl_->entries.find(handle.name);
450 if (pending == impl_->entries.end() || pending->second.generation != handle.generation) {
451 return Result<ScriptProviderHandle>::failure(providerDiagnostic(
452 DiagnosticCode::Failed, "the published provider vanished before its binder ran", handle.name));
453 }
454 pending->second.capability = std::string(capability);
455 pending->second.unbinder = std::move(unbinder);
456 }
457
458 auto bound = binder(handle);
459 if (!bound) {
460 // Roll back so a caller never observes a published provider that serves
461 // nothing. `unload` also runs the unbinder, which is what a partial bind
462 // needs; its own outcome is secondary to the binder's.
463 auto rolledBack = unload(handle);
464 rolledBack.ignore("a failed bind is already reported; the rollback only has to retire the generation");
465 return Result<ScriptProviderHandle>::failure(bound.status());
466 }
467
468 // Re-lookup after the binder: it is user code and may have re-published this same
469 // name, which invalidates the iterator above.
470 auto entry = impl_->entries.find(handle.name);
471 if (entry == impl_->entries.end() || entry->second.generation != handle.generation) {
472 // The binder re-published the same name; that generation owns its own bind.
473 return Result<ScriptProviderHandle>::failure(providerDiagnostic(
474 DiagnosticCode::Failed, "the capability binder replaced the provider it was binding", handle.name));
475 }
476 return Result<ScriptProviderHandle>::success(handle);
477}

479void ScriptProviderRegistry::clearAll() {
480 // Detach every adapter before invoking any unbinder: an unbinder is
481 // user-provided and may publish again, which would rehash the map under a live
482 // iterator. Declared capability seams survive, because they are composition-time
483 // declarations rather than published state.
484 std::vector<std::pair<std::string, ScriptProviderHandle>> retired;
485 std::vector<ScriptCapabilityUnbinder> unbinders;
486 for (auto& [name, entry] : impl_->entries) {
487 if (!entry.unbinder) continue;
488 retired.emplace_back(name, ScriptProviderHandle{name, entry.generation});
489 unbinders.push_back(std::move(entry.unbinder));
490 entry.unbinder = {};
491 entry.capability.clear();
492 }
493 impl_->entries.clear();
494
495 for (std::size_t i = 0; i < unbinders.size(); ++i) {
496 auto removed = unbinders[i](retired[i].second);
497 // Shutdown teardown: the adapter dies with the VM either way, so the only
498 // requirement is that the unbinder ran.
499 removed.ignore("clearAll cannot report per-adapter teardown failures");
500 }
501}
