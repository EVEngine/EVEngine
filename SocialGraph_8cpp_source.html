121bool SocialGraph::setRelation(const EntityId& source, const EntityId& target, const std::string& type, double value) {
122 if (!valid(source) || !valid(target) || type.empty()) return false;
123 const EdgeKey key{source, target, type};
124 const auto found = relations_.find(key);
125 if (found != relations_.end() && found->second == value) return false;
126 const double old = found == relations_.end() ? 0.0 : found->second;
127 relations_[key] = value;
128 relationTargets_[{source, type}][target] = true;

130 emit(found == relations_.end() ? "relation_added" : "relation_changed", source, target, type, old, value);
131 return true;
132}

188bool SocialGraph::removeEntity(const EntityId& entity) {
189 if (!valid(entity)) return false;
190 bool changed = setOwner(entity, {}) | setController(entity, {});
191
192 const auto owned = ownedBy(entity);
193 for (const auto& child : owned) changed = setOwner(child, {}) || changed;
194 const auto controlled = controlledBy(entity);
195 for (const auto& child : controlled) changed = setController(child, {}) || changed;
196
197 std::vector<std::tuple<EntityId, std::string, EntityId>> assignments;
198 for (const auto& [key, targets] : assignmentTargets_) {
199 for (const auto& [target, ignored] : targets) {
200 if (key.first == entity || target == entity) assignments.emplace_back(key.first, key.second, target);
201 }
202 }
203 for (const auto& [source, role, target] : assignments) changed = unassign(source, role, target) || changed;
204
205 std::vector<EdgeKey> relations;
206 for (const auto& [key, value] : relations_) {
207 if (std::get<0>(key) == entity || std::get<1>(key) == entity) relations.push_back(key);
208 }
209 for (const auto& [source, target, type] : relations) changed = removeRelation(source, target, type) || changed;
210
211 if (changed) emit("entity_removed", entity);
212 return changed;
213}
