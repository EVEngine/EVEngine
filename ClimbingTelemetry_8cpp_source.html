30ClimbingTelemetrySummary ClimbingTelemetryBuffer::summarize(ClimbingWorkload workload) const noexcept {
31 std::array<std::uint64_t, Capacity> durations{};
32 std::array<std::uint32_t, Capacity> queries{};
33 std::size_t count = 0;
34 std::uint32_t exceeded = 0;
35 std::uint32_t maximum = 0;
36 for (std::size_t index = 0; index < size_; ++index) {
37 const ClimbingTelemetrySample& sample = samples_[index];
38 if (sample.counters.workload != workload) continue;
39 durations[count] = sample.elapsedNanoseconds;
40 queries[count] = sample.counters.queryCount;
41 maximum = std::max(maximum, sample.counters.queryCount);
42 if (sample.counters.budgetState == ClimbingQueryBudgetState::Exceeded) ++exceeded;
43 ++count;
44 }
45 std::sort(durations.begin(), durations.begin() + static_cast<std::ptrdiff_t>(count));
46 std::sort(queries.begin(), queries.begin() + static_cast<std::ptrdiff_t>(count));
47
48 ClimbingTelemetrySummary summary;
49 summary.workload = workload;
50 summary.sampleCount = static_cast<std::uint32_t>(count);
51 summary.p50Nanoseconds = nearestRank(durations, count, 50);
52 summary.p95Nanoseconds = nearestRank(durations, count, 95);
53 summary.p50QueryCount = nearestRank(queries, count, 50);
54 summary.p95QueryCount = nearestRank(queries, count, 95);
55 summary.maxQueryCount = maximum;
56 summary.budgetExceededCount = exceeded;
57 return summary;
58}
