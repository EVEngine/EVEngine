28Result<std::vector<uint8_t>> quantize(std::span<const float> input, float scale, int zero, bool sign) {
29 if (!(scale > 0) || !std::isfinite(scale) || !validZero(zero, sign))
30 return fail<std::vector<uint8_t>>("Invalid affine scale or zero point");
31 std::vector<uint8_t> out;
32 out.reserve(input.size());
33 for (float x : input) {
34 if (!std::isfinite(x)) return fail<std::vector<uint8_t>>("Nonfinite quantization input");
35 const double rounded = evenRound(static_cast<double>(x) / scale) + zero;
36 const int q = static_cast<int>(std::clamp(rounded, sign ? -128.0 : 0.0, sign ? 127.0 : 255.0));
37 out.push_back(static_cast<uint8_t>(q));
38 }
39 return Result<std::vector<uint8_t>>::success(std::move(out));
40}

42Result<QuantizedActivation> dynamicQuantize(std::span<const float> input) {
43 float lo = 0, hi = 0;
44 for (float x : input) {
45 if (!std::isfinite(x)) return fail<QuantizedActivation>("Nonfinite dynamic quantization input");
46 lo = std::min(lo, x);
47 hi = std::max(hi, x);
48 }
49 QuantizedActivation result;
50 result.scale = hi == lo ? 1.f : static_cast<float>((static_cast<double>(hi) - lo) / 255.0);
51 if (!(result.scale > 0) || !std::isfinite(result.scale))
52 return fail<QuantizedActivation>("Dynamic quantization scale is not representable");
53 result.zeroPoint = static_cast<uint8_t>(std::clamp(evenRound(-static_cast<double>(lo) / result.scale), 0.0, 255.0));
54 auto bytes = quantize(input, result.scale, result.zeroPoint, false);
55 if (!bytes.ok()) return Result<QuantizedActivation>::failure(bytes.status());
56 result.values = std::move(bytes.value());
57 return Result<QuantizedActivation>::success(std::move(result));
58}

60Result<std::vector<float>> dequantize(ByteView input, std::span<const float> scales, std::span<const int32_t> zeros,
61 size_t inner) {
62 if (scales.empty() || scales.size() != zeros.size() || inner == 0 || !product(scales.size(), inner, SIZE_MAX) ||
63 input.bytes.size() % (scales.size() * inner))
64 return fail<std::vector<float>>("Invalid affine channel layout");
65 for (size_t i = 0; i < scales.size(); ++i)
66 if (!(scales[i] > 0) || !std::isfinite(scales[i]) || !validZero(zeros[i], input.signedValues))
67 return fail<std::vector<float>>("Invalid affine scale or zero point");
68 std::vector<float> out(input.bytes.size());
69 for (size_t i = 0; i < out.size(); ++i) {
70 const size_t channel = (i / inner) % scales.size();
71 out[i] = static_cast<float>(value(input, i) - zeros[channel]) * scales[channel];
72 }
73 return Result<std::vector<float>>::success(std::move(out));
74}

76Result<std::vector<int32_t>> matmul(ByteView a, ByteView b, size_t m, size_t k, size_t n, int aZero, int bZero,
77 OnnxCompute* compute) {
78 if (!validZero(aZero, a.signedValues) || !validZero(bZero, b.signedValues) || !product(m, k, SIZE_MAX) ||
79 !product(k, n, SIZE_MAX) || !product(m, n, INT32_MAX) || a.bytes.size() != m * k || b.bytes.size() != k * n)
80 return fail<std::vector<int32_t>>("Invalid integer matmul dimensions or zero points");
81 if (compute && m && n && k) {
82 try {
83 const int32_t z = bZero;
84 return Result<std::vector<int32_t>>::success(
85 onnx_detail::gpuMatmul(*compute, a, b, m, k, n, aZero, {&z, 1}));
86 } catch (const std::exception& e) {
87 return Result<std::vector<int32_t>>::failure(Diagnostic::error(DiagnosticCode::Failed, e.what()));
88 }
89 }
90 std::vector<int32_t> out(m * n);
91 for (size_t r = 0; r < m; ++r)
92 for (size_t c = 0; c < n; ++c) {
93 int64_t sum = 0;
94 for (size_t j = 0; j < k; ++j)
95 sum += static_cast<int64_t>(value(a, r * k + j) - aZero) * (value(b, j * n + c) - bZero);
96 if (!fits(sum)) return fail<std::vector<int32_t>>("Integer matmul accumulator overflow");
97 out[r * n + c] = static_cast<int32_t>(sum);
98 }
99 return Result<std::vector<int32_t>>::success(std::move(out));
100}

102Result<detail::ConvExtent> detail::validateConv(size_t xSize, size_t wSize, bool xSigned, bool wSigned,
103 const ConvShape& s, int xZero, std::span<const int32_t> wZeros) {
104 for (int d : {s.batch, s.channels, s.height, s.width, s.outputs, s.kernelH, s.kernelW, s.strideH, s.strideW,
105 s.dilationH, s.dilationW, s.groups})
106 if (d <= 0 || d > 65536) return fail<detail::ConvExtent>("Invalid integer convolution dimensions");
107 if (s.channels % s.groups || s.outputs % s.groups || s.padTop < 0 || s.padBottom < 0 || s.padLeft < 0 ||
108 s.padRight < 0 || !validZero(xZero, xSigned) ||
109 (wZeros.size() != 1 && wZeros.size() != static_cast<size_t>(s.outputs)))
110 return fail<detail::ConvExtent>("Invalid convolution groups, padding or zero points");
111 for (int z : wZeros)
112 if (!validZero(z, wSigned)) return fail<detail::ConvExtent>("Invalid convolution weight zero point");
113 const int64_t h = static_cast<int64_t>(s.height) + s.padTop + s.padBottom -
114 static_cast<int64_t>(s.dilationH) * (s.kernelH - 1) - 1;
115 const int64_t width = static_cast<int64_t>(s.width) + s.padLeft + s.padRight -
116 static_cast<int64_t>(s.dilationW) * (s.kernelW - 1) - 1;
117 if (h < 0 || width < 0) return fail<detail::ConvExtent>("Convolution kernel exceeds padded input");
118 const int64_t oh = h / s.strideH + 1, ow = width / s.strideW + 1;
119 int64_t outCount = 1;
120 for (int64_t d : {static_cast<int64_t>(s.batch), static_cast<int64_t>(s.outputs), oh, ow}) {
121 if (d > INT32_MAX / outCount) return fail<detail::ConvExtent>("Convolution output too large");
122 outCount *= d;
123 }
124 const int cg = s.channels / s.groups;
125 size_t inputCount = 1, weightCount = 1;
126 for (int d : {s.batch, s.channels, s.height, s.width}) {
127 if (!product(inputCount, d, INT32_MAX)) return fail<detail::ConvExtent>("Convolution input too large");
128 inputCount *= d;
129 }
130 for (int d : {s.outputs, cg, s.kernelH, s.kernelW}) {
131 if (!product(weightCount, d, INT32_MAX)) return fail<detail::ConvExtent>("Convolution weights too large");
132 weightCount *= d;
133 }
134 if (outCount > INT32_MAX || xSize != inputCount || wSize != weightCount)
135 return fail<detail::ConvExtent>("Convolution buffer size mismatch or output too large");
136 return Result<detail::ConvExtent>::success({oh, ow, outCount});
137}
138Result<std::vector<int32_t>> conv(ByteView x, ByteView w, const ConvShape& s, int xZero,
139 std::span<const int32_t> wZeros, OnnxCompute* compute) {
140 auto extent =
141 detail::validateConv(x.bytes.size(), w.bytes.size(), x.signedValues, w.signedValues, s, xZero, wZeros);
142 if (!extent.ok()) return Result<std::vector<int32_t>>::failure(extent.status());
143 const auto [oh, ow, outCount] = extent.value();
144 const int cg = s.channels / s.groups;
145 if (compute) {
146 try {
147 return Result<std::vector<int32_t>>::success(onnx_detail::gpuConv(*compute, x, w, s, xZero, wZeros));
148 } catch (const std::exception& e) {
149 return Result<std::vector<int32_t>>::failure(Diagnostic::error(DiagnosticCode::Failed, e.what()));
150 }
151 }
152 std::vector<int32_t> out(static_cast<size_t>(outCount));
153 for (int b = 0; b < s.batch; ++b)
154 for (int o = 0; o < s.outputs; ++o)
155 for (int64_t y = 0; y < oh; ++y)
156 for (int64_t z = 0; z < ow; ++z) {
157 int64_t sum = 0;
158 for (int c = 0; c < cg; ++c)
159 for (int ky = 0; ky < s.kernelH; ++ky)
160 for (int kx = 0; kx < s.kernelW; ++kx) {
161 const int64_t iy = y * s.strideH - s.padTop + static_cast<int64_t>(ky) * s.dilationH;
162 const int64_t ix = z * s.strideW - s.padLeft + static_cast<int64_t>(kx) * s.dilationW;
163 if (iy < 0 || ix < 0 || iy >= s.height || ix >= s.width) continue;
164 const size_t xi =
165 ((static_cast<size_t>(b) * s.channels + (o / (s.outputs / s.groups)) * cg + c) *
166 s.height +
167 iy) *
168 s.width +
169 ix;
170 const size_t wi = ((static_cast<size_t>(o) * cg + c) * s.kernelH + ky) * s.kernelW + kx;
171 sum += static_cast<int64_t>(value(x, xi) - xZero) *
172 (value(w, wi) - wZeros[wZeros.size() == 1 ? 0 : o]);
173 }
174 if (!fits(sum)) return fail<std::vector<int32_t>>("Integer convolution accumulator overflow");
175 out[((static_cast<size_t>(b) * s.outputs + o) * oh + y) * ow + z] = static_cast<int32_t>(sum);
176 }
177 return Result<std::vector<int32_t>>::success(std::move(out));
178}
