56eve::Result<ObservationState> parseObservation(const eve::Value::Object& object,
57 const eve::SnapshotEnvelope& snapshot) {
58 auto tick = readUint64(object, "tick");
59 if (!tick) return eve::Result<ObservationState>::failure(tick.status());
60 auto revision = readUint64(object, "revision");
61 if (!revision) return eve::Result<ObservationState>::failure(revision.status());
62 auto stepCount = readUint64(object, "stepCount");
63 if (!stepCount) return eve::Result<ObservationState>::failure(stepCount.status());
64 auto duration = readInt64(object, "simulatedDurationNs");
65 if (!duration) return eve::Result<ObservationState>::failure(duration.status());
66 auto lastDelta = readFloat(object, "lastDeltaSeconds");
67 if (!lastDelta) return eve::Result<ObservationState>::failure(lastDelta.status());
68 if (duration.value() < 0 || revision.value() != stepCount.value() || tick.value() != snapshot.tick.value() ||
69 stepCount.value() != snapshot.revision.value()) {
70 return eve::Result<ObservationState>::failure(eve::Diagnostic::error(
71 eve::DiagnosticCode::Conflict, "physics snapshot payload progress disagrees with its envelope",
72 "payload.observation"));
73 }
74
75 ObservationState result;
76 result.value.stepCount = stepCount.value();
77 result.value.lastTick = eve::SimulationTick(tick.value());
78 result.value.simulatedDuration = eve::Duration::fromNanoseconds(duration.value());
79 result.value.simulatedSeconds = result.value.simulatedDuration.seconds();
80 result.value.lastDeltaSeconds = lastDelta.value();
81 auto valid = detail::validateSimulationObservation(result.value, "physics.snapshot.observation");
82 if (!valid) return eve::Result<ObservationState>::failure(valid.status());
83 return eve::Result<ObservationState>::success(std::move(result));
84}

178 return eve::Result<Body3DState>::failure(eve::Diagnostic::error(
179 eve::DiagnosticCode::ParseError, "3D physics snapshot body id is invalid", "payload.bodies.id"));

183 return eve::Result<Body3DState>::failure(eve::Diagnostic::error(
184 eve::DiagnosticCode::InvalidArgument, "3D physics snapshot body type is invalid", "payload.bodies.type"));
185 state.type = std::move(type).takeValue();
186#define EV_READ_BODY3D_FLOAT(name) \
187 do { \
188 auto value = readFloat(*object, #name); \
189 if (!value) return eve::Result<Body3DState>::failure(value.status()); \
190 state.name = value.value(); \
191 } while (false)
192 EV_READ_BODY3D_FLOAT(x);
193 EV_READ_BODY3D_FLOAT(y);

204 EV_READ_BODY3D_FLOAT(wz);
205#undef EV_READ_BODY3D_FLOAT
206 const double quaternionLengthSquared =
207 static_cast<double>(state.qx) * state.qx + static_cast<double>(state.qy) * state.qy +
208 static_cast<double>(state.qz) * state.qz + static_cast<double>(state.qw) * state.qw;
209 if (!std::isfinite(quaternionLengthSquared) || std::abs(quaternionLengthSquared - 1.0) > 1e-4)
210 return eve::Result<Body3DState>::failure(
211 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
212 "3D physics snapshot rotation must be normalized", "payload.bodies.rotation"));
213#define EV_READ_BODY3D_BOOL(name) \
214 do { \
215 auto value = readBool(*object, #name); \
216 if (!value) return eve::Result<Body3DState>::failure(value.status()); \
217 state.name = value.value(); \
218 } while (false)
219 EV_READ_BODY3D_BOOL(active);
220 EV_READ_BODY3D_BOOL(bullet);
221 EV_READ_BODY3D_BOOL(awake);
222 EV_READ_BODY3D_BOOL(fixedRotation);

224 const auto* dynamics = field(*object, "dynamics")->getIf<Value::Object>();

226 !hasExactFields(*dynamics, {"linearDamping", "angularDamping", "gravityScale", "sleepThreshold", "allowSleep"}))
227 return Result<Body3DState>::failure(

229 for (auto [name, target] : {std::pair{"linearDamping", &state.linearDamping},

236 }

239 state.allowSleep = allowSleep.value();
240 if (state.linearDamping < 0 || state.angularDamping < 0 || state.sleepThreshold < 0)
241 return Result<Body3DState>::failure(
242 Diagnostic::error(DiagnosticCode::InvalidArgument, "3D damping and sleep threshold must be nonnegative"));
243 return eve::Result<Body3DState>::success(std::move(state));

298eve::Value fixtureTopologyValue(const Body& body) {
299 eve::Value::Array fixtures;
300 const b2Body* rawBody = body.raw();
301 for (const b2Fixture* fixture = rawBody ? rawBody->GetFixtureList() : nullptr; fixture;
302 fixture = fixture->GetNext()) {
303 eve::Value::Object object;
304 const b2Shape* shape = fixture->GetShape();
305 object.emplace("type", eve::Value(std::to_string(static_cast<int>(shape->GetType()))));
306 object.emplace("radius", eve::Value(shape->m_radius));
307 object.emplace("material", fixtureMaterialValue(*fixture));
308 eve::Value::Array geometry;
309 switch (shape->GetType()) {
310 case b2Shape::e_circle: {
311 const auto* circle = static_cast<const b2CircleShape*>(shape);
312 geometry.emplace_back(circle->m_p.x);
313 geometry.emplace_back(circle->m_p.y);
314 break;
315 }
316 case b2Shape::e_polygon: {
317 const auto* polygon = static_cast<const b2PolygonShape*>(shape);
318 for (int index = 0; index < polygon->m_count; ++index) {
319 geometry.emplace_back(polygon->m_vertices[index].x);
320 geometry.emplace_back(polygon->m_vertices[index].y);
321 }
322 break;
323 }
324 case b2Shape::e_chain: {
325 const auto* chain = static_cast<const b2ChainShape*>(shape);
326 for (int index = 0; index < chain->m_count; ++index) {
327 geometry.emplace_back(chain->m_vertices[index].x);
328 geometry.emplace_back(chain->m_vertices[index].y);
329 }
330 geometry.emplace_back(chain->m_hasPrevVertex);
331 geometry.emplace_back(chain->m_hasNextVertex);
332 break;
333 }
334 case b2Shape::e_edge: {
335 const auto* edge = static_cast<const b2EdgeShape*>(shape);
336 geometry.emplace_back(edge->m_vertex1.x);
337 geometry.emplace_back(edge->m_vertex1.y);
338 geometry.emplace_back(edge->m_vertex2.x);
339 geometry.emplace_back(edge->m_vertex2.y);
340 break;
341 }
342 default: break;
343 }
344 object.emplace("geometry", eve::Value(std::move(geometry)));
345 fixtures.emplace_back(std::move(object));
346 }
347 return eve::Value(std::move(fixtures));
348}

435 object.emplace("shapes", shapesTopologyValue(shapes));
436 object.emplace("joints", jointsTopologyValue(joints));
437 return eve::Value(std::move(object));
438}

472eve::Result<std::unique_ptr<World>> prepareWorld2D(const std::vector<Body2DState>& states, float gravityX,
473 float gravityY, float meter,
474 const SimulationObservation& observation,
475 eve::PersistentId instanceId) {
476 try {
477 auto prepared = std::make_unique<World>(gravityX, gravityY, false, meter, instanceId);
478 int maxId = 0;
479 for (const Body2DState& state : states) {
480 Body* body = prepared->newBody(state.type, state.x, state.y);
481 WorldSnapshotAccess::ownPreparedBody(*prepared, std::unique_ptr<Body>(body));
482 WorldSnapshotAccess::setBodyId(*body, state.id);
483 maxId = std::max(maxId, state.id);
484 const auto* fixtures = state.fixtures.getIf<eve::Value::Array>();
485 for (auto it = fixtures->rbegin(); it != fixtures->rend(); ++it) {
486 const auto* fixture = it->getIf<eve::Value::Object>();
487 if (!fixture || !hasExactFields(*fixture, {"geometry", "radius", "type", "material"}))
488 return eve::Result<std::unique_ptr<World>>::failure(
489 eve::Diagnostic::error(eve::DiagnosticCode::ParseError,
490 "2D fixture has unknown or missing fields", "payload.bodies.fixtures"));
491 auto type = readInt64(*fixture, "type");
492 if (!type) return Result<std::unique_ptr<World>>::failure(type.status());
493 auto radius = readFloat(*fixture, "radius");
494 if (!radius) return Result<std::unique_ptr<World>>::failure(radius.status());
495 const eve::Value* geometryValue = field(*fixture, "geometry");
496 const auto* geometry = geometryValue ? geometryValue->getIf<eve::Value::Array>() : nullptr;
497 if (!geometry)
498 return eve::Result<std::unique_ptr<World>>::failure(
499 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "2D fixture geometry is malformed",
500 "payload.bodies.fixtures.geometry"));
501 std::vector<float> values;
502 values.reserve(geometry->size());
503 for (const eve::Value& component : *geometry) {
504 if (!component.isNumeric()) {
505 if (component.isBool()) continue;
506 return eve::Result<std::unique_ptr<World>>::failure(eve::Diagnostic::error(
507 eve::DiagnosticCode::ParseError, "2D fixture coordinate must be numeric",
508 "payload.bodies.fixtures.geometry"));
509 }
510 values.push_back(
511 static_cast<float>(component.isDouble() ? component.asDouble() : component.asInt()) * meter);
512 }
513 Fixture* created = nullptr;
514 if (type.value() == b2Shape::e_circle && values.size() == 2)
515 created = body->newCircleFixture(radius.value() * meter);
516 else if (type.value() == b2Shape::e_polygon && values.size() >= 6)
517 created = body->newPolygonFixture(values);
518 else if (type.value() == b2Shape::e_chain && values.size() >= 4) {
519 const bool loop = geometry->size() >= 2 && (*geometry)[geometry->size() - 2].isBool() &&
520 (*geometry)[geometry->size() - 2].asBool() && geometry->back().isBool() &&
521 geometry->back().asBool();
522 // Box2D stores the closing vertex explicitly; the body factory closes loops itself.
523 if (loop && values.size() >= 8 && values[0] == values[values.size() - 2] &&
524 values[1] == values.back())
525 values.resize(values.size() - 2);
526 created = body->newChainFixture(values, loop);
527 }
528 if (!created)
529 return eve::Result<std::unique_ptr<World>>::failure(eve::Diagnostic::error(
530 eve::DiagnosticCode::Unsupported, "2D fixture shape cannot be reconstructed",
531 "payload.bodies.fixtures.type"));
532 WorldSnapshotAccess::ownPreparedFixture(*prepared, std::unique_ptr<Fixture>(created));
533 auto material = restoreFixtureMaterial(*created, *field(*fixture, "material"));
534 if (!material) return Result<std::unique_ptr<World>>::failure(material.status());
535 }
536 body->raw()->SetLinearDamping(state.linearDamping);
537 body->raw()->SetAngularDamping(state.angularDamping);
538 body->raw()->SetGravityScale(state.gravityScale);
539 body->raw()->SetSleepingAllowed(state.allowSleep);
540 body->setFixedRotation(state.fixedRotation);
541 body->setBullet(state.bullet);
542 body->setActive(state.active);
543 body->setAngle(state.angle);
544 body->setLinearVelocity(state.vx, state.vy);
545 body->setAngularVelocity(state.angularVelocity);
546 body->setAwake(state.awake);
547 }
548 auto restored = WorldSnapshotAccess::finishPrepared(*prepared, maxId + 1, observation);
549 if (!restored) return eve::Result<std::unique_ptr<World>>::failure(restored.status());
550 return eve::Result<std::unique_ptr<World>>::success(std::move(prepared));
551 } catch (const std::exception& error) {
552 return eve::Result<std::unique_ptr<World>>::failure(eve::Diagnostic::error(
553 eve::DiagnosticCode::Failed, std::string("2D topology preparation failed: ") + error.what(),
554 "physics.world.restore.prepare"));
555 }
556}

601eve::Result<std::unique_ptr<World3D>> prepareWorld3D(const std::vector<Body3DState>& states,
602 const eve::Value::Array& shapes, const eve::Value::Array& joints,
603 float gravityX, float gravityY, float gravityZ,
604 const SimulationObservation& observation,
605 eve::PersistentId instanceId, bool legacyParameters,
606 bool allowSleep) {
607 try {
608 auto prepared = WorldSnapshotAccess::makeDetachedWorld3D(gravityX, gravityY, gravityZ, instanceId, allowSleep);
609 std::unordered_map<int, Body3D*> bodyById;
610 int maxBodyId = 0;
611 for (const Body3DState& state : states) {
612 Body3D* body = prepared->newBody(state.type, state.x, state.y, state.z);
613 WorldSnapshotAccess::ownPreparedBody(*prepared, std::unique_ptr<Body3D>(body));
614 WorldSnapshotAccess::setBodyId(*body, state.id);
615 bodyById.emplace(state.id, body);
616 maxBodyId = std::max(maxBodyId, state.id);
617 body->setFixedRotation(state.fixedRotation);
618 body->setBullet(state.bullet);
619 body->setActive(state.active);
620 body->setRotation(state.qx, state.qy, state.qz, state.qw);
621 }
622 std::vector<std::pair<Shape3D*, float>> densities;
623 int maxShapeId = 0;
624 std::set<int> shapeIds;
625 for (const eve::Value& value : shapes) {
626 const auto* object = value.getIf<eve::Value::Object>();
627 if (!object)
628 return eve::Result<std::unique_ptr<World3D>>::failure(eve::Diagnostic::error(
629 eve::DiagnosticCode::ParseError, "3D shape must be an object", "payload.shapes"));
630 const std::initializer_list<std::string_view> sourceFields{"bodyId",
631 "id",
632 "kind",
633 "a",
634 "b",
635 "c",
636 "hullVertices",
637 "hullMaxVertices",
638 "meshVertices",
639 "meshIndices",
640 "meshWeldVertices",
641 "meshWeldTolerance",
642 "meshIdentifyEdges",
643 "meshUseMedianSplit",
644 "meshMaterialIndices",
645 "heightValues",
646 "heightCountX",
647 "heightCountZ",
648 "heightCellSizeX",
649 "heightCellSizeZ",
650 "heightGlobalMin",
651 "heightGlobalMax",
652 "heightClockwise"};
653 if (object->size() != sourceFields.size() + (legacyParameters ? 0 : 2))
654 return Result<std::unique_ptr<World3D>>::failure(Diagnostic::error(
655 DiagnosticCode::ParseError, "Shape has unknown or missing fields", "payload.shapes"));
656 for (auto name : sourceFields)
657 if (!field(*object, name))
658 return Result<std::unique_ptr<World3D>>::failure(Diagnostic::error(
659 DiagnosticCode::ParseError, "Shape source field is missing", std::string(name)));
660 auto bodyId = readInt64(*object, "bodyId");
661 if (!bodyId) return Result<std::unique_ptr<World3D>>::failure(bodyId.status());
662 auto id = readInt64(*object, "id");
663 if (!id) return Result<std::unique_ptr<World3D>>::failure(id.status());
664 auto kind = readString(*object, "kind");
665 if (!kind) return Result<std::unique_ptr<World3D>>::failure(kind.status());
666 auto a = readFloat(*object, "a");
667 if (!a) return Result<std::unique_ptr<World3D>>::failure(a.status());
668 auto b = readFloat(*object, "b");
669 if (!b) return Result<std::unique_ptr<World3D>>::failure(b.status());
670 auto c = readFloat(*object, "c");
671 if (!c) return Result<std::unique_ptr<World3D>>::failure(c.status());
672 if (bodyId.value() <= 0 || bodyId.value() >= std::numeric_limits<int>::max() || !id || id.value() <= 0 ||
673 id.value() >= std::numeric_limits<int>::max() ||
674 !shapeIds.insert(static_cast<int>(id.value())).second || !kind || !a || !b || !c ||
675 !bodyById.contains(static_cast<int>(bodyId.value())))
676 return eve::Result<std::unique_ptr<World3D>>::failure(eve::Diagnostic::error(
677 eve::DiagnosticCode::Conflict, "3D shape references invalid identity or source", "payload.shapes"));
678 Body3D* body = bodyById.at(static_cast<int>(bodyId.value()));
679 Shape3D* shape = nullptr;
680 if (kind.value() == "box")
681 shape = body->newBoxShape(a.value() * 2.f, b.value() * 2.f, c.value() * 2.f);
682 else if (kind.value() == "sphere")
683 shape = body->newSphereShape(a.value());
684 else if (kind.value() == "capsule")
685 shape = body->newCapsuleShape(a.value() * 2.f, b.value());
686 else if (kind.value() == "convexHull") {
687 auto vertices = readFloatArray(*object, "hullVertices");
688 if (!vertices) return Result<std::unique_ptr<World3D>>::failure(vertices.status());
689 auto maxVertices = readInt64(*object, "hullMaxVertices");
690 if (!maxVertices) return Result<std::unique_ptr<World3D>>::failure(maxVertices.status());
691 if (!vertices || !maxVertices)
692 return eve::Result<std::unique_ptr<World3D>>::failure(eve::Diagnostic::error(
693 eve::DiagnosticCode::ParseError, "convex hull source is invalid", "payload.shapes"));
694 shape = body->newConvexHullShape(vertices.value(), static_cast<int>(maxVertices.value()));
695 } else if (kind.value() == "triangleMesh") {
696 auto vertices = readFloatArray(*object, "meshVertices");
697 if (!vertices) return Result<std::unique_ptr<World3D>>::failure(vertices.status());
698 auto indices = readIntArray(*object, "meshIndices");
699 if (!indices) return Result<std::unique_ptr<World3D>>::failure(indices.status());
700 auto weld = readBool(*object, "meshWeldVertices");
701 if (!weld) return Result<std::unique_ptr<World3D>>::failure(weld.status());
702 auto tolerance = readFloat(*object, "meshWeldTolerance");
703 if (!tolerance) return Result<std::unique_ptr<World3D>>::failure(tolerance.status());
704 auto edges = readBool(*object, "meshIdentifyEdges");
705 if (!edges) return Result<std::unique_ptr<World3D>>::failure(edges.status());
706 auto median = readBool(*object, "meshUseMedianSplit");
707 if (!median) return Result<std::unique_ptr<World3D>>::failure(median.status());
708 if (!vertices || !indices || !weld || !tolerance || !edges || !median)
709 return eve::Result<std::unique_ptr<World3D>>::failure(eve::Diagnostic::error(
710 eve::DiagnosticCode::ParseError, "mesh source is invalid", "payload.shapes"));
711 shape = body->newTriangleMeshShape(vertices.value(), indices.value(), weld.value(), tolerance.value(),
712 edges.value(), median.value());
713 } else if (kind.value() == "heightField") {
714 auto heights = readFloatArray(*object, "heightValues");
715 if (!heights) return Result<std::unique_ptr<World3D>>::failure(heights.status());
716 auto cx = readInt64(*object, "heightCountX");
717 if (!cx) return Result<std::unique_ptr<World3D>>::failure(cx.status());
718 auto cz = readInt64(*object, "heightCountZ");
719 if (!cz) return Result<std::unique_ptr<World3D>>::failure(cz.status());
720 auto sx = readFloat(*object, "heightCellSizeX");
721 if (!sx) return Result<std::unique_ptr<World3D>>::failure(sx.status());
722 auto sz = readFloat(*object, "heightCellSizeZ");
723 if (!sz) return Result<std::unique_ptr<World3D>>::failure(sz.status());
724 auto mn = readFloat(*object, "heightGlobalMin");
725 if (!mn) return Result<std::unique_ptr<World3D>>::failure(mn.status());
726 auto mx = readFloat(*object, "heightGlobalMax");
727 if (!mx) return Result<std::unique_ptr<World3D>>::failure(mx.status());
728 auto clockwise = readBool(*object, "heightClockwise");
729 if (!clockwise) return Result<std::unique_ptr<World3D>>::failure(clockwise.status());
730 if (!heights || !cx || !cz || !sx || !sz || !mn || !mx || !clockwise)
731 return eve::Result<std::unique_ptr<World3D>>::failure(eve::Diagnostic::error(
732 eve::DiagnosticCode::ParseError, "height field source is invalid", "payload.shapes"));
733 shape =
734 body->newHeightFieldShape(static_cast<int>(cx.value()), static_cast<int>(cz.value()), sx.value(),
735 sz.value(), heights.value(), mn.value(), mx.value(), clockwise.value());
736 }
737 if (!shape)
738 return eve::Result<std::unique_ptr<World3D>>::failure(eve::Diagnostic::error(
739 eve::DiagnosticCode::Unsupported, "3D shape kind cannot be reconstructed", "payload.shapes.kind"));
740 WorldSnapshotAccess::ownPreparedShape(*prepared, std::unique_ptr<Shape3D>(shape));
741 WorldSnapshotAccess::setShapeId(*shape, static_cast<int>(id.value()));
742 if (legacyParameters) {
743 if (object->contains("localTransform"))
744 return Result<std::unique_ptr<World3D>>::failure(
745 Diagnostic::error(DiagnosticCode::ParseError, "Legacy shape has unexpected local transform"));
746 } else {
747 auto transform = readFloatArray(*object, "localTransform");
748 if (!transform) return Result<std::unique_ptr<World3D>>::failure(transform.status());
749 const auto& t = transform.value();
750 if (t.size() != 7 || !b3IsNormalizedQuat({{t[3], t[4], t[5]}, t[6]}))
751 return Result<std::unique_ptr<World3D>>::failure(Diagnostic::error(
752 DiagnosticCode::InvalidArgument, "Shape transform requires a position and unit quaternion"));
753 shape->setLocalTransform(t[0], t[1], t[2], t[3], t[4], t[5], t[6]);
754 auto contact = WorldSnapshotAccess::restoreShapeContact(*shape, *object);
755 if (!contact) return Result<std::unique_ptr<World3D>>::failure(contact.status());
756 densities.emplace_back(shape, contact.value());
757 }
758 maxShapeId = std::max(maxShapeId, static_cast<int>(id.value()));
759 }
760 for (auto [shape, density] : densities)
761 b3Shape_SetDensity(eve::physics::box3d::Access::shapeId(*shape), density, false);
762 auto mass = WorldSnapshotAccess::finishShapeMass(*prepared);
763 if (!mass) return Result<std::unique_ptr<World3D>>::failure(mass.status());
764 int maxJointId = 0;
765 std::set<int> jointIds;
766 for (const eve::Value& value : joints) {
767 const auto* object = value.getIf<eve::Value::Object>();
768 if (!object ||
769 !(legacyParameters
770 ? hasExactFields(*object, {"bodyAId", "bodyBId", "id", "kind", "localFrames"})
771 : hasExactFields(*object, {"bodyAId", "bodyBId", "id", "kind", "localFrames", "parameters"})))
772 return eve::Result<std::unique_ptr<World3D>>::failure(eve::Diagnostic::error(
773 eve::DiagnosticCode::ParseError, "3D joint has unknown or missing fields", "payload.joints"));
774 auto bodyAId = readInt64(*object, "bodyAId");
775 if (!bodyAId) return eve::Result<std::unique_ptr<World3D>>::failure(bodyAId.status());
776 auto bodyBId = readInt64(*object, "bodyBId");
777 if (!bodyBId) return eve::Result<std::unique_ptr<World3D>>::failure(bodyBId.status());
778 auto id = readInt64(*object, "id");
779 if (!id) return eve::Result<std::unique_ptr<World3D>>::failure(id.status());
780 auto kind = readString(*object, "kind");
781 if (!kind) return eve::Result<std::unique_ptr<World3D>>::failure(kind.status());
782 auto frames = readFloatArray(*object, "localFrames");
783 if (!frames) return eve::Result<std::unique_ptr<World3D>>::failure(frames.status());
784 if (id.value() <= 0 || id.value() >= std::numeric_limits<int>::max() || bodyAId.value() <= 0 ||
785 bodyAId.value() >= std::numeric_limits<int>::max() || bodyBId.value() <= 0 ||
786 bodyBId.value() >= std::numeric_limits<int>::max() || bodyAId.value() == bodyBId.value() ||
787 !jointIds.insert(static_cast<int>(id.value())).second || frames.value().size() != 14 ||
788 !bodyById.contains(static_cast<int>(bodyAId.value())) ||
789 !bodyById.contains(static_cast<int>(bodyBId.value())))
790 return eve::Result<std::unique_ptr<World3D>>::failure(eve::Diagnostic::error(
791 eve::DiagnosticCode::Conflict, "3D joint references invalid bodies or frames", "payload.joints"));
792 Body3D* aBody = bodyById.at(static_cast<int>(bodyAId.value()));
793 Body3D* bBody = bodyById.at(static_cast<int>(bodyBId.value()));
794 const auto& f = frames.value();
795 const b3Pos anchorA =
796 b3Body_GetWorldPoint(eve::physics::box3d::Access::bodyId(*aBody), b3Pos{f[0], f[1], f[2]});
797 const b3Pos anchorB =
798 b3Body_GetWorldPoint(eve::physics::box3d::Access::bodyId(*bBody), b3Pos{f[7], f[8], f[9]});
799 const b3Quat qa{{f[3], f[4], f[5]}, f[6]};
800 const b3Quat qb{{f[10], f[11], f[12]}, f[13]};
801 if (!b3IsNormalizedQuat(qa) || !b3IsNormalizedQuat(qb))
802 return eve::Result<std::unique_ptr<World3D>>::failure(
803 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
804 "Joint frame rotation must be normalized", "payload.joints.localFrames"));
805 Joint3D* joint = nullptr;
806 if (kind.value() == "distance") {
807 const b3Vec3 d = anchorB - anchorA;
808 const float length = std::sqrt(b3Dot(d, d));
809 joint = prepared->newDistanceJoint(aBody, bBody, anchorA.x, anchorA.y, anchorA.z, anchorB.x, anchorB.y,
810 anchorB.z, length);
811 } else if (kind.value() == "revolute") {
812 const b3Vec3 axis = b3Body_GetWorldVector(eve::physics::box3d::Access::bodyId(*aBody),
813 b3RotateVector(qa, b3Vec3_axisZ));
814 joint =
815 prepared->newRevoluteJoint(aBody, bBody, anchorA.x, anchorA.y, anchorA.z, axis.x, axis.y, axis.z);
816 } else if (kind.value() == "prismatic") {
817 const b3Vec3 axis = b3Body_GetWorldVector(eve::physics::box3d::Access::bodyId(*aBody),
818 b3RotateVector(qa, b3Vec3_axisX));
819 joint =
820 prepared->newPrismaticJoint(aBody, bBody, anchorA.x, anchorA.y, anchorA.z, axis.x, axis.y, axis.z);
821 } else if (kind.value() == "spherical") {
822 const b3Vec3 axis = b3Body_GetWorldVector(eve::physics::box3d::Access::bodyId(*aBody),
823 b3RotateVector(qa, b3Vec3_axisZ));
824 joint =
825 prepared->newSphericalJoint(aBody, bBody, anchorA.x, anchorA.y, anchorA.z, axis.x, axis.y, axis.z);
826 } else if (kind.value() == "wheel") {
827 const b3Vec3 suspension = b3Body_GetWorldVector(eve::physics::box3d::Access::bodyId(*aBody),
828 b3RotateVector(qa, b3Vec3_axisX));
829 const b3Vec3 wheel = b3Body_GetWorldVector(eve::physics::box3d::Access::bodyId(*bBody),
830 b3RotateVector(qb, b3Vec3_axisZ));
831 joint = prepared->newWheelJoint(aBody, bBody, anchorA.x, anchorA.y, anchorA.z, suspension.x,
832 suspension.y, suspension.z, wheel.x, wheel.y, wheel.z);
833 } else if (kind.value() == "weld") {
834 joint = prepared->newWeldJoint(aBody, bBody, anchorA.x, anchorA.y, anchorA.z);
835 } else if (kind.value() == "motor") {
836 joint = prepared->newMotorJoint(aBody, bBody);
837 } else if (kind.value() == "parallel") {
838 const b3Vec3 axis =
839 b3Body_GetWorldVector(box3d::Access::bodyId(*aBody), b3RotateVector(qa, b3Vec3_axisZ));
840 joint = prepared->newParallelJoint(aBody, bBody, axis.x, axis.y, axis.z);
841 } else if (kind.value() == "filter") {
842 joint = prepared->newFilterJoint(aBody, bBody);
843 }
844 if (!joint)
845 return eve::Result<std::unique_ptr<World3D>>::failure(eve::Diagnostic::error(
846 eve::DiagnosticCode::Unsupported, "3D joint kind cannot be reconstructed", "payload.joints.kind"));
