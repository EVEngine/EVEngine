26void wheelMove2D(VehicleEntity& v, eve::physics::Body* b, float dt) {
27 const VehicleDefinition* def = v.definition()->def;
28 if (def == nullptr) return;
29 auto in = v.input();
30 auto mo = v.motion();
31 mo->x = b->getX();
32 mo->y = b->getY();
33
34 const float speedFactor = std::clamp(std::fabs(mo->speed) / def->maxSpeed, 0.f, 1.f);
35 mo->heading = normalizeDeg(mo->heading + in->steer * def->turnRate * (0.35f + 0.65f * speedFactor) * dt);
36
37 float target = in->throttle * def->maxSpeed;
38 if (in->brake > 0.f || in->handbrake) target = 0.f;
39 const float dv = target - mo->speed;
40 const float maxDv = def->accel * dt;
41 mo->speed += std::clamp(dv, -maxDv, maxDv);
42
43 const float rad = mo->heading * kPi / 180.f;
44 b->setAngle(rad);
45 b->setLinearVelocity(std::cos(rad) * mo->speed, std::sin(rad) * mo->speed);
46}

70void suspensionMove3D(VehicleEntity& v, eve::physics::Body3D* b, float dt) {
71 const VehicleDefinition* def = v.definition()->def;
72 if (def == nullptr) return;
73 eve::physics::World3D* world = b->getWorld();
74 if (world == nullptr) {
75 if (IVehicleMobility* kinematic = VehicleSystem::findMobility("kinematic")) kinematic->update(v, dt);
76 return;
77 }
78
79 auto in = v.input();
80 auto mo = v.motion();
81 mo->x = b->getX();
82 mo->y = b->getZ();
83
84 const float qx = b->getRotX();
85 const float qy = b->getRotY();
86 const float qz = b->getRotZ();
87 const float qw = b->getRotW();
88 const float yaw = std::atan2(2.f * (qw * qy + qx * qz), 1.f - 2.f * (qy * qy + qz * qz));
89 mo->heading = normalizeDeg(yaw * 180.f / kPi);
90 const float yawRad = mo->heading * kPi / 180.f;
91 const float fx = std::sin(yawRad);
92 const float fz = std::cos(yawRad);
93
94 const auto& wheels = def->suspension.wheels;
95 auto sus = v.suspension();
96 if (sus->wheels.size() != wheels.size()) sus->wheels.resize(wheels.size());
97 const uint64_t chassisMask = ~uint64_t{2};
98
99 for (size_t i = 0; i < wheels.size(); ++i) {
100 const SuspensionWheel& w = wheels[i];
101 const float wx = mo->x + w.x * std::cos(yawRad) + w.z * std::sin(yawRad);
102 const float wz = mo->y - w.x * std::sin(yawRad) + w.z * std::cos(yawRad);
103 const float mountY = b->getY() + w.y + w.restLength;
104 const float rayLen = w.restLength + w.radius + def->suspension.maxTravel;
105 const int hit = world->rayCastFiltered(wx, mountY, wz, wx, mountY - rayLen, wz, chassisMask);
106
107 float compression = 0.f;
108 if (hit >= 0) {
109 const float hitDist = mountY - world->getRayHitY();
110 compression = std::clamp(w.restLength - (hitDist - w.radius), 0.f, def->suspension.maxTravel);
111 }
112
113 auto& ws = sus->wheels[i];
114 const float vel = (compression - ws.prevCompression) / std::max(dt, 1e-4f);
115 const float force = w.stiffness * compression + w.damping * vel;
116 if (force > 0.f) b->applyForceAt(0.f, force, 0.f, wx, mountY, wz);
117 ws.prevCompression = compression;
118 ws.grounded = hit >= 0;
119 }
120
121 const float drive = (in->brake > 0.f || in->handbrake) ? 0.f : in->throttle;
122 const float targetSpeed = drive * def->maxSpeed;
123 const float fwd =
124 std::fabs(targetSpeed) > 0.01f ? (b->getLinearVelocityX() * fx + b->getLinearVelocityZ() * fz) : 0.f;
125 float driveForce = def->suspension.driveForce * drive;
126 if (std::fabs(targetSpeed) > 0.01f) {
127 driveForce *= std::clamp(1.f - fwd / targetSpeed, 0.f, 1.f);
128 }
129 b->applyForce(fx * driveForce, 0.f, fz * driveForce);
130 b->setAngularVelocity(0.f, in->steer * def->turnRate * kPi / 180.f, 0.f);
131
132 const float vx = b->getLinearVelocityX();
133 const float vy = b->getLinearVelocityY();
134 const float vz = b->getLinearVelocityZ();
135 const float rx = fz;
136 const float rz = -fx;
137 const float lat = vx * rx + vz * rz;
138 const float grip = std::max(0.f, 1.f - def->suspension.lateralGrip * dt);
139 b->setLinearVelocity(fx * fwd + rx * lat * grip, vy, fz * fwd + rz * lat * grip);
140 mo->speed = fwd;
141}

159VehiclePhysicsStatus VehiclePhysics::attach2D(VehicleEntity* v, eve::physics::World* world) {
160 if (v == nullptr || world == nullptr) return VehiclePhysicsStatus::Unavailable;
161 (void)detach(v);

186 const float rad = mo->heading * kPi / 180.f;

196VehiclePhysicsStatus VehiclePhysics::detach(VehicleEntity* v) {
197 if (v == nullptr) return VehiclePhysicsStatus::Unavailable;
198 auto pb = v->physicsBody();
199 bool had = false;
200 if (pb->body2d != nullptr) {
201 pb->body2d->destroy();
202 had = true;
203 }
204 if (pb->body3d != nullptr) {
205 pb->body3d->destroy();
206 had = true;
207 }
208 pb->body2d = nullptr;
209 pb->body3d = nullptr;
210 pb->space.clear();
211 return had ? VehiclePhysicsStatus::Applied : VehiclePhysicsStatus::Unavailable;
212}
