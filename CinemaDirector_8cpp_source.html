37Result<void> CinemaDirector::defineOverview(const std::string& name, float eyeX, float eyeY, float eyeZ,
38 float targetX, float targetY, float targetZ, float fov) {
39 if (name.empty()) return fail("overview name is required", "camera.cinema.defineOverview");
40 const glm::vec3 eye(eyeX, eyeY, eyeZ);
41 const glm::vec3 target(targetX, targetY, targetZ);
42 if (!finite3(eye) || !finite3(target) || !std::isfinite(fov))
43 return fail("overview pose must be finite", "camera.cinema.defineOverview");
44 if (fov <= 0.f || fov >= 180.f) return fail("overview fov must be in (0,180)", "camera.cinema.defineOverview");
45 if (glm::length(target - eye) < 1e-5f)
46 return fail("overview eye and target must differ", "camera.cinema.defineOverview");
47 for (auto& shot : overviews_) {
48 if (shot.name == name) {
49 shot.eye = eye;
50 shot.target = target;
51 shot.fov = fov;
52 return Result<void>::success();
53 }
54 }
55 overviews_.push_back(OverviewShot{name, eye, target, fov});
56 return Result<void>::success();
57}

83Result<void> CinemaDirector::writeActivePose(const glm::vec3& eye, const glm::vec3& target, float fov) {
84 if (stack_.empty()) return fail("no active overview to write", "camera.cinema.writePose");
85 const std::string& name = stack_.back().overview;
86 auto written = controller_->setView(name, eye.x, eye.y, eye.z, target.x, target.y, target.z, fov);
87 if (!written) return written;
88 // Keep the active cinematic index on this view without restarting a blend.
89 if (!controller_->switchTo(name, 0.f))
90 return fail("active overview view is missing", "camera.cinema.writePose");
91 // Film moves author the pose each frame; snap so damping does not lag the track.
92 controller_->snap();
93 return Result<void>::success();
94}

111 auto bound = requireBound();

134Result<void> CinemaDirector::beginMove(MoveKind kind, float duration, float param) {
135 auto bound = requireBound();
136 if (!bound) return bound;
137 if (!isActive()) return fail("push an overview before playing a film move", "camera.cinema.move");
138 if (!std::isfinite(duration) || duration <= 0.f)
139 return fail("move duration must be finite and > 0", "camera.cinema.move");
140 if (!std::isfinite(param)) return fail("move parameter must be finite", "camera.cinema.move");
141
142 glm::vec3 eye, target;
143 float fov = 60.f;
144 auto sampled = sampleLivePose(eye, target, fov);
145 if (!sampled) return sampled;
146
147 MoveState next;
148 next.active = true;
149 next.kind = kind;
150 next.duration = duration;
151 next.param = param;
152 next.fromEye = eye;
153 next.fromTarget = target;
154 next.fromFov = fov;
155 next.toTarget = target;
156 next.toFov = fov;
157
158 const glm::vec3 forward = target - eye;
159 const float length = glm::length(forward);
160 if (length < 1e-5f) return fail("camera look direction is degenerate", "camera.cinema.move");
161 const glm::vec3 dir = forward / length;
162
163 if (kind == MoveKind::Dolly) {
164 next.toEye = eye + dir * param;
165 if (glm::length(target - next.toEye) < 1e-4f)
166 return fail("dolly would collapse eye onto target", "camera.cinema.dolly");
167 } else if (kind == MoveKind::Crane) {
168 next.toEye = eye + glm::vec3(0.f, param, 0.f);
169 } else if (kind == MoveKind::OrbitReveal) {
170 next.orbitCenter = target;
171 const glm::vec3 rel = eye - target;
172 next.orbitRadius = glm::length(glm::vec2(rel.x, rel.z));
173 if (next.orbitRadius < 1e-4f) return fail("orbit reveal needs horizontal radius", "camera.cinema.orbitReveal");
174 next.orbitElevation = rel.y;
175 next.orbitAzimuth0 = std::atan2(rel.x, rel.z);
176 const float endAz = next.orbitAzimuth0 + glm::radians(param);
177 next.toEye =
178 target + glm::vec3(std::sin(endAz) * next.orbitRadius, next.orbitElevation, std::cos(endAz) * next.orbitRadius);
179 } else {
180 return fail("unsupported move kind", "camera.cinema.move");
181 }
182
183 move_ = next;
184 return Result<void>::success();
185}

199Result<void> CinemaDirector::pop(float blendTime) {
200 auto bound = requireBound();
201 if (!bound) return bound;
202 if (stack_.empty()) return fail("story camera stack is empty", "camera.cinema.pop");
203 if (!std::isfinite(blendTime) || blendTime < 0.f)
204 return fail("blendTime must be finite and >= 0", "camera.cinema.pop");
205
206 cancelMove();
207 const StackEntry top = stack_.back();
208 stack_.pop_back();
209
210 if (stack_.empty()) {
211 auto restored = controller_->restoreFraming(top.gameplay, blendTime);
212 if (!restored) {
213 stack_.push_back(top);
214 return restored;
215 }
216 return Result<void>::success();
217 }
218
219 const OverviewShot* previous = findOverview(stack_.back().overview);
220 if (!previous) {
221 stack_.push_back(top);
222 return fail("previous overview is missing", "camera.cinema.pop");
223 }
224 return applyOverviewPose(*previous, blendTime);
225}

227Result<void> CinemaDirector::update(float dt) {
228 auto bound = requireBound();
229 if (!bound) return bound;
230 if (!std::isfinite(dt) || dt < 0.f) return fail("dt must be finite and >= 0", "camera.cinema.update");
231 if (!move_.active) return Result<void>::success();
232
233 move_.age += dt;
234 const float t = smoothstep01(move_.duration > 0.f ? move_.age / move_.duration : 1.f);
235 glm::vec3 eye = glm::mix(move_.fromEye, move_.toEye, t);
236 glm::vec3 target = glm::mix(move_.fromTarget, move_.toTarget, t);
237 float fov = glm::mix(move_.fromFov, move_.toFov, t);
238
239 if (move_.kind == MoveKind::OrbitReveal) {
240 const float az = move_.orbitAzimuth0 + glm::radians(move_.param) * t;
241 eye = move_.orbitCenter +
242 glm::vec3(std::sin(az) * move_.orbitRadius, move_.orbitElevation, std::cos(az) * move_.orbitRadius);
243 target = move_.orbitCenter;
244 fov = move_.fromFov;
245 }
246
247 auto written = writeActivePose(eye, target, fov);
248 if (!written) return written;
249
250 if (move_.age >= move_.duration) {
251 auto finalWrite = writeActivePose(move_.kind == MoveKind::OrbitReveal ? eye : move_.toEye,
252 move_.kind == MoveKind::OrbitReveal ? target : move_.toTarget,
253 move_.kind == MoveKind::OrbitReveal ? fov : move_.toFov);
254 cancelMove();
255 return finalWrite;
256 }
257 return Result<void>::success();
258}
