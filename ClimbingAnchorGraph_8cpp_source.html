214eve::Result<void> validateClimbingAnchorGraphDefinition(const ClimbingAnchorGraphDefinition& graph) {
215 if (graph.id.empty() || graph.sourceGeometryContentId.empty() || graph.buildSettingsHash.empty())
216 return failure<void>(eve::DiagnosticCode::InvalidArgument,
217 "graph id, source geometry content id, and build settings hash are required",
218 "graph.identity");
219 if (graph.nodes.empty() || graph.nodes.size() > maxGraphNodes || graph.edges.size() > maxGraphEdges)
220 return failure<void>(eve::DiagnosticCode::InvalidArgument,
221 "anchor graph node and edge counts exceed supported bounds", "graph.topology");
222 std::unordered_set<std::string> nodeIds;
223 for (std::size_t index = 0; index < graph.nodes.size(); ++index) {
224 const auto& node = graph.nodes[index];
225 const std::string path = "nodes." + std::to_string(index);
226 if (node.id.empty() || !nodeIds.emplace(node.id).second)
227 return failure<void>(eve::DiagnosticCode::AlreadyExists, "anchor node ids must be non-empty and unique",
228 path + ".id");
229 if (!finite(node.localPosition) || !finite(node.localNormal) || !finite(node.localTangent) ||
230 !finite(node.leftHandSocket) || !finite(node.rightHandSocket) || !finite(node.feetSocket))
231 return failure<void>(eve::DiagnosticCode::InvalidArgument, "anchor frame and sockets must be finite", path);
232 const float normalLength = lengthSquared(node.localNormal);
233 const float tangentLength = lengthSquared(node.localTangent);
234 if (std::fabs(normalLength - 1.f) > frameTolerance || std::fabs(tangentLength - 1.f) > frameTolerance ||
235 std::fabs(dot(node.localNormal, node.localTangent)) > frameTolerance)
236 return failure<void>(eve::DiagnosticCode::InvalidArgument,
237 "anchor normal and tangent must form an orthonormal local frame", path + ".frame");
238 if (node.occupancySlots == 0 || node.occupancySlots > maxOccupancySlots)
239 return failure<void>(eve::DiagnosticCode::InvalidArgument,
240 "anchor occupancy slots must be between one and eight", path + ".occupancySlots");
241 if (!validUniqueStrings(node.tags))
242 return failure<void>(eve::DiagnosticCode::InvalidArgument,
243 "anchor tags must be non-empty and unique", path + ".tags");
244 }
245 std::vector<ClimbingAnchorEdgeDefinition> edges = graph.edges;
246 std::sort(edges.begin(), edges.end(), [](const auto& lhs, const auto& rhs) {
247 if (lhs.from != rhs.from) return lhs.from < rhs.from;
248 if (lhs.to != rhs.to) return lhs.to < rhs.to;
249 return lhs.kind < rhs.kind;
250 });
251 for (std::size_t index = 0; index < edges.size(); ++index) {
252 const auto& edge = edges[index];
253 const std::string path = "edges." + std::to_string(index);
254 if (edge.from.empty() || edge.to.empty() || edge.from == edge.to || !nodeIds.contains(edge.from) ||
255 !nodeIds.contains(edge.to))
256 return failure<void>(eve::DiagnosticCode::InvalidArgument,
257 "anchor edge endpoints must name two distinct existing nodes", path);
258 if (index != 0 && sameEdgeIdentity(edges[index - 1], edge))
259 return failure<void>(eve::DiagnosticCode::AlreadyExists,
260 "anchor edge identity must be unique", path);
261 if (!validUniqueStrings(edge.requiredTags))
262 return failure<void>(eve::DiagnosticCode::InvalidArgument,
263 "edge required tags must be non-empty and unique", path + ".requiredTags");
264 }
265 return eve::Result<void>::success();
266}

268eve::Result<eve::Value> encodeClimbingAnchorGraphDefinition(const ClimbingAnchorGraphDefinition& graph) {
269 auto valid = validateClimbingAnchorGraphDefinition(graph);
270 if (!valid) return eve::Result<eve::Value>::failure(valid.status());
271 ClimbingAnchorGraphDefinition canonical = graph;
272 canonicalize(canonical);
273 eve::Value::Array nodes;
274 nodes.reserve(canonical.nodes.size());
275 for (const auto& node : canonical.nodes) {
276 eve::Value::Object object = node.extensionMetadata;
277 object["id"] = eve::Value(node.id);
278 object["kind"] = eve::Value(std::string(nodeKindName(node.kind)));
279 object["localPosition"] = vecValue(node.localPosition);
280 object["localNormal"] = vecValue(node.localNormal);
281 object["localTangent"] = vecValue(node.localTangent);
282 object["leftHandSocket"] = vecValue(node.leftHandSocket);
283 object["rightHandSocket"] = vecValue(node.rightHandSocket);
284 object["feetSocket"] = vecValue(node.feetSocket);
285 object["occupancySlots"] = eve::Value(static_cast<std::int64_t>(node.occupancySlots));
286 object["tags"] = eve::Value(stringArray(node.tags));
287 nodes.emplace_back(std::move(object));
288 }
289 eve::Value::Array edges;
290 edges.reserve(canonical.edges.size());
291 for (const auto& edge : canonical.edges) {
292 eve::Value::Object object = edge.extensionMetadata;
293 object["from"] = eve::Value(edge.from);
294 object["to"] = eve::Value(edge.to);
295 object["kind"] = eve::Value(std::string(edgeKindName(edge.kind)));
296 object["bidirectional"] = eve::Value(edge.bidirectional);
297 object["requiredTags"] = eve::Value(stringArray(edge.requiredTags));
298 edges.emplace_back(std::move(object));
299 }
300 eve::Value::Object root = canonical.extensionMetadata;
301 root["schemaId"] = eve::Value(std::string(ClimbingAnchorGraphDefinition::SchemaId));
302 root["schemaVersion"] = eve::Value(ClimbingAnchorGraphDefinition::SchemaVersion);
303 root["id"] = eve::Value(canonical.id);
304 root["sourceGeometryContentId"] = eve::Value(canonical.sourceGeometryContentId);
305 root["buildSettingsHash"] = eve::Value(canonical.buildSettingsHash);
306 root["nodes"] = eve::Value(std::move(nodes));
307 root["edges"] = eve::Value(std::move(edges));
308 return eve::Result<eve::Value>::success(eve::Value(std::move(root)));
309}

311eve::Result<ClimbingAnchorGraphDefinition> decodeClimbingAnchorGraphDefinition(const eve::Value& value) {
312 const auto* root = value.getIf<eve::Value::Object>();
313 if (!root)
314 return failure<ClimbingAnchorGraphDefinition>(eve::DiagnosticCode::ParseError,
315 "anchor graph definition must be an object");
316 ClimbingAnchorGraphDefinition graph;
317 std::string schemaId;
318 std::int64_t version = -1;
319 const eve::Value* nodesValue = field(*root, "nodes");
320 const auto* nodes = nodesValue ? nodesValue->getIf<eve::Value::Array>() : nullptr;
321 const eve::Value* edgesValue = field(*root, "edges");
322 const auto* edges = edgesValue ? edgesValue->getIf<eve::Value::Array>() : nullptr;
323 if (!readString(*root, "schemaId", schemaId) || schemaId != ClimbingAnchorGraphDefinition::SchemaId ||
324 !readInt64(*root, "schemaVersion", version) ||
325 (version != ClimbingAnchorGraphDefinition::SchemaVersion &&
326 version != ClimbingAnchorGraphDefinition::SchemaVersion - 1) || !readString(*root, "id", graph.id) ||
327 !readString(*root, "sourceGeometryContentId", graph.sourceGeometryContentId) ||
328 !readString(*root, "buildSettingsHash", graph.buildSettingsHash) || !nodes || !edges) {
329 if (version > ClimbingAnchorGraphDefinition::SchemaVersion)
330 return failure<ClimbingAnchorGraphDefinition>(eve::DiagnosticCode::UnknownVersion,
331 "anchor graph schema version is unsupported",
332 "schemaVersion");
333 return failure<ClimbingAnchorGraphDefinition>(eve::DiagnosticCode::ParseError,
334 "anchor graph envelope has missing or invalid known fields");
335 }
336 graph.nodes.reserve(nodes->size());
337 for (std::size_t index = 0; index < nodes->size(); ++index) {
338 const auto* object = (*nodes)[index].getIf<eve::Value::Object>();
339 ClimbingAnchorNodeDefinition node;
340 std::string kind;
341 std::int64_t occupancySlots = 0;
342 if (!object || !readString(*object, "id", node.id) || !readString(*object, "kind", kind) ||
343 !readNodeKind(kind, node.kind) || !readVec(*object, "localPosition", node.localPosition) ||
344 !readVec(*object, "localNormal", node.localNormal) ||
345 !readVec(*object, "localTangent", node.localTangent) ||
346 !readVec(*object, "leftHandSocket", node.leftHandSocket) ||
347 !readVec(*object, "rightHandSocket", node.rightHandSocket) ||
348 !readVec(*object, "feetSocket", node.feetSocket) ||
349 !readInt64(*object, "occupancySlots", occupancySlots) || occupancySlots < 0 ||
350 occupancySlots > std::numeric_limits<std::uint32_t>::max() ||
351 !readStringArray(*object, "tags", node.tags))
352 return failure<ClimbingAnchorGraphDefinition>(eve::DiagnosticCode::ParseError,
353 "anchor node has missing or invalid known fields",
354 "nodes." + std::to_string(index));
355 node.occupancySlots = static_cast<std::uint32_t>(occupancySlots);
356 node.extensionMetadata = unknownFields(*object, {"id", "kind", "localPosition", "localNormal",
357 "localTangent", "leftHandSocket", "rightHandSocket",
358 "feetSocket", "occupancySlots", "tags"});
359 graph.nodes.push_back(std::move(node));
360 }
361 graph.edges.reserve(edges->size());
362 for (std::size_t index = 0; index < edges->size(); ++index) {
363 const auto* object = (*edges)[index].getIf<eve::Value::Object>();
364 ClimbingAnchorEdgeDefinition edge;
365 std::string kind;
366 if (!object || !readString(*object, "from", edge.from) || !readString(*object, "to", edge.to) ||
367 !readString(*object, "kind", kind) || !readEdgeKind(kind, edge.kind) ||
368 !readBool(*object, "bidirectional", edge.bidirectional) ||
369 !readStringArray(*object, "requiredTags", edge.requiredTags))
370 return failure<ClimbingAnchorGraphDefinition>(eve::DiagnosticCode::ParseError,
371 "anchor edge has missing or invalid known fields",
372 "edges." + std::to_string(index));
373 edge.extensionMetadata = unknownFields(*object, {"from", "to", "kind", "bidirectional", "requiredTags"});
374 graph.edges.push_back(std::move(edge));
375 }
376 graph.extensionMetadata = unknownFields(*root, {"schemaId", "schemaVersion", "id",
377 "sourceGeometryContentId", "buildSettingsHash", "nodes", "edges"});
378 auto valid = validateClimbingAnchorGraphDefinition(graph);
379 if (!valid) return eve::Result<ClimbingAnchorGraphDefinition>::failure(valid.status());
380 canonicalize(graph);
381 return eve::Result<ClimbingAnchorGraphDefinition>::success(std::move(graph));
382}

422eve::Result<ResolvedClimbingAnchorNode> ClimbingAnchorGraphInstance::resolveNodeKinematics(
423 physics::World3D& world, const ClimbingAnchorNodeRef& reference) const {
424 if (reference.graphId != graph_.id)
425 return failure<ResolvedClimbingAnchorNode>(eve::DiagnosticCode::Conflict,
426 "anchor node reference belongs to another graph",
427 "reference.graphId");
428 if (reference.graphGeneration != generation_)
429 return failure<ResolvedClimbingAnchorNode>(eve::DiagnosticCode::StaleHandle,
430 "anchor node reference belongs to a stale graph generation",
431 "reference.graphGeneration");
432 const auto* node = findNode(reference.nodeId);
433 if (!node)
434 return failure<ResolvedClimbingAnchorNode>(eve::DiagnosticCode::NotFound,
435 "anchor node was not found", "reference.nodeId");
436 if (world.runtimeHandle() != world_)
437 return failure<ResolvedClimbingAnchorNode>(eve::DiagnosticCode::StaleHandle,
438 "anchor graph belongs to another or stale Physics world", "world");
439 physics::Body3D* body = world.findBody(body_);
440 if (!body)
441 return failure<ResolvedClimbingAnchorNode>(eve::DiagnosticCode::StaleHandle,
442 "anchor graph target body handle is stale", "body");
443 ResolvedClimbingAnchorNode result;
444 result.reference = reference;
445 result.kind = node->kind;
446 result.body = body_;
447 auto position = body->localToWorldPointOwned(node->localPosition.x, node->localPosition.y,
448 node->localPosition.z);
449 if (!position) return eve::Result<ResolvedClimbingAnchorNode>::failure(position.status());
450 auto normal = body->localToWorldVectorOwned(node->localNormal.x, node->localNormal.y,
451 node->localNormal.z);
452 if (!normal) return eve::Result<ResolvedClimbingAnchorNode>::failure(normal.status());
453 auto tangent = body->localToWorldVectorOwned(node->localTangent.x, node->localTangent.y,
454 node->localTangent.z);
455 if (!tangent) return eve::Result<ResolvedClimbingAnchorNode>::failure(tangent.status());
456 auto leftHand = body->localToWorldPointOwned(node->leftHandSocket.x, node->leftHandSocket.y,
457 node->leftHandSocket.z);
458 if (!leftHand) return eve::Result<ResolvedClimbingAnchorNode>::failure(leftHand.status());
459 auto rightHand = body->localToWorldPointOwned(node->rightHandSocket.x, node->rightHandSocket.y,
460 node->rightHandSocket.z);
461 if (!rightHand) return eve::Result<ResolvedClimbingAnchorNode>::failure(rightHand.status());
462 auto feet = body->localToWorldPointOwned(node->feetSocket.x, node->feetSocket.y, node->feetSocket.z);
463 if (!feet) return eve::Result<ResolvedClimbingAnchorNode>::failure(feet.status());
464 auto velocity = body->getLocalPointVelocityOwned(node->localPosition.x, node->localPosition.y,
465 node->localPosition.z);
466 if (!velocity) return eve::Result<ResolvedClimbingAnchorNode>::failure(velocity.status());

478 const ClimbingAnchorNodeRef& reference) const {
479 if (reference.graphId != graph_.id)

496 result.push_back(std::move(reverse));
497 }
498 }
499 std::sort(result.begin(), result.end(), [](const auto& lhs, const auto& rhs) {
500 if (lhs.kind != rhs.kind) return lhs.kind < rhs.kind;
501 return lhs.to < rhs.to;
502 });
503 return eve::Result<std::vector<ClimbingAnchorEdgeDefinition>>::success(std::move(result));

506eve::Result<ClimbingAnchorRoute> ClimbingAnchorGraphInstance::planRoute(
507 const ClimbingAnchorRouteRequest& request) const {
508 if (request.start.graphId != graph_.id || request.goal.graphId != graph_.id)
509 return failure<ClimbingAnchorRoute>(eve::DiagnosticCode::Conflict,
510 "route endpoints belong to another anchor graph", "request");
511 if (request.start.graphGeneration != generation_ || request.goal.graphGeneration != generation_)
512 return failure<ClimbingAnchorRoute>(eve::DiagnosticCode::StaleHandle,
513 "route endpoints belong to a stale graph generation", "request");
514 if (!findNode(request.start.nodeId) || !findNode(request.goal.nodeId))
515 return failure<ClimbingAnchorRoute>(eve::DiagnosticCode::NotFound,
516 "route endpoint was not found", "request");
517 if (request.maxVisitedNodes == 0 || request.maxVisitedNodes > 65536)
518 return failure<ClimbingAnchorRoute>(eve::DiagnosticCode::InvalidArgument,
519 "route maxVisitedNodes must be between one and 65536",
520 "request.maxVisitedNodes");
521 const bool requesterHasAgent = !request.requester.agentId.isZero();
522 const bool requesterHasExecution = !request.requester.executionId.isZero();
523 if (requesterHasAgent != requesterHasExecution)
524 return failure<ClimbingAnchorRoute>(eve::DiagnosticCode::InvalidArgument,
525 "route requester must provide both agent and execution ids",
526 "request.requester");
527
528 std::vector<ClimbingAnchorEdgeKind> allowed = request.allowedEdgeKinds;
529 std::sort(allowed.begin(), allowed.end());
530 allowed.erase(std::unique(allowed.begin(), allowed.end()), allowed.end());
531 if (!validUniqueStrings(request.availableTags))
532 return failure<ClimbingAnchorRoute>(eve::DiagnosticCode::InvalidArgument,
533 "route capability tags must be non-empty and unique",
534 "request.availableTags");
535 std::vector<std::string> availableTags = request.availableTags;
536 std::sort(availableTags.begin(), availableTags.end());
537 const auto permits = [&allowed](ClimbingAnchorEdgeKind kind) {
538 return allowed.empty() || std::binary_search(allowed.begin(), allowed.end(), kind);
539 };
540 const auto isFull = [this, &request](std::string_view nodeId) {
541 if (request.occupancyPolicy == ClimbingRouteOccupancyPolicy::Ignore) return false;
542 const auto* node = findNode(nodeId);
543 if (!node) return true;
544 std::uint32_t occupied = 0;
545 for (const auto& [id, record] : reservations_) {
546 (void)id;
547 if (record.nodeId != nodeId) continue;
548 if (record.occupant == request.requester) return false;
549 ++occupied;
550 }
551 return occupied >= node->occupancySlots;
552 };
553
554 struct Previous {
555 std::string from;
556 ClimbingAnchorEdgeKind kind = ClimbingAnchorEdgeKind::Shimmy;
557 };
558 std::deque<std::string> queue;
559 std::unordered_map<std::string, Previous> previous;
560 previous.emplace(request.start.nodeId, Previous{});
561 queue.push_back(request.start.nodeId);
562 bool found = request.start.nodeId == request.goal.nodeId;
563 while (!queue.empty() && !found) {
564 if (previous.size() > request.maxVisitedNodes)
565 return failure<ClimbingAnchorRoute>(eve::DiagnosticCode::PreconditionViolation,
566 "route planning exceeded maxVisitedNodes",
567 "request.maxVisitedNodes");
568 const std::string current = std::move(queue.front());
569 queue.pop_front();
570 auto reference = nodeRef(current);
571 if (!reference) return eve::Result<ClimbingAnchorRoute>::failure(reference.status());
572 auto outgoing = edgesFrom(reference.value());
573 if (!outgoing) return eve::Result<ClimbingAnchorRoute>::failure(outgoing.status());
574 for (const auto& edge : outgoing.value()) {
575 const bool hasRequiredTags = std::all_of(
576 edge.requiredTags.begin(), edge.requiredTags.end(), [&availableTags](const std::string& tag) {
577 return std::binary_search(availableTags.begin(), availableTags.end(), tag);
578 });
579 if (!permits(edge.kind) || !hasRequiredTags || previous.contains(edge.to) || isFull(edge.to))
580 continue;
581 if (previous.size() >= request.maxVisitedNodes)
582 return failure<ClimbingAnchorRoute>(eve::DiagnosticCode::PreconditionViolation,
583 "route planning exceeded maxVisitedNodes",
584 "request.maxVisitedNodes");
585 previous.emplace(edge.to, Previous{current, edge.kind});
586 if (edge.to == request.goal.nodeId) {
587 found = true;
588 break;
589 }
590 queue.push_back(edge.to);
591 }
592 }
593 if (!found)
594 return failure<ClimbingAnchorRoute>(eve::DiagnosticCode::NotFound,
595 "no permitted anchor route reaches the goal", "request.goal");
596
597 std::vector<std::string> reversedNodes;
598 std::vector<ClimbingAnchorEdgeKind> reversedKinds;
599 for (std::string cursor = request.goal.nodeId;;) {
600 reversedNodes.push_back(cursor);
601 if (cursor == request.start.nodeId) break;
602 const auto foundPrevious = previous.find(cursor);
603 if (foundPrevious == previous.end())
604 return failure<ClimbingAnchorRoute>(eve::DiagnosticCode::InvariantViolation,
605 "route predecessor chain is incomplete", "route");
606 reversedKinds.push_back(foundPrevious->second.kind);
607 cursor = foundPrevious->second.from;
608 }
609 std::reverse(reversedNodes.begin(), reversedNodes.end());
610 std::reverse(reversedKinds.begin(), reversedKinds.end());
611
612 ClimbingAnchorRoute route;
613 route.graphId = graph_.id;
614 route.graphGeneration = generation_;
615 route.nodes.reserve(reversedNodes.size());
616 for (const auto& nodeId : reversedNodes) route.nodes.push_back({graph_.id, nodeId, generation_});
617 route.steps.reserve(reversedKinds.size());
618 for (std::size_t index = 0; index < reversedKinds.size(); ++index)
619 route.steps.push_back({route.nodes[index], route.nodes[index + 1], reversedKinds[index]});
620 return eve::Result<ClimbingAnchorRoute>::success(std::move(route));
621}

624 const ClimbingAnchorNodeRef& reference, ClimbingAnchorOccupant occupant) {
625 if (reference.graphId != graph_.id)
626 return failure<ClimbingAnchorReservation>(eve::DiagnosticCode::Conflict,
627 "anchor node reference belongs to another graph",
628 "reference.graphId");
629 if (reference.graphGeneration != generation_)
630 return failure<ClimbingAnchorReservation>(eve::DiagnosticCode::StaleHandle,
631 "anchor node reference belongs to a stale graph generation",
632 "reference.graphGeneration");
633 const auto* node = findNode(reference.nodeId);
634 if (!node)
635 return failure<ClimbingAnchorReservation>(eve::DiagnosticCode::NotFound,
636 "anchor node was not found", "reference.nodeId");
637 if (occupant.agentId.isZero() || occupant.executionId.isZero())
638 return failure<ClimbingAnchorReservation>(eve::DiagnosticCode::InvalidArgument,
639 "anchor reservation requires non-zero agent and execution ids",
640 "occupant");
641 std::vector<bool> occupied(node->occupancySlots, false);
642 for (const auto& [id, record] : reservations_) {
643 (void)id;
644 if (record.nodeId != node->id) continue;
645 if (record.occupant == occupant)
646 return failure<ClimbingAnchorReservation>(eve::DiagnosticCode::Conflict,
647 "occupant already owns a slot on this anchor", "occupant");
648 if (record.slot < occupied.size()) occupied[record.slot] = true;
649 }
650 const auto free = std::find(occupied.begin(), occupied.end(), false);
651 if (free == occupied.end())
652 return failure<ClimbingAnchorReservation>(eve::DiagnosticCode::Conflict,
653 "all anchor occupancy slots are reserved", "reference.nodeId");
654 const auto next = nextReservationId_.incremented();
655 if (!next)
656 return failure<ClimbingAnchorReservation>(eve::DiagnosticCode::PreconditionViolation,
657 "anchor reservation identity is exhausted", "reservationId");
658 const ClimbingAnchorReservationId id = nextReservationId_;
659 const std::uint32_t slot = static_cast<std::uint32_t>(std::distance(occupied.begin(), free));
660 reservations_.emplace(id, ReservationRecord{node->id, slot, occupant, 1});
661 nextReservationId_ = *next;
662 return eve::Result<ClimbingAnchorReservation>::success(
663 {id, generation_, 1, node->id, slot, occupant}, eve::Status::success(eve::StatusCode::Applied));
664}
