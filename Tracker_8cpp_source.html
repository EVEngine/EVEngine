44eve::Result<std::string> Tracker::snapshotJson() const {
45 eve::Value::Array encodedEntries;
46 encodedEntries.reserve(order.size());
47 for (const auto &id : order) {
48 const auto found = entries.find(id);
49 if (found == entries.end())
50 return eve::Result<std::string>::failure(eve::Diagnostic::error(
51 eve::DiagnosticCode::InvariantViolation, "quest tracker order references a missing entry", id));
52 eve::Value::Array objectives;
53 objectives.reserve(found->second.objectives.size());
54 for (const auto &objective : found->second.objectives) {
55 eve::Value::Object encodedObjective;
56 encodedObjective.emplace("current", eve::Value(objective.current));
57 encodedObjective.emplace("id", eve::Value(objective.id));
58 objectives.emplace_back(std::move(encodedObjective));
59 }
60 eve::Value::Object encodedEntry;
61 encodedEntry.emplace("id", eve::Value(found->second.id));
62 encodedEntry.emplace("objectives", eve::Value(std::move(objectives)));
63 encodedEntry.emplace("state", eve::Value(found->second.state));
64 encodedEntries.emplace_back(std::move(encodedEntry));
65 }
66 eve::Value::Object root;
67 root.emplace("entries", eve::Value(std::move(encodedEntries)));
68 root.emplace("schema", eve::Value("eve.rpg.quest-tracker"));
69 root.emplace("version", eve::Value(1));
70 return eve::Value(std::move(root)).toJson();
71}

73eve::Result<void> Tracker::restoreSnapshotJson(std::string_view json) {
74 auto parsed = eve::Value::fromJson(json);
75 if (!parsed.ok()) return eve::Result<void>::failure(parsed.status());
76 const eve::Value &root = parsed.value();
77 auto fail = [](eve::DiagnosticCode code, std::string message, std::string path) {
78 return eve::Result<void>::failure(eve::Diagnostic::error(code, std::move(message), std::move(path)));
79 };
80 if (!root.isObject())
81 return fail(eve::DiagnosticCode::ParseError, "quest tracker root must be an object", "$");
82 const eve::Value *schema = root.find("schema");
83 if (!schema || !schema->isString() || schema->asString() != "eve.rpg.quest-tracker")
84 return fail(eve::DiagnosticCode::InvalidArgument, "snapshot does not belong to RPG Tracker", "$.schema");
85 const eve::Value *version = root.find("version");
86 if (!version || !version->isInt64() || version->asInt() != 1)
87 return fail(eve::DiagnosticCode::UnknownVersion, "unsupported RPG Tracker snapshot version", "$.version");
88 const eve::Value *encodedEntries = root.find("entries");
89 if (!encodedEntries || !encodedEntries->isArray())
90 return fail(eve::DiagnosticCode::ParseError, "quest tracker entries must be an array", "$.entries");
91
92 Tracker candidate;
93 if (encodedEntries->arraySize() != candidate.order.size())
94 return fail(eve::DiagnosticCode::Conflict,
95 "quest tracker snapshot does not match the current quest registry", "$.entries");
96 std::unordered_set<std::string> seenEntries;
97 for (std::size_t index = 0; index < candidate.order.size(); ++index) {
98 const eve::Value &encoded = encodedEntries->at(index);
99 const std::string path = "$.entries[" + std::to_string(index) + "]";
100 if (!encoded.isObject())
101 return fail(eve::DiagnosticCode::ParseError, "quest tracker entry must be an object", path);
102 const eve::Value *id = encoded.find("id");
103 const eve::Value *state = encoded.find("state");
104 const eve::Value *objectives = encoded.find("objectives");
105 if (!id || !id->isString() || !state || !state->isString() || !objectives || !objectives->isArray())
106 return fail(eve::DiagnosticCode::ParseError, "quest tracker entry fields have invalid types", path);
107 if (!seenEntries.emplace(id->asString()).second)
108 return fail(eve::DiagnosticCode::Conflict, "quest tracker entry id is duplicated", path + ".id");
109 if (id->asString() != candidate.order[index])
110 return fail(eve::DiagnosticCode::Conflict,
111 "quest tracker entry order or id does not match the current registry", path + ".id");
112 auto entryIt = candidate.entries.find(id->asString());
113 if (entryIt == candidate.entries.end())
114 return fail(eve::DiagnosticCode::NotFound, "quest definition is not registered", path + ".id");
115 const std::string &stateName = state->asString();
116 if (stateName != "locked" && stateName != "inactive" && stateName != "active" &&
117 stateName != "ready" && stateName != "completed" && stateName != "failed")
118 return fail(eve::DiagnosticCode::InvalidArgument, "quest tracker state is invalid", path + ".state");
119 if (objectives->arraySize() != entryIt->second.objectives.size())
120 return fail(eve::DiagnosticCode::Conflict,
121 "quest objective list does not match the current definition", path + ".objectives");
122 bool allDone = true;
123 for (std::size_t objectiveIndex = 0; objectiveIndex < entryIt->second.objectives.size(); ++objectiveIndex) {
124 const eve::Value &encodedObjective = objectives->at(objectiveIndex);
125 const std::string objectivePath = path + ".objectives[" + std::to_string(objectiveIndex) + "]";
126 if (!encodedObjective.isObject())
127 return fail(eve::DiagnosticCode::ParseError, "quest objective must be an object", objectivePath);
128 const eve::Value *objectiveId = encodedObjective.find("id");
129 const eve::Value *current = encodedObjective.find("current");
130 auto &runtime = entryIt->second.objectives[objectiveIndex];
131 if (!objectiveId || !objectiveId->isString() || !current || !current->isInt64())
132 return fail(eve::DiagnosticCode::ParseError, "quest objective fields have invalid types", objectivePath);
133 if (objectiveId->asString() != runtime.id)
134 return fail(eve::DiagnosticCode::Conflict,
135 "quest objective id or order does not match the current definition", objectivePath + ".id");
136 if (current->asInt() < 0 || current->asInt() > runtime.count)
137 return fail(eve::DiagnosticCode::InvalidArgument,
138 "quest objective progress is outside its definition bounds", objectivePath + ".current");
139 runtime.current = static_cast<int>(current->asInt());
140 runtime.done = runtime.current >= runtime.count;
141 allDone = allDone && runtime.done;
142 }
143 if ((stateName == "ready" || stateName == "completed") && !allDone)
144 return fail(eve::DiagnosticCode::InvariantViolation,
145 "ready or completed quest must have every objective complete", path + ".state");
146 if ((stateName == "locked" || stateName == "inactive") &&
147 std::any_of(entryIt->second.objectives.begin(), entryIt->second.objectives.end(),
148 [](const ObjectiveRuntime &objective) { return objective.current != 0; }))
149 return fail(eve::DiagnosticCode::InvariantViolation,
150 "locked or inactive quest cannot contain progress", path + ".state");
151 if (allDone && stateName != "ready" && stateName != "completed" && stateName != "failed")
152 return fail(eve::DiagnosticCode::InvariantViolation,
153 "fully progressed quest must be ready, completed, or failed", path + ".state");
154 entryIt->second.state = stateName;
155 }
156 candidate.pending.clear();
157 candidate.polled.clear();
158 entries.swap(candidate.entries);
159 order.swap(candidate.order);
160 pending.swap(candidate.pending);
161 polled.swap(candidate.polled);
162 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
163}
