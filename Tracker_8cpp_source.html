51eve::Result<std::string> Tracker::snapshotJson() const {
52 eve::Value::Array encodedEntries;
53 encodedEntries.reserve(order.size());
54 for (const auto &id : order) {
55 const auto found = entries.find(id);
56 if (found == entries.end())
57 return eve::Result<std::string>::failure(eve::Diagnostic::error(
58 eve::DiagnosticCode::InvariantViolation, "quest tracker order references a missing entry", id));
59 eve::Value::Array objectives;
60 objectives.reserve(found->second.objectives.size());
61 for (const auto &objective : found->second.objectives) {
62 eve::Value::Object encodedObjective;
63 encodedObjective.emplace("current", eve::Value(objective.current));
64 encodedObjective.emplace("id", eve::Value(objective.id));
65 objectives.emplace_back(std::move(encodedObjective));
66 }
67 eve::Value::Object encodedEntry;
68 encodedEntry.emplace("deadlineAt", eve::Value(found->second.deadlineAt));
69 encodedEntry.emplace("id", eve::Value(found->second.id));
70 encodedEntry.emplace("objectives", eve::Value(std::move(objectives)));
71 encodedEntry.emplace("refreshStamp", eve::Value(found->second.refreshStamp));
72 encodedEntry.emplace("state", eve::Value(found->second.state));
73 encodedEntries.emplace_back(std::move(encodedEntry));
74 }
75 eve::Value::Object root;
76 root.emplace("entries", eve::Value(std::move(encodedEntries)));
77 root.emplace("schema", eve::Value("eve.rpg.quest-tracker"));
78 root.emplace("version", eve::Value(1));
79 return eve::Value(std::move(root)).toJson();
80}

82eve::Result<void> Tracker::restoreSnapshotJson(std::string_view json) {
83 auto parsed = eve::Value::fromJson(json);
84 if (!parsed.ok()) return eve::Result<void>::failure(parsed.status());
85 const eve::Value &root = parsed.value();
86 auto fail = [](eve::DiagnosticCode code, std::string message, std::string path) {
87 return eve::Result<void>::failure(eve::Diagnostic::error(code, std::move(message), std::move(path)));
88 };
89 if (!root.isObject())
90 return fail(eve::DiagnosticCode::ParseError, "quest tracker root must be an object", "$");
91 const eve::Value *schema = root.find("schema");
92 if (!schema || !schema->isString() || schema->asString() != "eve.rpg.quest-tracker")
93 return fail(eve::DiagnosticCode::InvalidArgument, "snapshot does not belong to RPG Tracker", "$.schema");
94 const eve::Value *version = root.find("version");
95 if (!version || !version->isInt64() || version->asInt() != 1)
96 return fail(eve::DiagnosticCode::UnknownVersion, "unsupported RPG Tracker snapshot version", "$.version");
97 const eve::Value *encodedEntries = root.find("entries");
98 if (!encodedEntries || !encodedEntries->isArray())
99 return fail(eve::DiagnosticCode::ParseError, "quest tracker entries must be an array", "$.entries");
100
101 Tracker candidate;
102 if (encodedEntries->arraySize() != candidate.order.size())
103 return fail(eve::DiagnosticCode::Conflict,
104 "quest tracker snapshot does not match the current quest registry", "$.entries");
105 std::unordered_set<std::string> seenEntries;
106 for (std::size_t index = 0; index < candidate.order.size(); ++index) {
107 const eve::Value &encoded = encodedEntries->at(index);
108 const std::string path = "$.entries[" + std::to_string(index) + "]";
109 if (!encoded.isObject())
110 return fail(eve::DiagnosticCode::ParseError, "quest tracker entry must be an object", path);
111 const eve::Value *id = encoded.find("id");
112 const eve::Value *state = encoded.find("state");
113 const eve::Value *objectives = encoded.find("objectives");
114 if (!id || !id->isString() || !state || !state->isString() || !objectives || !objectives->isArray())
115 return fail(eve::DiagnosticCode::ParseError, "quest tracker entry fields have invalid types", path);
116 if (!seenEntries.emplace(id->asString()).second)
117 return fail(eve::DiagnosticCode::Conflict, "quest tracker entry id is duplicated", path + ".id");
118 if (id->asString() != candidate.order[index])
119 return fail(eve::DiagnosticCode::Conflict,
120 "quest tracker entry order or id does not match the current registry", path + ".id");
121 auto entryIt = candidate.entries.find(id->asString());
122 if (entryIt == candidate.entries.end())
123 return fail(eve::DiagnosticCode::NotFound, "quest definition is not registered", path + ".id");
124 const std::string &stateName = state->asString();
125 if (stateName != "locked" && stateName != "inactive" && stateName != "active" &&
126 stateName != "ready" && stateName != "completed" && stateName != "failed")
127 return fail(eve::DiagnosticCode::InvalidArgument, "quest tracker state is invalid", path + ".state");
128 if (objectives->arraySize() != entryIt->second.objectives.size())
129 return fail(eve::DiagnosticCode::Conflict, "quest objective list does not match the current definition",
130 path + ".objectives");
131 for (std::size_t objectiveIndex = 0; objectiveIndex < entryIt->second.objectives.size(); ++objectiveIndex) {
132 const eve::Value &encodedObjective = objectives->at(objectiveIndex);
133 const std::string objectivePath = path + ".objectives[" + std::to_string(objectiveIndex) + "]";
134 if (!encodedObjective.isObject())
135 return fail(eve::DiagnosticCode::ParseError, "quest objective must be an object", objectivePath);
136 const eve::Value *objectiveId = encodedObjective.find("id");
137 const eve::Value *current = encodedObjective.find("current");
138 auto &runtime = entryIt->second.objectives[objectiveIndex];
139 if (!objectiveId || !objectiveId->isString() || !current || !current->isInt64())
140 return fail(eve::DiagnosticCode::ParseError, "quest objective fields have invalid types", objectivePath);
141 if (objectiveId->asString() != runtime.id)
142 return fail(eve::DiagnosticCode::Conflict,
143 "quest objective id or order does not match the current definition", objectivePath + ".id");
144 if (current->asInt() < 0 || current->asInt() > runtime.count)
145 return fail(eve::DiagnosticCode::InvalidArgument,
146 "quest objective progress is outside its definition bounds", objectivePath + ".current");
147 runtime.current = static_cast<int>(current->asInt());
148 runtime.done = runtime.current >= runtime.count;
149 }
150 const eve::Value *deadlineAt = encoded.find("deadlineAt");
151 if (deadlineAt) {
152 if (!deadlineAt->isNumeric() || !std::isfinite(deadlineAt->asDouble()) || deadlineAt->asDouble() < 0.0)
153 return fail(eve::DiagnosticCode::InvalidArgument,
154 "quest deadlineAt must be a finite non-negative number", path + ".deadlineAt");
155 entryIt->second.deadlineAt = deadlineAt->asDouble();
156 } else {
157 entryIt->second.deadlineAt = 0.0;
158 }
159 const eve::Value *refreshStamp = encoded.find("refreshStamp");
160 if (refreshStamp) {
161 if (!refreshStamp->isString())
162 return fail(eve::DiagnosticCode::ParseError, "quest refreshStamp must be a string",
163 path + ".refreshStamp");
164 entryIt->second.refreshStamp = refreshStamp->asString();
165 } else {
166 entryIt->second.refreshStamp.clear();
167 }
168 entryIt->second.state = stateName;
169 const bool satisfied = QuestSystem::isObjectivesSatisfied(candidate, id->asString());
170 if ((stateName == "ready" || stateName == "completed") && !satisfied)
171 return fail(eve::DiagnosticCode::InvariantViolation,
172 "ready or completed quest must satisfy its objective policy", path + ".state");
173 if ((stateName == "locked" || stateName == "inactive") &&
174 std::any_of(entryIt->second.objectives.begin(), entryIt->second.objectives.end(),
175 [](const ObjectiveRuntime &objective) { return objective.current != 0; }))
176 return fail(eve::DiagnosticCode::InvariantViolation,
177 "locked or inactive quest cannot contain progress", path + ".state");
178 if (satisfied && stateName != "ready" && stateName != "completed" && stateName != "failed")
179 return fail(eve::DiagnosticCode::InvariantViolation,
180 "objective-satisfied quest must be ready, completed, or failed", path + ".state");
181 if ((stateName == "locked" || stateName == "inactive" || stateName == "completed" || stateName == "failed") &&
182 entryIt->second.deadlineAt != 0.0)
183 return fail(eve::DiagnosticCode::InvariantViolation,
184 "non-listening quest states cannot carry an armed deadline", path + ".deadlineAt");
185 }
186 candidate.pending.clear();
187 candidate.polled.clear();
188 entries.swap(candidate.entries);
189 order.swap(candidate.order);
190 pending.swap(candidate.pending);
191 polled.swap(candidate.polled);
192 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
193}
