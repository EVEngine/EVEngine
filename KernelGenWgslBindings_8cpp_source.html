20void specializeSource(std::string &source, const std::vector<int> &representatives, int inputCount) {
21 std::vector<std::string> names(static_cast<size_t>(inputCount));
22 std::map<std::string, std::string> aliases;
23 for (int i = 0; i < inputCount; ++i) {
24 const std::string prefix = "@group(0) @binding(" + std::to_string(i) + ") var<storage, read_write> ";
25 const auto begin = source.find(prefix);
26 if (begin == std::string::npos) throw eve::Exception("Tensor WGSL: missing input declaration");
27 const auto nameBegin = begin + prefix.size();
28 const auto nameEnd = source.find(':', nameBegin);
29 names[static_cast<size_t>(i)] = source.substr(nameBegin, nameEnd - nameBegin);
30 if (representatives[static_cast<size_t>(i)] == i) continue;
31 aliases.emplace(names[static_cast<size_t>(i)],
32 names[static_cast<size_t>(representatives[static_cast<size_t>(i)])]);
33 source.erase(begin, source.find('\n', nameEnd) + 1 - begin);
34 }
35 if (aliases.empty()) return;
36 std::string renamed;
37 for (size_t i = 0; i < source.size();) {
38 const unsigned char ch = static_cast<unsigned char>(source[i]);
39 if (std::isalpha(ch) || ch == '_') {
40 const auto begin = i++;
41 while (i < source.size() && (std::isalnum(static_cast<unsigned char>(source[i])) || source[i] == '_')) ++i;
42 const std::string word = source.substr(begin, i - begin);
43 const auto found = aliases.find(word);
44 renamed += found == aliases.end() ? word : found->second;
45 } else {
46 renamed += source[i++];
47 }
48 }
49 source = std::move(renamed);
50}
51} // namespace
