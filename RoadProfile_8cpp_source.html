15 for (float v : values) {

50Result<RoadProfile> makeRoadProfile(const RoadStyle& style, int lanesForward, int lanesBackward) {
51 auto ok = validateStyle(style);
52 if (!ok.ok()) return Result<RoadProfile>::failure(ok.status());
53 if (lanesForward < 1 || lanesForward > 8)
54 return Result<RoadProfile>::failure(
55 Diagnostic::error(DiagnosticCode::InvalidArgument, "lanesForward must be in [1,8]", "lanesForward"));
56 if (lanesBackward < 0 || lanesBackward > 8)
57 return Result<RoadProfile>::failure(
58 Diagnostic::error(DiagnosticCode::InvalidArgument, "lanesBackward must be in [0,8]", "lanesBackward"));
59
60 const float asphaltHalf =
61 0.5f * style.laneWidth * static_cast<float>(lanesForward + lanesBackward);
62 const float barrierInner = asphaltHalf;
63 const float barrierOuter = asphaltHalf + style.curbWidth;
64 const float sidewalkOuter = barrierOuter + style.sidewalkWidth;
65 const float barrierTop = style.curbHeight;
66 const float walkH = style.sidewalkHeight;
67
68 RoadProfile profile;
69 profile.halfWidth = sidewalkOuter;
70
71 // Jersey-barrier U-channel matching the reference interchange look:
72 // sidewalk → tall barrier outer/top/inner → asphalt deck → mirror.
73 pushPoint(profile, -sidewalkOuter, walkH, RoadMaterial::Sidewalk);
74 pushPoint(profile, -barrierOuter, walkH, RoadMaterial::Sidewalk);
75 pushPoint(profile, -barrierOuter, barrierTop, RoadMaterial::Curb);
76 pushPoint(profile, -barrierInner, barrierTop, RoadMaterial::Curb);
77 pushPoint(profile, -barrierInner, 0.f, RoadMaterial::Curb);
78 pushPoint(profile, barrierInner, 0.f, RoadMaterial::Asphalt);
79 pushPoint(profile, barrierInner, barrierTop, RoadMaterial::Curb);
80 pushPoint(profile, barrierOuter, barrierTop, RoadMaterial::Curb);
81 pushPoint(profile, barrierOuter, walkH, RoadMaterial::Sidewalk);
82 pushPoint(profile, sidewalkOuter, walkH, RoadMaterial::Sidewalk);
83
84 return Result<RoadProfile>::success(std::move(profile));
85}
