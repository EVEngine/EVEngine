55Result<RoadProfile> makeRoadProfile(const RoadStyle& style, int lanesForward, int lanesBackward) {
56 auto ok = validateStyle(style);
57 if (!ok.ok()) return Result<RoadProfile>::failure(ok.status());
58 if (lanesForward < 0 || lanesForward > 8)
59 return Result<RoadProfile>::failure(
60 Diagnostic::error(DiagnosticCode::InvalidArgument, "lanesForward must be in [0,8]", "lanesForward"));
61 if (lanesBackward < 0 || lanesBackward > 8)
62 return Result<RoadProfile>::failure(
63 Diagnostic::error(DiagnosticCode::InvalidArgument, "lanesBackward must be in [0,8]", "lanesBackward"));
64 if (lanesForward + lanesBackward < 1)
65 return Result<RoadProfile>::failure(Diagnostic::error(
66 DiagnosticCode::InvalidArgument, "road must contain at least one directional lane", "lanes"));
67
68 const float asphaltHalf = 0.5f * style.laneWidth * static_cast<float>(lanesForward + lanesBackward);
69 if (!style.crossSection.empty()) {
70 std::vector<std::pair<float, float>> coverage;
71 const std::size_t segments =
72 style.crossSectionClosed ? style.crossSection.size() : style.crossSection.size() - 1;
73 for (std::size_t i = 0; i < segments; ++i) {
74 const auto& a = style.crossSection[i];
75 const auto& b = style.crossSection[(i + 1) % style.crossSection.size()];
76 if (a.material == RoadMaterial::Asphalt && std::fabs(a.side - b.side) > 1e-5f)
77 coverage.emplace_back(std::min(a.side, b.side), std::max(a.side, b.side));
78 }
79 std::sort(coverage.begin(), coverage.end());
80 for (std::size_t i = 1; i < coverage.size(); ++i)
81 if (coverage[i].first < coverage[i - 1].second - 1e-5f)
82 return Result<RoadProfile>::failure(Diagnostic::error(
83 DiagnosticCode::InvalidArgument,
84 "custom driving surface must be single-valued across its lateral range", "style.crossSection"));
85 float covered = -asphaltHalf;
86 for (const auto& span : coverage) {
87 if (span.first > covered + 1e-4f) break;
88 covered = std::max(covered, span.second);
89 }
90 if (coverage.empty() || coverage.front().first > -asphaltHalf + 1e-4f || covered < asphaltHalf - 1e-4f)
91 return Result<RoadProfile>::failure(
92 Diagnostic::error(DiagnosticCode::InvalidArgument,
93 "custom asphalt section does not cover every driving lane", "style.crossSection"));
94 RoadProfile profile;
95 profile.points = style.crossSection;
96 profile.closed = style.crossSectionClosed;
97 profile.explicitMaterials = true;
98 for (const auto& point : profile.points) profile.halfWidth = std::max(profile.halfWidth, std::fabs(point.side));
99 return Result<RoadProfile>::success(std::move(profile));
100 }
101 const float barrierInner = asphaltHalf;
102 const float barrierOuter = asphaltHalf + style.curbWidth;
103 const float sidewalkOuter = barrierOuter + style.sidewalkWidth;
104 const float barrierTop = style.curbHeight;
105 const float walkH = style.sidewalkHeight;
106
107 RoadProfile profile;
108 profile.halfWidth = sidewalkOuter;
109
110 // Jersey-barrier U-channel matching the reference interchange look:
111 // sidewalk → tall barrier outer/top/inner → asphalt deck → mirror.
112 pushPoint(profile, -sidewalkOuter, walkH, RoadMaterial::Sidewalk);
113 pushPoint(profile, -barrierOuter, walkH, RoadMaterial::Sidewalk);
114 pushPoint(profile, -barrierOuter, barrierTop, RoadMaterial::Curb);
115 pushPoint(profile, -barrierInner, barrierTop, RoadMaterial::Curb);
116 pushPoint(profile, -barrierInner, 0.f, RoadMaterial::Curb);
117 pushPoint(profile, barrierInner, 0.f, RoadMaterial::Asphalt);
118 pushPoint(profile, barrierInner, barrierTop, RoadMaterial::Curb);
119 pushPoint(profile, barrierOuter, barrierTop, RoadMaterial::Curb);
120 pushPoint(profile, barrierOuter, walkH, RoadMaterial::Sidewalk);
121 pushPoint(profile, sidewalkOuter, walkH, RoadMaterial::Sidewalk);
122
123 return Result<RoadProfile>::success(std::move(profile));
124}
