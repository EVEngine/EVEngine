69BeamBuildResult BeamSkillEffect::build(const glm::vec3& cameraForward) const {
70 BeamBuildResult merged;
71 if (path_.size() < 2) {
72 merged.status = BeamBuildStatus::DegenerateInput;
73 return merged;
74 }
75 for (std::size_t segment = 1; segment < path_.size(); ++segment) {
76 BeamEffectConfig segmentConfig = geometry_;
77 segmentConfig.randomSeed = geometry_.randomSeed ^
78 static_cast<std::uint32_t>(segment * 0x9e3779b9u);
79 auto built = buildBeamEffect(path_[segment - 1u], path_[segment], cameraForward,
80 segmentConfig);
81 if (built.status != BeamBuildStatus::Built) return built;
82 const std::uint32_t base = static_cast<std::uint32_t>(merged.mesh.vertices.size());
83 merged.mesh.vertices.insert(merged.mesh.vertices.end(), built.mesh.vertices.begin(),
84 built.mesh.vertices.end());
85 merged.mesh.indices.reserve(merged.mesh.indices.size() + built.mesh.indices.size());
86 for (const auto index : built.mesh.indices) merged.mesh.indices.push_back(base + index);
87 }
88 merged.status = BeamBuildStatus::Built;
89 return merged;
90}
