44// Merge coplanar source triangles before clipping. Otherwise triangle diagonals
45// create almost-collinear cap vertices and ill-conditioned sliver triangles.
46std::vector<Vec> boundary(std::vector<Vec> points, Vec n, double eps) {
47 const Vec helper = std::abs(n.x) < .8 ? Vec{1, 0, 0} : Vec{0, 1, 0};
48 Vec u = cross(helper, n);
49 u = u * (1.0 / length(u));
50 const Vec v = cross(n, u);
51 std::sort(points.begin(), points.end(), [&](Vec a, Vec b) {
52 const double ax = dot(a, u), bx = dot(b, u);
53 return ax == bx ? dot(a, v) < dot(b, v) : ax < bx;
54 });
55 auto turn = [&](Vec a, Vec b, Vec c) { return dot(cross(b - a, c - a), n); };
56 std::vector<Vec> hull;
57 for (auto p : points) {
58 while (hull.size() >= 2 && turn(hull[hull.size() - 2], hull.back(), p) <= eps * eps) hull.pop_back();
59 hull.push_back(p);
60 }
61 const auto lower = hull.size();
62 for (std::size_t i = points.size() - 1; i > 0; --i) {
63 const auto p = points[i - 1];
64 while (hull.size() > lower && turn(hull[hull.size() - 2], hull.back(), p) <= eps * eps) hull.pop_back();
65 hull.push_back(p);
66 }
67 if (!hull.empty()) hull.pop_back();
68 return hull;
69}
70Poly sourceFaces(const FractureFragment& source, double eps) {
71 struct Group {
72 Vec n;
73 double d;
74 std::vector<Vec> points;
75 };
76 std::vector<Group> groups;
77 for (std::size_t t = 0; t < source.indices.size(); t += 3) {
78 const Vec a = point(source, source.indices[t]), b = point(source, source.indices[t + 1]),
79 c = point(source, source.indices[t + 2]);
80 Vec n = cross(b - a, c - a);
81 n = n * (1.0 / length(n));
82 const double d = dot(n, a);
83 auto found = std::find_if(groups.begin(), groups.end(),
84 [&](const Group& g) { return dot(g.n, n) > 1 - 1e-10 && std::abs(g.d - d) < eps; });
85 if (found == groups.end()) {
86 groups.push_back({n, d, {}});
87 found = groups.end() - 1;
88 }
89 uniquePush(found->points, a, eps);
90 uniquePush(found->points, b, eps);
91 uniquePush(found->points, c, eps);
92 }
93 Poly faces;
94 for (auto& g : groups) faces.push_back({boundary(std::move(g.points), g.n, eps), -1});
95 return faces;
96}
97// Keep n dot x <= d. The new cap points outwards towards the removed half-space.
98Poly clip(const Poly& input, Vec n, double d, int neighbor, double eps) {
99 Poly result;
100 std::vector<Vec> cap;
101 for (const auto& face : input) {
102 Face out{{}, face.neighbor};
103 for (std::size_t i = 0; i < face.points.size(); ++i) {
104 const Vec a = face.points[i], b = face.points[(i + 1) % face.points.size()];
105 const double da = dot(n, a) - d, db = dot(n, b) - d;
106 const bool ina = da <= eps, inb = db <= eps;
107 if (ina) uniquePush(out.points, a, eps);
108 if (ina != inb) {
109 const Vec q = a + (b - a) * (da / (da - db));
110 uniquePush(out.points, q, eps);
111 uniquePush(cap, q, eps);
112 }
113 if (std::abs(da) <= eps) uniquePush(cap, a, eps);
114 }
115 if (out.points.size() >= 3) result.push_back(std::move(out));
116 }
117 if (cap.size() >= 3) {
118 cap = boundary(std::move(cap), n, eps);
119 if (cap.size() >= 3) result.push_back({std::move(cap), neighbor});
120 }
121 return result;
122}

186Result<FracturedCollection> cookVoronoi(const eve::asset::CanonicalMeshData& mesh, const FractureRecipe& recipe) {
187 auto rv = recipe.validate();
188 if (!rv) return Result<FracturedCollection>::failure(rv.status());
189 if ((recipe.mode != FractureMode::UniformVoronoi && recipe.mode != FractureMode::ClusteredVoronoi) ||
190 recipe.siteCountMax > 64 || recipe.siteCountMax > recipe.maximumBones)
191 return Result<FracturedCollection>::failure(
192 bad("convex Voronoi cook requires 1..64 sites within maximumBones"));
193 Vec origin, lo, hi;
194 double eps = 0;
195 auto source = sourceMesh(mesh, origin, lo, hi, eps);
196 if (!source) return Result<FracturedCollection>::failure(source.status());
197 if (std::min({hi.x - lo.x, hi.y - lo.y, hi.z - lo.z}) < recipe.minimumThickness)
198 return Result<FracturedCollection>::failure(bad("source is thinner than minimumThickness"));
199 Poly original = sourceFaces(source.value(), eps);
200 Rng rng{seed(recipe)};
201 const int count =
202 recipe.siteCountMin +
203 static_cast<int>(rng.next() % static_cast<std::uint64_t>(recipe.siteCountMax - recipe.siteCountMin + 1));
204 auto inside = [&](Vec p) {
205 for (const auto& f : original) {
206 Vec n = cross(f.points[1] - f.points[0], f.points[2] - f.points[0]);
207 if (dot(n, p - f.points[0]) > eps * length(n)) return false;
208 }
209 return true;
210 };
211 auto uniform = [&]() {
212 return Vec{lo.x + (hi.x - lo.x) * rng.unit(), lo.y + (hi.y - lo.y) * rng.unit(),
213 lo.z + (hi.z - lo.z) * rng.unit()};
214 };
215 std::vector<Vec> centers, sites;
216 std::vector<int> clusters;
217 if (recipe.mode == FractureMode::ClusteredVoronoi) {
218 for (int c = 0; c < recipe.clusterCount; ++c) {
219 bool found = false;
220 for (int trial = 0; trial < 10000; ++trial) {
221 auto p = uniform();
222 if (inside(p)) {
223 centers.push_back(p);
224 found = true;
225 break;
226 }
227 }
228 if (!found) return Result<FracturedCollection>::failure(bad("cannot place cluster center inside source"));
229 }
230 }
231 for (int i = 0; i < count; ++i) {
232 bool found = false;
233 const int cluster = centers.empty() ? 0 : i % recipe.clusterCount;
234 for (int trial = 0; trial < 10000; ++trial) {
235 Vec p = uniform();
236 if (!centers.empty()) {
237 auto c = centers[static_cast<std::size_t>(cluster)];
238 p = c + Vec{(rng.unit() * 2 - 1) * recipe.clusterRadius, (rng.unit() * 2 - 1) * recipe.clusterRadius,
239 (rng.unit() * 2 - 1) * recipe.clusterRadius};
240 }
241 if (!inside(p)) continue;
242 bool duplicate = false;
243 for (auto q : sites)
244 if (length(q - p) < eps * 100) {
245 duplicate = true;
246 break;
247 }
248 if (duplicate) continue;
249 sites.push_back(p);
250 clusters.push_back(cluster);
251 found = true;
252 break;
253 }
254 if (!found)
255 return Result<FracturedCollection>::failure(bad("cannot place distinct Voronoi sites inside source"));
256 }
257 FracturedCollection result;
258 std::map<std::pair<int, int>, double> contacts;
259 for (int i = 0; i < count; ++i) {
260 Poly cell = original;
261 for (int j = 0; j < count; ++j)
262 if (i != j) {
263 const Vec diff = sites[static_cast<std::size_t>(j)] - sites[static_cast<std::size_t>(i)];
264 const double len = length(diff);
265 const Vec n = diff * (1.0 / len);
266 const double d = dot(n, (sites[static_cast<std::size_t>(i)] + sites[static_cast<std::size_t>(j)]) * .5);
267 cell = clip(cell, n, d, j, eps);
268 if (cell.empty()) return Result<FracturedCollection>::failure(bad("Voronoi produced empty cell"));
269 }
270 Vec center;
271 std::size_t points = 0;
272 for (const auto& f : cell)
273 for (auto p : f.points) {
274 center = center + p;
275 ++points;
276 }
277 center = center * (1.0 / static_cast<double>(points));
278 FractureFragment fragment;
279 double volume = 0;
280 Vec extents;
281 for (const auto& face : cell) {
282 Vec faceCenter;
283 for (auto p : face.points) faceCenter = faceCenter + p;
284 faceCenter = faceCenter * (1.0 / static_cast<double>(face.points.size()));
285 double area = 0;
286 for (std::size_t k = 0; k < face.points.size(); ++k) {
287 Vec a = faceCenter - center, b = face.points[k] - center,
288 c = face.points[(k + 1) % face.points.size()] - center;
289 const double triArea = length(cross(b - a, c - a)) * .5;
290 if (triArea <= eps * eps) continue;
291 const auto ia = vertex(fragment, a, eps), ib = vertex(fragment, b, eps), ic = vertex(fragment, c, eps);
292 fragment.indices.insert(fragment.indices.end(), {ia, ib, ic});
293 fragment.materials.push_back(face.neighbor < 0 ? 0U : 1U);
294 volume += dot(a, cross(b, c)) / 6;
295 area += triArea;
296 }
297 if (face.neighbor >= 0 && i < face.neighbor && area > eps * eps * 10) contacts[{i, face.neighbor}] = area;
298 }
299 for (std::uint32_t k = 0; k < fragment.positions.size() / 3; ++k) {
300 auto v = point(fragment, k);
301 extents = {std::max(extents.x, std::abs(v.x)), std::max(extents.y, std::abs(v.y)),
302 std::max(extents.z, std::abs(v.z))};
303 }
304 GeometryCollectionBone bone;
305 bone.localX = static_cast<float>(origin.x + center.x);
306 bone.localY = static_cast<float>(origin.y + center.y);
307 bone.localZ = static_cast<float>(origin.z + center.z);
308 bone.halfExtentX = static_cast<float>(extents.x);
309 bone.halfExtentY = static_cast<float>(extents.y);
310 bone.halfExtentZ = static_cast<float>(extents.z);
311 bone.mass = static_cast<float>(volume * recipe.defaultDensity);
312 bone.density = recipe.defaultDensity;
313 bone.friction = recipe.defaultFriction;
314 bone.restitution = recipe.defaultRestitution;
315 bone.clusterId = clusters[static_cast<std::size_t>(i)];
316 result.collection.bones.push_back(bone);
317 result.geometry.fragments.push_back(std::move(fragment));
318 }
319 for (const auto& [pair, area] : contacts) {
320 (void)area;
321 float threshold = recipe.defaultStrainThreshold;
322 if (recipe.mode == FractureMode::ClusteredVoronoi)
323 threshold *=
324 clusters[static_cast<std::size_t>(pair.first)] == clusters[static_cast<std::size_t>(pair.second)] ? 2.f
325 : .5f;
326 result.collection.edges.push_back({pair.first, pair.second, threshold});
327 }
328 auto gv = result.geometry.validate();
329 if (!gv) return Result<FracturedCollection>::failure(gv.status());
330 auto av = result.collection.validate();
331 if (!av) return Result<FracturedCollection>::failure(av.status());
332 return Result<FracturedCollection>::success(std::move(result));
333}

347 FractureComponentPolicy policy{0, recipe.defaultStrainThreshold, 1.f};
348 if (auto g = settings.groups.find(assembly.bones[i].clusterId); g != settings.groups.end()) policy = g->second;
349 if (auto c = settings.components.find(static_cast<int>(i)); c != settings.components.end()) policy = c->second;
350 return policy;
351 };

356 }

362 return Result<FracturedCollection>::failure(bad("component override is outside assembly"));

395 }
396 // Preserve explicit authoring supports; do not invent support between unrelated components.
397 for (const auto& parent : assembly.edges) {

399 auto [b0, b1] = ranges[static_cast<std::size_t>(parent.boneB)];

401 for (int b = b0; b < b1; ++b) {

407 result.collection.edges.push_back(

409 parent.strainThreshold *
