218 eve::Result<uint64_t> replaceBatch(const std::string& batchId, uint64_t targetRevision,
219 const std::vector<eve::ProcgenInstanceDesc>& instances) override {
220 const uint64_t currentRevision = batchRevision(batchId);
221 if (batchId.empty() || targetRevision == 0)
222 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
223 eve::DiagnosticCode::InvalidArgument,
224 "procedural scene snapshot requires a batch id and non-zero target revision", "targetRevision"));
225 if (currentRevision >= targetRevision)
226 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
227 eve::DiagnosticCode::Conflict, "procedural scene snapshot revision is stale", "targetRevision"));
228 std::unordered_set<std::string> instanceIds;
229 std::unordered_set<uint64_t> pointIds;
230 instanceIds.reserve(instances.size());
231 pointIds.reserve(instances.size());
232 for (const auto& instance : instances) {
233 if (instance.id.empty() || instance.sourcePointId == 0 || !instanceIds.insert(instance.id).second ||
234 !pointIds.insert(instance.sourcePointId).second)
235 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
236 eve::DiagnosticCode::Conflict,
237 "procedural scene snapshot requires unique instance ids and source PointIds", "instances"));
238 }
239 if (!applyBatch(batchId, instances))
240 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
241 eve::DiagnosticCode::Failed, "procedural scene provider rejected snapshot commit", "batchId"));
242 revisions_[batchId] = targetRevision;
243 return eve::Result<uint64_t>::success(targetRevision);
244 }

253 const eve::ProcgenBatchSnapshot* snapshot = nullptr;
254 SceneHost* host = nullptr;
255 SceneHost::Tree tree;
256 std::unordered_set<std::string> ids;
257 std::unordered_map<uint64_t, std::string> pointIds;

260 std::vector<std::string> removedIds;
261 Stats stats;
262 };
263 std::vector<PreparedBatch> prepared;
264 prepared.reserve(snapshots.size());
265 std::unordered_set<std::string> batchIds;
266 batchIds.reserve(snapshots.size());
267
268 for (const auto& snapshot : snapshots) {
269 if (snapshot.batchId.empty() || snapshot.targetRevision == 0)
270 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
271 eve::DiagnosticCode::InvalidArgument,
272 "procedural scene transaction requires batch ids and non-zero revisions", "snapshots"));
273 if (!batchIds.insert(snapshot.batchId).second)
274 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
275 eve::DiagnosticCode::Conflict, "procedural scene transaction repeats a batch id", "snapshots"));
276 if (batchRevision(snapshot.batchId) >= snapshot.targetRevision)
277 return eve::Result<uint64_t>::failure(
278 eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
279 "procedural scene transaction contains a stale revision", snapshot.batchId));
280
281 PreparedBatch batch;
282 batch.snapshot = &snapshot;
283 batch.ids.reserve(snapshot.instances.size());
284 batch.pointIds.reserve(snapshot.instances.size());
285 for (const auto& instance : snapshot.instances) {
286 if (instance.id.empty() || instance.sourcePointId == 0 || !batch.ids.insert(instance.id).second ||
287 !batch.pointIds.emplace(instance.sourcePointId, instance.id).second)
288 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
289 eve::DiagnosticCode::Conflict,
290 "procedural scene transaction requires unique instance ids and source PointIds",
291 snapshot.batchId));
292 }
293 const auto previous = ids_.find(snapshot.batchId);
294 const auto& previousIds = previous == ids_.end() ? emptyIds_ : previous->second;
295 for (const auto& id : batch.ids) {
296 if (previousIds.find(id) == previousIds.end()) {
297 ++batch.stats.created;
298 batch.addedIds.push_back(id);
299 } else {
300 ++batch.stats.reused;
301 batch.reusedIds.push_back(id);
302 }
303 }
304 for (const auto& id : previousIds) {
305 if (batch.ids.find(id) != batch.ids.end()) continue;
306 ++batch.stats.removed;
307 batch.removedIds.push_back(id);
308 }
309
310 batch.host = hostByName(hostName(snapshot.batchId));
311 NodeDesc root = makeRoot(snapshot.batchId, snapshot.instances);
312 auto tree = SceneHost::buildDetachedTree(batch.host ? &*batch.host->tree() : nullptr, std::move(root));
313 if (!tree) return eve::Result<uint64_t>::failure(tree.status());
314 batch.tree = std::move(tree).takeValue();
315 prepared.push_back(std::move(batch));
316 }

363 eve::Result<uint64_t> applyDelta(const std::string& batchId,
364 const eve::ProcgenInstanceDelta& delta) override {
365 const auto current = revisions_.find(batchId);
366 if (batchId.empty() || current == revisions_.end())
367 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
368 eve::DiagnosticCode::NotFound, "procedural scene batch is not published", "batchId"));
369 if (delta.baseRevision == 0 || current->second != delta.baseRevision ||
370 delta.targetRevision != delta.baseRevision + 1)
371 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
372 eve::DiagnosticCode::Conflict, "procedural scene delta revision is stale", "baseRevision"));
373
374 std::unordered_map<std::string, eve::ProcgenInstanceDesc> staged;
375 for (const auto& instance : instances_.at(batchId)) staged.emplace(instance.id, instance);
376 std::unordered_map<uint64_t, std::string> stagedPointIds = pointIds_.at(batchId);
377 std::unordered_set<std::string> removedIds;
378 removedIds.reserve(delta.removedPointIds.size() + delta.removed.size());
379 for (const auto pointId : delta.removedPointIds) {
380 const auto found = stagedPointIds.find(pointId);
381 if (pointId == 0 || found == stagedPointIds.end() || !removedIds.insert(found->second).second)
382 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
383 eve::DiagnosticCode::Conflict, "procedural scene delta removes an unknown source PointId",
384 "removedPointIds"));
385 }
386 for (const auto& id : delta.removed)
387 if (id.empty() || !removedIds.insert(id).second)
388 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
389 eve::DiagnosticCode::Conflict, "procedural scene delta repeats a removed identity", "removed"));
390 for (const auto& id : removedIds) {
391 const auto found = staged.find(id);
392 if (found == staged.end())
393 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
394 eve::DiagnosticCode::Conflict, "procedural scene delta removes an unknown identity", "removed"));
395 if (found->second.sourcePointId != 0) stagedPointIds.erase(found->second.sourcePointId);
396 staged.erase(found);
397 }
398 for (const auto& instance : delta.updated) {
399 if (instance.id.empty() || instance.sourcePointId == 0)
400 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
401 eve::DiagnosticCode::InvalidArgument,
402 "procedural scene delta updates require an id and source PointId", "updated"));
403 const auto identity = stagedPointIds.find(instance.sourcePointId);
404 if (identity == stagedPointIds.end())
405 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
406 eve::DiagnosticCode::Conflict, "procedural scene delta updates an unknown identity", "updated"));
407 if (identity->second != instance.id && staged.find(instance.id) != staged.end())
408 return eve::Result<uint64_t>::failure(
409 eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
410 "procedural scene delta update renames to a duplicate identity", "updated"));
411 staged.erase(identity->second);
412 staged.emplace(instance.id, instance);
413 identity->second = instance.id;
414 }
415 for (const auto& instance : delta.added) {
416 if (instance.id.empty() || instance.sourcePointId == 0 ||
417 stagedPointIds.find(instance.sourcePointId) != stagedPointIds.end() ||
418 !staged.emplace(instance.id, instance).second)
419 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
420 eve::DiagnosticCode::Conflict, "procedural scene delta adds a duplicate identity", "added"));
421 stagedPointIds.emplace(instance.sourcePointId, instance.id);
422 }
423 if (!delta.targetPointOrder.empty() && !delta.targetOrder.empty())
424 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
425 eve::DiagnosticCode::InvalidArgument,
426 "procedural scene delta cannot mix PointId and compatibility target orders", "targetPointOrder"));
427 std::vector<std::string> resolvedOrder;
428 if (!delta.targetPointOrder.empty()) {
429 resolvedOrder.reserve(delta.targetPointOrder.size());
430 std::unordered_set<uint64_t> orderedPointIds;
431 for (const auto pointId : delta.targetPointOrder) {
432 const auto found = stagedPointIds.find(pointId);
433 if (pointId == 0 || found == stagedPointIds.end() || !orderedPointIds.insert(pointId).second)
434 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
435 eve::DiagnosticCode::Conflict,
436 "procedural scene delta target order has an unknown or duplicate source PointId",
437 "targetPointOrder"));
438 resolvedOrder.push_back(found->second);
439 }
440 } else {
441 resolvedOrder = delta.targetOrder;
442 }
443 if (staged.size() != resolvedOrder.size())
444 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
445 eve::DiagnosticCode::Conflict, "procedural scene delta target order is incomplete", "targetOrder"));
446 std::vector<eve::ProcgenInstanceDesc> target;
447 target.reserve(resolvedOrder.size());
448 for (const auto& id : resolvedOrder) {
449 auto found = staged.find(id);
450 if (found == staged.end())
451 return eve::Result<uint64_t>::failure(
452 eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
453 "procedural scene delta target order has unknown identity", "targetOrder"));
454 target.push_back(found->second);
455 staged.erase(found);
456 }
457 if (!staged.empty())
458 return eve::Result<uint64_t>::failure(
459 eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
460 "procedural scene delta target order repeats an identity", "targetOrder"));
461 return replaceBatch(batchId, delta.targetRevision, target);
462 }

497 const std::function<eve::Result<void>()>& commitPreparedOwner) override {
498 if (batchIds.empty())

501 "procedural scene removal transaction requires at least one batch id", "batchIds"));

504 SceneHost* host = nullptr;
505 SceneHost::Tree tree;
506 std::vector<std::string> removedIds;

511 std::unordered_set<std::string> uniqueBatchIds;
512 uniqueBatchIds.reserve(batchIds.size());

526 removal.host = host;
527 removal.removedIds.assign(ids->second.begin(), ids->second.end());
528 std::sort(removal.removedIds.begin(), removal.removedIds.end());
529 removal.stats.removed = int(ids->second.size());
530 auto tree = SceneHost::buildDetachedTree(&*host->tree(), makeRoot(batchId, {}));
531 if (!tree) return eve::Result<uint64_t>::failure(tree.status());
532 removal.tree = std::move(tree).takeValue();
533 prepared.push_back(std::move(removal));

549 }

558 removal.host->setVisible(false);
559 }
560 counts_.swap(nextCounts);
561 ids_.swap(nextIds);
562 instances_.swap(nextInstances);
563 pointIds_.swap(nextPointIds);
564 revisions_.swap(nextRevisions);
565 stats_.swap(nextStats);
566 for (const auto& removal : prepared)
567 for (const auto& id : removal.removedIds) removal.host->fireEvent("node_removed", id);
568 return eve::Result<uint64_t>::success(static_cast<uint64_t>(prepared.size()));
569 }
