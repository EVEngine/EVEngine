33float interpTo(float current, float target, float dt, float speed) {
34 const float delta = wrapPi(target - current);
35 if (speed <= 0.f) return target;
36 const float step = std::clamp(dt * speed, 0.f, 1.f);
37 return wrapPi(current + delta * step);
38}

40void setWorldRotation(AnimPose& pose, const AnimSkeleton& skeleton, int bone, float qx, float qy, float qz, float qw) {
41 const int parent = skeleton.getParent(bone);
42 float lx, ly, lz, lw;
43 if (parent < 0) {
44 lx = qx;
45 ly = qy;
46 lz = qz;
47 lw = qw;
48 } else {
49 const auto& parentWorld = pose.world(parent);
50 multiplyQuat(-parentWorld.qx, -parentWorld.qy, -parentWorld.qz, parentWorld.qw, qx, qy, qz, qw, lx, ly, lz, lw);
51 }
52 pose.setLocalRotation(bone, lx, ly, lz, lw);
53}

55void addWorldYaw(AnimPose& pose, const AnimSkeleton& skeleton, int bone, float yaw) {
56 pose.computeWorld(&skeleton);
57 const auto& world = pose.world(bone);
58 float qx, qy, qz, qw;
59 const float half = yaw * 0.5f;
60 multiplyQuat(0.f, std::sin(half), 0.f, std::cos(half), world.qx, world.qy, world.qz, world.qw, qx, qy, qz, qw);
61 setWorldRotation(pose, skeleton, bone, qx, qy, qz, qw);
62}

64bool validBoneList(const AnimSkeleton& skeleton, int root, std::span<const int> bones, int maxCount) {
65 if (bones.size() > static_cast<std::size_t>(maxCount)) return false;
66 for (std::size_t i = 0; i < bones.size(); ++i) {
67 if (bones[i] < 0 || bones[i] >= skeleton.getBoneCount() || bones[i] == root) return false;
68 for (std::size_t j = 0; j < i; ++j)
69 if (bones[i] == bones[j]) return false;
70 }
71 return true;

92eve::Result<void> OrientationWarping::configure(AnimSkeleton& skeleton, int rootBone, std::span<const int> spineBones,
93 std::span<const int> ikBones) {
94 if (rootBone < 0 || rootBone >= skeleton.getBoneCount())
95 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
96 "root bone must belong to the skeleton",
97 "orientationWarping", {}, "animation"));
98 if (!validBoneList(skeleton, rootBone, spineBones, kMaxSpineBones))
99 return eve::Result<void>::failure(eve::Diagnostic::error(
100 eve::DiagnosticCode::InvalidArgument, "spine bones must be unique valid indices distinct from the root",
101 "orientationWarping", {}, "animation"));
102 if (!validBoneList(skeleton, rootBone, ikBones, kMaxIkBones))
103 return eve::Result<void>::failure(eve::Diagnostic::error(
104 eve::DiagnosticCode::InvalidArgument, "ik bones must be unique valid indices distinct from the root",
105 "orientationWarping", {}, "animation"));
106 for (int spine : spineBones)
107 for (int ik : ikBones)
108 if (spine == ik)
109 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
110 "a bone cannot be both spine and ik",
111 "orientationWarping", {}, "animation"));
112 impl_->skeleton = &skeleton;
113 impl_->rootBone = rootBone;
114 impl_->spineBones.assign(spineBones.begin(), spineBones.end());
115 impl_->ikBones.assign(ikBones.begin(), ikBones.end());
116 impl_->appliedAngle = 0.f;
117 impl_->targetAngle = 0.f;
118 return ok(eve::StatusCode::Applied);
119}

173eve::Result<void> OrientationWarping::apply(AnimPose& pose, float locomotionX, float locomotionZ, float animatedX,
174 float animatedZ, float dt) {
175 if (!impl_->skeleton)
176 return eve::Result<void>::failure(eve::Diagnostic::error(
177 eve::DiagnosticCode::InvalidArgument, "configure a skeleton before applying orientation warp",
178 "orientationWarping", {}, "animation"));
179 if (pose.getBoneCount() != impl_->skeleton->getBoneCount())
180 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
181 "pose bone count must match the skeleton",
182 "orientationWarping", {}, "animation"));
183 if (!std::isfinite(dt) || dt < 0.f || dt > 1.f)
184 return eve::Result<void>::failure(eve::Diagnostic::error(
185 eve::DiagnosticCode::InvalidArgument, "dt must be finite in [0,1]", "orientationWarping", {}, "animation"));
186 for (float value : {locomotionX, locomotionZ, animatedX, animatedZ})
187 if (!std::isfinite(value))
188 return eve::Result<void>::failure(eve::Diagnostic::error(
189 eve::DiagnosticCode::InvalidArgument, "locomotion and animated velocities must be finite",
190 "orientationWarping", {}, "animation"));
191 for (int i = 0; i < pose.getBoneCount(); ++i) {
192 const auto& t = pose.local(i);
193 for (float value : {t.px, t.py, t.pz, t.qx, t.qy, t.qz, t.qw, t.sx, t.sy, t.sz})
194 if (!std::isfinite(value))
195 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
196 "pose locals must be finite",
197 "orientationWarping", {}, "animation"));
198 }
199
200 float target = 0.f;
201 if (planarSpeed(locomotionX, locomotionZ) >= impl_->minSpeed &&
202 planarSpeed(animatedX, animatedZ) >= impl_->minSpeed) {
203 const float angle = signedPlanarAngle(animatedX, animatedZ, locomotionX, locomotionZ);
204 if (std::fabs(angle) < impl_->angleThreshold) target = angle;
205 }
206 const float applied = interpTo(impl_->appliedAngle, target, dt, impl_->interpSpeed);
207 if (!impl_->enabled) {
208 impl_->targetAngle = target;
209 impl_->appliedAngle = applied;
210 return ok(eve::StatusCode::NoOp);
211 }
212 if (std::fabs(applied) <= 1e-6f) {
213 impl_->targetAngle = target;
214 impl_->appliedAngle = 0.f;
215 return ok(eve::StatusCode::NoOp);
216 }
217
218 AnimPose next;
219 next.copyFrom(&pose);
220 next.computeWorld(impl_->skeleton);
221 struct SavedRotation {
222 int bone = 0;
223 float qx = 0.f, qy = 0.f, qz = 0.f, qw = 1.f;
224 };
225 std::vector<SavedRotation> ikWorld;
226 ikWorld.reserve(impl_->ikBones.size());
227 for (int bone : impl_->ikBones) {
228 const auto& world = next.world(bone);
229 ikWorld.push_back({bone, world.qx, world.qy, world.qz, world.qw});
230 }
231
232 const float spineShare = impl_->spineBones.empty() ? 0.f : impl_->distributedAlpha;
233 addWorldYaw(next, *impl_->skeleton, impl_->rootBone, applied * (1.f - spineShare));
234 if (!impl_->spineBones.empty()) {
235 const float perBone = applied * spineShare / static_cast<float>(impl_->spineBones.size());
236 for (int bone : impl_->spineBones) addWorldYaw(next, *impl_->skeleton, bone, perBone);
237 }
238 next.computeWorld(impl_->skeleton);
239 for (const auto& saved : ikWorld)
240 setWorldRotation(next, *impl_->skeleton, saved.bone, saved.qx, saved.qy, saved.qz, saved.qw);
241 next.computeWorld(impl_->skeleton);
242
243 pose.copyFrom(&next);
244 impl_->targetAngle = target;
245 impl_->appliedAngle = applied;
246 return ok(eve::StatusCode::Applied);
247}
