14HexCoordinates HexCoordinates::fromWorldPosition(HexVec3 position) noexcept {
15 const float diameter = HexMetrics::innerDiameter();
16 if (diameter <= 0.f) return HexCoordinates{};
17 const float fx = position.x / diameter;
18 const float fy = -fx;
19 const float offset = position.z / (HexMetrics::outerRadius() * 3.f);
20
21 const float ax = fx - offset;
22 const float ay = fy - offset;
23
24 std::int32_t ix = static_cast<std::int32_t>(std::lround(ax));
25 std::int32_t iy = static_cast<std::int32_t>(std::lround(ay));
26 std::int32_t iz = static_cast<std::int32_t>(std::lround(-ax - ay));
27
28 if (ix + iy + iz != 0) {
29 const float dx = std::fabs(ax - static_cast<float>(ix));
30 const float dy = std::fabs(ay - static_cast<float>(iy));
31 const float dz = std::fabs(-ax - ay - static_cast<float>(iz));
32 if (dx > dy && dx > dz)
33 ix = -iy - iz;
34 else if (dz > dy)
35 iz = -ix - iy;
36 }
37 return HexCoordinates{ix, iz};
38}

79[[nodiscard]] float neighbourhoodTopY(const HexMap& map, HexCoordinates coordinates) noexcept {
80 float top = cellTopY(map, coordinates);
81 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
82 HexCoordinates neighbour{};
83 if (!map.getNeighbor(coordinates, static_cast<HexDirection>(i), neighbour)) continue;
84 const float candidate = cellTopY(map, neighbour);
85 if (candidate > top) top = candidate;
86 }
87 return top;
88}

112void appendFogColumn(const HexMap& map, HexCoordinates coordinates, float shade, HexMeshData& out) {
113 const HexNoise& noise = map.noise();
114 const HexVec3 ground = map.cellGroundPosition(coordinates);
115 const float top = neighbourhoodTopY(map, coordinates) + kFogCapLift;
116 const float bottom = HexMetrics::elevationY(HexMetrics::kMinElevation) - kFogSkirtDepth;
117
118 HexVec3 ring[kHexDirectionCount];
119 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
120 HexVec3 corner = noise.perturb(ground + HexMetrics::corner(static_cast<HexDirection>(i)));
121 corner.y = top;
122 ring[i] = corner;
123 }
124 HexVec3 center = noise.perturb(HexVec3{ground.x, 0.f, ground.z});
125 center.y = top;
126
127 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
128 const std::int32_t next = (i + 1) % kHexDirectionCount;
129 emitTriangle(out, center, ring[i], ring[next], shade);
130
131 HexVec3 topNear = ring[i];
132 HexVec3 topFar = ring[next];
133 HexVec3 botFar = ring[next];
134 HexVec3 botNear = ring[i];
135 botNear.y = bottom;
136 botFar.y = bottom;
137 const auto i0 = static_cast<std::uint32_t>(out.vertexCount());
138 out.addVertex(topNear, shade, 0.f);
139 out.addVertex(topFar, shade, 0.f);
140 out.addVertex(botFar, shade, 0.f);
141 out.addVertex(botNear, shade, 0.f);
142 out.addQuad(i0, i0 + 1u, i0 + 2u, i0 + 3u);
143 }
144}

148void buildFogMesh(const HexMap& map, const HexVisibility& visibility, std::int32_t chunkIndex, HexMeshData& out) {
149 out.clear();
150 if (map.empty() || visibility.cellCount() != map.cellCount() || chunkIndex < 0 || chunkIndex >= map.chunkCount()) {
151 out.finalize();
152 return;
153 }
154
155 for (std::int32_t row = 0; row < HexMetrics::kChunkSizeZ; ++row) {
156 for (std::int32_t column = 0; column < HexMetrics::kChunkSizeX; ++column) {
157 const HexCoordinates coordinates = map.chunkCell(chunkIndex, column, row);
158 if (!map.contains(coordinates)) continue;
159 const std::int32_t index = map.indexOf(coordinates);
160 if (index < 0) continue;
161
162 float shade = 0.f;
163 if (!map.isExplored(coordinates)) {
164 shade = kFogShadeUnknown;
165 } else if (!visibility.isVisible(index)) {
166 shade = kFogShadeSeen;
167 } else {
168 // Fully visible: no overlay at all.
169 continue;
170 }
171 appendFogColumn(map, coordinates, shade, out);
172 }
173 }
174 out.finalize();
175}

182void buildChunkSurfaceMesh(const HexMap& map, std::int32_t chunkIndex, HexSurface surface, HexMeshData& out) {
183 switch (surface) {
184 case HexSurface::Terrain: buildTerrainMesh(map, chunkIndex, out); return;
185 case HexSurface::Water: buildWaterMesh(map, chunkIndex, out); return;
186 case HexSurface::River: buildRiverMesh(map, chunkIndex, out); return;
187 case HexSurface::Road: buildRoadMesh(map, chunkIndex, out); return;
188 case HexSurface::Fog:
189 // The fog overlay also depends on the visibility counters, which a grid-only
190 // helper cannot see; `HexMapModule::rebuildChunk` dispatches it to `buildFogMesh`.
191 break;
192 case HexSurface::Wall:
193 case HexSurface::Feature:
194 // Both are declared by `HexFeatures.h` and dispatched by
195 // `HexMapModule::rebuildChunk`; `HexMapMesh` stays free of that header.
196 break;
197 }
198 out.clear();
199 out.finalize();
200}

235void HexMeshData::finalize() noexcept {
236 normals_.assign(positions_.size(), 0.f);
237 for (std::size_t t = 0; t + 2 < indices_.size(); t += 3) {
238 const std::uint32_t ia = indices_[t];
239 const std::uint32_t ib = indices_[t + 1];
240 const std::uint32_t ic = indices_[t + 2];
241 const auto at = [this](std::uint32_t i) {
242 const std::size_t base = static_cast<std::size_t>(i) * 3u;
243 return HexVec3{positions_[base], positions_[base + 1], positions_[base + 2]};
244 };
245 const HexVec3 a = at(ia);
246 const HexVec3 b = at(ib);
247 const HexVec3 c = at(ic);
248 const HexVec3 ab = b - a;
249 const HexVec3 ac = c - a;
250 HexVec3 n{ab.y * ac.z - ab.z * ac.y, ab.z * ac.x - ab.x * ac.z, ab.x * ac.y - ab.y * ac.x};
251 const float len = std::sqrt(n.x * n.x + n.y * n.y + n.z * n.z);
252 if (len > 1e-8f) {
253 n.x /= len;
254 n.y /= len;
255 n.z /= len;
256 } else {
257 n = HexVec3{0.f, 1.f, 0.f};
258 }
259 for (std::uint32_t i : {ia, ib, ic}) {
260 const std::size_t base = static_cast<std::size_t>(i) * 3u;
261 normals_[base] = n.x;
262 normals_[base + 1] = n.y;
263 normals_[base + 2] = n.z;
264 }
265 }
266}
