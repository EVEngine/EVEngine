38Result<std::size_t> AttributeTable::appendRowFrom(const AttributeTable& source, std::size_t sourceRow) {
39 if (sourceRow >= source.rows_)
40 return Result<std::size_t>::failure(
41 Diagnostic::error(DiagnosticCode::InvalidArgument, "source attribute row is out of range", "sourceRow"));
42
43 for (const std::string& name : source.order_) {
44 const auto sourceColumn = source.columns_.find(name);
45 const auto targetColumn = columns_.find(name);
46 if (targetColumn != columns_.end() && targetColumn->second.type != sourceColumn->second.type)
47 return Result<std::size_t>::failure(
48 Diagnostic::error(DiagnosticCode::TypeMismatch,
49 "attribute '" + name + "' has incompatible source and target schema types", "name"));
50 }
51
52 AttributeTable staged = *this;
53 const std::size_t row = staged.appendRow();
54 for (const std::string& name : source.order_) {
55 const auto& sourceColumn = source.columns_.at(name);
56 auto ensured = staged.ensureColumn(name, sourceColumn.type);
57 if (!ensured.ok()) return Result<std::size_t>::failure(ensured.status());
58 Column& targetColumn = *ensured.value();
59 std::visit(
60 [sourceRow, row, &sourceColumn](auto& targetValues) {
61 using Values = std::decay_t<decltype(targetValues)>;
62 const auto& sourceValues = std::get<Values>(sourceColumn.storage);
63 targetValues[row] = sourceValues[sourceRow];
64 },
65 targetColumn.storage);
66 }
67 *this = std::move(staged);
68 return Result<std::size_t>::success(row);
69}

94Result<AttributeTable::Column*> AttributeTable::ensureColumn(std::string_view name, ProcgenAttributeType type) {
95 const std::string key(name);
96 auto found = columns_.find(key);
97 if (found != columns_.end()) {
98 if (found->second.type != type) {
99 auto mismatch = attributeTypeMismatch(name, type, found->second.type);
100 return Result<Column*>::failure(mismatch.status());
101 }
102 return Result<Column*>::success(&found->second);
103 }
104 Column replacement;
105 replacement.type = type;
106 switch (type) {
107 case ProcgenAttributeType::Float: replacement.storage = makeColumn<float>(rows_); break;
108 case ProcgenAttributeType::Int: replacement.storage = makeColumn<std::int64_t>(rows_); break;
109 case ProcgenAttributeType::Bool: replacement.storage = makeColumn<bool>(rows_); break;
110 case ProcgenAttributeType::Vector: replacement.storage = makeColumn<ProcgenAttributeVector>(rows_); break;
111 case ProcgenAttributeType::String: replacement.storage = makeColumn<std::string>(rows_); break;
112 }
113 auto inserted = columns_.emplace(key, std::move(replacement));
114 try {
115 order_.push_back(key);
116 } catch (...) {
117 columns_.erase(inserted.first);
118 throw;
119 }
120 return Result<Column*>::success(&inserted.first->second);
121}
