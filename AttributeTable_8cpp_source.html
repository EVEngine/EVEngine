34Result<std::size_t> AttributeTable::appendRowFrom(const AttributeTable& source, std::size_t sourceRow) {
35 if (sourceRow >= source.rows_)
36 return Result<std::size_t>::failure(
37 Diagnostic::error(DiagnosticCode::InvalidArgument, "source attribute row is out of range", "sourceRow"));
38
39 for (const std::string& name : source.order_) {
40 const auto sourceColumn = source.columns_.find(name);
41 const auto targetColumn = columns_.find(name);
42 if (targetColumn != columns_.end() && targetColumn->second.type != sourceColumn->second.type)
43 return Result<std::size_t>::failure(
44 Diagnostic::error(DiagnosticCode::TypeMismatch,
45 "attribute '" + name + "' has incompatible source and target schema types", "name"));
46 }
47
48 AttributeTable staged = *this;
49 const std::size_t row = staged.appendRow();
50 for (const std::string& name : source.order_) {
51 const auto& sourceColumn = source.columns_.at(name);
52 auto ensured = staged.ensureColumn(name, sourceColumn.type);
53 if (!ensured.ok()) return Result<std::size_t>::failure(ensured.status());
54 Column& targetColumn = *ensured.value();
55 std::visit(
56 [sourceRow, row, &sourceColumn](auto& targetValues) {
57 using Values = std::decay_t<decltype(targetValues)>;
58 const auto& sourceValues = std::get<Values>(sourceColumn.storage);
59 targetValues[row] = sourceValues[sourceRow];
60 },
61 targetColumn.storage);
62 }
63 *this = std::move(staged);
64 return Result<std::size_t>::success(row);
65}

90Result<void> AttributeTable::renameColumn(std::string_view from, std::string_view to) {
91 if (from.empty() || to.empty()) return invalidAttribute("attribute name must not be empty", "name");
92 if (!from.empty() && from.front() == '$')
93 return invalidAttribute("builtin selectors cannot be renamed", "from");
94 if (!to.empty() && to.front() == '$')
95 return invalidAttribute("builtin selectors cannot be rename targets", "to");
96 if (from == to) return Result<void>::success();
97 const std::string fromKey(from);
98 const std::string toKey(to);
99 const auto source = columns_.find(fromKey);
100 if (source == columns_.end())
101 return Result<void>::failure(
102 Diagnostic::error(DiagnosticCode::NotFound, "attribute '" + fromKey + "' does not exist", "from"));
103 if (columns_.find(toKey) != columns_.end())
104 return Result<void>::failure(
105 Diagnostic::error(DiagnosticCode::AlreadyExists, "attribute '" + toKey + "' already exists", "to"));
106
107 AttributeTable staged = *this;
108 auto moved = staged.columns_.extract(fromKey);
109 moved.key() = toKey;
110 staged.columns_.insert(std::move(moved));
111 for (std::string& name : staged.order_)
112 if (name == fromKey) {
113 name = toKey;
114 break;
115 }
116 *this = std::move(staged);
117 return Result<void>::success();
118}

162Result<AttributeTable::Column*> AttributeTable::ensureColumn(std::string_view name, ProcgenAttributeType type) {
163 const std::string key(name);
164 auto found = columns_.find(key);
165 if (found != columns_.end()) {
166 if (found->second.type != type) {
167 auto mismatch = attributeTypeMismatch(name, type, found->second.type);
168 return Result<Column*>::failure(mismatch.status());
169 }
170 return Result<Column*>::success(&found->second);
171 }
172 Column replacement;
173 replacement.type = type;
174 switch (type) {
175 case ProcgenAttributeType::Float: replacement.storage = std::vector<std::optional<float>>(rows_); break;
176 case ProcgenAttributeType::Int: replacement.storage = std::vector<std::optional<std::int64_t>>(rows_); break;
177 case ProcgenAttributeType::Bool: replacement.storage = std::vector<std::optional<bool>>(rows_); break;
178 case ProcgenAttributeType::Vector:
179 replacement.storage = std::vector<std::optional<ProcgenAttributeVector>>(rows_);
180 break;
181 case ProcgenAttributeType::String: replacement.storage = std::vector<std::optional<std::string>>(rows_); break;
182 }
183 auto inserted = columns_.emplace(key, std::move(replacement));
184 try {
185 order_.push_back(key);
186 } catch (...) {
187 columns_.erase(inserted.first);
188 throw;
189 }
190 return Result<Column*>::success(&inserted.first->second);
191}
