58Graphics::~Graphics() {
59 if (!initialized) return;
60 device->waitIdle();
61 // Pipeline objects hold raw Shader* owned by ownedShaders. Drop them
62 // first so their destructors do not see freed shader pointers.
63 pipelineAA_.reset();
64 pipelineAO_.reset();
65 pipelineGI_.reset();
66 renderControl_.reset();
67 ownedCanvases.clear();
68 ownedMeshes.clear();
69 ownedGpuMeshes.clear();
70 ownedTextures.clear();
71 for (auto &g : ownedGpuTextures) {
72 if (g->sampler) device->destroySampler(g->sampler);
73 }
74 ownedGpuTextures.clear();
75 texturesByPath.clear();
76 for (auto &g : ownedGpuShaders) {
77 if (g->swapchainPipeline) device->destroyPipeline(g->swapchainPipeline);
78 if (g->offscreenPipeline) device->destroyPipeline(g->offscreenPipeline);
79 if (g->mesh3dPipeline) device->destroyPipeline(g->mesh3dPipeline);
80 if (g->mesh3dXrayPipeline) device->destroyPipeline(g->mesh3dXrayPipeline);
81 // pipelineLayout is shared; do not destroy per-shader
82 }
83 ownedGpuShaders.clear();
84 ownedShaders.clear();
85 destroySwapchainResources();
86 if (pipeline) device->destroyPipeline(pipeline);
87 if (solidAlphaPipeline) device->destroyPipeline(solidAlphaPipeline);
88 if (additiveSolidPipeline) device->destroyPipeline(additiveSolidPipeline);
89 if (pipelineLayout) device->destroyPipelineLayout(pipelineLayout);
90 if (texPipeline) device->destroyPipeline(texPipeline);
91 if (additiveTexPipeline) device->destroyPipeline(additiveTexPipeline);
92 if (opaqueTexPipeline) device->destroyPipeline(opaqueTexPipeline);
93 if (texPipelineLayout) device->destroyPipelineLayout(texPipelineLayout);
94 if (shaderPipelineLayout) device->destroyPipelineLayout(shaderPipelineLayout);
95 if (mesh3dPipeline) device->destroyPipeline(mesh3dPipeline);
96 destroyOffscreen3DResources();
97 if (mesh3dPipelineLayout) device->destroyPipelineLayout(mesh3dPipelineLayout);
98 if (mesh3dShaderPipelineLayout) device->destroyPipelineLayout(mesh3dShaderPipelineLayout);
99 mesh3dSetLayoutUnique.reset();
100 mesh3dFrameSlots.clear();
101 destroyVoxelRectResources();
102 if (mesh3dClusteredPipeline) device->destroyPipeline(mesh3dClusteredPipeline);
103 if (mesh3dClusteredPipelineLayout) device->destroyPipelineLayout(mesh3dClusteredPipelineLayout);
104 mesh3dClusteredSetLayoutUnique.reset();
105 mesh3dClusteredFrameSlots.clear();
106 destroyShadowResources();
107 destroyGBufferResources();
108 destroySceneColorResources();
109 destroyUiColorResources();
110 auto destroyBuf = [&](vkb::GenericBuffer &b) { b.release(); };
111 for (auto &st : clusteredStorages) {
112 destroyBuf(st.lightsBuf);
113 destroyBuf(st.tableBuf);
114 destroyBuf(st.indicesBuf);
115 }
116 clusteredStorages.clear();
117 if (lit2dPipeline) device->destroyPipeline(lit2dPipeline);
118 if (offscreenLitPipeline) device->destroyPipeline(offscreenLitPipeline);
119 if (lit2dPipelineLayout) device->destroyPipelineLayout(lit2dPipelineLayout);

122 lit2dSets.clear();
123 offscreenLit2dSets.clear();
124 destroyBuf(offscreenLighting2dUbo);
125 if (offscreenSolidPipeline) device->destroyPipeline(offscreenSolidPipeline);
126 if (offscreenSolidAlphaPipeline) device->destroyPipeline(offscreenSolidAlphaPipeline);

128 if (offscreenTexPipeline) device->destroyPipeline(offscreenTexPipeline);

130 if (offscreenOpaqueTexPipeline) device->destroyPipeline(offscreenOpaqueTexPipeline);
131 if (offscreenRenderPass) device->destroyRenderPass(offscreenRenderPass);
132 texSetLayoutUnique.reset();
133 if (descriptorPool) device->destroyDescriptorPool(descriptorPool);
134 if (uploadPool) device->destroyCommandPool(uploadPool);
135 if (renderpass) device->destroyRenderPass(renderpass);
136 if (surface) inst.instance.destroySurfaceKHR(surface);

139void Graphics::createInstanceAndDevice(const std::vector<const char *> &extNames, void *nativeWindow,
140 vk::SurfaceKHR *surfaceOut) {
141 vkb::InstanceBuilder builder;
142 builder.require_api_version(1, 0);
143#if !defined(EVENGINE_IOS)
144 // Khronos validation on every draw / descriptor update is typically 5–20×
145 // slower. Opt in with EVENGINE_VULKAN_VALIDATION=1 (any value except "0").
146 const char *vkVal = std::getenv("EVENGINE_VULKAN_VALIDATION");
147 const bool wantValidation = vkVal && vkVal[0] != '\0' && vkVal[0] != '0';
148 if (wantValidation) {
149 builder.request_validation_layers();
150 builder.use_default_debug_messenger();
151 std::printf("EVEngine: Vulkan validation layers enabled (EVENGINE_VULKAN_VALIDATION)\n");
152 }
153#endif
154 for (auto *extName : extNames) builder.enable_extension(extName);
155 // No window / no surface: create a truly headless instance so the device
156 // selector does not demand a presentable queue family or a VkSurfaceKHR.
157 if (nativeWindow == nullptr) builder.set_headless(true);
158#if defined(EVENGINE_MACOSX) || defined(EVENGINE_IOS)
159 // SDL already supplies surface extensions; avoid duplicating them via
160 // InstanceBuilder's non-headless window path, and enable MoltenVK portability.
161 builder.set_headless(true);
162 // Portability enumeration comes from the Khronos loader. iOS links MoltenVK
163 // directly, so the extension is absent there and asking for it aborts
164 // instance creation.
165 if (vkb::SystemInfo::query().is_extension_available(VK_KHR_PORTABILITY_ENUMERATION_EXTENSION_NAME)) {
166 builder.enable_extension(VK_KHR_PORTABILITY_ENUMERATION_EXTENSION_NAME);
167 builder.add_flags(vk::InstanceCreateFlagBits::eEnumeratePortabilityKHR);
168 }
169#endif
170 {
171 StartupStage stage(" vulkan: instance + surface");
172 inst = builder.build();
173
174 if (nativeWindow != nullptr && surfaceOut != nullptr) {
175 auto *window = static_cast<SDL_Window *>(nativeWindow);
176 VkSurfaceKHR rawSurface = VK_NULL_HANDLE;
177 if (!SDL_Vulkan_CreateSurface(window, static_cast<VkInstance>(inst.instance), &rawSurface))
178 throw Exception("SDL_Vulkan_CreateSurface failed: %s", SDL_GetError());
179 surface = rawSurface;
180 *surfaceOut = rawSurface;
181 }
182 }
183
184 {
185 StartupStage stage(" vulkan: physical device + device");
186 auto selector = inst.selectPhysicalDevice();
187 selector.set_minimum_version(1, 0);
188 if (surface) {
189 selector.set_surface(surface);
190 } else {
191 // Headless: pick a device without requiring a presentable surface.
192 // VKBuilder's instance.headless flag is not propagated from
193 // InstanceBuilder, so defer_surface_initialization() is the
194 // supported way to select a device with no VkSurfaceKHR.
195 selector.defer_surface_initialization();
196 }
197#if defined(EVENGINE_MACOSX) || defined(EVENGINE_IOS)
198 selector.add_required_extension("VK_KHR_portability_subset");
199#endif
200 auto phys = selector.select();
201 {
202 const vk::PhysicalDeviceFeatures supported = phys->getFeatures();
203 if (supported.samplerAnisotropy) {
204 phys.features.samplerAnisotropy = VK_TRUE;
205 maxSamplerAnisotropy = phys.properties.limits.maxSamplerAnisotropy;
206 if (maxSamplerAnisotropy < 1.f) maxSamplerAnisotropy = 1.f;
207 } else {
208 maxSamplerAnisotropy = 1.f;
209 }
210 // Probe Vulkan 1.2 GPU-driven capabilities (bindless tables + GPU
211 // cull chain). The vendored headers omit `computeShader` / draw
212 // features from VkPhysicalDeviceFeatures; compute is core and
213 // universal, so treat it as present.
214 gpuDrivenCaps_ = GpuDrivenCaps{};
215 vk::PhysicalDeviceVulkan12Features vk12{};
216 vk::PhysicalDeviceFeatures2 features2{};
217 vk12.sType = vk::StructureType::ePhysicalDeviceVulkan12Features;
218 features2.sType = vk::StructureType::ePhysicalDeviceFeatures2;
219 features2.pNext = &vk12;
220 phys->getFeatures2(&features2);
221 gpuDrivenCaps_.api12 = phys.properties.apiVersion >= VK_API_VERSION_1_2;
222 gpuDrivenCaps_.computeShader = true;
223 gpuDrivenCaps_.multiDrawIndirect = supported.multiDrawIndirect == VK_TRUE;
224 gpuDrivenCaps_.shaderSampledImageArrayDynamicIndexing =
225 supported.shaderSampledImageArrayDynamicIndexing == VK_TRUE;
226 gpuDrivenCaps_.drawIndirectCount = vk12.drawIndirectCount == VK_TRUE;
227 gpuDrivenCaps_.descriptorIndexing = vk12.descriptorIndexing == VK_TRUE;
228 gpuDrivenCaps_.samplerArrayCapacity =
229 phys.properties.limits.maxPerStageDescriptorSamplers >= kMaxBindlessTextures;
230 if (gpuDrivenCaps_.gpuDrivenAvailable()) {
231 phys.features.shaderSampledImageArrayDynamicIndexing = VK_TRUE;
232 phys.features.shaderStorageBufferArrayDynamicIndexing = VK_TRUE;
233 phys.features.multiDrawIndirect = VK_TRUE;
234 }
235 }
236 vkb::DeviceBuilder deviceBuilder = phys.createDevice();
237 // Vulkan 1.2 feature: vkCmdDrawIndirectCount (VG cluster draws).
238 vk::PhysicalDeviceVulkan12Features vk12Enable{};
239 vk12Enable.sType = vk::StructureType::ePhysicalDeviceVulkan12Features;
240 if (gpuDrivenCaps_.drawIndirectCount) vk12Enable.drawIndirectCount = VK_TRUE;
241 deviceBuilder.add_pNext(&vk12Enable);
242 device = deviceBuilder.build();
243 maxSamplerAnisotropy = device.caps.maxSamplerAnisotropy;
244 }
245}

247void Graphics::initHeadless(int width, int height) {
248 StartupStage initStage("graphics: initHeadless (total)");
249 if (initialized) {
250 if (headless_) {
251 // Re-entrant: test cases share the process-wide Graphics singleton.
252 // Update the logical viewport and keep the existing device.
253 setViewportSize(width, height, width, height);
254 return;
255 }
256 throw Exception("Graphics::initHeadless: already initialized with a window");
257 }
258 if (width <= 0 || height <= 0) throw Exception("Graphics::initHeadless: invalid size");
259
260 // No window, no surface extensions, no swapchain: a bare Vulkan device that
261 // renders into offscreen canvases and reads pixels back on the CPU.
262 createInstanceAndDevice({}, nullptr, nullptr);
263
264 uploadPool = device.createCommandPool();
265 setViewportSize(width, height, width, height);
266 headless_ = true;
267
268 {
269 StartupStage stage(" vulkan: headless renderpass + pipelines");
270 // Plain color+depth render pass mirroring the swapchain pass format so
271 // the shared pipeline builders can reuse it; nothing is ever presented.
272 vkb::RenderPassBuilder rpBuilder{device};
273 renderpass =
274 rpBuilder.addPresentAttachment(vk::Format::eB8G8R8A8Unorm, vk::AttachmentLoadOp::eClear)
275 .addDepthAttachment(depthFormat, vk::AttachmentLoadOp::eClear,
276 vk::AttachmentStoreOp::eDontCare)
277 .addSubpass(vkb::SubpassBuilder()
278 .addAttachmentRef(0, vk::ImageLayout::eColorAttachmentOptimal)
279 .setDepthStencilAttachment(
280 1, vk::ImageLayout::eDepthStencilAttachmentOptimal))
281 .addDependency(VK_SUBPASS_EXTERNAL, 0,
282 vk::PipelineStageFlagBits::eColorAttachmentOutput |
283 vk::PipelineStageFlagBits::eEarlyFragmentTests,
284 vk::PipelineStageFlagBits::eColorAttachmentOutput |
285 vk::PipelineStageFlagBits::eEarlyFragmentTests,
286 {},
287 vk::AccessFlagBits::eColorAttachmentRead |
288 vk::AccessFlagBits::eColorAttachmentWrite |
289 vk::AccessFlagBits::eDepthStencilAttachmentWrite)
290 .build();
291 depthImage = vkb::DepthStencilImage{device, uint32_t(width), uint32_t(height), depthFormat};
292 pipelineLayout = createPipelineLayout(device);
293 createTexturedPipeline();
294 createMesh3DPipeline();
295 createMesh3DClusteredPipeline();
296 createVoxelRectPipeline();
297 }
298
299 // Must be set before createShadowResources(): it clears the shadow cascade
300 // layers by calling beginShadowPass()/endShadowPass(), and beginShadowPass()
301 // asserts `initialized` is already true.
302 initialized = true;
303 {
304 StartupStage stage(" vulkan: shadow resources (3x2048^2 maps + pipelines)");
305 createShadowResources();
306 }
307 {
308 StartupStage stage(" vulkan: white texture");
309 const uint8_t whitePixel[4] = {255, 255, 255, 255};
310 whiteTexture = newTexture(1, 1, whitePixel);
311 const std::vector<uint8_t> cubePx(6 * 4, 255);
312 defaultBindlessCube = newCubemap(1, cubePx.data());
313 }
314 {
315 StartupStage stage(" vulkan: gpu-driven bindless set + pipeline");
316 initGpuDrivenResources();
317 }
318}

320void Graphics::initWithWindow(void *nativeWindow) {
321 StartupStage initStage("graphics: initWithWindow (total)");
322 if (initialized) {
323 // Window module may destroy/recreate the SDL window (tests, setWindowSettings).
324 // The Vulkan surface is tied to the native window. SDL can hand back the
325 // same pointer for a freshly recreated window, so pointer identity alone
326 // can't tell whether the surface is still valid: Window::close() drops it
327 // via onNativeWindowDestroyed(), and a missing surface must be rebuilt too.
328 if (sdlWindow == nativeWindow && surface) return;
329 sdlWindow = nativeWindow;
330 recreateSurfaceForResume();
331 // Restore the initWithWindow contract (device + swapchain valid on
332 // return): rebuild the swapchain synchronously. On desktop this runs
333 // immediately; on Android/iOS isRenderSurfaceStable() defers it until
334 // the native surface settles, leaving swapchainDirty set (same as before).
335 rebuildSwapchainIfNeeded();
336 return;
337 }
338 sdlWindow = nativeWindow;
339 auto *window = static_cast<SDL_Window *>(nativeWindow);
340 ASSERT(window != nullptr);
341 if (!window) throw Exception("Graphics::initWithWindow: null SDL_Window");
342
343 unsigned int count = 0;
344 if (!SDL_Vulkan_GetInstanceExtensions(window, &count, nullptr))
345 throw Exception("SDL_Vulkan_GetInstanceExtensions failed: %s", SDL_GetError());
346
347 std::vector<const char *> extNames(count);
348 if (!SDL_Vulkan_GetInstanceExtensions(window, &count, extNames.data()))
349 throw Exception("SDL_Vulkan_GetInstanceExtensions failed: %s", SDL_GetError());
350
351 createInstanceAndDevice(extNames, nativeWindow, &surface);
352
353 uploadPool = device.createCommandPool();
354
355 int pw = 0, ph = 0;
356 SDL_Vulkan_GetDrawableSize(window, &pw, &ph);
357 int lw = 0, lh = 0;
358 SDL_GetWindowSize(window, &lw, &lh);
359 setViewportSize(lw, lh, pw, ph);
360
361 {
362 StartupStage stage(" vulkan: swapchain + solid pipelines");
363 createSwapchainAndPipeline();
364 }
365 {
366 StartupStage stage(" vulkan: textured/lit2d pipelines");
367 createTexturedPipeline();
368 }
369 {
370 StartupStage stage(" vulkan: mesh3d pipeline");
371 createMesh3DPipeline();
372 }
373 {
374 StartupStage stage(" vulkan: clustered pipeline");
375 createMesh3DClusteredPipeline();
376 }
377 {
378 StartupStage stage(" vulkan: voxel pipeline");
379 createVoxelRectPipeline();
380 }
381 // Must be set before createShadowResources(): it clears the shadow cascade
382 // layers by calling beginShadowPass()/endShadowPass(), and beginShadowPass()
383 // asserts `initialized` is already true.
384 initialized = true;
385 {
386 StartupStage stage(" vulkan: shadow resources (3x2048^2 maps + pipelines)");
387 createShadowResources();
388 }
389 {
390 StartupStage stage(" vulkan: white texture");
391 const uint8_t whitePixel[4] = {255, 255, 255, 255};
392 whiteTexture = newTexture(1, 1, whitePixel);
393 const std::vector<uint8_t> cubePx(6 * 4, 255);
394 defaultBindlessCube = newCubemap(1, cubePx.data());
395 }
396 {
397 StartupStage stage(" vulkan: gpu-driven bindless set + pipeline");
398 initGpuDrivenResources();
399 }
400}

402void Graphics::onNativeWindowDestroyed() {
403 if (!initialized) return;
404 if (swapchainPassOpen || sceneColorPassOpen) abortOpen3DFrame();
405 // Fire registered window-destroyed callbacks (UI backend tears down its
406 // ImGui context here) before the device/surface resources are released.
407 for (auto &cb : windowDestroyedCallbacks_) cb.first(cb.second);
408 windowDestroyedCallbacks_.clear();
409 clearPresentOverlay();
410
411 device->waitIdle();
412 destroySwapchainResources();
413 swapchain.destroy();
414 swapchain = vkb::Swapchain{};
415 if (surface) {
416 inst.instance.destroySurfaceKHR(surface);
417 surface = VK_NULL_HANDLE;
418 }
419 sdlWindow = nullptr;
420 swapchainDirty = true;
421}

423void Graphics::destroySwapchainResources() {
424 presentRecording = {};
425 swapchainPass = {};
426 presentModel.destroy();
427 presentModel = vkb::Present{};
428 destroyGpuDrivenCullResources();
429 depthImage = vkb::DepthStencilImage{};
430 destroyReadbackResources();
431 // Release reused per-frame vertex buffers. Callers hold a device-wide
432 // waitIdle before this runs.
433 for (auto &fb : frame2dBuffers) releaseFrame2dBuffers(fb);
434 frame2dBuffers.clear();
435 releaseFrame2dBuffers(offscreenBuffers);
436 // Per-frame lit-2D UBOs are keyed to the in-flight slot count; release them
437 // here (slot count may change across recreation).
438 for (auto &ubo : lighting2dUboSlots) ubo.release();
439 lighting2dUboSlots.clear();
440 // Descriptor sets cached against the released UBO handles must be dropped;
441 // they live in the shared descriptor pool, so only the cache is cleared.
442 lit2dSets.clear();
443}

612bool Graphics::isRenderSurfaceReady() const {
613#if defined(EVENGINE_ANDROID) || defined(EVENGINE_IOS)
614 auto *window = static_cast<SDL_Window *>(sdlWindow);
615 if (!window) return false;

617 // torn down and rebuilt; SDL reports a zero drawable size until it settles.
618 int pw = 0, ph = 0;

620 return pw > 0 && ph > 0;
621#else
622 return true;
623#endif
624}

626bool Graphics::isRenderSurfaceStable() {
627#if defined(EVENGINE_ANDROID) || defined(EVENGINE_IOS)
628 auto *window = static_cast<SDL_Window *>(sdlWindow);
629 if (!window) return false;
630 int pw = 0, ph = 0;
631 SDL_Vulkan_GetDrawableSize(window, &pw, &ph);
632 if (pw <= 0 || ph <= 0) {
633 surfaceStableFrames = 0;
634 return false;
635 }
636 if (pw != pendingSurfaceW || ph != pendingSurfaceH) {
637 // Size still changing (mid-rotation / Split View); wait for it to settle.
638 pendingSurfaceW = pw;
639 pendingSurfaceH = ph;
640 surfaceStableFrames = 1;
641 return false;
642 }
643 // Require a couple of consecutive identical polls before rebuilding.
644 if (surfaceStableFrames < 3) {
645 ++surfaceStableFrames;
646 return false;
647 }
648 return true;
649#else
650 return true;
651#endif
652}

654void Graphics::recreateSurfaceForResume() {
655 // Mobile platforms destroy the native window when backgrounded, which
656 // invalidates the VkSurfaceKHR. On resume SDL creates a fresh native
657 // window, so we must rebuild the surface (and the swapchain that depends on
658 // it) before presenting again. Runs on the render thread.
659 if (!initialized) return;
660 auto *window = static_cast<SDL_Window *>(sdlWindow);
661 if (!window) return;
662
663 device->waitIdle();
664
665 // Tear down surface-dependent resources first.
666 destroySwapchainResources();
667 swapchain.destroy();
668 swapchain = vkb::Swapchain{};
669 if (surface) {
670 inst.instance.destroySurfaceKHR(surface);
671 surface = VK_NULL_HANDLE;
672 }
673
674 // Create a new surface from the freshly created native window.
675 VkSurfaceKHR rawSurface = VK_NULL_HANDLE;
676 if (!SDL_Vulkan_CreateSurface(window, static_cast<VkInstance>(inst.instance), &rawSurface))
677 throw Exception("resume: SDL_Vulkan_CreateSurface failed: %s", SDL_GetError());
678 surface = rawSurface;
679 device.surface = surface;
680
681 // Refresh drawable size (orientation / Split View may have changed).
682 int pw = 0, ph = 0, lw = 0, lh = 0;
683 SDL_Vulkan_GetDrawableSize(window, &pw, &ph);
684 SDL_GetWindowSize(window, &lw, &lh);
685 if (pw > 0 && ph > 0)
686 setViewportSize(lw, lh, pw, ph);
687
688 swapchainDirty = true;
689}

691void Graphics::present() {
692 if (!initialized) return;
693 if (headless_) return;
694 if (!isActive()) return;
695 if (isCanvasActive()) throw Exception("present: cannot present while a Canvas is active");
696 if (!isRenderSurfaceReady()) {
697 // Native window is mid-(re)creation / rotation; drop this frame's work
698 // instead of touching an invalid swapchain (which crashes the driver).
699 if (swapchainPassOpen)
700 abortOpen3DFrame();
701 else
702 clear2DBatches();
703 hasPendingClear = false;
704 return;
705 }
706 flushBatch();
707}
