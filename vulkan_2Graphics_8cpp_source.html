97std::vector<const char *> collectFastInstanceExtensions(const std::vector<vk::ExtensionProperties> &props,
98 vk::InstanceCreateFlags *flagsOut) {
99 std::vector<const char *> exts;
100 addIfAvailable(exts, props, "VK_KHR_surface");
101#if defined(_WIN32)
102 addIfAvailable(exts, props, "VK_KHR_win32_surface");
103#elif defined(__ANDROID__)
104 addIfAvailable(exts, props, "VK_KHR_android_surface");
105#elif defined(__APPLE__)
106 addIfAvailable(exts, props, "VK_EXT_metal_surface");
107 if (extensionAvailable(props, "VK_KHR_portability_enumeration")) {
108 exts.push_back("VK_KHR_portability_enumeration");
109 if (flagsOut) *flagsOut |= vk::InstanceCreateFlagBits::eEnumeratePortabilityKHR;
110 }
111#else
112 addIfAvailable(exts, props, "VK_KHR_xcb_surface");
113 addIfAvailable(exts, props, "VK_KHR_xlib_surface");
114 addIfAvailable(exts, props, "VK_KHR_wayland_surface");
115#endif
116 return exts;
117}

134vkb::Instance createInstanceFast() {
135 initVulkanDispatcher();
136 disableImplicitLayersIfSafe();
137 // Enumerate instance extensions only. vk-bootstrap's SystemInfo::query() also
138 // walks every implicit layer (and each layer's extensions), which is the
139 // bulk of "instance + surface" on a Windows SDK install.
140 const auto props = vk::enumerateInstanceExtensionProperties();
141 vk::InstanceCreateFlags flags{};
142 const auto exts = collectFastInstanceExtensions(props, &flags);
143
144 if (wantVulkanValidation()) {
145 vkb::InstanceBuilder builder;
146 builder.require_api_version(1, 0);
147 builder.request_validation_layers();
148 builder.use_default_debug_messenger();
149 for (auto *e : exts) builder.enable_extension(e);
150#if defined(EVENGINE_MACOSX) || defined(EVENGINE_IOS)
151 builder.set_headless(true);
152 builder.add_flags(flags);
153#endif
154 std::printf("EVEngine: Vulkan validation layers enabled (EVENGINE_VULKAN_VALIDATION)\n");
155 return builder.build();
156 }
157
158 vk::ApplicationInfo app{};
159 app.pApplicationName = "EVEngine";
160 app.pEngineName = "EVEngine";
161 app.apiVersion = VK_MAKE_VERSION(1, 0, 0);
162
163 vk::InstanceCreateInfo ci{};
164 ci.flags = flags;
165 ci.pApplicationInfo = &app;
166 ci.enabledExtensionCount = static_cast<uint32_t>(exts.size());
167 ci.ppEnabledExtensionNames = exts.data();
168
169 vkb::Instance inst;
170 inst.instance = vk::createInstance(ci);
171 VULKAN_HPP_DEFAULT_DISPATCHER.init(inst.instance);
172 return inst;
173}

187void startVulkanInstanceWarmupImpl() {
188#if defined(EVENGINE_MACOSX) || defined(EVENGINE_IOS)
189 // MoltenVK / SDL loader affinity is main-thread on Apple.
190 return;
191#else
192 auto &w = instanceWarmup();
193 std::lock_guard<std::mutex> lock(w.mu);
194 if (w.started) return;
195 w.started = true;
196 w.future = std::async(std::launch::async, [] {
197 StartupStage stage(" vulkan: instance (warmup thread)");
198 return createInstanceFast();
199 });
200#endif
201}
202
203vkb::Instance consumeWarmedInstance() {
204#if defined(EVENGINE_MACOSX) || defined(EVENGINE_IOS)
205 StartupStage stage(" vulkan: instance");
206 return createInstanceFast();
207#else
208 startVulkanInstanceWarmupImpl();
209 std::future<vkb::Instance> fut;
210 {
211 auto &w = instanceWarmup();
212 std::lock_guard<std::mutex> lock(w.mu);
213 if (w.consumed) {
214 throw Exception("Vulkan instance warmup already consumed");
215 }
216 w.consumed = true;
217 fut = std::move(w.future);
218 }
219 StartupStage waitStage(" vulkan: wait instance warmup");
220 return fut.get();
221#endif
222}
223
224void discardWarmedInstance() noexcept {
225#if !defined(EVENGINE_MACOSX) && !defined(EVENGINE_IOS)
226 std::future<vkb::Instance> fut;
227 {
228 auto &w = instanceWarmup();
229 std::lock_guard<std::mutex> lock(w.mu);
230 if (!w.started || w.consumed) return;
231 w.consumed = true;
232 fut = std::move(w.future);
233 }
234 try {
235 vkb::Instance warmed = fut.get();
236 if (static_cast<VkInstance>(warmed.instance) != VK_NULL_HANDLE) warmed.destroy();
237 } catch (const std::exception &e) {
238 std::fprintf(stderr, "[vulkan] instance warmup cleanup failed: %s\n", e.what());
239 } catch (...) {
240 std::fprintf(stderr, "[vulkan] instance warmup cleanup failed with an unknown error\n");
241 }
242#endif
243}

260void VmaAllocatorOwner::create(const vkb::Instance &instance,
261 const vkb::PhysicalDevice &physicalDevice,
262 vkb::Device &device) {
263 VmaAllocatorCreateInfo createInfo{};
264 createInfo.instance = static_cast<VkInstance>(instance.instance);
265 createInfo.physicalDevice = static_cast<VkPhysicalDevice>(physicalDevice.instance);
266 createInfo.device = static_cast<VkDevice>(device.instance);
267 VmaVulkanFunctions vulkanFunctions{};
268 // The headless backend does not ask SDL to load Vulkan, so its loader
269 // accessor may legitimately be null. Reuse the dispatcher that created
270 // this instance; it is also the portable path on Android.
271 vulkanFunctions.vkGetInstanceProcAddr = VULKAN_HPP_DEFAULT_DISPATCHER.vkGetInstanceProcAddr;
272 if (!vulkanFunctions.vkGetInstanceProcAddr) {
273 throw Exception("VMA could not resolve vkGetInstanceProcAddr from the Vulkan dispatcher");
274 }
275 vulkanFunctions.vkGetDeviceProcAddr = reinterpret_cast<PFN_vkGetDeviceProcAddr>(
276 vulkanFunctions.vkGetInstanceProcAddr(createInfo.instance, "vkGetDeviceProcAddr"));
277 if (!vulkanFunctions.vkGetDeviceProcAddr) {
278 throw Exception("VMA could not resolve vkGetDeviceProcAddr");
279 }
280 createInfo.pVulkanFunctions = &vulkanFunctions;
281 // InstanceBuilder requests Vulkan 1.0. VMA requires this value to describe
282 // the application's instance contract, not the physical device maximum.
283 createInfo.vulkanApiVersion = VK_API_VERSION_1_0;
284 const VkResult result = vmaCreateAllocator(&createInfo, &allocator_);
285 if (result != VK_SUCCESS) throw Exception("vmaCreateAllocator failed: %d", int(result));
286 device.attachVmaAllocator(allocator_);
287}

418void Graphics::createInstanceAndDevice(const std::vector<const char *> &extNames, void *nativeWindow,
419 vk::SurfaceKHR *surfaceOut) {
420 (void)extNames;
421 if (static_cast<VkInstance>(inst.instance) == VK_NULL_HANDLE)
422 inst = consumeWarmedInstance();
423
424 if (nativeWindow != nullptr && surfaceOut != nullptr) {
425 StartupStage stage(" vulkan: surface");
426 auto *window = static_cast<SDL_Window *>(nativeWindow);
427 VkSurfaceKHR rawSurface = VK_NULL_HANDLE;
428 if (!SDL_Vulkan_CreateSurface(window, static_cast<VkInstance>(inst.instance), &rawSurface))
429 throw Exception("SDL_Vulkan_CreateSurface failed: %s", SDL_GetError());
430 surface = rawSurface;
431 *surfaceOut = rawSurface;
432 }
433
434 {
435 StartupStage stage(" vulkan: physical device + device");
436 auto selector = inst.selectPhysicalDevice();
437 selector.set_minimum_version(1, 0);
438 if (surface) {
439 selector.set_surface(surface);
440 } else {
441 // Headless: pick a device without requiring a presentable surface.
442 // VKBuilder's instance.headless flag is not propagated from
443 // InstanceBuilder, so defer_surface_initialization() is the
444 // supported way to select a device with no VkSurfaceKHR.
445 selector.defer_surface_initialization();
446 }
447#if defined(EVENGINE_MACOSX) || defined(EVENGINE_IOS)
448 selector.add_required_extension("VK_KHR_portability_subset");
449#endif
450 auto phys = selector.select();
451 {
452 // Record the GPU identity into the crash/error log before any Vulkan
453 // work, so a device/driver crash can be tied to the exact GPU+driver.
454 const auto &pp = phys.properties;
455 const char *vendor = "unknown";
456 switch (pp.vendorID) {
457 case 0x10DE: vendor = "NVIDIA"; break;
458 case 0x1002: vendor = "AMD"; break;
459 case 0x8086: vendor = "Intel"; break;
460 case 0x13B5: vendor = "ARM"; break;
461 case 0x5143: vendor = "Qualcomm"; break;
462 default: break;
463 }
464 char vendorHex[16], deviceHex[16], driverHex[16];
465 std::snprintf(vendorHex, sizeof(vendorHex), "%04X", pp.vendorID);
466 std::snprintf(deviceHex, sizeof(deviceHex), "%04X", pp.deviceID);
467 std::snprintf(driverHex, sizeof(driverHex), "%08X", pp.driverVersion);
468 const uint32_t maj = VK_API_VERSION_MAJOR(pp.apiVersion);
469 const uint32_t min = VK_API_VERSION_MINOR(pp.apiVersion);
470 const uint32_t pat = VK_API_VERSION_PATCH(pp.apiVersion);
471 eve::recordLogEvent(
472 "info",
473 std::string("gpu: ") + vendor + " '" + std::string(pp.deviceName.data()) +
474 "' vendorId=0x" + vendorHex + " deviceId=0x" + deviceHex + " api=" +
475 std::to_string(maj) + "." + std::to_string(min) + "." + std::to_string(pat) +
476 " driver=0x" + driverHex + (nativeWindow ? "" : " [headless]"));
477 }
478 {
479 const vk::PhysicalDeviceFeatures supported = phys->getFeatures();
480 if (supported.samplerAnisotropy) {
481 phys.features.samplerAnisotropy = VK_TRUE;
482 maxSamplerAnisotropy = phys.properties.limits.maxSamplerAnisotropy;
483 if (maxSamplerAnisotropy < 1.f) maxSamplerAnisotropy = 1.f;
484 } else {
485 maxSamplerAnisotropy = 1.f;
486 }
487 // Probe Vulkan 1.2 GPU-driven capabilities (bindless tables + GPU
488 // cull chain). The vendored headers omit `computeShader` / draw
489 // features from VkPhysicalDeviceFeatures; compute is core and
490 // universal, so treat it as present.
491 gpuDrivenCaps_ = GpuDrivenCaps{};
492 vk::PhysicalDeviceVulkan12Features vk12{};
493 vk::PhysicalDeviceFeatures2 features2{};
494 vk12.sType = vk::StructureType::ePhysicalDeviceVulkan12Features;
495 features2.sType = vk::StructureType::ePhysicalDeviceFeatures2;
496 features2.pNext = &vk12;
497 phys->getFeatures2(&features2);
498 gpuDrivenCaps_.api12 = phys.properties.apiVersion >= VK_API_VERSION_1_2;
499 gpuDrivenCaps_.computeShader = true;
500 gpuDrivenCaps_.multiDrawIndirect = supported.multiDrawIndirect == VK_TRUE;
501 gpuDrivenCaps_.shaderSampledImageArrayDynamicIndexing =
502 supported.shaderSampledImageArrayDynamicIndexing == VK_TRUE;
503 gpuDrivenCaps_.drawIndirectCount = vk12.drawIndirectCount == VK_TRUE;
504 gpuDrivenCaps_.descriptorIndexing = vk12.descriptorIndexing == VK_TRUE;
505 gpuDrivenCaps_.samplerArrayCapacity =
506 phys.properties.limits.maxPerStageDescriptorSamplers >= kMaxBindlessTextures;
507 if (gpuDrivenCaps_.gpuDrivenAvailable()) {
508 phys.features.shaderSampledImageArrayDynamicIndexing = VK_TRUE;
509 phys.features.shaderStorageBufferArrayDynamicIndexing = VK_TRUE;
510 phys.features.multiDrawIndirect = VK_TRUE;
511 }
512 }
513 vkb::DeviceBuilder deviceBuilder = phys.createDevice();
514 // Vulkan 1.2 feature: vkCmdDrawIndirectCount (VG cluster draws).
515 vk::PhysicalDeviceVulkan12Features vk12Enable{};
516 vk12Enable.sType = vk::StructureType::ePhysicalDeviceVulkan12Features;
517 if (gpuDrivenCaps_.drawIndirectCount) vk12Enable.drawIndirectCount = VK_TRUE;
518 deviceBuilder.add_pNext(&vk12Enable);
519 device = deviceBuilder.build();
520#if defined(VKB_ENABLE_VMA)
521 vmaAllocatorOwner_.create(inst, phys, device);
522#endif
523 maxSamplerAnisotropy = device.caps.maxSamplerAnisotropy;
524 eve::recordLogEvent("info",
525 "gpu: logical device created (gpuDriven=" +
526 std::string(gpuDrivenCaps_.gpuDrivenAvailable() ? "on" : "off") +
527#if defined(VKB_ENABLE_VMA)
528 ", allocator=VMA" +
529#else
530 ", allocator=native" +
531#endif
532 ", maxAniso=" + std::to_string(maxSamplerAnisotropy) + ")");
533 }

607void Graphics::initWithWindow(void *nativeWindow) {
608 eve::cap::provide<eve::service::IGpuTimer>(this);
609 StartupStage initStage("graphics: initWithWindow (total)");
610 if (initialized) {
611 // Window module may destroy/recreate the SDL window (tests, setWindowSettings).
612 // The Vulkan surface is tied to the native window. SDL can hand back the
613 // same pointer for a freshly recreated window, so pointer identity alone
614 // can't tell whether the surface is still valid: Window::close() drops it
615 // via onNativeWindowDestroyed(), and a missing surface must be rebuilt too.
616 if (sdlWindow == nativeWindow && surface) return;
617 sdlWindow = nativeWindow;
618 recreateSurfaceForResume();
619 // Restore the initWithWindow contract (device + swapchain valid on
620 // return): rebuild the swapchain synchronously. On desktop this runs
621 // immediately; on Android/iOS isRenderSurfaceStable() defers it until
622 // the native surface settles, leaving swapchainDirty set (same as before).
623 rebuildSwapchainIfNeeded();
624 return;
625 }
626 sdlWindow = nativeWindow;
627 auto *window = static_cast<SDL_Window *>(nativeWindow);
628 ASSERT(window != nullptr);
629 if (!window) throw Exception("Graphics::initWithWindow: null SDL_Window");
630

632 if (!SDL_Vulkan_GetInstanceExtensions(window, &count, nullptr))
633 throw Exception("SDL_Vulkan_GetInstanceExtensions failed: %s", SDL_GetError());
634

636 if (!SDL_Vulkan_GetInstanceExtensions(window, &count, extNames.data()))
637 throw Exception("SDL_Vulkan_GetInstanceExtensions failed: %s", SDL_GetError());
638
639 createInstanceAndDevice(extNames, nativeWindow, &surface);
640
641 uploadPool = device.createCommandPool();
642
643 int pw = 0, ph = 0;
644 SDL_Vulkan_GetDrawableSize(window, &pw, &ph);
645 int lw = 0, lh = 0;
646 SDL_GetWindowSize(window, &lw, &lh);
647 setViewportSize(lw, lh, pw, ph);
648
649 {
650 StartupStage stage(" vulkan: swapchain + solid pipelines");
651 createSwapchainAndPipeline();
652 eve::recordLogEvent("info", "gfx: swapchain + solid pipelines created");
653 }
654 {
655 StartupStage stage(" vulkan: textured/lit2d pipelines");
656 createTexturedPipeline();
657 }
658 {
659 StartupStage stage(" vulkan: mesh3d pipeline");
660 createMesh3DPipeline();
661 }
662 {
663 StartupStage stage(" vulkan: clustered pipeline");
664 createMesh3DClusteredPipeline();
665 }
666 {
667 StartupStage stage(" vulkan: voxel pipeline");
668 createVoxelRectPipeline();
669 }
670 initialized = true;
671 {
672 StartupStage stage(" vulkan: shadow resources (3x2048^2 maps + pipelines)");
673 createShadowResources();
674 }

676 StartupStage stage(" vulkan: white texture");

876bool Graphics::rebuildSwapchainIfNeeded() {
877 const bool wantRecreate =
878 surfaceNeedsRecreate.load() || swapchainDirty || presentModel.needs_recreate;
879 if (!wantRecreate) return true;

895bool Graphics::beginPresentCommandBuffer() {
896 for (int attempt = 0; attempt < 3; ++attempt) {
897 if (!rebuildSwapchainIfNeeded()) return false;
898 presentRecording = presentModel.begin();
899 if (presentRecording) {
900 writeGpuTimestampBegin();
901 return true;
902 }
903 // Acquire failed without beginning the CB (see Present::begin). Rebuild once.
904 if (presentModel.needs_recreate) {
905 presentModel.needs_recreate = false;
906 swapchainDirty = true;
907 continue;
908 }
909 return false;
910 }
911 return false;

928bool Graphics::isRenderSurfaceStable() {
929#if defined(EVENGINE_ANDROID) || defined(EVENGINE_IOS)
930 auto *window = static_cast<SDL_Window *>(sdlWindow);
931 if (!window) return false;
932 int pw = 0, ph = 0;
933 SDL_Vulkan_GetDrawableSize(window, &pw, &ph);
934 if (pw <= 0 || ph <= 0) {
935 surfaceStableFrames = 0;
936 return false;
937 }
938 if (pw != pendingSurfaceW || ph != pendingSurfaceH) {
939 // Size still changing (mid-rotation / Split View); wait for it to settle.
940 pendingSurfaceW = pw;
941 pendingSurfaceH = ph;
942 surfaceStableFrames = 1;
943 return false;
944 }
945 // Require a couple of consecutive identical polls before rebuilding.
946 if (surfaceStableFrames < 3) {
947 ++surfaceStableFrames;
948 return false;
949 }
950 return true;
951#else
952 return true;
953#endif
954}

956void Graphics::recreateSurfaceForResume() {
957 // Mobile platforms destroy the native window when backgrounded, which
958 // invalidates the VkSurfaceKHR. On resume SDL creates a fresh native
959 // window, so we must rebuild the surface (and the swapchain that depends on
960 // it) before presenting again. Runs on the render thread.
961 if (!initialized) return;
962 auto *window = static_cast<SDL_Window *>(sdlWindow);
963 if (!window) return;
964
965 device->waitIdle();
966
967 // Tear down surface-dependent resources first.
968 destroySwapchainResources();
969 swapchain.destroy();
970 swapchain = vkb::Swapchain{};
971 if (surface) {
972 inst.instance.destroySurfaceKHR(surface);
973 surface = VK_NULL_HANDLE;
974 }
975
976 // Create a new surface from the freshly created native window.
977 VkSurfaceKHR rawSurface = VK_NULL_HANDLE;
978 if (!SDL_Vulkan_CreateSurface(window, static_cast<VkInstance>(inst.instance), &rawSurface))
979 throw Exception("resume: SDL_Vulkan_CreateSurface failed: %s", SDL_GetError());
980 surface = rawSurface;
981 device.surface = surface;
982
983 // Refresh drawable size (orientation / Split View may have changed).
984 int pw = 0, ph = 0, lw = 0, lh = 0;
985 SDL_Vulkan_GetDrawableSize(window, &pw, &ph);
986 SDL_GetWindowSize(window, &lw, &lh);
987 if (pw > 0 && ph > 0)
988 setViewportSize(lw, lh, pw, ph);
989
990 swapchainDirty = true;
991}

1044void Graphics::readGpuFrameTiming() {
1045 if (!gpuTimingReady_ || !gpuQueryPool_) return;
1046 // drawFrame() waits on the in-flight fence, so results are valid here.
1047 std::array<uint64_t, 2> ts{};
1048 const auto r = device->getQueryPoolResults(
1049 gpuQueryPool_, 0, 2, sizeof(ts), ts.data(), sizeof(uint64_t),
1050 vk::QueryResultFlagBits::e64 | vk::QueryResultFlagBits::eWait);
1051 if (r == vk::Result::eSuccess && ts[1] > ts[0])
1052 gpuFrameMs_ = float(double(ts[1] - ts[0]) * double(timestampPeriod_) / 1'000'000.0);
1053 else
1054 gpuFrameMs_ = 0.f;
1055}
