22class OrderParticipant final : public transaction::ITransactionParticipant {
23public:
24 OrderParticipant(orders::CommandQueue& queue, std::string kind, int priority, double timeout, std::string product,
25 std::string owner)
26 : queue_(queue),
27 kind_(std::move(kind)),
28 priority_(priority),
29 timeout_(timeout),
30 product_(std::move(product)),
31 owner_(std::move(owner)) {}
32
33 [[nodiscard]] std::string_view name() const noexcept override { return "rts-build-order"; }
34
35 [[nodiscard]] eve::Result<void> prepare(const transaction::TransactionContext&) override {
36 if (prepared_ || committed_)
37 return failure(eve::DiagnosticCode::Conflict, "RTS order participant is already in flight", "orders");
38 try {
39 before_ = queue_;
40 staged_ = queue_;
41 } catch (const std::exception& exception) {
42 return failure(eve::DiagnosticCode::Failed, std::string("failed to stage RTS order: ") + exception.what(),
43 "orders");
44 }
45 auto id = staged_->append(kind_, priority_, timeout_);
46 if (!id) return eve::Result<void>::failure(id.status());
47 orderId_ = std::move(id).takeValue();
48 auto order = staged_->find(orderId_);
49 if (!order) return failure(eve::DiagnosticCode::InvariantViolation, "staged RTS order disappeared", "orders");
50 order->get().payload.setString("owner", owner_);
51 order->get().payload.setString("product", product_);
52 prepared_ = true;
53 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
54 }
55
56 [[nodiscard]] eve::Result<void> commit(const transaction::TransactionContext&) override {
57 if (!prepared_ || committed_)
58 return failure(eve::DiagnosticCode::Conflict, "RTS order participant has no prepared stage", "orders");
59 try {
60 queue_ = *staged_;
61 } catch (const std::exception& exception) {
62 return failure(eve::DiagnosticCode::Failed, std::string("failed to publish RTS order: ") + exception.what(),
63 "orders");
64 }
65 committed_ = true;
66 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
67 }
68
69 [[nodiscard]] eve::Result<void> rollback(const transaction::TransactionContext&) override {
70 if (committed_)
71 return failure(eve::DiagnosticCode::Conflict, "committed RTS order requires compensation", "orders");
72 staged_.reset();
73 prepared_ = false;
74 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
75 }
76
77 [[nodiscard]] eve::Result<void> compensate(const transaction::TransactionContext&) override {
78 if (!committed_)
79 return failure(eve::DiagnosticCode::Conflict, "RTS order has no committed state to compensate", "orders");
80 try {
81 queue_ = *before_;
82 } catch (const std::exception& exception) {
83 return failure(eve::DiagnosticCode::Failed,
84 std::string("failed to compensate RTS order: ") + exception.what(), "orders");
85 }
86 staged_.reset();
87 before_.reset();
88 prepared_ = false;
89 committed_ = false;
90 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
91 }
92
93 [[nodiscard]] const std::string& orderId() const noexcept { return orderId_; }

205public:
206 ActionParticipant(action::ActionRuntime& runtime, action::ActionDefinition definition,
207 action::ActionRequest request, SimulationTick tick, Duration delta)
208 : runtime_(runtime),
209 definition_(std::move(definition)),
210 request_(std::move(request)),
211 tick_(tick),
212 delta_(delta) {}
213
214 [[nodiscard]] std::string_view name() const noexcept override { return "rts-build-action"; }
215
216 [[nodiscard]] eve::Result<void> prepare(const transaction::TransactionContext&) override {
217 if (execution_)
218 return failure(eve::DiagnosticCode::Conflict, "RTS Action participant is already in flight", "action");
219 auto submitted = runtime_.submit(std::move(definition_), std::move(request_));
220 if (!submitted) return eve::Result<void>::failure(submitted.status());
221 execution_ = std::move(submitted).takeValue();
222 prepared_ = true;
223 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
224 }
225
226 [[nodiscard]] eve::Result<void> commit(const transaction::TransactionContext&) override {
227 if (!prepared_ || committed_ || !execution_)
228 return failure(eve::DiagnosticCode::Conflict, "RTS Action participant has no prepared execution", "action");
229 auto advanced = runtime_.advance(*execution_, tick_, delta_);
230 if (!advanced) return eve::Result<void>::failure(advanced.status());
231 std::move(advanced).takeValue();
232 committed_ = true;
233 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
234 }
235
236 [[nodiscard]] eve::Result<void> rollback(const transaction::TransactionContext&) override {
237 if (committed_)
238 return failure(eve::DiagnosticCode::Conflict, "committed RTS Action requires compensation", "action");
239 if (!execution_) return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
240 const auto* current = runtime_.find(*execution_);
241 if (current != nullptr && current->phase() != action::ActionPhase::Failed &&
242 current->phase() != action::ActionPhase::Cancelled && current->phase() != action::ActionPhase::Completed) {
243 auto cancelled = runtime_.cancel(*execution_, tick_);
244 if (!cancelled) return cancelled;
245 }
246 prepared_ = false;
247 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
248 }
249
250 [[nodiscard]] eve::Result<void> compensate(const transaction::TransactionContext&) override {
251 // ActionRuntime's terminal record is an audit record. It has no
252 // mutable business state once the other participants are compensated.
253 committed_ = false;
254 prepared_ = false;
255 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
256 }
257
258 [[nodiscard]] action::ActionExecutionId executionId() const noexcept {
259 return execution_.value_or(action::ActionExecutionId{});
260 }

263 action::ActionRuntime& runtime_;
264 action::ActionDefinition definition_;

266 SimulationTick tick_;
267 Duration delta_;

269 bool prepared_ = false;
270 bool committed_ = false;

308eve::Result<RTSBuildReceipt> RTSProductionActionAdapter::build(RTSBuildRequest request) {
309 if (request.production == nullptr || request.orders == nullptr || request.action == nullptr ||
310 request.account == nullptr)
311 return failure<RTSBuildReceipt>(eve::DiagnosticCode::InvalidArgument,
312 "RTS build requires production, orders, Action and account ports", "request");
313 if (request.owner.empty() || request.productionKind.empty() || request.product.empty())
314 return failure<RTSBuildReceipt>(eve::DiagnosticCode::InvalidArgument,
315 "RTS build requires owner, kind and product", "request");
316 if (!request.cost.isValid())
317 return failure<RTSBuildReceipt>(eve::DiagnosticCode::InvalidArgument,
318 "RTS build requires a validated resource cost", "cost");
319 if (request.duration.nanoseconds() <= 0 || request.actionDelta.nanoseconds() < 0)
320 return failure<RTSBuildReceipt>(eve::DiagnosticCode::InvalidArgument, "RTS build durations are invalid",
321 "duration");
322
323 std::vector<resource::ResourceCost> protectedItems = request.cost.items();
324 for (const auto& reserve : request.resourceReserves) {
325 if (!reserve.resource.resource.isValid() || reserve.resource.amount.isZero())
326 return failure<RTSBuildReceipt>(eve::DiagnosticCode::InvalidArgument,
327 "RTS production reserve must contain a positive resource amount",
328 "resourceReserves");
329 if (request.priority < reserve.minimumPriority) protectedItems.push_back(reserve.resource);
330 }
331 auto protectedCost = resource::CostSpec::create(std::move(protectedItems));
332 if (!protectedCost) return eve::Result<RTSBuildReceipt>::failure(protectedCost.status());
333 auto protectedAffordability = request.account->canAfford(protectedCost.value());
334 if (!protectedAffordability)
335 return eve::Result<RTSBuildReceipt>::failure(protectedAffordability.status());
336 if (!protectedAffordability.value().isAffordable())
337 return failure<RTSBuildReceipt>(eve::DiagnosticCode::PreconditionViolation,
338 "RTS production would consume a protected resource reserve",
339 "resourceReserves");
340
341 if (!request.actionDefinition.id.isValid()) {
342 auto id = eve::LogicalId::fromParts("rts", "build." + request.product);
343 if (!id)
344 return failure<RTSBuildReceipt>(eve::DiagnosticCode::InvalidArgument,
345 "RTS product cannot form an Action id", "product");
346 request.actionDefinition.id = *id;
347 }
348 if (request.actionDefinition.cost || request.actionDefinition.activeExecutionRequired ||
349 !request.actionDefinition.effectIds.empty())
350 return failure<RTSBuildReceipt>(eve::DiagnosticCode::InvalidArgument,
351 "RTS build Action must be effect-free; effect is a transaction port",
352 "actionDefinition");
353 request.actionDefinition.timing = {};
354 if (!request.actionRequest.actionId.isValid()) request.actionRequest.actionId = request.actionDefinition.id;
355 if (request.actionRequest.actionId != request.actionDefinition.id)
356 return failure<RTSBuildReceipt>(eve::DiagnosticCode::InvalidArgument,
357 "RTS build Action request does not match definition", "actionRequest");
358 request.actionRequest.requestedTick = request.tick;
359
360 if (request.transactionId.empty()) request.transactionId = "rts.build." + request.product;
361 transaction::TransactionContext context(std::move(request.transactionId));
362 OrderParticipant order(*request.orders, request.orderKind, request.orderPriority, request.orderTimeoutSeconds,
363 request.product, request.owner);
364 ProductionParticipant production(*request.production, request.owner, request.productionKind, request.product,
365 std::move(request.context), request.duration, request.priority);
366 ActionParticipant action(*request.action, std::move(request.actionDefinition), std::move(request.actionRequest),
367 request.tick, request.actionDelta);
368 std::vector<transaction::ITransactionParticipant*> participants;
369 participants.reserve(request.effect ? 4u : 3u);
370 participants.push_back(&order);
371 participants.push_back(&production);
372 if (request.effect != nullptr) participants.push_back(request.effect);
373 participants.push_back(&action);
374
375 auto committed = transaction::AtomicResourcePayment::execute(
376 context, *request.account, request.cost, std::span<transaction::ITransactionParticipant*>(participants));
377 if (!committed) return eve::Result<RTSBuildReceipt>::failure(committed.status());
378 auto receipt = std::move(committed).takeValue();
379 RTSBuildReceipt result{std::move(receipt), order.orderId(), production.taskId(), action.executionId()};
380 return eve::Result<RTSBuildReceipt>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
381}

402eve::Result<RTSCancelProductionReceipt> RTSProductionActionAdapter::cancel(
403 RTSCancelProductionRequest request) {
404 if (request.production == nullptr || request.orders == nullptr || request.account == nullptr)
405 return failure<RTSCancelProductionReceipt>(eve::DiagnosticCode::InvalidArgument,
406 "RTS production cancellation requires production, orders and account ports", "request");
407 if (!request.refund.isValid() || request.productionTaskId.empty() || request.orderId.empty() ||
408 request.reason.empty())
409 return failure<RTSCancelProductionReceipt>(eve::DiagnosticCode::InvalidArgument,
410 "RTS production cancellation requires task, order, refund and reason", "request");
411
412 auto productionBefore = request.production->snapshot();
413 if (!productionBefore)
414 return eve::Result<RTSCancelProductionReceipt>::failure(productionBefore.status());
415 production::WorkQueue stagedProduction;
416 auto stagedRestore = stagedProduction.restore(productionBefore.value());
417 if (!stagedRestore)
418 return eve::Result<RTSCancelProductionReceipt>::failure(stagedRestore.status());
419 auto cancelledTask = stagedProduction.cancel(request.productionTaskId, request.reason);
420 if (!cancelledTask)
421 return eve::Result<RTSCancelProductionReceipt>::failure(cancelledTask.status());
422
423 orders::CommandQueue ordersBefore;
424 orders::CommandQueue stagedOrders;
425 try {
426 ordersBefore = *request.orders;
427 stagedOrders = ordersBefore;
428 } catch (const std::exception& exception) {
429 return failure<RTSCancelProductionReceipt>(eve::DiagnosticCode::Failed,
430 std::string("failed to stage RTS cancellation orders: ") + exception.what(), "orders");
431 }
432 auto cancelledOrder = stagedOrders.cancel(request.orderId, request.reason);
433 if (!cancelledOrder)
434 return eve::Result<RTSCancelProductionReceipt>::failure(cancelledOrder.status());
435 auto productionAfter = stagedProduction.snapshot();
436 if (!productionAfter)
437 return eve::Result<RTSCancelProductionReceipt>::failure(productionAfter.status());
438
439 auto publishedProduction = request.production->restore(productionAfter.value());
440 if (!publishedProduction)
441 return eve::Result<RTSCancelProductionReceipt>::failure(publishedProduction.status());
442 try {
443 *request.orders = stagedOrders;
444 } catch (const std::exception& exception) {
445 auto restored = request.production->restore(productionBefore.value());
446 restored.ignore("best-effort production cancellation rollback");
447 return failure<RTSCancelProductionReceipt>(eve::DiagnosticCode::Failed,
448 std::string("failed to publish RTS cancellation orders: ") + exception.what(), "orders");
449 }
450
451 auto credited = request.account->credit(request.refund);
452 if (!credited) {
453 auto restoredProduction = request.production->restore(productionBefore.value());
454 try {
455 *request.orders = ordersBefore;
456 } catch (...) {
457 return failure<RTSCancelProductionReceipt>(eve::DiagnosticCode::InvariantViolation,
458 "RTS cancellation refund failed and order rollback could not be restored", "orders");
459 }
460 if (!restoredProduction)
461 return failure<RTSCancelProductionReceipt>(eve::DiagnosticCode::InvariantViolation,
462 "RTS cancellation refund failed and production rollback could not be restored", "production");
463 return eve::Result<RTSCancelProductionReceipt>::failure(credited.status());
464 }
465
466 RTSCancelProductionReceipt result{std::move(credited).takeValue(),
467 std::move(request.productionTaskId), std::move(request.orderId)};
468 return eve::Result<RTSCancelProductionReceipt>::success(
469 std::move(result), eve::Status::success(eve::StatusCode::Applied));
470}
