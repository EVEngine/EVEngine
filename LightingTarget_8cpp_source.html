120EditorResult<void> LightingPropertyTargetBase::applyDomainOperation(const DomainOperation& operation) {
121 if (operation.target != TargetId(id_))
122 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.lighting.target-mismatch"),
123 "Lighting operation targets another document");
124 if (operation.type != "lighting.property.set.v1")
125 return eve::editing::failed<void>(EditorStatus::Unsupported, RuleId("editor.lighting.operation-unsupported"),
126 "Lighting operation is unsupported: " + operation.type);
127 const EditorValue* pathValue = field(operation.payload, "path");
128 const EditorValue* assigned = field(operation.payload, "value");
129 const auto* path = pathValue ? pathValue->getIf<std::string>() : nullptr;
130 auto descriptor = path ? schema_.find(PropertyPath(*path)) : PropertyDescriptorLookup{};
131 if (!descriptor || !assigned)
132 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.lighting.invalid-operation"),
133 "Lighting operation requires a known path and value");
134 auto valid = validatePropertyValue(*descriptor, *assigned);
135 if (!valid.ok()) return valid;
136 values_[*path] = *assigned;
137 bumpRevision();
138 widenDirty(0, 0);
139 return eve::editing::applied<void>();
140}

160EditorResult<DomainOperation> LightingPropertyTargetBase::makeSet(const SelectionSnapshot& selection,
161 const PropertyPath& path, const EditorValue& assigned,
162 PropertySetMode mode) const {
163 if (!selectionMatches(selection))
164 return eve::editing::failed<DomainOperation>(EditorStatus::Rejected, RuleId("editor.lighting.selection"),
165 "Selection does not belong to this lighting target");
166 if (mode == PropertySetMode::Reset) return makeReset(selection, path);
167 if (mode != PropertySetMode::Absolute)
168 return eve::editing::failed<DomainOperation>(EditorStatus::Unsupported, RuleId("editor.lighting.set-mode"),
169 "Lighting properties require absolute assignment");
170 auto descriptor = schema_.find(path);
171 if (!descriptor)
172 return eve::editing::failed<DomainOperation>(EditorStatus::Unsupported,
173 RuleId("editor.lighting.property-unsupported"),
174 "Lighting property is unknown: " + path.value());
175 auto valid = validatePropertyValue(*descriptor, assigned);
176 if (!valid.ok()) return EditorResult<DomainOperation>::failure(valid.status());
177 const auto previous = values_.find(path.value());
178 EditorValue::Object payload{{"path", path.value()}, {"value", assigned}};
179 EditorValue::Object inverse{{"path", path.value()}, {"value", previous->second}};
180 DomainOperation operation;
181 operation.type = "lighting.property.set.v1";
182 operation.target = TargetId(id_);
183 operation.payload = EditorValue(std::move(payload));
184 operation.inverse = EditorValue(std::move(inverse));
185 operation.hasInverse = true;
186 operation.affectedProperties.push_back(path.value());
187 operation.mergeKey = "lighting:" + id_ + ":" + path.value();
188 return eve::editing::applied<DomainOperation>(std::move(operation));
189}

207EditorResult<void> LightingPropertyTargetBase::loadSnapshot(const EditorValue& snapshot) {
208 const EditorValue* versionValue = field(snapshot, "schemaVersion");
209 const EditorValue* propertiesValue = field(snapshot, "properties");
210 const auto* version = versionValue ? versionValue->getIf<int64_t>() : nullptr;
211 const auto* properties = propertiesValue ? propertiesValue->getIf<EditorValue::Object>() : nullptr;
212 if (!version || *version != 1 || !properties)
213 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.lighting.invalid-snapshot"),
214 "Lighting snapshot requires schema version one and properties");
215 std::map<std::string, EditorValue> candidate;
216 for (const PropertyDescriptor& descriptor : schema_.properties)
217 candidate[descriptor.path.value()] = descriptor.defaultValue;
218 for (const auto& [path, assigned] : *properties) {
219 auto descriptor = schema_.find(PropertyPath(path));
220 if (!descriptor)
221 return eve::editing::failed<void>(EditorStatus::Unsupported, RuleId("editor.lighting.snapshot-property"),
222 "Lighting snapshot contains an unknown property: " + path);
223 auto valid = validatePropertyValue(*descriptor, assigned);
224 if (!valid.ok()) return valid;
225 candidate[path] = assigned;
226 }
227 values_ = std::move(candidate);
228 bumpRevision();
229 clearDirtyRegion();
230 return eve::editing::applied<void>();
231}

245std::vector<EditorDiagnostic> Light3DDocumentTarget::validate() const {
246 std::vector<EditorDiagnostic> diagnostics;
247 const auto* type = value("light.type")->getIf<std::string>();
248 const auto* direction = value("transform.direction")->getIf<EditorValue::Array>();
249 const auto* castShadow = value("shadow.cast")->getIf<bool>();
250 if (direction && direction->size() == 3) {
251 double lengthSquared = 0.0;
252 for (const EditorValue& component : *direction) {
253 const auto* number = component.getIf<double>();
254 if (number) lengthSquared += *number * *number;
255 }
256 if (lengthSquared <= 1e-12)
257 diagnostics.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument,
258 RuleId("editor.light.zero-direction"), DiagnosticSeverity::Error, "Light direction cannot be zero"));
259 }
260 if (type && *type == "point" && castShadow && *castShadow)
261 diagnostics.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::Unsupported,
262 RuleId("editor.light.point-shadow-unsupported"), DiagnosticSeverity::Warning,
263 "The current renderer only selects directional shadow casters"));
264 return diagnostics;
265}

283 "DayNight and Weather cannot both own the environment"));
