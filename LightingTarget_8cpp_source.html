125EditorResult<void> LightingPropertyTargetBase::applyDomainOperation(const DomainOperation& operation) {
126 if (operation.target != TargetId(id_))
127 return lightingError<void>(EditorStatus::Rejected, "editor.lighting.target-mismatch",
128 "Lighting operation targets another document");
129 if (operation.type != "lighting.property.set.v1")
130 return lightingError<void>(EditorStatus::Unsupported, "editor.lighting.operation-unsupported",
131 "Lighting operation is unsupported: " + operation.type);
132 const EditorValue* pathValue = field(operation.payload, "path");
133 const EditorValue* assigned = field(operation.payload, "value");
134 const auto* path = pathValue ? pathValue->getIf<std::string>() : nullptr;
135 auto descriptor = path ? schema_.find(PropertyPath(*path)) : PropertyDescriptorLookup{};
136 if (!descriptor || !assigned)
137 return lightingError<void>(EditorStatus::Rejected, "editor.lighting.invalid-operation",
138 "Lighting operation requires a known path and value");
139 auto valid = validatePropertyValue(*descriptor, *assigned);
140 if (!valid.ok()) return valid;
141 values_[*path] = *assigned;
142 ++revision_;
143 dirty_.include(0, 0);
144 return eve::editing::applied<void>();
145}

165EditorResult<DomainOperation> LightingPropertyTargetBase::makeSet(const SelectionSnapshot& selection,
166 const PropertyPath& path, const EditorValue& assigned,
167 PropertySetMode mode) const {
168 if (!selectionMatches(selection))
169 return lightingError<DomainOperation>(EditorStatus::Rejected, "editor.lighting.selection",
170 "Selection does not belong to this lighting target");
171 if (mode == PropertySetMode::Reset) return makeReset(selection, path);
172 if (mode != PropertySetMode::Absolute)
173 return lightingError<DomainOperation>(EditorStatus::Unsupported, "editor.lighting.set-mode",
174 "Lighting properties require absolute assignment");
175 auto descriptor = schema_.find(path);
176 if (!descriptor)
177 return lightingError<DomainOperation>(EditorStatus::Unsupported, "editor.lighting.property-unsupported",
178 "Lighting property is unknown: " + path.value());
179 auto valid = validatePropertyValue(*descriptor, assigned);
180 if (!valid.ok()) return EditorResult<DomainOperation>::failure(valid.status());
181 const auto previous = values_.find(path.value());
182 EditorValue::Object payload{{"path", path.value()}, {"value", assigned}};
183 EditorValue::Object inverse{{"path", path.value()}, {"value", previous->second}};
184 DomainOperation operation;
185 operation.type = "lighting.property.set.v1";
186 operation.target = TargetId(id_);
187 operation.payload = EditorValue(std::move(payload));
188 operation.inverse = EditorValue(std::move(inverse));
189 operation.hasInverse = true;
190 operation.affectedProperties.push_back(path.value());
191 operation.mergeKey = "lighting:" + id_ + ":" + path.value();
192 return eve::editing::applied<DomainOperation>(std::move(operation));
193}

210EditorResult<void> LightingPropertyTargetBase::loadSnapshot(const EditorValue& snapshot) {
211 const EditorValue* versionValue = field(snapshot, "schemaVersion");
212 const EditorValue* propertiesValue = field(snapshot, "properties");
213 const auto* version = versionValue ? versionValue->getIf<int64_t>() : nullptr;
214 const auto* properties = propertiesValue ? propertiesValue->getIf<EditorValue::Object>() : nullptr;
215 if (!version || *version != 1 || !properties)
216 return lightingError<void>(EditorStatus::Rejected, "editor.lighting.invalid-snapshot",
217 "Lighting snapshot requires schema version one and properties");
218 std::map<std::string, EditorValue> candidate;
219 for (const PropertyDescriptor& descriptor : schema_.properties)
220 candidate[descriptor.path.value()] = descriptor.defaultValue;
221 for (const auto& [path, assigned] : *properties) {
222 auto descriptor = schema_.find(PropertyPath(path));
223 if (!descriptor)
224 return lightingError<void>(EditorStatus::Unsupported, "editor.lighting.snapshot-property",
225 "Lighting snapshot contains an unknown property: " + path);
226 auto valid = validatePropertyValue(*descriptor, assigned);
227 if (!valid.ok()) return valid;
228 candidate[path] = assigned;
229 }
230 values_ = std::move(candidate);
231 ++revision_;
232 dirty_.clear();
233 return eve::editing::applied<void>();
234}

248std::vector<EditorDiagnostic> Light3DDocumentTarget::validate() const {
249 std::vector<EditorDiagnostic> diagnostics;
250 const auto* type = value("light.type")->getIf<std::string>();
251 const auto* direction = value("transform.direction")->getIf<EditorValue::Array>();
252 const auto* castShadow = value("shadow.cast")->getIf<bool>();
253 if (direction && direction->size() == 3) {
254 double lengthSquared = 0.0;
255 for (const EditorValue& component : *direction) {
256 const auto* number = component.getIf<double>();
257 if (number) lengthSquared += *number * *number;
258 }
259 if (lengthSquared <= 1e-12)
260 diagnostics.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument,
261 RuleId("editor.light.zero-direction"), DiagnosticSeverity::Error, "Light direction cannot be zero"));
262 }
263 if (type && *type == "point" && castShadow && *castShadow)
264 diagnostics.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::Unsupported,
265 RuleId("editor.light.point-shadow-unsupported"), DiagnosticSeverity::Warning,
266 "The current renderer only selects directional shadow casters"));
267 return diagnostics;
268}

273std::vector<EditorDiagnostic> EnvironmentDocumentTarget::validate() const {
274 std::vector<EditorDiagnostic> diagnostics;

276 const auto* lightning = value("weather.lightning")->getIf<bool>();

279 diagnostics.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::PreconditionViolation,
280 RuleId("editor.environment.lightning-preset"), DiagnosticSeverity::Warning,
281 "Lightning is enabled outside the storm preset"));
282 const auto* weatherOwned = value("weather.environment-enabled")->getIf<bool>();
283 if (mode && *mode == "daynight" && weatherOwned && *weatherOwned)
284 diagnostics.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::Conflict,
285 RuleId("editor.environment.multiple-owners"), DiagnosticSeverity::Error,
286 "DayNight and Weather cannot both own the environment"));
287 return diagnostics;
