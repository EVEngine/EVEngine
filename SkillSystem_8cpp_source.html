95bool SkillSystem::canCast(RPGActor *actor, const std::string &skillId, std::string *reason) {
96 auto fail = [&](const std::string &r) {
97 if (reason) *reason = r;
98 return false;
99 };
100 if (!actor) return fail("no_actor");
101 const SkillDefinition *def = SkillRegistry::find(skillId);
102 if (!def) return fail("unknown_skill");
103 if (!knows(actor, skillId)) return fail("not_learned");
104 if (isCasting(actor)) return fail("already_casting");
105 if (getCooldownRemaining(actor, skillId) > 0.f) return fail("on_cooldown");
106 if (!checkCostsAffordable(actor, *def)) return fail("insufficient_cost");
107
108 for (auto &kv : conditions()) {
109 std::string r;
110 if (!kv.second(actor, *def, r)) return fail(r.empty() ? kv.first : r);
111 }
112 const auto condition = SkillConditionAdapter::evaluate(actor, *def, def->castCondition);
113 if (!condition.passed()) return fail(decision::conditionReasonCodeName(condition.reasonCode()));
114 return true;
115}

117bool SkillSystem::beginCast(RPGActor *actor, const std::string &skillId, RPGActor *target,
118 std::string *reason) {
119 if (!canCast(actor, skillId, reason)) return false;
120 const SkillDefinition *def = SkillRegistry::find(skillId);
121
122 deductCosts(actor, *def);
123 setCooldownRemaining(actor, skillId, def->cooldown);
124
125 if (def->castTime <= 0.f) {
126 resolveCast(actor, *def, target);
127 return true;
128 }
129
130 CastingState &cs = actor->skills()->casting;
131 cs.active = true;
132 cs.skillId = skillId;
133 cs.remaining = def->castTime;
134 cs.totalCastTime = def->castTime;
135 cs.target = target;
136 return true;
137}

150float SkillSystem::getCastProgress(RPGActor *actor) {
151 if (!actor) return 0.f;
152 const CastingState &cs = actor->skills()->casting;
153 if (!cs.active || cs.totalCastTime <= 0.f) return 0.f;
154 float p = 1.f - (cs.remaining / cs.totalCastTime);

158void SkillSystem::update(float dt) {
159 for (RPGActor *actor : RPGActor::liveActors()) {
160 for (auto &kv : actor->skills()->known)
161 if (kv.second.cooldownRemaining > 0.f) kv.second.cooldownRemaining -= dt;
162
163 CastingState &cs = actor->skills()->casting;
164 if (!cs.active) continue;
165 cs.remaining -= dt;
166 if (cs.remaining <= 0.f) {
167 const SkillDefinition *def = SkillRegistry::find(cs.skillId);
168 RPGActor *target = cs.target;
169 cs.active = false; // clear before resolveCast in case it re-enters casting
170 if (def) resolveCast(actor, *def, target);
171 }
172 }
173}
