83bool SkillSystem::canCast(RPGActor *actor, const std::string &skillId, std::string *reason) {
84 auto fail = [&](const std::string &r) {
85 if (reason) *reason = r;
86 return false;
87 };
88 if (!actor) return fail("no_actor");
89 const SkillDefinition *def = SkillRegistry::find(skillId);
90 if (!def) return fail("unknown_skill");
91 if (!knows(actor, skillId)) return fail("not_learned");
92 if (isCasting(actor)) return fail("already_casting");
93 if (getCooldownRemaining(actor, skillId) > 0.f) return fail("on_cooldown");
94 if (!checkCostsAffordable(actor, *def)) return fail("insufficient_cost");
95
96 for (auto &kv : conditions()) {
97 std::string r;
98 if (!kv.second(actor, *def, r)) return fail(r.empty() ? kv.first : r);
99 }
100 return true;
101}

103bool SkillSystem::beginCast(RPGActor *actor, const std::string &skillId, RPGActor *target,
104 std::string *reason) {
105 if (!canCast(actor, skillId, reason)) return false;
106 const SkillDefinition *def = SkillRegistry::find(skillId);
107
108 deductCosts(actor, *def);
109 setCooldownRemaining(actor, skillId, def->cooldown);
110
111 if (def->castTime <= 0.f) {
112 resolveCast(actor, *def, target);
113 return true;
114 }
115
116 CastingState &cs = actor->skills()->casting;
117 cs.active = true;
118 cs.skillId = skillId;
119 cs.remaining = def->castTime;
120 cs.totalCastTime = def->castTime;
121 cs.target = target;
122 return true;
123}

136float SkillSystem::getCastProgress(RPGActor *actor) {

144void SkillSystem::update(float dt) {
145 for (RPGActor *actor : RPGActor::liveActors()) {
146 for (auto &kv : actor->skills()->known)
147 if (kv.second.cooldownRemaining > 0.f) kv.second.cooldownRemaining -= dt;
148
149 CastingState &cs = actor->skills()->casting;
150 if (!cs.active) continue;
151 cs.remaining -= dt;
152 if (cs.remaining <= 0.f) {
153 const SkillDefinition *def = SkillRegistry::find(cs.skillId);
154 RPGActor *target = cs.target;
155 cs.active = false; // clear before resolveCast in case it re-enters casting
156 if (def) resolveCast(actor, *def, target);
157 }
158 }
159}
