20static void b64_encode_block(char in[3], char out[4], int len)
21{
22 out[0] = (char) cb64[(int)((in[0] & 0xfc) >> 2)];
23 out[1] = (char) cb64[(int)(((in[0] & 0x03) << 4) | ((in[1] & 0xf0) >> 4))];
