73bool unpackEvModel(const void *data, size_t size, const uint8_t **payload, size_t *payloadSize,
74 std::string *hint) {
75 if (!data || size < 6 || std::memcmp(data, kEvModelMagic, 4) != 0) return false;
76 const auto *bytes = static_cast<const uint8_t *>(data);
77 const uint16_t n = uint16_t(bytes[4]) | (uint16_t(bytes[5]) << 8u);
78 if (n == 0 || size < size_t(6u + n)) return false;
79 hint->assign(reinterpret_cast<const char *>(bytes + 6), n);
80 *payload = bytes + 6 + n;
81 *payloadSize = size - 6 - n;
82 return *payloadSize > 0;
83}

91ModelData *Model3D::newModelData(Data *data, std::string hintExt,
92 const ModelLoadOptions &options) {
93 if (data == nullptr || data->getData() == nullptr || data->getSize() == 0)
94 throw eve::Exception("Cannot decode empty model data");
95
96 std::string hint = ensureDotExt(std::move(hintExt));
97 if (hint.empty()) {
98 if (auto *fd = dynamic_cast<filesystem::FileData *>(data))
99 hint = ensureDotExt(fd->getExtension());
100 }
101
102 const void *encoded = data->getData();
103 size_t encodedSize = data->getSize();
104 if (hint == ".evmodel") {
105 const uint8_t *payload = nullptr;
106 std::string packedHint;
107 if (!unpackEvModel(encoded, encodedSize, &payload, &encodedSize, &packedHint))
108 throw eve::Exception("Invalid or truncated .evmodel envelope");
109 encoded = payload;
110 hint = ensureDotExt(std::move(packedHint));
111 }
112
113 // Prefer VFS for sidecar resolution when FileData carries a filename.
114 filesystem::Filesystem *fs = ModuleManager::getInstance<filesystem::Filesystem>("Filesystem");
115 if (!fs)
116 fs = filesystem::Filesystem::create();
117
118 EveFileSystem eveFs(fs);
119 medialoader::ModelLoader loader(&eveFs);
120
121 auto scene = loadOrThrow(loader, encoded, encodedSize, hint.c_str(),
122 toMedialoader(options));
123 if (scene.empty())
124 throw eve::Exception("Could not decode model data");
125
126 std::string uri;
127 if (auto *fd = dynamic_cast<filesystem::FileData *>(data))
128 uri = "file://" + fd->getFilename();
129
130 return new ModelData(std::move(scene), std::move(uri));
131}

159bool Model3D::bakeModel(const std::string &sourcePath, const std::string &destinationPath) {
160 if (sourcePath.empty() || destinationPath.empty()) return false;
161 filesystem::Filesystem *fs = ModuleManager::getInstance<filesystem::Filesystem>("Filesystem");
162 if (!fs) fs = filesystem::Filesystem::create();
163 filesystem::FileData *sourceRaw = fs->read(sourcePath);
164 if (!sourceRaw) return false;
165 eve::ref<filesystem::FileData> source(sourceRaw);
166 if (!source->getData() || source->getSize() == 0) return false;
167 const std::string hint = ensureDotExt(source->getExtension());
168 if (hint.empty() || hint == ".evmodel" || hint.size() > 65535u) return false;
169 std::vector<uint8_t> packed;
170 packed.reserve(6u + hint.size() + source->getSize());
171 packed.insert(packed.end(), kEvModelMagic, kEvModelMagic + 4);
172 const uint16_t n = static_cast<uint16_t>(hint.size());
173 packed.push_back(static_cast<uint8_t>(n & 0xffu));
174 packed.push_back(static_cast<uint8_t>((n >> 8u) & 0xffu));
175 packed.insert(packed.end(), hint.begin(), hint.end());
176 const auto *bytes = static_cast<const uint8_t *>(source->getData());
177 packed.insert(packed.end(), bytes, bytes + source->getSize());
178 try {
179 fs->write(destinationPath, packed.data(), static_cast<int64_t>(packed.size()));
180 return true;
181 } catch (...) {
182 return false;
183 }
