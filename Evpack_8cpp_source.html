37 for (unsigned shift = 0; shift != 32; shift += 8) bytes.push_back(static_cast<std::uint8_t>(value >> shift));
38}

79bool validText(std::string_view value, const EvpackLimits& limits) {
80 if (value.empty() || value.size() > limits.maximumStringBytes ||
81 !isValidUtf8(value, Utf8NullPolicy::Reject))
82 return false;
83 for (unsigned char byte : value)
84 if (byte == 0 || byte < 0x20 || byte == 0x7f) return false;
85 return true;

89 if (values.size() > limits.maximumStringBytes) return false;
90 std::set<std::string> unique;
91 for (const auto& value : values)
92 if (!validText(value, limits) || !unique.emplace(value).second) return false;

96class Reader {
97public:
98 Reader(std::span<const std::uint8_t> bytes, std::size_t begin, std::size_t end)
99 : bytes_(bytes), cursor_(begin), end_(end) {}
100
101 bool raw(std::size_t count, std::span<const std::uint8_t>& value) {
102 if (cursor_ > end_ || count > end_ - cursor_) return false;
103 value = bytes_.subspan(cursor_, count);
104 cursor_ += count;
105 return true;
106 }
107 bool u16(std::uint16_t& value) {
108 std::span<const std::uint8_t> data;
109 if (!raw(2, data)) return false;
110 value = std::uint16_t(data[0]) | (std::uint16_t(data[1]) << 8);
111 return true;
112 }
113 bool u32(std::uint32_t& value) {
114 std::span<const std::uint8_t> data;
115 if (!raw(4, data)) return false;
116 value = 0;
117 for (unsigned index = 0; index != 4; ++index) value |= std::uint32_t(data[index]) << (index * 8);
118 return true;
119 }
120 bool u64(std::uint64_t& value) {
121 std::span<const std::uint8_t> data;
122 if (!raw(8, data)) return false;
123 value = 0;
124 for (unsigned index = 0; index != 8; ++index) value |= std::uint64_t(data[index]) << (index * 8);
125 return true;
126 }
127 bool id(PersistentId& value) {
128 std::span<const std::uint8_t> data;
129 if (!raw(16, data)) return false;
130 auto parsed = PersistentId::fromBytes(data);
131 if (!parsed || parsed->isNil()) return false;
132 value = *parsed;
133 return true;
134 }
135 bool string(std::string& value, const EvpackLimits& limits) {
136 std::uint32_t size = 0;
137 std::span<const std::uint8_t> data;
138 if (!u32(size) || size > limits.maximumStringBytes || !raw(size, data)) return false;
139 value.assign(reinterpret_cast<const char*>(data.data()), data.size());
140 return validText(value, limits);
141 }
142 bool strings(std::vector<std::string>& values, const EvpackLimits& limits) {
143 std::uint32_t count = 0;
144 if (!u32(count) || count > limits.maximumStringBytes) return false;
145 values.reserve(count);
146 for (std::uint32_t index = 0; index < count; ++index) {
147 std::string value;
148 if (!string(value, limits)) return false;
149 values.push_back(std::move(value));
150 }
151 return validStrings(values, limits);
152 }
153 [[nodiscard]] std::size_t cursor() const { return cursor_; }
154 [[nodiscard]] bool finished() const { return cursor_ == end_; }
155
156private:
157 std::span<const std::uint8_t> bytes_;
158 std::size_t cursor_;
159 std::size_t end_;
160};

163 std::vector<std::uint8_t> bytes(kManifestMagic.begin(), kManifestMagic.end());
164 put32(bytes, static_cast<std::uint32_t>(variants.size()));
165 put32(bytes, 0);

167 putString(bytes, variant.os); putString(bytes, variant.arch); putString(bytes, variant.graphics);

181std::vector<std::uint8_t> encodeToc(const std::vector<BuildRecord>& records) {
182 std::vector<std::uint8_t> bytes(kTocMagic.begin(), kTocMagic.end());
183 put32(bytes, static_cast<std::uint32_t>(records.size())); put32(bytes, 0);
184 for (const auto& record : records) {
185 const auto& input = record.input;
186 putId(bytes, input.assetId); put64(bytes, input.schemaVersion.value());
187 put32(bytes, input.variantIndex); put16(bytes, static_cast<std::uint16_t>(input.kind));
188 put16(bytes, static_cast<std::uint16_t>(input.codec)); put32(bytes, input.chunkId);
189 put32(bytes, input.alignment);
190 put64(bytes, record.offset); put64(bytes, record.stored.size()); put64(bytes, input.bytes.size());
191 bytes.insert(bytes.end(), record.hash.begin(), record.hash.end());
192 put32(bytes, static_cast<std::uint32_t>(input.dependencies.size()));
193 putString(bytes, input.type);
194 for (const auto& dependency : input.dependencies) putId(bytes, dependency);
195 }
196 return bytes;
197}

203bool variantMatches(const EvpackVariant& variant, const EvpackCapabilities& capabilities) {
204 if (variant.os != capabilities.os || variant.arch != capabilities.arch ||
205 variant.graphics != capabilities.graphics || !contains(capabilities.shaderFormats, variant.shaderFormat) ||
206 !contains(capabilities.qualities, variant.quality)) return false;
207 if (!variant.textureFamilies.empty() &&
208 std::none_of(variant.textureFamilies.begin(), variant.textureFamilies.end(),
209 [&](const auto& family) { return contains(capabilities.textureFamilies, family); })) return false;
210 return std::all_of(variant.features.begin(), variant.features.end(),
211 [&](const auto& feature) { return contains(capabilities.features, feature); });
212}

214Result<std::vector<std::uint8_t>> encodeSignature(const EvpackSignature& signature) {
215 if (signature.algorithm.empty() || signature.algorithm.size() > 15 ||
216 !validText(signature.algorithm, EvpackLimits{}))
217 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
218 "signature algorithm identifier is invalid");
219 std::vector<std::uint8_t> bytes(kSignatureSize, 0);
220 std::copy(kSignatureMagic.begin(), kSignatureMagic.end(), bytes.begin());
221 std::copy(signature.algorithm.begin(), signature.algorithm.end(), bytes.begin() + 8);
222 std::copy(signature.keyId.begin(), signature.keyId.end(), bytes.begin() + 24);
223 std::copy(signature.bytes.begin(), signature.bytes.end(), bytes.begin() + 56);
224 return Result<std::vector<std::uint8_t>>::success(std::move(bytes));
225}

227Result<EvpackSignature> decodeSignature(std::span<const std::uint8_t> bytes) {
228 if (bytes.size() != kSignatureSize ||
229 !std::equal(kSignatureMagic.begin(), kSignatureMagic.end(), bytes.begin()))
230 return packFailure<EvpackSignature>(DiagnosticCode::ParseError,
231 "runtime package signature block is invalid");
232 const auto terminator = std::find(bytes.begin() + 8, bytes.begin() + 24, std::uint8_t(0));
233 if (terminator == bytes.begin() + 8)
234 return packFailure<EvpackSignature>(DiagnosticCode::ParseError,
235 "runtime package signature algorithm is empty");
236 if (std::any_of(terminator, bytes.begin() + 24, [](std::uint8_t value) { return value != 0; }) ||
237 std::any_of(bytes.begin() + 120, bytes.end(), [](std::uint8_t value) { return value != 0; }))
238 return packFailure<EvpackSignature>(DiagnosticCode::ParseError,
239 "runtime package signature padding is not canonical");
240 EvpackSignature signature;
241 signature.algorithm.assign(reinterpret_cast<const char*>(&bytes[8]),
242 static_cast<std::size_t>(terminator - (bytes.begin() + 8)));
243 if (!validText(signature.algorithm, EvpackLimits{}))
244 return packFailure<EvpackSignature>(DiagnosticCode::ParseError,
245 "runtime package signature algorithm is invalid");
246 std::copy_n(bytes.begin() + 24, 32, signature.keyId.begin());
247 std::copy_n(bytes.begin() + 56, 64, signature.bytes.begin());
248 return Result<EvpackSignature>::success(std::move(signature));
249}

253Result<std::vector<std::uint8_t>> buildEvpack(EvpackBuild build, const EvpackLimits& limits) {
254 if (build.packageId.isNil() || build.buildId.isNil())
255 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
256 "packageId and buildId must be non-nil");
257 if (build.variants.empty() || build.variants.size() > limits.maximumVariants ||
258 build.chunks.size() > limits.maximumChunks)
259 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
260 "variant or chunk count is outside limits");
261 for (const auto& variant : build.variants) {
262 if (!validText(variant.os, limits) || !validText(variant.arch, limits) ||
263 !validText(variant.graphics, limits) || !validText(variant.shaderFormat, limits) ||
264 !validText(variant.quality, limits) || !validStrings(variant.textureFamilies, limits) ||
265 !validStrings(variant.features, limits))
266 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
267 "variant contains invalid capability text");
268 }
269
270 std::vector<BuildRecord> records;
271 std::uint64_t decodedTotal = 0;
272 for (auto& input : build.chunks) {
273 if (input.assetId.isNil() || !validText(input.type, limits) || input.schemaVersion.value() == 0 ||
274 input.variantIndex >= build.variants.size() || !powerOfTwo(input.alignment) ||
275 input.alignment > 16 * 1024 * 1024 || input.dependencies.size() > limits.maximumDependencies ||
276 (input.codec != EvpackCodec::None && input.codec != EvpackCodec::Zstd) ||
277 input.bytes.size() > limits.maximumChunkBytes ||
278 input.bytes.size() > limits.maximumDecodedBytes ||
279 decodedTotal > limits.maximumDecodedBytes - input.bytes.size())
280 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
281 "chunk metadata or resource budget is invalid", input.type);
282 for (const auto& dependency : input.dependencies)
283 if (dependency.isNil())
284 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
285 "chunk dependency must be non-nil", input.type);
286 std::sort(input.dependencies.begin(), input.dependencies.end());
287 if (std::adjacent_find(input.dependencies.begin(), input.dependencies.end()) != input.dependencies.end())
288 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::Conflict,
289 "chunk dependencies contain duplicates", input.type);
290 decodedTotal += input.bytes.size();
291 records.push_back({std::move(input), {}, {}, 0});
292 records.back().hash = sha256(records.back().input.bytes);
293 auto compressed = compressEvpackChunk(records.back().input.codec, records.back().input.bytes);
294 if (!compressed) return Result<std::vector<std::uint8_t>>::failure(compressed.status());
295 records.back().stored = std::move(compressed).takeValue();
296 if (records.back().stored.size() > limits.maximumChunkBytes)
297 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
298 "stored chunk exceeds size limits",
299 records.back().input.type);
300 }
301 std::sort(records.begin(), records.end(), [](const BuildRecord& left, const BuildRecord& right) {
302 return std::tie(left.input.assetId, left.input.type, left.input.schemaVersion, left.input.variantIndex,
303 left.input.kind, left.input.chunkId) <
304 std::tie(right.input.assetId, right.input.type, right.input.schemaVersion, right.input.variantIndex,
305 right.input.kind, right.input.chunkId);
306 });
307 for (std::size_t index = 1; index < records.size(); ++index) {
308 const auto key = [](const BuildRecord& value) {
309 return std::tie(value.input.assetId, value.input.type, value.input.schemaVersion,
310 value.input.variantIndex, value.input.kind, value.input.chunkId);
311 };
312 if (key(records[index - 1]) == key(records[index]))
313 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::Conflict,
314 "duplicate runtime chunk key", records[index].input.type);
315 }
316
317 auto manifest = encodeManifest(build.variants);
318 auto preliminaryToc = encodeToc(records);
319 if (manifest.size() > limits.maximumMetadataBytes || preliminaryToc.size() > limits.maximumMetadataBytes ||
320 manifest.size() > limits.maximumMetadataBytes - preliminaryToc.size())
321 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
322 "runtime package metadata exceeds size limits");
323 const std::uint64_t signatureSize = build.signer ? kSignatureSize : 0;
324 std::uint64_t cursor = kHeaderSize + manifest.size() + preliminaryToc.size() + signatureSize;
325 for (auto& record : records) {
326 if (!alignUp(cursor, record.input.alignment, record.offset) ||
327 !checkedAdd(record.offset, record.stored.size(), cursor) || cursor > limits.maximumPackageBytes)
328 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
329 "encoded package exceeds size limits");
330 }
331 auto toc = encodeToc(records);
332 std::vector<std::uint8_t> bytes(static_cast<std::size_t>(cursor), 0);
333 std::copy(manifest.begin(), manifest.end(), bytes.begin() + kHeaderSize);
334 const std::uint64_t tocOffset = kHeaderSize + manifest.size();
335 std::copy(toc.begin(), toc.end(), bytes.begin() + static_cast<std::size_t>(tocOffset));
336 for (const auto& record : records)
337 std::copy(record.stored.begin(), record.stored.end(),
338 bytes.begin() + static_cast<std::size_t>(record.offset));
339
340 std::vector<std::uint8_t> header(kPackMagic.begin(), kPackMagic.end());
341 put32(header, kHeaderSize); put32(header, build.signer ? kSignedFlag : 0);
342 putId(header, build.packageId); putId(header, build.buildId);
343 put64(header, tocOffset); put64(header, toc.size()); put64(header, kHeaderSize); put64(header, manifest.size());
344 put64(header, build.signer ? tocOffset + toc.size() : 0); put64(header, signatureSize);
345 header.resize(kHeaderSize, 0);
346 std::vector<std::uint8_t> metadataHashInput = header;
347 metadataHashInput.insert(metadataHashInput.end(), manifest.begin(), manifest.end());
348 metadataHashInput.insert(metadataHashInput.end(), toc.begin(), toc.end());
349 const auto headerHash = sha256(metadataHashInput);
350 std::copy(headerHash.begin(), headerHash.end(), header.begin() + 96);
351 std::copy(header.begin(), header.end(), bytes.begin());
352 if (build.signer) {
353 auto signature = build.signer->sign(headerHash);
354 if (!signature) return Result<std::vector<std::uint8_t>>::failure(signature.status());
355 auto encoded = encodeSignature(signature.value());
356 if (!encoded) return encoded;
357 std::copy(encoded.value().begin(), encoded.value().end(),
358 bytes.begin() + static_cast<std::size_t>(tocOffset + toc.size()));
359 }
360 return Result<std::vector<std::uint8_t>>::success(std::move(bytes));
361}

363Result<Evpack> parseEvpackMetadata(std::span<const std::uint8_t> bytes, std::uint64_t packageSize,
364 const EvpackLimits& limits, const EvpackTrust& trust) {
365 if (packageSize < kHeaderSize || packageSize > limits.maximumPackageBytes ||
366 bytes.size() < kHeaderSize || bytes.size() > packageSize)
367 return packFailure<Evpack>(DiagnosticCode::ParseError, "package size is outside limits");
368 const bool completePackage = bytes.size() == packageSize;
369 Reader header(bytes, 0, kHeaderSize);
370 std::span<const std::uint8_t> magic;
371 std::uint32_t headerSize = 0, flags = 0;
372 PersistentId packageId, buildId;
373 std::uint64_t tocOffset = 0, tocSize = 0, manifestOffset = 0, manifestSize = 0, signatureOffset = 0,
374 signatureSize = 0;
375 std::span<const std::uint8_t> storedHeaderHash;
376 if (!header.raw(8, magic) || !std::equal(magic.begin(), magic.end(), kPackMagic.begin()) ||
377 !header.u32(headerSize) || headerSize != kHeaderSize || !header.u32(flags) ||
378 (flags & ~kSignedFlag) != 0 ||
379 !header.id(packageId) || !header.id(buildId) || !header.u64(tocOffset) || !header.u64(tocSize) ||
380 !header.u64(manifestOffset) || !header.u64(manifestSize) || !header.u64(signatureOffset) ||
381 !header.u64(signatureSize) ||
382 !header.raw(32, storedHeaderHash) || !header.finished())
383 return packFailure<Evpack>(DiagnosticCode::ParseError, "runtime package header is invalid");
384 const auto validRange = [&](std::uint64_t offset, std::uint64_t size) {
385 return offset >= kHeaderSize && offset <= packageSize && size <= packageSize - offset;
386 };
387 if (!validRange(manifestOffset, manifestSize) || !validRange(tocOffset, tocSize) ||
388 manifestOffset != kHeaderSize || manifestOffset + manifestSize != tocOffset ||
389 manifestSize > limits.maximumMetadataBytes || tocSize > limits.maximumMetadataBytes ||
390 manifestSize > limits.maximumMetadataBytes - tocSize)
391 return packFailure<Evpack>(DiagnosticCode::ParseError, "manifest or TOC range is invalid");
392 const bool signedPackage = (flags & kSignedFlag) != 0;
393 if (signedPackage != (signatureSize != 0) ||
394 (signedPackage && (signatureSize != kSignatureSize || signatureOffset != tocOffset + tocSize ||
395 !validRange(signatureOffset, signatureSize))) ||
396 (!signedPackage && (signatureOffset != 0 || signatureSize != 0)))
397 return packFailure<Evpack>(DiagnosticCode::ParseError, "signature range is invalid");
398 if (tocOffset + tocSize > bytes.size())
399 return packFailure<Evpack>(DiagnosticCode::ParseError, "metadata prefix does not contain the complete TOC");
400 std::vector<std::uint8_t> metadataHashInput(bytes.begin(), bytes.begin() + kHeaderSize);
401 std::fill(metadataHashInput.begin() + 96, metadataHashInput.end(), 0);
402 metadataHashInput.insert(metadataHashInput.end(), bytes.begin() + static_cast<std::size_t>(manifestOffset),
403 bytes.begin() + static_cast<std::size_t>(manifestOffset + manifestSize));
404 metadataHashInput.insert(metadataHashInput.end(), bytes.begin() + static_cast<std::size_t>(tocOffset),
405 bytes.begin() + static_cast<std::size_t>(tocOffset + tocSize));
406 const auto computedMetadataHash = sha256(metadataHashInput);
407 if (!std::equal(storedHeaderHash.begin(), storedHeaderHash.end(), computedMetadataHash.begin()))
408 return packFailure<Evpack>(DiagnosticCode::HashMismatch,
409 "runtime package header and metadata hash does not match");
410
411 Evpack pack;
412 pack.packageId_ = packageId;
413 pack.buildId_ = buildId;
414 if (signedPackage) {
415 if (signatureOffset + signatureSize > bytes.size())
416 return packFailure<Evpack>(DiagnosticCode::ParseError,
417 "metadata prefix does not contain the signature block");
418 auto signature = decodeSignature(bytes.subspan(static_cast<std::size_t>(signatureOffset),
419 static_cast<std::size_t>(signatureSize)));
420 if (!signature) return Result<Evpack>::failure(signature.status());
421 if (!trust.verifier)
422 return packFailure<Evpack>(DiagnosticCode::PreconditionViolation,
423 "signed package requires a configured trust verifier");
424 auto verified = trust.verifier->verify(computedMetadataHash, signature.value());
425 if (!verified) return Result<Evpack>::failure(verified.status());
426 pack.signature_ = signature.value();
427 pack.trustedSignature_ = true;
428 } else if (trust.policy == EvpackSignaturePolicy::RequireTrustedSignature) {
429 return packFailure<Evpack>(DiagnosticCode::PreconditionViolation,
430 "project policy requires a trusted package signature");
431 }
432
433 Reader manifestReader(bytes, static_cast<std::size_t>(manifestOffset),
434 static_cast<std::size_t>(manifestOffset + manifestSize));
435 std::uint32_t variantCount = 0, reserved = 0;
436 if (!manifestReader.raw(8, magic) || !std::equal(magic.begin(), magic.end(), kManifestMagic.begin()) ||
437 !manifestReader.u32(variantCount) || variantCount == 0 || variantCount > limits.maximumVariants ||
438 !manifestReader.u32(reserved) || reserved != 0)
439 return packFailure<Evpack>(DiagnosticCode::ParseError, "binary Cook manifest is invalid");
440 pack.variants_.reserve(variantCount);
441 for (std::uint32_t index = 0; index < variantCount; ++index) {
442 EvpackVariant variant;
443 if (!manifestReader.string(variant.os, limits) || !manifestReader.string(variant.arch, limits) ||
444 !manifestReader.string(variant.graphics, limits) ||
445 !manifestReader.strings(variant.textureFamilies, limits) ||
446 !manifestReader.string(variant.shaderFormat, limits) || !manifestReader.string(variant.quality, limits) ||
447 !manifestReader.strings(variant.features, limits))
448 return packFailure<Evpack>(DiagnosticCode::ParseError, "variant record is invalid");
449 pack.variants_.push_back(std::move(variant));
450 }
451 if (!manifestReader.finished())
452 return packFailure<Evpack>(DiagnosticCode::ParseError, "binary Cook manifest has trailing bytes");
453
454 Reader tocReader(bytes, static_cast<std::size_t>(tocOffset), static_cast<std::size_t>(tocOffset + tocSize));
455 std::uint32_t chunkCount = 0;
456 if (!tocReader.raw(8, magic) || !std::equal(magic.begin(), magic.end(), kTocMagic.begin()) ||
457 !tocReader.u32(chunkCount) || chunkCount > limits.maximumChunks ||
458 !tocReader.u32(reserved) || reserved != 0)
459 return packFailure<Evpack>(DiagnosticCode::ParseError, "runtime package TOC is invalid");
460 std::uint64_t decodedTotal = 0;
461 std::uint64_t previousEnd = signedPackage ? signatureOffset + signatureSize : tocOffset + tocSize;
462 using ChunkKey = std::tuple<PersistentId, std::string, std::uint64_t, std::uint32_t,
463 std::uint16_t, std::uint32_t>;
464 std::optional<ChunkKey> previousKey;
465 for (std::uint32_t index = 0; index < chunkCount; ++index) {
466 EvpackChunk chunk;
467 std::uint16_t kind = 0, codec = 0;
468 std::uint64_t schemaVersion = 0;
469 std::span<const std::uint8_t> hash;
470 std::uint32_t dependencyCount = 0;
471 if (!tocReader.id(chunk.assetId) || !tocReader.u64(schemaVersion) || schemaVersion == 0 ||
472 !tocReader.u32(chunk.variantIndex) ||
473 chunk.variantIndex >= variantCount || !tocReader.u16(kind) || !tocReader.u16(codec) ||
474 !tocReader.u32(chunk.chunkId) || !tocReader.u32(chunk.alignment) || !powerOfTwo(chunk.alignment) ||
475 chunk.alignment > 16 * 1024 * 1024 || !tocReader.u64(chunk.offset) ||
476 !tocReader.u64(chunk.storedSize) || !tocReader.u64(chunk.decodedSize) ||
477 !tocReader.raw(32, hash) || !tocReader.u32(dependencyCount) ||
478 dependencyCount > limits.maximumDependencies || !tocReader.string(chunk.type, limits))
479 return packFailure<Evpack>(DiagnosticCode::ParseError, "runtime chunk record is invalid");
480 chunk.schemaVersion = SchemaVersion(schemaVersion);
481 chunk.kind = static_cast<EvpackChunkKind>(kind);
482 chunk.codec = static_cast<EvpackCodec>(codec);
483 if (chunk.kind < EvpackChunkKind::Definition || chunk.kind > EvpackChunkKind::DebugName ||
484 (chunk.codec != EvpackCodec::None && chunk.codec != EvpackCodec::Zstd) ||
485 (chunk.codec == EvpackCodec::None && chunk.storedSize != chunk.decodedSize) ||
486 chunk.storedSize > limits.maximumChunkBytes || chunk.decodedSize > limits.maximumDecodedBytes ||
487 decodedTotal > limits.maximumDecodedBytes - chunk.decodedSize || chunk.offset % chunk.alignment != 0 ||
488 chunk.offset < previousEnd || chunk.offset > packageSize || chunk.storedSize > packageSize - chunk.offset)
489 return packFailure<Evpack>(DiagnosticCode::Unsupported, "unsupported or unsafe runtime chunk", chunk.type);
490 ChunkKey key{chunk.assetId, chunk.type, chunk.schemaVersion.value(), chunk.variantIndex, kind, chunk.chunkId};
491 if (previousKey && !(*previousKey < key))
492 return packFailure<Evpack>(DiagnosticCode::Conflict, "runtime chunk TOC is not canonical", chunk.type);
493 previousKey = std::move(key);
494 std::copy(hash.begin(), hash.end(), chunk.contentHash.begin());
495 chunk.dependencies.reserve(dependencyCount);
496 for (std::uint32_t dependency = 0; dependency < dependencyCount; ++dependency) {
497 PersistentId id;
498 if (!tocReader.id(id))
499 return packFailure<Evpack>(DiagnosticCode::ParseError, "chunk dependency is invalid", chunk.type);
500 chunk.dependencies.push_back(id);
501 }
502 if (!std::is_sorted(chunk.dependencies.begin(), chunk.dependencies.end()) ||
503 std::adjacent_find(chunk.dependencies.begin(), chunk.dependencies.end()) != chunk.dependencies.end())
504 return packFailure<Evpack>(DiagnosticCode::Conflict, "chunk dependencies are not canonical", chunk.type);
505 if (completePackage) {
506 const auto payload = bytes.subspan(static_cast<std::size_t>(chunk.offset),
507 static_cast<std::size_t>(chunk.storedSize));
508 auto decoded = decompressEvpackChunk(chunk.codec, payload, chunk.decodedSize,
509 limits.maximumDecodedBytes);
510 if (!decoded) return Result<Evpack>::failure(decoded.status());
511 if (sha256(decoded.value()) != chunk.contentHash)
512 return packFailure<Evpack>(DiagnosticCode::HashMismatch, "runtime chunk hash does not match", chunk.type);
513 }
514 decodedTotal += chunk.decodedSize;
515 previousEnd = chunk.offset + chunk.storedSize;
516 pack.chunks_.push_back(std::move(chunk));
517 }
518 if (!tocReader.finished())
519 return packFailure<Evpack>(DiagnosticCode::ParseError, "runtime package TOC has trailing bytes");
520 if (completePackage) pack.bytes_.assign(bytes.begin(), bytes.end());
521 return Result<Evpack>::success(std::move(pack));
522}

545Result<std::vector<std::uint8_t>> Evpack::decodeChunk(
546 std::size_t index, std::uint64_t maximumDecodedBytes) const {
547 if (index >= chunks_.size())
548 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
549 "chunk index is outside the TOC");
550 if (bytes_.empty())
551 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::PreconditionViolation,
552 "metadata-only package requires a range source");
553 const auto& chunk = chunks_[index];
554 const auto stored = std::span<const std::uint8_t>(bytes_).subspan(
555 static_cast<std::size_t>(chunk.offset), static_cast<std::size_t>(chunk.storedSize));
556 auto decoded = decompressEvpackChunk(chunk.codec, stored, chunk.decodedSize, maximumDecodedBytes);
557 if (!decoded) return decoded;
558 if (sha256(decoded.value()) != chunk.contentHash)
559 return packFailure<std::vector<std::uint8_t>>(DiagnosticCode::HashMismatch,
560 "runtime chunk hash does not match", chunk.type);
561 return decoded;
562}
