16bool sameValue(const ReflectedValue& a, const ReflectedValue& b) {
17 if (a.kind != b.kind) return false;
18 switch (a.kind) {
19 case ReflectedValueKind::Bool: return a.boolean == b.boolean;
20 case ReflectedValueKind::Integer: return a.integer == b.integer;
21 case ReflectedValueKind::Float: return a.floating == b.floating;
22 case ReflectedValueKind::String: return a.text == b.text;
23 default: return true;
24 }

129void DatabasePanel::rebuildCache() {
130 entries_ = ObjectRegistry::instance().entries(selectedClass_);
131 members_.clear();
132 if (const ReflectedClass* cls =
133 ModuleManager::runtime() ? ModuleManager::runtime()->reflectedClass(
134 selectedClass_)
135 : nullptr) {
136 for (const ReflectedMember& member : cls->members) {
137 if (!member.method) members_.push_back(member);
138 }
139 }
140 cached_.assign(entries_.size(), std::vector<ReflectedValue>(members_.size()));
141 Runtime* rt = ModuleManager::runtime();
142 if (!rt) {
143 entries_.clear();
144 return;
145 }
146 for (size_t r = 0; r < entries_.size(); ++r) {
147 for (size_t c = 0; c < members_.size(); ++c)
148 cached_[r][c] = rt->readProperty(entries_[r].object, members_[c].name);
149 }
150}

152WidgetDesc DatabasePanel::cellWidget(const ObjectEntry& entry,
153 const ReflectedMember& member,
154 const ReflectedValue& value) {
155 const std::string id =
156 "cell_" + std::to_string(entry.id) + "_" + member.name;
157 const std::string label = cellLabel(member.name, entry.id);
158 const std::string editor = member.attrString("editor");
159 const std::vector<std::string> options = member.attrOptions("options");
160 WidgetDesc cell;
161 if (value.kind == ReflectedValueKind::Bool) {
162 cell = checkbox(label, value.asBool(), id,
163 [this, entryId = entry.id, name = member.name](bool v) {
164 ReflectedValue out;
165 out.kind = ReflectedValueKind::Bool;
166 out.boolean = v;
167 if (const ObjectEntry* e =
168 ObjectRegistry::instance().entry(entryId))
169 ModuleManager::runtime()->writeProperty(e->object, name,
170 out);
171 });
172 } else if (editor == "combo" && !options.empty()) {
173 int index = 0;
174 const std::string current =
175 value.kind == ReflectedValueKind::String ? value.text : valueText(value);
176 const auto it = std::find(options.begin(), options.end(), current);
177 if (it != options.end()) index = int(it - options.begin());
178 cell = combo(label, options, index, id,
179 [this, entryId = entry.id, name = member.name,
180 kind = value.kind, options](float i) {
181 const int idx = static_cast<int>(i);
182 if (idx < 0 || idx >= int(options.size())) return;
183 const ObjectEntry* e =
184 ObjectRegistry::instance().entry(entryId);
185 if (!e) return;
186 ReflectedValue out;
187 if (kind == ReflectedValueKind::Integer) {
188 out.kind = ReflectedValueKind::Integer;
189 out.integer =
190 std::strtoll(options[size_t(idx)].c_str(), nullptr, 10);
191 } else if (kind == ReflectedValueKind::Float) {
192 out.kind = ReflectedValueKind::Float;
193 out.floating =
194 std::strtod(options[size_t(idx)].c_str(), nullptr);
195 } else {
196 out.kind = ReflectedValueKind::String;
197 out.text = options[size_t(idx)];
198 }
199 ModuleManager::runtime()->writeProperty(e->object, name, out);
200 });
201 } else if (value.kind == ReflectedValueKind::Array ||
202 value.kind == ReflectedValueKind::Table ||
203 value.kind == ReflectedValueKind::Instance ||
204 value.kind == ReflectedValueKind::None ||
205 value.kind == ReflectedValueKind::Other) {
206 std::string shown;
207 switch (value.kind) {
208 case ReflectedValueKind::Array: shown = "array"; break;
209 case ReflectedValueKind::Table: shown = "table"; break;
210 case ReflectedValueKind::Instance: shown = "instance"; break;
211 default: shown = "null"; break;
212 }
213 cell = text(member.name + " = " + shown, id);
214 } else {
215 cell = inputText(label, valueText(value), id,
216 [this, entryId = entry.id, name = member.name,
217 kind = value.kind](const std::string& text) {
218 const ObjectEntry* e =
219 ObjectRegistry::instance().entry(entryId);
220 if (!e) return;
221 ReflectedValue out;
222 out.kind = ReflectedValueKind::String;
223 out.text = text;
224 ModuleManager::runtime()->writeProperty(e->object, name,
225 out);
226 });
227 }
228 cell.withSize(kCellWidth, 0.f);
229 return cell;
230}

232WidgetDesc DatabasePanel::build() {
233 std::vector<WidgetDesc> children;
234 if (classNames_.empty()) {
235 children.push_back(text(
236 "No reflected script classes. Run a script that defines classes, "
237 "then call ui.dbRefresh().",
238 "db_hint"));
239 } else {
240 const auto it =
241 std::find(classNames_.begin(), classNames_.end(), selectedClass_);
242 const int classIndex = it == classNames_.end() ? 0 : int(it - classNames_.begin());
243 children.push_back(row(
244 {
245 text("Class", "db_lbl_class"),
246 spacer("db_class_spacer"),
247 combo("##db_class", classNames_, classIndex, "db_class",
248 [this](float index) {
249 const int idx = static_cast<int>(index);
250 if (idx >= 0 && idx < int(classNames_.size()))
251 selectClass(classNames_[size_t(idx)]);
252 }),
253 button("+", "db_add", [this]() { createInstance(); }),
254 },
255 "db_toolbar"));
256 children.push_back(separator("db_sep"));
257
258 // Header row: one label per reflected member.
259 std::vector<WidgetDesc> header;
260 for (const ReflectedMember& member : members_)
261 header.push_back(text(member.name, "db_hdr_" + member.name).withSize(kCellWidth, 0.f));
262 children.push_back(row(std::move(header), "db_header"));
263
264 // Body: one row per instance, cells bound to the live objects.
265 std::vector<WidgetDesc> rows;
266 for (size_t r = 0; r < entries_.size(); ++r) {
267 const ObjectEntry& entry = entries_[r];
268 std::vector<WidgetDesc> cells;
269 for (size_t c = 0; c < members_.size(); ++c) {
270 const ReflectedValue value = ModuleManager::runtime()->readProperty(
271 entry.object, members_[c].name);
272 cells.push_back(cellWidget(entry, members_[c], value));
273 }
274 cells.push_back(
275 button("x##db_" + std::to_string(entry.id),
276 "db_del_" + std::to_string(entry.id),
277 [this, entryId = entry.id]() { unregister(entryId); }));
278 rows.push_back(row(std::move(cells), "db_row_" + std::to_string(entry.id)));
279 }
280 if (rows.empty()) {
281 children.push_back(text("No instances of " + selectedClass_ +
282 ". Press + to create one.",
283 "db_empty"));
284 } else {
285 children.push_back(scrollList("db_rows", std::move(rows), 0.f, kRowHeight));
286 }
287 }
288 return window("Database", std::move(children), "root");
289}

296void DatabasePanel::sync() {
297 if (!host_ || !host_->meta()->visible) return;
298 Runtime* rt = ModuleManager::runtime();
299 if (!rt || entries_.size() != cached_.size()) return;
300 for (size_t r = 0; r < entries_.size(); ++r) {
301 for (size_t c = 0; c < members_.size(); ++c) {
302 const ReflectedValue live =
303 rt->readProperty(entries_[r].object, members_[c].name);
304 if (sameValue(live, cached_[r][c])) continue;
305 cached_[r][c] = live;
306 const std::string id =
307 "cell_" + std::to_string(entries_[r].id) + "_" + members_[c].name;
308 const ReflectedMember& member = members_[c];
309 const std::string editor = member.attrString("editor");
310 const std::vector<std::string> options = member.attrOptions("options");
311 if (live.kind == ReflectedValueKind::Bool) {
312 host_->setCheckedById(id, live.asBool());
313 } else if (editor == "combo" && !options.empty()) {
314 const std::string current =
315 live.kind == ReflectedValueKind::String ? live.text
316 : valueText(live);
317 const auto it = std::find(options.begin(), options.end(), current);
318 host_->setValueById(id, it == options.end() ? 0.f
319 : float(it - options.begin()));
320 } else if (live.kind == ReflectedValueKind::Array ||
321 live.kind == ReflectedValueKind::Table ||
322 live.kind == ReflectedValueKind::Instance ||
323 live.kind == ReflectedValueKind::None ||
324 live.kind == ReflectedValueKind::Other) {
325 // read-only cell; nothing to patch
326 } else {
327 host_->setValueTextById(id, valueText(live));
328 }
329 }
330 }
