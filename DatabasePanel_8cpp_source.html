16bool sameValue(const ReflectedValue& a, const ReflectedValue& b) {
17 if (a.kind != b.kind) return false;
18 switch (a.kind) {
19 case ReflectedValueKind::Bool: return a.boolean == b.boolean;
20 case ReflectedValueKind::Integer: return a.integer == b.integer;
21 case ReflectedValueKind::Float: return a.floating == b.floating;
22 case ReflectedValueKind::String: return a.text == b.text;
23 default: return true;
24 }

77void DatabasePanel::refresh() {
78 classNames_.clear();
79 if (Runtime* rt = ModuleManager::runtime()) {
80 rt->scanClasses();
81 for (const ReflectedClass& cls : rt->reflectedClasses())
82 classNames_.push_back(cls.name);
83 }
84 if (!selectedClass_.empty() &&
85 std::find(classNames_.begin(), classNames_.end(), selectedClass_) ==
86 classNames_.end()) {
87 selectedClass_.clear();
88 }
89 if (selectedClass_.empty() && !classNames_.empty())
90 selectedClass_ = classNames_.front();
91 rebuildCache();
92 rebuildHost();
93}

141void DatabasePanel::rebuildCache() {
142 entries_ = ObjectRegistry::instance().entries(selectedClass_);
143 members_.clear();
144 if (const ReflectedClass* cls =
145 ModuleManager::runtime() ? ModuleManager::runtime()->reflectedClass(
146 selectedClass_)
147 : nullptr) {
148 for (const ReflectedMember& member : cls->members) {
149 if (!member.method) members_.push_back(member);
150 }
151 }
152 cached_.assign(entries_.size(), std::vector<ReflectedValue>(members_.size()));
153 Runtime* rt = ModuleManager::runtime();
154 if (!rt) {
155 entries_.clear();
156 return;
157 }
158 for (size_t r = 0; r < entries_.size(); ++r) {
159 for (size_t c = 0; c < members_.size(); ++c)
160 cached_[r][c] = rt->readProperty(entries_[r].object, members_[c].name);
161 }
162}

164WidgetDesc DatabasePanel::cellWidget(const ObjectEntry& entry,
165 const ReflectedMember& member,
166 const ReflectedValue& value) {
167 const std::string id = "cell_" + std::to_string(entry.handle.packed()) + "_" + member.name;
168 const std::string label = cellLabel(member.name, entry.handle);
169 const std::string editor = member.attrString("editor");
170 const std::vector<std::string> options = member.attrOptions("options");
171 WidgetDesc cell;
172 if (value.kind == ReflectedValueKind::Bool) {
173 cell = checkbox(label, value.asBool(), id, [this, entryHandle = entry.handle, name = member.name](bool v) {
174 ReflectedValue out;
175 out.kind = ReflectedValueKind::Bool;
176 out.boolean = v;
177 if (const ObjectEntry* e = ObjectRegistry::instance().entry(entryHandle))
178 ModuleManager::runtime()->writeProperty(e->object, name, out);
179 });
180 } else if (editor == "combo" && !options.empty()) {
181 int index = 0;
182 const std::string current =
183 value.kind == ReflectedValueKind::String ? value.text : valueText(value);
184 const auto it = std::find(options.begin(), options.end(), current);
185 if (it != options.end()) index = int(it - options.begin());
186 cell = combo(label, options, index, id,
187 [this, entryHandle = entry.handle, name = member.name, kind = value.kind, options](float i) {
188 const int idx = static_cast<int>(i);
189 if (idx < 0 || idx >= int(options.size())) return;
190 const ObjectEntry* e = ObjectRegistry::instance().entry(entryHandle);
191 if (!e) return;
192 ReflectedValue out;
193 if (kind == ReflectedValueKind::Integer) {
194 out.kind = ReflectedValueKind::Integer;
195 out.integer =
196 std::strtoll(options[size_t(idx)].c_str(), nullptr, 10);
197 } else if (kind == ReflectedValueKind::Float) {
198 out.kind = ReflectedValueKind::Float;
199 out.floating =
200 std::strtod(options[size_t(idx)].c_str(), nullptr);
201 } else {
202 out.kind = ReflectedValueKind::String;
203 out.text = options[size_t(idx)];
204 }
205 ModuleManager::runtime()->writeProperty(e->object, name, out);
206 });
207 } else if (value.kind == ReflectedValueKind::Array ||
208 value.kind == ReflectedValueKind::Table ||
209 value.kind == ReflectedValueKind::Instance ||
210 value.kind == ReflectedValueKind::None ||
211 value.kind == ReflectedValueKind::Other) {
212 std::string shown;
213 switch (value.kind) {
214 case ReflectedValueKind::Array: shown = "array"; break;
215 case ReflectedValueKind::Table: shown = "table"; break;
216 case ReflectedValueKind::Instance: shown = "instance"; break;
217 default: shown = "null"; break;
218 }
219 cell = text(member.name + " = " + shown, id);
220 } else {
221 cell = inputText(
222 label, valueText(value), id,
223 [this, entryHandle = entry.handle, name = member.name, kind = value.kind](const std::string& text) {
224 const ObjectEntry* e = ObjectRegistry::instance().entry(entryHandle);
225 if (!e) return;
226 ReflectedValue out;
227 out.kind = ReflectedValueKind::String;
228 out.text = text;
229 ModuleManager::runtime()->writeProperty(e->object, name, out);
230 });
231 }
232 cell.withSize(kCellWidth, 0.f);
233 return cell;
234}

236WidgetDesc DatabasePanel::build() {
237 std::vector<WidgetDesc> children;
238 if (classNames_.empty()) {
239 children.push_back(text(
240 "No reflected script classes. Run a script that defines classes, "
241 "then call ui.dbRefresh().",
242 "db_hint"));
243 } else {
244 const auto it =
245 std::find(classNames_.begin(), classNames_.end(), selectedClass_);
246 const int classIndex = it == classNames_.end() ? 0 : int(it - classNames_.begin());
247 children.push_back(row(
248 {
249 text("Class", "db_lbl_class"),
250 spacer("db_class_spacer"),
251 combo("##db_class", classNames_, classIndex, "db_class",
252 [this](float index) {
253 const int idx = static_cast<int>(index);
254 if (idx >= 0 && idx < int(classNames_.size()))
255 selectClass(classNames_[size_t(idx)]);
256 }),
257 button("+", "db_add", [this]() { createInstance().ignore("create UI database row from button"); }),
258 },
259 "db_toolbar"));
260 children.push_back(separator("db_sep"));
261
262 // Header row: one label per reflected member.
263 std::vector<WidgetDesc> header;
264 for (const ReflectedMember& member : members_)
265 header.push_back(text(member.name, "db_hdr_" + member.name).withSize(kCellWidth, 0.f));
266 children.push_back(row(std::move(header), "db_header"));
267
268 // Body: one row per instance, cells bound to the live objects.
269 std::vector<WidgetDesc> rows;
270 for (size_t r = 0; r < entries_.size(); ++r) {
271 const ObjectEntry& entry = entries_[r];
272 std::vector<WidgetDesc> cells;
273 for (size_t c = 0; c < members_.size(); ++c) {
274 const ReflectedValue value = ModuleManager::runtime()->readProperty(
275 entry.object, members_[c].name);
276 cells.push_back(cellWidget(entry, members_[c], value));
277 }
278 cells.push_back(button(
279 "x##db_" + std::to_string(entry.handle.packed()), "db_del_" + std::to_string(entry.handle.packed()),
280 [this, entryHandle = entry.handle]() { unregister(entryHandle).ignore("delete UI database row"); }));
281 rows.push_back(row(std::move(cells), "db_row_" + std::to_string(entry.handle.packed())));
282 }
283 if (rows.empty()) {
284 children.push_back(text("No instances of " + selectedClass_ +
285 ". Press + to create one.",
286 "db_empty"));
287 } else {
288 children.push_back(scrollList("db_rows", std::move(rows), 0.f, kRowHeight));
289 }
290 }
291 return window("Database", std::move(children), "root");
292}

300void DatabasePanel::sync() {
301 auto host = UIHost::resolve(host_);
302 if (!host || !host->get().meta()->visible) return;
303 Runtime* rt = ModuleManager::runtime();
304 if (!rt || entries_.size() != cached_.size()) return;
305 for (size_t r = 0; r < entries_.size(); ++r) {
306 for (size_t c = 0; c < members_.size(); ++c) {
307 const ReflectedValue live =
308 rt->readProperty(entries_[r].object, members_[c].name);
309 if (sameValue(live, cached_[r][c])) continue;
310 cached_[r][c] = live;
311 const std::string id = "cell_" + std::to_string(entries_[r].handle.packed()) + "_" + members_[c].name;
312 const ReflectedMember& member = members_[c];
313 const std::string editor = member.attrString("editor");
314 const std::vector<std::string> options = member.attrOptions("options");
315 if (live.kind == ReflectedValueKind::Bool) {
316 host->get().setCheckedById(id, live.asBool());
317 } else if (editor == "combo" && !options.empty()) {
318 const std::string current =
319 live.kind == ReflectedValueKind::String ? live.text
320 : valueText(live);
321 const auto it = std::find(options.begin(), options.end(), current);
322 host->get().setValueById(id, it == options.end() ? 0.f : float(it - options.begin()));
323 } else if (live.kind == ReflectedValueKind::Array ||
324 live.kind == ReflectedValueKind::Table ||
325 live.kind == ReflectedValueKind::Instance ||
326 live.kind == ReflectedValueKind::None ||
327 live.kind == ReflectedValueKind::Other) {
328 // read-only cell; nothing to patch
329 } else {
330 host->get().setValueTextById(id, valueText(live));
331 }
332 }
333 }
