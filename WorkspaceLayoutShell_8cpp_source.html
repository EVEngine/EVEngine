38WidgetDesc regionContent(const std::vector<WorkspacePanelDescriptor> &panels,
39 const std::string &activePanel, bool tabStack,
40 WorkspaceLayoutShell::ContentBuilder content) {
41 if (panels.empty()) {
42 return ui::panelFrame("Empty", {ui::text("No panel", "empty")}, "empty");
43 }
44 if (panels.size() == 1 || !tabStack) {
45 return fillPanel(panels.front(), content);
46 }
47 std::vector<WidgetDesc> pages;
48 pages.reserve(panels.size());
49 std::string active = activePanel;
50 bool activeFound = false;
51 for (const auto &panel : panels) {
52 if (panel.id == active) activeFound = true;
53 pages.push_back(fillPanel(panel, content));
54 }
55 if (!activeFound) active = panels.front().id;
56 return ui::tabContainer(std::move(pages), active, panels.front().region + "/tabs");
57}

65std::vector<WorkspacePanelDescriptor> WorkspaceLayoutShell::panelsInRegion(
66 const editor::EditorWorkspace &workspace, std::string_view region) {
67 std::vector<WorkspacePanelDescriptor> out;
68 const int count = workspace.getPanelCount();
69 out.reserve(size_t(std::max(0, count)));
70 for (int i = 0; i < count; ++i) {
71 auto panel = workspace.panelAt(size_t(i));
72 if (!panel.ok()) continue;
73 if (!panel.value().visible) continue;
74 if (panel.value().region != region) continue;
75 out.push_back(panel.value());
76 }
77 std::stable_sort(out.begin(), out.end(),
78 [](const WorkspacePanelDescriptor &a, const WorkspacePanelDescriptor &b) {
79 if (a.order != b.order) return a.order < b.order;
80 return a.id < b.id;
81 });
82 return out;
83}

85ui::WidgetDesc WorkspaceLayoutShell::buildDockBody(const editor::EditorWorkspace &workspace,

88 const auto left = panelsInRegion(workspace, "left");

91 const auto bottom = panelsInRegion(workspace, "bottom");

93 const std::string &active = workspace.getActivePanel();
94 const bool tabStack = options.tabStackRegions;
95
96 WidgetDesc centerPane =
97 wrapDock("center", regionContent(center, active, tabStack, content));
98 if (!right.empty()) {
99 centerPane = ui::splitPane(ui::FlexDirection::Row, std::move(centerPane),
100 wrapDock("right", regionContent(right, active, tabStack, content)),
101 1.f - clampRatio(options.rightRatio), "split/center-right");
102 }
103 WidgetDesc mainRow = std::move(centerPane);
104 if (!left.empty()) {
105 mainRow = ui::splitPane(ui::FlexDirection::Row,
106 wrapDock("left", regionContent(left, active, tabStack, content)),
107 std::move(mainRow), clampRatio(options.leftRatio), "split/left-main");
108 }
109
110 WidgetDesc body = std::move(mainRow);
111 if (!top.empty()) {
112 body = ui::splitPane(ui::FlexDirection::Column,
113 wrapDock("top", regionContent(top, active, tabStack, content)),
114 std::move(body), 0.12f, "split/top-main");
115 }
116 if (!bottom.empty()) {
117 body = ui::splitPane(ui::FlexDirection::Column, std::move(body),
118 wrapDock("bottom", regionContent(bottom, active, tabStack, content)),
119 1.f - clampRatio(options.bottomRatio), "split/main-bottom");
120 }
121 body.flexGrow = 1.f;
122 return body;
123}
