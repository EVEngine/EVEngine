24TerrainStreamStats TerrainStreamingCache::streamAround(int worldX, int worldY, int radius,
25 int maxLoads, std::string *error) {
26 TerrainStreamStats stats;
27 if (radius < 0 || asset_.getChunkSize() <= 0) { stats.resident = getResidentCount(); return stats; }
28 const int chunkSize = asset_.getChunkSize();
29 const int centerX = worldX >= 0 ? worldX / chunkSize : -((-worldX - 1) / chunkSize) - 1;
30 const int centerY = worldY >= 0 ? worldY / chunkSize : -((-worldY - 1) / chunkSize) - 1;
31 const int64_t radiusSq = int64_t(radius) * radius;
32
33 for (auto it = resident_.begin(); it != resident_.end();) {
34 const int cx = int32_t(uint32_t(it->first));
35 const int cy = int32_t(uint32_t(it->first >> 32));
36 const int64_t dx = int64_t(cx) - centerX, dy = int64_t(cy) - centerY;
37 if (dx * dx + dy * dy > radiusSq) { it = resident_.erase(it); ++stats.evicted; }
38 else ++it;
39 }
40
41 struct Request { int x, y; int64_t distanceSq; };
42 std::vector<Request> requests;
43 for (const TerrainChunkEntry &entry : asset_.chunks()) {
44 const int64_t dx = int64_t(entry.chunkX) - centerX, dy = int64_t(entry.chunkY) - centerY;
45 const int64_t distanceSq = dx * dx + dy * dy;
46 if (distanceSq <= radiusSq && !getChunk(entry.chunkX, entry.chunkY))
47 requests.push_back({entry.chunkX, entry.chunkY, distanceSq});
48 }
49 std::sort(requests.begin(), requests.end(), [](const Request &a, const Request &b) {
50 return std::tie(a.distanceSq, a.y, a.x) < std::tie(b.distanceSq, b.y, b.x);
51 });
52 const size_t loadCount = maxLoads > 0 ? std::min(requests.size(), size_t(maxLoads)) : requests.size();
53 for (size_t i = 0; i < loadCount; ++i) {
54 TerrainChunkData chunk;
55 std::string chunkError;
56 if (!asset_.loadChunk(requests[i].x, requests[i].y, chunk, &chunkError)) {
57 ++stats.failed;
58 if (error && error->empty()) *error = chunkError;
59 continue;
60 }
61 resident_.emplace(key(requests[i].x, requests[i].y), std::move(chunk));
62 ++stats.loaded;
63 }
64 stats.pending = int(requests.size() - loadCount);
65 stats.resident = getResidentCount();
66 return stats;
67}

74bool TerrainStreamingCache::sampleCell(int worldX, int worldY, TerrainSample &out) const {
75 if (worldX < 0 || worldY < 0 || worldX >= asset_.getWidth() || worldY >= asset_.getHeight()) return false;
76 const int cs = asset_.getChunkSize(), cx = worldX / cs, cy = worldY / cs;
77 const TerrainChunkData *chunk = getChunk(cx, cy);
78 if (!chunk) return false;
79 const int lx = worldX - cx * cs, ly = worldY - cy * cs;
80 if (lx >= chunk->width || ly >= chunk->height) return false;
81 const size_t i = size_t(ly) * size_t(chunk->width) + size_t(lx);
82 out.height = chunk->heights.data()[i]; out.flowAccumulation = chunk->flowAccumulation[i];
83 out.flowDirection = int(chunk->flowDirection[i]);
84 out.flowVectorX = chunk->flowVectorX[i]; out.flowVectorY = chunk->flowVectorY[i];
85 out.streamOrder = int(chunk->streamOrder[i]);
86 out.lakeDepth = chunk->lakeDepth[i]; out.lake = out.lakeDepth > 0.001f;
87 out.temperature = chunk->temperature[i]; out.moisture = chunk->moisture[i];
88 out.river = chunk->rivers[i] != 0; out.biome = chunk->biomes[i];
89 return true;
90}

92bool TerrainStreamingCache::getReceiver(int worldX, int worldY, int &receiverX,
93 int &receiverY) const {
94 static constexpr int dx[8] = {-1, 0, 1, -1, 1, -1, 0, 1};
95 static constexpr int dy[8] = {-1, -1, -1, 0, 0, 1, 1, 1};
96 TerrainSample source;
97 if (!sampleCell(worldX, worldY, source) || source.flowDirection < 0 ||
98 source.flowDirection >= 8) return false;
99 receiverX = worldX + dx[source.flowDirection];
100 receiverY = worldY + dy[source.flowDirection];
101 TerrainSample receiver;
102 return sampleCell(receiverX, receiverY, receiver);

105bool TerrainStreamingCache::traceFlow(int worldX, int worldY, int maxSteps,
106 std::vector<std::pair<int, int>> &out) const {
107 out.clear();
108 TerrainSample start;
109 if (maxSteps <= 0 || !sampleCell(worldX, worldY, start)) return false;
110 out.emplace_back(worldX, worldY);
111 for (int step = 0; step < maxSteps; ++step) {
112 TerrainSample current;
113 if (!sampleCell(worldX, worldY, current)) return false;
114 if (current.flowDirection < 0) return true;
115 int nextX = 0, nextY = 0;
116 if (!getReceiver(worldX, worldY, nextX, nextY)) return false;
117 if (std::find(out.begin(), out.end(), std::pair<int, int>{nextX, nextY}) != out.end())
118 return false;
119 out.emplace_back(nextX, nextY);
120 worldX = nextX; worldY = nextY;
121 }
122 TerrainSample current;
123 return sampleCell(worldX, worldY, current) && current.flowDirection < 0;
124}

141 next.climate.temperature.resize(count); next.climate.moisture.resize(count);

163bool TerrainStreamingCache::sampleHeight(float worldX, float worldY, float &out) const {
164 if (!std::isfinite(worldX) || !std::isfinite(worldY) || worldX < 0.f || worldY < 0.f ||
165 worldX > float(asset_.getWidth() - 1) || worldY > float(asset_.getHeight() - 1)) return false;
166 const int x0 = int(std::floor(worldX)), y0 = int(std::floor(worldY));
167 const int x1 = std::min(x0 + 1, asset_.getWidth() - 1), y1 = std::min(y0 + 1, asset_.getHeight() - 1);
168 TerrainSample a, b, c, d;
169 if (!sampleCell(x0, y0, a) || !sampleCell(x1, y0, b) || !sampleCell(x0, y1, c) || !sampleCell(x1, y1, d)) return false;
170 const float tx = worldX - float(x0), ty = worldY - float(y0);
171 const float top = a.height + (b.height - a.height) * tx;
172 const float bottom = c.height + (d.height - c.height) * tx;
173 out = top + (bottom - top) * ty;
174 return true;
175}
