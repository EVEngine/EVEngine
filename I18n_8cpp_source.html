46bool validIdentifier(const std::string& value) {
47 if (value.empty() || value.size() > 64) return false;
48 for (const unsigned char ch : value)
49 if (!((ch >= 'a' && ch <= 'z') || (ch >= 'A' && ch <= 'Z') || (ch >= '0' && ch <= '9') || ch == '-' ||
50 ch == '_'))
51 return false;
52 return true;
53}

55bool collectPlaceholders(const std::string& text, std::set<std::string>& result, std::string& error) {
56 result.clear();
57 for (size_t offset = 0; offset < text.size();) {
58 if (text[offset] == '}') {
59 error = "unmatched closing placeholder brace";
60 return false;
61 }
62 if (text[offset] != '{') {
63 ++offset;
64 continue;
65 }
66 const size_t close = text.find('}', offset + 1);
67 if (close == std::string::npos) {
68 error = "unterminated placeholder";
69 return false;
70 }
71 const std::string name = text.substr(offset + 1, close - offset - 1);
72 if (!validIdentifier(name)) {
73 error = "placeholder names must be stable identifiers";
74 return false;
75 }
76 result.insert(name);
77 offset = close + 1;
78 }
79 return true;
80}

82bool flattenStrict(eve::json::Value node, const std::string& prefix, int depth,
83 std::unordered_map<std::string, std::string>& strings,
84 std::unordered_map<std::string, std::unordered_map<std::string, std::string>>& plurals,
85 std::string& error, std::string& errorPath) {
86 if (depth > 16) {
87 error = "locale nesting exceeds 16 levels";
88 errorPath = prefix;
89 return false;
90 }
91 if (!node.isObject() || node.size() == 0) {
92 error = "translation namespaces must be non-empty objects";
93 errorPath = prefix;
94 return false;
95 }
96 const std::vector<std::string> names = node.keys();
97 bool allPlural = true;
98 bool anyPlural = false;
99 for (const auto& name : names) {
100 if (isPluralCategory(name))
101 anyPlural = true;
102 else
103 allPlural = false;
104 if (!validIdentifier(name)) {
105 error = "translation key segments must be stable identifiers";
106 errorPath = prefix + "." + name;
107 return false;
108 }
109 }
110 if (anyPlural && !allPlural) {
111 error = "plural categories cannot be mixed with namespace keys";
112 errorPath = prefix;
113 return false;
114 }
115 if (allPlural) {
116 if (prefix.empty() || !node.has("other")) {
117 error = "plural tables require a non-empty key and an other form";
118 errorPath = prefix;
119 return false;
120 }
121 std::unordered_map<std::string, std::string> forms;
122 for (const auto& name : names) {
123 const auto value = node.get(name.c_str());
124 if (!value.isString() || value.asString().empty() || value.asString().size() > 16384) {
125 error = "plural forms must be non-empty strings of at most 16384 bytes";
126 errorPath = prefix + "." + name;
127 return false;
128 }
129 forms.emplace(name, value.asString());
130 }
131 plurals.emplace(prefix, std::move(forms));
132 return true;
133 }
134 for (const auto& name : names) {
135 const auto value = node.get(name.c_str());
136 const std::string path = prefix.empty() ? name : prefix + "." + name;
137 if (value.isString()) {
138 if (value.asString().empty() || value.asString().size() > 16384) {
139 error = "translations must be non-empty strings of at most 16384 bytes";
140 errorPath = path;
141 return false;
142 }
143 strings.emplace(path, value.asString());
144 } else if (value.isObject()) {
145 if (!flattenStrict(value, path, depth + 1, strings, plurals, error, errorPath)) return false;
146 } else {
147 error = "translations must be strings, namespaces, or plural tables";
148 errorPath = path;
149 return false;
150 }
151 if (strings.size() + plurals.size() > 10000) {
152 error = "a locale may contain at most 10000 translation keys";
153 errorPath = path;
154 return false;
155 }
156 }
157 return true;
158}

160void flatten(eve::json::Value node, const std::string& prefix, std::unordered_map<std::string, std::string>& strings,
161 std::unordered_map<std::string, std::unordered_map<std::string, std::string>>& plurals) {
162 if (node.isObject()) {
163 const std::vector<std::string> names = node.keys();
164 // A plain object of plural categories is a plural form table.
165 bool allCats = !names.empty();
166 bool allStrings = true;
167 for (const auto& k : names) {
168 if (!isPluralCategory(k)) {
169 allCats = false;
170 break;
171 }
172 if (!node.get(k.c_str()).isString()) allStrings = false;
173 }
174 if (allCats && allStrings && !prefix.empty()) {
175 std::unordered_map<std::string, std::string> forms;
176 for (const auto& k : names) forms[k] = node.getString(k.c_str());
177 plurals[prefix] = std::move(forms);
178 return;
179 }
180 for (const auto& k : names) {
181 const std::string path = prefix.empty() ? k : prefix + "." + k;
182 flatten(node.get(k.c_str()), path, strings, plurals);
183 }
184 return;
185 }
186 if (prefix.empty()) return;
187 // Scalars stringify; arrays and nulls are not translatable and are skipped.
188 if (node.isString() || node.isNumber() || node.isBool()) strings[prefix] = node.asString();
189}

191bool parseLocale(const std::string& text, std::unordered_map<std::string, std::string>& strings,
192 std::unordered_map<std::string, std::unordered_map<std::string, std::string>>& plurals,
193 std::string* error) {
194 const eve::json::Document doc = eve::json::Document::parse(text, error);
195 if (!doc.valid()) return false;
196 if (!doc.root().isObject()) {
197 if (error) *error = "locale root must be a JSON object";
198 return false;
199 }
200 strings.clear();
201 plurals.clear();
202 flatten(doc.root(), "", strings, plurals);
203 return true;
204}

214// Convert a Squirrel table (or null) into a string params map.
215std::unordered_map<std::string, std::string> readParams(ssq::Object params) {
216 std::unordered_map<std::string, std::string> out;
217 HSQUIRRELVM vm = params.getHandle();
218 if (!vm) return out;
219 const HSQOBJECT raw = params.getRaw();
220 if (raw._type != OT_TABLE) return out;
221
222 const SQInteger top = sq_gettop(vm);
223 sq_pushobject(vm, raw);
224 sq_pushnull(vm); // iterator
225 while (SQ_SUCCEEDED(sq_next(vm, -2))) {
226 // key at -2, value at -1
227 const SQChar* k = nullptr;
228 if (sq_gettype(vm, -2) == OT_STRING && SQ_SUCCEEDED(sq_getstring(vm, -2, &k)) && k) {
229 std::string v;
230 const SQObjectType vt = sq_gettype(vm, -1);
231 if (vt == OT_STRING) {
232 const SQChar* sv = nullptr;
233 if (SQ_SUCCEEDED(sq_getstring(vm, -1, &sv)) && sv) v = sv;
234 } else if (vt == OT_INTEGER) {
235 SQInteger iv = 0;
236 if (SQ_SUCCEEDED(sq_getinteger(vm, -1, &iv))) v = std::to_string(iv);
237 } else if (vt == OT_FLOAT) {
238 SQFloat fv = 0;
239 if (SQ_SUCCEEDED(sq_getfloat(vm, -1, &fv))) v = numberToString(fv);
240 } else if (vt == OT_BOOL) {
241 SQBool bv = SQFalse;
242 if (SQ_SUCCEEDED(sq_getbool(vm, -1, &bv))) v = bv ? "true" : "false";
243 }
244 out[k] = std::move(v);
245 }
246 sq_pop(vm, 2);
247 }
248 sq_settop(vm, top);
249 return out;
250}

287eve::Result<int> I18n::replaceBundleFromJson(const std::string& json) {
288 std::string parseError;
289 const auto document = eve::json::Document::parse(json, &parseError);
290 if (!document.valid())
291 return bundleFailure<int>(eve::DiagnosticCode::ParseError,
292 parseError.empty() ? "invalid localization bundle JSON" : parseError, "$");
293 const auto root = document.root();
294 if (!root.isObject())
295 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument, "localization bundle root must be an object",
296 "$");
297 const std::unordered_set<std::string> rootFields = {"schema", "version", "defaultLocale", "locales"};
298 for (const auto& field : root.keys())
299 if (!rootFields.contains(field))
300 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument,
301 "localization bundle contains an unknown field", "$." + field);
302 const auto schema = root.get("schema");
303 const auto version = root.get("version");
304 const auto defaultLocale = root.get("defaultLocale");
305 const auto locales = root.get("locales");
306 if (!schema.isString() || schema.asString() != "eve.i18n.bundle")
307 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument, "localization bundle schema id is invalid",
308 "$.schema");
309 if (!version.isInt64() || version.asInt64() != 1)
310 return bundleFailure<int>(eve::DiagnosticCode::UnknownVersion, "unsupported localization bundle schema version",
311 "$.version");
312 if (!defaultLocale.isString() || !validIdentifier(defaultLocale.asString()))
313 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument,
314 "defaultLocale must be a stable locale identifier", "$.defaultLocale");
315 if (!locales.isObject() || locales.size() == 0 || locales.size() > 32)
316 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument,
317 "locales must contain between 1 and 32 locale objects", "$.locales");
318 if (!locales.has(defaultLocale.asString().c_str()))
319 return bundleFailure<int>(eve::DiagnosticCode::NotFound, "defaultLocale is not present in locales",
320 "$.defaultLocale");
321
322 std::unordered_map<std::string, Locale> proposed;
323 for (const auto& lang : locales.keys()) {
324 if (!validIdentifier(lang))
325 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument, "locale names must be stable identifiers",
326 "$.locales." + lang);
327 Locale candidate;
328 std::string error;
329 std::string errorPath;
330 if (!flattenStrict(locales.get(lang.c_str()), "", 0, candidate.strings, candidate.plurals, error, errorPath))
331 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument, std::move(error),
332 "$.locales." + lang + (errorPath.empty() ? "" : "." + errorPath));
333 proposed.emplace(lang, std::move(candidate));
334 }
335
336 const Locale& source = proposed.at(defaultLocale.asString());
337 for (const auto& [lang, locale] : proposed) {
338 for (const auto& [key, sourceText] : source.strings) {
339 const auto translated = locale.strings.find(key);
340 if (translated == locale.strings.end())
341 return bundleFailure<int>(eve::DiagnosticCode::NotFound,
342 "locale is missing a required singular translation",
343 "$.locales." + lang + "." + key);
344 std::set<std::string> expected;
345 std::set<std::string> actual;
346 std::string placeholderError;
347 if (!collectPlaceholders(sourceText, expected, placeholderError))
348 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument, placeholderError,
349 "$.locales." + defaultLocale.asString() + "." + key);
350 if (!collectPlaceholders(translated->second, actual, placeholderError))
351 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument, placeholderError,
352 "$.locales." + lang + "." + key);
353 if (expected != actual)
354 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument,
355 "translation placeholders do not match the default locale",
356 "$.locales." + lang + "." + key);
357 }
358 for (const auto& [key, translated] : locale.strings)
359 if (!source.strings.contains(key))
360 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument,
361 "locale contains a singular key absent from the default locale",
362 "$.locales." + lang + "." + key);
363 for (const auto& [key, sourceForms] : source.plurals) {
364 const auto translated = locale.plurals.find(key);
365 if (translated == locale.plurals.end())
366 return bundleFailure<int>(eve::DiagnosticCode::NotFound,
367 "locale is missing a required plural translation",
368 "$.locales." + lang + "." + key);
369 for (const auto& [form, translatedText] : translated->second) {
370 const auto exactSource = sourceForms.find(form);
371 const auto fallbackSource = sourceForms.find("other");
372 const std::string& sourceText =
373 exactSource != sourceForms.end() ? exactSource->second : fallbackSource->second;
374 std::set<std::string> expected;
375 std::set<std::string> actual;
376 std::string placeholderError;
377 if (!collectPlaceholders(sourceText, expected, placeholderError))
378 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument, placeholderError,
379 "$.locales." + defaultLocale.asString() + "." + key + "." + form);
380 if (!collectPlaceholders(translatedText, actual, placeholderError))
381 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument, placeholderError,
382 "$.locales." + lang + "." + key + "." + form);
383 if (expected != actual)
384 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument,
385 "plural placeholders do not match the default locale",
386 "$.locales." + lang + "." + key + "." + form);
387 }
388 }
389 for (const auto& [key, forms] : locale.plurals) {
390 (void)forms;
391 if (!source.plurals.contains(key))
392 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument,
393 "locale contains a plural key absent from the default locale",
394 "$.locales." + lang + "." + key);
395 }
396 }
397
398 const std::string nextDefault = defaultLocale.asString();
399 const std::string nextLanguage = proposed.contains(language_) ? language_ : nextDefault;
400 locales_ = std::move(proposed);
401 defaultLanguage_ = nextDefault;
402 language_ = nextLanguage;
403 return eve::Result<int>::success(static_cast<int>(locales_.size()), eve::Status::success(eve::StatusCode::Applied));
404}

406eve::Result<void> I18n::replaceLocaleFromFile(const std::string& lang, const std::string& path) {
407 if (lang.empty() || path.empty())
408 return bundleFailure<void>(eve::DiagnosticCode::InvalidArgument,
409 "locale identifier and VFS path must not be empty", "locale.file");
410
411 auto* fs = ModuleManager::getInstance<filesystem::Filesystem>("Filesystem");
412 if (!fs) fs = filesystem::Filesystem::create();
413
414 filesystem::FileData* fd = nullptr;
415 try {
416 fd = fs->read(path);
417 } catch (...) {
418 delete fd;
419 return bundleFailure<void>(eve::DiagnosticCode::NotFound, "locale file could not be read", path);
420 }
421 if (fd == nullptr || fd->getData() == nullptr || fd->getSize() == 0) {
422 delete fd;
423 return bundleFailure<void>(eve::DiagnosticCode::NotFound, "locale file is missing or empty", path);
424 }
425
426 std::string text(static_cast<const char*>(fd->getData()), fd->getSize());
427 delete fd;
428
429 Locale loc;
430 std::string error;
431 if (!parseLocale(text, loc.strings, loc.plurals, &error))
432 return bundleFailure<void>(eve::DiagnosticCode::ParseError, error.empty() ? "invalid locale JSON" : error,
433 path);
434
435 loc.path = path;
436 loc.modtime = fileModtime(path);
437 fs->watch(path);
438 locales_[lang] = std::move(loc);
439 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
440}

475eve::Result<int> I18n::validateKeyCoverage(const std::string& key) const {
476 if (key.empty() || key.back() == '.')
477 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument, "translation key must not be empty",
478 "requiredKey");
479 size_t offset = 0;
480 while (offset < key.size()) {
481 const size_t separator = key.find('.', offset);
482 const std::string segment = key.substr(offset, separator == std::string::npos ? std::string::npos
483 : separator - offset);
484 if (!validIdentifier(segment))
485 return bundleFailure<int>(eve::DiagnosticCode::InvalidArgument,
486 "translation key segments must be stable identifiers", "requiredKey." + key);
487 if (separator == std::string::npos) break;
488 offset = separator + 1;
489 }
490 if (locales_.empty())
491 return bundleFailure<int>(eve::DiagnosticCode::NotFound, "no locales have been admitted", "locales");
492 std::vector<std::string> languages;
493 languages.reserve(locales_.size());
494 for (const auto& [language, locale] : locales_) languages.push_back(language);
495 std::sort(languages.begin(), languages.end());
496 for (const auto& language : languages)
497 if (!hasInLanguage(language, key))
498 return bundleFailure<int>(eve::DiagnosticCode::NotFound, "required translation key is missing",
499 "locales." + language + "." + key);
500 return eve::Result<int>::success(static_cast<int>(languages.size()));
501}

534std::string I18n::formatString(const std::string& tpl,
535 const std::unordered_map<std::string, std::string>& params) const {
536 std::string out;
537 out.reserve(tpl.size());
538 for (size_t i = 0; i < tpl.size();) {
539 if (tpl[i] == '{') {
540 const size_t close = tpl.find('}', i + 1);
541 if (close != std::string::npos) {
542 const std::string key = tpl.substr(i + 1, close - i - 1);
543 const auto it = params.find(key);
544 if (it != params.end()) {
545 out += it->second;
546 i = close + 1;
547 continue;
548 }
549 }
550 }
551 out += tpl[i++];
552 }
553 return out;
554}

563std::string I18n::getPluralWithParams(const std::string& key, int n,
564 const std::unordered_map<std::string, std::string>& params) const {
565 std::unordered_map<std::string, std::string> merged = params;
566 merged["n"] = std::to_string(n);
567
568 const Locale* cur = findLocale(language_);
569 const Locale* def = (defaultLanguage_ != language_) ? findLocale(defaultLanguage_) : nullptr;
570 if (!cur && !def) return key;
571
572 // Look up in the current language first, then fall back to the default.
573 std::string current;
574 if (const Locale* loc = cur) {
575 const auto pit = loc->plurals.find(key);
576 if (pit != loc->plurals.end()) {
577 const std::string form = pluralForm(language_, n);
578 auto it = pit->second.find(form);
579 if (it == pit->second.end()) it = pit->second.find("other");
580 if (it == pit->second.end()) it = pit->second.find("one");
581 if (it != pit->second.end()) return formatString(it->second, merged);
582 } else {
583 const auto sit = loc->strings.find(key);
584 if (sit != loc->strings.end()) return formatString(sit->second, merged);
585 }
586 }
587 if (const Locale* loc = def) {
588 const auto pit = loc->plurals.find(key);
589 if (pit != loc->plurals.end()) {
590 const std::string form = pluralForm(defaultLanguage_, n);
591 auto it = pit->second.find(form);
592 if (it == pit->second.end()) it = pit->second.find("other");
593 if (it == pit->second.end()) it = pit->second.find("one");
594 if (it != pit->second.end()) return formatString(it->second, merged);
595 } else {
596 const auto sit = loc->strings.find(key);
597 if (sit != loc->strings.end()) return formatString(sit->second, merged);
598 }
599 }
600 return key;
601}

603std::string I18n::pluralForm(const std::string& lang, int n) const {
604 std::string base = lang;
605 if (const size_t dash = base.find('-'); dash != std::string::npos) base = base.substr(0, dash);
606 if (const size_t under = base.find('_'); under != std::string::npos) base = base.substr(0, under);
607
608 if (base == "zh" || base == "ja" || base == "ko" || base == "th" || base == "vi" || base == "id" || base == "ms" ||
609 base == "tr" || base == "my")
610 return "other";
611
612 if (base == "fr" || base == "pt") return (n == 0 || n == 1) ? "one" : "other";
613
614 if (base == "ru" || base == "uk" || base == "be") {
615 const int mod10 = n % 10;
616 const int mod100 = n % 100;
617 if (mod10 == 1 && mod100 != 11) return "one";
618 if (mod10 >= 2 && mod10 <= 4 && (mod100 < 12 || mod100 > 14)) return "few";
619 return "many";
620 }
621
622 if (base == "pl") {
623 const int mod10 = n % 10;
624 const int mod100 = n % 100;
625 if (n == 1) return "one";
626 if (mod10 >= 2 && mod10 <= 4 && (mod100 < 12 || mod100 > 14)) return "few";
627 return "many";
628 }
629
630 if (base == "cs" || base == "sk") {
631 if (n == 1) return "one";
632 if (n >= 2 && n <= 4) return "few";
633 return "other";
634 }
635
636 return (n == 1) ? "one" : "other";
637}

639int I18n::update(float dt) {
640 (void)dt;
641 if (!autoReload_) return 0;
642
643 // Collect dirty entries first so reloading cannot invalidate the iterator.
644 std::vector<std::string> dirty;
645 for (const auto& [lang, loc] : locales_) {
646 if (loc.path.empty()) continue;
647 const int64_t mt = fileModtime(loc.path);
648 if (mt >= 0 && mt != loc.modtime) dirty.push_back(lang);
649 }
650
651 int reloaded = 0;
652 for (const std::string& lang : dirty) {
653 if (replaceLocaleFromFile(lang, locales_[lang].path).ok()) ++reloaded;
654 }
655 return reloaded;
656}
