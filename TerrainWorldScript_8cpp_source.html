49ResultRef<T> borrowRoadArgument(const ssq::Object& object, const char* path) {
50 if (object.isNull())
51 return ResultRef<T>::failure(Diagnostic::error(
52 DiagnosticCode::InvalidArgument, "road terrain argument is required", path, {}, "procgen.squirrel"));
53 if (object.getType() != ssq::Type::INSTANCE || object.getTypeTag() != ssq::detail::stableTypeHash<T*>())
54 return ResultRef<T>::failure(
55 Diagnostic::error(DiagnosticCode::TypeMismatch,
56 std::string("road terrain argument has the wrong instance type: ") + object.getTypeStr(),
57 path, {}, "procgen.squirrel"));
58 auto* value = object.toPtrUnsafe<T*>();
59 if (!value)
60 return ResultRef<T>::failure(Diagnostic::error(DiagnosticCode::StaleHandle,
61 "road terrain argument has no live native instance", path, {},
62 "procgen.squirrel"));
63 if constexpr (std::is_same_v<T, MeshBuild> || std::is_same_v<T, Heightmap>) {
64 using Ref =
65 std::conditional_t<std::is_same_v<T, MeshBuild>, ProcgenMeshBuildHandleRef, ProcgenHeightmapHandleRef>;
66 if (auto reference = nativeProxyReference<Ref>(value)) {
67 auto* owner = liveProcgen();
68 auto resolved = [&]() -> eve::script::Borrowed<T> {
69 if (!owner) return {};
70 if constexpr (std::is_same_v<T, MeshBuild>)
71 return owner->resolveMeshBuild(*reference);
72 else
73 return owner->resolveHeightmap(*reference);
74 }();
75 if (!resolved.isBound() || resolved.get() != value)
76 return ResultRef<T>::failure(Diagnostic::error(DiagnosticCode::StaleHandle,
77 "road terrain argument owner or generation is stale",
78 path, {}, "procgen.squirrel"));
79 }
80 }
81 return ResultRef<T>::success(std::ref(*value));
82}
83Result<int> missingRaster() {
84 return Result<int>::failure(
85 Diagnostic::error(DiagnosticCode::InvalidArgument, "terrain operation requires non-null rasters and settings"));
86}
