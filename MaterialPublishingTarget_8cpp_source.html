41EditorResult<void> MaterialPublishingTarget::commitDomainState(
42 std::unique_ptr<IDomainOperationTarget> candidate) {
43 auto* typed = dynamic_cast<MaterialPublishingTarget*>(candidate.get());
44 if (!typed || typed->targetId() != targetId() || typed->sink_ != sink_ || !typed->staging_)
45 return publishingError<void>(EditorStatus::Conflict,
46 "editor.material.publishing-candidate-mismatch",
47 "Material candidate belongs to another live target");
48 if (!sink_)
49 return publishingError<void>(EditorStatus::Rejected,
50 "editor.material.publishing-sink-missing",
51 "Material publishing target requires a live runtime sink");
52 EditorResult<void> published = sink_->publish(typed->document_);
53 if (!published.ok()) return published;
54 document_ = typed->document_;
55 return eve::editing::applied<void>();
56}
