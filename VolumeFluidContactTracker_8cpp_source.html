21bool keyLess(const VolumeFluidContact& a, const VolumeFluidContact& b) {
22 return a.colliderLabel < b.colliderLabel || (a.colliderLabel == b.colliderLabel && a.actorGroup < b.actorGroup);

27bool finiteContact(const VolumeFluidContact& contact) {
28 const auto finite3 = [](glm::vec3 value) {
29 return std::isfinite(value.x) && std::isfinite(value.y) && std::isfinite(value.z);
30 };
31 return finite3(contact.point) && finite3(contact.normal) && finite3(contact.impulse) &&
32 std::isfinite(contact.distance);

36Result<std::vector<VolumeFluidContactEvent>> VolumeFluidContactTracker::advance(
37 std::span<const VolumeFluidContact> contacts, float distanceThreshold) {
38 using Output = Result<std::vector<VolumeFluidContactEvent>>;
39 if (contacts.size() > 65536 || !std::isfinite(distanceThreshold) || distanceThreshold < 0.f ||
40 distanceThreshold > 1.f ||
41 std::any_of(contacts.begin(), contacts.end(), [](const auto& contact) { return !finiteContact(contact); }))
42 return Output::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "Invalid contact event batch",
43 "fluids.volume.contacts"));
44
45 scratch_.clear();
46 scratch_.reserve(std::max(scratch_.capacity(), contacts.size()));
47 for (const auto& contact : contacts)
48 if (contact.distance <= distanceThreshold) scratch_.push_back(contact);
49 std::stable_sort(scratch_.begin(), scratch_.end(), keyLess);
50 scratch_.erase(std::unique(scratch_.begin(), scratch_.end(), sameKey), scratch_.end());
51
52 std::vector<VolumeFluidContactEvent> events;
53 events.reserve(scratch_.size() + previous_.size());
54 size_t current = 0, previous = 0;
55 while (current < scratch_.size() && previous < previous_.size()) {
56 if (keyLess(scratch_[current], previous_[previous]))
57 events.push_back({VolumeFluidContactEventType::Enter, scratch_[current++]});
58 else if (keyLess(previous_[previous], scratch_[current]))
59 events.push_back({VolumeFluidContactEventType::Exit, previous_[previous++]});
60 else {
61 events.push_back({VolumeFluidContactEventType::Stay, scratch_[current++]});
62 ++previous;
63 }
64 }
65 while (current < scratch_.size()) events.push_back({VolumeFluidContactEventType::Enter, scratch_[current++]});
66 while (previous < previous_.size()) events.push_back({VolumeFluidContactEventType::Exit, previous_[previous++]});
67 previous_ = scratch_;
68 return Output::success(std::move(events));
69}

150Result<unsigned> VolumeFluidCoupling::step(physics::World3D& world, VolumeFluid& fluid, float dt, unsigned substeps,
151 std::span<const VolumeFluidThermalRule> rules) {
152 if (!world.isValid())
153 return Result<unsigned>::failure(
154 Diagnostic::error(DiagnosticCode::StaleHandle, "Rigid world is no longer valid", "fluids.volume.coupling"));
155 auto& bodies = impl_->bodies;
156 auto& samples = impl_->samples;
157 auto& sdfSamples = impl_->sdfSamples;
158 auto& centers = impl_->centers;
159 auto& labels = impl_->labels;
160 bodies.clear();
161 samples.clear();
162 sdfSamples.clear();
163 centers.clear();
164 labels.clear();
165 const size_t totalBindings = impl_->bindings.size() + impl_->sdfBindings.size();
166 bodies.reserve(totalBindings);
167 samples.reserve(impl_->bindings.size());
168 sdfSamples.reserve(impl_->sdfBindings.size());
169 centers.reserve(totalBindings);
170 labels.reserve(totalBindings);
171 for (const auto& binding : impl_->bindings) {
172 auto resolved = binding.link.resolve(world);
173 if (!resolved) return Result<unsigned>::failure(resolved.status());
174 auto* body = resolved.value();
175 const glm::quat orientation(body->getRotW(), body->getRotX(), body->getRotY(), body->getRotZ());
176 auto sample = binding.shape;
177 const auto center = body->localToWorldPointOwned(sample.center.x, sample.center.y, sample.center.z);
178 if (!center) return Result<unsigned>::failure(center.status());
179 const auto velocity = body->getLocalPointVelocityOwned(sample.center.x, sample.center.y, sample.center.z);
180 if (!velocity) return Result<unsigned>::failure(velocity.status());
181 sample.center = {center.value().x, center.value().y, center.value().z};
182 sample.velocity = {velocity.value().x, velocity.value().y, velocity.value().z};
183 sample.angularVelocity = {body->getAngularVelocityX(), body->getAngularVelocityY(),
184 body->getAngularVelocityZ()};
185 const auto q =
186 orientation * glm::quat(sample.rotation.w, sample.rotation.x, sample.rotation.y, sample.rotation.z);
187 sample.rotation = {q.x, q.y, q.z, q.w};
188 labels.emplace_back(sample.label, bodies.size());
189 bodies.push_back(body);
190 samples.push_back(sample);
191 centers.push_back({body->getWorldCenterX(), body->getWorldCenterY(), body->getWorldCenterZ()});
192 }
193 for (const auto& binding : impl_->sdfBindings) {
194 auto resolved = binding.link.resolve(world);
195 if (!resolved) return Result<unsigned>::failure(resolved.status());
196 auto* body = resolved.value();
197 const glm::quat orientation(body->getRotW(), body->getRotX(), body->getRotY(), body->getRotZ());
198 auto sample = binding.local;
199 const auto origin = body->localToWorldPointOwned(sample.position.x, sample.position.y, sample.position.z);
200 if (!origin) return Result<unsigned>::failure(origin.status());
201 const auto velocity = body->getLocalPointVelocityOwned(sample.position.x, sample.position.y, sample.position.z);
202 if (!velocity) return Result<unsigned>::failure(velocity.status());
203 sample.position = {origin.value().x, origin.value().y, origin.value().z};
204 sample.velocity = {velocity.value().x, velocity.value().y, velocity.value().z};
205 sample.angularVelocity = {body->getAngularVelocityX(), body->getAngularVelocityY(),
206 body->getAngularVelocityZ()};
207 const auto q =
208 orientation * glm::quat(sample.rotation.w, sample.rotation.x, sample.rotation.y, sample.rotation.z);
209 sample.rotation = {q.x, q.y, q.z, q.w};
210 labels.emplace_back(sample.label, bodies.size());
211 bodies.push_back(body);
212 sdfSamples.push_back(std::move(sample));
213 centers.push_back({body->getWorldCenterX(), body->getWorldCenterY(), body->getWorldCenterZ()});
214 }
215 std::sort(labels.begin(), labels.end());
216 // Allocate all output aggregation before advancing either domain.
217 auto& linear = impl_->linear;
218 auto& angular = impl_->angular;
219 linear.assign(bodies.size(), glm::vec3(0.f));
220 angular.assign(bodies.size(), glm::vec3(0.f));
221 Result<void> advanced = impl_->sdfBindings.empty()
222 ? fluid.stepWithColliders(dt, substeps, samples, rules)
223 : fluid.stepWithColliders(dt, substeps, samples, sdfSamples, rules);
224 if (!advanced) return Result<unsigned>::failure(advanced.status());
225 const auto contacts = fluid.contacts();
226 for (const auto& contact : contacts) {
227 const auto at =
228 std::lower_bound(labels.begin(), labels.end(), std::pair<unsigned, size_t>{contact.colliderLabel, 0});
229 if (at == labels.end() || at->first != contact.colliderLabel) continue;
230 const size_t i = at->second;
231 linear[i] += contact.impulse;
232 angular[i] += glm::cross(contact.point - centers[i], contact.impulse);
233 }
234 const auto reactions = fluid.attachmentReactions();
235 for (const auto& reaction : reactions) {
236 const auto at =
237 std::lower_bound(labels.begin(), labels.end(), std::pair<unsigned, size_t>{reaction.colliderLabel, 0});
238 if (at == labels.end() || at->first != reaction.colliderLabel) continue;
239 const size_t i = at->second;
240 linear[i] += reaction.impulse;
241 angular[i] += glm::cross(reaction.point - centers[i], reaction.impulse);
242 angular[i] += reaction.angularImpulse;
243 }
244 unsigned clampedBodyCount = 0;
245 for (size_t i = 0; i < bodies.size(); ++i) {
246 bool clamped = false;
247 const float mass = bodies[i]->getMass();
248 const float linearLength = glm::length(linear[i]);
249 if (mass > 0.f && linearLength > mass * impl_->maximumLinearDelta) {
250 linear[i] *= mass * impl_->maximumLinearDelta / linearLength;
251 clamped = true;
252 }
253 if (mass > 0.f && glm::dot(angular[i], angular[i]) > 0.f) {
254 const glm::mat3 localInertia(
255 bodies[i]->getInertiaXX(), bodies[i]->getInertiaXY(), bodies[i]->getInertiaXZ(),
256 bodies[i]->getInertiaXY(), bodies[i]->getInertiaYY(), bodies[i]->getInertiaYZ(),
257 bodies[i]->getInertiaXZ(), bodies[i]->getInertiaYZ(), bodies[i]->getInertiaZZ());
258 const float inertiaScale =
259 std::max({std::abs(localInertia[0][0]), std::abs(localInertia[0][1]), std::abs(localInertia[0][2]),
260 std::abs(localInertia[1][0]), std::abs(localInertia[1][1]), std::abs(localInertia[1][2]),
261 std::abs(localInertia[2][0]), std::abs(localInertia[2][1]), std::abs(localInertia[2][2])});
262 if (std::isfinite(inertiaScale) && inertiaScale > 0.f) {
263 const auto normalizedInertia = localInertia / inertiaScale;
264 const float determinant = glm::determinant(normalizedInertia);
265 if (std::isfinite(determinant) && std::abs(determinant) > 1e-6f) {
266 const glm::quat q(bodies[i]->getRotW(), bodies[i]->getRotX(), bodies[i]->getRotY(),
267 bodies[i]->getRotZ());
268 const auto rotation = glm::mat3_cast(q);
269 const auto angularDelta = rotation * (glm::inverse(normalizedInertia) / inertiaScale) *
270 glm::transpose(rotation) * angular[i];
271 const float angularLength = glm::length(angularDelta);
272 if (angularLength > impl_->maximumAngularDelta) {
273 angular[i] *= impl_->maximumAngularDelta / angularLength;
274 clamped = true;
275 }
276 }
277 }
278 }
279 if (glm::dot(linear[i], linear[i]) > 0.f) bodies[i]->applyLinearImpulse(linear[i].x, linear[i].y, linear[i].z);
280 if (glm::dot(angular[i], angular[i]) > 0.f)
281 bodies[i]->applyAngularImpulse(angular[i].x, angular[i].y, angular[i].z);
282 if (clamped) ++clampedBodyCount;
283 }
284 impl_->clampedBodyCount = clampedBodyCount;
285 return Result<unsigned>::success(unsigned(contacts.size() + reactions.size()));
286}

309Result<void> VolumeFluidDiffuse::advance(const VolumeFluid& fluid, float seconds, unsigned minimumNeighbors) {
310 if (!std::isfinite(seconds) || seconds <= 0.f || seconds > 1.f / 30.f || minimumNeighbors > 1000000)
311 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
312 "Invalid diffuse timestep or neighbor threshold",
313 "fluids.volume.diffuse"));
314 candidateScratch_.clear();
315 positionScratch_.clear();
316 candidateScratch_.reserve(particles_.size());
317 positionScratch_.reserve(particles_.size());
318 for (auto particle : particles_) {
319 particle.life -= seconds;
320 if (particle.life <= 0.f) continue;
321 candidateScratch_.push_back(particle);
322 positionScratch_.push_back(particle.position);
323 }
324 if (candidateScratch_.empty()) {
325 particles_.clear();
326 return Result<void>::success();
327 }
328 auto sampled = fluid.sampleFieldInto(positionScratch_, sampleScratch_);
329 if (!sampled) return Result<void>::failure(sampled.status());
330 size_t remaining = 0;
331 for (size_t i = 0; i < candidateScratch_.size(); ++i) {
332 const auto& sample = sampleScratch_[i];
333 if (sample.neighborCount < minimumNeighbors) continue;
334 auto particle = candidateScratch_[i];
335 particle.velocity = sample.velocity;
336 particle.position += particle.velocity * seconds;
337 if (!valid(particle))
338 return Result<void>::failure(Diagnostic::error(
339 DiagnosticCode::InvalidArgument, "Invalid diffuse advection result", "fluids.volume.diffuse"));
340 candidateScratch_[remaining++] = particle;
341 }
342 candidateScratch_.resize(remaining);
343 particles_.swap(candidateScratch_);
344 return Result<void>::success();
345}

411Result<unsigned> VolumeFluidFoam::advanceFiltered(const VolumeFluid& source, VolumeFluidDiffuse& pool, float seconds,
412 bool filterActor, unsigned actorGroup, float alpha) {
413 if (!range(seconds, 0.f, 1.f / 30.f) || !range(alpha, 0.f, 1.f))
414 return Result<unsigned>::failure(invalid("Invalid foam timestep or interpolation alpha"));
415 const double credit = credit_ + double(seconds) * settings_.rate;
416 const auto whole = unsigned(std::floor(credit));
417 const double remainder = credit - whole;
418 const unsigned budget = std::min({whole, settings_.maxPerStep, pool.availableCapacity()});
419 if (budget == 0) {
420 credit_ = remainder;
421 return Result<unsigned>::success(0);
422 }
423 if (source.particleCount() > 65536)
424 return Result<unsigned>::failure(invalid("Foam source exceeds 65536 particles"));
425 const auto particles = source.particleView();
426 positionScratch_.clear();
427 positionScratch_.reserve(particles.size());
428 for (size_t i = 0; i < particles.size(); ++i) {
429 const auto& p = particles[i];
430 if ((!filterActor || p.actorGroup == actorGroup) &&
431 (p.material.phase == VolumeFluidPhase::Liquid || p.material.phase == VolumeFluidPhase::Gas))
432 positionScratch_.push_back(source.interpolatedPositionUnchecked(i, alpha));
433 }
434 auto sampled = source.sampleFieldInto(positionScratch_, sampleScratch_);
435 if (!sampled) return Result<unsigned>::failure(sampled.status());
436 size_t eligibleCount = 0;
437 for (size_t i = 0; i < positionScratch_.size(); ++i) {
438 const auto& field = sampleScratch_[i];
439 const float curl2 = glm::dot(field.vorticity, field.vorticity);
440 if (!std::isfinite(curl2) || !std::isfinite(field.density))
441 return Result<unsigned>::failure(invalid("Invalid foam field sample"));
442 if (curl2 > settings_.vorticityThreshold * settings_.vorticityThreshold &&
443 field.density < settings_.densityThreshold)
444 ++eligibleCount;
445 }
446 const unsigned count = std::min(budget, unsigned(eligibleCount));
447 auto rng = randomState_;
448 batchScratch_.clear();
449 batchScratch_.reserve(count);
450 if (count != 0) {
451 const unsigned stride = unsigned(eligibleCount) / count;
452 const unsigned start = next(rng) % stride;
453 unsigned eligibleOrdinal = 0;
454 unsigned nextOrdinal = start;
455 for (size_t at = 0; at < positionScratch_.size() && batchScratch_.size() < count; ++at) {
456 const auto& field = sampleScratch_[at];
457 const float curl2 = glm::dot(field.vorticity, field.vorticity);
458 if (!(curl2 > settings_.vorticityThreshold * settings_.vorticityThreshold &&
459 field.density < settings_.densityThreshold))
460 continue;
461 if (eligibleOrdinal++ != nextOrdinal) continue;
462 batchScratch_.push_back(
463 {positionScratch_[at] + jitter(rng, settings_.randomness), field.velocity, settings_.lifetime});
464 nextOrdinal += stride;
465 }
466 }
467 auto admitted = pool.emit(batchScratch_);
468 if (!admitted) return Result<unsigned>::failure(admitted.status());
469 credit_ = remainder;
470 randomState_ = rng;
471 return Result<unsigned>::success(count);
472}
