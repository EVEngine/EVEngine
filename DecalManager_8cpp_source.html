31Frustum extractFrustum(const glm::mat4 &m) {
32 Frustum f;
33 const glm::vec4 r0(m[0][0], m[1][0], m[2][0], m[3][0]);
34 const glm::vec4 r1(m[0][1], m[1][1], m[2][1], m[3][1]);
35 const glm::vec4 r2(m[0][2], m[1][2], m[2][2], m[3][2]);
36 const glm::vec4 r3(m[0][3], m[1][3], m[2][3], m[3][3]);
37 auto norm = [](glm::vec4 &v) {
38 const float l = glm::length(glm::vec3(v));
39 if (l > 1e-8f) v /= l;
40 };
41 f.p[0] = r3 + r0; // left
42 f.p[1] = r3 - r0; // right
43 f.p[2] = r3 + r1; // bottom
44 f.p[3] = r3 - r1; // top
45 f.p[4] = r2; // near (Vulkan zero-to-one depth)
46 f.p[5] = r3 - r2; // far
47 for (auto &pl : f.p) norm(pl);
48 return f;
49}

51glm::mat4 decalModel(const DecalInstance &d) {
52 glm::vec3 normal(d.nx, d.ny, d.nz);
53 const float len = glm::length(normal);
54 if (len < 1e-6f) normal = glm::vec3(0.f, 1.f, 0.f);
55 else normal /= len;
56
57 // Lift slightly along the normal to avoid z-fighting with the surface.
58 glm::mat4 m = glm::translate(glm::mat4(1.f),
59 glm::vec3(d.x, d.y, d.z) + normal * 0.008f);
60 const glm::quat orient = glm::rotation(glm::vec3(0.f, 0.f, 1.f), normal);
61 m *= glm::mat4_cast(orient);
62 m = glm::rotate(m, d.yaw, glm::vec3(0.f, 0.f, 1.f));
63 m = glm::scale(m, glm::vec3(d.size, d.size, d.depth));
64 return m;
65}

81int DecalManager::project(float x, float y, float z, float nx, float ny, float nz,
82 graphics::Texture *albedo, const std::string &kind, float size,
83 float depth, bool randomYaw, int seed, float fadeIn, float lifetime,
84 float fadeOut, float normalStrength, float roughnessStrength,
85 float metalStrength, float emissiveStrength) {
86 const int limit = limitFor(kind);
87 if (limit > 0) {
88 int ofKind = 0;
89 for (const auto &d : decals_) {
90 if (d.kind == kind) ++ofKind;
91 }
92 while (ofKind >= limit) {
93 // Evict the oldest instance of this kind.
94 auto it = decals_.end();
95 for (auto i = decals_.begin(); i != decals_.end(); ++i) {
96 if (i->kind != kind) continue;
97 if (it == decals_.end() || i->age > it->age) it = i;
98 }
99 if (it == decals_.end()) break;
100 decals_.erase(it);
101 --ofKind;
102 }
103 }
104
105 DecalInstance d;
106 d.id = nextId_++;
107 d.x = x;
108 d.y = y;
109 d.z = z;
110 d.nx = nx;
111 d.ny = ny;
112 d.nz = nz;
113 d.yaw = randomYaw ? (seed != 0 ? float((seed * 2654435761u) % 6283u) / 1000.f
114 : float(std::rand()) / float(RAND_MAX) * 6.28318f)
115 : 0.f;
116 d.size = size > 0.f ? size : 0.5f;
117 d.depth = depth > 0.f ? depth : 0.15f;
118 d.lifetime = lifetime > 0.f ? lifetime : 0.f;
119 d.fadeIn = fadeIn > 0.f ? fadeIn : 0.f;
120 d.fadeOut = fadeOut > 0.f ? fadeOut : 0.f;
121 d.albedo = albedo;
122 d.normalStrength = normalStrength;
123 d.roughnessStrength = roughnessStrength;
124 d.metalStrength = metalStrength;
125 d.emissiveStrength = emissiveStrength;
126 d.kind = kind;
127 decals_.push_back(d);
128 return d.id;
129}

192int DecalManager::replace(int previousId, DecalInstance candidate) {
193 const float normalLength2 = candidate.nx * candidate.nx + candidate.ny * candidate.ny +
194 candidate.nz * candidate.nz;
195 if (!std::isfinite(normalLength2) || normalLength2 <= 1e-12f ||
196 !std::isfinite(candidate.size) || candidate.size <= 0.f ||
197 !std::isfinite(candidate.depth) || candidate.depth <= 0.f ||
198 candidate.blendMode < 0 || candidate.blendMode > 1)
199 return 0;
200 std::vector<DecalInstance> next = decals_;
201 if (previousId != 0) {
202 auto previous = std::find_if(next.begin(), next.end(),
203 [previousId](const DecalInstance& decal) { return decal.id == previousId; });
204 if (previous == next.end()) return 0;
205 next.erase(previous);
206 }
207 const int limit = limitFor(candidate.kind);
208 if (limit > 0) {
209 while (std::count_if(next.begin(), next.end(), [&](const DecalInstance& decal) {
210 return decal.kind == candidate.kind;
211 }) >= limit) {
212 auto oldest = next.end();
213 for (auto it = next.begin(); it != next.end(); ++it)
214 if (it->kind == candidate.kind &&
215 (oldest == next.end() || it->age > oldest->age)) oldest = it;
216 if (oldest == next.end()) break;
217 next.erase(oldest);
218 }
219 }
220 candidate.id = nextId_++;
221 candidate.age = 0.f;
222 next.push_back(std::move(candidate));
223 decals_.swap(next);
224 return decals_.back().id;
225}

237void DecalManager::update(float dt) {
238 if (dt <= 0.f) return;
239 for (auto it = decals_.begin(); it != decals_.end();) {
240 it->age += dt;
241 const bool expired =
242 it->lifetime > 0.f && it->age >= it->lifetime + std::max(it->fadeOut, 0.f);
243 if (expired)
244 it = decals_.erase(it);
245 else
246 ++it;
247 }

250void DecalManager::drawAll(graphics::Graphics &gfx, float eyeX, float eyeY, float eyeZ,
251 const glm::mat4 &viewProj, float aspect) {
252 (void)aspect;
253 const Frustum frustum = extractFrustum(viewProj);
254 const glm::vec3 eye(eyeX, eyeY, eyeZ);
255 constexpr float kMaxDist = 80.f;
256 constexpr float kMaxDistSq = kMaxDist * kMaxDist;
257 for (const auto &d : decals_) {
258 if (!d.albedo) continue;
259 const glm::vec3 c(d.x, d.y, d.z);
260 const glm::vec3 toEye = c - eye;
261 if (glm::dot(toEye, toEye) > kMaxDistSq) continue; // distance LOD
262 if (!frustum.sphereVisible(c, d.size * 1.25f)) continue;
263 float fade = 1.f;
264 if (d.fadeIn > 0.f && d.age < d.fadeIn) fade = std::min(1.f, d.age / d.fadeIn);
265 if (d.lifetime > 0.f) {
266 const float outT = d.age - d.lifetime;
267 if (outT >= 0.f) {
268 const float span = std::max(d.fadeOut, 0.001f);
269 fade *= std::max(0.f, 1.f - outT / span);
270 }
271 }
272 if (fade <= 0.001f) continue;
273 gfx.drawDecal(decalModel(d), d.albedo, d.normal, d.params, d.uvRect, fade,
274 d.normalStrength, d.roughnessStrength, d.metalStrength, d.emissiveStrength,
275 d.blendMode);
276 }
