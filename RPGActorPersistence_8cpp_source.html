25eve::Result<std::string> RPGActor::checkpointJson() const {
26 RPGActor *actor = const_cast<RPGActor *>(this);
27 auto statusRef = actor->statuses();
28 auto skillRef = actor->skills();
29 auto attributeRef = actor->attributes();
30 auto progressionRef = actor->progression();
31 auto vitalRef = actor->vitals();
32 auto traitRef = actor->traits();
33 auto classRef = actor->classInfo();
34 if (statusRef->container.effectCount() != 0 || skillRef->casting.active)
35 return checkpointFailure<std::string>(eve::DiagnosticCode::PreconditionViolation,
36 "actor checkpoint requires a safe point without active effects or casts");
37
38 eve::Value::Array baseAttributes;
39 for (int index = 0; index < attributeRef->values.attributeCount(); ++index) {
40 const std::string name = attributeRef->values.attributeAt(index);
41 eve::Value::Object entry;
42 entry.emplace("id", eve::Value(name));
43 entry.emplace("value", eve::Value(attributeRef->values.getBase(name)));
44 baseAttributes.emplace_back(std::move(entry));
45 }
46
47 eve::Value::Object progressionValue;
48 progressionValue.emplace("level", eve::Value(progressionRef->level));
49 progressionValue.emplace("xp", eve::Value(progressionRef->xp));
50 progressionValue.emplace("xpToNext", eve::Value(progressionRef->xpToNext));
51
52 eve::Value::Object vitalsValue;
53 for (const auto &[name, value] : vitalRef->current) vitalsValue.emplace(name, eve::Value(value));
54
55 std::vector<std::string> skillIds;
56 skillIds.reserve(skillRef->known.size());
57 for (const auto &[id, runtime] : skillRef->known) {
58 (void)runtime;
59 skillIds.push_back(id);
60 }
61 std::sort(skillIds.begin(), skillIds.end());
62 eve::Value::Array learnedSkills;
63 for (const auto &id : skillIds) learnedSkills.emplace_back(id);
64
65 eve::Value::Array traitValues;
66 for (const auto &trait : traitRef->active) {
67 eve::Value::Object encoded;
68 encoded.emplace("source", eve::Value(trait.source));
69 encoded.emplace("traitId", eve::Value(trait.traitId));
70 traitValues.emplace_back(std::move(encoded));
71 }
72
73 eve::Value::Object classValue;
74 classValue.emplace("id", eve::Value(classRef->classId));
75 classValue.emplace("skillsSyncedUpTo", eve::Value(classRef->skillsSyncedUpTo));
76
77 eve::Value::Object root;
78 root.emplace("baseAttributes", eve::Value(std::move(baseAttributes)));
79 root.emplace("class", eve::Value(std::move(classValue)));
80 root.emplace("learnedSkills", eve::Value(std::move(learnedSkills)));
81 root.emplace("progression", eve::Value(std::move(progressionValue)));
82 root.emplace("schema", eve::Value("eve.rpg.actor-checkpoint"));
83 root.emplace("traits", eve::Value(std::move(traitValues)));
84 root.emplace("version", eve::Value(1));
85 root.emplace("vitals", eve::Value(std::move(vitalsValue)));
86 return eve::Value(std::move(root)).toJson();
87}

89eve::Result<RPGActor::CheckpointCandidate> RPGActor::prepareCheckpointJson(std::string_view json) const {
90 RPGActor *actor = const_cast<RPGActor *>(this);
91 auto statusRef = actor->statuses();
92 auto skillRef = actor->skills();
93 auto attributeRef = actor->attributes();
94 auto progressionRef = actor->progression();
95 auto traitRef = actor->traits();
96 auto classRef = actor->classInfo();
97 if (statusRef->container.effectCount() != 0 || skillRef->casting.active)
98 return checkpointFailure<CheckpointCandidate>(
99 eve::DiagnosticCode::PreconditionViolation,
100 "actor checkpoint restore requires a clean preconfigured destination");
101 auto parsed = eve::Value::fromJson(json);
102 if (!parsed.ok()) return eve::Result<CheckpointCandidate>::failure(parsed.status());
103 const eve::Value &root = parsed.value();
104 if (!root.isObject())
105 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::ParseError,
106 "actor checkpoint root must be an object", "$");
107 const eve::Value *schema = root.find("schema");
108 if (!schema || !schema->isString() || schema->asString() != "eve.rpg.actor-checkpoint")
109 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::InvalidArgument,
110 "snapshot does not belong to RPGActor", "$.schema");
111 const eve::Value *version = root.find("version");
112 if (!version || !version->isInt64() || version->asInt() != 1)
113 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::UnknownVersion,
114 "unsupported RPG actor checkpoint version", "$.version");
115
116 const eve::Value *encodedClass = root.find("class");
117 const eve::Value *encodedTraits = root.find("traits");
118 const eve::Value *encodedAttributes = root.find("baseAttributes");
119 const eve::Value *encodedProgression = root.find("progression");
120 const eve::Value *encodedVitals = root.find("vitals");
121 const eve::Value *encodedSkills = root.find("learnedSkills");
122 if (!encodedClass || !encodedClass->isObject() || !encodedTraits || !encodedTraits->isArray() ||
123 !encodedAttributes || !encodedAttributes->isArray() || !encodedProgression || !encodedProgression->isObject() ||
124 !encodedVitals || !encodedVitals->isObject() || !encodedSkills || !encodedSkills->isArray())
125 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::ParseError,
126 "actor checkpoint participant fields have invalid types", "$");
127
128 const eve::Value *classId = encodedClass->find("id");
129 const eve::Value *skillsSynced = encodedClass->find("skillsSyncedUpTo");
130 if (!classId || !classId->isString() || !skillsSynced || !skillsSynced->isInt64() ||
131 classId->asString() != classRef->classId || skillsSynced->asInt() < 0)
132 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::Conflict,
133 "actor checkpoint class does not match its destination", "$.class");
134 if (encodedTraits->arraySize() != traitRef->active.size())
135 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::Conflict,
136 "actor checkpoint traits do not match its destination", "$.traits");
137 for (std::size_t index = 0; index < traitRef->active.size(); ++index) {
138 const eve::Value &encoded = encodedTraits->at(index);
139 const eve::Value *traitId = encoded.isObject() ? encoded.find("traitId") : nullptr;
140 const eve::Value *source = encoded.isObject() ? encoded.find("source") : nullptr;
141 if (!traitId || !traitId->isString() || !source || !source->isString() ||
142 traitId->asString() != traitRef->active[index].traitId || source->asString() != traitRef->active[index].source)
143 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::Conflict,
144 "actor checkpoint trait identity does not match", "$.traits");
145 }
146
147 CheckpointCandidate candidate{*attributeRef, {}, *progressionRef, {}, *classRef};
148 candidate.classInfo.skillsSyncedUpTo = static_cast<int>(skillsSynced->asInt());
149 if (encodedAttributes->arraySize() != static_cast<std::size_t>(candidate.attributes.values.attributeCount()))
150 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::Conflict,
151 "actor checkpoint attribute layout does not match", "$.baseAttributes");
152 std::unordered_set<std::string> seenAttributes;
153 for (std::size_t index = 0; index < encodedAttributes->arraySize(); ++index) {
154 const eve::Value &encoded = encodedAttributes->at(index);
155 const eve::Value *id = encoded.isObject() ? encoded.find("id") : nullptr;
156 const eve::Value *value = encoded.isObject() ? encoded.find("value") : nullptr;
157 if (!id || !id->isString() || !value || !value->isNumeric() || !std::isfinite(numericValue(*value)))
158 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::ParseError,
159 "actor base attribute is invalid", "$.baseAttributes");
160 if (!seenAttributes.emplace(id->asString()).second || !candidate.attributes.values.has(id->asString()))
161 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::Conflict,
162 "actor base attribute layout does not match", "$.baseAttributes");
163 candidate.attributes.values.setBase(id->asString(), numericValue(*value));
164 }
165
166 const eve::Value *level = encodedProgression->find("level");
167 const eve::Value *xp = encodedProgression->find("xp");
168 const eve::Value *xpToNext = encodedProgression->find("xpToNext");
169 if (!level || !level->isInt64() || level->asInt() < 1 || !xp || !xp->isNumeric() ||
170 !xpToNext || !xpToNext->isNumeric() || !std::isfinite(numericValue(*xp)) ||
171 !std::isfinite(numericValue(*xpToNext)) || numericValue(*xp) < 0.0 ||
172 numericValue(*xpToNext) <= numericValue(*xp))
173 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::InvalidArgument,
174 "actor checkpoint progression is invalid", "$.progression");
175 candidate.progression = {static_cast<int>(level->asInt()), numericValue(*xp), numericValue(*xpToNext)};
176
177 for (const auto &name : encodedVitals->keys()) {
178 const eve::Value *value = encodedVitals->find(name);
179 if (!value || !value->isNumeric() || !std::isfinite(numericValue(*value)) || numericValue(*value) < 0.0 ||
180 numericValue(*value) > candidate.attributes.values.getFinal(name))
181 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::InvalidArgument,
182 "actor checkpoint vital is outside its attribute bounds", "$.vitals." + name);
183 candidate.vitals.current.emplace(name, numericValue(*value));
184 }
185
186 std::unordered_set<std::string> seenSkills;
187 for (std::size_t index = 0; index < encodedSkills->arraySize(); ++index) {
188 const eve::Value &id = encodedSkills->at(index);
189 if (!id.isString() || id.asString().empty() || !seenSkills.emplace(id.asString()).second ||
190 !SkillRegistry::find(id.asString()))
191 return checkpointFailure<CheckpointCandidate>(eve::DiagnosticCode::NotFound,
192 "actor checkpoint skill is missing or invalid", "$.learnedSkills");
193 candidate.skills.known.emplace(id.asString(), SkillRuntime{});
194 }
195 candidate.skills.casting = CastingState{};
196 return eve::Result<CheckpointCandidate>::success(std::move(candidate));
197}
