66double readMotionNumber(const ssq::Object& table, const char* field, const char* what) {
67 if (table.getType() != ssq::Type::TABLE) throw Exception("%s must be a table (x/y/z or r/g/b/a)", what);
68 const ssq::Object value = table.find(field);
69 const ssq::Type type = value.getType();
70 if (type != ssq::Type::FLOAT && type != ssq::Type::INTEGER)
71 throw Exception("%s requires a numeric '%s'", what, field);
72 return value.toFloat();

236template <class Adapter>
237Adapter* adoptScriptMotionSink(ScriptMotionSinkStore<Adapter>& store, const ssq::Object& sink) {
238 const std::string name = "animation.motion-sink." + std::to_string(++scriptMotionSinkSequence());
239 auto published = eve::ScriptProviderRegistry::instance().publish(name, sink);
240 if (!published.ok()) throwMotionError(published.error());
241 auto adapter = std::make_unique<Adapter>(published.value());
242 Adapter* raw = adapter.get();
243 store.providerNames.push_back(name);
244 store.adapters.push_back(std::move(adapter));
245 return raw;
246}

256 store.providerNames.clear();

561template <class Traits>
562void exposeMotionTraitsBuilder(ssq::Table& table, const char* className) {
563 using Builder = ScriptMotionBuilderT<Traits>;
564 auto builder =
565 table.addClass<Builder>(className, std::function<Builder*()>([]() -> Builder* { return nullptr; }), true);
566 // These are lambdas rather than `&Builder::ease` (the shape the float builder uses)
567 // because `Builder` is a template-dependent name: the binding-contract generator indexes
568 // members by concrete class, and the same method name also exists on the concrete float
569 // builder with a different parameter type, so `&Builder::ease` resolved to two shapes and
570 // was reported as an unresolved binding. A lambda states the parameters directly, which is
571 // also the shape `to`/`bind` below already use.
572 builder.addFunc("ease", [](Builder* self, std::string kind) {
573 if (!self) throw Exception("%s.ease: builder is null", Traits::kType);
574 self->ease(std::move(kind));
575 });
576 builder.addFunc("delay", [](Builder* self, float seconds) {
577 if (!self) throw Exception("%s.delay: builder is null", Traits::kType);
578 self->delay(seconds);
579 });
580 builder.addFunc("loops", [](Builder* self, int count, std::string mode) {
581 if (!self) throw Exception("%s.loops: builder is null", Traits::kType);
582 self->loops(count, std::move(mode));
583 });
584 builder.addFunc("cancelOnError", [](Builder* self, bool enabled) {
585 if (!self) throw Exception("%s.cancelOnError: builder is null", Traits::kType);
586 self->cancelOnError(enabled);
587 });
588 if constexpr (Traits::kHasStyle) {
589 builder.addFunc("stylePunch", &Builder::stylePunch);
590 builder.addFunc("styleShake", &Builder::styleShake);
591 builder.addFunc("frequency", [](Builder* self, int count) {
592 if (!self) throw Exception("%s.frequency: builder is null", Traits::kType);
593 self->frequency(count);
594 });
595 builder.addFunc("dampingRatio", [](Builder* self, float ratio) {
596 if (!self) throw Exception("%s.dampingRatio: builder is null", Traits::kType);
597 self->dampingRatio(ratio);
598 });
599 builder.addFunc("seed", [](Builder* self, int value) {
600 if (!self) throw Exception("%s.seed: builder is null", Traits::kType);
601 self->seed(value);
602 });
603 }
604 builder.addFunc("to", [](Builder* self, const ssq::Object& sink) {
605 if (!self) throw Exception("%s.to: builder is null", Traits::kType);
606 self->to(sink);
607 });
608 builder.addFunc("bind", [](Builder* self, const ssq::Object& sink) -> ScriptMotionHandle* {
609 if (!self) throw Exception("%s.bind: builder is null", Traits::kType);
610 return self->bind(sink);
611 });
612 builder.addFunc("run", [](Builder* self) -> ScriptMotionHandle* {
613 if (!self) throw Exception("%s.run: builder is null", Traits::kType);
614 return self->run();
615 });
616}
