145 void append(ScriptMotionBuilder *builder) {
146 if (!builder) throw Exception("MotionSequence.append: builder is null");
147 auto taken = builder->takeDesc();
148 if (!taken.ok()) throwMotionError(taken.error());
149 auto desc = std::move(taken).value();
150 MotionBuilder bridge(animation_->motions(), desc.from, desc.to, desc.duration);
151 bridge.ease(desc.ease).delay(desc.delay).loops(desc.loops, desc.loopMode).cancelOnError(desc.cancelOnError)
152 .style(desc.style).frequency(desc.frequency).dampingRatio(desc.dampingRatio).seed(desc.seed);
153 if (desc.sink) bridge.to(*desc.sink);
154 auto scheduled = sequence_.append(std::move(bridge));
155 if (!scheduled.ok()) throwMotionError(scheduled.error());
156 }
