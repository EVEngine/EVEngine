45eve::Result<void> validateRpgDefinition(const EffectDefinition &definition) {
46 if (definition.id.empty())
47 return rejected<void>(eve::DiagnosticCode::InvalidArgument, "effect definition id must not be empty");
48 if (definition.durationPolicy != "instant" && definition.durationPolicy != "duration" &&
49 definition.durationPolicy != "infinite") {
50 return rejected<void>(eve::DiagnosticCode::InvalidArgument,
51 "unknown effect duration policy: " + definition.durationPolicy);
52 }
53 if (!std::isfinite(definition.duration) || definition.duration < 0.0f)
54 return rejected<void>(eve::DiagnosticCode::InvalidArgument,
55 "effect definition duration must be finite and non-negative");
56 if (!std::isfinite(definition.period) || definition.period < 0.0f)
57 return rejected<void>(eve::DiagnosticCode::InvalidArgument,
58 "effect definition period must be finite and non-negative");
59 if (definition.maxStacks < 0)
60 return rejected<void>(eve::DiagnosticCode::InvalidArgument, "effect definition maxStacks must be non-negative");
61 if (!isBuiltinStackPolicy(definition.stackPolicy) && !StatusSystem::hasStackPolicy(definition.stackPolicy)) {
62 return rejected<void>(eve::DiagnosticCode::InvalidArgument,
63 "unknown effect stack policy: " + definition.stackPolicy);
64 }
65
66 for (const auto &modifier : definition.modifiers) {
67 if (modifier.attribute.empty())
68 return rejected<void>(eve::DiagnosticCode::InvalidArgument, "effect modifier attribute must not be empty");
69 if (!std::isfinite(modifier.value))
70 return rejected<void>(eve::DiagnosticCode::InvalidArgument, "effect modifier value must be finite");
71
72 auto parsed = ::eve::attributes::parseAttributeOperation(modifier.op, modifier.value);
73 if (parsed.ok()) {
74 std::move(parsed).takeValue();
75 continue;
76 }
77 const eve::Status diagnostic = parsed.status();
78 if (!AttributeSystem::customOps().has(modifier.op)) return eve::Result<void>::failure(diagnostic);
79 parsed.ignore("custom RPG attribute operation is validated by the registry");
80 }
81
82 return eve::Result<void>::success();
83}

85LifecycleDefinition makeLifecycleDefinition(const EffectDefinition &definition, ::eve::effects::StackMode stackMode,
86 ::eve::effects::StackCountPolicy stackCountPolicy,
87 ::eve::effects::DurationPolicy durationPolicy, std::uint32_t stackCount = 1,
88 double duration = -1.0) {
89 LifecycleDefinition result;
90 result.id = definition.id;
91 // One RPG actor owns one container, so the actor-local subject is enough;
92 // the canonical string instance id remains the identity across all maps.
93 result.stackKey = definition.id;
94 result.priority = 0;
95 result.duration =
96 duration >= 0.0 ? duration : (definition.durationPolicy == "duration" ? double(definition.duration) : 0.0);
97 result.magnitude = 0.0;
98 result.stackCount = stackCount;
99 result.policy.stackMode = stackMode;
100 result.policy.stackCount = stackCountPolicy;
101 result.policy.duration = durationPolicy;
102 result.policy.magnitude = ::eve::effects::MagnitudePolicy::Keep;
103 result.policy.overflow = ::eve::effects::OverflowPolicy::Reject;
104 result.policy.maxStacks = definition.maxStacks == 0 ? 0u : static_cast<std::uint32_t>(definition.maxStacks);
105 result.tags = definition.tags;
106 for (const auto &[key, value] : definition.extra) result.payload.setString(key, value);
107 return result;
108}

144 : static_cast<int>(effect.stackCount);

166 const EffectDefinition &definition) {
167 (void)definition;
168 if (candidate.instanceId != current.instanceId || candidate.effectId != current.effectId)
169 return rejected<void>(eve::DiagnosticCode::InvariantViolation,

192eve::Result<std::vector<ModifierBinding>> prepareModifierBindings(::eve::attributes::AttributeSet &attributes,
193 const EffectDefinition &definition,
194 const LifecycleInstance &effect,
195 const StatusMetadata &oldMetadata) {
196 auto removed = removeBindings(attributes, oldMetadata);
197 if (!removed.ok()) return eve::Result<std::vector<ModifierBinding>>::failure(removed.status());
198
199 // A definition reload or adapter re-application may change an effect from
200 // a direct modifier into a periodic settlement effect. Remove bindings
201 // first, then leave the periodic effect to the RPG tick/settlement path.
202 if (definition.period > 0.0f) return eve::Result<std::vector<ModifierBinding>>::success({});
203
204 std::vector<ModifierBinding> bindings;
205 bindings.reserve(definition.modifiers.size());
206 const double stackScale = static_cast<double>(effect.stackCount);
207 for (std::size_t index = 0; index < definition.modifiers.size(); ++index) {
208 const auto &spec = definition.modifiers[index];
209 const double value = spec.value * stackScale;
210 if (!std::isfinite(value))
211 return rejected<std::vector<ModifierBinding>>(eve::DiagnosticCode::InvalidArgument,
212 "effect modifier value overflowed during stack scaling");
213
214 AttributeModifier modifier;
215 modifier.id = "status:" + effect.id + ":modifier:" + std::to_string(index);
216 modifier.attribute = spec.attribute;
217 modifier.source = "status:" + definition.id + ":" + effect.id;
218 modifier.priority = spec.priority;
219
220 auto parsed = ::eve::attributes::parseAttributeOperation(spec.op, value);
221 if (parsed.ok()) {
222 const auto normalized = std::move(parsed).takeValue();
223 modifier.operation = normalized.operation;
224 modifier.value = normalized.value;
225 } else {
226 const eve::Status diagnostic = parsed.status();
227 if (!AttributeSystem::customOps().has(spec.op))
228 return eve::Result<std::vector<ModifierBinding>>::failure(diagnostic);
229 parsed.ignore("custom RPG attribute operation is resolved by AttributeSet");
230 modifier.operation = AttributeOperation::Custom;
231 modifier.policyId = spec.op;
232 modifier.value = value;
233 }
234
235 auto added = attributes.addModifier(std::move(modifier));
236 if (!added.ok()) return eve::Result<std::vector<ModifierBinding>>::failure(added.status());
237 const std::string modifierId = std::move(added).takeValue();
238 bindings.emplace_back(spec.attribute, modifierId);
239 }
240 return eve::Result<std::vector<ModifierBinding>>::success(std::move(bindings));
241}

315eve::Result<int> StatusSystem::apply(RPGActor *actor, const std::string &effectId, const std::string &source) {
316 if (!actor) return rejected<int>(eve::DiagnosticCode::InvalidArgument, "status application requires an actor");
317
318 const EffectDefinition *definition = EffectRegistry::find(effectId);
319 if (!definition)
320 return rejected<int>(eve::DiagnosticCode::NotFound, "effect definition was not found: " + effectId,
321 eve::StatusCode::NotFound);
322
323 auto definitionValidation = validateRpgDefinition(*definition);
324 if (!definitionValidation.ok()) return eve::Result<int>::failure(definitionValidation.status());
325
326 for (const auto &[name, condition] : applyConditions()) {
327 if (!condition) continue;
328 std::string reason;
329 if (!condition(actor, *definition, source, reason)) {
330 StatusChangeEvent event;
331 event.actor = actor;
332 event.effectId = effectId;
333 event.source = source;
334 event.action = "reject";
335 event.reason = reason.empty() ? name : reason;
336 const std::string eventReason = event.reason;
337 emitChange(std::move(event));
338 return rejected<int>(eve::DiagnosticCode::PreconditionViolation,
339 "status application condition rejected the effect: " + eventReason);
340 }
341 }
342
343 if (definition->durationPolicy == "instant") {
344 auto candidateAttributes = actor->attributes()->values;
345 for (const auto &modifier : definition->modifiers) {
346 const double next = candidateAttributes.getBase(modifier.attribute) + modifier.value;
347 if (!std::isfinite(next))
348 return rejected<int>(eve::DiagnosticCode::InvalidArgument,
349 "instant effect modifier overflowed the base attribute");
350 candidateAttributes.modifyBase(modifier.attribute, modifier.value);
351 }
352 actor->attributes()->values = std::move(candidateAttributes);
353
354 StatusChangeEvent event;
355 event.actor = actor;
356 event.effectId = effectId;
357 event.source = source;
358 event.action = "apply";
359 emitChange(std::move(event));
360 return eve::Result<int>::success(0, eve::Status::success(eve::StatusCode::Applied));
361 }
362
363 auto statuses = actor->statuses();
364 LifecycleInstance *existing = findByType(*statuses, effectId);
365 if (existing) {
366 const std::string canonicalId = existing->id;
367 const int legacyId = legacyIdFor(*statuses, canonicalId);
368 const auto metadataIt = statuses->metadata.find(canonicalId);
369 if (legacyId <= 0 || metadataIt == statuses->metadata.end())
370 return rejected<int>(eve::DiagnosticCode::InvariantViolation,
371 "RPG status adapter mapping is missing for an active effect", eve::StatusCode::Failed);
372
373 const StatusMetadata oldMetadata = metadataIt->second;
374 const StatusInstance current = project(*existing, oldMetadata, legacyId);
375 const std::string &policy = definition->stackPolicy;
376
377 if (!isBuiltinStackPolicy(policy)) {
378 const auto custom = stackPolicies().find(policy);
379 if (custom == stackPolicies().end() || !custom->second)
380 return rejected<int>(eve::DiagnosticCode::InvalidArgument,
381 "effect stack policy is not callable: " + policy);
382
383 StatusInstance candidate = current;
384 const int policyResult = custom->second(actor, candidate, *definition, source);
385 if (policyResult < 0) {
386 StatusChangeEvent event;
387 event.actor = actor;
388 event.instanceId = legacyId;
389 event.effectId = effectId;
390 event.source = source;
391 event.action = "reject";
392 event.stacks = current.stacks;
393 event.reason = "stackPolicy:" + policy;
394 emitChange(std::move(event));
395 return rejected<int>(eve::DiagnosticCode::Conflict,
396 "custom effect stack policy rejected the application", eve::StatusCode::Conflict);
397 }
398 if (policyResult == 0) return eve::Result<int>::success(0, eve::Status::success(eve::StatusCode::NoOp));
399
400 auto candidateValidation = validateCandidate(candidate, current, *definition);
401 if (!candidateValidation.ok()) return eve::Result<int>::failure(candidateValidation.status());
402 if (policyResult != legacyId)
403 return rejected<int>(eve::DiagnosticCode::InvariantViolation,
404 "custom effect stack policy returned a foreign status id",
405 eve::StatusCode::Failed);
406
407 const double candidateDuration = candidate.remaining < 0.0f ? 0.0 : candidate.remaining;
408 auto lifecycle =
409 makeLifecycleDefinition(*definition, ::eve::effects::StackMode::Reuse,
410 ::eve::effects::StackCountPolicy::Set, ::eve::effects::DurationPolicy::Replace,
411 static_cast<std::uint32_t>(candidate.stacks), candidateDuration);
412 // A registered custom policy owns its own stack limit semantics.
413 // The generic container still supplies atomic instance replacement,
414 // but must not impose the legacy RPG default maxStacks of one.
415 lifecycle.policy.maxStacks = 0;
416 auto candidateContainer = statuses->container;
417 auto applied = candidateContainer.apply(lifecycle, "actor", source);
418 if (!applied.ok()) return eve::Result<int>::failure(applied.status());
419 const std::string updatedId = std::move(applied).takeValue();
420 if (updatedId != canonicalId)
421 return rejected<int>(eve::DiagnosticCode::InvariantViolation,
422 "effect container changed an existing status identity", eve::StatusCode::Failed);
423 LifecycleInstance *updated = candidateContainer.find(canonicalId);
424 if (!updated)
425 return rejected<int>(eve::DiagnosticCode::InvariantViolation,
426 "effect container lost an updated status instance", eve::StatusCode::Failed);
427 updated->source = candidate.source;
428 const int updatedStacks = candidate.stacks;
429
430 auto candidateAttributes = actor->attributes()->values;
431 auto bindings = prepareModifierBindings(candidateAttributes, *definition, *updated, oldMetadata);
432 if (!bindings.ok()) return eve::Result<int>::failure(bindings.status());
433
434 auto candidateMetadata = statuses->metadata;
435 auto &newMetadata = candidateMetadata[canonicalId];
436 newMetadata.periodAccum = candidate.periodAccum;
437 newMetadata.props = candidate.props;
438 newMetadata.appliedModifiers = std::move(bindings).takeValue();
439
440 auto candidateLegacyIdByEffect = statuses->legacyIdByEffect;
441 auto candidateEffectByLegacyId = statuses->effectByLegacyId;
442 commit(actor, std::move(candidateContainer), std::move(candidateMetadata),
443 std::move(candidateLegacyIdByEffect), std::move(candidateEffectByLegacyId), statuses->nextInstanceId,
444 std::move(candidateAttributes));
445
446 StatusChangeEvent event;
447 event.actor = actor;
448 event.instanceId = legacyId;
449 event.effectId = effectId;
450 event.source = source;
451 event.action = "stack";
452 event.stacks = updatedStacks;
453 emitChange(std::move(event));
454 return eve::Result<int>::success(legacyId, eve::Status::success(eve::StatusCode::Applied));
455 }
456
457 if (policy == "none") {
458 StatusChangeEvent event;
459 event.actor = actor;
460 event.instanceId = legacyId;
461 event.effectId = effectId;
462 event.source = source;
463 event.action = "reject";
464 event.stacks = current.stacks;
465 event.reason = "stackPolicy:none";
466 emitChange(std::move(event));
467 return rejected<int>(eve::DiagnosticCode::Conflict, "effect is already active and uses stackPolicy:none",
468 eve::StatusCode::Conflict);
469 }
470
471 ::eve::effects::StackCountPolicy stackCount = ::eve::effects::StackCountPolicy::Keep;
472 ::eve::effects::DurationPolicy durationPolicy = ::eve::effects::DurationPolicy::Replace;
473 ::eve::effects::OverflowPolicy overflow = ::eve::effects::OverflowPolicy::Reject;
474 std::string action = policy;
475 if (policy == "refresh") {
476 durationPolicy = ::eve::effects::DurationPolicy::Replace;
477 } else if (policy == "extend") {
478 durationPolicy = ::eve::effects::DurationPolicy::Extend;
479 } else if (policy == "stack") {
480 stackCount = ::eve::effects::StackCountPolicy::Increment;
481 overflow = ::eve::effects::OverflowPolicy::Clamp;
482 } else {
483 return rejected<int>(eve::DiagnosticCode::InvalidArgument,
484 "unknown built-in effect stack policy: " + policy);
485 }
486
487 auto lifecycle =
488 makeLifecycleDefinition(*definition, ::eve::effects::StackMode::Reuse, stackCount, durationPolicy);
489 lifecycle.policy.overflow = overflow;
490 auto candidateContainer = statuses->container;
491 auto applied = candidateContainer.apply(lifecycle, "actor", source);
492 if (!applied.ok()) return eve::Result<int>::failure(applied.status());
493 const std::string updatedId = std::move(applied).takeValue();
494 if (updatedId != canonicalId)
495 return rejected<int>(eve::DiagnosticCode::InvariantViolation,
496 "effect container changed an existing status identity", eve::StatusCode::Failed);
497 LifecycleInstance *updated = candidateContainer.find(canonicalId);
498 if (!updated)
499 return rejected<int>(eve::DiagnosticCode::InvariantViolation,
500 "effect container lost an updated status instance", eve::StatusCode::Failed);
501
502 auto candidateAttributes = actor->attributes()->values;
503 auto bindings = prepareModifierBindings(candidateAttributes, *definition, *updated, oldMetadata);
504 if (!bindings.ok()) return eve::Result<int>::failure(bindings.status());
505 const int updatedStacks = updated->stackCount > static_cast<std::uint32_t>(std::numeric_limits<int>::max())
506 ? std::numeric_limits<int>::max()
507 : static_cast<int>(updated->stackCount);
508 auto candidateMetadata = statuses->metadata;
509 candidateMetadata[canonicalId].appliedModifiers = std::move(bindings).takeValue();
510 auto candidateLegacyIdByEffect = statuses->legacyIdByEffect;
511 auto candidateEffectByLegacyId = statuses->effectByLegacyId;
512 commit(actor, std::move(candidateContainer), std::move(candidateMetadata), std::move(candidateLegacyIdByEffect),
513 std::move(candidateEffectByLegacyId), statuses->nextInstanceId, std::move(candidateAttributes));
514
515 StatusChangeEvent event;
516 event.actor = actor;
517 event.instanceId = legacyId;
518 event.effectId = effectId;
519 event.source = source;
520 event.action = action;
521 event.stacks = updatedStacks;
522 emitChange(std::move(event));
523 return eve::Result<int>::success(legacyId, eve::Status::success(eve::StatusCode::Applied));
524 }
525
526 auto candidateContainer = statuses->container;
527 const auto lifecycle = makeNewLifecycleDefinition(*definition);
528 auto applied = candidateContainer.apply(lifecycle, "actor", source);
529 if (!applied.ok()) return eve::Result<int>::failure(applied.status());
530 const std::string canonicalId = std::move(applied).takeValue();
531 LifecycleInstance *created = candidateContainer.find(canonicalId);
532 if (!created)
533 return rejected<int>(eve::DiagnosticCode::InvariantViolation,
534 "effect container did not retain a newly applied status", eve::StatusCode::Failed);
535

596 auto candidateLegacyIdByEffect = statuses->legacyIdByEffect;

598 candidateMetadata.erase(canonicalId);
599 candidateLegacyIdByEffect.erase(canonicalId);
600 candidateEffectByLegacyId.erase(instanceId);

602 std::move(candidateEffectByLegacyId), statuses->nextInstanceId, std::move(candidateAttributes));

604 StatusChangeEvent event;

606 event.instanceId = old.instanceId;
607 event.effectId = old.effectId;
608 event.source = old.source;
609 event.action = "remove";

729eve::Result<StatusUpdateSummary> StatusSystem::update(double dt) {
730 if (!std::isfinite(dt) || dt < 0.0)
731 return rejected<StatusUpdateSummary>(eve::DiagnosticCode::InvalidArgument,
732 "status update delta must be finite and non-negative");
733
734 StatusUpdateSummary total;
735 total.elapsedSeconds = dt;
736
737 for (RPGActor *actor : RPGActor::liveActors()) {
738 if (!actor) continue;
739 auto statuses = actor->statuses();
740 std::vector<std::string> activeIds;
741 activeIds.reserve(static_cast<std::size_t>(statuses->container.effectCount()));
742 for (int i = 0; i < statuses->container.effectCount(); ++i) {
743 const LifecycleInstance *effect = statuses->container.effectAt(i);
744 if (effect) activeIds.push_back(effect->id);
745 }
746
747 auto candidateContainer = statuses->container;
748 auto candidateMetadata = statuses->metadata;
749 auto candidateLegacyIdByEffect = statuses->legacyIdByEffect;
750 auto candidateEffectByLegacyId = statuses->effectByLegacyId;
751 auto candidateAttributes = actor->attributes()->values;
752 std::vector<StatusTickEvent> pendingTicks;
753 std::vector<StatusChangeEvent> pendingChanges;
754
755 for (const auto &canonicalId : activeIds) {
756 const LifecycleInstance *effect = statuses->container.find(canonicalId);
757 auto metadataIt = candidateMetadata.find(canonicalId);
758 if (!effect || metadataIt == candidateMetadata.end())
759 return rejected<StatusUpdateSummary>(eve::DiagnosticCode::InvariantViolation,
760 "RPG status executor metadata is out of sync",
761 eve::StatusCode::Failed);
762 const EffectDefinition *definition = EffectRegistry::find(effect->type);
763 if (!definition || definition->period <= 0.0f) continue;
764
765 double available = dt;
766 if (effect->remaining >= 0.0) available = std::min(available, effect->remaining);
767 double accumulated = metadataIt->second.periodAccum + available;
768 const double period = static_cast<double>(definition->period);
769 const int legacyId = legacyIdFor(*statuses, canonicalId);
770 if (legacyId <= 0)
771 return rejected<StatusUpdateSummary>(eve::DiagnosticCode::InvariantViolation,
772 "RPG status executor has no legacy id mapping",
773 eve::StatusCode::Failed);
774 while (accumulated >= period) {
775 accumulated -= period;
776 StatusTickEvent tick;
777 tick.actor = actor;
778 tick.instanceId = legacyId;
779 tick.effectId = effect->type;
780 tick.source = effect->source;
781 tick.stacks = effect->stackCount > static_cast<std::uint32_t>(std::numeric_limits<int>::max())
782 ? std::numeric_limits<int>::max()
783 : static_cast<int>(effect->stackCount);
784 pendingTicks.push_back(std::move(tick));
785 }
786 metadataIt->second.periodAccum = accumulated;
787 }
788
789 auto advanced = candidateContainer.update(dt);
790 if (!advanced.ok()) return eve::Result<StatusUpdateSummary>::failure(advanced.status());
791 const auto updateSummary = std::move(advanced).takeValue();
792
793 for (const auto &canonicalId : activeIds) {
794 if (candidateContainer.find(canonicalId)) continue;
795 const LifecycleInstance *oldEffect = statuses->container.find(canonicalId);
796 const auto oldMetadata = statuses->metadata.find(canonicalId);
797 const auto mapping = statuses->legacyIdByEffect.find(canonicalId);
798 if (!oldEffect || oldMetadata == statuses->metadata.end() || mapping == statuses->legacyIdByEffect.end())
799 return rejected<StatusUpdateSummary>(eve::DiagnosticCode::InvariantViolation,
800 "expired RPG status has incomplete adapter metadata",
801 eve::StatusCode::Failed);
802
803 auto bindingsRemoved = removeBindings(candidateAttributes, oldMetadata->second);
804 if (!bindingsRemoved.ok()) return eve::Result<StatusUpdateSummary>::failure(bindingsRemoved.status());
805
806 StatusChangeEvent expired;
807 expired.actor = actor;
808 expired.instanceId = mapping->second;
809 expired.effectId = oldEffect->type;
810 expired.source = oldEffect->source;
811 expired.action = "expire";
812 expired.stacks = oldEffect->stackCount > static_cast<std::uint32_t>(std::numeric_limits<int>::max())
813 ? std::numeric_limits<int>::max()
814 : static_cast<int>(oldEffect->stackCount);
815 pendingChanges.push_back(std::move(expired));
816 candidateMetadata.erase(canonicalId);
817 candidateLegacyIdByEffect.erase(canonicalId);
818 candidateEffectByLegacyId.erase(mapping->second);
819 }
820
821 commit(actor, std::move(candidateContainer), std::move(candidateMetadata), std::move(candidateLegacyIdByEffect),
822 std::move(candidateEffectByLegacyId), statuses->nextInstanceId, std::move(candidateAttributes));
823 for (auto &tick : pendingTicks) {
824 tickQueue().push_back(std::move(tick));
825 ++total.ticks;
826 }
827 for (auto &change : pendingChanges) {
828 emitChange(std::move(change));
829 ++total.expired;
830 }
831 (void)updateSummary;
832 }
833 return eve::Result<StatusUpdateSummary>::success(std::move(total), eve::Status::success(eve::StatusCode::Applied));
834}
