32eve::Result<HouseComponent> parseComponent(Value object) {
33 if (!object.isObject())
34 return failure<HouseComponent>(eve::DiagnosticCode::ParseError, "component must be an object");
35
36 HouseComponent out;
37 out.id = object.getString("id");
38 out.modelPath = object.getString("model");
39 out.category = object.getString("category");
40 out.width = object.getInt("width", 1);
41 out.depth = object.getInt("depth", 1);
42 out.height = object.getInt("height", 1);
43 out.weight = object.getInt("weight", 1);
44 out.tags = object.getStringArray("tags");
45 if (object.get("rotations").isArray()) out.rotations = object.getIntArray("rotations");
46
47 const Value sockets = object.get("sockets");
48 for (size_t i = 0; i < sockets.size(); ++i) {
49 const Value socketObject = sockets.at(i);
50 if (!socketObject.isObject())
51 return failure<HouseComponent>(eve::DiagnosticCode::ParseError, "socket must be an object", "sockets");
52 auto direction = parseDirection(socketObject.getString("direction"));
53 if (!direction.ok()) return eve::Result<HouseComponent>::failure(direction.status());
54
55 HouseSocket socket;
56 socket.direction = std::move(direction).takeValue();
57 socket.type = socketObject.getString("type");
58 socket.accepts = socketObject.getStringArray("accepts");
59 if (socket.type.empty())
60 return failure<HouseComponent>(eve::DiagnosticCode::InvalidArgument,
61 "socket needs a valid direction and type", "sockets");
62 out.sockets.push_back(std::move(socket));
63 }
64
65 const Value material = object.get("material");
66 if (material.isObject()) {
67 const Value color = material.get("baseColor");
68 if (color.isArray()) {
69 if (color.size() != 3 && color.size() != 4)
70 return failure<HouseComponent>(eve::DiagnosticCode::InvalidArgument,
71 "material baseColor needs 3 or 4 values", "material.baseColor");
72 out.material.hasBaseColor = true;
73 out.material.baseColorR = color.at(0).asFloat();
74 out.material.baseColorG = color.at(1).asFloat();
75 out.material.baseColorB = color.at(2).asFloat();
76 out.material.baseColorA = color.size() == 4 ? color.at(3).asFloat() : 1.f;
77 }
78 out.material.baseColorTexture = material.getString("baseColorTexture");
79 out.material.normalTexture = material.getString("normalTexture");
80 out.material.heightTexture = material.getString("heightTexture");
81 if (material.has("metallic")) {
82 out.material.hasMetallic = true;
83 out.material.metallic = material.getFloat("metallic");
84 }
85 if (material.has("roughness")) {
86 out.material.hasRoughness = true;
87 out.material.roughness = material.getFloat("roughness");
88 }
89 out.material.parallaxScale = material.getFloat("parallaxScale", 0.f);
90 out.material.parallaxMinLayers = material.getFloat("parallaxMinLayers", 8.f);
91 out.material.parallaxMaxLayers = material.getFloat("parallaxMaxLayers", 32.f);
92 out.material.cellBombScale = material.getFloat("cellBombScale", 4.f);
93 out.material.cellBombStrength = material.getFloat("cellBombStrength", 0.f);
94 out.material.cellBombRotation = material.getFloat("cellBombRotation", 1.f);
95 }

101eve::Result<void> HouseComponentLibrary::registerComponent(const HouseComponent &component) {
102 if (component.id.empty() || component.modelPath.empty() || component.category.empty())
103 return failure<void>(eve::DiagnosticCode::InvalidArgument, "component needs id, model and category");
104 if (component.width < 1 || component.depth < 1 || component.height < 1 || component.weight < 1)
105 return failure<void>(eve::DiagnosticCode::InvalidArgument, "component dimensions and weight must be positive");
106 for (const int rotation : component.rotations) {
107 if (rotation < 0 || rotation >= 360 || rotation % 90 != 0)
108 return failure<void>(eve::DiagnosticCode::InvalidArgument, "rotations must be cardinal degrees",
109 "rotations");
110 }
111
112 const auto inUnitRange = [](float value) { return value >= 0.f && value <= 1.f; };
113 if ((component.material.hasBaseColor &&
114 (!inUnitRange(component.material.baseColorR) || !inUnitRange(component.material.baseColorG) ||
115 !inUnitRange(component.material.baseColorB) || !inUnitRange(component.material.baseColorA))) ||
116 (component.material.hasMetallic && !inUnitRange(component.material.metallic)) ||
117 (component.material.hasRoughness && !inUnitRange(component.material.roughness)) ||
118 !inUnitRange(component.material.cellBombStrength) || !inUnitRange(component.material.cellBombRotation))
119 return failure<void>(eve::DiagnosticCode::InvalidArgument, "material values must be between 0 and 1",
120 "material");
121 if (component.material.parallaxScale < 0.f || component.material.parallaxScale > 0.2f ||
122 component.material.parallaxMinLayers < 1.f ||
123 component.material.parallaxMaxLayers < component.material.parallaxMinLayers ||
124 component.material.cellBombScale <= 0.f)
125 return failure<void>(eve::DiagnosticCode::InvalidArgument, "material sampling parameters are invalid",
126 "material");
127 if (components_.contains(component.id))
128 return failure<void>(eve::DiagnosticCode::Conflict, "duplicate component id: " + component.id, "id");
129
130 components_.emplace(component.id, component);
131 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
132}

134eve::Result<void> HouseComponentLibrary::loadFromJson(std::string_view json) {
135 std::string parseError;
136 const eve::json::Document document = eve::json::Document::parse(std::string(json), &parseError);
137 if (!document.valid())
138 return failure<void>(eve::DiagnosticCode::ParseError,
139 parseError.empty() ? "invalid component manifest" : parseError);
140
141 const Value root = document.root();
142 const Value values = root.isArray() ? root : root.get("components");
143 if (!values.isArray())
144 return failure<void>(eve::DiagnosticCode::ParseError, "expected components array", "components");
145
146 HouseComponentLibrary candidate;
147 for (size_t i = 0; i < values.size(); ++i) {
148 auto component = parseComponent(values.at(i));
149 if (!component.ok()) return eve::Result<void>::failure(component.status());
150 auto registered = candidate.registerComponent(std::move(component).takeValue());
151 if (!registered.ok()) return registered;
152 }
153 components_ = std::move(candidate.components_);
154 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
155}

157eve::Result<void> HouseComponentLibrary::loadFromFile(std::string_view filename) {
158 const std::string path(filename);
159 std::ifstream input(path, std::ios::binary);
160 if (!input) return failure<void>(eve::DiagnosticCode::NotFound, "cannot open component manifest: " + path, path);
161
162 std::ostringstream json;
163 json << input.rdbuf();
164 HouseComponentLibrary candidate;
165 auto loaded = candidate.loadFromJson(json.str());
166 if (!loaded.ok()) return loaded;
167
168 const std::filesystem::path assetRoot = std::filesystem::absolute(std::filesystem::path(path)).parent_path();
169 for (auto &[_, component] : candidate.components_) {
170 std::filesystem::path model(component.modelPath);
171 if (model.is_relative()) component.modelPath = (assetRoot / model).lexically_normal().string();
172 }
173 components_ = std::move(candidate.components_);
174 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
175}

195std::vector<std::reference_wrapper<const HouseComponent>> HouseComponentLibrary::byCategory(
196 std::string_view category, std::string_view style) const {
