123eve::Result<void> HouseComponentLibrary::registerComponent(const HouseComponent &component) {
124 if (component.id.empty() || component.modelPath.empty() || component.category.empty())
125 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
126 "component needs id, model and category", {}, {},
127 "housegen.components"));
128 if (component.width < 1 || component.depth < 1 || component.height < 1 || component.weight < 1)
129 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
130 "component dimensions and weight must be positive", {},
131 {}, "housegen.components"));
132 for (const int rotation : component.rotations) {
133 if (rotation < 0 || rotation >= 360 || rotation % 90 != 0)
134 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
135 "rotations must be cardinal degrees", "rotations",
136 {}, "housegen.components"));
137 }
138
139 const auto inUnitRange = [](float value) { return value >= 0.f && value <= 1.f; };
140 if ((component.material.hasBaseColor &&
141 (!inUnitRange(component.material.baseColorR) || !inUnitRange(component.material.baseColorG) ||
142 !inUnitRange(component.material.baseColorB) || !inUnitRange(component.material.baseColorA))) ||
143 (component.material.hasMetallic && !inUnitRange(component.material.metallic)) ||
144 (component.material.hasRoughness && !inUnitRange(component.material.roughness)) ||
145 !inUnitRange(component.material.cellBombStrength) || !inUnitRange(component.material.cellBombRotation))
146 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
147 "material values must be between 0 and 1", "material",
148 {}, "housegen.components"));
149 if (component.material.parallaxScale < 0.f || component.material.parallaxScale > 0.2f ||
150 component.material.parallaxMinLayers < 1.f ||
151 component.material.parallaxMaxLayers < component.material.parallaxMinLayers ||
152 component.material.cellBombScale <= 0.f)
153 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
154 "material sampling parameters are invalid", "material",
155 {}, "housegen.components"));
156 if (components_.contains(component.id))
157 return eve::Result<void>::failure(eve::Diagnostic::error(
158 eve::DiagnosticCode::Conflict, "duplicate component id: " + component.id, "id", {}, "housegen.components"));
159
160 components_.emplace(component.id, component);
161 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
162}

165 std::string parseError;
166 const eve::json::Document document = eve::json::Document::parse(std::string(json), &parseError);

183 if (!registered.ok()) return registered;

186 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
187}

464graphics::Texture *textureFromFile(graphics::Graphics *gfx, const std::string &modelPath,
465 const std::string &texturePath) {
466 if (texturePath.empty()) return nullptr;
467 std::filesystem::path resolved(texturePath);
468 if (resolved.is_relative()) resolved = std::filesystem::path(modelPath).parent_path() / resolved;
469 try {
470 if (std::filesystem::is_regular_file(resolved)) {
471 std::ifstream input(resolved, std::ios::binary | std::ios::ate);
472 const std::streamsize size = input.tellg();
473 if (size <= 0) return nullptr;
474 input.seekg(0, std::ios::beg);
475 std::vector<uint8_t> bytes(static_cast<size_t>(size));
476 if (!input.read(reinterpret_cast<char *>(bytes.data()), size)) return nullptr;
477 data::ByteData source(bytes.data(), bytes.size());
478 std::unique_ptr<image::ImageData> decoded(image::Image::create()->newImageData(&source));
479 return gfx->newTexture(decoded.get(), graphics::TextureCreateInfo::withMipmaps(true));
480 }
481 return gfx->newTextureFromFile(resolved.lexically_normal().string());
482 } catch (...) {
483 return nullptr;
484 }

487graphics::Texture *assimpTexture(graphics::Graphics *gfx, const aiScene *scene, const aiMaterial *material,
488 const std::string &modelPath, aiTextureType primary, aiTextureType fallback) {
489 if (!scene || !material) return nullptr;
490 aiString texturePath;
491 if (material->GetTexture(primary, 0, &texturePath) != AI_SUCCESS &&
492 material->GetTexture(fallback, 0, &texturePath) != AI_SUCCESS)
493 return nullptr;
494 const std::string path = texturePath.C_Str();
495 if (path.empty()) return nullptr;
496 if (path.front() == '*') {
497 const int index = std::atoi(path.c_str() + 1);
498 if (index < 0 || static_cast<unsigned>(index) >= scene->mNumTextures) return nullptr;
499 try {
500 return textureFromEmbedded(gfx, scene->mTextures[index]);
501 } catch (...) {
502 return nullptr;
503 }
504 }
505 return textureFromFile(gfx, modelPath, path);

514struct ModelHandle {
515 eve::script::Owned<model3d::ModelData> owned;
516 eve::ResourcePin pinned;
517
519 [[nodiscard]] model3d::ModelData *get() const noexcept {
520 // The cache registry stores Resource, so a pinned entry needs the same explicit
521 // downcast the cache's own loaders use.
522 return owned ? owned.get() : static_cast<model3d::ModelData *>(pinned.get());
523 }
524};
525
526ModelHandle loadModel(model3d::Model3D *models, const std::string &path) {
527 if (!std::filesystem::is_regular_file(std::filesystem::path(path))) {
528 // The module hands back the cache-owned resource, so pin it: this layout keeps
529 // using the model after the cache entry may have been unloaded.
530 model3d::ModelData *cached = models->newModelDataFromFile(path);
531 if (cached == nullptr) return {};
532 auto pinned = eve::ResourceManager::getInstance().pin(cached);
533 if (!pinned.ok()) return {};
534 ModelHandle handle;
535 handle.pinned = std::move(pinned).takeValue();
536 return handle;
537 }
538 std::ifstream input(path, std::ios::binary | std::ios::ate);
539 if (!input) throw std::runtime_error("cannot open model: " + path);
540 const std::streamsize size = input.tellg();
541 if (size <= 0) throw std::runtime_error("model is empty: " + path);
542 input.seekg(0, std::ios::beg);
543 std::vector<uint8_t> bytes(static_cast<size_t>(size));
544 if (!input.read(reinterpret_cast<char *>(bytes.data()), size))
545 throw std::runtime_error("cannot read model: " + path);
546 data::ByteData source(bytes.data(), bytes.size());
547 ModelHandle handle;
548 handle.owned.reset(models->newModelData(&source, std::filesystem::path(path).extension().string()));
549 return handle;
550}

676eve::Result<void> HouseLayout::validate(const HouseComponentLibrary &library) const {
677 if (!std::isfinite(moduleSize) || moduleSize <= 0.f || !std::isfinite(floorHeight) || floorHeight <= 0.f)
678 return eve::Result<void>::failure(eve::Diagnostic::error(
679 eve::DiagnosticCode::InvalidArgument, "moduleSize and floorHeight must be finite and positive", {}, {},
680 "housegen.layout"));
681 if (footprintWidth <= 0 || footprintDepth <= 0 ||
682 footprintMask.size() != size_t(footprintWidth) * size_t(footprintDepth))
683 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
684 "house layout has no canonical footprint mask",
685 "footprintMask", {}, "housegen.layout"));
686 for (const auto &instance : instances)
687 if (instance.x >= footprintWidth || instance.y >= footprintDepth ||
688 !footprintMask[size_t(instance.y * footprintWidth + instance.x)])
689 return eve::Result<void>::failure(eve::Diagnostic::error(
690 eve::DiagnosticCode::InvalidArgument, "house instance origin is outside the canonical footprint",
691 instance.componentId, {}, "housegen.layout"));
692 for (const HouseRoom &room : rooms) {
693 if (room.x < 0 || room.y < 0 || room.width <= 0 || room.depth <= 0 || room.x + room.width > footprintWidth ||
694 room.y + room.depth > footprintDepth)
695 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
696 "room lies outside the canonical footprint",
697 "rooms", {}, "housegen.layout"));
698 for (int y = room.y; y < room.y + room.depth; ++y)
699 for (int x = room.x; x < room.x + room.width; ++x)
700 if (!footprintMask[size_t(y * footprintWidth + x)])
701 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
702 "room contains an inactive footprint cell",
703 "rooms", {}, "housegen.layout"));
704 }
705 std::unordered_set<std::string> occupied;
706 std::unordered_set<std::string> floorCells, roofCells;
707 std::vector<std::tuple<int, int, int>> floors;
708 auto cellKey = [](int x, int y, int z) {
709 return std::to_string(x) + ":" + std::to_string(y) + ":" + std::to_string(z);
710 };
711 bool entrance = false, roof = false;
712 for (const auto &i : instances) {
713 const auto component = library.find(i.componentId);
714 if (!component)
715 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::NotFound,
716 "unknown component: " + i.componentId,
717 "componentId", {}, "housegen.layout"));
718 const HouseComponent &c = component->get();
719 const int normalizedRotation = (i.rotationDeg % 360 + 360) % 360;
720 if (normalizedRotation % 90 != 0)
721 return eve::Result<void>::failure(eve::Diagnostic::error(
722 eve::DiagnosticCode::InvalidArgument, "non-cardinal rotation", "rotationDeg", {}, "housegen.layout"));
723 const bool rotationAllowed = std::any_of(c.rotations.begin(), c.rotations.end(), [&](int allowed) {
724 return (allowed % 360 + 360) % 360 == normalizedRotation;
725 });
726 if (!rotationAllowed)
727 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
728 "component does not allow the requested rotation",
729 "rotationDeg", {}, "housegen.layout"));
730 const bool quarter = (normalizedRotation / 90) % 2 != 0;
731 const int w = quarter ? c.depth : c.width, d = quarter ? c.width : c.depth;
732 for (int y = 0; y < d; ++y)
733 for (int x = 0; x < w; ++x) {
734 // Boundary cells legitimately carry two perpendicular wall modules at corners.
735 const bool orientedBoundary = c.category == "wall" || c.category == "door" ||
736 c.category == "interior_wall" || c.category == "interior_door";
737 const std::string orientation =
738 orientedBoundary ? ":" + std::to_string((i.rotationDeg % 360 + 360) % 360) : "";
739 const std::string key = std::to_string(i.x + x) + ":" + std::to_string(i.y + y) + ":" +
740 std::to_string(i.z) + ":" + c.category + orientation;
741 if (!occupied.insert(key).second)
742 return eve::Result<void>::failure(
743 eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
744 "overlapping " + c.category + " components", {}, {}, "housegen.layout"));
745 if (c.category == "floor") {
746 floorCells.insert(cellKey(i.x + x, i.y + y, i.z));
747 floors.emplace_back(i.x + x, i.y + y, i.z);
748 } else if (c.category == "roof") {
749 roofCells.insert(cellKey(i.x + x, i.y + y, i.z));
750 }
751 }
752 entrance = entrance || (c.category == "door" && i.z == 0);
753 roof = roof || c.category == "roof";
754 }
755 if (!entrance)
756 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
757 "house has no entrance", {}, {}, "housegen.layout"));
758 if (!roof)
759 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
760 "house has no roof", {}, {}, "housegen.layout"));
761 for (const auto &[x, y, z] : floors) {
762 if (z > 0 && !floorCells.contains(cellKey(x, y, z - 1))) {
763 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
764 "upper floor has no structural support", {}, {},
765 "housegen.layout"));
766 }
767 if (!floorCells.contains(cellKey(x, y, z + 1)) && !roofCells.contains(cellKey(x, y, z + 1))) {
768 return eve::Result<void>::failure(eve::Diagnostic::error(
769 eve::DiagnosticCode::InvalidArgument, "floor cell has no roof coverage", {}, {}, "housegen.layout"));
770 }
771 }
772 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
773}

775eve::Result<void> HouseLayout::writeFootprintGrid(procgen::Grid2D &out) const {
776 if (!std::isfinite(moduleSize) || moduleSize <= 0.f || !std::isfinite(floorHeight) || floorHeight <= 0.f)
777 return eve::Result<void>::failure(eve::Diagnostic::error(
778 eve::DiagnosticCode::InvalidArgument, "moduleSize and floorHeight must be finite and positive", {}, {},
779 "housegen.layout"));
780 for (const auto &instance : instances) {
781 if (instance.x < 0 || instance.y < 0 || instance.z < 0)
782 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
783 "house instance has a negative grid coordinate",
784 instance.componentId, {}, "housegen.layout"));
785 }
786 if (footprintWidth <= 0 || footprintDepth <= 0 ||
787 footprintMask.size() != size_t(footprintWidth) * size_t(footprintDepth))
788 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
789 "house layout has no canonical footprint mask",
790 "footprintMask", {}, "housegen.layout"));
791
792 procgen::Grid2D candidate;
793 candidate.resize(footprintWidth, footprintDepth);
794 candidate.fill(int(procgen::Semantic::Empty));
795 for (int y = 0; y < footprintDepth; ++y)
796 for (int x = 0; x < footprintWidth; ++x)
797 if (footprintMask[size_t(y * footprintWidth + x)] != 0)
798 candidate.setCell(x, y, int(procgen::Semantic::Floor));
799 for (const auto &instance : instances) {
800 if (instance.x >= footprintWidth || instance.y >= footprintDepth) continue;
801 candidate.setDetail(instance.x, instance.y,
802 std::min(255, std::max(candidate.getDetail(instance.x, instance.y), instance.z + 1)));
803 }
804 candidate.setMeta("domain", "house.footprint");
805 candidate.setMeta("seed", std::to_string(seed));
806 candidate.setMeta("footprint", footprintStyle);
807 candidate.setMeta("roof", roofStyle);
808 candidate.setMeta("entrance", entranceSide);
809 out = std::move(candidate);
810 return eve::Result<void>::success();
811}

813eve::Result<void> HouseLayout::writeComponentPoints(procgen::PointSet &out) const {
814 if (!std::isfinite(moduleSize) || moduleSize <= 0.f || !std::isfinite(floorHeight) || floorHeight <= 0.f)
815 return eve::Result<void>::failure(eve::Diagnostic::error(
816 eve::DiagnosticCode::InvalidArgument, "moduleSize and floorHeight must be finite and positive", {}, {},
817 "housegen.layout"));
818 if (footprintWidth <= 0 || footprintDepth <= 0 ||
819 footprintMask.size() != size_t(footprintWidth) * size_t(footprintDepth))
820 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
821 "house layout has no canonical footprint mask",
822 "footprintMask", {}, "housegen.layout"));
823 for (const auto &instance : instances) {
824 const int rotation = (instance.rotationDeg % 360 + 360) % 360;
825 if (instance.x < 0 || instance.y < 0 || instance.z < 0 || instance.x >= footprintWidth ||
826 instance.y >= footprintDepth || !footprintMask[size_t(instance.y * footprintWidth + instance.x)] ||
827 rotation % 90 != 0)
828 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
829 "invalid house component point placement",
830 instance.componentId, {}, "housegen.layout"));
831 }
832 procgen::PointSet candidate;
833 candidate.reserve(instances.size());
834 std::uint64_t ordinal = 0;
835 const std::uint64_t pointNamespace = (std::uint64_t(seed) << 32u) | 0x484f5553u;
836 for (const auto &instance : instances) {
837 const int index = candidate.add(float(instance.x) * moduleSize, float(instance.z) * floorHeight,
838 float(instance.y) * moduleSize);
839 candidate.setYaw(index, float(instance.rotationDeg));
840 candidate.setPointSeed(index, seed);
841 candidate.setBounds(index, 0.f, 0.f, 0.f, moduleSize, floorHeight, moduleSize);
842 auto id = candidate.trySetPointId(index, procgen::derivePointId(pointNamespace, ++ordinal));
843 if (!id.ok()) return eve::Result<void>::failure(id.status());
844 auto component = candidate.trySetStringAttribute(index, "component_id", instance.componentId);
845 if (!component.ok()) return eve::Result<void>::failure(component.status());
846 auto cellX = candidate.trySetIntAttribute(index, "cell_x", instance.x);
847 if (!cellX.ok()) return eve::Result<void>::failure(cellX.status());
848 auto cellY = candidate.trySetIntAttribute(index, "cell_y", instance.y);
849 if (!cellY.ok()) return eve::Result<void>::failure(cellY.status());
850 auto floor = candidate.trySetIntAttribute(index, "floor", instance.z);
851 if (!floor.ok()) return eve::Result<void>::failure(floor.status());
852 }
853 out = std::move(candidate);
854 return eve::Result<void>::success();
855}

857eve::Result<std::vector<ecs::EntityHandle>> HouseLayout::instantiate(graphics::Graphics &gfx, model3d::Model3D &models,
858 const HouseComponentLibrary &library) const {
859 const auto valid = validate(library);
860 if (!valid.ok()) return eve::Result<std::vector<ecs::EntityHandle>>::failure(valid.status());
861 std::vector<ecs::EntityHandle> entities;
862 const auto destroyCreated = [&entities]() noexcept {
863 for (const auto &handle : entities) {
864 if (auto *entity = ecs::try_get(handle)) ecs::DestroyEntity(entity);
865 }
866 entities.clear();
867 };
868 std::unordered_map<std::string, ModelHandle> data;
869 struct CachedPart {
870 graphics::Mesh *mesh = nullptr;
871 graphics::Texture *texture = nullptr;
872 graphics::Texture *normalTexture = nullptr;
873 graphics::Texture *heightTexture = nullptr;
874 float r = 1.f, g = 1.f, b = 1.f, a = 1.f;
875 float metallic = 0.f, roughness = 0.72f;
876 float parallaxScale = 0.f, parallaxMinLayers = 8.f, parallaxMaxLayers = 32.f;
877 float cellBombScale = 4.f, cellBombStrength = 0.f, cellBombRotation = 1.f;
878 };
879 std::unordered_map<std::string, std::vector<CachedPart>> parts;
880 try {
881 for (const auto &i : instances) {
882 const auto component = library.find(i.componentId);
883 if (!component) {
884 destroyCreated();
885 return eve::Result<std::vector<ecs::EntityHandle>>::failure(
886 eve::Diagnostic::error(eve::DiagnosticCode::NotFound, "unknown component: " + i.componentId,
887 "componentId", {}, "housegen.layout"));
888 }
889 const HouseComponent &c = component->get();
890 // Loads once per path: the handle either owns the decode or pins the cache entry.
891 auto modelIt = data.find(c.modelPath);
892 if (modelIt == data.end()) modelIt = data.emplace(c.modelPath, loadModel(&models, c.modelPath)).first;
893 model3d::ModelData *model = modelIt->second.get();
894 // Material overrides belong to a component, so two components may safely reuse the
895 // same GLB with different architectural finishes.
896 auto &cached = parts[c.id];
897 if (cached.empty()) {
898 const aiScene *scene = model->getScene();
899 if (!scene || !scene->mRootNode) throw std::runtime_error("model has no scene root: " + c.modelPath);
900 std::function<void(const aiNode *, const aiMatrix4x4 &)> walk = [&](const aiNode *node,
901 const aiMatrix4x4 &parent) {
902 const aiMatrix4x4 world = parent * node->mTransformation;
903 for (unsigned ni = 0; ni < node->mNumMeshes; ++ni) {
904 const unsigned mi = node->mMeshes[ni];
905 if (mi >= scene->mNumMeshes || !scene->mMeshes[mi]) continue;
906 const aiMesh *source = scene->mMeshes[mi];
907 CachedPart part;
908 part.mesh = gfx.newMeshFromAssimp(*source, world);
909 if (scene->mMaterials && source->mMaterialIndex < scene->mNumMaterials) {
910 const aiMaterial *material = scene->mMaterials[source->mMaterialIndex];
911 aiColor4D color(1.f, 1.f, 1.f, 1.f);
912 material->Get(AI_MATKEY_COLOR_DIFFUSE, color);
913 material->Get(AI_MATKEY_BASE_COLOR, color);
914 part.r = color.r;
915 part.g = color.g;
916 part.b = color.b;
917 part.a = color.a;
918 material->Get(AI_MATKEY_METALLIC_FACTOR, part.metallic);
919 material->Get(AI_MATKEY_ROUGHNESS_FACTOR, part.roughness);
920 part.texture = assimpTexture(&gfx, scene, material, c.modelPath, aiTextureType_BASE_COLOR,
921 aiTextureType_DIFFUSE);
922 part.normalTexture = assimpTexture(&gfx, scene, material, c.modelPath,
923 aiTextureType_NORMALS, aiTextureType_NORMAL_CAMERA);
924 part.heightTexture = assimpTexture(&gfx, scene, material, c.modelPath, aiTextureType_HEIGHT,
925 aiTextureType_DISPLACEMENT);
926 }
927 if (c.material.hasBaseColor) {
928 part.r = c.material.baseColorR;
929 part.g = c.material.baseColorG;
930 part.b = c.material.baseColorB;
931 part.a = c.material.baseColorA;
932 }
933 if (!c.material.baseColorTexture.empty())
934 part.texture = textureFromFile(&gfx, c.modelPath, c.material.baseColorTexture);
935 if (!c.material.normalTexture.empty())
936 part.normalTexture = textureFromFile(&gfx, c.modelPath, c.material.normalTexture);
937 if (!c.material.heightTexture.empty())
938 part.heightTexture = textureFromFile(&gfx, c.modelPath, c.material.heightTexture);
939 if (c.material.hasMetallic) part.metallic = c.material.metallic;
940 if (c.material.hasRoughness) part.roughness = c.material.roughness;
941 part.parallaxScale = c.material.parallaxScale;
942 part.parallaxMinLayers = c.material.parallaxMinLayers;
943 part.parallaxMaxLayers = c.material.parallaxMaxLayers;
944 part.cellBombScale = c.material.cellBombScale;
945 part.cellBombStrength = c.material.cellBombStrength;
946 part.cellBombRotation = c.material.cellBombRotation;
947 cached.push_back(part);
948 }
949 for (unsigned child = 0; child < node->mNumChildren; ++child) walk(node->mChildren[child], world);
950 };
951 walk(scene->mRootNode, aiMatrix4x4());
952 }
953 for (const auto &part : cached) {
954 auto *e = graphics::Renderable3D::create();
