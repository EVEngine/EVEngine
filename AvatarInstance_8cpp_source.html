36std::string trimCopy(const std::string &s) {
37 size_t b = 0;
38 while (b < s.size() && std::isspace(static_cast<unsigned char>(s[b]))) ++b;
39 size_t e = s.size();
40 while (e > b && std::isspace(static_cast<unsigned char>(s[e - 1]))) --e;
41 return s.substr(b, e - b);
42}

44bool parseBoolish(const std::string &v, bool &out) {
45 std::string t;
46 t.reserve(v.size());
47 for (char c : v) t.push_back(static_cast<char>(std::tolower(static_cast<unsigned char>(c))));

345EquipmentVisualChange AvatarInstance::syncEquipmentAppearance() {
346 if (!equipment_) return EquipmentVisualChange::Rejected;
347 std::string signature;
348 std::unordered_map<std::string, std::string> equipped;
349 for (int index = 0; index < equipment_->getSlotCount(); ++index) {
350 const std::string slot = equipment_->getSlotName(index);
351 const std::string item = equipment_->getSlotItemId(slot);
352 signature += slot + "=" + item + "#" + std::to_string(equipment_->getSlotInstanceId(slot)) + ";";
353 if (!item.empty()) equipped.emplace(slot, item);
354 }
355 if (signature == equipmentSignature_) return EquipmentVisualChange::Unchanged;
356
357 hideEquipmentVisuals();
358 projectedEquipment_.clear();
359 equipmentRenderStack_.clear();
360 std::vector<std::string> occludedLayers;
361 for (const auto& visual : equipmentVisuals_) {
362 const auto active = equipped.find(visual.equipmentSlot);
363 if (active == equipped.end() || active->second != visual.itemId) continue;
364 const auto layer = equipmentLayers_.find(visual.wearLayer);
365 if (layer == equipmentLayers_.end()) continue;
366 occludedLayers.insert(occludedLayers.end(), layer->second.occludes.begin(),
367 layer->second.occludes.end());
368 }
369 for (auto& visual : equipmentVisuals_) {
370 const auto active = equipped.find(visual.equipmentSlot);
371 if (active == equipped.end() || active->second != visual.itemId) continue;
372 if (std::find(occludedLayers.begin(), occludedLayers.end(), visual.wearLayer) !=
373 occludedLayers.end())
374 continue;
375 const int renderOrder = equipmentLayerOrder(visual.wearLayer) * 1000 +
376 visual.withinLayerOrder;
377 if (kind_ == "image" && !visual.layerName.empty()) {
378 if (!hasLayer(visual.layerName)) addLayer(visual.layerName, visual.texture, visual.zIndex);
379 setLayerTexture(visual.layerName, visual.texture);
380 setLayerZ(visual.layerName, renderOrder + visual.zIndex);
381 setLayerVisible(visual.layerName, true);
382 }
383 if (visual.renderable) {
384 visual.renderable->setVisible(visible_);
385 if (visual.boneName.empty()) {
386 visual.renderable->setPosition(x3_ + visual.ox, y3_ + visual.oy, z3_ + visual.oz);
387 visual.renderable->setRotation(yaw_, pitch_, roll_);
388 visual.renderable->setScale(sx3_, sy3_, sz3_);
389 } else {
390 attachToBone("equipment:" + visual.equipmentSlot, visual.boneName, visual.renderable,
391 visual.ox, visual.oy, visual.oz);
392 }
393 }
394 if (visual.partIndex >= 0 && visual.skinnedMesh && renderable3d_) {
395 renderable3d_->setPart(visual.partIndex, visual.layerName, visual.skinnedMesh,
396 visual.skinnedMaterial);
397 renderable3d_->setPartSortPriority(visual.partIndex, renderOrder);
398 }
399 projectedEquipment_[visual.equipmentSlot] = visual.itemId;
400 equipmentRenderStack_.push_back({visual.itemId, visual.wearLayer, renderOrder});
401 }
402 std::stable_sort(equipmentRenderStack_.begin(), equipmentRenderStack_.end(),
403 [](const EquipmentRenderEntry& a, const EquipmentRenderEntry& b) {
404 if (a.order != b.order) return a.order < b.order;
405 return a.itemId < b.itemId;
406 });
407 equipmentSignature_ = std::move(signature);
408 return EquipmentVisualChange::Applied;
409}

781eve::Result<eve::AttachmentPoint> AvatarInstance::sampleAttachmentPoint(
782 std::string_view name, eve::AttachmentPoint localOffset) const {
783 if (!animSkeleton_ || attachmentPoseMatrices_.empty())
784 return eve::Result<eve::AttachmentPoint>::failure(eve::Diagnostic::error(
785 eve::DiagnosticCode::PreconditionViolation, "Avatar has no evaluated skeletal pose",
786 "avatar.attachment.pose"));
787 std::string boneName(name);
788 const auto semantic = humanoidBones_.find(boneName);
789 if (semantic != humanoidBones_.end()) boneName = semantic->second;
790 const int bone = animSkeleton_->findBone(boneName);
791 if (bone < 0 || static_cast<std::size_t>(bone) >= attachmentPoseMatrices_.size())
792 return eve::Result<eve::AttachmentPoint>::failure(eve::Diagnostic::error(
793 eve::DiagnosticCode::NotFound, "Avatar attachment bone was not found", boneName));
794 const auto& matrix = attachmentPoseMatrices_[static_cast<std::size_t>(bone)];
795 const float bx = matrix[0] * localOffset.x + matrix[4] * localOffset.y + matrix[8] * localOffset.z + matrix[12];
796 const float by = matrix[1] * localOffset.x + matrix[5] * localOffset.y + matrix[9] * localOffset.z + matrix[13];
797 const float bz = matrix[2] * localOffset.x + matrix[6] * localOffset.y + matrix[10] * localOffset.z + matrix[14];
798 const float sinYaw = std::sin(yaw_);
799 const float cosYaw = std::cos(yaw_);
800 return eve::Result<eve::AttachmentPoint>::success(
801 {x3_ + bx * sx3_ * cosYaw + bz * sz3_ * sinYaw, y3_ + by * sy3_,
802 z3_ - bx * sx3_ * sinYaw + bz * sz3_ * cosYaw});
803}

805animation::AnimPose* AvatarInstance::updateSkeletalAnimation(float dt) {
806 animation::AnimPose* pose = nullptr;
807 rootMotionLoopCount_ = 0;
808 if (animLayerMixer_) {
809 const float oldTime = animPlayer_ ? animPlayer_->getTime() : 0.f;
810 animation::AnimClip* oldClip = animPlayer_ ? animPlayer_->getClip() : nullptr;
811 animLayerMixer_->update(dt);
812 pose = animLayerMixer_->getPose();
813 if (animPlayer_ && oldClip && oldClip == animPlayer_->getClip() && animPlayer_->getLoop() &&
814 oldClip->getDuration() > 0.f) {
815 const float duration = oldClip->getDuration();
816 rootMotionLoopCount_ =
817 static_cast<int>(std::floor(animPlayer_->getTime() / duration) - std::floor(oldTime / duration));
818 }
819 } else if (animStateMachine_) {
820 animStateMachine_->update(dt);
821 pose = animStateMachine_->getPose();
822 } else if (animPlayer_) {
823 const float oldTime = animPlayer_->getTime();
824 animation::AnimClip* oldClip = animPlayer_->getClip();
825 animPlayer_->update(dt);
826 pose = animPlayer_->getPose();
827 if (oldClip && oldClip == animPlayer_->getClip() && animPlayer_->getLoop() && oldClip->getDuration() > 0.f) {
828 const float duration = oldClip->getDuration();
829 rootMotionLoopCount_ =
830 static_cast<int>(std::floor(animPlayer_->getTime() / duration) - std::floor(oldTime / duration));
831 }
832 }
833 if (!pose || !animSkeleton_) {
834 attachmentPoseMatrices_.clear();
835 return nullptr;
836 }
837 pose->computeWorld(animSkeleton_);
838 applyLookAt(pose);
839 if (footIKSolver_) footIKSolver_->apply(pose, dt);
840 if (animConstraintStack_) animConstraintStack_->apply(pose);
841 if (dynamicBoneSolver_) dynamicBoneSolver_->update(pose, dt);
842 updateRootMotion(pose);
843 updateSkin(pose);
844 updateAttachments(pose);
845 attachmentPoseMatrices_.resize(static_cast<std::size_t>(pose->getBoneCount()));
846 for (int bone = 0; bone < pose->getBoneCount(); ++bone)
847 pose->getWorldMatrix(bone, attachmentPoseMatrices_[static_cast<std::size_t>(bone)].data());
848 return pose;
849}

882void AvatarInstance::updateRootMotion(animation::AnimPose* pose) {
883 rootMotionDeltaX_ = 0.f;
884 rootMotionDeltaZ_ = 0.f;
885 if (!pose || pose->getBoneCount() <= 0) return;
886 const float rootX = pose->getWorldPositionX(0);
887 const float rootZ = pose->getWorldPositionZ(0);
888 if (hasPreviousRoot_) {
889 rootMotionDeltaX_ = rootX - previousRootX_;
890 rootMotionDeltaZ_ = rootZ - previousRootZ_;
891 if (rootMotionLoopCount_ > 0 && animPlayer_ && animPlayer_->getClip()) {
892 animation::AnimClip* clip = animPlayer_->getClip();
893 const int keyCount = clip->getPositionKeyCount(0);
894 if (keyCount > 1) {
895 const float cycleX = clip->getPositionKeyX(0, keyCount - 1) - clip->getPositionKeyX(0, 0);
896 const float cycleZ = clip->getPositionKeyZ(0, keyCount - 1) - clip->getPositionKeyZ(0, 0);
897 rootMotionDeltaX_ += cycleX * static_cast<float>(rootMotionLoopCount_);
898 rootMotionDeltaZ_ += cycleZ * static_cast<float>(rootMotionLoopCount_);
899 }
900 }
901 if (applyRootMotion_) {
902 x3_ += rootMotionDeltaX_;
903 z3_ += rootMotionDeltaZ_;
904 }
905 }
906 previousRootX_ = rootX;
907 previousRootZ_ = rootZ;
908 hasPreviousRoot_ = true;
909}

949void AvatarInstance::updateAttachments(animation::AnimPose* pose) {
950 if (!pose || !animSkeleton_) return;
951 const float sinYaw = std::sin(yaw_);
952 const float cosYaw = std::cos(yaw_);
953 for (auto& attachment : attachments_) {
954 if (!attachment.renderable) continue;
955 const int bone = animSkeleton_->findBone(attachment.boneName);
956 if (bone < 0 || bone >= pose->getBoneCount()) continue;
957 const animation::TransformTRS& world = pose->world(bone);
958 const animation::Mat4 boneMatrix = animation::Mat4::fromTRS(world);
959 float bx = 0.f, by = 0.f, bz = 0.f;
960 boneMatrix.transformPoint(attachment.ox, attachment.oy, attachment.oz, bx, by, bz);
961 const float scaledX = bx * sx3_;
962 const float scaledZ = bz * sz3_;
963 attachment.renderable->setPosition(x3_ + scaledX * cosYaw + scaledZ * sinYaw, y3_ + by * sy3_,
964 z3_ - scaledX * sinYaw + scaledZ * cosYaw);
965
966 const float sinPitch = 2.f * (world.qw * world.qx - world.qy * world.qz);
967 const float bonePitch = std::asin(std::clamp(sinPitch, -1.f, 1.f));
968 const float boneYaw = std::atan2(2.f * (world.qw * world.qy + world.qx * world.qz),
969 1.f - 2.f * (world.qx * world.qx + world.qy * world.qy));
970 const float boneRoll = std::atan2(2.f * (world.qw * world.qz + world.qx * world.qy),
971 1.f - 2.f * (world.qx * world.qx + world.qz * world.qz));
972 attachment.renderable->setRotation(yaw_ + boneYaw, pitch_ + bonePitch, roll_ + boneRoll);
973 }
974}

976void AvatarInstance::applyLookAt(animation::AnimPose* pose) {
977 if (!pose || !animSkeleton_) return;
978 const auto mapped = humanoidBones_.find("head");
979 if (mapped == humanoidBones_.end()) return;
980 const int head = animSkeleton_->findBone(mapped->second);
981 if (head < 0 || head >= pose->getBoneCount()) return;
982
983 animation::TransformTRS& local = pose->local(head);
984 if (lookAtApplied_) {
985 const float similarity = std::fabs(local.qx * lookAtPostQx_ + local.qy * lookAtPostQy_ +
986 local.qz * lookAtPostQz_ + local.qw * lookAtPostQw_);
987 if (similarity > 0.9999f) {
988 local.qx = lookAtBaseQx_;
989 local.qy = lookAtBaseQy_;
990 local.qz = lookAtBaseQz_;
991 local.qw = lookAtBaseQw_;
992 }
993 lookAtApplied_ = false;
994 pose->computeWorld(animSkeleton_);
995 }
996 if (!lookAtEnabled_ || lookAtWeight_ <= 0.f) return;
997
998 const animation::TransformTRS& headWorld = pose->world(head);
999 const float dx = lookAtX_ - (x3_ + headWorld.px * sx3_);
1000 const float dy = lookAtY_ - (y3_ + headWorld.py * sy3_);
1001 const float dz = lookAtZ_ - (z3_ + headWorld.pz * sz3_);
1002 const float cosYaw = std::cos(yaw_);
1003 const float sinYaw = std::sin(yaw_);
1004 const float localX = dx * cosYaw - dz * sinYaw;
1005 const float localZ = dx * sinYaw + dz * cosYaw;
1006 const float horizontal = std::sqrt(localX * localX + localZ * localZ);
1007 if (horizontal < 1e-6f && std::fabs(dy) < 1e-6f) return;
1008
1009 constexpr float kMaxYaw = 1.04719755f;
1010 constexpr float kMaxPitch = 0.61086524f;
1011 const float targetYaw = std::clamp(std::atan2(localX, localZ), -kMaxYaw, kMaxYaw) * lookAtWeight_;
1012 const float targetPitch = std::clamp(-std::atan2(dy, horizontal), -kMaxPitch, kMaxPitch) * lookAtWeight_;
1013 const float sinHalfYaw = std::sin(targetYaw * 0.5f);
1014 const float cosHalfYaw = std::cos(targetYaw * 0.5f);
1015 const float sinHalfPitch = std::sin(targetPitch * 0.5f);
1016 const float cosHalfPitch = std::cos(targetPitch * 0.5f);
1017 const float deltaX = cosHalfYaw * sinHalfPitch;
1018 const float deltaY = sinHalfYaw * cosHalfPitch;
1019 const float deltaZ = -sinHalfYaw * sinHalfPitch;
1020 const float deltaW = cosHalfYaw * cosHalfPitch;
1021
1022 lookAtBaseQx_ = local.qx;
1023 lookAtBaseQy_ = local.qy;
1024 lookAtBaseQz_ = local.qz;
1025 lookAtBaseQw_ = local.qw;
1026 const float qx = local.qw * deltaX + local.qx * deltaW + local.qy * deltaZ - local.qz * deltaY;
1027 const float qy = local.qw * deltaY - local.qx * deltaZ + local.qy * deltaW + local.qz * deltaX;
1028 const float qz = local.qw * deltaZ + local.qx * deltaY - local.qy * deltaX + local.qz * deltaW;
1029 const float qw = local.qw * deltaW - local.qx * deltaX - local.qy * deltaY - local.qz * deltaZ;
1030 local.qx = qx;
1031 local.qy = qy;
1032 local.qz = qz;
1033 local.qw = qw;
1034 local.normalizeRotation();
1035 lookAtPostQx_ = local.qx;
1036 lookAtPostQy_ = local.qy;
1037 lookAtPostQz_ = local.qz;
1038 lookAtPostQw_ = local.qw;
1039 lookAtApplied_ = true;
1040 pose->computeWorld(animSkeleton_);
1041}

1269bool AvatarInstance::transitionExpression(const std::string& name, float duration) {
1270 const auto definition = expressionDefs_.find(name);
1271 if (definition == expressionDefs_.end()) return false;
1272 if (duration <= 0.f) return applyExpression(name);
1273
1274 std::unordered_map<std::string, float> targets;
1275 if (kind_ == "vroid" && boundMesh_ && boundMesh_->hasMorphData()) {
1276 for (int i = 0; i < boundMesh_->getMorphCount(); ++i) targets[boundMesh_->getMorphName(i)] = 0.f;
1277 }
1278 std::stringstream stream(definition->second);
1279 std::string part;
1280 while (std::getline(stream, part, ';')) {
1281 part = trimCopy(part);
1282 const auto eq = part.find('=');
1283 if (eq == std::string::npos) continue;
1284 const std::string key = trimCopy(part.substr(0, eq));
1285 const std::string value = trimCopy(part.substr(eq + 1));
1286 if (key.empty()) continue;
1287 bool flag = false;
1288 if (parseBoolish(value, flag)) {
1289 targets[key] = flag ? 1.f : 0.f;
1290 continue;
1291 }
1292 char* end = nullptr;
1293 const float numeric = std::strtof(value.c_str(), &end);
1294 if (!end || end == value.c_str() || *end != '\0') return applyExpression(name);
1295 targets[key] = numeric;
1296 }
1297 if (targets.empty()) return applyExpression(name);
1298
1299 expressionBlendFrom_.clear();
1300 expressionBlendTo_ = std::move(targets);
1301 for (const auto& [key, target] : expressionBlendTo_) {
1302 (void)target;
1303 if (kind_ == "image") {
1304 const Layer* layer = findLayer(key);
1305 expressionBlendFrom_[key] = layer ? layer->entity->sprite()->a : getParameter(key);
1306 } else {
1307 expressionBlendFrom_[key] = getParameter(key);
1308 }
1309 }
1310 expression_ = name;
1311 expressionBlendElapsed_ = 0.f;
1312 expressionBlendDuration_ = duration;
1313 return true;
1314}

1316void AvatarInstance::updateExpressionTransition(float dt) {
1317 if (expressionBlendDuration_ <= 0.f || expressionBlendTo_.empty()) return;
1318 expressionBlendElapsed_ += std::max(0.f, dt);
1319 const float t = std::clamp(expressionBlendElapsed_ / expressionBlendDuration_, 0.f, 1.f);
1320 for (const auto& [key, target] : expressionBlendTo_) {
1321 const auto start = expressionBlendFrom_.find(key);
1322 const float from = start == expressionBlendFrom_.end() ? getParameter(key) : start->second;
1323 setParameter(key, from + (target - from) * t);
1324 }
1325 if (t >= 1.f) {
1326 expressionBlendFrom_.clear();
1327 expressionBlendTo_.clear();
1328 expressionBlendDuration_ = 0.f;
1329 }
