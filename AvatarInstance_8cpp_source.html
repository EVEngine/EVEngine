24std::string trimCopy(const std::string &s) {
25 size_t b = 0;
26 while (b < s.size() && std::isspace(static_cast<unsigned char>(s[b]))) ++b;
27 size_t e = s.size();
28 while (e > b && std::isspace(static_cast<unsigned char>(s[e - 1]))) --e;
29 return s.substr(b, e - b);
30}

32bool parseBoolish(const std::string &v, bool &out) {
33 std::string t;
34 t.reserve(v.size());
35 for (char c : v) t.push_back(static_cast<char>(std::tolower(static_cast<unsigned char>(c))));
36 if (t == "1" || t == "true" || t == "on" || t == "yes") {
37 out = true;
38 return true;
39 }
40 if (t == "0" || t == "false" || t == "off" || t == "no") {
41 out = false;
42 return true;
43 }
44 return false;
45}

56void AvatarInstance::release() {
57 if (released_) return;
58 released_ = true;
59 // Notify owners (e.g. Dialogue) before the object is destroyed so they can
60 // drop dangling raw pointers. Runs exactly once: the destructor re-enters
61 // release() but the guard above short-circuits.
62 for (auto &[hookId, hook] : destroyHooks_) {
63 try {
64 hook(this);
65 } catch (...) {
66 }
67 }
68 destroyHooks_.clear();
69 tween_ = nullptr;
70 if (auto *mod = ModuleManager::getInstance<Avatar>("Avatar"))
71 mod->unregisterInstance(this);
72 destroyLayers();
73 destroyVroid();
74 delete live2d_;
75 live2d_ = nullptr;
76}

224 name == "y3" || name == "z3" || name == "yaw")

324bool AvatarInstance::applyExpressionSpec(const std::string &spec) {
325 if (spec.empty()) return true;
326 std::stringstream ss(spec);
327 std::string part;
328 while (std::getline(ss, part, ';')) {
329 part = trimCopy(part);
330 if (part.empty()) continue;
331 const auto eq = part.find('=');
332 if (eq == std::string::npos) continue;
333 const std::string key = trimCopy(part.substr(0, eq));
334 const std::string val = trimCopy(part.substr(eq + 1));
335 if (key.empty()) continue;
336
337 if (kind_ == "vroid") {
338 bool flag = false;
339 if (parseBoolish(val, flag)) {
340 setParameter(key, flag ? 1.f : 0.f);
341 } else {
342 char *end = nullptr;
343 const float f = std::strtof(val.c_str(), &end);
344 if (end && end != val.c_str())
345 setParameter(key, f);
346 else
347 setParameter(key, 1.f);
348 }
349 continue;
350 }
351
352 Layer *L = findLayer(key);
353 if (!L) {
354 // Unknown layer name still recorded as parameter for tooling.
355 setParameter(key, 1.f);
356 continue;
357 }
358 bool flag = false;
359 if (parseBoolish(val, flag)) {
360 L->visible = flag;
361 } else {
362 // Non-bool value: show layer and stash as parameter (texture swap by name).
363 L->visible = true;
364 setParameter(key, 1.f);
365 setParameter(key + ".variant", float(std::hash<std::string>{}(val) & 0xffff));
366 }
367 }
368 return true;
369}

380void AvatarInstance::syncImageLayers() {
381 for (Layer &L : layers_) {
382 if (!L.entity) L.entity = graphics::Renderable2D::create();
383 auto tf = L.entity->transform();
384 auto sp = L.entity->sprite();
385 tf->x = x_ + L.ox * sx_;
386 tf->y = y_ + L.oy * sy_;
387 tf->rot = 0.f;
388 tf->sx = sx_;
389 tf->sy = sy_;
390 float w = L.w;
391 float h = L.h;
392 if ((w <= 0.f || h <= 0.f) && L.texture) {
393 w = float(L.texture->getWidth());
394 h = float(L.texture->getHeight());
395 }
396 if (w <= 0.f) w = 32.f;
397 if (h <= 0.f) h = 32.f;
398 sp->width = w;
399 sp->height = h;
400 sp->r = L.r;
401 sp->g = L.g;
402 sp->b = L.b;
403 sp->a = L.a;
404 sp->layer = layer_ + L.zIndex;
405 sp->visible = visible_ && L.visible;
406 sp->texture = L.texture;
407 sp->quad = nullptr;
408 sp->shader = nullptr;
409 sp->canvas = nullptr;
410 sp->camera = nullptr;
411 }
