25FlexResult flexArrange(bool row, float gap, float availMain, float availCross,
26 FlexAlign containerAlign, FlexJustify justify,
27 const std::vector<FlexItemSpec> &items) {
28 FlexResult res;
29 res.items.resize(items.size());
30 if (availMain < 0.f) availMain = 0.f;
31 if (availCross < 0.f) availCross = 0.f;
32
33 const int n = int(items.size());
34 std::vector<float> mainSize(size_t(n), 0.f);
35 std::vector<float> crossSize(size_t(n), 0.f);
36 std::vector<float> mainPos(size_t(n), 0.f);
37 std::vector<float> crossPos(size_t(n), 0.f);
38
39 float flowTotal = 0.f;
40 int flowCount = 0;
41 float growSum = 0.f;
42
43 for (int i = 0; i < n; ++i) {
44 const FlexItemSpec &s = items[size_t(i)];
45 float m = s.explicitMain > 0.f
46 ? s.explicitMain
47 : (s.percentMain > 0.f ? s.percentMain * availMain : s.basisMain);
48 m = clampV(m, s.minMain, s.maxMain);
49 float c = s.explicitCross > 0.f
50 ? s.explicitCross
51 : (s.percentCross > 0.f ? s.percentCross * availCross : s.basisCross);
52 c = clampV(c, s.minCross, s.maxCross);
53 mainSize[size_t(i)] = m;
54 crossSize[size_t(i)] = c;
55
56 if (s.absolute) continue;
57 growSum += effectiveGrow(s);
58 flowTotal += m + s.marginBefore + s.marginAfter;
59 ++flowCount;
60 }
61
62 const int gapCount = std::max(0, flowCount - 1);
63 float total = flowTotal + float(gapCount) * gap;
64 float freeSpace = availMain - total;
65 if (freeSpace < 0.f) freeSpace = 0.f;
66
67 float leading = 0.f;
68 float between = gap;
69 if (growSum > 0.f) {
70 for (int i = 0; i < n; ++i) {
71 const FlexItemSpec &s = items[size_t(i)];
72 if (s.absolute) continue;
73 const float g = effectiveGrow(s);
74 if (g > 0.f) mainSize[size_t(i)] += freeSpace * (g / growSum);
75 }
76 } else if (freeSpace > 0.f && flowCount > 0) {
77 switch (justify) {
78 case FlexJustify::Center:
79 leading = freeSpace * 0.5f;
80 break;
81 case FlexJustify::End:
82 leading = freeSpace;
83 break;
84 case FlexJustify::SpaceBetween:
85 if (gapCount > 0) between = gap + freeSpace / float(gapCount);
86 break;
87 case FlexJustify::SpaceAround:
88 if (flowCount > 0) {
89 const float edge = freeSpace / float(flowCount * 2);
90 leading = edge;
91 between = gap + edge * 2.f;
92 }
93 break;
94 case FlexJustify::Start:
95 default:
96 break;
97 }
98 }
99
100 float cur = leading;
101 for (int i = 0; i < n; ++i) {
102 const FlexItemSpec &s = items[size_t(i)];
103 if (s.absolute) {
104 mainPos[size_t(i)] =
105 s.anchorMain * availMain + s.posMain - s.anchorMain * mainSize[size_t(i)];
106 continue;
107 }
108 mainPos[size_t(i)] = cur + s.marginBefore;
109 cur += s.marginBefore + mainSize[size_t(i)] + s.marginAfter + between;
110 }
111 const float contentMain = flowCount > 0 ? cur - between : 0.f;
112
113 float contentCross = 0.f;
114 for (int i = 0; i < n; ++i) {
115 const FlexItemSpec &s = items[size_t(i)];
116 if (s.absolute) {
117 crossPos[size_t(i)] =
118 s.anchorCross * availCross + s.posCross - s.anchorCross * crossSize[size_t(i)];
119 contentCross = std::max(contentCross,
120 crossPos[size_t(i)] + crossSize[size_t(i)] +
121 s.marginCrossAfter);
122 continue;
123 }
124 const float cb = s.marginCrossBefore;
125 const float ca = s.marginCrossAfter;
126 const bool stretch = s.alignSelf >= 0
127 ? s.alignSelf == int(FlexAlign::Stretch)
128 : containerAlign == FlexAlign::Stretch;
129 if (stretch && s.explicitCross <= 0.f && s.percentCross <= 0.f) {
130 float c = availCross - cb - ca;
131 if (c < 0.f) c = 0.f;
132 crossSize[size_t(i)] = c;
133 }
134 switch (containerAlign) {
135 case FlexAlign::Center:
136 crossPos[size_t(i)] = (availCross - crossSize[size_t(i)] - cb - ca) * 0.5f + cb;
137 break;
138 case FlexAlign::End:
139 crossPos[size_t(i)] = availCross - crossSize[size_t(i)] - ca;
140 break;
141 case FlexAlign::Start:
142 case FlexAlign::Stretch:
143 default:
144 crossPos[size_t(i)] = cb;
145 break;
146 }
147 contentCross = std::max(contentCross,
148 crossPos[size_t(i)] + crossSize[size_t(i)] + ca);
149 }
150
151 for (int i = 0; i < n; ++i) {
152 FlexRect &r = res.items[size_t(i)];
153 if (row) {
154 r.x = mainPos[size_t(i)];
155 r.y = crossPos[size_t(i)];
156 r.w = mainSize[size_t(i)];
157 r.h = crossSize[size_t(i)];
158 } else {
159 r.x = crossPos[size_t(i)];
160 r.y = mainPos[size_t(i)];
161 r.w = crossSize[size_t(i)];
162 r.h = mainSize[size_t(i)];
163 }
164 }
165
166 res.contentW = row ? contentMain : contentCross;
167 res.contentH = row ? contentCross : contentMain;
168 return res;
169}
