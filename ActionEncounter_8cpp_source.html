43Result<double> ActionEncounter::hit(bool playerAttacks, double amount, int tick) {
44 if (!impl_)
45 return Result<double>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation, "Encounter is closed"));
46 if (tick < 0 || tick < impl_->lastTick || !std::isfinite(amount) || amount <= 0.0)
47 return Result<double>::failure(
48 Diagnostic::error(DiagnosticCode::InvalidArgument, "Positive finite damage and monotonic tick required"));
49 if (impl_->pending)
50 return Result<double>::failure(Diagnostic::error(DiagnosticCode::Conflict, "Reward delivery is pending"));
51 const auto source = playerAttacks ? impl_->player : impl_->enemy;
52 const auto target = playerAttacks ? impl_->enemy : impl_->player;
53 auto* attacker = dynamic_cast<RPGActor*>(ecs::try_get(source.actor));
54 auto* defender = dynamic_cast<RPGActor*>(ecs::try_get(target.actor));
55 if (!attacker || !defender)
56 return Result<double>::failure(
57 Diagnostic::error(DiagnosticCode::StaleHandle, "Encounter participant was destroyed"));
58 if (attacker->getCurrent("hp") <= 0 || defender->getCurrent("hp") <= 0)
59 return Result<double>::failure(
60 Diagnostic::error(DiagnosticCode::PreconditionViolation, "Both participants must be alive"));
61 // The registry chooses its first accepting provider. Reject ambiguous ownership
62 // before any provider can commit damage or enqueue a receipt for another encounter.
63 std::size_t accepting = 0;
64 bool ownsTarget = false;
65 cap::forEach<action::IActionDamageSink>([&](action::IActionDamageSink* sink) {
66 if (!sink->supports(target.actor)) return;
67 ++accepting;
68 ownsTarget = ownsTarget || sink == &impl_->sink;
69 });
70 if (accepting != 1 || !ownsTarget)
71 return Result<double>::failure(Diagnostic::error(
72 DiagnosticCode::Conflict, "Encounter target must have exactly one owning damage provider"));
73 auto registry = action::ActionNotifyRegistry::withBuiltins();
74 if (!registry) return Result<double>::failure(registry.status());
75 action::ActionTimelineEvent event;
76 event.type = *LogicalId::parse("combat:damage");
77 event.payload = {{"damageType", "Damage.Physical"}, {"amount", amount}};
78 action::ActionNotifyContext context;
79 context.source = source.actor;
80 context.targets = {target.actor};
81 context.tick = SimulationTick(static_cast<unsigned long long>(tick));
82 auto dispatched = registry.value().dispatch(event, context);
83 if (!dispatched) return Result<double>::failure(dispatched.status());
84 auto receipts = impl_->sink.takeReceipts();
85 if (receipts.size() != 1)
86 return Result<double>::failure(
87 Diagnostic::error(DiagnosticCode::InvariantViolation, "Encounter requires sole damage provider ownership"));
88 impl_->lastTick = tick;
89 if (playerAttacks && receipts.front().killed) impl_->pending = receipts.front();
90 return Result<double>::success(receipts.front().applied);
91}
