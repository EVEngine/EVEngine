45inline size_t elementSize(OnnxElement e) {
46 switch (e) {
47 case OnnxElement::Float32:
48 case OnnxElement::Int32: return 4;
49 case OnnxElement::Int64: return 8;
50 case OnnxElement::UInt8:
51 case OnnxElement::Int8:
52 case OnnxElement::Bool: return 1;
53 }
54 throw Failure("Unsupported ONNX element type", DiagnosticCode::Unsupported);
55}
56inline size_t count(const std::vector<int64_t>& shape) {
57 if (shape.size() > 6) throw Failure("ONNX rank exceeds six", DiagnosticCode::Unsupported);
58 size_t n = 1;
59 for (int64_t d : shape) {
60 if (d < 0 || d > INT32_MAX || (d && n > (128u * 1024u * 1024u) / static_cast<size_t>(d)))
61 throw Failure("Invalid or excessive tensor shape");
62 n *= static_cast<size_t>(d);
63 }
64 return n;
65}
66inline void validate(const RuntimeTensor& v) {
67 if (count(v.shape) * elementSize(v.element) != v.bytes.size()) throw Failure("Tensor byte count mismatch");
68}

82inline int64_t integer(const RuntimeTensor& v, size_t i = 0) {
83 if (i >= count(v.shape)) throw Failure("Integer index out of bounds");
84 switch (v.element) {
85 case OnnxElement::Int64: return read<int64_t>(v, i);
86 case OnnxElement::Int32: return read<int32_t>(v, i);
87 case OnnxElement::Int8: return v.bytes[i] >= 128 ? int(v.bytes[i]) - 256 : v.bytes[i];
88 case OnnxElement::UInt8:
89 case OnnxElement::Bool: return v.bytes[i];
90 default: throw Failure("Expected integer tensor");
91 }
92}
