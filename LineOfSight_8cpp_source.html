34Cell fromCube(double x, double y, double z) {
35 std::int64_t rx = static_cast<std::int64_t>(x < 0 ? x - 0.5 : x + 0.5);
36 std::int64_t ry = static_cast<std::int64_t>(y < 0 ? y - 0.5 : y + 0.5);
37 std::int64_t rz = static_cast<std::int64_t>(z < 0 ? z - 0.5 : z + 0.5);
38 const std::int64_t dx = std::llabs(rx - static_cast<std::int64_t>(x));
39 const std::int64_t dy = std::llabs(ry - static_cast<std::int64_t>(y));
40 const std::int64_t dz = std::llabs(rz - static_cast<std::int64_t>(z));
41 if (dx > dy && dx > dz) {
42 rx = -ry - rz;
43 } else if (dy > dz) {
44 ry = -rx - rz;
45 } else {
46 rz = -rx - ry;
47 }
48 return Cell{static_cast<int>(rx), static_cast<int>(ry), 0};
49}

58std::vector<Cell> traceInterior(const BoardState& board, Cell from, Cell to, bool& reverse) {
59 std::vector<Cell> interior;
60 if (from.layer != to.layer) return interior;
61 const int layer = from.layer;
62
63 Cell start = from;
64 Cell end = to;
65 reverse = (to < from);
66 if (reverse) std::swap(start, end);
67
68 if (board.topology() == BoardTopology::HexAxial) {
69 // Cube-space interpolation: the standard hex line, deduplicated so a trace that clips
70 // two hexes in one step does not repeat them.
71 const Hex a = toHex(start);
72 const Hex b = toHex(end);
73 const int dx = b.q - a.q;
74 const int dy = b.r - a.r;
75 const int dz = -dx - dy;
76 const int steps = std::max({std::abs(dx), std::abs(dy), std::abs(dz)});
77 Cell previous = start;
78 for (int step = 1; step < steps; ++step) {
79 const double t = static_cast<double>(step) / static_cast<double>(steps);
80 Cell cell = fromCube(static_cast<double>(a.q) + static_cast<double>(dx) * t,
81 static_cast<double>(a.r) + static_cast<double>(dy) * t,
82 static_cast<double>(dz) * t);
83 cell.layer = layer;
84 if (cell == previous) continue;
85 interior.push_back(cell);
86 previous = cell;
87 }
88 return interior;
89 }
90
91 // Square topologies: a supercover walk. When both error conditions hold in one step the
92 // trace takes a corner, so a diagonal that squeezes between two cells is blocked if
93 // either of them blocks: a shot does not pass through a keyhole.
94 const int dx = end.x - start.x;
95 const int dy = end.y - start.y;
96 const int stepX = (dx > 0) - (dx < 0);
97 const int stepY = (dy > 0) - (dy < 0);
98 const int absX = std::abs(dx);
99 const int absY = std::abs(dy);
100 int x = start.x;
101 int y = start.y;
102 int error = absX - absY;
103 while (x != end.x || y != end.y) {
104 const int doubled = 2 * error;
105 if (doubled > -absY) {
106 error -= absY;
107 x += stepX;
108 }
109 if (doubled < absX) {
110 error += absX;
111 y += stepY;
112 }
113 if (x == end.x && y == end.y) break;
114 Cell cell{x, y, layer};
115 interior.push_back(cell);
116 }
117 return interior;
118}

131Result<bool> GridLineOfSightPolicy::visible(const BoardState& board, Cell from, Cell to) const {
132 if (from.layer != to.layer)
133 return Result<bool>::failure(
134 Diagnostic::error(DiagnosticCode::Unsupported, "line of sight does not cross layers", "from.layer"));
135 if (!board.contains(from))
136 return Result<bool>::failure(
137 Diagnostic::error(DiagnosticCode::NotFound, "line of sight origin is not on the board", "from"));
138 if (!board.contains(to))
139 return Result<bool>::failure(
140 Diagnostic::error(DiagnosticCode::NotFound, "line of sight target is not on the board", "to"));
141 if (from == to) return Result<bool>::success(true);
142
143 bool reverse = false;
144 auto interior = traceInterior(board, from, to, reverse);
145 for (const Cell cell : interior) {
146 if (blocksSight(board, cell)) return Result<bool>::success(false);
147 }
148 return Result<bool>::success(true);
149}

151Result<CoverLevel> GridCoverPolicy::cover(const BoardState& board, Cell attacker, Cell target) const {
152 if (attacker.layer != target.layer)
153 return Result<CoverLevel>::failure(
154 Diagnostic::error(DiagnosticCode::Unsupported, "cover does not cross layers", "attacker.layer"));
155 if (!board.contains(target))
156 return Result<CoverLevel>::failure(
157 Diagnostic::error(DiagnosticCode::NotFound, "cover target is not on the board", "target"));
158 if (!board.contains(attacker))
159 return Result<CoverLevel>::failure(
160 Diagnostic::error(DiagnosticCode::NotFound, "cover attacker is not on the board", "attacker"));
161
162 // Full cover first: something sits on the incoming line.
163 bool reverse = false;
164 for (const Cell cell : traceInterior(board, attacker, target, reverse)) {
165 if (blocksSight(board, cell)) return Result<CoverLevel>::success(CoverLevel::Full);
166 }
167 // Half cover: a blocker next to the target, but not between attacker and target.
168 for (const Cell neighbour : board.neighbours(target)) {
169 // A neighbour the board does not contain is not cover a target can use.
170 if (board.contains(neighbour) && blocksSight(board, neighbour))
171 return Result<CoverLevel>::success(CoverLevel::Half);
172 }
173 return Result<CoverLevel>::success(CoverLevel::None);
174}

186Result<std::vector<Cell>> visibleCellsInRange(const BoardState& board, const ILineOfSightPolicy& policy,
187 Cell origin, int minimum, int maximum, CellRangeMetric metric) {
188 auto inRange = PathQuery::cellsInRange(board, origin, minimum, maximum, metric);
189 if (!inRange) return Result<std::vector<Cell>>::failure(inRange.status());
190 const std::vector<Cell> candidates = std::move(inRange).takeValue();
191
192 std::vector<Cell> visible;
193 visible.reserve(candidates.size());
194 for (const Cell cell : candidates) {
195 auto sight = policy.visible(board, origin, cell);
196 if (!sight) return Result<std::vector<Cell>>::failure(sight.status());
197 if (sight.value()) visible.push_back(cell);
198 }
199 return Result<std::vector<Cell>>::success(std::move(visible));
200}
