56void AtmosphereVolume::injectHeightFog(float baseExtinction, const glm::vec3 &albedo,
57 float baseHeight, float heightFalloff, float minWorldY,
58 float maxWorldY) {
59 const float sigma = std::max(baseExtinction, 0.f);
60 const glm::vec3 omega = glm::clamp(albedo, glm::vec3(0.f), glm::vec3(1.f));
61 const float falloff = std::max(heightFalloff, 0.f);
62 for (int y = 0; y < height_; ++y) {
63 const float ty = (float(y) + 0.5f) / float(height_);
64 const float worldY = minWorldY + (maxWorldY - minWorldY) * ty;
65 const float extinction = sigma * std::exp(-falloff * std::max(worldY - baseHeight, 0.f));
66 for (int z = 0; z < depth_; ++z) {
67 for (int x = 0; x < width_; ++x) {
68 FogFroxel &f = at(x, y, z);
69 f.extinction += extinction;
70 f.scattering += omega * extinction;
71 }
72 }
73 }

76void AtmosphereVolume::injectLocalVolume(const FogVolume &volume, const glm::vec3 &worldMin,
77 const glm::vec3 &worldMax) {
78 for (int z = 0; z < depth_; ++z) {
79 for (int y = 0; y < height_; ++y) {
80 for (int x = 0; x < width_; ++x) {
81 const glm::vec3 uvw((float(x) + 0.5f) / float(width_),
82 (float(y) + 0.5f) / float(height_),
83 (float(z) + 0.5f) / float(depth_));
84 const glm::vec3 world = worldMin + (worldMax - worldMin) * uvw;
85 const float signedExtinction = volume.sampleExtinction(world);
86 if (std::fabs(signedExtinction) <= 1e-8f) continue;
87 FogFroxel &f = at(x, y, z);
88 const float previous = f.extinction;
89 f.extinction = std::max(0.f, previous + signedExtinction);
90 if (signedExtinction > 0.f) {
91 f.scattering += volume.getAlbedo() * signedExtinction;
92 f.emissive += volume.getEmissive() * signedExtinction;
93 f.anisotropy = volume.getAnisotropy();
94 } else if (previous > 1e-6f) {
95 const float scale = f.extinction / previous;
96 f.scattering *= scale;
97 f.emissive *= scale;
98 }
99 }
100 }
101 }

107 glm::vec4 world = invViewProj * glm::vec4(ndcX, ndcY, ndcZ, 1.f);
108 return glm::vec3(world) / world.w;

110 for (int y = 0; y < height_; ++y) {
111 const float v = (float(y) + 0.5f) / float(height_);
112 for (int x = 0; x < width_; ++x) {

114 const glm::vec3 nearPoint = unproject(u * 2.f - 1.f, v * 2.f - 1.f, 0.f);

139void AtmosphereVolume::injectDensityGraph(const VolumeDensityGraph &graph,
140 const glm::vec3 &worldMin,
141 const glm::vec3 &worldMax, float extinctionScale,
142 const glm::vec3 &albedo, float time) {
143 const glm::vec3 omega = glm::clamp(albedo, glm::vec3(0.f), glm::vec3(1.f));
144 for (int z = 0; z < depth_; ++z) {
145 for (int y = 0; y < height_; ++y) {
146 for (int x = 0; x < width_; ++x) {
147 const glm::vec3 uvw((float(x) + 0.5f) / float(width_),
148 (float(y) + 0.5f) / float(height_),
149 (float(z) + 0.5f) / float(depth_));
150 const glm::vec3 world = worldMin + (worldMax - worldMin) * uvw;
151 const float extinction = graph.evaluate(world, time) * extinctionScale;
152 FogFroxel &f = at(x, y, z);
153 const float previous = f.extinction;
154 f.extinction = std::max(0.f, previous + extinction);
155 if (extinction >= 0.f)
156 f.scattering += omega * extinction;
157 else if (previous > 1e-6f)
158 f.scattering *= f.extinction / previous;
159 }
160 }

182void AtmosphereVolume::integrate(const glm::vec3 &lightColor, float phaseScale) {
183 const glm::vec3 light = glm::max(lightColor, glm::vec3(0.f)) * std::max(phaseScale, 0.f);
184 for (int y = 0; y < height_; ++y) {
185 for (int x = 0; x < width_; ++x) {
186 glm::vec3 radiance(0.f);
187 float transmittance = 1.f;
188 float previousDistance = nearDistance_;
189 for (int z = 0; z < depth_; ++z) {
190 const float distance = sliceDistance(z);
191 const float stepLength = std::max(distance - previousDistance, 0.f);
192 previousDistance = distance;
193 const FogFroxel &f = at(x, y, z);
194 const float opticalDepth = f.extinction * stepLength;
195 const float stepTransmittance = std::exp(-opticalDepth);
196 const float integral = f.extinction > 1e-6f
197 ? (1.f - stepTransmittance) / f.extinction
198 : stepLength;
199 radiance += transmittance *
200 (f.scattering * light * f.lightVisibility + f.emissive) * integral;
201 transmittance *= stepTransmittance;
202 integrated_[index(x, y, z)] = glm::vec4(radiance, transmittance);
203 }
204 }
205 }

208void AtmosphereVolume::integrateLocalLights(const std::vector<VolumetricLight> &lights,
209 const glm::vec3 &worldMin,
210 const glm::vec3 &worldMax,
211 const glm::vec3 &ambientLight) {
212 for (int y = 0; y < height_; ++y) {
213 for (int x = 0; x < width_; ++x) {
214 glm::vec3 radiance(0.f);
215 float transmittance = 1.f;
216 float previousDistance = nearDistance_;
217 for (int z = 0; z < depth_; ++z) {
218 const glm::vec3 uvw((float(x) + 0.5f) / float(width_),
219 (float(y) + 0.5f) / float(height_),
220 (float(z) + 0.5f) / float(depth_));
221 const glm::vec3 world = worldMin + (worldMax - worldMin) * uvw;
222 glm::vec3 incident = glm::max(ambientLight, glm::vec3(0.f));
223 for (const VolumetricLight &light : lights) {
224 if (!light.enabled || light.radius <= 0.f || light.intensity <= 0.f) continue;
225 const float d = glm::length(world - light.position);
226 if (d >= light.radius) continue;
227 const float radial = 1.f - d / light.radius;
228 incident += glm::max(light.color, glm::vec3(0.f)) * light.intensity *
229 radial * radial;
230 }
231 const float distance = sliceDistance(z);
232 const float stepLength = std::max(distance - previousDistance, 0.f);
233 previousDistance = distance;
234 const FogFroxel &f = at(x, y, z);
235 const float opticalDepth = f.extinction * stepLength;
236 const float stepTransmittance = std::exp(-opticalDepth);
237 const float integral = f.extinction > 1e-6f
238 ? (1.f - stepTransmittance) / f.extinction
239 : stepLength;
240 radiance += transmittance *
241 (f.scattering * incident * f.lightVisibility + f.emissive) * integral;
242 transmittance *= stepTransmittance;
243 integrated_[index(x, y, z)] = glm::vec4(radiance, transmittance);
244 }
245 }
246 }

253std::size_t AtmosphereVolume::blendHistory(const AtmosphereVolume &history, float historyWeight,
254 float rejectionThreshold) {
255 if (history.width_ != width_ || history.height_ != height_ || history.depth_ != depth_)
256 return integrated_.size();
257 const float baseWeight = std::clamp(historyWeight, 0.f, 1.f);
258 const float threshold = std::max(rejectionThreshold, 0.f);
259 std::size_t rejected = 0;
260 for (std::size_t i = 0; i < integrated_.size(); ++i) {
261 const glm::vec4 current = integrated_[i];
262 const glm::vec4 previous = history.integrated_[i];
263 const float currentLuma = glm::dot(glm::vec3(current), glm::vec3(0.2126f, 0.7152f, 0.0722f));
264 const float previousLuma = glm::dot(glm::vec3(previous), glm::vec3(0.2126f, 0.7152f, 0.0722f));
265 const float relativeDelta = std::fabs(currentLuma - previousLuma) /
266 std::max(std::max(currentLuma, previousLuma), 1e-4f);
267 const bool reject = relativeDelta > threshold;
268 if (reject) ++rejected;
269 const float weight = reject ? 0.f : baseWeight;
270 integrated_[i] = glm::mix(current, previous, weight);
271 }
272 return rejected;
273}
