26bool validMesh(const MeshBuild& mesh) noexcept {
27 if (mesh.empty() || mesh.positions().size() % 3u != 0u || mesh.indices().size() % 3u != 0u) return false;
28 const std::size_t vertices = mesh.positions().size() / 3u;
29 if ((!mesh.normals().empty() && mesh.normals().size() != mesh.positions().size()) ||
30 (!mesh.uvs().empty() && mesh.uvs().size() != vertices * 2u) ||
31 (!mesh.colors().empty() && mesh.colors().size() != vertices * 4u))
32 return false;
33 return std::all_of(mesh.positions().begin(), mesh.positions().end(),
34 [](float value) { return std::isfinite(value); }) &&
35 std::all_of(mesh.normals().begin(), mesh.normals().end(),
36 [](float value) { return std::isfinite(value); }) &&

38 std::all_of(mesh.colors().begin(), mesh.colors().end(), [](float value) { return std::isfinite(value); }) &&
39 std::all_of(mesh.indices().begin(), mesh.indices().end(),
40 [vertices](std::uint32_t index) { return index < vertices; });

70 if (!array || (!array->empty() && array->size() != vertices * 4u)) return fail();

82 }
