153std::vector<EditorDiagnostic> TextureRecipeTarget::validate() const {
154 std::vector<EditorDiagnostic> d;
155 auto& r = procgen::TextureRecipeRegistry::instance();
156 const auto* desc = r.descriptor(recipe_);
157 if (!desc) {
158 d.push_back(diagnostic("editor.texture-recipe.unknown", DiagnosticSeverity::Error,
159 "Texture recipe is not registered"));
160 return d;
161 }
162 const auto s = schema({});
163 for (const auto& p : desc->params) {
164 auto i = values_.find(p.key);
165 auto x = s.find(PropertyPath("param." + p.key));
166 if (i == values_.end() || !x || !validatePropertyValue(*x, i->second).ok())
167 d.push_back(diagnostic("editor.texture-recipe.parameter", DiagnosticSeverity::Error,
168 "Invalid texture parameter: " + p.key));
169 }
170 if (values_.size() != desc->params.size())
171 d.push_back(diagnostic("editor.texture-recipe.extra", DiagnosticSeverity::Error,
172 "Texture recipe contains unknown parameters"));
173 return d;
174}
175EditorResult<void> TextureRecipeTarget::applyDomainOperation(const DomainOperation& op) {
176 if (op.target != TargetId(id_) || op.type != "texture-recipe.replace.v1")
177 return fail<void>(EditorStatus::Rejected, "editor.texture-recipe.operation",
178 "Texture recipe operation is invalid");
179 const auto *recipe = field(op.payload, "recipe"), *values = field(op.payload, "values");
180 const auto* rs = recipe ? recipe->getIf<std::string>() : nullptr;
181 const auto* vo = values ? values->getIf<EditorValue::Object>() : nullptr;
182 if (!rs || *rs != recipe_ || !vo || vo->size() > 128)
183 return fail<void>(EditorStatus::Rejected, "editor.texture-recipe.payload", "Texture recipe payload is invalid");
184 TextureRecipeTarget candidate = *this;
185 candidate.values_ = *vo;
186 if (hasErrors(candidate.validate()))
187 return fail<void>(EditorStatus::Rejected, "editor.texture-recipe.invalid", "Texture recipe validation failed");
188 values_ = std::move(candidate.values_);
189 ++revision_;
190 dirty_.include(0, 0);
191 return eve::editing::applied<void>();

224EditorResult<TextureRecipePreviewArtifact> TextureRecipePreviewRuntime::generate(
225 const TextureRecipeTarget& target) {
226 const auto diagnostics = target.validate();
227 if (hasErrors(diagnostics)) {
228 return EditorResult<TextureRecipePreviewArtifact>::failure(
229 eve::Status(EditorStatus::Rejected, diagnostics));
230 }
231 procgen::Params params;
232 for (const auto& [key, v] : target.values()) {
233 if (auto* x = v.getIf<int64_t>())
234 params.setInt(key, static_cast<int>(*x));
235 else if (auto* x = v.getIf<double>())
236 params.setFloat(key, static_cast<float>(*x));
237 else if (auto* x = v.getIf<bool>())
238 params.setBool(key, *x);
239 else if (auto* x = v.getIf<std::string>())
240 params.setString(key, *x);
241 }
242 std::string error;
243 auto candidate = procgen::TextureRecipeRegistry::instance().generate(target.recipe(), params, error);
244 if (!candidate)
245 return fail<TextureRecipePreviewArtifact>(EditorStatus::Failed, "editor.texture-recipe.generate", error);
246 std::uint64_t hash = 1469598103934665603ull;
247 auto* bytes = static_cast<const unsigned char*>(candidate->getData());
248 for (std::size_t i = 0; i < candidate->getSize(); ++i) {
249 hash ^= bytes[i];
250 hash *= 1099511628211ull;
251 }
252 TextureRecipePreviewArtifact artifact{target.revision(), candidate->getWidth(), candidate->getHeight(), hash};
253 image_ = std::move(candidate);
254 revision_ = target.revision();
255 return eve::editing::applied<TextureRecipePreviewArtifact>(artifact, diagnostics);
256}
