21std::string extensionOf(const std::string &path) {
22 const auto pos = path.find_last_of('.');
23 if (pos == std::string::npos) return {};
24 std::string ext = path.substr(pos);
25 for (char &c : ext) c = static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
26 return ext;
27}

29bool isImagePath(const std::string &path) {
30 const std::string ext = extensionOf(path);
31 return ext == ".png" || ext == ".jpg" || ext == ".jpeg" || ext == ".bmp" || ext == ".tga" ||
32 ext == ".gif" || ext == ".webp" || ext == ".exr" || ext == ".hdr";
33}

36std::string normalize(std::string path) {
37 for (char &c : path) {
38 if (c == '\\') c = '/';
39 }
40 while (path.size() >= 2 && path[0] == '.' && path[1] == '/') path.erase(0, 2);
41 while (path.size() > 1 && path.back() == '/') path.pop_back();
42 return path;
43}

45class TileLayerReloader : public eve::caps::IAssetReloader {
46public:
47 const char *reloadKind() const override { return "tilemap"; }
48
49 bool handlesPath(const std::string &normPath) const override {
50 const std::string ext = extensionOf(normPath);
51 return ext == ".json" || ext == ".tmj" || ext == ".tsj" || ext == ".tsx" || isImagePath(normPath);
52 }
53
54 eve::Result<bool> reload(const std::string &normPath) override {
55 if (ecs::current()->getManager<TileLayer>() == nullptr) return eve::Result<bool>::success(false);
56 return eve::Result<bool>::success(isImagePath(normPath) ? rebindTileset(normPath) : reloadConfig(normPath));
57 }
58
59private:
60 static bool reloadConfig(const std::string &normPath) {
61 int reloaded = 0;
62 auto view = ecs::View<TileLayer, TileLayer::Config, TileLayer::Resource>();
63 for (auto it = view.begin(); it != view.end(); ++it) {
64 auto [cfg, res] = *it;
65 if (!cfg->entity || res->path.empty()) continue;
66 bool matches = normalize(res->path) == normPath;
67 for (const std::string &dependency : res->dependencyPaths)
68 matches = matches || normalize(dependency) == normPath;
69 if (!matches) continue;
70 if (reloadConfigFile(cfg->entity, nullptr)) ++reloaded;
71 }
72 return reloaded > 0;
73 }
74
75 static bool rebindTileset(const std::string &normPath) {
76 auto *gfx = eve::ModuleManager::getInstance<eve::graphics::Graphics>("Graphics");
77 if (!gfx) return false;
78
79 bool any = false;
80 auto view = ecs::View<TileLayer, TileLayer::Config, TileLayer::Resource, TileLayer::Tileset>();
81 for (auto it = view.begin(); it != view.end(); ++it) {
82 auto [cfg, res, ts] = *it;
83 if (!cfg->entity) continue;
84 try {
85 for (auto &atlas : ts->atlases) {
86 if (normalize(atlas.imagePath) != normPath) continue;
87 atlas.texture = gfx->newTextureFromFile(normPath);
88 if (atlas.firstGid == ts->firstGid) ts->texture = atlas.texture;
89 any = true;
90 }
91 } catch (...) {
92 }
93 }
94 return any;
95 }
96};
