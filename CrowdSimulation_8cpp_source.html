22void Crowd::Impl::stepAgents(float dt) {
23 const int n = int(actions.size());
24 if (avoidance.enabled) {
25 avoidanceMaxSpeed = n == 0 ? 0.f : *std::max_element(maxSpeeds.begin(), maxSpeeds.end());
26 avoidanceMaxRadius = n == 0 ? 0.f : *std::max_element(radii.begin(), radii.end());
27 }
28 spendPreferredWork(std::int64_t(n) * 5);
29 nextVxs.resize(size_t(n));
30 nextVys.resize(size_t(n));
31 preferredVxs.resize(size_t(n));
32 preferredVys.resize(size_t(n));
33 arriveFactors.resize(size_t(n), 1.f);
34 preparePreferredVelocities(dt);
35 if (avoidance.enabled) prepareAvoidanceNeighbors();
36 dispatchWork(size_t(n), [&](std::size_t first, std::size_t last) {
37 for (int i = int(first); i < int(last); ++i) {
38 if (interactions[size_t(i)].holdPosition) {
39 nextVxs[size_t(i)] = nextVys[size_t(i)] = speeds[size_t(i)] = 0.f;
40 continue;
41 }
42 const float desX = preferredVxs[size_t(i)], desY = preferredVys[size_t(i)];
43 const float maxSpeed = maxSpeeds[size_t(i)];
44 const float arriveK = arriveFactors[size_t(i)];
45 // 加速度限幅。
46 float avx = vxs[size_t(i)];
47 float avy = vys[size_t(i)];
48 float ddx = desX - avx;
49 float ddy = desY - avy;
50 const float ddl = std::sqrt(ddx * ddx + ddy * ddy);
51 const float maxDelta = std::max(maxAccels[size_t(i)] * dt, 0.f);
52 if (ddl > maxDelta && ddl > 1e-6f) {
53 ddx *= maxDelta / ddl;
54 ddy *= maxDelta / ddl;
55 }
56 avx += ddx;
57 avy += ddy;
58
59 if (avoidance.enabled) selectAvoidanceVelocity(size_t(i), dt, avx, avy);
60
61 // 速度限幅。
62 float sp = std::sqrt(avx * avx + avy * avy);
63 if (sp > maxSpeed && sp > 1e-6f) {
64 avx *= maxSpeed / sp;
65 avy *= maxSpeed / sp;
66 sp = maxSpeed;
67 }
68 nextVxs[size_t(i)] = avx;
69 nextVys[size_t(i)] = avy;
70 speeds[size_t(i)] = sp;
71
72 // 接近目标时对速度做阻尼，快速消除目标点附近的来回振荡。
73 if (arriveK < 1.f) {
74 const float damp = 1.f - (1.f - arriveK) * std::min(1.f, dt * 4.f);
75 nextVxs[size_t(i)] *= damp;
76 nextVys[size_t(i)] *= damp;
77 speeds[size_t(i)] *= damp;
78 }
79
80 // 转向：heading 以 turnRate·dt 为上限向速度方向收敛。
81 const float spAfter = speeds[size_t(i)];
82 if (spAfter > 1e-4f) {
83 const float targetAng = std::atan2(nextVys[size_t(i)], nextVxs[size_t(i)]);
84 const float delta = wrapPi(targetAng - headings[size_t(i)]);
85 const float maxTurn = std::max(turnRates[size_t(i)] * dt, 0.f);
86 headings[size_t(i)] += std::clamp(delta, -maxTurn, maxTurn);
87 }
88 }
89 });
90 vxs.swap(nextVxs);
91 vys.swap(nextVys);
92 for (int i = 0; i < n; ++i) {
93 xs[size_t(i)] += vxs[size_t(i)] * dt;
94 ys[size_t(i)] += vys[size_t(i)] * dt;
95 }
96}

98void Crowd::Impl::resolveWalls() {
99 if (!field.valid()) return;
100 const float width = float(field.getWidth()) * field.getCellSize();
101 const float height = float(field.getHeight()) * field.getCellSize();
102 for (size_t i = 0; i < xs.size(); ++i) {
103 for (int pass = 0; pass < 4; ++pass) {
104 const float oldX = xs[i], oldY = ys[i];
105 field.resolvePenetration(xs[i], ys[i], radii[i]);
106 if (clampToField) {
107 // Field origin is the lower cell corner, consistently with sampling.
108 const float rx = std::min(radii[i], width * 0.5f);
109 const float ry = std::min(radii[i], height * 0.5f);
110 xs[i] = std::clamp(xs[i], field.getOriginX() + rx, field.getOriginX() + width - rx);
111 ys[i] = std::clamp(ys[i], field.getOriginY() + ry, field.getOriginY() + height - ry);
112 }
113 const float dx = xs[i] - oldX, dy = ys[i] - oldY;
114 const float length2 = dx * dx + dy * dy;
115 if (length2 == 0.f) break;
116 // Remove velocity into the constraint; retaining it causes permanent
117 // walking-in-place when a requested destination lies outside the field.
118 const float inward = vxs[i] * dx + vys[i] * dy;
119 if (inward < 0.f) {
120 vxs[i] -= dx * inward / length2;
121 vys[i] -= dy * inward / length2;
122 speeds[i] = std::hypot(vxs[i], vys[i]);
123 }
124 }
125 }
