35std::uint8_t paeth(std::uint8_t left, std::uint8_t above, std::uint8_t upperLeft) {
36 const int prediction = int(left) + int(above) - int(upperLeft);
37 const int leftDistance = std::abs(prediction - int(left));
38 const int aboveDistance = std::abs(prediction - int(above));
39 const int cornerDistance = std::abs(prediction - int(upperLeft));
40 if (leftDistance <= aboveDistance && leftDistance <= cornerDistance) return left;
41 return aboveDistance <= cornerDistance ? above : upperLeft;
42}

44Result<std::vector<std::uint8_t>> decodePng(std::span<const std::uint8_t> bytes,
45 std::uint32_t expectedWidth,
46 std::uint32_t expectedHeight,
47 std::uint64_t maximumDecodedBytes) {
48 static constexpr std::uint8_t signature[] = {0x89, 'P', 'N', 'G', 0x0d, 0x0a, 0x1a, 0x0a};
49 if (bytes.size() < 33 || !std::equal(std::begin(signature), std::end(signature), bytes.begin()))
50 return failure<std::vector<std::uint8_t>>(DiagnosticCode::ParseError,
51 "PNG signature or header is invalid");
52 std::size_t cursor = 8;
53 bool sawHeader = false;
54 bool sawEnd = false;
55 std::uint32_t width = 0, height = 0;
56 std::uint8_t channels = 0;
57 std::vector<std::uint8_t> compressed;
58 while (cursor < bytes.size()) {
59 if (bytes.size() - cursor < 12)
60 return failure<std::vector<std::uint8_t>>(DiagnosticCode::ParseError,
61 "PNG chunk header is truncated");
62 const std::uint32_t size = big32(bytes, cursor);
63 if (size > bytes.size() - cursor - 12)
64 return failure<std::vector<std::uint8_t>>(DiagnosticCode::ParseError,
65 "PNG chunk payload is truncated");
66 const auto type = bytes.subspan(cursor + 4, 4);
67 const auto payload = bytes.subspan(cursor + 8, size);
68 const std::uint32_t storedCrc = big32(bytes, cursor + 8 + size);
69 uLong crc = crc32(0, Z_NULL, 0);
70 crc = crc32(crc, type.data(), 4);
71 if (!payload.empty()) crc = crc32(crc, payload.data(), static_cast<uInt>(payload.size()));
72 if (static_cast<std::uint32_t>(crc) != storedCrc)
73 return failure<std::vector<std::uint8_t>>(DiagnosticCode::HashMismatch,
74 "PNG chunk CRC does not match");
75 const std::string_view name(reinterpret_cast<const char*>(type.data()), 4);
76 if (!sawHeader) {
77 if (name != "IHDR" || size != 13)
78 return failure<std::vector<std::uint8_t>>(DiagnosticCode::ParseError,
79 "PNG IHDR must be first");
80 width = big32(payload, 0);
81 height = big32(payload, 4);
82 const std::uint8_t bitDepth = payload[8];
83 const std::uint8_t colorType = payload[9];
84 if (width == 0 || height == 0 || width != expectedWidth || height != expectedHeight ||
85 bitDepth != 8 || (colorType != 2 && colorType != 6) || payload[10] != 0 ||
86 payload[11] != 0 || payload[12] != 0)
87 return failure<std::vector<std::uint8_t>>(
88 DiagnosticCode::Unsupported,
89 "PNG Cook supports non-interlaced 8-bit RGB/RGBA matching the image definition");
90 channels = colorType == 6 ? 4 : 3;
91 sawHeader = true;
92 } else if (name == "IHDR") {
93 return failure<std::vector<std::uint8_t>>(DiagnosticCode::Conflict,
94 "PNG contains duplicate IHDR");
95 } else if (name == "IDAT") {
96 if (compressed.size() > maximumDecodedBytes ||
97 payload.size() > maximumDecodedBytes - compressed.size())
98 return failure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
99 "PNG compressed data exceeds Cook budget");
100 compressed.insert(compressed.end(), payload.begin(), payload.end());
101 } else if (name == "IEND") {
102 if (size != 0 || cursor + 12 != bytes.size())
103 return failure<std::vector<std::uint8_t>>(DiagnosticCode::ParseError,
104 "PNG IEND or trailing data is invalid");
105 sawEnd = true;
106 } else if ((type[0] & 0x20u) == 0) {
107 return failure<std::vector<std::uint8_t>>(DiagnosticCode::Unsupported,
108 "PNG contains an unsupported critical chunk",
109 std::string(name));
110 }
111 cursor += std::size_t(size) + 12;
112 if (sawEnd) break;
113 }
114 const std::uint64_t rowBytes = std::uint64_t(width) * channels;
115 const std::uint64_t filteredSize = std::uint64_t(height) * (rowBytes + 1);
116 const std::uint64_t rgbaSize = std::uint64_t(width) * height * 4;
117 if (!sawHeader || !sawEnd || compressed.empty() || compressed.size() > (std::numeric_limits<uLong>::max)() ||
118 filteredSize > maximumDecodedBytes ||
119 rgbaSize > maximumDecodedBytes || rgbaSize + 24 > maximumDecodedBytes ||
120 filteredSize > std::numeric_limits<uLongf>::max())
121 return failure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
122 "PNG decoded dimensions exceed Cook budget");
123 std::vector<std::uint8_t> filtered(static_cast<std::size_t>(filteredSize));
124 uLongf outputSize = static_cast<uLongf>(filtered.size());
125 if (uncompress(filtered.data(), &outputSize, compressed.data(), static_cast<uLong>(compressed.size())) != Z_OK ||
126 outputSize != filtered.size())
127 return failure<std::vector<std::uint8_t>>(DiagnosticCode::ParseError,
128 "PNG zlib stream is invalid or has unexpected size");
129 std::vector<std::uint8_t> raw(static_cast<std::size_t>(rowBytes) * height);
130 for (std::uint32_t row = 0; row < height; ++row) {
131 const std::size_t sourceBase = std::size_t(row) * (std::size_t(rowBytes) + 1);
132 const std::uint8_t filter = filtered[sourceBase];
133 if (filter > 4)
134 return failure<std::vector<std::uint8_t>>(DiagnosticCode::ParseError,
135 "PNG row uses an invalid filter");
136 const std::size_t destinationBase = std::size_t(row) * std::size_t(rowBytes);
137 for (std::size_t column = 0; column < rowBytes; ++column) {
138 const std::uint8_t encoded = filtered[sourceBase + 1 + column];
139 const std::uint8_t left = column >= channels ? raw[destinationBase + column - channels] : 0;
140 const std::uint8_t above = row ? raw[destinationBase + column - rowBytes] : 0;
141 const std::uint8_t corner = row && column >= channels
142 ? raw[destinationBase + column - rowBytes - channels]
143 : 0;
144 std::uint8_t predictor = 0;
145 if (filter == 1) predictor = left;
146 else if (filter == 2) predictor = above;
147 else if (filter == 3) predictor = static_cast<std::uint8_t>((int(left) + int(above)) / 2);
148 else if (filter == 4) predictor = paeth(left, above, corner);
149 raw[destinationBase + column] = static_cast<std::uint8_t>(encoded + predictor);
150 }
151 }
152 std::vector<std::uint8_t> rgba;
153 rgba.reserve(static_cast<std::size_t>(rgbaSize));
154 for (std::size_t pixel = 0; pixel < std::size_t(width) * height; ++pixel) {
155 rgba.insert(rgba.end(), raw.begin() + pixel * channels,
156 raw.begin() + pixel * channels + channels);
157 if (channels == 3) rgba.push_back(255);
158 }
159 return Result<std::vector<std::uint8_t>>::success(std::move(rgba));
160}

164Result<CookedCanonicalImage> cookCanonicalImageRgba8(
165 std::span<const std::uint8_t> definition,
166 std::span<const std::uint8_t> encodedSource,
167 std::uint64_t maximumDecodedBytes) {
168 auto parsed = Value::fromJson(std::string_view(reinterpret_cast<const char*>(definition.data()),
169 definition.size()));
170 if (!parsed) return Result<CookedCanonicalImage>::failure(parsed.status());
171 auto* object = parsed.value().getIf<Value::Object>();
172 const Value* schema = object ? field(*object, "schema") : nullptr;
173 const Value* version = object ? field(*object, "schemaVersion") : nullptr;
174 const Value* widthValue = object ? field(*object, "width") : nullptr;
175 const Value* heightValue = object ? field(*object, "height") : nullptr;
176 const Value* encoding = object ? field(*object, "encoding") : nullptr;
177 const Value* color = object ? field(*object, "color") : nullptr;
178 const auto* colorObject = color ? color->getIf<Value::Object>() : nullptr;
179 const Value* transfer = colorObject ? field(*colorObject, "transfer") : nullptr;
180 if (!schema || !schema->isString() || schema->asString() != "eve.image" || !version ||
181 !version->isInt64() || version->asInt() != 2 || !widthValue || !widthValue->isInt64() ||
182 widthValue->asInt() <= 0 || !heightValue || !heightValue->isInt64() ||
183 heightValue->asInt() <= 0 || !encoding || !encoding->isString() || !transfer ||
184 !transfer->isString() || (transfer->asString() != "srgb" && transfer->asString() != "linear"))
185 return failure<CookedCanonicalImage>(DiagnosticCode::ParseError,
186 "eve.image/2 runtime Cook definition is malformed");
187 if (encoding->asString() != "png")
188 return failure<CookedCanonicalImage>(DiagnosticCode::Unsupported,
189 "runtime RGBA8 Cook currently supports PNG source encoding");
190 const auto width = static_cast<std::uint32_t>(widthValue->asInt());
191 const auto height = static_cast<std::uint32_t>(heightValue->asInt());
192 auto rgba = decodePng(encodedSource, width, height, maximumDecodedBytes);
193 if (!rgba) return Result<CookedCanonicalImage>::failure(rgba.status());
194 const bool sourceSrgb = transfer->asString() == "srgb";
195 if (sourceSrgb) {
196 for (std::size_t index = 0; index < rgba.value().size(); index += 4) {
197 for (std::size_t channel = 0; channel < 3; ++channel) {
198 const double encoded = rgba.value()[index + channel] / 255.0;
199 const double linear = encoded <= 0.04045
200 ? encoded / 12.92
201 : std::pow((encoded + 0.055) / 1.055, 2.4);
202 rgba.value()[index + channel] = static_cast<std::uint8_t>(
203 std::clamp(std::lround(linear * 255.0), 0l, 255l));
204 }
205 }
206 }
207 std::vector<std::uint8_t> bulk = {'E', 'V', 'I', 'M', 'G', 0, 1, 0};
208 put32(bulk, width);
209 put32(bulk, height);
210 put32(bulk, 0);
211 put32(bulk, 0);
212 bulk.insert(bulk.end(), rgba.value().begin(), rgba.value().end());
213 (*object)["encoding"] = Value("rgba8");
214 (*object)["sourceEncoding"] = Value("png");
215 (*object)["blob"] = Value("chunk:1");
216 auto runtimeColor = *colorObject;
217 runtimeColor["sourceTransfer"] = Value(transfer->asString());
218 runtimeColor["transfer"] = Value("linear");
219 (*object)["color"] = Value(std::move(runtimeColor));
220 auto runtimeDefinition = parsed.value().toJson();
221 if (!runtimeDefinition) return Result<CookedCanonicalImage>::failure(runtimeDefinition.status());
222 return Result<CookedCanonicalImage>::success(
223 {{runtimeDefinition.value().begin(), runtimeDefinition.value().end()}, std::move(bulk)});
224}
