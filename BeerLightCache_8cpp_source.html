27Result<void> BeerLightCache::rebuild(const FogDensityField& field, const FogProfile& profile,
28 const glm::vec3& lightDir, const glm::vec3& lightColor,
29 float intensity, int samplesPerCell) {
30 if (field.width() < 2) {
31 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Failed,
32 "density field is empty", "rebuild", {},
33 "graphics.fog"));
34 }
35 if (!std::isfinite(intensity) || intensity < 0.f || samplesPerCell < 1 || samplesPerCell > 64) {
36 return Result<void>::failure(Diagnostic::error(
37 DiagnosticCode::InvalidArgument, "intensity/samples out of range", "rebuild", {},
38 "graphics.fog"));
39 }
40 const float len = glm::length(lightDir);
41 if (!(len > 1e-6f) || !std::isfinite(len)) {
42 return Result<void>::failure(Diagnostic::error(
43 DiagnosticCode::InvalidArgument, "light direction must be non-zero and finite",
44 "lightDir", {}, "graphics.fog"));
45 }
46 const glm::vec3 dir = lightDir / len;
47
48 width_ = field.width();
49 height_ = field.height();
50 depth_ = field.depth();
51 transmittance_.assign(static_cast<std::size_t>(width_) * height_ * depth_, 1.f);
52
53 const glm::vec3 cell = field.cellSize();
54 const float stepLen = 0.5f * std::min({cell.x, cell.y, cell.z});
55 const FogWorldBounds& bounds = field.bounds();
56
57 for (int z = 0; z < depth_; ++z) {
58 for (int y = 0; y < height_; ++y) {
59 for (int x = 0; x < width_; ++x) {
60 glm::vec3 p = bounds.minimum + glm::vec3((x + 0.5f) * cell.x, (y + 0.5f) * cell.y,
61 (z + 0.5f) * cell.z);
62 float optical = 0.f;
63 for (int s = 0; s < samplesPerCell; ++s) {
64 // March toward the light (opposite the "direction toward the surface").
65 p -= dir * stepLen;
66 if (!bounds.contains(p)) break;
67 optical += profile.extinctionAt(field.sampleDensity(p)) * stepLen;
68 }
69 const std::size_t idx =
70 (static_cast<std::size_t>(z) * height_ + y) * width_ + static_cast<std::size_t>(x);
71 transmittance_[idx] = std::exp(-optical);
72 }
73 }
74 }
75
76 key_.direction = dir;
77 key_.color = lightColor;
78 key_.intensity = intensity;
79 key_.densityRevision = field.revision();
80 ++key_.version;
81 valid_ = true;
82 return Result<void>::success();
83}

85float BeerLightCache::sampleTransmittance(const FogDensityField& field,
86 const glm::vec3& world) const noexcept {
87 if (!valid_ || transmittance_.empty()) return 1.f;
88 const FogWorldBounds& bounds = field.bounds();
89 const glm::vec3 cell = field.cellSize();
90 const glm::vec3 local = (world - bounds.minimum) / cell;
91 const int x = std::clamp(static_cast<int>(local.x), 0, width_ - 1);
92 const int y = std::clamp(static_cast<int>(local.y), 0, height_ - 1);
93 const int z = std::clamp(static_cast<int>(local.z), 0, depth_ - 1);
94 const std::size_t idx =
95 (static_cast<std::size_t>(z) * height_ + y) * width_ + static_cast<std::size_t>(x);
96 return transmittance_[idx];
97}
