108void decomposeNode(const aiMatrix4x4 &m, float &x, float &y, float &z, float &yaw, float &pitch,
109 float &roll, float &sx, float &sy, float &sz) {
110 aiVector3D pos, scale;
111 aiQuaternion rot;
112 m.Decompose(scale, rot, pos);
113 x = pos.x;
114 y = pos.y;
115 z = pos.z;
116 sx = scale.x;
117 sy = scale.y;
118 sz = scale.z;
119 glm::quat q(rot.w, rot.x, rot.y, rot.z);
120 glm::vec3 e = glm::eulerAngles(q);
121 yaw = e.y;
122 pitch = e.x;
123 roll = e.z;
124}

136 const size_t n = std::char_traits<char>::length(prefix);

188SamplerSpec samplerFor(const aiMaterial *mat, aiTextureType type, bool wantMips) {
189 SamplerSpec s;
190 if (!mat) {
191 s.mips = wantMips;
192 return s;
193 }
194 int modeU = aiTextureMapMode_Wrap;
195 int modeV = aiTextureMapMode_Wrap;
196 mat->Get(AI_MATKEY_MAPPINGMODE_U(type, 0), modeU);
197 mat->Get(AI_MATKEY_MAPPINGMODE_V(type, 0), modeV);
198 s.repeatU = (modeU != aiTextureMapMode_Clamp && modeU != aiTextureMapMode_Mirror);
199 s.repeatV = (modeV != aiTextureMapMode_Clamp && modeV != aiTextureMapMode_Mirror);
200
201 // glTF sampler filter values (AI_MATKEY_GLTF_MAPPINGFILTER_*).
202 int mag = 0, min = 0;
203 mat->Get(AI_MATKEY_GLTF_MAPPINGFILTER_MAG(type, 0), mag);
204 mat->Get(AI_MATKEY_GLTF_MAPPINGFILTER_MIN(type, 0), min);
205 s.filter = (mag == 9728) ? "nearest" : "linear"; // 0 / 9729(linear) -> linear
206 switch (min) {
207 case 9728: // NEAREST
208 case 9729: // LINEAR — no mipmaps requested by the source
209 s.mips = false;
210 s.mipmap = "none";
211 break;
212 case 9986: // NEAREST_MIPMAP_LINEAR
213 case 9987: // LINEAR_MIPMAP_LINEAR
214 s.mips = wantMips;
215 s.mipmap = "linear";
216 break;
217 case 9984: // NEAREST_MIPMAP_NEAREST
218 case 9985: // LINEAR_MIPMAP_NEAREST
219 s.mips = wantMips;
220 s.mipmap = "nearest";
221 break;
222 default: // unspecified -> honor the global toggle
223 s.mips = wantMips;
224 s.mipmap = wantMips ? "linear" : "none";
225 break;
226 }
227 return s;
228}

230graphics::Texture *resolveTexture(graphics::Graphics *gfx, const aiScene *scene,
231 const aiMaterial *mat, aiTextureType type,
232 SceneLoader::TextureCache &cache, bool wantMips,
233 const SceneLoader::CpuImageMap *predecoded = nullptr) {
234 if (!gfx || !scene || !mat) return nullptr;
235 aiString path;
236 if (mat->GetTexture(type, 0, &path) != AI_SUCCESS) return nullptr;
237 const char *p = path.C_Str();
238 if (!p || !p[0]) return nullptr;
239
240 const SamplerSpec s = samplerFor(mat, type, wantMips);
241 auto *const imageModule = eve::image::Image::create();
242
243 const std::string keySuffix =
244 std::string(s.repeatU ? "|1" : "|0") + (s.repeatV ? "1" : "0") + "|" + s.filter + "|" +
245 s.mipmap;
246
247 // Embedded texture ("*0", "*1", ...).
248 if (p[0] == '*') {
249 int idx = std::atoi(p + 1);
250 if (idx < 0 || static_cast<unsigned>(idx) >= scene->mNumTextures) return nullptr;
251 const aiTexture *tex = scene->mTextures[idx];
252 if (!tex || !tex->pcData) return nullptr;
253 const std::string key = "*" + std::to_string(idx) + keySuffix;
254 auto it = cache.find(key);
255 if (it != cache.end()) return it->second;
256
257 graphics::Texture *t = nullptr;
258 if (tex->mHeight == 0) {
259 eve::data::ByteData bytes(tex->pcData, static_cast<size_t>(tex->mWidth));
260 try {
261 eve::image::ImageData *img = imageModule->newImageData(&bytes);
262 t = gfx->newTextureWithSampler(img, s.repeatU, s.repeatV, s.mips, 8.f, s.filter,
263 s.mipmap);
264 delete img;
265 } catch (...) {
266 return nullptr;
267 }
268 } else {
269 const unsigned w = tex->mWidth;
270 const unsigned h = tex->mHeight;
271 std::vector<uint8_t> rgba(size_t(w) * size_t(h) * 4);
272 const aiTexel *src = tex->pcData;
273 for (unsigned i = 0; i < w * h; ++i) {
274 rgba[i * 4 + 0] = src[i].r;
275 rgba[i * 4 + 1] = src[i].g;
276 rgba[i * 4 + 2] = src[i].b;
277 rgba[i * 4 + 3] = src[i].a;
278 }
279 eve::image::ImageData img(int(w), int(h), "RGBA8", rgba.data(), false);
280 t = gfx->newTextureWithSampler(&img, s.repeatU, s.repeatV, s.mips, 8.f, s.filter,
281 s.mipmap);
282 }
283 if (t) cache[key] = t;
284 return t;
285 }
286
287 // External file through the VFS.
288 const std::string key = normPath(p) + keySuffix;
289 auto it = cache.find(key);
290 if (it != cache.end()) return it->second;
291
292 // Off-thread pre-decoded CPU image (async path): upload directly, no disk IO.
293 if (predecoded) {
294 auto pit = predecoded->find(key);
295 if (pit != predecoded->end()) {
296 const SceneLoader::CpuImage &ci = pit->second;
297 eve::image::ImageData img(ci.w, ci.h, "RGBA8",
298 const_cast<uint8_t *>(ci.rgba.data()), false);
299 graphics::Texture *t = gfx->newTextureWithSampler(&img, s.repeatU, s.repeatV, s.mips,
300 8.f, s.filter, s.mipmap);
301 if (t) cache[key] = t;
302 return t;
303 }
304 }
305
306 try {
307 auto *fs = eve::filesystem::Filesystem::create();
308 std::unique_ptr<eve::filesystem::FileData> fd(fs->read(p));
309 if (fd && fd->getSize() > 0) {
310 eve::image::ImageData *img = imageModule->newImageData(fd.get());
311 graphics::Texture *t = gfx->newTextureWithSampler(img, s.repeatU, s.repeatV, s.mips,
312 8.f, s.filter, s.mipmap);
313 delete img;
314 if (t) cache[key] = t;
315 return t;
316 }
317 } catch (...) {
318 }
319 return nullptr;
320}

322// Pre-decode external (non-embedded) texture files off the calling thread so the
323// async loader can skip disk IO + image decode on the render thread.
324void collectCpuImages(const aiScene *scene, const MeshSlotMap &slots, bool wantMips,
325 SceneLoader::CpuImageMap &out) {
326 if (!scene) return;
327 auto *const imageModule = eve::image::Image::create();
328 const aiTextureType kTypes[4] = {aiTextureType_BASE_COLOR, aiTextureType_DIFFUSE,
329 aiTextureType_NORMALS, aiTextureType_HEIGHT};
330 for (const auto &kv : slots) {
331 for (const MeshSlot &slot : kv.second) {
332 if (!slot.scene || slot.materialIndex >= scene->mNumMaterials) continue;
333 const aiMaterial *mat = scene->mMaterials[slot.materialIndex];
334 if (!mat) continue;
335 for (aiTextureType type : kTypes) {
336 aiString p;
337 if (mat->GetTexture(type, 0, &p) != AI_SUCCESS) continue;
338 const char *c = p.C_Str();
339 if (!c || !c[0] || c[0] == '*') continue; // embedded handled on main thread
340 const SamplerSpec s = samplerFor(mat, type, wantMips);
341 const std::string key = normPath(c) + std::string(s.repeatU ? "|1" : "|0") +
342 (s.repeatV ? "1" : "0") + "|" + s.filter + "|" + s.mipmap;
343 if (out.count(key)) continue;
344 try {
345 auto *fs = eve::filesystem::Filesystem::create();
346 std::unique_ptr<eve::filesystem::FileData> fd(fs->read(c));
347 if (!fd || fd->getSize() == 0) continue;
348 eve::image::ImageData *img = imageModule->newImageData(fd.get());
349 if (!img) continue;
350 if (img->getFormat() == "RGBA8") {
351 SceneLoader::CpuImage ci;
352 ci.w = img->getWidth();
353 ci.h = img->getHeight();
354 const size_t n = size_t(ci.w) * size_t(ci.h) * 4;
355 ci.rgba.assign(reinterpret_cast<const uint8_t *>(img->getData()),
356 reinterpret_cast<const uint8_t *>(img->getData()) + n);
357 out[key] = std::move(ci);
358 }
359 delete img;
360 } catch (...) {
361 }
362 }
363 }
364 }

369graphics::Renderable3D *makeRenderable(graphics::Graphics *gfx, const MeshSlot &slot,
370 SceneLoader::TextureCache &textures, bool mipmaps) {
371 if (!gfx || !slot.mesh) return nullptr;
372 graphics::Mesh *mesh = gfx->newMeshFromAssimp(*slot.mesh); // local-space (hierarchy transform)
373 if (!mesh) return nullptr;
374 auto *r = graphics::Renderable3D::create();
375 r->meshRenderer()->visible = true;
376 r->setMesh(mesh);
377
378 const aiScene *scene = slot.scene;
379 const aiMaterial *mat = nullptr;
380 if (scene && scene->mMaterials && slot.materialIndex < scene->mNumMaterials)
381 mat = scene->mMaterials[slot.materialIndex];
382
383 aiColor3D base(1.f, 1.f, 1.f);
384 if (mat) {
385 if (mat->Get(AI_MATKEY_BASE_COLOR, base) != AI_SUCCESS)
386 mat->Get(AI_MATKEY_COLOR_DIFFUSE, base);
387 float metallic = 0.f;
388 float roughness = 0.45f;
389 mat->Get(AI_MATKEY_METALLIC_FACTOR, metallic);
390 mat->Get(AI_MATKEY_ROUGHNESS_FACTOR, roughness);
391 r->setMetallic(metallic);
392 r->setRoughness(roughness);
393 }
394
395 graphics::Texture *albedo =
396 resolveTexture(gfx, scene, mat, aiTextureType_BASE_COLOR, textures, mipmaps);
397 if (!albedo) albedo = resolveTexture(gfx, scene, mat, aiTextureType_DIFFUSE, textures, mipmaps);
398 graphics::Texture *normal =
399 resolveTexture(gfx, scene, mat, aiTextureType_NORMALS, textures, mipmaps);
400 graphics::Texture *height =
401 resolveTexture(gfx, scene, mat, aiTextureType_HEIGHT, textures, mipmaps);
402
403 r->setTint(base.r, base.g, base.b, 1.f);
404 if (albedo) r->setTexture(albedo);
405 if (normal) r->setNormalTexture(normal);
406 if (height) r->setHeightTexture(height);
407 return r;
408}

431 const aiMesh *mesh = slot.mesh;

457void unionChildBounds(scene::SceneHost::Tree &tree, int nodeIndex) {
458 scene::SceneNode &n = tree.nodes[size_t(nodeIndex)];
459 for (int c = n.firstChild; c >= 0; c = tree.nodes[size_t(c)].nextSibling) {
460 unionChildBounds(tree, c);
461 }
462
463 bool any = false;
464 float mn[3] = {std::numeric_limits<float>::max(),
465 std::numeric_limits<float>::max(),
466 std::numeric_limits<float>::max()};
467 float mx[3] = {std::numeric_limits<float>::lowest(),
468 std::numeric_limits<float>::lowest(),
469 std::numeric_limits<float>::lowest()};
470 for (int c = n.firstChild; c >= 0; c = tree.nodes[size_t(c)].nextSibling) {
471 scene::SceneNode &ch = tree.nodes[size_t(c)];
472 if (!ch.hasBounds) continue;
473 any = true;
474 const glm::mat4 lm = nodeLocalMatrix(ch);
475 for (int i = 0; i < 8; ++i) {
476 const glm::vec3 p((i & 1) ? ch.bmaxX : ch.bminX,
477 (i & 2) ? ch.bmaxY : ch.bminY,
478 (i & 4) ? ch.bmaxZ : ch.bminZ);
479 const glm::vec4 w = lm * glm::vec4(p, 1.f);
480 for (int k = 0; k < 3; ++k) {
481 mn[k] = std::min(mn[k], w[k]);
482 mx[k] = std::max(mx[k], w[k]);
483 }
484 }
485 }
486 if (!any) return;
487 if (!n.hasBounds) {
488 n.bminX = mn[0];
489 n.bminY = mn[1];
490 n.bminZ = mn[2];
491 n.bmaxX = mx[0];
492 n.bmaxY = mx[1];
493 n.bmaxZ = mx[2];
494 n.hasBounds = true;
495 } else {
496 n.bminX = std::min(n.bminX, mn[0]);
497 n.bminY = std::min(n.bminY, mn[1]);
498 n.bminZ = std::min(n.bminZ, mn[2]);
499 n.bmaxX = std::max(n.bmaxX, mx[0]);
500 n.bmaxY = std::max(n.bmaxY, mx[1]);
501 n.bmaxZ = std::max(n.bmaxZ, mx[2]);
502 }
503}

507void buildNodeRecursive(const aiScene *scene, const aiNode *node, scene::NodeDesc &out,
508 MeshSlotMap *slots, std::unordered_map<std::string, int> &counts) {
509 const std::string rawName = (node && node->mName.length) ? node->mName.C_Str() : "<root>";
510 const std::string id = uniqueId(rawName, counts);
511
512 out.id = id;
513 out.key = id;
514 out.name = rawName;
515 out.space = "3d";
516 out.visible = true;
517 if (startsWithInsensitive(rawName, "SOCKET_")) out.tags.push_back("model-socket");
518 if (startsWithInsensitive(rawName, "UCX_") || startsWithInsensitive(rawName, "UBX_") ||
519 startsWithInsensitive(rawName, "USP_") || startsWithInsensitive(rawName, "UCP_")) {
520 out.tags.push_back("collision");
521 out.visible = false;
522 }
523 const int lod = lodFromName(rawName);
524 if (lod >= 0) {
525 out.tags.push_back("lod");
526 out.tags.push_back("lod:" + std::to_string(lod));
527 out.visible = lod == 0;
528 }
529 decomposeNode(node->mTransformation, out.x, out.y, out.z, out.yaw, out.pitch, out.roll,
530 out.sx, out.sy, out.sz);
531
532 for (unsigned i = 0; node && i < node->mNumMeshes; ++i) {
533 const unsigned mi = node->mMeshes[i];
534 if (mi >= scene->mNumMeshes) continue;
535 const aiMesh *mesh = scene->mMeshes[mi];
536 if (!mesh || mesh->mNumFaces == 0) continue;
537
538 const std::string cid = id + "_mesh" + std::to_string(i);
539 scene::NodeDesc child;
540 child.id = cid;
541 child.key = cid;
542 child.name = std::string("mesh") + std::to_string(i);
543 child.space = "3d";
544 child.visible = out.visible;
545 child.tags = out.tags;
546 out.children.push_back(std::move(child));
547
548 if (slots) {
549 MeshSlot s;
550 s.scene = scene;
551 s.mesh = mesh;
552 s.materialIndex = mesh->mMaterialIndex;
553 (*slots)[cid].push_back(std::move(s));
554 }
555 }
556
557 for (unsigned c = 0; node && c < node->mNumChildren; ++c) {
558 scene::NodeDesc child;
559 buildNodeRecursive(scene, node->mChildren[c], child, slots, counts);
560 out.children.push_back(std::move(child));
561 }
562}
563
564bool propsChanged(const scene::SceneNode &n, const scene::NodeDesc &d) {
565 if (n.visible != d.visible) return true;
566 return !(approx(n.x, d.x) && approx(n.y, d.y) && approx(n.z, d.z) && approx(n.yaw, d.yaw) &&
567 approx(n.pitch, d.pitch) && approx(n.roll, d.roll) && approx(n.sx, d.sx) &&
568 approx(n.sy, d.sy) && approx(n.sz, d.sz));
569}

571// Walk the new tree in DFS order, emitting Add / Move / Modify entries.
572void walkNew(const scene::NodeDesc &d, const std::string &parentId,
573 const std::unordered_map<std::string, const scene::SceneNode *> &oldNodes,
574 const std::unordered_map<std::string, std::string> &oldParent, SceneDiff &out) {
575 auto it = oldNodes.find(d.id);
576 if (it == oldNodes.end()) {
577 out.entries.push_back({SceneDiffEntry::Action::Add, d.id, parentId});
578 ++out.added;
579 } else {
580 auto pit = oldParent.find(d.id);
581 const std::string op = (pit != oldParent.end()) ? pit->second : "";
582 if (op != parentId) {
583 out.entries.push_back({SceneDiffEntry::Action::Move, d.id, parentId});
584 ++out.moved;
585 } else if (propsChanged(*it->second, d)) {
586 out.entries.push_back({SceneDiffEntry::Action::Modify, d.id, ""});
587 ++out.modified;
588 }
589 }
590 for (const auto &child : d.children) walkNew(child, d.id, oldNodes, oldParent, out);
591}

667SceneDiff SceneLoader::diffTree(scene::SceneHost *host, const scene::NodeDesc &newRoot) {
668 SceneDiff out;
669 std::unordered_map<std::string, const scene::SceneNode *> oldNodes;
670 std::unordered_map<std::string, std::string> oldParent;
671 if (host) {
672 auto t = host->tree();
673 for (size_t i = 0; i < t->nodes.size(); ++i) {
674 const scene::SceneNode &n = t->nodes[i];
675 oldNodes[n.id] = &n;
676 oldParent[n.id] = (n.parent >= 0) ? t->nodes[size_t(n.parent)].id : "";
677 }
678 }
679
680 // Collect the set of ids present in the new tree.
681 std::unordered_set<std::string> newIds;
682 std::function<void(const scene::NodeDesc &)> collect = [&](const scene::NodeDesc &d) {
683 newIds.insert(d.id);
684 for (const auto &c : d.children) collect(c);
685 };
686 collect(newRoot);
687
688 // Removed: present in old, absent in new.
689 for (const auto &kv : oldNodes) {
690 if (!newIds.count(kv.first)) {
691 out.entries.push_back({SceneDiffEntry::Action::Remove, kv.first, ""});
692 ++out.removed;
693 }
694 }
695
696 // Add / Move / Modify: walk the new tree (DFS so parents precede children).
697 walkNew(newRoot, "", oldNodes, oldParent, out);
698 return out;
699}

701bool SceneLoader::applyTreeDiff(scene::SceneHost *host, const scene::NodeDesc &newRoot,
702 const SceneDiff &diff, graphics::Graphics *gfx,
703 const MeshSlotMap *slots) {
704 if (!host || diff.empty()) return false;
705
706 // Snapshot the mounted tree by id so kept GameObjects can keep their linked
707 // Renderable3D (no re-upload / no object rebuild) across the update.
708 std::unordered_map<std::string, const scene::SceneNode *> oldNodes;
709 {
710 auto t = host->tree();
711 for (const auto &n : t->nodes) oldNodes[n.id] = &n;
712 }
713
714 // Collect the ids present in the new tree.
715 std::unordered_set<std::string> newIds;
716 std::function<void(const scene::NodeDesc &)> collect = [&](const scene::NodeDesc &d) {
717 newIds.insert(d.id);
718 for (const auto &c : d.children) collect(c);
719 };
720 collect(newRoot);
721
722 // Destroy Renderable3D of removed GameObjects.
723 for (const auto &kv : oldNodes) {
724 if (!newIds.count(kv.first)) {
725 if (const auto *l = host->findLink(kv.second, scene::findLinkKind("renderable3d"))) {
726 destroyRenderable(static_cast<graphics::Renderable3D *>(l->target));
727 }
728 }
729 }
730
731 // Rebuild the arena from the new tree (DFS). Kept nodes copy their link from
732 // the old node (identity preserved); added mesh nodes get a fresh Renderable3D.
733 std::vector<scene::SceneNode> nodes;
734 nodes.reserve(newIds.size());
735 std::function<int(const scene::NodeDesc &, int)> build = [&](const scene::NodeDesc &d,
736 int parentIndex) -> int {
737 const int idx = int(nodes.size());
738 scene::SceneNode n;
739 n.id = d.id;
740 n.key = d.key.empty() ? d.id : d.key;
741 n.name = d.name.empty() ? d.id : d.name;
742 n.space = d.space.empty() ? "3d" : d.space;
743 n.visible = d.visible;
744 n.x = d.x;
745 n.y = d.y;
746 n.z = d.z;
747 n.yaw = d.yaw;
748 n.pitch = d.pitch;
749 n.roll = d.roll;
750 n.sx = d.sx;
751 n.sy = d.sy;
752 n.sz = d.sz;
753 n.localDirty = true;
754 n.world = glm::mat4(1.f);
755 n.firstChild = -1;
756 n.nextSibling = -1;
757 n.parent = parentIndex;
758
759 auto oldIt = oldNodes.find(d.id);
760 if (oldIt != oldNodes.end()) {
761 n.links = oldIt->second->links;
762 n.objectId = oldIt->second->objectId;
763 n.bminX = oldIt->second->bminX;
764 n.bminY = oldIt->second->bminY;
765 n.bminZ = oldIt->second->bminZ;
766 n.bmaxX = oldIt->second->bmaxX;
767 n.bmaxY = oldIt->second->bmaxY;
768 n.bmaxZ = oldIt->second->bmaxZ;
769 n.hasBounds = oldIt->second->hasBounds;
770 }
771
772 nodes.push_back(std::move(n));
773
774 int prevChild = -1;
775 int firstChild = -1;
776 for (const auto &c : d.children) {
777 const int ci = build(c, idx);
778 if (firstChild < 0) firstChild = ci;
779 if (prevChild >= 0) nodes[size_t(prevChild)].nextSibling = ci;
780 prevChild = ci;
781 }
782 nodes[size_t(idx)].firstChild = firstChild;
783 return idx;
784 };
785 const int root = build(newRoot, -1);
786
787 // Fresh Renderable3D for newly added mesh GameObjects (only changed ones).
788 if (gfx && slots) {
789 TextureCache unused;
790 for (auto &n : nodes) {
791 if (!n.links.empty()) continue;
792 if (std::find(n.tags.begin(), n.tags.end(), "collision") != n.tags.end()) continue;
793 auto sit = slots->find(n.id);
794 if (sit == slots->end() || sit->second.empty()) continue;
795 graphics::Renderable3D *r = nullptr;
796 try {
797 r = makeRenderable(gfx, sit->second[0], unused, true);
798 } catch (...) {
799 r = nullptr;
800 }
801 if (r) {
802 n.links.push_back(scene::SceneLink{scene::findLinkKind("renderable3d"), r, 0});
803 }
804 }
805 }
806
807 host->tree()->nodes = std::move(nodes);
808 host->tree()->root = root;
809 host->invalidateIndex();
810 host->markTransformDirty();
811 if (slots) fillSceneBounds(host, *slots);
812 return true;
813}

829void SceneLoader::linkMeshNodes(scene::SceneHost *host, const MeshSlotMap &slots,
830 graphics::Graphics *gfx, const LoadOptions &options,
831 TextureCache &textures, MeshCache &shared,
832 const CpuImageMap *predecoded) {
833 if (!gfx) return;
834 for (const auto &kv : slots) {
835 scene::SceneNode *n = borrowResult(host->findById(kv.first));
836 if (!n || kv.second.empty()) continue;
837 if (std::find(n->tags.begin(), n->tags.end(), "collision") != n->tags.end()) continue;
838 if (host->findLink(n, scene::findLinkKind("renderable3d"))) continue;
839 const MeshSlot &slot = kv.second[0];
840 graphics::Mesh *mesh = nullptr;
841 auto it = shared.find(slot.mesh);
842 if (options.sharedMeshes && it != shared.end()) {
843 mesh = it->second;
844 } else {
845 try {
846 mesh = gfx->newMeshFromAssimp(*slot.mesh);
847 } catch (...) {
848 mesh = nullptr;
849 }
850 if (mesh && options.sharedMeshes) shared[slot.mesh] = mesh;
851 }
852 if (!mesh) continue;
853 auto *r = graphics::Renderable3D::create();
854 r->meshRenderer()->visible = true;
855 r->setMesh(mesh);
856
857 const aiScene *scene = slot.scene;
858 const aiMaterial *mat = nullptr;
859 if (scene && scene->mMaterials && slot.materialIndex < scene->mNumMaterials)
860 mat = scene->mMaterials[slot.materialIndex];
861 aiColor3D base(1.f, 1.f, 1.f);
862 if (mat) {
863 if (mat->Get(AI_MATKEY_BASE_COLOR, base) != AI_SUCCESS)
864 mat->Get(AI_MATKEY_COLOR_DIFFUSE, base);
865 float metallic = 0.f;
866 float roughness = 0.45f;
867 mat->Get(AI_MATKEY_METALLIC_FACTOR, metallic);
868 mat->Get(AI_MATKEY_ROUGHNESS_FACTOR, roughness);
869 r->setMetallic(metallic);
870 r->setRoughness(roughness);
871 }
872 graphics::Texture *albedo = resolveTexture(gfx, scene, mat, aiTextureType_BASE_COLOR,
873 textures, options.mipmaps, predecoded);
874 if (!albedo)
875 albedo = resolveTexture(gfx, scene, mat, aiTextureType_DIFFUSE, textures,
876 options.mipmaps, predecoded);
877 graphics::Texture *normal = resolveTexture(gfx, scene, mat, aiTextureType_NORMALS,
878 textures, options.mipmaps, predecoded);
879 graphics::Texture *height = resolveTexture(gfx, scene, mat, aiTextureType_HEIGHT,
880 textures, options.mipmaps, predecoded);
881 r->setTint(base.r, base.g, base.b, 1.f);
882 if (albedo) r->setTexture(albedo);
883 if (normal) r->setNormalTexture(normal);
884 if (height) r->setHeightTexture(height);
885 n->links.push_back(scene::SceneLink{scene::findLinkKind("renderable3d"), r, 0});
886 }
887}

889bool SceneLoader::decode(const std::string &path, const LoadOptions &options, DecodedScene *out) {
890 const std::string key = normPath(path);
891 // The unified resource cache may hold a decode from an earlier load; a
892 // reload/diff must see the file as it is now, so refresh the cached entry
893 // before asking for it (no-op when nothing is cached yet).
894 eve::ResourceManager::getInstance().reload(path).ignore("scene reload");
895
896 auto *mod3d = ModuleManager::getInstance<model3d::Model3D>("Model3D");
897 if (!mod3d) mod3d = model3d::Model3D::create();
898 model3d::ModelData *md = nullptr;
899 try {
900 md = mod3d->newModelDataFromFile(path, toModelOptions(options));
901 } catch (const std::exception &e) {
902 std::lock_guard<std::mutex> lock(pendingMu_);
903 lastErrors_[key] = e.what();
904 return false;
905 } catch (...) {
906 std::lock_guard<std::mutex> lock(pendingMu_);
907 lastErrors_[key] = "unknown model decode error";
908 return false;
909 }
910 if (!md) {
911 std::lock_guard<std::mutex> lock(pendingMu_);
912 lastErrors_[key] = "model decoder returned no scene";
913 return false;
914 }
915
916 std::vector<std::string> importWarnings;
917 for (int i = 0; i < md->getMeshCount(); ++i) {
918 if (!md->hasNormals(i))
919 importWarnings.push_back("mesh " + std::to_string(i) + " has no normals");
920 if (!md->hasTexCoords(i))
921 importWarnings.push_back("mesh " + std::to_string(i) + " has no UV0");
922 }
923 {
924 std::lock_guard<std::mutex> lock(pendingMu_);
925 lastErrors_.erase(key);
926 warnings_[key] = std::move(importWarnings);
927 }
928
929 out->path = normPath(path);
930 out->md = md;
931 out->options = options;
932 out->root = std::make_unique<scene::NodeDesc>(buildNodeDesc(md->getScene(), &out->slots));
933 collectCpuImages(md->getScene(), out->slots, options.mipmaps, out->cpuImages);
934 return true;
935}

937scene::SceneHost *SceneLoader::mount(DecodedScene &d) {
938 scene::SceneHost *host = borrowResult(scene::SceneHost::createHost(d.path));
939 if (!host) return nullptr;
940 host->setTree(std::move(*d.root));
941
942 graphics::Graphics *gfx = currentGraphics();
943 if (!d.slots.empty()) fillSceneBounds(host, d.slots);
944
945 if (gfx) {
946 MeshCache shared;
947 linkMeshNodes(host, d.slots, gfx, d.options, textures_, shared, &d.cpuImages);
948 if (d.options.importLights) importLights(d.md->getScene(), d.lights);
949 if (d.options.importCameras) importCameras(d.md->getScene(), d.cameras);
950 importAnimations(d.md->getScene(), d.options, &d.skeleton, d.clips);
951 }
952 fillSceneBounds(host, d.slots);
953 scene::TransformSystem::updateHost(host);
954 scenes_[d.path] = Loaded{d.path, host, gfx, d.options, std::move(d.lights),
955 std::move(d.cameras), d.skeleton, std::move(d.clips)};
956 return host;
957}

997eve::Result<bool> SceneLoader::reload(const std::string &path, SceneDiff *out, const LoadOptions &options) {
998 const std::string key = normPath(path);
999 auto it = scenes_.find(key);
1000 if (it == scenes_.end()) {
1001 if (!load(path, options)) {
1002 const std::string message = decodeErrorFor(path);
1003 return eve::Result<bool>::failure(eve::Diagnostic::error(
1004 eve::DiagnosticCode::Failed, message.empty() ? "scene reload failed to load the scene" : message, key,
1005 {}, "sceneloader.reload"));
1006 }
1007 if (out) *out = SceneDiff{};
1008 return eve::Result<bool>::success(true, eve::Status::success(eve::StatusCode::Applied));
1009 }
1010 Loaded &ld = it->second;
1011
1012 DecodedScene d;
1013 if (!decode(path, options, &d)) {
1014 const std::string message = decodeErrorFor(path);
1015 return eve::Result<bool>::failure(eve::Diagnostic::error(
1016 eve::DiagnosticCode::Failed, message.empty() ? "scene reload failed to decode the scene" : message, key, {},
1017 "sceneloader.reload"));
1018 }
1019
1020 SceneDiff diff = diffTree(ld.host, *d.root);
1021 if (out) *out = diff;
1022 if (!diff.empty()) {
1023 applyTreeDiff(ld.host, *d.root, diff, nullptr, nullptr);
1024 MeshCache shared;
1025 linkMeshNodes(ld.host, d.slots, ld.gfx, options, textures_, shared, &d.cpuImages);
1026 scene::TransformSystem::updateHost(ld.host);
1027 }
1028 if (diff.empty()) return eve::Result<bool>::success(false, eve::Status::success(eve::StatusCode::NoOp));
1029 return eve::Result<bool>::success(true, eve::Status::success(eve::StatusCode::Applied));
1030}

1032SceneDiff SceneLoader::diff(const std::string &path) {
1033 const std::string key = normPath(path);
1034 eve::ResourceManager::getInstance().reload(path).ignore("scene diff reload"); // fresh decode for diffing
1035 auto *mod3d = ModuleManager::getInstance<model3d::Model3D>("Model3D");
1036 if (!mod3d) mod3d = model3d::Model3D::create();
1037 model3d::ModelData *md = nullptr;
1038 try {
1039 md = mod3d->newModelDataFromFile(path);
1040 } catch (...) {
1041 return SceneDiff{};
1042 }
1043 if (!md) return SceneDiff{};
1044 MeshSlotMap slots;
1045 scene::NodeDesc newRoot = buildNodeDesc(md->getScene(), &slots);
1046 SceneDiff d;
1047 auto it = scenes_.find(key);
1048 if (it != scenes_.end() && it->second.host) {
1049 d = diffTree(it->second.host, newRoot);
1050 } else {
1051 // Nothing mounted: every node in the new tree is an add.
1052 std::function<void(const scene::NodeDesc &, const std::string &)> walk =
1053 [&](const scene::NodeDesc &n, const std::string &parent) {
1054 d.entries.push_back({SceneDiffEntry::Action::Add, n.id, parent});
1055 ++d.added;
1056 for (const auto &c : n.children) walk(c, n.id);
1057 };
1058 walk(newRoot, "");
1059 }
1060 return d;
1061}

1063eve::Result<SceneInspection> SceneLoader::inspect(const std::string &path,
1064 const LoadOptions &options) {
1065 DecodedScene decoded;
1066 if (!decode(path, options, &decoded)) {
1067 const std::string message = decodeErrorFor(path);
1068 return eve::Result<SceneInspection>::failure(eve::Diagnostic::error(
1069 eve::DiagnosticCode::Failed,
1070 message.empty() ? "scene inspection failed to decode the source" : message,
1071 normPath(path), {}, "sceneloader.inspect"));
1072 }
1073 SceneInspection result;
1074 const auto mounted = scenes_.find(normPath(path));
1075 if (mounted != scenes_.end() && mounted->second.host)
1076 result.diff = diffTree(mounted->second.host, *decoded.root);
1077 else {
1078 std::function<void(const scene::NodeDesc&, const std::string&)> add =
1079 [&](const scene::NodeDesc& node, const std::string& parent) {
1080 result.diff.entries.push_back({SceneDiffEntry::Action::Add, node.id, parent});
1081 ++result.diff.added;
1082 for (const auto& child : node.children) add(child, node.id);
1083 };
1084 add(*decoded.root, "");
1085 }
1086 std::function<void(const scene::NodeDesc&)> collect = [&](const scene::NodeDesc& node) {
1087 ++result.nodeCount;
1088 if (decoded.slots.find(node.id) != decoded.slots.end()) ++result.meshNodeCount;
1089 if (std::find(node.tags.begin(), node.tags.end(), "model-socket") != node.tags.end())
1090 result.sockets.push_back(node.name);
1091 if (std::find(node.tags.begin(), node.tags.end(), "collision") != node.tags.end())
1092 result.collisions.push_back(node.name);
1093 for (const auto& child : node.children) collect(child);
1094 };
1095 collect(*decoded.root);
1096 const int count = warningCount(path);
1097 for (int i = 0; i < count; ++i) result.warnings.push_back(warning(path, i));
1098 return eve::Result<SceneInspection>::success(std::move(result));
1099}

1113void SceneLoader::unload(const std::string &path) {

1115 if (it == scenes_.end()) return;
1116 if (it->second.host) {
