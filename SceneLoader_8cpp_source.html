96void decomposeNode(const aiMatrix4x4 &m, float &x, float &y, float &z, float &yaw, float &pitch,
97 float &roll, float &sx, float &sy, float &sz) {
98 aiVector3D pos, scale;
99 aiQuaternion rot;
100 m.Decompose(scale, rot, pos);
101 x = pos.x;
102 y = pos.y;
103 z = pos.z;
104 sx = scale.x;
105 sy = scale.y;
106 sz = scale.z;
107 glm::quat q(rot.w, rot.x, rot.y, rot.z);
108 glm::vec3 e = glm::eulerAngles(q);
109 yaw = e.y;
110 pitch = e.x;
111 roll = e.z;
112}

128 bool repeatV = true;

131 std::string mipmap = "linear"; // "none" | "linear" | "nearest"

134SamplerSpec samplerFor(const aiMaterial *mat, aiTextureType type, bool wantMips) {
135 SamplerSpec s;
136 if (!mat) {
137 s.mips = wantMips;
138 return s;
139 }
140 int modeU = aiTextureMapMode_Wrap;
141 int modeV = aiTextureMapMode_Wrap;
142 mat->Get(AI_MATKEY_MAPPINGMODE_U(type, 0), modeU);
143 mat->Get(AI_MATKEY_MAPPINGMODE_V(type, 0), modeV);
144 s.repeatU = (modeU != aiTextureMapMode_Clamp && modeU != aiTextureMapMode_Mirror);
145 s.repeatV = (modeV != aiTextureMapMode_Clamp && modeV != aiTextureMapMode_Mirror);
146
147 // glTF sampler filter values (AI_MATKEY_GLTF_MAPPINGFILTER_*).
148 int mag = 0, min = 0;
149 mat->Get(AI_MATKEY_GLTF_MAPPINGFILTER_MAG(type, 0), mag);
150 mat->Get(AI_MATKEY_GLTF_MAPPINGFILTER_MIN(type, 0), min);
151 s.filter = (mag == 9728) ? "nearest" : "linear"; // 0 / 9729(linear) -> linear
152 switch (min) {
153 case 9728: // NEAREST
154 case 9729: // LINEAR — no mipmaps requested by the source
155 s.mips = false;
156 s.mipmap = "none";
157 break;
158 case 9986: // NEAREST_MIPMAP_LINEAR
159 case 9987: // LINEAR_MIPMAP_LINEAR
160 s.mips = wantMips;
161 s.mipmap = "linear";
162 break;
163 case 9984: // NEAREST_MIPMAP_NEAREST
164 case 9985: // LINEAR_MIPMAP_NEAREST
165 s.mips = wantMips;
166 s.mipmap = "nearest";
167 break;
168 default: // unspecified -> honor the global toggle
169 s.mips = wantMips;
170 s.mipmap = wantMips ? "linear" : "none";
171 break;
172 }
173 return s;
174}

176graphics::Texture *resolveTexture(graphics::Graphics *gfx, const aiScene *scene,
177 const aiMaterial *mat, aiTextureType type,
178 SceneLoader::TextureCache &cache, bool wantMips,
179 const SceneLoader::CpuImageMap *predecoded = nullptr) {
180 if (!gfx || !scene || !mat) return nullptr;
181 aiString path;
182 if (mat->GetTexture(type, 0, &path) != AI_SUCCESS) return nullptr;
183 const char *p = path.C_Str();
184 if (!p || !p[0]) return nullptr;
185
186 const SamplerSpec s = samplerFor(mat, type, wantMips);
187 eve::image::Image::create();
188
189 const std::string keySuffix =
190 std::string(s.repeatU ? "|1" : "|0") + (s.repeatV ? "1" : "0") + "|" + s.filter + "|" +
191 s.mipmap;
192
193 // Embedded texture ("*0", "*1", ...).
194 if (p[0] == '*') {
195 int idx = std::atoi(p + 1);
196 if (idx < 0 || static_cast<unsigned>(idx) >= scene->mNumTextures) return nullptr;
197 const aiTexture *tex = scene->mTextures[idx];
198 if (!tex || !tex->pcData) return nullptr;
199 const std::string key = "*" + std::to_string(idx) + keySuffix;
200 auto it = cache.find(key);
201 if (it != cache.end()) return it->second;
202
203 graphics::Texture *t = nullptr;
204 if (tex->mHeight == 0) {
205 eve::data::ByteData bytes(tex->pcData, static_cast<size_t>(tex->mWidth));
206 try {
207 eve::image::ImageData *img = eve::image::Image::create()->newImageData(&bytes);
208 t = gfx->newTextureWithSampler(img, s.repeatU, s.repeatV, s.mips, 8.f, s.filter,
209 s.mipmap);
210 delete img;
211 } catch (...) {
212 return nullptr;
213 }
214 } else {
215 const unsigned w = tex->mWidth;
216 const unsigned h = tex->mHeight;
217 std::vector<uint8_t> rgba(size_t(w) * size_t(h) * 4);
218 const aiTexel *src = tex->pcData;
219 for (unsigned i = 0; i < w * h; ++i) {
220 rgba[i * 4 + 0] = src[i].r;
221 rgba[i * 4 + 1] = src[i].g;
222 rgba[i * 4 + 2] = src[i].b;
223 rgba[i * 4 + 3] = src[i].a;
224 }
225 eve::image::ImageData img(int(w), int(h), "RGBA8", rgba.data(), false);
226 t = gfx->newTextureWithSampler(&img, s.repeatU, s.repeatV, s.mips, 8.f, s.filter,
227 s.mipmap);
228 }
229 if (t) cache[key] = t;
230 return t;
231 }
232
233 // External file through the VFS.
234 const std::string key = normPath(p) + keySuffix;
235 auto it = cache.find(key);
236 if (it != cache.end()) return it->second;
237
238 // Off-thread pre-decoded CPU image (async path): upload directly, no disk IO.
239 if (predecoded) {
240 auto pit = predecoded->find(key);
241 if (pit != predecoded->end()) {
242 const SceneLoader::CpuImage &ci = pit->second;
243 eve::image::ImageData img(ci.w, ci.h, "RGBA8",
244 const_cast<uint8_t *>(ci.rgba.data()), false);
245 graphics::Texture *t = gfx->newTextureWithSampler(&img, s.repeatU, s.repeatV, s.mips,
246 8.f, s.filter, s.mipmap);
247 if (t) cache[key] = t;
248 return t;
249 }
250 }
251
252 try {
253 auto *fs = eve::filesystem::Filesystem::create();
254 std::unique_ptr<eve::filesystem::FileData> fd(fs->read(p));
255 if (fd && fd->getSize() > 0) {
256 eve::image::ImageData *img = eve::image::Image::create()->newImageData(fd.get());
257 graphics::Texture *t = gfx->newTextureWithSampler(img, s.repeatU, s.repeatV, s.mips,
258 8.f, s.filter, s.mipmap);
259 delete img;
260 if (t) cache[key] = t;
261 return t;
262 }
263 } catch (...) {
264 }
265 return nullptr;
266}

268// Pre-decode external (non-embedded) texture files off the calling thread so the
269// async loader can skip disk IO + image decode on the render thread.
270void collectCpuImages(const aiScene *scene, const MeshSlotMap &slots, bool wantMips,
271 SceneLoader::CpuImageMap &out) {
272 if (!scene) return;
273 eve::image::Image::create();
274 const aiTextureType kTypes[4] = {aiTextureType_BASE_COLOR, aiTextureType_DIFFUSE,
275 aiTextureType_NORMALS, aiTextureType_HEIGHT};
276 for (const auto &kv : slots) {
277 for (const MeshSlot &slot : kv.second) {
278 if (!slot.scene || slot.materialIndex >= scene->mNumMaterials) continue;
279 const aiMaterial *mat = scene->mMaterials[slot.materialIndex];
280 if (!mat) continue;
281 for (aiTextureType type : kTypes) {
282 aiString p;
283 if (mat->GetTexture(type, 0, &p) != AI_SUCCESS) continue;
284 const char *c = p.C_Str();
285 if (!c || !c[0] || c[0] == '*') continue; // embedded handled on main thread
286 const SamplerSpec s = samplerFor(mat, type, wantMips);
287 const std::string key = normPath(c) + std::string(s.repeatU ? "|1" : "|0") +
288 (s.repeatV ? "1" : "0") + "|" + s.filter + "|" + s.mipmap;
289 if (out.count(key)) continue;
290 try {
291 auto *fs = eve::filesystem::Filesystem::create();
292 std::unique_ptr<eve::filesystem::FileData> fd(fs->read(c));
293 if (!fd || fd->getSize() == 0) continue;
294 eve::image::ImageData *img =
295 eve::image::Image::create()->newImageData(fd.get());
296 if (!img) continue;
297 if (img->getFormat() == "RGBA8") {
298 SceneLoader::CpuImage ci;
299 ci.w = img->getWidth();
300 ci.h = img->getHeight();
301 const size_t n = size_t(ci.w) * size_t(ci.h) * 4;
302 ci.rgba.assign(reinterpret_cast<const uint8_t *>(img->getData()),
303 reinterpret_cast<const uint8_t *>(img->getData()) + n);
304 out[key] = std::move(ci);
305 }
306 delete img;
307 } catch (...) {
308 }
309 }
310 }
311 }

316graphics::Renderable3D *makeRenderable(graphics::Graphics *gfx, const MeshSlot &slot,
317 SceneLoader::TextureCache &textures, bool mipmaps) {
318 if (!gfx || !slot.mesh) return nullptr;
319 graphics::Mesh *mesh = gfx->newMeshFromAssimp(*slot.mesh); // local-space (hierarchy transform)
320 if (!mesh) return nullptr;
321 auto *r = graphics::Renderable3D::create();
322 r->meshRenderer()->visible = true;
323 r->setMesh(mesh);
324
325 const aiScene *scene = slot.scene;
326 const aiMaterial *mat = nullptr;
327 if (scene && scene->mMaterials && slot.materialIndex < scene->mNumMaterials)
328 mat = scene->mMaterials[slot.materialIndex];
329
330 aiColor3D base(1.f, 1.f, 1.f);
331 if (mat) {
332 if (mat->Get(AI_MATKEY_BASE_COLOR, base) != AI_SUCCESS)
333 mat->Get(AI_MATKEY_COLOR_DIFFUSE, base);
334 float metallic = 0.f;
335 float roughness = 0.45f;
336 mat->Get(AI_MATKEY_METALLIC_FACTOR, metallic);
337 mat->Get(AI_MATKEY_ROUGHNESS_FACTOR, roughness);
338 r->setMetallic(metallic);
339 r->setRoughness(roughness);
340 }
341
342 graphics::Texture *albedo =
343 resolveTexture(gfx, scene, mat, aiTextureType_BASE_COLOR, textures, mipmaps);
344 if (!albedo) albedo = resolveTexture(gfx, scene, mat, aiTextureType_DIFFUSE, textures, mipmaps);
345 graphics::Texture *normal =
346 resolveTexture(gfx, scene, mat, aiTextureType_NORMALS, textures, mipmaps);
347 graphics::Texture *height =
348 resolveTexture(gfx, scene, mat, aiTextureType_HEIGHT, textures, mipmaps);
349
350 r->setTint(base.r, base.g, base.b, 1.f);
351 if (albedo) r->setTexture(albedo);
352 if (normal) r->setNormalTexture(normal);
353 if (height) r->setHeightTexture(height);
354 return r;
355}

364 glm::mat4 m(1.f);
365 m = glm::translate(m, glm::vec3(n.x, n.y, n.z));
366 if (n.space == "2d") {

368 } else {

370 m = glm::rotate(m, n.pitch, glm::vec3(1.f, 0.f, 0.f));
371 m = glm::rotate(m, n.roll, glm::vec3(0.f, 0.f, 1.f));
372 }
373 m = glm::scale(m, glm::vec3(n.sx, n.sy, n.sz));
374 return m;
375}

377void fillMeshBoundsFromSlot(scene::SceneNode &n, const MeshSlot &slot) {
378 const aiMesh *mesh = slot.mesh;
379 if (!mesh || !mesh->mVertices || mesh->mNumVertices == 0) return;
380 float mn[3] = {std::numeric_limits<float>::max(),

391 mx[c] = std::max(mx[c], f);

393 }
394 n.bminX = mn[0];
395 n.bminY = mn[1];
396 n.bminZ = mn[2];
397 n.bmaxX = mx[0];
398 n.bmaxY = mx[1];
399 n.bmaxZ = mx[2];
400 n.hasBounds = true;

404void unionChildBounds(scene::SceneHost::Tree &tree, int nodeIndex) {
405 scene::SceneNode &n = tree.nodes[size_t(nodeIndex)];
406 for (int c = n.firstChild; c >= 0; c = tree.nodes[size_t(c)].nextSibling) {
407 unionChildBounds(tree, c);
408 }
409
410 bool any = false;
411 float mn[3] = {std::numeric_limits<float>::max(),
412 std::numeric_limits<float>::max(),
413 std::numeric_limits<float>::max()};
414 float mx[3] = {std::numeric_limits<float>::lowest(),
415 std::numeric_limits<float>::lowest(),
416 std::numeric_limits<float>::lowest()};
417 for (int c = n.firstChild; c >= 0; c = tree.nodes[size_t(c)].nextSibling) {
418 scene::SceneNode &ch = tree.nodes[size_t(c)];
419 if (!ch.hasBounds) continue;
420 any = true;
421 const glm::mat4 lm = nodeLocalMatrix(ch);
422 for (int i = 0; i < 8; ++i) {
423 const glm::vec3 p((i & 1) ? ch.bmaxX : ch.bminX,
424 (i & 2) ? ch.bmaxY : ch.bminY,
425 (i & 4) ? ch.bmaxZ : ch.bminZ);
426 const glm::vec4 w = lm * glm::vec4(p, 1.f);
427 for (int k = 0; k < 3; ++k) {
428 mn[k] = std::min(mn[k], w[k]);
429 mx[k] = std::max(mx[k], w[k]);
430 }
431 }
432 }
433 if (!any) return;
434 if (!n.hasBounds) {
435 n.bminX = mn[0];
436 n.bminY = mn[1];
437 n.bminZ = mn[2];
438 n.bmaxX = mx[0];
439 n.bmaxY = mx[1];
440 n.bmaxZ = mx[2];
441 n.hasBounds = true;
442 } else {
443 n.bminX = std::min(n.bminX, mn[0]);
444 n.bminY = std::min(n.bminY, mn[1]);
445 n.bminZ = std::min(n.bminZ, mn[2]);
446 n.bmaxX = std::max(n.bmaxX, mx[0]);
447 n.bmaxY = std::max(n.bmaxY, mx[1]);
448 n.bmaxZ = std::max(n.bmaxZ, mx[2]);
449 }

454void buildNodeRecursive(const aiScene *scene, const aiNode *node, scene::NodeDesc &out,
455 MeshSlotMap *slots, std::unordered_map<std::string, int> &counts) {
456 const std::string rawName = (node && node->mName.length) ? node->mName.C_Str() : "<root>";
457 const std::string id = uniqueId(rawName, counts);
458
459 out.id = id;
460 out.key = id;
461 out.name = rawName;
462 out.space = "3d";
463 out.visible = true;
464 decomposeNode(node->mTransformation, out.x, out.y, out.z, out.yaw, out.pitch, out.roll,
465 out.sx, out.sy, out.sz);
466
467 for (unsigned i = 0; node && i < node->mNumMeshes; ++i) {
468 const unsigned mi = node->mMeshes[i];
469 if (mi >= scene->mNumMeshes) continue;
470 const aiMesh *mesh = scene->mMeshes[mi];
471 if (!mesh || mesh->mNumFaces == 0) continue;
472
473 const std::string cid = id + "_mesh" + std::to_string(i);
474 scene::NodeDesc child;
475 child.id = cid;
476 child.key = cid;
477 child.name = std::string("mesh") + std::to_string(i);
478 child.space = "3d";
479 child.visible = true;
480 out.children.push_back(std::move(child));
481
482 if (slots) {
483 MeshSlot s;
484 s.scene = scene;
485 s.mesh = mesh;
486 s.materialIndex = mesh->mMaterialIndex;
487 (*slots)[cid].push_back(std::move(s));
488 }
489 }
490
491 for (unsigned c = 0; node && c < node->mNumChildren; ++c) {
492 scene::NodeDesc child;
493 buildNodeRecursive(scene, node->mChildren[c], child, slots, counts);
494 out.children.push_back(std::move(child));
495 }

505// Walk the new tree in DFS order, emitting Add / Move / Modify entries.
506void walkNew(const scene::NodeDesc &d, const std::string &parentId,
507 const std::unordered_map<std::string, const scene::SceneNode *> &oldNodes,
508 const std::unordered_map<std::string, std::string> &oldParent, SceneDiff &out) {
509 auto it = oldNodes.find(d.id);
510 if (it == oldNodes.end()) {
511 out.entries.push_back({SceneDiffEntry::Action::Add, d.id, parentId});
512 ++out.added;
513 } else {
514 auto pit = oldParent.find(d.id);
515 const std::string op = (pit != oldParent.end()) ? pit->second : "";
516 if (op != parentId) {
517 out.entries.push_back({SceneDiffEntry::Action::Move, d.id, parentId});
518 ++out.moved;
519 } else if (propsChanged(*it->second, d)) {
520 out.entries.push_back({SceneDiffEntry::Action::Modify, d.id, ""});
521 ++out.modified;
522 }
523 }
524 for (const auto &child : d.children) walkNew(child, d.id, oldNodes, oldParent, out);
525}

529void importLights(const aiScene *scene, std::vector<graphics::Light3D *> &out) {
530 if (!scene) return;
531 for (unsigned i = 0; i < scene->mNumLights; ++i) {

534 std::string type = "point";

538 break;
539 case aiLightSource_POINT:
540 case aiLightSource_SPOT:
541 case aiLightSource_AREA:

543 type = "point";
544 break;

601SceneDiff SceneLoader::diffTree(scene::SceneHost *host, const scene::NodeDesc &newRoot) {
602 SceneDiff out;
603 std::unordered_map<std::string, const scene::SceneNode *> oldNodes;
604 std::unordered_map<std::string, std::string> oldParent;
605 if (host) {
606 auto t = host->tree();
607 for (size_t i = 0; i < t->nodes.size(); ++i) {
608 const scene::SceneNode &n = t->nodes[i];
609 oldNodes[n.id] = &n;
610 oldParent[n.id] = (n.parent >= 0) ? t->nodes[size_t(n.parent)].id : "";
611 }
612 }
613
614 // Collect the set of ids present in the new tree.
615 std::unordered_set<std::string> newIds;
616 std::function<void(const scene::NodeDesc &)> collect = [&](const scene::NodeDesc &d) {
617 newIds.insert(d.id);
618 for (const auto &c : d.children) collect(c);
619 };
620 collect(newRoot);
621
622 // Removed: present in old, absent in new.
623 for (const auto &kv : oldNodes) {
624 if (!newIds.count(kv.first)) {
625 out.entries.push_back({SceneDiffEntry::Action::Remove, kv.first, ""});
626 ++out.removed;
627 }
628 }
629
630 // Add / Move / Modify: walk the new tree (DFS so parents precede children).
631 walkNew(newRoot, "", oldNodes, oldParent, out);
632 return out;
633}

635bool SceneLoader::applyTreeDiff(scene::SceneHost *host, const scene::NodeDesc &newRoot,
636 const SceneDiff &diff, graphics::Graphics *gfx,
637 const MeshSlotMap *slots) {
638 if (!host || diff.empty()) return false;
639
640 // Snapshot the mounted tree by id so kept GameObjects can keep their linked
641 // Renderable3D (no re-upload / no object rebuild) across the update.
642 std::unordered_map<std::string, const scene::SceneNode *> oldNodes;
643 {
644 auto t = host->tree();
645 for (const auto &n : t->nodes) oldNodes[n.id] = &n;
646 }
647
648 // Collect the ids present in the new tree.
649 std::unordered_set<std::string> newIds;
650 std::function<void(const scene::NodeDesc &)> collect = [&](const scene::NodeDesc &d) {
651 newIds.insert(d.id);
652 for (const auto &c : d.children) collect(c);
653 };
654 collect(newRoot);
655
656 // Destroy Renderable3D of removed GameObjects.
657 for (const auto &kv : oldNodes) {
658 if (!newIds.count(kv.first)) {
659 if (const auto *l = host->findLink(kv.second, scene::findLinkKind("renderable3d"))) {
660 destroyRenderable(static_cast<graphics::Renderable3D *>(l->target));
661 }
662 }
663 }
664
665 // Rebuild the arena from the new tree (DFS). Kept nodes copy their link from
666 // the old node (identity preserved); added mesh nodes get a fresh Renderable3D.
667 std::vector<scene::SceneNode> nodes;
668 nodes.reserve(newIds.size());
669 std::function<int(const scene::NodeDesc &, int)> build = [&](const scene::NodeDesc &d,
670 int parentIndex) -> int {
671 const int idx = int(nodes.size());
672 scene::SceneNode n;
673 n.id = d.id;
674 n.key = d.key.empty() ? d.id : d.key;
675 n.name = d.name.empty() ? d.id : d.name;
676 n.space = d.space.empty() ? "3d" : d.space;
677 n.visible = d.visible;
678 n.x = d.x;
679 n.y = d.y;
680 n.z = d.z;
681 n.yaw = d.yaw;
682 n.pitch = d.pitch;
683 n.roll = d.roll;
684 n.sx = d.sx;
685 n.sy = d.sy;
686 n.sz = d.sz;
687 n.localDirty = true;
688 n.world = glm::mat4(1.f);
689 n.firstChild = -1;
690 n.nextSibling = -1;
691 n.parent = parentIndex;
692
693 auto oldIt = oldNodes.find(d.id);
694 if (oldIt != oldNodes.end()) {
695 n.links = oldIt->second->links;
696 n.objectId = oldIt->second->objectId;
697 n.bminX = oldIt->second->bminX;
698 n.bminY = oldIt->second->bminY;
699 n.bminZ = oldIt->second->bminZ;
700 n.bmaxX = oldIt->second->bmaxX;
701 n.bmaxY = oldIt->second->bmaxY;
702 n.bmaxZ = oldIt->second->bmaxZ;
703 n.hasBounds = oldIt->second->hasBounds;
704 }
705
706 nodes.push_back(std::move(n));
707
708 int prevChild = -1;
709 int firstChild = -1;
710 for (const auto &c : d.children) {
711 const int ci = build(c, idx);
712 if (firstChild < 0) firstChild = ci;
713 if (prevChild >= 0) nodes[size_t(prevChild)].nextSibling = ci;
714 prevChild = ci;
715 }
716 nodes[size_t(idx)].firstChild = firstChild;
717 return idx;
718 };
719 const int root = build(newRoot, -1);
720
721 // Fresh Renderable3D for newly added mesh GameObjects (only changed ones).
722 if (gfx && slots) {
723 TextureCache unused;
724 for (auto &n : nodes) {
725 if (!n.links.empty()) continue;
726 auto sit = slots->find(n.id);
727 if (sit == slots->end() || sit->second.empty()) continue;
728 graphics::Renderable3D *r = nullptr;
729 try {
730 r = makeRenderable(gfx, sit->second[0], unused, true);
731 } catch (...) {
732 r = nullptr;
733 }
734 if (r) {
735 n.links.push_back(scene::SceneLink{scene::findLinkKind("renderable3d"), r, 0});
736 }
737 }
738 }
739
740 host->tree()->nodes = std::move(nodes);
741 host->tree()->root = root;
742 host->invalidateIndex();
743 host->markTransformDirty();
744 if (slots) fillSceneBounds(host, *slots);
745 return true;
746}

762void SceneLoader::linkMeshNodes(scene::SceneHost *host, const MeshSlotMap &slots,
763 graphics::Graphics *gfx, const LoadOptions &options,
764 TextureCache &textures, MeshCache &shared,
765 const CpuImageMap *predecoded) {
766 if (!gfx) return;
767 for (const auto &kv : slots) {
768 scene::SceneNode *n = host->findById(kv.first);
769 if (!n || kv.second.empty()) continue;
770 if (host->findLink(n, scene::findLinkKind("renderable3d"))) continue;
771 const MeshSlot &slot = kv.second[0];
772 graphics::Mesh *mesh = nullptr;
773 auto it = shared.find(slot.mesh);
774 if (options.sharedMeshes && it != shared.end()) {
775 mesh = it->second;
776 } else {
777 try {
778 mesh = gfx->newMeshFromAssimp(*slot.mesh);
779 } catch (...) {
780 mesh = nullptr;
781 }
782 if (mesh && options.sharedMeshes) shared[slot.mesh] = mesh;
783 }
784 if (!mesh) continue;
785 auto *r = graphics::Renderable3D::create();
786 r->meshRenderer()->visible = true;
787 r->setMesh(mesh);
788
789 const aiScene *scene = slot.scene;
790 const aiMaterial *mat = nullptr;
791 if (scene && scene->mMaterials && slot.materialIndex < scene->mNumMaterials)
792 mat = scene->mMaterials[slot.materialIndex];
793 aiColor3D base(1.f, 1.f, 1.f);
794 if (mat) {
795 if (mat->Get(AI_MATKEY_BASE_COLOR, base) != AI_SUCCESS)
796 mat->Get(AI_MATKEY_COLOR_DIFFUSE, base);
797 float metallic = 0.f;
798 float roughness = 0.45f;
799 mat->Get(AI_MATKEY_METALLIC_FACTOR, metallic);
800 mat->Get(AI_MATKEY_ROUGHNESS_FACTOR, roughness);
801 r->setMetallic(metallic);
802 r->setRoughness(roughness);
803 }
804 graphics::Texture *albedo = resolveTexture(gfx, scene, mat, aiTextureType_BASE_COLOR,
805 textures, options.mipmaps, predecoded);
806 if (!albedo)
807 albedo = resolveTexture(gfx, scene, mat, aiTextureType_DIFFUSE, textures,
808 options.mipmaps, predecoded);
809 graphics::Texture *normal = resolveTexture(gfx, scene, mat, aiTextureType_NORMALS,
810 textures, options.mipmaps, predecoded);
811 graphics::Texture *height = resolveTexture(gfx, scene, mat, aiTextureType_HEIGHT,
812 textures, options.mipmaps, predecoded);
813 r->setTint(base.r, base.g, base.b, 1.f);
814 if (albedo) r->setTexture(albedo);
815 if (normal) r->setNormalTexture(normal);
816 if (height) r->setHeightTexture(height);
817 n->links.push_back(scene::SceneLink{scene::findLinkKind("renderable3d"), r, 0});
818 }
