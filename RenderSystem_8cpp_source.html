172void applyCamera(float wx, float wy, float ww, float wh, const ViewCam &cam, int viewW, int viewH,
173 float &sx, float &sy, float &sw, float &sh) {
174 if (!cam.valid) {
175 sx = wx;
176 sy = wy;
177 sw = ww;
178 sh = wh;
179 return;
180 }
181 ASSERT_GT(viewW, 0);
182 ASSERT_GT(viewH, 0);
183 const float z = clampZoom(cam.zoom);
184 ASSERT_GT(z, 0.f);
185 sx = (wx - cam.x) * z + float(viewW) * 0.5f;
186 sy = (wy - cam.y) * z + float(viewH) * 0.5f;
187 sw = ww * z;
188 sh = wh * z;

196void collectLights(Canvas *canvasKey, std::vector<PackedLight> &out) {
197 out.clear();
198 if (ecs::current()->getManager<Light2D>() == nullptr) return;
199 auto view = ecs::View<Light2D, Light2D::Data>();
200 for (auto it = view.begin(); it != view.end(); ++it) {
201 auto [d] = *it;
202 if (!d->enabled) continue;
203 if (d->canvas != canvasKey) continue;
204 PackedLight pl;
205 pl.data = d;
206 pl.isPoint = (d->type != "dir");
207 out.push_back(pl);
208 }
209 std::stable_sort(out.begin(), out.end(), [](const PackedLight &a, const PackedLight &b) {
210 if (a.isPoint != b.isPoint) return a.isPoint && !b.isPoint;
211 return a.data->intensity > b.data->intensity;
212 });
213 if (out.size() > size_t(Lighting2DUBO::kMaxLights))
214 out.resize(size_t(Lighting2DUBO::kMaxLights));
215}

217Lighting2DUBO packLights(const std::vector<PackedLight> &lights, const ViewCam &cam, int viewW,
218 int viewH) {
219 Lighting2DUBO ubo{};
220 ubo.ambient = glm::vec4(cam.ambientR, cam.ambientG, cam.ambientB, 0.f);
221 ubo.meta = glm::vec4(float(lights.size()), float(viewW), float(viewH), 0.f);
222 for (size_t i = 0; i < lights.size(); ++i) {
223 const auto *d = lights[i].data;
224 Light2DGpu &g = ubo.lights[i];
225 g.color = glm::vec4(d->r * d->intensity, d->g * d->intensity, d->b * d->intensity, 1.f);
226 if (lights[i].isPoint) {
227 float lx = d->x, ly = d->y;
228 float lr = d->radius;
229 if (cam.valid) {
230 const float z = clampZoom(cam.zoom);
231 lx = (d->x - cam.x) * z + float(viewW) * 0.5f;
232 ly = (d->y - cam.y) * z + float(viewH) * 0.5f;
233 lr = d->radius * z;
234 }
235 g.posRadius = glm::vec4(lx, ly, 0.f, lr);
236 } else {
237 float dx = d->dx, dy = d->dy;
238 const float len = std::sqrt(dx * dx + dy * dy);
239 if (len > 1e-6f) {
240 dx /= len;
241 dy /= len;
242 } else {
243 dx = 0.f;
244 dy = -1.f;
245 }
246 g.posRadius = glm::vec4(dx, dy, 0.f, 0.f);
247 }
248 }
249 return ubo;
250}

252Color modulateUnlit(Color base, float cx, float cy, bool receiveLight, const Lighting2DUBO &ubo) {
253 if (!receiveLight) return base;
254 glm::vec3 lit(ubo.ambient.r, ubo.ambient.g, ubo.ambient.b);
255 const int count = int(ubo.meta.x + 0.5f);
256 for (int i = 0; i < count; ++i) {
257 const auto &L = ubo.lights[i];
258 glm::vec3 col(L.color.r, L.color.g, L.color.b);
259 if (L.posRadius.w <= 0.f) {
260 // Directional: approximate as constant contribution for unlit sprites.
261 lit += col * 0.65f;
262 } else {
263 const float dx = L.posRadius.x - cx;
264 const float dy = L.posRadius.y - cy;
265 const float dist = std::sqrt(dx * dx + dy * dy);
266 float atten = 1.f - dist / std::max(L.posRadius.w, 1.f);
267 if (atten < 0.f) atten = 0.f;
268 atten *= atten;
269 lit += col * atten;
270 }
271 }
272 return Color(base.r * lit.r, base.g * lit.g, base.b * lit.b, base.a);
273}

278 std::unordered_map<Canvas *, Camera2D *> defaultCam;
279 if (ecs::current()->getManager<Camera2D>() != nullptr) {
280 auto camView = ecs::View<Camera2D, Camera2D::Data>();
281 for (auto it = camView.begin(); it != camView.end(); ++it) {
282 auto [data] = *it;
283 if (!data->active) continue;
284 ASSERT(data->entity != nullptr);
285 if (!data->entity) continue;
286 Canvas *key = data->canvas;
287 if (defaultCam.find(key) == defaultCam.end()) defaultCam[key] = data->entity;
288 }
289 }
290
291 auto view = ecs::View<Renderable2D, Renderable2D::Transform2D, Renderable2D::Sprite>();
292 for (auto it = view.begin(); it != view.end(); ++it) {
293 auto [xf, sp] = *it;
294 if (!sp->visible) continue;
295
296 Camera2D *camEnt = sp->camera;
297 if (!camEnt) {
298 auto found = defaultCam.find(sp->canvas);
299 camEnt = (found != defaultCam.end()) ? found->second : nullptr;

302 ViewCam cam = fromEntity(camEnt);
303 DrawItem2D item;
304 item.x = xf->x + float(sp->offsetX) * xf->sx;
305 item.y = xf->y + float(sp->offsetY) * xf->sy;
306 item.w = float(sp->trimW > 0 ? sp->trimW : sp->width) * xf->sx;
307 item.h = float(sp->trimH > 0 ? sp->trimH : sp->height) * xf->sy;
308 item.depthY = xf->y + sp->height * xf->sy;
309 item.rotation = xf->rot;
310 item.anchorX = sp->trimW > 0
311 ? (sp->anchorX * sp->width - float(sp->offsetX)) / float(sp->trimW)
312 : sp->anchorX;
313 item.anchorY = sp->trimH > 0
314 ? (sp->anchorY * sp->height - float(sp->offsetY)) / float(sp->trimH)
315 : sp->anchorY;
316 item.flipX = sp->flipX;
317 item.flipY = sp->flipY;
318 item.color = Color(sp->r, sp->g, sp->b, sp->a);
319 item.layer = sp->layer;
320 item.blend = sp->blend;
321 item.texture = sp->texture;
322 item.normal = sp->normalTexture;
323 item.quad = sp->quad;
324 item.shader = sp->shader;
325 item.canvas = sp->canvas;
326 item.camera = camEnt;
327 item.camValid = cam.valid;
328 item.camX = cam.x;
329 item.camY = cam.y;
330 item.camZoom = cam.zoom;
331 item.camClear = cam.clearColor;
332 item.camAmbientR = cam.ambientR;
333 item.camAmbientG = cam.ambientG;
334 item.camAmbientB = cam.ambientB;
335 item.receiveLight = sp->receiveLight;
336 item.litPath = sp->receiveLight && sp->normalTexture != nullptr && sp->texture != nullptr &&
337 sp->shader == nullptr;
338 if (item.quad && item.texture) {
339 item.hasUV = true;
340 item.quad->getUV(item.texture->getWidth(), item.texture->getHeight(), item.u0, item.v0,
341 item.u1, item.v1);
342 }
343 out.push_back(item);

347void RenderSystem::drawItems(Graphics &gfx, std::vector<DrawItem2D> &items, bool present) {
348 eve::debug::RenderPassScope pass("RenderSystem2D");
349 sortDrawItems2D(items);
350
351 std::unordered_map<Canvas *, Camera2D *> defaultCam;
352 if (ecs::current()->getManager<Camera2D>() != nullptr) {
353 auto camView = ecs::View<Camera2D, Camera2D::Data>();
354 for (auto it = camView.begin(); it != camView.end(); ++it) {
355 auto [data] = *it;
356 if (!data->active) continue;
357 if (!data->entity) continue;
358 Canvas *key = data->canvas;
359 if (defaultCam.find(key) == defaultCam.end()) defaultCam[key] = data->entity;
360 }
361 }
362
363 Canvas *current = reinterpret_cast<Canvas *>(static_cast<uintptr_t>(1));
364 Lighting2DUBO currentLights{};
365 for (size_t i = 0; i < items.size(); ++i) {
366 Canvas *next = items[i].canvas;
367 if (i == 0 || next != current) {
368 Color clearCol = gfx.getBackgroundColor();
369 ViewCam groupCam{};
370 auto defIt = defaultCam.find(next);
371 if (defIt != defaultCam.end()) {
372 groupCam = fromEntity(defIt->second);
373 clearCol = groupCam.clearColor;
374 } else if (items[i].camValid) {
375 groupCam.valid = true;
376 groupCam.x = items[i].camX;
377 groupCam.y = items[i].camY;
378 groupCam.zoom = items[i].camZoom;
379 groupCam.clearColor = items[i].camClear;
380 groupCam.ambientR = items[i].camAmbientR;
381 groupCam.ambientG = items[i].camAmbientG;
382 groupCam.ambientB = items[i].camAmbientB;
383 clearCol = groupCam.clearColor;
384 }
385
386 gfx.setCanvas(next);
387 eve::debug::rtTarget(next ? "canvas" : "screen");
388 // Only clear when presenting a full sprite pass; map-unified draws must not wipe.
389 if (present) gfx.clear(clearCol, std::nullopt, std::nullopt);
390 current = next;
391
392 int viewW = next ? next->getWidth() : gfx.getWidth();
393 int viewH = next ? next->getHeight() : gfx.getHeight();
394 std::vector<PackedLight> packed;
395 collectLights(next, packed);
396 currentLights = packLights(packed, groupCam, viewW, viewH);
397 gfx.setLighting2D(currentLights);
398 }
399
400 const auto &it = items[i];
401 ViewCam cam{};
402 if (it.camValid) {
403 cam.valid = true;
404 cam.x = it.camX;
405 cam.y = it.camY;
406 cam.zoom = it.camZoom;
407 } else if (it.camera) {
408 cam = fromEntity(it.camera);
409 }
410 int viewW = it.canvas ? it.canvas->getWidth() : gfx.getWidth();
411 int viewH = it.canvas ? it.canvas->getHeight() : gfx.getHeight();
412 float sx, sy, sw, sh;
413 applyCamera(it.x, it.y, it.w, it.h, cam, viewW, viewH, sx, sy, sw, sh);
414
415 float u0 = it.hasUV ? it.u0 : 0.f;
416 float v0 = it.hasUV ? it.v0 : 0.f;
417 float u1 = it.hasUV ? it.u1 : 1.f;
418 float v1 = it.hasUV ? it.v1 : 1.f;
419 if (!it.hasUV && it.quad && it.texture)
420 it.quad->getUV(it.texture->getWidth(), it.texture->getHeight(), u0, v0, u1, v1);
421 if (it.flipX) std::swap(u0, u1);
422 if (it.flipY) std::swap(v0, v1);
423
424 const float pivotX = sx + sw * it.anchorX;
425 const float pivotY = sy + sh * it.anchorY;
426 const float radians = it.rotation * 3.14159265358979323846f / 180.f;
427 const float ox = sw * (0.5f - it.anchorX);
428 const float oy = sh * (0.5f - it.anchorY);
429 const float centerX = pivotX + ox * std::cos(radians) - oy * std::sin(radians);
430 const float centerY = pivotY + ox * std::sin(radians) + oy * std::cos(radians);
431
432 if (it.sceneColorDistortion) {
433 eve::debug::rtBind("texture", "distortion");
434 eve::debug::rtBind("texture", "sceneColor");
435 eve::debug::rtDraw("drawSceneColorDistortionUVRotated", "distortion");
436 gfx.drawSceneColorDistortionUVRotated(it.texture, centerX, centerY, sw, sh, it.rotation,
437 u0, v0, u1, v1, it.distortionStrength,
438 it.color.a, it.rotatedUV);
439 } else if (it.litPath) {
440 eve::debug::rtBind("texture", "albedo");
441 eve::debug::rtBind("texture", "normal");
442 eve::debug::rtDraw("drawTexturedRectLitUV", "lit2d");
443 gfx.drawTexturedRectLitUV(it.texture, it.normal, sx, sy, sw, sh, u0, v0, u1, v1,
444 it.color);
445 } else if (it.texture) {
446 Color c = it.color;
447 if (it.receiveLight && !it.normal)
448 c = modulateUnlit(c, sx + sw * 0.5f, sy + sh * 0.5f, true, currentLights);
449 eve::debug::rtBind("texture", "sprite");
450 if (it.shader) eve::debug::rtBind("shader", "custom");
451 if (it.rotation != 0.f) {
452 eve::debug::rtDraw("drawTexturedRectShaderUVRotated",
453 it.shader ? "shader" : "textured");
454 gfx.drawTexturedRectShaderUVRotated(it.texture, it.shader, centerX,
455 centerY, sw, sh, it.rotation, u0, v0, u1,
456 v1, c, it.rotatedUV, it.blend);
457 } else {
458 eve::debug::rtDraw("drawTexturedRectShaderUV", it.shader ? "shader" : "textured");
459 gfx.drawTexturedRectShaderUV(it.texture, it.shader, sx, sy, sw, sh, u0, v0, u1, v1, c,
460 it.rotatedUV, it.blend);
461 }
462 } else {
463 Color c = it.color;
464 if (it.receiveLight)
465 c = modulateUnlit(c, sx + sw * 0.5f, sy + sh * 0.5f, true, currentLights);
466 eve::debug::rtDraw("drawSolidRect", "solid");
467 if (it.rotation != 0.f)
468 gfx.drawSolidRectRotated(centerX, centerY, sw, sh, it.rotation, c,
469 it.blend);
470 else
471 gfx.drawSolidRect(sx, sy, sw, sh, c, it.blend);
472 }
473 }
474
475 if (present) {
476 gfx.setCanvas();
477 eve::debug::rtPassBegin("Present");
478 gfx.present();
479 eve::debug::rtPassEnd("Present");
480 }
