66Result<void> CollapseAnimation::capture(const GeometryCollectionInstance& source, float seconds) {
67 const auto sourceGeometry = source.fragmentGeometry();
68 if (!frames_.empty() && sourceGeometry != geometry_)
69 return Result<void>::failure(invalid("recording source geometry changed"));
70 const int count = source.boneCount();
71 if (!std::isfinite(seconds) || (frames_.empty() ? seconds != 0 : seconds <= duration()) || count <= 0 ||
72 (!frames_.empty() && count != boneCount()) || (frames_.size() + 1) * static_cast<size_t>(count) > MaxSamples)
73 return Result<void>::failure(invalid("invalid capture time, topology or sample budget"));
74 Frame frame{seconds, {}};
75 frame.bones.reserve(static_cast<size_t>(count));
76 for (int i = 0; i < count; ++i) {
77 auto pose = source.bonePresentation(i);
78 if (!pose) return Result<void>::failure(pose.status());
79 if (!valid(pose.value())) return Result<void>::failure(invalid("invalid fragment pose"));
80 if (!frames_.empty()) {
81 const auto& a = frames_.front().bones[static_cast<size_t>(i)];
82 const auto& b = pose.value();
83 if (a.halfExtentX != b.halfExtentX || a.halfExtentY != b.halfExtentY || a.halfExtentZ != b.halfExtentZ ||
84 a.clusterId != b.clusterId || a.fractureLevel != b.fractureLevel)
85 return Result<void>::failure(invalid("fragment topology changed while recording"));
86 }
87 frame.bones.push_back(pose.value());
88 }
89 if (frames_.empty()) geometry_ = sourceGeometry;
90 frames_.push_back(std::move(frame));
91 return Result<void>::success();
92}
93Result<BonePresentation> CollapseAnimation::sample(float seconds, int bone) const {
94 if (!std::isfinite(seconds) || bone < 0 || bone >= boneCount())
95 return Result<BonePresentation>::failure(invalid("invalid sample time or fragment"));
96 auto upper = std::upper_bound(frames_.begin(), frames_.end(), seconds,
97 [](float t, const Frame& f) { return t < f.seconds; });
98 const size_t index = static_cast<size_t>(bone);
99 if (upper == frames_.begin()) return Result<BonePresentation>::success(upper->bones[index]);
100 const auto& a = *(upper - 1);
101 if (upper == frames_.end()) return Result<BonePresentation>::success(a.bones[index]);
102 auto result = a.bones[index];
103 const auto& b = upper->bones[index];
104 const float t = (seconds - a.seconds) / (upper->seconds - a.seconds);
105 result.worldX = std::lerp(result.worldX, b.worldX, t);
106 result.worldY = std::lerp(result.worldY, b.worldY, t);
107 result.worldZ = std::lerp(result.worldZ, b.worldZ, t);
108 const float sign =
109 (result.rotX * b.rotX + result.rotY * b.rotY + result.rotZ * b.rotZ + result.rotW * b.rotW) < 0 ? -1.f : 1.f;
110 result.rotX = std::lerp(result.rotX, sign * b.rotX, t);
111 result.rotY = std::lerp(result.rotY, sign * b.rotY, t);
112 result.rotZ = std::lerp(result.rotZ, sign * b.rotZ, t);
113 result.rotW = std::lerp(result.rotW, sign * b.rotW, t);
114 const float norm = std::sqrt(result.rotX * result.rotX + result.rotY * result.rotY + result.rotZ * result.rotZ +
115 result.rotW * result.rotW);
116 result.rotX /= norm;
117 result.rotY /= norm;
118 result.rotZ /= norm;
119 result.rotW /= norm;
120 return Result<BonePresentation>::success(result);
121}

158Result<CollapseAnimation> CollapseAnimation::load(std::string_view path) {

181 if (!in || vertices < 4 || vertices > 254 || triangles == 0 || triangles > 2730 || total > 4000000)

184 fragment.positions.resize(3ULL * vertices);
185 fragment.indices.resize(3ULL * triangles);
186 fragment.materials.resize(triangles);
187 for (auto& v : fragment.positions) v = readFloat(in);
188 for (auto& i : fragment.indices) i = read32(in);
189 for (auto& tag : fragment.materials) tag = read32(in);
190 geometry.fragments.push_back(std::move(fragment));
191 }

200 for (std::uint32_t f = 0; f < frames; ++f) {

218 if (f > 0) {
