12Result<void> evaluateFollowPlayer(FollowPlayerOutput* output, const FollowPlayerSettings* settings,
13 const FollowPlayerInput* input) {
14 if (!output || !settings || !input || !finite(settings->offsetX) || !finite(settings->offsetY) ||
15 !finite(settings->offsetZ) || !finite(settings->scaleX) || !finite(settings->scaleY) ||
16 !finite(settings->scaleZ) || !finite(input->playerX) || !finite(input->playerY) || !finite(input->playerZ))
17 return Result<void>::failure(Diagnostic::error(
18 DiagnosticCode::InvalidArgument, "scene.followPlayer: objects and finite transform values required"));
19
20 FollowPlayerOutput candidate;
21 candidate.applyScale = settings->useScale;
22 candidate.scaleX = settings->scaleX;
23 candidate.scaleY = settings->scaleY;
24 candidate.scaleZ = settings->scaleZ;
25 candidate.applyPosition = settings->followPlayer && input->hasPlayer;
26 if (candidate.applyPosition) {
27 candidate.x = input->playerX;
28 candidate.y = input->playerY;
29 candidate.z = input->playerZ;
30 if (settings->useOffset) {
31 candidate.x += settings->offsetX;
32 candidate.z -= settings->offsetZ;
33 if (settings->waterObject)
34 candidate.y += (input->playerY < 1.f ? 70.f : 10.f) - settings->offsetY;
35 else
36 candidate.y -= settings->offsetY;
37 }
38 }
39 *output = candidate;
40 return Result<void>::success();
41}
