8Result<void> appendTriangle(asset::CanonicalMeshData& mesh, const std::array<std::array<double, 3>, 3>& triangle,
9 const AssetImportLimits& limits) {
10 const auto vertices = mesh.positions.size() / 3;
11 if (vertices + 3 > limits.maximumVerticesPerPrimitive ||
12 mesh.indices.size() + 3 > limits.maximumIndicesPerPrimitive ||
13 (vertices + 3) * 28 + 24 > limits.maximumDecodedBytes)
14 return Result<void>::failure(Diagnostic::error(
15 DiagnosticCode::InvalidArgument, "structured model triangle budget exceeded", {}, {}, "asset.import"));
16 for (const auto& point : triangle)
17 for (double value : point)
18 if (!std::isfinite(value) || std::abs(value) > std::numeric_limits<float>::max())
19 return Result<void>::failure(Diagnostic::error(
20 DiagnosticCode::ParseError, "nonfinite or overflowing model coordinate", {}, {}, "asset.import"));
21 auto rounded = triangle;
22 for (auto& point : rounded)
23 for (auto& value : point) value = static_cast<float>(value);
24 std::array<double, 3> a{}, b{}, normal{};
25 for (std::size_t i = 0; i < 3; ++i) {
26 a[i] = rounded[1][i] - rounded[0][i];
27 b[i] = rounded[2][i] - rounded[0][i];
28 }
29 normal = {a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]};
30 const double length = std::hypot(normal[0], normal[1], normal[2]);
31 if (!std::isfinite(length) || length == 0)
32 return Result<void>::failure(
33 Diagnostic::error(DiagnosticCode::ParseError, "degenerate model triangle", {}, {}, "asset.import"));
34 for (std::size_t vertex = 0; vertex < 3; ++vertex) {
35 for (std::size_t axis = 0; axis < 3; ++axis) {
36 mesh.positions.push_back(float(triangle[vertex][axis]));
37 mesh.normals.push_back(float(normal[axis] / length));
38 }
39 mesh.indices.push_back(static_cast<std::uint32_t>(vertices + vertex));
40 }
41 return Result<void>::success();
42}
