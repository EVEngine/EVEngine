54 const EditorValue* output = field(compiled.configuration, "output");
55 if (emitter->getBufferSize() != integer(*output, "bufferSize"))
56 return runtimeError<void>(EditorStatus::Conflict, "editor.particles.runtime-buffer-size",
57 "Live emitter buffer size differs from the graph output; recreate the emitter");
58 const std::string texture = text(*renderer, "texture");
59 graphics::Texture* resolvedTexture = nullptr;

67 "Particle texture asset could not be resolved: " + texture);
68 }
