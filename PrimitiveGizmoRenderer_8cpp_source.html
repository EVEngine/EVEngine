54eve::Result<graphics::PrimitiveDrawStatistics> PrimitiveGizmoRenderer::render(
55 const editing::GizmoSnapshot& snapshot, const graphics::SceneDrawContext& context,
56 graphics::IGraphics3D& graphicsBackend) const {
57 for (std::size_t index = 0; index < snapshot.primitives.size(); ++index) {
58 const auto& primitive = snapshot.primitives[index];
59 const std::string path = "primitives[" + std::to_string(index) + "]";
60 if (!supported(primitive.kind))
61 return failure("unsupported gizmo primitive kind: " + primitive.kind, path + ".kind");
62 if (!finite(primitive)) return failure("gizmo primitive must be finite", path);
63 if ((primitive.kind == "sphere" || primitive.kind == "capsule") && primitive.radius <= 0.0)
64 return failure("gizmo radius must be positive", path + ".radius");
65 if ((primitive.kind == "line" || primitive.kind == "capsule" || primitive.kind == "arrow" ||
66 primitive.kind == "camera" || primitive.kind == "frustum") &&
67 primitive.length <= 0.0)
68 return failure("gizmo length must be positive", path + ".length");
69 }
70
71 try {
72 graphics::PrimitiveSceneCanvas3D canvas(context);
73 for (const auto& primitive : snapshot.primitives) {
74 const glm::vec3 origin = vector(primitive.position);
75 glm::vec3 direction = vector(primitive.direction);
76 if (glm::length(direction) <= 1e-6f) direction = {0.f, 0.f, -1.f};
77 direction = glm::normalize(direction);
78 auto style = paint(primitive);
79 if (primitive.kind == "point") {
80 style.stroke.width = primitive.radius > 0.0 ? static_cast<float>(primitive.radius) : 5.f;
81 canvas.drawPoint(origin, style);
82 } else if (primitive.kind == "line") {
83 canvas.drawLine(origin, origin + direction * static_cast<float>(primitive.length), style);
84 } else if (primitive.kind == "box") {
85 const glm::vec3 half = vector(primitive.size) * 0.5f;
86 canvas.drawAabb(origin - half, origin + half, style);
87 } else if (primitive.kind == "sphere") {
88 canvas.drawSphere(origin, static_cast<float>(primitive.radius), style);
89 } else if (primitive.kind == "capsule") {
90 canvas.drawCapsule(origin, origin + direction * static_cast<float>(primitive.length),
91 static_cast<float>(primitive.radius), style);
92 } else if (primitive.kind == "arrow" || primitive.kind == "camera") {
93 const float length = static_cast<float>(primitive.length);
94 canvas.drawArrow(origin, origin + direction * length, length * 0.2f, std::max(0.04f, length * 0.08f),
95 style);
96 } else if (primitive.kind == "area") {
97 const glm::vec3 half = vector(primitive.size) * 0.5f;
98 canvas.drawQuad(origin + glm::vec3(-half.x, 0.f, -half.z), origin + glm::vec3(half.x, 0.f, -half.z),
99 origin + glm::vec3(half.x, 0.f, half.z), origin + glm::vec3(-half.x, 0.f, half.z),
100 style);
101 } else {
102 const float length = static_cast<float>(primitive.length);
103 const float radius = std::max(0.01f, static_cast<float>(primitive.radius));
104 glm::vec3 up = std::abs(direction.y) < 0.99f ? glm::vec3(0.f, 1.f, 0.f) : glm::vec3(1.f, 0.f, 0.f);
105 const glm::vec3 right = glm::normalize(glm::cross(direction, up)) * radius;
106 up = glm::normalize(glm::cross(right, direction)) * radius;
107 const glm::vec3 farCenter = origin + direction * length;
108 const std::array<glm::vec3, 8> corners{origin,
109 origin,
110 origin,
111 origin,
112 farCenter - right - up,
113 farCenter + right - up,
114 farCenter - right + up,
115 farCenter + right + up};
116 canvas.drawFrustum(corners, style);
117 }
118 }
119 graphicsBackend.drawPrimitiveScene(canvas);
120 return eve::Result<graphics::PrimitiveDrawStatistics>::success(canvas.statistics());
121 } catch (const std::exception& error) {
122 return failure(error.what(), "snapshot");
123 }
124}
