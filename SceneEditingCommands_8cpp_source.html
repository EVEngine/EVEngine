15bool assignVector3(const editing::Value& payload, const char* key, double& x, double& y, double& z) {
16 const editing::Value* entry = field(payload, key);
17 if (!entry) return true;
18 const auto* values = entry->getIf<editing::Value::Array>();
19 if (!values || values->size() != 3) return false;
20 const auto number = [](const editing::Value& value, double& output) {
21 if (const auto* real = value.getIf<double>()) {
22 output = *real;
23 return true;
24 }
25 if (const auto* integer = value.getIf<std::int64_t>()) {
26 output = static_cast<double>(*integer);
27 return true;
28 }
29 return false;
30 };
31 return number((*values)[0], x) && number((*values)[1], y) && number((*values)[2], z);
32}

41editing::Result<void> registerEditingCommands(editing::IEditingCommandRegistry& registry) {
42 editing::EditingCommandDescriptor descriptor;
43 descriptor.id = editing::CommandId("scene.transform.set.v1");
44 descriptor.ownerModule = "scene_editing";
45 descriptor.displayName = "Set scene transform";
46 descriptor.category = "Scene";
47 descriptor.automationAllowed = true;
48 return registry.registerPlannedCommand(
49 std::move(descriptor), [](editing::IEditableTarget& target, const editing::CommandRequest& request) {
50 auto* capability = static_cast<ITransformEditTarget*>(
51 target.queryCapability(ITransformEditTarget::editingCapabilityId()));
52 const editing::Value* objectValue = field(request.payload, "object");
53 const auto* object = objectValue ? objectValue->getIf<std::string>() : nullptr;
54 if (!capability || !object)
55 return error<editing::CommandPlan>(editing::Status::Rejected, "scene.editing.transform-payload",
56 "Scene transform requires a transform target and object id");
57 auto current = capability->readTransform(editing::ObjectId(*object));
58 if (!current.ok())
59 return error<editing::CommandPlan>(current.code(), "scene.editing.transform-object",
60 "Scene object transform is unavailable");
61 SceneTransformValue transform = current.value();
62 if (!assignVector3(request.payload, "position", transform.x, transform.y, transform.z) ||
63 !assignVector3(request.payload, "rotation", transform.rotationX, transform.rotationY,
64 transform.rotationZ) ||
65 !assignVector3(request.payload, "scale", transform.scaleX, transform.scaleY, transform.scaleZ))
66 return error<editing::CommandPlan>(editing::Status::Rejected, "scene.editing.transform-vector",
67 "Position, rotation and scale must contain three numbers");
68 auto operation = capability->makeSetTransform(editing::ObjectId(*object), transform);
69 if (!operation.ok())
70 return error<editing::CommandPlan>(operation.code(), "scene.editing.transform-operation",
71 "Scene target rejected the transform");
72 editing::CommandPlan plan;
73 plan.operations.push_back(std::move(operation.value()));
74 plan.summary = editing::Value::Object{{"object", *object}};
75 return eve::editing::applied<editing::CommandPlan>(std::move(plan));
76 });
77}
