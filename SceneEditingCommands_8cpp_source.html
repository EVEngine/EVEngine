10const editing::Value* field(const editing::Value& value, const char* key) {
11 const auto* object = value.getIf<editing::Value::Object>();
12 if (!object) return nullptr;
13 const auto found = object->find(key);
14 return found == object->end() ? nullptr : &found->second;
15}

17bool assignVector3(const editing::Value& payload, const char* key, double& x, double& y, double& z) {
18 const editing::Value* entry = field(payload, key);
19 if (!entry) return true;
20 const auto* values = entry->getIf<editing::Value::Array>();
21 if (!values || values->size() != 3) return false;
22 const auto number = [](const editing::Value& value, double& output) {
23 if (const auto* real = value.getIf<double>()) {
24 output = *real;
25 return true;
26 }
27 if (const auto* integer = value.getIf<std::int64_t>()) {
28 output = static_cast<double>(*integer);
29 return true;
30 }
31 return false;
32 };
33 return number((*values)[0], x) && number((*values)[1], y) && number((*values)[2], z);
34}
