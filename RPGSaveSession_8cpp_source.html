39eve::SnapshotHashProvider saveIntegrityProvider() {
40 return [](std::string_view input) -> eve::Result<eve::ContentId> {
41 std::uint64_t left = 14695981039346656037ull;
42 std::uint64_t right = 1099511628211ull;
43 for (const unsigned char byte : input) {
44 left ^= byte;
45 left *= 1099511628211ull;
46 right ^= static_cast<std::uint64_t>(byte) + 0x9e3779b97f4a7c15ull;
47 right *= 14029467366897019727ull;
48 }
49 eve::ContentId::Bytes bytes{};
50 for (int index = 0; index < 8; ++index) {
51 bytes[static_cast<std::size_t>(index)] = static_cast<std::uint8_t>(left >> (56 - index * 8));
52 bytes[static_cast<std::size_t>(index + 8)] = static_cast<std::uint8_t>(right >> (56 - index * 8));
53 }
54 return eve::Result<eve::ContentId>::success(eve::ContentId(bytes));
55 };
56}

72std::size_t applyIdRename(eve::Value &payload, RPGSaveIdDomain domain, std::string_view oldId,
73 std::string_view newId) {
74 eve::Value *actor = payload.find("actor");
75 eve::Value *tracker = payload.find("questTracker");
76 eve::Value *inventory = payload.find("inventory");
77 switch (domain) {
78 case RPGSaveIdDomain::Item: {
79 std::size_t changed = 0;
80 eve::Value *bag = inventory ? inventory->find("bag") : nullptr;
81 changed += visitArray(bag ? bag->find("slots") : nullptr, [&](eve::Value &stack) {
82 return renameStringValue(stack.find("itemId"), oldId, newId);
83 });
84 eve::Value *equipment = inventory ? inventory->find("equipment") : nullptr;
85 changed += visitArray(equipment ? equipment->find("slots") : nullptr, [&](eve::Value &slot) {
86 eve::Value *stack = slot.find("stack");
87 return renameStringValue(stack ? stack->find("itemId") : nullptr, oldId, newId);
88 });
89 return changed;
90 }
91 case RPGSaveIdDomain::Quest:
92 return visitArray(tracker ? tracker->find("entries") : nullptr, [&](eve::Value &entry) {
93 return renameStringValue(entry.find("id"), oldId, newId);
94 });
95 case RPGSaveIdDomain::QuestObjective:
96 return visitArray(tracker ? tracker->find("entries") : nullptr, [&](eve::Value &entry) {
97 return visitArray(entry.find("objectives"), [&](eve::Value &objective) {
98 return renameStringValue(objective.find("id"), oldId, newId);
99 });
100 });
101 case RPGSaveIdDomain::Skill:
102 return visitArray(actor ? actor->find("learnedSkills") : nullptr, [&](eve::Value &skill) {
103 return renameStringValue(&skill, oldId, newId);
104 });
105 case RPGSaveIdDomain::Class: {
106 eve::Value *classValue = actor ? actor->find("class") : nullptr;
107 return renameStringValue(classValue ? classValue->find("id") : nullptr, oldId, newId);
108 }
109 case RPGSaveIdDomain::Trait:
110 return visitArray(actor ? actor->find("traits") : nullptr, [&](eve::Value &trait) {
111 return renameStringValue(trait.find("traitId"), oldId, newId);
112 });
113 case RPGSaveIdDomain::TraitSource:
114 return visitArray(actor ? actor->find("traits") : nullptr, [&](eve::Value &trait) {
115 return renameStringValue(trait.find("source"), oldId, newId);
116 });
117 case RPGSaveIdDomain::EquipmentSlot: {
118 eve::Value *equipment = inventory ? inventory->find("equipment") : nullptr;
119 return visitArray(equipment ? equipment->find("slots") : nullptr, [&](eve::Value &slot) {
120 return renameStringValue(slot.find("name"), oldId, newId);
121 });
122 }
123 }
124 return 0;
125}

127eve::Result<void> applyQuestAdditions(eve::Value &payload, const std::set<std::string> &questIds) {
128 eve::Value *tracker = payload.find("questTracker");
129 eve::Value *entries = tracker ? tracker->find("entries") : nullptr;
130 if (!tracker || !tracker->isObject() || !entries || !entries->isArray())
131 return saveFailure<void>(eve::DiagnosticCode::ParseError,
132 "quest-addition migration requires a tracker entries array",
133 "payload.questTracker.entries");
134
135 Tracker baseline;
136 auto baselineJson = baseline.snapshotJson();
137 if (!baselineJson.ok()) return eve::Result<void>::failure(baselineJson.status());
138 auto parsedBaseline = eve::Value::fromJson(baselineJson.value());
139 if (!parsedBaseline.ok()) return eve::Result<void>::failure(parsedBaseline.status());
140 eve::Value baselineRoot = std::move(parsedBaseline).takeValue();
141 eve::Value *baselineEntries = baselineRoot.find("entries");
142 if (!baselineEntries || !baselineEntries->isArray())
143 return saveFailure<void>(eve::DiagnosticCode::InvariantViolation,
144 "current tracker did not produce an entries array", "questTracker.entries");
145
146 std::unordered_map<std::string, eve::Value> oldEntries;
147 for (std::size_t index = 0; index < entries->arraySize(); ++index) {
148 const eve::Value &entry = entries->at(index);
149 const eve::Value *id = entry.isObject() ? entry.find("id") : nullptr;
150 if (!id || !id->isString())
151 return saveFailure<void>(eve::DiagnosticCode::ParseError,
152 "quest-addition migration found an invalid old tracker entry",
153 "payload.questTracker.entries[" + std::to_string(index) + "].id");
154 if (!oldEntries.emplace(id->asString(), entry).second)
155 return saveFailure<void>(eve::DiagnosticCode::Conflict,
156 "quest-addition migration found a duplicate old quest id",
157 "payload.questTracker.entries[" + std::to_string(index) + "].id");
158 }
159 for (const auto &questId : questIds) {
160 if (oldEntries.contains(questId))
161 return saveFailure<void>(eve::DiagnosticCode::Conflict,
162 "quest-addition migration target already exists in the old tracker",
163 "payload.questTracker.entries." + questId);
164 }
165
166 eve::Value::Array rebuilt;
167 rebuilt.reserve(baselineEntries->arraySize());
168 std::set<std::string> appliedAdditions;
169 for (std::size_t index = 0; index < baselineEntries->arraySize(); ++index) {
170 const eve::Value &initial = baselineEntries->at(index);
171 const eve::Value *id = initial.find("id");
172 if (!id || !id->isString())
173 return saveFailure<void>(eve::DiagnosticCode::InvariantViolation,
174 "current tracker produced an invalid quest id", "questTracker.entries");
175 auto old = oldEntries.find(id->asString());
176 if (old != oldEntries.end()) {
177 rebuilt.push_back(std::move(old->second));
178 oldEntries.erase(old);
179 } else if (questIds.contains(id->asString())) {
180 rebuilt.push_back(initial);
181 appliedAdditions.emplace(id->asString());
182 } else {
183 return saveFailure<void>(eve::DiagnosticCode::Conflict,
184 "old tracker is missing a quest without an additive migration",
185 "payload.questTracker.entries." + id->asString());
186 }
187 }
188 if (!oldEntries.empty())
189 return saveFailure<void>(eve::DiagnosticCode::Conflict,
190 "old tracker contains a quest absent from the current registry",
191 "payload.questTracker.entries." + oldEntries.begin()->first);
192 if (appliedAdditions != questIds)
193 return saveFailure<void>(eve::DiagnosticCode::NotFound,
194 "quest-addition migration target is absent from the current registry",
195 "questTracker.entries");
196 entries->operator=(eve::Value(std::move(rebuilt)));
197 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
198}

256 if (!compatibleContentVersions_.emplace(contentVersion).second)

338eve::Result<std::string> RPGSaveSession::snapshotJson() const {
339 if (!gameState_ || !tracker_ || (!actor_ && !party_) || !bag_ || !equipment_)
340 return saveFailure<std::string>(eve::DiagnosticCode::PreconditionViolation,
341 "RPG save session requires bound participants");
342 if (contentVersion_.empty())
343 return saveFailure<std::string>(eve::DiagnosticCode::PreconditionViolation,
344 "RPG save session requires a content version");
345 auto gameStateJson = gameState_->snapshotJson();
346 if (!gameStateJson.ok()) return eve::Result<std::string>::failure(gameStateJson.status());
347 auto trackerJson = tracker_->snapshotJson();
348 if (!trackerJson.ok()) return eve::Result<std::string>::failure(trackerJson.status());
349 auto actorJson = actor_ ? actor_->checkpointJson() : party_->checkpointJson();
350 if (!actorJson.ok()) return eve::Result<std::string>::failure(actorJson.status());
351 inventory::InventorySaveSession inventorySession;
352 inventorySession.bind(*bag_, *equipment_);
353 auto inventoryJson = inventorySession.snapshotJson();
354 if (!inventoryJson.ok()) return eve::Result<std::string>::failure(inventoryJson.status());
355 auto gameStateValue = eve::Value::fromJson(gameStateJson.value());
356 if (!gameStateValue.ok()) return eve::Result<std::string>::failure(gameStateValue.status());
357 auto trackerValue = eve::Value::fromJson(trackerJson.value());
358 if (!trackerValue.ok()) return eve::Result<std::string>::failure(trackerValue.status());
359 auto actorValue = eve::Value::fromJson(actorJson.value());
360 if (!actorValue.ok()) return eve::Result<std::string>::failure(actorValue.status());
361 auto inventoryValue = eve::Value::fromJson(inventoryJson.value());
362 if (!inventoryValue.ok()) return eve::Result<std::string>::failure(inventoryValue.status());
363 eve::Value::Object payload;
364 payload.emplace(actor_ ? "actor" : "party", std::move(actorValue).takeValue());
365 payload.emplace("contentVersion", eve::Value(contentVersion_));
366 payload.emplace("gameState", std::move(gameStateValue).takeValue());
367 payload.emplace("inventory", std::move(inventoryValue).takeValue());
368 payload.emplace("questTracker", std::move(trackerValue).takeValue());
369 const eve::SchemaVersion schemaVersion(actor_ ? 1 : 2);
370 auto envelope = eve::makeSnapshotEnvelope("rpg.save-session", saveSchema(), schemaVersion,
371 eve::PersistentId::nil(), eve::Revision(0), eve::SimulationTick(0),
372 eve::Value(std::move(payload)), saveIntegrityProvider());
373 if (!envelope.ok()) return eve::Result<std::string>::failure(envelope.status());
374 return eve::serializeSnapshotEnvelope(envelope.value());
375}

385eve::Result<void> RPGSaveSession::restoreSnapshotJsonImpl(std::string_view json, bool publish) {
386 if (!gameState_ || !tracker_ || (!actor_ && !party_) || !bag_ || !equipment_)
387 return saveFailure<void>(eve::DiagnosticCode::PreconditionViolation,
388 "RPG save session requires bound participants");
389 if (contentVersion_.empty())
390 return saveFailure<void>(eve::DiagnosticCode::PreconditionViolation,
391 "RPG save session requires a content version");
392 auto parsed = eve::parseSnapshotEnvelope(json, saveIntegrityProvider());
393 if (!parsed.ok()) return eve::Result<void>::failure(parsed.status());
394 const eve::SnapshotEnvelope &envelope = parsed.value();
395 if (envelope.type != "rpg.save-session")
396 return saveFailure<void>(eve::DiagnosticCode::InvalidArgument,
397 "snapshot does not belong to RPGSaveSession", "type");
398 if (envelope.schema != saveSchema())
399 return saveFailure<void>(eve::DiagnosticCode::InvalidArgument,
400 "snapshot does not belong to RPGSaveSession", "schema");
401 eve::Value root = envelope.payload;
402 if (!root.isObject())
403 return saveFailure<void>(eve::DiagnosticCode::ParseError, "RPG save session payload must be an object",
404 "payload");
405 eve::Value *contentVersion = root.find("contentVersion");
406 if (!contentVersion || !contentVersion->isString())
407 return saveFailure<void>(eve::DiagnosticCode::ParseError,
408 "RPG save-session content version is missing or invalid",
409 "payload.contentVersion");
410 const std::string sourceContentVersion = contentVersion->asString();
411 const bool migrateSingleActor = party_ && envelope.schemaVersion == eve::SchemaVersion(1) &&
412 singleActorPartyMigrations_.contains(sourceContentVersion);
413 const eve::SchemaVersion expectedVersion(actor_ ? 1 : 2);
414 if (envelope.schemaVersion != expectedVersion && !migrateSingleActor)
415 return saveFailure<void>(eve::DiagnosticCode::UnknownVersion,
416 "unsupported RPG save-session snapshot version", "schemaVersion");
417 if (migrateSingleActor) {
418 const eve::Value *oldActor = root.find("actor");
419 if (!oldActor || !oldActor->isObject())
420 return saveFailure<void>(eve::DiagnosticCode::ParseError,
421 "single-actor migration requires an actor payload", "payload.actor");
422 auto baselineJson = party_->checkpointJson();
423 if (!baselineJson.ok()) return eve::Result<void>::failure(baselineJson.status());
424 auto baseline = eve::Value::fromJson(baselineJson.value());
425 if (!baseline.ok()) return eve::Result<void>::failure(baseline.status());
426 eve::Value migratedParty = std::move(baseline).takeValue();
427 eve::Value *members = migratedParty.find("members");
428 if (!members || !members->isArray() || members->arraySize() == 0)
429 return saveFailure<void>(eve::DiagnosticCode::PreconditionViolation,
430 "single-actor migration requires a non-empty live party", "party");
431 members->at(0).set("actor", *oldActor);
432 root.set("party", std::move(migratedParty));
433 }
434 if (sourceContentVersion != contentVersion_ &&
435 !compatibleContentVersions_.contains(sourceContentVersion)) {
436 bool hasMigration = false;
437 for (const auto &[key, destination] : idRenameMigrations_) {
438 if (key.fromContentVersion != sourceContentVersion) continue;
439 hasMigration = true;
440 applyIdRename(root, key.domain, key.oldId, destination);
441 }
442 const auto questAdditions = questAdditionMigrations_.find(sourceContentVersion);
443 if (questAdditions != questAdditionMigrations_.end()) {
444 hasMigration = true;
445 auto added = applyQuestAdditions(root, questAdditions->second);
446 if (!added.ok()) return added;
447 }
448 if (migrateSingleActor) hasMigration = true;
449 if (!hasMigration)
450 return saveFailure<void>(eve::DiagnosticCode::Conflict,
451 "RPG save-session content version has no compatible or migration route",
452 "payload.contentVersion");
453 root.set("contentVersion", eve::Value(contentVersion_));
454 }
455 const eve::Value *gameStateValue = root.find("gameState");
456 const eve::Value *trackerValue = root.find("questTracker");
457 const eve::Value *actorValue = root.find(actor_ ? "actor" : "party");
458 const eve::Value *inventoryValue = root.find("inventory");
459 if (!gameStateValue || !gameStateValue->isObject() || !trackerValue || !trackerValue->isObject() ||
460 !actorValue || !actorValue->isObject() || !inventoryValue || !inventoryValue->isObject())
461 return saveFailure<void>(eve::DiagnosticCode::ParseError,
462 "RPG save-session participants must be objects", "payload");
463 auto gameStateJson = gameStateValue->toJson();
464 if (!gameStateJson.ok()) return eve::Result<void>::failure(gameStateJson.status());
465 auto trackerJson = trackerValue->toJson();
466 if (!trackerJson.ok()) return eve::Result<void>::failure(trackerJson.status());
467 auto actorJson = actorValue->toJson();
468 if (!actorJson.ok()) return eve::Result<void>::failure(actorJson.status());
469 auto inventoryJson = inventoryValue->toJson();
470 if (!inventoryJson.ok()) return eve::Result<void>::failure(inventoryJson.status());
471
472 GameState candidateGameState = *gameState_;
473 Tracker candidateTracker = *tracker_;
474 auto gameStateRestore = candidateGameState.restoreSnapshotJson(gameStateJson.value());
475 if (!gameStateRestore.ok()) return gameStateRestore;
476 auto trackerRestore = candidateTracker.restoreSnapshotJson(trackerJson.value());
477 if (!trackerRestore.ok()) return trackerRestore;
478 inventory::InventorySaveSession inventorySession;
479 inventorySession.bind(*bag_, *equipment_);
480 auto inventoryCandidate = inventorySession.prepareRestoreSnapshotJson(inventoryJson.value());
481 if (!inventoryCandidate.ok()) return eve::Result<void>::failure(inventoryCandidate.status());
482
483 std::optional<RPGActor::CheckpointCandidate> actorCandidate;
484 std::optional<Party::CheckpointCandidate> partyCandidate;
485 if (actor_) {
