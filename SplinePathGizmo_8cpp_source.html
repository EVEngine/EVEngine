26void addLine(editing::GizmoSnapshot& output, std::string id, const std::array<double, 3>& from,
27 const std::array<double, 3>& to, std::array<double, 4> color, bool dashed = false) {
28 std::array<double, 3> direction{to[0] - from[0], to[1] - from[1], to[2] - from[2]};
29 const double length =
30 std::sqrt(direction[0] * direction[0] + direction[1] * direction[1] + direction[2] * direction[2]);
31 if (length <= 1e-12) return;
32 for (double& value : direction) value /= length;
33 output.primitives.push_back({std::move(id), "line", from, {}, direction, color, 0.0, length, dashed});
34}

36bool normalizeRay(double x, double y, double z, std::array<double, 3>& output) {
37 const double length = std::sqrt(x * x + y * y + z * z);
38 if (!std::isfinite(length) || length < 1e-9) return false;
39 output = {x / length, y / length, z / length};
40 return true;
41}

45editing::GizmoSnapshot SplinePathGizmoBuilder::build(const SplinePathDocument& document, int sampleCount,
46 int maximumSamples, SplineGizmoStyle style) const {
47 if (sampleCount < 2 || maximumSamples < 2 || sampleCount > maximumSamples)
48 return failedSnapshot(document, "editor.spline.gizmo-budget",
49 "Spline gizmo sample count exceeds its bounded budget");
50 const auto validColor = [](const auto& color) {
51 return std::all_of(color.begin(), color.end(),
52 [](double value) { return std::isfinite(value) && value >= 0.0 && value <= 1.0; });
53 };
54 if (!std::isfinite(style.nodeSize) || !std::isfinite(style.handleSize) || style.nodeSize <= 0.0 ||
55 style.handleSize <= 0.0 || !validColor(style.lineColor) || !validColor(style.textColor))
56 return failedSnapshot(document, "editor.spline.gizmo-style-invalid",
57 "Spline gizmo sizes must be positive and colors must be finite normalized values");
58 editing::GizmoSnapshot result;
59 result.target = document.targetId().value();
60 result.targetRevision = document.revision();
61 for (const auto& point : document.points()) {
62 const auto anchor = position(point);
63 result.primitives.push_back(
64 {"spline.anchor." + point.id.value(), "sphere", anchor, {}, {}, {1.0, 0.72, 0.18, 1.0}, style.nodeSize});
65 if (style.showPointLabels) {
66 editing::GizmoPrimitive label;
67 label.id = "spline.label." + point.id.value();
68 label.kind = "text";
69 label.position = anchor;
70 label.color = style.textColor;
71 label.text = point.id.value();
72 result.primitives.push_back(std::move(label));
73 }
74 if (document.settings().kind != "bezier") continue;
75 const std::array<double, 3> incoming{point.x + point.inX, point.y + point.inY, point.z + point.inZ};
76 const std::array<double, 3> outgoing{point.x + point.outX, point.y + point.outY, point.z + point.outZ};
77 addLine(result, "spline.in-line." + point.id.value(), anchor, incoming, {0.95, 0.4, 0.3, 0.9}, true);
78 addLine(result, "spline.out-line." + point.id.value(), anchor, outgoing, {0.35, 0.9, 0.45, 0.9}, true);
79 result.primitives.push_back(
80 {"spline.in." + point.id.value(), "sphere", incoming, {}, {}, {0.95, 0.4, 0.3, 1.0}, style.handleSize});
81 result.primitives.push_back(
82 {"spline.out." + point.id.value(), "sphere", outgoing, {}, {}, {0.35, 0.9, 0.45, 1.0}, style.handleSize});
83 }
84 auto path = document.compilePath();
85 if (!path.ok()) {
86 if (document.points().size() >= 2)
87 return failedSnapshot(document, "editor.spline.gizmo-incomplete", path.status().describe());
88 result.diagnostics.push_back(editing::ruleDiagnostic(
89 DiagnosticCode::PreconditionViolation, RuleId("editor.spline.gizmo-incomplete"),
90 DiagnosticSeverity::Warning, "Add at least two points to preview the spline curve"));
91 result.status = EditorStatus::Applied;
92 return result;
93 }
94 auto polyline = path.value().polylineResult(sampleCount, false);
95 if (!polyline.ok()) return failedSnapshot(document, "editor.spline.gizmo-evaluate", polyline.status().describe());
96 for (int chunk = 0; chunk < polyline.value().chunkCount(); ++chunk) {
97 auto previous = polyline.value().chunkPointResult(chunk, 0);
98 if (!previous.ok())
99 return failedSnapshot(document, "editor.spline.gizmo-evaluate", previous.status().describe());
100 for (int sample = 1; sample < polyline.value().chunkPointCount(chunk); ++sample) {
101 auto current = polyline.value().chunkPointResult(chunk, sample);
102 if (!current.ok())
103 return failedSnapshot(document, "editor.spline.gizmo-evaluate", current.status().describe());
104 addLine(result, "spline.chunk." + std::to_string(chunk) + ".segment." + std::to_string(sample - 1),
105 {previous.value().x, previous.value().y, previous.value().z},
106 {current.value().x, current.value().y, current.value().z}, style.lineColor);
107 previous = std::move(current);
108 }
109 }
110 result.status = EditorStatus::Applied;
111 return result;
112}

136EditorResult<SplinePathDragPreview> SplinePathDragSession::beginDrag(double rayOriginX, double rayOriginY,
137 double rayOriginZ, double rayDirectionX,
138 double rayDirectionY, double rayDirectionZ) {
139 cancelDrag();
140 if (!document_)
141 return gizmoError<SplinePathDragPreview>(EditorStatus::Rejected, "editor.spline.drag-target-required",
142 "Spline drag target is unavailable");
143 std::array<double, 3> direction;
144 if (!normalizeRay(rayDirectionX, rayDirectionY, rayDirectionZ, direction))
145 return gizmoError<SplinePathDragPreview>(EditorStatus::Rejected, "editor.spline.drag-ray-invalid",
146 "Spline drag requires a finite non-zero pointer ray");
147 const std::array<double, 3> origin{rayOriginX, rayOriginY, rayOriginZ};
148 const auto gizmo = SplinePathGizmoBuilder().build(*document_, sampleCount_);
149 if (gizmo.status != EditorStatus::Applied)
150 return gizmoError<SplinePathDragPreview>(EditorStatus::Rejected, "editor.spline.drag-gizmo-invalid",
151 "Spline gizmo is not available for picking");
152 double closest = std::numeric_limits<double>::infinity();
153 std::string pickedId;
154 for (const auto& primitive : gizmo.primitives) {
155 if (primitive.kind != "sphere" || primitive.id.rfind("spline.", 0) != 0) continue;
156 const std::array<double, 3> offset{origin[0] - primitive.position[0], origin[1] - primitive.position[1],
157 origin[2] - primitive.position[2]};
158 const double b = offset[0] * direction[0] + offset[1] * direction[1] + offset[2] * direction[2];
159 const double c =
160 offset[0] * offset[0] + offset[1] * offset[1] + offset[2] * offset[2] - primitive.radius * primitive.radius;
161 const double discriminant = b * b - c;
162 if (discriminant < 0.0) continue;
163 const double nearHit = -b - std::sqrt(discriminant);
164 const double farHit = -b + std::sqrt(discriminant);
165 const double hit = nearHit >= 0.0 ? nearHit : farHit;
166 if (hit < 0.0 || hit >= closest) continue;
167 closest = hit;
168 pickedId = primitive.id;
169 dragPlanePoint_ = primitive.position;
170 }
171 if (pickedId.empty())
172 return gizmoError<SplinePathDragPreview>(EditorStatus::NotFound, "editor.spline.handle-not-hit",
173 "Pointer ray did not hit a spline handle");
174 const auto roleEnd = pickedId.find('.', 7);
175 activeHandle_ = pickedId.substr(7, roleEnd - 7);
176 activePoint_ = StableId(pickedId.substr(roleEnd + 1));
177 const auto points = document_->points();
178 const auto found =
179 std::find_if(points.begin(), points.end(), [&](const auto& point) { return point.id == activePoint_; });
180 if (found == points.end())
181 return gizmoError<SplinePathDragPreview>(EditorStatus::NotFound, "editor.spline.point-not-found",
182 "Picked spline point no longer exists");
183 draft_ = *found;
184 dragPlaneNormal_ = direction;
185 baseRevision_ = document_->revision();
186 dragging_ = true;
187 return previewCurrent();
188}

190EditorResult<SplinePathDragPreview> SplinePathDragSession::updateDrag(double rayOriginX, double rayOriginY,
191 double rayOriginZ, double rayDirectionX,
192 double rayDirectionY, double rayDirectionZ) {
193 if (!dragging_ || !document_)
194 return gizmoError<SplinePathDragPreview>(EditorStatus::Rejected, "editor.spline.drag-not-active",
195 "Spline drag has not begun");
196 if (document_->revision() != baseRevision_) {
197 cancelDrag();
198 return gizmoError<SplinePathDragPreview>(EditorStatus::Conflict, "editor.spline.drag-stale",
199 "Spline document changed while dragging");
200 }
201 std::array<double, 3> direction;
202 if (!normalizeRay(rayDirectionX, rayDirectionY, rayDirectionZ, direction))
203 return gizmoError<SplinePathDragPreview>(EditorStatus::Rejected, "editor.spline.drag-ray-invalid",
204 "Spline drag requires a finite non-zero pointer ray");
205 const std::array<double, 3> origin{rayOriginX, rayOriginY, rayOriginZ};
206 const double denominator =
207 direction[0] * dragPlaneNormal_[0] + direction[1] * dragPlaneNormal_[1] + direction[2] * dragPlaneNormal_[2];
208 if (std::abs(denominator) < 1e-9)
209 return gizmoError<SplinePathDragPreview>(EditorStatus::Rejected, "editor.spline.drag-ray-parallel",
210 "Pointer ray is parallel to the spline drag plane");
211 const double distance = ((dragPlanePoint_[0] - origin[0]) * dragPlaneNormal_[0] +
212 (dragPlanePoint_[1] - origin[1]) * dragPlaneNormal_[1] +
213 (dragPlanePoint_[2] - origin[2]) * dragPlaneNormal_[2]) /
214 denominator;
215 if (distance < 0.0)
216 return gizmoError<SplinePathDragPreview>(EditorStatus::Rejected, "editor.spline.drag-behind-camera",
217 "Spline drag plane is behind the pointer ray origin");
218 const std::array<double, 3> world{origin[0] + direction[0] * distance, origin[1] + direction[1] * distance,
219 origin[2] + direction[2] * distance};
220 if (activeHandle_ == "anchor") {
221 draft_.x = world[0];
222 draft_.y = world[1];
223 draft_.z = world[2];
224 } else if (activeHandle_ == "in") {
225 draft_.inX = world[0] - draft_.x;
226 draft_.inY = world[1] - draft_.y;
227 draft_.inZ = world[2] - draft_.z;
228 } else {
229 draft_.outX = world[0] - draft_.x;
230 draft_.outY = world[1] - draft_.y;
231 draft_.outZ = world[2] - draft_.z;
232 }
233 return previewCurrent();
234}
