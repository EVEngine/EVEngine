25Result<void> writeExclusiveAndFlush(const std::filesystem::path& path,
26 std::span<const std::uint8_t> bytes) {
27#if defined(_WIN32)
28 HANDLE file = CreateFileW(path.c_str(), GENERIC_WRITE, 0, nullptr, CREATE_NEW,
29 FILE_ATTRIBUTE_NORMAL | FILE_FLAG_WRITE_THROUGH, nullptr);
30 if (file == INVALID_HANDLE_VALUE)
31 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Failed, "cannot create temporary package",
32 path.string(), {}, "asset.package.store"));
33 std::size_t cursor = 0;
34 bool written = true;
35 while (cursor < bytes.size()) {
36 const DWORD request = static_cast<DWORD>(std::min<std::size_t>(bytes.size() - cursor, MAXDWORD));
37 DWORD count = 0;
38 if (!WriteFile(file, bytes.data() + cursor, request, &count, nullptr) || count != request) {
39 written = false;
40 break;
41 }
42 cursor += count;
43 }
44 const bool flushed = written && FlushFileBuffers(file) != 0;
45 CloseHandle(file);
46 if (!flushed)
47 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Failed, "cannot flush temporary package",
48 path.string(), {}, "asset.package.store"));
49#else
50 const int file = ::open(path.c_str(), O_WRONLY | O_CREAT | O_EXCL | O_CLOEXEC, 0600);
51 if (file < 0)
52 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Failed, "cannot create temporary package",
53 path.string(), {}, "asset.package.store"));
54 std::size_t cursor = 0;
55 bool written = true;
56 while (cursor < bytes.size()) {
57 const ssize_t count = ::write(file, bytes.data() + cursor, bytes.size() - cursor);
58 if (count < 0 && errno == EINTR) continue;
59 if (count <= 0) { written = false; break; }
60 cursor += static_cast<std::size_t>(count);
61 }
62 const bool flushed = written && ::fsync(file) == 0;
63 ::close(file);
64 if (!flushed)
65 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Failed, "cannot flush temporary package",
66 path.string(), {}, "asset.package.store"));
67#endif
68 return Result<void>::success();
69}

71Result<std::vector<std::uint8_t>> readBounded(const std::filesystem::path& path, std::uint64_t maximumBytes) {
72 std::error_code ec;
73 const auto size = std::filesystem::file_size(path, ec);
74 if (ec || size > maximumBytes || size > (std::numeric_limits<std::size_t>::max)())
75 return storeFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
76 "temporary package size is outside limits", path);
77 std::ifstream input(path, std::ios::binary);
78 if (!input)
79 return storeFailure<std::vector<std::uint8_t>>(DiagnosticCode::Failed,
80 "cannot reopen temporary package", path);
81 std::vector<std::uint8_t> bytes(static_cast<std::size_t>(size));
82 if (!bytes.empty()) input.read(reinterpret_cast<char*>(bytes.data()), static_cast<std::streamsize>(bytes.size()));
83 if (!input || input.peek() != std::ifstream::traits_type::eof())
84 return storeFailure<std::vector<std::uint8_t>>(DiagnosticCode::Failed,
85 "temporary package changed while reading", path);
86 return Result<std::vector<std::uint8_t>>::success(std::move(bytes));
87}

89Result<void> replaceFile(const std::filesystem::path& temporary,

91#if defined(_WIN32)

95 "atomic package replacement failed",

138Result<PackagePublishReceipt> AtomicAssetPackageStore::publishEva(
139 const std::filesystem::path& destination, const EvaManifest& manifest,
140 std::vector<EvaArchiveEntry> entries, const EvaArchiveLimits& limits) {
141 if (destination.empty() || destination.filename().empty())
142 return storeFailure<PackagePublishReceipt>(DiagnosticCode::InvalidArgument,
143 "package destination is empty", destination);
144 auto built = buildEvaArchive(manifest, std::move(entries), limits);
145 if (!built) return Result<PackagePublishReceipt>::failure(built.status());
146 std::error_code ec;
147 if (!destination.parent_path().empty()) std::filesystem::create_directories(destination.parent_path(), ec);
148 if (ec) return storeFailure<PackagePublishReceipt>(DiagnosticCode::Failed, ec.message(), destination);
149 const auto temporary = temporaryPath(destination, ++sequence_);
150 TemporaryCleanup cleanup(temporary);
151 auto written = writeExclusiveAndFlush(temporary, built.value());
152 if (!written) return Result<PackagePublishReceipt>::failure(written.status());
153 auto reopened = readBounded(temporary, limits.maximumArchiveBytes);
154 if (!reopened) return Result<PackagePublishReceipt>::failure(reopened.status());
155 auto verified = parseEvaArchive(reopened.value(), limits);
156 if (!verified) return Result<PackagePublishReceipt>::failure(verified.status());
157 if (beforeReplace_ && beforeReplace_(temporary, destination) == PackagePublishGateDecision::Reject)
158 return storeFailure<PackagePublishReceipt>(DiagnosticCode::Cancelled,
159 "package replacement rejected before commit", destination);
160 auto replaced = replaceFile(temporary, destination);
161 if (!replaced) return Result<PackagePublishReceipt>::failure(replaced.status());
162 cleanup.release();
163 return Result<PackagePublishReceipt>::success(
164 {destination, verified.value().manifest.packageId, std::nullopt, reopened.value().size()});
165}
