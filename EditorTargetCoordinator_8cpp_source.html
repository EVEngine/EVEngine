56 EditorResult<TransactionReceipt> execute(const CommandPlan& plan) {
57 TargetEntry* selected = entry(plan.target);
58 if (!selected || !selected->target)
59 return coordinatorError<TransactionReceipt>(EditorStatus::NotFound, "editor.target.not-found",
60 "Editing target is not registered: " + plan.target.value());
61 if (selected->target->revision() != plan.baseRevision)
62 return coordinatorError<TransactionReceipt>(EditorStatus::Conflict, "editor.target.revision-conflict",
63 "Editing target changed after planning");
64 if (selected->generation != plan.targetGeneration)
65 return coordinatorError<TransactionReceipt>(EditorStatus::Conflict, "editor.target.generation-conflict",
66 "Editing target lifetime changed after planning");
67 TransactionSpec specification;
68 specification.id = TransactionId(plan.id.value() + ".transaction");
69 specification.label = plan.command.value();
70 specification.origin = ActionOrigin::Automation;
71 specification.target = plan.target;
72 specification.baseRevision = plan.baseRevision;
73 auto begun = selected->transactions.begin(std::move(specification));
74 if (!begun.ok())
75 return coordinatorError<TransactionReceipt>(begun.code(), "editor.target.begin-failed",
76 "Could not begin the editing transaction");
77 for (const DomainOperation& operation : plan.operations) {
78 auto appended = selected->transactions.append(operation);
79 if (appended.ok()) continue;
80 auto discarded = selected->transactions.discard();
81 if (!discarded.ok())
82 return coordinatorError<TransactionReceipt>(EditorStatus::Failed, "editor.target.discard-failed",
83 "Could not discard a rejected transaction");
84 return coordinatorError<TransactionReceipt>(appended.code(), "editor.target.append-failed",
85 "Could not stage the editing operation");
86 }
87 return selected->transactions.commit();
88 }

113eve::editing::Result<void> EditorTargetCoordinator::registerPlannedCommand(
114 eve::editing::EditingCommandDescriptor descriptor, eve::editing::EditingCommandPlanner planner) {
115 const CommandId commandId = descriptor.id;
116 const std::string ownerModule = descriptor.ownerModule;
117 CommandDescriptor hostDescriptor;
118 hostDescriptor.id = descriptor.id;
119 hostDescriptor.ownerModule = std::move(descriptor.ownerModule);
120 hostDescriptor.displayName = std::move(descriptor.displayName);
121 hostDescriptor.category = std::move(descriptor.category);
122 hostDescriptor.automationAllowed = descriptor.automationAllowed;
123 auto registered = impl_->commands->registerPlannedCommand(
124 std::move(hostDescriptor),
125 [state = impl_.get(), planner = std::move(planner)](const CommandRequest& request) {
126 Impl::TargetEntry* selected = state->entry(request.context.target);
127 if (!selected || !selected->target)
128 return coordinatorError<CommandPlan>(EditorStatus::NotFound, "editor.target.not-found",
129 "Editing target is not registered");
130 EditorResult<CommandPlan> result = planner(*selected->target, request);
131 if (result.ok()) result.value().targetGeneration = selected->generation;
132 return result;
133 },
134 [state = impl_.get()](const CommandRequest&, const CommandPlan& plan) { return state->execute(plan); });
135 if (!registered.ok()) return EditorResult<void>::failure(registered.status());
136 impl_->registeredCommands.emplace_back(commandId, ownerModule);
137 return eve::editing::applied<void>(registered.diagnostics());
138}

148EditorResult<void> EditorTargetCoordinator::registerTarget(IEditableTarget& target) {
149 auto* operations = dynamic_cast<IDomainOperationTarget*>(&target);
150 if (!operations)
151 return coordinatorError<void>(EditorStatus::Unsupported, "editor.target.operations-unsupported",
152 "Editing target does not support domain operations");
153 const TargetId id(target.targetId());
154 if (id.empty())
155 return coordinatorError<void>(EditorStatus::Rejected, "editor.target.empty-id",
156 "Editing target id must not be empty");
157 if (const auto found = impl_->targets.find(id); found != impl_->targets.end()) {
158 if (found->second->target == &target) {
159 return eve::editing::noOp();
160 }
161 return coordinatorError<void>(EditorStatus::Conflict, "editor.target.duplicate-id",
162 "Another editing target already uses id: " + id.value());
163 }
164 impl_->targets.emplace(id,
165 std::make_unique<Impl::TargetEntry>(target, *operations, impl_->nextTargetGeneration++));
166 return eve::editing::applied<void>();
167}

169EditorResult<void> EditorTargetCoordinator::unregisterTarget(const TargetId& target) {
170 Impl::TargetEntry* selected = impl_->entry(target);
171 if (selected) {
172 std::vector<EditorSession*> sessions;
173 for (const auto& [session, binding] : impl_->sessions)
174 if (binding.target == target && binding.generation == selected->generation) sessions.push_back(session);
175 for (EditorSession* session : sessions)
176 if (session) session->invalidateTrackedTarget(*this, target, selected->generation);
177 impl_->targets.erase(target);
178 return eve::editing::applied<void>();
179 }
180 return eve::editing::noOp();
181}
