36 if (!finite(forward) || !std::isfinite(strength)) return invalid<void>();

45Result<void> advanceVegetationWind(VegetationWindState& state, glm::vec3 forward, float strength, float dt) {
46 if (!valid(state) || !finite(forward) || !std::isfinite(strength) || !std::isfinite(dt) || dt < 0)
47 return invalid<void>();
48 const double alpha = saturate(double(dt) * 0.25);
49 const glm::dvec3 target{
50 forward.x, -std::max(double(forward.y), std::hypot(double(forward.x), double(forward.z)) * 0.5), forward.z};
51 const auto direction = glm::mix(glm::dvec3(state.direction), target, alpha);
52 const double main =
53 double(state.strength) + (std::clamp(double(strength), 0.0, double(1.2F)) - state.strength) * alpha;
54 double phase = double(state.updatePhase) + double(dt) * std::pow(main * 0.5 + 0.5, 3) * 0.1;
55 if (phase > 100) phase -= 100;
56 if (!representable(direction) || !std::isfinite(phase) || phase > std::numeric_limits<float>::max())
57 return invalid<void>();
58 state = {glm::vec3(direction), float(main), float(phase), float(phase)};
59 return Result<void>::success();
60}

88Result<glm::vec3> evaluateVegetationWind(const VegetationWindInput& in, const VegetationWindState& state) {
89 if (!valid(state) || !finite(in.position) || !finite(in.worldOffset) || !finite(in.cameraPosition) ||
90 !finite(in.flex) || !finite(in.frequency) || !std::isfinite(in.widthHeight.x) || in.widthHeight.x <= 0 ||
91 !std::isfinite(in.widthHeight.y) || in.widthHeight.y <= 0 || !std::isfinite(in.maximumDistance) ||
92 in.maximumDistance <= 0 || !std::isfinite(in.sinTimeQuarter) || std::abs(in.sinTimeQuarter) > 1 ||
93 !std::isfinite(in.sinTimeFull) || std::abs(in.sinTimeFull) > 1)
94 return invalid<glm::vec3>();
95 glm::dmat3 matrix(in.objectToWorld);
96 for (int c = 0; c < 3; ++c)
97 if (!finite(matrix[c])) return invalid<glm::vec3>();
98 const double determinant = glm::determinant(matrix);
99 if (!std::isfinite(determinant) || determinant == 0) return invalid<glm::vec3>();
100 if (!in.enabled) return Result<glm::vec3>::success(in.position);
101 const auto distance = (glm::dvec3(in.worldOffset) - glm::dvec3(in.cameraPosition)) / double(in.maximumDistance);
102 double attenuation = 1 - saturate(glm::dot(distance, distance));
103 attenuation *= attenuation;
104 if (attenuation == 0 || state.strength == 0) return Result<glm::vec3>::success(in.position);
105 const double main = state.strength;
106 glm::dvec3 flex = glm::dvec3(in.flex) * glm::dvec3(saturate(main * 3), saturate(main * 2), 1 - main * main * 0.5) *
107 main * attenuation;
108 const auto up = matrix * glm::dvec3(0, 1, 0);
109 const double length = glm::length(up), scale = std::max(length, 0.4), upDot = saturate(up.y / length);
110 const auto dimensions =
111 glm::mix(glm::dvec2(double(in.widthHeight.y) * 2), glm::dvec2(in.widthHeight), upDot * upDot) * scale;
112 flex *= scale;
113 const auto frequency = glm::dvec3(in.frequency) * scale;
114 auto local = matrix * glm::dvec3(in.position);
115 const auto world = local + glm::dvec3(in.worldOffset);
116 const double phase = double(state.phase) * 6, time = -(phase - std::floor(phase)) * 6.283185;
117 const auto norm = local / glm::dvec3(dimensions.x, dimensions.y, dimensions.x);
118 const double branch = norm.x * norm.x + norm.z * norm.z, stem = norm.y, lengthA = glm::dot(local, local);
119 double gust =
120 ((std::sin(time + frequency.x * (double(in.worldOffset.x) + in.worldOffset.y + in.worldOffset.z)) * 0.3 +
121 main * 0.5) +
122 (double(in.sinTimeQuarter) * 0.4 + main) * main) *
123 (double(in.sinTimeFull) * 0.3 + 0.7);
124 const auto direction = glm::dvec3(state.direction);
125 glm::dvec3 tally{direction.x * stem * stem * gust * flex.x, 0, direction.z * stem * stem * gust * flex.x};
126 gust = gust * 0.7 + 0.3;
127 if (!in.billboard) {
128 const auto displaced = world + tally * 0.25;
129 tally += direction * stem * stem *
130 (std::sin(time * 2 + (displaced.x + displaced.y + displaced.z) * frequency.y) * branch * 0.7 + 0.3) *
131 gust * flex.y;
132 }
133 const auto offset = local + tally;
134 const double denominator = glm::dot(offset, offset);
135 const double normalization = denominator == 0 ? 0 : saturate(lengthA / denominator);
136 tally = offset * normalization;
137 local = tally;
138 if (!in.billboard && in.alphaTest) {
139 const auto offsets = (world + tally) * frequency.z;
140 glm::dvec3 wave{std::sin(time * 5 + offsets.x), std::sin(time * 5 + offsets.y), std::sin(time * 5 + offsets.z)};
141 local += (wave * direction + direction) * glm::dvec3(branch, branch * 0.75, branch) * (stem * gust * flex.z) *
142 (normalization + 0.5);
143 }
144 const auto result = glm::inverse(matrix) * local;
145 if (!representable(result)) return invalid<glm::vec3>();
146 return Result<glm::vec3>::success(glm::vec3(result));
147}
