75 void buildCell(HexCoordinates coordinates) {
76 if (!map_.contains(coordinates)) return;
77 const HexCellData* cellData = map_.cell(coordinates);
78 if (cellData == nullptr) return;
79
80 const HexVec3 center = map_.cellPosition(coordinates);
81 for (std::int32_t index = 0; index < kHexDirectionCount; ++index) {
82 const auto direction = static_cast<HexDirection>(index);
83
84 EdgeVertices near = solidEdge(center, direction);
85 if (riverThrough(coordinates, direction, cellData)) {
86 // Carve a shallow channel along the river centreline (the middle sample only).
87 near.v3.y = HexMetrics::streamBedY(cellData->values.elevation());
88 }
89 appendEdgeFan(center, near, cellData);
90

92 if (!map_.getNeighbor(coordinates, direction, neighbourCoordinates)) continue;
93 const HexCellData* neighbour = map_.cell(neighbourCoordinates);
94 if (neighbour == nullptr) continue;
95 appendConnection(coordinates, center, near, direction, cellData, neighbour, neighbourCoordinates);

100 [[nodiscard]] bool riverThrough(HexCoordinates coordinates, HexDirection direction,
101 const HexCellData* cellData) const noexcept {
102 if (cellData != nullptr && cellData->flags.hasRiverThrough(direction)) return true;
103 HexCoordinates neighbour{};
104 if (!map_.getNeighbor(coordinates, direction, neighbour)) return false;
105 const HexCellData* other = map_.cell(neighbour);
106 return other != nullptr && other->flags.hasRiverThrough(opposite(direction));
107 }

110 void appendConnection(HexCoordinates coordinates, HexVec3 center, const EdgeVertices& near, HexDirection direction,
111 const HexCellData* cellData, const HexCellData* neighbour,
112 HexCoordinates neighbourCoordinates) {
113 // Chunk ownership rule: only the NE, E and SE edges of a cell belong to the
114 // owning chunk, so every shared boundary is emitted exactly once - by the cell
115 // that owns that edge, never by both. The reference gates its whole
116 // `TriangulateConnection` call the same way. Emitting the strip from the other
117 // side too puts two coplanar (or, on a slope, two interpenetrating) copies on
118 // every boundary, which is what makes the terrain look like it has seams.
119 //
120 // The gate has to come before the strip is built: it also covers the corner
121 // below, which is why the corner matrix only ever runs for these three
122 // directions.
123 if (static_cast<std::int32_t>(direction) > static_cast<std::int32_t>(HexDirection::SE)) return;
124
125 const EdgeVertices far = shiftedEdge(center, direction, map_.cellPosition(neighbourCoordinates).y);
126
127 const HexTerrainWeights nearWeights = HexTerrainWeights::primary();
128 const HexTerrainWeights farWeights = HexTerrainWeights::blend(1.f);
129
130 switch (map_.edgeTypeTo(coordinates, neighbourCoordinates)) {
131 case HexEdgeType::Flat: appendBlendStrip(near, far, nearWeights, farWeights, cellData, neighbour); break;
132 case HexEdgeType::Slope: appendEdgeTerraces(near, nearWeights, far, farWeights, cellData, neighbour); break;
133 case HexEdgeType::Cliff:
134 // The blend strip would already be a vertical wall here, but its winding flips with the
135 // sign of the elevation step; the vertical winding is correct in both directions.
136 emitQuadVertical(near.v1, near.v5, far.v1, far.v5, nearWeights, nearWeights, farWeights, farWeights,
137 cellData, neighbour, cellData);
138 break;
139 }
140
141 HexCoordinates nextCoordinates{};
142 if (!map_.getNeighbor(coordinates, next(direction), nextCoordinates)) return;
143 const HexCellData* nextCell = map_.cell(nextCoordinates);
144
145 CornerCells corner{};
146 corner.up = near.v5;
147 // Third corner vertex: this cell's corner carried across by the bridge of the
148 // *next* edge, exactly as the reference builds `e1.v5 + GetBridge(d.Next())`.
149 corner.left = near.v5 + HexMetrics::bridge(next(direction));
150 corner.left.y = map_.cellPosition(nextCoordinates).y;
151 corner.right = far.v5;
152 corner.upCell = cellData;
153 corner.leftCell = neighbour;
154 corner.rightCell = nextCell;
155 appendCorner(corner);
156 }

161 void appendCorner(const CornerCells& corner) {
162 const std::array<HexVec3, 3> positions{corner.up, corner.left, corner.right};
163 const std::array<const HexCellData*, 3> cells{corner.upCell, corner.leftCell, corner.rightCell};
164 std::array<std::int32_t, 3> order{0, 1, 2};
165 for (std::int32_t i = 0; i < 2; ++i) {
166 for (std::int32_t j = i + 1; j < 3; ++j) {
167 if (elevationOf(cells[static_cast<std::size_t>(order[static_cast<std::size_t>(j)])]) <
168 elevationOf(cells[static_cast<std::size_t>(order[static_cast<std::size_t>(i)])])) {
169 std::swap(order[static_cast<std::size_t>(i)], order[static_cast<std::size_t>(j)]);
170 }
171 }
172 }
173
174 const HexVec3 bottom = positions[static_cast<std::size_t>(order[0])];
175 const HexVec3 low = positions[static_cast<std::size_t>(order[1])];
176 const HexVec3 high = positions[static_cast<std::size_t>(order[2])];
177 const HexCellData* bottomCell = cells[static_cast<std::size_t>(order[0])];
178 const HexCellData* lowCell = cells[static_cast<std::size_t>(order[1])];
179 const HexCellData* highCell = cells[static_cast<std::size_t>(order[2])];
180
181 const HexEdgeType lowEdge = edgeType(elevationOf(bottomCell), elevationOf(lowCell));
182 const HexEdgeType highEdge = edgeType(elevationOf(bottomCell), elevationOf(highCell));
183
184 if (lowEdge == HexEdgeType::Slope && highEdge == HexEdgeType::Slope) {
185 cornerTerraces(bottom, low, high, bottomCell, lowCell, highCell);
186 } else if (lowEdge == HexEdgeType::Slope && highEdge == HexEdgeType::Flat) {
187 cornerTerraces(low, high, bottom, lowCell, highCell, bottomCell);
188 } else if (lowEdge == HexEdgeType::Flat && highEdge == HexEdgeType::Slope) {
189 cornerTerraces(high, bottom, low, highCell, bottomCell, lowCell);
190 } else if (lowEdge == HexEdgeType::Slope && highEdge == HexEdgeType::Cliff) {
191 cornerTerracesCliff(bottom, low, high, bottomCell, lowCell, highCell, lowEdge);
192 } else if (lowEdge == HexEdgeType::Cliff && highEdge == HexEdgeType::Slope) {
193 cornerCliffTerraces(bottom, low, high, bottomCell, lowCell, highCell, highEdge);
194 } else if (edgeType(elevationOf(lowCell), elevationOf(highCell)) == HexEdgeType::Slope) {
195 if (elevationOf(lowCell) < elevationOf(highCell)) {
196 cornerCliffTerraces(high, bottom, low, highCell, bottomCell, lowCell, highEdge);
197 } else {
198 cornerTerracesCliff(low, high, bottom, lowCell, highCell, bottomCell, lowEdge);
199 }
200 } else {
201 // Mirror of the reference winding: this engine's front face is the opposite
202 // handedness, so the flat corner fan is emitted high-before-low.
203 emitTriangle(bottom, high, low, HexTerrainWeights::primary(), bottomCell, lowCell, highCell);
204 }

208 void cornerTerraces(const HexVec3& bottom, const HexVec3& left, const HexVec3& right, const HexCellData* bottomCell,
209 const HexCellData* leftCell, const HexCellData* rightCell) {
210 const HexTerrainWeights bottomWeights = HexTerrainWeights::primary();
211 const HexTerrainWeights leftWeights = HexTerrainWeights::primary();
212 const HexTerrainWeights rightWeights = HexTerrainWeights::primary();
213
214 HexVec3 lastLeft = bottom;
215 HexVec3 lastRight = bottom;
216 HexTerrainWeights lastLeftWeights = bottomWeights;
217 HexTerrainWeights lastRightWeights = bottomWeights;
218 for (std::int32_t step = 1; step < HexMetrics::kTerracesPerSlope * 2; ++step) {
219 const float t = static_cast<float>(step) * HexMetrics::kHorizontalTerraceStepSize;
220 const HexTerrainWeights wl = HexTerrainWeights::lerp(bottomWeights, leftWeights, t);
221 const HexTerrainWeights wr = HexTerrainWeights::lerp(bottomWeights, rightWeights, t);
222 const HexVec3 boundaryLeft = HexMetrics::terraceLerp(bottom, left, step);
223 const HexVec3 boundaryRight = HexMetrics::terraceLerp(bottom, right, step);
224
225 emitQuadForward(boundaryLeft, boundaryRight, lastLeft, lastRight, wl, wr, lastLeftWeights, lastRightWeights,
226 bottomCell, leftCell, rightCell);
227 lastLeft = boundaryLeft;
228 lastRight = boundaryRight;
229 lastLeftWeights = wl;
230 lastRightWeights = wr;
231 }
232 emitQuadForward(lastLeft, lastRight, left, right, lastLeftWeights, lastRightWeights, leftWeights, rightWeights,
233 bottomCell, leftCell, rightCell);
234 }

242 void appendBoundaryTriangle(HexVec3 bottom, const HexTerrainWeights& bottomWeights, HexVec3 left,
243 const HexTerrainWeights& leftWeights, HexVec3 right,
244 const HexTerrainWeights& rightWeights, const HexCellData* bottomCell,
245 const HexCellData* leftCell, const HexCellData* rightCell) {
246 HexVec3 leftSide = HexMetrics::terraceLerp(bottom, left, 1);
247 HexVec3 rightSide = HexMetrics::terraceLerp(bottom, right, 1);
248 for (std::int32_t step = 1; step < HexMetrics::kTerracesPerSlope * 2; ++step) {
249 const float t = static_cast<float>(step) * HexMetrics::kHorizontalTerraceStepSize;
250 const HexTerrainWeights wl = HexTerrainWeights::lerp(bottomWeights, leftWeights, t);
251 const HexTerrainWeights wr = HexTerrainWeights::lerp(bottomWeights, rightWeights, t);
252
253 emitTriangle(leftSide, rightSide, bottom, wl, wr, bottomWeights, bottomCell, leftCell, rightCell);
254
255 leftSide = rightSide;
256 rightSide = HexMetrics::terraceLerp(bottom, right, step + 1);
257 }
258 emitTriangle(leftSide, rightSide, bottom, leftWeights, rightWeights, bottomWeights, bottomCell, leftCell,
259 rightCell);

263 void cornerTerracesCliff(const HexVec3& bottom, const HexVec3& low, const HexVec3& high,
264 const HexCellData* bottomCell, const HexCellData* lowCell, const HexCellData* highCell,
265 HexEdgeType lowEdge) {
266 const HexTerrainWeights bottomWeights = HexTerrainWeights::primary();
267 const HexTerrainWeights lowWeights = HexTerrainWeights::primary();
268 const HexTerrainWeights highWeights = HexTerrainWeights::primary();
269
270 const float span = static_cast<float>(elevationOf(highCell) - elevationOf(bottomCell));
271 const float blend = span > 1e-6f ? 1.f / span : 1.f;
272 const HexVec3 boundary = lerp(bottom, high, blend);
273 const HexTerrainWeights boundaryWeights = HexTerrainWeights::lerp(bottomWeights, highWeights, blend);

400 void appendBlendStrip(const EdgeVertices& near, const EdgeVertices& far, const HexTerrainWeights& nearWeights,
401 const HexTerrainWeights& farWeights, const HexCellData* nearCell,
402 const HexCellData* farCell) {
403 const HexVec3 nearPositions[5] = {near.v1, near.v2, near.v3, near.v4, near.v5};
404 const HexVec3 farPositions[5] = {far.v1, far.v2, far.v3, far.v4, far.v5};
405 for (std::int32_t i = 0; i < 4; ++i) {
406 const float t0 = static_cast<float>(i) * 0.25f;
407 const float t1 = static_cast<float>(i + 1) * 0.25f;
408 const HexTerrainWeights w0 = HexTerrainWeights::lerp(nearWeights, farWeights, t0);
409 const HexTerrainWeights w1 = HexTerrainWeights::lerp(nearWeights, farWeights, t1);
410 // Both vertices of an edge share their parameter: the near pair carries w0
411 // and the far pair carries w1. Swapping them makes every quad blend along the
412 // wrong axis and turns the band into a sawtooth.
413 emitQuadForward(nearPositions[i], nearPositions[i + 1], farPositions[i], farPositions[i + 1], w0, w0, w1,
414 w1, nearCell, farCell, nearCell);
415 }
416 }

421 const HexCellData* farCell) {
422 for (std::int32_t step = 1; step < HexMetrics::kTerracesPerSlope * 2; ++step) {

426 emitQuadForward(near.v1, near.v5, middle.v1, middle.v5, nearWeights, nearWeights, w1, w1, nearCell, farCell,
427 nearCell);
428 near = middle;
429 }
430 emitQuadForward(near.v1, near.v5, far.v1, far.v5, nearWeights, nearWeights, farWeights, farWeights, nearCell,
431 farCell, nearCell);
432 }
