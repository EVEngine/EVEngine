63Result<void> Appearance::configure(const Value& definition) {
64 try {
65 keys(definition, {"schema", "version", "profile", "parameters", "drivers", "slots", "items"});
66 envelope(definition, "eve.appearance.definition");
67 Appearance candidate;
68 candidate.profile_ = id(field(definition, "profile"));
69 for (const auto& entry : array(field(definition, "parameters"))) {
70 keys(entry, {"id", "min", "max", "default"});
71 const auto name = id(field(entry, "id"));
72 Parameter p{number(field(entry, "min")), number(field(entry, "max")), number(field(entry, "default"))};
73 if (p.minimum >= p.maximum || p.initial < p.minimum || p.initial > p.maximum ||
74 !std::isfinite(p.maximum - p.minimum) || !candidate.parameters_.emplace(name, p).second)
75 throw std::invalid_argument("Invalid or duplicate parameter: " + name);
76 candidate.state_.parameters.emplace(name, p.initial);
77 }
78 for (const auto& entry : array(field(definition, "drivers"))) {
79 keys(entry, {"output", "scale", "inputs"});
80 Driver driver;
81 driver.output = id(field(entry, "output"));
82 driver.scale = number(field(entry, "scale"));
83 // Keep the accumulated output finite even for adversarial definitions.
84 if (std::abs(driver.scale) > 1000 || candidate.drivers_.size() >= 4096)
85 throw std::invalid_argument("Appearance driver budget exceeded");
86 for (const auto& input : array(field(entry, "inputs"))) {
87 keys(input, {"parameter", "start", "end"});
88 Input in{id(field(input, "parameter")), number(field(input, "start")), number(field(input, "end"))};
89 auto parameter = candidate.parameters_.find(in.parameter);
90 if (parameter == candidate.parameters_.end() || in.start == in.end ||
91 !std::isfinite(in.end - in.start) || std::min(in.start, in.end) < parameter->second.minimum ||
92 std::max(in.start, in.end) > parameter->second.maximum)
93 throw std::invalid_argument("Invalid driver input: " + in.parameter);
94 driver.inputs.push_back(std::move(in));
95 }
96 if (driver.inputs.empty()) throw std::invalid_argument("Driver needs at least one input");
97 candidate.drivers_.push_back(std::move(driver));
98 }
99 for (const auto& entry : array(field(definition, "slots"))) {
100 keys(entry, {"id", "default"});
101 if (!candidate.state_.selections.emplace(id(field(entry, "id")), text(field(entry, "default"))).second)
102 throw std::invalid_argument("Duplicate wardrobe slot");
103 }
104 std::set<std::string> allParts;
105 for (const auto& entry : array(field(definition, "items"))) {
106 keys(entry, {"id", "slot", "parts", "hides", "excludes"});
107 Item item{id(field(entry, "slot")), names(field(entry, "parts")), names(field(entry, "hides")),
108 names(field(entry, "excludes"))};
109 for (const auto& part : item.parts) allParts.insert(part);
110 const auto name = id(field(entry, "id"));
111 if (!candidate.state_.selections.contains(item.slot) || !candidate.items_.emplace(name, item).second)
112 throw std::invalid_argument("Invalid or duplicate wardrobe item: " + name);
113 }
114 for (const auto& [name, item] : candidate.items_) {
115 for (const auto& other : item.excludes)
116 if (other == name || !candidate.items_.contains(other))
117 throw std::invalid_argument("Invalid exclusion: " + other);
118 for (const auto& part : item.hides)
119 if (!allParts.contains(part)) throw std::invalid_argument("Unknown hidden part: " + part);
120 }
121 auto valid = candidate.validate(candidate.state_);
122 if (!valid.ok()) return valid;
123 *this = std::move(candidate);
124 return Result<void>::success();
125 } catch (const std::invalid_argument& error) {
126 return invalid(error.what());
127 }
128}

130Result<void> Appearance::validate(const State& candidate) const {
131 for (const auto& [name, value] : candidate.parameters) {
132 const auto found = parameters_.find(name);
133 if (found == parameters_.end() || !std::isfinite(value) || value < found->second.minimum ||
134 value > found->second.maximum)
135 return invalid("Unknown or out-of-range parameter: " + name);
136 }
137 std::set<std::string> selected;
138 for (const auto& [slot, name] : candidate.selections) {
139 if (!state_.selections.contains(slot)) return invalid("Unknown wardrobe slot: " + slot);
140 if (name.empty()) continue;
141 const auto found = items_.find(name);
142 if (found == items_.end() || found->second.slot != slot)
143 return invalid("Wrong item for wardrobe slot: " + slot);
144 selected.insert(name);
145 }
146 for (const auto& name : selected)
147 for (const auto& excluded : items_.at(name).excludes)
148 if (selected.contains(excluded)) return invalid("Incompatible wardrobe items: " + name + " / " + excluded);
149 return Result<void>::success();
150}

163Result<void> Appearance::apply(const Value& patch) {
164 if (profile_.empty()) return invalid("Configure an appearance definition first");
165 try {
166 keys(patch, {"parameters", "selections"});
167 auto candidate = state_;
168 if (const auto* values = patch.find("parameters")) {
169 if (!values->isObject()) return invalid("Expected parameter object");
170 for (const auto& key : values->keys()) candidate.parameters[key] = number(field(*values, key.c_str()));
171 }
172 if (const auto* values = patch.find("selections")) {
173 if (!values->isObject()) return invalid("Expected selection object");
174 for (const auto& key : values->keys()) candidate.selections[key] = text(field(*values, key.c_str()));
175 }
176 return publish(std::move(candidate));
177 } catch (const std::invalid_argument& error) {
178 return invalid(error.what());
179 }

182Result<void> Appearance::restore(const Value& snapshot) {
183 try {
184 keys(snapshot, {"schema", "version", "profile", "parameters", "selections"});
185 envelope(snapshot, "eve.appearance.state");
186 if (profile_.empty() || text(field(snapshot, "profile")) != profile_)
187 return invalid("Appearance profile mismatch");
188 const auto& parameters = field(snapshot, "parameters");
189 const auto& selections = field(snapshot, "selections");
190 if (!parameters.isObject() || !selections.isObject() || parameters.keys().size() != parameters_.size() ||
191 selections.keys().size() != state_.selections.size())
192 return invalid("Incomplete appearance snapshot");
193 return apply(Value(Object{{"parameters", parameters}, {"selections", selections}}));
194 } catch (const std::invalid_argument& error) {
195 return invalid(error.what());

225Value Appearance::evaluate() const {
226 std::map<std::string, double> outputs;
227 for (const auto& driver : drivers_) {
228 double value = driver.scale;
229 for (const auto& input : driver.inputs)
230 value *=
231 std::clamp((state_.parameters.at(input.parameter) - input.start) / (input.end - input.start), 0.0, 1.0);
232 outputs[driver.output] += value;
233 }
234 Object channels;
235 for (const auto& [key, value] : outputs) channels.emplace(key, value);
236 std::set<std::string> parts, hidden;
237 for (const auto& [slot, name] : state_.selections) {
238 (void)slot;
239 if (name.empty()) continue;
240 const auto& item = items_.at(name);
241 parts.insert(item.parts.begin(), item.parts.end());
242 hidden.insert(item.hides.begin(), item.hides.end());
243 }
244 Array visible;
245 for (const auto& part : parts)
246 if (!hidden.contains(part)) visible.emplace_back(part);
247 return Value(Object{{"channels", Value(std::move(channels))}, {"visibleParts", Value(std::move(visible))}});
248}
