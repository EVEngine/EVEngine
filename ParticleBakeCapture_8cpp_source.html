22 ParticleTrajectory trajectory;

40 if (found == births.end()) {
41 if (event.kind != particles::ParticleRecordEventKind::Snapshot || births.size() == 65536)
42 return fail("death without birth or birth budget exceeded");

48 const float time = static_cast<float>(event.seconds);
49 if (!birth.trajectory.keys.empty() && event.seconds != birth.lastTime &&
50 time <= birth.trajectory.keys.back().seconds)
51 return fail("particle times cannot be represented distinctly in asset precision");

55 birth.trajectory.keys.back() = key;

67 }

72 const BakeRecipe& recipe) {

94 for (auto& value : values) tracks.push_back(std::move(value));
95 }

97}
