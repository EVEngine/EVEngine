27int LootSystem::registerTablesFromJson(const std::string &json) {
28 std::string err;
29 const eve::json::Document doc = eve::json::Document::parse(json, &err);
30 if (!doc.valid()) return 0;
31 const Value root = doc.root();
32 int n = 0;
33 const auto parseTable = [&](Value o) {
34 if (!o.isObject()) return;
35 LootTableDefinition def;
36 def.id = o.getString("id");
37 if (def.id.empty()) return;
38 const Value entries = o.get("entries");
39 for (size_t i = 0; i < entries.size(); ++i) {
40 const Value e = entries.at(i);
41 LootEntry entry;
42 entry.itemId = e.getString("itemId");
43 entry.chance = e.getDouble("chance", 0.0);
44 entry.minQty = e.getInt("minQty", 1);
45 entry.maxQty = e.getInt("maxQty", 1);
46 if (entry.itemId.empty()) continue;
47 if (entry.maxQty < entry.minQty) entry.maxQty = entry.minQty;
48 if (entry.maxQty <= 0) entry.maxQty = 1;
49 if (entry.minQty <= 0) entry.minQty = 1;
50 def.entries.push_back(std::move(entry));
51 }
52 lootTables()[def.id] = std::move(def);
53 ++n;
54 };
55 if (root.isArray()) {
56 for (size_t i = 0; i < root.size(); ++i) parseTable(root.at(i));
57 } else if (root.isObject()) {
58 parseTable(root);
59 }
60 return n;
61}

74eve::Result<int> rollAtomic(const LootTableDefinition &definition, inventory::Bag &bag, std::mt19937 &rng) {
75 // Snapshot before inventory policy/hook callbacks can reload the registry.
76 const auto entries = definition.entries;
77 if (entries.size() > static_cast<std::size_t>(std::numeric_limits<int>::max()))
78 return eve::Result<int>::failure(
79 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "Too many loot entries"));
80 for (const auto &entry : entries) {
81 if (entry.itemId.empty() || !std::isfinite(entry.chance) || entry.chance < 0.0 || entry.chance > 1.0 ||
82 entry.minQty <= 0 || entry.maxQty < entry.minQty)
83 return eve::Result<int>::failure(
84 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "Invalid loot entry", entry.itemId));
85 }
86 auto candidateRng = rng;
87 std::uniform_real_distribution<double> chanceDist(0.0, 1.0);
88 std::vector<inventory::InventoryItemGrant> grants;
89 for (const auto &entry : entries) {
90 if (chanceDist(candidateRng) >= entry.chance) continue;
91 std::uniform_int_distribution<int> qtyDist(entry.minQty, entry.maxQty);
92 grants.push_back({entry.itemId, qtyDist(candidateRng)});
93 }
94 if (grants.empty()) {
95 rng = candidateRng;
96 return eve::Result<int>::success(0);
97 }
98 auto prepared = inventory::InventorySystem::prepareAddBatch(&bag, grants);
99 if (!prepared) return eve::Result<int>::failure(prepared.status());
100 const auto before = rng;
101 // Publish the random stream before inventory's post-commit hooks run.
102 rng = candidateRng;
103 auto committed = inventory::InventorySystem::commitAddBatch(std::move(prepared).takeValue());
104 if (!committed) {
105 rng = before;
106 return eve::Result<int>::failure(committed.status());
107 }
108 return eve::Result<int>::success(static_cast<int>(grants.size()));
109}
110} // namespace
