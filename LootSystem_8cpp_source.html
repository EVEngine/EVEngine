24int LootSystem::registerTablesFromJson(const std::string &json) {
25 std::string err;
26 const eve::json::Document doc = eve::json::Document::parse(json, &err);
27 if (!doc.valid()) return 0;
28 const Value root = doc.root();
29 int n = 0;
30 const auto parseTable = [&](Value o) {
31 if (!o.isObject()) return;
32 LootTableDefinition def;
33 def.id = o.getString("id");
34 if (def.id.empty()) return;
35 const Value entries = o.get("entries");
36 for (size_t i = 0; i < entries.size(); ++i) {
37 const Value e = entries.at(i);
38 LootEntry entry;
39 entry.itemId = e.getString("itemId");
40 entry.chance = e.getDouble("chance", 0.0);
41 entry.minQty = e.getInt("minQty", 1);
42 entry.maxQty = e.getInt("maxQty", 1);
43 if (entry.itemId.empty()) continue;
44 if (entry.maxQty < entry.minQty) entry.maxQty = entry.minQty;
45 if (entry.maxQty <= 0) entry.maxQty = 1;
46 if (entry.minQty <= 0) entry.minQty = 1;
47 def.entries.push_back(std::move(entry));
48 }
49 lootTables()[def.id] = std::move(def);
50 ++n;
51 };
52 if (root.isArray()) {
53 for (size_t i = 0; i < root.size(); ++i) parseTable(root.at(i));
54 } else if (root.isObject()) {
55 parseTable(root);
56 }
57 return n;
58}

70int LootSystem::roll(const std::string &tableId, eve::inventory::Bag *bag, std::mt19937 &rng) {
71 if (!bag) return 0;
72 const LootTableDefinition *def = find(tableId);
73 if (!def) return 0;
74 std::uniform_real_distribution<double> chanceDist(0.0, 1.0);
75 int drops = 0;
76 for (const auto &entry : def->entries) {
77 if (chanceDist(rng) >= entry.chance) continue;
78 std::uniform_int_distribution<int> qtyDist(entry.minQty, entry.maxQty);
79 const int qty = qtyDist(rng);
80 if (qty <= 0) continue;
81 bag->addItem(entry.itemId, qty);
82 ++drops;
83 }
84 return drops;
85}
