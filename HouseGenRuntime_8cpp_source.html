20 for (const auto& diagnostic : diagnostics)
21 if (diagnostic.severity() == DiagnosticSeverity::Error)
22 return EditorResult<void>::failure(eve::Status(EditorStatus::Rejected, diagnostics));

24 std::set<std::string> categories;
25 for (const auto& component : document.components()) {
26 auto registered = library->registerComponent(component.component);
27 if (!registered.ok()) return fail<void>(EditorStatus::Rejected,"editor.housegen.runtime-component","House runtime rejected an authored component");
28 categories.insert(component.component.category);
29 }
30 for (const char* required : {"foundation","floor","wall","door","roof"})
31 if (!categories.contains(required))
32 return fail<void>(EditorStatus::Rejected,"editor.housegen.runtime-category","House kit is missing required category: "+std::string(required));
33 auto layout = std::make_unique<housegen::HouseLayout>();
34 housegen::HouseGenerator generator(*library);
35 auto generated = generator.generate(document.request(), *layout);
36 if (!generated.ok()) return fail<void>(EditorStatus::Failed,"editor.housegen.generate","House generator rejected the candidate kit or request");
37 library_=std::move(library); layout_=std::move(layout); revision_=document.revision();
38 return eve::editing::applied<void>(diagnostics);
