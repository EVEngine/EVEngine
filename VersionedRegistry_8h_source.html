421 [[nodiscard]] Result<void> restoreState(State candidate) {
422 auto sequenceOk = detail::validateRegistryNextSequence(candidate.nextEventSequence);
423 if (!sequenceOk) return sequenceOk;
424
425 try {
426 for (auto& [key, entry] : candidate.entries) {
427 if (entry.generation.isZero())
428 return Result<void>::failure(
429 Diagnostic::error(DiagnosticCode::ParseError, "registry entry generation must be positive"));
430 if (entry.value.has_value()) generationProject(*entry.value, entry.generation);
431 }
432 } catch (const std::exception&) {
433 return Result<void>::failure(
434 Diagnostic::error(DiagnosticCode::Failed, "registry generation projection failed"));
435 } catch (...) {
436 return Result<void>::failure(
437 Diagnostic::error(DiagnosticCode::Failed, "registry generation projection failed"));
438 }
439 detail::EventLogValidator validator;
440 for (const auto& event : candidate.events) {
441 auto accepted = validator.accept(event.sequence, event.generation,
442 event.operation == RegistryOperation::Remove, event.tombstone);
443 if (!accepted) return accepted;
444 }
445 auto boundary = validator.finish(candidate.nextEventSequence);
446 if (!boundary) return boundary;
447
448 state_.swap(candidate);
449 return Result<void>::success();
450 }

468 [[nodiscard]] Result<Handle> mutate(RegistryOperation operation, Key key, std::optional<Value> value,
469 EventData data, std::string label) {
470 try {
471 // Every potentially-throwing operation happens on a detached
472 // candidate. The live state is touched only by the nothrow swap
473 // below, so Value/EventData/projector failures cannot partially
474 // publish a registry mutation.
475 State candidate = state_;
476 auto it = candidate.entries.find(key);
477 const bool live = it != candidate.entries.end() && it->second.value.has_value();
478 if (operation == RegistryOperation::Insert && live)
479 return Result<Handle>::failure(
480 Diagnostic::error(DiagnosticCode::AlreadyExists, "registry key already exists"));
481 if (operation == RegistryOperation::Replace && !live)
482 return Result<Handle>::failure(Diagnostic::error(DiagnosticCode::NotFound, "registry key is not live"));
483 if (operation == RegistryOperation::Remove && !live)
484 return Result<Handle>::failure(Diagnostic::error(DiagnosticCode::NotFound, "registry key is not live"));
485 if (operation != RegistryOperation::Remove && !value.has_value())
486 return Result<Handle>::failure(
487 Diagnostic::error(DiagnosticCode::InvariantViolation, "live registry mutation has no value"));
488
489 const auto current = it == candidate.entries.end() ? nullptr : &it->second;
490 auto next =
491 detail::nextRegistryGeneration(current ? current->generation : Generation{}, current != nullptr);
492 if (!next.ok()) return Result<Handle>::failure(next.status());
493 const auto nextSequence = detail::nextRegistryEventSequence(candidate.nextEventSequence);
494 if (!nextSequence.ok()) return Result<Handle>::failure(nextSequence.status());
495
496 if (operation != RegistryOperation::Remove) generationProject(*value, next.value());
497
498 candidate.events.reserve(candidate.events.size() + 1);
499 Event event{candidate.nextEventSequence,
500 operation,
501 key,
502 next.value(),
503 operation == RegistryOperation::Remove,
504 std::move(label),
505 std::move(data)};
506
507 if (operation == RegistryOperation::Insert && it == candidate.entries.end()) {
508 auto [inserted, wasInserted] =
509 candidate.entries.try_emplace(std::move(key), Entry{next.value(), std::move(value)});
510 (void)wasInserted;
511 it = inserted;
512 } else if (operation == RegistryOperation::Remove) {
513 it->second.generation = next.value();
514 it->second.value.reset();
515 } else {
516 it->second.generation = next.value();
517 it->second.value = std::move(value);
518 }
519
520 candidate.events.push_back(std::move(event));
521 candidate.nextEventSequence = nextSequence.value();
522 // Keep the notification independent of state_.events. Observer
523 // dispatch is reentrant; a nested mutation is allowed to swap or
524 // reallocate the canonical event vector while callbacks run.
525 const Event notification = candidate.events.back();
526 const Handle handle{notification.key, notification.generation};
527 Status appliedStatus = Status::success(StatusCode::Applied);
528 Status callbackWarningStatus(
529 StatusCode::Applied, {Diagnostic::warning(DiagnosticCode::CallbackFailure,
530 "registry mutation committed; observer callback failed")});
531 state_.swap(candidate);
532
533 // State is already committed. A user callback is outside the
534 // transaction boundary: never roll back an event that another
535 // callback may already have observed. Surface the exception as a
536 // successful Applied result carrying a warning instead.
537 try {
538 observers_.notify(notification);
539 } catch (const std::exception&) {
540 return Result<Handle>::success(std::move(handle), std::move(callbackWarningStatus));
541 } catch (...) {
542 return Result<Handle>::success(std::move(handle), std::move(callbackWarningStatus));
543 }
544 return Result<Handle>::success(std::move(handle), std::move(appliedStatus));
545 } catch (const std::exception&) {
546 return Result<Handle>::failure(
547 Diagnostic::error(DiagnosticCode::Failed, "registry mutation preparation failed"));
548 } catch (...) {
549 return Result<Handle>::failure(
550 Diagnostic::error(DiagnosticCode::Failed, "registry mutation preparation failed"));
551 }
552 }
