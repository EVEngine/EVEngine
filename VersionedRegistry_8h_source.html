375 [[nodiscard]] Result<void> restoreState(State candidate) {
376 if (candidate.nextEventSequence.isZero())
377 return failure<void>(DiagnosticCode::ParseError, "registry next event sequence must be positive");
378
379 EventSequence previous{};
380 try {
381 for (auto& [key, entry] : candidate.entries) {
382 if (entry.generation.isZero())
383 return failure<void>(DiagnosticCode::ParseError, "registry entry generation must be positive");
384 if (entry.value.has_value()) generationProject(*entry.value, entry.generation);
385 }
386 } catch (const std::exception&) {
387 return failure<void>(DiagnosticCode::Failed, "registry generation projection failed");
388 } catch (...) {
389 return failure<void>(DiagnosticCode::Failed, "registry generation projection failed");
390 }
391 for (const auto& event : candidate.events) {
392 if (event.sequence.isZero() || (!previous.isZero() && event.sequence <= previous))
393 return failure<void>(DiagnosticCode::ParseError, "registry event sequence is not increasing");
394 if (event.generation.isZero())
395 return failure<void>(DiagnosticCode::ParseError, "registry event generation must be positive");
396 if (event.operation == RegistryOperation::Remove && !event.tombstone)
397 return failure<void>(DiagnosticCode::ParseError, "remove event must describe a tombstone");
398 if (event.operation != RegistryOperation::Remove && event.tombstone)
399 return failure<void>(DiagnosticCode::ParseError, "only remove events may describe tombstones");
400 previous = event.sequence;
401 }
402 if (!previous.isZero() && candidate.nextEventSequence <= previous)
403 return failure<void>(DiagnosticCode::ParseError, "next event sequence must exceed retained events");
404
405 state_.swap(candidate);
406 return Result<void>::success();
407 }

438 [[nodiscard]] Result<Handle> mutate(RegistryOperation operation, Key key, std::optional<Value> value,
439 EventData data, std::string label) {
440 try {
441 // Every potentially-throwing operation happens on a detached
442 // candidate. The live state is touched only by the nothrow swap
443 // below, so Value/EventData/projector failures cannot partially
444 // publish a registry mutation.
445 State candidate = state_;
446 auto it = candidate.entries.find(key);
447 const bool live = it != candidate.entries.end() && it->second.value.has_value();
448 if (operation == RegistryOperation::Insert && live)
449 return failure<Handle>(DiagnosticCode::AlreadyExists, "registry key already exists");
450 if (operation == RegistryOperation::Replace && !live)
451 return failure<Handle>(DiagnosticCode::NotFound, "registry key is not live");
452 if (operation == RegistryOperation::Remove && !live)
453 return failure<Handle>(DiagnosticCode::NotFound, "registry key is not live");
454 if (operation != RegistryOperation::Remove && !value.has_value())
455 return failure<Handle>(DiagnosticCode::InvariantViolation, "live registry mutation has no value");
456
457 const auto current = it == candidate.entries.end() ? nullptr : &it->second;
458 auto next = nextGeneration(current);
459 if (!next.ok()) return Result<Handle>::failure(next.status());
460 const auto nextSequence = candidate.nextEventSequence.incremented();
461 if (!nextSequence) return failure<Handle>(DiagnosticCode::Failed, "registry event sequence exhausted");
462
463 if (operation != RegistryOperation::Remove) generationProject(*value, next.value());
464
465 candidate.events.reserve(candidate.events.size() + 1);
466 Event event{candidate.nextEventSequence,
467 operation,
468 key,
469 next.value(),
470 operation == RegistryOperation::Remove,
471 std::move(label),
472 std::move(data)};
473
474 if (operation == RegistryOperation::Insert && it == candidate.entries.end()) {
475 auto [inserted, wasInserted] =
476 candidate.entries.try_emplace(std::move(key), Entry{next.value(), std::move(value)});
477 (void)wasInserted;
478 it = inserted;
479 } else if (operation == RegistryOperation::Remove) {
480 it->second.generation = next.value();
481 it->second.value.reset();
482 } else {
483 it->second.generation = next.value();
484 it->second.value = std::move(value);
485 }
486
487 candidate.events.push_back(std::move(event));
488 candidate.nextEventSequence = *nextSequence;
489 // Keep the notification independent of state_.events. Observer
490 // dispatch is reentrant; a nested mutation is allowed to swap or
491 // reallocate the canonical event vector while callbacks run.
492 const Event notification = candidate.events.back();
493 const Handle handle{notification.key, notification.generation};
494 Status appliedStatus = Status::success(StatusCode::Applied);
495 Status callbackWarningStatus(
496 StatusCode::Applied, {Diagnostic::warning(DiagnosticCode::CallbackFailure,
497 "registry mutation committed; observer callback failed")});
498 state_.swap(candidate);
499
500 // State is already committed. A user callback is outside the
501 // transaction boundary: never roll back an event that another
502 // callback may already have observed. Surface the exception as a
503 // successful Applied result carrying a warning instead.
504 try {
505 observers_.notify(notification);
506 } catch (const std::exception&) {
507 return Result<Handle>::success(std::move(handle), std::move(callbackWarningStatus));
508 } catch (...) {
509 return Result<Handle>::success(std::move(handle), std::move(callbackWarningStatus));
510 }
511 return Result<Handle>::success(std::move(handle), std::move(appliedStatus));
512 } catch (const std::exception&) {
513 return failure<Handle>(DiagnosticCode::Failed, "registry mutation preparation failed");
514 } catch (...) {
515 return failure<Handle>(DiagnosticCode::Failed, "registry mutation preparation failed");
516 }
517 }
