37eve::Result<MeshEffectPlayback> parsePlayback(const eve::Value* value, const std::string& path) {
38 MeshEffectPlayback result{};
39 if (!value) return eve::Result<MeshEffectPlayback>::success(result);
40 const auto* object = value->getIf<eve::Value::Object>();
41 if (!object) return failure<MeshEffectPlayback>(eve::DiagnosticCode::InvalidArgument, "expected object", path);
42 auto known = rejectUnknown(*object, {"fadeIn", "hold", "fadeOut", "loop"}, path);
43 if (!known) return eve::Result<MeshEffectPlayback>::failure(known.status());
44 const auto read = [&](const char* key, float& target) -> eve::Result<void> {
45 const auto it = object->find(key);
46 if (it == object->end()) return eve::Result<void>::success();
47 auto parsed = number(it->second, path + "." + key);
48 if (!parsed) return eve::Result<void>::failure(parsed.status());
49 target = std::move(parsed).takeValue();
50 return eve::Result<void>::success();
51 };
52 auto status = read("fadeIn", result.fadeIn);
53 if (!status) return eve::Result<MeshEffectPlayback>::failure(status.status());
54 status = read("hold", result.duration);
55 if (!status) return eve::Result<MeshEffectPlayback>::failure(status.status());
56 status = read("fadeOut", result.fadeOut);
57 if (!status) return eve::Result<MeshEffectPlayback>::failure(status.status());
58 const auto loop = object->find("loop");
59 if (loop != object->end()) {
60 const auto* enabled = loop->second.getIf<bool>();
61 if (!enabled) return failure<MeshEffectPlayback>(eve::DiagnosticCode::InvalidArgument, "expected boolean", path + ".loop");
62 result.loop = *enabled;
63 }
64 return eve::Result<MeshEffectPlayback>::success(result);
65}

67eve::Result<MeshVfxLayerAsset> parseLayer(const eve::Value& value, std::size_t index) {
68 const std::string path = "layers[" + std::to_string(index) + "]";
69 const auto* object = value.getIf<eve::Value::Object>();
70 if (!object) return failure<MeshVfxLayerAsset>(eve::DiagnosticCode::InvalidArgument, "expected object", path);
71 auto known = rejectUnknown(*object, {"style", "playback", "parameters", "curves"}, path);
72 if (!known) return eve::Result<MeshVfxLayerAsset>::failure(known.status());
73 const auto style = object->find("style");
74 if (style == object->end() || !style->second.isString())
75 return failure<MeshVfxLayerAsset>(eve::DiagnosticCode::InvalidArgument, "style must be a string", path + ".style");
76 MeshVfxLayerAsset layer;
77 layer.style = style->second.asString();
78 const auto playback = object->find("playback");
79 auto parsedPlayback = parsePlayback(playback == object->end() ? nullptr : &playback->second, path + ".playback");
80 if (!parsedPlayback) return eve::Result<MeshVfxLayerAsset>::failure(parsedPlayback.status());
81 layer.playback = std::move(parsedPlayback).takeValue();
82 const auto parameters = object->find("parameters");
83 if (parameters != object->end()) {
84 const auto* values = parameters->second.getIf<eve::Value::Object>();
85 if (!values) return failure<MeshVfxLayerAsset>(eve::DiagnosticCode::InvalidArgument, "expected object", path + ".parameters");
86 for (const auto& [name, raw] : *values) {
87 auto parsed = number(raw, path + ".parameters." + name);
88 if (!parsed) return eve::Result<MeshVfxLayerAsset>::failure(parsed.status());
89 layer.floatParameters.emplace(name, std::move(parsed).takeValue());
90 }
91 }
92 const auto curves = object->find("curves");
93 if (curves != object->end()) {
94 const auto* values = curves->second.getIf<eve::Value::Object>();
95 if (!values) return failure<MeshVfxLayerAsset>(eve::DiagnosticCode::InvalidArgument, "expected object", path + ".curves");
96 for (const auto& [name, rawCurve] : *values) {
97 const auto* rawKeys = rawCurve.getIf<eve::Value::Array>();
98 if (!rawKeys) return failure<MeshVfxLayerAsset>(eve::DiagnosticCode::InvalidArgument, "curve must be an array", path + ".curves." + name);
99 MeshVfxFloatCurve curve;
100 for (std::size_t keyIndex = 0; keyIndex < rawKeys->size(); ++keyIndex) {
101 const auto* pair = (*rawKeys)[keyIndex].getIf<eve::Value::Array>();
102 const std::string keyPath = path + ".curves." + name + "[" + std::to_string(keyIndex) + "]";
103 if (!pair || pair->size() != 2)
104 return failure<MeshVfxLayerAsset>(eve::DiagnosticCode::InvalidArgument, "curve key must be [time, value]", keyPath);
105 auto time = number((*pair)[0], keyPath + "[0]");
106 if (!time) return eve::Result<MeshVfxLayerAsset>::failure(time.status());
107 auto keyValue = number((*pair)[1], keyPath + "[1]");
108 if (!keyValue) return eve::Result<MeshVfxLayerAsset>::failure(keyValue.status());
109 curve.keys.push_back({std::move(time).takeValue(), std::move(keyValue).takeValue()});
110 }
111 layer.floatCurves.emplace(name, std::move(curve));
112 }
113 }
114 return eve::Result<MeshVfxLayerAsset>::success(std::move(layer));
115}

247eve::Result<MeshVfxAsset> MeshVfxAsset::fromJson(std::string_view json) {
248 auto parsed = eve::Value::fromJson(json);
249 if (!parsed) return eve::Result<MeshVfxAsset>::failure(parsed.status());
250 const auto* root = parsed.value().getIf<eve::Value::Object>();
251 if (!root) return failure<MeshVfxAsset>(eve::DiagnosticCode::InvalidArgument, "mesh VFX root must be an object");
252 const auto versionIt = root->find("schemaVersion");
253 if (versionIt == root->end() || !versionIt->second.getIf<std::int64_t>())
254 return failure<MeshVfxAsset>(eve::DiagnosticCode::InvalidArgument, "schemaVersion must be an integer", "schemaVersion");
255 const auto version = *versionIt->second.getIf<std::int64_t>();
256 MeshVfxAsset asset;
257 if (version == 0) {
258 auto known = rejectUnknown(*root, {"schemaVersion", "style", "playback", "parameters", "curves", "trail", "events", "animationTriggers"}, "$root");
259 if (!known) return eve::Result<MeshVfxAsset>::failure(known.status());
260 eve::Value::Object layerObject;
261 for (const char* key : {"style", "playback", "parameters", "curves"}) {
262 const auto it = root->find(key);
263 if (it != root->end()) layerObject.emplace(key, it->second);
264 }
265 auto layer = parseLayer(eve::Value(std::move(layerObject)), 0);
266 if (!layer) return eve::Result<MeshVfxAsset>::failure(layer.status());
267 asset.layers.push_back(std::move(layer).takeValue());
268 } else if (version == schemaVersion) {
269 auto known = rejectUnknown(*root, {"schema", "schemaVersion", "layers", "trail", "events", "animationTriggers"}, "$root");
270 if (!known) return eve::Result<MeshVfxAsset>::failure(known.status());
271 const auto schema = root->find("schema");
272 if (schema == root->end() || !schema->second.isString() || schema->second.asString() != schemaId)
273 return failure<MeshVfxAsset>(eve::DiagnosticCode::InvalidArgument, "unsupported mesh VFX schema", "schema");
274 const auto layers = root->find("layers");
275 const auto* array = layers == root->end() ? nullptr : layers->second.getIf<eve::Value::Array>();
276 if (!array) return failure<MeshVfxAsset>(eve::DiagnosticCode::InvalidArgument, "layers must be an array", "layers");
277 for (std::size_t index = 0; index < array->size(); ++index) {
278 auto layer = parseLayer((*array)[index], index);
279 if (!layer) return eve::Result<MeshVfxAsset>::failure(layer.status());
280 asset.layers.push_back(std::move(layer).takeValue());
281 }
282 } else {
283 return failure<MeshVfxAsset>(eve::DiagnosticCode::UnknownVersion, "unsupported mesh VFX schema version", "schemaVersion");
284 }
285 const auto trail = root->find("trail");
286 if (trail != root->end()) {
287 auto parsedTrail = parseTrail(trail->second);
288 if (!parsedTrail) return eve::Result<MeshVfxAsset>::failure(parsedTrail.status());
289 asset.trail = std::move(parsedTrail).takeValue();
290 auto binding = parseTrailBinding(trail->second);
291 if (!binding) return eve::Result<MeshVfxAsset>::failure(binding.status());
292 asset.trailBinding = std::move(binding).takeValue();
293 }
294 const auto events = root->find("events");
295 if (events != root->end()) {
296 auto parsedEvents = parseEvents(events->second);
297 if (!parsedEvents) return eve::Result<MeshVfxAsset>::failure(parsedEvents.status());
298 asset.events = std::move(parsedEvents).takeValue();
299 }
300 const auto animationTriggers = root->find("animationTriggers");
301 if (animationTriggers != root->end()) {
302 auto parsedTriggers = parseAnimationTriggers(animationTriggers->second);
303 if (!parsedTriggers) return eve::Result<MeshVfxAsset>::failure(parsedTriggers.status());
304 asset.animationTriggers = std::move(parsedTriggers).takeValue();
305 }
306 auto valid = asset.validate();
307 if (!valid) return eve::Result<MeshVfxAsset>::failure(valid.status());
308 return eve::Result<MeshVfxAsset>::success(std::move(asset));
309}

311eve::Result<std::string> MeshVfxAsset::toJson() const {
312 auto valid = validate();
313 if (!valid) return eve::Result<std::string>::failure(valid.status());
314 eve::Value::Object root;
315 root.emplace("schema", eve::Value(std::string(schemaId)));
316 root.emplace("schemaVersion", eve::Value(static_cast<std::int64_t>(schemaVersion)));
317 eve::Value::Array layerValues;
318 layerValues.reserve(layers.size());
319 for (const auto& layer : layers) {
320 eve::Value::Object object;
321 object.emplace("style", eve::Value(layer.style));
322 eve::Value::Object playback;
323 playback.emplace("fadeIn", eve::Value(static_cast<double>(layer.playback.fadeIn)));
324 playback.emplace("hold", eve::Value(static_cast<double>(layer.playback.duration)));
325 playback.emplace("fadeOut", eve::Value(static_cast<double>(layer.playback.fadeOut)));
326 playback.emplace("loop", eve::Value(layer.playback.loop));
327 object.emplace("playback", eve::Value(std::move(playback)));
328 if (!layer.floatParameters.empty()) {
329 eve::Value::Object parameters;
330 for (const auto& [name, value] : layer.floatParameters)
331 parameters.emplace(name, eve::Value(static_cast<double>(value)));
332 object.emplace("parameters", eve::Value(std::move(parameters)));
333 }
334 if (!layer.floatCurves.empty()) {
335 eve::Value::Object curves;
336 for (const auto& [name, curve] : layer.floatCurves) {
337 eve::Value::Array keys;
338 for (const auto& key : curve.keys) {
339 eve::Value::Array pair;
340 pair.emplace_back(static_cast<double>(key.time));
341 pair.emplace_back(static_cast<double>(key.value));
342 keys.emplace_back(std::move(pair));
343 }
344 curves.emplace(name, eve::Value(std::move(keys)));
345 }
346 object.emplace("curves", eve::Value(std::move(curves)));
347 }
348 layerValues.emplace_back(std::move(object));
349 }
350 root.emplace("layers", eve::Value(std::move(layerValues)));
351 if (trail) {
352 eve::Value::Object value;
353 value.emplace("maxSamples", eve::Value(static_cast<std::int64_t>(trail->maxSamples)));
354 value.emplace("lifetime", eve::Value(static_cast<double>(trail->lifetime)));
355 value.emplace("minDistance", eve::Value(static_cast<double>(trail->minSampleDistance)));
356 value.emplace("teleportDistance", eve::Value(static_cast<double>(trail->teleportDistance)));
357 if (trailBinding) {
358 value.emplace("rootAttachment", eve::Value(trailBinding->rootAttachment));
359 value.emplace("tipAttachment", eve::Value(trailBinding->tipAttachment));
360 value.emplace("rootOffset", eve::Value::Array{static_cast<double>(trailBinding->rootOffset.x),
361 static_cast<double>(trailBinding->rootOffset.y),
362 static_cast<double>(trailBinding->rootOffset.z)});
363 value.emplace("tipOffset", eve::Value::Array{static_cast<double>(trailBinding->tipOffset.x),
364 static_cast<double>(trailBinding->tipOffset.y),
365 static_cast<double>(trailBinding->tipOffset.z)});
366 }
367 root.emplace("trail", eve::Value(std::move(value)));
368 }
369 if (!events.empty()) {
370 eve::Value::Array values;
371 for (const auto& event : events) {
372 eve::Value::Object value;
373 value.emplace("time", eve::Value(static_cast<double>(event.time)));
374 value.emplace("name", eve::Value(event.name));
375 values.emplace_back(std::move(value));
376 }
377 root.emplace("events", eve::Value(std::move(values)));
378 }
379 if (!animationTriggers.play.empty() || !animationTriggers.stop.empty() || !animationTriggers.trailBreak.empty()) {
380 eve::Value::Object value;
381 const auto append = [&](const char* key, const std::vector<std::string>& names) {
382 if (names.empty()) return;
383 eve::Value::Array array;
384 for (const auto& name : names) array.emplace_back(name);
385 value.emplace(key, eve::Value(std::move(array)));
386 };
387 append("play", animationTriggers.play);
388 append("stop", animationTriggers.stop);
389 append("trailBreak", animationTriggers.trailBreak);
390 root.emplace("animationTriggers", eve::Value(std::move(value)));
391 }
392 return eve::Value(std::move(root)).toJson();
393}

395eve::Result<void> MeshVfxAsset::validate() const {
396 if (layers.empty()) return failure<void>(eve::DiagnosticCode::InvalidArgument, "mesh VFX requires at least one layer", "layers");
397 for (std::size_t index = 0; index < layers.size(); ++index) {
398 const auto& layer = layers[index];
399 const std::string path = "layers[" + std::to_string(index) + "]";
400 if (layer.style.empty()) return failure<void>(eve::DiagnosticCode::InvalidArgument, "style must not be empty", path + ".style");
401 if (!finiteNonNegative(layer.playback.fadeIn) || !finiteNonNegative(layer.playback.duration) ||
402 !finiteNonNegative(layer.playback.fadeOut))
403 return failure<void>(eve::DiagnosticCode::InvalidArgument, "playback durations must be finite and non-negative", path + ".playback");
404 for (const auto& [name, value] : layer.floatParameters)
405 if (name.empty() || !std::isfinite(value))
406 return failure<void>(eve::DiagnosticCode::InvalidArgument, "parameter names must be non-empty and values finite", path + ".parameters");
407 for (const auto& [name, curve] : layer.floatCurves) {
408 if (name.empty() || curve.keys.empty())
409 return failure<void>(eve::DiagnosticCode::InvalidArgument, "curve names and key arrays must not be empty", path + ".curves");
410 float previous = -1.f;
411 for (const auto& key : curve.keys) {
412 if (!std::isfinite(key.time) || !std::isfinite(key.value) || key.time < 0.f || key.time > 1.f || key.time <= previous)
413 return failure<void>(eve::DiagnosticCode::InvalidArgument, "curve times must be finite, normalized, and strictly increasing", path + ".curves." + name);
414 previous = key.time;
415 }
416 }
417 }
418 if (trail && (trail->maxSamples == 0 || !finiteNonNegative(trail->lifetime) ||
419 !finiteNonNegative(trail->minSampleDistance) || !finiteNonNegative(trail->teleportDistance)))
420 return failure<void>(eve::DiagnosticCode::InvalidArgument, "invalid trail settings", "trail");
421 if (trailBinding && (!trail || trailBinding->rootAttachment.empty() || trailBinding->tipAttachment.empty()))
422 return failure<void>(eve::DiagnosticCode::InvalidArgument,
423 "trail attachment binding requires a trail and two non-empty attachment names", "trail");
424 float previousEventTime = -1.f;
425 for (const auto& event : events) {
426 if (!std::isfinite(event.time) || event.time < 0.f || event.time > 1.f || event.time < previousEventTime || event.name.empty())
427 return failure<void>(eve::DiagnosticCode::InvalidArgument,
428 "events require non-empty names and non-decreasing normalized times", "events");
429 previousEventTime = event.time;
430 }
431 std::set<std::string> triggerNames;
432 const auto validateTriggers = [&](const std::vector<std::string>& names) -> eve::Result<void> {
433 for (const auto& name : names)
434 if (name.empty() || !triggerNames.insert(name).second)
435 return failure<void>(eve::DiagnosticCode::InvalidArgument,
436 "animation trigger names must be non-empty and map to exactly one action",
437 "animationTriggers");
438 return eve::Result<void>::success();
439 };
440 auto triggerStatus = validateTriggers(animationTriggers.play);
441 if (!triggerStatus) return triggerStatus;
442 triggerStatus = validateTriggers(animationTriggers.stop);
443 if (!triggerStatus) return triggerStatus;
444 triggerStatus = validateTriggers(animationTriggers.trailBreak);
445 if (!triggerStatus) return triggerStatus;
446 return eve::Result<void>::success();
447}

449float MeshVfxFloatCurve::evaluate(float normalizedTime) const noexcept {
450 if (keys.empty()) return 0.f;
451 const float time = std::clamp(normalizedTime, 0.f, 1.f);
452 if (time <= keys.front().time) return keys.front().value;
453 if (time >= keys.back().time) return keys.back().value;
454 const auto upper = std::upper_bound(keys.begin(), keys.end(), time,
455 [](float value, const MeshVfxFloatKey& key) { return value < key.time; });
456 const auto& right = *upper;
457 const auto& left = *(upper - 1);
458 const float alpha = (time - left.time) / (right.time - left.time);
459 return left.value + (right.value - left.value) * alpha;
460}

512void MeshVfxAssetInstance::update(float dtSeconds) {
513 for (std::size_t index = 0; index < layers_.size(); ++index) {
514 auto& layer = *layers_[index];
515 layer.update(dtSeconds);
516 const auto playback = layer.playback();
517 const float cycle = playback.fadeIn + playback.duration + playback.fadeOut;
518 const float normalized = cycle > 0.f ? std::clamp(layer.elapsed() / cycle, 0.f, 1.f) : 1.f;
519 for (const auto& [name, curve] : curves_[index]) layer.style().setFloat(name, curve.evaluate(normalized));
520 }
521 if (trail_) trail_->update(dtSeconds);
522 if (layers_.empty()) return;
523 const auto playback = layers_.front()->playback();
524 const float cycle = playback.fadeIn + playback.duration + playback.fadeOut;
525 const float current = cycle > 0.f ? std::clamp(layers_.front()->elapsed() / cycle, 0.f, 1.f) : 1.f;
526 const bool wrapped = playback.loop && current < previousNormalizedTime_;
527 for (const auto& event : events_) {
528 const bool crossed = wrapped ? (event.time > previousNormalizedTime_ || event.time <= current)
529 : (event.time > previousNormalizedTime_ && event.time <= current);
530 if (crossed) pendingEvents_.push_back(event.name);
531 }
532 previousNormalizedTime_ = current;
533}

555MeshVfxAnimationDispatch MeshVfxAssetInstance::processAnimationEvents(const eve::IAnimationEventSource& source) {
556 MeshVfxAnimationDispatch dispatch;
557 const auto contains = [](const std::vector<std::string>& values, const std::string& name) {
558 return std::find(values.begin(), values.end(), name) != values.end();
559 };
560 for (std::size_t index = 0; index < source.animationEventCount(); ++index) {
561 const std::string name = source.animationEventName(index);
562 if (contains(animationTriggers_.play, name)) {
563 play();
564 ++dispatch.played;
565 } else if (contains(animationTriggers_.stop, name)) {
566 stop();
567 ++dispatch.stopped;
568 } else if (contains(animationTriggers_.trailBreak, name)) {
569 if (trail_) trail_->breakTrail();
570 ++dispatch.trailBreaks;
571 } else {
572 continue;
573 }
574 ++dispatch.handled;
575 }
576 return dispatch;
577}
