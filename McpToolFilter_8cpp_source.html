29std::vector<std::string> splitComma(std::string_view raw) {
30 std::vector<std::string> out;
31 std::size_t start = 0;
32 while (start < raw.size()) {
33 const std::size_t comma = raw.find(',', start);
34 const auto end = comma == std::string_view::npos ? raw.size() : comma;
35 std::string part = toLower(raw.substr(start, end - start));
36 while (!part.empty() && std::isspace(static_cast<unsigned char>(part.front()))) part.erase(part.begin());
37 while (!part.empty() && std::isspace(static_cast<unsigned char>(part.back()))) part.pop_back();
38 if (!part.empty()) out.push_back(std::move(part));
39 if (comma == std::string_view::npos) break;
40 start = comma + 1;
41 }
42 return out;
43}

56void loadConfigUnlocked() {
57 g_cfg = FilterConfig{};
58 if (const char* toolset = std::getenv("EVE_MCP_TOOLSET")) {
59 const auto parts = splitComma(toolset);
60 g_cfg.all = parts.empty();
61 for (const auto& part : parts) {
62 if (part == "all") {
63 g_cfg.all = true;
64 g_cfg.tags.clear();
65 break;
66 }
67 g_cfg.all = false;
68 g_cfg.tags.insert(part);
69 }
70 }
71 if (const char* ro = std::getenv("EVE_MCP_READONLY")) {
72 const std::string value = toLower(ro);
73 g_cfg.readOnly = !(value.empty() || value == "0" || value == "false" || value == "off" || value == "no");
74 }
75 if (const char* exclude = std::getenv("EVE_MCP_EXCLUDE")) {
76 for (auto& name : splitComma(exclude)) g_cfg.exclude.insert(std::move(name));
77 }
78 g_loaded = true;
79}

81const FilterConfig& config() {

242bool mcpToolAllowed(std::string_view name) {
243 const FilterConfig& cfg = config();
244 if (cfg.exclude.count(toLower(name)) != 0) return false;
245 if (cfg.readOnly && mcpToolMutates(name)) return false;
246 if (!cfg.all) {
247 const auto tags = mcpToolTags(name);
248 bool hit = false;
249 for (const auto& tag : tags) {
250 if (cfg.tags.count(tag) != 0) {
251 hit = true;
252 break;
253 }
254 }
255 if (!hit) return false;
256 }
257 const std::string processProfile = mcpProcessProfile();
258 if (!mcpProfileAllowsTool(processProfile, name)) return false;
259 const std::string sessionProfile = mcpSessionProfile();
260 if (!mcpProfileAllowsTool(sessionProfile, name)) return false;
261 return true;
262}

264std::string filterMcpToolsBody(std::string_view toolsBody) {
265 if (toolsBody.empty()) return {};
266 const std::string wrapped = std::string("[") + std::string(toolsBody) + "]";
267 try {
268 Poco::JSON::Parser parser;
269 auto array = parser.parse(wrapped).extract<Poco::JSON::Array::Ptr>();
270 if (!array) return std::string(toolsBody);
271 Poco::JSON::Array::Ptr kept = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
272 for (std::size_t i = 0; i < array->size(); ++i) {
273 auto tool = array->getObject(static_cast<unsigned>(i));
274 if (!tool || !tool->has("name")) continue;
275 const std::string name = tool->getValue<std::string>("name");
276 if (mcpToolAllowed(name)) kept->add(tool);
277 }
278 if (kept->size() == 0) return {};
279 std::string out = mcpStringify(Poco::Dynamic::Var(kept));
280 if (out.size() >= 2 && out.front() == '[' && out.back() == ']') return out.substr(1, out.size() - 2);
281 return out;
282 } catch (...) {
283 return std::string(toolsBody);
284 }
285}

287std::string mcpToolFilteredPayload(std::string_view name) {
288 const std::string processProfile = mcpProcessProfile();
289 const std::string sessionProfile = mcpSessionProfile();
290 const std::string activeProfile = !sessionProfile.empty() ? sessionProfile : processProfile;
291
292 std::string error = "tool '" + std::string(name) +
293 "' is hidden by EVE_MCP_TOOLSET / EVE_MCP_READONLY / EVE_MCP_EXCLUDE / EVE_MCP_PROFILE";
294 std::string hint = "Adjust env filters or call eve_status; use EVE_MCP_TOOLSET=all to advertise every tool";
295 std::vector<std::string> next = {"eve_status"};
296 std::optional<McpProfileEscalate> escalate;
297 if (!activeProfile.empty() && !mcpProfileAllowsTool(activeProfile, name)) {
298 error = "tool '" + std::string(name) + "' is outside task profile '" + activeProfile + "'";
299 hint = "Call eve_agent_session_escalate with an allowed target, or start a new session with the needed profile";
300 next = {"eve_agent_session_status", "eve_agent_session_escalate"};
301 escalate = mcpProfileEscalateHint(activeProfile, name);
302 if (escalate) {
303 hint = "Escalate to profile '" + escalate->to + "': " + escalate->reason;
304 next = {"eve_agent_session_escalate", "eve_agent_session_status"};
305 }
306 }
307
308 Poco::JSON::Object::Ptr out = mcpFailureObject(error, hint, false, next);
309 out->set("filtered", true);
310 out->set("tool", std::string(name));
311 if (!activeProfile.empty()) out->set("profile", activeProfile);
312 if (!sessionProfile.empty()) out->set("sessionProfile", sessionProfile);
313 if (!processProfile.empty()) out->set("processProfile", processProfile);
314 if (escalate) {
315 Poco::JSON::Object::Ptr esc(new Poco::JSON::Object());
316 esc->set("to", escalate->to);
317 esc->set("reason", escalate->reason);
318 esc->set("errorCode", escalate->errorCode);
319 out->set("escalate", esc);
320 }
321 Poco::JSON::Array::Ptr tags = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
322 for (const auto& tag : mcpToolTags(name)) tags->add(tag);
323 out->set("tags", tags);
324 out->set("mutates", mcpToolMutates(name));
325 const std::string payload = mcpStringify(Poco::Dynamic::Var(out));
326 mcpFilterNoteCall(false, payload.size(), escalate.has_value());
327 return payload;
328}
