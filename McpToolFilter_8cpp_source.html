27std::vector<std::string> splitComma(std::string_view raw) {
28 std::vector<std::string> out;
29 std::size_t start = 0;
30 while (start < raw.size()) {
31 const std::size_t comma = raw.find(',', start);
32 const auto end = comma == std::string_view::npos ? raw.size() : comma;
33 std::string part = toLower(raw.substr(start, end - start));
34 while (!part.empty() && std::isspace(static_cast<unsigned char>(part.front()))) part.erase(part.begin());
35 while (!part.empty() && std::isspace(static_cast<unsigned char>(part.back()))) part.pop_back();
36 if (!part.empty()) out.push_back(std::move(part));
37 if (comma == std::string_view::npos) break;
38 start = comma + 1;
39 }
40 return out;
41}

54void loadConfigUnlocked() {
55 g_cfg = FilterConfig{};
56 if (const char* toolset = std::getenv("EVE_MCP_TOOLSET")) {
57 const auto parts = splitComma(toolset);
58 g_cfg.all = parts.empty();
59 for (const auto& part : parts) {
60 if (part == "all") {
61 g_cfg.all = true;
62 g_cfg.tags.clear();
63 break;
64 }
65 g_cfg.all = false;
66 g_cfg.tags.insert(part);
67 }
68 }
69 if (const char* ro = std::getenv("EVE_MCP_READONLY")) {
70 const std::string value = toLower(ro);
71 g_cfg.readOnly =
72 !(value.empty() || value == "0" || value == "false" || value == "off" || value == "no");
73 }
74 if (const char* exclude = std::getenv("EVE_MCP_EXCLUDE")) {
75 for (auto& name : splitComma(exclude)) g_cfg.exclude.insert(std::move(name));
76 }
77 g_loaded = true;
78}

207bool mcpToolAllowed(std::string_view name) {
208 const FilterConfig& cfg = config();
209 if (cfg.exclude.count(toLower(name)) != 0) return false;
210 if (cfg.readOnly && mcpToolMutates(name)) return false;
211 if (cfg.all) return true;
212 const auto tags = mcpToolTags(name);
213 for (const auto& tag : tags) {
214 if (cfg.tags.count(tag) != 0) return true;
215 }
216 return false;
217}

219std::string filterMcpToolsBody(std::string_view toolsBody) {
220 if (toolsBody.empty()) return {};
221 const std::string wrapped = std::string("[") + std::string(toolsBody) + "]";
222 try {
223 Poco::JSON::Parser parser;
224 auto array = parser.parse(wrapped).extract<Poco::JSON::Array::Ptr>();
225 if (!array) return std::string(toolsBody);
226 Poco::JSON::Array::Ptr kept = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
227 for (std::size_t i = 0; i < array->size(); ++i) {
228 auto tool = array->getObject(static_cast<unsigned>(i));
229 if (!tool || !tool->has("name")) continue;
230 const std::string name = tool->getValue<std::string>("name");
231 if (mcpToolAllowed(name)) kept->add(tool);
232 }
233 if (kept->size() == 0) return {};
234 std::string out = mcpStringify(Poco::Dynamic::Var(kept));
235 if (out.size() >= 2 && out.front() == '[' && out.back() == ']') return out.substr(1, out.size() - 2);
236 return out;
237 } catch (...) {
238 return std::string(toolsBody);
239 }
240}

243 Poco::JSON::Object::Ptr out = mcpFailureObject(

245 "Adjust env filters or call eve_status; use EVE_MCP_TOOLSET=all to advertise every tool", false,
246 {"eve_status"});
247 out->set("filtered", true);
248 out->set("tool", std::string(name));
249 Poco::JSON::Array::Ptr tags = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
250 for (const auto& tag : mcpToolTags(name)) tags->add(tag);
251 out->set("tags", tags);
252 out->set("mutates", mcpToolMutates(name));
253 return mcpStringify(Poco::Dynamic::Var(out));
254}
