64MotionEchoAdvanceReport ProjectileTrailEffect::update(float deltaSeconds,
65 const glm::vec3& currentPosition) {
66 MotionEchoAdvanceReport report;
67 if (!std::isfinite(deltaSeconds) || deltaSeconds < 0.f ||
68 !std::isfinite(currentPosition.x) || !std::isfinite(currentPosition.y) ||
69 !std::isfinite(currentPosition.z)) {
70 report.alive = static_cast<std::uint32_t>(samples_.size());
71 return report;
72 }
73 for (auto& sample : samples_) sample.age += deltaSeconds;
74 const auto expired = std::remove_if(samples_.begin(), samples_.end(), [&](const Sample& sample) {
75 return sample.age >= config_.sampleLifetimeSeconds;
76 });
77 report.expired += static_cast<std::uint32_t>(std::distance(expired, samples_.end()));
78 samples_.erase(expired, samples_.end());
79
80 if (!hasPreviousInput_) {
81 previousInput_ = currentPosition;
82 hasPreviousInput_ = true;
83 (void)append(currentPosition, 0.f, report);
84 } else {
85 const float previousAccumulator = sampleAccumulator_;
86 sampleAccumulator_ += deltaSeconds;
87 const std::uint32_t count = static_cast<std::uint32_t>(
88 std::floor(sampleAccumulator_ / config_.sampleIntervalSeconds));
89 for (std::uint32_t i = 0; i < count; ++i) {
90 const float elapsed = config_.sampleIntervalSeconds - previousAccumulator +
91 static_cast<float>(i) * config_.sampleIntervalSeconds;
92 const float t = deltaSeconds > 1e-6f ? std::clamp(elapsed / deltaSeconds, 0.f, 1.f) : 1.f;
93 (void)append(previousInput_ + (currentPosition - previousInput_) * t,
94 std::max(0.f, deltaSeconds - elapsed), report);
95 }
96 sampleAccumulator_ -= static_cast<float>(count) * config_.sampleIntervalSeconds;
97 previousInput_ = currentPosition;
98 }
99 report.alive = static_cast<std::uint32_t>(samples_.size());
100 return report;
101}

103TrailMeshSnapshot ProjectileTrailEffect::build(const glm::vec3& cameraForward) const {
104 TrailMeshSnapshot mesh;
105 if (samples_.size() < 2) return mesh;
106 mesh.vertices.reserve(samples_.size() * 2u);
107 mesh.indices.reserve((samples_.size() - 1u) * 6u);
108 const glm::vec3 view = safeNormal(cameraForward, glm::vec3(0.f, 0.f, -1.f));
109 for (std::size_t i = 0; i < samples_.size(); ++i) {
110 const float t = static_cast<float>(i) / static_cast<float>(samples_.size() - 1u);
111 const auto& previous = samples_[i == 0 ? 0 : i - 1u].position;
112 const auto& next = samples_[std::min(i + 1u, samples_.size() - 1u)].position;
113 const glm::vec3 tangent = safeNormal(next - previous, glm::vec3(0.f, 1.f, 0.f));
114 glm::vec3 side = safeNormal(glm::cross(view, tangent), glm::vec3(1.f, 0.f, 0.f));
115 const float width = config_.widthTail + (config_.widthHead - config_.widthTail) * t;
116 const float alpha = std::clamp(1.f - samples_[i].age / config_.sampleLifetimeSeconds,
117 0.f, 1.f);
118 mesh.vertices.push_back({samples_[i].position - side * (width * 0.5f), {t, 0.f}, alpha});
119 mesh.vertices.push_back({samples_[i].position + side * (width * 0.5f), {t, 1.f}, alpha});
120 if (i == 0) continue;
121 const std::uint32_t left0 = static_cast<std::uint32_t>((i - 1u) * 2u);
122 const std::uint32_t right0 = left0 + 1u;
123 const std::uint32_t left1 = static_cast<std::uint32_t>(i * 2u);
124 const std::uint32_t right1 = left1 + 1u;
125 mesh.indices.insert(mesh.indices.end(), {left0, right0, left1, left1, right0, right1});
126 }
127 return mesh;
128}
