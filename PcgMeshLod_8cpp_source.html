50Result<int> combinePcgStaticMeshesInto(MeshBuild& output, const PcgMeshCombinePlan& plan) {
51 if (plan.sources_.empty()) return invalid<int>("Pcg static mesh combine plan is empty");
52 std::uint64_t vertexCount = 0, indexCount = 0;
53 bool anyColors = false;
54 for (const auto& source : plan.sources_) {
55 vertexCount += static_cast<std::uint64_t>(source.mesh.getVertexCount());
56 indexCount += static_cast<std::uint64_t>(source.mesh.getIndexCount());
57 anyColors = anyColors || source.mesh.hasVertexColors();
58 }
59 if (vertexCount > 8U * 1024U * 1024U || indexCount > 48U * 1024U * 1024U ||
60 vertexCount > static_cast<std::uint64_t>(std::numeric_limits<int>::max()) ||
61 indexCount > static_cast<std::uint64_t>(std::numeric_limits<int>::max()))
62 return invalid<int>("Pcg static mesh combine output exceeds native mesh budgets");
63 MeshBuild candidate;
64 candidate.reserve(static_cast<int>(vertexCount), static_cast<int>(indexCount));
65 std::vector<float> colors;
66 if (anyColors) colors.reserve(static_cast<std::size_t>(vertexCount) * 4U);
67 struct Bucket { std::string material; std::vector<std::uint32_t> indices; };
68 std::vector<Bucket> buckets;
69 std::unordered_map<std::string, std::size_t> materialMap;
70 std::uint32_t vertexOffset = 0;
71 for (const auto& source : plan.sources_) {
72 const auto& mesh = source.mesh;
73 const auto& m = source.transform.matrix_;
74 for (int vertex = 0; vertex < mesh.getVertexCount(); ++vertex) {
75 const float x=mesh.getPositionX(vertex),y=mesh.getPositionY(vertex),z=mesh.getPositionZ(vertex);
76 const float nx=mesh.getNormalX(vertex),ny=mesh.getNormalY(vertex),nz=mesh.getNormalZ(vertex);
77 const float px=m[0]*x+m[1]*y+m[2]*z+m[3],py=m[4]*x+m[5]*y+m[6]*z+m[7];
78 const float pz=m[8]*x+m[9]*y+m[10]*z+m[11];
79 const float tx=m[0]*nx+m[1]*ny+m[2]*nz,ty=m[4]*nx+m[5]*ny+m[6]*nz;
80 const float tz=m[8]*nx+m[9]*ny+m[10]*nz;
81 if(!std::isfinite(px)||!std::isfinite(py)||!std::isfinite(pz)||!std::isfinite(tx)||
82 !std::isfinite(ty)||!std::isfinite(tz))return invalid<int>("Pcg static mesh transform overflowed");
83 candidate.addVertex(px,py,pz,tx,ty,tz,mesh.getUvU(vertex),mesh.getUvV(vertex));
84 if(anyColors)for(int component=0;component<4;++component)
85 colors.push_back(mesh.hasVertexColors()?mesh.getColor(vertex,component):1.F);
86 }
87 for (int triangle = 0; triangle < mesh.getIndexCount() / 3; ++triangle) {
88 const int group = mesh.getTriangleGroup(triangle);
89 std::string material = group >= 0 ? mesh.getGroupName(group) : source.defaultMaterialId;
90 if(material.empty())return invalid<int>("Pcg static mesh triangle has no material identity");
91 auto [found, inserted] = materialMap.emplace(material,buckets.size());
92 if(inserted)buckets.push_back(Bucket{material,{}});
93 auto& indices=buckets[found->second].indices;
94 for(int corner=0;corner<3;++corner)
95 indices.push_back(vertexOffset+static_cast<std::uint32_t>(mesh.getIndex(triangle*3+corner)));
96 }
97 vertexOffset += static_cast<std::uint32_t>(mesh.getVertexCount());
98 }
99 for(auto& bucket:buckets){candidate.setActiveGroup(bucket.material);
100 for(std::size_t i=0;i<bucket.indices.size();i+=3)
101 candidate.addTriangle(bucket.indices[i],bucket.indices[i+1],bucket.indices[i+2]);}
102 if(anyColors){auto set=candidate.setVertexColors(std::move(colors));if(!set.ok())return Result<int>::failure(set.status());}
103 candidate.setMeta("kind","pcg.mesh.combined");
104 candidate.setMeta("sourceCount",std::to_string(plan.sources_.size()));
105 output=std::move(candidate);
106 return Result<int>::success(static_cast<int>(indexCount/3U));
107}

207float PcgMeshLodSet::getSwitchDistance(int level, float diameter, float fov) const noexcept {
208 if (level < 0 || level >= static_cast<int>(levels_.size()) || !std::isfinite(diameter) || diameter <= 0.F ||
209 !std::isfinite(fov) || fov <= 0.F || fov >= 180.F) return -1.F;

216Result<void> buildPcgMeshLodsInto(PcgMeshLodSet& output, const MeshBuild& source,
217 const PcgMeshLodProfile& profile) {
218 const auto& levels = profile.levels();
219 if (source.empty() || levels.empty() || levels.size() > 4)
220 return invalid<void>("Pcg mesh LOD requires a non-empty mesh and one through four levels");
221 PcgMeshLodSet candidate;
222 candidate.levels_ = levels;
223 candidate.fadeMode_ = profile.getFadeMode();
224 candidate.animateCrossFading_ = profile.getAnimateCrossFading();
225 candidate.crossFadeAnimationDuration_ = profile.getCrossFadeAnimationDuration();
226 candidate.meshes_.reserve(levels.size());
227 for (const PcgMeshLodLevel& level : levels) {
228 MeshBuild mesh;
229 auto simplified = simplifyGtsMesh(mesh, source, level.quality, level.simplification);
230 if (!simplified.ok()) return Result<void>::failure(simplified.status());
231 candidate.meshes_.push_back(std::move(mesh));
232 }
233 output = std::move(candidate);
234 return Result<void>::success();
235}
