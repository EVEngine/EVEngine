403Result<std::map<std::string, graphics::VegetationMask>> loadVegetationSceneElementMasks(
404 const asset::EvpackResourceReader& reader, const LoadedVegetationScene& scene,
405 const asset::EvpackCapabilities& capabilities, std::uint64_t maximumDecodedBytes) {
406 try {
407 std::map<std::string, std::string> required;
408 auto admit = [&](const std::string& guid, const std::string& asset) -> Result<void> {
409 if (guid.empty()) return Result<void>::success();
410 if (asset.empty())
411 return fail<void>(DiagnosticCode::NotFound, "vegetation element texture dependency is unavailable", guid);
412 const auto [position, inserted] = required.emplace(guid, asset);
413 if (!inserted && position->second != asset)
414 return fail<void>(DiagnosticCode::Conflict, "vegetation element texture GUID has conflicting assets", guid);
415 return Result<void>::success();
416 };
417 for (const auto& element : scene.elements) {
418 auto admitted = admit(element.textureGuid, element.textureAsset);
419 if (!admitted) return Result<std::map<std::string, graphics::VegetationMask>>::failure(admitted.status());
420 for (const auto& property : element.properties) {
421 if (property.type != 0 || (property.name != "_MainTex" && property.name != "_NoiseTex")) continue;
422 admitted = admit(property.textureGuid, property.textureAsset);
423 if (!admitted) return Result<std::map<std::string, graphics::VegetationMask>>::failure(admitted.status());
424 }
425 }
426 std::map<std::string, graphics::VegetationMask> output;
427 std::uint64_t usedBytes = 0;
428 for (const auto& [guid, encodedAsset] : required) {
429 if (usedBytes >= maximumDecodedBytes)
430 return fail<std::map<std::string, graphics::VegetationMask>>(
431 DiagnosticCode::InvalidArgument, "vegetation element masks exceed aggregate decoded budget");
432 auto image = AssetRef::parse(encodedAsset);
433 if (!image) return Result<std::map<std::string, graphics::VegetationMask>>::failure(image.status());
434 asset::EvpackImageDecodeLimits limits;
435 limits.maximumDimension = 2048;
436 limits.maximumPixels = 4 * 1024 * 1024;
437 limits.maximumDecodedBytes = maximumDecodedBytes - usedBytes;
438 auto decoded = asset::decodeEvpackImage(reader, image.value(), capabilities, limits);
439 if (!decoded) return Result<std::map<std::string, graphics::VegetationMask>>::failure(decoded.status());
440 const std::uint64_t pixelCount = std::uint64_t(decoded.value().width) * decoded.value().height;
441 const std::uint64_t decodedBytes = decoded.value().pixels.size();
442 const std::uint64_t maskBytes = pixelCount * sizeof(glm::vec4);
443 if (decodedBytes > maximumDecodedBytes - usedBytes ||
444 maskBytes > maximumDecodedBytes - usedBytes - decodedBytes)
445 return fail<std::map<std::string, graphics::VegetationMask>>(
446 DiagnosticCode::InvalidArgument, "vegetation element masks exceed aggregate decoded budget");
447 graphics::VegetationMask mask;
448 mask.width = decoded.value().width;
449 mask.height = decoded.value().height;
450 mask.pixels.resize(std::size_t(pixelCount));
451 for (std::size_t index = 0; index < mask.pixels.size(); ++index) {
452 const auto offset = index * 4;
453 mask.pixels[index] = {decoded.value().pixels[offset] / 255.f, decoded.value().pixels[offset + 1] / 255.f,
454 decoded.value().pixels[offset + 2] / 255.f,
455 decoded.value().pixels[offset + 3] / 255.f};
456 }
457 usedBytes += decodedBytes + maskBytes;
458 output.emplace(guid, std::move(mask));
459 }
460 return Result<std::map<std::string, graphics::VegetationMask>>::success(std::move(output));
461 } catch (const std::bad_alloc&) {
462 return fail<std::map<std::string, graphics::VegetationMask>>(DiagnosticCode::Failed,
463 "vegetation element mask allocation failed");
464 }
465}

467Result<VegetationSceneElementPixel> evaluateVegetationSceneElementPixel(
468 const VegetationSceneElement& element, const VegetationSceneElementPixelInput& input) {
469 if (!finite(input.localUv) || !finite(input.mainSample) || !finite(input.elementParams) ||
470 !finite(input.vertexColor) || !std::isfinite(input.worldPosition.x) || !std::isfinite(input.worldPosition.y) ||
471 !std::isfinite(input.worldPosition.z) || !std::isfinite(input.worldNormal.x) || !std::isfinite(input.worldNormal.y) ||
472 !std::isfinite(input.worldNormal.z) || !finite(input.velocityDirection) || !finite(input.noiseSample) ||
473 !std::isfinite(input.terrainHeight) ||
474 !unit(input.volumeFade) || !std::isfinite(input.season) || input.season < 0.f ||
475 input.season > 4.f || !unit(element.intensity) || element.blendRgb < 0 || element.blendRgb > 2 ||
476 element.blendAlpha < 0 || element.blendAlpha > 1)
477 return fail<VegetationSceneElementPixel>(DiagnosticCode::InvalidArgument,
478 "vegetation element pixel inputs are invalid");
479
480 VegetationSceneElementPixel output;
481 output.blendRgb = element.blendRgb;
482 output.blendAlpha = element.blendAlpha;
483 if (!element.enabled) return Result<VegetationSceneElementPixel>::success(output);
484
485 const glm::vec3 remappedRgb(
486 remap(input.mainSample.r, element.remap[0], element.remap[1], 0.f),
487 remap(input.mainSample.g, element.remap[0], element.remap[1], 0.f),
488 remap(input.mainSample.b, element.remap[0], element.remap[1], 0.f));
489 const float remappedAlpha = remap(input.mainSample.a, element.remap[2], element.remap[3], .0001f);
490 const glm::vec2 centered = input.localUv * 2.f - 1.f;
491 const float radial = std::clamp(std::clamp(1.f - glm::length(centered), 0.f, 1.f), .0001f, .9999f);
492 const float falloff = remap(radial, element.remap[4], element.remap[5], .0001f);
493 const float volume = element.volumeFade ? input.volumeFade : 1.f;
494 const float alpha = element.intensity * remappedAlpha * input.elementParams.a * input.vertexColor.a * falloff * volume;
495 const glm::vec4 selected = seasonal(element, input.season);
496 const float scalar = selected.x * remappedRgb.x * input.elementParams.x * input.vertexColor.r;
497 const float alphaBlend = element.blendAlpha == 0 ? glm::mix(1.f, scalar, alpha) : scalar * alpha;
498
499 if (element.kind == "color-effect") {
500 output.value = {0.f, 0.f, 0.f, alphaBlend};
501 output.colorMask = 1;
502 output.rgbOperation = VegetationScenePixelBlend::Replace;
503 output.alphaOperation = element.blendAlpha == 0 ? VegetationScenePixelBlend::Multiply
504 : VegetationScenePixelBlend::Add;
505 } else if (element.kind == "color-map") {
506 output.value = {selected.r * remappedRgb.r, selected.g * remappedRgb.g, selected.b * remappedRgb.b,
507 selected.a * alpha};
508 } else if (element.kind == "color-noise") {
509 const float minimum = propertyScalar(element, "_NoiseMinValue", 0.f);
510 const float maximum = propertyScalar(element, "_NoiseMaxValue", 1.f);
511 const float noise = maximum == minimum ? (input.noiseSample.r >= maximum ? 1.f : 0.f)
512 : std::clamp((input.noiseSample.r - minimum) / (maximum - minimum), 0.f, 1.f);
513 const glm::vec4 one = propertyVector(element, "_NoiseColorOne", glm::vec4(1.f));
514 const glm::vec4 two = propertyVector(element, "_NoiseColorTwo", glm::vec4(1.f));
515 const glm::vec4 color = glm::mix(one, two, noise);
516 output.value = {color.r, color.g, color.b, color.a * alpha * selected.a};
517 } else if (element.kind == "color-tint") {
518 output.value = {selected.r * remappedRgb.r * input.elementParams.r * input.vertexColor.r,
519 selected.g * remappedRgb.g * input.elementParams.g * input.vertexColor.g,
520 selected.b * remappedRgb.b * input.elementParams.b * input.vertexColor.b,
521 selected.a * alpha};
522 output.alphaOperation = VegetationScenePixelBlend::Add;
523 } else if (element.kind == "extras-alpha") {
524 output.value = {scalar, 0.f, 0.f, alphaBlend};
525 output.colorMask = 1;
526 output.rgbOperation = VegetationScenePixelBlend::Replace;
527 output.alphaOperation = element.blendAlpha == 0 ? VegetationScenePixelBlend::Multiply
528 : VegetationScenePixelBlend::Add;
529 } else if (element.kind == "extras-emissive") {
530 output.value = {scalar, 0.f, 0.f, alpha};
531 output.colorMask = 8;
532 } else if (element.kind == "extras-overlay") {
533 output.value = {0.f, 0.f, scalar, alpha};
534 output.colorMask = 2;
535 } else if (element.kind == "extras-wetness") {
536 output.value = {0.f, scalar, 0.f, alpha};
537 output.colorMask = 4;
538 } else if (element.kind == "motion-wind-power") {
539 output.value = {0.f, 0.f, scalar, alpha};
540 output.colorMask = 2;
541 } else if (element.kind == "motion-advanced") {
542 const glm::vec3 forward3 = rotate(element.rotation, glm::vec3(0.f, 0.f, 1.f));
543 const glm::vec3 texture3 = rotate(
544 element.rotation, glm::vec3(remappedRgb.r * 2.f - 1.f, 0.f, remappedRgb.g * 2.f - 1.f));
545 glm::vec2 direction;
546 switch (element.directionMode) {
547 case 10: direction = {forward3.x, forward3.z}; break;
548 case 20: direction = {texture3.x, texture3.z}; break;
549 case 30: direction = glm::vec2(input.vertexColor) * 2.f - 1.f; break;
550 case 40: direction = input.velocityDirection; break;
551 default:
552 return fail<VegetationSceneElementPixel>(DiagnosticCode::InvalidArgument,
553 "TVE advanced motion direction mode is invalid");
554 }
555 if (element.invertDirection) direction = -direction;
556 const float noiseMinimum = propertyScalar(element, "_NoiseMinValue", 0.f);
557 const float noiseMaximum = propertyScalar(element, "_NoiseMaxValue", 1.f);
558 const float noiseDenominator = noiseMaximum - noiseMinimum;
559 glm::vec2 noise = noiseDenominator == 0.f
560 ? glm::step(glm::vec2(noiseMaximum), glm::vec2(input.noiseSample))
561 : glm::clamp((glm::vec2(input.noiseSample) - noiseMinimum) / noiseDenominator,
562 glm::vec2(0.f), glm::vec2(1.f));
563 const float noiseIntensity = propertyScalar(element, "_NoiseIntensityValue", 0.f);
564 const glm::vec2 encoded = glm::clamp(glm::mix(direction * .5f + .5f, noise, noiseIntensity),
565 glm::vec2(0.f), glm::vec2(1.f));
566 output.value = {encoded.x, encoded.y, element.motionPower,
567 element.intensity * remappedAlpha * input.elementParams.a * input.vertexColor.a * volume};
568 output.colorMask = std::uint8_t(element.motionMode);
569 output.alphaOperation = VegetationScenePixelBlend::Add;
570 } else if (element.kind == "vertex-size") {
571 output.value = {scalar, 0.f, 0.f, alphaBlend};
572 output.colorMask = 1;
573 output.rgbOperation = VegetationScenePixelBlend::Replace;
574 output.alphaOperation = element.blendAlpha == 0 ? VegetationScenePixelBlend::Multiply
575 : VegetationScenePixelBlend::Add;
576 } else if (element.kind == "motion-interaction") {
577 const glm::vec3 localDirection(remappedRgb.r * 2.f - 1.f, 0.f, remappedRgb.g * 2.f - 1.f);
578 glm::vec3 worldDirection = rotate(element.rotation, localDirection);
579 if (element.invertDirection) worldDirection = -worldDirection;
580 output.value = {std::clamp(worldDirection.x * .5f + .5f, 0.f, 1.f),
581 std::clamp(worldDirection.z * .5f + .5f, 0.f, 1.f), element.motionPower,
582 element.intensity * remappedAlpha * input.elementParams.a * input.vertexColor.a * volume};
583 output.colorMask = std::uint8_t(element.motionMode);
584 output.alphaOperation = VegetationScenePixelBlend::Add;
585 } else if (element.kind == "vertex-conform-model") {
586 output.value = {0.f, 0.f, input.worldPosition.y + propertyScalar(element, "_HeightOffsetValue", 0.f),
587 element.intensity * volume};
588 output.colorMask = 2;
589 } else if (element.kind == "vertex-conform-simple") {
590 output.value = {0.f, 0.f, input.mainSample.r * propertyScalar(element, "_HeightValue", 1.f) +
591 propertyScalar(element, "_HeightOffsetValue", 0.f),
592 element.intensity * volume};
593 output.colorMask = 2;
594 } else if (element.kind == "vertex-conform-terrain") {
595 output.value = {0.f, 0.f, input.terrainHeight + propertyScalar(element, "_HeightOffsetValue", 0.f),
596 element.intensity * volume};
597 output.colorMask = 2;
598 } else if (element.kind == "vertex-height-offset") {
599 output.value = {0.f, 0.f, propertyScalar(element, "_HeightOffsetValue", 0.f) * remappedAlpha *
600 element.intensity * volume,
601 0.f};
602 output.colorMask = 2;
603 output.rgbOperation = VegetationScenePixelBlend::Add;
604 } else if (element.kind == "vertex-height") {
605 output.value = {0.f, 0.f, scalar * propertyScalar(element, "_HeightValue", 1.f) +
606 propertyScalar(element, "_HeightOffsetValue", 0.f),
607 alpha};
608 output.colorMask = 2;
609 } else if (element.kind == "vertex-orientation-model") {
610 const glm::vec3 normal = glm::length(input.worldNormal) > 0.f ? glm::normalize(input.worldNormal)
611 : glm::vec3(0.f, 1.f, 0.f);
612 output.value = {normal.x * .5f + .5f, normal.z * .5f + .5f, 0.f, element.intensity * volume};
613 output.colorMask = 12;
614 } else if (element.kind == "vertex-orientation-terrain") {
615 output.value = {input.mainSample.r, input.mainSample.b, 0.f, element.intensity * volume};
616 output.colorMask = 12;
617 } else {
618 return fail<VegetationSceneElementPixel>(DiagnosticCode::Unsupported,
619 "TVE Element pixel shader is not implemented", element.kind);
620 }
621 if (!finite(output.value))
622 return fail<VegetationSceneElementPixel>(DiagnosticCode::InvalidArgument,
623 "vegetation element pixel evaluation overflowed");
624 return Result<VegetationSceneElementPixel>::success(output);
625}
626
627glm::vec4 composeVegetationSceneElementPixel(glm::vec4 destination,
628 const VegetationSceneElementPixel& source) noexcept {
629 auto apply = [&](float dst, float src, VegetationScenePixelBlend operation) {
630 switch (operation) {
631 case VegetationScenePixelBlend::Alpha: return src * source.value.a + dst * (1.f - source.value.a);
632 case VegetationScenePixelBlend::Multiply: return src * dst;
633 case VegetationScenePixelBlend::Add: return src + dst;
634 case VegetationScenePixelBlend::Replace: return src;
635 }
636 return dst;
637 };
638 glm::vec4 result = destination;
639 if (source.colorMask & 8) result.r = apply(destination.r, source.value.r, source.rgbOperation);
640 if (source.colorMask & 4) result.g = apply(destination.g, source.value.g, source.rgbOperation);
641 if (source.colorMask & 2) result.b = apply(destination.b, source.value.b, source.rgbOperation);
642 if (source.colorMask & 1) result.a = apply(destination.a, source.value.a, source.alphaOperation);
643 return result;
644}

646Result<graphics::VegetationChannelAtlas> bakeVegetationSceneChannel(
647 const LoadedVegetationScene& scene, const std::map<std::string, graphics::VegetationMask>& masks,
648 graphics::VegetationChannel channel, const graphics::VegetationChannelAtlas& base, std::uint8_t layer,
649 std::span<const glm::vec3> worldNormals, std::span<const float> terrainHeights,
650 std::span<const glm::vec4> noiseSamples) {
651 const std::uint64_t pixelCount64 = std::uint64_t(base.width) * base.height;
652 if (base.width == 0 || base.height == 0 || base.width > 2048 || base.height > 2048 ||
653 pixelCount64 > 4 * 1024 * 1024 || !std::isfinite(base.center.x) || !std::isfinite(base.center.y) ||
654 !std::isfinite(base.center.z) || !std::isfinite(base.extent.x) || !std::isfinite(base.extent.y) ||
655 !std::isfinite(base.extent.z) || base.extent.x <= 0.f || base.extent.y <= 0.f || base.extent.z <= 0.f ||
656 layer > 8)
657 return fail<graphics::VegetationChannelAtlas>(DiagnosticCode::InvalidArgument,
658 "vegetation scene channel geometry or layer is invalid");
659 const std::size_t pixelCount = std::size_t(pixelCount64);
660 if (base.pixels.size() != pixelCount ||
661 (!worldNormals.empty() && worldNormals.size() != pixelCount) ||
662 (!terrainHeights.empty() && terrainHeights.size() != pixelCount) ||
663 (!noiseSamples.empty() && noiseSamples.size() != pixelCount))
664 return fail<graphics::VegetationChannelAtlas>(DiagnosticCode::InvalidArgument,
665 "vegetation scene channel input sizes are inconsistent");
666 try {
667 graphics::VegetationChannelAtlas output = base;
668 const std::size_t channelIndex = static_cast<std::size_t>(channel);
669 const double stepX = 2.0 * base.extent.x / base.width;
670 const double stepZ = 2.0 * base.extent.z / base.height;
671 for (const auto& element : scene.elements) {
672 if (!element.enabled || element.channel != channelIndex || !(element.layers & (1u << layer))) continue;
673 const graphics::VegetationMask* mask = nullptr;
674 if (!element.textureGuid.empty()) {
675 const auto found = masks.find(element.textureGuid);
676 if (found == masks.end())
677 return fail<graphics::VegetationChannelAtlas>(DiagnosticCode::NotFound,
678 "vegetation scene element mask is missing",
679 element.textureGuid);
680 mask = &found->second;
681 }
682 for (std::uint32_t y = 0; y < base.height; ++y) {
683 for (std::uint32_t x = 0; x < base.width; ++x) {
684 const std::size_t index = std::size_t(y) * base.width + x;
685 const glm::vec3 world(float(double(base.center.x) - base.extent.x + (x + .5) * stepX),
686 base.center.y,
687 float(double(base.center.z) - base.extent.z + (y + .5) * stepZ));
688 const glm::vec3 delta = world - glm::vec3(element.position[0], element.position[1], element.position[2]);
689 const glm::vec3 local = inverseRotate(element.rotation, delta) /
690 glm::vec3(element.scale[0], element.scale[1], element.scale[2]);
691 const glm::vec2 uv(local.x + .5f, local.z + .5f);
692 if (uv.x < 0.f || uv.x > 1.f || uv.y < 0.f || uv.y > 1.f) continue;
693 VegetationSceneElementPixelInput input;
694 input.localUv = uv;
695 const bool flipped = element.kind == "motion-interaction" || element.kind == "motion-advanced" ||
696 element.kind == "vertex-conform-simple" ||
697 element.kind == "vertex-orientation-terrain" ||
698 element.kind == "vertex-height-offset";
699 input.mainSample = mask ? sampleMask(*mask, flipped ? glm::vec2(1.f) - uv : uv) : glm::vec4(1.f);
700 input.worldPosition = world;
701 if (!worldNormals.empty()) input.worldNormal = worldNormals[index];
702 if (!terrainHeights.empty()) input.terrainHeight = terrainHeights[index];
703 if (!noiseSamples.empty()) input.noiseSample = noiseSamples[index];
704 input.season = scene.control.season;
705 if (element.volumeFade) {
706 const glm::vec2 atlasUv((world.x - base.center.x) / (2.f * base.extent.x) + .5f,
707 (world.z - base.center.z) / (2.f * base.extent.z) + .5f);
708 const glm::vec2 edge = glm::abs(atlasUv * 2.002f - 1.001f);
709 if (scene.volume.edgeFade >= 1.f)
710 input.volumeFade = edge.x <= 1.f && edge.y <= 1.f ? 1.f : 0.f;
711 else {
712 const glm::vec2 faded = glm::clamp((edge - scene.volume.edgeFade) /
713 (1.f - scene.volume.edgeFade),
714 glm::vec2(0.f), glm::vec2(1.f));
715 input.volumeFade = 1.f - std::clamp(glm::dot(faded, faded), 0.f, 1.f);
716 }
717 }
718 auto source = evaluateVegetationSceneElementPixel(element, input);
719 if (!source) return Result<graphics::VegetationChannelAtlas>::failure(source.status());
720 output.pixels[index] = composeVegetationSceneElementPixel(output.pixels[index], source.value());
721 }
722 }
723 }
724 return Result<graphics::VegetationChannelAtlas>::success(std::move(output));
725 } catch (const std::bad_alloc&) {
726 return fail<graphics::VegetationChannelAtlas>(DiagnosticCode::Failed,
727 "vegetation scene channel allocation failed");
728 }
729}

799Result<std::unique_ptr<VegetationSceneGpuRuntime>> VegetationSceneGpuRuntime::create(
800 graphics::IResourceFactory& factory, const graphics::PbrSurface& baseSurface,
801 const graphics::VegetationMotion& baseMotion, const LoadedVegetationScene& scene,
802 const std::map<std::string, graphics::VegetationMask>& masks, const VegetationSceneGpuBuild& build) {
803 if (build.sourceRevision == 0)
804 return fail<std::unique_ptr<VegetationSceneGpuRuntime>>(DiagnosticCode::InvalidArgument,
805 "vegetation scene GPU revision must be nonzero");
806 auto projected = projectVegetationScene(baseSurface, baseMotion, scene);
807 if (!projected) return Result<std::unique_ptr<VegetationSceneGpuRuntime>>::failure(projected.status());
808 try {
809 std::array<std::array<graphics::VegetationChannelAtlas, 9>, 4> nativeChannels;
810 for (std::size_t channel = 0; channel < nativeChannels.size(); ++channel) {
811 const auto& input = build.channels[channel];
812 for (std::uint8_t layer = 0; layer < nativeChannels[channel].size(); ++layer) {
813 auto tve = bakeVegetationSceneChannel(scene, masks,
814 static_cast<graphics::VegetationChannel>(channel),
815 input.baseLayers[layer], layer, input.worldNormals,
816 input.terrainHeights, input.noiseSamples);
817 if (!tve) return Result<std::unique_ptr<VegetationSceneGpuRuntime>>::failure(tve.status());
818 auto native = convertVegetationSceneChannelToNative(
819 static_cast<graphics::VegetationChannel>(channel), tve.value());
820 if (!native) return Result<std::unique_ptr<VegetationSceneGpuRuntime>>::failure(native.status());
821 nativeChannels[channel][layer] = std::move(native).takeValue();
822 }
823 }
824
825 auto runtime = std::unique_ptr<VegetationSceneGpuRuntime>(new VegetationSceneGpuRuntime(factory));
826 runtime->projection_ = std::move(projected).takeValue();
827 runtime->sourceRevision_ = build.sourceRevision;
828 auto uploaded = graphics::uploadVegetationGpuFieldSet(factory, nativeChannels[0], nativeChannels[1],
829 nativeChannels[2], nativeChannels[3],
830 build.sourceRevision);
831 if (!uploaded) return Result<std::unique_ptr<VegetationSceneGpuRuntime>>::failure(uploaded.status());
832 runtime->fields_ = std::move(uploaded).takeValue();
833
834 graphics::VegetationGpuRuntime gpu = build.runtime;
835 const auto& motion = runtime->projection_.surface.vegetationMotion;
836 gpu.motionDirection = motion.globalDirection;
837 gpu.worldOrigin = motion.worldOrigin;
838 gpu.time = runtime->projection_.motion.time;
839 gpu.globalBending = motion.globalBending;
840 gpu.globalBranch = motion.globalBranch;
841 gpu.globalFlutter = motion.globalFlutter;
842 gpu.noiseTiling = motion.noiseTiling;
843 gpu.motionFadeDistance = motion.fadeDistance;
844 gpu.cameraFadeMin = runtime->projection_.surface.vegetationAlpha.cameraFadeMin;
845 gpu.cameraFadeMax = runtime->projection_.surface.vegetationAlpha.cameraFadeMax;
846 gpu.fadeNoiseTiling = runtime->projection_.surface.vegetationAlpha.noiseTiling;
847 auto bound = graphics::bindVegetationGpuFields(runtime->projection_.surface, runtime->fields_,
848 build.sourceRevision, gpu);
849 if (!bound) {
850 auto released = runtime->release();
851 if (!released)
852 return fail<std::unique_ptr<VegetationSceneGpuRuntime>>(
853 DiagnosticCode::Failed, "vegetation scene GPU binding rollback failed");
854 return Result<std::unique_ptr<VegetationSceneGpuRuntime>>::failure(bound.status());
855 }
856 return Result<std::unique_ptr<VegetationSceneGpuRuntime>>::success(std::move(runtime));
857 } catch (const std::bad_alloc&) {
858 return fail<std::unique_ptr<VegetationSceneGpuRuntime>>(DiagnosticCode::Failed,
859 "vegetation scene GPU runtime allocation failed");
860 }

863Result<VegetationSceneGpuPublication> VegetationSceneGpuRuntime::replace(
864 std::uint64_t expectedRevision, const graphics::PbrSurface& baseSurface,
865 const graphics::VegetationMotion& baseMotion, const LoadedVegetationScene& scene,
866 const std::map<std::string, graphics::VegetationMask>& masks, const VegetationSceneGpuBuild& build) {
867 if (!factory_)
868 return fail<VegetationSceneGpuPublication>(DiagnosticCode::Conflict,
869 "released vegetation scene runtime cannot be replaced");
870 if (expectedRevision != sourceRevision_)
871 return fail<VegetationSceneGpuPublication>(DiagnosticCode::Conflict,
872 "vegetation scene GPU revision changed before replacement");
873 if (build.sourceRevision <= sourceRevision_)
874 return fail<VegetationSceneGpuPublication>(DiagnosticCode::InvalidArgument,
875 "vegetation scene replacement revision must increase");
876
877 auto candidate = create(*factory_, baseSurface, baseMotion, scene, masks, build);
878 if (!candidate) return Result<VegetationSceneGpuPublication>::failure(candidate.status());
879 try {
880 retiredFields_.reserve(retiredFields_.size() + 1);
881 retiredFields_.push_back(std::move(fields_));
882 } catch (const std::bad_alloc&) {
883 return fail<VegetationSceneGpuPublication>(DiagnosticCode::Failed,
884 "vegetation scene replacement retirement allocation failed");
885 }
886
887 fields_ = std::move(candidate.value()->fields_);
888 std::swap(projection_, candidate.value()->projection_);
889 sourceRevision_ = build.sourceRevision;
890 candidate.value()->factory_ = nullptr;
891
892 for (auto current = retiredFields_.begin(); current != retiredFields_.end();) {
893 auto released = graphics::releaseVegetationGpuFieldSet(*factory_, *current);
894 if (released)
895 current = retiredFields_.erase(current);
896 else
897 ++current;
898 }
899 return Result<VegetationSceneGpuPublication>::success({sourceRevision_, retiredFields_.size()});
900}
