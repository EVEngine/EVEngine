11PropertyType propertyType(const schema::FieldDefinition& field) {
12 if (!field.enumValues.empty()) return PropertyType::Enum;
13 if (!field.ref.empty() || !field.reference.empty()) return PropertyType::ObjectRef;
14 switch (field.type) {
15 case schema::ValueType::Boolean: return PropertyType::Bool;
16 case schema::ValueType::Integer: return PropertyType::Int;
17 case schema::ValueType::Number: return PropertyType::Float;
18 case schema::ValueType::String: return PropertyType::String;
19 case schema::ValueType::Array: return PropertyType::Array;
20 case schema::ValueType::Object: return PropertyType::Struct;
21 case schema::ValueType::Any:
22 case schema::ValueType::Null: return PropertyType::ReadOnlyText;
23 }
24 return PropertyType::ReadOnlyText;
25}

69PropertySchema DefinitionSchemaFormTarget::schema(const SelectionSnapshot&) const {
70 PropertySchema result;
71 if (!schema_) return result;
72 result.typeId = "definition." + schema_->id;
73 result.version = static_cast<std::uint32_t>(schema_->version);
74 for (const schema::FieldDefinition& field : schema_->fields) {
75 PropertyDescriptor descriptor;
76 descriptor.path = PropertyPath(field.name);
77 descriptor.displayNameKey = field.title.empty() ? field.name : field.title;
78 descriptor.descriptionKey = field.description;
79 descriptor.category = "definition";
80 descriptor.type = propertyType(field);
81 descriptor.defaultValue = defaultValue(field);
82 descriptor.numeric.minimum = field.minimum;
83 descriptor.numeric.maximum = field.maximum;
84 descriptor.enumItems = field.enumValues;
85 descriptor.presenterHint = !field.ref.empty() || !field.reference.empty()
86 ? "definition-reference-picker" : "schema-field";
87 if (descriptor.type == PropertyType::ReadOnlyText) descriptor.flags = PropertyFlag::ReadOnly;
88 if (field.required) descriptor.validators.push_back(RuleId("editor.definition.required"));
89 result.properties.push_back(std::move(descriptor));
90 }
91 return result;
92}

94PropertyReadResult DefinitionSchemaFormTarget::read(const SelectionSnapshot& selection,
95 const PropertyPath& path) const {
96 const auto descriptor = schema(selection).find(path);
97 if (!matches(selection) || !descriptor) return {};
98 auto parsed = editorValueFromJson(document_->json());
99 const auto* object = parsed.ok() ? parsed.value().getIf<EditorValue::Object>() : nullptr;
100 if (!object) return {PropertyReadState::Error, {}, parsed.diagnostics()};
101 const auto found = object->find(path.value());
102 if (found == object->end()) return {PropertyReadState::Missing, {}, {}};
103 if (descriptor->type == PropertyType::Float) {
104 if (const auto* integer = found->second.getIf<int64_t>())
105 return {PropertyReadState::Value, static_cast<double>(*integer), {}};
106 }
107 return {PropertyReadState::Value, found->second, {}};
108}

110EditorResult<DomainOperation> DefinitionSchemaFormTarget::makeSet(
111 const SelectionSnapshot& selection, const PropertyPath& path, const EditorValue& value,
112 PropertySetMode mode) const {
113 if (mode == PropertySetMode::Reset) return makeReset(selection, path);
114 if (!matches(selection) || !schema_)
115 return formError(EditorStatus::Rejected, "editor.definition.form-selection",
116 "Definition form selection does not match its document");
117 auto descriptor = schema(selection).find(path);
118 if (!descriptor)
119 return formError(EditorStatus::Unsupported, "editor.definition.form-field",
120 "Definition schema does not contain field: " + path.value());
121 if (hasPropertyFlag(descriptor->flags, PropertyFlag::ReadOnly))
122 return formError(EditorStatus::Rejected, "editor.definition.form-read-only",
123 "Definition field is not directly editable");
124 auto valid = validatePropertyValue(*descriptor, value);
125 if (!valid.ok()) return EditorResult<DomainOperation>::failure(valid.status());
126 return document_->makeSetField(path.value(), value);
127}

138std::vector<EditorDiagnostic> DefinitionSchemaFormTarget::validate() const {
139 std::vector<EditorDiagnostic> diagnostics;
140 if (!document_ || !schema_ || document_->type() != schema_->id || document_->version() != schema_->version) {
141 diagnostics.push_back(eve::editing::ruleDiagnostic(
142 eve::DiagnosticCode::PreconditionViolation,
143 RuleId("editor.definition.form-schema-mismatch"), DiagnosticSeverity::Error,
144 "Definition document and exact schema identity do not match"));
145 return diagnostics;
146 }
147 auto parsed = editorValueFromJson(document_->json());
148 const auto* object = parsed.ok() ? parsed.value().getIf<EditorValue::Object>() : nullptr;
149 if (!object) {
150 diagnostics.push_back(eve::editing::ruleDiagnostic(
151 eve::DiagnosticCode::PreconditionViolation,
152 RuleId("editor.definition.form-object-required"), DiagnosticSeverity::Error,
153 "Schema-driven definition forms require an object payload"));
154 return diagnostics;
155 }
156 const PropertySchema formSchema = schema({});
157 for (const schema::FieldDefinition& field : schema_->fields) {
158 const auto found = object->find(field.name);
159 if (found == object->end()) {
160 if (field.required)
161 diagnostics.push_back(eve::editing::ruleDiagnostic(
162 eve::DiagnosticCode::PreconditionViolation,
163 RuleId("editor.definition.required-field-missing"), DiagnosticSeverity::Error,
164 "Required definition field is missing: " + field.name));
165 continue;
166 }
167 auto descriptor = formSchema.find(PropertyPath(field.name));
168 auto valid = validatePropertyValue(*descriptor, found->second);
169 diagnostics.insert(diagnostics.end(), valid.diagnostics().begin(), valid.diagnostics().end());
170 }
171 return diagnostics;
172}
