82std::string substitute(std::string text, const QuestGenerationArchetype &archetype, const EntityFact &entity,
83 const std::string &giverId, int n) {
84 auto replaceAll = [&](const std::string &token, const std::string &value) {
85 for (;;) {
86 const auto pos = text.find(token);
87 if (pos == std::string::npos) break;
88 text.replace(pos, token.size(), value);
89 }
90 };
91 replaceAll("{archetype}", archetype.id);
92 replaceAll("{entity}", entity.id);
93 replaceAll("{giver}", giverId);
94 replaceAll("{domain}", entity.domain);
95 replaceAll("{n}", std::to_string(n));
96 return text;
97}

99std::string resolveSlot(const std::string &slot, const EntityFact &entity, const std::string &giverId,
100 std::string *error) {
101 if (slot.empty()) return {};
102 if (slot == "entity") return entity.id;
103 if (slot == "giver") return giverId;
104 if (slot == "domain") return entity.domain;
105 constexpr std::string_view kVarPrefix = "var:";
106 if (slot.rfind(kVarPrefix.data(), 0) == 0) {
107 const std::string key = slot.substr(kVarPrefix.size());
108 auto sit = entity.stringVars.find(key);
109 if (sit != entity.stringVars.end()) return sit->second;
110 auto nit = entity.numberVars.find(key);
111 if (nit != entity.numberVars.end()) {
112 std::ostringstream oss;
113 oss << nit->second;
114 return oss.str();
115 }
116 if (error) *error = "missing entity var '" + key + "'";
117 return {};
118 }
119 if (error) *error = "unknown slot '" + slot + "'";
120 return {};
121}

123eve::Result<QuestDefinition> instantiate(const QuestGenerationArchetype &archetype, const EntityFact &entity,
124 const std::string &giverId, int n) {
125 QuestDefinition def;
126 const std::string pattern = archetype.idPattern.empty() ? "gen.{archetype}.{entity}.{n}" : archetype.idPattern;
127 def.id = substitute(pattern, archetype, entity, giverId, n);
128 if (!validContentId(def.id))
129 return eve::Result<QuestDefinition>::failure(eve::Diagnostic::error(
130 eve::DiagnosticCode::InvalidArgument, "generated quest id is not a stable content id", "idPattern",
131 {{"id", def.id}}, "rpg.quest-generation"));
132
133 def.startPolicy = archetype.startPolicy;
134 def.completePolicy = archetype.completePolicy;
135 def.objectivePolicy = archetype.objectivePolicy;
136 def.timeLimitSeconds = archetype.timeLimitSeconds;
137 def.refreshKey = archetype.refreshKey;
138 def.tags = archetype.tags;
139 if (std::find(def.tags.begin(), def.tags.end(), "generated") == def.tags.end()) def.tags.push_back("generated");
140
141 for (const auto &[key, value] : archetype.extra) def.extra[key] = substitute(value, archetype, entity, giverId, n);
142
143 def.extra["generatedArchetype"] = archetype.id;
144 def.extra["generatedEntity"] = entity.id;
145 if (!giverId.empty()) def.extra["generatedGiver"] = giverId;
146 if (!entity.domain.empty()) def.extra["generatedDomain"] = entity.domain;
147
148 for (size_t i = 0; i < archetype.objectives.size(); ++i) {
149 const auto &tmpl = archetype.objectives[i];
150 QuestObjective obj;
151 obj.id = tmpl.id;
152 obj.topic = tmpl.topic;
153 obj.optional = tmpl.optional;
154 obj.count = tmpl.count;
155 if (!tmpl.countVar.empty()) {
156 auto it = entity.numberVars.find(tmpl.countVar);
157 if (it == entity.numberVars.end() || !std::isfinite(it->second) || it->second <= 0.0 ||
158 it->second > static_cast<double>(INT_MAX) || std::floor(it->second) != it->second)
159 return eve::Result<QuestDefinition>::failure(eve::Diagnostic::error(
160 eve::DiagnosticCode::InvalidArgument, "objective countVar must resolve to a positive integer",
161 "objectives[" + std::to_string(i) + "].countVar", {{"var", tmpl.countVar}},
162 "rpg.quest-generation"));
163 obj.count = static_cast<int>(it->second);
164 }
165 if (obj.count <= 0) obj.count = 1;
166
167 std::string slotError;
168 if (!tmpl.targetSlot.empty()) {
169 obj.target = resolveSlot(tmpl.targetSlot, entity, giverId, &slotError);
170 if (!slotError.empty())
171 return eve::Result<QuestDefinition>::failure(eve::Diagnostic::error(
172 eve::DiagnosticCode::InvalidArgument, slotError, "objectives[" + std::to_string(i) + "].targetSlot",
173 {}, "rpg.quest-generation"));
174 } else {
175 obj.target = tmpl.target;
176 }
177 def.objectives.push_back(std::move(obj));
178 }
179
180 for (size_t i = 0; i < archetype.rewards.size(); ++i) {
181 const auto &tmpl = archetype.rewards[i];
182 RewardSpec reward;
183 reward.type = tmpl.type;
184 std::string slotError;
185 if (!tmpl.idSlot.empty()) {
186 reward.id = resolveSlot(tmpl.idSlot, entity, giverId, &slotError);
187 if (!slotError.empty())
188 return eve::Result<QuestDefinition>::failure(
189 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, slotError,
190 "rewards[" + std::to_string(i) + "].idSlot", {}, "rpg.quest-generation"));
191 } else {
192 reward.id = tmpl.id;
193 }
194 reward.amount = tmpl.amount;
195 if (!tmpl.amountVar.empty()) {
196 auto it = entity.numberVars.find(tmpl.amountVar);
197 if (it == entity.numberVars.end() || !std::isfinite(it->second) || it->second <= 0.0)
198 return eve::Result<QuestDefinition>::failure(eve::Diagnostic::error(
199 eve::DiagnosticCode::InvalidArgument, "reward amountVar must resolve to a finite positive number",
200 "rewards[" + std::to_string(i) + "].amountVar", {{"var", tmpl.amountVar}}, "rpg.quest-generation"));
201 reward.amount = it->second;
202 }
203 if (reward.id.empty() || !(reward.amount > 0.0) || !std::isfinite(reward.amount))
204 return eve::Result<QuestDefinition>::failure(eve::Diagnostic::error(
205 eve::DiagnosticCode::InvalidArgument, "generated reward requires id and positive amount",
206 "rewards[" + std::to_string(i) + "]", {}, "rpg.quest-generation"));
207 def.rewards.push_back(std::move(reward));
208 }
209
210 return eve::Result<QuestDefinition>::success(std::move(def));
211}

213eve::Value encodeQuest(const QuestDefinition &def) {
214 eve::Value::Object root;
215 root.emplace("id", eve::Value(def.id));
216 root.emplace("startPolicy", eve::Value(def.startPolicy));
217 root.emplace("completePolicy", eve::Value(def.completePolicy));
218 root.emplace("objectivePolicy", eve::Value(def.objectivePolicy));
219 if (def.timeLimitSeconds > 0.0) root.emplace("timeLimitSeconds", eve::Value(def.timeLimitSeconds));
220 if (!def.refreshKey.empty()) root.emplace("refreshKey", eve::Value(def.refreshKey));
221
222 eve::Value::Array requiresIds;
223 for (const auto &id : def.requiresIds) requiresIds.emplace_back(eve::Value(id));
224 root.emplace("requires", eve::Value(std::move(requiresIds)));
225
226 eve::Value::Array objectives;
227 for (const auto &obj : def.objectives) {
228 eve::Value::Object item;
229 item.emplace("id", eve::Value(obj.id));
230 item.emplace("topic", eve::Value(obj.topic));
231 item.emplace("target", eve::Value(obj.target));
232 item.emplace("count", eve::Value(obj.count));
233 if (obj.optional) item.emplace("optional", eve::Value(true));
234 objectives.emplace_back(std::move(item));
235 }
236 root.emplace("objectives", eve::Value(std::move(objectives)));
237
238 eve::Value::Array rewards;
239 for (const auto &reward : def.rewards) {
240 eve::Value::Object item;
241 item.emplace("type", eve::Value(reward.type));
242 item.emplace("id", eve::Value(reward.id));
243 item.emplace("amount", eve::Value(reward.amount));
244 rewards.emplace_back(std::move(item));
245 }
246 root.emplace("rewards", eve::Value(std::move(rewards)));
247
248 eve::Value::Array tags;
249 for (const auto &tag : def.tags) tags.emplace_back(eve::Value(tag));
250 root.emplace("tags", eve::Value(std::move(tags)));
251
252 eve::Value::Object extra;
253 for (const auto &[key, value] : def.extra) extra.emplace(key, eve::Value(value));
254 root.emplace("extra", eve::Value(std::move(extra)));
255 return eve::Value(std::move(root));
256}

573std::vector<Candidate> buildCandidates(const std::vector<EntityFact> &entities) {
574 std::vector<Candidate> candidates;
575 for (const auto &entity : entities) {
576 for (const auto &id : QuestGenerationCatalogue::ids()) {
577 const QuestGenerationArchetype *archetype = QuestGenerationCatalogue::find(id);
578 if (!archetype) continue;
579 if (!entityHasAllTags(entity, archetype->requiresEntityTags)) continue;
580 Candidate candidate;
581 candidate.archetype = archetype;
582 candidate.entity = &entity;
583 candidate.score = std::max(0.0, entity.urgency) * std::max(0.0, archetype->weight);
584 if (candidate.score <= 0.0) continue;
585 candidates.push_back(candidate);
586 }
587 }
588 return candidates;

591std::vector<Candidate> selectCandidates(std::vector<Candidate> candidates, const std::string &selection, int maxQuests,
592 std::mt19937 &rng) {
593 std::vector<Candidate> chosen;
594 if (candidates.empty() || maxQuests <= 0) return chosen;
595
596 auto takeOne = [&](size_t index) {
597 chosen.push_back(candidates[index]);
598 const std::string entityId = candidates[index].entity->id;
599 candidates.erase(std::remove_if(candidates.begin(), candidates.end(),
600 [&](const Candidate &c) { return c.entity->id == entityId; }),
601 candidates.end());
602 };
603
604 while (!candidates.empty() && static_cast<int>(chosen.size()) < maxQuests) {
605 if (selection == "equal") {
606 std::uniform_int_distribution<size_t> dist(0, candidates.size() - 1);
607 takeOne(dist(rng));
608 continue;
609 }
610 if (selection == "weighted") {
611 double total = 0.0;
612 for (const auto &c : candidates) total += c.score;
613 if (!(total > 0.0)) break;
614 std::uniform_real_distribution<double> dist(0.0, total);
615 double pick = dist(rng);
616 size_t index = 0;
617 for (; index + 1 < candidates.size(); ++index) {
618 pick -= candidates[index].score;
619 if (pick <= 0.0) break;
620 }
621 takeOne(index);
622 continue;
623 }
624 // mostUrgent (default)
625 const auto best = std::max_element(candidates.begin(), candidates.end(),
626 [](const Candidate &a, const Candidate &b) { return a.score < b.score; });
627 takeOne(static_cast<size_t>(best - candidates.begin()));
628 }
629 return chosen;
630}

637eve::Result<GeneratePlan> generatePlan(const std::string &requestJson) {
638 std::string parseError;
639 const eve::json::Document doc = eve::json::Document::parse(requestJson, &parseError);
640 if (!doc.valid())
641 return eve::Result<GeneratePlan>::failure(
642 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, parseError.empty() ? "invalid JSON" : parseError,
643 "$", {}, "rpg.quest-generation"));
644 const Value root = doc.root();
645 if (!root.isObject())
646 return eve::Result<GeneratePlan>::failure(eve::Diagnostic::error(eve::DiagnosticCode::ParseError,
647 "quest generate request must be an object",
648 "$", {}, "rpg.quest-generation"));
649
650 const Value schema = root.get("schema");
651 if (schema && (!schema.isString() || schema.asString() != "eve.rpg.quest-generate"))
652 return eve::Result<GeneratePlan>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
653 "schema must be eve.rpg.quest-generate",
654 "$.schema", {}, "rpg.quest-generation"));
655 const Value version = root.get("version");
656 if (version && (!version.isInt64() || version.asInt64() != 1))
657 return eve::Result<GeneratePlan>::failure(eve::Diagnostic::error(eve::DiagnosticCode::UnknownVersion,
658 "unsupported quest-generate version",
659 "$.version", {}, "rpg.quest-generation"));
660
661 if (QuestGenerationCatalogue::count() == 0)
662 return eve::Result<GeneratePlan>::failure(eve::Diagnostic::error(
663 eve::DiagnosticCode::PreconditionViolation, "no quest generation archetypes are registered", "archetypes",
664 {}, "rpg.quest-generation"));
665
666 std::uint64_t seed = 0;
667 const Value seedValue = root.get("seed");
668 if (seedValue) {
669 if (!seedValue.isInt64() || seedValue.asInt64() < 0)
670 return eve::Result<GeneratePlan>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
671 "seed must be a non-negative integer",
672 "$.seed", {}, "rpg.quest-generation"));
673 seed = static_cast<std::uint64_t>(seedValue.asInt64());
674 }
675
676 int maxQuests = 1;
677 const Value maxValue = root.get("maxQuests");
678 if (maxValue) {
679 if (!maxValue.isInt64() || maxValue.asInt64() <= 0 || maxValue.asInt64() > 64)
680 return eve::Result<GeneratePlan>::failure(
681 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "maxQuests must be an integer in [1, 64]",
682 "$.maxQuests", {}, "rpg.quest-generation"));
683 maxQuests = static_cast<int>(maxValue.asInt64());
684 }
685
686 std::string selection = "mostUrgent";
687 const Value selectionValue = root.get("selection");
688 if (selectionValue) {
689 if (!selectionValue.isString() ||
690 (selectionValue.asString() != "mostUrgent" && selectionValue.asString() != "weighted" &&
691 selectionValue.asString() != "equal"))
692 return eve::Result<GeneratePlan>::failure(eve::Diagnostic::error(
693 eve::DiagnosticCode::InvalidArgument, "selection must be mostUrgent, weighted, or equal", "$.selection",
694 {}, "rpg.quest-generation"));
695 selection = selectionValue.asString();
696 }
697
698 std::string giverId;
699 const Value giver = root.get("giverId");
700 if (giver) {
701 if (!giver.isString() || (!giver.asString().empty() && !validContentId(giver.asString())))
702 return eve::Result<GeneratePlan>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
703 "giverId must be empty or a stable id",
704 "$.giverId", {}, "rpg.quest-generation"));
705 giverId = giver.asString();
706 }
707
708 bool excludeExisting = true;
709 const Value exclude = root.get("excludeExisting");
710 if (exclude) {
711 if (!exclude.isBool())
712 return eve::Result<GeneratePlan>::failure(
713 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "excludeExisting must be a boolean",
714 "$.excludeExisting", {}, "rpg.quest-generation"));
715 excludeExisting = exclude.asBool();
716 }
717
718 const Value entitiesValue = root.get("entities");
719 if (!entitiesValue || !entitiesValue.isArray() || entitiesValue.size() == 0)
720 return eve::Result<GeneratePlan>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
721 "entities must be a non-empty array",
722 "$.entities", {}, "rpg.quest-generation"));
723
724 std::vector<EntityFact> entities;
725 std::unordered_set<std::string> seenEntities;
726 entities.reserve(entitiesValue.size());
727 for (size_t i = 0; i < entitiesValue.size(); ++i) {
728 auto parsed = parseEntity(entitiesValue.at(i), "$.entities[" + std::to_string(i) + "]");
729 if (!parsed.ok()) return eve::Result<GeneratePlan>::failure(parsed.status());
730 EntityFact entity = std::move(parsed).takeValue();
731 if (!seenEntities.insert(entity.id).second)
732 return eve::Result<GeneratePlan>::failure(
733 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "duplicate entity id in generate request",
734 "$.entities[" + std::to_string(i) + "].id", {}, "rpg.quest-generation"));
735 entities.push_back(std::move(entity));
736 }
737
738 auto candidates = buildCandidates(entities);
739 if (candidates.empty())
740 return eve::Result<GeneratePlan>::failure(eve::Diagnostic::error(eve::DiagnosticCode::NotFound,
741 "no entity matched any generation archetype",
742 "$.entities", {}, "rpg.quest-generation"));
743
744 std::mt19937 rng(static_cast<unsigned>(seed & 0xffffffffu) ^ static_cast<unsigned>((seed >> 32) & 0xffffffffu));
745 std::vector<Candidate> selected = selectCandidates(std::move(candidates), selection, maxQuests, rng);
746
747 GeneratePlan plan;
748 std::unordered_set<std::string> usedIds;
749 int n = 1;
750 for (const auto &candidate : selected) {
751 auto built = instantiate(*candidate.archetype, *candidate.entity, giverId, n);
752 if (!built.ok()) return eve::Result<GeneratePlan>::failure(built.status());
753 QuestDefinition def = std::move(built).takeValue();
754 if (!usedIds.insert(def.id).second)
755 return eve::Result<GeneratePlan>::failure(
756 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "generated quest id collided within this batch",
757 "idPattern", {{"id", def.id}}, "rpg.quest-generation"));
758 if (excludeExisting && QuestRegistry::contains(def.id)) continue;
759 plan.ids.push_back(def.id);
760 plan.quests.push_back(std::move(def));
761 ++n;
762 }
763 if (plan.quests.empty())
764 return eve::Result<GeneratePlan>::failure(eve::Diagnostic::error(
765 eve::DiagnosticCode::Conflict, "every generated quest id already exists in the registry", "ids", {},
766 "rpg.quest-generation"));
767 return eve::Result<GeneratePlan>::success(std::move(plan));
768}

772eve::Result<int> QuestGenerationCatalogue::replaceFromJsonStrict(const std::string &json) {
773 std::string parseError;
774 const eve::json::Document doc = eve::json::Document::parse(json, &parseError);
775 if (!doc.valid())
776 return failInt(eve::DiagnosticCode::ParseError, parseError.empty() ? "invalid JSON" : parseError, "$");

778 const Value root = doc.root();
779 std::vector<Value> objects;
780 if (root.isArray()) {
781 if (root.size() == 0)
782 return failInt(eve::DiagnosticCode::InvalidArgument, "archetype catalogue must not be empty", "$");

797 proposed.emplace(archetype.id, std::move(archetype));
798 }
799 archetypeTable() = std::move(proposed);
800 return eve::Result<int>::success(static_cast<int>(archetypeTable().size()));

842eve::Result<int> QuestGenerator::generateAndRegister(const std::string &requestJson) {
843 std::string parseError;
844 const eve::json::Document doc = eve::json::Document::parse(requestJson, &parseError);
845 if (!doc.valid())
846 return failInt(eve::DiagnosticCode::ParseError, parseError.empty() ? "invalid JSON" : parseError, "$");
847 bool replaceGenerated = true;
848 if (doc.root().isObject()) {
849 const Value flag = doc.root().get("replaceGenerated");
850 if (flag) {
851 if (!flag.isBool())
852 return failInt(eve::DiagnosticCode::InvalidArgument, "replaceGenerated must be a boolean",
853 "$.replaceGenerated");
854 replaceGenerated = flag.asBool();
855 }
856 }
857
858 // Remove previous generated board before planning so excludeExisting only sees hand-authored ids.
859 if (replaceGenerated) {
860 const auto existing = QuestRegistry::ids();
861 for (const auto &id : existing) {
862 const QuestDefinition *def = QuestRegistry::find(id);
863 if (def && def->hasTag("generated")) QuestRegistry::remove(id);
864 }
865 }
866
867 auto plan = generatePlan(requestJson);
868 if (!plan.ok()) return eve::Result<int>::failure(plan.status());
869 GeneratePlan value = std::move(plan).takeValue();
870
871 int registered = 0;
872 for (const auto &def : value.quests) {
873 const bool existed = QuestRegistry::contains(def.id);
874 QuestRegistry::registerQuest(def);
875 if (!QuestRegistry::contains(def.id))
876 return failInt(eve::DiagnosticCode::Conflict,
877 "generated quest was rejected by the registry (likely a requires cycle)", def.id);
878 if (!existed) ++registered;
879 }
880 return eve::Result<int>::success(registered);
881}
