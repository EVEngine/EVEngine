40[[nodiscard]] bool raySphereSpan(HexVec3 origin, HexVec3 direction, float radius, float& enter,
41 float& exit) noexcept {
42 const float b = dot3(origin, direction);
43 const float c = dot3(origin, origin) - radius * radius;
44 const float discriminant = b * b - c;
45 if (discriminant < 0.f) return false;
46 const float root = std::sqrt(discriminant);
47 enter = -b - root;
48 exit = -b + root;
49 return exit > 0.f;
50}

56Result<void> HexSphereMap::reset(std::int32_t subdivision, float radius, std::uint32_t seed) {
57 if (subdivision < 0 || subdivision > kMaxHexSphereSubdivision)
58 return invalidArgument("subdivision level is out of range");
59 if (!std::isfinite(radius) || radius <= 0.f) return invalidArgument("sphere radius must be finite and positive");
60
61 auto built = HexSphereTopology::build(subdivision);
62 if (!built.ok()) return Result<void>::failure(built.status());
63
64 topology_ = std::move(built).takeValue();
65 sphereRadius_ = radius;
66 elevationStep_ = radius * kDefaultElevationRatio;
67 noise_.reset(seed);
68 cells_.assign(static_cast<std::size_t>(topology_.cellCount()), HexCellData{});
69 cellDirty_.assign(cells_.size(), 0u);
70 dirtyQueue_.clear();
71 dirtyHead_ = 0;
72 revision_ = 0;
73 markAllDirty();
74 return Result<void>::success();
75}

89float HexSphereMap::cellSpacing() const noexcept {
90 const std::int32_t count = cellCount();
91 if (count < 2) return 0.f;
92 // Mean area per cell is `4 * pi / count` of the sphere, so the mean spacing of a
93 // hexagonal packing of that area is the square root of it.
94 return sphereRadius_ * std::sqrt(4.f * 3.14159265358979f / static_cast<float>(count));
95}

191Result<HexSphereCell> HexSphereMap::pickCell(HexVec3 rayOrigin, HexVec3 rayDirection) const {
192 if (empty())
193 return Result<HexSphereCell>::failure(
194 Diagnostic::error(DiagnosticCode::NotFound, "hex sphere map is empty", "hexmap.sphere.pick"));
195 const float directionLength = length3(rayDirection);
196 if (!(directionLength > 1e-6f))
197 return Result<HexSphereCell>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
198 "ray direction must be non-zero",
199 "hexmap.sphere.pick"));
200 const HexVec3 direction = rayDirection * (1.f / directionLength);
201
202 // The surface is a step function of direction, so the ray is marched against
203 // the bounding sphere and then bisected. Intersecting a single sphere and
204 // refining would return the far-side cell whenever the camera sits between
205 // the surface and that bounding sphere, which is exactly the low-orbit case
206 // this map exists for.
207 const float boundingRadius =
208 sphereRadius_ + static_cast<float>(HexMetrics::kMaxElevation) * elevationStep_;
209 float enter = 0.f;
210 float exit = 0.f;
211 if (!raySphereSpan(rayOrigin, direction, boundingRadius, enter, exit))
212 return Result<HexSphereCell>::failure(
213 Diagnostic::error(DiagnosticCode::NotFound, "ray misses the hex planet", "hexmap.sphere.pick"));
214
215 const float marchStep = std::max(boundingRadius * 0.01f, 1e-4f);
216 float tInside = 0.f;
217 HexSphereCell hit = kNoHexSphereCell;
218 for (float t = std::max(0.f, enter); t <= exit + marchStep; t += marchStep) {
219 const HexVec3 point = rayOrigin + direction * t;
220 const float span = length3(point);
221 if (span <= 1e-6f) continue;
222 const HexSphereCell cell = topology_.cellAt(point);
223 if (cell == kNoHexSphereCell) continue;
224 if (span <= surfaceRadius(cell)) {
225 hit = cell;
226 tInside = t;
227 break;
228 }
229 }
230 if (hit == kNoHexSphereCell)
231 return Result<HexSphereCell>::failure(Diagnostic::error(
232 DiagnosticCode::NotFound, "ray never reached the hex planet surface", "hexmap.sphere.pick"));
233
234 // Bisect between the last sample above the surface and the first one below it.
235 float tOutside = std::max(std::max(0.f, enter), tInside - marchStep);
236 for (std::int32_t iteration = 0; iteration < 24; ++iteration) {
237 const float middle = 0.5f * (tOutside + tInside);
238 const HexVec3 point = rayOrigin + direction * middle;
239 const float span = length3(point);
240 if (span <= 1e-6f) break;
241 const HexSphereCell cell = topology_.cellAt(point);
242 if (cell == kNoHexSphereCell) break;
243 if (span <= surfaceRadius(cell)) {
244 hit = cell;
245 tInside = middle;
246 } else {
247 tOutside = middle;
248 }
249 }
250 return Result<HexSphereCell>::success(hit);
251}

255std::int32_t HexSphereMap::distance(HexSphereCell a, HexSphereCell b) const {
256 if (!contains(a) || !contains(b)) return kNoHexSphereCell;
257 if (a == b) return 0;
258
259 std::vector<std::uint8_t> visited(cells_.size(), 0u);
260 std::vector<HexSphereCell> frontier;
261 std::vector<HexSphereCell> next;
262 frontier.push_back(a);
263 visited[static_cast<std::size_t>(a)] = 1u;
264
265 std::int32_t depth = 0;
266 while (!frontier.empty()) {
267 ++depth;
268 next.clear();
269 for (const HexSphereCell cell : frontier) {
270 const std::int32_t count = topology_.neighborCount(cell);
271 for (std::int32_t d = 0; d < count; ++d) {
272 const HexSphereCell neighbour = topology_.neighbor(cell, d);
273 if (neighbour == kNoHexSphereCell) continue;
274 if (visited[static_cast<std::size_t>(neighbour)]) continue;
275 if (neighbour == b) return depth;
276 visited[static_cast<std::size_t>(neighbour)] = 1u;
277 next.push_back(neighbour);
278 }
279 }
280 frontier.swap(next);
281 }
282 // The neighbour graph of a sphere is connected, so this is unreachable.
283 return kNoHexSphereCell;
284}

286void HexSphereMap::collectBrush(HexSphereCell center, std::int32_t radius,
287 std::vector<HexSphereCell>& out) const {
288 out.clear();
289 if (!contains(center) || radius < 0) return;
290
291 std::vector<std::uint8_t> visited(cells_.size(), 0u);
292 out.push_back(center);
293 visited[static_cast<std::size_t>(center)] = 1u;
294
295 std::size_t head = 0;
296 for (std::int32_t step = 0; step < radius; ++step) {
297 const std::size_t levelEnd = out.size();
298 for (; head < levelEnd; ++head) {
299 const HexSphereCell cell = out[head];
300 const std::int32_t count = topology_.neighborCount(cell);
301 for (std::int32_t d = 0; d < count; ++d) {
302 const HexSphereCell neighbour = topology_.neighbor(cell, d);
303 if (neighbour == kNoHexSphereCell) continue;
304 if (visited[static_cast<std::size_t>(neighbour)]) continue;
305 visited[static_cast<std::size_t>(neighbour)] = 1u;
306 out.push_back(neighbour);
307 }
308 }
309 }

339std::int32_t HexSphereMap::takeDirtyCell() noexcept {
340 while (dirtyHead_ < dirtyQueue_.size()) {
341 const HexSphereCell cell = dirtyQueue_[dirtyHead_++];
342 if (!contains(cell)) continue;
343 if (!cellDirty_[static_cast<std::size_t>(cell)]) continue;
344 cellDirty_[static_cast<std::size_t>(cell)] = 0u;
345 if (dirtyHead_ >= 4096u && dirtyHead_ * 2u >= dirtyQueue_.size()) {
346 dirtyQueue_.erase(dirtyQueue_.begin(),
347 dirtyQueue_.begin() + static_cast<std::ptrdiff_t>(dirtyHead_));
348 dirtyHead_ = 0u;
349 }
350 return cell;
351 }
352 dirtyQueue_.clear();
353 dirtyHead_ = 0u;
354 return kNoHexSphereCell;
355}

364void HexSphereMap::validateRivers(HexSphereCell cell) noexcept {
365 const HexCellData* data = cellAt(cell);
366 if (!data || !data->flags.hasRiver()) return;
367 // A river may only leave a cell that is at least as high as its neighbour,
368 // unless the cell itself is a lake at exactly its own elevation.
369 const std::int32_t fromElevation = elevation(cell);
370 const std::int32_t fromWater = waterLevel(cell);
371 const std::int32_t count = neighborCount(cell);
372 for (std::int32_t d = 0; d < count; ++d) {
373 if (!data->flags.hasRiverOut(static_cast<HexDirection>(d))) continue;
374 const HexSphereCell neighbour = topology_.neighbor(cell, d);
375 if (neighbour == kNoHexSphereCell) continue;
376 const bool canFlow = fromElevation >= elevation(neighbour) || fromWater == fromElevation;
377 if (!canFlow) {
378 removeRiver(cell).ignore("river validation must keep the map consistent");
379 return;
380 }
381 }

384Result<void> HexSphereMap::setElevation(HexSphereCell c, std::int32_t value) {
385 HexCellData* data = mutableCell(c);
386 if (!data) return invalidArgument("cell is outside the hex sphere map");
387 const std::int32_t clamped = clampInt(value, HexMetrics::kMinElevation, HexMetrics::kMaxElevation);
388 if (data->values.elevation() == clamped) return Result<void>::success();
389 data->values = data->values.withElevation(clamped);
390
391 // Roads cannot span a difference of more than one elevation step.
392 bool hasInvalidRoad = false;
393 const std::int32_t count = neighborCount(c);
394 for (std::int32_t d = 0; d < count && !hasInvalidRoad; ++d) {
395 if (!data->flags.hasRoad(static_cast<HexDirection>(d))) continue;
396 const HexSphereCell neighbour = topology_.neighbor(c, d);
397 if (neighbour == kNoHexSphereCell) continue;
398 const std::int32_t delta = elevation(neighbour) - clamped;
399 if (delta > 1 || delta < -1) hasInvalidRoad = true;
400 }
401 if (hasInvalidRoad) removeRoads(c).ignore("elevation change must drop invalid roads");
402 refreshCellDependents(c);
403 validateRivers(c);
404 for (std::int32_t d = 0; d < count; ++d) {
405 const HexSphereCell neighbour = topology_.neighbor(c, d);
406 if (neighbour != kNoHexSphereCell) validateRivers(neighbour);
407 }
408 return Result<void>::success();
409}

499Result<void> HexSphereMap::setOutgoingRiver(HexSphereCell c, std::int32_t direction) {
500 HexCellData* data = mutableCell(c);
501 if (!data) return invalidArgument("cell is outside the hex sphere map");
502 if (direction < 0 || direction >= neighborCount(c))
503 return invalidArgument("river direction is not an edge of the cell");
504 const HexSphereCell neighbour = topology_.neighbor(c, direction);
505 if (neighbour == kNoHexSphereCell) return invalidArgument("river direction has no neighbour");
506 const std::int32_t back = topology_.oppositeDirection(c, direction);
507 if (back < 0) return invalidArgument("river direction has no opposite edge");
508
509 const auto out = static_cast<HexDirection>(direction);
510 const auto in = static_cast<HexDirection>(back);
511 if (data->flags.hasAnyRiverOut() && data->flags.hasRiverOut(out)) return Result<void>::success();
512 if (elevation(c) < elevation(neighbour) && waterLevel(c) != elevation(c))
513 return invalidArgument("a river cannot flow uphill");
514
515 // Drop only the links this call replaces, mirroring the reference cell's
516 // `RemoveOutgoingRiver` + same-edge `RemoveIncomingRiver`. Calling the full
517 // `removeRiver` here also erased the *incoming* link from the upstream cell,
518 // so carving a channel step by step wiped the step behind it and only the
519 // last edge survived.
520 const std::int32_t count = neighborCount(c);
521 for (std::int32_t d = 0; d < count; ++d) {
522 const auto previous = static_cast<HexDirection>(d);
523 if (!data->flags.hasRiverOut(previous)) continue;
524 data->flags = data->flags.withoutRiverOut(previous);
525 const HexSphereCell previousCell = topology_.neighbor(c, d);
526 const std::int32_t previousBack = topology_.oppositeDirection(c, d);
527 if (previousCell != kNoHexSphereCell && previousBack >= 0) {
528 HexCellData* previousData = mutableCell(previousCell);
529 if (previousData) previousData->flags = previousData->flags.withoutRiverIn(static_cast<HexDirection>(previousBack));
530 refreshCellDependents(previousCell);
531 }
532 }
533
534 data = mutableCell(c);
535 // A river may not leave and enter through the same edge.
536 if (data->flags.hasRiverIn(out)) {
537 data->flags = data->flags.withoutRiverIn(out);
538 HexCellData* other = mutableCell(neighbour);
539 if (other) other->flags = other->flags.withoutRiverOut(in);
540 }
541
542 data = mutableCell(c);
543 data->flags = data->flags.withRiverOut(out).withoutRoad(out);
544 data->values = data->values.withSpecialIndex(0);
545 HexCellData* other = mutableCell(neighbour);
546 other->flags = other->flags.withRiverIn(in).withoutRoad(in);
547 other->values = other->values.withSpecialIndex(0);
548 refreshCellDependents(c);
549 refreshCellDependents(neighbour);
550 return Result<void>::success();
551}

553Result<void> HexSphereMap::removeRiver(HexSphereCell c) {
554 HexCellData* data = mutableCell(c);
555 if (!data) return invalidArgument("cell is outside the hex sphere map");
556 if (!data->flags.hasRiver()) return Result<void>::success();
557 for (std::int32_t d = 0; d < kHexDirectionCount; ++d) {
558 const auto direction = static_cast<HexDirection>(d);
559 data->flags = data->flags.withoutRiverIn(direction).withoutRiverOut(direction);
560 }
561 const std::int32_t count = neighborCount(c);
562 for (std::int32_t d = 0; d < count; ++d) {
563 const HexSphereCell neighbour = topology_.neighbor(c, d);
564 const std::int32_t back = topology_.oppositeDirection(c, d);
565 if (neighbour == kNoHexSphereCell || back < 0) continue;
566 const auto in = static_cast<HexDirection>(back);
567 HexCellData* other = mutableCell(neighbour);
568 if (other) other->flags = other->flags.withoutRiverIn(in).withoutRiverOut(in);
569 refreshCellDependents(neighbour);
570 }
571 refreshCellDependents(c);
572 return Result<void>::success();
573}

575Result<void> HexSphereMap::addRoad(HexSphereCell c, std::int32_t direction) {
576 HexCellData* data = mutableCell(c);
577 if (!data) return invalidArgument("cell is outside the hex sphere map");
578 if (direction < 0 || direction >= neighborCount(c))
579 return invalidArgument("road direction is not an edge of the cell");
580 const HexSphereCell neighbour = topology_.neighbor(c, direction);
581 if (neighbour == kNoHexSphereCell) return invalidArgument("road direction has no neighbour");
582 const std::int32_t back = topology_.oppositeDirection(c, direction);
583 if (back < 0) return invalidArgument("road direction has no opposite edge");
584
585 const auto out = static_cast<HexDirection>(direction);
586 const auto in = static_cast<HexDirection>(back);
587 if (data->flags.hasRoad(out)) return Result<void>::success();
588 if (data->flags.hasRiverThrough(out)) return Result<void>::success();
589 if (data->flags.hasRiver() && !data->flags.hasRiverBeginOrEnd()) {
590 // A road may only join a river at a begin or end cell.
591 return Result<void>::success();
592 }
593 HexCellData* other = mutableCell(neighbour);
594 if (other->values.specialIndex() != 0 || data->values.specialIndex() != 0) return Result<void>::success();
595 if (other->flags.hasRiverThrough(in)) return Result<void>::success();
596 const std::int32_t delta = elevation(neighbour) - elevation(c);
597 if (delta > 1 || delta < -1) return Result<void>::success();
598
599 data->flags = data->flags.withRoad(out);
600 other->flags = other->flags.withRoad(in);
601 refreshCellDependents(c);
602 refreshCellDependents(neighbour);
603 return Result<void>::success();
604}

606Result<void> HexSphereMap::removeRoads(HexSphereCell c) {
607 HexCellData* data = mutableCell(c);
608 if (!data) return invalidArgument("cell is outside the hex sphere map");
609 if (!data->flags.hasRoad()) return Result<void>::success();
610 data->flags = data->flags.without(HexFlags::roadMask());
611 const std::int32_t count = neighborCount(c);
612 for (std::int32_t d = 0; d < count; ++d) {
613 const HexSphereCell neighbour = topology_.neighbor(c, d);
614 const std::int32_t back = topology_.oppositeDirection(c, d);
615 if (neighbour == kNoHexSphereCell || back < 0) continue;
616 HexCellData* other = mutableCell(neighbour);
617 if (other) other->flags = other->flags.withoutRoad(static_cast<HexDirection>(back));
618 refreshCellDependents(neighbour);
619 }
620 refreshCellDependents(c);
621 return Result<void>::success();
622}

658Result<void> HexSphereMap::editFeatureLevel(HexSphereCell center, std::int32_t radius, std::int32_t feature,
659 std::int32_t delta) {
660 if (!contains(center)) return invalidArgument("brush centre is outside the hex sphere map");
661 if (radius < 0) return invalidArgument("brush radius must be non-negative");
662 std::vector<HexSphereCell> cells;
663 collectBrush(center, radius, cells);
664 const std::int32_t layer = layerIndex(feature);
665 for (const HexSphereCell cell : cells) {
666 const HexCellData* data = cellAt(cell);
667 if (!data || data->flags.hasRiver()) continue;
668 if (data->values.specialIndex() != 0) continue;
669 const std::int32_t current = layer == 0 ? data->values.urbanLevel()
670 : layer == 1 ? data->values.farmLevel()
671 : data->values.plantLevel();
672 const std::int32_t next = clampInt(current + delta, 0, 3);
673 if (layer == 0)
674 setUrbanLevel(cell, next).ignore("brush urban edit clamps per cell");
675 else if (layer == 1)
676 setFarmLevel(cell, next).ignore("brush farm edit clamps per cell");
677 else
678 setPlantLevel(cell, next).ignore("brush plant edit clamps per cell");
679 }
680 return Result<void>::success();
681}
