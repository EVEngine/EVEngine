63int Cmdline::Doc(std::string symbol, bool noOpen) {
64 // The user/API manual lives on the organization GitHub Pages site.
65 // Doxygen has no stable per-symbol URL, so open the manual root and echo
66 // the requested symbol (useful for headless/agent invocations).
67 string url = "https://evengine.github.io/EVEngine/";
68 if (!symbol.empty())
69 cout << "EVEngine docs for '" << symbol << "': " << url << "\n";
70 else
71 cout << "EVEngine docs: " << url << "\n";
72 if (noOpen) return 0;
73 return openWebPage(url);
