353eve::Result<ProcgenPointSetHandleRef> Procgen::projectToWorldHandle(ProcgenPointSetHandleRef input, float maxY,
354 float minY, std::uint64_t maskBits,
355 bool keepUnmatched) {
356 auto points = resolvePointSet(input);
357 if (!points.isBound())
358 return eve::Result<ProcgenPointSetHandleRef>::failure(
359 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "projectToWorld point-set handle is stale",
360 "points", {}, "procgen.squirrel"));
361 if (!std::isfinite(maxY) || !std::isfinite(minY) || maxY < minY)
362 return eve::Result<ProcgenPointSetHandleRef>::failure(
363 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "projectToWorld requires finite maxY >= minY",
364 "maxY", {}, "procgen.squirrel"));
365 auto* query = eve::cap::query<eve::IProcgenWorldQuery>();
366 if (!query)
367 return eve::Result<ProcgenPointSetHandleRef>::failure(eve::Diagnostic::error(
368 eve::DiagnosticCode::Unsupported, "projectToWorld requires an IProcgenWorldQuery provider", "worldQuery",
369 {}, "procgen.squirrel"));
370
371 PointSet output;
372 output.reserve(points->points().size());
373 for (size_t sourceIndex = 0; sourceIndex < points->points().size(); ++sourceIndex) {
374 const auto& source = points->points()[sourceIndex];
375 auto queryResult = query->projectDown(source.x, source.z, maxY, minY, maskBits);
376 if (!queryResult.ok())
377 return eve::Result<ProcgenPointSetHandleRef>::failure(
378 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "world-query provider failed to execute projection",
379 "worldQuery", {}, "procgen.squirrel"));
380 const auto& hit = queryResult.value();
381 if (!hit.hit) {
382 if (keepUnmatched)
383 std::move(output.appendPointFrom(*points, sourceIndex)).expect("projectToWorld attribute schema");
384 continue;
385 }
386 ProcgenPoint projected = source;
387 projected.x = hit.x;
388 projected.y = hit.y;
389 projected.z = hit.z;
390 projected.normalX = hit.normalX;
391 projected.normalY = hit.normalY;
392 projected.normalZ = hit.normalZ;
393 const int outputIndex =
394 std::move(output.appendPointFrom(*points, sourceIndex)).expect("projectToWorld attribute schema");
395 output.mutablePoint(size_t(outputIndex)) = std::move(projected);
396 output.trySetIntAttribute(outputIndex, "worldObjectId", hit.objectId).expect("projectToWorld metadata schema");
397 }
398 return ownProcgenObject(ownership_->points, std::make_unique<PointSet>(std::move(output)));
399}
