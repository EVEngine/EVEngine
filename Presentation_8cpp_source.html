21PresentationCommand revertible(const BattleEvent& event, const PresentationFrame& frame, SubjectRef subject,
22 UnitVisualState state, PresentationRevertTrigger trigger) {
23 PresentationCommand command;
24 command.intent.sequence = event.sequence;
25 command.intent.subject = subject;
26 command.intent.state = state;
27 command.intent.revision = frame.revision;
28 command.intent.tick = event.tick;
29 command.revert.restingState = restingFor(frame, subject);
30 command.revert.trigger = trigger;
31 command.revert.sequence = event.sequence;
32 if (trigger == PresentationRevertTrigger::OnExpiry)
33 command.intent.expiresAtTick = SimulationTick(event.tick.value() + frame.transientTicks);
34 return command;
35}

72Result<std::vector<PresentationCommand>> PresentationProjector::project(const BattleEvent& event,
73 const BattleCommand* command,
74 const PresentationFrame& frame) const {
75 (void)transientTicks_;
76 std::vector<PresentationCommand> commands;
77 // The event type is the stable protocol spelling owned by the simulation; matching on it
78 // keeps presentation a pure function of the event stream instead of a second opinion.
79 if (event.type == "turn.pending") {
80 commands.push_back(revertible(event, frame, event.subject, UnitVisualState::Selected,
81 PresentationRevertTrigger::OnNextTurn));
82 } else if (event.type == "turn.completed") {
83 // Finishing an activation lasts until the round boundary, which is exactly the kind
84 // of lifetime a view-object API cannot state.
85 commands.push_back(revertible(event, frame, event.subject, UnitVisualState::Finished,
86 PresentationRevertTrigger::OnRoundStart));
87 } else if (event.type == "round.pending") {
88 // The round boundary clears every per-round marker, so it emits no new intent; the
89 // consumer reverts everything this trigger names.
90 return Result<std::vector<PresentationCommand>>::success(std::move(commands));
91 } else if (event.type == "unit.moved") {
92 PresentationCommand moved = revertible(event, frame, event.subject, UnitVisualState::Moving,
93 PresentationRevertTrigger::OnExpiry);
94 if (command != nullptr && command->kind == BattleCommandKind::Move) {
95 moved.intent.from = command->cell;
96 moved.intent.to = command->cell;
97 moved.intent.path = {command->cell};
98 }
99 commands.push_back(std::move(moved));
100 } else if (event.type == "action.declared") {
101 PresentationCommand declared = revertible(event, frame, event.subject, UnitVisualState::Attacking,
102 PresentationRevertTrigger::OnExpiry);
103 if (command != nullptr && command->kind == BattleCommandKind::UseAbility) {
104 declared.intent.to = command->cell;
105 declared.intent.other = command->targetUnit;
106 }
107 commands.push_back(std::move(declared));
108 } else if (event.type == "unit.defeated") {
109 // Terminal: a destroyed unit is not coming back, so the revert is refused by contract.
110 commands.push_back(durable(event, frame, event.subject, UnitVisualState::Destroyed));
111 }
112 return Result<std::vector<PresentationCommand>>::success(std::move(commands));
113}

115Result<PresentationIntent> PresentationProjector::revert(const PresentationIntent& intent,
116 const PresentationRevert& spec) const {
117 if (!spec.isRevertable())
118 return Result<PresentationIntent>::failure(Diagnostic::error(
119 DiagnosticCode::Unsupported, "durable presentation intent cannot be reverted", "presentation.revert"));
120 PresentationIntent inverse;
121 inverse.sequence = spec.sequence;
122 inverse.subject = intent.subject;
123 inverse.other = intent.other;
124 inverse.state = spec.restingState;
125 inverse.from = intent.to;
126 inverse.to = intent.from;
127 inverse.revision = intent.revision;
128 inverse.tick = intent.expiresAtTick;
129 // The inverse is itself a transient instruction, so it must expire too; otherwise the
130 // "undo" would become the next stale state.
131 inverse.expiresAtTick = intent.expiresAtTick;
132 return Result<PresentationIntent>::success(std::move(inverse));
133}
