46 std::uint64_t size) const {
47 if (size > std::numeric_limits<std::size_t>::max() ||
48 offset > static_cast<std::uint64_t>(std::numeric_limits<std::streamoff>::max()) ||

61 return rangeFailure<std::vector<std::uint8_t>>(DiagnosticCode::Failed,
62 "runtime package range is truncated", path_.string());

66Result<EvpackRangeMount> prepareEvpackRangeMount(std::shared_ptr<const EvpackRangeSource> source,
67 const EvpackLimits& limits,
68 const EvpackTrust& trust) {
69 if (!source) return rangeFailure<EvpackRangeMount>(DiagnosticCode::InvalidArgument,
70 "range source is required");
71 auto sourceSize = source->size();
72 if (!sourceSize) return Result<EvpackRangeMount>::failure(sourceSize.status());
73 if (sourceSize.value() < 128 || sourceSize.value() > limits.maximumPackageBytes)
74 return rangeFailure<EvpackRangeMount>(DiagnosticCode::InvalidArgument,
75 "range source size is outside package limits");
76 auto header = source->read(0, 128);
77 if (!header) return Result<EvpackRangeMount>::failure(header.status());
78 if (header.value().size() != 128)
79 return rangeFailure<EvpackRangeMount>(DiagnosticCode::Failed, "range source returned a short header");
80 // Header offsets are still untrusted here; only use them to request the bounded metadata prefix.
81 const std::uint64_t tocOffset = little64(header.value(), 48);
82 const std::uint64_t tocSize = little64(header.value(), 56);
83 const std::uint64_t signatureOffset = little64(header.value(), 80);
84 const std::uint64_t signatureSize = little64(header.value(), 88);
85 if (tocOffset > sourceSize.value() || tocSize > sourceSize.value() - tocOffset ||
86 tocOffset + tocSize > limits.maximumMetadataBytes + 128)
87 return rangeFailure<EvpackRangeMount>(DiagnosticCode::ParseError,
88 "runtime package metadata extent is invalid");
89 std::uint64_t metadataSize = tocOffset + tocSize;
90 if (signatureSize != 0) {
91 if (signatureOffset > sourceSize.value() || signatureSize > sourceSize.value() - signatureOffset)
92 return rangeFailure<EvpackRangeMount>(DiagnosticCode::ParseError,
93 "runtime package signature extent is invalid");
94 metadataSize = std::max(metadataSize, signatureOffset + signatureSize);
95 }
96 auto metadata = source->read(0, metadataSize);
97 if (!metadata) return Result<EvpackRangeMount>::failure(metadata.status());
98 if (metadata.value().size() != metadataSize)
99 return rangeFailure<EvpackRangeMount>(DiagnosticCode::Failed,
100 "range source returned a short metadata prefix");
101 auto index = parseEvpackMetadata(metadata.value(), sourceSize.value(), limits, trust);
102 if (!index) return Result<EvpackRangeMount>::failure(index.status());
103 return Result<EvpackRangeMount>::success(
104 EvpackRangeMount(std::move(source), std::move(index).takeValue(), limits));
105}

107Result<std::vector<std::uint8_t>> EvpackRangeMount::readChunk(std::size_t index) const {
108 if (index >= index_.chunks().size())
109 return rangeFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
110 "chunk index is outside the TOC");
111 const auto& chunk = index_.chunks()[index];
112 if (chunk.storedSize > limits_.maximumChunkBytes)
113 return rangeFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
114 "range chunk exceeds configured budget", chunk.type);
115 auto bytes = source_->read(chunk.offset, chunk.storedSize);
116 if (!bytes) return Result<std::vector<std::uint8_t>>::failure(bytes.status());
117 if (bytes.value().size() != chunk.storedSize)
118 return rangeFailure<std::vector<std::uint8_t>>(DiagnosticCode::Failed,
119 "range source returned a short chunk", chunk.type);
120 auto decoded = decompressEvpackChunk(chunk.codec, bytes.value(), chunk.decodedSize,
121 limits_.maximumDecodedBytes);
122 if (!decoded) return Result<std::vector<std::uint8_t>>::failure(decoded.status());
123 if (sha256(decoded.value()) != chunk.contentHash)
124 return rangeFailure<std::vector<std::uint8_t>>(DiagnosticCode::HashMismatch,
125 "range chunk hash does not match", chunk.type);
126 return decoded;
127}
