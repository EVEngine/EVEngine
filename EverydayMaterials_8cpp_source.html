31eve::Result<std::unique_ptr<PbrTextureSet>> EverydayMaterials::generate(EverydayMaterial family, const Params &params) {
32 using Output = eve::Result<std::unique_ptr<PbrTextureSet>>;
33 const auto index = static_cast<size_t>(family);
34 if (index >= profiles.size() || params.getWidth() > 2048 || params.getHeight() > 2048)
35 return Output::failure(
36 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
37 "Everyday material requires a known family and dimensions at most 2048"));
38 const auto profile = profiles[index];
39 const float detail = control(params, "detail", .5f);
40 const float variation = control(params, "variation", .4f);
41 const float finish = control(params, "finish", .5f);
42 // Material scale is independent of output resolution. Supersample unresolved maps
43 // instead of changing the size or count of physical features.
44 const int repeats = int(std::round(control(params, "scale", 4.f, 1.f, 32.f)));
45 const float du = 1.f / float(params.getWidth()), dv = 1.f / float(params.getHeight());
46 const bool prefilter = std::min(params.getWidth(), params.getHeight()) < repeats * 180;
47 auto sample = [family, seed = params.getSeed(), repeats, du, dv, prefilter](float u, float v) {
48 if (!prefilter) return detail::sampleEveryday(family, u, v, seed, repeats, 0);
49 detail::EverydaySample sum{0.f, 0.f, 0.f};
50 for (float y : {-.25f, .25f})
51 for (float x : {-.25f, .25f}) {
52 const auto s = detail::sampleEveryday(family, u + x * du, v + y * dv, seed, repeats, 0);
53 sum.pigment += s.pigment * .25f;
54 sum.relief += s.relief * .25f;
55 sum.roughness += s.roughness * .25f;
56 }
57 return sum;
58 };
59 TextureRecipeDef def;
60 def.id = "everyday";
61 def.height = [sample, profile, detail](float u, float v, const NoiseField &) {
62 return .5f + (sample(u, v).relief - .5f) * profile.relief * (.25f + 1.5f * detail);
63 };
64 def.colorField = [sample, variation](float u, float v, const NoiseField &) {
65 return .5f + (sample(u, v).pigment - .5f) * (variation * 2.f);
66 };
67 def.roughnessField = [sample](float u, float v, const NoiseField &) { return sample(u, v).roughness; };
68 const Rgba8 tint{static_cast<uint8_t>(control(params, "tintR", profile.color.r / 255.f) * 255.f),
69 static_cast<uint8_t>(control(params, "tintG", profile.color.g / 255.f) * 255.f),
70 static_cast<uint8_t>(control(params, "tintB", profile.color.b / 255.f) * 255.f), 255};
71 if (family == EverydayMaterial::Marble || family == EverydayMaterial::Wood) {
72 auto mineral = [&](float t, float r, float g, float b) {
73 def.albedo.add(t, uint8_t(std::clamp(r * tint.r, 0.f, 255.f)), uint8_t(std::clamp(g * tint.g, 0.f, 255.f)),
74 uint8_t(std::clamp(b * tint.b, 0.f, 255.f)));
75 };
76 if (family == EverydayMaterial::Wood) {
77 mineral(0.f, .25f, .19f, .13f);
78 mineral(.30f, .55f, .50f, .40f);
79 mineral(.50f, 1.f, 1.f, 1.f);
80 mineral(.70f, 1.25f, 1.32f, 1.45f);
81 mineral(1.f, 1.35f, 1.55f, 1.90f);
82 } else {
83 mineral(0.f, .25f, .26f, .27f);
84 mineral(.28f, .63f, .59f, .51f);
85 mineral(.40f, .91f, .86f, .76f);
86 mineral(.50f, 1.f, 1.f, 1.f);
87 mineral(.75f, 1.03f, 1.04f, 1.07f);
88 mineral(1.f, 1.08f, 1.10f, 1.14f);
89 }
90 } else {
91 def.albedo.add(0.f, uint8_t(tint.r * .35f), uint8_t(tint.g * .35f), uint8_t(tint.b * .35f));
92 def.albedo.add(.5f, tint.r, tint.g, tint.b);
93 def.albedo.add(1.f, uint8_t(tint.r + (255 - tint.r) * .2f), uint8_t(tint.g + (255 - tint.g) * .2f),
94 uint8_t(tint.b + (255 - tint.b) * .2f));
95 }
96 const float roughness = std::clamp(profile.roughness + (.5f - finish) * .26f, .025f, 1.f);
97 def.pbr.roughnessLow = std::max(.025f, roughness - .09f);
98 def.pbr.roughnessHigh = std::min(1.f, roughness + .09f);
99 def.pbr.metallic = profile.metallic;
100 def.pbr.normalStrength = float(std::min(params.getWidth(), params.getHeight())) / 80.f;
101 def.pbr.aoStrength = family == EverydayMaterial::Glass ? 0.f : 1.f;
102 Params effective;
103 effective.setSize(params.getWidth(), params.getHeight());
104 effective.setSeed(params.getSeed());
105 effective.setFloat("scale", 1.f);
106 effective.setInt("colors", 256);
107 effective.setBool("seamless", true);
108 std::string error;
109 auto maps = generatePbrSet(def, effective, error);
110 if (!maps) return Output::failure(eve::Diagnostic::error(eve::DiagnosticCode::Failed, error));
111 if (family == EverydayMaterial::Glass) {
112 maps->transmission = .96f;
113 maps->ior = 1.5f;
114 }
115 return Output::success(std::move(maps));
116}
