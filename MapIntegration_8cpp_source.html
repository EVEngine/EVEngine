29MapObjectImportPlan MapObjectImporter::plan(const MapDocumentTarget& document, const StableId& layer,
30 const std::vector<MapObjectImportRecord>& records,
31 const std::string& idPrefix) const {
32 MapObjectImportPlan result;
33 result.documentRevision = document.revision();
34 if (layer.empty() || idPrefix.empty()) {
35 result.status = EditorStatus::Rejected;
36 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
37 eve::DiagnosticCode::InvalidArgument, RuleId("editor.map.invalid-import-target"),
38 DiagnosticSeverity::Error, "Map object import requires a target layer and stable id prefix"));
39 return result;
40 }
41 std::set<std::string> sourceIds;
42 std::set<StableId> existing;
43 for (const auto& placement : document.mapPlacements()) existing.insert(placement.id);
44 for (const auto& record : records) {
45 if (record.sourceId.empty() || record.asset.empty() || !sourceIds.insert(record.sourceId).second) {
46 result.status = EditorStatus::Conflict;
47 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
48 eve::DiagnosticCode::Conflict, RuleId("editor.map.invalid-import-object"), DiagnosticSeverity::Error,
49 "Imported objects require unique source ids and asset references"));
50 return result;
51 }
52 const StableId id(idPrefix + "/" + record.sourceId);
53 if (existing.contains(id)) {
54 result.status = EditorStatus::Conflict;
55 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
56 eve::DiagnosticCode::Conflict, RuleId("editor.map.import-id-conflict"), DiagnosticSeverity::Error,
57 "Imported placement id already exists: " + id.value()));
58 return result;
59 }
60 MapPlacementRecord placement{id,
61 layer,
62 record.asset,
63 record.x,
64 record.y,
65 record.z,
66 record.rotationX,
67 record.rotationY,
68 record.rotationZ,
69 record.scaleX,
70 record.scaleY,
71 record.scaleZ};
72 auto operation = document.makeSetPlacement(placement);
73 if (!operation.ok()) {
74 result.status = operation.code();
75 result.diagnostics.insert(result.diagnostics.end(), operation.diagnostics().begin(),
76 operation.diagnostics().end());
77 return result;
78 }
79 result.operations.push_back(std::move(operation).value());
80 existing.insert(id);
81 }
82 result.status = EditorStatus::Applied;
83 return result;
