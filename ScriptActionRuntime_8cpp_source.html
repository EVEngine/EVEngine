195Result<Value> ScriptActionRuntime::submitAction(const std::string& actionIdText, std::int64_t windupNs,
196 std::int64_t activeNs, std::int64_t recoverNs, std::int64_t tickValue) {
197 auto actionId = LogicalId::parse(actionIdText);
198 if (!actionId) return fail(DiagnosticCode::InvalidArgument, "action id is invalid", "actionId");
199 if (tickValue < 0 || windupNs < 0 || activeNs < 0 || recoverNs < 0)
200 return fail(DiagnosticCode::InvalidArgument, "action timing or tick is invalid", "timing");
201 ActionDefinition definition;
202 definition.id = *actionId;
203 definition.timing.windup = Duration::fromNanoseconds(static_cast<std::uint64_t>(windupNs));
204 definition.timing.active = Duration::fromNanoseconds(static_cast<std::uint64_t>(activeNs));
205 definition.timing.recover = Duration::fromNanoseconds(static_cast<std::uint64_t>(recoverNs));
206 ActionRequest request;
207 request.actionId = *actionId;
208 request.requestedTick = tickOf(tickValue);
209 auto submitted = actions_.submit(std::move(definition), std::move(request));
210 if (!submitted) return propagate(submitted);
211 return applied({{"executionId", static_cast<std::int64_t>(submitted.value().value())}});
212}

214Result<Value> ScriptActionRuntime::advanceAction(std::int64_t executionValue, std::int64_t tickValue, float seconds) {
215 if (executionValue <= 0 || tickValue < 0)
216 return fail(DiagnosticCode::InvalidArgument, "action execution or tick is invalid", "executionId");
217 auto delta = Duration::fromSeconds(seconds);
218 if (!delta) return propagate(delta);
219 auto advanced = actions_.advance(executionOf(executionValue), tickOf(tickValue), delta.value());
220 if (!advanced) return propagate(advanced);
221 Value::Array transitions;
222 for (const auto& transition : advanced.value().transitions) {
223 transitions.emplace_back(Value::Object{{"from", std::string(actionPhaseName(transition.from))},
224 {"to", std::string(actionPhaseName(transition.to))},
225 {"tick", static_cast<std::int64_t>(transition.tick.value())}});
226 }
227 return applied({{"executionId", executionValue},
228 {"phase", std::string(actionPhaseName(advanced.value().phase))},
229 {"elapsedSeconds", advanced.value().totalElapsed.seconds()},
230 {"transitions", std::move(transitions)}});
231}
