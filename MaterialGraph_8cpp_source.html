27MaterialCompileResult MaterialGraphDomain::compile(const GraphDocumentData& graph) const {
28 MaterialCompileResult result;
29 result.documentRevision = graph.revision;
30 if (graph.domain != domain()) {
31 result.status = EditorStatus::Rejected;
32 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
33 eve::DiagnosticCode::InvalidArgument, RuleId("editor.material.wrong-domain"),
34 DiagnosticSeverity::Error, "Graph is not a material domain document"));
35 return result;
36 }
37 const std::size_t outputs =
38 static_cast<std::size_t>(std::count_if(graph.nodes.begin(), graph.nodes.end(), [](const GraphNodeRecord& node) {
39 return node.type == "material.output";
40 }));
41 const bool errorNode = std::any_of(graph.nodes.begin(), graph.nodes.end(),
42 [](const GraphNodeRecord& node) { return node.type == "material.error"; });
43 if (outputs != 1 || errorNode) {
44 result.status = EditorStatus::Failed;
45 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
46 eve::DiagnosticCode::Failed, RuleId("editor.material.compile-failed"), DiagnosticSeverity::Error,
47 outputs != 1 ? "Material graph requires exactly one output node"
48 : "Material graph contains an invalid node"));
49 return result;
50 }
51 result.status = EditorStatus::Applied;
52 result.shaderArtifact = "material-shader:r" + std::to_string(graph.revision) + ":n" +
53 std::to_string(graph.nodes.size()) + ":e" + std::to_string(graph.edges.size());
54 return result;
55}

67EditorResult<TaskId> MaterialEditorService::compileAsync(const DocumentId& document, GraphDocumentData graph,
68 const MaterialGraphDomain& domain, TaskService& tasks) {
69 if (document.empty())
70 return graphError<TaskId>(EditorStatus::Rejected, "editor.material.missing-document",
71 "Material document id is required");
72 const MaterialGraphDomain domainCopy = domain;
73 auto queued = tasks.submit("Compile material " + document.value(), [graph = std::move(graph),
74 domainCopy](const TaskContext& context) {
75 TaskOutcome outcome;
76 if (context.isCancellationRequested()) {
77 outcome.status = EditorStatus::Cancelled;
78 return outcome;
79 }
80 context.reportProgress(0.25F);
81 const MaterialCompileResult compiled = domainCopy.compile(graph);
82 EditorValue::Object value;
83 value["status"] = EditorValue(static_cast<std::int64_t>(compiled.status));
84 value["revision"] = EditorValue(static_cast<std::int64_t>(compiled.documentRevision));
85 value["artifact"] = EditorValue(compiled.shaderArtifact);
86 outcome.status = compiled.status;
87 outcome.output = EditorValue(std::move(value));
88 outcome.diagnostics = compiled.diagnostics;
89 context.reportProgress(1.0F);
90 return outcome;
91 });
92 if (!queued.ok()) return queued;
93 tasks_.emplace(queued.value(), Task{document, {}, &tasks});
94 return queued;
95}

105EditorResult<MaterialCompileResult> MaterialEditorService::result(const TaskId& task) const {
106 auto found = tasks_.find(task);
107 if (found == tasks_.end())
108 return graphError<MaterialCompileResult>(EditorStatus::NotFound, "editor.material.task-not-found",
109 "Material compile task does not exist");
110 if (found->second.service) {
111 const auto snapshot = found->second.service->snapshot(task);
112 if (!snapshot.ok())
113 return graphError<MaterialCompileResult>(EditorStatus::NotFound, "editor.material.task-not-found",
114 "Material compile task does not exist");
115 if (snapshot.value().state == TaskState::Queued || snapshot.value().state == TaskState::Running)
116 return EditorResult<MaterialCompileResult>::success(
117 MaterialCompileResult{}, eve::Status::success(EditorStatus::Pending));
118 MaterialCompileResult compiled;
119 compiled.diagnostics = snapshot.value().diagnostics;
120 if (snapshot.value().state == TaskState::Cancelled) {
121 compiled.status = EditorStatus::Cancelled;
122 return eve::editing::applied<MaterialCompileResult>(std::move(compiled));
123 }
124 const auto* object = snapshot.value().output.getIf<EditorValue::Object>();
125 if (!object)
126 return graphError<MaterialCompileResult>(EditorStatus::Failed, "editor.material.invalid-task-output",
127 "Material worker returned an invalid result");
128 const auto status = object->find("status");
129 const auto revision = object->find("revision");
130 const auto artifact = object->find("artifact");
131 const auto* statusValue = status == object->end() ? nullptr : status->second.getIf<std::int64_t>();
132 const auto* revisionValue = revision == object->end() ? nullptr : revision->second.getIf<std::int64_t>();
133 const auto* artifactValue = artifact == object->end() ? nullptr : artifact->second.getIf<std::string>();
134 if (!statusValue || !revisionValue || !artifactValue)
135 return graphError<MaterialCompileResult>(EditorStatus::Failed, "editor.material.invalid-task-output",
136 "Material worker result fields are invalid");
137 compiled.status = static_cast<EditorStatus>(*statusValue);
138 compiled.documentRevision = static_cast<Revision>(*revisionValue);
139 compiled.shaderArtifact = *artifactValue;
140 return eve::editing::applied<MaterialCompileResult>(std::move(compiled));
141 }
142 return eve::editing::applied<MaterialCompileResult>(found->second.result);
143}

145EditorResult<void> MaterialEditorService::publishPreview(const DocumentId& document, Revision currentRevision,
146 const TaskId& task) {
147 auto found = tasks_.find(task);
148 if (found == tasks_.end() || found->second.document != document)
149 return graphError<void>(EditorStatus::NotFound, "editor.material.task-not-found",
150 "Material compile task does not belong to this document");
151 const auto compileResult = result(task);
152 if (compileResult.code() == EditorStatus::Pending)
153 return graphError<void>(EditorStatus::Conflict, "editor.material.compile-pending",
154 "Material compile task has not completed");
155 if (!compileResult.ok())
156 return graphError<void>(EditorStatus::Failed, "editor.material.compile-result-unavailable",
157 "Material compile result is unavailable");
158 const MaterialCompileResult& compiled = compileResult.value();
159 if (compiled.status != EditorStatus::Applied)
160 return graphError<void>(EditorStatus::Rejected, "editor.material.compile-not-successful",
161 "Failed material output cannot replace the current preview");
162 if (compiled.documentRevision != currentRevision)
163 return graphError<void>(EditorStatus::Conflict, "editor.material.stale-compile",
164 "Material changed after this compile task started");
165 previews_.insert_or_assign(document, compiled.shaderArtifact);
166 return eve::editing::applied<void>();
167}
