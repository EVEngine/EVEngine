31void emitTriangle(MeshBuild& mesh, V3 a, V3 b, V3 c, float u0 = 0.f, float u1 = 1.f, float u2 = 1.f) {
32 const V3 normal = normalized(cross(b - a, c - a));

119Result<MeshBuild> GeometryStroke::buildMeshResult() const {
120 if (points_.size() < 2)
121 return strokeFailure<MeshBuild>(DiagnosticCode::InvalidArgument,
122 "geometry stroke requires at least two accepted points", "points");
123 MeshBuild output;
124 output.setActiveGroup(shape_);
125 std::vector<std::vector<V3>> rings;
126 rings.reserve(points_.size());
127 for (std::size_t index = 0; index < points_.size(); ++index) {
128 const auto& point = points_[index];
129 const V3 center{point.x, point.y, point.z};
130 const auto& previousPoint = points_[index == 0 ? 0 : index - 1];
131 const auto& nextPoint = points_[std::min(index + 1, points_.size() - 1)];
132 V3 forward{nextPoint.x - previousPoint.x, nextPoint.y - previousPoint.y, nextPoint.z - previousPoint.z};
133 forward = normalized(forward);
134 const V3 reference = std::abs(forward.y) < 0.95f ? V3{0.f, 1.f, 0.f} : V3{1.f, 0.f, 0.f};
135 const V3 side = normalized(cross(reference, forward));
136 const V3 up = normalized(cross(forward, side));
137 const float halfWidth = width_ * 0.5f;
138 const float halfDepth = depth_ * 0.5f;
139 if (shape_ == "quad") {
140 rings.push_back({center - side * halfWidth, center + side * halfWidth});
141 } else if (shape_ == "triangularPrism") {
142 rings.push_back({center - side * halfWidth - up * halfDepth, center + side * halfWidth - up * halfDepth,
143 center + up * halfDepth});
144 } else {
145 rings.push_back({center - side * halfWidth - up * halfDepth, center + side * halfWidth - up * halfDepth,
146 center + side * halfWidth + up * halfDepth, center - side * halfWidth + up * halfDepth});
147 }
148 }
149 const std::size_t ringSize = rings.front().size();
150 for (std::size_t index = 1; index < rings.size(); ++index) {
151 if (shape_ == "quad") {
152 emitQuad(output, rings[index - 1][0], rings[index][0], rings[index][1], rings[index - 1][1]);
153 emitQuad(output, rings[index - 1][1], rings[index][1], rings[index][0], rings[index - 1][0]);
154 continue;
155 }
156 for (std::size_t side = 0; side < ringSize; ++side) {
157 const std::size_t next = (side + 1) % ringSize;
158 emitQuad(output, rings[index - 1][side], rings[index][side], rings[index][next], rings[index - 1][next]);
159 }
160 }
161 if (shape_ == "triangularPrism") {
162 emitTriangle(output, rings.front()[2], rings.front()[1], rings.front()[0]);
163 emitTriangle(output, rings.back()[0], rings.back()[1], rings.back()[2]);
164 } else if (shape_ == "cube") {
165 emitQuad(output, rings.front()[3], rings.front()[2], rings.front()[1], rings.front()[0]);
166 emitQuad(output, rings.back()[0], rings.back()[1], rings.back()[2], rings.back()[3]);
167 }
168 output.setMeta("generator", "mesh.geometryStroke");
169 output.setMeta("shape", shape_);
170 output.setMeta("inputSpace", inputSpace_);
171 output.setMeta("pointCount", std::to_string(points_.size()));
172 return Result<MeshBuild>::success(std::move(output));
173}
