20std::vector<std::size_t> elementIds(const asset::CanonicalMeshData& mesh, std::size_t count) {
21 std::vector<std::size_t> parent(count);
22 for (std::size_t i = 0; i < count; ++i) parent[i] = i;
23 auto root = [&](std::size_t value) {
24 while (parent[value] != value) {
25 parent[value] = parent[parent[value]];
26 value = parent[value];
27 }
28 return value;
29 };
30 for (std::size_t i = 0; i + 2 < mesh.indices.size(); i += 3) {
31 const auto a = mesh.indices[i], b = mesh.indices[i + 1], c = mesh.indices[i + 2];
32 if (a >= count || b >= count || c >= count) continue;
33 const auto ra = root(a), rb = root(b), rc = root(c);
34 parent[rb] = ra;
35 parent[rc] = ra;
36 }
37 std::map<std::size_t, std::size_t> ids;
38 std::vector<std::size_t> result(count);
39 for (std::size_t i = 0; i < count; ++i) {
40 const auto r = root(i);
41 auto [found, inserted] = ids.try_emplace(r, ids.size());
42 result[i] = found->second;
43 }
44 return result;
45}

66Result<std::vector<float>> mask(const asset::CanonicalMeshData& mesh, const std::vector<std::string>& rule,
67 const std::map<std::string, VegetationPresetImage>& textures, std::size_t count,
68 float radius, float height, float seed) {
69 std::vector<float> result(count, 1.f);
70 if (rule.empty() || rule[0] == "NONE") return Result<std::vector<float>>::success(std::move(result));
71 if (rule.size() < 2) return failure<std::vector<float>>(DiagnosticCode::ParseError, "mesh mask rule lacks option");
72 auto option = integer(rule[1]);
73 if (!option) return Result<std::vector<float>>::failure(option.status());
74 if (rule[0] == "GET_MASK_FROM_CHANNEL") {
75 auto color = four(mesh, "COLOR_0", count, {1, 1, 1, 1});
76 const int channel = option.value();
77 for (std::size_t i = 0; i < count; ++i) {
78 if (channel < 4)
79 result[i] = color[i * 4 + channel];
80 else {
81 const int uv = (channel - 4) / 4, component = (channel - 4) % 4;
82 auto values = four(mesh, ("_UNITY_UV" + std::to_string(uv)).c_str(), count, {0, 0, 0, 0});
83 result[i] = values[i * 4 + component];
84 }
85 }
86 } else if (rule[0] == "GET_MASK_PROCEDURAL") {
87 const int mode = option.value();
88 const auto elements = (mode == 2 || mode == 3) ? elementIds(mesh, count) : std::vector<std::size_t>{};
89 const auto elementCount =
90 elements.empty() ? std::size_t(0) : *std::max_element(elements.begin(), elements.end()) + 1;
91 for (std::size_t i = 0; i < count; ++i) {
92 const float x = mesh.positions[i * 3], y = mesh.positions[i * 3 + 1], z = mesh.positions[i * 3 + 2];
93 const float h = height > 0 ? y / height : 0.f, radial = radius > 0 ? std::hypot(x, z) / radius : 0.f;
94 switch (mode) {
95 case 0: result[i] = 0; break;
96 case 1: result[i] = 1; break;
97 case 2: {
98 std::uint32_t hash = std::uint32_t(elements[i]) ^ std::bit_cast<std::uint32_t>(seed);
99 hash ^= hash >> 16;
100 hash *= 0x7feb352du;
101 hash ^= hash >> 15;
102 hash *= 0x846ca68bu;
103 hash ^= hash >> 16;
104 result[i] = float(hash & 0x00ffffffu) / float(0x01000000u);
105 break;
106 }
107 case 3:
108 result[i] = std::fmod(float(elements[i]) / std::max<std::size_t>(elementCount, 1) * seed, 1.f);
109 break;
110 case 4: result[i] = clamp01(h); break;
111 case 5: result[i] = clamp01(std::sqrt(x * x + y * y + z * z) / std::max(radius, 1e-6f)); break;
112 case 6: result[i] = clamp01((radial - .1f) / .9f); break;
113 case 7: {
114 const float cap = clamp01((h - .8f) / .2f), base = clamp01(h / .1f);
115 result[i] = clamp01(clamp01((radial - .1f) / .9f) + cap) * base;
116 break;
117 }
118 case 8: result[i] = 1.f - clamp01(h); break;
119 case 9: result[i] = clamp01(-mesh.normals[i * 3 + 1] * .5f + .5f); break;
120 case 10: result[i] = clamp01(mesh.normals[i * 3 + 1] * .5f + .5f); break;
121 case 11: result[i] = clamp01((clamp01(h) - .2f) / .8f); break;
122 case 12: result[i] = clamp01((clamp01(h) - .4f) / .6f); break;
123 case 13: result[i] = clamp01((clamp01(h) - .6f) / .4f); break;
124 case 14: result[i] = 1.f - std::pow(1.f - clamp01(h), 4.f); break;
125 case 15:
126 result[i] = std::pow(clamp01(std::sqrt(x * x + y * y + z * z) / std::max(radius, 1e-6f)), 2.f);
127 break;
128 case 16: result[i] = std::pow(clamp01((radial - .1f) / .9f), 2.f); break;
129 case 17: {
130 const float cap = clamp01((h - .8f) / .2f), base = clamp01(h / .1f);
131 const float capsule = clamp01(clamp01((radial - .1f) / .9f) + cap) * base;
132 result[i] = capsule * capsule;
133 break;
134 }
135 case 18: result[i] = x / std::max(radius, 1e-6f); break;
136 case 19: result[i] = h; break;
137 case 20: result[i] = z / std::max(radius, 1e-6f); break;
138 default:
139 return failure<std::vector<float>>(DiagnosticCode::Unsupported,
140 "unsupported procedural mesh mask mode");
141 }
142 }
143 } else if (rule[0] == "GET_MASK_FROM_TEXTURE") {
144 if (rule.size() < 3)
145 return failure<std::vector<float>>(DiagnosticCode::ParseError, "texture mask lacks property");
146 const auto image = textures.find(rule[2]);
147 if (image == textures.end()) return Result<std::vector<float>>::success(std::move(result));
148 const auto pixels = std::uint64_t(image->second.width) * image->second.height;
149 if (!image->second.width || !image->second.height || pixels > 16ull * 1024ull * 1024ull ||
150 image->second.pixels.size() != std::size_t(pixels) * 4 || option.value() < 0 || option.value() > 3)
151 return failure<std::vector<float>>(DiagnosticCode::InvalidArgument, "invalid texture mask input");
152 int coord = 0;
153 for (std::size_t i = 3; i + 1 < rule.size(); ++i)
154 if (rule[i] == "GET_COORD") {
155 auto parsed = integer(rule[i + 1]);
156 if (!parsed) return Result<std::vector<float>>::failure(parsed.status());
157 coord = parsed.value();
158 }
159 auto uv = four(mesh, ("_UNITY_UV" + std::to_string(coord)).c_str(), count, {0, 0, 0, 0});
160 const auto texel = [&](std::uint32_t x, std::uint32_t y) {
161 return image->second.pixels[(std::size_t(y) * image->second.width + x) * 4 + option.value()] / 255.f;
162 };
163 for (std::size_t i = 0; i < count; ++i) {
164 const float px = std::clamp(uv[i * 4] * image->second.width - .5f, 0.f, float(image->second.width - 1));
165 const float py =
166 std::clamp(uv[i * 4 + 1] * image->second.height - .5f, 0.f, float(image->second.height - 1));
167 const auto x0 = std::uint32_t(px), y0 = std::uint32_t(py), x1 = std::min(x0 + 1, image->second.width - 1),
168 y1 = std::min(y0 + 1, image->second.height - 1);
169 result[i] = std::lerp(std::lerp(texel(x0, y0), texel(x1, y0), px - x0),
170 std::lerp(texel(x0, y1), texel(x1, y1), px - x0), py - y0);
171 }
172 } else if (rule[0] == "GET_MASK_3RD_PARTY") {
173 const auto uv2 = four(mesh, "_UNITY_UV2", count, {0, 0, 0, 0});
174 const auto uv1 = four(mesh, "_UNITY_UV1", count, {0, 0, 0, 0});
175 const auto uv3 = four(mesh, "_UNITY_UV3", count, {0, 0, 0, 0});
176 for (std::size_t i = 0; i < count; ++i) {
177 const float x = mesh.positions[i * 3], y = mesh.positions[i * 3 + 1], z = mesh.positions[i * 3 + 2];
178 if (option.value() == 0 || option.value() == 1) {
179 const float packed = uv2[i * 4], scale = uv2[i * 4 + 1];
180 const float pivotX = std::fmod(packed, 1.f) * 2.f - 1.f;
181 const float pivotZ = std::fmod(32768.f * packed, 1.f) * 2.f - 1.f;
182 const float pivotY = std::sqrt(1.f - clamp01(pivotX * pivotX + pivotZ * pivotZ));
183 if (option.value() == 0) {
184 result[i] = std::sqrt(std::pow(x - pivotX * scale, 2.f) + std::pow(y - pivotY * scale, 2.f) +
185 std::pow(z - pivotZ * scale, 2.f)) /
186 std::max(radius, 1e-6f);
187 } else {
188 auto repeat = [](float value) { return value - std::floor(value); };
189 result[i] = packed < .01f ? 0.f
190 : repeat(pivotX * scale * 33.3f) + repeat(pivotY * scale * 33.3f) +
191 repeat(pivotZ * scale * 33.3f);
192 }
193 } else if (option.value() == 2) {
194 const float ax = uv1[i * 4 + 2] - x, ay = uv1[i * 4 + 3] - y, az = uv2[i * 4 + 3] - z;
195 result[i] = uv3[i * 4 + 3] == 0.f ? clamp01(y / std::max(height, 1e-6f))
196 : std::sqrt(ax * ax + ay * ay + az * az) *
197 (uv1[i * 4 + 3] * uv3[i * 4 + 3]) / std::max(radius, 1e-6f);
198 } else {
199 return failure<std::vector<float>>(DiagnosticCode::Unsupported,
200 "unsupported third-party vegetation mask mode");
201 }
202 }
203 } else {
204 return failure<std::vector<float>>(DiagnosticCode::Unsupported,
205 "mesh mask source requires texture or vendor adapter");
206 }
207 if (rule.size() >= 3 && rule.back().starts_with("ACTION_")) {
208 const auto& action = rule.back();
209 if (action == "ACTION_ONE_MINUS")
210 for (auto& v : result) v = 1.f - v;
211 else if (action == "ACTION_NEGATIVE")
212 for (auto& v : result) v = -v;
213 else if (action == "ACTION_POWER_2")
214 for (auto& v : result) v *= v;
215 else if (action == "ACTION_MULTIPLY_BY_HEIGHT")
216 for (std::size_t i = 0; i < count; ++i)
217 result[i] *= clamp01(mesh.positions[i * 3 + 1] / std::max(height, 1e-6f));
218 else if (action == "ACTION_CLAMP_NEGATIVE_VALUES") {
219 for (std::size_t i = 0; i < count; ++i)
220 if (mesh.positions[i * 3 + 1] < 0) result[i] = 0;
221 } else if (action == "ACTION_FRACTIONAL_VALUES") {
222 for (auto& v : result) v -= std::floor(v);
223 } else if (action == "ACTION_REMAP_01") {
224 auto [lo, hi] = std::minmax_element(result.begin(), result.end());
225 const float range = *hi - *lo;
226 for (auto& v : result) v = range == 0 ? 0 : (v - *lo) / range;
227 } else
228 return failure<std::vector<float>>(DiagnosticCode::Unsupported, "unsupported mesh mask action");
229 }
230 return Result<std::vector<float>>::success(std::move(result));
231}

244Result<void> applyNormals(asset::CanonicalMeshData& mesh, const std::vector<std::string>& rule, float height) {
245 if (rule.size() != 2 || rule[0] != "GET_NORMALS_PROCEDURAL")
246 return failure<void>(DiagnosticCode::Unsupported, "unsupported normal rule");
247 auto mode = integer(rule[1]);
248 if (!mode || mode.value() < 0 || mode.value() > 6)
249 return failure<void>(DiagnosticCode::ParseError, "invalid normal mode");
250 const std::size_t count = mesh.positions.size() / 3;
251 if (mode.value() == 0) {
252 std::fill(mesh.normals.begin(), mesh.normals.end(), 0.f);
253 for (std::size_t i = 0; i + 2 < mesh.indices.size(); i += 3) {
254 const auto a = mesh.indices[i], b = mesh.indices[i + 1], c = mesh.indices[i + 2];
255 if (a >= count || b >= count || c >= count)
256 return failure<void>(DiagnosticCode::InvalidArgument, "mesh index out of range");
257 const float ax = mesh.positions[b * 3] - mesh.positions[a * 3],
258 ay = mesh.positions[b * 3 + 1] - mesh.positions[a * 3 + 1],
259 az = mesh.positions[b * 3 + 2] - mesh.positions[a * 3 + 2];
260 const float bx = mesh.positions[c * 3] - mesh.positions[a * 3],
261 by = mesh.positions[c * 3 + 1] - mesh.positions[a * 3 + 1],
262 bz = mesh.positions[c * 3 + 2] - mesh.positions[a * 3 + 2];
263 const float nx = ay * bz - az * by, ny = az * bx - ax * bz, nz = ax * by - ay * bx;
264 for (auto v : {a, b, c}) {
265 mesh.normals[v * 3] += nx;
266 mesh.normals[v * 3 + 1] += ny;
267 mesh.normals[v * 3 + 2] += nz;
268 }
269 }
270 for (std::size_t i = 0; i < count; ++i)
271 normalize3(mesh.normals[i * 3], mesh.normals[i * 3 + 1], mesh.normals[i * 3 + 2]);
272 return Result<void>::success();
273 }
274 for (std::size_t i = 0; i < count; ++i) {
275 float tx = 0, ty = 1, tz = 0, blend = 1;
276 if (mode.value() <= 3) {
277 blend = mode.value() == 1 ? clamp01(mesh.positions[i * 3 + 1] / std::max(height, 1e-6f))
278 : mode.value() == 2 ? clamp01(clamp01(mesh.positions[i * 3 + 1] / std::max(height, 1e-6f)) + .5f)
279 : 1.f;
280 } else {
281 tx = mesh.positions[i * 3];
282 ty = mesh.positions[i * 3 + 1];
283 tz = mesh.positions[i * 3 + 2];
284 normalize3(tx, ty, tz);
285 blend = mode.value() == 4 ? .5f : mode.value() == 5 ? .75f : 1.f;
286 }
287 mesh.normals[i * 3] = std::lerp(mesh.normals[i * 3], tx, blend);
288 mesh.normals[i * 3 + 1] = std::lerp(mesh.normals[i * 3 + 1], ty, blend);
289 mesh.normals[i * 3 + 2] = std::lerp(mesh.normals[i * 3 + 2], tz, blend);
290 }
291 return Result<void>::success();
292}

294Result<void> recalculateTangents(asset::CanonicalMeshData& mesh) {
295 const auto count = mesh.positions.size() / 3;
296 const auto uv = mesh.texcoords.find(0);
297 if (uv == mesh.texcoords.end()) {
298 mesh.attributes.erase("TANGENT");
299 return Result<void>::success();
300 }
301 if (uv->second.size() != count * 2)
302 return failure<void>(DiagnosticCode::InvalidArgument, "canonical UV0 has an invalid tangent input size");
303 std::vector<float> tan1(count * 3), tan2(count * 3);
304 for (std::size_t k = 0; k + 2 < mesh.indices.size(); k += 3) {
305 const auto a = mesh.indices[k], b = mesh.indices[k + 1], c = mesh.indices[k + 2];
306 if (a >= count || b >= count || c >= count)
307 return failure<void>(DiagnosticCode::InvalidArgument, "mesh index out of range while rebuilding tangents");
308 const float x1 = mesh.positions[b * 3] - mesh.positions[a * 3],
309 x2 = mesh.positions[c * 3] - mesh.positions[a * 3],
310 y1 = mesh.positions[b * 3 + 1] - mesh.positions[a * 3 + 1],
311 y2 = mesh.positions[c * 3 + 1] - mesh.positions[a * 3 + 1],
312 z1 = mesh.positions[b * 3 + 2] - mesh.positions[a * 3 + 2],
313 z2 = mesh.positions[c * 3 + 2] - mesh.positions[a * 3 + 2],
314 s1 = uv->second[b * 2] - uv->second[a * 2], s2 = uv->second[c * 2] - uv->second[a * 2],
315 t1 = uv->second[b * 2 + 1] - uv->second[a * 2 + 1],
316 t2 = uv->second[c * 2 + 1] - uv->second[a * 2 + 1], det = s1 * t2 - s2 * t1;
317 if (std::abs(det) < 1e-12f) continue;
318 const float r = 1.f / det, sx = (x1 * t2 - x2 * t1) * r, sy = (y1 * t2 - y2 * t1) * r,
319 sz = (z1 * t2 - z2 * t1) * r, tx = (x2 * s1 - x1 * s2) * r, ty = (y2 * s1 - y1 * s2) * r,
320 tz = (z2 * s1 - z1 * s2) * r;
321 for (const auto vertex : {a, b, c}) {
322 tan1[vertex * 3] += sx;
323 tan1[vertex * 3 + 1] += sy;
324 tan1[vertex * 3 + 2] += sz;
325 tan2[vertex * 3] += tx;
326 tan2[vertex * 3 + 1] += ty;
327 tan2[vertex * 3 + 2] += tz;
328 }
329 }
330 asset::CanonicalMeshAttribute tangents{4, std::vector<float>(count * 4)};
331 for (std::size_t i = 0; i < count; ++i) {
332 const float nx = mesh.normals[i * 3], ny = mesh.normals[i * 3 + 1], nz = mesh.normals[i * 3 + 2];
333 float tx = tan1[i * 3], ty = tan1[i * 3 + 1], tz = tan1[i * 3 + 2];
334 const float projection = nx * tx + ny * ty + nz * tz;
335 tx -= nx * projection;
336 ty -= ny * projection;
337 tz -= nz * projection;
338 if (std::sqrt(tx * tx + ty * ty + tz * tz) <= 1e-8f) {
339 if (std::abs(ny) < .999f) {
340 tx = nz;
341 ty = 0;
342 tz = -nx;
343 } else {
344 tx = 1;
345 ty = 0;
346 tz = 0;
347 }
348 }
349 normalize3(tx, ty, tz);
350 const float bx = ny * tz - nz * ty, by = nz * tx - nx * tz, bz = nx * ty - ny * tx;
351 tangents.values[i * 4] = tx;
352 tangents.values[i * 4 + 1] = ty;
353 tangents.values[i * 4 + 2] = tz;
354 tangents.values[i * 4 + 3] = bx * tan2[i * 3] + by * tan2[i * 3 + 1] + bz * tan2[i * 3 + 2] < 0 ? -1.f : 1.f;
355 }
356 mesh.attributes["TANGENT"] = std::move(tangents);
357 return Result<void>::success();
358}

361Result<asset::CanonicalMeshData> executeVegetationMeshRules(
362 const VegetationConversionCandidate& candidate, asset::CanonicalMeshData source,
363 const std::map<std::string, VegetationPresetImage>& textures, float variationSeed) {
364 const std::size_t count = source.positions.size() / 3;
365 if (!count || source.positions.size() != count * 3 || source.normals.size() != count * 3 ||
366 !std::isfinite(variationSeed))
367 return failure<asset::CanonicalMeshData>(DiagnosticCode::InvalidArgument, "invalid mesh geometry or seed");
368 try {
369 float radius = 0, height = 0;
370 for (std::size_t i = 0; i < count; ++i) {
371 radius =
372 std::max(radius, std::max(std::abs(source.positions[i * 3]), std::abs(source.positions[i * 3 + 2])));
373 height = std::max(height, std::abs(source.positions[i * 3 + 1]));
374 }
375 auto color = four(source, "COLOR_0", count, {1, 1, 1, 1});
376 auto uv0 = four(source, "_UNITY_UV0", count, {0, 0, 0, 0});
377 auto uv1 = four(source, "_UNITY_UV1", count, {0, 0, 0, 0});
378 auto apply = [&](const char* ruleName, unsigned component) -> Result<void> {
379 auto found = candidate.meshRules.find(ruleName);
380 if (found == candidate.meshRules.end()) return Result<void>::success();
381 auto values = mask(source, found->second, textures, count, radius, height, variationSeed);
382 if (!values) return Result<void>::failure(values.status());
383 for (std::size_t i = 0; i < count; ++i) color[i * 4 + component] = values.value()[i];
384 return Result<void>::success();
385 };
386 for (auto [name, component] : {std::pair<const char*, unsigned>{"SetVariation", 0u},
387 {"SetOcclusion", 1u},
388 {"SetDetailMask", 2u},
389 {"SetHeight", 3u}}) {
390 auto result = apply(name, component);
391 if (!result) return Result<asset::CanonicalMeshData>::failure(result.status());
392 }
393 auto motion2 =
394 candidate.meshRules.contains("SetMotion2")
395 ? mask(source, candidate.meshRules.at("SetMotion2"), textures, count, radius, height, variationSeed)
396 : Result<std::vector<float>>::success(std::vector<float>(count, 1));
397 auto motion3 =
