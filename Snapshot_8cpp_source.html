117StateValue sqToStateValue(HSQUIRRELVM vm, SQInteger idx, SeenSet& seen, int depth, std::string* firstError) {
118 if (depth > 32) return StateValue::null();
119 if (idx < 0) idx = sq_gettop(vm) + idx + 1;
120
121 const SQObjectType t = sq_gettype(vm, idx);
122 switch (t) {
123 case OT_NULL: return StateValue::null();
124 case OT_INTEGER: {
125 SQInteger v = 0;
126 sq_getinteger(vm, idx, &v);
127 return StateValue::integer(static_cast<int64_t>(v));
128 }
129 case OT_FLOAT: {
130 SQFloat v = 0;
131 sq_getfloat(vm, idx, &v);
132 return StateValue::number(static_cast<double>(v));
133 }
134 case OT_BOOL: {
135 SQBool v = SQFalse;
136 sq_getbool(vm, idx, &v);
137 return StateValue::boolean(v != SQFalse);
138 }
139 case OT_STRING: {
140 const SQChar* s = nullptr;
141 sq_getstring(vm, idx, &s);
142 return StateValue::string(s ? std::string(s) : std::string{});
143 }
144 case OT_ARRAY: {
145 const void* id = objectId(vm, idx);
146 if (id && !seen.insert(id).second) return StateValue::null(); // cycle -> null
147 StateValue arr = StateValue::array();
148 const SQInteger top = sq_gettop(vm);
149 const SQInteger size = sq_getsize(vm, idx);
150 for (SQInteger i = 0; i < size; ++i) {
151 sq_pushinteger(vm, i);
152 if (SQ_SUCCEEDED(sq_get(vm, idx))) {
153 arr.pushBack(sqToStateValue(vm, -1, seen, depth + 1, firstError));
154 sq_poptop(vm);
155 } else {
156 arr.pushBack(StateValue::null());
157 }
158 }
159 sq_settop(vm, top);
160 return arr;
161 }
162 case OT_TABLE:
163 case OT_INSTANCE: {
164 const void* id = objectId(vm, idx);
165 if (id && !seen.insert(id).second) return StateValue::null();
166 StateValue obj = StateValue::object();
167 const SQInteger top = sq_gettop(vm);
168 sq_pushnull(vm);
169 while (SQ_SUCCEEDED(sq_next(vm, idx))) {
170 if (sq_gettype(vm, -2) == OT_STRING) {
171 const SQChar* key = nullptr;
172 sq_getstring(vm, -2, &key);
173 if (key) {
174 const SQObjectType vt = sq_gettype(vm, -1);
175 if (vt == OT_CLOSURE || vt == OT_NATIVECLOSURE || vt == OT_CLASS || vt == OT_INSTANCE ||
176 vt == OT_USERDATA || vt == OT_USERPOINTER || vt == OT_THREAD || vt == OT_GENERATOR ||
177 vt == OT_WEAKREF) {
178 if (firstError && firstError->empty()) {
179 *firstError = std::string("'") + key + "' (" + sqTypeName(vt) + ")";
180 }
181 } else {
182 obj.set(std::string(key), sqToStateValue(vm, -1, seen, depth + 1, firstError));
183 }
184 }
185 }
186 sq_pop(vm, 2);
187 }
188 sq_settop(vm, top);
189 return obj;
190 }
191 default: return StateValue::null();
192 }
193}

195bool pushStateValue(HSQUIRRELVM vm, const StateValue& var, int depth) {
196 if (depth > 32) {
197 sq_pushnull(vm);
198 return true;
199 }
200 switch (var.kind()) {
201 case StateValue::Kind::Null: sq_pushnull(vm); return true;
202 case StateValue::Kind::Bool: sq_pushbool(vm, var.asBool() ? SQTrue : SQFalse); return true;
203 case StateValue::Kind::Int: sq_pushinteger(vm, static_cast<SQInteger>(var.asInt())); return true;
204 case StateValue::Kind::Float: sq_pushfloat(vm, static_cast<SQFloat>(var.asDouble())); return true;
205 case StateValue::Kind::String: {
206 const std::string& s = var.asString();
207 sq_pushstring(vm, s.c_str(), static_cast<SQInteger>(s.size()));
208 return true;
209 }
210 case StateValue::Kind::Array:
211 sq_newarray(vm, 0);
212 for (size_t i = 0; i < var.arraySize(); ++i) {
213 if (!pushStateValue(vm, var.at(i), depth + 1)) return false;
214 sq_arrayappend(vm, -2);
215 }
216 return true;
217 case StateValue::Kind::Object:
218 sq_newtable(vm);
219 for (const auto& key : var.keys()) {
220 sq_pushstring(vm, key.c_str(), static_cast<SQInteger>(key.size()));
221 if (!pushStateValue(vm, *var.find(key), depth + 1)) return false;
222 sq_newslot(vm, -3, SQFalse);
223 }
224 return true;
225 }
226 return false;
227}

229Poco::Dynamic::Var stateToVar(const StateValue& var) {
230 switch (var.kind()) {
231 case StateValue::Kind::Null: return Poco::Dynamic::Var();
232 case StateValue::Kind::Bool: return Poco::Dynamic::Var(var.asBool());
233 case StateValue::Kind::Int: return Poco::Dynamic::Var(static_cast<Poco::Int64>(var.asInt()));
234 case StateValue::Kind::Float: return Poco::Dynamic::Var(var.asDouble());
235 case StateValue::Kind::String: return Poco::Dynamic::Var(var.asString());
236 case StateValue::Kind::Array: {
237 Poco::JSON::Array::Ptr arr(new Poco::JSON::Array());
238 for (size_t i = 0; i < var.arraySize(); ++i) arr->add(stateToVar(var.at(i)));
239 return Poco::Dynamic::Var(arr);
240 }
241 case StateValue::Kind::Object: {
242 Poco::JSON::Object::Ptr obj(new Poco::JSON::Object());
243 for (const auto& key : var.keys()) obj->set(key, stateToVar(*var.find(key)));
244 return Poco::Dynamic::Var(obj);
245 }
246 }
247 return Poco::Dynamic::Var();
248}

250StateValue varToState(const Poco::Dynamic::Var& var) {
251 if (var.isEmpty()) return StateValue::null();
252 if (var.isBoolean()) return StateValue::boolean(var.convert<bool>());
253 if (var.isInteger()) return StateValue::integer(static_cast<int64_t>(var.convert<Poco::Int64>()));
254 if (var.isNumeric()) return StateValue::number(var.convert<double>());
255 if (var.isString()) return StateValue::string(var.convert<std::string>());
256
257 try {
258 if (var.type() == typeid(Poco::JSON::Array::Ptr)) {
259 StateValue arr = StateValue::array();
260 Poco::JSON::Array::Ptr a = var.extract<Poco::JSON::Array::Ptr>();
261 if (a)
262 for (size_t i = 0; i < a->size(); ++i)
263 arr.pushBack(varToState(a->get(static_cast<unsigned int>(i))));
264 return arr;
265 }
266 } catch (const Poco::BadCastException&) {
267 // fall through
268 }
269
270 try {
271 if (var.type() == typeid(Poco::JSON::Object::Ptr)) {
272 StateValue obj = StateValue::object();
273 Poco::JSON::Object::Ptr o = var.extract<Poco::JSON::Object::Ptr>();
274 if (o)
275 for (const auto& name : o->getNames()) obj.set(name, varToState(o->get(name)));
276 return obj;
277 }
278 } catch (const Poco::BadCastException&) {
279 // fall through
280 }
281 return StateValue::null();
282}

309std::vector<std::string> Snapshot::resolveRoots(HSQUIRRELVM vm) const {
310 if (!marked_.empty()) return marked_;
311 std::vector<std::string> out;
312 if (!vm) return out;
313
314 for (const char* pref : {"eve_state", "gameState", "state"}) {
315 const SQInteger top = sq_gettop(vm);
316 sq_pushroottable(vm);
317 sq_pushstring(vm, pref, -1);
318 if (SQ_SUCCEEDED(sq_get(vm, -2))) {
319 const SQObjectType t = sq_gettype(vm, -1);
320 if (t == OT_TABLE || t == OT_INSTANCE || t == OT_ARRAY) out.emplace_back(pref);
321 }
322 sq_settop(vm, top);
323 }
324 if (!out.empty()) return out;
325
326 const SQInteger top = sq_gettop(vm);
327 sq_pushroottable(vm);
328 const SQInteger rootIdx = sq_gettop(vm);
329 sq_pushnull(vm);
330 while (SQ_SUCCEEDED(sq_next(vm, rootIdx))) {
331 if (sq_gettype(vm, -2) == OT_STRING) {
332 const SQChar* key = nullptr;
333 sq_getstring(vm, -2, &key);
334 if (key && !isEngineName(key)) {
335 const SQObjectType vt = sq_gettype(vm, -1);
336 if (vt == OT_INTEGER || vt == OT_FLOAT || vt == OT_BOOL || vt == OT_STRING || vt == OT_TABLE ||
337 vt == OT_ARRAY) {
338 out.emplace_back(key);
339 }
340 }
341 }
342 sq_pop(vm, 2);
343 }
344 sq_settop(vm, top);
345 return out;
346}

348bool Snapshot::captureState(HSQUIRRELVM vm, StateValue& out, std::string* error) const {
349 if (!vm) {
350 if (error) *error = "no vm";
351 return false;
352 }
353 try {
354 auto rootNames = resolveRoots(vm);
355 StateValue roots = StateValue::object();
356 for (const auto& name : rootNames) {
357 SeenSet seen;
358 const SQInteger top = sq_gettop(vm);
359 sq_pushroottable(vm);
360 sq_pushstring(vm, name.c_str(), -1);
361 if (SQ_SUCCEEDED(sq_get(vm, -2))) {
362 const SQObjectType rt = sq_gettype(vm, -1);
363 if (rt == OT_CLOSURE || rt == OT_NATIVECLOSURE || rt == OT_CLASS || rt == OT_INSTANCE ||
364 rt == OT_USERDATA || rt == OT_USERPOINTER || rt == OT_THREAD || rt == OT_GENERATOR ||
365 rt == OT_WEAKREF) {
366 sq_settop(vm, top);
367 if (error) {
368 *error = "root '" + name + "' is a " + sqTypeName(rt) + " (state roots must be plain data)";
369 }
370 return false;
371 }
372 std::string firstError;
373 roots.set(name, sqToStateValue(vm, -1, seen, 0, &firstError));
374 if (!firstError.empty()) {
375 sq_settop(vm, top);
376 if (error) {
377 *error = "root '" + name + "' contains non-serializable value: " + firstError;
378 }
379 return false;
380 }
381 }
382 sq_settop(vm, top);
383 }
384
385 StateValue native = StateValue::object();
386 eve::cap::forEach<eve::caps::IStateProvider>([&](eve::caps::IStateProvider* p) {
387 StateValue captured;
388 if (p->captureState(captured)) native.set(p->stateKind(), std::move(captured));
389 });
390
391 out = StateValue::object();
392 out.set("version", StateValue::integer(2));
393 out.set("roots", std::move(roots));
394 out.set("native", std::move(native));
395 return true;
396 } catch (const std::exception& e) {
397 if (error) *error = e.what();
398 return false;
399 }

419bool Snapshot::restoreState(HSQUIRRELVM vm, const StateValue& state, std::string* error) const {
420 if (!vm) {
421 if (error) *error = "no vm";
422 return false;
423 }
424 try {
425 const StateValue* roots = state.find("roots");
426 if (!roots || !roots->isObject()) {
427 if (error) *error = "invalid snapshot: missing roots";
428 return false;
429 }
430 for (const auto& name : roots->keys()) {
431 const SQInteger top = sq_gettop(vm);
432 sq_pushroottable(vm);
433 sq_pushstring(vm, name.c_str(), static_cast<SQInteger>(name.size()));
434 if (!pushStateValue(vm, *roots->find(name), 0)) {
435 sq_settop(vm, top);
436 if (error) *error = "failed to restore " + name;
437 return false;
438 }
439 sq_newslot(vm, -3, SQFalse);
440 sq_settop(vm, top);
441 }
442
443 const StateValue* native = state.find("native");
444 if (native && native->isObject()) {
445 bool failed = false;
446 std::string nativeErr;
447 eve::cap::forEach<eve::caps::IStateProvider>([&](eve::caps::IStateProvider* p) {
448 const StateValue* v = native->find(p->stateKind());
449 if (!v) return; // provider not present in the snapshot
450 std::string perr;
451 if (!p->restoreState(*v, &perr)) {
452 failed = true;
453 nativeErr += std::string(p->stateKind()) + ": " + perr + "; ";
454 p->resetToDefaults();
455 }
456 });
457 if (failed) {
458 if (error) *error = nativeErr;
459 return false;
460 }
461 }
462 return true;
463 } catch (const std::exception& e) {
464 if (error) *error = e.what();
465 return false;
466 }
467}
