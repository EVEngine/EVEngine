40eve::Result<std::vector<AccountCost>> buildCosts(const DialogueAccountBindings& bindings, const PaymentSpec& payment) {
41 auto valid = payment.validate();
42 if (!valid) return eve::Result<std::vector<AccountCost>>::failure(valid.status());
43
44 struct Raw {
45 eve::resource::IResourceAccount* account;
46 const std::string* resource;
47 std::optional<std::int64_t> amount;
48 const char* path;
49 };
50 const std::array<Raw, 2> raw{{
51 {bindings.money, &bindings.moneyResource, payment.money, "money"},
52 {bindings.reputation, &bindings.reputationResource, payment.reputation, "reputation"},
53 }};
54
55 std::vector<AccountCost> result;
56 for (const auto& item : raw) {
57 if (!item.amount) continue;
58 if (item.account == nullptr)
59 return failure<std::vector<AccountCost>>(
60 eve::DiagnosticCode::Unsupported,
61 std::string("dialogue ") + item.path + " payment has no account adapter", item.path);
62 auto costItem = eve::resource::ResourceCost::create(*item.resource, *item.amount);
63 if (!costItem) return eve::Result<std::vector<AccountCost>>::failure(costItem.status());
64
65 auto found = std::find_if(result.begin(), result.end(),
66 [&](const AccountCost& candidate) { return candidate.account == item.account; });
67 if (found == result.end()) {
68 std::vector<eve::resource::ResourceCost> items;
69 items.push_back(std::move(costItem).takeValue());
70 auto cost = eve::resource::CostSpec::create(std::move(items));
71 if (!cost) return eve::Result<std::vector<AccountCost>>::failure(cost.status());
72 result.push_back({item.account, std::move(cost).takeValue()});
73 } else {
74 std::vector<eve::resource::ResourceCost> items = found->cost.items();
75 items.push_back(std::move(costItem).takeValue());
76 auto cost = eve::resource::CostSpec::create(std::move(items));
77 if (!cost) return eve::Result<std::vector<AccountCost>>::failure(cost.status());
78 found->cost = std::move(cost).takeValue();
79 }
80 }
81 return eve::Result<std::vector<AccountCost>>::success(std::move(result));
82}

99eve::Result<PaymentSpec> PaymentSpec::fromValue(const eve::Value& value) {
100 if (!value.isObject())
101 return failure<PaymentSpec>(eve::DiagnosticCode::InvalidArgument, "dialogue payment must be an object",
102 "payment");
103 PaymentSpec result;
104 for (const auto& key : value.keys()) {
105 const eve::Value* member = value.find(key);
106 if (key == "money") {
107 auto amount = positiveInteger(*member, "money");
108 if (!amount) return eve::Result<PaymentSpec>::failure(amount.status());
109 result.money = std::move(amount).takeValue();
110 } else if (key == "reputation") {
111 auto amount = positiveInteger(*member, "reputation");
112 if (!amount) return eve::Result<PaymentSpec>::failure(amount.status());
113 result.reputation = std::move(amount).takeValue();
114 } else {
115 return failure<PaymentSpec>(eve::DiagnosticCode::InvalidArgument, "unknown dialogue payment field",
116 "payment." + key);
117 }
118 }
119 return eve::Result<PaymentSpec>::success(std::move(result));
120}

168eve::Result<eve::transaction::TransactionReceipt> DialoguePaymentAdapter::execute(
169 std::string transactionId, const PaymentSpec& payment,
170 std::span<eve::transaction::ITransactionParticipant*> effects) const {
171 if (transactionId.empty())
172 return failure<eve::transaction::TransactionReceipt>(
173 eve::DiagnosticCode::InvalidArgument, "dialogue payment transaction id must not be empty", "transactionId");
174
175 for (std::size_t index = 0; index < effects.size(); ++index) {
176 if (effects[index] == nullptr)
177 return failure<eve::transaction::TransactionReceipt>(eve::DiagnosticCode::InvalidArgument,
178 "dialogue payment effect must not be null",
179 "effects[" + std::to_string(index) + "]");
180 for (std::size_t prior = 0; prior < index; ++prior) {
181 if (effects[prior] == effects[index])
182 return failure<eve::transaction::TransactionReceipt>(eve::DiagnosticCode::Conflict,
183 "dialogue payment effects must be unique",
184 "effects[" + std::to_string(index) + "]");
185 }
186 }
187
188 auto costs = buildCosts(bindings_, payment);
189 if (!costs) return eve::Result<eve::transaction::TransactionReceipt>::failure(costs.status());
190 auto accountCosts = std::move(costs).takeValue();
191
192 std::vector<std::unique_ptr<eve::transaction::ResourceDebitParticipant>> debitParticipants;
193 debitParticipants.reserve(accountCosts.size());
194 std::vector<eve::transaction::ITransactionParticipant*> participants;
195 participants.reserve(accountCosts.size() + effects.size());
196 for (std::size_t index = 0; index < accountCosts.size(); ++index) {
197 debitParticipants.push_back(std::make_unique<eve::transaction::ResourceDebitParticipant>(
198 *accountCosts[index].account, std::move(accountCosts[index].cost),
199 "dialogue.resource-debit[" + std::to_string(index) + "]"));
200 participants.push_back(debitParticipants.back().get());
201 }
202 for (auto* effect : effects) participants.push_back(effect);
203 if (participants.empty())
204 return failure<eve::transaction::TransactionReceipt>(eve::DiagnosticCode::InvalidArgument,
205 "dialogue payment has no account or effect", "payment");
206
207 eve::transaction::Coordinator coordinator;
208 return coordinator.execute(eve::transaction::TransactionContext(std::move(transactionId)), participants);
209}
