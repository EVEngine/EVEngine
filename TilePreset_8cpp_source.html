78std::uint16_t toTilePresetConfiguration(std::uint8_t mask) noexcept {
79 std::uint16_t c = 1u << 4u;
80 if (mask & (1u << 0u)) c |= 1u << 1u; // N -> top
81 if (mask & (1u << 1u)) c |= 1u << 5u; // E -> right
82 if (mask & (1u << 2u)) c |= 1u << 7u; // S -> bottom
83 if (mask & (1u << 3u)) c |= 1u << 3u; // W -> left
84 if (mask & (1u << 4u)) c |= 1u << 2u; // NE -> top-right
85 if (mask & (1u << 5u)) c |= 1u << 8u; // SE -> bottom-right
86 if (mask & (1u << 6u)) c |= 1u << 6u; // SW -> bottom-left
87 if (mask & (1u << 7u)) c |= 1u << 0u; // NW -> top-left
88 return c;
89}

91TilePresetVariant resolveTilePreset(std::uint8_t mask) noexcept {
92 const auto c = toTilePresetConfiguration(mask);
