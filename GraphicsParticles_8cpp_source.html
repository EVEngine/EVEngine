156void Graphics::releaseGpuParticleEmitter(GpuParticleHandle handle) {
157 auto found = gpuParticles_.find(handle);
158 if (found == gpuParticles_.end()) return;
159 waitForSharedGpuResources();
160 std::vector<vk::DescriptorSet> sets;

181bool Graphics::updateGpuParticleEmitter(GpuParticleHandle handle, const GpuParticleUpdate& update,
182 const GpuParticleSpawn* spawns, std::uint32_t spawnCount) {
183 if (!canSubmitGpuParticles()) return false;
184 auto found = gpuParticles_.find(handle);
185 if (found == gpuParticles_.end()) return false;
186 auto& resource = *found->second;
187 resource.update = update;
188 const std::uint32_t accepted = std::min(spawnCount, resource.capacity);
189 if (accepted > 0 && spawns)
190 resource.spawns.assign(spawns, spawns + accepted);
191 else
192 resource.spawns.clear();
193 resource.pendingUpdate = true;
194 return true;
195}

212void Graphics::createGpuParticlePipelines() {
213 if (gpuParticleComputePipeline_ || !descriptorPool || !renderpass) return;
214
215 std::array<vk::DescriptorSetLayoutBinding, 5> computeBindings{};
216 for (std::uint32_t binding = 0; binding < computeBindings.size(); ++binding) {
217 computeBindings[binding] = {binding, vk::DescriptorType::eStorageBuffer, 1, vk::ShaderStageFlagBits::eCompute};
218 }
219 gpuParticleComputeSetLayout_ =
220 device->createDescriptorSetLayout(vk::DescriptorSetLayoutCreateInfo({}, computeBindings));
221
222 const std::array drawBindings{
223 vk::DescriptorSetLayoutBinding(0, vk::DescriptorType::eCombinedImageSampler, 1,
224 vk::ShaderStageFlagBits::eFragment),
225 vk::DescriptorSetLayoutBinding(1, vk::DescriptorType::eStorageBuffer, 1, vk::ShaderStageFlagBits::eVertex),
226 vk::DescriptorSetLayoutBinding(2, vk::DescriptorType::eCombinedImageSampler, 1,
227 vk::ShaderStageFlagBits::eFragment),
228 };
229 gpuParticleDrawSetLayout_ = device->createDescriptorSetLayout(vk::DescriptorSetLayoutCreateInfo({}, drawBindings));
230
231 const auto computePush = pushConstantRange(vk::ShaderStageFlagBits::eCompute, sizeof(ParticleComputePush));
232 gpuParticleComputeLayout_ = createPipelineLayout(device, gpuParticleComputeSetLayout_, &computePush);
233 const auto drawPush = pushConstantRange(vk::ShaderStageFlagBits::eVertex, sizeof(ParticleDrawPush));
234 gpuParticleDrawLayout_ = createPipelineLayout(device, gpuParticleDrawSetLayout_, &drawPush);
235
236 const auto computeSpv = std::vector<std::uint32_t>(particle_resident_comp_spv,
237 particle_resident_comp_spv + particle_resident_comp_spv_count);
238 vk::ShaderModule computeModule = vkb::PipelineBuilder::createShaderModule(device.instance, computeSpv);
239 vk::PipelineShaderStageCreateInfo stage{};
240 stage.stage = vk::ShaderStageFlagBits::eCompute;
241 stage.module = computeModule;
242 stage.pName = "main";
243 vk::ComputePipelineCreateInfo computeInfo{};
244 computeInfo.stage = stage;
245 computeInfo.layout = gpuParticleComputeLayout_;
246 auto computeResult = device->createComputePipeline({}, computeInfo);
247 if (computeResult.result == vk::Result::eSuccess) gpuParticleComputePipeline_ = computeResult.value;
248 device->destroyShaderModule(computeModule);
249
250 const auto vertSpv = std::vector<std::uint32_t>(particle_resident_vert_spv,
251 particle_resident_vert_spv + particle_resident_vert_spv_count);
252 const auto fragSpv = std::vector<std::uint32_t>(particle_resident_frag_spv,
253 particle_resident_frag_spv + particle_resident_frag_spv_count);
254 vk::ShaderModule vert = vkb::PipelineBuilder::createShaderModule(device.instance, vertSpv);
255 vk::ShaderModule frag = vkb::PipelineBuilder::createShaderModule(device.instance, fragSpv);
256 gpuParticleAlphaPipeline_ =
257 createParticleDrawPipeline(device, renderpass, gpuParticleDrawLayout_, vert, frag, BlendMode::Alpha);
258 gpuParticleAdditivePipeline_ =
259 createParticleDrawPipeline(device, renderpass, gpuParticleDrawLayout_, vert, frag, BlendMode::Additive);
260 gpuParticlePremultipliedPipeline_ =
261 createParticleDrawPipeline(device, renderpass, gpuParticleDrawLayout_, vert, frag, BlendMode::Premultiplied);
262 gpuParticleMultiplyPipeline_ =
263 createParticleDrawPipeline(device, renderpass, gpuParticleDrawLayout_, vert, frag, BlendMode::Multiply);
264 gpuParticleOpaquePipeline_ =
265 createParticleDrawPipeline(device, renderpass, gpuParticleDrawLayout_, vert, frag, BlendMode::Opaque);
266 device->destroyShaderModule(vert);
267 device->destroyShaderModule(frag);
268}

303void Graphics::recordGpuParticleCompute(vk::CommandBuffer cb) {
304 if (!cb || !gpuParticleComputePipeline_) return;
305 const std::size_t frameSlot = currentFrameSlot();
306 const std::size_t requiredSlots = std::max<std::size_t>(2, frameSlotCount());
307 using BufferUsage = vk::BufferUsageFlagBits;
308 using Memory = vk::MemoryPropertyFlagBits;
309
310 for (auto& [handle, owned] : gpuParticles_) {
311 (void)handle;
312 auto& resource = *owned;
313 if (!resource.pendingUpdate && !resource.pendingReset) continue;
314 if (resource.slots.size() < requiredSlots) resource.slots.resize(requiredSlots);
315 auto& slot = resource.slots[frameSlot % resource.slots.size()];
316 const vk::DeviceSize stateBytes = vk::DeviceSize(resource.capacity) * sizeof(GpuParticleSpawn);
317 const vk::DeviceSize metaBytes = sizeof(ParticleMeta);
318
319 if (!slot.state.buffer) {
320 const auto stateUsage = BufferUsage::eStorageBuffer | BufferUsage::eTransferSrc | BufferUsage::eTransferDst;
321 const auto metaUsage = BufferUsage::eStorageBuffer | BufferUsage::eTransferSrc | BufferUsage::eTransferDst;
322 slot.state.allocate(frameToken(), device, stateUsage, stateBytes);
323 slot.scratchState.allocate(frameToken(), device, stateUsage, stateBytes);
324 slot.meta.allocate(frameToken(), device, metaUsage, metaBytes,
325 Memory::eHostVisible | Memory::eHostCoherent);
326 slot.indirect.allocate(frameToken(), device, BufferUsage::eIndirectBuffer | BufferUsage::eTransferDst,
327 sizeof(vk::DrawIndirectCommand));
328 slot.scratchMeta.allocate(frameToken(), device, metaUsage, metaBytes);
329 slot.spawns.allocate(frameToken(), device, BufferUsage::eStorageBuffer, stateBytes,
330 Memory::eHostVisible | Memory::eHostCoherent);
331 const ParticleMeta zero{};
332 slot.meta.updateLocal(frameToken(), zero);
333 }
334
335 if (slot.initialized && slot.serial > resource.statsSerial) {
336 const auto* meta = static_cast<const ParticleMeta*>(slot.meta.map());
337 if (meta) {
338 resource.stats.alive = meta->alive;
339 resource.stats.instances = meta->instanceCount;
340 resource.stats.spawned = meta->spawned;
341 resource.stats.killed = meta->killed;
342 resource.stats.dropped = meta->dropped;
343 slot.meta.unmap();
344 resource.statsSerial = slot.serial;
345 }
346 }
347
348 if (!resource.spawns.empty()) slot.spawns.updateLocal(frameToken(), resource.spawns);
349
350 vk::Buffer previousState = slot.scratchState.buffer;
351 vk::Buffer previousMeta = slot.scratchMeta.buffer;
352 vk::AccessFlags previousStateAccess{};
353 vk::AccessFlags previousMetaAccess{};
354 if (!resource.pendingReset && resource.lastOutputSlot >= 0) {
355 auto& previous = resource.slots[std::size_t(resource.lastOutputSlot)];
356 if (&previous == &slot) {
357 std::array<vk::BufferMemoryBarrier, 2> toTransfer{};
358 toTransfer[0].buffer = slot.state.buffer;
359 toTransfer[0].size = VK_WHOLE_SIZE;
360 toTransfer[0].srcAccessMask = vk::AccessFlagBits::eShaderWrite;
361 toTransfer[0].dstAccessMask = vk::AccessFlagBits::eTransferRead;
362 toTransfer[1].buffer = slot.meta.buffer;
363 toTransfer[1].size = VK_WHOLE_SIZE;
364 toTransfer[1].srcAccessMask = vk::AccessFlagBits::eShaderWrite;
365 toTransfer[1].dstAccessMask = vk::AccessFlagBits::eTransferRead;
366 cb.pipelineBarrier(vk::PipelineStageFlagBits::eComputeShader, vk::PipelineStageFlagBits::eTransfer, {},
367 nullptr, toTransfer, nullptr);
368 cb.copyBuffer(slot.state.buffer, slot.scratchState.buffer, vk::BufferCopy{0, 0, stateBytes});
369 cb.copyBuffer(slot.meta.buffer, slot.scratchMeta.buffer, vk::BufferCopy{0, 0, metaBytes});
370 previousStateAccess = vk::AccessFlagBits::eTransferWrite;
371 previousMetaAccess = vk::AccessFlagBits::eTransferWrite;
372 } else {
373 previousState = previous.state.buffer;
374 previousMeta = previous.meta.buffer;
375 previousStateAccess = vk::AccessFlagBits::eShaderWrite;
376 previousMetaAccess = vk::AccessFlagBits::eShaderWrite;
377 }
378 } else {
379 cb.fillBuffer(slot.scratchMeta.buffer, 0, metaBytes, 0);
380 previousMetaAccess = vk::AccessFlagBits::eTransferWrite;
381 }
382
383 cb.fillBuffer(slot.meta.buffer, 0, metaBytes, 0);
384
385 std::array<vk::BufferMemoryBarrier, 5> toCompute{};
386 const std::array<vk::Buffer, 5> computeBuffers{previousState, previousMeta, slot.spawns.buffer,
387 slot.meta.buffer, slot.state.buffer};
388 const std::array<vk::AccessFlags, 5> sourceAccess{
389 previousStateAccess,
390 previousMetaAccess,
391 vk::AccessFlagBits::eHostWrite,
392 vk::AccessFlagBits::eTransferWrite,
393 vk::AccessFlagBits::eShaderRead | vk::AccessFlagBits::eShaderWrite,
394 };
395 const std::array<vk::AccessFlags, 5> destinationAccess{
396 vk::AccessFlagBits::eShaderRead, vk::AccessFlagBits::eShaderRead,
397 vk::AccessFlagBits::eShaderRead, vk::AccessFlagBits::eShaderRead | vk::AccessFlagBits::eShaderWrite,
398 vk::AccessFlagBits::eShaderWrite,
399 };
400 for (std::size_t i = 0; i < toCompute.size(); ++i) {
401 toCompute[i].buffer = computeBuffers[i];
402 toCompute[i].size = VK_WHOLE_SIZE;
403 toCompute[i].srcAccessMask = sourceAccess[i];
404 toCompute[i].dstAccessMask = destinationAccess[i];
405 }
406 cb.pipelineBarrier(vk::PipelineStageFlagBits::eTransfer | vk::PipelineStageFlagBits::eHost |
407 vk::PipelineStageFlagBits::eComputeShader | vk::PipelineStageFlagBits::eVertexShader,
408 vk::PipelineStageFlagBits::eComputeShader, {}, nullptr, toCompute, nullptr);
409
410 if (!slot.computeSet) {
411 vk::DescriptorSetAllocateInfo allocate{};
412 allocate.descriptorPool = descriptorPool;
413 allocate.descriptorSetCount = 1;
414 allocate.pSetLayouts = &gpuParticleComputeSetLayout_;
415 slot.computeSet = device->allocateDescriptorSets(allocate).front();
416 }
417 std::array<vk::DescriptorBufferInfo, 5> infos{};
418 std::array<vk::WriteDescriptorSet, 5> writes{};
419 updateStorageBinding(writes[0], infos[0], slot.computeSet, 0, previousState, stateBytes);
420 updateStorageBinding(writes[1], infos[1], slot.computeSet, 1, slot.state.buffer, stateBytes);
421 updateStorageBinding(writes[2], infos[2], slot.computeSet, 2, slot.spawns.buffer, stateBytes);
422 updateStorageBinding(writes[3], infos[3], slot.computeSet, 3, previousMeta, metaBytes);
423 updateStorageBinding(writes[4], infos[4], slot.computeSet, 4, slot.meta.buffer, metaBytes);
424 device->updateDescriptorSets(writes, nullptr);
425
426 ParticleComputePush push{};
427 push.stepEmitterX[0] = resource.update.dt;
428 push.stepEmitterX[1] = resource.update.emitterX;
429 push.stepEmitterX[2] = resource.update.emitterY;
430 push.stepEmitterX[3] = resource.update.gravityX;
431 push.forces0[0] = resource.update.gravityY;
432 push.forces0[1] = resource.update.damping;
433 push.forces0[2] = resource.update.velocityLimit;
434 push.forces0[3] = resource.update.noiseStrength;
435 push.forces1[0] = resource.update.noiseFrequency;
436 push.forces1[1] = resource.update.noiseSpeed;
437 push.forces1[2] = resource.update.time;
438 push.forces1[3] = resource.update.frameRate;
439 push.localOffset[0] = resource.update.localOffsetX;
440 push.localOffset[1] = resource.update.localOffsetY;
441 push.capacity = resource.capacity;
442 push.spawnCount = std::uint32_t(resource.spawns.size());
443
444 cb.bindPipeline(vk::PipelineBindPoint::eCompute, gpuParticleComputePipeline_);
445 cb.bindDescriptorSets(vk::PipelineBindPoint::eCompute, gpuParticleComputeLayout_, 0, slot.computeSet, nullptr);
446 cb.pushConstants(gpuParticleComputeLayout_, vk::ShaderStageFlagBits::eCompute, 0, sizeof(push), &push);
447 cb.dispatch((resource.capacity + 63u) / 64u, 1, 1);
448
449 std::array<vk::BufferMemoryBarrier, 2> afterCompute{};
450 afterCompute[0].buffer = slot.state.buffer;
451 afterCompute[0].size = VK_WHOLE_SIZE;
452 afterCompute[0].srcAccessMask = vk::AccessFlagBits::eShaderWrite;
453 afterCompute[0].dstAccessMask = vk::AccessFlagBits::eShaderRead;
454 afterCompute[1].buffer = slot.meta.buffer;
455 afterCompute[1].size = sizeof(vk::DrawIndirectCommand);
456 afterCompute[1].srcAccessMask = vk::AccessFlagBits::eShaderWrite;
457 afterCompute[1].dstAccessMask = vk::AccessFlagBits::eTransferRead;
458 cb.pipelineBarrier(vk::PipelineStageFlagBits::eComputeShader,
459 vk::PipelineStageFlagBits::eVertexShader | vk::PipelineStageFlagBits::eTransfer, {}, nullptr,
460 afterCompute, nullptr);
461 cb.copyBuffer(slot.meta.buffer, slot.indirect.buffer, vk::BufferCopy{0, 0, sizeof(vk::DrawIndirectCommand)});
462 vk::BufferMemoryBarrier indirectReady{};
463 indirectReady.buffer = slot.indirect.buffer;
464 indirectReady.size = sizeof(vk::DrawIndirectCommand);
465 indirectReady.srcAccessMask = vk::AccessFlagBits::eTransferWrite;
466 indirectReady.dstAccessMask = vk::AccessFlagBits::eIndirectCommandRead;
467 cb.pipelineBarrier(vk::PipelineStageFlagBits::eTransfer, vk::PipelineStageFlagBits::eDrawIndirect, {}, nullptr,
468 indirectReady, nullptr);
469
470 slot.initialized = true;
471 slot.serial = ++resource.stats.submittedFrames;
472 resource.lastOutputSlot = int(frameSlot % resource.slots.size());
473 resource.pendingUpdate = false;
474 resource.pendingReset = false;
475 resource.spawns.clear();
476 }

479void Graphics::drawGpuParticleRequest(vk::CommandBuffer cb, const GpuParticleDrawRequest& request) {
480 auto found = gpuParticles_.find(request.handle);
481 if (found == gpuParticles_.end()) return;
482 auto& resource = *found->second;
483 if (resource.lastOutputSlot < 0) return;
484 auto& slot = resource.slots[std::size_t(resource.lastOutputSlot)];
485 if (!slot.state.buffer || !slot.meta.buffer) return;
486
487 Texture* texture = request.draw.texture ? request.draw.texture : whiteTexture;
