16 const auto agents = document.agents();
17 if (agents.size() > static_cast<std::size_t>(runtime->getMaxAgents()))
18 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.crowd.runtime-capacity"),
19 "Crowd document exceeds the runtime agent capacity");
20 std::map<StableId, CrowdPathRecord> paths;
21 for (const auto& path : document.paths()) paths.emplace(path.id, path);
22 runtime->clearAgents();
23 for (const auto& agent : agents) {

25 static_cast<float>(agent.y), static_cast<float>(agent.heading), static_cast<float>(agent.radius));
26 if (index < 0) {
27 runtime->clearAgents();
28 return eve::editing::failed<void>(EditorStatus::Failed, RuleId("editor.crowd.runtime-capacity"),
29 "Crowd runtime rejected an agent or reached capacity");
30 }

32 std::string action = agent.behavior;
33 if (action == "path") {

35 if (found == paths.end() || found->second.points.empty()) {
36 runtime->clearAgents();
37 return eve::editing::failed<void>(EditorStatus::Conflict, RuleId("editor.crowd.runtime-path"),
