349 [[nodiscard]] eve::Result<ReloadOutcome> reload(const DefinitionHandle& next, ReloadPolicy policy,
350 const State& defaults, RebuildFunction rebuild = {}) {
351 RuntimeInstanceCore::RebuildFunction erased;
352 if (rebuild) {

354 const DefinitionHandle& handle) -> eve::Result<void*> {
355 auto rebuilt = callback(*static_cast<const State*>(state), identity, handle);
356 if (!rebuilt) return eve::Result<void*>::failure(rebuilt.status());
357 return eve::Result<void*>::success(new State(std::move(rebuilt).takeValue()));
358 };
