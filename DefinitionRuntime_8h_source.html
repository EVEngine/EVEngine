362 [[nodiscard]] eve::Result<ReloadOutcome> reload(const DefinitionHandle& next, ReloadPolicy policy,
363 const State& defaults, RebuildFunction rebuild = {}) {
364 RuntimeInstanceCore::RebuildFunction erased;
365 if (rebuild) {
366 erased = [callback = std::move(rebuild)](const void* state, const InstanceIdentity& identity,
367 const DefinitionHandle& handle) -> eve::Result<void*> {
368 auto rebuilt = callback(*static_cast<const State*>(state), identity, handle);
369 if (!rebuilt) return eve::Result<void*>::failure(rebuilt.status());
371 return eve::Result<void*>::success(new State(std::move(rebuilt).takeValue()));
372 };
373 }
374 return core_.reload(next, policy, &defaults, erased);
375 }
