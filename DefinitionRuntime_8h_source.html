248 [[nodiscard]] eve::Result<ReloadOutcome> reload(const DefinitionHandle& next, ReloadPolicy policy,
249 const State& defaults, RebuildFunction rebuild = {}) {
250 if (!next.isValid())
251 return eve::Result<ReloadOutcome>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
252 "next definition handle is invalid",
253 "next", {}, "common.definitions"));
254 if (next.reference != identity_.definition)
255 return eve::Result<ReloadOutcome>::failure(eve::Diagnostic::error(
256 eve::DiagnosticCode::Conflict, "definition reload targets a different logical definition",
257 "next.reference", {}, "common.definitions"));
258 if (next.generation < identity_.definitionGeneration)
259 return eve::Result<ReloadOutcome>::failure(eve::Diagnostic::error(
260 eve::DiagnosticCode::StaleHandle, "definition reload handle is older than the instance generation",
261 "next.generation", {}, "common.definitions"));
262
263 ReloadOutcome outcome{
264 identity_.instanceId, identity_.definition, identity_.definitionGeneration, next.generation, policy,
265 ReloadDisposition::Unchanged};
266 if (next.generation == identity_.definitionGeneration)
267 return eve::Result<ReloadOutcome>::success(std::move(outcome), eve::Status::success(eve::StatusCode::NoOp));
268
269 if (policy == ReloadPolicy::RejectWhileActive && active_)
270 return eve::Result<ReloadOutcome>::failure(eve::Diagnostic::error(
271 eve::DiagnosticCode::Conflict, "active runtime instance rejects definition reload", "policy", {},
272 "common.definitions"));
273
274 try {
275 State candidate(*state_);
276 ReloadDisposition disposition = ReloadDisposition::Kept;
277 if (policy == ReloadPolicy::ReapplyDefaults) {
278 candidate = defaults;
279 disposition = ReloadDisposition::DefaultsReapplied;
280 } else if (policy == ReloadPolicy::RebuildInstance) {
281 if (!rebuild)
282 return eve::Result<ReloadOutcome>::failure(eve::Diagnostic::error(
283 eve::DiagnosticCode::InvalidArgument, "rebuild policy requires a rebuild callback", "rebuild",
284 {}, "common.definitions"));
285 auto rebuilt = rebuild(*state_, identity_, next);
286 if (!rebuilt) return eve::Result<ReloadOutcome>::failure(rebuilt.status());
287 candidate = std::move(rebuilt).takeValue();
288 disposition = ReloadDisposition::Rebuilt;
289 } else if (policy == ReloadPolicy::RejectWhileActive) {
290 // The active guard was handled above. Inactive instances have
291 // the documented keep-values behavior.
292 disposition = ReloadDisposition::Kept;
293 }
294
295 auto committed = std::make_unique<State>(std::move(candidate));
296 state_.swap(committed);
297 identity_.definitionGeneration = next.generation;
298 outcome.disposition = disposition;
299 return eve::Result<ReloadOutcome>::success(std::move(outcome),
300 eve::Status::success(eve::StatusCode::Applied));
301 } catch (const std::exception&) {
302 return eve::Result<ReloadOutcome>::failure(
303 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "definition reload candidate preparation failed",
304 {}, {}, "common.definitions"));
305 } catch (...) {
306 return eve::Result<ReloadOutcome>::failure(
307 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "definition reload candidate preparation failed",
308 {}, {}, "common.definitions"));
309 }
310 }
