52Result<Capture> capture(std::string_view name) {
53 if (name.empty() || name.size() > 256)
54 return failure<Capture>(DiagnosticCode::InvalidArgument, "host must contain 1..256 bytes");
55 SceneHost* host = nullptr;
56 if (ecs::current()->getManager<SceneHost>()) {
57 for (auto [meta] : ecs::View<SceneHost, SceneHost::Meta>()) {
58 if (meta->name == name) {
59 if (host) return failure<Capture>(DiagnosticCode::Conflict, "ambiguous host name");
60 host = meta->entity;
61 }
62 }
63 }
64 if (!host) return failure<Capture>(DiagnosticCode::NotFound, "scene host not found");
65 const auto& nodes = host->tree()->nodes;
66 if (nodes.size() > maxNodes)
67 return failure<Capture>(DiagnosticCode::InvalidArgument, "snapshot node budget exceeded (4096)");
68 Capture out;
69 out.snapshot.host = std::string(name);
70 std::ostringstream source;
71 source.imbue(std::locale::classic());
72 source << std::hexfloat;
73 const auto handle = ecs::handle_of(host);
74 // Process-local identity only; never serialized or exposed as a persistent object identity.
75 source << handle.table << ':' << handle.id << ':' << handle.generation;
76 std::ostringstream geometry;
77 geometry.imbue(std::locale::classic());
78 geometry << std::hexfloat;
79 std::vector<glm::mat4> worlds(nodes.size());
80 std::vector<bool> visibility(nodes.size());
81 std::vector<unsigned char> visited(nodes.size(), 0);
82 std::set<std::string> ids;
83 for (std::size_t i = 0; i < nodes.size(); ++i) {
84 std::vector<std::size_t> chain;
85 int cursor = int(i);
86 while (cursor >= 0) {
87 if (std::size_t(cursor) >= nodes.size() || visited[std::size_t(cursor)] == 1)
88 return failure<Capture>(DiagnosticCode::InvalidArgument, "invalid scene ancestry or cycle");
89 const auto index = std::size_t(cursor);
90 if (visited[index] == 2) break;
91 visited[index] = 1;
92 chain.push_back(index);
93 cursor = nodes[index].parent;
94 if (cursor < -1) return failure<Capture>(DiagnosticCode::InvalidArgument, "invalid scene parent");
95 }
96 for (auto it = chain.rbegin(); it != chain.rend(); ++it) {
97 const auto index = *it;
98 const auto& node = nodes[index];
99 worlds[index] =
100 (node.parent < 0 ? glm::mat4(1.f) : worlds[std::size_t(node.parent)]) * detail::localMatrix(node);
101 visibility[index] =
102 node.visible && (node.parent < 0 ? host->meta()->visible : visibility[std::size_t(node.parent)]);
103 visited[index] = 2;
104 }
105 }
106 for (std::size_t i = 0; i < nodes.size(); ++i) {
107 const auto& node = nodes[i];
108 if (node.id.empty() || !ids.insert(node.id).second)
109 return failure<Capture>(DiagnosticCode::InvalidArgument, "empty or duplicate scene node id");
110 SceneSearchRecord record;
111 record.id = node.id;
112 record.persistentId = node.persistentId.isNil() ? "" : node.persistentId.format();
113 record.name = node.name;
114 record.space = node.space;
115 record.tags = node.tags;
116 record.visible = visibility[i];
117 record.layer = node.layer;
118 record.position = {worlds[i][3][0], worlds[i][3][1], worlds[i][3][2]};
119 const auto forward = glm::vec3(worlds[i][2]);
120 const auto up = glm::vec3(worlds[i][1]);
121 if (glm::length(forward) > 1e-6f && glm::length(glm::cross(up, forward)) > 1e-6f) {
122 const auto z = glm::normalize(forward);
123 const auto x = glm::normalize(glm::cross(up, z));
124 const auto y = glm::cross(z, x);
125 record.axes = {{{x.x, x.y, x.z}, {y.x, y.y, y.z}, {z.x, z.y, z.z}}};
126 }
127 if (node.hasBounds) {
128 std::array<double, 6> bounds{INFINITY, INFINITY, INFINITY, -INFINITY, -INFINITY, -INFINITY};
129 for (int corner = 0; corner < 8; ++corner) {
130 const auto point =
131 worlds[i] * glm::vec4(corner & 1 ? node.bmaxX : node.bminX, corner & 2 ? node.bmaxY : node.bminY,
132 corner & 4 ? node.bmaxZ : node.bminZ, 1.f);
133 for (std::size_t axis = 0; axis < 3; ++axis) {
134 const double value = point[int(axis)];
135 if (!std::isfinite(value))
136 return failure<Capture>(DiagnosticCode::InvalidArgument, "nonfinite bounds");
137 bounds[axis] = std::min(bounds[axis], value);
138 bounds[axis + 3] = std::max(bounds[axis + 3], value);
139 }
140 }
141 record.bounds = bounds;
142 }
143 for (double number : record.position)
144 if (!std::isfinite(number))
145 return failure<Capture>(DiagnosticCode::InvalidArgument, "nonfinite scene transform");
146 record.kinds.push_back("scene-node");
147 for (const auto& link : node.links) record.kinds.emplace_back(linkKindName(link.kind));
148 std::sort(record.kinds.begin(), record.kinds.end());
149 record.kinds.erase(std::unique(record.kinds.begin(), record.kinds.end()), record.kinds.end());
150 std::sort(record.tags.begin(), record.tags.end());
151 record.text = record.name;
152 for (const auto& tag : record.tags) record.text += "\n" + tag;
153 if (record.text.size() > maxTextBytes || record.id.size() > 256 || record.kinds.size() > 64 ||
154 record.tags.size() > 64)
155 return failure<Capture>(DiagnosticCode::InvalidArgument, "scene search metadata budget exceeded");
156 source << std::quoted(record.id) << std::quoted(record.persistentId) << std::quoted(record.text)
157 << std::quoted(record.name) << std::quoted(record.space) << ':' << node.parent;
158 for (const auto& tag : record.tags) source << std::quoted(tag);
159 geometry << std::quoted(record.id) << record.visible << ':' << record.layer;
160 for (double number : record.position) geometry << ':' << number;
161 for (const auto& axis : record.axes)
162 for (double number : axis) geometry << ':' << number;
163 geometry << ':' << record.bounds.has_value();
164 if (record.bounds)
165 for (double number : *record.bounds) geometry << ':' << number;
166 for (const auto& kind : record.kinds) source << std::quoted(kind);
167 source << ';';
168 out.snapshot.records.push_back(std::move(record));
169 }
170 out.source = source.str();
171 out.geometry = geometry.str();
172 return Result<Capture>::success(std::move(out));
173}

175Result<std::vector<double>> normalized(const std::vector<double>& vector) {
176 if (vector.empty() || vector.size() > maxDimensions)
177 return failure<std::vector<double>>(DiagnosticCode::InvalidArgument, "embedding dimension must be 1..1024");
178 double norm = 0;
179 for (double number : vector) {
180 if (!std::isfinite(number) || std::abs(number) > 1e10)
181 return failure<std::vector<double>>(DiagnosticCode::InvalidArgument, "invalid embedding scalar");
182 norm += number * number;
183 }
184 if (norm < 1e-20) return failure<std::vector<double>>(DiagnosticCode::InvalidArgument, "zero embedding");
185 auto out = vector;
186 for (auto& number : out) number /= std::sqrt(norm);
187 return Result<std::vector<double>>::success(std::move(out));
188}

190std::string queryKey(const SceneSearchQuery& query) {
191 std::ostringstream out;
192 out.imbue(std::locale::classic());
193 out << std::hexfloat << std::quoted(query.snapshot) << std::quoted(query.model) << std::quoted(query.kind)
194 << std::quoted(query.space) << ':' << query.limit << ':' << query.scanBudget;
195 for (const auto& tag : query.tags) out << std::quoted(tag);
196 out << ':' << query.visible.has_value();
197 if (query.visible) out << ':' << *query.visible;
198 out << ':' << query.layer.has_value();
199 if (query.layer) out << ':' << *query.layer;
200 out << ':' << query.center.has_value();
201 if (query.center)
202 for (double number : *query.center) out << ':' << number;
203 out << ':' << query.radius;
204 out << ':' << query.candidates;
205 for (const auto& relation : query.spatial) {
206 out << ':' << int(relation.relation) << ':' << int(relation.frame) << std::quoted(relation.reference)
207 << std::quoted(relation.observer) << ':' << relation.distance << ':' << relation.margin;
208 }
209 for (double number : query.embedding) out << ':' << number;
210 return out.str();
211}

214 SceneSpatialConstraint constraint;
215 SceneSearchRecord reference;
216 std::array<std::array<double, 3>, 3> axes{{{1, 0, 0}, {0, 1, 0}, {0, 0, 1}}};
217};

219Result<std::vector<ResolvedSpatial>> resolveSpatial(const std::vector<SceneSearchRecord>& records,
220 const std::vector<SceneSpatialConstraint>& constraints) {
221 if (constraints.size() > 16)
222 return failure<std::vector<ResolvedSpatial>>(DiagnosticCode::InvalidArgument, "at most 16 spatial predicates");
223 std::vector<ResolvedSpatial> out;
224 for (const auto& constraint : constraints) {
225 if (!std::isfinite(constraint.distance) || constraint.distance < 0 || !std::isfinite(constraint.margin) ||
226 constraint.margin < 0 || int(constraint.relation) < 0 || int(constraint.relation) > 7 ||
227 int(constraint.frame) < 0 || int(constraint.frame) > 2)
228 return failure<std::vector<ResolvedSpatial>>(DiagnosticCode::InvalidArgument, "invalid spatial predicate");
229 const auto reference = std::find_if(records.begin(), records.end(),
230 [&](const auto& record) { return record.id == constraint.reference; });
231 if (reference == records.end())
232 return failure<std::vector<ResolvedSpatial>>(DiagnosticCode::NotFound, "spatial reference not found");
233 ResolvedSpatial resolved;
234 resolved.constraint = constraint;
235 resolved.reference = *reference;
236 if (constraint.frame == SceneSpatialFrame::Reference) resolved.axes = reference->axes;
237 if (constraint.frame == SceneSpatialFrame::Observer) {
238 const auto observer = std::find_if(records.begin(), records.end(),
239 [&](const auto& record) { return record.id == constraint.observer; });
240 if (observer == records.end() || observer->space != reference->space)
241 return failure<std::vector<ResolvedSpatial>>(DiagnosticCode::NotFound, "compatible observer not found");
242 resolved.axes = observer->axes;
243 } else if (!constraint.observer.empty()) {
244 return failure<std::vector<ResolvedSpatial>>(DiagnosticCode::InvalidArgument,
245 "observer requires observer frame");
246 }
247 for (const auto& axis : resolved.axes) {
248 double length = 0;
249 for (double value : axis) length += value * value;
250 if (!std::isfinite(length) || length < .5)
251 return failure<std::vector<ResolvedSpatial>>(DiagnosticCode::InvalidArgument,
252 "degenerate reference frame");
253 }
254 if (constraint.relation == SceneSpatialRelation::Inside && !reference->bounds)
255 return failure<std::vector<ResolvedSpatial>>(DiagnosticCode::Unsupported,
256 "inside requires authored reference bounds");
257 out.push_back(std::move(resolved));
258 }
259 return Result<std::vector<ResolvedSpatial>>::success(std::move(out));
260}

262std::optional<SceneSpatialEvidence> spatialEvidence(const SceneSearchRecord& record, const ResolvedSpatial& filter) {
263 if (record.space != filter.reference.space) return std::nullopt;
264 SceneSpatialEvidence out;
265 out.reference = filter.constraint.reference;
266 out.observer = filter.constraint.observer;
267 out.relation = filter.constraint.relation;
268 out.frame = filter.constraint.frame;
269 for (std::size_t coordinate = 0; coordinate < 3; ++coordinate) {
270 const double delta = record.position[coordinate] - filter.reference.position[coordinate];
271 out.distance = std::hypot(out.distance, delta);
272 for (std::size_t axis = 0; axis < 3; ++axis) out.offset[axis] += delta * filter.axes[axis][coordinate];
273 }
274 // Scene's 2D coordinates have +Y down; 3D has +Y up. Direction evidence always reports +up.
275 if (record.space == "2d") out.offset[1] = -out.offset[1];
276 const double margin = filter.constraint.margin;
277 bool matches = false;
278 switch (filter.constraint.relation) {
279 case SceneSpatialRelation::Left: matches = out.offset[0] < -margin; break;
280 case SceneSpatialRelation::Right: matches = out.offset[0] > margin; break;
281 case SceneSpatialRelation::Above: matches = out.offset[1] > margin; break;
282 case SceneSpatialRelation::Below: matches = out.offset[1] < -margin; break;
283 case SceneSpatialRelation::Front: matches = out.offset[2] > margin; break;
284 case SceneSpatialRelation::Behind: matches = out.offset[2] < -margin; break;
285 case SceneSpatialRelation::Near: matches = out.distance <= filter.constraint.distance; break;
286 case SceneSpatialRelation::Inside: {
287 const auto& bounds = *filter.reference.bounds;
288 matches = true;
289 for (std::size_t axis = 0; axis < 3; ++axis)
290 matches = matches && record.position[axis] >= bounds[axis] - margin &&
291 record.position[axis] <= bounds[axis + 3] + margin;
292 break;
293 }
294 }
295 return matches ? std::optional(out) : std::nullopt;
296}

342Result<void> SceneSearch::publish(const SceneEmbeddingBatch& batch) {
343 if (snapshot_.token.empty() || batch.snapshot != snapshot_.token)
344 return failure<void>(DiagnosticCode::StaleHandle, "stale snapshot token");
345 auto current = capture(snapshot_.host);
346 if (!current.ok()) return Result<void>::failure(current.status());
347 if (current.value().source != source_)
348 return failure<void>(DiagnosticCode::Conflict, "scene source changed; prepare again");
349 if (batch.model.empty() || batch.model.size() > 512 || batch.vectors.size() != snapshot_.records.size())
350 return failure<void>(DiagnosticCode::InvalidArgument,
351 "model identity and exactly one vector per record required");
352 std::vector<std::vector<double>> candidate;
353 for (const auto& vector : batch.vectors) {
354 if (!candidate.empty() && vector.size() != candidate.front().size())
355 return failure<void>(DiagnosticCode::InvalidArgument, "mixed embedding dimensions");
356 auto value = normalized(vector);
357 if (!value.ok()) return Result<void>::failure(value.status());
358 candidate.push_back(std::move(value).takeValue());
359 }
360 std::unique_ptr<SceneHnswIndex> index;
361 try {
362 if (!candidate.empty()) {
363 index = std::make_unique<SceneHnswIndex>(candidate.front().size(), candidate.size());
364 for (std::size_t i = 0; i < candidate.size(); ++i) {
365 std::vector<float> values;
366 for (double value : candidate[i]) values.push_back(static_cast<float>(value));
367 index->graph.addPoint(values.data(), i);
368 }
369 }
370 } catch (const std::exception& error) {
371 return failure<void>(DiagnosticCode::Failed, std::string("HNSW build failed: ") + error.what());
372 }
373 vectors_ = std::move(candidate);
374 index_ = std::move(index);
375 model_ = batch.model;
376 cursor_.clear();
377 return Result<void>::success();
378}

380Result<SceneSearchResult> SceneSearch::query(const SceneSearchQuery& query) {
381 if (snapshot_.token.empty() || query.snapshot != snapshot_.token)
382 return failure<SceneSearchResult>(DiagnosticCode::StaleHandle, "stale snapshot token");
383 if (query.tags.size() > 64 || query.model.size() > 512 || query.kind.size() > 256 || query.space.size() > 256 ||
384 std::any_of(query.tags.begin(), query.tags.end(), [](const auto& tag) { return tag.size() > 256; }))
385 return failure<SceneSearchResult>(DiagnosticCode::InvalidArgument, "query metadata budget exceeded");
386 if (query.limit < 1 || query.limit > 256 || query.scanBudget < 1 || query.scanBudget > maxNodes ||
387 !std::isfinite(query.radius) || query.radius < 0 || (!query.center && query.radius != 0))
388 return failure<SceneSearchResult>(DiagnosticCode::InvalidArgument, "invalid query budget or spatial filter");
389 if (query.center)
390 for (double number : *query.center)
391 if (!std::isfinite(number))
392 return failure<SceneSearchResult>(DiagnosticCode::InvalidArgument, "nonfinite center");
393 auto current = capture(snapshot_.host);
394 if (!current.ok()) return Result<SceneSearchResult>::failure(current.status());
395 if (current.value().source != source_)
396 return failure<SceneSearchResult>(DiagnosticCode::Conflict, "scene source changed; prepare again");
397 if (snapshot_.records.size() > query.scanBudget)
398 return failure<SceneSearchResult>(DiagnosticCode::InvalidArgument,
399 "scan budget exceeded; no partial results returned");
400 std::vector<double> embedding;
401 if (!query.embedding.empty()) {
402 if (model_.empty()) return failure<SceneSearchResult>(DiagnosticCode::Unsupported, "vector-index-unavailable");
403 if (query.model != model_)
404 return failure<SceneSearchResult>(DiagnosticCode::Conflict, "embedding model mismatch");
405 if (!vectors_.empty() && query.embedding.size() != vectors_.front().size())
406 return failure<SceneSearchResult>(DiagnosticCode::InvalidArgument, "query embedding dimension mismatch");
407 auto vector = normalized(query.embedding);
408 if (!vector.ok()) return Result<SceneSearchResult>::failure(vector.status());
409 embedding = std::move(vector).takeValue();
410 }
411 if (query.candidates < 1 || query.candidates > maxNodes)
412 return failure<SceneSearchResult>(DiagnosticCode::InvalidArgument, "candidate budget must be 1..4096");
413 auto spatial = resolveSpatial(current.value().snapshot.records, query.spatial);
414 if (!spatial.ok()) return Result<SceneSearchResult>::failure(spatial.status());
415 const auto key = queryKey(query) + current.value().geometry;
416 std::size_t start = 0;
417 if (!query.pageToken.empty()) {
418 if (query.pageToken != cursor_ || key != queryKey_)
419 return failure<SceneSearchResult>(DiagnosticCode::StaleHandle, "stale or cross-query page token");
420 start = offset_;
421 }
422 SceneSearchResult out;
423 out.snapshot = snapshot_.token;
424 out.indexSnapshot = model_.empty() ? "" : snapshot_.token;
425 out.model = model_;
426 out.scanned = snapshot_.records.size();
427 EligibleLabels labels;
428 labels.eligible.resize(snapshot_.records.size());
429 for (std::size_t i = 0; i < snapshot_.records.size(); ++i) {
430 const auto& record = current.value().snapshot.records[i];
431 if ((!query.kind.empty() &&
432 std::find(record.kinds.begin(), record.kinds.end(), query.kind) == record.kinds.end()) ||
433 (!query.space.empty() && query.space != record.space) ||
434 (query.visible && record.visible != *query.visible) || (query.layer && record.layer != *query.layer))
435 continue;
436 if (!std::all_of(query.tags.begin(), query.tags.end(), [&](const auto& tag) {
437 return std::find(record.tags.begin(), record.tags.end(), tag) != record.tags.end();
438 }))
439 continue;
440 if (query.center) {
441 double distance = 0;
442 for (std::size_t axis = 0; axis < 3; ++axis)
443 distance = std::hypot(distance, record.position[axis] - (*query.center)[axis]);
444 if (distance > query.radius) continue;
445 }
446 std::vector<SceneSpatialEvidence> evidence;
447 for (const auto& filter : spatial.value()) {
448 auto measured = spatialEvidence(record, filter);
449 if (!measured) break;
450 evidence.push_back(std::move(*measured));
451 }
452 if (evidence.size() != spatial.value().size()) continue;
453 labels.eligible[i] = true;
454 out.hits.push_back({record, 0, std::move(evidence)});
455 }
456 out.matched = out.hits.size();
457 if (!embedding.empty() && !out.hits.empty()) {
458 const auto window = std::min(out.hits.size(), std::max(query.limit, query.candidates));
459 std::vector<float> values;
460 for (double value : embedding) values.push_back(static_cast<float>(value));
461 try {
462 index_->graph.setEf(std::max(std::size_t{128}, window));
463 auto nearest = index_->graph.searchKnn(values.data(), window, &labels);
464 std::vector<SceneSearchHit> hits;
465 while (!nearest.empty()) {
466 const auto [distance, label] = nearest.top();
467 nearest.pop();
468 const auto found = std::find_if(out.hits.begin(), out.hits.end(), [&](const auto& hit) {
469 return hit.record.id == current.value().snapshot.records[label].id;
470 });
471 if (found != out.hits.end()) {
472 auto hit = *found;
473 hit.score = std::clamp(1.0 - double(distance), -1.0, 1.0);
474 hits.push_back(std::move(hit));
475 }
476 }
477 out.hits = std::move(hits);
478 } catch (const std::exception& error) {
479 return failure<SceneSearchResult>(DiagnosticCode::Failed,
480 std::string("HNSW query failed: ") + error.what());
481 }

484 }

487 });

494 offset_ = end;

496 } else

498 out.hits =
