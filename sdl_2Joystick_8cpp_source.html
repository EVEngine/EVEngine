63Pad* Joystick::addJoystick(int deviceindex) {
64 if (deviceindex < 0 || deviceindex >= SDL_NumJoysticks()) return nullptr;
65
66 std::string guidstr = getDeviceGUID(deviceindex);
67 Pad* pad = nullptr;
68 bool reused = false;
69
70 for (auto* stick : joysticks_) {
71 if (!stick->isConnected() && stick->getGUID() == guidstr) {
72 pad = stick;
73 reused = true;
74 break;
75 }
76 }
77
78 if (!pad) {
79 pad = new Pad(static_cast<int>(joysticks_.size()));
80 joysticks_.push_back(pad);
81 }
82
83 removeJoystick(pad);
84
85 if (!pad->open(deviceindex)) return nullptr;
86
87 for (auto* active : activeSticks_) {
88 if (pad->getHandle() == active->getHandle()) {
89 pad->close();
90 if (!reused) {
91 joysticks_.remove(pad);
92 delete pad;
93 }
94 return active;
95 }
96 }
97
98 if (pad->isGamepad()) recentGamepadGUIDs_[pad->getGUID()] = true;
99
100 activeSticks_.push_back(pad);
101 return pad;
102}

113void Joystick::checkGamepads(const std::string& guid) const {
114 for (int d_index = 0; d_index < SDL_NumJoysticks(); d_index++) {
115 if (!SDL_IsGameController(d_index)) continue;
116 if (guid.compare(getDeviceGUID(d_index)) != 0) continue;
117
118 for (auto* stick : activeSticks_) {
119 if (guid.compare(stick->getGUID()) != 0) continue;
120
121 SDL_GameController* controller = SDL_GameControllerOpen(d_index);
122 if (!controller) continue;
123
124 SDL_Joystick* sdlstick = SDL_GameControllerGetJoystick(controller);
125 bool open_gamepad = (sdlstick == static_cast<SDL_Joystick*>(stick->getHandle()));
126 SDL_GameControllerClose(controller);
127
128 if (open_gamepad) stick->openGamepad(d_index);
129 }
130 }
131}

142void Joystick::loadGamepadMappings(const std::string& mappings) {
143 std::stringstream ss(mappings);
144 std::string mapping;
145 bool success = false;
146
147 while (std::getline(ss, mapping)) {
148 if (mapping.empty()) continue;
149 if (mapping[0] == '#') continue;
150
151 size_t pstartpos = mapping.find("platform:");
152 if (pstartpos != std::string::npos) {
153 pstartpos += strlen("platform:");
154 size_t pendpos = mapping.find_first_of(',', pstartpos);
155 std::string platform = mapping.substr(pstartpos, pendpos - pstartpos);
156 if (platform.compare(SDL_GetPlatform()) != 0) {
157 success = true;
158 continue;
159 }
160 pstartpos -= strlen("platform:");
161 mapping.erase(pstartpos, pendpos - pstartpos + 1);
162 }
163
164 if (SDL_GameControllerAddMapping(mapping.c_str()) != -1) {
165 success = true;
166 std::string guid = mapping.substr(0, mapping.find_first_of(','));
167 recentGamepadGUIDs_[guid] = true;
168 checkGamepads(guid);
169 }
170 }
171
172 if (!success && !mappings.empty()) throw eve::Exception("Invalid gamepad mappings.");
173}
