61void transitionSampledColorForTransfer(vk::CommandBuffer cb, vkb::ColorTarget &image,
62 vk::ImageLayout newLayout) {
63 vk::ImageMemoryBarrier barrier{};
64 barrier.srcQueueFamilyIndex = VK_QUEUE_FAMILY_IGNORED;
65 barrier.dstQueueFamilyIndex = VK_QUEUE_FAMILY_IGNORED;
66 barrier.oldLayout = image.currentLayout();
67 barrier.newLayout = newLayout;
68 barrier.image = image.image();
69 barrier.subresourceRange = {vk::ImageAspectFlagBits::eColor, 0, 1, 0, 1};
70 vk::PipelineStageFlags srcStages;
71 vk::PipelineStageFlags dstStages;
72 if (newLayout == vk::ImageLayout::eTransferSrcOptimal) {
73 barrier.srcAccessMask = vk::AccessFlagBits::eShaderRead;
74 barrier.dstAccessMask = vk::AccessFlagBits::eTransferRead;
75 srcStages = vk::PipelineStageFlagBits::eVertexShader |
76 vk::PipelineStageFlagBits::eFragmentShader |
77 vk::PipelineStageFlagBits::eComputeShader;
78 dstStages = vk::PipelineStageFlagBits::eTransfer;
79 } else {
80 barrier.srcAccessMask = vk::AccessFlagBits::eTransferRead;
81 barrier.dstAccessMask = vk::AccessFlagBits::eShaderRead;
82 srcStages = vk::PipelineStageFlagBits::eTransfer;
83 dstStages = vk::PipelineStageFlagBits::eVertexShader |
84 vk::PipelineStageFlagBits::eFragmentShader |
85 vk::PipelineStageFlagBits::eComputeShader;
86 }
87 cb.pipelineBarrier(srcStages, dstStages, {}, 0, nullptr, 0, nullptr, 1, &barrier);
88 image.setCurrentLayout(newLayout);
89}

99std::vector<uint32_t> loadSpirvBytes(const void *data, size_t size) {
100 if (!data || size < 4 || (size % 4) != 0)
101 throw Exception("SPIR-V: invalid size %zu", size);
102 const auto *words = static_cast<const uint32_t *>(data);
103 if (words[0] != 0x07230203)
104 throw Exception("SPIR-V: bad magic (expected 0x07230203)");
105 return std::vector<uint32_t>(words, words + size / 4);
106}

115std::vector<uint32_t> compileGlslWithGlslc(const std::string &source, const char *stage) {
116 if (source.empty()) throw Exception("newShader: empty %s GLSL", stage);
117#if defined(_WIN32)
118 (void)source;
119 (void)stage;
120 throw Exception("newShader: GLSL compile via glslc is not supported on Windows; "
121 "use newShaderFromSpv / newShaderFromSpvFile");
122#else
123 char inPath[] = "/tmp/eve_shader_XXXXXX";
124 int fd = mkstemp(inPath);
125 if (fd < 0) throw Exception("newShader: mkstemp failed");
126 std::string outPath = std::string(inPath) + ".spv";
127 {
128 ssize_t n = write(fd, source.data(), source.size());
129 close(fd);
130 if (n < 0 || size_t(n) != source.size()) {
131 unlink(inPath);
132 throw Exception("newShader: failed to write temp GLSL");
133 }
134 }
135
136 std::string cmd = std::string("glslc -fshader-stage=") + stage + " \"" + inPath + "\" -o \"" +
137 outPath + "\" 2>&1";
138 FILE *pipe = popen(cmd.c_str(), "r");
139 std::string err;
140 if (pipe) {
141 char buf[256];
142 while (fgets(buf, sizeof(buf), pipe)) err += buf;
143 int status = pclose(pipe);
144 unlink(inPath);
145 if (status != 0) {
146 unlink(outPath.c_str());
147 throw Exception("newShader: glslc failed for %s:\n%s", stage, err.c_str());
148 }
149 } else {
150 unlink(inPath);
151 throw Exception("newShader: glslc not available (popen failed)");
152 }
153
154 FILE *f = fopen(outPath.c_str(), "rb");
155 if (!f) {
156 unlink(outPath.c_str());
157 throw Exception("newShader: failed to open compiled SPIR-V");
158 }
159 fseek(f, 0, SEEK_END);
160 long sz = ftell(f);
161 fseek(f, 0, SEEK_SET);
162 std::vector<uint8_t> bytes(static_cast<size_t>(sz > 0 ? sz : 0));
163 if (sz > 0 && fread(bytes.data(), 1, static_cast<size_t>(sz), f) != static_cast<size_t>(sz)) {
164 fclose(f);
165 unlink(outPath.c_str());
166 throw Exception("newShader: failed to read compiled SPIR-V");
167 }
168 fclose(f);
169 unlink(outPath.c_str());
170 return loadSpirvBytes(bytes.data(), bytes.size());
171#endif
172}

178 // flushToSwapchain / abortOpen3DFrame), so the present model must never

185void Graphics::ensureReadbackSlots() {
186 if (pixelWidth <= 0 || pixelHeight <= 0) return;
187 const size_t bytes = size_t(pixelWidth) * size_t(pixelHeight) * 4;
188 const size_t want = std::max<size_t>(2, frameSlotCount());
189 if (!screenReadbackSlots.empty() && screenReadbackSlots.size() >= want &&
190 screenReadbackBytes == bytes)
191 return;
192 // Recreating the staging ring must never race in-flight copies. Callers
193 // run on the render thread, and a size change implies the swapchain was
194 // rebuilt under waitIdle (rebuildSwapchainIfNeeded / recreate path).
195 for (auto &slot : screenReadbackSlots) {
196 if (slot.mapped) {
197 slot.staging.unmap();
198 slot.mapped = nullptr;
199 }
200 slot.staging.release();
201 }
202 screenReadbackSlots.clear();
203 screenReadbackSlots.resize(want);
204 for (auto &slot : screenReadbackSlots) {
205 slot.staging = vkb::GenericBuffer(device, vk::BufferUsageFlagBits::eTransferDst,
206 vk::DeviceSize(bytes),
207 vk::MemoryPropertyFlagBits::eHostVisible |
208 vk::MemoryPropertyFlagBits::eHostCoherent);
209 }
210 screenReadbackBytes = bytes;
211 readbackReady = false;
212 readbackCpuSynced = false;
213 readbackWriteSlot = 0;
214}

216bool Graphics::recordSwapchainReadback(vk::CommandBuffer cb) {
217 if (!screenReadbackEnabled || pixelWidth <= 0 || pixelHeight <= 0) return false;
218 if (!presentModel.has_acquired_image) return false;
219 const vk::Format fmt = swapchain.image_format;
220 const bool bgra = (fmt == vk::Format::eB8G8R8A8Unorm || fmt == vk::Format::eB8G8R8A8Srgb);
221 const bool rgba = (fmt == vk::Format::eR8G8B8A8Unorm || fmt == vk::Format::eR8G8B8A8Srgb);
222 if (!bgra && !rgba) return false;
223 readbackBgra = bgra;
224
225 const uint32_t imageIndex = presentModel.acquired_image_index;
226 auto &images = swapchain.get_images();
227 if (imageIndex >= images.size()) return false;
228
229 ensureReadbackSlots();
230 if (screenReadbackSlots.empty()) return false;
231
232 const size_t slot = size_t(presentRecording.slot().index) % screenReadbackSlots.size();
233 readbackWriteSlot = slot;
234 const vk::Image image = images[imageIndex];
235 // Recorded at the end of the present command buffer: the swapchain render
236 // pass has already transitioned the image back to PresentSrcKHR, and the
237 // copy runs in the same submit+present — no extra queue submit, no
238 // waitIdle, no per-frame buffer allocation.
239 vk::ImageMemoryBarrier toTransfer{};
240 toTransfer.srcQueueFamilyIndex = VK_QUEUE_FAMILY_IGNORED;
241 toTransfer.dstQueueFamilyIndex = VK_QUEUE_FAMILY_IGNORED;
242 toTransfer.oldLayout = vk::ImageLayout::ePresentSrcKHR;
243 toTransfer.newLayout = vk::ImageLayout::eTransferSrcOptimal;
244 toTransfer.image = image;
245 toTransfer.subresourceRange = {vk::ImageAspectFlagBits::eColor, 0, 1, 0, 1};
246 // The render pass has already transitioned the image to PresentSrcKHR, but
247 // there is no fence between it and this copy — the barrier must order the
248 // render-pass color writes against the transfer explicitly.
249 toTransfer.srcAccessMask = vk::AccessFlagBits::eColorAttachmentWrite;
250 toTransfer.dstAccessMask = vk::AccessFlagBits::eTransferRead;
251 cb.pipelineBarrier(vk::PipelineStageFlagBits::eColorAttachmentOutput,
252 vk::PipelineStageFlagBits::eTransfer, {}, 0, nullptr, 0, nullptr, 1,
253 &toTransfer);
254
255 vk::BufferImageCopy region{};
256 region.imageSubresource = {vk::ImageAspectFlagBits::eColor, 0, 0, 1};
257 region.imageExtent = vk::Extent3D{uint32_t(pixelWidth), uint32_t(pixelHeight), 1};
258 cb.copyImageToBuffer(image, vk::ImageLayout::eTransferSrcOptimal,
259 screenReadbackSlots[slot].staging.buffer, region);
260
261 vk::ImageMemoryBarrier toPresent = toTransfer;
262 toPresent.oldLayout = vk::ImageLayout::eTransferSrcOptimal;
263 toPresent.newLayout = vk::ImageLayout::ePresentSrcKHR;
264 toPresent.srcAccessMask = vk::AccessFlagBits::eTransferRead;
265 toPresent.dstAccessMask = vk::AccessFlagBits::eMemoryRead;
266 cb.pipelineBarrier(vk::PipelineStageFlagBits::eTransfer,
267 vk::PipelineStageFlagBits::eBottomOfPipe, {}, 0, nullptr, 0, nullptr, 1,
268 &toPresent);
269 return true;
270}

272void Graphics::syncReadbackCpu() {
273 if (!readbackReady || screenReadbackSlots.empty()) return;
274 if (readbackCpuSynced && !lastFrameRgba.empty()) return;
275 if (readbackWriteSlot >= screenReadbackSlots.size()) return;
276
277 const size_t bytes = screenReadbackBytes;
278 if (bytes == 0) return;
279
280 // The newest copy lives in the present submission of this frame slot; wait
281 // only that slot's fence instead of a device-wide waitIdle.
282 presentModel.waitForFrameSlot(readbackWriteSlot);
283
284 auto &slot = screenReadbackSlots[readbackWriteSlot];
285 if (!slot.mapped)
286 slot.mapped = slot.staging.map();
287
288 lastFrameRgba.resize(bytes);
289 if (readbackBgra) {
290 // BGRA -> RGBA byte swap, two pixels per 64-bit word (masked swap).
291 const size_t words = bytes / 8;
292 const uint64_t *src64 = static_cast<const uint64_t *>(slot.mapped);
293 uint64_t *dst64 = reinterpret_cast<uint64_t *>(lastFrameRgba.data());
294 for (size_t i = 0; i < words; ++i) {
295 const uint64_t v = src64[i];
296 dst64[i] = (v & 0xFF00FF00FF00FF00ull) |
297 ((v & 0x000000FF000000FFull) << 16) |
298 ((v >> 16) & 0x000000FF000000FFull);
299 }
300 for (size_t i = words * 8; i < bytes; i += 4) {
301 const uint32_t v = *reinterpret_cast<const uint32_t *>(
302 static_cast<const uint8_t *>(slot.mapped) + i);
303 const uint32_t out = (v & 0xFF00FF00u) | ((v & 0xFFu) << 16) | ((v >> 16) & 0xFFu);
304 std::memcpy(lastFrameRgba.data() + i, &out, 4);
305 }
306 } else {
307 std::memcpy(lastFrameRgba.data(), slot.mapped, bytes);
308 }
309 readbackCpuSynced = true;
310}

336Color Graphics::getPixel(int x, int y) {
337 syncReadbackCpu();
338 if (!hasPresentedFrame || lastFrameRgba.empty())
339 throw Exception("Graphics::getPixel: no presented frame");
340 if (x < 0 || y < 0 || x >= width || y >= height)
341 throw Exception("Graphics::getPixel: out of bounds (%d,%d)", x, y);
342
343 const int pxX = (width > 0) ? int((int64_t(x) * pixelWidth) / width) : x;
344 const int pxY = (height > 0) ? int((int64_t(y) * pixelHeight) / height) : y;
345 const int cx = std::min(std::max(pxX, 0), pixelWidth - 1);
346 const int cy = std::min(std::max(pxY, 0), pixelHeight - 1);
347 const size_t i = (size_t(cy) * size_t(pixelWidth) + size_t(cx)) * 4;
348 float r = lastFrameRgba[i + 0] / 255.f;
349 float g = lastFrameRgba[i + 1] / 255.f;
350 float b = lastFrameRgba[i + 2] / 255.f;
351 float a = lastFrameRgba[i + 3] / 255.f;
352 // If surface ended up sRGB, convert encoded bytes back to linear Color space.
353 const vk::Format fmt = swapchain.image_format;
354 if (fmt == vk::Format::eB8G8R8A8Srgb || fmt == vk::Format::eR8G8B8A8Srgb) {
355 auto toLinear = [](float u) {
356 return (u <= 0.04045f) ? (u / 12.92f) : std::pow((u + 0.055f) / 1.055f, 2.4f);
357 };
358 r = toLinear(r);
359 g = toLinear(g);
360 b = toLinear(b);
361 }
362 return Color(r, g, b, a);
363}

365image::ImageData *Graphics::renderEntityIdMask(
366 const std::vector<eve::graphics::Graphics::EntityIdDraw> &draws, const glm::mat4 &viewProj,
367 int maskW, int maskH) {
368 if (!initialized || maskW <= 0 || maskH <= 0) return nullptr;
369 // G-buffer pipeline / render pass are created lazily by createGBufferResources,
370 // so that must run before the availability check.
371 createGBufferResources(maskW, maskH);
372 if (!gbufferPipeline || !gbufferRenderPass) return nullptr;
373 auto *slot = currentGBufferSlot();
374 if (!slot || !slot->framebuffer || !whiteTexture) return nullptr;
375
376 // Render each mesh with a flat idColor into the G-buffer albedo attachment
377 // (location 2) by reusing the G-buffer pipeline (its fragment shader writes
378 // outAlbedo = albedo * tint, so passing a white texture + idColor tint gives
379 // a per-pixel flat entity-ID color).
380 std::vector<GBufferDraw> idDraws;
381 idDraws.reserve(draws.size());
382 auto u8 = [](float x) -> uint32_t {
383 return uint32_t(std::lround(std::clamp(x, 0.f, 1.f) * 255.f));
384 };
385 for (const auto &d : draws) {
386 if (!d.mesh || !d.mesh->gpuHandle) continue;
387 GBufferDraw gd{};
388 gd.mesh = d.mesh;
389 gd.albedo = whiteTexture;
390 gd.push.mvp = viewProj * d.model;
391 gd.push.modelR0 = glm::vec4(d.model[0][0], d.model[1][0], d.model[2][0], d.model[3][0]);
392 gd.push.modelR1 = glm::vec4(d.model[0][1], d.model[1][1], d.model[2][1], d.model[3][1]);
393 gd.push.modelR2 = glm::vec4(d.model[0][2], d.model[1][2], d.model[2][2], d.model[3][2]);
394 const uint32_t packed = u8(d.idColor.r) | (u8(d.idColor.g) << 8) |
395 (u8(d.idColor.b) << 16) | (u8(d.idColor.a) << 24);
396 gd.push.clip = glm::vec4(0.1f, 100.f, glm::uintBitsToFloat(packed), 0.f);
397 idDraws.push_back(gd);
398 }
399 if (idDraws.empty()) return nullptr;
400
401 const uint32_t w = uint32_t(maskW);
402 const uint32_t h = uint32_t(maskH);
403 const vk::DeviceSize byteSize = vk::DeviceSize(w) * vk::DeviceSize(h) * 4;
404 vkb::GenericBuffer staging(device, vk::BufferUsageFlagBits::eTransferDst, byteSize,
405 vk::MemoryPropertyFlagBits::eHostVisible |
406 vk::MemoryPropertyFlagBits::eHostCoherent);
407
408 vkb::executeImmediately(device.instance, uploadPool, device.getQueue(vkb::QueueType::graphics),
409 [&](vk::CommandBuffer cb) {
410 std::array<vk::ClearValue, 4> clears{};
411 clears[0].color = vk::ClearColorValue(std::array<float, 4>{0, 0, 0, 0});
412 clears[1].color = vk::ClearColorValue(std::array<float, 4>{0, 0, 0, 0});
413 clears[2].color = vk::ClearColorValue(std::array<float, 4>{0, 0, 0, 0});
414 clears[3].depthStencil = vk::ClearDepthStencilValue{1.0f, 0};
415 vk::RenderPassBeginInfo rpBegin{};
416 rpBegin.renderPass = gbufferRenderPass;
417 rpBegin.framebuffer = slot->framebuffer;
418 rpBegin.renderArea = vk::Rect2D{{0, 0}, {w, h}};
419 rpBegin.clearValueCount = uint32_t(clears.size());
420 rpBegin.pClearValues = clears.data();
421 slot->normal.beginColorAttachment();
422 slot->depthColor.beginColorAttachment();
423 slot->albedo.beginColorAttachment();
424 slot->depth.beginDepthAttachment();
425 cb.beginRenderPass(rpBegin, vk::SubpassContents::eInline);
426 setViewportAndScissor(cb, w, h);
427 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, gbufferPipeline);
428 for (const auto &d : idDraws) {
429 auto *gpuMesh = static_cast<GpuMesh *>(d.mesh->gpuHandle);
430 if (!gpuMesh) continue;
431 if (whiteTexture && whiteTexture->gpuHandle && texSetLayout) {
432 auto *gpuTex = static_cast<GpuTexture *>(whiteTexture->gpuHandle);
433 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics,
434 gbufferPipelineLayout, 0, 1,
435 gpuTex->descriptorSet.ptr(), 0, nullptr);
436 }
437 cb.pushConstants(gbufferPipelineLayout,
438 vk::ShaderStageFlagBits::eVertex |
439 vk::ShaderStageFlagBits::eFragment,
440 0, sizeof(GBufferPush), &d.push);
441 drawIndexedMesh(cb, *gpuMesh);
442 }
443 cb.endRenderPass();
444 slot->normal.endSampledLayout();
445 slot->depthColor.endSampledLayout();
446 slot->albedo.endSampledLayout();
447 slot->depth.endSampledLayout();
448
449 // Copy the albedo attachment (location 2 = ID colors) to CPU.
450 slot->albedo.setLayout(cb, vk::ImageLayout::eTransferSrcOptimal);
451 vk::BufferImageCopy region{};
452 region.imageSubresource = {vk::ImageAspectFlagBits::eColor, 0, 0, 1};
453 region.imageExtent = vk::Extent3D{w, h, 1};
454 cb.copyImageToBuffer(slot->albedo.image(),
455 vk::ImageLayout::eTransferSrcOptimal, staging.buffer,
456 region);
457 slot->albedo.setLayout(cb, vk::ImageLayout::eShaderReadOnlyOptimal);
458 });
459
460 auto *img = new image::ImageData(int(w), int(h), "RGBA8");
461 void *mapped = staging.map();
462 std::memcpy(img->getData(), mapped, size_t(byteSize));
463 staging.unmap();
464 staging.release();
465
466 // 让 RenderControl 的 GBuffer 也指向该槽位（镜像 endGBufferPass），这样
467 // 上层可通过 getDepthTexture()/getNormalTexture() 读取本次离屏 ID 渲染
468 // 生成的深度/法线（供 capture_render_frame 的 depth/normal 复用）。
469 if (RenderControl *rc = getRenderControl()) {
470 rc->getGBuffer()->setTargets(int(w), int(h), &slot->depthColorTex, &slot->normalTex,
471 &slot->albedoTex, &slot->depthTex);
472 }
473 return img;
474}

476image::ImageData *Graphics::readGBufferToImageData(const std::string &attachment) {
477 if (!initialized) return nullptr;
478 auto *slot = currentGBufferSlot();
479 if (!slot) return nullptr;
480 vkb::ColorTarget *src = nullptr;
481 if (attachment == "depth")
482 src = &slot->depthColor; // RGBA8 linear depth
483 else if (attachment == "normal")
484 src = &slot->normal;
485 else if (attachment == "albedo")
486 src = &slot->albedo;
487 else
488 return nullptr;
489
490 const uint32_t w = uint32_t(gbufferWidth);
491 const uint32_t h = uint32_t(gbufferHeight);
492 if (w == 0 || h == 0) return nullptr;
493
494 const vk::DeviceSize byteSize = vk::DeviceSize(w) * vk::DeviceSize(h) * 4;
495 if (gbufferPending && !gbufferPassDraws.empty()) recordDeferredFrameGraph();

498 vk::MemoryPropertyFlagBits::eHostCoherent);

510 });

705void Graphics::drawPrimitiveCanvas(const PrimitiveCanvas2D &canvas) {
706 const int targetWidth = activeCanvas ? activeCanvas->getWidth() : getWidth();
707 const int targetHeight = activeCanvas ? activeCanvas->getHeight() : getHeight();
708 const auto triangles = resolvePrimitiveStrokes2D(canvas, {targetWidth, targetHeight});
709 if (triangles.vertices.empty()) return;
710 auto logicalPoint = [targetWidth, targetHeight](const PrimitiveTriangleVertex &vertex) {
711 const glm::vec2 ndc = glm::vec2(vertex.clipPosition) / vertex.clipPosition.w;
712 return glm::vec2((ndc.x + 1.f) * 0.5f * static_cast<float>(targetWidth),
713 (ndc.y + 1.f) * 0.5f * static_cast<float>(targetHeight));
714 };
715 auto &spans = recordingEngine3D_ ? engine3DSpans : overlaySpans;
716 for (const ResolvedPrimitiveBatch2D &resolvedBatch : triangles.batches2D) {
717 auto it = std::find_if(solidBatches.begin(), solidBatches.end(),
718 [&](const SolidBatch &batch) { return batch.blend == resolvedBatch.blend; });
719 if (it == solidBatches.end()) {
720 solidBatches.push_back(SolidBatch{resolvedBatch.blend, Batcher{}});
721 it = solidBatches.end() - 1;
722 }
723 for (std::size_t i = resolvedBatch.firstVertex; i < resolvedBatch.firstVertex + resolvedBatch.vertexCount;
724 i += 3) {
725 it->batch.addTriangle(logicalPoint(triangles.vertices[i]), logicalPoint(triangles.vertices[i + 1]),
726 logicalPoint(triangles.vertices[i + 2]), triangles.vertices[i].color,
727 triangles.vertices[i + 1].color, triangles.vertices[i + 2].color);
728 }
729 const auto batchIndex = static_cast<std::uint32_t>(it - solidBatches.begin());
730 const std::uint32_t count = static_cast<std::uint32_t>(resolvedBatch.vertexCount);
731 const std::uint32_t end = static_cast<std::uint32_t>(it->batch.vertices().size());
732 spans.push_back({OverlayKind::Solid, batchIndex, end - count, count});
733 }
734}

751vk::Sampler Graphics::createVkSampler(const TextureSampler &sampler, uint32_t mipLevels) const {
752 auto wrapMode = [](bool repeat) {
753 return repeat ? vk::SamplerAddressMode::eRepeat : vk::SamplerAddressMode::eClampToEdge;
754 };
755 auto toFilter = [](FilterMode m) {
756 return m == FilterMode::Nearest ? vk::Filter::eNearest : vk::Filter::eLinear;
757 };
758 auto toMip = [](MipmapMode m) {
759 return m == MipmapMode::Nearest ? vk::SamplerMipmapMode::eNearest
760 : vk::SamplerMipmapMode::eLinear;
761 };
762
763 const bool useMips = sampler.mipmap != MipmapMode::Disabled && mipLevels > 1;
764 float maxLod = useMips ? std::min(sampler.maxLod, float(mipLevels - 1)) : 0.f;
765 if (maxLod < sampler.minLod) maxLod = sampler.minLod;
766
767 float aniso = 1.f;
768 bool enableAniso = false;
769 if (sampler.maxAnisotropy > 1.f && maxSamplerAnisotropy > 1.f) {
770 enableAniso = true;
771 aniso = std::min(sampler.maxAnisotropy, maxSamplerAnisotropy);
772 }
773
774 vkb::SamplerBuilder sb;
775 return sb.magFilter(toFilter(sampler.mag))
776 .minFilter(toFilter(sampler.min))
777 .mipmapMode(useMips ? toMip(sampler.mipmap) : vk::SamplerMipmapMode::eNearest)
778 .addressModeU(wrapMode(sampler.repeatU))
779 .addressModeV(wrapMode(sampler.repeatV))
780 .addressModeW(wrapMode(sampler.repeatW))
781 .mipLodBias(sampler.lodBias)
782 .anisotropyEnable(enableAniso ? VK_TRUE : VK_FALSE)
783 .maxAnisotropy(aniso)
784 .minLod(useMips ? sampler.minLod : 0.f)
785 .maxLod(maxLod)
786 .build(device);
787}

936void Graphics::drawUiTextureRects(void *commandBuffer, const std::vector<UiTextureDraw> &draws) {
937 if (!commandBuffer || draws.empty() || !uiTexturePipeline || uiColorWidth <= 0 ||
938 uiColorHeight <= 0)
939 return;
940
941 vk::CommandBuffer cb(static_cast<VkCommandBuffer>(commandBuffer));
942 auto &buffers = currentFrame2DBuffers().uiTexBufs;
943 std::size_t bufferIndex = 0;
944 setViewportAndScissor(cb, uint32_t(uiColorWidth), uint32_t(uiColorHeight));
945
946 for (const UiTextureDraw &draw : draws) {
947 if (!draw.texture || !draw.texture->gpuHandle || draw.w <= 0.f || draw.h <= 0.f)
948 continue;
949 auto *gpu = static_cast<GpuTexture *>(draw.texture->gpuHandle);
950 const vk::DescriptorSet set = gpu->descriptorSet;
951 if (!set) continue;
952
953 Batcher batch;
954 batch.addTexturedRect(draw.x, draw.y, draw.w, draw.h, draw.tint, draw.u0, draw.v0,
955 draw.u1, draw.v1);
956 batch.toNDC(uiColorWidth, uiColorHeight);
957 std::vector<TexturedVertex> vertices;
958 vertices.reserve(batch.vertices().size());
959 for (const auto &vertex : batch.vertices())
960 vertices.push_back(TexturedVertex{vertex.pos, vertex.color, vertex.uv});
961
962 if (bufferIndex >= buffers.size()) buffers.emplace_back();
963 vkb::HostVertexBuffer &vertexBuffer = buffers[bufferIndex++];
964 vertexBuffer.allocate<TexturedVertex>(frameToken(), device, vertices);
965
966 const int clipX = std::clamp(int(std::floor(draw.clipX)), 0, uiColorWidth);
967 const int clipY = std::clamp(int(std::floor(draw.clipY)), 0, uiColorHeight);
968 const int clipRight = std::clamp(int(std::ceil(draw.clipX + draw.clipW)), 0, uiColorWidth);
969 const int clipBottom = std::clamp(int(std::ceil(draw.clipY + draw.clipH)), 0, uiColorHeight);
970 if (clipRight <= clipX || clipBottom <= clipY) continue;
971 const vk::Rect2D scissor{{clipX, clipY},
972 {uint32_t(clipRight - clipX), uint32_t(clipBottom - clipY)}};
973 cb.setScissor(0, 1, &scissor);
974 cb.bindPipeline(vk::PipelineBindPoint::eGraphics,
975 draw.opaque ? uiTextureOpaquePipeline : uiTexturePipeline);
976 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, texPipelineLayout, 0, 1, &set, 0,
977 nullptr);
978 const vk::DeviceSize offset = 0;
979 cb.bindVertexBuffers(0, 1, vertexBuffer, &offset);
980 cb.draw(uint32_t(vertices.size()), 1, 0, 0);
981 }

1254bool Graphics::releaseShader(Shader *shader) {
1255 if (!shader || !shader->gpuHandle) return false;
1256
1257 auto *gpu = static_cast<GpuShader *>(shader->gpuHandle);
1258 auto gpuIt = std::find_if(ownedGpuShaders.begin(), ownedGpuShaders.end(),
1259 [&](const std::unique_ptr<GpuShader> &g) {
1260 return g.get() == gpu;
1261 });
1262 if (gpuIt == ownedGpuShaders.end()) return false;
1263
1264 auto shIt = std::find_if(ownedShaders.begin(), ownedShaders.end(),
1265 [&](const std::unique_ptr<Shader> &s) {
1266 return s.get() == shader;
1267 });
1268 if (shIt == ownedShaders.end()) return false;
1269
1270 // Mirror ~Graphics: pipelines are raw handles that must be destroyed here;
1271 // pipelineLayout is shared and must not be destroyed per-shader.
1272 waitForSharedGpuResources();
1273 if (gpu->swapchainPipeline) device->destroyPipeline(gpu->swapchainPipeline);
1274 if (gpu->offscreenPipeline) device->destroyPipeline(gpu->offscreenPipeline);
1275 if (gpu->swapchainOpaquePipeline) device->destroyPipeline(gpu->swapchainOpaquePipeline);
1276 if (gpu->offscreenOpaquePipeline) device->destroyPipeline(gpu->offscreenOpaquePipeline);
1277 if (gpu->hdrOffscreenPipeline) device->destroyPipeline(gpu->hdrOffscreenPipeline);
1278 if (gpu->hdrOffscreenOpaquePipeline)
1279 device->destroyPipeline(gpu->hdrOffscreenOpaquePipeline);
1280 if (gpu->mesh3dPipeline) device->destroyPipeline(gpu->mesh3dPipeline);
1281 if (gpu->mesh3dXrayPipeline) device->destroyPipeline(gpu->mesh3dXrayPipeline);
1282 if (gpu->mesh3dOffscreenPipeline)
1283 device->destroyPipeline(gpu->mesh3dOffscreenPipeline);
1284 if (gpu->mesh3dHdrOffscreenPipeline)
1285 device->destroyPipeline(gpu->mesh3dHdrOffscreenPipeline);
1286 shader->gpuHandle = nullptr;
1287 ownedGpuShaders.erase(gpuIt);
1288 // Transfer the CPU facade to the caller instead of destroying it.
1289 (void)shIt->release();
1290 ownedShaders.erase(shIt);
1291 return true;
1292}

1324void Graphics::flushToOffscreen(OffscreenCanvas *canvas) {
1325 auto solid = std::move(solidBatches);
1326 auto textured = std::move(texturedBatches);
1327 auto lit = std::move(litBatches);
1328 auto spans = std::move(overlaySpans);
1329 clear2DBatches();
1330
1331 const Color cc = canvas->pendingClearColor();
1332 const bool needClear = canvas->takePendingClear();
1333 bool hasSolid = false;
1334 for (const auto &sb : solid)
1335 if (!sb.batch.empty()) hasSolid = true;
1336 if (!hasSolid && textured.empty() && lit.empty() && !needClear) return;
1337
1338 // Offscreen color is a single shared image. An in-flight swapchain frame
1339 // may still be sampling it (draw canvas to screen last frame), so drain
1340 // those frames before transitioning it back to a color attachment.
1341 waitForSharedGpuResources();
1342
1343 auto recordOffscreen = [&](vk::CommandBuffer cb) {
1344 canvas->colorImage().setLayout(cb, vk::ImageLayout::eColorAttachmentOptimal);
1345
1346 vk::ClearValue cv{
1347 vk::ClearColorValue(std::array<float, 4>{cc.r, cc.g, cc.b, cc.a})};
1348 vk::RenderPassBeginInfo rpBegin{};
1349 rpBegin.renderPass = canvas->isHDR() ? hdrOffscreenRenderPass
1350 : offscreenRenderPass;
1351 rpBegin.framebuffer = canvas->framebuffer();
1352 rpBegin.renderArea.extent =
1353 vk::Extent2D{uint32_t(canvas->getWidth()), uint32_t(canvas->getHeight())};
1354 rpBegin.clearValueCount = 1;
1355 rpBegin.pClearValues = &cv;
1356 canvas->colorImage().beginColorAttachment();
1357 cb.beginRenderPass(rpBegin, vk::SubpassContents::eInline);
1358
1359 setViewportAndScissor(cb, uint32_t(canvas->getWidth()),
1360 uint32_t(canvas->getHeight()));
1361
1362 std::vector<vkb::HostVertexBuffer> &solidBufs =
1363 offscreenBuffers.solidBufs;
1364 std::vector<vkb::HostVertexBuffer> &texBufs = offscreenBuffers.texBufs;
1365 size_t texBufIndex = 0;
1366 const int vw = canvas->getWidth();
1367 const int vh = canvas->getHeight();
1368
1369 auto offscreenTexPipe = [&](BlendMode mode) -> vk::Pipeline {
1370 switch (mode) {
1371 case BlendMode::Additive:
1372 return offscreenAdditiveTexPipeline;
1373 case BlendMode::Premultiplied:
1374 return offscreenPremultipliedTexPipeline;
1375 case BlendMode::Multiply:
1376 return offscreenMultiplyTexPipeline;
1377 case BlendMode::Opaque:
1378 return canvas->isHDR() ? hdrOffscreenOpaqueTexPipeline
1379 : offscreenOpaqueTexPipeline;
1380 case BlendMode::Alpha:
1381 default:
1382 return canvas->isHDR() ? hdrOffscreenTexPipeline
1383 : offscreenTexPipeline;
1384 }
1385 };
1386 auto offscreenSolidPipe = [&](BlendMode mode) -> vk::Pipeline {
1387 switch (mode) {
1388 case BlendMode::Additive:
1389 return offscreenAdditiveSolidPipeline;
1390 case BlendMode::Premultiplied:
1391 return offscreenPremultipliedSolidPipeline;
1392 case BlendMode::Multiply:
1393 return offscreenMultiplySolidPipeline;
1394 case BlendMode::Alpha:
1395 return offscreenSolidAlphaPipeline;
1396 case BlendMode::Opaque:
1397 default:
1398 return offscreenSolidPipeline;
1399 }
1400 };
1401
1402 auto drawOffscreenTextured = [&](TexturedBatch &tb) {
1403 if (tb.batch.empty() || !tb.texture || !tb.texture->gpuHandle) return;
1404 auto *gpu = static_cast<GpuTexture *>(tb.texture->gpuHandle);
1405 vk::DescriptorSet texSet = gpu->descriptorSet;
1406 if ((tb.depth && tb.depth->gpuHandle) ||
1407 (tb.motion && tb.motion->gpuHandle) ||
1408 (tb.extra && tb.extra->gpuHandle)) {
1409 auto *depthGpu = tb.depth
1410 ? static_cast<GpuTexture *>(tb.depth->gpuHandle)
1411 : nullptr;
1412 auto *motionGpu = tb.motion
1413 ? static_cast<GpuTexture *>(tb.motion->gpuHandle)
1414 : nullptr;
1415 auto *extraGpu = tb.extra
1416 ? static_cast<GpuTexture *>(tb.extra->gpuHandle)
1417 : nullptr;
1418 auto *specularGpu =
1419 tb.specular
1420 ? static_cast<GpuTexture *>(tb.specular->gpuHandle)
1421 : nullptr;
1422 if (vk::DescriptorSet combo =
1423 post2SetFor(gpu, depthGpu, motionGpu, extraGpu,
1424 specularGpu))
1425 texSet = combo;
1426 }
1427 Batcher ndc = tb.batch;
1428 ndc.toNDC(vw, vh);
1429 std::vector<TexturedVertex> gpuVerts;
1430 gpuVerts.reserve(ndc.vertices().size());
1431 for (const auto &v : ndc.vertices())
1432 gpuVerts.push_back(TexturedVertex{v.pos, v.color, v.uv});
1433
1434 if (texBufIndex >= texBufs.size()) texBufs.emplace_back();
1435 vkb::HostVertexBuffer &vb = texBufs[texBufIndex++];
1436 vb.allocate<TexturedVertex>(frameToken(), device, gpuVerts);
1437
1438 if (tb.shader && tb.shader->gpuHandle) {
1439 if (canvas->isHDR())
1440 ensureShaderHdrOffscreenPipeline(tb.shader);
1441 else
1442 ensureShaderOffscreenPipeline(tb.shader);
1443 auto *gs = static_cast<GpuShader *>(tb.shader->gpuHandle);
1444 vk::Pipeline alphaPipe = canvas->isHDR()
1445 ? gs->hdrOffscreenPipeline
1446 : gs->offscreenPipeline;
1447 vk::Pipeline opaquePipe =
1448 canvas->isHDR() ? gs->hdrOffscreenOpaquePipeline
1449 : gs->offscreenOpaquePipeline;
1450 if (!alphaPipe) return;
1451 vk::Pipeline customPipeline =
1452 tb.blend == BlendMode::Opaque && opaquePipe
1453 ? opaquePipe
1454 : alphaPipe;
1455 cb.bindPipeline(vk::PipelineBindPoint::eGraphics,
1456 customPipeline);
1457 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics,
1458 shaderPipelineLayout, 0, 1,
1459 &texSet, 0, nullptr);
1460 cb.pushConstants(shaderPipelineLayout,
1461 vk::ShaderStageFlagBits::eVertex |
1462 vk::ShaderStageFlagBits::eFragment,
1463 0, Shader::kPushConstantBytes,
1464 tb.shader->pushConstantData());
1465 } else {
1466 vk::Pipeline pipe = offscreenTexPipe(tb.blend);
1467 if (!pipe) return;
1468 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, pipe);
1469 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics,
1470 texPipelineLayout, 0, 1,
1471 &texSet, 0, nullptr);
1472 }
1473 vk::DeviceSize offset = 0;
1474 cb.bindVertexBuffers(0, 1, vb, &offset);
1475 cb.draw(uint32_t(gpuVerts.size()), 1, 0, 0);
1476 };
1477
1478 std::vector<bool> solidUploaded(solid.size(), false);
1479 auto uploadSolid = [&](size_t idx) {
1480 if (idx >= solid.size() || solidUploaded[idx] ||
1481 solid[idx].batch.empty())
1482 return;
1483 vk::Pipeline pipe = offscreenSolidPipe(solid[idx].blend);
1484 if (!pipe) return;
1485 Batcher ndc = solid[idx].batch;
1486 ndc.toNDC(vw, vh);
1487 std::vector<ColorVertex> gpuVerts;
1488 gpuVerts.reserve(ndc.vertices().size());
1489 for (const auto &v : ndc.vertices())
1490 gpuVerts.push_back(ColorVertex{v.pos, v.color});
1491 if (solidBufs.size() <= idx) solidBufs.resize(idx + 1);
1492 solidBufs[idx].allocate<ColorVertex>(frameToken(), device,
1493 gpuVerts);
1494 solidUploaded[idx] = true;
1495 };
1496
1497 auto drawSolidSpan = [&](uint32_t batchIndex, uint32_t begin,
1498 uint32_t count) {
1499 if (batchIndex >= solid.size() || count == 0 ||
1500 solid[batchIndex].batch.empty())
1501 return;
1502 vk::Pipeline pipe = offscreenSolidPipe(solid[batchIndex].blend);
1503 if (!pipe) return;
1504 uploadSolid(batchIndex);
1505 vk::DeviceSize offset = 0;
1506 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, pipe);
1507 cb.bindVertexBuffers(0, 1, solidBufs[batchIndex], &offset);
1508 cb.draw(count, 1, begin, 0);
1509 };
1510
1511 if (!spans.empty()) {
1512 for (const auto &sp : spans) {
1513 if (sp.kind == OverlayKind::Solid)
1514 drawSolidSpan(sp.index, sp.vertBegin, sp.vertCount);
1515 else if (sp.kind == OverlayKind::Textured &&
1516 sp.index < textured.size())
1517 drawOffscreenTextured(textured[sp.index]);
