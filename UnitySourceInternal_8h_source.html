84inline bool validPath(std::string_view path, const AssetImportLimits& limits) {
85 if (path.empty() || path.size() > limits.maximumStringBytes || path.front() == '/' ||
86 path.find_first_of("\\:<>\"|?*") != std::string_view::npos || !isValidUtf8(path, Utf8NullPolicy::Reject))
87 return false;
88 for (unsigned char c : path)
89 if (c < 32 || c == 127) return false;
90 std::size_t start = 0;
91 while (start < path.size()) {
92 auto end = path.find('/', start);
93 if (end == std::string_view::npos) end = path.size();
94 const auto part = path.substr(start, end - start);
95 if (part.empty() || part == "." || part == ".." || part.back() == '.' || part.back() == ' ') return false;
96 start = end + 1;
97 }
98 return path.back() != '/';
99}
