60inline bool validPath(std::string_view path, const AssetImportLimits& limits) {
61 if (path.empty() || path.size() > limits.maximumStringBytes || path.front() == '/' ||
62 path.find_first_of("\\:<>\"|?*") != std::string_view::npos || !isValidUtf8(path, Utf8NullPolicy::Reject))
63 return false;
64 for (unsigned char c : path)
65 if (c < 32 || c == 127) return false;
66 std::size_t start = 0;
67 while (start < path.size()) {
68 auto end = path.find('/', start);
69 if (end == std::string_view::npos) end = path.size();
70 const auto part = path.substr(start, end - start);
71 if (part.empty() || part == "." || part == ".." || part.back() == '.' || part.back() == ' ') return false;
72 start = end + 1;
73 }
74 return path.back() != '/';
75}
