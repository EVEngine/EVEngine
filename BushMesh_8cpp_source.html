40void basisFor(V3 axis, V3 &right, V3 &forward) {
41 axis = norm(axis);

47// Closed ellipsoid. The per-vertex normal is the analytic ellipsoid normal so the
48// bush shades smoothly under directional light. UVs fall inside [uMin, uMax].
49void addEllipsoidBlob(MeshBuild &out, V3 c, V3 r, int rings, int sides, float uMin, float uMax,
50 float phase = 0.f) {
51 const uint32_t base = uint32_t(out.getVertexCount());
52 for (int y = 0; y <= rings; ++y) {
53 const float v = float(y) / float(rings);
54 const float phi = v * kPi;
55 for (int x = 0; x < sides; ++x) {
56 const float u = float(x) / float(sides);
57 const float theta = u * 2.f * kPi;
58 const V3 n = {std::sin(phi) * std::cos(theta), std::cos(phi), std::sin(phi) * std::sin(theta)};
59 // Two low-frequency harmonics break the unmistakable primitive-sphere
60 // silhouette without producing noisy, faceted normals. Keep both poles
61 // fixed so neighbouring lobes still close cleanly.
62 const float equator = std::sin(phi);
63 const float swell = 1.f + equator * (0.075f * std::sin(theta * 3.f + phase) +
64 0.035f * std::sin(theta * 5.f - phase * 0.7f));
65 const V3 p = {c.x + n.x * r.x * swell, c.y + n.y * r.y, c.z + n.z * r.z * swell};
66 // Ellipsoid normal = N/R divided by its length (finite since radii > 0).
67 const V3 normal = norm({n.x / r.x, n.y / r.y, n.z / r.z});
68 out.addVertex(p.x, p.y, p.z, normal.x, normal.y, normal.z, uMin + u * (uMax - uMin), v);
69 }
70 }
71 for (int y = 0; y < rings; ++y) {
72 for (int x = 0; x < sides; ++x) {
73 const int nx = (x + 1) % sides;
74 const uint32_t a = base + uint32_t(y * sides + x);
75 const uint32_t b = base + uint32_t(y * sides + nx);
76 const uint32_t c = base + uint32_t((y + 1) * sides + x);
77 const uint32_t d = base + uint32_t((y + 1) * sides + nx);
78 out.addTriangle(a, c, b);
79 out.addTriangle(b, c, d);
80 }
81 }
82}

84// Open tapered cylinder (a woody twig). Vertices use the bark UV half.
85void addTwig(MeshBuild &out, V3 a, V3 b, float r0, float r1, int sides, float uMin, float uMax) {
86 const V3 axis = norm(sub(b, a));
87 V3 right, forward;
88 basisFor(axis, right, forward);
89 const uint32_t base = uint32_t(out.getVertexCount());
90 for (int ring = 0; ring < 2; ++ring) {
91 const V3 center = ring ? b : a;
92 const float radius = ring ? r1 : r0;
93 for (int i = 0; i < sides; ++i) {
94 const float t = float(i) / float(sides);
95 const float angle = t * 2.f * kPi;
96 const V3 radial = add(mul(right, std::cos(angle)), mul(forward, std::sin(angle)));
97 const V3 p = add(center, mul(radial, radius));
98 out.addVertex(p.x, p.y, p.z, radial.x, radial.y, radial.z, uMin + t * (uMax - uMin), float(ring));
99 }
100 }
101 for (int i = 0; i < sides; ++i) {
102 const uint32_t n = uint32_t((i + 1) % sides);
103 const uint32_t i0 = base + uint32_t(i), i1 = base + n;
104 const uint32_t i2 = base + uint32_t(sides) + uint32_t(i);
105 const uint32_t i3 = base + uint32_t(sides) + n;
106 out.addTriangle(i0, i2, i1);
107 out.addTriangle(i1, i2, i3);
108 }
109}

111// A rounded, slightly cupped leaf. The eight-point outline avoids both the
112// rectangular-card look and the long triangular spikes of a diamond leaf.
113void addLeafCard(MeshBuild &out, std::mt19937 &rng, V3 c, V3 direction, float size, float uMin, float uMax) {
114 V3 right, up;
115 basisFor(norm(direction), right, up);
116 const float twist = randomRange(rng, 0.f, 2.f * kPi);
117 right = add(mul(right, std::cos(twist)), mul(up, std::sin(twist)));
118 up = norm(cross(norm(direction), right));
119 const V3 normal = norm(cross(right, up));
120 const float halfWidth = size * randomRange(rng, 0.34f, 0.44f);
121 const float halfLength = size * randomRange(rng, 0.58f, 0.72f);
122 const uint32_t base = uint32_t(out.getVertexCount());
123 const V3 center = add(c, mul(normal, size * 0.06f));
124 out.addVertex(center.x, center.y, center.z, normal.x, normal.y, normal.z,
125 (uMin + uMax) * 0.5f, 0.5f);
126 constexpr int outline = 8;
127 for (int i = 0; i < outline; ++i) {
128 const float angle = -0.5f * kPi + float(i) * 2.f * kPi / float(outline);
129 const float x = std::cos(angle) * halfWidth;
130 const float y = std::sin(angle) * halfLength;
131 const V3 point = add(c, add(mul(right, x), mul(up, y)));
132 out.addVertex(point.x, point.y, point.z, normal.x, normal.y, normal.z,
133 uMin + (0.5f + x / (2.f * halfWidth)) * (uMax - uMin), 0.5f + y / (2.f * halfLength));
134 }
135 for (int i = 0; i < outline; ++i) {
136 const uint32_t a = base + 1u + uint32_t(i);
137 const uint32_t b = base + 1u + uint32_t((i + 1) % outline);
138 out.addTriangle(base, a, b);
139 out.addTriangle(base, b, a);
140 }

145bool generateBushMesh(const Params &params, MeshBuild &out, std::string &error) {
146 const std::string style = params.getString("style", "mound");
147 const std::string leafMode = params.getString("leafMode", "mixed");
148 if (style != "mound" && style != "sphere") {
149 error = "mesh.bush: style must be mound|sphere";
150 return false;
151 }
152 if (leafMode != "blobs" && leafMode != "cards" && leafMode != "mixed" && leafMode != "none") {
153 error = "mesh.bush: leafMode must be blobs|cards|mixed|none";
154 return false;
155 }
156
157 const float height = std::max(0.3f, params.getFloat("height", 1.4f));
158 const float width = std::max(0.4f, params.getFloat("width", 2.2f));
159 const float halfW = width * 0.5f;
160 const int blobs = std::clamp(params.getInt("blobs", 9), 1, 40);
161 const int rings = std::clamp(params.getInt("rings", 3), 2, 10);
162 const int sides = std::clamp(params.getInt("radialSegments", 7), 4, 24);
163 const float density = std::clamp(params.getFloat("leafDensity", 0.62f), 0.f, 1.f);
164 const float leafSize = std::max(0.02f, params.getFloat("leafSize", height * 0.16f));
165 const int twigs = std::clamp(params.getInt("twigs", 4), 0, 16);
166 const float twigLen = std::max(0.05f, params.getFloat("twigLength", height * 0.30f));
167 const float twigScale = std::clamp(twigLen / (height * 0.30f), 0.35f, 1.65f);
168 const bool sphere = style == "sphere";
169
170 std::mt19937 rng(params.getSeed());
171 out.clear();
172
173 // Build the woody skeleton first. Each stem bends outward and forks once;
174 // foliage is then placed around these endpoints instead of being scattered
175 // independently. The clear lower third keeps the branch structure readable.
176 std::vector<V3> crownAnchors;
177 const float branchR = width * 0.022f;
178 for (int i = 0; i < twigs; ++i) {
179 const float angle = float(i) * 2.399963f + randomRange(rng, -0.12f, 0.12f);
180 const V3 root{std::cos(angle) * halfW * 0.05f, 0.02f, std::sin(angle) * halfW * 0.05f};
181 const V3 joint{std::cos(angle) * halfW * randomRange(rng, 0.22f, 0.34f) * twigScale,
182 height * randomRange(rng, 0.30f, 0.43f) * twigScale,
183 std::sin(angle) * halfW * randomRange(rng, 0.22f, 0.34f) * twigScale};
184 const float reach = halfW * randomRange(rng, 0.58f, 0.76f) * twigScale;
185 const V3 tip{std::cos(angle) * reach, height * randomRange(rng, 0.58f, 0.78f) * twigScale,
186 std::sin(angle) * reach};
187 const float forkAngle = angle + (i & 1 ? 0.38f : -0.38f);
188 const V3 forkTip{joint.x + std::cos(forkAngle) * halfW * randomRange(rng, 0.28f, 0.40f) * twigScale,
189 joint.y + height * randomRange(rng, 0.22f, 0.33f) * twigScale,
190 joint.z + std::sin(forkAngle) * halfW * randomRange(rng, 0.28f, 0.40f) * twigScale};
191 addTwig(out, root, joint, branchR * 1.35f, branchR, std::max(5, sides - 2), kBarkUMin, kBarkUMax);
192 addTwig(out, joint, tip, branchR, branchR * 0.38f, std::max(4, sides - 3), kBarkUMin, kBarkUMax);
193 addTwig(out, joint, forkTip, branchR * 0.78f, branchR * 0.28f, std::max(4, sides - 3), kBarkUMin,
194 kBarkUMax);
195 crownAnchors.push_back(tip);
196 crownAnchors.push_back(forkTip);
197 }
198
199 // Cluster squashed lobes under a dome silhouette so the bush reads as one
200 // rounded mound rather than a set of disconnected balls.
201 for (int i = 0; i < blobs; ++i) {
202 const float radial = halfW * 0.58f * std::sqrt(random01(rng));
203 const float theta = randomRange(rng, 0.f, 2.f * kPi);
204 const float heightFactor = 1.f - (radial / halfW) * (radial / halfW);
205 const float cy = height * (sphere ? 0.5f + 0.20f * random01(rng)
206 : 0.18f + heightFactor * (0.34f + 0.34f * random01(rng)));
207 const float rx = halfW * 0.29f * randomRange(rng, 0.82f, 1.16f);
208 const float ry = height * (sphere ? 0.21f : 0.22f) * randomRange(rng, 0.65f, 1.02f);
209 const float rz = rx * randomRange(rng, 0.80f, 1.20f);
210 V3 center{std::cos(theta) * radial, cy, std::sin(theta) * radial};
211 if (!crownAnchors.empty() && i < int(crownAnchors.size())) {
212 center = crownAnchors[size_t(i)];
213 center.x += randomRange(rng, -0.10f, 0.10f) * halfW;
214 center.y += randomRange(rng, -0.04f, 0.08f) * height;
215 center.z += randomRange(rng, -0.10f, 0.10f) * halfW;
216 }
217 addEllipsoidBlob(out, center, {rx, ry, rz}, rings, sides, kFoliageUMin, kFoliageUMax,
218 randomRange(rng, 0.f, 2.f * kPi));
219 }
220 // Always cap the top so the dome has no gap at its peak.
221 const float topRx = halfW * 0.18f;
222 addEllipsoidBlob(out, {0.f, height * (sphere ? 0.62f : 0.72f), 0.f}, {topRx, height * 0.20f, topRx},
223 rings, sides, kFoliageUMin, kFoliageUMax, randomRange(rng, 0.f, 2.f * kPi));
224
225 // Optional loose leaf cards across the canopy for a fuller look.
226 if (leafMode == "cards" || leafMode == "mixed") {
227 const int cards = std::max(1, int(std::round(float(blobs) * 8.f * density)));
228 for (int i = 0; i < cards; ++i) {
229 const float radial = halfW * 0.86f * std::sqrt(random01(rng));
230 const float theta = randomRange(rng, 0.f, 2.f * kPi);
231 const float heightFactor = 1.f - (radial / halfW) * (radial / halfW);
232 const V3 c{std::cos(theta) * radial,
233 height * (0.20f + heightFactor * randomRange(rng, 0.38f, 0.72f)),
234 std::sin(theta) * radial};
235 const V3 face = norm({randomRange(rng, -1.f, 1.f), randomRange(rng, 0.2f, 0.9f),
236 randomRange(rng, -1.f, 1.f)});
237 addLeafCard(out, rng, c, face, leafSize * randomRange(rng, 0.70f, 1.20f), kFoliageUMin, kFoliageUMax);
238 }
239 }
240
241 out.setMeta("recipe", "mesh.bush");
242 out.setMeta("style", style);
243 out.setMeta("leafMode", leafMode);
244 out.setMeta("seed", std::to_string(params.getSeed()));
245 if (out.empty()) {
246 error = "mesh.bush: generated an empty mesh";
247 return false;
248 }
249 return true;
250}
