91[[nodiscard]] inline MeshArrays toMeshArrays(const MeshBake& bake) {
92 MeshArrays out;
93 out.positions.reserve(bake.positions.size() * 3);
94 out.normals.reserve(bake.positions.size() * 3);
95 out.uvs.reserve(bake.positions.size() * 2);
96 out.indices = bake.indices;
97 for (const Vec3& p : bake.positions) {
98 out.positions.push_back(static_cast<float>(p.x));
99 out.positions.push_back(static_cast<float>(p.y));
100 out.positions.push_back(static_cast<float>(p.z));
101 out.normals.push_back(0.f);
102 out.normals.push_back(1.f);
103 out.normals.push_back(0.f);
104 out.uvs.push_back(static_cast<float>(p.x));
105 out.uvs.push_back(static_cast<float>(p.z));
106 }
107 for (std::size_t i = 0; i + 2 < bake.indices.size(); i += 3) {
108 const std::uint32_t i0 = bake.indices[i], i1 = bake.indices[i + 1], i2 = bake.indices[i + 2];
109 if (i0 >= bake.positions.size() || i1 >= bake.positions.size() || i2 >= bake.positions.size()) continue;
110 const Vec3& a = bake.positions[i0];
111 const Vec3& b = bake.positions[i1];
112 const Vec3& c = bake.positions[i2];
113 Vec3 n{ (b.y - a.y) * (c.z - a.z) - (b.z - a.z) * (c.y - a.y),
114 (b.z - a.z) * (c.x - a.x) - (b.x - a.x) * (c.z - a.z),
115 (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x) };
116 const double len = std::sqrt(n.x * n.x + n.y * n.y + n.z * n.z);
117 if (len > 1e-8) {
118 n.x /= len;
119 n.y /= len;
120 n.z /= len;
121 } else {
122 n = {0, 1, 0};
123 }
124 for (std::uint32_t idx : {i0, i1, i2}) {
125 out.normals[idx * 3 + 0] = static_cast<float>(n.x);
126 out.normals[idx * 3 + 1] = static_cast<float>(n.y);
127 out.normals[idx * 3 + 2] = static_cast<float>(n.z);
128 }
129 }
130 return out;
131}
