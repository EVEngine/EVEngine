161[[nodiscard]] inline MeshArrays toMeshArrays(const MeshBake& bake) {
162 MeshArrays out;
163 out.positions.reserve(bake.positions.size() * 3);
164 out.normals.reserve(bake.positions.size() * 3);
165 out.uvs.reserve(bake.positions.size() * 2);
166 out.indices = bake.indices;
167 for (const Vec3& p : bake.positions) {
168 out.positions.push_back(static_cast<float>(p.x));
169 out.positions.push_back(static_cast<float>(p.y));
170 out.positions.push_back(static_cast<float>(p.z));
171 out.normals.push_back(0.f);
172 out.normals.push_back(1.f);
173 out.normals.push_back(0.f);
174 out.uvs.push_back(static_cast<float>(p.x));
175 out.uvs.push_back(static_cast<float>(p.z));
176 }
177 for (std::size_t i = 0; i + 2 < bake.indices.size(); i += 3) {
178 const std::uint32_t i0 = bake.indices[i], i1 = bake.indices[i + 1], i2 = bake.indices[i + 2];
179 if (i0 >= bake.positions.size() || i1 >= bake.positions.size() || i2 >= bake.positions.size()) continue;
180 const Vec3& a = bake.positions[i0];
181 const Vec3& b = bake.positions[i1];
182 const Vec3& c = bake.positions[i2];
183 Vec3 n{(b.y - a.y) * (c.z - a.z) - (b.z - a.z) * (c.y - a.y),
184 (b.z - a.z) * (c.x - a.x) - (b.x - a.x) * (c.z - a.z),
185 (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x)};
186 const double len = std::sqrt(n.x * n.x + n.y * n.y + n.z * n.z);
187 if (len > 1e-8) {
188 n.x /= len;
189 n.y /= len;
190 n.z /= len;
191 } else {
192 n = {0, 1, 0};
193 }
194 for (std::uint32_t idx : {i0, i1, i2}) {
195 out.normals[idx * 3 + 0] = static_cast<float>(n.x);
196 out.normals[idx * 3 + 1] = static_cast<float>(n.y);
197 out.normals[idx * 3 + 2] = static_cast<float>(n.z);
198 }
199 }
200 return out;
201}
