18bool parseCsv(const std::string& source, CsvRows& rows, std::string& error) {
19 std::vector<std::string> row;
20 std::string field;
21 bool quoted = false;
22 for (size_t i = 0; i <= source.size(); ++i) {
23 const char ch = i < source.size() ? source[i] : '\n';
24 if (quoted) {
25 if (ch == '"' && i + 1 < source.size() && source[i + 1] == '"') {
26 field.push_back('"');
27 ++i;
28 } else if (ch == '"')
29 quoted = false;
30 else
31 field.push_back(ch);
32 } else if (ch == '"' && field.empty())
33 quoted = true;
34 else if (ch == ',') {
35 row.push_back(std::move(field));
36 field.clear();
37 } else if (ch == '\n') {
38 row.push_back(trim(std::move(field)));
39 field.clear();
40 if (!(row.size() == 1 && row.front().empty())) rows.push_back(std::move(row));
41 row.clear();
42 } else if (ch != '\r')
43 field.push_back(ch);
44 }
45 if (quoted) {
46 error = "unterminated quoted CSV field";
47 return false;
48 }
49 if (!rows.empty() && !rows[0].empty() && rows[0][0].rfind("\xEF\xBB\xBF", 0) == 0) rows[0][0].erase(0, 3);
50 return true;
51}

53std::string quote(std::string value) {
54 size_t position = 0;
55 while ((position = value.find('"', position)) != std::string::npos) {
56 value.insert(position, 1, '"');
57 position += 2;
58 }
59 return '"' + value + '"';
60}

70int ConversationLocalizationCatalog::importCsv(const std::string& csv, const std::string& defaultLocale,
71 std::vector<ConversationDiagnostic>& diagnostics) {
72 CsvRows rows;
73 std::string error;
74 if (!parseCsv(csv, rows, error) || rows.empty()) {
75 diagnostics.push_back({ConversationDiagnostic::Severity::Error, "<localization.csv>", 0,
76 error.empty() ? "localization CSV is empty" : error});
77 return 0;
78 }
79 std::unordered_map<std::string, size_t> columns;
80 for (size_t i = 0; i < rows.front().size(); ++i) columns[rows.front()[i]] = i;
81 if (!columns.count("i18n_key")) {
82 diagnostics.push_back({ConversationDiagnostic::Severity::Error, "<localization.csv>", 1,
83 "localization CSV requires an i18n_key column"});
84 return 0;
85 }
86 const auto cell = [&](const std::vector<std::string>& row, const char* name) -> std::string {
87 const auto it = columns.find(name);
88 return it != columns.end() && it->second < row.size() ? row[it->second] : std::string{};
89 };
90 int imported = 0;
91 for (size_t line = 1; line < rows.size(); ++line) {
92 const std::string key = cell(rows[line], "i18n_key");
93 std::string locale = cell(rows[line], "locale");
94 if (locale.empty()) locale = defaultLocale;
95 if (key.empty()) {
96 diagnostics.push_back({ConversationDiagnostic::Severity::Warning, "<localization.csv>",
97 static_cast<int>(line + 1), "row has no i18n_key and was ignored"});
98 continue;
99 }
100 auto& entry = entries_[mapKey(key, locale)];
101 std::string text = cell(rows[line], "translation");
102 if (text.empty()) text = cell(rows[line], "text");
103 if (!text.empty()) entry.text = std::move(text);
104 const std::string voice = cell(rows[line], "voice");
105 if (!voice.empty()) entry.voice = voice;
106 const std::string status = cell(rows[line], "status");
107 if (!status.empty()) entry.status = status;
108 const std::string duration = cell(rows[line], "duration");
109 if (!duration.empty()) {
110 char* end = nullptr;
111 errno = 0;
112 const double value = std::strtod(duration.c_str(), &end);
113 if (end == duration.c_str() + duration.size() && errno != ERANGE)
114 entry.duration = value;
115 else
116 diagnostics.push_back({ConversationDiagnostic::Severity::Warning, "<localization.csv>",
117 static_cast<int>(line + 1), "invalid duration was ignored"});
118 }
119 ++imported;
120 }
121 return imported;
122}

124const ConversationLocalizationEntry* ConversationLocalizationCatalog::find(const std::string& key,
125 const std::string& locale) const {
126 const auto exact = entries_.find(mapKey(key, locale));
127 if (exact != entries_.end()) return &exact->second;
128 const size_t separator = locale.find_first_of("-_");
129 if (separator != std::string::npos) {
130 const auto language = entries_.find(mapKey(key, locale.substr(0, separator)));
131 if (language != entries_.end()) return &language->second;
132 }
133 const auto fallback = entries_.find(mapKey(key, {}));
134 return fallback == entries_.end() ? nullptr : &fallback->second;
