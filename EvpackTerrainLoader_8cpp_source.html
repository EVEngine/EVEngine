42Result<LoadedTerrain> EvpackTerrainLoader::load(
43 const AssetRef& terrainRef, const asset::EvpackCapabilities& capabilities,
44 const TerrainAssetLoadLimits& limits) const {
45 auto payload = reader_.read(terrainRef, "eve.terrain/1", capabilities,
46 limits.maximumDecodedBytes);
47 if (!payload) return Result<LoadedTerrain>::failure(payload.status());
48 const asset::RuntimeAssetChunk* definition = nullptr;
49 const asset::RuntimeAssetChunk* bulk = nullptr;
50 for (const auto& chunk : payload.value().chunks) {
51 if (chunk.kind == asset::EvpackChunkKind::Definition) {
52 if (definition)
53 return failure<LoadedTerrain>(DiagnosticCode::Conflict,
54 "terrain has duplicate definition chunks");
55 definition = &chunk;
56 } else if (chunk.kind == asset::EvpackChunkKind::Bulk) {
57 if (bulk)
58 return failure<LoadedTerrain>(DiagnosticCode::Conflict,
59 "terrain has duplicate bulk chunks");
60 bulk = &chunk;
61 }
62 }
63 static constexpr std::uint8_t magic[] = {'E', 'V', 'T', 'R', 'N', 0, 1, 0};
64 if (!definition || !bulk || bulk->bytes.size() < 24 ||
65 !std::equal(std::begin(magic), std::end(magic), bulk->bytes.begin()))
66 return failure<LoadedTerrain>(DiagnosticCode::ParseError,
67 "terrain definition or EVTRN payload is invalid");
68 const std::uint32_t width = little32(bulk->bytes, 8);
69 const std::uint32_t height = little32(bulk->bytes, 12);
70 const float spacingX = littleFloat(bulk->bytes, 16);
71 const float spacingZ = littleFloat(bulk->bytes, 20);
72 const std::uint64_t samples = std::uint64_t(width) * height;
73 if (width == 0 || height == 0 || width > limits.maximumDimension ||
74 height > limits.maximumDimension || samples > limits.maximumSamples ||
75 !std::isfinite(spacingX) || !std::isfinite(spacingZ) || spacingX <= 0.f ||
76 spacingZ <= 0.f || samples > (std::numeric_limits<std::uint64_t>::max() - 24) / 4 ||
77 samples * 4 + 24 != bulk->bytes.size())
78 return failure<LoadedTerrain>(DiagnosticCode::InvalidArgument,
79 "terrain dimensions, spacing, or payload size is invalid");
80 asset::RuntimeDefinitionLimits definitionLimits;
81 definitionLimits.maximumBytes = limits.maximumDecodedBytes;
82 auto metadata = asset::decodeRuntimeDefinition(definition->bytes, definitionLimits);
83 if (!metadata) return Result<LoadedTerrain>::failure(metadata.status());
84 const auto* root = metadata.value().getIf<Value::Object>();
85 const Value* schema = root ? field(*root, "schema") : nullptr;
86 const Value* version = root ? field(*root, "schemaVersion") : nullptr;
87 const Value* coordinate = root ? field(*root, "coordinateSystem") : nullptr;
88 const Value* unit = root ? field(*root, "heightUnit") : nullptr;
89 if (!schema || !schema->isString() || schema->asString() != "eve.terrain" ||
90 !version || !version->isInt64() || version->asInt() != 1 || !coordinate ||
91 !coordinate->isString() ||
92 coordinate->asString() != "right-handed-x-right-y-up-minus-z-forward" ||
93 !unit || !unit->isString() || unit->asString() != "meter" ||
94 !numericEquals(root ? field(*root, "width") : nullptr, width) ||
95 !numericEquals(root ? field(*root, "height") : nullptr, height) ||
96 !numericEquals(root ? field(*root, "spacingX") : nullptr, spacingX) ||
97 !numericEquals(root ? field(*root, "spacingZ") : nullptr, spacingZ))
98 return failure<LoadedTerrain>(DiagnosticCode::ParseError,
99 "terrain metadata does not match EVTRN payload");
100 procgen::Heightmap heightmap(static_cast<int>(width), static_cast<int>(height));
101 std::size_t cursor = 24;
102 for (std::uint32_t z = 0; z < height; ++z) {
103 for (std::uint32_t x = 0; x < width; ++x) {
104 const float value = littleFloat(bulk->bytes, cursor);
105 cursor += 4;
106 if (!std::isfinite(value))
107 return failure<LoadedTerrain>(DiagnosticCode::ParseError,
108 "terrain contains a non-finite height");
109 heightmap.setHeight(static_cast<int>(x), static_cast<int>(z), value);
110 }
111 }
112 // SpatialData currently has one cell-size axis. Preserve the exact map and reject
113 // anisotropic sampling rather than silently distorting the procgen domain.
114 if (spacingX != spacingZ)
115 return failure<LoadedTerrain>(DiagnosticCode::Unsupported,
116 "anisotropic terrain spacing is not supported by SpatialData");
117 auto spatial = procgen::SpatialData::heightfield(heightmap, 0.f, 0.f, spacingX, 1.f);
118 return Result<LoadedTerrain>::success(
119 {terrainRef, std::move(heightmap), std::move(spatial), spacingX, spacingZ,
120 std::move(payload).takeValue().variant});
121}
