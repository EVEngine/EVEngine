32eve::Result<std::unique_ptr<IJointBackend3D>> WorldBackend::newWeldJoint(Body3D& bodyARef, Body3D& bodyBRef,
33 float anchorX, float anchorY, float anchorZ,
34 bool collideConnected) {
35 return checked("physics.world3d.newWeldJoint", [&] {
36 auto* bodyA = &bodyARef;
37 auto* bodyB = &bodyBRef;
38
39 requireDistinctBodies(owner_, bodyA, bodyB, "World3D.newWeldJoint");
40 const float values[] = {anchorX, anchorY, anchorZ};
41 requireFiniteJointParams(values, 3, "World3D.newWeldJoint");
42 const b3Pos anchor{anchorX, anchorY, anchorZ};
43 const b3WorldTransform xfA = b3Body_GetTransform(box3d::Access::bodyId(*bodyA));
44 const b3WorldTransform xfB = b3Body_GetTransform(box3d::Access::bodyId(*bodyB));
45 b3WeldJointDef def = b3DefaultWeldJointDef();
46 def.base.bodyIdA = box3d::Access::bodyId(*bodyA);
47 def.base.bodyIdB = box3d::Access::bodyId(*bodyB);
48 def.base.localFrameA.p = b3Body_GetLocalPoint(box3d::Access::bodyId(*bodyA), anchor);
49 def.base.localFrameB.p = b3Body_GetLocalPoint(box3d::Access::bodyId(*bodyB), anchor);
50 def.base.localFrameA.q = b3InvMulQuat(xfA.q, b3Quat_identity);
51 def.base.localFrameB.q = b3InvMulQuat(xfB.q, b3Quat_identity);
52 def.base.collideConnected = collideConnected;
