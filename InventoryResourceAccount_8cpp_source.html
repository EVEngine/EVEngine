115eve::Result<std::int64_t> InventoryResourceAccount::activeReservationsFor(const ResourceId& resource) const {
116 std::int64_t total = 0;
117 for (const auto& [id, record] : reservations_) {
118 (void)id;
119 if (record.state != ReservationState::Active) continue;
120 for (const auto& item : record.cost.items()) {
121 if (item.resource != resource) continue;
122 if (item.amount.value() > std::numeric_limits<std::int64_t>::max() - total)
123 return failure<std::int64_t>(eve::DiagnosticCode::InvariantViolation,
124 "active inventory reservations overflowed", resource.value());

127 }

131eve::Result<Affordability> InventoryResourceAccount::canAfford(const CostSpec& cost) const {
132 if (accountNonce_.isZero()) return eve::Result<Affordability>::failure(identityUnavailable());
133 auto valid = validateCost(cost);
134 if (!valid) return eve::Result<Affordability>::failure(valid.status());
135
136 Affordability result;
137 result.affordable = true;
138 for (const auto& item : cost.items()) {
139 auto balance = balanceOf(item.resource);
140 if (!balance) return eve::Result<Affordability>::failure(balance.status());
141 auto reserved = activeReservationsFor(item.resource);
142 if (!reserved) return eve::Result<Affordability>::failure(reserved.status());
143 const std::int64_t available = balance.value() >= reserved.value() ? balance.value() - reserved.value() : 0;
144 if (available < item.amount.value()) {
145 result.affordable = false;
146 result.shortfalls.push_back({item.resource, item.amount, Amount(available)});
147 }
148 }
149 return eve::Result<Affordability>::success(std::move(result));
150}

152eve::Result<Reservation> InventoryResourceAccount::reserve(const CostSpec& cost) {
153 if (accountNonce_.isZero()) return eve::Result<Reservation>::failure(identityUnavailable());
154 auto affordability = canAfford(cost);
155 if (!affordability) return eve::Result<Reservation>::failure(affordability.status());
156 const auto& checked = affordability.value();
157 if (!checked.affordable) return eve::Result<Reservation>::failure(insufficient(checked));
158 if (nextReservation_.value() == std::numeric_limits<std::uint64_t>::max())
159 return failure<Reservation>(eve::DiagnosticCode::InvariantViolation,
160 "inventory resource reservation id exhausted", "reservation");
161
162 const ReservationId id = nextReservation_;
163 const auto [it, inserted] = reservations_.emplace(id, ReservationRecord{cost, ReservationState::Active});
164 if (!inserted)
165 return failure<Reservation>(eve::DiagnosticCode::Conflict, "inventory resource reservation id was already used",
166 "reservation");
167 nextReservation_ = ReservationId(id.value() + 1);
168 return eve::Result<Reservation>::success(Reservation{accountNonce_, id, it->second.cost},
169 eve::Status::success(eve::StatusCode::Applied));
170}

172eve::Result<void> InventoryResourceAccount::applyDelta(const CostSpec& cost, bool debit) {
173 auto valid = validateCost(cost);
174 if (!valid) return valid;
175 for (const auto& item : cost.items()) {
176 if (item.amount.value() > std::numeric_limits<int>::max())
177 return failure<void>(eve::DiagnosticCode::Unsupported,
178 "inventory item quantity exceeds the Bag integer range", item.resource.value());
179 }
180
181 const auto originalSlots = bag_.slots();
182 const int originalInstanceCounter = InventorySystem::instanceCounter();
183 const std::size_t originalEventCount = InventorySystem::eventQueue().size();
184 const bool oldHookSuppression = InventorySystem::changeHooksSuppressed();
185 InventorySystem::changeHooksSuppressed() = true;
186
187 auto restore = [&] {
188 bag_.slots() = originalSlots;
189 InventorySystem::instanceCounter() = originalInstanceCounter;
190 InventorySystem::eventQueue().resize(originalEventCount);
191 InventorySystem::changeHooksSuppressed() = oldHookSuppression;
192 };
193
194 try {
195 for (const auto& item : cost.items()) {
196 const int amount = static_cast<int>(item.amount.value());
197 if (debit) {
198 auto balance = balanceOf(item.resource);
199 if (!balance || balance.value() < amount) {
200 restore();
201 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
202 "inventory debit would make item quantity negative", item.resource.value());
203 }
204 if (InventorySystem::removeItem(&bag_, item.resource.value(), amount) != amount) {
205 restore();
206 return failure<void>(eve::DiagnosticCode::Failed, "inventory debit was not applied in full",
207 item.resource.value());
208 }
209 } else {
210 std::string reason;
211 if (!InventorySystem::canAdd(&bag_, item.resource.value(), amount, &reason)) {
212 restore();
213 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
214 "inventory credit was rejected by the Bag policy: " + reason,
215 item.resource.value());
216 }
217 if (InventorySystem::addItem(&bag_, item.resource.value(), amount) != amount) {
218 restore();
219 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
220 "inventory credit was not applied in full", item.resource.value());
221 }
222 }
223 }
224 } catch (const std::exception& exception) {
225 restore();
226 return exceptionFailure<void>(debit ? "debit" : "credit", exception);
227 } catch (...) {
228 restore();
229 return unknownExceptionFailure<void>(debit ? "debit" : "credit");
230 }
231
232 InventorySystem::changeHooksSuppressed() = oldHookSuppression;
233
234 // The state is already committed. Deliver a snapshot of the events after
235 // the mutation, outside the suppressed section and without retaining any
236 // internal container reference across callbacks.
237 const std::vector<InventoryChangeEvent> emitted(
238 InventorySystem::eventQueue().begin() + static_cast<std::ptrdiff_t>(originalEventCount),
239 InventorySystem::eventQueue().end());
240 const auto hooks = InventorySystem::changeHooks();
241 for (const auto& event : emitted) {
242 for (const auto& [name, hook] : hooks) {
243 (void)name;
244 if (!hook) continue;
245 try {
246 hook(event);
247 } catch (...) {
248 // Event hooks are observers, not part of the account commit.
249 }
250 }
251 }
252 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
253}

255eve::Result<Receipt> InventoryResourceAccount::debit(const CostSpec& cost) {

282eve::Result<void> InventoryResourceAccount::activeReservationsAreCovered() const {
283 for (const auto& [id, record] : reservations_) {
284 (void)id;
285 if (record.state != ReservationState::Active) continue;
286 for (const auto& item : record.cost.items()) {
287 auto balance = balanceOf(item.resource);
288 if (!balance) return eve::Result<void>::failure(balance.status());
289 auto reserved = activeReservationsFor(item.resource);
290 if (!reserved) return eve::Result<void>::failure(reserved.status());
291 if (balance.value() < reserved.value())
292 return failure<void>(eve::DiagnosticCode::Conflict,
293 "external Bag mutation invalidated active item reservations",
294 item.resource.value());
295 }
296 }
297 return eve::Result<void>::success();
298}

300eve::Result<Receipt> InventoryResourceAccount::commit(const Reservation& reservation) {
301 if (accountNonce_.isZero()) return eve::Result<Receipt>::failure(identityUnavailable());
302 if (!reservation.isValid())
303 return failure<Receipt>(eve::DiagnosticCode::InvalidArgument, "inventory resource reservation is invalid",
304 "reservation");
305 if (reservation.account != accountNonce_) return eve::Result<Receipt>::failure(foreignReservation());
306 const auto it = reservations_.find(reservation.id);
307 if (it == reservations_.end()) return eve::Result<Receipt>::failure(unknownReservation());
308 if (it->second.cost != reservation.cost)
309 return failure<Receipt>(eve::DiagnosticCode::Conflict,
310 "inventory reservation cost does not match its account record", "reservation.cost");
311 if (it->second.state != ReservationState::Active) return eve::Result<Receipt>::failure(terminalReservation());
312 auto covered = activeReservationsAreCovered();
313 if (!covered) return eve::Result<Receipt>::failure(covered.status());
314 if (nextReceipt_.value() == std::numeric_limits<std::uint64_t>::max())
315 return eve::Result<Receipt>::failure(receiptExhausted());
316
317 Receipt receipt{accountNonce_, nextReceipt_, reservation.id, ReceiptOperation::Debit, it->second.cost};
318 auto applied = applyDelta(it->second.cost, true);
319 if (!applied) return eve::Result<Receipt>::failure(applied.status());
320 it->second.state = ReservationState::Committed;
321 nextReceipt_ = eve::resource::ReceiptId(nextReceipt_.value() + 1);
322 return eve::Result<Receipt>::success(std::move(receipt), eve::Status::success(eve::StatusCode::Applied));
323}
