36void StyleRecipe::compile(graphics::Graphics* gfx) {
37 if (!gfx) throw eve::Exception("StyleRecipe.compile: null graphics");
38 if (instances_.empty()) throw eve::Exception("StyleRecipe.compile: no styles");
39
40 if (graphics_ != gfx) {
41 scratch_ = nullptr;
42 scratchWidth_ = 0;
43 scratchHeight_ = 0;
44 }
45 std::stable_sort(instances_.begin(), instances_.end(),
46 [](StyleInstance* a, StyleInstance* b) { return a->getPriority() < b->getPriority(); });
47 stage_ = instances_.front()->getStage();
48 passes_.clear();
49 for (StyleInstance* instance : instances_) {
50 if (instance->getStage() != stage_)
51 throw eve::Exception("StyleRecipe.compile: cannot mix stages '%s' and '%s'", stage_.c_str(),
52 instance->getStage().c_str());
53 passes_.emplace_back(instance->newPass(gfx));
54 }
55 graphics_ = gfx;
56 compiled_ = true;
57}

59void StyleRecipe::ensureScratch(graphics::Graphics* gfx, graphics::Canvas* dest) {
60 if (passes_.size() < 2) return;
61 const int width = dest->getWidth();
62 const int height = dest->getHeight();
63 if (graphics_ != gfx || !scratch_ || scratchWidth_ != width || scratchHeight_ != height) {
64 scratch_ = gfx->newCanvas(width, height);
65 scratchWidth_ = width;
66 scratchHeight_ = height;
67 graphics_ = gfx;
68 }

71void StyleRecipe::apply(graphics::Graphics* gfx, graphics::Texture* source, graphics::Canvas* dest) {
72 if (!compiled_) throw eve::Exception("StyleRecipe.apply: compile first");
73 if (graphics_ != gfx) throw eve::Exception("StyleRecipe.apply: compiled for another Graphics");
74 if (!source || !dest) throw eve::Exception("StyleRecipe.apply: null source or destination");
75 ensureScratch(gfx, dest);
76
77 StyleChain chain;
78 for (const auto& pass : passes_) chain.add(pass.get());
79 chain.apply(gfx, source, dest, scratch_);
80}
