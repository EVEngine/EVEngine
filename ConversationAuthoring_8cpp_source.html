22const std::vector<Field>& fieldsFor(const std::string& type) {
23 // Authoring / editor field names stay stable; setField maps them onto the
24 // SequenceAsset payload keys used by L1 dialogue vocabulary.
25 static const std::vector<Field> line{{"next", "node"}, {"speaker", "string"}, {"text", "multiline"},
26 {"pool", "asset"}, {"i18nKey", "string"}, {"voice", "asset"},
27 {"expression", "string"}};
28 static const std::vector<Field> branch;
29 static const std::vector<Field> choice;
30 static const std::vector<Field> call{
31 {"target", "asset"}, {"returnNode", "node"}, {"next", "node"}, {"arguments", "json"}};
32 static const std::vector<Field> command{
33 {"target", "string"}, {"expression", "string"}, {"next", "node"}, {"arguments", "json"}};
34 static const std::vector<Field> wait{{"next", "node"}};
35 static const std::vector<Field> end;
36 if (type == "line") return line;
37 if (type == "branch") return branch;
38 if (type == "choice") return choice;
39 if (type == "call") return call;
40 if (type == "command") return command;
41 if (type == "wait") return wait;
42 return end;
43}

146bool ConversationDocument::removeNode(const std::string& nodeId) {
147 const auto found =
148 std::find_if(asset_.nodes.begin(), asset_.nodes.end(), [&](const auto& node) { return node.id == nodeId; });
149 if (found == asset_.nodes.end()) return false;
150 asset_.nodes.erase(found);
151 if (asset_.entry == nodeId) asset_.entry.clear();
152 for (auto& node : asset_.nodes) {
153 if (node.next == nodeId) node.next.clear();
154 if (node.type == "call" && sequencePayloadString(node, "return") == nodeId)
155 node.payload.set("return", eve::Value::string(""));
156 node.routes.erase(std::remove_if(node.routes.begin(), node.routes.end(),
157 [&](const auto& route) { return route.target == nodeId; }),
158 node.routes.end());
159 }
160 return true;
161}

207std::string ConversationDocument::getField(const std::string& nodeId, const std::string& field) const {
208 const auto* node = findNode(nodeId);
209 if (!node) return {};
210 if (field == "next") return node->next;
211 const std::string key = payloadKeyForField(field);
212 const eve::Value* value = node->payload.find(key);
213 if (!value) return {};
214 if (field == "arguments") {
215 auto json = value->toJson();
216 return json.ok() ? std::move(json).takeValue() : std::string{};
217 }
218 return value->isString() ? value->asString() : std::string{};
219}

221bool ConversationDocument::setField(const std::string& nodeId, const std::string& field, const std::string& value) {
222 auto* node = findNode(nodeId);
223 if (!node) return fail("node not found: " + nodeId);
224 if (field == "next") {
225 node->next = value;
226 } else if (field == "arguments") {
227 auto parsed = eve::Value::fromJson(value);
228 if (!parsed.ok()) {
229 failureMessage_ = parsed.status().describe();
230 return false;
231 }
232 eve::Value arguments = std::move(parsed).takeValue();
233 if (!arguments.isObject()) return fail("arguments must be a JSON object");
234 node->payload.set("arguments", std::move(arguments));
235 } else {
236 bool known = false;
237 for (const auto& [name, kind] : fieldsFor(node->type)) {
238 (void)kind;
239 if (field == name) {
240 known = true;
241 break;
242 }
243 }
244 if (!known) return fail("unknown node field: " + field);
245 node->payload.set(payloadKeyForField(field), eve::Value::string(value));
246 }
247 failureMessage_.clear();
248 return true;
249}
