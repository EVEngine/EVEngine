14bool parseKind(const std::string& name, Kind& out) {
15 static constexpr std::array<std::pair<const char*, Kind>, 7> kinds{{
16 {"line", Kind::Line},
17 {"branch", Kind::Branch},
18 {"choice", Kind::Choice},
19 {"call", Kind::Call},
20 {"command", Kind::Command},
21 {"wait", Kind::Wait},
22 {"end", Kind::End},
23 }};
24 for (const auto& [key, value] : kinds) {
25 if (name == key) {
26 out = value;
27 return true;
28 }
29 }
30 return false;
31}

63 case Kind::Call: return call;

157bool ConversationDocument::removeNode(const std::string& nodeId) {
158 const auto it =
159 std::find_if(asset_.nodes.begin(), asset_.nodes.end(), [&](const auto& node) { return node.id == nodeId; });
160 if (it == asset_.nodes.end()) return false;
161 asset_.nodes.erase(it);
162 if (asset_.entry == nodeId) asset_.entry.clear();
163 for (auto& node : asset_.nodes) {
164 if (node.next == nodeId) node.next.clear();
165 if (node.returnNode == nodeId) node.returnNode.clear();
166 node.routes.erase(std::remove_if(node.routes.begin(), node.routes.end(),
167 [&](const auto& route) { return route.second == nodeId; }),
168 node.routes.end());
169 }
170 return true;
171}
