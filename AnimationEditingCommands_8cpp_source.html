10const editing::Value* field(const editing::Value& value, const char* key) {
11 const auto* object = value.getIf<editing::Value::Object>();
12 if (!object) return nullptr;
13 const auto found = object->find(key);
14 return found == object->end() ? nullptr : &found->second;
15}

17bool readNumber(const editing::Value& value, const char* key, double& out) {
18 const editing::Value* entry = field(value, key);
19 if (!entry) return false;
20 if (const auto* real = entry->getIf<double>()) {
21 out = *real;
22 return true;
23 }
24 if (const auto* integer = entry->getIf<std::int64_t>()) {
25 out = static_cast<double>(*integer);
26 return true;
27 }
28 return false;
29}

54editing::Result<void> registerEditingCommands(editing::IEditingCommandRegistry& registry) {
55 auto settings = registerOne(
56 registry, "animation.clip.settings.set.v1", "Set animation clip settings",
57 [](editing::IEditableTarget& target, const editing::CommandRequest& request) {
58 auto* capability = clipTarget(target);
59 double duration = 0, sampleRate = 0;
60 const editing::Value* loopEntry = field(request.payload, "loop");
61 const auto* loop = loopEntry ? loopEntry->getIf<bool>() : nullptr;
62 if (!capability || !readNumber(request.payload, "duration", duration) ||
63 !readNumber(request.payload, "sampleRate", sampleRate) || !loop)
64 return error<editing::CommandPlan>(editing::Status::Rejected, "animation.editing.settings-payload",
65 "Clip settings require duration, sampleRate and loop");
66 auto operation = capability->makeSetSettings(duration, sampleRate, *loop);
67 if (!operation.ok())
68 return error<editing::CommandPlan>(operation.code(), "animation.editing.settings-operation",
69 "Animation clip rejected the settings");
70 editing::CommandPlan plan;
71 plan.operations.push_back(std::move(operation).takeValue());
72 plan.summary = editing::Value::Object{{"duration", duration}, {"sampleRate", sampleRate}, {"loop", *loop}};
73 return eve::editing::applied<editing::CommandPlan>(std::move(plan));
74 });
75 if (!settings.ok()) return settings;
