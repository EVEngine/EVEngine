186Result<RoadNetwork> RoadNetwork::makeCross(float span, int lanes) {
187 if (!std::isfinite(span) || span < 16.f || lanes < 1 || lanes > 4)
188 return Result<RoadNetwork>::failure(
189 Diagnostic::error(DiagnosticCode::InvalidArgument, "span>=16, lanes in [1,4] required", "cross"));
190 RoadNetwork network;
191 const float half = span * 0.5f;
192 RoadStyle style = groundStyle();
193 style.deckThickness = 0.08f;
194 style.sidewalkWidth = 1.2f;
195 style.curbWidth = 0.35f;
196 style.curbHeight = 0.45f;
197 // Push arms outward past the asphalt half-width so sidewalks do not overlap;
198 // the leftover ring is filled by quarter-circle curb returns.
199 const float asphaltHalf = 0.5f * style.laneWidth * static_cast<float>(lanes);
200 const float cornerR = 2.8f;
201 const float jr = asphaltHalf + cornerR;
202 auto nC = network.addNode(0.f, 0.f, 0.f, jr);
203 auto nN = network.addNode(0.f, 0.f, -half, 2.f);
204 auto nS = network.addNode(0.f, 0.f, half, 2.f);
205 auto nW = network.addNode(-half, 0.f, 0.f, 2.f);
206 auto nE = network.addNode(half, 0.f, 0.f, 2.f);
207 for (auto* r : {&nC, &nN, &nS, &nW, &nE}) {
208 if (!r->ok()) return Result<RoadNetwork>::failure(r->status());
209 }
210 // Arms run into the hub; bake trims each end by junctionRadius so strips
211 // stop outside the corner arcs.
212 auto e1 = network.addEdge(nN.value(), nC.value(), {P(0.f, 0.f, -half), P(0.f, 0.f, 0.f)}, lanes, 0, style);
213 auto e2 = network.addEdge(nC.value(), nS.value(), {P(0.f, 0.f, 0.f), P(0.f, 0.f, half)}, lanes, 0, style);
214 auto e3 = network.addEdge(nW.value(), nC.value(), {P(-half, 0.f, 0.f), P(0.f, 0.f, 0.f)}, lanes, 0, style);
215 auto e4 = network.addEdge(nC.value(), nE.value(), {P(0.f, 0.f, 0.f), P(half, 0.f, 0.f)}, lanes, 0, style);
216 for (auto* e : {&e1, &e2, &e3, &e4}) {
217 if (!e->ok()) return Result<RoadNetwork>::failure(e->status());
218 }
219 auto turns = network.connectAllTurns(nC.value());
220 if (!turns.ok()) return Result<RoadNetwork>::failure(turns.status());
221 return Result<RoadNetwork>::success(std::move(network));
222}

263Result<RoadNetwork> RoadNetwork::makeFan(float span, int lanes, std::vector<float> armAnglesDeg,
264 std::vector<bool> intoHub) {
265 if (!std::isfinite(span) || span < 16.f || lanes < 1 || lanes > 4)
266 return Result<RoadNetwork>::failure(
267 Diagnostic::error(DiagnosticCode::InvalidArgument, "span>=16, lanes in [1,4] required", "fan"));
268 if (armAnglesDeg.size() < 2u)
269 return Result<RoadNetwork>::failure(
270 Diagnostic::error(DiagnosticCode::InvalidArgument, "fan needs at least two arm angles", "fan"));
271 if (!intoHub.empty() && intoHub.size() != armAnglesDeg.size())
272 return Result<RoadNetwork>::failure(
273 Diagnostic::error(DiagnosticCode::InvalidArgument, "intoHub size must match armAnglesDeg", "fan"));
274
275 struct ArmSpec {
276 float degrees = 0.f;
277 bool intoHub = true;
278 };
279 std::vector<ArmSpec> specs;
280 specs.reserve(armAnglesDeg.size());
281 for (std::size_t index = 0; index < armAnglesDeg.size(); ++index) {
282 if (!std::isfinite(armAnglesDeg[index]))
283 return Result<RoadNetwork>::failure(
284 Diagnostic::error(DiagnosticCode::InvalidArgument, "arm angles must be finite", "fan"));
285 float degrees = std::fmod(armAnglesDeg[index], 360.f);
286 if (degrees < 0.f) degrees += 360.f;
287 specs.push_back({degrees, intoHub.empty() || intoHub[index]});
288 }
289 std::sort(specs.begin(), specs.end(),
290 [](const ArmSpec& first, const ArmSpec& second) { return first.degrees < second.degrees; });
291 if (intoHub.empty())
292 for (std::size_t index = 0; index < specs.size(); ++index) specs[index].intoHub = index % 2u == 0u;
293 std::vector<ArmSpec> unique;
294 unique.reserve(specs.size());
295 for (const auto& spec : specs)
296 if (unique.empty() || std::fabs(spec.degrees - unique.back().degrees) > 1.f) unique.push_back(spec);
297 if (unique.size() >= 2u && std::fabs(unique.front().degrees + 360.f - unique.back().degrees) <= 1.f)
298 unique.pop_back();
299 if (unique.size() < 2u)
300 return Result<RoadNetwork>::failure(
301 Diagnostic::error(DiagnosticCode::InvalidArgument, "fan needs at least two distinct arm angles", "fan"));
302
303 RoadNetwork network;
304 const float half = span * 0.5f;
305 RoadStyle style = groundStyle();
306 style.deckThickness = 0.08f;
307 style.sidewalkWidth = 1.2f;
308 style.curbWidth = 0.35f;
309 style.curbHeight = 0.45f;
310 const float asphaltHalf = 0.5f * style.laneWidth * static_cast<float>(lanes);
311 auto center = network.addNode(0.f, 0.f, 0.f, asphaltHalf + 2.8f);
312 if (!center.ok()) return Result<RoadNetwork>::failure(center.status());
313 for (const auto& spec : unique) {
314 const float radians = spec.degrees * 0.01745329252f;
315 const float x = half * std::cos(radians), z = half * std::sin(radians);
316 auto leaf = network.addNode(x, 0.f, z, 2.f);
317 if (!leaf.ok()) return Result<RoadNetwork>::failure(leaf.status());
318 auto edge =
319 spec.intoHub
320 ? network.addEdge(leaf.value(), center.value(),
321 {P(x, 0.f, z), P(x * 0.5f, 0.f, z * 0.5f), P(0.f, 0.f, 0.f)}, lanes, 0, style)
322 : network.addEdge(center.value(), leaf.value(),
323 {P(0.f, 0.f, 0.f), P(x * 0.5f, 0.f, z * 0.5f), P(x, 0.f, z)}, lanes, 0, style);
324 if (!edge.ok()) return Result<RoadNetwork>::failure(edge.status());
325 }
326 auto turns = network.connectAllTurns(center.value());
327 if (!turns.ok()) return Result<RoadNetwork>::failure(turns.status());
328 return Result<RoadNetwork>::success(std::move(network));
329}

335Result<RoadNetwork> RoadNetwork::makeRoundabout(float span, int lanes) {
336 if (!std::isfinite(span) || span < 32.f || lanes < 1 || lanes > 3)
337 return Result<RoadNetwork>::failure(
338 Diagnostic::error(DiagnosticCode::InvalidArgument, "span>=32, lanes in [1,3] required", "roundabout"));
339 RoadNetwork network;
340 const float outerRadius = span * 0.5f;
341 const float ringRadius = std::clamp(span * 0.25f, 8.f, 16.f);
342 constexpr float pi = 3.14159265358979323846f;
343 constexpr int ringCount = 8;
344 constexpr float ringStep = 2.f * pi / static_cast<float>(ringCount);
345 std::vector<std::uint32_t> ringNodes;
346 ringNodes.reserve(ringCount);
347 for (int i = 0; i < ringCount; ++i) {
348 const float angle = -0.5f * pi + static_cast<float>(i) * ringStep;
349 const float seamRadius = (i % 2 == 0) ? 1.5f : 0.25f;
350 auto node = network.addNode(std::cos(angle) * ringRadius, 0.f, std::sin(angle) * ringRadius, seamRadius);
351 if (!node.ok()) return Result<RoadNetwork>::failure(node.status());
352 ringNodes.push_back(node.value());
353 }
354
355 RoadStyle ringStyle = groundStyle();
356 ringStyle.laneWidth = 3.25f;
357 ringStyle.curbWidth = 0.3f;
358 ringStyle.sidewalkWidth = 0.75f;
359 ringStyle.trafficPriority = 10;
360 RoadStyle approachStyle = groundStyle();
361 approachStyle.laneWidth = 3.f;
362 approachStyle.trafficPriority = 1;
363 for (int i = 0; i < ringCount; ++i) {
364 const float a0 = -0.5f * pi + static_cast<float>(i) * ringStep;
365 const float a3 = a0 + ringStep;
366 const float handle = 2.f * ringRadius * std::sin(ringStep * 0.5f) / 3.f;
367 const RoadControlPoint p0 = P(std::cos(a0) * ringRadius, 0.f, std::sin(a0) * ringRadius);
368 const RoadControlPoint p3 = P(std::cos(a3) * ringRadius, 0.f, std::sin(a3) * ringRadius);
369 const RoadControlPoint p1 = P(p0.x - std::sin(a0) * handle, 0.f, p0.z + std::cos(a0) * handle);
370 const RoadControlPoint p2 = P(p3.x + std::sin(a3) * handle, 0.f, p3.z - std::cos(a3) * handle);
371 auto edge = network.addEdge(ringNodes[static_cast<std::size_t>(i)],
372 ringNodes[static_cast<std::size_t>((i + 1) % ringCount)], {p0, p1, p2, p3}, lanes,
373 0, ringStyle);
374 if (!edge.ok()) return Result<RoadNetwork>::failure(edge.status());
375 }
376 for (int i = 0; i < ringCount; i += 2) {
377 const float angle = -0.5f * pi + static_cast<float>(i) * ringStep;
378 auto outer = network.addNode(std::cos(angle) * outerRadius, 0.f, std::sin(angle) * outerRadius, 2.f);
379 if (!outer.ok()) return Result<RoadNetwork>::failure(outer.status());
380 auto approach = network.addEdge(outer.value(), ringNodes[static_cast<std::size_t>(i)], {{}, {}}, lanes, lanes,
381 approachStyle);
382 if (!approach.ok()) return Result<RoadNetwork>::failure(approach.status());
383 }
384 for (int i = 0; i < ringCount; i += 2) {
385 const auto node = ringNodes[static_cast<std::size_t>(i)];
386 auto control = network.setNodeJunctionControl(node, RoadJunctionControl::Yield);
387 if (!control.ok()) return Result<RoadNetwork>::failure(control.status());
388 auto connected = network.connectAllTurns(node);
389 if (!connected.ok()) return Result<RoadNetwork>::failure(connected.status());
390 }
391 return Result<RoadNetwork>::success(std::move(network));
392}

395 std::uint32_t seed) {
396 if (scene == "straight") return makeStraight(span, lanes);
397 if (scene == "curve") return makeCurve(std::max(8.f, span * 0.5f), lanes);
398 if (scene == "bridge") return makeBridge(span, bridgeHeight, lanes);
399 if (scene == "cross") return makeCross(span, lanes);
400 if (scene == "tee" || scene == "t") return makeTee(span, lanes);
401 if (scene == "y") return makeY(span, lanes);
402 if (scene == "fork") return makeFork(span, lanes);
403 if (scene == "skew") return makeSkew(span, lanes);
404 if (scene == "t-junction" || scene == "y-junction") return makeThreeArmScene(scene, lanes);
405 if (scene == "sloped-t") return makeSlopedJunctionScene(lanes, false);
406 if (scene == "curve-uphill") return makeSlopedJunctionScene(lanes, true);
407 if (scene == "tight-turn") return makeTightTurnScene(lanes);
408 if (scene == "roundabout") return makeRoundabout(span, std::min(lanes, 3));
409 if (scene == "interchange" || scene.empty()) return makeInterchange(span, bridgeHeight, lanes, seed);
410 return Result<RoadNetwork>::failure(
411 Diagnostic::error(DiagnosticCode::InvalidArgument,
412 "scene must be "
413 "straight|curve|bridge|cross|tee|t|y|fork|skew|t-junction|y-junction|sloped-t|curve-uphill|"
414 "tight-turn|roundabout|interchange",
415 "scene"));
416}

418Result<RoadNetwork> RoadNetwork::makeInterchange(float span, float bridgeHeight, int lanes, std::uint32_t seed) {
419 if (!std::isfinite(span) || span < 16.f || !std::isfinite(bridgeHeight) || bridgeHeight < 1.f || lanes < 1 ||
420 lanes > 4)
421 return Result<RoadNetwork>::failure(Diagnostic::error(
422 DiagnosticCode::InvalidArgument, "span>=16, bridgeHeight>=1, lanes in [1,4] required", "interchange"));
423
424 RoadNetwork network;
425 const float half = span * 0.5f;
426 const float h = bridgeHeight;
427 // Tiny seed wobble on elevated mid-handles only — ground cross stays axis-aligned
428 // so bakeJunction keeps the same outward-center curb returns as makeCross.
429 const float wobble = 0.25f * static_cast<float>(static_cast<int>(seed % 7u) - 3);
430
431 RoadStyle ground = groundStyle();
432 ground.deckThickness = 0.08f;
433 ground.sidewalkWidth = 1.2f;
434 ground.curbWidth = 0.35f;
435 ground.curbHeight = 0.45f;
436 const float asphaltHalf = 0.5f * ground.laneWidth * static_cast<float>(lanes);
437 const float cornerR = 2.8f;
438 const float jr = asphaltHalf + cornerR;
439
440 RoadStyle bridge = bridgeStyle();
441 bridge.pierSpacing = 14.0f;
442 bridge.sidewalkWidth = 1.0f;
443 bridge.curbWidth = 0.38f;
444 bridge.curbHeight = 0.50f;
445 RoadStyle ramp = bridge;
446 ramp.sidewalkWidth = 0.5f;
447 ramp.pierSpacing = 10.f;
448
449 auto nGround = network.addNode(0.f, 0.f, 0.f, jr);
450 auto nN = network.addNode(0.f, 0.f, -half, 2.f);
451 auto nS = network.addNode(0.f, 0.f, half, 2.f);
452 auto nW = network.addNode(-half, 0.f, 0.f, 2.f);
453 auto nE = network.addNode(half, 0.f, 0.f, 2.f);
454
455 // The elevated diagonal crosses the ground graph in XZ but has its own node
456 // at bridge height. Equal planar coordinates never imply connectivity.
457 auto nBridgeNW = network.addNode(-half, h, -half, 2.0f);
458 auto nElev = network.addNode(0.f, h, 0.f, 2.0f);
459 auto nBridgeSE = network.addNode(half, h, half, 2.0f);
460 for (auto* res : {&nGround, &nN, &nS, &nW, &nE, &nBridgeNW, &nElev, &nBridgeSE}) {
461 if (!res->ok()) return Result<RoadNetwork>::failure(res->status());
462 }
463
464 auto add = [&](std::uint32_t a, std::uint32_t b, std::vector<RoadControlPoint> pts, const RoadStyle& style) {
465 return network.addEdge(a, b, std::move(pts), lanes, 0, style);
466 };
467
468 // Ground cross — same fillet construction as makeCross.
469 const auto g = nGround.value();
470 auto e1 = add(nN.value(), g, {P(0.f, 0.f, -half), P(0.f, 0.f, 0.f)}, ground);
471 auto e2 = add(g, nS.value(), {P(0.f, 0.f, 0.f), P(0.f, 0.f, half)}, ground);
472 auto e3 = add(nW.value(), g, {P(-half, 0.f, 0.f), P(0.f, 0.f, 0.f)}, ground);
473 auto e4 = add(g, nE.value(), {P(0.f, 0.f, 0.f), P(half, 0.f, 0.f)}, ground);
474
475 // The central split is a smooth continuation and therefore creates no
476 // junction apron. Seed wobble remains bounded and tangent-continuous.
477 auto e5 = add(nBridgeNW.value(), nElev.value(),
478 {P(-half, h, -half), P(-half * 0.45f + wobble, h, -half * 0.45f), P(0.f, h, 0.f)}, bridge);
479 auto e6 = add(nElev.value(), nBridgeSE.value(),
480 {P(0.f, h, 0.f), P(half * 0.45f, h, half * 0.45f + wobble), P(half, h, half)}, bridge);
481
482 // Four outer ramps connect both ground approaches to the elevated diagonal
483 // without inventing planar connectivity at the central grade separation.
484 auto r1 = network.addEdge(
485 nN.value(), nBridgeNW.value(),
486 {P(0.f, 0.f, -half), P(-half * 0.35f, h * 0.2f, -half), P(-half * 0.75f, h * 0.75f, -half), P(-half, h, -half)},
487 1, 1, ramp);
488 auto r2 = network.addEdge(
489 nW.value(), nBridgeNW.value(),
490 {P(-half, 0.f, 0.f), P(-half, h * 0.2f, -half * 0.35f), P(-half, h * 0.75f, -half * 0.75f), P(-half, h, -half)},
491 1, 1, ramp);
492 auto r3 = network.addEdge(
493 nBridgeSE.value(), nS.value(),
494 {P(half, h, half), P(half * 0.75f, h * 0.75f, half), P(half * 0.35f, h * 0.2f, half), P(0.f, 0.f, half)}, 1, 1,
495 ramp);
496 auto r4 = network.addEdge(
497 nBridgeSE.value(), nE.value(),
498 {P(half, h, half), P(half, h * 0.75f, half * 0.75f), P(half, h * 0.2f, half * 0.35f), P(half, 0.f, 0.f)}, 1, 1,
499 ramp);
500
501 for (auto* edge : {&e1, &e2, &e3, &e4, &e5, &e6, &r1, &r2, &r3, &r4}) {
502 if (!edge->ok()) return Result<RoadNetwork>::failure(edge->status());
503 }
504
505 // Route links exist only at the ground intersection. The elevated road is
506 // a separate graph despite sharing the same XZ crossing coordinate.
507 auto turns = network.connectAllTurns(g);
508 if (!turns.ok()) return Result<RoadNetwork>::failure(turns.status());
509 for (const auto node : {nN.value(), nW.value(), nBridgeNW.value(), nBridgeSE.value(), nS.value(), nE.value()}) {
510 turns = network.connectAllTurns(node);
511 if (!turns.ok()) return Result<RoadNetwork>::failure(turns.status());
512 }
513
514 return Result<RoadNetwork>::success(std::move(network));
515}
