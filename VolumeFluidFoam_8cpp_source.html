26glm::vec3 jitter(uint32_t& state, float radius) {
27 // Fixed three draws, uniformly distributed ball; no unbounded rejection loop.
28 const float z = 2.f * unit(state) - 1.f;
29 const float angle = 6.28318530718f * unit(state);
30 const float r = radius * std::cbrt(unit(state));
31 const float xy = std::sqrt(std::max(0.f, 1.f - z * z));
32 return r * glm::vec3(xy * std::cos(angle), xy * std::sin(angle), z);

50Result<unsigned> VolumeFluidFoam::advanceFiltered(const VolumeFluid& source, VolumeFluidDiffuse& pool, float seconds,
51 bool filterActor, unsigned actorGroup, float alpha) {
52 if (!range(seconds, 0.f, 1.f / 30.f) || !range(alpha, 0.f, 1.f))
53 return Result<unsigned>::failure(invalid("Invalid foam timestep or interpolation alpha"));
54 const double credit = credit_ + double(seconds) * settings_.rate;
55 const auto whole = unsigned(std::floor(credit));
56 const double remainder = credit - whole;
57 const unsigned budget = std::min({whole, settings_.maxPerStep, pool.availableCapacity()});
58 if (budget == 0) {
59 credit_ = remainder;
60 return Result<unsigned>::success(0);
61 }
62 if (source.particleCount() > 65536)
63 return Result<unsigned>::failure(invalid("Foam source exceeds 65536 particles"));
64 const auto particles = source.particleView();
65 positionScratch_.clear();
66 positionScratch_.reserve(particles.size());
67 for (size_t i = 0; i < particles.size(); ++i) {
68 const auto& p = particles[i];
69 if ((!filterActor || p.actorGroup == actorGroup) &&
70 (p.material.phase == VolumeFluidPhase::Liquid || p.material.phase == VolumeFluidPhase::Gas))
71 positionScratch_.push_back(source.interpolatedPositionUnchecked(i, alpha));
72 }
73 auto sampled = source.sampleFieldInto(positionScratch_, sampleScratch_);
74 if (!sampled) return Result<unsigned>::failure(sampled.status());
75 size_t eligibleCount = 0;
76 for (size_t i = 0; i < positionScratch_.size(); ++i) {
77 const auto& field = sampleScratch_[i];
78 const float curl2 = glm::dot(field.vorticity, field.vorticity);
79 if (!std::isfinite(curl2) || !std::isfinite(field.density))
80 return Result<unsigned>::failure(invalid("Invalid foam field sample"));
81 if (curl2 > settings_.vorticityThreshold * settings_.vorticityThreshold &&
82 field.density < settings_.densityThreshold)
83 ++eligibleCount;
84 }
85 const unsigned count = std::min(budget, unsigned(eligibleCount));
86 auto rng = randomState_;
87 batchScratch_.clear();
88 batchScratch_.reserve(count);
89 if (count != 0) {
90 const unsigned stride = unsigned(eligibleCount) / count;
91 const unsigned start = next(rng) % stride;
92 unsigned eligibleOrdinal = 0;
93 unsigned nextOrdinal = start;
94 for (size_t at = 0; at < positionScratch_.size() && batchScratch_.size() < count; ++at) {
95 const auto& field = sampleScratch_[at];
96 const float curl2 = glm::dot(field.vorticity, field.vorticity);
97 if (!(curl2 > settings_.vorticityThreshold * settings_.vorticityThreshold &&
98 field.density < settings_.densityThreshold))
99 continue;
100 if (eligibleOrdinal++ != nextOrdinal) continue;
101 batchScratch_.push_back(
102 {positionScratch_[at] + jitter(rng, settings_.randomness), field.velocity, settings_.lifetime});
103 nextOrdinal += stride;
104 }
105 }
106 auto admitted = pool.emit(batchScratch_);
107 if (!admitted) return Result<unsigned>::failure(admitted.status());
108 credit_ = remainder;
109 randomState_ = rng;
110 return Result<unsigned>::success(count);
111}
