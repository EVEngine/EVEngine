39// 环检测：图的边是 id -> requires[i]；未在表中的前置 id 不算环。自依赖是环。
40bool idOnCycle(const std::unordered_map<std::string, QuestDefinition> &table, const std::string &start) {
41 std::unordered_set<std::string> visiting;
42 std::unordered_set<std::string> visited;
43 std::function<bool(const std::string &)> dfs = [&](const std::string &id) -> bool {
44 if (visiting.count(id)) return true;
45 if (visited.count(id)) return false;
46 auto it = table.find(id);
47 if (it == table.end()) return false;
48 visiting.insert(id);
49 for (const auto &req : it->second.requiresIds) {
50 if (dfs(req)) return true;
51 }
52 visiting.erase(id);
53 visited.insert(id);
54 return false;
55 };
56 return dfs(start);
57}

79// 返回 id 为空的定义表示该 JSON 对象被拒绝（缺 id / 缺目标 id / 目标 id 重复）。
80QuestDefinition parseQuestObject(Value o) {
81 QuestDefinition def;
82 if (!o.isObject()) return def;
83 def.id = o.getString("id");
84 def.startPolicy = normalizeStart(o.getString("startPolicy", "manual"));
85 def.completePolicy = normalizeComplete(o.getString("completePolicy", "auto"));
86 def.requiresIds = o.getStringArray("requires");
87 def.tags = o.getStringArray("tags");
88 def.extra = o.getStringMap("extra");
89
90 const Value objectives = o.get("objectives");
91 std::unordered_set<std::string> seen;
92 bool invalid = false;
93 for (size_t i = 0; i < objectives.size(); ++i) {
94 QuestObjective obj = parseObjective(objectives.at(i));
95 if (obj.id.empty() || !seen.insert(obj.id).second) {
96 invalid = true;
97 break;
98 }
99 def.objectives.push_back(std::move(obj));
100 }
101 if (invalid) {
102 def.id.clear(); // 拒绝整条定义
103 return def;
104 }
105
106 const Value rewards = o.get("rewards");
107 for (size_t i = 0; i < rewards.size(); ++i) {
108 RewardSpec reward = parseReward(rewards.at(i));
109 def.rewards.push_back(std::move(reward));
110 }
111 return def;
112}

140void QuestRegistry::registerQuest(const QuestDefinition &def) {
141 if (def.id.empty()) return;
142 auto &t = questTable();
143 auto old = t.find(def.id);
144 auto saved = old != t.end() ? old->second : QuestDefinition{};
145 bool hadOld = old != t.end();
146
147 QuestDefinition normalized = def;
148 normalized.startPolicy = normalizeStart(def.startPolicy);
149 normalized.completePolicy = normalizeComplete(def.completePolicy);
150 t[def.id] = normalized;
151 if (idOnCycle(t, def.id)) {
152 if (hadOld) {
153 t[def.id] = saved;
154 } else {
155 t.erase(def.id);
156 }
157 }
158}

183int QuestRegistry::loadFromJson(const std::string &json, std::string *error) {
184 std::string err;
185 const eve::json::Document doc = eve::json::Document::parse(json, &err);
186 if (!doc.valid()) {
187 if (error) *error = err.empty() ? "invalid json" : err;
188 return 0;
189 }
190
191 const Value root = doc.root();
192 std::vector<Value> objects;
193 if (root.isArray()) {
194 for (size_t i = 0; i < root.size(); ++i) objects.push_back(root.at(i));
195 } else if (root.isObject()) {
196 objects.push_back(root);
197 }
198
199 auto &t = questTable();
200 std::vector<QuestDefinition> parsed;
201 std::vector<std::string> candidateIds;
202 for (const auto &o : objects) {
203 QuestDefinition def = parseQuestObject(o);
204 if (def.id.empty()) continue;
205 parsed.push_back(def);
206 candidateIds.push_back(def.id);
207 }
208
209 // proposed = 当前表 + 本批候选；对本批候选在 proposed 上做环检测。
210 auto proposed = t;
211 for (const auto &def : parsed) proposed[def.id] = def;
212 std::unordered_set<std::string> rejected;
213 for (const auto &id : candidateIds) {
214 if (idOnCycle(proposed, id)) rejected.insert(id);
215 }
216
217 int n = 0;
218 for (const auto &def : parsed) {
219 if (rejected.count(def.id)) continue;
220 t[def.id] = def;
221 ++n;
222 }
223 return n;
224}

226eve::Result<int> QuestRegistry::replaceFromJsonStrict(const std::string &json) {
227 std::string parseError;
228 const eve::json::Document doc = eve::json::Document::parse(json, &parseError);
229 if (!doc.valid()) return strictFailure(parseError.empty() ? "invalid JSON" : parseError, "$");
230
231 const Value root = doc.root();
232 if (!root.isArray() && !root.isObject())
233 return strictFailure("quest catalogue must be an object or array", "$");
234
235 std::vector<Value> objects;
236 if (root.isArray()) {
237 if (root.size() == 0) return strictFailure("quest catalogue must not be empty", "$");
238 for (size_t i = 0; i < root.size(); ++i) objects.push_back(root.at(i));
239 } else {
240 objects.push_back(root);
241 }
242
243 std::unordered_map<std::string, QuestDefinition> proposed;
244 for (size_t questIndex = 0; questIndex < objects.size(); ++questIndex) {
245 const Value object = objects[questIndex];
246 const std::string base = "$[" + std::to_string(questIndex) + "]";
247 if (!object.isObject()) return strictFailure("quest entry must be an object", base);
248
249 const Value idValue = object.get("id");
250 if (!idValue.isString() || !validContentId(idValue.asString()))
251 return strictFailure("quest id must be a non-empty stable id of at most 256 bytes", base + ".id");
252
253 QuestDefinition definition;
254 definition.id = idValue.asString();
255 if (proposed.count(definition.id) != 0)
256 return strictFailure("duplicate quest id", base + ".id");
257
258 const Value startPolicy = object.get("startPolicy");
259 if (startPolicy && (!startPolicy.isString() ||
260 (startPolicy.asString() != "manual" && startPolicy.asString() != "auto")))
261 return strictFailure("startPolicy must be 'manual' or 'auto'", base + ".startPolicy");
262 definition.startPolicy = startPolicy ? startPolicy.asString() : "manual";
263
264 const Value completePolicy = object.get("completePolicy");
265 if (completePolicy && (!completePolicy.isString() ||
266 (completePolicy.asString() != "auto" && completePolicy.asString() != "claim")))
267 return strictFailure("completePolicy must be 'auto' or 'claim'", base + ".completePolicy");
268 definition.completePolicy = completePolicy ? completePolicy.asString() : "auto";
269
270 const Value prerequisites = object.get("requires");
271 if (prerequisites && !stringArrayIsStrict(prerequisites))
272 return strictFailure("requires must be an array of quest ids", base + ".requires");
273 std::unordered_set<std::string> seenRequirements;
274 definition.requiresIds = prerequisites ? prerequisites.toStringArray() : std::vector<std::string>{};
275 for (size_t i = 0; i < definition.requiresIds.size(); ++i) {
276 const auto &required = definition.requiresIds[i];
277 if (!validContentId(required) || !seenRequirements.insert(required).second)
278 return strictFailure("requires contains an invalid or duplicate quest id",
279 base + ".requires[" + std::to_string(i) + "]");
280 }
281
282 const Value objectives = object.get("objectives");
283 if (objectives && !objectives.isArray())
284 return strictFailure("objectives must be an array", base + ".objectives");
285 std::unordered_set<std::string> seenObjectives;
286 for (size_t i = 0; i < objectives.size(); ++i) {
287 const Value item = objectives.at(i);
288 const auto path = base + ".objectives[" + std::to_string(i) + "]";
289 if (!item.isObject()) return strictFailure("objective must be an object", path);
290 const Value objectiveId = item.get("id");
291 const Value topic = item.get("topic");
292 if (!objectiveId.isString() || !validContentId(objectiveId.asString()) ||
293 !seenObjectives.insert(objectiveId.asString()).second)
294 return strictFailure("objective id must be unique and stable", path + ".id");
295 if (!topic.isString() || !validContentId(topic.asString()))
296 return strictFailure("objective topic must be a stable id", path + ".topic");
297 const Value target = item.get("target");
298 if (target && (!target.isString() || (!target.asString().empty() && !validContentId(target.asString()))))
299 return strictFailure("objective target must be empty or a stable id", path + ".target");
300 const Value count = item.get("count");
301 if (count && (!count.isInt64() || count.asInt64() <= 0 || count.asInt64() > INT_MAX))
302 return strictFailure("objective count must be a positive integer", path + ".count");
303 definition.objectives.push_back(
304 {objectiveId.asString(), topic.asString(), target ? target.asString() : std::string{},
305 count ? int(count.asInt64()) : 1});
306 }
307
308 const Value rewards = object.get("rewards");
309 if (rewards && !rewards.isArray()) return strictFailure("rewards must be an array", base + ".rewards");
310 for (size_t i = 0; i < rewards.size(); ++i) {
311 const Value item = rewards.at(i);
312 const auto path = base + ".rewards[" + std::to_string(i) + "]";
313 if (!item.isObject()) return strictFailure("reward must be an object", path);
314 const Value type = item.get("type");
315 const Value id = item.get("id");
316 const Value amount = item.get("amount");
317 if (!type.isString() || !validContentId(type.asString()))
318 return strictFailure("reward type must be a stable id", path + ".type");
319 if (!id.isString() || !validContentId(id.asString()))
320 return strictFailure("reward id must be a stable id", path + ".id");
321 if (!amount.isNumber() || !std::isfinite(amount.asDouble()) || amount.asDouble() <= 0.0)
322 return strictFailure("reward amount must be finite and positive", path + ".amount");
323 definition.rewards.push_back({type.asString(), id.asString(), amount.asDouble()});
324 }
325
326 const Value tags = object.get("tags");
327 if (tags && !stringArrayIsStrict(tags))
328 return strictFailure("tags must be an array of strings", base + ".tags");
329 definition.tags = tags ? tags.toStringArray() : std::vector<std::string>{};
330 for (size_t i = 0; i < definition.tags.size(); ++i)
331 if (!validContentId(definition.tags[i]))
332 return strictFailure("tag must be a stable id", base + ".tags[" + std::to_string(i) + "]");
333
334 const Value extra = object.get("extra");
335 if (extra && !stringMapIsStrict(extra))
336 return strictFailure("extra must be an object containing only string values", base + ".extra");
337 if (extra) {
338 for (const auto &key : extra.keys()) definition.extra[key] = extra.get(key.c_str()).asString();
339 }
340 proposed.emplace(definition.id, std::move(definition));
341 }
342
343 for (const auto &[id, definition] : proposed) {
344 for (const auto &required : definition.requiresIds)
345 if (proposed.count(required) == 0)
346 return strictFailure("required quest is missing from replacement catalogue", id + ".requires");
347 if (idOnCycle(proposed, id)) return strictFailure("quest prerequisite cycle detected", id + ".requires");
348 }
349
350 questTable() = std::move(proposed);
351 return eve::Result<int>::success(int(questTable().size()));
352}
