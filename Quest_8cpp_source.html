42// 环检测：图的边是 id -> requires[i]；未在表中的前置 id 不算环。自依赖是环。
43bool idOnCycle(const std::unordered_map<std::string, QuestDefinition> &table, const std::string &start) {
44 std::unordered_set<std::string> visiting;
45 std::unordered_set<std::string> visited;
46 std::function<bool(const std::string &)> dfs = [&](const std::string &id) -> bool {
47 if (visiting.count(id)) return true;
48 if (visited.count(id)) return false;
49 auto it = table.find(id);
50 if (it == table.end()) return false;
51 visiting.insert(id);
52 for (const auto &req : it->second.requiresIds) {
53 if (dfs(req)) return true;
54 }
55 visiting.erase(id);
56 visited.insert(id);
57 return false;
58 };
59 return dfs(start);
60}

82// 返回 id 为空的定义表示该 JSON 对象被拒绝（缺 id / 缺目标 id / 目标 id 重复）。
83QuestDefinition parseQuestObject(Value o) {
84 QuestDefinition def;
85 if (!o.isObject()) return def;
86 def.id = o.getString("id");
87 def.startPolicy = normalizeStart(o.getString("startPolicy", "manual"));
88 def.completePolicy = normalizeComplete(o.getString("completePolicy", "auto"));
89 def.requiresIds = o.getStringArray("requires");
90 def.tags = o.getStringArray("tags");
91 def.extra = o.getStringMap("extra");
92
93 const Value objectives = o.get("objectives");
94 std::unordered_set<std::string> seen;
95 bool invalid = false;
96 for (size_t i = 0; i < objectives.size(); ++i) {
97 QuestObjective obj = parseObjective(objectives.at(i));
98 if (obj.id.empty() || !seen.insert(obj.id).second) {
99 invalid = true;
100 break;
101 }
102 def.objectives.push_back(std::move(obj));
103 }
104 if (invalid) {
105 def.id.clear(); // 拒绝整条定义
106 return def;
107 }
108
109 const Value rewards = o.get("rewards");
110 for (size_t i = 0; i < rewards.size(); ++i) {
111 RewardSpec reward = parseReward(rewards.at(i));
112 def.rewards.push_back(std::move(reward));
113 }
114 return def;
115}

143void QuestRegistry::registerQuest(const QuestDefinition &def) {
144 if (def.id.empty()) return;
145 auto &t = questTable();
146 auto old = t.find(def.id);
147 auto saved = old != t.end() ? old->second : QuestDefinition{};
148 bool hadOld = old != t.end();
149
150 QuestDefinition normalized = def;
151 normalized.startPolicy = normalizeStart(def.startPolicy);
152 normalized.completePolicy = normalizeComplete(def.completePolicy);
153 t[def.id] = normalized;
154 if (idOnCycle(t, def.id)) {
155 if (hadOld) {
156 t[def.id] = saved;
157 } else {
158 t.erase(def.id);
159 }
160 }
161}

186int QuestRegistry::loadFromJson(const std::string &json, std::string *error) {
187 std::string err;
188 const eve::json::Document doc = eve::json::Document::parse(json, &err);
189 if (!doc.valid()) {
190 if (error) *error = err.empty() ? "invalid json" : err;
191 return 0;
192 }
193
194 const Value root = doc.root();
195 std::vector<Value> objects;
196 if (root.isArray()) {
197 for (size_t i = 0; i < root.size(); ++i) objects.push_back(root.at(i));
198 } else if (root.isObject()) {
199 objects.push_back(root);
200 }
201
202 auto &t = questTable();
203 std::vector<QuestDefinition> parsed;
204 std::vector<std::string> candidateIds;
205 for (const auto &o : objects) {
206 QuestDefinition def = parseQuestObject(o);
207 if (def.id.empty()) continue;
208 parsed.push_back(def);
209 candidateIds.push_back(def.id);
210 }
211
212 // proposed = 当前表 + 本批候选；对本批候选在 proposed 上做环检测。
213 auto proposed = t;
214 for (const auto &def : parsed) proposed[def.id] = def;
215 std::unordered_set<std::string> rejected;
216 for (const auto &id : candidateIds) {
217 if (idOnCycle(proposed, id)) rejected.insert(id);
218 }
219
220 int n = 0;
221 for (const auto &def : parsed) {
222 if (rejected.count(def.id)) continue;
223 t[def.id] = def;
224 ++n;
225 }
226 return n;
227}

229eve::Result<int> QuestRegistry::replaceFromJsonStrict(const std::string &json) {
230 std::string parseError;
231 const eve::json::Document doc = eve::json::Document::parse(json, &parseError);
232 if (!doc.valid()) return strictFailure(parseError.empty() ? "invalid JSON" : parseError, "$");
233
234 const Value root = doc.root();
235 if (!root.isArray() && !root.isObject()) return strictFailure("quest catalogue must be an object or array", "$");
236
237 std::vector<Value> objects;
238 if (root.isArray()) {
239 if (root.size() == 0) return strictFailure("quest catalogue must not be empty", "$");
240 for (size_t i = 0; i < root.size(); ++i) objects.push_back(root.at(i));
241 } else {
242 objects.push_back(root);
243 }
244
245 std::unordered_map<std::string, QuestDefinition> proposed;
246 for (size_t questIndex = 0; questIndex < objects.size(); ++questIndex) {
247 const Value object = objects[questIndex];
248 const std::string base = "$[" + std::to_string(questIndex) + "]";
249 if (!object.isObject()) return strictFailure("quest entry must be an object", base);
250
251 const Value idValue = object.get("id");
252 if (!idValue.isString() || !validContentId(idValue.asString()))
253 return strictFailure("quest id must be a non-empty stable id of at most 256 bytes", base + ".id");
254
255 QuestDefinition definition;
256 definition.id = idValue.asString();
257 if (proposed.count(definition.id) != 0) return strictFailure("duplicate quest id", base + ".id");
258
259 const Value startPolicy = object.get("startPolicy");
260 if (startPolicy &&
261 (!startPolicy.isString() || (startPolicy.asString() != "manual" && startPolicy.asString() != "auto")))
262 return strictFailure("startPolicy must be 'manual' or 'auto'", base + ".startPolicy");
263 definition.startPolicy = startPolicy ? startPolicy.asString() : "manual";
264
265 const Value completePolicy = object.get("completePolicy");
266 if (completePolicy && (!completePolicy.isString() ||
267 (completePolicy.asString() != "auto" && completePolicy.asString() != "claim")))
268 return strictFailure("completePolicy must be 'auto' or 'claim'", base + ".completePolicy");
269 definition.completePolicy = completePolicy ? completePolicy.asString() : "auto";
270
271 const Value prerequisites = object.get("requires");
272 if (prerequisites && !stringArrayIsStrict(prerequisites))
273 return strictFailure("requires must be an array of quest ids", base + ".requires");
274 std::unordered_set<std::string> seenRequirements;
275 definition.requiresIds = prerequisites ? prerequisites.toStringArray() : std::vector<std::string>{};
276 for (size_t i = 0; i < definition.requiresIds.size(); ++i) {
277 const auto &required = definition.requiresIds[i];
278 if (!validContentId(required) || !seenRequirements.insert(required).second)
279 return strictFailure("requires contains an invalid or duplicate quest id",
280 base + ".requires[" + std::to_string(i) + "]");
281 }
282
283 const Value objectives = object.get("objectives");
284 if (objectives && !objectives.isArray())
285 return strictFailure("objectives must be an array", base + ".objectives");
286 std::unordered_set<std::string> seenObjectives;
287 for (size_t i = 0; i < objectives.size(); ++i) {
288 const Value item = objectives.at(i);
289 const auto path = base + ".objectives[" + std::to_string(i) + "]";
290 if (!item.isObject()) return strictFailure("objective must be an object", path);
291 const Value objectiveId = item.get("id");
292 const Value topic = item.get("topic");
293 if (!objectiveId.isString() || !validContentId(objectiveId.asString()) ||
294 !seenObjectives.insert(objectiveId.asString()).second)
295 return strictFailure("objective id must be unique and stable", path + ".id");
296 if (!topic.isString() || !validContentId(topic.asString()))
297 return strictFailure("objective topic must be a stable id", path + ".topic");
298 const Value target = item.get("target");
299 if (target && (!target.isString() || (!target.asString().empty() && !validContentId(target.asString()))))
300 return strictFailure("objective target must be empty or a stable id", path + ".target");
301 const Value count = item.get("count");
302 if (count && (!count.isInt64() || count.asInt64() <= 0 || count.asInt64() > INT_MAX))
303 return strictFailure("objective count must be a positive integer", path + ".count");
304 definition.objectives.push_back({objectiveId.asString(), topic.asString(),
305 target ? target.asString() : std::string{},
306 count ? int(count.asInt64()) : 1});
307 }
308
309 const Value rewards = object.get("rewards");
310 if (rewards && !rewards.isArray()) return strictFailure("rewards must be an array", base + ".rewards");
311 for (size_t i = 0; i < rewards.size(); ++i) {
312 const Value item = rewards.at(i);
313 const auto path = base + ".rewards[" + std::to_string(i) + "]";
314 if (!item.isObject()) return strictFailure("reward must be an object", path);
315 const Value type = item.get("type");
316 const Value id = item.get("id");
317 const Value amount = item.get("amount");
318 if (!type.isString() || !validContentId(type.asString()))
319 return strictFailure("reward type must be a stable id", path + ".type");
320 if (!id.isString() || !validContentId(id.asString()))
321 return strictFailure("reward id must be a stable id", path + ".id");
322 if (!amount.isNumber() || !std::isfinite(amount.asDouble()) || amount.asDouble() <= 0.0)
323 return strictFailure("reward amount must be finite and positive", path + ".amount");
324 definition.rewards.push_back({type.asString(), id.asString(), amount.asDouble()});
325 }
326
327 const Value tags = object.get("tags");
328 if (tags && !stringArrayIsStrict(tags))
329 return strictFailure("tags must be an array of strings", base + ".tags");
330 definition.tags = tags ? tags.toStringArray() : std::vector<std::string>{};
331 for (size_t i = 0; i < definition.tags.size(); ++i)
332 if (!validContentId(definition.tags[i]))
333 return strictFailure("tag must be a stable id", base + ".tags[" + std::to_string(i) + "]");
334
335 const Value extra = object.get("extra");
336 if (extra && !stringMapIsStrict(extra))
337 return strictFailure("extra must be an object containing only string values", base + ".extra");
338 if (extra) {
339 for (const auto &key : extra.keys()) definition.extra[key] = extra.get(key.c_str()).asString();
340 }
341 proposed.emplace(definition.id, std::move(definition));
342 }
343
344 for (const auto &[id, definition] : proposed) {
345 for (const auto &required : definition.requiresIds)
346 if (proposed.count(required) == 0)
347 return strictFailure("required quest is missing from replacement catalogue", id + ".requires");
348 if (idOnCycle(proposed, id)) return strictFailure("quest prerequisite cycle detected", id + ".requires");
349 }
350
351 questTable() = std::move(proposed);
352 return eve::Result<int>::success(int(questTable().size()));
353}

368eve::Result<int> QuestReward::claim(Tracker &tracker, GameState &gameState, inventory::Bag &bag,
369 const std::string &questId) {
370 const QuestDefinition *definition = QuestRegistry::find(questId);
371 if (!definition)
372 return rewardFailure(eve::DiagnosticCode::NotFound, "quest reward claim references an unknown quest", "questId",
373 questId);
374 if (tracker.getState(questId) != "ready")
375 return rewardFailure(eve::DiagnosticCode::PreconditionViolation,
376 "quest must be ready before rewards can be claimed", "quest.state", questId);
377
378 std::vector<inventory::InventoryItemGrant> itemGrants;
379 itemGrants.reserve(definition->rewards.size());
380 for (std::size_t index = 0; index < definition->rewards.size(); ++index) {
381 const RewardSpec &reward = definition->rewards[index];
382 const std::string path = "quest.rewards[" + std::to_string(index) + "]";
383 if (reward.id.empty() || !std::isfinite(reward.amount) || reward.amount <= 0.0)
384 return rewardFailure(eve::DiagnosticCode::InvalidArgument,
385 "quest reward requires an id and finite positive amount", path, questId);
386 if (reward.type == "item") {
387 if (reward.amount > static_cast<double>(INT_MAX) || std::floor(reward.amount) != reward.amount)
388 return rewardFailure(eve::DiagnosticCode::InvalidArgument,
389 "item reward amount must be a positive integer", path + ".amount", questId);
390 itemGrants.push_back({reward.id, static_cast<int>(reward.amount)});
391 } else if (reward.type == "attribute") {
392 const double committed = gameState.getVariable(reward.id) + reward.amount;
393 if (!std::isfinite(committed))
394 return rewardFailure(eve::DiagnosticCode::InvalidArgument,
395 "attribute reward would produce a non-finite value", path + ".amount", questId);
396 } else {
397 return rewardFailure(eve::DiagnosticCode::Unsupported,
398 "quest reward type is not supported by the canonical claim transaction",
399 path + ".type", questId);
400 }
401 }
402
403 std::optional<inventory::PreparedInventoryAdd> preparedInventory;
404 if (!itemGrants.empty()) {
405 auto prepared = inventory::InventorySystem::prepareAddBatch(&bag, itemGrants);
406 if (!prepared.ok()) return eve::Result<int>::failure(prepared.status());
407 preparedInventory.emplace(std::move(prepared).takeValue());
408 }
409
410 auto gameStateBefore = gameState.snapshotJson();
411 if (!gameStateBefore.ok()) return eve::Result<int>::failure(gameStateBefore.status());
412 std::string gameStateSnapshot = std::move(gameStateBefore).takeValue();
413 auto entriesBefore = tracker.entries;
414 auto orderBefore = tracker.order;
415 auto pendingBefore = tracker.pending;
416 auto polledBefore = tracker.polled;
417
418 for (const RewardSpec &reward : definition->rewards)
419 if (reward.type == "attribute") gameState.addVariable(reward.id, reward.amount);
420
421 std::string claimReason;
422 if (!QuestSystem::claim(&tracker, questId, &claimReason)) {
423 auto restored = gameState.restoreSnapshotJson(gameStateSnapshot);
424 const bool restoredOk = restored.ok();
425 tracker.entries = std::move(entriesBefore);
426 tracker.order = std::move(orderBefore);
427 tracker.pending = std::move(pendingBefore);
428 tracker.polled = std::move(polledBefore);
429 if (!restoredOk)
430 return rewardFailure(eve::DiagnosticCode::InvariantViolation,
431 "quest reward rollback could not restore GameState", "rollback", questId);
432 return rewardFailure(eve::DiagnosticCode::Conflict, "quest state changed before reward commit: " + claimReason,
433 "quest.state", questId);
434 }
435
436 if (preparedInventory) {
437 auto committed = inventory::InventorySystem::commitAddBatch(std::move(*preparedInventory));
438 if (!committed.ok()) {
439 auto restored = gameState.restoreSnapshotJson(gameStateSnapshot);
440 const bool restoredOk = restored.ok();
441 tracker.entries = std::move(entriesBefore);
442 tracker.order = std::move(orderBefore);
443 tracker.pending = std::move(pendingBefore);
444 tracker.polled = std::move(polledBefore);
445 if (!restoredOk)
446 return rewardFailure(eve::DiagnosticCode::InvariantViolation,
447 "quest reward rollback could not restore GameState", "rollback", questId);
448 return eve::Result<int>::failure(committed.status());
449 }
450 }
451
452 return eve::Result<int>::success(static_cast<int>(definition->rewards.size()),
453 eve::Status::success(eve::StatusCode::Applied));
454}

510void QuestSystem::syncAuto(Tracker *t) {
511 if (!t) return;
512 // 1. 补建 Registry 里尚未出现在本 Tracker 上的条目。
513 for (const auto &id : QuestRegistry::ids()) {
514 if (t->entries.count(id)) continue;
515 const QuestDefinition *def = QuestRegistry::find(id);
516 if (!def) continue;
517 QuestRuntime rt;
518 rt.id = id;
519 rt.state = "locked";
520 for (const auto &obj : def->objectives) {
521 ObjectiveRuntime orun;
522 orun.id = obj.id;
523 orun.current = 0;
524 orun.count = obj.count > 0 ? obj.count : 1;
525 orun.done = false;
526 rt.objectives.push_back(std::move(orun));
527 }
528 t->entries[id] = std::move(rt);
529 t->order.push_back(id);
530 }
531 // 2. 反复解开前置已齐的 locked 条目。
532 for (size_t pass = 0; pass <= t->entries.size(); ++pass) {
533 bool changed = false;
534 for (auto &[id, rt] : t->entries) {
535 if (rt.state != "locked") continue;
536 const QuestDefinition *def = QuestRegistry::find(id);
537 if (!def) continue;
538 if (!hasMetRequirements(*t, *def)) continue;
539 const std::string newState = stateForUnlocked(*def);
540 rt.state = newState;
541 if (newState == "active") pushEvent(t, {id, "", "activate", "", "", 0, ""});
542 if (newState == "active" && objectivesFull(rt)) onObjectivesFull(t, rt);
543 changed = true;
544 }
545 if (!changed) break;
546 }
547}

549std::string QuestSystem::canActivateReason(Tracker *t, const std::string &id) {
550 if (!t) return "unknown";
551 auto it = t->entries.find(id);
552 if (it == t->entries.end()) return "unknown";
553 const QuestDefinition *def = QuestRegistry::find(id);
554 if (!def) return "unknown";
555 const std::string &state = it->second.state;
556 if (state == "locked") return "locked";
557 if (state == "active") return "alreadyActive";

594void QuestSystem::notify(Tracker *t, const std::string &topic, const std::string &target, int amount) {
595 if (!t || amount <= 0) return;
596 std::vector<std::string> fullIds;
597 for (auto &[id, rt] : t->entries) {
598 if (rt.state != "active") continue;
599 const QuestDefinition *def = QuestRegistry::find(id);
600 if (!def) continue;
601 const size_t n = std::min(def->objectives.size(), rt.objectives.size());
602 for (size_t i = 0; i < n; ++i) {
603 const auto &objDef = def->objectives[i];
604 auto &rtObj = rt.objectives[i];
605 if (rtObj.done) continue;
606 if (rtObj.count <= 0) rtObj.count = 1;
607 const bool match = objDef.topic == topic && (objDef.target.empty() || objDef.target == target);
608 if (!match) continue;
609 const int added = std::min(amount, rtObj.count - rtObj.current);
610 rtObj.current += added;
611 if (rtObj.current >= rtObj.count) rtObj.done = true;
612 pushEvent(t, {id, rtObj.id, "progress", topic, target, added, ""});
613 }
614 if (objectivesFull(rt)) fullIds.push_back(id);
615 }
616 for (const auto &fid : fullIds) {
617 auto it = t->entries.find(fid);
618 if (it != t->entries.end()) onObjectivesFull(t, it->second);
619 }
620}

644bool QuestSystem::reset(Tracker *t, const std::string &id) {
645 if (!t) return false;
646 auto it = t->entries.find(id);
647 if (it == t->entries.end()) return false; // 未知 id：false，无事件
648 const QuestDefinition *def = QuestRegistry::find(id);
649 QuestRuntime &rt = it->second;
650 rt.objectives.clear();
651 if (def) {
652 for (const auto &obj : def->objectives) {
653 ObjectiveRuntime orun;
654 orun.id = obj.id;
655 orun.current = 0;
656 orun.count = obj.count > 0 ? obj.count : 1;
657 orun.done = false;
658 rt.objectives.push_back(std::move(orun));
659 }
660 }
661 rt.state = (def && hasMetRequirements(*t, *def)) ? stateForUnlocked(*def) : std::string("locked");
662 pushEvent(t, {id, "", "reset", "", "", 0, ""});
663 if (rt.state == "active" && objectivesFull(rt)) onObjectivesFull(t, rt);
664 return true;
665}
