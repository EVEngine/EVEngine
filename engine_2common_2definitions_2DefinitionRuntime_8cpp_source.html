125eve::Result<ReloadOutcome> RuntimeInstanceCore::reload(const DefinitionHandle& next, ReloadPolicy policy,
126 const void* defaults, const RebuildFunction& rebuild) {
127 if (!next.isValid())
128 return eve::Result<ReloadOutcome>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
129 "next definition handle is invalid", "next",
130 {}, "common.definitions"));
131 if (next.reference != identity_.definition)
132 return eve::Result<ReloadOutcome>::failure(eve::Diagnostic::error(
133 eve::DiagnosticCode::Conflict, "definition reload targets a different logical definition", "next.reference",
134 {}, "common.definitions"));
135 if (next.generation < identity_.definitionGeneration)
136 return eve::Result<ReloadOutcome>::failure(eve::Diagnostic::error(
137 eve::DiagnosticCode::StaleHandle, "definition reload handle is older than the instance generation",
138 "next.generation", {}, "common.definitions"));
139
140 ReloadOutcome outcome{
141 identity_.instanceId, identity_.definition, identity_.definitionGeneration, next.generation, policy,
142 ReloadDisposition::Unchanged};
143 if (next.generation == identity_.definitionGeneration)
144 return eve::Result<ReloadOutcome>::success(std::move(outcome), eve::Status::success(eve::StatusCode::NoOp));
145
146 if (policy == ReloadPolicy::RejectWhileActive && active_)
147 return eve::Result<ReloadOutcome>::failure(
148 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "active runtime instance rejects definition reload",
149 "policy", {}, "common.definitions"));
150 if (policy == ReloadPolicy::ReapplyDefaults && (defaults == nullptr || assign_ == nullptr))
151 return eve::Result<ReloadOutcome>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
152 "reapply policy requires typed defaults",
153 "defaults", {}, "common.definitions"));
154
155 try {
156 PayloadGuard candidate(clone_(state_), destroy_);
157 ReloadDisposition disposition = ReloadDisposition::Kept;
158 if (policy == ReloadPolicy::ReapplyDefaults) {
159 assign_(candidate.get(), defaults);
160 disposition = ReloadDisposition::DefaultsReapplied;
161 } else if (policy == ReloadPolicy::RebuildInstance) {
162 if (!rebuild)
163 return eve::Result<ReloadOutcome>::failure(eve::Diagnostic::error(
164 eve::DiagnosticCode::InvalidArgument, "rebuild policy requires a rebuild callback", "rebuild", {},
165 "common.definitions"));
166 auto rebuilt = rebuild(state_, identity_, next);
167 if (!rebuilt) return eve::Result<ReloadOutcome>::failure(rebuilt.status());
168 candidate.adopt(std::move(rebuilt).takeValue());
169 disposition = ReloadDisposition::Rebuilt;
170 } else if (policy == ReloadPolicy::RejectWhileActive) {
171 // The active guard was handled above. Inactive instances have
172 // the documented keep-values behavior.
173 disposition = ReloadDisposition::Kept;
174 }
175
176 if (state_ != nullptr && destroy_ != nullptr) destroy_(state_);
177 state_ = candidate.release();
178 identity_.definitionGeneration = next.generation;
179 outcome.disposition = disposition;
180 return eve::Result<ReloadOutcome>::success(std::move(outcome), eve::Status::success(eve::StatusCode::Applied));
181 } catch (const std::exception&) {
182 return eve::Result<ReloadOutcome>::failure(
183 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "definition reload candidate preparation failed", {},
184 {}, "common.definitions"));
185 } catch (...) {
186 return eve::Result<ReloadOutcome>::failure(
187 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "definition reload candidate preparation failed", {},
188 {}, "common.definitions"));
189 }
190}

192eve::Result<void> RuntimeInstanceCore::restoreExact(const InstanceIdentity& identity, void* state, bool active) {
193 PayloadGuard candidate(state, destroy_);
194 if (!identity.isValid())
195 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
196 "runtime snapshot identity is invalid", "identity", {},
197 "common.definitions"));
198 if (identity.instanceId != identity_.instanceId || identity.definition != identity_.definition)
199 return eve::Result<void>::failure(eve::Diagnostic::error(
200 eve::DiagnosticCode::Conflict, "runtime snapshot belongs to a different instance or definition", "identity",
201 {}, "common.definitions"));
202 if (identity.definitionGeneration != identity_.definitionGeneration)
203 return eve::Result<void>::failure(eve::Diagnostic::error(
204 eve::DiagnosticCode::StaleHandle, "runtime snapshot definition generation is not current",
205 "identity.definitionGeneration", {}, "common.definitions"));
206
207 if (state_ != nullptr && destroy_ != nullptr) destroy_(state_);
208 state_ = candidate.release();
209 active_ = active;
210 return eve::Result<void>::success();
211}
