72bool readStringArray(const eve::Value::Object& object, std::string_view name,
73 std::vector<std::string>& output) {
74 const auto* value = field(object, name);
75 if (value == nullptr) return true;
76 const auto* array = value->getIf<eve::Value::Array>();
77 if (array == nullptr) return false;
78 std::vector<std::string> candidate;
79 candidate.reserve(array->size());
80 for (const auto& entry : *array) {
81 const auto* text = entry.getIf<std::string>();
82 if (text == nullptr || text->empty()) return false;
83 candidate.push_back(*text);
84 }
85 std::sort(candidate.begin(), candidate.end());
86 if (std::adjacent_find(candidate.begin(), candidate.end()) != candidate.end()) return false;
87 output = std::move(candidate);
88 return true;
89}

117eve::Result<WeaponDefinition> parseDefinition(const eve::definitions::Definition& source,
118 const eve::DefinitionRef& reference) {
119 auto parsed = eve::Value::fromJson(source.json);
120 if (!parsed) return eve::Result<WeaponDefinition>::failure(parsed.status());
121 auto value = std::move(parsed).takeValue();
122 const auto* object = value.getIf<eve::Value::Object>();
123 if (object == nullptr) return invalid<WeaponDefinition>("weapon definition must be an object", "json");
124
125 WeaponDefinition result;
126 result.id = std::string(reference.id().name());
127 std::string id = result.id;
128 if (!readText(*object, "id", id) || id != result.id)
129 return invalid<WeaponDefinition>("weapon definition id does not match its logical reference", "id");
130 std::string kind = weaponKindName(result.kind);
131 std::string logic = result.logic;
132 if (!readText(*object, "kind", kind) || !readText(*object, "logic", logic))
133 return invalid<WeaponDefinition>("weapon kind and logic must be strings", "definition");
134 result.kind = weaponKindFromName(kind);
135 result.logic = std::move(logic);
136
137 if (!readNumber(*object, "damage", result.damage) || !readNumber(*object, "penetration", result.penetration) ||
138 !readNumber(*object, "range", result.range) || !readNumber(*object, "spread", result.spread) ||
139 !readNumber(*object, "falloffStart", result.falloffStart) ||
140 !readNumber(*object, "minimumDamageFactor", result.minimumDamageFactor) ||
141 !readNumber(*object, "splashMinimumDamageFactor", result.splashMinimumDamageFactor) ||
142 !readNumber(*object, "preferredTargetBonus", result.preferredTargetBonus) ||
143 !readNumber(*object, "accuracy", result.accuracy) ||
144 !readNumber(*object, "scatterRadius", result.scatterRadius) ||
145 !readNumber(*object, "spreadMin", result.spreadMin) || !readNumber(*object, "spreadMax", result.spreadMax) ||
146 !readNumber(*object, "spreadPerShot", result.spreadPerShot) ||
147 !readNumber(*object, "spreadRecover", result.spreadRecover) ||
148 !readNumber(*object, "recoilPitch", result.recoilPitch) ||
149 !readNumber(*object, "recoilYaw", result.recoilYaw) ||
150 !readNumber(*object, "recoilRecover", result.recoilRecover) ||
151 !readNumber(*object, "zoomFov", result.zoomFov) || !readNumber(*object, "cooldown", result.cooldown) ||
152 !readNumber(*object, "arc", result.arc))
153 return invalid<WeaponDefinition>("weapon numeric fields must be finite numbers", "definition");
154 if (result.falloffStart < 0.0f || result.falloffStart > result.range ||
155 result.minimumDamageFactor < 0.0f || result.minimumDamageFactor > 1.0f ||
156 result.splashMinimumDamageFactor < 0.0f || result.splashMinimumDamageFactor > 1.0f ||
157 result.accuracy < 0.0f || result.accuracy > 1.0f || result.scatterRadius < 0.0f ||
158 result.preferredTargetBonus < 1.0f)
159 return invalid<WeaponDefinition>("weapon damage falloff fields are outside valid ranges", "definition");
160
161 if (!readText(*object, "damageType", result.damageType) || !readText(*object, "element", result.element) ||
162 !readBoolean(*object, "targetsGround", result.targetsGround) ||
163 !readBoolean(*object, "targetsAir", result.targetsAir) ||
164 !readBoolean(*object, "friendlyFire", result.friendlyFire) ||
165 !readBoolean(*object, "blockedByObstacles", result.blockedByObstacles) ||
166 !readStringArray(*object, "requiredTargetTags", result.requiredTargetTags) ||
167 !readStringArray(*object, "excludedTargetTags", result.excludedTargetTags) ||
168 !readStringArray(*object, "preferredTargetTags", result.preferredTargetTags))
169 return invalid<WeaponDefinition>("weapon target policy fields are invalid", "definition");
170 if (!result.targetsGround && !result.targetsAir)
171 return invalid<WeaponDefinition>("weapon must target ground, air, or both", "definition");
172
173 if (const auto* modes = field(*object, "fireModes")) {
174 const auto* array = modes->getIf<eve::Value::Array>();
175 if (array == nullptr) return invalid<WeaponDefinition>("fireModes must be an array", "fireModes");
176 for (const auto& entry : *array) {
177 const auto* text = entry.getIf<std::string>();
178 if (text == nullptr) return invalid<WeaponDefinition>("fire mode must be a string", "fireModes");
179 result.selectableModes.push_back(fireModeFromName(*text));
180 }
181 }
182 std::string fireMode = fireModeName(result.fireMode);
183 if (!readText(*object, "fireMode", fireMode))
184 return invalid<WeaponDefinition>("fireMode must be a string", "fireMode");
185 result.fireMode = fireModeFromName(fireMode);
186
187 if (const auto* resource = field(*object, "resource")) {
188 const auto* nested = resource->getIf<eve::Value::Object>();
189 if (nested == nullptr) return invalid<WeaponDefinition>("resource must be an object", "resource");
190 std::string resourceName = resourceKindName(result.resource.kind);
191 if (!readText(*nested, "kind", resourceName) || !readNumber(*nested, "max", result.resource.max) ||
192 !readNumber(*nested, "regen", result.resource.regen) ||
193 !readNumber(*nested, "cost", result.resource.cost) ||
194 !readBoolean(*nested, "infinite", result.resource.infinite))
195 return invalid<WeaponDefinition>("weapon resource fields are invalid", "resource");
196 result.resource.kind = resourceKindFromName(resourceName);
197 }
198 if (const auto* stages = field(*object, "stages")) {
199 const auto* nested = stages->getIf<eve::Value::Object>();
200 if (nested == nullptr || !readNumber(*nested, "windup", result.stages.windupTime) ||
201 !readNumber(*nested, "active", result.stages.activeTime) ||
202 !readNumber(*nested, "recover", result.stages.recoverTime))
203 return invalid<WeaponDefinition>("stages fields are invalid", "stages");
204 }
205 if (const auto* burst = field(*object, "burst")) {
206 const auto* nested = burst->getIf<eve::Value::Object>();
207 if (nested == nullptr || !readInteger(*nested, "size", result.burstSize) ||
208 !readNumber(*nested, "interval", result.burstInterval))
209 return invalid<WeaponDefinition>("burst fields are invalid", "burst");
210 }
211 if (const auto* ammo = field(*object, "ammo")) {
212 const auto* nested = ammo->getIf<eve::Value::Object>();
213 if (nested == nullptr || !readInteger(*nested, "mag", result.magSize) ||
214 !readInteger(*nested, "reserve", result.reserveSize) || !readNumber(*nested, "reload", result.reloadTime))
215 return invalid<WeaponDefinition>("ammo fields are invalid", "ammo");
216 }
217 if (const auto* projectile = field(*object, "projectile")) {
218 const auto* nested = projectile->getIf<eve::Value::Object>();
219 if (nested == nullptr || !readText(*nested, "type", result.projectile.type) ||
220 !readNumber(*nested, "speed", result.projectile.speed) ||
221 !readNumber(*nested, "gravity", result.projectile.gravity) ||
222 !readNumber(*nested, "aoe", result.projectile.aoe) ||
223 !readInteger(*nested, "pelletCount", result.projectile.pelletCount) ||
224 !readNumber(*nested, "pelletSpread", result.projectile.pelletSpread))
225 return invalid<WeaponDefinition>("projectile fields are invalid", "projectile");
226 }
227 if (const auto* effects = field(*object, "effects")) {
228 const auto* nested = effects->getIf<eve::Value::Object>();
229 if (nested == nullptr || !readText(*nested, "muzzle", result.effectMuzzle) ||
230 !readText(*nested, "sound", result.effectSound))
231 return invalid<WeaponDefinition>("effects fields are invalid", "effects");
232 }
233 if (result.resource.kind == ResourceKind::None) result.resource.infinite = true;
234 return eve::Result<WeaponDefinition>::success(std::move(result));
235}

315eve::Result<WeaponRuntimeState> decodeState(const eve::Value& value) {
316 const auto* object = value.getIf<eve::Value::Object>();
317 if (object == nullptr) return invalid<WeaponRuntimeState>("weapon runtime state must be an object", "state");
318 static const std::set<std::string> fields = {"aiming", "burstRemaining", "burstTimer", "cooldown",
319 "currentSpread", "jammed", "recoilPitch", "recoilYaw",
320 "resource", "selector", "stage", "stageTimer"};
321 for (const auto& [name, unused] : *object) {
322 (void)unused;
323 if (!fields.contains(name))
324 return invalid<WeaponRuntimeState>("unknown weapon runtime state field", "state." + name);
325 }
326 for (const auto& name : fields)
327 if (!object->contains(name))
328 return invalid<WeaponRuntimeState>("weapon runtime state is missing a field", "state." + name);
329
330 WeaponRuntimeState result;
331 std::string selector;
332 std::string stage;
333 if (!readBoolean(*object, "aiming", result.aiming) ||
334 !readInteger(*object, "burstRemaining", result.burstRemaining) ||
335 !readStateNumber(*object, "burstTimer", result.burstTimer) ||
336 !readStateNumber(*object, "cooldown", result.cooldown) ||
337 !readStateNumber(*object, "currentSpread", result.currentSpread) ||
338 !readBoolean(*object, "jammed", result.jammed) ||
339 !readStateNumber(*object, "recoilPitch", result.recoilPitch) ||
340 !readStateNumber(*object, "recoilYaw", result.recoilYaw) || !readText(*object, "selector", selector) ||
341 !readText(*object, "stage", stage) || !readStateNumber(*object, "stageTimer", result.stageTimer))
342 return invalid<WeaponRuntimeState>("weapon runtime scalar field is invalid", "state");
343 result.selector = fireModeFromName(selector);
344 result.stage = attackStageFromName(stage);
345 const auto* resourceValue = object->at("resource").getIf<eve::Value::Object>();
346 if (resourceValue == nullptr)
347 return invalid<WeaponRuntimeState>("weapon runtime resource is invalid", "state.resource");
348 std::string kind;
349 if (!readText(*resourceValue, "kind", kind) || !readStateNumber(*resourceValue, "value", result.resource.value) ||
350 !readStateNumber(*resourceValue, "max", result.resource.max) ||
351 !readStateNumber(*resourceValue, "regen", result.resource.regen) ||
352 !readStateNumber(*resourceValue, "cost", result.resource.cost) ||
353 !readBoolean(*resourceValue, "infinite", result.resource.infinite) ||
354 !readInteger(*resourceValue, "reserve", result.resource.reserve) ||
355 !readBoolean(*resourceValue, "reloading", result.resource.reloading) ||
356 !readStateNumber(*resourceValue, "reloadProgress", result.resource.reloadProgress))
357 return invalid<WeaponRuntimeState>("weapon runtime resource fields are invalid", "state.resource");
358 result.resource.kind = resourceKindFromName(kind);
359 return eve::Result<WeaponRuntimeState>::success(std::move(result));
360}

428eve::Result<void> WeaponDefinitionRuntime::applyTo(WeaponEntity* entity) const {
429 if (entity == nullptr)
430 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
431 "weapon runtime cannot project to a null WeaponEntity",
432 "entity", {}, "weapon.definition_runtime"));
433 auto typed = definition();
434 if (!typed) return eve::Result<void>::failure(typed.status());
435 try {
436 auto definitionOwner = std::make_shared<const WeaponDefinition>(typed.value());
437 auto definitionComponent = *entity->definition();
438 auto binding = *entity->definitionBinding();
439 auto targetState = *entity->state();
440 definitionComponent.owned = std::move(definitionOwner);
441 definitionComponent.def = definitionComponent.owned.get();
442 binding.identity = identity();
443 binding.reloadPolicy = policy_;
444 binding.active = isActive();
445 const auto& source = state();
446 targetState.resource = source.resource;
447 targetState.cooldown = source.cooldown;
448 targetState.burstRemaining = source.burstRemaining;
449 targetState.burstTimer = source.burstTimer;
450 targetState.jammed = source.jammed;
451 targetState.stage = source.stage;
452 targetState.stageTimer = source.stageTimer;
453 targetState.stages = &definitionComponent.def->stages;
454 targetState.currentSpread = source.currentSpread;
455 targetState.recoilPitch = source.recoilPitch;
456 targetState.recoilYaw = source.recoilYaw;
457 targetState.selector = source.selector;
458 targetState.aiming = source.aiming;
459 using std::swap;
460 swap(*entity->definition(), definitionComponent);
461 swap(*entity->definitionBinding(), binding);
462 swap(*entity->state(), targetState);
463 return eve::Result<void>::success();
464 } catch (const std::exception&) {
465 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::Failed,
466 "weapon runtime projection allocation failed",
467 "entity", {}, "weapon.definition_runtime"));

471eve::Result<eve::definition::ReloadOutcome> WeaponDefinitionRuntime::reload(eve::definition::ReloadPolicy policy) {
472 if (registry_ == nullptr)
473 return invalid<eve::definition::ReloadOutcome>("weapon definition registry is not bound", "registry");
474 auto next = currentHandle(*registry_, identity().definition);
475 if (!next) return eve::Result<eve::definition::ReloadOutcome>::failure(next.status());
476 auto typed = definitionFor(*registry_, next.value(), identity().definition);
477 if (!typed) return eve::Result<eve::definition::ReloadOutcome>::failure(typed.status());
478 const WeaponRuntimeState defaults = initialState(typed.value());
479 const float maxSpread = std::max(defaults.currentSpread, typed.value().spreadMax);
480 auto rebuild = [defaults, maxSpread](const WeaponRuntimeState& oldState, const eve::definition::InstanceIdentity&,
481 const eve::definition::DefinitionHandle&) {
482 WeaponRuntimeState rebuilt = oldState;
483 if (oldState.resource.kind != defaults.resource.kind) {
484 rebuilt.resource = defaults.resource;
485 } else {
486 rebuilt.resource.kind = defaults.resource.kind;
487 rebuilt.resource.max = defaults.resource.max;
488 rebuilt.resource.regen = defaults.resource.regen;
489 rebuilt.resource.cost = defaults.resource.cost;
490 rebuilt.resource.infinite = defaults.resource.infinite;
491 if (rebuilt.resource.infinite) {
492 rebuilt.resource.value = 0.f;
493 } else {
494 rebuilt.resource.value = std::clamp(rebuilt.resource.value, 0.f, rebuilt.resource.max);
495 }
496 rebuilt.resource.reserve = defaults.resource.reserve < 0 ? 0 : rebuilt.resource.reserve;
497 }
498 rebuilt.selector = defaults.selector;
499 rebuilt.currentSpread = std::clamp(oldState.currentSpread, 0.f, maxSpread);
500 return eve::Result<WeaponRuntimeState>::success(std::move(rebuilt));
501 };
502 auto result = runtime_.reload(next.value(), policy, defaults, std::move(rebuild));
503 if (result.ok()) policy_ = policy;
504 return result;
505}
