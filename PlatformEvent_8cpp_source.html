55std::unique_ptr<Message> PlatformEvent::pollOwned() {
56 std::unique_ptr<Message> msg;
57 {
58 std::lock_guard<std::mutex> lock(queueMu_);
59 if (queue.empty()) return nullptr;
60 msg = std::move(queue.front());
61 queue.pop();
62 }
63 // Report consumption outside the lock so observers may re-enter push/poll.
64 // The queue mutation is already committed; callback exceptions are
65 // contained and counted instead of making a successful poll look failed.
66 if (msg) {
67 static_cast<void>(pollObservers_.notifyChecked([this]() noexcept { ++pollObserverFailures_; }, *msg));
68 }
69 return msg;
70}
