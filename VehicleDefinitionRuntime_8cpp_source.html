438 eve::DiagnosticCode::InvalidArgument, "vehicle runtime cannot project to a null VehicleEntity", "entity",
439 {}, "vehicle.definition_runtime"));
440 auto typed = definition();
441 if (!typed) return eve::Result<void>::failure(typed.status());
442 try {
443 auto definitionOwner = std::make_shared<const VehicleDefinition>(typed.value());
444 auto identity = *entity->identity();

446 auto binding = *entity->definitionBinding();
447 auto motion = *entity->motion();
448 auto health = *entity->health();
449 auto flags = *entity->stateFlags();
450
451 identity.id = this->identity().instanceId.format();

459 motion.x = state().x;

464 health.maxHp = typed.value().maxHealth;
465 health.hp = std::clamp(state().health, 0.f, health.maxHp);

583eve::Result<VehicleEffectUpdate> VehicleEffectExecutor::settle(VehicleEffectTarget& target,
584 effects::EffectUpdateSummary lifecycle) const {
585 auto valid = validate(target);
586 if (!valid) return eve::Result<VehicleEffectUpdate>::failure(valid.status());
587 VehicleEffectUpdate result;
588 result.lifecycle = std::move(lifecycle);
589 for (const auto& tick : result.lifecycle.periodicTicks) {
590 if (tick.stackCount == 0 || tick.magnitude < 0.0 || !std::isfinite(tick.magnitude))
591 return eve::Result<VehicleEffectUpdate>::failure(eve::Diagnostic::error(
592 eve::DiagnosticCode::InvalidArgument, "vehicle periodic effect value is invalid", "periodic"));
593 const double amount = tick.magnitude * static_cast<double>(tick.stackCount);
594 if (!std::isfinite(amount))
595 return eve::Result<VehicleEffectUpdate>::failure(eve::Diagnostic::error(
596 eve::DiagnosticCode::InvalidArgument, "vehicle periodic settlement overflows", "periodic"));
597 const auto has = [&tick](const char* tag) {
598 return std::find(tick.tags.begin(), tick.tags.end(), tag) != tick.tags.end();
599 };
600 if (has("vehicle:repair")) {
601 target.hull = std::min(target.maxHull, target.hull + amount);
602 } else if (has("vehicle:disable")) {
603 target.disabled = true;
604 } else {
605 const double damage = amount * zoneMultiplier(target, tick.tags);
606 if (!std::isfinite(damage))
607 return eve::Result<VehicleEffectUpdate>::failure(eve::Diagnostic::error(
608 eve::DiagnosticCode::InvalidArgument, "vehicle armor settlement overflows", "periodic"));
609 target.hull = std::max(0.0, target.hull - damage);
610 if (damage >= target.maxHull * 0.5) {
611 ++target.criticalHits;
612 ++result.criticalHits;
613 }
614 }
615 ++result.settled;
616 }
617 return eve::Result<VehicleEffectUpdate>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
618}
