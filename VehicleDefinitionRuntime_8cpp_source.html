261eve::Result<VehicleRuntimeState> decodeState(const eve::Value& value) {
262 const auto* object = value.getIf<eve::Value::Object>();
263 if (object == nullptr) return invalid<VehicleRuntimeState>("vehicle runtime state must be an object", "state");
264 static const std::set<std::string> fields = {"destroyed", "faction", "heading", "health", "speed", "x", "y"};
265 for (const auto& [name, unused] : *object) {
266 (void)unused;
267 if (!fields.contains(name))
268 return invalid<VehicleRuntimeState>("unknown vehicle runtime state field", "state." + name);
269 }
270 for (const auto& name : fields)
271 if (!object->contains(name))
272 return invalid<VehicleRuntimeState>("vehicle runtime state is missing a field", "state." + name);
273 auto number = [&](std::string_view name, float& output, bool allowNegative) -> bool {
274 const auto* real = object->at(std::string(name)).getIf<double>();
275 if (real == nullptr || !std::isfinite(*real) || (!allowNegative && *real < 0.0) ||
276 *real < -std::numeric_limits<float>::max() || *real > std::numeric_limits<float>::max())
277 return false;
278 output = static_cast<float>(*real);
279 return true;
280 };
281 const auto* destroyed = object->at("destroyed").getIf<bool>();
282 const auto* faction = object->at("faction").getIf<std::string>();
283 VehicleRuntimeState result;
284 if (destroyed == nullptr || faction == nullptr || !number("heading", result.heading, false) ||
285 !number("health", result.health, false) || !number("speed", result.speed, false) ||
286 !number("x", result.x, true) || !number("y", result.y, true))
287 return invalid<VehicleRuntimeState>("vehicle runtime state field is invalid", "state");
288 result.destroyed = *destroyed;
289 result.faction = *faction;
290 return eve::Result<VehicleRuntimeState>::success(std::move(result));
291}

368 auto binding = *entity->definitionBinding();
369 auto motion = *entity->motion();
370 auto health = *entity->health();
371 auto flags = *entity->stateFlags();
372

374 identity.defId = this->identity().definition.id().name();
375 definition.owned = std::move(definitionOwner);

404eve::Result<eve::definition::ReloadOutcome> VehicleDefinitionRuntime::reload(eve::definition::ReloadPolicy policy) {
405 if (registry_ == nullptr)
406 return invalid<eve::definition::ReloadOutcome>("vehicle definition registry is not bound", "registry");
407 auto next = currentHandle(*registry_, identity().definition);
408 if (!next) return eve::Result<eve::definition::ReloadOutcome>::failure(next.status());
409 auto resolved = registry_->resolveHandle(next.value());
410 if (!resolved) return eve::Result<eve::definition::ReloadOutcome>::failure(resolved.status());
411 auto typed = parseVehicleDefinition(resolved.value().get());
412 if (!typed) return eve::Result<eve::definition::ReloadOutcome>::failure(typed.status());
413 VehicleRuntimeState defaults;
414 defaults.health = typed.value().maxHealth;
415 const float maxHealth = typed.value().maxHealth;
416 auto rebuild = [maxHealth](const VehicleRuntimeState& oldState, const eve::definition::InstanceIdentity&,
417 const eve::definition::DefinitionHandle&) {
418 VehicleRuntimeState rebuilt = oldState;
419 rebuilt.health = std::clamp(oldState.health, 0.f, maxHealth);
420 return eve::Result<VehicleRuntimeState>::success(std::move(rebuilt));
421 };
422 auto result = runtime_.reload(next.value(), policy, defaults, std::move(rebuild));
423 if (result.ok()) policy_ = policy;
424 return result;
425}
