54Result<std::unique_ptr<Lightning>> Lightning::prepare(graphics::Graphics& graphics, const LightningSettings& settings) {
55 auto valid = validateLightning(settings);
56 if (!valid) return Result<std::unique_ptr<Lightning>>::failure(valid.status());
57 auto impl = std::make_unique<Impl>();
58 impl->graphics = &graphics;
59 impl->lifetime = graphics.resourceLifetime();
60 impl->settings = settings;
61 auto subscription = graphics.onResourcesRetiring([state = impl.get()] { state->cleanup(); });
62 if (!subscription) return Result<std::unique_ptr<Lightning>>::failure(subscription.status());
63 impl->retiring = std::move(subscription).takeValue();
64 try {
65 impl->meshes.reserve(static_cast<size_t>(settings.variants));
66 if (graphics.getBackendName() == "webgpu")
67 impl->shader = graphics.newMeshShaderFromWgsl(shaders::kBoltVertWgsl, shaders::kBoltFragWgsl);
68 else
69 impl->shader = graphics.newMeshShaderFromSpv({bolt_vert_spv, bolt_vert_spv + bolt_vert_spv_count},
70 {bolt_frag_spv, bolt_frag_spv + bolt_frag_spv_count});
71 if (!impl->shader)
72 return Result<std::unique_ptr<Lightning>>::failure(Diagnostic::error(
73 DiagnosticCode::InvalidArgument, "Lightning shader unavailable", "weather.lightning"));
74 impl->material = std::make_unique<graphics::Material>();
75 auto& m = *impl->material;
76 m.setShader(impl->shader);
77 m.setShadingModel("unlit");
78 m.setReceiveLight(false);
79 m.setReceiveShadow(false);
80 m.setCastShadow(false);
81 m.setCastOcclusion(false);
82 m.setTint(1, 1, 1, 1);
83 impl->shader->setPushConstantBlock({});
84 for (int i = 0; i < settings.variants; ++i) {
85 auto geometry = buildLightningGeometry(settings, static_cast<unsigned>(i));
86 if (!geometry) return Result<std::unique_ptr<Lightning>>::failure(geometry.status());
87 auto& g = geometry.value();
88 auto* mesh = graphics.newMeshFromArrays(g.positions.data(), g.normals.data(), g.uv.data(),
89 static_cast<int>(g.positions.size() / 3), g.indices.data(),
90 static_cast<int>(g.indices.size()));
91 if (!mesh)
92 return Result<std::unique_ptr<Lightning>>::failure(Diagnostic::error(
93 DiagnosticCode::InvalidArgument, "Lightning mesh unavailable", "weather.lightning"));
94 impl->meshes.push_back(mesh);
95 }
96 auto* entity = graphics::Renderable3D::create();
97 impl->entity = ecs::handle_of(entity);
98 entity->setMesh(impl->meshes.front());
99 entity->setMaterial(impl->material.get());
100 entity->setVisible(false);
101 } catch (const std::exception& e) {
102 return Result<std::unique_ptr<Lightning>>::failure(
103 Diagnostic::error(DiagnosticCode::InvalidArgument, e.what(), "weather.lightning"));
104 }
105 return Result<std::unique_ptr<Lightning>>::success(std::unique_ptr<Lightning>(new Lightning(std::move(impl))));
106}
107Result<void> Lightning::setEndpoints(float ax, float ay, float az, float bx, float by, float bz) {
108 auto* e = impl_->renderable();
109 if (!e) return invalid("Lightning renderer retired", DiagnosticCode::StaleHandle);
110 const float dx = bx - ax, dy = by - ay, dz = bz - az;
111 const float length = std::hypot(dx, dy, dz);
112 if (!std::isfinite(ax) || !std::isfinite(ay) || !std::isfinite(az) || !std::isfinite(length) || length < .0001f ||
113 length > 1e6f)
114 return invalid("Lightning endpoints must be finite and distinct, length <= 1e6");
115 e->setPosition(ax, ay, az);
116 e->setRotation(std::atan2(dx, dz), std::acos(std::clamp(dy / length, -1.f, 1.f)), 0);
117 e->setScale(1, length, 1);
118 return Result<void>::success();
119}

128Result<void> Lightning::trigger() {
129 auto* e = impl_->renderable();

131 e->setMesh(impl_->meshes[impl_->variant++ % impl_->meshes.size()]);
132 impl_->playing = true;

134 impl_->flash = 1;

141Result<void> Lightning::update(float dt) {
142 if (!std::isfinite(dt) || dt < 0) return invalid("Lightning dt must be finite and nonnegative");
143 auto* e = impl_->renderable();
144 if (!e) return invalid("Lightning renderer retired", DiagnosticCode::StaleHandle);
145 if (!impl_->playing) return Result<void>::success();
146 impl_->age = std::min(impl_->age + dt, static_cast<double>(impl_->settings.duration));
147 impl_->flash = lightningEnvelope(static_cast<float>(impl_->age / impl_->settings.duration));
148 std::array<float, graphics::Shader::kMaxFloats> block{};
149 block[11] = impl_->flash;
150 impl_->shader->setPushConstantBlock(block);
151 e->setVisible(impl_->flash > .02f);
152 if (impl_->age >= impl_->settings.duration) impl_->playing = false;
153 return Result<void>::success();
154}
