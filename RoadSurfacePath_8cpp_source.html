7Result<RoadSurfacePolyline> prepareSurfacePolyline(std::span<const SplineFrameSample> frames) {
8 if (frames.size() < 2 || frames.size() > 4099)
9 return bakeFail<RoadSurfacePolyline>(DiagnosticCode::InvalidArgument, "surface path needs 2..4099 rows",
10 "sidePlacement.pathSampling");
11 RoadSurfacePolyline result;
12 result.frames = frames;
13 result.lastVisible = frames.size() - 1;
14 result.distances.reserve(frames.size());
15 result.distances.push_back(0.);
16 for (std::size_t i = 1; i < frames.size(); ++i) {
17 const auto& a = frames[i - 1].sample;
18 const auto& b = frames[i].sample;
19 const double step =
20 std::hypot(static_cast<double>(b.x) - a.x, static_cast<double>(b.y) - a.y, static_cast<double>(b.z) - a.z);
21 if (!std::isfinite(step) || step <= 1e-7)
22 return bakeFail<RoadSurfacePolyline>(DiagnosticCode::InvalidArgument,
23 "surface path contains a degenerate segment",
24 "sidePlacement.pathSampling");
25 result.distances.push_back(result.distances.back() + step);
26 }
27 if (result.distances.back() > std::numeric_limits<float>::max())
28 return bakeFail<RoadSurfacePolyline>(DiagnosticCode::InvalidArgument, "surface length is not representable",
29 "sidePlacement.pathSampling");
30 result.length = static_cast<float>(result.distances.back());
31 return Result<RoadSurfacePolyline>::success(std::move(result));
32}

34Result<SplineFrameSample> sampleSurfacePolyline(const RoadSurfacePolyline& path, float distance) {
35 if (!std::isfinite(distance) || distance < 0 || path.frames.size() < 2 ||
36 path.distances.size() != path.frames.size())
37 return bakeFail<SplineFrameSample>(DiagnosticCode::InvalidArgument, "invalid surface sampling request",
38 "sidePlacement.pathSampling");
39 auto end = std::lower_bound(path.distances.begin() + 1, path.distances.end(), static_cast<double>(distance));
40 const auto index = end == path.distances.end() ? path.frames.size() - 2
41 : static_cast<std::size_t>(end - path.distances.begin() - 1);
42 const auto& a = path.frames[index];
43 const auto& b = path.frames[index + 1];
44 const float t =
45 static_cast<float>((distance - path.distances[index]) / (path.distances[index + 1] - path.distances[index]));
46 const float attributeT = std::clamp(t, 0.f, 1.f);
47 auto result = a;
48 result.sample.x = std::lerp(a.sample.x, b.sample.x, t);
49 result.sample.y = std::lerp(a.sample.y, b.sample.y, t);
50 result.sample.z = std::lerp(a.sample.z, b.sample.z, t);
51 result.sample.scaleX = std::lerp(a.sample.scaleX, b.sample.scaleX, attributeT);
52 result.sample.scaleY = std::lerp(a.sample.scaleY, b.sample.scaleY, attributeT);
53 result.sample.normalizedDistance = distance / path.length;
54 const V3 forward = normalize(V3{b.sample.x - a.sample.x, b.sample.y - a.sample.y, b.sample.z - a.sample.z});
55 V3 up{std::lerp(a.upX, b.upX, attributeT), std::lerp(a.upY, b.upY, attributeT),
56 std::lerp(a.upZ, b.upZ, attributeT)};
57 up = up - forward * dot(up, forward);
58 if (length(up) < 1e-6f) {
59 up = V3{a.upX, a.upY, a.upZ} - forward * dot(V3{a.upX, a.upY, a.upZ}, forward);
60 if (length(up) < 1e-6f) {
61 const V3 axis = std::fabs(forward.y) < .9f ? V3{0, 1, 0} : V3{1, 0, 0};
62 up = axis - forward * dot(axis, forward);
63 }
64 }
65 const V3 side = normalize(cross(up, forward));
66 up = normalize(cross(forward, side));
67 result.forwardX = forward.x;
68 result.forwardY = forward.y;
69 result.forwardZ = forward.z;
70 result.sideX = side.x;
71 result.sideY = side.y;
72 result.sideZ = side.z;
73 result.upX = up.x;
74 result.upY = up.y;
75 result.upZ = up.z;
76 result.sample.tangentX = forward.x;
77 result.sample.tangentY = forward.y;
78 result.sample.tangentZ = forward.z;
79 return Result<SplineFrameSample>::success(result);
80}

103 if (!std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z) || path.frames.size() < 2)

105 "sidePlacement.pathSampling");
106 const V3 target{x, y, z};
107 float best = std::numeric_limits<float>::max(), along = 0;
108 for (std::size_t i = path.firstVisible + 1; i <= path.lastVisible; ++i) {
109 const auto& a = path.frames[i - 1].sample;
110 const auto& b = path.frames[i].sample;
111 const V3 start{a.x, a.y, a.z}, delta{b.x - a.x, b.y - a.y, b.z - a.z};
112 const float t = std::clamp(dot(target - start, delta) / dot(delta, delta), 0.f, 1.f);
113 const V3 difference = target - (start + delta * t);
114 const float squared = dot(difference, difference);
115 if (squared < best) {
116 best = squared;
117 along = static_cast<float>(std::lerp(path.distances[i - 1], path.distances[i], static_cast<double>(t)));
118 }
119 }

121 if (!sampled.ok()) return Result<SplineSample>::failure(sampled.status());
122 return Result<SplineSample>::success(sampled.value().sample);
