18 return failure<std::vector<std::uint32_t>>("Shader stage size is invalid");

20 words.reserve(input->size());
21 for (const auto& word : *input) {
22 if (!word.isInt64() || word.asInt() < 0 || word.asInt() > std::numeric_limits<std::uint32_t>::max())

31 const auto count = words[cursor] >> 16;
32 const auto opcode = words[cursor] & 0xffff;
33 if (count == 0 || count > words.size() - cursor)
34 return failure<std::vector<std::uint32_t>>("Truncated SPIR-V instruction");
35 if (opcode == 15) {

44 }

50Result<ShaderAsset> decodeShaderAsset(const Value& definition, const ShaderAssetLimits& limits) {
51 const auto* fields = definition.getIf<Value::Object>();
52 if (!fields || fields->size() != 7)
53 return failure<ShaderAsset>("Shader definition requires exactly seven known fields");
54 for (const auto name : {"schema", "schemaVersion", "interface", "format", "vertex", "fragment", "parameters"})
55 if (!fields->contains(name)) return failure<ShaderAsset>("Shader field is missing or unknown");
56 if (!fields->at("schema").isString() || fields->at("schema").asString() != "eve.shader" ||
57 !fields->at("schemaVersion").isInt64() || fields->at("schemaVersion").asInt() != 1)
58 return Result<ShaderAsset>::failure(Diagnostic::error(
59 DiagnosticCode::UnknownVersion, "Unsupported shader schema/version", {}, {}, "asset.shader"));
60 if (!fields->at("format").isString() || fields->at("format").asString() != "spirv-1.6" ||
61 !fields->at("interface").isString())
62 return failure<ShaderAsset>("Unsupported shader format/interface");
63 ShaderAsset result;
64 const auto& interface = fields->at("interface").asString();
65 if (interface == "sprite2d")
66 result.interface = ShaderAssetInterface::Sprite2D;
67 else if (interface != "mesh3d")
68 return failure<ShaderAsset>("Unsupported shader interface");
69 const auto* parameters = fields->at("parameters").getIf<Value::Array>();
70 if (!parameters || parameters->size() > 32) return failure<ShaderAsset>("Invalid shader parameter count");
71 std::set<std::string> names;
72 std::size_t slots = 0;
73 for (const auto& parameter : *parameters) {
74 const auto* object = parameter.getIf<Value::Object>();
75 if (!object || object->size() != 2 || !object->contains("name") || !object->contains("default") ||
76 !object->at("name").isString())
77 return failure<ShaderAsset>("Invalid parameter fields");
78 const auto& name = object->at("name").asString();
79 const auto* defaults = object->at("default").getIf<Value::Array>();
80 if (name.empty() || name.size() > 128 || name.find('\0') != std::string::npos || !names.insert(name).second ||
81 !defaults || defaults->empty() || defaults->size() > 4 || (slots += defaults->size()) > 32)
82 return failure<ShaderAsset>("Invalid parameter name or push-constant capacity");
83 ShaderAssetParameter output{name, {}};
84 for (const auto& value : *defaults) {
85 if (!value.isDouble() && !value.isInt64()) return failure<ShaderAsset>("Non-numeric default");
86 const double number = value.isDouble() ? value.asDouble() : static_cast<double>(value.asInt());
87 if (!std::isfinite(number) || std::abs(number) > std::numeric_limits<float>::max())
88 return failure<ShaderAsset>("Non-finite float default");
89 output.defaults.push_back(static_cast<float>(number));

91 result.parameters.push_back(std::move(output));
92 }

99 return Result<ShaderAsset>::success(std::move(result));
