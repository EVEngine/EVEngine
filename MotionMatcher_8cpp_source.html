72void MotionMatcher::buildQuery(std::vector<float>& query) const {
73 if (!database_->isBaked()) throw Exception("MotionMatcher: database not baked");
74 const int n = database_->getFeatureSize();
75 query.assign(static_cast<size_t>(n), 0.f);
76
77 // Character-space desired velocity (assume current facing = desiredYaw for query).
78 const float cs = std::cos(desiredYaw_);
79 const float sn = std::sin(desiredYaw_);
80 query[0] = desiredVelX_ * cs + desiredVelZ_ * sn;
81 query[1] = -desiredVelX_ * sn + desiredVelZ_ * cs;
82
83 // Desired trajectory: integrate constant velocity for 0.33/0.66/1.0s in char space.
84 const float horizons[3] = {0.33f, 0.66f, 1.0f};
85 for (int h = 0; h < 3; ++h) {
86 query[static_cast<size_t>(2 + h * 2)] = query[0] * horizons[h];
87 query[static_cast<size_t>(2 + h * 2 + 1)] = query[1] * horizons[h];
88 }
89 // Facing at +1s: same as desired (relative facing 0 → forward +Z in char space).
90 float fx, fz;
91 yawToForward(0.f, fx, fz);
92 query[8] = fx;
93 query[9] = fz;
94
95 // Pose features from current pose (character-relative).
96 AnimPose cur;
97 cur.copyFrom(&pose_);
98 cur.computeWorld(skeleton_);
99 const int root = database_->getRootBone();
100 const float rootX = cur.getWorldPositionX(root);
101 const float rootZ = cur.getWorldPositionZ(root);
102 // Estimate current yaw from root rotation.
103 const float qy = cur.getWorldRotationY(root);
104 const float qw = cur.getWorldRotationW(root);
105 const float yaw = std::atan2(2.f * (qw * qy), 1.f - 2.f * (qy * qy));
106 const float c2 = std::cos(yaw);
107 const float s2 = std::sin(yaw);
108
109 int base = 10;
110 for (int i = 0; i < database_->getFeatureBoneCount(); ++i) {
111 const int b = database_->getFeatureBone(i);
112 const float dx = cur.getWorldPositionX(b) - rootX;
113 const float dz = cur.getWorldPositionZ(b) - rootZ;
114 const float lx = dx * c2 + dz * s2;
115 const float lz = -dx * s2 + dz * c2;
116 query[static_cast<size_t>(base)] = lx;
117 query[static_cast<size_t>(base + 1)] = cur.getWorldPositionY(b);
118 query[static_cast<size_t>(base + 2)] = lz;
119 base += 3;
120 }
121 database_->normalizeFeature(query);
122}

124float MotionMatcher::cost(const std::vector<float>& query, const std::vector<float>& cand, float upperBound) const {
125 const int n = static_cast<int>(query.size());
126 float sum = 0.f;
127 for (int i = 0; i < n; ++i) {
128 const float d = query[static_cast<size_t>(i)] - cand[static_cast<size_t>(i)];
129 float w = poseWeight_;
130 if (i < 2)
131 w = velWeight_;
132 else if (i < 10)
133 w = trajWeight_;
134 sum += w * d * d;
135 if (sum >= upperBound) return sum;
136 }
137 return sum;
138}

150void MotionMatcher::search() {
151 if (!database_->isBaked()) throw Exception("MotionMatcher.search: database not baked");
152 if (database_->getFrameCount() == 0) {
153 throw Exception("MotionMatcher.search: empty database");
154 }
155
156 std::vector<float> query;
157 buildQuery(query);
158
159 float bestCost = std::numeric_limits<float>::infinity();
160 int best = 0;
161 for (int i = 0; i < database_->getFrameCount(); ++i) {
162 if (matchedFrame_ >= 0 && std::abs(i - matchedFrame_) <= ignoreRadius_) {
163 // Still allow if different clip continues — but prefer diversity slightly.
164 // Skip only exact-neighborhood of current match when already playing.
165 if (playing_ && database_->getFrameClipIndex(i) == getMatchedClipIndex()) {
166 continue;
167 }
168 }
169 const auto& f = database_->frameAt(i);
170 const float c = cost(query, f.feature, bestCost);
171 if (c < bestCost) {
172 bestCost = c;
173 best = i;
174 }
175 }
176
177 // If everything skipped, fall back to exhaustive.
178 if (!std::isfinite(bestCost)) {
179 for (int i = 0; i < database_->getFrameCount(); ++i) {
180 const auto& f = database_->frameAt(i);
181 const float c = cost(query, f.feature, bestCost);
182 if (c < bestCost) {
183 bestCost = c;
184 best = i;
185 }
186 }
187 }
188
189 lastCost_ = bestCost;
190 if (best != matchedFrame_) {
191 fromPose_.copyFrom(&pose_);
192 matchedFrame_ = best;
193 matchedTime_ = database_->getFrameTime(best);
194 sampleMatched(&matchedPose_);
195 if (blendTime_ > 1e-6f && playing_) {
196 blending_ = true;
197 blendElapsed_ = 0.f;
198 } else {
199 pose_.copyFrom(&matchedPose_);
200 blending_ = false;
201 }
202 playing_ = true;
203 }
204}

206void MotionMatcher::updateUnchecked(float dt) {
207 if (dt < 0.f) throw Exception("MotionMatcher.update: dt must be >= 0");
208 if (!database_->isBaked()) return;
209
210 if (!playing_) {
211 search();
212 searchTimer_ = 0.f;
213 } else {
214 searchTimer_ += dt;
215 if (searchTimer_ >= searchInterval_) {
216 searchTimer_ = 0.f;
217 search();
218 }
219 }
220
221 if (matchedFrame_ >= 0) {
222 matchedTime_ += dt;
223 sampleMatched(&matchedPose_);
224 }
225
226 if (blending_) {
227 blendElapsed_ += dt;
228 float t = blendTime_ > 1e-8f ? blendElapsed_ / blendTime_ : 1.f;
229 if (t >= 1.f) {
230 blending_ = false;
231 pose_.copyFrom(&matchedPose_);
232 } else {
233 pose_.blendFrom(&fromPose_, &matchedPose_, t);
234 }
235 } else if (matchedFrame_ >= 0) {
236 pose_.copyFrom(&matchedPose_);
237 }
238}

240eve::Result<void> MotionMatcher::advance(const eve::SimulationStep& step) {
241 auto seconds = detail::secondsForStep(step, hasLastTick_, lastTick_, "MotionMatcher");
242 if (!seconds) return eve::Result<void>::failure(seconds.status());
243 updateUnchecked(std::move(seconds).takeValue());
244 lastTick_ = step.tick;
245 hasLastTick_ = true;
246 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
247}
