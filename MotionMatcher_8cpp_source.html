107void MotionMatcher::buildQuery(std::vector<float>& query) const {
108 if (database_->hasFeatureLayout()) {
109 buildSchemaQuery(query);
110 return;
111 }
112 if (!database_->isBaked()) throw Exception("MotionMatcher: database not baked");
113 if (database_->hasLocomotionFeatures()) {
114 buildLocomotionQuery(query);
115 return;
116 }
117 const int n = database_->getFeatureSize();
118 query.assign(static_cast<size_t>(n), 0.f);
119
120 // Character-space desired velocity (assume current facing = desiredYaw for query).
121 const float cs = std::cos(desiredYaw_);
122 const float sn = std::sin(desiredYaw_);
123 query[0] = desiredVelX_ * cs - desiredVelZ_ * sn;
124 query[1] = desiredVelX_ * sn + desiredVelZ_ * cs;
125
126 // Desired trajectory: integrate constant velocity for 0.33/0.66/1.0s in char space.
127 const float horizons[3] = {0.33f, 0.66f, 1.0f};
128 for (int h = 0; h < 3; ++h) {
129 query[static_cast<size_t>(2 + h * 2)] = query[0] * horizons[h];
130 query[static_cast<size_t>(2 + h * 2 + 1)] = query[1] * horizons[h];
131 if (hasTrajectory_) {
132 query[static_cast<size_t>(2 + h * 2)] = trajectory_[h].x * cs - trajectory_[h].z * sn;
133 query[static_cast<size_t>(3 + h * 2)] = trajectory_[h].x * sn + trajectory_[h].z * cs;
134 }
135 }
136 // Facing at +1s: same as desired (relative facing 0 → forward +Z in char space).
137 float fx, fz;
138 yawToForward(hasTrajectory_ ? trajectory_[2].yaw - desiredYaw_ : 0.f, fx, fz);
139 query[8] = fx;
140 query[9] = fz;
141
142 // Pose features from current pose (character-relative).
143 AnimPose cur;
144 cur.copyFrom(hasQueryPose_ ? &queryPose_ : (playing_ ? &matchedPose_ : &pose_));
145 cur.computeWorld(skeleton_);
146 const int root = database_->getRootBone();
147 const float rootX = cur.getWorldPositionX(root);
148 const float rootZ = cur.getWorldPositionZ(root);
149 // Estimate current yaw from root rotation.
150 const float qy = cur.getWorldRotationY(root);
151 const float qw = cur.getWorldRotationW(root);
152 const float yaw = std::atan2(2.f * (qw * qy), 1.f - 2.f * (qy * qy));
153 const float c2 = std::cos(yaw);
154 const float s2 = std::sin(yaw);
155
156 int base = 10;
157 for (int i = 0; i < database_->getFeatureBoneCount(); ++i) {
158 const int b = database_->getFeatureBone(i);
159 const float dx = cur.getWorldPositionX(b) - rootX;
160 const float dz = cur.getWorldPositionZ(b) - rootZ;
161 const float lx = dx * c2 - dz * s2;
162 const float lz = dx * s2 + dz * c2;
163 query[static_cast<size_t>(base)] = lx;
164 query[static_cast<size_t>(base + 1)] = cur.getWorldPositionY(b);
165 query[static_cast<size_t>(base + 2)] = lz;
166 base += 3;
167 }
168 database_->normalizeFeature(query);
169}

171float MotionMatcher::cost(const std::vector<float>& query, const std::vector<float>& cand, float upperBound) const {
172 const int n = static_cast<int>(query.size());
173 float sum = 0.f;
174 const auto* schema = database_->schema_.get();
175 const float schemaWeight = schema ? schema->weightSums[0] * poseWeight_ + schema->weightSums[1] * velWeight_ +
176 schema->weightSums[2] * trajWeight_
177 : 0.f;
178 for (int i = 0; i < n; ++i) {
179 const float d = query[static_cast<size_t>(i)] - cand[static_cast<size_t>(i)];
180 float w = poseWeight_;
181 if (schema) {
182 w = schema->sources[i] == MotionFeatureSource::Pose
183 ? poseWeight_
184 : (schema->kinds[i] == MotionFeatureKind::Velocity ? velWeight_ : trajWeight_);
185 w *= schema->weights[i] / std::max(1e-8f, schemaWeight);
186 } else if (database_->hasLocomotionFeatures()) {
187 w = i < detail::locomotionPoseOffset ? trajWeight_ : poseWeight_;
188 if (i < 2 || (i >= 12 && i < 14) || (i >= 16 && i < 19)) w = velWeight_;
189 w *= detail::locomotionWeight(i);
190 w /= std::max(1e-8f, 8.5f * velWeight_ + 10.6f * trajWeight_ + 5.f * poseWeight_);
191 } else if (i < 2)
192 w = velWeight_;
193 else if (i < 10)
194 w = trajWeight_;
195 sum += w * d * d;
196 if (sum >= upperBound) return sum;
197 }
198 return sum;
199}

211void MotionMatcher::search() {
212 if (!database_->isBaked()) throw Exception("MotionMatcher.search: database not baked");
213 if (database_->getFrameCount() == 0) {
214 throw Exception("MotionMatcher.search: empty database");
215 }
216
217 std::vector<float> query;
218 buildQuery(query);
219
220 // Compare against the live playhead, not the original entry frame.
221 // Equal-cost searches must keep advancing instead of restarting a blend.
222 int continuation = -1;
223 float nearest = std::numeric_limits<float>::infinity();
224 const int currentClip = getMatchedClipIndex();
225 const auto* currentAsset = currentClip >= 0 ? database_->getClip(currentClip) : nullptr;
226 const bool ended = currentAsset && !currentAsset->getLoop() && matchedTime_ >= currentAsset->getDuration();
227 if (playing_ && currentClip >= 0 && !ended) {
228 const auto* clip = database_->getClip(currentClip);
229 const float time = clip->wrapTime(matchedTime_);
230 const int count = filtered_ ? static_cast<int>(candidateFrames_.size()) : database_->getFrameCount();
231 for (int candidate = 0; candidate < count; ++candidate) {
232 const int i = filtered_ ? candidateFrames_[candidate] : candidate;
233 if (database_->getFrameClipIndex(i) != currentClip) continue;
234 float distance = std::abs(database_->getFrameTime(i) - time);
235 if (clip->getLoop()) distance = std::min(distance, clip->getDuration() - distance);
236 if (distance < nearest) {
237 nearest = distance;
238 continuation = i;
239 }
240 }
241 // Candidate ranges are a live search context (for example a Chooser
242 // changing from grounded to airborne). A different valid interval in
243 // the same clip must not make an out-of-range playhead eligible as a
244 // continuing pose. At the bake rate, a live time is at most half a
245 // sample from its representing frame.
246 if (filtered_ && continuation >= 0) {
247 const float rate = currentAsset->getSampleRate() > 0.f ? currentAsset->getSampleRate() : 30.f;
248 if (nearest > .5f / rate + 1e-5f) continuation = -1;
249 }
250 }
251 float bestCost = std::numeric_limits<float>::infinity();
252 if (continuation >= 0) {
253 auto feature = database_->frameAt(continuation).feature;
254 // Continuing has no pose discontinuity. Use the evaluated playhead
255 // pose rather than its nearest baked sample (which is quantized).
256 if (database_->schema_) {
257 for (std::size_t i = 0; i < query.size(); ++i)
258 if (database_->schema_->queries[i] == MotionFeatureQuery::Continuing) query[i] = feature[i];
259 } else if (database_->hasLocomotionFeatures()) {
260 // Authored pose channels use UseContinuingPose; trajectory channels
261 // still use the character's actual/predicted movement.
262 std::copy(feature.begin() + detail::locomotionPoseOffset, feature.end(),
263 query.begin() + detail::locomotionPoseOffset);
264 } else
265 std::copy(query.begin() + 10, query.end(), feature.begin() + 10);
266 bestCost = cost(query, feature, bestCost);
267 if (filtered_) bestCost += candidateBias_[continuation] + candidateContinuingBias_[continuation];
268 }
269 int best = continuation;
270 float switchCost = bestCost;
271 const int searchCount = filtered_ ? static_cast<int>(candidateFrames_.size()) : database_->getFrameCount();
272 for (int candidate = 0; candidate < searchCount; ++candidate) {
273 const int i = filtered_ ? candidateFrames_[candidate] : candidate;
274 // Authored protected phases can be played through, but never jumped into.
275 if (filtered_ && transitionBlocked_[i]) continue;
276 if (std::any_of(poseHistory_.begin(), poseHistory_.end(), [i](const auto& item) { return item.first == i; }))
277 continue;
278 // DisableReselection belongs to the current asset, including its end.
279 // An expired one-shot cannot become its own fresh candidate again.
280 if (playing_ && filtered_ && disableReselection_[currentClip] && database_->getFrameClipIndex(i) == currentClip)
281 continue;
282 if (continuation >= 0 && database_->getFrameClipIndex(i) == currentClip) {
283 if (filtered_ && disableReselection_[currentClip]) continue;
284 const auto* clip = database_->getClip(currentClip);
285 float distance = std::abs(database_->getFrameTime(i) - clip->wrapTime(matchedTime_));
286 if (clip->getLoop()) distance = std::min(distance, clip->getDuration() - distance);
287 const float rate = clip->getSampleRate() > 0.f ? clip->getSampleRate() : 30.f;
288 if (distance <= static_cast<float>(ignoreRadius_) / rate + 1e-5f) continue;
289 }
290 const float bias = filtered_ ? candidateBias_[i] : 0.f;
291 const float c = cost(query, database_->frameAt(i).feature, switchCost - bias) + bias;
292 if (c + 1e-5f < switchCost) {
293 bestCost = c;
294 switchCost = c;
295 best = i;
296 }
297 }
298 if (best == continuation && continuation >= 0) {
299 matchedFrame_ = continuation;
300 lastCost_ = bestCost;
301 updatePlayRate(continuation);
302 return;
303 }
304
305 // A database containing only an exhausted, non-reselectable one-shot holds
306 // its terminal pose. It remains searchable after the caller changes candidates.
307 if (best < 0) return;
308 lastCost_ = bestCost;
309 updatePlayRate(best);
310 if (best != matchedFrame_ || ended) {
311 matchedFrame_ = best;
312 matchedTime_ = database_->getFrameTime(best);
313 sampleMatched(&matchedPose_);
314 if (blendTime_ > 1e-6f && playing_) {
315 AnimPose previousTarget;
316 database_->getClip(getMatchedClipIndex())
317 ->sample(matchedTime_ - inertia_->historyInterval * playRate_, &previousTarget, skeleton_);
318 inertia_->begin(pose_, matchedPose_, previousTarget, blendTime_);
319 blending_ = true;
320 blendElapsed_ = 0.f;
321 } else {
322 pose_.copyFrom(&matchedPose_);
323 blending_ = false;
324 }
325 playing_ = true;
326 }
327}

329void MotionMatcher::updateUnchecked(float dt) {
330 if (dt < 0.f) throw Exception("MotionMatcher.update: dt must be >= 0");
331 if (!database_->isBaked()) return;
332
333 if (!playing_) {
334 search();
335 searchTimer_ = 0.f;
336 } else {
337 searchTimer_ += dt;
338 const auto* clip = matchedFrame_ >= 0 ? database_->getClip(getMatchedClipIndex()) : nullptr;
339 const bool ended = clip && !clip->getLoop() && matchedTime_ >= clip->getDuration();
340 if (searchTimer_ >= searchInterval_ || ended) {
341 searchTimer_ = 0.f;
342 search();
343 }
344 }
345
346 inertia_->remember(pose_, dt);
347 if (matchedFrame_ >= 0) {
348 matchedTime_ += dt * playRate_;
349 const auto* clip = database_->getClip(getMatchedClipIndex());
350 if (!clip->getLoop()) matchedTime_ = std::min(matchedTime_, clip->getDuration());
351 sampleMatched(&matchedPose_);
352 }
353
354 if (blending_) {
355 blendElapsed_ += dt;
356 if (blendElapsed_ >= inertia_->duration) {
357 blending_ = false;
358 pose_.copyFrom(&matchedPose_);
359 } else {
360 inertia_->apply(matchedPose_, blendElapsed_, pose_);
361 }
362 } else if (matchedFrame_ >= 0) {
363 pose_.copyFrom(&matchedPose_);
364 }
365 updatePoseHistory(dt);
366}

368void MotionMatcher::updatePoseHistory(float dt) {
369 if (poseReselectHistory_ <= 0.f || dt <= 0.f || matchedFrame_ < 0) return;
370 for (auto& item : poseHistory_) item.second += dt;
371 std::erase_if(poseHistory_, [&](const auto& item) { return item.second > poseReselectHistory_; });
372 const int clipIndex = getMatchedClipIndex();
373 const auto* clip = database_->getClip(clipIndex);
374 const float time = clip->wrapTime(matchedTime_);
375 int nearestFrame = -1;
376 float nearest = std::numeric_limits<float>::infinity();
377 const int count = filtered_ ? static_cast<int>(candidateFrames_.size()) : database_->getFrameCount();
378 for (int candidate = 0; candidate < count; ++candidate) {
379 const int i = filtered_ ? candidateFrames_[candidate] : candidate;
380 if (database_->getFrameClipIndex(i) != clipIndex) continue;
381 float distance = std::abs(database_->getFrameTime(i) - time);
382 if (clip->getLoop()) distance = std::min(distance, clip->getDuration() - distance);
383 if (distance < nearest) {
384 nearest = distance;
385 nearestFrame = i;
386 }
387 }
388 if (nearestFrame < 0) return;
389 auto found = std::find_if(poseHistory_.begin(), poseHistory_.end(),
390 [nearestFrame](const auto& item) { return item.first == nearestFrame; });
391 if (found == poseHistory_.end())
392 poseHistory_.push_back({nearestFrame, 0.f});
393 else
394 found->second = 0.f;
395}

419eve::Result<void> MotionMatcher::setLocomotionQuery(const AnimPose& current, const AnimPose& previous,
420 float elapsedSeconds,
421 std::span<const MotionLocomotionSample> samples) {
422 auto finitePose = [](const AnimPose& pose) {
423 for (int i = 0; i < pose.getBoneCount(); ++i) {
424 const auto& t = pose.local(i);
425 for (float v : {t.px, t.py, t.pz, t.qx, t.qy, t.qz, t.qw, t.sx, t.sy, t.sz})
426 if (!std::isfinite(v)) return false;
427 }
428 return true;
429 };
430 if (!database_->hasLocomotionFeatures() || current.getBoneCount() != pose_.getBoneCount() ||
431 previous.getBoneCount() != pose_.getBoneCount() || !std::isfinite(elapsedSeconds) || elapsedSeconds <= 0.f ||
432 elapsedSeconds > 1.f || samples.size() != 5 || !finitePose(current) || !finitePose(previous) ||
433 std::any_of(samples.begin(), samples.end(), [](const auto& s) {
434 for (float v : {s.x, s.y, s.z, s.vx, s.vy, s.vz, s.yaw})
435 if (!std::isfinite(v)) return true;
436 return false;
437 }))
438 return eve::Result<void>::failure(eve::Diagnostic::error(
439 eve::DiagnosticCode::InvalidArgument,
440 "locomotion query requires the locomotion layout, finite matching poses, five samples, and dt in (0,1]",
441 "locomotionQuery", {}, "animation"));
442 AnimPose nextCurrent, nextPrevious;
443 nextCurrent.copyFrom(&current);
444 nextPrevious.copyFrom(&previous);
445 queryPose_ = std::move(nextCurrent);
446 previousQueryPose_ = std::move(nextPrevious);
447 std::copy(samples.begin(), samples.end(), locomotionTrajectory_.begin());
448 queryPoseInterval_ = elapsedSeconds;
449 hasLocomotionQuery_ = true;
450 hasQueryPose_ = true;
451 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
452}

454void MotionMatcher::buildLocomotionQuery(std::vector<float>& query) const {
455 if (!hasLocomotionQuery_) throw Exception("MotionMatcher: configure a locomotion query before searching");
456 query.assign(detail::locomotionFeatureCount, 0.f);
457 detail::encodeLocomotionTrajectory(std::span(query), locomotionTrajectory_, desiredYaw_);
458 AnimPose current, previous;
459 current.copyFrom(&queryPose_);
460 previous.copyFrom(&previousQueryPose_);
461 current.computeWorld(skeleton_);
462 previous.computeWorld(skeleton_);
463 std::array<TransformTRS, 4> now{}, past{};
464 now[0] = current.world(database_->getRootBone());
465 past[0] = previous.world(database_->getRootBone());
466 for (int i = 0; i < 3; ++i) {
467 const int bone = database_->getFeatureBone(i);
468 now[i + 1] = current.world(bone);
469 past[i + 1] = previous.world(bone);
470 }
471 detail::encodeLocomotionPose(query, now, past, queryPoseInterval_);
472 database_->normalizeFeature(query);
473}

478eve::Result<void> MotionMatcher::setFeatureQuery(const AnimPose& current, const AnimPose& previous, float dt,
479 std::span<const MotionFeatureTrajectorySample> samples,
480 std::span<const MotionFeatureCurveSample> curves) {
481 auto fail = [](const char* message) {
482 return eve::Result<void>::failure(
483 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, message, "featureQuery", {}, "animation"));
484 };
485 if (!database_->schema_ || !database_->isBaked() || !std::isfinite(dt) || dt <= 0.f || dt > 1.f ||
486 current.getBoneCount() != skeleton_->getBoneCount() || previous.getBoneCount() != skeleton_->getBoneCount() ||
487 samples.size() > 256 || curves.size() > 256)
488 return fail("feature query requires a baked variable layout, matching poses and finite dt in (0,1]");
489 for (const auto* pose : {&current, &previous})
490 for (int i = 0; i < pose->getBoneCount(); ++i) {
491 const auto& t = pose->local(i);
492 for (float v : {t.px, t.py, t.pz, t.qx, t.qy, t.qz, t.qw, t.sx, t.sy, t.sz})
493 if (!std::isfinite(v)) return fail("feature query poses must be finite");
494 }
495 for (std::size_t i = 0; i < samples.size(); ++i) {
496 const auto& s = samples[i];
497 for (float v : {s.seconds, s.x, s.y, s.z, s.vx, s.vy, s.vz, s.yaw})
498 if (!std::isfinite(v)) return fail("trajectory samples must be finite");
499 for (std::size_t j = 0; j < i; ++j)
500 if (std::abs(s.seconds - samples[j].seconds) < 1e-6f) return fail("trajectory sample times must be unique");
501 }
502 const auto& schema = *database_->schema_;
503 for (std::size_t i = 0; i < curves.size(); ++i) {
504 const auto& c = curves[i];
505 if (c.curve.empty() || c.curve.size() > 4096 || !std::isfinite(c.seconds) || !std::isfinite(c.value))
506 return fail("curve queries require bounded names and finite times/values");
507 for (std::size_t j = 0; j < i; ++j)
508 if (c.curve == curves[j].curve && std::abs(c.seconds - curves[j].seconds) < 1e-6f)
509 return fail("curve query names and times must be unique");
510 }
511 std::vector<float> scalarValues(schema.layout.channels.size(), 0.f);
512 for (std::size_t i = 0; i < schema.layout.channels.size(); ++i) {
513 const auto& c = schema.layout.channels[i];
514 if (c.kind != MotionFeatureKind::Curve) continue;
515 auto sample = std::find_if(curves.begin(), curves.end(), [&](const auto& s) {
516 return s.curve == c.curve && std::abs(s.seconds - c.sampleTime) < 1e-6f;
517 });
518 if (sample == curves.end()) return fail("curve query does not cover a configured channel/time");
519 scalarValues[i] = sample->value;
520 }
521 std::vector<const MotionFeatureTrajectorySample*> selected;
522 for (const auto& c : schema.layout.channels) {
523 const MotionFeatureTrajectorySample* sample = nullptr;
524 if (c.source == MotionFeatureSource::Trajectory) {
525 for (const auto& s : samples)
526 if (std::abs(s.seconds - c.sampleTime) < 1e-6f) {
527 sample = &s;
528 break;
529 }
530 if (!sample) return fail("trajectory query does not cover a configured sample time");
531 }
532 selected.push_back(sample);
533 }
534 AnimPose now, past;
535 now.copyFrom(&current);
536 past.copyFrom(&previous);
537 now.computeWorld(skeleton_);
538 past.computeWorld(skeleton_);
539 const auto& root = now.world(database_->getRootBone());
540 const auto& previousRoot = past.world(database_->getRootBone());
541 TransformTRS character;
542 character.qy = std::sin(desiredYaw_ * .5f);
543 character.qw = std::cos(desiredYaw_ * .5f);
544 std::vector<float> query(schema.dimension, 0.f);
545 for (std::size_t i = 0; i < schema.layout.channels.size(); ++i) {
546 const auto& c = schema.layout.channels[i];
547 std::array<float, 3> value{};
548 if (c.kind == MotionFeatureKind::Curve)
549 value[0] = scalarValues[i];
550 else if (const auto* sample = selected[i]) {
551 if (c.kind == MotionFeatureKind::Position)
552 value = detail::schemaRotate({sample->x, sample->y, sample->z}, character, true);
553 else if (c.kind == MotionFeatureKind::Velocity)
554 value = detail::schemaRotate({sample->vx, sample->vy, sample->vz}, character, true);
555 else {
556 TransformTRS rotation;
557 rotation.qy = std::sin(sample->yaw * .5f);
558 rotation.qw = std::cos(sample->yaw * .5f);
559 value = detail::schemaHeading(rotation, character, c.headingAxis);
560 }
561 } else {
562 const auto& bone = now.world(c.bone);
563 if (c.kind == MotionFeatureKind::Position)
564 value = detail::schemaDifference(bone, now.world(c.origin), root);
565 else if (c.kind == MotionFeatureKind::Heading)
566 value = detail::schemaHeading(bone, root, c.headingAxis);
567 else if (c.characterSpaceVelocity) {
568 value = detail::schemaDifference(bone, now.world(c.origin), root);
569 const auto old = detail::schemaDifference(past.world(c.bone), past.world(c.origin), previousRoot);
570 for (int axis = 0; axis < 3; ++axis) value[axis] = (value[axis] - old[axis]) / dt;
571 } else {
572 value = detail::schemaDifference(bone, past.world(c.bone), root);
573 for (float& v : value) v /= dt;
574 }
575 }
576 detail::schemaEncode(std::span(query).subspan(schema.offsets[i]), c, value,
577 schema.layout.normalizationLengthScale);
578 }
579 for (float v : query)
580 if (!std::isfinite(v)) return fail("encoded feature query exceeds finite range");
581 const float queryTrajectorySpeed = database_->schemaTrajectorySpeed(query);
582 database_->normalizeFeature(query);
583 for (float v : query)
584 if (!std::isfinite(v)) return fail("normalized feature query exceeds finite range");
585 schemaQuery_.swap(query);
586 queryTrajectorySpeed_ = queryTrajectorySpeed;
587 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
588}

598eve::Result<int> MotionMatcher::setCandidateRanges(std::span<const MotionSearchRange> ranges) {
599 auto fail = [](const char* message) {
600 return eve::Result<int>::failure(
601 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, message, "candidateRanges", {}, "animation"));
602 };
603 if (!database_->isBaked() || ranges.empty() || ranges.size() > 100000)
604 return fail("candidate selection requires a baked database and bounded nonempty ranges");
605 std::vector<std::vector<MotionSearchRange>> perClip(database_->getClipCount());
606 std::size_t intervalCount = 0;
607 for (const auto& r : ranges) {
608 if (r.clipIndex < 0 || r.clipIndex >= database_->getClipCount() || !std::isfinite(r.start) ||
609 !std::isfinite(r.end) || !std::isfinite(r.costBias) || r.start < 0.f || r.end < r.start ||
610 r.end > database_->getClip(r.clipIndex)->getDuration() + 0.001f)
611 return fail("invalid candidate clip, time interval, or cost bias");
612 intervalCount += r.transitionBlocks.size() + r.costOverrides.size() + r.continuingCostOverrides.size();
613 if (intervalCount > 100000) return fail("too many candidate metadata intervals");
614 for (const auto& block : r.transitionBlocks)
615 if (!std::isfinite(block.start) || !std::isfinite(block.end) || block.start < 0.f ||
616 block.end <= block.start)
617 return fail("invalid transition block interval");
618 if (!std::isfinite(r.continuingCostBias)) return fail("invalid continuing cost bias");
619 for (const auto* overrides : {&r.costOverrides, &r.continuingCostOverrides})
620 for (const auto& cost : *overrides)
621 if (!std::isfinite(cost.start) || !std::isfinite(cost.end) || !std::isfinite(cost.costBias) ||
622 cost.start < 0.f || cost.end <= cost.start)
623 return fail("invalid cost override interval");
624 perClip[r.clipIndex].push_back(r);
625 }
626 const int count = database_->getFrameCount();
627 std::vector<int> selected;
628 std::vector<float> bias(count, 0.f);
629 std::vector<float> continuingBias(count, 0.f);
630 std::vector<unsigned char> mask(count, 0), noReselect(database_->getClipCount(), 0);
631 std::vector<unsigned char> blocked(count, 0);
632 for (int clipIndex = 0; clipIndex < database_->getClipCount(); ++clipIndex) {
633 const auto& clipRanges = perClip[clipIndex];
634 if (clipRanges.empty()) continue;
635 const int begin = database_->clipFrameOffsets_[static_cast<std::size_t>(clipIndex)];
636 const int end = database_->clipFrameOffsets_[static_cast<std::size_t>(clipIndex + 1)];
637 for (int i = begin; i < end; ++i) {
638 const auto& frame = database_->frameAt(i);
639 float best = std::numeric_limits<float>::infinity();
640 for (const auto& r : clipRanges) {
641 for (const auto& block : r.transitionBlocks)
642 if (frame.time >= block.start && frame.time < block.end) blocked[i] = 1;
643 if (frame.time + 1e-5f < r.start || frame.time - 1e-5f > r.end) continue;
644 float rangeBias = r.costBias;
645 for (const auto& cost : r.costOverrides)
646 if (frame.time >= cost.start && frame.time < cost.end) rangeBias = cost.costBias;
647 float rangeContinuingBias = r.continuingCostBias;
648 for (const auto& cost : r.continuingCostOverrides)
649 if (frame.time >= cost.start && frame.time < cost.end) rangeContinuingBias = cost.costBias;
650 if (rangeBias < best) {
651 best = rangeBias;
652 continuingBias[i] = rangeContinuingBias;
653 }
654 if (r.disableReselection) noReselect[frame.clipIndex] = 1;
655 }
656 if (!std::isfinite(best)) continue;
657 selected.push_back(i);
658 mask[i] = 1;
659 bias[i] = best;
660 }
661 }
662 if (selected.empty()) return fail("candidate selection contains no sampled frames");
663 candidateFrames_.swap(selected);
664 candidateBias_.swap(bias);
665 candidateContinuingBias_.swap(continuingBias);
666 candidateMask_.swap(mask);
667 transitionBlocked_.swap(blocked);
668 disableReselection_.swap(noReselect);
669 filtered_ = true;
670 return eve::Result<int>::success(static_cast<int>(candidateFrames_.size()),
671 eve::Status::success(eve::StatusCode::Applied));
672}
