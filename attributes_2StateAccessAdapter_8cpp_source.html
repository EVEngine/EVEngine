46eve::Result<eve::MutationReceipt> AttributeSetStateAdapter::apply(std::span<const eve::StateMutation> mutations,
47 const eve::MutationContext& context) {
48 for (std::size_t index = 0; index < mutations.size(); ++index) {
49 const auto& mutation = mutations[index];
50 if (mutation.persistent)
51 return failure(eve::DiagnosticCode::Unsupported,
52 "AttributeSet mutations are volatile; use StatePatch for persistence",
53 "mutations[" + std::to_string(index) + "]");
54 if (!owns(mutation.subject) || mutation.key.empty())
55 return failure(eve::DiagnosticCode::InvalidArgument,
56 "attribute mutation requires this set's subject and a key",
57 "mutations[" + std::to_string(index) + "]");
58 if (mutation.kind != eve::MutationKind::Set && mutation.kind != eve::MutationKind::AddNumber)
59 return failure(eve::DiagnosticCode::Unsupported, "AttributeSet supports Set and AddNumber only",
60 "mutations[" + std::to_string(index) + "]");
61 double ignored = 0.0;
62 if (!numeric(mutation.value, ignored))
63 return failure(eve::DiagnosticCode::InvalidArgument, "attribute mutation value must be a finite number",
64 "mutations[" + std::to_string(index) + "].value");
65 }
66
67 AttributeSet candidate = attributes_;
68 std::size_t changed = 0;
69 for (const auto& mutation : mutations) {
70 const double amount =
71 mutation.value.isInt64() ? static_cast<double>(mutation.value.asInt()) : mutation.value.asDouble();
72 const double before = candidate.getFinal(mutation.key);
73 if (mutation.kind == eve::MutationKind::Set)
74 candidate.setBase(mutation.key, amount);
75 else
76 candidate.modifyBase(mutation.key, amount);
77 const double after = candidate.getFinal(mutation.key);
78 if (before != after || !attributes_.has(mutation.key)) ++changed;
79 }
80 attributes_ = std::move(candidate);
81 return eve::Result<eve::MutationReceipt>::success(
82 eve::MutationReceipt{context.transactionId, changed},
83 eve::Status::success(changed == 0 ? eve::StatusCode::NoOp : eve::StatusCode::Applied));
84}
