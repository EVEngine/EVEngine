21float SurfaceDropletSimulation::dropletRadius(float volume) const {
22 const float theta = glm::radians(std::clamp(params_.contactAngleDegrees, 5.f, 175.f));
23 const float cosine = std::cos(theta);
24 const float sine = std::max(1e-3f, std::sin(theta));
25 constexpr float pi = 3.14159265358979323846f;
26 const float factor = pi * (2.f - 3.f * cosine + cosine * cosine * cosine) / (3.f * sine * sine * sine);
27 return std::cbrt(std::max(0.f, volume) / std::max(1e-6f, factor));
28}

30bool SurfaceDropletSimulation::addDroplet(const SurfaceLocation& location, float volume,
31 const glm::vec3& relativeVelocity) {
32 if (!binding_ || !binding_->isValid() || location.triangle >= uint32_t(binding_->triangleCount()) || volume <= 0.f)
33 return false;
34 SurfaceDroplet droplet;
35 droplet.id = nextDropletId_++;
36 droplet.location = location;
37 droplet.volume = volume;
38 const SurfaceSample sample = binding_->evaluate(location, 0.f);
39 droplet.relativeVelocity = relativeVelocity - sample.normal * glm::dot(relativeVelocity, sample.normal);
40 droplets_.push_back(droplet);
41 return true;
42}

44void SurfaceDropletSimulation::step(float dt) {
45 detached_.clear();
46 if (!binding_ || !binding_->isValid() || dt <= 0.f || (droplets_.empty() && airborne_.empty())) return;
47 float remaining = dt;
48 while (remaining > 1e-7f) {
49 const float substep = std::min(remaining, 0.05f);
50 stepSubstep(substep, dt);
51 remaining -= substep;
52 }

55void SurfaceDropletSimulation::stepSubstep(float dt, float poseDt) {
56 std::vector<SurfaceDroplet> attached;
57 attached.reserve(droplets_.size());
58 for (SurfaceDroplet droplet : droplets_) {
59 const SurfaceLocation previousLocation = droplet.location;
60 const SurfaceSample previousSample = binding_->evaluate(previousLocation, poseDt);
61 const SurfaceSample sample = binding_->evaluate(droplet.location, poseDt);
62 glm::vec3 surfaceAcceleration(0.f);
63 if (droplet.hasPreviousSurfaceVelocity)
64 surfaceAcceleration = (sample.velocity - droplet.previousSurfaceVelocity) / poseDt;
65
66 const glm::vec3 relativeAcceleration = params_.gravity - surfaceAcceleration;
67 const float outwardAcceleration = glm::dot(relativeAcceleration, sample.normal);
68 if (outwardAcceleration > std::max(0.f, params_.adhesionAcceleration)) {
69 detached_.push_back({sample.position, sample.velocity + droplet.relativeVelocity, droplet.volume});
70 airborne_.push_back({sample.position, sample.velocity + droplet.relativeVelocity, droplet.volume});
71 continue;
72 }
73
74 glm::vec3 tangentAcceleration =
75 relativeAcceleration - sample.normal * glm::dot(relativeAcceleration, sample.normal);
76 droplet.relativeVelocity += tangentAcceleration * dt;
77 droplet.relativeVelocity *= std::exp(-std::max(0.f, params_.friction) * dt);
78 const float speed2 = glm::length2(droplet.relativeVelocity);
79 const float maxSpeed = std::max(0.f, params_.maxSpeed);
80 if (maxSpeed > 0.f && speed2 > maxSpeed * maxSpeed) droplet.relativeVelocity *= maxSpeed / std::sqrt(speed2);
81
82 const SurfaceWalkResult walk = binding_->walkAcrossSurface(droplet.location, droplet.relativeVelocity * dt,
83 std::max(0, params_.maxCrossings));
84 if (!walk.valid) continue;
85 droplet.location = walk.location;
86 if (walk.reachedBoundary) {
87 const SurfaceSample edge = binding_->evaluate(walk.location, poseDt);
88 detached_.push_back({edge.position, edge.velocity + droplet.relativeVelocity, droplet.volume});
89 airborne_.push_back({edge.position, edge.velocity + droplet.relativeVelocity, droplet.volume, 0.f});
90 continue;
91 }
92
93 const SurfaceSample end = binding_->evaluate(droplet.location, poseDt);
94 droplet.relativeVelocity -= end.normal * glm::dot(droplet.relativeVelocity, end.normal);
95 droplet.previousSurfaceVelocity = end.velocity;
96 droplet.hasPreviousSurfaceVelocity = true;
97 if (wetness_) {
98 const float distance = glm::distance(previousSample.position, end.position);
99 wetness_->deposit(previousLocation, distance * params_.trailDeposition * droplet.volume);
100 wetness_->deposit(droplet.location, distance * params_.trailDeposition * droplet.volume);
101 }
102 attached.push_back(droplet);
103 }
104 droplets_.swap(attached);
105
106 // Merge overlapping spherical caps without depending on mesh UV seams.
107 for (size_t i = 0; i < droplets_.size(); ++i) {
108 const SurfaceSample a = binding_->evaluate(droplets_[i].location, poseDt);
109 for (size_t j = i + 1u; j < droplets_.size();) {
110 const SurfaceSample b = binding_->evaluate(droplets_[j].location, poseDt);
111 const float mergeDistance =
112 params_.mergeRadiusScale * (dropletRadius(droplets_[i].volume) + dropletRadius(droplets_[j].volume));
113 if (glm::distance2(a.position, b.position) > mergeDistance * mergeDistance) {
114 ++j;
115 continue;
116 }
117 const float combined = droplets_[i].volume + droplets_[j].volume;
118 droplets_[i].relativeVelocity = (droplets_[i].relativeVelocity * droplets_[i].volume +
119 droplets_[j].relativeVelocity * droplets_[j].volume) /
120 combined;
121 droplets_[i].volume = combined;
122 droplets_.erase(droplets_.begin() + std::ptrdiff_t(j));
123 }
124 }
125
126 std::vector<AirborneDroplet> flying;
127 flying.reserve(airborne_.size());
128 for (AirborneDroplet drop : airborne_) {
129 drop.age += dt;
130 drop.velocity += params_.gravity * dt;
131 drop.velocity *= std::exp(-std::max(0.f, params_.airDrag) * dt);
132 drop.position += drop.velocity * dt;
133 SurfaceLocation hit;
134 if (drop.age > 0.08f && binding_->project(drop.position, std::max(0.f, params_.reattachDistance), hit)) {
135 const SurfaceSample surface = binding_->evaluate(hit, poseDt);
136 if (glm::dot(drop.velocity - surface.velocity, surface.normal) <= 0.f) {
137 addDroplet(hit, drop.volume, drop.velocity - surface.velocity);
138 if (wetness_) wetness_->deposit(hit, drop.volume * 0.15f);
139 continue;
140 }
141 }
142 flying.push_back(drop);
143 }
144 airborne_.swap(flying);
145}

353void SurfaceFluidRenderData::update(const FluidSurfaceBinding& binding, const SurfaceDropletSimulation& simulation,
354 const SurfaceWetnessField* wetness, const SurfaceFluidRenderParams& params) {
355 droplets_.clear();
356 droplets_.reserve(simulation.droplets().size());
357 const float stretch = std::max(0.f, params.velocityStretch);
358 const float maximumAspect = std::max(1.f, params.maxAspectRatio);
359 const float theta = glm::radians(std::clamp(simulation.params().contactAngleDegrees, 5.f, 175.f));
360
361 for (const SurfaceDroplet& droplet : simulation.droplets()) {
362 const SurfaceSample surface = binding.evaluate(droplet.location, 0.f);
363 const float radius = simulation.dropletRadius(droplet.volume);
364 const float speed = glm::length(droplet.relativeVelocity);
365 const float aspect = std::clamp(1.f + speed * stretch, 1.f, maximumAspect);
366 const float axisScale = std::sqrt(aspect);
367
368 glm::vec3 direction = droplet.relativeVelocity;
369 direction -= surface.normal * glm::dot(direction, surface.normal);
370 if (glm::length2(direction) > 1e-12f)
371 direction = glm::normalize(direction);
372 else
373 direction = surface.tangent;
374 glm::vec3 across = glm::cross(surface.normal, direction);
375 if (glm::length2(across) > 1e-12f)
376 across = glm::normalize(across);
377 else
378 across = surface.bitangent;
379
380 SurfaceDropletRenderInstance instance;
381 instance.id = droplet.id;
382 instance.position = surface.position + surface.normal * std::max(0.f, params.surfaceOffset);
383 instance.normal = surface.normal;
384 instance.majorAxis = direction * (radius * axisScale);
385 instance.minorAxis = across * (radius / axisScale);
386 instance.velocity = surface.velocity + droplet.relativeVelocity;
387 instance.capHeight = radius * std::tan(theta * 0.5f);
388 instance.wetness = wetness ? std::clamp(wetness->sample(droplet.location), 0.f, 1.f) : 0.f;
389 droplets_.push_back(instance);
390 }
391}

393WetSurfaceMaterialSample SurfaceFluidRenderData::evaluateMaterial(float wetness,
394 const WetSurfaceMaterialParams& params) {
395 WetSurfaceMaterialSample sample;
396 sample.wetness = std::clamp(wetness, 0.f, 1.f);
397 const float response = sample.wetness * sample.wetness * (3.f - 2.f * sample.wetness);
398 sample.roughness = glm::mix(params.dryRoughness, params.wetRoughness, response);
399 sample.specular = glm::mix(params.drySpecular, params.wetSpecular, response);
400 sample.darkening = std::max(0.f, params.wetDarkening) * response;
401 sample.normalStrength = std::max(0.f, params.normalStrength) * response;
402 return sample;
403}
