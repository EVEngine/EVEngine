243 const float phase = static_cast<float>(seed);
244 return std::sin(value.x * 1.13f + value.y * 0.71f + value.z * 0.47f + phase * 0.123f) * 0.5f +

246 std::sin(-value.x * 0.83f + value.y * 0.29f + value.z * 1.67f + phase * 0.037f) * 0.2f;
247}

249void applyVertexOperation(const auto& node, Vec3& position, Vec3& normal) {
250 constexpr float radiansPerDegree = 0.01745329251994329577f;
251 if (node.operation == "deform.transform") {
252 const float sx = parameter(node, "scaleX", 1.f);
253 const float sy = parameter(node, "scaleY", 1.f);
254 const float sz = parameter(node, "scaleZ", 1.f);
255 position = {position.x * sx, position.y * sy, position.z * sz};
256 normal = {normal.x / sx, normal.y / sy, normal.z / sz};
257 position = rotateZ(rotateY(rotateX(position, parameter(node, "pitch", 0.f) * radiansPerDegree),
258 parameter(node, "yaw", 0.f) * radiansPerDegree),
259 parameter(node, "roll", 0.f) * radiansPerDegree);
260 normal = rotateZ(rotateY(rotateX(normal, parameter(node, "pitch", 0.f) * radiansPerDegree),
261 parameter(node, "yaw", 0.f) * radiansPerDegree),
262 parameter(node, "roll", 0.f) * radiansPerDegree);
263 position.x += parameter(node, "x", 0.f);
264 position.y += parameter(node, "y", 0.f);
265 position.z += parameter(node, "z", 0.f);
266 } else if (node.operation == "deform.twist" || node.operation == "deform.bend") {
267 const std::string axis = stringParameter(node, "axis", "y");
268 const float extent = std::max(std::abs(parameter(node, "extent", 1.f)), 1e-6f);
269 const float along = axis == "x" ? position.x : axis == "z" ? position.z : position.y;
270 const float angle = parameter(node, "angle", 0.f) * radiansPerDegree * along / extent;
271 if (node.operation == "deform.twist") {
272 if (axis == "x") {
273 position = rotateX(position, angle);
274 normal = rotateX(normal, angle);
275 } else if (axis == "z") {
276 position = rotateZ(position, angle);
277 normal = rotateZ(normal, angle);
278 } else {
279 position = rotateY(position, angle);
280 normal = rotateY(normal, angle);
281 }
282 } else {
283 if (axis == "x") {
284 position = rotateZ(position, angle);
285 normal = rotateZ(normal, angle);
286 } else if (axis == "z") {
287 position = rotateX(position, -angle);
288 normal = rotateX(normal, -angle);
289 } else {
290 position = rotateZ(position, -angle);
291 normal = rotateZ(normal, -angle);
292 }
293 }
294 } else if (node.operation == "deform.noise") {
295 const float frequency = parameter(node, "frequency", 1.f);
296 const Vec3 sample{position.x * frequency, position.y * frequency, position.z * frequency};
297 const int seed = intParameter(node, "seed", 1);
298 const float amplitude = parameter(node, "amplitude", 0.1f);
299 position.x += coherentNoise(sample, seed) * amplitude;
300 position.y += coherentNoise({sample.y, sample.z, sample.x}, seed + 1013) * amplitude;
301 position.z += coherentNoise({sample.z, sample.x, sample.y}, seed + 2029) * amplitude;
302 } else if (node.operation == "deform.soundReact") {
303 const std::string axis = stringParameter(node, "axis", "y");
304 const float coordinate = axis == "x" ? position.x : (axis == "z" ? position.z : position.y);
305 const float level = parameter(node, "level", 0.f);
306 const float threshold = parameter(node, "threshold", 0.f);
307 const float response = std::max(level - threshold, 0.f) * parameter(node, "strength", 1.f);
308 const float wave = 0.5f + 0.5f * std::sin(coordinate * parameter(node, "frequency", 1.f) * 6.283185307f +
309 parameter(node, "phase", 0.f));
310 Vec3 radial = position;
311 if (axis == "x") radial.x = 0.f;
312 if (axis == "y") radial.y = 0.f;
313 if (axis == "z") radial.z = 0.f;
314 const float radialLength = std::sqrt(radial.x * radial.x + radial.y * radial.y + radial.z * radial.z);
315 if (radialLength > 1e-7f) {
316 const float amount = response * wave / radialLength;
317 position.x += radial.x * amount;
318 position.y += radial.y * amount;
319 position.z += radial.z * amount;
320 }
321 } else if (node.operation == "deform.radial") {
322 const Vec3 center{parameter(node, "x", 0.f), parameter(node, "y", 0.f), parameter(node, "z", 0.f)};
323 Vec3 delta{position.x - center.x, position.y - center.y, position.z - center.z};
324 const float distance = std::sqrt(delta.x * delta.x + delta.y * delta.y + delta.z * delta.z);
325 const float radius = std::max(parameter(node, "radius", 1.f), 1e-6f);
326 if (distance < radius && distance > 1e-7f) {
327 const float falloff = std::max(parameter(node, "falloff", 1.f), 0.01f);
328 const float amount = parameter(node, "strength", 0.1f) * std::pow(1.f - distance / radius, falloff);
329 position.x += delta.x / distance * amount;
330 position.y += delta.y / distance * amount;
331 position.z += delta.z / distance * amount;
332 }
333 } else if (node.operation == "deform.effector") {
334 const int pointCount = std::clamp(intParameter(node, "pointCount", 1), 1, 4);
335 const float density = std::max(parameter(node, "density", 1.f), 0.01f);
336 const float multiplier = parameter(node, "multiplier", 1.f);
337 const Vec3 source = position;
338 for (int point = 0; point < pointCount; ++point) {
339 const std::string prefix = "p" + std::to_string(point);
340 const Vec3 center{parameter(node, prefix + "x", 0.f), parameter(node, prefix + "y", 0.f),
341 parameter(node, prefix + "z", 0.f)};
342 const Vec3 offset{parameter(node, prefix + "dx", 0.f), parameter(node, prefix + "dy", 1.f),
343 parameter(node, prefix + "dz", 0.f)};
344 const float dx = source.x - center.x;
345 const float dy = source.y - center.y;
346 const float dz = source.z - center.z;
347 const float distance = std::sqrt(dx * dx + dy * dy + dz * dz);
348 const float radius = parameter(node, prefix + "radius", 1.f);
349 if (distance >= radius) continue;
350 const float influence =
351 std::pow(1.f - distance / radius, density) * parameter(node, prefix + "weight", 1.f) * multiplier;
352 position.x += offset.x * influence;
353 position.y += offset.y * influence;
354 position.z += offset.z * influence;
355 }
356 } else if (node.operation == "deform.spherify") {
357 const Vec3 center{parameter(node, "x", 0.f), parameter(node, "y", 0.f), parameter(node, "z", 0.f)};
358 Vec3 delta{position.x - center.x, position.y - center.y, position.z - center.z};
359 const float length = std::sqrt(delta.x * delta.x + delta.y * delta.y + delta.z * delta.z);
360 if (length > 1e-7f) {
361 const float radius = std::max(parameter(node, "radius", 1.f), 0.f);
362 const float weight = std::clamp(parameter(node, "weight", 1.f), 0.f, 1.f);
363 const Vec3 target{center.x + delta.x * radius / length, center.y + delta.y * radius / length,
364 center.z + delta.z * radius / length};
365 position = {std::lerp(position.x, target.x, weight), std::lerp(position.y, target.y, weight),
366 std::lerp(position.z, target.z, weight)};
367 normal = {delta.x / length, delta.y / length, delta.z / length};
368 }
369 }
370 const float length = std::sqrt(normal.x * normal.x + normal.y * normal.y + normal.z * normal.z);
371 if (length > 1e-7f) normal = {normal.x / length, normal.y / length, normal.z / length};
372}

374void applyMaskedVertexOperation(const auto& node, Vec3& position, Vec3& normal) {
375 const Vec3 originalPosition = position;
376 const Vec3 originalNormal = normal;
377 applyVertexOperation(node, position, normal);
378 const float radius = parameter(node, "maskRadius", 0.f);
379 if (radius <= 0.f) return;
380 const float dx = originalPosition.x - parameter(node, "maskX", 0.f);
381 const float dy = originalPosition.y - parameter(node, "maskY", 0.f);
382 const float dz = originalPosition.z - parameter(node, "maskZ", 0.f);
383 const float distance = std::sqrt(dx * dx + dy * dy + dz * dz);
384 float weight =
385 std::pow(std::clamp(1.f - distance / radius, 0.f, 1.f), std::max(parameter(node, "maskFalloff", 1.f), 0.01f));
386 if (intParameter(node, "maskInvert", 0) != 0) weight = 1.f - weight;
387 position = {std::lerp(originalPosition.x, position.x, weight), std::lerp(originalPosition.y, position.y, weight),
388 std::lerp(originalPosition.z, position.z, weight)};
389 normal = {std::lerp(originalNormal.x, normal.x, weight), std::lerp(originalNormal.y, normal.y, weight),
390 std::lerp(originalNormal.z, normal.z, weight)};
391 const float normalLength = std::sqrt(normal.x * normal.x + normal.y * normal.y + normal.z * normal.z);
392 if (normalLength > 1e-7f) normal = {normal.x / normalLength, normal.y / normalLength, normal.z / normalLength};
393}

403 return graphFailure(DiagnosticCode::InvariantViolation, "mesh index is outside the vertex stream",
404 std::string(path));
405 return Result<void>::success();

408void recalculateNormals(MeshBuild& mesh) {
409 auto& normals = mesh.normals();
410 std::fill(normals.begin(), normals.end(), 0.f);
411 const auto& positions = mesh.positions();
412 const auto& indices = mesh.indices();
413 for (std::size_t i = 0; i + 2 < indices.size(); i += 3) {
414 const std::size_t a = std::size_t(indices[i]) * 3u;
415 const std::size_t b = std::size_t(indices[i + 1]) * 3u;
416 const std::size_t c = std::size_t(indices[i + 2]) * 3u;
417 const Vec3 ab{positions[b] - positions[a], positions[b + 1] - positions[a + 1],
418 positions[b + 2] - positions[a + 2]};
419 const Vec3 ac{positions[c] - positions[a], positions[c + 1] - positions[a + 1],
420 positions[c + 2] - positions[a + 2]};
421 const Vec3 cross{ab.y * ac.z - ab.z * ac.y, ab.z * ac.x - ab.x * ac.z, ab.x * ac.y - ab.y * ac.x};
422 for (const auto vertex : {a, b, c}) {
423 normals[vertex] += cross.x;
424 normals[vertex + 1] += cross.y;
425 normals[vertex + 2] += cross.z;
426 }
427 }
428 for (std::size_t i = 0; i + 2 < normals.size(); i += 3) {
429 const float length =
430 std::sqrt(normals[i] * normals[i] + normals[i + 1] * normals[i + 1] + normals[i + 2] * normals[i + 2]);
431 if (length > 1e-7f) {
432 normals[i] /= length;
433 normals[i + 1] /= length;
434 normals[i + 2] /= length;
435 }
436 }
437}

439Result<MeshBuild> smoothMesh(const MeshBuild& input, float strength, int iterations) {
440 MeshBuild output = input;
441 const int count = output.getVertexCount();
442 std::vector<std::unordered_set<std::uint32_t>> neighbors(static_cast<std::size_t>(count));
443 const auto& indices = output.indices();
444 for (std::size_t i = 0; i + 2 < indices.size(); i += 3) {
445 const std::array<std::uint32_t, 3> tri{indices[i], indices[i + 1], indices[i + 2]};
446 for (int corner = 0; corner < 3; ++corner) {
447 neighbors[tri[corner]].insert(tri[(corner + 1) % 3]);
448 neighbors[tri[corner]].insert(tri[(corner + 2) % 3]);
449 }
450 }
451 strength = std::clamp(strength, 0.f, 1.f);
452 iterations = std::clamp(iterations, 1, 64);
453 std::vector<float> next = output.positions();
454 for (int iteration = 0; iteration < iterations; ++iteration) {
455 const auto current = output.positions();
456 for (int vertex = 0; vertex < count; ++vertex) {
457 const auto& adjacent = neighbors[static_cast<std::size_t>(vertex)];
458 if (adjacent.empty()) continue;
459 Vec3 average;
460 for (const auto neighbor : adjacent) {
461 average.x += current[std::size_t(neighbor) * 3u];
462 average.y += current[std::size_t(neighbor) * 3u + 1u];
463 average.z += current[std::size_t(neighbor) * 3u + 2u];
464 }
465 const float divisor = float(adjacent.size());
466 const std::size_t base = std::size_t(vertex) * 3u;
467 next[base] = std::lerp(current[base], average.x / divisor, strength);
468 next[base + 1] = std::lerp(current[base + 1], average.y / divisor, strength);
469 next[base + 2] = std::lerp(current[base + 2], average.z / divisor, strength);
470 }
471 output.positions() = next;
472 }
473 recalculateNormals(output);
474 return Result<MeshBuild>::success(std::move(output));
475}

481Result<MeshBuild> angularBendMesh(const MeshBuild& input, const auto& node) {
482 MeshBuild output = input;
483 auto& positions = output.positions();
484 if (positions.empty()) return Result<MeshBuild>::success(std::move(output));
485 const std::string axis = stringParameter(node, "axis", "y");
486 const int component = axis == "x" ? 0 : axis == "z" ? 2 : 1;
487 float minimum = positions[static_cast<std::size_t>(component)];
488 float maximum = minimum;
489 for (std::size_t i = static_cast<std::size_t>(component); i < positions.size(); i += 3u) {
490 minimum = std::min(minimum, positions[i]);
491 maximum = std::max(maximum, positions[i]);
492 }
493 const float extent = std::max(maximum - minimum, 1e-6f);
494 const float direction = parameter(node, "direction", 0.f) * 0.01745329251994329577f;
495 const Vec3 bendAxis{std::sin(direction), 0.f, std::cos(direction)};
496 const float angle = parameter(node, "angle", 0.f) * 0.01745329251994329577f;
497 auto& normals = output.normals();
498 for (int vertex = 0; vertex < output.getVertexCount(); ++vertex) {
499 const std::size_t base = static_cast<std::size_t>(vertex) * 3u;
500 const float t = (positions[base + static_cast<std::size_t>(component)] - minimum) / extent;
501 const float theta = angle * t;
502 const float c = std::cos(theta), s = std::sin(theta);
503 auto rotate = [&](Vec3 value) {
504 const float dot = value.x * bendAxis.x + value.y * bendAxis.y + value.z * bendAxis.z;
505 const Vec3 cross{bendAxis.y * value.z - bendAxis.z * value.y, bendAxis.z * value.x - bendAxis.x * value.z,
506 bendAxis.x * value.y - bendAxis.y * value.x};
507 return Vec3{value.x * c + cross.x * s + bendAxis.x * dot * (1.f - c),
508 value.y * c + cross.y * s + bendAxis.y * dot * (1.f - c),
509 value.z * c + cross.z * s + bendAxis.z * dot * (1.f - c)};
510 };
511 const Vec3 position = rotate({positions[base], positions[base + 1], positions[base + 2]});
512 const Vec3 normal = rotate({normals[base], normals[base + 1], normals[base + 2]});
513 positions[base] = position.x;
514 positions[base + 1] = position.y;
515 positions[base + 2] = position.z;
516 normals[base] = normal.x;
517 normals[base + 1] = normal.y;
518 normals[base + 2] = normal.z;
519 }
520 return Result<MeshBuild>::success(std::move(output));
521}

523Result<MeshBuild> splineMesh(const MeshBuild& input, const auto& node) {
524 MeshBuild output = input;
525 auto& positions = output.positions();
526 if (positions.empty()) return Result<MeshBuild>::success(std::move(output));
527 const std::string axis = stringParameter(node, "axis", "y");
528 const int component = axis == "x" ? 0 : axis == "z" ? 2 : 1;
529 float minimum = positions[static_cast<std::size_t>(component)];
530 float maximum = minimum;
531 for (std::size_t i = static_cast<std::size_t>(component); i < positions.size(); i += 3u) {
532 minimum = std::min(minimum, positions[i]);
533 maximum = std::max(maximum, positions[i]);
534 }
535 const float extent = std::max(maximum - minimum, 1e-6f);
536 std::array<Vec3, 4> controls;
537 for (int i = 0; i < 4; ++i) {
538 const std::string prefix = "c" + std::to_string(i);
539 controls[static_cast<std::size_t>(i)] = {parameter(node, prefix + "x", 0.f), parameter(node, prefix + "y", 0.f),
540 parameter(node, prefix + "z", 0.f)};
541 }
542 const float weight = std::clamp(parameter(node, "weight", 1.f), 0.f, 1.f);
543 auto bezier = [&](float t) {
544 const float u = 1.f - t;
545 const std::array<float, 4> b{u * u * u, 3.f * u * u * t, 3.f * u * t * t, t * t * t};
546 Vec3 result;
547 for (int i = 0; i < 4; ++i) {
548 result.x += controls[static_cast<std::size_t>(i)].x * b[static_cast<std::size_t>(i)];
549 result.y += controls[static_cast<std::size_t>(i)].y * b[static_cast<std::size_t>(i)];
550 result.z += controls[static_cast<std::size_t>(i)].z * b[static_cast<std::size_t>(i)];
551 }
552 return result;
553 };
554 for (std::size_t base = 0; base < positions.size(); base += 3u) {
555 Vec3 source{positions[base], positions[base + 1], positions[base + 2]};
556 const float t = std::clamp((component == 0 ? source.x
557 : component == 2 ? source.z
558 : source.y) -
559 minimum,
560 0.f, extent) /
561 extent;
562 const Vec3 offset = bezier(t);
563 positions[base] = source.x + offset.x * weight;
564 positions[base + 1] = source.y + offset.y * weight;
565 positions[base + 2] = source.z + offset.z * weight;
566 }
567 recalculateNormals(output);
568 return Result<MeshBuild>::success(std::move(output));
569}

571Result<MeshBuild> splinePathMesh(const MeshBuild& input, const auto& node) {
572 MeshBuild output = input;
573 auto& positions = output.positions();
574 if (positions.empty()) return Result<MeshBuild>::success(std::move(output));
575 const std::string axis = stringParameter(node, "axis", "y");
576 const int component = axis == "x" ? 0 : axis == "z" ? 2 : 1;
577 float minimum = positions[static_cast<std::size_t>(component)];
578 float maximum = minimum;
579 for (std::size_t i = static_cast<std::size_t>(component); i < positions.size(); i += 3u) {
580 minimum = std::min(minimum, positions[i]);
581 maximum = std::max(maximum, positions[i]);
582 }
583 const float extent = std::max(maximum - minimum, 1e-6f);
584 const float weight = std::clamp(parameter(node, "weight", 1.f), 0.f, 1.f);
585 const float scale = parameter(node, "scale", 1.f);
586 const float roll = parameter(node, "roll", 0.f) * 0.01745329251994329577f;
587 for (std::size_t base = 0; base < positions.size(); base += 3u) {
588 const Vec3 source{positions[base], positions[base + 1], positions[base + 2]};
589 const float along = component == 0 ? source.x : component == 2 ? source.z : source.y;
590 auto sampled = node.splinePath.evaluateResult((along - minimum) / extent);
591 if (!sampled.ok()) return Result<MeshBuild>::failure(sampled.status());
592 const auto& sample = sampled.value();
593 Vec3 tangent{sample.tangentX, sample.tangentY, sample.tangentZ};
594 Vec3 side, up;
595 if (tangent.z < -0.999999f) {
596 side = {0.f, -1.f, 0.f};
597 up = {-1.f, 0.f, 0.f};
598 } else {
599 const float a = 1.f / (1.f + tangent.z);
600 const float b = -tangent.x * tangent.y * a;
601 side = {1.f - tangent.x * tangent.x * a, b, -tangent.x};
602 up = {b, 1.f - tangent.y * tangent.y * a, -tangent.y};
603 }
604 const float c = std::cos(roll), s = std::sin(roll);
605 const Vec3 rolledSide{side.x * c + up.x * s, side.y * c + up.y * s, side.z * c + up.z * s};
606 const Vec3 rolledUp{up.x * c - side.x * s, up.y * c - side.y * s, up.z * c - side.z * s};
607 const float localA = component == 0 ? source.y : source.x;
608 const float localB = component == 2 ? source.y : source.z;
609 const Vec3 target{sample.x + (rolledSide.x * localA + rolledUp.x * localB) * scale,
610 sample.y + (rolledSide.y * localA + rolledUp.y * localB) * scale,
611 sample.z + (rolledSide.z * localA + rolledUp.z * localB) * scale};
612 positions[base] = std::lerp(source.x, target.x, weight);
613 positions[base + 1] = std::lerp(source.y, target.y, weight);
614 positions[base + 2] = std::lerp(source.z, target.z, weight);
615 }
616 recalculateNormals(output);
617 return Result<MeshBuild>::success(std::move(output));
618}

620Result<MeshBuild> splineTubeMesh(const auto& node) {
621 if (node.splinePath.chunkCount() > 1) {
622 MeshBuild output;
623 for (int chunk = 0; chunk < node.splinePath.chunkCount(); ++chunk) {
624 auto path = node.splinePath.chunkPathResult(chunk);
625 if (!path.ok()) return Result<MeshBuild>::failure(path.status());
626 auto chunkNode = node;
627 chunkNode.splinePath = std::move(path).takeValue();
628 auto mesh = splineTubeMesh(chunkNode);
629 if (!mesh.ok()) return mesh;
630 if (!output.appendTransformed(&mesh.value(), 0.f, 0.f, 0.f, 0.f, 1.f, 1.f, 1.f))
631 return Result<MeshBuild>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
632 "failed to compose spline tube chunks", {}, {},
633 "procgen.meshModifierGraph"));
634 }
635 output.setMeta("generator", "mesh.splineTube");
636 output.setMeta("chunks", std::to_string(node.splinePath.chunkCount()));
637 return Result<MeshBuild>::success(std::move(output));
638 }
639 const float radius = parameter(node, "radius", 0.5f);
640 const int pathSegments = intParameter(node, "pathSegments", 32);
641 const int radialSegments = intParameter(node, "radialSegments", 12);
642 const float roll = parameter(node, "roll", 0.f);
643 const bool capped = intParameter(node, "cap", 1) != 0 && !node.splinePath.isClosed();
644 const int ringStride = radialSegments + 1;
645 const int ringCount = pathSegments + 1;
646 MeshBuild output;
647 output.reserve(ringCount * ringStride + (capped ? 2 * (ringStride + 1) : 0),
648 pathSegments * radialSegments * 6 + (capped ? radialSegments * 6 : 0));
649
650 constexpr float tau = 6.28318530717958647692f;
651 auto sampledFrames = node.splinePath.sampleFramesResult(pathSegments, true, roll, 24);
652 if (!sampledFrames.ok()) return Result<MeshBuild>::failure(sampledFrames.status());
653 const auto& frames = sampledFrames.value();
654 for (int ring = 0; ring < ringCount; ++ring) {
655 const float u = static_cast<float>(ring) / static_cast<float>(pathSegments);
656 const auto& frame = frames[static_cast<std::size_t>(ring)];
657 const auto& sample = frame.sample;
658 const Vec3 rolledSide{frame.sideX, frame.sideY, frame.sideZ};
659 const Vec3 rolledUp{frame.upX, frame.upY, frame.upZ};
660 for (int radial = 0; radial <= radialSegments; ++radial) {
661 const float v = static_cast<float>(radial) / static_cast<float>(radialSegments);
662 const float angle = v * tau;
663 const float c = std::cos(angle), s = std::sin(angle);
664 const Vec3 offset{rolledSide.x * c * sample.scaleX + rolledUp.x * s * sample.scaleY,
665 rolledSide.y * c * sample.scaleX + rolledUp.y * s * sample.scaleY,
666 rolledSide.z * c * sample.scaleX + rolledUp.z * s * sample.scaleY};
667 const float inverseX = 1.f / sample.scaleX, inverseY = 1.f / sample.scaleY;
668 Vec3 normal{rolledSide.x * c * inverseX + rolledUp.x * s * inverseY,
669 rolledSide.y * c * inverseX + rolledUp.y * s * inverseY,
670 rolledSide.z * c * inverseX + rolledUp.z * s * inverseY};
671 const float normalLength = std::sqrt(normal.x * normal.x + normal.y * normal.y + normal.z * normal.z);
672 normal = {normal.x / normalLength, normal.y / normalLength, normal.z / normalLength};
673 output.addVertex(sample.x + offset.x * radius, sample.y + offset.y * radius, sample.z + offset.z * radius,
674 normal.x, normal.y, normal.z, u, v);
675 }
676 }
677
678 output.setActiveGroup("tube");
679 for (int ring = 0; ring < pathSegments; ++ring) {
680 for (int radial = 0; radial < radialSegments; ++radial) {
681 const auto a = static_cast<std::uint32_t>(ring * ringStride + radial);
682 const auto b = a + static_cast<std::uint32_t>(ringStride);
683 output.addTriangle(a, a + 1u, b);
684 output.addTriangle(a + 1u, b + 1u, b);
685 }
686 }
687 if (capped) {
688 output.setActiveGroup("caps");
689 auto startSample = node.splinePath.evaluateResult(0.f);
690 auto endSample = node.splinePath.evaluateResult(1.f);
691 if (!startSample.ok()) return Result<MeshBuild>::failure(startSample.status());
692 if (!endSample.ok()) return Result<MeshBuild>::failure(endSample.status());
693 const Vec3 startNormal{-startSample.value().tangentX, -startSample.value().tangentY,
694 -startSample.value().tangentZ};
695 const Vec3 endNormal{endSample.value().tangentX, endSample.value().tangentY, endSample.value().tangentZ};
696 const auto startCenter = static_cast<std::uint32_t>(output.getVertexCount());
697 output.addVertex(startSample.value().x, startSample.value().y, startSample.value().z, startNormal.x,
698 startNormal.y, startNormal.z, 0.5f, 0.5f);
699 const auto startRim = static_cast<std::uint32_t>(output.getVertexCount());
700 for (int radial = 0; radial <= radialSegments; ++radial) {
701 const float angle = static_cast<float>(radial) / static_cast<float>(radialSegments) * tau;
702 output.addVertex(output.getPositionX(radial), output.getPositionY(radial), output.getPositionZ(radial),
703 startNormal.x, startNormal.y, startNormal.z, 0.5f + std::cos(angle) * 0.5f,
704 0.5f + std::sin(angle) * 0.5f);
705 }
706 const auto endCenter = static_cast<std::uint32_t>(output.getVertexCount());
707 output.addVertex(endSample.value().x, endSample.value().y, endSample.value().z, endSample.value().tangentX,
708 endSample.value().tangentY, endSample.value().tangentZ, 0.5f, 0.5f);
709 const auto lastRing = static_cast<std::uint32_t>(pathSegments * ringStride);
710 const auto endRim = static_cast<std::uint32_t>(output.getVertexCount());
711 for (int radial = 0; radial <= radialSegments; ++radial) {
712 const float angle = static_cast<float>(radial) / static_cast<float>(radialSegments) * tau;
713 output.addVertex(output.getPositionX(static_cast<int>(lastRing) + radial),
714 output.getPositionY(static_cast<int>(lastRing) + radial),
715 output.getPositionZ(static_cast<int>(lastRing) + radial), endNormal.x, endNormal.y,
716 endNormal.z, 0.5f + std::cos(angle) * 0.5f, 0.5f + std::sin(angle) * 0.5f);
717 }
718 for (int radial = 0; radial < radialSegments; ++radial) {
719 const auto r = static_cast<std::uint32_t>(radial);
720 output.addTriangle(startCenter, startRim + r + 1u, startRim + r);
721 output.addTriangle(endCenter, endRim + r, endRim + r + 1u);
722 }
723 }
724 output.setMeta("generator", "mesh.splineTube");
725 output.setMeta("pathSegments", std::to_string(pathSegments));
726 output.setMeta("radialSegments", std::to_string(radialSegments));
727 output.setMeta("closed", node.splinePath.isClosed() ? "1" : "0");
728 return Result<MeshBuild>::success(std::move(output));
729}

731Result<MeshBuild> splineRibbonMesh(const auto& node) {
732 if (node.splinePath.chunkCount() > 1) {
733 MeshBuild output;
734 for (int chunk = 0; chunk < node.splinePath.chunkCount(); ++chunk) {
735 auto path = node.splinePath.chunkPathResult(chunk);
736 if (!path.ok()) return Result<MeshBuild>::failure(path.status());
737 auto chunkNode = node;
738 chunkNode.splinePath = std::move(path).takeValue();
739 auto mesh = splineRibbonMesh(chunkNode);
740 if (!mesh.ok()) return mesh;
741 if (!output.appendTransformed(&mesh.value(), 0.f, 0.f, 0.f, 0.f, 1.f, 1.f, 1.f))
742 return Result<MeshBuild>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
743 "failed to compose spline ribbon chunks", {}, {},
744 "procgen.meshModifierGraph"));
745 }
746 output.setMeta("generator", "mesh.splineRibbon");
747 output.setMeta("chunks", std::to_string(node.splinePath.chunkCount()));
748 return Result<MeshBuild>::success(std::move(output));
749 }
750 const float width = parameter(node, "width", 2.f);
751 const float thickness = parameter(node, "thickness", 0.f);
752 const int pathSegments = intParameter(node, "pathSegments", 32);
753 const bool solid = thickness > 1e-7f;
754 const bool capped = solid && intParameter(node, "cap", 1) != 0 && !node.splinePath.isClosed();
755 auto frames = node.splinePath.sampleFramesResult(pathSegments, true, parameter(node, "roll", 0.f), 24);
756 if (!frames.ok()) return Result<MeshBuild>::failure(frames.status());
757
758 const int ringStride = solid ? 8 : 2;
759 MeshBuild output;
760 output.reserve((pathSegments + 1) * ringStride + (capped ? 8 : 0),
761 pathSegments * (solid ? 24 : 6) + (capped ? 12 : 0));
762 const float halfWidth = width * 0.5f;
763 for (int ring = 0; ring <= pathSegments; ++ring) {
764 const auto& frame = frames.value()[static_cast<std::size_t>(ring)];
765 const auto& p = frame.sample;
766 const Vec3 side{frame.sideX, frame.sideY, frame.sideZ};
767 const Vec3 up{frame.upX, frame.upY, frame.upZ};
768 const float u = static_cast<float>(ring) / static_cast<float>(pathSegments);
769 const float ringHalfWidth = halfWidth * p.scaleX;
770 const float ringThickness = thickness * p.scaleY;
771 const Vec3 left{p.x - side.x * ringHalfWidth, p.y - side.y * ringHalfWidth, p.z - side.z * ringHalfWidth};
772 const Vec3 right{p.x + side.x * ringHalfWidth, p.y + side.y * ringHalfWidth, p.z + side.z * ringHalfWidth};
773 output.addVertex(left.x, left.y, left.z, up.x, up.y, up.z, u, 0.f);
774 output.addVertex(right.x, right.y, right.z, up.x, up.y, up.z, u, 1.f);
775 if (!solid) continue;
776 const Vec3 down{-up.x, -up.y, -up.z};
777 const Vec3 bottomLeft{left.x - up.x * ringThickness, left.y - up.y * ringThickness,
778 left.z - up.z * ringThickness};
779 const Vec3 bottomRight{right.x - up.x * ringThickness, right.y - up.y * ringThickness,
780 right.z - up.z * ringThickness};
781 output.addVertex(bottomRight.x, bottomRight.y, bottomRight.z, down.x, down.y, down.z, u, 0.f);
782 output.addVertex(bottomLeft.x, bottomLeft.y, bottomLeft.z, down.x, down.y, down.z, u, 1.f);
783 output.addVertex(left.x, left.y, left.z, -side.x, -side.y, -side.z, u, 0.f);
784 output.addVertex(bottomLeft.x, bottomLeft.y, bottomLeft.z, -side.x, -side.y, -side.z, u, 1.f);
785 output.addVertex(right.x, right.y, right.z, side.x, side.y, side.z, u, 0.f);
786 output.addVertex(bottomRight.x, bottomRight.y, bottomRight.z, side.x, side.y, side.z, u, 1.f);
787 }
788
789 output.setActiveGroup("surface");
790 for (int ring = 0; ring < pathSegments; ++ring) {
791 const auto a = static_cast<std::uint32_t>(ring * ringStride);
792 const auto b = a + static_cast<std::uint32_t>(ringStride);
793 output.addTriangle(a, b, a + 1u);
794 output.addTriangle(a + 1u, b, b + 1u);
795 }
796 if (solid) {
797 output.setActiveGroup("bottom");
798 for (int ring = 0; ring < pathSegments; ++ring) {
799 const auto a = static_cast<std::uint32_t>(ring * ringStride + 2);
800 const auto b = a + static_cast<std::uint32_t>(ringStride);
801 output.addTriangle(a, b, a + 1u);
802 output.addTriangle(a + 1u, b, b + 1u);
803 }
804 output.setActiveGroup("sides");
805 for (int ring = 0; ring < pathSegments; ++ring) {
806 const auto base = static_cast<std::uint32_t>(ring * ringStride);
807 const auto next = base + static_cast<std::uint32_t>(ringStride);
808 output.addTriangle(base + 4u, base + 5u, next + 4u);
809 output.addTriangle(base + 5u, next + 5u, next + 4u);
810 output.addTriangle(base + 6u, next + 6u, base + 7u);
811 output.addTriangle(base + 7u, next + 6u, next + 7u);
812 }
813 if (capped) {
814 output.setActiveGroup("caps");
815 for (int end = 0; end < 2; ++end) {
816 const int sourceRing = end == 0 ? 0 : pathSegments;
817 const auto& frame = frames.value()[static_cast<std::size_t>(sourceRing)];
818 const Vec3 normal{(end == 0 ? -1.f : 1.f) * frame.sample.tangentX,
819 (end == 0 ? -1.f : 1.f) * frame.sample.tangentY,
820 (end == 0 ? -1.f : 1.f) * frame.sample.tangentZ};
821 const auto source = static_cast<std::uint32_t>(sourceRing * ringStride);
822 const auto base = static_cast<std::uint32_t>(output.getVertexCount());
823 for (const auto offset : {0u, 1u, 2u, 3u}) {
824 const auto vertex = source + offset;
825 output.addVertex(output.getPositionX(static_cast<int>(vertex)),
826 output.getPositionY(static_cast<int>(vertex)),
827 output.getPositionZ(static_cast<int>(vertex)), normal.x, normal.y, normal.z,
828 offset == 0u || offset == 3u ? 0.f : 1.f, offset < 2u ? 0.f : 1.f);
829 }
830 if (end == 0) {
831 output.addTriangle(base, base + 1u, base + 3u);
832 output.addTriangle(base + 1u, base + 2u, base + 3u);
833 } else {
834 output.addTriangle(base, base + 3u, base + 1u);
835 output.addTriangle(base + 1u, base + 3u, base + 2u);
836 }
837 }
838 }
839 }
840 output.setMeta("generator", "mesh.splineRibbon");
841 output.setMeta("pathSegments", std::to_string(pathSegments));
842 output.setMeta("closed", node.splinePath.isClosed() ? "1" : "0");
843 return Result<MeshBuild>::success(std::move(output));
844}

846float profileArea(const SplineProfile& profile) {
847 float area = 0.f;
848 for (std::size_t i = 0; i < profile.points.size(); ++i) {
849 const auto& a = profile.points[i];
850 const auto& b = profile.points[(i + 1u) % profile.points.size()];
851 area += a.x * b.y - b.x * a.y;
852 }
853 return area * 0.5f;

856float cross2(const SplineProfilePoint& a, const SplineProfilePoint& b, const SplineProfilePoint& c) {
857 return (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x);

863 return cross2(a, b, p) >= -epsilon && cross2(b, c, p) >= -epsilon && cross2(c, a, p) >= -epsilon;
864}

866Result<std::vector<std::uint32_t>> triangulateProfile(const SplineProfile& profile) {
867 std::vector<std::uint32_t> polygon(profile.points.size());
868 if (profileArea(profile) > 0.f)
869 for (std::size_t i = 0; i < polygon.size(); ++i) polygon[i] = static_cast<std::uint32_t>(i);
870 else
871 for (std::size_t i = 0; i < polygon.size(); ++i)
872 polygon[i] = static_cast<std::uint32_t>(polygon.size() - 1u - i);
873 std::vector<std::uint32_t> triangles;
874 while (polygon.size() > 3u) {
875 bool clipped = false;
876 for (std::size_t i = 0; i < polygon.size(); ++i) {
877 const auto previous = polygon[(i + polygon.size() - 1u) % polygon.size()];
878 const auto current = polygon[i];
879 const auto next = polygon[(i + 1u) % polygon.size()];
880 if (cross2(profile.points[previous], profile.points[current], profile.points[next]) <= 1e-7f) continue;
881 bool contains = false;
882 for (const auto candidate : polygon) {
883 if (candidate == previous || candidate == current || candidate == next) continue;
884 if (insideTriangle(profile.points[candidate], profile.points[previous], profile.points[current],
885 profile.points[next])) {
886 contains = true;
887 break;
888 }
889 }
890 if (contains) continue;
891 triangles.insert(triangles.end(), {previous, current, next});
892 polygon.erase(polygon.begin() + static_cast<std::ptrdiff_t>(i));
893 clipped = true;
894 break;
895 }
896 if (!clipped)
897 return Result<std::vector<std::uint32_t>>::failure(Diagnostic::error(
898 DiagnosticCode::InvalidArgument, "closed spline profile must be a simple non-degenerate polygon",
899 "profile", {}, "procgen.meshModifierGraph"));
900 }
901 triangles.insert(triangles.end(), {polygon[0], polygon[1], polygon[2]});
902 return Result<std::vector<std::uint32_t>>::success(std::move(triangles));
903}

905Result<MeshBuild> splineExtrudeMesh(const auto& node) {
906 if (node.splinePath.chunkCount() > 1) {
907 MeshBuild output;
908 for (int chunk = 0; chunk < node.splinePath.chunkCount(); ++chunk) {
909 auto path = node.splinePath.chunkPathResult(chunk);
910 if (!path.ok()) return Result<MeshBuild>::failure(path.status());
911 auto chunkNode = node;
912 chunkNode.splinePath = std::move(path).takeValue();
913 auto mesh = splineExtrudeMesh(chunkNode);
914 if (!mesh.ok()) return mesh;
915 if (!output.appendTransformed(&mesh.value(), 0.f, 0.f, 0.f, 0.f, 1.f, 1.f, 1.f))
916 return Result<MeshBuild>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
917 "failed to compose spline extrusion chunks", {}, {},
918 "procgen.meshModifierGraph"));
919 }
920 output.setMeta("generator", "mesh.splineExtrude");
921 output.setMeta("chunks", std::to_string(node.splinePath.chunkCount()));
922 return Result<MeshBuild>::success(std::move(output));
923 }
924 const int pathSegments = intParameter(node, "pathSegments", 32);
925 const bool capped = node.splineProfile.closed && intParameter(node, "cap", 1) != 0 && !node.splinePath.isClosed();
926 auto frames = node.splinePath.sampleFramesResult(pathSegments, true, parameter(node, "roll", 0.f), 24);
927 if (!frames.ok()) return Result<MeshBuild>::failure(frames.status());
928 const auto& profile = node.splineProfile;
929 const int pointCount = static_cast<int>(profile.points.size());
930 const int profileSegments = profile.closed ? pointCount : pointCount - 1;
931 const int ringStride = pointCount + (profile.closed ? 1 : 0);
932 MeshBuild output;
933 output.reserve((pathSegments + 1) * ringStride + (capped ? pointCount * 2 : 0),
934 pathSegments * profileSegments * 6 + (capped ? (pointCount - 2) * 6 : 0));
935 const float orientation = profile.closed && profileArea(profile) < 0.f ? -1.f : 1.f;
936 for (int ring = 0; ring <= pathSegments; ++ring) {
937 const auto& frame = frames.value()[static_cast<std::size_t>(ring)];
938 const auto& sample = frame.sample;
939 for (int profileIndex = 0; profileIndex < ringStride; ++profileIndex) {
940 const int i = profileIndex % pointCount;
941 const int previous = profile.closed ? (i + pointCount - 1) % pointCount : std::max(0, i - 1);
942 const int next = profile.closed ? (i + 1) % pointCount : std::min(pointCount - 1, i + 1);
943 const auto& p = profile.points[static_cast<std::size_t>(i)];
944 const auto& a = profile.points[static_cast<std::size_t>(previous)];
945 const auto& b = profile.points[static_cast<std::size_t>(next)];
946 float nx = orientation * (b.y - a.y), ny = orientation * (a.x - b.x);
947 const float inverseX = 1.f / sample.scaleX, inverseY = 1.f / sample.scaleY;
948 nx *= inverseX;
949 ny *= inverseY;
950 const float nl = std::sqrt(nx * nx + ny * ny);
951 nx /= nl;
952 ny /= nl;
953 output.addVertex(sample.x + frame.sideX * p.x * sample.scaleX + frame.upX * p.y * sample.scaleY,
954 sample.y + frame.sideY * p.x * sample.scaleX + frame.upY * p.y * sample.scaleY,
955 sample.z + frame.sideZ * p.x * sample.scaleX + frame.upZ * p.y * sample.scaleY,
956 frame.sideX * nx + frame.upX * ny, frame.sideY * nx + frame.upY * ny,
957 frame.sideZ * nx + frame.upZ * ny,
958 static_cast<float>(ring) / static_cast<float>(pathSegments),
959 static_cast<float>(profileIndex) / static_cast<float>(profileSegments));
960 }
961 }
962 output.setActiveGroup("profile");
963 for (int ring = 0; ring < pathSegments; ++ring)
964 for (int segment = 0; segment < profileSegments; ++segment) {
965 const auto a = static_cast<std::uint32_t>(ring * ringStride + segment);
966 const auto b = a + static_cast<std::uint32_t>(ringStride);
967 output.addTriangle(a, a + 1u, b);
968 output.addTriangle(a + 1u, b + 1u, b);
969 }
970 if (capped) {
971 auto triangles = triangulateProfile(profile);
972 if (!triangles.ok()) return Result<MeshBuild>::failure(triangles.status());
973 output.setActiveGroup("caps");
974 for (int end = 0; end < 2; ++end) {
975 const int ring = end == 0 ? 0 : pathSegments;
976 const auto& frame = frames.value()[static_cast<std::size_t>(ring)];
977 const auto base = static_cast<std::uint32_t>(output.getVertexCount());
978 for (int i = 0; i < pointCount; ++i) {
979 const int source = ring * ringStride + i;
980 const float direction = end == 0 ? -1.f : 1.f;
981 output.addVertex(output.getPositionX(source), output.getPositionY(source), output.getPositionZ(source),
982 frame.sample.tangentX * direction, frame.sample.tangentY * direction,
983 frame.sample.tangentZ * direction, profile.points[static_cast<std::size_t>(i)].x,
984 profile.points[static_cast<std::size_t>(i)].y);
985 }
986 for (std::size_t i = 0; i < triangles.value().size(); i += 3u) {
987 const auto a = base + triangles.value()[i], b = base + triangles.value()[i + 1u];
988 const auto c = base + triangles.value()[i + 2u];
989 if (end == 0)
990 output.addTriangle(a, c, b);
991 else
992 output.addTriangle(a, b, c);
993 }
994 }
995 }
996 output.setMeta("generator", "mesh.splineExtrude");
997 output.setMeta("profileClosed", profile.closed ? "1" : "0");
998 output.setMeta("closed", node.splinePath.isClosed() ? "1" : "0");
999 return Result<MeshBuild>::success(std::move(output));
1000}

1002bool rayTriangle(Vec3 origin, Vec3 direction, Vec3 a, Vec3 b, Vec3 c, float& distance, Vec3& normal) {
1003 const Vec3 edge1{b.x - a.x, b.y - a.y, b.z - a.z};
1004 const Vec3 edge2{c.x - a.x, c.y - a.y, c.z - a.z};
1005 const Vec3 p{direction.y * edge2.z - direction.z * edge2.y, direction.z * edge2.x - direction.x * edge2.z,
1006 direction.x * edge2.y - direction.y * edge2.x};
1007 const float determinant = edge1.x * p.x + edge1.y * p.y + edge1.z * p.z;
1008 if (std::abs(determinant) < 1e-7f) return false;
1009 const float inverse = 1.f / determinant;
1010 const Vec3 offset{origin.x - a.x, origin.y - a.y, origin.z - a.z};
1011 const float u = (offset.x * p.x + offset.y * p.y + offset.z * p.z) * inverse;
1012 if (u < 0.f || u > 1.f) return false;
1013 const Vec3 q{offset.y * edge1.z - offset.z * edge1.y, offset.z * edge1.x - offset.x * edge1.z,
1014 offset.x * edge1.y - offset.y * edge1.x};
1015 const float v = (direction.x * q.x + direction.y * q.y + direction.z * q.z) * inverse;
1016 if (v < 0.f || u + v > 1.f) return false;
1017 distance = (edge2.x * q.x + edge2.y * q.y + edge2.z * q.z) * inverse;
1018 if (distance < 1e-6f) return false;
1019 normal = {edge1.y * edge2.z - edge1.z * edge2.y, edge1.z * edge2.x - edge1.x * edge2.z,
1020 edge1.x * edge2.y - edge1.y * edge2.x};
1021 const float normalLength = std::sqrt(normal.x * normal.x + normal.y * normal.y + normal.z * normal.z);
1022 if (normalLength < 1e-7f) return false;
1023 normal = {normal.x / normalLength, normal.y / normalLength, normal.z / normalLength};
1024 return true;
1025}

1027Result<MeshBuild> meshFitMesh(const MeshBuild& input, const MeshBuild& surface, const auto& node) {
1028 MeshBuild output = input;
1029 auto& positions = output.positions();
1030 Vec3 direction{parameter(node, "directionX", 0.f), parameter(node, "directionY", -1.f),
1031 parameter(node, "directionZ", 0.f)};
1032 const float directionLength =
1033 std::sqrt(direction.x * direction.x + direction.y * direction.y + direction.z * direction.z);
1034 direction = {direction.x / directionLength, direction.y / directionLength, direction.z / directionLength};
1035 const float maximum = parameter(node, "maxDistance", 10.f);
1036 const float surfaceOffset = parameter(node, "surfaceOffset", 0.f);
1037 const bool bidirectional = intParameter(node, "bidirectional", 0) != 0;
1038 const auto& targetPositions = surface.positions();
1039 const auto& targetIndices = surface.indices();
1040 for (int vertex = 0; vertex < output.getVertexCount(); ++vertex) {
1041 const std::size_t base = static_cast<std::size_t>(vertex) * 3u;
1042 const Vec3 origin{positions[base], positions[base + 1], positions[base + 2]};
1043 float best = maximum + 1.f;
1044 Vec3 bestDirection{}, bestNormal{};
1045 for (const float sign : {1.f, -1.f}) {
1046 if (sign < 0.f && !bidirectional) continue;
1047 const Vec3 ray{direction.x * sign, direction.y * sign, direction.z * sign};
1048 for (std::size_t triangle = 0; triangle + 2u < targetIndices.size(); triangle += 3u) {
1049 const auto point = [&](std::uint32_t index) {
1050 const std::size_t target = static_cast<std::size_t>(index) * 3u;
1051 return Vec3{targetPositions[target], targetPositions[target + 1], targetPositions[target + 2]};
1052 };
1053 float distance = 0.f;
1054 Vec3 normal{};
1055 if (!rayTriangle(origin, ray, point(targetIndices[triangle]), point(targetIndices[triangle + 1u]),
1056 point(targetIndices[triangle + 2u]), distance, normal) ||
1057 distance > maximum || distance >= best)
1058 continue;
1059 best = distance;
1060 bestDirection = ray;
1061 bestNormal = normal;
1062 }
1063 }
1064 if (best > maximum) continue;
1065 positions[base] = origin.x + bestDirection.x * best + bestNormal.x * surfaceOffset;
1066 positions[base + 1] = origin.y + bestDirection.y * best + bestNormal.y * surfaceOffset;
1067 positions[base + 2] = origin.z + bestDirection.z * best + bestNormal.z * surfaceOffset;
1068 }
1069 recalculateNormals(output);
1070 output.setMeta("deformer", "deform.meshFit");
1071 return Result<MeshBuild>::success(std::move(output));
1072}

1074Result<MeshBuild> ffdMesh(const MeshBuild& input, const auto& node) {
1075 MeshBuild output = input;
1076 auto& positions = output.positions();
1077 if (positions.empty()) return Result<MeshBuild>::success(std::move(output));
1078 Vec3 minimum{positions[0], positions[1], positions[2]};
1079 Vec3 maximum = minimum;
1080 for (std::size_t i = 0; i < positions.size(); i += 3u) {
1081 minimum = {std::min(minimum.x, positions[i]), std::min(minimum.y, positions[i + 1]),
1082 std::min(minimum.z, positions[i + 2])};
1083 maximum = {std::max(maximum.x, positions[i]), std::max(maximum.y, positions[i + 1]),
1084 std::max(maximum.z, positions[i + 2])};
1085 }
1086 const Vec3 extent{std::max(maximum.x - minimum.x, 1e-6f), std::max(maximum.y - minimum.y, 1e-6f),
1087 std::max(maximum.z - minimum.z, 1e-6f)};
1088 std::array<Vec3, 8> offsets;
1089 const std::array<const char*, 8> names{"p000", "p001", "p010", "p011", "p100", "p101", "p110", "p111"};
1090 for (std::size_t i = 0; i < names.size(); ++i) {
1091 offsets[i] = {parameter(node, std::string(names[i]) + "x", 0.f),
1092 parameter(node, std::string(names[i]) + "y", 0.f),
1093 parameter(node, std::string(names[i]) + "z", 0.f)};
1094 }
1095 const float weight = std::clamp(parameter(node, "weight", 1.f), 0.f, 1.f);
1096 for (std::size_t base = 0; base < positions.size(); base += 3u) {
1097 const float x = std::clamp((positions[base] - minimum.x) / extent.x, 0.f, 1.f);
1098 const float y = std::clamp((positions[base + 1] - minimum.y) / extent.y, 0.f, 1.f);
1099 const float z = std::clamp((positions[base + 2] - minimum.z) / extent.z, 0.f, 1.f);
1100 Vec3 displacement;
1101 for (int ix = 0; ix < 2; ++ix)
1102 for (int iy = 0; iy < 2; ++iy)
1103 for (int iz = 0; iz < 2; ++iz) {
1104 const float blend = (ix ? x : 1.f - x) * (iy ? y : 1.f - y) * (iz ? z : 1.f - z);
1105 const auto& offset = offsets[static_cast<std::size_t>(ix * 4 + iy * 2 + iz)];
1106 displacement.x += offset.x * blend;
1107 displacement.y += offset.y * blend;
1108 displacement.z += offset.z * blend;
1109 }
1110 positions[base] += displacement.x * weight;
1111 positions[base + 1] += displacement.y * weight;
1112 positions[base + 2] += displacement.z * weight;
1113 }
1114 recalculateNormals(output);
1115 return Result<MeshBuild>::success(std::move(output));
1116}

1131Result<MeshBuild> subdivideMesh(const MeshBuild& input, int levels) {
1132 MeshBuild current = input;
1133 levels = std::clamp(levels, 1, 4);
1134 for (int level = 0; level < levels; ++level) {
1135 MeshBuild next;
1136 next.reserve(current.getIndexCount() * 2, current.getIndexCount() * 4);
1137 std::vector<int> assignments;
1138 auto addSource = [&](std::uint32_t index) {
1139 next.addVertex(current.getPositionX(static_cast<int>(index)), current.getPositionY(static_cast<int>(index)),
1140 current.getPositionZ(static_cast<int>(index)), current.getNormalX(static_cast<int>(index)),
1141 current.getNormalY(static_cast<int>(index)), current.getNormalZ(static_cast<int>(index)),
1142 current.getUvU(static_cast<int>(index)), current.getUvV(static_cast<int>(index)));
1143 return static_cast<std::uint32_t>(next.getVertexCount() - 1);
1144 };
1145 auto midpoint = [&](std::uint32_t a, std::uint32_t b) {
1146 next.addVertex((current.getPositionX(a) + current.getPositionX(b)) * 0.5f,
1147 (current.getPositionY(a) + current.getPositionY(b)) * 0.5f,
1148 (current.getPositionZ(a) + current.getPositionZ(b)) * 0.5f,
1149 (current.getNormalX(a) + current.getNormalX(b)) * 0.5f,
1150 (current.getNormalY(a) + current.getNormalY(b)) * 0.5f,
1151 (current.getNormalZ(a) + current.getNormalZ(b)) * 0.5f,
1152 (current.getUvU(a) + current.getUvU(b)) * 0.5f,
1153 (current.getUvV(a) + current.getUvV(b)) * 0.5f);
1154 return static_cast<std::uint32_t>(next.getVertexCount() - 1);
1155 };
1156 for (int triangle = 0; triangle * 3 + 2 < current.getIndexCount(); ++triangle) {
1157 const auto a0 = static_cast<std::uint32_t>(current.getIndex(triangle * 3));
1158 const auto b0 = static_cast<std::uint32_t>(current.getIndex(triangle * 3 + 1));
1159 const auto c0 = static_cast<std::uint32_t>(current.getIndex(triangle * 3 + 2));
1160 const auto a = addSource(a0), b = addSource(b0), c = addSource(c0);
1161 const auto ab = midpoint(a0, b0), bc = midpoint(b0, c0), ca = midpoint(c0, a0);
1162 for (const auto& tri :
1163 {std::array{a, ab, ca}, std::array{ab, b, bc}, std::array{ca, bc, c}, std::array{ab, bc, ca}}) {
1164 next.addTriangle(tri[0], tri[1], tri[2]);
1165 assignments.push_back(current.getTriangleGroup(triangle));
1166 }
1167 }
1168 std::vector<std::string> names;
1169 for (int i = 0; i < current.getGroupCount(); ++i) names.push_back(current.getGroupName(i));
1170 auto restored = next.restoreGroupData(std::move(names), std::move(assignments), -1);
1171 if (!restored.ok()) return Result<MeshBuild>::failure(restored.status());
1172 copyMetadata(current, next);
1173 recalculateNormals(next);
1174 current = std::move(next);
1175 }
1176 return Result<MeshBuild>::success(std::move(current));

1195Result<MeshBuild> cutPlaneMesh(const MeshBuild& input, const auto& node) {
1196 Vec3 normal{parameter(node, "normalX", 0.f), parameter(node, "normalY", 1.f), parameter(node, "normalZ", 0.f)};
1197 const float length = std::sqrt(normal.x * normal.x + normal.y * normal.y + normal.z * normal.z);
1198 if (length < 1e-7f)
1199 return Result<MeshBuild>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
1200 "mesh.cutPlane requires a non-zero normal", "normal", {},
1201 "procgen.meshModifierGraph"));
1202 normal = {normal.x / length, normal.y / length, normal.z / length};
1203 const float distance = parameter(node, "distance", 0.f);
1204 const float sign = intParameter(node, "keepPositive", 1) != 0 ? 1.f : -1.f;
1205 auto signedDistance = [&](const ClipVertex& vertex) {
1206 return sign *
1207 (vertex.position.x * normal.x + vertex.position.y * normal.y + vertex.position.z * normal.z - distance);
1208 };
1209 MeshBuild output;
1210 output.reserve(input.getIndexCount() * 2, input.getIndexCount() * 2);
1211 std::vector<int> assignments;
1212 std::vector<std::array<Vec3, 2>> cutSegments;
1213 for (int triangle = 0; triangle * 3 + 2 < input.getIndexCount(); ++triangle) {
1214 std::vector<ClipVertex> polygon;
1215 for (int corner = 0; corner < 3; ++corner) {
1216 const int index = input.getIndex(triangle * 3 + corner);
1217 polygon.push_back({{input.getPositionX(index), input.getPositionY(index), input.getPositionZ(index)},
1218 {input.getNormalX(index), input.getNormalY(index), input.getNormalZ(index)},
1219 input.getUvU(index),
1220 input.getUvV(index)});
1221 }
1222 std::vector<ClipVertex> clipped;
1223 std::vector<Vec3> intersections;
1224 for (std::size_t i = 0; i < polygon.size(); ++i) {
1225 const auto& a = polygon[i];
1226 const auto& b = polygon[(i + 1u) % polygon.size()];
1227 const float da = signedDistance(a), db = signedDistance(b);
1228 const bool insideA = da >= 0.f, insideB = db >= 0.f;
1229 if (insideA) clipped.push_back(a);
1230 if (insideA != insideB) {
1231 auto intersection = interpolate(a, b, da / (da - db));
1232 clipped.push_back(intersection);
1233 intersections.push_back(intersection.position);
1234 }
1235 }
1236 if (intersections.size() == 2u) cutSegments.push_back({intersections[0], intersections[1]});
1237 if (clipped.size() < 3u) continue;
1238 const auto base = static_cast<std::uint32_t>(output.getVertexCount());
1239 for (const auto& vertex : clipped)
1240 output.addVertex(vertex.position.x, vertex.position.y, vertex.position.z, vertex.normal.x, vertex.normal.y,
1241 vertex.normal.z, vertex.u, vertex.v);
1242 for (std::size_t i = 1; i + 1 < clipped.size(); ++i) {
1243 output.addTriangle(base, base + static_cast<std::uint32_t>(i), base + static_cast<std::uint32_t>(i + 1));
1244 assignments.push_back(input.getTriangleGroup(triangle));
1245 }
1246 }
1247 std::vector<std::string> names;
1248 for (int i = 0; i < input.getGroupCount(); ++i) names.push_back(input.getGroupName(i));
1249 if (intParameter(node, "cap", 0) != 0 && !cutSegments.empty()) {
1250 const int capGroup = static_cast<int>(names.size());
1251 names.push_back("__cut_cap");
1252 std::vector<Vec3> points;
1253 std::vector<std::array<int, 2>> edges;
1254 auto pointIndex = [&](const Vec3& point) {
1255 constexpr float toleranceSquared = 1e-10f;
1256 for (std::size_t i = 0; i < points.size(); ++i) {
1257 const float dx = points[i].x - point.x, dy = points[i].y - point.y, dz = points[i].z - point.z;
1258 if (dx * dx + dy * dy + dz * dz <= toleranceSquared) return static_cast<int>(i);
1259 }
1260 points.push_back(point);
1261 return static_cast<int>(points.size() - 1u);
1262 };
1263 for (const auto& segment : cutSegments) {
1264 const int a = pointIndex(segment[0]), b = pointIndex(segment[1]);
1265 if (a == b) continue;
1266 const auto duplicate = std::find_if(edges.begin(), edges.end(), [&](const auto& edge) {
1267 return (edge[0] == a && edge[1] == b) || (edge[0] == b && edge[1] == a);
1268 });
1269 if (duplicate == edges.end()) edges.push_back({a, b});
1270 }
1271 std::vector<bool> used(edges.size(), false);
1272 for (std::size_t seed = 0; seed < edges.size(); ++seed) {
1273 if (used[seed]) continue;
1274 std::vector<int> loop{edges[seed][0], edges[seed][1]};
1275 used[seed] = true;
1276 while (loop.back() != loop.front()) {
1277 bool extended = false;
1278 for (std::size_t edge = 0; edge < edges.size(); ++edge) {
1279 if (used[edge]) continue;
1280 if (edges[edge][0] == loop.back() || edges[edge][1] == loop.back()) {
1281 loop.push_back(edges[edge][0] == loop.back() ? edges[edge][1] : edges[edge][0]);
1282 used[edge] = true;
1283 extended = true;
1284 break;
1285 }
1286 }
1287 if (!extended || loop.size() > edges.size() + 1u) break;
1288 }
1289 if (loop.size() < 4u || loop.back() != loop.front()) continue;
1290 loop.pop_back();
1291 Vec3 centroid;
1292 for (const int index : loop) {
1293 centroid.x += points[static_cast<std::size_t>(index)].x;
1294 centroid.y += points[static_cast<std::size_t>(index)].y;
1295 centroid.z += points[static_cast<std::size_t>(index)].z;
1296 }
1297 const float inverseCount = 1.f / static_cast<float>(loop.size());
1298 centroid = {centroid.x * inverseCount, centroid.y * inverseCount, centroid.z * inverseCount};
1299 const Vec3 capNormal{-normal.x * sign, -normal.y * sign, -normal.z * sign};
1300 Vec3 winding;
1301 for (std::size_t i = 0; i < loop.size(); ++i) {
1302 const auto& a = points[static_cast<std::size_t>(loop[i])];
1303 const auto& b = points[static_cast<std::size_t>(loop[(i + 1u) % loop.size()])];
1304 winding.x += a.y * b.z - a.z * b.y;
1305 winding.y += a.z * b.x - a.x * b.z;
1306 winding.z += a.x * b.y - a.y * b.x;
1307 }
1308 if (winding.x * capNormal.x + winding.y * capNormal.y + winding.z * capNormal.z < 0.f)
1309 std::reverse(loop.begin(), loop.end());
1310 const auto centerIndex = static_cast<std::uint32_t>(output.getVertexCount());
1311 output.addVertex(centroid.x, centroid.y, centroid.z, capNormal.x, capNormal.y, capNormal.z, 0.5f, 0.5f);
1312 for (const int index : loop) {
1313 const auto& point = points[static_cast<std::size_t>(index)];
1314 output.addVertex(point.x, point.y, point.z, capNormal.x, capNormal.y, capNormal.z,
1315 0.5f + (point.x - centroid.x), 0.5f + (point.z - centroid.z));
1316 }
1317 for (std::size_t i = 0; i < loop.size(); ++i) {
1318 const auto a = centerIndex + 1u + static_cast<std::uint32_t>(i);
1319 const auto b = centerIndex + 1u + static_cast<std::uint32_t>((i + 1u) % loop.size());
1320 output.addTriangle(centerIndex, a, b);
1321 assignments.push_back(capGroup);
1322 }
1323 }
1324 }
1325 auto restored = output.restoreGroupData(std::move(names), std::move(assignments), -1);
1326 if (!restored.ok()) return Result<MeshBuild>::failure(restored.status());
1327 copyMetadata(input, output);
1328 return Result<MeshBuild>::success(std::move(output));
1329}

1331struct WeldKey {
1332 std::int64_t x, y, z;
1333 bool operator==(const WeldKey&) const = default;
1334};
1335
1336struct WeldKeyHash {
1337 std::size_t operator()(const WeldKey& key) const noexcept {
1338 std::size_t value = std::hash<std::int64_t>{}(key.x);
1339 value ^= std::hash<std::int64_t>{}(key.y) + 0x9e3779b9u + (value << 6u) + (value >> 2u);
1340 value ^= std::hash<std::int64_t>{}(key.z) + 0x9e3779b9u + (value << 6u) + (value >> 2u);
1341 return value;
1342 }
1343};

1345Result<MeshBuild> weldMesh(const MeshBuild& input, float tolerance) {
1346 if (!std::isfinite(tolerance) || tolerance <= 0.f)
1347 return Result<MeshBuild>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
1348 "mesh.weld tolerance must be finite and positive",
1349 "tolerance", {}, "procgen.meshModifierGraph"));
1350 MeshBuild output;
1351 output.reserve(input.getVertexCount(), input.getIndexCount());
1352 std::unordered_map<WeldKey, std::uint32_t, WeldKeyHash> vertices;
1353 std::vector<std::uint32_t> remap(static_cast<std::size_t>(input.getVertexCount()));
1354 for (int i = 0; i < input.getVertexCount(); ++i) {
1355 const WeldKey key{std::llround(input.getPositionX(i) / tolerance),
1356 std::llround(input.getPositionY(i) / tolerance),
1357 std::llround(input.getPositionZ(i) / tolerance)};
1358 const auto found = vertices.find(key);
1359 if (found != vertices.end()) {
1360 remap[static_cast<std::size_t>(i)] = found->second;
1361 continue;
1362 }
1363 const auto next = static_cast<std::uint32_t>(output.getVertexCount());
1364 vertices.emplace(key, next);
1365 remap[static_cast<std::size_t>(i)] = next;
1366 output.addVertex(input.getPositionX(i), input.getPositionY(i), input.getPositionZ(i), input.getNormalX(i),
1367 input.getNormalY(i), input.getNormalZ(i), input.getUvU(i), input.getUvV(i));
1368 }
1369 std::vector<int> assignments;
1370 for (int i = 0; i + 2 < input.getIndexCount(); i += 3) {
1371 const auto a = remap[static_cast<std::size_t>(input.getIndex(i))];
1372 const auto b = remap[static_cast<std::size_t>(input.getIndex(i + 1))];
1373 const auto c = remap[static_cast<std::size_t>(input.getIndex(i + 2))];
1374 if (a == b || b == c || a == c) continue;
1375 output.addTriangle(a, b, c);
1376 assignments.push_back(input.getTriangleGroup(i / 3));
1377 }
1378 std::vector<std::string> names;
1379 for (int i = 0; i < input.getGroupCount(); ++i) names.push_back(input.getGroupName(i));
1380 auto restored = output.restoreGroupData(std::move(names), std::move(assignments), -1);
1381 if (!restored.ok()) return Result<MeshBuild>::failure(restored.status());
1382 for (const auto& [key, value] : input.metadata()) output.setMeta(key, value);
1383 recalculateNormals(output);
1384 return Result<MeshBuild>::success(std::move(output));
1385}

1551Result<void> MeshModifierGraph::validateNode(std::string_view id, std::unordered_map<std::string, int>& states) const {
1552 const std::string key(id);
1553 if (states[key] == 2) return Result<void>::success();
1554 if (states[key] == 1) return graphFailure(DiagnosticCode::Conflict, "cycle at mesh modifier node: " + key, key);
1555 const auto found = nodes_.find(key);
1556 if (found == nodes_.end()) return graphFailure(DiagnosticCode::NotFound, "unknown mesh modifier node: " + key, key);
1557 states[key] = 1;
1558 const Node& node = found->second;
1559 if (node.operation == "mesh.input") {
1560 if (!node.hasInputMesh)
1561 return graphFailure(DiagnosticCode::PreconditionViolation, "mesh.input has no bound mesh", key);
1562 auto valid = validateMesh(node.inputMesh, key);
1563 if (!valid.ok()) return valid;
1564 }
1565 for (std::size_t input = 0; input < node.inputs.size(); ++input) {
1566 if (node.inputs[input].empty())
1567 return graphFailure(DiagnosticCode::PreconditionViolation,
1568 node.operation + " requires mesh input " + std::to_string(input), key);
1569 auto valid = validateNode(node.inputs[input], states);
1570 if (!valid.ok()) return valid;
1571 }
1572 const std::string axis = stringParameter(node, "axis", "y");
1573 if ((node.operation == "deform.bend" || node.operation == "deform.twist" ||
1574 node.operation == "deform.angularBend" || node.operation == "deform.spline" ||
1575 node.operation == "deform.splinePath") &&
1576 axis != "x" && axis != "y" && axis != "z")
1577 return graphFailure(DiagnosticCode::InvalidArgument, "deformation axis must be x, y, or z", key + ".axis");
1578 if ((node.operation == "deform.splinePath" || node.operation == "mesh.splineTube" ||
1579 node.operation == "mesh.splineRibbon" || node.operation == "mesh.splineExtrude") &&
1580 !node.hasSplinePath)
1581 return graphFailure(DiagnosticCode::PreconditionViolation, node.operation + " has no bound spline path", key);
1582 if (node.hasSplinePath && node.splinePath.chunkCount() > 1) {
1583 if (node.operation == "deform.splinePath")
1584 return graphFailure(DiagnosticCode::PreconditionViolation, "deform.splinePath requires one connected chunk",
1585 key);
1586 for (int chunk = 0; chunk < node.splinePath.chunkCount(); ++chunk) {
1587 auto path = node.splinePath.chunkPathResult(chunk);
1588 if (!path.ok()) return Result<void>::failure(path.status());
1589 }
1590 }
1591 if (node.operation == "deform.splinePath" && std::abs(parameter(node, "scale", 1.f)) < 1e-7f)
1592 return graphFailure(DiagnosticCode::InvalidArgument, "spline path scale must be non-zero", key + ".scale");
1593 if (node.operation == "mesh.splineTube") {
1594 const int pathSegments = intParameter(node, "pathSegments", 32);
1595 const int radialSegments = intParameter(node, "radialSegments", 12);
1596 if (parameter(node, "radius", 0.5f) <= 0.f)
1597 return graphFailure(DiagnosticCode::InvalidArgument, "spline tube radius must be positive",
1598 key + ".radius");
1599 if (pathSegments < 1 || pathSegments > 4096)
1600 return graphFailure(DiagnosticCode::InvalidArgument, "spline tube pathSegments must be in [1, 4096]",
1601 key + ".pathSegments");
1602 if (radialSegments < 3 || radialSegments > 256)
1603 return graphFailure(DiagnosticCode::InvalidArgument, "spline tube radialSegments must be in [3, 256]",
1604 key + ".radialSegments");
1605 const std::uint64_t capVertices = node.splinePath.isClosed() || intParameter(node, "cap", 1) == 0
1606 ? 0u
1607 : 2u * static_cast<std::uint64_t>(radialSegments + 2);
1608 const std::uint64_t vertexCount =
1609 static_cast<std::uint64_t>(pathSegments + 1) * static_cast<std::uint64_t>(radialSegments + 1) + capVertices;
1610 if (vertexCount > 1'000'000u)
1611 return graphFailure(DiagnosticCode::InvalidArgument, "spline tube exceeds the one-million vertex budget",
1612 key);
1613 }
1614 if (node.operation == "mesh.splineRibbon") {
1615 const int pathSegments = intParameter(node, "pathSegments", 32);
1616 if (parameter(node, "width", 2.f) <= 0.f)
1617 return graphFailure(DiagnosticCode::InvalidArgument, "spline ribbon width must be positive",
1618 key + ".width");
1619 if (parameter(node, "thickness", 0.f) < 0.f)
1620 return graphFailure(DiagnosticCode::InvalidArgument, "spline ribbon thickness must be non-negative",
1621 key + ".thickness");
1622 if (pathSegments < 1 || pathSegments > 4096)
1623 return graphFailure(DiagnosticCode::InvalidArgument, "spline ribbon pathSegments must be in [1, 4096]",
1624 key + ".pathSegments");
1625 }
1626 if (node.operation == "mesh.splineExtrude") {
1627 if (!node.hasSplineProfile)
1628 return graphFailure(DiagnosticCode::PreconditionViolation, "mesh.splineExtrude has no bound profile", key);
1629 const int pathSegments = intParameter(node, "pathSegments", 32);
1630 if (pathSegments < 1 || pathSegments > 4096)
1631 return graphFailure(DiagnosticCode::InvalidArgument, "spline extrusion pathSegments must be in [1, 4096]",
1632 key + ".pathSegments");
1633 const std::uint64_t stride = node.splineProfile.points.size() + (node.splineProfile.closed ? 1u : 0u);
1634 if (static_cast<std::uint64_t>(pathSegments + 1) * stride > 1'000'000u)
1635 return graphFailure(DiagnosticCode::InvalidArgument,
1636 "spline extrusion exceeds the one-million vertex budget", key);
1637 }
1638 if (node.operation == "deform.transform" &&
1639 (std::abs(parameter(node, "scaleX", 1.f)) < 1e-7f || std::abs(parameter(node, "scaleY", 1.f)) < 1e-7f ||
1640 std::abs(parameter(node, "scaleZ", 1.f)) < 1e-7f))
1641 return graphFailure(DiagnosticCode::InvalidArgument, "deformation scale components must be non-zero", key);
1642 if (node.operation == "deform.soundReact") {
1643 const float level = parameter(node, "level", 0.f), threshold = parameter(node, "threshold", 0.f);
1644 const std::string axis = stringParameter(node, "axis", "y");
1645 if (level < 0.f || level > 1.f)
1646 return graphFailure(DiagnosticCode::InvalidArgument, "sound-react level must be in [0, 1]", key + ".level");
1647 if (threshold < 0.f || threshold > 1.f)
1648 return graphFailure(DiagnosticCode::InvalidArgument, "sound-react threshold must be in [0, 1]",
1649 key + ".threshold");
1650 if (axis != "x" && axis != "y" && axis != "z")
1651 return graphFailure(DiagnosticCode::InvalidArgument, "sound-react axis must be x, y, or z", key + ".axis");
1652 if (parameter(node, "frequency", 1.f) < 0.f)
1653 return graphFailure(DiagnosticCode::InvalidArgument, "sound-react frequency must be non-negative",
1654 key + ".frequency");
1655 }
1656 if (node.operation == "deform.effector") {
1657 const int pointCount = intParameter(node, "pointCount", 1);
1658 if (pointCount < 1 || pointCount > 4)
1659 return graphFailure(DiagnosticCode::InvalidArgument, "mesh effector pointCount must be in [1, 4]",
1660 key + ".pointCount");
1661 if (parameter(node, "density", 1.f) <= 0.f)
1662 return graphFailure(DiagnosticCode::InvalidArgument, "mesh effector density must be positive",
1663 key + ".density");
1664 for (int point = 0; point < pointCount; ++point) {
1665 const std::string radius = "p" + std::to_string(point) + "radius";
1666 if (parameter(node, radius, 1.f) <= 0.f)
1667 return graphFailure(DiagnosticCode::InvalidArgument, "mesh effector radius must be positive",
1668 key + "." + radius);
1669 }
1670 }
1671 if (node.operation == "deform.meshFit") {
1672 const float x = parameter(node, "directionX", 0.f), y = parameter(node, "directionY", -1.f);
1673 const float z = parameter(node, "directionZ", 0.f);
1674 if (x * x + y * y + z * z < 1e-12f)
1675 return graphFailure(DiagnosticCode::InvalidArgument, "mesh fit direction must be non-zero",
1676 key + ".direction");
1677 if (parameter(node, "maxDistance", 10.f) <= 0.f)
1678 return graphFailure(DiagnosticCode::InvalidArgument, "mesh fit maxDistance must be positive",
1679 key + ".maxDistance");
1680 }
1681 states[key] = 2;
1682 return Result<void>::success();
1683}

1694Result<std::vector<MeshModifierGraph::Segment>> MeshModifierGraph::compilePlan(std::string_view outputId) const {
1695 std::vector<std::string> order;
1696 std::unordered_map<std::string, int> states;
1697 std::function<Result<void>(const std::string&)> visit = [&](const std::string& id) -> Result<void> {
1698 if (states[id] == 2) return Result<void>::success();
1699 if (states[id] == 1) return graphFailure(DiagnosticCode::Conflict, "cycle at mesh modifier node: " + id, id);
1700 const auto found = nodes_.find(id);
1701 if (found == nodes_.end())
1702 return graphFailure(DiagnosticCode::NotFound, "unknown mesh modifier output: " + id, id);
1703 states[id] = 1;
1704 for (const auto& input : found->second.inputs) {
1705 if (input.empty())
1706 return graphFailure(DiagnosticCode::PreconditionViolation,
1707 found->second.operation + " has an unconnected input", id);
1708 auto result = visit(input);
1709 if (!result.ok()) return result;
1710 }
1711 states[id] = 2;
1712 order.push_back(id);
1713 return Result<void>::success();
1714 };
1715 auto visited = visit(std::string(outputId));
1716 if (!visited.ok()) return Result<std::vector<Segment>>::failure(visited.status());

1856Result<MeshBuild> MeshModifierGraph::executeResult(std::string_view outputId) {
1857 const auto requested = nodes_.find(std::string(outputId));
1858 if (requested == nodes_.end())
1859 return Result<MeshBuild>::failure(Diagnostic::error(DiagnosticCode::NotFound,
1860 "unknown mesh modifier output: " + std::string(outputId),
1861 std::string(outputId), {}, "procgen.meshModifierGraph"));
1862 if (requested->second.cacheValid) {
1863 metrics_ = {{requested->first, requested->second.cache.getVertexCount(),
1864 requested->second.cache.getIndexCount() / 3, 0.f, true, false}};
1865 compiledSegmentCount_ = 0;
1866 fusedOperationCount_ = 0;
1867 return Result<MeshBuild>::success(requested->second.cache);
1868 }
1869 std::unordered_map<std::string, int> validationStates;
1870 auto valid = validateNode(outputId, validationStates);
1871 if (!valid.ok()) return Result<MeshBuild>::failure(valid.status());
1872 auto planned = compilePlan(outputId);
1873 if (!planned.ok()) return Result<MeshBuild>::failure(planned.status());
1874 const auto plan = std::move(planned).takeValue();
1875 std::unordered_map<std::string, MeshBuild> outputs;
1876 std::vector<MeshModifierNodeMetric> candidateMetrics;
1877 int candidateFused = 0;
1878 for (const auto& segment : plan) {
1879 const auto started = std::chrono::steady_clock::now();
1880 Result<MeshBuild> evaluated = segment.fusedVertexTraversal
1881 ? executeFused(segment, outputs)
1882 : executeNode(nodes_.at(segment.nodes.front()), outputs);
1883 if (!evaluated.ok()) return Result<MeshBuild>::failure(evaluated.status());
1884 MeshBuild output = std::move(evaluated).takeValue();
1885 const auto elapsed =
1886 std::chrono::duration<float, std::milli>(std::chrono::steady_clock::now() - started).count();
1887 const std::string& outputNode = segment.nodes.back();
1888 candidateMetrics.push_back({outputNode, output.getVertexCount(), output.getIndexCount() / 3, elapsed, false,
1889 segment.fusedVertexTraversal});
1890 if (segment.fusedVertexTraversal) candidateFused += static_cast<int>(segment.nodes.size());
1891 outputs.emplace(outputNode, std::move(output));
1892 }
1893 const auto found = outputs.find(std::string(outputId));
1894 if (found == outputs.end())
1895 return Result<MeshBuild>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
1896 "mesh modifier output was not produced",
1897 std::string(outputId), {}, "procgen.meshModifierGraph"));
1898 for (auto& [id, node] : nodes_) {
1899 const auto output = outputs.find(id);
1900 if (output != outputs.end()) {
1901 node.cache = output->second;
1902 node.cacheValid = true;
1903 }
1904 }
1905 metrics_ = std::move(candidateMetrics);
1906 compiledSegmentCount_ = static_cast<int>(plan.size());
1907 fusedOperationCount_ = candidateFused;
1908 ++executionPlanBuildCount_;
1909 return Result<MeshBuild>::success(found->second);
1910}
