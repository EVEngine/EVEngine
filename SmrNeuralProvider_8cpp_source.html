23Result<void> SmrNeuralProvider::ensureOnnx(const AnimRetargetProfile& profile) const {
24 const std::string& path = profile.getNeuralModelPath();
25 if (path.empty()) {
26 return Result<void>::failure(
27 Diagnostic::error(DiagnosticCode::PreconditionViolation, "SmrNeuralProvider: empty ONNX path"));
28 }
29 if (modelLoaded_ && loadedPath_ == path) return Result<void>::success();
30 auto loaded = onnx_.loadFile(path);
31 if (!loaded.ok()) return loaded;
32 loadedPath_ = path;
33 modelLoaded_ = true;
34 return Result<void>::success();

37Result<SmrNeuralResult> SmrNeuralProvider::applyRot6d(const SmrNeuralRequest& request, const SmrFeatureBatch& features,
38 const std::vector<float>& rot6d,
39 const std::string& backend) const {
40 if (!request.targetClip || !request.targetSkeleton) {
41 return Result<SmrNeuralResult>::failure(
42 Diagnostic::error(DiagnosticCode::InvalidArgument, "SmrNeuralProvider: null clip/skeleton"));
43 }
44 if (static_cast<int>(rot6d.size()) < features.frames * features.joints * 6) {
45 return Result<SmrNeuralResult>::failure(
46 Diagnostic::error(DiagnosticCode::InvariantViolation, "SmrNeuralProvider: rot6d size mismatch"));
47 }
48
49 AnimClip& clip = *request.targetClip;
50 const float duration = clip.getDuration();
51 for (int joint = 0; joint < features.joints; ++joint) {
52 clip.clearTrack(joint);
53 for (int frame = 0; frame < features.frames; ++frame) {
54 const float time = duration <= 0.f ? 0.f
55 : duration * static_cast<float>(frame) /
56 static_cast<float>(std::max(features.frames - 1, 1));
57 const float* r6 = rot6d.data() + static_cast<size_t>((frame * features.joints + joint) * 6);
58 float qx = 0.f, qy = 0.f, qz = 0.f, qw = 1.f;
59 rot6dToQuat(r6, qx, qy, qz, qw);
60 clip.addRotationKey(joint, time, qx, qy, qz, qw);
61 }
62 }
63
64 SmrNeuralResult result;
65 result.framesWritten = features.frames;
66 result.backend = backend;
67 result.sensorCount = static_cast<size_t>(features.sensors);
68 result.pairCount = static_cast<size_t>(features.pairs);
69 return Result<SmrNeuralResult>::success(std::move(result));
70}

72Result<SmrNeuralResult> SmrNeuralProvider::retarget(const SmrNeuralRequest& request) {
73 if (!request.sourceClip || !request.targetClip || !request.sourceSkeleton || !request.targetSkeleton ||
74 !request.profile) {
75 return Result<SmrNeuralResult>::failure(
76 Diagnostic::error(DiagnosticCode::InvalidArgument, "SmrNeuralProvider.retarget: null argument"));
77 }
78
79 SmrFeatureBatch features = buildSmrFeatures(*request.sourceClip, request.sourceSkeleton, request.targetSkeleton,
80 request.sourceSkin, request.targetSkin, 2, 4);
81
82 const std::string& backendHint = request.profile->getNeuralBackend();
83 const bool preferOnnx =
84 backendHint == "onnx" || (backendHint != "tensor" && !request.profile->getNeuralModelPath().empty());
85 if (preferOnnx) {
86 auto ready = ensureOnnx(*request.profile);
87 if (ready.ok()) {
88 auto ran = onnx_.run(features);
89 if (ran.ok()) return applyRot6d(request, features, ran.value(), "onnx");
90 if (backendHint == "onnx") return Result<SmrNeuralResult>::failure(ran.status());
91 } else if (backendHint == "onnx") {
92 return Result<SmrNeuralResult>::failure(ready.status());
93 }
94 }
95
96 auto ran = net_.forward(features);
97 if (!ran.ok()) return Result<SmrNeuralResult>::failure(ran.status());
98 return applyRot6d(request, features, ran.value(), "tensor-meshret");
99}
