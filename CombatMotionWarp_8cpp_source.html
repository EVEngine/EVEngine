23Result<void> CombatWarpRequest::validate() const {
24 if (!finite3(attacker) || !finite3(target) || !finite3(attackerFacing))

39Result<CombatWarpResult> CombatMotionWarp::solve(const CombatWarpRequest& request) {
40 auto valid = request.validate();
41 if (!valid) return Result<CombatWarpResult>::failure(valid.status());
42
43 CombatWarpResult result;
44 result.facing = request.attackerFacing;
45 CombatVector3 delta{request.target.x - request.attacker.x, 0.0, request.target.z - request.attacker.z};
46 const double distance = lengthXZ(delta);
47 if (request.facing && distance > 0.0) result.facing = normalizedXZ(delta);
48 if (!request.horizontal || distance <= 0.0)
49 return Result<CombatWarpResult>::success(result, Status::success(StatusCode::NoOp));
50
51 const double error = distance - request.desiredDistance;
52 if (std::abs(error) <= 1e-6) return Result<CombatWarpResult>::success(result, Status::success(StatusCode::NoOp));
53
54 CombatVector3 direction = normalizedXZ(delta);
55 if (error < 0.0) direction = {-direction.x, 0.0, -direction.z};
56 double step = std::min(std::abs(error), request.maxTranslation);
57 if (step > request.remainingBudget) {
58 result.budgetExceeded = true;
59 return Result<CombatWarpResult>::failure(
60 Diagnostic::error(DiagnosticCode::Conflict, "motion warp would exceed remaining budget", "budget"));
61 }
62 result.translation = {direction.x * step, 0.0, direction.z * step};
63 result.budgetConsumed = step;
64 return Result<CombatWarpResult>::success(result, Status::success(StatusCode::Applied));
65}

73std::vector<CombatTargetCandidate> sortedUnique(std::vector<CombatTargetCandidate> candidates, SubjectRef owner) {
74 candidates.erase(std::remove_if(candidates.begin(), candidates.end(),
75 [&](const CombatTargetCandidate& candidate) {
76 return !candidate.subject.isValid() || candidate.subject == owner ||
77 !std::isfinite(candidate.score);

79 candidates.end());
80 std::stable_sort(candidates.begin(), candidates.end(),
81 [](const CombatTargetCandidate& a, const CombatTargetCandidate& b) {

83 return a.subject.format() < b.subject.format();
84 });
85 return candidates;
