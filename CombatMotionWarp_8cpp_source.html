36Result<CombatWarpResult> CombatMotionWarp::solve(const CombatWarpRequest& request) {
37 auto valid = request.validate();
38 if (!valid) return Result<CombatWarpResult>::failure(valid.status());
39
40 CombatWarpResult result;
41 result.facing = request.attackerFacing;
42 CombatVector3 delta{request.target.x - request.attacker.x, 0.0, request.target.z - request.attacker.z};
43 const double distance = lengthXZ(delta);
44 if (request.facing && distance > 0.0) result.facing = normalizedXZ(delta);
45 if (!request.horizontal || distance <= 0.0)
46 return Result<CombatWarpResult>::success(result, Status::success(StatusCode::NoOp));
47
48 const double error = distance - request.desiredDistance;
49 if (std::abs(error) <= 1e-6) return Result<CombatWarpResult>::success(result, Status::success(StatusCode::NoOp));
50
51 CombatVector3 direction = normalizedXZ(delta);
52 if (error < 0.0) direction = {-direction.x, 0.0, -direction.z};
53 double step = std::min(std::abs(error), request.maxTranslation);
54 if (step > request.remainingBudget) {
55 result.budgetExceeded = true;
56 return Result<CombatWarpResult>::failure(
57 Diagnostic::error(DiagnosticCode::Conflict, "motion warp would exceed remaining budget", "budget"));
58 }
59 result.translation = {direction.x * step, 0.0, direction.z * step};
60 result.budgetConsumed = step;
61 return Result<CombatWarpResult>::success(result, Status::success(StatusCode::Applied));
62}
