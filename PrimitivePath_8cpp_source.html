32float pointLineDistance(glm::vec2 point, glm::vec2 a, glm::vec2 b) {
33 const glm::vec2 line = b - a;
34 const float lengthSquared = glm::dot(line, line);
35 if (lengthSquared <= 1e-20f) return glm::length(point - a);
36 const float t = glm::clamp(glm::dot(point - a, line) / lengthSquared, 0.f, 1.f);
37 return glm::length(point - (a + line * t));
38}

40void flattenQuad(glm::vec2 a, glm::vec2 control, glm::vec2 b, float tolerance, std::uint32_t depth,
41 std::uint32_t maxDepth, std::vector<glm::vec2>& output) {
42 if (depth >= maxDepth || pointLineDistance(control, a, b) <= tolerance) {
43 output.push_back(b);
44 return;
45 }
46 const glm::vec2 ac = (a + control) * 0.5f;
47 const glm::vec2 cb = (control + b) * 0.5f;
48 const glm::vec2 middle = (ac + cb) * 0.5f;
49 flattenQuad(a, ac, middle, tolerance, depth + 1, maxDepth, output);
50 flattenQuad(middle, cb, b, tolerance, depth + 1, maxDepth, output);
51}

53void flattenCubic(glm::vec2 a, glm::vec2 c1, glm::vec2 c2, glm::vec2 b, float tolerance, std::uint32_t depth,
54 std::uint32_t maxDepth, std::vector<glm::vec2>& output) {
55 const float deviation = std::max(pointLineDistance(c1, a, b), pointLineDistance(c2, a, b));
56 if (depth >= maxDepth || deviation <= tolerance) {
57 output.push_back(b);
58 return;
59 }
60 const glm::vec2 a1 = (a + c1) * 0.5f;
61 const glm::vec2 c12 = (c1 + c2) * 0.5f;
62 const glm::vec2 c2b = (c2 + b) * 0.5f;
63 const glm::vec2 leftControl = (a1 + c12) * 0.5f;
64 const glm::vec2 rightControl = (c12 + c2b) * 0.5f;
65 const glm::vec2 middle = (leftControl + rightControl) * 0.5f;
66 flattenCubic(a, a1, leftControl, middle, tolerance, depth + 1, maxDepth, output);
67 flattenCubic(middle, rightControl, c2b, b, tolerance, depth + 1, maxDepth, output);
68}

123std::vector<FlattenedContour2D> Path2D::flatten(float tolerance, std::uint32_t maxDepth) const {
124 const bool validTolerance = std::isfinite(tolerance) && tolerance > 0.f;
125 EV_PARAM_CHECK(validTolerance, "path flatten tolerance must be finite and positive");
126 const bool validMaxDepth = maxDepth > 0 && maxDepth <= 24;
127 EV_PARAM_CHECK(validMaxDepth, "path flatten maxDepth must be in [1, 24]");
128
129 std::vector<FlattenedContour2D> result;
130 std::size_t pointIndex = 0;
131 glm::vec2 current{0.f};
132 for (Verb verb : verbs_) {
133 switch (verb) {
134 case Verb::Move:
135 result.push_back({});
136 current = points_[pointIndex++];
137 result.back().points.push_back(current);
138 break;
139 case Verb::Line:
140 current = points_[pointIndex++];
141 result.back().points.push_back(current);
142 break;
143 case Verb::Quad: {
144 const glm::vec2 control = points_[pointIndex++];
145 const glm::vec2 end = points_[pointIndex++];
146 flattenQuad(current, control, end, tolerance, 0, maxDepth, result.back().points);
147 current = end;
148 break;
149 }
150 case Verb::Cubic: {
151 const glm::vec2 control1 = points_[pointIndex++];
152 const glm::vec2 control2 = points_[pointIndex++];
153 const glm::vec2 end = points_[pointIndex++];
154 flattenCubic(current, control1, control2, end, tolerance, 0, maxDepth, result.back().points);
155 current = end;
156 break;
157 }
158 case Verb::Close: result.back().closed = true; break;
159 }
160 }
161 return result;
162}

170bool finite(glm::vec3 value) { return std::isfinite(value.x) && std::isfinite(value.y) && std::isfinite(value.z); }

172bool nonZero(glm::vec3 value) {
173 constexpr float minimumLengthSquared = 1e-12f;
174 return finite(value) && glm::dot(value, value) > minimumLengthSquared;
175}
176
177bool independent(glm::vec3 a, glm::vec3 b) {
178 constexpr float minimumRelativeAreaSquared = 1e-12f;
179 const float aLengthSquared = glm::dot(a, a);
180 const float bLengthSquared = glm::dot(b, b);
181 const glm::vec3 crossValue = glm::cross(a, b);
182 const float areaSquared = glm::dot(crossValue, crossValue);
183 return aLengthSquared > 0.f && bLengthSquared > 0.f &&
184 areaSquared > minimumRelativeAreaSquared * aLengthSquared * bLengthSquared;
185}

187bool formsBasis(glm::vec3 a, glm::vec3 b, glm::vec3 c) {
188 constexpr float minimumRelativeVolumeSquared = 1e-12f;
189 const float volume = glm::dot(a, glm::cross(b, c));
190 return volume * volume > minimumRelativeVolumeSquared * glm::dot(a, a) * glm::dot(b, b) * glm::dot(c, c);
191}

193eve::Result<void> validateDescriptor(const PrimitiveDescriptor3D& descriptor) {
194 const auto& paint = descriptor.paint;
195 const auto& stroke = paint.stroke;
196 if (!std::isfinite(stroke.width) || stroke.width < 0.f || !std::isfinite(stroke.miterLimit) ||
197 stroke.miterLimit <= 0.f || !std::isfinite(paint.color.r) || !std::isfinite(paint.color.g) ||
198 !std::isfinite(paint.color.b) || !std::isfinite(paint.color.a))
199 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
200 "primitive paint must be finite and valid",
201 "primitive", {}, "graphics"));
202 if (stroke.dash) {
203 const auto& dash = *stroke.dash;
204 float period = 0.f;
205 if (dash.intervals.empty() || dash.intervals.size() % 2 != 0 || !std::isfinite(dash.phase))
206 return eve::Result<void>::failure(eve::Diagnostic::error(
207 eve::DiagnosticCode::InvalidArgument, "invalid primitive dash pattern", "primitive", {}, "graphics"));
208 for (float interval : dash.intervals) {
209 if (!std::isfinite(interval) || interval <= 0.f)
210 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
211 "invalid primitive dash interval", "primitive",
212 {}, "graphics"));
213 period += interval;
214 }
215 if (!std::isfinite(period))
216 return eve::Result<void>::failure(eve::Diagnostic::error(
217 eve::DiagnosticCode::InvalidArgument, "primitive dash period overflow", "primitive", {}, "graphics"));
218 }
219 try {
220 descriptor.paint.validate();
221 } catch (const std::exception& error) {
222 return eve::Result<void>::failure(
223 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, error.what(), "primitive", {}, "graphics"));
224 }
225 for (glm::length_t column = 0; column < 4; ++column)
226 for (glm::length_t row = 0; row < 4; ++row)
227 if (!std::isfinite(descriptor.transform[column][row]))
228 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
229 "primitive transform must be finite",
230 "primitive", {}, "graphics"));
231 const bool valid = std::visit(
232 [](const auto& geometry) {
233 using Geometry = std::decay_t<decltype(geometry)>;
234 if constexpr (std::is_same_v<Geometry, PrimitivePolyline3D>) {
235 return geometry.points.size() >= (geometry.closed ? 3u : 2u) &&
236 std::all_of(geometry.points.begin(), geometry.points.end(), finite);
237 } else if constexpr (std::is_same_v<Geometry, PrimitiveAabb3D>) {
238 return finite(geometry.minimum) && finite(geometry.maximum) &&
239 geometry.minimum.x <= geometry.maximum.x && geometry.minimum.y <= geometry.maximum.y &&
240 geometry.minimum.z <= geometry.maximum.z;
241 } else if constexpr (std::is_same_v<Geometry, PrimitiveObb3D>) {
242 return finite(geometry.center) && nonZero(geometry.halfAxes[0]) && nonZero(geometry.halfAxes[1]) &&
243 nonZero(geometry.halfAxes[2]) && independent(geometry.halfAxes[0], geometry.halfAxes[1]) &&
244 independent(geometry.halfAxes[1], geometry.halfAxes[2]) &&
245 independent(geometry.halfAxes[2], geometry.halfAxes[0]) &&
246 formsBasis(geometry.halfAxes[0], geometry.halfAxes[1], geometry.halfAxes[2]);
247 } else if constexpr (std::is_same_v<Geometry, PrimitiveDisk3D>) {
248 return finite(geometry.center) && nonZero(geometry.normal) && std::isfinite(geometry.radius) &&
249 geometry.radius > 0.f && geometry.segments >= 3;
250 } else if constexpr (std::is_same_v<Geometry, PrimitiveArc3D>) {
251 return finite(geometry.center) && nonZero(geometry.normal) && nonZero(geometry.zeroDirection) &&
252 independent(geometry.normal, geometry.zeroDirection) && std::isfinite(geometry.radius) &&
253 geometry.radius > 0.f && std::isfinite(geometry.startRadians) &&
254 std::isfinite(geometry.sweepRadians) && geometry.segments >= 1;
255 } else if constexpr (std::is_same_v<Geometry, PrimitiveSphere3D>) {
256 return finite(geometry.center) && std::isfinite(geometry.radius) && geometry.radius > 0.f &&
257 geometry.segments >= 3;
258 } else if constexpr (std::is_same_v<Geometry, PrimitiveCapsule3D> ||
259 std::is_same_v<Geometry, PrimitiveCylinder3D>) {
260 return finite(geometry.a) && finite(geometry.b) && nonZero(geometry.b - geometry.a) &&
261 std::isfinite(geometry.radius) && geometry.radius > 0.f && geometry.segments >= 3;
262 } else if constexpr (std::is_same_v<Geometry, PrimitiveCone3D>) {
263 return finite(geometry.apex) && nonZero(geometry.axis) && std::isfinite(geometry.height) &&
264 geometry.height > 0.f && std::isfinite(geometry.radius) && geometry.radius > 0.f &&
265 geometry.segments >= 3;
266 } else if constexpr (std::is_same_v<Geometry, PrimitiveGrid3D>) {
267 return finite(geometry.origin) && nonZero(geometry.axisU) && nonZero(geometry.axisV) &&
268 independent(geometry.axisU, geometry.axisV) && geometry.cellsU > 0 && geometry.cellsV > 0;
269 } else if constexpr (std::is_same_v<Geometry, PrimitiveArrow3D>) {
270 return finite(geometry.from) && finite(geometry.to) && nonZero(geometry.to - geometry.from) &&
271 std::isfinite(geometry.headLength) && geometry.headLength > 0.f &&
272 std::isfinite(geometry.headRadius) && geometry.headRadius > 0.f;
273 } else {
274 return std::all_of(geometry.corners.begin(), geometry.corners.end(), finite);
275 }
276 },
277 descriptor.geometry);
278 if (!valid)
279 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
280 "primitive geometry is invalid or non-finite",
281 "primitive", {}, "graphics"));
282 return eve::Result<void>::success();
283}

344eve::Result<std::size_t> PrimitiveScene::updateMany(std::span<const PrimitiveBatchUpdate> updates) {
345 for (const PrimitiveBatchUpdate& update : updates) {
346 if (!matches(update.handle))
347 return eve::Result<std::size_t>::failure(eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle,
348 "primitive batch contains a stale handle",
349 "primitive", {}, "graphics"));
350 auto validation = validateDescriptor(update.descriptor);
351 if (!validation) return eve::Result<std::size_t>::failure(validation.status());
352 }
353 // Copy every owning payload before publishing any mutation. A later allocation
354 // failure must not leave earlier descriptors in the batch already replaced.
355 std::vector<PrimitiveBatchUpdate> prepared(updates.begin(), updates.end());
356 static_assert(std::is_nothrow_move_assignable_v<PrimitiveDescriptor3D>);

358 *slots_[update.handle.index()].descriptor = std::move(update.descriptor);
359 slots_[update.handle.index()].cache.reset();
360 }
361 return eve::Result<std::size_t>::success(updates.size());
362}

394eve::Result<void> PrimitiveScene::tryRender(PrimitiveSceneCanvas3D& destination) const {
395 for (const Slot& slot : slots_) {
396 if (!slot.descriptor || !slot.descriptor->visible) continue;
397 const PrimitiveDescriptor3D& descriptor = *slot.descriptor;
398 const bool hit = slot.cache.has_value();
399 if (!hit) {
400 PrimitiveSceneCanvas3D canvas(destination.context_, std::numeric_limits<std::size_t>::max());
401 canvas.concat(descriptor.transform);
402 std::visit(
403 [&](const auto& geometry) {
404 using Geometry = std::decay_t<decltype(geometry)>;
405 if constexpr (std::is_same_v<Geometry, PrimitivePolyline3D>) {
406 canvas.drawPolyline(geometry.points, geometry.closed, descriptor.paint);
407 } else if constexpr (std::is_same_v<Geometry, PrimitiveAabb3D>) {
408 canvas.drawAabb(geometry.minimum, geometry.maximum, descriptor.paint);
409 } else if constexpr (std::is_same_v<Geometry, PrimitiveObb3D>) {
410 canvas.drawObb(geometry.center, geometry.halfAxes, descriptor.paint);
411 } else if constexpr (std::is_same_v<Geometry, PrimitiveDisk3D>) {
412 canvas.drawDisk(geometry.center, geometry.normal, geometry.radius, descriptor.paint,
413 geometry.segments);
414 } else if constexpr (std::is_same_v<Geometry, PrimitiveArc3D>) {
415 canvas.drawArc(geometry.center, geometry.normal, geometry.zeroDirection, geometry.radius,
416 geometry.startRadians, geometry.sweepRadians, descriptor.paint,
417 geometry.segments);
418 } else if constexpr (std::is_same_v<Geometry, PrimitiveSphere3D>) {
419 canvas.drawSphere(geometry.center, geometry.radius, descriptor.paint, geometry.segments);
420 } else if constexpr (std::is_same_v<Geometry, PrimitiveCapsule3D>) {
421 canvas.drawCapsule(geometry.a, geometry.b, geometry.radius, descriptor.paint,
422 geometry.segments);
423 } else if constexpr (std::is_same_v<Geometry, PrimitiveCylinder3D>) {
424 canvas.drawCylinder(geometry.a, geometry.b, geometry.radius, descriptor.paint,
425 geometry.segments);
426 } else if constexpr (std::is_same_v<Geometry, PrimitiveCone3D>) {
427 canvas.drawCone(geometry.apex, geometry.axis, geometry.height, geometry.radius,
428 descriptor.paint, geometry.segments);
429 } else if constexpr (std::is_same_v<Geometry, PrimitiveGrid3D>) {
430 canvas.drawGrid(geometry.origin, geometry.axisU, geometry.axisV, geometry.cellsU,
431 geometry.cellsV, descriptor.paint);
432 } else if constexpr (std::is_same_v<Geometry, PrimitiveArrow3D>) {
433 canvas.drawArrow(geometry.from, geometry.to, geometry.headLength, geometry.headRadius,
434 descriptor.paint);
435 } else if constexpr (std::is_same_v<Geometry, PrimitiveFrustum3D>) {
436 canvas.drawFrustum(geometry.corners, descriptor.paint);
437 }
438 },
439 descriptor.geometry);
440 slot.cache = std::move(canvas);
441 }
442 const auto& cached = *slot.cache;
443 const auto count = cached.commands_.size() + cached.triangles_.size();
444 const auto used = destination.commands_.size() + destination.triangles_.size();
445 if (used > destination.hardCommandLimit_ || count > destination.hardCommandLimit_ - used) {
446 destination.statistics_.droppedCommands += count + cached.statistics_.droppedCommands;
447 return eve::Result<void>::failure(eve::Diagnostic::error(
448 eve::DiagnosticCode::Failed, "primitive scene command budget exceeded", "primitive", {}, "graphics"));
449 }
450 // Prepare all owning payloads and capacity before publishing this slot.
451 auto lines = cached.commands_;
452 auto triangles = cached.triangles_;
453 for (auto& command : lines) command.transform = destination.transform_ * command.transform;
454 for (auto& command : triangles) command.transform = destination.transform_ * command.transform;
455 destination.commands_.reserve(destination.commands_.size() + lines.size());
456 destination.triangles_.reserve(destination.triangles_.size() + triangles.size());
457 static_assert(std::is_nothrow_move_constructible_v<PolylineCommand3D>);
458 static_assert(std::is_nothrow_move_constructible_v<TriangleCommand3D>);
459 for (auto& command : lines) destination.commands_.push_back(std::move(command));
460 for (auto& command : triangles) destination.triangles_.push_back(std::move(command));
461 destination.statistics_.commandCount += cached.statistics_.commandCount;
462 destination.statistics_.segmentCount += cached.statistics_.segmentCount;
463 destination.statistics_.triangleCount += cached.statistics_.triangleCount;
464 destination.statistics_.cacheHits += hit ? 1u : 0u;
465 }
466 return eve::Result<void>::success();
467}

510template <class Mutator>
511eve::Result<PrimitiveUpdateStatus> mutatePrimitive(ScriptPrimitive3D* value, Mutator&& mutator) {
512 auto descriptor = primitiveDescriptor(value);
513 if (!descriptor) return eve::Result<PrimitiveUpdateStatus>::failure(descriptor.status());
514 PrimitiveDescriptor3D copy = std::move(descriptor).takeValue();
515 std::invoke(std::forward<Mutator>(mutator), copy);
516 auto scene = value->scene.lock();
517 if (!scene)
518 return eve::Result<PrimitiveUpdateStatus>::failure(
519 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "primitive scene owner no longer exists",
520 "primitive", {}, "graphics.primitive.binding"));
521 return scene->update(value->handle, std::move(copy));
522}

524ssq::Table makePrimitiveProxy(HSQUIRRELVM vm, const std::shared_ptr<PrimitiveScene>& scene,
525 eve::Result<PrimitiveHandle>&& result) {
526 if (!result) return eve::script::projectStatusResult(vm, result.status());
527 const PrimitiveHandle handle = std::move(result).takeValue();
528 auto object = eve::script::makeOwnedSquirrelInstance<ScriptPrimitive3D>(
529 vm, std::make_unique<ScriptPrimitive3D>(scene, handle));
530 if (!object) {
531 const eve::Status status = object.status();
532 object.ignore("failed to create owned primitive proxy");
533 scene->remove(handle).ignore("rollback failed primitive proxy allocation");
534 return eve::script::projectStatusResult(vm, status);
535 }
536 auto projected = eve::script::projectStatusResult(vm, eve::Status::success(eve::StatusCode::Applied));
537 projected.set("value", std::move(object).takeValue());
538 projected.set("ownership", std::string("owned"));
539 projected.set("owner", static_cast<std::int64_t>(handle.owner()));
540 projected.set("index", static_cast<std::int64_t>(handle.index()));
541 projected.set("generation", static_cast<std::int64_t>(handle.generation()));
542 return projected;
543}

554eve::Result<std::vector<glm::vec3>> scriptPoints(ssq::Array values, std::size_t minimum, std::size_t maximum) {
555 if (values.size() % 3 != 0 || values.size() / 3 < minimum || values.size() / 3 > maximum)
556 return eve::Result<std::vector<glm::vec3>>::failure(eve::Diagnostic::error(
557 eve::DiagnosticCode::InvalidArgument, "expected a flat xyz array with the required point count", "points",
558 {}, "graphics.primitive.binding"));
559 std::vector<glm::vec3> points;
560 points.reserve(values.size() / 3);
561 try {
562 for (std::size_t i = 0; i < values.size(); i += 3)
563 points.push_back({values.get<float>(i), values.get<float>(i + 1), values.get<float>(i + 2)});
564 } catch (const std::exception& error) {
565 return eve::Result<std::vector<glm::vec3>>::failure(eve::Diagnostic::error(
566 eve::DiagnosticCode::InvalidArgument, error.what(), "points", {}, "graphics.primitive.binding"));
567 }
568 return eve::Result<std::vector<glm::vec3>>::success(std::move(points));
569}

1033std::uint32_t resolveRadialSegments(const RadialTessellation& tessellation, float projectedRadiusPixels) {
1034 const bool validError = std::isfinite(tessellation.maxScreenErrorPixels) && tessellation.maxScreenErrorPixels > 0.f;
1035 EV_PARAM_CHECK(validError, "radial tessellation screen error must be finite and positive");
1036 if (tessellation.customSegments != 0) {
1037 const bool validCustom = tessellation.customSegments >= 3 && tessellation.customSegments <= 4096;
1038 EV_PARAM_CHECK(validCustom, "custom radial segment count must be in [3, 4096]");
1039 return tessellation.customSegments;
1040 }
1041 switch (tessellation.quality) {
1042 case PrimitiveQuality::Low: return 16;
1043 case PrimitiveQuality::Medium: return 32;
1044 case PrimitiveQuality::High: return 64;
1045 case PrimitiveQuality::Adaptive: break;
1046 }
1047 const bool validRadius = std::isfinite(projectedRadiusPixels) && projectedRadiusPixels >= 0.f;
1048 EV_PARAM_CHECK(validRadius, "adaptive radial projected radius must be finite and non-negative");
1049 if (projectedRadiusPixels <= tessellation.maxScreenErrorPixels) return 8;
1050 const float cosine = glm::clamp(1.f - tessellation.maxScreenErrorPixels / projectedRadiusPixels, -1.f, 1.f);
1051 const float halfAngle = std::acos(cosine);
1052 if (halfAngle <= 1e-6f) return 256;
1053 return std::clamp(static_cast<std::uint32_t>(std::ceil(glm::pi<float>() / halfAngle)), 8u, 256u);
1054}
