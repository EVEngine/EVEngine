33Result<void> replace(Graphics& graphics, Shader& shader, const std::string& vertex, const std::string& fragment,
34 const Value& descriptions, const std::string& constants, const std::string& instances = {}) {
35 try {
36 if (!descriptions.isArray() || !descriptions.arraySize() || descriptions.arraySize() > 16)
37 throw std::runtime_error("Expected 1-16 image resource descriptions");
38 std::vector<uint32_t> vert;
39 if (!vertex.empty()) {
40 auto stage = readShaderStageFile(vertex);
41 if (!stage) return Result<void>::failure(stage.status());
42 vert = std::move(stage).takeValue();
43 }
44 auto frag = readShaderStageFile(fragment);
45 if (!frag) return Result<void>::failure(frag.status());
46 std::vector<ShaderImageInput> images;
47 std::vector<std::shared_ptr<const filesystem::FileData>> files;
48 constexpr std::string_view keys[] = {"binding", "format", "dimension", "width", "height", "layers",
49 "mips", "path", "filter", "wrapU", "wrapV", "anisotropy"};
50 constexpr std::pair<std::string_view, ShaderImageFormat> formats[] = {
51 {"r8-unorm", ShaderImageFormat::R8}, {"rg8-unorm", ShaderImageFormat::RG8},
52 {"r16-unorm", ShaderImageFormat::R16}, {"rgba8-unorm", ShaderImageFormat::RGBA8},
53 {"bgra8-unorm", ShaderImageFormat::BGRA8}, {"rgba8-srgb", ShaderImageFormat::RGBA8Srgb},
54 {"bgra8-srgb", ShaderImageFormat::BGRA8Srgb}, {"bc3-unorm", ShaderImageFormat::BC3},
55 {"bc1-unorm", ShaderImageFormat::BC1}, {"bc1-srgb", ShaderImageFormat::BC1Srgb},
56 {"bc3-srgb", ShaderImageFormat::BC3Srgb}, {"bc7-unorm", ShaderImageFormat::BC7},
57 {"bc7-srgb", ShaderImageFormat::BC7Srgb}};
58 size_t totalBytes = 0;
59 for (size_t i = 0; i < descriptions.arraySize(); ++i) {
60 const auto& row = descriptions.at(i);
61 if (!row.isObject()) throw std::runtime_error("Image description must be an object");
62 for (const auto& key : row.keys())
63 if (std::find(std::begin(keys), std::end(keys), key) == std::end(keys))
64 throw std::runtime_error("Unknown image resource field: " + key);
65 ShaderImageInput image;
66 image.binding = number(row, "binding");
67 image.width = number(row, "width");
68 image.height = number(row, "height");
69 image.layers = number(row, "layers");
70 image.mipLevels = number(row, "mips");
71 const auto format = field(row, "format", Value::Type::String).asString();
72 auto found = std::find_if(std::begin(formats), std::end(formats),
73 [&](const auto& item) { return item.first == format; });
74 if (found == std::end(formats)) throw std::runtime_error("Unknown image format: " + format);
75 image.format = found->second;
76 const auto dimension = field(row, "dimension", Value::Type::String).asString();
77 if (dimension == "2d")
78 image.dimension = ShaderImageDimension::Image2D;
79 else if (dimension == "array2d")
80 image.dimension = ShaderImageDimension::Array2D;
81 else if (dimension == "cube")
82 image.dimension = ShaderImageDimension::Cube;
83 else
84 throw std::runtime_error("Unknown image view dimension: " + dimension);
85 auto filter = number(row, "filter");
86 auto wrapU = number(row, "wrapU"), wrapV = number(row, "wrapV");
87 if (filter > 2 || wrapU > 1 || wrapV > 1) throw std::runtime_error("Unknown resource sampler mode");
88 image.sampler.min = image.sampler.mag = filter == 0 ? FilterMode::Nearest : FilterMode::Linear;
89 image.sampler.mipmap = image.mipLevels <= 1 ? MipmapMode::Disabled
90 : filter == 2 ? MipmapMode::Linear
91 : MipmapMode::Nearest;
92 image.sampler.repeatU = wrapU == 0;
93 image.sampler.repeatV = wrapV == 0;
94 image.sampler.maxAnisotropy = float(number(row, "anisotropy"));
95 const auto path = field(row, "path", Value::Type::String).asString();
96 files.push_back(readBytes(path, 1024ull * 1024 * 1024));
97 const auto& bytes = *files.back();
98 totalBytes += bytes.getSize();
99 if (totalBytes > 2ull * 1024 * 1024 * 1024) throw std::runtime_error("Resource set exceeds 2 GiB");
100 image.bytes = {static_cast<const std::byte*>(bytes.getData()), size_t(bytes.getSize())};
101 image.contentOwner = files.back();
102 images.push_back(image);
103 }
104 std::shared_ptr<const filesystem::FileData> uniform;
105 std::shared_ptr<const filesystem::FileData> matrices;
106 ShaderResourceInputs inputs{images, {}};
107 if (!constants.empty()) {
108 uniform = readBytes(constants, 65536);
109 inputs.constants = {static_cast<const std::byte*>(uniform->getData()), size_t(uniform->getSize())};
110 }
111 if (!instances.empty()) {
112 matrices = readBytes(instances, 64ull * 1024 * 1024);
113 inputs.instanceMatrices = {static_cast<const std::byte*>(matrices->getData()), size_t(matrices->getSize())};
114 }
115 return graphics.replaceMeshShaderResources(shader, vert, frag.value(), inputs);
116 } catch (const std::exception& error) {
117 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, error.what(), "resources"));
118 }
