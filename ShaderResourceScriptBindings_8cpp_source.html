32 auto value = field(object, name, Value::Type::Int64).asInt();
33 if (value < 0 || uint64_t(value) > UINT32_MAX) throw std::runtime_error("Image integer out of range: " + name);

44 const Value& descriptions, const std::string& constants, const std::string& instances = {}) {
45 try {
46 if (!descriptions.isArray() || descriptions.arraySize() > 16)

95 image.format = found->second;

99 else if (dimension == "array2d")
100 image.dimension = ShaderImageDimension::Array2D;
101 else if (dimension == "cube")
102 image.dimension = ShaderImageDimension::Cube;
103 else if (dimension == "3d")
104 image.dimension = ShaderImageDimension::Image3D;
105 else
106 throw std::runtime_error("Unknown image view dimension: " + dimension);
107 if (row.find("depth") || image.dimension == ShaderImageDimension::Image3D)
108 image.depth = number(row, "depth");
109 auto filter = number(row, "filter");
110 auto wrapU = number(row, "wrapU"), wrapV = number(row, "wrapV");
111 if (filter > 2 || wrapU > 1 || wrapV > 1) throw std::runtime_error("Unknown resource sampler mode");
112 image.sampler.min = image.sampler.mag = filter == 0 ? FilterMode::Nearest : FilterMode::Linear;
113 image.sampler.mipmap = image.mipLevels <= 1 ? MipmapMode::Disabled
114 : filter == 2 ? MipmapMode::Linear
115 : MipmapMode::Nearest;
116 image.sampler.repeatU = wrapU == 0;
117 image.sampler.repeatV = wrapV == 0;
118 if (row.find("wrapW") || image.dimension == ShaderImageDimension::Image3D) {
119 const auto wrapW = number(row, "wrapW");
120 if (wrapW > 1) throw std::runtime_error("Unknown resource W sampler mode");
121 image.sampler.repeatW = wrapW == 0;
122 }
123 image.sampler.maxAnisotropy = float(number(row, "anisotropy"));
124 const auto path = field(row, "path", Value::Type::String).asString();
125 files.push_back(readBytes(path, 1024ull * 1024 * 1024));
126 const auto& bytes = *files.back();
127 totalBytes += bytes.getSize();
128 if (totalBytes > 2ull * 1024 * 1024 * 1024) throw std::runtime_error("Resource set exceeds 2 GiB");
129 image.bytes = {static_cast<const std::byte*>(bytes.getData()), size_t(bytes.getSize())};
130 image.contentOwner = files.back();
131 images.push_back(image);
132 }
133 std::shared_ptr<const filesystem::FileData> uniform;
134 std::shared_ptr<const filesystem::FileData> matrices;
135 ShaderResourceInputs inputs{images, {}};
136 if (!constants.empty()) {
137 uniform = readBytes(constants, 65536);
138 inputs.constants = {static_cast<const std::byte*>(uniform->getData()), size_t(uniform->getSize())};
139 }
140 if (!instances.empty()) {
141 matrices = readBytes(instances, 64ull * 1024 * 1024);
142 inputs.instanceMatrices = {static_cast<const std::byte*>(matrices->getData()), size_t(matrices->getSize())};
143 }
144 return graphics.replaceMeshShaderResources(shader, vert, frag.value(), inputs);
145 } catch (const std::exception& error) {
146 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, error.what(), "resources"));
147 }

313Result<std::vector<ShaderImageRegion>> shaderImageRegions(const ShaderImageInput& image) {
314 const auto invalid = [](const char* message) {
315 return Result<std::vector<ShaderImageRegion>>::failure(
316 Diagnostic::error(DiagnosticCode::InvalidArgument, message, "image"));
317 };
318 if (!image.width || !image.height || image.width > 32768 || image.height > 32768 || !image.layers ||
319 image.layers > 2048 || !image.depth || image.depth > 2048 || !image.mipLevels ||
320 image.mipLevels > static_cast<uint32_t>(std::bit_width(std::max({image.width, image.height, image.depth}))) ||
321 image.binding > 31)
322 return invalid("Invalid shader image dimensions, mip count or binding");
323 if (image.dimension != ShaderImageDimension::Image3D && image.depth != 1)
324 return invalid("Non-volume images require depth one");
325 switch (image.dimension) {
326 case ShaderImageDimension::Image2D:
327 if (image.layers != 1) return invalid("A 2D image must have one layer");
328 break;
329 case ShaderImageDimension::Array2D: break;
330 case ShaderImageDimension::Image3D:
331 if (image.layers != 1 || image.width > 2048 || image.height > 2048)
332 return invalid("A volume requires one layer and at most 2048 texels per dimension");
333 break;
334 case ShaderImageDimension::Cube:
335 if (image.layers != 6 || image.width != image.height) return invalid("A cube image needs six square faces");
336 break;
337 default: return invalid("Unknown image dimension");
338 }
339 std::size_t block = 1, stride = 0;
340 switch (image.format) {
341 case ShaderImageFormat::R8: stride = 1; break;
342 case ShaderImageFormat::RG8:
343 case ShaderImageFormat::R16: stride = 2; break;
344 case ShaderImageFormat::RGBA8:
345 case ShaderImageFormat::BGRA8:
346 case ShaderImageFormat::RGBA8Srgb:
347 case ShaderImageFormat::BGRA8Srgb: stride = 4; break;
348 case ShaderImageFormat::RGBA16Unorm:
349 case ShaderImageFormat::RGBA16Float: stride = 8; break;
350 case ShaderImageFormat::RGBA32Float: stride = 16; break;
351 case ShaderImageFormat::BC1:
352 case ShaderImageFormat::BC1Srgb:
353 block = 4;
354 stride = 8;
355 break;
356 case ShaderImageFormat::BC3:
357 case ShaderImageFormat::BC7:
358 case ShaderImageFormat::BC7Srgb:
359 case ShaderImageFormat::BC3Srgb:
360 block = 4;
361 stride = 16;
362 break;
363 default: return invalid("Unknown image format");
364 }
365 if (image.dimension == ShaderImageDimension::Image3D && block != 1)
366 return invalid("Block-compressed volume images are unsupported");
367 if (!std::isfinite(image.sampler.maxAnisotropy) || image.sampler.maxAnisotropy < 1 ||
368 !std::isfinite(image.sampler.lodBias) || !std::isfinite(image.sampler.minLod) ||
369 !std::isfinite(image.sampler.maxLod) || image.sampler.minLod < 0 || image.sampler.maxLod < image.sampler.minLod)
370 return invalid("Invalid shader image sampler");
371 std::vector<ShaderImageRegion> result;
372 std::size_t offset = 0;
373 for (std::uint32_t layer = 0; layer < image.layers; ++layer) {
374 for (std::uint32_t mip = 0; mip < image.mipLevels; ++mip) {
375 const auto width = std::max(1u, image.width >> mip);
376 const auto height = std::max(1u, image.height >> mip);
377 const auto depth = std::max(1u, image.depth >> mip);
378 const uint64_t size =
379 uint64_t((width + block - 1) / block) * ((height + block - 1) / block) * stride * depth;
380 if (size > 1024ull * 1024 * 1024 || offset > 1024ull * 1024 * 1024 - size)
381 return invalid("Shader image exceeds 1 GiB upload limit");
382 result.push_back({layer, mip, width, height, offset, static_cast<size_t>(size), depth});
383 offset += static_cast<size_t>(size);
384 }
385 }
386 if (image.bytes.size() != offset) return invalid("Shader image payload size does not match its layout");
387 return Result<std::vector<ShaderImageRegion>>::success(std::move(result));
388}
389Result<std::uint32_t> shaderInstanceMatrixCount(std::span<const std::byte> bytes) {
390 auto bad = [](const char* message) {
391 return Result<std::uint32_t>::failure(
392 Diagnostic::error(DiagnosticCode::InvalidArgument, message, "shader.instances"));
393 };
394 if (bytes.size() % 64 || bytes.size() > 64ull * 1024 * 1024)
395 return bad("Instance matrices require complete 64-byte records, at most 64 MiB");
396 for (std::size_t offset = 0; offset < bytes.size(); offset += 64) {
397 float m[16];
398 std::memcpy(m, bytes.data() + offset, sizeof(m));
399 for (float value : m)
400 if (!std::isfinite(value)) return bad("Nonfinite instance transform");
401 if (m[3] != 0 || m[7] != 0 || m[11] != 0 || m[15] != 1) return bad("Instance transform must be affine");
402 const double determinant = double(m[0]) * (double(m[5]) * m[10] - double(m[9]) * m[6]) -
403 double(m[4]) * (double(m[1]) * m[10] - double(m[9]) * m[2]) +
404 double(m[8]) * (double(m[1]) * m[6] - double(m[5]) * m[2]);
405 if (!std::isfinite(determinant) || determinant == 0) return bad("Instance transform must be invertible");
406 }
407 return Result<std::uint32_t>::success(static_cast<std::uint32_t>(bytes.size() / 64));
408}

414Result<Value> buildGrassPoints(double radius, double extent, int seed) {
415 if (!std::isfinite(radius) || !std::isfinite(extent) || radius <= 0 || extent <= 0 || radius < extent / 1000 ||
416 seed < 0)
417 return Result<Value>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
418 "Invalid bounded grass scatter dimensions or seed", "scatter"));
419 const double cell = radius / std::sqrt(2.0);
420 const int side = int(std::ceil(2 * extent / cell)) + 1;
421 std::vector<int> grid(size_t(side) * size_t(side), -1), active;
422 struct Point {
423 double x, z;
424 };
425 std::vector<Point> points;
426 int64_t state = int64_t(seed) % 2147483646 + 1;
427 auto next = [&]() {
428 state = state * 48271 % 2147483647;
429 return double(state) / 2147483647.0;
430 };
431 auto insert = [&](double x, double z) {
432 const auto id = int(points.size());
433 grid[size_t(int((z + extent) / cell)) * size_t(side) + size_t(int((x + extent) / cell))] = id;
434 points.push_back({x, z});
435 active.push_back(id);
436 };
437 const double firstX = (next() * 2 - 1) * extent, firstZ = (next() * 2 - 1) * extent;
438 insert(firstX, firstZ);
439 while (!active.empty()) {
440 const auto index = size_t(next() * double(active.size()));
441 const auto p = points[size_t(active[index])];
442 bool found = false;
443 for (int trial = 0; trial < 30; ++trial) {
444 const double angle = next() * 6.283185307179586, distance = radius * std::sqrt(1 + 3 * next());
445 const double x = p.x + std::cos(angle) * distance, z = p.z + std::sin(angle) * distance;
446 if (std::abs(x) > extent || std::abs(z) > extent) continue;
447 const int gx = int((x + extent) / cell), gz = int((z + extent) / cell);
448 bool valid = true;
449 for (int dz = -2; dz <= 2 && valid; ++dz)
450 for (int dx = -2; dx <= 2; ++dx) {
451 const int cx = gx + dx, cz = gz + dz;
452 if (cx < 0 || cz < 0 || cx >= side || cz >= side) continue;
453 const auto other = grid[size_t(cz) * size_t(side) + size_t(cx)];
454 if (other < 0) continue;
455 const auto q = points[size_t(other)];
456 const double ox = x - q.x, oz = z - q.z;
457 if (ox * ox + oz * oz < radius * radius) {
458 valid = false;
459 break;
460 }
461 }
462 if (valid) {
463 insert(x, z);
464 found = true;
465 break;
466 }
467 }
468 // Two coordinates per point plus the array container stay inside the
469 // common Squirrel Value projection's 100,000-element limit.
470 if (points.size() > 49999)
471 return Result<Value>::failure(
472 Diagnostic::error(DiagnosticCode::InvalidArgument, "Grass scatter exceeds 49,999 points", "scatter"));
473 if (!found) {
474 active[index] = active.back();
475 active.pop_back();
476 }
477 }
478 Value::Array output;
479 output.reserve(points.size() * 2);
480 for (const auto& p : points) {
481 output.emplace_back(p.x);
482 output.emplace_back(p.z);
483 }
484 return Result<Value>::success(Value(std::move(output)));
485}

487// CPU-authored vector blades rasterized once per field. No texture file or
488// shader-time shape generation is involved; RGB remains untinted white.
489std::vector<std::byte> grassMaskPixels() {
490 constexpr int size = 96, width = size * 4, height = size * 2, samples = 3;
491 struct Blade {
492 float root, height, bend, width;
493 };
494 constexpr Blade blades[2][4] = {
495 {{0.18f, 0.65f, -0.17f, 0.075f}, {0.48f, 0.96f, 0.08f, 0.10f}, {0.79f, 0.76f, 0.15f, 0.085f}, {}},
496 {{0.12f, 0.50f, -0.15f, 0.07f},
497 {0.32f, 0.84f, -0.10f, 0.065f},
498 {0.57f, 0.97f, 0.17f, 0.085f},
499 {0.83f, 0.61f, 0.16f, 0.075f}}};
500 std::vector<std::byte> pixels(width * height * 4, std::byte{255});
501 for (int variant = 0; variant < 2; ++variant)
502 for (int frame = 0; frame < 4; ++frame) {
503 constexpr float pi = 3.14159265358979323846f;
504 const float sway = std::sin(float(frame) * pi * 0.5f) * 0.075f;
505 const float flutter = std::cos(float(frame) * pi * 0.5f) * 0.12f;
506 for (int y = 0; y < size; ++y)
507 for (int x = 0; x < size; ++x) {
508 int covered = 0;
509 for (int sy = 0; sy < samples; ++sy)
510 for (int sx = 0; sx < samples; ++sx) {
511 const float u = (float(x) + (float(sx) + 0.5f) / samples) / size;
512 const float v = 1.f - (float(y) + (float(sy) + 0.5f) / samples) / size;
513 for (int leaf = 0; leaf < (variant == 0 ? 3 : 4); ++leaf) {
514 const auto& b = blades[variant][leaf];
515 const float t = v / b.height;
516 if (t < 0.f || t > 1.f) continue;
517 const float center = b.root + (b.bend + sway) * t * t;
518 const float half = b.width * std::pow(1.f - t, 0.8f) * (1.f + flutter * t);
519 if (std::abs(u - center) <= half) {
520 ++covered;
521 break;
522 }
523 }
524 }
525 const auto offset = size_t(((variant * size + y) * width + frame * size + x) * 4 + 3);
526 pixels[offset] = std::byte((covered * 255 + samples * samples / 2) / (samples * samples));
527 }
528 }
529 return pixels;
530}

532ResultRef<Shader> buildGrassShader(Graphics& graphics, ssq::Array records, const std::string& pass) {
533 using namespace detail::grass_spv;
534 try {
535 if (graphics.getBackendName() != "vulkan")
536 return ResultRef<Shader>::failure(
537 Diagnostic::error(DiagnosticCode::Unsupported, "Stylized grass requires Vulkan", "backend"));
538 if (pass != "grass" && pass != "ground")
539 return ResultRef<Shader>::failure(
540 Diagnostic::error(DiagnosticCode::InvalidArgument, "Expected grass or ground pass", "pass"));
541 if (records.size() == 0 || records.size() % 16 != 0 || records.size() > 64u * 1024u * 1024u / sizeof(float))
542 return ResultRef<Shader>::failure(Diagnostic::error(
543 DiagnosticCode::InvalidArgument, "Expected nonempty affine matrix records within 64 MiB", "instances"));
544 std::vector<float> matrices;
545 matrices.reserve(records.size());
546 for (size_t i = 0; i < records.size(); ++i) {
547 const auto type = records.get<ssq::Object>(i).getType();
548 if (type != ssq::Type::FLOAT && type != ssq::Type::INTEGER)
549 return ResultRef<Shader>::failure(
550 Diagnostic::error(DiagnosticCode::InvalidArgument, "Matrix elements must be numbers", "instances"));
551 matrices.push_back(type == ssq::Type::INTEGER ? float(records.get<SQInteger>(i)) : records.get<float>(i));
552 }
553 ShaderResourceInputs inputs{};
554 inputs.instanceMatrices = std::as_bytes(std::span(matrices));
555 auto count = shaderInstanceMatrixCount(inputs.instanceMatrices);
556 if (!count) return ResultRef<Shader>::failure(count.status());
557 std::vector<std::byte> pixels;
558 ShaderImageInput image;
559 if (pass == "grass") {
560 pixels = grassMaskPixels();
561 image.binding = 0;
562 image.width = 384;
563 image.height = 192;
564 image.sampler.min = image.sampler.mag = FilterMode::Linear;
565 image.sampler.repeatU = image.sampler.repeatV = false;
566 image.bytes = pixels;
567 inputs.images = std::span(&image, 1);
568 }
569 const std::vector<uint32_t> vertex =
570 pass == "grass" ? std::vector<uint32_t>(std::begin(meadow_vert_spv), std::end(meadow_vert_spv))
571 : std::vector<uint32_t>(std::begin(ground_vert_spv), std::end(ground_vert_spv));
572 const std::vector<uint32_t> fragment =
573 pass == "grass" ? std::vector<uint32_t>(std::begin(meadow_frag_spv), std::end(meadow_frag_spv))
574 : std::vector<uint32_t>(std::begin(ground_frag_spv), std::end(ground_frag_spv));
575 auto* shader = graphics.newMeshShaderFromSpv({std::begin(bootstrap_vert_spv), std::end(bootstrap_vert_spv)},
576 {std::begin(bootstrap_frag_spv), std::end(bootstrap_frag_spv)});
577 if (!shader)
578 return ResultRef<Shader>::failure(
579 Diagnostic::error(DiagnosticCode::Failed, "Shader allocation failed", "shader"));
580 auto uploaded = graphics.replaceMeshShaderResources(*shader, vertex, fragment, inputs);
581 if (!uploaded) return ResultRef<Shader>::failure(uploaded.status());
582 return ResultRef<Shader>::success(std::ref(*shader));
583 } catch (const std::exception& error) {
584 return ResultRef<Shader>::failure(Diagnostic::error(DiagnosticCode::Failed, error.what(), "stylized-grass"));
585 }
586}

588Result<std::vector<uint32_t>> shaderWords(ssq::Array array) {
589 std::vector<uint32_t> words;
590 words.reserve(array.size());
591 for (size_t i = 0; i < array.size(); ++i) {
592 if (array.get<ssq::Object>(i).getType() != ssq::Type::INTEGER)
593 return Result<std::vector<uint32_t>>::failure(
594 Diagnostic::error(DiagnosticCode::InvalidArgument, "SPIR-V words must be integers"));

597 return Result<std::vector<uint32_t>>::failure(
598 Diagnostic::error(DiagnosticCode::InvalidArgument, "SPIR-V words must be unsigned 32-bit integers"));
599 words.push_back(static_cast<uint32_t>(word));
600 }
601 return Result<std::vector<uint32_t>>::success(std::move(words));
602}

606Result<std::vector<std::uint32_t>> readShaderStageFile(const std::string& path) {
607 try {
608 auto* fs = filesystem::Filesystem::create();
609 std::unique_ptr<filesystem::FileData> bytes(fs->read(path));
610 if (!bytes || bytes->getSize() < 20 || bytes->getSize() % 4 != 0)
611 return Result<std::vector<std::uint32_t>>::failure(Diagnostic::error(
612 DiagnosticCode::ParseError, "SPIR-V must contain a complete word-aligned header", path));
613 std::vector<std::uint32_t> words(bytes->getSize() / 4);
614 std::memcpy(words.data(), bytes->getData(), bytes->getSize());
615 if (words[0] != 0x07230203)
616 return Result<std::vector<std::uint32_t>>::failure(
617 Diagnostic::error(DiagnosticCode::ParseError, "SPIR-V magic mismatch", path));
618 return Result<std::vector<std::uint32_t>>::success(std::move(words));
619 } catch (const std::exception& error) {
620 return Result<std::vector<std::uint32_t>>::failure(
621 Diagnostic::error(DiagnosticCode::Failed, error.what(), path));
622 }
623}

628ResultRef<Shader> loadMeshShader(Graphics& graphics, const std::string& vertex, const std::string& fragment) {
629 if (graphics.getBackendName() != "vulkan")
630 return ResultRef<Shader>::failure(Diagnostic::error(
631 DiagnosticCode::Unsupported, "SPIR-V mesh shaders require the Vulkan backend", "backend"));
632 if (fragment.empty())
633 return ResultRef<Shader>::failure(
634 Diagnostic::error(DiagnosticCode::InvalidArgument, "fragment path must not be empty", "fragment"));
635 std::vector<std::uint32_t> vert;
636 if (!vertex.empty()) {
637 auto decoded = detail::readShaderStageFile(vertex);
638 if (!decoded) return ResultRef<Shader>::failure(decoded.status());
639 vert = std::move(decoded).takeValue();
640 }
641 auto frag = detail::readShaderStageFile(fragment);
642 if (!frag) return ResultRef<Shader>::failure(frag.status());
643 try {
644 auto* shader = graphics.newMeshShaderFromSpv(vert, frag.value());
645 if (!shader)
646 return ResultRef<Shader>::failure(
647 Diagnostic::error(DiagnosticCode::Failed, "Mesh shader allocation failed", "shader"));
648 return ResultRef<Shader>::success(std::ref(*shader));
649 } catch (const std::exception& error) {
650 return ResultRef<Shader>::failure(Diagnostic::error(DiagnosticCode::Failed, error.what(), "shader"));
651 }
