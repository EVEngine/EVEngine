50Result<std::vector<Cell>> bindingCellsJson(const std::string& json) {
51 auto value = Value::fromJson(json);
52 if (!value) return Result<std::vector<Cell>>::failure(value.status());
53 const auto* rows = value.value().getIf<Value::Array>();
54 if (!rows || rows->empty())
55 return failure<std::vector<Cell>>(DiagnosticCode::InvalidArgument,
56 "objective cells JSON must be a non-empty array", "cellsJson");
57 std::vector<Cell> result;
58 result.reserve(rows->size());
59 for (std::size_t index = 0; index < rows->size(); ++index) {
60 const auto* tuple = (*rows)[index].getIf<Value::Array>();
61 if (!tuple || tuple->size() != 3 || !(*tuple)[0].isInt64() || !(*tuple)[1].isInt64() ||
62 !(*tuple)[2].isInt64())
63 return failure<std::vector<Cell>>(DiagnosticCode::InvalidArgument,
64 "objective cell must be [x,y,layer] integers",
65 "cellsJson[" + std::to_string(index) + "]");
66 const auto convert = [&](std::size_t coordinate) -> Result<int> {
67 const auto value = (*tuple)[coordinate].asInt();
68 if (value < std::numeric_limits<int>::min() || value > std::numeric_limits<int>::max())
69 return failure<int>(DiagnosticCode::InvalidArgument, "objective cell coordinate is out of range",
70 "cellsJson[" + std::to_string(index) + "]");
71 return Result<int>::success(static_cast<int>(value));
72 };
73 auto x = convert(0);
74 auto y = convert(1);
75 auto layer = convert(2);
76 if (!x) return Result<std::vector<Cell>>::failure(x.status());
77 if (!y) return Result<std::vector<Cell>>::failure(y.status());
78 if (!layer) return Result<std::vector<Cell>>::failure(layer.status());
79 result.push_back({x.value(), y.value(), layer.value()});
80 }
81 return Result<std::vector<Cell>>::success(std::move(result));
82}

244Result<GameplayObservation> Tactics::observeGameplay(const GameplaySession& session,
245 SubjectRef instance) const {
246 Battle* battle = resolveBattle(instance);
247 if (!battle)
248 return failure<GameplayObservation>(DiagnosticCode::NotFound, "tactics gameplay instance was not found",
249 "instance");
250 if (session.access == GameplayAccess::PlayerEquivalent) {
251 const bool ownsParticipant = std::any_of(battle->turn()->units.begin(), battle->turn()->units.end(),
252 [&](const ecs::EntityHandle& handle) {
253 auto* unit = dynamic_cast<TacticalUnit*>(ecs::try_get(handle));
254 return unit && controls(session, unit->identity()->subject);
255 });
256 if (!ownsParticipant)
257 return failure<GameplayObservation>(DiagnosticCode::PreconditionViolation,
258 "session controls no participant in this tactics battle",
259 "session.controlledSubjects");
260 }
261
262 Value::Array units;
263 for (const auto& handle : battle->turn()->units) {
264 auto* unit = dynamic_cast<TacticalUnit*>(ecs::try_get(handle));
265 if (!unit) continue;
266 units.emplace_back(Value::Object{
267 {"acted", Value(unit->turn()->acted)},
268 {"actionPoints", Value(unit->turn()->actionPoints)},
269 {"alive", Value(unit->turn()->alive)},
270 {"cell", cellValue(unit->position()->cell)},
271 {"controlled", Value(controls(session, unit->identity()->subject))},
272 {"facing", Value(unit->position()->facing)},
273 {"movePoints", Value(unit->turn()->movePoints)},
274 {"subject", Value(unit->identity()->subject.format())},
275 });
276 }
277 const auto active = battle->turn()->activeUnit
278 ? dynamic_cast<TacticalUnit*>(ecs::try_get(*battle->turn()->activeUnit))
279 : nullptr;
280 GameplayObservation observation;
281 observation.domain = gameplayId("gameplay:tactics");
282 observation.instance = instance;
283 observation.tick = battle->turn()->tick;
284 observation.revision = battle->turn()->revision.value();
285 observation.state = Value(Value::Object{
286 {"activeUnit", Value(active ? active->identity()->subject.format() : std::string{})},
287 {"phase", Value(std::string(phaseName(battle->turn()->phase)))},
288 {"round", Value(static_cast<std::int64_t>(battle->turn()->round))},
289 {"status", Value(std::string(statusName(battle->turn()->status)))},
290 {"units", Value(std::move(units))},
291 });
292 return Result<GameplayObservation>::success(std::move(observation));
293}

324Result<GameplayCommandReceipt> Tactics::submitGameplay(const GameplaySession& session, SubjectRef instance,
325 const GameplayCommand& command) {
326 Battle* battle = resolveBattle(instance);
327 if (!battle)
328 return failure<GameplayCommandReceipt>(DiagnosticCode::NotFound, "tactics gameplay instance was not found",
329 "instance");
330 if (command.id.empty())
331 return failure<GameplayCommandReceipt>(DiagnosticCode::InvalidArgument, "command id must not be empty",
332 "command.id");
333 if (!controls(session, command.subject))
334 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation,
335 "session does not control the command subject", "command.subject");
336 if (command.observedTick != battle->turn()->tick ||
337 command.expectedRevision != battle->turn()->revision.value())
338 return failure<GameplayCommandReceipt>(DiagnosticCode::Conflict,
339 "gameplay command was based on a stale observation",
340 "command.expectedRevision");
341
342 Value details(Value::Object{});
343 if (command.action == gameplayId("tactics:move")) {
344 auto x = integerParameter(command.parameters, "x");
345 auto y = integerParameter(command.parameters, "y");
346 auto layer = integerParameter(command.parameters, "layer");
347 if (!x) return Result<GameplayCommandReceipt>::failure(x.status());
348 if (!y) return Result<GameplayCommandReceipt>::failure(y.status());
349 if (!layer) return Result<GameplayCommandReceipt>::failure(layer.status());
350 auto moved = BattleSystem::moveUnit(*battle, command.subject, {x.value(), y.value(), layer.value()},
351 battle->turn()->tick);
352 if (!moved) return Result<GameplayCommandReceipt>::failure(moved.status());
353 const MoveReceipt receipt = std::move(moved).takeValue();
354 details = Value(Value::Object{{"cost", Value(receipt.cost)},
355 {"remainingMovePoints", Value(receipt.remainingMovePoints)}});
356 } else if (command.action == gameplayId("tactics:face")) {
357 auto facing = integerParameter(command.parameters, "facing");
358 if (!facing) return Result<GameplayCommandReceipt>::failure(facing.status());
359 auto faced = BattleSystem::faceUnit(*battle, command.subject, facing.value(), battle->turn()->tick);
360 if (!faced) return Result<GameplayCommandReceipt>::failure(faced.status());
361 } else if (command.action == gameplayId("tactics:wait")) {
362 auto waited = BattleSystem::waitUnit(*battle, command.subject, battle->turn()->tick);
363 if (!waited) return Result<GameplayCommandReceipt>::failure(waited.status());
364 } else if (command.action == gameplayId("tactics:end-turn")) {
365 auto ended = BattleSystem::endTurn(*battle, command.subject);
366 if (!ended) return Result<GameplayCommandReceipt>::failure(ended.status());
367 } else {
368 return failure<GameplayCommandReceipt>(DiagnosticCode::Unsupported,
369 "unsupported tactics gameplay action", "command.action");
370 }
371
372 GameplayCommandReceipt receipt;
373 receipt.commandId = command.id;
374 receipt.executionId = battle->commands()->values.empty()
375 ? std::string{}
376 : "tactics:" + std::to_string(battle->commands()->values.back().sequence);
377 receipt.acceptedTick = battle->turn()->tick;
378 receipt.resultingRevision = battle->turn()->revision.value();
379 receipt.details = std::move(details);
380 return Result<GameplayCommandReceipt>::success(std::move(receipt), Status::success(StatusCode::Applied));
381}
