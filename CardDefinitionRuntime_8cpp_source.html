42eve::Result<CardRuntimeState> parseState(const eve::definitions::Definition& definition,
43 const eve::DefinitionRef& reference) {
44 auto parsed = eve::Value::fromJson(definition.json);
45 if (!parsed) return eve::Result<CardRuntimeState>::failure(parsed.status());
46 auto value = std::move(parsed).takeValue();
47 const auto* object = value.getIf<eve::Value::Object>();
48 if (!object) return invalid<CardRuntimeState>("card definition payload must be an object", "json");
49
50 CardRuntimeState state;
51 state.name = std::string(reference.id().name());
52 if (const auto* value = field(*object, "name")) {
53 const auto* text = value->getIf<std::string>();
54 if (!text) return invalid<CardRuntimeState>("card name must be a string", "name");
55 state.name = *text;
56 }
57 if (const auto* value = field(*object, "kind")) {
58 const auto* text = value->getIf<std::string>();
59 if (!text) return invalid<CardRuntimeState>("card kind must be a string", "kind");
60 state.kind = *text;
61 }
62 if (!readInt(*object, "cost", state.cost) || !readInt(*object, "attack", state.attack) ||
63 !readInt(*object, "health", state.maxHealth))
64 return invalid<CardRuntimeState>("card cost, attack and health must be Int64 fields", "stats");
65 if (state.cost < 0 || state.attack < 0 || state.maxHealth < 0)
66 return invalid<CardRuntimeState>("card stats must not be negative", "stats");
67 state.currentHealth = state.maxHealth;
68
69 if (const auto* value = field(*object, "tint")) {
70 const auto* array = value->getIf<eve::Value::Array>();
71 if (!array || array->size() < 3)
72 return invalid<CardRuntimeState>("card tint must contain three numeric channels", "tint");
73 float* channels[] = {&state.tint.r, &state.tint.g, &state.tint.b};
74 for (std::size_t i = 0; i < 3; ++i) {
75 const auto* number = (*array)[i].getIf<double>();
76 const auto* integer = (*array)[i].getIf<std::int64_t>();
77 const double channel = number ? *number : integer ? static_cast<double>(*integer) : -1.0;
78 if (!std::isfinite(channel) || channel < 0.0 || channel > 1.0)
79 return invalid<CardRuntimeState>("card tint channel must be finite in [0,1]", "tint");
80 *channels[i] = static_cast<float>(channel);
81 }
82 }
83 if (const auto* value = field(*object, "tags")) {
84 const auto* array = value->getIf<eve::Value::Array>();
85 if (!array) return invalid<CardRuntimeState>("card tags must be an array", "tags");
86 for (const auto& item : *array) {
87 const auto* tag = item.getIf<std::string>();
88 if (!tag || tag->empty()) return invalid<CardRuntimeState>("card tag must be a non-empty string", "tags");
89 state.tags.push_back(*tag);
90 }
91 std::sort(state.tags.begin(), state.tags.end());
92 state.tags.erase(std::unique(state.tags.begin(), state.tags.end()), state.tags.end());
93 }
94 return eve::Result<CardRuntimeState>::success(std::move(state));
95}

131eve::Result<CardRuntimeState> decodeState(const eve::Value& value) {
132 const auto* object = value.getIf<eve::Value::Object>();
133 if (object == nullptr) return invalid<CardRuntimeState>("card runtime state must be an object", "state");
134 static const std::set<std::string> fields = {"attack", "cost", "currentHealth", "kind",
135 "maxHealth", "name", "tags", "tint"};
136 for (const auto& [name, unused] : *object) {
137 (void)unused;
138 if (!fields.contains(name))
139 return invalid<CardRuntimeState>("unknown card runtime state field", "state." + name);
140 }
141 for (const auto& name : fields)
142 if (!object->contains(name))
143 return invalid<CardRuntimeState>("card runtime state is missing a field", "state." + name);
144 CardRuntimeState result;
145 const auto readIntField = [&](const char* name, int& output) {
146 const auto* integer = object->at(name).getIf<std::int64_t>();
147 if (integer == nullptr || *integer < std::numeric_limits<int>::min() ||
148 *integer > std::numeric_limits<int>::max())
149 return false;
150 output = static_cast<int>(*integer);
151 return true;
152 };
153 const auto* name = object->at("name").getIf<std::string>();
154 const auto* kind = object->at("kind").getIf<std::string>();
155 const auto* tags = object->at("tags").getIf<eve::Value::Array>();
156 const auto* tint = object->at("tint").getIf<eve::Value::Array>();
157 if (name == nullptr || kind == nullptr || tags == nullptr || tint == nullptr || tint->size() != 3 ||
158 !readIntField("attack", result.attack) || !readIntField("cost", result.cost) ||
159 !readIntField("currentHealth", result.currentHealth) || !readIntField("maxHealth", result.maxHealth))
160 return invalid<CardRuntimeState>("card runtime state field is invalid", "state");
161 result.name = *name;
162 result.kind = *kind;
163 if (result.cost < 0 || result.attack < 0 || result.maxHealth < 0 || result.currentHealth < 0 ||
164 result.currentHealth > result.maxHealth)
165 return invalid<CardRuntimeState>("card runtime health/stat values are invalid", "state");
166 float* channels[] = {&result.tint.r, &result.tint.g, &result.tint.b};
167 for (std::size_t index = 0; index < 3; ++index) {
168 const auto* number = (*tint)[index].getIf<double>();
169 if (number == nullptr || !std::isfinite(*number) || *number < 0.0 || *number > 1.0)
170 return invalid<CardRuntimeState>("card tint channel is invalid", "state.tint");
171 *channels[index] = static_cast<float>(*number);
172 }
173 for (const auto& tag : *tags) {
174 const auto* text = tag.getIf<std::string>();
175 if (text == nullptr || text->empty())
176 return invalid<CardRuntimeState>("card runtime tag is invalid", "state.tags");
177 result.tags.push_back(*text);
178 }
179 return eve::Result<CardRuntimeState>::success(std::move(result));
180}

184eve::Result<CardDefinitionRuntime> CardDefinitionRuntime::create(eve::definitions::DefinitionRegistry& registry,

187 eve::definition::ReloadPolicy policy) {

195 if (!runtime) return eve::Result<CardDefinitionRuntime>::failure(runtime.status());
196 return eve::Result<CardDefinitionRuntime>::success(
197 CardDefinitionRuntime(registry, std::move(runtime).takeValue(), policy));
198}

216eve::Result<void> CardDefinitionRuntime::applyTo(CardData* card) const {
217 if (card == nullptr)

219 "card runtime cannot project to a null CardData",
220 "card", {}, "card.definition_runtime"));
221
222 CardData::Identity identity = *card->identity().operator->();
223 CardData::Stats stats = *card->stats().operator->();
224 CardData::Visual visual = *card->visual().operator->();
225 CardData::DefinitionBinding binding = *card->definitionBinding().operator->();
226

228 identity.name = state().name;
229 identity.kind = state().kind;
230 stats.cost = state().cost;
231 stats.attack = state().attack;
232 stats.health = state().currentHealth;
233 visual.tint = state().tint;
234 binding.identity = this->identity();
235 binding.reloadPolicy = policy_;
236 binding.active = isActive();
237
238 using std::swap;
239 swap(*card->identity().operator->(), identity);
240 swap(*card->stats().operator->(), stats);
241 swap(*card->visual().operator->(), visual);
242 swap(*card->definitionBinding().operator->(), binding);
243 return eve::Result<void>::success();

246eve::Result<eve::definition::ReloadOutcome> CardDefinitionRuntime::reload(eve::definition::ReloadPolicy policy) {
247 if (registry_ == nullptr)
248 return invalid<eve::definition::ReloadOutcome>("card definition registry is not bound", "registry");
249 auto next = currentHandle(*registry_, identity().definition);
250 if (!next) return eve::Result<eve::definition::ReloadOutcome>::failure(next.status());
251 auto defaults = resolveState(*registry_, identity().definition, next.value());
252 if (!defaults) return eve::Result<eve::definition::ReloadOutcome>::failure(defaults.status());
253 CardRuntimeState defaultState = std::move(defaults).takeValue();
254 auto rebuild = [defaultState](const CardRuntimeState& oldState, const eve::definition::InstanceIdentity&,
255 const eve::definition::DefinitionHandle&) mutable -> eve::Result<CardRuntimeState> {
256 CardRuntimeState rebuilt = defaultState;
257 rebuilt.currentHealth = std::clamp(oldState.currentHealth, 0, rebuilt.maxHealth);
258 return eve::Result<CardRuntimeState>::success(std::move(rebuilt));
259 };
260 auto result = runtime_.reload(next.value(), policy, defaultState, std::move(rebuild));
261 if (result.ok()) policy_ = policy;
262 return result;
263}
