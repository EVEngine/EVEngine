31 editor::EditorResult<editor::AutomationOwnedTarget> create(
32 const editor::TargetId& target, std::string_view type,
33 const editor::EditorValue::Object& request) override {
34 (void)type;
35 auto biome = std::make_unique<biome_editing::BiomeDocumentTarget>(target.value());
36 const std::string layer = stringField(request, "layer");
37 if (!layer.empty()) {
38 biome_editing::BiomeLayerValue created;
39 created.id = editor::ObjectId(layer);
40 created.name = stringField(request, "name");
41 if (created.name.empty()) created.name = layer;
42 created.spatialAsset = stringField(request, "spatial");
43 auto operation = biome->makeCreateLayer(created);
44 if (!operation.ok())
45 return editor::EditorResult<editor::AutomationOwnedTarget>::failure(operation.status());
46 auto applied = biome->applyDomainOperation(operation.value());
47 if (!applied.ok())
48 return editor::EditorResult<editor::AutomationOwnedTarget>::failure(applied.status());
49 }
50 editor::AutomationOwnedTarget owned;
51 owned.target = std::move(biome);
52 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
53 }
54};
