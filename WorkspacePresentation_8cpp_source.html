10Result<void> WorkspacePresentation::stage(IWorkspacePresentationSource& source, PropertyPath path, EditorValue value) {
11 if (draft_ && draft_->path != path)
12 return failure(EditorStatus::Conflict, "Finish or cancel the active field before editing another field");
13 const auto state = source.state();
14 if (draft_ && (draft_->start.revision != state.revision || draft_->start.context != state.context)) {
15 cancel();
16 return failure(EditorStatus::Conflict, "Document or selection changed during editing");
17 }
18 const auto descriptor = source.schema().find(path);
19 if (!descriptor) return failure(EditorStatus::NotFound, "Property is unavailable: " + path.value());
20 if (hasPropertyFlag(descriptor->flags, PropertyFlag::ReadOnly))
21 return failure(EditorStatus::Rejected, "Property is read-only: " + path.value());
22 auto valid = validatePropertyValue(*descriptor, value);
23 if (!valid.ok()) {
24 cancel();
25 return valid;
26 }
27 if (descriptor->type == PropertyType::Float)
28 if (const auto* integer = value.getIf<std::int64_t>()) value = static_cast<double>(*integer);
29 const auto current = source.read(path);
30 if (current.state != PropertyReadState::Value)
31 return failure(EditorStatus::Rejected, "Property has no single editable value: " + path.value());
32 if (!draft_)
33 draft_ = Draft{std::move(path), std::move(value), current.value, state};
34 else
35 draft_->value = std::move(value);
36 return editing::applied<void>();

39Result<void> WorkspacePresentation::commit(IWorkspacePresentationSource& source) {
40 if (!draft_) return editing::noOp();
41 Draft draft = std::move(*draft_);
42 cancel();
43 const auto state = source.state();
44 if (draft.start.revision != state.revision || draft.start.context != state.context)
45 return failure(EditorStatus::Conflict, "Document or selection changed during editing");
46 if (draft.value == draft.original) return editing::noOp();
47 const auto descriptor = source.schema().find(draft.path);
48 if (!descriptor || hasPropertyFlag(descriptor->flags, PropertyFlag::ReadOnly))
49 return failure(EditorStatus::Rejected, "Property became unavailable during editing");
50 auto valid = validatePropertyValue(*descriptor, draft.value);
51 if (!valid.ok()) return valid;
52 return source.write(draft.path, draft.value);
53}

71Value WorkspacePresentation::snapshot(const IWorkspacePresentationSource& source) const {
72 const auto state = source.state();
73 Value::Array rows;
74 for (const auto& descriptor : source.schema().properties) {
75 const auto current = source.read(descriptor.path);
76 std::string kind;
77 switch (descriptor.type) {
78 case PropertyType::Bool: kind = "bool"; break;
79 case PropertyType::Float:
80 case PropertyType::Int: kind = "number"; break;
81 case PropertyType::Enum: kind = "enum"; break;
82 default: kind = "text"; break;
83 }
84 Value::Array choices;
85 for (const auto& choice : descriptor.enumItems) choices.emplace_back(choice);
86 const bool editable = !hasPropertyFlag(descriptor.flags, PropertyFlag::ReadOnly) &&
87 current.state == PropertyReadState::Value &&
88 (kind != "text" || descriptor.type == PropertyType::String ||
89 descriptor.type == PropertyType::AssetRef || descriptor.type == PropertyType::ObjectRef);
90 EditorValue value = current.value;
91 if (draft_ && draft_->path == descriptor.path) value = draft_->value;
92 auto projected = toPresentationValue(value);
93 std::string display;
94 if (current.state == PropertyReadState::Mixed)
95 display = "Mixed values";
96 else if (current.state != PropertyReadState::Value)
97 display = "Unavailable";
98 else if (const auto* text = projected.getIf<std::string>())
99 display = *text;
100 else {
101 auto encoded = projected.toJson();
102 display = encoded.ok() ? encoded.value() : "Unavailable";
103 }
104 rows.emplace_back(Value::Object{{"path", descriptor.path.value()},
105 {"label", descriptor.displayNameKey},
106 {"kind", kind},
107 {"value", std::move(projected)},
108 {"display", std::move(display)},
109 {"editable", editable},
110 {"minimum", descriptor.numeric.minimum.value_or(-1000000.0)},
111 {"maximum", descriptor.numeric.maximum.value_or(1000000.0)},
112 {"choices", std::move(choices)}});
113 }
114 return Value::Object{{"revision", static_cast<std::int64_t>(state.revision)},
115 {"context", state.context},
116 {"canUndo", state.canUndo},
117 {"canRedo", state.canRedo},
118 {"playing", state.playing},
119 {"rows", std::move(rows)}};
120}
