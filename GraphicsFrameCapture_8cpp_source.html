170void Graphics::pollFrameCapture() {
171 frameCaptures().collect();
172 if (!captureToken_.empty() && !captureSubmitted_ && device.instance) {
173 // A new main-thread poll before submission means present threw. Drain
174 // this exceptional path rather than publish unsubmitted staging bytes.
175 try {
176 device->waitIdle();
177 } catch (const vk::SystemError&) {
178 }
179 auto failed = frameCaptures().fail(captureToken_, "CaptureSubmitFailed");
180 failed.ignore();
181 destroyFrameCapture();
182 return;
183 }
184 if (!captureSubmitted_ || !captureFence_ || !device.instance) return;
185 vk::Result status;
186 try {
187 status = device->getFenceStatus(captureFence_);
188 } catch (const vk::SystemError&) {
189 status = vk::Result::eErrorDeviceLost;
190 }
191 if (status == vk::Result::eNotReady) return;
192 if (status != vk::Result::eSuccess) {
193 auto failed = frameCaptures().fail(captureToken_, "DeviceLost");
194 failed.ignore();
195 } else {
196 auto state = frameCaptures().snapshot(captureToken_);
197 if (state && state.value().phase == FrameCapturePhase::ReadbackPending) {
198 try {
199 std::vector<std::shared_ptr<const std::vector<uint8_t>>> storage(captureCopies_.size());
200 for (size_t i = 0; i < captureCopies_.size(); ++i) {
201 auto& copy = captureCopies_[i];
202 if (copy.staging.buffer && copy.storageIndex == i) {
203 auto bytes = std::make_shared<std::vector<uint8_t>>(size_t(copy.metadata.width) *
204 copy.metadata.height * 4);
205 const void* mapped = copy.staging.map();
206 if (!mapped) throw std::runtime_error("ReadbackMapFailed");
207 std::memcpy(bytes->data(), mapped, bytes->size());
208 copy.staging.unmap();
209 storage[i] = std::move(bytes);
210 }
211 copy.metadata.bytes = storage[copy.storageIndex];
212 captureSnapshot_.buffers.push_back(copy.metadata);
213 }
214 auto done = frameCaptures().complete(captureToken_, std::move(captureSnapshot_));
215 if (!done) {
216 auto failed = frameCaptures().fail(captureToken_, "CapturePublicationFailed");
217 failed.ignore();
218 }
219 } catch (const std::exception& e) {
220 auto failed = frameCaptures().fail(captureToken_, std::string("ReadbackFailed: ") + e.what());
221 failed.ignore();
222 }
223 }
224 }
225 frameCaptures().retire(captureToken_);
226 captureCopies_.clear();
227 captureSnapshot_ = {};
228 captureToken_.clear();
229 captureFence_ = nullptr;
230 captureSubmitted_ = false;
231}
