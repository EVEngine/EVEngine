25Result<PbrVariantPlan> planPbrVariant(const PbrSurface& surface, bool environment, bool shadows,
26 const PbrVariantLimits& limits) {
27 auto valid = validatePbrSurface(surface);
28 if (!valid) return Result<PbrVariantPlan>::failure(valid.status());
29
30 PbrVariantPlan plan;
31 for (std::uint32_t slot = 0; slot < std::uint32_t(PbrTextureSlot::Count); ++slot)
32 if (surface.textures[slot].texture) plan.canonicalTextureMask |= 1u << slot;
33
34 if (surface.vegetationDetail.value > 0.f) {
35 // The TVE detail stage always evaluates albedo and normal through neutral
36 // fallbacks. Its packed mask is reachable only when authored.
37 plan.detailTextureMask = 0b011u;
38 if (surface.vegetationDetail.textures[2].texture) plan.detailTextureMask |= 0b100u;
39 }
40 if (environment) plan.fragmentFlags |= PbrEnvironment;
41 if (shadows) plan.fragmentFlags |= PbrShadows;
42 if (surface.vegetationExtras.texture) plan.fragmentFlags |= PbrExtras;
43 if (surface.vegetationColors.texture) plan.fragmentFlags |= PbrColors;
44 if (surface.vegetationAlpha.enabled && surface.vegetationAlpha.noise) plan.fragmentFlags |= PbrFadeNoise;
45
46 if (surface.vegetationVertex.source == PbrVegetationDeformationSource::GpuFields &&
47 surface.vegetationVertex.texture)
48 plan.vertexFlags |= PbrVertexField;
49 if (surface.vegetationMotion.mode == PbrVegetationMotionMode::Object) {
50 if (surface.vegetationMotion.texture) plan.vertexFlags |= PbrMotionField;
51 if (surface.vegetationMotion.noise) plan.vertexFlags |= PbrMotionNoise;
52 }
53
54 plan.fragmentTextures = std::popcount(plan.canonicalTextureMask) + std::popcount(plan.detailTextureMask) +
55 std::popcount(plan.fragmentFlags);
56 std::vector<std::pair<SamplerKey, std::uint32_t>> fragmentSamplers;
57 for (std::uint32_t slot = 0; slot < std::uint32_t(PbrTextureSlot::Count); ++slot)
58 if ((plan.canonicalTextureMask & (1u << slot)) != 0u) {
59 plan.canonicalSamplerRepresentatives[slot] =
60 addSampler(fragmentSamplers, samplerKey(surface.textures[slot]), slot);
61 }
62 for (std::uint32_t slot = 0; slot < surface.vegetationDetail.textures.size(); ++slot)
63 if ((plan.detailTextureMask & (1u << slot)) != 0u) {
64 const auto resource = std::uint32_t(PbrTextureSlot::Count) + slot;
65 plan.detailSamplerRepresentatives[slot] =
66 addSampler(fragmentSamplers, samplerKey(surface.vegetationDetail.textures[slot]), resource);
67 }
68 // These fixed-function resources currently have distinct backend sampler
69 // contracts; a later layout may merge them only after proving equivalence.
70 plan.fragmentSamplers = std::uint32_t(fragmentSamplers.size()) + std::popcount(plan.fragmentFlags);
71 plan.vertexTextures = std::popcount(plan.vertexFlags);
72 plan.vertexSamplers = plan.vertexTextures;
73
74 if (plan.fragmentTextures > limits.sampledTexturesPerStage ||
75 plan.fragmentSamplers > limits.samplersPerStage || plan.vertexTextures > limits.sampledTexturesPerStage ||
76 plan.vertexSamplers > limits.samplersPerStage)
77 return Result<PbrVariantPlan>::failure(Diagnostic::error(
78 DiagnosticCode::Unsupported, "WebGPU PBR material exceeds per-stage texture or sampler limits"));
79 return Result<PbrVariantPlan>::success(plan);
80}
