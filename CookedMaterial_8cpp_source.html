51 const asset::EvpackCapabilities& caps) {
52 auto payload = reader.read(ref, "eve.material/2", caps, 1024 * 1024);
53 if (!payload && payload.error()->code() == DiagnosticCode::TypeMismatch)
54 payload = reader.read(ref, "eve.material/1", caps, 1024 * 1024);
55 if (!payload) return Result<CookedMaterial>::failure(payload.status());
56 try {
57 if (payload.value().chunks.size() != 1) throw Invalid{};
58 auto decoded = asset::decodeRuntimeDefinition(payload.value().chunks.front().bytes);
59 if (!decoded) return Result<CookedMaterial>::failure(decoded.status());
60 auto* object = decoded.value().getIf<Value::Object>();
61 if (!object) throw Invalid{};
62 auto& o = *object;
63 CookedMaterial m;
64 auto& s = m.surface;
65 auto version = integer(o, "schemaVersion", 0);
66 if (string(o, "schema", "") != "eve.material" || (version != 1 && version != 2)) throw Invalid{};
67 const auto shading = string(o, "shadingModel", "");
68 if (shading != "pbr" && shading != "unlit") throw Invalid{};
69 s.unlit = shading == "unlit";
70 auto mode = string(o, "surfaceMode", "opaque");
71 if (mode != "opaque" && mode != "masked" && mode != "transparent") throw Invalid{};
72 m.transparent = mode == "transparent";
73 m.masked = mode == "masked";
74 auto blend = string(o, "blendMode", "alpha");
75 if (blend != "alpha" && blend != "premultiplied") throw Invalid{};
76 m.blend = blend == "alpha" ? graphics::BlendMode::Alpha : graphics::BlendMode::Premultiplied;
77 m.doubleSided = boolean(o, "doubleSided", false);
78 m.alphaCutoff = number(o, "alphaCutoff", .5f);
79 std::array<float, 4> color{1, 1, 1, 1};
80 vector(o, "baseColor", color);
81 for (float c : color)
82 if (c < 0 || c > 1) throw Invalid{};
83 m.color = {color[0], color[1], color[2], color[3]};
84 m.metallic = number(o, "metallic", 0);
85 m.roughness = number(o, "roughness", 1);
86 if (m.metallic < 0 || m.metallic > 1 || m.roughness < 0 || m.roughness > 1 || m.alphaCutoff < 0 ||
87 m.alphaCutoff > 1)
88 throw Invalid{};
89 vector(o, "emissive", s.emissive);
90 vector(o, "specularColorFactor", s.specularColor);
91 s.normalScale = number(o, "normalScale", 1);
92 s.occlusionStrength = number(o, "occlusionStrength", 1);
93 s.emissiveStrength = number(o, "emissiveStrength", 1);
94 s.specularFactor = number(o, "specularFactor", 1);
95 s.ior = number(o, "ior", 1.5f);
96 s.anisotropyStrength = number(o, "anisotropyStrength", 0);
97 s.anisotropyRotation = number(o, "anisotropyRotation", 0);
98 s.clearcoatFactor = number(o, "clearcoatFactor", 0);
99 s.clearcoatRoughness = number(o, "clearcoatRoughnessFactor", 0);
100 s.clearcoatNormalScale = number(o, "clearcoatNormalScale", 1);
101 constexpr const char* roles[] = {"baseColorTexture", "metallicRoughnessTexture", "normalTexture",
102 "occlusionTexture", "emissiveTexture", "specularTexture",
103 "specularColorTexture", "anisotropyTexture", "clearcoatTexture",
104 "clearcoatRoughnessTexture", "clearcoatNormalTexture"};
105 for (size_t i = 0; i < 11; i++) {
106 std::string role = roles[i];
107 auto* image = field(o, role);
108 if (!image) continue;
109 if (!image->isString()) throw Invalid{};
110 auto ref = AssetRef::parse(image->asString());
111 if (!ref) return Result<CookedMaterial>::failure(ref.status());
112 m.images[i] = std::move(ref).takeValue();
113 auto& b = s.textures[i];
114 b.srgbDecode = false;
115 b.texcoord = integer(o, role + "TexCoord", 0);
116 if (auto* transform = field(o, role + "Transform")) {
117 auto* t = transform->getIf<Value::Object>();
118 if (!t) throw Invalid{};
119 vector(*t, "offset", b.offset);
120 vector(*t, "scale", b.scale);
121 b.rotation = number(*t, "rotation", 0);
122 b.texcoord = integer(*t, "texCoord", b.texcoord);
123 }
124 if (auto* sampler = field(o, role + "Sampler")) {
125 auto* t = sampler->getIf<Value::Object>();
126 if (!t) throw Invalid{};
127 b.wrapS = integer(*t, "wrapS", 10497);
128 b.wrapT = integer(*t, "wrapT", 10497);
129 b.minFilter = integer(*t, "minFilter", 9987);
130 b.magFilter = integer(*t, "magFilter", 9729);
131 }
132 }
133 auto validated = graphics::validatePbrSurface(s);
134 if (!validated) return Result<CookedMaterial>::failure(validated.status());
135 return Result<CookedMaterial>::success(std::move(m));
136 } catch (const Invalid&) {

138 DiagnosticCode::InvalidArgument, "invalid cooked PBR material", {}, {}, "asset.graphics.material"));
139 }
