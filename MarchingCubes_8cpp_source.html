54inline float grad3(uint32_t h, float x, float y, float z) {
55 const uint32_t g = h & 15u;
56 const float u = g < 8 ? x : y;
57 const float v = g < 4 ? y : (g == 12 || g == 14 ? x : z);
58 return ((g & 1u) ? -u : u) + ((g & 2u) ? -v : v);

61float valueNoise3(float x, float y, float z, uint32_t seed) {
62 const int xi = int(std::floor(x));
63 const int yi = int(std::floor(y));
64 const int zi = int(std::floor(z));
65 const float xf = x - float(xi);
66 const float yf = y - float(yi);
67 const float zf = z - float(zi);
68 const float u = fade(xf);
69 const float v = fade(yf);
70 const float w = fade(zf);
71
72 auto h = [&](int ix, int iy, int iz) -> uint32_t {
73 return uint32_t(ix) * 374761393u + uint32_t(iy) * 668265263u + uint32_t(iz) * 1274126177u +
74 seed * 2246822519u;
75 };
76
77 const float n000 = grad3(h(xi, yi, zi), xf, yf, zf);
78 const float n100 = grad3(h(xi + 1, yi, zi), xf - 1, yf, zf);
79 const float n010 = grad3(h(xi, yi + 1, zi), xf, yf - 1, zf);
80 const float n110 = grad3(h(xi + 1, yi + 1, zi), xf - 1, yf - 1, zf);
81 const float n001 = grad3(h(xi, yi, zi + 1), xf, yf, zf - 1);
82 const float n101 = grad3(h(xi + 1, yi, zi + 1), xf - 1, yf, zf - 1);
83 const float n011 = grad3(h(xi, yi + 1, zi + 1), xf, yf - 1, zf - 1);
84 const float n111 = grad3(h(xi + 1, yi + 1, zi + 1), xf - 1, yf - 1, zf - 1);
85
86 const float x00 = lerp(n000, n100, u);
87 const float x10 = lerp(n010, n110, u);
88 const float x01 = lerp(n001, n101, u);
89 const float x11 = lerp(n011, n111, u);
90 const float y0 = lerp(x00, x10, v);
91 const float y1 = lerp(x01, x11, v);
92 return lerp(y0, y1, w); // roughly [-1,1]
93}

95float fbm3(float x, float y, float z, uint32_t seed, int octaves) {
96 float sum = 0.f;
97 float amp = 0.5f;
98 float freq = 1.f;
99 float norm = 0.f;
100 for (int i = 0; i < octaves; ++i) {
101 sum += valueNoise3(x * freq, y * freq, z * freq, seed + uint32_t(i) * 1013u) * amp;
102 norm += amp;
103 amp *= 0.5f;
104 freq *= 2.f;
105 }
106 return norm > 0.f ? sum / norm : 0.f;
107}

129uint64_t edgeKey(uint32_t a, uint32_t b) {
130 if (a > b) std::swap(a, b);

134uint32_t midpoint(uint32_t a, uint32_t b, std::vector<Vec3> &vertices,
135 std::map<uint64_t, uint32_t> &cache) {
136 const uint64_t key = edgeKey(a, b);
137 const auto it = cache.find(key);
138 if (it != cache.end()) return it->second;
139 const uint32_t index = uint32_t(vertices.size());
140 vertices.push_back(normalized(add(vertices[a], vertices[b])));
141 cache.emplace(key, index);
142 return index;
143}

146 p = normalized(p);
147 constexpr float kPi = 3.14159265358979323846f;
148 const float u = std::atan2(p.z, p.x) / (2.f * kPi) + 0.5f;
149 const float v = std::asin(std::clamp(p.y, -1.f, 1.f)) / kPi + 0.5f;
150 out.addVertex(p.x * radius, p.y * radius, p.z * radius, p.x, p.y, p.z, u, v);
151}

155bool generateHexPlanetMesh(const Params &params, MeshBuild &out, std::string &error) {
156 const int subdivisions = params.getInt("subdivisions", 2);
157 const float radius = params.getFloat("radius", 1.f);
158 const float inset = params.getFloat("tileInset", 0.06f);
159 if (subdivisions < 0 || subdivisions > 7) {
160 error = "mesh.hexplanet: subdivisions must be in [0, 7]";
161 return false;
162 }
163 if (!(radius > 0.f)) {
164 error = "mesh.hexplanet: radius must be positive";
165 return false;
166 }
167 if (inset < 0.f || inset >= 0.5f) {
168 error = "mesh.hexplanet: tileInset must be in [0, 0.5)";
169 return false;
170 }
171
172 const float phi = (1.f + std::sqrt(5.f)) * 0.5f;
173 std::vector<Vec3> vertices = {
174 {-1, phi, 0}, {1, phi, 0}, {-1, -phi, 0}, {1, -phi, 0},
175 {0, -1, phi}, {0, 1, phi}, {0, -1, -phi}, {0, 1, -phi},
176 {phi, 0, -1}, {phi, 0, 1}, {-phi, 0, -1}, {-phi, 0, 1},
177 };
178 for (Vec3 &v : vertices) v = normalized(v);
179 std::vector<Triangle> faces = {
180 {0, 11, 5}, {0, 5, 1}, {0, 1, 7}, {0, 7, 10}, {0, 10, 11},
181 {1, 5, 9}, {5, 11, 4}, {11, 10, 2}, {10, 7, 6}, {7, 1, 8},
182 {3, 9, 4}, {3, 4, 2}, {3, 2, 6}, {3, 6, 8}, {3, 8, 9},
183 {4, 9, 5}, {2, 4, 11}, {6, 2, 10}, {8, 6, 7}, {9, 8, 1},
184 };
185 for (int level = 0; level < subdivisions; ++level) {
186 std::map<uint64_t, uint32_t> cache;
187 std::vector<Triangle> next;
188 next.reserve(faces.size() * 4u);
189 for (const Triangle &f : faces) {
190 const uint32_t ab = midpoint(f.a, f.b, vertices, cache);
191 const uint32_t bc = midpoint(f.b, f.c, vertices, cache);
192 const uint32_t ca = midpoint(f.c, f.a, vertices, cache);
193 next.insert(next.end(), {{f.a, ab, ca}, {f.b, bc, ab}, {f.c, ca, bc}, {ab, bc, ca}});
194 }
195 faces.swap(next);
196 }
197
198 std::vector<std::vector<uint32_t>> incident(vertices.size());
199 std::vector<Vec3> faceCenters;
200 faceCenters.reserve(faces.size());
201 for (uint32_t i = 0; i < faces.size(); ++i) {
202 const Triangle &f = faces[i];
203 faceCenters.push_back(normalized(add(add(vertices[f.a], vertices[f.b]), vertices[f.c])));
204 incident[f.a].push_back(i);
205 incident[f.b].push_back(i);
206 incident[f.c].push_back(i);
207 }
208
209 out.clear();
210 int pentagons = 0, hexagons = 0;
211 for (uint32_t cell = 0; cell < vertices.size(); ++cell) {
212 const Vec3 center = vertices[cell];
213 const Vec3 reference = std::fabs(center.y) < 0.9f ? normalized(cross({0, 1, 0}, center))
214 : normalized(cross({1, 0, 0}, center));
215 const Vec3 tangent = cross(center, reference);
216 auto &ring = incident[cell];
217 std::sort(ring.begin(), ring.end(), [&](uint32_t lhs, uint32_t rhs) {
218 const Vec3 a = sub(faceCenters[lhs], mul(center, dot(faceCenters[lhs], center)));
219 const Vec3 b = sub(faceCenters[rhs], mul(center, dot(faceCenters[rhs], center)));
220 return std::atan2(dot(a, tangent), dot(a, reference)) <
221 std::atan2(dot(b, tangent), dot(b, reference));
222 });
223 if (ring.size() == 5) ++pentagons;
224 else if (ring.size() == 6) ++hexagons;
225
226 const uint32_t base = uint32_t(out.getVertexCount());
227 addPlanetVertex(out, center, radius);
228 for (uint32_t faceIndex : ring) {
229 // Pull dual corners toward this cell's center. Re-normalizing keeps every tile spherical.
230 addPlanetVertex(out, normalized(add(mul(faceCenters[faceIndex], 1.f - inset),
231 mul(center, inset))), radius);
232 }
233 for (uint32_t i = 0; i < ring.size(); ++i) {
234 out.addTriangle(base, base + 1u + i, base + 1u + (i + 1u) % uint32_t(ring.size()));
235 }
236 }
237 out.setMeta("algorithm", "mesh.hexplanet");
238 out.setMeta("cells", std::to_string(vertices.size()));
239 out.setMeta("pentagons", std::to_string(pentagons));
240 out.setMeta("hexagons", std::to_string(hexagons));
241 out.setMeta("subdivisions", std::to_string(subdivisions));
242 return true;
243}

245bool marchingCubes(const float *density, int nx, int ny, int nz, float isolevel, MeshBuild &out,
246 std::string *error) {
247 if (!density) {
248 if (error) *error = "marchingCubes: null density";
249 return false;
250 }
251 if (nx < 2 || ny < 2 || nz < 2) {
252 if (error) *error = "marchingCubes: volume must be at least 2x2x2";
253 return false;
254 }
255
256 out.clear();
257 out.reserve((nx * ny * nz) / 2, (nx * ny * nz) * 3);
258
259 auto at = [&](int x, int y, int z) -> float {
260 return density[size_t(x) + size_t(y) * size_t(nx) + size_t(z) * size_t(nx) * size_t(ny)];
261 };
262
263 // World-space mapping: unit cube centered at origin spanning [-0.5, 0.5]^3.
264 const float sx = 1.f / float(nx - 1);
265 const float sy = 1.f / float(ny - 1);
266 const float sz = 1.f / float(nz - 1);
267
268 for (int z = 0; z < nz - 1; ++z) {
269 for (int y = 0; y < ny - 1; ++y) {
270 for (int x = 0; x < nx - 1; ++x) {
271 float val[8];
272 int cubeIndex = 0;
273 for (int i = 0; i < 8; ++i) {
274 const int cx = x + int(kCornerOffset[i][0]);
275 const int cy = y + int(kCornerOffset[i][1]);
276 const int cz = z + int(kCornerOffset[i][2]);
277 val[i] = at(cx, cy, cz);
278 if (val[i] < isolevel) cubeIndex |= (1 << i);
279 }
280 const int edges = kEdgeTable[cubeIndex];
281 if (edges == 0) continue;
282
283 float vertList[12][3];
284 for (int e = 0; e < 12; ++e) {
285 if (!(edges & (1 << e))) continue;
286 const int a = kEdgeCorners[e][0];
287 const int b = kEdgeCorners[e][1];
288 const float va = val[a];
289 const float vb = val[b];
290 float t = (isolevel - va) / (vb - va + 1e-12f);
291 t = std::clamp(t, 0.f, 1.f);
292 const float px =
293 (float(x) + lerp(kCornerOffset[a][0], kCornerOffset[b][0], t)) * sx - 0.5f;
294 const float py =
295 (float(y) + lerp(kCornerOffset[a][1], kCornerOffset[b][1], t)) * sy - 0.5f;
296 const float pz =
297 (float(z) + lerp(kCornerOffset[a][2], kCornerOffset[b][2], t)) * sz - 0.5f;
298 vertList[e][0] = px;
299 vertList[e][1] = py;
300 vertList[e][2] = pz;
301 }
302
303 for (int i = 0; kTriTable[cubeIndex][i] != -1; i += 3) {
304 const int e0 = kTriTable[cubeIndex][i];
305 const int e1 = kTriTable[cubeIndex][i + 1];
306 const int e2 = kTriTable[cubeIndex][i + 2];
307 const float *p0 = vertList[e0];
308 const float *p1 = vertList[e1];
309 const float *p2 = vertList[e2];
310
311 float ax = p1[0] - p0[0], ay = p1[1] - p0[1], az = p1[2] - p0[2];
312 float bx = p2[0] - p0[0], by = p2[1] - p0[1], bz = p2[2] - p0[2];
313 float nxn = ay * bz - az * by;
314 float nyn = az * bx - ax * bz;
315 float nzn = ax * by - ay * bx;
316 normalize3(nxn, nyn, nzn);
317
318 // Flip so normals point toward empty (lower density / outside).
319 // With cubeIndex bits for val < isolevel, winding already tends outward.
320 const uint32_t base = uint32_t(out.getVertexCount());
321 const float u0 = p0[0] + 0.5f, v0 = p0[1] + 0.5f;
322 const float u1 = p1[0] + 0.5f, v1 = p1[1] + 0.5f;
323 const float u2 = p2[0] + 0.5f, v2 = p2[1] + 0.5f;
324 out.addVertex(p0[0], p0[1], p0[2], nxn, nyn, nzn, u0, v0);
325 out.addVertex(p1[0], p1[1], p1[2], nxn, nyn, nzn, u1, v1);
326 out.addVertex(p2[0], p2[1], p2[2], nxn, nyn, nzn, u2, v2);
327 out.addTriangle(base, base + 1, base + 2);
328 }
329 }
330 }
331 }
332
333 out.setMeta("algorithm", "mesh.marchingcubes");
334 return true;
335}

337bool fillDensityField(const Params &params, std::vector<float> &density, int &nx, int &ny, int &nz,
338 std::string &error) {
339 const int res = params.getInt("resolution",
340 params.getWidth() > 0 ? params.getWidth() : 24);
341 nx = params.getInt("nx", res);
342 ny = params.getInt("ny", params.getHeight() > 0 ? params.getHeight() : res);
343 nz = params.getInt("nz", params.getInt("depth", res));
344 if (nx < 2 || ny < 2 || nz < 2) {
345 error = "mesh.marchingcubes: resolution must be at least 2 in each axis";
346 return false;
347 }
348 if (nx > 128 || ny > 128 || nz > 128) {
349 error = "mesh.marchingcubes: resolution capped at 128 per axis";
350 return false;
351 }
352
353 const std::string field = params.getString("field", "sphere");
354 const float scale = params.getFloat("scale", 1.f);
355 const int octaves = std::max(1, params.getInt("octaves", 3));
356 const uint32_t seed = params.getSeed();
357
358 density.assign(size_t(nx) * size_t(ny) * size_t(nz), 0.f);
359 for (int z = 0; z < nz; ++z) {
360 for (int y = 0; y < ny; ++y) {
361 for (int x = 0; x < nx; ++x) {
362 const float px = (float(x) / float(nx - 1) - 0.5f) * 2.f;
363 const float py = (float(y) / float(ny - 1) - 0.5f) * 2.f;
364 const float pz = (float(z) / float(nz - 1) - 0.5f) * 2.f;
365 float d = 0.f;
366 if (field == "sphere") {
367 const float r = params.getFloat("radius", 0.7f);
368 d = r - std::sqrt(px * px + py * py + pz * pz);
369 } else if (field == "rock") {
370 // An ellipsoid SDF whose radius is displaced by low-frequency strata and
371 // higher-frequency erosion. Quantising the direction before sampling the
372 // strata creates broad, natural fracture planes instead of a noisy sphere.
373 const float radius = params.getFloat("radius", 0.68f);
374 const float flattening =
375 std::clamp(params.getFloat("flattening", 0.22f), 0.f, 0.7f);
376 const float angularity =
377 std::clamp(params.getFloat("angularity", 0.35f), 0.f, 1.f);
378 const float erosion =
379 std::clamp(params.getFloat("erosion", 0.18f), 0.f, 0.45f);
380 const float detailScale = std::max(0.25f, scale);
381 const float sy = std::max(0.3f, 1.f - flattening);
382 const float ex = px;
383 const float ey = py / sy;
384 const float ez = pz;
385 const float len = std::sqrt(ex * ex + ey * ey + ez * ez);
386 const float invLen = len > 1e-5f ? 1.f / len : 0.f;
387 const float steps = 3.f + angularity * 9.f;
388 const float qx = std::round(ex * invLen * steps) / steps;
389 const float qy = std::round(ey * invLen * steps) / steps;
390 const float qz = std::round(ez * invLen * steps) / steps;
391 const float strata = fbm3((qx + 2.3f) * detailScale,
392 (qy + 4.7f) * detailScale,
393 (qz + 8.1f) * detailScale, seed, octaves);
394 const float pits = fbm3((px + 7.2f) * detailScale * 2.7f,
395 (py + 1.9f) * detailScale * 2.7f,
396 (pz + 5.4f) * detailScale * 2.7f,
397 seed + 7919u, std::max(2, octaves - 1));
398 const float displacement = strata * (0.08f + angularity * 0.16f) -
399 std::max(0.f, pits) * erosion;
400 d = radius + displacement - len;
401 } else if (field == "torus") {
402 const float R = params.getFloat("majorRadius", 0.55f);
403 const float r = params.getFloat("minorRadius", 0.22f);
404 const float q = std::sqrt(px * px + pz * pz) - R;
405 d = r - std::sqrt(q * q + py * py);
406 } else if (field == "terrain") {
407 const float h =
408 fbm3(px * scale + 3.1f, 0.f, pz * scale + 1.7f, seed, octaves) * 0.45f;
409 d = h - py;
410 } else if (field == "noise") {
411 const float n =
412 fbm3(px * scale + 2.f, py * scale + 5.f, pz * scale + 9.f, seed, octaves);
413 d = n - params.getFloat("threshold", 0.05f);
414 } else {
415 error = "mesh.marchingcubes: unknown field '" + field +
416 "' (use sphere|rock|torus|noise|terrain)";
417 return false;
418 }
419 // Soft boundary falloff so surfaces close.
420 const float margin = 0.92f;
421 const float bx = std::max(0.f, std::fabs(px) - margin);
422 const float by = std::max(0.f, std::fabs(py) - margin);
423 const float bz = std::max(0.f, std::fabs(pz) - margin);
424 d -= (bx * bx + by * by + bz * bz) * 4.f;
425 density[size_t(x) + size_t(y) * size_t(nx) + size_t(z) * size_t(nx) * size_t(ny)] =
426 d;
427 }
428 }
429 }
430 return true;
431}

433bool generateMarchingCubesMesh(const Params &params, MeshBuild &out, std::string &error) {
434 std::vector<float> density;
435 int nx = 0, ny = 0, nz = 0;
436 if (!fillDensityField(params, density, nx, ny, nz, error)) return false;
437 const float isolevel = params.getFloat("isolevel", 0.f);

441 error = "mesh.marchingcubes: empty mesh (adjust field/isolevel/resolution)";
442 return false;
443 }
444 return true;

468bool MeshRecipeRegistry::generate(const std::string &id, const Params &params, MeshBuild &out,
469 std::string &error) const {
470 auto it = recipes_.find(id);
471 if (it == recipes_.end()) {
472 error = "unknown mesh recipe '" + id + "'";
473 return false;
474 }
475 return it->second.fn(params, out, error);
