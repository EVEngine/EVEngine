14eve::Result<QueryVolume> boxVolume(float x, float y, float z, float w, float h, float d, float qx, float qy, float qz,
15 float qw) {
16 JPH::Quat rotation(qx, qy, qz, qw);
17 if (!isFinite(x, y, z) || !isFinite(w, h, d) || w <= 0 || h <= 0 || d <= 0 || !std::isfinite(rotation.LengthSq()) ||
18 rotation.LengthSq() < 1.e-12f)
19 return rejected<QueryVolume>("Invalid box query", eve::DiagnosticCode::InvalidArgument);
20 QueryVolume volume;
21 volume.shape = new JPH::BoxShape({w * .5f, h * .5f, d * .5f}, std::min({.05f, w * .5f, h * .5f, d * .5f}));
22 volume.transform = JPH::RMat44::sRotationTranslation(rotation.Normalized(), {x, y, z});
23 return eve::Result<QueryVolume>::success(std::move(volume));
24}

26eve::Result<int> WorldBackend::overlapQuery(const QueryVolume& volume, QueryFilter3D filter) {
27 if (!isValid()) return rejected<int>("World expired", eve::DiagnosticCode::StaleHandle);
28 auto hits = overlapVolume(*state, volume);
29 std::vector<std::pair<int, int>> shapes;
30 std::vector<int> bodies;
31 for (const auto& hit : hits.mHits) {
32 if (!isAcceptedHit(*state, hit.mBodyID2, hit.mSubShapeID2, filter)) continue;
33 auto value = queryHit(*state, hit.mBodyID2, hit.mSubShapeID2, JPH::RVec3::sZero(), JPH::Vec3::sZero(), 0);
34 shapes.emplace_back(value.bodyId, value.shapeId);
35 bodies.push_back(value.bodyId);
36 }
37 std::sort(shapes.begin(), shapes.end());
38 shapes.erase(std::unique(shapes.begin(), shapes.end()), shapes.end());
39 std::sort(bodies.begin(), bodies.end());
40 bodies.erase(std::unique(bodies.begin(), bodies.end()), bodies.end());
41 owner->queryShapes_ = std::move(shapes);
42 owner->queryBodyIds_ = std::move(bodies);
43 return eve::Result<int>::success(static_cast<int>(owner->queryBodyIds_.size()));
44}

70eve::Result<int> WorldBackend::castQuery(const QueryVolume& volume, float dx, float dy, float dz, int maxHits) {
71 if (!isValid()) return rejected<int>("World expired", eve::DiagnosticCode::StaleHandle);
72 if (!isFinite(dx, dy, dz) || maxHits < 1 || maxHits > 4096)
73 return rejected<int>("Invalid cast", eve::DiagnosticCode::InvalidArgument);
74 const QueryFilter3D filter{owner->queryCategoryBits_, owner->queryMaskBits_, owner->queryIgnoredBodyId_,
75 owner->queryIgnoredShapeId_};
76 auto hits = sweepVolume(*state, volume, {dx, dy, dz});
77 std::vector<World3D::ShapeCastResult> values;
78 for (const auto& hit : hits.mHits) {
79 if (!isAcceptedHit(*state, hit.mBodyID2, hit.mSubShapeID2, filter)) continue;
80 auto r =
81 queryHit(*state, hit.mBodyID2, hit.mSubShapeID2, volume.transform.GetTranslation() + hit.mContactPointOn2,
82 -hit.mPenetrationAxis.NormalizedOr(JPH::Vec3::sAxisY()), hit.mFraction);
83 values.push_back({r.bodyId, r.shapeId, r.shapeTag, r.materialId, r.triangleIndex, r.x, r.y, r.z, r.normalX,
84 r.normalY, r.normalZ, r.fraction});
85 }
86 std::sort(values.begin(), values.end(), [](const auto& a, const auto& b) {
87 return std::tie(a.fraction, a.shapeId, a.triangleIndex) < std::tie(b.fraction, b.shapeId, b.triangleIndex);
88 });
89 if (values.size() > size_t(maxHits)) values.resize(size_t(maxHits));
90 const auto first = values.empty() ? World3D::ShapeCastResult{} : values.front();
91 owner->shapeCastResults_ = std::move(values);
92 owner->shapeCastBodyId_ = first.bodyId;
93 owner->shapeCastShapeId_ = first.shapeId;
94 owner->shapeCastShapeTag_ = first.shapeTag;
95 owner->shapeCastMaterialId_ = first.materialId;
96 owner->shapeCastTriangleIndex_ = first.triangleIndex;
97 owner->shapeCastX_ = first.x;
98 owner->shapeCastY_ = first.y;
99 owner->shapeCastZ_ = first.z;
100 owner->shapeCastNormalX_ = first.normalX;
101 owner->shapeCastNormalY_ = first.normalY;
102 owner->shapeCastNormalZ_ = first.normalZ;
103 owner->shapeCastFraction_ = first.fraction;
104 return eve::Result<int>::success(static_cast<int>(owner->shapeCastResults_.size()));
105}

122eve::Result<CameraSphereHit3D> WorldBackend::sphereCast(float x, float y, float z, float x2, float y2, float z2,
123 float radius, uint64_t mask, int ignore) const {
124 const float dx = x2 - x, dy = y2 - y, dz = z2 - z;
125 if (!isValid()) return rejected<CameraSphereHit3D>("World expired", eve::DiagnosticCode::StaleHandle);
126 if (!isFinite(dx, dy, dz))
127 return rejected<CameraSphereHit3D>("Invalid sweep", eve::DiagnosticCode::InvalidArgument);
128 if (radius == 0) {
129 if (!isFinite(x, y, z)) return rejected<CameraSphereHit3D>("Invalid ray", eve::DiagnosticCode::InvalidArgument);
130 const JPH::RRayCast ray({x, y, z}, {dx, dy, dz});
131 JPH::AllHitCollisionCollector<JPH::CastRayCollector> hits;
132 state->system.GetNarrowPhaseQuery().CastRay(ray, JPH::RayCastSettings(), hits);
133 hits.Sort();
134 CameraSphereHit3D result;
135 for (const auto& hit : hits.mHits) {
136 const auto* native = state->system.GetBodyLockInterfaceNoLock().TryGetBody(hit.mBodyID);
137 if (!native) continue;
138 const auto* shape = child(*native, hit.mSubShapeID2);
139 if (!shape || !shape->owner || shape->sensor || !(shape->filter.categoryBits & mask)) continue;
140 auto body = shape->body.lock();
141 if (!body || !body->owner || body->owner->getId() == ignore) continue;
142 auto point = ray.GetPointOnRay(hit.mFraction);
143 auto normal = native->GetWorldSpaceSurfaceNormal(hit.mSubShapeID2, point);
144 result = {true,
145 body->owner->getId(),
146 hit.mFraction,
147 float(point.GetX()),
148 float(point.GetY()),
149 float(point.GetZ()),
150 normal.GetX(),
151 normal.GetY(),
152 normal.GetZ()};
153 break;
154 }
155 return eve::Result<CameraSphereHit3D>::success(result);
156 }
157 auto volume = capsuleVolume(x, y, z, x, y, z, radius);
158 if (!volume) return eve::Result<CameraSphereHit3D>::failure(volume.status());
159 auto hits = sweepVolume(*state, volume.value(), {dx, dy, dz});
160 CameraSphereHit3D result;
161 for (const auto& hit : hits.mHits) {
162 const auto* native = state->system.GetBodyLockInterfaceNoLock().TryGetBody(hit.mBodyID2);
163 if (!native) continue;
164 const auto* shape = child(*native, hit.mSubShapeID2);
165 if (!shape || !shape->owner || shape->sensor || !(shape->filter.categoryBits & mask)) continue;
166 auto body = shape->body.lock();
167 if (!body || !body->owner || body->owner->getId() == ignore) continue;
168 auto normal = -hit.mPenetrationAxis.NormalizedOr(JPH::Vec3::sAxisY());
169 auto point = volume.value().transform.GetTranslation() + hit.mContactPointOn2;
170 result = {true,
171 body->owner->getId(),
172 hit.mFraction,
173 float(point.GetX()),
174 float(point.GetY()),
175 float(point.GetZ()),
176 normal.GetX(),
177 normal.GetY(),
178 normal.GetZ()};
179 break;
180 }
181 return eve::Result<CameraSphereHit3D>::success(result);
182}
