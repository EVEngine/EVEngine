237[[nodiscard]] inline EventId deterministicContainerEventId(const ContainerId& container, const MembershipId& object,
238 ContainerEventKind kind, std::uint64_t serial) noexcept {
239 auto mix = [](std::uint64_t value, std::string_view text) noexcept {
240 for (const unsigned char byte : text) {
241 value ^= byte;
242 value *= 1099511628211ull;
243 }
244 return value;
245 };
246 std::uint64_t first = mix(14695981039346656037ull, container.value());
247 first = mix(first, object.value());
248 first = mix(first, std::string_view(containerEventTypeName(kind)));
249 std::uint64_t second = mix(1099511628211ull ^ serial, object.value());
250 second = mix(second, container.value());
251 second ^= serial + 0x9e3779b97f4a7c15ull;
252
253 EventId::Bytes bytes{};
254 for (std::size_t index = 0; index < sizeof(first); ++index)
255 bytes[index] = static_cast<std::uint8_t>((first >> (index * 8u)) & 0xffu);
256 for (std::size_t index = 0; index < sizeof(second); ++index)
257 bytes[sizeof(first) + index] = static_cast<std::uint8_t>((second >> (index * 8u)) & 0xffu);
258 // Mark the derived value as a conventional UUID variant/version while
259 // retaining deterministic bytes.
260 bytes[6] = static_cast<std::uint8_t>((bytes[6] & 0x0fu) | 0x50u);
261 bytes[8] = static_cast<std::uint8_t>((bytes[8] & 0x3fu) | 0x80u);
262 return EventId(bytes);
