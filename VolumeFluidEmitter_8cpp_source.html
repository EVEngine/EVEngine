65 size_t distributionOffset, size_t actorFree, float stepOffset, float dt) {
66 if (!valid(e) || count > 4096) return invalidEmission("Invalid nozzle or burst work limit");
67 if (count > 0 && e.shape == VolumeFluidEmissionShape::Distribution && e.distribution.empty())
68 return invalidEmission("Cannot emit a nonempty burst from an empty distribution");
69 if (count > solver.availableCapacity()) return invalidEmission("Burst exceeds available pool");
70 if (count > actorFree) return invalidEmission("Burst exceeds emitter actor capacity");
71 const auto forward = glm::normalize(e.direction);
72 const auto reference = std::abs(forward.y) < 0.9f ? glm::vec3(0, 1, 0) : glm::vec3(1, 0, 0);
73 const auto right = glm::normalize(glm::cross(reference, forward));
74 const auto up = glm::cross(forward, right);
75 const auto nozzleRotation = glm::quat_cast(glm::mat3(right, up, forward));
76 const auto prototypeRotation = glm::quat(e.prototype.orientation.w, e.prototype.orientation.x,
77 e.prototype.orientation.y, e.prototype.orientation.z);
78 uint32_t rng = e.seed + sequence * 0x9e3779b9u;
79 std::vector<VolumeFluidParticle> particles;
80 particles.reserve(count);
81 for (unsigned i = 0; i < count; ++i) {
82 const float a = sample(rng), b = sample(rng), c = sample(rng);
83 glm::vec3 local(0.f);
84 const VolumeFluidDistributionPoint* distributionPoint = nullptr;
85 switch (e.shape) {
86 case VolumeFluidEmissionShape::Distribution:
87 distributionPoint = &e.distribution[(distributionOffset + i) % e.distribution.size()];
88 local = distributionPoint->position;
89 break;
90 case VolumeFluidEmissionShape::Edge: local.x = (2.f * a - 1.f) * e.extent.x; break;
91 case VolumeFluidEmissionShape::Square:
92 local = {(2.f * a - 1.f) * e.extent.x, (2.f * b - 1.f) * e.extent.y, 0.f};
93 break;
94 case VolumeFluidEmissionShape::Cube:
95 local = {(2.f * a - 1.f) * e.extent.x, (2.f * b - 1.f) * e.extent.y, (2.f * c - 1.f) * e.extent.z};
96 break;
97 case VolumeFluidEmissionShape::Disk: {
98 const float radius = std::sqrt(a) * e.extent.x, angle = 6.28318530718f * b;
99 local = {radius * std::cos(angle), radius * std::sin(angle), 0.f};
100 break;
101 }
102 case VolumeFluidEmissionShape::Sphere: {
103 const float z = 2.f * a - 1.f, angle = 6.28318530718f * b;
104 const float xy = std::sqrt(std::max(0.f, 1.f - z * z)), radius = std::cbrt(c) * e.extent.x;
105 local = radius * glm::vec3(xy * std::cos(angle), xy * std::sin(angle), z);
106 break;
107 }
108 }
109 auto particle = e.prototype;
110 randomizeGranularRadius(particle, solver.spacing(), e.granularRadiusRandomness, rng);
111 const auto worldOrientation = glm::normalize(nozzleRotation * prototypeRotation);
112 particle.orientation = {worldOrientation.x, worldOrientation.y, worldOrientation.z, worldOrientation.w};
113 if (distributionPoint && e.useShapeColor) particle.color = distributionPoint->color;
114 particle.position = e.origin + right * local.x + up * local.y + forward * local.z;
115 const float jx = 2.f * sample(rng) - 1.f, jy = 2.f * sample(rng) - 1.f, jz = 2.f * sample(rng) - 1.f;
116 const auto emissionDirection =
117 distributionPoint ? nozzleRotation * glm::normalize(distributionPoint->direction) : forward;
118 const float randomZ = jx;
119 const float randomAngle = 3.14159265359f * (jy + 1.f);
120 const float randomXY = std::sqrt(std::max(0.f, 1.f - randomZ * randomZ));
121 const auto randomDirection =
122 glm::vec3(randomXY * std::cos(randomAngle), randomXY * std::sin(randomAngle), randomZ);
123 const auto spawnDirection = emissionDirection * (1.f - e.randomVelocity) + randomDirection * e.randomVelocity;
124 particle.velocity = spawnDirection * e.speed + e.jitter * glm::vec3(jx, jy, jz);
125 particle.position += spawnDirection * e.speed * dt * stepOffset;
126 particles.push_back(particle);
127 }
128 auto admitted = solver.emit(particles);
129 if (!admitted) return Result<unsigned>::failure(admitted.status());
130 return Result<unsigned>::success(count);
131}

134 state += 0x9e3779b9u;
135 uint32_t x = state;
136 x = (x ^ (x >> 16u)) * 0x85ebca6bu;
137 x = (x ^ (x >> 13u)) * 0xc2b2ae35u;

253 minimumPoolFraction < 0.f || minimumPoolFraction > 1.f)
254 return invalidEmission("Invalid stream duration, rate, threshold or work limit");

266 actorFree == 0 ||

268 credit_ = 0.0;

270 return Result<unsigned>::success(0);
271 }

273 const double whole = std::floor(accumulated);

283}

406 const float age = e.speed > 0.f ? distance / e.speed : 0.f;
407 const float fraction = dt > 0.f ? std::clamp(1.f - age / dt, 0.f, 1.f) : 1.f;
408 const auto orientation = glm::slerp(q0, q1, fraction);
409 const auto origin = glm::mix(begin.position, end.position, fraction);
410 for (const auto& point : points) {
411 auto p = e.prototype;
412 randomizeGranularRadius(p, spacing, e.granularRadiusRandomness, rng);
413 const auto worldOrientation = glm::normalize(orientation * prototypeRotation);
414 p.orientation = {worldOrientation.x, worldOrientation.y, worldOrientation.z, worldOrientation.w};
415 const auto offset = orientation * point.position;
416 const float jx = 2.f * sample(rng) - 1.f, jy = 2.f * sample(rng) - 1.f, jz = 2.f * sample(rng) - 1.f;
417 const auto emissionDirection = orientation * glm::normalize(point.direction);
418 p.velocity = emissionDirection * e.speed + e.jitter * glm::vec3(jx, jy, jz) +
419 inheritVelocity * (linear + glm::cross(omega, offset));
420 p.position = origin + offset + p.velocity * age;
421 if (e.useShapeColor) p.color = point.color;
422 batch.push_back(p);
423 }
424 }
425 auto admitted = solver.emit(batch);
426 if (!admitted) return Result<unsigned>::failure(admitted.status());
427 distance_ = remainder;
428 if (layers != 0) ++sequence_;
429 emitting_ = free - batch.size() >= points.size() && actorFree - batch.size() >= points.size();
430 return Result<unsigned>::success(unsigned(batch.size()));
431}

451 return distributionFailure("Sphere radius and spacing must be finite and positive", source);

453 if (!surface) {
454 const double ratio = double(radius) / spacing;
455 const double countValue = std::ceil(ratio - std::max(1.0, ratio) * 1e-7);
456 if (countValue <= 0.0 || countValue > 32767.0)
457 return distributionFailure("Sphere lattice exceeds 65536 candidates", source);
458 const uint64_t side = uint64_t(2.0 * countValue + 1.0);
459 if (side * side * side > 65536) return distributionFailure("Sphere lattice exceeds 65536 candidates", source);
460 const int count = int(countValue);

463 for (int y = -count; y <= count; ++y)
464 for (int x = -count; x <= count; ++x) {

480 uint64_t candidates = 0;
481 for (int ring = 0; ring <= rings; ++ring) {
482 const double polar = 3.141592653589793 * ring / rings;

484 const double sampleValue = std::ceil(2.0 * 3.141592653589793 * ringRadius / spacing);

486 return distributionFailure("Sphere lattice exceeds 65536 candidates", source);

489 if (candidates > 65536) return distributionFailure("Sphere lattice exceeds 65536 candidates", source);
490 for (int sample = 0; sample < samples; ++sample) {

493 if (!appendDistributionPoint(output, point, glm::normalize(point)))
494 return distributionFailure("Sphere distribution exceeds 4096 points", source);
495 }
496 }

582Result<VolumeFluidEmission> composeVolumeFluidEmitterShapes(const VolumeFluidEmission& base,
583 std::span<const VolumeFluidEmission> shapes) {
584 constexpr const char* source = "fluids.volume.emitterShapes";
585 if (shapes.size() > 64)
586 return Result<VolumeFluidEmission>::failure(
587 Diagnostic::error(DiagnosticCode::InvalidArgument, "Emitter shape count exceeds 64", source));
588
589 VolumeFluidEmission combined = base;
590 combined.shape = VolumeFluidEmissionShape::Distribution;
591 combined.origin = glm::vec3(0.f);
592 combined.direction = glm::vec3(0.f, 0.f, 1.f);
593 combined.distribution.clear();
594
595 const auto appendPoint = [&](glm::vec3 position, glm::vec3 direction, glm::vec4 color) {
596 if (combined.distribution.size() == 4096) return false;
597 combined.distribution.push_back({position, color, direction});
598 return true;
599 };
600 const auto appendShape = [&](const VolumeFluidEmission& shape) {
601 if (!valid(shape) || shape.shape != VolumeFluidEmissionShape::Distribution) return false;
602 const auto forward = glm::normalize(shape.direction);
603 const auto reference = std::abs(forward.y) < .9f ? glm::vec3(0, 1, 0) : glm::vec3(1, 0, 0);
604 const auto right = glm::normalize(glm::cross(reference, forward));
605 const auto up = glm::cross(forward, right);
606 if (shape.distribution.empty()) return appendPoint(shape.origin, forward, glm::vec4(1.f));
607 for (const auto& point : shape.distribution) {
608 const auto position =
609 shape.origin + right * point.position.x + up * point.position.y + forward * point.position.z;
610 const auto direction =
611 glm::normalize(right * point.direction.x + up * point.direction.y + forward * point.direction.z);
612 if (!appendPoint(position, direction, point.color)) return false;
613 }
614 return true;
615 };
616
617 if (shapes.empty()) {
618 if (!valid(base) || !appendPoint(base.origin, glm::normalize(base.direction), glm::vec4(1.f)))
619 return Result<VolumeFluidEmission>::failure(
620 Diagnostic::error(DiagnosticCode::InvalidArgument, "Invalid emitter-wide description", source));
621 } else {
622 for (const auto& shape : shapes)
623 if (!appendShape(shape))
624 return Result<VolumeFluidEmission>::failure(
625 Diagnostic::error(DiagnosticCode::InvalidArgument,
626 "Invalid emitter shape or distribution exceeds 4096 points", source));
627 }
628 if (!valid(combined))
629 return Result<VolumeFluidEmission>::failure(
630 Diagnostic::error(DiagnosticCode::InvalidArgument, "Invalid emitter-wide description", source));
631 return Result<VolumeFluidEmission>::success(std::move(combined));
632}
