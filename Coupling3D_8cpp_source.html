36Result<void> coupling_detail::validateResponse(World3D& world, const Row& row) {
37 if (!row.active) return Result<void>::success();
38 double response = 0;
39 for (int i = 0; i < row.count; ++i) {
40 const auto& t = row.terms[i];
41 auto* body = world.findBody(t.body);
42 if (!body || !body->isValid()) return fail<void>(DiagnosticCode::StaleHandle, "Coupling endpoint is stale");
43 if (b3Body_GetType(eve::physics::box3d::Access::bodyId(*body)) != b3_dynamicBody) continue;
44 response += b3Body_GetInverseMass(eve::physics::box3d::Access::bodyId(*body)) * b3Dot(t.linear, t.linear) +
45 b3Dot(t.angular,
46 b3MulMV(b3Body_GetWorldInverseRotationalInertia(eve::physics::box3d::Access::bodyId(*body)),
47 t.angular));
48 }
49 if (!std::isfinite(response) || response <= 0)
50 return fail<void>(DiagnosticCode::InvalidArgument, "Coupling has no finite movable degree of freedom");
51 return Result<void>::success();
52}
53Result<coupling_detail::Row> Coupling3D::Data::row() const {
54 using R = coupling_detail::Row;
55 if (!world || !world->isValid()) return fail<R>(DiagnosticCode::StaleHandle, "Coupling world is dead");
56 R out;
57 out.active = enabled;
58 std::array<b3BodyId, 4> raw{};
59 const int count = gear ? 4 : 2;
60 for (int i = 0; i < count; ++i) {
61 auto* body = world->findBody(bodies[i]);
62 if (!body || !body->isValid()) return fail<R>(DiagnosticCode::StaleHandle, "Coupling body is dead");
63 raw[i] = eve::physics::box3d::Access::bodyId(*body);
64 out.active = out.active && b3Body_IsEnabled(raw[i]);
65 }
66 if (gear) {
67 for (int i = 0; i < 2; ++i) {
68 auto* joint = world->findJoint(sources[i]);
69 if (!joint || !joint->isValid()) return fail<R>(DiagnosticCode::StaleHandle, "Gear source is dead");
70 const auto a = raw[2 * i], b = raw[2 * i + 1];
71 const auto frame = b3Joint_GetLocalFrameA(eve::physics::box3d::Access::jointId(*joint));
72 const auto transform = b3Body_GetTransform(a);
73 const b3Vec3 axis =
74 b3RotateVector(transform.q, b3RotateVector(frame.q, revolute[i] ? b3Vec3_axisZ : b3Vec3_axisX));
75 const float scale = i == 0 ? 1.f : gearSettings.ratio;
76 if (revolute[i]) {
77 const float angle = b3RevoluteJoint_GetAngle(eve::physics::box3d::Access::jointId(*joint));
78 out.wrapped[i] = angle;
79 out.coordinate[i] =
80 coordinate[i] + std::remainder(static_cast<double>(angle) - previous[i], 2 * std::numbers::pi);
81 out.add(bodies[2 * i], {}, b3Neg(axis), scale);
82 out.add(bodies[2 * i + 1], {}, axis, scale);
83 } else {
84 const auto anchorA = b3Body_GetWorldPoint(a, frame.p);
85 const auto anchorB =
86 b3Body_GetWorldPoint(b, b3Joint_GetLocalFrameB(eve::physics::box3d::Access::jointId(*joint)).p);
87 const auto d = b3SubPos(anchorB, anchorA);
88 const auto rA = b3SubPos(anchorA, b3Body_GetWorldCenterOfMass(a));
89 const auto rB = b3SubPos(anchorB, b3Body_GetWorldCenterOfMass(b));
90 out.coordinate[i] = b3Dot(axis, d);
91 out.add(bodies[2 * i], b3Neg(axis), b3Neg(b3Cross(b3Add(rA, d), axis)), scale);
92 out.add(bodies[2 * i + 1], axis, b3Cross(rB, axis), scale);
93 }
94 }
95 out.error = out.coordinate[0] + gearSettings.ratio * out.coordinate[1] - reference;
96 } else {
97 const std::array<CouplingPoint3D, 2> local{pulleySettings.localAnchorA, pulleySettings.localAnchorB};
98 const std::array<CouplingPoint3D, 2> guide{pulleySettings.groundA, pulleySettings.groundB};
99 for (int i = 0; i < 2; ++i) {
100 const auto anchor = b3Body_GetWorldPoint(raw[i], vec(local[i]));
101 const auto d = b3SubPos(anchor, pos(guide[i]));
102 const float length = b3Length(d);
103 out.coordinate[i] = length;
104 if (!std::isfinite(length) || (out.active && length <= 1e-6f))
105 return fail<R>(DiagnosticCode::InvalidArgument, "Pulley segment is singular or non-finite");
106 const auto direction = length > 1e-6f ? b3MulSV(1.f / length, d) : b3Vec3{};
107 const auto offset = b3SubPos(anchor, b3Body_GetWorldCenterOfMass(raw[i]));
108 out.add(bodies[i], direction, b3Cross(offset, direction), i == 0 ? 1.f : pulleySettings.ratio);
109 }
110 out.error = out.coordinate[0] + pulleySettings.ratio * out.coordinate[1] - pulleySettings.length;
111 }
112 if (!std::isfinite(out.error)) return fail<R>(DiagnosticCode::InvalidArgument, "Coupling coordinate overflow");
113 for (int i = 0; i < out.count; ++i)
114 if (!b3IsValidVec3(out.terms[i].linear) || !b3IsValidVec3(out.terms[i].angular))
115 return fail<R>(DiagnosticCode::InvalidArgument, "Coupling Jacobian overflow");
116 return Result<R>::success(out);
117}

163 const GearCoupling3DSettings& settings) {

167 if (!world.isValid() || !first.isValid() || !second.isValid() || &first == &second || first.getWorld() != &world ||
168 second.getWorld() != &world || !valid(settings.limits) || !std::isfinite(settings.ratio) || settings.ratio == 0)
169 return fail<R>(DiagnosticCode::InvalidArgument, "Live same-world sources and finite gear settings required");

178 for (int i = 0; i < 2; ++i) {

180 if (type != b3_revoluteJoint && type != b3_prismaticJoint)
181 return fail<R>(DiagnosticCode::Unsupported, "Gear requires revolute/prismatic sources");
182 d->revolute[i] = type == b3_revoluteJoint;
183 d->sources[i] = sources[i]->runtimeHandle();
184 auto* a = world.findBodyById(sources[i]->getBodyAId());
185 auto* b = world.findBodyById(sources[i]->getBodyBId());
186 if (!a || !b) return fail<R>(DiagnosticCode::StaleHandle, "Gear body is missing");
187 d->bodies[2 * i] = a->runtimeHandle();
188 d->bodies[2 * i + 1] = b->runtimeHandle();
189 }
190 auto row = d->row();
191 if (!row) return Result<R>::failure(row.status());
192 auto movable = coupling_detail::validateResponse(world, row.value());
193 if (!movable) return Result<R>::failure(movable.status());
194 d->reference = row.value().coordinate[0] + settings.ratio * row.value().coordinate[1];
195 d->previous = row.value().wrapped;
196 d->coordinate = row.value().coordinate;
197 d->id = world.nextCoupling3DId_++;
198 R owned(new Coupling3D(std::move(d)));
199 world.couplings3D_.emplace(owned->getId(), owned.get());
200 return Result<R>::success(std::move(owned));
