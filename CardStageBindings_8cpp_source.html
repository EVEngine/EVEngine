40Result<void> Card::stagePlay(transaction::AtomicBatch& batch, CardData& card) const {
41 auto cost = playCost(card);
42 if (!cost) return batch.reject(cost.status());
43 auto hand = prepareHandConsumption(card);
44 if (!hand) return batch.reject(hand.status());
45 std::vector<std::unique_ptr<transaction::ITransactionParticipant>> effects;
46 effects.push_back(std::make_unique<PlayRules>(card, *findDef(card.identity()->definitionId), batch));
47 effects.push_back(std::make_unique<CardPlayParticipant>(card));
48 effects.push_back(std::move(hand).takeValue());
49 return batch.add(std::move(effects), std::move(cost).takeValue());
