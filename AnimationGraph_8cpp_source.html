40GraphConnectionDecision AnimationStateGraphDomain::canConnect(const GraphPinRecord& from,
41 const GraphPinRecord& to) const {
42 GraphConnectionDecision decision;
43 decision.allowed = from.direction == GraphPinDirection::Output &&
44 to.direction == GraphPinDirection::Input && from.type == "animation.flow" &&
45 to.type == "animation.flow" && from.node != to.node;
46 if (!decision.allowed)
47 decision.diagnostics.push_back(eve::editing::ruleDiagnostic(
48 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.animation.invalid-connection"),
49 DiagnosticSeverity::Error,
50 "Animation graph requires flow output-to-input between distinct nodes"));
51 return decision;
52}

94AnimationGraphCompileResult AnimationStateGraphDomain::compile(const GraphDocumentData& graph) const {
95 AnimationGraphCompileResult result;
96 result.documentRevision = graph.revision;
97 if (graph.domain != domain()) {
98 result.status = EditorStatus::Rejected;
99 diagnostic(result, "editor.animation.wrong-domain", "Graph is not an animation state document");
100 return result;
101 }
102 if (graph.schemaVersion != 1) {
103 result.status = EditorStatus::Unsupported;
104 diagnostic(result, "editor.animation.unsupported-schema", "Animation graph schema version is unsupported");
105 return result;
106 }
107 const auto* parameters = graph.parameters.getIf<EditorValue::Object>();
108 const EditorValue* entryValue = field(graph.parameters, "entry");
109 const auto* entry = entryValue ? entryValue->getIf<std::string>() : nullptr;
110 if (!parameters || !entry || entry->empty()) {
111 result.status = EditorStatus::Failed;
112 diagnostic(result, "editor.animation.entry-required", "Animation graph requires an entry state parameter");
113 return result;
114 }
115
116 std::map<GraphNodeId, const GraphNodeRecord*> nodes;
117 std::map<GraphPinId, const GraphNodeRecord*> pinOwners;
118 for (const GraphNodeRecord& node : graph.nodes) {
119 if (node.type != "animation.state" && node.type != "animation.transition") {
120 diagnostic(result, "editor.animation.unknown-node", "Unknown animation node type: " + node.type);
121 continue;
122 }
123 if (!nodes.emplace(node.id, &node).second)
124 diagnostic(result, "editor.animation.duplicate-node", "Duplicate animation node id: " + node.id.value());
125 for (const GraphPinRecord& pin : node.pins) pinOwners.emplace(pin.id, &node);
126 }
127 const auto entryNode = nodes.find(GraphNodeId(*entry));
128 if (entryNode == nodes.end() || entryNode->second->type != "animation.state")
129 diagnostic(result, "editor.animation.invalid-entry", "Entry does not reference an animation state node");
130
131 std::map<GraphNodeId, GraphNodeId> transitionFrom;
132 std::map<GraphNodeId, GraphNodeId> transitionTo;
133 for (const GraphEdgeRecord& edge : graph.edges) {
134 const auto fromOwner = pinOwners.find(edge.from);
135 const auto toOwner = pinOwners.find(edge.to);
136 if (fromOwner == pinOwners.end() || toOwner == pinOwners.end()) {
137 diagnostic(result, "editor.animation.dangling-edge", "Animation edge references a missing pin");
138 continue;
139 }
140 const GraphNodeRecord* from = fromOwner->second;
141 const GraphNodeRecord* to = toOwner->second;
142 if (from->type == "animation.state" && to->type == "animation.transition") {
143 if (!transitionFrom.emplace(to->id, from->id).second)
144 diagnostic(result, "editor.animation.multiple-transition-sources",
145 "Transition has more than one source state: " + to->id.value());
146 } else if (from->type == "animation.transition" && to->type == "animation.state") {
147 if (!transitionTo.emplace(from->id, to->id).second)
148 diagnostic(result, "editor.animation.multiple-transition-destinations",
149 "Transition has more than one destination state: " + from->id.value());
150 } else
151 diagnostic(result, "editor.animation.invalid-flow",
152 "Flow must alternate state and transition nodes");
153 }
154
155 std::ostringstream definition;
156 definition << "EVANIM_STATE_GRAPH 1\n" << "entry " << quote(*entry) << '\n';
157 for (const auto& [id, node] : nodes) {
158 if (node->type != "animation.state") continue;
159 const auto* clipValue = field(node->properties, "clip");
160 const auto* speedValue = field(node->properties, "speed");
161 const auto* loopValue = field(node->properties, "loop");
162 const auto* clip = clipValue ? clipValue->getIf<std::string>() : nullptr;
163 const auto* speed = speedValue ? speedValue->getIf<double>() : nullptr;
164 const auto* loop = loopValue ? loopValue->getIf<bool>() : nullptr;
165 if (!clip || clip->empty() || !speed || *speed <= 0.0 || !loop) {
166 diagnostic(result, "editor.animation.invalid-state-properties",
167 "State requires clip, positive speed and loop properties: " + id.value());
168 continue;
169 }
170 definition << "state " << quote(id.value()) << ' ' << quote(*clip) << ' ' << *speed << ' '
171 << (*loop ? 1 : 0) << '\n';
172 }
173 for (const auto& [id, node] : nodes) {
174 if (node->type != "animation.transition") continue;
175 if (!transitionFrom.contains(id) || !transitionTo.contains(id)) {
176 diagnostic(result, "editor.animation.incomplete-transition",
177 "Transition requires exactly one source and destination: " + id.value());
178 continue;
179 }
180 const auto* blend = field(node->properties, "blendSeconds");
181 const auto* hasExit = field(node->properties, "hasExitTime");
182 const auto* exitTime = field(node->properties, "exitTime");
183 const auto* kind = field(node->properties, "conditionKind");
184 const auto* parameter = field(node->properties, "parameter");
185 const auto* op = field(node->properties, "operator");
186 const auto* threshold = field(node->properties, "threshold");
187 const auto* boolValue = field(node->properties, "boolValue");
188 const auto* blendNumber = blend ? blend->getIf<double>() : nullptr;
189 const auto* exitEnabled = hasExit ? hasExit->getIf<bool>() : nullptr;
190 const auto* exitNumber = exitTime ? exitTime->getIf<double>() : nullptr;
191 const auto* kindName = kind ? kind->getIf<std::string>() : nullptr;
192 const auto* parameterName = parameter ? parameter->getIf<std::string>() : nullptr;
193 const auto* operatorName = op ? op->getIf<std::string>() : nullptr;
194 const auto* thresholdNumber = threshold ? threshold->getIf<double>() : nullptr;
195 const auto* boolFlag = boolValue ? boolValue->getIf<bool>() : nullptr;
196 const bool kindValid = kindName && (*kindName == "none" || *kindName == "float" ||
197 *kindName == "bool" || *kindName == "trigger");
198 const bool opValid = operatorName && (*operatorName == ">" || *operatorName == ">=" ||
199 *operatorName == "<" || *operatorName == "<=" ||
200 *operatorName == "==" || *operatorName == "!=");
201 if (!blendNumber || *blendNumber < 0.0 || !exitEnabled || !exitNumber || *exitNumber < 0.0 ||
202 *exitNumber > 1.0 || !kindValid || !parameterName || !operatorName || !thresholdNumber ||
203 !boolFlag || (*kindName != "none" && parameterName->empty()) ||
204 (*kindName == "float" && !opValid)) {
205 diagnostic(result, "editor.animation.invalid-transition-properties",
206 "Transition metadata is invalid: " + id.value());
207 continue;
208 }
209 definition << "transition " << quote(id.value()) << ' ' << quote(transitionFrom[id].value()) << ' '
210 << quote(transitionTo[id].value()) << ' ' << *blendNumber << ' '
211 << (*exitEnabled ? 1 : 0) << ' ' << *exitNumber << ' ' << quote(*kindName) << ' '
212 << quote(*parameterName) << ' ' << quote(*operatorName) << ' ' << *thresholdNumber << ' '
213 << (*boolFlag ? 1 : 0) << '\n';
214 }
215 if (!result.diagnostics.empty()) {
216 result.status = EditorStatus::Failed;
217 return result;
218 }
219 result.status = EditorStatus::Applied;
220 result.definition = definition.str();
221 return result;
222}
