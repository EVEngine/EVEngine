41void Event::pump() {
42 EV_PROFILE_MODULE("event", "Event::pump");
43 SDL_Event e;
44
45 while (SDL_PollEvent(&e)) {
46 Message *msg = convert(e);
47 if (msg) push(msg);
48 releaseOwnedEventPayload(e);
49 // Ownership stays in the queue until poll()/clear() — do not delete here.
50 }

52 // Per-frame work that modules hang off the pump (audio streaming, ...).
53 cap::forEach<IPlatformEventSink>([](IPlatformEventSink *sink) { sink->onPumpFinished(); });
54}

56Message *Event::wait() {
57 for (;;) {
58 if (Message *queued = poll()) return queued;
59
60 SDL_Event e;
61 if (SDL_WaitEvent(&e) != 1) return nullptr;
62 if (e.type == wakeEventType_) continue;
63
64 Message *message = convert(e);
65 releaseOwnedEventPayload(e);
66 if (message) return message;
67 }
68}

88Message *Event::convert(const SDL_Event &e) {
89 // Modules that own an event family translate it themselves; see
90 // event/PlatformEventSink.h. Only window-close and quit are handled here,
91 // because they are about the event loop rather than any one module.
92 Message *translated = nullptr;
93 const bool consumed = cap::forEachUntil<IPlatformEventSink>([&](IPlatformEventSink *sink) {
94 if (sink->shouldConsumePlatformEvent(&e)) return true;
95 translated = sink->translatePlatformEvent(&e);
96 return translated != nullptr;
97 });
98 if (translated) return translated;
99 if (consumed) return nullptr;
100
101 switch (e.type) {
102 case SDL_QUIT: return new Message("quit");
103 case SDL_WINDOWEVENT:
104 if (e.window.event == SDL_WINDOWEVENT_CLOSE) return new Message("quit");
105 break;
106 default: break;
107 }
108 return nullptr;
109}
