19Result<void> CrowdFlowNavigation::replaceGroups(crowd::Crowd& crowd, std::vector<GroupInput> groups) try {
20 if (groups.size() > 64)
21 return groupFailure<void>(DiagnosticCode::InvalidArgument, "Expected at most 64 shared groups");
22 const auto source = crowd.getFieldStatus();
23 std::vector<GroupBinding> staged;
24 for (const auto& group : groups) {
25 if (group.members.empty() || group.members.size() > 1024)
26 return groupFailure<void>(DiagnosticCode::InvalidArgument, "Expected 1..1024 group members");
27 auto normalized = goalSet(crowd, group.targets);
28 if (!normalized) return Result<void>::failure(normalized.status());
29 std::shared_ptr<const std::vector<crowd::FlowGoal>> goals;
30 const auto priorGroup = std::lower_bound(
31 groups_.begin(), groups_.end(), group.members.front().unit,
32 [](const auto& candidate, const auto& handle) { return handleLess(candidate.member.unit, handle); });
33 if (priorGroup != groups_.end() && !handleLess(group.members.front().unit, priorGroup->member.unit) &&
34 priorGroup->owner == source.owner && priorGroup->input == source.inputRevision &&
35 *priorGroup->goals == normalized.value())
36 goals = priorGroup->goals;
37 else
38 goals = std::make_shared<const std::vector<crowd::FlowGoal>>(std::move(normalized).takeValue());
39 for (const auto& member : group.members) {
40 if (member.unit.table == nullptr || member.unit.generation == 0 || member.orderId.empty())
41 return groupFailure<void>(DiagnosticCode::InvalidArgument, "Invalid group member binding");
42 GroupBinding binding{member, goals, false, source.owner, source.inputRevision};
43 const auto previous = std::lower_bound(
44 groups_.begin(), groups_.end(), member.unit,
45 [](const auto& candidate, const auto& handle) { return handleLess(candidate.member.unit, handle); });
46 if (previous != groups_.end() && !handleLess(member.unit, previous->member.unit) &&
47 previous->member.orderId == member.orderId && previous->owner == source.owner &&
48 previous->input == source.inputRevision && previous->goals == goals)
49 binding.reform = previous->reform;
50 staged.push_back(std::move(binding));
51 }
52 }
53 std::sort(staged.begin(), staged.end(),
54 [](const auto& left, const auto& right) { return handleLess(left.member.unit, right.member.unit); });
55 if (std::adjacent_find(staged.begin(), staged.end(), [](const auto& left, const auto& right) {
56 return !handleLess(left.member.unit, right.member.unit);
57 }) != staged.end())
58 return groupFailure<void>(DiagnosticCode::Conflict, "Unit belongs to multiple shared groups");
59 groups_.swap(staged);
60 return Result<void>::success();
61} catch (const std::exception& error) {
62 return groupFailure<void>(DiagnosticCode::Failed, error.what());
63}

65Result<CrowdFlowNavigation::Route> CrowdFlowNavigation::requestRoute(crowd::Crowd& crowd, ecs::EntityHandle unit,
66 std::string_view orderId, WorldPosition position,
67 WorldPosition target, SimulationTick tick) {
68 auto exact = goalCell(crowd, target);
69 if (!exact) return Result<Route>::failure(exact.status());
70 auto cell = goalCell(crowd, position);
71 if (!cell) return Result<Route>::failure(cell.status());
72 const auto source = crowd.getFieldStatus();
73 Route route;
74 route.exactGoal_ = exact.value();
75 route.target_ = target;
76 route.owner_ = source.owner;
77 route.input_ = source.inputRevision;
78 auto member = std::lower_bound(groups_.begin(), groups_.end(), unit, [](const auto& binding, const auto& handle) {
79 return handleLess(binding.member.unit, handle);
80 });
81 bool reform = false;
82 const bool matched = member != groups_.end() && !handleLess(unit, member->member.unit) &&
83 member->member.orderId == orderId && member->owner == source.owner &&
84 member->input == source.inputRevision &&
85 std::binary_search(member->goals->begin(), member->goals->end(), exact.value());
86 if (matched) {
87 // Once a member enters any slot cell, it follows its own terminal route until membership changes.
88 // Keeping this derived phase across group refresh prevents oscillation back to the nearest wrong slot.
89 reform = member->reform || std::binary_search(member->goals->begin(), member->goals->end(), cell.value());
90 if (!reform) route.sharedGoals_ = member->goals;
91 }
92 const auto goals = route.sharedGoals_ ? std::span<const crowd::FlowGoal>(*route.sharedGoals_)
93 : std::span<const crowd::FlowGoal>(&route.exactGoal_, 1);
94 auto requested = requestGoals(goals, tick);
95 if (!requested) return Result<Route>::failure(requested.status());
96 if (matched) member->reform = reform;
97 return Result<Route>::success(std::move(route));
98}
