16UiThemePreviewSnapshot UiThemePreviewService::build(const UiThemeCatalogTarget& catalog, const ObjectId& themeId,
17 Revision expectedRevision) const {
18 UiThemePreviewSnapshot snapshot;
19 snapshot.documentRevision = catalog.revision();
20 snapshot.themeId = themeId;
21 if (expectedRevision != catalog.revision()) {
22 snapshot.status = EditorStatus::Conflict;
23 snapshot.diagnostics.push_back(
24 previewDiagnostic("editor.ui-theme.stale-preview", "Theme preview requires the current catalog revision"));
25 return snapshot;
26 }
27 auto asset = catalog.theme(themeId);
28 if (!asset.ok()) {
29 snapshot.status = EditorStatus::NotFound;
30 snapshot.diagnostics.insert(snapshot.diagnostics.end(), asset.status().diagnostics().begin(),
31 asset.status().diagnostics().end());
32 return snapshot;
33 }
34 snapshot.theme = asset.value().tokens;
35 snapshot.runtimeName = catalog.runtimeName(themeId);
36 snapshot.status = EditorStatus::Applied;
37 return snapshot;
38}

40EditorResult<void> UiThemeRuntimePublisher::publish(const UiThemeCatalogTarget& catalog) const {
41 auto asset = catalog.theme(catalog.activeId());
42 if (!asset.ok()) return EditorResult<void>::failure(asset.status());
43 auto diagnostics = validateThemeTokens(asset.value().tokens);
44 if (!diagnostics.empty())
45 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.ui-theme.publish"),
46 "Active theme tokens are invalid");
47 ui::setGlobalTheme(asset.value().tokens, catalog.runtimeName(catalog.activeId()));
48 return eve::editing::applied<void>();
