20eve::Result<void> lintConversationWorkspace(const std::vector<ConversationAsset>& assets, const std::string& label,
21 std::vector<ConversationDiagnostic>& diagnostics) {
22 const size_t diagnosticsBegin = diagnostics.size();
23 auto linted = lintConversations(assets, label, diagnostics);
24 bool valid = linted.ok();
25 std::unordered_set<std::string> assetIds;
26 for (const auto& asset : assets) assetIds.insert(asset.id);
27 for (const auto& asset : assets) {
28 for (const auto& node : asset.nodes) {
29 if (node.kind != ConversationAsset::Node::Kind::Call || node.target.empty()) continue;
30 if (assetIds.find(node.target) == assetIds.end()) {
31 diagnostics.push_back({ConversationDiagnostic::Severity::Error, label, node.sourceLine,
32 "conversation '" + asset.id + "': call node '" + node.id +
33 "' references missing conversation '" + node.target + "'",
34 "MissingConversation", node.sourceColumn, asset.id + "/" + node.id});
35 valid = false;
36 continue;
37 }
38 const auto target = std::find_if(assets.begin(), assets.end(), [&](const auto& candidate) {
39 return candidate.id == node.target;
40 });
41 for (const auto& parameter : target->parameters) {
42 if (parameter.required && !node.arguments.find(parameter.name)) {
43 diagnostics.push_back({ConversationDiagnostic::Severity::Error, label, node.sourceLine,
44 "conversation '" + asset.id + "': call node '" + node.id +
45 "' omits required argument '" + parameter.name + "'",
46 "MissingCallArgument", node.sourceColumn, asset.id + "/" + node.id});
47 valid = false;
48 }
49 }
50 for (const auto& argument : node.arguments.keys()) {
51 if (std::none_of(target->parameters.begin(), target->parameters.end(), [&](const auto& parameter) {
52 return parameter.name == argument;
53 })) {
54 diagnostics.push_back({ConversationDiagnostic::Severity::Error, label, node.sourceLine,
55 "conversation '" + asset.id + "': call node '" + node.id +
56 "' passes undeclared argument '" + argument + "'",
57 "UnexpectedCallArgument", node.sourceColumn, asset.id + "/" + node.id});
58 valid = false;
59 }
60 }
61 }
62 }
63 valid = valid &&
64 std::none_of(diagnostics.begin() + static_cast<std::ptrdiff_t>(diagnosticsBegin), diagnostics.end(),
65 [](const auto& diagnostic) {
66 return diagnostic.severity == ConversationDiagnostic::Severity::Error;
67 });
68 if (!valid) {
69 const std::string message = diagnostics.size() > diagnosticsBegin ? diagnostics[diagnosticsBegin].message
70 : "dialogue workspace lint failed";
71 return eve::Result<void>::failure(eve::Diagnostic::error(
72 eve::DiagnosticCode::InvalidArgument, message, label, {}, "dialogue.workspace.lint"));
73 }
74 return eve::Result<void>::success();
75}

77eve::Result<void> renameConversationAsset(std::vector<ConversationAsset>& assets, const std::string& oldId,
78 const std::string& newId) {
79 if (oldId.empty() || newId.empty()) return renameFailure("conversation IDs must not be empty", oldId);
80 auto source = std::find_if(assets.begin(), assets.end(), [&](const auto& asset) { return asset.id == oldId; });
81 if (source == assets.end()) return renameFailure("conversation not found: " + oldId, oldId);
82 if (std::any_of(assets.begin(), assets.end(), [&](const auto& asset) { return asset.id == newId; }))
83 return renameFailure("conversation already exists: " + newId, newId);
84 source->id = newId;
85 ++source->version;
86 for (auto& asset : assets)
87 for (auto& node : asset.nodes)
88 if (node.kind == ConversationAsset::Node::Kind::Call && node.target == oldId) node.target = newId;
89 return eve::Result<void>::success();
90}

93 const std::string& oldId, const std::string& newId) {

97 auto node =
98 std::find_if(asset->nodes.begin(), asset->nodes.end(), [&](const auto& item) { return item.id == oldId; });
99 if (node == asset->nodes.end()) return renameFailure("node not found: " + oldId, assetId + "/" + oldId);
100 if (asset->findNode(newId)) return renameFailure("node already exists: " + newId, assetId + "/" + newId);
101 node->id = newId;
102 rewriteNodeReference(asset->entry, oldId, newId);
103 for (auto& item : asset->nodes) {
104 rewriteNodeReference(item.next, oldId, newId);
105 rewriteNodeReference(item.returnNode, oldId, newId);
106 for (auto& route : item.routes) rewriteNodeReference(route.second, oldId, newId);
107 }
108 ++asset->version;
109 return eve::Result<void>::success();
