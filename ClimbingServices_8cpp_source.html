16 if (requirement == ClimbingCommandRequirement::Any) return true;

26eve::Result<ClimbingServiceStart> ClimbingServiceSelectionSystem::tryStart(
27 ClimbingRuntime& runtime, physics::World3D& world, const ClimbingPose& pose, ClimbingIntent& intent,
28 ClimbingCommand command, eve::SimulationTick tick, eve::SimulationTick lastGroundedTick,
29 ClimbingServiceSubject subject) {
30 if (subject.isZero())
31 return failure<ClimbingServiceStart>(eve::DiagnosticCode::InvalidArgument,
32 "service subject must be non-zero", "subject");
33 if (!ClimbingInputSystem::peek(intent, command, tick))
34 return failure<ClimbingServiceStart>(eve::DiagnosticCode::NotFound, "climbing.input.no_match",
35 "intent.commands");
36
37 ClimbingPose effectivePose = pose;
38 if (!effectivePose.grounded && command != ClimbingCommand::Drop &&
39 ClimbingInputSystem::coyoteWindowState(tick, lastGroundedTick, runtime.profile_.coyoteTicks) ==
40 ClimbingCoyoteState::Eligible)
41 effectivePose.grounded = true;
42 auto prepared = runtime.prepareBegin(world, effectivePose, tick);
43 if (!prepared) return eve::Result<ClimbingServiceStart>::failure(prepared.status());
44 if (!commandMatches(prepared.value().action.requiredCommand, command))
45 return failure<ClimbingServiceStart>(eve::DiagnosticCode::NotFound,
46 "selected action does not accept the buffered command",
47 "action.requiredCommand");
48
49 if (!prepared.value().action.requiredConditionTags.empty()) {
50 IClimbingConditionAuthority* conditions = eve::cap::query<IClimbingConditionAuthority>();
51 if (!conditions)
52 return failure<ClimbingServiceStart>(eve::DiagnosticCode::Unsupported,
53 "required climbing condition provider is absent", "conditions");
54 auto decision = conditions->evaluate(subject, prepared.value().action.requiredConditionTags, tick);
55 if (!decision) return eve::Result<ClimbingServiceStart>::failure(decision.status());
56 if (decision.value() != ClimbingConditionDecision::Allowed)
57 return failure<ClimbingServiceStart>(eve::DiagnosticCode::PreconditionViolation,
58 "climbing action conditions were denied", "conditions");
59 prepared.value().conditionsSatisfied = true;
60 }
61
62 IClimbingStaminaAuthority* stamina = nullptr;
63 ClimbingStaminaReservation reservation = ClimbingStaminaReservation::zero();
64 ClimbingOptionalServiceState staminaState = ClimbingOptionalServiceState::Disabled;
65 if (prepared.value().action.staminaCost > 0.f &&
66 runtime.profile_.staminaPolicy != ClimbingStaminaPolicy::Disabled) {
67 stamina = eve::cap::query<IClimbingStaminaAuthority>();
68 if (!stamina && runtime.profile_.staminaPolicy == ClimbingStaminaPolicy::RequireProvider)
69 return failure<ClimbingServiceStart>(eve::DiagnosticCode::Unsupported,
70 "required climbing stamina provider is absent", "stamina");
71 if (!stamina) {
72 staminaState = ClimbingOptionalServiceState::ProviderAbsent;
73 } else {
74 if (!runtime.profile_.staminaAdapter.empty() &&
75 stamina->adapterId() != runtime.profile_.staminaAdapter)
76 return failure<ClimbingServiceStart>(eve::DiagnosticCode::Unsupported,
77 "configured climbing stamina adapter is not available",
78 "staminaAdapter");
79 auto reserved = stamina->prepare(subject, prepared.value().action.staminaCost, tick);
80 if (!reserved) return eve::Result<ClimbingServiceStart>::failure(reserved.status());
81 reservation = reserved.value();
82 if (reservation.isZero())
83 return failure<ClimbingServiceStart>(eve::DiagnosticCode::InvariantViolation,
84 "stamina provider returned a zero reservation", "stamina");
85 staminaState = ClimbingOptionalServiceState::Applied;
86 }
87 }
88
89 const ClimbingIntent originalIntent = intent;
90 auto consumed = ClimbingInputSystem::consume(intent, command, tick, prepared.value().executionId);
91 if (!consumed || !consumed.value()) {
92 intent = originalIntent;
93 if (stamina) stamina->cancelPrepared(reservation);
94 if (!consumed) return eve::Result<ClimbingServiceStart>::failure(consumed.status());
95 return failure<ClimbingServiceStart>(eve::DiagnosticCode::Conflict,
96 "eligible climbing input changed before commit", "intent.commands");
97 }
98 auto committed = runtime.commitBegin(std::move(prepared).takeValue());
99 if (!committed) {
100 intent = originalIntent;
101 if (stamina) stamina->cancelPrepared(reservation);
102 return eve::Result<ClimbingServiceStart>::failure(committed.status());
103 }
104 if (stamina) stamina->commitPrepared(reservation, committed.value().executionId);
105 return eve::Result<ClimbingServiceStart>::success({std::move(committed).takeValue(), staminaState},
106 eve::Status::success(eve::StatusCode::Applied));
107}

109eve::Result<ClimbingOptionalServiceState> dispatchClimbingEvents(
110 ClimbingServiceSubject subject, std::span<const ClimbingEvent> events) {
111 if (subject.isZero())
