22bool commandMatches(ClimbingCommandRequirement requirement, ClimbingCommand command) {
23 if (requirement == ClimbingCommandRequirement::Any) return true;
24 return (requirement == ClimbingCommandRequirement::Jump && command == ClimbingCommand::Jump) ||
25 (requirement == ClimbingCommandRequirement::Climb && command == ClimbingCommand::Climb) ||
26 (requirement == ClimbingCommandRequirement::Drop && command == ClimbingCommand::Drop) ||
27 (requirement == ClimbingCommandRequirement::Sprint && command == ClimbingCommand::Sprint) ||
28 (requirement == ClimbingCommandRequirement::Crouch && command == ClimbingCommand::Crouch);
29}

33eve::Result<ClimbingServiceStart> ClimbingServiceSelectionSystem::tryStart(
34 ClimbingRuntime& runtime, physics::World3D& world, const ClimbingPose& pose, ClimbingIntent& intent,
35 ClimbingCommand command, eve::SimulationTick tick, eve::SimulationTick lastGroundedTick,
36 ClimbingServiceSubject subject) {
37 if (subject.isZero())
38 return eve::Result<ClimbingServiceStart>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
39 "service subject must be non-zero",
40 "subject", {}, "climbing.services"));
41 if (!ClimbingInputSystem::peek(intent, command, tick))
42 return eve::Result<ClimbingServiceStart>::failure(eve::Diagnostic::error(
43 eve::DiagnosticCode::NotFound, "climbing.input.no_match", "intent.commands", {}, "climbing.services"));
44
45 ClimbingPose effectivePose = pose;
46 if (!effectivePose.grounded && command != ClimbingCommand::Drop &&
47 ClimbingInputSystem::coyoteWindowState(tick, lastGroundedTick, runtime.profile_.coyoteTicks) ==
48 ClimbingCoyoteState::Eligible)
49 effectivePose.grounded = true;
50 auto prepared = runtime.prepareBegin(world, effectivePose, tick);
51 if (!prepared) return eve::Result<ClimbingServiceStart>::failure(prepared.status());
52 if (!commandMatches(prepared.value().action.requiredCommand, command))
53 return eve::Result<ClimbingServiceStart>::failure(eve::Diagnostic::error(
54 eve::DiagnosticCode::NotFound, "selected action does not accept the buffered command",
55 "action.requiredCommand", {}, "climbing.services"));
56
57 if (!prepared.value().action.requiredConditionTags.empty()) {
58 IClimbingConditionAuthority* conditions = eve::cap::query<IClimbingConditionAuthority>();
59 if (!conditions)
60 return eve::Result<ClimbingServiceStart>::failure(eve::Diagnostic::error(
61 eve::DiagnosticCode::Unsupported, "required climbing condition provider is absent", "conditions", {},
62 "climbing.services"));
63 auto decision = conditions->evaluate(subject, prepared.value().action.requiredConditionTags, tick);
64 if (!decision) return eve::Result<ClimbingServiceStart>::failure(decision.status());
65 if (decision.value() != ClimbingConditionDecision::Allowed)
66 return eve::Result<ClimbingServiceStart>::failure(eve::Diagnostic::error(
67 eve::DiagnosticCode::PreconditionViolation, "climbing action conditions were denied", "conditions", {},
68 "climbing.services"));
69 prepared.value().conditionsSatisfied = true;
70 }
71
72 IClimbingStaminaAuthority* stamina = nullptr;
73 ClimbingStaminaReservation reservation = ClimbingStaminaReservation::zero();
74 ClimbingOptionalServiceState staminaState = ClimbingOptionalServiceState::Disabled;
75 if (prepared.value().action.staminaCost > 0.f &&
76 runtime.profile_.staminaPolicy != ClimbingStaminaPolicy::Disabled) {
77 stamina = eve::cap::query<IClimbingStaminaAuthority>();
78 if (!stamina && runtime.profile_.staminaPolicy == ClimbingStaminaPolicy::RequireProvider)
79 return eve::Result<ClimbingServiceStart>::failure(
80 eve::Diagnostic::error(eve::DiagnosticCode::Unsupported, "required climbing stamina provider is absent",
81 "stamina", {}, "climbing.services"));
82 if (!stamina) {
83 staminaState = ClimbingOptionalServiceState::ProviderAbsent;
84 } else {
85 if (!runtime.profile_.staminaAdapter.empty() &&
86 stamina->adapterId() != runtime.profile_.staminaAdapter)
87 return eve::Result<ClimbingServiceStart>::failure(eve::Diagnostic::error(
88 eve::DiagnosticCode::Unsupported, "configured climbing stamina adapter is not available",
89 "staminaAdapter", {}, "climbing.services"));
90 auto reserved = stamina->prepare(subject, prepared.value().action.staminaCost, tick);
91 if (!reserved) return eve::Result<ClimbingServiceStart>::failure(reserved.status());
92 reservation = reserved.value();
93 if (reservation.isZero())
94 return eve::Result<ClimbingServiceStart>::failure(eve::Diagnostic::error(
95 eve::DiagnosticCode::InvariantViolation, "stamina provider returned a zero reservation", "stamina",
96 {}, "climbing.services"));
97 staminaState = ClimbingOptionalServiceState::Applied;
98 }
99 }
100
101 const ClimbingIntent originalIntent = intent;
102 auto consumed = ClimbingInputSystem::consume(intent, command, tick, prepared.value().executionId);
103 if (!consumed || !consumed.value()) {
104 intent = originalIntent;
105 if (stamina) stamina->cancelPrepared(reservation);
106 if (!consumed) return eve::Result<ClimbingServiceStart>::failure(consumed.status());
107 return eve::Result<ClimbingServiceStart>::failure(
108 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "eligible climbing input changed before commit",
109 "intent.commands", {}, "climbing.services"));
110 }
111 auto committed = runtime.commitBegin(std::move(prepared).takeValue());
112 if (!committed) {
113 intent = originalIntent;
114 if (stamina) stamina->cancelPrepared(reservation);
115 return eve::Result<ClimbingServiceStart>::failure(committed.status());
116 }
117 if (stamina) stamina->commitPrepared(reservation, committed.value().executionId);
118 return eve::Result<ClimbingServiceStart>::success({std::move(committed).takeValue(), staminaState},
119 eve::Status::success(eve::StatusCode::Applied));
120}

198EVENGINE_API_DOMAINS void declareScriptClimbingEventSeam() {
199 ScriptClimbingEventProviders& owned = scriptClimbingEventProviders();
200 eve::ScriptProviderRegistry::instance().declareCapability(
201 IClimbingEventSink::capabilityName,
202 [&owned](const eve::ScriptProviderHandle& handle) -> eve::Result<void> {
203 // 单实现槽位：`provide` 会顶掉当前实现，顶掉必须可见（否则原生 sink 会"悄悄失效"，
204 // 而症状只是事件不再被派发）。
205 if (eve::cap::query<IClimbingEventSink>() != nullptr) {
206 eve::recordLogEvent("warn", "script climbing event provider '" + handle.name +
207 "' replaced the provider that owned IClimbingEventSink");
208 }
209 auto adapter = std::make_unique<ScriptClimbingEventSink>(handle);
210 ScriptClimbingEventSink* raw = adapter.get();
211 eve::cap::provide<IClimbingEventSink>(raw);
212 owned.live.push_back(std::move(adapter));
213 return eve::Result<void>::success();
214 },
215 [&owned](const eve::ScriptProviderHandle& handle) -> eve::Result<void> {
216 const auto match = std::find_if(owned.live.begin(), owned.live.end(),
217 [&handle](const std::unique_ptr<ScriptClimbingEventSink>& adapter) {
218 return adapter->handle() == handle;
219 });
220 if (match == owned.live.end()) {
221 // 幂等：后置条件是"这一代没有适配器"，模块退出时可能已经撤过了。
222 return eve::Result<void>::success();
223 }
224 eve::cap::revoke<IClimbingEventSink>(match->get());
225 owned.retired.push_back(std::move(*match));
226 owned.live.erase(match);
227 return eve::Result<void>::success();
228 });
229}

231EVENGINE_API_DOMAINS void withdrawScriptClimbingEventSeam() {
232 ScriptClimbingEventProviders& owned = scriptClimbingEventProviders();
233 for (auto& adapter : owned.live) {
234 eve::cap::revoke<IClimbingEventSink>(adapter.get());
235 owned.retired.push_back(std::move(adapter));
236 }
237 owned.live.clear();
238 eve::ScriptProviderRegistry::instance()
239 .removeCapability(IClimbingEventSink::capabilityName)
240 .ignore("module teardown may already have dropped this seam");
241}
