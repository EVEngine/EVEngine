55Result<void> validateSkeleton(const ActionPreviewSkeleton& skeleton, std::string path) {
56 if (skeleton.bones.empty()) return invalid("skeleton has no bones", std::move(path));
57 for (std::size_t index = 0; index < skeleton.bones.size(); ++index) {
58 const auto& bone = skeleton.bones[index];
59 if (bone.name.empty()) return invalid("skeleton bone name is empty", path + ".bones[" + std::to_string(index) + "]");
60 if (bone.parent < -1 || bone.parent >= static_cast<int>(index))
61 return invalid("skeleton parent must precede its child", path + ".bones[" + std::to_string(index) + "].parent");
62 }
63 return Result<void>::success();
64}

66ActionPreviewCompatibility compareSkeletons(const ActionPreviewSkeleton& target,
67 const ActionPreviewSkeleton& candidate) {
68 if (target.bones.size() != candidate.bones.size()) return ActionPreviewCompatibility::Incompatible;
69 bool exact = true;
70 std::map<std::string, std::string> targetParents;
71 for (std::size_t index = 0; index < target.bones.size(); ++index) {
72 const auto& bone = target.bones[index];
73 const std::string parent = bone.parent < 0 ? std::string{} : normalizedBone(target.bones[bone.parent].name);
74 targetParents.emplace(normalizedBone(bone.name), parent);
75 }
76 for (std::size_t index = 0; index < candidate.bones.size(); ++index) {
77 const auto& bone = candidate.bones[index];
78 const std::string normalized = normalizedBone(bone.name);
79 const std::string parent = bone.parent < 0 ? std::string{} : normalizedBone(candidate.bones[bone.parent].name);
80 const auto found = targetParents.find(normalized);
81 if (found == targetParents.end() || found->second != parent) return ActionPreviewCompatibility::Incompatible;
82 exact = exact && target.bones[index].name == bone.name && target.bones[index].parent == bone.parent;
83 }
84 return exact ? ActionPreviewCompatibility::Compatible : ActionPreviewCompatibility::Retargetable;
85}

188Result<ActionPreviewCompatibilityReport> ActionPreviewCompatibilityAnalyzer::analyze(
189 const action::ActionTimeline& timeline, const ActionPreviewProfile& profile,
190 IActionPreviewAssetProvider* provider) const {
191 auto timelineValid = timeline.validate();
192 if (!timelineValid) return Result<ActionPreviewCompatibilityReport>::failure(timelineValid.status());
193 auto profileValid = profile.validate();
194 if (!profileValid) return Result<ActionPreviewCompatibilityReport>::failure(profileValid.status());
195
196 ActionPreviewCompatibilityReport report;
197 if (!provider) {
198 report.profileDiagnostics.push_back(Diagnostic::error(
199 DiagnosticCode::Unsupported, "animation/model preview asset provider is not available", "provider"));
200 for (const auto& section : timeline.animationSections)
201 report.sections.push_back({section.id, ActionPreviewCompatibility::Unsupported,
202 {Diagnostic::error(DiagnosticCode::Unsupported,
203 "skeleton compatibility was not evaluated", "provider")}});
204 return Result<ActionPreviewCompatibilityReport>::success(std::move(report));
205 }
206
207 if (profile.previewModelUri.empty() && timeline.animationSections.empty()) {
208 report.profileDiagnostics.push_back(Diagnostic::error(
209 DiagnosticCode::NotFound, "preview requires a model or at least one animation section", "previewModelUri"));
210 return Result<ActionPreviewCompatibilityReport>::success(std::move(report));
211 }
212 Result<ActionPreviewSkeleton> target = profile.previewModelUri.empty()
213 ? provider->animationSkeleton(timeline.animationSections.front().animationUri)
214 : provider->modelSkeleton(profile.previewModelUri);
215 if (!target) return Result<ActionPreviewCompatibilityReport>::failure(target.status());
216 auto targetValid = validateSkeleton(target.value(), "targetSkeleton");
217 if (!targetValid) return Result<ActionPreviewCompatibilityReport>::failure(targetValid.status());
218 if (!profile.boneMaskUri.empty()) {
219 auto mask = provider->validateBoneMask(profile.boneMaskUri, target.value());
220 if (!mask) {
221 if (!mask.status().diagnostics().empty())
222 report.profileDiagnostics.push_back(mask.status().diagnostics().front());
223 else
224 report.profileDiagnostics.push_back(Diagnostic::error(
225 DiagnosticCode::Conflict, "preview bone mask is incompatible with the target skeleton",
226 "boneMaskUri"));
227 }
228 }
229
230 for (const auto& section : timeline.animationSections) {
231 ActionPreviewSectionDiagnostic projected;
232 projected.sectionId = section.id;
233 auto skeleton = provider->animationSkeleton(section.animationUri);
234 if (!skeleton) {
235 projected.compatibility = ActionPreviewCompatibility::Incompatible;
236 projected.diagnostics = skeleton.status().diagnostics();
237 report.sections.push_back(std::move(projected));
238 continue;
239 }
240 auto valid = validateSkeleton(skeleton.value(), "animationSkeleton");
241 if (!valid) {
242 projected.compatibility = ActionPreviewCompatibility::Incompatible;
243 projected.diagnostics = valid.status().diagnostics();
244 report.sections.push_back(std::move(projected));
245 continue;
246 }
247 projected.compatibility = compareSkeletons(target.value(), skeleton.value());
248 if (projected.compatibility == ActionPreviewCompatibility::Retargetable)
249 projected.diagnostics.push_back(Diagnostic::warning(
250 DiagnosticCode::Conflict, "animation skeleton requires normalized-name retargeting", "animationUri"));
251 else if (projected.compatibility == ActionPreviewCompatibility::Incompatible)
252 projected.diagnostics.push_back(Diagnostic::error(
253 DiagnosticCode::Conflict, "animation skeleton is incompatible with the preview target", "animationUri"));
254 report.sections.push_back(std::move(projected));
255 }
256 return Result<ActionPreviewCompatibilityReport>::success(std::move(report));
257}

270Result<void> ActionPreviewProfileStore::save(const ActionPreviewProfile& profile) const {
271 auto encoded = profile.toValue();
272 if (!encoded) return Result<void>::failure(encoded.status());
273 auto json = encoded.value().toJson();
274 if (!json) return Result<void>::failure(json.status());
275
276 const auto destination = pathFor(profile.actionAsset);
277 const auto temporary = std::filesystem::path(destination.string() + ".tmp");
278 std::error_code error;
279 std::filesystem::create_directories(destination.parent_path(), error);
280 if (error)
281 return Result<void>::failure(Diagnostic::error(
282 DiagnosticCode::Failed, "could not create action preview sidecar directory", destination.string()));
283 {
284 std::ofstream output(temporary, std::ios::binary | std::ios::trunc);
285 if (!output)
286 return Result<void>::failure(Diagnostic::error(
287 DiagnosticCode::Failed, "could not open action preview sidecar temporary file", temporary.string()));
288 output << json.value();
289 output.flush();
290 if (!output)
291 return Result<void>::failure(Diagnostic::error(
292 DiagnosticCode::Failed, "could not write action preview sidecar", temporary.string()));
293 }
294 std::filesystem::rename(temporary, destination, error);
295 if (error) {
296 std::error_code removeError;
297 std::filesystem::remove(destination, removeError);
298 error.clear();
299 std::filesystem::rename(temporary, destination, error);
300 }
301 if (error)
302 return Result<void>::failure(Diagnostic::error(
303 DiagnosticCode::Failed, "could not atomically replace action preview sidecar", destination.string()));
304 return Result<void>::success(Status::success(StatusCode::Applied));
305}

307Result<ActionPreviewProfile> ActionPreviewProfileStore::load(const AssetGuid& actionAsset) const {
308 const auto source = pathFor(actionAsset);
309 std::ifstream input(source, std::ios::binary);
310 if (!input)
311 return Result<ActionPreviewProfile>::failure(
312 Diagnostic::error(DiagnosticCode::NotFound, "action preview sidecar was not found", source.string()));
313 std::string json((std::istreambuf_iterator<char>(input)), std::istreambuf_iterator<char>());
314 if (!input.good() && !input.eof())
315 return Result<ActionPreviewProfile>::failure(
316 Diagnostic::error(DiagnosticCode::Failed, "could not read action preview sidecar", source.string()));
317 auto value = Value::fromJson(json);
318 if (!value) return Result<ActionPreviewProfile>::failure(value.status());
319 auto profile = ActionPreviewProfile::fromValue(value.value());
320 if (!profile) return profile;
321 if (profile.value().actionAsset != actionAsset)
322 return invalid<ActionPreviewProfile>("preview sidecar asset identity does not match its filename",
323 "actionAssetGuid");
324 return profile;
325}
