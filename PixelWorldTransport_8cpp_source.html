49eve::Result<std::vector<PixelChunkTransferPart>> ReliablePixelChunkSender::capture(
50 const eve::pixelworld::PixelWorld& source, eve::pixelworld::PixelChunkRegion interest) {
51 if (inFlight_.size() >= config_.maximumInFlightTransfers)
52 return fail<std::vector<PixelChunkTransferPart>>(
53 eve::DiagnosticCode::PreconditionViolation, "reliable send window is full",
54 "maximumInFlightTransfers");
55 if (nextTransferId_ == std::numeric_limits<std::uint64_t>::max())
56 return fail<std::vector<PixelChunkTransferPart>>(
57 eve::DiagnosticCode::PreconditionViolation, "transfer id space is exhausted",
58 "transferId");
59 auto candidateCursor = cursor_;
60 auto captured = candidateCursor.capture(source, interest);
61 if (!captured.ok())
62 return eve::Result<std::vector<PixelChunkTransferPart>>::failure(captured.status());
63 auto update = std::move(captured).takeValue();
64 if (captured_ && !update.fullResync && update.batch.chunks.empty() &&
65 update.batch.sourceRevision == lastEnqueuedRevision_) {
66 cursor_ = std::move(candidateCursor);
67 return eve::Result<std::vector<PixelChunkTransferPart>>::success({});
68 }
69
70 const std::size_t chunkCount = update.batch.chunks.size();
71 const std::size_t partCount = std::max<std::size_t>(
72 1, (chunkCount + config_.maximumChunksPerPart - 1) / config_.maximumChunksPerPart);
73 if (partCount > config_.maximumPartsPerTransfer)
74 return fail<std::vector<PixelChunkTransferPart>>(
75 eve::DiagnosticCode::PreconditionViolation, "captured update exceeds part budget",
76 "maximumPartsPerTransfer");
77 std::vector<PixelChunkTransferPart> parts;
78 parts.reserve(partCount);
79 for (std::size_t index = 0; index < partCount; ++index) {
80 PixelChunkTransferPart part;
81 part.streamId = streamId_;
82 part.transferId = nextTransferId_;
83 part.partIndex = std::uint32_t(index);
84 part.partCount = std::uint32_t(partCount);
85 part.interest = interest;
86 part.batch.catalogFingerprint = update.batch.catalogFingerprint;
87 part.batch.sourceSeed = update.batch.sourceSeed;
88 part.batch.sourceRevision = update.batch.sourceRevision;
89 part.batch.sourceTick = update.batch.sourceTick;
90 part.batch.sourceLastEditSequence = update.batch.sourceLastEditSequence;
91 part.batch.fullResync = update.batch.fullResync;
92 const std::size_t begin = index * config_.maximumChunksPerPart;
93 const std::size_t end = std::min(chunkCount, begin + config_.maximumChunksPerPart);
94 if (begin < end)
95 part.batch.chunks.assign(update.batch.chunks.begin() + std::ptrdiff_t(begin),
96 update.batch.chunks.begin() + std::ptrdiff_t(end));
97 parts.push_back(std::move(part));
98 }
99 inFlight_.emplace(nextTransferId_, parts);
100 cursor_ = std::move(candidateCursor);
101 ++nextTransferId_;
102 captured_ = true;
103 lastEnqueuedRevision_ = update.batch.sourceRevision;
104 return eve::Result<std::vector<PixelChunkTransferPart>>::success(std::move(parts));
105}

116eve::Result<PixelChunkAckReceipt> ReliablePixelChunkSender::acknowledge(
117 PixelChunkTransferAck ack) {
118 if (ack.streamId != streamId_)
119 return fail<PixelChunkAckReceipt>(eve::DiagnosticCode::Conflict,
120 "ACK belongs to another stream", "streamId");
121 if (ack.acknowledgedThrough <= acknowledgedThrough_)
122 return eve::Result<PixelChunkAckReceipt>::success(
123 {0, 0, acknowledgedThrough_});
124 if (ack.acknowledgedThrough >= nextTransferId_)
125 return fail<PixelChunkAckReceipt>(eve::DiagnosticCode::Conflict,
126 "ACK advances beyond the sent window",
127 "acknowledgedThrough");
128 const auto acknowledged = inFlight_.find(ack.acknowledgedThrough);
129 if (acknowledged == inFlight_.end() || acknowledged->second.empty() ||
130 acknowledged->second.front().batch.sourceRevision != ack.appliedRevision)
131 return fail<PixelChunkAckReceipt>(eve::DiagnosticCode::Conflict,
132 "ACK revision does not match the transfer",
133 "appliedRevision");
134 PixelChunkAckReceipt receipt;
135 for (auto iterator = inFlight_.begin(); iterator != inFlight_.end() &&
136 iterator->first <= ack.acknowledgedThrough;) {
137 for (const auto& part : iterator->second)
138 receipt.tombstonesReleased += std::uint32_t(std::count_if(
139 part.batch.chunks.begin(), part.batch.chunks.end(),
140 [](const auto& chunk) { return chunk.removed; }));
141 iterator = inFlight_.erase(iterator);
142 ++receipt.transfersReleased;
143 }
144 acknowledgedThrough_ = ack.acknowledgedThrough;
145 receipt.acknowledgedThrough = acknowledgedThrough_;
146 return eve::Result<PixelChunkAckReceipt>::success(receipt);
147}

177eve::Result<PixelChunkReceiveReceipt> ReliablePixelChunkReceiver::receive(
178 PixelChunkTransferPart part, eve::pixelworld::PixelWorld& replica) {
179 if (part.streamId != streamId_ || part.transferId == 0 || part.partCount == 0 ||
180 part.partCount > config_.maximumPartsPerTransfer || part.partIndex >= part.partCount ||
181 part.batch.chunks.size() > config_.maximumChunksPerPart)
182 return fail<PixelChunkReceiveReceipt>(eve::DiagnosticCode::InvalidArgument,
183 "transfer part metadata exceeds policy", "part");
184 PixelChunkReceiveReceipt receipt;
185 if (part.transferId < expectedTransferId_) {
186 receipt.duplicate = true;
187 receipt.ack = {streamId_, expectedTransferId_ - 1, replica.revision()};
188 receipt.partsBuffered = std::uint32_t(bufferedPartCount());
189 return eve::Result<PixelChunkReceiveReceipt>::success(receipt);
190 }
191 auto found = buffered_.find(part.transferId);
192 if (found == buffered_.end()) {
193 if (buffered_.size() >= config_.maximumBufferedTransfers)
194 return fail<PixelChunkReceiveReceipt>(
195 eve::DiagnosticCode::PreconditionViolation,
196 "out-of-order receive window is full", "maximumBufferedTransfers");
197 Assembly assembly;
198 assembly.partCount = part.partCount;
199 assembly.parts.resize(part.partCount);
200 found = buffered_.emplace(part.transferId, std::move(assembly)).first;
201 }
202 Assembly& assembly = found->second;
203 if (assembly.partCount != part.partCount)
204 return fail<PixelChunkReceiveReceipt>(eve::DiagnosticCode::Conflict,
205 "transfer part count changed", "partCount");
206 auto& slot = assembly.parts[part.partIndex];
207 if (slot) {
208 if (*slot != part)
209 return fail<PixelChunkReceiveReceipt>(eve::DiagnosticCode::Conflict,
210 "duplicate part payload conflicts", "part");
211 receipt.duplicate = true;
212 } else {
213 slot = std::make_unique<PixelChunkTransferPart>(std::move(part));
214 }
215
216 while (true) {
217 auto next = buffered_.find(expectedTransferId_);
218 if (next == buffered_.end()) break;
219 const bool complete = std::all_of(next->second.parts.begin(), next->second.parts.end(),
220 [](const auto& value) { return bool(value); });
221 if (!complete) break;
222 const PixelChunkTransferPart& first = *next->second.parts.front();
223 eve::pixelworld::PixelChunkBatch combined = first.batch;
224 combined.chunks.clear();
225 for (const auto& value : next->second.parts) {
226 if (!sameMetadata(first, *value))
227 return fail<PixelChunkReceiveReceipt>(eve::DiagnosticCode::Conflict,
228 "transfer parts disagree on authority metadata",
229 "part.batch");
230 combined.chunks.insert(combined.chunks.end(), value->batch.chunks.begin(),
231 value->batch.chunks.end());
232 }
233 auto applied = replica.applyChunkBatch(combined, replica.revision());
234 if (!applied.ok()) {
235 buffered_.erase(next);
236 return eve::Result<PixelChunkReceiveReceipt>::failure(applied.status());
237 }
238 buffered_.erase(next);
239 ++expectedTransferId_;
240 ++receipt.transfersApplied;
241 }
242 receipt.ack = {streamId_, expectedTransferId_ - 1, replica.revision()};
243 receipt.partsBuffered = std::uint32_t(bufferedPartCount());
244 return eve::Result<PixelChunkReceiveReceipt>::success(receipt);
245}
