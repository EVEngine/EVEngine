44void AnimImporter::collectBones(const aiNode *node, int parent, AnimSkeleton *skeleton,
45 std::vector<const aiNode *> &order) {
46 if (!node) return;
47 const std::string name = node->mName.C_Str();
48 const std::string boneName = name.empty() ? ("node_" + std::to_string(order.size())) : name;
49 const int id = skeleton->addBone(boneName, parent);
50 order.push_back(node);
51
52 float px, py, pz, qx, qy, qz, qw, sx, sy, sz;
53 decomposeBind(node->mTransformation, px, py, pz, qx, qy, qz, qw, sx, sy, sz);
54 skeleton->setBindPosition(id, px, py, pz);
55 skeleton->setBindRotation(id, qx, qy, qz, qw);
56 skeleton->setBindScale(id, sx, sy, sz);
57
58 for (unsigned i = 0; i < node->mNumChildren; ++i) {
59 collectBones(node->mChildren[i], id, skeleton, order);
60 }

77AnimClip *AnimImporter::loadClip(const aiScene *scene, const AnimSkeleton *skeleton,
78 int animIndex) {
79 if (!scene || !skeleton) {
80 throw Exception("AnimImporter.loadClip: null scene/skeleton");
81 }
82 if (animIndex < 0 || animIndex >= getAnimationCount(scene)) {
83 throw Exception("AnimImporter.loadClip: invalid anim index %d", animIndex);
84 }
85 const aiAnimation *anim = scene->mAnimations[animIndex];
86 auto *clip = new AnimClip(anim->mName.length ? anim->mName.C_Str() : "anim");
87 clip->setDuration(ticksToSeconds(anim, anim->mDuration));
88 clip->setLoop(true);
89 clip->setSampleRate(30.f);
90
91 for (unsigned c = 0; c < anim->mNumChannels; ++c) {
92 const aiNodeAnim *channel = anim->mChannels[c];
93 if (!channel) continue;
94 const int bone = skeleton->findBone(channel->mNodeName.C_Str());
95 if (bone < 0) continue;
96
97 for (unsigned i = 0; i < channel->mNumPositionKeys; ++i) {
98 const aiVectorKey &k = channel->mPositionKeys[i];
99 clip->addPositionKey(bone, ticksToSeconds(anim, k.mTime), k.mValue.x, k.mValue.y,
100 k.mValue.z);
101 }
102 for (unsigned i = 0; i < channel->mNumRotationKeys; ++i) {
103 const aiQuatKey &k = channel->mRotationKeys[i];
104 clip->addRotationKey(bone, ticksToSeconds(anim, k.mTime), k.mValue.x, k.mValue.y,
105 k.mValue.z, k.mValue.w);
106 }
107 for (unsigned i = 0; i < channel->mNumScalingKeys; ++i) {
108 const aiVectorKey &k = channel->mScalingKeys[i];
109 clip->addScaleKey(bone, ticksToSeconds(anim, k.mTime), k.mValue.x, k.mValue.y,
110 k.mValue.z);
111 }
112 }
