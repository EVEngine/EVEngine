43Result<EditorValue> EditorMcpProjector::auditCompleteness() const {
44 EditorValue::Array incomplete;
45 eve::cap::forEach<editing::IEditorPackageProvider>([&](editing::IEditorPackageProvider* provider) {
46 if (!provider) return;
47 const editing::EditorPackage& package = provider->package();
48 if (!package.incomplete) return;
49 EditorValue::Object item;
50 item["owner"] = package.ownerModule;
51 item["reason"] = "package.incomplete";
52 incomplete.emplace_back(std::move(item));
53 });
54
55 std::set<std::string> packagedTypes;
56 eve::cap::forEach<editing::IEditorPackageProvider>([&](editing::IEditorPackageProvider* provider) {
57 if (!provider) return;
58 const editing::EditorPackage& package = provider->package();
59 if (package.incomplete) return;
60 for (const editing::EditorTargetTypeSpec& target : package.targetTypes) packagedTypes.insert(target.type);
61 });
62
63 EditorValue::Array orphans;
64 eve::cap::forEach<IEditorAutomationTargetFactory>([&](IEditorAutomationTargetFactory* factory) {
65 if (!factory) return;
66 for (const std::string_view type : factory->types()) {
67 if (packagedTypes.contains(std::string(type))) continue;
68 orphans.emplace_back(std::string(type));
69 }
70 });
71 std::sort(orphans.begin(), orphans.end(), [](const EditorValue& left, const EditorValue& right) {
72 const auto* leftString = left.getIf<std::string>();
73 const auto* rightString = right.getIf<std::string>();
74 return (leftString ? *leftString : std::string{}) < (rightString ? *rightString : std::string{});
75 });
76
77 EditorValue::Object report;
78 report["incompletePackages"] = std::move(incomplete);
79 report["orphanFactoryTypes"] = std::move(orphans);
80 report["mode"] = "commands";
81 report["attached"] = isAttached();
82 if (host_) report["hostId"] = host_->workspace().getId();
83 return eve::editing::applied<EditorValue>(EditorValue(std::move(report)));
84}
