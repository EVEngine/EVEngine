24inline Result<std::vector<SubjectRef>> parseScriptSubjects(const std::vector<std::string>& texts) {
25 std::vector<SubjectRef> subjects;
26 subjects.reserve(texts.size());
27 for (std::size_t index = 0; index < texts.size(); ++index) {
28 auto subject = parseScriptSubject(texts[index], "subjects[" + std::to_string(index) + "]");
29 if (!subject) return Result<std::vector<SubjectRef>>::failure(subject.status());
30 subjects.push_back(std::move(subject).takeValue());
31 }
32 return Result<std::vector<SubjectRef>>::success(std::move(subjects));
33}
