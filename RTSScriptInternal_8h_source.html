20inline Result<std::vector<SubjectRef>> parseScriptSubjects(const std::vector<std::string>& texts) {
21 std::vector<SubjectRef> subjects;
22 subjects.reserve(texts.size());
23 for (std::size_t index = 0; index < texts.size(); ++index) {
