921int Scene::forEachEntity(const std::string &hostName, const std::string &nodeId,
922 ssq::Object cb) {
923 if (!vm_) return 0;
924 SceneHost *h = resolveHost(hostName);
925 if (!h) return 0;
926 SceneNode *n = borrowSceneResult(h->findById(nodeId));
927 if (!n) return 0;
928 SceneObject *obj = n->objectId ? findSceneObjectById(n->objectId) : nullptr;
929 if (!obj) return 0;
930 // Copy + addref: the callback may attach/detach and mutate the live list.
931 auto vec = obj->scriptBindings()->instances;
932 for (auto &b : vec) sq_addref(vm_, &b);
933 HSQOBJECT fn = cb.getRaw();
934 sq_addref(vm_, &fn);
935 int count = 0;
936 for (size_t i = 0; i < vec.size(); ++i) {
937 if (callCallback(fn, vec[i], int(i))) ++count;
938 }
939 sq_release(vm_, &fn);
940 for (auto &b : vec) sq_release(vm_, &b);
941 return count;
942}

986void Scene::pruneOrphanObjects() {
987 if (ecs::current()->getManager<SceneObject>() == nullptr) return;
988 std::vector<SceneObject *> orphans;
989 {
990 auto view = ecs::View<SceneObject, SceneObject::Meta, SceneObject::ScriptBindings>();
991 for (auto it = view.begin(); it != view.end(); ++it) {
992 auto [meta, sb] = *it;
993 if (!meta->entity) continue;
994 SceneHost *h = findHostByName(meta->hostName);
995 if (!h) {
996 orphans.push_back(meta->entity);
997 continue;
998 }
999 SceneNode *n = borrowSceneResult(h->findById(meta->nodeId));
1000 if (!n || n->objectId != uint32_t(meta->entity->id)) {
1001 orphans.push_back(meta->entity);
1002 continue;
1003 }
1004 if (!n->persistentId.isNil()) meta->persistentId = n->persistentId;
1005 (void)sb;
1006 // Self-heal: node id / host renamed (reconcile patch), keep binding.
1007 if (meta->hostName != h->getName() || meta->nodeId != n->id) {
1008 meta->hostName = h->getName();
1009 meta->nodeId = n->id;
1010 syncBindingRefs(meta->entity);
1011 }
1012 }
1013 }
1014 for (SceneObject *o : orphans) teardownBindings(o);
1015}
