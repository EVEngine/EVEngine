99bool kitMatchesFilters(Poco::JSON::Object::Ptr kit, const std::string& genre, const std::string& module,
100 const std::string& tag, const std::string& q) {
101 if (!kit) return false;
102 if (!genre.empty() && kit->optValue<std::string>("genre", "") != genre) return false;
103 if (!module.empty()) {
104 if (!kit->has("modules") || !kit->isArray("modules") ||
105 !arrayContainsString(kit->getArray("modules"), module))
106 return false;
107 }
108 if (!tag.empty()) {
109 if (!kit->has("tags") || !kit->isArray("tags") || !arrayContainsString(kit->getArray("tags"), tag))
110 return false;
111 }
112 if (!q.empty()) {
113 const std::string qLower = toLower(q);
114 std::string blob = kit->optValue<std::string>("id", "") + " " +
115 kit->optValue<std::string>("title", "") + " " +
116 kit->optValue<std::string>("summary", "") + " " +
117 kit->optValue<std::string>("genre", "");
118 if (kit->has("tags") && kit->isArray("tags")) {
119 auto tags = kit->getArray("tags");
120 for (std::size_t i = 0; i < tags->size(); ++i) {
121 try {
122 blob += " ";
123 blob += tags->getElement<std::string>(static_cast<unsigned>(i));
124 } catch (...) {
125 }
126 }
127 }
128 if (toLower(blob).find(qLower) == std::string::npos) return false;
129 }
130 return true;
131}

133std::string fillTemplate(const std::string& templ, const std::unordered_map<std::string, std::string>& values,
134 std::vector<std::string>* unresolved) {
135 std::string out;
136 out.reserve(templ.size());
137 for (std::size_t i = 0; i < templ.size();) {
138 if (templ[i] != '{') {
139 out.push_back(templ[i++]);
140 continue;
141 }
142 const std::size_t end = templ.find('}', i + 1);
143 if (end == std::string::npos) {
144 out.push_back(templ[i++]);
145 continue;
146 }
147 const std::string key = templ.substr(i + 1, end - i - 1);
148 auto it = values.find(key);
149 if (it == values.end() || it->second.empty()) {
150 if (unresolved) unresolved->push_back(key);
151 out.append(templ, i, end - i + 1);
152 } else {
153 out += it->second;
154 }
155 i = end + 1;
156 }
157 return out;
158}

160std::string assetKitList(Poco::JSON::Object::Ptr args) {
161 auto catalog = parseCatalog();
162 if (!catalog)
163 return errorPayload("embedded asset-kit catalog failed to parse",
164 "Rebuild after running python3 scripts/embed_asset_kits.py");
165
166 const std::string genre = getArgString(args, "genre", "");
167 const std::string module = getArgString(args, "module", "");
168 const std::string tag = getArgString(args, "tag", "");
169 const std::string q = getArgString(args, "q", "");
170
171 Poco::JSON::Object::Ptr out = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
172 Poco::JSON::Array::Ptr kits = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
173 int count = 0;
174 if (catalog->has("kits") && catalog->isArray("kits")) {
175 auto arr = catalog->getArray("kits");
176 for (std::size_t i = 0; i < arr->size(); ++i) {
177 auto kit = arr->getObject(static_cast<unsigned>(i));
178 if (!kitMatchesFilters(kit, genre, module, tag, q)) continue;
179 kits->add(summarizeKit(kit));

181 }

185 out->set("count", count);
186 out->set("kits", kits);
187 out->set("hint",

189 "Unity/Godot sizes");
190 return textPayload(out);

231std::string assetKitResolve(Poco::JSON::Object::Ptr args) {
232 const std::string kitId = getArgString(args, "kit", getArgString(args, "id", ""));
233 const std::string slotId = getArgString(args, "slot", "");
234 if (kitId.empty() || slotId.empty())
235 return errorPayload("kit and slot are required",
236 "Pass {kit:\"card.playing\", slot:\"face\", vars:{...}}");
237
238 auto catalog = parseCatalog();
239 if (!catalog) return errorPayload("embedded asset-kit catalog failed to parse");
240
241 Poco::JSON::Object::Ptr kit;
242 std::vector<std::string> knownKits;
243 if (catalog->has("kits") && catalog->isArray("kits")) {
244 auto arr = catalog->getArray("kits");
245 for (std::size_t i = 0; i < arr->size(); ++i) {
246 auto candidate = arr->getObject(static_cast<unsigned>(i));
247 if (!candidate) continue;
248 const std::string kid = candidate->optValue<std::string>("id", "");
249 knownKits.push_back(kid);
250 if (kid == kitId) kit = candidate;
251 }
252 }
253 if (!kit) {
254 std::string knownList;
255 for (std::size_t i = 0; i < knownKits.size(); ++i) {
256 if (i) knownList += ", ";
257 knownList += knownKits[i];
258 }
259 return errorPayload("unknown kit id: " + kitId, "known ids: " + knownList);
260 }
261
262 Poco::JSON::Object::Ptr slot;
263 std::vector<std::string> knownSlots;
264 if (kit->has("slots") && kit->isArray("slots")) {
265 auto slots = kit->getArray("slots");
266 for (std::size_t i = 0; i < slots->size(); ++i) {
267 auto candidate = slots->getObject(static_cast<unsigned>(i));
268 if (!candidate) continue;
269 const std::string sid = candidate->optValue<std::string>("id", "");
270 knownSlots.push_back(sid);
271 if (sid == slotId) slot = candidate;
272 }
273 }
274 if (!slot) {
275 std::string knownList;
276 for (std::size_t i = 0; i < knownSlots.size(); ++i) {
277 if (i) knownList += ", ";
278 knownList += knownSlots[i];
279 }
280 return errorPayload("unknown slot id: " + slotId, "kit " + kitId + " slots: " + knownList);
281 }
282
283 std::unordered_map<std::string, std::string> values;
284 auto size = slot->has("pixelSize") && slot->isObject("pixelSize")
285 ? slot->getObject("pixelSize")
286 : Poco::JSON::Object::Ptr{};

288 values["width"] = std::to_string(size->optValue<int>("width", 0));
289 values["height"] = std::to_string(size->optValue<int>("height", 0));

295

297 auto vars = args->getObject("vars");
298 for (const auto& name : vars->getNames()) {
299 try {
300 values[name] = vars->getValue<std::string>(name);
301 } catch (...) {
302 try {
303 values[name] = std::to_string(vars->getValue<int>(name));
304 } catch (...) {

306 values[name] = std::to_string(vars->getValue<double>(name));

308 }
309 }
310 }
311 }
312 }

314 std::vector<std::string> unresolved;

320 unresolved.assign(uniq.begin(), uniq.end());
321 std::sort(unresolved.begin(), unresolved.end());
322
323 Poco::JSON::Object::Ptr varsOut = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
324 for (const auto& [key, value] : values) varsOut->set(key, value);

326 Poco::JSON::Array::Ptr unresolvedArr = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
327 for (const auto& key : unresolved) unresolvedArr->add(key);
328

331 auto arr = kit->getArray("bindings");

333 }

335 auto arr = slot->getArray("bindings");
336 for (std::size_t i = 0; i < arr->size(); ++i) bindings->add(arr->get(static_cast<unsigned>(i)));
337 }

340 job->set("kitId", kitId);
341 job->set("slotId", slotId);
342 job->set("role", slot->optValue<std::string>("role", ""));
343 job->set("required", slot->optValue<bool>("required", false));
344 if (size) job->set("pixelSize", size);
345 if (slot->has("displaySize")) job->set("displaySize", slot->get("displaySize"));
346 job->set("aspectRatio", slot->optValue<std::string>("aspectRatio", ""));

348 job->set("colorType", slot->optValue<std::string>("colorType", ""));
349 job->set("alpha", slot->optValue<std::string>("alpha", ""));

353 job->set("prompt", prompt);

355 if (slot->has("constraints")) job->set("constraints", slot->get("constraints"));
356 if (slot->has("consumer")) job->set("consumer", slot->get("consumer"));
357 job->set("bindings", bindings);
358 job->set("vars", varsOut);
359 job->set("unresolved", unresolvedArr);
360 job->set("source", "engine");
361
362 Poco::JSON::Object::Ptr out = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
363 stampApi(out, catalog);
364 out->set("ok", true);
365 out->set("job", job);
366 out->set("hint", unresolved.empty()
367 ? "Generate at pixelSize; write job.path; bind via consumer/bindings"
368 : "Fill unresolved placeholders then generate at pixelSize");
369 return textPayload(out);
370}
