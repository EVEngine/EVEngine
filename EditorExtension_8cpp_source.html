8EditorResult<void> EditorExtensionRegistry::load(IGameEditorExtension& extension) {
9 if (!commands_)
10 return invalid("editor.extension.missing-command-service", "Extension registry has no command service");
11 const std::string owner = extension.ownerModule();
12 if (owner.empty()) return invalid("editor.extension.missing-owner", "Game editor extension owner is required");
13 if (std::any_of(commandVisibility_.begin(), commandVisibility_.end(),
14 [&](const CommandVisibility& command) { return command.ownerModule == owner; }) ||
15 std::any_of(tools_.begin(), tools_.end(),
16 [&](const ExtensionToolDescriptor& tool) { return tool.ownerModule == owner; }))
17 return invalid("editor.extension.duplicate-owner", "Game editor extension owner is already loaded");
18 activeOwner_ = owner;
19 try {
20 extension.registerEditor(*this);
21 } catch (const std::exception& exception) {
22 unload(owner);
23 activeOwner_.clear();
24 return invalid("editor.extension.exception", exception.what());
25 } catch (...) {
26 unload(owner);
27 activeOwner_.clear();
28 return invalid("editor.extension.exception", "Game editor extension threw an unknown exception");
29 }
30 activeOwner_.clear();
31 return eve::editing::applied<void>();
32}

34std::size_t EditorExtensionRegistry::unload(const std::string& ownerModule) {
35 std::size_t removed = commands_ ? commands_->unregisterOwner(ownerModule) : 0;
36 auto eraseOwner = [&](auto& values) {
37 const std::size_t before = values.size();
38 std::erase_if(values, [&](const auto& value) { return value.ownerModule == ownerModule; });
39 removed += before - values.size();
40 };
41 eraseOwner(commandVisibility_);
42 eraseOwner(tools_);
43 eraseOwner(palettes_);
44 eraseOwner(rules_);
45 return removed;
46}

78EditorResult<EditorValue> EditorExtensionRegistry::registerCommand(CommandDescriptor descriptor,

91EditorResult<void> EditorExtensionRegistry::registerTool(ExtensionToolDescriptor descriptor) {
92 if (activeOwner_.empty() || descriptor.id.empty())

94 if (std::any_of(tools_.begin(), tools_.end(),

96 return invalid("editor.extension.duplicate-tool", "Tool id is already registered");
97 descriptor.ownerModule = activeOwner_;
