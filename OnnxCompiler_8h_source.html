14// Run, buffers or callbacks. Joining the workers needs no device access.

19class CompilerQueue {
20 using Job = std::packaged_task<CompiledProgram()>;
21 std::mutex mutex;
22 std::condition_variable ready;
23 std::deque<Job> jobs;
24 bool stopping = false;
25 size_t inFlight = 0;

28 void stop() noexcept {
29 {
30 std::lock_guard lock(mutex);
31 stopping = true;
32 }
33 ready.notify_all();
34 for (auto& worker : workers)
35 if (worker.joinable()) worker.join();
36 }

38public:
39 explicit CompilerQueue(uint32_t count) {
40 try {
41 for (uint32_t i = 0; i < count; ++i)
42 workers.emplace_back([this] {
43 for (;;) {
44 Job job;
45 {
46 std::unique_lock lock(mutex);
47 ready.wait(lock, [this] { return stopping || !jobs.empty(); });
48 if (jobs.empty()) return;
49 job = std::move(jobs.front());
50 jobs.pop_front();
51 ++inFlight;
52 }
53 const auto n = ++running;
54 auto p = peak.load();
55 while (p < n && !peak.compare_exchange_weak(p, n)) {
56 }
57 job(); // packaged_task transports failures to the owning device thread.
58 --running;
59 {
60 std::lock_guard lock(mutex);
61 --inFlight;
62 }
63 ready.notify_all();
64 }
65 });
66 } catch (...) {

69 }
70 }

74 std::future<CompiledProgram> enqueue(std::string source) {
75 Job job([source = std::move(source)] {
76 const auto start = std::chrono::steady_clock::now();
77 auto result = gpgpu::compileComputeSpirv(source);
78 if (!result.ok()) throw std::runtime_error(result.error()->message() + "\nCompute source:\n" + source);
79 return CompiledProgram{
80 std::move(result.value()),
81 std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - start).count()};
82 });
83 auto future = job.get_future();
84 {
85 std::lock_guard lock(mutex);
86 if (stopping || jobs.size() >= 48) throw std::runtime_error("ONNX compiler queue unavailable");
87 jobs.push_back(std::move(job));
88 }
89 ready.notify_one();
90 return future;
91 }
