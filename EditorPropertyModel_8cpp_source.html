28TransactionId newPropertyTransactionId() {
29 static std::atomic<std::uint64_t> entropySequence{1};
30 const eve::UuidV7Generator generator([](std::span<std::uint8_t> bytes) {
31 const std::uint64_t seed = entropySequence.fetch_add(1, std::memory_order_relaxed);
32 for (std::size_t index = 0; index < bytes.size(); ++index)
33 bytes[index] = static_cast<std::uint8_t>(seed >> ((index % sizeof(seed)) * 8u));
34 return true;
35 });
36 const auto generated = generator.generate(std::chrono::system_clock::now());
37 if (!generated) {
38 // A clock before the UUID epoch is not expected for a live editor, but
39 // retain a valid canonical UUID path if a platform clock is malformed.
40 const auto epoch = generator.generate(std::chrono::system_clock::time_point{});
41 if (!epoch) return TransactionId{};
42 using CanonicalUuid = typename TransactionId::Uuid;
43 return TransactionId::fromUuid(CanonicalUuid(epoch->bytes()));
44 }
45 using CanonicalUuid = typename TransactionId::Uuid;
46 return TransactionId::fromUuid(CanonicalUuid(generated->bytes()));
47}

143property_access::WriteResult EditorPropertyModel::write(const std::string& path, const eve::Value& value) {
144 if (!provider_) return property_access::WriteResult::reject("editor.property.provider", "No property provider");
145 const EditorResult<void> revisionCheck = ensureCurrentRevision();
146 if (!revisionCheck.ok())
147 return writeFailure(revisionCheck, "editor.property.revision", "Property revision is unavailable");
148 if (transactionBackend_ && transactionBackend_->active() && pendingPaths_.contains(path))
149 return property_access::WriteResult::reject("editor.property.duplicate-path",
150 "A property may be written only once per transaction");
151
152 const EditorValue editorValue = toEditorValue(value);
153 EditorResult<PropertyEditIntent> intent = [&]() {
154 if (surface_ == PropertyModelSurface::Runtime) {
155 RuntimePropertyPresenter presenter;
156 return presenter.editIntent(editorSchema_, selection_, PropertyPath(path), editorValue, profile_);
157 }
158 DeveloperPropertyPresenter presenter;
159 return presenter.editIntent(editorSchema_, selection_, PropertyPath(path), editorValue);
160 }();
161 if (!intent.ok()) {
162 return writeFailure(intent, "editor.property.intent", "Property edit was rejected");
163 }
164
165 if (transactionBackend_) {
166 EditorResult<DomainOperation> operation =
167 provider_->makeSet(selection_, PropertyPath(path), editorValue, PropertySetMode::Absolute);
168 if (!operation.ok())
169 return writeFailure(operation, "editor.property.operation", "Property operation was rejected");
170 if (operation.value().target.empty())
171 return property_access::WriteResult::reject("editor.property.target", "Property operation has no target");
172 if (operation.value().mergeKey.empty())
173 operation.value().mergeKey = intent.value().command.value() + ":" + path;
174
175 bool started = false;
176 if (!transactionBackend_->active()) {
177 TransactionSpec specification;
178 specification.id = newPropertyTransactionId();
179 specification.label = intent.value().command.value();
180 specification.target = operation.value().target;
181 specification.mergeKey = operation.value().mergeKey;
182 specification.baseRevision = targetRevision_.value();
183 if (specification.id.empty())
184 return property_access::WriteResult::reject("editor.property.transaction-id",
185 "Could not allocate a property transaction identity");
186 EditorResult<TransactionId> begun = transactionBackend_->begin(std::move(specification));
187 if (!begun.ok() || !begun.ok())
188 return writeFailure(begun, "editor.property.transaction.begin", "Could not begin property transaction");
189 started = true;
190 }
191
192 EditorResult<void> appended = transactionBackend_->append(std::move(operation.value()));
193 if (!appended.ok()) {
194 if (started) {
195 EditorResult<void> discarded = transactionBackend_->discard();
196 if (!discarded.ok())
197 return writeFailure(discarded, "editor.property.transaction.discard",
198 "Could not discard rejected property transaction");
199 }
200 return writeFailure(appended, "editor.property.transaction.append", "Could not stage property operation");
201 }
202 pendingPaths_.insert(path);
203
204 // An already-open transaction is intentionally left staged for the
205 // caller's explicit preview/commit boundary.
206 if (!started) return property_access::WriteResult::success();
207
208 EditorResult<EditorDryRunReport> previewed = transactionBackend_->preview();
209 if (!previewed.ok() || !previewed.ok()) {
210 EditorResult<void> discarded = transactionBackend_->discard();
211 pendingPaths_.clear();
212 if (!discarded.ok())
213 return writeFailure(discarded, "editor.property.transaction.discard",
214 "Could not discard failed property preview");
215 return writeFailure(previewed, "editor.property.transaction.preview",
216 "Property transaction preview was rejected");
217 }
218
219 EditorResult<TransactionReceipt> committed = transactionBackend_->commit();
220 if (!committed.ok() || !committed.ok())
221 // Commit failure is deliberately retained by the backend so the
222 // caller can inspect diagnostics, retry, or explicitly discard.
223 return writeFailure(committed, "editor.property.transaction.commit", "Property transaction commit failed");
224 targetRevision_ = eve::Revision(committed.value().afterRevision);
225 pendingPaths_.clear();
226 // The mutation already committed. Do not report a successful commit
227 // as a rejected write; force an explicit refresh if observation fails.
228 if (!refresh().ok()) bound_ = false;
229 return property_access::WriteResult::success();
230 }
231
232 if (!sink_) return property_access::WriteResult::reject("editor.property.sink", "No command sink is connected");
233 // The sink is a compatibility path and may not have an authority that can
234 // perform a commit-time CAS. Recheck immediately before handing it the
235 // intent; stale compatibility writes must fail closed as well.
236 const EditorResult<void> sinkRevisionCheck = ensureCurrentRevision();
237 if (!sinkRevisionCheck.ok())
238 return writeFailure(sinkRevisionCheck, "editor.property.revision", "Property revision is unavailable");
239 const EditorResult<void> applied = sink_(intent.value());
240 if (!applied.ok()) return writeFailure(applied, "editor.property.command", "Property command failed");
241 // A legacy sink may mutate the provider without returning a receipt. Read
242 // the authoritative post-command revision so the next write does not use
243 // the pre-command baseline. If observation fails, retain command success
244 // and require an explicit refresh before another write.
245 if (!refresh().ok()) bound_ = false;
246 return property_access::WriteResult::success();
247}

406EditorResult<void> EditorPropertyModel::refresh() {
407 if (!provider_)
408 return eve::editing::failed<void>(EditorStatus::Unsupported, RuleId("editor.property.provider"),
409 "No property provider is connected");
410 if (transactionBackend_ && transactionBackend_->active())
411 return eve::editing::failed<void>(EditorStatus::Conflict, RuleId("editor.property.active-transaction"),
412 "Cannot refresh or rebase while a property transaction is active");
413
414 const EditorResult<eve::Revision> first = readProviderRevision();
415 if (!first.ok()) return EditorResult<void>::failure(first.status());
416
417 std::map<std::string, CachedProperty> nextProperties;
418 for (const property_access::PropertyDescriptor& property : presentationSchema_.properties) {
419 const PropertyReadResult current = provider_->read(selection_, PropertyPath(property.path));
420 if (current.state == PropertyReadState::Error) {
421 if (!current.diagnostics.empty())
422 return EditorResult<void>::failure(eve::Status(EditorStatus::Failed, current.diagnostics));
423 return eve::editing::failed<void>(EditorStatus::Failed, RuleId("editor.property.read"),
424 "Property provider returned a read error");
425 }
426 CachedProperty cached;
427 switch (current.state) {
428 case PropertyReadState::Value:
429 cached.state = property_access::PropertyChangeState::Value;
430 cached.value = toPresentationValue(current.value);
431 break;
432 case PropertyReadState::Mixed: cached.state = property_access::PropertyChangeState::Mixed; break;
433 case PropertyReadState::Missing: cached.state = property_access::PropertyChangeState::Missing; break;
434 case PropertyReadState::Error: break;
435 }
436 nextProperties.emplace(property.path, std::move(cached));
437 }
438
439 const EditorResult<eve::Revision> last = readProviderRevision();
440 if (!last.ok()) return EditorResult<void>::failure(last.status());
441 if (first.value() != last.value()) return externalRevisionConflict(first.value(), last.value());
442
443 std::vector<property_access::PropertyChange> changes;
444 for (const auto& [path, property] : nextProperties) {
445 const auto found = cachedProperties_.find(path);
446 if (found == cachedProperties_.end() || found->second != property)
447 changes.push_back({path, property.value.value_or(eve::Value{}), 0, property.state});
448 }
449
450 if (!changes.empty()) {
451 const std::uint64_t snapshotRevision = ++revision_;
452 for (property_access::PropertyChange& change : changes) change.revision = snapshotRevision;
453 }
454 cachedProperties_ = std::move(nextProperties);
455 targetRevision_ = last.value();
456 bound_ = true;
457 dispatch(std::move(changes));
458 return eve::editing::applied<void>();
459}

461void EditorPropertyModel::dispatch(std::vector<property_access::PropertyChange> changes) {
462 if (changes.empty()) return;
463 notificationQueue_.push_back(std::move(changes));
464 if (dispatchingNotifications_) return;
465
466 dispatchingNotifications_ = true;
467 try {
468 while (!notificationQueue_.empty()) {
469 std::vector<property_access::PropertyChange> batch = std::move(notificationQueue_.front());
470 notificationQueue_.erase(notificationQueue_.begin());
471 for (const property_access::PropertyChange& change : batch) {
472 const auto observers = observers_->entries;
473 for (const ObserverState::Entry& entry : observers)
474 if (entry.callback) entry.callback(change);
475 }
476 }
477 dispatchingNotifications_ = false;
478 } catch (...) {
479 notificationQueue_.clear();
480 dispatchingNotifications_ = false;
481 throw;
482 }
483}
