41bool parsePoints(const std::string& text, Polygon& out) {
42 out.clear();
43 for (const std::string& pair : split(text, ';')) {
44 const auto xy = split(pair, ',');
45 if (xy.size() != 2) return false;
46 try {
47 out.push_back({std::stod(xy[0]), std::stod(xy[1])});
48 } catch (...) {
49 return false;
50 }
51 }
52 return out.size() >= 3;

55bool makePresetLand(const std::string& preset, double w, double h, Polygon& out) {
56 out.clear();
57 const std::string p = preset.empty() ? "rect" : preset;
58 if (p == "rect") {
59 out = {{0, 0}, {w, 0}, {w, h}, {0, h}};
60 } else if (p == "triangle") {
61 out = {{0, 0}, {w, 0}, {w * 0.5, h}};
62 } else if (p == "ellipse" || p == "circle") {
63 const int segs = 28;
64 const double rx = w * 0.5;
65 const double ry = h * 0.5;
66 for (int i = 0; i < segs; ++i) {
67 const double a = double(i) * 2.0 * kPi / double(segs);
68 out.push_back({w * 0.5 + rx * std::cos(a), h * 0.5 + ry * std::sin(a)});
69 }
70 } else if (p == "l") {
71 out = {{0, 0}, {w, 0}, {w, h * 0.55}, {w * 0.45, h * 0.55}, {w * 0.45, h}, {0, h}};
72 } else if (p == "hexagon") {
73 const double cx = w * 0.5;
74 const double cy = h * 0.5;
75 for (int i = 0; i < 6; ++i) {
76 const double a = -kPi * 0.5 + double(i) * kPi / 3.0;
77 out.push_back({cx + w * 0.5 * std::cos(a), cy + h * 0.5 * std::sin(a)});
78 }
79 } else {
80 return false;
81 }
82 return out.size() >= 3;
83}

99// Intersect inward edge half-planes. The result is an interior convex footprint;
100// parcels too narrow for the requested setback have no buildable footprint.
101Polygon buildingFootprint(const Polygon& parcel, double setback) {
102 if (setback <= 0.0) return parcel;
103 Polygon clipped = parcel;
104 for (size_t edge = 0; edge < parcel.size() && !clipped.empty(); ++edge) {
105 const Vec2 a = parcel[edge];
106 const Vec2 normal = perpendicular(normalize(parcel[(edge + 1) % parcel.size()] - a));
107 Polygon next;
108 Vec2 previous = clipped.back();
109 double previousDistance = dot(previous - a, normal) - setback;
110 for (const Vec2 current : clipped) {
111 const double currentDistance = dot(current - a, normal) - setback;
112 if ((currentDistance >= 0.0) != (previousDistance >= 0.0)) {
113 const double t = previousDistance / (previousDistance - currentDistance);
114 next.push_back(previous + (current - previous) * t);
115 }
116 if (currentDistance >= 0.0) next.push_back(current);
117 previous = current;
118 previousDistance = currentDistance;
119 }
120 clipped = std::move(next);
121 }
122 cleanupRing(clipped);
123 if (clipped.size() < 3 || area(clipped) < 1e-6) return {};
124 return clipped;
125}

192bool generateUrbanGrid(const Params& params, Grid2D& out, std::string& error) {
193 UrbanOptions opts;
194 if (!parseUrbanOptions(params, opts, error)) return false;
195 UrbanGenerator gen(opts);
196 if (!gen.generate(&error)) return false;
197 const UrbanLayout& layout = gen.layout();
198 if (layout.parcels.empty()) {
199 error = "urban.parcels: no parcels generated";
200 return false;
201 }
202
203 double minX = opts.land[0].x, minY = opts.land[0].y;
204 double maxX = opts.land[0].x, maxY = opts.land[0].y;
205 for (const Vec2& p : opts.land) {
206 minX = std::min(minX, p.x);
207 minY = std::min(minY, p.y);
208 maxX = std::max(maxX, p.x);
209 maxY = std::max(maxY, p.y);
210 }
211 const double landW = std::max(1e-6, maxX - minX);
212 const double landH = std::max(1e-6, maxY - minY);
213 const double cellSize = std::max(0.05, double(params.getFloat("cellSize", 1.f)));
214 const int gridW = std::clamp(int(std::ceil(landW / cellSize)) + 2, 3, 1024);
215 const int gridH = std::clamp(int(std::ceil(landH / cellSize)) + 2, 3, 1024);
216 const double ox = minX - cellSize;
217 const double oy = minY - cellSize;
218 const double halfStreet = opts.streetWidth * 0.5;
219
220 out.resize(gridW, gridH);
221 out.fill(Semantic::Wall);
222 for (int gy = 0; gy < gridH; ++gy) {
223 for (int gx = 0; gx < gridW; ++gx) {
224 const Vec2 c{ox + (double(gx) + 0.5) * cellSize, oy + (double(gy) + 0.5) * cellSize};
225 if (!pointInPolygon(c, opts.land)) continue;
226 bool onStreet = false;
227 for (const Street& s : layout.streets) {
228 for (size_t i = 1; i < s.pts.size(); ++i) {
229 if (distanceToSegment(c, s.pts[i - 1], s.pts[i]) <= halfStreet) {
230 onStreet = true;
231 break;
232 }
233 }
234 if (onStreet) break;
235 }
236 if (onStreet) {
237 out.setCell(gx, gy, int(Semantic::Road));
238 continue;
239 }
240 int pid = 0;
241 for (size_t p = 0; p < layout.parcels.size(); ++p) {
242 Polygon poly;
243 for (const int ci : layout.parcels[p].ring) poly.push_back(layout.corners[size_t(ci)]);
244 if (pointInPolygon(c, poly)) {
245 pid = int(p) + 1;
246 break;
247 }
248 }
249 out.setCell(gx, gy, pid > 0 ? int(Semantic::Floor) : int(Semantic::Wall));
250 if (pid > 0) out.setDetail(gx, gy, std::min(pid, 254));
251 }
252 }
253
254 // Parcel anchors for building placement.
255 out.clearObjects();
256 for (size_t p = 0; p < layout.parcels.size(); ++p) {
257 Polygon poly;
258 for (const int ci : layout.parcels[p].ring) poly.push_back(layout.corners[size_t(ci)]);
259 const Vec2 c = centroid(poly);
260 out.addObjectAt("parcel" + std::to_string(p), "parcel", float((c.x - ox) / cellSize),
261 float((c.y - oy) / cellSize));
262 }
263
264 out.setMeta("algorithm", "urban.parcels");
265 out.setMeta("seed", std::to_string(opts.seed));
266 out.setMeta("parcels", std::to_string(layout.parcels.size()));
267 out.setMeta("streets", std::to_string(layout.streets.size()));
268 out.setMeta("junctions", std::to_string(layout.streetJunctions));
269 out.setMeta("streetLength", std::to_string(layout.totalStreetLength));
270 out.setMeta("avgIrregularity", std::to_string(layout.avgIrregularity));
271 out.setMeta("levels", std::to_string(layout.levelsUsed));
272 out.setMeta("streetPattern", params.getString("streetPattern", "default"));
273 out.setMeta("optimize", opts.optimize ? "1" : "0");
274 out.setMeta("cellSize", std::to_string(cellSize));
275 return true;
276}

459 out.addTriangle(base + lawn[t], base + lawn[t + 2], base + lawn[t + 1]);
460 center = center * (1. / double(poly.size()));
461 for (size_t i = 0; i < poly.size(); i += 2) tree(center * .55 + poly[i] * .45, .85f);
462 ++parkCount;
463 continue;
464 }
465 const int firstVertex = out.getVertexCount();
466 // A named recipe seed and parcel index own the variation; independent of frame time.
467 const uint32_t parcelHash = uint32_t(opts.seed) ^ (uint32_t(p) + 1u) * 0x9e3779b9u;
468 const float noise = float((parcelHash ^ (parcelHash >> 16)) & 0xffffu) / 65535.f;
469 const float parcelHeight = extrude * (1.f + variation * (2.f * noise - 1.f));
470 std::vector<int> tris;
471 if (!triangulatePolygon(poly, tris)) {
472 error = "mesh.urban: failed to triangulate parcel " + std::to_string(p);
473 return false;
474 }
475 auto emitVert = [&](const Vec2& v, float y, float nx, float ny, float nz) {
476 out.addVertex(float(v.x), y, float(v.y), nx, ny, nz, float(v.x) * uvScale, float(v.y) * uvScale);
477 };
478 if (extrude <= 0.f) {
479 const int base = out.getVertexCount();
480 for (size_t i = 0; i < poly.size(); ++i) emitVert(poly[i], 0.f, 0.f, 1.f, 0.f);
481 for (size_t t = 0; t + 2 < tris.size(); t += 3)
482 // Ear-clipping emits CCW (x,y) which is -Y in the XZ plane; flip for +Y.
483 out.addTriangle(uint32_t(base + tris[t]), uint32_t(base + tris[t + 2]), uint32_t(base + tris[t + 1]));
484 } else {
485 const double yTop = double(parcelHeight);
486 const int baseTop = out.getVertexCount();
487 for (size_t i = 0; i < poly.size(); ++i) emitVert(poly[i], float(yTop), 0.f, 1.f, 0.f);
488 for (size_t t = 0; t + 2 < tris.size(); t += 3)
489 out.addTriangle(uint32_t(baseTop + tris[t]), uint32_t(baseTop + tris[t + 2]),
490 uint32_t(baseTop + tris[t + 1]));
491 const int baseBottom = out.getVertexCount();
492 for (size_t i = 0; i < poly.size(); ++i) emitVert(poly[i], 0.f, 0.f, -1.f, 0.f);
493 for (size_t t = 0; t + 2 < tris.size(); t += 3)
494 out.addTriangle(uint32_t(baseBottom + tris[t]), uint32_t(baseBottom + tris[t + 1]),
495 uint32_t(baseBottom + tris[t + 2]));
496 const size_t m = poly.size();
497 for (size_t i = 0; i < m; ++i) {
498 const Vec2& a = poly[i];
499 const Vec2& b = poly[(i + 1) % m];
500 const Vec2 n = normalize(perpendicular(a - b));
501 const int base = out.getVertexCount();
502 emitVert(a, 0.f, float(n.x), 0.f, float(n.y));
503 emitVert(b, 0.f, float(n.x), 0.f, float(n.y));
504 emitVert(b, float(yTop), float(n.x), 0.f, float(n.y));
505 emitVert(a, float(yTop), float(n.x), 0.f, float(n.y));
506 out.addTriangle(uint32_t(base), uint32_t(base + 2), uint32_t(base + 1));
507 out.addTriangle(uint32_t(base), uint32_t(base + 3), uint32_t(base + 2));
508 }
509 }
510 if (vertexColors) {
511 const float warm = float(parcelHash % 3u) * 0.055f;
512 for (int v = firstVertex; v < out.getVertexCount(); ++v) {
513 const bool roof = out.getNormalY(v) > 0.5f;
514 if (details) {
515 const float palette[3][3] = {{.45f, .40f, .31f}, {.26f, .36f, .39f}, {.49f, .28f, .19f}};
516 const auto& wall = palette[parcelHash % 3];
517 colors.insert(colors.end(),
518 {roof ? .12f : wall[0], roof ? .16f : wall[1], roof ? .17f : wall[2], 1.f});
519 continue;
520 }
521 colors.insert(colors.end(), {roof ? 0.27f + warm : 0.62f + warm, roof ? 0.34f + warm : 0.64f,
522 roof ? 0.40f : 0.65f - warm, 1.f});
523 }
524 }
525 if (details) {
526 // Windows are inset within each actual facade edge, never projected onto unrelated parcels.
527 for (size_t edge = 0; edge < poly.size(); ++edge) {
528 Vec2 a = poly[edge], b = poly[(edge + 1) % poly.size()], delta = b - a;
529 double length = std::sqrt(dot(delta, delta));
530 Vec2 normal = normalize(perpendicular(a - b));
531 int columns = int(length / 1.25), floors = int((parcelHeight - .65f) / 1.25f);
532 for (int floor = 0; floor < floors; ++floor)
533 for (int column = 0; column < columns; ++column) {
534 double u = (column + .5) / columns, half = .35 / length;
535 Vec2 left = a + delta * (u - half) + normal * .025,
536 right = a + delta * (u + half) + normal * .025;
537 bool lit = ((column + floor + int(parcelHash % 11)) % 11) == 0;
538 quad(left, right, .65f + floor * 1.25f, 1.35f + floor * 1.25f, float(normal.x), 0,
539 float(normal.y), lit ? .60f : .025f, lit ? .40f : .095f, lit ? .16f : .13f);
540 ++windowCount;
541 }
542 // A shallow cornice closes each roof edge.
543 quad(a + normal * .035, b + normal * .035, parcelHeight, parcelHeight + .18f, float(normal.x), 0,
544 float(normal.y), .22f, .25f, .24f);
545 if (edge == 0 && length > 2.) {
546 Vec2 mid = (a + b) * .5;
547 quad(mid - delta * (.4 / length) + normal * .035, mid + delta * (.4 / length) + normal * .035, .11f,
548 1.3f, float(normal.x), 0, float(normal.y), .035f, .055f, .06f);
549 }
550 }
551 Vec2 center{};
552 for (const auto& v : poly) center = center + v;
553 center = center * (1. / double(poly.size()));
554 // A roof service cap, scaled inside the convex building footprint.
555 std::vector<int> cap;
556 Polygon equipment;
557 for (const auto& v : poly) equipment.push_back(center * .75 + v * .25);
558 if (!triangulatePolygon(equipment, cap)) {
559 error = "mesh.urban: roof equipment triangulation failed";
560 return false;
561 }
562 int base = out.getVertexCount();
563 for (const auto& v : equipment) {
564 out.addVertex(float(v.x), parcelHeight + .4f, float(v.y), 0, 1, 0, 0, 0);
565 colors.insert(colors.end(), {.26f, .29f, .27f, 1.f});
566 }
567 for (size_t t = 0; t + 2 < cap.size(); t += 3)
568 out.addTriangle(base + cap[t], base + cap[t + 2], base + cap[t + 1]);
569 for (size_t i = 0; i < equipment.size(); ++i) {
570 Vec2 a = equipment[i], b = equipment[(i + 1) % equipment.size()], n = normalize(perpendicular(a - b));
571 quad(a, b, parcelHeight, parcelHeight + .4f, float(n.x), 0, float(n.y), .19f, .22f, .21f);
572 }
573 // Trees stand in the paved setback, outside the building edge.
574 if (p % 2 == 0 && !sidewalk.empty()) {
575 Vec2 site = sidewalk[0] * .65 + poly[0] * .35;
576 tree(site, .65f);
577 }
578 }
579 }
580
581 if (vertexColors) {
582 auto colored = out.setVertexColors(std::move(colors));
583 if (!colored) {
584 error = colored.status().describe();
585 return false;
586 }
587 }
588 out.setMeta("detail.crossings", std::to_string(crossingCount));
589 out.setMeta("detail.windows", std::to_string(windowCount));
590 out.setMeta("detail.trees", std::to_string(treeCount));
591 out.setMeta("detail.markings", std::to_string(markingCount));
592 out.setMeta("detail.parks", std::to_string(parkCount));
593 out.setMeta("omittedFootprints", std::to_string(omittedFootprints));
594 out.setMeta("parcelInset", std::to_string(inset));
595
596 out.setMeta("algorithm", "mesh.urban");
597 out.setMeta("seed", std::to_string(opts.seed));
598 out.setMeta("parcels", std::to_string(layout.parcels.size()));
599 out.setMeta("streets", std::to_string(layout.streets.size()));
600 out.setMeta("junctions", std::to_string(layout.streetJunctions));
601 out.setMeta("streetLength", std::to_string(layout.totalStreetLength));
602 out.setMeta("avgIrregularity", std::to_string(layout.avgIrregularity));
603 out.setMeta("extrude", std::to_string(extrude));
604 return true;
605}
