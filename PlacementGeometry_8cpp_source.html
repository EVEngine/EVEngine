60FreeFootprint definitionFreeFootprint(const BuildingDefinition &definition, const PlacementWorld &world, float x,
61 float y, float rotationDeg) {
62 FreeFootprint result;
63 result.x = x;
64 result.y = y;
65 result.radius = definition.freeRadiusCells * std::min(world.getGrid().cellW, world.getGrid().cellH);
66 if (!definition.freeFootprintVertices.empty()) {
67 for (size_t i = 0; i < definition.freeFootprintVertices.size(); i += 2) {
68 result.vertices.push_back({definition.freeFootprintVertices[i] * world.getGrid().cellW,
69 definition.freeFootprintVertices[i + 1] * world.getGrid().cellH});
70 result.halfWidth = std::max(result.halfWidth, std::fabs(result.vertices.back().x));
71 result.halfHeight = std::max(result.halfHeight, std::fabs(result.vertices.back().y));
72 }
73 } else if (definition.freeFootprintWidthCells > 0.f && definition.freeFootprintHeightCells > 0.f) {
74 result.halfWidth = definition.freeFootprintWidthCells * world.getGrid().cellW * 0.5f;
75 result.halfHeight = definition.freeFootprintHeightCells * world.getGrid().cellH * 0.5f;
76 result.vertices = {{-result.halfWidth, -result.halfHeight},
77 {result.halfWidth, -result.halfHeight},
78 {result.halfWidth, result.halfHeight},
79 {-result.halfWidth, result.halfHeight}};
80 }
81 result.rotationRadians = wrapDeg(rotationDeg) * 3.14159265358979323846f / 180.f;
82 return result;
83}

119std::vector<FreeFootprint::Point> worldVertices(const FreeFootprint &footprint) {
120 std::vector<FreeFootprint::Point> result;
121 result.reserve(footprint.vertices.size());
122 const float c = std::cos(footprint.rotationRadians);
123 const float s = std::sin(footprint.rotationRadians);
124 for (const auto &point : footprint.vertices)

129bool isPointInPolygon(const FreeFootprint &polygon, float x, float y) {
130 const auto vertices = worldVertices(polygon);
131 float winding = 0.f;
132 for (size_t i = 0; i < vertices.size(); ++i) {
133 const auto &a = vertices[i];
134 const auto &b = vertices[(i + 1) % vertices.size()];
135 const float cross = (b.x - a.x) * (y - a.y) - (b.y - a.y) * (x - a.x);
136 if (std::fabs(cross) <= 1e-5f) continue;
137 if (winding == 0.f)
138 winding = cross;
139 else if (cross * winding < 0.f)
140 return false;
141 }
142 return true;
143}

145bool hasFreeFootprintOverlap(const FreeFootprint &a, const FreeFootprint &b) {
146 const float dx = b.x - a.x;
147 const float dy = b.y - a.y;
148 if (!a.isPolygon() && !b.isPolygon()) return std::hypot(dx, dy) < a.radius + b.radius;
149
150 auto circlePolygonOverlap = [](const FreeFootprint &circle, const FreeFootprint &polygon) {
151 if (isPointInPolygon(polygon, circle.x, circle.y)) return true;
152 const auto vertices = worldVertices(polygon);
153 const float radiusSquared = circle.radius * circle.radius;
154 for (size_t i = 0; i < vertices.size(); ++i) {
155 const auto &a = vertices[i];
156 const auto &b = vertices[(i + 1) % vertices.size()];
157 const float edgeX = b.x - a.x;
158 const float edgeY = b.y - a.y;
159 const float lengthSquared = edgeX * edgeX + edgeY * edgeY;
160 const float t =
161 lengthSquared > 0.f
162 ? std::clamp(((circle.x - a.x) * edgeX + (circle.y - a.y) * edgeY) / lengthSquared, 0.f, 1.f)
163 : 0.f;
164 const float closestX = a.x + edgeX * t;
165 const float closestY = a.y + edgeY * t;
166 const float distanceX = circle.x - closestX;
167 const float distanceY = circle.y - closestY;
168 if (distanceX * distanceX + distanceY * distanceY < radiusSquared) return true;
169 }
170 return false;
171 };
172 if (!a.isPolygon()) return circlePolygonOverlap(a, b);
173 if (!b.isPolygon()) return circlePolygonOverlap(b, a);
174
175 const auto aVertices = worldVertices(a);
176 const auto bVertices = worldVertices(b);
177 for (const auto *vertices : {&aVertices, &bVertices}) {
178 for (size_t i = 0; i < vertices->size(); ++i) {
179 const auto &p = vertices->at(i);
180 const auto &q = vertices->at((i + 1) % vertices->size());
181 const float axisX = -(q.y - p.y);
182 const float axisY = q.x - p.x;
183 float aMin = std::numeric_limits<float>::max();
184 float aMax = std::numeric_limits<float>::lowest();
185 float bMin = std::numeric_limits<float>::max();
186 float bMax = std::numeric_limits<float>::lowest();
187 for (const auto &vertex : aVertices) {
188 const float projection = vertex.x * axisX + vertex.y * axisY;
189 aMin = std::min(aMin, projection);
190 aMax = std::max(aMax, projection);
191 }
192 for (const auto &vertex : bVertices) {
193 const float projection = vertex.x * axisX + vertex.y * axisY;
194 bMin = std::min(bMin, projection);
195 bMax = std::max(bMax, projection);
196 }
197 if (aMax <= bMin || bMax <= aMin) return false;
198 }
199 }
200 return true;
201}

216SurfaceFrameStatus normalizeSurfaceFrame(PlacementSystem::PlacementHit &hit) {
217 if (!std::isfinite(hit.worldX) || !std::isfinite(hit.worldY) || !std::isfinite(hit.worldZ) ||
218 !std::isfinite(hit.normalX) || !std::isfinite(hit.normalY) || !std::isfinite(hit.normalZ)) {
219 return SurfaceFrameStatus::Invalid;
220 }
221 const float normalLength = length3(hit.normalX, hit.normalY, hit.normalZ);
222 if (normalLength <= 1e-5f) return SurfaceFrameStatus::Invalid;
223 hit.normalX /= normalLength;
224 hit.normalY /= normalLength;
225 hit.normalZ /= normalLength;
226
227 const float tangentDot = hit.tangentX * hit.normalX + hit.tangentY * hit.normalY + hit.tangentZ * hit.normalZ;
228 hit.tangentX -= tangentDot * hit.normalX;
229 hit.tangentY -= tangentDot * hit.normalY;
230 hit.tangentZ -= tangentDot * hit.normalZ;
231 float tangentLength = length3(hit.tangentX, hit.tangentY, hit.tangentZ);
232 if (tangentLength <= 1e-5f) {
233 if (std::fabs(hit.normalY) < 0.9f) {
234 hit.tangentX = 0.f;
235 hit.tangentY = 1.f;
236 hit.tangentZ = 0.f;
237 } else {
238 hit.tangentX = 1.f;
239 hit.tangentY = 0.f;
240 hit.tangentZ = 0.f;
241 }
242 const float fallbackDot = hit.tangentX * hit.normalX + hit.tangentY * hit.normalY + hit.tangentZ * hit.normalZ;
243 hit.tangentX -= fallbackDot * hit.normalX;
244 hit.tangentY -= fallbackDot * hit.normalY;
245 hit.tangentZ -= fallbackDot * hit.normalZ;
246 tangentLength = length3(hit.tangentX, hit.tangentY, hit.tangentZ);
247 }
248 hit.tangentX /= tangentLength;
249 hit.tangentY /= tangentLength;
250 hit.tangentZ /= tangentLength;
251 cross3(hit.tangentX, hit.tangentY, hit.tangentZ, hit.normalX, hit.normalY, hit.normalZ, hit.bitangentX,
252 hit.bitangentY, hit.bitangentZ);
253 return SurfaceFrameStatus::Normalized;
254}
