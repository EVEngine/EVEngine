40Result<std::uint64_t> GtsTerrainColliderRuntime::replace(const procgen::GtsTerrainLodSet& lods, physics::World3D& world,
41 int level, float ox, float oy, float oz) {
42 if (!world.isValid() || level < 0 || level >= lods.getLevelCount() || !std::isfinite(ox) || !std::isfinite(oy) ||
43 !std::isfinite(oz))
44 return Result<std::uint64_t>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
45 "GTS collider world, level or origin is invalid",
46 "procgen.physics.gtsTerrainCollider"));
47 std::vector<ColliderLink> candidate(static_cast<std::size_t>(lods.getTileCount()));
48 try {
49 for (int tileIndex = 0; tileIndex < lods.getTileCount(); ++tileIndex) {
50 const auto* tile = lods.tileAt(tileIndex);
51 if (!tile || level >= static_cast<int>(tile->levels.size())) {
52 destroy(candidate);
53 return Result<std::uint64_t>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
54 "GTS collider tile level is missing",
55 "procgen.physics.gtsTerrainCollider"));
56 }
57 const auto& mesh = tile->levels[static_cast<std::size_t>(level)];
58 if (mesh.empty()) continue;
59 auto* body = world.newBody("static", ox + tile->offsetX, oy, oz + tile->offsetZ);
60 if (!body) {
61 destroy(candidate);
62 return Result<std::uint64_t>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
63 "GTS collider body creation failed",
64 "procgen.physics.gtsTerrainCollider"));
65 }
66 ColliderLink link{world.runtimeHandle(), body->runtimeHandle()};
67 candidate[static_cast<std::size_t>(tileIndex)] = link;
68 std::vector<std::int32_t> indices;
69 indices.reserve(mesh.indices().size());
70 for (auto index : mesh.indices()) {
71 if (index > static_cast<std::uint32_t>(std::numeric_limits<std::int32_t>::max()))
72 throw eve::Exception("mesh index exceeds physics range");
73 indices.push_back(static_cast<std::int32_t>(index));
74 }
75 try {
76 if (!body->newTriangleMeshShape(mesh.positions(), indices))
77 throw eve::Exception("shape creation returned null");
78 } catch (...) {
79 world.destroyBody(body);
80 candidate[static_cast<std::size_t>(tileIndex)] = {};
81 throw;
82 }
83 }
84 } catch (const std::exception& error) {
85 destroy(candidate);
86 return Result<std::uint64_t>::failure(
87 Diagnostic::error(DiagnosticCode::Failed, error.what(), "procgen.physics.gtsTerrainCollider"));
88 }
89 destroy(impl_->tiles);
90 impl_->tiles = std::move(candidate);
91 ++impl_->revision;
92 if (!impl_->revision) ++impl_->revision;
93 return Result<std::uint64_t>::success(impl_->revision);
94}
