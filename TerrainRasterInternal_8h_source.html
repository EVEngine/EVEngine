31inline double textureSample(const Heightmap& map, double u, double v) {
32 const double x = std::clamp(u * map.getWidth() - 0.5, 0.0, double(map.getWidth() - 1));
33 const double z = std::clamp(v * map.getHeight() - 0.5, 0.0, double(map.getHeight() - 1));
34 const int x0 = int(x), z0 = int(z), x1 = std::min(x0 + 1, map.getWidth() - 1),
35 z1 = std::min(z0 + 1, map.getHeight() - 1);
36 return std::lerp(std::lerp(double(map.height(x0, z0)), double(map.height(x1, z0)), x - x0),
37 std::lerp(double(map.height(x0, z1)), double(map.height(x1, z1)), x - x0), z - z0);
38}

40// Double intermediates avoid overflow when interpolating large finite floats.
41inline double sample(const Heightmap& map, double u, double v) {
42 const double x = u * (map.getWidth() - 1), y = v * (map.getHeight() - 1);
43 const int x0 = int(x), y0 = int(y);
44 const int x1 = std::min(x0 + 1, map.getWidth() - 1), y1 = std::min(y0 + 1, map.getHeight() - 1);
45 const double top = std::lerp(double(map.height(x0, y0)), double(map.height(x1, y0)), x - x0);
46 const double bottom = std::lerp(double(map.height(x0, y1)), double(map.height(x1, y1)), x - x0);
47 return std::lerp(top, bottom, y - y0);
48}
