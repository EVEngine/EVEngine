60 const auto* value = member(object, name);

62 if (const auto* number = value->getIf<double>()) {

64 output = static_cast<float>(*number);

115eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>> SceneArtifactProvider::prepare(
116 const eve::artifact::PublicationView& publication) {
117 if (failPrepare_)
118 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(
119 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "scene artifact prepare failure was injected"));
120 if (publication.id.isNil() || publication.buildKey.empty() || publication.parts.empty())
121 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(eve::Diagnostic::error(
122 eve::DiagnosticCode::InvalidArgument, "scene publication requires identity, build key and parts"));
123 if (find(publication.id) != nullptr)
124 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(
125 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "scene artifact identity is already published"));
126 if (nextIndex_ == SceneArtifactHandle::invalidIndex)
127 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(
128 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "scene artifact handle index exhausted"));
129
130 SceneArtifactRecord record;
131 record.id = publication.id;
132 record.buildKey = std::string(publication.buildKey);
133 record.handle = SceneArtifactHandle(nextIndex_, 1);
134 record.nodes.reserve(publication.parts.size());
135 for (const auto& part : publication.parts) {
136 if (part.id.isNil() || part.role.empty() || !validBounds(part.bounds))
137 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(eve::Diagnostic::error(
138 eve::DiagnosticCode::InvalidArgument, "scene artifact part has invalid identity, role or bounds"));
139 record.nodes.push_back({std::string(part.role), part.bounds});
140 }
141 try {
142 records_.reserve(records_.size() + 1);
143 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::success(
144 std::make_unique<SceneArtifactStage>(*this, std::move(record)));
145 } catch (...) {
146 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(
147 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "scene artifact prepare allocation failed"));
148 }
149}

168eve::Result<eve::Value> SceneArtifactProvider::snapshotState() const {
169 try {
170 eve::Value::Array records;
171 records.reserve(records_.size());
172 for (const auto& record : records_) {
173 eve::Value::Array nodes;
174 nodes.reserve(record.nodes.size());
175 for (const auto& node : record.nodes) {
176 eve::Value::Object item;
177 item.emplace("role", eve::Value(node.role));
178 item.emplace("bounds", boundsValue(node.bounds));
179 nodes.emplace_back(std::move(item));
180 }
181 eve::Value::Object item;
182 item.emplace("artifactId", eve::Value(record.id.format()));
183 item.emplace("buildKey", eve::Value(record.buildKey));
184 item.emplace("handle", eve::Value(std::to_string(record.handle.packed())));
185 item.emplace("nodes", eve::Value(std::move(nodes)));
186 records.emplace_back(std::move(item));
187 }
188 eve::Value::Object state;
189 state.emplace("provider", eve::Value("scene.cpu-registry"));
190 state.emplace("version", eve::Value(std::int64_t(1)));
191 state.emplace("records", eve::Value(std::move(records)));
192 return eve::Result<eve::Value>::success(eve::Value(std::move(state)));
193 } catch (...) {
194 return failureValue(eve::DiagnosticCode::Failed, "scene artifact snapshot allocation failed");
195 }

198eve::Result<void> SceneArtifactProvider::restoreState(const eve::Value& state) {
199 if (!records_.empty())
200 return eve::Result<void>::failure(
201 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "scene artifact restore requires an empty registry"));
202 const auto* object = state.getIf<eve::Value::Object>();
203 const auto* provider = object ? member(*object, "provider") : nullptr;
204 const auto* providerName = provider ? provider->getIf<std::string>() : nullptr;
205 const auto* records = object ? member(*object, "records") : nullptr;
206 const auto* array = records ? records->getIf<eve::Value::Array>() : nullptr;
207 if (!object || !providerName || *providerName != "scene.cpu-registry" || !hasSupportedVersion(*object) || !array)
208 return eve::Result<void>::failure(
209 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "invalid scene artifact provider state"));
210
211 std::vector<SceneArtifactRecord> candidate;
212 std::unordered_set<eve::PersistentId> identities;
213 std::uint32_t next = 0;
214 try {
215 candidate.reserve(array->size());
216 for (const eve::Value& encoded : *array) {
217 const auto* item = encoded.getIf<eve::Value::Object>();
218 std::string idText;
219 std::string buildKey;
220 std::uint64_t packed = 0;
221 if (!item || !readString(*item, "artifactId", idText) || !readString(*item, "buildKey", buildKey) ||
222 !readUint64String(*item, "handle", packed))
223 return eve::Result<void>::failure(
224 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "invalid scene artifact record"));
225 const auto parsedId = eve::PersistentId::parse(idText);
226 const auto handle = SceneArtifactHandle::fromPacked(packed);
227 const auto* encodedNodes = member(*item, "nodes");
228 const auto* nodes = encodedNodes ? encodedNodes->getIf<eve::Value::Array>() : nullptr;
229 if (!parsedId || parsedId->isNil() || handle.isInvalid() || !nodes ||
230 !identities.emplace(*parsedId).second || handle.index() == SceneArtifactHandle::invalidIndex)
231 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::ParseError,
232 "invalid scene artifact identity or handle"));
233 SceneArtifactRecord record{*parsedId, std::move(buildKey), handle, {}};
234 record.nodes.reserve(nodes->size());
235 for (const eve::Value& encodedNode : *nodes) {
236 const auto* node = encodedNode.getIf<eve::Value::Object>();
237 std::string role;
238 const auto* boundsEncoded = node ? member(*node, "bounds") : nullptr;
239 const auto* boundsObject = boundsEncoded ? boundsEncoded->getIf<eve::Value::Object>() : nullptr;
240 if (!node || !readString(*node, "role", role) || !boundsObject)
241 return eve::Result<void>::failure(
242 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "invalid scene artifact node"));
243 eve::artifact::Bounds bounds;
244 if (!readBounds(*node, bounds))
245 return eve::Result<void>::failure(
246 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "invalid scene artifact node bounds"));
247 record.nodes.push_back({std::move(role), bounds});
248 }
249 next = std::max(next, handle.index() == std::numeric_limits<std::uint32_t>::max() ? handle.index()
250 : handle.index() + 1u);
251 candidate.push_back(std::move(record));
252 }
253 } catch (...) {
254 return eve::Result<void>::failure(
255 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "scene artifact restore allocation failed"));
256 }
257 records_.swap(candidate);
258 nextIndex_ = next;
259 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
260}
