80struct ArithOp {
81 ArithKind kind = ArithKind::Add;
82 double value = 0.0;

85[[nodiscard]] double applyArith(double base, const std::vector<ArithOp>& ops) {
86 double result = base;
87 for (const auto& op : ops) {
88 if (op.kind == ArithKind::Add)
89 result += op.value;
90 else
91 result *= op.value;
92 }
93 return result;
94}

261Result<CarrierRecipe> buildProjectile(const Value::Object& object, const ModifierStack& stack, std::string path) {
262 if (field(object, "triggers") || field(object, "impacts") || field(object, "motionOps") ||
263 (field(object, "motion") && field(object, "motion")->isArray())) {
264 return failT<CarrierRecipe>(
265 DiagnosticCode::Unsupported,
266 "Projectile fragments must use preset fields, not full-form motion/triggers/impacts", path);
267 }
268
269 std::string id;
270 if (!readString(object, "id", id) || id.empty())
271 return failT<CarrierRecipe>(DiagnosticCode::InvalidArgument, "projectile requires id", path + ".id");
272
273 auto speedField = optionalDouble(object, "speed", path + ".speed");
274 if (!speedField.ok()) return Result<CarrierRecipe>::failure(speedField.status());
275 double speed = applyArith(speedField.value().value_or(10.0), stack.speedOps);
276 if (!(speed > 0.0) || !std::isfinite(speed))
277 return failT<CarrierRecipe>(DiagnosticCode::InvalidArgument, "projectile speed must be positive", path + ".speed");
278
279 auto lifetimeField = optionalDouble(object, "lifetime", path + ".lifetime");
280 if (!lifetimeField.ok()) return Result<CarrierRecipe>::failure(lifetimeField.status());
281 double lifetime = applyArith(lifetimeField.value().value_or(2.0), stack.lifetimeOps);
282 if (!(lifetime > 0.0) || !std::isfinite(lifetime))
283 return failT<CarrierRecipe>(DiagnosticCode::InvalidArgument, "projectile lifetime must be positive",
284 path + ".lifetime");
285
286 auto damageField = optionalDouble(object, "damage", path + ".damage");
287 if (!damageField.ok()) return Result<CarrierRecipe>::failure(damageField.status());
288 double damage = applyArith(damageField.value().value_or(0.0), stack.damageOps);
289
290 ModifierStack merged = stack;
291 auto seedDouble = [&](const char* name, double& dest) -> Result<void> {
292 auto value = optionalDouble(object, name, std::string(path) + "." + name);
293 if (!value.ok()) return Result<void>::failure(value.status());
294 if (dest <= 0.0 && value.value().has_value()) dest = *value.value();
295 return Result<void>::success();
296 };
297 if (auto seeded = seedDouble("homing", merged.homing); !seeded)
298 return Result<CarrierRecipe>::failure(seeded.status());
299 if (auto seeded = seedDouble("maxTurnRateDegrees", merged.homing); !seeded)
300 return Result<CarrierRecipe>::failure(seeded.status());
301 if (auto seeded = seedDouble("gravity", merged.gravity); !seeded)
302 return Result<CarrierRecipe>::failure(seeded.status());
303 if (auto seeded = seedDouble("accelerate", merged.accelerate); !seeded)
304 return Result<CarrierRecipe>::failure(seeded.status());
305
306 std::string motionPreset;
307 (void)readString(object, "motion", motionPreset);
308 if (motionPreset == "homing" && merged.homing <= 0.0) merged.homing = 90.0;
309 if (motionPreset == "ballistic" && merged.gravity <= 0.0) merged.gravity = 9.8;
310
311 auto pierceField = optionalInt(object, "pierce", path + ".pierce");
312 if (!pierceField.ok()) return Result<CarrierRecipe>::failure(pierceField.status());
313 if (merged.pierce <= 0 && pierceField.value().has_value()) merged.pierce = *pierceField.value();
314 auto bounceField = optionalInt(object, "bounce", path + ".bounce");
315 if (!bounceField.ok()) return Result<CarrierRecipe>::failure(bounceField.status());
316 if (merged.bounce <= 0 && bounceField.value().has_value()) merged.bounce = *bounceField.value();
317
318 Value::Object preset;
319 preset["id"] = Value(id);
320 preset["speed"] = Value(speed);
321 preset["lifetime"] = Value(lifetime);
322 preset["damage"] = Value(damage);
323 if (merged.gravity > 0.0) preset["gravity"] = Value(merged.gravity);
324 if (merged.homing > 0.0) preset["homing"] = Value(merged.homing);
325 if (merged.accelerate > 0.0) preset["accelerate"] = Value(merged.accelerate);
326 if (merged.pierce > 0) preset["pierce"] = Value(static_cast<std::int64_t>(merged.pierce));
327 if (merged.bounce > 0) preset["bounce"] = Value(static_cast<std::int64_t>(merged.bounce));
328 if (merged.fuse > 0.0) preset["fuse"] = Value(merged.fuse);
329 if (merged.splash > 0.0) preset["splash"] = Value(merged.splash);
330
331 std::string element;
332 std::string damageType;
333 (void)readString(object, "element", element);
334 (void)readString(object, "damageType", damageType);
335 if (!merged.element.empty()) element = merged.element;
336 if (!merged.damageType.empty()) damageType = merged.damageType;
337 if (!element.empty()) preset["element"] = Value(element);
338 if (!damageType.empty()) preset["damageType"] = Value(damageType);
339
340 auto recipe = decodeCarrierRecipe(Value(std::move(preset)));
341 if (!recipe.ok()) return recipe;
342 CarrierRecipe built = std::move(recipe).takeValue();
343
344 // Preserve preset motion from decode; only inject curve ops that presets cannot express.
345 if (merged.swayAmplitude > 0.0 || merged.helixAmplitude > 0.0) {
346 std::vector<CarrierMotionOp> motion = built.motionOps;
347 // Insert curve ops before the final IntegrateLinear.
348 if (!motion.empty() && motion.back().kind == CarrierMotionOpKind::IntegrateLinear) motion.pop_back();
349 if (merged.swayAmplitude > 0.0) {
350 CarrierMotionOp op;
351 op.kind = CarrierMotionOpKind::CurveSway;
352 op.curveAmplitude = merged.swayAmplitude;
353 op.curveFrequencyHz = merged.swayFrequency > 0.0 ? merged.swayFrequency : 2.0;
354 motion.push_back(op);
355 }
356 if (merged.helixAmplitude > 0.0) {
357 CarrierMotionOp op;
358 op.kind = CarrierMotionOpKind::CurveHelix;
359 op.curveAmplitude = merged.helixAmplitude;
360 op.curveFrequencyHz = merged.helixFrequency > 0.0 ? merged.helixFrequency : 2.0;
361 motion.push_back(op);
362 }
363 motion.push_back({CarrierMotionOpKind::IntegrateLinear});
364 built.motionOps = std::move(motion);
365 }
366
367 if (merged.bounce > 0) {
368 for (auto& impact : built.impacts) {
369 if (impact.kind == CarrierImpactKind::Bounce) impact.restitution = merged.restitution;
370 }
371 }
372
373 auto valid = built.validate();
374 if (!valid) return Result<CarrierRecipe>::failure(valid.status());
375 return Result<CarrierRecipe>::success(std::move(built));
376}

380Result<SpellCastPlan> compileSpellFragments(const Value& fragments) {
381 if (!fragments.isArray())
382 return failT<SpellCastPlan>(DiagnosticCode::InvalidArgument, "Fragments must be an array", "");
383 if (fragments.arraySize() == 0)
384 return failT<SpellCastPlan>(DiagnosticCode::InvalidArgument, "Fragments must not be empty", "");
385
386 ModifierStack stack;
387 std::optional<CarrierRecipe> recipe;
388 for (std::size_t i = 0; i < fragments.arraySize(); ++i) {
389 const std::string path = "[" + std::to_string(i) + "]";
390 const auto* object = fragments.at(i).getIf<Value::Object>();
391 if (!object)
392 return failT<SpellCastPlan>(DiagnosticCode::InvalidArgument, "Fragment must be an object", path);
393 std::string kind;
394 if (!readString(*object, "kind", kind) || kind.empty())
395 return failT<SpellCastPlan>(DiagnosticCode::InvalidArgument, "Fragment requires kind", path + ".kind");
396
397 if (kind == "projectile" || kind == "bolt" || kind == "grenade" || kind == "missile") {
398 if (recipe.has_value())
399 return failT<SpellCastPlan>(DiagnosticCode::InvalidArgument,
400 "Only one projectile fragment is supported per cast", path);
401 auto built = buildProjectile(*object, stack, path);
402 if (!built.ok()) return Result<SpellCastPlan>::failure(built.status());
403 recipe = std::move(built).takeValue();
404 continue;
405 }
406
407 if (recipe.has_value())
408 return failT<SpellCastPlan>(DiagnosticCode::InvalidArgument,
409 "Modifiers must appear before the projectile fragment", path);
410
411 auto applied = applyModifier(stack, *object, kind, path);
412 if (!applied) return Result<SpellCastPlan>::failure(applied.status());
413 }
414
415 if (!recipe.has_value())
416 return failT<SpellCastPlan>(DiagnosticCode::InvalidArgument,
417 "Fragment sequence requires one projectile fragment", "");
418
419 SpellCastPlan plan;
420 plan.recipe = std::move(*recipe);
421 plan.volley = stack.volley;
422 return Result<SpellCastPlan>::success(std::move(plan));
423}
