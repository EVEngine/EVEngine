35Result<std::unique_ptr<OceanSprayRenderer>> OceanSprayRendererBuilder::build() const {
36 auto lifetime = graphics_.resourceLifetime();
37 if (lifetime.expired())
38 return Result<std::unique_ptr<OceanSprayRenderer>>::failure(
39 failure(DiagnosticCode::StaleHandle, "Graphics retired"));
40 const auto backend = graphics_.getBackendName();
41 if (backend != "vulkan" && backend != "webgpu")
42 return Result<std::unique_ptr<OceanSprayRenderer>>::failure(
43 failure(DiagnosticCode::Unsupported, "Spray requires Vulkan or WebGPU"));
44 std::unique_ptr<OceanSprayRenderer> result(new OceanSprayRenderer);
45 auto& state = *result->impl_;
46 state.graphics = &graphics_;
47 state.lifetime = lifetime;
48 try {
49 if (backend == "webgpu")
50 state.shader = graphics_.newMeshShaderFromWgsl(shaders::sprayVertex, shaders::sprayFragment);
51 else
52 state.shader = graphics_.newMeshShaderFromSpv(
53 std::vector<std::uint32_t>(ocean_spray_vert_spv, ocean_spray_vert_spv + ocean_spray_vert_spv_count),
54 std::vector<std::uint32_t>(ocean_spray_frag_spv, ocean_spray_frag_spv + ocean_spray_frag_spv_count));
55 if (!state.shader)
56 return Result<std::unique_ptr<OceanSprayRenderer>>::failure(
57 failure(DiagnosticCode::Unsupported, "Spray shader unavailable"));
58 auto configured = graphics_.configureMeshShaderSurface(*state.shader, graphics::BlendMode::Alpha, false, true);
59 if (!configured) return Result<std::unique_ptr<OceanSprayRenderer>>::failure(configured.status());
60 const float positions[9] = {0, 0, 0, 1, 0, 0, 0, 1, 0}, attributes[9] = {}, uv[6] = {};
61 const std::uint32_t indices[3] = {0, 1, 2};
62 for (auto& mesh : state.mesh) {
63 mesh = graphics_.newMeshFromArrays(positions, attributes, uv, 3, indices, 3);

67 }

70 }
71 return Result<std::unique_ptr<OceanSprayRenderer>>::success(std::move(result));

73Result<void> OceanSprayRenderer::publish(const OceanSprayMeshFrame& frame) {
74 auto& s = *impl_;
75 if (s.lifetime.expired()) return Result<void>::failure(failure(DiagnosticCode::StaleHandle, "Graphics retired"));
76 const auto vertices = frame.positions.size() / 3;
77 auto finite = [](float v) { return std::isfinite(v); };
78 if (frame.positions.size() % 12 || vertices > 16384 * 4 || frame.attributes.size() != frame.positions.size() ||
79 frame.uv.size() != vertices * 2 || frame.indices.size() != vertices / 4 * 6 || !std::isfinite(frame.time) ||
80 !std::all_of(frame.positions.begin(), frame.positions.end(), finite) ||
81 !std::all_of(frame.attributes.begin(), frame.attributes.end(), finite) ||
82 !std::all_of(frame.uv.begin(), frame.uv.end(), finite))
83 return Result<void>::failure(failure(DiagnosticCode::InvalidArgument, "Invalid spray batch"));
84 for (std::size_t i = 0; i < vertices; ++i)
85 if (frame.attributes[i * 3] < 0 || frame.attributes[i * 3] > 1 || frame.attributes[i * 3 + 1] != 0 ||
86 frame.attributes[i * 3 + 2] != 0)
87 return Result<void>::failure(failure(DiagnosticCode::InvalidArgument, "Invalid spray coverage"));
88 for (auto index : frame.indices)
89 if (index >= vertices)
90 return Result<void>::failure(failure(DiagnosticCode::InvalidArgument, "Invalid spray index"));
91 if (vertices == 0) {
92 s.empty = true;
93 s.published = true;
94 return Result<void>::success();
95 }
96 const auto next = 1 - s.front;
97 if (!s.graphics->updateMeshVertices(s.mesh[next], frame.positions.data(), frame.attributes.data(), frame.uv.data(),
98 static_cast<int>(vertices), frame.indices.data(),
99 static_cast<int>(frame.indices.size())))
100 return Result<void>::failure(failure(DiagnosticCode::Unsupported, "Spray upload rejected"));
101 s.front = next;
102 s.empty = false;
103 s.published = true;
104 return Result<void>::success();
105}
