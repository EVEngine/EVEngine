21void registerCppEntityClassForScript(const ssq::Class& cls) {
22 eve::registerCppEntityView(cls, [](ssq::Array& out) {
23 HSQUIRRELVM vm = out.getHandle();
24 sq_pushobject(vm, out.getRaw());
25 ecs::Table* table = ecs::current();
26 if (table == nullptr) {
27 sq_pop(vm, 1);
28 return;
29 }
30 ecs::IComponentManager& cm = table->getOrCreateManager<T>();
31 auto* reg = cm.getOrCreateRegistryComponentBuffer<T>();
32 std::vector<ecs::IComponentBuffer*> stack;
33 stack.push_back(reg);
34 while (!stack.empty()) {
35 ecs::IComponentBuffer* buf = stack.back();
36 stack.pop_back();
37 auto* r = dynamic_cast<ecs::IRegistryComponentBuffer*>(buf);
38 if (r != nullptr) {
39 for (uint32_t i = 0; i < r->entity_count(); ++i) {
40 ecs::Entity* ent = r->entity_at(i);
41 if (ent != nullptr && ecs::is_entity_visible(ent)) {
42 ssq::detail::pushByPtr<T>(vm, static_cast<T*>(ent));
43 sq_arrayappend(vm, -2);
44 }
45 }
46 }
47 if (buf->children != nullptr) stack.push_back(buf->children);
48 if (buf->next != nullptr) stack.push_back(buf->next);
49 }
50 sq_pop(vm, 1);
51 });
52}

97Result<WarshipEntity*> Warship::newWarship(const std::string& defId, float x, float y, float heading,
98 const std::string& faction) {
99 const WarshipDefinition* def = WarshipRegistry::findHull(defId);
100 if (!def) {
101 return Result<WarshipEntity*>::failure(
102 Diagnostic::error(DiagnosticCode::InvalidArgument, "unknown hull definition", "warship.new"));
103 }
104 if (!std::isfinite(x) || !std::isfinite(y) || !std::isfinite(heading)) {
105 return Result<WarshipEntity*>::failure(
106 Diagnostic::error(DiagnosticCode::InvalidArgument, "spawn transform must be finite", "warship.new"));
107 }
108
109 WarshipEntity* ship = WarshipEntity::createWarship();
110 ship->identity()->id = defId + "#" + std::to_string(reinterpret_cast<std::uintptr_t>(ship));
111 ship->identity()->defId = defId;
112 ship->identity()->faction = faction;
113 ship->identity()->category = def->category;
114 ship->definition()->def = def;
115 ship->motion()->x = x;
116 ship->motion()->y = y;
117 ship->motion()->heading = heading;
118 ship->health()->hp = def->maxHealth;
119 ship->health()->maxHp = def->maxHealth;
120 ship->hull()->length = def->length;
121 ship->hull()->beam = def->beam;
122 ship->hull()->decks = def->grid.decks;
123 ship->layout()->grid = def->grid;
124 ship->layout()->cells.assign(static_cast<std::size_t>(def->grid.cellCount()), CompartmentCell{});
125
126 for (const LayoutSeed& seed : def->defaultLayout) {
127 const int index = ship->layout()->grid.indexOf(seed.deck, seed.row, seed.col);
128 if (index < 0) continue;
129 if (!WarshipRegistry::hasFunction(seed.functionId)) continue;
130 ship->layout()->cells[static_cast<std::size_t>(index)].functionId = seed.functionId;
131 }
132 WarshipSystem::recomputeCapabilities(*ship);
133 return Result<WarshipEntity*>::success(ship);
134}
