50eve::MonotonicTimestamp Timer::monotonicNow() const {
51 if (freq_ == 0) return eve::MonotonicTimestamp::zero();
52 const long double counter = static_cast<long double>(SDL_GetPerformanceCounter());
53 const long double nanos = counter * 1000000000.0L / static_cast<long double>(freq_);
54 if (nanos >= static_cast<long double>(std::numeric_limits<std::int64_t>::max()))
55 return eve::MonotonicTimestamp(std::numeric_limits<std::int64_t>::max());
56 return eve::MonotonicTimestamp(static_cast<std::int64_t>(nanos));
57}
