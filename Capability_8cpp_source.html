56void addListenerRaw(const char* name, void* impl, int priority) {
57 if (!name || !impl) return;
58 auto& list = slots()[name].listeners;
59 for (const auto& l : list)
60 if (l.impl == impl) return; // idempotent
61 // stable_sort keeps registration order within one priority.
62 list.push_back({impl, priority});
63 std::stable_sort(list.begin(), list.end(),
64 [](const Listener& a, const Listener& b) { return a.priority < b.priority; });
65}

67void removeListenerRaw(const char* name, void* impl) {
68 if (!name) return;
69 auto& table = slots();
70 auto it = table.find(name);
71 if (it == table.end()) return;
72 auto& list = it->second.listeners;
73 list.erase(std::remove_if(list.begin(), list.end(),
74 [impl](const Listener& l) { return l.impl == impl; }),
75 list.end());
76}
