105Result<void> HexMapModule::newGrid(graphics::Graphics* gfx, std::int32_t cellCountX, std::int32_t cellCountZ,
106 std::uint32_t seed) {
107 if (!gfx) return invalidArgument("newGrid requires a graphics device");
108 if (cellCountX <= 0 || cellCountZ <= 0) return invalidArgument("hex map size must be positive");
109 // Script-supplied sizes reach a raw cell allocation, so bound them here rather than
110 // let std::bad_alloc escape a native binding. The save format caps the same
111 // quantity at kMaxHexGridDimension.
112 if (cellCountX > kMaxHexGridDimension || cellCountZ > kMaxHexGridDimension)
113 return invalidArgument("hex map size exceeds the " + std::to_string(kMaxHexGridDimension) +
114 " cell per side limit");
115 // Reset before releasing: `HexMap::reset` validates before it mutates, so a
116 // rejected size now leaves the old grid *and* its meshes untouched. Releasing
117 // first detached every GPU mesh while the renderables scripts already hold kept
118 // pointing at them, and the next draw threw on a null mesh.
119 auto reset = map_.reset(cellCountX, cellCountZ, seed);
120 if (!reset) return reset;
121 if (!meshes_.empty()) releaseMeshes(gfx);
122 visibility_.reset(map_.cellCount());
123 units_.removeAll(map_, visibility_);
124 syncScratch();
125 clearMeshSlots();
126 return Result<void>::success();
127}

129graphics::Mesh* HexMapModule::chunkMeshAt(std::int32_t chunkIndex, HexSurface surface) const {
130 if (chunkIndex < 0 || chunkIndex >= map_.chunkCount()) return nullptr;
131 const std::int32_t slot = meshSlot(chunkIndex, surface);
132 if (slot < 0 || static_cast<std::size_t>(slot) >= meshes_.size()) return nullptr;
133 // A surface that lost all geometry keeps its last GPU mesh allocated but is
134 // no longer exposed, so a renderable can never be asked to draw stale data.
135 if (meshPopulated_[static_cast<std::size_t>(slot)] == 0u) return nullptr;
136 return meshes_[static_cast<std::size_t>(slot)];
137}

139std::int32_t HexMapModule::rebuildChunk(graphics::Graphics* gfx, std::int32_t chunkIndex) {
140 if (!gfx || map_.empty()) return 0;
141 if (chunkIndex < 0 || chunkIndex >= map_.chunkCount()) return 0;
142
143 std::int32_t rebuilt = 0;
144 HexMeshData meshData;
145 for (std::int32_t s = 0; s < kHexSurfaceCount; ++s) {
146 const auto surface = static_cast<HexSurface>(s);
147 const std::int32_t slot = meshSlot(chunkIndex, surface);
148 const auto slotIndex = static_cast<std::size_t>(slot);
149 graphics::Mesh*& mesh = meshes_[slotIndex];
150 std::uint8_t& populated = meshPopulated_[slotIndex];
151 if (surface == HexSurface::Fog) {
152 buildFogMesh(map_, visibility_, chunkIndex, meshData);
153 } else if (surface == HexSurface::Wall) {
154 buildWallMesh(map_, chunkIndex, meshData);
155 } else if (surface == HexSurface::Feature) {
156 buildFeatureMesh(map_, chunkIndex, meshData);
157 } else {
158 buildChunkSurfaceMesh(map_, chunkIndex, surface, meshData);
159 }
160 if (meshData.empty()) {
161 // The surface lost all geometry (for example the last road was removed).
162 // The GPU mesh is retired lazily: it stays allocated until the grid is
163 // replaced, so a renderable that still references it is never dangling.
164 populated = 0u;
165 continue;
166 }
167
168 const int vertexCount = static_cast<int>(meshData.vertexCount());
169 const int indexCount = static_cast<int>(meshData.indices().size());
170 const auto upload = [&]() {
171 return gfx->newMeshFromArrays(meshData.positions().data(), meshData.normals().data(), meshData.uvs().data(),
172 vertexCount, meshData.indices().data(), indexCount);
173 };
174
175 if (mesh &&
176 !gfx->updateMeshVertices(mesh, meshData.positions().data(), meshData.normals().data(),
177 meshData.uvs().data(), vertexCount, meshData.indices().data(), indexCount)) {
178 // The backend rejected the in-place update; recreate rather than keep stale geometry.
179 (void)gfx->releaseMesh(mesh);
180 mesh = nullptr;
181 }
182 if (!mesh) mesh = upload();
183 populated = mesh ? 1u : 0u;
184 if (mesh) ++rebuilt;
185 }
186 return rebuilt;
187}

189std::int32_t HexMapModule::rebuildDirtyChunks(graphics::Graphics* gfx) {
190 if (!gfx || map_.empty()) return 0;
191 std::int32_t rebuilt = 0;
192 for (;;) {
193 const std::int32_t chunkIndex = map_.takeDirtyChunk();
194 if (chunkIndex < 0) break;
195 rebuilt += rebuildChunk(gfx, chunkIndex);
196 }
197 return rebuilt;
198}

211Result<void> HexMapModule::newSphere(graphics::Graphics* gfx, std::int32_t subdivision, float radius,
212 std::uint32_t seed) {
213 if (gfx == nullptr) return invalidArgument("hex sphere needs a graphics device");
214 // Same ordering rule as `newGrid`: validate through `reset` first, because
215 // `HexSphereMap::reset` rejects its arguments before touching the map. Releasing
216 // the sphere meshes up front left the planet example's renderables pointing at
217 // GPU meshes that had already been destroyed whenever the arguments were bad.
218 auto reset = sphere_.reset(subdivision, radius, seed);
219 if (!reset) return reset;
220 releaseSphereMeshes(gfx);
221 return Result<void>::success();
222}

225 std::int32_t waterLevel) {

241Result<void> HexMapModule::rebuildSphere(graphics::Graphics* gfx) {
242 if (gfx == nullptr) return invalidArgument("hex sphere needs a graphics device");
243 if (sphere_.empty()) return invalidArgument("hex sphere map has no cells");
244
245 HexMeshData terrain;
246 buildSphereTerrainMesh(sphere_, terrain);
247 HexMeshData water;
248 buildSphereWaterMesh(sphere_, water);
249
250 bool uploaded = true;
251 const auto upload = [&](HexMeshData& data, graphics::Mesh*& mesh) {
252 if (data.empty()) {
253 // Nothing is flooded any more, for example: retire the mesh rather than draw
254 // stale geometry.
255 if (mesh != nullptr) {
256 (void)gfx->releaseMesh(mesh);
257 mesh = nullptr;
258 }
259 return;
260 }
261 const int vertexCount = static_cast<int>(data.vertexCount());
262 const int indexCount = static_cast<int>(data.indices().size());
263 if (mesh != nullptr &&
264 !gfx->updateMeshVertices(mesh, data.positions().data(), data.normals().data(), data.uvs().data(),
265 vertexCount, data.indices().data(), indexCount)) {
266 // The backend rejected the in-place update; recreate rather than keep stale geometry.
267 (void)gfx->releaseMesh(mesh);
268 mesh = nullptr;
269 }
270 if (mesh == nullptr) {
271 mesh = gfx->newMeshFromArrays(data.positions().data(), data.normals().data(), data.uvs().data(),
272 vertexCount, data.indices().data(), indexCount);
273 }
274 if (mesh == nullptr) uploaded = false;
275 };
276
277 upload(terrain, sphereTerrain_);
278 upload(water, sphereWater_);
279 if (!uploaded) return invalidArgument("the graphics backend refused the hex sphere mesh upload");
280 return Result<void>::success();
281}
