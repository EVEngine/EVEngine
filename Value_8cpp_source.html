15Result<Value> copyJsonValue(const json::Value& source) {
16 if (source.isNull()) return Result<Value>::success(Value());
17 if (source.isBool()) return Result<Value>::success(Value(source.asBool()));
18 if (source.isString()) return Result<Value>::success(Value(source.asString()));
19
20 if (source.isNumber()) {
21 if (source.isIntegerLiteral()) {
22 if (!source.isInt64()) {
23 return Result<Value>::failure(
24 Diagnostic::error(DiagnosticCode::ParseError, "JSON integer is outside the canonical Int64 range"));
25 }
26 return Result<Value>::success(Value(source.asInt64()));
27 }
28 return Result<Value>::success(Value(source.asDouble()));
29 }
30
31 if (source.isArray()) {
32 Value::Array result;
33 result.reserve(source.size());
34 for (std::size_t index = 0; index < source.size(); ++index) {
35 auto element = copyJsonValue(source.at(index));
36 if (!element.ok()) return Result<Value>::failure(element.status());
37 result.push_back(std::move(element).takeValue());
38 }
39 return Result<Value>::success(Value(std::move(result)));
40 }
41
42 if (source.isObject()) {
43 Value::Object result;
44 for (const std::string& key : source.keys()) {
45 auto member = copyJsonValue(source.get(key.c_str()));
46 if (!member.ok()) return Result<Value>::failure(member.status());
47 result.emplace(key, std::move(member).takeValue());
48 }
49 return Result<Value>::success(Value(std::move(result)));
50 }
51
52 return Result<Value>::failure(Diagnostic::error(DiagnosticCode::ParseError, "unsupported JSON value kind"));
53}

57Result<Value> Value::fromJson(std::string_view jsonText) {
58 const std::string input(jsonText);
59 std::string error;

61 if (!document.valid()) {
