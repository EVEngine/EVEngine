81Result<float> readFiniteFloatValue(const Value& value) {
82 double number = 0.0;
83 if (const auto* integer = value.getIf<std::int64_t>())
84 number = static_cast<double>(*integer);
85 else if (const auto* real = value.getIf<double>())
86 number = *real;
87 else
88 return Result<float>::failure(Diagnostic::error(DiagnosticCode::TypeMismatch, "expected Integer or Double"));
89 const double maximum = static_cast<double>(std::numeric_limits<float>::max());
90 if (!std::isfinite(number) || number < -maximum || number > maximum)
91 return Result<float>::failure(
92 Diagnostic::error(DiagnosticCode::InvalidArgument, "number exceeds finite Float range"));
93 return Result<float>::success(static_cast<float>(number));
94}

135Result<Value> copyJsonValue(const json::Value& source) {
136 if (source.isNull()) return Result<Value>::success(Value());
137 if (source.isBool()) return Result<Value>::success(Value(source.asBool()));
138 if (source.isString()) return Result<Value>::success(Value(source.asString()));
139
140 if (source.isNumber()) {
141 if (source.isIntegerLiteral()) {
142 if (!source.isInt64()) {
143 return Result<Value>::failure(
144 Diagnostic::error(DiagnosticCode::ParseError, "JSON integer is outside the canonical Int64 range"));
145 }
146 return Result<Value>::success(Value(source.asInt64()));
147 }
148 return Result<Value>::success(Value(source.asDouble()));
149 }
150
151 if (source.isArray()) {
152 Value::Array result;
153 result.reserve(source.size());
154 for (std::size_t index = 0; index < source.size(); ++index) {
155 auto element = copyJsonValue(source.at(index));
156 if (!element.ok()) return Result<Value>::failure(element.status());
157 result.push_back(std::move(element).takeValue());
158 }
159 return Result<Value>::success(Value(std::move(result)));
160 }
161
162 if (source.isObject()) {
163 Value::Object result;
164 for (const std::string& key : source.keys()) {
165 auto member = copyJsonValue(source.get(key.c_str()));
166 if (!member.ok()) return Result<Value>::failure(member.status());
167 result.emplace(key, std::move(member).takeValue());
168 }
169 return Result<Value>::success(Value(std::move(result)));
170 }
171
172 return Result<Value>::failure(Diagnostic::error(DiagnosticCode::ParseError, "unsupported JSON value kind"));
173}
