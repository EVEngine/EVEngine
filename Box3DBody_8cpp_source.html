35b3Vec3 checkedVector(float x, float y, float z, const char *operation, const char *parameter) {
36 if (!std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z))

41b3Quat normalizedQuaternion(float qx, float qy, float qz, float qw, const char *operation) {
42 requireFinite(qx, operation, "qx");
43 requireFinite(qy, operation, "qy");
44 requireFinite(qz, operation, "qz");
45 requireFinite(qw, operation, "qw");
46 const float lengthSquared = qx * qx + qy * qy + qz * qz + qw * qw;
47 if (lengthSquared <= 1e-16f) throw eve::Exception("%s: quaternion length must be > 0", operation);
48 const float inverseLength = 1.f / std::sqrt(lengthSquared);
49 return b3Quat{{qx * inverseLength, qy * inverseLength, qz * inverseLength}, qw * inverseLength};
50}

111 return checked("physics.body3d.setRotation", [&] {
112 b3Quat q;
113 q.v = b3Vec3{qx, qy, qz};
114 q.s = qw;

116 if (len > 1e-8f) {
117 q.v.x /= len;
118 q.v.y /= len;
119 q.v.z /= len;
120 q.s /= len;
121 } else {
122 q = b3Quat_identity;

125

467 eve::Result<void> setMassProperties(float mass, float centerX, float centerY, float centerZ, float inertiaXX,
468 float inertiaYY, float inertiaZZ, float inertiaXY, float inertiaXZ,
469 float inertiaYZ) override {
470 return checked("physics.body3d.setMassProperties", [&] {
471 constexpr const char *operation = "Body3D.setMassProperties";
472 if (b3Body_GetType(bodyId_) != b3_dynamicBody) throw eve::Exception("%s: body must be dynamic", operation);
473 requireFinite(mass, operation, "mass");
474 requireFinite(centerX, operation, "centerX");
475 requireFinite(centerY, operation, "centerY");
476 requireFinite(centerZ, operation, "centerZ");
477 requireFinite(inertiaXX, operation, "inertiaXX");
478 requireFinite(inertiaYY, operation, "inertiaYY");
479 requireFinite(inertiaZZ, operation, "inertiaZZ");
480 requireFinite(inertiaXY, operation, "inertiaXY");
481 requireFinite(inertiaXZ, operation, "inertiaXZ");
482 requireFinite(inertiaYZ, operation, "inertiaYZ");
483 if (!(mass > 0.f)) throw eve::Exception("%s: mass must be > 0", operation);
484
485 // Sylvester's criterion for a symmetric positive-definite 3x3 tensor.
486 const double xx = inertiaXX, yy = inertiaYY, zz = inertiaZZ;
487 const double xy = inertiaXY, xz = inertiaXZ, yz = inertiaYZ;
488 const double minor2 = xx * yy - xy * xy;
489 const double determinant = xx * (yy * zz - yz * yz) - xy * (xy * zz - yz * xz) + xz * (xy * yz - yy * xz);
490 if (!(xx > 0.0 && minor2 > 0.0 && determinant > 0.0) || !std::isfinite(minor2) ||
491 !std::isfinite(determinant))
492 throw eve::Exception("%s: inertia tensor must be positive definite", operation);
493
494 b3MassData data{};
495 data.mass = mass;
496 data.center = {centerX, centerY, centerZ};
497 data.inertia = {{inertiaXX, inertiaXY, inertiaXZ},
498 {inertiaXY, inertiaYY, inertiaYZ},
499 {inertiaXZ, inertiaYZ, inertiaZZ}};
500 b3Body_SetMassData(bodyId_, data);
501
502 return eve::Result<void>::success();
503 });
504 }
