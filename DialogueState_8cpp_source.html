14std::optional<T> queryProviders(const std::string& subject, eve::IStateQuery* explicitProvider, Query&& query) {
15 if (explicitProvider) {
16 if (auto result = query(explicitProvider, subject); result.has_value()) return result;
17 }
18
19 eve::IStateQuery* service = eve::cap::query<eve::IStateQuery>();
20 if (service && service != explicitProvider) {
21 if (auto result = query(service, subject); result.has_value()) return result;
22 }
23
24 std::optional<T> result;
25 eve::cap::forEach<eve::IStateQuery>([&](eve::IStateQuery* provider) {
26 if (result.has_value() || provider == explicitProvider || provider == service) return;
27 result = query(provider, subject);
28 });
29 return result;
30}

72eve::Result<eve::decision::Condition> compileCondition(const eve::Value& specification) {
73 if (specification.isArray()) {
74 std::vector<eve::decision::Condition> children;
75 children.reserve(specification.arraySize());
76 for (std::size_t i = 0; i < specification.arraySize(); ++i) {
77 auto child = compileCondition(specification.at(i));
78 if (!child) return eve::Result<eve::decision::Condition>::failure(child.status());
79 children.push_back(std::move(child).takeValue());
80 }
81 return eve::Result<eve::decision::Condition>::success(eve::decision::Condition::all(std::move(children)));
82 }
83 if (!specification.isObject()) return conditionError("dialogue condition must be an object or array");
84
85 if (const eve::Value* all = specification.find("all")) {
86 if (!all->isArray()) return conditionError("condition 'all' must be an array", "all");
87 std::vector<eve::decision::Condition> children;
88 children.reserve(all->arraySize());
89 for (std::size_t i = 0; i < all->arraySize(); ++i) {
90 auto child = compileCondition(all->at(i));
91 if (!child) return eve::Result<eve::decision::Condition>::failure(child.status());
92 children.push_back(std::move(child).takeValue());
93 }
94 return eve::Result<eve::decision::Condition>::success(eve::decision::Condition::all(std::move(children)));
95 }
96 if (const eve::Value* any = specification.find("any")) {
97 if (!any->isArray()) return conditionError("condition 'any' must be an array", "any");
98 std::vector<eve::decision::Condition> children;
99 children.reserve(any->arraySize());
100 for (std::size_t i = 0; i < any->arraySize(); ++i) {
101 auto child = compileCondition(any->at(i));
102 if (!child) return eve::Result<eve::decision::Condition>::failure(child.status());
103 children.push_back(std::move(child).takeValue());
104 }
105 return eve::Result<eve::decision::Condition>::success(eve::decision::Condition::any(std::move(children)));
106 }
107 if (const eve::Value* negated = specification.find("not")) {
108 auto child = compileCondition(*negated);
109 if (!child) return eve::Result<eve::decision::Condition>::failure(child.status());
110 return eve::Result<eve::decision::Condition>::success(
111 eve::decision::Condition::not_(std::move(child).takeValue()));
112 }
113
114 const eve::Value* tag = specification.find("tag");
115 if (!tag) tag = specification.find("hasTag");
116 if (tag) {
117 if (!tag->isString()) return conditionError("tag condition requires a string tag", "tag");
118 return eve::Result<eve::decision::Condition>::success(eve::decision::Condition::hasTag(tag->asString()));
119 }
120
121 const eve::Value* authority = specification.find("authority");
122 if (authority) {
123 if (!authority->isString()) return conditionError("authority condition requires a string scope", "authority");
124 return eve::Result<eve::decision::Condition>::success(
125 eve::decision::Condition::authorityCheck(authority->asString()));
126 }
127
128 const eve::Value* policy = specification.find("policy");
129 if (!policy) policy = specification.find("script");
130 if (policy) {
131 if (!policy->isString()) return conditionError("policy condition requires a string name", "policy");
132 eve::Value arguments = eve::Value::Object{};
133 if (const eve::Value* supplied = specification.find("arguments")) arguments = *supplied;
134 return eve::Result<eve::decision::Condition>::success(
135 eve::decision::Condition::policyCall(policy->asString(), std::move(arguments)));
136 }
137
138 const eve::Value* state = specification.find("state");
139 if (state && state->isString()) {
140 const eve::Value* expected = specification.find("equals");
141 if (!expected) expected = specification.find("value");
142 if (!expected)
143 return eve::Result<eve::decision::Condition>::success(
144 eve::decision::Condition::hasAttribute(state->asString()));
145 return eve::Result<eve::decision::Condition>::success(
146 eve::decision::Condition::stateEquals(state->asString(), *expected));
147 }
148
149 const eve::Value* attribute = specification.find("attribute");
150 const eve::Value* nameValue = specification.find("key");
151 if (!nameValue) nameValue = specification.find("var");
152 if (attribute && attribute->isString()) nameValue = attribute;
153
154 const eve::Value* operation = specification.find("op");
155 if (nameValue && operation) {
156 if (!nameValue->isString() || !operation->isString())
157 return conditionError("comparison condition requires string key and op");
158 const std::string key = nameValue->asString();
159 const std::string op = operation->asString();
160 if (op == "has")
161 return eve::Result<eve::decision::Condition>::success(eve::decision::Condition::hasAttribute(key));
162 if (op == "missing")
163 return eve::Result<eve::decision::Condition>::success(
164 eve::decision::Condition::not_(eve::decision::Condition::hasAttribute(key)));
165 eve::decision::CompareOperator parsed;
166 if (!readOperator(op, parsed)) return conditionError("unknown condition operator: " + op, "op");
167 const eve::Value* expected = specification.find("value");
168 if (!expected) return conditionError("comparison condition is missing value", "value");
169 return eve::Result<eve::decision::Condition>::success(
170 eve::decision::Condition::compare(key, parsed, *expected));
171 }
172
173 if (attribute && attribute->isString())
174 return eve::Result<eve::decision::Condition>::success(
175 eve::decision::Condition::hasAttribute(attribute->asString()));
176
177 if (state && state->isObject()) {
178 std::string key;
179 if (!readString(*state, "key", key)) return conditionError("state condition is missing key", "state.key");
180 const eve::Value* expected = state->find("equals");
181 if (!expected) expected = state->find("value");
182 if (!expected) return conditionError("state condition is missing value", "state.value");
183 return eve::Result<eve::decision::Condition>::success(
184 eve::decision::Condition::stateEquals(std::move(key), *expected));
185 }
186
187 if (const eve::Value* compare = specification.find("compare")) {
188 if (!compare->isObject()) return conditionError("compare condition must be an object", "compare");
189 return compileCondition(*compare);
190 }
191
192 return conditionError("unrecognized dialogue condition");
193}

203eve::Value toCanonicalValue(const eve::StateValue& value) {
204 switch (value.kind()) {
205 case eve::StateValue::Kind::Null: return eve::Value::null();
206 case eve::StateValue::Kind::Int: return eve::Value::integer(value.asInt());
207 case eve::StateValue::Kind::Float: return eve::Value::number(value.asDouble());
208 case eve::StateValue::Kind::Bool: return eve::Value::boolean(value.asBool());
209 case eve::StateValue::Kind::String: return eve::Value::string(value.asString());
210 case eve::StateValue::Kind::Array: {
211 eve::Value::Array result;
212 result.reserve(value.arraySize());
213 for (std::size_t i = 0; i < value.arraySize(); ++i) result.push_back(toCanonicalValue(value.at(i)));
214 return eve::Value::array(std::move(result));
215 }
216 case eve::StateValue::Kind::Object: {
217 eve::Value::Object result;
218 for (const auto& key : value.keys()) result.emplace(key, toCanonicalValue(*value.find(key)));
219 return eve::Value::object(std::move(result));
220 }
221 }
222 return eve::Value::null();
223}

225eve::StateValue toDialogueStateValue(const eve::Value& value) {
226 switch (value.type()) {
227 case eve::Value::Type::Null: return eve::StateValue::null();
228 case eve::Value::Type::Int64: return eve::StateValue::integer(value.asInt());
229 case eve::Value::Type::Double: return eve::StateValue::number(value.asDouble());
230 case eve::Value::Type::Bool: return eve::StateValue::boolean(value.asBool());
231 case eve::Value::Type::String: return eve::StateValue::string(value.asString());
232 case eve::Value::Type::Array: {
233 eve::StateValue result = eve::StateValue::array();
234 for (std::size_t i = 0; i < value.arraySize(); ++i) result.pushBack(toDialogueStateValue(value.at(i)));
235 return result;
236 }
237 case eve::Value::Type::Object: {
238 eve::StateValue result = eve::StateValue::object();
239 for (const auto& key : value.keys()) result.set(key, toDialogueStateValue(*value.find(key)));
240 return result;
241 }
242 }
243 return eve::StateValue::null();
244}

341eve::decision::ConditionResult DialogueStateContext::evaluate(const eve::Value& specification) const {
342 auto compiled = compileCondition(specification);
343 if (!compiled) {
344 const eve::Diagnostic* diagnostic = compiled.error();
345 const std::string message = diagnostic ? diagnostic->message() : "invalid dialogue condition";
346 return eve::decision::ConditionResult::failed(eve::decision::ConditionReasonCode::InvalidCondition,
347 eve::Value(), eve::Value::Object{{"error", eve::Value(message)}});
348 }
349 eve::decision::Condition condition = std::move(compiled).takeValue();
350 return evaluate(condition);
