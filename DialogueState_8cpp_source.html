16 if (explicitProvider) {
17 if (auto result = query(explicitProvider, subject); result.has_value()) return result;
18 }

77eve::Result<eve::condition::Condition> compileCondition(const eve::Value& specification) {
78 if (specification.isArray()) {
79 std::vector<eve::condition::Condition> children;
80 children.reserve(specification.arraySize());
81 for (std::size_t i = 0; i < specification.arraySize(); ++i) {
82 auto child = compileCondition(specification.at(i));
83 if (!child) return eve::Result<eve::condition::Condition>::failure(child.status());
84 children.push_back(std::move(child).takeValue());
85 }
86 return eve::Result<eve::condition::Condition>::success(eve::condition::Condition::all(std::move(children)));
87 }
88 if (!specification.isObject()) return conditionError("dialogue condition must be an object or array");
89
90 if (const eve::Value* all = specification.find("all")) {
91 if (!all->isArray()) return conditionError("condition 'all' must be an array", "all");
92 std::vector<eve::condition::Condition> children;
93 children.reserve(all->arraySize());
94 for (std::size_t i = 0; i < all->arraySize(); ++i) {
95 auto child = compileCondition(all->at(i));
96 if (!child) return eve::Result<eve::condition::Condition>::failure(child.status());
97 children.push_back(std::move(child).takeValue());
98 }
99 return eve::Result<eve::condition::Condition>::success(eve::condition::Condition::all(std::move(children)));
100 }
101 if (const eve::Value* any = specification.find("any")) {
102 if (!any->isArray()) return conditionError("condition 'any' must be an array", "any");
103 std::vector<eve::condition::Condition> children;
104 children.reserve(any->arraySize());
105 for (std::size_t i = 0; i < any->arraySize(); ++i) {
106 auto child = compileCondition(any->at(i));
107 if (!child) return eve::Result<eve::condition::Condition>::failure(child.status());
108 children.push_back(std::move(child).takeValue());
109 }
110 return eve::Result<eve::condition::Condition>::success(eve::condition::Condition::any(std::move(children)));
111 }
112 if (const eve::Value* negated = specification.find("not")) {
113 auto child = compileCondition(*negated);
114 if (!child) return eve::Result<eve::condition::Condition>::failure(child.status());
115 return eve::Result<eve::condition::Condition>::success(
116 eve::condition::Condition::not_(std::move(child).takeValue()));
117 }
118
119 const eve::Value* tag = specification.find("tag");
120 if (!tag) tag = specification.find("hasTag");
121 if (tag) {
122 if (!tag->isString()) return conditionError("tag condition requires a string tag", "tag");
123 return eve::Result<eve::condition::Condition>::success(eve::condition::tagPresent(tag->asString()));
124 }
125
126 const eve::Value* authority = specification.find("authority");
127 if (authority) {
128 if (!authority->isString()) return conditionError("authority condition requires a string scope", "authority");
129 return eve::Result<eve::condition::Condition>::success(eve::condition::authorityAllow(authority->asString()));
130 }
131
132 const eve::Value* policy = specification.find("policy");
133 if (!policy) policy = specification.find("script");
134 if (policy) {
135 if (!policy->isString()) return conditionError("policy condition requires a string name", "policy");
136 eve::Value arguments = eve::Value::Object{};
137 if (const eve::Value* supplied = specification.find("arguments")) arguments = *supplied;
138 return eve::Result<eve::condition::Condition>::success(
139 eve::condition::Condition::policyCall(policy->asString(), std::move(arguments)));
140 }
141
142 const eve::Value* state = specification.find("state");
143 if (state && state->isString()) {
144 const eve::Value* expected = specification.find("equals");
145 if (!expected) expected = specification.find("value");
146 if (!expected)
147 return eve::Result<eve::condition::Condition>::success(eve::condition::attributeExists(state->asString()));
148 return eve::Result<eve::condition::Condition>::success(
149 eve::condition::stateEquals(state->asString(), *expected));
150 }
151
152 const eve::Value* attribute = specification.find("attribute");
153 const eve::Value* nameValue = specification.find("key");
154 if (!nameValue) nameValue = specification.find("var");
155 if (attribute && attribute->isString()) nameValue = attribute;
156
157 const eve::Value* operation = specification.find("op");
158 if (nameValue && operation) {
159 if (!nameValue->isString() || !operation->isString())
160 return conditionError("comparison condition requires string key and op");
161 const std::string key = nameValue->asString();
162 const std::string op = operation->asString();
163 if (op == "has") return eve::Result<eve::condition::Condition>::success(eve::condition::attributeExists(key));
164 if (op == "missing")
165 return eve::Result<eve::condition::Condition>::success(
166 eve::condition::Condition::not_(eve::condition::attributeExists(key)));
167 eve::condition::CompareOperator parsed;
168 if (!readOperator(op, parsed)) return conditionError("unknown condition operator: " + op, "op");
169 const eve::Value* expected = specification.find("value");
170 if (!expected) return conditionError("comparison condition is missing value", "value");
171 return eve::Result<eve::condition::Condition>::success(
172 eve::condition::Condition::compare(key, parsed, *expected));
173 }
174
175 if (attribute && attribute->isString())
176 return eve::Result<eve::condition::Condition>::success(eve::condition::attributeExists(attribute->asString()));
177
178 if (state && state->isObject()) {
179 std::string key;
180 if (!readString(*state, "key", key)) return conditionError("state condition is missing key", "state.key");
181 const eve::Value* expected = state->find("equals");
182 if (!expected) expected = state->find("value");
183 if (!expected) return conditionError("state condition is missing value", "state.value");
184 return eve::Result<eve::condition::Condition>::success(eve::condition::stateEquals(std::move(key), *expected));
185 }
186
187 if (const eve::Value* compare = specification.find("compare")) {
188 if (!compare->isObject()) return conditionError("compare condition must be an object", "compare");
189 return compileCondition(*compare);
190 }
191
192 return conditionError("unrecognized dialogue condition");
193}

203eve::Value toCanonicalValue(const eve::StateValue& value) {
204 switch (value.kind()) {
205 case eve::StateValue::Kind::Null: return eve::Value::null();
206 case eve::StateValue::Kind::Int: return eve::Value::integer(value.asInt());
207 case eve::StateValue::Kind::Float: return eve::Value::number(value.asDouble());
208 case eve::StateValue::Kind::Bool: return eve::Value::boolean(value.asBool());
209 case eve::StateValue::Kind::String: return eve::Value::string(value.asString());
210 case eve::StateValue::Kind::Array: {
211 eve::Value::Array result;
212 result.reserve(value.arraySize());
213 for (std::size_t i = 0; i < value.arraySize(); ++i) result.push_back(toCanonicalValue(value.at(i)));
214 return eve::Value::array(std::move(result));
215 }
216 case eve::StateValue::Kind::Object: {
217 eve::Value::Object result;
218 for (const auto& key : value.keys()) result.emplace(key, toCanonicalValue(*value.find(key)));
219 return eve::Value::object(std::move(result));
220 }
221 }
222 return eve::Value::null();
223}

225eve::StateValue toDialogueStateValue(const eve::Value& value) {
226 switch (value.type()) {
227 case eve::Value::Type::Null: return eve::StateValue::null();
228 case eve::Value::Type::Int64: return eve::StateValue::integer(value.asInt());
229 case eve::Value::Type::Double: return eve::StateValue::number(value.asDouble());
230 case eve::Value::Type::Bool: return eve::StateValue::boolean(value.asBool());
231 case eve::Value::Type::String: return eve::StateValue::string(value.asString());
232 case eve::Value::Type::Array: {
233 eve::StateValue result = eve::StateValue::array();
234 for (std::size_t i = 0; i < value.arraySize(); ++i) result.pushBack(toDialogueStateValue(value.at(i)));
235 return result;
236 }
237 case eve::Value::Type::Object: {
238 eve::StateValue result = eve::StateValue::object();
239 for (const auto& key : value.keys()) result.set(key, toDialogueStateValue(*value.find(key)));
240 return result;
241 }
242 }
243 return eve::StateValue::null();
244}

251std::optional<eve::Value> StateEvaluationContext::value(std::string_view key) const {
252 if (auto tag = eve::condition::stripPrefix(key, eve::condition::kTagPrefix)) {
253 auto result = queryProviders<bool>(
254 subject_, provider_,
255 [&](eve::IStateQuery* provider, const std::string& subject) { return provider->hasTag(subject, *tag); });
256 return result.has_value() ? std::optional<eve::Value>(eve::Value(*result)) : std::nullopt;
257 }
258 if (auto attr = eve::condition::stripPrefix(key, eve::condition::kAttrPrefix)) {
259 auto result = queryProviders<eve::Value>(subject_, provider_,
260 [&](eve::IStateQuery* provider, const std::string& subject) {
261 return provider->attribute(subject, *attr);
262 });
263 if (result.has_value()) return result;
264 return queryProviders<eve::Value>(
265 subject_, provider_,
266 [&](eve::IStateQuery* provider, const std::string& subject) { return provider->value(subject, *attr); });
267 }
268 if (auto resource = eve::condition::stripPrefix(key, eve::condition::kResourcePrefix)) {
269 return queryProviders<eve::Value>(subject_, provider_,
270 [&](eve::IStateQuery* provider, const std::string& subject) {
271 return provider->resource(subject, *resource);
272 });
273 }
274 if (auto state = eve::condition::stripPrefix(key, eve::condition::kStatePrefix)) {
275 auto result = queryProviders<eve::Value>(
276 subject_, provider_,
277 [&](eve::IStateQuery* provider, const std::string& subject) { return provider->state(subject, *state); });
278 if (result.has_value()) return result;
279 return queryProviders<eve::Value>(
280 subject_, provider_,
281 [&](eve::IStateQuery* provider, const std::string& subject) { return provider->value(subject, *state); });
282 }
283 return queryProviders<eve::Value>(subject_, provider_, [&](eve::IStateQuery* provider, const std::string& subject) {
