21Result<void> LifeFieldMaterialBinding::syncFrom(const SurfaceField& field) {
22 if (field.resolution < 2 || field.lifeField.size() != static_cast<size_t>(field.resolution) * field.resolution) {
23 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
24 "surface field is not initialized for life binding", "field", {},
25 "gpuagents"));
26 }
27 const int res = field.resolution;
28 uniforms_.origin = field.origin;
29 uniforms_.worldSize = field.worldSize;
30 uniforms_.resolution = res;
31
32 lifePixels_.assign(static_cast<size_t>(res) * res * 4u, 0);
33 surfacePixels_.assign(static_cast<size_t>(res) * res * 4u, 0);
34
35 float minH = field.height.empty() ? 0.f : field.height[0];
36 float maxH = minH;
37 for (float h : field.height) {
38 minH = std::min(minH, h);
39 maxH = std::max(maxH, h);
40 }
41
42 for (int z = 0; z < res; ++z) {
43 for (int x = 0; x < res; ++x) {
44 const size_t i = static_cast<size_t>(x + res * z);
45 const auto& life = field.lifeField[i];
46 const size_t p = i * 4u;
47 lifePixels_[p + 0] = encode01(life.r);
48 lifePixels_[p + 1] = encode01(life.g);
49 lifePixels_[p + 2] = encode01(life.b);
50 lifePixels_[p + 3] = encode01(life.a);
51
52 const float h = field.height.empty() ? 0.f : field.height[i];
53 const glm::vec3 n = field.normals.empty() ? glm::vec3(0.f, 1.f, 0.f) : field.normals[i];
54 surfacePixels_[p + 0] = encode01(height01(h, minH, maxH));
55 surfacePixels_[p + 1] = encode01(n.x * 0.5f + 0.5f);
56 surfacePixels_[p + 2] = encode01(n.y * 0.5f + 0.5f);
57 surfacePixels_[p + 3] = encode01(n.z * 0.5f + 0.5f);
58 }
59 }
