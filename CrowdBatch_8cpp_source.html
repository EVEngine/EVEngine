70Result<AgentBatchReceipt> Crowd::removeAgents(const AgentRemovalBatch& batch) {
71 std::vector<std::uint64_t> identities;
72 try {
73 identities.reserve(batch.agents.size());
74 for (const auto& reference : batch.agents) {
75 if (!isAgentAlive(reference)) return failed(DiagnosticCode::NotFound, "Stale or wrong-owner removal");
76 identities.push_back(reference.handle.packed());
77 }
78 } catch (const std::bad_alloc&) {
79 return failed(DiagnosticCode::Failed, "Cannot allocate removal validation workspace");
80 }
81 if (duplicates(identities)) return failed(DiagnosticCode::Conflict, "Duplicate agent removal");
82 // All allocation and reference validation completed. Re-resolve after every swap-pop.
83 for (const auto& reference : batch.agents) {
84 const bool removed = removeAgent(impl_->resolveAgent(reference));
85 EV_ASSERT(removed, "Validated Crowd removal must succeed without reentry");
86 }
87 return Result<AgentBatchReceipt>::success({batch.agents.size()});
88}
