39Graph buildGraph() {
40 Graph graph;
41 for (const auto& contract : eve::script::generatedBindingContracts().snapshot()) {
42 const auto api = "api:" + contract.key();
43 const auto binding = "binding:" + contract.key();
44 graph.node(api, "api", contract.key(), {}, contract.architecture ? "mapped" : "unknown-owner");
45 graph.node(binding, "binding", contract.key(), contract.source);
46 graph.edge(api, binding, "binds", "static-binding");
47 if (contract.cppSymbol.empty()) continue;
48 const auto implementation = "implementation:" + contract.module + ":" + contract.cppSymbol;
49 graph.node(implementation, "implementation", contract.cppSymbol,
50 contract.architecture ? contract.architecture->implementationSource : std::string{});
51 graph.edge(binding, implementation, "calls", "static-direct-binding");
52 if (!contract.architecture) continue;
53 const auto& mapping = *contract.architecture;
54 const auto owner = "owner:" + mapping.stateOwner;
55 graph.node(owner, "owner", mapping.stateOwner, mapping.ownerSource);
56 graph.edge(implementation, owner, "state_owner", "reviewed-owner", mapping.inputHash);
57 const auto addAccess = [&](const std::vector<std::string>& fields, const std::string& access) {
58 for (const auto& field : fields) {
59 const auto state = "state:" + field;
60 graph.node(state, "state", field);
61 graph.edge(owner, state, "owns", "reviewed-owner", mapping.inputHash);
62 graph.edge(implementation, state, access, "reviewed-access", mapping.inputHash);
63 }
64 };
65 addAccess(mapping.writes, "writes");
66 addAccess(mapping.reads, "reads");
67 for (const auto& test : mapping.tests) {
68 const auto id = "test:" + test;
69 graph.node(id, "test", test, test);
70 graph.edge(implementation, id, "tested_by", "test-reference", mapping.inputHash);
71 }
72 }
73 return graph;
74}

84Object::Ptr query(std::string_view mode, Object::Ptr args) {
85 const auto seedArgument = args ? args->optValue<std::string>("seed", "") : std::string{};
86 const auto seed = seedArgument.find(':') == std::string::npos ? "api:" + seedArgument : seedArgument;
87 const int limit = args ? args->optValue<int>("maxNodes", 64) : 64;
88 const int depth = args ? args->optValue<int>("maxDepth", 6) : 6;
89 const int bytes = args ? args->optValue<int>("maxBytes", 16384) : 16384;
90 if (seedArgument.empty() || limit < 1 || limit > 256 || depth < 0 || depth > 8 || bytes < 1024 || bytes > 65536 ||
91 (mode == "eve_task_context" && (!args || args->optValue<std::string>("task", "").empty())))
92 return mcpFailureObject("invalid_query", "seed required; maxNodes 1..256, maxDepth 0..8, maxBytes 1024..65536");
93 // Generated contracts are immutable for this linked process. Retain owning
94 // evidence once; hot reload does not replace this build-generated registry.
95 static const Graph graph = buildGraph();
96 if (!graph.nodes.contains(seed))
97 return mcpFailureObject("unknown_seed", "Use api: followed by a binding key or a returned node ID");
98 const auto expectedHash = args->optValue<std::string>("expectedInputHash", "");
99 if (!expectedHash.empty()) {
100 const auto contract = eve::script::generatedBindingContracts().find(graph.nodes.at(seed).label);
101 if (!contract || !contract->architecture || contract->architecture->inputHash != expectedHash)
102 return mcpFailureObject("stale_evidence",
103 "Refresh the seed contract fingerprint before using this task context");
104 }
105 std::set<std::string> selected{seed};
106 std::queue<std::pair<std::string, int>> pending;
107 pending.emplace(seed, 0);
108 bool truncated = false;
109 while (!pending.empty()) {
110 const auto [current, level] = pending.front();
111 pending.pop();
112 for (const auto index : graph.adjacency.at(current)) {
113 const auto& edge = graph.edges[index];
114 const auto next = edge.from == current ? edge.to : edge.from;
115 if (selected.contains(next)) continue;
116 // Impact flows from writers to field consumers; owner/test nodes
117 // supply context but cannot fan out into unrelated state.
118 bool terminal = false;
119 if (mode != "eve_architecture_query") {
120 const auto& kind = graph.nodes.at(current).kind;
121 if (kind == "owner" || kind == "test") continue;
122 if (edge.relation == "reads" && edge.from == current) {
123 if (mode == "eve_architecture_impact") continue;
124 terminal = true;
125 }
126 }
127 if (level >= depth || selected.size() >= static_cast<size_t>(limit)) {
128 truncated = true;
129 continue;
130 }
131 selected.insert(next);
132 if (!terminal) pending.emplace(next, level + 1);
133 }
134 }
135 Object::Ptr result = new Object();
136 result->set("ok", true);
137 result->set("schema", "eve.architecture-graph");
138 result->set("schemaVersion", 1);
139 result->set("seed", seed);
140 result->set("mode", std::string(mode));
141 result->set("coverage", "reviewed-subset");
142 result->set("limitations",
143 "Candidate static impact; no runtime causality, dynamic keys or complete call graph. Test references "
144 "do not prove tests passed.");
145 Array::Ptr nodes = new Array(), edges = new Array(), tests = new Array(), unknowns = new Array(),
146 fingerprints = new Array(), contracts = new Array();
147 std::set<std::string> seenHashes;
148 for (const auto& id : selected) {
149 const auto& node = graph.nodes.at(id);
150 nodes->add(nodeJson(node));
151 if (node.kind == "api") {
152 const auto* contract = eve::script::generatedBindingContracts().find(node.label);
153 if (contract) {
154 Poco::JSON::Parser parser;
155 contracts->add(parser.parse(contract->toJson()));
156 }
157 }
158 if (node.kind == "test") tests->add(node.label);
159 if (node.coverage == "unknown-owner") unknowns->add(id);
160 }
161 for (const auto& edge : graph.edges) {
162 if (!selected.contains(edge.from) || !selected.contains(edge.to)) continue;
163 Object::Ptr item = new Object();
164 item->set("from", edge.from);
165 item->set("to", edge.to);
166 item->set("relation", edge.relation);
167 item->set("evidence", edge.evidence);
168 item->set("inputHash", edge.fingerprint);
169 edges->add(item);
170 if (!edge.fingerprint.empty()) seenHashes.insert(edge.fingerprint);
171 }
172 for (const auto& hash : seenHashes) fingerprints->add(hash);
173 result->set("nodes", nodes);
174 result->set("edges", edges);
175 result->set("tests", tests);
176 result->set("unknowns", unknowns);
177 result->set("inputHashes", fingerprints);
178 result->set("contracts", contracts);
179 result->set("truncated", truncated);
180 if (mode == "eve_task_context") {
181 result->set("schema", "eve.task-context");
182 result->set("task", args->optValue<std::string>("task", ""));
183 result->set("consistency", "build-generated source evidence; not a live-world snapshot");
184 }
185 // Return a bounded, explicit failure rather than silently discarding evidence.
186 if (mcpStringify(result).size() > static_cast<size_t>(bytes))
187 return mcpFailureObject("context_budget_exceeded",
188 "Reduce maxNodes/maxDepth or increase maxBytes; no partial evidence was returned");
189 return result;
190}
