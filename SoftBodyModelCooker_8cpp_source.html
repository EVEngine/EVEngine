18 const SoftBodyModelDefinition& definition) {

22 definition.volumeSampling != SoftBodyVolumeSampling::None)
23 return failure<SoftBodyModel>(eve::DiagnosticCode::Unsupported,
24 "only vertex surface sampling without volume voxels is implemented",
25 "sampling");
26 if (mesh.positions.empty() || mesh.positions.size() % 3 != 0)
27 return failure<SoftBodyModel>(eve::DiagnosticCode::InvalidArgument,
28 "mesh positions must contain complete xyz vertices", "mesh.positions");
29 const std::size_t vertexCount = mesh.positions.size() / 3;
30 if (mesh.indices.empty() || mesh.indices.size() % 3 != 0)
31 return failure<SoftBodyModel>(eve::DiagnosticCode::InvalidArgument,
32 "mesh indices must contain complete triangles", "mesh.indices");
33
34 SoftBodyModel model;
35 model.sourceMesh = definition.sourceMesh;
36 model.particles.reserve(vertexCount);
37 model.surfaceBindings.reserve(vertexCount);
38 for (std::size_t index = 0; index < vertexCount; ++index) {
39 const float x = mesh.positions[index * 3];
40 const float y = mesh.positions[index * 3 + 1];
41 const float z = mesh.positions[index * 3 + 2];
42 if (!std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z))
43 return failure<SoftBodyModel>(eve::DiagnosticCode::InvalidArgument,
44 "mesh position must be finite", "mesh.positions");
45 model.particles.push_back({x, y, z});
46 model.surfaceBindings.push_back({static_cast<std::uint32_t>(index)});
47 }
48 model.surfaceIndices = mesh.indices;
49
50 std::vector<std::set<std::uint32_t>> neighborhoods(vertexCount);
51 for (std::size_t triangle = 0; triangle < mesh.indices.size(); triangle += 3) {
52 const auto a = mesh.indices[triangle];
53 const auto b = mesh.indices[triangle + 1];
54 const auto c = mesh.indices[triangle + 2];
55 if (a >= vertexCount || b >= vertexCount || c >= vertexCount)
56 return failure<SoftBodyModel>(eve::DiagnosticCode::InvalidArgument,
57 "mesh triangle index exceeds vertex count", "mesh.indices");
58 if (a == b || b == c || a == c)
59 return failure<SoftBodyModel>(eve::DiagnosticCode::InvalidArgument,
60 "mesh contains a degenerate index triangle", "mesh.indices");
61 for (const auto center : {a, b, c}) {
62 neighborhoods[center].insert(a);
63 neighborhoods[center].insert(b);
64 neighborhoods[center].insert(c);
65 }
66 }
67 for (const auto& neighborhood : neighborhoods) {
68 if (neighborhood.size() < 3) continue;
69 SoftBodyModelCluster cluster;
70 cluster.particleIndices.assign(neighborhood.begin(), neighborhood.end());
71 model.clusters.push_back(std::move(cluster));
72 }
73 auto modelValid = model.validate();
74 if (!modelValid) return eve::Result<SoftBodyModel>::failure(modelValid.status());
75 return eve::Result<SoftBodyModel>::success(std::move(model));
76}
