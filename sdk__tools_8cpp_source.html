55// 校验 zip 条目路径安全：拒绝绝对路径与 ".."/"." 组件（zip-slip 防护）。
56bool zipEntryPathSafe(const std::string& name) {
57 if (name.empty()) return false;
58 if (name[0] == '/' || name[0] == '\\') return false;
59 std::string comp;
60 for (const char c : name) {
61 if (c == '/' || c == '\\') {
62 if (comp == ".." || comp == ".") return false;
63 comp.clear();
64 } else {
65 comp += c;
66 }
67 }
68 return comp != ".." && comp != ".";
69}

71// 进程内 ZIP 解压（STORED + DEFLATE）：宿主 tar 不一定能读 zip（Linux 的
72// GNU tar 就不支持），而 SDK 安装不能要求用户机器预装 unzip/bsdtar。
73bool extractZip(const std::string& zipPath, const std::string& destDir) {
74 std::ifstream in(zipPath, std::ios::binary);
75 if (!in) return false;
76 const std::vector<char> data((std::istreambuf_iterator<char>(in)),
77 std::istreambuf_iterator<char>());
78 if (data.size() < 22) return false;
79
80 const auto le16 = [&](size_t off) -> uint16_t {
81 return static_cast<uint16_t>(static_cast<unsigned char>(data[off]) |
82 (static_cast<unsigned char>(data[off + 1]) << 8));
83 };
84 const auto le32 = [&](size_t off) -> uint32_t {
85 return static_cast<uint32_t>(static_cast<unsigned char>(data[off]) |
86 (static_cast<unsigned char>(data[off + 1]) << 8) |
87 (static_cast<unsigned char>(data[off + 2]) << 16) |
88 (static_cast<unsigned char>(data[off + 3]) << 24));
89 };
90
91 // EOCD 在文件末尾 64KB+22 字节内。
92 size_t eocd = std::string::npos;
93 const size_t searchStart = data.size() > 65557 ? data.size() - 65557 : 0;
94 for (size_t i = data.size() - 22;; --i) {
95 if (data[i] == 'P' && data[i + 1] == 'K' && data[i + 2] == 5 && data[i + 3] == 6) {
96 eocd = i;
97 break;
98 }
99 if (i == searchStart) break;
100 }
101 if (eocd == std::string::npos) return false;
102 const uint32_t cdOffset = le32(eocd + 16);
103 const uint32_t cdSize = le32(eocd + 12);
104 if (cdOffset > data.size() || cdSize > data.size() - cdOffset) return false;
105
106 size_t pos = cdOffset;
107 const size_t cdEnd = cdOffset + cdSize;
108 while (cdEnd - pos >= 46 && data[pos] == 'P' && data[pos + 1] == 'K' && data[pos + 2] == 1 && data[pos + 3] == 2) {
109 const size_t entryStart = pos;
110 const uint16_t method = le16(pos + 10);
111 const uint32_t entryCrc = le32(pos + 16);
112 const uint32_t compSz = le32(pos + 20);
113 const uint32_t uncompSz = le32(pos + 24);
114 const uint16_t nameLen = le16(pos + 28);
115 const uint16_t extraLen = le16(pos + 30);
116 const uint16_t commentLen = le16(pos + 32);
117 const uint32_t localOff = le32(pos + 42);
118 const size_t entrySize = static_cast<size_t>(46) + nameLen + extraLen + commentLen;
119 if (entrySize > cdEnd - pos) return false;
120 const std::string name(data.data() + entryStart + 46, nameLen);
121 // Read the central-directory attributes before advancing to the next entry.
122 // Reading them from `pos` after the increment was the vector overrun seen
123 // when installing the local release fixture.
124#if !defined(_WIN32)
125 const uint32_t extAttr = le32(entryStart + 38);
126#endif
127 pos += entrySize;
128
129 if (!zipEntryPathSafe(name)) return false;
130 const bool isDir = !name.empty() && name.back() == '/';
131 const auto dest = std::filesystem::path(destDir) / name;
132 std::error_code ec;
133 if (isDir) {
134 std::filesystem::create_directories(dest, ec);
135 continue;
136 }
137 if (localOff > data.size() || data.size() - localOff < 30 || data[localOff] != 'P' ||
138 data[localOff + 1] != 'K' || data[localOff + 2] != 3 || data[localOff + 3] != 4)
139 return false;
140 const size_t localHeaderSize = static_cast<size_t>(30) + le16(localOff + 26) + le16(localOff + 28);
141 if (localHeaderSize > data.size() - localOff) return false;
142 const size_t lData = localOff + localHeaderSize;
143 if (compSz > data.size() - lData || compSz > static_cast<uint32_t>(std::numeric_limits<uInt>::max()))
144 return false;
145
146 std::vector<char> out;
147 if (method == 0) {
148 out.assign(data.begin() + static_cast<ptrdiff_t>(lData),
149 data.begin() + static_cast<ptrdiff_t>(lData + compSz));
150 } else if (method == 8) {
151 z_stream zs = {};
152 if (inflateInit2(&zs, -15) != Z_OK) return false;
153 zs.next_in = reinterpret_cast<Bytef*>(const_cast<char*>(data.data() + lData));
154 zs.avail_in = static_cast<uInt>(compSz);
155 std::vector<char> buf(65536);
156 int ret;
157 do {
158 zs.next_out = reinterpret_cast<Bytef*>(buf.data());
159 zs.avail_out = static_cast<uInt>(buf.size());
160 ret = inflate(&zs, Z_NO_FLUSH);
161 if (ret != Z_OK && ret != Z_STREAM_END) {
162 inflateEnd(&zs);
163 return false;
164 }
165 out.insert(out.end(), buf.begin(), buf.end() - zs.avail_out);
166 } while (ret != Z_STREAM_END);
167 inflateEnd(&zs);
168 } else {
169 return false; // 不支持的方法（zip 里极罕见）。
170 }
171 if (out.size() != uncompSz) return false;
172 if (static_cast<uint32_t>(
173 crc32(0, reinterpret_cast<const Bytef*>(out.data()), static_cast<uInt>(out.size()))) != entryCrc)
174 return false;
175
176 std::filesystem::create_directories(dest.parent_path(), ec);
177 if (ec) return false;
178 std::ofstream ofs(dest, std::ios::binary | std::ios::trunc);
179 if (!ofs) return false;
180 ofs.write(out.data(), static_cast<std::streamsize>(out.size()));
181 if (!ofs) return false;
182#if !defined(_WIN32)
183 // Preserve the unix mode recorded in the zip's external attributes
184 // (bits 16-23) so extracted tools stay executable: gradle, sdkmanager,
185 // aapt/adb all ship as scripts/binaries with +x in their archives.
186 const auto mode = static_cast<unsigned>((extAttr >> 16) & 0xFFFF);
187 if (mode & 0111) {
188 std::error_code pec;
189 std::filesystem::permissions(dest,
190 std::filesystem::perms::owner_exec | std::filesystem::perms::group_exec |
191 std::filesystem::perms::others_exec,
192 std::filesystem::perm_options::add, pec);
193 }
194#endif
195 }
196 return pos == cdEnd;
197}

270std::string fileSha256(const std::string& path) {
271 std::error_code ec;
272 const auto tmp = std::filesystem::temp_directory_path(ec) / "eve-sha256.txt";
273 if (ec) return "";
274 const std::string cmd =
275#if defined(_WIN32)
276 "certutil -hashfile \"" + path + "\" SHA256 > \"" + tmp.string() + "\"";
277#else
278 "sha256sum \"" + path + "\" > \"" + tmp.string() + "\"";
279#endif
280 if (runShell(cmd) != 0) return "";
281 std::ifstream in(tmp);
282 std::string line, hex;
283 while (std::getline(in, line)) {
284 std::istringstream iss(line);
285 std::string tok;
286 while (iss >> tok) {
287 if (tok.size() != 64) continue;
288 bool ok = true;
289 for (const char c : tok) {
290 if (!((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f') ||
291 (c >= 'A' && c <= 'F'))) {
292 ok = false;
293 break;
294 }
295 }
296 if (ok) {
297 hex = lower(tok);
298 break;
299 }
300 }
301 if (!hex.empty()) break;
302 }
303 std::filesystem::remove(tmp, ec);
304 return hex;
305}

317std::string findSdkRoot(const std::string& sdkArg) {
318 std::error_code ec;
319 if (!sdkArg.empty()) {
320 const auto p = std::filesystem::absolute(sdkArg, ec);
321 if (!ec && std::filesystem::is_directory(p, ec)) return p.string();
322 return "";
323 }
324 const std::string env = getEnv("EVENGINE_SDK");
325 if (!env.empty() && std::filesystem::is_directory(env, ec))
326 return std::filesystem::absolute(env, ec).string();
327
328 // Running from <sdk>/bin/<runtime> -> root is two directories up.
329#if defined(_WIN32)
330 wchar_t buf[MAX_PATH + 1] = {0};
331 if (GetModuleFileNameW(nullptr, buf, MAX_PATH) != 0) {
332 const auto exeDir = std::filesystem::path(buf).parent_path();
333 if (isSdkRoot(exeDir.parent_path())) return exeDir.parent_path().string();
334 }
335#else
336 char buf[4096] = {0};
337 const ssize_t n = readlink("/proc/self/exe", buf, sizeof(buf) - 1);
338 if (n > 0) {
339 const auto exeDir = std::filesystem::path(std::string(buf, static_cast<size_t>(n))).parent_path();
340 if (isSdkRoot(exeDir.parent_path())) return exeDir.parent_path().string();
341 }
342#endif
343
344 const auto cwd = std::filesystem::current_path(ec);
345 if (!ec && isSdkRoot(cwd)) return cwd.string();
346
347 // `eve get <platform>` 安装的 SDK（<installRoot>/<platform>）。优先返回
348 // android（build 目前唯一支持的平台），其次任意 SDK 根。
349 const auto installRoot = std::filesystem::path(eveSdkInstallRoot());
350 std::string fallback;
351 if (std::filesystem::is_directory(installRoot, ec)) {
352 std::filesystem::directory_iterator it(installRoot, ec), end;
353 for (; it != end; it.increment(ec)) {
354 if (ec) {
355 ec.clear();
356 continue;
357 }
358 if (!it->is_directory(ec) || !isSdkRoot(it->path())) continue;
359 const std::string plat = sdkTargetPlatform(it->path().string());
360 if (plat == "android") return it->path().string();
361 if (fallback.empty()) fallback = it->path().string();
362 }
363 }
364 return fallback;
365}

369// 从 SHA256SUMS 文本里找 "<hex> <name>" 行，返回小写 hex；找不到返回空串。
370std::string sha256ForName(const std::string& sums, const std::string& name) {
371 std::istringstream iss(sums);
372 std::string line;
373 while (std::getline(iss, line)) {
374 const size_t sp = line.find(' ');
375 if (sp == std::string::npos) continue;
376 const std::string hex = line.substr(0, sp);
377 std::string rest = line.substr(sp + 1);
378 while (!rest.empty() && (rest.front() == ' ' || rest.front() == '\t'))
379 rest.erase(rest.begin());
380 while (!rest.empty() &&
381 (rest.back() == '\r' || rest.back() == ' ' || rest.back() == '\t'))
382 rest.pop_back();
383 if (rest == name) return lower(hex);
384 }
385 return "";
386}

402eve::Result<void> installEveSdk(Platform p) {
403 namespace fs = std::filesystem;
404 if (p == Platform::Unknown) return sdkFailure(eve::DiagnosticCode::InvalidArgument, "unknown SDK platform");
405 const std::string plat = platformName(p);
406 const std::string tag = sdkVersionTag();
407 if (tag.empty() || tag[0] != 'v') {
408 return sdkFailure(eve::DiagnosticCode::InvalidArgument,
409 "cannot determine the current EVEngine version; set EVE_SDK_TAG", "EVE_SDK_TAG");
410 }
411 const std::string expectedVer = tag.substr(1);
412 const std::string installRoot = eveSdkInstallRoot();
413 const std::string destRoot = (fs::path(installRoot) / plat).string();
414 std::error_code ec;
415
416 // 已安装同版本则直接复用，避免重复下载。
417 const std::string currentVer = sdkVersion(destRoot);
418 if (!currentVer.empty() && currentVer == expectedVer) {
419 std::cout << "eve get: EVEngine " << plat << " SDK already installed at "
420 << destRoot << " (" << tag << ")" << std::endl;
421 return eve::Result<void>::success();
422 }
423
424 const std::string base = eveSdkBaseUrl();
425 const std::string zipName = "eve-sdk-" + plat + "-" + tag + ".zip";
426 const std::string zipUrl = base + "/" + zipName;
427 const std::string sumsUrl = base + "/SHA256SUMS";
428 const auto work =
429 fs::temp_directory_path(ec) / ("eve-get-" + plat + "-" + std::to_string(processId()));
430 if (ec) {
431 return sdkFailure(eve::DiagnosticCode::Failed, "no temporary directory available");
432 }
433 fs::remove_all(work, ec);
434 fs::create_directories(work, ec);
435 if (ec) {
436 return sdkFailure(eve::DiagnosticCode::Failed, "cannot create SDK download workspace: " + ec.message(),
437 work.string());
438 }
439 const std::string zipPath = (work / zipName).string();
440 const std::string sumsPath = (work / "SHA256SUMS").string();
441
442 std::cout << "eve get: downloading " << zipName << " (" << zipUrl << ")...\n";
443 if (runShell("curl -fL --retry 3 -o \"" + zipPath + "\" \"" + zipUrl + "\"") != 0) {
444 fs::remove_all(work, ec);
445 return sdkFailure(eve::DiagnosticCode::Failed, "SDK archive download failed", zipUrl);
446 }
447 if (!nonEmptyRegularFile(zipPath, ec)) {
448 fs::remove_all(work, ec);
449 return sdkFailure(eve::DiagnosticCode::Failed, "SDK archive download returned an empty or unreadable response",
450 zipUrl);
451 }
452 if (runShell("curl -fL --retry 3 -o \"" + sumsPath + "\" \"" + sumsUrl + "\"") != 0) {
453 fs::remove_all(work, ec);
454 return sdkFailure(eve::DiagnosticCode::Failed, "SDK checksum file download failed", sumsUrl);
455 }
456 if (!nonEmptyRegularFile(sumsPath, ec)) {
457 fs::remove_all(work, ec);
458 return sdkFailure(eve::DiagnosticCode::Failed, "SDK checksum download returned an empty or unreadable response",
459 sumsUrl);
460 }
461 {
462 std::ifstream in(sumsPath);
463 const std::string sums(
464 (std::istreambuf_iterator<char>(in)), std::istreambuf_iterator<char>());
465 const std::string expected = sha256ForName(sums, zipName);
466 const std::string actual = fileSha256(zipPath);
467 if (expected.empty() || actual.empty() || lower(expected) != lower(actual)) {
468 fs::remove_all(work, ec);
469 return sdkFailure(eve::DiagnosticCode::HashMismatch, "SDK SHA-256 verification failed for " + zipName,
470 zipPath);
471 }
472 }
473
474 const std::string extractDir = (work / "extract").string();
475 fs::create_directories(extractDir, ec);
476 if (ec) {
477 fs::remove_all(work, ec);
478 return sdkFailure(eve::DiagnosticCode::Failed, "cannot create SDK extraction directory: " + ec.message(),
479 extractDir);
480 }
481 if (!extractZip(zipPath, extractDir)) {
482 fs::remove_all(work, ec);
483 return sdkFailure(eve::DiagnosticCode::ParseError,
484 "SDK archive is empty, truncated, or not a supported ZIP archive", zipPath);
485 }
486 const auto srcRoot = fs::path(extractDir) / "eve-sdk" / plat;
487 if (!fs::is_directory(srcRoot, ec)) {
488 fs::remove_all(work, ec);
489 return sdkFailure(eve::DiagnosticCode::ParseError,
490 "unexpected SDK archive layout; expected eve-sdk/" + plat + "/", zipPath);
491 }
492 {
493 const std::string platMarker = sdkTargetPlatform(srcRoot.string());
494 const std::string verMarker = sdkVersion(srcRoot.string());
495 if (platMarker != plat) {
496 fs::remove_all(work, ec);
497 return sdkFailure(eve::DiagnosticCode::InvalidArgument,
498 "SDK archive targets '" + platMarker + "', not '" + plat + "'",
499 (srcRoot / "share" / "eve" / "TARGET_PLATFORM").string());
500 }
501 if (verMarker != expectedVer) {
502 fs::remove_all(work, ec);
503 return sdkFailure(eve::DiagnosticCode::InvalidArgument,
504 "SDK archive contains version '" + verMarker + "', expected '" + expectedVer + "'",
505 (srcRoot / "share" / "eve" / "VERSION").string());
506 }
507 }
508
509 fs::create_directories(installRoot, ec);
510 if (ec) {
511 fs::remove_all(work, ec);
512 return sdkFailure(eve::DiagnosticCode::Failed, "cannot create SDK install directory: " + ec.message(),
513 installRoot);
514 }
515 if (!moveOrCopy(srcRoot, fs::path(destRoot))) {
516 fs::remove_all(work, ec);
517 return sdkFailure(eve::DiagnosticCode::Failed, "failed to install the SDK", destRoot);
518 }
519 fs::remove_all(work, ec);
520 std::cout << "eve get: EVEngine " << plat << " SDK " << tag << " installed at "
521 << destRoot << std::endl;
522 return eve::Result<void>::success();
523}

548void applyEnvFile(const std::string& path) {
549 std::ifstream in(path);
550 if (!in) return;

552 while (std::getline(in, line)) {

662[[nodiscard]] eve::Result<void> downloadAndExtract(const std::string& url, const std::string& zipPath,
663 const std::string& extractDir) {
664 if (runShell("curl -fL --retry 3 -o \"" + zipPath + "\" \"" + url + "\"") != 0) {
665 return sdkFailure(eve::DiagnosticCode::Failed, "download failed", url);
666 }
667 std::error_code ec;
668 if (!nonEmptyRegularFile(zipPath, ec)) {
669 return sdkFailure(eve::DiagnosticCode::Failed, "download returned an empty or unreadable response", url);
670 }
671 std::filesystem::remove_all(extractDir, ec);
672 std::filesystem::create_directories(extractDir, ec);
673 if (ec) {
674 return sdkFailure(eve::DiagnosticCode::Failed, "cannot create extraction directory: " + ec.message(),
675 extractDir);
676 }
677 if (!extractZip(zipPath, extractDir)) {
678 return sdkFailure(eve::DiagnosticCode::ParseError, "downloaded archive is empty, truncated, or unsupported",
679 zipPath);
680 }
681 std::filesystem::remove(zipPath, ec);
682 if (ec) {
683 return sdkFailure(eve::DiagnosticCode::Failed, "cannot remove temporary downloaded archive: " + ec.message(),
684 zipPath);
685 }
686 return eve::Result<void>::success();
687}

689// move 失败（跨盘符等）时退化为递归 copy。
690bool moveOrCopy(const std::filesystem::path& from, const std::filesystem::path& to) {
691 std::error_code ec;
692 std::filesystem::remove_all(to, ec);
693 std::filesystem::rename(from, to, ec);
694 if (!ec) return true;
695 ec.clear();
696 std::filesystem::copy(from, to,
697 std::filesystem::copy_options::recursive |
698 std::filesystem::copy_options::overwrite_existing,
699 ec);
700 return !ec;
701}

758eve::Result<void> installAndroidSdk() {
759 // EVEngine 官方 android SDK 优先：预编译 native 库 + APK 模板按当前 eve
760 // 版本从 GitHub Release 下载（校验 SHA256）。
761 auto eveSdk = installEveSdk(Platform::Android);
762 if (!eveSdk.ok()) return eve::Result<void>::failure(eveSdk.status());
763
764 const std::string root = androidSdkRoot();
765 std::error_code ec;
766 std::filesystem::create_directories(root, ec);
767 if (ec) {
768 return sdkFailure(eve::DiagnosticCode::Failed, "cannot create Android SDK directory: " + ec.message(), root);
769 }
770
771 const std::string sdkManager =
772 root + "/cmdline-tools/latest/bin/sdkmanager" + sdkmanagerSuffix();
773 if (isAndroidSdkInstalled(root)) {
774 std::cout << "eve get: Android SDK already installed at " << root << "\n";
775 } else {
776 std::cout << "eve get: downloading Android command-line tools...\n";
777 const std::string url =
778 getEnv("EVE_ANDROID_CMDLINE_TOOLS_URL", androidCmdlineToolsUrl());
779 const std::string tmp = root + "/.cmdline-tools-extract";
780 auto downloaded = downloadAndExtract(url, root + "/cmdline-tools.zip", tmp);
781 if (!downloaded.ok()) return eve::Result<void>::failure(downloaded.status());
782 if (!std::filesystem::is_directory(std::filesystem::path(tmp) / "cmdline-tools", ec)) {
783 return sdkFailure(eve::DiagnosticCode::ParseError, "unexpected Android command-line tools archive layout",
784 tmp);
785 }
786 if (!moveOrCopy(std::filesystem::path(tmp) / "cmdline-tools",
787 std::filesystem::path(root) / "cmdline-tools" / "latest")) {
788 return sdkFailure(eve::DiagnosticCode::Failed, "failed to install Android command-line tools",
789 root + "/cmdline-tools/latest");
790 }
791 std::filesystem::remove_all(tmp, ec);
792 }
793
794 // JDK 17 优先复用 $JAVA_HOME；否则下载 Temurin 17 到 <sdk>/jdk17。
795 std::string javaHome = getEnv("JAVA_HOME");
796 if (javaHome.empty()) {
797 javaHome = (std::filesystem::path(root) / "jdk17").string();
798 if (!jdkHomeExists(javaHome)) {
799 auto jdk = installJdk(root);
800 if (!jdk.ok()) return eve::Result<void>::failure(jdk.status());
801 }
802 } else if (!jdkHomeExists(javaHome)) {
803 std::cerr << "eve get: warning: JAVA_HOME=" << javaHome
804 << " does not contain bin/java; make sure it points at a JDK 17+.\n";
805 }
806 setEnv("JAVA_HOME", javaHome);
807
808 // 接受许可并安装组件（sdkmanager 幂等，可重复执行补齐）。
809 const std::string lic = root + "/.eve-licenses.txt";
810 {
811 std::ofstream f(lic, std::ios::binary | std::ios::trunc);
812 if (!f) {
813 return sdkFailure(eve::DiagnosticCode::Failed, "cannot create Android SDK license input", lic);
814 }
815 for (int i = 0; i < 200; ++i) f << "y\n";
816 if (!f) {
817 return sdkFailure(eve::DiagnosticCode::Failed, "cannot write Android SDK license input", lic);
818 }
819 }
820#if defined(_WIN32)
821 // cmd /c strips the first/last quote of a command that starts with one, so
822 // batch files with spaces in their path need the `call` prefix.
823 const std::string sdkCmd = "call \"" + sdkManager + "\"";
824#else
825 const std::string sdkCmd = "\"" + sdkManager + "\"";
826#endif
827 std::cout << "eve get: accepting Android SDK licenses...\n";
828 if (runShell(sdkCmd + " --licenses < \"" + lic + "\"") != 0) {
829 return sdkFailure(eve::DiagnosticCode::Failed, "failed to accept Android SDK licenses", sdkManager);
830 }
831 std::cout << "eve get: installing SDK packages (platform-tools, android-34, "
832 "build-tools)...\n";
833 const std::string packages = "\"platform-tools\" \"" + std::string(kAndroidPlatform) +
834 "\" \"" + std::string(kAndroidBuildTools) + "\"";
835 if (runShell(sdkCmd + " " + packages) != 0) {
836 return sdkFailure(eve::DiagnosticCode::Failed, "failed to install Android SDK packages", sdkManager);
837 }
838 std::filesystem::remove(lic, ec);
839
840 // Gradle：APK 模板没有 gradlew.bat，直接用发行版启动器，无需额外工具。
841 std::string gradleHome = (std::filesystem::path(root) / ("gradle-" + std::string(kGradleVersion))).string();
842 if (!gradleInstalled(gradleHome)) {
843 auto gradle = installGradle(root);
844 if (!gradle.ok()) return eve::Result<void>::failure(gradle.status());
845 }
846
847 // 记录环境，供 `eve build android` 在环境变量未设置时使用。
848 const std::string eveSdkRoot =
849 (std::filesystem::path(eveSdkInstallRoot()) / "android").string();
850 {
851 std::ofstream f(std::filesystem::path(root) / "eve-android.env",
852 std::ios::binary | std::ios::trunc);
853 if (!f) {
854 return sdkFailure(eve::DiagnosticCode::Failed, "cannot write Android SDK environment file",
855 (std::filesystem::path(root) / "eve-android.env").string());
856 }
857 f << "ANDROID_HOME=" << root << "\n"
858 << "ANDROID_SDK_ROOT=" << root << "\n"
859 << "JAVA_HOME=" << javaHome << "\n"
860 << "GRADLE_HOME=" << gradleHome << "\n"
861 << "EVENGINE_SDK=" << eveSdkRoot << "\n";
862 if (!f) {
863 return sdkFailure(eve::DiagnosticCode::Failed, "cannot finish writing Android SDK environment file",
864 (std::filesystem::path(root) / "eve-android.env").string());
865 }
866 }
867
868 std::cout << "eve get: Android SDK ready at " << root << "\n"
869 << "eve get: run `eve build android` to build an APK "
870 "(or `eve build -d android` for a debug APK).\n";
871 return eve::Result<void>::success();
872}

876void copyOne(const std::filesystem::path& from, const std::filesystem::path& to,
877 const std::vector<std::string>& skipDirs, bool contentsOnly) {
878 std::error_code ec;
879 if (!contentsOnly) std::filesystem::create_directories(to, ec);
880 std::filesystem::recursive_directory_iterator it(
881 from, std::filesystem::directory_options::skip_permission_denied, ec);
882 std::filesystem::recursive_directory_iterator end;
883 for (; it != end; it.increment(ec)) {
884 if (ec) {
885 ec.clear();
886 continue;
887 }
888 const auto rel = std::filesystem::relative(it->path(), from, ec).lexically_normal();
889 if (ec) continue;
890 const std::string relStr = rel.generic_string();
891 bool skip = false;
892 for (const auto& d : skipDirs) {
893 if (relStr == d || relStr.rfind(d + "/", 0) == 0) {
894 skip = true;
895 break;
896 }
897 }
898 if (skip) {
899 it.disable_recursion_pending();
900 continue;
901 }
902 // 无论是否 contentsOnly，目标都按相对路径落位（contentsOnly 时 to 就是
903 // 目标根目录，rel 相对源根目录）。
904 const auto dst = to / rel;
905 if (it->is_directory()) {
906 std::filesystem::create_directories(dst, ec);
907 } else if (it->is_regular_file()) {
908 std::filesystem::create_directories(dst.parent_path(), ec);
909 std::filesystem::copy_file(it->path(), dst,
910 std::filesystem::copy_options::overwrite_existing, ec);
911 }
912 }
913}
