58 struct Prepared {
59 Coupling3D* owner;
60 coupling_detail::Row row;
61 std::array<std::size_t, 4> index{};
62 double response = 0, impulse = 0, maximum = 0, bias = 0;
63 bool tensionOnly = false;

74 p.owner = coupling;
75 p.row = std::move(result).takeValue();
76 p.maximum = data.limits().maxForce * dt;

78 const double gain = p.tensionOnly && p.row.error < 0 ? 1.0 : data.limits().positionCorrection;
79 p.bias = std::clamp(gain * p.row.error / dt, -static_cast<double>(data.limits().maxCorrectionSpeed),
80 static_cast<double>(data.limits().maxCorrectionSpeed));
81 for (int i = 0; i < p.row.count; ++i) {
82 const auto& term = p.row.terms[i];
83 auto found = indices.find(term.body);
84 if (found == indices.end()) {
85 auto* body = findBody(term.body);
86 if (!body || !body->isValid()) return rejected("Coupling body became stale during preparation");
87 const auto raw = eve::physics::box3d::Access::bodyId(*body);
88 const auto linear = b3Body_GetLinearVelocity(raw), angular = b3Body_GetAngularVelocity(raw);
89 const auto inertia = b3Body_GetWorldInverseRotationalInertia(raw);
90 const float mass = b3Body_GetInverseMass(raw);
91 if (!b3IsValidVec3(linear) || !b3IsValidVec3(angular) || !std::isfinite(mass) || mass < 0 ||
92 !b3IsValidVec3(inertia.cx) || !b3IsValidVec3(inertia.cy) || !b3IsValidVec3(inertia.cz))
93 return rejected("Non-finite coupling body state");

96 b3Body_GetType(raw) == b3_dynamicBody, b3Body_IsAwake(raw)});

98 }

100 const auto& v = transaction.values[found->second];

102 p.response += v.inverseMass * b3Dot(term.linear, term.linear) +
103 b3Dot(term.angular, b3MulMV(v.inverseInertia, term.angular));
104 }
105 if (p.row.active && (!std::isfinite(p.response) || p.response <= 0))
106 return rejected("Coupling has no finite movable degree of freedom");
107 rows.push_back(std::move(p));
108 }
109 // Project all transmissions together. Shared endpoints accumulate one Jacobian before effective-mass evaluation.
110 for (int iteration = 0; iteration < settings.velocityIterations; ++iteration)
111 for (auto& p : rows) {
112 if (!p.row.active) continue;
113 double speed = 0;
114 for (int i = 0; i < p.row.count; ++i) {
115 const auto& v = transaction.values[p.index[i]];
116 const auto& t = p.row.terms[i];
117 speed += b3Dot(t.linear, v.linear) + b3Dot(t.angular, v.angular);
118 }
119 const double next =
120 std::clamp(p.impulse - (speed + p.bias) / p.response, -p.maximum, p.tensionOnly ? 0.0 : p.maximum);
121 const float impulse = static_cast<float>(next - p.impulse);
122 if (!std::isfinite(impulse)) return rejected("Coupling impulse overflow");
123 p.impulse = next;
124 for (int i = 0; i < p.row.count; ++i) {
125 auto& v = transaction.values[p.index[i]];
126 const auto& t = p.row.terms[i];
127 if (!v.dynamic) continue;
128 v.linear = b3Add(v.linear, b3MulSV(impulse * v.inverseMass, t.linear));
129 v.angular = b3Add(v.angular, b3MulMV(v.inverseInertia, b3MulSV(impulse, t.angular)));
130 if (!b3IsValidVec3(v.linear) || !b3IsValidVec3(v.angular))
131 return rejected("Coupling velocity overflow");
132 }
133 }
134 for (const auto& p : rows) {
135 const auto& d = *p.owner->data_;
136 if (!d.gear) continue;
137 for (int i = 0; i < 2; ++i)
138 if (d.revolute[i]) {
139 const auto& a = transaction.values[indices.at(d.bodies[2 * i])];
140 const auto& b = transaction.values[indices.at(d.bodies[2 * i + 1])];
141 if (b3Length(b3Sub(b.angular, a.angular)) * dt >= std::numbers::pi)
142 return rejected("Revolute coupling requires less than pi radians of relative rotation per step");
143 }
144 }
145 transaction.apply();
146 auto result = simulation_->step(step, settings);
147 if (!result) return result;
148 transaction.committed = true;
149 for (const auto& p : rows) {
150 auto& d = *p.owner->data_;
151 d.previous = p.row.wrapped;
152 d.coordinate = p.row.coordinate;
153 d.lastImpulse = p.impulse;
154 d.saturated = p.row.active && std::abs(p.impulse) >= p.maximum;
155 }
156 return result;
157}
