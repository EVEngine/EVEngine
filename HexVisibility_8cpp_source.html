56Result<void> HexVisibility::increase(HexMap& map, HexSearchContext& scratch, HexCoordinates from,
57 std::int32_t range) {
58 Result<void> valid = validateView(map, *this, from);
59 if (!valid.ok()) return valid;
60
61 std::vector<std::int32_t> cells;
62 Result<void> collected = collectVisibleCells(map, scratch, from, range, cells);
63 if (!collected.ok()) return collected;
64
65 for (const std::int32_t index : cells) {
66 // `collectVisibleCells` only reports indices of this map, so the counter is in range.
67 if (++counts_[static_cast<std::size_t>(index)] != 1) continue;
68 // First viewer of this cell: latch it explored and remember the counter so
69 // `clear` can zero the map without a full scan.
70 touched_.push_back(index);
71 map.setExplored(map.coordinatesAt(index), true)
72 .ignore("the cell index comes from this map, so the explored latch cannot be rejected");
73 // The fog overlay of this cell changed shade, so its chunk must be rebuilt.
74 map.markChunkDirtyAndNeighbors(map.chunkIndexOf(map.coordinatesAt(index)));
75 }
76 return Result<void>::success();
77}

79Result<void> HexVisibility::decrease(HexMap& map, HexSearchContext& scratch, HexCoordinates from,
80 std::int32_t range) {
81 Result<void> valid = validateView(map, *this, from);
82 if (!valid.ok()) return valid;
83
84 std::vector<std::int32_t> cells;
85 Result<void> collected = collectVisibleCells(map, scratch, from, range, cells);
86 if (!collected.ok()) return collected;
87
88 for (const std::int32_t index : cells) {
89 std::int32_t& counter = counts_[static_cast<std::size_t>(index)];
90 if (counter == 0) continue;
91 if (--counter != 0) continue;
92 // The cell lost its last viewer. `touched_` order is unspecified, so a
93 // swap-and-pop removal is fine.
94 const auto entry = std::find(touched_.begin(), touched_.end(), index);
95 if (entry != touched_.end()) {
96 *entry = touched_.back();
97 touched_.pop_back();
98 }
99 // The cell became invisible again: its fog overlay shade changed.
100 map.markChunkDirtyAndNeighbors(map.chunkIndexOf(map.coordinatesAt(index)));
101 }
102 return Result<void>::success();
103}
