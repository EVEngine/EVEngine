31 for (const auto& asset : before.assets) left.emplace(asset.asset.id(), &asset);

34 for (const auto& [id, asset] : left) {
35 const auto found = right.find(id);
36 if (found == right.end()) {
37 result.assets.push_back({id, EvaAssetChangeKind::Removed, asset->type,
38 asset->schemaVersion, asset->contentHash, {}, {}, {}});
39 } else if (asset->type != found->second->type || asset->schemaVersion != found->second->schemaVersion ||
40 asset->contentHash != found->second->contentHash) {
41 result.assets.push_back({id, EvaAssetChangeKind::Changed, asset->type,
42 asset->schemaVersion, asset->contentHash, found->second->type,
43 found->second->schemaVersion, found->second->contentHash});

45 }
46 for (const auto& [id, asset] : right)
47 if (!left.contains(id))

49 asset->schemaVersion, asset->contentHash});
50 std::sort(result.assets.begin(), result.assets.end(), [](const auto& a, const auto& b) {
51 return a.assetId < b.assetId;
