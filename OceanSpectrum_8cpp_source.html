25Result<OceanSpectrum> OceanSpectrumBuilder::build() const {
26 const auto& s = settings;
27 if (!std::isfinite(s.significantWaveHeight) || s.significantWaveHeight < 0.0 || s.significantWaveHeight > 30.0 ||
28 !std::isfinite(s.peakPeriod) || s.peakPeriod < 1.0 || s.peakPeriod > 30.0 ||
29 !std::isfinite(s.peakEnhancement) || s.peakEnhancement < 1.0 || s.peakEnhancement > 10.0 ||
30 !std::isfinite(s.gravity) || s.gravity <= 0.0 || s.gravity > 100.0 || !std::isfinite(s.seaLevel) ||
31 std::abs(s.seaLevel) > 1e6 || !std::isfinite(s.choppiness) || s.choppiness < 0.0 || s.choppiness > 1.0 ||
32 !finite(s.windDirection) || glm::length(s.windDirection) < 1e-8 || glm::length(s.windDirection) > 1e6 ||
33 !finite(s.current) || glm::length(s.current) > 100.0 || s.bands.empty() || s.bands.size() > 4)
34 return invalid();
35 double previous = 0.0;
36 for (const auto& b : s.bands) {
37 if (b.resolution < 8 || b.resolution > 512 || (b.resolution & (b.resolution - 1)) != 0 ||
38 !std::isfinite(b.length) || b.length < 1.0 || b.length > 1e6 || !std::isfinite(b.minimumWavelength) ||
39 !std::isfinite(b.maximumWavelength) || b.minimumWavelength < 2.0 * b.length / b.resolution ||
40 b.maximumWavelength > b.length || b.minimumWavelength >= b.maximumWavelength ||
41 b.minimumWavelength < previous)
42 return invalid();
43 previous = b.maximumWavelength;
44 }
45 auto data = std::make_shared<OceanSpectrumData>();
46 data->settings = s;
47 const auto wind = glm::normalize(s.windDirection);
48 std::uint64_t random = s.seed;
49 double energy = 0.0;
50 const double peak = tau / s.peakPeriod;
51 for (const auto& b : s.bands) {
52 OceanSpectrumData::Band band{b, {}};
53 band.coefficients.resize(b.resolution * b.resolution);
54 const double dk = tau / b.length;
55 for (unsigned z = 0; z < b.resolution; ++z) {
56 for (unsigned x = 0; x < b.resolution; ++x) {
57 const int ix =
58 x < b.resolution / 2 ? static_cast<int>(x) : static_cast<int>(x) - static_cast<int>(b.resolution);
59 const int iz =
60 z < b.resolution / 2 ? static_cast<int>(z) : static_cast<int>(z) - static_cast<int>(b.resolution);
61 // Exclude Nyquist axes so odd displacement multipliers remain Hermitian.
62 if ((ix == 0 && iz == 0) || x == b.resolution / 2 || z == b.resolution / 2) continue;
63 const glm::dvec2 kvec{ix * dk, iz * dk};
64 const double k = glm::length(kvec), wavelength = tau / k;
65 if (wavelength < b.minimumWavelength || wavelength >= b.maximumWavelength) continue;
66 const double omega = std::sqrt(s.gravity * k);
67 const double sigma = omega <= peak ? 0.07 : 0.09;
68 const double r = std::exp(-0.5 * std::pow((omega - peak) / (sigma * peak), 2.0));
69 const double direction = std::pow(std::max(glm::dot(kvec / k, wind), 0.0), 4.0);
70 const double density = std::pow(omega, -5.0) * std::exp(-1.25 * std::pow(peak / omega, 4.0)) *
71 std::pow(s.peakEnhancement, r) * direction * (s.gravity / (2.0 * omega)) / k;
72 const double magnitude = std::sqrt(density * dk * dk * -std::log(uniform(random)));
73 const auto coefficient = std::polar(magnitude, tau * uniform(random));
74 band.coefficients[z * b.resolution + x] = coefficient;
75 energy += 2.0 * std::norm(coefficient);
76 }
77 }
78 data->bands.push_back(std::move(band));
79 }
80 if (s.significantWaveHeight > 0.0 && energy < 1e-30) return invalid();
81 const double variance = s.significantWaveHeight * s.significantWaveHeight / 16.0;
82 const double scale = energy > 0.0 ? std::sqrt(variance / energy) : 0.0;
83 for (auto& band : data->bands)
84 for (auto& coefficient : band.coefficients) coefficient *= scale;
85 data->variance = variance;
86 return Result<OceanSpectrum>::success(OceanSpectrum(std::move(data)));
87}
