19eve::Result<double> validateTrade(const GameState &gameState, const std::string &currencyId,
20 const std::string &itemId, int quantity, double unitPrice,
21 bool purchase) {
22 if (currencyId.empty() || itemId.empty() || quantity <= 0 ||
23 !std::isfinite(unitPrice) || unitPrice < 0.0)
24 return eve::Result<double>::failure(eve::Diagnostic::error(
25 eve::DiagnosticCode::InvalidArgument,
26 "shop trade requires currency/item ids, positive quantity, and finite non-negative unit price",
27 "trade", {{"itemId", itemId}}, "rpg.shop-transaction"));
28 const double total = unitPrice * static_cast<double>(quantity);
29 const double balance = gameState.getVariable(currencyId);
30 const double result = purchase ? balance - total : balance + total;
31 if (!std::isfinite(total) || !std::isfinite(balance) || !std::isfinite(result))
32 return eve::Result<double>::failure(eve::Diagnostic::error(
33 eve::DiagnosticCode::InvalidArgument, "shop trade would produce a non-finite currency value",
34 "unitPrice", {{"itemId", itemId}}, "rpg.shop-transaction"));
35 if (purchase && balance < total)
36 return eve::Result<double>::failure(eve::Diagnostic::error(
37 eve::DiagnosticCode::PreconditionViolation, "shop purchase has insufficient currency",
38 "currency", {{"itemId", itemId}}, "rpg.shop-transaction"));
39 return eve::Result<double>::success(total);
40}

75eve::Result<int> ShopTransaction::buy(GameState &gameState, inventory::Bag &bag,
76 const std::string &currencyId, const std::string &itemId,
77 int quantity, double unitPrice) {
78 auto validated = validateTrade(gameState, currencyId, itemId, quantity, unitPrice, true);
79 if (!validated.ok()) return eve::Result<int>::failure(validated.status());
80 const double total = std::move(validated).takeValue();
81 auto prepared = inventory::InventorySystem::prepareAddBatch(&bag, {{itemId, quantity}});
82 if (!prepared.ok()) return eve::Result<int>::failure(prepared.status());
83 auto before = gameState.snapshotJson();
84 if (!before.ok()) return eve::Result<int>::failure(before.status());
85 std::string snapshot = std::move(before).takeValue();
86
87 gameState.addVariable(currencyId, -total);
88 auto committed = inventory::InventorySystem::commitAddBatch(std::move(prepared).takeValue());
89 if (!committed.ok()) return rollbackFailure(gameState, snapshot, committed.status(), itemId);
90 return committed;
91}
