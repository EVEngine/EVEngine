77void addLeafCard(MeshBuild& out, V3 c, V3 direction, float size, float twist, std::mt19937& rng) {
78 V3 right, up;
79 basisFor(norm(direction), right, up);
80 right = add(mul(right, std::cos(twist)), mul(up, std::sin(twist)));
81 up = norm(cross(norm(direction), right));
82 const V3 normal = norm(cross(right, up));
83 const float card = size * 1.25f;
84 const float halfW = card * randomRange(rng, 0.55f, 0.72f);
85 const float halfH = card * randomRange(rng, 0.55f, 0.72f);
86 const V3 r = mul(right, halfW), h = mul(up, halfH);
87 const V3 center = add(c, mul(normal, size * 0.04f));
88 const V3 points[4] = {sub(sub(center, r), h), add(sub(center, h), r), add(add(center, r), h),
89 add(sub(center, r), h)};
90
91 // One of six leaf-card panels (8–16 blue-noise ovate stamps each).
92 constexpr float kCardU0 = 0.52f;
93 constexpr float kCardU1 = 1.00f;
94 constexpr int kCols = 2;
95 constexpr int kRows = 3;
96 const int panel = int(randomRange(rng, 0.f, float(kCols * kRows))) % (kCols * kRows);
97 const int col = panel % kCols;
98 const int row = panel / kCols;
99 const float inset = 0.04f;
100 const float cellWU = (kCardU1 - kCardU0) / float(kCols);
101 const float cellWV = 1.f / float(kRows);
102 const float u0 = kCardU0 + (float(col) + inset) * cellWU;
103 const float u1 = kCardU0 + (float(col) + 1.f - inset) * cellWU;
104 const float v0 = (float(row) + inset) * cellWV;
105 const float v1 = (float(row) + 1.f - inset) * cellWV;
106 const bool flipU = randomRange(rng, 0.f, 1.f) > 0.5f;
107 const bool flipV = randomRange(rng, 0.f, 1.f) > 0.5f;
108 const float uv[4][2] = {
109 {flipU ? u1 : u0, flipV ? v1 : v0},
110 {flipU ? u0 : u1, flipV ? v1 : v0},
111 {flipU ? u0 : u1, flipV ? v0 : v1},
112 {flipU ? u1 : u0, flipV ? v0 : v1},
113 };
114 const uint32_t base = uint32_t(out.getVertexCount());
115 for (int i = 0; i < 4; ++i)
116 out.addVertex(points[i].x, points[i].y, points[i].z, normal.x, normal.y, normal.z, uv[i][0], uv[i][1]);
117 out.addTriangle(base, base + 1, base + 2);
118 out.addTriangle(base, base + 2, base + 3);
119 out.addTriangle(base + 2, base + 1, base);
120 out.addTriangle(base + 3, base + 2, base);
121}

189int addFoliageClusters(MeshBuild& out, const std::vector<Tip>& anchors, std::mt19937& rng, float crownRadius,
190 float leafSize, float density, uint32_t seed, const ClusterSettings& settings) {
191 if (anchors.empty() || settings.limit <= 0) return 0;
192
193 std::vector<size_t> order(anchors.size());
194 for (size_t i = 0; i < order.size(); ++i) order[i] = i;
195 for (size_t i = order.size(); i > 1; --i)
196 std::swap(order[i - 1], order[size_t(randomRange(rng, 0.f, float(i) - 1e-4f))]);
197
198 const float baseRadius = std::max(1e-3f, crownRadius * settings.size);
199 const float separation = std::max(0.05f, settings.separation) * baseRadius * 2.f;
200 const float separationSq = separation * separation;
201 // Higher density packs the same cluster volume with more, closer leaves.
202 const float spacing = std::max(0.05f, settings.spacing * (1.45f - 0.75f * density));
203
204 std::vector<V3> accepted;
205 accepted.reserve(size_t(settings.limit));
206 int clusters = 0;
207 for (size_t k = 0; k < order.size() && clusters < settings.limit; ++k) {
208 const Tip& tip = anchors[order[k]];
209 bool blocked = false;
210 for (const V3& taken : accepted) {
211 if (distanceSquared(taken, tip.p) < separationSq) {
212 blocked = true;
213 break;
214 }
215 }
216 if (blocked) continue;
217
218 const float radius = baseRadius * (0.62f + 0.38f * tip.scale) * randomRange(rng, 0.88f, 1.12f);
219 // Pull the cluster back along the shoot so the twig visibly enters the
220 // foliage instead of stopping just short of it.
221 const V3 center = add(sub(tip.p, mul(tip.dir, radius * 0.10f)), {0.f, radius * 0.06f, 0.f});
222
223 FoliageClusterDesc cluster;
224 cluster.seed = seed ^ (uint32_t(clusters + 1) * 2654435761u);
225 cluster.centerX = center.x;
226 cluster.centerY = center.y;
227 cluster.centerZ = center.z;
228 cluster.radius = radius;
229 cluster.leafSize = leafSize * settings.leafScale * randomRange(rng, 0.85f, 1.15f);
230 cluster.leafSpacing = spacing;
231 cluster.planes = settings.planes;
232 cluster.capPlanes = settings.caps;
233 cluster.tiltDegrees = settings.tilt;
234 cluster.maxLeavesPerPlane = settings.leavesPerPlane;
235 cluster.normalRounding = 1.f;
236 cluster.doubleSided = true;
237 addFoliageCluster(out, cluster);
238
239 accepted.push_back(tip.p);
240 ++clusters;
241 }
242 return clusters;
243}

250StemPath makeCurvedStem(V3 start, V3 direction, V3 bendAxis, float length, float curve, float curveBack,
251 float verticalAcceleration, int segments) {
252 StemPath path;
253 path.points.reserve(size_t(segments + 1));
254 direction = norm(direction);
255 bendAxis = norm(sub(bendAxis, mul(direction, dot(bendAxis, direction))));
256 const V3 up{0.f, 1.f, 0.f};
257 for (int i = 0; i <= segments; ++i) {
258 const float t = float(i) / float(segments);
259 // Both offsets have zero slope at the branch base, avoiding an immediate kink.
260 const float primary = curve * t * t;
261 const float back = curveBack * std::sin(2.f * kPi * t) * t * (1.f - t);
262 const V3 p = add(start, add(mul(direction, length * t), add(mul(bendAxis, length * (primary + back)),
263 mul(up, length * verticalAcceleration * t * t))));
264 path.points.push_back(p);
265 }
266 path.endDirection = norm(sub(path.points.back(), path.points[path.points.size() - 2]));
267 return path;
268}

288// One continuous sweep: transported frames, shared rings and arc-length UVs.
289// Local input/output only; does not consume the skeleton/foliage RNG stream.
290void addStem(MeshBuild& out, const StemPath& path, float r0, float r1, int sides, const StemSurface& surface) {
291 const int sourceSegments = int(path.points.size()) - 1;
292 const int segments = std::max(sourceSegments, surface.segments);
293 std::vector<V3> centers;
294 centers.reserve(size_t(segments + 1));
295 for (int i = 0; i <= segments; ++i) {
296 const float at = float(i) * float(sourceSegments) / float(segments);
297 const int edge = std::min(int(at), sourceSegments - 1);
298 centers.push_back(add(path.points[size_t(edge)],
299 mul(sub(path.points[size_t(edge + 1)], path.points[size_t(edge)]), at - float(edge))));
300 }
301 const int rings = int(centers.size());
302 std::vector<float> arc(size_t(rings), 0.f);
303 std::vector<V3> tangent(size_t(rings), V3{});
304 for (int i = 0; i < rings; ++i) {
305 const V3 before = centers[size_t(std::max(0, i - 1))];
306 const V3 after = centers[size_t(std::min(rings - 1, i + 1))];
307 tangent[size_t(i)] = norm(sub(after, before));
308 if (i > 0) arc[size_t(i)] = arc[size_t(i - 1)] + std::sqrt(distanceSquared(centers[size_t(i)], before));
309 }
310 const float length = std::max(arc.back(), 1e-5f);
311 const float phase = hash01(surface.seed ^ 0xb5297a4du) * 2.f * kPi;
312 const int stride = sides + 1;
313 std::vector<V3> positions(size_t(rings * stride));
314 NoiseField coarse, fine;
315 coarse.seed = surface.seed ^ 0x68e31da4u;
316 coarse.periodX = 4;
317 fine.seed = surface.seed ^ 0x1b56c4e9u;
318 fine.periodX = 8;
319 struct Knot {
320 float angle, distance, width, height, strength;
321 };
322 std::vector<Knot> knots;
323 for (int i = 0; i < surface.knots; ++i) {
324 const uint32_t key = surface.seed ^ 0x9e3779b9u ^ (uint32_t(i + 1) * 0x85ebca6bu);
325 knots.push_back({2.f * kPi * hash01(key), length * (0.12f + 0.70f * hash01(key ^ 0x514e28b7u)),
326 0.45f + 0.4f * hash01(key ^ 0x7feb352du), r0 * (1.4f + 1.8f * hash01(key ^ 0x846ca68bu)),
327 surface.knotStrength * (0.7f + 0.3f * hash01(key ^ 0xa511e9b3u))});
328 }
329 const float noiseLength = std::max(r0 * surface.noiseScale, 1e-4f);
330 V3 right, forward;
331 basisFor(tangent[0], right, forward);
332 for (int i = 0; i < rings; ++i) {
333 const V3 axis = tangent[size_t(i)];
334 // Project the preceding frame onto this ring instead of switching the
335 // helper axis at an arbitrary direction threshold on every segment.
336 const V3 transported = sub(right, mul(axis, dot(right, axis)));
337 if (dot(transported, transported) < 1e-8f)
338 basisFor(axis, right, forward);
339 else {
340 right = norm(transported);
341 forward = norm(cross(axis, right));
342 }
343 const float t = arc[size_t(i)] / length;
344 const float radius = stemRadius(t, r0, r1, surface.taper);
345 const float foot = surface.rootFlare * std::exp(-3.f * arc[size_t(i)] / surface.flareLength);
346 const float collar = surface.collar * std::exp(-2.5f * arc[size_t(i)] / std::max(r0 * 2.f, 1e-4f));
347 for (int j = 0; j <= sides; ++j) {
348 const float angle = 2.f * kPi * float(j % sides) / float(sides);
349 const V3 radial = add(mul(right, std::cos(angle)), mul(forward, std::sin(angle)));
350 const float roots = 0.22f + 1.25f * std::pow(std::max(0.f, std::cos(5.f * angle + phase)), 4.f) +
351 0.55f * std::pow(std::max(0.f, std::cos(3.f * angle - phase)), 6.f);
352 const float u = float(j % sides) / float(sides);
353 const float growthMask = std::min(1.f, t * 20.f) * std::min(1.f, (1.f - t) * 8.f);
354 const float body =
355 surface.noiseAmount * (2.f * coarse.valueNoise(u * 4.f, arc[size_t(i)] / noiseLength) - 1.f);
356 const float grain =
357 surface.relief * (2.f * fine.valueNoise(u * 8.f, arc[size_t(i)] / (noiseLength * 0.4f)) - 1.f);
358 float localKnots = 0.f;
359 for (const Knot& knot : knots) {
360 const float angular = 2.f * std::sin((angle - knot.angle) * 0.5f) / knot.width;
361 const float axial = (arc[size_t(i)] - knot.distance) / knot.height;
362 const float d2 = angular * angular + axial * axial;
363 if (d2 >= 1.f) continue;
364 // A raised lip around a shallow centre gives each knot local
365 // structure instead of inflating a ring around the whole stem.
366 const float falloff = 1.f - d2;
367 localKnots += knot.strength * falloff * falloff * (0.35f + 2.f * d2);
368 }
369 const float detail = (body + grain + std::clamp(localKnots, 0.f, 0.65f)) * growthMask;
370 const float shaped = radius * (1.f + detail + collar) + r0 * foot * roots;
371 positions[size_t(i * stride + j)] = add(centers[size_t(i)], mul(radial, shaped));
372 }
373 }
374 const uint32_t base = uint32_t(out.getVertexCount());
375 for (int i = 0; i < rings; ++i) {
376 for (int j = 0; j <= sides; ++j) {
377 const int column = j % sides;
378 const V3 around = sub(positions[size_t(i * stride + (column + 1) % sides)],
379 positions[size_t(i * stride + (column + sides - 1) % sides)]);
380 const V3 along = sub(positions[size_t(std::min(rings - 1, i + 1) * stride + column)],
381 positions[size_t(std::max(0, i - 1) * stride + column)]);
382 const V3 normal = norm(cross(norm(around), norm(along)));
383 const V3 p = positions[size_t(i * stride + j)];
384 out.addVertex(p.x, p.y, p.z, normal.x, normal.y, normal.z, 0.005f + 0.47f * float(j) / float(sides),
385 arc[size_t(i)] / length);
386 }
387 }
388 for (int i = 0; i + 1 < rings; ++i) {
389 for (int j = 0; j < sides; ++j) {
390 const uint32_t a = base + uint32_t(i * stride + j);
391 const uint32_t b = a + 1;
392 const uint32_t c = a + uint32_t(stride);
393 const uint32_t d = c + 1;
394 out.addTriangle(a, b, c);
395 out.addTriangle(b, d, c);
396 }
397 }
398 for (int end = 0; end < 2; ++end) {
399 const int row = end ? rings - 1 : 0;
400 const V3 n = mul(tangent[size_t(row)], end ? 1.f : -1.f);
401 const V3 c = centers[size_t(row)];
402 const uint32_t cap = uint32_t(out.getVertexCount());
403 out.addVertex(c.x, c.y, c.z, n.x, n.y, n.z, 0.24f, end ? 1.f : 0.f);
404 for (int j = 0; j <= sides; ++j) {
405 const V3 p = positions[size_t(row * stride + j)];
406 out.addVertex(p.x, p.y, p.z, n.x, n.y, n.z, 0.24f, end ? 1.f : 0.f);
407 }
408 for (int j = 0; j < sides; ++j) {
409 if (end)
410 out.addTriangle(cap, cap + uint32_t(j + 1), cap + uint32_t(j + 2));
411 else
412 out.addTriangle(cap, cap + uint32_t(j + 2), cap + uint32_t(j + 1));
413 }
414 }
415}

465 samplePath(trunk, settings.foliageStart + (1.f - settings.foliageStart) * 0.53f, crownCenter, ignoredDirection);

468 std::vector<V3> attractors;
469 attractors.reserve(size_t(settings.attractorCount));
470 while (int(attractors.size()) < settings.attractorCount) {
471 const V3 q{randomRange(rng, -1.f, 1.f), randomRange(rng, -1.f, 1.f), randomRange(rng, -1.f, 1.f)};
472 if (dot(q, q) > 1.f) continue;
473 // A modest upward bias avoids a perfectly symmetric, balloon-like crown.
474 attractors.push_back(add(crownCenter, {q.x * settings.crownRadius,
475 q.y * crownHalfHeight + 0.12f * crownHalfHeight * (1.f - q.y * q.y),
476 q.z * settings.crownRadius}));
477 }
478
479 std::vector<ColonyNode> nodes;
480 // Multiple trunk buds provide the primary scaffold. Colonization then shapes and
481 // splits those limbs; starting from a single bud produces a shrub-like fan.
482 const int seedCount = 9;
483 for (int i = 0; i < seedCount; ++i) {
484 V3 p, dir;
485 const float t =
486 settings.foliageStart + (1.f - settings.foliageStart) * (0.08f + 0.78f * float(i) / float(seedCount - 1));
487 samplePath(trunk, t, p, dir);
488 const float crownHeight = float(i) / float(seedCount - 1);
489 const float lengthScale = 1.f - settings.lengthFalloff * crownHeight;
490 const float vigor = 1.f - settings.radiusFalloff * crownHeight;
491 nodes.push_back({p, dir, dir, -1, 1, 0, 0, settings.crownRadius * 1.35f * lengthScale, vigor, crownHeight});
492 }
493
494 std::vector<uint8_t> alive(size_t(settings.attractorCount), 1);
495 const float influenceSquared = settings.influenceRadius * settings.influenceRadius;
496 const float killSquared = settings.killRadius * settings.killRadius;
497 const float separationSquared = settings.step * settings.step * 0.22f;
498 const float maxTurn = settings.maxTurnAngle * kPi / 180.f;
499 const float maxTurnCos = std::cos(maxTurn);
500 const float maxTurnSin = std::sin(maxTurn);
501 const float cumulativeTurn = settings.maxCumulativeAngle * kPi / 180.f;
502 const float cumulativeTurnCos = std::cos(cumulativeTurn);
503 const float cumulativeTurnSin = std::sin(cumulativeTurn);
504 for (int iteration = 0; iteration < settings.iterations; ++iteration) {
505 const size_t nodeCount = nodes.size();
506 std::vector<V3> directionSums(nodeCount);
507 std::vector<int> directionCounts(nodeCount, 0);
508 int liveAttractors = 0;
509 for (size_t ai = 0; ai < attractors.size(); ++ai) {
510 if (!alive[ai]) continue;
511 ++liveAttractors;
512 size_t nearest = 0;
513 float best = influenceSquared;
514 bool found = false;
515 for (size_t ni = 0; ni < nodeCount; ++ni) {
516 const V3 toward = sub(attractors[ai], nodes[ni].p);
517 const float d = dot(toward, toward);
518 if (d < killSquared) {
519 alive[ai] = 0;
520 found = false;
521 break;
522 }
523 // Once a shoot has left the trunk, attraction from behind may not
524 // reverse it. This is the main distinction between a branch and a vine.
525 if (nodes[ni].depth > 0) {
526 const V3 attractionDirection = norm(toward);
527 if (dot(attractionDirection, nodes[ni].dir) < -0.08f ||
528 dot(attractionDirection, nodes[ni].heading) < 0.04f)
529 continue;
530 }
531 if (d < best) {
532 best = d;
533 nearest = ni;
534 found = true;
535 }
536 }
537 if (found) {
538 directionSums[nearest] = add(directionSums[nearest], norm(sub(attractors[ai], nodes[nearest].p)));
539 ++directionCounts[nearest];
540 }
541 }
542 if (liveAttractors == 0) break;
543
544 std::vector<ColonyNode> additions;
545 for (size_t ni = 0; ni < nodeCount; ++ni) {
546 if (directionCounts[ni] == 0 || nodes[ni].children >= settings.maxChildren ||
547 nodes[ni].remainingLength < settings.step)
548 continue;
549 V3 direction = norm(
550 add(mul(directionSums[ni], 1.f / float(directionCounts[ni])), mul(nodes[ni].dir, settings.inertia)));
551 const float vertical = settings.tropism - settings.droop * std::max(0.f, 1.f - direction.y);
552 direction = norm(add(direction, {0.f, vertical, 0.f}));
553 // Clamp curvature per growth step. Linear blending alone can still turn
554 // sharply when the remaining attractors move to the other side of a tip.
555 const float alignment = std::clamp(dot(nodes[ni].dir, direction), -1.f, 1.f);
556 if (nodes[ni].depth > 0 && alignment < maxTurnCos) {
557 V3 tangent = sub(direction, mul(nodes[ni].dir, alignment));
558 if (dot(tangent, tangent) > 1e-8f)
559 direction = norm(add(mul(nodes[ni].dir, maxTurnCos), mul(norm(tangent), maxTurnSin)));
560 else
561 direction = nodes[ni].dir;
562 }
563 if (nodes[ni].depth > 0) {
564 const float headingAlignment = std::clamp(dot(nodes[ni].heading, direction), -1.f, 1.f);
565 if (headingAlignment < cumulativeTurnCos) {
566 V3 tangent = sub(direction, mul(nodes[ni].heading, headingAlignment));
567 if (dot(tangent, tangent) > 1e-8f)
568 direction =
569 norm(add(mul(nodes[ni].heading, cumulativeTurnCos), mul(norm(tangent), cumulativeTurnSin)));
570 else
571 direction = nodes[ni].heading;
572 }
573 } else if (direction.y < -0.05f) {
574 // Primary limbs may spread horizontally, but do not launch downward.
575 direction = norm({direction.x, -0.05f, direction.z});
576 }
577 const V3 candidate = add(nodes[ni].p, mul(direction, settings.step));
578 bool separated = true;
579 for (size_t oi = 0; oi < nodes.size() && separated; ++oi)
580 separated = distanceSquared(candidate, nodes[oi].p) >= separationSquared;
581 for (size_t oi = 0; oi < additions.size() && separated; ++oi)
582 separated = distanceSquared(candidate, additions[oi].p) >= separationSquared;
583 if (separated) {
584 const V3 heading = nodes[ni].depth == 0 ? direction : nodes[ni].heading;
585 additions.push_back({candidate, direction, heading, int(ni), 1, 0, nodes[ni].depth + 1,
586 nodes[ni].remainingLength - settings.step, nodes[ni].vigor, nodes[ni].budHeight});
587 ++nodes[ni].children;
588 }
589 }
590 if (additions.empty()) break;
591 nodes.insert(nodes.end(), additions.begin(), additions.end());
592 }
593
594 for (int i = int(nodes.size()) - 1; i >= 0; --i) {
595 const int parent = nodes[size_t(i)].parent;
596 if (parent >= 0) {
597 nodes[size_t(parent)].descendants += nodes[size_t(i)].descendants;
598 }
599 }
600 const float twigRadius = settings.trunkRadius * 0.055f;
601 for (size_t i = 0; i < nodes.size(); ++i) {
602 const ColonyNode& node = nodes[i];
603 if (node.parent < 0) continue;
604 const ColonyNode& parent = nodes[size_t(node.parent)];
605 const float r0 = std::min(settings.trunkRadius * 0.34f * parent.vigor,
606 twigRadius * parent.vigor * (1.f + std::sqrt(float(parent.descendants))));
607 const float r1 = std::min(r0 * 0.92f, twigRadius * node.vigor * (1.f + std::sqrt(float(node.descendants))));
608 addTaperedCylinder(out, parent.p, node.p, r0, r1, std::max(3, settings.sides - 2));
609 if (node.children == 0) {
610 foliageAnchors.push_back({node.p, node.dir, 0.42f + 0.22f * node.vigor});
611 } else if (node.depth >= 2 && (node.depth % 2) == 0) {
612 // Older lower limbs carry foliage along the branch, whereas young upper
613 // shoots keep most foliage near their tips.
614 const float coverage =
615 settings.lowerLeafCoverage + (settings.upperLeafCoverage - settings.lowerLeafCoverage) * node.budHeight;
616 if (hash01(uint32_t(i) * 747796405u + 2891336453u) < coverage)
617 foliageAnchors.push_back({node.p, node.dir, 0.48f + 0.20f * (1.f - node.budHeight)});
618 }
619 }

685 .size = std::clamp(params.getFloat("clusterSize", 0.30f), 0.04f, 0.8f),

691 .caps = std::clamp(params.getInt("clusterCaps", 2), 0, 8),
692 .leavesPerPlane = std::clamp(params.getInt("clusterLeaves", 28), 1, 256),

698 const V3 root{0.f, 0.f, 0.f};

704 // A second broad bend avoids constraining the entire trunk to one plane.

711 .rootFlare = rootFlare,
712 .flareLength = height * rootFlareHeight,
713 .relief = barkRelief,
714 .seed = params.getSeed(),

716 .noiseScale = trunkNoiseScale,
717 .knots = trunkKnots,
718 .knotStrength = trunkKnotStrength,

751 const float levelScale = std::pow(0.62f, float(level));
752 std::vector<Tip> next;
753 for (int i = 0; i < count; ++i) {
754 const float h = foliageStart + (1.f - foliageStart) * (float(i) + 0.35f) / float(count);

845 const float r = crownRadius * randomRange(rng, lobeMin, lobeMax) * (0.72f + 0.28f * density) *

847 V3 c = sub(tip.p, mul(tip.dir, r * 0.22f));
848 c.y += r * randomRange(rng, 0.08f, 0.22f);
849 addCanopyBlob(out, c, {r, r * randomRange(rng, 0.72f, 1.15f), r}, realistic ? 8 : 4, realistic ? 12 : 7);
850 ++emitted;
851 }
852 }
