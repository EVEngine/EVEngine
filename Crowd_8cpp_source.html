42float wrapPi(float a) {
43 const float kTwoPi = 6.28318530717958647692f;
44 while (a > M_PI) a -= kTwoPi;
45 while (a < -M_PI) a += kTwoPi;

90 void rebuildGrid() {
91 const int n = int(actions.size());
92 if (n == 0) {
93 gridW = gridH = 0;
94 return;
95 }
96 gridCell = std::max(std::max(sepRadius, perceiveRadius), 1.f);
97 float minX = xs[0], maxX = xs[0], minY = ys[0], maxY = ys[0];
98 for (int i = 1; i < n; ++i) {
99 minX = std::min(minX, xs[size_t(i)]);
100 maxX = std::max(maxX, xs[size_t(i)]);
101 minY = std::min(minY, ys[size_t(i)]);
102 maxY = std::max(maxY, ys[size_t(i)]);
103 }
104 minX -= gridCell;
105 minY -= gridCell;
106 maxX += gridCell;
107 maxY += gridCell;
108 gridOriginX = minX;
109 gridOriginY = minY;
110 gridW = int((maxX - minX) / gridCell) + 1;
111 gridH = int((maxY - minY) / gridCell) + 1;
112
113 const size_t cells = size_t(gridW) * size_t(gridH);
114 cellCount.assign(cells, 0);
115 for (int i = 0; i < n; ++i) {
116 int cx = int((xs[size_t(i)] - minX) / gridCell);
117 int cy = int((ys[size_t(i)] - minY) / gridCell);
118 cx = std::clamp(cx, 0, gridW - 1);
119 cy = std::clamp(cy, 0, gridH - 1);
120 ++cellCount[size_t(cy * gridW + cx)];
121 }
122 cellStart.assign(cells, 0);
123 int running = 0;
124 for (size_t c = 0; c < cells; ++c) {
125 cellStart[c] = running;
126 running += cellCount[c];
127 }
128 cursor = cellStart;
129 sorted.resize(size_t(n));
130 for (int i = 0; i < n; ++i) {
131 int cx = int((xs[size_t(i)] - minX) / gridCell);
132 int cy = int((ys[size_t(i)] - minY) / gridCell);
133 cx = std::clamp(cx, 0, gridW - 1);
134 cy = std::clamp(cy, 0, gridH - 1);
135 const int c = cy * gridW + cx;
136 sorted[size_t(cursor[size_t(c)]++)] = i;
137 }
138 }

140 template <typename Fn>
141 void forEachNeighbor(float qx, float qy, float radius, Fn &&fn) const {
142 if (gridW <= 0 || gridH <= 0 || radius <= 0.f) return;
143 const float cell = gridCell;
144 const int x0 = std::max(0, int(std::floor((qx - radius - gridOriginX) / cell)));
145 const int x1 = std::min(gridW - 1, int(std::floor((qx + radius - gridOriginX) / cell)));
146 const int y0 = std::max(0, int(std::floor((qy - radius - gridOriginY) / cell)));
147 const int y1 = std::min(gridH - 1, int(std::floor((qy + radius - gridOriginY) / cell)));
148 const float r2 = radius * radius;
149 for (int cy = y0; cy <= y1; ++cy) {
150 const int rowBase = cy * gridW;
151 for (int cx = x0; cx <= x1; ++cx) {
152 const int c = rowBase + cx;
153 const int begin = cellStart[size_t(c)];
154 const int end = begin + cellCount[size_t(c)];
155 for (int k = begin; k < end; ++k) {
156 const int j = sorted[size_t(k)];
157 const float dx = xs[size_t(j)] - qx;
158 const float dy = ys[size_t(j)] - qy;
159 if (dx * dx + dy * dy <= r2) fn(j);
160 }
161 }
162 }

165 void stepAgents(float dt) {
166 const int n = int(actions.size());
167 const bool hasField = field.isBuilt();
168 for (int i = 0; i < n; ++i) {
169 const float x = xs[size_t(i)];
170 const float y = ys[size_t(i)];
171 const float maxSpeed = maxSpeeds[size_t(i)];
172 const int action = actions[size_t(i)];
173 float arriveK = 1.f; // flow 模式到达减速系数（1=全速，0=停）
174
175 // 基础期望方向。
176 float desX = 0.f, desY = 0.f;
177 if (action == kFlow) {
178 float fx = 0.f, fy = 0.f;
179 if (hasField) field.flowAtWorld(x, y, fx, fy);
180 // 接近目标时按积分代价线性减速（arrive），避免在目标格附近振荡。
181 float k = 1.f;
182 if (hasField) {
183 const float c = field.costAtWorld(x, y);
184 const float cell = std::max(field.getCellSize(), 1e-6f);
185 const float arriveCells = arriveRadius / cell;
186 if (c < CrowdField::kUnreachable && arriveCells > 0.f && c < arriveCells) {
187 k = c / arriveCells;
188 arriveK = k;
189 }
190 }
191 desX = fx * k;
192 desY = fy * k;
193 } else if (action == kSeek && hasTargets[size_t(i)]) {
194 const float tx = targetXs[size_t(i)] - x;
195 const float ty = targetYs[size_t(i)] - y;
196 const float d = std::sqrt(tx * tx + ty * ty);
197 if (d > 1e-3f) {
198 desX = tx / d;
199 desY = ty / d;
200 if (d < arriveRadius) {
201 const float k = d / arriveRadius;
202 desX *= k;
203 desY *= k;
204 }
205 }
206 } else if (action == kBoids && hasTargets[size_t(i)] && goalWeight > 0.f) {
207 const float tx = targetXs[size_t(i)] - x;
208 const float ty = targetYs[size_t(i)] - y;
209 const float d = std::sqrt(tx * tx + ty * ty);
210 if (d > 1e-3f) {
211 desX = tx / d * goalWeight;
212 desY = ty / d * goalWeight;
213 }
214 }
215 desX *= maxSpeed;
216 desY *= maxSpeed;
217
218 // Boids 三力。
219 if (sepWeight > 0.f || alignWeight > 0.f || cohesionWeight > 0.f) {
220 const float queryR = std::max(sepRadius, perceiveRadius);
221 float sx = 0.f, sy = 0.f;
222 float axv = 0.f, ayv = 0.f;
223 float cxv = 0.f, cyv = 0.f;
224 int alignN = 0, cohN = 0;
225 forEachNeighbor(x, y, queryR, [&](int j) {
226 const float dx = xs[size_t(j)] - x;
227 const float dy = ys[size_t(j)] - y;
228 const float d2 = dx * dx + dy * dy;
229 if (d2 <= 0.f) return;
230 if (sepWeight > 0.f && d2 <= sepRadius * sepRadius) {
231 const float d = std::sqrt(d2);
232 const float fall = 1.f - d / sepRadius;
233 sx += (x - xs[size_t(j)]) / d * fall;
234 sy += (y - ys[size_t(j)]) / d * fall;
235 }
236 if (alignWeight > 0.f && d2 <= perceiveRadius * perceiveRadius) {
237 axv += vxs[size_t(j)];
238 ayv += vys[size_t(j)];
239 ++alignN;
240 }
241 if (cohesionWeight > 0.f && d2 <= perceiveRadius * perceiveRadius) {
242 cxv += xs[size_t(j)];
243 cyv += ys[size_t(j)];
244 ++cohN;
245 }
246 });
247 desX += sx * sepWeight * maxSpeed;
248 desY += sy * sepWeight * maxSpeed;
249 if (alignN > 0) {
250 const float al = std::sqrt(axv * axv + ayv * ayv);
251 if (al > 1e-4f) {
252 desX += axv / al * alignWeight * maxSpeed;
253 desY += ayv / al * alignWeight * maxSpeed;
254 }
255 }
256 if (cohN > 0) {
257 const float cmx = cxv / float(cohN) - x;
258 const float cmy = cyv / float(cohN) - y;
259 const float cl = std::sqrt(cmx * cmx + cmy * cmy);
260 if (cl > 1e-4f) {
261 desX += cmx / cl * cohesionWeight * maxSpeed;
262 desY += cmy / cl * cohesionWeight * maxSpeed;
263 }
264 }
265 }
266
267 // Boids wander。
268 if (action == kBoids && wanderWeight > 0.f) {
269 wanderPhases[size_t(i)] += dt * 2.5f;
270 desX += std::cos(wanderPhases[size_t(i)]) * wanderWeight * maxSpeed;
271 desY += std::sin(wanderPhases[size_t(i)]) * wanderWeight * maxSpeed;
272 }
273
274 // 期望速度限幅。
275 float dl = std::sqrt(desX * desX + desY * desY);
276 if (dl > maxSpeed && dl > 1e-6f) {
277 desX *= maxSpeed / dl;
278 desY *= maxSpeed / dl;
279 }
280
281 // 加速度限幅。
282 float avx = vxs[size_t(i)];
283 float avy = vys[size_t(i)];
284 float ddx = desX - avx;
285 float ddy = desY - avy;
286 const float ddl = std::sqrt(ddx * ddx + ddy * ddy);
287 const float maxDelta = std::max(maxAccels[size_t(i)] * dt, 0.f);
288 if (ddl > maxDelta && ddl > 1e-6f) {
289 ddx *= maxDelta / ddl;
290 ddy *= maxDelta / ddl;
291 }
292 avx += ddx;
293 avy += ddy;
294
295 // 速度限幅。
296 float sp = std::sqrt(avx * avx + avy * avy);
297 if (sp > maxSpeed && sp > 1e-6f) {
298 avx *= maxSpeed / sp;
299 avy *= maxSpeed / sp;
300 sp = maxSpeed;
301 }
302 vxs[size_t(i)] = avx;
303 vys[size_t(i)] = avy;
304 speeds[size_t(i)] = sp;
305
306 // 接近目标时对速度做阻尼，快速消除目标点附近的来回振荡。
307 if (arriveK < 1.f) {
308 const float damp = 1.f - (1.f - arriveK) * std::min(1.f, dt * 4.f);
309 vxs[size_t(i)] *= damp;
310 vys[size_t(i)] *= damp;
311 speeds[size_t(i)] *= damp;
312 }
313
314 // 转向：heading 以 turnRate·dt 为上限向速度方向收敛。
315 const float spAfter = speeds[size_t(i)];
316 if (spAfter > 1e-4f) {
317 const float targetAng = std::atan2(vys[size_t(i)], vxs[size_t(i)]);
318 const float delta = wrapPi(targetAng - headings[size_t(i)]);
319 const float maxTurn = std::max(turnRates[size_t(i)] * dt, 0.f);
320 headings[size_t(i)] += std::clamp(delta, -maxTurn, maxTurn);
321 }
322
323 xs[size_t(i)] += vxs[size_t(i)] * dt;
324 ys[size_t(i)] += vys[size_t(i)] * dt;
325
326 // 钳制在场边界内。
327 if (clampToField && field.valid()) {
328 const float cs = field.getCellSize();
329 const float loX = field.getOriginX() - cs * 0.5f;
330 const float hiX = field.getOriginX() + (float(field.getWidth()) - 0.5f) * cs;
331 const float loY = field.getOriginY() - cs * 0.5f;
332 const float hiY = field.getOriginY() + (float(field.getHeight()) - 0.5f) * cs;
333 xs[size_t(i)] = std::clamp(xs[size_t(i)], loX, hiX);
334 ys[size_t(i)] = std::clamp(ys[size_t(i)], loY, hiY);
335 }
336 }
337 }

339 void resolveOverlapsPass() {
340 const int n = int(actions.size());
341 for (int i = 0; i < n; ++i) {
342 const float xi = xs[size_t(i)];
343 const float yi = ys[size_t(i)];
344 forEachNeighbor(xi, yi, sepRadius, [&](int j) {
345 if (j <= i) return;
346 float dx = xs[size_t(j)] - xi;
347 float dy = ys[size_t(j)] - yi;
348 float d2 = dx * dx + dy * dy;
349 const float minD = radii[size_t(i)] + radii[size_t(j)];
350 if (d2 >= minD * minD) return;
351 const bool exactOverlap = d2 <= 1e-9f;
352 if (exactOverlap) {
353 const std::string left = stableIds[size_t(i)].empty()
354 ? std::to_string(i) : stableIds[size_t(i)];
355 const std::string right = stableIds[size_t(j)].empty()
356 ? std::to_string(j) : stableIds[size_t(j)];
357 const std::string& first = left < right ? left : right;
358 const std::string& second = left < right ? right : left;
359 std::uint64_t hash = 1469598103934665603ULL;
360 for (const unsigned char value : first + "\n" + second) {
361 hash ^= value;
362 hash *= 1099511628211ULL;
363 }
364 const float angle = static_cast<float>(hash % 104729ULL) /
365 104729.0f * 6.28318530717958647692f;
366 dx = std::cos(angle);
367 dy = std::sin(angle);
368 if (left > right) { dx = -dx; dy = -dy; }
369 d2 = 1.0f;
370 }
371 const float d = std::sqrt(d2);
372 const float push = minD - (exactOverlap ? 0.0f : d);
373 const float nx = dx / d;
374 const float ny = dy / d;
375 float leftShare = 0.5f;
376 float rightShare = 0.5f;
377 if (avoidancePriorities[size_t(i)] > avoidancePriorities[size_t(j)]) {
378 leftShare = 0.0f; rightShare = 1.0f;
379 } else if (avoidancePriorities[size_t(j)] > avoidancePriorities[size_t(i)]) {
380 leftShare = 1.0f; rightShare = 0.0f;
381 }
382 xs[size_t(i)] -= nx * push * leftShare;
383 ys[size_t(i)] -= ny * push * leftShare;
384 xs[size_t(j)] += nx * push * rightShare;
385 ys[size_t(j)] += ny * push * rightShare;
386 });
387 }
388 }

390 void resolveWalls() {
391 if (!field.valid()) return;
392 const int n = int(actions.size());
393 for (int i = 0; i < n; ++i) {
394 // 墙角处一次推出可能仍与相邻阻挡格重叠，迭代 2 次。
395 for (int pass = 0; pass < 2; ++pass) {
396 if (!field.resolvePenetration(xs[size_t(i)], ys[size_t(i)], radii[size_t(i)]))
397 break;
398 }
399 }
400 }

461int Crowd::addAgent(float x, float y, float heading, float radius) {
462 auto &d = *impl_;
463 if (int(d.actions.size()) >= d.maxAgents) return -1;
464 const int id = int(d.xs.size());
465 d.xs.push_back(x);
466 d.ys.push_back(y);
467 d.headings.push_back(heading);
468 d.vxs.push_back(0.f);
469 d.vys.push_back(0.f);
470 d.speeds.push_back(0.f);
471 d.radii.push_back(radius > 0.f ? radius : d.defaultRadius);
472 d.maxSpeeds.push_back(d.defaultSpeed);
473 d.maxAccels.push_back(d.defaultSpeed * 2.f);
474 d.turnRates.push_back(d.defaultTurnRate);
475 d.actions.push_back(kFlow);
476 d.datas.push_back(0);
477 d.avoidancePriorities.push_back(0);
478 d.hasTargets.push_back(0);
479 d.targetXs.push_back(0.f);
480 d.targetYs.push_back(0.f);
481 d.wanderPhases.push_back(float(id) * 2.399963f);
482 d.stableIds.emplace_back();
483 return id;
484}

514bool Crowd::removeAgent(int id) {
515 auto &d = *impl_;
516 if (!d.validId(id)) return false;
517 const int last = int(d.xs.size()) - 1;
518 const std::string removedStableId = d.stableIds[static_cast<size_t>(id)];
519 if (id != last) {
520 d.xs[size_t(id)] = d.xs[size_t(last)];
521 d.ys[size_t(id)] = d.ys[size_t(last)];
522 d.headings[size_t(id)] = d.headings[size_t(last)];
523 d.vxs[size_t(id)] = d.vxs[size_t(last)];
524 d.vys[size_t(id)] = d.vys[size_t(last)];
525 d.speeds[size_t(id)] = d.speeds[size_t(last)];
526 d.radii[size_t(id)] = d.radii[size_t(last)];
