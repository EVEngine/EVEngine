15Vertex mix(const Vertex& a, const Vertex& b, float t) {
16 Vertex r{};
17 const float* pa = &a.x;
18 const float* pb = &b.x;
19 float* pr = &r.x;

23template <class Distance>
24std::vector<Vertex> clip(const std::vector<Vertex>& in, Distance distance) {
25 std::vector<Vertex> out;
26 if (in.empty()) return out;
27 out.reserve(in.size() + 1);
28 Vertex previous = in.back();
29 float dp = distance(previous);
30 bool previousInside = dp >= 0.f;
31 for (const auto& current : in) {
32 const float dc = distance(current);
33 const bool currentInside = dc >= 0.f;
34 if (currentInside != previousInside) {
35 const float denominator = dp - dc;
36 const float t = std::abs(denominator) > 1e-20f ? dp / denominator : 0.f;
37 out.push_back(mix(previous, current, std::clamp(t, 0.f, 1.f)));
38 }
39 if (currentInside) out.push_back(current);
40 previous = current;
41 dp = dc;
42 previousInside = currentInside;
43 }
44 return out;
45}

53bool nonDegenerate(const Vertex& a, const Vertex& b, const Vertex& c) {
54 const double abx = double(b.x) - a.x, aby = double(b.y) - a.y, abz = double(b.z) - a.z;
55 const double acx = double(c.x) - a.x, acy = double(c.y) - a.y, acz = double(c.z) - a.z;
56 const double nx = aby * acz - abz * acy, ny = abz * acx - abx * acz, nz = abx * acy - aby * acx;
57 return nx * nx + ny * ny + nz * nz > 1e-24;
58}

77Result<GtsMeshSplitResult> splitGtsMesh(const MeshBuild& source, int xSplits, int zSplits, GtsMeshPivot pivot) {
78 const int vertexCount = source.getVertexCount(), indexCount = source.getIndexCount();
79 if (vertexCount < 3 || indexCount < 3 || indexCount % 3 != 0 || xSplits < 0 || zSplits < 0 || xSplits > 255 ||
80 zSplits > 255 || static_cast<int>(pivot) < 0 || static_cast<int>(pivot) > 2)
81 return fail<GtsMeshSplitResult>("GTS mesh split requires a valid triangle mesh, split counts and pivot");
82 float minX = std::numeric_limits<float>::max(), maxX = -minX, minZ = minX, maxZ = -minX;
83 for (int i = 0; i < vertexCount; ++i) {
84 Vertex v{source.getPositionX(i), source.getPositionY(i), source.getPositionZ(i), source.getNormalX(i),
85 source.getNormalY(i), source.getNormalZ(i), source.getUvU(i), source.getUvV(i),
86 source.getColor(i, 0), source.getColor(i, 1), source.getColor(i, 2), source.getColor(i, 3)};
87 if (!finiteVertex(v)) return fail<GtsMeshSplitResult>("GTS mesh streams must contain only finite values");
88 minX = std::min(minX, v.x);
89 maxX = std::max(maxX, v.x);
90 minZ = std::min(minZ, v.z);
91 maxZ = std::max(maxZ, v.z);
92 }
93 for (int i = 0; i < indexCount; ++i)
94 if (source.getIndex(i) < 0 || source.getIndex(i) >= vertexCount)
95 return fail<GtsMeshSplitResult>("GTS mesh index is out of range");
96 if (!(maxX > minX) || !(maxZ > minZ)) return fail<GtsMeshSplitResult>("GTS mesh must have non-zero X/Z bounds");
97 GtsMeshSplitResult result;
98 result.columns_ = xSplits + 1;
99 result.rows_ = zSplits + 1;
100 result.tiles_.resize(static_cast<std::size_t>(result.columns_ * result.rows_));
101 const float cellX = (maxX - minX) / result.columns_, cellZ = (maxZ - minZ) / result.rows_;
102 for (int row = 0; row < result.rows_; ++row)
103 for (int column = 0; column < result.columns_; ++column) {
104 const float loX = minX + column * cellX,
105 hiX = column + 1 == result.columns_ ? maxX : minX + (column + 1) * cellX;
106 const float loZ = minZ + row * cellZ, hiZ = row + 1 == result.rows_ ? maxZ : minZ + (row + 1) * cellZ;
107 auto& tile = result.tiles_[static_cast<std::size_t>(row * result.columns_ + column)];
108 float pivotX = 0.f, pivotZ = 0.f;
109 if (pivot == GtsMeshPivot::MinimumXZ) {
110 pivotX = loX;
111 pivotZ = loZ;
112 } else if (pivot == GtsMeshPivot::CenterXZ) {
113 pivotX = (loX + hiX) * .5f;
114 pivotZ = (loZ + hiZ) * .5f;
115 }
116 tile.offsetX = pivotX;
117 tile.offsetZ = pivotZ;
118 std::vector<float> tileColors;
119 for (int i = 0; i < indexCount; i += 3) {
120 std::vector<Vertex> polygon;
121 polygon.reserve(7);
122 for (int k = 0; k < 3; ++k) {
123 const int n = source.getIndex(i + k);
124 polygon.push_back({source.getPositionX(n), source.getPositionY(n), source.getPositionZ(n),
125 source.getNormalX(n), source.getNormalY(n), source.getNormalZ(n),
126 source.getUvU(n), source.getUvV(n), source.getColor(n, 0), source.getColor(n, 1),
127 source.getColor(n, 2), source.getColor(n, 3)});
128 }
129 polygon = clip(polygon, [&](const Vertex& v) { return v.x - loX; });
130 polygon = clip(polygon, [&](const Vertex& v) { return hiX - v.x; });
131 polygon = clip(polygon, [&](const Vertex& v) { return v.z - loZ; });
132 polygon = clip(polygon, [&](const Vertex& v) { return hiZ - v.z; });
133 if (polygon.size() < 3) continue;
134 const int group = source.getTriangleGroup(i / 3);
135 if (group >= 0)
136 tile.mesh.setActiveGroup(source.getGroupName(group));
137 else
138 tile.mesh.setActiveGroup("");
139 const auto add = [&](const Vertex& v) {
140 const auto n = static_cast<std::uint32_t>(tile.mesh.getVertexCount());
141 tile.mesh.addVertex(v.x - pivotX, v.y, v.z - pivotZ, v.nx, v.ny, v.nz, v.u, v.v);
142 if (source.hasVertexColors()) tileColors.insert(tileColors.end(), {v.r, v.g, v.b, v.a});
143 return n;
144 };
145 for (std::size_t k = 1; k + 1 < polygon.size(); ++k) {
146 if (!nonDegenerate(polygon[0], polygon[k], polygon[k + 1])) continue;
147 tile.mesh.addTriangle(add(polygon[0]), add(polygon[k]), add(polygon[k + 1]));
148 }
149 }
150 if (!tileColors.empty())
151 tile.mesh.setVertexColors(std::move(tileColors)).ignore("validated clipped color stream");
152 }
153 return Result<GtsMeshSplitResult>::success(std::move(result));
154}
