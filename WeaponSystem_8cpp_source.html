98void updateFeel(WeaponEntity& w, const WeaponDefinition* def, float dt) {
99 auto state = w.state();
100 // 散布回稳：从当前连射散布衰减回 spreadMin。
101 if (def->spreadRecover > 0.f && state->currentSpread > def->spreadMin) {
102 state->currentSpread = std::max(def->spreadMin, state->currentSpread - def->spreadRecover * dt);
103 }
104 // 后坐回正：向 0 收敛。
105 if (def->recoilRecover > 0.f) {
106 const float p = std::min(std::fabs(state->recoilPitch), def->recoilRecover * dt);
107 state->recoilPitch += state->recoilPitch > 0.f ? -p : p;
108 const float y = std::min(std::fabs(state->recoilYaw), def->recoilRecover * dt);
109 state->recoilYaw += state->recoilYaw > 0.f ? -y : y;
110 }
111}

114float applyBloom(WeaponEntity& w, const WeaponDefinition* def) {
115 auto state = w.state();
116 if (def->spreadPerShot > 0.f && def->spreadMax > def->spreadMin) {
117 state->currentSpread = std::min(def->spreadMax, state->currentSpread + def->spreadPerShot);
118 }
119 return state->currentSpread;
120}

157bool canFireRanged(WeaponEntity& w, const WeaponDefinition* def) {
158 const auto state = w.state();
159 if (state->resource.reloading || state->jammed) return false;
160 if (state->cooldown > 0.f) return false;
161 if (state->resource.value <= 0.f) return false;
162 const IWeaponLogic* logic = WeaponSystem::findLogic(def->logic);
163 return logic == nullptr || logic->canFire(w);
164}

166bool tryFireRanged(WeaponEntity& w, const AttackRequest& req, const WeaponDefinition* def) {
167 if (!canFireRanged(w, def)) return false;
168
169 auto state = w.state();
170 state->resource.value -= 1.f;
171
172 const float effSpread = applyBloom(w, def);
173 state->recoilPitch += def->recoilPitch;
174 state->recoilYaw += def->recoilYaw;
175
176 AttackRequest r = req;
177 r.yaw = w.aim()->yaw;
178 r.pitch = w.aim()->pitch;
179 r.spread = effSpread;
180 const int pellets = spawnPellets(w, def, r);
181
182 state->cooldown = def->cooldown;
183 if (state->selector == FireMode::Burst && def->burstSize > 1) {
184 state->burstRemaining = def->burstSize - 1;
185 state->burstTimer = def->burstInterval;
186 }
187
188 r.pelletCount = pellets;
189 pushFireEvent(w, def, r, static_cast<int>(state->resource.value));
190 if (state->resource.value <= 0.f) pushEmptyEvent(w, def);
191 return true;
192}

194void updateRanged(WeaponEntity& w, const WeaponDefinition* def, float dt) {
195 auto state = w.state();
196 Resource& r = state->resource;
197 state->cooldown = std::max(0.f, state->cooldown - dt);
198 updateFeel(w, def, dt);
199
200 // 自动装填：弹匣打空、冷却结束且有备弹时自动开始（事件照发，游戏侧可覆盖/取消）。
201 if (!r.reloading && r.value <= 0.f && state->cooldown <= 0.f && !state->jammed) {
202 WeaponSystem::startReload(w);
203 }
204
205 // 装填推进
206 if (r.reloading) {
207 r.reloadProgress += dt;
208 if (r.reloadProgress >= def->reloadTime) {
209 const int missing = static_cast<int>(r.max) - static_cast<int>(r.value);
210 if (def->reserveSize < 0) {
211 r.value = r.max;
212 } else if (AmmoPoolEntity* pool = state->ammoPool) {
213 const int take = std::min(missing, pool->state()->count);
214 r.value += static_cast<float>(take);
215 pool->state()->count -= take;
216 } else {
217 const int take = std::min(missing, r.reserve);
218 r.value += static_cast<float>(take);
219 r.reserve -= take;
220 }
221 r.reloading = false;
222 r.reloadProgress = 0.f;
223 pushReloadEnd(w, def, static_cast<int>(r.value));
224 }
225 }
226
227 // 连发续射（第一次扣弹在 tryFire，续发在这里）
228 if (!r.reloading && state->burstRemaining > 0) {
229 state->burstTimer -= dt;
230 if (state->burstTimer <= 0.f) {
231 if (r.value > 0.f) {
232 r.value -= 1.f;
233 AttackRequest req;
234 req.yaw = w.aim()->yaw;
235 req.pitch = w.aim()->pitch;
236 req.spread = applyBloom(w, def);
237 state->recoilPitch += def->recoilPitch;
238 state->recoilYaw += def->recoilYaw;
239 const int pellets = spawnPellets(w, def, req);
240 state->cooldown = def->cooldown;
241 state->burstRemaining--;
242 state->burstTimer = def->burstInterval;
243
244 req.pelletCount = pellets;
245 pushFireEvent(w, def, req, static_cast<int>(r.value));
246
247 if (r.value <= 0.f) {
248 state->burstRemaining = 0;
249 pushEmptyEvent(w, def);
250 }
251 } else {
252 state->burstRemaining = 0;
253 }
254 }
255 }
256
257 if (IWeaponLogic* logic = WeaponSystem::findLogic(def->logic)) logic->update(w, dt);
258 WeaponSystem::updateAim(w, dt);
259}

265bool canAttackStage(WeaponEntity& w, const WeaponDefinition* def) {
266 const auto state = w.state();
267 if (state->stage != AttackStage::Idle) return false;
268 if (state->jammed) return false;
269 if (state->cooldown > 0.f) return false;
270 const Resource& r = state->resource;
271 if (r.kind == ResourceKind::Mana || r.kind == ResourceKind::Stamina) {
272 const auto attribute = r.kind == ResourceKind::Mana ? WeaponAttributeAdapter::manaAttribute
273 : WeaponAttributeAdapter::staminaAttribute;
274 auto current = WeaponAttributeAdapter::read(w, attribute);
275 if (!current) {
276 current.ignore("legacy bool canFire cannot expose weapon attribute diagnostics");
277 return false;
278 }
279 if (!r.infinite && current.value() < static_cast<double>(r.cost)) return false;
280 } else if (!r.infinite && r.value < r.cost) {
281 return false;
282 }
283 const IWeaponLogic* logic = WeaponSystem::findLogic(def->logic);
284 return logic == nullptr || logic->canFire(w);
285}

287bool tryAttackStage(WeaponEntity& w, const AttackRequest& req, const WeaponDefinition* def) {
288 if (!canAttackStage(w, def)) return false;
289
290 auto state = w.state();
291 Resource& r = state->resource;
292 if (r.kind == ResourceKind::Mana || r.kind == ResourceKind::Stamina) {
293 auto consumed = WeaponAttributeAdapter::consumeTriggerResource(w);
294 if (!consumed) {
295 consumed.ignore("legacy bool tryFire cannot expose weapon attribute diagnostics");
296 return false;
297 }
298 } else if (!r.infinite) {
299 r.value = std::max(0.f, r.value - r.cost);
300 }
301
302 state->stage = AttackStage::Windup;
303 state->stageTimer = 0.f;
304 state->lastRequest = req;
305 if (IWeaponLogic* logic = WeaponSystem::findLogic(def->logic)) logic->begin(w, req);
306
307 WeaponEvent e;
308 e.type = WeaponEventType::WindupStart;
309 e.weaponId = w.identity()->id;
310 e.defId = def->id;
311 e.ammoLeft = static_cast<int>(r.value);
312 pushEvent(e);
313 return true;
314}

316void updateStage(WeaponEntity& w, const WeaponDefinition* def, float dt) {
317 auto state = w.state();
318 state->cooldown = std::max(0.f, state->cooldown - dt);
319 updateFeel(w, def, dt);
320
321 // 资源回复（法力 / 充能）
322 Resource& r = state->resource;
323 if (r.regen > 0.f && r.value < r.max) r.value = std::min(r.max, r.value + r.regen * dt);
324
325 const AttackStageSpec* stages = state->stages != nullptr ? state->stages : &def->stages;
326 state->stageTimer += dt;
327
328 if (state->stage == AttackStage::Windup && state->stageTimer >= stages->windupTime) {
329 // Windup → Active：命中/生效时刻
330 state->stage = AttackStage::Active;
331 state->stageTimer = 0.f;
332 AttackRequest req = state->lastRequest;
333 req.yaw = w.aim()->yaw;
334 req.pitch = w.aim()->pitch;
335 req.spread = applyBloom(w, def);
336 const int pellets = spawnPellets(w, def, req);
337 req.pelletCount = pellets;
338 pushFireEvent(w, def, req, static_cast<int>(r.value));
339 if (!r.infinite && r.value <= 0.f) pushEmptyEvent(w, def);
340 } else if (state->stage == AttackStage::Active) {
341 if (IWeaponLogic* logic = WeaponSystem::findLogic(def->logic)) logic->channel(w, state->lastRequest, dt);
342 if (state->stageTimer >= stages->activeTime) {
343 state->stage = AttackStage::Recover;
344 state->stageTimer = 0.f;
345 if (IWeaponLogic* logic = WeaponSystem::findLogic(def->logic)) logic->end(w, state->lastRequest);
346 WeaponEvent e;
347 e.type = WeaponEventType::AttackEnd;
348 e.weaponId = w.identity()->id;
349 e.defId = def->id;
350 e.ammoLeft = static_cast<int>(r.value);
351 pushEvent(e);
352 }
353 } else if (state->stage == AttackStage::Recover && state->stageTimer >= stages->recoverTime) {
354 state->stage = AttackStage::Idle;
355 state->stageTimer = 0.f;
356 }
357
358 if (IWeaponLogic* logic = WeaponSystem::findLogic(def->logic)) logic->update(w, dt);
359 WeaponSystem::updateAim(w, dt);
360}

433bool WeaponSystem::updateAim(WeaponEntity& w, float dt) {
434 auto aim = w.aim();
435 if (aim->turnSpeed <= 0.f) {
436 aim->yaw = aim->desiredYaw;
437 aim->pitch = aim->desiredPitch;
438 return true;
439 }
440 const float step = aim->turnSpeed * dt;
441 const float dyaw = angleDelta(aim->yaw, aim->desiredYaw);
442 const float dpitch = aim->desiredPitch - aim->pitch;
443 aim->yaw += (std::fabs(dyaw) <= step) ? dyaw : (dyaw > 0.f ? step : -step);
444 aim->pitch += (std::fabs(dpitch) <= step) ? dpitch : (dpitch > 0.f ? step : -step);
445 return std::fabs(angleDelta(aim->yaw, aim->desiredYaw)) < 0.01f &&
446 std::fabs(aim->pitch - aim->desiredPitch) < 0.01f;
447}
