111void updateFeel(WeaponEntity& w, const WeaponDefinition* def, float dt) {
112 auto state = w.state();
113 // 散布回稳：从当前连射散布衰减回 spreadMin。
114 if (def->spreadRecover > 0.f && state->currentSpread > def->spreadMin) {
115 state->currentSpread = std::max(def->spreadMin, state->currentSpread - def->spreadRecover * dt);
116 }
117 // 后坐回正：向 0 收敛。
118 if (def->recoilRecover > 0.f) {
119 const float p = std::min(std::fabs(state->recoilPitch), def->recoilRecover * dt);
120 state->recoilPitch += state->recoilPitch > 0.f ? -p : p;
121 const float y = std::min(std::fabs(state->recoilYaw), def->recoilRecover * dt);
122 state->recoilYaw += state->recoilYaw > 0.f ? -y : y;
123 }
124}

127float applyBloom(WeaponEntity& w, const WeaponDefinition* def) {
128 auto state = w.state();
129 if (def->spreadPerShot > 0.f && def->spreadMax > def->spreadMin) {
130 state->currentSpread = std::min(def->spreadMax, state->currentSpread + def->spreadPerShot);
131 }
132 return state->currentSpread;
133}

170bool canFireRanged(WeaponEntity& w, const WeaponDefinition* def) {
171 const auto state = w.state();
172 if (state->resource.reloading || state->jammed) return false;
173 if (state->cooldown > 0.f) return false;
174 if (state->resource.value <= 0.f) return false;
175 const IWeaponLogic* logic = WeaponSystem::findLogic(def->logic);
176 return logic == nullptr || logic->canFire(w);
177}
178
179bool tryFireRanged(WeaponEntity& w, const AttackRequest& req, const WeaponDefinition* def) {
180 if (!canFireRanged(w, def)) return false;
181
182 auto state = w.state();
183 state->resource.value -= 1.f;
184
185 const float effSpread = applyBloom(w, def);
186 state->recoilPitch += def->recoilPitch;
187 state->recoilYaw += def->recoilYaw;
188
189 AttackRequest r = req;
190 r.yaw = w.aim()->yaw;
191 r.pitch = w.aim()->pitch;
192 r.spread = effSpread;
193 const int pellets = spawnPellets(w, def, r);
194
195 state->cooldown = def->cooldown;
196 if (state->selector == FireMode::Burst && def->burstSize > 1) {
197 state->burstRemaining = def->burstSize - 1;
198 state->burstTimer = def->burstInterval;
199 }
200
201 r.pelletCount = pellets;
202 pushFireEvent(w, def, r, static_cast<int>(state->resource.value));
203 if (state->resource.value <= 0.f) pushEmptyEvent(w, def);
204 return true;
205}

207void updateRanged(WeaponEntity& w, const WeaponDefinition* def, float dt) {
208 auto state = w.state();
209 Resource& r = state->resource;
210 state->cooldown = std::max(0.f, state->cooldown - dt);
211 updateFeel(w, def, dt);
212
213 // 自动装填：弹匣打空、冷却结束且有备弹时自动开始（事件照发，游戏侧可覆盖/取消）。
214 if (!r.reloading && r.value <= 0.f && state->cooldown <= 0.f && !state->jammed) {
215 WeaponSystem::startReload(w);
216 }
217
218 // 装填推进
219 if (r.reloading) {
220 r.reloadProgress += dt;
221 if (r.reloadProgress >= def->reloadTime) {
222 const int missing = static_cast<int>(r.max) - static_cast<int>(r.value);
223 if (def->reserveSize < 0) {
224 r.value = r.max;
225 } else if (AmmoPoolEntity* pool = state->ammoPool) {
226 const int take = std::min(missing, pool->state()->count);
227 r.value += static_cast<float>(take);
228 pool->state()->count -= take;
229 } else {
230 const int take = std::min(missing, r.reserve);
231 r.value += static_cast<float>(take);
232 r.reserve -= take;
233 }
234 r.reloading = false;
235 r.reloadProgress = 0.f;
236 pushReloadEnd(w, def, static_cast<int>(r.value));
237 }
238 }
239
240 // 连发续射（第一次扣弹在 tryFire，续发在这里）
241 if (!r.reloading && state->burstRemaining > 0) {
242 state->burstTimer -= dt;
243 if (state->burstTimer <= 0.f) {
244 if (r.value > 0.f) {
245 r.value -= 1.f;
246 AttackRequest req;
247 req.yaw = w.aim()->yaw;
248 req.pitch = w.aim()->pitch;
249 req.spread = applyBloom(w, def);
250 state->recoilPitch += def->recoilPitch;
251 state->recoilYaw += def->recoilYaw;
252 const int pellets = spawnPellets(w, def, req);
253 state->cooldown = def->cooldown;
254 state->burstRemaining--;
255 state->burstTimer = def->burstInterval;
256
257 req.pelletCount = pellets;
258 pushFireEvent(w, def, req, static_cast<int>(r.value));
259
260 if (r.value <= 0.f) {
261 state->burstRemaining = 0;
262 pushEmptyEvent(w, def);
263 }
264 } else {
265 state->burstRemaining = 0;
266 }
267 }
268 }
269
270 if (IWeaponLogic* logic = WeaponSystem::findLogic(def->logic)) logic->update(w, dt);
271 WeaponSystem::updateAim(w, dt);
272}

278bool canAttackStage(WeaponEntity& w, const WeaponDefinition* def) {
279 const auto state = w.state();
280 if (state->stage != AttackStage::Idle) return false;
281 if (state->jammed) return false;
282 if (state->cooldown > 0.f) return false;
283 const Resource& r = state->resource;
284 if (r.kind == ResourceKind::Mana || r.kind == ResourceKind::Stamina) {
285 const auto attribute = r.kind == ResourceKind::Mana ? WeaponAttributeAdapter::manaAttribute
286 : WeaponAttributeAdapter::staminaAttribute;
287 auto current = WeaponAttributeAdapter::read(w, attribute);
288 if (!current) {
289 current.ignore("legacy bool canFire cannot expose weapon attribute diagnostics");
290 return false;
291 }
292 if (!r.infinite && current.value() < static_cast<double>(r.cost)) return false;
293 } else if (!r.infinite && r.value < r.cost) {
294 return false;
295 }
296 const IWeaponLogic* logic = WeaponSystem::findLogic(def->logic);
297 return logic == nullptr || logic->canFire(w);
298}
299
300bool tryAttackStage(WeaponEntity& w, const AttackRequest& req, const WeaponDefinition* def) {
301 if (!canAttackStage(w, def)) return false;
302
303 auto state = w.state();
304 Resource& r = state->resource;
305 if (r.kind == ResourceKind::Mana || r.kind == ResourceKind::Stamina) {
306 auto consumed = WeaponAttributeAdapter::consumeTriggerResource(w);
307 if (!consumed) {
308 consumed.ignore("legacy bool tryFire cannot expose weapon attribute diagnostics");
309 return false;
310 }
311 } else if (!r.infinite) {
312 r.value = std::max(0.f, r.value - r.cost);
313 }
314
315 state->stage = AttackStage::Windup;
316 state->stageTimer = 0.f;
317 state->lastRequest = req;
318 if (IWeaponLogic* logic = WeaponSystem::findLogic(def->logic)) logic->begin(w, req);
319
320 WeaponEvent e;
321 e.type = WeaponEventType::WindupStart;
322 e.weaponId = w.identity()->id;
323 e.defId = def->id;
324 e.ammoLeft = static_cast<int>(r.value);
325 pushEvent(e);
326 return true;
327}

329void updateStage(WeaponEntity& w, const WeaponDefinition* def, float dt) {
330 auto state = w.state();
331 state->cooldown = std::max(0.f, state->cooldown - dt);
332 updateFeel(w, def, dt);
333
334 // 资源回复（法力 / 充能）
335 Resource& r = state->resource;
336 if (r.regen > 0.f && r.value < r.max) r.value = std::min(r.max, r.value + r.regen * dt);
337
338 const AttackStageSpec* stages = state->stages != nullptr ? state->stages : &def->stages;
339 state->stageTimer += dt;
340
341 if (state->stage == AttackStage::Windup && state->stageTimer >= stages->windupTime) {
342 // Windup → Active：命中/生效时刻
343 state->stage = AttackStage::Active;
344 state->stageTimer = 0.f;
345 AttackRequest req = state->lastRequest;
346 req.yaw = w.aim()->yaw;
347 req.pitch = w.aim()->pitch;
348 req.spread = applyBloom(w, def);
349 const int pellets = spawnPellets(w, def, req);
350 req.pelletCount = pellets;
351 pushFireEvent(w, def, req, static_cast<int>(r.value));
352 if (!r.infinite && r.value <= 0.f) pushEmptyEvent(w, def);
353 } else if (state->stage == AttackStage::Active) {
354 if (IWeaponLogic* logic = WeaponSystem::findLogic(def->logic)) logic->channel(w, state->lastRequest, dt);
355 if (state->stageTimer >= stages->activeTime) {
356 state->stage = AttackStage::Recover;
357 state->stageTimer = 0.f;
358 if (IWeaponLogic* logic = WeaponSystem::findLogic(def->logic)) logic->end(w, state->lastRequest);
359 WeaponEvent e;
360 e.type = WeaponEventType::AttackEnd;
361 e.weaponId = w.identity()->id;
362 e.defId = def->id;
363 e.ammoLeft = static_cast<int>(r.value);
364 pushEvent(e);
365 }
366 } else if (state->stage == AttackStage::Recover && state->stageTimer >= stages->recoverTime) {
367 state->stage = AttackStage::Idle;
368 state->stageTimer = 0.f;
369 }
370
371 if (IWeaponLogic* logic = WeaponSystem::findLogic(def->logic)) logic->update(w, dt);
372 WeaponSystem::updateAim(w, dt);
373}

417void WeaponSystem::startReload(WeaponEntity& w) {
418 const WeaponDefinition* def = defOf(w);
419 if (def == nullptr) return;
420 auto state = w.state();

436 e.defId = def->id;

446bool WeaponSystem::updateAim(WeaponEntity& w, float dt) {
447 auto aim = w.aim();
448 if (aim->turnSpeed <= 0.f) {
449 aim->yaw = aim->desiredYaw;
450 aim->pitch = aim->desiredPitch;
451 return true;
452 }
453 const float step = aim->turnSpeed * dt;
454 const float dyaw = angleDelta(aim->yaw, aim->desiredYaw);
455 const float dpitch = aim->desiredPitch - aim->pitch;
456 aim->yaw += (std::fabs(dyaw) <= step) ? dyaw : (dyaw > 0.f ? step : -step);
457 aim->pitch += (std::fabs(dpitch) <= step) ? dpitch : (dpitch > 0.f ? step : -step);
458 return std::fabs(angleDelta(aim->yaw, aim->desiredYaw)) < 0.01f &&
459 std::fabs(aim->pitch - aim->desiredPitch) < 0.01f;
460}

503class ScriptProjectileService final : public IProjectileService {
504public:
505 explicit ScriptProjectileService(ScriptProviderHandle handle) : handle_(std::move(handle)) {}
506
508 [[nodiscard]] const ScriptProviderHandle& handle() const noexcept { return handle_; }
509
510 void spawnProjectile(const WeaponEntity& w, const AttackRequest& req) override {
511 // 接口借出的是 const 引用（提供方不该改武器），而 ECS 的组件访问器按宏设计就是
512 // 非 const（`external/ECS.hpp` 的 `COMPONENT(T,name)`）。这里只转一次、只读：
513 // 下面不写 `w` 的任何字段，并在脚本运行前把需要的值全部拷进 ScriptArg。
514 WeaponEntity& mutableWeapon = const_cast<WeaponEntity&>(w);
515 const auto identity = mutableWeapon.identity();
516
517 const ScriptArg args[] = {
518 ScriptArg::fromText(identity->id), ScriptArg::fromText(identity->defId),
519 ScriptArg::fromInteger(identity->ownerId), ScriptArg::fromNumber(double(req.yaw)),
520 ScriptArg::fromNumber(double(req.pitch)), ScriptArg::fromInteger(req.shooterId),
521 ScriptArg::fromBoolean(req.hasTarget), ScriptArg::fromNumber(double(req.targetX)),
522 ScriptArg::fromNumber(double(req.targetY)), ScriptArg::fromNumber(double(req.targetZ)),
523 ScriptArg::fromNumber(double(req.muzzleX)), ScriptArg::fromNumber(double(req.muzzleY)),
524 ScriptArg::fromNumber(double(req.muzzleZ)), ScriptArg::fromNumber(double(req.spread)),
525 ScriptArg::fromInteger(req.pelletIndex), ScriptArg::fromInteger(req.pelletCount),
526 };
527
528 auto spawned = ScriptProviderRegistry::instance().call(handle_, "spawnProjectile", args);
529 if (!spawned) {
530 // Fail visibly, once per generation: a provider that raises or declares the wrong
531 // arity means "no projectile appeared", and silence would look like a weapon that
532 // simply does not fire.
533 if (reportedGeneration_ != handle_.generation) {
534 reportedGeneration_ = handle_.generation;
535 recordLogEvent("error", "projectile provider '" + handle_.name +
536 "' failed and no projectile was spawned: " + spawned.status().describe());
537 }
538 }
539 }
540
541private:
542 ScriptProviderHandle handle_;
544 std::uint64_t reportedGeneration_ = 0;
545};

568void declareScriptProjectileSeam() {
569 ScriptProjectileProviders& owned = scriptProjectileProviders();
570 ScriptProviderRegistry::instance().declareCapability(
571 IProjectileService::capabilityName,
572 [&owned](const ScriptProviderHandle& handle) -> eve::Result<void> {
573 // 单实现槽位：`provide` 会顶掉当前实现。顶掉必须可见，否则原生提供方会
574 // "悄悄失效"，而症状只是投射物不再生成。
575 if (eve::cap::query<IProjectileService>() != nullptr) {
576 recordLogEvent("warn", "script projectile provider '" + handle.name +
577 "' replaced the provider that owned IProjectileService");
578 }
579 auto adapter = std::make_unique<ScriptProjectileService>(handle);
580 ScriptProjectileService* raw = adapter.get();
581 // 先注册再交所有权：从这里起能力槽位持有裸指针，`owned.live` 是让它有效的唯一原因。
582 eve::cap::provide<IProjectileService>(raw);
583 owned.live.push_back(std::move(adapter));
584 return eve::Result<void>::success();
585 },
586 [&owned](const ScriptProviderHandle& handle) -> eve::Result<void> {
587 const auto match = std::find_if(owned.live.begin(), owned.live.end(),
588 [&handle](const std::unique_ptr<ScriptProjectileService>& adapter) {
589 return adapter->handle() == handle;
590 });
591 if (match == owned.live.end()) {

595 }
596 eve::cap::revoke<IProjectileService>(match->get());
597 owned.retired.push_back(std::move(*match));
598 owned.live.erase(match);
599 return eve::Result<void>::success();
600 });
601}

605 for (auto& adapter : owned.live) {

608 }

610

612 ScriptProviderRegistry::instance()
613 .removeCapability(IProjectileService::capabilityName)
614 .ignore("module teardown may already have dropped this seam");
615}
