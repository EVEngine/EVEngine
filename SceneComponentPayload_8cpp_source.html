18EditorResult<void> SceneComponentPropertyBindings::bind(
19 SceneComponentPayloadRef component, SelectionItem moduleSelection,
20 IPropertyProvider* properties, IDomainOperationTarget* operations, Validator validator) {
21 if (componentType_.empty() || component.target.empty() || component.object.empty() ||
22 component.component.empty() || component.type != componentType_ || !properties || !operations ||
23 moduleSelection.target != TargetId(operations->targetId()))
24 return payloadError<void>(EditorStatus::Rejected, "editor.scene.component-binding-invalid",
25 "Component binding requires matching stable ids and module targets");
26 if (find(component.target, component.component))
27 return payloadError<void>(EditorStatus::Conflict, "editor.scene.component-binding-duplicate",
28 "Scene component is already bound: " + component.component.value());
29 component.revision = operations->revision();
30 bindings_.push_back({std::move(component), std::move(moduleSelection), properties, operations,
31 std::move(validator)});
32 return eve::editing::applied<void>();
33}

64EditorResult<std::pair<const SceneComponentPropertyBindings::Binding*, SelectionSnapshot>>
65SceneComponentPropertyBindings::translate(const SelectionSnapshot& selection) const {
66 if (selection.items.empty())
67 return payloadError<std::pair<const Binding*, SelectionSnapshot>>(
68 EditorStatus::Rejected, "editor.scene.component-selection-empty",
69 "Component property selection cannot be empty");
70 const Binding* first = find(selection.items.front().target, selection.items.front().item);
71 if (!first || selection.items.front().type != componentType_)
72 return payloadError<std::pair<const Binding*, SelectionSnapshot>>(
73 EditorStatus::NotFound, "editor.scene.component-binding-missing",
74 "Selected scene component is not bound to this provider");
75 SelectionSnapshot translated;
76 translated.channel = selection.channel;
77 translated.sequence = selection.sequence;
78 for (const SelectionItem& item : selection.items) {
79 const Binding* binding = find(item.target, item.item);
80 if (!binding || item.type != componentType_ || binding->properties != first->properties ||
81 binding->operations != first->operations)
82 return payloadError<std::pair<const Binding*, SelectionSnapshot>>(
83 EditorStatus::Rejected, "editor.scene.component-binding-mixed-target",
84 "Multi-edit requires components owned by one module property target");
85 translated.items.push_back(binding->moduleSelection);
86 }
87 translated.primary = translated.items.front();
88 return eve::editing::applied<std::pair<const Binding*, SelectionSnapshot>>(
89 {first, std::move(translated)});
90}

168EditorResult<std::vector<SceneComponentPayloadRef>> SceneComponentPayloadRegistry::componentPayloads(
169 const TargetId& scene, const ObjectId& object) const {
170 if (scene.empty() || object.empty())
171 return payloadError<std::vector<SceneComponentPayloadRef>>(
172 EditorStatus::Rejected, "editor.scene.component-owner-invalid",
173 "Scene target and object ids are required for component discovery");
174 std::vector<SceneComponentPayloadRef> result;
175 for (const auto& [type, provider] : providers_) {
176 auto components = provider->components(scene, object);
177 for (SceneComponentPayloadRef& component : components) {
178 if (component.target != scene || component.object != object || component.type != type ||
179 component.component.empty())
180 return payloadError<std::vector<SceneComponentPayloadRef>>(
181 EditorStatus::Failed, "editor.scene.component-provider-contract",
182 "Component provider returned an invalid or foreign stable reference: " + type);
183 result.push_back(std::move(component));
184 }
185 }
186 std::sort(result.begin(), result.end(), [](const auto& left, const auto& right) {
187 if (left.type != right.type) return left.type < right.type;
188 return left.component < right.component;
189 });
190 return eve::editing::applied<std::vector<SceneComponentPayloadRef>>(std::move(result));
191}

193EditorResult<ISceneComponentPayloadProvider*> SceneComponentPayloadRegistry::resolve(
194 const SelectionSnapshot& selection) const {
195 if (selection.items.empty())
196 return payloadError<ISceneComponentPayloadProvider*>(
197 EditorStatus::Rejected, "editor.scene.component-selection-empty",
198 "Component property selection cannot be empty");
199 const std::string& type = selection.items.front().type;
200 if (type.empty())
201 return payloadError<ISceneComponentPayloadProvider*>(
202 EditorStatus::Rejected, "editor.scene.component-selection-type",
203 "Component property selection requires a stable component type");
204 for (const SelectionItem& item : selection.items) {
205 if (item.domain != SelectionDomain::Scene || item.type != type || item.item.empty())
206 return payloadError<ISceneComponentPayloadProvider*>(
207 EditorStatus::Rejected, "editor.scene.component-selection-mixed",
208 "Component property selections must contain one non-empty component type");
209 }
210 const auto found = providers_.find(type);
211 if (found == providers_.end())
212 return payloadError<ISceneComponentPayloadProvider*>(
213 EditorStatus::Unsupported, "editor.scene.component-provider-missing",
214 "No component payload provider is registered for type: " + type);
215 return eve::editing::applied<ISceneComponentPayloadProvider*>(found->second);
216}

232EditorResult<std::vector<EditorDiagnostic>> SceneComponentPayloadRegistry::validatePayload(
233 const SceneComponentPayloadRef& component) const {
234 const auto found = providers_.find(component.type);
235 if (found == providers_.end())
236 return payloadError<std::vector<EditorDiagnostic>>(
237 EditorStatus::Unsupported, "editor.scene.component-provider-missing",
238 "No component payload provider is registered for type: " + component.type);
239 const auto current = found->second->components(component.target, component.object);
240 const auto matching = std::find_if(current.begin(), current.end(), [&](const auto& candidate) {
241 return candidate.component == component.component && candidate.type == component.type;
242 });
243 if (matching == current.end())
244 return payloadError<std::vector<EditorDiagnostic>>(
245 EditorStatus::NotFound, "editor.scene.component-reference-missing",
246 "The referenced component no longer exists: " + component.component.value());
247 if (matching->generation != component.generation || matching->revision != component.revision)
248 return payloadError<std::vector<EditorDiagnostic>>(
249 EditorStatus::Conflict, "editor.scene.component-reference-stale",
250 "The component changed since its inspector reference was captured: " +
251 component.component.value());
252 return eve::editing::applied<std::vector<EditorDiagnostic>>(
253 found->second->validateComponent(component));
254}

256EditorResult<SelectionSnapshot> makeSceneComponentSelection(
257 std::string channel, const std::vector<SceneComponentPayloadRef>& components,
258 std::uint64_t sequence) {
259 if (components.empty())
260 return payloadError<SelectionSnapshot>(EditorStatus::Rejected,

262 "At least one component is required");
263 const std::string& type = components.front().type;
264 SelectionSnapshot result;
265 result.channel = std::move(channel);
266 result.sequence = sequence;
267 for (const SceneComponentPayloadRef& component : components) {
268 if (component.component.empty() || component.target.empty() || component.type.empty() ||
269 component.type != type)
270 return payloadError<SelectionSnapshot>(EditorStatus::Rejected,
271 "editor.scene.component-selection-mixed",
272 "All selected components must have one non-empty type");
273 result.items.push_back(
274 {SelectionDomain::Scene, component.target, component.component, component.type});
275 }
276 result.primary = result.items.front();
277 return eve::editing::applied<SelectionSnapshot>(std::move(result));
