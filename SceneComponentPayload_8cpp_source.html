18 IPropertyProvider* properties, IDomainOperationTarget* operations,
19 Validator validator) {

27 "Scene component is already bound: " + component.component.value());

31 return eve::editing::applied<void>();

63Result<std::pair<const SceneComponentPropertyBindings::Binding*, SelectionSnapshot>>
64SceneComponentPropertyBindings::translate(const SelectionSnapshot& selection) const {
65 if (selection.items.empty())
66 return eve::editing::failed<std::pair<const Binding*, SelectionSnapshot>>(
67 EditorStatus::Rejected, RuleId("editor.scene.component-selection-empty"),
68 "Component property selection cannot be empty");
69 const Binding* first = find(selection.items.front().target, selection.items.front().item);
70 if (!first || selection.items.front().type != componentType_)
71 return eve::editing::failed<std::pair<const Binding*, SelectionSnapshot>>(
72 EditorStatus::NotFound, RuleId("editor.scene.component-binding-missing"),
73 "Selected scene component is not bound to this provider");
74 SelectionSnapshot translated;
75 translated.channel = selection.channel;
76 translated.sequence = selection.sequence;
77 for (const SelectionItem& item : selection.items) {
78 const Binding* binding = find(item.target, item.item);
79 if (!binding || item.type != componentType_ || binding->properties != first->properties ||
80 binding->operations != first->operations)
81 return eve::editing::failed<std::pair<const Binding*, SelectionSnapshot>>(
82 EditorStatus::Rejected, RuleId("editor.scene.component-binding-mixed-target"),
83 "Multi-edit requires components owned by one module property target");
84 translated.items.push_back(binding->moduleSelection);
85 }
86 translated.primary = translated.items.front();
87 return eve::editing::applied<std::pair<const Binding*, SelectionSnapshot>>({first, std::move(translated)});
88}

164Result<std::vector<SceneComponentPayloadRef>> SceneComponentPayloadRegistry::componentPayloads(
165 const TargetId& scene, const ObjectId& object) const {
166 if (scene.empty() || object.empty())
167 return eve::editing::failed<std::vector<SceneComponentPayloadRef>>(
168 EditorStatus::Rejected, RuleId("editor.scene.component-owner-invalid"),
169 "Scene target and object ids are required for component discovery");
170 std::vector<SceneComponentPayloadRef> result;
171 for (const auto& [type, provider] : providers_) {
172 auto components = provider->components(scene, object);
173 for (SceneComponentPayloadRef& component : components) {
174 if (component.target != scene || component.object != object || component.type != type ||
175 component.component.empty())
176 return eve::editing::failed<std::vector<SceneComponentPayloadRef>>(
177 EditorStatus::Failed, RuleId("editor.scene.component-provider-contract"),
178 "Component provider returned an invalid or foreign stable reference: " + type);
179 result.push_back(std::move(component));
180 }
181 }
182 std::sort(result.begin(), result.end(), [](const auto& left, const auto& right) {
183 if (left.type != right.type) return left.type < right.type;
184 return left.component < right.component;
185 });
186 return eve::editing::applied<std::vector<SceneComponentPayloadRef>>(std::move(result));
187}

208 return eve::editing::failed<ISceneComponentPayloadProvider*>(
209 EditorStatus::Unsupported, RuleId("editor.scene.component-provider-missing"),

227Result<std::vector<EditorDiagnostic>> SceneComponentPayloadRegistry::validatePayload(
228 const SceneComponentPayloadRef& component) const {
229 const auto found = providers_.find(component.type);
230 if (found == providers_.end())
231 return eve::editing::failed<std::vector<EditorDiagnostic>>(
232 EditorStatus::Unsupported, RuleId("editor.scene.component-provider-missing"),
233 "No component payload provider is registered for type: " + component.type);
234 const auto current = found->second->components(component.target, component.object);
235 const auto matching = std::find_if(current.begin(), current.end(), [&](const auto& candidate) {
236 return candidate.component == component.component && candidate.type == component.type;
237 });
238 if (matching == current.end())
239 return eve::editing::failed<std::vector<EditorDiagnostic>>(
240 EditorStatus::NotFound, RuleId("editor.scene.component-reference-missing"),
241 "The referenced component no longer exists: " + component.component.value());
242 if (matching->generation != component.generation || matching->revision != component.revision)
243 return eve::editing::failed<std::vector<EditorDiagnostic>>(
244 EditorStatus::Conflict, RuleId("editor.scene.component-reference-stale"),
245 "The component changed since its inspector reference was captured: " + component.component.value());
246 return eve::editing::applied<std::vector<EditorDiagnostic>>(found->second->validateComponent(component));
247}

285const editing::Value* field(const editing::Value& value, const char* key) {
286 const auto* object = value.getIf<editing::Value::Object>();
287 if (!object) return nullptr;
288 const auto found = object->find(key);
289 return found == object->end() ? nullptr : &found->second;

292bool assignVector3(const editing::Value& payload, const char* key, double& x, double& y, double& z) {
293 const editing::Value* entry = field(payload, key);
294 if (!entry) return true;
295 const auto* values = entry->getIf<editing::Value::Array>();
296 if (!values || values->size() != 3) return false;
297 const auto number = [](const editing::Value& value, double& output) {
298 if (const auto* real = value.getIf<double>()) {
299 output = *real;
300 return true;
301 }
302 if (const auto* integer = value.getIf<std::int64_t>()) {
303 output = static_cast<double>(*integer);
304 return true;
305 }
306 return false;
307 };
308 return number((*values)[0], x) && number((*values)[1], y) && number((*values)[2], z);
309}

489 plan.operations.push_back(std::move(operation.value()));
490 }

492 });

525Result<std::map<ObjectId, SceneObjectSnapshot>> SceneTargetBase::parseSnapshot(const EditorValue& value) {
526 using Objects = std::map<ObjectId, SceneObjectSnapshot>;
527 const auto reject = [](const char* message) {
528 return editing::failed<Objects>(EditorStatus::Rejected, RuleId("scene.snapshot.invalid"), message);
529 };
530 const auto* root = value.getIf<EditorValue::Object>();
531 if (!root || !root->contains("schemaVersion") || !root->contains("objects") ||
532 root->size() != (root->contains("schemaId") ? 3u : 2u))
533 return reject("Scene snapshot requires schemaId, schemaVersion and objects");
534 if (root->contains("schemaId")) {
535 const auto* schema = root->at("schemaId").getIf<std::string>();
536 if (!schema || *schema != "eve.scene.hierarchy") return reject("Unknown scene schema id");
537 }
538 const auto* version = root->at("schemaVersion").getIf<std::int64_t>();
539 const auto* entries = root->at("objects").getIf<EditorValue::Array>();
540 if (!version || *version != 1 || !entries) return reject("Unsupported scene snapshot schema");
541 Objects objects;
542 for (const auto& entry : *entries) {
543 const auto* fields = entry.getIf<EditorValue::Object>();
544 if (!fields || fields->size() != 4 || !fields->contains("transform"))
545 return reject("Scene object requires id, parent, name and transform");
546 const auto* transform = fields->at("transform").getIf<EditorValue::Object>();
547 if (!transform || transform->size() != 9) return reject("Scene transform requires all nine TRS fields");
548 // JSON parsers may decode integral floating point values as integers.
549 EditorValue normalized = entry;
550 auto& normalizedTransform =
551 *normalized.getIf<EditorValue::Object>()->at("transform").getIf<EditorValue::Object>();
552 for (const auto* key : {"x", "y", "z", "rotationX", "rotationY", "rotationZ", "scaleX", "scaleY", "scaleZ"}) {
553 const auto found = transform->find(key);
554 if (found == transform->end()) return reject("Unknown or missing transform field");
555 double number;
556 if (const auto* real = found->second.getIf<double>())
557 number = *real;
558 else if (const auto* integer = found->second.getIf<std::int64_t>())
559 number = static_cast<double>(*integer);
560 else
561 return reject("Transform fields must be numbers");
562 if (!std::isfinite(number) || std::abs(number) > std::numeric_limits<float>::max())
563 return reject("Transform must be finite and representable by the scene runtime");
564 if (std::string_view(key).starts_with("scale") && static_cast<float>(number) == 0.f)
565 return reject("Scene scale must be nonzero");
566 normalizedTransform[key] = number;
567 }
568 auto parsed = parseObject(normalized);
569 if (!parsed.ok()) return Result<Objects>::failure(parsed.status());
570 auto object = std::move(parsed.value());
571 if (object.id.empty() || object.name.empty() || objects.contains(object.id))
572 return reject("Object ids must be unique and names nonempty");
573 objects.emplace(object.id, std::move(object));
574 }
575 for (const auto& [id, object] : objects) {
576 std::set<ObjectId> visited{id};
577 auto parent = object.parent;
578 while (!parent.empty()) {
579 if (!objects.contains(parent)) return reject("Scene parent does not exist");
580 if (!visited.insert(parent).second) return reject("Scene hierarchy contains a cycle");
581 parent = objects.at(parent).parent;
582 }
583 }
584 return editing::applied<Objects>(std::move(objects));
585}

587Result<DomainOperation> SceneTargetBase::makeRestore(const EditorValue& snapshot) const {
588 auto validated = parseSnapshot(snapshot);
589 if (!validated.ok()) return Result<DomainOperation>::failure(validated.status());
590 DomainOperation operation;
591 operation.type = operation.inverseType = "scene.snapshot.restore.v1";
592 operation.target = targetId();
593 operation.payload = snapshot;
594 operation.inverse = snapshotValue();
595 operation.hasInverse = true;
596 for (const auto& [id, object] : objects_) operation.affectedObjects.push_back({targetId(), id.value(), 0});
597 for (const auto& [id, object] : validated.value())
598 if (!objects_.contains(id)) operation.affectedObjects.push_back({targetId(), id.value(), 0});
599 return editing::applied<DomainOperation>(std::move(operation));
600}
