64image::ImageData *grayscaleImage(const std::vector<float> &values, int w, int h) {
65 auto *img = new image::ImageData(w, h, "RGBA8");
66 auto *pixels = static_cast<uint8_t *>(img->getData());
67 for (int i = 0; i < w * h && i < int(values.size()); ++i) {
68 const uint8_t v = uint8_t(std::lround(std::clamp(values[size_t(i)], 0.f, 1.f) * 255.f));
69 const size_t o = size_t(i) * 4u;
70 pixels[o + 0] = pixels[o + 1] = pixels[o + 2] = v;
71 pixels[o + 3] = 255;
72 }
73 return img;
74}

76PbrTextureSet *generatePbrSet(const TextureRecipeDef &def, const Params &params,
77 std::string &error) {
78 const auto ctx = TextureGenContext::fromParams(params);
79 if (ctx.width > 4096 || ctx.height > 4096) {
80 error = "texture size too large (max 4096)";
81 return nullptr;
82 }
83
84 // Recipe defaults, then per-call overrides.
85 PbrParams pbr = def.pbr;
86 pbr.roughnessLow = std::clamp(params.getFloat("roughnessLow", pbr.roughnessLow), 0.f, 1.f);
87 pbr.roughnessHigh = std::clamp(params.getFloat("roughnessHigh", pbr.roughnessHigh), 0.f, 1.f);
88 pbr.metallic = std::clamp(params.getFloat("metallic", pbr.metallic), 0.f, 1.f);
89 pbr.normalStrength = std::max(0.01f, params.getFloat("normalStrength", pbr.normalStrength));
90 pbr.aoStrength = std::clamp(params.getFloat("aoStrength", pbr.aoStrength), 0.f, 5.f);
91 pbr.heightStrength = std::max(0.f, params.getFloat("heightStrength", pbr.heightStrength));
92
93 std::vector<float> height;
94 fillHeightField(ctx, def.height, height);
95 const int w = ctx.width;
96 const int h = ctx.height;
97
98 auto *set = new PbrTextureSet();
99 set->albedo = new image::ImageData(w, h, "RGBA8");
100 paintHeightToImage(*set->albedo, height, w, h, def.albedo, ctx.colors, ctx.pixelSize);
101 set->normal = heightToNormalImage(height, w, h, pbr.normalStrength, ctx.seamless);
102
103 std::vector<float> rough(size_t(w * h));
104 std::vector<float> heightMap(size_t(w * h));
105 std::vector<float> ao(size_t(w * h));
106 const int r = std::max(1, std::min(w, h) / 64); // cavity sampling radius
107 for (int y = 0; y < h; ++y) {
108 for (int x = 0; x < w; ++x) {
109 const size_t i = size_t(y) * size_t(w) + size_t(x);
110 const float hh = height[i];
111 rough[size_t(i)] =
112 pbr.roughnessLow + (pbr.roughnessHigh - pbr.roughnessLow) * hh;
113 heightMap[i] = std::clamp(0.5f + (hh - 0.5f) * pbr.heightStrength, 0.f, 1.f);
114
115 // Cavity AO: darker where a pixel sits below its neighbourhood mean.
116 float sum = 0.f;
117 int cnt = 0;
118 for (int dy = -r; dy <= r; ++dy) {
119 for (int dx = -r; dx <= r; ++dx) {
120 int xx = x + dx, yy = y + dy;
121 if (ctx.seamless) {
122 xx = ((xx % w) + w) % w;
123 yy = ((yy % h) + h) % h;
124 } else {
125 if (xx < 0 || yy < 0 || xx >= w || yy >= h) continue;
126 }
127 sum += height[size_t(yy) * size_t(w) + size_t(xx)];
128 ++cnt;
129 }
130 }
131 const float local = cnt > 0 ? sum / float(cnt) : hh;
132 ao[i] = std::clamp(1.f - pbr.aoStrength * std::max(0.f, local - hh), 0.f, 1.f);
133 }
134 }
135 set->roughness = grayscaleImage(rough, w, h);
136 set->metallic = grayscaleImage(std::vector<float>(size_t(w * h), pbr.metallic), w, h);
137 set->height = grayscaleImage(heightMap, w, h);
138 set->ao = grayscaleImage(ao, w, h);
139 return set;
140}
