39bool isAnimationFixtureTextPath(const std::string& path) {
40 constexpr std::string_view suffix = ".anim.txt";
41 if (path.size() < suffix.size()) return false;
42 std::string ext = path.substr(path.size() - suffix.size());
43 for (char& c : ext) c = static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
44 return ext == suffix;
45}

104eve::Result<int> AnimClipRegistry::reloadPath(const std::string& path) {
105 const std::string norm = normalizePath(path);
106 const auto it = entries().find(norm);
107 if (it == entries().end() || it->second.empty()) {
108 notifyReload({norm, 0, false});
109 return eve::Result<int>::success(0, eve::Status::success(eve::StatusCode::NoOp));
110 }
111
112 AnimSkeleton* skeleton = nullptr;
113 AnimClip* fresh = nullptr;
114 try {
115 AnimImporter::importAnimationFixtureTextFile(norm, &skeleton, &fresh);
116 } catch (...) {
117 delete skeleton;
118 delete fresh;
119 notifyReload({norm, 0, false});
120 return eve::Result<int>::failure(
121 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "animation clip reload import failed", norm));
122 }
123 if (!fresh) {
124 delete skeleton;
125 notifyReload({norm, 0, false});
126 return eve::Result<int>::failure(
127 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "animation clip reload produced no clip", norm));
128 }
129 for (AnimClip* clip : it->second) clip->adopt(*fresh);
130 const int refreshed = static_cast<int>(it->second.size());
131 delete fresh;
132 delete skeleton;
133 notifyReload({norm, refreshed, true});
134 return eve::Result<int>::success(refreshed, eve::Status::success(eve::StatusCode::Applied));
135}
