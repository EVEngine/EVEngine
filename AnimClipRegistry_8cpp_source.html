85int AnimClipRegistry::reloadPath(const std::string& path) {
86 const std::string norm = normalizePath(path);
87 const auto it = entries().find(norm);
88 if (it == entries().end() || it->second.empty()) return 0;
89
90 AnimSkeleton* skeleton = nullptr;
91 AnimClip* fresh = nullptr;
92 try {
93 AnimImporter::importEvaFile(norm, &skeleton, &fresh);
94 } catch (...) {
95 delete skeleton;
96 delete fresh;
97 return 0;
98 }
99 if (!fresh) {
100 delete skeleton;
101 return 0;
102 }
103 for (AnimClip* clip : it->second) clip->adopt(*fresh);
104 delete fresh;
105 delete skeleton;
106 return static_cast<int>(it->second.size());
107}
