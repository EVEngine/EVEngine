20Result<RoadIntersection> findUniqueInteriorIntersection(const procgen::road::RoadEdge& first,
21 const procgen::road::RoadEdge& second,
22 float maximumHeightDelta) {
23 struct ParameterSample {
24 float parameter = 0.f;
25 procgen::road::RoadControlPoint point;
26 };
27 auto sampleEdge = [](const procgen::road::RoadEdge& edge) -> Result<std::vector<ParameterSample>> {
28 procgen::SplinePath path;
29 auto kind = path.setKindResult("catmullRom");
30 if (!kind.ok()) return Result<std::vector<ParameterSample>>::failure(kind.status());
31 path.setClosed(false);
32 for (const auto& point : edge.controlPoints) {
33 procgen::SplinePoint splinePoint;
34 splinePoint.alignStraightStart = point.alignStraightStart;
35 splinePoint.alignStraightEnd = point.alignStraightEnd;
36 splinePoint.circularFromEnd = point.circularFromEnd;
37 splinePoint.spanControl = static_cast<eve::procgen::SplineSpanControl>(point.pathControl);
38 splinePoint.spanStrength = point.splineStrength;
39 splinePoint.x = point.x;
40 splinePoint.y = point.y;
41 splinePoint.z = point.z;
42 auto added = path.addPointResult(splinePoint);
43 if (!added.ok()) return Result<std::vector<ParameterSample>>::failure(added.status());
44 }
45 const int intervals = std::min(4096, std::max(32, path.segmentCount() * 24));
46 std::vector<ParameterSample> samples;
47 samples.reserve(static_cast<std::size_t>(intervals + 1));
48 for (int i = 0; i <= intervals; ++i) {
49 const float parameter = static_cast<float>(i) / static_cast<float>(intervals);
50 auto evaluated = path.evaluateResult(parameter);
51 if (!evaluated.ok()) return Result<std::vector<ParameterSample>>::failure(evaluated.status());
52 samples.push_back({parameter, {evaluated.value().x, evaluated.value().y, evaluated.value().z}});
53 }
54 return Result<std::vector<ParameterSample>>::success(std::move(samples));
55 };
56 auto firstSamples = sampleEdge(first);
57 if (!firstSamples.ok()) return Result<RoadIntersection>::failure(firstSamples.status());
58 auto secondSamples = sampleEdge(second);
59 if (!secondSamples.ok()) return Result<RoadIntersection>::failure(secondSamples.status());
60
61 std::vector<RoadIntersection> crossings;
62 constexpr float epsilon = 1e-5f;
63 const auto cross2 = [](float ax, float az, float bx, float bz) { return ax * bz - az * bx; };
64 for (std::size_t i = 0; i + 1 < firstSamples.value().size(); ++i) {
65 const auto& a = firstSamples.value()[i].point;
66 const auto& b = firstSamples.value()[i + 1].point;
67 const float rx = b.x - a.x, rz = b.z - a.z;
68 for (std::size_t j = 0; j + 1 < secondSamples.value().size(); ++j) {
69 const auto& c = secondSamples.value()[j].point;
70 const auto& d = secondSamples.value()[j + 1].point;
71 const float sx = d.x - c.x, sz = d.z - c.z;
72 const float denominator = cross2(rx, rz, sx, sz);
73 if (std::fabs(denominator) <= epsilon) continue;
74 const float qx = c.x - a.x, qz = c.z - a.z;
75 const float t = cross2(qx, qz, sx, sz) / denominator;
76 const float u = cross2(qx, qz, rx, rz) / denominator;
77 if (t < -epsilon || t > 1.f + epsilon || u < -epsilon || u > 1.f + epsilon) continue;
78 const float clampedT = std::clamp(t, 0.f, 1.f), clampedU = std::clamp(u, 0.f, 1.f);
79 const float x = a.x + rx * clampedT, z = a.z + rz * clampedT;
80 const float firstParameter = firstSamples.value()[i].parameter +
81 (firstSamples.value()[i + 1].parameter - firstSamples.value()[i].parameter) * clampedT;
82 const float secondParameter = secondSamples.value()[j].parameter +
83 (secondSamples.value()[j + 1].parameter - secondSamples.value()[j].parameter) * clampedU;
84 if (firstParameter <= epsilon || firstParameter >= 1.f - epsilon || secondParameter <= epsilon ||
85 secondParameter >= 1.f - epsilon)
86 continue;
87 const float firstY = a.y + (b.y - a.y) * clampedT;
88 const float secondY = c.y + (d.y - c.y) * clampedU;
89 if (std::fabs(firstY - secondY) > maximumHeightDelta) continue;
90 const bool duplicate = std::any_of(crossings.begin(), crossings.end(), [&](const auto& crossing) {
91 return std::fabs(crossing.mergedPoint.x - x) <= 1e-3f &&
92 std::fabs(crossing.mergedPoint.z - z) <= 1e-3f;
93 });
94 if (!duplicate) crossings.push_back({1, firstParameter, secondParameter, {x, firstY, z},
95 {x, secondY, z}, {x, (firstY + secondY) * 0.5f, z}});
96 }
97 }
98 if (crossings.size() == 1) return Result<RoadIntersection>::success(crossings.front());
99 RoadIntersection result;
100 result.count = crossings.empty() ? 0 : 2;
101 return Result<RoadIntersection>::success(result);
102}

675EditorResult<void> RoadNetworkEditTarget::loadSnapshot(const EditorValue& snapshot) {
676 const auto* schemaValue = field(snapshot, "schema");
677 const auto* versionValue = field(snapshot, "schemaVersion");
678 const auto* nodesValue = field(snapshot, "nodes");
679 const auto* edgesValue = field(snapshot, "edges");
680 const auto* linksValue = field(snapshot, "laneLinks");
681 const auto* blockedValue = field(snapshot, "blockedLaneLinks");
682 const auto* schema = schemaValue ? schemaValue->getIf<std::string>() : nullptr;
683 const auto* version = versionValue ? versionValue->getIf<std::int64_t>() : nullptr;
684 const auto* nodes = nodesValue ? nodesValue->getIf<EditorValue::Array>() : nullptr;
685 const auto* edges = edgesValue ? edgesValue->getIf<EditorValue::Array>() : nullptr;
686 const auto* links = linksValue ? linksValue->getIf<EditorValue::Array>() : nullptr;
687 const auto* blocked = blockedValue ? blockedValue->getIf<EditorValue::Array>() : nullptr;
688 if (!schema || *schema != "eve.procgen.roadNetwork" || !version || *version != 1 || !nodes || !edges ||
689 !links)
690 return reject<void>("editor.road.invalid-snapshot", "Road snapshot schema is unsupported",
691 EditorStatus::Unsupported);
692 constexpr std::size_t maximumNodes = 65536u, maximumEdges = 65536u, maximumLinks = 262144u;
693 if (nodes->size() > maximumNodes || edges->size() > maximumEdges || links->size() > maximumLinks ||
694 (blocked && blocked->size() > maximumLinks))
695 return reject<void>("editor.road.snapshot-budget", "Road snapshot exceeds the graph budget");
696
697 procgen::road::RoadNetwork candidate;
698 for (const auto& encoded : *nodes) {
699 procgen::road::RoadNode node;
700 if (!decodeNodeRecord(encoded, node))
701 return reject<void>("editor.road.invalid-snapshot-node", "Road snapshot contains an invalid node");
702 auto restored = candidate.restoreNode(node);
703 if (!restored.ok()) return roadFailure<void>(restored.status());
704 }
705 for (const auto& encoded : *edges) {
706 procgen::road::RoadEdge edge;
707 std::vector<procgen::road::RoadLaneConnection> embeddedLinks;
708 if (!decodeEdgeRecord(encoded, edge, embeddedLinks) || !embeddedLinks.empty())
709 return reject<void>("editor.road.invalid-snapshot-edge", "Road snapshot contains an invalid edge");
710 auto restored = candidate.restoreEdge(std::move(edge));
711 if (!restored.ok()) return roadFailure<void>(restored.status());
712 }
713 for (const auto& encoded : *links) {
714 procgen::road::RoadLaneConnection link;
715 if (!decodeLink(encoded, link))
716 return reject<void>("editor.road.invalid-snapshot-link", "Road snapshot contains an invalid lane link");
717 auto restored = candidate.addLaneLink(link);
718 if (!restored.ok()) return roadFailure<void>(restored.status());
719 }
720 if (blocked) {
721 for (const auto& encoded : *blocked) {
722 procgen::road::RoadLaneConnection link;
723 if (!decodeLink(encoded, link))
724 return reject<void>("editor.road.invalid-snapshot-block",
725 "Road snapshot contains an invalid blocked lane link");
726 auto restored = candidate.blockLaneLink(link);
727 if (!restored.ok()) return roadFailure<void>(restored.status());
728 if (restored.value())
729 return reject<void>("editor.road.invalid-snapshot-block",
730 "Road snapshot marks the same lane link active and blocked");
731 }
732 }
733 auto valid = candidate.validate();
734 if (!valid.ok()) return roadFailure<void>(valid.status());
735 *network_ = std::move(candidate);
736 dirty_.include(0, 0);
737 return editing::applied();
738}
