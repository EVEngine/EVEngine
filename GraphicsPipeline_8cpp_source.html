78vk::Pipeline createSolidColorPipeline(vkb::Device &device, const vkb::BuiltRenderPass &renderPass,
79 vk::PipelineLayout layout,
80 BlendMode mode = BlendMode::Opaque) {
81 if (mode == BlendMode::Additive) {
82 // cbs/attachments must outlive build(); the builder must stay a single
83 // expression — copying the builder into a named local leaves its
84 // shader-stage pName pointers dangling into the temporary's storage.
85 std::vector<vk::PipelineColorBlendAttachmentState> attachments(1,
86 makeBlendAttachment(mode));
87 vk::PipelineColorBlendStateCreateInfo cbs{};
88 cbs.logicOpEnable = false;
89 cbs.attachmentCount = 1;
90 cbs.pAttachments = attachments.data();
91 return device.createPipeline()
92 .useClassicPipeline(embeddedSpirv(color_vert_spv), embeddedSpirv(color_frag_spv))
93 .setPipelineLayout(layout)
94 .setVertexInputState(vkb::VertexInputStateBuilder()
95 .addInputBinding<ColorVertex>()
96 .addAttributeDescription<ColorVertex>())
97 .setDynamicStatesViewportScissor()
98 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
99 vk::CullModeFlagBits::eNone, vk::FrontFace::eCounterClockwise)
100 .setColorBlending(cbs)
101 .build(renderPass);
102 }
103 if (mode == BlendMode::Alpha) {
104 return device.createPipeline()
105 .useClassicPipeline(embeddedSpirv(color_vert_spv), embeddedSpirv(color_frag_spv))
106 .setPipelineLayout(layout)
107 .setVertexInputState(vkb::VertexInputStateBuilder()
108 .addInputBinding<ColorVertex>()
109 .addAttributeDescription<ColorVertex>())
110 .setDynamicStatesViewportScissor()
111 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
112 vk::CullModeFlagBits::eNone, vk::FrontFace::eCounterClockwise)
113 .setAlphaBlending(1)
114 .build(renderPass);
115 }
116 // Opaque: keep the original single-expression chain (no explicit blend state).
117 return device.createPipeline()
118 .useClassicPipeline(embeddedSpirv(color_vert_spv), embeddedSpirv(color_frag_spv))
119 .setPipelineLayout(layout)
120 .setVertexInputState(vkb::VertexInputStateBuilder()
121 .addInputBinding<ColorVertex>()
122 .addAttributeDescription<ColorVertex>())
123 .setDynamicStatesViewportScissor()
124 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f, vk::CullModeFlagBits::eNone,
125 vk::FrontFace::eCounterClockwise)
126 .build(renderPass);
127}

132void Graphics::createSwapchainAndPipeline() {
133 // Framebuffers / command buffers alias the current swapchain images; tear
134 // them down before replacing the swapchain (destroy() waitIdles first).
135 presentRecording = {};
136 swapchainPass = {};
137 presentModel.destroy();
138 presentModel = vkb::Present{};
139
140 vkb::SwapchainBuilder swapchainBuilder = device.createSwapchain();
141 if (pixelWidth > 0 && pixelHeight > 0)
142 swapchainBuilder.set_desired_extent(uint32_t(pixelWidth), uint32_t(pixelHeight));
143 // Prefer UNORM so clear/draw Color floats match getPixel without sRGB encode.
144 swapchainBuilder.set_desired_format(
145 {vk::Format::eB8G8R8A8Unorm, vk::ColorSpaceKHR::eSrgbNonlinear});
146 if (vsyncEnabled) {
147 swapchainBuilder.set_desired_present_mode(vk::PresentModeKHR::eMailbox);
148 swapchainBuilder.add_fallback_present_mode(vk::PresentModeKHR::eFifo);
149 } else {
150 swapchainBuilder.set_desired_present_mode(vk::PresentModeKHR::eImmediate);
151 swapchainBuilder.add_fallback_present_mode(vk::PresentModeKHR::eMailbox);
152 swapchainBuilder.add_fallback_present_mode(vk::PresentModeKHR::eFifo);
153 }
154 // Allow screen getPixel / newImageData readback after present.
155 swapchainBuilder.add_image_usage_flags(vk::ImageUsageFlagBits::eTransferSrc);
156 auto swapRet = swapchainBuilder.set_old_swapchain(swapchain).build();
157 swapchain.destroy();
158 swapchain = swapRet;
159
160 depthImage = vkb::DepthStencilImage{device, swapchain.extent.width, swapchain.extent.height, depthFormat};
161
162 if (!renderpass) {
163 vkb::RenderPassBuilder rpBuilder{device};
164 renderpass =
165 rpBuilder.addPresentAttachment(swapchain.image_format, vk::AttachmentLoadOp::eClear)
166 .addDepthAttachment(depthFormat, vk::AttachmentLoadOp::eClear,
167 vk::AttachmentStoreOp::eDontCare)
168 .addSubpass(vkb::SubpassBuilder()
169 .addAttachmentRef(0, vk::ImageLayout::eColorAttachmentOptimal)
170 .setDepthStencilAttachment(
171 1, vk::ImageLayout::eDepthStencilAttachmentOptimal))
172 .addDependency(VK_SUBPASS_EXTERNAL, 0,
173 vk::PipelineStageFlagBits::eColorAttachmentOutput |
174 vk::PipelineStageFlagBits::eEarlyFragmentTests,
175 vk::PipelineStageFlagBits::eColorAttachmentOutput |
176 vk::PipelineStageFlagBits::eEarlyFragmentTests,
177 {},
178 vk::AccessFlagBits::eColorAttachmentRead |
179 vk::AccessFlagBits::eColorAttachmentWrite |
180 vk::AccessFlagBits::eDepthStencilAttachmentWrite)
181 .build();
182 }
183
184 if (!pipeline) {
185 pipelineLayout = createPipelineLayout(device);
186 pipeline = createSolidColorPipeline(device, renderpass, pipelineLayout);
187 solidAlphaPipeline = createSolidColorPipeline(device, renderpass, pipelineLayout,
188 BlendMode::Alpha);
189 additiveSolidPipeline = createSolidColorPipeline(device, renderpass, pipelineLayout,
190 BlendMode::Additive);
191 }
192
193 // Scene-pass pipelines are initially built for the swapchain render pass at
194 // 1x; ensureScenePassPipelines rebuilds them only when the active scene pass
195 // (handle or sample count) differs.
196 scenePassPipelineTarget = vk::RenderPass(renderpass);
197 scenePassPipelineSamples = vk::SampleCountFlagBits::e1;
198
199 presentModel = device.createPresent(swapchain).build(renderpass, depthImage.imageView());
200 // Multi-frame overlap: submit + present without waiting on this frame's
201 // fence, so the CPU can build frame N+1 while the GPU still renders frame N.
202 // Present caps frames_in_flight at 2 (independent of swapchain image count)
203 // so present-wait semaphores are not reused while WSI still holds them.
204 // Per-frame mutable GPU resources must be multi-buffered (see *FrameSlots).
205 presentModel.synchronous_frames = false;
206 ensurePresentCaptureHook();
207 swapchainDirty = false;
208}

254vk::Pipeline Graphics::createTexturedStylePipeline(const std::vector<uint32_t> &vert,
255 const std::vector<uint32_t> &frag,
256 const vkb::BuiltRenderPass &rp,
257 vk::PipelineLayout layout, BlendMode mode) {
258 ShaderModulePair modules(device, vert, frag);

264 vk::PipelineColorBlendStateCreateInfo cbs{};
265 cbs.logicOpEnable = false;
266 cbs.attachmentCount = 1;
267 cbs.pAttachments = attachments.data();
268 return device.createPipeline()
269 .useClassicPipeline(modules.vert, modules.frag)
270 .setPipelineLayout(layout)
271 .setVertexInputState(vkb::VertexInputStateBuilder()
272 .addInputBinding<TexturedVertex>()
273 .addAttributeDescription<TexturedVertex>())
274 .setDynamicStatesViewportScissor()
275 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
276 vk::CullModeFlagBits::eNone, vk::FrontFace::eCounterClockwise)
277 .setDepthStencil(false, false)
278 .setColorBlending(cbs)
279 .build(rp);
280 }
281 if (mode == BlendMode::Opaque) {
282 return device.createPipeline()
283 .useClassicPipeline(modules.vert, modules.frag)
284 .setPipelineLayout(layout)
285 .setVertexInputState(vkb::VertexInputStateBuilder()
286 .addInputBinding<TexturedVertex>()
287 .addAttributeDescription<TexturedVertex>())
288 .setDynamicStatesViewportScissor()
289 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
290 vk::CullModeFlagBits::eNone, vk::FrontFace::eCounterClockwise)
291 .setDepthStencil(false, false)
292 .setColorBlending()
293 .build(rp);
294 }
295 // Alpha (original behavior).
296 return device.createPipeline()
297 .useClassicPipeline(modules.vert, modules.frag)
298 .setPipelineLayout(layout)
299 .setVertexInputState(vkb::VertexInputStateBuilder()
300 .addInputBinding<TexturedVertex>()
301 .addAttributeDescription<TexturedVertex>())
302 .setDynamicStatesViewportScissor()
303 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f, vk::CullModeFlagBits::eNone,
304 vk::FrontFace::eCounterClockwise)
305 .setDepthStencil(false, false)
306 .setAlphaBlending(1)
307 .build(rp);
308}

501void Graphics::createUiColorResources(int uiW, int uiH) {
502 if (uiW <= 0 || uiH <= 0) return;
503 const vk::Format colorFmt = swapchain.image_format;
504 if (colorFmt == vk::Format::eUndefined) return;
505
506 const vk::SampleCountFlags supported =
507 device.physical_device.properties.limits.framebufferColorSampleCounts;
508 const vk::SampleCountFlagBits samples =
509 (supported & vk::SampleCountFlagBits::e4) ? vk::SampleCountFlagBits::e4
510 : vk::SampleCountFlagBits::e1;
511
512 // The render pass is size-independent and must remain stable for ImGui's
513 // lifetime: ImGui_ImplVulkan_Init builds its pipeline against it. Create it
514 // once, before any early-returns below.
515 if (!uiRenderPass) {
516 if (samples != vk::SampleCountFlagBits::e1) {
517 uiRenderPass =
518 device.createRenderPass()
519 .addColorAttachment(colorFmt, vk::AttachmentLoadOp::eClear,

1024void Graphics::createShadowResources() {
1025 if (!shadowMaps.empty()) return;
1026
1027 const uint32_t size = uint32_t(ShadowConfig::kMapSize);
1028 const uint32_t layers = uint32_t(ShadowConfig::kCascades);
1029
1030 shadowMaps.resize(kAsyncResourceCopies);
1031 for (auto &slot : shadowMaps)
1032 slot.image = device.createDepthArray(size, size, layers, vk::Format::eD32Sfloat);
1033
1034 vkb::SamplerBuilder sb;
1035 // Hardware PCF: linear filtering + depth compare on the D32 shadow map.
1036 // The compare result of each filtered depth sample is blended by the driver,
1037 // giving a soft penumbra instead of the hard 1-texel boundary.
1038 shadowSampler = sb.linearClamp().compareEnable(VK_TRUE).compareOp(vk::CompareOp::eLess).buildDepthPcf(device);
1039
1040 auto shadowPass =
1041 device.createRenderPass()
1042 .addSampledDepthAttachment(vk::Format::eD32Sfloat)
1043 .addSubpass(vkb::SubpassBuilder().setDepthStencilAttachment(
1044 0, vk::ImageLayout::eDepthStencilAttachmentOptimal))
1045 .addExternalShaderReadDependencies()
1046 .build();
1047 shadowRenderPass = shadowPass;
1048
1049 for (auto &slot : shadowMaps) {
1050 for (uint32_t i = 0; i < layers; ++i) {
1051 slot.framebuffers[i] = shadowPass.createFramebuffer(
1052 device, size, size, {slot.image.layerAttachment(i)});
1053 }
1054 }
1055
1056 shadowPipelineLayout =
1057 device.createPipelineLayout()
1058 .push<glm::mat4>(vk::ShaderStageFlagBits::eVertex)
1059 .build();
1060
1061 std::vector<uint32_t> vert(mesh3d_shadow_vert_spv,
1062 mesh3d_shadow_vert_spv + mesh3d_shadow_vert_spv_count);
1063 std::vector<uint32_t> frag(mesh3d_shadow_frag_spv,
1064 mesh3d_shadow_frag_spv + mesh3d_shadow_frag_spv_count);
1065 vk::ShaderModule vertModule = vkb::PipelineBuilder::createShaderModule(device.instance, vert);
1066 vk::ShaderModule fragModule = vkb::PipelineBuilder::createShaderModule(device.instance, frag);
1067 shadowPipeline =
1068 device.createPipeline()
1069 .useClassicPipeline(vertModule, fragModule)
1070 .setPipelineLayout(shadowPipelineLayout)
