91vk::Pipeline createSolidColorPipeline(vkb::Device &device, const vkb::BuiltRenderPass &renderPass,
92 vk::PipelineLayout layout,
93 BlendMode mode = BlendMode::Opaque) {
94 if (mode == BlendMode::Additive || mode == BlendMode::Premultiplied ||
95 mode == BlendMode::Multiply) {
96 // cbs/attachments must outlive build(); the builder must stay a single
97 // expression — copying the builder into a named local leaves its
98 // shader-stage pName pointers dangling into the temporary's storage.
99 std::vector<vk::PipelineColorBlendAttachmentState> attachments(1,
100 makeBlendAttachment(mode));
101 vk::PipelineColorBlendStateCreateInfo cbs{};
102 cbs.logicOpEnable = false;
103 cbs.attachmentCount = 1;
104 cbs.pAttachments = attachments.data();
105 return device.createPipeline()
106 .useClassicPipeline(embeddedSpirv(color_vert_spv), embeddedSpirv(color_frag_spv))
107 .setPipelineLayout(layout)
108 .setVertexInputState(vkb::VertexInputStateBuilder()
109 .addInputBinding<ColorVertex>()
110 .addAttributeDescription<ColorVertex>())
111 .setDynamicStatesViewportScissor()
112 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
113 vk::CullModeFlagBits::eNone, vk::FrontFace::eCounterClockwise)
114 .setColorBlending(cbs)
115 .build(renderPass);
116 }
117 if (mode == BlendMode::Alpha) {
118 return device.createPipeline()
119 .useClassicPipeline(embeddedSpirv(color_vert_spv), embeddedSpirv(color_frag_spv))
120 .setPipelineLayout(layout)
121 .setVertexInputState(vkb::VertexInputStateBuilder()
122 .addInputBinding<ColorVertex>()
123 .addAttributeDescription<ColorVertex>())
124 .setDynamicStatesViewportScissor()
125 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
126 vk::CullModeFlagBits::eNone, vk::FrontFace::eCounterClockwise)
127 .setAlphaBlending(1)
128 .build(renderPass);
129 }
130 // Opaque: keep the original single-expression chain (no explicit blend state).
131 return device.createPipeline()
132 .useClassicPipeline(embeddedSpirv(color_vert_spv), embeddedSpirv(color_frag_spv))
133 .setPipelineLayout(layout)
134 .setVertexInputState(vkb::VertexInputStateBuilder()
135 .addInputBinding<ColorVertex>()
136 .addAttributeDescription<ColorVertex>())
137 .setDynamicStatesViewportScissor()
138 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f, vk::CullModeFlagBits::eNone,
139 vk::FrontFace::eCounterClockwise)
140 .build(renderPass);
141}

143vk::Pipeline createPrimitive3DPipeline(vkb::Device &device, const vkb::BuiltRenderPass &renderPass,
144 vk::PipelineLayout layout, PrimitiveDepthMode depth, BlendMode blend,
145 PrimitiveCullMode cull, vk::SampleCountFlagBits samples) {
146 const bool depthTest = depth != PrimitiveDepthMode::Ignore;
147 const bool depthWrite = depth == PrimitiveDepthMode::TestAndWrite;
148 const vk::CullModeFlags cullFlags =
149 cull == PrimitiveCullMode::Back
150 ? vk::CullModeFlagBits::eBack
151 : (cull == PrimitiveCullMode::Front ? vk::CullModeFlagBits::eFront : vk::CullModeFlagBits::eNone);
152 std::vector<vk::PipelineColorBlendAttachmentState> attachments(1, makeBlendAttachment(blend));
153 vk::PipelineColorBlendStateCreateInfo blending{};
154 blending.attachmentCount = 1;
155 blending.pAttachments = attachments.data();
156 return device.createPipeline()
157 .useClassicPipeline(embeddedSpirv(primitive3d_vert_spv), embeddedSpirv(primitive3d_frag_spv))
158 .setPipelineLayout(layout)
159 .setVertexInputState(vkb::VertexInputStateBuilder()
160 .addInputBinding<Primitive3DVertex>()
161 .addAttributeDescription<Primitive3DVertex>())
162 .setDynamicStatesViewportScissor()
163 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.f, cullFlags, vk::FrontFace::eCounterClockwise)
164 .setDepthStencil(depthTest, depthWrite, depthTest ? vk::CompareOp::eLessOrEqual : vk::CompareOp::eAlways)
165 .setMultisampler(false, samples)
166 .setColorBlending(blending)
167 .build(renderPass);
168}

173void Graphics::createSwapchainAndPipeline() {
174 initGpuTiming();
175 // Framebuffers / command buffers alias the current swapchain images; tear
176 // them down before replacing the swapchain (destroy() waitIdles first).
177 presentRecording = {};
178 swapchainPass = {};
179 presentModel.destroy();
180 presentModel = vkb::Present{};
181
182 vkb::SwapchainBuilder swapchainBuilder = device.createSwapchain();
183 if (pixelWidth > 0 && pixelHeight > 0)
184 swapchainBuilder.set_desired_extent(uint32_t(pixelWidth), uint32_t(pixelHeight));
185 // Prefer UNORM so clear/draw Color floats match getPixel without sRGB encode.
186 swapchainBuilder.set_desired_format(
187 {vk::Format::eB8G8R8A8Unorm, vk::ColorSpaceKHR::eSrgbNonlinear});
188 if (vsyncEnabled) {
189 swapchainBuilder.set_desired_present_mode(vk::PresentModeKHR::eMailbox);
190 swapchainBuilder.add_fallback_present_mode(vk::PresentModeKHR::eFifo);
191 } else {
192 swapchainBuilder.set_desired_present_mode(vk::PresentModeKHR::eImmediate);
193 swapchainBuilder.add_fallback_present_mode(vk::PresentModeKHR::eMailbox);
194 swapchainBuilder.add_fallback_present_mode(vk::PresentModeKHR::eFifo);
195 }
196 // Allow screen getPixel / newImageData readback after present.
197 swapchainBuilder.add_image_usage_flags(vk::ImageUsageFlagBits::eTransferSrc);
198 auto swapRet = swapchainBuilder.set_old_swapchain(swapchain).build();
199 swapchain.destroy();
200 swapchain = swapRet;
201
202 {
203 StartupStage stage(" vulkan: depth image");
204 depthImage = vkb::DepthStencilImage{device, swapchain.extent.width, swapchain.extent.height, depthFormat};
205 }
206
207 if (!renderpass) {
208 vkb::RenderPassBuilder rpBuilder{device};
209 renderpass =
210 rpBuilder.addPresentAttachment(swapchain.image_format, vk::AttachmentLoadOp::eClear)
211 .addDepthAttachment(depthFormat, vk::AttachmentLoadOp::eClear,
212 vk::AttachmentStoreOp::eDontCare)
213 .addSubpass(vkb::SubpassBuilder()
214 .addAttachmentRef(0, vk::ImageLayout::eColorAttachmentOptimal)
215 .setDepthStencilAttachment(
216 1, vk::ImageLayout::eDepthStencilAttachmentOptimal))
217 .addDependency(VK_SUBPASS_EXTERNAL, 0,
218 vk::PipelineStageFlagBits::eColorAttachmentOutput |
219 vk::PipelineStageFlagBits::eEarlyFragmentTests,
220 vk::PipelineStageFlagBits::eColorAttachmentOutput |
221 vk::PipelineStageFlagBits::eEarlyFragmentTests,
222 {},
223 vk::AccessFlagBits::eColorAttachmentRead |
224 vk::AccessFlagBits::eColorAttachmentWrite |
225 vk::AccessFlagBits::eDepthStencilAttachmentWrite)
226 .build();
227 }
228
229 if (!pipeline) {
230 pipelineLayout = createPipelineLayout(device);
231 pipeline = createSolidColorPipeline(device, renderpass, pipelineLayout);
232 solidAlphaPipeline = createSolidColorPipeline(device, renderpass, pipelineLayout,
233 BlendMode::Alpha);
234 additiveSolidPipeline = createSolidColorPipeline(device, renderpass, pipelineLayout,
235 BlendMode::Additive);
236 premultipliedSolidPipeline = createSolidColorPipeline(
237 device, renderpass, pipelineLayout, BlendMode::Premultiplied);
238 multiplySolidPipeline = createSolidColorPipeline(device, renderpass, pipelineLayout,
239 BlendMode::Multiply);
240 }
241
242 // Scene-pass pipelines are initially built for the swapchain render pass at
243 // 1x; ensureScenePassPipelines rebuilds them only when the active scene pass
244 // (handle or sample count) differs.
245 scenePassPipelineTarget = vk::RenderPass(renderpass);
246 scenePassPipelineSamples = vk::SampleCountFlagBits::e1;
247
248 presentModel = device.createPresent(swapchain).build(renderpass, depthImage.imageView());
249 // Multi-frame overlap: submit + present without waiting on this frame's
250 // fence, so the CPU can build frame N+1 while the GPU still renders frame N.
251 // Present caps frames_in_flight at 2 (independent of swapchain image count)
252 // so present-wait semaphores are not reused while WSI still holds them.
253 // Per-frame mutable GPU resources must be multi-buffered (see *FrameSlots).
254 presentModel.synchronous_frames = false;
255 ensurePresentCaptureHook();
256 swapchainDirty = false;
257}

318 createGpuParticlePipelines();

376 .setDepthStencil(false, false)
377 .setMultisampler(false, samples)
378 .setAlphaBlending(1)
379 .build(rp);

473 }
474 }

624void Graphics::createUiColorResources(int uiW, int uiH) {
625 if (uiW <= 0 || uiH <= 0) return;
626 // Keep scene lighting in linear HDR. The swapchain remains display
627 // encoded and receives a separate final composite; regular script Canvas
628 // targets intentionally stay RGBA8.
629 const vk::Format colorFmt = vk::Format::eR16G16B16A16Sfloat;
630
631 const vk::SampleCountFlags supported =
632 device.physical_device.properties.limits.framebufferColorSampleCounts;
633 const vk::SampleCountFlagBits samples =
634 (supported & vk::SampleCountFlagBits::e4) ? vk::SampleCountFlagBits::e4
635 : vk::SampleCountFlagBits::e1;
636
637 // The render pass is size-independent and must remain stable for ImGui's
638 // lifetime: ImGui_ImplVulkan_Init builds its pipeline against it. Create it
639 // once, before any early-returns below.
640 if (!uiRenderPass) {
641 if (samples != vk::SampleCountFlagBits::e1) {
642 uiRenderPass =
643 device.createRenderPass()
644 .addColorAttachment(colorFmt, vk::AttachmentLoadOp::eClear,
645 vk::AttachmentStoreOp::eDontCare, samples)
646 .addResolveColorAttachment(colorFmt, vk::AttachmentLoadOp::eDontCare)
647 .addSubpass(vkb::SubpassBuilder()
648 .addAttachmentRef(0, vk::ImageLayout::eColorAttachmentOptimal)
649 .addResolveAttachment(1, vk::ImageLayout::eColorAttachmentOptimal))
650 .addExternalShaderReadDependencies()
651 .build();
652 } else {
653 uiRenderPass =
654 device.createRenderPass()
655 .addSampledColorAttachment(colorFmt)
656 .addSubpass(vkb::SubpassBuilder()
657 .addAttachmentRef(0, vk::ImageLayout::eColorAttachmentOptimal))
658 .addExternalShaderReadDependencies()
659 .build();
660 }
661 }
662 // Keep sample count in sync with the render pass even if target creation
663 // returns early (ImGui builds its pipeline from getUiMsaaSamples()).
664 uiColorSamples = samples;
665
666 if (!texSetLayout || !descriptorPool) return;
667 if (!uiTexturePipeline) {
668 const auto vert = embeddedSpirv(textured_vert_spv);
669 const auto frag = embeddedSpirv(textured_frag_spv);
670 const auto opaqueFrag = embeddedSpirv(textured_opaque_frag_spv);
671 uiTexturePipeline = createTexturedStylePipeline(vert, frag, uiRenderPass,
672 texPipelineLayout, BlendMode::Alpha,
673 uiColorSamples);
674 uiTextureOpaquePipeline = createTexturedStylePipeline(
675 vert, opaqueFrag, uiRenderPass, texPipelineLayout, BlendMode::Opaque,
676 uiColorSamples);
677 }
678
679 if (!uiColorSlots.empty() && uiColorWidth == uiW && uiColorHeight == uiH &&
680 uiColorFormat == colorFmt && uiColorSamples == samples)
681 return;
682 destroyUiColorTargets();
683
684 uiColorWidth = uiW;
685 uiColorHeight = uiH;
686 uiColorFormat = colorFmt;
687 uiColorSamples = samples;
688 const uint32_t w = uint32_t(uiW);
689 const uint32_t h = uint32_t(uiH);
690 const bool msaa = samples != vk::SampleCountFlagBits::e1;
691
692 uiColorSlots.resize(kAsyncResourceCopies);
693 for (auto &slot : uiColorSlots) {
694 slot.color = device.createColorTarget(w, h, colorFmt);
695 if (msaa) slot.msaaColor = device.createColorTarget(w, h, colorFmt, samples);
696 }
697 auto uiPass = uiRenderPass;
698
699 for (auto &slot : uiColorSlots) {
700 if (msaa) {
701 slot.framebuffer = uiPass.createFramebuffer(
702 device, w, h, {slot.msaaColor.asAttachment(), slot.color.asAttachment()});
703 } else {
704 slot.framebuffer = uiPass.createFramebuffer(device, w, h, {slot.color.asAttachment()});
705 }
706 }

1595void Graphics::createShadowResources() {
1596 if (!shadowMaps.empty()) return;
1597
1598 const uint32_t size = uint32_t(ShadowConfig::kMapSize);
1599 const uint32_t layers = uint32_t(ShadowConfig::kCascades);
1600
1601 shadowMaps.resize(kAsyncResourceCopies);
1602 for (auto &slot : shadowMaps)
1603 slot.image = device.createDepthArray(size, size, layers, vk::Format::eD32Sfloat);
1604
1605 vkb::SamplerBuilder sb;
1606 // Hardware PCF: linear filtering + depth compare on the D32 shadow map.
1607 // The compare result of each filtered depth sample is blended by the driver,
1608 // giving a soft penumbra instead of the hard 1-texel boundary.
1609 shadowSampler = sb.linearClamp().compareEnable(VK_TRUE).compareOp(vk::CompareOp::eLess).buildDepthPcf(device);
1610 vkb::SamplerBuilder rawBuilder;
1611 shadowRawSampler = rawBuilder.magFilter(vk::Filter::eNearest)
1612 .minFilter(vk::Filter::eNearest)
1613 .addressModeU(vk::SamplerAddressMode::eClampToEdge)
1614 .addressModeV(vk::SamplerAddressMode::eClampToEdge)
1615 .build(device);
1616
1617 auto shadowPass =
1618 device.createRenderPass()
1619 .addSampledDepthAttachment(vk::Format::eD32Sfloat)
1620 .addSubpass(vkb::SubpassBuilder().setDepthStencilAttachment(
1621 0, vk::ImageLayout::eDepthStencilAttachmentOptimal))
1622 .addExternalShaderReadDependencies()
1623 .build();
1624 shadowRenderPass = shadowPass;
1625
1626 for (auto &slot : shadowMaps) {
1627 for (uint32_t i = 0; i < layers; ++i) {
1628 slot.framebuffers[i] = shadowPass.createFramebuffer(
1629 device, size, size, {slot.image.layerAttachment(i)});
1630 }
1631 }
1632
1633 shadowPipelineLayout =
1634 device.createPipelineLayout()
1635 .push<glm::mat4>(vk::ShaderStageFlagBits::eVertex)
1636 .build();
1637
1638 std::vector<uint32_t> vert(mesh3d_shadow_vert_spv,
1639 mesh3d_shadow_vert_spv + mesh3d_shadow_vert_spv_count);
1640 std::vector<uint32_t> frag(mesh3d_shadow_frag_spv,
1641 mesh3d_shadow_frag_spv + mesh3d_shadow_frag_spv_count);
1642 vk::ShaderModule vertModule = vkb::PipelineBuilder::createShaderModule(device.instance, vert);
1643 vk::ShaderModule fragModule = vkb::PipelineBuilder::createShaderModule(device.instance, frag);
1644 shadowPipeline =
1645 device.createPipeline()
1646 .useClassicPipeline(vertModule, fragModule)
1647 .setPipelineLayout(shadowPipelineLayout)
1648 .setVertexInputState(vkb::VertexInputStateBuilder()
1649 .addInputBinding<MeshVertex>()
1650 .addAttributeDescription<MeshVertex>())
1651 .setDynamicStatesViewportScissor()
1652 // No cull: Cornell-style one-sided interiors keep writing when the
1653 // ceiling/walls are back-facing the sun. Closest depth still wins
1654 // on closed meshes, so floors are not punched through.
1655 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f, vk::CullModeFlagBits::eNone,
1656 vk::FrontFace::eClockwise)
1657 .setDepthBias(0.0f, 0.5f)
1658 .build(shadowPass);
1659 device->destroyShaderModule(vertModule);
1660 device->destroyShaderModule(fragModule);
1661
1662 // Alpha-cutout variant for billboard/card shadow casters (sprite-stack
1663 // slices): same transform push constant, fragment samples the albedo and
1664 // discards transparent texels so shadows follow the silhouette.
1665 shadowAlphaPipelineLayout = device.createPipelineLayout()
1666 .set(texSetLayout)
1667 .push<glm::mat4>(vk::ShaderStageFlagBits::eVertex)
1668 .build();
1669 std::vector<uint32_t> alphaVert(mesh3d_shadow_alpha_vert_spv,
1670 mesh3d_shadow_alpha_vert_spv +
1671 mesh3d_shadow_alpha_vert_spv_count);
1672 std::vector<uint32_t> alphaFrag(mesh3d_shadow_alpha_frag_spv,
1673 mesh3d_shadow_alpha_frag_spv +
1674 mesh3d_shadow_alpha_frag_spv_count);
1675 vk::ShaderModule alphaVertModule =
1676 vkb::PipelineBuilder::createShaderModule(device.instance, alphaVert);
1677 vk::ShaderModule alphaFragModule =
1678 vkb::PipelineBuilder::createShaderModule(device.instance, alphaFrag);
1679 shadowAlphaPipeline =
1680 device.createPipeline()
1681 .useClassicPipeline(alphaVertModule, alphaFragModule)
1682 .setPipelineLayout(shadowAlphaPipelineLayout)
1683 .setVertexInputState(vkb::VertexInputStateBuilder()
1684 .addInputBinding<MeshVertex>()
1685 .addAttributeDescription<MeshVertex>())
1686 .setDynamicStatesViewportScissor()
1687 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f, vk::CullModeFlagBits::eNone,
1688 vk::FrontFace::eClockwise)
1689 .setDepthBias(0.0f, 0.5f)
1690 .build(shadowPass);
1691 device->destroyShaderModule(alphaVertModule);
1692 device->destroyShaderModule(alphaFragModule);
1693
1694 auto skinVert = embeddedSpirv(mesh3d_shadow_skin_vert_spv);
1695 shadowSkinPipeline = device.createPipeline()
1696 .useClassicPipeline(skinVert, frag)
1697 .setPipelineLayout(skinPassPipelineLayout)
1698 .setVertexInputState(vkb::VertexInputStateBuilder()
1699 .addInputBinding<MeshVertex>()
1700 .addAttributeDescription<MeshVertex>())
1701 .setDynamicStatesViewportScissor()
1702 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
1703 vk::CullModeFlagBits::eNone,
1704 vk::FrontFace::eClockwise)
1705 .setDepthBias(0.0f, 0.5f)
1706 .build(shadowPass);
1707 auto skinAlphaFrag = embeddedSpirv(mesh3d_shadow_skin_alpha_frag_spv);
1708 shadowSkinAlphaPipeline = device.createPipeline()
1709 .useClassicPipeline(skinVert, skinAlphaFrag)
1710 .setPipelineLayout(skinPassPipelineLayout)
1711 .setVertexInputState(vkb::VertexInputStateBuilder()
1712 .addInputBinding<MeshVertex>()
1713 .addAttributeDescription<MeshVertex>())
1714 .setDynamicStatesViewportScissor()
1715 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
1716 vk::CullModeFlagBits::eNone,
1717 vk::FrontFace::eClockwise)
1718 .setDepthBias(0.0f, 0.5f)
1719 .build(shadowPass);
1720
1721 // Clear every ping-pong copy so sampling before the first real shadow pass
1722 // sees SHADER_READ_ONLY rather than UNDEFINED.
1723 vkb::executeImmediately(device.instance, uploadPool, device.getQueue(vkb::QueueType::graphics),
1724 [&](vk::CommandBuffer cb) {
1725 vk::ClearValue clear{};
1726 clear.depthStencil = vk::ClearDepthStencilValue{1.0f, 0};
1727 for (auto &slot : shadowMaps) {
1728 slot.image.beginDepthAttachment();
1729 for (int c = 0; c < ShadowConfig::kCascades; ++c) {
1730 vk::RenderPassBeginInfo rpBegin{};
1731 rpBegin.renderPass = shadowRenderPass;
1732 rpBegin.framebuffer = slot.framebuffers[c];
1733 rpBegin.renderArea = vk::Rect2D{{0, 0}, {size, size}};
1734 rpBegin.clearValueCount = 1;
1735 rpBegin.pClearValues = &clear;
1736 cb.beginRenderPass(rpBegin, vk::SubpassContents::eInline);
1737 cb.endRenderPass();
1738 }
1739 slot.image.endSampledLayout();
1740 }
1741 });
1742}

1744void Graphics::ensureClusteredBuffers(size_t lightsBytes, size_t tableBytes, size_t indicesBytes) {
1745 auto &st = currentClusteredStorage();
1746 auto ensure = [&](vkb::GenericBuffer &buf, size_t &cap, size_t need) {
1747 if (need == 0) need = 4;
1748 if (cap >= need && buf.buffer) return;
1749 buf.release();
1750 // Grow with some slack.
1751 size_t alloc = std::max(need, cap ? cap * 2 : need);
1752 buf.allocate(frameToken(), device, vk::BufferUsageFlagBits::eStorageBuffer, vk::DeviceSize(alloc),
1753 vk::MemoryPropertyFlagBits::eHostVisible |
1754 vk::MemoryPropertyFlagBits::eHostCoherent);
1755 cap = alloc;
1756 };
1757 ensure(st.lightsBuf, st.lightsCap, lightsBytes);
1758 ensure(st.tableBuf, st.tableCap, tableBytes);
1759 ensure(st.indicesBuf, st.indicesCap, indicesBytes);
1760 // Reallocated handles invalidate this frame's cached descriptor sets; drop
1761 // them so mesh3dClusteredSetFor rebinds against the new buffers.
1762 currentMesh3dClusteredFrameSlots().sets.clear();
1763}
