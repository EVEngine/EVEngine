16Result<double> numeric(const Value::Object& payload, std::string_view key, double fallback) {
17 const auto found = payload.find(std::string(key));
18 if (found == payload.end()) return Result<double>::success(fallback);
19 double value = 0.0;
20 if (const auto* integer = found->second.getIf<std::int64_t>())
21 value = static_cast<double>(*integer);
22 else if (const auto* decimal = found->second.getIf<double>())
23 value = *decimal;
24 else
25 return invalid<double>("audio field must be numeric", std::string(key));
26 if (!std::isfinite(value)) return invalid<double>("audio field must be finite", std::string(key));
27 return Result<double>::success(value);
28}
