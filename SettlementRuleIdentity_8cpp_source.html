54eve::Result<std::string> SettlementRuleSet::canonicalJson() const {
55 std::vector<const SettlementRule*> ordered;
56 ordered.reserve(rules_.size());
57 for (const auto& rule : rules_) ordered.push_back(&rule);
58 std::sort(ordered.begin(), ordered.end(), [](const auto* left, const auto* right) {
59 const auto leftStage = static_cast<std::uint8_t>(left->stage);
60 const auto rightStage = static_cast<std::uint8_t>(right->stage);
61 if (leftStage != rightStage) return leftStage < rightStage;
62 if (left->priority != right->priority) return left->priority < right->priority;
63 return left->id < right->id;
64 });
65
66 eve::Value::Array rules;
67 rules.reserve(ordered.size());
68 for (const auto* rule : ordered) rules.push_back(encodeRule(*rule));
69 return eve::Value(eve::Value::Object{
70 {"conditionLanguageVersion", 1},
71 {"rules", std::move(rules)},
72 {"schema", "settlement.rules"},
73 {"version", static_cast<std::int64_t>(schemaVersion())},
74 })
75 .toJson();
76}
