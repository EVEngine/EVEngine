33Result<CanonicalMeshData> decodeCanonicalMesh(std::span<const std::uint8_t> bytes, const CanonicalMeshLimits& limits) {
34 static constexpr std::uint8_t magic[] = {'E', 'V', 'M', 'E', 'S', 'H', 0};
35 if (bytes.size() < 24 || !std::equal(std::begin(magic), std::end(magic), bytes.begin()) ||
36 (bytes[7] != 1 && bytes[7] != 2 && bytes[7] != 3))
37 return failure<CanonicalMeshData>(DiagnosticCode::ParseError, "canonical mesh header is invalid");
38 const auto vertexCount = little32(bytes, 8), indexCount = little32(bytes, 12);
39 const auto flags = little32(bytes, 16), countField = little32(bytes, 20);
40 const bool legacy = bytes[7] == 1;
41 const bool colored = bytes[7] == 3 && (flags & 2u);
42 if (!vertexCount || !indexCount || indexCount % 3 ||
43 flags > (legacy ? 3u : (bytes[7] == 3 ? 3u : 1u)) || (legacy && countField))
44 return failure<CanonicalMeshData>(DiagnosticCode::ParseError, "canonical mesh metadata is invalid");
45 if (vertexCount > limits.maximumVertices || indexCount > limits.maximumIndices)
46 return failure<CanonicalMeshData>(DiagnosticCode::InvalidArgument, "canonical mesh exceeds limits");
47 const uint32_t uvCount = legacy ? ((flags & 2u) ? 1u : 0u) : countField;
48 const uint64_t header = 24ull + (legacy ? 0ull : uint64_t(uvCount) * 4);
49 const uint64_t floatsPerVertex =
50 3ull + ((flags & 1u) ? 3 : 0) + uint64_t(uvCount) * 2 + (colored ? 4 : 0);
51 // Bound the product through the budget before multiplying untrusted counts.
52 const uint64_t indexBytes = uint64_t(indexCount) * 4;
53 if (header > bytes.size() || header > limits.maximumDecodedBytes ||
54 indexBytes > limits.maximumDecodedBytes - header ||
55 floatsPerVertex > (limits.maximumDecodedBytes - header - indexBytes) / 4 / vertexCount)
56 return failure<CanonicalMeshData>(DiagnosticCode::InvalidArgument, "canonical mesh exceeds byte budget");
57 const uint64_t expected = header + uint64_t(vertexCount) * floatsPerVertex * 4 + uint64_t(indexCount) * 4;
58 if (expected != bytes.size() || expected > limits.maximumDecodedBytes)
59 return failure<CanonicalMeshData>(DiagnosticCode::InvalidArgument, "canonical mesh byte size is invalid");
60 CanonicalMeshData result;
61 result.positions.reserve(std::size_t(vertexCount) * 3);
62 if (flags & 1u) result.normals.reserve(std::size_t(vertexCount) * 3);
63 if (colored) result.colors.reserve(std::size_t(vertexCount) * 4);
64 uint32_t previous = 0;
65 for (uint32_t set = 0; set < uvCount; ++set) {
66 const auto id = legacy ? 0u : little32(bytes, 24 + size_t(set) * 4);
67 if (set && id <= previous)
68 return failure<CanonicalMeshData>(DiagnosticCode::ParseError,
69 "UV set descriptors must be strictly increasing");
70 previous = id;
71 result.texcoords[id].reserve(size_t(vertexCount) * 2);
72 }
73 std::size_t cursor = size_t(header);
74 auto appendFloats = [&](std::vector<float>& output, std::uint32_t count) -> Result<void> {
75 for (std::uint32_t index = 0; index < count; ++index) {
76 const float value = littleFloat(bytes, cursor);
77 cursor += 4;
78 if (!std::isfinite(value))
79 return Result<void>::failure(Diagnostic::error(
80 DiagnosticCode::ParseError, "canonical mesh contains a non-finite value", {}, {}, "asset.mesh"));
81 output.push_back(value);
82 }
83 return Result<void>::success();
84 };
85 for (std::uint32_t vertex = 0; vertex < vertexCount; ++vertex) {
86 auto positions = appendFloats(result.positions, 3);
87 if (!positions) return Result<CanonicalMeshData>::failure(positions.status());
88 if (flags & 1u) {
89 auto normals = appendFloats(result.normals, 3);
90 if (!normals) return Result<CanonicalMeshData>::failure(normals.status());
91 }
92 for (auto& [set, values] : result.texcoords) {
93 auto texcoords = appendFloats(values, 2);
94 if (!texcoords) return Result<CanonicalMeshData>::failure(texcoords.status());
95 }
96 if (colored) {
97 auto colors = appendFloats(result.colors, 4);
98 if (!colors) return Result<CanonicalMeshData>::failure(colors.status());
99 }
100 }
101 result.indices.reserve(indexCount);
102 for (std::uint32_t index = 0; index < indexCount; ++index) {
103 const std::uint32_t value = little32(bytes, cursor);
104 cursor += 4;
105 if (value >= vertexCount)
106 return failure<CanonicalMeshData>(DiagnosticCode::ParseError, "canonical mesh index exceeds vertex count");
107 result.indices.push_back(value);
108 }
109 return Result<CanonicalMeshData>::success(std::move(result));
110}
