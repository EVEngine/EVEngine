27Result<CanonicalMeshData> decodeCanonicalMesh(std::span<const std::uint8_t> bytes, const CanonicalMeshLimits& limits) {
28 static constexpr std::uint8_t magic[] = {'E', 'V', 'M', 'E', 'S', 'H', 0};
29 if (bytes.size() < 24 || !std::equal(std::begin(magic), std::end(magic), bytes.begin()) ||
30 (bytes[7] != 1 && bytes[7] != 2))
31 return failure<CanonicalMeshData>(DiagnosticCode::ParseError, "canonical mesh header is invalid");
32 const auto vertexCount = little32(bytes, 8), indexCount = little32(bytes, 12);
33 const auto flags = little32(bytes, 16), countField = little32(bytes, 20);
34 const bool legacy = bytes[7] == 1;
35 if (!vertexCount || !indexCount || indexCount % 3 || flags > (legacy ? 3u : 1u) || (legacy && countField))
36 return failure<CanonicalMeshData>(DiagnosticCode::ParseError, "canonical mesh metadata is invalid");
37 if (vertexCount > limits.maximumVertices || indexCount > limits.maximumIndices)
38 return failure<CanonicalMeshData>(DiagnosticCode::InvalidArgument, "canonical mesh exceeds limits");
39 const uint32_t uvCount = legacy ? ((flags & 2u) ? 1u : 0u) : countField;
40 const uint64_t header = 24ull + (legacy ? 0ull : uint64_t(uvCount) * 4);
41 const uint64_t floatsPerVertex = 3ull + ((flags & 1u) ? 3 : 0) + uint64_t(uvCount) * 2;
42 // Bound the product through the budget before multiplying untrusted counts.
43 const uint64_t indexBytes = uint64_t(indexCount) * 4;
44 if (header > bytes.size() || header > limits.maximumDecodedBytes ||
45 indexBytes > limits.maximumDecodedBytes - header ||
46 floatsPerVertex > (limits.maximumDecodedBytes - header - indexBytes) / 4 / vertexCount)
47 return failure<CanonicalMeshData>(DiagnosticCode::InvalidArgument, "canonical mesh exceeds byte budget");
48 const uint64_t expected = header + uint64_t(vertexCount) * floatsPerVertex * 4 + uint64_t(indexCount) * 4;
49 if (expected != bytes.size() || expected > limits.maximumDecodedBytes)
50 return failure<CanonicalMeshData>(DiagnosticCode::InvalidArgument, "canonical mesh byte size is invalid");
51 CanonicalMeshData result;
52 result.positions.reserve(std::size_t(vertexCount) * 3);
53 if (flags & 1u) result.normals.reserve(std::size_t(vertexCount) * 3);
54 uint32_t previous = 0;
55 for (uint32_t set = 0; set < uvCount; ++set) {
56 const auto id = legacy ? 0u : little32(bytes, 24 + size_t(set) * 4);
57 if (set && id <= previous)
58 return failure<CanonicalMeshData>(DiagnosticCode::ParseError,
59 "UV set descriptors must be strictly increasing");
60 previous = id;
61 result.texcoords[id].reserve(size_t(vertexCount) * 2);
62 }
63 std::size_t cursor = size_t(header);
64 auto appendFloats = [&](std::vector<float>& output, std::uint32_t count) -> Result<void> {
65 for (std::uint32_t index = 0; index < count; ++index) {
66 const float value = littleFloat(bytes, cursor);
67 cursor += 4;
68 if (!std::isfinite(value))
69 return Result<void>::failure(Diagnostic::error(
70 DiagnosticCode::ParseError, "canonical mesh contains a non-finite value", {}, {}, "asset.mesh"));
71 output.push_back(value);
72 }
73 return Result<void>::success();
74 };
75 for (std::uint32_t vertex = 0; vertex < vertexCount; ++vertex) {
76 auto positions = appendFloats(result.positions, 3);
77 if (!positions) return Result<CanonicalMeshData>::failure(positions.status());
78 if (flags & 1u) {
79 auto normals = appendFloats(result.normals, 3);
80 if (!normals) return Result<CanonicalMeshData>::failure(normals.status());
81 }
82 for (auto& [set, values] : result.texcoords) {
83 auto texcoords = appendFloats(values, 2);
84 if (!texcoords) return Result<CanonicalMeshData>::failure(texcoords.status());
85 }
86 }
87 result.indices.reserve(indexCount);
88 for (std::uint32_t index = 0; index < indexCount; ++index) {
89 const std::uint32_t value = little32(bytes, cursor);
90 cursor += 4;
91 if (value >= vertexCount)
92 return failure<CanonicalMeshData>(DiagnosticCode::ParseError, "canonical mesh index exceeds vertex count");
93 result.indices.push_back(value);
94 }
95 return Result<CanonicalMeshData>::success(std::move(result));
96}
