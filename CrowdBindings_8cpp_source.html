8Result<SpawnBatch> readSpawnBatch(ssq::Array agents, ssq::Table options) {
9 try {
10 if (agents.size() > 1024)
11 return Result<SpawnBatch>::failure(Diagnostic::error(
12 DiagnosticCode::InvalidArgument, "Spawn batches support at most 1024 agents", "agents"));
13 SpawnBatch batch;
14 const auto mode = options.get<std::string>("policy");
15 if (mode == "reject")
16 batch.policy = SpawnPolicy::RejectOverlap;
17 else if (mode == "nearestFree")
18 batch.policy = SpawnPolicy::NearestFree;
19 else if (mode == "pushNeighbors")
20 batch.policy = SpawnPolicy::PushNeighbors;
21 else
22 return Result<SpawnBatch>::failure(
23 Diagnostic::error(DiagnosticCode::InvalidArgument, "Unknown spawn policy", "policy"));
24 batch.maxDistance = options.get<float>("maxDistance");
25 batch.searchSpacing = options.get<float>("searchSpacing");
26 batch.maxPasses = options.get<int>("maxPasses");
27 batch.maxChecks = options.get<int>("maxChecks");
28 for (size_t i = 0; i < agents.size(); ++i) {
29 auto item = agents.get<ssq::Table>(i);
30 SpawnRequest request;
31 request.stableId = item.get<std::string>("stableId");
32 request.x = item.get<float>("x");
33 request.y = item.get<float>("y");
34 request.heading = item.get<float>("heading");
35 request.radius = item.get<float>("radius");
36 request.interaction = {item.get<float>("pushability"), item.get<bool>("holdPosition"),
37 item.get<int>("layer"), item.get<int>("mask")};
38 batch.agents.push_back(std::move(request));
39 }
40 return Result<SpawnBatch>::success(std::move(batch));
41 } catch (const std::exception &error) {
42 return Result<SpawnBatch>::failure(
43 Diagnostic::error(DiagnosticCode::InvalidArgument, error.what(), "spawnBatch"));
44 }
45}
