178 [[nodiscard]] static HexVec3 corner(HexDirection direction, float scale = 1.f) noexcept {
179 const float r = kOuterRadius * scale;
180 const float i = kInnerRadius * scale;
181 switch (direction) {
182 case HexDirection::NE: return HexVec3{0.f, 0.f, r};
183 case HexDirection::E: return HexVec3{i, 0.f, 0.5f * r};
184 case HexDirection::SE: return HexVec3{i, 0.f, -0.5f * r};
185 case HexDirection::SW: return HexVec3{0.f, 0.f, -r};
186 case HexDirection::W: return HexVec3{-i, 0.f, -0.5f * r};
187 case HexDirection::NW: return HexVec3{-i, 0.f, 0.5f * r};
188 }
189 return HexVec3{0.f, 0.f, r};

253 if (len <= 1e-6f) return HexVec3{0.f, 0.f, 0.f};
254 const float s = (kWallThickness * 0.5f) / len;
