183 [[nodiscard]] static HexVec3 corner(HexDirection direction, float scale = 1.f) noexcept {
184 const float r = kOuterRadius * scale;
185 const float i = kInnerRadius * scale;
186 switch (direction) {
187 case HexDirection::NE: return HexVec3{0.f, 0.f, r};
188 case HexDirection::E: return HexVec3{i, 0.f, 0.5f * r};
189 case HexDirection::SE: return HexVec3{i, 0.f, -0.5f * r};
190 case HexDirection::SW: return HexVec3{0.f, 0.f, -r};
191 case HexDirection::W: return HexVec3{-i, 0.f, -0.5f * r};
192 case HexDirection::NW: return HexVec3{-i, 0.f, 0.5f * r};
193 }
194 return HexVec3{0.f, 0.f, r};

255 [[nodiscard]] static HexVec3 wallThicknessOffset(HexVec3 near, HexVec3 far) noexcept {

259 if (len <= 1e-6f) return HexVec3{0.f, 0.f, 0.f};
260 const float s = (kWallThickness * 0.5f) / len;
261 return HexVec3{dx * s, 0.f, dz * s};
