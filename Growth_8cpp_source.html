37Result<void> Growth::stepSeconds(double seconds) {
38 if (!std::isfinite(seconds) || seconds < 0.0) {
39 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
40 "seconds must be finite and >= 0", "growth.stepSeconds"));
41 }
42 auto delta = Duration::fromSeconds(seconds);
43 if (!delta) return Result<void>::failure(delta.status());
44 SimulationStep step;
45 step.delta = std::move(delta).takeValue();
46 step.tick = SimulationTick::zero();
47 return GrowthWorld::instance().step(step);
48}
