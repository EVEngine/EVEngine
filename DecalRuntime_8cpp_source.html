36EditorResult<void> DecalRuntimeBinding::publish(const DecalDocumentTarget& document) {
37 if (!manager_)
38 return fail<void>(EditorStatus::Rejected, "editor.decal.manager", "Live DecalManager is required");
39 const auto diagnostics = document.validate();
40 if (std::any_of(diagnostics.begin(), diagnostics.end(),
41 [](const auto& diagnostic) { return diagnostic.severity() == DiagnosticSeverity::Error; }))
42 return EditorResult<void>::failure(eve::Status(EditorStatus::Rejected, diagnostics));
43
44 auto albedo = resolve(document, assets_, "texture.albedo");
45 if (!albedo.ok()) return EditorResult<void>::failure(albedo.status());
46 auto normal = resolve(document, assets_, "texture.normal");
47 if (!normal.ok()) return EditorResult<void>::failure(normal.status());
48 auto params = resolve(document, assets_, "texture.params");
49 if (!params.ok()) return EditorResult<void>::failure(params.status());
50
51 decal::DecalInstance candidate;
52 const auto& position = array(document, "transform.position");
53 const auto& direction = array(document, "transform.normal");
54 const auto& uv = array(document, "texture.uvRect");
55 candidate.x = static_cast<float>(*position[0].getIf<double>());
56 candidate.y = static_cast<float>(*position[1].getIf<double>());
57 candidate.z = static_cast<float>(*position[2].getIf<double>());
58 candidate.nx = static_cast<float>(*direction[0].getIf<double>());
59 candidate.ny = static_cast<float>(*direction[1].getIf<double>());
60 candidate.nz = static_cast<float>(*direction[2].getIf<double>());
61 candidate.yaw = static_cast<float>(number(document, "transform.yaw"));
62 candidate.size = static_cast<float>(number(document, "projection.size"));
63 candidate.depth = static_cast<float>(number(document, "projection.depth"));
64 candidate.kind = *document.value("decal.kind")->getIf<std::string>();
65 candidate.albedo = albedo.value();
66 candidate.normal = normal.value();
67 candidate.params = params.value();
68 for (int index = 0; index < 4; ++index)
69 candidate.uvRect[index] = static_cast<float>(*uv[index].getIf<double>());
70 candidate.normalStrength = static_cast<float>(number(document, "channel.normal"));
71 candidate.roughnessStrength = static_cast<float>(number(document, "channel.roughness"));
72 candidate.metalStrength = static_cast<float>(number(document, "channel.metal"));
73 candidate.emissiveStrength = static_cast<float>(number(document, "channel.emissive"));
74 candidate.blendMode = *document.value("blend.mode")->getIf<std::string>() == "add" ? 1 : 0;
75 candidate.lifetime = static_cast<float>(number(document, "lifetime.seconds"));
76 candidate.fadeIn = static_cast<float>(number(document, "lifetime.fadeIn"));
77 candidate.fadeOut = static_cast<float>(number(document, "lifetime.fadeOut"));
78
79 const int next = manager_->replace(runtimeId_, std::move(candidate));
80 if (next == 0)
81 return fail<void>(EditorStatus::Conflict, "editor.decal.replace",
82 "Decal runtime generation is stale or invalid");
83 runtimeId_ = next;
84 return eve::editing::applied<void>(diagnostics);

110EditorResult<void> DecalPublishingTarget::commitDomainState(
