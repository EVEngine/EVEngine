15float distanceToCell(float x, float z, int cellX, int cellZ, float size) {
16 const float centerX = (float(cellX) + 0.5f) * size;
17 const float centerZ = (float(cellZ) + 0.5f) * size;
18 const float dx = centerX - x;
19 const float dz = centerZ - z;
20 return std::sqrt(dx * dx + dz * dz);
21}

93int RuntimeGeneration::trimToResidentPoints(int target) {
94 target = std::max(0, target);
95 uint64_t projectedResident = 0;
96 for (const auto& [key, cell] : cells_)
97 if (cell.state != State::Cleanup)
98 projectedResident += uint64_t(cell.output.getCount());
99 if (projectedResident <= uint64_t(target)) return 0;
100 std::vector<CellKey> candidates;
101 for (const auto& [key, cell] : cells_)
102 if (cell.state == State::Active) candidates.push_back(key);
103 std::sort(candidates.begin(), candidates.end(), [this](const CellKey& a, const CellKey& b) {
104 const float aPriority = cells_.at(a).priority;
105 const float bPriority = cells_.at(b).priority;
106 if (aPriority != bPriority) return aPriority > bPriority;
107 if (a.level != b.level) return a.level > b.level;
108 if (a.z != b.z) return a.z > b.z;
109 return a.x > b.x;
110 });
111 int scheduled = 0;
112 for (const auto& key : candidates) {
113 if (projectedResident <= uint64_t(target)) break;
114 auto& cell = cells_.at(key);
115 projectedResident -= uint64_t(cell.output.getCount());
116 transitionCellState(cell, State::Cleanup);
117 cell.trimmed = true;
118 cell.ticket = ++nextTicket_;
119 cleanupQueue_.push_back(key);
120 ++scheduled;
121 }
122 sortQueues();
123 return scheduled;
124}

215void RuntimeGeneration::startRefreshPlan() {
216 constexpr float degreesToRadians = 0.017453292519943295f;
217 RefreshPlan plan;
218 plan.revision = sourceRevision_;
219 plan.levels = levels_;
220 plan.directionWeight = directionWeight_;
221 plan.frustumHalfAngle = frustumHalfAngle_;
222 plan.coneCosine = std::cos(frustumHalfAngle_ * degreesToRadians);
223 plan.frustumBehindRadius = frustumBehindRadius_;
224 plan.frustumCulling = frustumCulling_;
225 plan.sources.reserve(sourceOrder_.size());
226 for (const auto& sourceId : sourceOrder_) plan.sources.push_back(sources_.at(sourceId));

230uint64_t RuntimeGeneration::advanceRefreshPlan(uint64_t candidateBudget) {
231 uint64_t processed = 0;
232 while (refreshPlan_) {
233 auto& plan = *refreshPlan_;
234 if (plan.sourceIndex >= plan.sources.size() || plan.levels.empty()) {
235 const uint64_t committedRevision = plan.revision;
236 commitRefreshPlan(plan);
237 committedRefreshRevision_ = committedRevision;
238 refreshPlan_.reset();
239 if (committedRefreshRevision_ != sourceRevision_) {
240 startRefreshPlan();
241 continue;
242 }
243 break;
244 }
245 if (plan.levelIndex >= plan.levels.size()) {
246 ++plan.sourceIndex;
247 plan.levelIndex = 0;
248 plan.rangeReady = false;
249 continue;
250 }
251 if (processed >= candidateBudget) break;
252
253 const auto& source = plan.sources[plan.sourceIndex];
254 const size_t candidateLevelIndex = plan.levelIndex;
255 const auto& level = plan.levels[candidateLevelIndex];
256 const float generationRadius = level.generationRadius * source.radiusScale;
257 if (!plan.rangeReady) {
258 plan.minX = int(std::floor((source.x - generationRadius) / level.cellSize));
259 plan.maxX = int(std::floor((source.x + generationRadius) / level.cellSize));
260 plan.minZ = int(std::floor((source.z - generationRadius) / level.cellSize));
261 plan.maxZ = int(std::floor((source.z + generationRadius) / level.cellSize));
262 plan.cellX = plan.minX;
263 plan.cellZ = plan.minZ;
264 plan.rangeReady = true;
265 }
266 const int cellX = plan.cellX;
267 const int cellZ = plan.cellZ;
268 if (++plan.cellX > plan.maxX) {
269 plan.cellX = plan.minX;
270 if (++plan.cellZ > plan.maxZ) {
271 ++plan.levelIndex;
272 plan.rangeReady = false;
273 }
274 }
275 ++processed;
276
277 const float centerX = (float(cellX) + 0.5f) * level.cellSize;
278 const float centerZ = (float(cellZ) + 0.5f) * level.cellSize;
279 const float dx = centerX - source.x;
280 const float dz = centerZ - source.z;
281 const float distance = std::sqrt(dx * dx + dz * dz);
282 if (distance > generationRadius) continue;
283 const bool hasDirection = source.directionX != 0.f || source.directionZ != 0.f;
284 const float forward =
285 distance > 0.f && hasDirection ? (dx * source.directionX + dz * source.directionZ) / distance : 1.f;
286 if (plan.frustumCulling && hasDirection && distance > plan.frustumBehindRadius && forward < plan.coneCosine)
287 continue;
288 const CellKey key{int(candidateLevelIndex), cellX, cellZ};
289 const float priority = distance / std::max(generationRadius, 0.0001f) - plan.directionWeight * forward;
290 const auto [found, inserted] = plan.desiredPriorities.emplace(key, priority);
291 if (!inserted) found->second = std::min(found->second, priority);
292 }
293 return processed;
294}

296void RuntimeGeneration::commitRefreshPlan(const RefreshPlan& plan) {
297 for (auto& [key, cell] : cells_)
298 if (cell.state != State::Cleanup) cell.priority = std::numeric_limits<float>::max();
299 for (const auto& [key, priority] : plan.desiredPriorities) {
300 const auto existing = cells_.find(key);
301 if (existing != cells_.end()) {
302 if (existing->second.state == State::Cleanup) {
303 if (existing->second.trimmed) continue;
304 existing->second.ticket = ++nextTicket_;
305 transitionCellState(existing->second, existing->second.revision > 0 ? State::Active : State::Pending);
306 existing->second.trimmed = false;
307 cleanupQueue_.erase(std::remove(cleanupQueue_.begin(), cleanupQueue_.end(), key), cleanupQueue_.end());
308 if (existing->second.state == State::Pending) generateQueue_.push_back(key);
309 }
310 existing->second.priority = std::min(existing->second.priority, priority);
311 continue;
312 }
313 Cell cell;
314 cell.priority = priority;
315 cells_.emplace(key, cell);
316 generateQueue_.push_back(key);
317 }
318 for (auto& [key, cell] : cells_) {
319 if (cell.state == State::Cleanup) continue;
320 if ((cell.state == State::Pending || cell.state == State::Generating) &&
321 plan.desiredPriorities.find(key) == plan.desiredPriorities.end()) {
322 if (cell.state == State::Generating) ++cancelledGenerationCount_;
323 transitionCellState(cell, State::Cleanup);
324 cell.trimmed = false;
325 cell.ticket = ++nextTicket_;
326 cleanupQueue_.push_back(key);
327 continue;
328 }
329 const auto& level = plan.levels[size_t(key.level)];
330 bool retained = false;
331 for (const auto& source : plan.sources) {
332 if (distanceToCell(source.x, source.z, key.x, key.z, level.cellSize) <=
333 level.cleanupRadius * source.radiusScale) {
334 retained = true;
335 break;
336 }
337 }
338 if (retained) continue;
339 transitionCellState(cell, State::Cleanup);
340 cell.trimmed = false;
341 cell.ticket = ++nextTicket_;
342 cleanupQueue_.push_back(key);
343 }
344 generateQueue_.erase(
345 std::remove_if(generateQueue_.begin(), generateQueue_.end(), [this](const CellKey& key) {
346 const auto found = cells_.find(key);
347 return found == cells_.end() || found->second.state != State::Pending;
348 }),
349 generateQueue_.end());
350 sortQueues();
351}

353void RuntimeGeneration::sortQueues() {
354 const auto priorityLess = [this](const CellKey& a, const CellKey& b) {
355 const float aPriority = cells_.at(a).priority;
356 const float bPriority = cells_.at(b).priority;
357 if (aPriority != bPriority) return aPriority < bPriority;
358 if (a.level != b.level) return a.level < b.level;
359 if (a.z != b.z) return a.z < b.z;
360 return a.x < b.x;
361 };
362 std::sort(generateQueue_.begin(), generateQueue_.end(), priorityLess);
363 std::sort(cleanupQueue_.begin(), cleanupQueue_.end(), [](const CellKey& a, const CellKey& b) {
364 if (a.level != b.level) return a.level < b.level;
365 if (a.z != b.z) return a.z < b.z;
366 return a.x < b.x;
367 });
368}

395ProcgenCellRequest* RuntimeGeneration::nextGenerate() {
396 if (generateQueue_.empty() || getGeneratingCount() >= maxGenerating_) return nullptr;
397 if (maxActiveCells_ > 0 && getActiveCellCount() + getGeneratingCount() >= maxActiveCells_)
398 return nullptr;
399 if (frameTimeBudgetMs_ > 0.f && frameStartedNs_ != 0) {
400 const uint64_t now = uint64_t(std::chrono::duration_cast<std::chrono::nanoseconds>(
401 std::chrono::steady_clock::now().time_since_epoch())
402 .count());
403 if (float(now - frameStartedNs_) * 0.000001f >= frameTimeBudgetMs_) return nullptr;
404 }
405 while (!generateQueue_.empty()) {
406 const CellKey key = generateQueue_.front();
407 generateQueue_.pop_front();
408 const auto found = cells_.find(key);
409 if (found == cells_.end() || found->second.state != State::Pending) continue;
410 transitionCellState(found->second, State::Generating);
411 found->second.ticket = ++nextTicket_;
412 return makeRequest(key);
413 }
414 return nullptr;
415}

438bool RuntimeGeneration::completeGeneration(ProcgenCellRequest* request, PointSet* output) {
439 if (!request || !output) return false;
440 const CellKey key{request->level_, request->x_, request->z_};
441 const auto found = cells_.find(key);
442 if (found == cells_.end() || found->second.state != State::Generating ||
443 request->seed_ != cellSeed(key) || request->ticket_ != found->second.ticket)
444 return false;
445 const int outputPoints = output->getCount();
446 if ((maxPointsPerCell_ > 0 && outputPoints > maxPointsPerCell_) ||
447 (maxResidentPoints_ > 0 && getResidentPointCount() > maxResidentPoints_ - outputPoints)) {
448 ++rejectedOutputCount_;
449 if (maxResidentPoints_ > 0 && outputPoints <= maxResidentPoints_)
450 trimToResidentPoints(maxResidentPoints_ - outputPoints);
451 return false;
452 }
453 found->second.output = *output;
454 found->second.hasDelta = false;
455 found->second.failures = 0;
456 ++found->second.revision;
457 transitionCellState(found->second, State::Active);
458 return true;
459}

493Result<void> RuntimeGeneration::validateCleanups(const std::vector<const ProcgenCellRequest*>& requests) const {
494 if (requests.empty())
495 return Result<void>::failure(Diagnostic::error(
496 DiagnosticCode::InvalidArgument, "cleanup transaction requires at least one request", "requests"));
497
498 std::unordered_set<CellKey, CellKeyHash> uniqueKeys;
499 uniqueKeys.reserve(requests.size());
500 for (const auto* request : requests) {
501 if (!request)
502 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
503 "cleanup transaction contains a null request", "requests"));
504 const CellKey key{request->level_, request->x_, request->z_};
505 if (!uniqueKeys.insert(key).second)
506 return Result<void>::failure(Diagnostic::error(
507 DiagnosticCode::Conflict, "cleanup transaction contains a duplicate cell", "requests"));
508 const auto found = cells_.find(key);
509 if (found == cells_.end() || found->second.state != State::Cleanup || request->seed_ != cellSeed(key) ||
510 request->ticket_ != found->second.ticket)
511 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Conflict,
512 "cleanup transaction contains a stale request", "requests"));
513 }
514 return Result<void>::success();
515}

538Result<uint64_t> RuntimeGeneration::applyCellUpdate(int level, int x, int z, uint64_t expectedRevision,
539 const PointSet& output) {
540 const auto found = cells_.find({level, x, z});
541 if (found == cells_.end() || found->second.state != State::Active)
542 return Result<uint64_t>::failure(
543 Diagnostic::error(DiagnosticCode::NotFound, "active runtime-generation cell was not found", "cell"));
544 if (expectedRevision == 0 || found->second.revision != expectedRevision)
545 return Result<uint64_t>::failure(
546 Diagnostic::error(DiagnosticCode::Conflict, "runtime-generation cell revision is stale", "revision"));
547
548 auto delta = diffPointSets(found->second.output, output);
549 if (!delta.ok()) return Result<uint64_t>::failure(delta.status());
550 auto staged = applyPointDelta(found->second.output, delta.value());
551 if (!staged.ok()) return Result<uint64_t>::failure(staged.status());
552
553 const int outputPoints = staged.value().getCount();
554 const std::int64_t projectedResident = std::int64_t(getResidentPointCount()) - found->second.output.getCount() +
555 outputPoints;
556 if ((maxPointsPerCell_ > 0 && outputPoints > maxPointsPerCell_) ||
557 (maxResidentPoints_ > 0 && projectedResident > maxResidentPoints_)) {
558 ++rejectedOutputCount_;
559 return Result<uint64_t>::failure(
560 Diagnostic::error(DiagnosticCode::PreconditionViolation,
561 "runtime-generation cell update exceeds the configured point budget", "output"));
562 }
563
564 found->second.output = std::move(staged).takeValue();
565 found->second.delta = std::move(delta).takeValue();
566 found->second.hasDelta = true;
567 ++found->second.revision;
568 return Result<uint64_t>::success(found->second.revision);
569}
570
571Result<uint64_t> RuntimeGeneration::migrateCellPointIds(int level, int x, int z, uint64_t expectedRevision) {
572 const CellKey key{level, x, z};
573 const auto found = cells_.find(key);
574 if (found == cells_.end() || found->second.state != State::Active)
575 return Result<uint64_t>::failure(
576 Diagnostic::error(DiagnosticCode::NotFound, "active runtime-generation cell was not found", "cell"));
577 if (expectedRevision == 0 || found->second.revision != expectedRevision)
578 return Result<uint64_t>::failure(
579 Diagnostic::error(DiagnosticCode::Conflict, "runtime-generation cell revision is stale", "revision"));
580
581 PointSet staged = found->second.output;
582 const std::uint64_t identityNamespace =
583 derivePointId((std::uint64_t(worldSeed_) << 32u) | cellSeed(key), found->second.revision);
584 auto assigned = staged.assignPointIds(identityNamespace);
585 if (!assigned.ok()) return Result<uint64_t>::failure(assigned.status());
586 found->second.output = std::move(staged);
587 found->second.hasDelta = false;
588 ++found->second.revision;
589 return Result<uint64_t>::success(found->second.revision);
590}

596 : nullptr;

613size_t RuntimeGeneration::CellKeyHash::operator()(const CellKey& key) const {
614 size_t hash = size_t(uint32_t(key.level));
615 hash ^= size_t(uint32_t(key.x)) * 0x9e3779b1u + (hash << 6u) + (hash >> 2u);
616 hash ^= size_t(uint32_t(key.z)) * 0x85ebca77u + (hash << 6u) + (hash >> 2u);
617 return hash;
618}
