38EditorResult<TaskId> MaterialPreviewService::render(const DocumentId& document,
39 const MaterialDocumentTarget& material,
40 MaterialPreviewSettings settings,
41 IMaterialPreviewRenderer& renderer) {
42 if (document.empty())
43 return previewError<TaskId>(EditorStatus::Rejected, "editor.material.preview-document",
44 "Material preview document id is required");
45 auto diagnostics = material.validate();
46 auto settingsDiagnostics = validateSettings(settings);
47 diagnostics.insert(diagnostics.end(), std::make_move_iterator(settingsDiagnostics.begin()),
48 std::make_move_iterator(settingsDiagnostics.end()));
49 for (const EditorDiagnostic& diagnostic : diagnostics)
50 if (diagnostic.severity() == DiagnosticSeverity::Error)
51 return EditorResult<TaskId>::failure(
52 Status(EditorStatus::Rejected, std::move(diagnostics)));
53 const TaskId task("material-preview-" + std::to_string(++sequence_));
54 MaterialPreviewRenderRequest request;
55 request.sceneId = StableId(task.value() + ":isolated-scene");
56 request.document = document;
57 request.documentRevision = material.revision();
58 request.material = material.snapshotValue();
59 request.settings = std::move(settings);
60 MaterialPreviewRenderResult rendered = renderer.render(request);
61 rendered.diagnostics.insert(rendered.diagnostics.begin(), diagnostics.begin(), diagnostics.end());
62 if (rendered.status == EditorStatus::Applied && rendered.artifact.empty()) {
63 rendered.status = EditorStatus::Failed;
64 rendered.diagnostics.push_back(eve::editing::ruleDiagnostic(
65 DiagnosticCode::Failed, RuleId("editor.material.preview-artifact-required"),
66 DiagnosticSeverity::Error, "Preview renderer succeeded without an artifact"));
67 }
68 tasks_.emplace(task, Task{document, request.documentRevision, std::move(rendered)});
69 return eve::editing::applied<TaskId>(task);
70}

81EditorResult<void> MaterialPreviewService::publish(const DocumentId& document, Revision currentRevision,
82 const TaskId& task) {
83 const auto found = tasks_.find(task);
84 if (found == tasks_.end())
85 return previewError<void>(EditorStatus::NotFound, "editor.material.preview-task-not-found",
86 "Material preview task was not found");
87 if (found->second.document != document)
88 return previewError<void>(EditorStatus::Rejected, "editor.material.preview-document-mismatch",
89 "Material preview belongs to another document");
90 if (found->second.revision != currentRevision)
91 return previewError<void>(EditorStatus::Conflict, "editor.material.preview-stale",
92 "Material changed after the preview was rendered");
93 if (found->second.result.status != EditorStatus::Applied || found->second.result.artifact.empty()) {
94 const EditorStatus status = found->second.result.status == EditorStatus::Applied
95 ? EditorStatus::Failed
96 : found->second.result.status;
97 return EditorResult<void>::failure(Status(status, found->second.result.diagnostics));
98 }
99 published_[document] = found->second.result.artifact;
100 return eve::editing::applied<void>();
101}
