29bool validElement(const VegetationSceneElement& element) {
30 static const std::set<std::string> kinds{
31 "color-effect", "color-map", "color-noise", "color-tint", "extras-alpha", "extras-emissive",
32 "extras-overlay", "extras-wetness", "motion-advanced", "motion-interaction", "motion-wind-power",
33 "vertex-conform-model", "vertex-conform-simple", "vertex-conform-terrain", "vertex-height-offset",
34 "vertex-height", "vertex-orientation-model", "vertex-orientation-terrain", "vertex-size"};
35 const std::uint32_t expectedChannel = element.kind.starts_with("color-") ? 0u
36 : element.kind.starts_with("extras-") ? 1u
37 : element.kind.starts_with("motion-") ? 2u
38 : 3u;
39 const float rotationLength = std::sqrt(std::inner_product(element.rotation.begin(), element.rotation.end(),
40 element.rotation.begin(), 0.f));
41 std::set<std::string> propertyNames;
42 const auto finiteValues = [](const auto& values) {
43 return std::all_of(values.begin(), values.end(), [](float value) { return finite(value); });
44 };
45 if (element.sourceFileId == 0 || !validGuid(element.shaderGuid, false) ||
46 !validGuid(element.textureGuid, true) || !kinds.contains(element.kind) || element.channel != expectedChannel ||
47 (element.visibility != -1 && element.visibility != 0 && element.visibility != 10 &&
48 element.visibility != 20) ||
49 element.layers == 0 || (element.layers & ~0x1ffu) || !finite(element.intensity) || element.intensity < 0.f ||
50 element.intensity > 1.f || !finite(element.motionPower) || element.motionPower < 0.f ||
51 element.motionPower > 1.f || !finiteValues(element.position) || !finiteValues(element.rotation) ||
52 !finiteValues(element.scale) || !finiteValues(element.value) || !finiteValues(element.remap) ||
53 !finiteValues(element.seasons[0]) || !finiteValues(element.seasons[1]) ||
54 !finiteValues(element.seasons[2]) || !finiteValues(element.seasons[3]) ||
55 std::abs(rotationLength - 1.f) > 1e-3f ||
56 std::any_of(element.scale.begin(), element.scale.end(), [](float value) { return value <= 0.f; }) ||
57 element.remap[0] > element.remap[1] || element.remap[2] > element.remap[3] ||
58 element.remap[4] > element.remap[5] || element.blendRgb < 0 || element.blendRgb > 2 ||
59 element.blendAlpha < 0 || element.blendAlpha > 1 ||
60 (element.directionMode != 10 && element.directionMode != 20 && element.directionMode != 30 &&
61 element.directionMode != 40) ||
62 (element.motionMode != 13 && element.motionMode != 15) || element.properties.size() > 256 ||
63 (!element.textureAsset.empty() && !AssetRef::parse(element.textureAsset)))
64 return false;
65 for (const auto& property : element.properties) {
66 if (property.name.empty() || property.name.size() > 128 || property.type < 0 || property.type > 2 ||
67 !validGuid(property.textureGuid, true) ||
68 !finiteValues(property.vector) || !finite(property.value) ||
69 (!property.textureAsset.empty() && !AssetRef::parse(property.textureAsset)) ||
70 !propertyNames.emplace(property.name).second)
71 return false;
72 }
73 return true;
74}

109Result<VegetationSceneElementPublication> VegetationSceneLiveElements::registerElement(
110 std::uint64_t expectedRevision, std::uint64_t expectedGpuRevision, const VegetationSceneElement& element,
111 VegetationSceneGpuRuntime& runtime, const graphics::PbrSurface& baseSurface,
112 const graphics::VegetationMotion& baseMotion, const std::map<std::string, graphics::VegetationMask>& masks,
113 const VegetationSceneGpuBuild& build) {
114 if (expectedRevision != revision_ || expectedGpuRevision != runtime.sourceRevision())
115 return fail<VegetationSceneElementPublication>(DiagnosticCode::Conflict,
116 "vegetation live-element state changed before registration");
117 if (revision_ == std::numeric_limits<std::uint64_t>::max() || !validElement(element))
118 return fail<VegetationSceneElementPublication>(DiagnosticCode::InvalidArgument,
119 "vegetation live Element is invalid");
120 std::size_t slotIndex = 0;
121 while (slotIndex < slots_.size() && (slots_[slotIndex].element || slots_[slotIndex].retired)) ++slotIndex;
122 const auto activeCount = std::count_if(slots_.begin(), slots_.end(), [](const Slot& slot) {
123 return slot.element.has_value();
124 });
125 if (baseScene_.elements.size() + std::size_t(activeCount) + 1 > 4096)
126 return fail<VegetationSceneElementPublication>(DiagnosticCode::InvalidArgument,
127 "vegetation live-element capacity exceeded");
128 auto candidate = snapshot();
129 if (!candidate) return Result<VegetationSceneElementPublication>::failure(candidate.status());
130 try {
131 candidate.value().elements.push_back(element);
132 if (slotIndex == slots_.size()) slots_.reserve(slots_.size() + 1);
133 } catch (const std::bad_alloc&) {
134 return fail<VegetationSceneElementPublication>(DiagnosticCode::Failed,
135 "vegetation live-element registration allocation failed");
136 }
137 auto published = runtime.replace(expectedGpuRevision, baseSurface, baseMotion, candidate.value(), masks, build);
138 if (!published) return Result<VegetationSceneElementPublication>::failure(published.status());
139 if (slotIndex == slots_.size()) slots_.push_back({});
140 auto& slot = slots_[slotIndex];
141 slot.element.emplace(std::move(candidate.value().elements.back()));
142 ++revision_;
143 return Result<VegetationSceneElementPublication>::success(
144 {{owner_, static_cast<VegetationSceneElementHandle::index_type>(slotIndex), slot.generation}, revision_,
145 published.value()});

148Result<VegetationSceneElementPublication> VegetationSceneLiveElements::withdrawElement(
149 std::uint64_t expectedRevision, std::uint64_t expectedGpuRevision, VegetationSceneElementHandle handle,
150 VegetationSceneGpuRuntime& runtime, const graphics::PbrSurface& baseSurface,
151 const graphics::VegetationMotion& baseMotion, const std::map<std::string, graphics::VegetationMask>& masks,
152 const VegetationSceneGpuBuild& build) {
153 if (expectedRevision != revision_ || expectedGpuRevision != runtime.sourceRevision())
154 return fail<VegetationSceneElementPublication>(DiagnosticCode::Conflict,
155 "vegetation live-element state changed before withdrawal");
156 if (!handle.isValid() || handle.owner() != owner_ || handle.index() >= slots_.size())
157 return fail<VegetationSceneElementPublication>(DiagnosticCode::NotFound,
158 "vegetation live Element handle is foreign or missing");
159 auto& slot = slots_[handle.index()];
160 if (!slot.element || slot.generation != handle.generation())
161 return fail<VegetationSceneElementPublication>(DiagnosticCode::NotFound,
162 "vegetation live Element handle is stale or withdrawn");
163 if (revision_ == std::numeric_limits<std::uint64_t>::max())
164 return fail<VegetationSceneElementPublication>(DiagnosticCode::InvalidArgument,
165 "vegetation live-element revision is exhausted");
166 auto candidate = snapshot();
167 if (!candidate) return Result<VegetationSceneElementPublication>::failure(candidate.status());
168 const std::size_t dynamicOffset = baseScene_.elements.size();
169 std::size_t activeBefore = 0;
170 for (std::size_t index = 0; index < handle.index(); ++index)
171 if (slots_[index].element) ++activeBefore;
172 candidate.value().elements.erase(candidate.value().elements.begin() + dynamicOffset + activeBefore);
173 auto published = runtime.replace(expectedGpuRevision, baseSurface, baseMotion, candidate.value(), masks, build);
174 if (!published) return Result<VegetationSceneElementPublication>::failure(published.status());
175 slot.element.reset();
176 const auto next = VegetationSceneElementHandle::LocalHandle::nextGeneration(slot.generation);
177 if (next)
178 slot.generation = *next;
179 else
180 slot.retired = true;
181 ++revision_;
182 return Result<VegetationSceneElementPublication>::success({handle, revision_, published.value()});
183}
