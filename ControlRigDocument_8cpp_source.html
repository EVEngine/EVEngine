115Result<editing::DomainOperation> ControlRigDocument::replaceGraph(const editing::GraphDocumentData& graph) const {
116 auto fail = [](std::string message) {
117 return Result<editing::DomainOperation>::failure(
118 Diagnostic::error(DiagnosticCode::InvalidArgument, std::move(message), "controlRig.graph"));
119 };
120 if (graph.domain != "animation.control-rig" || graph.schemaVersion != 1 || graph.revision != revision())
121 return fail("graph schema or revision mismatch");
122 auto encoded = animation::encodeControlRig(impl_->definition);
123 if (!encoded.ok()) return Result<editing::DomainOperation>::failure(encoded.status());
124 Value nodes = Value::array({});
125 std::map<std::string, size_t> ids;
126 editing::GraphDocument validation;
127 for (const auto& node : graph.nodes) {
128 if (node.type != "rig.solve" || !ids.emplace(node.id.value(), nodes.arraySize()).second)
129 return fail("invalid graph node");
130 if (node.pins.size() != 2 || !node.properties.isWithinLimits(16, 100000, 16000000))
131 return fail("invalid solve pins or properties");
132 bool input = false, output = false;
133 for (const auto& pin : node.pins) {
134 if (pin.node != node.id || pin.type != "rig.execution") return fail("invalid solve pin owner or type");
135 if (pin.id.value() == node.id.value() + ":in" && pin.direction == editing::GraphPinDirection::Input &&
136 !input)
137 input = true;
138 else if (pin.id.value() == node.id.value() + ":out" &&
139 pin.direction == editing::GraphPinDirection::Output && !output)
140 output = true;
141 else
142 return fail("invalid solve pin");
143 }
144 auto added = validation.createNode(node);
145 if (!added.ok()) return Result<editing::DomainOperation>::failure(added.status());
146 auto properties = Value::fromJson(editing::editorValueToJson(node.properties));
147 if (!properties.ok()) return Result<editing::DomainOperation>::failure(properties.status());
148 auto* id = properties.value().find("id");
149 if (!id || !id->isString() || id->asString() != node.id.value()) return fail("node identity mismatch");
150 properties.value().set("after", Value::array({}));
151 nodes.pushBack(std::move(properties).takeValue());
152 }
153 for (const auto& edge : graph.edges) {
154 auto connected = validation.connect(edge, {true, {}});
155 if (!connected.ok()) return Result<editing::DomainOperation>::failure(connected.status());
156 std::string from, to;
157 for (const auto& [id, index] : ids) {
158 (void)index;
159 if (edge.from.value() == id + ":out") from = id;
160 if (edge.to.value() == id + ":in") to = id;
161 }
162 if (from.empty() || to.empty()) return fail("edge must connect solve output to input");
163 nodes.at(ids.at(to)).find("after")->pushBack(from);
164 }
165 encoded.value().set("nodes", std::move(nodes));
166 auto candidate = animation::decodeControlRig(encoded.value());
167 if (!candidate.ok()) return Result<editing::DomainOperation>::failure(candidate.status());
168 return replace(candidate.value());
169}
