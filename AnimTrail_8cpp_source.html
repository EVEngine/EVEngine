107void AnimTrail::sampleBoneOffset(const AnimPose *pose, int boneIndex, float ox, float oy, float oz,
108 const std::string &plane) {
109 if (!pose) throw Exception("AnimTrail.sampleBone: pose is null");
110 if (boneIndex < 0 || boneIndex >= pose->getBoneCount()) {
111 throw Exception("AnimTrail.sampleBone: boneIndex %d out of range (count=%d)", boneIndex,
112 pose->getBoneCount());
113 }
114 const float wx = pose->getWorldPositionX(boneIndex) + ox;
115 const float wy = pose->getWorldPositionY(boneIndex) + oy;
116 const float wz = pose->getWorldPositionZ(boneIndex) + oz;
117 float px = 0.f, py = 0.f;
118 projectPlane(wx, wy, wz, plane, &px, &py);
119 pushPoint(px, py, plane == "xy" || plane.empty() ? wz : (plane == "xz" ? wy : wx));
120}

122void AnimTrail::pushPoint(float x, float y, float z) {
123 if (!points_.empty() && minDistance_ > 0.f) {
124 const Point &last = points_.back();
125 const float dx = x - last.x;
126 const float dy = y - last.y;
127 const float dz = z - last.z;
128 if (dx * dx + dy * dy + dz * dz < minDistance_ * minDistance_) return;
129 }
130 Point p;
131 p.x = x;
132 p.y = y;
133 p.z = z;
134 p.age = 0.f;
135 points_.push_back(p);
136 while (static_cast<int>(points_.size()) > capacity_) {
137 points_.erase(points_.begin());
138 }
139}

143void AnimTrail::updateUnchecked(float dt) {
144 if (dt < 0.f) dt = 0.f;
145 for (Point &p : points_) p.age += dt;
146 if (duration_ > 0.f) {
147 auto it = points_.begin();
148 while (it != points_.end()) {
149 if (it->age > duration_) {
150 it = points_.erase(it);
151 } else {
152 ++it;
153 }
154 }
155 }

158eve::Result<void> AnimTrail::advance(const eve::SimulationStep &step) {
159 auto seconds = detail::secondsForStep(step, hasLastTick_, lastTick_, "AnimTrail");
160 if (!seconds) return eve::Result<void>::failure(seconds.status());
161 updateUnchecked(std::move(seconds).takeValue());
162 lastTick_ = step.tick;

228void AnimTrail::drawSegment(graphics::Graphics *gfx, float x0, float y0, float x1, float y1,
229 float a0, float a1) const {
230 if (!gfx) return;
231 float sx0 = 0.f, sy0 = 0.f, sx1 = 0.f, sy1 = 0.f;
232 toScreen(x0, y0, &sx0, &sy0);
233 toScreen(x1, y1, &sx1, &sy1);
234 const float dx = sx1 - sx0;
235 const float dy = sy1 - sy0;
236 const float len = std::sqrt(dx * dx + dy * dy);
237 if (len < 1e-4f) {
238 drawDot(gfx, x1, y1, a1);
239 return;
240 }
241 const float step = std::max(1.f, width_ * 0.75f);
242 const int steps = std::max(1, int(len / step));
243 const float half = width_ * 0.5f;
244 for (int i = 0; i <= steps; ++i) {
245 const float t = float(i) / float(steps);
246 const float a = a0 + (a1 - a0) * t;
247 if (a <= 0.f) continue;
248 const float px = sx0 + dx * t;
249 const float py = sy0 + dy * t;
250 gfx->drawSolidRect(px - half, py - half, width_, width_,
251 Color(colorR_, colorG_, colorB_, a));
252 }
253}

255void AnimTrail::draw(graphics::Graphics *gfx) {
256 if (!gfx || points_.empty()) return;
257
258 if (style_ == Style::Points) {
259 for (const Point &p : points_) {
260 drawDot(gfx, p.x, p.y, alphaFor(p));
261 }
262 return;
263 }
264
265 // Line: connect oldest → newest with per-vertex fade.
266 for (size_t i = 1; i < points_.size(); ++i) {
267 const Point &a = points_[i - 1];
268 const Point &b = points_[i];
269 drawSegment(gfx, a.x, a.y, b.x, b.y, alphaFor(a), alphaFor(b));
270 }
271 // Emphasize the head.
272 if (!points_.empty()) {
273 const Point &head = points_.back();
274 drawDot(gfx, head.x, head.y, alphaFor(head));
275 }
276}
