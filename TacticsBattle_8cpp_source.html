46void record(Battle& battle, BattleCommand command) {
47 auto commands = battle.commands();
48 command.sequence = commands->nextSequence++;
49 command.resultingRevision = battle.turn()->revision;
50 std::uint64_t correlation = command.sequence;
51 if (command.triggerSequence != 0) {
52 const auto trigger = std::find_if(battle.events()->values.begin(), battle.events()->values.end(),
53 [&](const auto& event) { return event.sequence == command.triggerSequence; });
54 if (trigger != battle.events()->values.end())
55 correlation = trigger->correlationCommand != 0 ? trigger->correlationCommand
56 : trigger->causationCommand;
57 if (correlation == 0) correlation = command.sequence;
58 }
59 for (auto& event : battle.events()->values) {
60 if (event.causationCommand != 0) continue;
61 event.causationCommand = command.sequence;
62 event.correlationCommand = correlation;
63 }
64 commands->values.push_back(std::move(command));
65}

72bool objectiveSatisfied(Battle& battle, const ObjectiveSpec& objective) {
73 switch (objective.kind) {
74 case ObjectiveKind::EliminateSide:
75 for (const auto& handle : battle.turn()->units) {
76 auto* unit = resolve<TacticalUnit>(handle);
77 if (unit && unit->turn()->alive && sideSubject(unit->membership()->side) == objective.targetSide)
78 return false;
79 }
80 return true;
81 case ObjectiveKind::SurviveRounds: return battle.turn()->round >= objective.requiredRound;
82 case ObjectiveKind::OccupyCells:
83 for (const Cell cell : objective.requiredCells) {
84 const auto occupant = battle.board()->value.occupant(cell);
85 if (!occupant) return false;
86 TacticalUnit* occupyingUnit = nullptr;
87 for (const auto& handle : battle.turn()->units) {
88 auto* unit = resolve<TacticalUnit>(handle);
89 if (unit && unit->identity()->subject == *occupant) {
90 occupyingUnit = unit;
91 break;
92 }
93 }
94 if (!occupyingUnit || !occupyingUnit->turn()->alive ||
95 sideSubject(occupyingUnit->membership()->side) != objective.beneficiarySide)
96 return false;
97 }
98 return true;
99 }
100 return false;
101}

158Result<void> validateChargeProgress(Battle& battle, ChargeModel model) {
159 const Status usable = Status::success(StatusCode::NoOp);
160 if (!model.isChargeBased()) return Result<void>::success(usable);
161 for (const auto& handle : battle.turn()->units) {
162 TacticalUnit* unit = resolve<TacticalUnit>(handle);
163 if (unit == nullptr || !unit->turn()->alive) continue;
164 const std::int64_t gain = static_cast<std::int64_t>(model.initiativeGain) *
165 static_cast<std::int64_t>(unit->turn()->initiative);
166 if (gain > 0) return Result<void>::success(usable);
167 }
168 return Result<void>::failure(
169 Diagnostic::error(DiagnosticCode::PreconditionViolation,
170 "charge-time turn policy requires a living unit that can gain charge", "battle.units"));
171}

180void applyChargeGain(Battle& battle, ChargeModel model) {
181 if (model.initiativeGain == 0) return;
182 constexpr int kMaxCharge = std::numeric_limits<int>::max();
183 for (const auto& handle : battle.turn()->units) {
184 TacticalUnit* unit = resolve<TacticalUnit>(handle);
185 if (unit == nullptr || !unit->turn()->alive) continue;
186 const std::int64_t gain = static_cast<std::int64_t>(model.initiativeGain) *
187 static_cast<std::int64_t>(unit->turn()->initiative);
188 const std::int64_t next = static_cast<std::int64_t>(unit->turn()->charge) + gain;
189 unit->turn()->charge = next < 0 ? 0 : (next > kMaxCharge ? kMaxCharge : static_cast<int>(next));
190 }
191}

302Result<void> BattleSystem::start(Battle& battle, std::string policyId) {
303 auto turn = battle.turn();
304 const Revision expectedRevision = turn->revision;
305 if (turn->status != BattleStatus::Setup)
306 return failure(DiagnosticCode::Conflict, "tactics battle has already started", "battle.status");
307 if (turn->sides.empty() || turn->units.empty())
308 return failure(DiagnosticCode::PreconditionViolation, "tactics battle requires sides and units",
309 "battle.setup");
310 // Resolve the policy before mutating anything: an unregistered id must be a
311 // refusal that leaves the battle in Setup, not a battle that starts and then
312 // cannot schedule.
313 auto policy = TurnPolicyRegistry::builtins().find(policyId);
314 if (!policy) return Result<void>::failure(policy.status());
315 // A charge rule that can never make progress is refused here, before any
316 // mutation, so an unschedulable battle stays in Setup instead of advancing
317 // forever without an activation.
318 auto chargeUsable = validateChargeProgress(battle, policy.value()->chargeModel(battle));
319 if (!chargeUsable) return Result<void>::failure(chargeUsable.status());
320 auto boardValid = battle.board()->value.validateInvariants();
321 if (!boardValid) return Result<void>::failure(boardValid.status());
322 auto units = orderedUnits(battle);
323 if (!units) return Result<void>::failure(units.status());
324 auto revision = nextRevision(battle);
325 if (!revision) return Result<void>::failure(revision.status());
326
327 turn->policyId = std::move(policyId);
328 turn->status = BattleStatus::Running;
329 turn->phase = BattlePhase::BattleStart;
330 turn->tick = SimulationTick::zero();
331 turn->round = 0;
332 turn->cursor = 0;
333 turn->revision = std::move(revision).takeValue();
334 emit(battle, BattlePhase::Setup, BattlePhase::BattleStart, turn->tick, "battle.started",
335 battle.identity()->subject);
336 BattleCommand command;
337 command.kind = BattleCommandKind::Start;
338 command.expectedRevision = expectedRevision;
339 command.policyId = turn->policyId;
340 record(battle, std::move(command));
341 return Result<void>::success(Status::success(StatusCode::Applied));
342}

351Result<AbilityReceipt> validateAbility(Battle& battle, SubjectRef actor, const LogicalId& action, Cell targetCell,
352 SubjectRef targetUnit, std::string_view payload) {
353 auto turn = battle.turn();
354 if (turn->status != BattleStatus::Running || turn->phase != BattlePhase::Acting || !turn->activeUnit)
355 return Result<AbilityReceipt>::failure(Diagnostic::error(
356 DiagnosticCode::PreconditionViolation, "tactics battle is not accepting an ability", "battle.phase"));
357 if (!action.isValid())
358 return Result<AbilityReceipt>::failure(
359 Diagnostic::error(DiagnosticCode::InvalidArgument, "tactics ability requires a valid action id", "action"));
360 TacticalUnit* unit = resolve<TacticalUnit>(*turn->activeUnit);
361 if (unit == nullptr)
362 return Result<AbilityReceipt>::failure(
363 Diagnostic::error(DiagnosticCode::StaleHandle, "tactics active unit is stale", "battle.activeUnit"));
364 if (unit->identity()->subject != actor)
365 return Result<AbilityReceipt>::failure(Diagnostic::error(
366 DiagnosticCode::PreconditionViolation, "tactics actor does not own the active turn", "actor"));
367 // The action economy is enforced here: an activation costs one action point,
368 // and a unit with none left cannot act.
369 if (unit->turn()->actionPoints <= 0)
370 return Result<AbilityReceipt>::failure(Diagnostic::error(
371 DiagnosticCode::PreconditionViolation, "tactics unit has no action points left", "unit.actionPoints"));
372 if (targetCell.layer != unit->position()->cell.layer)
373 return Result<AbilityReceipt>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
374 "tactics ability target must share the actor's layer",
375 "targetCell.layer"));
376 auto targetState = battle.board()->value.cell(targetCell);
377 if (!targetState) return Result<AbilityReceipt>::failure(targetState.status());
378 if (targetUnit.isValid()) {
379 // A named target must be a living unit of this battle; a stale or foreign subject
380 // is a refusal, never a silently dropped target.
381 TacticalUnit* targeted = findUnit(battle, targetUnit);
382 if (targeted == nullptr)
383 return Result<AbilityReceipt>::failure(Diagnostic::error(
384 DiagnosticCode::NotFound, "tactics ability target is not a unit of this battle", "targetUnit"));
385 if (!targeted->turn()->alive)
386 return Result<AbilityReceipt>::failure(Diagnostic::error(
387 DiagnosticCode::PreconditionViolation, "tactics ability target is already defeated", "targetUnit"));
388 }
389 if (payload.size() > kMaxAbilityPayloadBytes)
390 return Result<AbilityReceipt>::failure(Diagnostic::error(
391 DiagnosticCode::InvalidArgument, "tactics ability payload exceeds the persisted size bound", "payload"));
392
393 AbilityReceipt receipt;
394 receipt.actor = actor;
395 receipt.action = action;
396 receipt.targetUnit = targetUnit;
397 receipt.targetCell = targetCell;
398 receipt.remainingActionPoints = unit->turn()->actionPoints - 1;
399 return Result<AbilityReceipt>::success(std::move(receipt));
400}

439Result<BattlePhase> BattleSystem::advance(Battle& battle, const SimulationStep& step) {
440 auto turn = battle.turn();
441 if (turn->status != BattleStatus::Running)
442 return Result<BattlePhase>::failure(
443 Diagnostic::error(DiagnosticCode::PreconditionViolation, "tactics battle is not running", "battle.status"));
444 if (step.delta.nanoseconds() < 0 || step.tick <= turn->tick)
445 return Result<BattlePhase>::failure(Diagnostic::error(
446 DiagnosticCode::InvalidArgument, "tactics step requires a non-negative delta and increasing tick", "step"));
447 auto revision = nextRevision(battle);
448 if (!revision) return Result<BattlePhase>::failure(revision.status());
449 const Revision committedRevision = std::move(revision).takeValue();
450 BattleCommand command;
451 command.kind = BattleCommandKind::Advance;
452 command.expectedRevision = turn->revision;
453 command.step = step;
454
455 switch (turn->phase) {
456 case BattlePhase::BattleStart:
457 return transition(battle, BattlePhase::RoundStart, step.tick, "round.pending", committedRevision,
458 std::move(command));
459 case BattlePhase::RoundStart: {
460 auto policy = schedulingPolicy(battle);
461 if (!policy) return Result<BattlePhase>::failure(policy.status());
462 const ChargeModel charge = policy.value()->chargeModel(battle);
463 applyChargeGain(battle, charge);
464 auto ordered = orderedUnits(battle);
465 if (!ordered) return Result<BattlePhase>::failure(ordered.status());
466 turn->units = std::move(ordered).takeValue();
467 turn->cursor = 0;
468 ++turn->round;
469 for (const auto& handle : turn->units) {
470 TacticalUnit* unit = resolve<TacticalUnit>(handle);
471 if (unit != nullptr) {
472 unit->turn()->actionPoints = unit->turn()->roundActionPoints;
473 unit->turn()->movePoints = unit->turn()->roundMovePoints;
474 unit->turn()->reactionPoints = unit->turn()->roundReactionPoints;
475 unit->turn()->acted = false;
476 }
477 }
478 if (turn->units.empty())
479 return transition(battle, BattlePhase::BattleEnd, step.tick, "battle.empty", committedRevision,
480 std::move(command));
481 // A charge rule may leave units out of this round's queue. That is a real
482 // scheduling outcome, so it is reported as an unchanged phase with NoOp
483 // rather than as a new phase only one policy can reach: the caller can
484 // tell "time passed, nobody acted" apart from an activation, and the
485 // round counter and resource refresh above still describe a real round.
486 turn->schedule = readySchedule(battle, charge, turn->units);
487 if (turn->schedule.empty()) {
488 auto progress = validateChargeProgress(battle, charge);
489 if (!progress) return Result<BattlePhase>::failure(progress.status());
490 turn->tick = step.tick;
491 turn->revision = committedRevision;
492 record(battle, std::move(command));
493 return Result<BattlePhase>::success(BattlePhase::RoundStart, Status::success(StatusCode::NoOp));
494 }
495 consumeActivationCharge(battle, charge, turn->schedule);
496 TacticalUnit* unit = resolve<TacticalUnit>(turn->schedule.front());
497 if (unit == nullptr)
498 return Result<BattlePhase>::failure(Diagnostic::error(
499 DiagnosticCode::StaleHandle, "tactics schedule contains a stale unit", "battle.schedule"));
500 turn->activeUnit = turn->schedule.front();
501 turn->activeSide = unit->membership()->side;
502 return transition(battle, BattlePhase::TurnStart, step.tick, "turn.pending", committedRevision,
503 std::move(command),
504 unit->identity()->subject);
505 }
506 case BattlePhase::TurnStart: {
507 TacticalUnit* unit = turn->activeUnit ? resolve<TacticalUnit>(*turn->activeUnit) : nullptr;
508 if (unit == nullptr)
509 return Result<BattlePhase>::failure(Diagnostic::error(
510 DiagnosticCode::StaleHandle, "tactics active unit is stale", "battle.activeUnit"));
511 return transition(battle, BattlePhase::Acting, step.tick, "turn.started", committedRevision,
512 std::move(command),
513 unit->identity()->subject);
514 }
515 case BattlePhase::TurnEnd: {
516 // The round's queue is fixed when the round starts, so the remaining
517 // order cannot drift. A unit that died before its activation is skipped
518 // rather than shifting the cursor onto somebody else's turn.
519 do {
520 ++turn->cursor;
521 } while (turn->cursor < turn->schedule.size() && !scheduleEntryAlive(turn->schedule[turn->cursor]));
522 if (turn->cursor >= turn->schedule.size()) {
523 turn->activeUnit.reset();
524 turn->activeSide.reset();
525 return transition(battle, BattlePhase::RoundEnd, step.tick, "round.pending", committedRevision,
526 std::move(command));
527 }
528 TacticalUnit* unit = resolve<TacticalUnit>(turn->schedule[turn->cursor]);
529 if (unit == nullptr)
530 return Result<BattlePhase>::failure(Diagnostic::error(
531 DiagnosticCode::StaleHandle, "tactics schedule contains a stale unit", "battle.schedule"));
532 turn->activeUnit = turn->schedule[turn->cursor];
533 turn->activeSide = unit->membership()->side;
534 return transition(battle, BattlePhase::TurnStart, step.tick, "turn.pending", committedRevision,
535 std::move(command),
536 unit->identity()->subject);
537 }
538 case BattlePhase::RoundEnd:
539 turn->tick = step.tick;
540 if (evaluateObjectives(battle, committedRevision)) {
541 turn->status = BattleStatus::Ended;
542 return transition(battle, BattlePhase::BattleEnd, step.tick, "battle.objective_completed",
543 committedRevision, std::move(command));
544 }
545 return transition(battle, BattlePhase::RoundStart, step.tick, "round.completed", committedRevision,
546 std::move(command));
547 case BattlePhase::BattleEnd:
548 turn->status = BattleStatus::Ended;
549 turn->tick = step.tick;
550 turn->revision = committedRevision;
551 record(battle, std::move(command));
552 return Result<BattlePhase>::success(BattlePhase::BattleEnd, Status::success(StatusCode::NoOp));
553 case BattlePhase::Acting:
554 case BattlePhase::Reaction:
555 return Result<BattlePhase>::failure(
556 Diagnostic::error(DiagnosticCode::PreconditionViolation,
557 "tactics phase requires an explicit command before advancing", "battle.phase"));
558 case BattlePhase::Setup:
559 return Result<BattlePhase>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
560 "running tactics battle cannot remain in setup phase",
561 "battle.phase"));
562 }
563 return Result<BattlePhase>::failure(
564 Diagnostic::error(DiagnosticCode::InvariantViolation, "tactics battle phase is invalid", "battle.phase"));
565}

775Result<std::size_t> BattleSystem::openReaction(Battle& battle, std::uint64_t triggerSequence,
776 std::vector<ReactionCandidate> candidates) {
777 auto turn = battle.turn();
778 const Revision expectedRevision = turn->revision;
779 auto reactions = battle.reactions();
780 if (turn->status != BattleStatus::Running ||
781 (turn->phase != BattlePhase::Acting && turn->phase != BattlePhase::Reaction))
782 return Result<std::size_t>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
783 "tactics reactions require an acting or reaction phase",
784 "battle.phase"));
785 if (triggerSequence == 0 || candidates.empty())
786 return Result<std::size_t>::failure(Diagnostic::error(
787 DiagnosticCode::InvalidArgument, "tactics reaction window requires a trigger and candidates", "reaction"));
788 const auto triggerEvent = std::find_if(battle.events()->values.begin(), battle.events()->values.end(),
789 [&](const auto& event) { return event.sequence == triggerSequence; });
790 if (triggerEvent == battle.events()->values.end())
791 return Result<std::size_t>::failure(Diagnostic::error(
792 DiagnosticCode::NotFound, "tactics reaction trigger event does not exist", "reaction.triggerSequence"));
793 const std::size_t depth = reactions->stack.size() + 1;
794 if (depth > reactions->maxDepth) {
795 return Result<std::size_t>::failure(Diagnostic::error(
796 DiagnosticCode::PreconditionViolation, "tactics reaction stack depth exceeded", "reaction.depth"));
797 }
798 for (const auto& candidate : candidates) {
799 if (!candidate.reactor.isValid() || !candidate.action.isValid())
800 return Result<std::size_t>::failure(
801 Diagnostic::error(DiagnosticCode::InvalidArgument,
802 "tactics reaction candidate requires valid identities", "reaction.candidate"));
803 }
804 std::sort(candidates.begin(), candidates.end(), [](const auto& left, const auto& right) {
805 if (left.priority != right.priority) return left.priority > right.priority;
806 if (left.initiative != right.initiative) return left.initiative > right.initiative;
807 if (left.reactor.format() != right.reactor.format()) return left.reactor.format() < right.reactor.format();
808 return left.action.format() < right.action.format();
809 });
810 candidates.erase(std::unique(candidates.begin(), candidates.end(), [](const auto& left, const auto& right) {
811 return left.reactor == right.reactor && left.action == right.action;
812 }),
813 candidates.end());
814 auto revision = nextRevision(battle);
815 if (!revision) return Result<std::size_t>::failure(revision.status());
816 const auto recordedCandidates = candidates;
817 reactions->stack.push_back({triggerSequence, depth, std::move(candidates)});
818 const auto from = turn->phase;
819 turn->phase = BattlePhase::Reaction;
820 turn->revision = std::move(revision).takeValue();
821 emit(battle, from, BattlePhase::Reaction, turn->tick, "reaction.opened");
822 BattleCommand command;
823 command.kind = BattleCommandKind::OpenReaction;
824 command.expectedRevision = expectedRevision;
825 command.triggerSequence = triggerSequence;
826 command.candidates = recordedCandidates;
827 record(battle, std::move(command));
828 return Result<std::size_t>::success(depth, Status::success(StatusCode::Applied));
829}

831Result<ReactionReceipt> BattleSystem::acceptReaction(Battle& battle, SubjectRef reactor, const LogicalId& action) {
832 auto turn = battle.turn();
833 const Revision expectedRevision = turn->revision;
834 auto reactions = battle.reactions();
835 if (turn->status != BattleStatus::Running || turn->phase != BattlePhase::Reaction || reactions->stack.empty())
836 return Result<ReactionReceipt>::failure(Diagnostic::error(
837 DiagnosticCode::PreconditionViolation, "tactics battle has no open reaction window", "reaction"));
838 ReactionWindow& window = reactions->stack.back();
839 const auto candidate = std::find_if(window.candidates.begin(), window.candidates.end(), [&](const auto& value) {
840 return value.reactor == reactor && value.action == action;
841 });
842 if (candidate == window.candidates.end())
843 return Result<ReactionReceipt>::failure(Diagnostic::error(
844 DiagnosticCode::NotFound, "tactics reaction is not eligible in the top window", "reaction.candidate"));
845 const std::string key = std::to_string(window.triggerSequence) + ":" + reactor.format() + ":" + action.format();
846 if (std::find(reactions->seen.begin(), reactions->seen.end(), key) != reactions->seen.end())
847 return Result<ReactionReceipt>::failure(Diagnostic::error(
848 DiagnosticCode::Conflict, "tactics reaction already fired for this trigger chain", "reaction.cycle"));
849
850 TacticalUnit* reactingUnit = nullptr;
851 for (const auto& handle : turn->units) {
852 TacticalUnit* unit = resolve<TacticalUnit>(handle);
853 if (unit != nullptr && unit->identity()->subject == reactor) {
854 reactingUnit = unit;
855 break;
856 }
857 }
858 if (reactingUnit == nullptr)
859 return Result<ReactionReceipt>::failure(Diagnostic::error(
860 DiagnosticCode::NotFound, "tactics reacting unit is not part of the battle", "reaction.reactor"));
861 if (!reactingUnit->turn()->alive || reactingUnit->turn()->reactionPoints <= 0)
862 return Result<ReactionReceipt>::failure(
863 Diagnostic::error(DiagnosticCode::PreconditionViolation,
864 "tactics reacting unit has no available reaction point", "reaction.resource"));
865 auto revision = nextRevision(battle);
866 if (!revision) return Result<ReactionReceipt>::failure(revision.status());
867
868 --reactingUnit->turn()->reactionPoints;
869 reactions->seen.push_back(key);
870 const std::uint64_t trigger = window.triggerSequence;
871 reactions->stack.pop_back();
872 turn->phase = reactions->stack.empty() ? BattlePhase::Acting : BattlePhase::Reaction;
873 turn->revision = std::move(revision).takeValue();
874 emit(battle, BattlePhase::Reaction, turn->phase, turn->tick, "reaction.accepted", reactor);
875 BattleCommand command;
876 command.kind = BattleCommandKind::AcceptReaction;
877 command.expectedRevision = expectedRevision;
878 command.actor = reactor;
879 command.action = action;
880 command.triggerSequence = trigger;
881 record(battle, std::move(command));
882 return Result<ReactionReceipt>::success(
883 {trigger, reactor, action, reactingUnit->turn()->reactionPoints}, Status::success(StatusCode::Applied));
884}

1013Result<std::vector<ecs::EntityHandle>> BattleSystem::orderedUnits(Battle& battle) {
1014 // Resolve the scheduling policy once: an unregistered id cannot be scheduled.
1015 auto policy = TurnPolicyRegistry::builtins().find(battle.turn()->policyId);
1016 if (!policy) return Result<std::vector<ecs::EntityHandle>>::failure(policy.status());
1017
1018 const auto sideIndexOf = [&](const ecs::EntityHandle& side) {
1019 const auto& sides = battle.turn()->sides;
1020 const auto found =
1021 std::find_if(sides.begin(), sides.end(), [&](const auto& value) { return sameHandle(value, side); });
1022 return static_cast<std::size_t>(std::distance(sides.begin(), found));
1023 };
1024
1025 // Project once, sort the projection, then map back, so the comparator never
1026 // resolves a unit and the policy sees values only.
1027 struct Entry {
1028 ecs::EntityHandle handle;
1029 UnitOrder order;
1030 std::string canonical;
1031 };
1032 std::vector<Entry> entries;
1033 std::vector<ecs::EntityHandle> result;
1034 for (const auto& handle : battle.turn()->units) {
1035 TacticalUnit* unit = resolve<TacticalUnit>(handle);
1036 if (unit == nullptr)
1037 return Result<std::vector<ecs::EntityHandle>>::failure(
1038 Diagnostic::error(DiagnosticCode::StaleHandle, "tactics battle contains a stale unit", "battle.units"));
1039 if (!unit->turn()->alive) continue;
1040 result.push_back(handle);
1041 entries.push_back({handle,
1042 UnitOrder{unit->turn()->initiative, sideIndexOf(unit->membership()->side),
1043 unit->turn()->charge},
1044 unit->identity()->subject.format()});
1045 }
1046 std::sort(entries.begin(), entries.end(), [&](const Entry& left, const Entry& right) {
1047 const ITurnPolicy* resolved = policy.value();
1048 switch (resolved->order(battle, left.order, right.order)) {
1049 case TurnOrder::LeftFirst: return true;
1050 case TurnOrder::RightFirst: return false;
