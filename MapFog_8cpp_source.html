19float hashWrap(int x, int y, int period, uint32_t seed) {
20 x = ((x % period) + period) % period;
21 y = ((y % period) + period) % period;
22 uint32_t n = uint32_t(x) * 374761393u + uint32_t(y) * 668265263u + seed;
23 n = (n ^ (n >> 13u)) * 1274126177u;
24 return float(n & 0x00ffffffu) / float(0x00ffffffu);
25}

27float valueNoiseWrap(float x, float y, int period, uint32_t seed) {
28 const int x0 = int(std::floor(x));
29 const int y0 = int(std::floor(y));
30 const float fx = x - float(x0);
31 const float fy = y - float(y0);
32 // Quintic fade — softer ridges than the classic cubic hermite.
33 const float ux = fx * fx * fx * (fx * (fx * 6.f - 15.f) + 10.f);
34 const float uy = fy * fy * fy * (fy * (fy * 6.f - 15.f) + 10.f);
35 const float n00 = hashWrap(x0, y0, period, seed);
36 const float n10 = hashWrap(x0 + 1, y0, period, seed);
37 const float n01 = hashWrap(x0, y0 + 1, period, seed);
38 const float n11 = hashWrap(x0 + 1, y0 + 1, period, seed);
39 const float nx0 = n00 + (n10 - n00) * ux;
40 const float nx1 = n01 + (n11 - n01) * ux;
41 return nx0 + (nx1 - nx0) * uy;
42}

44// True tileable fBm: each octave period is an integer multiple of the base so
45// u=0 and u=1 hit identical lattice corners (no fmod phase-shift seams).
46float fbmSeamless(float u, float v, int basePeriod, uint32_t seed, int octaves) {
47 float sum = 0.f;
48 float amp = 0.55f;
49 float norm = 0.f;
50 int period = std::max(basePeriod, 2);
51 for (int i = 0; i < octaves; ++i) {
52 sum += amp * valueNoiseWrap(u * float(period), v * float(period), period,
53 seed + uint32_t(i) * 97u);
54 norm += amp;
55 period *= 2;
56 amp *= 0.48f;
57 }
58 return (norm > 1e-6f) ? (sum / norm) : 0.f;
59}

61float billow(float n) { return 1.f - std::fabs(n * 2.f - 1.f); }
62
63float ridged(float n) {
64 const float r = 1.f - std::fabs(n * 2.f - 1.f);
65 return r * r;
66}

264Texture *MapFog::makeCloudTexture(int size) {
265 const int n = std::clamp(size, 16, 512);
266 std::vector<uint8_t> rgba(size_t(n * n * 4));
267 // Integer-period wrap fBm is seamless by construction. Prefer LOW base
268 // periods so one tile reads as a few large soft billows (article look),
269 // not wallpaper static. Do NOT scale UV by non-integers here.
270 for (int y = 0; y < n; ++y) {
271 for (int x = 0; x < n; ++x) {
272 const float u = float(x) / float(n);
273 const float v = float(y) / float(n);
274
275 // Period 1/2/3 => a few huge soft masses per tile (not wallpaper micro-noise).
276 const float large = fbmSeamless(u, v, 1, 0xA11CE001u, 5);
277 const float mid = fbmSeamless(u, v, 2, 0xBEEF42u, 4);
278 const float fine = fbmSeamless(u, v, 3, 0xC0FFEEu, 3);
279
280 // Soft puffy volumes: billow dominates, fine ridge only for breakup.
281 const float soft = billow(large) * 0.68f + billow(mid) * 0.24f + ridged(fine) * 0.08f;
282 // Lift into a pale cloud range so dual-scroll multiply stays readable.
283 const float c = std::clamp(0.38f + soft * 0.55f, 0.f, 1.f);
284
285 const float cool = std::clamp(c * 0.96f + 0.03f, 0.f, 1.f);
286 const float warm = std::clamp(c * 1.02f - 0.01f, 0.f, 1.f);
287 const size_t i = size_t((y * n + x) * 4);
288 rgba[i + 0] = uint8_t(warm * 255.f + 0.5f);
289 rgba[i + 1] = uint8_t(c * 255.f + 0.5f);
290 rgba[i + 2] = uint8_t(cool * 255.f + 0.5f);
291 rgba[i + 3] = 255;
292 }
293 }
294 return graphics_->newTexture(n, n, rgba.data(), true, true);
295}
