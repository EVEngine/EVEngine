44Result<float> numberField(const Value::Object& object, std::string_view name, float fallback) {
45 const auto* value = field(object, name);
46 if (!value) return Result<float>::success(fallback);
47 if (!value->isNumeric())
48 return failure<float>(DiagnosticCode::ParseError, "terrain number is invalid", std::string(name));
49 const double number = value->isInt64() ? double(value->asInt()) : value->asDouble();
50 if (!std::isfinite(number) || std::abs(number) > std::numeric_limits<float>::max())
51 return failure<float>(DiagnosticCode::InvalidArgument, "terrain number is outside float range",
52 std::string(name));
53 return Result<float>::success(float(number));
54}

56template <std::size_t N>
57Result<std::array<float, N>> vectorField(const Value::Object& object, std::string_view name,
58 std::array<float, N> fallback) {
59 const auto* value = field(object, name);
60 if (!value) return Result<std::array<float, N>>::success(fallback);
61 const auto* array = value->getIf<Value::Array>();
62 if (!array || array->size() != N)
63 return failure<std::array<float, N>>(DiagnosticCode::ParseError, "terrain vector is invalid",
64 std::string(name));
65 for (std::size_t i = 0; i < N; ++i) {
66 Value::Object scalar{{"value", (*array)[i]}};
67 auto parsed = numberField(scalar, "value", 0);
68 if (!parsed) return Result<std::array<float, N>>::failure(parsed.status());
69 fallback[i] = parsed.value();
70 }
71 return Result<std::array<float, N>>::success(fallback);
72}

76Result<LoadedTerrainMaterial> EvpackTerrainMaterialLoader::load(const AssetRef& material,
77 const asset::EvpackCapabilities& capabilities,
78 const TerrainMaterialLoadLimits& limits) const {
79 auto payload = reader_.read(material, "eve.terrain-material/3", capabilities, limits.maximumDecodedBytes);
80 if (!payload && payload.error()->code() == DiagnosticCode::TypeMismatch)
81 payload = reader_.read(material, "eve.terrain-material/2", capabilities, limits.maximumDecodedBytes);
82 if (!payload) return Result<LoadedTerrainMaterial>::failure(payload.status());
83 const asset::RuntimeAssetChunk* definition = nullptr;
84 for (const auto& chunk : payload.value().chunks) {
85 if (chunk.kind != asset::EvpackChunkKind::Definition)
86 return failure<LoadedTerrainMaterial>(DiagnosticCode::TypeMismatch,
87 "terrain material cannot contain runtime bulk");
88 if (definition)
89 return failure<LoadedTerrainMaterial>(DiagnosticCode::Conflict,
90 "terrain material has duplicate definitions");
91 definition = &chunk;
92 }
93 if (!definition)
94 return failure<LoadedTerrainMaterial>(DiagnosticCode::NotFound, "terrain material definition is missing");
95 asset::RuntimeDefinitionLimits definitionLimits;
96 definitionLimits.maximumBytes = limits.maximumDecodedBytes;
97 definitionLimits.maximumStringBytes = limits.maximumStringBytes;
98 auto metadata = asset::decodeRuntimeDefinition(definition->bytes, definitionLimits);
99 if (!metadata) return Result<LoadedTerrainMaterial>::failure(metadata.status());
100 const auto* root = metadata.value().getIf<Value::Object>();
101 const Value* schema = root ? field(*root, "schema") : nullptr;
102 const Value* version = root ? field(*root, "schemaVersion") : nullptr;
103 const Value* layersValue = root ? field(*root, "layers") : nullptr;
104 const auto* layers = layersValue ? layersValue->getIf<Value::Array>() : nullptr;
105 if (!schema || !schema->isString() || schema->asString() != "eve.terrain-material" || !version ||
106 !version->isInt64() || (version->asInt() != 2 && version->asInt() != 3) || !layers || layers->empty() ||
107 layers->size() > limits.maximumLayers)
108 return failure<LoadedTerrainMaterial>(DiagnosticCode::ParseError, "terrain material envelope is invalid");
109 std::vector<RuntimeTerrainLayer> decoded;
110 decoded.reserve(layers->size());
111 for (std::size_t index = 0; index < layers->size(); ++index) {
112 const auto* layer = (*layers)[index].getIf<Value::Object>();
113 if (!layer)
114 return failure<LoadedTerrainMaterial>(DiagnosticCode::ParseError, "terrain layer must be an object",
115 std::to_string(index));
116 auto name = textField(*layer, "name", limits, false);
117 auto diffuse = textField(*layer, "diffuseSource", limits, true);
118 auto normal = textField(*layer, "normalSource", limits, true);
119 auto weight = textField(*layer, "weightSource", limits, true);
120 auto convention = textField(*layer, "normalConvention", limits, false);
121 const Value* tile = field(*layer, "tileSizeMeters");
122 if (!name || !diffuse || !normal || !weight || !convention || !tile || (!tile->isDouble() && !tile->isInt64()))
123 return failure<LoadedTerrainMaterial>(DiagnosticCode::ParseError, "terrain layer fields are incomplete",
124 std::to_string(index));
125 const double tileSize = tile->isDouble() ? tile->asDouble() : double(tile->asInt());
126 if (!std::isfinite(tileSize) || tileSize <= 0 || tileSize > 1'000'000.0 ||
127 (convention.value() != "opengl" && convention.value() != "directx"))
128 return failure<LoadedTerrainMaterial>(
129 DiagnosticCode::InvalidArgument, "terrain layer physical semantics are invalid", std::to_string(index));
130 RuntimeTerrainLayer runtime;
131 runtime.name = std::move(name).takeValue();
132 runtime.diffuseSource = std::move(diffuse).takeValue();
133 runtime.normalSource = std::move(normal).takeValue();
134 runtime.weightSource = std::move(weight).takeValue();
135 runtime.normalConvention = std::move(convention).takeValue();
136 runtime.tileSizeMeters = static_cast<float>(tileSize);
137 if (version->asInt() >= 2) {
138 auto mask = textField(*layer, "maskSource", limits, true);
139 auto tileScale = vectorField<2>(*layer, "tileScaleMeters", {1, 1});
140 auto tileOffset = vectorField<2>(*layer, "tileOffsetMeters", {});
141 auto remapMin = vectorField<4>(*layer, "maskRemapMinimum", {});
142 auto remapMax = vectorField<4>(*layer, "maskRemapMaximum", {1, 1, 1, 1});
143 auto specular = vectorField<4>(*layer, "specular", {});
144 auto metallic = numberField(*layer, "metallic", 0);
145 auto normalScale = numberField(*layer, "normalScale", 1);
146 auto smoothness = numberField(*layer, "smoothness", 0);
147 if (!mask || !tileScale || !tileOffset || !remapMin || !remapMax || !specular || !metallic ||
148 !normalScale || !smoothness)
149 return failure<LoadedTerrainMaterial>(DiagnosticCode::ParseError,
150 "terrain layer v2 fields are incomplete", std::to_string(index));
151 if (tileScale.value()[0] <= 0 || tileScale.value()[1] <= 0 || metallic.value() < 0 ||
152 metallic.value() > 1 || normalScale.value() < -8 || normalScale.value() > 8 || smoothness.value() < 0 ||
153 smoothness.value() > 1)
154 return failure<LoadedTerrainMaterial>(DiagnosticCode::InvalidArgument,
155 "terrain layer v2 physical semantics are invalid",
156 std::to_string(index));
157 runtime.maskSource = std::move(mask).takeValue();
158 runtime.tileScaleMeters = std::move(tileScale).takeValue();
159 runtime.tileOffsetMeters = std::move(tileOffset).takeValue();
160 runtime.maskRemapMinimum = std::move(remapMin).takeValue();
161 runtime.maskRemapMaximum = std::move(remapMax).takeValue();
162 runtime.specular = std::move(specular).takeValue();
163 runtime.metallic = metallic.value();
164 runtime.normalScale = normalScale.value();
165 runtime.smoothness = smoothness.value();
166 if (version->asInt() >= 3) {
167 auto diffuseAsset = assetField(*layer, "diffuseAsset", limits);
168 auto normalAsset = assetField(*layer, "normalAsset", limits);
169 auto weightAsset = assetField(*layer, "weightAsset", limits);
170 auto maskAsset = assetField(*layer, "maskAsset", limits);
171 if (!diffuseAsset || !normalAsset || !weightAsset || !maskAsset)
172 return failure<LoadedTerrainMaterial>(
173 DiagnosticCode::ParseError, "terrain layer v3 AssetRefs are invalid", std::to_string(index));
174 runtime.diffuseAsset = std::move(diffuseAsset).takeValue();
175 runtime.normalAsset = std::move(normalAsset).takeValue();
176 runtime.weightAsset = std::move(weightAsset).takeValue();
177 runtime.maskAsset = std::move(maskAsset).takeValue();
178 }
179 }
180 decoded.push_back(std::move(runtime));
181 }
182 std::string holes;
183 std::array<std::string, 4> controls;
184 std::optional<AssetRef> holesAsset;
185 std::array<std::optional<AssetRef>, 4> controlAssets;
186 float boundsMultiplier = 1;
187 if (version->asInt() >= 2) {
188 auto parsedHoles = textField(*root, "holesSource", limits, true);
189 const auto* encodedControls = field(*root, "controlSources");
190 const auto* controlArray = encodedControls ? encodedControls->getIf<Value::Array>() : nullptr;
191 auto parsedBounds = numberField(*root, "boundsMultiplier", 1);
192 if (!parsedHoles || !controlArray || controlArray->size() != controls.size() || !parsedBounds)
193 return failure<LoadedTerrainMaterial>(DiagnosticCode::ParseError,
194 "terrain material v2 fields are incomplete");
195 for (std::size_t i = 0; i < controls.size(); ++i) {
196 if (!(*controlArray)[i].isString() || (*controlArray)[i].asString().size() > limits.maximumStringBytes ||
197 !isValidUtf8((*controlArray)[i].asString(), Utf8NullPolicy::Reject))
198 return failure<LoadedTerrainMaterial>(DiagnosticCode::ParseError, "terrain control source is invalid",
199 std::to_string(i));
200 controls[i] = (*controlArray)[i].asString();
201 }
202 if (parsedBounds.value() <= 0 || parsedBounds.value() > 1'000'000)
203 return failure<LoadedTerrainMaterial>(DiagnosticCode::InvalidArgument,
204 "terrain bounds multiplier is invalid");
205 holes = std::move(parsedHoles).takeValue();
206 boundsMultiplier = parsedBounds.value();
207 if (version->asInt() >= 3) {
208 auto parsedHolesAsset = assetField(*root, "holesAsset", limits);
209 const auto* encodedControlAssets = field(*root, "controlAssets");
210 const auto* controlAssetArray =
211 encodedControlAssets ? encodedControlAssets->getIf<Value::Array>() : nullptr;
212 if (!parsedHolesAsset || !controlAssetArray || controlAssetArray->size() != controlAssets.size())
213 return failure<LoadedTerrainMaterial>(DiagnosticCode::ParseError,
214 "terrain material v3 AssetRefs are incomplete");
215 holesAsset = std::move(parsedHolesAsset).takeValue();
216 for (std::size_t i = 0; i < controlAssets.size(); ++i) {
217 Value::Object encoded{{"asset", (*controlAssetArray)[i]}};
218 auto parsed = assetField(encoded, "asset", limits);
219 if (!parsed)
220 return failure<LoadedTerrainMaterial>(DiagnosticCode::ParseError,
221 "terrain control AssetRef is invalid", std::to_string(i));
222 controlAssets[i] = std::move(parsed).takeValue();
223 }
224 }
225 }
226 LoadedTerrainMaterial result{material};
227 result.layers = std::move(decoded);
228 result.holesSource = std::move(holes);
229 result.controlSources = std::move(controls);
230 result.holesAsset = std::move(holesAsset);
231 result.controlAssets = std::move(controlAssets);
232 result.boundsMultiplier = boundsMultiplier;
233 result.variant = std::move(payload).takeValue().variant;
234 return Result<LoadedTerrainMaterial>::success(std::move(result));
235}
