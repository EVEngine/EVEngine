35Result<LoadedTerrainMaterial> EvpackTerrainMaterialLoader::load(
36 const AssetRef& material, const asset::EvpackCapabilities& capabilities,
37 const TerrainMaterialLoadLimits& limits) const {
38 auto payload = reader_.read(material, "eve.terrain-material/1", capabilities,
39 limits.maximumDecodedBytes);
40 if (!payload) return Result<LoadedTerrainMaterial>::failure(payload.status());
41 const asset::RuntimeAssetChunk* definition = nullptr;
42 for (const auto& chunk : payload.value().chunks) {
43 if (chunk.kind != asset::EvpackChunkKind::Definition)
44 return failure<LoadedTerrainMaterial>(DiagnosticCode::TypeMismatch,
45 "terrain material cannot contain runtime bulk");
46 if (definition)
47 return failure<LoadedTerrainMaterial>(DiagnosticCode::Conflict,
48 "terrain material has duplicate definitions");
49 definition = &chunk;
50 }
51 if (!definition)
52 return failure<LoadedTerrainMaterial>(DiagnosticCode::NotFound,
53 "terrain material definition is missing");
54 asset::RuntimeDefinitionLimits definitionLimits;
55 definitionLimits.maximumBytes = limits.maximumDecodedBytes;
56 definitionLimits.maximumStringBytes = limits.maximumStringBytes;
57 auto metadata = asset::decodeRuntimeDefinition(definition->bytes, definitionLimits);
58 if (!metadata) return Result<LoadedTerrainMaterial>::failure(metadata.status());
59 const auto* root = metadata.value().getIf<Value::Object>();
60 const Value* schema = root ? field(*root, "schema") : nullptr;
61 const Value* version = root ? field(*root, "schemaVersion") : nullptr;
62 const Value* layersValue = root ? field(*root, "layers") : nullptr;
63 const auto* layers = layersValue ? layersValue->getIf<Value::Array>() : nullptr;
64 if (!schema || !schema->isString() || schema->asString() != "eve.terrain-material" ||
65 !version || !version->isInt64() || version->asInt() != 1 || !layers ||
66 layers->empty() || layers->size() > limits.maximumLayers)
67 return failure<LoadedTerrainMaterial>(DiagnosticCode::ParseError,
68 "terrain material envelope is invalid");
69 std::vector<RuntimeTerrainLayer> decoded;
70 decoded.reserve(layers->size());
71 for (std::size_t index = 0; index < layers->size(); ++index) {
72 const auto* layer = (*layers)[index].getIf<Value::Object>();
73 if (!layer)
74 return failure<LoadedTerrainMaterial>(DiagnosticCode::ParseError,
75 "terrain layer must be an object",
76 std::to_string(index));
77 auto name = textField(*layer, "name", limits, false);
78 auto diffuse = textField(*layer, "diffuseSource", limits, true);
79 auto normal = textField(*layer, "normalSource", limits, true);
80 auto weight = textField(*layer, "weightSource", limits, true);
81 auto convention = textField(*layer, "normalConvention", limits, false);
82 const Value* tile = field(*layer, "tileSizeMeters");
83 if (!name || !diffuse || !normal || !weight || !convention || !tile ||
84 (!tile->isDouble() && !tile->isInt64()))
85 return failure<LoadedTerrainMaterial>(DiagnosticCode::ParseError,
86 "terrain layer fields are incomplete",
87 std::to_string(index));
88 const double tileSize = tile->isDouble() ? tile->asDouble() : double(tile->asInt());
89 if (!std::isfinite(tileSize) || tileSize <= 0 || tileSize > 1'000'000.0 ||
90 (convention.value() != "opengl" && convention.value() != "directx"))
91 return failure<LoadedTerrainMaterial>(DiagnosticCode::InvalidArgument,
92 "terrain layer physical semantics are invalid",
93 std::to_string(index));
94 decoded.push_back({std::move(name).takeValue(), std::move(diffuse).takeValue(),
95 std::move(normal).takeValue(), std::move(weight).takeValue(),
96 std::move(convention).takeValue(), static_cast<float>(tileSize)});
97 }
98 return Result<LoadedTerrainMaterial>::success(
99 {material, std::move(decoded), std::move(payload).takeValue().variant});
100}
