421eve::Result<void> Procgen::commitSystem(ProcgenContextHandleRef reference) {
422 auto view = Procgen::resolve(reference);
423 if (!view.isBound())
424 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle,
425 "commitSystem context handle is stale", "context", {},
426 "procgen.squirrel"));
427 ProcgenContext* context = view.get();
428 if (!context->isActive())
429 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
430 "commitSystem: transaction is closed", "context", {},
431 "procgen.squirrel"));
432 if (context->hasFailed()) {
433 const std::string error = context->getError();
434 context->close();
435 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::Failed, "commitSystem: " + error,
436 "context", {}, "procgen.squirrel"));
437 }
438 if (!context->openTraces_.empty()) {
439 const std::string trace = context->openTraces_.back().name;
440 context->close();
441 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
442 "commitSystem: unfinished trace '" + trace + "'",
443 "context", {}, "procgen.squirrel"));
444 }
445
446 const auto current = systems_.find(context->name_);
447 if (current != systems_.end()) previousSystems_[context->name_] = current->second;
448 auto& snapshot = systems_[context->name_];
449 snapshot.seed = context->seed_;
450 snapshot.revision = snapshot.revision + 1u;
451 snapshot.buildKey = context->buildKey_;
452 snapshot.outputs = context->outputs_;
453 snapshot.outputOrder = context->outputOrder_;
454 snapshot.debugStages = context->debugStages_;
455 snapshot.debugStageOrder = context->debugStageOrder_;
456 snapshot.stageCache = context->stageCache_;
457 snapshot.stageCacheHits = context->stageCacheHits_;
458 snapshot.stageCacheMisses = context->stageCacheMisses_;
459 snapshot.traces = context->traces_;
460 context->close();
461 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
462}

568 const auto found = systems_.find(name);
569 if (found == systems_.end()) return "system '" + name + "' is not committed";
570 const auto& snapshot = found->second;
571 std::ostringstream report;
572 report << name << " revision=" << snapshot.revision << " seed=" << snapshot.seed;
573 if (!snapshot.buildKey.empty()) report << " buildKey=" << snapshot.buildKey;
574 report << " stageCache=" << snapshot.stageCacheHits << " hit/" << snapshot.stageCacheMisses << " miss";
575 for (const auto& trace : snapshot.traces) {
576 report << "\n " << trace.name << " input=" << trace.inputCount << " output=" << trace.outputCount
577 << " ms=" << trace.milliseconds;
578 }
579 for (const auto& outputName : snapshot.outputOrder) {
580 report << "\n output " << outputName << " points=" << snapshot.outputs.at(outputName).getCount();
581 }
582 for (const auto& stageName : snapshot.debugStageOrder) {
583 report << "\n debug " << stageName << " points=" << snapshot.debugStages.at(stageName).getCount();
584 }
585 return report.str();

588std::string Procgen::getSystemDebugDiffReport(const std::string& name) const {
589 const auto current = systems_.find(name);
590 if (current == systems_.end()) return "system '" + name + "' is not committed";
591 const auto previous = previousSystems_.find(name);
592 if (previous == previousSystems_.end()) return name + " has no previous revision";
593
594 std::ostringstream report;
595 report << name << " revision=" << previous->second.revision << " -> " << current->second.revision;
596 for (const auto& stageName : current->second.debugStageOrder) {
597 const int currentCount = current->second.debugStages.at(stageName).getCount();
598 const auto oldStage = previous->second.debugStages.find(stageName);
599 const int oldCount = oldStage == previous->second.debugStages.end() ? 0 : oldStage->second.getCount();
600 report << "\n debug " << stageName << " points=" << currentCount << " delta=";
601 if (currentCount >= oldCount) report << "+";
602 report << currentCount - oldCount;
603 }
604 for (const auto& stageName : previous->second.debugStageOrder) {
605 if (current->second.debugStages.find(stageName) != current->second.debugStages.end()) continue;
606 report << "\n debug " << stageName << " removed delta=-"
607 << previous->second.debugStages.at(stageName).getCount();
608 }
609 return report.str();
610}

1744float diagnosticScale(const std::vector<float>& values, float exposure) {
1745 if (std::isfinite(exposure) && exposure > 0.f) return exposure;
1746 std::vector<float> positive;
1747 positive.reserve(values.size());
1748 for (float value : values)
1749 if (std::isfinite(value) && value > 0.f) positive.push_back(value);
1750 if (positive.empty()) return 1.f;
1751 const size_t percentile = std::min(positive.size() - 1, size_t(std::floor(float(positive.size() - 1) * 0.99f)));
1752 std::nth_element(positive.begin(), positive.begin() + ptrdiff_t(percentile), positive.end());
1753 return 1.f / std::max(1e-8f, positive[percentile]);
1754}
