18 void quad(float ax, float ay, float az, float bx, float by, float bz, float cx, float cy,

21 mesh.addVertex(ax, ay, az, nx, ny, nz, 0, 0);
22 mesh.addVertex(bx, by, bz, nx, ny, nz, std::hypot(bx-ax, bz-az)*uv, 0);
23 mesh.addVertex(cx, cy, cz, nx, ny, nz, std::hypot(cx-dx, cz-dz)*uv,
24 std::fabs(cy-by)*uv);
25 mesh.addVertex(dx, dy, dz, nx, ny, nz, 0, std::fabs(dy-ay)*uv);

39 void cylinder(float cx, float y0, float cz, float radius, float height, int sides) {
40 const float y1 = y0 + height;
41 for (int i=0; i<sides; ++i) {
42 float a0=2*kPi*float(i)/float(sides), a1=2*kPi*float(i+1)/float(sides);
43 float x0=cx+std::cos(a0)*radius, z0=cz+std::sin(a0)*radius;
44 float x1=cx+std::cos(a1)*radius, z1=cz+std::sin(a1)*radius;
45 float am=(a0+a1)*0.5f;
46 quad(x0,y0,z0, x1,y0,z1, x1,y1,z1, x0,y1,z0,
47 std::cos(am),0,std::sin(am));

53void crenelsX(Builder &b, float x0, float x1, float y, float z, float depth,
54 float merlonW, float merlonH) {

67void stairX(Builder &b, float x0, float y0, float z, float run, float rise, float width,
68 float stepHeight, bool reverse=false) {
69 int steps=std::max(2,int(std::ceil(rise/stepHeight)));
70 float dx=run/float(steps), dy=rise/float(steps);
71 for(int i=0;i<steps;++i) {
72 // Both directions occupy the same [x0, x0 + run] footprint; reverse
73 // changes only which end reaches the upper landing.
74 float xa=reverse ? x0+(steps-1-i)*dx : x0+i*dx;
75 b.box(xa,y0,z-width*.5f,xa+dx,y0+(i+1)*dy,z+width*.5f);
76 }

79void tower(Builder &b,float x,float z,float radius,float height,int sides,float merlonW,
80 int detail,int &merlons) {
81 b.cylinder(x,0,z,radius,height,sides);
82 if(detail>0) {
83 int n=std::max(6,int(std::floor(2*kPi*radius/merlonW)));
84 for(int i=0;i<n;i+=2) {
85 float a=2*kPi*(float(i)+.5f)/float(n), tang=2*kPi*radius/float(n)*.42f;
86 float cx=x+std::cos(a)*(radius-tang*.35f), cz=z+std::sin(a)*(radius-tang*.35f);
87 b.box(cx-tang*.5f,height,cz-tang*.5f,cx+tang*.5f,height+radius*.3f,cz+tang*.5f);
88 ++merlons;
89 }

95bool generateCastleMesh(const Params &p, MeshBuild &out, std::string &error) {
96 const float width=std::max(16.f,p.getFloat("width",42.f));
97 const float depth=std::max(16.f,p.getFloat("depth",36.f));
98 const int rings=std::clamp(p.getInt("rings",2),1,4);
99 const float wallH=std::max(2.f,p.getFloat("wallHeight",6.f));
100 const float wallT=std::clamp(p.getFloat("wallThickness",1.8f),.5f,5.f);
101 const float towerR=std::max(wallT,p.getFloat("towerRadius",3.3f));
102 const float towerH=std::max(wallH+1.f,p.getFloat("towerHeight",9.f));
103 const int sides=std::clamp(p.getInt("towerSides",12),6,32);
104 const float spacing=std::max(8.f,p.getFloat("towerSpacing",16.f));
105 const float gateW=std::clamp(p.getFloat("gateWidth",5.f),2.f,width*.3f);
106 const float merlonW=std::max(.35f,p.getFloat("merlonWidth",1.25f));
107 const float stairW=std::max(1.f,p.getFloat("stairWidth",2.f));
108 const float stepH=std::clamp(p.getFloat("stepHeight",.28f),.12f,.5f);
109 const int keepFloors=std::clamp(p.getInt("keepFloors",3),1,8);
110 const int detail=std::clamp(p.getInt("detail",2),0,2);
111 const float ringInset=std::max(wallT*2.5f,p.getFloat("ringInset",std::min(width,depth)*.13f));
112 const float ringHeightStep=std::clamp(p.getFloat("ringHeightStep",.22f),-.25f,.75f);
113 const float towerHeightStep=std::clamp(p.getFloat("towerHeightStep",.14f),-.25f,.75f);
114 const float floorH=std::max(2.f,p.getFloat("floorHeight",3.2f));
115 const float keepW=std::clamp(p.getFloat("keepWidth",width*.25f),6.f,width*.55f);
116 const float keepD=std::clamp(p.getFloat("keepDepth",depth*.24f),6.f,depth*.55f);
117 const int courtyardBuildings=std::clamp(p.getInt("courtyardBuildings",3),0,12);
118 if(width <= 2*towerR+2*wallT || depth <= 2*towerR+2*wallT) {
119 error="mesh.castle: width/depth too small for towerRadius and wallThickness"; return false;
120 }
121 out.clear(); out.reserve(40000,120000); Builder b{out,std::max(0.f,p.getFloat("uvRepeat",.5f))};
122 std::mt19937 rng(p.getSeed());
123 int towers=0,walls=0,stairs=0,merlons=0,actualRings=0;
124 for(int ring=0;ring<rings;++ring) {
125 float inset=float(ring)*ringInset;
126 float hx=width*.5f-inset, hz=depth*.5f-inset;
127 if(hx<towerR*1.8f || hz<towerR*1.8f) break;
128 ++actualRings;
129 float h=wallH*(1.f+ringHeightStep*float(ring));
130 // South wall is split around a real gate opening.
131 b.mesh.setActiveGroup("walls");
132 b.box(-hx,0,-hz,-gateW*.5f,h,-hz+wallT); b.box(gateW*.5f,0,-hz,hx,h,-hz+wallT);
133 b.box(-hx,0,hz-wallT,hx,h,hz); b.box(-hx,0,-hz+wallT,-hx+wallT,h,hz-wallT);
134 b.box(hx-wallT,0,-hz+wallT,hx,h,hz-wallT); walls+=4;
135 if(detail>0) {
136 b.mesh.setActiveGroup("battlements");
137 crenelsX(b,-hx,hx,h,-hz,wallT,merlonW,wallT*.7f);
138 crenelsX(b,-hx,hx,h,hz,wallT,merlonW,wallT*.7f);
139 crenelsZ(b,-hz,hz,h,-hx,wallT,merlonW,wallT*.7f);
140 crenelsZ(b,-hz,hz,h,hx,wallT,merlonW,wallT*.7f);
141 }
142 float tr=towerR*(1.f-.12f*float(ring)), th=towerH*(1.f+towerHeightStep*float(ring));
143 b.mesh.setActiveGroup("towers");
144 for(float sx:{-1.f,1.f}) for(float sz:{-1.f,1.f}) {
145 tower(b,sx*hx,sz*hz,tr,th,sides,merlonW,detail,merlons); ++towers;
146 }
147 // Interval towers make large castles read as defensible rather than four-corner boxes.
148 if(detail>1) {
149 int nx=std::max(0,int(std::floor(2*hx/spacing))-1);
150 int nz=std::max(0,int(std::floor(2*hz/spacing))-1);
151 for(int i=1;i<=nx;++i){float x=-hx+2*hx*i/float(nx+1); tower(b,x,hz,tr*.78f,th*.88f,sides,merlonW,detail,merlons); ++towers;}
152 for(int i=1;i<=nz;++i){float z=-hz+2*hz*i/float(nz+1); tower(b,hx,z,tr*.78f,th*.88f,sides,merlonW,detail,merlons); ++towers;}
153 }
154 // Gatehouse towers and lintel preserve the opening below.
155 const float gateTowerR=tr*.72f;
156 const float gateTowerX=gateW*.5f+gateTowerR*.72f;
157 b.mesh.setActiveGroup("gatehouses");
158 tower(b,-gateTowerX,-hz,gateTowerR,th*.9f,sides,merlonW,detail,merlons);
159 tower(b, gateTowerX,-hz,gateTowerR,th*.9f,sides,merlonW,detail,merlons); towers+=2;
160 b.box(-gateW*.5f,h*.62f,-hz-wallT*.2f,gateW*.5f,h,-hz+wallT*1.2f);
161 // Each ring has a physical flight onto its wall walk.
162 b.mesh.setActiveGroup("stairs");
163 stairX(b, -hx + wallT, 0.f, -hz + wallT + stairW * .55f,
164 h * 1.35f, h, stairW, stepH);
165 b.box(-hx+wallT+h*1.35f-stairW*.5f,h-wallT*.25f,-hz+wallT,
166 -hx+wallT+h*1.35f+stairW*.5f,h,-hz+wallT+stairW);
167 ++stairs;
168 }
169 // Central keep: stacked floors with setbacks, battlements, and an external stair.
170 float kw=keepW,kd=keepD;
171 b.mesh.setActiveGroup("keep");
172 for(int f=0;f<keepFloors;++f) {
173 float setback=float(f)*.35f, y=float(f)*floorH;
174 b.box(-kw*.5f+setback,y,-kd*.5f+setback,kw*.5f-setback,y+floorH,kd*.5f-setback);
175 if(f+1<keepFloors){
176 b.mesh.setActiveGroup("stairs");
177 stairX(b,-kw*.5f+setback,y,-kd*.5f-stairW*.55f,kw-2*setback,floorH,stairW,stepH,(f%2)==1);
178 b.mesh.setActiveGroup("keep");
179 ++stairs;
180 }
181 }
182 float ky=keepFloors*floorH;
183 crenelsX(b,-kw*.5f,kw*.5f,ky,-kd*.5f,wallT*.55f,merlonW*.8f,wallT*.65f);
184 crenelsX(b,-kw*.5f,kw*.5f,ky,kd*.5f,wallT*.55f,merlonW*.8f,wallT*.65f);
185 crenelsZ(b,-kd*.5f,kd*.5f,ky,-kw*.5f,wallT*.55f,merlonW*.8f,wallT*.65f);
186 crenelsZ(b,-kd*.5f,kd*.5f,ky,kw*.5f,wallT*.55f,merlonW*.8f,wallT*.65f);
187 // A continuous raised threshold makes the aligned passages through all
188 // rings readable and guarantees a walkable route from outside to the keep.
189 b.mesh.setActiveGroup("gatehouses");
190 b.box(-gateW*.38f, .02f, -depth*.5f-wallT*2.5f,
191 gateW*.38f, .18f, -kd*.5f);
192 // Seeded courtyard buildings add controlled variation without changing topology.
193 b.mesh.setActiveGroup("courtyard");
194 if(detail>1) for(int i=0;i<courtyardBuildings;++i) {
195 std::uniform_real_distribution<float> jitter(-1.f,1.f);
196 const float lane=float(i)-float(courtyardBuildings-1)*.5f;
197 float x=lane*kw*.72f+jitter(rng), z=kd*.85f+jitter(rng);
198 b.box(x-2,0,z-1.8f,x+2,2.6f+float(i%3)*.35f,z+1.8f);
199 }
200 float scale=std::max(.01f,p.getFloat("scale",1.f));
201 if(scale!=1.f) for(float &v:out.positions()) v*=scale;
202 if(out.empty()){error="mesh.castle: generated empty mesh";return false;}
203 out.setMeta("algorithm","mesh.castle"); out.setMeta("rings",std::to_string(actualRings));
204 out.setMeta("wallSections",std::to_string(walls)); out.setMeta("towerCount",std::to_string(towers));
205 out.setMeta("stairFlights",std::to_string(stairs)); out.setMeta("keepFloors",std::to_string(keepFloors));
206 out.setMeta("merlonCount",std::to_string(merlons)); out.setMeta("seed",std::to_string(p.getSeed()));
207 out.setMeta("detail",std::to_string(detail));
208 out.setMeta("courtyardBuildings",std::to_string(detail>1?courtyardBuildings:0));
209 return true;
210}
