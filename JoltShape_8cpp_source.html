16eve::Result<JPH::RefConst<JPH::Shape>> cook(const ShapeGeometry3D& g, float density, JPH::uint64 userData) {
17 using R = eve::Result<JPH::RefConst<JPH::Shape>>;
18 if (!std::isfinite(density) || density < 0 || !isFinite(g.localX, g.localY, g.localZ) ||
19 !isFinite(g.localQx, g.localQy, g.localQz) || !std::isfinite(g.localQw))
20 return rejected<JPH::RefConst<JPH::Shape>>("Invalid shape parameters", eve::DiagnosticCode::InvalidArgument);
21 std::unique_ptr<JPH::ConvexShapeSettings> settings;
22 if (g.kind == ShapeKind3D::Box && isFinite(g.a, g.b, g.c) && g.a > 0 && g.b > 0 && g.c > 0)
23 settings = std::make_unique<JPH::BoxShapeSettings>(JPH::Vec3(g.a, g.b, g.c), std::min({0.05f, g.a, g.b, g.c}));
24 else if (g.kind == ShapeKind3D::Sphere && std::isfinite(g.a) && g.a > 0)
25 settings = std::make_unique<JPH::SphereShapeSettings>(g.a);
26 else if (g.kind == ShapeKind3D::Capsule && isFinite(g.a, g.b, 0) && g.a >= 0 && g.b > 0)
27 settings = std::make_unique<JPH::CapsuleShapeSettings>(g.a, g.b);
28 else if (g.kind == ShapeKind3D::ConvexHull) {
29 if (g.hullVertices.size() < 12 || g.hullVertices.size() % 3 || g.hullVertices.size() / 3 > 100000 ||
30 g.hullMaxVertices < 4 || g.hullMaxVertices > 254)
31 return rejected<JPH::RefConst<JPH::Shape>>("Invalid hull", eve::DiagnosticCode::InvalidArgument);
32 auto hull = std::make_unique<JPH::ConvexHullShapeSettings>();
33 for (size_t i = 0; i < g.hullVertices.size(); i += 3) {
34 if (!isFinite(g.hullVertices[i], g.hullVertices[i + 1], g.hullVertices[i + 2]))
35 return rejected<JPH::RefConst<JPH::Shape>>("Invalid hull", eve::DiagnosticCode::InvalidArgument);
36 hull->mPoints.emplace_back(g.hullVertices[i], g.hullVertices[i + 1], g.hullVertices[i + 2]);
37 }
38 JPH::ConvexHullBuilder builder(hull->mPoints);
39 const char* error = nullptr;
40 const auto built = builder.Initialize(g.hullMaxVertices, hull->mHullTolerance, error);
41 if (built != JPH::ConvexHullBuilder::EResult::Success &&
42 built != JPH::ConvexHullBuilder::EResult::MaxVerticesReached)
43 return rejected<JPH::RefConst<JPH::Shape>>(error ? error : "Invalid hull",
44 eve::DiagnosticCode::InvalidArgument);
45 JPH::Vec3 center;
46 float volume = 0;
47 builder.GetCenterOfMassAndVolume(center, volume);
48 if (!std::isfinite(volume) || volume <= 0)
49 return rejected<JPH::RefConst<JPH::Shape>>("Hull must have positive volume",
50 eve::DiagnosticCode::InvalidArgument);
51 // Keep source order for deterministic cooking; faces borrow indices from the original points.
52 std::vector<bool> used(hull->mPoints.size(), false);
53 for (const auto* face : builder.GetFaces()) {
54 const auto* edge = face->mFirstEdge;
55 do {
56 used[static_cast<size_t>(edge->mStartIdx)] = true;
57 edge = edge->mNextEdge;
58 } while (edge != face->mFirstEdge);
59 }
60 JPH::Array<JPH::Vec3> reduced;
61 for (size_t i = 0; i < used.size(); ++i)
62 if (used[i]) reduced.push_back(hull->mPoints[i]);
63 hull->mPoints = std::move(reduced);
64 settings = std::move(hull);
65 } else if (g.kind == ShapeKind3D::TriangleMesh || g.kind == ShapeKind3D::HeightField)
66 return cookMesh(g, userData);
67 else
68 return rejected<JPH::RefConst<JPH::Shape>>("Invalid primitive dimensions",
69 eve::DiagnosticCode::InvalidArgument);
70 settings->mDensity = density;
71 settings->mUserData = userData;
72 auto result = settings->Create();
73 if (result.HasError())
74 return rejected<JPH::RefConst<JPH::Shape>>(result.GetError().c_str(), eve::DiagnosticCode::InvalidArgument);
75 return R::success(result.Get());
76}
77eve::Result<void> replaceShapes(BodyState& body, const std::vector<std::shared_ptr<ShapeState>>& shapes) {
78 if (!body.valid()) return rejected<void>("Body expired", eve::DiagnosticCode::StaleHandle);
79 for (const auto& shape : shapes)
80 if (shape->native->MustBeStatic() && body.api().GetMotionType(body.id) != JPH::EMotionType::Static)
81 return rejected<void>("Concave shapes require a static body", eve::DiagnosticCode::InvalidArgument);
82 JPH::RefConst<JPH::Shape> native;
83 if (shapes.empty())
84 native = new JPH::EmptyShape;
85 else {
86 JPH::StaticCompoundShapeSettings settings;
87 for (const auto& s : shapes) {
88 const auto& g = s->geometry;
89 JPH::Quat q(g.localQx, g.localQy, g.localQz, g.localQw);
90 if (!std::isfinite(q.LengthSq()) || q.LengthSq() < 1.e-12f)
91 return rejected<void>("Invalid local rotation", eve::DiagnosticCode::InvalidArgument);
92 settings.AddShape({g.localX, g.localY, g.localZ}, q.Normalized(), s->native);
93 }
94 auto result = settings.Create();
95 if (result.HasError()) return rejected<void>(result.GetError().c_str(), eve::DiagnosticCode::InvalidArgument);
96 native = result.Get();
97 }
98 auto mass = body.mass;
99 if (!native->MustBeStatic()) {
100 mass = native->GetMassProperties();
101 if (!(mass.mMass > 0)) mass.SetMassAndInertiaOfSolidBox({1, 1, 1}, 1);
102 }
103 const auto previous = body.api().GetShape(body.id)->GetCenterOfMass();
104 std::erase_if(body.world.lock()->listener->solved, [&](const auto& entry) {
105 return entry.first.GetBody1ID() == body.id || entry.first.GetBody2ID() == body.id;
106 });
107 body.api().SetShape(body.id, native, false, JPH::EActivation::Activate);
108 if (!native->MustBeStatic()) {
109 auto* nativeBody = body.world.lock()->system.GetBodyLockInterfaceNoLock().TryGetBody(body.id);
110 nativeBody->GetMotionPropertiesUnchecked()->SetMassProperties(body.dofs, mass);
111 body.mass = mass;
112 }
113 notifyJointShapeChanged(*body.world.lock(), body.id, native->GetCenterOfMass() - previous);
114 return success();
115}

136eve::Result<void> ShapeBackend::rebuild(const ShapeGeometry3D& geometry, ShapeBehavior3D behavior,
137 std::span<const SurfaceMaterial3D> materials) {
138 if (!isValid()) return rejected<void>("Shape expired", eve::DiagnosticCode::StaleHandle);
139 for (const auto& material : materials) {
140 auto checked = validateMaterial(material);
141 if (!checked) return checked;
142 }
143 if (!geometry.meshMaterialIndices.empty() &&
144 *std::max_element(geometry.meshMaterialIndices.begin(), geometry.meshMaterialIndices.end()) >= materials.size())
145 return rejected<void>("Missing mesh material", eve::DiagnosticCode::InvalidArgument);
146 auto cooked = cook(geometry, shape->density, reinterpret_cast<JPH::uint64>(shape.get()));
147 if (!cooked) return eve::Result<void>::failure(cooked.status());
148 auto old = shape->native;
149 auto oldGeometry = shape->geometry;
150 shape->native = std::move(cooked).takeValue();
151 shape->geometry = geometry;
152 auto b = shape->body.lock();
153 auto result = replaceShapes(*b, b->shapes);
154 if (!result) {
155 shape->native = old;
156 shape->geometry = std::move(oldGeometry);
157 return result;
158 }
159 shape->meshMaterials.assign(materials.begin(), materials.end());
160 shape->sensor = behavior.sensor;
161 return success();
162}
