54std::string AttributeSystem::addModifier(RPGActor* actor, const std::string& attribute, const std::string& source,
55 const std::string& operation, double value, int priority) {
56 if (!actor || attribute.empty()) return {};
57
58 auto parsed = ::eve::attributes::parseAttributeOperation(operation, value);
59 AttributeModifier modifier;
60 modifier.attribute = attribute;
61 modifier.source = source;
62 modifier.priority = priority;
63 if (parsed.ok()) {
64 modifier.operation = parsed.value().operation;
65 modifier.value = parsed.value().value;
66 } else if (customOps().has(operation)) {
67 parsed.ignore("RPG custom operation compatibility fallback");
68 modifier.operation = AttributeOperation::Custom;
69 modifier.policyId = operation;
70 modifier.value = value;
71 } else {
72 parsed.ignore("unknown RPG attribute operation");
73 return {};
74 }
75
76 auto result = addModifier(actor, std::move(modifier));
77 if (!result.ok()) return {};
78 return result.value();
