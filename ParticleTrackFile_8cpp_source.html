17struct Reader {
18 std::string_view bytes;
19 std::size_t cursor = 0;
20 std::string_view take(std::size_t count) {
21 if (count > bytes.size() - cursor) throw std::runtime_error("truncated particle asset");
22 auto value = bytes.substr(cursor, count);
23 cursor += count;
24 return value;
25 }
26 std::uint32_t word() {
27 const auto data = take(4);
28 std::uint32_t value = 0;
29 for (unsigned i = 0; i < 4; ++i) value |= std::uint32_t(static_cast<unsigned char>(data[i])) << (i * 8);
30 return value;
31 }
32 float real() { return std::bit_cast<float>(word()); }
33};

64 std::error_code error;
65 const auto status = std::filesystem::status(std::filesystem::path(path), error);

80 if (bakedAssetContentKey(payload) != std::string_view(bytes).substr(bytes.size() - 32))
81 return Result<ParticleTrack>::failure(
82 Diagnostic::error(DiagnosticCode::HashMismatch, "particle asset checksum mismatch", "particleTrack"));
83 try {
84 Reader reader{payload};
85 if (reader.take(8) != "EVEPART1" || reader.word() != 1)
86 return Result<ParticleTrack>::failure(malformed("unknown particle schema/version"));
87 const float duration = reader.real();
88 const auto count = reader.word(), recipeSize = reader.word();

133Result<ParticleTrack> ParticleTrack::loadForRecipe(std::string_view path, const BakeRecipe& expected) {
134 auto expectedBytes = expected.encode();
