97 std::unordered_map<std::string, glm::dmat4>& cache,
98 std::unordered_set<std::string>& visiting) {
99 if (const auto cached = cache.find(id); cached != cache.end()) return editing::applied<glm::dmat4>(cached->second);
100 const auto node = graph.find(id);
101 if (node == graph.end())
102 return editing::failed<glm::dmat4>(editing::Status::NotFound,

108 "Scene hierarchy contains a cycle");

110 if (!node->second.parent.empty()) {

114 }
115 visiting.erase(id);
116 cache.insert_or_assign(id, world);
117 return editing::applied<glm::dmat4>(world);

148bool spansVolume(const std::vector<float>& vertices) {
149 constexpr double epsilonSquared = 1.0e-12;
150 const auto point = [&](std::size_t index) {
151 return std::array<double, 3>{vertices[index], vertices[index + 1], vertices[index + 2]};
152 };
153 const auto subtract = [](const auto& left, const auto& right) {
154 return std::array<double, 3>{left[0] - right[0], left[1] - right[1], left[2] - right[2]};
155 };
156 const auto cross = [](const auto& left, const auto& right) {
157 return std::array<double, 3>{left[1] * right[2] - left[2] * right[1], left[2] * right[0] - left[0] * right[2],
158 left[0] * right[1] - left[1] * right[0]};
159 };
160 const auto dot = [](const auto& left, const auto& right) {
161 return left[0] * right[0] + left[1] * right[1] + left[2] * right[2];
162 };
163 const auto origin = point(0);
164 std::array<double, 3> first{};
165 std::array<double, 3> normal{};
166 bool hasFirst = false;
167 bool hasNormal = false;
168 for (std::size_t index = 3; index < vertices.size(); index += 3) {
169 const auto edge = subtract(point(index), origin);
170 if (!hasFirst) {
171 if (dot(edge, edge) <= epsilonSquared) continue;
172 first = edge;
173 hasFirst = true;
174 continue;
175 }
176 if (!hasNormal) {
177 const auto candidate = cross(first, edge);
178 if (dot(candidate, candidate) <= epsilonSquared) continue;
179 normal = candidate;
180 hasNormal = true;
181 continue;
182 }
183 if (std::abs(dot(normal, edge)) > 1.0e-9) return true;
184 }
185 return false;
186}

226 const auto* shapeValue = valueField(value, "shape");
227 const auto* sourceValue = valueField(value, "source");
228 const auto* halfValue = valueField(value, "halfExtents");
229 const auto* localValue = valueField(value, "localPosition");
230 const auto* rotationValue = valueField(value, "localRotation");
231 const auto* verticesValue = valueField(value, "vertices");
232 const auto* maximumValue = valueField(value, "maxVertices");
233 const auto* shape = shapeValue ? shapeValue->getIf<std::string>() : nullptr;
234 const auto* source = sourceValue ? sourceValue->getIf<std::string>() : nullptr;
235 const auto* half = halfValue ? halfValue->getIf<editing::Value::Array>() : nullptr;
236 const auto* local = localValue ? localValue->getIf<editing::Value::Array>() : nullptr;
237 const auto* rotation = rotationValue ? rotationValue->getIf<editing::Value::Array>() : nullptr;
238 const auto* vertices = verticesValue ? verticesValue->getIf<editing::Value::Array>() : nullptr;
239 const auto* maximum = maximumValue ? maximumValue->getIf<std::int64_t>() : nullptr;
240 PhysicsPlacementCollider collider;

242 rotation->size() != 3 || !vertices || !maximum || !valueNumber((*half)[0], collider.halfExtentX) ||
243 !valueNumber((*half)[1], collider.halfExtentY) || !valueNumber((*half)[2], collider.halfExtentZ) ||
244 !valueNumber((*local)[0], collider.localX) || !valueNumber((*local)[1], collider.localY) ||
245 !valueNumber((*local)[2], collider.localZ) || !valueNumber((*rotation)[0], collider.localRotationX) ||
246 !valueNumber((*rotation)[1], collider.localRotationY) || !valueNumber((*rotation)[2], collider.localRotationZ))
247 return editing::failed<PhysicsPlacementCollider>(editing::Status::Rejected,
248 editing::RuleId("scene.physics-placement.cache-part"),
249 "Collider cache part has an invalid schema");
250 if (*shape == "sphere")
251 collider.shape = PhysicsPlacementShape::Sphere;
252 else if (*shape == "capsule")
253 collider.shape = PhysicsPlacementShape::Capsule;
254 else if (*shape == "convex")
255 collider.shape = PhysicsPlacementShape::ConvexHull;
256 else if (*shape != "box")
257 return editing::failed<PhysicsPlacementCollider>(editing::Status::Rejected,
258 editing::RuleId("scene.physics-placement.cache-shape"),
259 "Collider cache shape is unsupported");

261 collider.source = PhysicsPlacementColliderSource::Existing;
262 else if (*source != "generated")
263 return editing::failed<PhysicsPlacementCollider>(editing::Status::Rejected,
264 editing::RuleId("scene.physics-placement.cache-source"),

267 collider.convexVertices.reserve(vertices->size());
268 for (const auto& component : *vertices) {
269 double numeric = 0.0;
270 if (!valueNumber(component, numeric))
271 return editing::failed<PhysicsPlacementCollider>(editing::Status::Rejected,
272 editing::RuleId("scene.physics-placement.cache-vertex"),
273 "Collider cache vertices must be numeric");
274 collider.convexVertices.push_back(static_cast<float>(numeric));
275 }
276 return editing::applied<PhysicsPlacementCollider>(std::move(collider));
277}

425 auto decoded = editing::editorValueFromJson(json);
426 if (!decoded.ok()) return editing::Result<void>::failure(decoded.status());
427 const auto* schemaValue = valueField(decoded.value(), "schema");
428 const auto* versionValue = valueField(decoded.value(), "version");
429 const auto* entriesValue = valueField(decoded.value(), "entries");

431 const auto* version = versionValue ? versionValue->getIf<std::int64_t>() : nullptr;
432 const auto* entries = entriesValue ? entriesValue->getIf<editing::Value::Array>() : nullptr;

437 for (const auto& entry : *entries) {
438 const auto* objectValue = valueField(entry, "object");
439 const auto* keyValue = valueField(entry, "resourceKey");
440 const auto* collidersValue = valueField(entry, "colliders");
441 const auto* object = objectValue ? objectValue->getIf<std::string>() : nullptr;
442 const auto* key = keyValue ? keyValue->getIf<std::string>() : nullptr;
443 const auto* colliders = collidersValue ? collidersValue->getIf<editing::Value::Array>() : nullptr;
444 if (!object || object->empty() || !key || !colliders || colliders->empty() || colliders->size() > 64 ||
445 replacement.contains(*object))

447 editing::RuleId("scene.physics-placement.cache-entry"),

449 std::vector<PhysicsPlacementCollider> parsed;
450 parsed.reserve(colliders->size());
451 for (const auto& value : *colliders) {
