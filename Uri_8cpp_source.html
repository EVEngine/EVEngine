20[[nodiscard]] bool isSchemeCharacter(char value, bool first) noexcept {
21 const bool alpha = (value >= 'a' && value <= 'z') || (value >= 'A' && value <= 'Z');
22 const bool digit = value >= '0' && value <= '9';
23 return first ? alpha : (alpha || digit || value == '+' || value == '-' || value == '.');

26[[nodiscard]] bool isPathCharacter(char value) noexcept {
27 const auto byte = static_cast<unsigned char>(value);
28 if (byte < 0x21 || byte > 0x7e) return false;
29 switch (value) {
30 case '/':
31 case '_':
32 case '-':
33 case '.':
34 case '~':
35 case '+':
36 case '@': return true;
37 default: return std::isalnum(byte) != 0;
38 }
39}

67[[nodiscard]] bool validPath(std::string_view path) noexcept {
68 if (path.empty() || path.front() == '/' || path.back() == '/') return false;
69 char previous = '\0';
70 std::size_t segmentStart = 0;
71 for (std::size_t i = 0; i < path.size(); ++i) {
72 const char value = path[i];
73 if (!isPathCharacter(value) || (value == '/' && previous == '/')) return false;
74 if (value == '/') {
75 const auto segment = path.substr(segmentStart, i - segmentStart);
76 if (segment == "." || segment == ".." || segment.empty()) return false;
77 segmentStart = i + 1;
78 }
79 previous = value;
80 }
81 const auto lastSegment = path.substr(segmentStart);
82 return lastSegment != "." && lastSegment != ".." && !lastSegment.empty();
83}

87 return std::all_of(value.begin(), value.end(), isQueryOrFragmentCharacter);

90[[nodiscard]] Result<Uri> parseUri(std::string_view text) {
91 const auto delimiter = text.find("://");
92 if (delimiter == std::string_view::npos || delimiter == 0) {
93 return uriFailure("resource URI must use scheme://path and cannot be a bare path", text);
94 }
95 for (std::size_t i = 0; i < delimiter; ++i) {
96 if (!isSchemeCharacter(text[i], i == 0)) {
97 return uriFailure("resource URI contains an invalid scheme", text);
98 }
99 }
100
101 std::string scheme(text.substr(0, delimiter));
102 std::transform(scheme.begin(), scheme.end(), scheme.begin(),
103 [](unsigned char value) { return static_cast<char>(std::tolower(value)); });
104 const auto schemeKind = schemeFromText(scheme);
105 if (!schemeKind) return uriFailure("unsupported resource URI scheme", text);
106
107 std::string_view remainder = text.substr(delimiter + 3);
108 const auto fragmentDelimiter = remainder.find('#');
109 std::string_view fragment;
110 if (fragmentDelimiter != std::string_view::npos) {
111 fragment = remainder.substr(fragmentDelimiter + 1);
112 remainder = remainder.substr(0, fragmentDelimiter);
113 if (!validQueryOrFragment(fragment)) return uriFailure("URI fragment is empty or invalid", text);
114 }
115
116 const auto queryDelimiter = remainder.find('?');
117 std::string_view query;
118 if (queryDelimiter != std::string_view::npos) {
119 query = remainder.substr(queryDelimiter + 1);
120 remainder = remainder.substr(0, queryDelimiter);
121 if (!validQueryOrFragment(query)) return uriFailure("URI query is empty or invalid", text);
122 }
123 if (!validPath(remainder)) return uriFailure("URI path is empty, absolute, or invalid", text);
124
125 return Uri::fromParts(*schemeKind, remainder, query, fragment);
126}

159Result<Uri> Uri::fromParts(UriScheme scheme, std::string_view path, std::string_view query, std::string_view fragment) {
160 if (!isKnownScheme(scheme)) return uriFailure("unknown resource URI scheme", path);
161 if (!validPath(path)) return uriFailure("URI path is empty, absolute, or invalid", path);
162 if ((!query.empty() && !validQueryOrFragment(query)) || (!fragment.empty() && !validQueryOrFragment(fragment))) {
163 return uriFailure("URI query or fragment is invalid", path);
164 }
165 return Result<Uri>::success(Uri(scheme, std::string(path), std::string(query), std::string(fragment)));
