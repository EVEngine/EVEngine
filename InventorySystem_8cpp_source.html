268void InventorySystem::emit(InventoryChangeEvent ev) {
269 if (!changeHooksSuppressed()) {
270 for (auto &kv : changeHooks()) {
271 if (!kv.second) continue;
272 try {
273 kv.second(ev);
274 } catch (...) {
275 // Hooks are post-commit observers. One faulty observer must not
276 // unwind an already committed authoritative inventory change or
277 // prevent the remaining observers and poll queue from seeing it.
278 }
279 }
280 }
281 eventQueue().push_back(std::move(ev));
282}

327int InventorySystem::freeSpaceInSlot(const Bag &bag, int slot, const ItemDefinition &def) {
328 if (slot < 0 || slot >= bag.getSlotCount()) return 0;
329 const auto &s = bag.slots()[size_t(slot)];
330 int maxStack = def.maxStack > 0 ? def.maxStack : 1;
331 // If the bag's stack rule rejects merging two identical probes, treat every
332 // slot as capacity 1 (e.g. built-in "never").
333 {
334 ItemStack probeA;
335 probeA.itemId = def.id;
336 probeA.quantity = 1;
337 ItemStack probeB = probeA;
338 probeB.instanceId = 1;
339 if (!canStackTogether(bag, probeA, probeB, def)) maxStack = 1;
340 }
341 if (s.empty()) return maxStack;
342 ItemStack probe;
343 probe.itemId = def.id;
344 probe.quantity = 1;
345 if (!canStackTogether(bag, s, probe, def)) return 0;
346 return std::max(0, maxStack - s.quantity);
347}

369int InventorySystem::addItemImpl(Bag *bag, const std::string &itemId, int quantity,
370 bool publish) {
371 if (!bag || quantity <= 0) return 0;
372 const ItemDefinition *def = ItemRegistry::find(itemId);
373 if (!def) return 0;
374 // Accept is all-or-nothing for the requested item type (not quantity-scaled).
375 if (!checkAccept(*bag, *def, 1, nullptr)) return 0;
376
377 int remaining = quantity;
378 int added = 0;
379
380 auto maxFittable = [&](int space) -> int {
381 int lo = 0, hi = std::min(space, remaining);
382 while (lo < hi) {
383 int mid = lo + (hi - lo + 1) / 2;
384 if (checkCapacity(*bag, *def, mid, nullptr))
385 lo = mid;
386 else
387 hi = mid - 1;
388 }
389 return lo;
390 };
391
392 // Pass 1: fill existing stacks.
393 for (int i = 0; i < bag->getSlotCount() && remaining > 0; ++i) {
394 int space = freeSpaceInSlot(*bag, i, *def);
395 auto &s = bag->slots()[size_t(i)];
396 if (s.empty() || space <= 0) continue;
397 int put = maxFittable(space);
398 if (put <= 0) continue;
399 s.quantity += put;
400 remaining -= put;
401 added += put;
402 }
403
404 // Pass 2: empty slots.
405 for (int i = 0; i < bag->getSlotCount() && remaining > 0; ++i) {
406 auto &s = bag->slots()[size_t(i)];
407 if (!s.empty()) continue;
408 int space = freeSpaceInSlot(*bag, i, *def);
409 int put = maxFittable(space);
410 if (put <= 0) continue;
411 s.instanceId = nextInstanceId();
412 s.itemId = def->id;
413 s.quantity = put;
414 s.durability = -1.f;
415 s.props.clear();
416 s.tags.clear();
417 remaining -= put;
418 added += put;
419 }
420
421 if (publish && added > 0) {
422 InventoryChangeEvent ev;
423 ev.action = "add";
424 ev.bagId = bag->getId();
425 ev.itemId = itemId;
426 ev.quantity = added;
427 emit(std::move(ev));
428 }
429 return added;
430}

432eve::Result<PreparedInventoryAdd>
433InventorySystem::prepareAddBatch(Bag *bag, const std::vector<InventoryItemGrant> &grants) {
434 if (!bag)
435 return eve::Result<PreparedInventoryAdd>::failure(eve::Diagnostic::error(
436 eve::DiagnosticCode::InvalidArgument, "inventory batch requires a Bag", "bag"));
437 if (grants.empty())
438 return eve::Result<PreparedInventoryAdd>::failure(eve::Diagnostic::error(
439 eve::DiagnosticCode::InvalidArgument, "inventory batch requires at least one grant", "grants"));
440
441 PreparedInventoryAdd prepared;
442 prepared.state_ = std::make_unique<PreparedInventoryAdd::State>();
443 prepared.state_->target = bag;
444 prepared.state_->baseline = cloneBag(*bag);
445 prepared.state_->candidate = cloneBag(*bag);
446 prepared.state_->events.reserve(grants.size());
447 for (std::size_t index = 0; index < grants.size(); ++index) {
448 const auto &grant = grants[index];
449 if (grant.itemId.empty() || grant.quantity <= 0)
450 return eve::Result<PreparedInventoryAdd>::failure(eve::Diagnostic::error(
451 eve::DiagnosticCode::InvalidArgument, "inventory grant requires an item id and positive quantity",
452 "grants[" + std::to_string(index) + "]"));
453 const int added = addItemImpl(prepared.state_->candidate.get(), grant.itemId, grant.quantity, false);
454 if (added != grant.quantity)
455 return eve::Result<PreparedInventoryAdd>::failure(eve::Diagnostic::error(
456 ItemRegistry::find(grant.itemId) ? eve::DiagnosticCode::PreconditionViolation
457 : eve::DiagnosticCode::NotFound,
458 ItemRegistry::find(grant.itemId) ? "inventory cannot fit the complete reward batch"
459 : "inventory reward references an unknown item",
460 "grants[" + std::to_string(index) + "].itemId", {{"itemId", grant.itemId}},
461 "inventory.prepare-add-batch"));
462 prepared.state_->totalAdded += added;
463 InventoryChangeEvent event;
464 event.action = "add";
465 event.bagId = bag->getId();
466 event.itemId = grant.itemId;
467 event.quantity = added;
468 prepared.state_->events.push_back(std::move(event));
469 }
470 return eve::Result<PreparedInventoryAdd>::success(std::move(prepared),
471 eve::Status::success(eve::StatusCode::Pending));
472}

474eve::Result<int> InventorySystem::commitAddBatch(PreparedInventoryAdd prepared) {
475 if (!prepared.state_ || !prepared.state_->target || !prepared.state_->baseline ||
476 !prepared.state_->candidate)
477 return eve::Result<int>::failure(eve::Diagnostic::error(
478 eve::DiagnosticCode::InvalidArgument, "inventory batch is empty or already consumed", "prepared"));
479 Bag &target = *prepared.state_->target;
480 const Bag &baseline = *prepared.state_->baseline;
481 if (!bagsEqual(target, baseline))
482 return eve::Result<int>::failure(eve::Diagnostic::error(
483 eve::DiagnosticCode::Conflict, "Bag changed after inventory batch preparation", "bag"));
484
485 target.slots_.swap(prepared.state_->candidate->slots_);
486 const int totalAdded = prepared.state_->totalAdded;
487 for (auto &event : prepared.state_->events) emit(std::move(event));
488 prepared.state_.reset();
489 return eve::Result<int>::success(totalAdded, eve::Status::success(eve::StatusCode::Applied));
490}

492eve::Result<PreparedInventoryRemove>
493InventorySystem::prepareRemove(Bag *bag, const std::string &itemId, int quantity) {
494 if (!bag || itemId.empty() || quantity <= 0)
495 return eve::Result<PreparedInventoryRemove>::failure(eve::Diagnostic::error(
496 eve::DiagnosticCode::InvalidArgument,
497 "inventory removal requires a Bag, item id, and positive quantity", "removal"));
498 if (!ItemRegistry::find(itemId))
499 return eve::Result<PreparedInventoryRemove>::failure(eve::Diagnostic::error(
500 eve::DiagnosticCode::NotFound, "inventory removal references an unknown item", "itemId",
501 {{"itemId", itemId}}, "inventory.prepare-remove"));
502 if (countItem(bag, itemId) < quantity)
503 return eve::Result<PreparedInventoryRemove>::failure(eve::Diagnostic::error(
504 eve::DiagnosticCode::PreconditionViolation,
505 "inventory does not contain the complete removal quantity", "quantity",
506 {{"itemId", itemId}}, "inventory.prepare-remove"));
507
508 PreparedInventoryRemove prepared;
509 prepared.state_ = std::make_unique<PreparedInventoryRemove::State>();
510 prepared.state_->target = bag;
511 prepared.state_->baseline = cloneBag(*bag);
512 prepared.state_->candidate = cloneBag(*bag);
513 int remaining = quantity;
514 for (auto &stack : prepared.state_->candidate->slots_) {
515 if (remaining <= 0) break;
516 if (stack.empty() || stack.itemId != itemId) continue;
517 const int take = std::min(stack.quantity, remaining);
518 stack.quantity -= take;
519 remaining -= take;
520 if (stack.quantity <= 0) stack.clear();
521 }
522 prepared.state_->totalRemoved = quantity;
523 prepared.state_->event.action = "remove";
524 prepared.state_->event.bagId = bag->getId();
525 prepared.state_->event.itemId = itemId;
526 prepared.state_->event.quantity = quantity;
527 return eve::Result<PreparedInventoryRemove>::success(
528 std::move(prepared), eve::Status::success(eve::StatusCode::Pending));
529}

589bool InventorySystem::swapSlots(Bag *bag, int slotA, int slotB) {
590 if (!bag || slotA < 0 || slotB < 0 || slotA >= bag->getSlotCount() ||

673int InventorySystem::transfer(Bag *from, Bag *to, const std::string &itemId, int quantity) {
674 if (!from || !to || quantity <= 0 || itemId.empty()) return 0;
675 int available = countItem(from, itemId);
676 int want = std::min(available, quantity);
677 if (want <= 0) return 0;
678
679 for (int n = want; n >= 1; --n) {
680 if (!canAdd(to, itemId, n, nullptr)) continue;
681
682 // Detach from source without emitting per-slot remove events.
683 int remaining = n;
684 for (int i = 0; i < from->getSlotCount() && remaining > 0; ++i) {
685 auto &s = from->slots()[size_t(i)];
686 if (s.empty() || s.itemId != itemId) continue;
687 int take = std::min(s.quantity, remaining);
688 s.quantity -= take;
689 remaining -= take;
690 if (s.quantity <= 0) s.clear();
691 }
692
693 int added = addItem(to, itemId, n);
694 if (added < n) {
695 // Roll back shortfall into source (best-effort).
696 int needRestore = n - added;
697 addItem(from, itemId, needRestore);
698 }
699 if (added > 0) {
700 InventoryChangeEvent ev;
701 ev.action = "transfer";
702 ev.bagId = from->getId();
703 ev.otherBagId = to->getId();
704 ev.itemId = itemId;
705 ev.quantity = added;
706 emit(std::move(ev));
707 }
708 return added;
709 }
710 return 0;
711}

713int InventorySystem::transferSlot(Bag *from, int fromSlot, Bag *to, int quantity) {
714 if (!from || !to || fromSlot < 0 || fromSlot >= from->getSlotCount() || quantity <= 0)
715 return 0;
716 auto &s = from->slots()[size_t(fromSlot)];
717 if (s.empty()) return 0;
718 int want = std::min(s.quantity, quantity);
719 std::string itemId = s.itemId;
720
721 for (int n = want; n >= 1; --n) {
722 if (!canAdd(to, itemId, n, nullptr)) continue;
723
724 // Prefer preserving instance/props when moving a whole stack into an empty target slot.
725 if (n == s.quantity) {
726 int empty = -1;
727 for (int i = 0; i < to->getSlotCount(); ++i) {
728 if (to->slots()[size_t(i)].empty()) {
729 empty = i;
730 break;
731 }
732 }
733 const ItemDefinition *def = ItemRegistry::find(itemId);
734 bool canPlaceWhole = empty >= 0 && def != nullptr;
735 if (canPlaceWhole) {
736 // Also allow merging into an existing compatible stack instead.
737 bool merged = false;
738 for (int i = 0; i < to->getSlotCount(); ++i) {
739 auto &ts = to->slots()[size_t(i)];
740 if (ts.empty()) continue;
741 if (!canStackTogether(*to, s, ts, *def)) continue;
742 int maxStack = def->maxStack > 0 ? def->maxStack : 1;
743 int space = maxStack - ts.quantity;
744 if (space < n) continue;
745 if (!checkCapacity(*to, *def, n, nullptr)) continue;
746 ts.quantity += n;
747 s.clear();
748 InventoryChangeEvent ev;
749 ev.action = "transfer";
750 ev.bagId = from->getId();
751 ev.otherBagId = to->getId();
752 ev.itemId = itemId;
753 ev.quantity = n;
754 ev.slot = fromSlot;
755 ev.otherSlot = i;
756 emit(std::move(ev));
757 merged = true;
758 break;
759 }
760 if (merged) return n;
761
762 if (checkCapacity(*to, *def, n, nullptr) && checkAccept(*to, *def, n, nullptr)) {
763 to->slots()[size_t(empty)] = s;
764 s.clear();
765 InventoryChangeEvent ev;
766 ev.action = "transfer";
767 ev.bagId = from->getId();
768 ev.otherBagId = to->getId();
769 ev.itemId = itemId;
770 ev.quantity = n;
771 ev.slot = fromSlot;
772 ev.otherSlot = empty;
773 emit(std::move(ev));
774 return n;
775 }
776 }
777 }
778
779 s.quantity -= n;
780 if (s.quantity <= 0) s.clear();
781 int added = addItem(to, itemId, n);
782 if (added < n) {
783 // Restore remainder into original slot if possible.
784 auto &fs = from->slots()[size_t(fromSlot)];
785 if (fs.empty()) {
786 fs.itemId = itemId;
787 fs.quantity = n - added;
788 fs.instanceId = nextInstanceId();
789 } else if (fs.itemId == itemId) {
790 fs.quantity += (n - added);
791 } else {
792 addItem(from, itemId, n - added);
793 }
794 }
795 if (added > 0) {
796 InventoryChangeEvent ev;
797 ev.action = "transfer";
798 ev.bagId = from->getId();
799 ev.otherBagId = to->getId();
800 ev.itemId = itemId;
801 ev.quantity = added;
802 ev.slot = fromSlot;
803 emit(std::move(ev));
804 }
805 return added;
806 }
807 return 0;
808}

916bool InventorySystem::unequipToBag(EquipmentSet *eq, const std::string &equipSlot, Bag *bag) {
917 if (!eq || !bag) return false;
918 auto *src = eq->stackAt(equipSlot);
919 if (!src || src->empty()) return false;
920
921 // Find empty slot first to preserve instance/props.
922 int empty = -1;
923 for (int i = 0; i < bag->getSlotCount(); ++i) {
924 if (bag->slots()[size_t(i)].empty()) {
925 empty = i;
926 break;
927 }
928 }
929 if (empty < 0) {
930 // Fall back to addItem (may stack, losing instance identity).
931 if (!canAdd(bag, src->itemId, src->quantity, nullptr)) return false;
932 std::string itemId = src->itemId;
933 int qty = src->quantity;
934 src->clear();

936 if (added < qty) {
937 // restore
938 src->itemId = itemId;
939 src->quantity = qty - added;
940 src->instanceId = nextInstanceId();
941 }
942 InventoryChangeEvent ev;
943 ev.action = "unequip";
944 ev.bagId = bag->getId();
945 ev.itemId = itemId;
946 ev.quantity = added;
947 ev.equipSlot = equipSlot;
948 emit(std::move(ev));
949 return added > 0;
950 }
951
952 bag->slots()[size_t(empty)] = *src;
953 std::string itemId = src->itemId;
954 int qty = src->quantity;
955 src->clear();
956
957 InventoryChangeEvent ev;
958 ev.action = "unequip";
959 ev.bagId = bag->getId();
960 ev.itemId = itemId;
961 ev.quantity = qty;
962 ev.slot = empty;
963 ev.equipSlot = equipSlot;
964 emit(std::move(ev));
