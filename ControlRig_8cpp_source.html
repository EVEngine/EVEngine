25Result<void> validateRigDefinition(const RigDefinition& d) {
26 if (!std::isfinite(d.duration) || d.duration <= 0 || d.duration > 36000 || d.controls.size() > 4096 ||
27 d.nodes.size() > 4096)
28 return error<void>("invalid duration or rig capacity");
29 std::set<std::string> names;
30 for (const auto& c : d.controls) {
31 if (c.id.empty() || !names.insert(c.id).second || !validTRS(c.initial) || !std::isfinite(c.shapeSize) ||
32 c.shapeSize <= 0 || c.keys.size() > 100000)
33 return error<void>("invalid/duplicate control or transform: " + c.id);
34 auto constrained = constrainRigControl(c.limits, c.initial);
35 if (!constrained.ok()) return Result<void>::failure(constrained.status());
36 double previous = -1;
37 for (const auto& k : c.keys) {
38 if (!std::isfinite(k.time) || k.time < 0 || k.time > d.duration || k.time <= previous ||
39 !validTRS(k.transform))
40 return error<void>("invalid or unordered key: " + c.id);
41 previous = k.time;
42 }
43 }
44 auto order = controlOrder(d);
45 if (!order.ok()) return Result<void>::failure(order.status());
46 names.clear();
47 for (const auto& n : d.nodes) {
48 if (n.id.empty() || !names.insert(n.id).second || controlIndex(d, n.control) < 0 || n.bone.empty() ||
49 !std::isfinite(n.weight) || n.weight < 0 || n.weight > 1 ||
50 (n.operation != RigOperation::FK && n.operation != RigOperation::Aim &&
51 n.operation != RigOperation::TwoBoneIK))
52 return error<void>("invalid solve node: " + n.id);
53 if (n.operation == RigOperation::TwoBoneIK && (n.middle.empty() || n.tip.empty()))
54 return error<void>("IK requires middle and tip bones: " + n.id);
55 std::set<std::string> dependencies;
56 for (const auto& dependency : n.after)
57 if (!dependencies.insert(dependency).second) return error<void>("duplicate solve dependency: " + n.id);
58 }
59 std::set<std::string> ready;
60 while (ready.size() < d.nodes.size()) {
61 bool progress = false;
62 for (const auto& n : d.nodes) {
63 if (ready.contains(n.id)) continue;
64 bool available = true;
65 for (const auto& dependency : n.after) {
66 if (!names.contains(dependency)) return error<void>("missing solve dependency: " + dependency);
67 if (!ready.contains(dependency)) available = false;
68 }
69 if (available) {
70 ready.insert(n.id);
71 progress = true;
72 }
73 }
74 if (!progress) return error<void>("solve graph cycle");
75 }
76 return Result<void>::success();
77}

79Result<std::unique_ptr<ControlRig>> ControlRig::compile(const RigDefinition& d, const AnimSkeleton& skeleton) {
80 auto valid = validateRigDefinition(d);
81 if (!valid.ok()) return Result<std::unique_ptr<ControlRig>>::failure(valid.status());
82 if (skeleton.getBoneCount() <= 0) return error<std::unique_ptr<ControlRig>>("empty skeleton");
83 auto p = std::make_unique<Impl>();
84 p->definition = d;
85 for (auto& c : p->definition.controls) {
86 auto initial = constrainRigControl(c.limits, c.initial);
87 if (!initial.ok()) return Result<std::unique_ptr<ControlRig>>::failure(initial.status());
88 c.initial = initial.value();
89 for (auto& key : c.keys) {
90 auto projected = constrainRigControl(c.limits, key.transform);
91 if (!projected.ok()) return Result<std::unique_ptr<ControlRig>>::failure(projected.status());
92 key.transform = projected.value();
93 }
94 }
95 std::set<std::string> boneNames;
96 for (int i = 0; i < skeleton.getBoneCount(); ++i) {
97 const auto name = skeleton.getBoneName(i);
98 const int parent = skeleton.getParent(i);
99 const auto& t = skeleton.bindLocal(i);
100 if (!boneNames.insert(name).second || parent >= i || parent < -1 || !validTRS(t))
101 return error<std::unique_ptr<ControlRig>>(
102 "skeleton requires unique names, ordered parents and uniform positive TRS");
103 const int index = p->skeleton.addBone(name, parent);
104 p->skeleton.setBindPosition(index, t.px, t.py, t.pz);
105 p->skeleton.setBindRotation(index, t.qx, t.qy, t.qz, t.qw);
106 p->skeleton.setBindScale(index, t.sx, t.sy, t.sz);
107 }
108 auto order = controlOrder(d);
109 if (!order.ok()) return Result<std::unique_ptr<ControlRig>>::failure(order.status());
110 p->order = std::move(order).takeValue();
111 for (const auto& c : d.controls) p->parents.push_back(controlIndex(d, c.parent));
112 p->jointControls.assign(static_cast<size_t>(skeleton.getBoneCount()), -1);
113 for (const auto& n : d.nodes) {
114 if (n.operation != RigOperation::FK) continue;
115 const int ci = controlIndex(d, n.control), bone = skeleton.findBone(n.bone);
116 if (bone < 0) continue;
117 const auto& c = d.controls[static_cast<size_t>(ci)];
118 const auto& l = c.limits;
119 const bool active = l.extension || l.scale.mode != RigAxisMode::Free ||
120 std::any_of(l.translation.begin(), l.translation.end(),
121 [](const auto& a) { return a.mode != RigAxisMode::Free; }) ||
122 std::any_of(l.rotation.begin(), l.rotation.end(),
123 [](const auto& a) { return a.mode != RigAxisMode::Free; });
124 if (!active) continue;
125 const int parent = skeleton.getParent(bone);
126 const int pc = controlIndex(d, c.parent);
127 bool matched = parent < 0 && pc < 0;
128 for (const auto& pn : d.nodes)
129 if (parent >= 0 && pn.operation == RigOperation::FK && pn.bone == skeleton.getBoneName(parent) &&
130 pn.control == c.parent && pn.weight == 1)
131 matched = true;
132 const auto bindings = std::count_if(d.nodes.begin(), d.nodes.end(), [&](const auto& other) {
133 return other.operation == RigOperation::FK && (other.bone == n.bone || other.control == n.control);
134 });
135 if (!matched || n.weight != 1 || bindings != 1 || p->jointControls[static_cast<size_t>(bone)] >= 0)
136 return error<std::unique_ptr<ControlRig>>(
137 "limited joints need unique full-weight FK and matching parent hierarchy");
138 p->jointControls[static_cast<size_t>(bone)] = ci;
139 }
140 std::set<std::string> ready;
141 while (ready.size() < d.nodes.size()) {
142 for (const auto& n : d.nodes) {
143 if (ready.contains(n.id)) continue;
144 if (!std::all_of(n.after.begin(), n.after.end(), [&](const auto& id) { return ready.contains(id); }))
145 continue;
146 const int bone = skeleton.findBone(n.bone), middle = skeleton.findBone(n.middle),
147 tip = skeleton.findBone(n.tip);
148 if (bone < 0) return error<std::unique_ptr<ControlRig>>("missing bone: " + n.bone);
149 if (n.operation == RigOperation::TwoBoneIK &&
150 (middle < 0 || tip < 0 || skeleton.getParent(middle) != bone || skeleton.getParent(tip) != middle))
151 return error<std::unique_ptr<ControlRig>>("IK requires direct root-middle-tip chain: " + n.id);
152 p->nodes.push_back({n.operation, controlIndex(d, n.control), bone, middle, tip, n.weight});
153 ready.insert(n.id);
154 }
155 }
156 return Result<std::unique_ptr<ControlRig>>::success(std::unique_ptr<ControlRig>(new ControlRig(std::move(p))));
157}

159Result<std::vector<TransformTRS>> ControlRig::sampleControls(double seconds) const {
160 if (!std::isfinite(seconds)) return error<std::vector<TransformTRS>>("nonfinite sample time");
161 seconds = std::clamp(seconds, 0.0, impl_->definition.duration);
162 std::vector<TransformTRS> result;
163 for (const auto& c : impl_->definition.controls) {
164 if (c.keys.empty()) {
165 result.push_back(c.initial);
166 } else {
167 const auto upper = std::upper_bound(c.keys.begin(), c.keys.end(), seconds,
168 [](double t, const RigKey& k) { return t < k.time; });
169 if (upper == c.keys.begin())
170 result.push_back(upper->transform);
171 else if (upper == c.keys.end())
172 result.push_back(c.keys.back().transform);
173 else {
174 const auto& a = *(upper - 1);
175 result.push_back(blendTRS(a.transform, upper->transform,
176 static_cast<float>((seconds - a.time) / (upper->time - a.time))));
177 }
178 }
179 auto projected = constrainRigControl(c.limits, result.back());
180 if (!projected.ok()) return Result<std::vector<TransformTRS>>::failure(projected.status());
181 result.back() = projected.value();
182 }
183 return Result<std::vector<TransformTRS>>::success(std::move(result));
184}

186Result<std::vector<TransformTRS>> ControlRig::controlWorlds(const std::vector<TransformTRS>& locals) const {
187 if (locals.size() != impl_->parents.size()) return error<std::vector<TransformTRS>>("control count mismatch");
188 for (const auto& t : locals)
189 if (!validTRS(t)) return error<std::vector<TransformTRS>>("invalid control transform");
190 auto worlds = locals;
191 auto constrained = locals;
192 for (size_t i = 0; i < locals.size(); ++i) {
193 auto projected = constrainRigControl(impl_->definition.controls[i].limits, locals[i]);
194 if (!projected.ok()) return Result<std::vector<TransformTRS>>::failure(projected.status());
195 worlds[i] = constrained[i] = projected.value();
196 }
197 for (int i : impl_->order) {
198 const int parent = impl_->parents[static_cast<size_t>(i)];
199 if (parent >= 0)
200 worlds[static_cast<size_t>(i)] =
201 detail::mulTRS(worlds[static_cast<size_t>(parent)], constrained[static_cast<size_t>(i)]);
202 if (!validTRS(worlds[static_cast<size_t>(i)]))
203 return error<std::vector<TransformTRS>>("control hierarchy overflow");
204 }
205 return Result<std::vector<TransformTRS>>::success(std::move(worlds));
206}

208Result<void> ControlRig::evaluate(double seconds, const AnimPose& input, AnimPose& output,
209 const std::vector<TransformTRS>& overrides) const {
210 if (!std::isfinite(seconds) || input.getBoneCount() != impl_->skeleton.getBoneCount())
211 return error<void>("pose/time mismatch");
212 for (int i = 0; i < input.getBoneCount(); ++i)
213 if (!validTRS(input.local(i))) return error<void>("invalid input bone TRS");
214 auto sampled = overrides.empty() ? sampleControls(seconds) : Result<std::vector<TransformTRS>>::success(overrides);
215 if (!sampled.ok()) return Result<void>::failure(sampled.status());
216 auto worlds = controlWorlds(sampled.value());
217 if (!worlds.ok()) return Result<void>::failure(worlds.status());
218 AnimPose candidate;
219 candidate.copyFrom(&input);
220 candidate.computeWorld(&impl_->skeleton);
221 for (int i = 0; i < candidate.getBoneCount(); ++i)
222 if (!validTRS(candidate.world(i))) return error<void>("input hierarchy overflow");
223 for (const auto& n : impl_->nodes) {
224 if (n.weight == 0) continue;
225 const auto& target = worlds.value()[static_cast<size_t>(n.control)];
226 if (n.operation == RigOperation::FK) {
227 const int parent = impl_->skeleton.getParent(n.bone);
228 auto local = parent < 0 ? target : relative(candidate.world(parent), target);
229 candidate.local(n.bone) = blendTRS(candidate.local(n.bone), local, n.weight);
230 } else if (n.operation == RigOperation::Aim) {
231 if (!candidate.aimBone(&impl_->skeleton, n.bone, target.px, target.py, target.pz, n.weight))
232 return error<void>("aim target coincides with bone");
233 } else if (!candidate.solveTwoBoneIK(&impl_->skeleton, n.bone, n.middle, n.tip, target.px, target.py, target.pz,
234 n.weight)) {
235 return error<void>("degenerate IK chain");
236 }
237 for (int bone = 0; bone < candidate.getBoneCount(); ++bone) {
238 const int ci = impl_->jointControls[static_cast<size_t>(bone)];
239 if (ci < 0) continue;
240 auto projected =
241 constrainRigControl(impl_->definition.controls[static_cast<size_t>(ci)].limits, candidate.local(bone));
242 if (!projected.ok()) return Result<void>::failure(projected.status());
243 candidate.local(bone) = projected.value();
244 }
245 candidate.computeWorld(&impl_->skeleton);
246 }
247 for (int i = 0; i < candidate.getBoneCount(); ++i)
248 if (!validTRS(candidate.local(i)) || !validTRS(candidate.world(i)))
249 return error<void>("solve produced invalid transform");
250 output = std::move(candidate);
251 return Result<void>::success();
252}

298Result<std::unique_ptr<AnimClip>> ControlRig::bake(double rate, const AnimPose& input) const {
299 if (!std::isfinite(rate) || rate <= 0 || rate > 240)
300 return error<std::unique_ptr<AnimClip>>("sample rate must be in (0,240]");
301 const double frameCount = std::ceil(impl_->definition.duration * rate);
302 if ((frameCount + 1) * impl_->skeleton.getBoneCount() > 2000000)
303 return error<std::unique_ptr<AnimClip>>("bake exceeds two million bone samples");
304 auto clip = std::make_unique<AnimClip>("ControlRig bake");
305 clip->setDuration(static_cast<float>(impl_->definition.duration));
306 clip->setLoop(false);
307 clip->setSampleRate(static_cast<float>(rate));
308 AnimPose pose;
309 for (int frame = 0; frame <= static_cast<int>(frameCount); ++frame) {
310 const double time = std::min(static_cast<double>(frame) / rate, impl_->definition.duration);
311 auto evaluated = evaluate(time, input, pose);
312 if (!evaluated.ok()) return Result<std::unique_ptr<AnimClip>>::failure(evaluated.status());
313 for (int bone = 0; bone < pose.getBoneCount(); ++bone) {
314 const auto& t = pose.local(bone);
315 const float seconds = static_cast<float>(time);
316 clip->addPositionKey(bone, seconds, t.px, t.py, t.pz);
317 clip->addRotationKey(bone, seconds, t.qx, t.qy, t.qz, t.qw);
318 clip->addScaleKey(bone, seconds, t.sx, t.sy, t.sz);
319 }
320 }
321 return Result<std::unique_ptr<AnimClip>>::success(std::move(clip));
322}
