23 character == '/')
24 character = '_';

34std::string valueText(const Value &value) {
35 if (const auto *text = value.getIf<std::string>()) return *text;
36 if (const auto *boolean = value.getIf<bool>()) return *boolean ? "true" : "false";
37 if (const auto *integer = value.getIf<std::int64_t>()) return std::to_string(*integer);
38 if (const auto *number = value.getIf<double>()) return std::to_string(*number);
39 if (const auto *array = value.getIf<Value::Array>())
40 return std::to_string(array->size()) + " items";
41 if (const auto *object = value.getIf<Value::Object>())
42 return std::to_string(object->size()) + " fields";
43 return "null";
44}

68WidgetDesc makePropertyField(property_access::IPropertyAccess &model, const PropertyViewOptions &options,
69 const PropertyDescriptor &property, const Value &value) {
70 const std::string id = propertyWidgetId(options, property.path);
71 const std::string label = displayName(property);
72 const bool readOnly =
73 property_access::hasFlag(property.flags, PropertyFlag::ReadOnly) || property.kind == PropertyKind::ReadOnlyText;
74 WidgetDesc result;
75
76 if (readOnly) {
77 result = text(label + ": " + valueText(value), id);
78 } else {
79 switch (property.kind) {
80 case PropertyKind::Bool: {
81 const bool current = value.getIf<bool>() ? *value.getIf<bool>() : false;
82 const std::string path = property.path;
83 result = checkbox(label, current, id, [&model, path](bool next) {
84 model.write(path, Value(next));
85 });
86 break;
87 }
88 case PropertyKind::Integer:
89 case PropertyKind::Number: {
90 const bool integer = property.kind == PropertyKind::Integer;
91 const std::string path = property.path;
92 if (property.numeric.minimum && property.numeric.maximum) {
93 result = slider(label, static_cast<float>(numericValue(value)),
94 static_cast<float>(*property.numeric.minimum),
95 static_cast<float>(*property.numeric.maximum), id,
96 [&model, path, integer](float next) {
97 if (integer)
98 model.write(path, Value(static_cast<std::int64_t>(next)));
99 else
100 model.write(path, Value(static_cast<double>(next)));
101 });
102 } else {
103 result = inputText(label, valueText(value), id,
104 [&model, path, integer](const std::string &next) {
105 if (integer) {
106 std::int64_t parsed = 0;
107 const auto converted = std::from_chars(
108 next.data(), next.data() + next.size(), parsed);
109 if (converted.ec == std::errc() &&
110 converted.ptr == next.data() + next.size())
111 model.write(path, Value(parsed));
112 } else {
113 char *end = nullptr;
114 errno = 0;
115 const double parsed = std::strtod(next.c_str(), &end);
116 if (!next.empty() && errno != ERANGE &&
117 end == next.c_str() + next.size())
118 model.write(path, Value(parsed));
119 }
120 });
121 }
122 break;
123 }
124 case PropertyKind::Enum: {
125 const std::string selected = valueText(value);
126 int selectedIndex = 0;
127 for (std::size_t index = 0; index < property.choices.size(); ++index)
128 if (property.choices[index] == selected) selectedIndex = static_cast<int>(index);
129 const std::string path = property.path;
130 const std::vector<std::string> choices = property.choices;
131 result = combo(label, choices, selectedIndex, id,
132 [&model, path, choices](float next) {
133 const int index = static_cast<int>(next);
134 if (index >= 0 && static_cast<std::size_t>(index) < choices.size())
135 model.write(path, Value(choices[static_cast<std::size_t>(index)]));
136 });
137 break;
138 }
139 case PropertyKind::Action: {
140 const std::string path = property.path;
141 result = button(label, id, [&model, path]() { model.write(path, Value(true)); });
142 break;
143 }
144 case PropertyKind::String:
145 case PropertyKind::AssetRef:
146 case PropertyKind::ObjectRef:
147 case PropertyKind::Auto: {
148 const std::string path = property.path;
149 result = inputText(label, valueText(value), id,
150 [&model, path](const std::string &next) {
151 model.write(path, Value(next));
152 });
153 break;
154 }
155 case PropertyKind::Color:
156 case PropertyKind::Vec2:
157 case PropertyKind::Vec3:
158 case PropertyKind::Vec4:
159 case PropertyKind::Struct:
160 case PropertyKind::Array:
161 case PropertyKind::Map:
162 case PropertyKind::ReadOnlyText:
163 result = text(label + ": " + valueText(value), id);
164 break;
165 }
166 }
167 result.tooltip = property.description;
168 result.accessibilityRole = accessibilityRole(property.kind, readOnly);
169 result.accessibilityName = label;
170 result.accessibilityDescription = property.description;
171 if (readOnly) {
172 result.focusMode = FocusMode::None;
173 result.mouseFilter = MouseFilter::Ignore;
174 }
175 return result;
176}

178bool visible(const PropertyDescriptor &property, const PropertyViewOptions &options) {
179 if (!options.showAdvanced && property_access::hasFlag(property.flags, PropertyFlag::Advanced)) return false;
180 if (!options.showEditorOnly && property_access::hasFlag(property.flags, PropertyFlag::EditorOnly)) return false;
181 if (!options.showReadOnly && (property_access::hasFlag(property.flags, PropertyFlag::ReadOnly) ||
182 property.kind == PropertyKind::ReadOnlyText))
183 return false;
184 return true;
185}

193WidgetDesc buildPropertyField(property_access::IPropertyAccess &model, const std::string &path,
194 const PropertyViewOptions &options) {

205WidgetDesc buildPropertyView(property_access::IPropertyAccess &model, const PropertyViewOptions &options) {
206 std::vector<WidgetDesc> children;
207 if (!options.title.empty()) children.push_back(sectionHeader(options.title, options.idPrefix + "title"));
208
209 std::string category;
210 for (const PropertyDescriptor &property : model.schema().properties) {
211 if (!visible(property, options)) continue;
212 const std::optional<Value> value = model.read(property.path);
213 if (!value) continue;
214 if (options.groupCategories && !property.category.empty() && property.category != category) {
215 category = property.category;
216 children.push_back(sectionHeader(category, options.idPrefix + "category/" + sanitize(category)));
217 }
218 children.push_back(makePropertyField(model, options, property, *value));
219 }
220 return column(std::move(children), options.idPrefix + "root");
221}

223void syncPropertyView(UIHost &host, const property_access::IPropertyAccess &model, const PropertyViewOptions &options) {
224 for (const PropertyDescriptor &property : model.schema().properties) {
225 const std::optional<Value> value = model.read(property.path);
226 if (!value) continue;
227 const std::string id = propertyWidgetId(options, property.path);
228 const bool readOnly = property_access::hasFlag(property.flags, PropertyFlag::ReadOnly) ||
229 property.kind == PropertyKind::ReadOnlyText;
230 if (readOnly) {
231 host.setTextById(id, displayName(property) + ": " + valueText(*value));
232 } else if (property.kind == PropertyKind::Bool) {
233 if (const auto *boolean = value->getIf<bool>()) host.setCheckedById(id, *boolean);
234 } else if (property.kind == PropertyKind::Enum) {
235 const std::string selected = valueText(*value);
236 const auto found = std::find(property.choices.begin(), property.choices.end(), selected);
237 host.setValueById(id, found == property.choices.end()
238 ? 0.f
239 : static_cast<float>(found - property.choices.begin()));
240 } else if (property.kind == PropertyKind::Integer ||
241 property.kind == PropertyKind::Number) {
242 if (property.numeric.minimum && property.numeric.maximum)
243 host.setValueById(id, static_cast<float>(numericValue(*value)));
244 else
245 host.setValueTextById(id, valueText(*value));
246 } else if (const auto *textValue = value->getIf<std::string>()) {
247 host.setValueTextById(id, *textValue);
248 }
249 }
