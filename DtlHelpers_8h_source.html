49inline void placeSpawnAndStairs(Grid2D &grid, uint32_t seed) {
50 std::vector<std::pair<int, int>> walkable;
51 walkable.reserve(size_t(grid.getWidth() * grid.getHeight() / 4));
52 for (int y = 0; y < grid.getHeight(); ++y) {
53 for (int x = 0; x < grid.getWidth(); ++x) {
54 const uint32_t c = uint32_t(grid.getCell(x, y));
55 if (c == Semantic::Floor || c == Semantic::Corridor || c == Semantic::Grass ||
56 c == Semantic::Dirt || c == Semantic::Sand) {
57 walkable.emplace_back(x, y);
58 }
59 }
60 }
61 if (walkable.empty()) return;
62
63 auto pick = [&](uint32_t salt) -> std::pair<int, int> {
64 const uint32_t idx = (seed * 1664525u + salt * 1013904223u) % uint32_t(walkable.size());
65 return walkable[idx];
66 };
67 auto a = pick(1);
68 auto b = pick(7);
69 if (a == b && walkable.size() > 1) b = walkable[(size_t(seed) + 1) % walkable.size()];
70
71 grid.clearObjects();
72 grid.addObjectAt("spawn", "spawn", float(a.first), float(a.second));
73 grid.addObjectAt("stairs", "stairs", float(b.first), float(b.second));
74}
