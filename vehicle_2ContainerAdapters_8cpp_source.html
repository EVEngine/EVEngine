27[[nodiscard]] std::string quote(std::string_view value) {
28 std::string result;
29 result.reserve(value.size() + 2);
30 result.push_back('"');
31 for (const unsigned char byte : value) {
32 switch (byte) {
33 case '"': result += "\\\""; break;
34 case '\\': result += "\\\\"; break;
35 case '\n': result += "\\n"; break;
36 case '\r': result += "\\r"; break;
37 case '\t': result += "\\t"; break;
38 default:
39 if (byte < 0x20u) {
40 result += "\\u00";
41 constexpr char digits[] = "0123456789abcdef";
42 result.push_back(digits[(byte >> 4u) & 0x0fu]);
43 result.push_back(digits[byte & 0x0fu]);
44 } else {
45 result.push_back(static_cast<char>(byte));
46 }
47 }
48 }
49 result.push_back('"');
50 return result;
51}

53[[nodiscard]] int parseOccupant(const eve::container::ContainerObject& object) noexcept {
54 if (object.type != "vehicle.occupant" || object.quantity != 1u) return 0;
55 if (object.id.value().size() <= kOccupantPrefix.size() ||
56 object.id.value().compare(0, kOccupantPrefix.size(), kOccupantPrefix) != 0)
57 return 0;
58 const auto text = object.id.value().substr(kOccupantPrefix.size());
59 int value = 0;
60 const auto parsed = std::from_chars(text.data(), text.data() + text.size(), value);
61 if (parsed.ec != std::errc{} || parsed.ptr != text.data() + text.size() || value <= 0) return 0;
62 return value;
63}

74[[nodiscard]] bool sameLayout(const eve::container::ContainerSnapshot& lhs,
75 const eve::container::ContainerSnapshot& rhs) {
76 if (lhs.id != rhs.id || lhs.revision != rhs.revision || lhs.entries.size() != rhs.entries.size()) return false;
77 for (const auto& left : lhs.entries) {
78 const auto right = std::find_if(rhs.entries.begin(), rhs.entries.end(), [&](const auto& entry) {
79 return entry.membership.object == left.membership.object;
80 });
81 if (right == rhs.entries.end() || left.membership.slot != right->membership.slot ||
82 left.membership.generation != right->membership.generation || left.object.id != right->object.id ||

140class VehicleSeatContainerAdapter::PreparedState final : public eve::container::IContainer::PreparedState {
141public:
142 PreparedState(VehicleSeatContainerAdapter& owner, std::vector<std::optional<int>> staged, eve::Revision revision)
143 : owner_(owner), staged_(std::move(staged)), revision_(revision) {}
144
145 void commit() noexcept override {
146 if (committed_) return;
147 auto* vehicle = owner_.vehicle();
148 auto* seats = seatsOf(vehicle);
149 if (seats == nullptr || seats->list.size() != staged_.size()) std::terminate();
150 for (std::size_t index = 0; index < staged_.size(); ++index) {
151 auto& seat = seats->list[index];
152 seat.occupied = staged_[index].has_value();
153 seat.occupant = staged_[index].value_or(0);
154 }
155 seats->revision = revision_;
156 committed_ = true;
157 }
158
159 void rollback() noexcept override {
160 if (!committed_) staged_.clear();
161 }
162
163private:
164 VehicleSeatContainerAdapter& owner_;
165 std::vector<std::optional<int>> staged_;
166 eve::Revision revision_;
167 bool committed_ = false;
168};

209eve::Result<eve::container::ContainerSnapshot> VehicleSeatContainerAdapter::snapshot() const {
210 const auto* value = vehicle();
211 const auto* seats = seatsOf(value);
212 if (!descriptor_.id.isValid() || seats == nullptr)
213 return eve::Result<eve::container::ContainerSnapshot>::failure(
214 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "vehicle seat adapter is stale or unbound"));
215 eve::container::ContainerSnapshot result{descriptor_.id, seats->revision, {}};
216 std::unordered_set<eve::container::MembershipId> identities;
217 result.entries.reserve(seats->list.size());
218 for (std::size_t index = 0; index < seats->list.size(); ++index) {
219 const auto& seat = seats->list[index];
220 if (!seat.occupied) continue;
221 if (seat.occupant <= 0 || index > static_cast<std::size_t>(std::numeric_limits<std::int32_t>::max()))
222 return eve::Result<eve::container::ContainerSnapshot>::failure(eve::Diagnostic::error(
223 eve::DiagnosticCode::InvariantViolation, "vehicle seat contains an invalid occupant"));
224 const auto id = occupantId(seat.occupant);
225 if (!identities.insert(id).second)
226 return eve::Result<eve::container::ContainerSnapshot>::failure(
227 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "vehicle occupant appears in multiple seats"));
228 result.entries.push_back({{id, eve::container::SlotIndex(static_cast<std::int32_t>(index)), eve::Generation(1)},
229 describe(seat.occupant, seat.driver)});
230 }
231 return eve::Result<eve::container::ContainerSnapshot>::success(std::move(result));
232}

234eve::Result<void> VehicleSeatContainerAdapter::validateInsert(
235 const eve::container::ContainerObject& object, std::optional<eve::container::SlotIndex> destination,
236 std::optional<eve::container::MembershipId> ignoredObject) const {
237 const auto* value = vehicle();
238 const auto* seats = seatsOf(value);
239 if (!descriptor_.id.isValid() || seats == nullptr)
240 return eve::Result<void>::failure(
241 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "vehicle seat adapter is stale or unbound"));
242 if (!object.id.isValid() || object.type != "vehicle.occupant" || object.quantity != 1u)
243 return eve::Result<void>::failure(
244 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "vehicle seat object facts are invalid"));
245 const int occupant = payloadOccupant(object);
246 if (occupant <= 0)
247 return eve::Result<void>::failure(
248 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "vehicle occupant identity is stale"));
249
250 const auto occupiedBy = [&](std::size_t index) {
251 const auto& seat = seats->list[index];
252 return seat.occupied && occupantId(seat.occupant) == object.id;
253 };
254 for (std::size_t index = 0; index < seats->list.size(); ++index) {
255 if (occupiedBy(index) && (!ignoredObject || *ignoredObject != object.id))
256 return eve::Result<void>::failure(
257 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "vehicle occupant is already seated"));
258 }
259 if (destination) {
260 if (!destination->isValid() || static_cast<std::size_t>(destination->value()) >= seats->list.size())
261 return eve::Result<void>::failure(
262 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "vehicle seat slot is out of range"));
263 const auto& seat = seats->list[static_cast<std::size_t>(destination->value())];
264 if (seat.occupied && (!ignoredObject || occupantId(seat.occupant) != *ignoredObject))
265 return eve::Result<void>::failure(
266 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "vehicle seat slot is occupied"));
267 }
268 auto accepted = descriptor_.filter.evaluate(object);
269 if (!accepted) return accepted;
270 std::size_t count = 0;
271 for (const auto& seat : seats->list)
272 if (seat.occupied && (!ignoredObject || occupantId(seat.occupant) != *ignoredObject)) ++count;
273 if (count >= seats->list.size())
274 return eve::Result<void>::failure(
275 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "vehicle has no free seat"));
276 return eve::Result<void>::success();
277}

279eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>> VehicleSeatContainerAdapter::prepare(
280 const eve::container::ContainerSnapshot& expected, const eve::container::ContainerSnapshot& candidate) {
281 const auto* value = vehicle();
282 const auto* seats = seatsOf(value);
283 if (!descriptor_.id.isValid() || seats == nullptr)
284 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
285 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "vehicle seat adapter is stale or unbound"));
286 if (expected.id != descriptor_.id || candidate.id != descriptor_.id)
287 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(eve::Diagnostic::error(
288 eve::DiagnosticCode::InvalidArgument, "vehicle seat prepare targets another container"));
289 auto current = snapshot();
290 if (!current)
291 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(current.status());
292 if (!sameLayout(current.value(), expected))
293 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
294 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "vehicle seat expected snapshot is stale"));
295 const auto next = expected.revision.incremented();
296 if (!next || candidate.revision != *next)
297 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(eve::Diagnostic::error(
298 eve::DiagnosticCode::InvariantViolation, "vehicle seat candidate revision is not the next revision"));
299 if (candidate.entries.size() > seats->list.size())
300 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
301 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "vehicle seat candidate exceeds capacity"));
302
303 std::vector<std::optional<int>> staged(seats->list.size());
304 std::unordered_set<eve::container::MembershipId> identities;
305 for (const auto& entry : candidate.entries) {
306 if (!entry.membership.object.isValid() || !entry.membership.slot.isValid() ||
307 static_cast<std::size_t>(entry.membership.slot.value()) >= staged.size() ||
308 entry.membership.generation.isZero() || entry.object.id != entry.membership.object ||
309 entry.object.type != "vehicle.occupant" || entry.object.quantity != 1u ||
310 !identities.insert(entry.membership.object).second)
311 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
312 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
313 "vehicle seat candidate contains invalid membership"));
314 const int occupant = payloadOccupant(entry.object);
315 if (occupant <= 0)
316 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
317 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "vehicle seat candidate payload is stale"));
318 auto& slot = staged[static_cast<std::size_t>(entry.membership.slot.value())];
319 if (slot)
320 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
321 eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
322 "vehicle seat candidate contains duplicate slots"));
323 slot = occupant;
324 }
325 std::unique_ptr<eve::container::IContainer::PreparedState> prepared =
326 std::make_unique<PreparedState>(*this, std::move(staged), *next);
327 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::success(std::move(prepared));
328}

330eve::Result<eve::container::MembershipId> VehicleSeatContainerAdapter::enter(eve::container::SlotIndex seatIndex,
331 int occupant,
332 eve::container::GameEventSink sink,
333 eve::SimulationTick tick) {
334 const auto serial = nextEventSerial();
335 const auto objectIdValue = occupantId(occupant);
336 auto current = snapshot();
337 if (!current) {
338 const auto status = current.status();
339 bool ignored = false;
340 const auto* diagnostic = status.primaryDiagnostic();
341 notify(sink, descriptor_.id, objectIdValue, eve::container::ContainerEventKind::Rejected, seatIndex, revision(),
342 serial, tick, diagnostic == nullptr ? "stale_container" : diagnostic->message(), ignored);
343 return eve::Result<eve::container::MembershipId>::failure(status);
344 }
345 auto object = describe(occupant);
346 auto accepted = validateInsert(object, seatIndex, std::nullopt);
347 if (!accepted) {
348 const auto status = accepted.status();
349 bool ignored = false;
350 const auto* diagnostic = status.primaryDiagnostic();
351 notify(sink, descriptor_.id, object.id, eve::container::ContainerEventKind::Rejected, seatIndex,
352 current.value().revision, serial, tick, diagnostic == nullptr ? "rejected" : diagnostic->message(),
353 ignored);
354 return eve::Result<eve::container::MembershipId>::failure(status);
355 }
356 auto candidate = current.value();
357 const auto next = candidate.revision.incremented();
358 if (!next) {
359 const auto status = eve::Status::failure(
360 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation, "vehicle seat revision exhausted"));
361 bool ignored = false;
362 notify(sink, descriptor_.id, object.id, eve::container::ContainerEventKind::Rejected, seatIndex,
363 candidate.revision, serial, tick, "revision_exhausted", ignored);
364 return eve::Result<eve::container::MembershipId>::failure(status);
365 }
366 candidate.revision = *next;
367 candidate.entries.push_back({{object.id, seatIndex, eve::Generation(1)}, std::move(object)});
368 auto prepared = prepare(current.value(), candidate);
369 if (!prepared) {
370 const auto status = prepared.status();
371 bool ignored = false;
372 const auto* diagnostic = status.primaryDiagnostic();
373 notify(sink, descriptor_.id, objectIdValue, eve::container::ContainerEventKind::Rejected, seatIndex,
374 current.value().revision, serial, tick, diagnostic == nullptr ? "prepare_failed" : diagnostic->message(),
375 ignored);
376 return eve::Result<eve::container::MembershipId>::failure(status);
377 }
378 auto staged = std::move(prepared).takeValue();
379 staged->commit();
380 bool callbackFailed = false;
381 notify(sink, descriptor_.id, objectIdValue, eve::container::ContainerEventKind::Accepted, seatIndex, *next, serial,
382 tick, {}, callbackFailed);
383 notify(sink, descriptor_.id, objectIdValue, eve::container::ContainerEventKind::Enter, seatIndex, *next, serial,
384 tick, {}, callbackFailed);
385 return eve::Result<eve::container::MembershipId>::success(objectIdValue, appliedStatus(callbackFailed));
386}

389 eve::container::GameEventSink sink,
390 eve::SimulationTick tick) {
391 const auto serial = nextEventSerial();
392 auto current = snapshot();
393 eve::container::MembershipId objectIdValue;
394 if (current) {
395 for (const auto& entry : current.value().entries)
396 if (entry.membership.slot == seatIndex) objectIdValue = entry.membership.object;
397 }
398 if (!current || !objectIdValue.isValid()) {
399 const auto status = current ? eve::Status::failure(eve::Diagnostic::error(eve::DiagnosticCode::NotFound,
400 "vehicle seat is already empty"))
401 : current.status();
402 bool ignored = false;
403 const auto* diagnostic = status.primaryDiagnostic();
404 notify(sink, descriptor_.id, objectIdValue, eve::container::ContainerEventKind::Rejected, seatIndex, revision(),
405 serial, tick, diagnostic == nullptr ? "rejected" : diagnostic->message(), ignored);
406 return eve::Result<eve::container::MembershipId>::failure(status);
407 }
408 auto candidate = current.value();
409 candidate.entries.erase(std::remove_if(candidate.entries.begin(), candidate.entries.end(),
410 [&](const auto& entry) { return entry.membership.slot == seatIndex; }),
411 candidate.entries.end());
412 const auto next = candidate.revision.incremented();
413 if (!next) {
414 const auto status = eve::Status::failure(
415 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation, "vehicle seat revision exhausted"));
416 bool ignored = false;
417 notify(sink, descriptor_.id, objectIdValue, eve::container::ContainerEventKind::Rejected, seatIndex,
418 current.value().revision, serial, tick, "revision_exhausted", ignored);
419 return eve::Result<eve::container::MembershipId>::failure(status);
420 }
421 candidate.revision = *next;
422 auto prepared = prepare(current.value(), candidate);
423 if (!prepared) {
424 const auto status = prepared.status();
425 bool ignored = false;
426 const auto* diagnostic = status.primaryDiagnostic();
427 notify(sink, descriptor_.id, objectIdValue, eve::container::ContainerEventKind::Rejected, seatIndex,
428 current.value().revision, serial, tick, diagnostic == nullptr ? "prepare_failed" : diagnostic->message(),
429 ignored);
430 return eve::Result<eve::container::MembershipId>::failure(status);
431 }
432 auto staged = std::move(prepared).takeValue();
433 staged->commit();
434 bool callbackFailed = false;
435 notify(sink, descriptor_.id, objectIdValue, eve::container::ContainerEventKind::Accepted, seatIndex, *next, serial,
436 tick, {}, callbackFailed);
437 notify(sink, descriptor_.id, objectIdValue, eve::container::ContainerEventKind::Exit, seatIndex, *next, serial,
