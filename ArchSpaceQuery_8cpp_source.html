16bool pointInPolygon(Vec2 p, const std::vector<Vec2>& poly) {
17 if (poly.size() < 3) return false;
18 bool inside = false;
19 for (std::size_t i = 0, j = poly.size() - 1; i < poly.size(); j = i++) {
20 const Vec2& a = poly[i];
21 const Vec2& b = poly[j];
22 const bool intersect = ((a.z > p.z) != (b.z > p.z)) &&
23 (p.x < (b.x - a.x) * (p.z - a.z) / ((b.z - a.z) + (b.z == a.z ? kEps : 0.0)) + a.x);
24 if (intersect) inside = !inside;
25 }
26 return inside;
27}

45std::string nearestRegionId(const Document& document, const std::string& levelId, Vec2 xz) {
46 std::string best;
47 double bestDist2 = 0.0;
48 bool any = false;
49 for (const auto& [id, node] : document.nodes()) {
50 (void)id;
51 if (node.kind != NodeKind::Zone || node.parentId != levelId || node.polygon.empty()) continue;
52 double cx = 0.0, cz = 0.0;
53 for (const auto& p : node.polygon) {
54 cx += p.x;
55 cz += p.z;
56 }
57 cx /= static_cast<double>(node.polygon.size());
58 cz /= static_cast<double>(node.polygon.size());
59 const double dx = cx - xz.x;
60 const double dz = cz - xz.z;
61 const double d2 = dx * dx + dz * dz;
62 if (!any || d2 < bestDist2) {
63 any = true;
64 bestDist2 = d2;
65 best = node.id;
66 }
67 }
68 return best;
69}

71SpacePortal toPortalFromOpening(const Document& document, const Node& opening) {
72 SpacePortal portal;
73 portal.id = opening.id;
74 portal.name = opening.name;
75 portal.wallHosted = true;
76 portal.passageMask = opening.openingKind == OpeningKind::Door ? kPassageWalk : PassageMask{0};
77 portal.position = openingWorldPosition(document, opening);
78 const Node* wall = document.find(opening.parentId);
79 portal.levelId = wall ? wall->parentId : std::string{};
80 if (wall) {
81 const Vec2 mid{(wall->start.x + wall->end.x) * 0.5, (wall->start.z + wall->end.z) * 0.5};
82 // Sample slightly off the wall midline toward both normals to find adjacent rooms.
83 Vec2 dir{wall->end.x - wall->start.x, wall->end.z - wall->start.z};
84 const double len = std::sqrt(dir.x * dir.x + dir.z * dir.z);
85 if (len > kEps) {
86 dir.x /= len;
87 dir.z /= len;
88 const Vec2 n{-dir.z, dir.x};
89 const double offset = std::max(wall->thickness, 0.2);
90 const auto a = nearestRegionId(document, portal.levelId, Vec2{mid.x + n.x * offset, mid.z + n.z * offset});
91 const auto b = nearestRegionId(document, portal.levelId, Vec2{mid.x - n.x * offset, mid.z - n.z * offset});
92 portal.regionAId = a;
93 portal.regionBId = (b != a) ? b : std::string{};
94 }
95 }
96 return portal;
97}

171eve::Result<bool> SpaceQuery::regionsConnected(const std::string& regionAId, const std::string& regionBId) const {
172 if (!document_)
173 return eve::Result<bool>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
174 "ArchSpace SpaceQuery has no document", "document", {},
175 "archspace.query"));
176 if (regionAId.empty() || regionBId.empty())
177 return eve::Result<bool>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
178 "ArchSpace region ids must be non-empty", "regionId",
179 {}, "archspace.query"));
180 if (!document_->find(regionAId) || !document_->find(regionBId))
181 return eve::Result<bool>::failure(eve::Diagnostic::error(
182 eve::DiagnosticCode::NotFound, "ArchSpace region id is unknown", "regionId", {}, "archspace.query"));
183 if (regionAId == regionBId) return eve::Result<bool>::success(true);
184 std::unordered_map<std::string, std::vector<std::string>> adj;
185 for (const auto& portal : portals()) {
186 if (portal.regionAId.empty()) continue;
187 if (!portal.regionBId.empty()) {
188 adj[portal.regionAId].push_back(portal.regionBId);
189 adj[portal.regionBId].push_back(portal.regionAId);
190 }
191 }
192 std::unordered_set<std::string> seen;
193 std::queue<std::string> q;
194 q.push(regionAId);
195 seen.insert(regionAId);
196 while (!q.empty()) {
197 const std::string cur = std::move(q.front());
198 q.pop();
199 if (cur == regionBId) return eve::Result<bool>::success(true);
200 const auto it = adj.find(cur);
201 if (it == adj.end()) continue;
202 for (const auto& next : it->second) {
203 if (seen.insert(next).second) q.push(next);
204 }
205 }
206 return eve::Result<bool>::success(false);
207}
