22EditorResult<SocialEntityRecord> parseEntity(const EditorValue& value) {
23 const auto* i = field(value, "id");
24 const auto* l = field(value, "label");
25 const auto* c = field(value, "category");
26 const auto* id = i ? i->getIf<std::string>() : nullptr;
27 const auto* label = l ? l->getIf<std::string>() : nullptr;
28 const auto* category = c ? c->getIf<std::string>() : nullptr;
29 if (!id || id->empty() || !label || label->empty() || !category)
30 return fail<SocialEntityRecord>(EditorStatus::Rejected, "editor.social.invalid-entity",

39EditorResult<SocialEdgeRecord> parseEdge(const EditorValue& value) {
40 auto str = [&](const char* k) -> const std::string* {
41 const auto* v = field(value, k);
42 return v ? v->getIf<std::string>() : nullptr;
43 };
44 const auto* w = field(value, "weight");
45 const auto* weight = w ? w->getIf<double>() : nullptr;
46 const auto *id = str("id"), *source = str("source"), *target = str("target"), *kind = str("kind"),
47 *type = str("type");
48 static const std::set<std::string> kinds{"owner", "controller", "assignment", "relation"};
49 if (!id || id->empty() || !source || source->empty() || !target || target->empty() || !kind ||
50 !kinds.contains(*kind) || !type || !weight || !std::isfinite(*weight) ||
51 ((*kind == "assignment" || *kind == "relation") && type->empty()))
52 return fail<SocialEdgeRecord>(EditorStatus::Rejected, "editor.social.invalid-edge",
53 "Social edge requires ids, supported kind/type and finite weight");
54 return eve::editing::applied<SocialEdgeRecord>(
55 {StableId(*id), StableId(*source), StableId(*target), *kind, *type, *weight});
56}

87EditorResult<void> SocialDocumentTarget::applyDomainOperation(const DomainOperation& o) {
88 if (o.target != TargetId(id_))
89 return fail<void>(EditorStatus::Rejected, "editor.social.target", "Operation targets another social document");
90 if (o.type == "social.entity.set.v1") {
91 auto v = parseEntity(o.payload);
92 if (!v.ok()) return fail<void>(v.code(), "editor.social.entity-payload", "Invalid entity payload");
93 entities_.insert_or_assign(v.value().id, v.value());
94 } else if (o.type == "social.edge.set.v1") {
95 auto v = parseEdge(o.payload);
96 if (!v.ok()) return fail<void>(v.code(), "editor.social.edge-payload", "Invalid edge payload");
97 if (!entities_.contains(v.value().source) || !entities_.contains(v.value().target))
98 return fail<void>(EditorStatus::Conflict, "editor.social.missing-endpoint",
99 "Social edge endpoints must exist");
100 edges_.insert_or_assign(v.value().id, v.value());
101 } else if (o.type == "social.entity.delete.v1" || o.type == "social.edge.delete.v1") {
102 const auto* id = o.payload.getIf<std::string>();
103 if (!id)
104 return fail<void>(EditorStatus::Rejected, "editor.social.delete-payload", "Delete requires a stable id");
105 const StableId stable(*id);
106 if (o.type == "social.entity.delete.v1") {
107 for (const auto& [edgeId, e] : edges_) {
108 (void)edgeId;
109 if (e.source == stable || e.target == stable)
110 return fail<void>(EditorStatus::Conflict, "editor.social.entity-in-use",
111 "Entity is referenced by an edge");
112 }
113 if (!entities_.erase(stable))
114 return fail<void>(EditorStatus::NotFound, "editor.social.entity-not-found", "Entity was not found");
115 } else if (!edges_.erase(stable))
116 return fail<void>(EditorStatus::NotFound, "editor.social.edge-not-found", "Edge was not found");
117 } else
118 return fail<void>(EditorStatus::Rejected, "editor.social.operation", "Unsupported social operation");
119 ++revision_;
120 dirty_.include(0, 0);
121 return eve::editing::applied<void>();
122}

157EditorResult<DomainOperation> SocialDocumentTarget::makeSetEdge(const SocialEdgeRecord& e) const {
158 if (!parseEdge(edgeValue(e)).ok())
159 return fail<DomainOperation>(EditorStatus::Rejected, "editor.social.invalid-edge", "Cannot plan invalid edge");
160 if (!entities_.contains(e.source) || !entities_.contains(e.target))
161 return fail<DomainOperation>(EditorStatus::Conflict, "editor.social.missing-endpoint",
162 "Social edge endpoints must exist");
163 for (const auto& [id, current] : edges_)
164 if (id != e.id && current.source == e.source && current.target == e.target && current.kind == e.kind &&
165 current.type == e.type)

167 "Equivalent social edge already exists");
168 const auto f = edges_.find(e.id);
169 return eve::editing::applied<DomainOperation>(
170 op("social.edge.set.v1", f == edges_.end() ? "social.edge.delete.v1" : "social.edge.set.v1", id_, edgeValue(e),
171 f == edges_.end() ? EditorValue(e.id.value()) : edgeValue(f->second), e.id));

180std::vector<EditorDiagnostic> SocialDocumentTarget::validate() const {
181 std::vector<EditorDiagnostic> r;
182 std::set<std::tuple<StableId, StableId, std::string, std::string>> unique;
183 std::map<StableId, int> owners, controllers;
184 for (const auto& [id, e] : edges_) {
185 if (!entities_.contains(e.source) || !entities_.contains(e.target))
186 r.push_back(eve::editing::ruleDiagnostic(
187 eve::DiagnosticCode::NotFound, RuleId("editor.social.missing-endpoint"),
188 DiagnosticSeverity::Error, "Edge " + id.value() + " has a missing endpoint"));
189 if (!unique.emplace(e.source, e.target, e.kind, e.type).second)
190 r.push_back(eve::editing::ruleDiagnostic(
191 eve::DiagnosticCode::Conflict, RuleId("editor.social.duplicate-edge"),
192 DiagnosticSeverity::Error, "Duplicate social edge semantics"));
193 if (e.kind == "owner" && ++owners[e.source] > 1)
194 r.push_back(eve::editing::ruleDiagnostic(
195 eve::DiagnosticCode::Conflict, RuleId("editor.social.multiple-owners"),
196 DiagnosticSeverity::Error, "Entity has multiple owners"));
197 if (e.kind == "controller" && ++controllers[e.source] > 1)
198 r.push_back(eve::editing::ruleDiagnostic(
199 eve::DiagnosticCode::Conflict, RuleId("editor.social.multiple-controllers"),
200 DiagnosticSeverity::Error, "Entity has multiple controllers"));
201 }
202 return r;
203}

217EditorResult<void> SocialDocumentTarget::loadSnapshot(const EditorValue& s) {
218 const auto *v = field(s, "schemaVersion"), *ns = field(s, "entities"), *es = field(s, "edges");
219 const auto* version = v ? v->getIf<int64_t>() : nullptr;
220 const auto* nodes = ns ? ns->getIf<EditorValue::Array>() : nullptr;
221 const auto* edges = es ? es->getIf<EditorValue::Array>() : nullptr;
222 if (!version || *version != 1 || !nodes || !edges)
223 return fail<void>(EditorStatus::Rejected, "editor.social.snapshot", "Invalid social snapshot envelope");
224 SocialDocumentTarget c(id_);
225 for (const auto& e : *nodes) {
226 auto p = parseEntity(e);
227 if (!p.ok() || !c.entities_.emplace(p.value().id, p.value()).second)
228 return fail<void>(EditorStatus::Rejected, "editor.social.snapshot-entity",
229 "Invalid or duplicate snapshot entity");
230 }
231 for (const auto& e : *edges) {
232 auto p = parseEdge(e);
233 if (!p.ok() || !c.edges_.emplace(p.value().id, p.value()).second)
234 return fail<void>(EditorStatus::Rejected, "editor.social.snapshot-edge",
235 "Invalid or duplicate snapshot edge");
236 }
237 for (const auto& d : c.validate())
238 if (d.severity() == DiagnosticSeverity::Error)
239 return fail<void>(EditorStatus::Rejected, "editor.social.snapshot-invariant",
240 "Social snapshot violates graph invariants");
241 entities_ = std::move(c.entities_);
242 edges_ = std::move(c.edges_);
243 ++revision_;
244 dirty_.include(0, 0);
245 return eve::editing::applied<void>();
246}
