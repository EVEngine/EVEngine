35EditorResult<SocialEdgeRecord> parseEdge(const EditorValue& value) {
36 auto str = [&](const char* k) -> const std::string* {
37 const auto* v = field(value, k);
38 return v ? v->getIf<std::string>() : nullptr;
39 };
40 const auto* w = field(value, "weight");
41 const auto* weight = w ? w->getIf<double>() : nullptr;
42 const auto *id = str("id"), *source = str("source"), *target = str("target"), *kind = str("kind"),
43 *type = str("type");
44 static const std::set<std::string> kinds{"owner", "controller", "assignment", "relation"};
45 if (!id || id->empty() || !source || source->empty() || !target || target->empty() || !kind ||
46 !kinds.contains(*kind) || !type || !weight || !std::isfinite(*weight) ||
47 ((*kind == "assignment" || *kind == "relation") && type->empty()))
48 return eve::editing::failed<SocialEdgeRecord>(
49 EditorStatus::Rejected, RuleId("editor.social.invalid-edge"),
50 "Social edge requires ids, supported kind/type and finite weight");
51 return eve::editing::applied<SocialEdgeRecord>(
52 {StableId(*id), StableId(*source), StableId(*target), *kind, *type, *weight});
53}

84EditorResult<void> SocialDocumentTarget::applyDomainOperation(const DomainOperation& o) {
85 if (o.target != TargetId(id_))
86 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.social.target"),
87 "Operation targets another social document");
88 if (o.type == "social.entity.set.v1") {
89 auto v = parseEntity(o.payload);
90 if (!v.ok())
91 return eve::editing::failed<void>(v.code(), RuleId("editor.social.entity-payload"),
92 "Invalid entity payload");
93 entities_.insert_or_assign(v.value().id, v.value());
94 } else if (o.type == "social.edge.set.v1") {
95 auto v = parseEdge(o.payload);
96 if (!v.ok())
97 return eve::editing::failed<void>(v.code(), RuleId("editor.social.edge-payload"), "Invalid edge payload");
98 if (!entities_.contains(v.value().source) || !entities_.contains(v.value().target))
99 return eve::editing::failed<void>(EditorStatus::Conflict, RuleId("editor.social.missing-endpoint"),
100 "Social edge endpoints must exist");
101 edges_.insert_or_assign(v.value().id, v.value());
102 } else if (o.type == "social.entity.delete.v1" || o.type == "social.edge.delete.v1") {
103 const auto* id = o.payload.getIf<std::string>();
104 if (!id)
105 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.social.delete-payload"),
106 "Delete requires a stable id");
107 const StableId stable(*id);
108 if (o.type == "social.entity.delete.v1") {
109 for (const auto& [edgeId, e] : edges_) {
110 (void)edgeId;
111 if (e.source == stable || e.target == stable)
112 return eve::editing::failed<void>(EditorStatus::Conflict, RuleId("editor.social.entity-in-use"),
113 "Entity is referenced by an edge");
114 }
115 if (!entities_.erase(stable))
116 return eve::editing::failed<void>(EditorStatus::NotFound, RuleId("editor.social.entity-not-found"),
117 "Entity was not found");
118 } else if (!edges_.erase(stable))
119 return eve::editing::failed<void>(EditorStatus::NotFound, RuleId("editor.social.edge-not-found"),
120 "Edge was not found");
121 } else
122 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.social.operation"),
123 "Unsupported social operation");
124 bumpRevision();
125 widenDirty(0, 0);
126 return eve::editing::applied<void>();
127}

189std::vector<EditorDiagnostic> SocialDocumentTarget::validate() const {
190 std::vector<EditorDiagnostic> r;
191 std::set<std::tuple<StableId, StableId, std::string, std::string>> unique;
192 std::map<StableId, int> owners, controllers;
193 for (const auto& [id, e] : edges_) {
194 if (!entities_.contains(e.source) || !entities_.contains(e.target))
195 r.push_back(eve::editing::ruleDiagnostic(
196 eve::DiagnosticCode::NotFound, RuleId("editor.social.missing-endpoint"),
197 DiagnosticSeverity::Error, "Edge " + id.value() + " has a missing endpoint"));
198 if (!unique.emplace(e.source, e.target, e.kind, e.type).second)
199 r.push_back(eve::editing::ruleDiagnostic(
200 eve::DiagnosticCode::Conflict, RuleId("editor.social.duplicate-edge"),
201 DiagnosticSeverity::Error, "Duplicate social edge semantics"));
202 if (e.kind == "owner" && ++owners[e.source] > 1)
203 r.push_back(eve::editing::ruleDiagnostic(
204 eve::DiagnosticCode::Conflict, RuleId("editor.social.multiple-owners"),
205 DiagnosticSeverity::Error, "Entity has multiple owners"));
206 if (e.kind == "controller" && ++controllers[e.source] > 1)
207 r.push_back(eve::editing::ruleDiagnostic(
208 eve::DiagnosticCode::Conflict, RuleId("editor.social.multiple-controllers"),
209 DiagnosticSeverity::Error, "Entity has multiple controllers"));
210 }
211 return r;

226EditorResult<void> SocialDocumentTarget::loadSnapshot(const EditorValue& s) {
227 const auto *v = field(s, "schemaVersion"), *ns = field(s, "entities"), *es = field(s, "edges");
228 const auto* version = v ? v->getIf<int64_t>() : nullptr;
229 const auto* nodes = ns ? ns->getIf<EditorValue::Array>() : nullptr;
230 const auto* edges = es ? es->getIf<EditorValue::Array>() : nullptr;
231 if (!version || *version != 1 || !nodes || !edges)
232 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.social.snapshot"),
233 "Invalid social snapshot envelope");
234 SocialDocumentTarget c(id_);
235 for (const auto& e : *nodes) {
236 auto p = parseEntity(e);
237 if (!p.ok() || !c.entities_.emplace(p.value().id, p.value()).second)
238 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.social.snapshot-entity"),
239 "Invalid or duplicate snapshot entity");
240 }
241 for (const auto& e : *edges) {
242 auto p = parseEdge(e);
243 if (!p.ok() || !c.edges_.emplace(p.value().id, p.value()).second)
244 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.social.snapshot-edge"),
245 "Invalid or duplicate snapshot edge");
246 }
247 for (const auto& d : c.validate())
248 if (d.severity() == DiagnosticSeverity::Error)
249 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.social.snapshot-invariant"),
250 "Social snapshot violates graph invariants");
251 entities_ = std::move(c.entities_);
252 edges_ = std::move(c.edges_);
253 bumpRevision();
254 widenDirty(0, 0);
255 return eve::editing::applied<void>();
256}
