95Result<BuildingBlueprint> AssemblyService::decodeBlueprint(const Value& v) {
96 try {
97 knownFields(v, {"schema", "version", "id", "assetVersion", "cellSize", "floorHeight", "parts", "curves",
98 "definitions"});
99 schema(v, "eve.building.blueprint");
100 BuildingBlueprint b;
101 b.id = identity<AssetGuid>(required(v, "id"));
102 b.version = uint64_t(integer(required(v, "assetVersion"), 1));
103 b.cellSize = number(required(v, "cellSize"));
104 b.floorHeight = number(required(v, "floorHeight"));
105 if (b.cellSize <= 0 || b.floorHeight <= 0) throw std::runtime_error("Invalid blueprint scale");
106 b.definitions = required(v, "definitions");
107 auto defs = checkDefinitions(b.definitions);
108 if (!defs.ok())
109 return fail<BuildingBlueprint>("Blueprint definitions missing or changed", DiagnosticCode::Conflict);
110 std::set<int> keys;
111 for (const auto& p : array(required(v, "parts"))) {
112 knownFields(p, {"key", "instance"});
113 knownInstance(required(p, "instance"));
114 int key = integer(required(p, "key"), 1, std::numeric_limits<int>::max() - 1);
115 auto part = codec::parseInstance(required(p, "instance"));
116 const bool partOk = part.ok();
117 if (!keys.insert(key).second || !partOk || part.value().instanceId != key ||
118 !part.value().garrison.empty() || !b.definitions.find(part.value().buildingId))
119 throw std::runtime_error("Invalid blueprint part or external entity reference");
120 b.parts.push_back({{uint32_t(key)}, std::move(part).takeValue()});
121 }
122 if (b.parts.empty() || b.parts.size() > 100000)
123 throw std::runtime_error("Blueprint must contain 1..100000 parts");
124 std::set<int> curveMembers;
125 for (const auto& c : array(required(v, "curves"))) {
126 knownCurve(c);
127 auto curve = codec::parseEdgeCurve(c);
128 if (!curve.ok()) throw std::runtime_error("Invalid curve payload");
129 for (const auto& p : curve.value().instances) {
130 auto found = std::find_if(b.parts.begin(), b.parts.end(),
131 [&](const auto& part) { return part.local.instanceId == p.instanceId; });
132 if (found == b.parts.end() || !curveMembers.insert(p.instanceId).second ||
133 codec::instanceValue(found->local) != codec::instanceValue(p))
134 throw std::runtime_error("Curve membership differs from blueprint parts");
135 }
136 b.curves.push_back(std::move(curve).takeValue());
137 }
138 return Result<BuildingBlueprint>::success(std::move(b));
139 } catch (const std::exception& e) {
140 return fail<BuildingBlueprint>(e.what());
141 }

199Result<std::unique_ptr<PlacementWorld>> AssemblyService::decodeWorld(const Value& v) {
200 try {
201 knownFields(
202 v, {"schema", "version", "worldId", "width", "height", "cellSize", "plane",
203 "originX", "originY", "floorHeight", "activeLevel", "snapMode", "validateRule", "extra",
204 "nextInstance", "nextCurve", "terrain", "instances", "curves", "assemblies", "definitions"});
205 schema(v, "eve.building.world");
206 auto defs = checkDefinitions(required(v, "definitions"));
207 if (!defs.ok()) throw std::runtime_error("World definitions missing or changed");
208 int width = integer(required(v, "width"), 1, 4096), height = integer(required(v, "height"), 1, 4096);
209 if (int64_t(width) * height > 4000000) throw std::runtime_error("World exceeds cell budget");
210 float size = number(required(v, "cellSize"));
211 if (size <= 0) throw std::runtime_error("Invalid cell size");
212 auto w = std::make_unique<PlacementWorld>(width, height, size);
213 w->publishEvents_ = false;
214 w->id_ = text(required(v, "worldId"));
215 auto plane = text(required(v, "plane"));
216 if (plane != "xz" && plane != "xy") throw std::runtime_error("Invalid plane");
217 w->setGridPlane(plane);
218 w->setOrigin(number(required(v, "originX")), number(required(v, "originY")));
219 w->floorHeight_ = number(required(v, "floorHeight"));
220 if (w->floorHeight_ <= 0) throw std::runtime_error("Invalid floor height");
221 w->activeLevel_ = integer(required(v, "activeLevel"));
222 w->snapMode_ = text(required(v, "snapMode"));
223 w->validateRule_ = text(required(v, "validateRule"));
224 const auto& extra = required(v, "extra");
225 if (!extra.isObject()) throw std::runtime_error("Invalid extras");
226 for (const auto& key : extra.keys()) w->extra_[key] = text(*extra.find(key));
227 const auto& terrain = array(required(v, "terrain"));
228 if (terrain.size() != size_t(width) * height) throw std::runtime_error("Terrain dimensions mismatch");
229 for (size_t i = 0; i < terrain.size(); ++i) w->terrain_[i] = integer(terrain[i]);
230 w->terrainOverrides_.assign(terrain.size(), 1);
231 w->instanceCounter_ = integer(required(v, "nextInstance"), 0);
232 const auto& nextCurve = required(v, "nextCurve");
233 if (!nextCurve.isInt64() || nextCurve.asInt() <= 0) throw std::runtime_error("Invalid curve sequence");
234 w->nextEdgeCurveGroupId_ = uint64_t(nextCurve.asInt());
235 for (const auto& record : array(required(v, "instances"))) {
236 knownInstance(record);
237 auto p = codec::parseInstance(record);
238 if (!p.ok()) throw std::runtime_error("Invalid instance payload");
239 int id = p.value().instanceId;
240 if (id > w->instanceCounter_ || !required(v, "definitions").find(p.value().buildingId) ||
241 !w->buildings_.emplace(id, codec::runtime(p.value())).second)
242 throw std::runtime_error("Duplicate/unpinned instance record");
243 w->instanceOrder_.push_back(id);
244 }
245 for (const auto& record : array(required(v, "curves"))) {
246 knownCurve(record);
247 auto c = codec::parseEdgeCurve(record);
248 if (!c.ok()) throw std::runtime_error("Invalid curve");
249 auto& payload = c.value();
250 for (const auto& p : payload.instances) {
251 auto it = w->buildings_.find(p.instanceId);
252 if (it == w->buildings_.end() ||
253 codec::instanceValue(codec::snapshot(it->second)) != codec::instanceValue(p))
254 throw std::runtime_error("Curve member payload mismatch");
255 }
256 if (payload.group.id.value >= w->nextEdgeCurveGroupId_ ||
257 !w->edgeCurveGroups_.emplace(payload.group.id.value, payload.group).second)
258 throw std::runtime_error("Duplicate curve record");
259 }
260 for (const auto& record : array(required(v, "assemblies"))) {
261 knownFields(record, {"id", "blueprint", "blueprintVersion", "frame", "members", "nextPartKey"});
262 BuildingAssembly a;
263 a.id = identity<AssemblyId>(required(record, "id"));
264 a.blueprint = identity<AssetGuid>(required(record, "blueprint"));
265 a.blueprintVersion = uint64_t(integer(required(record, "blueprintVersion"), 1));
266 a.nextPartKey = uint32_t(integer(required(record, "nextPartKey"), 1));
267 auto pose = parsePose(required(record, "frame"));
268 if (!pose.ok()) throw std::runtime_error("Invalid house frame");
269 a.frame = pose.value();
270 for (const auto& m : array(required(record, "members"))) {
271 knownFields(m, {"key", "record", "persistentId"});
272 int key = integer(required(m, "key"), 1), member = integer(required(m, "record"), 1);
273 if (!a.members.emplace(uint32_t(key), member).second ||
274 text(required(m, "persistentId")) != a.id.child(std::to_string(key)).format())
275 throw std::runtime_error("Invalid persistent part identity");
276 }
277 if (!w->assemblyState_->records.emplace(a.id, a).second)
278 throw std::runtime_error("Duplicate house identity");
279 w->assemblyState_->handles[a.id] = eve::RuntimeHandle<AssemblyHandleTag>(w->assemblyState_->nextSlot++, 1);
280 }
281 auto valid = validateFinal(*w);
282 if (!valid.ok())
283 return fail<std::unique_ptr<PlacementWorld>>("Saved world failed final-state validation",
284 DiagnosticCode::Conflict);
285 return Result<std::unique_ptr<PlacementWorld>>::success(std::move(w));
286 } catch (const std::exception& e) {
287 return fail<std::unique_ptr<PlacementWorld>>(e.what());
288 }
