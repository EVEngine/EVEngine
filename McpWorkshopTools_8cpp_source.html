109std::string workshopPlay(const char* op, Poco::JSON::Object::Ptr args) {
110 Poco::JSON::Object::Ptr req = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
111 req->set("schemaId", "evengine.play-request");
112 req->set("schemaVersion", 1);
113 req->set("op", std::string(op));
114 if (args) {
115 if (args->has("id")) req->set("id", args->get("id"));
116 if (args->has("observation")) req->set("observation", args->get("observation"));
117 if (args->has("action")) req->set("action", args->get("action"));
118 if (args->has("params")) req->set("params", args->get("params"));
119 if (args->has("fields")) req->set("fields", args->get("fields"));
120 }
121 auto playResponse = executePlayJson(mcpStringify(Poco::Dynamic::Var(req)));
122 if (!playResponse) {
123 return errorPayload(playResponse.status().describe(),
124 "Play Host needs game.agent.json with paths.gameState / observations");
125 }
126 auto playObj = parseJsonObject(playResponse.value());
127 if (!playObj) {
128 Poco::JSON::Object::Ptr out = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
129 out->set("ok", true);
130 out->set("op", std::string(op));
131 out->set("raw", playResponse.value());
132 return textPayload(out);
133 }
134 playObj->set("workshopOp", std::string("play.") + op);
135 return textPayload(playObj, !playObj->optValue<bool>("ok", true));
136}

138std::string workshopDispatch(Poco::JSON::Object::Ptr args) {
139 if (!args || !args->has("op")) {
140 return errorPayload("op is required", "ops: status|set_mode|recipe.apply|map.brush|play.observe|play.submit");
141 }
142 const std::string op = getArgString(args, "op");
143 if (op == "status") return workshopStatus();
144 if (op == "set_mode") return workshopSetMode(args);
145 if (op == "recipe.apply") return workshopRecipeApply(args);
146 if (op == "map.brush") return workshopMapBrush(args);
147 if (op == "play.observe") return workshopPlay("observe", args);
148 if (op == "play.submit" || op == "play.act") return workshopPlay("act", args);
149 return errorPayload("unknown workshop op '" + op + "'",
150 "ops: status|set_mode|recipe.apply|map.brush|play.observe|play.submit");
151}
