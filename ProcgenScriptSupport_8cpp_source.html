65eve::Value artifactProjection(GeneratedArtifact&& artifact) {
66 eve::Value::Array dependencies;
67 dependencies.reserve(artifact.dependencies.size());
68 for (const ArtifactId& dependency : artifact.dependencies) dependencies.emplace_back(dependency.format());
69
70 eve::Value::Object result;
71 result.emplace("id", eve::Value(artifact.id.format()));
72 result.emplace("type", eve::Value(std::string(artifactTypeName(artifact.type))));
73 result.emplace("schemaVersion", eve::Value(static_cast<std::int64_t>(artifact.schemaVersion.value())));
74 result.emplace("buildKey", eve::Value(artifact.buildKey.format()));
75 result.emplace("bounds", boundsValue(artifact.bounds));
76 result.emplace("dependencies", eve::Value(std::move(dependencies)));
77 result.emplace("metadata", eve::Value(std::move(artifact.metadata)));
78 if (artifact.type == ArtifactType::Composite) {
79 const auto* composite = std::get_if<CompositeArtifact>(&artifact.payload);
80 result.emplace("partCount", eve::Value(static_cast<std::int64_t>(composite ? composite->children.size() : 0)));
81 } else {
82 result.emplace("partCount", eve::Value(std::int64_t(1)));
83 }
84 return eve::Value(std::move(result));
85}

137ssq::Table projectRecipeDescriptorResult(HSQUIRRELVM vm, eve::Result<RecipeDescriptor>&& result) {
138 const bool ok = result.ok();

140 if (!ok) return eve::script::projectStatusResult(vm, status);
141

145 if (!object) {
