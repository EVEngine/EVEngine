47EditorResult<SplinePathControlPoint> parsePoint(const EditorValue& value) {
48 const auto string = [&](const char* name) {
49 const auto* entry = field(value, name);
50 return entry ? entry->getIf<std::string>() : nullptr;
51 };
52 const auto integer = [&](const char* name) {
53 const auto* entry = field(value, name);
54 return entry ? entry->getIf<std::int64_t>() : nullptr;
55 };
56 const auto number = [&](const char* name) {
57 const auto* entry = field(value, name);
58 return entry ? entry->getIf<double>() : nullptr;
59 };
60 const auto* id = string("id");
61 const auto* order = integer("order");
62 const auto* x = number("x");
63 const auto* y = number("y");
64 const auto* z = number("z");
65 const auto* inX = number("inX");
66 const auto* inY = number("inY");
67 const auto* inZ = number("inZ");
68 const auto* outX = number("outX");
69 const auto* outY = number("outY");
70 const auto* outZ = number("outZ");
71 const auto* rollValue = field(value, "rollDegrees");
72 const auto* scaleXValue = field(value, "scaleX");
73 const auto* scaleYValue = field(value, "scaleY");
74 const auto* breakValue = field(value, "breakBefore");
75 const auto* pitchValue = field(value, "pitchDegrees");
76 const auto* yawValue = field(value, "yawDegrees");
77 const auto* roll = rollValue ? rollValue->getIf<double>() : nullptr;
78 const auto* scaleX = scaleXValue ? scaleXValue->getIf<double>() : nullptr;
79 const auto* scaleY = scaleYValue ? scaleYValue->getIf<double>() : nullptr;
80 const double resolvedRoll = roll ? *roll : 0.0;
81 const double resolvedScaleX = scaleX ? *scaleX : 1.0;
82 const double resolvedScaleY = scaleY ? *scaleY : 1.0;
83 const auto* breakBefore = breakValue ? breakValue->getIf<bool>() : nullptr;
84 const auto* pitch = pitchValue ? pitchValue->getIf<double>() : nullptr;
85 const auto* yaw = yawValue ? yawValue->getIf<double>() : nullptr;
86 const double resolvedPitch = pitch ? *pitch : 0.0;
87 const double resolvedYaw = yaw ? *yaw : 0.0;
88 if (!id || id->empty() || !order || !x || !y || !z || !inX || !inY || !inZ || !outX || !outY || !outZ ||
89 !std::isfinite(*x) || !std::isfinite(*y) || !std::isfinite(*z) || !std::isfinite(*inX) ||
90 !std::isfinite(*inY) || !std::isfinite(*inZ) || !std::isfinite(*outX) || !std::isfinite(*outY) ||
91 !std::isfinite(*outZ) || !std::isfinite(resolvedRoll) || !std::isfinite(resolvedScaleX) ||

93 resolvedScaleX <= 0.0 || resolvedScaleY <= 0.0)
94 return splineError<SplinePathControlPoint>(EditorStatus::Rejected, "editor.spline.invalid-point",
95 "Spline point requires a stable id, order and finite coordinates");
96 return eve::editing::applied<SplinePathControlPoint>(
97 {StableId(*id), *order, *x, *y, *z, *inX, *inY, *inZ, *outX, *outY, *outZ, resolvedRoll, resolvedScaleX,
98 resolvedScaleY, breakBefore ? *breakBefore : false, resolvedPitch, resolvedYaw});

144EditorResult<void> SplinePathDocument::applyDomainOperation(const DomainOperation& value) {
145 if (value.target != TargetId(id_))
146 return splineError<void>(EditorStatus::Rejected, "editor.spline.target-mismatch",
147 "Spline operation targets another document");
148 if (value.type == "spline.snapshot.replace.v1") {
149 SplinePathDocument candidate(id_);
150 auto loaded = candidate.loadSnapshot(value.payload);
151 if (!loaded.ok())
152 return splineError<void>(loaded.code(), "editor.spline.invalid-replacement", loaded.status().describe());
153 candidate.revision_ = revision_ + 1;
154 candidate.dirty_.include(0, 0);
155 *this = std::move(candidate);
156 return eve::editing::applied<void>();
157 }
158 if (value.type == "spline.point.set.v1") {
159 auto parsed = parsePoint(value.payload);
160 if (!parsed.ok())
161 return splineError<void>(parsed.code(), "editor.spline.invalid-point", "Invalid spline point");
162 points_[parsed.value().id] = std::move(parsed.value());
163 } else if (value.type == "spline.point.delete.v1") {
164 auto parsed = parsePoint(value.payload);
165 if (!parsed.ok() || !points_.erase(parsed.value().id))
166 return splineError<void>(EditorStatus::NotFound, "editor.spline.point-not-found",
167 "Spline point was not found");
168 } else if (value.type == "spline.settings.set.v1") {
169 auto parsed = parseSettings(value.payload);
170 if (!parsed.ok())
171 return splineError<void>(parsed.code(), "editor.spline.invalid-settings", "Invalid spline settings");
172 settings_ = std::move(parsed.value());
173 } else {
174 return splineError<void>(EditorStatus::Unsupported, "editor.spline.operation-unsupported",
175 "Spline operation is unsupported");
176 }
177 ++revision_;
178 dirty_.include(0, 0);
179 return eve::editing::applied<void>();
180}

182EditorResult<DomainOperation> SplinePathDocument::makeInsertPointAt(int segment, double t,
183 const StableId& point) const {
184 const auto ordered = points();
185 std::vector<std::pair<int, int>> segments;
186 for (int i = 0; i + 1 < static_cast<int>(ordered.size()); ++i)
187 if (!ordered[static_cast<std::size_t>(i + 1)].breakBefore) segments.emplace_back(i, i + 1);
188 if (settings_.closed && ordered.size() > 1u) segments.emplace_back(static_cast<int>(ordered.size()) - 1, 0);
189 const int segmentCount = static_cast<int>(segments.size());
190 if (point.empty() || points_.contains(point) || segment < 0 || segment >= segmentCount || !std::isfinite(t) ||
191 t <= 0.0 || t >= 1.0)
192 return splineError<DomainOperation>(EditorStatus::Rejected, "editor.spline.insert-invalid",
193 "Spline insertion requires a unique id, valid segment and t in (0,1)");
194 auto runtime = compilePath();
195 if (!runtime.ok())
196 return splineError<DomainOperation>(runtime.code(), "editor.spline.insert-incomplete",
197 runtime.status().describe());
198 auto sampled = runtime.value().evaluateResult((static_cast<float>(segment) + static_cast<float>(t)) /
199 static_cast<float>(segmentCount));
200 if (!sampled.ok())
201 return splineError<DomainOperation>(EditorStatus::Rejected, "editor.spline.insert-sample-failed",
202 sampled.status().describe());
203 SplinePathDocument candidate(*this);
204 auto candidatePoints = ordered;
205 const int firstIndex = segments[static_cast<std::size_t>(segment)].first;
206 const int nextIndex = segments[static_cast<std::size_t>(segment)].second;
207 SplinePathControlPoint inserted;
208 inserted.id = point;
209 inserted.x = sampled.value().x;
210 inserted.y = sampled.value().y;
211 inserted.z = sampled.value().z;
212 inserted.rollDegrees = sampled.value().rollDegrees;
213 inserted.scaleX = sampled.value().scaleX;
214 inserted.scaleY = sampled.value().scaleY;
215 if (settings_.kind == "bezier") {
216 auto& first = candidatePoints[static_cast<std::size_t>(firstIndex)];
217 auto& second = candidatePoints[static_cast<std::size_t>(nextIndex)];
218 const auto lerp = [t](double a, double b) { return a + (b - a) * t; };
219 const double q0x = lerp(first.x, first.x + first.outX), q0y = lerp(first.y, first.y + first.outY),
220 q0z = lerp(first.z, first.z + first.outZ);
221 const double q1x = lerp(first.x + first.outX, second.x + second.inX),
222 q1y = lerp(first.y + first.outY, second.y + second.inY),
223 q1z = lerp(first.z + first.outZ, second.z + second.inZ);
224 const double q2x = lerp(second.x + second.inX, second.x), q2y = lerp(second.y + second.inY, second.y),
225 q2z = lerp(second.z + second.inZ, second.z);
226 const double r0x = lerp(q0x, q1x), r0y = lerp(q0y, q1y), r0z = lerp(q0z, q1z);
227 const double r1x = lerp(q1x, q2x), r1y = lerp(q1y, q2y), r1z = lerp(q1z, q2z);
228 inserted.x = lerp(r0x, r1x);
229 inserted.y = lerp(r0y, r1y);
230 inserted.z = lerp(r0z, r1z);
231 first.outX = q0x - first.x;
232 first.outY = q0y - first.y;
233 first.outZ = q0z - first.z;
234 inserted.inX = r0x - inserted.x;
235 inserted.inY = r0y - inserted.y;
236 inserted.inZ = r0z - inserted.z;
237 inserted.outX = r1x - inserted.x;
238 inserted.outY = r1y - inserted.y;
239 inserted.outZ = r1z - inserted.z;
240 second.inX = q2x - second.x;
241 second.inY = q2y - second.y;
242 second.inZ = q2z - second.z;
243 }
244 const int insertionIndex = nextIndex == 0 ? static_cast<int>(candidatePoints.size()) : nextIndex;
245 candidatePoints.insert(candidatePoints.begin() + insertionIndex, inserted);
246 candidate.points_.clear();
247 for (std::size_t i = 0; i < candidatePoints.size(); ++i) {
248 candidatePoints[i].order = static_cast<std::int64_t>(i);
249 candidate.points_[candidatePoints[i].id] = candidatePoints[i];
250 }
251 return eve::editing::applied<DomainOperation>(operation("spline.snapshot.replace.v1", "spline.snapshot.replace.v1",
252 id_, candidate.snapshotValue(), snapshotValue(), point));
253}

286EditorResult<DomainOperation> SplinePathDocument::makeFlipDirection() const {
287 if (points_.size() < 2u)
288 return splineError<DomainOperation>(EditorStatus::Rejected, "editor.spline.flip-incomplete",
289 "Spline direction requires at least two points");
290 std::vector<std::vector<SplinePathControlPoint>> chunks(1);
291 for (const auto& point : points()) {
292 if (point.breakBefore) chunks.emplace_back();
293 chunks.back().push_back(point);
294 }
295 std::reverse(chunks.begin(), chunks.end());
296 std::vector<SplinePathControlPoint> reversed;
297 for (auto& chunk : chunks) {
298 std::reverse(chunk.begin(), chunk.end());
299 for (std::size_t i = 0; i < chunk.size(); ++i) {
300 chunk[i].breakBefore = !reversed.empty() && i == 0;
301 reversed.push_back(chunk[i]);
302 }
303 }
304 SplinePathDocument candidate(*this);
305 candidate.points_.clear();
306 for (std::size_t i = 0; i < reversed.size(); ++i) {
307 auto& point = reversed[i];
308 point.order = static_cast<std::int64_t>(i);
309 std::swap(point.inX, point.outX);
310 std::swap(point.inY, point.outY);
311 std::swap(point.inZ, point.outZ);
312 candidate.points_[point.id] = point;
313 }
314 return eve::editing::applied<DomainOperation>(operation("spline.snapshot.replace.v1", "spline.snapshot.replace.v1",
315 id_, candidate.snapshotValue(), snapshotValue(),
316 reversed.front().id));
317}

319EditorResult<DomainOperation> SplinePathDocument::makeSetChunkBreak(const StableId& point, bool disconnected) const {
320 if (settings_.closed && disconnected)
321 return splineError<DomainOperation>(EditorStatus::Rejected, "editor.spline.chunk-break-closed",
322 "Closed splines cannot contain disconnected chunks");
323 auto ordered = points();
324 const auto found =
325 std::find_if(ordered.begin(), ordered.end(), [&](const auto& value) { return value.id == point; });
326 if (found == ordered.end() || found == ordered.begin())
327 return splineError<DomainOperation>(EditorStatus::NotFound, "editor.spline.chunk-break-point-invalid",
328 "Chunk boundary requires an existing non-first point");
329 const auto index = static_cast<std::size_t>(found - ordered.begin());
330 if (disconnected && !found->breakBefore) {
331 std::size_t chunkStart = index;
332 while (chunkStart > 0u && !ordered[chunkStart].breakBefore) --chunkStart;
333 std::size_t chunkEnd = index;
334 while (chunkEnd + 1u < ordered.size() && !ordered[chunkEnd + 1u].breakBefore) ++chunkEnd;
335 if (index - chunkStart < 2u || chunkEnd - index + 1u < 2u)
336 return splineError<DomainOperation>(EditorStatus::Rejected, "editor.spline.chunk-break-singleton",
337 "Splitting must leave at least two points in each chunk");
338 }
339 auto updated = *found;
340 updated.breakBefore = disconnected;
341 return makeSetPoint(updated);
342}

344EditorResult<DomainOperation> SplinePathDocument::makeAppendChunk(const SplinePathControlPoint& first,
345 const SplinePathControlPoint& second) const {
346 if (settings_.closed)
347 return splineError<DomainOperation>(EditorStatus::Rejected, "editor.spline.append-chunk-closed",
348 "Closed splines cannot append disconnected chunks");
349 if (first.id.empty() || second.id.empty() || first.id == second.id || points_.contains(first.id) ||
350 points_.contains(second.id))
351 return splineError<DomainOperation>(EditorStatus::Rejected, "editor.spline.append-chunk-id-invalid",
352 "Appended chunk points require two new distinct stable ids");
353 auto parsedFirst = parsePoint(pointValue(first));
354 auto parsedSecond = parsePoint(pointValue(second));
355 if (!parsedFirst.ok() || !parsedSecond.ok())
356 return splineError<DomainOperation>(EditorStatus::Rejected, "editor.spline.append-chunk-point-invalid",
357 "Appended chunk points must be finite and valid");
358 SplinePathDocument candidate(*this);
359 auto ordered = candidate.points();
360 auto appendedFirst = parsedFirst.value();
361 auto appendedSecond = parsedSecond.value();
362 appendedFirst.order = static_cast<std::int64_t>(ordered.size());
363 appendedFirst.breakBefore = !ordered.empty();
364 appendedSecond.order = appendedFirst.order + 1;
365 appendedSecond.breakBefore = false;
366 candidate.points_[appendedFirst.id] = appendedFirst;
367 candidate.points_[appendedSecond.id] = appendedSecond;
368 return eve::editing::applied<DomainOperation>(operation("spline.snapshot.replace.v1", "spline.snapshot.replace.v1",
369 id_, candidate.snapshotValue(), snapshotValue(),
370 appendedFirst.id));

373EditorResult<DomainOperation> SplinePathDocument::makeDeleteChunk(int chunk) const {
374 if (chunk < 0)
375 return splineError<DomainOperation>(EditorStatus::NotFound, "editor.spline.chunk-not-found",
376 "Spline chunk was not found");
377 auto ordered = points();
378 std::vector<std::vector<SplinePathControlPoint>> chunks;
379 for (const auto& point : ordered) {
380 if (chunks.empty() || point.breakBefore) chunks.emplace_back();
381 chunks.back().push_back(point);
382 }
383 if (static_cast<std::size_t>(chunk) >= chunks.size())
384 return splineError<DomainOperation>(EditorStatus::NotFound, "editor.spline.chunk-not-found",
385 "Spline chunk was not found");
386 chunks.erase(chunks.begin() + chunk);
387 SplinePathDocument candidate(*this);
388 candidate.points_.clear();
389 std::int64_t order = 0;
390 for (std::size_t chunkIndex = 0; chunkIndex < chunks.size(); ++chunkIndex) {
391 for (std::size_t pointIndex = 0; pointIndex < chunks[chunkIndex].size(); ++pointIndex) {
392 auto point = chunks[chunkIndex][pointIndex];
393 point.order = order++;
394 point.breakBefore = chunkIndex > 0u && pointIndex == 0u;
395 candidate.points_[point.id] = point;
396 }
397 }
398 return eve::editing::applied<DomainOperation>(operation("spline.snapshot.replace.v1", "spline.snapshot.replace.v1",
399 id_, candidate.snapshotValue(), snapshotValue(),
400 StableId("delete-chunk")));
401}

423EditorResult<DomainOperation> SplinePathDocument::makeCenterPoint(const StableId& point) const {
424 const auto ordered = points();
425 const auto found =
426 std::find_if(ordered.begin(), ordered.end(), [&](const auto& value) { return value.id == point; });
427 if (found == ordered.end())
428 return splineError<DomainOperation>(EditorStatus::NotFound, "editor.spline.center-point-missing",
429 "Spline point was not found");
430 const auto index = static_cast<std::size_t>(found - ordered.begin());
431 if (index == 0u || index + 1u >= ordered.size() || found->breakBefore || ordered[index + 1u].breakBefore)
432 return splineError<DomainOperation>(EditorStatus::Rejected, "editor.spline.center-point-endpoint",
433 "Only an interior point of one connected chunk can be centered");
434 auto updated = *found;
435 updated.x = (ordered[index - 1u].x + ordered[index + 1u].x) * 0.5;
436 updated.y = (ordered[index - 1u].y + ordered[index + 1u].y) * 0.5;
437 updated.z = (ordered[index - 1u].z + ordered[index + 1u].z) * 0.5;
438 return makeSetPoint(updated);

441EditorResult<DomainOperation> SplinePathDocument::makeMirrorAxis(std::string_view axis) const {
442 if (axis != "x" && axis != "y" && axis != "z")
443 return splineError<DomainOperation>(EditorStatus::Rejected, "editor.spline.mirror-axis-invalid",
444 "Spline mirror axis must be x, y, or z");
445 SplinePathDocument candidate(*this);
446 for (auto& [unused, point] : candidate.points_) {
447 (void)unused;
448 double* position = axis == "x" ? &point.x : axis == "y" ? &point.y : &point.z;
449 double* incoming = axis == "x" ? &point.inX : axis == "y" ? &point.inY : &point.inZ;
450 double* outgoing = axis == "x" ? &point.outX : axis == "y" ? &point.outY : &point.outZ;
451 *position = -*position;
452 *incoming = -*incoming;
453 *outgoing = -*outgoing;
454 }

507std::vector<SplinePathControlPoint> SplinePathDocument::points() const {
508 std::vector<SplinePathControlPoint> result;
509 result.reserve(points_.size());
510 for (const auto& [id, point] : points_) {
511 static_cast<void>(id);
512 result.push_back(point);
513 }
514 std::sort(result.begin(), result.end(), [](const auto& left, const auto& right) {
515 return left.order == right.order ? left.id < right.id : left.order < right.order;
516 });
517 return result;
518}

520EditorResult<procgen::SplinePath> SplinePathDocument::compilePath() const {
521 procgen::SplinePath path;
522 auto kind = path.setKindResult(settings_.kind);
523 if (!kind.ok())
524 return splineError<procgen::SplinePath>(EditorStatus::Rejected, "editor.spline.compile-kind",
525 kind.status().describe());
526 for (const auto& point : points()) {
527 auto added = path.addPointResult(
528 {static_cast<float>(point.x), static_cast<float>(point.y), static_cast<float>(point.z),
529 static_cast<float>(point.inX), static_cast<float>(point.inY), static_cast<float>(point.inZ),
530 static_cast<float>(point.outX), static_cast<float>(point.outY), static_cast<float>(point.outZ),
531 static_cast<float>(point.rollDegrees), static_cast<float>(point.scaleX), static_cast<float>(point.scaleY),
532 point.breakBefore, static_cast<float>(point.pitchDegrees), static_cast<float>(point.yawDegrees)});
533 if (!added.ok())
534 return splineError<procgen::SplinePath>(EditorStatus::Rejected, "editor.spline.compile-point",
535 added.status().describe());
536 }
537 path.setClosed(settings_.closed);
538 auto ready = path.evaluateResult(0.f);
539 if (!ready.ok())
540 return splineError<procgen::SplinePath>(EditorStatus::Rejected, "editor.spline.compile-incomplete",
541 ready.status().describe());
542 return eve::editing::applied<procgen::SplinePath>(std::move(path));
543}

555EditorResult<void> SplinePathDocument::loadSnapshot(const EditorValue& snapshot) {
556 const auto* schemaValue = field(snapshot, "schema");
557 const auto* versionValue = field(snapshot, "schemaVersion");
558 const auto* kindValue = field(snapshot, "kind");
559 const auto* closedValue = field(snapshot, "closed");
560 const auto* pointsValue = field(snapshot, "points");
561 const auto* schema = schemaValue ? schemaValue->getIf<std::string>() : nullptr;
562 const auto* version = versionValue ? versionValue->getIf<std::int64_t>() : nullptr;
563 const auto* points = pointsValue ? pointsValue->getIf<EditorValue::Array>() : nullptr;
564 if (!schema || *schema != "eve.procgen.splinePath" || !version || *version != 1 || !kindValue || !closedValue ||
565 !points)
566 return splineError<void>(EditorStatus::Unsupported, "editor.spline.invalid-snapshot",
567 "Spline snapshot schema is unsupported");
568 auto settings = parseSettings(EditorValue::Object{{"kind", *kindValue}, {"closed", *closedValue}});
569 if (!settings.ok())
570 return splineError<void>(settings.code(), "editor.spline.invalid-snapshot-settings",
571 "Spline snapshot settings are invalid");
572 SplinePathDocument candidate(id_);
573 candidate.settings_ = std::move(settings.value());
574 for (const auto& value : *points) {
575 auto parsed = parsePoint(value);
576 if (!parsed.ok() || candidate.points_.contains(parsed.value().id))
577 return splineError<void>(EditorStatus::Rejected, "editor.spline.invalid-snapshot-point",
578 "Spline snapshot contains invalid or duplicate points");
579 candidate.points_[parsed.value().id] = std::move(parsed.value());
580 }
581 if (candidate.settings_.closed && std::any_of(candidate.points_.begin(), candidate.points_.end(),
582 [](const auto& entry) { return entry.second.breakBefore; }))
583 return splineError<void>(EditorStatus::Rejected, "editor.spline.invalid-snapshot-breaks",
584 "Closed spline snapshot cannot contain disconnected chunks");
585 candidate.revision_ = revision_ + 1;
586 candidate.dirty_.clear();
587 *this = std::move(candidate);
588 return eve::editing::applied<void>();
589}

591EditorResult<procgen::SplinePath> compileSplinePathSnapshot(const EditorValue& snapshot) {
592 SplinePathDocument document("spline-preview");
593 auto loaded = document.loadSnapshot(snapshot);
594 if (!loaded.ok())
595 return splineError<procgen::SplinePath>(loaded.code(), "editor.spline.snapshot-compile",
596 loaded.status().describe());
597 return document.compilePath();
598}
