16bool finitePose(const AnimPose& pose, int count) {
17 if (count <= 0 || pose.getBoneCount() != count) return false;
18 for (int i = 0; i < count; ++i) {
19 const auto& t = pose.local(i);
20 for (float v : {t.px, t.py, t.pz, t.qx, t.qy, t.qz, t.qw, t.sx, t.sy, t.sz})
21 if (!std::isfinite(v)) return false;
22 const double norm = double(t.qx) * t.qx + double(t.qy) * t.qy + double(t.qz) * t.qz + double(t.qw) * t.qw;
23 if (norm < 1e-12 || norm > 1e12) return false;
24 }
25 return true;
26}
