23std::vector<FragmentCollisionRect> decompose(const eve::pixelworld::PixelFragment& fragment) {
24 const int width = fragment.width, height = fragment.height;
25 std::vector<std::uint8_t> consumed(std::size_t(width) * std::size_t(height));
26 std::vector<FragmentCollisionRect> result;
27 const auto occupied = [&](int x, int y) {
28 return fragment.cells[std::size_t(y) * std::size_t(width) + std::size_t(x)].material !=
29 eve::pixelworld::MaterialId::Air;
30 };
31 for (int y = 0; y < height; ++y)
32 for (int x = 0; x < width; ++x) {
33 const std::size_t start = std::size_t(y) * std::size_t(width) + std::size_t(x);
34 if (consumed[start] || !occupied(x, y)) continue;
35 int rectWidth = 1;
36 while (x + rectWidth < width && !consumed[start + std::size_t(rectWidth)] &&
37 occupied(x + rectWidth, y))
38 ++rectWidth;
39 int rectHeight = 1;
40 for (; y + rectHeight < height; ++rectHeight) {
41 bool complete = true;
42 for (int ox = 0; ox < rectWidth; ++ox) {
43 const std::size_t index = std::size_t(y + rectHeight) * std::size_t(width) +
44 std::size_t(x + ox);
45 if (consumed[index] || !occupied(x + ox, y + rectHeight)) {
46 complete = false;
47 break;
48 }
49 }
50 if (!complete) break;
51 }
52 for (int oy = 0; oy < rectHeight; ++oy)
53 for (int ox = 0; ox < rectWidth; ++ox)
54 consumed[std::size_t(y + oy) * std::size_t(width) + std::size_t(x + ox)] = 1;
55 result.push_back({x, y, rectWidth, rectHeight});
56 }
57 return result;
58}

60eve::pixelworld::PixelFragment rotated(const eve::pixelworld::PixelFragment& source, int turns) {
61 turns = ((turns % 4) + 4) % 4;
62 eve::pixelworld::PixelFragment result = source;
63 if (turns == 0) return result;
64 result.width = (turns % 2 == 0) ? source.width : source.height;
65 result.height = (turns % 2 == 0) ? source.height : source.width;
66 result.cells.assign(std::size_t(result.width) * std::size_t(result.height), {});
67 for (int y = 0; y < source.height; ++y)
68 for (int x = 0; x < source.width; ++x) {
69 int rx = 0, ry = 0;
70 if (turns == 1) {
71 rx = source.height - 1 - y;
72 ry = x;
73 } else if (turns == 2) {
74 rx = source.width - 1 - x;
75 ry = source.height - 1 - y;
76 } else {
77 rx = y;
78 ry = source.width - 1 - x;
79 }
80 result.cells[std::size_t(ry) * std::size_t(result.width) + std::size_t(rx)] =
81 source.cells[std::size_t(y) * std::size_t(source.width) + std::size_t(x)];
82 }
83 return result;
84}

97int direction(const GridEdge& edge) {
98 if (edge.to.x > edge.from.x) return 0;
99 if (edge.to.y > edge.from.y) return 1;
100 if (edge.to.x < edge.from.x) return 2;
101 return 3;
102}

104int turnRank(int previous, int next) {
105 const int turn = (next - previous + 4) % 4;
106 if (turn == 1) return 0;
107 if (turn == 0) return 1;
108 if (turn == 3) return 2;
109 return 3;
110}

112bool collinear(GridPoint a, GridPoint b, GridPoint c) {
113 return (b.x - a.x) * (c.y - b.y) == (b.y - a.y) * (c.x - b.x);
114}

124eve::Result<std::vector<TerrainCollisionContour>> extractTerrainContours(
125 const eve::pixelworld::PixelWorld& pixelWorld, int chunkX, int chunkY,
126 std::uint32_t maximumVertices) {
127 if (maximumVertices < 2)
128 return failure<std::vector<TerrainCollisionContour>>(
129 eve::DiagnosticCode::InvalidArgument, "maximumVertices must be at least two",
130 "maximumVertices");
131 constexpr int size = eve::pixelworld::kPixelChunkSize;
132 const std::int64_t wideOriginX = std::int64_t(chunkX) * size;
133 const std::int64_t wideOriginY = std::int64_t(chunkY) * size;
134 if (wideOriginX < std::numeric_limits<int>::min() ||
135 wideOriginX + size > std::numeric_limits<int>::max() ||
136 wideOriginY < std::numeric_limits<int>::min() ||
137 wideOriginY + size > std::numeric_limits<int>::max())
138 return failure<std::vector<TerrainCollisionContour>>(
139 eve::DiagnosticCode::InvalidArgument, "Chunk coordinates exceed world coordinate range",
140 "chunk");
141 const int originX = int(wideOriginX), originY = int(wideOriginY);
142 const auto solid = [&](int localX, int localY) {
143 return pixelWorld.isSolidMaterial(
144 pixelWorld.getCell(originX + localX, originY + localY).material);
145 };
146 std::vector<GridEdge> edges;
147 edges.reserve(size * 4);
148 for (int y = 0; y < size; ++y)
149 for (int x = 0; x < size; ++x) {
150 if (!solid(x, y)) continue;
151 if (!solid(x, y - 1)) edges.push_back({{x, y}, {x + 1, y}});
152 if (!solid(x + 1, y)) edges.push_back({{x + 1, y}, {x + 1, y + 1}});
153 if (!solid(x, y + 1)) edges.push_back({{x + 1, y + 1}, {x, y + 1}});
154 if (!solid(x - 1, y)) edges.push_back({{x, y + 1}, {x, y}});
155 }
156 std::map<GridPoint, std::vector<std::size_t>> outgoing;
157 for (std::size_t index = 0; index < edges.size(); ++index)
158 outgoing[edges[index].from].push_back(index);
159 std::vector<std::uint8_t> used(edges.size());
160 std::vector<TerrainCollisionContour> contours;
161 std::uint64_t vertexCount = 0;
162 for (std::size_t first = 0; first < edges.size(); ++first) {
163 if (used[first]) continue;
164 std::vector<GridPoint> points{edges[first].from};
165 std::size_t current = first;
166 bool loop = false;
167 while (true) {
168 used[current] = 1;
169 const GridPoint end = edges[current].to;
170 if (end == points.front()) {
171 loop = true;
172 break;
173 }
174 points.push_back(end);
175 const auto found = outgoing.find(end);
176 if (found == outgoing.end()) break;
177 std::size_t next = edges.size();
178 int bestRank = 5;
179 for (const std::size_t candidate : found->second) {
180 if (used[candidate]) continue;
181 const int rank = turnRank(direction(edges[current]), direction(edges[candidate]));
182 if (rank < bestRank || (rank == bestRank && candidate < next)) {
183 bestRank = rank;
184 next = candidate;
185 }
186 }
187 if (next == edges.size()) break;
188 current = next;
189 }
190 bool changed = true;
191 while (changed && points.size() > (loop ? 3U : 2U)) {
192 changed = false;
193 for (std::size_t index = 0; index < points.size(); ++index) {
194 if (!loop && (index == 0 || index + 1 == points.size())) continue;
195 const std::size_t before = (index + points.size() - 1) % points.size();
196 const std::size_t after = (index + 1) % points.size();
197 if (collinear(points[before], points[index], points[after])) {
198 points.erase(points.begin() + std::ptrdiff_t(index));
199 changed = true;
200 break;
201 }
202 }
203 }
204 if (points.size() < (loop ? 3U : 2U)) continue;
205 vertexCount += points.size();
206 if (vertexCount > maximumVertices)
207 return failure<std::vector<TerrainCollisionContour>>(
208 eve::DiagnosticCode::PreconditionViolation,
209 "terrain contour exceeds maximumVertices", "maximumVertices");
210 TerrainCollisionContour contour;
211 contour.loop = loop;
212 contour.vertices.reserve(points.size() * 2);
213 for (const GridPoint point : points) {
214 contour.vertices.push_back(float(point.x));
215 contour.vertices.push_back(float(point.y));
216 }
217 contours.push_back(std::move(contour));
218 }
219 return eve::Result<std::vector<TerrainCollisionContour>>::success(std::move(contours));
220}

222eve::Result<PixelTerrainContact> probeTerrainCircle(
223 const eve::pixelworld::PixelWorld& pixelWorld, float centerX, float centerY, float radius,
224 std::uint32_t maximumCells) {
225 if (!std::isfinite(centerX) || !std::isfinite(centerY) || !std::isfinite(radius) ||
226 radius <= 0.f || maximumCells == 0)
227 return failure<PixelTerrainContact>(eve::DiagnosticCode::InvalidArgument,
228 "circle probe parameters are invalid", "probe");
229 if (!validCellBounds(centerX - radius, centerX + radius) ||
230 !validCellBounds(centerY - radius, centerY + radius))
231 return failure<PixelTerrainContact>(eve::DiagnosticCode::InvalidArgument,
232 "circle probe exceeds world coordinate range", "probe");
233 const int minX = int(std::floor(centerX - radius));
234 const int maxX = int(std::floor(centerX + radius));
235 const int minY = int(std::floor(centerY - radius));
236 const int maxY = int(std::floor(centerY + radius));
237 const std::uint64_t width = std::uint64_t(std::int64_t(maxX) - minX + 1);
238 const std::uint64_t height = std::uint64_t(std::int64_t(maxY) - minY + 1);
239 if (width > maximumCells || height > maximumCells || width * height > maximumCells)
240 return failure<PixelTerrainContact>(eve::DiagnosticCode::PreconditionViolation,
241 "circle probe exceeds maximumCells", "maximumCells");
242 PixelTerrainContact best;
243 for (int y = minY; y <= maxY; ++y)
244 for (int x = minX; x <= maxX; ++x) {
245 const auto material = pixelWorld.getCell(x, y).material;
246 if (!pixelWorld.isSolidMaterial(material)) continue;
247 const float closestX = std::clamp(centerX, float(x), float(x + 1));
248 const float closestY = std::clamp(centerY, float(y), float(y + 1));
249 const float dx = centerX - closestX, dy = centerY - closestY;
250 const float distanceSquared = dx * dx + dy * dy;
251 if (distanceSquared >= radius * radius) continue;
252 const float distance = std::sqrt(distanceSquared);
253 const float depth = radius - distance;
254 if (best.hit && depth < best.depth) continue;
255 if (best.hit && depth == best.depth && std::pair(y, x) >= std::pair(best.cellY, best.cellX))
256 continue;
257 best.hit = true;
258 best.material = material;
259 best.cellX = x;
260 best.cellY = y;
261 best.pointX = closestX;
262 best.pointY = closestY;
263 best.depth = depth;
264 if (distance > 0.f) {
265 best.normalX = dx / distance;
266 best.normalY = dy / distance;
267 } else {
268 const float left = centerX - float(x), right = float(x + 1) - centerX;
269 const float top = centerY - float(y), bottom = float(y + 1) - centerY;
270 const float nearest = std::min({left, right, top, bottom});
271 if (nearest == left) best.normalX = -1.f;
272 else if (nearest == right) best.normalX = 1.f;
273 else if (nearest == top) best.normalY = -1.f;
274 else best.normalY = 1.f;
275 }
276 }
277 return eve::Result<PixelTerrainContact>::success(best);
278}

280eve::Result<PixelTerrainContact> sweepTerrainCircle(
281 const eve::pixelworld::PixelWorld& pixelWorld, float startX, float startY, float endX,
282 float endY, float radius, std::uint32_t maximumCells) {
283 if (!std::isfinite(startX) || !std::isfinite(startY) || !std::isfinite(endX) ||
284 !std::isfinite(endY) || !std::isfinite(radius) || radius <= 0.f || maximumCells == 0)
285 return failure<PixelTerrainContact>(eve::DiagnosticCode::InvalidArgument,
286 "circle sweep parameters are invalid", "sweep");
287 if (!validCellBounds(std::min(startX, endX) - radius,
288 std::max(startX, endX) + radius) ||
289 !validCellBounds(std::min(startY, endY) - radius,
290 std::max(startY, endY) + radius))
291 return failure<PixelTerrainContact>(eve::DiagnosticCode::InvalidArgument,
292 "circle sweep exceeds world coordinate range", "sweep");
293 auto initial = probeTerrainCircle(pixelWorld, startX, startY, radius, maximumCells);
294 if (!initial.ok()) return initial;
295 if (initial.value().hit) return initial;
296 const int minX = int(std::floor(std::min(startX, endX) - radius));
297 const int maxX = int(std::floor(std::max(startX, endX) + radius));
298 const int minY = int(std::floor(std::min(startY, endY) - radius));
299 const int maxY = int(std::floor(std::max(startY, endY) + radius));
300 const std::uint64_t width = std::uint64_t(std::int64_t(maxX) - minX + 1);
301 const std::uint64_t height = std::uint64_t(std::int64_t(maxY) - minY + 1);
302 if (width > maximumCells || height > maximumCells || width * height > maximumCells)
303 return failure<PixelTerrainContact>(eve::DiagnosticCode::PreconditionViolation,
304 "circle sweep exceeds maximumCells", "maximumCells");
305 const float deltaX = endX - startX, deltaY = endY - startY;
306 PixelTerrainContact best;
307 best.fraction = 1.f;
308 for (int y = minY; y <= maxY; ++y)
309 for (int x = minX; x <= maxX; ++x) {
310 const auto material = pixelWorld.getCell(x, y).material;
311 if (!pixelWorld.isSolidMaterial(material)) continue;
312 float cellFraction = 2.f, cellNormalX = 0.f, cellNormalY = 0.f;
313 const auto consider = [&](float fraction, float nx, float ny) {
314 if (fraction >= 0.f && fraction <= 1.f && fraction < cellFraction) {
315 cellFraction = fraction;
316 cellNormalX = nx;
317 cellNormalY = ny;
318 }
319 };
320 if (deltaX > 0.f) {
321 const float fraction = (float(x) - radius - startX) / deltaX;
322 const float centerY = startY + deltaY * fraction;
323 if (centerY >= float(y) && centerY <= float(y + 1))
324 consider(fraction, -1.f, 0.f);
325 } else if (deltaX < 0.f) {
326 const float fraction = (float(x + 1) + radius - startX) / deltaX;
327 const float centerY = startY + deltaY * fraction;
328 if (centerY >= float(y) && centerY <= float(y + 1))
329 consider(fraction, 1.f, 0.f);
330 }
331 if (deltaY > 0.f) {
332 const float fraction = (float(y) - radius - startY) / deltaY;
333 const float centerX = startX + deltaX * fraction;
334 if (centerX >= float(x) && centerX <= float(x + 1))
335 consider(fraction, 0.f, -1.f);
336 } else if (deltaY < 0.f) {
337 const float fraction = (float(y + 1) + radius - startY) / deltaY;
338 const float centerX = startX + deltaX * fraction;
339 if (centerX >= float(x) && centerX <= float(x + 1))
340 consider(fraction, 0.f, 1.f);
341 }
342 const float quadraticA = deltaX * deltaX + deltaY * deltaY;
343 if (quadraticA > 0.f) {
344 struct Corner {
345 float x;
346 float y;
347 int quadrantX;
348 int quadrantY;
349 };
350 const Corner corners[] = {
351 {float(x), float(y), -1, -1}, {float(x + 1), float(y), 1, -1},
352 {float(x + 1), float(y + 1), 1, 1}, {float(x), float(y + 1), -1, 1}};
353 for (const Corner corner : corners) {
354 const float offsetX = startX - corner.x, offsetY = startY - corner.y;
355 const float quadraticB = 2.f * (offsetX * deltaX + offsetY * deltaY);
356 const float quadraticC = offsetX * offsetX + offsetY * offsetY - radius * radius;
357 const float discriminant = quadraticB * quadraticB - 4.f * quadraticA * quadraticC;
358 if (discriminant < 0.f) continue;
359 const float fraction =
360 (-quadraticB - std::sqrt(discriminant)) / (2.f * quadraticA);
361 if (fraction < 0.f || fraction > 1.f) continue;
362 const float hitX = startX + deltaX * fraction;
363 const float hitY = startY + deltaY * fraction;
364 if ((corner.quadrantX < 0 && hitX > corner.x) ||
365 (corner.quadrantX > 0 && hitX < corner.x) ||
366 (corner.quadrantY < 0 && hitY > corner.y) ||
367 (corner.quadrantY > 0 && hitY < corner.y))
368 continue;
369 consider(fraction, (hitX - corner.x) / radius,
370 (hitY - corner.y) / radius);
371 }
372 }
373 if (cellFraction > 1.f) continue;
374 if (best.hit && cellFraction > best.fraction) continue;
375 if (best.hit && cellFraction == best.fraction &&
376 std::pair(y, x) >= std::pair(best.cellY, best.cellX))
377 continue;
378 best.hit = true;
379 best.material = material;
380 best.cellX = x;
381 best.cellY = y;
382 best.fraction = cellFraction;
383 best.pointX = startX + deltaX * cellFraction;
384 best.pointY = startY + deltaY * cellFraction;
385 best.normalX = cellNormalX;
386 best.normalY = cellNormalY;
387 }
388 if (!best.hit) best.fraction = 1.f;
389 return eve::Result<PixelTerrainContact>::success(best);
390}

397eve::Result<std::unique_ptr<PixelFragmentBody>> PixelFragmentBody::create(
398 eve::physics::World& world, eve::pixelworld::PixelFragment fragment, FragmentBodyConfig config) {
399 if (fragment.width <= 0 || fragment.height <= 0 || fragment.id == 0 ||
400 std::uint64_t(fragment.width) * std::uint64_t(fragment.height) != fragment.cells.size())
401 return failure<std::unique_ptr<PixelFragmentBody>>(
402 eve::DiagnosticCode::InvalidArgument, "fragment bitmap metadata is invalid", "fragment");
403 if (!std::isfinite(config.density) || config.density <= 0.f ||
404 !std::isfinite(config.friction) || config.friction < 0.f ||
405 !std::isfinite(config.restitution) || config.restitution < 0.f || config.maximumFixtures == 0)
406 return failure<std::unique_ptr<PixelFragmentBody>>(
407 eve::DiagnosticCode::InvalidArgument, "fragment body config is invalid", "config");
408 auto rectangles = decompose(fragment);
409 if (rectangles.empty() || rectangles.size() > config.maximumFixtures)
410 return failure<std::unique_ptr<PixelFragmentBody>>(
411 eve::DiagnosticCode::PreconditionViolation,
412 "fragment collision decomposition is empty or exceeds maximumFixtures", "maximumFixtures");
413
414 eve::physics::Body* body = nullptr;
415 try {
416 const float centerX = float(fragment.originX) + float(fragment.width) * 0.5f;
417 const float centerY = float(fragment.originY) + float(fragment.height) * 0.5f;
418 body = world.newBody("dynamic", centerX, centerY);
419 for (const FragmentCollisionRect& rect : rectangles) {
420 const float offsetX = float(rect.x) + float(rect.width) * 0.5f - float(fragment.width) * 0.5f;
421 const float offsetY = float(rect.y) + float(rect.height) * 0.5f - float(fragment.height) * 0.5f;
422 body->newRectangleFixtureAt(float(rect.width), float(rect.height), offsetX, offsetY,
423 config.density, config.friction, config.restitution);
424 }
425 auto link = eve::physics::PhysicsLink::fromBody(*body);
426 if (!link.ok()) {
427 body->destroy();
428 return eve::Result<std::unique_ptr<PixelFragmentBody>>::failure(link.status());
429 }
430 return eve::Result<std::unique_ptr<PixelFragmentBody>>::success(
431 std::unique_ptr<PixelFragmentBody>(new PixelFragmentBody(
432 std::move(fragment), std::move(link).takeValue(), std::move(rectangles))));
433 } catch (const std::exception& error) {
434 if (body && body->isValid()) body->destroy();
435 return failure<std::unique_ptr<PixelFragmentBody>>(
436 eve::DiagnosticCode::Failed, std::string("failed to create fragment fixtures: ") + error.what(),
437 "physics");
438 }
439}

449eve::Result<FragmentSettleReceipt> PixelFragmentBody::settleIfSleeping(
450 eve::physics::World& physicsWorld, eve::pixelworld::PixelWorld& pixelWorld) {
451 if (rasterized_ || physicsReleased_)
452 return failure<FragmentSettleReceipt>(eve::DiagnosticCode::PreconditionViolation,
453 "fragment body is no longer active", "state");
454 auto resolved = link_.resolve(physicsWorld);
455 if (!resolved.ok()) return eve::Result<FragmentSettleReceipt>::failure(resolved.status());
456 eve::physics::Body* body = resolved.value();
457 if (body->isAwake())
458 return eve::Result<FragmentSettleReceipt>::success(
459 {FragmentSettleDisposition::StillAwake, fragment_.id, 0, 0});
460
461 constexpr double halfPi = 1.57079632679489661923;
462 const int turns = int(std::llround(double(body->getAngle()) / halfPi));
463 auto candidate = rotated(fragment_, turns);
464 const int originX = int(std::lround(double(body->getX()) - double(candidate.width) * 0.5));
465 const int originY = int(std::lround(double(body->getY()) - double(candidate.height) * 0.5));
466 auto rasterized = pixelWorld.rasterizeFragment(candidate, originX, originY);
467 if (!rasterized.ok()) return eve::Result<FragmentSettleReceipt>::failure(rasterized.status());
468 body->destroy();
469 physicsReleased_ = true;
470 rasterized_ = true;
471 return eve::Result<FragmentSettleReceipt>::success(
472 {FragmentSettleDisposition::Rasterized, fragment_.id,
473 rasterized.value().cellsPlaced, ((turns % 4) + 4) % 4});
474}

498eve::Result<TerrainCollisionSyncReceipt> PixelTerrainCollisionCache::sync(
499 eve::physics::World& physicsWorld, const eve::pixelworld::PixelWorld& pixelWorld,
500 std::uint32_t maximumFixturesPerChunk) {
501 if (maximumFixturesPerChunk == 0)
502 return failure<TerrainCollisionSyncReceipt>(eve::DiagnosticCode::InvalidArgument,
503 "maximumFixturesPerChunk must be positive",
504 "maximumFixturesPerChunk");
505 const bool samePixelWorld = impl_->pixelWorld == pixelWorld.worldLink();
506 const bool samePhysicsWorld = impl_->physicsWorld == physicsWorld.runtimeHandle();
507 const std::uint64_t since = (samePixelWorld && samePhysicsWorld) ? impl_->revision : 0;
508 const auto changed = pixelWorld.snapshotChangedChunks(since);
509 std::set<Impl::Coord> rebuildCoords;
510 for (const auto& chunk : changed) {
511 const Impl::Coord coord{chunk.x, chunk.y};
512 rebuildCoords.insert(coord);
513 constexpr Impl::Coord neighbors[] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};
514 for (const auto& [dx, dy] : neighbors) {
515 const Impl::Coord neighbor{chunk.x + dx, chunk.y + dy};
516 if (impl_->bodies.contains(neighbor)) rebuildCoords.insert(neighbor);
517 }
518 }
519
520 struct Candidate {
521 Impl::Coord coord;
522 eve::physics::PhysicsLink link;
523 std::uint32_t fixtures = 0;
524 };
525 std::vector<Candidate> staged;
526 staged.reserve(rebuildCoords.size());
527 const auto destroyStaged = [&] {
528 for (const Candidate& candidate : staged) {
529 auto resolved = candidate.link.resolve(physicsWorld);
530 if (resolved.ok()) resolved.value()->destroy();
531 }
532 };
533
534 for (const auto& coord : rebuildCoords) {
535 auto extracted = extractTerrainContours(pixelWorld, coord.first, coord.second);
536 if (!extracted.ok()) {
537 destroyStaged();
538 return eve::Result<TerrainCollisionSyncReceipt>::failure(extracted.status());
539 }
540 auto contours = std::move(extracted).takeValue();
541 if (contours.size() > maximumFixturesPerChunk) {
542 destroyStaged();
543 return failure<TerrainCollisionSyncReceipt>(
544 eve::DiagnosticCode::PreconditionViolation,
545 "terrain Chunk contours exceed maximumFixturesPerChunk", "chunk");
546 }
547 if (contours.empty()) {
548 staged.push_back({coord, {}, 0});
549 continue;
550 }
551 eve::physics::Body* body = nullptr;
552 try {
553 body = physicsWorld.newBody(
554 "static", float(coord.first) * float(eve::pixelworld::kPixelChunkSize),
555 float(coord.second) * float(eve::pixelworld::kPixelChunkSize));
556 for (const TerrainCollisionContour& contour : contours)
557 body->newChainFixture(contour.vertices, contour.loop, 0.5f, 0.f);
558 auto link = eve::physics::PhysicsLink::fromBody(*body);
559 if (!link.ok()) {
560 body->destroy();
561 destroyStaged();
562 return eve::Result<TerrainCollisionSyncReceipt>::failure(link.status());
563 }
564 staged.push_back({coord, std::move(link).takeValue(),
565 std::uint32_t(contours.size())});
566 } catch (const std::exception& error) {
567 if (body && body->isValid()) body->destroy();
568 destroyStaged();
569 return failure<TerrainCollisionSyncReceipt>(
570 eve::DiagnosticCode::Failed,
571 std::string("failed to stage terrain collision body: ") + error.what(), "physics");
572 }
573 }
574
575 TerrainCollisionSyncReceipt receipt;
576 receipt.sourceRevision = pixelWorld.revision();
577 receipt.chunksRebuilt = std::uint32_t(staged.size());
578 if (!samePhysicsWorld) {
579 impl_->bodies.clear();
580 } else if (!samePixelWorld) {
581 for (const auto& [coord, link] : impl_->bodies) {
582 (void)coord;
583 auto resolved = link.resolve(physicsWorld);
584 if (resolved.ok()) resolved.value()->destroy();
585 ++receipt.bodiesRemoved;
586 }
587 impl_->bodies.clear();
588 }
589 for (const Candidate& candidate : staged) {
590 const auto old = impl_->bodies.find(candidate.coord);
591 if (old != impl_->bodies.end()) {
592 auto resolved = old->second.resolve(physicsWorld);
593 if (resolved.ok()) resolved.value()->destroy();
594 impl_->bodies.erase(old);
595 ++receipt.bodiesRemoved;
596 }
597 if (candidate.fixtures != 0) {
598 impl_->bodies.emplace(candidate.coord, candidate.link);
599 receipt.fixturesCreated += candidate.fixtures;
600 }
601 }
602 impl_->pixelWorld = pixelWorld.worldLink();
603 impl_->physicsWorld = physicsWorld.runtimeHandle();
604 impl_->revision = pixelWorld.revision();
605 return eve::Result<TerrainCollisionSyncReceipt>::success(receipt);
606}

608eve::Result<void> PixelTerrainCollisionCache::clearPhysics(eve::physics::World& physicsWorld) {
609 if (impl_->physicsWorld.isValid() && impl_->physicsWorld != physicsWorld.runtimeHandle())
610 return failure<void>(eve::DiagnosticCode::StaleHandle,
611 "terrain collision cache belongs to another physics world", "physicsWorld");
612 for (const auto& [coord, link] : impl_->bodies) {
613 (void)coord;
614 auto resolved = link.resolve(physicsWorld);
615 if (resolved.ok()) resolved.value()->destroy();
616 }
617 impl_->bodies.clear();
618 impl_->pixelWorld = {};
619 impl_->physicsWorld = eve::physics::PhysicsWorldHandle::invalid();
620 impl_->revision = 0;
621 return eve::Result<void>::success();
622}
