64eve::Result<int> StoryEventCatalogue::replaceFromJsonStrict(const std::string& json) {
65 std::string parseError;
66 const auto document = eve::json::Document::parse(json, &parseError);
67 if (!document.valid())
68 return storyFailure<int>(eve::DiagnosticCode::ParseError, parseError.empty() ? "invalid JSON" : parseError,
69 "$");
70 const auto root = document.root();
71 if (!root.isObject())
72 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument, "story-event document root must be an object",
73 "$");
74 const std::unordered_set<std::string> rootFields = {"schema", "version", "events"};
75 for (const auto& key : root.keys())
76 if (!rootFields.contains(key))
77 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument,
78 "story-event document contains an unknown field", "$." + key);
79 const auto schema = root.get("schema");
80 const auto version = root.get("version");
81 const auto events = root.get("events");
82 if (!schema.isString() || schema.asString() != "eve.rpg.story-events")
83 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument, "story-event schema id is invalid", "$.schema");
84 if (!version.isInt64() || version.asInt64() != 1)
85 return storyFailure<int>(eve::DiagnosticCode::UnknownVersion, "unsupported story-event schema version",
86 "$.version");
87 if (!events.isArray() || events.size() == 0 || events.size() > 256)
88 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument, "events must contain between 1 and 256 entries",
89 "$.events");
90
91 const std::unordered_set<std::string> eventFields = {"id", "repeatable", "steps"};
92 const std::unordered_set<std::string> stepFields = {"kind", "reference", "actorId", "x", "y", "duration"};
93 std::unordered_map<std::string, StoryEventDefinition> proposed;
94 for (std::size_t index = 0; index < events.size(); ++index) {
95 const auto object = events.at(index);
96 const std::string path = "$.events[" + std::to_string(index) + "]";
97 if (!object.isObject())
98 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument, "story event must be an object", path);
99 for (const auto& key : object.keys())
100 if (!eventFields.contains(key))
101 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument, "story event contains an unknown field",
102 path + "." + key);
103 const auto id = object.get("id");
104 const auto repeatable = object.get("repeatable");
105 const auto steps = object.get("steps");
106 if (!id.isString() || !validText(id.asString()))
107 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument,
108 "story event id must be a stable non-empty id", path + ".id");
109 if (proposed.contains(id.asString()))
110 return storyFailure<int>(eve::DiagnosticCode::AlreadyExists, "duplicate story event id", path + ".id");
111 if (!repeatable.isBool())
112 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument, "repeatable must be boolean",
113 path + ".repeatable");
114 if (!steps.isArray() || steps.size() == 0 || steps.size() > 128)
115 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument,
116 "steps must contain between 1 and 128 entries", path + ".steps");
117 StoryEventDefinition definition;
118 definition.id = id.asString();
119 definition.repeatable = repeatable.asBool();
120 for (std::size_t stepIndex = 0; stepIndex < steps.size(); ++stepIndex) {
121 const auto value = steps.at(stepIndex);
122 const std::string stepPath = path + ".steps[" + std::to_string(stepIndex) + "]";
123 if (!value.isObject())
124 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument, "story-event step must be an object",
125 stepPath);
126 for (const auto& key : value.keys())
127 if (!stepFields.contains(key))
128 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument,
129 "story-event step contains an unknown field", stepPath + "." + key);
130 const auto kindValue = value.get("kind");
131 const auto reference = value.get("reference");
132 const auto actorId = value.get("actorId");
133 const auto x = value.get("x");
134 const auto y = value.get("y");
135 const auto duration = value.get("duration");
136 StoryEventStep parsed;
137 if (!kindValue.isString() || !parseKind(kindValue.asString(), parsed.kind))
138 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument, "story-event step kind is invalid",
139 stepPath + ".kind");
140 if (!reference.isString() || !actorId.isString() || !validText(reference.asString(), true) ||
141 !validText(actorId.asString(), true) || !x.isNumber() || !y.isNumber() || !duration.isNumber() ||
142 !std::isfinite(x.asDouble()) || !std::isfinite(y.asDouble()) || !std::isfinite(duration.asDouble()) ||
143 duration.asDouble() < 0.0)
144 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument, "story-event step fields are invalid",
145 stepPath);
146 const bool needsReference =
147 parsed.kind == StoryEventStepKind::Dialogue || parsed.kind == StoryEventStepKind::Message;
148 const bool needsActor = parsed.kind == StoryEventStepKind::Move;
149 const bool needsDuration = parsed.kind == StoryEventStepKind::Wait;
150 if (needsReference != !reference.asString().empty())
151 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument,
152 "reference presence does not match step kind", stepPath + ".reference");
153 if (needsActor != !actorId.asString().empty())
154 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument,
155 "actorId presence does not match step kind", stepPath + ".actorId");
156 if (needsDuration && duration.asDouble() <= 0.0)
157 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument, "wait duration must be positive",
158 stepPath + ".duration");
159 parsed.reference = reference.asString();
160 parsed.actorId = actorId.asString();
161 parsed.x = x.asDouble();
162 parsed.y = y.asDouble();
163 parsed.duration = duration.asDouble();
164 definition.steps.push_back(std::move(parsed));
165 }
166 proposed.emplace(definition.id, std::move(definition));
167 }
168 definitions() = std::move(proposed);
169 return eve::Result<int>::success(static_cast<int>(definitions().size()));
170}

180eve::Result<void> StoryEventSession::begin(const std::string& eventId, GameState* gameState) {
181 if (!gameState)
182 return storyFailure<void>(eve::DiagnosticCode::InvalidArgument, "begin requires a GameState owner",
183 "gameState");
184 const auto* definition = StoryEventCatalogue::find(eventId);
185 if (!definition)
186 return storyFailure<void>(eve::DiagnosticCode::NotFound, "story event is not registered", "eventId");
187 const std::string scope = eventScope(eventId);
188 const bool hasCompleted = gameState->hasSelfVariable(scope, "completed");
189 const double completedValue = hasCompleted ? gameState->getSelfVariable(scope, "completed") : 0.0;
190 if (hasCompleted && completedValue != 0.0 && completedValue != 1.0)
191 return storyFailure<void>(eve::DiagnosticCode::InvariantViolation,
192 "persisted story-event completion flag is invalid", scope + ".completed");
193 const bool completed = completedValue == 1.0;
194 if (completed && !definition->repeatable)
195 return storyFailure<void>(eve::DiagnosticCode::Conflict, "non-repeatable story event is already complete",
196 "eventId");
197 double storedCursor =
198 gameState->hasSelfVariable(scope, "cursor") ? gameState->getSelfVariable(scope, "cursor") : 0.0;
199 if (completed && definition->repeatable) storedCursor = 0.0;
200 if (!std::isfinite(storedCursor) || std::floor(storedCursor) != storedCursor || storedCursor < 0.0 ||
201 storedCursor >= static_cast<double>(definition->steps.size()))
202 return storyFailure<void>(eve::DiagnosticCode::InvariantViolation, "persisted story-event cursor is invalid",
203 scope + ".cursor");
204
205 StoryEventDefinition candidate = *definition;
206 definition_ = std::move(candidate);
207 cursor_ = static_cast<int>(storedCursor);
208 active_ = true;
209 finished_ = false;
210 if (completed) {
211 gameState->setSelfVariable(scope, "cursor", 0.0);
212 gameState->setSelfVariable(scope, "completed", 0.0);
213 }
214 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
215}

217eve::Result<int> StoryEventSession::advance(GameState* gameState) {
218 if (!gameState)
219 return storyFailure<int>(eve::DiagnosticCode::InvalidArgument, "advance requires a GameState owner",
220 "gameState");
221 if (!active_ || !current())
222 return storyFailure<int>(eve::DiagnosticCode::PreconditionViolation, "story event has no active step",
223 "session");
224 const std::string scope = eventScope(definition_.id);
225 const double persistedCursor =
226 gameState->hasSelfVariable(scope, "cursor") ? gameState->getSelfVariable(scope, "cursor") : 0.0;
227 const double persistedCompleted =
228 gameState->hasSelfVariable(scope, "completed") ? gameState->getSelfVariable(scope, "completed") : 0.0;
229 if (!std::isfinite(persistedCursor) || persistedCursor != static_cast<double>(cursor_) || persistedCompleted != 0.0)
230 return storyFailure<int>(eve::DiagnosticCode::Conflict,
231 "story-event cursor changed since this step was presented", scope + ".cursor");
232 const int next = cursor_ + 1;
233 gameState->setSelfVariable(scope, "cursor", static_cast<double>(next));
234 if (next == static_cast<int>(definition_.steps.size())) {
235 gameState->setSelfVariable(scope, "completed", 1.0);
236 cursor_ = next;
237 active_ = false;
238 finished_ = true;
239 return eve::Result<int>::success(0, eve::Status::success(eve::StatusCode::Applied));
240 }
241 cursor_ = next;
242 return eve::Result<int>::success(static_cast<int>(definition_.steps.size()) - cursor_,
243 eve::Status::success(eve::StatusCode::Applied));
244}
