39std::string jsonEscape(const std::string &s) {
40 std::string out;
41 out.reserve(s.size() + 8);
42 for (char c : s) {
43 switch (c) {
44 case '"': out += "\\\""; break;
45 case '\\': out += "\\\\"; break;
46 case '\n': out += "\\n"; break;
47 case '\r': out += "\\r"; break;
48 case '\t': out += "\\t"; break;
49 default: out += c;
50 }
51 }
52 return out;
53}

98 bool savePng(const std::string &path, int *outWidth, int *outHeight,
99 std::string *err) override {
100 auto *g = gfx();
101 if (!g) {
102 if (err) *err = "Graphics module not available";
103 return false;
104 }
105 g->setScreenReadbackEnabled(true);
106 image::ImageData *frame = nullptr;
107 try {
108 frame = g->newImageData();
109 } catch (...) {
110 if (err)
111 *err = "no presented frame yet: readback was just enabled and the "
112 "copy lands on the next present — call again after one frame "
113 "(screenshots need a running, presented game loop)";
114 return false;
115 }
116 if (!frame) {
117 if (err) *err = "no presented frame (enable readback and render a frame first)";
118 return false;
119 }
120 filesystem::FileData *png = nullptr;
121 try {
122 png = frame->encode(image::ImageData::FormatHandler::ENCODED_PNG, "frame.png", false);
123 } catch (...) {
124 delete frame;
125 if (err) *err = "png encode failed";
126 return false;
127 }
128 if (!png) {
129 delete frame;
130 if (err) *err = "png encode returned null";
131 return false;
132 }
133 std::error_code ec;
134 std::filesystem::create_directories(std::filesystem::path(path).parent_path(), ec);
135 std::ofstream out(path, std::ios::binary);
136 if (!out.good()) {
137 delete png;
138 delete frame;
139 if (err) *err = "cannot open output file: " + path;
140 return false;
141 }
142 out.write(static_cast<const char *>(png->getData()),
143 static_cast<std::streamsize>(png->getSize()));
144 const bool ok = out.good();
145 delete png;
146 delete frame;
147 if (outWidth) *outWidth = g->getPixelWidth();
148 if (outHeight) *outHeight = g->getPixelHeight();
149 if (!ok && err) *err = "failed writing: " + path;
150 return ok;
151 }

241 std::string entityIdMaskJson(int *outWidth, int *outHeight, bool *ok) override {
242 auto *g = gfx();
243 if (ok) *ok = false;
244 if (!g) return "{\"error\":\"Graphics module not available\"}";
245 auto *cam = camera();
246 if (!cam) return "{\"error\":\"no camera\"}";
247 auto d = cam->data();
248 const glm::vec3 eye(d->eyeX, d->eyeY, d->eyeZ);
249 const glm::vec3 target(d->targetX, d->targetY, d->targetZ);
250 const glm::vec3 up(d->upX, d->upY, d->upZ);
251 const int vw = g->getPixelWidth() > 0 ? g->getPixelWidth() : g->getWidth();
252 const int vh = g->getPixelHeight() > 0 ? g->getPixelHeight() : g->getHeight();
253 if (outWidth) *outWidth = vw;
254 if (outHeight) *outHeight = vh;
255 if (vw <= 0 || vh <= 0) return "{\"error\":\"empty viewport\"}";
256 const glm::mat4 viewM = glm::lookAtRH(eye, target, up);
257 const glm::mat4 projM = perspectiveVulkanRH_ZO(degToRad(d->fovYDeg), float(vw) / float(vh),
258 d->nearZ, d->farZ);
259 const glm::mat4 viewProj = projM * viewM;
260
261 std::vector<Graphics::EntityIdDraw> draws;
262 int id = 1;
263 std::string entities;
264 scene::TransformSystem::updateAll();
265 if (ecs::current()->getManager<scene::SceneHost>() != nullptr) {
266 auto hostView = ecs::View<scene::SceneHost, scene::SceneHost::Meta,
267 scene::SceneHost::Tree>();
268 for (auto it = hostView.begin(); it != hostView.end(); ++it) {
269 auto [meta, tree] = *it;
270 if (!meta || !meta->entity || !tree) continue;
271 auto *host = meta->entity;
272 host->walkDepthFirst([&](scene::SceneHost *, int, scene::SceneNode &node) {
273 if (!node.visible || node.space != "3d") return;
274 const auto *l = host->findLink(&node, scene::findLinkKind("renderable3d"));
275 if (!l || !l->target) return;
276 auto *r = static_cast<Renderable3D *>(l->target);
277 auto mr = r->meshRenderer();
278 Mesh *mesh = mr->mesh;
279 if (!mesh || !mesh->gpuHandle) return;
280 Graphics::EntityIdDraw draw;
281 draw.mesh = mesh;
282 draw.model = node.world;
283 const uint32_t packed = uint32_t(id);
284 draw.idColor = glm::vec4((packed & 255u) / 255.f,
285 ((packed >> 8) & 255u) / 255.f,
286 ((packed >> 16) & 255u) / 255.f, 1.f);
287 draws.push_back(draw);
288 if (!entities.empty()) entities += ",";
289 entities += "{\"id\":" + std::to_string(id) +
290 ",\"asset\":\"" + jsonEscape(node.name.empty() ? node.id : node.name) +
291 "\",\"node\":\"" + jsonEscape(node.id.empty() ? node.name : node.id) + "\"}";
292 ++id;
293 });
294 }
295 }
296 image::ImageData *img = g->renderEntityIdMask(draws, viewProj, vw, vh);
297 if (!img) return "{\"error\":\"id mask unsupported\"}";
298 delete img;
299 if (ok) *ok = true;
300 return "{\"camera\":" + vec3Json(eye) + ",\"viewportWidth\":" + std::to_string(vw) +
301 ",\"viewportHeight\":" + std::to_string(vh) + ",\"entities\":[" + entities +
302 "],\"count\":" + std::to_string(id - 1) + "}";
303 }

426 std::string visibleAt(const glm::vec3 *eye, const glm::vec3 *target, float fovYDeg,
427 bool *ok) {
428 auto *g = gfx();
429 if (ok) *ok = false;
430 auto *cam = camera();
431
432 glm::vec3 camEye(0.f, 1.8f, 3.f);
433 glm::vec3 camTarget(0.f, 1.2f, 0.f);
434 float camFov = 60.f;
435 float camNear = 0.1f, camFar = 100.f;
436 if (cam) {
437 auto d = cam->data();
438 camEye = glm::vec3(d->eyeX, d->eyeY, d->eyeZ);
439 camTarget = glm::vec3(d->targetX, d->targetY, d->targetZ);
440 camFov = d->fovYDeg;
441 camNear = d->nearZ;
442 camFar = d->farZ;
443 }
444 if (eye && target) {
445 camEye = *eye;
446 camTarget = *target;
447 }
448 if (fovYDeg > 0.f) camFov = fovYDeg;
449
450 const int vw = g ? g->getPixelWidth() : 1280;
451 const int vh = g ? g->getPixelHeight() : 720;
452 const glm::vec3 up(0.f, 1.f, 0.f);
453 const glm::mat4 view = glm::lookAtRH(camEye, camTarget, up);
454 const glm::mat4 proj = perspectiveVulkanRH_ZO(degToRad(camFov), float(vw) / float(vh),
455 camNear, camFar);
456 const glm::mat4 viewProj = proj * view;
457
458 static const glm::vec3 kLocalCorners[8] = {
459 {-0.5f, -0.5f, -0.5f}, {0.5f, -0.5f, -0.5f}, {-0.5f, 0.5f, -0.5f},
460 {-0.5f, -0.5f, 0.5f}, {0.5f, 0.5f, -0.5f}, {0.5f, -0.5f, 0.5f},
461 {-0.5f, 0.5f, 0.5f}, {0.5f, 0.5f, 0.5f}};
462
463 std::string entities;
464 int count = 0;
465 scene::TransformSystem::updateAll();
466 if (ecs::current()->getManager<scene::SceneHost>() != nullptr) {
467 auto hostView = ecs::View<scene::SceneHost, scene::SceneHost::Meta,
468 scene::SceneHost::Tree>();
469 for (auto it = hostView.begin(); it != hostView.end(); ++it) {
470 auto [meta, tree] = *it;
471 if (!meta || !meta->entity || !tree) continue;
472 auto *host = meta->entity;
473 host->walkDepthFirst([&](scene::SceneHost *, int, scene::SceneNode &node) {
474 if (!node.visible || node.space != "3d") return;
475 float minX = 1e30f, minY = 1e30f, minZ = 1e30f;
476 float maxX = -1e30f, maxY = -1e30f, maxZ = -1e30f;
477 float scMinX = 1e30f, scMinY = 1e30f;
478 float scMaxX = -1e30f, scMaxY = -1e30f;
479 bool anyInFront = false;
480 for (const auto &c : kLocalCorners) {
481 const glm::vec4 w = node.world * glm::vec4(c, 1.f);
482 minX = std::min(minX, w.x);
483 minY = std::min(minY, w.y);
484 minZ = std::min(minZ, w.z);
485 maxX = std::max(maxX, w.x);
486 maxY = std::max(maxY, w.y);
487 maxZ = std::max(maxZ, w.z);
488 const glm::vec4 clip = viewProj * w;
489 if (clip.w <= 1e-6f) continue;
490 anyInFront = true;
491 const float ndcX = clip.x / clip.w;
492 const float ndcY = clip.y / clip.w;
493 const float sx = (ndcX * 0.5f + 0.5f) * float(vw);
494 const float sy = (ndcY * 0.5f + 0.5f) * float(vh);
495 scMinX = std::min(scMinX, sx);
496 scMinY = std::min(scMinY, sy);
497 scMaxX = std::max(scMaxX, sx);
498 scMaxY = std::max(scMaxY, sy);
499 }
500 if (!anyInFront) return;
501 if (scMaxX < 0.f || scMinX > float(vw) || scMaxY < 0.f ||
502 scMinY > float(vh))
503 return;
504 if (!entities.empty()) entities += ",";
505 entities += "{\"id\":\"" + jsonEscape(node.id.empty() ? node.name : node.id) +
506 "\",\"asset\":\"" +
507 jsonEscape(node.name.empty() ? node.id : node.name) +
508 "\",\"world_aabb\":{\"min\":" + vec3Json(glm::vec3(minX, minY, minZ)) +
509 ",\"max\":" + vec3Json(glm::vec3(maxX, maxY, maxZ)) +
510 "},\"screen_bbox\":{\"x\":" + num(scMinX) + ",\"y\":" + num(scMinY) +
511 ",\"w\":" + num(scMaxX - scMinX) + ",\"h\":" + num(scMaxY - scMinY) +
512 "}}";
513 ++count;
514 });
515 }
516 }
517 if (ok) *ok = true;
518 return "{\"camera\":" + vec3Json(camEye) + ",\"cameraTarget\":" + vec3Json(camTarget) +
519 ",\"fov\":" + num(camFov) + ",\"viewportWidth\":" + std::to_string(vw) +
520 ",\"viewportHeight\":" + std::to_string(vh) + ",\"entities\":[" + entities +
521 "],\"count\":" + std::to_string(count) + "}";
522 }
