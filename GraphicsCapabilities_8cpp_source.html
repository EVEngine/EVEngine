39std::string jsonEscape(const std::string &s) {
40 std::string out;
41 out.reserve(s.size() + 8);
42 for (char c : s) {
43 switch (c) {
44 case '"': out += "\\\""; break;
45 case '\\': out += "\\\\"; break;
46 case '\n': out += "\\n"; break;
47 case '\r': out += "\\r"; break;
48 case '\t': out += "\\t"; break;
49 default: out += c;
50 }
51 }
52 return out;
53}

98 bool savePng(const std::string &path, int *outWidth, int *outHeight,
99 std::string *err) override {
100 auto *g = gfx();
101 if (!g) {
102 if (err) *err = "Graphics module not available";
103 return false;
104 }
105 g->setScreenReadbackEnabled(true);
106 image::ImageData *frame = nullptr;
107 try {
108 frame = g->newImageData();
109 } catch (...) {
110 if (err)
111 *err = "no presented frame yet: readback was just enabled and the "
112 "copy lands on the next present — call again after one frame "
113 "(screenshots need a running, presented game loop)";
114 return false;
115 }
116 if (!frame) {
117 if (err) *err = "no presented frame (enable readback and render a frame first)";
118 return false;
119 }
120 filesystem::FileData *png = nullptr;
121 try {
122 png = frame->encode(image::ImageData::FormatHandler::ENCODED_PNG, "frame.png", false);
123 } catch (...) {
124 delete frame;
125 if (err) *err = "png encode failed";
126 return false;
127 }
128 if (!png) {
129 delete frame;
130 if (err) *err = "png encode returned null";
131 return false;
132 }
133 std::error_code ec;
134 std::filesystem::create_directories(std::filesystem::path(path).parent_path(), ec);
135 std::ofstream out(path, std::ios::binary);
136 if (!out.good()) {
137 delete png;
138 delete frame;
139 if (err) *err = "cannot open output file: " + path;
140 return false;
141 }
142 out.write(static_cast<const char *>(png->getData()),
143 static_cast<std::streamsize>(png->getSize()));
144 const bool ok = out.good();
145 delete png;
146 delete frame;
147 if (outWidth) *outWidth = g->getPixelWidth();
148 if (outHeight) *outHeight = g->getPixelHeight();
149 if (!ok && err) *err = "failed writing: " + path;
150 return ok;

276 std::string entityIdMaskJson(int *outWidth, int *outHeight, bool *ok) override {
277 auto *g = gfx();
278 if (ok) *ok = false;
279 if (!g) return "{\"error\":\"Graphics module not available\"}";
280 auto *cam = camera();
281 if (!cam) return "{\"error\":\"no camera\"}";
282 auto d = cam->data();
283 const glm::vec3 eye(d->eyeX, d->eyeY, d->eyeZ);
284 const glm::vec3 target(d->targetX, d->targetY, d->targetZ);
285 const glm::vec3 up(d->upX, d->upY, d->upZ);
286 const int vw = g->getPixelWidth() > 0 ? g->getPixelWidth() : g->getWidth();
287 const int vh = g->getPixelHeight() > 0 ? g->getPixelHeight() : g->getHeight();
288 if (outWidth) *outWidth = vw;
289 if (outHeight) *outHeight = vh;
290 if (vw <= 0 || vh <= 0) return "{\"error\":\"empty viewport\"}";
291 const glm::mat4 viewM = glm::lookAtRH(eye, target, up);
292 const glm::mat4 projM = cameraProjectionVulkanRH_ZO(
293 d->orthographic, degToRad(d->fovYDeg), d->orthoHeight,
294 float(vw) / float(vh), d->nearZ, d->farZ);
295 const glm::mat4 viewProj = projM * viewM;
296
297 std::vector<Graphics::EntityIdDraw> draws;
298 int id = 1;
299 std::string entities;
300 scene::TransformSystem::updateAll();
301 if (ecs::current()->getManager<scene::SceneHost>() != nullptr) {
302 auto hostView = ecs::View<scene::SceneHost, scene::SceneHost::Meta,
303 scene::SceneHost::Tree>();
304 for (auto it = hostView.begin(); it != hostView.end(); ++it) {
305 auto [meta, tree] = *it;
306 if (!meta || !meta->entity || !tree) continue;
307 auto *host = meta->entity;
308 host->walkDepthFirst([&](scene::SceneHost *, int, scene::SceneNode &node) {
309 if (!node.visible || node.space != "3d") return;
310 const auto *l = host->findLink(&node, scene::findLinkKind("renderable3d"));
311 if (!l || !l->target) return;
312 auto *r = static_cast<Renderable3D *>(l->target);
313 auto mr = r->meshRenderer();
314 Mesh *mesh = mr->mesh;
315 if (!mesh || !mesh->gpuHandle) return;
316 Graphics::EntityIdDraw draw;
317 draw.mesh = mesh;
318 draw.model = node.world;
319 const uint32_t packed = uint32_t(id);
320 draw.idColor = glm::vec4((packed & 255u) / 255.f,
321 ((packed >> 8) & 255u) / 255.f,
322 ((packed >> 16) & 255u) / 255.f, 1.f);
323 draws.push_back(draw);
324 if (!entities.empty()) entities += ",";
325 entities += "{\"id\":" + std::to_string(id) +
326 ",\"asset\":\"" + jsonEscape(node.name.empty() ? node.id : node.name) +
327 "\",\"node\":\"" + jsonEscape(node.id.empty() ? node.name : node.id) + "\"}";
328 ++id;
329 });
330 }
331 }
332 image::ImageData *img = g->renderEntityIdMask(draws, viewProj, vw, vh);
333 if (!img) return "{\"error\":\"id mask unsupported\"}";
334 delete img;
335 if (ok) *ok = true;
336 return "{\"camera\":" + vec3Json(eye) + ",\"viewportWidth\":" + std::to_string(vw) +
337 ",\"viewportHeight\":" + std::to_string(vh) + ",\"entities\":[" + entities +
338 "],\"count\":" + std::to_string(id - 1) + "}";
339 }

463 std::string visibleAt(const glm::vec3 *eye, const glm::vec3 *target, float fovYDeg,
464 bool *ok) {
465 auto *g = gfx();
466 if (ok) *ok = false;
467 auto *cam = camera();
468
469 glm::vec3 camEye(0.f, 1.8f, 3.f);
470 glm::vec3 camTarget(0.f, 1.2f, 0.f);
471 float camFov = 60.f;
472 float camNear = 0.1f, camFar = 100.f;
473 if (cam) {
474 auto d = cam->data();
475 camEye = glm::vec3(d->eyeX, d->eyeY, d->eyeZ);
476 camTarget = glm::vec3(d->targetX, d->targetY, d->targetZ);
477 camFov = d->fovYDeg;
478 camNear = d->nearZ;
479 camFar = d->farZ;
480 }
481 if (eye && target) {
482 camEye = *eye;
483 camTarget = *target;
484 }
485 if (fovYDeg > 0.f) camFov = fovYDeg;
486
487 const int vw = g ? g->getPixelWidth() : 1280;
488 const int vh = g ? g->getPixelHeight() : 720;
489 const glm::vec3 up(0.f, 1.f, 0.f);
490 const glm::mat4 view = glm::lookAtRH(camEye, camTarget, up);
491 const glm::mat4 proj = perspectiveVulkanRH_ZO(degToRad(camFov), float(vw) / float(vh),
492 camNear, camFar);
493 const glm::mat4 viewProj = proj * view;
494
495 static const glm::vec3 kLocalCorners[8] = {
496 {-0.5f, -0.5f, -0.5f}, {0.5f, -0.5f, -0.5f}, {-0.5f, 0.5f, -0.5f},
497 {-0.5f, -0.5f, 0.5f}, {0.5f, 0.5f, -0.5f}, {0.5f, -0.5f, 0.5f},
498 {-0.5f, 0.5f, 0.5f}, {0.5f, 0.5f, 0.5f}};
499
500 std::string entities;
501 int count = 0;
502 scene::TransformSystem::updateAll();
503 if (ecs::current()->getManager<scene::SceneHost>() != nullptr) {
504 auto hostView = ecs::View<scene::SceneHost, scene::SceneHost::Meta,
505 scene::SceneHost::Tree>();
506 for (auto it = hostView.begin(); it != hostView.end(); ++it) {
507 auto [meta, tree] = *it;
508 if (!meta || !meta->entity || !tree) continue;
509 auto *host = meta->entity;
510 host->walkDepthFirst([&](scene::SceneHost *, int, scene::SceneNode &node) {
511 if (!node.visible || node.space != "3d") return;
512 float minX = 1e30f, minY = 1e30f, minZ = 1e30f;
513 float maxX = -1e30f, maxY = -1e30f, maxZ = -1e30f;
514 float scMinX = 1e30f, scMinY = 1e30f;
515 float scMaxX = -1e30f, scMaxY = -1e30f;
516 bool anyInFront = false;
517 for (const auto &c : kLocalCorners) {
518 const glm::vec4 w = node.world * glm::vec4(c, 1.f);
519 minX = std::min(minX, w.x);
520 minY = std::min(minY, w.y);
521 minZ = std::min(minZ, w.z);
522 maxX = std::max(maxX, w.x);
523 maxY = std::max(maxY, w.y);
524 maxZ = std::max(maxZ, w.z);
525 const glm::vec4 clip = viewProj * w;
526 if (clip.w <= 1e-6f) continue;
527 anyInFront = true;
528 const float ndcX = clip.x / clip.w;
529 const float ndcY = clip.y / clip.w;
530 const float sx = (ndcX * 0.5f + 0.5f) * float(vw);
531 const float sy = (ndcY * 0.5f + 0.5f) * float(vh);
532 scMinX = std::min(scMinX, sx);
533 scMinY = std::min(scMinY, sy);
534 scMaxX = std::max(scMaxX, sx);
535 scMaxY = std::max(scMaxY, sy);
536 }
537 if (!anyInFront) return;
538 if (scMaxX < 0.f || scMinX > float(vw) || scMaxY < 0.f ||
539 scMinY > float(vh))
540 return;
541 if (!entities.empty()) entities += ",";
542 entities += "{\"id\":\"" + jsonEscape(node.id.empty() ? node.name : node.id) +
