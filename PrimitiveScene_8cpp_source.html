25bool independent(glm::vec3 a, glm::vec3 b) {
26 constexpr float minimumRelativeAreaSquared = 1e-12f;
27 const float aLengthSquared = glm::dot(a, a);
28 const float bLengthSquared = glm::dot(b, b);
29 const glm::vec3 crossValue = glm::cross(a, b);
30 const float areaSquared = glm::dot(crossValue, crossValue);
31 return aLengthSquared > 0.f && bLengthSquared > 0.f &&
32 areaSquared > minimumRelativeAreaSquared * aLengthSquared * bLengthSquared;
33}

35bool formsBasis(glm::vec3 a, glm::vec3 b, glm::vec3 c) {
36 constexpr float minimumRelativeVolumeSquared = 1e-12f;
37 const float volume = glm::dot(a, glm::cross(b, c));
38 return volume * volume > minimumRelativeVolumeSquared * glm::dot(a, a) * glm::dot(b, b) * glm::dot(c, c);
39}

41eve::Result<void> validateDescriptor(const PrimitiveDescriptor3D& descriptor) {
42 const auto& paint = descriptor.paint;
43 const auto& stroke = paint.stroke;
44 if (!std::isfinite(stroke.width) || stroke.width < 0.f ||
45 !std::isfinite(stroke.miterLimit) || stroke.miterLimit <= 0.f ||
46 !std::isfinite(paint.color.r) || !std::isfinite(paint.color.g) ||
47 !std::isfinite(paint.color.b) || !std::isfinite(paint.color.a))
48 return primitiveFailure<void>(eve::DiagnosticCode::InvalidArgument, "primitive paint must be finite and valid");
49 if (stroke.dash) {
50 const auto& dash = *stroke.dash;
51 float period = 0.f;
52 if (dash.intervals.empty() || dash.intervals.size() % 2 != 0 || !std::isfinite(dash.phase))
53 return primitiveFailure<void>(eve::DiagnosticCode::InvalidArgument, "invalid primitive dash pattern");
54 for (float interval : dash.intervals) {
55 if (!std::isfinite(interval) || interval <= 0.f)
56 return primitiveFailure<void>(eve::DiagnosticCode::InvalidArgument, "invalid primitive dash interval");
57 period += interval;
58 }
59 if (!std::isfinite(period))
60 return primitiveFailure<void>(eve::DiagnosticCode::InvalidArgument, "primitive dash period overflow");
61 }
62 try {
63 descriptor.paint.validate();
64 } catch (const std::exception& error) {
65 return primitiveFailure<void>(eve::DiagnosticCode::InvalidArgument, error.what());
66 }
67 for (glm::length_t column = 0; column < 4; ++column)
68 for (glm::length_t row = 0; row < 4; ++row)
69 if (!std::isfinite(descriptor.transform[column][row]))
70 return primitiveFailure<void>(eve::DiagnosticCode::InvalidArgument,
71 "primitive transform must be finite");
72 const bool valid = std::visit(
73 [](const auto& geometry) {
74 using Geometry = std::decay_t<decltype(geometry)>;
75 if constexpr (std::is_same_v<Geometry, PrimitivePolyline3D>) {
76 return geometry.points.size() >= (geometry.closed ? 3u : 2u) &&
77 std::all_of(geometry.points.begin(), geometry.points.end(), finite);
78 } else if constexpr (std::is_same_v<Geometry, PrimitiveAabb3D>) {
79 return finite(geometry.minimum) && finite(geometry.maximum) &&
80 geometry.minimum.x <= geometry.maximum.x && geometry.minimum.y <= geometry.maximum.y &&
81 geometry.minimum.z <= geometry.maximum.z;
82 } else if constexpr (std::is_same_v<Geometry, PrimitiveObb3D>) {
83 return finite(geometry.center) && nonZero(geometry.halfAxes[0]) && nonZero(geometry.halfAxes[1]) &&
84 nonZero(geometry.halfAxes[2]) && independent(geometry.halfAxes[0], geometry.halfAxes[1]) &&
85 independent(geometry.halfAxes[1], geometry.halfAxes[2]) &&
86 independent(geometry.halfAxes[2], geometry.halfAxes[0]) &&
87 formsBasis(geometry.halfAxes[0], geometry.halfAxes[1], geometry.halfAxes[2]);
88 } else if constexpr (std::is_same_v<Geometry, PrimitiveDisk3D>) {
89 return finite(geometry.center) && nonZero(geometry.normal) && std::isfinite(geometry.radius) &&
90 geometry.radius > 0.f && geometry.segments >= 3;
91 } else if constexpr (std::is_same_v<Geometry, PrimitiveArc3D>) {
92 return finite(geometry.center) && nonZero(geometry.normal) && nonZero(geometry.zeroDirection) &&
93 independent(geometry.normal, geometry.zeroDirection) &&
94 std::isfinite(geometry.radius) && geometry.radius > 0.f &&
95 std::isfinite(geometry.startRadians) && std::isfinite(geometry.sweepRadians) &&
96 geometry.segments >= 1;
97 } else if constexpr (std::is_same_v<Geometry, PrimitiveSphere3D>) {
98 return finite(geometry.center) && std::isfinite(geometry.radius) && geometry.radius > 0.f &&
99 geometry.segments >= 3;
100 } else if constexpr (std::is_same_v<Geometry, PrimitiveCapsule3D> ||
101 std::is_same_v<Geometry, PrimitiveCylinder3D>) {
102 return finite(geometry.a) && finite(geometry.b) && nonZero(geometry.b - geometry.a) &&
103 std::isfinite(geometry.radius) &&
104 geometry.radius > 0.f && geometry.segments >= 3;
105 } else if constexpr (std::is_same_v<Geometry, PrimitiveCone3D>) {
106 return finite(geometry.apex) && nonZero(geometry.axis) && std::isfinite(geometry.height) &&
107 geometry.height > 0.f && std::isfinite(geometry.radius) && geometry.radius > 0.f &&
108 geometry.segments >= 3;
109 } else if constexpr (std::is_same_v<Geometry, PrimitiveGrid3D>) {
110 return finite(geometry.origin) && nonZero(geometry.axisU) && nonZero(geometry.axisV) &&
111 independent(geometry.axisU, geometry.axisV) &&
112 geometry.cellsU > 0 && geometry.cellsV > 0;
113 } else if constexpr (std::is_same_v<Geometry, PrimitiveArrow3D>) {
114 return finite(geometry.from) && finite(geometry.to) && nonZero(geometry.to - geometry.from) &&
115 std::isfinite(geometry.headLength) &&
116 geometry.headLength > 0.f && std::isfinite(geometry.headRadius) && geometry.headRadius > 0.f;
117 } else {
118 return std::all_of(geometry.corners.begin(), geometry.corners.end(), finite);
119 }
120 },
121 descriptor.geometry);
122 if (!valid)
123 return primitiveFailure<void>(eve::DiagnosticCode::InvalidArgument,
124 "primitive geometry is invalid or non-finite");
125 return eve::Result<void>::success();
126}

136eve::Result<PrimitiveHandle> PrimitiveScene::add(PrimitiveDescriptor3D descriptor) {
137 auto validation = validateDescriptor(descriptor);
138 if (!validation) return eve::Result<PrimitiveHandle>::failure(validation.status());
139 PrimitiveHandle::index_type index;
140 if (!freeSlots_.empty()) {
141 index = freeSlots_.back();
142 freeSlots_.pop_back();
143 slots_[index].descriptor = std::move(descriptor);
144 } else {
145 if (slots_.size() >= PrimitiveHandle::invalidIndex) {
146 return primitiveFailure<PrimitiveHandle>(eve::DiagnosticCode::Failed,
147 "primitive scene slot capacity exceeded");
148 }
149 index = static_cast<PrimitiveHandle::index_type>(slots_.size());
150 slots_.push_back(Slot{std::move(descriptor), 1, false, {}});
151 }
152 ++liveCount_;
153 return eve::Result<PrimitiveHandle>::success(PrimitiveHandle(owner_, index, slots_[index].generation));
154}

185eve::Result<std::size_t> PrimitiveScene::updateMany(std::span<const PrimitiveBatchUpdate> updates) {
186 for (const PrimitiveBatchUpdate& update : updates) {
187 if (!matches(update.handle))
188 return primitiveFailure<std::size_t>(eve::DiagnosticCode::StaleHandle,
189 "primitive batch contains a stale handle");
190 auto validation = validateDescriptor(update.descriptor);
191 if (!validation) return eve::Result<std::size_t>::failure(validation.status());
192 }
193 // Copy every owning payload before publishing any mutation. A later allocation
194 // failure must not leave earlier descriptors in the batch already replaced.
195 std::vector<PrimitiveBatchUpdate> prepared(updates.begin(), updates.end());
196 static_assert(std::is_nothrow_move_assignable_v<PrimitiveDescriptor3D>);
197 for (PrimitiveBatchUpdate& update : prepared) {
198 *slots_[update.handle.index()].descriptor = std::move(update.descriptor);
199 slots_[update.handle.index()].cache.reset();
200 }
201 return eve::Result<std::size_t>::success(updates.size());
202}

234eve::Result<void> PrimitiveScene::tryRender(PrimitiveSceneCanvas3D& destination) const {
235 for (const Slot& slot : slots_) {
236 if (!slot.descriptor || !slot.descriptor->visible) continue;
237 const PrimitiveDescriptor3D& descriptor = *slot.descriptor;
238 const bool hit = slot.cache.has_value();
239 if (!hit) {
240 PrimitiveSceneCanvas3D canvas(destination.context_, std::numeric_limits<std::size_t>::max());
241 canvas.concat(descriptor.transform);
242 std::visit(
243 [&](const auto& geometry) {
244 using Geometry = std::decay_t<decltype(geometry)>;
245 if constexpr (std::is_same_v<Geometry, PrimitivePolyline3D>) {
246 canvas.drawPolyline(geometry.points, geometry.closed, descriptor.paint);
247 } else if constexpr (std::is_same_v<Geometry, PrimitiveAabb3D>) {
248 canvas.drawAabb(geometry.minimum, geometry.maximum, descriptor.paint);
249 } else if constexpr (std::is_same_v<Geometry, PrimitiveObb3D>) {
250 canvas.drawObb(geometry.center, geometry.halfAxes, descriptor.paint);
251 } else if constexpr (std::is_same_v<Geometry, PrimitiveDisk3D>) {
252 canvas.drawDisk(geometry.center, geometry.normal, geometry.radius, descriptor.paint,
253 geometry.segments);
254 } else if constexpr (std::is_same_v<Geometry, PrimitiveArc3D>) {
255 canvas.drawArc(geometry.center, geometry.normal, geometry.zeroDirection, geometry.radius,
256 geometry.startRadians, geometry.sweepRadians, descriptor.paint,
257 geometry.segments);
258 } else if constexpr (std::is_same_v<Geometry, PrimitiveSphere3D>) {
259 canvas.drawSphere(geometry.center, geometry.radius, descriptor.paint, geometry.segments);
260 } else if constexpr (std::is_same_v<Geometry, PrimitiveCapsule3D>) {
261 canvas.drawCapsule(geometry.a, geometry.b, geometry.radius, descriptor.paint,
262 geometry.segments);
263 } else if constexpr (std::is_same_v<Geometry, PrimitiveCylinder3D>) {
264 canvas.drawCylinder(geometry.a, geometry.b, geometry.radius, descriptor.paint,
265 geometry.segments);
266 } else if constexpr (std::is_same_v<Geometry, PrimitiveCone3D>) {
267 canvas.drawCone(geometry.apex, geometry.axis, geometry.height, geometry.radius,
268 descriptor.paint, geometry.segments);
269 } else if constexpr (std::is_same_v<Geometry, PrimitiveGrid3D>) {
270 canvas.drawGrid(geometry.origin, geometry.axisU, geometry.axisV, geometry.cellsU,
271 geometry.cellsV, descriptor.paint);
272 } else if constexpr (std::is_same_v<Geometry, PrimitiveArrow3D>) {
273 canvas.drawArrow(geometry.from, geometry.to, geometry.headLength, geometry.headRadius,
274 descriptor.paint);
275 } else if constexpr (std::is_same_v<Geometry, PrimitiveFrustum3D>) {
276 canvas.drawFrustum(geometry.corners, descriptor.paint);
277 }
278 },
279 descriptor.geometry);
280 slot.cache = std::move(canvas);
281 }
282 const auto& cached = *slot.cache;
283 const auto count = cached.commands_.size() + cached.triangles_.size();
284 const auto used = destination.commands_.size() + destination.triangles_.size();
285 if (used > destination.hardCommandLimit_ || count > destination.hardCommandLimit_ - used) {
286 destination.statistics_.droppedCommands += count + cached.statistics_.droppedCommands;
287 return primitiveFailure<void>(eve::DiagnosticCode::Failed, "primitive scene command budget exceeded");
288 }
289 // Prepare all owning payloads and capacity before publishing this slot.
290 auto lines = cached.commands_;
291 auto triangles = cached.triangles_;
292 for (auto& command : lines) command.transform = destination.transform_ * command.transform;
293 for (auto& command : triangles) command.transform = destination.transform_ * command.transform;
294 destination.commands_.reserve(destination.commands_.size() + lines.size());
295 destination.triangles_.reserve(destination.triangles_.size() + triangles.size());
296 static_assert(std::is_nothrow_move_constructible_v<PolylineCommand3D>);
297 static_assert(std::is_nothrow_move_constructible_v<TriangleCommand3D>);
298 for (auto& command : lines) destination.commands_.push_back(std::move(command));
299 for (auto& command : triangles) destination.triangles_.push_back(std::move(command));
300 destination.statistics_.commandCount += cached.statistics_.commandCount;
301 destination.statistics_.segmentCount += cached.statistics_.segmentCount;
302 destination.statistics_.triangleCount += cached.statistics_.triangleCount;
303 destination.statistics_.cacheHits += hit ? 1u : 0u;
304 }
305 return eve::Result<void>::success();
306}
