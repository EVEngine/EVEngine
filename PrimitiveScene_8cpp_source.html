19eve::Result<void> validateDescriptor(const PrimitiveDescriptor3D& descriptor) {
20 try {
21 descriptor.paint.validate();
22 } catch (const std::exception& error) {
23 return primitiveFailure<void>(eve::DiagnosticCode::InvalidArgument, error.what());
24 }
25 for (glm::length_t column = 0; column < 4; ++column)
26 for (glm::length_t row = 0; row < 4; ++row)
27 if (!std::isfinite(descriptor.transform[column][row]))
28 return primitiveFailure<void>(eve::DiagnosticCode::InvalidArgument,
29 "primitive transform must be finite");
30 const bool valid = std::visit(
31 [](const auto& geometry) {
32 using Geometry = std::decay_t<decltype(geometry)>;
33 if constexpr (std::is_same_v<Geometry, PrimitivePolyline3D>) {
34 return geometry.points.size() >= 2 &&
35 std::all_of(geometry.points.begin(), geometry.points.end(), finite);
36 } else if constexpr (std::is_same_v<Geometry, PrimitiveAabb3D>) {
37 return finite(geometry.minimum) && finite(geometry.maximum);
38 } else if constexpr (std::is_same_v<Geometry, PrimitiveObb3D>) {
39 return finite(geometry.center) && finite(geometry.halfAxes[0]) && finite(geometry.halfAxes[1]) &&
40 finite(geometry.halfAxes[2]);
41 } else if constexpr (std::is_same_v<Geometry, PrimitiveDisk3D>) {
42 return finite(geometry.center) && finite(geometry.normal) && std::isfinite(geometry.radius) &&
43 geometry.radius > 0.f && geometry.segments >= 3;
44 } else if constexpr (std::is_same_v<Geometry, PrimitiveArc3D>) {
45 return finite(geometry.center) && finite(geometry.normal) && finite(geometry.zeroDirection) &&
46 std::isfinite(geometry.radius) && geometry.radius > 0.f &&
47 std::isfinite(geometry.startRadians) && std::isfinite(geometry.sweepRadians) &&
48 geometry.segments >= 1;
49 } else if constexpr (std::is_same_v<Geometry, PrimitiveSphere3D>) {
50 return finite(geometry.center) && std::isfinite(geometry.radius) && geometry.radius > 0.f &&
51 geometry.segments >= 3;
52 } else if constexpr (std::is_same_v<Geometry, PrimitiveCapsule3D> ||
53 std::is_same_v<Geometry, PrimitiveCylinder3D>) {
54 return finite(geometry.a) && finite(geometry.b) && std::isfinite(geometry.radius) &&
55 geometry.radius > 0.f && geometry.segments >= 3;
56 } else if constexpr (std::is_same_v<Geometry, PrimitiveCone3D>) {
57 return finite(geometry.apex) && finite(geometry.axis) && std::isfinite(geometry.height) &&
58 geometry.height > 0.f && std::isfinite(geometry.radius) && geometry.radius > 0.f &&
59 geometry.segments >= 3;
60 } else if constexpr (std::is_same_v<Geometry, PrimitiveGrid3D>) {
61 return finite(geometry.origin) && finite(geometry.axisU) && finite(geometry.axisV) &&
62 geometry.cellsU > 0 && geometry.cellsV > 0;
63 } else if constexpr (std::is_same_v<Geometry, PrimitiveArrow3D>) {
64 return finite(geometry.from) && finite(geometry.to) && std::isfinite(geometry.headLength) &&
65 geometry.headLength > 0.f && std::isfinite(geometry.headRadius) && geometry.headRadius > 0.f;
66 } else {
67 return std::all_of(geometry.corners.begin(), geometry.corners.end(), finite);
68 }
69 },
70 descriptor.geometry);
71 if (!valid)
72 return primitiveFailure<void>(eve::DiagnosticCode::InvalidArgument,
73 "primitive geometry is invalid or non-finite");
74 return eve::Result<void>::success();
75}

85eve::Result<PrimitiveHandle> PrimitiveScene::add(PrimitiveDescriptor3D descriptor) {
86 auto validation = validateDescriptor(descriptor);
87 if (!validation) return eve::Result<PrimitiveHandle>::failure(validation.status());
88 PrimitiveHandle::index_type index;
89 if (!freeSlots_.empty()) {
90 index = freeSlots_.back();
91 freeSlots_.pop_back();
92 slots_[index].descriptor = std::move(descriptor);
93 } else {
94 if (slots_.size() >= PrimitiveHandle::invalidIndex) {
95 return primitiveFailure<PrimitiveHandle>(eve::DiagnosticCode::Failed,
96 "primitive scene slot capacity exceeded");
97 }
98 index = static_cast<PrimitiveHandle::index_type>(slots_.size());
99 slots_.push_back(Slot{std::move(descriptor), 1, false});
100 }
101 ++liveCount_;
102 return eve::Result<PrimitiveHandle>::success(PrimitiveHandle(owner_, index, slots_[index].generation));
103}

115eve::Result<PrimitiveRemoveStatus> PrimitiveScene::remove(PrimitiveHandle handle) {
116 if (!matches(handle)) {
117 return primitiveFailure<PrimitiveRemoveStatus>(eve::DiagnosticCode::StaleHandle, "primitive handle is stale");
118 }
119 Slot& slot = slots_[handle.index()];
120 slot.descriptor.reset();
121 --liveCount_;
122 const auto next = PrimitiveHandle::nextGeneration(slot.generation);
123 if (next) {
124 slot.generation = *next;
125 freeSlots_.push_back(handle.index());
126 } else {
127 slot.retired = true;
128 }
129 return eve::Result<PrimitiveRemoveStatus>::success(PrimitiveRemoveStatus::Removed);

132eve::Result<std::size_t> PrimitiveScene::updateMany(std::span<const PrimitiveBatchUpdate> updates) {
133 for (const PrimitiveBatchUpdate& update : updates) {
134 if (!matches(update.handle))
135 return primitiveFailure<std::size_t>(eve::DiagnosticCode::StaleHandle,
136 "primitive batch contains a stale handle");
137 auto validation = validateDescriptor(update.descriptor);
138 if (!validation) return eve::Result<std::size_t>::failure(validation.status());
139 }
140 for (const PrimitiveBatchUpdate& update : updates) slots_[update.handle.index()].descriptor = update.descriptor;
141 return eve::Result<std::size_t>::success(updates.size());
142}

168void PrimitiveScene::render(PrimitiveSceneCanvas3D& canvas) const {
169 for (const Slot& slot : slots_) {
170 if (!slot.descriptor || !slot.descriptor->visible) continue;
171 const PrimitiveDescriptor3D& descriptor = *slot.descriptor;
172 canvas.save();
173 canvas.concat(descriptor.transform);
174 std::visit(
175 [&](const auto& geometry) {
176 using Geometry = std::decay_t<decltype(geometry)>;
177 if constexpr (std::is_same_v<Geometry, PrimitivePolyline3D>) {
178 canvas.drawPolyline(geometry.points, geometry.closed, descriptor.paint);
179 } else if constexpr (std::is_same_v<Geometry, PrimitiveAabb3D>) {
180 canvas.drawAabb(geometry.minimum, geometry.maximum, descriptor.paint);
181 } else if constexpr (std::is_same_v<Geometry, PrimitiveObb3D>) {
182 canvas.drawObb(geometry.center, geometry.halfAxes, descriptor.paint);
183 } else if constexpr (std::is_same_v<Geometry, PrimitiveDisk3D>) {
184 canvas.drawDisk(geometry.center, geometry.normal, geometry.radius, descriptor.paint,
185 geometry.segments);
186 } else if constexpr (std::is_same_v<Geometry, PrimitiveArc3D>) {
187 canvas.drawArc(geometry.center, geometry.normal, geometry.zeroDirection, geometry.radius,
188 geometry.startRadians, geometry.sweepRadians, descriptor.paint, geometry.segments);
189 } else if constexpr (std::is_same_v<Geometry, PrimitiveSphere3D>) {
190 canvas.drawSphere(geometry.center, geometry.radius, descriptor.paint, geometry.segments);
191 } else if constexpr (std::is_same_v<Geometry, PrimitiveCapsule3D>) {
192 canvas.drawCapsule(geometry.a, geometry.b, geometry.radius, descriptor.paint, geometry.segments);
193 } else if constexpr (std::is_same_v<Geometry, PrimitiveCylinder3D>) {
194 canvas.drawCylinder(geometry.a, geometry.b, geometry.radius, descriptor.paint, geometry.segments);
195 } else if constexpr (std::is_same_v<Geometry, PrimitiveCone3D>) {
196 canvas.drawCone(geometry.apex, geometry.axis, geometry.height, geometry.radius, descriptor.paint,
197 geometry.segments);
198 } else if constexpr (std::is_same_v<Geometry, PrimitiveGrid3D>) {
199 canvas.drawGrid(geometry.origin, geometry.axisU, geometry.axisV, geometry.cellsU, geometry.cellsV,
200 descriptor.paint);
201 } else if constexpr (std::is_same_v<Geometry, PrimitiveArrow3D>) {
202 canvas.drawArrow(geometry.from, geometry.to, geometry.headLength, geometry.headRadius,
203 descriptor.paint);
204 } else if constexpr (std::is_same_v<Geometry, PrimitiveFrustum3D>) {
205 canvas.drawFrustum(geometry.corners, descriptor.paint);
206 }
207 },

213bool PrimitiveScene::matches(PrimitiveHandle handle) const noexcept {
214 if (!handle.isValid() || handle.owner() != owner_ || handle.index() >= slots_.size()) return false;
215 const Slot& slot = slots_[handle.index()];
216 return !slot.retired && slot.descriptor.has_value() && slot.generation == handle.generation();
