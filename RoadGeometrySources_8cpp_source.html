17bool geometryValid(const MeshBuild& m) {
18 const auto vertices = m.positions().size() / 3;
19 if (vertices == 0 || m.positions().size() % 3 || m.indices().empty() || m.indices().size() % 3 ||
20 m.positions().size() > static_cast<std::size_t>(std::numeric_limits<int>::max()) ||
21 m.indices().size() > static_cast<std::size_t>(std::numeric_limits<int>::max()) ||
22 m.normals().size() != vertices * 3 || m.uvs().size() != vertices * 2 ||
23 (!m.colors().empty() && m.colors().size() != vertices * 4) ||
24 m.triangleGroups().size() != m.indices().size() / 3)
25 return false;
26 for (const auto* values : {&m.positions(), &m.normals(), &m.uvs(), &m.colors()})
27 for (float v : *values)
28 if (!std::isfinite(v)) return false;
29 for (auto i : m.indices())
30 if (i >= vertices) return false;
31 for (auto g : m.triangleGroups())
32 if (g < 0 || static_cast<std::size_t>(g) >= m.groupNames().size()) return false;
33 return true;
34}
35std::string digest(const MeshBuild& m) {
36 std::uint64_t hash = 14695981039346656037ull;
37 auto byte = [&](std::uint8_t b) {
38 hash ^= b;
39 hash *= 1099511628211ull;
40 };
41 auto integer = [&](std::uint64_t v) {
42 for (int n = 0; n < 8; ++n) byte(static_cast<std::uint8_t>(v >> (8 * n)));
43 };
44 for (const auto* values : {&m.positions(), &m.normals(), &m.uvs(), &m.colors()}) {
45 integer(values->size());
46 for (float v : *values) integer(std::bit_cast<std::uint32_t>(v));
47 }
48 integer(m.indices().size());
49 for (auto i : m.indices()) integer(i);
50 integer(m.triangleGroups().size());
51 for (auto g : m.triangleGroups()) integer(static_cast<std::uint32_t>(g));
52 integer(m.groupNames().size());
53 for (const auto& name : m.groupNames()) {
54 integer(name.size());
55 for (unsigned char b : name) byte(b);
56 }
57 std::array<char, 16> out{};
58 constexpr char digits[] = "0123456789abcdef";
59 for (std::size_t i = 0; i < out.size(); ++i) out[15 - i] = digits[(hash >> (4 * i)) & 15u];
60 return std::string(out.data(), out.size());
61}
62bool partitionValid(std::span<const RoadGeometrySourceSpan> spans, std::size_t triangles) {
63 std::size_t next = 0;
