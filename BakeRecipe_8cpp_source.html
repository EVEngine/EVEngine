26void append32(std::string& out, std::uint32_t value) {
27 for (unsigned shift = 0; shift < 32; shift += 8) out.push_back(static_cast<char>((value >> shift) & 255));

33struct Reader {
34 std::string_view bytes;
35 std::size_t cursor = 0;
36 bool valid = true;
37 std::string_view take(std::size_t size) {
38 if (size > bytes.size() - cursor) {
39 valid = false;
40 return {};
41 }
42 auto result = bytes.substr(cursor, size);
43 cursor += size;
44 return result;
45 }
46 std::uint32_t word() {
47 auto data = take(4);
48 if (!valid) return 0;
49 std::uint32_t result = 0;
50 for (unsigned i = 0; i < 4; ++i) result |= std::uint32_t(static_cast<unsigned char>(data[i])) << (8 * i);
51 return result;
52 }
53 std::string text() {
54 const auto size = word();
55 if (size > MaxToken) {
56 valid = false;
57 return {};
58 }
59 return std::string(take(size));
60 }
61};

105 if (!token(producer_) || !token(backend_) || !digest(parameters_) || !std::isfinite(stepSeconds_) ||
106 stepSeconds_ <= 0 || sampleStride_ == 0 || sources_.empty())

108 const bool planar = coordinates_ == BakeCoordinates::PlanarPixels;

129Result<BakeRecipe> BakeRecipe::decode(std::string_view bytes) {
130 auto failure = [] { return Result<BakeRecipe>::failure(invalidRecipe("invalid bake provenance schema or data")); };
131 if (bytes.size() > 65536) return failure();
132 Reader reader{bytes};
133 const auto schema = reader.take(8);
134 const auto coordinates = reader.text();
135 BakeRecipe recipe;
136 if (schema == "EVECREC1" && coordinates == "metres-rh-y-up")
137 recipe.coordinates_ = BakeCoordinates::MetresRightHandedYUp;
138 else if (schema == "EVEBREC1" && coordinates == "planar-pixels-xy")
139 recipe.coordinates_ = BakeCoordinates::PlanarPixels;
140 else
141 return failure();
142 recipe.producer_ = reader.text();
143 recipe.backend_ = reader.text();
144 recipe.parameters_ = reader.take(32);
145 recipe.stepSeconds_ = std::bit_cast<float>(reader.word());
146 recipe.sampleStride_ = reader.word();
147 const auto sources = reader.word();
148 if (sources == 0 || sources > MaxEntries) return failure();
149 for (std::uint32_t i = 0; i < sources; ++i) {
150 auto name = reader.text();
151 auto key = reader.take(32);
152 if (!reader.valid || !token(name) || !digest(key) || !recipe.sources_.emplace(name, key).second)
153 return failure();
154 }
155 const auto streams = reader.word();
156 if (streams > MaxEntries) return failure();
157 for (std::uint32_t i = 0; i < streams; ++i) {
158 auto name = reader.text();
159 const auto low = reader.word();
160 const auto high = reader.word();
161 if (!reader.valid || !token(name) ||
162 !recipe.randomStreams_.emplace(name, std::uint64_t(low) | (std::uint64_t(high) << 32)).second)
163 return failure();
164 }
165 if (!reader.valid || reader.cursor != bytes.size()) return failure();
166 auto canonical = recipe.encode();
167 if (!canonical || canonical.value() != bytes) return failure();
168 return Result<BakeRecipe>::success(std::move(recipe));
169}
170Result<void> BakeRecipe::validateSampleTime(std::uint32_t frame, float seconds) const {
171 const double expected = double(frame) * stepSeconds_ * sampleStride_;
172 const double tolerance =
173 std::min(4 * std::numeric_limits<float>::epsilon() * std::max(expected, double(stepSeconds_)),
174 0.25 * double(stepSeconds_) * sampleStride_);
175 if (!std::isfinite(seconds) || stepSeconds_ <= 0 || sampleStride_ == 0 ||
176 expected > std::numeric_limits<float>::max() ||
177 (frame == 0 ? seconds != 0 : std::abs(double(seconds) - expected) > tolerance))
178 return Result<void>::failure(invalidRecipe("sample time does not match bake timing"));
179 return Result<void>::success();
180}
