39eve::Value nodeContract(const SchemaNode& node) {
40 eve::Value::Object result;
41 result.emplace("type", eve::Value(valueTypeName(node.type)));
42 if (node.elementType != ValueType::Any) result.emplace("elementType", eve::Value(valueTypeName(node.elementType)));
43 if (!node.ref.empty()) {
44 result.emplace("ref", eve::Value(node.ref));
45 if (node.refVersion > 0) result.emplace("refVersion", eve::Value(node.refVersion));
46 }
47 if (node.objectSchema) result.emplace("object", definitionContract(*node.objectSchema));
48 if (node.itemSchema) result.emplace("items", nodeContract(*node.itemSchema));
49 if (!node.variants.empty()) {
50 eve::Value::Array variants;
51 variants.reserve(node.variants.size());
52 for (const auto& variant : node.variants) variants.push_back(nodeContract(variant));
53 result.emplace("union", eve::Value(std::move(variants)));
54 }
55 if (!node.discriminator.empty()) result.emplace("discriminator", eve::Value(node.discriminator));
56 if (!node.discriminatorMapping.empty()) {
57 eve::Value::Object mapping;
58 for (const auto& [value, target] : node.discriminatorMapping) mapping.emplace(value, eve::Value(target));
59 result.emplace("discriminatorMapping", eve::Value(std::move(mapping)));
60 }
61 if (node.minimum) result.emplace("minimum", eve::Value(*node.minimum));
62 if (node.maximum) result.emplace("maximum", eve::Value(*node.maximum));
63 if (node.minLength) result.emplace("minLength", eve::Value(*node.minLength));
64 if (node.maxLength) result.emplace("maxLength", eve::Value(*node.maxLength));
65 if (node.minItems) result.emplace("minItems", eve::Value(*node.minItems));
66 if (node.maxItems) result.emplace("maxItems", eve::Value(*node.maxItems));
67 if (!node.enumValues.empty()) {
68 eve::Value::Array values;
69 for (const auto& value : node.enumValues) values.emplace_back(value);
70 result.emplace("enum", eve::Value(std::move(values)));
71 }
72 return eve::Value(std::move(result));
73}

75std::string markdownEscape(std::string_view text) {
76 std::string result;
77 result.reserve(text.size());
78 for (const char character : text) {

80 result += "\\|";

82 result += ' ';

84 result += character;

89void appendFieldDocumentation(std::ostringstream& output, const FieldDefinition& field, const std::string& parentPath,
90 int depth) {
91 const std::string path = parentPath + "/" + field.name;
92 output << std::string(static_cast<size_t>(depth), ' ') << "- `" << markdownEscape(path) << "` — `"
93 << valueTypeName(field.type) << "`, " << (field.required ? "required" : "optional");
94 if (!field.ref.empty()) {
95 output << ", ref=`" << markdownEscape(field.ref);
96 if (field.refVersion > 0) output << "@" << field.refVersion;
97 output << '`';
98 }
99 if (!field.discriminator.empty()) output << ", discriminator=`" << markdownEscape(field.discriminator) << '`';
100 if (!field.description.empty()) output << " — " << markdownEscape(field.description);
101 output << '\n';
102 if (field.objectSchema) {
103 for (const auto& nested : field.objectSchema->fields) appendFieldDocumentation(output, nested, path, depth + 2);
104 }
105 if (field.itemSchema && field.itemSchema->objectSchema) {
106 output << std::string(static_cast<size_t>(depth + 2), ' ') << "- array item shape:\n";
107 for (const auto& nested : field.itemSchema->objectSchema->fields)
108 appendFieldDocumentation(output, nested, path + "[]", depth + 4);
109 }
110}

114eve::Result<std::string> SchemaRegistry::generateDocumentation(const std::string& schemaId, int schemaVersion) {
115 const auto* schema = resolve(schemaId, schemaVersion);
116 if (!schema)
117 return failure<std::string>(eve::DiagnosticCode::UnknownVersion, "schema version is not registered",
118 "schemaVersion");
119
120 std::ostringstream output;
121 output << "# " << markdownEscape(schema->id) << " (v" << schema->version << ")\n\n";
122 output << "Language: Eve Schema v1\n\n";
123 output << (schema->additionalProperties ? "Unknown object fields: allow\n\n" : "Unknown object fields: reject\n\n");
124 if (!schema->title.empty()) output << "Title: " << markdownEscape(schema->title) << "\n\n";
125 if (!schema->description.empty()) output << markdownEscape(schema->description) << "\n\n";
126 output << "## Fields\n\n";
127 for (const auto& field : schema->fields) appendFieldDocumentation(output, field, {}, 0);
128 if (schema->fields.empty()) output << "(none)\n";
129 output << "\nGenerated contract: `SchemaRegistry::generateBindingContract`\n";
130 return eve::Result<std::string>::success(output.str());
131}

133eve::Result<std::string> SchemaRegistry::generateBindingContract(const std::string& schemaId, int schemaVersion) {
134 const auto* schema = resolve(schemaId, schemaVersion);
135 if (!schema)
136 return failure<std::string>(eve::DiagnosticCode::UnknownVersion, "schema version is not registered",
137 "schemaVersion");
138 eve::Value::Object result;
139 result.emplace("language", eve::Value("eve-schema-v1"));
140 result.emplace("schemaId", eve::Value(schema->id));
141 result.emplace("schemaVersion", eve::Value(schema->version));
142 result.emplace("unknownFields", eve::Value(schema->additionalProperties ? "allow" : "reject"));
143 result.emplace("root", definitionContract(*schema));
144 auto encoded = eve::Value(std::move(result)).toJson();
145 if (!encoded.ok()) return eve::Result<std::string>::failure(encoded.status());
146 return eve::Result<std::string>::success(std::move(encoded).takeValue());
147}
