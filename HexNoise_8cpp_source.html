32HexVec4 HexNoise::sample(float worldX, float worldZ) const noexcept {
33 const float fx = worldX * (1.f / kCellScale);
34 const float fz = worldZ * (1.f / kCellScale);
35 const float bx = std::floor(fx);
36 const float bz = std::floor(fz);
37 const int ix = static_cast<int>(bx);
38 const int iz = static_cast<int>(bz);
39 const float tx = smooth(fx - bx);
40 const float tz = smooth(fz - bz);
41
42 HexVec4 result;
43 float* channels[4] = {&result.x, &result.y, &result.z, &result.w};
44 for (int channel = 0; channel < 4; ++channel) {
45 const float c00 = lattice(ix, iz, channel);
46 const float c10 = lattice(ix + 1, iz, channel);
47 const float c01 = lattice(ix, iz + 1, channel);
48 const float c11 = lattice(ix + 1, iz + 1, channel);
49 const float top = c00 + (c10 - c00) * tx;
50 const float bottom = c01 + (c11 - c01) * tx;
51 *channels[channel] = top + (bottom - top) * tz;
52 }
53 return result;
54}
