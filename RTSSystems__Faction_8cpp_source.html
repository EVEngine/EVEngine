9Result<int> VeterancySystem::award(Unit& unit, float experience) {
10 auto veterancy = unit.veterancy();
11 auto durability = unit.durability();
12 auto combatPolicy = unit.combat();
13 if (!durability->alive || !std::isfinite(experience) || experience <= 0.0f)
14 return Result<int>::failure(
15 Diagnostic::error(DiagnosticCode::InvalidArgument,
16 "RTS veterancy award requires a live unit and positive finite experience", "experience"));
17 if (veterancy->veteranThreshold <= 0.0f)
18 return Result<int>::success(0, Status::success(StatusCode::NoOp));
19 if (!std::isfinite(veterancy->experience) || veterancy->experience < 0.0f ||
20 !std::isfinite(veterancy->veteranThreshold) || !std::isfinite(veterancy->eliteThreshold) ||
21 veterancy->eliteThreshold <= veterancy->veteranThreshold ||
22 !std::isfinite(veterancy->veteranDamageFactor) ||
23 !std::isfinite(veterancy->eliteDamageFactor) ||
24 !std::isfinite(veterancy->veteranHealthFactor) ||
25 !std::isfinite(veterancy->eliteHealthFactor) ||
26 veterancy->veteranDamageFactor < 1.0f ||
27 veterancy->eliteDamageFactor < veterancy->veteranDamageFactor ||
28 veterancy->veteranHealthFactor < 1.0f ||
29 veterancy->eliteHealthFactor < veterancy->veteranHealthFactor ||
30 veterancy->level < 0 || veterancy->level > 2)
31 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
32 "RTS veterancy thresholds and factors are inconsistent",
33 "unit.veterancy"));
34 veterancy->experience += experience;
35 const int oldLevel = veterancy->level;
36 if (veterancy->eliteThreshold > 0.0f && veterancy->experience >= veterancy->eliteThreshold)
37 veterancy->level = 2;
38 else if (veterancy->experience >= veterancy->veteranThreshold)
39 veterancy->level = 1;
40 if (veterancy->level == oldLevel)
41 return Result<int>::success(0, Status::success(StatusCode::Applied));
42 const auto damageAt = [&](int level) {
43 return level >= 2 ? veterancy->eliteDamageFactor
44 : level >= 1 ? veterancy->veteranDamageFactor : 1.0f;
45 };
46 const auto healthAt = [&](int level) {
47 return level >= 2 ? veterancy->eliteHealthFactor
48 : level >= 1 ? veterancy->veteranHealthFactor : 1.0f;
49 };
50 combatPolicy->upgradeDamageFactor *= damageAt(veterancy->level) / damageAt(oldLevel);
51 const double healthRatio = static_cast<double>(healthAt(veterancy->level) / healthAt(oldLevel));
52 durability->state.maxHealth *= healthRatio;
53 durability->state.health = std::min(durability->state.maxHealth,
54 durability->state.health * healthRatio);
55 return Result<int>::success(veterancy->level - oldLevel, Status::success(StatusCode::Applied));
56}

58bool FactionRelationSystem::isAllied(Faction* left, Faction* right) noexcept {
59 if (left == nullptr || right == nullptr) return false;
60 if (left == right) return true;
61 auto matches = ecs::View<Match, Match::Participants>();
62 for (auto it = matches.begin(); it != matches.end(); ++it) {
63 auto [participants] = *it;
64 const Match::Participants::Entry* leftEntry = nullptr;
65 const Match::Participants::Entry* rightEntry = nullptr;
66 for (const auto& entry : participants->entries) {
67 if (entry.eliminated) continue;
68 if (entry.faction.resolve() == left) leftEntry = &entry;
69 if (entry.faction.resolve() == right) rightEntry = &entry;
70 }
71 if (leftEntry != nullptr && rightEntry != nullptr)
72 return leftEntry->team == rightEntry->team;
73 }
74 return false;
75}
