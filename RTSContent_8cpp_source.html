46void normalizeWeapon(Value::Object& object) {
47 if (field(object, "kind") == nullptr) object.emplace("kind", Value("ranged"));
48 const double speed = number(object, "projectileSpeed", 0.0);
49 if (field(object, "logic") == nullptr) object.emplace("logic", Value(speed > 0.0 ? "projectile" : "hitscan"));
50 if (field(object, "ammo") == nullptr) {
51 const int magazine = std::max(1, static_cast<int>(number(object, "magazineSize", 0.0)));
52 const int reserve = number(object, "magazineSize", 0.0) > 0.0 ? 0 : -1;
53 object.emplace("ammo", Value(Value::Object{{"mag", Value(magazine)},
54 {"reload", Value(number(object, "reloadTime", 1.0))},
55 {"reserve", Value(reserve)}}));
56 }
57 if (field(object, "projectile") == nullptr && speed > 0.0) {
58 object.emplace("projectile", Value(Value::Object{{"aoe", Value(number(object, "splashRadius", 0.0))},
59 {"pelletCount", Value(1)},
60 {"pelletSpread", Value(0.0)},
61 {"speed", Value(speed)},
62 {"type", Value("rts")}}));
63 }
64 const std::string targets = text(object, "targets");
65 if (field(object, "targetsGround") == nullptr)
66 object.emplace("targetsGround", Value(targets != "air"));
67 if (field(object, "targetsAir") == nullptr)
68 object.emplace("targetsAir", Value(targets != "ground"));
69}

71bool available(const definitions::DefinitionRegistry& registry, const std::vector<StagedDefinition>& staged,
72 std::string_view type, std::string_view id) {
73 if (id.empty()) return true;
74 if (std::any_of(staged.begin(), staged.end(), [&](const auto& value) {
75 return value.type == type && value.id == id;
76 })) return true;
77 auto resolved = registry.resolve(std::string(type), std::string(id));
78 if (resolved) return true;
79 resolved.ignore("optional RTS content cross-reference lookup");
80 return false;
81}

83Result<void> requireReference(const definitions::DefinitionRegistry& registry,
84 const std::vector<StagedDefinition>& staged, const StagedDefinition& owner,
85 const Value::Object& object, std::string_view member, std::string_view targetType) {
86 const auto* source = field(object, member);
87 if (source == nullptr) return Result<void>::success(Status::success(StatusCode::NoOp));
88 const auto* id = source->getIf<std::string>();
89 if (id == nullptr)
90 return invalid<void>("RTS content reference must be a string", owner.type + "." + owner.id + "." +
91 std::string(member));
92 if (!available(registry, staged, targetType, *id))
93 return invalid<void>("RTS content references an unknown " + std::string(targetType) + " '" + *id + "'",
94 owner.type + "." + owner.id + "." + std::string(member));
95 return Result<void>::success(Status::success(StatusCode::Applied));
96}

100Result<ContentImportReceipt> RTSContentLoader::load(definitions::DefinitionRegistry& registry,
101 std::string_view json) {
102 auto parsed = Value::fromJson(json);
103 if (!parsed) return Result<ContentImportReceipt>::failure(parsed.status());
104 Value root = std::move(parsed).takeValue();
105 const auto* object = root.getIf<Value::Object>();
106 if (object == nullptr) return invalid<ContentImportReceipt>("RTS content root must be an object", "root");
107 static constexpr std::array<std::pair<std::string_view, std::string_view>, 7> arrays{{
108 {"weapons", "weapon"}, {"units", "unit"}, {"buildings", "building"}, {"upgrades", "upgrade"},
109 {"statusEffects", "effect"}, {"abilities", "ability"}, {"damageMultipliers", "damage_multiplier"},
110 }};
111 std::vector<StagedDefinition> staged;
112 std::set<std::pair<std::string, std::string>> keys;
113 for (const auto& [member, type] : arrays) {
114 const auto* value = field(*object, member);
115 if (value == nullptr) continue;
116 const auto* entries = value->getIf<Value::Array>();
117 if (entries == nullptr)
118 return invalid<ContentImportReceipt>("RTS content collection must be an array", std::string(member));
119 for (std::size_t index = 0; index < entries->size(); ++index) {
120 const auto* item = (*entries)[index].getIf<Value::Object>();
121 if (item == nullptr)
122 return invalid<ContentImportReceipt>("RTS content item must be an object",
123 std::string(member) + "[" + std::to_string(index) + "]");
124 Value::Object normalized = *item;
125 std::string id = text(normalized, "id");
126 if (type == "damage_multiplier" && id.empty()) {
127 const std::string damage = text(normalized, "damageType");
128 const std::string armor = text(normalized, "armorType");
129 if (!damage.empty() && !armor.empty()) {
130 id = damage + "." + armor;
131 normalized.emplace("id", Value(id));
132 }
133 }
134 if (id.empty())
135 return invalid<ContentImportReceipt>("RTS content item requires a non-empty id",
136 std::string(member) + "[" + std::to_string(index) + "].id");
137 if (!keys.emplace(std::string(type), id).second)
138 return invalid<ContentImportReceipt>("RTS content pack contains a duplicate definition",
139 std::string(type) + ":" + id);
140 if (type == "weapon") normalizeWeapon(normalized);
141 staged.push_back({std::string(type), std::move(id), Value(std::move(normalized))});
142 }
143 }
144 for (const auto& definition : staged) {
145 const auto* item = definition.value.getIf<Value::Object>();
146 if (definition.type == "unit") {
147 for (const auto& [member, target] : {std::pair{"weaponType", "weapon"},
148 {"producer", "building"}, {"prerequisite", "building"}}) {
149 auto checked = requireReference(registry, staged, definition, *item, member, target);
150 if (!checked) return Result<ContentImportReceipt>::failure(checked.status());
151 }
152 } else if (definition.type == "building") {
153 for (const auto& [member, target] : {std::pair{"weaponType", "weapon"},
154 {"prerequisite", "building"}}) {
155 auto checked = requireReference(registry, staged, definition, *item, member, target);
156 if (!checked) return Result<ContentImportReceipt>::failure(checked.status());
157 }
158 } else if (definition.type == "upgrade") {
159 for (const auto& [member, target] : {std::pair{"producer", "building"},
160 {"prerequisiteBuilding", "building"},
161 {"prerequisiteUpgrade", "upgrade"},
162 {"targetUnit", "unit"}, {"targetBuilding", "building"}}) {
163 auto checked = requireReference(registry, staged, definition, *item, member, target);
164 if (!checked) return Result<ContentImportReceipt>::failure(checked.status());
165 }
166 } else if (definition.type == "ability") {
167 for (const auto& [member, target] : {std::pair{"casterUnit", "unit"},
168 {"statusEffect", "effect"}}) {
169 auto checked = requireReference(registry, staged, definition, *item, member, target);
170 if (!checked) return Result<ContentImportReceipt>::failure(checked.status());
171 }
172 }
173 }
174
175 const std::string rollback = registry.snapshotJson();
176 ContentImportReceipt receipt;
177 for (const auto& definition : staged) {
178 auto encoded = definition.value.toJson();
179 if (!encoded) {
180 auto restored = registry.restoreJson(rollback);
181 if (!restored) return Result<ContentImportReceipt>::failure(restored.status());
182 return Result<ContentImportReceipt>::failure(encoded.status());
183 }
184 auto existing = registry.handle(definition.type, definition.id);
185 const bool replace = existing.ok();
186 if (!replace) existing.ignore("RTS content insert lookup");
187 auto published = replace ? registry.replace(definition.type, definition.id, 1, encoded.value())
188 : registry.insert(definition.type, definition.id, 1, encoded.value());
189 if (!published) {
190 const Status failure = published.status();
191 auto restored = registry.restoreJson(rollback);
192 if (!restored) return Result<ContentImportReceipt>::failure(restored.status());
193 return Result<ContentImportReceipt>::failure(failure);
194 }
195 replace ? ++receipt.replaced : ++receipt.inserted;
196 }
197 return Result<ContentImportReceipt>::success(receipt,
198 Status::success(staged.empty() ? StatusCode::NoOp : StatusCode::Applied));
199}
