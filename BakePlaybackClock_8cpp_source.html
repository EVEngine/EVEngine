11Result<void> BakePlaybackClock::advance(double elapsedSeconds, double duration) {
12 if (!std::isfinite(duration) || duration < 0 || !std::isfinite(elapsedSeconds) || elapsedSeconds < 0)
13 return Result<void>::failure(clockError("invalid playback elapsed time"));
14 const double delta = elapsedSeconds * rate_;
15 const double next = time_ + delta;
16 if (!std::isfinite(delta) || !std::isfinite(next))
17 return Result<void>::failure(clockError("playback time overflow"));
18 if (paused_ || delta == 0) return Result<void>::success();
19 double candidate = std::clamp(next, 0.0, duration);
20 if (wrap_ == BakePlaybackWrap::Loop && duration > 0) {
21 candidate = std::fmod(next, duration);
22 if (candidate < 0) candidate += duration;
23 // Rounding of a tiny negative remainder can reach the excluded endpoint.
24 if (candidate >= duration) candidate = 0;
25 }
26 time_ = candidate;
27 return Result<void>::success();
28}
