18void applySnowToHeightmap(const SnowField &field, const procgen::Heightmap &terrain,
19 procgen::Heightmap &out, float heightScale) {
20 const int w = terrain.getWidth();
21 const int h = terrain.getHeight();
22 out.resize(w, h);
23 for (int y = 0; y < h; ++y) {
24 for (int x = 0; x < w; ++x) {
25 const float s = field.inBounds(x, y) ? field.height(x, y) : 0.f;
26 out.setHeight(x, y, terrain.height(x, y) + s * heightScale);
27 }
28 }
29}
