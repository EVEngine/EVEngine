21float fade(float value) { return value * value * value * (value * (value * 6.f - 15.f) + 10.f); }
22float seamHash(int x, int y) {
23 std::uint32_t value = std::uint32_t(x) * 0x8da6b343u ^ std::uint32_t(y) * 0xd8163841u;
24 value ^= value >> 13;
25 value *= 0xcb1ab31fu;
26 value ^= value >> 16;
27 return float(value & 0x00ffffffu) / float(0x00ffffffu);
28}
29float seamNoise(float x, float y) {
30 const int x0 = static_cast<int>(std::floor(x)), y0 = static_cast<int>(std::floor(y));
31 const float tx = fade(x - float(x0)), ty = fade(y - float(y0));
32 const float a = std::lerp(seamHash(x0, y0), seamHash(x0 + 1, y0), tx);
33 const float b = std::lerp(seamHash(x0, y0 + 1), seamHash(x0 + 1, y0 + 1), tx);
34 return std::lerp(a, b, ty);
35}

119Result<int> alignTerrainSplatTextures(TerrainSplatmap& terrainA, TerrainSplatmap& terrainB,
120 const TerrainTextureAlignSettings& settings) {
121 if (&terrainA == &terrainB || !terrainA.impl_ || !terrainB.impl_ || terrainA.impl_->layers <= 0 ||
122 terrainB.impl_->layers <= 0 || terrainA.impl_->width <= 0 || terrainA.impl_->height <= 0 ||
123 terrainB.impl_->width <= 0 || terrainB.impl_->height <= 0 ||
124 terrainA.impl_->width != terrainA.impl_->height || terrainB.impl_->width != terrainB.impl_->height ||
125 !std::isfinite(settings.terrainAOriginX) || !std::isfinite(settings.terrainAOriginZ) ||
126 !std::isfinite(settings.terrainAWidth) || !std::isfinite(settings.terrainADepth) ||
127 !std::isfinite(settings.terrainBOriginX) || !std::isfinite(settings.terrainBOriginZ) ||
128 !std::isfinite(settings.terrainBWidth) || !std::isfinite(settings.terrainBDepth) ||
129 !std::isfinite(settings.blendStrength) || !std::isfinite(settings.adjacencyTolerance) ||
130 settings.terrainAWidth <= 0 || settings.terrainADepth <= 0 || settings.terrainBWidth <= 0 ||
131 settings.terrainBDepth <= 0 || settings.blendStrength < 0 || settings.blendStrength > 1 ||
132 settings.blendWidth <= 0 || settings.adjacencyTolerance <= 0)
133 return invalid<int>("terrain.textureAlign: distinct initialized square maps and valid finite settings required");
134
135 enum class Edge { None, Left, Right, Top, Bottom };
136 Edge edgeA = Edge::None, edgeB = Edge::None;
137 const double aMinX = settings.terrainAOriginX, aMaxX = aMinX + settings.terrainAWidth;
138 const double aMinZ = settings.terrainAOriginZ, aMaxZ = aMinZ + settings.terrainADepth;
139 const double bMinX = settings.terrainBOriginX, bMaxX = bMinX + settings.terrainBWidth;
140 const double bMinZ = settings.terrainBOriginZ, bMaxZ = bMinZ + settings.terrainBDepth;
141 const auto near = [&](double a, double b) { return std::abs(a - b) < settings.adjacencyTolerance; };
142 const bool xOverlaps = (bMaxX > aMinX && bMinX < aMaxX) || near(bMaxX, aMinX) ||
143 near(bMinX, aMaxX) || near(aMinX, bMinX) || near(aMaxX, bMaxX);
144 if (xOverlaps && near(aMaxZ, bMinZ)) edgeA = Edge::Right, edgeB = Edge::Left;
145 else if (xOverlaps && near(aMinZ, bMaxZ)) edgeA = Edge::Left, edgeB = Edge::Right;
146 else {
147 const bool zOverlaps = (bMaxZ > aMinZ && bMinZ < aMaxZ) || near(bMaxZ, aMinZ) ||
148 near(bMinZ, aMaxZ) || near(aMinZ, bMinZ) || near(aMaxZ, bMaxZ);
149 if (zOverlaps && near(aMaxX, bMinX)) edgeA = Edge::Top, edgeB = Edge::Bottom;
150 else if (zOverlaps && near(aMinX, bMaxX)) edgeA = Edge::Bottom, edgeB = Edge::Top;
151 }
152 if (edgeA == Edge::None)
153 return invalid<int>("terrain.textureAlign: terrain bounds are not adjacent");
154
155 auto nextA = *terrainA.impl_, nextB = *terrainB.impl_;
156 const int resolution = std::min(nextA.width, nextB.width);
157 const int layers = std::min(nextA.layers, nextB.layers);
158 const bool horizontal = edgeA == Edge::Left || edgeA == Edge::Right;
159 int changedA = 0, changedB = 0;
160 for (int i = 0; i < settings.blendWidth; ++i) {
161 const float t = std::clamp(float(i) / float(settings.blendWidth), 0.f, 1.f);
162 const float blend = settings.blendStrength * (1.f - t * t * (3.f - 2.f * t));
163 for (int j = 0; j < resolution; ++j) {
164 const int depthA = std::min(i, resolution - 1), depthB = std::min(i, resolution - 1);
165 const int xA = horizontal ? (edgeA == Edge::Left ? depthA : resolution - 1 - depthA) : j;
166 const int yA = horizontal ? j : (edgeA == Edge::Bottom ? depthA : resolution - 1 - depthA);
167 const int xB = horizontal ? (edgeB == Edge::Left ? depthB : resolution - 1 - depthB) : j;
168 const int yB = horizontal ? j : (edgeB == Edge::Bottom ? depthB : resolution - 1 - depthB);
169 const float noise = seamNoise(horizontal ? j * 0.1f : i * 0.1f,
170 horizontal ? i * 0.1f : j * 0.1f) * 0.5f + 0.5f;
171 const float weightA = noise * blend, weightB = (1.f - noise) * blend;
172 std::vector<float> oldA(static_cast<std::size_t>(layers));
173 std::vector<float> oldB(static_cast<std::size_t>(layers));
174 for (int layer = 0; layer < layers; ++layer) {
175 oldA[static_cast<std::size_t>(layer)] = terrainA.impl_->weights[terrainA.impl_->index(layer, xA, yA)];
176 oldB[static_cast<std::size_t>(layer)] = terrainB.impl_->weights[terrainB.impl_->index(layer, xB, yB)];
177 }
178 float sumA = 0, sumB = 0;
179 for (int layer = 0; layer < layers; ++layer) {
180 const float a = oldA[static_cast<std::size_t>(layer)], b = oldB[static_cast<std::size_t>(layer)];
181 nextA.weights[nextA.index(layer, xA, yA)] = std::lerp(a, b, weightA);
182 nextB.weights[nextB.index(layer, xB, yB)] = std::lerp(b, a, weightB);
183 sumA += nextA.weights[nextA.index(layer, xA, yA)];
184 sumB += nextB.weights[nextB.index(layer, xB, yB)];
185 }
186 if (sumA > 0 && sumB > 0) for (int layer = 0; layer < layers; ++layer) {
187 nextA.weights[nextA.index(layer, xA, yA)] /= sumA;
188 nextB.weights[nextB.index(layer, xB, yB)] /= sumB;
189 }
190 bool aChanged = false, bChanged = false;
191 for (int layer = 0; layer < layers; ++layer) {
192 aChanged |= nextA.weights[nextA.index(layer, xA, yA)] != oldA[static_cast<std::size_t>(layer)];
193 bChanged |= nextB.weights[nextB.index(layer, xB, yB)] != oldB[static_cast<std::size_t>(layer)];
194 }
195 changedA += aChanged;
196 changedB += bChanged;
197 }
198 }
199 nextA.lastChanged = changedA;
200 nextB.lastChanged = changedB;
201 *terrainA.impl_ = std::move(nextA);
202 *terrainB.impl_ = std::move(nextB);
203 return Result<int>::success(changedA + changedB);
204}

206Result<int> paintTerrainSplatLayer(TerrainSplatmap& target, const Heightmap& paint, int targetLayer) {
207 using namespace raster_detail;
208 if (!target.impl_ || target.impl_->layers <= 0 || !validRaster(paint) ||
209 paint.getWidth() != target.impl_->width || paint.getHeight() != target.impl_->height ||
210 targetLayer < 0 || targetLayer >= target.impl_->layers ||
211 !std::all_of(paint.data().begin(), paint.data().end(), [](float value) {
212 return std::isfinite(value) && value >= 0 && value <= 1;
213 }) || (target.impl_->layers == 1 && std::any_of(paint.data().begin(), paint.data().end(),
214 [](float value) { return value != 1; })))
215 return invalid<int>("terrain.paintSplat: initialized matching topology, layer and normalized paint required");
216 auto next = *target.impl_;
217 int changed = 0;
218 for (int y = 0; y < next.height; ++y) {
219 for (int x = 0; x < next.width; ++x) {
220 const auto targetIndex = next.index(targetLayer, x, y);
221 const float oldTarget = next.weights[targetIndex], replacement = paint.height(x, y);
222 if (oldTarget != replacement) ++changed;
223 const double oldOther = 1.0 - oldTarget;
224 if (oldOther > 1e-7) {
225 const double factor = (1.0 - replacement) / oldOther;
226 for (int layer = 0; layer < next.layers; ++layer)
227 if (layer != targetLayer) next.weights[next.index(layer, x, y)] =
228 static_cast<float>(next.weights[next.index(layer, x, y)] * factor);
229 } else {
230 const float share = next.layers > 1 ? (1 - replacement) / float(next.layers - 1) : 0;
231 for (int layer = 0; layer < next.layers; ++layer)
232 if (layer != targetLayer) next.weights[next.index(layer, x, y)] = share;
233 }
234 next.weights[targetIndex] = replacement;
235 }
236 }
237 next.lastChanged = changed;
238 *target.impl_ = std::move(next);
239 return Result<int>::success(changed);
240}

242Result<int> applyGtsHeightBlend(TerrainSplatmap& output, const TerrainSplatmap& input,
243 const GtsHeightBlendSet& heights, float blendFactor) {
244 if (!input.impl_ || input.impl_->layers <= 0 || !heights.impl_ ||
245 heights.impl_->layers.size() != static_cast<std::size_t>(input.impl_->layers) ||
246 !std::isfinite(blendFactor) || blendFactor < 0 || blendFactor > 1)
247 return invalid<int>("terrain.gtsHeightBlend: initialized splat, matching heights and normalized factor required");
248 for (const auto& layer : heights.impl_->layers)
249 if (layer.height.getWidth() != input.impl_->width || layer.height.getHeight() != input.impl_->height)
250 return invalid<int>("terrain.gtsHeightBlend: topology mismatch");
251 auto next = *input.impl_;
252 int changed = 0;
253 std::vector<float> weighted(static_cast<std::size_t>(next.layers));
254 for (int y = 0; y < next.height; ++y) for (int x = 0; x < next.width; ++x) {
255 float maxHeight = 0;
256 for (int layer = 0; layer < next.layers; ++layer) {
257 const auto& source = heights.impl_->layers[static_cast<std::size_t>(layer)];
258 const float transformed = std::pow(std::abs(source.height.height(x, y)), source.contrast) *
259 source.brightness + source.increase;
260 weighted[static_cast<std::size_t>(layer)] = transformed * input.impl_->weights[input.impl_->index(layer, x, y)];
261 maxHeight = std::max(maxHeight, weighted[static_cast<std::size_t>(layer)]);
262 }
263 float sum = 0;
264 const float transition = std::max(blendFactor, 1e-5F);
265 for (int layer = 0; layer < next.layers; ++layer) {
266 float& value = weighted[static_cast<std::size_t>(layer)];
267 value = std::max(0.0F, value + transition - maxHeight);
268 value = (value + 1e-6F) * input.impl_->weights[input.impl_->index(layer, x, y)];
269 sum += value;
270 }
271 sum = std::max(sum, 1e-6F);
272 bool texelChanged = false;
273 for (int layer = 0; layer < next.layers; ++layer) {
274 const auto index = next.index(layer, x, y);
275 const float value = weighted[static_cast<std::size_t>(layer)] / sum;
276 texelChanged |= value != next.weights[index]; next.weights[index] = value;
277 }
278 changed += texelChanged;
279 }
280 next.lastChanged = changed;
281 if (!output.impl_) output.impl_ = std::make_unique<TerrainSplatmap::Impl>();
282 *output.impl_ = std::move(next);
283 return Result<int>::success(changed);
284}

287 const std::vector<TerrainSplatTile>& tiles, const Heightmap& operationPaint, int targetLayer,
288 const TerrainStampSettings& operationSettings, bool worldMapOperation,
289 const std::vector<std::string>& validTerrainNames) {
290 using namespace raster_detail;
291 if (tiles.empty() || !validRaster(operationPaint) ||
292 !std::all_of(operationPaint.data().begin(), operationPaint.data().end(), [](float value) {
293 return std::isfinite(value) && value >= 0 && value <= 1;
294 }))
295 return invalid<TerrainMultiTileReport>("terrain.paintSplat.multitile: tiles and normalized paint required");
296 std::vector<TerrainOperationTile> descriptors;
297 std::unordered_set<TerrainSplatmap*> owners;
298 descriptors.reserve(tiles.size());
299 for (const auto& tile : tiles) {
300 if (!tile.splatmap || !tile.splatmap->impl_ || tile.splatmap->impl_->layers <= 0 ||

308 worldMapOperation, validTerrainNames);

311 if (operationPaint.getWidth() != report.operationWidth || operationPaint.getHeight() != report.operationHeight)
312 return invalid<TerrainMultiTileReport>(
313 "terrain.paintSplat.multitile: paint dimensions must match the shared operation window");
314 struct Candidate { TerrainSplatmap* target; TerrainSplatmap::Impl value; };
315 std::vector<Candidate> candidates;
316 candidates.reserve(report.mappings.size());
317 for (const auto& mapping : report.mappings) {
318 const auto tile = std::find_if(tiles.begin(), tiles.end(),
319 [&](const auto& value) { return value.name == mapping.terrainName; });
320 if (tile == tiles.end())

322 DiagnosticCode::InvariantViolation, "terrain.paintSplat.multitile: mapped terrain was not supplied"));
323 if (tile->splatmap->impl_->layers == 1) {
324 for (int y = 0; y < mapping.height; ++y)
325 for (int x = 0; x < mapping.width; ++x)
326 if (operationPaint.height(mapping.operationX + x, mapping.operationY + y) != 1)
327 return invalid<TerrainMultiTileReport>(
328 "terrain.paintSplat.multitile: a selected single-layer splatmap must remain one");
329 }
330 auto next = *tile->splatmap->impl_;
331 int changed = 0;
332 for (int y = 0; y < mapping.height; ++y) {
333 for (int x = 0; x < mapping.width; ++x) {
334 const int localX = mapping.localX + x, localY = mapping.localY + y;
335 const float replacement = operationPaint.height(mapping.operationX + x, mapping.operationY + y);
336 const auto index = next.index(targetLayer, localX, localY);
337 const float oldTarget = next.weights[index];
338 if (oldTarget != replacement) ++changed;
339 const double oldOther = 1.0 - oldTarget;
340 if (oldOther > 1e-7) {
341 const double factor = (1.0 - replacement) / oldOther;
342 for (int layer = 0; layer < next.layers; ++layer)
343 if (layer != targetLayer) next.weights[next.index(layer, localX, localY)] =
344 static_cast<float>(next.weights[next.index(layer, localX, localY)] * factor);
345 } else {
346 const float share = next.layers > 1 ? (1 - replacement) / float(next.layers - 1) : 0;
347 for (int layer = 0; layer < next.layers; ++layer)
348 if (layer != targetLayer) next.weights[next.index(layer, localX, localY)] = share;
349 }
350 next.weights[index] = replacement;
351 }
352 }
353 next.lastChanged = changed;
354 if (report.changedSamples > std::numeric_limits<int>::max() - changed)
355 return invalid<TerrainMultiTileReport>("terrain.paintSplat.multitile: changed texel count exceeds range");
356 report.changedSamples += changed;
357 candidates.push_back({tile->splatmap, std::move(next)});
358 }
359 for (auto& candidate : candidates) *candidate.target->impl_ = std::move(candidate.value);
360 return Result<TerrainMultiTileReport>::success(std::move(report));
361}

396Result<int> TerrainMultiSplatWorkspace::addTile(const std::string& name, const TerrainSplatmap& splatmap,
397 double originX, double originZ, double width, double depth,
398 bool worldMap) {
399 if (!impl_ || name.empty() || splatmap.getWidth() <= 0 || splatmap.getHeight() <= 0)
400 return invalid<int>("terrain.splat.workspace: initialized named tile required");
401 if (std::any_of(impl_->tiles.begin(), impl_->tiles.end(), [&](const auto& tile) { return tile.name == name; }))
402 return invalid<int>("terrain.splat.workspace: duplicate tile name");
403 if (impl_->history.size() > 1)
404 return invalid<int>("terrain.splat.workspace: topology is fixed after painting");
405 auto candidate = std::make_unique<Impl>(*impl_);
406 candidate->tiles.push_back({name, splatmap, originX, originZ, width, depth, worldMap});
407 std::vector<TerrainOperationTile> descriptors;
408 for (const auto& tile : candidate->tiles)
409 descriptors.push_back({tile.name, tile.originX, tile.originZ, tile.width, tile.depth,
410 tile.splatmap.getWidth(), tile.splatmap.getHeight(), tile.worldMap});
411 TerrainStampSettings bounds;
412 bounds.centerX = originX + width * 0.5;
413 bounds.centerZ = originZ + depth * 0.5;
414 bounds.width = width;
415 bounds.depth = depth;
416 auto checked = mapTerrainOperationMultiTile(descriptors, bounds, TerrainOperationDomain::Texture, worldMap, {name});
417 if (!checked.ok()) return Result<int>::failure(checked.status());
418 candidate->history.clear();
419 candidate->history.push_back(candidate->snapshot());
420 candidate->cursor = 0;
421 impl_.swap(candidate);
422 return Result<int>::success(static_cast<int>(impl_->tiles.size()));
423}

425Result<int> TerrainMultiSplatWorkspace::paint(const Heightmap& operationPaint, int targetLayer,
426 const TerrainStampSettings& operationSettings,
427 bool worldMapOperation) {
428 if (!impl_) return invalid<int>("terrain.splat.workspace: moved-from workspace");
429 auto candidate = std::make_unique<Impl>(*impl_);
430 std::vector<TerrainSplatTile> descriptors;
431 for (auto& tile : candidate->tiles)
432 descriptors.push_back({tile.name, &tile.splatmap, tile.originX, tile.originZ, tile.width, tile.depth,
433 tile.worldMap});
434 auto painted = paintTerrainSplatLayerMultiTile(descriptors, operationPaint, targetLayer, operationSettings,
435 worldMapOperation);
436 if (!painted.ok()) return Result<int>::failure(painted.status());
437 candidate->last = std::move(painted.value());
438 candidate->history.resize(static_cast<std::size_t>(candidate->cursor + 1));
439 candidate->history.push_back(candidate->snapshot());
440 ++candidate->cursor;
441 const int changed = candidate->last.changedSamples;
442 impl_.swap(candidate);
443 return Result<int>::success(changed);
444}
