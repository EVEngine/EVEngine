166Result<void> EvpackStaticPrefab::release() {
167 if (impl_->resourcePins)
168 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
169 "prefab resource pins must be released before GPU resources"));
170 impl_->released = true;
171 if (impl_->gpuGraphics) {
172 for (const auto& material : impl_->gpuMaterials) {
173 auto released = impl_->gpuGraphics->gpuDrivenReleaseMaterialRecord(material.get());
174 if (!released) return released;
175 }
176 impl_->gpuGraphics = nullptr;
177 }
178 impl_->gpuMaterials.clear();
179 impl_->draws.clear();
180 Status failure;
181 for (auto it = impl_->meshLeases.begin(); it != impl_->meshLeases.end();) {
182 auto result = impl_->meshes.releaseMesh(it->second);
183 if (result)
184 it = impl_->meshLeases.erase(it);
185 else {
186 failure = result.status();
187 ++it;
188 }
189 }
190 for (auto it = impl_->imageLeases.begin(); it != impl_->imageLeases.end();) {
191 auto result = impl_->images.releaseImage(it->second);
192 if (result)
193 it = impl_->imageLeases.erase(it);
194 else {
195 failure = result.status();
196 ++it;
197 }
198 }
199 if (failure.isFailure()) return Result<void>::failure(failure);
200 return Result<void>::success();
201}
202Result<std::vector<StaticPrefabGpuPart>> EvpackStaticPrefab::prepareGpuDriven(graphics::Graphics& gfx) {
203 if (impl_->released)
204 return Result<std::vector<StaticPrefabGpuPart>>::failure(Diagnostic::error(
205 DiagnosticCode::InvalidArgument, "prefab resources have been released", {}, {}, "asset.graphics.prefab"));
206 if (!impl_->gpuMaterials.empty())
207 return Result<std::vector<StaticPrefabGpuPart>>::failure(
208 Diagnostic::error(DiagnosticCode::InvalidArgument, "prefab GPU-driven parts are already registered", {}, {},
209 "asset.graphics.prefab"));
210 std::vector<std::unique_ptr<graphics::Material>> materials;
211 std::vector<StaticPrefabGpuPart> parts;
212 materials.reserve(impl_->draws.size());
213 parts.reserve(impl_->draws.size());
214 for (const auto& draw : impl_->draws) {
215 if (draw.material.transparent)
216 return Result<std::vector<StaticPrefabGpuPart>>::failure(Diagnostic::error(
217 DiagnosticCode::InvalidArgument, "transparent prefab Detail cannot use opaque GPU submission", {}, {},
218 "asset.graphics.prefab"));
219 bool unsupportedImage = false;
220 for (std::size_t role = 1; role < draw.material.images.size(); ++role) {
221 if (role != std::size_t(graphics::PbrTextureSlot::Normal) && draw.material.images[role])
222 unsupportedImage = true;
223 }
224 const auto conventionalBinding = [](const graphics::PbrTextureBinding& binding) {
225 return binding.texcoord == 0 && binding.offset == std::array<float, 2>{0.f, 0.f} &&
226 binding.scale == std::array<float, 2>{1.f, 1.f} && binding.rotation == 0.f;
227 };
228 if (unsupportedImage ||
229 std::any_of(draw.material.detailImages.begin(), draw.material.detailImages.end(),
230 [](const auto& image) { return image.has_value(); }) ||
231 draw.material.vegetationSurface || draw.material.surface.normalScale != 1.f ||
232 draw.material.surface.normalMode != graphics::PbrNormalMode::TangentXYZ ||
233 !conventionalBinding(draw.material.surface.textures[0]) ||
234 !conventionalBinding(draw.material.surface.textures[std::size_t(graphics::PbrTextureSlot::Normal)]))
235 return Result<std::vector<StaticPrefabGpuPart>>::failure(Diagnostic::error(
236 DiagnosticCode::InvalidArgument, "extended prefab material requires a GPU-driven PBR surface record",
237 {}, {}, "asset.graphics.prefab"));
238 auto material = std::make_unique<graphics::Material>();
239 material->setTint(draw.material.color.r, draw.material.color.g, draw.material.color.b, draw.material.color.a);
240 material->setMetallic(draw.material.metallic);
241 material->setRoughness(draw.material.roughness);
242 material->setSurfaceMode(draw.material.masked ? "masked" : "opaque");
243 material->setAlphaCutoff(draw.material.alphaCutoff);
244 material->setDoubleSided(draw.material.doubleSided);
245 material->setAlbedoTexture(draw.image);
246 material->setNormalTexture(
247 draw.material.surface.textures[std::size_t(graphics::PbrTextureSlot::Normal)].texture);
248 if (!gfx.gpuDrivenMaterialUsable(material.get()))
249 return Result<std::vector<StaticPrefabGpuPart>>::failure(Diagnostic::error(
250 DiagnosticCode::InvalidArgument, "prefab material is unavailable to GPU-driven rendering", {}, {},
251 "asset.graphics.prefab"));
252 materials.push_back(std::move(material));
253 }
254 const auto releaseRegistered = [&gfx](const std::vector<std::unique_ptr<graphics::Material>>& owned,
255 std::size_t count) {
256 for (std::size_t i = 0; i < count; ++i) {
257 auto released = gfx.gpuDrivenReleaseMaterialRecord(owned[i].get());
258 if (!released) return released;
259 }
260 return Result<void>::success();
261 };
262 for (std::size_t drawIndex = 0; drawIndex < impl_->draws.size(); ++drawIndex) {
263 const auto& draw = impl_->draws[drawIndex];
264 auto& material = materials[drawIndex];
265 StaticPrefabGpuPart part;
266 std::copy(glm::value_ptr(draw.transform), glm::value_ptr(draw.transform) + 16, part.localTransform.begin());
267 part.meshId = gfx.gpuDrivenMeshRecord(draw.mesh);
268 part.materialId = gfx.gpuDrivenMaterialRecord(material.get());
269 if (part.meshId == graphics::kInvalidGpuDrivenSlot || part.materialId == graphics::kInvalidGpuDrivenSlot) {
270 auto cleanup = releaseRegistered(materials, drawIndex + 1);
271 if (!cleanup) return Result<std::vector<StaticPrefabGpuPart>>::failure(cleanup.status());
272 return Result<std::vector<StaticPrefabGpuPart>>::failure(
273 Diagnostic::error(DiagnosticCode::InvalidArgument, "prefab GPU table registration failed", {}, {},
274 "asset.graphics.prefab"));
275 }
276 parts.push_back(part);
277 }
278 if (parts.empty())
279 return Result<std::vector<StaticPrefabGpuPart>>::failure(Diagnostic::error(
280 DiagnosticCode::InvalidArgument, "prefab has no enabled renderer", {}, {}, "asset.graphics.prefab"));
281 impl_->gpuMaterials = std::move(materials);
282 impl_->gpuGraphics = &gfx;
283 return Result<std::vector<StaticPrefabGpuPart>>::success(std::move(parts));
284}
285Result<std::unique_ptr<EvpackStaticPrefab>> EvpackStaticPrefab::load(const asset::EvpackResourceReader& reader,
286 graphics::IMeshResourceFactory& meshes,
287 graphics::IImageResourceFactory& images,
288 const AssetRef& ref,
289 const asset::EvpackCapabilities& caps) {
290 auto scene = asset_scene::EvpackSceneTemplateLoader(reader).load(ref, caps);
291 if (!scene) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(scene.status());
292 auto build = reader.buildId();
293 if (!build.ok()) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(build.status());
294 auto out =
295 std::unique_ptr<EvpackStaticPrefab>(new EvpackStaticPrefab(std::make_unique<Impl>(Impl{meshes, images})));
296 out->impl_->sourceAsset = ref;
297 out->impl_->sourceBuild = build.value();
298 for (unsigned axis = 0; axis < 3; ++axis) {
299 out->impl_->localBounds[axis] = std::numeric_limits<float>::max();
300 out->impl_->localBounds[axis + 3] = std::numeric_limits<float>::lowest();
301 out->impl_->placementBounds[axis] = std::numeric_limits<float>::max();
302 out->impl_->placementBounds[axis + 3] = std::numeric_limits<float>::lowest();
303 }
304 std::map<SceneObjectId, glm::mat4> transforms;
305 std::map<SceneObjectId, glm::mat4> placementTransforms;
306 std::function<void(const scene::NodeDesc&, glm::mat4, glm::mat4, bool)> walk =
307 [&](const scene::NodeDesc& node, glm::mat4 parent, glm::mat4 placedParent, bool root) {
308 if (!node.visible) return;
309 auto local = glm::translate(glm::mat4(1), glm::vec3(node.x, node.y, node.z));
310 local = glm::rotate(local, glm::radians(node.yaw), glm::vec3(0, 1, 0));
311 local = glm::rotate(local, glm::radians(node.pitch), glm::vec3(1, 0, 0));
312 local = glm::rotate(local, glm::radians(node.roll), glm::vec3(0, 0, 1));
313 local = glm::scale(local, glm::vec3(node.sx, node.sy, node.sz));
314 auto world = parent * local;
315 auto placed =
316 placedParent * (root ? glm::scale(glm::mat4(1), glm::vec3(node.sx, node.sy, node.sz)) : local);
317 transforms.emplace(node.persistentId, world);
318 placementTransforms.emplace(node.persistentId, placed);
319 for (const auto& child : node.children) walk(child, world, placed, false);
320 };
321 // The template decoder owns an identity-only synthetic container; its children
322 // are the authored source roots whose poses the placement convention replaces.
323 for (const auto& sourceRoot : scene.value().root.children) walk(sourceRoot, glm::mat4(1), glm::mat4(1), true);
324 EvpackGraphicsLoader meshLoader(reader, meshes);
325 EvpackImageLoader imageLoader(reader, images);
326 for (const auto& binding : scene.value().renderers) {
327 if (!binding.enabled || !transforms.contains(binding.object)) continue;
328 auto parameters = detail::readCookedMaterial(reader, binding.material, caps);
329 if (!parameters) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(parameters.status());
330 const auto meshKey = binding.mesh.format();
331 if (!out->impl_->meshLeases.contains(meshKey)) {
332 auto loaded = meshLoader.loadMesh(binding.mesh, caps, {}, 0, true);
333 if (!loaded) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(loaded.status());
334 out->impl_->meshLeases.emplace(meshKey, loaded.value().mesh);
335 out->impl_->meshUvSets.emplace(meshKey, loaded.value().availableTexcoordSets);
336 out->impl_->meshBounds.emplace(meshKey, loaded.value().localBounds);
337 }
338 graphics::Texture* image = nullptr;
339 for (size_t role = 0; role < 11; role++) {
340 if (!parameters.value().images[role]) continue;
341 const auto& sets = out->impl_->meshUvSets.at(meshKey);
342 auto selected = parameters.value().surface.textures[role].texcoord;
343 if (std::find(sets.begin(), sets.end(), selected) == sets.end())
344 return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(
345 Diagnostic::error(DiagnosticCode::InvalidArgument, "material references missing mesh UV channel",
346 {}, {}, "asset.graphics.prefab"));
347 const auto& ref = *parameters.value().images[role];
348 const auto imageKey = ref.format();
349 if (!out->impl_->imageLeases.contains(imageKey)) {
350 auto loaded = imageLoader.load(ref, caps);
351 if (!loaded) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(loaded.status());
352 out->impl_->imageLeases.emplace(imageKey, loaded.value().texture);
353 }
354 parameters.value().surface.textures[role].texture = out->impl_->imageLeases.at(imageKey);
355 }
356 for (size_t role = 0; role < 3; ++role) {
357 if (!parameters.value().detailImages[role]) continue;
358 const auto& ref = *parameters.value().detailImages[role];
359 const auto imageKey = ref.format();
360 if (!out->impl_->imageLeases.contains(imageKey)) {
361 auto loaded = imageLoader.load(ref, caps);
362 if (!loaded) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(loaded.status());
363 out->impl_->imageLeases.emplace(imageKey, loaded.value().texture);
364 }
365 parameters.value().surface.vegetationDetail.textures[role].texture = out->impl_->imageLeases.at(imageKey);
366 }
367 auto validatedSurface = graphics::validatePbrSurface(parameters.value().surface);
368 if (!validatedSurface) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(validatedSurface.status());
369 image = parameters.value().surface.textures[0].texture;
370 const auto& bounds = out->impl_->meshBounds.at(meshKey);
371 const auto& matrix = transforms.at(binding.object);
372 const auto& placementMatrix = placementTransforms.at(binding.object);
373 for (unsigned corner = 0; corner < 8; ++corner) {
374 const auto p = matrix * glm::vec4(bounds[(corner & 1) ? 3 : 0], bounds[(corner & 2) ? 4 : 1],
375 bounds[(corner & 4) ? 5 : 2], 1.f);
376 const auto placed = placementMatrix * glm::vec4(bounds[(corner & 1) ? 3 : 0], bounds[(corner & 2) ? 4 : 1],
377 bounds[(corner & 4) ? 5 : 2], 1.f);
378 for (unsigned axis = 0; axis < 3; ++axis) {
379 if (!std::isfinite(p[axis]) || !std::isfinite(placed[axis]))
380 return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(Diagnostic::error(
381 DiagnosticCode::InvalidArgument, "transformed prefab bounds are not representable", {}, {},
382 "asset.graphics.prefab"));
383 out->impl_->localBounds[axis] = std::min(out->impl_->localBounds[axis], p[axis]);
384 out->impl_->localBounds[axis + 3] = std::max(out->impl_->localBounds[axis + 3], p[axis]);
385 out->impl_->placementBounds[axis] = std::min(out->impl_->placementBounds[axis], placed[axis]);
386 out->impl_->placementBounds[axis + 3] = std::max(out->impl_->placementBounds[axis + 3], placed[axis]);
387 }
388 }
389 out->impl_->hasLocalBounds = true;
390 out->impl_->draws.push_back({transforms.at(binding.object), placementTransforms.at(binding.object),
391 out->impl_->meshLeases.at(meshKey), image, std::move(parameters).takeValue(),
392 binding.castShadows, binding.receiveShadows});
393 }
394 return Result<std::unique_ptr<EvpackStaticPrefab>>::success(std::move(out));
395}
396Result<void> EvpackStaticPrefab::draw(graphics::Graphics& gfx, const std::array<float, 16>& transform,
397 const std::array<float, 16>& cameraView) const {
398 if (impl_->released)
399 return Result<void>::failure(Diagnostic::error(
400 DiagnosticCode::InvalidArgument, "prefab resources have been released", {}, {}, "asset.graphics.prefab"));
401 for (float value : transform)
402 if (!std::isfinite(value))
403 return Result<void>::failure(Diagnostic::error(
404 DiagnosticCode::InvalidArgument, "nonfinite instance transform", {}, {}, "asset.graphics.prefab"));
405 for (float value : cameraView)
406 if (!std::isfinite(value))
407 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "nonfinite camera view", {},
408 {}, "asset.graphics.prefab"));
409 const auto instance = glm::make_mat4(transform.data());
410 const auto view = glm::make_mat4(cameraView.data());
411 std::vector<std::size_t> order(impl_->draws.size());
412 std::iota(order.begin(), order.end(), 0);
413 std::stable_sort(order.begin(), order.end(), [&](std::size_t a, std::size_t b) {
414 const auto& left = impl_->draws[a];
415 const auto& right = impl_->draws[b];
416 if (left.material.transparent != right.material.transparent) return !left.material.transparent;
417 return left.material.transparent &&
418 (view * instance * left.transform)[3].z < (view * instance * right.transform)[3].z;
419 });
420 for (auto index : order) {
421 const auto& draw = impl_->draws[index];
422 gfx.setMesh3DSurface(draw.material.transparent ? graphics::SurfaceMode::Transparent
423 : draw.material.masked ? graphics::SurfaceMode::Masked
424 : graphics::SurfaceMode::Opaque,
425 draw.material.blend, !draw.material.transparent, draw.material.doubleSided,
426 draw.material.alphaCutoff, "cutoff");
427 auto bound = gfx.setMesh3DPbrSurface(&draw.material.surface);
428 if (!bound) return bound;
429 gfx.setMesh3DMaterial(draw.material.metallic, draw.material.roughness);
430 gfx.setMesh3DNormalTexture(nullptr);
431 gfx.setMesh3DHeightTexture(nullptr);
432 gfx.setMesh3DVirtualTexture(false, 1, 1, 1, 1, 0.f);
433 gfx.setMesh3DTexCellBomb(1.f, 0.f, 0.f);

443 const std::array<float, 16>& transform,
444 const std::array<float, 16>& lightViewProjection) const {
445 if (impl_->released)
446 return Result<void>::failure(Diagnostic::error(
447 DiagnosticCode::InvalidArgument, "prefab resources have been released", {}, {}, "asset.graphics.prefab"));
448 for (float value : transform)
449 if (!std::isfinite(value))
450 return Result<void>::failure(Diagnostic::error(
451 DiagnosticCode::InvalidArgument, "nonfinite instance transform", {}, {}, "asset.graphics.prefab"));
452 for (float value : lightViewProjection)
453 if (!std::isfinite(value))
454 return Result<void>::failure(Diagnostic::error(
455 DiagnosticCode::InvalidArgument, "nonfinite shadow view projection", {}, {}, "asset.graphics.prefab"));
456 const glm::mat4 instance = glm::make_mat4(transform.data());
457 const glm::mat4 lightVP = glm::make_mat4(lightViewProjection.data());
458 for (const auto& draw : impl_->draws) {
459 if (!draw.castShadows || draw.material.transparent) continue;
460 const glm::mat4 lightMvp = lightVP * instance * draw.transform;
461 const bool doubleSided = draw.material.surface.cullMode == graphics::PbrCullMode::None ||
462 (draw.material.surface.cullMode == graphics::PbrCullMode::Inherit &&
463 draw.material.doubleSided);
464 if (draw.material.masked)
465 gfx.drawMeshShadowAlpha(draw.mesh, lightMvp, draw.image, doubleSided);
466 else
467 gfx.drawMeshShadow(draw.mesh, lightMvp, doubleSided);
468 }
469 return Result<void>::success();
470}
