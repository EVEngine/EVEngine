50Result<void> EvpackStaticPrefab::release() {
51 impl_->released = true;
52 impl_->draws.clear();
53 Status failure;
54 for (auto it = impl_->meshLeases.begin(); it != impl_->meshLeases.end();) {
55 auto result = impl_->meshes.releaseMesh(it->second);
56 if (result)
57 it = impl_->meshLeases.erase(it);
58 else {
59 failure = result.status();
60 ++it;
61 }
62 }
63 for (auto it = impl_->imageLeases.begin(); it != impl_->imageLeases.end();) {
64 auto result = impl_->images.releaseImage(it->second);
65 if (result)
66 it = impl_->imageLeases.erase(it);
67 else {
68 failure = result.status();
69 ++it;
70 }
71 }
72 if (failure.isFailure()) return Result<void>::failure(failure);
73 return Result<void>::success();
74}
75Result<std::unique_ptr<EvpackStaticPrefab>> EvpackStaticPrefab::load(const asset::EvpackResourceReader& reader,
76 graphics::IMeshResourceFactory& meshes,
77 graphics::IImageResourceFactory& images,
78 const AssetRef& ref,
79 const asset::EvpackCapabilities& caps) {
80 auto scene = asset_scene::EvpackSceneTemplateLoader(reader).load(ref, caps);
81 if (!scene) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(scene.status());
82 auto out =
83 std::unique_ptr<EvpackStaticPrefab>(new EvpackStaticPrefab(std::make_unique<Impl>(Impl{meshes, images})));
84 std::map<SceneObjectId, glm::mat4> transforms;
85 std::function<void(const scene::NodeDesc&, glm::mat4)> walk = [&](const scene::NodeDesc& node, glm::mat4 parent) {
86 if (!node.visible) return;
87 auto local = glm::translate(glm::mat4(1), glm::vec3(node.x, node.y, node.z));
88 local = glm::rotate(local, glm::radians(node.yaw), glm::vec3(0, 1, 0));
89 local = glm::rotate(local, glm::radians(node.pitch), glm::vec3(1, 0, 0));
90 local = glm::rotate(local, glm::radians(node.roll), glm::vec3(0, 0, 1));
91 local = glm::scale(local, glm::vec3(node.sx, node.sy, node.sz));
92 auto world = parent * local;
93 transforms.emplace(node.persistentId, world);
94 for (const auto& child : node.children) walk(child, world);
95 };
96 walk(scene.value().root, glm::mat4(1));
97 EvpackGraphicsLoader meshLoader(reader, meshes);
98 EvpackImageLoader imageLoader(reader, images);
99 for (const auto& binding : scene.value().renderers) {
100 if (!binding.enabled || !transforms.contains(binding.object)) continue;
101 auto parameters = detail::readCookedMaterial(reader, binding.material, caps);
102 if (!parameters) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(parameters.status());
103 const auto meshKey = binding.mesh.format();
104 if (!out->impl_->meshLeases.contains(meshKey)) {
105 auto loaded = meshLoader.loadMesh(binding.mesh, caps, {}, 0, true);
106 if (!loaded) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(loaded.status());
107 out->impl_->meshLeases.emplace(meshKey, loaded.value().mesh);
108 out->impl_->meshUvSets.emplace(meshKey, loaded.value().availableTexcoordSets);
109 }
110 graphics::Texture* image = nullptr;
111 for (size_t role = 0; role < 11; role++) {
112 if (!parameters.value().images[role]) continue;
113 const auto& sets = out->impl_->meshUvSets.at(meshKey);
114 auto selected = parameters.value().surface.textures[role].texcoord;
115 if (std::find(sets.begin(), sets.end(), selected) == sets.end())
116 return fail<std::unique_ptr<EvpackStaticPrefab>>("material references missing mesh UV channel");
117 const auto& ref = *parameters.value().images[role];
118 const auto imageKey = ref.format();
119 if (!out->impl_->imageLeases.contains(imageKey)) {
120 auto loaded = imageLoader.load(ref, caps);
121 if (!loaded) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(loaded.status());
122 out->impl_->imageLeases.emplace(imageKey, loaded.value().texture);
123 }
124 parameters.value().surface.textures[role].texture = out->impl_->imageLeases.at(imageKey);
125 }
126 image = parameters.value().surface.textures[0].texture;
127 out->impl_->draws.push_back({transforms.at(binding.object), out->impl_->meshLeases.at(meshKey), image,
128 std::move(parameters).takeValue()});
129 }
130 return Result<std::unique_ptr<EvpackStaticPrefab>>::success(std::move(out));
131}
132Result<void> EvpackStaticPrefab::draw(graphics::Graphics& gfx, const std::array<float, 16>& transform,
133 const std::array<float, 16>& cameraView) const {
134 if (impl_->released) return fail<void>("prefab resources have been released");
135 for (float value : transform)
136 if (!std::isfinite(value)) return fail<void>("nonfinite instance transform");
137 for (float value : cameraView)
138 if (!std::isfinite(value)) return fail<void>("nonfinite camera view");
139 const auto instance = glm::make_mat4(transform.data());
140 const auto view = glm::make_mat4(cameraView.data());
141 std::vector<std::size_t> order(impl_->draws.size());
142 std::iota(order.begin(), order.end(), 0);
143 std::stable_sort(order.begin(), order.end(), [&](std::size_t a, std::size_t b) {
144 const auto& left = impl_->draws[a];
145 const auto& right = impl_->draws[b];
146 if (left.material.transparent != right.material.transparent) return !left.material.transparent;
147 return left.material.transparent &&
148 (view * instance * left.transform)[3].z < (view * instance * right.transform)[3].z;
149 });
150 for (auto index : order) {
151 const auto& draw = impl_->draws[index];
152 gfx.setMesh3DSurface(draw.material.transparent ? graphics::SurfaceMode::Transparent
153 : draw.material.masked ? graphics::SurfaceMode::Masked
154 : graphics::SurfaceMode::Opaque,
155 draw.material.blend, !draw.material.transparent, draw.material.doubleSided,
156 draw.material.alphaCutoff, "cutoff");
157 auto bound = gfx.setMesh3DPbrSurface(&draw.material.surface);
158 if (!bound) return bound;
159 gfx.setMesh3DMaterial(draw.material.metallic, draw.material.roughness);
160 gfx.setMesh3DNormalTexture(nullptr);
161 gfx.setMesh3DHeightTexture(nullptr);
162 gfx.setMesh3DVirtualTexture(false, 1, 1, 1, 1, 0.f);
163 gfx.setMesh3DTexCellBomb(1.f, 0.f, 0.f);
164 gfx.setMesh3DParallax(0.f);
165 gfx.setMesh3DShadowReceive(true);
166 auto color = draw.material.color;
167 gfx.drawMesh(draw.mesh, instance * draw.transform, draw.image, color);
168 }
169 return gfx.setMesh3DPbrSurface(nullptr);
170}
