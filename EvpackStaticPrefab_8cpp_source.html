55Result<void> EvpackStaticPrefab::release() {
56 impl_->released = true;
57 if (impl_->gpuGraphics) {
58 for (const auto& material : impl_->gpuMaterials) {
59 auto released = impl_->gpuGraphics->gpuDrivenReleaseMaterialRecord(material.get());
60 if (!released) return released;
61 }
62 impl_->gpuGraphics = nullptr;
63 }
64 impl_->gpuMaterials.clear();
65 impl_->draws.clear();
66 Status failure;
67 for (auto it = impl_->meshLeases.begin(); it != impl_->meshLeases.end();) {
68 auto result = impl_->meshes.releaseMesh(it->second);
69 if (result)
70 it = impl_->meshLeases.erase(it);
71 else {
72 failure = result.status();
73 ++it;
74 }
75 }
76 for (auto it = impl_->imageLeases.begin(); it != impl_->imageLeases.end();) {
77 auto result = impl_->images.releaseImage(it->second);
78 if (result)
79 it = impl_->imageLeases.erase(it);
80 else {
81 failure = result.status();
82 ++it;
83 }
84 }
85 if (failure.isFailure()) return Result<void>::failure(failure);
86 return Result<void>::success();
87}
88Result<std::vector<StaticPrefabGpuPart>> EvpackStaticPrefab::prepareGpuDriven(graphics::Graphics& gfx) {
89 if (impl_->released) return fail<std::vector<StaticPrefabGpuPart>>("prefab resources have been released");
90 if (!impl_->gpuMaterials.empty())
91 return fail<std::vector<StaticPrefabGpuPart>>("prefab GPU-driven parts are already registered");
92 std::vector<std::unique_ptr<graphics::Material>> materials;
93 std::vector<StaticPrefabGpuPart> parts;
94 materials.reserve(impl_->draws.size());
95 parts.reserve(impl_->draws.size());
96 for (const auto& draw : impl_->draws) {
97 if (draw.material.transparent)
98 return fail<std::vector<StaticPrefabGpuPart>>("transparent prefab Detail cannot use opaque GPU submission");
99 bool unsupportedImage = false;
100 for (std::size_t role = 1; role < draw.material.images.size(); ++role) {
101 if (role != std::size_t(graphics::PbrTextureSlot::Normal) && draw.material.images[role])
102 unsupportedImage = true;
103 }
104 const auto conventionalBinding = [](const graphics::PbrTextureBinding& binding) {
105 return binding.texcoord == 0 && binding.offset == std::array<float, 2>{0.f, 0.f} &&
106 binding.scale == std::array<float, 2>{1.f, 1.f} && binding.rotation == 0.f;
107 };
108 if (unsupportedImage ||
109 std::any_of(draw.material.detailImages.begin(), draw.material.detailImages.end(),
110 [](const auto& image) { return image.has_value(); }) ||
111 draw.material.vegetationSurface || draw.material.surface.normalScale != 1.f ||
112 draw.material.surface.normalMode != graphics::PbrNormalMode::TangentXYZ ||
113 !conventionalBinding(draw.material.surface.textures[0]) ||
114 !conventionalBinding(draw.material.surface.textures[std::size_t(graphics::PbrTextureSlot::Normal)]))
115 return fail<std::vector<StaticPrefabGpuPart>>(
116 "extended prefab material requires a GPU-driven PBR surface record");
117 auto material = std::make_unique<graphics::Material>();
118 material->setTint(draw.material.color.r, draw.material.color.g, draw.material.color.b, draw.material.color.a);
119 material->setMetallic(draw.material.metallic);
120 material->setRoughness(draw.material.roughness);
121 material->setSurfaceMode(draw.material.masked ? "masked" : "opaque");
122 material->setAlphaCutoff(draw.material.alphaCutoff);
123 material->setDoubleSided(draw.material.doubleSided);
124 material->setAlbedoTexture(draw.image);
125 material->setNormalTexture(
126 draw.material.surface.textures[std::size_t(graphics::PbrTextureSlot::Normal)].texture);
127 if (!gfx.gpuDrivenMaterialUsable(material.get()))
128 return fail<std::vector<StaticPrefabGpuPart>>("prefab material is unavailable to GPU-driven rendering");
129 materials.push_back(std::move(material));
130 }
131 const auto releaseRegistered = [&gfx](const std::vector<std::unique_ptr<graphics::Material>>& owned,
132 std::size_t count) {
133 for (std::size_t i = 0; i < count; ++i) {
134 auto released = gfx.gpuDrivenReleaseMaterialRecord(owned[i].get());
135 if (!released) return released;
136 }
137 return Result<void>::success();
138 };
139 for (std::size_t drawIndex = 0; drawIndex < impl_->draws.size(); ++drawIndex) {
140 const auto& draw = impl_->draws[drawIndex];
141 auto& material = materials[drawIndex];
142 StaticPrefabGpuPart part;
143 std::copy(glm::value_ptr(draw.transform), glm::value_ptr(draw.transform) + 16, part.localTransform.begin());
144 part.meshId = gfx.gpuDrivenMeshRecord(draw.mesh);
145 part.materialId = gfx.gpuDrivenMaterialRecord(material.get());
146 if (part.meshId == graphics::kInvalidGpuDrivenSlot || part.materialId == graphics::kInvalidGpuDrivenSlot) {
147 auto cleanup = releaseRegistered(materials, drawIndex + 1);
148 if (!cleanup) return Result<std::vector<StaticPrefabGpuPart>>::failure(cleanup.status());
149 return fail<std::vector<StaticPrefabGpuPart>>("prefab GPU table registration failed");
150 }
151 parts.push_back(part);
152 }
153 if (parts.empty()) return fail<std::vector<StaticPrefabGpuPart>>("prefab has no enabled renderer");
154 impl_->gpuMaterials = std::move(materials);
155 impl_->gpuGraphics = &gfx;
156 return Result<std::vector<StaticPrefabGpuPart>>::success(std::move(parts));
157}
158Result<std::unique_ptr<EvpackStaticPrefab>> EvpackStaticPrefab::load(const asset::EvpackResourceReader& reader,
159 graphics::IMeshResourceFactory& meshes,
160 graphics::IImageResourceFactory& images,
161 const AssetRef& ref,
162 const asset::EvpackCapabilities& caps) {
163 auto scene = asset_scene::EvpackSceneTemplateLoader(reader).load(ref, caps);
164 if (!scene) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(scene.status());
165 auto out =
166 std::unique_ptr<EvpackStaticPrefab>(new EvpackStaticPrefab(std::make_unique<Impl>(Impl{meshes, images})));
167 std::map<SceneObjectId, glm::mat4> transforms;
168 std::function<void(const scene::NodeDesc&, glm::mat4)> walk = [&](const scene::NodeDesc& node, glm::mat4 parent) {
169 if (!node.visible) return;
170 auto local = glm::translate(glm::mat4(1), glm::vec3(node.x, node.y, node.z));
171 local = glm::rotate(local, glm::radians(node.yaw), glm::vec3(0, 1, 0));
172 local = glm::rotate(local, glm::radians(node.pitch), glm::vec3(1, 0, 0));
173 local = glm::rotate(local, glm::radians(node.roll), glm::vec3(0, 0, 1));
174 local = glm::scale(local, glm::vec3(node.sx, node.sy, node.sz));
175 auto world = parent * local;
176 transforms.emplace(node.persistentId, world);
177 for (const auto& child : node.children) walk(child, world);
178 };
179 walk(scene.value().root, glm::mat4(1));
180 EvpackGraphicsLoader meshLoader(reader, meshes);
181 EvpackImageLoader imageLoader(reader, images);
182 for (const auto& binding : scene.value().renderers) {
183 if (!binding.enabled || !transforms.contains(binding.object)) continue;
184 auto parameters = detail::readCookedMaterial(reader, binding.material, caps);
185 if (!parameters) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(parameters.status());
186 const auto meshKey = binding.mesh.format();
187 if (!out->impl_->meshLeases.contains(meshKey)) {
188 auto loaded = meshLoader.loadMesh(binding.mesh, caps, {}, 0, true);
189 if (!loaded) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(loaded.status());
190 out->impl_->meshLeases.emplace(meshKey, loaded.value().mesh);
191 out->impl_->meshUvSets.emplace(meshKey, loaded.value().availableTexcoordSets);
192 }
193 graphics::Texture* image = nullptr;
194 for (size_t role = 0; role < 11; role++) {
195 if (!parameters.value().images[role]) continue;
196 const auto& sets = out->impl_->meshUvSets.at(meshKey);
197 auto selected = parameters.value().surface.textures[role].texcoord;
198 if (std::find(sets.begin(), sets.end(), selected) == sets.end())
199 return fail<std::unique_ptr<EvpackStaticPrefab>>("material references missing mesh UV channel");
200 const auto& ref = *parameters.value().images[role];
201 const auto imageKey = ref.format();
202 if (!out->impl_->imageLeases.contains(imageKey)) {
203 auto loaded = imageLoader.load(ref, caps);
204 if (!loaded) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(loaded.status());
205 out->impl_->imageLeases.emplace(imageKey, loaded.value().texture);
206 }
207 parameters.value().surface.textures[role].texture = out->impl_->imageLeases.at(imageKey);
208 }
209 for (size_t role = 0; role < 3; ++role) {
210 if (!parameters.value().detailImages[role]) continue;
211 const auto& ref = *parameters.value().detailImages[role];
212 const auto imageKey = ref.format();
213 if (!out->impl_->imageLeases.contains(imageKey)) {
214 auto loaded = imageLoader.load(ref, caps);
215 if (!loaded) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(loaded.status());
216 out->impl_->imageLeases.emplace(imageKey, loaded.value().texture);
217 }
218 parameters.value().surface.vegetationDetail.textures[role].texture = out->impl_->imageLeases.at(imageKey);
219 }
220 auto validatedSurface = graphics::validatePbrSurface(parameters.value().surface);
221 if (!validatedSurface) return Result<std::unique_ptr<EvpackStaticPrefab>>::failure(validatedSurface.status());
222 image = parameters.value().surface.textures[0].texture;
223 out->impl_->draws.push_back({transforms.at(binding.object), out->impl_->meshLeases.at(meshKey), image,
224 std::move(parameters).takeValue(), binding.castShadows,
225 binding.receiveShadows});
226 }
227 return Result<std::unique_ptr<EvpackStaticPrefab>>::success(std::move(out));
228}
229Result<void> EvpackStaticPrefab::draw(graphics::Graphics& gfx, const std::array<float, 16>& transform,
230 const std::array<float, 16>& cameraView) const {
231 if (impl_->released) return fail<void>("prefab resources have been released");
232 for (float value : transform)
233 if (!std::isfinite(value)) return fail<void>("nonfinite instance transform");
234 for (float value : cameraView)
235 if (!std::isfinite(value)) return fail<void>("nonfinite camera view");
236 const auto instance = glm::make_mat4(transform.data());
237 const auto view = glm::make_mat4(cameraView.data());
238 std::vector<std::size_t> order(impl_->draws.size());
239 std::iota(order.begin(), order.end(), 0);
240 std::stable_sort(order.begin(), order.end(), [&](std::size_t a, std::size_t b) {
241 const auto& left = impl_->draws[a];
242 const auto& right = impl_->draws[b];
243 if (left.material.transparent != right.material.transparent) return !left.material.transparent;
244 return left.material.transparent &&
245 (view * instance * left.transform)[3].z < (view * instance * right.transform)[3].z;
246 });
247 for (auto index : order) {
248 const auto& draw = impl_->draws[index];
249 gfx.setMesh3DSurface(draw.material.transparent ? graphics::SurfaceMode::Transparent
250 : draw.material.masked ? graphics::SurfaceMode::Masked
251 : graphics::SurfaceMode::Opaque,
252 draw.material.blend, !draw.material.transparent, draw.material.doubleSided,
253 draw.material.alphaCutoff, "cutoff");
254 auto bound = gfx.setMesh3DPbrSurface(&draw.material.surface);
255 if (!bound) return bound;
256 gfx.setMesh3DMaterial(draw.material.metallic, draw.material.roughness);
257 gfx.setMesh3DNormalTexture(nullptr);
258 gfx.setMesh3DHeightTexture(nullptr);
259 gfx.setMesh3DVirtualTexture(false, 1, 1, 1, 1, 0.f);
260 gfx.setMesh3DTexCellBomb(1.f, 0.f, 0.f);
261 gfx.setMesh3DParallax(0.f);
262 gfx.setMesh3DShadowReceive(draw.receiveShadows);
263 auto color = draw.material.color;
264 gfx.drawMesh(draw.mesh, instance * draw.transform, draw.image, color);
265 }
266 return gfx.setMesh3DPbrSurface(nullptr);
267}

269Result<void> EvpackStaticPrefab::drawShadow(graphics::Graphics& gfx,
270 const std::array<float, 16>& transform,
271 const std::array<float, 16>& lightViewProjection) const {
272 if (impl_->released) return fail<void>("prefab resources have been released");
273 for (float value : transform)
274 if (!std::isfinite(value)) return fail<void>("nonfinite instance transform");
275 for (float value : lightViewProjection)
276 if (!std::isfinite(value)) return fail<void>("nonfinite shadow view projection");
277 const glm::mat4 instance = glm::make_mat4(transform.data());
278 const glm::mat4 lightVP = glm::make_mat4(lightViewProjection.data());
279 for (const auto& draw : impl_->draws) {
280 if (!draw.castShadows || draw.material.transparent) continue;
281 const glm::mat4 lightMvp = lightVP * instance * draw.transform;
282 const bool doubleSided = draw.material.surface.cullMode == graphics::PbrCullMode::None ||
283 (draw.material.surface.cullMode == graphics::PbrCullMode::Inherit &&
284 draw.material.doubleSided);
285 if (draw.material.masked)
286 gfx.drawMeshShadowAlpha(draw.mesh, lightMvp, draw.image, doubleSided);
287 else
288 gfx.drawMeshShadow(draw.mesh, lightMvp, doubleSided);
289 }
290 return Result<void>::success();
291}
