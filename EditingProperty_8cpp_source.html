65template <class ResultValue>
66Result<ResultValue> validationError(const eve::Result<void>& validation, const PropertyDescriptor& descriptor) {
67 std::string rule = "editor.property.invalid-value";
68 std::string message = "Property value was rejected";
69 if (const eve::Diagnostic* error = validation.error()) {
70 rule = legacyRule(std::string(property_access::diagnosticRule(*error)));
71 message = error->message();
72 }
73 return failed<ResultValue>(Status::Rejected, eve::DiagnosticCode::PreconditionViolation, RuleId(std::move(rule)),
74 message + ": " + descriptor.path.value());
75}

98eve::Value toPresentationValue(const Value& value) {
99 return std::visit(
100 [](const auto& current) -> eve::Value {
101 using T = std::decay_t<decltype(current)>;
102 if constexpr (std::is_same_v<T, std::monostate>) {
103 return {};
104 } else if constexpr (std::is_same_v<T, Value::Array>) {
105 eve::Value::Array result;
106 result.reserve(current.size());
107 for (const Value& entry : current) result.push_back(toPresentationValue(entry));
108 return result;
109 } else if constexpr (std::is_same_v<T, Value::Object>) {
110 eve::Value::Object result;
111 for (const auto& [key, entry] : current) result.emplace(key, toPresentationValue(entry));
112 return result;
113 } else {
114 return eve::Value(current);
115 }
116 },
117 value.storage());
118}

142Result<void> validatePropertyValue(const PropertyDescriptor& descriptor, const Value& value) {
143 eve::Result<void> validation =
144 property_access::validatePropertyValue(toPresentationDescriptor(descriptor), toPresentationValue(value));
145 if (!validation.ok()) return validationError<void>(validation, descriptor);
146 if (descriptor.type == PropertyType::Action && value.type() != Value::Type::Null)
147 return eve::editing::failed<void>(Status::Rejected, RuleId("editor.property.type-mismatch"),
148 "Value type does not match property: " + descriptor.path.value());
149 return eve::editing::applied<void>();
150}
