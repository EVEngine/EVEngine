94eve::Value toPresentationValue(const Value& value) {
95 return std::visit(
96 [](const auto& current) -> eve::Value {
97 using T = std::decay_t<decltype(current)>;
98 if constexpr (std::is_same_v<T, std::monostate>) {
99 return {};
100 } else if constexpr (std::is_same_v<T, Value::Array>) {
101 eve::Value::Array result;
102 result.reserve(current.size());
103 for (const Value& entry : current) result.push_back(toPresentationValue(entry));
104 return result;
105 } else if constexpr (std::is_same_v<T, Value::Object>) {
106 eve::Value::Object result;
107 for (const auto& [key, entry] : current) result.emplace(key, toPresentationValue(entry));
108 return result;
109 } else {
110 return eve::Value(current);
111 }
112 },
113 value.storage());
114}

138Result<void> validatePropertyValue(const PropertyDescriptor& descriptor, const Value& value) {
139 const property_access::WriteResult validation =
140 property_access::validatePropertyValue(toPresentationDescriptor(descriptor), toPresentationValue(value));
141 if (!validation.accepted) return validationError<void>(validation, descriptor);
142 if (descriptor.type == PropertyType::Action && value.type() != Value::Type::Null)
143 return eve::editing::failed<void>(Status::Rejected, RuleId("editor.property.type-mismatch"),
144 "Value type does not match property: " + descriptor.path.value());
145 return eve::editing::applied<void>();
146}
