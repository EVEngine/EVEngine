39bool startKindMatches(ClimbingAnchorKind nodeKind, ClimbingActionKind actionKind) {
40 if (nodeKind == ClimbingAnchorKind::LadderRung)
41 return actionKind == ClimbingActionKind::LadderMount || actionKind == ClimbingActionKind::LedgeGrab;
42 if (nodeKind == ClimbingAnchorKind::Beam) return actionKind == ClimbingActionKind::BeamBalance;
43 if (nodeKind == ClimbingAnchorKind::Pole) return actionKind == ClimbingActionKind::PoleSwing;
44 if (nodeKind == ClimbingAnchorKind::Bar) return actionKind == ClimbingActionKind::BarSwing;
45 return actionKind == ClimbingActionKind::LedgeGrab || actionKind == ClimbingActionKind::ClimbDown;
46}

48bool edgeKindMatches(ClimbingAnchorEdgeKind edgeKind, ClimbingActionKind actionKind) {
49 switch (edgeKind) {
50 case ClimbingAnchorEdgeKind::Shimmy: return actionKind == ClimbingActionKind::Shimmy;
51 case ClimbingAnchorEdgeKind::Corner:
52 return actionKind == ClimbingActionKind::CornerInner || actionKind == ClimbingActionKind::CornerOuter;
53 case ClimbingAnchorEdgeKind::Jump: return actionKind == ClimbingActionKind::LedgeJump;
54 case ClimbingAnchorEdgeKind::Drop: return actionKind == ClimbingActionKind::ClimbDown;
55 case ClimbingAnchorEdgeKind::Mount: return actionKind == ClimbingActionKind::LadderMount;
56 case ClimbingAnchorEdgeKind::Dismount: return actionKind == ClimbingActionKind::LadderDismount;
57 case ClimbingAnchorEdgeKind::Climb: return actionKind == ClimbingActionKind::LadderClimb;
58 case ClimbingAnchorEdgeKind::Balance: return actionKind == ClimbingActionKind::BeamBalance;
59 case ClimbingAnchorEdgeKind::Swing:
60 return actionKind == ClimbingActionKind::PoleSwing || actionKind == ClimbingActionKind::BarSwing;
61 }
62 return false;
63}

79eve::Result<void> validatePath(physics::World3D& world, Vec3 start, const ClimbingCandidate& candidate,
80 const ClimbingActionDefinition& action, const ClimbingProfileDefinition& profile,
81 std::uint32_t& queryCount) {
82 physics::QueryFilter3D filter = profile.queryFilter;
83 filter.ignoredBodyId = candidate.ignoredBodyId;
84 Vec3 current = start;
85 for (std::uint32_t segment = 1; segment <= profile.pathValidationSegments; ++segment) {
86 const float t = static_cast<float>(segment) / static_cast<float>(profile.pathValidationSegments);
87 const Vec3 target = detail::trajectoryPoint(start, candidate, action, profile, t);
88 const Vec3 desired = subtract(target, current);
89 const float lowerY = current.y + profile.capsuleRadius;
90 const float upperY = current.y + profile.capsuleHeight - profile.capsuleRadius;
91 auto moved = world.moveCapsuleOwned(current.x, lowerY, current.z, current.x, upperY, current.z,
92 profile.capsuleRadius, desired.x, desired.y, desired.z, filter);
93 ++queryCount;
94 if (!moved) return eve::Result<void>::failure(moved.status());
95 const Vec3 actual{moved.value().deltaX, moved.value().deltaY, moved.value().deltaZ};
96 if (length(subtract(desired, actual)) > profile.skin + 0.001f)
97 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
98 "climbing.anchor.path_blocked", "candidate.path." + std::to_string(segment));
99 current = target;
100 }
101 return eve::Result<void>::success();
102}

104eve::Result<ClimbingCandidate> makeCandidate(physics::World3D& world, ClimbingAnchorGraphInstance& graph,
105 const ResolvedClimbingAnchorNode& node,
106 const ClimbingActionDefinition& action,
107 const ClimbingProfileDefinition& profile,
108 const ClimbingPose& pose, std::uint64_t definitionGeneration) {
109 physics::Body3D* body = world.findBody(graph.body());
110 if (!body)
111 return failure<ClimbingCandidate>(eve::DiagnosticCode::StaleHandle,
112 "anchor graph target body is stale", "graph.body");
113 ClimbingCandidate candidate;
114 candidate.actionId = action.id;
115 candidate.definitionGeneration = definitionGeneration;
116 candidate.world = world.runtimeHandle();
117 candidate.obstacleBody = graph.body();
118 candidate.obstacleBodyId = body->getId();
119 candidate.ignoredBodyId = pose.ignoredBodyId;
120 candidate.frontPoint = node.position;
121 candidate.topPoint = node.position;
122 candidate.surfaceNormal = node.normal;
123 candidate.surfaceTangent = node.tangent;
124 candidate.leftHandAnchor = node.leftHandSocket;
125 candidate.rightHandAnchor = node.rightHandSocket;
126 candidate.landingFeet = action.kind == ClimbingActionKind::LadderDismount ||
127 action.kind == ClimbingActionKind::BeamBalance
128 ? node.position
129 : hangingFeet(node, action, profile);
130 candidate.obstacleHeight = length(subtract(candidate.landingFeet, pose.feet));
131 candidate.kind = action.kind;
132 candidate.support = action.kind == ClimbingActionKind::LadderDismount ||
133 action.kind == ClimbingActionKind::BeamBalance
134 ? HangSupport::None
135 : (node.kind == ClimbingAnchorKind::LadderRung ? HangSupport::Braced : HangSupport::Free);
136 auto localTop = body->worldToLocalPointOwned(node.position.x, node.position.y, node.position.z);
137 if (!localTop) return eve::Result<ClimbingCandidate>::failure(localTop.status());
138 auto localLanding = body->worldToLocalPointOwned(candidate.landingFeet.x, candidate.landingFeet.y,
139 candidate.landingFeet.z);
140 if (!localLanding) return eve::Result<ClimbingCandidate>::failure(localLanding.status());
141 candidate.bodyLocalTop = {localTop.value().x, localTop.value().y, localTop.value().z};
142 candidate.bodyLocalLanding = {localLanding.value().x, localLanding.value().y, localLanding.value().z};
143 return eve::Result<ClimbingCandidate>::success(std::move(candidate));
144}

153eve::Result<void> ClimbingRuntime::releaseAnchorReservation() {
154 if (!execution_ || execution_->anchorReservation.id.isZero())
155 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
156 auto graph = Climbing::resolveAnchorGraph(execution_->anchorGraph);
157 if (!graph.isBound()) {
158 execution_->anchorReservation = {};
159 execution_->anchorGraph = {};
160 execution_->anchorNode = {};
161 return failure<void>(eve::DiagnosticCode::StaleHandle,
162 "anchor graph instance is stale while releasing occupancy", "execution.anchorGraph");
163 }
164 auto released = graph->release(execution_->anchorReservation);
165 if (released || (released.error() &&
166 (released.error()->code() == eve::DiagnosticCode::StaleHandle ||
167 released.error()->code() == eve::DiagnosticCode::NotFound ||
168 released.error()->code() == eve::DiagnosticCode::Conflict))) {
169 execution_->anchorReservation = {};
170 execution_->anchorGraph = {};
171 execution_->anchorNode = {};
172 }
173 if (!released && released.error() && released.error()->code() == eve::DiagnosticCode::Conflict)
174 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
175 return released;
176}

185eve::Result<ClimbingCandidate> ClimbingRuntime::tryBeginAnchor(
186 physics::World3D& world, ClimbingAnchorGraphHandleRef graphReference, const ClimbingAnchorNodeRef& nodeReference,
187 ClimbingAnchorAgentId agentId, std::string_view actionId, const ClimbingPose& pose, eve::SimulationTick tick) {
188 if (activePhase(phase_) || execution_)
189 return failure<ClimbingCandidate>(eve::DiagnosticCode::Conflict,
190 "a climbing execution is already active", "runtime.phase");
191 if (nextExecutionId_ == 0 || nextExecutionId_ == std::numeric_limits<std::uint64_t>::max())
192 return failure<ClimbingCandidate>(eve::DiagnosticCode::PreconditionViolation,
193 "climbing execution id space is exhausted", "runtime.nextExecutionId");
194 if (agentId.isZero())
195 return failure<ClimbingCandidate>(eve::DiagnosticCode::InvalidArgument,
196 "anchor execution requires a stable non-zero agent id", "agentId");
197 auto graph = Climbing::resolveAnchorGraph(graphReference);
198 if (!graph.isBound())
199 return failure<ClimbingCandidate>(eve::DiagnosticCode::StaleHandle,
200 "anchor graph instance handle is stale", "graph");
201 auto resolved = graph->resolveNode(world, nodeReference);
202 if (!resolved) return eve::Result<ClimbingCandidate>::failure(resolved.status());
203 const ClimbingActionDefinition* action = findAction(profile_, actionId);
204 if (!action || !startKindMatches(resolved.value().kind, action->kind))
205 return failure<ClimbingCandidate>(eve::DiagnosticCode::InvalidArgument,
206 "action kind cannot approach the requested anchor node", "actionId");
207 if (!action->requiredNotifies.empty() && !validatedAnimationActions_.contains(action->id))
208 return failure<ClimbingCandidate>(eve::DiagnosticCode::PreconditionViolation,
209 "climbing.animation.notify_missing: action clip contract was not validated",
210 "actionId");
211 auto candidate = makeCandidate(world, *graph, resolved.value(), *action, profile_, pose, definitionGeneration_);
212 if (!candidate) return candidate;
213 if (candidate.value().obstacleHeight + epsilon < action->minHeight ||
214 candidate.value().obstacleHeight - epsilon > action->maxHeight)
215 return failure<ClimbingCandidate>(eve::DiagnosticCode::PreconditionViolation,
216 "anchor approach lies outside the action geometry range", "action.height");
217 auto path = validatePath(world, pose.feet, candidate.value(), *action, profile_, lastQueryCount_);
218 if (!path) return eve::Result<ClimbingCandidate>::failure(path.status());
219 auto eventCapacity = requireEventCapacity(1, tick);
220 if (!eventCapacity) return eve::Result<ClimbingCandidate>::failure(eventCapacity.status());
221 const ClimbingExecutionId executionId(nextExecutionId_);
222 auto reservation = graph->reserve(nodeReference, {agentId, executionId});
223 if (!reservation) return eve::Result<ClimbingCandidate>::failure(reservation.status());
224
225 PreparedBegin prepared;
226 prepared.executionId = executionId;
227 prepared.candidate = candidate.value();
228 prepared.action = *action;
229 prepared.startFeet = pose.feet;
230 prepared.tick = tick;
231 prepared.conditionsSatisfied = action->requiredConditionTags.empty();
232 auto committed = commitBegin(std::move(prepared));
233 if (!committed) {
234 auto rollback = graph->release(reservation.value());
235 rollback.ignore("rollback graph reservation after climbing begin commit failure");
236 return eve::Result<ClimbingCandidate>::failure(committed.status());
237 }
238 execution_->anchorGraph = graphReference;
239 execution_->anchorNode = nodeReference;
240 execution_->anchorReservation = std::move(reservation).takeValue();
241 return eve::Result<ClimbingCandidate>::success(std::move(committed).takeValue().candidate,
242 eve::Status::success(eve::StatusCode::Applied));
243}

245eve::Result<void> ClimbingRuntime::transitionAnchor(physics::World3D& world,
246 const ClimbingAnchorNodeRef& target,
247 ClimbingAnchorEdgeKind edgeKind,
248 std::string_view actionId,
249 eve::SimulationTick tick) {
250 if ((phase_ != ClimbingPhase::Hanging && phase_ != ClimbingPhase::Balanced &&
251 phase_ != ClimbingPhase::Swinging) || !execution_ || !execution_->anchorGraph.isValid() ||
252 execution_->anchorReservation.id.isZero())
253 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
254 "anchor transition requires a graph-bound hanging execution", "runtime.phase");
255 if (!execution_->branchWindowOpen)
256 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
257 "anchor transition branch window is closed", "execution.branchWindow");
258 if (tick <= execution_->lastTick)
259 return failure<void>(eve::DiagnosticCode::Conflict,
260 "anchor transition tick must be newer than the last execution tick", "tick");
261 auto graph = Climbing::resolveAnchorGraph(execution_->anchorGraph);
262 if (!graph.isBound())
263 return failure<void>(eve::DiagnosticCode::StaleHandle,
264 "anchor graph instance handle is stale", "execution.anchorGraph");
265 auto reservationValid = graph->validateReservation(execution_->anchorReservation);
266 if (!reservationValid) return reservationValid;
267 auto edges = graph->edgesFrom(execution_->anchorNode);
268 if (!edges) return eve::Result<void>::failure(edges.status());
269 const auto edge = std::find_if(edges.value().begin(), edges.value().end(), [&](const auto& candidate) {
270 return candidate.to == target.nodeId && candidate.kind == edgeKind;
271 });
272 if (edge == edges.value().end())
273 return failure<void>(eve::DiagnosticCode::NotFound,
274 "no authored edge connects the current and requested anchors", "target");
275 auto resolved = graph->resolveNode(world, target);
276 if (!resolved) return eve::Result<void>::failure(resolved.status());
277 if (!tagsContain(resolved.value().tags, edge->requiredTags))
278 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
279 "target anchor does not satisfy the edge tag contract", "target.tags");
280 const ClimbingActionDefinition* action = findAction(profile_, actionId);
281 if (!action || !edgeKindMatches(edgeKind, action->kind))
282 return failure<void>(eve::DiagnosticCode::InvalidArgument,
283 "action kind does not match the authored anchor edge", "actionId");
284 if (!action->requiredNotifies.empty() && !validatedAnimationActions_.contains(action->id))
285 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
286 "climbing.animation.notify_missing: action clip contract was not validated", "actionId");
287 ClimbingPose pose;
288 pose.feet = execution_->currentFeet;
289 pose.forward = scale(execution_->candidate.surfaceNormal, -1.f);
290 pose.ignoredBodyId = execution_->candidate.ignoredBodyId;
291 pose.grounded = false;
292 auto candidate = makeCandidate(world, *graph, resolved.value(), *action, profile_, pose, definitionGeneration_);
293 if (!candidate) return eve::Result<void>::failure(candidate.status());
294 const float span = length(subtract(candidate.value().landingFeet, execution_->currentFeet));
295 candidate.value().obstacleHeight = span;
296 if (span + epsilon < action->minHeight || span - epsilon > action->maxHeight)
297 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
298 "anchor transition lies outside the action geometry range", "action.height");
299 auto path = validatePath(world, execution_->currentFeet, candidate.value(), *action, profile_, lastQueryCount_);
300 if (!path) return eve::Result<void>::failure(path.status());
301 auto eventCapacity = requireEventCapacity(1, tick);
302 if (!eventCapacity) return eventCapacity;
303 auto nextReservation = graph->reserve(target, execution_->anchorReservation.occupant);
304 if (!nextReservation) return eve::Result<void>::failure(nextReservation.status());
305 auto released = graph->release(execution_->anchorReservation);
306 if (!released) {
307 auto rollback = graph->release(nextReservation.value());
308 rollback.ignore("rollback target reservation after source release failure");
309 return released;
310 }
311
312 execution_->candidate = std::move(candidate).takeValue();
313 execution_->action = *action;
314 execution_->definitionGeneration = definitionGeneration_;
315 execution_->startFeet = execution_->currentFeet;
316 execution_->lastPlannedFeet = execution_->currentFeet;
317 execution_->elapsed = eve::Duration::zero();
318 execution_->duration = action->duration;
319 execution_->lastTick = tick;
320 execution_->accumulatedResidual = {};
321 execution_->horizontalWarpUsed = 0.f;
322 execution_->verticalWarpUsed = 0.f;
323 execution_->facingWarpUsed = 0.f;
324 execution_->leftContactEmitted = false;
325 execution_->rightContactEmitted = false;
326 execution_->landContactReleased = false;
327 execution_->compactCollisionActive = false;
328 execution_->branchWindowOpen = false;
329 execution_->anchorNode = target;
330 execution_->anchorReservation = std::move(nextReservation).takeValue();
331 phase_ = ClimbingPhase::Launching;
332 terminalCode_.clear();
333 enqueueEvent({ClimbingEventKind::AnchorTransitionStarted, execution_->candidate.actionId, tick,
334 execution_->executionId});
335 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
336}
