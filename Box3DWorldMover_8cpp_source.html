13void WorldBackend::do_setMoverUp(float x, float y, float z) {
14 if (!std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z))
15 throw eve::Exception("World3D.setMoverUp: values must be finite");
16 const float lengthSquared = x * x + y * y + z * z;
17 if (lengthSquared <= 1e-12f) throw eve::Exception("World3D.setMoverUp: direction must be non-zero");
18 const float inverseLength = 1.f / std::sqrt(lengthSquared);
19 moverUp_ = b3Vec3{x * inverseLength, y * inverseLength, z * inverseLength};
20}

55 static bool callback(b3ShapeId shapeId, const b3PlaneResult *results, int resultCount, void *context) {
56 auto *self = static_cast<PlaneCollector *>(context);
57 if (self->world->do_shouldIgnoreQueryShape(shapeId)) return true;
58 for (int i = 0; i < resultCount && self->count < planeCapacity; ++i) {
59 const b3Plane &plane = results[i].plane;
60 self->planes[self->count++] =
61 b3CollisionPlane{.plane = plane, .pushLimit = FLT_MAX, .push = 0.f, .clipVelocity = true};
62 const float d = b3Dot(plane.normal, self->desired);
63 if (d < self->bestDot) {
64 self->bestDot = d;
65 self->bestNormal = plane.normal;
66 }
67 self->maxUpDot = std::max(self->maxUpDot, b3Dot(plane.normal, self->up));
68 }
69 return true;
70 }
71 } collector;
72 collector.world = this;
73 collector.up = moverUp_;

75 const b3Pos start{ax, ay, az};
76 b3Pos current = start;
77 const b3Pos target{ax + dx, ay + dy, az + dz};
78 b3Capsule mover{b3Vec3_zero, b3Vec3{bx - ax, by - ay, bz - az}, radius};
79 b3QueryFilter filter = do_makeQueryFilter();
80 constexpr float toleranceSquared = 1e-8f;
81 bool collided = false;
82
83 for (int pass = 0; pass < 5; ++pass) {
84 collector.count = 0;
85 collector.desired = target - current;
86 b3World_CollideMover(worldId_, current, &mover, filter, &PlaneCollector::callback, &collector);
87 owner_->moverPlaneCount_ += collector.count;
88 collided = collided || collector.count > 0;
89
90 const b3Vec3 targetDelta = target - current;
91 b3PlaneSolverResult solved = b3SolvePlanes(targetDelta, collector.planes, collector.count);
92 b3Vec3 delta = solved.delta;
93 struct MoverFilter {
94 WorldBackend *world = nullptr;
95 static bool callback(b3ShapeId shapeId, void *context) {
96 auto *self = static_cast<MoverFilter *>(context);
97 return !self->world->do_shouldIgnoreQueryShape(shapeId);
98 }
99 } moverFilter{this};
100 const float fraction =
101 b3World_CastMover(worldId_, current, &mover, delta, filter, &MoverFilter::callback, &moverFilter);
102 if (fraction < 1.f) collided = true;
103 delta = fraction * delta;
104 current = current + delta;
105 owner_->moverIterations_ = pass + 1;
106
107 if (b3LengthSquared(delta) <= toleranceSquared) break;
108 if (b3LengthSquared(target - current) <= toleranceSquared) break;
109 }
110
111 const b3Vec3 total = current - start;
112 owner_->moverDeltaX_ = total.x;
113 owner_->moverDeltaY_ = total.y;
114 owner_->moverDeltaZ_ = total.z;
115 owner_->moverNormalX_ = collector.bestNormal.x;
116 owner_->moverNormalY_ = collector.bestNormal.y;
117 owner_->moverNormalZ_ = collector.bestNormal.z;
118 owner_->moverGroundDot_ = collector.maxUpDot;
119 owner_->moverGrounded_ = owner_->moverGroundDot_ >= owner_->moverSlopeCos_;
120 return collided;
121}

146 const float inverseUpLength = 1.f / std::sqrt(upLengthSquared);
147 moverUp_ = {policy.upX * inverseUpLength, policy.upY * inverseUpLength, policy.upZ * inverseUpLength};

158 result.normalZ = owner_->getMoverNormalZ();
159 result.planeCount = owner_->getMoverPlaneCount();
160 result.iterations = owner_->getMoverIterations();
161 moverUp_ = previousUp;

177ClothProbeStatus WorldBackend::do_pointProbeFiltered(float x, float y, float z, float radius, ClothContact3D *out,
178 uint64_t categoryBits, uint64_t maskBits) const {
179 if (out) *out = ClothContact3D{};
180 if (!isValid() || radius <= 0.f || !out) return ClothProbeStatus::Miss;
181
182 const b3Vec3 target{x, y, z};
183 for (Shape3D *s : owner_->shapes_) {
184 if (!s || !s->isValid() || s->isSensor()) continue;
185 if ((maskBits & s->getCategoryBits()) == 0 || (s->getMaskBits() & categoryBits) == 0) continue;
186 // Box3D closest-point proxies are convex-only. Concave surfaces are handled by
187 // swept queries (and rigid contact generation), not b3Shape_GetClosestPoint.
188 const std::string kind = s->getKind();
189 if (kind == "triangleMesh" || kind == "heightField") continue;
190 const b3Vec3 closest = b3Shape_GetClosestPoint(box3d::Access::shapeId(*s), target);
191 const b3Vec3 delta = target - closest;
192 const float d = std::sqrt(delta.x * delta.x + delta.y * delta.y + delta.z * delta.z);
193 const float depth = radius - d;
194 if (depth > 0.f && depth > out->depth) {
195 out->hit = true;
196 out->depth = depth;
197 if (d > 1e-6f) {
198 out->nx = delta.x / d;
199 out->ny = delta.y / d;
200 out->nz = delta.z / d;
201 } else {
202 out->nx = 0.f;
203 out->ny = 1.f;
204 out->nz = 0.f;
205 }
206 out->body = s->getBody();
207 out->friction = s->getFriction();
208 out->restitution = s->getRestitution();
209 }
210 }
211 return out->hit ? ClothProbeStatus::Hit : ClothProbeStatus::Miss;
212}
