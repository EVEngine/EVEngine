65std::uint32_t crcOf(std::span<const std::uint8_t> bytes) {

69 checksum = crc32(checksum, bytes.data(), count);

75std::string sha256Text(std::span<const std::uint8_t> bytes) {
76 data::HashFunction::Value digest{};
77 data::HashFunction::getHashFunction("sha256")
78 ->hash("sha256", reinterpret_cast<const char*>(bytes.data()), bytes.size(), digest);
79 static constexpr char digits[] = "0123456789abcdef";
80 std::string result = "sha256:";
81 result.reserve(71);
82 for (std::size_t index = 0; index < 32; ++index) {
83 const auto byte = static_cast<unsigned char>(digest.data[index]);
84 result.push_back(digits[byte >> 4]);
85 result.push_back(digits[byte & 0x0f]);
86 }
87 return result;
88}

90bool validPath(std::string_view path, const EvaArchiveLimits& limits) {
91 if (path.empty() || path.size() > limits.maximumPathBytes ||
92 path.size() > std::numeric_limits<std::uint16_t>::max() || path.front() == '/' || path.back() == '/')
93 return false;
94 std::size_t segmentStart = 0;
95 for (std::size_t index = 0; index <= path.size(); ++index) {
96 if (index < path.size()) {
97 const unsigned char byte = static_cast<unsigned char>(path[index]);
98 if (byte == '\\' || byte == 0 || byte < 0x20 || byte == 0x7f) return false;
99 if (byte != '/') continue;
100 }
101 const std::string_view segment = path.substr(segmentStart, index - segmentStart);
102 if (segment.empty() || segment == "." || segment == "..") return false;
103 segmentStart = index + 1;
104 }
105 for (std::size_t index = 0; index < path.size();) {
106 const unsigned char lead = static_cast<unsigned char>(path[index]);
107 if (lead < 0x80) { ++index; continue; }
108 std::size_t continuation = 0;
109 std::uint32_t codepoint = 0;
110 if ((lead & 0xe0) == 0xc0) { continuation = 1; codepoint = lead & 0x1f; }
111 else if ((lead & 0xf0) == 0xe0) { continuation = 2; codepoint = lead & 0x0f; }
112 else if ((lead & 0xf8) == 0xf0) { continuation = 3; codepoint = lead & 0x07; }
113 else return false;
114 if (continuation >= path.size() - index) return false;
115 for (std::size_t part = 1; part <= continuation; ++part) {
116 const unsigned char byte = static_cast<unsigned char>(path[index + part]);
117 if ((byte & 0xc0) != 0x80) return false;
118 codepoint = (codepoint << 6) | (byte & 0x3f);
119 }
120 if ((continuation == 1 && codepoint < 0x80) || (continuation == 2 && codepoint < 0x800) ||
121 (continuation == 3 && codepoint < 0x10000) || codepoint > 0x10ffff ||
122 (codepoint >= 0xd800 && codepoint <= 0xdfff)) return false;
123 index += continuation + 1;
124 }
125 utf8proc_uint8_t* normalized = nullptr;
126 const auto normalizedSize = utf8proc_map(
127 reinterpret_cast<const utf8proc_uint8_t*>(path.data()),
128 static_cast<utf8proc_ssize_t>(path.size()), &normalized,
129 static_cast<utf8proc_option_t>(UTF8PROC_STABLE | UTF8PROC_COMPOSE));
130 if (normalizedSize < 0 || !normalized) return false;
131 const bool canonical = static_cast<std::size_t>(normalizedSize) == path.size() &&
132 std::equal(path.begin(), path.end(),
133 reinterpret_cast<const char*>(normalized));
134 std::free(normalized);
135 return canonical;
136}

149}

162std::vector<std::uint8_t> deflateRaw(std::span<const std::uint8_t> input) {
163 if (input.empty() || input.size() > std::numeric_limits<uInt>::max()) return {};
164 z_stream stream{};
165 if (deflateInit2(&stream, Z_BEST_COMPRESSION, Z_DEFLATED, -MAX_WBITS, 8,
166 Z_DEFAULT_STRATEGY) != Z_OK)
167 return {};
168 const uLong bound = deflateBound(&stream, static_cast<uLong>(input.size()));
169 if (bound > std::numeric_limits<uInt>::max()) {
170 deflateEnd(&stream);
171 return {};
172 }
173 std::vector<std::uint8_t> output(static_cast<std::size_t>(bound));
174 stream.next_in = const_cast<Bytef*>(input.data());
175 stream.avail_in = static_cast<uInt>(input.size());
176 stream.next_out = output.data();
177 stream.avail_out = static_cast<uInt>(output.size());
178 const int status = deflate(&stream, Z_FINISH);
179 const std::size_t size = stream.total_out;
180 deflateEnd(&stream);
181 if (status != Z_STREAM_END || size >= input.size()) return {};
182 output.resize(size);
183 return output;
184}

186Result<std::vector<std::uint8_t>> inflateRaw(std::span<const std::uint8_t> input,
187 std::uint64_t decodedSize) {
188 if (input.size() > std::numeric_limits<uInt>::max() ||
189 decodedSize > std::numeric_limits<uInt>::max())
190 return archiveFailure<std::vector<std::uint8_t>>(
191 DiagnosticCode::InvalidArgument, "Deflate entry exceeds decoder limits");
192 std::vector<std::uint8_t> output(static_cast<std::size_t>(decodedSize));
193 std::uint8_t emptyOutput = 0;
194 z_stream stream{};
195 stream.next_in = const_cast<Bytef*>(input.data());
196 stream.avail_in = static_cast<uInt>(input.size());
197 stream.next_out = output.empty() ? &emptyOutput : output.data();
198 stream.avail_out = output.empty() ? 1u : static_cast<uInt>(output.size());
199 if (inflateInit2(&stream, -MAX_WBITS) != Z_OK)
200 return archiveFailure<std::vector<std::uint8_t>>(DiagnosticCode::Failed,
201 "Deflate decoder initialization failed");
202 const int status = inflate(&stream, Z_FINISH);
203 const bool exact = status == Z_STREAM_END && stream.total_in == input.size() &&
204 stream.total_out == decodedSize;
205 inflateEnd(&stream);
206 if (!exact)
207 return archiveFailure<std::vector<std::uint8_t>>(DiagnosticCode::ParseError,
208 "Deflate payload is malformed");
209 return Result<std::vector<std::uint8_t>>::success(std::move(output));
210}

253Result<void> validateContentAddressedBlobs(const std::vector<EvaArchiveEntry>& entries) {
254 constexpr std::string_view prefix = "blobs/sha256/";
255 for (const auto& entry : entries) {
256 if (!entry.path.starts_with(prefix)) continue;
257 const std::string_view digest = std::string_view(entry.path).substr(prefix.size());
258 if (digest.size() != 64 ||
259 !std::all_of(digest.begin(), digest.end(), [](unsigned char character) {
260 return (character >= '0' && character <= '9') ||
261 (character >= 'a' && character <= 'f');
262 }) || sha256Text(entry.bytes) != "sha256:" + std::string(digest))
263 return Result<void>::failure(Diagnostic::error(

265 "content-addressed blob path does not match its bytes", entry.path, {},
266 "asset.eva.archive"));
267 }
268 return Result<void>::success();

414Result<std::vector<std::uint8_t>> buildEvaArchive(const EvaManifest& manifest,
415 std::vector<EvaArchiveEntry> entries,
416 const EvaArchiveLimits& limits) {
417 auto manifestJson = serializeEvaManifest(manifest);
418 if (!manifestJson) return Result<std::vector<std::uint8_t>>::failure(manifestJson.status());
419 std::string manifestText = std::move(manifestJson).takeValue();
420 auto validatedManifest = parseEvaManifest(manifestText);
421 if (!validatedManifest) return Result<std::vector<std::uint8_t>>::failure(validatedManifest.status());
422 if (manifestText.size() > limits.maximumManifestBytes)
423 return archiveFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
424 "manifest exceeds admission budget", "manifest.json");
425 entries.push_back({"manifest.json", {manifestText.begin(), manifestText.end()}});
426 if (entries.size() > limits.maximumEntries)
427 return archiveFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
428 "archive has too many entries");
429 std::sort(entries.begin(), entries.end(), [](const auto& left, const auto& right) {
430 return left.path < right.path;
431 });
432 std::uint64_t total = 0;
433 std::string previous;
434 std::set<std::string> foldedPaths;
435 for (const auto& entry : entries) {
436 if (!validPath(entry.path, limits))
437 return archiveFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
438 "entry path is not canonical", entry.path);
439 if (entry.path == previous)
440 return archiveFailure<std::vector<std::uint8_t>>(DiagnosticCode::Conflict,
441 "duplicate archive entry", entry.path);
442 previous = entry.path;
443 const std::string folded = unicodeCaseFold(entry.path);
444 if (folded.empty())
445 return archiveFailure<std::vector<std::uint8_t>>(
446 DiagnosticCode::InvalidArgument, "entry path cannot be case-folded", entry.path);
447 if (!foldedPaths.emplace(folded).second)
448 return archiveFailure<std::vector<std::uint8_t>>(DiagnosticCode::Conflict,
449 "case-colliding archive entry", entry.path);
450 if (entry.bytes.size() > limits.maximumEntryBytes || entry.bytes.size() > limits.maximumDecodedBytes ||
451 total > limits.maximumDecodedBytes - entry.bytes.size())
452 return archiveFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
453 "decoded entry budget exceeded", entry.path);
454 total += entry.bytes.size();
455 }
456 auto definitions = validateAssetDefinitions(validatedManifest.value(), entries, limits);
457 if (!definitions) return Result<std::vector<std::uint8_t>>::failure(definitions.status());
458 auto blobs = validateContentAddressedBlobs(entries);
459 if (!blobs) return Result<std::vector<std::uint8_t>>::failure(blobs.status());
460 auto report = validateImportReport(validatedManifest.value(), entries, limits);
461 if (!report) return Result<std::vector<std::uint8_t>>::failure(report.status());
462
463 std::vector<std::uint8_t> out;
464 std::vector<CentralRecord> records;
465 for (const auto& entry : entries) {
466 auto compressed = deflateRaw(entry.bytes);
467 const bool useDeflate = !compressed.empty();
468 const auto payload = useDeflate ? std::span<const std::uint8_t>(compressed)
469 : std::span<const std::uint8_t>(entry.bytes);
470 CentralRecord record{entry.path, crcOf(entry.bytes), entry.bytes.size(), payload.size(),
471 out.size(), static_cast<std::uint16_t>(useDeflate ? 8 : 0)};
472 put32(out, kLocalSignature); put16(out, 45); put16(out, kUtf8Flag); put16(out, record.method);
473 put16(out, 0); put16(out, 0x21); put32(out, record.crc);
474 put32(out, 0xffffffff); put32(out, 0xffffffff);
475 put16(out, static_cast<std::uint16_t>(entry.path.size())); put16(out, 20);
476 out.insert(out.end(), entry.path.begin(), entry.path.end());
477 put16(out, 0x0001); put16(out, 16); put64(out, record.decodedSize);
478 put64(out, record.compressedSize);
479 out.insert(out.end(), payload.begin(), payload.end());
480 records.push_back(std::move(record));
481 }
482 const std::uint64_t centralOffset = out.size();
483 for (const auto& record : records) {
484 put32(out, kCentralSignature); put16(out, 45); put16(out, 45); put16(out, kUtf8Flag);
485 put16(out, record.method);
486 put16(out, 0); put16(out, 0x21); put32(out, record.crc);
487 put32(out, 0xffffffff); put32(out, 0xffffffff);
488 put16(out, static_cast<std::uint16_t>(record.path.size())); put16(out, 28); put16(out, 0);
489 put16(out, 0); put16(out, 0); put32(out, 0); put32(out, 0xffffffff);
490 out.insert(out.end(), record.path.begin(), record.path.end());
491 put16(out, 0x0001); put16(out, 24); put64(out, record.decodedSize);
492 put64(out, record.compressedSize);
493 put64(out, record.localOffset);
494 }
495 const std::uint64_t centralSize = out.size() - centralOffset;
496 const std::uint64_t zip64EndOffset = out.size();
497 put32(out, kZip64EndSignature); put64(out, 44); put16(out, 45); put16(out, 45);
498 put32(out, 0); put32(out, 0); put64(out, records.size()); put64(out, records.size());
499 put64(out, centralSize); put64(out, centralOffset);
500 put32(out, kZip64LocatorSignature); put32(out, 0); put64(out, zip64EndOffset); put32(out, 1);
501 put32(out, kEndSignature); put16(out, 0); put16(out, 0); put16(out, 0xffff); put16(out, 0xffff);
502 put32(out, 0xffffffff); put32(out, 0xffffffff); put16(out, 0);
503 if (out.size() > limits.maximumArchiveBytes)
504 return archiveFailure<std::vector<std::uint8_t>>(DiagnosticCode::InvalidArgument,
505 "encoded archive exceeds admission budget");
506 return Result<std::vector<std::uint8_t>>::success(std::move(out));
507}

509Result<EvaArchive> parseEvaArchive(std::span<const std::uint8_t> bytes, const EvaArchiveLimits& limits) {
510 if (bytes.size() > limits.maximumArchiveBytes || bytes.size() < 98)
511 return archiveFailure<EvaArchive>(DiagnosticCode::ParseError, "archive size is outside admission limits");
512 std::uint32_t signature = 0;
513 std::uint16_t comment = 0;
514 std::size_t eocd = std::numeric_limits<std::size_t>::max();
515 const std::size_t earliest = bytes.size() > 22 + std::numeric_limits<std::uint16_t>::max()
516 ? bytes.size() - 22 - std::numeric_limits<std::uint16_t>::max()
517 : 0;
518 for (std::size_t candidate = bytes.size() - 22;; --candidate) {
519 if (get32(bytes, candidate, signature) && signature == kEndSignature &&
520 get16(bytes, candidate + 20, comment) &&
521 std::size_t(comment) == bytes.size() - candidate - 22) {
522 eocd = candidate;
523 break;
524 }
525 if (candidate == earliest) break;
526 }
527 if (eocd == std::numeric_limits<std::size_t>::max() || eocd < 20)
528 return archiveFailure<EvaArchive>(DiagnosticCode::ParseError, "canonical ZIP end record is missing");
529 const std::size_t locator = eocd - 20;
530 std::uint64_t zip64Offset = 0;
531 if (!get32(bytes, locator, signature) || signature != kZip64LocatorSignature ||
532 !get64(bytes, locator + 8, zip64Offset) || zip64Offset > locator)
533 return archiveFailure<EvaArchive>(DiagnosticCode::ParseError, "ZIP64 locator is invalid");
534 std::uint64_t entryCount = 0, centralSize = 0, centralOffset = 0;
535 if (!get32(bytes, static_cast<std::size_t>(zip64Offset), signature) || signature != kZip64EndSignature ||
536 !get64(bytes, static_cast<std::size_t>(zip64Offset) + 32, entryCount) ||
537 !get64(bytes, static_cast<std::size_t>(zip64Offset) + 40, centralSize) ||
538 !get64(bytes, static_cast<std::size_t>(zip64Offset) + 48, centralOffset) ||
539 entryCount > limits.maximumEntries || centralOffset > bytes.size() ||
540 centralSize > bytes.size() - centralOffset || centralOffset + centralSize != zip64Offset)
541 return archiveFailure<EvaArchive>(DiagnosticCode::ParseError, "ZIP64 central directory is invalid");
542
543 std::vector<CentralRecord> records;
544 std::set<std::string> names;
545 std::set<std::string> foldedNames;
546 std::uint64_t decodedTotal = 0;
547 std::size_t cursor = static_cast<std::size_t>(centralOffset);
548 for (std::uint64_t index = 0; index < entryCount; ++index) {
549 std::uint16_t flags = 0, method = 0, nameSize = 0, extraSize = 0, commentSize = 0, disk = 0;
550 std::uint32_t crc = 0, external = 0;
551 if (!get32(bytes, cursor, signature) || signature != kCentralSignature ||
552 !get16(bytes, cursor + 8, flags) || !get16(bytes, cursor + 10, method) ||
553 !get32(bytes, cursor + 16, crc) || !get16(bytes, cursor + 28, nameSize) ||
554 !get16(bytes, cursor + 30, extraSize) || !get16(bytes, cursor + 32, commentSize) ||
555 !get16(bytes, cursor + 34, disk) || !get32(bytes, cursor + 38, external) ||
556 flags != kUtf8Flag || (method != 0 && method != 8) || commentSize != 0 || disk != 0 || external != 0 ||
557 !rangeFits(cursor + 46, std::size_t(nameSize) + extraSize, bytes.size()))
558 return archiveFailure<EvaArchive>(DiagnosticCode::Unsupported,
559 "unsupported or malformed ZIP entry");
560 std::string path(reinterpret_cast<const char*>(bytes.data() + cursor + 46), nameSize);
561 const std::string folded = unicodeCaseFold(path);
562 if (!validPath(path, limits) || !names.emplace(path).second || folded.empty() ||
563 !foldedNames.emplace(folded).second)
564 return archiveFailure<EvaArchive>(DiagnosticCode::InvalidArgument,
565 "duplicate or non-canonical entry path", path);
566 const std::size_t extra = cursor + 46 + nameSize;
567 std::uint16_t extraId = 0, zip64Size = 0;
568 std::uint64_t decodedSize = 0, compressedSize = 0, localOffset = 0;
569 if (extraSize != 28 || !get16(bytes, extra, extraId) || extraId != 1 ||
570 !get16(bytes, extra + 2, zip64Size) || zip64Size != 24 ||
571 !get64(bytes, extra + 4, decodedSize) || !get64(bytes, extra + 12, compressedSize) ||
572 !get64(bytes, extra + 20, localOffset) ||
573 localOffset >= centralOffset ||
574 decodedSize > limits.maximumEntryBytes || decodedSize > limits.maximumDecodedBytes ||
575 decodedTotal > limits.maximumDecodedBytes - decodedSize)
576 return archiveFailure<EvaArchive>(DiagnosticCode::ParseError, "ZIP64 entry sizes are invalid", path);
577 decodedTotal += decodedSize;
578 records.push_back({std::move(path), crc, decodedSize, compressedSize, localOffset, method});
579 cursor += 46 + nameSize + extraSize;
580 }
581 if (cursor != zip64Offset)
582 return archiveFailure<EvaArchive>(DiagnosticCode::ParseError, "central directory length mismatch");
583
584 EvaArchive archive;
585 for (const auto& record : records) {
586 const auto local = static_cast<std::size_t>(record.localOffset);
587 std::uint16_t flags = 0, method = 0, nameSize = 0, extraSize = 0;
588 if (!get32(bytes, local, signature) || signature != kLocalSignature ||
589 !get16(bytes, local + 6, flags) || !get16(bytes, local + 8, method) ||
590 !get16(bytes, local + 26, nameSize) || !get16(bytes, local + 28, extraSize) ||
591 flags != kUtf8Flag || method != record.method || (method != 0 && method != 8) || extraSize != 20 ||
592 !rangeFits(local + 30, std::size_t(nameSize) + extraSize, bytes.size()))
593 return archiveFailure<EvaArchive>(DiagnosticCode::ParseError, "local ZIP header is invalid", record.path);
594 const std::string localName(reinterpret_cast<const char*>(bytes.data() + local + 30), nameSize);
595 std::uint16_t extraId = 0, zip64Size = 0;
596 std::uint64_t decodedSize = 0, compressedSize = 0;
597 const std::size_t extra = local + 30 + nameSize;
598 if (localName != record.path || !get16(bytes, extra, extraId) || extraId != 1 ||
599 !get16(bytes, extra + 2, zip64Size) || zip64Size != 16 ||
600 !get64(bytes, extra + 4, decodedSize) || !get64(bytes, extra + 12, compressedSize) ||
601 decodedSize != record.decodedSize || compressedSize != record.compressedSize)
602 return archiveFailure<EvaArchive>(DiagnosticCode::ParseError, "local and central ZIP records disagree",
603 record.path);
604 const std::size_t payload = extra + extraSize;
605 if (record.compressedSize > bytes.size() ||
606 !rangeFits(payload, static_cast<std::size_t>(record.compressedSize), bytes.size()) ||
607 payload + static_cast<std::size_t>(record.compressedSize) > centralOffset)
608 return archiveFailure<EvaArchive>(DiagnosticCode::ParseError, "entry payload is truncated", record.path);
609 auto payloadBytes = bytes.subspan(payload, static_cast<std::size_t>(record.compressedSize));
610 std::vector<std::uint8_t> decoded;
611 if (record.method == 0) {
612 if (record.compressedSize != record.decodedSize)
613 return archiveFailure<EvaArchive>(DiagnosticCode::ParseError,
614 "stored entry sizes disagree", record.path);
615 decoded.assign(payloadBytes.begin(), payloadBytes.end());
616 } else {
617 auto inflated = inflateRaw(payloadBytes, record.decodedSize);
618 if (!inflated) return Result<EvaArchive>::failure(inflated.status());
619 decoded = std::move(inflated).takeValue();
620 }
621 if (crcOf(decoded) != record.crc)
622 return archiveFailure<EvaArchive>(DiagnosticCode::HashMismatch, "entry CRC does not match", record.path);
623 EvaArchiveEntry entry{record.path, std::move(decoded)};
624 if (record.path == "manifest.json") {
625 if (record.decodedSize > limits.maximumManifestBytes)
626 return archiveFailure<EvaArchive>(DiagnosticCode::InvalidArgument, "manifest exceeds budget",
627 record.path);
628 std::string json(entry.bytes.begin(), entry.bytes.end());
629 auto manifest = parseEvaManifest(json);
630 if (!manifest) return Result<EvaArchive>::failure(manifest.status());
631 archive.manifest = std::move(manifest).takeValue();
632 } else {
633 archive.entries.push_back(std::move(entry));
634 }
635 }
636 if (!names.contains("manifest.json"))
637 return archiveFailure<EvaArchive>(DiagnosticCode::ParseError, "root manifest.json is required");
638 std::sort(archive.entries.begin(), archive.entries.end(), [](const auto& left, const auto& right) {
639 return left.path < right.path;
640 });
641 auto definitions = validateAssetDefinitions(archive.manifest, archive.entries, limits);
642 if (!definitions) return Result<EvaArchive>::failure(definitions.status());
643 auto blobs = validateContentAddressedBlobs(archive.entries);
644 if (!blobs) return Result<EvaArchive>::failure(blobs.status());
645 auto report = validateImportReport(archive.manifest, archive.entries, limits);
646 if (!report) return Result<EvaArchive>::failure(report.status());
647 return Result<EvaArchive>::success(std::move(archive));
648}
