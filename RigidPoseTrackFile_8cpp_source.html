18void word(std::ostream& out, std::uint32_t value) {
19 for (unsigned i = 0; i < 4; ++i) out.put(static_cast<char>((value >> (8 * i)) & 255));

21std::uint32_t word(std::istream& in) {
22 std::uint32_t result = 0;
23 for (unsigned i = 0; i < 4; ++i) {
24 const auto value = in.get();
25 if (value < 0) throw std::runtime_error("truncated rigid track");
26 result |= static_cast<std::uint32_t>(value) << (8 * i);
27 }
28 return result;
29}

45 }
46 for (const auto& sample : data_->frames.front().samples) {
47 const auto bytes = sample.id.bytes();

49 }

56 }

62Result<RigidPoseTrack> RigidPoseTrack::load(std::string_view path) {
63 if (path.empty() || path.find('\0') != std::string_view::npos)
64 return Result<RigidPoseTrack>::failure(malformed("invalid rigid track path"));
65 std::ifstream file(std::string(path), std::ios::binary | std::ios::ate);
66 if (!file) {
67 std::error_code error;
68 const auto status = std::filesystem::status(std::filesystem::path(path), error);
69 const bool missing =
70 error == std::errc::no_such_file_or_directory || (!error && !std::filesystem::exists(status));
71 return Result<RigidPoseTrack>::failure(
72 Diagnostic::error(missing ? DiagnosticCode::NotFound : DiagnosticCode::Failed,
73 "rigid track cannot be opened", std::string(path)));
74 }
75 const auto size = file.tellg();
76 if (size < 56 || static_cast<std::uint64_t>(size) > MaxBytes)
77 return Result<RigidPoseTrack>::failure(malformed("rigid track file size outside budget"));
78 std::string bytes(static_cast<std::size_t>(size), '\0');
79 file.seekg(0);
80 file.read(bytes.data(), static_cast<std::streamsize>(bytes.size()));
81 if (!file) return Result<RigidPoseTrack>::failure(malformed("rigid track read failed"));
82 if (bakedAssetContentKey(std::string_view(bytes).substr(0, bytes.size() - 32)) !=
83 std::string_view(bytes).substr(bytes.size() - 32))
84 return Result<RigidPoseTrack>::failure(
85 Diagnostic::error(DiagnosticCode::HashMismatch, "rigid track checksum mismatch", "rigidPoseTrack"));
86 bytes.resize(bytes.size() - 32);
87 const auto payloadSize = bytes.size();
88 std::istringstream in(std::move(bytes), std::ios::binary);
89 try {
90 char magic[8];
91 in.read(magic, 8);
92 const auto version = word(in);
93 if (!in || std::string_view(magic, 8) != "EVERIGD1" || (version != 1 && version != 2))
94 return Result<RigidPoseTrack>::failure(malformed("unknown rigid track schema/version"));
95 const auto spaceCode = word(in), tracks = word(in), frames = word(in);
96 if ((spaceCode != 1 && spaceCode != 2) || !tracks || tracks > MaxTracks || !frames ||
97 static_cast<std::uint64_t>(frames) * tracks > MaxSamples)
98 return Result<RigidPoseTrack>::failure(malformed("invalid rigid track counts or coordinates"));
99 std::shared_ptr<const BakeRecipe> provenance;
100 std::uint64_t metadataBytes = 0;
101 if (version == 2) {
102 const auto recipeSize = word(in);
103 if (recipeSize == 0 || recipeSize > 65536 || recipeSize > payloadSize - 28)
104 return Result<RigidPoseTrack>::failure(malformed("invalid rigid recipe length"));
105 std::string encoded(recipeSize, '\0');
106 in.read(encoded.data(), static_cast<std::streamsize>(recipeSize));
107 if (!in) return Result<RigidPoseTrack>::failure(malformed("truncated rigid recipe"));
108 auto decoded = BakeRecipe::decode(encoded);
109 if (!decoded) return Result<RigidPoseTrack>::failure(decoded.status());
110 provenance = std::make_shared<const BakeRecipe>(std::move(decoded).takeValue());
111 metadataBytes = 4ull + recipeSize;
112 }
113 const auto expected =
114 24 + metadataBytes + 16ull * tracks + static_cast<std::uint64_t>(frames) * (4 + 28ull * tracks);
115 if (payloadSize != expected)
116 return Result<RigidPoseTrack>::failure(malformed("truncated or trailing rigid track fields"));
117 std::vector<PersistentId> ids;
118 ids.reserve(tracks);
119 for (std::uint32_t i = 0; i < tracks; ++i) {
120 PersistentId::Bytes identity{};
121 in.read(reinterpret_cast<char*>(identity.data()), static_cast<std::streamsize>(identity.size()));
122 const PersistentId id(identity);
123 if (id.isNil() || (!ids.empty() && id <= ids.back()))
124 return Result<RigidPoseTrack>::failure(malformed("noncanonical rigid track identities"));
125 ids.push_back(id);
126 }
127 std::vector<RigidTrackFrame> candidate;
128 candidate.reserve(frames);
129 for (std::uint32_t f = 0; f < frames; ++f) {
130 RigidTrackFrame frame{real(in), {}};
131 frame.samples.reserve(tracks);
132 for (auto id : ids) {
133 RigidTrackPose p;
134 p.x = real(in);
135 p.y = real(in);
136 p.z = real(in);
137 p.qx = real(in);
138 p.qy = real(in);
139 p.qz = real(in);
140 p.qw = real(in);
141 frame.samples.push_back({id, p});
142 }
143 candidate.push_back(std::move(frame));
144 }
145 const auto space = spaceCode == 1 ? RigidTrackSpace::PlanarPixels : RigidTrackSpace::Metres3D;
146 if (provenance) return prepare(space, std::move(candidate), *provenance);
147 return prepare(space, std::move(candidate));
148 } catch (const std::exception&) {
149 return Result<RigidPoseTrack>::failure(malformed("rigid track decoding failed"));
150 }

152Result<RigidPoseTrack> RigidPoseTrack::loadForRecipe(std::string_view path, const BakeRecipe& expected) {
153 auto expectedBytes = expected.encode();
154 if (!expectedBytes) return Result<RigidPoseTrack>::failure(expectedBytes.status());
155 auto loaded = load(path);
156 if (!loaded) return loaded;
157 const auto provenance = loaded.value().recipe();
158 if (!provenance)
159 return Result<RigidPoseTrack>::failure(
160 Diagnostic::error(DiagnosticCode::PreconditionViolation,
161 "rigid asset has no provenance; explicit playback or rebake required", "rigidPoseTrack"));
162 auto actual = provenance->encode();
163 if (!actual) return Result<RigidPoseTrack>::failure(actual.status());
164 if (actual.value() != expectedBytes.value())
165 return Result<RigidPoseTrack>::failure(
166 Diagnostic::error(DiagnosticCode::HashMismatch, "rigid asset provenance mismatch", "rigidPoseTrack"));
167 return loaded;
168}
