12 std::smatch result;

16Result<double> property(const std::string& text, const std::string& name, double defaultValue) {
17 const auto value = match(text, "(?:^|\\n)\\s*- " + name + ": *([^\\r\\n]+)");
18 if (!value) return Result<double>::success(defaultValue);
19 auto parsed = Value::fromJson(*value);
20 if (!parsed || !parsed.value().isNumeric())
21 return detail::failure<double>(DiagnosticCode::ParseError, "invalid Unity material scalar", name);
22 const auto number = parsed.value().isInt64() ? double(parsed.value().asInt()) : parsed.value().asDouble();
23 if (!std::isfinite(number))
24 return detail::failure<double>(DiagnosticCode::ParseError, "nonfinite material scalar", name);
25 return Result<double>::success(number);
26}
27std::string refKey(const std::string& guid, const std::string& id) {
28 return "unity:" + unity_detail::foldAscii(guid) + "/" + id;

30struct Document {
31 std::string type, id, body;
32};
33std::vector<Document> documents(const std::string& text) {
34 const std::regex header(R"((?:^|\n)--- !u!([0-9]+) &(-?[0-9]+)[^\n]*\n)");
35 std::vector<Document> out;
36 std::size_t start = 0;
37 for (std::sregex_iterator it(text.begin(), text.end(), header), end; it != end; ++it) {
38 if (!out.empty()) out.back().body = text.substr(start, std::size_t(it->position()) - start);
39 out.push_back({(*it)[1].str(), (*it)[2].str(), {}});
40 start = std::size_t(it->position() + it->length());
41 }
42 if (!out.empty()) out.back().body = text.substr(start);
43 return out;
44}

92 {"baseColor", Value(std::move(color))},
93 {"metallic", Value(metallic.value())},

104 return detail::failure<PreparedAssetImport>(DiagnosticCode::Unsupported,

108 if (!image) return Result<PreparedAssetImport>::failure(image.status());

113 "baseColorTexture",
114 {},
115 "eve.image/3",
116 {}});
117 }
118 }

120 auto json = Value(std::move(definition)).toJson();
121 if (!json) return Result<PreparedAssetImport>::failure(json.status());
122 std::vector<std::uint8_t> encoded(json.value().begin(), json.value().end());
123 out.manifest.assets.push_back(

125 out.entries.push_back({path, std::move(encoded)});
126 out.manifest.entrypoints.emplace("default", ref.value());
127 out.sourceMappings.push_back({"2100000", ref.value()});
128 out.findings.push_back({source.path, "Material.Standard", ImportDisposition::Translated,
129 "metallic-roughness, base color and surface mode"});
130 out.findings.push_back({source.path, "Material.extraProperties", ImportDisposition::Unsupported,
131 "additional shader keywords, non-base texture maps and saved properties are not applied"});
132 return Result<PreparedAssetImport>::success(std::move(out));
133}

277 if (renderers.empty()) continue;
278 (*object)["schemaVersion"] = Value(std::int64_t(3));
279 (*object)["renderers"] = Value(std::move(renderers));
280 auto encoded = definition.value().toJson();

288 out.findings.push_back({source.path, "Prefab.renderBindings", ImportDisposition::Translated,
