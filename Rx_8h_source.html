92class Subscription {
93public:
94 Subscription() = default;
96 explicit Subscription(std::function<void()> dispose) : dispose_(std::move(dispose)) {}
97 Subscription(const Subscription&) = delete;
98 Subscription& operator=(const Subscription&) = delete;
99 Subscription(Subscription&& other) noexcept : dispose_(std::move(other.dispose_)), disposed_(other.disposed_) {
100 other.dispose_ = nullptr;
101 other.disposed_ = true;
102 }
103 Subscription& operator=(Subscription&& other) noexcept {
104 if (this != &other) {
105 dispose();
106 dispose_ = std::move(other.dispose_);
107 disposed_ = other.disposed_;
108 other.dispose_ = nullptr;
109 other.disposed_ = true;
110 }
111 return *this;
112 }
113 ~Subscription() { dispose(); }
114
116 void dispose() {
117 if (dispose_ && !disposed_) {
118 disposed_ = true;
119 auto fn = std::move(dispose_);
120 dispose_ = nullptr;
121 fn();
122 }
123 }
125 bool isDisposed() const { return disposed_; }
126
127private:
128 std::function<void()> dispose_;
129 bool disposed_ = false;

264Observable<T>* Observable<T>::take(int n) {
265 auto* self = this;
266 return new AnonymousObservable<T>([self, n](Observer<T> out) {
267 auto remaining = std::make_shared<int>(n);
268 Observer<T> in;
269 in.onNext = [out, remaining](const T& v) {
270 if (*remaining <= 0) return;
271 *remaining -= 1;
272 out.next(v);
273 if (*remaining == 0) {
274 out.completed();

323 auto* self = this;

349template <typename T>
350Observable<T>* Observable<T>::distinctUntilChanged() {
351 auto* self = this;
352 return new AnonymousObservable<T>([self](Observer<T> out) {
353 auto last = std::make_shared<T>();
354 auto has = std::make_shared<bool>(false);
355 Observer<T> in;
356 in.onNext = [out, last, has](const T& v) {
357 if (*has && *last == v) return;
358 *has = true;
359 *last = v;
360 out.next(v);
361 };
362 in.onError = [out](const std::string& e) { out.error(e); };
363 in.onCompleted = [out]() { out.completed(); };
364 return self->subscribe(std::move(in));

372template <typename T>
373class Subject : public Observable<T> {
374public:
375 using Observable<T>::subscribe;
376 ~Subject() override = default;
377
379 Subscription subscribe(Observer<T> obs) override {
380 auto slot = std::make_shared<Slot>();
381 slot->observer = std::move(obs);
382 {
383 std::lock_guard<std::mutex> lock(mu_);
384 if (stopped_) {
385 // Terminal state: deliver the terminal notification immediately
386 // to a fresh copy of the observer and never register the slot.
387 Observer<T> snapshot = slot->observer;
388 snapshot.completed();
389 return Subscription{};
390 }
391 slots_.push_back(slot);
392 }
393 return Subscription([this, slot]() { removeSlot(slot); });
394 }
395
397 void onNext(const T& v) { emit([&](Observer<T>& o) { o.next(v); }); }
399 void onError(const std::string& e) {
400 emit([&](Observer<T>& o) { o.error(e); });
401 markStopped();
402 }
404 void onCompleted() {
405 emit([&](Observer<T>& o) { o.completed(); });
406 markStopped();
407 }
408
410 bool hasObservers() const {
411 std::lock_guard<std::mutex> lock(mu_);
412 for (const auto& s : slots_)
413 if (!s->removed) return true;
414 return false;
415 }
417 int observerCount() const {
418 std::lock_guard<std::mutex> lock(mu_);
419 int n = 0;
420 for (const auto& s : slots_)
421 if (!s->removed) n++;
422 return n;
423 }
424
425private:
426 struct Slot {
427 Observer<T> observer;
428 bool removed = false;
429 };
430
431 void emit(const std::function<void(Observer<T>&)>& fn) {
432 std::vector<std::shared_ptr<Slot>> copy;
433 {
434 std::lock_guard<std::mutex> lock(mu_);
435 for (auto& s : slots_)
436 if (!s->removed) copy.push_back(s);
437 }
438 for (auto& s : copy) {
439 Observer<T> snapshot = s->observer;
440 fn(snapshot);
441 }
442 }
443
444 void removeSlot(const std::shared_ptr<Slot>& slot) {
445 std::lock_guard<std::mutex> lock(mu_);
446 slot->removed = true;
447 }
448
449 void markStopped() {
450 std::lock_guard<std::mutex> lock(mu_);
451 stopped_ = true;
452 slots_.clear();
453 }
454
455 mutable std::mutex mu_;
456 std::vector<std::shared_ptr<Slot>> slots_;
457 bool stopped_ = false;
458};
