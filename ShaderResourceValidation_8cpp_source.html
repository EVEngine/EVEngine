21 spvtools::SpirvTools validator(SPV_ENV_VULKAN_1_2);
22 if (!validator.Validate(words.data(), words.size()))
23 return fail("Invalid Vulkan 1.2 SPIR-V", DiagnosticCode::ParseError);
24 spirv_cross::Compiler compiler(words.data(), words.size());
25 const auto entries = compiler.get_entry_points_and_stages();
26 if (entries.size() != 1 || entries[0].name != "main" ||
27 entries[0].execution_model != (vertex ? spv::ExecutionModelVertex : spv::ExecutionModelFragment))
28 return fail("Expected one main entry point with the requested stage");
29 for (auto cap : compiler.get_declared_capabilities())
30 if (cap != spv::CapabilityShader && cap != spv::CapabilityMatrix)
31 return fail("Resource programs require only Shader/Matrix capabilities");
32 if (!compiler.get_declared_extensions().empty() || !compiler.get_specialization_constants().empty())
33 return fail("Extensions and specialization constants are unsupported");
34 const auto r = compiler.get_shader_resources();
35 if (!r.storage_images.empty() || !r.subpass_inputs.empty() || !r.separate_images.empty() ||
36 !r.separate_samplers.empty() || !r.atomic_counters.empty())
37 return fail("Resource programs require combined sampled images and uniform buffers");
38 if (r.storage_buffers.size() > 1) return fail("Only the readonly instance matrix buffer is supported");
39 for (const auto& buffer : r.storage_buffers) {
40 const auto& t = compiler.get_type(buffer.base_type_id);
41 if (!vertex || inputs.instanceMatrices.empty() ||
42 compiler.get_decoration(buffer.id, spv::DecorationDescriptorSet) != 1 ||
43 compiler.get_decoration(buffer.id, spv::DecorationBinding) != 33 ||
44 !compiler.get_type(buffer.type_id).array.empty() || t.member_types.size() != 1 ||
45 compiler.type_struct_member_offset(t, 0) != 0)
46 return fail("Instance storage must be vertex-only set 1 binding 33");
47 const auto& m = compiler.get_type(t.member_types[0]);
48 if ((!compiler.has_decoration(buffer.id, spv::DecorationNonWritable) &&
49 !compiler.has_member_decoration(t.self, 0, spv::DecorationNonWritable)) ||
50 m.basetype != spirv_cross::SPIRType::Float || m.width != 32 || m.vecsize != 4 || m.columns != 4 ||
51 m.array.size() != 1 || m.array[0] != 0 || compiler.type_struct_member_array_stride(t, 0) != 64 ||
52 compiler.type_struct_member_matrix_stride(t, 0) != 16 ||
53 compiler.has_member_decoration(t.self, 0, spv::DecorationRowMajor))
54 return fail("Instance storage must contain readonly column-major mat4 records");
55 }
56 for (const auto& image : r.sampled_images) {
57 const auto& t = compiler.get_type(image.type_id);
58 const auto set = compiler.get_decoration(image.id, spv::DecorationDescriptorSet);
59 const auto binding = compiler.get_decoration(image.id, spv::DecorationBinding);
60 if (!t.array.empty() || t.image.ms || t.image.depth ||
61 compiler.get_type(t.image.type).basetype != spirv_cross::SPIRType::Float)
62 return fail("Expected a non-shadow, non-MS float combined image");
63 ShaderImageDimension expected = ShaderImageDimension::Image2D;
64 if (set == 0 && binding == 1 && !vertex) {
65 // The existing per-material albedo binding.
66 } else if (set == 1) {
67 auto found = std::find_if(inputs.images.begin(), inputs.images.end(),
68 [&](const auto& item) { return item.binding == binding; });
69 if (found == inputs.images.end()) return fail("Shader image binding has no supplied resource");
70 expected = found->dimension;
71 } else
72 return fail("Image binding is outside the resource program ABI");
73 if (t.image.dim != (expected == ShaderImageDimension::Cube ? spv::DimCube : spv::Dim2D) ||
74 bool(t.image.arrayed) != (expected == ShaderImageDimension::Array2D))
75 return fail("Shader image view dimension does not match the supplied resource");
76 }
77 for (const auto& buffer : r.uniform_buffers) {
78 const auto set = compiler.get_decoration(buffer.id, spv::DecorationDescriptorSet);
79 const auto binding = compiler.get_decoration(buffer.id, spv::DecorationBinding);
80 const auto& t = compiler.get_type(buffer.base_type_id);
81 if (!compiler.get_type(buffer.type_id).array.empty()) return fail("Buffer descriptor arrays unsupported");
82 if (set == 1 && binding == 32) {
83 if (inputs.constants.empty() || compiler.get_declared_struct_size(t) > inputs.constants.size())
84 return fail("Uniform buffer is larger than its supplied bytes");
85 } else if (set == 0 && binding == 0) {
86 if (t.member_types.empty() || t.member_types.size() > 6) return fail("Expected the supported Frame prefix");
87 for (unsigned i = 0; i < t.member_types.size(); ++i) {
88 const auto& m = compiler.get_type(t.member_types[i]);
89 if (compiler.type_struct_member_offset(t, i) != (i < 2 ? i * 64 : 128 + (i - 2) * 16))
90 return fail("Frame member offset mismatch");
91 if (i < 2) {
92 if (m.basetype != spirv_cross::SPIRType::Float || m.width != 32 || m.columns != 4 ||
93 m.vecsize != 4 || !m.array.empty() || compiler.type_struct_member_matrix_stride(t, i) != 16 ||
94 compiler.has_member_decoration(t.self, i, spv::DecorationRowMajor))
95 return fail("Frame matrices must be column-major mat4");
96 } else if (!floats(m, 4))
97 return fail("Frame trailing members must be vec4");
98 }
99 } else
100 return fail("Uniform binding is outside the resource program ABI");
101 }
102 if (r.push_constant_buffers.size() > 1) return fail("Multiple push constant blocks unsupported");
103 for (const auto& buffer : r.push_constant_buffers) {
104 const auto& t = compiler.get_type(buffer.base_type_id);
105 if (t.member_types.size() != 1 || compiler.type_struct_member_offset(t, 0) != 0)
106 return fail("Push constants must contain float data[32]");
107 const auto& m = compiler.get_type(t.member_types[0]);
108 if (m.basetype != spirv_cross::SPIRType::Float || m.width != 32 || m.vecsize != 1 || m.columns != 1 ||
109 m.array.size() != 1 || !m.array_size_literal[0] || m.array[0] != 32 ||
110 compiler.type_struct_member_array_stride(t, 0) != 4)
111 return fail("Push constants must be 128 tightly packed float bytes");
112 }
113 if (vertex) {
114 for (const auto& input : r.stage_inputs) {
115 auto location = compiler.get_decoration(input.id, spv::DecorationLocation);
116 const uint32_t sizes[] = {3, 3, 2, 0, 0, 4};
117 if (location > 5 || sizes[location] == 0 || !floats(compiler.get_type(input.type_id), sizes[location]))
118 return fail("Resource vertex inputs require position/normal/UV or optional vec4 tangent at location 5");
119 }
120 } else if (r.stage_outputs.size() != 1 ||
121 compiler.get_decoration(r.stage_outputs[0].id, spv::DecorationLocation) != 0 ||
122 !floats(compiler.get_type(r.stage_outputs[0].type_id), 4))
123 return fail("Fragment output must be one vec4 at location 0");
124 return Result<void>::success();
