32 editor::EditorWorkspace candidate = workspace;
33 struct Panel {
34 const char* id;
35 const char* title;

73procgen_editing::Result<void> ProcgenScriptEditor::commit(procgen_editing::Result<editing::DomainOperation> operation,
74 std::string label) {
75 if (!operation.ok()) return procgen_editing::Result<void>::failure(operation.status());
76 editor::TransactionSpec spec;
77 spec.id = editor::TransactionId("procgen.script.tx." + std::to_string(++txSequence_));
78 spec.label = std::move(label);
79 spec.target = editor::TargetId(target_.targetId());
80 spec.baseRevision = target_.revision();
81 auto begun = transactions_.begin(std::move(spec));
82 if (!begun.ok())
83 return editorError(begun.code(), "editor.procgen.begin", "Could not begin the generator transaction");
84 auto appended = transactions_.append(std::move(operation).takeValue());
85 if (!appended.ok()) {
86 auto discarded = transactions_.discard();
87 if (!discarded.ok()) discarded.ignore("pending procgen script transaction already inactive");
88 return procgen_editing::Result<void>::failure(appended.status());
89 }
90 auto committed = transactions_.commit();
91 if (!committed.ok()) return procgen_editing::Result<void>::failure(committed.status());
92 dirty_ = true;
93 previewFailureSummary_.clear();
94 return eve::editing::applied<void>();
95}

132procgen_editing::Result<std::vector<ProcgenScriptEditor::PreviewPoint>> ProcgenScriptEditor::copyPoints(
133 const procgen::PointSet* points) const {
134 if (!points)
135 return editorError<std::vector<PreviewPoint>>(procgen_editing::EditorStatus::Rejected, "editor.procgen.preview",
136 "Preview PointSet must not be null");
137 if (pointBudget_ > 0 && points->getCount() > pointBudget_)
138 return editorError<std::vector<PreviewPoint>>(procgen_editing::EditorStatus::Rejected,
139 "editor.procgen.point-budget",
140 "Preview exceeded the point budget");
141 std::vector<PreviewPoint> next;
142 next.reserve(static_cast<std::size_t>(points->getCount()));
143 for (int i = 0; i < points->getCount(); ++i) {
144 PreviewPoint point;
145 point.x = points->getX(i);
146 point.z = points->getZ(i);
147 point.seed = points->getPointSeed(i);
148 next.push_back(point);
149 }
150 return eve::editing::applied(std::move(next));
151}

153procgen_editing::Result<void> ProcgenScriptEditor::publishPreview(const procgen::PointSet* points, std::string stage,
154 std::uint64_t expectedRevision) {
155 if (expectedRevision != target_.revision())
156 return editorError(procgen_editing::EditorStatus::Conflict, "editor.procgen.stale-preview",
157 "Preview revision does not match the generator document");
158 auto copied = copyPoints(points);
159 if (!copied.ok()) return procgen_editing::Result<void>::failure(copied.status());
160 if (stage.empty()) stage = "output";
161 if (std::find(stageOrder_.begin(), stageOrder_.end(), stage) == stageOrder_.end()) stageOrder_.push_back(stage);
162 stages_[stage] = std::move(copied).takeValue();
163 selectedStage_ = stage;
164 previewRevision_ = target_.revision();
165 dirty_ = false;
166 previewFailureSummary_.clear();
167 return eve::editing::applied<void>();
168}
