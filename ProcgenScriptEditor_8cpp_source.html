25procgen_editing::EditorResult<void> ProcgenScriptEditor::configureWorkspace(
26 editor::EditorWorkspace& workspace) const {
27 editor::EditorWorkspace candidate = workspace;
28 struct Panel {
29 const char* id;
30 const char* title;
31 const char* region;
32 const char* context;
33 int order;
34 };
35 constexpr Panel panels[] = {
36 {"procgen.modules", "Modules", "left", "list", 100},
37 {"procgen.preview", "World Preview", "center", "preview", 100},
38 {"procgen.inspector", "Generator Inspector", "right", "inspector", 100},
39 {"procgen.stages", "Debug Stages", "bottom", "stages", 100},
40 };
41 for (const auto& panel : panels) {
42 if (!candidate.registerPanel(panel.id, panel.title, panel.region, panel.order) ||
43 !candidate.setPanelCapability(panel.id, "procgen.script") ||
44 !candidate.setPanelContext(panel.id, panel.context))
45 return editorError(procgen_editing::EditorStatus::Rejected, "editor.procgen.workspace-conflict",
46 "Could not install the procgen script workspace composition");
47 }
48 if (!candidate.activatePanel("procgen.preview"))
49 return editorError(procgen_editing::EditorStatus::Rejected, "editor.procgen.workspace-activate",
50 "Could not activate the procgen preview panel");
51 workspace = std::move(candidate);
52 return eve::editing::applied<void>();

68procgen_editing::EditorResult<void> ProcgenScriptEditor::commit(
69 procgen_editing::EditorResult<editing::DomainOperation> operation, std::string label) {
70 if (!operation.ok()) return procgen_editing::EditorResult<void>::failure(operation.status());
71 editor::TransactionSpec spec;
72 spec.id = editor::TransactionId("procgen.script.tx." + std::to_string(++txSequence_));
73 spec.label = std::move(label);
74 spec.target = editor::TargetId(target_.targetId());
75 spec.baseRevision = target_.revision();
76 auto begun = transactions_.begin(std::move(spec));
77 if (!begun.ok())
78 return editorError(begun.code(), "editor.procgen.begin", "Could not begin the generator transaction");
79 auto appended = transactions_.append(std::move(operation).takeValue());
80 if (!appended.ok()) {
81 auto discarded = transactions_.discard();
82 if (!discarded.ok()) discarded.ignore("pending procgen script transaction already inactive");
83 return procgen_editing::EditorResult<void>::failure(appended.status());
84 }
85 auto committed = transactions_.commit();
86 if (!committed.ok()) return procgen_editing::EditorResult<void>::failure(committed.status());
87 dirty_ = true;
88 previewFailureSummary_.clear();
89 return eve::editing::applied<void>();
90}

129procgen_editing::EditorResult<std::vector<ProcgenScriptEditor::PreviewPoint>> ProcgenScriptEditor::copyPoints(
130 const procgen::PointSet* points) const {
131 if (!points)
132 return editorError<std::vector<PreviewPoint>>(procgen_editing::EditorStatus::Rejected,
133 "editor.procgen.preview", "Preview PointSet must not be null");
134 if (pointBudget_ > 0 && points->getCount() > pointBudget_)
135 return editorError<std::vector<PreviewPoint>>(procgen_editing::EditorStatus::Rejected,
136 "editor.procgen.point-budget",
137 "Preview exceeded the point budget");
138 std::vector<PreviewPoint> next;
139 next.reserve(static_cast<std::size_t>(points->getCount()));
140 for (int i = 0; i < points->getCount(); ++i) {
141 PreviewPoint point;
142 point.x = points->getX(i);
143 point.z = points->getZ(i);
144 point.seed = points->getPointSeed(i);
145 next.push_back(point);
146 }
147 return eve::editing::applied(std::move(next));
148}

150procgen_editing::EditorResult<void> ProcgenScriptEditor::publishPreview(const procgen::PointSet* points,
151 std::string stage,
152 std::uint64_t expectedRevision) {
153 if (expectedRevision != target_.revision())
154 return editorError(procgen_editing::EditorStatus::Conflict, "editor.procgen.stale-preview",
155 "Preview revision does not match the generator document");
156 auto copied = copyPoints(points);
157 if (!copied.ok()) return procgen_editing::EditorResult<void>::failure(copied.status());
158 if (stage.empty()) stage = "output";
159 if (std::find(stageOrder_.begin(), stageOrder_.end(), stage) == stageOrder_.end())
160 stageOrder_.push_back(stage);
161 stages_[stage] = std::move(copied).takeValue();
162 selectedStage_ = stage;
163 previewRevision_ = target_.revision();
164 dirty_ = false;
165 previewFailureSummary_.clear();
166 return eve::editing::applied<void>();
167}
