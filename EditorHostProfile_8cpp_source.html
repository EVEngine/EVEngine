288std::string EditorInspector::pollChangedId() {
289 const int n = static_cast<int>(fields_.size());
290 if (n == 0) return "";
291 for (int k = 0; k < n; ++k) {
292 int i = (pollCursor_ + k) % n;
293 if (fields_[i].dirty) {
294 fields_[i].dirty = false;
295 pollCursor_ = (i + 1) % n;
296 return fields_[i].id;
297 }
298 }
299 return "";

308EditorValue operationValue(const DomainOperation &operation) {
309 EditorValue::Object value;
310 value["type"] = EditorValue(operation.type);
311 value["inverseType"] = EditorValue(operation.inverseType);
312 value["target"] = EditorValue(operation.target.value());
313 value["payload"] = operation.payload;
314 value["inverse"] = operation.inverse;
315 value["hasInverse"] = EditorValue(operation.hasInverse);
316 value["mergeKey"] = EditorValue(operation.mergeKey);
317 return EditorValue(std::move(value));
318}
319
320DomainOperation decodeOperation(const EditorValue &value) {
321 DomainOperation operation;
322 const auto *object = value.getIf<EditorValue::Object>();
323 if (!object) return operation;
324 const auto text = [&](const char *key) {
325 auto found = object->find(key);
326 const auto *result = found == object->end() ? nullptr : found->second.getIf<std::string>();
327 return result ? *result : std::string{};
328 };
329 operation.type = text("type");
330 operation.inverseType = text("inverseType");
331 operation.target = TargetId(text("target"));
332 operation.mergeKey = text("mergeKey");
333 auto payload = object->find("payload");
334 if (payload != object->end()) operation.payload = payload->second;
335 auto inverse = object->find("inverse");
336 if (inverse != object->end()) operation.inverse = inverse->second;
337 auto hasInverse = object->find("hasInverse");
338 const auto *flag = hasInverse == object->end() ? nullptr : hasInverse->second.getIf<bool>();
339 operation.hasInverse = flag && *flag;
340 return operation;
341}

349Result<EditorPersistenceSnapshot> EditorPersistenceAdapter::load() const {
350 if (!store_ || resourceUri_.empty())
351 return persistenceError(EditorStatus::Rejected, "editor.persistence.invalid",
352 "Persistence store and URI are required");
353 auto stored = store_->read(resourceUri_);
354 if (!stored.ok()) {
355 if (stored.code() == EditorStatus::NotFound) {
356 EditorPersistenceSnapshot empty;
357 empty.kind = kind_;
358 return eve::editing::applied<EditorPersistenceSnapshot>(std::move(empty));
359 }
360 return Result<EditorPersistenceSnapshot>::failure(stored.status());
361 }
362 return decode(kind_, stored.value());
363}

365Result<EditorPersistenceSnapshot> EditorPersistenceAdapter::commit(const EditorPersistenceSnapshot &base,
366 EditorValue content,
367 std::vector<DomainOperation> journal) {
368 if (!store_ || base.kind != kind_)
369 return persistenceError(EditorStatus::Rejected, "editor.persistence.kind-mismatch",
370 "Persistence checkpoint belongs to another backend kind");
371 EditorValue::Array serializedJournal;
372 for (const DomainOperation &operation : journal) serializedJournal.push_back(operationValue(operation));
373 EditorValue::Object envelope;
374 envelope["schemaVersion"] = EditorValue(1);
375 envelope["content"] = content;
376 envelope["journal"] = EditorValue(std::move(serializedJournal));
377 auto stored =
378 store_->compareAndSwap(resourceUri_, base.revision, base.contentHash, EditorValue(std::move(envelope)));
379 if (!stored.ok()) return Result<EditorPersistenceSnapshot>::failure(stored.status());
380 return decode(kind_, stored.value());
381}
382
383Result<EditorPersistenceSnapshot> EditorPersistenceAdapter::decode(EditorPersistenceKind kind,
384 const StoredDocument &stored) {
385 const auto *envelope = stored.content.getIf<EditorValue::Object>();
386 if (!envelope)
387 return persistenceError(EditorStatus::Failed, "editor.persistence.invalid-envelope",
388 "Persistence checkpoint must be an object");
389 const auto content = envelope->find("content");
390 const auto journal = envelope->find("journal");
391 const auto *journalValues = journal == envelope->end() ? nullptr : journal->second.getIf<EditorValue::Array>();
392 if (content == envelope->end() || !journalValues)
393 return persistenceError(EditorStatus::Failed, "editor.persistence.invalid-envelope",
394 "Persistence checkpoint content or journal is missing");
395 EditorPersistenceSnapshot result;
396 result.kind = kind;
397 result.revision = stored.revision;
398 result.contentHash = stored.contentHash;
399 result.content = content->second;
400 for (const EditorValue &value : *journalValues) result.journal.push_back(decodeOperation(value));
401 return eve::editing::applied<EditorPersistenceSnapshot>(std::move(result));
402}

421eve::Result<void> writeFailure(const Result<T> &result, std::string fallbackCode, std::string fallbackMessage) {
422 if (!result.diagnostics().empty()) {
423 const EditorDiagnostic &diagnostic = result.diagnostics().front();
424 const RuleId rule = eve::editing::diagnosticRule(diagnostic);
425 if (!rule.empty() && !diagnostic.message().empty())
426 return property_access::rejected(rule.value(), diagnostic.message());
427 if (!diagnostic.message().empty())
428 return property_access::rejected(std::move(fallbackCode), diagnostic.message());
429 }
430 return property_access::rejected(std::move(fallbackCode), std::move(fallbackMessage));
431}

433TransactionId newPropertyTransactionId() {
434 static std::atomic<std::uint64_t> entropySequence{1};
435 const eve::UuidV7Generator generator([](std::span<std::uint8_t> bytes) {
436 const std::uint64_t seed = entropySequence.fetch_add(1, std::memory_order_relaxed);
437 for (std::size_t index = 0; index < bytes.size(); ++index)
438 bytes[index] = static_cast<std::uint8_t>(seed >> ((index % sizeof(seed)) * 8u));
439 return true;
440 });
441 const auto generated = generator.generate(std::chrono::system_clock::now());
442 if (!generated) {
443 // A clock before the UUID epoch is not expected for a live editor, but
444 // retain a valid canonical UUID path if a platform clock is malformed.
445 const auto epoch = generator.generate(std::chrono::system_clock::time_point{});
446 if (!epoch) return TransactionId{};
447 using CanonicalUuid = typename TransactionId::Uuid;
448 return TransactionId::fromUuid(CanonicalUuid(epoch->bytes()));
449 }
450 using CanonicalUuid = typename TransactionId::Uuid;
451 return TransactionId::fromUuid(CanonicalUuid(generated->bytes()));
452}

547eve::Result<void> EditorPropertyModel::write(const std::string &path, const eve::Value &value) {
548 if (!provider_) return property_access::rejected("editor.property.provider", "No property provider");
549 const Result<void> revisionCheck = ensureCurrentRevision();
550 if (!revisionCheck.ok())
551 return writeFailure(revisionCheck, "editor.property.revision", "Property revision is unavailable");
552 if (transactionBackend_ && transactionBackend_->active() && pendingPaths_.contains(path))
553 return property_access::rejected("editor.property.duplicate-path",
554 "A property may be written only once per transaction");
555
556 const EditorValue editorValue = toEditorValue(value);
557 Result<PropertyEditIntent> intent = [&]() {
558 if (surface_ == PropertyModelSurface::Runtime) {
559 RuntimePropertyPresenter presenter;
560 return presenter.editIntent(editorSchema_, selection_, PropertyPath(path), editorValue, profile_);
561 }
562 DeveloperPropertyPresenter presenter;
563 return presenter.editIntent(editorSchema_, selection_, PropertyPath(path), editorValue);
564 }();
565 if (!intent.ok()) {
566 return writeFailure(intent, "editor.property.intent", "Property edit was rejected");
567 }
568
569 if (transactionBackend_) {
570 Result<DomainOperation> operation =
571 provider_->makeSet(selection_, PropertyPath(path), editorValue, PropertySetMode::Absolute);
572 if (!operation.ok())
573 return writeFailure(operation, "editor.property.operation", "Property operation was rejected");
574 if (operation.value().target.empty())
575 return property_access::rejected("editor.property.target", "Property operation has no target");
576 if (operation.value().mergeKey.empty())
577 operation.value().mergeKey = intent.value().command.value() + ":" + path;
578
579 bool started = false;
580 if (!transactionBackend_->active()) {
581 TransactionSpec specification;
582 specification.id = newPropertyTransactionId();
583 specification.label = intent.value().command.value();
584 specification.target = operation.value().target;
585 specification.mergeKey = operation.value().mergeKey;
586 specification.baseRevision = targetRevision_.value();
587 if (specification.id.empty())
588 return property_access::rejected("editor.property.transaction-id",
589 "Could not allocate a property transaction identity");
590 Result<TransactionId> begun = transactionBackend_->begin(std::move(specification));
591 if (!begun.ok() || !begun.ok())
592 return writeFailure(begun, "editor.property.transaction.begin", "Could not begin property transaction");
593 started = true;
594 }
595
596 Result<void> appended = transactionBackend_->append(std::move(operation.value()));
597 if (!appended.ok()) {
598 if (started) {
599 Result<void> discarded = transactionBackend_->discard();
600 if (!discarded.ok())
601 return writeFailure(discarded, "editor.property.transaction.discard",
602 "Could not discard rejected property transaction");
603 }
604 return writeFailure(appended, "editor.property.transaction.append", "Could not stage property operation");
605 }
606 pendingPaths_.insert(path);
607
608 // An already-open transaction is intentionally left staged for the
609 // caller's explicit preview/commit boundary.
610 if (!started) return property_access::accepted();
611
612 Result<EditorDryRunReport> previewed = transactionBackend_->preview();
613 if (!previewed.ok() || !previewed.ok()) {
614 Result<void> discarded = transactionBackend_->discard();
615 pendingPaths_.clear();
616 if (!discarded.ok())
617 return writeFailure(discarded, "editor.property.transaction.discard",
618 "Could not discard failed property preview");
619 return writeFailure(previewed, "editor.property.transaction.preview",
620 "Property transaction preview was rejected");
621 }
622
623 Result<TransactionReceipt> committed = transactionBackend_->commit();
624 if (!committed.ok() || !committed.ok())
625 // Commit failure is deliberately retained by the backend so the
626 // caller can inspect diagnostics, retry, or explicitly discard.
627 return writeFailure(committed, "editor.property.transaction.commit", "Property transaction commit failed");
628 targetRevision_ = eve::Revision(committed.value().afterRevision);
629 pendingPaths_.clear();
630 // The mutation already committed. Do not report a successful commit
631 // as a rejected write; force an explicit refresh if observation fails.
632 if (!refresh().ok()) bound_ = false;
633 return property_access::accepted();
634 }
635
636 if (!sink_) return property_access::rejected("editor.property.sink", "No command sink is connected");
637 // The sink is a compatibility path and may not have an authority that can
638 // perform a commit-time CAS. Recheck immediately before handing it the
639 // intent; stale compatibility writes must fail closed as well.
640 const Result<void> sinkRevisionCheck = ensureCurrentRevision();
641 if (!sinkRevisionCheck.ok())
642 return writeFailure(sinkRevisionCheck, "editor.property.revision", "Property revision is unavailable");
643 const Result<void> applied = sink_(intent.value());
644 if (!applied.ok()) return writeFailure(applied, "editor.property.command", "Property command failed");
645 // A legacy sink may mutate the provider without returning a receipt. Read
646 // the authoritative post-command revision so the next write does not use
647 // the pre-command baseline. If observation fails, retain command success
648 // and require an explicit refresh before another write.
649 if (!refresh().ok()) bound_ = false;
650 return property_access::accepted();
651}

810Result<void> EditorPropertyModel::refresh() {
811 if (!provider_)
812 return eve::editing::failed<void>(EditorStatus::Unsupported, RuleId("editor.property.provider"),
813 "No property provider is connected");
814 if (transactionBackend_ && transactionBackend_->active())
815 return eve::editing::failed<void>(EditorStatus::Conflict, RuleId("editor.property.active-transaction"),
816 "Cannot refresh or rebase while a property transaction is active");
817
818 const Result<eve::Revision> first = readProviderRevision();
819 if (!first.ok()) return Result<void>::failure(first.status());
820
821 std::map<std::string, CachedProperty> nextProperties;
822 for (const property_access::PropertyDescriptor &property : presentationSchema_.properties) {
823 const PropertyReadResult current = provider_->read(selection_, PropertyPath(property.path));
824 if (current.state == PropertyReadState::Error) {
825 if (!current.diagnostics.empty())
826 return Result<void>::failure(eve::Status(EditorStatus::Failed, current.diagnostics));
827 return eve::editing::failed<void>(EditorStatus::Failed, RuleId("editor.property.read"),
828 "Property provider returned a read error");
829 }
830 CachedProperty cached;
831 switch (current.state) {
832 case PropertyReadState::Value:
833 cached.state = property_access::PropertyChangeState::Value;
834 cached.value = toPresentationValue(current.value);
835 break;
836 case PropertyReadState::Mixed: cached.state = property_access::PropertyChangeState::Mixed; break;
837 case PropertyReadState::Missing: cached.state = property_access::PropertyChangeState::Missing; break;
838 case PropertyReadState::Error: break;
839 }
840 nextProperties.emplace(property.path, std::move(cached));
841 }
842
843 const Result<eve::Revision> last = readProviderRevision();
844 if (!last.ok()) return Result<void>::failure(last.status());
845 if (first.value() != last.value()) return externalRevisionConflict(first.value(), last.value());
846
847 std::vector<property_access::PropertyChange> changes;
848 for (const auto &[path, property] : nextProperties) {
849 const auto found = cachedProperties_.find(path);
850 if (found == cachedProperties_.end() || found->second != property)
851 changes.push_back({path, property.value.value_or(eve::Value{}), 0, property.state});
852 }
853
854 if (!changes.empty()) {
855 const std::uint64_t snapshotRevision = ++revision_;
856 for (property_access::PropertyChange &change : changes) change.revision = snapshotRevision;
857 }
858 cachedProperties_ = std::move(nextProperties);
859 targetRevision_ = last.value();
860 bound_ = true;
861 dispatch(std::move(changes));
862 return eve::editing::applied<void>();
863}
864
865void EditorPropertyModel::dispatch(std::vector<property_access::PropertyChange> changes) {
866 if (changes.empty()) return;
867 notificationQueue_.push_back(std::move(changes));
868 if (dispatchingNotifications_) return;
869
870 dispatchingNotifications_ = true;
871 try {
872 while (!notificationQueue_.empty()) {
873 std::vector<property_access::PropertyChange> batch = std::move(notificationQueue_.front());
874 notificationQueue_.erase(notificationQueue_.begin());
875 for (const property_access::PropertyChange &change : batch) {
876 const auto observers = observers_->entries;
877 for (const ObserverState::Entry &entry : observers)
878 if (entry.callback) entry.callback(change);
879 }
880 }
881 dispatchingNotifications_ = false;
882 } catch (...) {
883 notificationQueue_.clear();
884 dispatchingNotifications_ = false;
885 throw;
886 }
887}
