32 const QuestDefinition *def = QuestRegistry::find(rt.id);

61void QuestSystem::syncAuto(Tracker *t) {
62 if (!t) return;
63 // 1. 补建 Registry 里尚未出现在本 Tracker 上的条目。
64 for (const auto &id : QuestRegistry::ids()) {
65 if (t->entries.count(id)) continue;
66 const QuestDefinition *def = QuestRegistry::find(id);
67 if (!def) continue;
68 QuestRuntime rt;
69 rt.id = id;
70 rt.state = "locked";
71 for (const auto &obj : def->objectives) {
72 ObjectiveRuntime orun;
73 orun.id = obj.id;
74 orun.current = 0;
75 orun.count = obj.count > 0 ? obj.count : 1;
76 orun.done = false;
77 rt.objectives.push_back(std::move(orun));
78 }
79 t->entries[id] = std::move(rt);
80 t->order.push_back(id);
81 }
82 // 2. 反复解开前置已齐的 locked 条目。
83 for (size_t pass = 0; pass <= t->entries.size(); ++pass) {
84 bool changed = false;
85 for (auto &[id, rt] : t->entries) {
86 if (rt.state != "locked") continue;
87 const QuestDefinition *def = QuestRegistry::find(id);
88 if (!def) continue;
89 if (!hasMetRequirements(*t, *def)) continue;
90 const std::string newState = stateForUnlocked(*def);
91 rt.state = newState;
92 if (newState == "active") pushEvent(t, {id, "", "activate", "", "", 0, ""});
93 if (newState == "active" && objectivesFull(rt)) onObjectivesFull(t, rt);
94 changed = true;
95 }
96 if (!changed) break;
97 }
98}

100std::string QuestSystem::canActivateReason(Tracker *t, const std::string &id) {
101 if (!t) return "unknown";
102 auto it = t->entries.find(id);
103 if (it == t->entries.end()) return "unknown";
104 const QuestDefinition *def = QuestRegistry::find(id);
105 if (!def) return "unknown";
106 const std::string &state = it->second.state;
107 if (state == "locked") return "locked";
108 if (state == "active") return "alreadyActive";
109 if (state == "completed") return "alreadyCompleted";
110 if (state == "failed") return "failed";
111 if (state == "ready") return "ready";
112 if (!hasMetRequirements(*t, *def)) return "locked";
113 return "";
114}

122bool QuestSystem::activate(Tracker *t, const std::string &id, std::string *reason) {
123 if (!t) {
124 if (reason) *reason = "unknown";
125 return false;
126 }
127 auto it = t->entries.find(id);
128 if (it == t->entries.end()) {
129 if (reason) *reason = "unknown";
130 pushReject(t, id, "unknown");
131 return false;

133 const std::string why = canActivateReason(t, id);
134 if (!why.empty()) {

136 pushReject(t, id, why);
137 return false;

141 if (objectivesFull(it->second)) onObjectivesFull(t, it->second);
142 return true;

145void QuestSystem::notify(Tracker *t, const std::string &topic, const std::string &target, int amount) {
146 if (!t || amount <= 0) return;
147 std::vector<std::string> fullIds;
148 for (auto &[id, rt] : t->entries) {
149 if (rt.state != "active") continue;
150 const QuestDefinition *def = QuestRegistry::find(id);
151 if (!def) continue;
152 const size_t n = std::min(def->objectives.size(), rt.objectives.size());
153 for (size_t i = 0; i < n; ++i) {
154 const auto &objDef = def->objectives[i];
155 auto & rtObj = rt.objectives[i];
156 if (rtObj.done) continue;
157 if (rtObj.count <= 0) rtObj.count = 1;
158 const bool match =
159 objDef.topic == topic && (objDef.target.empty() || objDef.target == target);
160 if (!match) continue;
161 const int added = std::min(amount, rtObj.count - rtObj.current);
162 rtObj.current += added;
163 if (rtObj.current >= rtObj.count) rtObj.done = true;
164 pushEvent(t, {id, rtObj.id, "progress", topic, target, added, ""});
165 }
166 if (objectivesFull(rt)) fullIds.push_back(id);
167 }
168 for (const auto &fid : fullIds) {
169 auto it = t->entries.find(fid);
170 if (it != t->entries.end()) onObjectivesFull(t, it->second);
171 }
172}

174bool QuestSystem::claim(Tracker *t, const std::string &id, std::string *reason) {
175 if (!t) {
176 if (reason) *reason = "unknown";
177 return false;
178 }
179 auto it = t->entries.find(id);
180 if (it == t->entries.end()) {
181 if (reason) *reason = "unknown";
182 pushReject(t, id, "unknown");
183 return false;
184 }
185 if (it->second.state != "ready") {
186 if (reason) *reason = it->second.state;
187 pushReject(t, id, it->second.state);
188 return false;
189 }
190 it->second.state = "completed";
191 pushEvent(t, {id, "", "complete", "", "", 0, ""});
192 syncAuto(t);
193 return true;
194}

196bool QuestSystem::reset(Tracker *t, const std::string &id) {
197 if (!t) return false;
198 auto it = t->entries.find(id);
199 if (it == t->entries.end()) return false; // 未知 id：false，无事件
200 const QuestDefinition *def = QuestRegistry::find(id);
201 QuestRuntime & rt = it->second;
202 rt.objectives.clear();
203 if (def) {
204 for (const auto &obj : def->objectives) {
205 ObjectiveRuntime orun;
206 orun.id = obj.id;
207 orun.current = 0;
208 orun.count = obj.count > 0 ? obj.count : 1;
209 orun.done = false;
210 rt.objectives.push_back(std::move(orun));
211 }
212 }
213 rt.state = (def && hasMetRequirements(*t, *def)) ? stateForUnlocked(*def) : std::string("locked");
214 pushEvent(t, {id, "", "reset", "", "", 0, ""});
215 if (rt.state == "active" && objectivesFull(rt)) onObjectivesFull(t, rt);
216 return true;
217}

221 if (!t) return false;
