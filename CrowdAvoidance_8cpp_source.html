30void Crowd::Impl::prepareAvoidanceNeighbors() {
31 EV_PROFILE_MODULE("crowd", "Crowd::prepareAvoidanceNeighbors");
32 // Arrival/settled tests depend only on a neighbour's own state, not the querying agent.
33 std::vector<std::uint8_t> settled(actions.size(), 0);
34 for (size_t j = 0; j < actions.size(); ++j) {
35 if (interactions[j].holdPosition || interactions[j].pushability <= 0.f) continue;
36 settled[j] = actions[j] == 0 || (actions[j] == seekAction && hasTargets[j] &&
37 std::hypot(targetXs[j] - xs[j], targetYs[j] - ys[j]) <= arriveRadius &&
38 std::hypot(vxs[j], vys[j]) <= maxSpeeds[j] * 0.05f);
39 }
40 predictionNeighbors.resize(actions.size());
41 const auto prepareRange = [&](size_t first, size_t last, StepReport* partitionReport, std::int64_t& checks,
42 int& truncations) {
43 for (size_t i = first; i < last; ++i) {
44 using Neighbor = AvoidanceNeighbor;
45 auto& neighbors = predictionNeighbors[i];
46 neighbors.clear();
47 if (maxSpeeds[i] <= 0.f || interactions[i].holdPosition) continue;
48 neighbors.reserve(size_t(avoidance.maxNeighbors) + 1);
49 const double query = double(radii[i]) + avoidanceMaxRadius + avoidance.margin +
50 double(avoidance.horizon) * (double(maxSpeeds[i]) + avoidanceMaxSpeed);
51 const bool arriving = resolveOverlaps && actions[i] == seekAction && hasTargets[i];
52 int encountered = 0;
53 const auto nearer = [&](const Neighbor& a, const Neighbor& b) {
54 if (a.distance2 != b.distance2) return a.distance2 < b.distance2;
55 return tieKeys[a.slot] < tieKeys[b.slot];
56 };
57 const double queryRadius = float(std::min(query, double(std::numeric_limits<float>::max())));
58 forEachNeighbor(
59 xs[i], ys[i], float(queryRadius),
60 [&](int slot) {
61 const size_t j = size_t(slot);
62 if (i == j || !canAvoid(i, j)) return;
63 // Settled, pushable neighbours yield through the contact solver during final arrival.
64 // Held bodies and pairs without physical contact still require predictive avoidance.
65 if (arriving && (settled[j] || maxSpeeds[j] <= maxSpeeds[i] * 0.05f) &&
66 !interactions[j].holdPosition && interactions[j].pushability > 0.f && canContact(i, j))
67 return;
68 ++encountered;
69 const double dx = double(xs[j]) - xs[i], dy = double(ys[j]) - ys[i];
70 Neighbor neighbor{j, dx * dx + dy * dy};
71 if (neighbors.size() == size_t(avoidance.maxNeighbors) && !nearer(neighbor, neighbors.back()))
72 return;
73 const auto position = std::lower_bound(neighbors.begin(), neighbors.end(), neighbor, nearer);
74 if (position != neighbors.end() || neighbors.size() < size_t(avoidance.maxNeighbors)) {
75 neighbors.insert(position, neighbor);
76 if (neighbors.size() > size_t(avoidance.maxNeighbors)) neighbors.pop_back();
77 }
78 },
79 partitionReport);
80 if (encountered > avoidance.maxNeighbors) ++truncations;
81 checks += static_cast<std::int64_t>(neighbors.size()) * scoresPerNeighbor;
82 }
83 };
84 std::int64_t checks = 0;
85 if (execution == CrowdExecution::Serial) {
86 prepareRange(0, actions.size(), nullptr, checks, avoidanceTruncations);
87 } else {
88 constexpr size_t grain = 256;
89 struct Partition {
90 StepReport report;
91 std::int64_t checks = 0;
92 int truncations = 0;
93 std::exception_ptr error;
94 };
95 std::vector<Partition> partitions((actions.size() + grain - 1) / grain);
96 dispatchWork(actions.size(), [&](size_t first, size_t last) {
97 auto& partition = partitions[first / grain];
98 try {
99 prepareRange(first, last, &partition.report, partition.checks, partition.truncations);
100 } catch (...) {
101 partition.error = std::current_exception();
102 }
103 });
104 for (const auto& partition : partitions) {
105 if (partition.error) std::rethrow_exception(partition.error);
106 if (budgetActive &&
107 partition.report.spatialVisits > activeBudget.maxSpatialVisits - workReport.spatialVisits)
108 throw WorkLimit{"Crowd spatial lookup budget exhausted"};
109 workReport.spatialVisits += partition.report.spatialVisits;
110 workReport.cellsVisited += partition.report.cellsVisited;
111 if (budgetActive &&
112 partition.report.candidatesVisited > activeBudget.maxNeighborVisits - workReport.candidatesVisited)
113 throw WorkLimit{"Crowd neighbor visit budget exhausted"};
114 workReport.candidatesVisited += partition.report.candidatesVisited;
115 avoidanceTruncations += partition.truncations;
116 checks += partition.checks;
117 }
118 }
119 if (budgetActive && checks > activeBudget.maxPredictionChecks - avoidanceChecks)
120 throw WorkLimit{"Crowd prediction check budget exhausted"};
121 avoidanceChecks += checks;
122}

124void Crowd::Impl::selectAvoidanceVelocity(size_t i, float dt, float& vx, float& vy) {
125 const auto& neighbors = predictionNeighbors[i];
126 if (neighbors.empty()) return;
127
128
129 const float speed = maxSpeeds[i];
130 const float preferredX = preferredVxs[i], preferredY = preferredVys[i];
131 const float preferredLength = std::hypot(preferredX, preferredY);
132 const float baseAngle = preferredLength > 1e-5f ? std::atan2(preferredY, preferredX) : headings[i];
133 const float acceleration = maxAccels[i] * dt;
134 const auto reachable = [&](float& x, float& y) {
135 const float changeX = x - vxs[i], changeY = y - vys[i];
136 const float change = std::hypot(changeX, changeY);
137 if (change > acceleration && change > 0.f) {
138 x = vxs[i] + changeX * acceleration / change;
139 y = vys[i] + changeY * acceleration / change;
140 }
141 const float length = std::hypot(x, y);
142 if (length > speed) {
143 x *= speed / length;
144 y *= speed / length;
145 }
146 };
147 const auto score = [&](float x, float y, double cutoff) {
148 double cost =
149 std::hypot(x - preferredX, y - preferredY) / speed + 0.15 * std::hypot(x - vxs[i], y - vys[i]) / speed;
150 // A small, persistent right-hand preference breaks mirror symmetry.
151 if (preferredLength > 1e-5f)
152 cost += 0.05 * std::max(0.f, preferredX * y - preferredY * x) / (preferredLength * speed);
153 // Remaining terms can only add cost. A candidate already above the winner cannot improve it.
154 if (cost >= cutoff) return cost;
155 for (const auto& neighbor : neighbors) {
156 const size_t j = neighbor.slot;
157 const double px = double(xs[j]) - xs[i], py = double(ys[j]) - ys[i];
158 const double dx = double(x) - vxs[j], dy = double(y) - vys[j];
159 const double radius = double(radii[i]) + radii[j] + avoidance.margin;
160 const double c = neighbor.distance2 - radius * radius;
161 const double a = dx * dx + dy * dy, b = px * dx + py * dy;
162 double collision = double(avoidance.horizon) + 1.0;
163 if (c < 0.0) {
164 // Prefer opening an existing contact, rather than freezing all candidates.
165 collision = 0.0;
166 cost += 2.0 * std::max(0.0, b) / (std::max(radius, 1e-6) * speed);
167 } else if (a > 1e-10 && b > 0.0) {
168 const double discriminant = b * b - a * c;
169 if (discriminant >= 0.0) collision = (b - std::sqrt(discriminant)) / a;
170 }
171 if (collision >= 0.0 && collision < avoidance.horizon) {
172 const double urgency = 1.0 - collision / avoidance.horizon;
173 // Priority adjusts yielding preference; it never removes collision cost.
174 const double priority =
175 std::clamp(double(avoidancePriorities[j]) - avoidancePriorities[i], -100.0, 100.0);
176 cost += (20.0 + 0.1 * priority) * urgency * urgency;
177 }
178 if (cost >= cutoff) return cost;
179 }
180 if (field.valid()) {
181 float px = xs[i] + x * dt, py = ys[i] + y * dt;
182 float projectedX = px, projectedY = py;
183 field.resolvePenetration(projectedX, projectedY, radii[i]);
184 if (std::hypot(projectedX - px, projectedY - py) > 1e-4f) cost += 100.0;
185 if (clampToField && (px - radii[i] < field.getOriginX() || py - radii[i] < field.getOriginY() ||
186 px + radii[i] > field.getOriginX() + float(field.getWidth()) * field.getCellSize() ||
187 py + radii[i] > field.getOriginY() + float(field.getHeight()) * field.getCellSize()))
188 cost += 100.0;
189 }
190 return cost;
191 };
192 double best = score(vx, vy, std::numeric_limits<double>::infinity());
193 // Every score term is non-negative; a zero-cost current choice cannot be improved.
194 if (best <= 1e-9) return;
195 const auto consider = [&](float x, float y) {
196 reachable(x, y);
197 const double value = score(x, y, best - 1e-9);
198 if (value < best - 1e-9) {
199 best = value;
200 vx = x;
201 vy = y;
202 }
203 };
204 consider(0.f, 0.f);
205 consider(vxs[i], vys[i]);
206 const float samplingSpeed = std::max(preferredLength, speed * 0.25f);
207 for (float fraction : predictionFractions)
208 for (float angle : predictionAngles)
209 consider(std::cos(baseAngle + angle) * samplingSpeed * fraction,
210 std::sin(baseAngle + angle) * samplingSpeed * fraction);
211}
