15Result<void> ActionBlockRuntime::apply(const ActionAdvance& advance, ActionNotifyContext context) {
16 if (context.executionId != advance.id) return invalidContext();
17 context.time = advance.totalElapsed;
18 for (const auto& event : advance.timelineEvents) {
19 auto dispatched = registry_.dispatch(event, context);
20 if (!dispatched) return dispatched;
21 }
22 for (const auto& block : advance.activeBlocks) {
23 auto updated = registry_.dispatchUpdate(block, context);
24 if (!updated) return updated;
25 }
26 auto maintained = registry_.advanceHandlers(context);
27 if (!maintained) return maintained;
28 if (advance.activeBlocks.empty())
29 active_.erase(advance.id);

31 active_[advance.id] = advance.activeBlocks;

35Result<void> ActionBlockRuntime::interrupt(ActionNotifyContext context) {
36 const auto found = active_.find(context.executionId);
37 if (found == active_.end()) return Result<void>::success(Status::success(StatusCode::NoOp));
38 context.interrupted = true;
39 auto& blocks = found->second;
40 while (!blocks.empty()) {
41 const auto& block = blocks.front();
42 ActionTimelineEvent event{ActionTimelineEventKind::StateExit, block.trackId, block.itemId, block.type,
43 context.time, block.payload};
44 auto dispatched = registry_.dispatch(event, context);
45 if (!dispatched) return dispatched;
46 blocks.erase(blocks.begin());
47 }
48 active_.erase(found);
49 return Result<void>::success(Status::success(StatusCode::Applied));
50}
