44void normalize3(float &x, float &y, float &z) {
45 const float len = std::sqrt(x * x + y * y + z * z);
46 if (len <= 1e-8f) {
47 x = 0.f;
48 y = 1.f;
49 z = 0.f;
50 return;
51 }
52 x /= len;
53 y /= len;
54 z /= len;
55}

66CardMeshData buildCard(float width, float height) {
67 CardMeshData mesh;
68 const float w = width > 1e-4f ? width : 1e-4f;
69 const float h = height > 1e-4f ? height : 1e-4f;
70 const float hw = 0.5f * w;
71 pushQuad(mesh, -hw, 0.f, 0.f, hw, 0.f, 0.f, hw, h, 0.f, -hw, h, 0.f, 0.f, 0.f, 1.f);

75CardMeshData buildCardsAlongPolyline(const float *pointsXYZ, int pointCount, float width) {
76 CardMeshData mesh;
77 if (!pointsXYZ || pointCount < 2) return mesh;
78 const float w = width > 1e-4f ? width : 1e-4f;
79 const float hw = 0.5f * w;
80 for (int i = 0; i + 1 < pointCount; ++i) {
81 const float *a = pointsXYZ + i * 3;
82 const float *b = pointsXYZ + (i + 1) * 3;
83 float dx = b[0] - a[0], dy = b[1] - a[1], dz = b[2] - a[2];
84 normalize3(dx, dy, dz);
85 float sx = 0.f, sy = 0.f, sz = 0.f;
86 cross3(dx, dy, dz, 0.f, 1.f, 0.f, sx, sy, sz);
87 if (sx * sx + sy * sy + sz * sz < 1e-8f)
88 cross3(dx, dy, dz, 1.f, 0.f, 0.f, sx, sy, sz);
89 normalize3(sx, sy, sz);
90 float nx = 0.f, ny = 0.f, nz = 0.f;
91 cross3(sx, sy, sz, dx, dy, dz, nx, ny, nz);
92 normalize3(nx, ny, nz);
93 pushQuad(mesh, a[0] - sx * hw, a[1] - sy * hw, a[2] - sz * hw, a[0] + sx * hw,
94 a[1] + sy * hw, a[2] + sz * hw, b[0] + sx * hw, b[1] + sy * hw, b[2] + sz * hw,
95 b[0] - sx * hw, b[1] - sy * hw, b[2] - sz * hw, nx, ny, nz);
96 }
97 return mesh;
98}

100CardMeshData buildCardsFromRoots(const CardRoot *roots, int rootCount) {
101 CardMeshData mesh;
102 if (!roots || rootCount <= 0) return mesh;
103 for (int i = 0; i < rootCount; ++i) {
104 const CardRoot &r = roots[i];
105 float dx = r.dirX, dy = r.dirY, dz = r.dirZ;
106 normalize3(dx, dy, dz);
107 const float len = r.length > 1e-4f ? r.length : 1e-4f;
108 const float tip[3] = {r.x + dx * len, r.y + dy * len, r.z + dz * len};
109 const float pts[6] = {r.x, r.y, r.z, tip[0], tip[1], tip[2]};
110 appendCardMesh(mesh, buildCardsAlongPolyline(pts, 2, r.width));
111 }
112 return mesh;
113}
