46void normalize3(float &x, float &y, float &z) {
47 const float len = std::sqrt(x * x + y * y + z * z);
48 if (len <= 1e-8f) {
49 x = 0.f;
50 y = 1.f;
51 z = 0.f;
52 return;
53 }
54 x /= len;
55 y /= len;
56 z /= len;
57}

59void cross3(float ax, float ay, float az, float bx, float by, float bz, float &ox, float &oy, float &oz) {
60 ox = ay * bz - az * by;
61 oy = az * bx - ax * bz;
62 oz = ax * by - ay * bx;

67CardMeshData buildCard(float width, float height) {
68 CardMeshData mesh;
69 const float w = width > 1e-4f ? width : 1e-4f;
70 const float h = height > 1e-4f ? height : 1e-4f;
71 const float hw = 0.5f * w;
72 pushQuad(mesh, -hw, 0.f, 0.f, hw, 0.f, 0.f, hw, h, 0.f, -hw, h, 0.f, 0.f, 0.f, 1.f);
73 return mesh;
74}

76CardMeshData buildCardsAlongPolyline(const float *pointsXYZ, int pointCount, float width) {
77 CardMeshData mesh;
78 if (!pointsXYZ || pointCount < 2) return mesh;
79 const float w = width > 1e-4f ? width : 1e-4f;
80 const float hw = 0.5f * w;
81 for (int i = 0; i + 1 < pointCount; ++i) {
82 const float *a = pointsXYZ + i * 3;
83 const float *b = pointsXYZ + (i + 1) * 3;
84 float dx = b[0] - a[0], dy = b[1] - a[1], dz = b[2] - a[2];
85 normalize3(dx, dy, dz);
86 float sx = 0.f, sy = 0.f, sz = 0.f;
87 cross3(dx, dy, dz, 0.f, 1.f, 0.f, sx, sy, sz);
88 if (sx * sx + sy * sy + sz * sz < 1e-8f) cross3(dx, dy, dz, 1.f, 0.f, 0.f, sx, sy, sz);
89 normalize3(sx, sy, sz);
90 float nx = 0.f, ny = 0.f, nz = 0.f;
91 cross3(sx, sy, sz, dx, dy, dz, nx, ny, nz);
92 normalize3(nx, ny, nz);
93 pushQuad(mesh, a[0] - sx * hw, a[1] - sy * hw, a[2] - sz * hw, a[0] + sx * hw, a[1] + sy * hw, a[2] + sz * hw,
94 b[0] + sx * hw, b[1] + sy * hw, b[2] + sz * hw, b[0] - sx * hw, b[1] - sy * hw, b[2] - sz * hw, nx, ny,
95 nz);
96 }
97 return mesh;
98}

100CardMeshData buildCardsFromRoots(const CardRoot *roots, int rootCount) {
101 CardMeshData mesh;
102 if (!roots || rootCount <= 0) return mesh;
103 for (int i = 0; i < rootCount; ++i) {
104 const CardRoot &r = roots[i];
105 float dx = r.dirX, dy = r.dirY, dz = r.dirZ;
106 normalize3(dx, dy, dz);
107 const float len = r.length > 1e-4f ? r.length : 1e-4f;
108 const float tip[3] = {r.x + dx * len, r.y + dy * len, r.z + dz * len};
109 const float pts[6] = {r.x, r.y, r.z, tip[0], tip[1], tip[2]};
110 appendCardMesh(mesh, buildCardsAlongPolyline(pts, 2, r.width));
111 }
112 return mesh;
113}

115void appendCardMesh(CardMeshData &dst, const CardMeshData &src) {
116 const uint32_t base = static_cast<uint32_t>(dst.positions.size() / 3);
117 dst.positions.insert(dst.positions.end(), src.positions.begin(), src.positions.end());
118 dst.normals.insert(dst.normals.end(), src.normals.begin(), src.normals.end());
119 dst.uvs.insert(dst.uvs.end(), src.uvs.begin(), src.uvs.end());
120 for (uint32_t idx : src.indices) dst.indices.push_back(base + idx);
121}
