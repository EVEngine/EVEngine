14std::string trim(std::string value) {
15 const auto first = std::find_if_not(value.begin(), value.end(), [](unsigned char c) {
16 return std::isspace(c) != 0;
17 });
18 const auto last = std::find_if_not(value.rbegin(), value.rend(), [](unsigned char c) {

20 }).base();
21 return first < last ? std::string(first, last) : std::string{};

24std::vector<std::string> words(const std::string& line) {
25 std::vector<std::string> out;
26 std::string word;
27 bool quoted = false;
28 bool escaped = false;
29 for (char c : line) {
30 if (escaped) {
31 word.push_back(c);
32 escaped = false;
33 } else if (c == '\\' && quoted) {
34 escaped = true;
35 } else if (c == '"') {
36 quoted = !quoted;
37 } else if (std::isspace(static_cast<unsigned char>(c)) && !quoted) {
38 if (!word.empty()) {
39 out.push_back(std::move(word));
40 word.clear();
41 }
42 } else {
43 word.push_back(c);
44 }
45 }
46 if (!word.empty()) out.push_back(std::move(word));
47 return out;
48}

85bool compileDnutConversations(const std::string& source, const std::string& path,
86 std::vector<ConversationAsset>& assets,
87 std::vector<ConversationDiagnostic>& diagnostics) {
88 assets.clear();
89 ConversationAsset* asset = nullptr;
90 ConversationAsset::Node* node = nullptr;
91 std::istringstream input(source);
92 std::string raw;
93 int lineNumber = 0;
94 const auto error = [&](const std::string& message) {
95 diagnostics.push_back(
96 {ConversationDiagnostic::Severity::Error, path, lineNumber, message});
97 };
98 while (std::getline(input, raw)) {
99 ++lineNumber;
100 const size_t comment = raw.find("//");
101 const std::string line = trim(raw.substr(0, comment));
102 if (line.empty()) continue;
103 const auto tokens = words(line);
104 if (tokens.empty()) continue;
105 if (tokens[0] == "conversation") {
106 if (tokens.size() < 2) {
107 error("conversation requires an id");
108 continue;
109 }
110 assets.push_back({});
111 asset = &assets.back();
112 asset->id = tokens[1];
113 const auto attrs = attributes(tokens, 2);
114 if (const auto it = attrs.find("entry"); it != attrs.end()) asset->entry = it->second;
115 if (const auto it = attrs.find("version"); it != attrs.end()) {
116 int version = 0;
117 std::from_chars(it->second.data(), it->second.data() + it->second.size(), version);
118 asset->version = version;
119 }
120 if (const auto it = attrs.find("params"); it != attrs.end()) {
121 std::istringstream params(it->second);
122 std::string parameter;
123 while (std::getline(params, parameter, ',')) asset->parameters.push_back(parameter);
124 }
125 node = nullptr;
126 } else if (tokens[0] == "endconversation") {
127 asset = nullptr;
128 node = nullptr;
129 } else if (!asset) {
130 // Pool syntax is compiled by DnutParser; conversation blocks may coexist in one file.
131 continue;
132 } else if (tokens[0] == "node") {
133 if (!asset || tokens.size() < 3) {
134 error("node requires an active conversation, id, and kind");
135 continue;
136 }
137 bool validKind = false;
138 asset->nodes.push_back({});
139 node = &asset->nodes.back();
140 node->id = tokens[1];
141 node->kind = nodeKind(tokens[2], validKind);
142 if (!validKind) error("unknown node kind '" + tokens[2] + "'");
143 const auto attrs = attributes(tokens, 3);
144 const auto set = [&](const char* name, std::string& value) {
145 if (const auto it = attrs.find(name); it != attrs.end()) value = it->second;
146 };
147 set("next", node->next);
148 set("speaker", node->speaker);
149 set("text", node->text);
150 set("pool", node->pool);
151 set("i18n", node->i18nKey);
152 set("voice", node->voice);
153 set("target", node->target);
154 set("return", node->returnNode);
155 set("result", node->expression);
156 } else if (tokens[0] == "when" || tokens[0] == "else" || tokens[0] == "option") {
157 if (!node) {
158 error("route requires a preceding node");
159 continue;
160 }
161 const size_t arrow = line.rfind("->");
162 if (arrow == std::string::npos) {
163 error("route requires '-> target'");
164 continue;
165 }
166 const std::string target = trim(line.substr(arrow + 2));
167 std::string expression;
168 if (tokens[0] == "else") expression = "else";
169 else expression = trim(line.substr(tokens[0].size(), arrow - tokens[0].size()));
170 node->routes.emplace_back(std::move(expression), target);
171 } else {
172 error("unexpected conversation statement '" + tokens[0] + "'");
173 }
174 }
175 if (asset) error("conversation is missing endconversation");
176 return lintConversations(assets, path, diagnostics);
177}

179bool lintConversations(const std::vector<ConversationAsset>& assets, const std::string& path,
180 std::vector<ConversationDiagnostic>& diagnostics) {
181 bool valid = true;
182 std::unordered_set<std::string> assetIds;
183 for (const auto& asset : assets) {
184 if (!assetIds.insert(asset.id).second) {
185 diagnostics.push_back({ConversationDiagnostic::Severity::Error, path, 0,
186 "duplicate conversation id '" + asset.id + "'"});
187 valid = false;
188 }
189 std::string error;
190 if (!asset.validate(&error)) {
191 diagnostics.push_back({ConversationDiagnostic::Severity::Error, path, 0, error});
192 valid = false;
193 continue;
194 }
195 std::unordered_set<std::string> reached;
196 std::deque<std::string> pending{asset.entry};
197 while (!pending.empty()) {
198 const std::string id = pending.front();
199 pending.pop_front();
200 if (!reached.insert(id).second) continue;
201 const auto* node = asset.findNode(id);
202 if (!node) continue;
203 if (!node->next.empty()) pending.push_back(node->next);
204 if (!node->returnNode.empty()) pending.push_back(node->returnNode);
205 for (const auto& route : node->routes) pending.push_back(route.second);
206 }
207 for (const auto& node : asset.nodes) {
208 if (reached.find(node.id) == reached.end())
209 diagnostics.push_back({ConversationDiagnostic::Severity::Warning, path, 0,
210 "conversation '" + asset.id + "': unreachable node '" +
211 node.id + "'"});
212 }
213 }
214 return valid && std::none_of(diagnostics.begin(), diagnostics.end(), [](const auto& item) {
215 return item.severity == ConversationDiagnostic::Severity::Error;
216 });
217}
