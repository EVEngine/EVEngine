24EditorResult<std::vector<AssetRecord>> publishEvaAssetProjection(
25 MemoryAssetDatabase& database, const asset::EvaManifest& manifest, std::string archiveUri,
26 std::string importerId) {
27 if (archiveUri.empty() || importerId.empty())
28 return eve::editing::failed<std::vector<AssetRecord>>(
29 EditorStatus::Rejected, RuleId("editor.asset.eva-projection.invalid-source"),
30 "Eva projection requires an archive URI and importer identity");
31 std::vector<AssetPublication> publications;
32 publications.reserve(manifest.assets.size());
33 for (const asset::EvaAssetEntry& source : manifest.assets) {
34 AssetPublication publication;
35 publication.record.guid = guid(source.asset);
36 publication.record.logicalUri = source.asset.format();
37 publication.record.typeId = source.type;
38 publication.record.schemaVersion = static_cast<std::uint32_t>(source.schemaVersion.value());
39 publication.record.sourceUri = archiveUri + "#" + source.definition;
40 publication.record.sourceHash = source.contentHash;
41 publication.record.importerId = importerId;
42 publication.record.importerVersion = 1;
43 publication.record.tags = source.tags;
44 publication.record.metadata = EditorValue::Object{
45 {"packageId", EditorValue(manifest.packageId.format())},
46 {"packageName", EditorValue(manifest.packageName)},
47 {"packageVersion", EditorValue(manifest.packageVersion)},
48 {"definition", EditorValue(source.definition)},
49 };
50 for (const asset::EvaDependency& sourceDependency : manifest.dependencies) {
51 if (sourceDependency.from != source.asset) continue;
52 publication.dependencies.push_back(
53 {guid(sourceDependency.from), guid(sourceDependency.to), projectionKind(sourceDependency.kind),
54 PropertyPath(sourceDependency.path.empty() ? "dependency" : sourceDependency.path)});
55 }
56 std::sort(publication.dependencies.begin(), publication.dependencies.end(),
57 [](const AssetDependency& left, const AssetDependency& right) {
58 if (left.to != right.to) return left.to < right.to;
59 return left.sourceProperty < right.sourceProperty;
60 });
61 publications.push_back(std::move(publication));
62 }
63 return database.publishBatch(std::move(publications));
64}
