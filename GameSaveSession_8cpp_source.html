40Result<std::string> GameSaveSession::snapshotJson(const SnapshotHashProvider& hashProvider) const {
41 if (contentVersion_.empty() || participants_.empty())
42 return Result<std::string>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
43 "game save requires content version and participants"));
44 Value::Object states;
45 for (const auto& [key, participant] : participants_) {
46 auto state = participant->capture();
47 if (!state.ok()) return Result<std::string>::failure(state.status());
48 states.emplace(key, std::move(state).takeValue());
49 }
50 Value::Object payload{{"contentVersion", Value(contentVersion_)}, {"participants", Value(std::move(states))}};
51 auto envelope = makeSnapshotEnvelope("game.save-session", gameSaveSchema(), SchemaVersion(1), PersistentId::nil(),
52 Revision(0), SimulationTick(0), Value(std::move(payload)), hashProvider);
53 if (!envelope.ok()) return Result<std::string>::failure(envelope.status());
54 return serializeSnapshotEnvelope(envelope.value());
55}

57Result<void> GameSaveSession::restoreJson(std::string_view json, const SnapshotHashProvider& hashProvider) {
58 if (contentVersion_.empty() || participants_.empty())
59 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
60 "game save requires content version and participants"));
61 auto decoded = parseSnapshotEnvelope(json, hashProvider);
62 if (!decoded.ok()) return Result<void>::failure(decoded.status());
63 const auto& envelope = decoded.value();
64 if (envelope.type != "game.save-session" || envelope.schema != gameSaveSchema())
65 return Result<void>::failure(
66 Diagnostic::error(DiagnosticCode::InvalidArgument, "snapshot is not a game save", "schema"));
67 if (envelope.schemaVersion != SchemaVersion(1))
68 return Result<void>::failure(
69 Diagnostic::error(DiagnosticCode::UnknownVersion, "unknown game-save schema version", "schemaVersion"));
70 const auto* payload = envelope.payload.getIf<Value::Object>();
71 if (!payload || payload->size() != 2)
72 return Result<void>::failure(
73 Diagnostic::error(DiagnosticCode::ParseError, "game-save payload requires exactly two fields", "payload"));
74 const Value* savedVersion = envelope.payload.find("contentVersion");
75 const Value* savedStates = envelope.payload.find("participants");
76 if (!savedVersion || !savedVersion->isString() || !savedStates || !savedStates->isObject())
77 return Result<void>::failure(
78 Diagnostic::error(DiagnosticCode::ParseError, "invalid game-save payload", "payload"));
79 Value migratedStates;
80 if (savedVersion->asString() != contentVersion_) {
81 const auto migration = migrations_.find(savedVersion->asString());
82 if (migration == migrations_.end())
83 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Conflict,
84 "game-save content version has no migration route",
85 "payload.contentVersion"));
86 auto migrated = migration->second(*savedStates);
87 if (!migrated.ok()) return Result<void>::failure(migrated.status());
88 migratedStates = std::move(migrated).takeValue();
89 savedStates = &migratedStates;
90 }
91 const auto* states = savedStates->getIf<Value::Object>();
92 if (!states || states->size() != participants_.size())
93 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Conflict,
94 "game-save participant set differs from the current game",
95 "payload.participants"));
96 std::vector<std::unique_ptr<IPreparedGameRestore>> prepared;
97 prepared.reserve(participants_.size());
98 for (const auto& [key, participant] : participants_) {
99 const auto state = states->find(key);
100 if (state == states->end())
101 return Result<void>::failure(
102 Diagnostic::error(DiagnosticCode::Unsupported, "game-save participant is absent", key));
103 auto candidate = participant->prepare(state->second);
104 if (!candidate.ok()) return Result<void>::failure(candidate.status());
105 if (!candidate.value())
106 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
107 "game-save participant returned no prepared state", key));
108 prepared.push_back(std::move(candidate).takeValue());
109 }
110 for (auto& candidate : prepared) candidate->commit();
111 return Result<void>::success(Status::success(StatusCode::Applied));
112}
