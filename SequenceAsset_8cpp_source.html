29eve::Result<void> SequenceAsset::validate() const {
30 if (id.empty())
31 return assetFailure(eve::DiagnosticCode::InvariantViolation, "sequence asset has no id", "id");
32 if (entry.empty())
33 return assetFailure(eve::DiagnosticCode::InvariantViolation, "sequence asset has no entry node", "entry");
34 if (version < 1)
35 return assetFailure(eve::DiagnosticCode::InvariantViolation, "sequence asset version must be positive",
36 "version");
37
38 std::unordered_set<std::string> ids;
39 for (const auto& node : nodes) {
40 if (node.id.empty())
41 return assetFailure(eve::DiagnosticCode::InvariantViolation, "sequence node has an empty id", "nodes");
42 if (!ids.insert(node.id).second)
43 return assetFailure(eve::DiagnosticCode::AlreadyExists, "duplicate sequence node id '" + node.id + "'",
44 "nodes." + node.id);
45 if (node.type.empty())
46 return assetFailure(eve::DiagnosticCode::InvariantViolation,
47 "sequence node '" + node.id + "' has no type", "nodes." + node.id + ".type");
48 if (!node.payload.isObject())
49 return assetFailure(eve::DiagnosticCode::InvariantViolation,
50 "sequence node '" + node.id + "' payload must be an object",
51 "nodes." + node.id + ".payload");
52 }
53
54 if (!findNode(entry))
55 return assetFailure(eve::DiagnosticCode::NotFound, "entry node '" + entry + "' does not exist", "entry");

57 const auto checkReference = [&](const std::string& owner, const std::string& field,
58 const std::string& reference) -> eve::Result<void> {
59 if (reference.empty() || findNode(reference)) return eve::Result<void>::success();
60 return assetFailure(eve::DiagnosticCode::NotFound,
61 "sequence node '" + owner + "' references missing node '" + reference + "'",
62 "nodes." + owner + "." + field);
63 };
64
65 for (const auto& node : nodes) {
66 auto nextResult = checkReference(node.id, "next", node.next);
67 if (!nextResult.ok()) return nextResult;
68
69 if (node.type == "call") {
70 const eve::Value* target = node.payload.find("target");
71 if (!target || !target->isString() || target->asString().empty())
72 return assetFailure(eve::DiagnosticCode::InvariantViolation,
73 "call node '" + node.id + "' requires a target asset id",
74 "nodes." + node.id + ".payload.target");
75 }
76 if (node.type == "choice") {
77 if (node.routes.empty())
78 return assetFailure(eve::DiagnosticCode::InvariantViolation,
79 "choice node '" + node.id + "' requires at least one route",
80 "nodes." + node.id + ".routes");
81 std::unordered_set<std::string> labels;
82 for (const auto& route : node.routes) {
83 if (route.label.empty())
84 return assetFailure(eve::DiagnosticCode::InvariantViolation,
85 "choice node '" + node.id + "' has a route without a label",
86 "nodes." + node.id + ".routes");
87 if (!labels.insert(route.label).second)
88 return assetFailure(eve::DiagnosticCode::AlreadyExists,
89 "choice node '" + node.id + "' has duplicate route label '" + route.label + "'",
90 "nodes." + node.id + ".routes");
91 }
92 }
93 if (node.type == "branch") {
94 if (node.routes.empty())
95 return assetFailure(eve::DiagnosticCode::InvariantViolation,
96 "branch node '" + node.id + "' requires at least one route",
97 "nodes." + node.id + ".routes");
98 for (const auto& route : node.routes) {
99 if (route.condition.isNull())
100 return assetFailure(eve::DiagnosticCode::InvariantViolation,
101 "branch node '" + node.id + "' has a route without a condition",
102 "nodes." + node.id + ".routes");
103 }
104 }
105 for (const auto& route : node.routes) {
106 auto routeResult = checkReference(node.id, "routes", route.target);
107 if (!routeResult.ok()) return routeResult;
108 }
109 }
110 return eve::Result<void>::success();
111}
