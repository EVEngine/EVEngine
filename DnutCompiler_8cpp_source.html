134 void synchronize() {
135 int depth = 0;
136 bool consumed = false;
137 while (!atEnd()) {
138 if (depth == 0) {
139 if (isPunct("}")) return;
140 if (consumed && cur().kind == DnutTokenKind::Identifier && cur().line > errorLine_) return;
141 }
142 if (isPunct("{")) {
143 ++depth;
144 } else if (isPunct("}")) {
145 if (depth == 0) return;
146 --depth;
147 }
148 advance();
149 consumed = true;
150 }
151 }

155 int depth = 0;
156 while (!atEnd()) {
157 if (depth == 0 && (isIdentifier("story") || isIdentifier("common"))) return;
158 if (isPunct("{")) {
159 ++depth;
160 } else if (isPunct("}")) {
161 if (depth > 0) --depth;
162 }
163 advance();
164 }

167 void parseStory() {
168 const bool isCommon = isIdentifier("common");
169 advance(); // 'story' | 'common'
170 invalidStep_ = false;
171 const std::string id = expectIdentifier(isCommon ? "a common-event id" : "a story id");
172 bool repeatable = isCommon; // common events are reusable by default
173 int version = 1;
174 while (cur().kind == DnutTokenKind::Identifier) {
175 const std::string modifier = cur().text;
176 if (modifier == "repeatable") {
177 repeatable = true;
178 advance();
179 continue;
180 }
181 if (modifier == "version") {
182 advance();
183 expectPunct("=");
184 if (cur().kind != DnutTokenKind::Number) errorHere("version requires an integer");
185 version = static_cast<int>(std::strtol(advance().text.c_str(), nullptr, 10));
186 if (version < 1) errorHere("version must be positive");
187 continue;
188 }
189 errorHere(std::string("unknown ") + (isCommon ? "common" : "story") + " modifier '" + modifier + "'");
190 }
191
192 std::vector<CompiledStatement> body;
193 parseBlockBody(body);
194
195 SequenceAsset asset;
196 asset.id = id;
197 asset.version = version;
198 asset.repeatable = repeatable;
199 const std::string endId = nextNodeId();
200 SequenceNode endNode;
201 endNode.id = endId;
202 endNode.type = "end";
203 loopStack_.clear();
204 pendingJumps_.clear();
205 labels_.clear();
206 const std::string entry = emitBlock(body, endId, asset);
207 asset.entry = entry.empty() ? endId : entry;
208 asset.nodes.push_back(std::move(endNode));
209 resolveJumps(asset);
210
211 bool publishable = !invalidStep_;
212 if (auto validated = asset.validate(); !validated.ok()) {
213 const auto* diagnostic = validated.error();
214 output_.diagnostics.push_back(
215 {DnutSeverity::Error, path_, 0, 0, diagnostic ? diagnostic->message() : "invalid sequence asset"});
216 publishable = false;
217 }
218 if (!publishedIds_.insert(asset.id).second) {
219 output_.diagnostics.push_back(
220 {DnutSeverity::Error, path_, 0, 0,
221 std::string("duplicate ") + (isCommon ? "common" : "story") + " id '" + asset.id + "'"});
222 publishable = false;
223 }
224 if (publishable) output_.assets.push_back(std::move(asset));
225 }

227 void parseBlockBody(std::vector<CompiledStatement>& out) {
228 expectPunct("{");
229 while (true) {
230 if (atEnd()) errorHere("unterminated block");
231 if (isPunct("}")) {
232 advance();
233 return;
234 }
235 try {
236 out.push_back(parseStatement());
237 } catch (const ParseAbort&) {
238 synchronize();
239 }
240 }

243 CompiledStatement parseStatement() {
244 if (cur().kind != DnutTokenKind::Identifier) errorHere("expected a statement");
245 const std::string keyword = cur().text;
246 if (keyword == "if") return parseIf();
247 if (keyword == "choice") return parseChoice();
248 if (keyword == "wait") return parseWait();
249 if (keyword == "call" || keyword == "common") return parseCall();
250 if (keyword == "loop") return parseLoop();
251 if (keyword == "break") return parseBreakOrContinue(true);
252 if (keyword == "continue") return parseBreakOrContinue(false);
253 if (keyword == "label") return parseLabel();
254 if (keyword == "jump") return parseJump();
255 if (keyword == "parallel") return parseParallel();
256 if (keyword == "end") {
257 advance();
258 CompiledStatement statement;
259 statement.type = "end";
260 statement.nodeId = nextNodeId();
261 return statement;
262 }
263 return parseStep();
264 }

266 CompiledStatement parseStep() {
267 const DnutToken start = cur();
268 CompiledStatement statement;
269 statement.type = advance().text;
270 statement.nodeId = nextNodeId();
271 while (cur().kind == DnutTokenKind::Identifier && peek(1).kind == DnutTokenKind::Punctuator &&
272 peek(1).text == "=") {
273 const std::string field = advance().text;
274 advance(); // '='
275 statement.payload.set(field, parseValue());
276 }
277
278 // Validate against the vocabulary here, while the statement's own source
279 // location is still available, so a bad field points at the authored line
280 // instead of at the whole document.
281 SequenceNode candidate;
282 candidate.id = statement.nodeId;
283 candidate.type = statement.type;
284 candidate.payload = statement.payload;
285 if (auto validated = registry_.validate(candidate); !validated.ok()) {
286 const auto* diagnostic = validated.error();
287 output_.diagnostics.push_back({DnutSeverity::Error, path_, start.line, start.column,
288 diagnostic ? diagnostic->message() : "invalid step"});
289 invalidStep_ = true;
290 }
291 return statement;
292 }

447 eve::Value parseOr() {
448 eve::Value left = parseAnd();
449 while (isPunct("||")) {
450 advance();
451 eve::Value right = parseAnd();
452 if (auto* object = left.getIf<eve::Value::Object>();
453 object != nullptr && object->contains("any") && object->at("any").isArray()) {
454 object->at("any").pushBack(std::move(right));
455 } else {
456 eve::Value::Array operands;
457 operands.reserve(2);
458 operands.emplace_back(std::move(left));
459 operands.emplace_back(std::move(right));
460 eve::Value::Object combined;
461 combined.emplace("any", std::move(operands));
462 left = eve::Value(std::move(combined));
463 }
464 }
465 return left;
466 }
467
468 eve::Value parseAnd() {
469 eve::Value left = parseNot();
470 while (isPunct("&&")) {
471 advance();
472 eve::Value right = parseNot();
473 if (auto* object = left.getIf<eve::Value::Object>();
474 object != nullptr && object->contains("all") && object->at("all").isArray()) {
475 object->at("all").pushBack(std::move(right));
476 } else {
477 eve::Value::Array operands;
478 operands.reserve(2);
479 operands.emplace_back(std::move(left));
480 operands.emplace_back(std::move(right));
481 eve::Value::Object combined;
482 combined.emplace("all", std::move(operands));
483 left = eve::Value(std::move(combined));
484 }
485 }
486 return left;
487 }

503 eve::Value parseComparison() {
504 if (cur().kind != DnutTokenKind::Identifier) errorHere("condition must start with a variable name");
505 const std::string variable = advance().text;
506 if (cur().kind != DnutTokenKind::Punctuator || !isComparisonOperator(cur().text))
507 errorHere("condition requires a comparison operator (== != > < >= <=)");
508 const std::string operation = advance().text;
509 eve::Value expected = parseValue();
510 return eve::Value(eve::Value::Object{{"var", eve::Value::string(variable)},
511 {"op", eve::Value::string(comparisonSpelling(operation))},
512 {"value", std::move(expected)}});
513 }

515 eve::Value parseValue() {
516 if (isPunct("-")) {
517 advance();
518 if (cur().kind != DnutTokenKind::Number) errorHere("'-' must be followed by a number");
519 eve::Value magnitude = numberValue(advance().text);
520 if (magnitude.isInt64()) return eve::Value::integer(-magnitude.asInt());
521 return eve::Value::number(-magnitude.asDouble());
522 }
523 switch (cur().kind) {
524 case DnutTokenKind::String: return eve::Value::string(advance().text);
525 case DnutTokenKind::Number: return numberValue(advance().text);
526 case DnutTokenKind::Identifier:
527 if (isIdentifier("true")) {
528 advance();
529 return eve::Value::boolean(true);
530 }
531 if (isIdentifier("false")) {
532 advance();
533 return eve::Value::boolean(false);
534 }
535 return eve::Value::string(advance().text);
536 default: errorHere("expected a literal value"); break;
537 }

548 std::string emitStatement(const CompiledStatement& statement, const std::string& next, SequenceAsset& asset) {
549 if (statement.kind == CompiledStatement::Kind::Step) {
550 SequenceNode node;
551 node.id = statement.nodeId;
552 node.type = statement.type;
553 node.next = next;
554 node.payload = statement.payload;
555 asset.nodes.push_back(std::move(node));
556 return statement.nodeId;
557 }
558 if (statement.kind == CompiledStatement::Kind::If) {
559 const std::string thenEntry = emitBlock(statement.thenBody, next, asset);
560 const std::string elseEntry = emitBlock(statement.elseBody, next, asset);
561 SequenceNode node;
562 node.id = statement.nodeId;
563 node.type = "branch";
564 node.next = elseEntry;
565 SequenceRoute route;
566 route.condition = statement.condition;
567 route.target = thenEntry;
568 node.routes.push_back(std::move(route));
569 asset.nodes.push_back(std::move(node));
570 return statement.nodeId;
571 }
572 if (statement.kind == CompiledStatement::Kind::Choice) {
573 SequenceNode node;
574 node.id = statement.nodeId;
575 node.type = "choice";
576 node.next = next;
577 for (const auto& option : statement.options) {
578 SequenceRoute route;
579 route.label = option.label;
580 route.condition = option.condition;
581 route.target = emitBlock(option.body, next, asset);
582 node.routes.push_back(std::move(route));
583 }
584 asset.nodes.push_back(std::move(node));
585 return statement.nodeId;
586 }
587 if (statement.kind == CompiledStatement::Kind::Loop) {
588 loopStack_.push_back({statement.nodeId, next});
589 const std::string bodyEntry = emitBlock(statement.thenBody, statement.nodeId, asset);
590 loopStack_.pop_back();
591 SequenceNode node;
592 node.id = statement.nodeId;
593 node.type = "loop";
594 node.next = next;
595 node.payload.set("mode", eve::Value::string(statement.name));
596 if (statement.name == "times") {
597 if (const eve::Value* count = statement.payload.find("count")) node.payload.set("count", *count);
598 }
599 SequenceRoute route;
600 if (statement.name == "while") route.condition = statement.condition;
601 route.target = bodyEntry;
602 node.routes.push_back(std::move(route));
603 asset.nodes.push_back(std::move(node));
604 return statement.nodeId;
605 }
606 if (statement.kind == CompiledStatement::Kind::Break ||
607 statement.kind == CompiledStatement::Kind::Continue) {
608 if (loopStack_.empty()) {
609 output_.diagnostics.push_back({DnutSeverity::Error, path_, statement.sourceLine,
610 statement.sourceColumn,
611 statement.type + " is not inside a loop"});
612 invalidStep_ = true;
613 }
614 SequenceNode node;
615 node.id = statement.nodeId;
616 node.type = statement.type;
617 node.next = loopStack_.empty()
618 ? next
619 : (statement.kind == CompiledStatement::Kind::Break ? loopStack_.back().afterNodeId
620 : loopStack_.back().loopNodeId);
621 asset.nodes.push_back(std::move(node));
622 return statement.nodeId;
623 }
624 if (statement.kind == CompiledStatement::Kind::Label) {
625 if (!labels_.emplace(statement.name, statement.nodeId).second) {
626 output_.diagnostics.push_back({DnutSeverity::Error, path_, statement.sourceLine,
627 statement.sourceColumn, "duplicate label '" + statement.name + "'"});
628 invalidStep_ = true;
629 }
630 SequenceNode node;
631 node.id = statement.nodeId;
632 node.type = "label";
633 node.next = next;
634 node.payload.set("name", eve::Value::string(statement.name));
635 asset.nodes.push_back(std::move(node));
636 return statement.nodeId;
637 }
638 if (statement.kind == CompiledStatement::Kind::Jump) {
639 SequenceNode node;
640 node.id = statement.nodeId;
641 node.type = "jump";
642 node.next = next;
643 node.payload.set("name", eve::Value::string(statement.name));
644 asset.nodes.push_back(std::move(node));
645 pendingJumps_.push_back({statement.nodeId, statement.name, statement.sourceLine, statement.sourceColumn});
646 return statement.nodeId;
647 }
648 if (statement.kind == CompiledStatement::Kind::Parallel) {
649 const std::string joinId = nextNodeId();
650 SequenceNode join;
651 join.id = joinId;
652 join.type = "join";
653 asset.nodes.push_back(std::move(join));
654
655 SequenceNode node;
656 node.id = statement.nodeId;
657 node.type = "parallel";
658 node.next = next;
659 node.payload.set("join", eve::Value::string(joinId));
660 for (const auto& lane : statement.lanes) {
661 SequenceRoute route;
662 route.target = emitBlock(lane, joinId, asset);
663 node.routes.push_back(std::move(route));
664 }
665 asset.nodes.push_back(std::move(node));
666 return statement.nodeId;
667 }
668 output_.diagnostics.push_back(
669 {DnutSeverity::Error, path_, 0, 0, "internal: unhandled compiled statement kind"});
670 invalidStep_ = true;
671 return next;
672 }

674 void resolveJumps(SequenceAsset& asset) {
675 for (const auto& jump : pendingJumps_) {
676 auto found = labels_.find(jump.name);
677 if (found == labels_.end()) {
678 output_.diagnostics.push_back({DnutSeverity::Error, path_, jump.line, jump.column,
679 "jump targets unknown label '" + jump.name + "'"});
680 invalidStep_ = true;
681 continue;
682 }
683 SequenceNode* node = nullptr;
684 for (auto& candidate : asset.nodes) {
685 if (candidate.id == jump.nodeId) {
686 node = &candidate;
687 break;
