107eve::Result<eve::decision::Condition> decodeNode(const eve::Value& value) {
108 const auto* object = value.getIf<eve::Value::Object>();
109 if (object == nullptr) return invalid<eve::decision::Condition>("condition node must be an object");
110 const auto* kindValue = field(*object, "kind");
111 if (kindValue == nullptr) return invalid<eve::decision::Condition>("condition node requires kind", "kind");
112 auto kind = parseKind(*kindValue);
113 if (!kind) return eve::Result<eve::decision::Condition>::failure(kind.status());
114 const auto parsedKind = kind.value();
115 using K = eve::decision::ConditionKind;
116 if (parsedKind == K::All || parsedKind == K::Any || parsedKind == K::Not) {
117 if (!exactFields(*object, {"kind", "children"}))
118 return invalid<eve::decision::Condition>("logical condition contains an unknown field");
119 auto children = decodeChildren(*object);
120 if (!children) return eve::Result<eve::decision::Condition>::failure(children.status());
121 auto values = std::move(children).takeValue();
122 if (parsedKind == K::Not && values.size() != 1)
123 return invalid<eve::decision::Condition>("not condition requires one child", "children");
124 if (parsedKind == K::All)
125 return eve::Result<eve::decision::Condition>::success(eve::decision::Condition::all(std::move(values)));
126 if (parsedKind == K::Any)
127 return eve::Result<eve::decision::Condition>::success(eve::decision::Condition::any(std::move(values)));
128 return eve::Result<eve::decision::Condition>::success(
129 eve::decision::Condition::not_(std::move(values.front())));
130 }
131 if (parsedKind == K::Compare) {
132 if (!exactFields(*object, {"kind", "key", "operator", "expected"}))
133 return invalid<eve::decision::Condition>("compare condition contains an unknown field");
134 auto key = requiredString(*object, "key");
135 const auto* op = field(*object, "operator");
136 const auto* expected = field(*object, "expected");
137 if (!key || op == nullptr || expected == nullptr)
138 return invalid<eve::decision::Condition>("compare condition is incomplete");
139 auto parsedOp = parseOperator(*op);
140 if (!parsedOp) return eve::Result<eve::decision::Condition>::failure(parsedOp.status());
141 return eve::Result<eve::decision::Condition>::success(
142 eve::decision::Condition::compare(std::move(key).takeValue(), parsedOp.value(), *expected));
143 }
144 if (parsedKind == K::StateEquals) {
145 if (!exactFields(*object, {"kind", "key", "expected"}))
146 return invalid<eve::decision::Condition>("state condition contains an unknown field");
147 auto key = requiredString(*object, "key");
148 const auto* expected = field(*object, "expected");
149 if (!key || expected == nullptr) return invalid<eve::decision::Condition>("state condition is incomplete");
150 return eve::Result<eve::decision::Condition>::success(
151 eve::decision::Condition::stateEquals(std::move(key).takeValue(), *expected));
152 }
153 if (parsedKind == K::PolicyCall) {
154 if (!exactFields(*object, {"kind", "key", "arguments", "scriptDeclaration"}))
155 return invalid<eve::decision::Condition>("policy condition contains an unknown field");
156 auto key = requiredString(*object, "key");
157 if (!key) return eve::Result<eve::decision::Condition>::failure(key.status());
158 eve::Value arguments = eve::Value::Object{};
159 if (const auto* value = field(*object, "arguments")) arguments = *value;
160 std::optional<eve::decision::ScriptConditionDeclaration> declaration;
161 if (const auto* value = field(*object, "scriptDeclaration")) {
162 const auto* declarationObject = value->getIf<eve::Value::Object>();
163 if (declarationObject == nullptr ||
164 !exactFields(*declarationObject, {"name", "dependencies", "determinism"}))
165 return invalid<eve::decision::Condition>("script declaration is invalid", "scriptDeclaration");
166 auto name = requiredString(*declarationObject, "name");
167 const auto* dependencyValue = field(*declarationObject, "dependencies");
168 const auto* dependencies =
169 dependencyValue == nullptr ? nullptr : dependencyValue->getIf<eve::Value::Array>();
170 const auto* determinism = field(*declarationObject, "determinism");
171 if (!name || dependencies == nullptr || determinism == nullptr)
172 return invalid<eve::decision::Condition>("script declaration is incomplete", "scriptDeclaration");
173 auto level = parseDeterminism(*determinism);
174 if (!level) return eve::Result<eve::decision::Condition>::failure(level.status());
175 eve::decision::ScriptConditionDeclaration parsed{std::move(name).takeValue(), {}, level.value()};
176 for (const auto& dependency : *dependencies) {
177 const auto* text = dependency.getIf<std::string>();
178 if (text == nullptr || text->empty())
179 return invalid<eve::decision::Condition>("script dependency is invalid",
180 "scriptDeclaration.dependencies");
181 parsed.dependencies.push_back(*text);
182 }
183 declaration = std::move(parsed);
184 }
185 return eve::Result<eve::decision::Condition>::success(eve::decision::Condition::policyCall(
186 std::move(key).takeValue(), std::move(arguments), std::move(declaration)));
187 }
188 if (!exactFields(*object, {"kind", "key"}))
189 return invalid<eve::decision::Condition>("leaf condition contains an unknown field");
190 auto key = requiredString(*object, "key");
191 if (!key) return eve::Result<eve::decision::Condition>::failure(key.status());
192 switch (parsedKind) {
193 case K::HasTag:
194 return eve::Result<eve::decision::Condition>::success(
195 eve::decision::Condition::hasTag(std::move(key).takeValue()));
196 case K::HasAttribute:
197 return eve::Result<eve::decision::Condition>::success(
198 eve::decision::Condition::hasAttribute(std::move(key).takeValue()));
199 case K::HasResource:
200 return eve::Result<eve::decision::Condition>::success(
201 eve::decision::Condition::hasResource(std::move(key).takeValue()));
202 case K::AuthorityCheck:
203 return eve::Result<eve::decision::Condition>::success(
204 eve::decision::Condition::authorityCheck(std::move(key).takeValue()));
205 default: return invalid<eve::decision::Condition>("unsupported condition node kind");
206 }
207}

211eve::Result<eve::Value> encodeSkillCondition(const eve::decision::Condition& condition) {
212 if (!condition.isValid()) return invalid<eve::Value>("skill condition is invalid");
213 using K = eve::decision::ConditionKind;
214 eve::Value::Object object{{"kind", eve::Value(eve::decision::conditionKindName(condition.kind()))}};
215 if (condition.kind() == K::All || condition.kind() == K::Any || condition.kind() == K::Not) {
216 eve::Value::Array children;
217 for (const auto& child : condition.children()) {
218 auto encoded = encodeSkillCondition(child);
219 if (!encoded) return eve::Result<eve::Value>::failure(encoded.status());
220 children.push_back(std::move(encoded).takeValue());
221 }
222 object.emplace("children", eve::Value(std::move(children)));
223 } else if (condition.kind() == K::Compare) {
224 object.emplace("key", eve::Value(condition.key()));
225 object.emplace("operator", eve::Value(eve::decision::compareOperatorName(condition.compareOperator())));
226 object.emplace("expected", condition.expected());
227 } else if (condition.kind() == K::StateEquals) {
228 object.emplace("key", eve::Value(condition.key()));
229 object.emplace("expected", condition.expected());
230 } else {
231 object.emplace("key", eve::Value(condition.key()));
232 if (condition.kind() == K::PolicyCall) {
233 object.emplace("arguments", condition.arguments());
234 if (condition.scriptDeclaration()) {
235 const auto& declaration = *condition.scriptDeclaration();
236 eve::Value::Array dependencies;
237 for (const auto& dependency : declaration.dependencies) dependencies.emplace_back(dependency);
238 std::string determinism;
239 switch (declaration.determinism) {
240 case eve::decision::DeterminismLevel::BitExact: determinism = "bit_exact"; break;
241 case eve::decision::DeterminismLevel::TickDeterministic: determinism = "tick_deterministic"; break;
242 case eve::decision::DeterminismLevel::ToleranceBounded: determinism = "tolerance_bounded"; break;
243 case eve::decision::DeterminismLevel::ExplicitlyNondeterministic:
244 determinism = "explicitly_nondeterministic";
245 break;
246 }
247 object.emplace("scriptDeclaration",
248 eve::Value(eve::Value::Object{{"name", eve::Value(declaration.name)},
249 {"dependencies", eve::Value(std::move(dependencies))},
250 {"determinism", eve::Value(std::move(determinism))}}));
251 }
252 }
253 }
254 return eve::Result<eve::Value>::success(eve::Value(std::move(object)));
255}
