34bool isValidGameplayTagName(std::string_view name) noexcept {
35 if (name.empty() || name.front() == '.' || name.back() == '.') return false;
36 bool segmentHasCharacter = false;
37 for (const unsigned char c : name) {
38 if (c == '.') {
39 if (!segmentHasCharacter) return false;
40 segmentHasCharacter = false;
41 continue;
42 }
43 if (!(std::isalnum(c) || c == '_' || c == '-')) return false;
44 segmentHasCharacter = true;
45 }
46 return segmentHasCharacter;
47}

64Result<GameplayTagId> GameplayTagRegistry::registerTag(std::string name, std::string description) {
65 if (!isValidGameplayTagName(name)) return invalid<GameplayTagId>("Invalid gameplay-tag name", "name");
66 const GameplayTagId id = gameplayTagId(name);
67 if (const auto existing = definitionsByName_.find(name); existing != definitionsByName_.end()) {
68 if (existing->second.description != description)
69 return Result<GameplayTagId>::failure(Diagnostic::error(
70 DiagnosticCode::AlreadyExists, "Gameplay tag is already registered with different metadata", name));
71 return Result<GameplayTagId>::success(id, Status::success(StatusCode::NoOp));
72 }
73 if (const auto collision = nameById_.find(id); collision != nameById_.end() && collision->second != name)
74 return Result<GameplayTagId>::failure(
75 Diagnostic::error(DiagnosticCode::Conflict, "Gameplay-tag hash collision", name));
76 definitionsByName_.emplace(name, GameplayTagDefinition{id, name, std::move(description)});
77 nameById_.emplace(id, std::move(name));
78 return Result<GameplayTagId>::success(id, Status::success(StatusCode::Applied));
79}

127 if (!version || !version->isInt64() || version->asInt() != 1)

130
131 GameplayTagRegistry candidate;
132 for (std::size_t i = 0; i < tags->arraySize(); ++i) {
133 const auto& entry = tags->at(i);

135 if (!entry.isObject()) return invalid<GameplayTagRegistry>("Gameplay-tag entry must be an object", path);
136 const Value* name = member(entry, "name");
137 const Value* description = member(entry, "description");
138 const Value* id = member(entry, "id");
139 if (!name || !name->isString())
140 return invalid<GameplayTagRegistry>("Gameplay-tag name must be a string", path + ".name");
141 if (!description || !description->isString())

145 auto registered = candidate.registerTag(name->asString(), description->asString());
