15ResourceManager& ResourceManager::getInstance() {
16 // Intentionally leaked: cached CPU resources may own third-party handles
17 // (FreeType faces, Assimp scenes, image decode handlers) whose libraries
18 // are torn down at process exit in an unspecified TU order. Destroying
19 // cached entries from the singleton destructor can therefore crash at
20 // exit. Keeping the singleton alive until the OS reclaims it avoids
21 // exit-time destructors entirely; explicit unload()/clear() still release
22 // entries during the run.
23 static ResourceManager* instance = new ResourceManager();
24 instance->ensureRegistered();
25 return *instance;
26}

28std::string ResourceManager::normalizePath(std::string path) {
29 for (char &c : path) {
30 if (c == '\\') c = '/';
31 }
32 while (path.size() >= 2 && path[0] == '.' && path[1] == '/') path.erase(0, 2);
33 while (path.size() > 1 && path.back() == '/') path.pop_back();
34 return path;
35}

75void ResourceManager::runLoadJob(std::string norm, uint64_t epoch) {
76 Resource *loaded = nullptr;
77 std::string error;
78 try {
79 loaded = loadUncached(norm);
80 } catch (const std::exception &ex) {
81 error = ex.what();
82 } catch (...) {
83 error = "unknown resource load failure";
84 }
85
86 std::lock_guard<std::mutex> lock(mu_);
87 auto pendingIt = pending_.find(norm);
88 if (epoch_ != epoch) {
89 delete loaded;
90 if (pendingIt != pending_.end()) {
91 pendingIt->second->done = true;
92 pending_.erase(pendingIt);
93 }
94 cv_.notify_all();
95 return;
96 }
97 if (!error.empty() || loaded == nullptr) {
98 if (pendingIt != pending_.end()) {
99 pendingIt->second->done = true;
100 pendingIt->second->failed = true;
101 pendingIt->second->error =
102 error.empty() ? std::string("no provider claimed key") : std::move(error);
103 pending_.erase(pendingIt);
104 }
105 delete loaded;
106 cv_.notify_all();
107 return;
108 }
109 auto [it, inserted] = resources.emplace(norm, loaded);
110 if (!inserted) delete loaded;
111 if (pendingIt != pending_.end()) {
112 pendingIt->second->done = true;
113 pending_.erase(pendingIt);
114 }
115 cv_.notify_all();
116}

126eve::Result<void> ResourceManager::request(std::string key) {
127 const std::string norm = makeKey(std::move(key));
128 if (norm.empty()) {
129 return eve::Result<void>::failure(
130 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "empty resource key"));
131 }
132 ensureRegistered();
133
134 std::shared_ptr<AsyncLoad> job;
135 uint64_t epoch = 0;
136 {
137 std::lock_guard<std::mutex> lock(mu_);
138 auto cached = resources.find(norm);
139 if (cached != resources.end())
140 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
141 auto pendingIt = pending_.find(norm);
142 if (pendingIt != pending_.end() && !pendingIt->second->done)
143 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Pending));
144 if (pendingIt != pending_.end()) pending_.erase(pendingIt);
145
146 bool claimed = false;
147 eve::cap::forEachUntil<eve::caps::IAssetReloader>([&](eve::caps::IAssetReloader *r) {
148 if (r == this) return false;
149 if (!r->handlesPath(norm)) return false;
150 claimed = true;
151 return true;
152 });
153 if (!claimed) {
154 return eve::Result<void>::failure(
155 eve::Diagnostic::error(eve::DiagnosticCode::NotFound, "no provider claimed key", norm));
156 }
157
158 job = std::make_shared<AsyncLoad>();
159 job->epoch = epoch_;
160 epoch = epoch_;
161 pending_.emplace(norm, job);
162 }
163
164 auto *executor = eve::cap::query<eve::caps::IAsyncWorkExecutor>();
165 if (!executor) {
166 runLoadJob(norm, epoch);
167 std::lock_guard<std::mutex> lock(mu_);
168 if (resources.find(norm) != resources.end())
169 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
170 return eve::Result<void>::failure(
171 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "inline resource load failed", norm));
172 }
173
174 auto submitted = executor->submit([this, norm, epoch]() { runLoadJob(norm, epoch); });
175 if (!submitted.ok()) {
176 std::lock_guard<std::mutex> lock(mu_);
177 pending_.erase(norm);
178 cv_.notify_all();
179 return submitted;
180 }
181 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
182}

201ResultRef<Resource> ResourceManager::waitFor(std::string key) {
202 const std::string norm = makeKey(std::move(key));
203 if (norm.empty()) {
204 return ResultRef<Resource>::failure(
205 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "empty resource key"));
206 }
207 ensureRegistered();
208
209 std::shared_ptr<AsyncLoad> job;
210 {
211 std::unique_lock<std::mutex> lock(mu_);
212 auto cached = resources.find(norm);
213 if (cached != resources.end())
214 return ResultRef<Resource>::success(std::ref(*cached->second.get()));
215 auto pendingIt = pending_.find(norm);
216 if (pendingIt != pending_.end()) job = pendingIt->second;
217 if (job) {
218 cv_.wait(lock, [&] { return job->done || resources.find(norm) != resources.end(); });
219 cached = resources.find(norm);
220 if (cached != resources.end())
221 return ResultRef<Resource>::success(std::ref(*cached->second.get()));
222 if (job->failed) {
223 return ResultRef<Resource>::failure(eve::Diagnostic::error(
224 eve::DiagnosticCode::Failed,
225 job->error.empty() ? "async resource load failed" : job->error, norm));
226 }
227 }
228 }
229
230 if (job && job->done) {
231 std::lock_guard<std::mutex> lock(mu_);
232 auto cached = resources.find(norm);
233 if (cached != resources.end())
234 return ResultRef<Resource>::success(std::ref(*cached->second.get()));
235 if (job->failed) {
236 return ResultRef<Resource>::failure(eve::Diagnostic::error(
237 eve::DiagnosticCode::Failed,
238 job->error.empty() ? "async resource load failed" : job->error, norm));
239 }
240 }
241
242 Resource *loaded = nullptr;
243 try {
244 loaded = loadUncached(norm);
245 } catch (const std::exception &ex) {
246 return ResultRef<Resource>::failure(
247 eve::Diagnostic::error(eve::DiagnosticCode::Failed, ex.what(), norm));
248 }
249 if (!loaded) {
250 return ResultRef<Resource>::failure(
251 eve::Diagnostic::error(eve::DiagnosticCode::NotFound, "no provider claimed key", norm));
252 }
253
254 std::lock_guard<std::mutex> lock(mu_);
255 auto [it, inserted] = resources.emplace(norm, loaded);
256 if (!inserted) delete loaded;
257 return ResultRef<Resource>::success(std::ref(*it->second.get()));
258}

310eve::Result<bool> ResourceManager::reload(const std::string &normPath) {
311 const std::string norm = normalizePath(normPath);
312 if (norm.empty()) return eve::Result<bool>::success(false);
313
314 struct Prepared {
315 std::string key;
316 Resource *cached = nullptr;
317 std::unique_ptr<Resource> replacement;
318 };
319
320 std::vector<std::string> keys;
321 std::set<std::string> selectedKeys;
322 std::set<Resource *> selectedResources;
323 {
324 std::lock_guard<std::mutex> lock(mu_);
325 for (auto &kv : resources) {
326 if (pathOfKey(kv.first) != norm) continue;
327 keys.push_back(kv.first);
328 selectedKeys.insert(kv.first);
329 selectedResources.insert(kv.second.get());
330 }
331
332 // Close over reverse dependencies before loading anything. This gives
333 // the transaction a stable, root-first order and makes cycles safe.
334 bool expanded = true;
335 while (expanded) {
336 expanded = false;
337 for (auto &kv : resources) {
338 if (selectedKeys.count(kv.first)) continue;
339 bool dependsOnSelection = false;
340 for (auto dependency : kv.second->getDependencies()) {
341 if (selectedResources.count(dependency.get())) {
342 dependsOnSelection = true;
343 break;
344 }
345 }
346 if (!dependsOnSelection) continue;
347 keys.push_back(kv.first);
348 selectedKeys.insert(kv.first);
349 selectedResources.insert(kv.second.get());
350 expanded = true;
351 }
352 }
353 }
354 if (keys.empty()) return eve::Result<bool>::success(false);
355
356 // Decode every root and dependent into detached candidates. A single
357 // invalid file aborts the whole graph without touching live objects.
358 std::vector<Prepared> prepared;
359 std::vector<ref<Resource>> keepAlive;
360 prepared.reserve(keys.size());
361 keepAlive.reserve(keys.size());
362 for (const auto &key : keys) {
363 Resource *cached = nullptr;
364 {
365 std::lock_guard<std::mutex> lock(mu_);
366 auto it = resources.find(key);
367 if (it == resources.end()) return eve::Result<bool>::success(false);
368 cached = it->second.get();
369 keepAlive.emplace_back(cached);
370 }
371 std::unique_ptr<Resource> replacement(loadReplacement(key));
372 if (!replacement) return eve::Result<bool>::success(false);
373 prepared.push_back({key, cached, std::move(replacement)});
374 }
375
376 std::lock_guard<std::mutex> lock(mu_);
377 for (const auto &item : prepared) {
378 auto it = resources.find(item.key);
379 if (it == resources.end() || it->second.get() != item.cached) return eve::Result<bool>::success(false);
380 }
381
382 size_t committed = 0;
383 try {
384 for (; committed < prepared.size(); ++committed)
385 prepared[committed].cached->adopt(*prepared[committed].replacement);
386 } catch (...) {
387 // adopt() is a payload swap. Candidates therefore hold the old state
388 // after a successful commit and can restore it in reverse order.
389 while (committed > 0) {
390 --committed;
391 try {
392 prepared[committed].cached->adopt(*prepared[committed].replacement);
393 } catch (...) {
394 // The adopt contract requires a non-mutating failure. Keep
395 // unwinding other entries even if a broken implementation
396 // violates it.
397 }
398 }
399 return eve::Result<bool>::success(false);
400 }
401 return eve::Result<bool>::success(true, eve::Status::success(eve::StatusCode::Applied));
402}
