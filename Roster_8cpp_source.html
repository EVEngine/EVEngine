64Result<void> Roster::assignPost(std::string_view postId, std::string subject, std::string role) {
65 if (emptyId(postId) || emptyId(subject))
66 return fail(DiagnosticCode::InvalidArgument, "post id and subject are required", "assignPost");
67 auto it = posts_.find(std::string(postId));
68 if (it == posts_.end()) return fail(DiagnosticCode::NotFound, "post was not found", "postId");
69 for (const auto& [id, post] : posts_) {
70 (void)id;
71 if (findOccupant(post.occupants, subject) != post.occupants.end())
72 return fail(DiagnosticCode::Conflict, "subject already occupies a post", "subject");
73 }
74 if (static_cast<int>(it->second.occupants.size()) >= it->second.capacity)
75 return fail(DiagnosticCode::Conflict, "post is at capacity", "postId");
76 if (role.empty()) role = it->second.defaultRole;
77 it->second.occupants.push_back(Occupant{std::move(subject), std::move(role)});
78 std::sort(it->second.occupants.begin(), it->second.occupants.end(),
79 [](const Occupant& a, const Occupant& b) { return a.subject < b.subject; });
80 return Result<void>::success(Status::success(StatusCode::Applied));
81}

202 if (emptyId(subject)) return failT<int>(DiagnosticCode::InvalidArgument, "subject is required", "subject");
203 int released = 0;
204 for (auto& [postId, post] : posts_) {

214 pushVacancy("squad", squadId, member->role, member->subject, tick);
215 squad.members.erase(member);
216 ++released;
217 }
218 return Result<int>::success(released, Status::success(released == 0 ? StatusCode::NoOp : StatusCode::Applied));

249Result<void> Roster::setFallback(std::string_view policyId, std::string preferredDef, std::string fallbackDef) {
250 auto it = policies_.find(std::string(policyId));
251 if (it == policies_.end()) return fail(DiagnosticCode::NotFound, "policy was not found", "policyId");
252 if (emptyId(preferredDef) || emptyId(fallbackDef) || preferredDef == fallbackDef)
253 return fail(DiagnosticCode::InvalidArgument, "fallback needs distinct preferred and fallback defs", "fallback");

279Result<int> Roster::step(std::uint64_t tick) {
280 int created = 0;
281 std::vector<Vacancy> remaining;
282 remaining.reserve(vacancies_.size());
283 for (auto& vacancy : vacancies_) {
284 if (vacancy.tick > tick) {
285 remaining.push_back(std::move(vacancy));
286 continue;
287 }
288 bool matched = false;
289 for (auto& [policyId, policy] : policies_) {
290 if (!policyMatches(policy, vacancy)) continue;
291 if (static_cast<std::uint64_t>(policy.delayTicks) > tick - vacancy.tick) {
292 remaining.push_back(std::move(vacancy));
293 matched = true;
294 break;
295 }
296 if (!budgetAvailable(policy)) {
297 // Budget exhausted: drop this vacancy for this policy family.
298 matched = true;
299 break;
300 }
301 SpawnRequest request;
302 request.requestId = nextRequestId();
303 request.policyId = policyId;
304 request.preferredDef = policy.preferredDef;
305 request.role = vacancy.role;
306 if (vacancy.kind == "post") request.postId = vacancy.targetId;
307 if (vacancy.kind == "squad") request.squadId = vacancy.targetId;
308 request.readyAtTick = tick;
309 spawnRequests_.push_back(std::move(request));
310 ++created;
311 matched = true;
312 break;
313 }
314 if (!matched) remaining.push_back(std::move(vacancy));
315 }
316 vacancies_ = std::move(remaining);
317 return Result<int>::success(created, Status::success(created == 0 ? StatusCode::NoOp : StatusCode::Applied));
318}

346Result<void> Roster::failSpawnRequest(std::string_view requestId, std::uint64_t tick) {
347 auto it = std::find_if(spawnRequests_.begin(), spawnRequests_.end(),
348 [&](const SpawnRequest& request) { return request.requestId == requestId; });
349 if (it == spawnRequests_.end()) return fail(DiagnosticCode::NotFound, "spawn request was not found", "requestId");
350 SpawnRequest failed = *it;
351 spawnRequests_.erase(it);
352 auto policy = policies_.find(failed.policyId);
353 if (policy == policies_.end()) return Result<void>::success(Status::success(StatusCode::NoOp));
354 auto fallback = policy->second.fallbacks.find(failed.preferredDef);
355 if (fallback == policy->second.fallbacks.end()) return Result<void>::success(Status::success(StatusCode::NoOp));
356 if (!budgetAvailable(policy->second)) return Result<void>::success(Status::success(StatusCode::NoOp));
357 SpawnRequest retry = failed;
358 retry.requestId = nextRequestId();
359 retry.preferredDef = fallback->second;
360 retry.readyAtTick = tick;
361 spawnRequests_.push_back(std::move(retry));
362 return Result<void>::success(Status::success(StatusCode::Applied));
363}
