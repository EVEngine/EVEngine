70Result<LogicalId> id(std::string_view domain, std::string_view name) {
71 auto parsed = LogicalId::fromParts(domain, name);
72 if (!parsed)

74 Diagnostic::error(DiagnosticCode::InvalidArgument, "Standard ability logical id is invalid",
75 std::string(domain) + ":" + std::string(name)));
76 return Result<LogicalId>::success(std::move(*parsed));

83Result<std::vector<action::AbilityDefinition>> standardCombatAbilities() {
84 std::vector<action::AbilityDefinition> result;
85 result.reserve(kSpecs.size());
86 for (const AbilitySpec& spec : kSpecs) {
87 auto abilityId = id("combat-ability", spec.name);
88 if (!abilityId) return Result<std::vector<action::AbilityDefinition>>::failure(abilityId.status());
89 auto actionId = id("combat-action", spec.name);
90 if (!actionId) return Result<std::vector<action::AbilityDefinition>>::failure(actionId.status());
91
92 action::AbilityDefinition definition;
93 definition.id = std::move(abilityId).takeValue();
94 definition.action.id = std::move(actionId).takeValue();
95 definition.action.timing.windup = milliseconds(spec.windupMs);
96 definition.action.timing.active = milliseconds(spec.activeMs);
97 definition.action.timing.recover = milliseconds(spec.recoverMs);
98 definition.action.metadata["combat.adapter"] = std::string(spec.adapter);
99 definition.cooldown = milliseconds(spec.cooldownMs);
100 definition.instancing = spec.instancing;
101 definition.activationGroup = spec.group;
102 auto valid = definition.validate();
103 if (!valid) return Result<std::vector<action::AbilityDefinition>>::failure(valid.status());
104 result.push_back(std::move(definition));
105 }
106 return Result<std::vector<action::AbilityDefinition>>::success(std::move(result));
107}
