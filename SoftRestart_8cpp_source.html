23void pushJsonValue(HSQUIRRELVM vm, const Poco::Dynamic::Var& value, int depth) {
24 if (depth > kMaxArgumentDepth || value.isEmpty()) {
25 sq_pushnull(vm);
26 return;
27 }
28 if (value.isBoolean()) {
29 sq_pushbool(vm, value.convert<bool>() ? SQTrue : SQFalse);
30 return;
31 }
32 if (value.isInteger()) {
33 sq_pushinteger(vm, static_cast<SQInteger>(value.convert<Poco::Int64>()));
34 return;
35 }
36 if (value.isNumeric()) {
37 sq_pushfloat(vm, static_cast<SQFloat>(value.convert<double>()));
38 return;
39 }
40 if (value.isString()) {
41 const std::string text = value.convert<std::string>();
42 sq_pushstring(vm, text.c_str(), static_cast<SQInteger>(text.size()));
43 return;
44 }
45 if (value.type() == typeid(Poco::JSON::Array::Ptr)) {
46 auto array = value.extract<Poco::JSON::Array::Ptr>();
47 sq_newarray(vm, 0);
48 const SQInteger target = sq_gettop(vm);
49 for (unsigned index = 0; index < array->size(); ++index) {
50 pushJsonValue(vm, array->get(index), depth + 1);
51 sq_arrayappend(vm, target);
52 }
53 return;
54 }
55 if (value.type() == typeid(Poco::JSON::Object::Ptr)) {
56 auto object = value.extract<Poco::JSON::Object::Ptr>();
57 sq_newtable(vm);
58 const SQInteger target = sq_gettop(vm);
59 for (const auto& entry : *object) {
60 sq_pushstring(vm, entry.first.c_str(), static_cast<SQInteger>(entry.first.size()));
61 pushJsonValue(vm, entry.second, depth + 1);
62 sq_newslot(vm, target, SQFalse);
63 }
64 return;
65 }
66 sq_pushnull(vm);
67}

76Result<void> installRestartArgs(HSQUIRRELVM vm, Poco::JSON::Object::Ptr args) {
77 const SQInteger top = sq_gettop(vm);
78 sq_pushroottable(vm);
79 sq_pushstring(vm, _SC("eve"), -1);
80 if (SQ_FAILED(sq_get(vm, -2)) || sq_gettype(vm, -1) != OT_TABLE) {
81 sq_settop(vm, top);
82 return Result<void>::failure(
83 Diagnostic::error(DiagnosticCode::NotFound, "eve table missing; is load.nut running?", "eve"));
84 }
85 sq_pushstring(vm, _SC("restartArgs"), -1);
86 if (args)
87 pushJsonValue(vm, Poco::Dynamic::Var(args), 0);
88 else
89 sq_newtable(vm);
90 if (SQ_FAILED(sq_newslot(vm, -3, SQFalse))) {
91 sq_settop(vm, top);
92 return Result<void>::failure(
93 Diagnostic::error(DiagnosticCode::Failed, "failed to set eve.restartArgs", "eve.restartArgs"));
94 }
95 sq_settop(vm, top);
96 return Result<void>::success();
97}

100Result<bool> callSoftRestartGame(HSQUIRRELVM vm, bool reloadScripts) {
101 const SQInteger top = sq_gettop(vm);
102 sq_pushroottable(vm);
103 sq_pushstring(vm, _SC("soft_restart_game"), -1);
104 if (SQ_FAILED(sq_get(vm, -2)) || sq_gettype(vm, -1) != OT_CLOSURE) {
105 sq_settop(vm, top);
106 return Result<bool>::failure(Diagnostic::error(
107 DiagnosticCode::NotFound,
108 "soft_restart_game is missing; start the game with load.nut (eve run --mcp-port ...)",
109 "soft_restart_game"));
110 }
111 sq_pushroottable(vm); // this
112 sq_pushbool(vm, reloadScripts ? SQTrue : SQFalse);
113 if (SQ_FAILED(sq_call(vm, 2, SQTrue, SQTrue))) {
114 const std::string err = lastVmError(vm);
115 sq_settop(vm, top);
116 return Result<bool>::failure(
117 Diagnostic::error(DiagnosticCode::Failed, err, "soft_restart_game"));
118 }
119 SQBool ok = SQFalse;
120 if (sq_gettype(vm, -1) == OT_BOOL) sq_getbool(vm, -1, &ok);
121 sq_settop(vm, top);
122 if (!ok) {
123 return Result<bool>::failure(Diagnostic::error(
124 DiagnosticCode::Failed, "soft_restart_game returned false (see console / eve_console_read)",
125 "soft_restart_game"));
126 }
127 return Result<bool>::success(true);
128}

142Result<SoftRestartReport> executeSoftRestart(HSQUIRRELVM vm, SoftRestartRequest request) {
143 if (!vm) {
144 return Result<SoftRestartReport>::failure(
145 Diagnostic::error(DiagnosticCode::NotFound, "no VM attached", "vm"));
146 }
147 if (ReloadSession::instance().active()) {
148 return Result<SoftRestartReport>::failure(Diagnostic::error(
149 DiagnosticCode::Conflict, "a state hot-reload session is already active", "ReloadSession"));
150 }
151
152 SoftRestartReport report;
153 report.resourceCountBefore = resourceCacheCount();
154 report.args = request.args ? request.args : Poco::JSON::Object::Ptr(new Poco::JSON::Object());
155 report.scriptsReloaded = request.reloadScripts;
156
157 auto installed = installRestartArgs(vm, report.args);
158 if (!installed) return Result<SoftRestartReport>::failure(installed.status());
159
160 auto restarted = callSoftRestartGame(vm, request.reloadScripts);
161 if (!restarted) return Result<SoftRestartReport>::failure(restarted.status());
162
163 report.initCalled = true;
164 report.resourceCountAfter = resourceCacheCount();
165 // Soft restart must not drop the unified resource cache. A drop here means
166 // someone called clear()/unload during the restart path — treat as failure
167 // so agents do not silently pay for a cold asset load on the next frame.
168 if (report.resourceCountAfter < report.resourceCountBefore) {
169 return Result<SoftRestartReport>::failure(Diagnostic::error(
170 DiagnosticCode::Failed,
171 "resource cache shrank during soft restart (expected resources to be kept)",
172 "ResourceManager"));
173 }
174 return Result<SoftRestartReport>::success(std::move(report));
175}
