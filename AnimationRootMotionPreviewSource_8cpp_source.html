23Result<std::vector<action::ActionPreviewPoint3>> AnimationRootMotionPreviewSource::sampleRootMotion(
24 std::string_view animationUri, Duration duration, std::uint32_t sampleCount) const {
25 if (animationUri.empty())
26 return sampleError(DiagnosticCode::InvalidArgument, "Animation URI must not be empty", "animationUri");
27 if (duration < Duration::zero())
28 return sampleError(DiagnosticCode::InvalidArgument, "Preview duration must be non-negative", "duration");
29 if (sampleCount < 2 || sampleCount > 1024)
30 return sampleError(DiagnosticCode::InvalidArgument, "Root-motion sample count must be between 2 and 1024",
31 "sampleCount");
32 if (rootBone_ > static_cast<std::uint32_t>(std::numeric_limits<int>::max()))
33 return sampleError(DiagnosticCode::InvalidArgument, "Root-bone index exceeds the animation API range",
34 "rootBone");
35
36 const std::string uri(animationUri);
37 auto clips = animation::AnimClipRegistry::findByPath(uri);
38 if (clips.empty())
39 return sampleError(DiagnosticCode::NotFound, "No registered animation clip matches the preview URI", uri);
40 if (clips.size() != 1)
41 return sampleError(DiagnosticCode::Conflict, "Preview URI resolves to more than one animation clip", uri);
42
43 animation::AnimClip* clip = clips.front();
44 if (!clip) return sampleError(DiagnosticCode::StaleHandle, "Animation registry returned a stale clip", uri);
45 const int rootBone = static_cast<int>(rootBone_);
46 if (rootBone >= clip->getTrackCount())
47 return sampleError(DiagnosticCode::InvalidArgument, "Root-bone track is not present in the animation clip",
48 "rootBone");
49
50 std::vector<action::ActionPreviewPoint3> path;
51 path.reserve(sampleCount);
52 animation::AnimPose pose;
53 try {
54 for (std::uint32_t index = 0; index < sampleCount; ++index) {
55 const double alpha = static_cast<double>(index) / static_cast<double>(sampleCount - 1);
56 const float sampleTime = static_cast<float>(duration.seconds() * alpha);
57 clip->sample(sampleTime, &pose);
58 const auto& root = pose.local(rootBone);
59 path.push_back({root.px, root.py, root.pz});
60 }
61 } catch (const Exception&) {
62 return sampleError(DiagnosticCode::Failed, "Animation clip failed while sampling root motion", uri);
63 }
64 return Result<std::vector<action::ActionPreviewPoint3>>::success(std::move(path));
65}

67} // namespace eve::editor
