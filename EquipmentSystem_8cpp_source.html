29int EquipmentSystem::registerItemStatsFromJson(const std::string &itemId, const std::string &json) {
30 if (itemId.empty()) return 0;
31 std::string err;
32 const eve::json::Document doc = eve::json::Document::parse(json, &err);
33 if (!doc.valid()) return 0;
34 const Value root = doc.root();
35 int n = 0;
36 const auto addObject = [&](Value o) {
37 if (!o.isObject()) return;
38 EquipmentStat stat;
39 stat.attribute = o.getString("attribute");
40 stat.op = o.getString("op", "add");
41 stat.value = o.getDouble("value", 0.0);
42 stat.priority = o.getInt("priority", 0);
43 if (stat.attribute.empty()) return;
44 itemStatsTable()[itemId].push_back(std::move(stat));
45 ++n;
46 };
47 if (root.isArray()) {
48 for (size_t i = 0; i < root.size(); ++i) addObject(root.at(i));
49 } else if (root.isObject()) {
50 addObject(root);
51 }
52 return n;
53}

61int EquipmentSystem::syncEquipModifiers(RPGActor *actor, eve::inventory::EquipmentSet *equip) {
62 if (!actor || !equip) return 0;
63 int applied = 0;
64 for (int i = 0; i < equip->getSlotCount(); ++i) {
65 const std::string slot = equip->getSlotName(i);
66 if (equip->isSlotEmpty(slot)) {
67 projectSlot(*actor, slot, {}).ignore("legacy equipment synchronization projection");
68 continue;
69 }
70 const std::string itemId = equip->getSlotItemId(slot);
71 auto it = itemStatsTable().find(itemId);
72 auto result = projectSlot(*actor, slot,
73 it == itemStatsTable().end() ? std::span<const EquipmentStat>{}
74 : std::span<const EquipmentStat>(it->second));
75 if (result) applied += result.value();
76 }
77 return applied;
78}

80eve::Result<int> EquipmentSystem::projectSlot(RPGActor& actor, std::string_view slot,
81 std::span<const EquipmentStat> stats) {
82 if (slot.empty())
83 return eve::Result<int>::failure(
84 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "Equipment slot must not be empty"));
85 auto staged = actor.attributes()->values;
86 const std::string source = "equip:" + std::string(slot);
87 auto removed = staged.removeBySource(source);
88 if (!removed) return eve::Result<int>::failure(removed.status());
89 for (const auto& stat : stats) {
90 if (stat.attribute.empty() || !std::isfinite(stat.value))
91 return eve::Result<int>::failure(
92 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "Invalid equipment statistic"));
93 auto parsed = eve::attributes::parseAttributeOperation(stat.op, stat.value);
94 AttributeModifier modifier;
95 modifier.attribute = stat.attribute;
96 modifier.source = source;
97 modifier.priority = stat.priority;
98 if (parsed) {
99 modifier.operation = parsed.value().operation;
100 modifier.value = parsed.value().value;
101 } else if (AttributeSystem::customOps().has(stat.op)) {
102 modifier.operation = AttributeOperation::Custom;
103 modifier.policyId = stat.op;
104 modifier.value = stat.value;
105 } else
106 return eve::Result<int>::failure(parsed.status());
107 auto added = staged.addModifier(std::move(modifier));
108 if (!added) return eve::Result<int>::failure(added.status());
109 }
110 actor.attributes()->values = std::move(staged);
111 return eve::Result<int>::success(static_cast<int>(stats.size()));
112}
