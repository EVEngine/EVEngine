28int EquipmentSystem::registerItemStatsFromJson(const std::string &itemId, const std::string &json) {
29 if (itemId.empty()) return 0;
30 std::string err;
31 const eve::json::Document doc = eve::json::Document::parse(json, &err);
32 if (!doc.valid()) return 0;
33 const Value root = doc.root();
34 int n = 0;
35 const auto addObject = [&](Value o) {
36 if (!o.isObject()) return;
37 EquipmentStat stat;
38 stat.attribute = o.getString("attribute");
39 stat.op = o.getString("op", "add");
40 stat.value = o.getDouble("value", 0.0);
41 stat.priority = o.getInt("priority", 0);
42 if (stat.attribute.empty()) return;
43 itemStatsTable()[itemId].push_back(std::move(stat));
44 ++n;
45 };
46 if (root.isArray()) {
47 for (size_t i = 0; i < root.size(); ++i) addObject(root.at(i));
48 } else if (root.isObject()) {
49 addObject(root);
50 }
51 return n;
52}

60int EquipmentSystem::syncEquipModifiers(RPGActor *actor, eve::inventory::EquipmentSet *equip) {
61 if (!actor || !equip) return 0;
62 int applied = 0;
63 for (int i = 0; i < equip->getSlotCount(); ++i) {
64 const std::string slot = equip->getSlotName(i);
65 const std::string source = "equip:" + slot;
66 // 先清除该槽位旧加成（覆盖换装/卸下的情况）。
67 AttributeSystem::removeAllModifiersBySource(actor, source);
68 if (equip->isSlotEmpty(slot)) continue;
69 const std::string itemId = equip->getSlotItemId(slot);
70 auto it = itemStatsTable().find(itemId);
71 if (it == itemStatsTable().end()) continue;
72 for (const auto &stat : it->second) {
73 AttributeSystem::addModifier(actor, stat.attribute, source, stat.op, stat.value,
74 stat.priority);
75 ++applied;
76 }
77 }
78 return applied;
79}
