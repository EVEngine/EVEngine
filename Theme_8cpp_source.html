67bool resolveStyle(const std::string &name, StyleClass &out,
68 std::unordered_set<std::string> &visiting) {
69 const auto it = g_styleClasses.find(name);
70 if (it == g_styleClasses.end() || !visiting.insert(name).second) return false;
71 const StyleClass &source = it->second;
72 if (!source.parent.empty() && !resolveStyle(source.parent, out, visiting)) return false;
73 overlayStyle(out, source);
74 out.name = source.name;
75 out.parent = source.parent;
76 visiting.erase(name);
77 return true;
78}

82StyleClassStatus defineStyleClass(const std::string &name, const std::string &parent) {
83 if (name.empty()) return StyleClassStatus::InvalidName;
84 if (!parent.empty() && g_styleClasses.find(parent) == g_styleClasses.end())
85 return StyleClassStatus::UnknownParent;
86 for (std::string current = parent; !current.empty();) {
87 if (current == name) return StyleClassStatus::InheritanceCycle;
88 const auto it = g_styleClasses.find(current);
89 if (it == g_styleClasses.end()) return StyleClassStatus::UnknownParent;
90 current = it->second.parent;
91 }
92 StyleClass replacement;
93 replacement.name = name;
94 replacement.parent = parent;
95 g_styleClasses[name] = std::move(replacement);
96 return StyleClassStatus::Applied;
97}

99StyleClassStatus setStyleClassColor(const std::string &name, const std::string &property, float r,
100 float g, float b, float a) {
101 const auto it = g_styleClasses.find(name);
102 if (it == g_styleClasses.end()) return StyleClassStatus::UnknownClass;
103 StyleClass &style = it->second;
104 float *color = nullptr;
105 const std::string key = toLower(property);
106 if (key == "text") {
107 style.hasTextColor = true;
108 color = style.textColor;
109 } else if (key == "background") {
110 style.hasBackgroundColor = true;
111 color = style.backgroundColor;
112 } else if (key == "border") {
113 style.hasBorderColor = true;
114 color = style.borderColor;
115 } else if (key == "accent") {
116 style.hasAccentColor = true;
117 color = style.accentColor;
118 } else {
119 return StyleClassStatus::UnknownProperty;
120 }
121 color[0] = std::clamp(r, 0.f, 1.f);
122 color[1] = std::clamp(g, 0.f, 1.f);
123 color[2] = std::clamp(b, 0.f, 1.f);
124 color[3] = std::clamp(a, 0.f, 1.f);
125 return StyleClassStatus::Applied;
126}

150StyleClassStatus resolveStyleClass(const std::string &name, StyleClass *out) {
151 if (!out || name.empty()) return StyleClassStatus::InvalidName;
152 if (g_styleClasses.find(name) == g_styleClasses.end()) return StyleClassStatus::UnknownClass;
153 StyleClass resolved;
154 std::unordered_set<std::string> visiting;
155 if (!resolveStyle(name, resolved, visiting)) return StyleClassStatus::InheritanceCycle;
156 *out = std::move(resolved);
157 return StyleClassStatus::Applied;
