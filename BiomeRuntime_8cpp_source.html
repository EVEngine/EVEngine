19EditorResult<void> BiomeDocumentRuntime::publish(const BiomeDocumentTarget& document,
20 const IBiomeSpatialResolver& resolver) {
21 const auto diagnostics = document.validate();
22 for (const auto& diagnostic : diagnostics)
23 if (diagnostic.severity() == DiagnosticSeverity::Error)
24 return EditorResult<void>::failure(eve::Status(EditorStatus::Rejected, diagnostics));
25
26 auto candidate = std::make_unique<procgen::BiomeRules>();
27 for (const auto& layer : document.layers()) {
28 auto spatial = resolver.resolve(layer.spatialAsset);
29 if (!spatial.ok()) return EditorResult<void>::failure(spatial.status());
30 if (!spatial.value())
31 return fail<void>(EditorStatus::NotFound, "editor.biome.runtime-spatial",
32 "Biome spatial resolver returned no spatial data");
33 if (!candidate->addLayer(layer.name, spatial.value(), layer.priority, layer.density))
34 return fail<void>(EditorStatus::Failed, "editor.biome.runtime-layer",
35 "Biome runtime rejected a validated layer");
36 for (const auto& asset : layer.assets)
37 if (!candidate->addAsset(layer.name, asset.asset, asset.weight, asset.minScale, asset.maxScale,
38 asset.randomYaw))
39 return fail<void>(EditorStatus::Failed, "editor.biome.runtime-asset",
40 "Biome runtime rejected a validated asset");
41 }
42 for (const auto& reference : document.exclusions()) {
43 auto spatial = resolver.resolve(reference);
44 if (!spatial.ok()) return EditorResult<void>::failure(spatial.status());
45 if (!spatial.value())
46 return fail<void>(EditorStatus::NotFound, "editor.biome.runtime-exclusion-spatial",
47 "Biome exclusion resolver returned no spatial data");
48 if (!candidate->addExclusion(spatial.value()))
49 return fail<void>(EditorStatus::Failed, "editor.biome.runtime-exclusion",
50 "Biome runtime rejected a validated exclusion");
51 }
52 rules_ = std::move(candidate);
53 revision_ = document.revision();
54 return eve::editing::applied<void>(diagnostics);
55}

57EditorResult<std::unique_ptr<procgen::PointSet>> BiomeDocumentRuntime::preview(procgen::SpatialData* domain,
58 float spacing, std::uint32_t seed,
59 float jitter,
60 Revision expectedRevision) {
61 if (expectedRevision != revision_)
62 return fail<std::unique_ptr<procgen::PointSet>>(EditorStatus::Conflict, "editor.biome.stale",
63 "Biome preview generation is stale");
64 if (!rules_ || !domain || !std::isfinite(spacing) || spacing <= 0 || !std::isfinite(jitter) || jitter < 0 ||
65 jitter > 1)
66 return fail<std::unique_ptr<procgen::PointSet>>(EditorStatus::Rejected, "editor.biome.preview",
67 "Biome preview inputs are invalid");
68 std::unique_ptr<procgen::PointSet> points(rules_->generate(domain, spacing, seed, jitter));
69 if (!points)
70 return fail<std::unique_ptr<procgen::PointSet>>(EditorStatus::Failed, "editor.biome.generate",
71 rules_->getError());
72 return eve::editing::applied<std::unique_ptr<procgen::PointSet>>(std::move(points));
73}
