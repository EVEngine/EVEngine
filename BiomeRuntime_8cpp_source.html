14EditorResult<void> BiomeDocumentRuntime::publish(const BiomeDocumentTarget& document,
15 const IBiomeSpatialResolver& resolver) {
16 const auto diagnostics = document.validate();
17 for (const auto& diagnostic : diagnostics)
18 if (diagnostic.severity() == DiagnosticSeverity::Error)
19 return EditorResult<void>::failure(eve::Status(EditorStatus::Rejected, diagnostics));
20
21 auto candidate = std::make_unique<procgen::BiomeRules>();
22 for (const auto& layer : document.layers()) {
23 auto spatial = resolver.resolve(layer.spatialAsset);
24 if (!spatial.ok()) return EditorResult<void>::failure(spatial.status());
25 if (!spatial.value())
26 return eve::editing::failed<void>(EditorStatus::NotFound, RuleId("editor.biome.runtime-spatial"),
27 "Biome spatial resolver returned no spatial data");
28 if (!candidate->addLayer(layer.name, spatial.value(), layer.priority, layer.density))
29 return eve::editing::failed<void>(EditorStatus::Failed, RuleId("editor.biome.runtime-layer"),
30 "Biome runtime rejected a validated layer");
31 for (const auto& asset : layer.assets)
32 if (!candidate->addAsset(layer.name, asset.asset, asset.weight, asset.minScale, asset.maxScale,
33 asset.randomYaw))
34 return eve::editing::failed<void>(EditorStatus::Failed, RuleId("editor.biome.runtime-asset"),
35 "Biome runtime rejected a validated asset");
36 }
37 for (const auto& reference : document.exclusions()) {
38 auto spatial = resolver.resolve(reference);
39 if (!spatial.ok()) return EditorResult<void>::failure(spatial.status());
40 if (!spatial.value())
41 return eve::editing::failed<void>(EditorStatus::NotFound, RuleId("editor.biome.runtime-exclusion-spatial"),
42 "Biome exclusion resolver returned no spatial data");
43 if (!candidate->addExclusion(spatial.value()))
44 return eve::editing::failed<void>(EditorStatus::Failed, RuleId("editor.biome.runtime-exclusion"),
45 "Biome runtime rejected a validated exclusion");
46 }
47 rules_ = std::move(candidate);
48 revision_ = document.revision();
49 return eve::editing::applied<void>(diagnostics);
50}
