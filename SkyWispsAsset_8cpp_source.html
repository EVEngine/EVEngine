292 const auto failure = [] {
293 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
294 "Invalid daylight material coefficients", "sky.daylight"));
295 };
296 for (const auto* v : {&sun, &moon, &dayTint, &duskTint, &nightTint, &glow, &controls, &stars, &twinkle})

310 }
311 for (const auto* v : {&rayDay, &rayDusk, &rayNight, &absorptionDay, &absorptionNight})
