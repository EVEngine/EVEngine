37void integrateHelm(WarshipEntity& ship, float dt, float speedScale, float turnScale) {
38 const WarshipDefinition* def = ship.definition()->def;
39 if (def == nullptr) return;
40 auto input = ship.input();
41 auto motion = ship.motion();
42 auto caps = ship.capabilities();
43
44 float thrustMul = caps->values.thrustMul;
45 if (caps->values.underpowered) thrustMul *= 0.55f;
46
47 float maxSpeed = def->maxSpeed * speedScale * thrustMul;
48 if (maxSpeed < 0.f) maxSpeed = 0.f;
49
50 if (motion->hasMove) {
51 const float dx = motion->targetX - motion->x;
52 const float dy = motion->targetY - motion->y;
53 const float dist = std::sqrt(dx * dx + dy * dy);
54 if (dist <= motion->arriveRadius) {
55 motion->hasMove = false;
56 motion->arrived = true;
57 input->throttle = 0.f;
58 input->steer = 0.f;
59 motion->speed = 0.f;
60 WarshipEvent event;
61 event.type = WarshipEventType::Arrived;
62 event.warshipId = ship.identity()->id;
63 event.defId = ship.identity()->defId;
64 event.x = motion->x;
65 event.y = motion->y;
66 WarshipSystem::pushEvent(std::move(event));
67 } else {
68 const float desired = normalizeHeading(std::atan2(dy, dx) * 180.f / kPi);
69 const float delta = shortestAngleDelta(motion->heading, desired);
70 input->steer = std::clamp(delta / 45.f, -1.f, 1.f);
71 input->throttle = dist < motion->arriveRadius * 3.f ? 0.45f : 1.f;
72 motion->arrived = false;
73 }
74 }
75
76 const float turn = def->turnRate * turnScale * input->steer * dt;
77 motion->heading = normalizeHeading(motion->heading + turn);
78
79 float targetSpeed = maxSpeed * std::clamp(input->throttle, -1.f, 1.f);
80 const float accel = def->accel * speedScale * (targetSpeed >= motion->speed ? 1.f : 1.5f);
81 if (motion->speed < targetSpeed)
82 motion->speed = std::min(motion->speed + accel * dt, targetSpeed);
83 else
84 motion->speed = std::max(motion->speed - accel * dt, targetSpeed);
85
86 const float rad = motion->heading * kPi / 180.f;
87 motion->x += std::cos(rad) * motion->speed * dt;
88 motion->y += std::sin(rad) * motion->speed * dt;
89}

174void WarshipSystem::recomputeCapabilities(WarshipEntity& ship) {
175 auto layout = ship.layout();
176 auto caps = ship.capabilities();
177
178 WarshipCapabilities values;
179 values.thrustMul = 1.f;
180 const bool hadUnderpowered = caps->values.underpowered;
181
182 for (const CompartmentCell& cell : layout->cells) {
183 if (cell.functionId.empty()) continue;
184 const CompartmentFunctionDef* fn = WarshipRegistry::findFunction(cell.functionId);
185 if (fn == nullptr) continue;
186 values.powerNet += fn->powerProvide - fn->powerDraw;
187 values.thrustMul += fn->thrustBonus;
188 values.hangarCapacity += fn->hangarCapacity;
189 values.weaponBayCount += fn->weaponBayCount;
190 values.crewCapacity += fn->crewCapacity;
191 values.sailPower += fn->sailPower;
192 }
193 if (values.thrustMul < 0.05f) values.thrustMul = 0.05f;
194 values.underpowered = values.powerNet < 0.f;
195 caps->values = values;
196 ++caps->revision;
197
198 if (values.underpowered && !hadUnderpowered) {
199 WarshipEvent event;
200 event.type = WarshipEventType::Underpowered;
201 event.warshipId = ship.identity()->id;
202 event.defId = ship.identity()->defId;
203 event.detail = "power_deficit";
204 pushEvent(std::move(event));
205 }
206}

259Result<void> WarshipSystem::rearrangeFromJson(WarshipEntity& ship, const std::string& json) {
260 auto layout = ship.layout();
261 auto parsed = Value::fromJson(json);
262 if (!parsed) return Result<void>::failure(parsed.status());
263 const auto* array = parsed.value().getIf<Value::Array>();
264 if (array == nullptr) {
265 return Result<void>::failure(
266 Diagnostic::error(DiagnosticCode::InvalidArgument, "layout JSON must be an array", "warship.layout"));
267 }
268
269 std::vector<CompartmentCell> next(layout->cells.size());
270 for (const Value& entry : *array) {
271 const auto* object = entry.getIf<Value::Object>();
272 if (object == nullptr) {
273 return Result<void>::failure(
274 Diagnostic::error(DiagnosticCode::InvalidArgument, "layout entries must be objects", "warship.layout"));
275 }
276 auto readCoord = [&](const char* key) -> Result<int> {
277 auto it = object->find(key);
278 if (it == object->end()) {
279 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
280 "layout entry missing coordinate", "warship.layout"));
281 }
282 return readIntValue(it->second);
283 };
284 auto deckR = readCoord("deck");
285 if (!deckR) return Result<void>::failure(deckR.status());
286 auto rowR = readCoord("row");
287 if (!rowR) return Result<void>::failure(rowR.status());
288 auto colR = readCoord("col");
289 if (!colR) return Result<void>::failure(colR.status());
290
291 std::string functionKey = "function";
292 if (object->find("function") == object->end()) functionKey = "functionId";
293 auto functionId = readStringMember(*object, functionKey);
294 if (!functionId) return Result<void>::failure(functionId.status());
295 if (functionId.value().empty() || !WarshipRegistry::hasFunction(functionId.value())) {
296 return Result<void>::failure(
297 Diagnostic::error(DiagnosticCode::InvalidArgument, "unknown compartment function", "warship.layout"));
298 }
299 const int index = layout->grid.indexOf(deckR.value(), rowR.value(), colR.value());
300 if (index < 0 || index >= static_cast<int>(next.size())) {
301 return Result<void>::failure(
302 Diagnostic::error(DiagnosticCode::InvalidArgument, "layout cell out of range", "warship.layout"));
303 }
304 if (!next[static_cast<std::size_t>(index)].functionId.empty()) {
305 return Result<void>::failure(
306 Diagnostic::error(DiagnosticCode::InvalidArgument, "duplicate layout cell", "warship.layout"));
307 }
308 next[static_cast<std::size_t>(index)].functionId = std::move(functionId).takeValue();
309 }
310
311 layout->cells = std::move(next);
312 recomputeCapabilities(ship);
313 WarshipEvent event;
314 event.type = WarshipEventType::LayoutChanged;
315 event.warshipId = ship.identity()->id;
316 event.defId = ship.identity()->defId;
317 event.detail = "rearranged";
318 pushEvent(std::move(event));
319 return Result<void>::success();
320}

363void WarshipSystem::update(float dt) {
364 if (!std::isfinite(dt) || dt < 0.f) return;
365 ensureBuiltinMobility();
366 ecs::Table* table = ecs::current();
367 if (table == nullptr) return;
368
369 ecs::IComponentManager& cm = table->getOrCreateManager<WarshipEntity>();
370 auto* reg = cm.getOrCreateRegistryComponentBuffer<WarshipEntity>();
371 std::vector<ecs::IComponentBuffer*> stack;
372 stack.push_back(reg);
373 while (!stack.empty()) {
374 ecs::IComponentBuffer* buf = stack.back();
375 stack.pop_back();
376 auto* r = dynamic_cast<ecs::IRegistryComponentBuffer*>(buf);
377 if (r != nullptr) {
378 for (uint32_t i = 0; i < r->entity_count(); ++i) {
379 ecs::Entity* ent = r->entity_at(i);
380 if (ent == nullptr || !ecs::is_entity_visible(ent)) continue;
381 auto* ship = static_cast<WarshipEntity*>(ent);
382 if (ship->stateFlags()->destroyed) continue;
383 const WarshipDefinition* def = ship->definition()->def;
384 if (def == nullptr) continue;
385 auto it = mobilityTable().find(def->mobility);
386 if (it == mobilityTable().end() || !it->second) {
387 it = mobilityTable().find(kMobilitySurface);
388 }
389 if (it != mobilityTable().end() && it->second) it->second->update(*ship, dt);
390 }
391 }
392 if (buf->children != nullptr) stack.push_back(buf->children);
393 if (buf->next != nullptr) stack.push_back(buf->next);
394 }
395}
