122inline float linearToPq(float v) noexcept {
123 v = std::max(v, 0.f);
124 constexpr float m1 = 2610.f / 4096.f / 4.f;
125 constexpr float m2 = 2523.f / 4096.f * 128.f;
126 constexpr float c1 = 3424.f / 4096.f;
127 constexpr float c2 = 2413.f / 4096.f * 32.f;
128 constexpr float c3 = 2392.f / 4096.f * 32.f;
129 const float cp = std::pow(v, m1);
131 return std::pow((c1 + c2 * cp) / (1.f + c3 * cp), m2);
132}

138inline void encodeHdr10(float &r, float &g, float &b, float paperWhiteNits, float peakNits) noexcept {
140 clampCalibration(paperWhiteNits, peakNits);
141 const float peakRatio = peakNits / paperWhiteNits;
142 r = std::clamp(r, 0.f, peakRatio);
143 g = std::clamp(g, 0.f, peakRatio);
144 b = std::clamp(b, 0.f, peakRatio);
146 rec709ToRec2020(r, g, b);
147 constexpr float st2084Max = 10000.f;
148 r = linearToPq(r * paperWhiteNits / st2084Max);
149 g = linearToPq(g * paperWhiteNits / st2084Max);
150 b = linearToPq(b * paperWhiteNits / st2084Max);

157inline void acesToDisplayLinear(float &r, float &g, float &b, float paperWhiteNits,
158 float peakNits) noexcept {
160 clampCalibration(paperWhiteNits, peakNits);
161 const float peakRatio = peakNits / paperWhiteNits;
162 const float inv = 1.f / peakRatio;
163 auto aces = [](float c) {
164 c = std::max(c, 0.f);
165 constexpr float a = 2.51f, b = 0.03f, c0 = 2.43f, d = 0.59f, e = 0.14f;
167 return std::clamp((c * (a * c + b)) / (c * (c0 * c + d) + e), 0.f, 1.f);
168 };
169 r = aces(r * inv) * peakRatio;
170 g = aces(g * inv) * peakRatio;
171 b = aces(b * inv) * peakRatio;
172}
