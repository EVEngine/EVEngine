38bool BuildingDefinition::maskAt(int localX, int localY) const {
39 if (localX < 0 || localY < 0 || localX >= footprintW || localY >= footprintH) return false;
40 if (footprintMask.empty()) return true;
41 const size_t idx = size_t(localY) * size_t(footprintW) + size_t(localX);

64void BuildingRegistry::registerBuilding(const BuildingDefinition &def) {
65 if (def.id.empty()) return;
66 BuildingDefinition copy = def;
67 if (copy.footprintW <= 0) copy.footprintW = 1;
68 if (copy.footprintH <= 0) copy.footprintH = 1;
69 if (copy.displayName.empty()) copy.displayName = copy.id;
70 if (copy.snapMode.empty()) copy.snapMode = "grid";
71 if (copy.rotationMode.empty()) copy.rotationMode = "cardinal";
72 if (copy.validateRule.empty()) copy.validateRule = "default";
73 const size_t expected = size_t(copy.footprintW) * size_t(copy.footprintH);
74 if (!copy.footprintMask.empty() && copy.footprintMask.size() != expected) {
75 copy.footprintMask.clear(); // 长度不匹配时回退实心矩形
76 }
77 table()[copy.id] = std::move(copy);
78}

94BuildingDefinition parseBuildingObject(Value o) {
95 BuildingDefinition def;
96 if (!o.isObject()) return def;
97 def.id = o.getString("id");
98 def.displayName = o.getString("displayName", def.id);
99 def.category = o.getString("category");
100 def.channel = o.getString("channel");
101 def.renderMode = o.getString("renderMode");
102 def.footprintW = o.getInt("footprintW", 1);
103 def.footprintH = o.getInt("footprintH", 1);
104 def.snapMode = o.getString("snapMode", "grid");
105 def.rotationMode = o.getString("rotationMode", "cardinal");
106 def.validateRule = o.getString("validateRule", "default");
107 def.tags = o.getStringArray("tags");
108 def.requireTerrain = o.getIntArray("requireTerrain");
109 def.forbidTerrain = o.getIntArray("forbidTerrain");
110 def.requireAdjacentTag = o.getString("requireAdjacentTag");
111 def.requireAdjacentTerrain = o.getInt("requireAdjacentTerrain", -1);
112 def.cost = o.getIntMap("cost");
113 def.extra = o.getStringMap("extra");
114 def.visual2d = o.getStringMap("visual2d");
115 def.visual3d = o.getStringMap("visual3d");
116
117 const Value mask = o.get("footprintMask");
118 def.footprintMask.reserve(mask.size());
119 for (size_t i = 0; i < mask.size(); ++i)
120 def.footprintMask.push_back(uint8_t(mask.at(i).asInt(0) != 0 ? 1 : 0));
121 return def;
122}

126int BuildingRegistry::loadFromJson(const std::string &json, std::string *error) {
127 std::string err;
128 const eve::json::Document doc = eve::json::Document::parse(json, &err);
129 if (!doc.valid()) {
130 if (error) *error = err.empty() ? "invalid json" : err;
131 return 0;
132 }
133
134 const Value root = doc.root();
135 int n = 0;
136 if (root.isArray()) {
137 for (size_t i = 0; i < root.size(); ++i) {
138 BuildingDefinition def = parseBuildingObject(root.at(i));
139 if (def.id.empty()) continue;
140 registerBuilding(def);
141 ++n;
142 }
143 } else if (root.isObject()) {
144 BuildingDefinition def = parseBuildingObject(root);
145 if (!def.id.empty()) {
146 registerBuilding(def);
147 ++n;
148 }
149 }
150 return n;
151}
