27inline void lightSpotCosines(float angleDeg, float softness, float &cosOuter, float &cosInner) {
28 constexpr float kPi = 3.14159265358979323846f;
29 const float outerDeg = std::clamp(angleDeg, 0.1f, 89.f);
30 const float soft = std::clamp(softness, 0.f, 1.f);
31 const float outerRad = outerDeg * (kPi / 180.f);
32 const float innerRad = outerRad * (1.f - soft);
33 cosOuter = std::cos(outerRad);
34 cosInner = std::cos(innerRad);
35 if (cosInner < cosOuter + 1e-4f) cosInner = cosOuter + 1e-4f;
36}
