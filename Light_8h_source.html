28inline void lightSpotCosines(float angleDeg, float softness, float &cosOuter, float &cosInner) {
29 constexpr float kPi = 3.14159265358979323846f;
30 const float outerDeg = std::clamp(angleDeg, 0.1f, 89.f);
31 const float soft = std::clamp(softness, 0.f, 1.f);
32 const float outerRad = outerDeg * (kPi / 180.f);
33 const float innerRad = outerRad * (1.f - soft);
34 cosOuter = std::cos(outerRad);
35 cosInner = std::cos(innerRad);
36 if (cosInner < cosOuter + 1e-4f) cosInner = cosOuter + 1e-4f;
37}
