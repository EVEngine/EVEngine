18int SpriteSheet::addFrameTrimmed(const std::string &name, int x, int y, int w, int h,
19 int sourceW, int sourceH, int offsetX, int offsetY) {
20 if (w <= 0 || h <= 0) throw Exception("SpriteSheet.addFrame: width/height must be > 0");
21 Frame f;
22 f.name = name.empty() ? ("frame" + std::to_string(frames_.size())) : name;
23 if (byName_.count(f.name))
24 throw Exception("SpriteSheet.addFrame: duplicate frame name '%s'", f.name.c_str());
25 f.x = x;
26 f.y = y;
27 f.w = w;
28 f.h = h;
29 f.sourceW = sourceW;
30 f.sourceH = sourceH;
31 f.offsetX = offsetX;
32 f.offsetY = offsetY;
33 int idx = static_cast<int>(frames_.size());
34 byName_[f.name] = idx;
35 frames_.push_back(std::move(f));
36 return idx;
37}

44int SpriteSheet::setGrid(int columns, int rows, int frameW, int frameH, int margin, int spacing,
45 int originX, int originY) {
46 if (columns <= 0 || rows <= 0)
47 throw Exception("SpriteSheet.setGrid: columns/rows must be > 0");
48 if (frameW <= 0 || frameH <= 0)
49 throw Exception("SpriteSheet.setGrid: frameW/frameH must be > 0");
50 if (margin < 0 || spacing < 0)
51 throw Exception("SpriteSheet.setGrid: margin/spacing must be >= 0");
52
53 clear();
54 int added = 0;
55 for (int row = 0; row < rows; ++row) {
56 for (int col = 0; col < columns; ++col) {
57 int x = originX + margin + col * (frameW + spacing);
58 int y = originY + margin + row * (frameH + spacing);
59 addFrame("", x, y, frameW, frameH);
60 ++added;
61 }
62 }
63 return added;
64}
