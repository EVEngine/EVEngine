11[[nodiscard]] float approach(float current, float target, float maxDelta) noexcept {
12 const float delta = target - current;
13 if (std::fabs(delta) <= maxDelta) return target;
14 return current + std::copysign(maxDelta, delta);
15}

87glm::vec3 SceneWind::sample(const glm::vec3& world, float timeSeconds) const noexcept {
88 const float t = timeSeconds * curlTimeScale_;
89 const float f = curlFrequency_;
90 // Analytic curl of a smooth potential → locally swirling, nearly div-free.
91 const float sx = std::sin(f * world.y + 0.7f * t);
92 const float sy = std::sin(f * world.z + 1.1f * t);
93 const float sz = std::sin(f * world.x + 0.5f * t);
94 const float cx = std::cos(f * world.z + 0.9f * t);
95 const float cy = std::cos(f * world.x + 0.3f * t);
96 const float cz = std::cos(f * world.y + 1.3f * t);
97 const glm::vec3 curl(currentCurl_ * (sy - cz), currentCurl_ * (sz - cx),
98 currentCurl_ * (sx - cy));
99 return currentMain_ + curl;
100}
