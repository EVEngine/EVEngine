47ProjectileVector steer(ProjectileVector velocity, ProjectileVector desired, double maxRadians) {
48 const double speed = length(velocity);
49 if (speed <= 0.0) return velocity;
50 ProjectileVector current = normalized(velocity);
51 desired = normalized(desired);
52 const double dot = std::clamp(current.x * desired.x + current.y * desired.y + current.z * desired.z, -1.0, 1.0);
53 const double angle = std::acos(dot);
54 if (angle <= maxRadians || angle <= 1e-12) return {desired.x * speed, desired.y * speed, desired.z * speed};
55 const double alpha = maxRadians / angle;
56 ProjectileVector blended{current.x + (desired.x - current.x) * alpha, current.y + (desired.y - current.y) * alpha,
57 current.z + (desired.z - current.z) * alpha};
58 blended = normalized(blended);
59 return {blended.x * speed, blended.y * speed, blended.z * speed};
60}

109Result<ProjectileHandle> ProjectileRuntime::spawn(const ProjectileDefinition& definition,
110 const ProjectileSpawnRequest& request) {
111 auto valid = validateSpawn(definition, request);
112 if (!valid) return Result<ProjectileHandle>::failure(valid.status());
113 auto free = std::find_if(slots_.begin(), slots_.end(), [](const Slot& slot) { return !slot.state.has_value(); });
114 if (free == slots_.end())
115 return projectileValueError<ProjectileHandle>(DiagnosticCode::Conflict, "Projectile pool is exhausted");
116 const std::size_t index = static_cast<std::size_t>(std::distance(slots_.begin(), free));
117 ++free->generation;
118 if (free->generation == 0) ++free->generation;
119 const ProjectileVector direction = normalized(request.direction);
120 ProjectileState state;
121 state.handle = {static_cast<std::uint32_t>(index), free->generation};
122 state.definitionId = definition.id;
123 state.mode = definition.mode;
124 state.position = request.position;
125 state.velocity = {direction.x * definition.speed, direction.y * definition.speed, direction.z * definition.speed};
126 state.gravity = definition.gravity;
127 state.maxTurnRateDegrees = definition.maxTurnRateDegrees;
128 state.lifetime = definition.lifetime;
129 state.target = request.target;
130 free->state = state;
131 ++activeCount_;
132 return Result<ProjectileHandle>::success(state.handle);
133}

135Result<ProjectileUpdate> ProjectileRuntime::update(Duration delta, const IProjectileTargetProvider* targets) {
136 if (delta < Duration::zero())
137 return projectileValueError<ProjectileUpdate>(DiagnosticCode::InvalidArgument,
138 "Projectile delta must be non-negative", "delta");
139 if (delta.isZero()) return Result<ProjectileUpdate>::success({}, Status::success(StatusCode::NoOp));
140
141 std::vector<Slot> staged = slots_;
142 ProjectileUpdate update;
143 const double seconds = delta.seconds();
144 for (Slot& slot : staged) {
145 if (!slot.state) continue;
146 ProjectileState& state = *slot.state;
147 auto nextAge = state.age.tryAdd(delta);
148 if (!nextAge) return Result<ProjectileUpdate>::failure(nextAge.status());
149 state.age = std::move(nextAge).takeValue();
150 if (state.age >= state.lifetime) {
151 update.released.push_back(state.handle);
152 slot.state.reset();
153 continue;
154 }
155 if (state.mode == ProjectileMode::Homing) {
156 if (!targets)
157 return projectileValueError<ProjectileUpdate>(DiagnosticCode::Unsupported,
158 "Homing projectile target provider is unavailable");
159 auto target = targets->position(*state.target);
160 if (!target) return Result<ProjectileUpdate>::failure(target.status());
161 ProjectileVector desired = directionTo(state.position, target.value());
162 if (length(desired) <= 1e-12)
163 return projectileValueError<ProjectileUpdate>(DiagnosticCode::Conflict,
164 "Homing projectile overlaps its target");
165 state.velocity = steer(state.velocity, desired, state.maxTurnRateDegrees * kPi / 180.0 * seconds);
166 }
167 if (state.mode == ProjectileMode::Ballistic) state.velocity.y -= state.gravity * seconds;
168 state.position.x += state.velocity.x * seconds;
169 state.position.y += state.velocity.y * seconds;
170 state.position.z += state.velocity.z * seconds;
171 update.advanced.push_back(state.handle);
172 }
173 slots_ = std::move(staged);
174 activeCount_ -= update.released.size();
175 return Result<ProjectileUpdate>::success(std::move(update));
176}

209Result<void> ProjectileRuntime::restore(const ProjectileRuntimeSnapshot& snapshot) {
210 if (snapshot.slots.empty() || snapshot.slots.size() > 1048576)
211 return projectileError(DiagnosticCode::InvalidArgument,
212 "Projectile snapshot pool capacity is outside 1..1048576");
213 std::vector<Slot> staged(snapshot.slots.size());
214 std::size_t active = 0;
215 for (std::size_t index = 0; index < snapshot.slots.size(); ++index) {
216 const auto& input = snapshot.slots[index];
217 staged[index].generation = input.generation;
218 if (!input.state) continue;
219 const auto& state = *input.state;
220 const auto finitePoint = [](const ProjectilePoint& point) {
221 return std::isfinite(point.x) && std::isfinite(point.y) && std::isfinite(point.z);
222 };
223 const auto finiteVector = [](const ProjectileVector& value) {
224 return std::isfinite(value.x) && std::isfinite(value.y) && std::isfinite(value.z);
225 };
226 if (state.handle.slot != index || state.handle.generation != input.generation ||
227 !state.definitionId.isValid() || !finitePoint(state.position) || !finiteVector(state.velocity) ||
228 !std::isfinite(state.gravity) || state.gravity < 0.0 ||
229 !std::isfinite(state.maxTurnRateDegrees) || state.maxTurnRateDegrees < 0.0 ||
230 state.age.nanoseconds() < 0 || state.lifetime.nanoseconds() <= 0 || state.age >= state.lifetime ||
231 (state.mode == ProjectileMode::Homing && !state.target))
232 return projectileError(DiagnosticCode::InvalidArgument,
233 "Projectile snapshot contains an invalid live slot");
234 staged[index].state = state;
235 ++active;
236 }
237 slots_ = std::move(staged);
238 activeCount_ = active;
239 return Result<void>::success();
240}
