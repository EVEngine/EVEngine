94eve::Result<EffectRuntimeState> parseState(const eve::definitions::Definition& definition,
95 const eve::DefinitionRef& reference) {
96 auto parsed = eve::Value::fromJson(definition.json);
97 if (!parsed) return eve::Result<EffectRuntimeState>::failure(parsed.status());
98 auto value = std::move(parsed).takeValue();
99 const auto* object = value.getIf<eve::Value::Object>();
100 if (!object) return invalid<EffectRuntimeState>("effect definition payload must be an object", "json");
101
102 EffectRuntimeState state;
103 state.stackKey = std::string(reference.id().name());
104 if (const auto* value = field(*object, "stackKey")) {
105 const auto* text = value->getIf<std::string>();
106 if (!text) return invalid<EffectRuntimeState>("effect stackKey must be a string", "stackKey");
107 state.stackKey = *text;
108 }
109 if (const auto* value = field(*object, "priority")) {
110 if (!readInt(*object, "priority", state.priority))
111 return invalid<EffectRuntimeState>("effect priority must be Int64", "priority");
112 }
113 if (const auto* value = field(*object, "duration")) {
114 (void)value;
115 if (!readDouble(*object, "duration", state.duration))
116 return invalid<EffectRuntimeState>("effect duration must be finite", "duration");
117 }
118 if (const auto* value = field(*object, "magnitude")) {
119 (void)value;
120 if (!readDouble(*object, "magnitude", state.magnitude))
121 return invalid<EffectRuntimeState>("effect magnitude must be finite", "magnitude");
122 }
123 if (const auto* value = field(*object, "stackCount")) {
124 (void)value;
125 if (!readUInt(*object, "stackCount", state.stackCount) || state.stackCount == 0)
126 return invalid<EffectRuntimeState>("effect stackCount must be positive", "stackCount");
127 }
128 if (const auto* value = field(*object, "maxStacks")) {
129 (void)value;
130 if (!readUInt(*object, "maxStacks", state.maxStacks))
131 return invalid<EffectRuntimeState>("effect maxStacks must be UInt32", "maxStacks");
132 }
133 if (state.maxStacks != 0 && state.stackCount > state.maxStacks)
134 return invalid<EffectRuntimeState>("effect stackCount exceeds maxStacks", "stackCount");
135 constexpr const char* stackModeNames[] = {"replace", "new_instance", "reuse", "accumulate"};
136 constexpr StackMode stackModeValues[] = {StackMode::Replace, StackMode::NewInstance, StackMode::Reuse,
137 StackMode::Accumulate};
138 if (!readOptionalEnum(*object, "stackMode", stackModeNames, stackModeValues, std::size(stackModeNames),
139 state.policy.stackMode))
140 return invalid<EffectRuntimeState>("effect stackMode is invalid", "stackMode");
141 constexpr const char* stackCountPolicyNames[] = {"keep", "increment", "set"};
142 constexpr StackCountPolicy stackCountPolicyValues[] = {StackCountPolicy::Keep, StackCountPolicy::Increment,
143 StackCountPolicy::Set};
144 if (!readOptionalEnum(*object, "stackCountPolicy", stackCountPolicyNames, stackCountPolicyValues,
145 std::size(stackCountPolicyNames), state.policy.stackCount))
146 return invalid<EffectRuntimeState>("effect stackCountPolicy is invalid", "stackCountPolicy");
147 constexpr const char* durationPolicyNames[] = {"keep", "replace", "extend"};
148 constexpr DurationPolicy durationPolicyValues[] = {DurationPolicy::Keep, DurationPolicy::Replace,
149 DurationPolicy::Extend};
150 if (!readOptionalEnum(*object, "durationPolicy", durationPolicyNames, durationPolicyValues,
151 std::size(durationPolicyNames), state.policy.duration))
152 return invalid<EffectRuntimeState>("effect durationPolicy is invalid", "durationPolicy");
153 constexpr const char* magnitudePolicyNames[] = {"keep", "replace", "add", "max"};
154 constexpr MagnitudePolicy magnitudePolicyValues[] = {MagnitudePolicy::Keep, MagnitudePolicy::Replace,
155 MagnitudePolicy::Add, MagnitudePolicy::Max};
156 if (!readOptionalEnum(*object, "magnitudePolicy", magnitudePolicyNames, magnitudePolicyValues,
157 std::size(magnitudePolicyNames), state.policy.magnitude))
158 return invalid<EffectRuntimeState>("effect magnitudePolicy is invalid", "magnitudePolicy");
159 constexpr const char* overflowPolicyNames[] = {"reject", "clamp", "replace_oldest"};
160 constexpr OverflowPolicy overflowPolicyValues[] = {OverflowPolicy::Reject, OverflowPolicy::Clamp,
161 OverflowPolicy::ReplaceOldest};
162 if (!readOptionalEnum(*object, "overflowPolicy", overflowPolicyNames, overflowPolicyValues,
163 std::size(overflowPolicyNames), state.policy.overflow))
164 return invalid<EffectRuntimeState>("effect overflowPolicy is invalid", "overflowPolicy");
165 state.policy.maxStacks = state.maxStacks;
166 state.remaining = state.duration > 0.0 ? state.duration : -1.0;
167
168 if (const auto* value = field(*object, "tags")) {
169 const auto* array = value->getIf<eve::Value::Array>();
170 if (!array) return invalid<EffectRuntimeState>("effect tags must be an array", "tags");
171 for (const auto& item : *array) {
172 const auto* tag = item.getIf<std::string>();
173 if (!tag || tag->empty())
174 return invalid<EffectRuntimeState>("effect tag must be a non-empty string", "tags");
175 state.tags.push_back(*tag);
176 }
177 std::sort(state.tags.begin(), state.tags.end());
178 state.tags.erase(std::unique(state.tags.begin(), state.tags.end()), state.tags.end());
179 }
180 if (const auto* value = field(*object, "payload")) {
181 const auto* payload = value->getIf<eve::Value::Object>();
182 if (!payload) return invalid<EffectRuntimeState>("effect payload must be an object", "payload");
183 for (const auto& [key, item] : *payload) {
184 auto json = item.toJson();
185 if (!json) return eve::Result<EffectRuntimeState>::failure(json.status());
186 auto stored = state.payload.setJson(key, std::move(json).takeValue());
187 if (!stored.ok())
188 return invalid<EffectRuntimeState>("effect payload member could not be stored", "payload." + key);
189 }
190 }
191 return eve::Result<EffectRuntimeState>::success(std::move(state));
192}

204eve::Result<EffectDefinitionRuntime> EffectDefinitionRuntime::create(eve::definitions::DefinitionRegistry& registry,
205 eve::DefinitionRef definition, std::string subject,
206 std::string source, eve::PersistentId instanceId,
207 eve::definition::ReloadPolicy policy) {
208 if (!definition.id().isValid()) return invalid<EffectDefinitionRuntime>("effect definition reference is invalid");
209 if (subject.empty()) return invalid<EffectDefinitionRuntime>("effect subject must not be empty", "subject");
210 auto handle = currentHandle(registry, definition);
211 if (!handle) return eve::Result<EffectDefinitionRuntime>::failure(handle.status());
212 auto state = resolveState(registry, definition, handle.value());
213 if (!state) return eve::Result<EffectDefinitionRuntime>::failure(state.status());
214 auto runtime = eve::definition::RuntimeInstance<EffectRuntimeState>::create(
215 instanceId, std::move(definition), handle.value().generation, std::move(state).takeValue());
216 if (!runtime) return eve::Result<EffectDefinitionRuntime>::failure(runtime.status());
217 return eve::Result<EffectDefinitionRuntime>::success(EffectDefinitionRuntime(
218 registry, std::move(runtime).takeValue(), std::move(subject), std::move(source), policy));
219}

237eve::Result<void> EffectDefinitionRuntime::applyTo(EffectInstance* effect) const {
238 if (effect == nullptr)
239 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
240 "effect runtime cannot project to a null instance",
241 "effect", {}, "effects.definition_runtime"));
242 EffectInstance candidate = *effect;
243 candidate.id = identity().instanceId.format();
244 candidate.subject = subject_;
245 candidate.type = identity().definition.id().name();
246 candidate.source = source_;
247 candidate.stackKey = state().stackKey;
248 candidate.priority = state().priority;
249 candidate.duration = state().duration;
250 candidate.remaining = state().remaining;
251 candidate.magnitude = state().magnitude;
252 candidate.stackCount = state().stackCount;
253 candidate.policy = state().policy;
254 candidate.payload = state().payload;
255 candidate.tags = state().tags;
256 candidate.identity = eve::EffectId::fromUuid(identity().instanceId);
257 candidate.definitionIdentity = identity();
258 candidate.reloadPolicy = policy_;
259 using std::swap;
260 swap(*effect, candidate);
261 return eve::Result<void>::success();
262}

264eve::Result<eve::definition::ReloadOutcome> EffectDefinitionRuntime::reload(eve::definition::ReloadPolicy policy) {
265 if (registry_ == nullptr)
266 return invalid<eve::definition::ReloadOutcome>("effect definition registry is not bound", "registry");
267 auto next = currentHandle(*registry_, identity().definition);
268 if (!next) return eve::Result<eve::definition::ReloadOutcome>::failure(next.status());
269 auto defaults = resolveState(*registry_, identity().definition, next.value());
270 if (!defaults) return eve::Result<eve::definition::ReloadOutcome>::failure(defaults.status());
271 EffectRuntimeState defaultState = std::move(defaults).takeValue();
272 auto rebuild = [defaultState](const EffectRuntimeState& oldState, const eve::definition::InstanceIdentity&,
273 const eve::definition::DefinitionHandle&) mutable -> eve::Result<EffectRuntimeState> {
274 EffectRuntimeState rebuilt = defaultState;
275 if (oldState.remaining >= 0.0)
276 rebuilt.remaining = rebuilt.duration > 0.0 ? std::min(oldState.remaining, rebuilt.duration) : -1.0;
277 rebuilt.stackCount =
278 rebuilt.maxStacks == 0 ? oldState.stackCount : std::min(oldState.stackCount, rebuilt.maxStacks);
279 if (rebuilt.stackCount == 0) rebuilt.stackCount = 1;
280 return eve::Result<EffectRuntimeState>::success(std::move(rebuilt));
281 };
282 auto result = runtime_.reload(next.value(), policy, defaultState, std::move(rebuild));
283 if (result.ok()) policy_ = policy;
284 return result;
285}
