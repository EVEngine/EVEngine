83eve::Result<EffectRuntimeState> parseState(const eve::definitions::Definition& definition,
84 const eve::DefinitionRef& reference) {
85 auto parsed = eve::Value::fromJson(definition.json);
86 if (!parsed) return eve::Result<EffectRuntimeState>::failure(parsed.status());
87 auto value = std::move(parsed).takeValue();
88 const auto* object = value.getIf<eve::Value::Object>();
89 if (!object)
90 return eve::Result<EffectRuntimeState>::failure(
91 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect definition payload must be an object",
92 "json", {}, "effects.definition_runtime"));
93
94 EffectRuntimeState state;
95 state.stackKey = std::string(reference.id().name());
96 if (const auto* value = field(*object, "stackKey")) {
97 const auto* text = value->getIf<std::string>();
98 if (!text)
99 return eve::Result<EffectRuntimeState>::failure(
100 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect stackKey must be a string",
101 "stackKey", {}, "effects.definition_runtime"));
102 state.stackKey = *text;
103 }
104 if (const auto* value = field(*object, "priority")) {
105 if (!readInt(*object, "priority", state.priority))
106 return eve::Result<EffectRuntimeState>::failure(
107 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect priority must be Int64",
108 "priority", {}, "effects.definition_runtime"));
109 }
110 if (const auto* value = field(*object, "duration")) {
111 (void)value;
112 if (!readDouble(*object, "duration", state.duration))
113 return eve::Result<EffectRuntimeState>::failure(
114 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect duration must be finite",
115 "duration", {}, "effects.definition_runtime"));
116 }
117 if (const auto* value = field(*object, "magnitude")) {
118 (void)value;
119 if (!readDouble(*object, "magnitude", state.magnitude))
120 return eve::Result<EffectRuntimeState>::failure(
121 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect magnitude must be finite",
122 "magnitude", {}, "effects.definition_runtime"));
123 }
124 if (const auto* value = field(*object, "stackCount")) {
125 (void)value;
126 if (!readUInt(*object, "stackCount", state.stackCount) || state.stackCount == 0)
127 return eve::Result<EffectRuntimeState>::failure(
128 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect stackCount must be positive",
129 "stackCount", {}, "effects.definition_runtime"));
130 }
131 if (const auto* value = field(*object, "maxStacks")) {
132 (void)value;
133 if (!readUInt(*object, "maxStacks", state.maxStacks))
134 return eve::Result<EffectRuntimeState>::failure(
135 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect maxStacks must be UInt32",
136 "maxStacks", {}, "effects.definition_runtime"));
137 }
138 if (state.maxStacks != 0 && state.stackCount > state.maxStacks)
139 return eve::Result<EffectRuntimeState>::failure(
140 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect stackCount exceeds maxStacks",
141 "stackCount", {}, "effects.definition_runtime"));
142 constexpr const char* stackModeNames[] = {"replace", "new_instance", "reuse", "accumulate"};
143 constexpr StackMode stackModeValues[] = {StackMode::Replace, StackMode::NewInstance, StackMode::Reuse,
144 StackMode::Accumulate};
145 if (!readOptionalEnum(*object, "stackMode", stackModeNames, stackModeValues, std::size(stackModeNames),
146 state.policy.stackMode))
147 return eve::Result<EffectRuntimeState>::failure(
148 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect stackMode is invalid", "stackMode", {},
149 "effects.definition_runtime"));
150 constexpr const char* stackCountPolicyNames[] = {"keep", "increment", "set"};
151 constexpr StackCountPolicy stackCountPolicyValues[] = {StackCountPolicy::Keep, StackCountPolicy::Increment,
152 StackCountPolicy::Set};
153 if (!readOptionalEnum(*object, "stackCountPolicy", stackCountPolicyNames, stackCountPolicyValues,
154 std::size(stackCountPolicyNames), state.policy.stackCount))
155 return eve::Result<EffectRuntimeState>::failure(
156 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect stackCountPolicy is invalid",
157 "stackCountPolicy", {}, "effects.definition_runtime"));
158 constexpr const char* durationPolicyNames[] = {"keep", "replace", "extend"};
159 constexpr DurationPolicy durationPolicyValues[] = {DurationPolicy::Keep, DurationPolicy::Replace,
160 DurationPolicy::Extend};
161 if (!readOptionalEnum(*object, "durationPolicy", durationPolicyNames, durationPolicyValues,
162 std::size(durationPolicyNames), state.policy.duration))
163 return eve::Result<EffectRuntimeState>::failure(
164 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect durationPolicy is invalid",
165 "durationPolicy", {}, "effects.definition_runtime"));
166 constexpr const char* magnitudePolicyNames[] = {"keep", "replace", "add", "max"};
167 constexpr MagnitudePolicy magnitudePolicyValues[] = {MagnitudePolicy::Keep, MagnitudePolicy::Replace,
168 MagnitudePolicy::Add, MagnitudePolicy::Max};
169 if (!readOptionalEnum(*object, "magnitudePolicy", magnitudePolicyNames, magnitudePolicyValues,
170 std::size(magnitudePolicyNames), state.policy.magnitude))
171 return eve::Result<EffectRuntimeState>::failure(
172 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect magnitudePolicy is invalid",
173 "magnitudePolicy", {}, "effects.definition_runtime"));
174 constexpr const char* overflowPolicyNames[] = {"reject", "clamp", "replace_oldest"};
175 constexpr OverflowPolicy overflowPolicyValues[] = {OverflowPolicy::Reject, OverflowPolicy::Clamp,
176 OverflowPolicy::ReplaceOldest};
177 if (!readOptionalEnum(*object, "overflowPolicy", overflowPolicyNames, overflowPolicyValues,
178 std::size(overflowPolicyNames), state.policy.overflow))
179 return eve::Result<EffectRuntimeState>::failure(
180 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect overflowPolicy is invalid",
181 "overflowPolicy", {}, "effects.definition_runtime"));
182 state.policy.maxStacks = state.maxStacks;
183 state.remaining = state.duration > 0.0 ? state.duration : -1.0;
184
185 if (const auto* value = field(*object, "tags")) {
186 const auto* array = value->getIf<eve::Value::Array>();
187 if (!array)
188 return eve::Result<EffectRuntimeState>::failure(
189 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect tags must be an array", "tags", {},
190 "effects.definition_runtime"));
191 for (const auto& item : *array) {
192 const auto* tag = item.getIf<std::string>();
193 if (!tag || tag->empty())
194 return eve::Result<EffectRuntimeState>::failure(eve::Diagnostic::error(
195 eve::DiagnosticCode::InvalidArgument, "effect tag must be a non-empty string", "tags", {},
196 "effects.definition_runtime"));
197 state.tags.push_back(*tag);
198 }
199 std::sort(state.tags.begin(), state.tags.end());
200 state.tags.erase(std::unique(state.tags.begin(), state.tags.end()), state.tags.end());
201 }
202 if (const auto* value = field(*object, "payload")) {
203 const auto* payload = value->getIf<eve::Value::Object>();
204 if (!payload)
205 return eve::Result<EffectRuntimeState>::failure(
206 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect payload must be an object",
207 "payload", {}, "effects.definition_runtime"));
208 for (const auto& [key, item] : *payload) {
209 auto json = item.toJson();
210 if (!json) return eve::Result<EffectRuntimeState>::failure(json.status());
211 auto stored = state.payload.setJson(key, std::move(json).takeValue());
212 if (!stored.ok())
213 return eve::Result<EffectRuntimeState>::failure(eve::Diagnostic::error(
214 eve::DiagnosticCode::InvalidArgument, "effect payload member could not be stored", "payload." + key,
215 {}, "effects.definition_runtime"));
216 }
217 }
218 return eve::Result<EffectRuntimeState>::success(std::move(state));
219}

232 eve::DefinitionRef definition, std::string subject,
233 std::string source, eve::PersistentId instanceId,
234 eve::definition::ReloadPolicy policy) {
235 if (!definition.id().isValid())
236 return eve::Result<EffectDefinitionRuntime>::failure(

250 return eve::Result<EffectDefinitionRuntime>::success(EffectDefinitionRuntime(
251 registry, std::move(runtime).takeValue(), std::move(subject), std::move(source), policy));
252}

270eve::Result<void> EffectDefinitionRuntime::applyTo(EffectInstance* effect) const {
271 if (effect == nullptr)
272 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,

297eve::Result<eve::definition::ReloadOutcome> EffectDefinitionRuntime::reload(eve::definition::ReloadPolicy policy) {
298 if (registry_ == nullptr)
299 return eve::Result<eve::definition::ReloadOutcome>::failure(
300 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "effect definition registry is not bound",
301 "registry", {}, "effects.definition_runtime"));
302 auto next = currentHandle(*registry_, identity().definition);
303 if (!next) return eve::Result<eve::definition::ReloadOutcome>::failure(next.status());
304 auto defaults = resolveState(*registry_, identity().definition, next.value());
305 if (!defaults) return eve::Result<eve::definition::ReloadOutcome>::failure(defaults.status());
306 EffectRuntimeState defaultState = std::move(defaults).takeValue();
307 auto rebuild = [defaultState](const EffectRuntimeState& oldState, const eve::definition::InstanceIdentity&,
308 const eve::definition::DefinitionHandle&) mutable -> eve::Result<EffectRuntimeState> {
309 EffectRuntimeState rebuilt = defaultState;
310 if (oldState.remaining >= 0.0)
311 rebuilt.remaining = rebuilt.duration > 0.0 ? std::min(oldState.remaining, rebuilt.duration) : -1.0;
312 rebuilt.stackCount =
313 rebuilt.maxStacks == 0 ? oldState.stackCount : std::min(oldState.stackCount, rebuilt.maxStacks);
314 if (rebuilt.stackCount == 0) rebuilt.stackCount = 1;
315 return eve::Result<EffectRuntimeState>::success(std::move(rebuilt));
316 };
317 auto result = runtime_.reload(next.value(), policy, defaultState, std::move(rebuild));
318 if (result.ok()) policy_ = policy;
319 return result;
320}
