248std::int64_t selectionCost(const ClimbingProfile& profile, const ClimbingActionDefinition& action,
249 const ClimbingPose& pose, Vec3 targetDirection, Vec3 targetDelta,
250 float heightError, float distance, std::int64_t stableTieBreak = 0) {
251 const Vec3 forward = normalizedHorizontal(pose.forward);
252 targetDirection = normalizedHorizontal(targetDirection);
253 if (lengthSquared(targetDirection) <= epsilon) targetDirection = forward;
254
255 Vec3 intentDirection = normalizedHorizontal(pose.inputMode == ClimbingInputMode::Precision
256 ? pose.lookIntent
257 : pose.moveIntent);
258 if (lengthSquared(intentDirection) <= epsilon) intentDirection = forward;
259
260 const float directionDot = std::clamp(forward.x * targetDirection.x + forward.z * targetDirection.z,
261 -1.f, 1.f);
262 const float intentDot = std::clamp(intentDirection.x * targetDirection.x +
263 intentDirection.z * targetDirection.z,
264 -1.f, 1.f);
265 const bool precision = pose.inputMode == ClimbingInputMode::Precision;
266 const float assistScale = 1.f - profile.autoAssistStrength * (precision ? 0.15f : 0.5f);
267 const float directionModeScale = precision ? 1.5f : 0.75f;
268 const float speedModeScale = precision ? 0.5f : 1.5f;
269 const float translationModeScale = precision ? 1.25f : 0.75f;
270 const float rotationModeScale = precision ? 1.5f : 0.75f;
271 const float intentModeScale = precision ? 1.5f : 0.5f;
272 const float horizontalDelta = std::sqrt(targetDelta.x * targetDelta.x + targetDelta.z * targetDelta.z);
273 const float warpTranslation = std::sqrt(horizontalDelta * horizontalDelta + targetDelta.y * targetDelta.y);
274 const float warpRotation = std::acos(directionDot);
275 const float intentMismatch = 1.f - intentDot;
276
277 return static_cast<std::int64_t>(action.selectionBias) * 1000000ll +
278 weightedMillimeters((1.f - directionDot) * assistScale * directionModeScale,
279 profile.scoreWeights.direction) +
280 weightedMillimeters(std::fabs(pose.speed - action.minSpeed) * speedModeScale,
281 profile.scoreWeights.approachSpeed) +
282 weightedMillimeters(heightError, profile.scoreWeights.height) +
283 weightedMillimeters(distance, profile.scoreWeights.distance) +
284 weightedMillimeters(warpTranslation * translationModeScale,
285 profile.scoreWeights.warpTranslation) +
286 weightedMillimeters(warpRotation * rotationModeScale, profile.scoreWeights.warpRotation) +
287 weightedMillimeters(intentMismatch * intentModeScale, profile.scoreWeights.intentMismatch) +
288 stableTieBreak;
289}

299void ClimbingCandidateSet::considerWithActionId(ClimbingCandidate candidate, std::string_view actionId) {
300 const auto store = [&](ClimbingCandidate& slot) {
301 std::string reused = std::move(slot.actionId);
302 slot = std::move(candidate);
303 if (reused.capacity() >= actionId.size()) {
304 reused.assign(actionId);
305 slot.actionId.swap(reused);
306 } else {
307 slot.actionId.assign(actionId);
308 }
309 };
310 if (size_ < Capacity) {
311 store(values_[size_++]);
312 return;
313 }
314 auto worst = std::max_element(values_.begin(), values_.end(), candidateLess);
315 const bool better = candidate.score != worst->score
316 ? candidate.score < worst->score
317 : actionId != worst->actionId
318 ? actionId < std::string_view(worst->actionId)
319 : candidate.obstacleBodyId != worst->obstacleBodyId
320 ? candidate.obstacleBodyId < worst->obstacleBodyId
321 : candidate.obstacleShapeId < worst->obstacleShapeId;
322 if (better) store(*worst);
323}

493 if (!localLanding) return eve::Result<void>::failure(localLanding.status());
494 candidate.bodyLocalTop = {localTop.value().x, localTop.value().y, localTop.value().z};
495 candidate.bodyLocalLanding = {localLanding.value().x, localLanding.value().y,
496 localLanding.value().z};
497 }
498 return eve::Result<void>::success();

502 filter.ignoredBodyId = pose.ignoredBodyId;
503 const float broadRadius = profile_.maxProbeDistance + profile_.capsuleRadius + profile_.skin;
504 auto broadPhase = world.queryAabbBroadPhaseOwned(
505 pose.feet.x - broadRadius, pose.feet.y - profile_.skin, pose.feet.z - broadRadius,
506 pose.feet.x + broadRadius, pose.feet.y + profile_.maxObstacleHeight + profile_.capsuleHeight,
507 pose.feet.z + broadRadius, filter);
508 if (!broadPhase) return eve::Result<void>::failure(broadPhase.status());
509 lastCounters_.broadPhaseQueryCount = 1;
510 lastCounters_.broadPhaseHitCount = static_cast<std::uint32_t>(broadPhase.value().count);
511 if (broadPhase.value().count == 0) return finalizeCandidates();
512 const auto broadPhaseContains = [&](physics::PhysicsShapeHandle shape) noexcept {
513 if (broadPhase.value().truncated) return true;
514 return std::any_of(broadPhase.value().hits.begin(),
515 broadPhase.value().hits.begin() +
516 static_cast<std::ptrdiff_t>(broadPhase.value().count),
517 [&](const physics::BroadPhaseHit3D& hit) { return hit.shape == shape; });

520 const bool hasSlide = std::any_of(profile_.actions.begin(), profile_.actions.end(), [&](const auto& action) {
521 return action.kind == ClimbingActionKind::Slide && actionEnabledForPose(profile_, action, pose) &&
522 probeRecipeMatchesKind(action);
523 });

526 const Vec3 groundEnd{pose.feet.x, pose.feet.y - 0.35f, pose.feet.z};
527 if (!reserveQuery()) return finalizeCandidates();

529 groundEnd.x, groundEnd.y, groundEnd.z, filter);
530 if (!groundResult) return eve::Result<void>::failure(groundResult.status());

532 if (debugCapture_ == ClimbingDebugCapture::Enabled)
533 boundedDebugPush(lastDebugQueries_, ClimbingDebugQuery{"", ground.hit ? "slide.ground" : "slide.no_ground",
534 groundStart, groundEnd});

538 !probeRecipeMatchesKind(action))

542 if (debugCapture_ == ClimbingDebugCapture::Enabled)
543 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "below_min_speed"});
544 continue;
545 }
546 if (!supportSelectorsMatch(action, ground.shapeTag, ground.materialId)) {
547 rejectCandidate();
548 if (debugCapture_ == ClimbingDebugCapture::Enabled)
549 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "support_tag_rejected"});
550 continue;
551 }
552 const Vec3 landing = pose.feet + forward * action.landingForward;

557 profile_.capsuleRadius, filter);
558 if (!overlap) return eve::Result<void>::failure(overlap.status());
559 if (overlap.value().bodyCount != 0) {
560 rejectCandidate();
561 if (debugCapture_ == ClimbingDebugCapture::Enabled)
562 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "clearance_blocked"});
563 continue;

566 candidate.definitionGeneration = definitionGeneration_;
567 candidate.world = world.runtimeHandle();
568 candidate.obstacleBody = ground.body;
569 candidate.obstacleShape = ground.shape;
570 candidate.obstacleBodyId = ground.bodyId;
571 candidate.obstacleShapeId = ground.shapeId;
572 candidate.ignoredBodyId = pose.ignoredBodyId;
573 candidate.frontPoint = {ground.x, ground.y, ground.z};
574 candidate.topPoint = candidate.frontPoint;
575 candidate.landingFeet = landing;
576 candidate.surfaceNormal = {ground.normalX, ground.normalY, ground.normalZ};

578 candidate.gapDistance = action.landingForward;

581 candidate.supportShapeTag = ground.shapeTag;
582 candidate.supportMaterialId = ground.materialId;
583 candidate.probeRecipe = action.probeRecipe;

587 candidate.kind = action.kind;
588 auto captured = captureBodyLocal(candidate);
589 if (!captured) return eve::Result<void>::failure(captured.status());
590 if (debugCapture_ == ClimbingDebugCapture::Enabled)
591 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{
592 action.id, "accepted", static_cast<std::int64_t>(action.selectionBias) * 1000000ll,
593 0, quantizeMillimeters(action.landingForward), candidate.score});
594 candidates.considerWithActionId(std::move(candidate), action.id);
595 }

599 const bool hasWallRun = std::any_of(profile_.actions.begin(), profile_.actions.end(), [&](const auto& action) {
600 return action.kind == ClimbingActionKind::WallRun && actionEnabledForPose(profile_, action, pose) &&
601 probeRecipeMatchesKind(action);
602 });
603 if (hasWallRun) {
604 const Vec3 chest{pose.feet.x, pose.feet.y + profile_.capsuleHeight * 0.55f, pose.feet.z};
605 const Vec3 sides[2] = {{-forward.z, 0.f, forward.x}, {forward.z, 0.f, -forward.x}};
606 for (std::size_t sideIndex = 0; sideIndex < 2; ++sideIndex) {
607 const Vec3 end = chest + sides[sideIndex] * profile_.maxProbeDistance;
608 if (!reserveQuery()) return finalizeCandidates();
609 auto sideResult = world.rayCastOwned(chest.x, chest.y, chest.z, end.x, end.y, end.z, filter);
610 if (!sideResult) return eve::Result<void>::failure(sideResult.status());
611 const physics::RayHit3D hit = std::move(sideResult).takeValue();
612 if (debugCapture_ == ClimbingDebugCapture::Enabled)
613 boundedDebugPush(lastDebugQueries_, ClimbingDebugQuery{"", hit.hit ? "wall_run.wall" : "wall_run.miss",
614 chest, end});
615 Vec3 normal = hit.hit ? normalizedHorizontal({hit.normalX, hit.normalY, hit.normalZ}) : Vec3{};
616 if (!hit.hit || !broadPhaseContains(hit.shape) || lengthSquared(normal) <= epsilon ||
617 std::fabs(hit.normalY) > 0.35f)
618 continue;
619 Vec3 tangent{normal.z, 0.f, -normal.x};
620 if (tangent.x * forward.x + tangent.z * forward.z < 0.f) tangent = tangent * -1.f;
621 for (const auto& action : profile_.actions) {
622 if (action.kind != ClimbingActionKind::WallRun || !actionEnabledForPose(profile_, action, pose) ||
623 !probeRecipeMatchesKind(action) || pose.speed + epsilon < action.minSpeed)
624 continue;
625 if (!supportSelectorsMatch(action, hit.shapeTag, hit.materialId)) {
626 rejectCandidate();
627 continue;
628 }
629 const float wallOffset = profile_.capsuleRadius + profile_.skin;
630 const Vec3 landing{hit.x + normal.x * wallOffset + tangent.x * action.landingForward,
631 pose.feet.y, hit.z + normal.z * wallOffset + tangent.z * action.landingForward};
632 const float lowerY = landing.y + profile_.capsuleRadius + profile_.skin;
633 const float upperY = landing.y + profile_.capsuleHeight - profile_.capsuleRadius + profile_.skin;
634 if (!reserveQuery()) return finalizeCandidates();
635 auto overlap = world.queryCapsuleOwned(landing.x, lowerY, landing.z, landing.x, upperY, landing.z,
636 profile_.capsuleRadius, filter);
637 if (!overlap) return eve::Result<void>::failure(overlap.status());
638 if (overlap.value().bodyCount != 0) {
639 rejectCandidate();
640 continue;
641 }
642 ClimbingCandidate candidate;
643 candidate.definitionGeneration = definitionGeneration_;
644 candidate.world = world.runtimeHandle();
645 candidate.obstacleBody = hit.body;
646 candidate.obstacleShape = hit.shape;
647 candidate.obstacleBodyId = hit.bodyId;
648 candidate.obstacleShapeId = hit.shapeId;
649 candidate.ignoredBodyId = pose.ignoredBodyId;
650 candidate.frontPoint = {hit.x, hit.y, hit.z};
651 candidate.topPoint = candidate.frontPoint;
652 candidate.landingFeet = landing;
653 candidate.surfaceNormal = normal;
654 candidate.surfaceTangent = tangent;
655 const float distance = std::sqrt((hit.x - chest.x) * (hit.x - chest.x) +
656 (hit.z - chest.z) * (hit.z - chest.z));
657 candidate.gapDistance = distance;
658 candidate.clearanceHeight = profile_.capsuleHeight;
659 candidate.slopeRadians = std::acos(std::clamp(hit.normalY, -1.f, 1.f));
660 candidate.supportShapeTag = hit.shapeTag;
661 candidate.supportMaterialId = hit.materialId;
662 candidate.probeRecipe = action.probeRecipe;
663 candidate.score = selectionCost(profile_, action, pose, tangent, landing - pose.feet,
664 0.f, distance, static_cast<std::int64_t>(sideIndex)) +
665 (previousActionId_ == action.id ? action.repetitionPenalty : 0) +
666 0;
667 candidate.kind = action.kind;
668 auto captured = captureBodyLocal(candidate);
669 if (!captured) return eve::Result<void>::failure(captured.status());
670 if (debugCapture_ == ClimbingDebugCapture::Enabled)
671 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{
672 action.id, "accepted", static_cast<std::int64_t>(action.selectionBias) * 1000000ll,
673 0, quantizeMillimeters(distance), candidate.score});
674 candidates.considerWithActionId(std::move(candidate), action.id);
675 }
676 }

679 const bool hasObstacleProbe = std::any_of(profile_.actions.begin(), profile_.actions.end(), [&](const auto& action) {
680 return isObstacleProbeKind(action.kind) && actionEnabledForPose(profile_, action, pose) &&
681 probeRecipeMatchesKind(action);
682 });
683 if (!hasObstacleProbe) return finalizeCandidates();
684 // A single chest-height ray skips low vault obstacles entirely. The obstacle recipe starts just above the
685 // capsule's lower hemisphere so it can see every valid obstacle above the profile's physical skin; top and
686 // landing queries still provide the authoritative height and clearance rejection.
687 const float obstacleProbeY = pose.feet.y + profile_.capsuleRadius + profile_.skin;
688 const Vec3 frontStart{pose.feet.x, obstacleProbeY, pose.feet.z};
689 const Vec3 frontEnd = frontStart + forward * profile_.maxProbeDistance;
690 if (!reserveQuery()) return finalizeCandidates();
691 auto frontResult =
692 world.rayCastOwned(frontStart.x, frontStart.y, frontStart.z, frontEnd.x, frontEnd.y, frontEnd.z, filter);
693 if (!frontResult) return eve::Result<void>::failure(frontResult.status());
694 const physics::RayHit3D front = std::move(frontResult).takeValue();
695 if (debugCapture_ == ClimbingDebugCapture::Enabled)
696 boundedDebugPush(lastDebugQueries_, ClimbingDebugQuery{"", front.hit ? "front.hit" : "front.miss",
697 frontStart, frontEnd});
698 if (!front.hit || !broadPhaseContains(front.shape)) return finalizeCandidates();
699
700 const Vec3 overFront{front.x + forward.x * (profile_.capsuleRadius + profile_.skin),
701 pose.feet.y + profile_.maxObstacleHeight + profile_.skin,
702 front.z + forward.z * (profile_.capsuleRadius + profile_.skin)};
703 const Vec3 topEnd{overFront.x, pose.feet.y + profile_.skin, overFront.z};
704 if (!reserveQuery()) return finalizeCandidates();
705 auto topResult = world.rayCastOwned(overFront.x, overFront.y, overFront.z, topEnd.x, topEnd.y, topEnd.z, filter);
706 if (!topResult) return eve::Result<void>::failure(topResult.status());
707 const physics::RayHit3D top = std::move(topResult).takeValue();
708 if (debugCapture_ == ClimbingDebugCapture::Enabled)
709 boundedDebugPush(lastDebugQueries_, ClimbingDebugQuery{"", top.hit ? "top.hit" : "top.miss", overFront,
710 topEnd});
711 if (!top.hit || top.body != front.body || top.normalY < profile_.minTopNormalY)
712 return finalizeCandidates();
713
714 const float height = top.y - pose.feet.y;
715 if (height < 0.f || height > profile_.maxObstacleHeight + profile_.skin)
716 return finalizeCandidates();
717
718 const float baseDepth = std::sqrt((top.x - front.x) * (top.x - front.x) +
719 (top.z - front.z) * (top.z - front.z));
720 float measuredDepth = baseDepth;
721 float measuredCurvature = 0.f;
722 const float topSlope = std::acos(std::clamp(top.normalY, -1.f, 1.f));
723 float depthProbeDistance = 0.f;
724 bool needsDepthEvidence = false;
725 for (const ClimbingActionDefinition& action : profile_.actions) {
726 if (!isObstacleProbeKind(action.kind) || !actionEnabledForPose(profile_, action, pose) ||
727 !probeRecipeMatchesKind(action))
728 continue;
729 if (action.minDepth > baseDepth + epsilon || action.maxDepth < 999.f || action.maxCurvature < 999.f) {
730 needsDepthEvidence = true;
731 depthProbeDistance = std::max(depthProbeDistance, std::max(action.minDepth, action.maxDepth < 999.f
732 ? action.maxDepth + profile_.skin
733 : 0.f));
734 }
735 }
736 depthProbeDistance = std::min(profile_.maxProbeDistance, std::max(0.f, depthProbeDistance - baseDepth));
737 if (needsDepthEvidence && depthProbeDistance > epsilon) {
738 const std::uint32_t sampleCount = std::min<std::uint32_t>(4, profile_.probeSectors);
739 Vec3 previousNormal{top.normalX, top.normalY, top.normalZ};
740 const float stepDistance = depthProbeDistance / static_cast<float>(sampleCount);
741 for (std::uint32_t sample = 1; sample <= sampleCount; ++sample) {
742 if (!reserveQuery()) break;
743 const float offset = stepDistance * static_cast<float>(sample);
744 const Vec3 sampleStart{top.x + forward.x * offset, top.y + profile_.skin + 0.2f,
745 top.z + forward.z * offset};
746 const Vec3 sampleEnd{sampleStart.x, top.y - profile_.skin - 0.2f, sampleStart.z};
747 auto sampled = world.rayCastOwned(sampleStart.x, sampleStart.y, sampleStart.z, sampleEnd.x,
748 sampleEnd.y, sampleEnd.z, filter);
749 if (!sampled) return eve::Result<void>::failure(sampled.status());
750 const physics::RayHit3D hit = std::move(sampled).takeValue();
751 if (!hit.hit || hit.body != top.body || hit.normalY < profile_.minTopNormalY) break;
752 measuredDepth = baseDepth + offset;
753 const Vec3 normal{hit.normalX, hit.normalY, hit.normalZ};
754 const float dot = std::clamp(previousNormal.x * normal.x + previousNormal.y * normal.y +
755 previousNormal.z * normal.z,
756 -1.f, 1.f);
757 measuredCurvature = std::max(measuredCurvature, std::acos(dot) / std::max(stepDistance, epsilon));
758 previousNormal = normal;
759 }
760 }
761
762 for (const ClimbingActionDefinition& action : profile_.actions) {
763 if (!actionEnabledForPose(profile_, action, pose)) {
764 rejectCandidate();
765 if (debugCapture_ == ClimbingDebugCapture::Enabled)
766 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "definition_policy_rejected"});
767 continue;
768 }
769 if (!probeRecipeMatchesKind(action)) {
770 rejectCandidate();
771 if (debugCapture_ == ClimbingDebugCapture::Enabled)
772 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "probe_recipe_rejected"});
773 continue;
774 }
775 if (!isObstacleProbeKind(action.kind)) {
776 if (isRuntimeProbeKind(action.kind)) continue;
777 rejectCandidate();
778 if (debugCapture_ == ClimbingDebugCapture::Enabled)
779 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "anchor_graph_only"});
780 continue;
781 }
782 if (height + epsilon < action.minHeight || height - epsilon > action.maxHeight) {
783 rejectCandidate();
784 if (debugCapture_ == ClimbingDebugCapture::Enabled)
785 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "out_of_height_range"});
786 continue;
787 }
788 if (measuredDepth + epsilon < action.minDepth || measuredDepth - epsilon > action.maxDepth ||
789 topSlope - epsilon > action.maxSlopeRadians || measuredCurvature - epsilon > action.maxCurvature ||
790 !supportSelectorsMatch(action, top.shapeTag, top.materialId)) {
791 rejectCandidate();
792 if (debugCapture_ == ClimbingDebugCapture::Enabled)
793 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "support_geometry_rejected"});
794 continue;
795 }
796 if (pose.speed + epsilon < action.minSpeed) {
797 rejectCandidate();
798 if (debugCapture_ == ClimbingDebugCapture::Enabled)
799 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "below_min_speed"});
800 continue;
801 }
802 const bool wantsHang = action.kind == ClimbingActionKind::LedgeGrab;
803 if ((wantsHang && (pose.grounded || pose.verticalSpeed > epsilon)) || (!wantsHang && !pose.grounded)) {
804 rejectCandidate();
805 if (debugCapture_ == ClimbingDebugCapture::Enabled)
806 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "source_mode_rejected"});
807 continue;
808 }
809
810 const Vec3 landing =
811 wantsHang
812 ? Vec3{front.x - forward.x * (profile_.capsuleRadius + profile_.skin + action.hangBodyOffset),
813 top.y - action.hangFeetBelowLedge,
814 front.z - forward.z * (profile_.capsuleRadius + profile_.skin + action.hangBodyOffset)}
815 : Vec3{top.x + forward.x * action.landingForward, top.y, top.z + forward.z * action.landingForward};
816 const float lowerY = landing.y + profile_.capsuleRadius + profile_.skin;
817 const float upperY = landing.y + profile_.capsuleHeight - profile_.capsuleRadius + profile_.skin;
818 if (!reserveQuery()) return finalizeCandidates();
819 auto overlap = world.queryCapsuleOwned(landing.x, lowerY, landing.z, landing.x, upperY, landing.z,
820 profile_.capsuleRadius, filter);
821 if (!overlap) return eve::Result<void>::failure(overlap.status());
822 if (debugCapture_ == ClimbingDebugCapture::Enabled)
823 boundedDebugPush(lastDebugQueries_,
824 ClimbingDebugQuery{action.id,
825 overlap.value().bodyCount == 0 ? "landing.clear" : "landing.blocked",
826 landing, landing, profile_.capsuleRadius, profile_.capsuleHeight});
827 if (overlap.value().bodyCount != 0) {
828 rejectCandidate();
829 if (debugCapture_ == ClimbingDebugCapture::Enabled)
830 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "clearance_blocked"});
831 continue;
832 }
833
834 HangSupport support = HangSupport::None;
835 if (wantsHang) {
836 const Vec3 supportStart{landing.x, landing.y + profile_.capsuleHeight * 0.45f, landing.z};
837 const Vec3 supportEnd =
838 supportStart + forward * (profile_.capsuleRadius + profile_.skin + action.hangBodyOffset + 0.25f);
839 if (!reserveQuery()) return finalizeCandidates();
840 auto supportHit = world.rayCastOwned(supportStart.x, supportStart.y, supportStart.z, supportEnd.x,
841 supportEnd.y, supportEnd.z, filter);
842 if (!supportHit) return eve::Result<void>::failure(supportHit.status());
843 if (debugCapture_ == ClimbingDebugCapture::Enabled)
844 boundedDebugPush(lastDebugQueries_,
845 ClimbingDebugQuery{action.id,
846 supportHit.value().hit ? "support.hit" : "support.miss",
847 supportStart, supportEnd});
848 support = supportHit.value().hit && supportHit.value().body == front.body ? HangSupport::Braced
849 : HangSupport::Free;
850 }
851
852 const float distance = std::sqrt((front.x - pose.feet.x) * (front.x - pose.feet.x) +
853 (front.z - pose.feet.z) * (front.z - pose.feet.z));
854 if (distance + epsilon < action.minDistance || distance - epsilon > action.maxDistance ||
855 front.normalY + epsilon < action.minSurfaceNormalY) {
856 rejectCandidate();
857 if (debugCapture_ == ClimbingDebugCapture::Enabled)
858 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "geometry_policy_rejected"});
859 continue;
860 }
861 const Vec3 approachNormal = normalizedHorizontal({-front.normalX, 0.f, -front.normalZ});
862 const float heightCenter = (action.minHeight + action.maxHeight) * 0.5f;
863 const std::int64_t score = selectionCost(profile_, action, pose, approachNormal,
864 landing - pose.feet, std::fabs(height - heightCenter),
865 distance) +
866 (previousActionId_ == action.id ? action.repetitionPenalty : 0);
867 if (debugCapture_ == ClimbingDebugCapture::Enabled) {
868 const std::int64_t biasCost = static_cast<std::int64_t>(action.selectionBias) * 1000000ll;
869 const std::int64_t heightCost = quantizeMillimeters(height) * 1000ll;
870 const std::int64_t distanceCost = quantizeMillimeters(distance);
871 boundedDebugPush(lastEvidence_, ClimbingCandidateEvidence{action.id, "accepted", biasCost, heightCost,
872 distanceCost, score});
873 }
874 ClimbingCandidate candidate;
875 candidate.definitionGeneration = definitionGeneration_;
876 candidate.world = world.runtimeHandle();
877 candidate.obstacleBody = front.body;
878 candidate.obstacleShape = front.shape;
879 candidate.obstacleBodyId = front.bodyId;
880 candidate.obstacleShapeId = front.shapeId;
881 candidate.ignoredBodyId = pose.ignoredBodyId;
882 candidate.frontPoint = {front.x, front.y, front.z};
883 candidate.topPoint = {top.x, top.y, top.z};
884 candidate.landingFeet = landing;
885 candidate.surfaceNormal = {front.normalX, front.normalY, front.normalZ};
886 candidate.surfaceTangent = normalizedHorizontal({front.normalZ, 0.f, -front.normalX});
887 candidate.leftHandAnchor = candidate.topPoint - candidate.surfaceTangent * (action.handSpacing * 0.5f);
888 candidate.rightHandAnchor = candidate.topPoint + candidate.surfaceTangent * (action.handSpacing * 0.5f);
889 candidate.obstacleHeight = height;
890 candidate.obstacleDepth = measuredDepth;
891 candidate.gapDistance = distance;
892 candidate.clearanceHeight = profile_.capsuleHeight;
893 candidate.slopeRadians = topSlope;
894 candidate.curvature = measuredCurvature;
895 candidate.supportShapeTag = top.shapeTag;
896 candidate.supportMaterialId = top.materialId;
897 candidate.probeRecipe = action.probeRecipe;
898 candidate.score = score;
899 candidate.kind = action.kind;
900 candidate.support = support;
901 if (physics::Body3D* body = world.findBody(front.body)) {
902 auto localTop = body->worldToLocalPointOwned(top.x, top.y, top.z);
903 if (!localTop) return eve::Result<void>::failure(localTop.status());
904 auto localLanding = body->worldToLocalPointOwned(landing.x, landing.y, landing.z);
905 if (!localLanding) return eve::Result<void>::failure(localLanding.status());
906 candidate.bodyLocalTop = {localTop.value().x, localTop.value().y, localTop.value().z};
907 candidate.bodyLocalLanding = {localLanding.value().x, localLanding.value().y,
908 localLanding.value().z};
909 }
910 candidates.considerWithActionId(std::move(candidate), action.id);
911 }
912 return finalizeCandidates();

919eve::Result<ClimbingAdvance> ClimbingRuntime::advance(physics::World3D& world, eve::SimulationStep step,
920 const ClimbingMotionInput& motion) {
921 lastQueryCount_ = 0;
922 lastCounters_ = {};
923 lastCounters_.workload = ClimbingWorkload::Active;
924 lastCounters_.queryBudget = ClimbingQueryBudgets::Active;
925 RuntimeTelemetryScope telemetryScope(telemetry_, lastCounters_, step.tick);
926 if (!isActivePhase(phase_) || !execution_)
927 return climbingFailure<ClimbingAdvance>(eve::DiagnosticCode::PreconditionViolation,
928 "no climbing execution is active", "runtime.phase");
929 Execution& execution = *execution_;
930 lastCounters_.selectedCost = execution.candidate.score;
931 if (world.runtimeHandle() != execution.candidate.world)
932 return climbingFailure<ClimbingAdvance>(eve::DiagnosticCode::StaleHandle,
933 "execution belongs to another or stale physics world", "world");
934 if (step.tick <= execution.lastTick)
935 return climbingFailure<ClimbingAdvance>(eve::DiagnosticCode::Conflict,
936 "simulation tick must increase exactly once per update", "step.tick");
937 if (step.delta.nanoseconds() <= 0)
938 return climbingFailure<ClimbingAdvance>(eve::DiagnosticCode::InvalidArgument,
939 "simulation delta must be positive", "step.delta");
940 if (!finite(motion.rootTranslation) || !finite(motion.facing) || !finite(motion.pelvisOffset) ||
941 !finite(motion.rootYawRadians) || length(motion.pelvisOffset) > profile_.maxPelvisDeviation + epsilon)
942 return climbingFailure<ClimbingAdvance>(
943 eve::DiagnosticCode::InvalidArgument,
944 "authored motion must be finite and pelvis deviation must remain inside the profile limit", "motion");
945 auto eventCapacity = requireEventCapacity(4, step.tick);

973 "graph anchor or occupancy reservation is stale",
974 "execution.anchor");

1001 terminalCode_ = "climbing.anchor.stale";

1115 execution.velocity.y += world.getGravityY() * dt;
1116 execution.velocity.z += world.getGravityZ() * dt;

1119 const float upperY = execution.currentFeet.y + profile_.capsuleHeight - profile_.capsuleRadius;

1121 execution.currentFeet.x, upperY, execution.currentFeet.z,
1122 profile_.capsuleRadius, desired.x, desired.y, desired.z, filter);
1123 if (!moved) return eve::Result<ClimbingAdvance>::failure(moved.status());
1124 ++lastQueryCount_;
1125 lastCounters_.queryCount = lastQueryCount_;
1126 const physics::CapsuleMove3D movement = std::move(moved).takeValue();

1128 const Vec3 actual{movement.deltaX, movement.deltaY, movement.deltaZ};
1129 lastCounters_.warpResidual = desired - actual;
1130 execution.currentFeet = execution.currentFeet + actual;

1150 terminalCode_ = "climbing.completed";
1151 output.phase = phase_;

1153 enqueueEvent({ClimbingEventKind::Completed, output.actionId, step.tick, execution.executionId});
1154 output.terminalVelocity = terminalVelocityFor(execution.action, actual, 1.f / dt);
1155 output.hasTerminalVelocity = true;
1156 execution_.reset();
1157 }
1158 return eve::Result<ClimbingAdvance>::success(std::move(output));
1159 }

1161 auto elapsed = execution.elapsed.tryAdd(step.delta);
1162 if (!elapsed) return eve::Result<ClimbingAdvance>::failure(elapsed.status());
1163 execution.elapsed = std::move(elapsed).takeValue();
1164 execution.lastTick = step.tick;
1165 const double durationSeconds = execution.duration.seconds();
1166 const float t = static_cast<float>(std::clamp(execution.elapsed.seconds() / durationSeconds, 0.0, 1.0));
1167 const Vec3 planned =
1168 detail::trajectoryPoint(execution.startFeet, execution.candidate, execution.action, profile_, t);
1169 const Vec3 proceduralDelta = planned - execution.currentFeet;
1170 Vec3 baseDelta = proceduralDelta;
1171 if (motion.hasRootMotion) {
1172 const float authoredLength = length(motion.rootTranslation);
1173 const float targetLength = length(proceduralDelta);
1174 const float scale = authoredLength > epsilon
1175 ? std::clamp(targetLength / authoredLength, execution.action.rootMotionScaleMin,
1176 execution.action.rootMotionScaleMax)
1177 : execution.action.rootMotionScaleMin;
1178 baseDelta = motion.rootTranslation * scale;
1179 }
1180
1181 const WarpChannels channels = activeWarpChannels(execution.action, t);
1182 const Vec3 warpError = planned - (execution.currentFeet + baseDelta);
1183 Vec3 appliedWarp;
1184 if (channels.horizontal) {
1185 Vec3 horizontal{warpError.x, 0.f, warpError.z};
1186 const float remaining = std::max(0.f, execution.action.horizontalWarpBudget - execution.horizontalWarpUsed);
1187 horizontal = clampMagnitude(horizontal, std::min(execution.action.maxTranslationWarpPerTick, remaining));
1188 appliedWarp.x = horizontal.x;
1189 appliedWarp.z = horizontal.z;
1190 }
1191 if (channels.vertical) {
1192 const float remaining = std::max(0.f, execution.action.verticalWarpBudget - execution.verticalWarpUsed);
1193 appliedWarp.y = std::clamp(warpError.y, -std::min(execution.action.maxTranslationWarpPerTick, remaining),
1194 std::min(execution.action.maxTranslationWarpPerTick, remaining));
1195 }
1196 const float remainingTotalWarp =
1197 std::max(0.f, profile_.maxTotalWarpBudget - execution.horizontalWarpUsed - execution.verticalWarpUsed);
1198 appliedWarp = clampMagnitude(appliedWarp, std::min(execution.action.maxTranslationWarpPerTick, remainingTotalWarp));
1199 execution.horizontalWarpUsed += length({appliedWarp.x, 0.f, appliedWarp.z});
1200 execution.verticalWarpUsed += std::fabs(appliedWarp.y);
1201 const Vec3 desired = baseDelta + appliedWarp;
1202
1203 float desiredYawDelta = motion.rootYawRadians;
1204 if (channels.facing) {
1205 const Vec3 targetFacing = execution.candidate.surfaceNormal * -1.f;
1206 const float yawError = signedHorizontalAngle(motion.facing, targetFacing) - motion.rootYawRadians;
1207 const float remaining = std::max(0.f, execution.action.facingWarpBudgetRadians - execution.facingWarpUsed);
1208 const float correction = std::clamp(yawError, -std::min(execution.action.maxYawWarpRadiansPerTick, remaining),
1209 std::min(execution.action.maxYawWarpRadiansPerTick, remaining));
1210 execution.facingWarpUsed += std::fabs(correction);
1211 desiredYawDelta += correction;
1212 }
1213 const float authoredLeftHand = activeContactWeight(execution.action, ClimbingContactTarget::LeftHand, t);
1214 const float authoredRightHand = activeContactWeight(execution.action, ClimbingContactTarget::RightHand, t);
1215 bool leftContact = authoredLeftHand > epsilon;
1216 bool rightContact = authoredRightHand > epsilon;
1217 bool landContact = false;
1218 bool compactRequested = false;
1219 bool compactForTick = execution.compactCollisionActive;
1220 bool branchForTick = execution.action.branchWindows.empty() ? execution.branchWindowOpen
1221 : activeBranchWindow(execution.action, t);
1222 for (ClimbingNotifyKind notify : motion.notifies) {
1223 switch (notify) {
1224 case ClimbingNotifyKind::ContactLeftHand: leftContact = true; break;
1225 case ClimbingNotifyKind::ContactRightHand: rightContact = true; break;
1226 case ClimbingNotifyKind::CollisionCompact:
1227 compactRequested = true;
1228 compactForTick = true;
1229 break;
1230 case ClimbingNotifyKind::BranchOpen: branchForTick = true; break;
1231 case ClimbingNotifyKind::BranchClose: branchForTick = false; break;
1232 case ClimbingNotifyKind::Land:
1233 landContact = true;
1234 compactForTick = false;
1235 branchForTick = false;
1236 break;
1237 }
1238 }
1239 const float collisionHeight = compactForTick ? profile_.compactCapsuleHeight : profile_.capsuleHeight;
1240 const float lowerY = execution.currentFeet.y + profile_.capsuleRadius;
1241 const float upperY = execution.currentFeet.y + collisionHeight - profile_.capsuleRadius;
1242 auto moved = world.moveCapsuleOwned(execution.currentFeet.x, lowerY, execution.currentFeet.z,
1243 execution.currentFeet.x, upperY, execution.currentFeet.z,
1244 profile_.capsuleRadius, desired.x, desired.y, desired.z, filter);
1245 if (!moved) return eve::Result<ClimbingAdvance>::failure(moved.status());
1246 ++lastQueryCount_;
1247 lastCounters_.queryCount = lastQueryCount_;
1248 const physics::CapsuleMove3D movement = std::move(moved).takeValue();
1249 lastCounters_.moverIterations = static_cast<std::uint32_t>(std::max(0, movement.iterations));
1250 const Vec3 actual{movement.deltaX, movement.deltaY, movement.deltaZ};
1251 execution.currentFeet = execution.currentFeet + actual;
1252 execution.lastPlannedFeet = planned;
1253 const Vec3 residual = desired - actual;
1254 lastCounters_.warpResidual = residual;
1255 execution.accumulatedResidual = execution.accumulatedResidual + residual;
1256 execution.compactCollisionActive = compactForTick;
1257 execution.branchWindowOpen = branchForTick;
1258 if (debugCapture_ == ClimbingDebugCapture::Enabled)
1259 boundedDebugPush(motionEvidence_, ClimbingMotionEvidence{step.tick, planned, execution.currentFeet, residual,
1260 collisionHeight, movement.constrained});
1261 ClimbingAdvance output{ClimbingPhase::Climbing,
1262 execution.candidate.actionId,
1263 execution.currentFeet,
1264 desired,
1265 actual,
1266 residual,
1267 t,
1268 movement.constrained,
1269 movement.grounded,
1270 execution.candidate.support};
1271 output.executionId = execution.executionId;
1272 output.appliedWarp = appliedWarp;
1273 output.desiredYawDelta = desiredYawDelta;
1274 output.cameraCueProfile = profile_.cameraCueProfile;
1275 output.cameraCue = execution.action.cameraCue;
1276 output.animationClipId = execution.action.animation.clipId;
1277 output.animationGraphNodeId = execution.action.animation.graphNodeId;
1278 output.animationMirrored = execution.action.animation.mirrored;
1279 output.leftHandAnchor = execution.candidate.leftHandAnchor;
1280 output.rightHandAnchor = execution.candidate.rightHandAnchor;
1281 if (leftContact && !execution.leftContactEmitted) {
1282 execution.leftContactEmitted = true;
1283 enqueueEvent({ClimbingEventKind::ContactLeftHand, output.actionId, step.tick, execution.executionId});
1284 }
1285 if (rightContact && !execution.rightContactEmitted) {
1286 execution.rightContactEmitted = true;
1287 enqueueEvent({ClimbingEventKind::ContactRightHand, output.actionId, step.tick, execution.executionId});
1288 }
1289 if (landContact) execution.landContactReleased = true;
1290 if (execution.action.contactConstraints.empty()) {
1291 output.leftHandWeight = execution.leftContactEmitted && !execution.landContactReleased ? 1.f : 0.f;
1292 output.rightHandWeight = execution.rightContactEmitted && !execution.landContactReleased ? 1.f : 0.f;
1293 } else if (!execution.landContactReleased) {
1294 output.leftHandWeight = authoredLeftHand;
1295 output.rightHandWeight = authoredRightHand;
1296 output.leftFootWeight = activeContactWeight(execution.action, ClimbingContactTarget::LeftFoot, t);
1297 output.rightFootWeight = activeContactWeight(execution.action, ClimbingContactTarget::RightFoot, t);
1298 output.pelvisWeight = activeContactWeight(execution.action, ClimbingContactTarget::Pelvis, t);
1299 }
1300 output.contactWeight = std::max({output.leftHandWeight, output.rightHandWeight, output.leftFootWeight,
1301 output.rightFootWeight, output.pelvisWeight});
1302 output.compactCollisionRequested = compactRequested;
1303 output.compactCollisionActive = execution.compactCollisionActive;
1304 output.branchWindowOpen = execution.branchWindowOpen;
1305 if (const std::string* comboTag = activeBranchComboTag(execution.action, t))
1306 output.branchComboTag = *comboTag;
1307 if (t < 0.1f)
1308 phase_ = ClimbingPhase::Aligning;
1309 else if (t < 0.2f)
1310 phase_ = ClimbingPhase::Launching;
1311 else if (t < 0.8f)
1312 phase_ = ClimbingPhase::Climbing;
1313 else if (t < 0.95f)
1314 phase_ = ClimbingPhase::Landing;
1315 else
1316 phase_ = ClimbingPhase::Recovering;
1317 output.phase = phase_;
1318 const Vec3 remainingTargetError = planned - execution.currentFeet;
1319 const bool authoredWarpMissed = motion.hasRootMotion && t + epsilon >= finalWarpWindowEnd(execution.action) &&
1320 length(remainingTargetError) > profile_.maxWarpResidual + epsilon;
1321 if (length(execution.accumulatedResidual) > profile_.maxWarpResidual + epsilon || authoredWarpMissed) {
1322 phase_ = ClimbingPhase::Failed;
1323 terminalCode_ = "climbing.warp.budget_exceeded";
1324 output.phase = phase_;
1325 enqueueEvent({ClimbingEventKind::Failed, output.actionId, step.tick, execution.executionId});
1326 if (graphBound) {
1327 auto released = releaseAnchorReservation();
1328 released.ignore("failed anchor motion releases graph occupancy");
1329 }
1330 execution_.reset();
1331 } else if (t >= 1.f - epsilon) {
1332 if (graphBound && endsAtAnchorHang(execution.candidate.kind)) {
1333 if (execution.candidate.kind == ClimbingActionKind::BeamBalance)
1334 phase_ = ClimbingPhase::Balanced;
1335 else if (execution.candidate.kind == ClimbingActionKind::PoleSwing ||
1336 execution.candidate.kind == ClimbingActionKind::BarSwing)
1337 phase_ = ClimbingPhase::Swinging;
1338 else
1339 phase_ = ClimbingPhase::Hanging;
1340 execution.branchWindowOpen = true;
1341 output.branchWindowOpen = true;
1342 enqueueEvent({execution.candidate.kind == ClimbingActionKind::LedgeGrab
1343 ? ClimbingEventKind::Hanging
1344 : ClimbingEventKind::AnchorReached,
1345 output.actionId, step.tick, execution.executionId});
1346 } else if (execution.candidate.kind == ClimbingActionKind::LedgeGrab) {
1347 phase_ = ClimbingPhase::Hanging;
1348 execution.branchWindowOpen = true;
1349 output.branchWindowOpen = true;
1350 enqueueEvent({ClimbingEventKind::Hanging, output.actionId, step.tick, execution.executionId});
1351 } else {
1352 phase_ = ClimbingPhase::Completed;
1353 terminalCode_ = "climbing.completed";
1354 if (execution.candidate.kind != ClimbingActionKind::WallRun)
1355 enqueueEvent({ClimbingEventKind::Landed, output.actionId, step.tick, execution.executionId});
1356 enqueueEvent({ClimbingEventKind::Completed, output.actionId, step.tick, execution.executionId});
1357 if (graphBound) {
1358 auto released = releaseAnchorReservation();
1359 released.ignore("completed graph dismount releases occupancy");
1360 }
1361 }
1362 output.phase = phase_;
1363 if (phase_ == ClimbingPhase::Completed) {
1364 output.terminalVelocity = terminalVelocityFor(execution.action, actual,
1365 static_cast<float>(1.0 / step.delta.seconds()));
1366 output.hasTerminalVelocity = true;
1367 execution_.reset();
1368 }
1369 }
1370 return eve::Result<ClimbingAdvance>::success(std::move(output));
1371}

1373eve::Result<void> ClimbingRuntime::drop(eve::SimulationTick tick) {
1374 if (phase_ != ClimbingPhase::Hanging || !execution_)
1375 return climbingFailure<void>(eve::DiagnosticCode::PreconditionViolation, "drop requires a hanging execution",
1376 "runtime.phase");
1377 if (tick <= execution_->lastTick)
1378 return climbingFailure<void>(eve::DiagnosticCode::Conflict,
1379 "drop tick must be newer than the last execution tick", "tick");
1380 auto eventCapacity = requireEventCapacity(1, tick);
1381 if (!eventCapacity) return eventCapacity;
1382 if (execution_->anchorGraph.isValid()) {
1383 auto released = releaseAnchorReservation();
1384 if (!released && released.error() && released.error()->code() != eve::DiagnosticCode::StaleHandle &&
1385 released.error()->code() != eve::DiagnosticCode::NotFound)
1386 return released;
1387 }
1388 enqueueEvent({ClimbingEventKind::Dropped, execution_->candidate.actionId, tick, execution_->executionId});
1389 execution_->velocity = {0.f, -profile_.dropInitialSpeed, 0.f};
1390 execution_->lastTick = tick;
1391 execution_->candidate.support = HangSupport::None;
1392 execution_->compactCollisionActive = false;
1393 execution_->branchWindowOpen = false;
1394 phase_ = ClimbingPhase::Dropping;
1395 terminalCode_.clear();
1396 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
1397}

1399eve::Result<void> ClimbingRuntime::climbUp(eve::SimulationTick tick) {
1400 if (phase_ != ClimbingPhase::Hanging || !execution_)
1401 return climbingFailure<void>(eve::DiagnosticCode::PreconditionViolation,
1402 "climb up requires a hanging execution", "runtime.phase");
1403 if (tick <= execution_->lastTick)
1404 return climbingFailure<void>(eve::DiagnosticCode::Conflict,
1405 "climb-up tick must be newer than the last execution tick", "tick");
1406 const auto found = std::find_if(profile_.actions.begin(), profile_.actions.end(), [&](const auto& action) {
1407 return (action.kind == ClimbingActionKind::ClimbUp || action.kind == ClimbingActionKind::Mantle) &&
1408 execution_->candidate.obstacleHeight + epsilon >= action.minHeight &&
1409 execution_->candidate.obstacleHeight - epsilon <= action.maxHeight;
1410 });
1411 if (found == profile_.actions.end())
1412 return climbingFailure<void>(eve::DiagnosticCode::NotFound, "no compatible climb-up action is registered",
1413 "profile.actions");
1414 if (execution_->anchorGraph.isValid()) {
1415 auto released = releaseAnchorReservation();
1416 if (!released && released.error() && released.error()->code() != eve::DiagnosticCode::StaleHandle &&
1417 released.error()->code() != eve::DiagnosticCode::NotFound)
1418 return released;
1419 }
1420 execution_->candidate.actionId = found->id;
1421 execution_->candidate.kind = ClimbingActionKind::ClimbUp;
1422 execution_->candidate.definitionGeneration = definitionGeneration_;

1442eve::Result<void> ClimbingRuntime::cancel(ClimbingCancelReason reason, eve::SimulationTick tick) {
1443 if (!isActivePhase(phase_) || !execution_)
1444 return climbingFailure<void>(eve::DiagnosticCode::PreconditionViolation, "no climbing execution is active",
1445 "runtime.phase");
1446 if (tick < execution_->lastTick)
1447 return climbingFailure<void>(eve::DiagnosticCode::Conflict,
1448 "cancel tick must not precede the last execution tick", "tick");
1449 const ClimbingActionDefinition& action = execution_->action;
1450 const float t =
1451 action.duration.nanoseconds() > 0
1452 ? static_cast<float>(std::clamp(execution_->elapsed.seconds() / action.duration.seconds(), 0.0, 1.0))
1453 : 0.f;
1454 if (phase_ != ClimbingPhase::Hanging && phase_ != ClimbingPhase::Dropping &&
1455 (t < action.cancelWindowStart || t > action.cancelWindowEnd))
1456 return climbingFailure<void>(eve::DiagnosticCode::PreconditionViolation,
1457 "the active definition does not allow cancellation in this window",
1458 "action.cancelWindow");
1459 auto eventCapacity = requireEventCapacity(1, tick);
1460 if (!eventCapacity) return eventCapacity;
1461 if (execution_->anchorGraph.isValid()) {
1462 auto released = releaseAnchorReservation();
1463 if (!released && released.error() && released.error()->code() != eve::DiagnosticCode::StaleHandle &&
1464 released.error()->code() != eve::DiagnosticCode::NotFound)
1465 return released;
1466 }
1467 enqueueEvent({ClimbingEventKind::Cancelled, execution_->candidate.actionId, tick, execution_->executionId});
1468 terminalCode_ = reason == ClimbingCancelReason::LinkStale ? "climbing.link.stale"
1469 : reason == ClimbingCancelReason::AnchorStale ? "climbing.anchor.stale"
1470 : reason == ClimbingCancelReason::MotionBlocked ? "climbing.motion.blocked"
1471 : reason == ClimbingCancelReason::WarpBudgetExceeded ? "climbing.warp.budget_exceeded"
1472 : "climbing.cancelled";
1473 execution_.reset();
1474 phase_ = ClimbingPhase::Cancelled;
1475 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
1476}
