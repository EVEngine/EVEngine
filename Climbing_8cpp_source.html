30 const auto store = [&](ClimbingCandidate& slot) {
31 std::string reused = std::move(slot.actionId);
32 slot = std::move(candidate);
33 if (reused.capacity() >= actionId.size()) {
34 reused.assign(actionId);
35 slot.actionId.swap(reused);

37 slot.actionId.assign(actionId);

40 if (size_ < Capacity) {
41 store(values_[size_++]);
42 return;
43 }
44 auto worst = std::max_element(values_.begin(), values_.end(), isCandidateLess);
45 const bool better = candidate.score != worst->score
46 ? candidate.score < worst->score
47 : actionId != worst->actionId
48 ? actionId < std::string_view(worst->actionId)
49 : candidate.obstacleBodyId != worst->obstacleBodyId

52 if (better) store(*worst);

483 measuredDepth = baseDepth + offset;

486 previousNormal.z * normal.z,

511 continue;

518 }

593 const float heightCenter = (action.minHeight + action.maxHeight) * 0.5f;

621 candidate.obstacleDepth = measuredDepth;
622 candidate.gapDistance = distance;

626 candidate.supportShapeTag = top.shapeTag;
627 candidate.supportMaterialId = top.materialId;

738 return released;
