91void RecipeDescriptor::applyDefaults(Params& values) const {
92 for (const ParamDescriptor& param : params) {
93 if (values.has(param.key)) continue;
94 switch (param.kind) {
95 case ParamKind::Integer: values.setInt(param.key, std::stoi(param.defaultValue)); break;
96 case ParamKind::Boolean: values.setBool(param.key, std::stoi(param.defaultValue) != 0); break;
97 case ParamKind::Float: values.setFloat(param.key, std::stof(param.defaultValue)); break;
98 case ParamKind::String:
99 case ParamKind::Choice: values.setString(param.key, param.defaultValue); break;
100 }
101 }
102}
