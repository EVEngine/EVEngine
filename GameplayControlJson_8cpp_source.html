73Result<GameplaySession> session(const Value::Object& root) {
74 const auto found = root.find("session");
75 if (found == root.end())
76 return failure<GameplaySession>(DiagnosticCode::ParseError,
77 "gameplay request requires a session", "request.session");
78 auto object = asObject(found->second, "request.session");
79 if (!object) return Result<GameplaySession>::failure(object.status());
80 auto known = knownFields(*object.value(), {"access", "controlledSubjects", "id"},
81 "request.session");
82 if (!known) return Result<GameplaySession>::failure(known.status());
83 auto id = stringMember(*object.value(), "id", "request.session");
84 auto access = stringMember(*object.value(), "access", "request.session");
85 auto subjects = member(*object.value(), "controlledSubjects", Value::Type::Array, "request.session");
86 if (!id) return Result<GameplaySession>::failure(id.status());
87 if (!access) return Result<GameplaySession>::failure(access.status());
88 if (!subjects) return Result<GameplaySession>::failure(subjects.status());
89 GameplaySession result;
90 result.id = id.value();
91 if (result.id.empty())
92 return failure<GameplaySession>(DiagnosticCode::ParseError,
93 "gameplay session id must not be empty", "request.session.id");
94 if (access.value() == "player") result.access = GameplayAccess::PlayerEquivalent;
95 else if (access.value() == "test-driver") result.access = GameplayAccess::TestDriver;
96 else if (access.value() == "developer-cheat") result.access = GameplayAccess::DeveloperCheat;
97 else
98 return failure<GameplaySession>(DiagnosticCode::ParseError,
99 "unknown gameplay access profile", "request.session.access");
100 const auto* values = subjects.value()->getIf<Value::Array>();
101 for (std::size_t index = 0; index < values->size(); ++index) {
102 if (!(*values)[index].isString())
103 return failure<GameplaySession>(DiagnosticCode::ParseError,
104 "controlled subject must be a UUID string",
105 "request.session.controlledSubjects." + std::to_string(index));
106 auto parsed = subject((*values)[index].asString(),
107 "request.session.controlledSubjects." + std::to_string(index));
108 if (!parsed) return Result<GameplaySession>::failure(parsed.status());
109 result.controlledSubjects.push_back(parsed.value());
110 }
111 return Result<GameplaySession>::success(std::move(result));
112}

114Result<IGameplayControlProvider*> provider(std::string_view domain) {
115 IGameplayControlProvider* match = nullptr;
116 bool duplicate = false;
117 cap::forEach<IGameplayControlProvider>([&](auto* candidate) {
118 if (candidate && candidate->gameplayDomain() == domain) {
119 duplicate = match != nullptr;
120 match = candidate;
121 }
122 });
123 if (duplicate)
124 return failure<IGameplayControlProvider*>(DiagnosticCode::Conflict,
125 "multiple gameplay providers publish the same domain",
126 "request.domain");
127 if (!match)
128 return failure<IGameplayControlProvider*>(DiagnosticCode::NotFound,
129 "gameplay provider domain was not found", "request.domain");
130 return Result<IGameplayControlProvider*>::success(match);
131}

196Result<Value> executeGameplayControlRequest(const Value& request) {
197 auto root = asObject(request, "request");
198 if (!root) return Result<Value>::failure(root.status());
199 auto valid = validateRoot(*root.value());
200 if (!valid) return Result<Value>::failure(valid.status());
201 auto operation = stringMember(*root.value(), "op", "request");
202 if (!operation) return Result<Value>::failure(operation.status());
203 if (operation.value() == "domains") {
204 std::vector<std::string> domains;
205 cap::forEach<IGameplayControlProvider>([&](auto* candidate) {
206 if (candidate) domains.emplace_back(candidate->gameplayDomain());
207 });
208 std::sort(domains.begin(), domains.end());
209 Value::Array output;
210 for (auto& domain : domains) output.emplace_back(std::move(domain));
211 return Result<Value>::success(Value(Value::Object{{"domains", Value(std::move(output))}}));
212 }
213 auto domain = stringMember(*root.value(), "domain", "request");
214 if (!domain) return Result<Value>::failure(domain.status());
215 auto target = provider(domain.value());
216 if (!target) return Result<Value>::failure(target.status());
217 auto access = session(*root.value());
218 if (!access) return Result<Value>::failure(access.status());
219 auto instance = rootSubject(*root.value(), "instance");
220 if (!instance) return Result<Value>::failure(instance.status());
221
222 if (operation.value() == "observe") {
223 auto result = target.value()->observeGameplay(access.value(), instance.value());
224 if (!result) return Result<Value>::failure(result.status());
225 return Result<Value>::success(encodeObservation(std::move(result).takeValue()));
226 }
227 if (operation.value() == "actions") {
228 auto actor = rootSubject(*root.value(), "subject");
229 if (!actor) return Result<Value>::failure(actor.status());
230 auto result = target.value()->availableGameplayActions(access.value(), instance.value(), actor.value());
231 if (!result) return Result<Value>::failure(result.status());
232 return Result<Value>::success(Value(Value::Object{{"actions", encodeActions(std::move(result).takeValue())}}));
233 }
234 if (operation.value() == "submit") {
235 const auto found = root.value()->find("command");
236 if (found == root.value()->end())
237 return failure<Value>(DiagnosticCode::ParseError, "submit requires a command", "request.command");
238 auto commandObject = asObject(found->second, "request.command");
239 if (!commandObject) return Result<Value>::failure(commandObject.status());
240 auto known = knownFields(*commandObject.value(),
241 {"action", "expectedRevision", "id", "observedTick", "parameters", "subject"},
242 "request.command");
243 if (!known) return Result<Value>::failure(known.status());
244 auto commandId = stringMember(*commandObject.value(), "id", "request.command");
245 auto actionText = stringMember(*commandObject.value(), "action", "request.command");
246 auto actorText = stringMember(*commandObject.value(), "subject", "request.command");
247 auto observedTick = uintMember(*commandObject.value(), "observedTick", "request.command");
248 auto expectedRevision = uintMember(*commandObject.value(), "expectedRevision", "request.command");
249 const auto parameters = commandObject.value()->find("parameters");
250 if (!commandId) return Result<Value>::failure(commandId.status());
251 if (!actionText) return Result<Value>::failure(actionText.status());
252 if (!actorText) return Result<Value>::failure(actorText.status());
253 if (!observedTick) return Result<Value>::failure(observedTick.status());
254 if (!expectedRevision) return Result<Value>::failure(expectedRevision.status());
255 if (parameters == commandObject.value()->end())
256 return failure<Value>(DiagnosticCode::ParseError, "submit command requires parameters",
257 "request.command.parameters");
258 const auto parsedAction = LogicalId::parse(actionText.value());
259 if (!parsedAction)
260 return failure<Value>(DiagnosticCode::ParseError, "command action must be a logical id",
261 "request.command.action");
262 auto actor = subject(actorText.value(), "request.command.subject");
263 if (!actor) return Result<Value>::failure(actor.status());
264 GameplayCommand command;
265 command.id = commandId.value();
266 command.action = *parsedAction;
267 command.subject = actor.value();
268 command.observedTick = SimulationTick(observedTick.value());
269 command.expectedRevision = expectedRevision.value();
270 command.parameters = parameters->second;
271 auto result = target.value()->submitGameplay(access.value(), instance.value(), command);
272 if (!result) return Result<Value>::failure(result.status());
273 return Result<Value>::success(Value(Value::Object{{"receipt", encodeReceipt(std::move(result).takeValue())}}));
274 }
275 if (operation.value() == "advance") {
276 auto tick = uintMember(*root.value(), "tick", "request");
277 auto delta = member(*root.value(), "deltaNanoseconds", Value::Type::Int64, "request");
278 if (!tick) return Result<Value>::failure(tick.status());
279 if (!delta) return Result<Value>::failure(delta.status());
280 if (delta.value()->asInt() < 0)
281 return failure<Value>(DiagnosticCode::ParseError, "delta must be non-negative",
282 "request.deltaNanoseconds");
283 SimulationStep step{SimulationTick(tick.value()), Duration::fromNanoseconds(delta.value()->asInt())};
284 auto result = target.value()->advanceGameplay(access.value(), instance.value(), step);
285 if (!result) return Result<Value>::failure(result.status());
286 return Result<Value>::success(encodeObservation(std::move(result).takeValue()));
287 }
288 if (operation.value() == "events") {
289 auto cursor = uintMember(*root.value(), "afterSequence", "request");
290 if (!cursor) return Result<Value>::failure(cursor.status());
291 auto result = target.value()->gameplayEvents(access.value(), instance.value(), cursor.value());
292 if (!result) return Result<Value>::failure(result.status());
293 return Result<Value>::success(Value(Value::Object{{"events", encodeEvents(std::move(result).takeValue())}}));
294 }
295 return failure<Value>(DiagnosticCode::Unsupported, "unsupported gameplay control operation", "request.op");
296}
