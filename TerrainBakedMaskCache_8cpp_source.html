37 const Heightmap& raster) {
38 using namespace raster_detail;
39 if (!impl_ || !validIdentity(terrainId) || !validIdentity(maskGuid) || !validRaster(raster))
40 return invalid("terrain.bakedMask.store: stable bounded identities and finite raster required");
41 auto candidate = impl_->entries;
42 auto it = std::find_if(candidate.begin(), candidate.end(), [&](const auto& entry) {
43 return entry.terrainId == terrainId && entry.maskGuid == maskGuid;

45 if (it == candidate.end()) candidate.push_back({terrainId, maskGuid, raster, false});

47 it->raster = raster;
48 it->dirty = false;

50 impl_->entries.swap(candidate);
