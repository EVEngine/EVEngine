121bool structureMatches(const UIHost::Tree &tree, int nodeIndex, const WidgetDesc &desc) {
122 if (nodeIndex < 0 || nodeIndex >= int(tree.nodes.size())) return false;
123 const UINode &n = tree.nodes[size_t(nodeIndex)];
124 if (n.type != desc.type) return false;
125 const std::string &dk = desc.reconcileKey();
126 if (!dk.empty() && !n.key.empty() && n.key != dk) return false;
127
128 std::vector<std::string> oldKeys;
129 for (int c = n.firstChild; c >= 0; c = tree.nodes[size_t(c)].nextSibling) {
130 oldKeys.push_back(tree.nodes[size_t(c)].key);
131 }
132 if (oldKeys.size() != desc.children.size()) return false;
133 for (size_t i = 0; i < desc.children.size(); ++i) {
134 const std::string &ck = desc.children[i].reconcileKey();
135 if (ck.empty() || oldKeys[i].empty()) {
136 if (!(ck.empty() && oldKeys[i].empty())) return false;
137 } else if (ck != oldKeys[i]) {
138 return false;
139 }
140 }
141
142 int child = n.firstChild;
143 for (const auto &ch : desc.children) {
144 if (!structureMatches(tree, child, ch)) return false;
145 child = tree.nodes[size_t(child)].nextSibling;
146 }
147 return true;
148}

683void applyTree(UIHost *host, WidgetDesc root) {
684 if (!host) return;
685 auto t = host->tree();
686 t->nodes.clear();
687 t->clickHandlers.clear();
688 t->toggleHandlers.clear();
689 t->valueHandlers.clear();
690 t->textHandlers.clear();
691 t->root = -1;
692 t->root = appendNode(*t, std::move(root));
693 t->dirty = true;
694}
695
696bool applyTreeReconcile(UIHost *host, WidgetDesc root) {
697 if (!host) return true;
698 auto t = host->tree();
699 if (t->root < 0 || t->nodes.empty() || !structureMatches(*t, t->root, root)) {
700 applyTree(host, std::move(root));
701 return true;
702 }
703 patchProps(*t, t->root, std::move(root));
704 t->dirty = false;
705 return false;
706}
