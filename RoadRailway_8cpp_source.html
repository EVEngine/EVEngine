76Result<RoadStyle> makeRailwayStyle(int tracks) {
77 if (tracks < 1 || tracks > 2)
78 return Result<RoadStyle>::failure(
79 Diagnostic::error(DiagnosticCode::InvalidArgument, "railway tracks must be 1 or 2", "tracks"));
80
81 RoadStyle style;
82 style.laneWidth = kStandardGauge;
83 style.curbWidth = 0.f;
84 style.curbHeight = 0.f;
85 style.sidewalkWidth = 0.f;
86 style.sidewalkHeight = 0.f;
87 style.markingWidth = 0.f;
88 style.dashLength = 0.f;
89 style.dashGap = 0.f;
90 style.deckThickness = 0.35f;
91 style.pierClearance = 100.f;
92 style.uvMeters = 4.f;
93 style.sideObjectsLeft = false;
94 style.sideObjectsRight = false;
95 style.automaticStructures = true;
96 style.speedLimitMps = 33.3333f; // ~120 km/h rail cruise default for nav overlay.
97 style.samplingSpacingMeters = 2.f;
98
99 const float halfExtent = tracks == 1 ? 1.6f : (0.5f * kTrackCenters + 1.6f);
100 const float toe = halfExtent + 0.45f;
101 // Open loft ribbon: left toe → ballast crown (asphalt driving span) → right toe.
102 style.crossSection = {{-toe, -0.18f, RoadMaterial::Deck},
103 {-halfExtent, 0.f, RoadMaterial::Asphalt},
104 {halfExtent, 0.f, RoadMaterial::Deck},
105 {toe, -0.18f, RoadMaterial::Deck}};
106
107 std::uint32_t nextId = 101;
108 for (int track = 0; track < tracks; ++track) {
109 const float center =
110 tracks == 1 ? 0.f : (static_cast<float>(track) - 0.5f * static_cast<float>(tracks - 1)) * kTrackCenters;
111 style.facilities.push_back(railFacility(nextId++, center - 0.5f * kStandardGauge));
112 style.facilities.push_back(railFacility(nextId++, center + 0.5f * kStandardGauge));
113 style.facilities.push_back(sleeperFacility(nextId++, center, 0.5f * kStandardGauge + 0.25f));
114 }
115
116 auto valid = makeRoadProfile(style, tracks, 0);
117 if (!valid.ok()) return Result<RoadStyle>::failure(valid.status());
118 return Result<RoadStyle>::success(std::move(style));
119}

121Result<void> applyRailwayRoadType(RoadNetwork& network, int tracks) {
122 auto style = makeRailwayStyle(tracks);
123 if (!style.ok()) return Result<void>::failure(style.status());
124
125 auto candidate = network;
126 for (const auto& edge : network.edges()) {
127 RoadStyle next = style.value();
128 next.structures = edge.style.structures;
129 next.automaticStructures = edge.style.automaticStructures;
130 next.deckThickness = edge.style.deckThickness;
131 next.pierWidth = edge.style.pierWidth;
132 next.pierDepth = edge.style.pierDepth;
133 next.pierSpacing = edge.style.pierSpacing;
134 next.pierClearance = edge.style.pierClearance;
135 next.terrainPolicy = edge.style.terrainPolicy;
136 auto styled = candidate.setEdgeStyle(edge.id, next);
137 if (!styled.ok()) return styled;
138 auto lanes = candidate.setEdgeLaneCounts(edge.id, tracks, 0);
139 if (!lanes.ok()) return Result<void>::failure(lanes.status());
140 }
141 for (const auto& node : candidate.nodes()) {
142 int incident = 0;
143 for (const auto& edge : candidate.edges())
144 if (edge.from == node.id || edge.to == node.id) ++incident;
145 if (incident < 2) continue;
146 auto linked = candidate.connectAllTurns(node.id);
147 if (!linked.ok()) return Result<void>::failure(linked.status());
148 }
149 network = std::move(candidate);
150 return Result<void>::success();
151}
