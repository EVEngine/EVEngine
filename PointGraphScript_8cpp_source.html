15SQInteger sqPointGraphConnect(HSQUIRRELVM vm) {
16 const SQInteger n = sq_gettop(vm);
17 if (n < 3 || n > 4) return sq_throwerror(vm, "PointGraph.connect expects (from, to [, inputIndex=0])");
18 PointGraph* self = nullptr;
19 if (SQ_FAILED(sq_getinstanceup(vm, 1, reinterpret_cast<SQUserPointer*>(&self), nullptr)) || !self)
20 return sq_throwerror(vm, "invalid PointGraph");
21 const SQChar* from = nullptr;
22 const SQChar* to = nullptr;
23 SQInteger inputIndex = 0;
24 if (SQ_FAILED(sq_getstring(vm, 2, &from)) || SQ_FAILED(sq_getstring(vm, 3, &to)))
25 return sq_throwerror(vm, "PointGraph.connect expects string node ids");
26 if (n >= 4 && SQ_FAILED(sq_getinteger(vm, 4, &inputIndex)))
27 return sq_throwerror(vm, "PointGraph.connect inputIndex must be an integer");
28 sq_pushbool(vm, self->connect(from ? from : "", to ? to : "", static_cast<int>(inputIndex)) ? SQTrue : SQFalse);
29 return 1;
30}
