124Value InventoryControl::bagState(const Entry& entry) const {
125 Bag* bag = entry.bag;
126 Value::Array slots;
127 if (bag != nullptr) {
128 const int slotCount = bag->getSlotCount();
129 for (int slot = 0; slot < slotCount; ++slot) {
130 const std::string itemId = bag->getSlotItemId(slot);
131 if (itemId.empty()) continue;
132 slots.emplace_back(
133 Value::Object{{"slot", Value(static_cast<std::int64_t>(slot))},
134 {"itemId", Value(itemId)},
135 {"quantity", Value(static_cast<std::int64_t>(bag->getSlotQuantity(slot)))},
136 {"instanceId", Value(static_cast<std::int64_t>(bag->getSlotInstanceId(slot)))}});
137 }
138 }
139 Value::Object state{{"bagId", Value(bag != nullptr ? bag->getId() : std::string{})},
140 {"kind", Value(bag != nullptr ? bag->getKind() : std::string{})},
141 {"slots", Value(std::move(slots))}};
142 if (bag != nullptr) {
143 state.emplace("slotCount", Value(static_cast<std::int64_t>(bag->getSlotCount())));
144 state.emplace("usedWeight", Value(static_cast<double>(bag->getUsedWeight())));
145 state.emplace("maxWeight", Value(static_cast<double>(bag->getMaxWeight())));
146 state.emplace("usedVolume", Value(static_cast<double>(bag->getUsedVolume())));
147 state.emplace("maxVolume", Value(static_cast<double>(bag->getMaxVolume())));
148 state.emplace("capacityPolicy", Value(bag->getCapacityPolicy()));
149 }
150 state.emplace("equipment", equipmentState(entry));
151 return Value(std::move(state));
152}

154Value InventoryControl::equipmentState(const Entry& entry) const {
155 EquipmentSet* equipment = entry.equipment;
156 Value::Array slots;
157 if (equipment != nullptr) {
158 const int slotCount = equipment->getSlotCount();
159 for (int index = 0; index < slotCount; ++index) {
160 const std::string slotName = equipment->getSlotName(index);
161 if (slotName.empty()) continue;
162 Value::Object slot{{"slot", Value(slotName)}};
163 const std::string itemId = equipment->getSlotItemId(slotName);
164 if (!itemId.empty()) {
165 slot.emplace("itemId", Value(itemId));
166 slot.emplace("quantity", Value(static_cast<std::int64_t>(equipment->getSlotQuantity(slotName))));
167 slot.emplace("instanceId", Value(static_cast<std::int64_t>(equipment->getSlotInstanceId(slotName))));
168 }
169 slots.emplace_back(std::move(slot));
170 }
171 }
172 return Value(std::move(slots));

193Result<std::vector<GameplayActionDescriptor>> InventoryControl::availableGameplayActions(const GameplaySession& session,
194 SubjectRef instance,
195 SubjectRef subject) const {
196 auto observed = observeGameplay(session, instance);
197 if (!observed) return Result<std::vector<GameplayActionDescriptor>>::failure(observed.status());
198 const Entry* entry = find(instance);
199 if (entry == nullptr)
200 return failure<std::vector<GameplayActionDescriptor>>(DiagnosticCode::NotFound,
201 "inventory gameplay instance was not found", "instance");
202 if (subject != entry->owner)
203 return failure<std::vector<GameplayActionDescriptor>>(DiagnosticCode::PreconditionViolation,
204 "inventory action subject must be its owner", "subject");
205
206 const Value stringType(Value::Object{{"type", Value("string")}});
207 const Value integerType(Value::Object{{"type", Value("integer")}});
208 std::vector<GameplayActionDescriptor> actions{
209 {id(kRemoveItem), Value(Value::Object{{"itemId", stringType}, {"quantity", integerType}})},
210 {id(kMoveSlot),
211 Value(Value::Object{{"fromSlot", integerType}, {"toSlot", integerType}, {"quantity", integerType}})},
212 };
213 // A set without slots carries no equipment authority, so discovery stays
214 // truthful: everything advertised here can actually succeed.
215 if (entry->equipment != nullptr && entry->equipment->getSlotCount() > 0) {
216 actions.push_back({id(kEquip), Value(Value::Object{{"bagSlot", integerType}, {"equipSlot", stringType}})});
217 actions.push_back({id(kUnequip), Value(Value::Object{{"equipSlot", stringType}})});
218 }
219 // Granting items is not a player action; it is advertized only to the
220 // profiles that are allowed to use it, so discovery never offers an action
221 // the same session would be refused.
222 if (session.access != GameplayAccess::PlayerEquivalent) {
223 actions.push_back({id(kAddItem), Value(Value::Object{{"itemId", stringType}, {"quantity", integerType}})});
224 }
225 return Result<std::vector<GameplayActionDescriptor>>::success(std::move(actions));
226}

228Result<GameplayCommandReceipt> InventoryControl::submitGameplay(const GameplaySession& session, SubjectRef instance,
229 const GameplayCommand& command) {
230 Entry* entry = find(instance);
231 if (entry == nullptr)
232 return failure<GameplayCommandReceipt>(DiagnosticCode::NotFound, "inventory gameplay instance was not found",
233 "instance");
234 if (!controls(session, *entry) || command.subject != entry->owner)
235 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation,
236 "session does not control the inventory owner", "command.subject");
237 if (command.id.empty())
238 return failure<GameplayCommandReceipt>(DiagnosticCode::InvalidArgument, "command id must not be empty",
239 "command.id");
240 if (command.observedTick != entry->tick || command.expectedRevision != entry->revision)
241 return failure<GameplayCommandReceipt>(
242 DiagnosticCode::Conflict, "inventory command was based on a stale observation", "command.expectedRevision");
243 auto values = object(command.parameters);
244 if (!values) return Result<GameplayCommandReceipt>::failure(values.status());
245 const Value::Object& parameters = *values.value();
246
247 Bag* bag = entry->bag;
248 EquipmentSet* equipment = entry->equipment;
249 if (bag == nullptr)
250 return failure<GameplayCommandReceipt>(DiagnosticCode::Failed, "inventory instance lost its container",
251 "instance");
252
253 const std::string action = command.action.format();
254 std::string detail;
255 std::string eventType;
256 // How much the authority actually moved. `Bag` removal is "take up to N",
257 // so an agent that asks for more than the container holds gets a success
258 // with the applied amount disclosed here rather than a silent full grant.
259 std::int64_t applied = 0;
260 if (action == kAddItem) {
261 if (session.access == GameplayAccess::PlayerEquivalent)
262 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation,
263 "adding items requires the test-driver or developer-cheat profile",
264 "session.access");
265 auto itemId = text(parameters, "itemId");
266 auto quantity = integer(parameters, "quantity", 1);
267 if (!itemId) return Result<GameplayCommandReceipt>::failure(itemId.status());
268 if (!quantity) return Result<GameplayCommandReceipt>::failure(quantity.status());
269 if (itemId.value().empty() || quantity.value() <= 0)
270 return failure<GameplayCommandReceipt>(DiagnosticCode::InvalidArgument,
271 "add-item needs a non-empty itemId and a positive quantity",
272 "command.parameters");
273 if (!bag->canAddItem(itemId.value(), quantity.value()))
274 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation,
275 bag->canAddItemReason(itemId.value(), quantity.value()),
276 "command.parameters.itemId");
277 const int added = bag->addItem(itemId.value(), quantity.value());
278 if (added <= 0)
279 return failure<GameplayCommandReceipt>(DiagnosticCode::Failed, "bag rejected the granted items",
280 "command.parameters");
281 detail = "added " + std::to_string(added) + " " + itemId.value();
282 eventType = "inventory.item-added";
283 applied = added;
284 } else if (action == kRemoveItem) {
285 auto itemId = text(parameters, "itemId");
286 auto quantity = integer(parameters, "quantity", 1);
287 if (!itemId) return Result<GameplayCommandReceipt>::failure(itemId.status());
288 if (!quantity) return Result<GameplayCommandReceipt>::failure(quantity.status());
289 if (itemId.value().empty() || quantity.value() <= 0)
290 return failure<GameplayCommandReceipt>(DiagnosticCode::InvalidArgument,
291 "remove-item needs a non-empty itemId and a positive quantity",
292 "command.parameters");
293 const int removed = bag->removeItem(itemId.value(), quantity.value());
294 if (removed <= 0)
295 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation, "bag does not hold that item",
296 "command.parameters.itemId");
297 detail = "removed " + std::to_string(removed) + " " + itemId.value();
298 eventType = "inventory.item-removed";
299 applied = removed;
300 } else if (action == kMoveSlot) {
301 auto fromSlot = integer(parameters, "fromSlot", -1);
302 auto toSlot = integer(parameters, "toSlot", -1);
303 auto quantity = integer(parameters, "quantity", 1);
304 if (!fromSlot) return Result<GameplayCommandReceipt>::failure(fromSlot.status());
305 if (!toSlot) return Result<GameplayCommandReceipt>::failure(toSlot.status());
306 if (!quantity) return Result<GameplayCommandReceipt>::failure(quantity.status());
307 if (fromSlot.value() < 0 || toSlot.value() < 0 || quantity.value() <= 0)
308 return failure<GameplayCommandReceipt>(DiagnosticCode::InvalidArgument,
309 "move-slot needs non-negative slots and a positive quantity",
310 "command.parameters");
311 if (!bag->splitStack(fromSlot.value(), quantity.value(), toSlot.value()))
312 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation,
313 "bag could not move that stack", "command.parameters.fromSlot");
314 detail = "moved " + std::to_string(quantity.value()) + " from slot " + std::to_string(fromSlot.value());
315 eventType = "inventory.slot-moved";
316 applied = quantity.value();
317 } else if (action == kEquip) {
318 if (equipment == nullptr)
319 return failure<GameplayCommandReceipt>(DiagnosticCode::Unsupported,
320 "this inventory instance has no equipment set", "command.action");
321 auto bagSlot = integer(parameters, "bagSlot", -1);
322 auto equipSlot = text(parameters, "equipSlot");
323 if (!bagSlot) return Result<GameplayCommandReceipt>::failure(bagSlot.status());
324 if (!equipSlot) return Result<GameplayCommandReceipt>::failure(equipSlot.status());
325 std::string reason;
326 if (bagSlot.value() < 0 || equipSlot.value().empty() ||
327 !equipment->canEquipFromBag(bag, bagSlot.value(), equipSlot.value(), &reason))
328 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation,
329 reason.empty() ? "item cannot be equipped into that slot" : reason,
330 "command.parameters.equipSlot");
331 if (!equipment->equipFromBag(equipSlot.value(), bag, bagSlot.value()))
332 return failure<GameplayCommandReceipt>(DiagnosticCode::Failed, "equip was rejected by the equipment set",
333 "command.parameters.equipSlot");
334 detail = "equipped into " + equipSlot.value();
335 eventType = "inventory.equipped";
336 applied = equipment->getSlotQuantity(equipSlot.value());
337 } else if (action == kUnequip) {
338 if (equipment == nullptr)
339 return failure<GameplayCommandReceipt>(DiagnosticCode::Unsupported,
340 "this inventory instance has no equipment set", "command.action");
341 auto equipSlot = text(parameters, "equipSlot");
342 if (!equipSlot) return Result<GameplayCommandReceipt>::failure(equipSlot.status());
343 if (equipSlot.value().empty() || equipment->isSlotEmpty(equipSlot.value()))
344 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation, "equipment slot is empty",
345 "command.parameters.equipSlot");
346 const int carried = equipment->getSlotQuantity(equipSlot.value());
347 if (!equipment->unequipToBag(equipSlot.value(), bag))
348 return failure<GameplayCommandReceipt>(DiagnosticCode::Failed, "unequip was rejected by the equipment set",
349 "command.parameters.equipSlot");
350 detail = "unequipped " + equipSlot.value();
351 eventType = "inventory.unequipped";
352 applied = carried;
353 } else {
354 return failure<GameplayCommandReceipt>(DiagnosticCode::Unsupported, "unsupported inventory action",
355 "command.action");
356 }
357
358 ++entry->revision;
359 GameplayEvent event;
360 event.sequence = entry->nextEventSequence++;
361 event.tick = entry->tick;
362 event.type = eventType;
363 event.subject = entry->owner;
364 event.causationCommandId = command.id;
365 event.correlationId = command.id;
366 event.payload = Value(Value::Object{
367 {"instance", Value(entry->instance.format())}, {"detail", Value(detail)}, {"quantity", Value(applied)}});
368 entry->events.push_back(std::move(event));
369
370 GameplayCommandReceipt receipt;
371 receipt.commandId = command.id;
372 receipt.executionId = command.id + ".executed";
373 receipt.acceptedTick = entry->tick;
374 receipt.resultingRevision = entry->revision;
375 receipt.details =
376 Value(Value::Object{{"action", Value(action)}, {"detail", Value(detail)}, {"quantity", Value(applied)}});
377 return Result<GameplayCommandReceipt>::success(std::move(receipt), Status::success(StatusCode::Applied));
378}
