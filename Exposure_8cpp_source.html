57void Exposure::ensureTargets(int width, int height) {
58 width = std::max(width, 1);
59 height = std::max(height, 1);
60 if (!meter_) {
61 meter_ = gfx_->newHDRCanvas(1, 1);
62 historyA_ = gfx_->newHDRCanvas(1, 1);
63 historyB_ = gfx_->newHDRCanvas(1, 1);
64 historyRead_ = historyA_;
65 historyWrite_ = historyB_;
66 }
67 if (!output_ || width_ != width || height_ != height) {
68 output_ = gfx_->newHDRCanvas(width, height);
69 width_ = width;
70 height_ = height;
71 historyValid_ = false;
72 }
73 if (!meter_ || !historyRead_ || !historyWrite_ || !output_)
74 throw eve::Exception("Exposure: failed to allocate HDR targets");

77Texture *Exposure::apply(Texture *source, float manualExposure, bool automatic, float minEV,
78 float maxEV, Texture *meterSource, float deltaSeconds) {
79 if (!source) throw eve::Exception("Exposure.apply: null source");
80 ensureTargets(source->getWidth(), source->getHeight());
81 Canvas *previousCanvas = gfx_->getCanvas();
82 Texture *exposureTexture = source;
83 if (!meterSource) meterSource = source;
84
85 if (automatic) {
86 meterShader_->sendFloat("minEV", std::min(minEV, maxEV));
87 meterShader_->sendFloat("maxEV", std::max(minEV, maxEV));
88 meter_->clear(Color(0.f, 0.f, 0.f, 1.f), {}, {});
89 gfx_->setCanvas(meter_);
90 gfx_->drawTexturedRectShaderUV(meterSource, meterShader_, 0.f, 0.f, 1.f, 1.f, 0.f, 0.f,
91 1.f, 1.f, Color(1.f, 1.f, 1.f, 1.f), false,
92 BlendMode::Opaque);
93
94 adaptShader_->sendFloat("deltaSeconds", std::clamp(deltaSeconds, 0.f, 0.25f));
95 adaptShader_->sendFloat("darkSpeed", 1.5f);
96 adaptShader_->sendFloat("brightSpeed", 3.f);
97 adaptShader_->sendFloat("resetHistory", historyValid_ ? 0.f : 1.f);
98 historyWrite_->clear(Color(1.f, 0.f, 0.f, 1.f), {}, {});
99 gfx_->setCanvas(historyWrite_);
100 Texture *meterTexture = meter_->getTexture();
101 Texture *previousExposure = historyValid_ ? historyRead_->getTexture() : meterTexture;
102 gfx_->drawTexturedRectShaderDepthMotion(meterTexture, previousExposure, meterTexture,
103 adaptShader_, 0.f, 0.f, 1.f, 1.f,
104 Color(1.f, 1.f, 1.f, 1.f));
105 exposureTexture = historyWrite_->getTexture();
106 historyValid_ = true;
107 std::swap(historyRead_, historyWrite_);
108 }
109
110 applyShader_->sendFloat("manualExposure", std::max(manualExposure, 0.f));
111 applyShader_->sendFloat("automatic", automatic ? 1.f : 0.f);
112 output_->clear(Color(0.f, 0.f, 0.f, 0.f), {}, {});
113 gfx_->setCanvas(output_);
114 gfx_->drawTexturedRectShaderDepthMotion(source, exposureTexture, source, applyShader_, 0.f, 0.f,
115 float(width_), float(height_),
116 Color(1.f, 1.f, 1.f, 1.f));
117 gfx_->setCanvas(previousCanvas);
118 Texture *result = output_->getTexture();
119 if (!result) throw eve::Exception("Exposure: output texture missing");
120 return result;
121}
