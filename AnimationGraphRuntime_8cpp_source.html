27EditorResult<animation::AnimStateMachine*> AnimationStateGraphRuntimeBuilder::build(
28 const GraphDocumentData& graph, animation::AnimSkeleton* skeleton, const ClipResolver& clips) const {
29 if (!skeleton || !clips)
30 return runtimeError<animation::AnimStateMachine*>(EditorStatus::Rejected,
31 "editor.animation.runtime-input",
32 "Animation skeleton and clip resolver are required");
33 AnimationStateGraphDomain domain;
34 const AnimationGraphCompileResult compiled = domain.compile(graph);
35 if (compiled.status != EditorStatus::Applied)
36 return EditorResult<animation::AnimStateMachine*>::failure(
37 eve::Status(compiled.status, compiled.diagnostics));
38
39 std::map<GraphNodeId, const GraphNodeRecord*> nodes;
40 std::map<GraphPinId, GraphNodeId> pinOwners;
41 for (const GraphNodeRecord& node : graph.nodes) {
42 nodes.emplace(node.id, &node);
43 for (const GraphPinRecord& pin : node.pins) pinOwners.emplace(pin.id, node.id);
44 }
45 std::map<GraphNodeId, GraphNodeId> from;
46 std::map<GraphNodeId, GraphNodeId> to;
47 for (const GraphEdgeRecord& edge : graph.edges) {
48 const GraphNodeId fromNode = pinOwners.at(edge.from);
49 const GraphNodeId toNode = pinOwners.at(edge.to);
50 if (nodes.at(fromNode)->type == "animation.state") from[toNode] = fromNode;
51 else to[fromNode] = toNode;
52 }
53
54 auto machine = std::make_unique<animation::AnimStateMachine>(skeleton);
55 for (const auto& [id, node] : nodes) {
56 if (node->type != "animation.state") continue;
57 const auto* clipAssetValue = runtimeField(node->properties, "clip");
58 const auto* clipAsset = clipAssetValue->getIf<std::string>();
59 animation::AnimClip* clip = clips(*clipAsset);
60 if (!clip)
61 return runtimeError<animation::AnimStateMachine*>(EditorStatus::NotFound,
62 "editor.animation.clip-not-found",
63 "Animation clip asset could not be resolved: " +
64 *clipAsset);
65 machine->addState(id.value(), clip);
66 }
67 const auto* entryValue = runtimeField(graph.parameters, "entry");
68 machine->setEntry(*entryValue->getIf<std::string>());
69
70 for (const auto& [id, node] : nodes) {
71 if (node->type != "animation.transition") continue;
72 const double blend = *runtimeField(node->properties, "blendSeconds")->getIf<double>();
73 const bool hasExit = *runtimeField(node->properties, "hasExitTime")->getIf<bool>();
74 const double exitTime = *runtimeField(node->properties, "exitTime")->getIf<double>();
75 const std::string kind = *runtimeField(node->properties, "conditionKind")->getIf<std::string>();
76 const std::string parameter = *runtimeField(node->properties, "parameter")->getIf<std::string>();
77 const std::string op = *runtimeField(node->properties, "operator")->getIf<std::string>();
78 const double threshold = *runtimeField(node->properties, "threshold")->getIf<double>();
79 const bool boolValue = *runtimeField(node->properties, "boolValue")->getIf<bool>();
80 const int transition = machine->addTransition(from.at(id).value(), to.at(id).value(),
81 static_cast<float>(blend));
82 machine->setHasExitTime(transition, hasExit);
83 if (hasExit) machine->setExitTime(transition, static_cast<float>(exitTime));
84 if (kind == "float")
85 machine->addFloatCondition(transition, parameter, op, static_cast<float>(threshold));
86 else if (kind == "bool")
87 machine->addBoolCondition(transition, parameter, boolValue);
88 else if (kind == "trigger")
89 machine->addTriggerCondition(transition, parameter);
90 }
91 return eve::editing::applied<animation::AnimStateMachine*>(machine.release());
92}
