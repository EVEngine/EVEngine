61bool validRelativeSavePath(const std::filesystem::path &path) {
62 if (path.empty() || path.is_absolute() || path.has_root_name() || path.has_root_directory()) return false;
63 for (const auto &part : path) {
64 if (part == ".." || part == ".") return false;
65 }
66 return true;

87eve::Result<void> Filesystem::writeTextAtomic(std::string_view filename, std::string_view text) {
88#if defined(EVENGINE_WEBGPU)
89 (void)filename;
90 (void)text;
91 return atomicWriteFailure(eve::DiagnosticCode::Unsupported,
92 "atomic save replacement is unavailable on the WebGPU filesystem", {});
93#else
94 const std::string relativeText(filename);
95 const std::filesystem::path relative = pathFromUtf8(relativeText).lexically_normal();
96 if (!validRelativeSavePath(relative))
97 return atomicWriteFailure(eve::DiagnosticCode::InvalidArgument,
98 "atomic save path must be relative and cannot traverse parents", relativeText);
99 const std::string saveDirectory = getSaveDirectory();
100 if (saveDirectory.empty())
101 return atomicWriteFailure(eve::DiagnosticCode::PreconditionViolation,
102 "filesystem write directory is not configured", relativeText);
103 const std::filesystem::path base = pathFromUtf8(saveDirectory).lexically_normal();
104 const std::filesystem::path target = (base / relative).lexically_normal();
105 if (target.parent_path().empty() || !std::filesystem::exists(target.parent_path()))
106 return atomicWriteFailure(eve::DiagnosticCode::NotFound,
107 "atomic save parent directory does not exist", relativeText);
108
109 const auto sequence = atomicWriteSequence().fetch_add(1, std::memory_order_relaxed);
110#if defined(EVENGINE_WINDOWS)
111 const std::wstring targetPath = target.wstring();
112 const std::wstring temporaryPath = targetPath + L".tmp." + std::to_wstring(GetCurrentProcessId()) +
113 L"." + std::to_wstring(sequence);
114 HANDLE file = CreateFileW(temporaryPath.c_str(), GENERIC_WRITE, 0, nullptr, CREATE_NEW,
115 FILE_ATTRIBUTE_NORMAL | FILE_FLAG_WRITE_THROUGH, nullptr);
116 if (file == INVALID_HANDLE_VALUE)
117 return atomicWriteFailure(eve::DiagnosticCode::Failed,
118 "could not create atomic save temporary file", relativeText);
119 bool written = true;
120 std::size_t offset = 0;
121 while (offset < text.size()) {
122 const DWORD chunk = static_cast<DWORD>(std::min<std::size_t>(text.size() - offset, 0x7fffffffu));
123 DWORD completed = 0;
124 if (!WriteFile(file, text.data() + offset, chunk, &completed, nullptr) || completed != chunk) {
125 written = false;
126 break;
127 }
128 offset += completed;
129 }
130 if (written && !FlushFileBuffers(file)) written = false;
131 if (!CloseHandle(file)) written = false;
132 if (!written) {
133 DeleteFileW(temporaryPath.c_str());
134 return atomicWriteFailure(eve::DiagnosticCode::Failed,
135 "could not write or flush atomic save temporary file", relativeText);
136 }
137 if (!MoveFileExW(temporaryPath.c_str(), targetPath.c_str(),
138 MOVEFILE_REPLACE_EXISTING | MOVEFILE_WRITE_THROUGH)) {
139 DeleteFileW(temporaryPath.c_str());
140 return atomicWriteFailure(eve::DiagnosticCode::Failed,
141 "could not atomically replace save file", relativeText);
142 }
143#else
144 const std::string targetPath = target.string();
145 const std::string temporaryPath = targetPath + ".tmp." + std::to_string(getpid()) + "." +
146 std::to_string(sequence);
147 const int file = ::open(temporaryPath.c_str(), O_WRONLY | O_CREAT | O_EXCL, 0600);
148 if (file < 0)
149 return atomicWriteFailure(eve::DiagnosticCode::Failed,
150 "could not create atomic save temporary file", relativeText);
151 bool written = true;
152 std::size_t offset = 0;
153 while (offset < text.size()) {
154 const ssize_t completed = ::write(file, text.data() + offset, text.size() - offset);
155 if (completed < 0) {
156 if (errno == EINTR) continue;
157 written = false;
158 break;
159 }
160 offset += static_cast<std::size_t>(completed);
161 }
162 if (written && ::fsync(file) != 0) written = false;
163 if (::close(file) != 0) written = false;
164 if (!written) {
165 ::unlink(temporaryPath.c_str());
166 return atomicWriteFailure(eve::DiagnosticCode::Failed,
167 "could not write or flush atomic save temporary file", relativeText);
168 }
169 if (::rename(temporaryPath.c_str(), targetPath.c_str()) != 0) {
170 ::unlink(temporaryPath.c_str());
171 return atomicWriteFailure(eve::DiagnosticCode::Failed,
172 "could not atomically replace save file", relativeText);
173 }
174#ifdef O_DIRECTORY
175 const int directory = ::open(target.parent_path().string().c_str(), O_RDONLY | O_DIRECTORY);
176 if (directory >= 0) {
177 (void)::fsync(directory);
178 (void)::close(directory);
179 }
180#endif
181#endif
182 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
183#endif
184}

275bool Filesystem::isRealDirectory(const std::string &path) const {
276#ifdef EVENGINE_WINDOWS
277 // make sure non-ASCII paths work.
278 std::wstring wpath = to_widestr(path);
279
280 struct _stat buf;
281 if (_wstat(wpath.c_str(), &buf) != 0) return false;
282 return (buf.st_mode & _S_IFDIR) == _S_IFDIR;
283#else
284 // Assume POSIX support...
285 struct stat buf;
286 if (stat(path.c_str(), &buf) != 0) return false;
287 return S_ISDIR(buf.st_mode) != 0;
288#endif
289}

291std::string Filesystem::getExecutablePath() const {
292#if defined(EVENGINE_MACOSX)
293 return macosx::getExecutablePath();
294#elif defined(EVENGINE_IOS)
295 return ios::getExecutablePath();
296#elif defined(EVENGINE_ANDROID)
297 return android::getExecutablePath();
298#elif defined(EVENGINE_WEBGPU)
299 return eve::webgpu_platform::getExecutablePath();
300#elif defined(EVENGINE_WINDOWS)
301 wchar_t buffer[MAX_PATH + 1] = {0};
302 if (GetModuleFileNameW(nullptr, buffer, MAX_PATH) == 0) return "";
303 return to_utf8(buffer);
304#elif defined(EVENGINE_LINUX)
305 char buffer[2048] = {0};
306 ssize_t len = readlink("/proc/self/exe", buffer, 2048);
307 if (len <= 0) return "";
308 return std::string(buffer, len);
309#else
310#error Missing implementation for Filesystem::getExecutablePath!
311#endif
312}
