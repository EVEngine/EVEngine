77Result<DiskAssetScanResult> DiskAssetCatalog::scan() {
78 if (!database_ || contentRoot_.empty())
79 return scanError(EditorStatus::Rejected, "editor.asset.catalog-invalid",
80 "Asset database and Content root are required");
81 DiskAssetScanResult result;
82 std::error_code ec;
83 if (!std::filesystem::exists(contentRoot_)) return eve::editing::applied<DiskAssetScanResult>(std::move(result));
84 std::set<AssetGuid> discovered;
85 for (std::filesystem::recursive_directory_iterator iterator(contentRoot_, ec), end; iterator != end && !ec;
86 iterator.increment(ec)) {
87 if (!iterator->is_regular_file() || iterator->path().extension() != ".evmeta") continue;
88 std::ifstream input(iterator->path(), std::ios::binary);
89 const std::string json{std::istreambuf_iterator<char>(input), std::istreambuf_iterator<char>()};
90 auto parsed = editorValueFromJson(json);
91 const auto* metadata = parsed.ok() ? parsed.value().getIf<EditorValue::Object>() : nullptr;
92 const AssetGuid guid(metadata ? textMember(*metadata, "guid") : std::string{});
93 const std::string type = metadata ? textMember(*metadata, "type") : std::string{};
94 if (!parsed.ok() || !metadata || guid.empty() || type.empty()) {
95 result.diagnostics.push_back(editing::ruleDiagnostic(
96 eve::DiagnosticCode::InvalidArgument, RuleId("editor.asset.sidecar-invalid"),
97 DiagnosticSeverity::Error,
98 "Invalid asset sidecar: " + iterator->path().generic_string()));
99 continue;
100 }
101 if (!discovered.emplace(guid).second) {
102 result.diagnostics.push_back(editing::ruleDiagnostic(
103 eve::DiagnosticCode::Conflict, RuleId("editor.asset.duplicate-guid"),
104 DiagnosticSeverity::Error, "Duplicate asset GUID: " + guid.value()));
105 continue;
106 }
107 std::filesystem::path source = iterator->path();
108 source.replace_extension();
109 if (!std::filesystem::is_regular_file(source)) {
110 result.diagnostics.push_back(editing::ruleDiagnostic(
111 eve::DiagnosticCode::NotFound, RuleId("editor.asset.source-missing"),
112 DiagnosticSeverity::Error,
113 "Asset sidecar source is missing: " + source.generic_string()));
114 continue;
115 }
116 const auto relative = source.lexically_relative(contentRoot_);
117 const std::string hash = fileHash(source);
118 const std::string fingerprint = relative.generic_string() + ":" + hash + ":" + json;
119 AssetRecord record;
120 record.guid = guid;
121 record.logicalUri = "content://" + relative.generic_string();
122 record.typeId = type;
123 record.sourceUri = source.generic_string();
124 record.sourceHash = hash;
125 record.schemaVersion = 1;
126 auto published = database_->publish(std::move(record));
127 if (!published.ok()) {
128 const auto& diagnostics = published.diagnostics();
129 result.diagnostics.insert(result.diagnostics.end(), diagnostics.begin(), diagnostics.end());
130 continue;
131 }
132 ++result.indexed;
133 const auto previous = fingerprints_.find(guid);
134 if (previous == fingerprints_.end() || previous->second != fingerprint) ++result.changed;
135 fingerprints_.insert_or_assign(guid, fingerprint);
136 }
137 if (ec) return scanError(EditorStatus::Failed, "editor.asset.scan-failed", ec.message());
138 auto diagnostics = result.diagnostics;
139 return eve::editing::applied<DiskAssetScanResult>(std::move(result), std::move(diagnostics));
140}

144std::string DiskAssetCatalog::fileHash(const std::filesystem::path& path) {
145 std::ifstream input(path, std::ios::binary);
146 std::uint64_t hash = 1469598103934665603ULL;
147 char buffer[8192];
148 while (input) {
149 input.read(buffer, sizeof(buffer));
150 for (std::streamsize index = 0; index < input.gcount(); ++index) {
151 hash ^= static_cast<unsigned char>(buffer[index]);
152 hash *= 1099511628211ULL;
153 }
154 }
155 std::ostringstream text;
156 text << std::hex << std::setfill('0') << std::setw(16) << hash;
157 return text.str();
158}

209std::filesystem::path DiskAtomicDocumentStore::resolve(const std::string& resourceUri) const {
210 if (root_.empty()) return {};
211 std::string relative;
212 if (resourceUri.rfind("content://", 0) == 0)
213 relative = "Content/" + resourceUri.substr(10);
214 else if (resourceUri.rfind("autosave://", 0) == 0)
215 relative = "Saved/Autosave/" + resourceUri.substr(11);
216 else
217 return {};
218 if (relative.empty()) return {};
219
220 std::filesystem::path candidate(relative);
221 if (candidate.is_absolute()) return {};
222 for (const auto& part : candidate)
223 if (part == "..") return {};
224
225 candidate = (root_ / candidate).lexically_normal();
226 const std::filesystem::path scoped = candidate.lexically_relative(root_);
227 if (scoped.empty()) return {};
228 for (const auto& part : scoped)
229 if (part == "..") return {};
230 return candidate;

260Result<StoredDocument> DiskAtomicDocumentStore::compareAndSwap(const std::string& resourceUri,
261 Revision expectedRevision,
262 const std::string& expectedContentHash,
263 const EditorValue& content) {
264 const std::filesystem::path path = resolve(resourceUri);
265 if (path.empty()) return failure(EditorStatus::Rejected, "editor.document.invalid-uri", "Rejected resource URI");
266
267 StoredDocument current;
268 Result<StoredDocument> existing = read(resourceUri);
269 if (existing.ok())
270 current = existing.value();
271 else if (existing.code() != EditorStatus::NotFound)
272 return existing;
273 else
274 current.contentHash.clear();
275 if (current.revision != expectedRevision || current.contentHash != expectedContentHash)
276 return failure(EditorStatus::Conflict, "editor.document.disk-conflict",
277 "Document content or revision changed before save");
278
279 StoredDocument next;
280 next.content = content;
281 next.revision = current.revision + 1;
282 next.contentHash = editorValueContentHash(content);
283 EditorValue::Object envelope;
284 envelope["schemaVersion"] = EditorValue(1);
285 envelope["revision"] = EditorValue(static_cast<std::int64_t>(next.revision));
286 envelope["contentHash"] = EditorValue(next.contentHash);
287 envelope["content"] = content;
288
289 std::error_code ec;
290 std::filesystem::create_directories(path.parent_path(), ec);
291 if (ec) return failure(EditorStatus::Failed, "editor.document.create-directory", ec.message());
292 const std::filesystem::path temporary = path.string() + ".tmp." + std::to_string(++sequence_);
293 {
294 std::ofstream output(temporary, std::ios::binary | std::ios::trunc);
295 if (!output) return failure(EditorStatus::Failed, "editor.document.temp-open", "Cannot open temp file");
296 output << editorValueToJson(EditorValue(std::move(envelope)));
297 output.flush();
298 if (!output.good()) {
299 output.close();
300 std::filesystem::remove(temporary, ec);
301 return failure(EditorStatus::Failed, "editor.document.temp-write", "Cannot flush temp file");
302 }
303 }
304 if ((beforeReplace_ && !beforeReplace_(path)) || !replace(temporary, path)) {
305 std::filesystem::remove(temporary, ec);
306 return failure(EditorStatus::Failed, "editor.document.atomic-replace", "Atomic replacement failed");
307 }
308 return eve::editing::applied<StoredDocument>(std::move(next));
309}

334Result<StoredDocument> AutosaveService::writeDraft(const DocumentSnapshot& snapshot, const EditorValue& content) {
335 if (!store_ || snapshot.id.empty())
336 return eve::editing::failed<StoredDocument>(EditorStatus::Rejected, RuleId("editor.autosave.invalid"),
