28float valueNoise(float x, float y, float z, uint32_t seed) {
29 const int x0 = int(std::floor(x));
30 const int y0 = int(std::floor(y));
31 const int z0 = int(std::floor(z));
32 const float tx = smooth(x - float(x0));
33 const float ty = smooth(y - float(y0));
34 const float tz = smooth(z - float(z0));
35 auto lerp = [](float a, float b, float t) { return a + (b - a) * t; };
36 const float x00 = lerp(cornerNoise(x0, y0, z0, seed), cornerNoise(x0 + 1, y0, z0, seed), tx);
37 const float x10 = lerp(cornerNoise(x0, y0 + 1, z0, seed), cornerNoise(x0 + 1, y0 + 1, z0, seed), tx);
38 const float x01 = lerp(cornerNoise(x0, y0, z0 + 1, seed), cornerNoise(x0 + 1, y0, z0 + 1, seed), tx);
39 const float x11 = lerp(cornerNoise(x0, y0 + 1, z0 + 1, seed), cornerNoise(x0 + 1, y0 + 1, z0 + 1, seed), tx);
40 return lerp(lerp(x00, x10, ty), lerp(x01, x11, ty), tz);

45CaveRoughnessSample sampleCaveWallRoughness(const CaveRoughnessInput& input) {
46 // Frequency bands remain below the default voxel Nyquist limit. Vertical frequencies
47 // are lower because bedding makes carbonate relief statistically anisotropic.
48 const float macro = valueNoise(input.x * 3.7f, input.y * 2.1f, input.z * 3.7f, input.seed ^ 0x21f0aaadu);
49 const float meso = valueNoise(input.x * 8.3f, input.y * 4.8f, input.z * 8.3f, input.seed ^ 0x91e10da5u);
50 const float fine = valueNoise(input.x * 16.7f, input.y * 9.5f, input.z * 16.7f, input.seed ^ 0xd1b54a35u);
51 const float relief = std::clamp((macro * 0.72f + meso * 0.24f + fine * 0.08f) / 1.04f, -1.f, 1.f);
52 return {relief, macro, meso, fine};
53}
