12Result<void> PcgColorPreviewSync::sync(float r, float g, float b, float a) {
13 if (!std::isfinite(r) || !std::isfinite(g) || !std::isfinite(b) || !std::isfinite(a))

16 if (r == 0.f && g == 0.f && b == 0.f) {
17 red_ = green_ = blue_ = 0.5f;

170Result<void> PcgLoadingScreen::endTimeout() {
171 if (!timeoutOpen_)
172 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
173 "beginTimeout must be called before endTimeout", {}, {},
174 "ui.pcgLoadingScreen"));
175 std::string message = "Pcg Loading Screen Timeout, closing down loading screen.\r\n";
176 message +=
177 "If you feel that there is no error but the terrain loading simply needs more time, try to increase the "
178 "Timeout value in the Terrain Loader Manager.\r\n";
179 message += "The following scenes are still not loaded:\r\n";
180 for (const auto& scene : pendingMissingScenes_) {
181 if (!scene.regularReferences.empty()) {
182 message += "\r\n " + scene.terrainName + "\r\n Referencing Objects:\r\n";
183 for (const auto& name : scene.regularReferences) message += name + ", ";
184 }
185 if (!scene.impostorReferences.empty()) {
186 message += "\r\n " + scene.impostorName + "\r\n Referencing Objects:\r\n";
187 for (const auto& name : scene.impostorReferences) message += name + ", ";
188 }
189 message += "\r\n";
190 }
191 timeoutMessage_ = std::move(message);
192 pendingMissingScenes_.clear();
193 timeoutOpen_ = false;
194 onLoadProgressEnded();
195 return Result<void>::success();
196}

198Result<void> PcgLoadingScreen::tick(float deltaTime) {
199 if (!std::isfinite(deltaTime) || deltaTime < 0.f)
200 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
201 "frame delta time must be finite and non-negative", {}, {},
202 "ui.pcgLoadingScreen"));
203 if (fading_) {
204 backgroundAlpha_ -= deltaTime * fadeOutSpeed_;
205 if (backgroundAlpha_ <= 0.f) {
206 backgroundAlpha_ = 0.f;
207 canvasVisible_ = false;
208 fading_ = false;
209 }
210 }
211 return Result<void>::success();
