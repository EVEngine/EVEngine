66 double number = 0.0;
67 if (const auto* integer = value.getIf<std::int64_t>())
68 number = static_cast<double>(*integer);

213eve::Result<std::vector<ClimbingEvent>> readEvents(const eve::Value& value, std::uint64_t nextExecutionId) {
214 const auto* array = value.getIf<eve::Value::Array>();
215 if (!array || array->size() > ClimbingRuntime::PendingEventCapacity)
216 return failure<std::vector<ClimbingEvent>>(eve::DiagnosticCode::ParseError,
217 "pending climbing events must be a bounded array",
218 "pendingEvents");
219 std::vector<ClimbingEvent> events;
220 events.reserve(array->size());
221 eve::SimulationTick previousTick = eve::SimulationTick::zero();
222 bool first = true;
223 for (std::size_t index = 0; index < array->size(); ++index) {
224 const auto* object = (*array)[index].getIf<eve::Value::Object>();
225 ClimbingEvent event;
226 std::string kind;
227 std::uint64_t tick = 0;
228 std::uint64_t executionId = 0;
229 if (!object || !readString(*object, "kind", kind) || !readEventKind(kind, event.kind) ||
230 !readString(*object, "actionId", event.actionId) || event.actionId.empty() ||
231 !readUint64String(*object, "tick", tick) || !readUint64String(*object, "executionId", executionId) ||
232 executionId == 0 || executionId >= nextExecutionId)
233 return failure<std::vector<ClimbingEvent>>(eve::DiagnosticCode::ParseError,
234 "pending climbing event is missing or inconsistent",
235 "pendingEvents." + std::to_string(index));
236 if (const eve::Value* metadataValue = field(*object, "metadata")) {
237 const auto* metadata = metadataValue->getIf<eve::Value::Array>();
238 if (!metadata)
239 return failure<std::vector<ClimbingEvent>>(eve::DiagnosticCode::ParseError,
240 "climbing event metadata must be an array",
241 "pendingEvents." + std::to_string(index) + ".metadata");
242 event.metadata.reserve(metadata->size());
243 for (const eve::Value& item : *metadata) {
244 const auto* text = item.getIf<std::string>();
245 if (!text || text->empty())
246 return failure<std::vector<ClimbingEvent>>(
247 eve::DiagnosticCode::ParseError, "climbing event metadata must contain non-empty strings",
248 "pendingEvents." + std::to_string(index) + ".metadata");
249 event.metadata.push_back(*text);
250 }
251 }
252 event.tick = eve::SimulationTick(tick);
253 event.executionId = ClimbingExecutionId(executionId);
254 if (!first && event.tick < previousTick)
255 return failure<std::vector<ClimbingEvent>>(eve::DiagnosticCode::InvariantViolation,
256 "pending climbing events must be ordered by tick",
257 "pendingEvents." + std::to_string(index) + ".tick");
258 first = false;
259 previousTick = event.tick;
260 events.push_back(std::move(event));
261 }
262 return eve::Result<std::vector<ClimbingEvent>>::success(std::move(events));
263}

317eve::Result<ClimbingCandidate> readCandidate(const eve::Value& value, std::int64_t snapshotVersion) {
318 const auto* object = value.getIf<eve::Value::Object>();
319 if (!object)
320 return failure<ClimbingCandidate>(eve::DiagnosticCode::ParseError, "runtime candidate must be an object",
321 "execution.candidate");
322 ClimbingCandidate result;
323 std::uint64_t world = 0, body = 0, shape = 0, definitionGeneration = 1;
324 std::int64_t bodyId = 0, shapeId = 0, ignoredBodyId = 0, score = 0, kind = 0, support = 0;
325 std::int64_t supportShapeTag = 0, supportMaterialId = 0, probeRecipe = 0;
326 const eve::Value* heightValue = field(*object, "obstacleHeight");
327 const eve::Value* depthValue = field(*object, "obstacleDepth");
328 const eve::Value* gapValue = field(*object, "gapDistance");
329 const eve::Value* clearanceValue = field(*object, "clearanceHeight");
330 const eve::Value* slopeValue = field(*object, "slopeRadians");
331 const eve::Value* curvatureValue = field(*object, "curvature");
332 if (!readString(*object, "actionId", result.actionId) ||
333 (field(*object, "definitionGeneration") &&
334 !readUint64String(*object, "definitionGeneration", definitionGeneration)) ||
335 definitionGeneration == 0 || !readUint64String(*object, "world", world) ||
336 !readUint64String(*object, "obstacleBody", body) || !readUint64String(*object, "obstacleShape", shape) ||
337 !readInt64(*object, "obstacleBodyId", bodyId) || bodyId < std::numeric_limits<int>::min() ||
338 bodyId > std::numeric_limits<int>::max() || !readInt64(*object, "obstacleShapeId", shapeId) ||
339 shapeId < std::numeric_limits<int>::min() || shapeId > std::numeric_limits<int>::max() ||
340 !readInt64(*object, "ignoredBodyId", ignoredBodyId) || ignoredBodyId < std::numeric_limits<int>::min() ||
341 ignoredBodyId > std::numeric_limits<int>::max() || !readVec(*object, "frontPoint", result.frontPoint) ||
342 !readVec(*object, "topPoint", result.topPoint) || !readVec(*object, "landingFeet", result.landingFeet) ||
343 !readVec(*object, "surfaceNormal", result.surfaceNormal) ||
344 !readVec(*object, "surfaceTangent", result.surfaceTangent) ||
345 !readVec(*object, "leftHandAnchor", result.leftHandAnchor) ||
346 !readVec(*object, "rightHandAnchor", result.rightHandAnchor) ||
347 !readVec(*object, "bodyLocalTop", result.bodyLocalTop) ||
348 !readVec(*object, "bodyLocalLanding", result.bodyLocalLanding) || !heightValue ||
349 !numericFloat(*heightValue, result.obstacleHeight) || !readInt64(*object, "score", score) ||
350 !readInt64(*object, "kind", kind) || kind < 0 ||
351 kind > static_cast<std::int64_t>(ClimbingActionKind::BarSwing) ||
352 !readInt64(*object, "support", support) ||
353 support < 0 || support > static_cast<std::int64_t>(HangSupport::Free))
354 return failure<ClimbingCandidate>(eve::DiagnosticCode::ParseError,
355 "runtime candidate has missing or invalid fields", "execution.candidate");
356 if ((snapshotVersion >= 4 && (!depthValue || !gapValue || !clearanceValue || !slopeValue || !curvatureValue ||
357 !field(*object, "supportShapeTag") || !field(*object, "supportMaterialId") ||
358 !field(*object, "probeRecipe"))) ||
359 (depthValue && !numericFloat(*depthValue, result.obstacleDepth)) ||
360 (gapValue && !numericFloat(*gapValue, result.gapDistance)) ||
361 (clearanceValue && !numericFloat(*clearanceValue, result.clearanceHeight)) ||
362 (slopeValue && !numericFloat(*slopeValue, result.slopeRadians)) ||
363 (curvatureValue && !numericFloat(*curvatureValue, result.curvature)) ||
364 (field(*object, "supportShapeTag") && !readInt64(*object, "supportShapeTag", supportShapeTag)) ||
365 supportShapeTag < std::numeric_limits<int>::min() || supportShapeTag > std::numeric_limits<int>::max() ||
366 (field(*object, "supportMaterialId") && !readInt64(*object, "supportMaterialId", supportMaterialId)) ||
367 supportMaterialId < std::numeric_limits<int>::min() ||
368 supportMaterialId > std::numeric_limits<int>::max() ||
369 (field(*object, "probeRecipe") && !readInt64(*object, "probeRecipe", probeRecipe)) || probeRecipe < 0 ||
370 probeRecipe > static_cast<std::int64_t>(ClimbingProbeRecipe::AnchorGraph))
371 return failure<ClimbingCandidate>(eve::DiagnosticCode::ParseError,
372 "runtime candidate metrics are missing or invalid", "execution.candidate");
373 result.definitionGeneration = definitionGeneration;
374 result.world = physics::PhysicsWorldHandle::fromPacked(world);
375 result.obstacleBody = physics::PhysicsBodyHandle::fromPacked(body);
376 result.obstacleShape = physics::PhysicsShapeHandle::fromPacked(shape);
377 result.obstacleBodyId = static_cast<int>(bodyId);
378 result.obstacleShapeId = static_cast<int>(shapeId);
379 result.ignoredBodyId = static_cast<int>(ignoredBodyId);
380 result.score = score;
381 result.kind = static_cast<ClimbingActionKind>(kind);
382 result.supportShapeTag = static_cast<int>(supportShapeTag);
383 result.supportMaterialId = static_cast<int>(supportMaterialId);
384 result.probeRecipe = static_cast<ClimbingProbeRecipe>(probeRecipe);
385 result.support = static_cast<HangSupport>(support);
386 return eve::Result<ClimbingCandidate>::success(std::move(result));
387}

396eve::Result<eve::Value> ClimbingRuntime::snapshot() const {
397 auto profile = encodeClimbingProfileDefinition(profile_);
398 if (!profile) return eve::Result<eve::Value>::failure(profile.status());
399 eve::Value::Object root{{"schemaId", eve::Value(std::string(SnapshotSchemaId))},
400 {"schemaVersion", eve::Value(SnapshotSchemaVersion)},
401 {"phase", eve::Value(std::string(phaseName(phase_)))},
402 {"terminalCode", eve::Value(terminalCode_)},
403 {"definitionGeneration", eve::Value(std::to_string(definitionGeneration_))},
404 {"profile", std::move(profile).takeValue()},
405 {"previousActionId", eve::Value(previousActionId_)},
406 {"nextExecutionId", eve::Value(std::to_string(nextExecutionId_))},
407 {"pendingEvents", eventArrayValue(pendingEvents_)}};
408 if (!execution_) {
409 root.emplace("execution", eve::Value());
410 return eve::Result<eve::Value>::success(eve::Value(std::move(root)));
411 }
412 auto action = encodeClimbingActionDefinition(execution_->action);
413 if (!action) return eve::Result<eve::Value>::failure(action.status());
414 eve::Value::Object state{
415 {"executionId", std::to_string(execution_->executionId.value())},
416 {"definitionGeneration", std::to_string(execution_->definitionGeneration)},
417 {"action", std::move(action).takeValue()},
418 {"candidate", candidateValue(execution_->candidate)},
419 {"startFeet", vecValue(execution_->startFeet)},
420 {"currentFeet", vecValue(execution_->currentFeet)},
421 {"lastPlannedFeet", vecValue(execution_->lastPlannedFeet)},
422 {"elapsedNs", execution_->elapsed.nanoseconds()},
423 {"durationNs", execution_->duration.nanoseconds()},
424 {"lastTick", std::to_string(execution_->lastTick.value())},
425 {"velocity", vecValue(execution_->velocity)},
426 {"accumulatedResidual", vecValue(execution_->accumulatedResidual)},
427 {"horizontalWarpUsed", execution_->horizontalWarpUsed},
428 {"verticalWarpUsed", execution_->verticalWarpUsed},
429 {"facingWarpUsed", execution_->facingWarpUsed},
430 {"leftContactEmitted", execution_->leftContactEmitted},
431 {"rightContactEmitted", execution_->rightContactEmitted},
432 {"landContactReleased", execution_->landContactReleased},
433 {"compactCollisionActive", execution_->compactCollisionActive},
434 {"branchWindowOpen", execution_->branchWindowOpen},
435 {"anchor", anchorValue(execution_->anchorGraph, execution_->anchorNode,
436 execution_->anchorReservation)},
437 };
438 root.emplace("execution", eve::Value(std::move(state)));
439 return eve::Result<eve::Value>::success(eve::Value(std::move(root)));
440}

442eve::Result<void> ClimbingRuntime::restore(const eve::Value& value, physics::World3D& world) {
443 const auto* root = value.getIf<eve::Value::Object>();
444 if (!root) return failure<void>(eve::DiagnosticCode::ParseError, "runtime snapshot must be an object");
445 std::string schemaId;
446 std::int64_t version = -1;
447 std::string phaseText;
448 std::string terminalCode;
449 std::string candidatePreviousActionId;
450 std::uint64_t candidateNextExecutionId = 1;
451 std::uint64_t candidateDefinitionGeneration = definitionGeneration_;
452 if (!readString(*root, "schemaId", schemaId) || schemaId != SnapshotSchemaId ||
453 !readInt64(*root, "schemaVersion", version) || version < 0 || version > SnapshotSchemaVersion ||
454 !readString(*root, "phase", phaseText) ||
455 (version >= 1 && !readString(*root, "terminalCode", terminalCode))) {
456 if (version > SnapshotSchemaVersion)
457 return failure<void>(eve::DiagnosticCode::UnknownVersion, "climbing.restore.version_unsupported",
458 "schemaVersion");
459 return failure<void>(eve::DiagnosticCode::ParseError, "invalid climbing runtime snapshot envelope");
460 }
461 if (version >= 4 && !readString(*root, "previousActionId", candidatePreviousActionId))
462 return failure<void>(eve::DiagnosticCode::ParseError, "invalid previous climbing action id",
463 "previousActionId");
464 if (field(*root, "nextExecutionId") && !readUint64String(*root, "nextExecutionId", candidateNextExecutionId))
465 return failure<void>(eve::DiagnosticCode::ParseError, "invalid next climbing execution id", "nextExecutionId");
466 if (field(*root, "definitionGeneration") &&
467 (!readUint64String(*root, "definitionGeneration", candidateDefinitionGeneration) ||
468 candidateDefinitionGeneration == 0))
469 return failure<void>(eve::DiagnosticCode::ParseError, "invalid climbing definition generation",
470 "definitionGeneration");
471 ClimbingProfile candidateProfile = profile_;
472 if (const eve::Value* profileValue = field(*root, "profile")) {
473 auto decodedProfile = decodeClimbingProfileDefinition(*profileValue);
474 if (!decodedProfile) return eve::Result<void>::failure(decodedProfile.status());
475 candidateProfile = std::move(decodedProfile).takeValue();
476 }
477 ClimbingPhase candidatePhase;
478 if (!readPhase(phaseText, candidatePhase))
