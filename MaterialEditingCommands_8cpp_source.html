32editing::Result<void> registerEditingCommands(editing::IEditingCommandRegistry& registry) {
33 editing::EditingCommandDescriptor descriptor;
34 descriptor.id = editing::CommandId("material.property.set.v1");
35 descriptor.ownerModule = "material_editing";
36 descriptor.displayName = "Set material property";
37 descriptor.category = "Material";
38 descriptor.automationAllowed = true;
39 return registry.registerPlannedCommand(
40 std::move(descriptor), [](editing::IEditableTarget& target, const editing::CommandRequest& request) {
41 auto* properties = static_cast<editing::IPropertyProvider*>(
42 target.queryCapability(editing::IPropertyProvider::editingCapabilityId()));
43 const editing::Value* pathValue = field(request.payload, "path");
44 const auto* path = pathValue ? pathValue->getIf<std::string>() : nullptr;
45 const editing::Value* value = field(request.payload, "value");
46 if (!properties || !path || !value)
47 return eve::editing::failed<editing::CommandPlan>(
48 editing::Status::Rejected, editing::RuleId("material.editing.property-payload"),
49 "Material property requires a property target, path and value");
50 auto operation = properties->makeSet(selectionFor(target), editing::PropertyPath(*path), *value,
51 editing::PropertySetMode::Absolute);
52 if (!operation.ok())
53 return eve::editing::failed<editing::CommandPlan>(
54 operation.code(), editing::RuleId("material.editing.property-operation"),
55 "Material target rejected the property value");
56 editing::CommandPlan plan;
57 plan.operations.push_back(std::move(operation.value()));
58 plan.summary = editing::Value::Object{{"path", *path}};
59 return eve::editing::applied<editing::CommandPlan>(std::move(plan));
60 });
61}

69GraphConnectionDecision MaterialGraphDomain::canConnect(const GraphPinRecord& from, const GraphPinRecord& to) const {
70 GraphConnectionDecision decision;
71 decision.allowed = from.direction == GraphPinDirection::Output && to.direction == GraphPinDirection::Input &&
72 from.type == to.type && from.node != to.node;
73 if (!decision.allowed)
74 decision.diagnostics.push_back(eve::editing::ruleDiagnostic(
75 eve::DiagnosticCode::InvalidArgument, RuleId("editor.material.pin-type-mismatch"),
76 DiagnosticSeverity::Error, "Material pins require matching types, output-to-input and distinct nodes"));
77 return decision;
78}

80MaterialCompileResult MaterialGraphDomain::compile(const GraphDocumentData& graph) const {
81 MaterialCompileResult result;
82 result.documentRevision = graph.revision;
83 if (graph.domain != domain()) {
84 result.status = EditorStatus::Rejected;
85 result.diagnostics.push_back(
86 eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument, RuleId("editor.material.wrong-domain"),
87 DiagnosticSeverity::Error, "Graph is not a material domain document"));
88 return result;
89 }
90 const std::size_t outputs =
91 static_cast<std::size_t>(std::count_if(graph.nodes.begin(), graph.nodes.end(), [](const GraphNodeRecord& node) {
92 return node.type == "material.output";
93 }));
94 const bool errorNode = std::any_of(graph.nodes.begin(), graph.nodes.end(),
95 [](const GraphNodeRecord& node) { return node.type == "material.error"; });
96 if (outputs != 1 || errorNode) {
97 result.status = EditorStatus::Failed;
98 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
99 eve::DiagnosticCode::Failed, RuleId("editor.material.compile-failed"), DiagnosticSeverity::Error,
100 outputs != 1 ? "Material graph requires exactly one output node"
101 : "Material graph contains an invalid node"));
102 return result;
103 }
104 result.status = EditorStatus::Applied;
105 result.shaderArtifact = "material-shader:r" + std::to_string(graph.revision) + ":n" +
106 std::to_string(graph.nodes.size()) + ":e" + std::to_string(graph.edges.size());
107 return result;
108}

120Result<TaskId> MaterialEditorService::compileAsync(const DocumentId& document, GraphDocumentData graph,
121 const MaterialGraphDomain& domain, TaskService& tasks) {
122 if (document.empty())
123 return eve::editing::failed<TaskId>(EditorStatus::Rejected, RuleId("editor.material.missing-document"),
124 "Material document id is required");
125 const MaterialGraphDomain domainCopy = domain;
126 auto queued = tasks.submit(
127 "Compile material " + document.value(), [graph = std::move(graph), domainCopy](const TaskContext& context) {
128 TaskOutcome outcome;
129 if (context.isCancellationRequested()) {
130 outcome.status = EditorStatus::Cancelled;
131 return outcome;
132 }
133 context.reportProgress(0.25F);
134 const MaterialCompileResult compiled = domainCopy.compile(graph);
135 EditorValue::Object value;
136 value["status"] = EditorValue(static_cast<std::int64_t>(compiled.status));
137 value["revision"] = EditorValue(static_cast<std::int64_t>(compiled.documentRevision));
138 value["artifact"] = EditorValue(compiled.shaderArtifact);
139 outcome.status = compiled.status;
140 outcome.output = EditorValue(std::move(value));
141 outcome.diagnostics = compiled.diagnostics;
142 context.reportProgress(1.0F);
143 return outcome;
144 });
145 if (!queued.ok()) return queued;
146 tasks_.emplace(queued.value(), Task{document, {}, &tasks});
147 return queued;
148}

158Result<MaterialCompileResult> MaterialEditorService::result(const TaskId& task) const {
159 auto found = tasks_.find(task);
160 if (found == tasks_.end())
161 return eve::editing::failed<MaterialCompileResult>(
162 EditorStatus::NotFound, RuleId("editor.material.task-not-found"), "Material compile task does not exist");
163 if (found->second.service) {
164 const auto snapshot = found->second.service->snapshot(task);
165 if (!snapshot.ok())
166 return eve::editing::failed<MaterialCompileResult>(EditorStatus::NotFound,
167 RuleId("editor.material.task-not-found"),
168 "Material compile task does not exist");
169 if (snapshot.value().state == TaskState::Queued || snapshot.value().state == TaskState::Running)
170 return Result<MaterialCompileResult>::success(MaterialCompileResult{},
171 eve::Status::success(EditorStatus::Pending));
172 MaterialCompileResult compiled;
173 compiled.diagnostics = snapshot.value().diagnostics;
174 if (snapshot.value().state == TaskState::Cancelled) {
175 compiled.status = EditorStatus::Cancelled;
176 return eve::editing::applied<MaterialCompileResult>(std::move(compiled));
177 }
178 const auto* object = snapshot.value().output.getIf<EditorValue::Object>();
179 if (!object)
180 return eve::editing::failed<MaterialCompileResult>(EditorStatus::Failed,
181 RuleId("editor.material.invalid-task-output"),
182 "Material worker returned an invalid result");
183 const auto status = object->find("status");
184 const auto revision = object->find("revision");
185 const auto artifact = object->find("artifact");
186 const auto* statusValue = status == object->end() ? nullptr : status->second.getIf<std::int64_t>();
187 const auto* revisionValue = revision == object->end() ? nullptr : revision->second.getIf<std::int64_t>();
188 const auto* artifactValue = artifact == object->end() ? nullptr : artifact->second.getIf<std::string>();
189 if (!statusValue || !revisionValue || !artifactValue)
190 return eve::editing::failed<MaterialCompileResult>(EditorStatus::Failed,
191 RuleId("editor.material.invalid-task-output"),
192 "Material worker result fields are invalid");
193 compiled.status = static_cast<EditorStatus>(*statusValue);
194 compiled.documentRevision = static_cast<Revision>(*revisionValue);
195 compiled.shaderArtifact = *artifactValue;
196 return eve::editing::applied<MaterialCompileResult>(std::move(compiled));
197 }
198 return eve::editing::applied<MaterialCompileResult>(found->second.result);
199}

201Result<void> MaterialEditorService::publishPreview(const DocumentId& document, Revision currentRevision,
202 const TaskId& task) {
203 auto found = tasks_.find(task);
204 if (found == tasks_.end() || found->second.document != document)
205 return eve::editing::failed<void>(EditorStatus::NotFound, RuleId("editor.material.task-not-found"),
206 "Material compile task does not belong to this document");
207 const auto compileResult = result(task);
208 if (compileResult.code() == EditorStatus::Pending)
209 return eve::editing::failed<void>(EditorStatus::Conflict, RuleId("editor.material.compile-pending"),
210 "Material compile task has not completed");
211 if (!compileResult.ok())
212 return eve::editing::failed<void>(EditorStatus::Failed, RuleId("editor.material.compile-result-unavailable"),
213 "Material compile result is unavailable");
214 const MaterialCompileResult& compiled = compileResult.value();
215 if (compiled.status != EditorStatus::Applied)
216 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.material.compile-not-successful"),
217 "Failed material output cannot replace the current preview");
218 if (compiled.documentRevision != currentRevision)
219 return eve::editing::failed<void>(EditorStatus::Conflict, RuleId("editor.material.stale-compile"),
220 "Material changed after this compile task started");
221 previews_.insert_or_assign(document, compiled.shaderArtifact);
222 return eve::editing::applied<void>();
223}
