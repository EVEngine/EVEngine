129eve::Result<std::int64_t> AttributeSetResourceAccount::balanceOf(const ResourceId& resource) const {
130 const double value = attributes_.getBase(resource.value(), 0.0);
131 if (!std::isfinite(value) || value < 0.0 || std::floor(value) != value || value > kMaxExactlyRepresentableInteger) {

135 return eve::Result<std::int64_t>::success(static_cast<std::int64_t>(value));

138eve::Result<std::int64_t> AttributeSetResourceAccount::activeReservationsFor(const ResourceId& resource) const {
139 std::int64_t total = 0;
140 for (const auto& [id, record] : reservations_) {
141 (void)id;
142 if (record.state != ReservationState::Active) continue;
143 for (const auto& item : record.cost.items()) {
144 if (item.resource != resource) continue;
145 if (item.amount.value() > std::numeric_limits<std::int64_t>::max() - total)
146 return eve::Result<std::int64_t>::failure(
147 invariantFailure("active resource reservations overflowed", resource.value()));
148 total += item.amount.value();
149 }
150 }
151 return eve::Result<std::int64_t>::success(total);

154eve::Result<Affordability> AttributeSetResourceAccount::canAfford(const CostSpec& cost) const {
155 try {
156 if (accountNonce_.isZero()) return eve::Result<Affordability>::failure(accountIdentityUnavailableStatus());
157 auto valid = validateCost(cost);
158 if (!valid) return eve::Result<Affordability>::failure(valid.status());
159
160 Affordability result;
161 result.affordable = true;
162 for (const auto& item : cost.items()) {
163 auto balance = balanceOf(item.resource);
164 if (!balance) return eve::Result<Affordability>::failure(balance.status());
165 auto reserved = activeReservationsFor(item.resource);
166 if (!reserved) return eve::Result<Affordability>::failure(reserved.status());
167
168 const std::int64_t available = balance.value() >= reserved.value() ? balance.value() - reserved.value() : 0;
169 if (available < item.amount.value()) {
170 result.affordable = false;
171 result.shortfalls.push_back({item.resource, item.amount, Amount(available)});
172 }
173 }
174 return eve::Result<Affordability>::success(std::move(result));
175 } catch (const std::exception& exception) {
176 return exceptionFailure<Affordability>("canAfford", exception);
177 } catch (...) {
178 return unknownExceptionFailure<Affordability>("canAfford");
179 }
180}

182eve::Result<Reservation> AttributeSetResourceAccount::reserve(const CostSpec& cost) {
183 try {
184 auto affordability = canAfford(cost);
185 if (!affordability) return eve::Result<Reservation>::failure(affordability.status());
186 const auto& checked = affordability.value();
187 if (!checked.affordable) return eve::Result<Reservation>::failure(insufficientStatus(checked));
188
189 if (nextReservation_.value() == std::numeric_limits<std::uint64_t>::max())
190 return eve::Result<Reservation>::failure(eve::Status::failure(
191 eve::StatusCode::Failed, invariantFailure("resource reservation id exhausted", "reservation")));
192
193 const ReservationId id = nextReservation_;
194 const auto [it, inserted] = reservations_.emplace(id, ReservationRecord{cost, ReservationState::Active});
195 if (!inserted)
196 return eve::Result<Reservation>::failure(eve::Status::failure(
197 eve::StatusCode::Conflict, conflict("resource reservation id was already used", "reservation")));
198 nextReservation_ = ReservationId(id.value() + 1);
199 return eve::Result<Reservation>::success(Reservation{accountNonce_, id, it->second.cost},
200 eve::Status::success(eve::StatusCode::Applied));
201 } catch (const std::exception& exception) {
202 return exceptionFailure<Reservation>("reserve", exception);
203 } catch (...) {
204 return unknownExceptionFailure<Reservation>("reserve");
205 }

208eve::Result<void> AttributeSetResourceAccount::applyDelta(const CostSpec& cost, bool debit) {
209 try {
210 AttributeSet candidate = attributes_;
211 for (const auto& item : cost.items()) {
212 auto balance = balanceOf(item.resource);
213 if (!balance) return eve::Result<void>::failure(balance.status());
214 const auto amount = item.amount.value();
215 if (debit) {
216 if (balance.value() < amount)
217 return eve::Result<void>::failure(eve::Status::failure(
218 eve::StatusCode::Rejected,
219 eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
220 "resource debit would make balance negative", item.resource.value())));
221 candidate.setBase(item.resource.value(), static_cast<double>(balance.value() - amount));
222 } else {
223 if (amount > std::numeric_limits<std::int64_t>::max() - balance.value())
224 return eve::Result<void>::failure(
225 eve::Status::failure(eve::StatusCode::Failed,
226 invariantFailure("resource credit overflowed", item.resource.value())));
227 candidate.setBase(item.resource.value(), static_cast<double>(balance.value() + amount));
228 }
229 }
230 attributes_ = std::move(candidate);
231 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
232 } catch (const std::exception& exception) {
233 return exceptionFailure<void>("applyDelta", exception);
234 } catch (...) {
235 return unknownExceptionFailure<void>("applyDelta");
236 }

311eve::Result<Receipt> AttributeSetResourceAccount::commit(const Reservation& reservation) {
312 try {
313 if (accountNonce_.isZero()) return eve::Result<Receipt>::failure(accountIdentityUnavailableStatus());
314 if (!reservation.isValid())
315 return eve::Result<Receipt>::failure(invalidArgument("resource reservation is invalid", "reservation"));
316 if (reservation.account != accountNonce_) return eve::Result<Receipt>::failure(foreignReservationStatus());
317 const auto it = reservations_.find(reservation.id);
318 if (it == reservations_.end()) return eve::Result<Receipt>::failure(unknownReservationStatus());
319 if (it->second.cost != reservation.cost) return eve::Result<Receipt>::failure(mismatchedReservationStatus());
320 if (it->second.state != ReservationState::Active)
321 return eve::Result<Receipt>::failure(duplicateReservationStatus());
322
323 auto covered = activeReservationsAreCovered();
324 if (!covered) return eve::Result<Receipt>::failure(covered.status());
325 if (nextReceipt_.value() == std::numeric_limits<std::uint64_t>::max())
326 return eve::Result<Receipt>::failure(receiptIdExhaustedStatus());
327
328 Receipt receipt{accountNonce_, nextReceipt_, reservation.id, ReceiptOperation::Debit, it->second.cost};
329 auto applied = applyDelta(it->second.cost, true);
330 if (!applied) return eve::Result<Receipt>::failure(applied.status());
331 it->second.state = ReservationState::Committed;
332 nextReceipt_ = eve::resource::ReceiptId(nextReceipt_.value() + 1);
333 return eve::Result<Receipt>::success(std::move(receipt), eve::Status::success(eve::StatusCode::Applied));
334 } catch (const std::exception& exception) {
335 return exceptionFailure<Receipt>("commit", exception);
336 } catch (...) {
337 return unknownExceptionFailure<Receipt>("commit");
338 }
