40EditorResult<UiThemeAsset> parseAsset(const EditorValue& value) {
41 const EditorValue* idField = field(value, "id");
42 const EditorValue* nameField = field(value, "name");
43 const EditorValue* presetField = field(value, "basePreset");
44 const EditorValue* tokensField = field(value, "tokens");
45 const auto* id = idField ? idField->getIf<std::string>() : nullptr;
46 const auto* name = nameField ? nameField->getIf<std::string>() : nullptr;
47 const auto* preset = presetField ? presetField->getIf<std::string>() : nullptr;
48 if (!id || id->empty() || !name || name->empty() || !preset || !tokensField)
49 return fail<UiThemeAsset>(EditorStatus::Rejected, "editor.ui-theme.asset",
50 "Theme asset id, name, basePreset and tokens are required");
51 auto base = parsePreset(*preset);
52 if (!base.ok()) return EditorResult<UiThemeAsset>::failure(base.status());
53 auto tokens = parseThemeTokens(*tokensField);
54 if (!tokens.ok()) return EditorResult<UiThemeAsset>::failure(tokens.status());
55 UiThemeAsset asset;
56 asset.id = ObjectId(*id);
57 asset.name = *name;
58 asset.basePreset = base.value();
59 asset.tokens = std::move(tokens).value();
60 return eve::editing::applied<UiThemeAsset>(std::move(asset));
61}

63EditorResult<void> parseCatalog(const EditorValue& content, std::vector<UiThemeAsset>& themes, ObjectId& activeId) {
64 const EditorValue* activeField = field(content, "activeId");
65 const EditorValue* themesField = field(content, "themes");
66 const auto* active = activeField ? activeField->getIf<std::string>() : nullptr;
67 const auto* list = themesField ? themesField->getIf<EditorValue::Array>() : nullptr;
68 if (!active || active->empty() || !list)
69 return fail<void>(EditorStatus::Rejected, "editor.ui-theme.catalog",
70 "Theme catalog requires activeId and a themes array");
71 std::vector<UiThemeAsset> parsed;
72 parsed.reserve(list->size());
73 for (const EditorValue& entry : *list) {
74 auto asset = parseAsset(entry);
75 if (!asset.ok()) return EditorResult<void>::failure(asset.status());
76 parsed.push_back(std::move(asset).value());
77 }
78 themes = std::move(parsed);
79 activeId = ObjectId(*active);
80 return eve::editing::applied<void>();
81}

161std::vector<EditorDiagnostic> UiThemeCatalogTarget::validate() const {
162 std::vector<EditorDiagnostic> diagnostics;
163 if (themes_.empty())
164 diagnostics.push_back(diagnostic("editor.ui-theme.empty", "Theme catalog must contain at least one asset"));
165 if (themes_.size() > 32)
166 diagnostics.push_back(diagnostic("editor.ui-theme.budget", "Theme catalog exceeds 32 named assets"));
167 std::set<std::string> ids;
168 std::set<std::string> names;
169 bool activeFound = false;
170 for (const UiThemeAsset& asset : themes_) {
171 if (asset.id.empty() || !ids.insert(asset.id.value()).second)
172 diagnostics.push_back(diagnostic("editor.ui-theme.id", "Theme asset id is empty or duplicated"));
173 if (asset.name.empty() || !names.insert(asset.name).second)
174 diagnostics.push_back(diagnostic("editor.ui-theme.name", "Theme asset name is empty or duplicated"));
175 if (asset.id == activeId_) activeFound = true;
176 auto tokenDiagnostics = validateThemeTokens(asset.tokens);
177 diagnostics.insert(diagnostics.end(), tokenDiagnostics.begin(), tokenDiagnostics.end());
178 }
179 if (!activeFound)
180 diagnostics.push_back(diagnostic("editor.ui-theme.active", "Active theme id is missing from the catalog"));
181 return diagnostics;
182}

184EditorResult<void> UiThemeCatalogTarget::applyDomainOperation(const DomainOperation& operation) {
185 if (operation.target != TargetId(id_))
186 return fail<void>(EditorStatus::Rejected, "editor.ui-theme.target", "Theme operation targets another catalog");
187 if (operation.type != "ui.theme.catalog.replace.v1")
188 return fail<void>(EditorStatus::Unsupported, "editor.ui-theme.operation",
189 "Unsupported theme operation: " + operation.type);
190 auto candidate = *this;
191 auto parsed = parseCatalog(operation.payload, candidate.themes_, candidate.activeId_);
192 if (!parsed.ok()) return parsed;
193 if (hasError(candidate.validate()))
194 return fail<void>(EditorStatus::Rejected, "editor.ui-theme.invalid",
195 "Theme catalog replacement is invalid");
196 themes_ = std::move(candidate.themes_);
197 activeId_ = candidate.activeId_;
198 ++revision_;
199 dirty_.include(0, 0);
200 return eve::editing::applied<void>();
201}
