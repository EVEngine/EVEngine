25bool looksLikeJsonValue(const std::string& json) {
26 const auto first = std::find_if_not(json.begin(), json.end(), [](unsigned char c) { return std::isspace(c); });
27 if (first == json.end()) return false;
28 const auto last = std::find_if_not(json.rbegin(), json.rend(), [](unsigned char c) { return std::isspace(c); });
29 const char open = *first;
30 const char close = *last;
31 return (open == '{' && close == '}') || (open == '[' && close == ']');
32}

44EditorResult<void> DefinitionDocument::applyDomainOperation(const DomainOperation& operation) {
45 if (operation.target != TargetId(targetId_) || operation.type != "definition.field.set.v1")
46 return definitionError<void>(EditorStatus::Rejected, "editor.definition.operation-mismatch",
47 "Definition field operation targets another document or type");
48 const auto* pathValue = field(operation.payload, "field");
49 const auto* assigned = field(operation.payload, "value");
50 const auto* presentValue = field(operation.payload, "present");
51 const auto* path = pathValue ? pathValue->getIf<std::string>() : nullptr;
52 const auto* present = presentValue ? presentValue->getIf<bool>() : nullptr;
53 auto parsed = editorValueFromJson(json_);
54 auto* object = parsed.ok() ? parsed.value().getIf<EditorValue::Object>() : nullptr;
55 if (!path || path->empty() || !assigned || !present || !object)
56 return definitionError<void>(EditorStatus::Rejected, "editor.definition.invalid-field-operation",
57 "Definition field operation requires an object payload and field name");
58 if (*present) (*object)[*path] = *assigned;
59 else object->erase(*path);
60 json_ = editorValueToJson(parsed.value());
61 ++revision_;
62 dirty_.include(0, 0);
63 return eve::editing::applied<void>();
64}

98EditorResult<DomainOperation> DefinitionDocument::makeSetField(const std::string& fieldName,
99 const EditorValue& value) const {
100 if (fieldName.empty())
101 return definitionError<DomainOperation>(EditorStatus::Rejected, "editor.definition.empty-field",
102 "Definition field name must not be empty");
103 auto parsed = editorValueFromJson(json_);
104 const auto* object = parsed.ok() ? parsed.value().getIf<EditorValue::Object>() : nullptr;
105 if (!object)
106 return definitionError<DomainOperation>(EditorStatus::Unsupported, "editor.definition.form-object-required",
107 "Schema form editing requires an object definition payload");
108 const auto current = object->find(fieldName);
109 const EditorValue before = current == object->end() ? EditorValue{} : current->second;
110 DomainOperation operation;
111 operation.type = "definition.field.set.v1";
112 operation.inverseType = operation.type;
113 operation.target = TargetId(targetId_);
114 operation.payload = EditorValue::Object{{"field", fieldName}, {"value", value}, {"present", true}};
115 operation.inverse = EditorValue::Object{{"field", fieldName}, {"value", before},
116 {"present", current != object->end()}};
117 operation.hasInverse = true;
118 operation.affectedObjects.push_back({TargetId(targetId_), id_, 0});
119 operation.affectedProperties.push_back(fieldName);
120 operation.mergeKey = targetId_ + ":" + fieldName;
121 return eve::editing::applied<DomainOperation>(std::move(operation));
122}

124std::vector<EditorDiagnostic> DefinitionDocument::validate(const SchemaValidator& schema,
125 const ReferenceResolver& references) const {
126 std::vector<EditorDiagnostic> diagnostics;
127 if (type_.empty() || id_.empty() || version_ <= 0)
128 diagnostics.push_back(eve::editing::ruleDiagnostic(
129 eve::DiagnosticCode::PreconditionViolation,
130 RuleId("editor.definition.invalid-identity"), DiagnosticSeverity::Error,
131 "Definition type, id and positive version are required"));
132 if (!looksLikeJsonValue(json_))
133 diagnostics.push_back(eve::editing::ruleDiagnostic(
134 eve::DiagnosticCode::InvalidArgument, RuleId("editor.definition.invalid-json-shape"),
135 DiagnosticSeverity::Error, "Definition payload must be a JSON object or array"));
136 if (schema) {
137 auto schemaDiagnostics = schema(type_, version_, json_);
138 diagnostics.insert(diagnostics.end(), std::make_move_iterator(schemaDiagnostics.begin()),
139 std::make_move_iterator(schemaDiagnostics.end()));
140 }
141 for (const DefinitionReferenceField& reference : references_) {
142 if (reference.type.empty() || reference.id.empty()) {
143 if (reference.required)
144 diagnostics.push_back(eve::editing::ruleDiagnostic(
145 eve::DiagnosticCode::PreconditionViolation,
146 RuleId("editor.definition.reference-required"), DiagnosticSeverity::Error,
147 "Required reference is empty: " + reference.path));
148 continue;
149 }
150 if (references && !references(reference.type, reference.id))
151 diagnostics.push_back(eve::editing::ruleDiagnostic(
152 eve::DiagnosticCode::NotFound, RuleId("editor.definition.reference-not-found"),
153 DiagnosticSeverity::Error,
154 "Definition reference could not be resolved: " + reference.path + " -> " +
155 reference.type + ":" + reference.id));
156 }
157 return diagnostics;
158}

170EditorResult<void> DefinitionDocument::loadSnapshot(const EditorValue& snapshot) {
171 const auto* schemaValue = field(snapshot, "schemaVersion");
172 const auto* typeValue = field(snapshot, "type");
173 const auto* idValue = field(snapshot, "id");
174 const auto* versionValue = field(snapshot, "definitionVersion");
175 const auto* jsonValue = field(snapshot, "json");
176 const auto* referencesValue = field(snapshot, "references");
177 const auto* schema = schemaValue ? schemaValue->getIf<int64_t>() : nullptr;
178 const auto* type = typeValue ? typeValue->getIf<std::string>() : nullptr;
179 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
180 const auto* version = versionValue ? versionValue->getIf<int64_t>() : nullptr;
181 const auto* json = jsonValue ? jsonValue->getIf<std::string>() : nullptr;
182 const auto* references = referencesValue ? referencesValue->getIf<EditorValue::Array>() : nullptr;
183 if (!schema || *schema != 1 || !type || type->empty() || !id || id->empty() || !version || *version <= 0 ||
184 !json || !looksLikeJsonValue(*json) || !references)
185 return definitionError<void>(EditorStatus::Rejected, "editor.definition.invalid-snapshot",
186 "Definition snapshot is invalid or unsupported");
187 std::vector<DefinitionReferenceField> parsed;
188 for (const EditorValue& value : *references) {
189 const auto* pathValue = field(value, "path");
190 const auto* refTypeValue = field(value, "type");
191 const auto* refIdValue = field(value, "id");
192 const auto* requiredValue = field(value, "required");
193 const auto* path = pathValue ? pathValue->getIf<std::string>() : nullptr;
194 const auto* refType = refTypeValue ? refTypeValue->getIf<std::string>() : nullptr;
195 const auto* refId = refIdValue ? refIdValue->getIf<std::string>() : nullptr;
196 const auto* required = requiredValue ? requiredValue->getIf<bool>() : nullptr;
197 if (!path || !refType || !refId || !required)
198 return definitionError<void>(EditorStatus::Rejected, "editor.definition.invalid-reference",
199 "Definition snapshot contains an invalid reference");
200 parsed.push_back({*path, *refType, *refId, *required});
201 }
202 DefinitionDocument candidate(*type, *id, static_cast<int>(*version));
203 if (!candidate.setJson(*json).ok() || !candidate.setReferences(std::move(parsed)).ok())
204 return definitionError<void>(EditorStatus::Rejected, "editor.definition.invalid-snapshot",
205 "Definition snapshot could not be applied");
206 type_ = std::move(candidate.type_);
207 id_ = std::move(candidate.id_);
208 version_ = candidate.version_;
209 json_ = std::move(candidate.json_);
210 references_ = std::move(candidate.references_);
211 targetId_ = "definition:" + type_ + ":" + id_;
212 ++revision_;
213 dirty_.clear();
214 return eve::editing::applied<void>();
215}
