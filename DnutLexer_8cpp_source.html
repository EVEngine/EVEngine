32std::size_t twoCharacterPunctuatorLength(std::string_view source, std::size_t i) {

92eve::Result<std::vector<DnutToken>> lexDnut(std::string_view source, const std::string& path) {
93 std::vector<DnutToken> tokens;
94 std::size_t i = 0;
95 int line = 1;
96 int column = 1;
97 const std::size_t size = source.size();
98
99 const auto advanceOne = [&]() {
100 if (source[i] == '\n') {
101 ++line;
102 column = 1;
103 } else {
104 ++column;
105 }
106 ++i;
107 };
108
109 while (i < size) {
110 const char c = source[i];
111
112 if (c == '\n' || c == ' ' || c == '\t' || c == '\r') {
113 advanceOne();
114 continue;
115 }
116 if (c == '/' && i + 1 < size && source[i + 1] == '/') {
117 while (i < size && source[i] != '\n') ++i;
118 continue;
119 }
120 if (c == '/' && i + 1 < size && source[i + 1] == '*') {
121 const int startLine = line;
122 const int startColumn = column;
123 advanceOne();
124 advanceOne();
125 bool closed = false;
126 while (i < size) {
127 if (source[i] == '*' && i + 1 < size && source[i + 1] == '/') {
128 advanceOne();
129 advanceOne();
130 closed = true;
131 break;
132 }
133 advanceOne();
134 }
135 if (!closed)
136 return lexFailure(path, startLine, startColumn, "unterminated block comment");
137 continue;
138 }
139 if (c == '"' || c == '\'') {
140 const char quote = c;
141 const int startLine = line;
142 const int startColumn = column;
143 advanceOne();
144 std::string text;
145 bool terminated = false;
146 while (i < size) {
147 if (source[i] == quote) {
148 advanceOne();
149 terminated = true;
150 break;
151 }
152 if (source[i] == '\\') {
153 advanceOne();
154 if (i >= size) break;
155 if (!appendEscape(source[i], text))
156 return lexFailure(path, line, column,
157 std::string("unknown escape sequence '\\") + source[i] + "'");
158 advanceOne();
159 continue;
160 }
161 text += source[i];
162 advanceOne();
163 }
164 if (!terminated) return lexFailure(path, startLine, startColumn, "unterminated string literal");
165 tokens.push_back({DnutTokenKind::String, std::move(text), startLine, startColumn});
166 continue;
167 }
168 if (isDigit(c)) {
169 const int startLine = line;
170 const int startColumn = column;
171 const std::size_t start = i;
172 while (i < size) {
173 const char d = source[i];
174 if (isDigit(d) || d == '.') {
175 advanceOne();
176 } else if ((d == 'e' || d == 'E') && i + 1 < size &&
177 (isDigit(source[i + 1]) || source[i + 1] == '+' || source[i + 1] == '-')) {
178 advanceOne();
179 advanceOne();
180 } else {
181 break;
182 }
183 }
184 tokens.push_back({DnutTokenKind::Number, std::string(source.substr(start, i - start)), startLine,
185 startColumn});
186 continue;
187 }
188 if (isIdentifierStart(c)) {
189 const int startLine = line;
190 const int startColumn = column;
191 const std::size_t start = i;
192 // A '-' joins an identifier only when a word character follows it, so
193 // dashed content ids (`iron-sword`) stay one token while arrow and
194 // minus punctuation (`->`, `-3`) are unaffected.
195 const auto continuesIdentifier = [&source, size](std::size_t position) {
196 if (isIdentifierBody(source[position])) return true;
197 return source[position] == '-' && position + 1 < size && isIdentifierBody(source[position + 1]);
198 };
199 while (i < size && continuesIdentifier(i)) advanceOne();
200 tokens.push_back(
201 {DnutTokenKind::Identifier, std::string(source.substr(start, i - start)), startLine, startColumn});
202 continue;
203 }
204 if (const std::size_t pairLength = twoCharacterPunctuatorLength(source, i); pairLength != 0) {
205 const int startLine = line;
206 const int startColumn = column;
207 const std::string text(source.substr(i, pairLength));
208 advanceOne();
209 advanceOne();
210 tokens.push_back({DnutTokenKind::Punctuator, text, startLine, startColumn});
211 continue;
212 }
213 if (isSingleCharacterPunctuator(c)) {
214 const int startLine = line;
215 const int startColumn = column;
216 const std::string text(1, c);
217 advanceOne();
218 tokens.push_back({DnutTokenKind::Punctuator, std::move(text), startLine, startColumn});
219 continue;
220 }
221 return lexFailure(path, line, column, std::string("unexpected character '") + c + "'");
222 }
223
224 tokens.push_back({DnutTokenKind::EndOfFile, std::string{}, line, column});
225 return eve::Result<std::vector<DnutToken>>::success(std::move(tokens));
226}
