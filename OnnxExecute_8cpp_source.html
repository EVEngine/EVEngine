211std::vector<RuntimeTensor> execute(const Node& n, const std::vector<const RuntimeTensor*>& in, OnnxCompute* compute) {
212 if (n.domain == "com.microsoft" && n.op == "DynamicQuantizeLSTM") return executeQuantLstm(n, in, compute);
213 if (n.op == "Constant") {
214 auto a = n.attrs.find("value");
215 if (a == n.attrs.end() || !a->second.tensor) throw Failure("Constant requires tensor value");
216 return single(*a->second.tensor);
217 }
218 if (n.op.find("Quantize") != std::string::npos || n.op == "DequantizeLinear" || n.op == "MatMulInteger" ||
219 n.op == "ConvInteger")
220 return executeQuant(n, in, compute);
221 const auto& x = required(in, 0);
222 if (n.op == "Identity") return single(x);
223 if (n.op == "Shape") {
224 const int64_t rank = x.shape.size();
225 auto norm = [rank](int64_t i) { return std::clamp(i < 0 ? i + rank : i, int64_t(0), rank); };
226 const auto begin = norm(attr(n, "start", 0)), end = norm(attr(n, "end", rank));
227 std::vector<int64_t> shape;
228 if (begin < end) shape.assign(x.shape.begin() + begin, x.shape.begin() + end);
229 return single(make(OnnxElement::Int64, {static_cast<int64_t>(shape.size())}, shape));
230 }
231 if (n.op == "Cast") {
232 const auto target = static_cast<OnnxElement>(attr(n, "to", 0));
233 if (compute && target == OnnxElement::Float32 && x.element == OnnxElement::Int32)
234 return single(dispatchFloat(*compute, {&x}, x.shape, "y[i]=float(x0[i]);"));
235 const auto size = elementSize(target);
236 RuntimeTensor out{target, x.shape, std::vector<uint8_t>(count(x.shape) * size)};
237 for (size_t i = 0; i < count(x.shape); ++i) {
238 if (target == OnnxElement::Float32) {
239 const float v =
240 x.element == OnnxElement::Float32 ? read<float>(x, i) : static_cast<float>(integer(x, i));
241 std::memcpy(out.bytes.data() + i * 4, &v, 4);
242 } else {
243 int64_t v;
244 if (x.element == OnnxElement::Float32) {
245 const double f = read<float>(x, i);
246 if (!std::isfinite(f) || f < -9223372036854775808.0 || f >= 9223372036854775808.0)
247 throw Failure("Cast overflow");
248 v = target == OnnxElement::Bool ? (f != 0) : static_cast<int64_t>(f);
249 } else
250 v = integer(x, i);
251 if (target == OnnxElement::Bool) v = v != 0;
252 std::memcpy(out.bytes.data() + i * size, &v, size);
253 }
254 }
255 return single(std::move(out));
256 }
257 if (n.op == "Gather") {
258 const auto& idx = required(in, 1);
259 const int a = axis(attr(n, "axis", 0), x.shape.size());
260 if (idx.element != OnnxElement::Int32 && idx.element != OnnxElement::Int64)
261 throw Failure("Gather indices must be int32/int64");
262 size_t inner = 1, outer = 1;
263 for (size_t i = a + 1; i < x.shape.size(); ++i) inner *= x.shape[i];
264 for (int i = 0; i < a; ++i) outer *= x.shape[i];
265 std::vector<int64_t> shape(x.shape.begin(), x.shape.begin() + a);
266 shape.insert(shape.end(), idx.shape.begin(), idx.shape.end());
267 shape.insert(shape.end(), x.shape.begin() + a + 1, x.shape.end());
268 const size_t stride = inner * elementSize(x.element), ni = count(idx.shape);
269 RuntimeTensor out{x.element, shape, std::vector<uint8_t>(count(shape) * elementSize(x.element))};
270 for (size_t o = 0; o < outer; ++o)
271 for (size_t i = 0; i < ni; ++i) {
272 auto j = integer(idx, i);
273 if (j < 0) j += x.shape[a];
274 if (j < 0 || j >= x.shape[a]) throw Failure("Gather index out of range");
275 if (stride)
276 std::memcpy(out.bytes.data() + (o * ni + i) * stride,
277 x.bytes.data() + (o * x.shape[a] + j) * stride, stride);
278 }
279 return single(std::move(out));
280 }
281 if (n.op == "Reshape" || n.op == "Unsqueeze" || n.op == "Squeeze") {
282 auto shape = x.shape;
283 if (n.op == "Reshape") {
284 shape = ints(required(in, 1));
285 int infer = -1;
286 size_t known = 1;
287 for (size_t i = 0; i < shape.size(); ++i) {
288 if (shape[i] == 0) {
289 if (i >= x.shape.size()) throw Failure("Reshape zero axis out of range");
290 shape[i] = x.shape[i];
291 }
292 if (shape[i] == -1) {
293 if (infer != -1) throw Failure("Multiple inferred dimensions");
294 infer = static_cast<int>(i);
295 } else {
296 if (shape[i] < 0 || shape[i] > INT32_MAX ||
297 (shape[i] && known > 128u * 1024u * 1024u / static_cast<size_t>(shape[i])))
298 throw Failure("Reshape overflow");
299 known *= static_cast<size_t>(shape[i]);
300 }
301 }
302 if (infer >= 0) {
303 if (!known || count(x.shape) % known) throw Failure("Invalid inferred shape");
304 shape[infer] = count(x.shape) / known;
305 }
306 } else {
307 std::vector<int64_t> axes;
308 if (in.size() > 1 && in[1])
309 axes = ints(*in[1]);
310 else if (n.op == "Unsqueeze")
311 throw Failure("Unsqueeze needs axes");
312 else
313 for (size_t i = 0; i < shape.size(); ++i)
314 if (shape[i] == 1) axes.push_back(i);
315 const size_t rank = shape.size() + (n.op == "Unsqueeze" ? axes.size() : 0);
316 std::set<int> positions;
317 for (auto a : axes)
318 if (!positions.insert(axis(a, rank)).second) throw Failure("Duplicate squeeze axis");
319 if (n.op == "Unsqueeze") {
320 for (int a : positions) shape.insert(shape.begin() + a, 1);
321 } else
322 for (auto it = positions.rbegin(); it != positions.rend(); ++it) {
323 if (shape[*it] != 1) throw Failure("Squeeze dimension not one");
324 shape.erase(shape.begin() + *it);
325 }
326 }
327 if (count(shape) != count(x.shape)) throw Failure("Reshape element count mismatch");
328 RuntimeTensor out = x;
329 out.shape = std::move(shape);
330 return single(std::move(out));
331 }
332 if (n.op == "Transpose") {
333 std::vector<int64_t> order = attrs(n, "perm", {});
334 if (order.empty())
335 for (size_t i = x.shape.size(); i > 0; --i) order.push_back(i - 1);
336 if (order.size() != x.shape.size()) throw Failure("Invalid transpose rank");
337 std::set<int64_t> used;
338 std::vector<int64_t> shape;
339 for (auto a : order) {
340 if (a < 0 || a >= static_cast<int64_t>(x.shape.size()) || !used.insert(a).second)
341 throw Failure("Invalid transpose permutation");
342 shape.push_back(x.shape[a]);
343 }
344 RuntimeTensor out{x.element, shape, std::vector<uint8_t>(x.bytes.size())};
345 const auto size = elementSize(x.element);
346 std::vector<size_t> strides(x.shape.size(), 1);
347 for (size_t i = x.shape.size(); i > 1; --i) strides[i - 2] = strides[i - 1] * x.shape[i - 1];
348 for (size_t i = 0; i < count(shape); ++i) {
349 size_t rest = i, source = 0;
350 for (size_t j = shape.size(); j > 0; --j) {
351 source += (rest % shape[j - 1]) * strides[order[j - 1]];
352 rest /= shape[j - 1];
353 }
354 std::memcpy(out.bytes.data() + i * size, x.bytes.data() + source * size, size);
355 }
356 return single(std::move(out));
357 }
358 if (auto indexResult = executeIndex(n, in)) return std::move(*indexResult);
359 if (auto shapeResult = executeShape(n, in)) return single(std::move(*shapeResult));
360 if (auto miscResult = executeMisc(n, in, compute)) return single(std::move(*miscResult));
361 if (auto neuralResult = executeNeural(n, in, compute)) return single(std::move(*neuralResult));
362 if (auto numericResult = executeNumeric(n, in, compute)) return single(std::move(*numericResult));
363 const auto a = floats(x);
364 if (n.op == "Softmax") {
365 auto xd = dims(x);
366 std::vector<float> out(a.size());
367 kernels::softmax(a.data(), xd.data(), static_cast<int>(xd.size()), axis(attr(n, "axis", -1), xd.size()), false,
368 out.data());
369 return single(make(OnnxElement::Float32, x.shape, out));
370 }
371 static const std::map<std::string, OpType> unary{
372 {"Sqrt", OpType::Sqrt}, {"Exp", OpType::Exp}, {"Log", OpType::Log}, {"Sin", OpType::Sin},
373 {"Cos", OpType::Cos}, {"Relu", OpType::Relu}, {"Sigmoid", OpType::Sigmoid}, {"Tanh", OpType::Tanh},
374 {"Neg", OpType::Neg}, {"Abs", OpType::Abs}};
375 auto op = unary.find(n.op);
376 if (op == unary.end()) throw Failure("Unsupported operator", DiagnosticCode::Unsupported);
377 std::vector<float> out(a.size());
378 kernels::unaryOp(op->second, a.data(), static_cast<int>(a.size()), out.data(), 0, 0);
379 return single(make(OnnxElement::Float32, x.shape, out));
380}

386std::string prefix(bool as, bool bs) {
387 std::ostringstream s;
388 s << "#version 450\nlayout(local_size_x=64) in;\n"
389 "layout(std430,binding=0) readonly buffer A {uint a[];};\n"
390 "layout(std430,binding=1) readonly buffer B {uint b[];};\n"
391 "layout(std430,binding=2) readonly buffer Z {int zp[];};\n"
392 "layout(std430,binding=3) writeonly buffer Y {int y[];};\n"
393 "int av(uint i){int v=int((a[i/4]>>((i%4)*8))&255u);return "
394 << (as ? "v>=128?v-256:v" : "v")
395 << ";}\n"
396 "int bv(uint i){int v=int((b[i/4]>>((i%4)*8))&255u);return "
397 << (bs ? "v>=128?v-256:v" : "v") << ";}\n";
398 return s.str();
399}

432OnnxBuffer enqueueInteger(OnnxCompute& d, const std::string& source, OnnxBuffer a, OnnxBuffer b,
433 std::span<const int32_t> z, size_t size, size_t terms) {
434 if (terms > INT32_MAX / 65025 || !size || size > 64u * 65535u)
435 throw Failure("GPU integer kernel exceeds exact accumulator or dispatch limit", DiagnosticCode::Unsupported);
436 std::vector<uint8_t> bytes(z.size_bytes());
437 std::memcpy(bytes.data(), z.data(), bytes.size());
438 auto host = std::make_shared<const std::vector<uint8_t>>(std::move(bytes));
439 const std::vector<OnnxBuffer> inputs{a, b, {host, {}, host->size()}};
440 auto r = d.enqueue(source, inputs, size * 4, static_cast<uint32_t>(size));
441 if (!r.ok()) throw Failure(r.error()->message(), r.error()->code());
442 return std::move(r.value());
443}
444OnnxBuffer gpuMatmulResident(OnnxCompute& d, OnnxBuffer a, OnnxBuffer b, bool as, bool bs, size_t m, size_t k, size_t n,
445 int az, std::span<const int32_t> bz, size_t bOffset) {
446 if (bz.size() != 1 && bz.size() != n) throw Failure("GPU matmul zero-point extent mismatch");
447 std::ostringstream s;
448 s << prefix(as, bs) << "void main(){uint i=gl_GlobalInvocationID.x;if(i>=" << m * n << "u)return;"
449 << "uint r=i/" << n << "u,c=i%" << n << "u;int v=0;for(uint j=0;j<" << k << "u;++j)"
450 << "v+=(av(r*" << k << "u+j)-(" << az << "))*(bv(" << bOffset << "u+j*" << n << "u+c)-zp["
451 << (bz.size() == 1 ? "0" : "c") << "]);y[i]=v;}";
452 return enqueueInteger(d, s.str(), a, b, bz, m * n, k);

474 << c.padLeft << "+kx*" << c.dilationW << ";"
475 << "if(iy<0||ix<0||iy>=" << c.height << "||ix>=" << c.width << ")continue;"
476 << "uint ai=uint(((batch*" << c.channels << "+o/" << c.outputs / c.groups << "*" << cg << "+ch)*" << c.height
477 << "+iy)*" << c.width << "+ix);"
478 << "uint bi=uint(((o*" << cg << "+ch)*" << c.kernelH << "+ky)*" << c.kernelW << "+kx);"
479 << "v+=(av(ai)-(" << xz << "))*(bv(bi)-zp[" << (wz.size() == 1 ? "0" : "o") << "]);}y[i]=v;}";

486std::optional<std::vector<RuntimeTensor>> executeIndex(const Node& n, const std::vector<const RuntimeTensor*>& in) {
487 const auto& x = required(in, 0);
488 auto one = [](RuntimeTensor t) { return std::vector<RuntimeTensor>{std::move(t)}; };
489 if (n.op == "Range") {
490 const auto& end = required(in, 1);
491 const auto& delta = required(in, 2);
492 if (count(x.shape) != 1 || count(end.shape) != 1 || count(delta.shape) != 1 || x.element != end.element ||
493 x.element != delta.element)
494 throw Failure("Range requires matching scalars");
495 if (x.element == OnnxElement::Float32) {
496 const double start = read<float>(x, 0), stop = read<float>(end, 0), step = read<float>(delta, 0);
497 if (!std::isfinite(start) || !std::isfinite(stop) || !std::isfinite(step) || step == 0)
498 throw Failure("Invalid Range");
499 const double length = std::max(0., std::ceil((stop - start) / step));
500 if (length > 128u * 1024u * 1024u) throw Failure("Range too large");
501 std::vector<float> v(static_cast<size_t>(length));
502 for (size_t i = 0; i < v.size(); ++i) v[i] = static_cast<float>(start + i * step);
503 return one(make(x.element, {static_cast<int64_t>(v.size())}, v));
504 }
505 if (x.element != OnnxElement::Int64 && x.element != OnnxElement::Int32)
506 throw Failure("Range integer dtype unsupported");
507 const int64_t start = integer(x), stop = integer(end), step = integer(delta);
508 if (!step || step == INT64_MIN) throw Failure("Invalid Range step");
509 uint64_t length = 0;
510 if (step > 0 && stop > start) length = 1 + (uint64_t(stop) - uint64_t(start) - 1) / uint64_t(step);
511 if (step < 0 && stop < start) length = 1 + (uint64_t(start) - uint64_t(stop) - 1) / uint64_t(-step);
512 if (length > 128u * 1024u * 1024u) throw Failure("Range too large");
513 RuntimeTensor out{
514 x.element, {static_cast<int64_t>(length)}, std::vector<uint8_t>(length * elementSize(x.element))};
515 int64_t v = start;
516 for (size_t i = 0; i < length; ++i) {
517 std::memcpy(out.bytes.data() + i * elementSize(x.element), &v, elementSize(x.element));
518 if (i + 1 < length) v += step;
519 }
520 return one(std::move(out));
521 }
522 if (n.op == "TopK") {
523 const auto& kt = required(in, 1);
524 if (count(kt.shape) != 1) throw Failure("TopK requires one K");
525 int a = axis(attr(n, "axis", -1), x.shape.size());
526 const int64_t k = integer(kt), size = x.shape[a];
527 if (k < 0 || k > size) throw Failure("Invalid TopK K");
528 size_t outer = 1, inner = 1;
529 for (int j = 0; j < a; ++j) outer *= x.shape[j];
530 for (size_t j = a + 1; j < x.shape.size(); ++j) inner *= x.shape[j];
531 auto shape = x.shape;
532 shape[a] = k;
533 RuntimeTensor values{x.element, shape, std::vector<uint8_t>(count(shape) * elementSize(x.element))};
534 std::vector<int64_t> indices(count(shape));
535 for (size_t o = 0; o < outer; ++o)
536 for (size_t i = 0; i < inner; ++i) {
537 std::vector<int64_t> order(size);
538 std::iota(order.begin(), order.end(), 0);
539 std::stable_sort(order.begin(), order.end(), [&](int64_t aa, int64_t bb) {
540 auto compare = [&](auto u, auto v) { return attr(n, "largest", 1) ? u > v : u < v; };
541 const size_t ai = (o * size + aa) * inner + i, bi = (o * size + bb) * inner + i;
542 return x.element == OnnxElement::Float32 ? compare(read<float>(x, ai), read<float>(x, bi))
543 : compare(integer(x, ai), integer(x, bi));
544 });
545 for (int64_t j = 0; j < k; ++j) {
546 size_t dst = (o * k + j) * inner + i, src = (o * size + order[j]) * inner + i;
547 indices[dst] = order[j];
548 std::memcpy(values.bytes.data() + dst * elementSize(x.element),
549 x.bytes.data() + src * elementSize(x.element), elementSize(x.element));
550 }
551 }
552 return std::vector<RuntimeTensor>{std::move(values), make(OnnxElement::Int64, shape, indices)};
553 }
554 if (n.op == "ScatterElements" || n.op == "ScatterND") {
555 const auto& idx = required(in, 1);
556 const auto& updates = required(in, 2);
557 if (updates.element != x.element || (idx.element != OnnxElement::Int64 && idx.element != OnnxElement::Int32))
558 throw Failure("Scatter dtype mismatch");
559 auto out = x;
560 const auto bytes = elementSize(x.element);
561 if (n.op == "ScatterElements") {
562 if (idx.shape != updates.shape || idx.shape.size() != x.shape.size())
563 throw Failure("ScatterElements shape mismatch");
564 int a = axis(attr(n, "axis", 0), x.shape.size());
565 for (size_t j = 0; j < x.shape.size(); ++j)
566 if (j != a && idx.shape[j] > x.shape[j]) throw Failure("Scatter index shape exceeds data");
567 for (size_t i = 0; i < count(idx.shape); ++i) {
568 size_t rest = i, source = 0, stride = 1;
569 for (size_t j = x.shape.size(); j > 0; --j) {
570 int64_t c = rest % idx.shape[j - 1];
571 rest /= idx.shape[j - 1];
572 if (j - 1 == a) {
573 c = integer(idx, i);
574 if (c < 0) c += x.shape[j - 1];
575 }
576 if (c < 0 || c >= x.shape[j - 1]) throw Failure("Scatter index out of range");
577 source += c * stride;
578 stride *= x.shape[j - 1];
579 }
580 std::memcpy(out.bytes.data() + source * bytes, updates.bytes.data() + i * bytes, bytes);
581 }
582 } else {
583 if (idx.shape.empty() || idx.shape.back() < 1 || static_cast<size_t>(idx.shape.back()) > x.shape.size())
584 throw Failure("ScatterND index rank mismatch");
585 const size_t depth = idx.shape.back();
586 std::vector<int64_t> expected(idx.shape.begin(), idx.shape.end() - 1);
587 expected.insert(expected.end(), x.shape.begin() + depth, x.shape.end());
588 if (expected != updates.shape) throw Failure("ScatterND updates shape mismatch");
589 size_t chunk = 1;
590 for (size_t j = depth; j < x.shape.size(); ++j) chunk *= x.shape[j];
591 for (size_t i = 0; i < count(idx.shape) / depth; ++i) {
592 size_t target = 0;
593 for (size_t j = 0; j < depth; ++j) {
594 auto c = integer(idx, i * depth + j);
595 if (c < 0) c += x.shape[j];
596 if (c < 0 || c >= x.shape[j]) throw Failure("ScatterND index out of range");
597 target = target * x.shape[j] + c;
598 }
599 if (chunk)
600 std::memcpy(out.bytes.data() + target * chunk * bytes, updates.bytes.data() + i * chunk * bytes,
601 chunk * bytes);
602 }
603 }
604 return one(std::move(out));
605 }
606 if (n.op == "ReduceProd" || n.op == "ReduceMax") {
607 auto axes = attrs(n, "axes", {});
608 if (axes.empty())
609 for (size_t i = 0; i < x.shape.size(); ++i) axes.push_back(i);
610 std::set<int> reduced;
611 for (auto a : axes)
612 if (!reduced.insert(axis(a, x.shape.size())).second) throw Failure("Duplicate reduction axis");
613 auto mapShape = x.shape;
614 std::vector<int64_t> shape;
615 for (size_t j = 0; j < x.shape.size(); ++j) {
616 if (reduced.contains(static_cast<int>(j))) {
617 mapShape[j] = 1;
618 if (attr(n, "keepdims", 1)) shape.push_back(1);
619 } else
620 shape.push_back(x.shape[j]);
621 }
622 if (x.element == OnnxElement::Float32) {
623 std::vector<float> out(count(shape), n.op == "ReduceProd" ? 1.f : -std::numeric_limits<float>::infinity());
624 for (size_t i = 0; i < count(x.shape); ++i) {
625 size_t t = broadcastIndex(i, mapShape, x.shape);
626 const float v = read<float>(x, i);
627 out[t] = n.op == "ReduceProd" ? out[t] * v : std::max(out[t], v);
628 }
629 return one(make(x.element, shape, out));
630 }
631 if (x.element != OnnxElement::Int64 && x.element != OnnxElement::Int32)
632 throw Failure("Integer reduction dtype unsupported");
633 std::vector<int64_t> out(count(shape), n.op == "ReduceProd" ? 1 : INT64_MIN);
634 for (size_t i = 0; i < count(x.shape); ++i) {
635 auto& a = out[broadcastIndex(i, mapShape, x.shape)];
636 const auto b = integer(x, i);
637 if (n.op == "ReduceMax")
638 a = std::max(a, b);
639 else {
640 if (a && b &&
641 (a > 0 ? (b > 0 ? a > INT64_MAX / b : b < INT64_MIN / a)
642 : (b > 0 ? a < INT64_MIN / b : a < INT64_MAX / b)))
643 throw Failure("ReduceProd overflow");
644 a *= b;
645 }
646 }
647 if (x.element == OnnxElement::Int64) return one(make(x.element, shape, out));
648 std::vector<int32_t> narrow;
649 for (auto v : out) {
650 if (v < INT32_MIN || v > INT32_MAX) throw Failure("Int32 reduction overflow");
651 narrow.push_back(static_cast<int32_t>(v));
652 }
653 return one(make(x.element, shape, narrow));
654 }
655 return std::nullopt;
656}

676struct Weights {
677 const RuntimeTensor& tensor;
678 std::vector<float> scales;
679 std::vector<int64_t> zeros;
680 size_t directions, gates, rows;
681 Weights(const RuntimeTensor& t, const RuntimeTensor& s, const RuntimeTensor& z, size_t d, size_t g, size_t r)
682 : tensor(t), scales(floats(s)), zeros(ints(z)), directions(d), gates(g), rows(r) {
683 if (t.element != OnnxElement::Int8 && t.element != OnnxElement::UInt8)
684 throw Failure("LSTM weights must be 8-bit");
685 if (t.element != z.element || s.shape != z.shape ||
686 (s.shape != std::vector<int64_t>{static_cast<int64_t>(d)} &&
687 s.shape != std::vector<int64_t>{static_cast<int64_t>(d), static_cast<int64_t>(g)}))
688 throw Failure("LSTM quantization parameter shape mismatch");
689 shape(t, {static_cast<int64_t>(d), static_cast<int64_t>(r), static_cast<int64_t>(g)});
690 for (float v : scales)
691 if (!(v > 0) || !std::isfinite(v)) throw Failure("Invalid LSTM weight scale");
692 }
693 std::vector<float> multiply(const affine::QuantizedActivation& a, size_t batch, size_t direction,
694 OnnxCompute* compute) const {
695 std::vector<float> out(batch * gates);
696 const bool sign = tensor.element == OnnxElement::Int8;
697 if (compute) {
698 std::vector<int32_t> z(scales.size() == directions ? 1 : gates);
699 for (size_t i = 0; i < z.size(); ++i) z[i] = static_cast<int32_t>(zeros[direction * z.size() + i]);
700 ByteStorage activation(a.values);
701 ByteStorage accumulator(gpuMatmulResident(*compute, activation.buffer(), tensor.bytes.buffer(), false, sign,
702 batch, rows, gates, a.zeroPoint, z, direction * rows * gates));
703 std::vector<int32_t> sums(batch * gates);
704 std::memcpy(sums.data(), static_cast<const ByteStorage&>(accumulator).data(),
705 sums.size() * sizeof(int32_t));
706 for (size_t b = 0; b < batch; ++b)
707 for (size_t g = 0; g < gates; ++g)
708 out[b * gates + g] = static_cast<float>(sums[b * gates + g]) * a.scale *
709 scales[direction * z.size() + (z.size() == 1 ? 0 : g)];
710 return out;
711 }
712 // Per-output-channel affine zero points do not require unpacking a weight matrix.
713 for (size_t b = 0; b < batch; ++b)
714 for (size_t g = 0; g < gates; ++g) {
715 const size_t p = scales.size() == directions ? direction : direction * gates + g;
716 int64_t sum = 0;
717 for (size_t k = 0; k < rows; ++k) {
718 const int byte = tensor.bytes[(direction * rows + k) * gates + g];
719 const int w = sign && byte >= 128 ? byte - 256 : byte;
720 sum += static_cast<int64_t>(int(a.values[b * rows + k]) - a.zeroPoint) * (w - zeros[p]);
721 }
722 if (sum < INT32_MIN || sum > INT32_MAX) throw Failure("LSTM integer accumulator overflow");
723 out[b * gates + g] = static_cast<float>(sum) * a.scale * scales[p];
724 }
725 return out;
726 }
727};

729std::vector<RuntimeTensor> executeQuantLstm(const Node& n, const std::vector<const RuntimeTensor*>& in,
730 OnnxCompute* compute) {
731 const auto& xt = required(in, 0);
732 if (xt.shape.size() != 3) throw Failure("LSTM X must be [time,batch,input]");
733 const auto x = floats(xt);
734 const int64_t time = xt.shape[0], batch = xt.shape[1], input = xt.shape[2], hidden = attr(n, "hidden_size", 0);
735 const auto direction = n.attrs.contains("direction") ? n.attrs.at("direction").text : "forward";
736 const int64_t dirs = direction == "bidirectional" ? 2 : 1;
737 if (time <= 0 || batch <= 0 || input <= 0 || hidden <= 0 || hidden > 16384)
738 throw Failure("Invalid LSTM dimensions");
739 const size_t gates = 4 * hidden;
740 const size_t projectionSize = count({time, batch, static_cast<int64_t>(gates)});
741 if (projectionSize > 128u * 1024u * 1024u) throw Failure("LSTM projection exceeds memory limit");
742 Weights w(required(in, 1), required(in, 8), required(in, 9), dirs, gates, input);
743 Weights r(required(in, 2), required(in, 10), required(in, 11), dirs, gates, hidden);
744 auto biases = optionalFloats(in, 3, {dirs, 8 * hidden});
745 auto h = optionalFloats(in, 5, {dirs, batch, hidden}), c = optionalFloats(in, 6, {dirs, batch, hidden});
746 auto p = optionalFloats(in, 7, {dirs, 3 * hidden});
747 std::vector<int64_t> lengths(batch, time);
748 if (in.size() > 4 && in[4]) {
749 if (in[4]->element != OnnxElement::Int32) throw Failure("LSTM sequence lengths require int32");
750 shape(*in[4], {batch});
751 lengths = ints(*in[4]);
752 }
753 for (auto len : lengths)
754 if (len < 0 || len > time) throw Failure("Invalid LSTM sequence length");
755 const int64_t coupled = attr(n, "input_forget", 0);
756 if (coupled != 0 && coupled != 1) throw Failure("Invalid coupled gate flag");
757 const float clip = n.attrs.contains("clip") ? n.attrs.at("clip").real : std::numeric_limits<float>::max();
758 if (!(clip > 0) || !std::isfinite(clip)) throw Failure("Invalid LSTM clipping threshold");
759 auto sigmoid = [clip](float z) { return 1.f / (1.f + std::exp(-std::clamp(z, -clip, clip))); };
760 auto tanh = [clip](float z) { return std::tanh(std::clamp(z, -clip, clip)); };
761 std::vector<float> y(count({time, dirs, batch, hidden}), 0);
762 const auto qx = checked(affine::dynamicQuantize(x));
763 for (int64_t d = 0; d < dirs; ++d) {
764 const auto projected = w.multiply(qx, time * batch, d, compute);
765 const bool reverse = direction == "reverse" || d == 1;
766 for (int64_t step = 0; step < time; ++step) {
767 const auto qh = checked(affine::dynamicQuantize(std::span(h).subspan(d * batch * hidden, batch * hidden)));
768 const auto recurrent = r.multiply(qh, batch, d, compute);
769 for (int64_t b = 0; b < batch; ++b) {
770 if (step >= lengths[b]) continue;
771 const int64_t t = reverse ? lengths[b] - 1 - step : step;
772 const size_t state = (d * batch + b) * hidden;
773 for (int64_t j = 0; j < hidden; ++j) {
774 auto gate = [&](int g) {
775 const size_t channel = g * hidden + j;
776 return projected[(t * batch + b) * gates + channel] + recurrent[b * gates + channel] +
777 biases[d * 8 * hidden + channel] + biases[d * 8 * hidden + gates + channel];
778 };
779 const float i = sigmoid(gate(0) + p[d * 3 * hidden + j] * c[state + j]);
780 const float f =
781 coupled ? 1 - i : sigmoid(gate(2) + p[d * 3 * hidden + 2 * hidden + j] * c[state + j]);
782 c[state + j] = f * c[state + j] + i * tanh(gate(3));
783 const float o = sigmoid(gate(1) + p[d * 3 * hidden + hidden + j] * c[state + j]);
784 h[state + j] = o * tanh(c[state + j]);
785 y[((t * dirs + d) * batch + b) * hidden + j] = h[state + j];
786 }
787 }
788 }
789 }
790 return {make(OnnxElement::Float32, {time, dirs, batch, hidden}, y),
791 make(OnnxElement::Float32, {dirs, batch, hidden}, h), make(OnnxElement::Float32, {dirs, batch, hidden}, c)};
792}
