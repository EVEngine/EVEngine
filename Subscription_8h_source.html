119 void notify(const Args &...args) {
120 const auto snapshot = state_->entries;
121 for (const auto &entry : snapshot) {
122 if (!entry->active || !entry->callback) continue;
123 // Copy outside any registry lock (the registry deliberately has no
124 // lock) so callback destruction/re-entrancy cannot invalidate the
125 // callable being invoked.
126 Callback callback = entry->callback;
127 if (entry->active && callback) callback(args...);
128 }

141 [[nodiscard]] std::size_t notifyChecked(FailureHandler &&onFailure, const Args &...args) {
142 const auto snapshot = state_->entries;
143 std::size_t failures = 0;
144 for (const auto &entry : snapshot) {
145 if (!entry->active || !entry->callback) continue;
146 Callback callback = entry->callback;
147 if (!entry->active || !callback) continue;
148 try {
149 callback(args...);
150 } catch (...) {
151 ++failures;
152 std::forward<FailureHandler>(onFailure)();
153 }
154 }
