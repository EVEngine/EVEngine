27eve::Result<AccountNonce> allocateAccountNonce() {
28 auto& counter = nextAccountNonce();
29 std::uint64_t value = counter.load(std::memory_order_relaxed);
30 for (;;) {
31 if (value == 0 || value == std::numeric_limits<std::uint64_t>::max())
32 return eve::Result<AccountNonce>::failure(invariantFailure("resource account nonce exhausted", "account"));
33 if (counter.compare_exchange_weak(value, value + 1, std::memory_order_relaxed, std::memory_order_relaxed))
34 return eve::Result<AccountNonce>::success(AccountNonce(value));
35 }
36}

74eve::Result<CostSpec> CostSpec::create(std::vector<ResourceCost> items) {
75 if (items.empty())
76 return eve::Result<CostSpec>::failure(invalidArgument("resource cost must contain at least one item", "cost"));
77 for (const auto& item : items) {
78 if (!item.resource.isValid())
79 return eve::Result<CostSpec>::failure(
80 invalidArgument("resource cost contains an invalid resource id", "cost.resource"));
81 if (item.amount.value() <= 0)
82 return eve::Result<CostSpec>::failure(
83 invalidArgument("resource cost amounts must be positive", "cost.amount"));
84 }
85
86 std::sort(items.begin(), items.end(),
87 [](const auto& left, const auto& right) { return left.resource.value() < right.resource.value(); });
88
89 std::vector<ResourceCost> canonical;
90 canonical.reserve(items.size());
91 for (auto& item : items) {
92 if (canonical.empty() || canonical.back().resource != item.resource) {
93 canonical.push_back(std::move(item));
94 continue;
95 }
96 auto combined = canonical.back().amount.checkedAdd(item.amount);
97 if (!combined) return eve::Result<CostSpec>::failure(combined.status());
98 canonical.back().amount = std::move(combined).takeValue();
99 }
100 return eve::Result<CostSpec>::success(CostSpec(std::move(canonical)));
101}
