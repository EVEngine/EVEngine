22eve::Result<void> AnimSyncGroup::advance(const eve::SimulationStep& step) {
23 auto seconds = detail::secondsForStep(step, hasLastTick_, lastTick_, "AnimSyncGroup");
24 if (!seconds) return eve::Result<void>::failure(seconds.status());
25 (void)std::move(seconds).takeValue();
26 usedMarkerSync_ = false;
27 if (entries_.empty()) {
28 lastTick_ = step.tick;
29 hasLastTick_ = true;
30 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
31 }
32 for (const Entry& entry : entries_) {
33 if (entry.player->hasCurrentTick() && step.tick <= entry.player->currentTick())
34 return eve::Result<void>::failure(eve::Diagnostic::error(
35 eve::DiagnosticCode::Conflict, "animation sync-group player already consumed this tick"));
36 }
37 Entry& leader = entries_[static_cast<size_t>(leader_)];
38 auto leaderResult = leader.player->advance(step);
39 if (!leaderResult) return eve::Result<void>::failure(leaderResult.status());
40 AnimClip* leaderClip = leader.player->getClip();
41 if (!leaderClip || leaderClip->getDuration() <= 1e-8f) {
42 lastTick_ = step.tick;
43 hasLastTick_ = true;
44 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
45 }
46 phase_ = leaderClip->wrapTime(leader.player->getTime()) / leaderClip->getDuration();
47 for (int i = 0; i < getCount(); ++i) {
48 if (i == leader_) continue;
49 AnimPlayer* player = entries_[static_cast<size_t>(i)].player;
50 auto playerResult = player->advance(step);
51 if (!playerResult) return eve::Result<void>::failure(playerResult.status());
52 AnimClip* clip = player->getClip();
53 if (!clip || clip->getDuration() <= 1e-8f) continue;
54 float mappedTime;
55 if (leaderClip->hasCompatibleSyncMarkers(clip)) {
56 mappedTime = leaderClip->mapSyncTimeTo(leader.player->getTime(), clip);
57 usedMarkerSync_ = true;
58 } else {
59 mappedTime = phase_ * clip->getDuration();
60 }
61 mappedTime += entries_[static_cast<size_t>(i)].phaseOffset * clip->getDuration();
62 player->setTime(clip->wrapTime(mappedTime));
63 }
64 lastTick_ = step.tick;
65 hasLastTick_ = true;
66 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
67}
