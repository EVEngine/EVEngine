45EditorResult<OffscreenPreviewArtifact> GraphicsOffscreenPreviewService::render(const OffscreenPreviewRequest& request,
46 const DrawCallback& draw,
47 int maximumDimension,
48 std::uint64_t maximumPixels) {
49 if (!impl_->graphics || !impl_->canvases || !impl_->targets || request.previewId.empty() || !draw ||
50 request.sourceRevision == 0 || request.width <= 0 || request.height <= 0 || maximumDimension <= 0 ||
51 request.width > maximumDimension || request.height > maximumDimension ||
52 static_cast<std::uint64_t>(request.width) * static_cast<std::uint64_t>(request.height) > maximumPixels)
53 return eve::editing::failed<OffscreenPreviewArtifact>(
54 EditorStatus::Rejected, RuleId("editor.preview.invalid-offscreen-request"),
55 "Offscreen preview requires graphics, identity, revision and bounded dimensions");
56 for (double component : request.clearColor)
57 if (!std::isfinite(component) || component < 0.0 || component > 1.0)
58 return eve::editing::failed<OffscreenPreviewArtifact>(EditorStatus::Rejected,
59 RuleId("editor.preview.invalid-clear-color"),
60 "Offscreen clear color must be normalized");
61 const auto key = std::make_pair(request.width, request.height);
62 graphics::Canvas*& canvas = impl_->canvasCache[key];
63 if (!canvas) canvas = impl_->canvases->newCanvas(request.width, request.height);
64 if (!canvas)
65 return eve::editing::failed<OffscreenPreviewArtifact>(
66 EditorStatus::Failed, RuleId("editor.preview.canvas-allocation"),
67 "Graphics backend could not allocate an offscreen Canvas");
68 graphics::Canvas* previous = impl_->targets->getCanvas();
69 impl_->targets->setCanvas(canvas);
70 CanvasTargetRestore restore{impl_->targets, previous};
71 canvas->clear(graphics::Color(static_cast<float>(request.clearColor[0]), static_cast<float>(request.clearColor[1]),
72 static_cast<float>(request.clearColor[2]), static_cast<float>(request.clearColor[3])),
73 0, 1.0);
74 EditorResult<void> drawn = [&]() -> EditorResult<void> {
75 try {
76 return draw(impl_->graphics, canvas);
77 } catch (const std::exception& exception) {
78 return eve::editing::failed<void>(EditorStatus::Failed, RuleId("editor.preview.draw-exception"),
79 exception.what());
80 } catch (...) {
81 return eve::editing::failed<void>(EditorStatus::Failed, RuleId("editor.preview.draw-exception"),
82 "Offscreen preview draw callback threw an exception");
83 }
84 }();
85 if (!drawn.ok()) {
86 return EditorResult<OffscreenPreviewArtifact>::failure(drawn.status());
87 }
88 std::unique_ptr<image::ImageData> pixels(canvas->newImageData());
89 if (!pixels)
90 return eve::editing::failed<OffscreenPreviewArtifact>(
91 EditorStatus::Failed, RuleId("editor.preview.readback-failed"),
92 "Graphics backend could not read back the preview Canvas");
93 OffscreenPreviewArtifact metadata;
94 metadata.previewId = request.previewId;
95 metadata.sourceRevision = request.sourceRevision;
96 metadata.width = request.width;
97 metadata.height = request.height;
98 metadata.handle = "preview://" + request.previewId.value() + "/" + std::to_string(request.sourceRevision) + "/" +
99 std::to_string(++impl_->sequence);
100 impl_->artifacts.insert_or_assign(metadata.handle, Impl::Artifact{metadata, std::move(pixels)});
101 return eve::editing::applied<OffscreenPreviewArtifact>(std::move(metadata));
102}

104EditorResult<const image::ImageData*> GraphicsOffscreenPreviewService::image(const std::string& handle,
105 Revision currentRevision) const {
106 const auto found = impl_->artifacts.find(handle);
107 if (found == impl_->artifacts.end())
108 return eve::editing::failed<const image::ImageData*>(EditorStatus::NotFound,
109 RuleId("editor.preview.artifact-not-found"),
110 "Offscreen preview artifact was not found");
111 if (found->second.metadata.sourceRevision != currentRevision)
112 return eve::editing::failed<const image::ImageData*>(
113 EditorStatus::Conflict, RuleId("editor.preview.stale-artifact"),
114 "Offscreen preview artifact belongs to an older source revision");
115 return eve::editing::applied<const image::ImageData*>(found->second.pixels.get());
116}
