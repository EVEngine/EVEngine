37eve::Result<void> AttributeResourceAccountAdapter::validateCost(const eve::resource::CostSpec& cost) const {
38 const char* name = resourceName(kind_);
39 if (name == nullptr)
40 return eve::Result<void>::failure(eve::Diagnostic::error(
41 eve::DiagnosticCode::InvalidArgument, "attribute resource kind is invalid", "resource.kind"));
42 if (!cost.isValid())
43 return eve::Result<void>::failure(eve::Diagnostic::error(
44 eve::DiagnosticCode::InvalidArgument, "attribute resource view requires a validated CostSpec", "cost"));
45 if (cost.size() != 1 || cost.items().front().resource.value() != name)
46 return eve::Result<void>::failure(
47 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
48 "named attribute resource view received a different resource", "cost.resource"));
49 return eve::Result<void>::success();
50}
