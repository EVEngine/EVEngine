57 [[nodiscard]] Result<PrefabLease> spawn(const std::string& uri, const WorldTransform& transform,
58 bool visible = true) {
59 for (std::uint32_t index = 0; index < slots_.size(); ++index) {
60 auto& slot = slots_[index];
61 if (slot.occupied || slot.retired || slot.uri != uri || !allEntitiesAlive(slot)) continue;
62 slot.occupied = true;
63 slot.independent = false;
64 setSlotVisible(slot, visible);
65 apply(slot, transform);
66 return Result<PrefabLease>::success(PrefabLease(index, slot.generation));
67 }
68
69 auto* graphics = ModuleManager::getInstance<graphics::Graphics>("Graphics");
70 auto* models = ModuleManager::getInstance<model3d::Model3D>("Model3D");
71 if (!graphics || !models)
72 return fail<PrefabLease>(DiagnosticCode::NotFound,
73 "Prefab spawn requires Graphics and Model3D modules", "uri");
74 try {
75 auto* model = models->newModelDataFromFile(uri);
76 auto renderables = model3d::buildRenderables(*graphics, model);
77 if (renderables.empty())
78 return fail<PrefabLease>(DiagnosticCode::Failed,
79 "Prefab model contains no renderable meshes", "uri");
80 Slot slot;
81 slot.uri = uri;
82 slot.occupied = true;
83 slot.entities.reserve(renderables.size());
84 for (auto* renderable : renderables) slot.entities.push_back(ecs::handle_of(renderable));
85 slots_.push_back(std::move(slot));
86 const auto index = static_cast<std::uint32_t>(slots_.size() - 1u);
87 apply(slots_.back(), transform);
88 setSlotVisible(slots_.back(), visible);
89 return Result<PrefabLease>::success(PrefabLease(index, slots_.back().generation));
90 } catch (const std::exception& error) {
91 return fail<PrefabLease>(DiagnosticCode::Failed, error.what(), "uri");
92 }
93 }

102 [[nodiscard]] Result<void> recycle(PrefabLease lease) {
103 auto slot = resolve(lease);
104 if (!slot) return Result<void>::failure(slot.status());

106 setSlotVisible(*value, false);
107 value->occupied = false;
108 value->independent = false;

160 [[nodiscard]] Result<Slot*> resolve(PrefabLease lease) {
161 if (!lease.isValid() || lease.index() >= slots_.size())

169 static bool allEntitiesAlive(const Slot& slot) {
170 if (slot.entities.empty()) return false;
171 for (const auto entity : slot.entities)
172 if (!ecs::try_get(entity)) return false;
173 return true;
174 }

176 static void setSlotVisible(Slot& slot, bool visible) {
177 for (const auto entity : slot.entities)

179 renderable->setVisible(visible);

182 static void apply(Slot& slot, const WorldTransform& transform) {
183 for (const auto entity : slot.entities) {
184 auto* renderable = dynamic_cast<graphics::Renderable3D*>(ecs::try_get(entity));
185 if (!renderable) continue;
186 renderable->setPosition(transform.x, transform.y, transform.z);
187 renderable->setRotation(transform.yaw, transform.pitch, transform.roll);
188 renderable->setScale(transform.sx, transform.sy, transform.sz);
189 }

216 Result<void> prepare(const action::ActionPreviewFrame& frame) override {
217 discardPrepared();
218 for (const auto& block : frame.activeBlocks) {
219 if (block.type.format() != "gameplay:prefab-spawn") continue;
220 auto binding = action::ActionPrefabSpawnBinding::fromPayload(block.payload);
221 if (!binding) return Result<void>::failure(binding.status());
222 if (binding.value().lifecycle == action::PrefabSpawnLifecycle::CustomDuration &&
223 block.localTime >= binding.value().customDuration)
224 continue;
225 auto lease = pool_.spawn(binding.value().uri, previewWorldTransform(binding.value().spatial), false);
226 if (!lease) {
227 discardPrepared();
228 return Result<void>::failure(lease.status());
229 }
230 staged_.push_back(lease.value());
231 }
232 return Result<void>::success(Status::success(StatusCode::Applied));
233 }
234
235 void present(const action::ActionPreviewFrame&) noexcept override {
236 recycleAll(current_);
237 current_ = std::move(staged_);
238 staged_.clear();
239 for (const auto lease : current_) {
240 auto visible = pool_.setVisible(lease, true);
241 visible.ignore();
242 }
243 }
244
245 void discardPrepared() noexcept override { recycleAll(staged_); }
246
247private:
248 void recycleAll(std::vector<PrefabLease>& leases) noexcept {
249 for (const auto lease : leases) {
250 auto recycled = pool_.recycle(lease);
251 recycled.ignore();
252 }
253 leases.clear();
254 }

258 std::vector<PrefabLease> staged_;

267 Result<void> handle(const action::ActionTimelineEvent& event,
268 const action::ActionNotifyContext& context) override {
269 const ActiveKey key{context.executionId, event.itemId.format()};
270 if (event.kind == action::ActionTimelineEventKind::StateExit) return exit(key, context);
271 if (event.kind != action::ActionTimelineEventKind::StateEnter)
272 return fail<void>(DiagnosticCode::InvalidArgument,
273 "Prefab spawn requires state enter or exit", "event.kind");
274 if (active_.contains(key))
275 return fail<void>(DiagnosticCode::Conflict, "Prefab spawn state is already active", "itemId");
276 auto binding = action::ActionPrefabSpawnBinding::fromPayload(event.payload);
277 if (!binding) return Result<void>::failure(binding.status());
278 auto pose = resolvePose(binding.value().spatial, context);
279 if (!pose) return Result<void>::failure(pose.status());
280 auto* instances = pool();
281 if (!instances)
282 return fail<void>(DiagnosticCode::NotFound,
283 "SceneLoader prefab instance service is unavailable", "sceneloader");
284 auto lease = instances->spawn(binding.value().uri, worldTransform(binding.value().spatial, pose.value()));
285 if (!lease) return Result<void>::failure(lease.status());
286 auto deadline = context.time.tryAdd(binding.value().customDuration);
287 if (!deadline) {
288 auto recycled = instances->recycle(lease.value());
289 recycled.ignore();
290 return Result<void>::failure(deadline.status());
291 }
292 active_.emplace(key, Active{lease.value(), std::move(binding).takeValue(),
293 std::move(pose).takeValue(), deadline.value()});
294 return Result<void>::success(Status::success(StatusCode::Applied));
295 }
296
297 Result<void> update(const action::ActionActiveBlock& block,
298 const action::ActionNotifyContext& context) override {
299 const auto found = active_.find({context.executionId, block.itemId.format()});
300 if (found == active_.end())
301 return fail<void>(DiagnosticCode::NotFound, "Active prefab state has no instance", "itemId");
302 auto& active = found->second;
303 if (active.recycled) return Result<void>::success(Status::success(StatusCode::NoOp));
304 if (active.binding.lifecycle == action::PrefabSpawnLifecycle::CustomDuration &&
305 context.time >= active.deadline) {
306 auto* instances = pool();
307 if (!instances)
308 return fail<void>(DiagnosticCode::NotFound,
309 "SceneLoader prefab instance service is unavailable", "sceneloader");
310 auto recycled = instances->recycle(active.lease);
311 if (!recycled) return recycled;
312 active.recycled = true;
313 return recycled;
314 }
315 if (active.binding.spatial.mode != action::ActionSpatialAttachmentMode::WorldTransformAtStart) {
316 auto pose = resolvePose(active.binding.spatial, context);
317 if (!pose) return Result<void>::failure(pose.status());
318 if (active.binding.spatial.mode == action::ActionSpatialAttachmentMode::FollowPositionOnly) {
319 active.pose.positionX = pose.value().positionX;
320 active.pose.positionY = pose.value().positionY;
321 active.pose.positionZ = pose.value().positionZ;
322 } else {
323 active.pose = std::move(pose).takeValue();
324 }
325 }
326 auto* instances = pool();
327 if (!instances)
328 return fail<void>(DiagnosticCode::NotFound,
329 "SceneLoader prefab instance service is unavailable", "sceneloader");
330 return instances->update(active.lease, worldTransform(active.binding.spatial, active.pose));
331 }

340 Result<void> advance(const action::ActionNotifyContext& context) override {
341 StatusCode outcome = StatusCode::NoOp;
342 for (std::size_t index = 0; index < timed_.size();) {
343 if (timed_[index].executionId != context.executionId || context.time < timed_[index].deadline) {
344 ++index;
345 continue;
346 }
347 auto* instances = pool();
348 if (!instances)
349 return fail<void>(DiagnosticCode::NotFound,
350 "SceneLoader prefab instance service is unavailable", "sceneloader");
351 auto recycled = instances->recycle(timed_[index].lease);
352 if (!recycled) return recycled;
353 timed_.erase(timed_.begin() + static_cast<std::ptrdiff_t>(index));
354 outcome = StatusCode::Applied;
355 }
356 return Result<void>::success(Status::success(outcome));
357 }

373 Result<void> exit(const ActiveKey& key, const action::ActionNotifyContext& context) {
374 const auto found = active_.find(key);
375 if (found == active_.end()) return Result<void>::success(Status::success(StatusCode::NoOp));
376 if (found->second.recycled) {
377 active_.erase(found);
378 return Result<void>::success(Status::success(StatusCode::NoOp));
379 }
380 if (found->second.binding.lifecycle == action::PrefabSpawnLifecycle::CustomDuration) {
381 timed_.push_back({context.executionId, found->second.lease, found->second.deadline});
382 active_.erase(found);
383 return Result<void>::success(Status::success(StatusCode::Applied));
384 }
385 auto* instances = pool();
386 if (!instances)
387 return fail<void>(DiagnosticCode::NotFound,
388 "SceneLoader prefab instance service is unavailable", "sceneloader");
389 auto result = found->second.binding.lifecycle == action::PrefabSpawnLifecycle::RecycleOnBlockExit
390 ? instances->recycle(found->second.lease)
391 : instances->makeIndependent(found->second.lease);
392 if (!result) return result;
393 active_.erase(found);
394 return result;
395 }

397 static Result<EntitySpatialPose> resolvePose(const action::ActionSpatialBinding& spatial,
398 const action::ActionNotifyContext& context) {
399 std::optional<ecs::EntityHandle> handle;
400 OptionalRef<const IAttachmentPointSource> attachments;
401 if (spatial.target == action::ActionSpatialTarget::Source) {
402 handle = context.source;
403 attachments = context.sourceAttachment;
404 } else {
405 if (spatial.targetIndex >= context.targets.size())
406 return fail<EntitySpatialPose>(DiagnosticCode::NotFound,
407 "Prefab target index is unavailable", "targetIndex");
408 handle = context.targets[spatial.targetIndex];
409 if (spatial.targetIndex < context.targetAttachments.size())
410 attachments = context.targetAttachments[spatial.targetIndex];
411 }
412 EntitySpatialPose pose;
413 if (handle) {
414 auto root = resolveEntitySpatialPose(*handle);
415 if (!root) return root;
416 pose = std::move(root).takeValue();
417 }
418 if (spatial.bone.empty()) return Result<EntitySpatialPose>::success(std::move(pose));
419 if (!attachments)
420 return fail<EntitySpatialPose>(DiagnosticCode::Unsupported,
421 "Prefab bone requires an attachment source", "bone");
422 auto point = attachments->get().sampleAttachmentPoint(
423 spatial.bone, {static_cast<float>(spatial.positionOffset.x),
424 static_cast<float>(spatial.positionOffset.y),
425 static_cast<float>(spatial.positionOffset.z)});
426 if (!point) return Result<EntitySpatialPose>::failure(point.status());
427 pose.positionX = point.value().x;
428 pose.positionY = point.value().y;
429 pose.positionZ = point.value().z;
430 return Result<EntitySpatialPose>::success(std::move(pose));
431 }

433 static WorldTransform worldTransform(const action::ActionSpatialBinding& spatial,
434 const EntitySpatialPose& pose) {
435 const bool bone = !spatial.bone.empty();
436 return {
437 static_cast<float>(pose.positionX + (bone ? 0.0 : spatial.positionOffset.x)),
438 static_cast<float>(pose.positionY + (bone ? 0.0 : spatial.positionOffset.y)),
439 static_cast<float>(pose.positionZ + (bone ? 0.0 : spatial.positionOffset.z)),
440 static_cast<float>((pose.rotationYDegrees + spatial.rotationOffsetDegrees.y) * kDegreesToRadians),
441 static_cast<float>((pose.rotationXDegrees + spatial.rotationOffsetDegrees.x) * kDegreesToRadians),
442 static_cast<float>((pose.rotationZDegrees + spatial.rotationOffsetDegrees.z) * kDegreesToRadians),
443 static_cast<float>(pose.scaleX * spatial.scale.x),
444 static_cast<float>(pose.scaleY * spatial.scale.y),
445 static_cast<float>(pose.scaleZ * spatial.scale.z),
446 };
