13 return [jobs](const std::vector<uint32_t> &layerPasses,
14 const std::function<void(uint32_t)> &recordOne) {
15 if (!jobs || layerPasses.size() <= 1) {
16 for (uint32_t order : layerPasses) recordOne(order);
17 return;
18 }
19 auto *group = jobs->createFrameTaskGroup();
20 for (uint32_t order : layerPasses)
21 group->fork([recordOne, order] { recordOne(order); });
22 // Join without help-execution: TaskGroup::wait() help-executes on the
23 // waiting thread, and that path races with the arena lifecycle in
24 // beginFrame/endFrame (a help-run job's completion can observe
25 // outstandingFrame == 0 and reset the per-frame arena while workers
26 // are still finishing other jobs). Polling the group's pending count
27 // keeps workers as the sole executors; every forked job still drains.
28 // (Follow-up: fix the JobSystem help-execution race upstream so this
29 // can return to a blocking join.)
30 while (group->getPendingCount() > 0)
31 std::this_thread::sleep_for(std::chrono::microseconds(50));
32 // Arena-owned group; never deleted.
33 };
34}
