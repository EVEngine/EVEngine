34Result<std::size_t> ParticleBakePresentation::draw(BakedParticlePlayback& player, graphics::Graphics& graphics,
35 std::span<const ParticleBakeMaterial> materials) {
36 if (materials.size() > 65536)
37 return Result<std::size_t>::failure(Diagnostic::error(
38 DiagnosticCode::InvalidArgument, "particle material budget exceeded", "particleBakePresentation"));
39 for (std::size_t i = 0; i < materials.size(); ++i)
40 if (materials[i].emitterSession.isNil() ||
41 (i && !(materials[i - 1].emitterSession < materials[i].emitterSession)))
42 return Result<std::size_t>::failure(Diagnostic::error(
43 DiagnosticCode::InvalidArgument, "particle materials must have sorted unique session identities",
44 "particleBakePresentation"));
45 return drawImpl(player, graphics, nullptr, materials);
46}
47Result<std::size_t> ParticleBakePresentation::drawImpl(BakedParticlePlayback& player, graphics::Graphics& graphics,
48 const graphics::DrawItem2D* uniformMaterial,
49 std::span<const ParticleBakeMaterial> materials) {
50 if (!impl_->track.isSameAsset(player.track()))
51 return Result<std::size_t>::failure(
52 Diagnostic::error(DiagnosticCode::InvalidArgument, "foreign particle player", "particleBakePresentation"));
53 auto sampled = player.sample(impl_->samples);
54 if (!sampled) return Result<std::size_t>::failure(sampled.status());
55 const auto count = sampled.value();
56 // Clear borrowed resource references even if the renderer throws; no material survives this call.
57 struct Clear {
58 std::vector<graphics::DrawItem2D>& items;
59 ~Clear() { items.clear(); }
60 } clear{impl_->items};
61 bool usedCanvas = false;
62 for (std::size_t i = 0; i < count; ++i) {
63 const auto& pose = impl_->samples[i].pose;
64 const graphics::DrawItem2D* selected = uniformMaterial;
65 if (!selected) {
66 const auto id = impl_->samples[i].emitter;
67 auto found = std::lower_bound(
68 materials.begin(), materials.end(), id,
69 [](const ParticleBakeMaterial& binding, PersistentId value) { return binding.emitterSession < value; });
70 if (found == materials.end() || found->emitterSession != id)
71 return Result<std::size_t>::failure(
72 Diagnostic::error(DiagnosticCode::NotFound, "visible particle emitter has no material binding",
73 "particleBakePresentation"));
74 selected = &found->material;
75 }
76 auto item = *selected;
77 usedCanvas = usedCanvas || item.canvas != nullptr;
78 item.x = pose.x;
79 item.y = pose.y;
80 item.w = pose.w;
81 item.h = pose.h;
82 item.rotation = pose.rotation;
83 item.color = {pose.r, pose.g, pose.b, pose.a};
84 item.u0 = pose.u0;
85 item.v0 = pose.v0;
86 item.u1 = pose.u1;
87 item.v1 = pose.v1;
88 item.hasUV = true;
89 item.rotatedUV = false;
90 item.anchorX = item.anchorY = 0.5f;
91 item.flipX = item.flipY = false;
92 item.order = static_cast<int>(i);
93 item.hasOrder = true;
94 item.quad = nullptr;
95 impl_->items.push_back(item);
96 }
97 if (!impl_->items.empty()) graphics::RenderSystem::drawItems(graphics, impl_->items, false);
98 if (usedCanvas) graphics.setCanvas();
99 return Result<std::size_t>::success(count);
100}
