14CardState parseCardState(const std::string& name) {

43bool captureCardState(StateValue& out) {
44 StateValue cards = StateValue::array();
45 auto view = ecs::View<CardData, CardData::Identity, CardData::State>();
46 for (auto it = view.begin(); it != view.end(); ++it) {
47 auto [id, st] = *it;
48 StateValue item = StateValue::object();
49 item.set("id", StateValue::string(id->id));
50 item.set("phase", StateValue::string(cardStateName(st->phase)));
51 cards.pushBack(std::move(item));
52 }
53 out = StateValue::object();
54 out.set("cards", std::move(cards));
55 return true;
56}

58bool restoreCardState(const StateValue& in, std::string* err) {
59 if (!in.isObject()) {
60 if (err) *err = "card: state is not an object";
61 return false;
62 }
63 const StateValue* cards = in.find("cards");
64 if (!cards || !cards->isArray()) {
65 if (err) *err = "card: missing cards";
66 return false;
67 }
68
69 auto view = ecs::View<CardData, CardData::Identity, CardData::State>();
70 for (auto it = view.begin(); it != view.end(); ++it) {
71 auto [id, st] = *it;
72 st->hovered = false;
73 st->dragging = false;
74 for (size_t i = 0; i < cards->arraySize(); ++i) {
75 const StateValue& item = cards->at(i);
76 const StateValue* cid = item.find("id");
77 const StateValue* phase = item.find("phase");
78 if (cid && phase && cid->isString() && phase->isString() && cid->asString() == id->id) {
79 st->phase = parseCardState(phase->asString());
80 break;
81 }
82 }
83 }
84
85 // In-flight drag is transient: dropped on reload.
86 auto hands = ecs::View<Hand, Hand::Drag>();
87 for (auto it = hands.begin(); it != hands.end(); ++it) {
88 auto [drag] = *it;
89 drag->card = nullptr;
90 drag->press = nullptr;
91 drag->wasDown = false;
92 }
93 return true;
94}
