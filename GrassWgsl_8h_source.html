88fn cascadeBias(cascade: i32) -> f32 {
89 var value = shadow.cascadeBias.z;
90 if (cascade == 0) { value = shadow.cascadeBias.x; }
91 if (cascade == 1) { value = shadow.cascadeBias.y; }
92 return select(shadow.bias.x, value, value > 0.00000001);
93}
94fn slopeScaledBias(cascade: i32, nDotL: f32) -> f32 {
95 return cascadeBias(cascade) * mix(0.75, 1.0, clamp(nDotL, 0.0, 1.0));
96}
97fn sampleCascade(worldPos: vec3f, cascade: i32, bias: f32) -> f32 {
98 let clip = shadow.lightVP[cascade] * vec4f(worldPos, 1.0);
99 let ndc = clip.xyz / max(clip.w, 0.000001);
100 // The WebGPU shadow writer mirrors clip Y; sampling must mirror it too.
101 let uv = vec2f(ndc.x, -ndc.y) * 0.5 + vec2f(0.5);
102 if (any(uv < vec2f(0.0)) || any(uv > vec2f(1.0)) || ndc.z < 0.0 || ndc.z > 1.0) {
103 return 1.0;
104 }
105 var sum = 0.0;
106 let offsets = array<vec2f, 9>(vec2f(-0.5,-0.5),vec2f(0.0,-0.6),vec2f(0.5,-0.5),
107 vec2f(-0.6,0.0),vec2f(0.0),vec2f(0.6,0.0),vec2f(-0.5,0.5),vec2f(0.0,0.6),vec2f(0.5,0.5));
108 let dims = vec2f(textureDimensions(shadowMap));
109 for (var i = 0; i < 9; i++) {
110 let sampleUv = uv + offsets[i] / dims;
111 if (all(sampleUv >= vec2f(0.0)) && all(sampleUv <= vec2f(1.0))) {
112 sum += textureSampleCompareLevel(shadowMap, shadowSampler, sampleUv, cascade,
113 ndc.z - bias);
114 }
115 }
116 return sum / 9.0;
117}
118fn grassShadow(worldPos: vec3f, viewDepth: f32) -> f32 {
119 if (shadow.bias.y < 0.5 || shadow.bias.z < 0.5 || shadow.splits.w < 0.0001) { return 1.0; }
120 var cascade = 2;
121 if (viewDepth < shadow.splits.x) { cascade = 0; }
122 else if (viewDepth < shadow.splits.y) { cascade = 1; }
123 var texel = shadow.cascadeTexel.z;
124 if (cascade == 0) { texel = shadow.cascadeTexel.x; }
125 else if (cascade == 1) { texel = shadow.cascadeTexel.y; }
126 let normal = vec3f(0.0, 1.0, 0.0);
127 let nDotL = max(dot(normal, normalize(frame.lightDir.xyz)), 0.0);
128 let samplePos = worldPos + normal * ((2.0 * max(texel, 0.000001)) / max(nDotL, 0.2));
129 var visibility = sampleCascade(samplePos, cascade, slopeScaledBias(cascade, nDotL));
130 var hi = shadow.splits.z;
131 var lo = shadow.splits.y;
132 if (cascade == 0) { hi = shadow.splits.x; lo = 0.0; }
133 else if (cascade == 1) { hi = shadow.splits.y; lo = shadow.splits.x; }
134 let band = max(0.5, (hi - lo) * 0.1);
135 let toPrev = 1.0 - clamp((viewDepth - lo) / band, 0.0, 1.0);
136 let toNext = 1.0 - clamp((hi - viewDepth) / band, 0.0, 1.0);
137 if (toPrev > 0.0 && cascade > 0) {
138 visibility = mix(visibility, sampleCascade(samplePos, cascade - 1,
139 slopeScaledBias(cascade - 1, nDotL)), toPrev);
140 }
141 if (toNext > 0.0 && cascade < 2) {
142 visibility = mix(visibility, sampleCascade(samplePos, cascade + 1,
143 slopeScaledBias(cascade + 1, nDotL)), toNext);
144 }
145 visibility = mix(1.0, visibility, clamp(shadow.splits.w, 0.0, 1.0));
146 return mix(0.04, 1.0, visibility);
147}

149fn fs_main(in: FSIn) -> @location(0) vec4f {
150 let frames = max(p(12u), 1.0); let duration = max(p(1u), 0.0001);
151 let random = fract(sin(in.instanceId * 12.9898) * 43758.5453);
152 let frameNumber = i32(p(0u) / duration + random * frames) % i32(frames);
153 let cols = i32(max(p(13u), 1.0)); let rows = i32(max(p(14u), 1.0));
154 let leaf = in.alwaysDark > 0.5 || p(5u) > 0.5;
155 let variants = select(i32(max(p(15u), 1.0)), i32(max(p(16u), 1.0)), leaf);
156 let variant = i32(in.instanceId) % variants;
157 var atlasUv: vec2f;
158 if (rows <= 1) {
159 atlasUv = vec2f((in.uv.x + f32(frameNumber)) / frames, 1.0 - in.uv.y);
160 } else {
161 let column = variant * 2 + frameNumber % 2;
162 let row = select(0, i32(p(17u) + 0.5), leaf) + frameNumber / 2;
163 atlasUv = vec2f((f32(column) + in.uv.x) / f32(cols),
164 (f32(row) + 1.0 - in.uv.y) / f32(rows));
165 }
166 let mask = textureSample(albedo, mainSampler, atlasUv) * in.tint;
167 if (mask.a < max(p(4u), 0.01)) { discard; }
168 // Dawn's depth precision self-shadows roots that lie exactly on the ground
169 // plane. Sample the visible blade point; the explicit normal offset in
170 // grassShadow still preserves occluder shadows without blanket acne.
171 var visibility = grassShadow(in.worldPos, max(-in.viewPos.z, 0.0));
172 if (p(5u) > 0.5 || in.alwaysDark > 0.5) { visibility = 0.0; }
173 let color = mix(vec3f(p(9u), p(10u), p(11u)), vec3f(p(6u), p(7u), p(8u)), visibility);
174 return vec4f(color * mask.rgb, mask.a);
175}
