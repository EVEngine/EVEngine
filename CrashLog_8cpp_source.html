78void recordCrashEvent(const std::string& report) {
79 std::unique_lock<std::mutex> lock(g_logMutex, std::try_to_lock);
80 if (!lock.owns_lock()) return;
81 ensureOpenLocked();
82 if (!g_open || !g_logFile) return;
83 g_logFile << "[" << nowStamp() << "] crash | ";
84 g_logFile.write(report.data(), static_cast<std::streamsize>(report.size()));
85 if (report.empty() || report.back() != '\n') g_logFile << "\n";
