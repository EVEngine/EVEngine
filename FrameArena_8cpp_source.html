35bool FrameArena::ensure(vkb::Device &device, vk::DeviceSize bytes, vk::BufferUsageFlags usage) {
36 if (capacity_ >= bytes) return true;
37 if (!device.instance) return false;
38
39 const vk::DeviceSize grow = (capacity_ > 0) ? capacity_ : 1;
40 vk::DeviceSize target = bytes;
41 while (target < bytes) target += grow; // grow at least by current capacity
42 target = std::max<vk::DeviceSize>(target, 1 << 20);
43
44 if (mapped_ && device_.instance) buf_.unmap();
45 mapped_ = nullptr;
46 vkb::GenericBuffer next(device, usage, target,
47 vk::MemoryPropertyFlagBits::eHostVisible | vk::MemoryPropertyFlagBits::eHostCoherent);
48 buf_ = std::move(next);
49 device_ = device;
50 capacity_ = target;
51 head_ = 0;
52 mapped_ = buf_.map();
53 return true;
54}

56FrameArena::Alloc FrameArena::alloc(vk::DeviceSize bytes, vk::DeviceSize align) {
57 Alloc out{};
58 if (bytes == 0) return out;
59 const vk::DeviceSize aligned = ((head_ + align - 1) / align) * align;
60 if (aligned + bytes > capacity_) return out; // overflow: caller truncates + counts
61 head_ = aligned + bytes;
62 out.offset = aligned;
63 out.size = bytes;
64 out.mapped = mapped_ ? static_cast<char *>(mapped_) + aligned : nullptr;
65 return out;
66}

77vkb::PassRecordExecutor jobSystemPassExecutor(eve::thread::JobSystem *jobs) {
78 return [jobs](const std::vector<uint32_t> &layerPasses, const std::function<void(uint32_t)> &recordOne) {
79 if (!jobs || layerPasses.size() <= 1) {
80 for (uint32_t order : layerPasses) recordOne(order);
81 return;
82 }
83 auto *group = jobs->createFrameTaskGroup();
84 for (uint32_t order : layerPasses) group->fork([recordOne, order] { recordOne(order); });
85 // Join without help-execution: TaskGroup::wait() help-executes on the
86 // waiting thread, and that path races with the arena lifecycle in
87 // beginFrame/endFrame (a help-run job's completion can observe
88 // outstandingFrame == 0 and reset the per-frame arena while workers
89 // are still finishing other jobs). Polling the group's pending count
90 // keeps workers as the sole executors; every forked job still drains.
91 // (Follow-up: fix the JobSystem help-execution race upstream so this
92 // can return to a blocking join.)
93 while (group->getPendingCount() > 0) std::this_thread::sleep_for(std::chrono::microseconds(50));
94 // Arena-owned group; never deleted.
95 };
96}
