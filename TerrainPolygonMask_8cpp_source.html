44Result<int> TerrainPolygonMask::rasterize(Heightmap& target, const Heightmap& brush, const TerrainSampleGrid& grid,
45 TerrainPolygonMaskType type) const {
46 if (!impl_ || !validRaster(target) || !validRaster(brush) || target.getWidth() != grid.width ||
47 target.getHeight() != grid.height || !std::isfinite(grid.originX) || !std::isfinite(grid.originZ) ||
48 !std::isfinite(grid.spacingX) || !std::isfinite(grid.spacingZ) || grid.spacingX <= 0 || grid.spacingZ <= 0 ||
49 type < TerrainPolygonMaskType::Open || type > TerrainPolygonMaskType::Closed)
50 return invalid("terrain.polyMask.rasterize: finite compatible rasters, grid and topology required");
51
52 auto inside = [&](double x, double z) {
53 bool hit = false;
54 for (size_t i = 0, j = impl_->nodes.size() - 1; i < impl_->nodes.size(); j = i++) {
55 const auto& a = impl_->nodes[i];
56 const auto& b = impl_->nodes[j];
57 if ((a.z > z) != (b.z > z) && x < (b.x - a.x) * (z - a.z) / (b.z - a.z) + a.x) hit = !hit;
58 }
59 return hit;
60 };
61
62 std::vector<float> output(target.data().size(), 0.0F);
63 for (int z = 0; z < target.getHeight(); ++z)
64 for (int x = 0; x < target.getWidth(); ++x) {
65 const double wx = grid.originX + x * grid.spacingX;
66 const double wz = grid.originZ + z * grid.spacingZ;
67 double value = type == TerrainPolygonMaskType::Closed && impl_->nodes.size() > 2 && inside(wx, wz) ? 1 : 0;
68 for (const auto& node : impl_->nodes) {
69 const double u = (wx - (node.x - node.radius)) / (2 * node.radius);
70 const double v = (wz - (node.z - node.radius)) / (2 * node.radius);
71 if (u >= 0 && u <= 1 && v >= 0 && v <= 1) value += textureSample(brush, u, v);
72 value = std::min(value, 1.0);
73 if (value < 0.25) value = 0;
74 }
75 output[size_t(z) * target.getWidth() + x] = static_cast<float>(value);
76 }
77 return publish(target, std::move(output));
78}
