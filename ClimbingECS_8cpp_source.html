16eve::Result<void> ClimbingCandidateBuffer::replace(std::span<const ClimbingCandidate> candidates,
17 eve::SimulationTick tick) {
18 if (candidates.size() > Capacity)
19 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
20 "climbing candidate buffer capacity exceeded", "candidates");
21 ClimbingCandidateSet replacement;
22 for (const ClimbingCandidate& candidate : candidates) replacement.consider(candidate);
23 replacement.sortAndLimit(Capacity);
24 values_.swap(replacement);
25 tick_ = tick;
