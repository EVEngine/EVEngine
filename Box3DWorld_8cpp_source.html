20constexpr uint64_t combineModeMask = 0x7ull;
21constexpr int frictionModeShift = 32;
22constexpr int restitutionModeShift = 35;

24float combineMaterialValue(float a, uint64_t materialA, float b, uint64_t materialB, int shift, bool friction) {
25 const uint64_t modeA = (materialA >> shift) & combineModeMask;
26 const uint64_t modeB = (materialB >> shift) & combineModeMask;
27 const uint64_t mode = std::max(modeA, modeB);
28 switch (mode) {
29 case 1: return 0.5f * (a + b);
30 case 2: return std::min(a, b);
31 case 3: return a * b;
32 case 4: return std::max(a, b);
33 default: return friction ? std::sqrt(a * b) : std::max(a, b);
34 }
35}

37float mixFriction(float a, uint64_t materialA, float b, uint64_t materialB) {
38 return combineMaterialValue(a, materialA, b, materialB, frictionModeShift, true);

45uint64_t stablePairKey(int idA, int idB) {
46 const uint32_t low = static_cast<uint32_t>(std::min(idA, idB));
47 const uint32_t high = static_cast<uint32_t>(std::max(idA, idB));
48 return (static_cast<uint64_t>(high) << 32) | low;
49}

258int WorldBackend::do_explode(float x, float y, float z, float radius, float falloff, float impulsePerArea,
259 int maskBits) {
260 owner_->explosionResults_.clear();
261 if (!std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z))
262 throw eve::Exception("World3D.explode: center must be finite");
263 if (!std::isfinite(radius) || radius < 0.f || !std::isfinite(falloff) || falloff < 0.f)
264 throw eve::Exception("World3D.explode: radius and falloff must be finite and >= 0");
265 if (!std::isfinite(impulsePerArea)) throw eve::Exception("World3D.explode: impulsePerArea must be finite");
266 if (!isValid() || impulsePerArea == 0.f) return 0;
267
268 struct Before {
269 Body3D *body = nullptr;
270 b3Vec3 linear{};
271 b3Vec3 angular{};
272 };
273 std::vector<Before> before;
274 before.reserve(owner_->bodies_.size());
275 for (Body3D *body : owner_->bodies_) {
276 if (!body || !body->isValid() || b3Body_GetType(box3d::Access::bodyId(*body)) != b3_dynamicBody) continue;
277 before.push_back({body, b3Body_GetLinearVelocity(box3d::Access::bodyId(*body)),
278 b3Body_GetAngularVelocity(box3d::Access::bodyId(*body))});
279 }
280
281 b3ExplosionDef def = b3DefaultExplosionDef();
282 def.position = b3Pos{x, y, z};
283 def.radius = radius;
284 def.falloff = falloff;
285 def.impulsePerArea = impulsePerArea;
286 def.maskBits = static_cast<uint32_t>(maskBits);
287 b3World_Explode(worldId_, &def);
288
289 for (const Before &entry : before) {
290 if (!entry.body->isValid()) continue;
291 const b3Vec3 linear = b3Body_GetLinearVelocity(box3d::Access::bodyId(*entry.body));
292 const b3Vec3 angular = b3Body_GetAngularVelocity(box3d::Access::bodyId(*entry.body));
293 const b3Vec3 dv = linear - entry.linear;
294 const b3Vec3 dw = angular - entry.angular;
295 if (b3LengthSquared(dv) <= 1e-12f && b3LengthSquared(dw) <= 1e-12f) continue;
296 owner_->explosionResults_.push_back({entry.body->getId(), dv.x, dv.y, dv.z, dw.x, dw.y, dw.z});
297 }
298 std::sort(owner_->explosionResults_.begin(), owner_->explosionResults_.end(),
299 [](const ExplosionResult &a, const ExplosionResult &b) { return a.bodyId < b.bodyId; });
300 return static_cast<int>(owner_->explosionResults_.size());
301}

537bool WorldBackend::do_customFilterCallback(b3ShapeId shapeIdA, b3ShapeId shapeIdB, void *context) {
538 auto *world = static_cast<WorldBackend *>(context);
539 Shape3D *shapeA = shapeFromShape(shapeIdA);
540 Shape3D *shapeB = shapeFromShape(shapeIdB);
541 Body3D *bodyA = bodyFromShape(shapeIdA);
542 Body3D *bodyB = bodyFromShape(shapeIdB);
543 if (!world || !shapeA || !shapeB || !bodyA || !bodyB) return false;

545 return world->owner_->disabledShapePairs_.count(stablePairKey(shapeA->getId(), shapeB->getId())) == 0;
546}

562bool WorldBackend::do_preSolveCallback(b3ShapeId shapeIdA, b3ShapeId shapeIdB, b3Pos point, b3Vec3 normal,
563 void *context) {
564 auto *world = static_cast<WorldBackend *>(context);
565 if (!world) return true;
566 auto findShape = [world](b3ShapeId id) -> Shape3D * {
567 const auto raw = world->owner_->shapeRawHandles_.find(b3StoreShapeId(id));
568 if (raw == world->owner_->shapeRawHandles_.end()) return nullptr;
569 const auto found = world->owner_->shapeHandles_.find(raw->second);
570 return found == world->owner_->shapeHandles_.end() ? nullptr : found->second;
571 };
572 Shape3D *shapeA = findShape(shapeIdA);
573 Shape3D *shapeB = findShape(shapeIdB);
574 if (shapeA && shapeA->isOneWay() &&
575 !shapeA->allowsOneWayContact(point.x, point.y, point.z, normal.x, normal.y, normal.z))
576 return false;
577 if (shapeB && shapeB->isOneWay() &&
578 !shapeB->allowsOneWayContact(point.x, point.y, point.z, -normal.x, -normal.y, -normal.z))
579 return false;
580 return true;
581}
