38uint32_t readBe32(const uint8_t* data) {
39 return (uint32_t(data[0]) << 24) | (uint32_t(data[1]) << 16) | (uint32_t(data[2]) << 8) | uint32_t(data[3]);
40}

42void be32(std::vector<uint8_t>& out, uint32_t value) {
43 for (int shift = 24; shift >= 0; shift -= 8) out.push_back(static_cast<uint8_t>(value >> shift));
44}

46void pngChunk(std::vector<uint8_t>& out, const char* type, const std::vector<uint8_t>& bytes) {
47 be32(out, static_cast<uint32_t>(bytes.size()));
48 const size_t start = out.size();
49 out.insert(out.end(), type, type + 4);
50 out.insert(out.end(), bytes.begin(), bytes.end());
51 Poco::Checksum checksum(Poco::Checksum::TYPE_CRC32);
52 checksum.update(reinterpret_cast<const char*>(out.data() + start), static_cast<unsigned>(out.size() - start));
53 be32(out, checksum.checksum());
54}

56Result<std::vector<uint8_t>> encodeRgbaPng(uint32_t width, uint32_t height,
57 const std::shared_ptr<const std::vector<uint8_t>>& rgba) {
58 if (!rgba || width == 0 || height == 0 || rgba->size() != size_t(width) * height * 4)
59 return fail<std::vector<uint8_t>>(DiagnosticCode::InvalidArgument, "invalid RGBA for PNG encode");
60 // Dedicated encoder (not frameCapturePreview): preview path forces alpha=255.
61 std::vector<uint8_t> scanlines((size_t(width) * 4 + 1) * height);
62 for (uint32_t y = 0; y < height; ++y) {
63 const size_t row = size_t(y) * (size_t(width) * 4 + 1);
64 scanlines[row] = 0; // filter None
65 std::memcpy(scanlines.data() + row + 1, rgba->data() + size_t(y) * width * 4, size_t(width) * 4);
66 }
67 std::ostringstream compressed;
68 Poco::DeflatingOutputStream zlib(compressed, Poco::DeflatingStreamBuf::STREAM_ZLIB);
69 zlib.write(reinterpret_cast<const char*>(scanlines.data()), static_cast<std::streamsize>(scanlines.size()));
70 zlib.close();
71 const auto data = compressed.str();
72 std::vector<uint8_t> png = {137, 80, 78, 71, 13, 10, 26, 10};
73 std::vector<uint8_t> header;
74 be32(header, width);
75 be32(header, height);
76 header.insert(header.end(), {8, 6, 0, 0, 0});
77 pngChunk(png, "IHDR", header);
78 pngChunk(png, "IDAT", std::vector<uint8_t>(data.begin(), data.end()));
79 pngChunk(png, "IEND", {});
80 return Result<std::vector<uint8_t>>::success(std::move(png));
81}

91Result<ActionGuideCandidateFrame> decodeActionGuideCandidatePng(std::span<const std::uint8_t> pngBytes) {
92 static const uint8_t kSig[8] = {137, 80, 78, 71, 13, 10, 26, 10};
93 if (pngBytes.size() < 8 || std::memcmp(pngBytes.data(), kSig, 8) != 0)
94 return fail<ActionGuideCandidateFrame>(DiagnosticCode::ParseError, "not a PNG");
95 size_t offset = 8;
96 uint32_t width = 0, height = 0;
97 bool sawIhdr = false, sawIend = false;
98 std::vector<uint8_t> idat;
99 while (offset + 12 <= pngBytes.size()) {
100 const uint32_t length = readBe32(pngBytes.data() + offset);
101 if (offset + 12 + length > pngBytes.size())
102 return fail<ActionGuideCandidateFrame>(DiagnosticCode::ParseError, "truncated PNG chunk");
103 const char* type = reinterpret_cast<const char*>(pngBytes.data() + offset + 4);
104 const auto* data = pngBytes.data() + offset + 8;
105 Poco::Checksum crc(Poco::Checksum::TYPE_CRC32);
106 crc.update(reinterpret_cast<const char*>(pngBytes.data() + offset + 4), static_cast<unsigned>(length + 4));
107 const uint32_t expected = readBe32(pngBytes.data() + offset + 8 + length);
108 if (crc.checksum() != expected)
109 return fail<ActionGuideCandidateFrame>(DiagnosticCode::ParseError, "PNG chunk CRC mismatch");
110 if (std::strncmp(type, "IHDR", 4) == 0) {
111 if (length != 13 || sawIhdr)
112 return fail<ActionGuideCandidateFrame>(DiagnosticCode::ParseError, "invalid IHDR");
113 width = readBe32(data);
114 height = readBe32(data + 4);
115 const auto bitDepth = data[8];
116 const auto colorType = data[9];
117 const auto interlace = data[12];
118 if (bitDepth != 8 || colorType != 6 || interlace != 0 || width == 0 || height == 0 || width > 4096 ||
119 height > 4096)
120 return fail<ActionGuideCandidateFrame>(DiagnosticCode::Unsupported,
121 "only non-interlaced 8-bit RGBA PNG is supported");
122 sawIhdr = true;
123 } else if (std::strncmp(type, "IDAT", 4) == 0) {
124 idat.insert(idat.end(), data, data + length);
125 } else if (std::strncmp(type, "IEND", 4) == 0) {
126 sawIend = true;
127 offset += 12 + length;
128 break;
129 }
130 offset += 12 + length;
131 }
132 if (!sawIhdr || !sawIend || idat.empty())
133 return fail<ActionGuideCandidateFrame>(DiagnosticCode::ParseError, "PNG missing IHDR/IDAT/IEND");
134
135 std::string compressed(reinterpret_cast<const char*>(idat.data()), idat.size());
136 std::istringstream input(compressed);
137 Poco::InflatingInputStream zlib(input, Poco::InflatingStreamBuf::STREAM_ZLIB);
138 std::ostringstream inflated;
139 inflated << zlib.rdbuf();
140 const auto raw = inflated.str();
141 const size_t stride = size_t(width) * 4 + 1;
142 if (raw.size() != stride * height)
143 return fail<ActionGuideCandidateFrame>(DiagnosticCode::ParseError, "PNG inflated size mismatch");
144
145 auto rgba = std::make_shared<std::vector<uint8_t>>(size_t(width) * height * 4);
146 for (uint32_t y = 0; y < height; ++y) {
147 const auto* row = reinterpret_cast<const uint8_t*>(raw.data() + size_t(y) * stride);
148 const auto filter = row[0];
149 if (filter != 0)
150 return fail<ActionGuideCandidateFrame>(DiagnosticCode::Unsupported,
151 "PNG filter methods other than None are unsupported");
152 std::memcpy(rgba->data() + size_t(y) * width * 4, row + 1, size_t(width) * 4);
153 }
154 ActionGuideCandidateFrame frame;
155 frame.width = width;
156 frame.height = height;
157 frame.rgba8 = std::move(rgba);
158 return Result<ActionGuideCandidateFrame>::success(std::move(frame));
159}

161Result<std::vector<ActionGuideCandidateFrame>> loadActionGuideCandidates(const std::filesystem::path& candidateDir,
162 const ActionGuidePack& pack) {
163 if (pack.shots.empty())
164 return fail<std::vector<ActionGuideCandidateFrame>>(DiagnosticCode::InvalidArgument, "pack has no shots");
165 if (!std::filesystem::is_directory(candidateDir))
166 return fail<std::vector<ActionGuideCandidateFrame>>(DiagnosticCode::NotFound, "candidateDir is missing",
167 candidateDir.generic_string());
168 std::vector<ActionGuideCandidateFrame> frames;
169 frames.reserve(pack.shots.size());
170 try {
171 for (size_t i = 0; i < pack.shots.size(); ++i) {
172 const auto folder = candidateDir / padFolder(i + 1);
173 const auto metaPath = folder / "meta.json";
174 const auto pngPath = folder / "frame.png";
175 if (!std::filesystem::is_regular_file(metaPath) || !std::filesystem::is_regular_file(pngPath))
176 return fail<std::vector<ActionGuideCandidateFrame>>(
177 DiagnosticCode::NotFound, "missing meta.json or frame.png", folder.generic_string());
178 const auto metaBytes = readAllBytes(metaPath);
179 auto metaJson =
180 Value::fromJson(std::string(reinterpret_cast<const char*>(metaBytes.data()), metaBytes.size()));
181 if (!metaJson) return Result<std::vector<ActionGuideCandidateFrame>>::failure(metaJson.status());
182 auto meta = decodeActionGuideCandidateMeta(metaJson.value());
183 if (!meta) return Result<std::vector<ActionGuideCandidateFrame>>::failure(meta.status());
184 const auto pngBytes = readAllBytes(pngPath);
185 auto frame = decodeActionGuideCandidatePng(pngBytes);
186 if (!frame) return Result<std::vector<ActionGuideCandidateFrame>>::failure(frame.status());
187 frame.value().meta = std::move(meta).takeValue();
188 frames.push_back(std::move(frame).takeValue());
189 }
190 } catch (const std::exception& e) {
191 return fail<std::vector<ActionGuideCandidateFrame>>(DiagnosticCode::Failed,
192 std::string("candidate load failed: ") + e.what());
193 }
194 return Result<std::vector<ActionGuideCandidateFrame>>::success(std::move(frames));
195}

197Result<Value> writeActionGuideIngestPack(const ActionGuideIngestDraft& draft,
198 std::span<const ActionGuideCandidateFrame> frames,
199 const std::filesystem::path& outputRoot) {
200 if (frames.empty() || static_cast<std::int64_t>(frames.size()) != draft.frameCount)
201 return fail<Value>(DiagnosticCode::InvalidArgument, "draft frameCount does not match frames");
202 const auto staging = outputRoot / ".staging";
203 try {
204 std::filesystem::remove_all(staging);
205 std::filesystem::create_directories(staging / "frames");
206 for (size_t i = 0; i < frames.size(); ++i) {
207 auto png = encodeRgbaPng(frames[i].width, frames[i].height, frames[i].rgba8);
208 if (!png) {
209 std::filesystem::remove_all(staging);
210 return Result<Value>::failure(png.status());
211 }
212 writeBytes(staging / ("frames/frame_" + std::to_string(i) + ".png"),
213 reinterpret_cast<const char*>(png.value().data()), png.value().size());
214 }
215 auto writeJson = [&](const char* name, const Value& value) -> Result<void> {
216 auto json = value.toJson();
217 if (!json) return Result<void>::failure(json.status());
218 writeBytes(staging / name, json.value().data(), json.value().size());
219 return Result<void>::success();
220 };
221 if (auto r = writeJson("clip.json", draft.clipDescriptor); !r) {
222 std::filesystem::remove_all(staging);
223 return Result<Value>::failure(r.status());
224 }
225 if (auto r = writeJson("ability.draft.json", draft.abilityDraft); !r) {
226 std::filesystem::remove_all(staging);
227 return Result<Value>::failure(r.status());
228 }
229 if (auto r = writeJson("attribution.gen.json", draft.attribution); !r) {
230 std::filesystem::remove_all(staging);
231 return Result<Value>::failure(r.status());
232 }
233 Value manifest =
234 Value::object({{"schemaId", Value("eve.action.guide-ingest.artifact")},
235 {"schemaVersion", Value(static_cast<std::int64_t>(1))},
236 {"sourceGuideRevision", Value(static_cast<std::int64_t>(draft.sourceGuideRevision))},
237 {"clipName", Value(draft.clipName)},
238 {"frameCount", Value(draft.frameCount)},
239 {"sequencePattern", Value(draft.sequencePattern)},
240 {"sequenceFirst", Value(draft.sequenceFirst)},
241 {"sequenceLast", Value(draft.sequenceLast)},
242 {"clipPath", Value("clip.json")},
243 {"abilityDraftPath", Value("ability.draft.json")},
244 {"attributionPath", Value("attribution.gen.json")}});
245 if (auto r = writeJson("ingest.manifest.json", manifest); !r) {
246 std::filesystem::remove_all(staging);
247 return Result<Value>::failure(r.status());
248 }
249
250 std::filesystem::create_directories(outputRoot);
251 // Replace published content atomically by swapping staging into place via temp rename.
252 const auto publishedTmp = outputRoot / (".published-" + std::to_string(draft.sourceGuideRevision));
253 std::filesystem::remove_all(publishedTmp);
254 std::filesystem::rename(staging, publishedTmp);
255 for (const auto& entry : std::filesystem::directory_iterator(publishedTmp)) {
256 const auto name = entry.path().filename();
257 if (name == "." || name == "..") continue;
258 const auto dest = outputRoot / name;
259 std::filesystem::remove_all(dest);
260 std::filesystem::rename(entry.path(), dest);
261 }
262 std::filesystem::remove_all(publishedTmp);
263
264 return Result<Value>::success(
265 Value::object({{"exportRoot", Value(outputRoot.generic_string())},
266 {"manifest", Value("ingest.manifest.json")},
267 {"clipPath", Value("clip.json")},
268 {"abilityDraftPath", Value("ability.draft.json")},
269 {"frameCount", Value(draft.frameCount)},
270 {"clipName", Value(draft.clipName)},
271 {"sourceGuideRevision", Value(static_cast<std::int64_t>(draft.sourceGuideRevision))}}));
272 } catch (const std::exception& e) {
273 std::error_code ec;
274 std::filesystem::remove_all(staging, ec);
275 return fail<Value>(DiagnosticCode::Failed, std::string("ingest write failed: ") + e.what());
276 }
