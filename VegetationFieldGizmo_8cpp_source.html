31Result<editing::GizmoSnapshot> VegetationFieldGizmoBuilder::build(std::string target, std::uint64_t expectedRevision,
32 const graphics::VegetationField& field,
33 std::size_t maximumElements) const {
34 if (target.empty() || maximumElements == 0 || maximumElements > 100000)
35 return failure(StatusCode::Rejected, DiagnosticCode::InvalidArgument,
36 "Vegetation field gizmo requires a target and a bounded element budget", "request");
37 if (field.revision() != expectedRevision)
38 return failure(StatusCode::Conflict, DiagnosticCode::StaleHandle,
39 "Vegetation field changed before gizmo generation", "revision");
40 auto elements = field.snapshotElements();
41 if (!elements) return Result<editing::GizmoSnapshot>::failure(elements.status());
42 if (field.revision() != expectedRevision)
43 return failure(StatusCode::Conflict, DiagnosticCode::StaleHandle,
44 "Vegetation field changed while gizmo generation was reading it", "revision");
45 if (elements.value().size() > maximumElements)
46 return failure(StatusCode::Rejected, DiagnosticCode::PreconditionViolation,
47 "Vegetation field exceeds the gizmo element budget", "elements");
48 try {
49 editing::GizmoSnapshot result;
50 result.status = StatusCode::Applied;
51 result.target = std::move(target);
52 result.targetRevision = expectedRevision;
53 result.primitives.reserve(elements.value().size());
54 for (std::size_t i = 0; i < elements.value().size(); ++i) {
55 const auto& element = elements.value()[i];
56 editing::GizmoPrimitive primitive;
57 primitive.id = "vegetation-field-" + std::to_string(i);
58 primitive.kind = element.shape == graphics::VegetationShape::Box ? "obb" : "ellipsoid";
59 primitive.position = {element.center.x, element.center.y, element.center.z};
60 primitive.size = {element.extents.x * 2.0, element.extents.y * 2.0, element.extents.z * 2.0};
61 primitive.color = channelColor(element.channel, element.opacity);
62 primitive.yaw = element.yaw;
63 primitive.dashed = !element.mask.pixels.empty();
64 result.primitives.push_back(std::move(primitive));
65 }
66 return Result<editing::GizmoSnapshot>::success(std::move(result));
67 } catch (const std::bad_alloc&) {
68 return failure(StatusCode::Failed, DiagnosticCode::Failed, "Vegetation field gizmo allocation failed");
69 }
70}

79bool readVec3(const editing::Value& value, glm::vec3& output) {
80 const auto* array = value.getIf<editing::Value::Array>();

294editing::Result<void> VegetationFieldTarget::validateAndPublish(const std::vector<Entry>& entries) {
295 std::vector<graphics::VegetationElement> values;
296 try {
297 values.reserve(entries.size());
298 for (const Entry& entry : entries) values.push_back(entry.value);
299 } catch (const std::bad_alloc&) {
300 return editing::failed<void>(editing::Status::Failed, editing::RuleId("editor.vegetation-field.allocation"),
301 "Vegetation field publication allocation failed");
302 }
303 if (!field_) {
304 graphics::VegetationField validator;
305 const auto result = validator.replace(globals_, values);
306 return result.ok() ? editing::applied<void>() : editing::Result<void>::failure(result.status());
307 }
308 if (field_->revision() != fieldRevision_)
309 return editing::failed<void>(editing::Status::Conflict,
310 editing::RuleId("editor.vegetation-field.runtime-stale"),
311 "Vegetation field changed outside this editor target");
312 const auto result = field_->replace(globals_, values);
313 if (!result.ok()) return editing::Result<void>::failure(result.status());
314 fieldRevision_ = field_->revision();
315 return editing::applied<void>();
316}

318editing::Result<void> VegetationFieldTarget::applyDomainOperation(const editing::DomainOperation& operation) {
319 if (operation.target != targetId() || operation.type != "vegetation-field.element.transform.v1")
320 return editing::failed<void>(editing::Status::Rejected, editing::RuleId("editor.vegetation-field.operation"),
321 "Vegetation field operation mismatch");
322 const auto* idValue = field(operation.payload, "id");
323 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
324 Entry* current = id ? selected(*id) : nullptr;
325 if (!current)
326 return editing::failed<void>(editing::Status::NotFound, editing::RuleId("editor.vegetation-field.element"),
327 "Vegetation field element was not found");
328 Entry replacementEntry = *current;
329 const auto* center = field(operation.payload, "center");
330 const auto* extents = field(operation.payload, "extents");
331 if (!center || !extents || !readVec3(*center, replacementEntry.value.center) ||
332 !readVec3(*extents, replacementEntry.value.extents))
333 return editing::failed<void>(editing::Status::Rejected, editing::RuleId("editor.vegetation-field.transform"),
334 "Vegetation field transform payload is invalid");
335 const auto* yawValue = field(operation.payload, "yaw");
336 const auto* yaw = yawValue ? yawValue->getIf<double>() : nullptr;
337 if (!yaw || !std::isfinite(*yaw))
338 return editing::failed<void>(editing::Status::Rejected, editing::RuleId("editor.vegetation-field.transform"),
339 "Vegetation field transform payload is invalid");
340 replacementEntry.value.yaw = static_cast<float>(*yaw);
341 auto candidate = entries_;
342 *std::find_if(candidate.begin(), candidate.end(), [&](const Entry& entry) { return entry.id == current->id; }) =
343 std::move(replacementEntry);
344 auto published = validateAndPublish(candidate);
345 if (!published.ok()) return published;
346 entries_ = std::move(candidate);
347 ++revision_;
348 dirty_.include(0, 0);
349 return editing::applied<void>();
350}
