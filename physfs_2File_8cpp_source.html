28bool File::open(std::string requestedMode) {
29 if (requestedMode == "c") return true;
30
31 // Love-style aliases → PhysFS binary modes.
32 if (requestedMode == "r") requestedMode = "rb";
33 else if (requestedMode == "w") requestedMode = "wb";
34 else if (requestedMode == "a") requestedMode = "ab";
35
36 if (!PHYSFS_isInit()) throw Exception("PhysFS is not initialized.");
37
38 // File must exist if read mode.
39 if ((requestedMode == "rb") && !PHYSFS_exists(filename.c_str()))
40 throw Exception("Could not open file %s. Does not exist.", filename.c_str());
41
42 // Check whether the write directory is set.
43 if ((requestedMode == "ab" || requestedMode == "wb") &&
44 (PHYSFS_getWriteDir() == nullptr) && !hack_setupWriteDirectory())
45 throw Exception("Could not set write directory.");
46
47 // File already open?
48 if (file != nullptr) return false;
49
50 PHYSFS_getLastErrorCode();
51 PHYSFS_File *handle = nullptr;
52
53 if (requestedMode == "rb") handle = PHYSFS_openRead(filename.c_str());
54 else if (requestedMode == "ab") handle = PHYSFS_openAppend(filename.c_str());
55 else if (requestedMode == "wb") handle = PHYSFS_openWrite(filename.c_str());
56 else
57 throw Exception("Invalid file mode %s.", requestedMode.c_str());
58
59 if (handle == nullptr) {
60 const char *err = PHYSFS_getErrorByCode(PHYSFS_getLastErrorCode());
61 if (err == nullptr) err = "unknown error";
62 throw Exception("Could not open file %s (%s)", filename.c_str(), err);
63 }
64
65 file = handle;
66
67 this->mode = requestedMode;
68
69 if (file != nullptr && !setBuffer(bufferMode, bufferSize)) {
70 // Revert to buffer defaults if we don't successfully set the buffer.
71 bufferMode = "none";
72 bufferSize = 0;
73 }
74
75 return (file != nullptr);
76}

102int64_t File::read(void *dst, int64_t size) {
103 if (!file || mode != "rb") throw Exception("File is not opened for reading.");
104
105 int64_t max = (int64_t)PHYSFS_fileLength(file);
106 int64_t cur = (int64_t)PHYSFS_tell(file);
107 if (max < 0 || cur < 0) throw Exception("Could not read from file.");
108 if (cur > max) return 0;
109
110 int64_t remaining = max - cur;
111 size = (size == ALL) ? remaining : size;
112 if (size > remaining) size = remaining;
113
114 if (size < 0) throw Exception("Invalid read size.");
115
116 return PHYSFS_readBytes(file, dst, (PHYSFS_uint64)size);
117}

119bool File::write(const void *data, int64_t size) {
120 if (!file || (mode != "wb" && mode != "ab")) throw Exception("File is not opened for writing.");
121
122 if (size < 0) throw Exception("Invalid write size.");
123
124 // Try to write.
125 int64_t written = PHYSFS_writeBytes(file, data, (PHYSFS_uint64)size);
126
127 // Check that correct amount of data was written.
128 if (written != size) return false;
129
130 // Manually flush the buffer in BUFFER_LINE mode if we find a newline.
131 if (bufferMode == "newline" && bufferSize > size) {
132 if (memchr(data, '\n', (size_t)size) != nullptr) flush();
133 }
134
135 return true;
136}

164bool File::setBuffer(std::string bufmode, int64_t size) {
165 if (size < 0) return false;
166
167 // If the file isn't open, we'll make sure the buffer values are set in
168 // File::open.
169 if (!isOpen()) {
170 bufferMode = bufmode;
171 bufferSize = size;
172 return true;
173 }
174
175 int ret = 1;
176 if (bufferMode == "newline" || bufferMode == "full") {
177 ret = PHYSFS_setBuffer(file, size);
178 } else {
179 ret = PHYSFS_setBuffer(file, 0);
180 size = 0;
181 }
182
183 if (ret == 0) return false;
184
185 bufferMode = bufmode;
186 bufferSize = size;
187
188 return true;
189}
