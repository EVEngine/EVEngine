24Result<TerrainVegetationGpuPlan> buildTerrainVegetationGpuPlan(const TerrainVegetationRealization& realization,
25 ITerrainVegetationGpuResolver& resolver,
26 const TerrainVegetationGpuFrame& frame) {
27 if (!std::isfinite(frame.timeSeconds) || std::abs(frame.timeSeconds) > 1.0e12)
28 return failure<TerrainVegetationGpuPlan>(DiagnosticCode::InvalidArgument,
29 "terrain vegetation frame time is invalid");
30 TerrainVegetationGpuPlan result;
31 result.instances.reserve(realization.instances.size());
32 std::map<std::string_view, const RuntimeInstancePrototype*, std::less<>> prototypeMetadata;
33 for (const auto& prototype : realization.prototypes)
34 if (prototype.prototype.empty() || !prototypeMetadata.emplace(prototype.prototype, &prototype).second)
35 return failure<TerrainVegetationGpuPlan>(
36 DiagnosticCode::Conflict, "terrain vegetation prototype metadata is invalid", prototype.prototype);
37 std::size_t expectedFirst = 0;
38 for (const auto& bucket : realization.buckets) {
39 if (bucket.prototype.empty() || bucket.firstInstance != expectedFirst || bucket.instanceCount == 0 ||
40 std::uint64_t(bucket.firstInstance) + bucket.instanceCount > realization.instances.size())
41 return failure<TerrainVegetationGpuPlan>(DiagnosticCode::InvalidArgument,
42 "terrain vegetation bucket layout is invalid", bucket.prototype);
43 const auto metadata = prototypeMetadata.find(bucket.prototype);
44 auto prototype =
45 resolver.resolve({bucket.prototype, metadata == prototypeMetadata.end() ? nullptr : metadata->second});
46 if (!prototype) return Result<TerrainVegetationGpuPlan>::failure(prototype.status());
47 if (prototype.value().parts.empty() &&
48 (prototype.value().meshId == graphics::kInvalidGpuDrivenSlot ||
49 prototype.value().materialId == graphics::kInvalidGpuDrivenSlot))
50 return failure<TerrainVegetationGpuPlan>(
51 DiagnosticCode::NotFound, "vegetation prototype has no GPU mesh or material", bucket.prototype);
52 for (const auto& part : prototype.value().parts)
53 if (part.meshId == graphics::kInvalidGpuDrivenSlot || part.materialId == graphics::kInvalidGpuDrivenSlot)
54 return failure<TerrainVegetationGpuPlan>(DiagnosticCode::NotFound,
55 "vegetation prefab part has no GPU mesh or material",
56 bucket.prototype);
57 for (std::uint32_t offset = 0; offset < bucket.instanceCount; ++offset) {
58 const auto& source = realization.instances[bucket.firstInstance + offset];
59 if (source.prototype != bucket.prototype)
60 return failure<TerrainVegetationGpuPlan>(
61 DiagnosticCode::Conflict, "vegetation bucket does not match instance prototype", bucket.prototype);
62 const glm::quat rotation(source.rotation[3], source.rotation[0], source.rotation[1], source.rotation[2]);
63 const glm::mat4 model =
64 glm::translate(glm::mat4(1.f), {source.position[0], source.position[1], source.position[2]}) *
65 glm::mat4_cast(rotation) *
66 glm::scale(glm::mat4(1.f), {source.scale[0], source.scale[1], source.scale[2]});
67 glm::vec4 instanceColor(1.f);
68 if (metadata != prototypeMetadata.end() && !metadata->second->useInstancing) {
69 const auto& detail = *metadata->second;
70 const float span = detail.maxHeight - detail.minHeight;
71 const float healthy = span > 0.f ? std::clamp((source.scale[1] - detail.minHeight) / span, 0.f, 1.f) : 1.f;
72 for (glm::length_t component = 0; component < 4; ++component)
73 instanceColor[component] =
74 detail.dryColor[component] + (detail.healthyColor[component] - detail.dryColor[component]) * healthy;
75 }
76 glm::vec4 terrainWave{};
77 glm::vec4 terrainWaveTint(1.f);
78 if (metadata != prototypeMetadata.end() && !metadata->second->usePrototypeMesh) {
79 terrainWave = {static_cast<float>(frame.timeSeconds * realization.wavingGrassSpeed),
80 realization.wavingGrassAmount, realization.wavingGrassStrength, 1.f};
81 terrainWaveTint = glm::make_vec4(realization.wavingGrassTint.data());
82 }
83 if (prototype.value().parts.empty()) {
84 graphics::GpuInstance instance{};
85 instance.model = model;
86 instance.meshId = prototype.value().meshId;
87 instance.materialId = prototype.value().materialId;
88 instance.flags = prototype.value().flags;
89 instance.lodGroupId = prototype.value().lodGroupId;
90 instance.color = instanceColor;
91 instance.terrainWave = terrainWave;
92 instance.terrainWaveTint = terrainWaveTint;
93 result.instances.push_back(instance);
94 } else {
95 for (const auto& part : prototype.value().parts) {
96 graphics::GpuInstance instance{};
97 instance.model = model * glm::make_mat4(part.localTransform.data());
98 instance.meshId = part.meshId;
99 instance.materialId = part.materialId;
100 instance.flags = part.flags;
101 instance.lodGroupId = part.lodGroupId;
102 instance.color = instanceColor;
103 instance.terrainWave = terrainWave;
104 instance.terrainWaveTint = terrainWaveTint;
105 result.instances.push_back(instance);
106 }
107 }
108 }
109 expectedFirst += bucket.instanceCount;
110 }
111 if (expectedFirst != realization.instances.size())
112 return failure<TerrainVegetationGpuPlan>(DiagnosticCode::InvalidArgument,
113 "terrain vegetation buckets do not cover every instance");
114 return Result<TerrainVegetationGpuPlan>::success(std::move(result));
115}

117Result<void> submitTerrainVegetation(graphics::Graphics& graphics, const TerrainVegetationRealization& realization,
118 ITerrainVegetationGpuResolver& resolver,
119 const TerrainVegetationGpuFrame& frame) {
120 auto plan = buildTerrainVegetationGpuPlan(realization, resolver, frame);
121 if (!plan) return Result<void>::failure(plan.status());
122 if (plan.value().instances.empty()) return Result<void>::success();
123 if (!graphics.gpuDrivenSubmitOpaque(plan.value().instances.data(),
124 static_cast<std::uint32_t>(plan.value().instances.size())))
125 return failure<void>(DiagnosticCode::Failed, "GPU-driven vegetation submission failed");
126 return Result<void>::success();
127}

143 drawerToken_ = graphics::RenderSystem3D::addGpuOpaqueCollector(
144 [this](graphics::Graphics&, const graphics::Camera3D::Data&, const glm::mat4&, float,
145 std::vector<graphics::GpuInstance>& instances) {

159 std::map<std::string_view, const RuntimeInstancePrototype*, std::less<>> metadata;
160 for (const auto& prototype : state_->realization.prototypes)
161 metadata.emplace(prototype.prototype, &prototype);
162 for (const auto& instance : state_->realization.instances) {
163 const auto found = metadata.find(instance.prototype);
164 const glm::quat rotation(instance.rotation[3], instance.rotation[0], instance.rotation[1],
165 instance.rotation[2]);
166 const glm::mat4 model =
167 glm::translate(glm::mat4(1.f), {instance.position[0], instance.position[1], instance.position[2]}) *
168 glm::mat4_cast(rotation) *
169 glm::scale(glm::mat4(1.f), {instance.scale[0], instance.scale[1], instance.scale[2]});
170 std::array<float, 16> modelValues{}, lightValues{};
171 std::copy(glm::value_ptr(model), glm::value_ptr(model) + 16, modelValues.begin());
172 std::copy(glm::value_ptr(lightViewProjection), glm::value_ptr(lightViewProjection) + 16,
173 lightValues.begin());
174 auto drawn = state_->resolver->drawShadow(
175 {instance.prototype, found == metadata.end() ? nullptr : found->second}, modelValues, lightValues);
176 if (!drawn) {
177 state_->shadowFailedThisFrame = true;
178 state_->shadowError = std::make_unique<Diagnostic>(*drawn.error());
179 return;
180 }
181 }
182 });
