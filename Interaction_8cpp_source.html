51InteractionSession::InteractionSession(InteractionContext context) : context_(std::move(context)) {
52 // The initial state is computed once from the context; every later transition is an
53 // explicit handler, so there is exactly one place that decides "can the player act".
54 if (context_.status != BattleStatus::Running) {
55 state_ = context_.status == BattleStatus::Ended ? InteractionState::Ended : InteractionState::Blocked;
56 } else if (context_.phase != BattlePhase::Acting) {
57 state_ = InteractionState::Blocked;
58 } else if (!context_.controlledUnit.isValid() || context_.activeUnit != context_.controlledUnit) {
59 state_ = InteractionState::Blocked;
60 } else {
61 state_ = InteractionState::AwaitSelection;
62 }
63}

80InteractionIntent InteractionSession::makeIntent(InteractionIntentKind kind, Cell cell) const {
81 InteractionIntent intent;
82 intent.kind = kind;
83 intent.actor = context_.controlledUnit;
84 intent.cell = cell;
85 intent.action = armedAction_;
86 const auto occupant = context_.unitsByCell.find(cell);
87 if (occupant != context_.unitsByCell.end()) intent.targetUnit = occupant->second;
88 intent.expected = context_.revision;
89 return intent;

92Result<InteractionIntent> InteractionSession::onCellClicked(Cell cell) {
93 if (state_ == InteractionState::Ended)
94 return Result<InteractionIntent>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
95 "interaction session belongs to a finished battle",
96 "interaction.state"));
97 if (state_ == InteractionState::Blocked)
98 return Result<InteractionIntent>::failure(Diagnostic::error(
99 DiagnosticCode::PreconditionViolation, "interaction session is blocked", "interaction.state"));
100 if (state_ == InteractionState::Resolving)
101 return Result<InteractionIntent>::failure(
102 Diagnostic::error(DiagnosticCode::Conflict,
103 "a pending interaction intent must be committed or dropped first", "interaction.state"));
104
105 if (state_ == InteractionState::Targeting) {
106 if (!cellInTargetable(cell)) return Result<InteractionIntent>::success(makeIntent(InteractionIntentKind::None, cell));
107 // The intent names both the cell and the occupying unit, so the caller can commit
108 // it without re-deriving which unit the click landed on.
109 pending_ = InteractionIntentKind::UseAbilityOn;
110 state_ = InteractionState::Resolving;
111 return Result<InteractionIntent>::success(makeIntent(pending_, cell));
112 }
113
114 // AwaitSelection / UnitSelected.
115 if (cell == context_.controlledCell) {
116 unitSelected_ = true;
117 state_ = InteractionState::UnitSelected;
118 pending_ = InteractionIntentKind::SelectUnit;
119 return Result<InteractionIntent>::success(makeIntent(pending_, cell));
120 }
121 if (cellInReachable(cell)) {
122 // A move is a complete intention on its own, so the machine reports it and lets the
123 // caller commit; the session is expected to be resolved or rebuilt afterwards.
124 pending_ = InteractionIntentKind::MoveTo;
125 state_ = InteractionState::Resolving;
126 return Result<InteractionIntent>::success(makeIntent(pending_, cell));
127 }
128 return Result<InteractionIntent>::success(makeIntent(InteractionIntentKind::None, cell));
129}

131Result<void> InteractionSession::armAbility(const LogicalId& action, std::vector<Cell> targetableCells) {
132 if (state_ == InteractionState::Ended || state_ == InteractionState::Blocked || state_ == InteractionState::Resolving)
133 return Result<void>::failure(Diagnostic::error(
134 DiagnosticCode::PreconditionViolation, "interaction session cannot arm an ability", "interaction.state"));
135 if (!action.isValid())
136 return Result<void>::failure(
137 Diagnostic::error(DiagnosticCode::InvalidArgument, "armed ability requires a logical id", "action"));
138 armedAction_ = action;
139 armedTargets_ = std::move(targetableCells);
140 unitSelected_ = true;
141 state_ = InteractionState::Targeting;
142 pending_ = InteractionIntentKind::None;
143 return Result<void>::success(Status::success(StatusCode::Applied));
144}

146Result<InteractionIntent> InteractionSession::onCancel() {
147 if (state_ == InteractionState::Ended)
148 return Result<InteractionIntent>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
149 "interaction session belongs to a finished battle",
150 "interaction.state"));
151 if (state_ == InteractionState::Blocked)
152 return Result<InteractionIntent>::failure(Diagnostic::error(
153 DiagnosticCode::PreconditionViolation, "interaction session is blocked", "interaction.state"));
154 if (state_ == InteractionState::Resolving)
155 return Result<InteractionIntent>::failure(
156 Diagnostic::error(DiagnosticCode::Conflict,
157 "a pending interaction intent must be committed or dropped first", "interaction.state"));
158 if (state_ == InteractionState::AwaitSelection && !unitSelected_)
159 // Nothing to cancel is not a failure, but it must not be reported as if the player
160 // had cancelled something either.
161 return Result<InteractionIntent>::success(makeIntent(InteractionIntentKind::None, context_.controlledCell));
162
163 armedAction_ = LogicalId{};
164 armedTargets_.clear();
165 unitSelected_ = false;
166 pending_ = InteractionIntentKind::None;
167 state_ = InteractionState::AwaitSelection;
168 return Result<InteractionIntent>::success(makeIntent(InteractionIntentKind::Cancel, context_.controlledCell));
169}

171Result<InteractionIntent> InteractionSession::onEndTurn() {
172 if (state_ == InteractionState::Ended)
173 return Result<InteractionIntent>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
174 "interaction session belongs to a finished battle",
175 "interaction.state"));
176 if (state_ == InteractionState::Blocked || state_ == InteractionState::Resolving)
177 return Result<InteractionIntent>::failure(Diagnostic::error(
178 DiagnosticCode::PreconditionViolation, "interaction session cannot end the turn", "interaction.state"));
179 // Ending the turn is terminal for this activation, so the session moves to Resolving and
180 // the caller decides whether to commit it or rebuild from a fresh context.
181 pending_ = InteractionIntentKind::EndTurn;
182 armedAction_ = LogicalId{};
183 armedTargets_.clear();
184 unitSelected_ = false;
185 state_ = InteractionState::Resolving;
186 InteractionIntent intent = makeIntent(pending_, context_.controlledCell);
187 // `makeIntent` echoed the armed action, which this handler just cleared.
188 intent.action = LogicalId{};
189 return Result<InteractionIntent>::success(std::move(intent));
190}

192Result<void> InteractionSession::markResolving() {
193 if (state_ == InteractionState::Ended || state_ == InteractionState::Blocked)
194 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
195 "interaction session cannot resolve", "interaction.state"));
196 if (pending_ == InteractionIntentKind::None)
197 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
198 "interaction session has no pending intent to confirm",
199 "interaction.pending"));
200 state_ = InteractionState::Resolving;
201 return Result<void>::success(Status::success(StatusCode::Applied));
202}
