22void EditorShell::open(UIHostHandle inspector, UIHostHandle database, UIHostHandle scene) {
23 inspector_ = inspector;
24 database_ = database;
25 scene_ = scene;
26 auto host = UIHost::resolve(host_);
27 if (!host) {
28 host_ = UIHost::createHost(kShellHostName);
29 host = UIHost::resolve(host_);
30 }
31 if (!host) return;
32 UIHost& shell = host->get();
33 auto meta = shell.meta();
34 meta->overlay = true; // no title bar
35 meta->hasPos = true;
36 meta->lockPos = true;
37 meta->posX = 0.f;
38 meta->posY = 0.f;
39 meta->hasSize = true;
40 meta->sizeX = 0.f; // width = full display (percentW below)
41 meta->sizeY = toolbarHeight_;
42 meta->lockSize = true;
43 meta->percentW = 1.f;
44 meta->visible = true;
45 meta->overlayBgAlpha = 1.f;
46 meta->overlayFlush = true;
47 shell.setTree(build());
48 relayout();
49 setHostVisible(inspector_, true);
50 setHostVisible(database_, true);
51 setHostVisible(scene_, true);
52}

66bool EditorShell::selectPanel(const std::string& name) {
67 auto host = UIHost::resolve(host_);
68 if (!host || !host->get().meta()->visible) return false;
69 if (name == "inspector") {
70 setHostVisible(inspector_, true);
71 setHostVisible(database_, false);
72 setHostVisible(scene_, false);
73 return true;
74 }
75 if (name == "database") {
76 setHostVisible(database_, true);
77 setHostVisible(inspector_, false);
78 setHostVisible(scene_, false);
79 return true;
80 }
81 if (name == "scene") {
82 setHostVisible(scene_, true);
83 setHostVisible(inspector_, false);
84 setHostVisible(database_, false);
85 return true;
86 }
87 if (name.empty()) {
88 setHostVisible(inspector_, false);
89 setHostVisible(database_, false);
90 setHostVisible(scene_, false);
91 return true;
92 }
93 return false;
94}

96bool EditorShell::togglePanel(const std::string& name) {
97 auto host = UIHost::resolve(host_);
98 if (!host || !host->get().meta()->visible) return false;
99 UIHostHandle handle{};
100 if (name == "inspector") handle = inspector_;
101 if (name == "database") handle = database_;
102 if (name == "scene") handle = scene_;
103 auto panel = UIHost::resolve(handle);
104 if (!panel) return false;
105 UIHost& panelHost = panel->get();
106 panelHost.setVisible(!panelHost.meta()->visible);
107 return true;
108}

110WidgetDesc EditorShell::build() {
111 std::vector<WidgetDesc> commands;
112 commands.push_back(icon(Icon::Cubes, "brand_icon"));
113 commands.push_back(text("EVEngine Editor", "lbl_title"));
114 commands.push_back(badge("Workspace", "workspace_badge"));
115 commands.push_back(spacer("menu_spacer"));
116 commands.push_back(iconButton(Icon::Sliders, "", "menu_inspector", [this]() {
117 togglePanel("inspector");
118 }).withTooltip("Show or hide the Inspector"));
119 commands.push_back(iconButton(Icon::Database, "", "menu_database", [this]() {
120 togglePanel("database");
121 }).withTooltip("Show or hide the object database"));
122 commands.push_back(iconButton(Icon::Layers, "", "menu_scene", [this]() {
123 togglePanel("scene");
124 }).withTooltip("Show or hide the scene hierarchy"));
125 commands.push_back(iconButton(Icon::EyeSlash, "", "menu_close", [this]() {
126 selectPanel("");
127 }).withTooltip("Hide all editor panels"));
128 return window("Editor", {toolbar(std::move(commands), "editor_toolbar")}, "root");
129}
