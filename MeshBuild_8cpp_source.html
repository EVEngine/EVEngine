90std::unique_ptr<MeshBuild> MeshBuild::copyGroup(int groupIndex) const {
91 if (groupIndex < 0 || groupIndex >= getGroupCount()) return {};
92 auto result = std::make_unique<MeshBuild>();
93 result->setActiveGroup(groupNames_[size_t(groupIndex)]);
94 std::unordered_map<uint32_t, uint32_t> remap;
95 for (int t = 0; t < int(triangleGroups_.size()); ++t) {
96 if (triangleGroups_[size_t(t)] != groupIndex) continue;
97 uint32_t dst[3];
98 for (int corner = 0; corner < 3; ++corner) {
99 const uint32_t src = indices_[size_t(t) * 3u + size_t(corner)];
100 auto found = remap.find(src);
101 if (found == remap.end()) {
102 const uint32_t next = uint32_t(result->getVertexCount());
103 result->addVertex(getPositionX(int(src)), getPositionY(int(src)), getPositionZ(int(src)),
104 getNormalX(int(src)), getNormalY(int(src)), getNormalZ(int(src)),
105 getUvU(int(src)), getUvV(int(src)));
106 remap.emplace(src, next); dst[corner] = next;
107 } else dst[corner] = found->second;
108 }
109 result->addTriangle(dst[0], dst[1], dst[2]);
110 }
111 if (result->empty()) return {};
112 result->meta_ = meta_;
113 result->setMeta("group", groupNames_[size_t(groupIndex)]);
114 return result;
115}

117bool MeshBuild::appendTransformed(const MeshBuild *other, float tx, float ty, float tz,
118 float yawDegrees, float sx, float sy, float sz) {
119 if (!other || other == this || other->empty() || std::fabs(sx) < 1e-8f ||
120 std::fabs(sy) < 1e-8f || std::fabs(sz) < 1e-8f)
121 return false;
122 const uint32_t base = uint32_t(getVertexCount());
123 std::vector<int> groupMap(size_t(other->getGroupCount()), -1);
124 const int previousGroup = activeGroup_;
125 for (int i = 0; i < other->getGroupCount(); ++i)
126 groupMap[size_t(i)] = setActiveGroup(other->getGroupName(i));
127 reserve(getVertexCount() + other->getVertexCount(), getIndexCount() + other->getIndexCount());
128 const float radians = yawDegrees * 0.01745329251994329577f;
129 const float c = std::cos(radians), s = std::sin(radians);
130 for (int i = 0; i < other->getVertexCount(); ++i) {
131 const float px = other->getPositionX(i) * sx;
132 const float py = other->getPositionY(i) * sy;
133 const float pz = other->getPositionZ(i) * sz;
134 const float ix = other->getNormalX(i) / sx;
135 const float iy = other->getNormalY(i) / sy;
136 const float iz = other->getNormalZ(i) / sz;
137 float nx = c * ix + s * iz;
138 float ny = iy;
139 float nz = -s * ix + c * iz;
140 const float nl = std::sqrt(nx * nx + ny * ny + nz * nz);
141 if (nl > 1e-8f) { nx /= nl; ny /= nl; nz /= nl; }
142 addVertex(c * px + s * pz + tx, py + ty, -s * px + c * pz + tz,
143 nx, ny, nz, other->getUvU(i), other->getUvV(i));
144 }
145 const bool mirrored = (sx * sy * sz) < 0.f;
146 for (int i = 0; i + 2 < other->getIndexCount(); i += 3) {
147 const int sourceGroup = other->getTriangleGroup(i / 3);
148 activeGroup_ = sourceGroup >= 0 ? groupMap[size_t(sourceGroup)] : -1;
149 const uint32_t a = base + uint32_t(other->getIndex(i));
150 const uint32_t b = base + uint32_t(other->getIndex(i + 1));
151 const uint32_t cidx = base + uint32_t(other->getIndex(i + 2));
152 if (mirrored) addTriangle(a, cidx, b); else addTriangle(a, b, cidx);
153 }
154 activeGroup_ = previousGroup;
155 return true;
156}
