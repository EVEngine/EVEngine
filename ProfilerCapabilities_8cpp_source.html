21void appendString(std::string& out, const std::string& value) {
22 out += '"';
23 for (char c : value) {
24 switch (c) {
25 case '"': out += "\\\""; break;
26 case '\\': out += "\\\\"; break;

55 std::string frameJson() const override {
56 const bool hasFrame = eve::prof::Profiler::hasFrame();
57 auto* timer = eve::cap::query<eve::service::IGpuTimer>();
58 const bool gpuKnown = timer != nullptr && timer->gpuTimingAvailable();
59
60 // Depth-0 samples are the roots of each thread's zone tree, so their self
61 // times are the CPU work the zones account for.
62 static const std::vector<eve::prof::ZoneSample> kNoZones;
63 const std::vector<eve::prof::ZoneSample>& zones = hasFrame ? eve::prof::Profiler::lastFrame() : kNoZones;
64 double cpuFrameMs = 0.0;
65 for (const auto& zone : zones) {
66 if (zone.minDepth == 0) cpuFrameMs += zone.selfMs;
67 }
68
69 std::string out = "{\"schema\":\"eve.profiler.frame\",\"version\":1,\"enabled\":";
70 out += enabled() ? "true" : "false";
71 out += ",\"hasFrame\":";
72 out += hasFrame ? "true" : "false";
73 out += ",\"cpuFrameMs\":";
74 appendNumber(out, cpuFrameMs);
75 out += ",\"gpuMs\":";
76 appendNumber(out, gpuKnown ? static_cast<double>(timer->gpuFrameMs()) : 0.0);
77 out += ",\"gpuTimingAvailable\":";
78 out += gpuKnown ? "true" : "false";
79 out += ",\"zoneCount\":" + std::to_string(zones.size()) + ",\"zones\":[";
80 for (std::size_t index = 0; index < zones.size(); ++index) {
81 const eve::prof::ZoneSample& zone = zones[index];
82 if (index) out += ',';
83 out += "{\"module\":";
84 appendString(out, zone.module);
85 out += ",\"name\":";
86 appendString(out, zone.name);
87 out += ",\"thread\":";
88 appendString(out, zone.thread);
89 out += ",\"selfMs\":";
90 appendNumber(out, zone.selfMs);
91 out += ",\"totalMs\":";
92 appendNumber(out, zone.totalMs);
93 out += ",\"count\":" + std::to_string(zone.count) + ",\"depth\":" + std::to_string(zone.minDepth) + "}";
94 }
95 out += "]}";
96 return out;
97 }
