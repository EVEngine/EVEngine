56struct FormulaParser {
57 const std::string &s;
58 size_t pos = 0;
59 RPGActor *a;
60 RPGActor *b;
61 bool validateOnly;
62 std::optional<eve::Diagnostic> error;
63
64 FormulaParser(const std::string &str, RPGActor *atk, RPGActor *tgt, bool syntaxOnly)
65 : s(str), a(atk), b(tgt), validateOnly(syntaxOnly) {}
66
67 void skipSpace() {
68 while (pos < s.size() && std::isspace(static_cast<unsigned char>(s[pos]))) ++pos;
69 }
70
71 double parseExpression() {
72 double lhs = parseTerm();
73 while (true) {
74 skipSpace();
75 if (pos >= s.size()) break;
76 const char c = s[pos];
77 if (c == '+') {
78 ++pos;
79 lhs += parseTerm();
80 } else if (c == '-') {
81 ++pos;
82 lhs -= parseTerm();
83 } else {
84 break;
85 }
86 }
87 return lhs;
88 }
89
90 double parseTerm() {
91 double lhs = parseFactor();
92 while (true) {
93 skipSpace();
94 if (pos >= s.size()) break;
95 const char c = s[pos];
96 if (c == '*') {
97 ++pos;
98 lhs *= parseFactor();
99 } else if (c == '/') {
100 ++pos;
101 const double rhs = parseFactor();
102 if (validateOnly) {
103 lhs = 1.0;
104 } else if (rhs == 0.0) {
105 if (!error)
106 error = eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
107 "damage formula cannot divide by zero",
108 "formula[" + std::to_string(pos) + "]");
109 lhs = 0.0;
110 } else {
111 lhs /= rhs;
112 }
113 } else {
114 break;
115 }
116 }
117 return lhs;
118 }
119
120 double parseFactor() {
121 skipSpace();
122 if (pos < s.size() && (s[pos] == '+' || s[pos] == '-')) {
123 const bool negate = s[pos++] == '-';
124 const double value = parseFactor();
125 return negate ? -value : value;
126 }
127 if (pos < s.size() && s[pos] == '(') {
128 ++pos;
129 double v = parseExpression();
130 skipSpace();
131 if (pos >= s.size() || s[pos] != ')') {
132 fail("damage formula is missing a closing parenthesis");
133 return 0.0;
134 }
135 ++pos;
136 return v;
137 }
138 if (pos < s.size() && (s[pos] == 'a' || s[pos] == 'b') && pos + 1 < s.size() &&
139 s[pos + 1] == '.') {
140 const bool isA = s[pos] == 'a';
141 pos += 2;
142 std::string name;
143 while (pos < s.size() && (std::isalnum(static_cast<unsigned char>(s[pos])) ||
144 s[pos] == '_')) {
145 name.push_back(s[pos++]);
146 }
147 if (name.empty()) {
148 fail("damage formula attribute name must not be empty");
149 return 0.0;
150 }
151 RPGActor *actor = isA ? a : b;
152 return actor ? AttributeSystem::getFinal(actor, name) : 0.0;
153 }
154 if (pos < s.size() && (std::isdigit(static_cast<unsigned char>(s[pos])) || s[pos] == '.')) {
155 std::string num;
156 while (pos < s.size() && (std::isdigit(static_cast<unsigned char>(s[pos])) ||
157 s[pos] == '.')) {
158 num.push_back(s[pos++]);
159 }
160 char *end = nullptr;
161 const double value = std::strtod(num.c_str(), &end);
162 if (end == num.c_str() || *end != '\0' || !std::isfinite(value)) {
163 fail("damage formula contains an invalid number");
164 return 0.0;
165 }
166 return value;
167 }
168 fail("damage formula expected a number, attribute, unary sign, or parenthesized expression");
169 return 0.0;
170 }
171
172 void fail(std::string message) {
173 if (error) return;
174 error = eve::Diagnostic::error(eve::DiagnosticCode::ParseError, std::move(message),
175 "formula[" + std::to_string(pos) + "]");
176 }
177};

179eve::Result<double> parseFormula(const std::string &formula, RPGActor *attacker, RPGActor *target,
180 bool validateOnly) {
181 if (formula.empty())
182 return eve::Result<double>::failure(eve::Diagnostic::error(
183 eve::DiagnosticCode::ParseError, "damage formula must not be empty", "formula"));
184 FormulaParser parser(formula, attacker, target, validateOnly);
185 const double value = parser.parseExpression();
186 parser.skipSpace();
187 if (parser.error) return eve::Result<double>::failure(std::move(*parser.error));
188 if (parser.pos != formula.size())
189 return eve::Result<double>::failure(eve::Diagnostic::error(
190 eve::DiagnosticCode::ParseError, "damage formula contains trailing input",
191 "formula[" + std::to_string(parser.pos) + "]"));
192 if (!std::isfinite(value))
193 return eve::Result<double>::failure(eve::Diagnostic::error(
194 eve::DiagnosticCode::InvalidArgument, "damage formula result must be finite", "formula"));
195 return eve::Result<double>::success(value);
196}

210DamageResult BattleSystem::resolveHit(RPGActor *attacker, RPGActor *target,
211 const SkillDamageSpec &spec, unsigned seed) {
212 DamageResult result;
213 if (!attacker || !target) return result;
214 std::mt19937 rng(seed);
215 std::uniform_real_distribution<double> dist(0.0, 1.0);
216
217 // 命中判定：攻击方 hit + (100 - 目标 evasion)
218 double attackerHit = 100.0 + TraitSystem::getExParam(attacker, "hit");
219 double targetEvasion = TraitSystem::getExParam(target, "evasion");
220 double hitChance = spec.hitChance + attackerHit - targetEvasion;
221 if (dist(rng) * 100.0 >= hitChance) {
222 result.hit = false;
223 return result;
224 }
225
226 // 公式（默认物理：a.atk - b.def，至少 1）
227 double amount;
228 if (spec.formula.empty()) {
229 amount = AttributeSystem::getFinal(attacker, "attack") -
230 AttributeSystem::getFinal(target, "defense");
231 } else {
232 amount = evaluateFormula(spec.formula, attacker, target);
233 }
234 if (amount <= 0.0) amount = 1.0;
235
236 // 元素耐性
237 if (!spec.element.empty()) {
238 result.elementRate = TraitSystem::getElementRate(target, spec.element);
239 amount *= result.elementRate;
240 }
241
242 // 暴击（RPG Maker 默认 3x）
243 double critChance = spec.critChance + TraitSystem::getExParam(attacker, "critRate");
244 if (critChance > 0.0 && dist(rng) < critChance) {
245 result.crit = true;
246 amount *= 3.0;
247 }
248
249 result.amount = amount;
250 return result;
251}
