16[[nodiscard]] std::string quote(std::string_view value) {
17 std::string result;
18 result.reserve(value.size() + 2);
19 result.push_back('"');
20 for (const unsigned char byte : value) {
21 switch (byte) {
22 case '"': result += "\\\""; break;
23 case '\\': result += "\\\\"; break;
24 case '\n': result += "\\n"; break;
25 case '\r': result += "\\r"; break;
26 case '\t': result += "\\t"; break;
27 default:
28 if (byte < 0x20u) {
29 result += "\\u00";
30 constexpr char digits[] = "0123456789abcdef";
31 result.push_back(digits[(byte >> 4u) & 0x0fu]);
32 result.push_back(digits[byte & 0x0fu]);
33 } else {
34 result.push_back(static_cast<char>(byte));
35 }
36 }
37 }
38 result.push_back('"');
39 return result;
40}

58[[nodiscard]] bool sameLayout(const eve::container::ContainerSnapshot& lhs,
59 const eve::container::ContainerSnapshot& rhs) {
60 if (lhs.id != rhs.id || lhs.revision != rhs.revision || lhs.entries.size() != rhs.entries.size()) return false;
61 for (const auto& left : lhs.entries) {
62 const auto right = std::find_if(rhs.entries.begin(), rhs.entries.end(), [&](const auto& entry) {
63 return entry.membership.object == left.membership.object;
64 });
65 if (right == rhs.entries.end() || left.membership.slot != right->membership.slot ||
66 left.membership.generation != right->membership.generation || left.object.id != right->object.id ||
67 left.object.type != right->object.type || left.object.quantity != right->object.quantity ||
68 left.object.tags != right->object.tags)
69 return false;
70 }
71 return true;
72}

125class BuildingGarrisonContainerAdapter::PreparedState final : public eve::container::IContainer::PreparedState {
126public:
127 PreparedState(BuildingGarrisonContainerAdapter& owner, std::vector<GarrisonMember> staged,
128 eve::Revision expectedRevision, std::size_t expectedSize, eve::Revision revision)
129 : owner_(owner),
130 staged_(std::move(staged)),
131 expectedRevision_(expectedRevision),
132 expectedSize_(expectedSize),
133 revision_(revision) {}
134
135 void commit() noexcept override {
136 if (committed_) return;
137 auto* building = owner_.building();
138 if (building == nullptr || building->garrisonRevision != expectedRevision_ ||
139 building->garrison.size() != expectedSize_)
140 std::terminate();
141 building->garrison.swap(staged_);
142 building->garrisonRevision = revision_;
143 committed_ = true;
144 }
145
146 void rollback() noexcept override {
147 if (!committed_) staged_.clear();
148 }
149
150private:
151 BuildingGarrisonContainerAdapter& owner_;
152 std::vector<GarrisonMember> staged_;
153 eve::Revision expectedRevision_;
154 std::size_t expectedSize_ = 0;
155 eve::Revision revision_;
156 bool committed_ = false;
157};

204 const auto* value = building();

206 return eve::Result<eve::container::ContainerSnapshot>::failure(

208 if (!descriptor_.capacity.isUnlimited() && value->garrison.size() > descriptor_.capacity.value())

224 result.entries.push_back(

231 const eve::container::ContainerObject& object, std::optional<eve::container::SlotIndex> destination,
232 std::optional<eve::container::MembershipId> ignoredObject) const {
233 const auto* value = building();
234 if (!descriptor_.id.isValid() || value == nullptr)
235 return eve::Result<void>::failure(
236 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "building garrison adapter is stale or unbound"));
237 if (!object.id.isValid() || object.type.empty() || object.quantity != 1u)
238 return eve::Result<void>::failure(
239 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "building garrison object facts are invalid"));
240 GarrisonMember fallback;
241 if (payloadMember(object, fallback) == nullptr)
242 return eve::Result<void>::failure(
243 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "building garrison payload is stale"));
244 for (const auto& member : value->garrison) {
245 if (member.id == object.id.value() && (!ignoredObject || *ignoredObject != object.id))
246 return eve::Result<void>::failure(
247 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "building garrison already contains member"));
248 }
249 const std::size_t effectiveSize =
250 value->garrison.size() -
251 ((ignoredObject && std::any_of(value->garrison.begin(), value->garrison.end(),
252 [&](const auto& member) { return member.id == ignoredObject->value(); }))
253 ? 1u
254 : 0u);
255 if (destination && (!destination->isValid() || static_cast<std::size_t>(destination->value()) > effectiveSize))
256 return eve::Result<void>::failure(
257 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "building garrison slot is out of range"));
258 auto accepted = descriptor_.filter.evaluate(object);
259 if (!accepted) return accepted;
260 if (!descriptor_.capacity.isUnlimited() && effectiveSize >= descriptor_.capacity.value())
261 return eve::Result<void>::failure(
262 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "building garrison capacity is full"));
263 return eve::Result<void>::success();

266eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>> BuildingGarrisonContainerAdapter::prepare(
267 const eve::container::ContainerSnapshot& expected, const eve::container::ContainerSnapshot& candidate) {
268 const auto* value = building();
269 if (!descriptor_.id.isValid() || value == nullptr)
270 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
271 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "building garrison adapter is stale or unbound"));
272 if (expected.id != descriptor_.id || candidate.id != descriptor_.id)
273 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(eve::Diagnostic::error(
274 eve::DiagnosticCode::InvalidArgument, "building garrison prepare targets another container"));
275 auto current = snapshot();
276 if (!current)
277 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(current.status());
278 if (!sameLayout(current.value(), expected))
279 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
280 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "building garrison expected snapshot is stale"));
281 const auto next = expected.revision.incremented();
282 if (!next || candidate.revision != *next)
283 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(eve::Diagnostic::error(
284 eve::DiagnosticCode::InvariantViolation, "building garrison candidate revision is not the next revision"));
285 if (!descriptor_.capacity.isUnlimited() && candidate.entries.size() > descriptor_.capacity.value())
286 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
287 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "building garrison candidate exceeds capacity"));
288
289 std::vector<GarrisonMember> staged;
290 staged.reserve(candidate.entries.size());
291 std::unordered_set<eve::container::MembershipId> identities;
292 for (std::size_t index = 0; index < candidate.entries.size(); ++index) {
293 const auto& entry = candidate.entries[index];
294 if (!entry.membership.object.isValid() || !entry.membership.slot.isValid() ||
295 entry.membership.slot.value() != static_cast<std::int32_t>(index) || entry.membership.generation.isZero() ||
296 entry.object.id != entry.membership.object || entry.object.quantity != 1u ||
297 !identities.insert(entry.membership.object).second)
298 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
299 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
300 "building garrison candidate contains invalid membership"));
301 GarrisonMember fallback;
302 const auto* member = payloadMember(entry.object, fallback);
303 if (member == nullptr)
304 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
305 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle,
306 "building garrison candidate payload is stale"));
307 staged.push_back(*member);
308 }
309 std::unique_ptr<eve::container::IContainer::PreparedState> prepared =
310 std::make_unique<PreparedState>(*this, std::move(staged), expected.revision, expected.entries.size(), *next);
311 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::success(std::move(prepared));
312}

314eve::Result<eve::container::MembershipId> BuildingGarrisonContainerAdapter::enter(std::string memberId,
315 std::string type,
316 std::vector<std::string> tags,
317 eve::container::GameEventSink sink,
318 eve::SimulationTick tick) {
319 const auto serial = nextEventSerial();
320 const eve::container::MembershipId objectId(std::move(memberId));
321 auto current = snapshot();
322 if (!current) {
323 const auto status = current.status();
324 bool ignored = false;
325 const auto* diagnostic = status.primaryDiagnostic();
326 notify(sink, descriptor_.id, objectId, eve::container::ContainerEventKind::Rejected,
327 eve::container::SlotIndex::invalid(), revision(), serial, tick,
328 diagnostic == nullptr ? "stale_container" : diagnostic->message(), ignored);
329 return eve::Result<eve::container::MembershipId>::failure(status);
330 }
331 eve::container::ContainerObject object;
332 object.id = objectId;
333 object.type = type.empty() ? "building.garrison.member" : std::move(type);
334 object.tags = std::move(tags);
335 object.quantity = 1;
336 auto accepted = validateInsert(object, std::nullopt, std::nullopt);
337 if (!accepted) {
338 const auto status = accepted.status();
339 bool ignored = false;
340 const auto* diagnostic = status.primaryDiagnostic();
341 notify(sink, descriptor_.id, objectId, eve::container::ContainerEventKind::Rejected,
342 eve::container::SlotIndex::invalid(), current.value().revision, serial, tick,
343 diagnostic == nullptr ? "rejected" : diagnostic->message(), ignored);
344 return eve::Result<eve::container::MembershipId>::failure(status);
345 }
346 auto candidate = current.value();
347 const auto next = candidate.revision.incremented();
348 if (!next) {
349 const auto status = eve::Status::failure(
350 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation, "building garrison revision exhausted"));
351 bool ignored = false;
352 notify(sink, descriptor_.id, objectId, eve::container::ContainerEventKind::Rejected,
353 eve::container::SlotIndex::invalid(), candidate.revision, serial, tick, "revision_exhausted", ignored);
354 return eve::Result<eve::container::MembershipId>::failure(status);

356 candidate.revision = *next;
357 candidate.entries.push_back(
358 {{objectId, eve::container::SlotIndex(static_cast<std::int32_t>(current.value().entries.size())),
359 eve::Generation(1)},

362 if (!prepared) {
363 const auto status = prepared.status();
364 bool ignored = false;
365 const auto* diagnostic = status.primaryDiagnostic();
366 notify(sink, descriptor_.id, objectId, eve::container::ContainerEventKind::Rejected,
367 eve::container::SlotIndex::invalid(), current.value().revision, serial, tick,
368 diagnostic == nullptr ? "prepare_failed" : diagnostic->message(), ignored);
369 return eve::Result<eve::container::MembershipId>::failure(status);
370 }
371 auto staged = std::move(prepared).takeValue();
372 staged->commit();
373 bool callbackFailed = false;
374 const auto slot = candidate.entries.back().membership.slot;
375 notify(sink, descriptor_.id, objectId, eve::container::ContainerEventKind::Accepted, slot, *next, serial, tick, {},
376 callbackFailed);
377 notify(sink, descriptor_.id, objectId, eve::container::ContainerEventKind::Enter, slot, *next, serial, tick, {},
378 callbackFailed);
379 return eve::Result<eve::container::MembershipId>::success(objectId, appliedStatus(callbackFailed));

382eve::Result<eve::container::MembershipId> BuildingGarrisonContainerAdapter::exit(eve::container::MembershipId memberId,
383 eve::container::GameEventSink sink,
384 eve::SimulationTick tick) {
385 const auto serial = nextEventSerial();
386 auto current = snapshot();
387 eve::container::SlotIndex slot = eve::container::SlotIndex::invalid();
388 if (current) {
389 for (const auto& entry : current.value().entries) {
390 if (entry.membership.object == memberId) {
391 slot = entry.membership.slot;
392 break;
393 }
394 }
395 }
396 if (!current || !slot.isValid()) {
397 const auto status = current ? eve::Status::failure(eve::Diagnostic::error(
398 eve::DiagnosticCode::NotFound, "building garrison member was not found"))
399 : current.status();
400 bool ignored = false;
401 const auto* diagnostic = status.primaryDiagnostic();
402 notify(sink, descriptor_.id, memberId, eve::container::ContainerEventKind::Rejected, slot, revision(), serial,
403 tick, diagnostic == nullptr ? "rejected" : diagnostic->message(), ignored);
404 return eve::Result<eve::container::MembershipId>::failure(status);
405 }
406 auto candidate = current.value();
407 candidate.entries.erase(std::remove_if(candidate.entries.begin(), candidate.entries.end(),
408 [&](const auto& entry) { return entry.membership.object == memberId; }),
409 candidate.entries.end());
410 for (std::size_t index = 0; index < candidate.entries.size(); ++index)
411 candidate.entries[index].membership.slot = eve::container::SlotIndex(static_cast<std::int32_t>(index));
412 const auto next = candidate.revision.incremented();
413 if (!next) {
414 const auto status = eve::Status::failure(
415 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation, "building garrison revision exhausted"));
416 bool ignored = false;
417 notify(sink, descriptor_.id, memberId, eve::container::ContainerEventKind::Rejected, slot,
418 current.value().revision, serial, tick, "revision_exhausted", ignored);
419 return eve::Result<eve::container::MembershipId>::failure(status);
420 }
421 candidate.revision = *next;
422 auto prepared = prepare(current.value(), candidate);
423 if (!prepared) {
424 const auto status = prepared.status();
425 bool ignored = false;
426 const auto* diagnostic = status.primaryDiagnostic();
427 notify(sink, descriptor_.id, memberId, eve::container::ContainerEventKind::Rejected, slot,
428 current.value().revision, serial, tick, diagnostic == nullptr ? "prepare_failed" : diagnostic->message(),
429 ignored);
430 return eve::Result<eve::container::MembershipId>::failure(status);
431 }
432 auto staged = std::move(prepared).takeValue();
433 staged->commit();
434 bool callbackFailed = false;
435 notify(sink, descriptor_.id, memberId, eve::container::ContainerEventKind::Accepted, slot, *next, serial, tick, {},
436 callbackFailed);
