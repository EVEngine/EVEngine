105std::string callSceneSearch(std::string_view name, Object::Ptr args) try {
106 if (!args) throw std::invalid_argument("arguments required");
107 std::set<std::string> allowed;
108 if (name == "eve_scene_search_prepare")
109 allowed = {"host"};
110 else if (name == "eve_scene_search_status")
111 allowed = {};
112 else if (name == "eve_scene_search_publish")
113 allowed = {"snapshot", "model", "vectors"};
114 else if (name == "eve_scene_search")
115 allowed = {"snapshot", "model", "embedding", "kind", "space", "tags", "visible", "layer",
116 "center", "radius", "limit", "scanBudget", "pageToken", "spatial", "candidates"};
117 else
118 return failed("unknown_tool", "Unknown scene search tool");
119 for (const auto& entry : *args)
120 if (!allowed.contains(entry.first)) throw std::invalid_argument("unknown field: " + entry.first);
121 auto* provider = cap::query<ISceneSearch>();
122 if (!provider) return failed("scene-unavailable", "Scene module is absent or has not been initialized");
123 auto out = envelope();
124 if (name == "eve_scene_search_prepare" || name == "eve_scene_search_status") {
125 auto value =
126 name == "eve_scene_search_prepare" ? provider->prepare(string(args, "host", true)) : provider->status();
127 if (!value.ok()) return failed(value.status());
128 const auto& snapshot = value.value();
129 out->set("snapshot", snapshot.token);
130 out->set("host", snapshot.host);
131 out->set("model", snapshot.model);
132 out->set("backend", snapshot.model.empty() ? "none" : "hnswlib-0.8.0");
133 out->set("spatialContract",
134 "Node origins; 3D world +X right, +Y up, +Z forward; 2D +Y down. Node frames use local +Z forward, "
135 "not Camera3D view -Z. Use frame=observer with an explicit observer node to resolve perspective. "
136 "inside means origin in authored world AABB, not mesh containment. Resolve ambiguous anchor names "
137 "with semantic search first; spatial.reference and observer are exact node IDs.");
138 Array::Ptr records = new Array;
139 for (const auto& item : snapshot.records) records->add(record(item));
140 out->set("records", records);
141 } else if (name == "eve_scene_search_publish") {
142 SceneEmbeddingBatch batch;
143 batch.snapshot = string(args, "snapshot", true);
144 batch.model = string(args, "model", true);
145 const auto values = array(args, "vectors", 4096);
146 // Bound aggregate payload work before allocating normalized copies.
147 std::size_t scalars = 0;
148 for (std::size_t i = 0; i < values->size(); ++i) {
149 auto row = values->getArray(unsigned(i));
150 if (!row || (scalars += row->size()) > 4194304)
151 throw std::invalid_argument("embedding scalar budget exceeded");
152 batch.vectors.push_back(vector(row));
153 }
154 auto result = provider->publish(batch);
155 if (!result.ok()) return failed(result.status());
156 out->set("snapshot", batch.snapshot);
157 out->set("model", batch.model);
158 out->set("backend", "hnswlib-0.8.0");
159 } else {
160 SceneSearchQuery query;
161 query.snapshot = string(args, "snapshot", true);
162 query.model = string(args, "model");
163 query.kind = string(args, "kind");
164 query.space = string(args, "space");
165 query.pageToken = string(args, "pageToken");
166 query.limit = std::size_t(integer(args, "limit", 20, 1, 256));
167 query.scanBudget = std::size_t(integer(args, "scanBudget", 4096, 1, 4096));
168 query.candidates = std::size_t(integer(args, "candidates", 128, 1, 4096));
169 if (args->has("spatial")) {
170 auto spatial = array(args, "spatial", 16);
171 for (std::size_t i = 0; i < spatial->size(); ++i) {
172 auto item = spatial->getObject(unsigned(i));
173 if (!item) throw std::invalid_argument("spatial predicates must be objects");
174 for (const auto& entry : *item)
175 if (!std::set<std::string>{"relation", "reference", "frame", "observer", "distance", "margin"}
176 .contains(entry.first))
177 throw std::invalid_argument("unknown spatial field: " + entry.first);
178 SceneSpatialConstraint constraint;
179 const auto relation = string(item, "relation", true);
180 const auto frame = item->has("frame") ? string(item, "frame", true) : "world";
181 const auto relationAt =
182 std::find(sceneSpatialRelationNames.begin(), sceneSpatialRelationNames.end(), relation);
183 const auto frameAt = std::find(sceneSpatialFrameNames.begin(), sceneSpatialFrameNames.end(), frame);
184 if (relationAt == sceneSpatialRelationNames.end() || frameAt == sceneSpatialFrameNames.end())
185 throw std::invalid_argument("unknown spatial relation/frame");
186 constraint.relation = SceneSpatialRelation(relationAt - sceneSpatialRelationNames.begin());
187 constraint.frame = SceneSpatialFrame(frameAt - sceneSpatialFrameNames.begin());
188 constraint.reference = string(item, "reference", true);
189 constraint.observer = string(item, "observer");
190 if (item->has("distance")) constraint.distance = number(item->get("distance"));
191 if (item->has("margin")) constraint.margin = number(item->get("margin"));
192 query.spatial.push_back(std::move(constraint));
193 }
194 }
195 if (args->has("visible")) {
196 if (!args->get("visible").isBoolean()) throw std::invalid_argument("visible must be boolean");
197 query.visible = args->getValue<bool>("visible");
198 }
199 if (args->has("layer")) query.layer = integer(args, "layer", 0, -2147483647, 2147483647);
200 if (args->has("embedding")) query.embedding = vector(array(args, "embedding", 1024));
201 if (args->has("center") != args->has("radius"))
202 throw std::invalid_argument("center and radius must be supplied together");
203 if (args->has("center")) {
204 const auto center = vector(array(args, "center", 3));
205 if (center.size() != 3 || !args->has("radius"))
206 throw std::invalid_argument("center requires three axes and radius");
207 query.center = std::array<double, 3>{center[0], center[1], center[2]};
208 }
209 if (args->has("radius")) query.radius = number(args->get("radius"));
210 if (args->has("tags")) {
211 auto tags = array(args, "tags", 64);
212 for (const auto& tag : *tags) {
213 if (!tag.isString() || tag.convert<std::string>().size() > 256)
214 throw std::invalid_argument("invalid tag");
215 query.tags.push_back(tag.convert<std::string>());
216 }
217 }
218 auto value = provider->query(query);
219 if (!value.ok()) return failed(value.status());
220 const auto& result = value.value();
221 out->set("snapshot", result.snapshot);
222 out->set("indexSnapshot", result.indexSnapshot);
223 out->set("model", result.model);
224 out->set("scanned", Poco::UInt64(result.scanned));
225 out->set("matched", Poco::UInt64(result.matched));
226 out->set("pageToken", result.pageToken);
227 out->set("truncated", !result.pageToken.empty());
228 out->set("consistency", "source-validated");
229 out->set("backend", query.embedding.empty() ? "structured" : "hnswlib-0.8.0");
230 out->set("approximate", result.approximate);
231 out->set("candidateTruncated", result.candidateTruncated);
232 Array::Ptr hits = new Array;
233 for (const auto& hit : result.hits) {
234 auto item = record(hit.record);
235 item->set("score", hit.score);
236 item->set("evidence", query.embedding.empty() ? "exact-filters" : "hnsw-cosine+exact-filters");
237 Array::Ptr spatial = new Array;
238 for (const auto& measured : hit.spatial) {
239 Object::Ptr evidence = new Object;
240 evidence->set("relation", std::string(sceneSpatialRelationNames[std::size_t(measured.relation)]));
241 evidence->set("frame", std::string(sceneSpatialFrameNames[std::size_t(measured.frame)]));
242 evidence->set("reference", measured.reference);
243 evidence->set("observer", measured.observer);
244 evidence->set("distance", measured.distance);
245 Array::Ptr offset = new Array;
246 for (double value : measured.offset) offset->add(value);
247 evidence->set("offsetRightUpForward", offset);
248 spatial->add(evidence);
249 }
250 item->set("spatialEvidence", spatial);
251 hits->add(item);
252 }
253 out->set("hits", hits);
254 }
255 return textContentResult(mcpStringify(out));
256} catch (const std::exception& error) {
257 return failed("invalid_argument", error.what());
258}
