380ssq::Table makeOwnedProxy(HSQUIRRELVM vm, eve::Result<Ref>&& reference, Release&& release) {
381 if (!reference) return eve::script::projectStatusResult(vm, reference.status(), false, false);
382 const Ref ref = std::move(reference).takeValue();
383 auto object = eve::script::makeOwnedSquirrelInstance<Proxy>(vm, std::make_unique<Proxy>(ref));
384 if (!object) {
385 const eve::Status status = object.status();
386 object.ignore("failed to create owned climbing proxy");
387 std::invoke(std::forward<Release>(release), ref).ignore("rollback failed climbing allocation");
388 return eve::script::projectStatusResult(vm, status, false, false);
389 }
390 ssq::Object owned = std::move(object).takeValue();
391 auto result = eve::script::projectStatusResult(vm, eve::Status::success(eve::StatusCode::Applied), true, false);
392 result.set("value", owned);
393 result.set("ownership", std::string("owned"));
394 result.set("ownerEpoch", static_cast<std::int64_t>(ref.ownerEpoch));
395 result.set("handle", static_cast<std::int64_t>(ref.packed()));
396 return result;

420 const HSQUIRRELVM vm = table.getHandle();
421 auto graph = table.addClass<ScriptClimbingAnchorGraph>(
422 "ClimbingAnchorGraph", std::function<ScriptClimbingAnchorGraph*()>([]() { return nullptr; }), false);
423 graph.addFunc("ownership", [](ScriptClimbingAnchorGraph*) { return std::string("owned"); });

432 return eve::script::projectResult(vm, Climbing::releaseAnchorGraph(value->reference));
433 });

435 if (!value)
436 return eve::script::projectResult(
437 vm, bindingFailure<ClimbingAnchorNodeRef>(eve::DiagnosticCode::InvalidArgument,
438 "climbing anchor graph proxy must not be null",
439 "anchorGraph"),
440 anchorNodeValue);

442 return eve::script::projectResult(
443 vm,
444 resolved.isBound()
445 ? resolved->nodeRef(nodeId)
446 : bindingFailure<ClimbingAnchorNodeRef>(eve::DiagnosticCode::StaleHandle,

454 "climbing anchor graph proxy must not be null",
455 "anchorGraph"),

458 if (!parsed)
459 return eve::script::projectResult(

461 auto definition = decodeClimbingAnchorGraphDefinition(parsed.value());
462 if (!definition)
463 return eve::script::projectResult(
464 vm, eve::Result<ClimbingAnchorGraphReload>::failure(definition.status()), graphReloadValue);
465 auto resolved = Climbing::resolveAnchorGraph(value->reference);
466 return eve::script::projectResult(
467 vm,
468 resolved.isBound()
469 ? resolved->reload(std::move(definition).takeValue())
470 : bindingFailure<ClimbingAnchorGraphReload>(eve::DiagnosticCode::StaleHandle,
471 "climbing anchor graph handle is stale", "anchorGraph"),
472 graphReloadValue);
473 });

475 const std::string& goalNodeId, bool avoidFull, std::int64_t agentId,
476 std::int64_t executionId, std::int64_t maxVisitedNodes) {
477 if (!value || agentId < 0 || executionId < 0 || (agentId == 0) != (executionId == 0) ||
478 maxVisitedNodes <= 0 || maxVisitedNodes > 65536)
479 return eve::script::projectResult(
480 vm, bindingFailure<ClimbingAnchorRoute>(eve::DiagnosticCode::InvalidArgument,
481 "graph, paired non-negative owner ids, and route bound are required",
482 "planRoute"),
483 anchorRouteValue);
484 auto resolved = Climbing::resolveAnchorGraph(value->reference);
485 if (!resolved.isBound())
486 return eve::script::projectResult(
487 vm, bindingFailure<ClimbingAnchorRoute>(eve::DiagnosticCode::StaleHandle,
488 "climbing anchor graph handle is stale", "anchorGraph"),
489 anchorRouteValue);
490 auto start = resolved->nodeRef(startNodeId);
491 if (!start)
492 return eve::script::projectResult(vm, eve::Result<ClimbingAnchorRoute>::failure(start.status()),
493 anchorRouteValue);
