53void rasterizeCrossSection(const std::vector<Tri2D> &tris, float dc, int w, int h,
54 uint8_t *rgba) {
55 const size_t pixelCount = size_t(w) * size_t(h);
56 std::vector<std::vector<float>> hits(pixelCount);
57 std::vector<float> bestY(pixelCount, 1e30f);
58 std::vector<uint32_t> bestRGB(pixelCount, 0xffffffu);
59
60 for (const auto &tri : tris) {
61 const glm::vec2 &a = tri.px[0];
62 const glm::vec2 &b = tri.px[1];
63 const glm::vec2 &c = tri.px[2];
64 int x0 = int(std::floor(std::min(a.x, std::min(b.x, c.x))));
65 int x1 = int(std::ceil(std::max(a.x, std::max(b.x, c.x))));
66 int y0 = int(std::floor(std::min(a.y, std::min(b.y, c.y))));
67 int y1 = int(std::ceil(std::max(a.y, std::max(b.y, c.y))));
68 x0 = std::max(0, x0);
69 y0 = std::max(0, y0);
70 x1 = std::min(w - 1, x1);
71 y1 = std::min(h - 1, y1);
72
73 const glm::vec2 e0 = b - a;
74 const glm::vec2 e1 = c - a;
75 const float denom = e0.x * e1.y - e0.y * e1.x;
76 if (std::fabs(denom) < 1e-12f) continue;
77
78 for (int y = y0; y <= y1; ++y) {
79 for (int x = x0; x <= x1; ++x) {
80 const glm::vec2 p(float(x) + 0.5f, float(y) + 0.5f);
81 const glm::vec2 d = p - a;
82 const float u = (d.x * e1.y - d.y * e1.x) / denom;
83 const float v = (e0.x * d.y - e0.y * d.x) / denom;
84 if (u < -1e-4f || v < -1e-4f || u + v > 1.f + 1e-4f) continue;
85 const float w0 = 1.f - u - v;
86 const float depthAt = w0 * tri.depth[0] + u * tri.depth[1] + v * tri.depth[2];
87 if (depthAt <= dc) continue;
88
89 const size_t i = size_t(y) * size_t(w) + size_t(x);
90 hits[i].push_back(depthAt);
91 if (depthAt < bestY[i]) {
92 bestY[i] = depthAt;
93 bestRGB[i] = (uint32_t(clampf(tri.color.r, 0.f, 1.f) * 255.f) << 16) |
94 (uint32_t(clampf(tri.color.g, 0.f, 1.f) * 255.f) << 8) |
95 uint32_t(clampf(tri.color.b, 0.f, 1.f) * 255.f);
96 }
97 }
98 }
99 }
100
101 std::vector<float> uniqueHits;
102 for (size_t i = 0; i < pixelCount; ++i) {
103 if (hits[i].empty()) continue;
104 std::sort(hits[i].begin(), hits[i].end());
105 uniqueHits.clear();
106 for (float d : hits[i]) {
107 // Shared edges/vertices report the same surface with slightly
108 // different interpolated depths (float weight sums); group them.
109 if (uniqueHits.empty() ||
110 d - uniqueHits.back() > 1e-5f * std::max(1.f, std::fabs(d)))
111 uniqueHits.push_back(d);
112 }
113 if (uniqueHits.size() % 2 == 0) continue; // even = outside the solid
114 uint8_t *px = rgba + i * 4;
115 px[0] = uint8_t(bestRGB[i] >> 16);
116 px[1] = uint8_t(bestRGB[i] >> 8);
117 px[2] = uint8_t(bestRGB[i]);
118 px[3] = 255;
119 }
120}

122std::vector<image::ImageData *> sliceArrays(const float *pos, const float *nrm, const float *rgb,
123 int vertexCount, const uint32_t *indices, int indexCount,
124 const SliceOptions &opt) {
125 if (!pos || vertexCount < 3 || !indices || indexCount < 3)
126 throw eve::Exception("SpriteStack.sliceMesh: invalid mesh arrays");
127 if (opt.layerCount <= 0) throw eve::Exception("SpriteStack.sliceMesh: layerCount must be > 0");
128 if (opt.imageW <= 0 || opt.imageH <= 0)
129 throw eve::Exception("SpriteStack.sliceMesh: image size must be > 0");
130
131 const Projection proj = projectionForAxis(opt.axis);
132
133 glm::vec3 mn(1e30f), mx(-1e30f);
134 for (int i = 0; i < vertexCount; ++i) {
135 const glm::vec3 p(pos[i * 3], pos[i * 3 + 1], pos[i * 3 + 2]);
136 mn = glm::min(mn, p);
137 mx = glm::max(mx, p);
138 }
139 if (mn.x > mx.x) throw eve::Exception("SpriteStack.sliceMesh: empty mesh AABB");
140
141 const float d0 = mn[proj.d];
142 const float d1 = opt.thickness > 0.f ? d0 + opt.thickness * float(opt.layerCount - 1)
143 : mx[proj.d];
144 const float slab = opt.thickness > 0.f ? opt.thickness : (d1 - d0) / float(opt.layerCount);
145 if (slab <= 0.f) throw eve::Exception("SpriteStack.sliceMesh: zero slice thickness");
146
147 const float uMin = mn[proj.u] - opt.padding * std::max(1.f, mx[proj.u] - mn[proj.u]);
148 const float uMax = mx[proj.u] + opt.padding * std::max(1.f, mx[proj.u] - mn[proj.u]);
149 const float vMin = mn[proj.v] - opt.padding * std::max(1.f, mx[proj.v] - mn[proj.v]);
150 const float vMax = mx[proj.v] + opt.padding * std::max(1.f, mx[proj.v] - mn[proj.v]);
151 const float uSpan = uMax - uMin;
152 const float vSpan = vMax - vMin;
153 if (uSpan <= 0.f || vSpan <= 0.f)
154 throw eve::Exception("SpriteStack.sliceMesh: degenerate projected AABB");
155
156 const glm::vec3 viewDir =
157 proj.d == 0 ? glm::vec3(1.f, 0.f, 0.f)
158 : (proj.d == 1 ? glm::vec3(0.f, 1.f, 0.f) : glm::vec3(0.f, 0.f, 1.f));
159
160 std::vector<image::ImageData *> layers;
161 layers.reserve(size_t(opt.layerCount));
162 const int triangleCount = indexCount / 3;
163 std::vector<Tri2D> tris;
164 tris.reserve(size_t(triangleCount));
165
166 for (int t = 0; t < triangleCount; ++t) {
167 const uint32_t i0 = indices[t * 3];
168 const uint32_t i1 = indices[t * 3 + 1];
169 const uint32_t i2 = indices[t * 3 + 2];
170 if (int(i0) >= vertexCount || int(i1) >= vertexCount || int(i2) >= vertexCount) continue;
171
172 const glm::vec3 a(pos[i0 * 3], pos[i0 * 3 + 1], pos[i0 * 3 + 2]);
173 const glm::vec3 b(pos[i1 * 3], pos[i1 * 3 + 1], pos[i1 * 3 + 2]);
174 const glm::vec3 c(pos[i2 * 3], pos[i2 * 3 + 1], pos[i2 * 3 + 2]);
175
176 image::ImageData::Colorf color{opt.tintR, opt.tintG, opt.tintB, 1.f};
177 if (rgb) {
178 const float rr = (rgb[i0 * 3] + rgb[i1 * 3] + rgb[i2 * 3]) / 3.f;
179 const float gg = (rgb[i0 * 3 + 1] + rgb[i1 * 3 + 1] + rgb[i2 * 3 + 1]) / 3.f;
180 const float bb = (rgb[i0 * 3 + 2] + rgb[i1 * 3 + 2] + rgb[i2 * 3 + 2]) / 3.f;
181 color.r *= rr;
182 color.g *= gg;
183 color.b *= bb;
184 }
185 if (opt.shade) {
186 glm::vec3 n = glm::cross(b - a, c - a);
187 if (glm::dot(n, n) > 1e-12f) {
188 n = glm::normalize(n);
189 const float s = 0.72f + 0.28f * std::abs(glm::dot(n, viewDir));
190 color.r *= s;
191 color.g *= s;
192 color.b *= s;
193 }
194 }
195
196 Tri2D tri;
197 tri.px[0] = glm::vec2((a[proj.u] - uMin) / uSpan * float(opt.imageW - 1),
198 (1.f - (a[proj.v] - vMin) / vSpan) * float(opt.imageH - 1));
199 tri.px[1] = glm::vec2((b[proj.u] - uMin) / uSpan * float(opt.imageW - 1),
200 (1.f - (b[proj.v] - vMin) / vSpan) * float(opt.imageH - 1));
201 tri.px[2] = glm::vec2((c[proj.u] - uMin) / uSpan * float(opt.imageW - 1),
202 (1.f - (c[proj.v] - vMin) / vSpan) * float(opt.imageH - 1));
203 tri.depth[0] = a[proj.d];
204 tri.depth[1] = b[proj.d];
205 tri.depth[2] = c[proj.d];
206 tri.color = color;
207 tris.push_back(tri);
208 }
209
210 for (int li = 0; li < opt.layerCount; ++li) {
211 auto *img = new image::ImageData(opt.imageW, opt.imageH, "RGBA8");
212 auto *data = static_cast<uint8_t *>(img->getData());
213 std::memset(data, 0, size_t(opt.imageW) * size_t(opt.imageH) * 4);
214 const float dc = d0 + (float(li) + 0.5f) * slab;
215 rasterizeCrossSection(tris, dc, opt.imageW, opt.imageH, data);
216 layers.push_back(img);
217 }
218 return layers;
219}

221void makeBox(std::vector<float> &pos, std::vector<float> &nrm, std::vector<uint32_t> &idx) {
222 const float h = 0.5f;
223 const glm::vec3 corners[8] = {
224 {-h, -h, -h}, {h, -h, -h}, {h, h, -h}, {-h, h, -h},
225 {-h, -h, h}, {h, -h, h}, {h, h, h}, {-h, h, h},
226 };
227 const glm::vec3 faces[6] = {
228 {0.f, 0.f, -1.f}, {0.f, 0.f, 1.f}, {-1.f, 0.f, 0.f},
229 {1.f, 0.f, 0.f}, {0.f, -1.f, 0.f}, {0.f, 1.f, 0.f},
230 };
231 const int quads[6][4] = {
232 {0, 1, 2, 3}, {5, 4, 7, 6}, {4, 0, 3, 7},
233 {1, 5, 6, 2}, {4, 5, 1, 0}, {3, 2, 6, 7},
234 };
235 for (int f = 0; f < 6; ++f) {
236 for (int k = 0; k < 4; ++k) {
237 const glm::vec3 &p = corners[quads[f][k]];
238 pos.insert(pos.end(), {p.x, p.y, p.z});
239 nrm.insert(nrm.end(), {faces[f].x, faces[f].y, faces[f].z});
240 }
241 const uint32_t base = uint32_t(f * 4);
242 idx.insert(idx.end(), {base, base + 1, base + 2, base, base + 2, base + 3});
243 }
244}

246void makeLathe(const std::string &kind, int slices, std::vector<float> &pos,
247 std::vector<float> &nrm, std::vector<uint32_t> &idx) {
248 const bool sphere = kind == "sphere";
249 const bool cone = kind == "cone";
250 const int stacks = sphere ? 12 : 1;
251 auto ring = [&](float y, float radius, float nY) {
252 // Push `slices` vertices forming one horizontal ring.
253 const uint32_t base = uint32_t(pos.size() / 3);
254 for (int s = 0; s < slices; ++s) {
255 const float a = float(s) / float(slices) * 2.f * kPi;
256 const glm::vec3 n(std::cos(a), nY, std::sin(a));
257 pos.insert(pos.end(), {std::cos(a) * radius, y, std::sin(a) * radius});
258 nrm.insert(nrm.end(), {n.x, n.y, n.z});
259 }
260 return base;
261 };
262 auto connectRings = [&](uint32_t r0, uint32_t r1) {
263 for (int s = 0; s < slices; ++s) {
264 const uint32_t s0 = r0 + uint32_t(s);
265 const uint32_t s1 = r0 + uint32_t((s + 1) % slices);
266 const uint32_t t0 = r1 + uint32_t(s);
267 const uint32_t t1 = r1 + uint32_t((s + 1) % slices);
268 idx.insert(idx.end(), {s0, t0, t1, s0, t1, s1});
269 }
270 };
271 auto cap = [&](float y, float radius, float nY) {
272 const uint32_t center = uint32_t(pos.size() / 3);
273 pos.insert(pos.end(), {0.f, y, 0.f});
274 nrm.insert(nrm.end(), {0.f, nY, 0.f});
275 for (int s = 0; s < slices; ++s) {
276 const float a = float(s) / float(slices) * 2.f * kPi;
277 pos.insert(pos.end(), {std::cos(a) * radius, y, std::sin(a) * radius});
278 nrm.insert(nrm.end(), {0.f, nY, 0.f});
279 const uint32_t ring0 = center + 1;
280 const uint32_t a0 = ring0 + uint32_t(s);
281 const uint32_t b0 = ring0 + uint32_t((s + 1) % slices);
282 if (nY > 0.f)
283 idx.insert(idx.end(), {center, b0, a0});
284 else
285 idx.insert(idx.end(), {center, a0, b0});
286 }
287 };
288
289 if (sphere) {
290 std::vector<uint32_t> rings;
291 for (int st = 0; st < stacks; ++st) {
292 const float t0 = float(st) / float(stacks);
293 const float y0 = std::cos(t0 * kPi) * 0.5f;
294 const float r0 = std::sin(t0 * kPi) * 0.5f;
295 rings.push_back(ring(y0, r0, -std::sin(t0 * kPi)));
296 }
297 for (int st = 0; st + 1 < stacks; ++st)
298 connectRings(rings[size_t(st)], rings[size_t(st + 1)]);
299 return;
300 }
301
302 if (cone) {
303 const uint32_t apex = uint32_t(pos.size() / 3);
304 pos.insert(pos.end(), {0.f, 0.5f, 0.f});
305 nrm.insert(nrm.end(), {0.f, 1.f, 0.f});
306 for (int s = 0; s < slices; ++s) {
307 const float a0 = float(s) / float(slices) * 2.f * kPi;
308 const float a1 = float(s + 1) / float(slices) * 2.f * kPi;
309 const uint32_t b0 = uint32_t(pos.size() / 3);
310 pos.insert(pos.end(), {std::cos(a0) * 0.5f, -0.5f, std::sin(a0) * 0.5f});
311 nrm.insert(nrm.end(), {std::cos(a0) * 0.7071f, 0.7071f, std::sin(a0) * 0.7071f});
312 pos.insert(pos.end(), {std::cos(a1) * 0.5f, -0.5f, std::sin(a1) * 0.5f});
313 nrm.insert(nrm.end(), {std::cos(a1) * 0.7071f, 0.7071f, std::sin(a1) * 0.7071f});
314 idx.insert(idx.end(), {apex, b0, b0 + 1});
315 }
316 cap(-0.5f, 0.5f, -1.f);
317 return;
318 }
319
320 // cylinder
321 const uint32_t bottom = ring(-0.5f, 0.5f, 0.f);
322 const uint32_t top = ring(0.5f, 0.5f, 0.f);
323 connectRings(bottom, top);
324 cap(0.5f, 0.5f, 1.f);
325 cap(-0.5f, 0.5f, -1.f);
326}

335std::vector<image::ImageData *> sliceModelToLayers(model3d::ModelData *model,
336 const SliceOptions &opt) {
337 if (!model) throw eve::Exception("SpriteStack.sliceModel: null model");
338 std::vector<float> pos, nrm, rgb;
339 std::vector<uint32_t> idx;
340 const int meshCount = model->getMeshCount();
341 for (int m = 0; m < meshCount; ++m) {
342 const aiMesh *am = model->getMesh(m);
343 if (!am) continue;
344 const uint32_t base = uint32_t(pos.size() / 3);
345 for (unsigned v = 0; v < am->mNumVertices; ++v) {
346 const aiVector3D &p = am->mVertices[v];
347 pos.insert(pos.end(), {p.x, p.y, p.z});
348 if (am->mNormals) {
349 const aiVector3D &n = am->mNormals[v];
350 nrm.insert(nrm.end(), {n.x, n.y, n.z});
351 }
352 if (am->mColors[0] != nullptr) {
353 const aiColor4D &c = am->mColors[0][v];
354 rgb.insert(rgb.end(), {c.r, c.g, c.b});
355 }
356 }
357 for (unsigned f = 0; f < am->mNumFaces; ++f) {
358 const auto &face = am->mFaces[f];
359 if (face.mNumIndices != 3) continue;
360 idx.insert(idx.end(), {base + face.mIndices[0], base + face.mIndices[1],
361 base + face.mIndices[2]});
362 }
363 }
364 SliceInput in{};
365 in.posXYZ = pos.data();
366 in.nrmXYZ = nrm.empty() ? nullptr : nrm.data();
367 in.rgb = rgb.empty() ? nullptr : rgb.data();
368 in.vertexCount = int(pos.size() / 3);
369 in.indices = idx.data();
370 in.indexCount = int(idx.size());
371 return sliceArrays(in.posXYZ, in.nrmXYZ, in.rgb, in.vertexCount, in.indices, in.indexCount, opt);
372}
