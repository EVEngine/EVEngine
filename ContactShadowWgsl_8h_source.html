33fn march_contact(view_proj: mat4x4f, origin: vec3f, light: vec3f, ray_len: f32, thickness: f32,
34 near_z: f32, far_z: f32, steps: i32, dither: f32) -> f32 {
35 let step_size = ray_len / f32(steps);
36 var prev_t = 0.0;
37 var hit_t = -1.0;
38 var hit_delta = 0.0;
39 let start_t = step_size * (0.35 + dither * 0.25);
40 for (var s = 0; s < steps; s = s + 1) {
41 let t = start_t + step_size * f32(s);
42 if (t > ray_len) { break; }
43 let sample_pos = origin + light * t;
44 let clip = view_proj * vec4f(sample_pos, 1.0);
45 if (abs(clip.w) < 0.000001) { prev_t = t; continue; }
46 let ndc = clip.xyz / clip.w;
47 let uv = ndc.xy * 0.5 + 0.5;
48 if (uv.x < 0.0 || uv.x > 1.0 || uv.y < 0.0 || uv.y > 1.0) { break; }
49 let scene_z = depth(uv);
50 if (scene_z >= 0.9999) { prev_t = t; continue; }
51 let scene_view = ndc_to_view_z(scene_z, near_z, far_z);
52 let ray_view = ndc_to_view_z(clamp(ndc.z, 0.0, 1.0), near_z, far_z);
53 let delta = ray_view - scene_view;
54 if (delta > 0.0 && delta < thickness) {
55 hit_t = t;
56 hit_delta = delta;
57 var lo = prev_t;
58 var hi = t;
59 for (var r = 0; r < 6; r = r + 1) {
60 let mid = mix(lo, hi, 0.5);
61 let mid_pos = origin + light * mid;
62 let mid_clip = view_proj * vec4f(mid_pos, 1.0);
63 if (abs(mid_clip.w) < 0.000001) { break; }
64 let mid_ndc = mid_clip.xyz / mid_clip.w;
65 let mid_uv = mid_ndc.xy * 0.5 + 0.5;
66 let mid_scene = ndc_to_view_z(depth(mid_uv), near_z, far_z);
67 let mid_ray = ndc_to_view_z(clamp(mid_ndc.z, 0.0, 1.0), near_z, far_z);
68 let mid_delta = mid_ray - mid_scene;
69 if (mid_delta > 0.0 && mid_delta < thickness) {
70 hi = mid; hit_t = mid; hit_delta = mid_delta;
71 } else {
72 lo = mid;
73 }
74 }
75 break;
76 }
77 prev_t = t;
78 }
79 if (hit_t < 0.0) { return -1.0; }
80 let dist_u = clamp(hit_t / ray_len, 0.0, 1.0);
81 let thick_u = clamp(hit_delta / thickness, 0.0, 1.0);
82 let dist_fade = 1.0 - (dist_u * dist_u * (3.0 - 2.0 * dist_u));
83 let soft = 1.0 - (thick_u * thick_u * (3.0 - 2.0 * thick_u));
84 return dist_fade * soft;
85}

124 let tangent = normalize(cross(light, up_ref));
