44Result<Assessment> assess(const Problem& p, const std::vector<double>& values, const std::vector<std::uint64_t>& seeds,
45 IEvaluator& evaluator, std::uint64_t& evaluations) {
46 Assessment a;
47 a.metrics.resize(p.targets.size());
48 std::size_t count = 0;
49 for (auto seed : seeds) {
50 auto sample = [&]() -> Result<std::vector<double>> {
51 try {
52 return evaluator.evaluate(values, seed, p.dt, p.steps);
53 } catch (const std::exception& error) {
54 return Result<std::vector<double>>::failure(
55 Diagnostic::error(DiagnosticCode::CallbackFailure, error.what(), {}, {}, "agent.balance"));
56 }
57 }();
58 ++evaluations;
59 if (!sample) return Result<Assessment>::failure(sample.status());
60 if (sample.value().size() != p.targets.size())
61 return Result<Assessment>::failure(invalid("Evaluator metric count differs from targets"));
62 ++count;
63 for (std::size_t i = 0; i < a.metrics.size(); ++i) {
64 const double value = sample.value()[i];
65 if (!finite(value))
66 return Result<Assessment>::failure(invalid("Evaluator returned nonfinite/oversized metric"));
67 auto& m = a.metrics[i];
68 const double delta = value - m.mean;
69 m.mean += delta / double(count);
70 m.deviation += delta * (value - m.mean);
71 }
72 }
73 for (std::size_t i = 0; i < a.metrics.size(); ++i) {
74 auto& m = a.metrics[i];
75 m.deviation = count > 1 ? std::sqrt(std::max(0.0, m.deviation / double(count - 1))) : 0;
76 const auto& t = p.targets[i];
77 const double distance = (m.mean - std::clamp(m.mean, t.minimum, t.maximum)) / t.scale;
78 a.loss += t.weight * distance * distance;
79 }
80 double drift = 0;
81 for (std::size_t i = 0; i < values.size(); ++i) {
82 const auto& x = p.parameters[i];
83 if (x.maximum == x.minimum) continue;
84 const double delta = (values[i] - x.initial) / (x.maximum - x.minimum);
85 drift += delta * delta;
86 }
87 a.score = a.loss + p.changeWeight * drift / double(values.size());
88 return Result<Assessment>::success(std::move(a));
89}

99Result<Report> BalanceSearch::run(IEvaluator& evaluator) {
100 if (active_) return Result<Report>::failure(Diagnostic::error(DiagnosticCode::Conflict, "Balance search reentry"));
101 if (!problem_) return Result<Report>::failure(invalid("Configure the balance search before run"));
102 struct Guard {
103 bool& active;
104 explicit Guard(bool& a) : active(a) { active = true; }
105 ~Guard() { active = false; }
106 } guard(active_);
107 Report report;
108 report.problem = *problem_;
109 const auto& p = report.problem;
110 std::vector<double> initial;
111 for (const auto& x : p.parameters) initial.push_back(x.initial);
112 report.parameters = initial;
113 auto baseline = assess(p, initial, p.trainingSeeds, evaluator, report.evaluations);
114 if (!baseline) return Result<Report>::failure(baseline.status());
115 report.baseline = std::move(baseline).takeValue();
116 report.tuned = report.baseline;
117 detail::Random random(p.searchSeed);
118 for (std::uint32_t trial = 1; trial < p.candidates; ++trial) {
119 auto values = report.parameters;
120 if (trial <= 2) {
121 // Boundary anchors expose interactions that small local mutations cannot reach.
122 for (std::size_t i = 0; i < values.size(); ++i)
123 values[i] = trial == 1 ? p.parameters[i].minimum : p.parameters[i].maximum;
124 } else if (trial == 3 && values.size() >= 4) {
125 // Probe a sparse corner: the outer parameters stay at their initial values.
126 // This can cross a coupled plateau that single-coordinate repairs cannot.
127 for (std::size_t i = 0; i < values.size(); ++i) values[i] = p.parameters[i].minimum;
128 values.front() = p.parameters.front().initial;
129 values.back() = p.parameters.back().initial;
130 } else if (report.tuned.loss == 0 && trial - 3 < values.size()) {
131 // Once feasible, spend bounded probes restoring one parameter at a time.
132 const auto index = std::size_t(trial - 3);
133 values[index] = p.parameters[index].initial;
134 } else {
135 const bool explore = random.unit() < 0.25;
136 const double radius = 0.25 * (1.0 - double(trial) / double(p.candidates)) + 0.01;
137 for (std::size_t i = 0; i < values.size(); ++i) {
138 const auto& x = p.parameters[i];
139 const double width = x.maximum - x.minimum;
140 values[i] =
141 explore ? x.minimum + random.unit() * width
142 : std::clamp(values[i] + (random.unit() * 2 - 1) * width * radius, x.minimum, x.maximum);
143 }
144 }
145 auto measured = assess(p, values, p.trainingSeeds, evaluator, report.evaluations);
146 if (!measured) return Result<Report>::failure(measured.status());
147 if (measured.value().score < report.tuned.score) {
148 report.parameters = std::move(values);
149 report.tuned = std::move(measured).takeValue();
150 }
151 }
152 auto before = assess(p, initial, p.validationSeeds, evaluator, report.evaluations);
153 if (!before) return Result<Report>::failure(before.status());
154 report.baselineValidation = std::move(before).takeValue();
155 auto after = assess(p, report.parameters, p.validationSeeds, evaluator, report.evaluations);
156 if (!after) return Result<Report>::failure(after.status());
157 report.validation = std::move(after).takeValue();
158 report.verdict = Verdict::WithinTargets;
159 for (std::size_t i = 0; i < p.targets.size(); ++i) {
160 const double mean = report.validation.metrics[i].mean;
161 if (mean < p.targets[i].minimum || mean > p.targets[i].maximum) report.verdict = Verdict::OutsideTargets;
162 }
163 return Result<Report>::success(std::move(report));
164}
