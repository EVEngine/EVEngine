146class ScriptVehicleDamageModifier final : public IVehicleDamageModifier {
147public:
148 explicit ScriptVehicleDamageModifier(ScriptProviderHandle handle) : handle_(std::move(handle)) {}
149
151 [[nodiscard]] const ScriptProviderHandle& handle() const noexcept { return handle_; }
152
153 float modifyDamage(VehicleEntity& v, float amount, const std::string& zone, int sourceId) override {
154 // Identity is copied out before the script call: the rule runs user code and
155 // may mutate the ECS world, so no component accessor stays live across it.
156 const auto identity = v.identity();
157 const ScriptArg args[] = {
158 ScriptArg::fromText(identity->id), ScriptArg::fromText(identity->defId),
159 ScriptArg::fromNumber(double(amount)), ScriptArg::fromText(zone),
160 ScriptArg::fromInteger(sourceId),
161 };
162
163 auto modified = ScriptProviderRegistry::instance().callNumber(handle_, "modifyDamage", args);
164 if (!modified) {
165 // Fail open on the incoming amount: the alternative is silently dropping
166 // a hit because a game rule has a typo. Report the first failure of each
167 // generation so the fallback stays observable without flooding the log.
168 if (reportedGeneration_ != handle_.generation) {
169 reportedGeneration_ = handle_.generation;
170 recordLogEvent("error",
171 "vehicle damage rule '" + handle_.name +
172 "' failed and the incoming amount was kept: " + modified.status().describe());
173 }
174 return amount;
175 }
176 return static_cast<float>(modified.value());
177 }
178
179private:
180 ScriptProviderHandle handle_;
182 std::uint64_t reportedGeneration_ = 0;
183};

217void declareScriptDamageSeam() {
218 ScriptDamageModifiers& owned = scriptDamageModifiers();
219 ScriptProviderRegistry::instance().declareCapability(
220 IVehicleDamageModifier::capabilityName,
221 [&owned](const ScriptProviderHandle& handle) -> eve::Result<void> {
222 auto adapter = std::make_unique<ScriptVehicleDamageModifier>(handle);
223 // Register before publishing ownership: from here on the listener table
224 // holds the raw pointer, and `owned.live` is what keeps it valid.
225 eve::cap::addListener<IVehicleDamageModifier>(adapter.get());
226 owned.live.push_back(std::move(adapter));
227 return eve::Result<void>::success();
228 },
229 [&owned](const ScriptProviderHandle& handle) -> eve::Result<void> {
230 const auto match = std::find_if(owned.live.begin(), owned.live.end(),
231 [&handle](const std::unique_ptr<ScriptVehicleDamageModifier>& adapter) {
232 return adapter->handle() == handle;
233 });
234 if (match == owned.live.end()) {
235 // Idempotent on purpose: the postcondition is "this generation has no
236 // adapter", and module teardown may already have withdrawn it. Returning
237 // a failure here would make a later `publish` of the same name fail for a
238 // state that is already correct.
239 return eve::Result<void>::success();
240 }
241 eve::cap::removeListener<IVehicleDamageModifier>(match->get());
242 owned.retired.push_back(std::move(*match));
243 owned.live.erase(match);
244 return eve::Result<void>::success();
245 });
246}

260void withdrawScriptDamageSeam() {
261 ScriptDamageModifiers& owned = scriptDamageModifiers();
262 for (auto& adapter : owned.live) {
263 eve::cap::removeListener<IVehicleDamageModifier>(adapter.get());
264 owned.retired.push_back(std::move(adapter));
265 }
266 owned.live.clear();
267
268 // Withdraw the declaration too: with no consumer left, a seam would advertise a
269 // binding whose only possible outcome is a rule nothing ever dispatches.
270 auto withdrawn = ScriptProviderRegistry::instance().removeCapability(IVehicleDamageModifier::capabilityName);
271 withdrawn.ignore("a module may be destroyed without ever having declared the seam");
272}

295int Vehicle::registerVehiclesFromJson(const std::string& json) {
296 auto canonical = eve::Value::fromJson(json);
297 if (!canonical) {
298 canonical.ignore("vehicle definition input could not be parsed as canonical Value");
299 return 0;
300 }
301
302 int n = 0;
303 const auto publish = [this, &n](const eve::Value& value) {
304 const auto* object = value.getIf<eve::Value::Object>();
305 if (object == nullptr) return;
306 const auto id = object->find("id");
307 const auto* name = id == object->end() ? nullptr : id->second.getIf<std::string>();
308 if (name == nullptr || name->empty()) return;
309
310 auto encoded = value.toJson();
311 if (!encoded) {
312 encoded.ignore("vehicle definition value could not be serialized canonically");
313 return;
314 }
315 eve::definitions::Definition source;
316 source.type = "vehicle";
317 source.id = *name;
318 source.version = eve::SchemaVersion(1);
319 source.json = encoded.value();
320 auto typed = parseVehicleDefinition(source);
321 if (!typed) {
322 typed.ignore("vehicle definition failed typed projection validation");
323 return;
324 }
325
326 auto current = definitionRegistry_.resolve("vehicle", *name);
327 const bool exists = current.ok();
328 if (!exists && current.status().code() != eve::StatusCode::NotFound) return;
329 auto stored = exists ? definitionRegistry_.replace("vehicle", *name, 1, source.json)
330 : definitionRegistry_.insert("vehicle", *name, 1, source.json);
331 if (stored) {
332 ++n;
333 } else {
334 stored.ignore("vehicle definition registry rejected canonical registration");
335 }
336 };
337 const eve::Value& owningRoot = canonical.value();
338 if (const auto* array = owningRoot.getIf<eve::Value::Array>()) {
339 for (const auto& value : *array) publish(value);
340 } else {
341 publish(owningRoot);
342 }
343 return n;
344}

394VehicleEntity* Vehicle::newVehicle(const std::string& defId, float x, float y, float heading,
395 const std::string& faction) {
396 auto definition = findDef(defId);
397 if (!definition) {
398 definition.ignore("legacy nullable factory returns null for an unknown definition");
399 return nullptr;
400 }
401 VehicleDefinition def = std::move(definition).takeValue();
402
403 VehicleEntity* v = VehicleEntity::createVehicle();
404 v->identity()->id = defId + "#" + std::to_string(nextInstance_++);
405 v->identity()->defId = defId;
406 v->identity()->faction = faction;
407 v->definition()->owned = std::make_shared<const VehicleDefinition>(def);
408 v->definition()->def = v->definition()->owned.get();
409 v->motion()->x = x;
410 v->motion()->y = y;
411 v->motion()->heading = normalizeHeading(heading);
412 v->health()->maxHp = def.maxHealth;
413 v->health()->hp = def.maxHealth;
414
415 if (!def.mounts.empty()) {
416 // Composition boundary: this factory is the single place Vehicle resolves the
417 // Weapon module, and mounts cannot be built before it exists. Spelled out
418 // rather than through the non-runtime `requireModInst` macro (R-MECH-3).
419 if (eve::weapon::Weapon* wmod = eve::ModuleManager::requireInstance<eve::weapon::Weapon>("Weapon")) {
420 for (const MountDef& md : def.mounts) {
421 eve::weapon::WeaponMountEntity* m = wmod->newMount(md.name, md.type);
422 if (m == nullptr) continue;
423 m->identity()->nodePath = md.name;
424 wmod->mountSetLimits(m, md.yawMin, md.yawMax, md.pitchMin, md.pitchMax, md.rotSpeed, md.firingArc);
425 if (!md.weapon.empty()) {
426 if (eve::weapon::WeaponEntity* w = wmod->newWeapon(md.weapon)) {
427 wmod->mountAttachWeapon(m, w);
428 }
429 }
430 VehicleEntity::MountSlot slot;
431 slot.mount = m;
432 slot.aimMode = md.aimMode;
433 v->mounts()->list.push_back(slot);
434 }
435 }
436 }
437
438 for (const SeatDef& sd : def.seats) {
439 VehicleEntity::SeatSlot slot;
440 slot.name = sd.name;
441 slot.driver = sd.driver;
442 slot.cameraMode = sd.cameraMode;
443 slot.mountIndex = sd.mountIndex;
444 v->seats()->list.push_back(slot);
445 }
446
447 vehicles_.push_back(ecs::handle_of(v));
448 return v;
449}

553void Vehicle::applyDamage(VehicleEntity* v, float amount, const std::string& zone, int sourceId) {
554 if (v == nullptr || amount <= 0.f) return;
555 if (v->stateFlags()->destroyed) return;
556
557 // 修饰器（多重监听，游戏侧叠加规则，如科技减伤/暴击）
558 eve::cap::forEach<IVehicleDamageModifier>(
559 [&](IVehicleDamageModifier* m) { amount = m->modifyDamage(*v, amount, zone, sourceId); });
560 if (amount <= 0.f) return;
561
562 // 装甲区倍率
563 float mult = 1.f;
564 if (!zone.empty()) {
565 for (const ArmorZone& z : v->definition()->def->armorZones) {
566 if (z.name == zone) {
567 mult = z.mult;
568 break;
569 }
570 }
571 }
572
573 const float hp = v->health()->hp - amount * mult;
574 v->health()->hp = std::max(0.f, hp);
575
576 VehicleEvent e;
577 e.vehicleId = v->identity()->id;
578 e.defId = v->identity()->defId;
579 e.x = v->motion()->x;
580 e.y = v->motion()->y;
581 if (hp <= 0.f) {
582 v->stateFlags()->destroyed = true;
583 e.type = VehicleEventType::Destroyed;
584 } else {
585 e.type = VehicleEventType::Damaged;
586 }
587 events_.push_back(e);
588}

734void Vehicle::updateSeats(VehicleEntity& v) {
735 auto seats = v.seats();
736 if (seats->list.empty()) return;
737
738 eve::weapon::Weapon* wmod = eve::ModuleManager::getInstance<eve::weapon::Weapon>("Weapon");
739 bool driverApplied = false;
740
741 for (const VehicleEntity::SeatSlot& s : seats->list) {
742 if (!s.occupied) continue;
743 IVehicleDriver* driver = VehicleSystem::findDriver(s.driver);
744 if (driver == nullptr) continue;
745
746 VehicleInput in;
747 if (!driver->sample(v, s.occupant, in)) continue;
748
749 // 驾驶员座位：写移动输入（只取第一个有效驾驶员）
750 if (!driverApplied && s.name == "driver") {
751 auto vi = v.input();
752 vi->throttle = in.throttle;
753 vi->steer = in.steer;
754 vi->brake = in.brake;
755 vi->handbrake = in.handbrake;
756 vi->fire = in.fire;
757 vi->aimYaw = in.aimYaw;
758 vi->aimPitch = in.aimPitch;
759 driverApplied = true;
760 }
761
762 // 武器座位：瞄准 + 开火
763 if (s.mountIndex >= 0 && static_cast<size_t>(s.mountIndex) < v.mounts()->list.size() && wmod != nullptr) {
764 eve::weapon::WeaponMountEntity* m = v.mounts()->list[s.mountIndex].mount;
765 if (m == nullptr || m->state()->destroyed) continue;
766 wmod->mountAimAt(m, in.aimYaw, in.aimPitch);
767 if (in.fire) {
768 if (eve::weapon::WeaponEntity* w = wmod->mountGetWeapon(m)) {
769 eve::weapon::FireRequest req;
770 req.yaw = in.aimYaw;
771 req.pitch = in.aimPitch;
772 req.shooterId = s.occupant;
773 wmod->fire(w, req);
774 }
775 }
776 }
777 }

780void Vehicle::autoAim(VehicleEntity& v) {
781 const VehicleOrder* cur = VehicleSystem::currentOrder(v);
782 if (cur == nullptr || (cur->type != VehicleOrderType::Attack && cur->type != VehicleOrderType::AttackMove)) {
783 return;
784 }
785 if (v.mounts()->list.empty()) return;
786
787 eve::weapon::Weapon* wmod = eve::ModuleManager::getInstance<eve::weapon::Weapon>("Weapon");
788 if (wmod == nullptr) return;
789
790 auto mo = v.motion();
791 const float worldYaw = std::atan2(cur->y - mo->y, cur->x - mo->x) * 180.f / kPi;
792 float localYaw = worldYaw - mo->heading;
793 localYaw = std::fmod(localYaw + 180.f, 360.f);
794 if (localYaw < 0.f) localYaw += 360.f;
795 localYaw -= 180.f;
796
797 const float dist = std::sqrt((cur->x - mo->x) * (cur->x - mo->x) +
798 (cur->y - mo->y) * (cur->y - mo->y));
799 for (const VehicleEntity::MountSlot& slot : v.mounts()->list) {
800 if (slot.mount == nullptr || slot.mount->state()->destroyed) continue;
801 if (slot.aimMode != "auto") continue;
802 wmod->mountAimAt(slot.mount, localYaw, 0.f);
803 // 攻击命令：炮塔在射程内自动开火；attackMove 只负责边走边瞄准。
804 if (cur->type != VehicleOrderType::Attack) continue;
805 eve::weapon::WeaponEntity* w = wmod->mountGetWeapon(slot.mount);
806 if (w == nullptr || !wmod->canFire(w)) continue;
807 const std::string& defId = w->identity()->defId;
808 const float range = wmod->getWeaponDefinitionRange(defId);
809 if (range > 0.f && dist > range) continue;
810 wmod->fireAt(w, cur->x, cur->y, 0.f, cur->targetId);
811 }
