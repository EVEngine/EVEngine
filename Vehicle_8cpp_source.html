37template <typename T>
38void registerCppEntityClassForScript(const ssq::Class& cls) {
39 eve::registerCppEntityView(cls, [](ssq::Array& out) {
40 HSQUIRRELVM vm = out.getHandle();
41 sq_pushobject(vm, out.getRaw());
42 ecs::Table* table = ecs::current();
43 if (table == nullptr) {
44 sq_pop(vm, 1);
45 return;
46 }
47 ecs::IComponentManager& cm = table->getOrCreateManager<T>();
48 auto* reg = cm.getOrCreateRegistryComponentBuffer<T>();
49 std::vector<ecs::IComponentBuffer*> stack;
50 stack.push_back(reg);
51 while (!stack.empty()) {
52 ecs::IComponentBuffer* buf = stack.back();
53 stack.pop_back();
54 auto* r = dynamic_cast<ecs::IRegistryComponentBuffer*>(buf);
55 if (r != nullptr) {
56 for (uint32_t i = 0; i < r->entity_count(); ++i) {
57 ecs::Entity* ent = r->entity_at(i);
58 if (ent != nullptr && ecs::is_entity_visible(ent)) {
59 ssq::detail::pushByPtr<T>(vm, static_cast<T*>(ent));
60 sq_arrayappend(vm, -2);
61 }
62 }
63 }
64 if (buf->children != nullptr) stack.push_back(buf->children);
65 if (buf->next != nullptr) stack.push_back(buf->next);
66 }
67 sq_pop(vm, 1);
68 });
69}

133int Vehicle::registerVehiclesFromJson(const std::string& json) {
134 auto canonical = eve::Value::fromJson(json);
135 if (!canonical) {
136 canonical.ignore("vehicle definition input could not be parsed as canonical Value");
137 return 0;
138 }
139
140 int n = 0;
141 const auto publish = [this, &n](const eve::Value& value) {
142 const auto* object = value.getIf<eve::Value::Object>();
143 if (object == nullptr) return;
144 const auto id = object->find("id");
145 const auto* name = id == object->end() ? nullptr : id->second.getIf<std::string>();
146 if (name == nullptr || name->empty()) return;
147
148 auto encoded = value.toJson();
149 if (!encoded) {
150 encoded.ignore("vehicle definition value could not be serialized canonically");
151 return;
152 }
153 eve::definitions::Definition source;
154 source.type = "vehicle";
155 source.id = *name;
156 source.version = eve::SchemaVersion(1);
157 source.json = encoded.value();
158 auto typed = parseVehicleDefinition(source);
159 if (!typed) {
160 typed.ignore("vehicle definition failed typed projection validation");
161 return;
162 }
163
164 auto current = definitionRegistry_.resolve("vehicle", *name);
165 const bool exists = current.ok();
166 if (!exists && current.status().code() != eve::StatusCode::NotFound) return;
167 auto stored = exists ? definitionRegistry_.replace("vehicle", *name, 1, source.json)
168 : definitionRegistry_.insert("vehicle", *name, 1, source.json);
169 if (stored) {
170 ++n;
171 } else {
172 stored.ignore("vehicle definition registry rejected canonical registration");
173 }
174 };
175 const eve::Value& owningRoot = canonical.value();
176 if (const auto* array = owningRoot.getIf<eve::Value::Array>()) {
177 for (const auto& value : *array) publish(value);
178 } else {
179 publish(owningRoot);
180 }
181 return n;
182}

232VehicleEntity* Vehicle::newVehicle(const std::string& defId, float x, float y, float heading,
233 const std::string& faction) {
234 auto definition = findDef(defId);
235 if (!definition) {
236 definition.ignore("legacy nullable factory returns null for an unknown definition");
237 return nullptr;
238 }
239 VehicleDefinition def = std::move(definition).takeValue();
240
241 VehicleEntity* v = VehicleEntity::createVehicle();
242 v->identity()->id = defId + "#" + std::to_string(nextInstance_++);
243 v->identity()->defId = defId;
244 v->identity()->faction = faction;
245 v->definition()->owned = std::make_shared<const VehicleDefinition>(def);
246 v->definition()->def = v->definition()->owned.get();
247 v->motion()->x = x;
248 v->motion()->y = y;
249 v->motion()->heading = normalizeHeading(heading);
250 v->health()->maxHp = def.maxHealth;
251 v->health()->hp = def.maxHealth;
252
253 if (!def.mounts.empty()) {
254 if (eve::weapon::Weapon* wmod = requireModInst(eve::weapon, Weapon)) {
255 for (const MountDef& md : def.mounts) {
256 eve::weapon::WeaponMountEntity* m = wmod->newMount(md.name, md.type);
257 if (m == nullptr) continue;
258 m->identity()->nodePath = md.name;
259 wmod->mountSetLimits(m, md.yawMin, md.yawMax, md.pitchMin, md.pitchMax, md.rotSpeed, md.firingArc);
260 if (!md.weapon.empty()) {
261 if (eve::weapon::WeaponEntity* w = wmod->newWeapon(md.weapon)) {
262 wmod->mountAttachWeapon(m, w);
263 }
264 }
265 VehicleEntity::MountSlot slot;
266 slot.mount = m;
267 slot.aimMode = md.aimMode;
268 v->mounts()->list.push_back(slot);
269 }
270 }
271 }
272
273 for (const SeatDef& sd : def.seats) {
274 VehicleEntity::SeatSlot slot;
275 slot.name = sd.name;
276 slot.driver = sd.driver;
277 slot.cameraMode = sd.cameraMode;
278 slot.mountIndex = sd.mountIndex;
279 v->seats()->list.push_back(slot);
280 }
281
282 vehicles_.push_back(ecs::handle_of(v));
283 return v;
284}

388void Vehicle::applyDamage(VehicleEntity* v, float amount, const std::string& zone, int sourceId) {
389 if (v == nullptr || amount <= 0.f) return;
390 if (v->stateFlags()->destroyed) return;
391
392 // 修饰器（多重监听，游戏侧叠加规则，如科技减伤/暴击）
393 eve::cap::forEach<IVehicleDamageModifier>(
394 [&](IVehicleDamageModifier* m) { amount = m->modifyDamage(*v, amount, zone, sourceId); });
395 if (amount <= 0.f) return;
396
397 // 装甲区倍率
398 float mult = 1.f;
399 if (!zone.empty()) {
400 for (const ArmorZone& z : v->definition()->def->armorZones) {
401 if (z.name == zone) {
402 mult = z.mult;
403 break;
404 }
405 }
406 }
407
408 const float hp = v->health()->hp - amount * mult;
409 v->health()->hp = std::max(0.f, hp);
410
411 VehicleEvent e;
412 e.vehicleId = v->identity()->id;
413 e.defId = v->identity()->defId;
414 e.x = v->motion()->x;
415 e.y = v->motion()->y;
416 if (hp <= 0.f) {
417 v->stateFlags()->destroyed = true;
418 e.type = VehicleEventType::Destroyed;
419 } else {
420 e.type = VehicleEventType::Damaged;
421 }
422 events_.push_back(e);
423}

425float Vehicle::getArmorZoneMult(VehicleEntity* v, const std::string& zone) {
426 if (v == nullptr || v->definition()->def == nullptr || zone.empty()) return 1.f;
427 for (const ArmorZone& z : v->definition()->def->armorZones) {
428 if (z.name == zone) return z.mult;
429 }
430 return 1.f;
431}

570void Vehicle::updateSeats(VehicleEntity& v) {
571 auto seats = v.seats();
572 if (seats->list.empty()) return;
573
574 eve::weapon::Weapon* wmod = eve::ModuleManager::getInstance<eve::weapon::Weapon>("Weapon");
575 bool driverApplied = false;
576
577 for (const VehicleEntity::SeatSlot& s : seats->list) {
578 if (!s.occupied) continue;
579 IVehicleDriver* driver = VehicleSystem::findDriver(s.driver);
580 if (driver == nullptr) continue;
581
582 VehicleInput in;
583 if (!driver->sample(v, s.occupant, in)) continue;
584
585 // 驾驶员座位：写移动输入（只取第一个有效驾驶员）
586 if (!driverApplied && s.name == "driver") {
587 auto vi = v.input();
588 vi->throttle = in.throttle;
589 vi->steer = in.steer;
590 vi->brake = in.brake;
591 vi->handbrake = in.handbrake;
592 vi->fire = in.fire;
593 vi->aimYaw = in.aimYaw;
594 vi->aimPitch = in.aimPitch;
595 driverApplied = true;
596 }
597
598 // 武器座位：瞄准 + 开火
599 if (s.mountIndex >= 0 && static_cast<size_t>(s.mountIndex) < v.mounts()->list.size() && wmod != nullptr) {
600 eve::weapon::WeaponMountEntity* m = v.mounts()->list[s.mountIndex].mount;
601 if (m == nullptr || m->state()->destroyed) continue;
602 wmod->mountAimAt(m, in.aimYaw, in.aimPitch);
603 if (in.fire) {
604 if (eve::weapon::WeaponEntity* w = wmod->mountGetWeapon(m)) {
605 eve::weapon::FireRequest req;
606 req.yaw = in.aimYaw;
607 req.pitch = in.aimPitch;
608 req.shooterId = s.occupant;
609 wmod->fire(w, req);

616void Vehicle::autoAim(VehicleEntity& v) {
617 const VehicleOrder* cur = VehicleSystem::currentOrder(v);
618 if (cur == nullptr || (cur->type != VehicleOrderType::Attack && cur->type != VehicleOrderType::AttackMove)) {
619 return;
620 }
621 if (v.mounts()->list.empty()) return;
622
623 eve::weapon::Weapon* wmod = eve::ModuleManager::getInstance<eve::weapon::Weapon>("Weapon");
624 if (wmod == nullptr) return;
625
626 auto mo = v.motion();
627 const float worldYaw = std::atan2(cur->y - mo->y, cur->x - mo->x) * 180.f / kPi;
628 float localYaw = worldYaw - mo->heading;
629 localYaw = std::fmod(localYaw + 180.f, 360.f);
630 if (localYaw < 0.f) localYaw += 360.f;
631 localYaw -= 180.f;
632
633 const float dist = std::sqrt((cur->x - mo->x) * (cur->x - mo->x) +
634 (cur->y - mo->y) * (cur->y - mo->y));
635 for (const VehicleEntity::MountSlot& slot : v.mounts()->list) {
636 if (slot.mount == nullptr || slot.mount->state()->destroyed) continue;
637 if (slot.aimMode != "auto") continue;
638 wmod->mountAimAt(slot.mount, localYaw, 0.f);
639 // 攻击命令：炮塔在射程内自动开火；attackMove 只负责边走边瞄准。
640 if (cur->type != VehicleOrderType::Attack) continue;
641 eve::weapon::WeaponEntity* w = wmod->mountGetWeapon(slot.mount);
642 if (w == nullptr || !wmod->canFire(w)) continue;
643 const std::string& defId = w->identity()->defId;
644 const float range = wmod->getWeaponDefinitionRange(defId);
645 if (range > 0.f && dist > range) continue;
646 wmod->fireAt(w, cur->x, cur->y, 0.f, cur->targetId);
647 }
