74Result<eve::Subscription> Graphics::onResourcesRetiring(std::function<void()> callback) const {
75 auto value = lifetime(this);
76 if (value->retired)
77 return Result<eve::Subscription>::failure(
78 Diagnostic::error(DiagnosticCode::Conflict, "Graphics resources are retiring"));

81void Graphics::retireResourceLifetime() const {
82 if (!registryAlive) return;
83 auto it = lifetimes().find(this);
84 if (it == lifetimes().end() || it->second->retired) return;
85 auto value = it->second;
86 value->retired = true;
87 const auto failures = value->retiring.notifyChecked([]() noexcept {});
88 if (failures) std::fprintf(stderr, "Graphics resource retirement: %zu callback failures\n", failures);
89 lifetimes().erase(this);
90}
