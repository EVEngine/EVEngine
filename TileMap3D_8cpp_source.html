25void atlasUvForGid(const map::TileLayer::Tileset &set, uint32_t gid, float &u0, float &v0,
26 float &u1, float &v1) {
27 const int cols = std::max(1, set.columns);
28 const int tileW = std::max(1, set.tileW);
29 const int tileH = std::max(1, set.tileH);
30 const int margin = std::max(0, set.margin);
31 const int spacing = std::max(0, set.spacing);
32 const int cell = int(gid) - set.firstGid;
33 if (cell < 0) {
34 u0 = 0.f;
35 v0 = 0.f;
36 u1 = 1.f;
37 v1 = 1.f;
38 return;
39 }
40 const int cx = cell % cols;
41 const int cy = cell / cols;
42 const float texW = set.texture ? float(std::max(1, set.texture->getWidth())) : float(tileW);
43 const float texH = set.texture ? float(std::max(1, set.texture->getHeight())) : float(tileH);
44 const float px0 = float(margin + cx * (tileW + spacing));
45 const float py0 = float(margin + cy * (tileH + spacing));
46 u0 = px0 / texW;
47 v0 = py0 / texH;
48 u1 = (px0 + float(tileW)) / texW;
49 v1 = (py0 + float(tileH)) / texH;
50}

81graphics::Mesh *TileMap3D::buildMesh(graphics::Graphics *gfx, map::TileLayer *layer) {
82 if (!gfx) throw eve::Exception("TileMap3D.buildMesh: null graphics");
83 if (!layer) throw eve::Exception("TileMap3D.buildMesh: null tile layer");
84
85 const int mapW = layer->getMapWidth();
86 const int mapH = layer->getMapHeight();
87 if (mapW <= 0 || mapH <= 0) throw eve::Exception("TileMap3D.buildMesh: empty tile layer");
88
89 const float tileW = std::max(1.f, layer->getTileWidth());
90 const float tileH = std::max(1.f, layer->getTileHeight());
91 const float hx = tileW * 0.5f;
92 const float hz = tileH * 0.5f;
93
94 auto set = layer->tileset(); // ecs::ComponentRef<Tileset>; always valid
95 const bool haveSet = set->texture != nullptr && set->columns > 0;
96
97 std::vector<float> pos;
98 std::vector<float> nrm;
99 std::vector<float> uv;
100 std::vector<uint32_t> idx;
101
102 // Unit-cube face topology (RH Y-up, outward CCW): faces = -Z,+Z,-X,+X,-Y,+Y.
103 // Face 5 (+Y) is the textured top; faces 0..4 are extruded side walls.
104 static const int kQuads[6][4] = {
105 {0, 1, 2, 3}, {5, 4, 7, 6}, {4, 0, 3, 7},
106 {1, 5, 6, 2}, {4, 5, 1, 0}, {3, 2, 6, 7},
107 };
108 static const glm::vec3 kFaces[6] = {
109 {0.f, 0.f, -1.f}, {0.f, 0.f, 1.f}, {-1.f, 0.f, 0.f},
110 {1.f, 0.f, 0.f}, {0.f, -1.f, 0.f}, {0.f, 1.f, 0.f},
111 };
112
113 const auto pushBoxTile = [&](float cx, float cz, float topY, float botY, float tu0, float tv0,
114 float tu1, float tv1) {
115 // Corners (index matches kQuads[kFace]).
116 const glm::vec3 corners[8] = {
117 {cx - hx, botY, cz - hz}, {cx + hx, botY, cz - hz}, {cx + hx, topY, cz - hz},
118 {cx - hx, topY, cz - hz}, {cx - hx, botY, cz + hz}, {cx + hx, botY, cz + hz},
119 {cx + hx, topY, cz + hz}, {cx - hx, topY, cz + hz},
120 };
121 const float sideU0 = wallU0_, sideV0 = wallV0_, sideU1 = wallU1_, sideV1 = wallV1_;
122 for (int f = 0; f < 6; ++f) {
123 const uint32_t base = uint32_t(pos.size() / 3);
124 const bool top = (f == 5);
125 const float su0 = top ? tu0 : sideU0;
126 const float sv0 = top ? tv0 : sideV0;
127 const float su1 = top ? tu1 : sideU1;
128 const float sv1 = top ? tv1 : sideV1;
129 const float us[4] = {su0, su1, su1, su0};
130 const float vs[4] = {sv0, sv0, sv1, sv1};
131 const glm::vec3 &n = kFaces[f];
132 for (int k = 0; k < 4; ++k) {
133 const glm::vec3 &p = corners[kQuads[f][k]];
134 pos.insert(pos.end(), {p.x, p.y, p.z});
135 nrm.insert(nrm.end(), {n.x, n.y, n.z});
136 uv.insert(uv.end(), {us[k], vs[k]});
137 }
138 idx.insert(idx.end(), {base, base + 1, base + 2, base, base + 2, base + 3});
139 }
140 };
141
142 tileCount_ = 0;
143
144 for (int ty = 0; ty < mapH; ++ty) {
145 for (int tx = 0; tx < mapW; ++tx) {
146 const uint32_t gid = eve::map::tileGid(uint32_t(layer->getTile(tx, ty)));
147 if (gid == 0) continue;
148
149 // World X/Z from the layer's 2D projection; elevation from metadata.
150 const float wx = layer->tileToWorldX(tx, ty);
151 const float wz = layer->tileToWorldY(tx, ty);
152 const float elev = layer->getTileDataNumber(int(gid), "height") * heightScale_;
153 const float topY = elev;
154 const float botY = elev - sideDepth_;
155
156 float tu0 = 0.f, tv0 = 0.f, tu1 = 1.f, tv1 = 1.f;
157 if (haveSet) atlasUvForGid(*set, gid, tu0, tv0, tu1, tv1);
158
159 pushBoxTile(wx, wz, topY, botY, tu0, tv0, tu1, tv1);
160 ++tileCount_;
161 }
162 }
163
164 if (pos.empty()) return nullptr;
165 return gfx->newMeshFromArrays(pos.data(), nrm.data(), uv.data(), int(pos.size() / 3),
166 idx.data(), int(idx.size()));
167}
