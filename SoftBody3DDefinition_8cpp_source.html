58eve::Result<void> SoftBody3DDefinition::validate() const {
59 if (cols < 2 || rows < 2 || layers < 2)
60 return invalid<void>("soft-body lattice dimensions must each be at least 2", "dimensions");
61 const std::int64_t count = std::int64_t(cols) * std::int64_t(rows) * std::int64_t(layers);
62 if (count > 1000000) return invalid<void>("soft-body exceeds the one-million-particle limit", "dimensions");
63 const auto finite = [](float value) { return std::isfinite(value); };
64 if (!finite(spacing) || spacing <= 0.f) return invalid<void>("spacing must be finite and positive", "spacing");
65 if (!finite(originX) || !finite(originY) || !finite(originZ))
66 return invalid<void>("origin must contain finite coordinates", "origin");
67 if (!finite(gravityX) || !finite(gravityY) || !finite(gravityZ))
68 return invalid<void>("gravity must contain finite coordinates", "gravity");
69 if (!finite(deformationResistance) || deformationResistance < 0.f || deformationResistance > 1.f)
70 return invalid<void>("deformationResistance must be in [0, 1]", "deformationResistance");
71 if (iterations < 1 || iterations > 32) return invalid<void>("iterations must be in [1, 32]", "iterations");
72 if (!finite(damping) || damping < 0.f || damping > 1.f)
73 return invalid<void>("damping must be in [0, 1]", "damping");
74 if (!finite(particleRadius) || particleRadius <= 0.f)
75 return invalid<void>("particleRadius must be finite and positive", "particleRadius");
76 if (!finite(particleMass) || particleMass <= 0.f)
77 return invalid<void>("particleMass must be finite and positive", "particleMass");
78 if (!finite(plasticYield) || !finite(plasticCreep) || !finite(plasticRecovery) || !finite(maxDeformation) ||
79 plasticYield < 0.f || plasticCreep < 0.f || plasticCreep > 1.f || plasticRecovery < 0.f ||
80 plasticRecovery > 1.f || maxDeformation < 0.f)
81 return invalid<void>("plasticity fields are outside their supported ranges", "plasticity");
82 return eve::Result<void>::success();
83}
