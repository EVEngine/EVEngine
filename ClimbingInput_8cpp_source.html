23eve::Result<void> ClimbingInputSystem::submit(ClimbingIntent& intent, ClimbingCommand command,
24 eve::SimulationTick pressedTick, std::uint64_t bufferTicks) {
25 if (bufferTicks == 0)
26 return failure<void>(eve::DiagnosticCode::InvalidArgument, "input buffer must contain at least one tick",
27 "bufferTicks");
28 if (pressedTick.value() > std::numeric_limits<std::uint64_t>::max() - bufferTicks)
29 return failure<void>(eve::DiagnosticCode::InvalidArgument, "input expiry tick would overflow", "pressedTick");
30 (void)prune(intent, pressedTick);
31 if (std::any_of(intent.commands.begin(), intent.commands.end(),
32 [&](const auto& entry) { return entry.command == command && entry.pressedTick == pressedTick; }))
33 return failure<void>(eve::DiagnosticCode::Conflict, "duplicate edge command for the same tick", "command");
34 if (intent.commands.size() >= MaxBufferedCommands)
35 return failure<void>(eve::DiagnosticCode::PreconditionViolation, "climbing input buffer is full", "commands");
36 intent.commands.push_back(
37 {command, pressedTick, eve::SimulationTick(pressedTick.value() + bufferTicks), ClimbingExecutionId::zero()});
38 std::stable_sort(intent.commands.begin(), intent.commands.end(), commandLess);
39 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
40}

48std::size_t ClimbingInputSystem::prune(ClimbingIntent& intent, eve::SimulationTick currentTick) noexcept {
49 const std::size_t before = intent.commands.size();
50 std::erase_if(intent.commands, [&](const BufferedClimbingCommand& entry) {
51 return !entry.consumedExecutionId.isZero() || entry.expiryTick < currentTick;
52 });
53 return before - intent.commands.size();
54}

56eve::Result<std::optional<BufferedClimbingCommand>> ClimbingInputSystem::consume(ClimbingIntent& intent,
57 ClimbingCommand command,
58 eve::SimulationTick currentTick,
59 ClimbingExecutionId executionId) {
60 if (executionId.isZero())
61 return failure<std::optional<BufferedClimbingCommand>>(
62 eve::DiagnosticCode::InvalidArgument, "consuming execution id must be non-zero", "executionId");
63 (void)prune(intent, currentTick);
64 const auto found = std::find_if(intent.commands.begin(), intent.commands.end(), [&](const auto& entry) {
65 return entry.command == command && entry.pressedTick <= currentTick && currentTick <= entry.expiryTick &&
66 entry.consumedExecutionId.isZero();
67 });
68 if (found == intent.commands.end())
69 return eve::Result<std::optional<BufferedClimbingCommand>>::success(
70 std::nullopt, eve::Status::success(eve::StatusCode::NoOp));
71 found->consumedExecutionId = executionId;
72 return eve::Result<std::optional<BufferedClimbingCommand>>::success(*found,
73 eve::Status::success(eve::StatusCode::Applied));
74}
