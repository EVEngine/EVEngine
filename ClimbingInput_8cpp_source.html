29eve::Result<void> ClimbingInputSystem::submit(ClimbingIntent& intent, ClimbingCommand command,
30 eve::SimulationTick pressedTick, std::uint64_t bufferTicks) {
31 if (bufferTicks == 0)
32 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
33 "input buffer must contain at least one tick",
34 "bufferTicks", {}, "climbing.input"));
35 if (pressedTick.value() > std::numeric_limits<std::uint64_t>::max() - bufferTicks)
36 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
37 "input expiry tick would overflow", "pressedTick", {},
38 "climbing.input"));
39 (void)prune(intent, pressedTick);
40 if (std::any_of(intent.commands.begin(), intent.commands.end(),
41 [&](const auto& entry) { return entry.command == command && entry.pressedTick == pressedTick; }))
42 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
43 "duplicate edge command for the same tick", "command",
44 {}, "climbing.input"));
45 if (intent.commands.size() >= MaxBufferedCommands)
46 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
47 "climbing input buffer is full", "commands", {},
48 "climbing.input"));
49 intent.commands.push_back(
50 {command, pressedTick, eve::SimulationTick(pressedTick.value() + bufferTicks), ClimbingExecutionId::zero()});
51 std::stable_sort(intent.commands.begin(), intent.commands.end(), commandLess);
52 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
53}

160Vec3 toModelSpace(Vec3 world, const ClimbingPoseBinding& binding) {
161 const float x = world.x - binding.modelWorldOrigin.x;
162 const float y = world.y - binding.modelWorldOrigin.y;
163 const float z = world.z - binding.modelWorldOrigin.z;
164 const float cosine = std::cos(binding.modelYawRadians);
165 const float sine = std::sin(binding.modelYawRadians);
166 return {cosine * x - sine * z, y, sine * x + cosine * z};
167}

171eve::Result<ClimbingPoseResult> applyClimbingPose(animation::AnimSkeleton& skeleton, animation::AnimPose& pose,
172 const ClimbingAdvance& advance, const ClimbingPoseBinding& binding) {
173 if (!std::isfinite(binding.modelYawRadians))
174 return poseFailure("model yaw must be finite", "binding.modelYawRadians");
175 const int boneCount = skeleton.getBoneCount();
176 if (binding.enableLeftHand && !validChain(binding.leftArm, boneCount))
177 return poseFailure("left arm chain contains an invalid bone", "binding.leftArm");
178 if (binding.enableRightHand && !validChain(binding.rightArm, boneCount))
179 return poseFailure("right arm chain contains an invalid bone", "binding.rightArm");
180
181 ClimbingPoseResult result;
182 const float legacyWeight = std::clamp(advance.contactWeight, 0.f, 1.f);
183 const float authoredLeft = std::clamp(advance.leftHandWeight, 0.f, 1.f);
184 const float authoredRight = std::clamp(advance.rightHandWeight, 0.f, 1.f);
185 const bool hasPerHandWeights = authoredLeft > 0.f || authoredRight > 0.f;
186 const float leftWeight = hasPerHandWeights ? authoredLeft : legacyWeight;
187 const float rightWeight = hasPerHandWeights ? authoredRight : legacyWeight;
188 result.weight = std::max(leftWeight, rightWeight);
189 if (result.weight <= 0.f)
190 return eve::Result<ClimbingPoseResult>::success(std::move(result), eve::Status::success(eve::StatusCode::NoOp));
191
192 animation::AnimConstraintStack constraints(&skeleton);
193 if (binding.enableLeftHand && leftWeight > 0.f) {
194 const Vec3 target = toModelSpace(advance.leftHandAnchor, binding);
195 constraints.addTwoBoneIK(binding.leftArm.root, binding.leftArm.mid, binding.leftArm.tip, target.x, target.y,
196 target.z, leftWeight);
197 result.leftHandApplied = true;
198 }
199 if (binding.enableRightHand && rightWeight > 0.f) {
200 const Vec3 target = toModelSpace(advance.rightHandAnchor, binding);
201 constraints.addTwoBoneIK(binding.rightArm.root, binding.rightArm.mid, binding.rightArm.tip, target.x, target.y,
202 target.z, rightWeight);
203 result.rightHandApplied = true;
204 }
205 constraints.apply(&pose);
206 return eve::Result<ClimbingPoseResult>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
207}

218void appendByte(std::uint64_t& hash, std::uint8_t value) noexcept {
219 hash ^= value;
220 hash *= fnvPrime;
221}

223template <class T>
224void appendInteger(std::uint64_t& hash, T value) noexcept {
225 using Unsigned = std::make_unsigned_t<T>;
226 const Unsigned bits = static_cast<Unsigned>(value);
227 for (std::size_t index = 0; index < sizeof(T); ++index)
228 appendByte(hash, static_cast<std::uint8_t>((bits >> (index * 8)) & static_cast<Unsigned>(0xff)));
229}

231void appendString(std::uint64_t& hash, std::string_view value) noexcept {
232 appendInteger(hash, static_cast<std::uint64_t>(value.size()));
233 for (const unsigned char character : value) appendByte(hash, character);
234}

236std::int64_t quantize(float value) noexcept {
237 if (!std::isfinite(value)) return std::numeric_limits<std::int64_t>::min();
238 const double scaled = std::round(static_cast<double>(value) * 1000.0);
239 if (scaled <= static_cast<double>(std::numeric_limits<std::int64_t>::min()))
240 return std::numeric_limits<std::int64_t>::min();
241 if (scaled >= static_cast<double>(std::numeric_limits<std::int64_t>::max()))
242 return std::numeric_limits<std::int64_t>::max();
243 return static_cast<std::int64_t>(scaled);
244}

360bool decodeKey(const eve::Value& value, ClimbingCandidateKey& key) {
361 const auto* object = value.getIf<eve::Value::Object>();
362 std::int64_t kind = 0;
363 std::int64_t rank = 0;
364 if (!object || !readString(*object, "actionId", key.actionId) || !readInt64(*object, "kind", kind) || kind < 0 ||
365 kind > static_cast<std::int64_t>(ClimbingActionKind::BarSwing) ||
366 !readUint64String(*object, "definitionGeneration", key.definitionGeneration) ||
367 !readInt64(*object, "sortedRank", rank) || rank < 0 || rank > std::numeric_limits<std::uint32_t>::max() ||
368 !readUint64String(*object, "fingerprintHex", key.fingerprint, 16))
369 return false;
370 key.kind = static_cast<ClimbingActionKind>(kind);
371 key.sortedRank = static_cast<std::uint32_t>(rank);
372 return true;
373}

433ClimbingCandidateKey makeClimbingCandidateKey(const ClimbingCandidate& candidate, std::uint32_t sortedRank) noexcept {
434 std::uint64_t hash = fnvOffset;
435 appendString(hash, candidate.actionId);
436 appendInteger(hash, static_cast<std::uint8_t>(candidate.kind));
437 appendInteger(hash, candidate.definitionGeneration);
438 appendInteger(hash, sortedRank);
439 appendInteger(hash, candidate.obstacleBodyId);
440 appendInteger(hash, candidate.obstacleShapeId);
441 appendVec(hash, candidate.frontPoint);
442 appendVec(hash, candidate.bodyLocalTop);
443 appendVec(hash, candidate.bodyLocalLanding);
444 appendVec(hash, candidate.surfaceNormal);
445 appendVec(hash, candidate.surfaceTangent);
446 appendInteger(hash, quantize(candidate.obstacleHeight));
447 appendInteger(hash, quantize(candidate.obstacleDepth));
448 appendInteger(hash, quantize(candidate.gapDistance));
449 appendInteger(hash, quantize(candidate.clearanceHeight));
450 appendInteger(hash, quantize(candidate.slopeRadians));
451 appendInteger(hash, quantize(candidate.curvature));
452 appendInteger(hash, candidate.supportShapeTag);
453 appendInteger(hash, candidate.supportMaterialId);
454 appendInteger(hash, static_cast<std::uint8_t>(candidate.probeRecipe));
455 appendInteger(hash, candidate.score);
456 appendInteger(hash, static_cast<std::uint8_t>(candidate.support));
457 if (hash == 0) hash = 1;
458 return {candidate.actionId, candidate.kind, candidate.definitionGeneration, sortedRank, hash};
459}

461eve::Result<eve::Value> encodeClimbingPredictionRequest(const ClimbingPredictionRequest& request) {
462 if (request.sequence.isZero() || !validKey(request.candidate))

464 eve::DiagnosticCode::InvalidArgument, "prediction request requires a non-zero sequence and candidate key",

468 object["schemaVersion"] = eve::Value(ClimbingPredictionRequest::SchemaVersion);
469 object["sequence"] = eve::Value(std::to_string(request.sequence.value()));
470 object["clientTick"] = eve::Value(std::to_string(request.clientTick.value()));
471 object["candidate"] = encodeKey(request.candidate);
472 return eve::Result<eve::Value>::success(eve::Value(std::move(object)));

475eve::Result<ClimbingPredictionRequest> decodeClimbingPredictionRequest(const eve::Value& value) {
476 auto root = readEnvelope(value, ClimbingPredictionRequest::SchemaId);
477 if (!root) return eve::Result<ClimbingPredictionRequest>::failure(root.status());
478 std::uint64_t sequence = 0;
479 std::uint64_t clientTick = 0;
480 ClimbingPredictionRequest request;
481 const eve::Value* candidate = field(root.value(), "candidate");
482 if (!readUint64String(root.value(), "sequence", sequence) || sequence == 0 ||
483 !readUint64String(root.value(), "clientTick", clientTick) || !candidate ||
484 !decodeKey(*candidate, request.candidate) || !validKey(request.candidate))
485 return eve::Result<ClimbingPredictionRequest>::failure(
486 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "prediction request has missing or invalid fields",
487 "request", {}, "climbing.prediction"));

516 auto root = readEnvelope(value, ClimbingPredictionDecision::SchemaId);
517 if (!root) return eve::Result<ClimbingPredictionDecision>::failure(root.status());
518 std::uint64_t sequence = 0;
519 std::uint64_t clientTick = 0;
520 std::uint64_t serverTick = 0;
521 std::string disposition;
522 std::string reason;
523 ClimbingPredictionDecision decision;
524 const eve::Value* key = field(root.value(), "authoritativeCandidate");
525 const eve::Value* snapshot = field(root.value(), "authoritativeSnapshot");
526 if (!readUint64String(root.value(), "sequence", sequence) || sequence == 0 ||
527 !readUint64String(root.value(), "clientTick", clientTick) ||
528 !readUint64String(root.value(), "serverTick", serverTick) ||
529 !readString(root.value(), "disposition", disposition) || !readDisposition(disposition, decision.disposition) ||
530 !readString(root.value(), "reason", reason) || !readReason(reason, decision.reason) || !key ||
531 !decodeKey(*key, decision.authoritativeCandidate) || !snapshot)
532 return eve::Result<ClimbingPredictionDecision>::failure(
533 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "prediction decision has missing or invalid fields",
534 "decision", {}, "climbing.prediction"));
535 decision.sequence = ClimbingPredictionSequence(sequence);
536 decision.clientTick = eve::SimulationTick(clientTick);
537 decision.serverTick = eve::SimulationTick(serverTick);
538 if (!consistentDecision(decision))
539 return eve::Result<ClimbingPredictionDecision>::failure(
540 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "prediction decision fields are inconsistent",
541 "decision", {}, "climbing.prediction"));
542 decision.authoritativeSnapshot = *snapshot;
543 auto snapshotEnvelope = validateSnapshotEnvelope(decision.authoritativeSnapshot);
544 if (!snapshotEnvelope) return eve::Result<ClimbingPredictionDecision>::failure(snapshotEnvelope.status());
545 decision.extensionMetadata =
546 unknownFields(root.value(), {"schemaId", "schemaVersion", "sequence", "clientTick", "serverTick", "disposition",
547 "reason", "authoritativeCandidate", "authoritativeSnapshot"});
548 return eve::Result<ClimbingPredictionDecision>::success(std::move(decision));

551eve::Result<ClimbingPredictionDecision> ClimbingRuntime::tryBeginPredicted(physics::World3D& world,
552 const ClimbingPose& pose,
553 const ClimbingPredictionRequest& request,
554 eve::SimulationTick serverTick,
555 std::uint64_t maxClientTickLead) {
556 ClimbingPredictionDecision decision;
557 decision.sequence = request.sequence;
558 decision.clientTick = request.clientTick;
559 decision.serverTick = serverTick;
560 const auto finish = [this](ClimbingPredictionDecision value) -> eve::Result<ClimbingPredictionDecision> {
561 auto state = snapshot();
562 if (!state) return eve::Result<ClimbingPredictionDecision>::failure(state.status());
563 value.authoritativeSnapshot = std::move(state).takeValue();
564 return eve::Result<ClimbingPredictionDecision>::success(std::move(value));
565 };
566
567 if (request.sequence.isZero() || !validKey(request.candidate)) {
568 decision.disposition = ClimbingPredictionDisposition::Rejected;
569 decision.reason = ClimbingPredictionReason::InvalidRequest;
570 return finish(std::move(decision));
571 }
572 if (request.clientTick > serverTick && request.clientTick.value() - serverTick.value() > maxClientTickLead) {
573 decision.disposition = ClimbingPredictionDisposition::Rejected;
574 decision.reason = ClimbingPredictionReason::TickTooFarAhead;
575 return finish(std::move(decision));
576 }
577 if (activePhase(phase_)) {
578 decision.disposition = ClimbingPredictionDisposition::Rejected;
579 decision.reason = ClimbingPredictionReason::RuntimeBusy;
580 return finish(std::move(decision));
581 }
582
583 auto candidates = probe(world, pose);
584 if (!candidates) return eve::Result<ClimbingPredictionDecision>::failure(candidates.status());
585 if (candidates.value().empty()) {
586 decision.disposition = ClimbingPredictionDisposition::Rejected;
587 decision.reason = ClimbingPredictionReason::NoCandidate;
588 return finish(std::move(decision));
589 }
590 decision.authoritativeCandidate = makeClimbingCandidateKey(candidates.value().front(), 0);
591 std::size_t match = candidates.value().size();
592 for (std::size_t index = 0; index < candidates.value().size(); ++index) {
593 if (index > std::numeric_limits<std::uint32_t>::max()) break;
594 if (makeClimbingCandidateKey(candidates.value()[index], static_cast<std::uint32_t>(index)) ==
595 request.candidate) {
596 match = index;
597 break;
598 }
599 }
600 if (match == candidates.value().size()) {
601 decision.disposition = ClimbingPredictionDisposition::Corrected;
602 decision.reason = ClimbingPredictionReason::CandidateMismatch;
603 return finish(std::move(decision));
604 }
605
606 decision.authoritativeCandidate = request.candidate;
607 auto prepared = prepareBeginCandidate(world, pose, serverTick, candidates.value()[match]);
608 if (!prepared) return eve::Result<ClimbingPredictionDecision>::failure(prepared.status());
609 auto committed = commitBegin(std::move(prepared).takeValue());
610 if (!committed) return eve::Result<ClimbingPredictionDecision>::failure(committed.status());
611 decision.disposition = ClimbingPredictionDisposition::Accepted;
612 decision.reason = ClimbingPredictionReason::None;
613 return finish(std::move(decision));
614}
