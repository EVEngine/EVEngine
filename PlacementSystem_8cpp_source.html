160eve::Result<PlacementSystem::PlacementHit>
161PlacementSystem::sampleSurface(const PlacementWorld &world, const std::string &name, float x,
162 float y) {
163 ensureBuiltins();
164 const std::string resolvedName = name.empty() ? "plane" : name;
165 auto it = surfaceProviders().find(resolvedName);
166 if (it == surfaceProviders().end() || !it->second) {
167 return eve::Result<PlacementHit>::failure(eve::Diagnostic::error(
168 eve::DiagnosticCode::NotFound, "placement surface provider is not registered",
169 resolvedName, {}, "building.surface"));
170 }
171 auto result = it->second(world, x, y);
172 if (!result.ok()) return eve::Result<PlacementHit>::failure(result.status());
173 PlacementHit hit = std::move(result).takeValue();
174 if (normalizeSurfaceFrame(hit) != SurfaceFrameStatus::Normalized) {
175 return eve::Result<PlacementHit>::failure(eve::Diagnostic::error(
176 eve::DiagnosticCode::InvalidArgument,
177 "placement surface returned a non-finite position or zero normal", resolvedName,
178 {}, "building.surface"));
179 }
180 if (hit.surfaceId.empty()) hit.surfaceId = resolvedName;
181 return eve::Result<PlacementHit>::success(std::move(hit));
182}

184eve::Result<PlacementSystem::SurfacePatch>
185PlacementSystem::sampleSurfacePatch(const PlacementWorld &world, const std::string &buildingId,
186 const std::string &surfaceName, float x, float y,
187 float rotationDegrees) {
188 const BuildingDefinition *def = BuildingRegistry::find(buildingId);
189 if (!def) {
190 return eve::Result<SurfacePatch>::failure(eve::Diagnostic::error(
191 eve::DiagnosticCode::NotFound, "building definition is not registered", buildingId,
192 {}, "building.surface-patch"));
193 }
194
195 const SnapResult snapped = snap(world, buildingId, x, y);
196 auto anchorResult = sampleSurface(world, surfaceName, snapped.worldX, snapped.worldY);
197 if (!anchorResult.ok()) return eve::Result<SurfacePatch>::failure(anchorResult.status());
198
199 SurfacePatch patch;
200 patch.anchor = std::move(anchorResult).takeValue();
201 float originX = 0.f;
202 float originY = 0.f;
203 world.cellToWorldPlane(snapped.cellX, snapped.cellY, originX, originY);
204
205 float minHeight = std::numeric_limits<float>::max();
206 float maxHeight = std::numeric_limits<float>::lowest();
207 const auto slopeDegrees = [&](const PlacementHit &hit) {
208 const float upDot = world.getGrid().plane == grid::GridPlane::XZ ? hit.normalY
209 : hit.normalZ;
210 return std::acos(std::clamp(upDot, -1.f, 1.f)) * 180.f / 3.14159265358979323846f;
211 };
212 patch.maxSlopeDegrees = 0.f;
213 bool failed = false;
214 eve::Status failureStatus;
215 const auto appendSample = [&](float sampleX, float sampleY, int cellX, int cellY,
216 bool anchor) {
217 if (failed) return false;
218 PlacementHit hit;
219 if (anchor) {
220 hit = patch.anchor;
221 } else {
222 auto sample = sampleSurface(world, surfaceName, sampleX, sampleY);
223 if (!sample.ok()) {
224 failed = true;
225 failureStatus = sample.status();
226 return false;
227 }
228 hit = std::move(sample).takeValue();
229 }
230 if (hit.surfaceId != patch.anchor.surfaceId ||
231 hit.surfaceRevision != patch.anchor.surfaceRevision) {
232 failed = true;
233 failureStatus = eve::Status::failure(eve::Diagnostic::error(
234 eve::DiagnosticCode::Conflict,
235 "footprint crosses a surface identity or revision boundary", hit.surfaceId,
236 {}, "building.surface-patch"));
237 return false;
238 }
239 const float height =
240 world.getGrid().plane == grid::GridPlane::XZ ? hit.worldY : hit.worldZ;
241 minHeight = std::min(minHeight, height);
242 maxHeight = std::max(maxHeight, height);
243 patch.maxSlopeDegrees = std::max(patch.maxSlopeDegrees, slopeDegrees(hit));
244 patch.samples.push_back({cellX, cellY, std::move(hit)});
245 return true;
246 };
247 if (def->placementKind == "free" &&
248 (!def->freeFootprintVertices.empty() ||
249 (def->freeFootprintWidthCells > 0.f && def->freeFootprintHeightCells > 0.f))) {
250 appendSample(snapped.worldX, snapped.worldY, snapped.cellX, snapped.cellY, true);
251 const FreeFootprint footprint = definitionFreeFootprint(
252 *def, world, snapped.worldX, snapped.worldY,
253 normalizeRotation(buildingId, rotationDegrees));
254 for (const auto &point : worldVertices(footprint)) {
255 int cellX = 0;
256 int cellY = 0;
257 grid::worldToCell(world.getGrid(), point.x, point.y, cellX, cellY,
258 world.getWidth(), world.getHeight());
259 if (!appendSample(point.x, point.y, cellX, cellY, false)) break;
260 }
261 } else {
262 foreachFootprintCell(
263 *def, snapped.cellX, snapped.cellY,
264 normalizeRotation(buildingId, rotationDegrees), [&](int cellX, int cellY) {
265 float cellWorldX = 0.f;
266 float cellWorldY = 0.f;
267 world.cellToWorldPlane(cellX, cellY, cellWorldX, cellWorldY);
268 return appendSample(snapped.worldX + cellWorldX - originX,
269 snapped.worldY + cellWorldY - originY, cellX, cellY,
270 cellX == snapped.cellX && cellY == snapped.cellY);
271 });
272 }
273 if (failed) return eve::Result<SurfacePatch>::failure(std::move(failureStatus));
274 if (patch.samples.empty()) {
275 return eve::Result<SurfacePatch>::failure(eve::Diagnostic::error(
276 eve::DiagnosticCode::InvalidArgument, "building footprint has no occupied cells",
277 buildingId, {}, "building.surface-patch"));
278 }
279 patch.heightDelta = maxHeight - minHeight;
280 return eve::Result<SurfacePatch>::success(std::move(patch));
281}

457float PlacementSystem::normalizeRotation(const std::string &buildingId, float rotationDeg) {
458 const BuildingDefinition *def = BuildingRegistry::find(buildingId);
459 const std::string mode = def ? def->rotationMode : "cardinal";
460 if (mode == "none") return 0.f;
461 if (mode == "free") return wrapDeg(rotationDeg);
462 if (mode == "hex") {
463 const float d = wrapDeg(rotationDeg);
464 const int steps = int(std::lround(d / 60.f)) % 6;
465 return float((steps < 0 ? steps + 6 : steps) * 60);
466 }
467 // cardinal（默认）
468 return float(cardinalQuarter(rotationDeg) * 90);
469}

471void PlacementSystem::effectiveFootprint(const BuildingDefinition &def, float rotationDeg, int *outW,
472 int *outH) {
473 int w = def.footprintW;
474 int h = def.footprintH;
475 if (def.rotationMode == "hex") {
476 const int steps = int(std::lround(wrapDeg(rotationDeg) / 60.f)) % 6;
477 grid::rotatedFootprintSize(def.footprintW, def.footprintH, def.footprintMask, steps, true,
478 w, h);
479 } else if (def.rotationMode == "cardinal" || def.rotationMode.empty()) {
480 const int q = cardinalQuarter(rotationDeg);
481 if (q == 1 || q == 3) std::swap(w, h);
482 }
483 if (outW) *outW = w;
484 if (outH) *outH = h;
485}
