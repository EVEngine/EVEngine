23inline Vec3 trajectoryPoint(Vec3 start, const ClimbingCandidate& candidate,
24 const ClimbingActionDefinition& action, const ClimbingProfileDefinition& profile,
25 float normalizedTime) noexcept {
26 const float t = std::clamp(normalizedTime, 0.f, 1.f);
27 const bool surfaceFollow = action.trajectory == ClimbingTrajectoryKind::SurfaceFollow ||
28 action.trajectory == ClimbingTrajectoryKind::AnchorToAnchor;
29 const bool ballistic = action.trajectory == ClimbingTrajectoryKind::BallisticArc;
30 if (!ballistic && !surfaceFollow &&
31 (action.kind == ClimbingActionKind::Mantle || action.kind == ClimbingActionKind::ClimbUp)) {
32 constexpr float riseEnd = 0.4f;
33 constexpr float crossEnd = 0.85f;
34 const float clearance = std::max({start.y, candidate.topPoint.y + profile.skin,
35 candidate.landingFeet.y + profile.skin});
36 const Vec3 raisedStart{start.x, clearance, start.z};
37 const Vec3 raisedLanding{candidate.landingFeet.x, clearance, candidate.landingFeet.z};
38 if (t < riseEnd) return interpolate(start, raisedStart, smoothStep(t / riseEnd));
39 if (t < crossEnd)
40 return interpolate(raisedStart, raisedLanding, smoothStep((t - riseEnd) / (crossEnd - riseEnd)));
41 return interpolate(raisedLanding, candidate.landingFeet, smoothStep((t - crossEnd) / (1.f - crossEnd)));
42 }
43
44 if (surfaceFollow || action.kind == ClimbingActionKind::Shimmy ||
45 action.kind == ClimbingActionKind::CornerInner ||
46 action.kind == ClimbingActionKind::CornerOuter || action.kind == ClimbingActionKind::ClimbDown ||
47 action.kind == ClimbingActionKind::LadderMount || action.kind == ClimbingActionKind::LadderClimb ||
48 action.kind == ClimbingActionKind::LadderDismount || action.kind == ClimbingActionKind::Slide ||
49 action.kind == ClimbingActionKind::BeamBalance)
50 return interpolate(start, candidate.landingFeet, smoothStep(t));
51
52 Vec3 point = interpolate(start, candidate.landingFeet, t);
53 point.y += 4.f * action.apexHeight * t * (1.f - t);
54 return point;
