22// Join the two support planes in the horizontal plane. Each eased leg stays
23// on its own support plane; a direct chord can cut through a convex corner.
24inline Vec3 cornerPathPoint(Vec3 start, Vec3 end, Vec3 sourceNormal, Vec3 targetNormal, float normalizedTime) noexcept {
25 const float determinant = sourceNormal.x * targetNormal.z - sourceNormal.z * targetNormal.x;
26 const float t = std::clamp(normalizedTime, 0.f, 1.f);
27 if (std::abs(determinant) < 1e-4f) return interpolate(start, end, smoothStep(t));
28 const float sourcePlane = sourceNormal.x * start.x + sourceNormal.z * start.z;
29 const float targetPlane = targetNormal.x * end.x + targetNormal.z * end.z;
30 const Vec3 join{(sourcePlane * targetNormal.z - sourceNormal.z * targetPlane) / determinant,
31 (start.y + end.y) * 0.5f,
32 (sourceNormal.x * targetPlane - sourcePlane * targetNormal.x) / determinant};
33 return t < 0.5f ? interpolate(start, join, smoothStep(t * 2.f))
34 : interpolate(join, end, smoothStep((t - 0.5f) * 2.f));
35}

43inline Vec3 trajectoryPoint(Vec3 start, const ClimbingCandidate& candidate, const ClimbingActionDefinition& action,
44 const ClimbingProfileDefinition& profile, float normalizedTime,
45 Vec3 sourceNormal = {}) noexcept {
46 const float t = std::clamp(normalizedTime, 0.f, 1.f);
47 if (action.kind == ClimbingActionKind::CornerInner || action.kind == ClimbingActionKind::CornerOuter)
48 return cornerPathPoint(start, candidate.landingFeet, sourceNormal, candidate.surfaceNormal, t);
49 const bool surfaceFollow = action.trajectory == ClimbingTrajectoryKind::SurfaceFollow ||
50 action.trajectory == ClimbingTrajectoryKind::AnchorToAnchor;
51 const bool ballistic = action.trajectory == ClimbingTrajectoryKind::BallisticArc;
52 const float frontX = start.x - candidate.frontPoint.x;
53 const float frontZ = start.z - candidate.frontPoint.z;
54 const float contactDistance = profile.capsuleRadius + 2.f * profile.skin;
55 const bool contactVault = action.kind == ClimbingActionKind::Vault &&
56 frontX * frontX + frontZ * frontZ <= contactDistance * contactDistance;
57 if (!ballistic && !surfaceFollow &&
58 (action.kind == ClimbingActionKind::Mantle || action.kind == ClimbingActionKind::ClimbUp || contactVault)) {
59 constexpr float riseEnd = 0.4f;
60 constexpr float crossEnd = 0.85f;
61 const float clearance = std::max({start.y, candidate.topPoint.y + profile.skin,
62 candidate.landingFeet.y + profile.skin});
63 const Vec3 raisedStart{start.x, clearance, start.z};
64 const Vec3 raisedLanding{candidate.landingFeet.x, clearance, candidate.landingFeet.z};
65 if (t < riseEnd) return interpolate(start, raisedStart, smoothStep(t / riseEnd));
66 if (t < crossEnd)
68 return interpolate(raisedStart, raisedLanding, smoothStep((t - riseEnd) / (crossEnd - riseEnd)));
70 return interpolate(raisedLanding, candidate.landingFeet, smoothStep((t - crossEnd) / (1.f - crossEnd)));
71 }
72
73 if (surfaceFollow || action.kind == ClimbingActionKind::Shimmy ||
74 action.kind == ClimbingActionKind::CornerInner ||
75 action.kind == ClimbingActionKind::CornerOuter || action.kind == ClimbingActionKind::ClimbDown ||
76 action.kind == ClimbingActionKind::LadderMount || action.kind == ClimbingActionKind::LadderClimb ||
77 action.kind == ClimbingActionKind::LadderDismount || action.kind == ClimbingActionKind::Slide ||
78 action.kind == ClimbingActionKind::BeamBalance)
80 return interpolate(start, candidate.landingFeet, smoothStep(t));
81
82 Vec3 point = interpolate(start, candidate.landingFeet, t);
83 point.y += 4.f * action.apexHeight * t * (1.f - t);
84 return point;
85}
