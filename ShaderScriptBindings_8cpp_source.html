17Result<std::vector<uint32_t>> shaderWords(ssq::Array array) {
18 std::vector<uint32_t> words;
19 words.reserve(array.size());
20 for (size_t i = 0; i < array.size(); ++i) {
21 if (array.get<ssq::Object>(i).getType() != ssq::Type::INTEGER)
22 return Result<std::vector<uint32_t>>::failure(
23 Diagnostic::error(DiagnosticCode::InvalidArgument, "SPIR-V words must be integers"));
24 const auto word = array.get<SQInteger>(i);
25 if (word < 0 || static_cast<uint64_t>(word) > UINT32_MAX)
26 return Result<std::vector<uint32_t>>::failure(
27 Diagnostic::error(DiagnosticCode::InvalidArgument, "SPIR-V words must be unsigned 32-bit integers"));
28 words.push_back(static_cast<uint32_t>(word));
29 }
30 return Result<std::vector<uint32_t>>::success(std::move(words));
31}

35Result<std::vector<std::uint32_t>> readShaderStageFile(const std::string& path) {
36 try {
37 auto* fs = filesystem::Filesystem::create();
38 std::unique_ptr<filesystem::FileData> bytes(fs->read(path));
39 if (!bytes || bytes->getSize() < 20 || bytes->getSize() % 4 != 0)
40 return Result<std::vector<std::uint32_t>>::failure(Diagnostic::error(
41 DiagnosticCode::ParseError, "SPIR-V must contain a complete word-aligned header", path));
42 std::vector<std::uint32_t> words(bytes->getSize() / 4);
43 std::memcpy(words.data(), bytes->getData(), bytes->getSize());
44 if (words[0] != 0x07230203)
45 return Result<std::vector<std::uint32_t>>::failure(Diagnostic::error(
46 DiagnosticCode::ParseError, "SPIR-V magic mismatch", path));
47 return Result<std::vector<std::uint32_t>>::success(std::move(words));
48 } catch (const std::exception& error) {
49 return Result<std::vector<std::uint32_t>>::failure(
50 Diagnostic::error(DiagnosticCode::Failed, error.what(), path));
51 }
52}

57ResultRef<Shader> loadMeshShader(Graphics& graphics, const std::string& vertex,
58 const std::string& fragment) {
59 if (graphics.getBackendName() != "vulkan")
60 return ResultRef<Shader>::failure(Diagnostic::error(
61 DiagnosticCode::Unsupported, "SPIR-V mesh shaders require the Vulkan backend", "backend"));
62 if (fragment.empty())
63 return ResultRef<Shader>::failure(Diagnostic::error(
64 DiagnosticCode::InvalidArgument, "fragment path must not be empty", "fragment"));
65 std::vector<std::uint32_t> vert;
66 if (!vertex.empty()) {
67 auto decoded = detail::readShaderStageFile(vertex);
68 if (!decoded) return ResultRef<Shader>::failure(decoded.status());
69 vert = std::move(decoded).takeValue();
70 }
71 auto frag = detail::readShaderStageFile(fragment);
72 if (!frag) return ResultRef<Shader>::failure(frag.status());
73 try {
74 auto* shader = graphics.newMeshShaderFromSpv(vert, frag.value());
75 if (!shader)
76 return ResultRef<Shader>::failure(Diagnostic::error(
77 DiagnosticCode::Failed, "Mesh shader allocation failed", "shader"));
78 return ResultRef<Shader>::success(std::ref(*shader));
79 } catch (const std::exception& error) {
80 return ResultRef<Shader>::failure(
81 Diagnostic::error(DiagnosticCode::Failed, error.what(), "shader"));
82 }
83}
