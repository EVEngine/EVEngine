308vec3 surfaceDerivative(vec3 forward, vec3 backward, bool forwardValid, bool backwardValid) {
309 if (!forwardValid && backwardValid) return backward;
310 if (!backwardValid && forwardValid) return forward;
311 // Preserve symmetry when differences are indistinguishable at depth-key precision.
312 float tolerance = 2.0 * (pc.d[20] - pc.d[19]) / 16777215.0;
313 if ((!forwardValid && !backwardValid) || abs(abs(forward.z) - abs(backward.z)) <= tolerance)
314 return (forward + backward) * 0.5;
315 return abs(forward.z) < abs(backward.z) ? forward : backward;

387 if (particleTint) mode -= 20;
388 bool uniformTint = !particleTint && mode >= 10;
389 if (uniformTint) mode -= 10;
390 vec3 outC;
391 float alpha;
392 if (uniformTint || particleTint) {
393 uint accumulatedAt = i * 5u;

395 if (accumulatedWeight == 0u) {

397 return;
398 }
399 vec4 tint = particleTint
400 ? vec4(accum.v[accumulatedAt + 1u], accum.v[accumulatedAt + 2u],
401 accum.v[accumulatedAt + 3u], accum.v[accumulatedAt + 4u]) / float(accumulatedWeight)
402 : vec4(pc.d[0], pc.d[1], pc.d[2], pc.d[3]);
403 vec3 reflectionColor = vec3(pc.d[4], pc.d[5], pc.d[6]);
404 bool lighting = pc.d[7] > 0.5;
405 float smoothness = pc.d[8];
406 float metalness = pc.d[9];
407 float ambient = pc.d[10];
408 float reflection = pc.d[11];
409 float opacity = pc.d[12];
410 float cutoff = pc.d[13];
411 if (thickness * 10.0 < cutoff) {
412 color.c[i] = 0u;
413 return;
414 }
415 float light = lighting ?
416 clamp(ambient + (1.0 - min(ambient, 1.0)) * diff, 0.0, 6.0) : 1.0;
417 float exponent = 4.0 + 124.0 * smoothness;
418 float spec = lighting ? pow(max(n.z, 0.0), exponent) * smoothness * 0.56 : 0.0;
419 float fresnel = 0.04 + 0.96 * pow(1.0 - max(n.z, 0.0), 5.0);
420 vec3 reflected = mix(reflectionColor, tint.rgb, metalness);
421 outC = tint.rgb * light + reflected * fresnel * reflection + vec3(spec);
422 alpha = clamp(thickness * opacity, 0.0, 1.0) * tint.a;
423 } else if (mode == 1) {
424 // Mud: diffuse brown, darker with thickness, rough specular.
425 vec3 base = vec3(0.36, 0.23, 0.12) * (0.45 + 0.55 * diff);
426 float attenuation = exp(-thickness * 1.8);
427 vec3 Hv = normalize(L + V);
428 float spec = pow(max(dot(n, Hv), 0.0), 8.0) * 0.12;
429 outC = base * attenuation + vec3(spec);
430 alpha = clamp(thickness * 0.6, 0.0, 1.0);
431 } else {
432 vec3 base = vec3(0.05, 0.32, 0.72) * (0.55 + 0.45 * diff);
433 float fresnel = 0.04 + 0.96 * pow(1.0 - max(dot(n, V), 0.0), 5.0);
434 vec3 Hv = normalize(L + V);
435 float spec = pow(max(dot(n, Hv), 0.0), 64.0) * 0.45;
436 outC = base + vec3(0.55, 0.72, 1.0) * fresnel * 0.75 + vec3(spec);
437 alpha = clamp(thickness * 0.35, 0.0, 1.0);
438 }
439 // Match the CPU output's clamp and truncation (packUnorm4x8 rounds instead).
440 uvec4 rgba = uvec4(clamp(vec4(outC, alpha), 0.0, 1.0) * 255.0);

442}
