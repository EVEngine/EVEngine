364vec3 surfaceDerivative(vec3 forward, vec3 backward, bool forwardValid, bool backwardValid) {
365 if (!forwardValid && backwardValid) return backward;
366 if (!backwardValid && forwardValid) return forward;
367 // Preserve symmetry when differences are indistinguishable at depth-key precision.

369 if ((!forwardValid && !backwardValid) || abs(abs(forward.z) - abs(backward.z)) <= tolerance)
370 return (forward + backward) * 0.5;
372 return abs(forward.z) < abs(backward.z) ? forward : backward;

456 bool uniformTint = !particleTint && mode >= 10;
457 if (uniformTint) mode -= 10;

459 float alpha;
460 if (uniformTint || particleTint) {
461 uint accumulatedAt = i * 5u;

466 }
467 vec4 tint = particleTint
468 ? vec4(accum.v[accumulatedAt + 1u], accum.v[accumulatedAt + 2u],
469 accum.v[accumulatedAt + 3u], accum.v[accumulatedAt + 4u]) / float(accumulatedWeight)

472 vec3 reflectionColor = vec3(pc.d[4], pc.d[5], pc.d[6]);
473 bool lighting = pc.d[7] > 0.5;
474 float smoothness = pc.d[8];
475 float metalness = pc.d[9];
476 float ambient = pc.d[10];
477 float reflection = pc.d[11];
478 float opacity = pc.d[12];
479 float cutoff = pc.d[13];

488 float spec = lighting ? pow(max(n.z, 0.0), exponent) * smoothness * 0.56 : 0.0;

490 vec3 reflected = mix(reflectionColor, tint.rgb, metalness);

492 alpha = clamp(thickness * opacity, 0.0, 1.0) * tint.a;
493 } else if (mode == 1) {

508 }
