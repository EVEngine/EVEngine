122float decode(uint key) {
123 float nearZ = max(pc.d[19], 1e-4);
124 float farZ = max(pc.d[20], nearZ + 1e-3);
125 return nearZ + (float(key) / 16777215.0) * (farZ - nearZ);

128void main() {
129 uint i = gl_GlobalInvocationID.x;
130 uint W = uint(pc.d[23]);
131 uint H = uint(pc.d[24]);
132 if (i >= W * H) return;
133 uint cx = i % W;
134 uint cy = i / W;
135 uint centerKey = depthA.d[i];
136 if (centerKey == 0xFFFFFFFFu) {
137 depthB.d[i] = centerKey;
138 return;
139 }
140 float center = decode(centerKey);
141 float falloff = max(pc.d[28], 1e-4); // depth edge falloff
142 float sum = center;
143 float wsum = 1.0;
144 for (int oy = -2; oy <= 2; ++oy) {
145 for (int ox = -2; ox <= 2; ++ox) {
146 if (ox == 0 && oy == 0) continue;
147 int xx = int(cx) + ox;
148 int yy = int(cy) + oy;
149 if (xx < 0 || yy < 0 || xx >= int(W) || yy >= int(H)) continue;
150 uint idx = uint(yy) * W + uint(xx);
151 uint nk = depthA.d[idx];
152 if (nk == 0xFFFFFFFFu) continue;
153 float nd = decode(nk);
154 float spatial = exp(-(float(ox * ox + oy * oy)) * 0.5);
155 float wDepth = exp(-abs(nd - center) / falloff);
156 float w = spatial * wDepth;
157 sum += nd * w;
158 wsum += w;
159 }
160 }
161 float outD = sum / max(wsum, 1e-5);
162 float nearZ = max(pc.d[19], 1e-4);
163 float farZ = max(pc.d[20], nearZ + 1e-3);
164 float t = clamp((outD - nearZ) / (farZ - nearZ), 0.0, 1.0);
165 depthB.d[i] = uint(t * 16777215.0);
166}
167)GLSL";

237void main() {
238 uint i = gl_GlobalInvocationID.x;
239 uint W = uint(pc.d[23]);
240 uint H = uint(pc.d[24]);
241 if (i >= W * H) return;
242 if (depthA.d[i] == 0xFFFFFFFFu) {
243 color.c[i] = vec4(0.0);
244 return;
245 }
246 vec3 n = normal.n[i].xyz;
247 float thickness = float(thick.t[i]) / 256.0;
248 vec3 L = normalize(vec3(0.35, 0.65, 0.55));
249 vec3 V = vec3(0.0, 0.0, 1.0);
250 float diff = max(dot(n, L), 0.0);
251 int mode = int(pc.d[26]);
252 vec3 outC;
253 float alpha;
254 if (mode == 1) {
255 // Mud: diffuse brown, darker with thickness, rough specular.
256 vec3 base = vec3(0.36, 0.23, 0.12) * (0.45 + 0.55 * diff);
257 float attenuation = exp(-thickness * 1.8);
258 vec3 Hv = normalize(L + V);
259 float spec = pow(max(dot(n, Hv), 0.0), 8.0) * 0.12;
260 outC = base * attenuation + vec3(spec);
261 alpha = clamp(thickness * 0.6, 0.0, 1.0);
262 } else {
263 // Water: blue base, Fresnel rim, sharp specular.
264 vec3 base = vec3(0.05, 0.32, 0.72) * (0.55 + 0.45 * diff);
265 float fresnel = 0.04 + 0.96 * pow(1.0 - max(dot(n, V), 0.0), 5.0);
266 vec3 Hv = normalize(L + V);
267 float spec = pow(max(dot(n, Hv), 0.0), 64.0) * 0.45;
268 outC = base + vec3(0.55, 0.72, 1.0) * fresnel * 0.75 + vec3(spec);
269 alpha = clamp(thickness * 0.35, 0.0, 1.0);
270 }
271 color.c[i] = vec4(outC, alpha);
272}
273)GLSL";
