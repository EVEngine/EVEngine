22Result<int> worldToCell(double coordinate, double origin, double cellSize, std::string path) {
23 const double projected = std::floor((coordinate - origin) / cellSize);
24 if (!std::isfinite(projected) || projected < static_cast<double>(std::numeric_limits<int>::min()) ||
25 projected > static_cast<double>(std::numeric_limits<int>::max()))
26 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
27 "navigation coordinate is outside grid range", std::move(path)));
28 return Result<int>::success(static_cast<int>(projected));
29}

54Result<CombatNavigationSteering> PathfinderCombatNavigationProvider::steer(
55 const CombatLocomotionState& state, const CombatNavigationGoal& goal, SimulationTick tick) {
56 (void)tick;
57 if (!finite(state.position) || !finite(goal.position) || !std::isfinite(goal.acceptanceRadius) ||
58 goal.acceptanceRadius < 0.0)
59 return Result<CombatNavigationSteering>::failure(
60 Diagnostic::error(DiagnosticCode::InvalidArgument, "combat navigation request is invalid", "navigation"));
61
62 const CombatVector2 goalDelta{goal.position.x - state.position.x,
63 goal.position.z - state.position.z};
64 if (length(goalDelta) <= goal.acceptanceRadius)
65 return Result<CombatNavigationSteering>::success(
66 {CombatNavigationPhase::Arrived, {}, 0.0});
67
68 auto startX = worldToCell(state.position.x, config_.origin.x, config_.cellSize, "state.position.x");
69 if (!startX) return Result<CombatNavigationSteering>::failure(startX.status());
70 auto startY = worldToCell(state.position.z, config_.origin.z, config_.cellSize, "state.position.z");
71 if (!startY) return Result<CombatNavigationSteering>::failure(startY.status());
72 auto goalX = worldToCell(goal.position.x, config_.origin.x, config_.cellSize, "goal.position.x");
73 if (!goalX) return Result<CombatNavigationSteering>::failure(goalX.status());
74 auto goalY = worldToCell(goal.position.z, config_.origin.z, config_.cellSize, "goal.position.z");
75 if (!goalY) return Result<CombatNavigationSteering>::failure(goalY.status());
76
77 std::unique_ptr<map::Path> path(pathfinder_.get().findPath(startX.value(), startY.value(),
78 goalX.value(), goalY.value()));
79 if (!path || path->empty())
80 return Result<CombatNavigationSteering>::failure(Diagnostic::error(
81 DiagnosticCode::NotFound, "combat navigation goal is unreachable", state.subject.format()));
82
83 CombatVector2 waypoint = goal.position;
84 if (path->getLength() > 1) {
85 waypoint.x = config_.origin.x + (static_cast<double>(path->getX(1)) + 0.5) * config_.cellSize;
86 waypoint.z = config_.origin.z + (static_cast<double>(path->getY(1)) + 0.5) * config_.cellSize;
87 }
88 return Result<CombatNavigationSteering>::success(
89 {CombatNavigationPhase::Moving,
90 normalized({waypoint.x - state.position.x, waypoint.z - state.position.z}), 1.0});
91}
