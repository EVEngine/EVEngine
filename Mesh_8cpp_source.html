16void Mesh::captureImportedAttributes(const ::aiMesh &mesh) {
17 importedUvs_.clear();
18 for (unsigned channel = 0; channel < mesh.GetNumUVChannels(); ++channel) {
19 if (!mesh.HasTextureCoords(channel)) continue;
20 std::vector<float> values;
21 values.reserve(size_t(mesh.mNumVertices) * 2u);
22 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
23 values.push_back(mesh.mTextureCoords[channel][i].x);
24 values.push_back(mesh.mTextureCoords[channel][i].y);
25 }
26 importedUvs_.push_back(std::move(values));
27 }
28 importedColors_.clear();
29 for (unsigned channel = 0; channel < mesh.GetNumColorChannels(); ++channel) {
30 if (!mesh.HasVertexColors(channel)) continue;
31 std::vector<float> values;
32 values.reserve(size_t(mesh.mNumVertices) * 4u);
33 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
34 const aiColor4D &c = mesh.mColors[channel][i];
35 values.insert(values.end(), {c.r, c.g, c.b, c.a});
36 }
37 importedColors_.push_back(std::move(values));
38 }
39 importedTangents_.clear();
40 importedBitangents_.clear();
41 if (mesh.HasTangentsAndBitangents()) {
42 importedTangents_.reserve(size_t(mesh.mNumVertices) * 3u);
43 importedBitangents_.reserve(size_t(mesh.mNumVertices) * 3u);
44 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
45 const aiVector3D &t = mesh.mTangents[i];
46 const aiVector3D &b = mesh.mBitangents[i];
47 importedTangents_.insert(importedTangents_.end(), {t.x, t.y, t.z});
48 importedBitangents_.insert(importedBitangents_.end(), {b.x, b.y, b.z});
49 }
50 }
51}

71void Mesh::computeBounds(const float *posXYZ, int vertexCount) {
72 boundsCx = 0.f;
73 boundsCy = 0.f;
74 boundsCz = 0.f;
75 boundsRadius = 0.f;
76 if (!posXYZ || vertexCount <= 0) return;
77
78 const size_t n = size_t(vertexCount);
79 glm::vec3 c(0.f);
80 for (size_t i = 0; i < n; ++i) {
81 c.x += posXYZ[i * 3u + 0u];
82 c.y += posXYZ[i * 3u + 1u];
83 c.z += posXYZ[i * 3u + 2u];
84 }
85 c /= float(n);
86 boundsCx = c.x;
87 boundsCy = c.y;
88 boundsCz = c.z;
89
90 float r = 0.f;
91 for (size_t i = 0; i < n; ++i) {
92 const float dx = posXYZ[i * 3u + 0u] - c.x;
93 const float dy = posXYZ[i * 3u + 1u] - c.y;
94 const float dz = posXYZ[i * 3u + 2u] - c.z;
95 const float d = std::sqrt(dx * dx + dy * dy + dz * dz);
96 if (d > r) r = d;
97 }
98 // Degenerate (single-point) meshes still get a tiny non-zero sphere so
99 // hasBounds() stays meaningful and the culler never drops point geometry.
100 boundsRadius = r > 0.f ? r : 1e-4f;
101}

103void Mesh::drawOcclusion(Graphics *gfx, const glm::mat4 &matrix) const {
104 if (!gfx || !getCastOcclusion()) return;
105 const float x = matrix[3][0];
106 const float y = matrix[3][1];
107 const float w = std::sqrt(matrix[0][0] * matrix[0][0] + matrix[0][1] * matrix[0][1]);
108 const float h = std::sqrt(matrix[1][0] * matrix[1][0] + matrix[1][1] * matrix[1][1]);
109 if (w <= 0.f || h <= 0.f) return;
110 gfx->drawOcclusionSolid(x, y, w, h);
111}

206void Mesh::computeMorphedPositions(std::vector<float> &outPos, std::vector<float> &outNrm) const {
207 outPos = basePos_;
208 outNrm = baseNrm_;
209 if (outPos.empty()) return;
210 for (const MorphTarget &t : morphs_) {
211 auto it = morphWeights_.find(t.name);
212 const float w = (it == morphWeights_.end()) ? 0.f : it->second;
213 if (std::fabs(w) < 1e-8f) continue;
214 for (size_t i = 0; i < outPos.size() && i < t.deltaPos.size(); ++i)
215 outPos[i] += t.deltaPos[i] * w;
216 }
217}
