32void StyleChain::apply(graphics::Graphics* gfx, graphics::Texture* source, graphics::Canvas* dest,
33 graphics::Canvas* temp) {
34 if (!gfx) throw eve::Exception("StyleChain.apply: null graphics");
35 if (!source) throw eve::Exception("StyleChain.apply: null source");
36 if (!dest) throw eve::Exception("StyleChain.apply: null dest");
37 if (passes_.empty()) throw eve::Exception("StyleChain.apply: no passes");
38
39 if (passes_.size() == 1) {
40 passes_[0]->applyTo(gfx, source, dest);
41 return;
42 }
43 if (!temp) throw eve::Exception("StyleChain.apply: temp canvas required for multi-pass");
44 if (!temp->getTexture()) throw eve::Exception("StyleChain.apply: temp canvas has no sampleable texture");
45 if (!dest->getTexture()) throw eve::Exception("StyleChain.apply: dest canvas has no sampleable texture");
46
47 // Alternate temp / dest. Final result must live in dest.
48 // n even: … → temp → dest
49 // n odd : … → dest → temp, then blit temp → dest
50 graphics::Texture* in = source;
51 const int n = int(passes_.size());
52 for (int i = 0; i < n; ++i) {
53 graphics::Canvas* out = (i & 1) ? dest : temp;
54 passes_[size_t(i)]->applyTo(gfx, in, out);
55 in = out->getTexture();
56 if (!in) throw eve::Exception("StyleChain.apply: intermediate texture missing");
57 }
58 if ((n & 1) != 0) {
59 // Last write was temp — blit into dest (no style shader).
60 graphics::Canvas* prev = gfx->getCanvas();
61 gfx->setCanvas(dest);
62 gfx->drawTexturedRect(temp->getTexture(), 0.f, 0.f, float(dest->getWidth()), float(dest->getHeight()),
63 glm::vec4(1.f, 1.f, 1.f, 1.f));
64 gfx->setCanvas(prev);
65 }
66}

262void StylePass::apply(graphics::Graphics* gfx, graphics::Texture* source) {
263 if (!gfx) throw eve::Exception("StylePass.apply: null graphics");
264 if (!source) throw eve::Exception("StylePass.apply: null source texture");
265 if (!shader_) throw eve::Exception("StylePass.apply: null shader");
266
267 const int w = source->getWidth();
268 const int h = source->getHeight();
269 uploadScreenUniforms(w, h);
270
271 // Prefer destination canvas size for pixel grid when available.
272 if (auto* dst = gfx->getCanvas()) {
273 if (hasParam("screenW")) setFloat("screenW", float(dst->getWidth()));
274 if (hasParam("screenH")) setFloat("screenH", float(dst->getHeight()));

276 if (hasParam("screenW")) setFloat("screenW", float(gfx->getWidth()));
277 if (hasParam("screenH")) setFloat("screenH", float(gfx->getHeight()));
278 }
279
280 const float dw = gfx->getCanvas() ? float(gfx->getCanvas()->getWidth()) : float(gfx->getWidth());
281 const float dh = gfx->getCanvas() ? float(gfx->getCanvas()->getHeight()) : float(gfx->getHeight());

283 gfx->drawTexturedRectShader(source, shader_, 0.f, 0.f, dw, dh, glm::vec4(1.f, 1.f, 1.f, 1.f));

334void StyleRecipe::compile(graphics::Graphics* gfx) {
335 if (!gfx) throw eve::Exception("StyleRecipe.compile: null graphics");
336 if (instances_.empty()) throw eve::Exception("StyleRecipe.compile: no styles");
337
338 if (graphics_ != gfx) {
339 scratch_ = nullptr;
340 scratchWidth_ = 0;
341 scratchHeight_ = 0;
342 }
343 std::stable_sort(instances_.begin(), instances_.end(),
344 [](StyleInstance* a, StyleInstance* b) { return a->getPriority() < b->getPriority(); });
345 stage_ = instances_.front()->getStage();
346 passes_.clear();
347 for (StyleInstance* instance : instances_) {
348 if (instance->getStage() != stage_)
349 throw eve::Exception("StyleRecipe.compile: cannot mix stages '%s' and '%s'", stage_.c_str(),
350 instance->getStage().c_str());
351 passes_.emplace_back(instance->newPass(gfx));
352 }
353 graphics_ = gfx;
354 compiled_ = true;
355}

357void StyleRecipe::ensureScratch(graphics::Graphics* gfx, graphics::Canvas* dest) {
358 if (passes_.size() < 2) return;
359 const int width = dest->getWidth();
360 const int height = dest->getHeight();
361 if (graphics_ != gfx || !scratch_ || scratchWidth_ != width || scratchHeight_ != height) {
362 scratch_ = gfx->newCanvas(width, height);
363 scratchWidth_ = width;
364 scratchHeight_ = height;
365 graphics_ = gfx;
366 }
367}

369void StyleRecipe::apply(graphics::Graphics* gfx, graphics::Texture* source, graphics::Canvas* dest) {
370 if (!compiled_) throw eve::Exception("StyleRecipe.apply: compile first");
371 if (graphics_ != gfx) throw eve::Exception("StyleRecipe.apply: compiled for another Graphics");
372 if (!source || !dest) throw eve::Exception("StyleRecipe.apply: null source or destination");
373 ensureScratch(gfx, dest);
374
375 StyleChain chain;
376 for (const auto& pass : passes_) chain.add(pass.get());
377 chain.apply(gfx, source, dest, scratch_);
378}
