37Poco::JSON::Object::Ptr parseToolTextObject(const std::string& envelopeOrText) {
38 try {
39 Poco::JSON::Parser parser;
40 auto root = parser.parse(envelopeOrText);
41 if (root.type() == typeid(Poco::JSON::Object::Ptr)) {
42 auto object = root.extract<Poco::JSON::Object::Ptr>();
43 if (object && object->has("content")) {
44 auto content = object->getArray("content");
45 if (content && content->size() > 0) {
46 auto item = content->getObject(0);
47 if (item && item->has("text")) {
48 auto inner = parser.parse(item->getValue<std::string>("text"));
49 if (inner.type() == typeid(Poco::JSON::Object::Ptr))
50 return inner.extract<Poco::JSON::Object::Ptr>();
51 }
52 }
53 }
54 return object;
55 }
56 } catch (...) {
57 }
58 return {};
59}

71std::string whiteboxPlaytest(Poco::JSON::Object::Ptr args) {
72 if (!args) {
73 return errorPayload("recipe object is required",
74 "Pass {reset?, scene:[...], inputs?, waitFrames?, capture?, capturePath?, play?}");
75 }
76
77 Debugger& dbg = Debugger::instance();
78 HSQUIRRELVM vm = dbg.vm();
79 if (!vm) return errorPayload("no VM", "Attach a live eve run --mcp-port session first");
80
81 std::string err;
82 if (!ensureSceneDirectorInstalled(vm, &err, /*allowHeavyWork=*/false)) {
83 if (err == "deferred-to-update") {
84 return errorPayload(
85 "deferred-to-update",
86 "scene_director install was queued for the next eve.dev.poll; retry eve_whitebox_playtest");
87 }
88 return errorPayload(err.empty() ? "scene_director kit unavailable" : err);
89 }
90
91 Poco::JSON::Object::Ptr report = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
92 report->set("schemaId", "eve.whitebox.playtest-report");
93 report->set("schemaVersion", 1);
94
95 const bool doReset = getArgBool(args, "reset", true);
96 if (doReset) {
97 const std::string resetOut = callSceneDirectorReturn(vm, "return ::scene_director.reset();", &err);
98 if (!err.empty()) return errorPayload(err);
99 if (auto resetObj = parseJsonObject(resetOut))
100 report->set("reset", resetObj);
101 else
102 report->set("reset", resetOut);
103 }
104
105 if (args->has("scene")) {
106 Poco::Dynamic::Var actionsVar;
107 try {
108 actionsVar = Poco::Dynamic::Var(args->getArray("scene"));
109 } catch (...) {
110 return errorPayload("scene must be a JSON array of {action,target?,params?}");
111 }
112 Poco::JSON::Object::Ptr options = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
113 options->set("stopOnError", getArgBool(args, "stopOnError", true));
114 auto actionsLiteral = sceneLiteral(actionsVar);
115 if (!actionsLiteral.ok()) return errorPayload(actionsLiteral.status().describe());
116 auto optionsLiteral = sceneLiteral(Poco::Dynamic::Var(options));
117 if (!optionsLiteral.ok()) return errorPayload(optionsLiteral.status().describe());
118 err.clear();
119 const std::string snippet =
120 "return ::scene_director.batch(" + actionsLiteral.value() + "," + optionsLiteral.value() + ");";
121 const std::string sceneOut = callSceneDirectorReturn(vm, snippet, &err);
122 if (!err.empty()) return errorPayload(err);
123 auto sceneObj = parseJsonObject(sceneOut);
124 if (!sceneObj) return errorPayload("scene batch returned non-object JSON", sceneOut);
125 report->set("scene", sceneObj);
126 if (!sceneObj->optValue<bool>("ok", false)) {
127 report->set("ok", false);
128 report->set("error", "scene batch failed");
129 report->set("hint", "Inspect scene.results for the first failing white-box action");
130 return textPayload(report, true);
131 }
132 }
133
134 if (args->has("inputs")) {
135 Poco::JSON::Object::Ptr inputArgs = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
136 try {
137 inputArgs->set("actions", args->getArray("inputs"));
138 } catch (...) {
139 return errorPayload("inputs must be a JSON array of eve_input_batch actions");
140 }
141 if (args->has("waitFrames")) inputArgs->set("waitFrames", getArgInt(args, "waitFrames", 0));
142 if (args->has("capture")) inputArgs->set("capture", getArgBool(args, "capture", false));
143 if (args->has("capturePath"))
144 inputArgs->set("capturePath", getArgString(args, "capturePath", "wb_playtest.png"));
145 auto inputObj = parseToolTextObject(callMcpInputTool("eve_input_batch", inputArgs));
146 if (!inputObj) return errorPayload("input batch returned an unexpected payload");
147 report->set("input", inputObj);
148 if (!inputObj->optValue<bool>("ok", false)) {
149 report->set("ok", false);
150 report->set("error", "input batch failed");
151 report->set("hint", "Fix inputs[] action ops; then call eve_input_poll if status=pending");
152 return textPayload(report, true);
153 }
154 }
155
156 if (args->has("play")) {
157 Poco::JSON::Array::Ptr playRequests;
158 try {
159 playRequests = args->getArray("play");
160 } catch (...) {
161 return errorPayload("play must be a JSON array of eve_play request objects");
162 }
163 Poco::JSON::Object::Ptr batchReq = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
164 batchReq->set("schemaId", "evengine.play-request");
165 batchReq->set("schemaVersion", 1);
166 batchReq->set("op", "batch");
167 Poco::JSON::Array::Ptr wrapped = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
168 for (std::size_t i = 0; i < playRequests->size(); ++i) {
169 Poco::JSON::Object::Ptr one;
170 try {
171 one = playRequests->getObject(static_cast<unsigned>(i));
172 } catch (...) {
173 return errorPayload("play[" + std::to_string(i) + "] must be an object");
174 }
175 if (!one) return errorPayload("play[" + std::to_string(i) + "] must be an object");
176 if (!one->has("schemaId")) one->set("schemaId", "evengine.play-request");
177 if (!one->has("schemaVersion")) one->set("schemaVersion", 1);
178 wrapped->add(one);
179 }
180 batchReq->set("requests", wrapped);
181 auto playResponse = executePlayJson(mcpStringify(Poco::Dynamic::Var(batchReq)));
182 if (!playResponse) {
183 report->set("ok", false);
184 report->set("error", playResponse.status().describe());
185 report->set("hint", "Play Host requires game.agent.json for observe/act; use clock/step/status freely");
186 return textPayload(report, true);
187 }
188 auto playObj = parseJsonObject(playResponse.value());
189 if (playObj)
190 report->set("play", playObj);
191 else
192 report->set("play", playResponse.value());
193 }
194
195 // Fresh scene truth for assertions / agent follow-up.
196 err.clear();
197 const std::string infoOut = callSceneDirectorReturn(vm, "return ::scene_director.info();", &err);
198 if (err.empty()) {
199 if (auto info = parseJsonObject(infoOut)) report->set("info", info);
200 }
201
202 report->set("ok", true);
203 if (report->has("input")) {
204 auto input = report->getObject("input");
205 if (input && input->optValue<std::string>("status", "") == "pending") {
206 report->set("status", "pending");
207 report->set("hint", "Call eve_input_poll until status=ready, then continue play asserts if needed");
208 Poco::JSON::Array::Ptr next = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
209 next->add("eve_input_poll");
210 report->set("nextTools", next);
211 return textPayload(report);
212 }
213 }
214 report->set("status", "ready");
215 report->set("hint", "White-box scene + playtest submitted; inspect info / play / input");
216 return textPayload(report);
217}
