46Result<void> OceanWake::inject(std::span<const OceanWakeImpulse> impulses) {
47 const auto& s = impl_->settings;
48 if (impl_->revision == std::numeric_limits<std::uint64_t>::max() || impulses.size() > 256) return invalid<void>();
49 for (const auto& p : impulses)
50 if (!finite(p.position) || !finite(p.direction) || !std::isfinite(p.radius) ||
51 p.radius < s.length / s.resolution || p.radius > s.length * 0.25 || !std::isfinite(p.velocity) ||
52 std::abs(p.velocity) > 5.0 || !std::isfinite(p.foam) || p.foam < 0.0 || p.foam > 1.0)
53 return invalid<void>();
54 for (unsigned z = 0; z < s.resolution; ++z) {
55 for (unsigned x = 0; x < s.resolution; ++x) {
56 const auto index = z * s.resolution + x;
57 const auto point = s.center + (glm::dvec2(x, z) / double(s.resolution) - 0.5) * s.length;
58 double rate = 0.0, foam = 0.0;
59 for (const auto& p : impulses) {
60 const auto delta = (point - p.position) / p.radius;
61 const double distance2 = glm::dot(delta, delta);
62 if (distance2 > 16.0) continue;
63 const double envelope = std::exp(-0.5 * distance2);
64 const double directionLength = glm::length(p.direction);
65 const double shape = directionLength > 1e-8 ? glm::dot(delta, p.direction / directionLength) : 1.0;
66 rate += p.velocity * envelope * shape;
67 foam += p.foam * envelope;
68 }
69 impl_->velocity[index] = std::clamp(impl_->velocity[index] + rate, -10.0, 10.0);
70 impl_->foam[index] = std::clamp(impl_->foam[index] + foam, 0.0, 1.0);
71 }
72 }
73 ++impl_->revision;
74 return Result<void>::success();
75}

97Result<void> OceanWake::advance(double dt, double endTime, const OceanSpectrumFrame& water) {
98 if (!std::isfinite(dt) || dt <= 0.0 || dt > 1.0 / 30.0 || !std::isfinite(endTime) ||
99 std::abs(endTime - impl_->time - dt) > 1e-8 || endTime > 1e9 || std::abs(water.time() - impl_->time) > 1e-8)
100 return invalid<void>();
101 const auto& s = impl_->settings;
102 const auto n = s.resolution;
103 // Stage transport before mutating any wave state: an inverse-query failure is atomic.
104 std::vector<double> foam(impl_->foam.size(), 0.0);
105 if (std::any_of(impl_->foam.begin(), impl_->foam.end(), [](double f) { return f > 0.0; })) {
106 const double cell = s.length / n;
107 auto coverage = [&](glm::dvec2 point) {
108 const auto uv = (point - s.center) / cell + double(n) * 0.5;
109 if (uv.x <= -1.0 || uv.y <= -1.0 || uv.x >= n || uv.y >= n) return 0.0;
110 const auto base = glm::floor(uv), fraction = uv - base;
111 double result = 0.0;
112 for (int dz = 0; dz < 2; ++dz)
113 for (int dx = 0; dx < 2; ++dx) {
114 const int x = int(base.x) + dx, z = int(base.y) + dz;
115 if (x < 0 || z < 0 || x >= int(n) || z >= int(n)) continue;
116 result += impl_->foam[z * n + x] * (dx ? fraction.x : 1.0 - fraction.x) *
117 (dz ? fraction.y : 1.0 - fraction.y);
118 }
119 return std::clamp(result, 0.0, 1.0);
120 };
121 for (unsigned z = 0; z < n; ++z)
122 for (unsigned x = 0; x < n; ++x) {
123 const auto point = s.center + (glm::dvec2(x, z) - double(n) * 0.5) * cell;
124 auto start = water.sample(point);
125 if (!start) return Result<void>::failure(start.status());
126 const auto v = start.value().velocity;
127 auto midpoint = water.sample(point - glm::dvec2(v.x, v.z) * (dt * 0.5));
128 if (!midpoint) return Result<void>::failure(midpoint.status());
129 const auto m = midpoint.value().velocity;
130 foam[z * n + x] = coverage(point - glm::dvec2(m.x, m.z) * dt);
131 }
132 }
133 std::vector<ocean_detail::Complex> height(impl_->height.begin(), impl_->height.end());
134 std::vector<ocean_detail::Complex> velocity(impl_->velocity.begin(), impl_->velocity.end());
135 ocean_detail::forward2D(height, n);
136 ocean_detail::forward2D(velocity, n);
137 for (unsigned z = 0; z < n; ++z) {
138 for (unsigned x = 0; x < n; ++x) {
139 const int ix = x < n / 2 ? int(x) : int(x) - int(n);
140 const int iz = z < n / 2 ? int(z) : int(z) - int(n);
141 const auto i = z * n + x;
142 const double k = std::hypot(double(ix), double(iz)) * ocean_detail::tau / s.length;
143 if (k == 0.0) {
144 height[i] = 0.0;
145 velocity[i] = 0.0;
146 continue;
147 }
148 const double omega = std::sqrt(s.gravity * k * std::tanh(k * s.depth));
149 const double sn = std::sin(omega * dt), cs = std::cos(omega * dt);
150 const auto oldHeight = height[i];
151 height[i] = oldHeight * cs + velocity[i] * (sn / omega);
152 velocity[i] = velocity[i] * cs - oldHeight * (omega * sn);
153 }
154 }
155 ocean_detail::inverse2D(height, n);
156 ocean_detail::inverse2D(velocity, n);
157 for (unsigned z = 0; z < n; ++z) {
158 for (unsigned x = 0; x < n; ++x) {
159 const auto i = z * n + x;
160 const double edge = double(std::min({x, z, n - 1 - x, n - 1 - z})) * s.length / n;
161 const double sponge = std::clamp(1.0 - edge / s.absorptionWidth, 0.0, 1.0);
162 const double decay = std::exp(-(s.damping + 8.0 * sponge * sponge) * dt);
163 impl_->height[i] = std::clamp(height[i].real() * decay, -5.0, 5.0);
164 impl_->velocity[i] = std::clamp(velocity[i].real() * decay, -10.0, 10.0);
165 impl_->foam[i] = foam[i] * std::exp(-(s.foamDecay + 8.0 * sponge * sponge) * dt);
166 }
167 }
168 impl_->time = endTime;
169 return Result<void>::success();
170}
171Result<void> OceanWake::recenter(glm::dvec2 center) {
172 if (impl_->revision == std::numeric_limits<std::uint64_t>::max() || !finite(center)) return invalid<void>();
173 auto& s = impl_->settings;
174 const double cell = s.length / s.resolution;
175 const auto shift = glm::round((center - s.center) / cell);
176 const auto oldHeight = impl_->height, oldVelocity = impl_->velocity, oldFoam = impl_->foam;
177 for (unsigned z = 0; z < s.resolution; ++z) {
178 for (unsigned x = 0; x < s.resolution; ++x) {
179 const auto i = z * s.resolution + x;
180 const double ox = x + shift.x, oz = z + shift.y;
181 if (ox >= 0.0 && ox < s.resolution && oz >= 0.0 && oz < s.resolution) {
182 const auto j = unsigned(oz) * s.resolution + unsigned(ox);
183 impl_->height[i] = oldHeight[j];
184 impl_->velocity[i] = oldVelocity[j];
185 impl_->foam[i] = oldFoam[j];
186 } else {
187 impl_->height[i] = 0.0;
188 impl_->velocity[i] = 0.0;
189 impl_->foam[i] = 0.0;
190 }
191 }
192 }
193 s.center += shift * cell;
194 ++impl_->revision;
195 return Result<void>::success();

197Result<OceanWakeSample> OceanWake::sample(glm::dvec2 worldXZ) const {
198 if (!finite(worldXZ)) return invalid<OceanWakeSample>();
199 const auto& s = impl_->settings;
200 const auto uv = ((worldXZ - s.center) / s.length + 0.5) * double(s.resolution);
201 OceanWakeSample out;
202 if (uv.x < 0.0 || uv.y < 0.0 || uv.x >= s.resolution - 1 || uv.y >= s.resolution - 1)
203 return Result<OceanWakeSample>::success(out);
204 const auto x = int(uv.x), z = int(uv.y);
205 const double fx = uv.x - x, fz = uv.y - z;
206 const auto wx = ocean_detail::weights(fx), wz = ocean_detail::weights(fz);
207 const auto dx = ocean_detail::derivatives(fx), dz = ocean_detail::derivatives(fz);
208 for (int j = 0; j < 4; ++j) {
209 for (int i = 0; i < 4; ++i) {
210 const int ix = x + i - 1, iz = z + j - 1;
211 if (ix < 0 || iz < 0 || ix >= int(s.resolution) || iz >= int(s.resolution)) continue;
212 const auto index = unsigned(iz) * s.resolution + unsigned(ix);
213 out.height += impl_->height[index] * wx[i] * wz[j];
214 out.velocity += impl_->velocity[index] * wx[i] * wz[j];
215 out.foam += impl_->foam[index] * wx[i] * wz[j];
216 out.gradient += impl_->height[index] * glm::dvec2(dx[i] * wz[j], wx[i] * dz[j]) * (s.resolution / s.length);
217 }
218 }
219 const double cell = s.length / s.resolution;
220 const auto edgeBlend = [&](double coordinate) {
221 const double far = double(s.resolution - 1) - coordinate;
222 const double distance = std::min(coordinate, far) * cell;
223 const double t = std::clamp(distance / s.absorptionWidth, 0.0, 1.0);
224 return glm::dvec2(t * t * (3.0 - 2.0 * t),
225 (coordinate < far ? 1.0 : -1.0) * 6.0 * t * (1.0 - t) / s.absorptionWidth);
226 };
227 const auto bx = edgeBlend(uv.x), bz = edgeBlend(uv.y);
228 const double blend = bx.x * bz.x;
229 out.gradient = out.gradient * blend + out.height * glm::dvec2(bx.y * bz.x, bx.x * bz.y);
230 out.height *= blend;
231 out.velocity *= blend;
232 out.foam = std::clamp(out.foam * blend, 0.0, 1.0);
233 return Result<OceanWakeSample>::success(out);
234}
