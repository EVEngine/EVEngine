13std::uint32_t resolveRadialSegments(const RadialTessellation& tessellation, float projectedRadiusPixels) {
14 const bool validError = std::isfinite(tessellation.maxScreenErrorPixels) && tessellation.maxScreenErrorPixels > 0.f;
15 EV_PARAM_CHECK(validError, "radial tessellation screen error must be finite and positive");
16 if (tessellation.customSegments != 0) {
17 const bool validCustom = tessellation.customSegments >= 3 && tessellation.customSegments <= 4096;
18 EV_PARAM_CHECK(validCustom, "custom radial segment count must be in [3, 4096]");
19 return tessellation.customSegments;
20 }
21 switch (tessellation.quality) {
22 case PrimitiveQuality::Low: return 16;
23 case PrimitiveQuality::Medium: return 32;
24 case PrimitiveQuality::High: return 64;
25 case PrimitiveQuality::Adaptive: break;
26 }
27 const bool validRadius = std::isfinite(projectedRadiusPixels) && projectedRadiusPixels >= 0.f;
28 EV_PARAM_CHECK(validRadius, "adaptive radial projected radius must be finite and non-negative");
29 if (projectedRadiusPixels <= tessellation.maxScreenErrorPixels) return 8;
30 const float cosine = glm::clamp(1.f - tessellation.maxScreenErrorPixels / projectedRadiusPixels, -1.f, 1.f);
31 const float halfAngle = std::acos(cosine);
32 if (halfAngle <= 1e-6f) return 256;
33 return std::clamp(static_cast<std::uint32_t>(std::ceil(glm::pi<float>() / halfAngle)), 8u, 256u);
34}
