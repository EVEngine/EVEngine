72void OS::limitFrame() {
73 const uint64_t now = SDL_GetPerformanceCounter();
74 const uint64_t frequency = SDL_GetPerformanceFrequency();
75 if (lastFrameCounter_ == 0 || frequency == 0) {
76 lastFrameCounter_ = now;
77 return;
78 }
79 if (verticalSyncCount_ == 0 && targetFramesPerSecond_ > 0) {
80 const double target = static_cast<double>(frequency) / static_cast<double>(targetFramesPerSecond_);
81 const double elapsed = static_cast<double>(now - lastFrameCounter_);
82 if (elapsed < target) {
83 const uint32_t milliseconds =
84 static_cast<uint32_t>((target - elapsed) * 1000.0 / static_cast<double>(frequency));
85 if (milliseconds > 0) SDL_Delay(milliseconds);
86 }
87 }
88 lastFrameCounter_ = SDL_GetPerformanceCounter();
89}

129int OS::getProcessMemoryMB() const {
130#if defined(EVENGINE_WINDOWS) || defined(_WIN32)
131 PROCESS_MEMORY_COUNTERS pmc{};
132 if (GetProcessMemoryInfo(GetCurrentProcess(), &pmc, sizeof(pmc)))
133 return static_cast<int>(pmc.WorkingSetSize / (1024ull * 1024ull));
134 return 0;
135#elif defined(__APPLE__)
136 task_vm_info_data_t info{};
137 mach_msg_type_number_t count = TASK_VM_INFO_COUNT;
138 if (task_info(mach_task_self(), TASK_VM_INFO, reinterpret_cast<task_info_t>(&info), &count) ==
139 KERN_SUCCESS) {
140 return static_cast<int>(info.phys_footprint / (1024ull * 1024ull));
141 }
142 return 0;
143#elif defined(__linux__) || defined(EVENGINE_ANDROID) || defined(EVENGINE_LINUX)
144 std::ifstream in("/proc/self/status");
145 if (!in) return 0;
146 std::string key;
147 while (in >> key) {
148 if (key == "VmRSS:") {
149 long kb = 0;
150 in >> kb;
151 return static_cast<int>(kb / 1024);
152 }
153 std::string rest;
154 std::getline(in, rest);
155 }
156 return 0;
157#else
158 return 0;
159#endif
160}
