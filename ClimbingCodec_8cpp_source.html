202 return eve::Result<const eve::Value::Object*>::success(object);

294eve::Result<void> validateClimbingActionDefinition(const ClimbingActionDefinition& action) {
295 if (action.id.empty())
296 return failure<void>(eve::DiagnosticCode::InvalidArgument, "action id must not be empty", "id");
297 if (!finite(action.minHeight) || !finite(action.maxHeight) || action.minHeight < 0.f ||
298 action.maxHeight < action.minHeight)
299 return failure<void>(eve::DiagnosticCode::InvalidArgument,
300 "action height range must be finite, non-negative, and ordered", "height");
301 if (!finite(action.minSpeed) || action.minSpeed < 0.f || !finite(action.landingForward) ||
302 action.landingForward < 0.f || !finite(action.apexHeight) || action.apexHeight < 0.f)
303 return failure<void>(eve::DiagnosticCode::InvalidArgument,
304 "action speed, landing distance, and apex must be finite and non-negative");
305 if (action.duration.nanoseconds() <= 0)
306 return failure<void>(eve::DiagnosticCode::InvalidArgument, "action duration must be positive", "durationNs");
307 if (!finite(action.hangBodyOffset) || action.hangBodyOffset < 0.f || !finite(action.hangFeetBelowLedge) ||
308 action.hangFeetBelowLedge <= 0.f || !finite(action.handSpacing) || action.handSpacing <= 0.f ||
309 !finite(action.cancelWindowStart) || !finite(action.cancelWindowEnd) || action.cancelWindowStart < 0.f ||
310 action.cancelWindowEnd > 1.f || action.cancelWindowEnd < action.cancelWindowStart)
311 return failure<void>(eve::DiagnosticCode::InvalidArgument,
312 "hang geometry and cancel window must be finite and ordered");
313 if (!finite(action.rootMotionScaleMin) || !finite(action.rootMotionScaleMax) || action.rootMotionScaleMin <= 0.f ||
314 action.rootMotionScaleMax < action.rootMotionScaleMin || action.rootMotionScaleMax > 4.f ||
315 !finite(action.maxTranslationWarpPerTick) || action.maxTranslationWarpPerTick <= 0.f ||
316 !finite(action.maxYawWarpRadiansPerTick) || action.maxYawWarpRadiansPerTick <= 0.f ||
317 !finite(action.horizontalWarpBudget) || action.horizontalWarpBudget < 0.f ||
318 !finite(action.verticalWarpBudget) || action.verticalWarpBudget < 0.f ||
319 !finite(action.facingWarpBudgetRadians) || action.facingWarpBudgetRadians < 0.f)
320 return failure<void>(eve::DiagnosticCode::InvalidArgument,
321 "root-motion scale and warp budgets must be finite, positive, and bounded", "warp");
322 float previousEnd = 0.f;
323 for (std::size_t index = 0; index < action.warpWindows.size(); ++index) {
324 const ClimbingWarpWindow& window = action.warpWindows[index];
325 if (!finite(window.start) || !finite(window.end) || window.start < 0.f || window.end > 1.f ||
326 window.end <= window.start || window.start < previousEnd ||
327 (!window.horizontal && !window.vertical && !window.facing))
328 return failure<void>(eve::DiagnosticCode::InvalidArgument,
329 "warp windows must be ordered, non-overlapping, and enable a channel",
330 "warpWindows." + std::to_string(index));
331 previousEnd = window.end;
332 }
333 std::vector<std::string> notifyNames = action.requiredNotifies;
334 std::sort(notifyNames.begin(), notifyNames.end());
335 if (std::any_of(notifyNames.begin(), notifyNames.end(), [](const std::string& name) { return name.empty(); }) ||
336 std::adjacent_find(notifyNames.begin(), notifyNames.end()) != notifyNames.end())
337 return failure<void>(eve::DiagnosticCode::InvalidArgument,
338 "required animation notifies must be non-empty and unique", "requiredNotifies");
339 const auto sourceBits = static_cast<std::uint8_t>(action.sourceModes);
340 if (sourceBits == 0 || (sourceBits & ~static_cast<std::uint8_t>(ClimbingSourceMode::Any)) != 0)
341 return failure<void>(eve::DiagnosticCode::InvalidArgument, "sourceModes must contain a known mode",
342 "sourceModes");
343 if (!finite(action.minDepth) || !finite(action.maxDepth) || action.minDepth < 0.f ||
344 action.maxDepth < action.minDepth || !finite(action.minDistance) || !finite(action.maxDistance) ||
345 action.minDistance < 0.f || action.maxDistance < action.minDistance ||
346 !finite(action.minSurfaceNormalY) || action.minSurfaceNormalY < -1.f || action.minSurfaceNormalY > 1.f ||
347 !finite(action.maxSlopeRadians) || action.maxSlopeRadians < 0.f || action.maxSlopeRadians > 3.1415927f ||
348 !finite(action.maxCurvature) || action.maxCurvature < 0.f)
349 return failure<void>(eve::DiagnosticCode::InvalidArgument, "action geometry constraints are invalid",
350 "geometry");
351 if (!uniqueNonEmpty(action.tags) || !uniqueNonEmpty(action.requiredSupportTags) ||
352 !uniqueNonEmpty(action.comboTags) || !uniqueNonEmpty(action.requiredConditionTags) ||
353 !uniqueNonEmpty(action.eventMetadata))
354 return failure<void>(eve::DiagnosticCode::InvalidArgument, "action tags and event metadata must be unique",
355 "metadata");
356 if (action.animation.rootBone.empty())
357 return failure<void>(eve::DiagnosticCode::InvalidArgument, "animation rootBone must not be empty",
358 "animation.rootBone");
359 for (std::size_t index = 0; index < action.branchWindows.size(); ++index) {
360 const auto& window = action.branchWindows[index];
361 if (!finite(window.start) || !finite(window.end) || window.start < 0.f || window.end > 1.f ||
362 window.end <= window.start || window.comboTag.empty())
363 return failure<void>(eve::DiagnosticCode::InvalidArgument, "branch window is invalid",
364 "branchWindows." + std::to_string(index));
365 if (std::find(action.comboTags.begin(), action.comboTags.end(), window.comboTag) == action.comboTags.end())
366 return failure<void>(eve::DiagnosticCode::InvalidArgument,
367 "branch window comboTag must be declared by the action",
368 "branchWindows." + std::to_string(index) + ".comboTag");
369 }
370 for (std::size_t index = 0; index < action.contactConstraints.size(); ++index) {
371 const auto& contact = action.contactConstraints[index];
372 if (!finite(contact.start) || !finite(contact.end) || !finite(contact.maxWeight) || contact.start < 0.f ||
373 contact.end > 1.f || contact.end <= contact.start || contact.maxWeight < 0.f || contact.maxWeight > 1.f)
374 return failure<void>(eve::DiagnosticCode::InvalidArgument, "contact constraint is invalid",
375 "contactConstraints." + std::to_string(index));
376 }
377 if (action.repetitionPenalty < 0 || !finite(action.staminaCost) || action.staminaCost < 0.f)
378 return failure<void>(eve::DiagnosticCode::InvalidArgument, "action cost metadata is invalid", "cost");
379 return eve::Result<void>::success();
380}
