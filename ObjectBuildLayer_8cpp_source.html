35std::uint64_t mix(std::uint64_t value) {
36 value += 0x9e3779b97f4a7c15ULL;
37 value = (value ^ (value >> 30U)) * 0xbf58476d1ce4e5b9ULL;
38 value = (value ^ (value >> 27U)) * 0x94d049bb133111ebULL;
39 return value ^ (value >> 31U);

42struct RandomStream {
43 std::uint64_t state;
44 std::uint64_t next() {
45 state = mix(state);
46 return state;
47 }
48 float unit() { return float(next() >> 40U) * (1.f / 16777216.f); }
49 float between(float minimum, float maximum) { return minimum + (maximum - minimum) * unit(); }
50};

52std::uint64_t pointKey(const ProcgenPoint& point, std::size_t index, std::uint32_t seed) {
53 if (point.id != 0) return mix(point.id ^ (std::uint64_t(seed) << 32U));
54 const auto qx = std::int64_t(std::llround(double(point.x) * 1000.0));
55 const auto qy = std::int64_t(std::llround(double(point.y) * 1000.0));
56 const auto qz = std::int64_t(std::llround(double(point.z) * 1000.0));
57 return mix(std::uint64_t(qx) ^ (mix(std::uint64_t(qy)) << 1U) ^ (mix(std::uint64_t(qz)) << 2U) ^
58 std::uint64_t(index) ^ (std::uint64_t(seed) << 32U));
59}

61float orientedYaw(const ProcgenPoint& point, const PointSet& orientation, float cellSize, bool invert) {
62 struct Direction {
63 float x;
64 float z;
65 float yaw;
66 };
67 const Direction cardinal[] = {
68 {0.f, 1.f, invert ? 90.f : 270.f}, {1.f, 0.f, 0.f}, {0.f, -1.f, invert ? 270.f : 90.f}, {-1.f, 0.f, 180.f}};
69 const float tolerance = std::max(0.0001f, cellSize * 0.01f);
70 const auto matches = [&](float x, float z) {
71 return std::any_of(orientation.points().begin(), orientation.points().end(), [&](const ProcgenPoint& other) {
72 return std::abs(other.x - x) <= tolerance && std::abs(other.z - z) <= tolerance;
73 });
74 };
75 for (const auto& direction : cardinal)
76 if (matches(point.x + direction.x * cellSize, point.z + direction.z * cellSize)) return direction.yaw;
77 const Direction diagonal[] = {{1.f, 1.f, 90.f}, {1.f, -1.f, 0.f}, {-1.f, -1.f, 180.f}, {-1.f, 1.f, 180.f}};
78 for (const auto& direction : diagonal)
79 if (matches(point.x + direction.x * cellSize, point.z + direction.z * cellSize)) return direction.yaw;
80 return 0.f;
81}

83void applyRandomTransform(ProcgenPoint& point, const ObjectTransformRange& range, RandomStream& random) {
84 point.pitch += random.between(range.minPitch, range.maxPitch);
85 point.yaw += random.between(range.minYaw, range.maxYaw);
86 point.roll += random.between(range.minRoll, range.maxRoll);
87 if (range.uniformScale) {
88 const float scale = random.between(range.minScaleX, range.maxScaleX);
89 point.scaleX *= scale;
90 point.scaleY *= scale;
91 point.scaleZ *= scale;
92 } else {
93 point.scaleX *= random.between(range.minScaleX, range.maxScaleX);
94 point.scaleY *= random.between(range.minScaleY, range.maxScaleY);
95 point.scaleZ *= random.between(range.minScaleZ, range.maxScaleZ);
96 }
97}

218Result<PointSet> ObjectBuildLayer::build(const PointSet& source, const PointSet* orientation) const {
219 if (assets_.empty()) return fail<PointSet>("object build layer has no assets", "assets");
220 if (orient_ && !orientation) return fail<PointSet>("orientation is enabled but no orientation layer was supplied");
221 float totalWeight = 0.f;
222 for (const auto& asset : assets_) totalWeight += asset.weight;
223 if (!std::isfinite(totalWeight) || totalWeight <= 0.f) return fail<PointSet>("total asset weight is invalid");
224
225 std::size_t childCount = 0;
226 for (const auto& child : children_) childCount += std::size_t(child.count);
227 PointSet output;
228 output.reserve(source.points().size() * (childCount + 1));
229 for (std::size_t sourceIndex = 0; sourceIndex < source.points().size(); ++sourceIndex) {
230 const auto& input = source.points()[sourceIndex];
231 RandomStream random{pointKey(input, sourceIndex, seed_)};
232 float selection = random.unit() * totalWeight;
233 const Asset* selected = &assets_.back();
234 for (const auto& asset : assets_) {
235 selection -= asset.weight;
236 if (selection < 0.f) {
237 selected = &asset;
238 break;
239 }
240 }
241
242 auto appended = output.appendPointFrom(source, sourceIndex);
243 if (!appended.ok()) return Result<PointSet>::failure(appended.status());
244 const int parentIndex = appended.value();
245 auto& parent = output.mutablePoint(std::size_t(parentIndex));
246 if (positionRadius_ > 0.f) {
247 const float angle = random.unit() * 6.283185307179586f;
248 const float radius = std::sqrt(random.unit()) * positionRadius_;
249 parent.x += std::cos(angle) * radius;
250 parent.z += std::sin(angle) * radius;
251 }
252 parent.x += offsetX_;
253 parent.y += offsetY_;
254 parent.z += offsetZ_;
255 if (placeOnTop_) {
256 const char* attribute = useLowest_ ? "surface_lowest_y" : "surface_highest_y";
257 if (source.hasFloatAttribute(int(sourceIndex), attribute))
258 parent.y = source.getFloatAttribute(int(sourceIndex), attribute, parent.y) + topOffset_ + offsetY_;
259 }
260 applyRandomTransform(parent, transform_, random);
261 parent.scaleX *= scaleX_;
262 parent.scaleY *= scaleY_;
263 parent.scaleZ *= scaleZ_;
264 if (orient_)
265 parent.yaw = orientedYaw(input, *orientation, orientationCellSize_, invertOrientation_) +
266 orientationYawOffset_ + random.between(transform_.minYaw, transform_.maxYaw);
267 auto tagged = tagPoint(output, parentIndex, selected->id, "parent", -1);
268 if (!tagged.ok()) return Result<PointSet>::failure(tagged.status());
269
270 for (std::size_t ruleIndex = 0; ruleIndex < children_.size(); ++ruleIndex) {
271 const auto& rule = children_[ruleIndex];
272 RandomStream childRandom{mix(pointKey(input, sourceIndex, seed_) ^ (std::uint64_t(ruleIndex + 1) << 32U))};
273 for (int childOrdinal = 0; childOrdinal < rule.count; ++childOrdinal) {
274 ProcgenPoint child = parent;
275 child.id = derivePointId(parent.id != 0 ? parent.id : pointKey(input, sourceIndex, seed_),
276 std::uint64_t(ruleIndex) * 1000000ULL + std::uint64_t(childOrdinal + 1));
277 const float angle = childRandom.unit() * 6.283185307179586f;
278 const float radius = std::sqrt(childRandom.unit()) * rule.radius;
279 child.x += std::cos(angle) * radius;
280 child.z += std::sin(angle) * radius;
281 child.pitch = 0.f;
282 child.yaw = 0.f;
283 child.roll = 0.f;
284 child.scaleX = 1.f;
285 child.scaleY = 1.f;
286 child.scaleZ = 1.f;
287 applyRandomTransform(child, rule.transform, childRandom);
288 const int childIndex = output.appendPoint(child);
289 tagged = tagPoint(output, childIndex, rule.asset, "child", parentIndex);
290 if (!tagged.ok()) return Result<PointSet>::failure(tagged.status());
291 }
292 }
293 }
294 return Result<PointSet>::success(std::move(output));
295}

323Result<void> ObjectBuildLayer::deserializeDefinition(std::string_view definition) {
324 ObjectBuildLayer replacement;
325 std::istringstream input{std::string(definition)};
326 std::string magic;
327 int version = 0;
328 if (!(input >> magic >> version) || magic != "EVPCG_OBJECT_LAYER" || version != 1)
329 return fail<void>("invalid object layer header");
330 std::string line;
331 std::getline(input, line);
332 bool ended = false;
333 while (std::getline(input, line)) {
334 if (line.empty()) continue;
335 std::istringstream record(line);
336 std::string kind;
337 record >> kind;
338 if (kind == "END") {
339 ended = true;
340 break;
341 }
342 if (kind == "SEED") {
343 std::uint32_t seed = 0;
344 if (!(record >> seed)) return fail<void>("invalid SEED record");
345 replacement.setSeed(seed);
346 } else if (kind == "OFFSET") {
347 float x = 0.f, y = 0.f, z = 0.f;
348 if (!(record >> x >> y >> z) || !std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z))
349 return fail<void>("invalid OFFSET record");
350 replacement.setLayerOffset(x, y, z);
351 } else if (kind == "SCALE") {
352 float x = 0.f, y = 0.f, z = 0.f;
353 if (!(record >> x >> y >> z)) return fail<void>("invalid SCALE record");
354 auto applied = replacement.setLayerScale(x, y, z);
355 if (!applied.ok()) return applied;
356 } else if (kind == "POSITION_RADIUS") {
357 float radius = 0.f;
358 if (!(record >> radius)) return fail<void>("invalid POSITION_RADIUS record");
359 auto applied = replacement.setPositionRadius(radius);
360 if (!applied.ok()) return applied;
361 } else if (kind == "RANDOM") {
362 ObjectTransformRange range;
363 int uniform = 0;
364 if (!(record >> range.minPitch >> range.maxPitch >> range.minYaw >> range.maxYaw >> range.minRoll >>
365 range.maxRoll >> range.minScaleX >> range.maxScaleX >> range.minScaleY >> range.maxScaleY >>
366 range.minScaleZ >> range.maxScaleZ >> uniform) ||
367 (uniform != 0 && uniform != 1))
368 return fail<void>("invalid RANDOM record");
369 range.uniformScale = uniform != 0;
370 auto applied = replacement.setRandomTransform(range);
371 if (!applied.ok()) return applied;
372 } else if (kind == "ORIENTATION") {
373 int enabled = 0, invert = 0;
374 float cell = 0.f, offset = 0.f;
375 if (!(record >> enabled >> cell >> offset >> invert) || (enabled != 0 && enabled != 1) ||
376 (invert != 0 && invert != 1))
377 return fail<void>("invalid ORIENTATION record");
378 if (enabled) {
379 auto applied = replacement.setOrientation(cell, offset, invert != 0);
380 if (!applied.ok()) return applied;
381 }
382 } else if (kind == "PLACE") {
383 int enabled = 0, lowest = 0;
384 float offset = 0.f;
385 if (!(record >> enabled >> lowest >> offset) || (enabled != 0 && enabled != 1) ||
386 (lowest != 0 && lowest != 1) || !std::isfinite(offset))
387 return fail<void>("invalid PLACE record");
388 replacement.setPlaceOnTop(enabled != 0, lowest != 0, offset);
389 } else if (kind == "ASSET") {
390 std::string asset;
391 float weight = 0.f;
392 if (!(record >> std::quoted(asset) >> weight)) return fail<void>("invalid ASSET record");
393 auto applied = replacement.addAsset(std::move(asset), weight);
394 if (!applied.ok()) return applied;
395 } else if (kind == "CHILD") {
396 std::string asset;
397 int count = 0, uniform = 0;
398 float radius = 0.f;
399 ObjectTransformRange range;
400 if (!(record >> std::quoted(asset) >> count >> radius >> range.minPitch >> range.maxPitch >> range.minYaw >>
401 range.maxYaw >> range.minRoll >> range.maxRoll >> range.minScaleX >> range.maxScaleX >>
402 range.minScaleY >> range.maxScaleY >> range.minScaleZ >> range.maxScaleZ >> uniform) ||
403 (uniform != 0 && uniform != 1))
404 return fail<void>("invalid CHILD record");
405 range.uniformScale = uniform != 0;
406 auto applied = replacement.addChildRule(std::move(asset), count, radius, range);
407 if (!applied.ok()) return applied;
408 } else {
409 return fail<void>("unknown object layer record: " + kind);
410 }
411 record >> std::ws;
412 if (!record.eof()) return fail<void>("trailing object layer record data");
413 }
414 if (!ended) return fail<void>("object layer END record is missing");
415 *this = std::move(replacement);
416 return Result<void>::success();
417}
