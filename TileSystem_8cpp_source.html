62bool atlasUV(const TileLayer::Tileset::Atlas &atlas, uint32_t gid, float &u0, float &v0, float &u1, float &v1) {
63 if (!atlas.texture || atlas.columns <= 0 || atlas.tileW <= 0 || atlas.tileH <= 0) return false;
64 if (gid < uint32_t(atlas.firstGid)) return false;
65 const int local = int(gid) - atlas.firstGid;
66 const int col = local % atlas.columns;
67 const int row = local / atlas.columns;
68 const float iw = float(atlas.texture->getWidth());
69 const float ih = float(atlas.texture->getHeight());
70 if (iw <= 0.f || ih <= 0.f) return false;
71
72 const float px = float(atlas.margin + col * (atlas.tileW + atlas.spacing));
73 const float py = float(atlas.margin + row * (atlas.tileH + atlas.spacing));
74 u0 = px / iw;
75 v0 = py / ih;
76 u1 = (px + float(atlas.tileW)) / iw;
77 v1 = (py + float(atlas.tileH)) / ih;
78 return true;
79}

133uint32_t animatedGid(const TileLayer::Tileset &ts, uint32_t gid) {
134 for (const auto &animation : ts.animations) {
135 if (animation.gid != int(gid) || animation.frames.empty()) continue;

138 if (total <= 0) return gid;

148void applyTiledTransform(graphics::DrawItem2D &item, uint32_t raw) {
149 const bool horizontal = (raw & 0x80000000u) != 0;
150 const bool vertical = (raw & 0x40000000u) != 0;
151 const bool diagonal = (raw & 0x20000000u) != 0;
152 if (!diagonal) {
153 item.flipX = horizontal;
154 item.flipY = vertical;
155 return;
156 }
157 item.rotation = horizontal ? 90.f : 270.f;
158 item.flipX = horizontal == vertical;
159 item.flipY = false;

162bool chunkVisible(const TileLayer::Config &cfg, const TileLayer::Tileset &ts, const ViewCam &cam,
163 int viewWidth, int viewHeight, int x0, int y0, int x1, int y1) {
164 if (!cam.valid || viewWidth <= 0 || viewHeight <= 0) return true;
165 float minX = 1e30f, minY = 1e30f, maxX = -1e30f, maxY = -1e30f;
166 const int xs[] = {x0, std::max(x0, x1 - 1)};
167 const int ys[] = {y0, std::max(y0, y1 - 1)};
168 for (int y : ys) {
169 for (int x : xs) {
170 float wx = 0.f, wy = 0.f;
171 tileToWorld(cfg, x, y, wx, wy);
172 minX = std::min(minX, wx);
173 minY = std::min(minY, wy);
174 maxX = std::max(maxX, wx + cfg.tileW);
175 maxY = std::max(maxY, wy + cfg.tileH);
176 }
177 }
178 float expandX = cfg.tileW;
179 float expandY = cfg.tileH;
180 for (const auto &visual : ts.visuals) {
181 expandX = std::max(expandX, float(visual.width) + std::fabs(visual.pivotX));
182 expandY = std::max(expandY, float(visual.height) + std::fabs(visual.pivotY));
183 }
184 minX -= expandX;
185 minY -= expandY;
186 maxX += expandX;
187 maxY += expandY;
188 const float zoom = cam.zoom > 0.f ? cam.zoom : 0.0001f;
189 const float left = cam.x - float(viewWidth) * 0.5f / zoom;
190 const float right = cam.x + float(viewWidth) * 0.5f / zoom;
191 const float top = cam.y - float(viewHeight) * 0.5f / zoom;
192 const float bottom = cam.y + float(viewHeight) * 0.5f / zoom;
193 return maxX >= left && minX <= right && maxY >= top && minY <= bottom;
194}

200void TileRenderSystem::collect(std::vector<graphics::DrawItem2D> &out, int viewWidth,
201 int viewHeight) {
202 gLastVisibleTileCount = 0;
203 gLastCustomVisualCount = 0;
204 gLastAtlasCount = 0;
205 gLastVisitedChunkCount = 0;
206 gLastVisitedCellCount = 0;
207 if (ecs::current()->getManager<TileLayer>() == nullptr) return;
208
209 std::unordered_set<graphics::Texture *> atlases;
210
211 auto view = ecs::View<TileLayer, TileLayer::Config, TileLayer::Tiles, TileLayer::Tileset,
212 TileLayer::Draw>();
213 for (auto it = view.begin(); it != view.end(); ++it) {
214 auto [cfg, tiles, ts, draw] = *it;
215 if (!draw->visible || cfg->mapW <= 0 || cfg->mapH <= 0) continue;
216 if (tiles->gids.size() < size_t(cfg->mapW * cfg->mapH)) continue;
217
218 const Color &tint = draw->tint;
219 const ViewCam cam = fromEntity(draw->camera);
220
221 const int chunkColumns = std::max(1, tiles->chunkColumns);
222 const int chunkRows = std::max(1, tiles->chunkRows);
223 for (int cy = 0; cy < chunkRows; ++cy) {
224 for (int cx = 0; cx < chunkColumns; ++cx) {
225 const int chunkIndex = cy * chunkColumns + cx;
226 if (!tiles->chunks.empty() &&
227 (chunkIndex >= int(tiles->chunks.size()) ||
228 tiles->chunks[size_t(chunkIndex)].nonEmpty == 0))
229 continue;
230 const int x0 = cx * TileLayer::Tiles::kChunkSize;
231 const int y0 = cy * TileLayer::Tiles::kChunkSize;
232 const int x1 = std::min(cfg->mapW, x0 + TileLayer::Tiles::kChunkSize);
233 const int y1 = std::min(cfg->mapH, y0 + TileLayer::Tiles::kChunkSize);
234 if (!chunkVisible(*cfg, *ts, cam, viewWidth, viewHeight, x0, y0, x1, y1))
235 continue;
236 ++gLastVisitedChunkCount;
237 for (int ty = y0; ty < y1; ++ty) {
238 for (int tx = x0; tx < x1; ++tx) {
239 ++gLastVisitedCellCount;
240 const uint32_t raw = tiles->gids[size_t(ty * cfg->mapW + tx)];
241 const uint32_t gid = animatedGid(*ts, tileGid(raw));
242 if (gid == 0) continue;
243
244 graphics::DrawItem2D item;
245 tileToWorld(*cfg, tx, ty, item.x, item.y);
246 item.w = cfg->tileW;
247 item.h = cfg->tileH;
248 item.depthY = tileToDepthY(*cfg, tx, ty);
249 item.layer = draw->layer;
250 item.canvas = draw->canvas;
251 item.camera = draw->camera;
252 item.camValid = cam.valid;
253 item.camX = cam.x;
254 item.camY = cam.y;
255 item.camZoom = cam.zoom;
256 item.receiveLight = false;
257 item.litPath = false;
258 applyTiledTransform(item, raw);
259
260 float u0, v0, u1, v1;
261 const auto* visual = visualForGid(*ts, gid);
262 const auto *atlas = atlasForGid(*ts, gid);
263 const auto *subtiles = visual ? subtileFrame(*visual) : nullptr;
264 if (subtiles && atlas) {
265 for (const auto &part : subtiles->parts) {
266 graphics::DrawItem2D quarter = item;
267 if (!subtileUV(*atlas, part, u0, v0, u1, v1)) continue;
268 float centerX = part.offsetX + float(part.width) * 0.5f - cfg->tileW * 0.5f;
269 float centerY = part.offsetY + float(part.height) * 0.5f - cfg->tileH * 0.5f;
270 if (raw & 0x20000000u) std::swap(centerX, centerY);
271 if (raw & 0x80000000u) centerX = -centerX;
272 if (raw & 0x40000000u) centerY = -centerY;
273 quarter.x += cfg->tileW * 0.5f + centerX - float(part.width) * 0.5f;
274 quarter.y += cfg->tileH * 0.5f + centerY - float(part.height) * 0.5f;
275 quarter.w = float(part.width);
276 quarter.h = float(part.height);
277 quarter.texture = atlas->texture;
278 quarter.hasUV = true;
279 quarter.u0 = u0;
280 quarter.v0 = v0;
281 quarter.u1 = u1;
282 quarter.v1 = v1;
283 quarter.color = tint;
284 out.push_back(quarter);
285 ++gLastCustomVisualCount;
286 }
287 ++gLastVisibleTileCount;
288 if (atlas->texture) atlases.insert(atlas->texture);
289 continue;
290 }
291 if (visual && atlas && visualUV(*atlas, *visual, u0, v0, u1, v1)) {
292 ++gLastCustomVisualCount;
293 item.x -= visual->pivotX;
294 item.y -= visual->pivotY;
295 item.w = float(visual->width);
296 item.h = float(visual->height);
297 item.depthY += visual->sortBias;
298 item.texture = atlas->texture;
299 item.hasUV = true;
300 item.u0 = u0;
301 item.v0 = v0;
302 item.u1 = u1;
303 item.v1 = v1;
304 item.color = tint;
305 } else if (atlas && atlasUV(*atlas, gid, u0, v0, u1, v1)) {
306 item.texture = atlas->texture;
307 item.hasUV = true;
308 item.u0 = u0;
309 item.v0 = v0;
310 item.u1 = u1;
311 item.v1 = v1;
312 item.color = tint;
313 } else {
314 item.texture = nullptr;
315 item.color = solidForGid(gid, tint);
316 }
317 out.push_back(item);
318 ++gLastVisibleTileCount;
319 if (item.texture) atlases.insert(item.texture);
320 }
321 }
322 }
323 }
324 }
325 gLastAtlasCount = int(atlases.size());
326}

346int TileConfigSystem::poll() {
347 if (ecs::current()->getManager<TileLayer>() == nullptr) return 0;
348
349 int reloaded = 0;
350 auto view = ecs::View<TileLayer, TileLayer::Config, TileLayer::Resource>();
351 for (auto it = view.begin(); it != view.end(); ++it) {
352 auto [cfg, res] = *it;
353 if (!res->autoReload || res->path.empty() || !cfg->entity) continue;
354
355 const int64_t mt = fileModtime(res->path);
356 bool changed = mt >= 0 && mt != res->modtime;
357 for (size_t index = 0; !changed && index < res->dependencyPaths.size(); ++index) {
358 const int64_t dependencyModtime = fileModtime(res->dependencyPaths[index]);
359 const int64_t known = index < res->dependencyModtimes.size() ? res->dependencyModtimes[index] : -1;
360 changed = dependencyModtime >= 0 && dependencyModtime != known;
361 }
362 if (!changed) continue;
363 if (reloadConfigFile(cfg->entity, nullptr)) ++reloaded;
364 }
365 return reloaded;
366}
