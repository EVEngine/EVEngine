40Result<BuildingEdgeCurveDragPreview> BuildingEdgeCurveDragSession::beginDrag(double rayOriginX, double rayOriginY,
41 double rayOriginZ, double rayDirectionX,
42 double rayDirectionY,
43 double rayDirectionZ) {
44 cancelDrag();
45 if (!target_ || memberInstanceId_ <= 0 || controls_.size() != 4)
46 return eve::editing::failed<BuildingEdgeCurveDragPreview>(
47 EditorStatus::Rejected, RuleId("editor.building.curve-drag-state-invalid"),
48 "Curve drag requires an existing group member and four controls");
49 const double directionLength =
50 std::sqrt(rayDirectionX * rayDirectionX + rayDirectionY * rayDirectionY + rayDirectionZ * rayDirectionZ);
51 if (!std::isfinite(directionLength) || directionLength < 1e-9)
52 return eve::editing::failed<BuildingEdgeCurveDragPreview>(EditorStatus::Rejected,
53 RuleId("editor.building.curve-drag-ray-invalid"),
54 "Curve drag requires a finite non-zero pointer ray");
55 const std::array<double, 3> origin{rayOriginX, rayOriginY, rayOriginZ};
56 const std::array<double, 3> direction{rayDirectionX / directionLength, rayDirectionY / directionLength,
57 rayDirectionZ / directionLength};
58 const editing::GizmoSnapshot gizmo =
59 target_->edgeCubicBezierGizmo(buildingId_, controls_, subdivisions_, 0, memberInstanceId_, surfaceName_);
60 double closest = std::numeric_limits<double>::infinity();
61 for (const editing::GizmoPrimitive& primitive : gizmo.primitives) {
62 if (primitive.kind != "sphere" || primitive.id.rfind("curve.control.", 0) != 0) continue;
63 const std::array<double, 3> offset{origin[0] - primitive.position[0], origin[1] - primitive.position[1],
64 origin[2] - primitive.position[2]};
65 const double b = offset[0] * direction[0] + offset[1] * direction[1] + offset[2] * direction[2];
66 const double c =
67 offset[0] * offset[0] + offset[1] * offset[1] + offset[2] * offset[2] - primitive.radius * primitive.radius;
68 const double discriminant = b * b - c;
69 if (discriminant < 0.0) continue;
70 const double nearHit = -b - std::sqrt(discriminant);
71 const double farHit = -b + std::sqrt(discriminant);
72 const double hit = nearHit >= 0.0 ? nearHit : farHit;
73 if (hit < 0.0 || hit >= closest) continue;
74 closest = hit;
75 activeControlPointIndex_ = std::stoi(primitive.id.substr(14));
76 dragPlanePoint_ = primitive.position;
77 }
78 if (activeControlPointIndex_ < 0)
79 return eve::editing::failed<BuildingEdgeCurveDragPreview>(EditorStatus::NotFound,
80 RuleId("editor.building.curve-control-not-hit"),
81 "Pointer ray did not hit a curve control handle");
82 dragPlaneNormal_ = direction;
83 baseRevision_ = target_->revision();
84 dragging_ = true;
85 lastPreviewValid_ = gizmo.status == EditorStatus::Applied;
86 return previewCurrent();
87}

89Result<BuildingEdgeCurveDragPreview> BuildingEdgeCurveDragSession::updateDrag(double rayOriginX, double rayOriginY,
90 double rayOriginZ, double rayDirectionX,
91 double rayDirectionY,
92 double rayDirectionZ) {
93 if (!dragging_ || !target_)
94 return eve::editing::failed<BuildingEdgeCurveDragPreview>(
95 EditorStatus::Rejected, RuleId("editor.building.curve-drag-not-active"), "Curve drag has not begun");
96 if (target_->revision() != baseRevision_) {
97 cancelDrag();
98 return eve::editing::failed<BuildingEdgeCurveDragPreview>(
99 EditorStatus::Conflict, RuleId("editor.building.curve-drag-stale"),
100 "Curve world changed while the control handle was being dragged");
101 }
102 const double length =
103 std::sqrt(rayDirectionX * rayDirectionX + rayDirectionY * rayDirectionY + rayDirectionZ * rayDirectionZ);
104 if (!std::isfinite(length) || length < 1e-9)
105 return eve::editing::failed<BuildingEdgeCurveDragPreview>(EditorStatus::Rejected,
106 RuleId("editor.building.curve-drag-ray-invalid"),
107 "Curve drag requires a finite non-zero pointer ray");
108 const std::array<double, 3> origin{rayOriginX, rayOriginY, rayOriginZ};
109 const std::array<double, 3> direction{rayDirectionX / length, rayDirectionY / length, rayDirectionZ / length};
110 const double denominator =
111 direction[0] * dragPlaneNormal_[0] + direction[1] * dragPlaneNormal_[1] + direction[2] * dragPlaneNormal_[2];
112 if (std::abs(denominator) < 1e-9)
113 return eve::editing::failed<BuildingEdgeCurveDragPreview>(EditorStatus::Rejected,
114 RuleId("editor.building.curve-drag-ray-parallel"),
115 "Pointer ray is parallel to the curve drag plane");
116 const double distance = ((dragPlanePoint_[0] - origin[0]) * dragPlaneNormal_[0] +
117 (dragPlanePoint_[1] - origin[1]) * dragPlaneNormal_[1] +
118 (dragPlanePoint_[2] - origin[2]) * dragPlaneNormal_[2]) /
119 denominator;
120 if (distance < 0.0)
121 return eve::editing::failed<BuildingEdgeCurveDragPreview>(EditorStatus::Rejected,
122 RuleId("editor.building.curve-drag-behind-camera"),
123 "Curve drag plane is behind the pointer ray origin");
124 auto logical = target_->curveLogicalPointFromWorld(
125 origin[0] + direction[0] * distance, origin[1] + direction[1] * distance, origin[2] + direction[2] * distance);
126 if (!logical.ok())
127 return eve::editing::failed<BuildingEdgeCurveDragPreview>(
128 logical.code(), RuleId("editor.building.curve-drag-projection-failed"),
129 "Pointer ray could not be projected to logical curve coordinates");
130 controls_[static_cast<size_t>(activeControlPointIndex_)] = logical.value();
131 auto preview = previewCurrent();
132 if (preview.ok()) lastPreviewValid_ = preview.value().status == EditorStatus::Applied;
133 return preview;
134}
