75 // frame-arena vertices. All other canvases retain shared-image semantics.

84 canvas->colorImage().setLayout(cb, vk::ImageLayout::eColorAttachmentOptimal);

88 rpBegin.renderPass = canvas->isHDR() ? hdrOffscreenRenderPass : offscreenRenderPass;
89 rpBegin.framebuffer = canvas->framebuffer();
90 rpBegin.renderArea.extent = vk::Extent2D{uint32_t(canvas->getWidth()), uint32_t(canvas->getHeight())};
91 rpBegin.clearValueCount = 1;
92 rpBegin.pClearValues = &cv;

94 cb.beginRenderPass(rpBegin, vk::SubpassContents::eInline);
95 if (needClear) {
96 vk::ClearAttachment attachment{};
97 attachment.aspectMask = vk::ImageAspectFlagBits::eColor;
98 attachment.colorAttachment = 0;
99 attachment.clearValue = cv;
100 vk::ClearRect rect{};
101 rect.rect = rpBegin.renderArea;
102 rect.layerCount = 1;
103 cb.clearAttachments(attachment, rect);
104 }

108 std::vector<vkb::HostVertexBuffer>& solidBufs = offscreenBuffers.solidBufs;

114 auto offscreenTexPipe = [&](BlendMode mode) -> vk::Pipeline {

139 auto drawOffscreenTextured = [&](TexturedBatch& tb) {
140 if (tb.batch.empty() || !tb.texture || !tb.texture->gpuHandle) return;
141 auto* gpu = static_cast<GpuTexture*>(tb.texture->gpuHandle);
142 vk::DescriptorSet texSet = gpu->descriptorSet;
143 if ((tb.depth && tb.depth->gpuHandle) || (tb.motion && tb.motion->gpuHandle) ||
144 (tb.extra && tb.extra->gpuHandle)) {
145 auto* depthGpu = tb.depth ? static_cast<GpuTexture*>(tb.depth->gpuHandle) : nullptr;
146 auto* motionGpu = tb.motion ? static_cast<GpuTexture*>(tb.motion->gpuHandle) : nullptr;
147 auto* extraGpu = tb.extra ? static_cast<GpuTexture*>(tb.extra->gpuHandle) : nullptr;
148 auto* specularGpu = tb.specular ? static_cast<GpuTexture*>(tb.specular->gpuHandle) : nullptr;

155 gpuVerts.reserve(ndc.vertices().size());

162 auto& arena = currentFrameArena();

164 vk::BufferUsageFlagBits::eVertexBuffer | vk::BufferUsageFlagBits::eStorageBuffer |

169 std::memcpy(allocation.mapped, gpuVerts.data(), size_t(allocation.size));
170 vertexBuffer = arena.buffer();
171 vertexOffset = allocation.offset;
172 } else {
173 if (texBufIndex >= texBufs.size()) texBufs.emplace_back();
174 vkb::HostVertexBuffer& vb = texBufs[texBufIndex++];
175 vb.allocate<TexturedVertex>(frameToken(), device, gpuVerts);
176 vertexBuffer = vb.buffer;
177 }

179 if (tb.shader && tb.shader->gpuHandle) {
180 if (canvas->isHDR())

182 else
183 ensureShaderOffscreenPipeline(tb.shader);
184 auto* gs = static_cast<GpuShader*>(tb.shader->gpuHandle);
185 vk::Pipeline alphaPipe = canvas->isHDR() ? gs->hdrOffscreenPipeline : gs->offscreenPipeline;
186 vk::Pipeline opaquePipe =
187 canvas->isHDR() ? gs->hdrOffscreenOpaquePipeline : gs->offscreenOpaquePipeline;
188 if (!alphaPipe) return;
189 vk::Pipeline customPipeline = tb.blend == BlendMode::Opaque && opaquePipe ? opaquePipe : alphaPipe;
190 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, customPipeline);
191 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, shaderPipelineLayout, 0, 1, &texSet, 0,
192 nullptr);
193 cb.pushConstants(shaderPipelineLayout,
194 vk::ShaderStageFlagBits::eVertex | vk::ShaderStageFlagBits::eFragment, 0,
195 Shader::kPushConstantBytes, tb.shader->pushConstantData());
196 } else {
197 vk::Pipeline pipe = offscreenTexPipe(tb.blend);
198 if (!pipe) return;
199 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, pipe);
200 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, texPipelineLayout, 0, 1, &texSet, 0, nullptr);
201 }
202 cb.bindVertexBuffers(0, 1, &vertexBuffer, &vertexOffset);
203 cb.draw(uint32_t(gpuVerts.size()), 1, 0, 0);
204 };
205
206 std::vector<bool> solidUploaded(solid.size(), false);
207 auto uploadSolid = [&](size_t idx) {
208 if (idx >= solid.size() || solidUploaded[idx] || solid[idx].batch.empty()) return;
209 vk::Pipeline pipe = offscreenSolidPipe(solid[idx].blend);
210 if (!pipe) return;
211 Batcher ndc = solid[idx].batch;
212 ndc.toNDC(vw, vh);
213 std::vector<ColorVertex> gpuVerts;
214 gpuVerts.reserve(ndc.vertices().size());
215 for (const auto& v : ndc.vertices()) gpuVerts.push_back(ColorVertex{v.pos, v.color});
216 if (solidBufs.size() <= idx) solidBufs.resize(idx + 1);
217 solidBufs[idx].allocate<ColorVertex>(frameToken(), device, gpuVerts);
218 solidUploaded[idx] = true;
219 };
220
221 auto drawSolidSpan = [&](uint32_t batchIndex, uint32_t begin, uint32_t count) {
222 if (batchIndex >= solid.size() || count == 0 || solid[batchIndex].batch.empty()) return;
223 vk::Pipeline pipe = offscreenSolidPipe(solid[batchIndex].blend);
224 if (!pipe) return;
225 uploadSolid(batchIndex);
226 vk::DeviceSize offset = 0;
227 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, pipe);
228 cb.bindVertexBuffers(0, 1, solidBufs[batchIndex], &offset);
229 cb.draw(count, 1, begin, 0);
230 };
231
232 const bool litOk = canvas->isHDR() ? bool(hdrOffscreenLitPipeline) : bool(offscreenLitPipeline);
233 if (!spans.empty()) {
234 for (const auto& sp : spans) {
235 if (sp.kind == OverlayKind::Solid)
236 drawSolidSpan(sp.index, sp.vertBegin, sp.vertCount);
237 else if (sp.kind == OverlayKind::Textured && sp.index < textured.size())
238 drawOffscreenTextured(textured[sp.index]);
239 else if (sp.kind == OverlayKind::Lit && litOk && sp.index < lit.size()) {
240 std::vector<LitBatch> one;
241 one.push_back(std::move(lit[sp.index]));
242 drawLitBatches(cb, vw, vh, one, texBufs, texBufIndex, true, canvas->isHDR());
243 }
244 }
245 } else {
246 for (size_t i = 0; i < solid.size(); ++i) {
247 if (!solid[i].batch.empty()) drawSolidSpan(uint32_t(i), 0, uint32_t(solid[i].batch.vertices().size()));
248 }
249 for (auto& tb : textured) drawOffscreenTextured(tb);
250 if (litOk) drawLitBatches(cb, vw, vh, lit, texBufs, texBufIndex, true, canvas->isHDR());
251 }
252
253 cb.endRenderPass();
254 canvas->colorImage().endSampledLayout();
255 };
256 if (canRecordPostSceneGpuWork()) {
257 recordOffscreen(currentPresentCb());
258 } else {
259 EV_PROFILE_MODULE("graphics", "Offscreen.SubmitAndWait");
260 vkb::executeImmediately(device.instance, uploadPool, device.getQueue(vkb::QueueType::graphics),
261 recordOffscreen);
262 }

277vk::Pipeline createPbrPipeline(vkb::Device& device, vk::PipelineLayout layout, const vkb::BuiltRenderPass& pass,
278 vk::SampleCountFlagBits samples, BlendMode blend, bool depthWrite, PbrCullMode cullMode,
279 bool alphaToCoverage) {
280 ShaderModulePair shaders(device, embeddedSpirv(pbr_surface_vert_spv), embeddedSpirv(pbr_surface_frag_spv));
281 auto input = vkb::VertexInputStateBuilder();
282 input.addInputBinding<MeshVertex>();
283 // The dedicated PBR shader reserves locations 5..15 for its eleven material
284 // UV streams. MeshVertex gained a color attribute at location 5 for the
285 // regular mesh pipelines, so describe only the five attributes consumed by
286 // pbr_surface.vert here instead of importing the shared six-attribute list.
287 input.input_attributes.emplace_back(0, 0, vk::Format::eR32G32B32Sfloat, uint32_t(offsetof(MeshVertex, pos)));
288 input.input_attributes.emplace_back(1, 0, vk::Format::eR32G32B32Sfloat, uint32_t(offsetof(MeshVertex, normal)));
289 input.input_attributes.emplace_back(2, 0, vk::Format::eR32G32Sfloat, uint32_t(offsetof(MeshVertex, uv)));
290 input.input_attributes.emplace_back(3, 0, vk::Format::eR16G16B16A16Uint, uint32_t(offsetof(MeshVertex, joints)));
291 input.input_attributes.emplace_back(4, 0, vk::Format::eR32G32B32A32Sfloat, uint32_t(offsetof(MeshVertex, weights)));
292 // Five mesh attributes plus eleven material UV streams fit Vulkan's minimum
293 // sixteen vertex attributes. Each role can select any canonical mesh UV set.
294 for (uint32_t i = 0; i < 11; ++i) {
295 input.input_bindings.emplace_back(i + 1, uint32_t(2 * sizeof(float)), vk::VertexInputRate::eVertex);
296 input.input_attributes.emplace_back(i + 5, i + 1, vk::Format::eR32G32Sfloat, 0);
297 }
298 auto attachment = makeBlendAttachment(blend);
299 attachment.blendEnable = blend != BlendMode::Opaque;
300 vk::PipelineColorBlendStateCreateInfo color{};
301 color.attachmentCount = 1;
302 color.pAttachments = &attachment;
303 vk::PipelineMultisampleStateCreateInfo multisampling{};
304 multisampling.rasterizationSamples = samples;
305 multisampling.alphaToCoverageEnable = alphaToCoverage && samples != vk::SampleCountFlagBits::e1;
306 return device.createPipeline()
307 .useClassicPipeline(shaders.vert, shaders.frag)
308 .setPipelineLayout(layout)
309 .setVertexInputState(input)
310 .setDynamicStatesViewportScissor()
311 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
312 cullMode == PbrCullMode::None ? vk::CullModeFlagBits::eNone
313 : (cullMode == PbrCullMode::Front ? vk::CullModeFlagBits::eFront
314 : vk::CullModeFlagBits::eBack),
315 vk::FrontFace::eCounterClockwise)
316 .setMultisampler(multisampling)
317 .setDepthStencil(true, depthWrite, vk::CompareOp::eLess)
318 .setColorBlending(color)
319 .build(pass);
320}

905size_t Graphics::uploadSkinPalette(Mesh* mesh, Mesh3dFrameSlots& fslots) {
906 const bool skinned = mesh && mesh->hasGpuSkinning() && mesh->getSkinPaletteCount() > 0;
907 const size_t slot = skinned ? fslots.drawIndex + 1 : 0;
908 const size_t bytes = skinned ? mesh->skinPalette().size() * sizeof(float) : sizeof(glm::mat4);
909 if (bytes > device.physical_device.properties.limits.maxStorageBufferRange)
910 throw Exception("Skin palette exceeds device maxStorageBufferRange");
911 if (fslots.palettes.size() <= slot) fslots.palettes.resize(slot + 1);
912 auto& buffer = fslots.palettes[slot];
913 if (!buffer.buffer || buffer.capacity < bytes) {
914 buffer.allocate(frameToken(), device, vk::BufferUsageFlagBits::eStorageBuffer, bytes, kHostVisibleCoherent);
915 // The fence protects earlier frames. Other slots used this frame retain
916 // their buffers and published descriptor sets unchanged.
917 fslots.sets.clear();
918 fslots.skinSets.clear();
919 }
920 const glm::mat4 identity(1.f);
921 updateRingLocal(buffer, 0, skinned ? mesh->skinPalette().data() : &identity[0][0], bytes);
922 fslots.activePalette = slot;
923 return slot;
924}

949bool Graphics::prepareSkinPass(Mesh* mesh, Texture* albedo, const glm::mat4& mvp, const glm::mat4& model,
950 const glm::vec4& clip, vk::DescriptorSet& set, uint32_t& uboOffset) {

954 // example, when script-side 2D clear work precedes render3D). In that

975 if (!texture || !texture->gpuHandle) return false;
