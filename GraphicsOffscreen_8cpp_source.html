99 vk::DescriptorSet texSet = gpu->descriptorSet;
100 if ((tb.depth && tb.depth->gpuHandle) ||
101 (tb.motion && tb.motion->gpuHandle) ||
102 (tb.extra && tb.extra->gpuHandle)) {

105 : nullptr;
106 auto *motionGpu = tb.motion

108 : nullptr;
109 auto *extraGpu = tb.extra

111 : nullptr;
112 auto *specularGpu =
113 tb.specular
114 ? static_cast<GpuTexture *>(tb.specular->gpuHandle)

122 ndc.toNDC(vw, vh);
123 std::vector<TexturedVertex> gpuVerts;
124 gpuVerts.reserve(ndc.vertices().size());

128 if (texBufIndex >= texBufs.size()) texBufs.emplace_back();
