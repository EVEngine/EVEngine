36 const eve::container::ContainerSnapshot& rhs) {
37 if (lhs.id != rhs.id || lhs.revision != rhs.revision || lhs.entries.size() != rhs.entries.size()) return false;

51class InventoryContainerAdapter::PreparedState final : public eve::container::IContainer::PreparedState {
52public:
53 PreparedState(InventoryContainerAdapter& owner, std::vector<ItemStack> staged, eve::Revision revision)
54 : owner_(owner), staged_(std::move(staged)), revision_(revision) {}
55
56 void commit() noexcept override {
57 if (committed_) return;
58 if (owner_.kind_ == InventoryContainerKind::Bag) {
59 if (owner_.bag_ == nullptr || owner_.bag_->slots().size() != staged_.size()) std::terminate();
60 owner_.bag_->slots().swap(staged_);
61 } else {
62 if (owner_.equipment_ == nullptr) std::terminate();
63 for (std::size_t index = 0; index < staged_.size(); ++index) {
64 auto* live = owner_.stackAt(eve::container::SlotIndex(static_cast<std::int32_t>(index)));
65 if (live == nullptr) std::terminate();
66 swapStacks(*live, staged_[index]);
67 }
68 }
69 owner_.revision_ = revision_;
70 committed_ = true;
71 }
72
73 void rollback() noexcept override {
74 if (!committed_) staged_.clear();
75 }
76
77private:
78 InventoryContainerAdapter& owner_;
79 std::vector<ItemStack> staged_;
80 eve::Revision revision_;
81 bool committed_ = false;
82};

144eve::Result<eve::container::ContainerSnapshot> InventoryContainerAdapter::snapshot() const {
145 if (!descriptor_.id.isValid() || (kind_ == InventoryContainerKind::Bag ? bag_ == nullptr : equipment_ == nullptr))
146 return eve::Result<eve::container::ContainerSnapshot>::failure(
147 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "inventory container adapter is not bound"));
148 eve::container::ContainerSnapshot snapshot{descriptor_.id, revision_, {}};
149 std::unordered_set<eve::container::MembershipId> ids;
150 const int count = kind_ == InventoryContainerKind::Bag ? bag_->getSlotCount() : equipment_->getSlotCount();
151 for (int index = 0; index < count; ++index) {
152 const auto slot = eve::container::SlotIndex(index);
153 const auto* stack = stackAt(slot);
154 if (stack == nullptr || stack->empty()) continue;
155 if (!descriptor_.capacity.isUnlimited() && static_cast<std::size_t>(index) >= descriptor_.capacity.value())
156 return eve::Result<eve::container::ContainerSnapshot>::failure(eve::Diagnostic::error(
157 eve::DiagnosticCode::InvariantViolation, "inventory membership exceeds configured capacity"));
158 if (stack->instanceId <= 0)
159 return eve::Result<eve::container::ContainerSnapshot>::failure(eve::Diagnostic::error(
160 eve::DiagnosticCode::InvariantViolation, "inventory stack has no instance identity"));
161 const auto id = objectId(*stack);
162 if (!ids.insert(id).second)
163 return eve::Result<eve::container::ContainerSnapshot>::failure(eve::Diagnostic::error(
164 eve::DiagnosticCode::Conflict, "inventory container contains duplicate instance identity"));
165 snapshot.entries.push_back({{id, slot, eve::Generation(1)}, describe(*stack, slotName(slot))});
166 }
167 if (!descriptor_.capacity.isUnlimited() && snapshot.entries.size() > descriptor_.capacity.value())
168 return eve::Result<eve::container::ContainerSnapshot>::failure(eve::Diagnostic::error(
169 eve::DiagnosticCode::InvariantViolation, "inventory container exceeds configured capacity"));
170 return eve::Result<eve::container::ContainerSnapshot>::success(std::move(snapshot));
171}

179eve::Result<void> InventoryContainerAdapter::validateInsert(
180 const eve::container::ContainerObject& object, std::optional<eve::container::SlotIndex> destination,
181 std::optional<eve::container::MembershipId> ignoredObject) const {
182 const auto* payload = dynamic_cast<const InventoryContainerObject*>(object.payload.get());
183 if (payload == nullptr) return error(eve::DiagnosticCode::StaleHandle, "inventory payload is stale");
184 auto validObject = validateObject(*payload);
185 if (!validObject) return validObject;
186 if (object.id != objectId(payload->stack))
187 return error(eve::DiagnosticCode::StaleHandle, "inventory membership identity is stale");
188 if (object.type != payload->stack.itemId || object.quantity != static_cast<std::uint32_t>(payload->stack.quantity))
189 return error(eve::DiagnosticCode::StaleHandle, "inventory object facts are stale");
190 const auto* definition = ItemRegistry::find(payload->stack.itemId);
191 if (definition == nullptr) return error(eve::DiagnosticCode::NotFound, "inventory item definition was not found");
192 if (definition->maxStack <= 0 || payload->stack.quantity > definition->maxStack)
193 return error(eve::DiagnosticCode::PreconditionViolation, "inventory stack exceeds item definition maxStack");
194
195 if (kind_ == InventoryContainerKind::Bag) {
196 if (bag_ == nullptr) return error(eve::DiagnosticCode::InvalidArgument, "bag adapter is unbound");
197 const std::size_t slotCount = static_cast<std::size_t>(bag_->getSlotCount());
198 const std::size_t slotLimit =
199 descriptor_.capacity.isUnlimited() ? slotCount : std::min(slotCount, descriptor_.capacity.value());
200 if (hasId(*bag_, object.id) && (!ignoredObject || *ignoredObject != object.id))
201 return error(eve::DiagnosticCode::Conflict, "bag already contains inventory object");
202 if (destination) {
203 if (!destination->isValid() || static_cast<std::size_t>(destination->value()) >= slotLimit)
204 return error(eve::DiagnosticCode::InvalidArgument, "bag destination slot is out of range");
205 if (!bag_->slots()[static_cast<std::size_t>(destination->value())].empty() &&
206 (!ignoredObject || *ignoredObject != object.id))
207 return error(eve::DiagnosticCode::Conflict, "bag destination slot is occupied");
208 } else {
209 const bool hasEmptySlot =
210 std::any_of(bag_->slots().begin(), bag_->slots().begin() + static_cast<std::ptrdiff_t>(slotLimit),
211 [](const ItemStack& stack) { return stack.empty(); });
212 if (!hasEmptySlot && (!ignoredObject || *ignoredObject != object.id))
213 return error(eve::DiagnosticCode::Conflict, "bag has no empty membership slot");
214 }
215 if (!ignoredObject || *ignoredObject != object.id) {
216 std::string reason;
217 if (!InventorySystem::canAdd(bag_, payload->stack.itemId, payload->stack.quantity, &reason))
218 return error(eve::DiagnosticCode::PreconditionViolation,
219 reason.empty() ? "bag rejected inventory object" : reason);
220 }
221 return eve::Result<void>::success();
222 }
223
224 if (equipment_ == nullptr) return error(eve::DiagnosticCode::InvalidArgument, "equipment is unbound");
225 if (!destination || !destination->isValid() ||
226 static_cast<std::size_t>(destination->value()) >= static_cast<std::size_t>(equipment_->getSlotCount()))
227 return error(eve::DiagnosticCode::PreconditionViolation,
228 "equipment transfer requires a valid destination slot");
229 const std::string target = slotName(*destination);
230 if (target.empty()) return error(eve::DiagnosticCode::NotFound, "equipment slot was not found");
231 const auto* current = equipment_->stackAt(target);
232 if (current != nullptr && !current->empty() && (!ignoredObject || object.id != objectId(*current)))
233 return error(eve::DiagnosticCode::Conflict, "equipment destination slot is occupied");
234 if (!definition->equipSlot.empty() && definition->equipSlot != target)
235 return error(eve::DiagnosticCode::PreconditionViolation, "inventory item does not fit equipment slot");
236 const auto* allowed = equipment_->allowedTags(target);
237 if (allowed != nullptr && !allowed->empty()) {
238 const bool accepted = std::any_of(allowed->begin(), allowed->end(), [&](const std::string& tag) {
239 return definition->hasTag(tag) || payload->stack.hasTag(tag);
240 });
241 if (!accepted)
242 return error(eve::DiagnosticCode::PreconditionViolation, "equipment slot rejected inventory tags");
243 }
244 return eve::Result<void>::success();
245}

247eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>> InventoryContainerAdapter::prepare(
248 const eve::container::ContainerSnapshot& expected, const eve::container::ContainerSnapshot& candidate) {
249 const bool bound = kind_ == InventoryContainerKind::Bag ? bag_ != nullptr : equipment_ != nullptr;
250 if (!bound || !descriptor_.id.isValid())
251 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
252 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "inventory adapter is not bound"));
253 if (expected.id != descriptor_.id || candidate.id != descriptor_.id)
254 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(eve::Diagnostic::error(
255 eve::DiagnosticCode::InvalidArgument, "inventory prepare targets another container"));
256 auto currentResult = snapshot();
257 if (!currentResult)
258 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(currentResult.status());
259 if (!sameLayout(currentResult.value(), expected))
260 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
261 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "inventory prepare expected snapshot is stale"));
262 const auto nextRevision = expected.revision.incremented();
263 if (!nextRevision || candidate.revision != *nextRevision)
264 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(eve::Diagnostic::error(
265 eve::DiagnosticCode::InvariantViolation, "inventory candidate revision is not the next revision"));
266 const int slotCount = kind_ == InventoryContainerKind::Bag ? bag_->getSlotCount() : equipment_->getSlotCount();
267 if (!descriptor_.capacity.isUnlimited() && candidate.entries.size() > descriptor_.capacity.value())
268 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
269 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "inventory candidate exceeds container capacity"));
270
271 std::vector<ItemStack> staged(static_cast<std::size_t>(slotCount));
272 std::unordered_set<eve::container::MembershipId> ids;
273 std::unordered_set<std::int32_t> slots;
274 for (const auto& entry : candidate.entries) {
275 if (!entry.membership.object.isValid() || !entry.membership.slot.isValid() ||
276 entry.membership.slot.value() >= slotCount ||
277 (!descriptor_.capacity.isUnlimited() &&
278 static_cast<std::size_t>(entry.membership.slot.value()) >= descriptor_.capacity.value()) ||
279 entry.membership.generation.isZero() || entry.object.id != entry.membership.object ||
280 !ids.insert(entry.membership.object).second || !slots.insert(entry.membership.slot.value()).second)
281 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
282 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
283 "inventory candidate contains invalid membership"));
284 const auto* payload = dynamic_cast<const InventoryContainerObject*>(entry.object.payload.get());
285 if (payload == nullptr)
286 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
287 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "inventory candidate payload is stale"));
288 auto validObject = validateObject(*payload);
289 if (!validObject)
290 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
291 validObject.status());
292 if (entry.object.type != payload->stack.itemId ||
293 entry.object.quantity != static_cast<std::uint32_t>(payload->stack.quantity) ||
294 objectId(payload->stack) != entry.membership.object ||
295 (kind_ == InventoryContainerKind::Equipment && slotName(entry.membership.slot).empty()))
296 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
297 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "inventory candidate object facts are stale"));
298 staged[static_cast<std::size_t>(entry.membership.slot.value())] = payload->stack;
299 }
300
301 std::unique_ptr<eve::container::IContainer::PreparedState> prepared =
302 std::make_unique<PreparedState>(*this, std::move(staged), *nextRevision);
303 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::success(std::move(prepared));
304}
