35Result<Coverage> coverage(const MeshBuild& mesh, const RoadTerrainTile& tile, int width, int height,
36 const RoadTerrainWriteOptions& options, std::size_t& work) {
37 Coverage out{Heightmap(width, height), Heightmap(width, height)};
38 std::vector<bool> protectedCells(static_cast<std::size_t>(width) * height, false);
39 const float dx = tile.sizeX / static_cast<float>(std::max(1, width - 1)),
40 dz = tile.sizeZ / static_cast<float>(std::max(1, height - 1));
41 for (int t = 0; t < mesh.getIndexCount() / 3; ++t) {
42 const auto group = mesh.getGroupName(mesh.getTriangleGroup(t));
43 if (group != "asphalt" && group != "asphalt.bridge" && group != "asphalt.tunnel" &&
44 group != "asphalt.tunnel.portal")
45 continue;
46 const int a = mesh.getIndex(t * 3), b = mesh.getIndex(t * 3 + 1), c = mesh.getIndex(t * 3 + 2);
47 const float ax = mesh.getPositionX(a), az = mesh.getPositionZ(a), bx = mesh.getPositionX(b),
48 bz = mesh.getPositionZ(b), cx = mesh.getPositionX(c), cz = mesh.getPositionZ(c);
49 const float denominator = (bz - cz) * (ax - cx) + (cx - bx) * (az - cz);
50 if (std::fabs(denominator) < 1e-7f) continue;
51 const float minX = std::clamp((std::min({ax, bx, cx}) - tile.originX) / dx, 0.f, static_cast<float>(width - 1));
52 const float maxX = std::clamp((std::max({ax, bx, cx}) - tile.originX) / dx, 0.f, static_cast<float>(width - 1));
53 const float minZ =
54 std::clamp((std::min({az, bz, cz}) - tile.originZ) / dz, 0.f, static_cast<float>(height - 1));
55 const float maxZ =
56 std::clamp((std::max({az, bz, cz}) - tile.originZ) / dz, 0.f, static_cast<float>(height - 1));
57 for (int z = static_cast<int>(std::floor(minZ)); z <= static_cast<int>(std::ceil(maxZ)); ++z)
58 for (int x = static_cast<int>(std::floor(minX)); x <= static_cast<int>(std::ceil(maxX)); ++x) {
59 if (++work > options.maximumWork) return invalid<Coverage>("channel rasterization exceeds work budget");
60 const float tx = width == 1 ? .5f : static_cast<float>(x) / static_cast<float>(width - 1);
61 const float tz = height == 1 ? .5f : static_cast<float>(z) / static_cast<float>(height - 1);
62 const float wx = tile.originX + tx * tile.sizeX, wz = tile.originZ + tz * tile.sizeZ;
63 const float u = ((bz - cz) * (wx - cx) + (cx - bx) * (wz - cz)) / denominator;
64 const float v = ((cz - az) * (wx - cx) + (ax - cx) * (wz - cz)) / denominator, w = 1.f - u - v;
65 // Portal boundaries come from sampled arc interpolation. Use a world-space
66 // 2 mm edge tolerance so a grid point on a portal border survives a road split.
67 const bool portal = group == "asphalt.tunnel.portal";
68 const float tu = portal ? .002f * std::hypot(bx - cx, bz - cz) / std::fabs(denominator) : 1e-4f;
69 const float tv = portal ? .002f * std::hypot(cx - ax, cz - az) / std::fabs(denominator) : 1e-4f;
70 const float tw = portal ? .002f * std::hypot(ax - bx, az - bz) / std::fabs(denominator) : 1e-4f;
71 if (u < -tu || v < -tv || w < -tw) continue;
72 const auto index = static_cast<std::size_t>(z) * width + x;
73 if (group == "asphalt.tunnel" || group == "asphalt.tunnel.portal") {
74 if (!options.tunnelHolesAtPortalsOnly || group == "asphalt.tunnel.portal")
75 out.tunnel.data()[index] = 1.f;
76 protectedCells[index] = true;
77 } else if (group == "asphalt.bridge")
78 protectedCells[index] = true;
79 else {
80 const float roadY = u * mesh.getPositionY(a) + v * mesh.getPositionY(b) + w * mesh.getPositionY(c);
81 const float terrainY =
82 tile.originY + tile.heights.sampleBilinear(tx * (tile.heights.getWidth() - 1),
83 tz * (tile.heights.getHeight() - 1)) *
84 tile.heightScale;
85 if (std::fabs(roadY - terrainY) <= options.maxVerticalDelta)
86 out.ground.data()[index] = 1.f;
87 else
88 protectedCells[index] = true;
89 }
90 }
91 }
92 const auto original = out.ground.data();
93 const int radiusX = static_cast<int>(std::ceil(options.blendDistance / dx)),
94 radiusZ = static_cast<int>(std::ceil(options.blendDistance / dz));
95 if (radiusX > 256 || radiusZ > 256) return invalid<Coverage>("channel blend radius exceeds 256 cells");
96 for (int z = 0; z < height; ++z)
97 for (int x = 0; x < width; ++x) {
98 if (original[static_cast<std::size_t>(z) * width + x] == 0.f) continue;
99 for (int oz = -radiusZ; oz <= radiusZ; ++oz)
100 for (int ox = -radiusX; ox <= radiusX; ++ox) {
101 if (++work > options.maximumWork) return invalid<Coverage>("channel blending exceeds work budget");
102 const int px = x + ox, pz = z + oz;
103 if (px < 0 || pz < 0 || px >= width || pz >= height) continue;
104 const auto index = static_cast<std::size_t>(pz) * width + px;
105 if (protectedCells[index] && original[index] == 0.f) continue;
106 const float distance = std::hypot(static_cast<float>(ox) * dx, static_cast<float>(oz) * dz);
107 const float weight =
108 options.blendDistance == 0.f ? 1.f : std::max(0.f, 1.f - distance / options.blendDistance);
109 out.ground.data()[index] = std::max(out.ground.data()[index], weight);
110 }
111 }
112 return Result<Coverage>::success(std::move(out));
113}

116Result<void> validateTerrainTiles(const std::vector<RoadTerrainTile>& tiles) {
117 if (tiles.empty() || tiles.size() > 256u) return invalid<void>("terrain requires 1..256 tiles");
118 std::set<std::string> tileIds;
119 std::size_t samples = 0;
120 for (const auto& tile : tiles) {
121 if (tile.id.empty() || tile.id.size() > 256u || !tileIds.insert(tile.id).second ||
122 !std::isfinite(tile.originX) || !std::isfinite(tile.originY) || !std::isfinite(tile.originZ) ||
123 !std::isfinite(tile.sizeX) || !std::isfinite(tile.sizeZ) || !std::isfinite(tile.heightScale) ||
124 tile.sizeX <= 0.f || tile.sizeZ <= 0.f || tile.heightScale <= 0.f || !rasterValid(tile.heights, false) ||
125 tile.heights.getWidth() < 2 || tile.heights.getHeight() < 2 || !rasterValid(tile.holes, true) ||
126 std::fabs(tile.sizeX / (tile.heights.getWidth() - 1) - tile.sizeZ / (tile.heights.getHeight() - 1)) >
127 1e-4f ||
128 tile.splatLayers.size() > 64u || tile.details.size() > 256u || tile.trees.size() > 1000000u)
129 return invalid<void>("invalid terrain tile identity, geometry or channel topology");
130 samples += tile.heights.data().size() + tile.holes.data().size() + tile.trees.size() * 8u;
131 for (float v : tile.holes.data())
132 if (v != 0.f && v != 1.f) return invalid<void>("holes must be binary");
133 for (const auto& layer : tile.splatLayers) {
134 if (!rasterValid(layer, true) || layer.getWidth() != tile.splatLayers.front().getWidth() ||
135 layer.getHeight() != tile.splatLayers.front().getHeight())
136 return invalid<void>("invalid splat topology");
137 samples += layer.data().size();
138 }
139 if (!tile.splatLayers.empty())
140 for (std::size_t i = 0; i < tile.splatLayers.front().data().size(); ++i) {
141 float sum = 0.f;
142 for (const auto& layer : tile.splatLayers) sum += layer.data()[i];
143 if (std::fabs(sum - 1.f) > 1e-4f) return invalid<void>("splat weights must sum to one");
144 }
145 std::set<std::string> resources, trees;
146 for (const auto& d : tile.details) {
147 if (d.resourceId.empty() || d.resourceId.size() > 256u || !resources.insert(d.resourceId).second ||
148 d.width < 1 || d.height < 1 || d.width > 4096 || d.height > 4096 ||
149 d.density.size() != static_cast<std::size_t>(d.width) * d.height)
150 return invalid<void>("invalid detail identity or topology");
151 for (int v : d.density)
152 if (v < 0) return invalid<void>("negative detail density");
153 samples += d.density.size();
154 }
155 for (const auto& tree : tile.trees) {
156 if (tree.id.empty() || tree.id.size() > 256u || tree.resourceId.empty() || tree.resourceId.size() > 256u ||
157 !trees.insert(tree.id).second || !std::isfinite(tree.x) || !std::isfinite(tree.y) ||
158 !std::isfinite(tree.z) || !std::isfinite(tree.yaw) || !std::isfinite(tree.width) ||
159 !std::isfinite(tree.height) || tree.width <= 0.f || tree.height <= 0.f)
160 return invalid<void>("invalid tree instance");
161 }
162 if (samples > 16u * 1024u * 1024u) return invalid<void>("terrain channel storage exceeds budget");
163 }
164 return Result<void>::success();
165}

184Result<int> RoadTerrainWorkspace::applyBatch(std::span<const RoadTerrainWrite> writes, std::size_t maximumWork) {
185 if (!impl_ || impl_->history.empty() || writes.empty() || writes.size() > 4096u || maximumWork == 0u)
186 return invalid<int>("invalid terrain batch or uninitialized owner");
187 for (const auto& write : writes) {
188 const auto& options = write.options;
189 if (!std::isfinite(options.blendDistance) || options.blendDistance < 0.f ||
190 !std::isfinite(options.maxVerticalDelta) || options.maxVerticalDelta < 0.f || options.splatLayer < -1 ||
191 options.maximumWork == 0u || !std::isfinite(options.splatOpacity) || options.splatOpacity < 0.f ||
192 options.splatOpacity > 1.f)
193 return invalid<int>("invalid terrain write options");
194 }
195 std::size_t historyStorage = storage(impl_->history[impl_->cursor]);
196 for (std::size_t i = 0; i <= impl_->cursor; ++i) historyStorage += storage(impl_->history[i]);
197 if (historyStorage > 16u * 1024u * 1024u || impl_->cursor >= 4095u)
198 return invalid<int>("terrain undo history exceeds save budget");
199 auto frame = impl_->history[impl_->cursor];
200 std::size_t work = 0;
201 std::int64_t changed = 0;
202 for (const auto& write : writes) {
203 if (work >= maximumWork) return invalid<int>("terrain batch exceeds shared work budget");
204 auto options = write.options;
205 options.maximumWork = work + std::min(maximumWork - work, options.maximumWork);
206 const auto& mesh = write.mesh.get();
207 for (auto& tile : frame) {
208 if (options.splatLayer >= static_cast<int>(tile.splatLayers.size()))
209 return invalid<int>("selected splat layer absent on tile");
210 RoadTerrainConformOptions conform;
211 conform.originX = tile.originX;
212 conform.originY = tile.originY;
213 conform.originZ = tile.originZ;
214 conform.cellSize = tile.sizeX / (tile.heights.getWidth() - 1);
215 conform.heightScale = tile.heightScale;
216 conform.blendDistance = options.blendDistance;
217 conform.maxVerticalDelta = options.maxVerticalDelta;
218 // Validate mesh through the canonical conformance path before any channel indexes its streams.
219 if (work >= options.maximumWork) return invalid<int>("terrain write exceeds shared work budget");
220 conform.maximumWork = options.maximumWork - work;
221 const auto baseline = tile.heights;
222 auto height = conformHeightmapToRoadWithReceipt(tile.heights, mesh, conform);
223 if (!height.ok()) return Result<int>::failure(height.status());
224 work += height.value().workPerformed;
225 changed += static_cast<int>(height.value().sampleIndices.size());
226 // Channel eligibility uses the original terrain height, so height writes cannot change bridge
227 // classification.
228 RoadTerrainTile mapping = tile;
229 mapping.heights = baseline;
230 if (options.splatLayer >= 0) {
231 auto mask = coverage(mesh, mapping, tile.splatLayers.front().getWidth(),
232 tile.splatLayers.front().getHeight(), options, work);
233 if (!mask.ok()) return Result<int>::failure(mask.status());
234 for (std::size_t i = 0; i < mask.value().ground.data().size(); ++i) {
235 const float weight = mask.value().ground.data()[i] * options.splatOpacity;
236 const auto target = static_cast<std::size_t>(options.splatLayer);
237 if (weight <= tile.splatLayers[target].data()[i]) continue;
238 float remainder = 0.f;
239 for (std::size_t l = 0; l < tile.splatLayers.size(); ++l)
240 if (l != target) remainder += tile.splatLayers[l].data()[i];
241 for (std::size_t l = 0; l < tile.splatLayers.size(); ++l) {
242 auto& value = tile.splatLayers[l].data()[i];
243 const float next = l == target ? (remainder > 0.f ? weight : 1.f)
244 : (remainder > 0.f ? value * ((1.f - weight) / remainder) : 0.f);
245 changed += value != next;
246 value = next;
247 }
248 }
249 }
250 if (options.clearDetails)
251 for (auto& d : tile.details) {
252 auto mask = coverage(mesh, mapping, d.width, d.height, options, work);
253 if (!mask.ok()) return Result<int>::failure(mask.status());
254 for (std::size_t i = 0; i < d.density.size(); ++i)
255 if (mask.value().ground.data()[i] > 0.f ||
256 (options.cutTunnelHoles && mask.value().tunnel.data()[i] > 0.f)) {
257 changed += d.density[i] != 0;
258 d.density[i] = 0;
259 }
260 }
261 if (options.clearTrees && !tile.trees.empty()) {
262 auto mask = coverage(mesh, mapping, tile.heights.getWidth(), tile.heights.getHeight(), options, work);
263 if (!mask.ok()) return Result<int>::failure(mask.status());
264 const auto oldCount = tile.trees.size();
265 std::erase_if(tile.trees, [&](const RoadTerrainTree& tree) {
266 const float x = (tree.x - tile.originX) / tile.sizeX, z = (tree.z - tile.originZ) / tile.sizeZ;
267 if (x < 0.f || z < 0.f || x > 1.f || z > 1.f) return false;
268 const float ground = mask.value().ground.sampleBilinear(x * (mask.value().ground.getWidth() - 1),
269 z * (mask.value().ground.getHeight() - 1));
270 const float tunnel = mask.value().tunnel.sampleBilinear(x * (mask.value().tunnel.getWidth() - 1),
271 z * (mask.value().tunnel.getHeight() - 1));
272 return ground > 0.f || (options.cutTunnelHoles && tunnel > 0.f);
273 });
274 changed += static_cast<int>(oldCount - tile.trees.size());
275 }
276 if (options.cutTunnelHoles) {
277 auto mask = coverage(mesh, mapping, tile.holes.getWidth(), tile.holes.getHeight(), options, work);
278 if (!mask.ok()) return Result<int>::failure(mask.status());
279 for (std::size_t i = 0; i < tile.holes.data().size(); ++i)
280 if (mask.value().tunnel.data()[i] > 0.f && mask.value().ground.data()[i] == 0.f) {
281 changed += tile.holes.data()[i] != 1.f;
282 tile.holes.data()[i] = 1.f;
283 }
284 }
285 }
286 }
287 if (changed > std::numeric_limits<int>::max())
288 return invalid<int>("terrain batch change count exceeds result range");
289 auto candidate = std::make_unique<Impl>(*impl_);
290 candidate->history.resize(candidate->cursor + 1u);
291 candidate->history.push_back(std::move(frame));
292 ++candidate->cursor;
293 impl_.swap(candidate);
294 return Result<int>::success(static_cast<int>(changed));
295}
