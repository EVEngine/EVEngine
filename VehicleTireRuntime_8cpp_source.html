60void driveInputs(physics::TireInput& target, const SuspensionConfig& config, const VehicleEntity::Input& input,
61 const SuspensionWheel& wheel, std::size_t driven, double maximumSpeed, double traction) {
62 const double throttle = std::clamp(double(input.throttle), -1.0, 1.0);
63 const double speedLimit = std::abs(throttle) * maximumSpeed;
64 const bool driving =
65 driven > 0 && wheel.drive && throttle * target.longitudinalVelocity < std::abs(throttle) * speedLimit;
66 target.driveTorque = driving ? traction * throttle * wheel.radius / static_cast<double>(driven) : 0;
67 target.brakeTorque = config.brakeTorque * (input.handbrake ? 1 : std::clamp(double(input.brake), 0.0, 1.0));
