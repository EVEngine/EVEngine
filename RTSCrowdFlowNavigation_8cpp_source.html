28Result<CrowdFlowNavigation::InputChange> CrowdFlowNavigation::begin(crowd::Crowd& crowd, const SimulationStep& step) {
29 if (!enabled_) return failure<InputChange>(DiagnosticCode::PreconditionViolation, "Shared-flow policy is disabled");
30 notices_.clear();
31 const auto source = crowd.getFieldStatus();
32 if (source.surface.surface.isZero())
33 return failure<InputChange>(DiagnosticCode::PreconditionViolation,
34 "Shared-flow mode requires an imported movement projection");
35 const bool ownerChanged = owner_ != source.owner;
36 const bool changed = ownerChanged || input_ != source.inputRevision;
37 if (changed) {
38 jobs_.clear();
39 workAfter_ = 0;
40 roundThrough_ = 0;
41 frame_.reset();
42 if (ownerChanged) pumped_.reset();
43 owner_ = source.owner;
44 input_ = source.inputRevision;
45 }
46 if (!frame_ || *frame_ != step.tick) {
47 if (frame_)
48 jobs_.erase(std::remove_if(jobs_.begin(), jobs_.end(), [&](const Job& job) { return job.used != *frame_; }),
49 jobs_.end());
50 frame_ = step.tick;
51 }
52 return Result<InputChange>::success(changed ? InputChange::Changed : InputChange::Unchanged);
53}
54Result<crowd::FlowGoal> CrowdFlowNavigation::goalCell(crowd::Crowd& crowd, WorldPosition target) const {
55 if (!std::isfinite(target.x) || !std::isfinite(target.y))
56 return failure<crowd::FlowGoal>(DiagnosticCode::InvalidArgument, "Nonfinite shared goal");
57 const double x = std::floor((double(target.x) - crowd.getFieldOriginX()) / crowd.getCellSize());
58 const double y = std::floor((double(target.y) - crowd.getFieldOriginY()) / crowd.getCellSize());
59 if (x < 0 || y < 0 || x >= crowd.getFieldWidth() || y >= crowd.getFieldHeight())
60 return failure<crowd::FlowGoal>(DiagnosticCode::InvalidArgument, "Shared goal outside navigation surface");
61 return Result<crowd::FlowGoal>::success({static_cast<int>(x), static_cast<int>(y)});

63Result<std::vector<crowd::FlowGoal>> CrowdFlowNavigation::goalSet(crowd::Crowd& crowd,
64 std::span<const WorldPosition> targets) const {

67 "Expected 1..1024 shared targets");

69 goals.reserve(targets.size());

92 return std::lexicographical_compare(job.goals.begin(), job.goals.end(), key.begin(), key.end());
93 });

102 return failure<void>(DiagnosticCode::PreconditionViolation, "Shared-goal admission sequence exhausted");

104 job.sequence = nextSequence_;
105 job.goals.assign(goals.begin(), goals.end());
106 job.used = tick;
107 jobs_.insert(found, std::move(job));

109 return Result<void>::success();

113Result<void> CrowdFlowNavigation::pump(crowd::Crowd& crowd, const SimulationStep& step) try {
114 if (step.delta.isZero() || (pumped_ && *pumped_ == step.tick))
115 return Result<void>::success(Status::success(StatusCode::NoOp));
116 // One attempt owns this tick's quota, even if a later stage fails. Input replacement
117 // may retire the partial plans, but does not refund work in the same owner timeline.
118 pumped_ = step.tick;
119 stats_.lastWork = stats_.lastStarts = 0;
120 stats_.cachePressure = false;
121 // Canonical key order serves lookup only. Admission order serves quotas, so new low keys
122 // cannot displace older work. These pointers are borrowed only while jobs_ is unchanged.
123 std::vector<Job*> ordered(jobs_.size(), nullptr);
124 std::size_t count = 0, activeBuilders = 0;
125 for (const auto& job : jobs_) {
126 const auto state = job.plan.state();
127 if (state == crowd::Crowd::FlowBuildState::Pending || state == crowd::Crowd::FlowBuildState::Ready ||
128 state == crowd::Crowd::FlowBuildState::Unreachable)
129 ++activeBuilders;
130 }
131 for (auto& job : jobs_)
132 if (job.used == step.tick) ordered[count++] = &job;
133 std::sort(ordered.begin(), ordered.begin() + count,
134 [](const Job* left, const Job* right) { return left->sequence < right->sequence; });
135 for (std::size_t index = 0; index < count; ++index) {
136 auto& job = *ordered[index];
137 if (job.used != step.tick || job.lease.generation() != 0) continue;
138 if (job.plan.state() == crowd::Crowd::FlowBuildState::Cancelled) {
139 auto cached = crowd.acquireFlowField(job.goals);
140 if (cached) {
141 job.lease = std::move(cached).takeValue();
142 ++stats_.cacheHits;
143 continue;
144 }
145 if (cached.code() != StatusCode::NotFound) return Result<void>::failure(cached.status());
146 cached.ignore("Ready cache miss; explicit bounded preparation follows");
147 if (stats_.lastStarts == settings_.maxStartsPerTick || activeBuilders == 64) continue;
148 ++stats_.lastStarts;
149 auto prepared = crowd.prepareFlowBuild(job.goals);
150 if (!prepared) return Result<void>::failure(prepared.status());
151 job.plan = std::move(prepared).takeValue();
152 ++stats_.prepared;
153 ++activeBuilders;
154 }
155 }
156 int remaining = settings_.maxWorkPerTick;
157 while (remaining > 0 && count != 0) {
158 Job* selected = nullptr;
159 for (std::size_t index = 0; index < count; ++index) {
160 auto* job = ordered[index];
161 if (job->sequence > workAfter_ && job->sequence <= roundThrough_ &&
162 job->plan.state() == crowd::Crowd::FlowBuildState::Pending) {
163 selected = job;
164 break;
165 }
166 }
167 if (!selected) {
168 // Freeze admission at each round boundary: newcomers wait until older
169 // pending requests have each had their quantum, even across ticks.
170 roundThrough_ = ordered[count - 1]->sequence;
171 workAfter_ = 0;
172 for (std::size_t index = 0; index < count; ++index)
173 if (ordered[index]->plan.state() == crowd::Crowd::FlowBuildState::Pending) {
174 selected = ordered[index];
175 break;
176 }
177 }
178 if (!selected) break;
179 const int reserved = std::min(remaining, 256);
180 stats_.lastWork += reserved;
181 auto progress = crowd.advanceFlowBuild(selected->plan, reserved);
182 if (!progress) {
183 // A failed builder may contain partial internal work; it must never become a ready field.
184 auto cancelled = selected->plan.cancel();
185 if (!cancelled) return Result<void>::failure(cancelled.status());
186 return Result<void>::failure(progress.status());
187 }
188 const int work = progress.value().heapPops + progress.value().directionCells;
189 workAfter_ = selected->sequence;
190 remaining -= work;
191 stats_.lastWork += work - reserved;
192 if (work == 0 && selected->plan.state() == crowd::Crowd::FlowBuildState::Pending) break;
193 }
194 for (auto& job : jobs_) {
195 if (job.used != step.tick || job.lease.generation() != 0) continue;
196 if (job.plan.state() != crowd::Crowd::FlowBuildState::Ready &&
197 job.plan.state() != crowd::Crowd::FlowBuildState::Unreachable)
198 continue;
199 auto acquired = crowd.acquireFlowField(std::move(job.plan));
200 if (acquired) {
201 job.lease = std::move(acquired).takeValue();
202 continue;
203 }
204 if (acquired.code() != StatusCode::Rejected) return Result<void>::failure(acquired.status());
205 acquired.ignore("Cache pressure retains completed work for the next quota phase");
206 stats_.cachePressure = true;
207 }
208 return Result<void>::success();
209} catch (const std::exception& error) {
210 return failure<void>(DiagnosticCode::Failed, error.what());
211}

227Result<CrowdFlowNavigation::DriveState> CrowdFlowNavigation::steerGoals(crowd::Crowd& crowd, WorldPosition position,
228 std::span<const crowd::FlowGoal> goals,
229 WorldPosition target, float radius,
230 crowd::MovementBinding movement,
231 crowd::AgentUpdate& update) {
232 auto destinationCell = goalCell(crowd, target);
233 if (!destinationCell) return Result<DriveState>::failure(destinationCell.status());
234 if (!std::binary_search(goals.begin(), goals.end(), destinationCell.value()))
235 return Result<DriveState>::failure(Diagnostic::error(
236 DiagnosticCode::InvalidArgument, "Exact destination must belong to the shared goal set", "target"));
237 auto found = std::lower_bound(jobs_.begin(), jobs_.end(), goals, [](const Job& job, const auto& key) {
238 return std::lexicographical_compare(job.goals.begin(), job.goals.end(), key.begin(), key.end());
239 });
240 update.flow = crowd::FlowLease{};
241 update.intent = {crowd::AgentAction::Idle, false, 0.f, 0.f};
242 if (found == jobs_.end() || !std::equal(found->goals.begin(), found->goals.end(), goals.begin(), goals.end()) ||
243 found->lease.generation() == 0) {
244 update.maxSpeed = 0.f;
245 return Result<DriveState>::success(DriveState::Pending);
246 }
247 update.flow = found->lease;
248 const auto source = crowd.getFieldStatus().surface;
249 if (radius > source.clearanceRadius ||
250 (!movement.movement.isZero() &&
251 (movement.movement != source.movement || movement.revision != source.movementRevision))) {
252 update.maxSpeed = 0.f;
253 return Result<DriveState>::success(DriveState::Unavailable);
254 }
255 auto sampled = crowd.sampleFlowField(found->lease, position.x, position.y);
256 if (!sampled) return Result<DriveState>::failure(sampled.status());
257 auto destination = crowd.sampleFlowField(found->lease, target.x, target.y);
258 if (!destination) return Result<DriveState>::failure(destination.status());
259 if (sampled.value().state == crowd::FlowSampleState::Unreachable ||
260 destination.value().state == crowd::FlowSampleState::Unreachable) {
261 update.maxSpeed = 0.f;
262 return Result<DriveState>::success(DriveState::Unreachable);
263 }
264 auto cell = goalCell(crowd, position);
265 if (!cell) return Result<DriveState>::failure(cell.status());
266 update.intent = cell.value() == destinationCell.value()
267 ? crowd::SteeringIntent{crowd::AgentAction::Seek, true, target.x, target.y}
268 : crowd::SteeringIntent{crowd::AgentAction::Flow, false, 0.f, 0.f};
269 return Result<DriveState>::success(DriveState::Ready);
270}
