16EditorResult<EnvironmentAssetCard> EnvironmentAssetBrowser::card(const AssetRecord& record) const {
17 EnvironmentAssetCard result;
18 result.asset = record.guid; result.logicalUri = record.logicalUri; result.status = record.status;
19 result.diagnostics = record.diagnostics;
20 if (!record.artifacts.empty()) result.previewArtifact = record.artifacts.front();
21 const auto* layoutEntry = field(record.metadata, "layout");
22 const auto* widthEntry = field(record.metadata, "width");
23 const auto* heightEntry = field(record.metadata, "height");
24 const auto* facesEntry = field(record.metadata, "faceCount");
25 const auto* layout = layoutEntry ? layoutEntry->getIf<std::string>() : nullptr;
26 const auto* width = widthEntry ? widthEntry->getIf<int64_t>() : nullptr;
27 const auto* height = heightEntry ? heightEntry->getIf<int64_t>() : nullptr;
28 const auto* faces = facesEntry ? facesEntry->getIf<int64_t>() : nullptr;
29 static const std::set<std::string> layouts{"cubemap", "horizontal-cross", "vertical-cross", "equirectangular"};
30 if (!layout || !layouts.contains(*layout) || !width || !height || *width <= 0 || *height <= 0) {
31 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
32 eve::DiagnosticCode::InvalidArgument, RuleId("editor.environment.invalid-map-metadata"),
33 DiagnosticSeverity::Error, "Environment map requires supported layout and positive dimensions"));
34 } else {
35 result.layout = *layout; result.width = static_cast<int>(*width); result.height = static_cast<int>(*height);
36 result.faceCount = faces ? static_cast<int>(*faces) : (*layout == "cubemap" ? 6 : 1);
37 if (*layout == "cubemap" && result.faceCount != 6)
38 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
39 eve::DiagnosticCode::InvalidArgument, RuleId("editor.environment.cubemap-face-count"),
40 DiagnosticSeverity::Error, "Cubemap assets require exactly six faces"));
41 if (*layout == "equirectangular" && *width != *height * 2)
42 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
43 eve::DiagnosticCode::InvalidArgument, RuleId("editor.environment.equirectangular-aspect"),
44 DiagnosticSeverity::Warning, "Equirectangular maps normally use a 2:1 aspect ratio"));
45 }
46 if (record.artifacts.empty())
47 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
48 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.environment.missing-artifact"),
49 DiagnosticSeverity::Error, "Environment asset has no imported artifact"));
50 if (record.status != AssetStatus::Ready && record.status != AssetStatus::Warning)
51 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
52 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.environment.asset-not-ready"),
53 DiagnosticSeverity::Error, "Environment asset is not ready for preview or assignment"));
54 const bool error = std::any_of(result.diagnostics.begin(), result.diagnostics.end(),
55 [](const EditorDiagnostic& diagnostic) {
56 return diagnostic.severity() == DiagnosticSeverity::Error ||
57 diagnostic.severity() == DiagnosticSeverity::Fatal;
58 });
59 if (error)
60 return EditorResult<EnvironmentAssetCard>::failure(
61 eve::Status(EditorStatus::Rejected, result.diagnostics));
62 return eve::editing::applied<EnvironmentAssetCard>(std::move(result));
63}

65EditorResult<EnvironmentAssetPage> EnvironmentAssetBrowser::query(
66 std::string text, std::size_t offset, std::size_t limit,
67 std::optional<std::uint64_t> generation) const {
68 if (!database_ || limit == 0)
69 return eve::editing::failed<EnvironmentAssetPage>(EditorStatus::Rejected,
70 RuleId("editor.environment.invalid-browser-query"), "Environment browser requires AssetDB and positive page size");
71 AssetQuery filter; filter.typeIds = {"cubemap", "image.cubemap", "environment-map"};
72 filter.text = std::move(text);
73 auto page = database_->query(filter, offset, limit, generation);
74 if (!page.ok()) return EditorResult<EnvironmentAssetPage>::failure(page.status());
75 EnvironmentAssetPage result; result.nextOffset = page.value().nextOffset;
76 result.hasMore = page.value().hasMore; result.generation = page.value().generation;
77 for (const auto& record : page.value().values) {
78 auto value = card(record);
79 if (value.ok()) result.values.push_back(value.value());
80 }
81 return eve::editing::applied<EnvironmentAssetPage>(std::move(result));
