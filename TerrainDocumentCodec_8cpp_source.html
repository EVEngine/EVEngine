49editor::EditorResult<editor::EditorValue> encodeTerrainDocument(const procgen::Heightmap& heightmap, float spacingX,
50 float spacingZ, const TerrainDocumentLimits& limits) {
51 const std::uint64_t width = static_cast<std::uint64_t>(heightmap.getWidth());
52 const std::uint64_t height = static_cast<std::uint64_t>(heightmap.getHeight());
53 const std::uint64_t samples = width * height;
54 if (width == 0 || height == 0 || width > limits.maximumDimension || height > limits.maximumDimension ||
55 samples > limits.maximumSamples || !std::isfinite(spacingX) || !std::isfinite(spacingZ) || spacingX <= 0.0F ||
56 spacingZ <= 0.0F || samples != heightmap.data().size())
57 return rejected<editor::EditorValue>("Terrain dimensions, spacing, or sample count is invalid");
58 editor::EditorValue::Array heights;
59 heights.reserve(heightmap.data().size());
60 for (float value : heightmap.data()) {
61 if (!std::isfinite(value)) return rejected<editor::EditorValue>("Terrain contains a non-finite height");
62 heights.emplace_back(static_cast<double>(value));
63 }
64 return editing::applied<editor::EditorValue>(
65 editor::EditorValue::Object{{"schema", "eve.terrain-editor-document"},
66 {"schemaVersion", std::int64_t{1}},
67 {"width", static_cast<std::int64_t>(width)},
68 {"height", static_cast<std::int64_t>(height)},
69 {"spacingX", static_cast<double>(spacingX)},
70 {"spacingZ", static_cast<double>(spacingZ)},
71 {"heights", std::move(heights)}});
72}

74editor::EditorResult<TerrainDocumentData> decodeTerrainDocument(const editor::EditorValue& value,
75 const TerrainDocumentLimits& limits) {
76 const auto* object = value.getIf<editor::EditorValue::Object>();
77 static const std::set<std::string, std::less<>> allowed{"schema", "schemaVersion", "width", "height",
78 "spacingX", "spacingZ", "heights"};
79 if (!object || object->size() != allowed.size())
80 return rejected<TerrainDocumentData>("Terrain document fields do not match schema version 1");
81 for (const auto& [name, ignored] : *object) {
82 (void)ignored;
83 if (!allowed.contains(name))
84 return rejected<TerrainDocumentData>("Terrain document contains an unknown field: " + name);
85 }
86 const auto* schema = field(*object, "schema");
87 const auto* version = field(*object, "schemaVersion");
88 const auto* schemaText = schema ? schema->getIf<std::string>() : nullptr;
89 const auto* versionNumber = version ? version->getIf<std::int64_t>() : nullptr;
90 if (!schemaText || *schemaText != "eve.terrain-editor-document" || !versionNumber || *versionNumber != 1)
91 return rejected<TerrainDocumentData>("Terrain document schema or version is unsupported");
92 std::uint32_t width = 0, height = 0;
93 float spacingX = 0.0F, spacingZ = 0.0F;
94 if (!dimension(field(*object, "width"), width) || !dimension(field(*object, "height"), height) ||
95 width > limits.maximumDimension || height > limits.maximumDimension ||
96 std::uint64_t(width) * height > limits.maximumSamples || !finiteNumber(field(*object, "spacingX"), spacingX) ||
97 !finiteNumber(field(*object, "spacingZ"), spacingZ) || spacingX <= 0.0F || spacingZ <= 0.0F)
98 return rejected<TerrainDocumentData>("Terrain document dimensions or spacing is invalid");
99 const auto* heightsValue = field(*object, "heights");
100 const auto* heights = heightsValue ? heightsValue->getIf<editor::EditorValue::Array>() : nullptr;
101 if (!heights || heights->size() != std::uint64_t(width) * height)
102 return rejected<TerrainDocumentData>("Terrain document height count does not match dimensions");
103 TerrainDocumentData result;
104 result.heightmap.resize(static_cast<int>(width), static_cast<int>(height));
105 result.spacingX = spacingX;
106 result.spacingZ = spacingZ;
107 for (std::size_t index = 0; index < heights->size(); ++index) {
108 float sample = 0.0F;
109 if (!finiteNumber(&(*heights)[index], sample))
110 return rejected<TerrainDocumentData>("Terrain document contains a non-finite height");
111 result.heightmap.data()[index] = sample;
112 }
113 return editing::applied<TerrainDocumentData>(std::move(result));
114}
