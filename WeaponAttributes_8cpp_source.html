72eve::Result<void> WeaponAttributeAdapter::ensure(WeaponEntity& weapon) {
73 auto bound = weapon.attributes()->values.bindOwner(ecs::handle_of(&weapon));
74 if (!bound) return failureVoid(bound.code(), "weapon attribute owner could not be bound", "owner");
75 if (!weapon.attributes()->values.initialized()) {
76 const Resource& resource = weapon.state()->resource;
77 const double mana = resource.kind == ResourceKind::Mana ? resource.value : 0.0;
78 const double maxMana = resource.kind == ResourceKind::Mana ? resource.max : 0.0;
79 const double stamina = resource.kind == ResourceKind::Stamina ? resource.value : 0.0;
80 const double maxStamina = resource.kind == ResourceKind::Stamina ? resource.max : 0.0;
81 const std::array<eve::attributes::AttributeSnapshotBase, 4> bases = {
82 eve::attributes::AttributeSnapshotBase{std::string(manaAttribute), mana},
83 eve::attributes::AttributeSnapshotBase{std::string(maxManaAttribute), maxMana},
84 eve::attributes::AttributeSnapshotBase{std::string(staminaAttribute), stamina},
85 eve::attributes::AttributeSnapshotBase{std::string(maxStaminaAttribute), maxStamina}};
86 auto initialized = weapon.attributes()->values.initialize(bases);
87 if (!initialized)
88 return failureVoid(initialized.code(), "weapon attribute state could not be initialized", "attributes");
89 }
90 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
91}

93eve::Result<void> WeaponAttributeAdapter::project(WeaponEntity& weapon) {
94 auto ready = ensure(weapon);
95 if (!ready) return failureVoid(ready.code(), "weapon attributes are not available", "attributes");
96 Resource& resource = weapon.state()->resource;
97 if (!isManaOrStamina(resource.kind)) return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
98 auto current = weapon.attributes()->values.getFinal(currentAttribute(resource.kind));
99 if (!current) return failureVoid(current.code(), "weapon resource attribute could not be read", "resource");
100 auto maximum = weapon.attributes()->values.getFinal(maxAttribute(resource.kind));
101 if (!maximum) return failureVoid(maximum.code(), "weapon maximum resource attribute could not be read", "resource");
102 if (!std::isfinite(current.value()) || !std::isfinite(maximum.value()) || current.value() < 0.0 ||
103 maximum.value() < 0.0 || current.value() > maximum.value() ||
104 maximum.value() > static_cast<double>(std::numeric_limits<float>::max()))
105 return failureVoid(eve::DiagnosticCode::InvariantViolation,
106 "weapon mana/stamina attributes are outside validated bounds", "resource");
107 resource.value = static_cast<float>(current.value());
108 resource.max = static_cast<float>(maximum.value());
109 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
110}

112eve::Result<double> WeaponAttributeAdapter::read(WeaponEntity& weapon, std::string_view attribute) {

118 auto value = weapon.attributes()->values.getFinal(attribute);
119 if (!value) return failure<double>(value.code(), "weapon attribute read failed", "attribute");
120 return eve::Result<double>::success(value.value());

134eve::Result<void> WeaponAttributeAdapter::consumeTriggerResource(WeaponEntity& weapon) {
135 auto ready = ensure(weapon);
136 if (!ready) return failureVoid(ready.code(), "weapon attributes are not available", "attributes");
137 const Resource& resource = weapon.state()->resource;
138 if (!isManaOrStamina(resource.kind))
139 return failureVoid(eve::DiagnosticCode::Unsupported, "weapon trigger resource is not mana or stamina",
140 "resource.kind");
141 const auto currentName = currentAttribute(resource.kind);
142 auto current = weapon.attributes()->values.getFinal(currentName);
143 if (!current) return failureVoid(current.code(), "weapon trigger resource could not be read", "resource");
144 if (!resource.infinite && current.value() < static_cast<double>(resource.cost))
145 return failureVoid(eve::DiagnosticCode::Conflict, "weapon does not have enough mana or stamina", "resource");
146 if (resource.infinite) return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
147 auto changed = weapon.attributes()->values.setBase(
148 currentName, std::max(0.0, current.value() - static_cast<double>(resource.cost)));
149 if (!changed) return changed;
150 return project(weapon);
151}
