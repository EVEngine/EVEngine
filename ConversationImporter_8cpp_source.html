18std::string trim(std::string value) {
19 const auto first =
20 std::find_if_not(value.begin(), value.end(), [](unsigned char ch) { return std::isspace(ch) != 0; });
21 const auto last =
22 std::find_if_not(value.rbegin(), value.rend(), [](unsigned char ch) { return std::isspace(ch) != 0; }).base();
23 return first < last ? std::string(first, last) : std::string{};
24}

26std::vector<std::pair<int, std::string>> splitLines(const std::string& source) {
27 std::vector<std::pair<int, std::string>> out;
28 std::istringstream stream(source);
29 std::string line;
30 int number = 0;
31 while (std::getline(stream, line)) {
32 ++number;
33 if (!line.empty() && line.back() == '\r') line.pop_back();
34 out.emplace_back(number, std::move(line));
35 }

39std::string assetIdFromPath(const std::string& path) {
40 const size_t slash = path.find_last_of("/\\");
41 const size_t dot = path.find_last_of('.');
42 const size_t begin = slash == std::string::npos ? 0 : slash + 1;
43 const size_t end = dot == std::string::npos || dot < begin ? path.size() : dot;
44 std::string id = path.substr(begin, end - begin);
45 for (char& ch : id)
46 if (!std::isalnum(static_cast<unsigned char>(ch)) && ch != '_' && ch != '-' && ch != '.') ch = '_';
47 return id.empty() ? "imported" : id;
48}

50bool parseLink(const std::string& text, std::string& label, std::string& target) {
51 if (text.size() < 4 || text.substr(0, 2) != "[[" || text.substr(text.size() - 2) != "]]") return false;
52 const std::string body = trim(text.substr(2, text.size() - 4));
53 size_t split = body.find("->");
54 size_t width = 2;
55 if (split == std::string::npos) {
56 split = body.find('|');
57 width = 1;
58 }
59 if (split == std::string::npos) {
60 split = body.find("<-");
61 if (split != std::string::npos) {
62 target = trim(body.substr(0, split));
63 label = trim(body.substr(split + 2));
64 return !label.empty() && !target.empty();
65 }
66 }
67 if (split == std::string::npos)
68 label = target = trim(body);
69 else {
70 label = trim(body.substr(0, split));
71 target = trim(body.substr(split + width));
72 }
73 return !label.empty() && !target.empty();
74}

79 return {};
80 return trim(text.substr(prefix.size(), text.size() - prefix.size() - 2));

83void consumeLineTags(std::string& text, ConversationAsset::Node& node) {
84 size_t tag = text.find(" #");
85 while (tag != std::string::npos) {
86 const size_t end = text.find(' ', tag + 2);
87 const std::string value = text.substr(tag + 2, end == std::string::npos ? std::string::npos : end - tag - 2);
88 if (value.rfind("line:", 0) == 0) node.i18nKey = value.substr(5);
89 if (value.rfind("voice:", 0) == 0) node.voice = value.substr(6);
90 text.erase(tag, end == std::string::npos ? std::string::npos : end - tag);
91 tag = text.find(" #", tag);
92 }
93 text = trim(text);
94}

101bool buildAsset(const std::vector<Passage>& passages, const std::string& path, std::vector<ConversationAsset>& assets,
102 std::vector<ConversationDiagnostic>& diagnostics) {
103 if (passages.empty()) {
104 addDiagnostic(diagnostics, ConversationDiagnostic::Severity::Error, path, 0, "no dialogue passages found");
105 return false;
106 }
107 ConversationAsset asset;
108 asset.id = assetIdFromPath(path);
109 asset.entry = passages.front().title;
110 std::unordered_set<std::string> titles;
111 for (const auto& passage : passages) {
112 if (passage.title.empty() || !titles.insert(passage.title).second)
113 addDiagnostic(diagnostics, ConversationDiagnostic::Severity::Error, path, passage.line,
114 "empty or duplicate passage title: " + passage.title);
115 }
116 for (const auto& passage : passages) {
117 std::vector<ConversationAsset::Node> nodes;
118 size_t index = 0;
119 for (size_t bodyIndex = 0; bodyIndex < passage.body.size();) {
120 const auto& [lineNumber, raw] = passage.body[bodyIndex];
121 const std::string text = trim(raw);
122 if (text.empty() || text == "---" || text == "===") {
123 ++bodyIndex;
124 continue;
125 }
126 std::string label;
127 std::string target;
128 if (parseLink(text, label, target) || text.rfind("->", 0) == 0) {
129 ConversationAsset::Node node;
130 node.kind = ConversationAsset::Node::Kind::Choice;
131 node.id = index == 0 ? passage.title : passage.title + "." + std::to_string(index);
132 ++index;
133 while (bodyIndex < passage.body.size()) {
134 const std::string candidate = trim(passage.body[bodyIndex].second);
135 if (!parseLink(candidate, label, target)) {
136 if (candidate.rfind("->", 0) != 0) break;
137 label = trim(candidate.substr(2));
138 target.clear();
139 if (bodyIndex + 1 < passage.body.size())
140 target = commandArgument(trim(passage.body[bodyIndex + 1].second), "jump");
141 if (target.empty()) {
142 addDiagnostic(diagnostics, ConversationDiagnostic::Severity::Error, path,
143 passage.body[bodyIndex].first,
144 "Yarn shortcut option requires a following <<jump Target>>");
145 break;
146 }
147 ++bodyIndex;
148 }
149 node.routes.emplace_back(label, target);
150 ++bodyIndex;
151 }
152 nodes.push_back(std::move(node));
153 continue;
154 }
155 ConversationAsset::Node node;
156 node.id = index == 0 ? passage.title : passage.title + "." + std::to_string(index);
157 ++index;
158 const std::string jump = commandArgument(text, "jump");
159 const std::string wait = commandArgument(text, "wait");
160 const std::string call = commandArgument(text, "call");
161 const std::string set = commandArgument(text, "set");
162 if (!jump.empty()) {
163 node.kind = ConversationAsset::Node::Kind::Branch;
164 node.routes.emplace_back("else", jump);
165 } else if (text == "<<stop>>") {
166 node.kind = ConversationAsset::Node::Kind::End;
167 } else if (!wait.empty()) {
168 node.kind = ConversationAsset::Node::Kind::Wait;
169 node.expression = wait;
170 } else if (!call.empty() || !set.empty()) {
171 node.kind = ConversationAsset::Node::Kind::Command;
172 node.target = !call.empty() ? "call" : "set";
173 node.expression = !call.empty() ? call : set;
174 } else {
175 node.kind = ConversationAsset::Node::Kind::Line;
176 std::string lineText = text;
177 consumeLineTags(lineText, node);
178 const size_t colon = lineText.find(':');
179 if (colon != std::string::npos && colon > 0 && lineText.find("[[") == std::string::npos) {
180 node.speaker = trim(lineText.substr(0, colon));
181 node.text = trim(lineText.substr(colon + 1));
182 } else {
183 node.text = lineText;
184 if (lineText.find("[[") != std::string::npos)
185 addDiagnostic(diagnostics, ConversationDiagnostic::Severity::Warning, path, lineNumber,
186 "inline passage links are imported as text");
187 }
188 }
189 nodes.push_back(std::move(node));
190 ++bodyIndex;
191 }
192 if (nodes.empty()) {
193 ConversationAsset::Node end;
194 end.id = passage.title;
195 end.kind = ConversationAsset::Node::Kind::End;
196 nodes.push_back(std::move(end));
197 }
198 const auto isSequential = [](ConversationAsset::Node::Kind kind) {
199 return kind == ConversationAsset::Node::Kind::Line || kind == ConversationAsset::Node::Kind::Command ||
200 kind == ConversationAsset::Node::Kind::Wait;
201 };
202 for (size_t i = 0; i + 1 < nodes.size(); ++i)
203 if (isSequential(nodes[i].kind)) nodes[i].next = nodes[i + 1].id;
204 if (isSequential(nodes.back().kind)) {
205 ConversationAsset::Node end;
206 end.id = passage.title + ".end";
207 end.kind = ConversationAsset::Node::Kind::End;
208 nodes.back().next = end.id;
209 nodes.push_back(std::move(end));
210 }
211 asset.nodes.insert(asset.nodes.end(), std::make_move_iterator(nodes.begin()),
212 std::make_move_iterator(nodes.end()));
213 }
214 if (std::any_of(diagnostics.begin(), diagnostics.end(), [](const auto& diagnostic) {
215 return diagnostic.severity == ConversationDiagnostic::Severity::Error;
216 }))
217 return false;
218 assets.push_back(std::move(asset));
219 return lintConversations(assets, path, diagnostics);
220}

224bool importYarnConversation(const std::string& source, const std::string& path, std::vector<ConversationAsset>& assets,
225 std::vector<ConversationDiagnostic>& diagnostics) {
226 std::vector<Passage> passages;
227 Passage current;
228 bool inBody = false;
229 for (const auto& [number, raw] : splitLines(source)) {
230 const std::string text = trim(raw);
231 if (text.rfind("title:", 0) == 0) {
232 if (!current.title.empty()) passages.push_back(std::move(current));
233 current = {};
234 current.title = trim(text.substr(6));
235 current.line = number;
236 inBody = false;
237 } else if (text == "---")
238 inBody = !current.title.empty();
239 else if (text == "===") {
240 if (!current.title.empty()) passages.push_back(std::move(current));
241 current = {};
242 inBody = false;
243 } else if (inBody)
244 current.body.emplace_back(number, raw);
245 }
246 if (!current.title.empty()) passages.push_back(std::move(current));
247 return buildAsset(passages, path, assets, diagnostics);
248}

251 std::vector<ConversationDiagnostic>& diagnostics) {

258 current = {};

262 current.line = number;
263 } else if (!current.title.empty())
264 current.body.emplace_back(number, raw);

268 [](const Passage& passage) {
269 return passage.title == "StoryTitle" || passage.title == "StoryData";
270 }),
