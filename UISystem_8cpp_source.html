52void renderDragDrop(UIHost *host, const UINode &node) {
53 if (UISystem::dragDropSupport() != DragDropSupport::Supported || !host ||
54 !node.enabled)
55 return;
56 const std::string hostName = host->meta()->name;
57 if (node.dragSource && !node.dragPayloadType.empty() && ImGui::BeginDragDropSource()) {
58 if (!g_activeDrag || g_activeDrag->sourceHostName != hostName ||
59 g_activeDrag->sourceNodeId != node.id ||
60 g_activeDrag->payloadType != node.dragPayloadType ||
61 g_activeDrag->payloadText != node.dragPayloadText) {
62 g_activeDrag = ActiveDragPayload{g_nextDragSerial++, hostName, node.id,
63 node.dragPayloadType, node.dragPayloadText};
64 }
65 const uint64_t serial = g_activeDrag->serial;
66 ImGui::SetDragDropPayload("EVE_UI_DND", &serial, sizeof(serial), ImGuiCond_Once);
67 ImGui::TextUnformatted(node.dragPayloadText.empty() ? node.dragPayloadType.c_str()
68 : node.dragPayloadText.c_str());
69 ImGui::EndDragDropSource();
70 }
71
72 if (node.dropTarget && ImGui::BeginDragDropTarget()) {
73 const bool acceptsActive = g_activeDrag &&
74 acceptsDropType(node, g_activeDrag->payloadType);
75 if (const ImGuiPayload *payload =
76 acceptsActive ? ImGui::AcceptDragDropPayload("EVE_UI_DND") : nullptr) {
77 if (payload->IsDelivery() && payload->DataSize == int(sizeof(uint64_t)) &&
78 g_activeDrag) {
79 const uint64_t serial = *static_cast<const uint64_t *>(payload->Data);
80 if (serial == g_activeDrag->serial) {
81 g_drops.push_back({DragDropOrigin::Internal,
82 g_activeDrag->sourceHostName,
83 g_activeDrag->sourceNodeId,
84 hostName,
85 node.id,
86 g_activeDrag->payloadType,
87 g_activeDrag->payloadText});
88 g_activeDrag.reset();
89 }
90 }
91 }
92 ImGui::EndDragDropTarget();
93 }
94
95 if (!g_pendingFileDrops.empty() && acceptsDropType(node, "file") &&
96 ImGui::IsItemHovered()) {
97 for (auto &path : g_pendingFileDrops) {
98 g_drops.push_back({DragDropOrigin::OperatingSystemFile, "", "", hostName,
99 node.id, "file", std::move(path)});
100 }
101 g_pendingFileDrops.clear();
102 }

115bool updateWorldAnchor(UIHost &host, graphics::Camera3D *camera, float width, float height) {
116 auto anchor = host.worldAnchor();
117 if (!anchor->enabled) {
118 anchor->state = WorldAnchorState::Disabled;
119 return true;
120 }
121 if (!camera || width <= 0.f || height <= 0.f) {
122 anchor->state = WorldAnchorState::NoCamera;
123 return false;
124 }
125 const auto data = camera->data();
126 const glm::vec3 eye(data->eyeX, data->eyeY, data->eyeZ);
127 const glm::vec3 target(data->targetX, data->targetY, data->targetZ);
128 const glm::vec3 up(data->upX, data->upY, data->upZ);
129 const glm::mat4 view = glm::lookAtRH(eye, target, up);
130 const glm::mat4 projection = graphics::cameraProjectionVulkanRH_ZO(
131 data->orthographic, glm::radians(data->fovYDeg), data->orthoHeight,
132 width / height, data->nearZ, data->farZ);
133 const auto projected = projectWorldAnchor(*anchor, projection * view, data->eyeX,
134 data->eyeY, data->eyeZ, width, height);
135 anchor->state = projected.state;
136 anchor->screenX = projected.screenX;
137 anchor->screenY = projected.screenY;
138 anchor->depth = projected.depth;
139 anchor->scale = projected.scale;
140 anchor->displacementX = 0.f;
141 anchor->displacementY = 0.f;
142 return projected.render;
143}

177bool privateUseGlyph(const std::string &text, ImWchar *out) {
178 if (text.size() != 3) return false;
179 const auto b0 = static_cast<unsigned char>(text[0]);
180 const auto b1 = static_cast<unsigned char>(text[1]);
181 const auto b2 = static_cast<unsigned char>(text[2]);
182 if ((b0 & 0xf0u) != 0xe0u || (b1 & 0xc0u) != 0x80u || (b2 & 0xc0u) != 0x80u)
183 return false;
184 const unsigned int codepoint =
185 ((b0 & 0x0fu) << 12u) | ((b1 & 0x3fu) << 6u) | (b2 & 0x3fu);
186 if (codepoint < 0xe000u || codepoint > 0xf8ffu) return false;
187 if (out) *out = static_cast<ImWchar>(codepoint);
188 return true;
189}

380void walkFlex(UIHost *host, UIHost::Tree *tree, UINode &flex) {
381 const bool row = flex.flexDirection == FlexDirection::Row;
382 const ImGuiStyle &style = ImGui::GetStyle();
383 const float gap = flex.gap >= 0.f ? flex.gap : (row ? style.ItemSpacing.x : style.ItemSpacing.y);
384
385 // Own content box: an explicit size (set by the parent's arrange) wins;
386 // otherwise use the available region (root flex inside a window/child).
387 const ImVec2 avail = ImGui::GetContentRegionAvail();
388 const bool fillsContainer =
389 flex.type == NodeType::Toolbar || flex.type == NodeType::StatusBar;
390 const float flexW = flex.sizeX > 0.f ? flex.sizeX : avail.x;
391 const float flexH =
392 flex.sizeY > 0.f ? flex.sizeY : (fillsContainer ? avail.y : flex.measuredH);
393 const float padMain = row ? flex.paddingL + flex.paddingR : flex.paddingT + flex.paddingB;
394 const float padCross = row ? flex.paddingT + flex.paddingB : flex.paddingL + flex.paddingR;
395 const float availMain = (row ? flexW : flexH) - padMain;
396 const float availCross = (row ? flexH : flexW) - padCross;
397
398 std::vector<int> kids;
399 std::vector<int> absKids;
400 std::vector<FlexItemSpec> specs;
401 for (int c = flex.firstChild; c >= 0; c = tree->nodes[size_t(c)].nextSibling) {
402 if (c >= int(tree->nodes.size())) break;
403 UINode &child = tree->nodes[size_t(c)];
404 if (!child.visible) continue;
405 if (child.absolute) {
406 absKids.push_back(c);
407 continue;
408 }
409 kids.push_back(c);
410 FlexItemSpec s;
411 s.basisMain = row ? child.measuredW : child.measuredH;
412 s.basisCross = row ? child.measuredH : child.measuredW;
413 s.marginBefore = row ? child.marginL : child.marginT;
414 s.marginAfter = row ? child.marginR : child.marginB;
415 s.marginCrossBefore = row ? child.marginT : child.marginL;
416 s.marginCrossAfter = row ? child.marginB : child.marginR;
417 s.flexGrow = child.flexGrow;
418 s.isSpacer = child.type == NodeType::Spacer;
419 s.minMain = row ? child.minSizeX : child.minSizeY;
420 s.maxMain = row ? child.maxSizeX : child.maxSizeY;
421 s.minCross = row ? child.minSizeY : child.minSizeX;
422 s.maxCross = row ? child.maxSizeY : child.maxSizeX;
423 s.percentMain = row ? child.percentW : child.percentH;
424 s.percentCross = row ? child.percentH : child.percentW;
425 s.explicitMain = row ? child.sizeX : child.sizeY;
426 s.explicitCross = row ? child.sizeY : child.sizeX;
427 specs.push_back(s);
428 }
429
430 const FlexResult res =
431 flexArrange(row, gap, std::max(0.f, availMain), std::max(0.f, availCross),
432 flex.alignItems, flex.justifyContent, specs);
433 const ImVec2 cursor = ImGui::GetCursorScreenPos();
434 ImGui::Dummy(ImVec2(std::max(0.f, flexW), std::max(0.f, flexH)));
435 const ImVec2 flowEnd = ImGui::GetCursorScreenPos();
436 const ImVec2 origin = ImVec2(cursor.x + flex.paddingL, cursor.y + flex.paddingT);
437
438 for (size_t i = 0; i < kids.size(); ++i) {
439 UINode &child = tree->nodes[size_t(kids[i])];
440 const FlexRect &r = res.items[i];
441 ImGui::SetCursorScreenPos(ImVec2(origin.x + r.x, origin.y + r.y));
442 const float oldX = child.sizeX;
443 const float oldY = child.sizeY;
444 bool pushedWidth = false;
445 if (r.w > 0.f) child.sizeX = r.w;
446 if (r.h > 0.f) child.sizeY = r.h;
447 if (row || r.w > 0.f) {
448 ImGui::PushItemWidth(r.w > 0.f ? r.w : 1.f);
449 pushedWidth = true;
450 }
451 walkNode(host, tree, kids[i]);
452 child.sizeX = oldX;
453 child.sizeY = oldY;
454 if (pushedWidth) ImGui::PopItemWidth();
455 }
456
457 // Absolutely placed children are excluded from flow; draw on top of it.
458 for (int c : absKids) {
459 UINode &child = tree->nodes[size_t(c)];
460 float w = child.percentW > 0.f ? child.percentW * flexW : child.measuredW;
461 float h = child.percentH > 0.f ? child.percentH * flexH : child.measuredH;
462 if (child.sizeX > 0.f) w = child.sizeX;
463 if (child.sizeY > 0.f) h = child.sizeY;
464 if (child.minSizeX > 0.f) w = std::max(w, child.minSizeX);
465 if (child.minSizeY > 0.f) h = std::max(h, child.minSizeY);
466 if (child.maxSizeX > 0.f) w = std::min(w, child.maxSizeX);
467 if (child.maxSizeY > 0.f) h = std::min(h, child.maxSizeY);
468 const float x = child.anchorX * flexW + child.posX - child.anchorX * w;
469 const float y = child.anchorY * flexH + child.posY - child.anchorY * h;
470 ImGui::SetCursorScreenPos(ImVec2(origin.x + x, origin.y + y));

1234 walkFlex(host, tree, n);

1302eve::Result<std::reference_wrapper<ViewportState>> UISystem::ensureViewport(const std::string &key, int w, int h) {
1303 using Return = eve::Result<std::reference_wrapper<ViewportState>>;
1304 if (key.empty() || w <= 0 || h <= 0) {
1305 return Return::failure(eve::Status::failure(
1306 eve::StatusCode::Rejected,
1307 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "viewport key and dimensions must be valid")));
1308 }
1309 auto *gfx = eve::ModuleManager::getInstance<eve::graphics::Graphics>("Graphics");
1310 if (!gfx) {
1311 return Return::failure(eve::Status::failure(
1312 eve::StatusCode::Unsupported,
1313 eve::Diagnostic::error(eve::DiagnosticCode::Unsupported, "viewport requires the Graphics module")));
1314 }
1315 ViewportState &vs = g_viewports[key];
1316 vs.key = key;
1317 if (!vs.canvas || vs.width != w || vs.height != h) {
1318 graphics::Canvas *canvas = gfx->newCanvas(w, h);
1319 if (!canvas) {
1320 return Return::failure(eve::Status::failure(
1321 eve::StatusCode::Failed,
1322 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "Graphics could not create the viewport canvas")));
1323 }
1324 if (vs.textureId && g_backend) g_backend->unregisterTexture(vs.textureId);
1325 vs.textureId = 0;
1326 vs.canvas = canvas;
1327 vs.width = w;
1328 vs.height = h;
1329 if (vs.canvas && g_backend)
1330 vs.textureId = g_backend->registerTexture(vs.canvas->getTexture());
1331 }
1332 return Return::success(std::ref(vs));
1333}

1366void UISystem::render() {
1367 EV_PROFILE_MODULE("ui", "UISystem::render");
1368 if (ecs::current()->getManager<UIHost>() == nullptr) return;
1369
1370 applyThemeToImGui(globalTheme());
1371
1372 struct Item {
1373 UIHost *host;
1374 UIHost::Meta *meta;
1375 UIHost::Tree *tree;
1376 bool render = true;
1377 };
1378 std::vector<Item> items;
1379
1380 graphics::Camera3D *camera = activeCamera();
1381 const ImVec2 display = ImGui::GetIO().DisplaySize;
1382 auto view = ecs::View<UIHost, UIHost::Meta, UIHost::Tree>();
1383 for (auto it = view.begin(); it != view.end(); ++it) {
1384 auto [meta, tree] = *it;
1385 if (!meta->visible) continue;
1386 auto host = UIHost::resolve(meta->entity);
1387 if (!host) continue;
1388 if (!updateWorldAnchor(host->get(), camera, display.x, display.y)) continue;
1389 items.push_back(Item{&host->get(), meta, tree, true});
1390 }
1391
1392 std::stable_sort(items.begin(), items.end(),
1393 [](const Item &a, const Item &b) { return a.meta->layer < b.meta->layer; });
1394
1395 g_stats.nodeCount = 0;
1396 g_stats.measureMs = 0.0;
1397 g_stats.walkMs = 0.0;
1398 for (auto &item : items) {
1399 item.tree->dirty = false;
1400 const auto m0 = std::chrono::steady_clock::now();
1401 measureTree(*item.tree);
1402 const auto m1 = std::chrono::steady_clock::now();
1403 g_stats.nodeCount += int(item.tree->nodes.size());
1404 g_stats.measureMs +=
1405 std::chrono::duration<double, std::milli>(m1 - m0).count();
1406 }
1407
1408 std::vector<WorldAnchorLayoutItem> anchorItems;
1409 anchorItems.reserve(items.size());
1410 const ImGuiStyle &style = ImGui::GetStyle();
1411 for (std::size_t i = 0; i < items.size(); ++i) {
1412 auto &item = items[i];
1413 auto anchor = item.host->worldAnchor();
1414 if (!anchor->enabled) continue;
1415 float width = item.meta->hasSize ? item.meta->sizeX : 0.f;
1416 float height = item.meta->hasSize ? item.meta->sizeY : 0.f;
1417 if (item.meta->percentW > 0.f) width = item.meta->percentW * display.x;
1418 if (item.meta->percentH > 0.f) height = item.meta->percentH * display.y;
1419 if (item.tree->root >= 0) {
1420 const auto &root = item.tree->nodes[std::size_t(item.tree->root)];
1421 if (width <= 0.f) width = root.measuredW + style.WindowPadding.x * 2.f;
1422 if (height <= 0.f) height = root.measuredH + style.WindowPadding.y * 2.f;
1423 }
1424 width *= anchor->scale;
1425 height *= anchor->scale;
1426 anchorItems.push_back({i,
1427 anchor->screenX,
1428 anchor->screenY,
1429 width,
1430 height,
1431 item.meta->pivotX,
1432 item.meta->pivotY,
1433 anchor->depth,
1434 anchor->safeMargin,
1435 anchor->overlapPadding,
1436 anchor->maxDisplacement,
1437 anchor->overlapPriority,
1438 anchor->overlapPolicy == WorldAnchorOverlapPolicy::Avoid});
1439 }
1440 for (const auto &resolved :
1441 resolveWorldAnchorOverlaps(std::move(anchorItems), display.x, display.y)) {
1442 auto &item = items[resolved.stableIndex];
1443 auto anchor = item.host->worldAnchor();
1444 anchor->screenX = resolved.screenX;
1445 anchor->screenY = resolved.screenY;
1446 anchor->displacementX = resolved.displacementX;
1447 anchor->displacementY = resolved.displacementY;
1448 item.render = resolved.render;
1449 if (!resolved.render) anchor->state = WorldAnchorState::Crowded;
1450 }
1451
1452 g_stats.hostCount = int(std::count_if(items.begin(), items.end(),
1453 [](const Item &item) { return item.render; }));
1454 for (auto &item : items) {
1455 if (!item.render) continue;
1456 const auto w0 = std::chrono::steady_clock::now();
1457 if (item.tree->root >= 0) walk(item.host, item.tree, item.tree->root);
1458 const auto w1 = std::chrono::steady_clock::now();
1459 g_stats.walkMs += std::chrono::duration<double, std::milli>(w1 - w0).count();
1460 }

1465 g_pending.clear();

1467 auto resolvedHost = UIHost::resolve(ev.host);
1468 if (!resolvedHost) continue;

1470

1472 UIClick c;
1473 c.hostName = ev.hostName.empty() ? host->meta()->name : ev.hostName;
1474 c.nodeId = ev.nodeId;
1475 g_clicks.push_back(std::move(c));
1476 }
1477
1478 if ((ev.kind == "toggle" || ev.kind == "value" || ev.kind == "text") && !ev.nodeId.empty()) {
1479 UIChange ch;
1480 ch.hostName = ev.hostName.empty() ? host->meta()->name : ev.hostName;
1481 ch.nodeId = ev.nodeId;
1482 ch.kind = ev.kind;
1483 g_changes.push_back(std::move(ch));
1484 }
1485

1487 auto tree = host->tree();
1488 int index = ev.nodeIndex;
1489 if (index < 0 || index >= int(tree->nodes.size()) ||
1490 tree->nodes[size_t(index)].id != ev.nodeId) {
1491 auto target = host->findById(ev.nodeId);
1492 index = target && !tree->nodes.empty() ? int(&target->get() - tree->nodes.data()) : -1;
1493 }
1494
1495 bool first = true;
1496 while (index >= 0 && index < int(tree->nodes.size())) {
1497 const UINode &node = tree->nodes[size_t(index)];
1498 if (!node.visible || !node.enabled) break;
1499 const uint32_t handler = first ? ev.handlerIndex : node.handlerClick;
1500 if (node.mouseFilter != MouseFilter::Ignore && handler != 0) {
1501 const size_t handlerSlot = size_t(handler - 1);
1502 if (handlerSlot < tree->clickHandlers.size() &&
1503 tree->clickHandlers[handlerSlot])
1504 tree->clickHandlers[handlerSlot]();
1505 }
1506 if (node.mouseFilter == MouseFilter::Stop) break;
1507 index = node.parent;
1508 first = false;
1509 }
1510 continue;
1511 }
1512
1513 if (ev.kind == "toggle" && ev.handlerIndex != 0) {
1514 auto t = host->tree();
1515 size_t idx = size_t(ev.handlerIndex - 1);
1516 if (idx < t->toggleHandlers.size() && t->toggleHandlers[idx])
1517 t->toggleHandlers[idx](ev.toggleValue);
1518 continue;
1519 }
1520 if (ev.kind == "value" && ev.handlerIndex != 0) {
1521 auto t = host->tree();

1523 if (idx < t->valueHandlers.size() && t->valueHandlers[idx])
1524 t->valueHandlers[idx](ev.floatValue);

1526 }

1528 auto t = host->tree();

1531 t->textHandlers[idx](ev.textValue);
1532 continue;
