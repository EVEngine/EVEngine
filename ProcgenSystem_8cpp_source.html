144bool ProcgenContext::endTrace(int outputCount) {
145 if (!active_) {
146 error_ = "endTrace: transaction is closed";
147 return false;
148 }
149 if (openTraces_.empty()) {
150 error_ = "endTrace: no stage timer is active";
151 return false;
152 }
153 const auto open = std::move(openTraces_.back());
154 openTraces_.pop_back();
155 const float milliseconds = float(monotonicNanoseconds() - open.startedNanoseconds) / 1000000.f;
156 traces_.push_back({open.name, open.inputCount, std::max(0, outputCount), milliseconds});
157 return true;
