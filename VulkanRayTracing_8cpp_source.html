92Result<void> DeviceAddressBuffer::allocate(vkb::Device& dev, vk::DeviceSize bytes, vk::BufferUsageFlags usage) {
93 if (bytes == 0) return invalidArg("DeviceAddressBuffer.allocate", "size is zero");
94 release();
95 device = &dev;
96
97 vk::BufferCreateInfo ci{};
98 ci.size = bytes;
99 ci.usage = usage | vk::BufferUsageFlagBits::eShaderDeviceAddress;
100 ci.sharingMode = vk::SharingMode::eExclusive;
101 buffer = (*device)->createBuffer(ci, device->allocation_callbacks);
102 size = bytes;
103
104 const auto memreq = (*device)->getBufferMemoryRequirements(buffer);
105
106 vk::MemoryAllocateFlagsInfo flagsInfo{};
107 flagsInfo.flags = vk::MemoryAllocateFlagBits::eDeviceAddress;
108
109 vk::MemoryAllocateInfo allocInfo{};
110 allocInfo.pNext = &flagsInfo;
111 allocInfo.allocationSize = memreq.size;
112 allocInfo.memoryTypeIndex =
113 device->physical_device.findMemoryTypeIndex(memreq.memoryTypeBits, vk::MemoryPropertyFlagBits::eDeviceLocal);
114 memory = (*device)->allocateMemory(allocInfo, device->allocation_callbacks);
115 (*device)->bindBufferMemory(buffer, memory, 0);
116
117 vk::BufferDeviceAddressInfo addrInfo{};
118 addrInfo.buffer = buffer;
119 address = (*device)->getBufferAddress(addrInfo);
120 if (address == 0) {
121 release();
122 return failed("DeviceAddressBuffer.allocate", "getBufferAddress returned 0");
123 }
124 return Result<void>::success();
125}

127Result<void> DeviceAddressBuffer::upload(vk::CommandPool pool, vk::Queue queue, const void* data,
128 vk::DeviceSize bytes) {
129 if (!device || !buffer) return failed("DeviceAddressBuffer.upload", "buffer not allocated");
130 if (!data || bytes == 0 || bytes > size) return invalidArg("DeviceAddressBuffer.upload", "invalid host data");
131
132 using pfb = vk::MemoryPropertyFlagBits;
133 vkb::GenericBuffer staging(*device, vk::BufferUsageFlagBits::eTransferSrc, bytes,
134 pfb::eHostVisible | pfb::eHostCoherent);
135 staging.updateLocal(vkb::FrameSlot::gpuIdle(), data, bytes);
136
137 auto cmd = (*device)->allocateCommandBuffers({pool, vk::CommandBufferLevel::ePrimary, 1}).front();
138 cmd.begin({vk::CommandBufferUsageFlagBits::eOneTimeSubmit});
139 vk::BufferCopy copy{};
140 copy.size = bytes;
141 cmd.copyBuffer(staging.buffer, buffer, copy);
142 cmd.end();
143 vk::SubmitInfo submit{};
144 submit.commandBufferCount = 1;
145 submit.pCommandBuffers = &cmd;
146 queue.submit(submit, vk::Fence{});
147 queue.waitIdle();
148 (*device)->freeCommandBuffers(pool, cmd);
149 return Result<void>::success();
150}

179Result<void> VulkanRayTracing::ensureAttached() {
180 auto* base = eve::ModuleManager::getInstance<eve::graphics::Graphics>("Graphics");
181 if (!base) base = eve::graphics::Graphics::create();
182 auto* vg = dynamic_cast<eve::graphics::vulkan::Graphics*>(base);
183 if (!vg || !vg->supportsRayTracing()) {
184 if (vg) caps_ = vg->rayTracingCaps();
185 return unsupported("ensureAttached");
186 }
187 vkb::Device* current = &vg->getDevice();
188 // Rebind whenever Graphics rebuilt the logical device (pointer identity).
189 if (device_ == current && caps_.rayTracingAvailable()) return Result<void>::success();
190 attachDevice(current, vg->rayTracingCaps(), vg->getUploadPool(),
191 vg->getDevice().getQueue(vkb::QueueType::graphics));
192 if (!isAvailable()) return unsupported("ensureAttached");
193 return Result<void>::success();
194}

260Result<uint32_t> VulkanRayTracing::addTriangleMesh(const float* positionsXYZ, int vertexCount, const uint32_t* indices,
261 int indexCount, const glm::mat4& transform) {
262 if (auto attached = ensureAttached(); !attached.ok()) return Result<uint32_t>::failure(attached.status());
263 if (!positionsXYZ || vertexCount < 3 || !indices || indexCount < 3 || (indexCount % 3) != 0)
264 return Result<uint32_t>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
265 "addTriangleMesh: invalid geometry", "graphics.raytracing"));
266
267 auto mesh = std::make_unique<TriangleMeshRecord>();
268 mesh->vertexCount = uint32_t(vertexCount);
269 mesh->indexCount = uint32_t(indexCount);
270 mesh->transform = transform;
271
272 const vk::DeviceSize vBytes = vk::DeviceSize(vertexCount) * 3u * sizeof(float);
273 const vk::DeviceSize iBytes = vk::DeviceSize(indexCount) * sizeof(uint32_t);
274 const auto usage = vk::BufferUsageFlagBits::eAccelerationStructureBuildInputReadOnlyKHR |
275 vk::BufferUsageFlagBits::eStorageBuffer | vk::BufferUsageFlagBits::eTransferDst;
276
277 if (auto r = mesh->vertexBuffer.allocate(*device_, vBytes, usage); !r.ok())
278 return Result<uint32_t>::failure(r.status());
279 if (auto r = mesh->indexBuffer.allocate(*device_, iBytes, usage); !r.ok())
280 return Result<uint32_t>::failure(r.status());
281 if (auto r = mesh->vertexBuffer.upload(uploadPool_, graphicsQueue_, positionsXYZ, vBytes); !r.ok())
282 return Result<uint32_t>::failure(r.status());
283 if (auto r = mesh->indexBuffer.upload(uploadPool_, graphicsQueue_, indices, iBytes); !r.ok())
284 return Result<uint32_t>::failure(r.status());
285 if (auto r = buildBlas(*mesh); !r.ok()) return Result<uint32_t>::failure(r.status());
286
287 const uint32_t id = uint32_t(meshes_.size());
288 meshes_.push_back(std::move(mesh));
289 return Result<uint32_t>::success(id);
290}

292Result<uint32_t> VulkanRayTracing::addMesh(Mesh* mesh, const glm::mat4& transform) {
293 if (auto attached = ensureAttached(); !attached.ok()) return Result<uint32_t>::failure(attached.status());
294 if (!mesh || !mesh->gpuHandle)
295 return Result<uint32_t>::failure(Diagnostic::error(
296 DiagnosticCode::InvalidArgument, "addMesh: null mesh or missing GPU handle", "graphics.raytracing"));
297 auto* gpu = static_cast<eve::graphics::vulkan::GpuMesh*>(mesh->gpuHandle);
298 if (gpu->vertexCount < 3 || gpu->indexCount < 3 || (gpu->indexCount % 3u) != 0)
299 return Result<uint32_t>::failure(
300 Diagnostic::error(DiagnosticCode::InvalidArgument, "addMesh: invalid geometry", "graphics.raytracing"));
301
302 auto& vb = eve::graphics::vulkan::meshDrawVertices(*gpu);
303 if (!vb.buffer)
304 return Result<uint32_t>::failure(
305 Diagnostic::error(DiagnosticCode::Failed, "addMesh: empty vertex buffer", "graphics.raytracing"));
306
307 std::vector<float> positions(size_t(gpu->vertexCount) * 3u);
308 {
309 void* mapped = vb.map();
310 const auto* verts = static_cast<const eve::graphics::vulkan::MeshVertex*>(mapped);
311 for (uint32_t i = 0; i < gpu->vertexCount; ++i) {
312 positions[size_t(i) * 3u + 0u] = verts[i].pos.x;
313 positions[size_t(i) * 3u + 1u] = verts[i].pos.y;
314 positions[size_t(i) * 3u + 2u] = verts[i].pos.z;
315 }
316 vb.unmap();
317 }
318
319 std::vector<uint32_t> indices(gpu->indexCount);
320 if (!gpu->cpuIndices.empty() && gpu->cpuIndices.size() >= gpu->indexCount) {
321 std::copy_n(gpu->cpuIndices.begin(), gpu->indexCount, indices.begin());
322 } else {
323 auto& ib = eve::graphics::vulkan::meshDrawIndices(*gpu);
324 if (!ib.buffer)
325 return Result<uint32_t>::failure(
326 Diagnostic::error(DiagnosticCode::Failed, "addMesh: empty index buffer", "graphics.raytracing"));
327 void* mapped = ib.map();
328 if (gpu->indexType == vk::IndexType::eUint16) {
329 const auto* src = static_cast<const uint16_t*>(mapped);
330 for (uint32_t i = 0; i < gpu->indexCount; ++i) indices[i] = src[i];
331 } else {
332 std::memcpy(indices.data(), mapped, size_t(gpu->indexCount) * sizeof(uint32_t));
333 }
334 ib.unmap();
335 }
336
337 return addTriangleMesh(positions.data(), int(gpu->vertexCount), indices.data(), int(gpu->indexCount), transform);
338}

340Result<void> VulkanRayTracing::buildBlas(TriangleMeshRecord& mesh) {
341 vk::AccelerationStructureGeometryTrianglesDataKHR triangles{};
342 triangles.vertexFormat = vk::Format::eR32G32B32Sfloat;
343 triangles.vertexData.deviceAddress = mesh.vertexBuffer.address;
344 triangles.vertexStride = sizeof(float) * 3;
345 triangles.maxVertex = mesh.vertexCount - 1;
346 triangles.indexType = vk::IndexType::eUint32;
347 triangles.indexData.deviceAddress = mesh.indexBuffer.address;
348
349 vk::AccelerationStructureGeometryKHR geometry{};
350 geometry.geometryType = vk::GeometryTypeKHR::eTriangles;
351 geometry.flags = vk::GeometryFlagBitsKHR::eOpaque;
352 geometry.geometry.triangles = triangles;
353
354 vk::AccelerationStructureBuildGeometryInfoKHR buildInfo{};
355 buildInfo.type = vk::AccelerationStructureTypeKHR::eBottomLevel;
356 buildInfo.flags = vk::BuildAccelerationStructureFlagBitsKHR::ePreferFastTrace;
357 buildInfo.mode = vk::BuildAccelerationStructureModeKHR::eBuild;
358 buildInfo.geometryCount = 1;
359 buildInfo.pGeometries = &geometry;
360
361 const uint32_t primitiveCount = mesh.indexCount / 3;
362 const auto sizes = (*device_)->getAccelerationStructureBuildSizesKHR(vk::AccelerationStructureBuildTypeKHR::eDevice,
363 buildInfo, primitiveCount);
364
365 auto storageResult = mesh.blas.storage.allocate(
366 *device_, sizes.accelerationStructureSize,
367 vk::BufferUsageFlagBits::eAccelerationStructureStorageKHR | vk::BufferUsageFlagBits::eShaderDeviceAddress);
368 if (!storageResult.ok()) return storageResult;
369
370 vk::AccelerationStructureCreateInfoKHR createInfo{};
371 createInfo.buffer = mesh.blas.storage.buffer;
372 createInfo.size = sizes.accelerationStructureSize;
373 createInfo.type = vk::AccelerationStructureTypeKHR::eBottomLevel;
374 mesh.blas.handle = (*device_)->createAccelerationStructureKHR(createInfo, device_->allocation_callbacks);
375 mesh.blas.device = device_;
376
377 vk::AccelerationStructureDeviceAddressInfoKHR addrInfo{};
378 addrInfo.accelerationStructure = mesh.blas.handle;
379 mesh.blas.deviceAddress = (*device_)->getAccelerationStructureAddressKHR(addrInfo);
380
381 DeviceAddressBuffer scratch;
382 if (auto r =
383 scratch.allocate(*device_, sizes.buildScratchSize,
384 vk::BufferUsageFlagBits::eStorageBuffer | vk::BufferUsageFlagBits::eShaderDeviceAddress);
385 !r.ok())
386 return r;
387
388 buildInfo.dstAccelerationStructure = mesh.blas.handle;
389 buildInfo.scratchData.deviceAddress = scratch.address;
390
391 vk::AccelerationStructureBuildRangeInfoKHR range{};
392 range.primitiveCount = primitiveCount;
393 const vk::AccelerationStructureBuildRangeInfoKHR* ranges = &range;
394
395 auto cmd = (*device_)->allocateCommandBuffers({uploadPool_, vk::CommandBufferLevel::ePrimary, 1}).front();
396 cmd.begin({vk::CommandBufferUsageFlagBits::eOneTimeSubmit});
397 cmd.buildAccelerationStructuresKHR(buildInfo, ranges);
398 cmd.end();
399 vk::SubmitInfo submit{};
400 submit.commandBufferCount = 1;
401 submit.pCommandBuffers = &cmd;
402 graphicsQueue_.submit(submit, vk::Fence{});
403 graphicsQueue_.waitIdle();
404 (*device_)->freeCommandBuffers(uploadPool_, cmd);
405 return Result<void>::success();
406}

409 tlas_.release();
410 if (meshes_.empty()) return Result<void>::success();
411
412 std::vector<vk::AccelerationStructureInstanceKHR> instances;
413 instances.reserve(meshes_.size());

503Result<void> VulkanRayTracing::ensurePipeline() {
504 if (pipeline_) return Result<void>::success();
505 if (!isAvailable()) return unsupported("ensurePipeline");
506
507 auto makeModule = [&](const uint32_t* words, size_t count) -> vk::ShaderModule {
508 vk::ShaderModuleCreateInfo ci{};
509 ci.codeSize = count * sizeof(uint32_t);
510 ci.pCode = words;
511 return (*device_)->createShaderModule(ci, device_->allocation_callbacks);
512 };
513 raygenModule_ = makeModule(rt_reflection_rgen_spv, rt_reflection_rgen_spv_count);
514 missModule_ = makeModule(rt_reflection_rmiss_spv, rt_reflection_rmiss_spv_count);
515 closestHitModule_ = makeModule(rt_reflection_rchit_spv, rt_reflection_rchit_spv_count);
516
517 vk::DescriptorSetLayoutBinding bindings[5]{};
518 bindings[0].binding = 0;
519 bindings[0].descriptorType = vk::DescriptorType::eAccelerationStructureKHR;
520 bindings[0].descriptorCount = 1;
521 bindings[0].stageFlags = vk::ShaderStageFlagBits::eRaygenKHR;
522 bindings[1].binding = 1;
523 bindings[1].descriptorType = vk::DescriptorType::eStorageImage;
524 bindings[1].descriptorCount = 1;

602Result<void> VulkanRayTracing::createShaderBindingTable() {
603 const uint32_t handleSize = caps_.shaderGroupHandleSize;
604 const uint32_t handleAlignment = caps_.shaderGroupHandleAlignment ? caps_.shaderGroupHandleAlignment : handleSize;
605 const uint32_t baseAlignment = caps_.shaderGroupBaseAlignment ? caps_.shaderGroupBaseAlignment : handleAlignment;
606 const uint32_t handleSizeAligned = alignedSize(handleSize, handleAlignment);
607 const uint32_t groupCount = 3;
608 const uint32_t sbtSize = groupCount * handleSizeAligned;
609
610 std::vector<uint8_t> handles(sbtSize);
611 auto result = (*device_)->getRayTracingShaderGroupHandlesKHR(pipeline_, 0, groupCount, sbtSize, handles.data());
612 if (result != vk::Result::eSuccess)
613 return failed("createShaderBindingTable", "getRayTracingShaderGroupHandlesKHR failed");
614
615 // Pack groups with base alignment between regions: raygen | miss | hit
616 const uint32_t raygenSize = alignedSize(handleSizeAligned, baseAlignment);
617 const uint32_t missSize = alignedSize(handleSizeAligned, baseAlignment);
618 const uint32_t hitSize = alignedSize(handleSizeAligned, baseAlignment);
619 const uint32_t total = raygenSize + missSize + hitSize;
620
621 std::vector<uint8_t> sbt(total, 0);
622 std::memcpy(sbt.data(), handles.data(), handleSize);
623 std::memcpy(sbt.data() + raygenSize, handles.data() + handleSizeAligned, handleSize);
624 std::memcpy(sbt.data() + raygenSize + missSize, handles.data() + 2 * handleSizeAligned, handleSize);
625
626 if (auto r = sbtBuffer_.allocate(*device_, total,
627 vk::BufferUsageFlagBits::eShaderBindingTableKHR |
628 vk::BufferUsageFlagBits::eShaderDeviceAddress |
629 vk::BufferUsageFlagBits::eTransferDst);
630 !r.ok())
631 return r;
632 if (auto r = sbtBuffer_.upload(uploadPool_, graphicsQueue_, sbt.data(), total); !r.ok()) return r;
633
634 raygenRegion_.deviceAddress = sbtBuffer_.address;
635 raygenRegion_.stride = raygenSize;
636 raygenRegion_.size = raygenSize;
637
638 missRegion_.deviceAddress = sbtBuffer_.address + raygenSize;
639 missRegion_.stride = handleSizeAligned;
640 missRegion_.size = missSize;
641
642 hitRegion_.deviceAddress = sbtBuffer_.address + raygenSize + missSize;
643 hitRegion_.stride = handleSizeAligned;
644 hitRegion_.size = hitSize;
645
646 callableRegion_ = vk::StridedDeviceAddressRegionKHR{};
647 return Result<void>::success();
648}

753Result<void> VulkanRayTracing::applyReflections(Graphics* gfx, Texture* sceneColor, Texture* hwDepth,
754 Texture* worldNormal, Canvas* dest, const glm::mat4& invViewProj,
755 const glm::mat4& viewProj, const glm::vec3& eyeWorld) {
756 if (auto attached = ensureAttached(); !attached.ok()) return attached;
757 if (!gfx || !sceneColor || !hwDepth || !worldNormal || !dest)
758 return invalidArg("applyReflections", "null argument");
759 if (meshes_.empty() || !tlas_.handle) {
760 // Empty scene: no usable RT output — callers must fall through to SSR.
761 return Result<void>::success(Status::success(StatusCode::NoOp));
762 }
763
764 auto* vkGfx = dynamic_cast<eve::graphics::vulkan::Graphics*>(gfx);
765 if (!vkGfx) return failed("applyReflections", "requires Vulkan Graphics backend");
766
767 if (auto r = ensurePipeline(); !r.ok()) return r;
768
769 auto textureView = [](Texture* tex) -> vk::ImageView {
770 if (!tex || !tex->gpuHandle) return {};
771 auto* img = static_cast<eve::graphics::vulkan::GpuTexture*>(tex->gpuHandle);
772 return img->imageView();
773 };
774
775 Texture* destTex = dest->getTexture();
776 if (!destTex) return failed("applyReflections", "destination canvas has no texture");
777 auto* destCanvas = dynamic_cast<eve::graphics::vulkan::OffscreenCanvas*>(dest);
778 if (!destCanvas) return failed("applyReflections", "destination must be an OffscreenCanvas");
779
780 const uint32_t width = uint32_t(dest->getWidth());
781 const uint32_t height = uint32_t(dest->getHeight());
782 if (auto r = ensureOutputImage(width, height); !r.ok()) return r;
783
784 const vk::ImageView outView = outputImage_.imageView();
785 const vk::ImageView sceneView = textureView(sceneColor);
786 const vk::ImageView depthView = textureView(hwDepth);
787 const vk::ImageView normalView = textureView(worldNormal);
788 if (!outView || !sceneView || !depthView || !normalView)
789 return failed("applyReflections", "missing texture image view");
790
791 if (auto r = ensureDescriptorSets(outView, sceneView, depthView, normalView); !r.ok()) return r;
792
793 PushConstants push{};
794 push.invViewProj = invViewProj;
795 push.viewProj = viewProj;
796 push.eye = glm::vec4(eyeWorld, 0.f);
797
798 auto record = [&](vk::CommandBuffer cmd) {
799 outputImage_.setLayout(cmd, vk::ImageLayout::eGeneral);
800
801 vk::MemoryBarrier memBarrier{};
802 memBarrier.srcAccessMask =
803 vk::AccessFlagBits::eColorAttachmentWrite | vk::AccessFlagBits::eDepthStencilAttachmentWrite;
804 memBarrier.dstAccessMask = vk::AccessFlagBits::eShaderRead | vk::AccessFlagBits::eShaderWrite;
805 cmd.pipelineBarrier(
806 vk::PipelineStageFlagBits::eColorAttachmentOutput | vk::PipelineStageFlagBits::eLateFragmentTests,
807 vk::PipelineStageFlagBits::eRayTracingShaderKHR, {}, memBarrier, {}, {});
808
809 cmd.bindPipeline(vk::PipelineBindPoint::eRayTracingKHR, pipeline_);
810 cmd.bindDescriptorSets(vk::PipelineBindPoint::eRayTracingKHR, pipelineLayout_, 0, descriptorSet_, {});
811 cmd.pushConstants(pipelineLayout_, vk::ShaderStageFlagBits::eRaygenKHR, 0, sizeof(PushConstants), &push);
812 cmd.traceRaysKHR(raygenRegion_, missRegion_, hitRegion_, callableRegion_, width, height, 1);
813
814 // Copy RT storage result into the reflection canvas for composite sampling.
815 outputImage_.setLayout(cmd, vk::ImageLayout::eTransferSrcOptimal);
816 destCanvas->colorImage().setLayout(cmd, vk::ImageLayout::eTransferDstOptimal);
817 vk::ImageCopy region{};
818 region.srcSubresource = {vk::ImageAspectFlagBits::eColor, 0, 0, 1};
819 region.dstSubresource = {vk::ImageAspectFlagBits::eColor, 0, 0, 1};
820 region.extent = vk::Extent3D{width, height, 1};
821 cmd.copyImage(outputImage_.image(), vk::ImageLayout::eTransferSrcOptimal, destCanvas->colorImage().image(),
822 vk::ImageLayout::eTransferDstOptimal, region);
823 destCanvas->colorImage().setLayout(cmd, vk::ImageLayout::eShaderReadOnlyOptimal);
824 };
825
826 if (vkGfx->canRecordPostSceneGpuWork()) {
827 record(vkGfx->postSceneCommandBuffer());
828 } else {
829 vkGfx->waitForSharedGpuResources();
830 vkb::executeImmediately(device_->instance, uploadPool_, graphicsQueue_, record);
831 }
832 return Result<void>::success();
833}
