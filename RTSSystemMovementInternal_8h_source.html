72// Service contact is based on occupied geometry, not exact overlap of agent centers.
73// The radius is derived from canonical node membership/building footprint and is
74// recomputed synchronously; no second range state or persistent format is created.
75inline float interactionArrivalRadius(Unit& unit, const OrderRecord& order) {
76 float radius = unit.motion()->arrivalRadius;
77 if (order.kind == OrderKind::Gather) {
78 if (auto* node = dynamic_cast<ResourceNode*>(ecs::try_get(order.targetEntity))) {
79 const float workers = static_cast<float>(std::max<std::size_t>(1, node->harvest()->workers.size()));
80 radius = std::max(radius, unit.crowd()->radius * 2.0f * std::sqrt(workers));
81 }
82 } else if (isInteractionOrder(order.kind)) {
83 if (auto* building = dynamic_cast<Building*>(ecs::try_get(order.targetEntity)))
84 radius = std::max(radius, building->dropoff()->radius + unit.crowd()->radius);
85 }
86 return radius;
87}

100inline WorldPosition movementGoal(Unit& unit, const OrderRecord& order) {
101 WorldPosition goal = order.target;
102 if (order.kind == OrderKind::Attack || order.kind == OrderKind::Resupply || order.kind == OrderKind::SupplyRelay) {
103 if (auto target = entityPosition(order.targetEntity)) goal = *target;
104 if (order.kind == OrderKind::SupplyRelay && unit.supply()->rendezvousActive)
105 goal = unit.supply()->rendezvousPoint;
106 } else if (order.kind == OrderKind::Escort && unit.tactics()->guardSet) {
107 goal = {unit.tactics()->guardX, unit.tactics()->guardY};
108 } else if (order.kind == OrderKind::Patrol && unit.navigation()->patrolInitialized &&
109 !unit.navigation()->patrolTowardTarget) {
110 goal = unit.navigation()->patrolOrigin;
111 }
112 return goal;
113}
