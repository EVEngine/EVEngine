15bool boundedValue(const eve::Value& value) {
16 std::vector<std::pair<const eve::Value*, std::size_t>> pending{{&value, 0}};
17 std::size_t elements = 0, bytes = 0;
18 while (!pending.empty()) {
19 const auto [entry, depth] = pending.back();
20 pending.pop_back();
21 if (depth > 64 || ++elements > 262144) return false;
22 if (const auto* text = entry->getIf<std::string>()) bytes += text->size();
23 if (const auto* array = entry->getIf<eve::Value::Array>()) {
24 if (array->size() > 262144 - elements - pending.size()) return false;
25 for (const auto& child : *array) pending.emplace_back(&child, depth + 1);
26 }
27 if (const auto* object = entry->getIf<eve::Value::Object>()) {
28 if (object->size() > 262144 - elements - pending.size()) return false;
29 for (const auto& [key, child] : *object) {
30 bytes += key.size();
31 pending.emplace_back(&child, depth + 1);
32 }
33 }
34 if (bytes > maximumJsonBytes) return false;
35 }
36 return true;
37}

330eve::Result<std::shared_ptr<State>> resolve(RoadPresetDraft draft) {
331 if (draft.types.size() > 256 || draft.bindings.size() > 4096 || draft.source.edges().size() > 4096 ||
332 draft.source.nodes().size() > 4096 || draft.metadata.size() > 4096)
333 return fail<std::shared_ptr<State>>("author graph exceeds preset document budget");
334 for (const auto& edge : draft.source.edges())
335 if (edge.controlPoints.size() > 4096)
336 return fail<std::shared_ptr<State>>("source marker count exceeds the persistent codec budget");
337 auto valid = draft.source.validate();
338 if (!valid.ok()) return eve::Result<std::shared_ptr<State>>::failure(valid.status());
339 std::map<std::string, const RoadTypePreset*> definitions;
340 for (const auto& type : draft.types) {
341 if (type.id.empty() || type.id.size() > 256 || type.parentId.size() > 256 ||
342 !definitions.emplace(type.id, &type).second)
343 return fail<std::shared_ptr<State>>("invalid or duplicate preset identity");
344 }
345 std::map<std::string, RoadStyle> styles;
346 std::map<std::string, RoadPointOverrides> pointDefaults;
347 std::set<std::string> visiting;
348 std::function<eve::Result<RoadStyle>(const std::string&)> visit =
349 [&](const std::string& id) -> eve::Result<RoadStyle> {
350 if (const auto found = styles.find(id); found != styles.end())
351 return eve::Result<RoadStyle>::success(found->second);
352 const auto found = definitions.find(id);
353 if (found == definitions.end() || !visiting.insert(id).second)
354 return fail<RoadStyle>("missing preset parent or inheritance cycle");
355 RoadStyle style;
356 if (!found->second->parentId.empty()) {
357 auto parent = visit(found->second->parentId);
358 if (!parent.ok()) return parent;
359 style = std::move(parent).takeValue();
360 }
361 apply(style, found->second->overrides);
362 RoadPointOverrides points;
363 if (!found->second->parentId.empty()) points = pointDefaults.at(found->second->parentId);
364 mergePoints(points, found->second->pointDefaults);
365 // Validate even unused presets through the same native road style boundary.
366 auto probe = RoadNetwork::makeStraight(36.f, 2);
367 if (!probe.ok()) return eve::Result<RoadStyle>::failure(probe.status());
368 auto checked = probe.value().setEdgeStyle(probe.value().edges()[0].id, style);
369 if (!checked.ok()) return eve::Result<RoadStyle>::failure(checked.status());
370 auto probePoints = probe.value().edges()[0].controlPoints;
371 for (auto& point : probePoints) applyPoint(point, points);
372 auto checkedPoints = probe.value().setEdgeControlPoints(probe.value().edges()[0].id, std::move(probePoints));
373 if (!checkedPoints.ok()) return eve::Result<RoadStyle>::failure(checkedPoints.status());
374 pointDefaults.emplace(id, points);
375 visiting.erase(id);
376 styles.emplace(id, style);
377 return eve::Result<RoadStyle>::success(std::move(style));
378 };
379 for (const auto& [id, definition] : definitions) {
380 (void)definition;
381 auto checked = visit(id);
382 if (!checked.ok()) return eve::Result<std::shared_ptr<State>>::failure(checked.status());
383 }
384 auto edges = draft.source.edges();
385 std::map<std::uint32_t, std::size_t> edgeIndices;
386 for (std::size_t i = 0; i < edges.size(); ++i) edgeIndices.emplace(edges[i].id, i);
387 std::set<std::uint32_t> bound;
388 for (const auto& binding : draft.bindings) {
389 if (!bound.insert(binding.edgeId).second || !styles.contains(binding.typeId))
390 return fail<std::shared_ptr<State>>("duplicate binding or missing preset identity");
391 auto style = styles.at(binding.typeId);
392 apply(style, binding.overrides);
393 if (binding.facilities.size() > 64 || binding.markers.size() > 4096)
394 return fail<std::shared_ptr<State>>("binding override budget exceeded");
395 std::set<std::uint32_t> facilityIds;
396 for (const auto& patch : binding.facilities) {
397 auto found = std::find_if(style.facilities.begin(), style.facilities.end(),
398 [&](const auto& facility) { return facility.id == patch.id; });
399 if (found == style.facilities.end() || !facilityIds.insert(patch.id).second)
400 return fail<std::shared_ptr<State>>("missing or duplicate facility override identity");
401 applyFacility(*found, patch);
402 }
403 if (!edgeIndices.contains(binding.edgeId))
404 return fail<std::shared_ptr<State>>("binding refers to a missing source edge");
405 auto& edge = edges[edgeIndices.at(binding.edgeId)];
406 auto points = edge.controlPoints;
407 auto defaults = pointDefaults.at(binding.typeId);
408 mergePoints(defaults, binding.pointDefaults);
409 for (auto& point : points) applyPoint(point, defaults);
410 std::set<std::uint32_t> markerIds;
411 for (const auto& marker : binding.markers) {
412 if (marker.markerIndex >= points.size() || !markerIds.insert(marker.markerIndex).second)
413 return fail<std::shared_ptr<State>>("missing or duplicate marker override index");
414 applyPoint(points[marker.markerIndex], marker.overrides);
415 }
416 edge.controlPoints = std::move(points);
417 edge.style = std::move(style);
418 }
419 std::set<std::uint32_t> metadataIds;
420 for (const auto& item : draft.metadata) {
421 auto edge = draft.source.edgeResult(item.edgeId);
422 if (!edge.ok()) return eve::Result<std::shared_ptr<State>>::failure(edge.status());
423 if (item.name.size() > 256 || !metadataIds.insert(item.edgeId).second)
424 return fail<std::shared_ptr<State>>("invalid author metadata");
425 }
426 // Construct complete final records before checking cross-edge seams; sequential setters
427 // would reject a valid combined width/profile update against intermediate geometry.
428 RoadNetwork resolved;
429 for (const auto& node : draft.source.nodes()) {
430 auto restored = resolved.restoreNode(node);
431 if (!restored.ok()) return eve::Result<std::shared_ptr<State>>::failure(restored.status());
432 }
433 for (auto& edge : edges) {
434 auto restored = resolved.restoreEdge(std::move(edge));
435 if (!restored.ok()) return eve::Result<std::shared_ptr<State>>::failure(restored.status());
436 }
437 for (const auto& link : draft.source.laneLinks()) {
438 auto restored = resolved.addLaneLink(link);
439 if (!restored.ok()) return eve::Result<std::shared_ptr<State>>::failure(restored.status());
440 }
441 for (const auto& link : draft.source.blockedLaneLinks()) {
442 auto restored = resolved.blockLaneLink(link);
443 if (!restored.ok()) return eve::Result<std::shared_ptr<State>>::failure(restored.status());
444 }
445 auto checked = resolved.validate();
446 if (!checked.ok()) return eve::Result<std::shared_ptr<State>>::failure(checked.status());
447 std::sort(draft.types.begin(), draft.types.end(), [](const auto& a, const auto& b) { return a.id < b.id; });
448 std::sort(draft.metadata.begin(), draft.metadata.end(),
449 [](const auto& a, const auto& b) { return a.edgeId < b.edgeId; });
450 std::sort(draft.bindings.begin(), draft.bindings.end(),
451 [](const auto& a, const auto& b) { return a.edgeId < b.edgeId; });
452 for (auto& binding : draft.bindings) {
453 std::sort(binding.markers.begin(), binding.markers.end(),
454 [](const auto& a, const auto& b) { return a.markerIndex < b.markerIndex; });
455 std::sort(binding.facilities.begin(), binding.facilities.end(),
456 [](const auto& a, const auto& b) { return a.id < b.id; });
457 }
458 return eve::Result<std::shared_ptr<State>>::success(
459 std::make_shared<State>(State{std::move(draft), std::move(resolved)}));
460}
461V encodeDraft(const RoadPresetDraft& draft) {
462 auto source = draft.source;
463 RoadNetworkEditTarget target("road:preset-codec", source);
464 V::Array types, bindings;
465 for (const auto& type : draft.types)
466 types.emplace_back(V::Object{{"id", type.id},
467 {"parentId", type.parentId},
468 {"overrides", encodePatch(type.overrides)},
469 {"pointDefaults", encodePoints(type.pointDefaults)}});
470 for (const auto& binding : draft.bindings) bindings.push_back(encodeBinding(binding));
471 V::Array metadata;
472 for (const auto& item : draft.metadata)
473 metadata.emplace_back(V::Object{
474 {"edgeId", static_cast<std::int64_t>(item.edgeId)}, {"name", item.name}, {"locked", item.locked}});
475 return V::Object{{"source", target.snapshotValue()},
476 {"types", std::move(types)},
477 {"bindings", std::move(bindings)},
478 {"metadata", std::move(metadata)}};
479}
480eve::Result<RoadPresetDraft> decodeDraft(const V& value) {
481 const auto* source = detail::field(value, "source");
482 const auto* types = detail::field(value, "types");
483 const auto* bindings = detail::field(value, "bindings");
484 const auto* typeArray = types ? types->getIf<V::Array>() : nullptr;
485 const auto* bindingArray = bindings ? bindings->getIf<V::Array>() : nullptr;
486 if (!source || !typeArray || !bindingArray || typeArray->size() > 256 || bindingArray->size() > 4096)
487 return fail<RoadPresetDraft>("invalid preset document frame");
488 RoadPresetDraft draft;
489 RoadNetworkEditTarget target("road:preset-codec", draft.source);
490 auto loaded = target.loadSnapshot(*source);
491 if (!loaded.ok()) return fail<RoadPresetDraft>("invalid author source snapshot");
492 for (const auto& encoded : *typeArray) {
493 const auto* id = detail::field(encoded, "id");
494 const auto* parent = detail::field(encoded, "parentId");
495 const auto* patch = detail::field(encoded, "overrides");
496 if (!id || !id->getIf<std::string>() || !parent || !parent->getIf<std::string>() || !patch)
497 return fail<RoadPresetDraft>("invalid type record");
498 auto decoded = decodePatch(*patch);
499 if (!decoded.ok()) return eve::Result<RoadPresetDraft>::failure(decoded.status());
500 draft.types.push_back(
501 {*id->getIf<std::string>(), *parent->getIf<std::string>(), std::move(decoded).takeValue()});
502 }
503 for (const auto& encoded : *bindingArray) {
504 std::uint32_t edgeId = 0;
505 const auto* id = detail::field(encoded, "typeId");
506 const auto* patch = detail::field(encoded, "overrides");
507 if (!detail::readId(encoded, "edgeId", edgeId) || !id || !id->getIf<std::string>() || !patch)
508 return fail<RoadPresetDraft>("invalid binding record");
509 auto decoded = decodePatch(*patch);
510 if (!decoded.ok()) return eve::Result<RoadPresetDraft>::failure(decoded.status());
511 draft.bindings.push_back({edgeId, *id->getIf<std::string>(), std::move(decoded).takeValue()});
512 }
513 for (std::size_t i = 0; i < typeArray->size(); ++i) {
514 auto points = decodePoints(detail::field((*typeArray)[i], "pointDefaults"));
515 if (!points.ok()) return eve::Result<RoadPresetDraft>::failure(points.status());
516 draft.types[i].pointDefaults = std::move(points).takeValue();
517 }
518 for (std::size_t i = 0; i < bindingArray->size(); ++i) {
519 const auto& encoded = (*bindingArray)[i];
520 auto& binding = draft.bindings[i];
521 auto points = decodePoints(detail::field(encoded, "pointDefaults"));
522 if (!points.ok()) return eve::Result<RoadPresetDraft>::failure(points.status());
523 binding.pointDefaults = std::move(points).takeValue();
524 if (const auto* markers = detail::field(encoded, "markers")) {
525 const auto* array = markers->getIf<V::Array>();
526 if (!array || array->size() > 4096) return fail<RoadPresetDraft>("invalid marker override array");
527 for (const auto& marker : *array) {
528 int index = 0;
529 if (!detail::readSmallInt(marker, "markerIndex", 0, 4095, index))
530 return fail<RoadPresetDraft>("invalid marker override index");
531 auto patch = decodePoints(detail::field(marker, "overrides"));
532 if (!patch.ok()) return eve::Result<RoadPresetDraft>::failure(patch.status());
533 binding.markers.push_back({static_cast<std::uint32_t>(index), std::move(patch).takeValue()});
534 }
535 }
536 if (const auto* facilities = detail::field(encoded, "facilities")) {
537 const auto* array = facilities->getIf<V::Array>();
538 if (!array || array->size() > 64) return fail<RoadPresetDraft>("invalid facility override array");
539 for (const auto& facility : *array) {
540 auto patch = decodeFacility(facility);
541 if (!patch.ok()) return eve::Result<RoadPresetDraft>::failure(patch.status());
542 binding.facilities.push_back(std::move(patch).takeValue());
543 }
544 }
545 }
546 if (const auto* metadata = detail::field(value, "metadata")) {
547 const auto* array = metadata->getIf<V::Array>();
548 if (!array || array->size() > 4096) return fail<RoadPresetDraft>("invalid author metadata array");
549 for (const auto& entry : *array) {
550 RoadAuthorMetadata item;
551 const auto* name = detail::field(entry, "name");
552 if (!detail::readId(entry, "edgeId", item.edgeId) || !name || !name->getIf<std::string>() ||
553 !detail::readOptionalBool(entry, "locked", item.locked))
554 return fail<RoadPresetDraft>("invalid author metadata record");
555 item.name = *name->getIf<std::string>();
556 draft.metadata.push_back(std::move(item));
557 }
558 }
559 return eve::Result<RoadPresetDraft>::success(std::move(draft));
560}

591eve::Result<PreparedRoadPresetEdit> RoadPresetDocument::prepare(RoadPresetDraft draft) const {
592 if (impl_->cursor + 1 >= maximumHistory) return fail<PreparedRoadPresetEdit>("preset undo history budget exceeded");
593 const auto& previous = impl_->history[impl_->cursor]->draft;
594 for (const auto& item : previous.metadata) {
595 if (!item.locked) continue;
596 const auto nextMeta = std::find_if(draft.metadata.begin(), draft.metadata.end(),
597 [&](const auto& next) { return next.edgeId == item.edgeId; });
598 if (nextMeta == draft.metadata.end())
599 return fail<PreparedRoadPresetEdit>("locked edge metadata must be explicitly unlocked");
600 if (!nextMeta->locked) continue;
601 auto oldEdge = previous.source.edgeResult(item.edgeId);
602 if (!oldEdge.ok()) return eve::Result<PreparedRoadPresetEdit>::failure(oldEdge.status());
603 auto nextEdge = draft.source.edgeResult(item.edgeId);
604 if (!nextEdge.ok()) return eve::Result<PreparedRoadPresetEdit>::failure(nextEdge.status());
605 if (detail::encodeEdgeRecord(oldEdge.value(), {}) != detail::encodeEdgeRecord(nextEdge.value(), {}))
606 return fail<PreparedRoadPresetEdit>("locked road source cannot be directly edited");
607 for (const auto nodeId : {oldEdge.value().from, oldEdge.value().to}) {
608 auto oldNode = previous.source.nodeResult(nodeId);
609 if (!oldNode.ok()) return eve::Result<PreparedRoadPresetEdit>::failure(oldNode.status());
610 auto nextNode = draft.source.nodeResult(nodeId);
611 if (!nextNode.ok()) return eve::Result<PreparedRoadPresetEdit>::failure(nextNode.status());
612 if (detail::encodeNodeRecord(oldNode.value()) != detail::encodeNodeRecord(nextNode.value()))
613 return fail<PreparedRoadPresetEdit>("locked road endpoint cannot be directly edited");
614 }
615 const auto oldBinding = std::find_if(previous.bindings.begin(), previous.bindings.end(),
616 [&](const auto& binding) { return binding.edgeId == item.edgeId; });
617 const auto newBinding = std::find_if(draft.bindings.begin(), draft.bindings.end(),
618 [&](const auto& binding) { return binding.edgeId == item.edgeId; });
619 if ((oldBinding == previous.bindings.end()) != (newBinding == draft.bindings.end()) ||
620 (oldBinding != previous.bindings.end() && encodeBinding(*oldBinding) != encodeBinding(*newBinding)))
621 return fail<PreparedRoadPresetEdit>("locked road binding cannot be directly edited");
622 }
623 auto state = resolve(std::move(draft));
624 if (!state.ok()) return eve::Result<PreparedRoadPresetEdit>::failure(state.status());
625 auto impl = std::make_unique<PreparedRoadPresetEdit::Impl>();
626 impl->owner = impl_->owner;
627 impl->revision = impl_->revision;
628 impl->states.push_back(std::move(state).takeValue());
629 return eve::Result<PreparedRoadPresetEdit>::success(PreparedRoadPresetEdit(std::move(impl)));
630}
631eve::Result<void> RoadPresetDocument::commit(PreparedRoadPresetEdit edit) {
632 if (!edit.impl_) return fail<void>("consumed preset edit");
633 if (edit.impl_->owner != impl_->owner)
634 return eve::Result<void>::failure(eve::Diagnostic::error(
635 eve::DiagnosticCode::PreconditionViolation, "preset edit belongs to another document", "road.presets"));
636 if (edit.impl_->revision != impl_->revision)
637 return eve::Result<void>::failure(
638 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "preset document revision changed", "road.presets"));
639 if (edit.impl_->restore) {
640 impl_->history.swap(edit.impl_->states);
641 impl_->cursor = edit.impl_->cursor;
642 } else {
643 std::vector<std::shared_ptr<State>> next(
644 impl_->history.begin(), impl_->history.begin() + static_cast<std::ptrdiff_t>(impl_->cursor + 1));
645 next.push_back(std::move(edit.impl_->states.front()));
646 impl_->history.swap(next);
647 ++impl_->cursor;
648 }
649 ++impl_->revision;
650 return eve::Result<void>::success();
651}
652eve::Result<void> RoadPresetDocument::undo() {
653 if (impl_->cursor == 0) return fail<void>("no preset undo state");
654 --impl_->cursor;
655 ++impl_->revision;
656 return eve::Result<void>::success();
657}
658eve::Result<void> RoadPresetDocument::redo() {
659 if (impl_->cursor + 1 >= impl_->history.size()) return fail<void>("no preset redo state");
660 ++impl_->cursor;
661 ++impl_->revision;
662 return eve::Result<void>::success();
663}

679eve::Result<PreparedRoadPresetEdit> RoadPresetDocument::prepareRestore(const eve::Value& canonical) const {
680 if (!boundedValue(canonical)) return fail<PreparedRoadPresetEdit>("preset snapshot exceeds tree budget");
681 auto json = canonical.toJson();
682 if (!json.ok()) return eve::Result<PreparedRoadPresetEdit>::failure(json.status());

715 if (!type.overrides.samplingPolicy)
716 type.overrides.samplingPolicy = procgen::road::RoadSamplingPolicy::MaximumIntervals;
717 }
718 }
719 auto state = resolve(std::move(draft).takeValue());
720 if (!state.ok()) return eve::Result<PreparedRoadPresetEdit>::failure(state.status());
721 prepared->states.push_back(std::move(state).takeValue());
722 }
723 return eve::Result<PreparedRoadPresetEdit>::success(PreparedRoadPresetEdit(std::move(prepared)));
724}
