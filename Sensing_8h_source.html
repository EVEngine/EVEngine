428[[nodiscard]] inline std::vector<PerceptionFact> perceptionFactsFrom(const CandidateQueryResult& result) {
429 std::vector<PerceptionFact> facts;
430 facts.reserve(result.size());
431 for (const auto& candidate : result.ranked()) {
432 PerceptionFact fact;
433 fact.subjectId = candidate.id;
434 fact.x = candidate.x;
435 fact.y = candidate.y;
436 fact.z = candidate.z;
437 fact.space = candidate.space;
438 fact.distance = candidate.distance;
439 fact.score = candidate.score;
440 fact.scoreReason = candidate.scoreReason;
441 facts.push_back(std::move(fact));
442 }
443 return facts;
444}
