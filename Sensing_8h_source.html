310[[nodiscard]] inline std::vector<PerceptionFact> perceptionFactsFrom(const CandidateQueryResult& result) {
311 std::vector<PerceptionFact> facts;
312 facts.reserve(result.size());
313 for (const auto& candidate : result.ranked()) {
314 PerceptionFact fact;
315 fact.subjectId = candidate.id;
316 fact.x = candidate.x;
317 fact.y = candidate.y;
318 fact.distance = candidate.distance;
319 fact.score = candidate.score;
320 fact.scoreReason = candidate.scoreReason;
321 facts.push_back(std::move(fact));
322 }
323 return facts;
324}
