377[[nodiscard]] inline std::vector<PerceptionFact> perceptionFactsFrom(const CandidateQueryResult& result) {
378 std::vector<PerceptionFact> facts;
379 facts.reserve(result.size());
380 for (const auto& candidate : result.ranked()) {
381 PerceptionFact fact;
382 fact.subjectId = candidate.id;
383 fact.x = candidate.x;
384 fact.y = candidate.y;
385 fact.distance = candidate.distance;
386 fact.score = candidate.score;
387 fact.scoreReason = candidate.scoreReason;
388 facts.push_back(std::move(fact));
389 }
390 return facts;
391}
