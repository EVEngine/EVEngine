321[[nodiscard]] inline std::vector<PerceptionFact> perceptionFactsFrom(const CandidateQueryResult& result) {
322 std::vector<PerceptionFact> facts;

324 for (const auto& candidate : result.ranked()) {

327 fact.x = candidate.x;
