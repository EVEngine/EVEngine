38static std::string sceneInstanceId(const PointSet& points, size_t pointIndex,
39 std::unordered_map<uint32_t, size_t>& seedOccurrences) {
40 const auto& point = points.points()[pointIndex];
41 const auto explicitId = points.attributes().getString(pointIndex, "instanceId");
42 if (explicitId && !explicitId->empty()) return std::string(*explicitId);
43 if (point.id != 0) return "pcg-id-" + std::to_string(point.id);
44 return "pcg-" + std::to_string(point.seed) + "-" + std::to_string(seedOccurrences[point.seed]++);
45}

47static eve::ProcgenInstanceDesc sceneInstanceDesc(const PointSet& points, size_t pointIndex,
48 const std::string& assetAttribute, const std::string& defaultAsset,
49 std::unordered_map<uint32_t, size_t>& seedOccurrences) {
50 const auto& point = points.points()[pointIndex];
51 eve::ProcgenInstanceDesc instance;
52 instance.sourcePointId = point.id;
53 instance.id = sceneInstanceId(points, pointIndex, seedOccurrences);
54 instance.asset = defaultAsset;
55 if (!assetAttribute.empty()) {
56 const auto found = points.attributes().getString(pointIndex, assetAttribute);
57 if (found) instance.asset = std::string(*found);
58 }
59 instance.x = point.x;
60 instance.y = point.y;
61 instance.z = point.z;
62 instance.yaw = point.yaw;
63 instance.pitch = point.pitch;
64 instance.roll = point.roll;
65 instance.scaleX = point.scaleX;
66 instance.scaleY = point.scaleY;
67 instance.scaleZ = point.scaleZ;
68 instance.seed = point.seed;
69 return instance;
70}

72static eve::Result<std::vector<eve::ProcgenInstanceDesc>> sceneInstanceDescs(const PointSet& points,
73 const std::string& assetAttribute,
74 const std::string& defaultAsset,
75 bool requireStablePointIds) {
76 std::vector<eve::ProcgenInstanceDesc> instances;
77 instances.reserve(points.points().size());
78 std::unordered_map<uint32_t, size_t> seedOccurrences;
79 std::unordered_set<std::string> instanceIds;
80 std::unordered_set<uint64_t> pointIds;
81 for (size_t pointIndex = 0; pointIndex < points.points().size(); ++pointIndex) {
82 auto instance = sceneInstanceDesc(points, pointIndex, assetAttribute, defaultAsset, seedOccurrences);
83 if (points.attributes().has(pointIndex, "asset_root_pose") &&
84 !points.attributes().has(pointIndex, "asset_build_id"))
85 return eve::Result<std::vector<eve::ProcgenInstanceDesc>>::failure(
86 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
87 "prefab root pose requires a typed asset/build binding", "asset_root_pose"));
88 if (points.attributes().has(pointIndex, "asset_build_id")) {
89 const auto buildText = points.attributes().getString(pointIndex, "asset_build_id");
90 if (!buildText)
91 return eve::Result<std::vector<eve::ProcgenInstanceDesc>>::failure(eve::Diagnostic::error(
92 eve::DiagnosticCode::TypeMismatch, "procedural asset_build_id must be a string", "asset_build_id"));
93 auto asset = eve::AssetRef::parse(instance.asset);
94 if (!asset.ok()) return eve::Result<std::vector<eve::ProcgenInstanceDesc>>::failure(asset.status());
95 auto build = eve::PersistentId::parse(*buildText);
96 if (!build || build->isNil() || build->format() != *buildText || asset.value().format() != instance.asset)
97 return eve::Result<std::vector<eve::ProcgenInstanceDesc>>::failure(eve::Diagnostic::error(
98 eve::DiagnosticCode::InvalidArgument,
99 "procedural prefab binding must use canonical asset/build identities", "asset_build_id"));
100 instance.assetBinding = eve::ProcgenInstanceAssetBinding{asset.value(), *build};
101 if (points.attributes().has(pointIndex, "asset_root_pose")) {
102 const auto pose = points.attributes().getString(pointIndex, "asset_root_pose");
103 if (!pose || (*pose != "authored" && *pose != "placement"))
104 return eve::Result<std::vector<eve::ProcgenInstanceDesc>>::failure(
105 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
106 "unknown procedural prefab root pose", "asset_root_pose"));
107 instance.assetBinding->rootPose = *pose == "authored" ? eve::PrefabRootPose::Authored
108 : eve::PrefabRootPose::ReplaceTranslationRotation;
109 }
110 }
111 if (!std::isfinite(instance.yaw) || !std::isfinite(instance.pitch) || !std::isfinite(instance.roll))
112 return eve::Result<std::vector<eve::ProcgenInstanceDesc>>::failure(eve::Diagnostic::error(
113 eve::DiagnosticCode::InvalidArgument, "procedural instance rotation must be finite", "rotation"));
114 if (!instanceIds.insert(instance.id).second)
115 return eve::Result<std::vector<eve::ProcgenInstanceDesc>>::failure(
116 eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
117 "procedural Scene snapshot contains a duplicate instance id", "instanceId"));
118 if (requireStablePointIds && (instance.sourcePointId == 0 || !pointIds.insert(instance.sourcePointId).second))
119 return eve::Result<std::vector<eve::ProcgenInstanceDesc>>::failure(eve::Diagnostic::error(
120 eve::DiagnosticCode::Conflict, "procedural Scene snapshot requires unique non-zero source PointIds",
121 "pointId"));
122 instances.push_back(std::move(instance));
123 }
124 return eve::Result<std::vector<eve::ProcgenInstanceDesc>>::success(std::move(instances));
125}

233eve::Result<uint64_t> Procgen::synchronizeCellInstances(const std::string& prefix, const RuntimeGeneration& runtime,
234 const ProcgenCellRequest& request,
235 const std::string& assetAttribute,
236 const std::string& defaultAsset) {
237 if (prefix.empty())
238 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
239 "synchronizeCellInstances requires a prefix",
240 "prefix", {}, "procgen.squirrel"));
241 const int level = request.getLevel();
242 const int x = request.getX();
243 const int z = request.getZ();
244 const uint64_t targetRevision = runtime.getCellRevision(level, x, z);
245 std::unique_ptr<PointSet> snapshot(runtime.getCellOutput(level, x, z));
246 if (!snapshot || targetRevision == 0)
247 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
248 eve::DiagnosticCode::NotFound, "synchronizeCellInstances requires an active runtime cell", "cell", {},
249 "procgen.squirrel"));
250
251 auto* sink = eve::cap::query<eve::IProcgenSceneSink>();
252 if (!sink)
253 return eve::Result<uint64_t>::failure(
254 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "synchronizeCellInstances scene sink is unavailable",
255 {}, {}, "procgen.squirrel"));
256 const std::string batchId =
257 prefix + "/L" + std::to_string(level) + "/" + std::to_string(x) + "/" + std::to_string(z);
258 const uint64_t sceneRevision = sink->batchRevision(batchId);
259 if (sceneRevision == targetRevision) return eve::Result<uint64_t>::success(targetRevision);
260 if (sceneRevision > targetRevision)
261 return eve::Result<uint64_t>::failure(
262 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "Scene cell revision is ahead of RuntimeGeneration",
263 "revision", {}, "procgen.squirrel"));
264
265 if (sceneRevision + 1 == targetRevision) {
266 std::unique_ptr<PointDelta> delta(runtime.getCellDelta(level, x, z));
267 if (delta)
268 return publishCellInstanceDelta(prefix, request, *delta, targetRevision, assetAttribute, defaultAsset);
269 }
270
271 auto instances = sceneInstanceDescs(*snapshot, assetAttribute, defaultAsset, true);
272 if (!instances.ok()) return eve::Result<uint64_t>::failure(instances.status());
273 return sink->replaceBatch(batchId, targetRevision, instances.value());
274}

364eve::Result<uint64_t> Procgen::completeCellCleanupAtomic(const std::string& prefix,
365 const std::vector<RuntimeGeneration*>& runtimes,
366 const std::vector<const ProcgenCellRequest*>& requests) {
367 if (prefix.empty() || requests.empty() || runtimes.size() != requests.size())
368 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
369 eve::DiagnosticCode::InvalidArgument,
370 "completeCellCleanupAtomic requires a prefix and equally sized runtime/request arrays", "requests", {},
371 "procgen.squirrel"));
372 auto* sink = eve::cap::query<eve::IProcgenSceneSink>();
373 if (!sink)
374 return eve::Result<uint64_t>::failure(
375 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "completeCellCleanupAtomic scene sink is unavailable",
376 {}, {}, "procgen.squirrel"));
377
378 struct RuntimeCleanupGroup {
379 RuntimeGeneration* runtime = nullptr;
380 std::vector<const ProcgenCellRequest*> requests;
381 };
382 std::vector<RuntimeCleanupGroup> groups;
383 std::vector<std::string> batchIds;
384 groups.reserve(runtimes.size());
385 batchIds.reserve(requests.size());
386 for (size_t index = 0; index < requests.size(); ++index) {
387 auto* runtime = runtimes[index];
388 auto* request = requests[index];
389 if (!runtime || !request)
390 return eve::Result<uint64_t>::failure(eve::Diagnostic::error(
391 eve::DiagnosticCode::InvalidArgument, "completeCellCleanupAtomic contains a null runtime or request",
392 "requests", {}, "procgen.squirrel"));
393 auto group = std::find_if(groups.begin(), groups.end(),
394 [runtime](const auto& candidate) { return candidate.runtime == runtime; });
395 if (group == groups.end()) {
396 groups.push_back({runtime, {}});
397 group = std::prev(groups.end());
398 }
399 group->requests.push_back(request);
400 batchIds.push_back(prefix + "/L" + std::to_string(request->getLevel()) + "/" + std::to_string(request->getX()) +
401 "/" + std::to_string(request->getZ()));
402 }
403 for (const auto& group : groups) {
404 auto validated = group.runtime->validateCleanups(group.requests);
405 if (!validated.ok()) return eve::Result<uint64_t>::failure(validated.status());
406 }
407 return sink->removeBatchesCoordinated(batchIds, [&groups] {
408 for (auto& group : groups) group.runtime->eraseValidatedCleanups(group.requests);
409 return eve::Result<void>::success();
410 });
411}
