34bool CardPlaneMapper::mapRay(float ox, float oy, float oz, float dx, float dy, float dz) {
35 const glm::vec3 normal = glm::cross(axisU_, axisV_);
36 const glm::vec3 direction(dx, dy, dz);
37 const float denominator = glm::dot(normal, direction);
38 if (std::abs(denominator) < 1e-6f) return false;
39 const float t = glm::dot(normal, origin_ - glm::vec3(ox, oy, oz)) / denominator;
40 if (t < 0.f) return false;
41 resultWorld_ = glm::vec3(ox, oy, oz) + direction * t;
42 const glm::vec3 relative = resultWorld_ - origin_;
43 const float uu = glm::dot(axisU_, axisU_);
44 const float uv = glm::dot(axisU_, axisV_);
45 const float vv = glm::dot(axisV_, axisV_);
46 const float determinant = uu * vv - uv * uv;
47 if (std::abs(determinant) < 1e-8f || logicalWidth_ == 0.f || logicalHeight_ == 0.f)
48 return false;
49 const float wu = glm::dot(relative, axisU_);
50 const float wv = glm::dot(relative, axisV_);
51 const float u = (wu * vv - wv * uv) / determinant;
52 const float v = (wv * uu - wu * uv) / determinant;
53 resultLogicalX_ = logicalX_ + u * logicalWidth_;
54 resultLogicalY_ = logicalY_ + v * logicalHeight_;
55 return true;
56}

267eve::Result<std::size_t> Card::step(const eve::SimulationStep &simulationStep) {
268 std::size_t settled = 0;
269 for (const auto &handle : cards_) {
270 auto *card = resolve<CardData>(handle);
271 if (card == nullptr || card->effects()->values.count() == 0) continue;
272 auto updated = card->effects()->values.advance(simulationStep);
273 if (!updated) return eve::Result<std::size_t>::failure(updated.status());
274 auto result = std::move(updated).takeValue();
275 auto projected = CardAttributeAdapter::setBase(*card, CardAttributeAdapter::healthAttribute,
276 static_cast<double>(card->effects()->values.target().health));
277 if (!projected) return eve::Result<std::size_t>::failure(projected.status());
278 settled += result.settled;
279 }
280 return eve::Result<std::size_t>::success(
281 settled, eve::Status::success(settled == 0 ? eve::StatusCode::NoOp : eve::StatusCode::Applied));
282}
