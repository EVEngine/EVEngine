113graphics::Texture *Decal::bakePresetTexture(graphics::Graphics *gfx, const std::string &preset,
114 std::uint32_t seed, int resolution,
115 const std::string &channel) {
116 if (!gfx) throw eve::Exception("Decal.bakePresetTexture: null gfx");
117 requireBakeChannel(channel);
118 const std::string source = "preset:" + preset;
119 if (proceduralBakeCache_->source != source || proceduralBakeCache_->seed != seed ||
120 proceduralBakeCache_->resolution != resolution) {
121 auto recipe = proceduralDecalPreset(preset, seed);
122 if (!recipe.ok()) throw eve::Exception("%s", recipe.status().describe().c_str());
123 recipe.value().width = resolution;
124 recipe.value().height = resolution;
125 auto baked = bakeProceduralDecal(recipe.value());
126 if (!baked.ok()) throw eve::Exception("%s", baked.status().describe().c_str());
127 proceduralBakeCache_->source = source;
128 proceduralBakeCache_->seed = seed;
129 proceduralBakeCache_->resolution = resolution;
130 proceduralBakeCache_->bake = std::move(baked.value());
131 }
132 const auto &bytes = bakeChannel(proceduralBakeCache_->bake, channel);
133 auto *texture = gfx->newTexture(resolution, resolution, bytes.data());
134 if (!texture) throw eve::Exception("Decal.bakePresetTexture: graphics upload failed");
135 return texture;
136}
