11using DetachedConnections = std::vector<std::pair<std::uint32_t, procgen::road::RoadConnection>>;
12Result<DetachedConnections> detachUndoBindings(procgen::road::RoadNetwork& candidate,
13 std::initializer_list<std::uint32_t> edgeIds) {
14 DetachedConnections owned;
15 const auto affected = [&](std::uint32_t id) {
16 return std::find(edgeIds.begin(), edgeIds.end(), id) != edgeIds.end();
17 };
18 for (const auto& node : candidate.nodes()) {
19 const auto& connection = node.connection;
20 if (std::any_of(connection.ports.begin(), connection.ports.end(),
21 [&](const auto& p) { return affected(p.edgeId); }) ||
22 std::any_of(connection.facilityRoutes.begin(), connection.facilityRoutes.end(),
23 [&](const auto& r) { return affected(r.firstEdge) || affected(r.secondEdge); }))
24 owned.emplace_back(node.id, connection);
25 }
26 // Iterate owning copies: setNodeConnection atomically replaces candidate's graph storage.
27 for (const auto& [id, connection] : owned) {
28 (void)connection;
29 auto cleared = candidate.setNodeConnection(id, {});
30 if (!cleared.ok()) return Result<DetachedConnections>::failure(cleared.status());
31 }
32 return Result<DetachedConnections>::success(std::move(owned));
33}
34Result<void> restoreUndoBindings(procgen::road::RoadNetwork& candidate, DetachedConnections owned,
35 std::initializer_list<std::pair<std::uint32_t, std::uint32_t>> replacements) {
36 auto remap = [&](std::uint32_t& edge) {
37 for (const auto& [oldId, newId] : replacements)
38 if (edge == oldId) {
39 edge = newId;
40 return;
41 }
42 };
43 for (auto& [id, connection] : owned) {
44 for (auto& p : connection.ports) remap(p.edgeId);
45 for (auto& r : connection.facilityRoutes) {
46 remap(r.firstEdge);
47 remap(r.secondEdge);
48 }
49 auto restored = candidate.setNodeConnection(id, std::move(connection));
50 if (!restored.ok()) return restored;
51 }
52 return Result<void>::success();
53}

435 std::vector<procgen::road::RoadLaneConnection> links;
436 std::vector<procgen::road::RoadLaneConnection> blockedLinks;
437 if (!decodeEdgeRecord(operation.payload, edge, links, &blockedLinks))
438 return reject<void>("editor.road.edge-payload", "Road edge restore payload is invalid");
439 auto candidate = *network_;
440 auto restored = candidate.restoreEdge(std::move(edge));
441 if (!restored.ok()) return roadFailure<void>(restored.status());
442 for (const auto& link : links) {
443 auto linked = candidate.addLaneLink(link);
444 if (!linked.ok()) return roadFailure<void>(linked.status());
445 }
446 for (const auto& link : blockedLinks) {
447 auto blocked = candidate.blockLaneLink(link);
448 if (!blocked.ok()) return roadFailure<void>(blocked.status());
449 }
450 *network_ = std::move(candidate);
451 } else if (operation.type == kRemoveEdge) {
452 std::uint32_t edgeId = 0;
453 if (!readId(operation.payload, "id", edgeId))
454 return reject<void>("editor.road.edge-payload", "Road edge removal payload is invalid");
455 auto candidate = *network_;
456 auto removed = candidate.removeEdge(edgeId);
457 if (!removed.ok()) return roadFailure<void>(removed.status());
458 *network_ = std::move(candidate);
459 } else if (operation.type == kRestoreLaneLink || operation.type == kRemoveLaneLink) {
460 procgen::road::RoadLaneConnection link;
461 if (!decodeLink(operation.payload, link))
462 return reject<void>("editor.road.lane-link-payload", "Road lane-link payload is invalid");
463 auto candidate = *network_;
464 auto changed =
465 operation.type == kRestoreLaneLink ? candidate.addLaneLink(link) : candidate.removeLaneLink(link);
466 if (!changed.ok()) return roadFailure<void>(changed.status());
467 *network_ = std::move(candidate);
468 } else if (operation.type == kBlockLaneLink || operation.type == kUnblockLaneLink) {
469 procgen::road::RoadLaneConnection link;
470 bool restoreActive = false;
471 if (!decodeBlockedLink(operation.payload, link, restoreActive))
472 return reject<void>("editor.road.lane-link-block-payload", "Road lane-link block payload is invalid");
473 auto candidate = *network_;
474 if (operation.type == kBlockLaneLink) {
475 auto changed = candidate.blockLaneLink(link);
476 if (!changed.ok()) return roadFailure<void>(changed.status());
477 } else {
478 auto changed = candidate.unblockLaneLink(link);
479 if (!changed.ok()) return roadFailure<void>(changed.status());
480 if (restoreActive) {
481 auto restored = candidate.addLaneLink(link);
482 if (!restored.ok()) return roadFailure<void>(restored.status());
483 }
484 }
485 *network_ = std::move(candidate);
486 } else if (operation.type == kRestoreLaneLinks || operation.type == kRemoveLaneLinks) {
487 std::vector<procgen::road::RoadLaneConnection> links;
488 if (!decodeLinks(operation.payload, links))
489 return reject<void>("editor.road.lane-links-payload", "Road lane-link batch payload is invalid");
490 if (links.empty()) return editing::noOp();
491 auto candidate = *network_;
492 for (const auto& link : links) {
493 auto changed =
494 operation.type == kRestoreLaneLinks ? candidate.addLaneLink(link) : candidate.removeLaneLink(link);
495 if (!changed.ok()) return roadFailure<void>(changed.status());
496 }
497 *network_ = std::move(candidate);
498 } else {
499 return reject<void>("editor.road.type", "Unsupported road operation type", EditorStatus::Unsupported);
500 }
501 dirty_.include(0, 0);
502 return editing::applied();
503}
