77std::function<eve::Result<void>(const SequenceNode&)> exactlyOneOf(std::vector<std::string> names,
78 std::string stepType) {
79 return [names = std::move(names), stepType = std::move(stepType)](const SequenceNode& node) -> eve::Result<void> {
80 int present = 0;
81 for (const auto& name : names) {
82 if (const eve::Value* value = node.payload.find(name); value && !value->isNull()) ++present;
83 }
84 if (present == 1) return eve::Result<void>::success();
85 std::string expected;
86 for (std::size_t index = 0; index < names.size(); ++index) {
87 if (index != 0) expected += " or ";
88 expected += "'" + names[index] + "'";
89 }
90 return descriptorFailure("step '" + stepType + "' requires exactly one of " + expected);
91 };
92}

95std::function<eve::Result<void>(const SequenceNode&)> oneOfValues(std::string fieldName,
96 std::vector<std::string> allowed,
97 std::string stepType) {
98 return [fieldName = std::move(fieldName), allowed = std::move(allowed),
99 stepType = std::move(stepType)](const SequenceNode& node) -> eve::Result<void> {
100 const eve::Value* value = node.payload.find(fieldName);
101 if (!value || !value->isString()) return eve::Result<void>::success();
102 for (const auto& candidate : allowed) {
103 if (value->asString() == candidate) return eve::Result<void>::success();
104 }
105 std::string accepted;
106 for (std::size_t index = 0; index < allowed.size(); ++index) {
107 if (index != 0) accepted += ", ";
108 accepted += allowed[index];
109 }
110 return descriptorFailure("step '" + stepType + "' field '" + fieldName + "' must be one of " + accepted);
111 };

124StepOutcome applySkill(const SequenceNode& node, const StepContext& context) {
125 auto* binding = static_cast<RpgStoryBinding*>(context.host);
126 const std::string actorId = stringField(node, "actor");
127 RPGActor* actor = resolveActor(binding, actorId);
128 if (!actor) return failed("skill step: actor '" + actorId + "' cannot be resolved");
129 if (const eve::Value* learn = field(node, "learn"); learn && learn->isString()) {
130 actor->learnSkill(learn->asString());
131 return completed();
132 }
133 const eve::Value* forget = field(node, "forget");
134 if (!forget || !forget->isString()) return failed("skill step: neither learn nor forget is set");
135 if (!actor->forgetSkill(forget->asString()))
136 return failed("skill step: actor '" + actorId + "' does not know skill '" + forget->asString() + "'");
137 return completed();
138}

140StepOutcome applyAttribute(const SequenceNode& node, const StepContext& context) {
141 auto* binding = static_cast<RpgStoryBinding*>(context.host);
142 const std::string actorId = stringField(node, "actor");
143 RPGActor* actor = resolveActor(binding, actorId);
144 if (!actor) return failed("attribute step: actor '" + actorId + "' cannot be resolved");
145 const std::string name = stringField(node, "name");
146 const std::string op = stringField(node, "op");
147 const double value = numberField(node, "value", 0.0);
148 if (op == "set") {
149 actor->setBaseAttribute(name, value);
150 } else if (op == "add") {
151 actor->modifyBaseAttribute(name, value);
152 } else if (op == "sub") {
153 actor->modifyBaseAttribute(name, -value);
154 } else if (op == "mul") {
155 actor->setBaseAttribute(name, actor->getBaseAttribute(name) * value);
156 } else {
157 return failed("attribute step: unsupported op '" + op + "'");
158 }
159 return completed();
160}

162StepOutcome applyItem(const SequenceNode& node, const StepContext& context) {
163 auto* binding = static_cast<RpgStoryBinding*>(context.host);
164 if (!binding || !binding->bag) return failed("item step: no bag is bound");
165 const std::int64_t requested = integerField(node, "count", 1);
166 if (requested < 1) return failed("item step: count must be at least 1");
167 const auto quantity = static_cast<int>(requested);
168
169 if (const eve::Value* add = field(node, "add"); add && add->isString()) {
170 if (!binding->bag->canAddItem(add->asString(), quantity)) {
171 const std::string reason = binding->bag->canAddItemReason(add->asString(), quantity);
172 return failed("item step: bag refuses '" + add->asString() + "' (" +
173 (reason.empty() ? std::string("no capacity") : reason) + ")");
174 }
175 if (binding->bag->addItem(add->asString(), quantity) != quantity)
176 return failed("item step: bag did not accept the whole stack of '" + add->asString() + "'");
177 return completed();
178 }
179 const eve::Value* remove = field(node, "remove");
180 if (!remove || !remove->isString()) return failed("item step: neither add nor remove is set");
181 if (binding->bag->countItem(remove->asString()) < quantity)
182 return failed("item step: bag does not hold " + std::to_string(quantity) + " of '" + remove->asString() +
183 "'");
184 if (binding->bag->removeItem(remove->asString(), quantity) != quantity)
185 return failed("item step: bag did not remove the whole stack of '" + remove->asString() + "'");
186 return completed();
187}

189StepOutcome applyEquipment(const SequenceNode& node, const StepContext& context) {
190 auto* binding = static_cast<RpgStoryBinding*>(context.host);
191 if (!binding || !binding->equipment) return failed("equipment step: no equipment set is bound");
192 if (const eve::Value* equip = field(node, "equip"); equip && equip->isString()) {
193 if (!binding->bag) return failed("equipment step: equipping requires a bound bag");
194 const std::string slotName = stringField(node, "slot");
195 const int bagSlot = binding->bag->findItem(equip->asString());
196 if (bagSlot < 0) return failed("equipment step: bag does not hold '" + equip->asString() + "'");
197 if (!binding->equipment->equipFromBag(slotName, binding->bag, bagSlot))
198 return failed("equipment step: cannot equip '" + equip->asString() + "' into slot '" + slotName + "'");
199 return completed();
200 }
201 const eve::Value* unequip = field(node, "unequip");
202 if (!unequip || !unequip->isString()) return failed("equipment step: neither equip nor unequip is set");
203 if (!binding->bag) return failed("equipment step: unequipping requires a bound bag");
204 if (!binding->equipment->unequipToBag(unequip->asString(), binding->bag))
205 return failed("equipment step: cannot unequip slot '" + unequip->asString() + "'");
206 return completed();
207}

250StepOutcome performMove(const SequenceNode& node, const StepContext& context) {
251 auto* binding = static_cast<RpgStoryBinding*>(context.host);
252 if (!binding || !binding->moveActor) {
253 StepOutcome presented;
254 presented.status = StepStatus::Blocked;
255 return presented;
256 }
257
258 StoryMoveRequest request;
259 request.actorId = stringField(node, "actor");
260 request.actor = resolveActor(binding, request.actorId);
261 request.x = numberField(node, "x", 0.0);
262 request.y = numberField(node, "y", 0.0);
263 if (const eve::Value* duration = field(node, "duration"); duration && duration->isNumeric()) {
264 request.hasDuration = true;
265 request.duration = numberField(node, "duration", 0.0);
266 }
267
268 auto started = binding->moveActor(request);
269 if (!started.ok()) {
270 const auto* diagnostic = started.error();
271 return failed(diagnostic ? diagnostic->message() : "move step: the movement controller failed");
272 }
273 switch (started.value()) {
274 case StoryMoveStatus::Arrived: return completed();
275 case StoryMoveStatus::Unreachable:
276 return failed("move step: actor '" + request.actorId + "' cannot reach (" + formatCoordinate(request.x) +
277 ", " + formatCoordinate(request.y) + ")");
278 case StoryMoveStatus::Travelling: break;
279 }
280 StepOutcome travelling;
281 travelling.status = StepStatus::Blocked;
282 return travelling;
283}

300StepOutcome performAnimation(const SequenceNode& node, const StepContext& context) {
301 auto* binding = static_cast<RpgStoryBinding*>(context.host);
302 if (!binding || !binding->playAnimation) {
303 StepOutcome presented;
304 presented.status = StepStatus::Blocked;
305 return presented;
306 }
307
308 StoryAnimationRequest request;
309 request.targetId = stringField(node, "target");
310 request.actor = resolveActor(binding, request.targetId);
311 request.clip = stringField(node, "clip");
312 request.loop = boolField(node, "loop", false);
313 request.hold = boolField(node, "hold", false);
314
315 auto started = binding->playAnimation(request);
316 if (!started.ok()) {
317 const auto* diagnostic = started.error();
318 return failed(diagnostic ? diagnostic->message() : "animation step: the playback controller failed");
319 }
320 switch (started.value()) {
321 case StoryAnimationStatus::Finished: return completed();
322 case StoryAnimationStatus::Unavailable:
323 return failed("animation step: target '" + request.targetId + "' cannot play clip '" + request.clip + "'");
324 case StoryAnimationStatus::Playing: break;
325 }
326 StepOutcome playing;
327 playing.status = StepStatus::Blocked;
328 return playing;
329}

333struct ResolvedState {
334 bool found = false;
335 double number = 0.0;
336};
337
338ResolvedState resolveStateName(const std::string& name, const RpgStoryBinding* binding) {
339 ResolvedState resolved;
340 if (!binding || !binding->gameState) return resolved;
341 if (name.rfind("switch.", 0) == 0) {
342 resolved.found = true;
343 resolved.number = binding->gameState->isSwitchOn(name.substr(7)) ? 1.0 : 0.0;
344 return resolved;
345 }
346 if (name.rfind("variable.", 0) == 0) {
347 resolved.found = true;
348 resolved.number = binding->gameState->getVariable(name.substr(9));
349 return resolved;
350 }
351 if (name.rfind("self.", 0) == 0) {
352 const std::string rest = name.substr(5);
353 const auto dot = rest.rfind('.');
354 if (dot == std::string::npos) return resolved;
355 const std::string scope = rest.substr(0, dot);
356 const std::string key = rest.substr(dot + 1);
357 if (!binding->gameState->hasSelfVariable(scope, key)) return resolved;
358 resolved.found = true;
359 resolved.number = binding->gameState->getSelfVariable(scope, key);
360 return resolved;
361 }
362 return resolved;
363}

365bool compareNumbers(double left, double right, const std::string& op) {
366 if (op == "eq") return std::fabs(left - right) <= kNumberEpsilon;
367 if (op == "ne") return std::fabs(left - right) > kNumberEpsilon;
368 if (op == "gt") return left > right + kNumberEpsilon;
369 if (op == "lt") return left < right - kNumberEpsilon;
370 if (op == "ge") return left >= right - kNumberEpsilon;
371 if (op == "le") return left <= right + kNumberEpsilon;
372 return false;
373}

375SequenceConditionOutcome evaluateCondition(const eve::Value& condition, const RpgStoryBinding* binding) {
376 if (!condition.isObject()) return {false, "invalid-condition"};
377
378 if (const eve::Value* all = condition.find("all"); all && all->isArray()) {
379 for (std::size_t index = 0; index < all->arraySize(); ++index) {
380 if (!evaluateCondition(all->at(index), binding).passed) return {false, "child-failed"};
381 }
382 return {true, {}};
383 }
384 if (const eve::Value* any = condition.find("any"); any && any->isArray()) {
385 for (std::size_t index = 0; index < any->arraySize(); ++index) {
386 if (evaluateCondition(any->at(index), binding).passed) return {true, {}};
387 }
388 return {false, "no-child-passed"};
389 }
390 if (const eve::Value* negated = condition.find("not"); negated) {
391 return evaluateCondition(*negated, binding).passed ? SequenceConditionOutcome{false, "negated"}
392 : SequenceConditionOutcome{true, {}};
393 }
394
395 const eve::Value* variable = condition.find("var");
396 const eve::Value* operation = condition.find("op");
397 const eve::Value* expected = condition.find("value");
398 if (!variable || !variable->isString() || !operation || !operation->isString() || !expected)
399 return {false, "invalid-condition"};
400
401 const ResolvedState resolved = resolveStateName(variable->asString(), binding);
402 if (!resolved.found) return {false, "missing-value"};
403
404 double expectedNumber = 0.0;
405 if (expected->isBool()) {
406 expectedNumber = expected->asBool() ? 1.0 : 0.0;
407 } else if (expected->isNumeric()) {
408 expectedNumber = expected->isInt64() ? static_cast<double>(expected->asInt()) : expected->asDouble();
409 } else {
410 return {false, "unsupported-operand"};
411 }
412 if (compareNumbers(resolved.number, expectedNumber, operation->asString())) return {true, {}};
413 return {false, "value-mismatch"};
414}

421std::function<eve::Result<void>(const SequenceNode&)> equipRequiresSlot() {
422 return [](const SequenceNode& node) -> eve::Result<void> {
423 if (const eve::Value* equip = node.payload.find("equip"); equip && !equip->isNull()) {

432StepKindRegistry buildRpgStoryRegistry() {
433 StepKindRegistry registry;
434 const auto declare = [&registry](StepKindDescriptor descriptor) {
435 registry.registerStep(std::move(descriptor)).expect("RPG story step descriptor must register");
436 };
437 const auto bind = [&registry](const char* type, StepHandler handler) {
438 registry.registerHandler(type, std::move(handler)).expect("RPG story step handler must bind");
439 };
440
441 // `move` carries a handler so a host can install a movement controller. With
442 // no controller bound the handler presents the step itself, so both paths
443 // suspend the story and acknowledge with advance().
444 declare({"move", "Move an actor", "presentation", StepShape::Await,
445 {requiredField("actor", StepFieldType::String), requiredField("x", StepFieldType::Number),
446 requiredField("y", StepFieldType::Number), optionalField("duration", StepFieldType::Number)}});
447 bind("move", performMove);
448
449 // `animation` carries a handler for the same reason `move` does: a host can
450 // install a playback controller and stop branching on the step kind. With no
451 // controller bound the handler presents the step itself, so it takes the
452 // same host-presented path as the steps below.
453 declare({"animation", "Play an animation clip", "presentation", StepShape::Await,
454 {requiredField("target", StepFieldType::String), requiredField("clip", StepFieldType::String),
455 optionalField("loop", StepFieldType::Boolean), optionalField("hold", StepFieldType::Boolean)}});
456 bind("animation", performAnimation);
457
458 // Host-presented steps: registered without a handler, so the runtime suspends
459 // and the host presents the payload before acknowledging with advance().
460 declare({"select", "Select a world object", "presentation", StepShape::Await,
461 {requiredField("object", StepFieldType::String), optionalField("prompt", StepFieldType::String)}});
462 declare({"camera", "Move the camera", "presentation", StepShape::Await,
463 {requiredField("x", StepFieldType::Number), requiredField("y", StepFieldType::Number),
464 optionalField("duration", StepFieldType::Number)}});
465 declare({"dialogue", "Play a conversation", "dialogue", StepShape::Await,

467 declare({"message", "Show a message", "dialogue", StepShape::Await,

560eve::Result<int> RpgStoryCatalogue::replaceFromDnutStrict(const std::string& source, const std::string& path) {
561 eve::dnut::DnutCompileOutput output = eve::dnut::compileDnut(source, path, rpgStoryStepRegistry());
562 if (output.hasErrors()) {
563 std::string message = "story document contains errors";
564 for (const auto& diagnostic : output.diagnostics) {

566 message = diagnostic.message;

568 }
569 }
570 return eve::Result<int>::failure(

572 }

575 eve::DiagnosticCode::InvalidArgument, "document declares no 'story' block", path, {}, "rpg.dnut-story"));
576
577 std::map<std::string, eve::dnut::SequenceAsset> proposed;
578 for (auto& asset : output.assets) {
579 const std::string id = asset.id;
580 proposed.emplace(id, std::move(asset));
581 }
582 storyCatalogue() = std::move(proposed);
583 return eve::Result<int>::success(static_cast<int>(storyCatalogue().size()));
584}

651eve::Result<void> RpgStorySession::begin(const std::string& storyId, RpgStoryBinding* binding) {
652 stop();
653 impl_->binding = binding;
654 if (!impl_->snapshot(storyId))
655 return makeStoryFailure(eve::DiagnosticCode::NotFound, "story '" + storyId + "' is not published", "storyId");
656
657 if (binding && binding->gameState) {
658 const std::string scope = completionScope(storyId);
659 const bool hasCompleted = binding->gameState->hasSelfVariable(scope, "completed");
660 const double completed = hasCompleted ? binding->gameState->getSelfVariable(scope, "completed") : 0.0;
661 if (hasCompleted && completed != 0.0 && completed != 1.0)
662 return makeStoryFailure(eve::DiagnosticCode::InvariantViolation,
663 "persisted story-completion flag is invalid", scope + ".completed");
664 if (completed == 1.0) {
665 const auto definition = impl_->assets.find(storyId);
666 if (definition != impl_->assets.end() && !definition->second.repeatable)
667 return makeStoryFailure(eve::DiagnosticCode::Conflict,
668 "non-repeatable story '" + storyId + "' is already complete", "storyId");
669 binding->gameState->setSelfVariable(scope, "completed", 0.0);
670 }
671 }
672
673 impl_->storyId = storyId;
674 impl_->configureRuntime();
675 const auto definition = impl_->assets.find(storyId);
676 if (definition == impl_->assets.end())
677 return makeStoryFailure(eve::DiagnosticCode::NotFound, "story '" + storyId + "' is not published", "storyId");
678 if (auto started = impl_->runtime.start(&definition->second); !started.ok()) {
679 const auto* diagnostic = started.error();
680 const std::string message = diagnostic ? diagnostic->message() : "story could not start";
681 stop();
682 return makeStoryFailure(eve::DiagnosticCode::Failed, message, "storyId");
683 }
684 impl_->publishCompletion();
685 return eve::Result<void>::success();
686}
