33Result<void> AbilityDefinition::validate() const {
34 if (!id.isValid()) return failure(DiagnosticCode::InvalidArgument, "Ability id is invalid", "id");
35 if (cooldown < Duration::zero())
36 return failure(DiagnosticCode::InvalidArgument, "Ability cooldown must be non-negative", "cooldown");
37 auto actionValid = action.validate();
38 if (!actionValid) return Result<void>::failure(actionValid.status());
39 for (std::size_t i = 0; i < triggers.size(); ++i) {

43 }
44 return Result<void>::success();

57Result<AbilityGrantId> AbilityRuntime::grant(std::string ownerId, const LogicalId& definitionId) {
58 if (ownerId.empty())
59 return failure<AbilityGrantId>(DiagnosticCode::InvalidArgument, "Ability owner is empty", "ownerId");
60 const auto definition = definitions_.find(definitionId.format());
61 if (definition == definitions_.end())
62 return failure<AbilityGrantId>(DiagnosticCode::NotFound, "Ability definition is not registered",
63 definitionId.format());
64 auto grantId = nextGrantId();
65 if (!grantId) return grantId;
66 AbilityInstanceId ownerInstance;
67 if (definition->second.instancing == AbilityInstancingPolicy::PerOwner) {
68 auto instance = nextInstanceId();
69 if (!instance) return Result<AbilityGrantId>::failure(instance.status());
70 ownerInstance = instance.value();
71 }
72 AbilityGrantState state{grantId.value(), std::move(ownerId), definitionId, Duration::zero()};
73 grants_.emplace(grantId.value(), GrantRecord{std::move(state), ownerInstance});
74 return Result<AbilityGrantId>::success(grantId.value(), Status::success(StatusCode::Applied));
75}

93Result<AbilityActivation> AbilityRuntime::activate(AbilityGrantId grantId, ActionRequest request, SimulationTick tick) {
94 auto grant = grants_.find(grantId);
95 if (grant == grants_.end())
96 return failure<AbilityActivation>(DiagnosticCode::NotFound, "Ability grant was not found", "grantId");
97 const auto definition = definitions_.find(grant->second.state.definitionId.format());
98 if (definition == definitions_.end())
99 return failure<AbilityActivation>(DiagnosticCode::NotFound, "Granted ability definition was removed",
100 "definitionId");
101 if (grant->second.state.cooldownRemaining > Duration::zero())
102 return failure<AbilityActivation>(DiagnosticCode::PreconditionViolation, "Ability is on cooldown", "cooldown");
103 if (request.actionId != definition->second.action.id)
104 return failure<AbilityActivation>(DiagnosticCode::InvalidArgument, "Ability request action id does not match",
105 "actionId");
106 if (definition->second.instancing != AbilityInstancingPolicy::PerExecution && grantHasActiveActivation(grantId))
107 return failure<AbilityActivation>(DiagnosticCode::Conflict, "Ability instance is already active", "grantId");
108
109 std::vector<AbilityActivationId> replaceable;
110 if (definition->second.activationGroup != AbilityActivationGroup::Independent) {
111 for (const auto& [id, active] : activations_) {
112 if (active.ownerId != grant->second.state.ownerId || active.group == AbilityActivationGroup::Independent)
113 continue;
114 const auto* execution = runtime_.find(active.executionId);
115 if (!execution || terminal(execution->phase())) continue;
116 if (active.group == AbilityActivationGroup::ExclusiveBlocking)
117 return failure<AbilityActivation>(DiagnosticCode::Conflict, "An exclusive-blocking ability is active",
118 "activationGroup");
119 replaceable.push_back(id);
120 }
121 }
122
123 auto execution = runtime_.submit(definition->second.action, std::move(request));
124 if (!execution) return Result<AbilityActivation>::failure(execution.status());
125 for (const auto replacedId : replaceable) {
126 auto cancelled = runtime_.cancel(activations_.at(replacedId).executionId, tick);
127 if (!cancelled) {
128 auto rollback = runtime_.cancel(execution.value(), tick);
129 if (!rollback) return Result<AbilityActivation>::failure(rollback.status());
130 return failure<AbilityActivation>(DiagnosticCode::Conflict,
131 "Could not replace the active exclusive ability", "activationGroup");
132 }
133 activations_.erase(replacedId);
134 }
135
136 auto activationId = nextActivationId();
137 if (!activationId) {
138 auto rollback = runtime_.cancel(execution.value(), tick);
139 if (!rollback) return Result<AbilityActivation>::failure(rollback.status());
140 return Result<AbilityActivation>::failure(activationId.status());
141 }
142 AbilityInstanceId instanceId;
143 if (definition->second.instancing == AbilityInstancingPolicy::PerOwner) {
144 instanceId = grant->second.ownerInstanceId;
145 } else if (definition->second.instancing == AbilityInstancingPolicy::PerExecution) {
146 auto instance = nextInstanceId();
147 if (!instance) {
148 auto rollback = runtime_.cancel(execution.value(), tick);
149 if (!rollback) return Result<AbilityActivation>::failure(rollback.status());
150 return Result<AbilityActivation>::failure(instance.status());
151 }
152 instanceId = instance.value();
153 }
154 AbilityActivation activation{activationId.value(),
155 grantId,
156 instanceId,
157 execution.value(),
158 grant->second.state.ownerId,
159 definition->second.activationGroup};
160 activations_.emplace(activation.id, activation);
161 grant->second.state.cooldownRemaining = definition->second.cooldown;
162 return Result<AbilityActivation>::success(std::move(activation), Status::success(StatusCode::Applied));
163}

165std::vector<AbilityGrantId> AbilityRuntime::matchingGrants(std::string_view ownerId,
166 std::string_view gameplayEventTag) const {
167 std::vector<AbilityGrantId> result;
168 for (const auto& [id, grant] : grants_) {
169 if (grant.state.ownerId != ownerId) continue;
170 const auto definition = definitions_.find(grant.state.definitionId.format());
171 if (definition == definitions_.end()) continue;
172 const bool matches = std::any_of(
173 definition->second.triggers.begin(), definition->second.triggers.end(), [&](const auto& trigger) {
174 return tags::gameplayTagMatches(gameplayEventTag, trigger.gameplayTag, trigger.match);
175 });
176 if (matches) result.push_back(id);
177 }
178 return result;
179}

181Result<void> AbilityRuntime::advanceCooldowns(Duration delta) {
182 if (delta < Duration::zero())
183 return failure(DiagnosticCode::InvalidArgument, "Ability cooldown delta must be non-negative", "delta");
184 for (auto& [id, grant] : grants_) {
185 const auto remaining = grant.state.cooldownRemaining.nanoseconds();
186 grant.state.cooldownRemaining =
187 Duration::fromNanoseconds(delta.nanoseconds() >= remaining ? 0 : remaining - delta.nanoseconds());
188 }
189 return Result<void>::success(Status::success(delta.isZero() ? StatusCode::NoOp : StatusCode::Applied));
190}

192Result<std::size_t> AbilityRuntime::synchronize() {
193 const std::size_t before = activations_.size();
194 std::erase_if(activations_, [&](const auto& item) {
195 const auto* execution = runtime_.find(item.second.executionId);
196 return !execution || terminal(execution->phase());
197 });
198 const std::size_t removed = before - activations_.size();
199 return Result<std::size_t>::success(removed,
200 Status::success(removed == 0 ? StatusCode::NoOp : StatusCode::Applied));
201}

216bool AbilityRuntime::grantHasActiveActivation(AbilityGrantId grantId) const {
217 return std::any_of(activations_.begin(), activations_.end(), [&](const auto& item) {
218 if (item.second.grantId != grantId) return false;
219 const auto* execution = runtime_.find(item.second.executionId);
220 return execution && !terminal(execution->phase());
221 });
