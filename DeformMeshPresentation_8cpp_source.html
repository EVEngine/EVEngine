28Result<std::unique_ptr<DeformMeshPresentation>> DeformMeshPresentation::prepare(DeformTrack track, float ropeRadius) {
29 using Output = Result<std::unique_ptr<DeformMeshPresentation>>;
30 if (track.recipe()->coordinates() != BakeCoordinates::MetresRightHandedYUp)
31 return Output::failure(invalid("3D deformation presentation requires metres"));
32 const auto topology = track.topology();
33 const bool triangles = topology.primitive() == DeformPrimitive::Triangles;
34 if (!triangles && (!std::isfinite(ropeRadius) || ropeRadius <= 0))
35 return Output::failure(invalid("rope radius must be finite and positive"));
36 const auto count = triangles ? topology.vertexCount() * 2 : topology.indices().size() / 2 * 18;
37 const auto indexCount = triangles ? topology.indices().size() * 2 : topology.indices().size() / 2 * 96;
38 if (count > 2000000 || indexCount > 12000000)
39 return Output::failure(invalid("deformation presentation budget exceeded"));
40 try {
41 auto impl = std::make_unique<Impl>(Impl{std::move(track), topology, ropeRadius, {}, {}, {}, {}, {}, {}});
42 impl->sampled.resize(topology.vertexCount());
43 impl->positions.resize(count * 3);
44 impl->normals.resize(count * 3);
45 impl->uv.resize(count * 2);
46 impl->indices.reserve(indexCount);
47 if (triangles) {
48 impl->accumulated.resize(topology.vertexCount());
49 for (auto index : topology.indices()) impl->indices.push_back(index);
50 const auto offset = static_cast<std::uint32_t>(topology.vertexCount());
51 for (std::size_t i = 0; i < topology.indices().size(); i += 3) {
52 impl->indices.push_back(topology.indices()[i] + offset);
53 impl->indices.push_back(topology.indices()[i + 2] + offset);
54 impl->indices.push_back(topology.indices()[i + 1] + offset);
55 }
56 if (!topology.uv().empty())
57 for (std::size_t i = 0; i < count; ++i) {
58 auto value = topology.uv()[i % topology.vertexCount()];
59 impl->uv[i * 2] = value.u;
60 impl->uv[i * 2 + 1] = value.v;
61 }
62 } else {
63 for (std::uint32_t base = 0; base < count; base += 18) {
64 for (std::uint32_t side = 0; side < 8; ++side) {
65 const auto next = (side + 1) % 8;
66 for (auto index :
67 {base + side, base + next, base + 8 + side, base + next, base + 8 + next, base + 8 + side,
68 base + 16, base + next, base + side, base + 17, base + 8 + side, base + 8 + next})
69 impl->indices.push_back(index);
70 }
71 }
72 }
73 return Output::success(std::unique_ptr<DeformMeshPresentation>(new DeformMeshPresentation(std::move(impl))));
74 } catch (const std::bad_alloc&) {
75 return Output::failure(Diagnostic::error(DiagnosticCode::Failed, "deformation mesh allocation failed"));
76 }
77}
78Result<void> DeformMeshPresentation::rebuild(DeformPlayback& player) {
79 if (!impl_->track.isSameAsset(player.track())) return Result<void>::failure(invalid("foreign deformation player"));
80 auto sampled = player.sample(impl_->sampled);
81 if (!sampled) return sampled;
82 auto point = [&](std::uint32_t index) {
83 auto p = impl_->sampled[index];
84 return glm::dvec3(p.x, p.y, p.z);
85 };
86 bool representable = true;
87 auto write = [&](std::size_t index, glm::dvec3 p, glm::dvec3 n) {
88 for (int axis = 0; axis < 3; ++axis) {
89 if (!std::isfinite(p[axis]) || std::abs(p[axis]) > std::numeric_limits<float>::max() ||
90 !std::isfinite(n[axis])) {
91 representable = false;
92 continue;
93 }
94 impl_->positions[index * 3 + axis] = static_cast<float>(p[axis]);
95 impl_->normals[index * 3 + axis] = static_cast<float>(n[axis]);
96 }
97 };
98 if (impl_->topology.primitive() == DeformPrimitive::Triangles) {
99 std::fill(impl_->accumulated.begin(), impl_->accumulated.end(), glm::dvec3(0));
100 const auto indices = impl_->topology.indices();
101 for (std::size_t i = 0; i < indices.size(); i += 3) {
102 const auto normal =
103 glm::cross(point(indices[i + 1]) - point(indices[i]), point(indices[i + 2]) - point(indices[i]));
104 for (int corner = 0; corner < 3; ++corner) impl_->accumulated[indices[i + corner]] += normal;
105 }
106 for (std::size_t i = 0; i < impl_->sampled.size(); ++i) {
107 auto normal = impl_->accumulated[i];
108 const double length = glm::length(normal);
109 normal = length > 0 ? normal / length : glm::dvec3(0, 1, 0);
110 write(i, point(static_cast<std::uint32_t>(i)), normal);
111 write(i + impl_->sampled.size(), point(static_cast<std::uint32_t>(i)), -normal);
112 }
113 } else {
114 const auto indices = impl_->topology.indices();
115 for (std::size_t i = 0; i < indices.size(); i += 2) {
116 auto a = point(indices[i]), b = point(indices[i + 1]);
117 auto axis = b - a;
118 const double length = glm::length(axis);
119 axis = length > 0 ? axis / length : glm::dvec3(0, 1, 0);
120 const auto reference = std::abs(axis.y) < 0.9 ? glm::dvec3(0, 1, 0) : glm::dvec3(1, 0, 0);
121 const auto u = glm::normalize(glm::cross(axis, reference)), v = glm::cross(axis, u);
122 const auto base = i / 2 * 18;
123 for (std::size_t side = 0; side < 8; ++side) {
124 const double angle = double(side) * std::numbers::pi * 0.25;
125 const auto normal = std::cos(angle) * u + std::sin(angle) * v;
126 write(base + side, a + double(impl_->radius) * normal, normal);
127 write(base + 8 + side, b + double(impl_->radius) * normal, normal);
128 }
129 write(base + 16, a, -axis);
130 write(base + 17, b, axis);
131 }
132 }
133 if (!representable) return Result<void>::failure(invalid("deformation mesh position overflow"));
134 for (float value : impl_->positions)
135 if (!std::isfinite(value)) return Result<void>::failure(invalid("deformation mesh position overflow"));
136 for (float value : impl_->normals)
137 if (!std::isfinite(value)) return Result<void>::failure(invalid("deformation mesh normal overflow"));
138 return Result<void>::success();
139}

141 auto ready = rebuild(player);

151Result<void> DeformMeshPresentation::upload(DeformPlayback& player, graphics::Graphics& graphics,
152 graphics::Mesh& mesh) {
153 if (!graphics.describeMesh(&mesh)) return Result<void>::failure(invalid("foreign or released deformation mesh"));
154 auto ready = rebuild(player);
155 if (!ready) return ready;
156 if (!graphics.updateMeshVertices(&mesh, impl_->positions.data(), impl_->normals.data(), impl_->uv.data(),
157 static_cast<int>(impl_->positions.size() / 3), impl_->indices.data(),
158 static_cast<int>(impl_->indices.size())))
159 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Failed, "deformation mesh upload failed"));
160 return Result<void>::success();
161}
