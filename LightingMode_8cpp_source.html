27LightingPreset suggestedLightingPreset() {
28 if (const char* presetEnv = std::getenv("EVENGINE_LIGHTING_PRESET")) {
29 const std::string p = toLower(presetEnv);
30 if (p == "desktop") return LightingPreset::Desktop;
31 if (p == "mobile") return LightingPreset::Mobile;
32 if (p == "ci") return LightingPreset::Ci;
33 }
34
35 const char* platform = SDL_GetPlatform();
36 if (platform) {
37 const std::string os = toLower(platform);
38 if (os.find("android") != std::string::npos || os.find("ios") != std::string::npos)
39 return LightingPreset::Mobile;
40 }
41
42 if (auto* gpu = eve::cap::query<eve::caps::IGpuInfo>()) {
43 if (gpu->gpuReady()) {
44 const std::string type = toLower(gpu->gpuDeviceType());
45 const std::string name = gpu->gpuName();
46 if (type == "cpu" || containsInsensitive(name, "llvmpipe") || containsInsensitive(name, "lavapipe"))
47 return LightingPreset::Ci;
48 }
49 }
50 // Hosted CI often runs before GPU info is ready; prefer ForwardPlus there.
51 if (std::getenv("CI") || std::getenv("GITHUB_ACTIONS") || std::getenv("EVENGINE_CI")) return LightingPreset::Ci;
52 return LightingPreset::Desktop;
53}

55Result<LightingMode> lightingModeFromEnv() {
