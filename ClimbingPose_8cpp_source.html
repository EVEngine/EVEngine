23Vec3 toModelSpace(Vec3 world, const ClimbingPoseBinding& binding) {
24 const float x = world.x - binding.modelWorldOrigin.x;
25 const float y = world.y - binding.modelWorldOrigin.y;
26 const float z = world.z - binding.modelWorldOrigin.z;
27 const float cosine = std::cos(binding.modelYawRadians);
28 const float sine = std::sin(binding.modelYawRadians);
29 return {cosine * x - sine * z, y, sine * x + cosine * z};
30}

34eve::Result<ClimbingPoseResult> applyClimbingPose(animation::AnimSkeleton& skeleton, animation::AnimPose& pose,
35 const ClimbingAdvance& advance, const ClimbingPoseBinding& binding) {
36 if (!std::isfinite(binding.modelYawRadians))
37 return poseFailure("model yaw must be finite", "binding.modelYawRadians");
38 const int boneCount = skeleton.getBoneCount();
39 if (binding.enableLeftHand && !validChain(binding.leftArm, boneCount))
40 return poseFailure("left arm chain contains an invalid bone", "binding.leftArm");
41 if (binding.enableRightHand && !validChain(binding.rightArm, boneCount))
42 return poseFailure("right arm chain contains an invalid bone", "binding.rightArm");
43
44 ClimbingPoseResult result;
45 const float legacyWeight = std::clamp(advance.contactWeight, 0.f, 1.f);
46 const float authoredLeft = std::clamp(advance.leftHandWeight, 0.f, 1.f);
47 const float authoredRight = std::clamp(advance.rightHandWeight, 0.f, 1.f);
48 const bool hasPerHandWeights = authoredLeft > 0.f || authoredRight > 0.f;
49 const float leftWeight = hasPerHandWeights ? authoredLeft : legacyWeight;
50 const float rightWeight = hasPerHandWeights ? authoredRight : legacyWeight;
51 result.weight = std::max(leftWeight, rightWeight);
52 if (result.weight <= 0.f)
53 return eve::Result<ClimbingPoseResult>::success(std::move(result), eve::Status::success(eve::StatusCode::NoOp));
54
55 animation::AnimConstraintStack constraints(&skeleton);
56 if (binding.enableLeftHand && leftWeight > 0.f) {
57 const Vec3 target = toModelSpace(advance.leftHandAnchor, binding);
58 constraints.addTwoBoneIK(binding.leftArm.root, binding.leftArm.mid, binding.leftArm.tip, target.x, target.y,
59 target.z, leftWeight);
60 result.leftHandApplied = true;
61 }
62 if (binding.enableRightHand && rightWeight > 0.f) {
63 const Vec3 target = toModelSpace(advance.rightHandAnchor, binding);
64 constraints.addTwoBoneIK(binding.rightArm.root, binding.rightArm.mid, binding.rightArm.tip, target.x, target.y,
65 target.z, rightWeight);
66 result.rightHandApplied = true;
67 }
68 constraints.apply(&pose);
69 return eve::Result<ClimbingPoseResult>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
70}
