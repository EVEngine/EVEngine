39Result<std::size_t> BakedParticlePlayback::sample(std::span<BakedParticleSample> output) {
40 if (output.size() < scratch_.size())
41 return Result<std::size_t>::failure(Diagnostic::error(
42 DiagnosticCode::InvalidArgument, "insufficient particle output extent", "bakedParticlePlayback"));
43 auto sampled = track_.sample(static_cast<float>(clock_.time()), scratch_);
44 if (!sampled) return Result<std::size_t>::failure(sampled.status());
45 const auto count = sampled.value();
46 const auto p = transform_.placement();
47 const double angle = 2 * std::atan2(double(p.rotationZ), double(p.rotationW));
48 const double c = std::cos(angle), s = std::sin(angle);
49 constexpr double radiansToDegrees = 180. / 3.14159265358979323846;
50 for (std::size_t i = 0; i < count; ++i) {
51 auto& value = scratch_[i].pose;
52 const double cx = double(value.x) + double(value.w) * .5;
53 const double cy = double(value.y) + double(value.h) * .5;
54 const double x = c * cx - s * cy + p.x - double(value.w) * .5;
55 const double y = s * cx + c * cy + p.y - double(value.h) * .5;
56 const double rotation = double(value.rotation) + angle * radiansToDegrees;
57 for (double candidate : {x, y, rotation})
58 if (!std::isfinite(candidate) || std::abs(candidate) > std::numeric_limits<float>::max())
59 return Result<std::size_t>::failure(Diagnostic::error(
60 DiagnosticCode::InvalidArgument, "particle placement overflow", "bakedParticlePlayback"));
61 value.x = static_cast<float>(x);
62 value.y = static_cast<float>(y);
63 value.rotation = static_cast<float>(rotation);
64 }
65 std::copy_n(scratch_.begin(), count, output.begin());
66 return Result<std::size_t>::success(count);
67}
