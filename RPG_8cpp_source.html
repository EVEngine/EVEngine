548void RPG::expose(ssq::Table &table) {
549 const HSQUIRRELVM vm = table.getHandle();
550 auto cls = table.addClass(name, RPG::create, false);
551 expose(cls);
552
553 auto actor = table.addClass<RPGActor>(
554 "RPGActor", std::function<RPGActor *()>([]() -> RPGActor * { return nullptr; }), true);
555 // 绑定为 float：SimpleSquirrel 只有在 squirrel 以 SQUSEDOUBLE 编译时才认识
