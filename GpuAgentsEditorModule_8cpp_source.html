24editor::Result<editor::AutomationOwnedTarget> makeTarget(const editor::TargetId& id,
25 const editor::EditorValue::Object& request) {
26 auto target = std::make_unique<gpuagents_editing::GpuAgentsDocumentTarget>(id.value());
27 const auto snapshot = request.find("snapshot");
28 if (snapshot != request.end()) {
29 auto loaded = target->loadSnapshot(snapshot->second);
30 if (!loaded.ok()) {
31 return editor::Result<editor::AutomationOwnedTarget>::failure(loaded.status());
32 }
33 }
34 editor::AutomationOwnedTarget owned;
35 owned.target = std::move(target);
36 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
37}

76GpuAgentsPreviewSnapshot GpuAgentsPreviewService::build(const GpuAgentsDocumentTarget& target,
77 const GpuAgentsPreviewRequest& request) const {
78 GpuAgentsPreviewSnapshot out;
79 out.documentRevision = target.describe().revision;
80 out.diagnostics = target.validate();
81 if (std::any_of(out.diagnostics.begin(), out.diagnostics.end(),
82 [](const auto& d) { return d.severity() == gpuagents_editing::DiagnosticSeverity::Error; })) {
83 out.status = EditorStatus::Failed;
84 return out;
85 }
86 if (request.agentCount <= 0 || request.seconds < 0.0 || request.fixedStep <= 0.0 || request.maximumSteps == 0) {
87 out.status = EditorStatus::Failed;
88 out.diagnostics.push_back(eve::editing::ruleDiagnostic(
89 eve::DiagnosticCode::InvalidArgument, gpuagents_editing::RuleId("editor.gpuagents.preview.request"),
90 gpuagents_editing::DiagnosticSeverity::Error, "Invalid GPU Agents preview request"));
91 return out;
92 }
93
94 gpuagents::GpuAgentWorld world;
95 auto backend = std::make_unique<gpuagents::EffectBackend>();
96 gpuagents_editing::GpuAgentsRuntimeApplier applier;
97 auto applied = applier.apply(target, backend.get(), &world);
98 if (!applied.ok()) {
99 out.status = EditorStatus::Failed;
100 out.diagnostics.push_back(eve::editing::ruleDiagnostic(
101 eve::DiagnosticCode::Failed, gpuagents_editing::RuleId("editor.gpuagents.preview.apply"),
102 gpuagents_editing::DiagnosticSeverity::Error, "Failed to apply GPU Agents document to preview runtime"));
103 return out;
104 }
105
106 const int count = std::min(request.agentCount, backend->simulation().capacity());
107 auto spawned = backend->simulation().initialize(count, [&](int i) {
108 gpuagents::AgentState s;
109 const float a = static_cast<float>(i) * 0.6180339f;
110 const float r = static_cast<float>(request.spawnSpread) * (0.2f + 0.8f * std::fmod(a, 1.f));
111 s.position = {static_cast<float>(request.centerX) + std::cos(a * 6.2831853f) * r,
112 static_cast<float>(request.centerY),
113 static_cast<float>(request.centerZ) + std::sin(a * 6.2831853f) * r};
114 s.velocity = {std::cos(a), 0.f, std::sin(a)};
115 if (target.settings().kind == "Petal") {
116 s.velocity = {0.f, -1.f, 0.f};
117 s.customData[1] = 1.f;
118 s.position.y += 2.f;
119 }
120 if (target.settings().kind == "LifeNetwork") s.position.y = static_cast<float>(request.centerY);
121 return s;
122 });
123 if (!spawned.ok()) {
124 out.status = EditorStatus::Failed;
125 return out;
126 }
127
128 auto registered = world.registerBackend("preview", backend.get());
129 if (!registered.ok()) {
130 out.status = EditorStatus::Failed;
131 return out;
132 }
133 const double step = request.fixedStep;
134 std::uint32_t steps = 0;
135 double t = 0.0;
136 while (t + 1e-12 < request.seconds && steps < request.maximumSteps) {
137 auto stepped = world.stepAll(static_cast<float>(step));
138 if (!stepped.ok()) {
139 out.status = EditorStatus::Failed;
140 return out;
141 }
142 t += step;
143 ++steps;
144 }
145 out.simulatedSeconds = t;
146 for (const auto& s : backend->simulation().states()) {
147 if (!s.alive) continue;
148 GpuAgentsPreviewAgent a;
149 a.x = s.position.x;
150 a.y = s.position.y;
151 a.z = s.position.z;
152 a.vx = s.velocity.x;
153 a.vy = s.velocity.y;
154 a.vz = s.velocity.z;
155 a.age = s.age;
156 a.custom0 = s.customData[0];
157 out.agents.push_back(a);
158 }
159 if (target.settings().kind == "LifeNetwork") {
160 for (const auto& c : world.surface().lifeField) out.lifeTrailEnergy += c.r;
161 }
162 out.status = EditorStatus::Applied;
163 return out;
164}
