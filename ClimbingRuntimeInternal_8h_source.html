103inline Vec3 terminalVelocityFor(const ClimbingActionDefinition& action, Vec3 actualDelta, float inverseDelta) {
104 Vec3 velocity = actualDelta * inverseDelta;
105 switch (action.landingPolicy) {
106 case ClimbingLandingPolicy::PreserveMomentum: break;
107 case ClimbingLandingPolicy::MatchGround: velocity.y = 0.f; break;

111 case ClimbingTerminalVelocityPolicy::Preserve: break;

114 }

122inline float signedHorizontalAngle(Vec3 from, Vec3 to) {
123 from = normalizedHorizontal(from);
124 to = normalizedHorizontal(to);
125 if (lengthSquared(from) <= epsilon || lengthSquared(to) <= epsilon) return 0.f;
126 const float crossY = from.z * to.x - from.x * to.z;
127 const float dot = std::clamp(from.x * to.x + from.z * to.z, -1.f, 1.f);
128 return std::atan2(crossY, dot);
129}

208inline std::optional<int> parseTagSelector(std::string_view selector, std::string_view prefix) {
209 if (!selector.starts_with(prefix)) return std::nullopt;
210 const std::string_view digits = selector.substr(prefix.size());
211 if (digits.empty()) return std::nullopt;
212 int value = 0;
213 const auto parsed = std::from_chars(digits.data(), digits.data() + digits.size(), value);
214 if (parsed.ec != std::errc{} || parsed.ptr != digits.data() + digits.size()) return std::nullopt;
215 return value;
216}

235inline std::int64_t selectionCost(const ClimbingProfile& profile, const ClimbingActionDefinition& action,
236 const ClimbingPose& pose, Vec3 targetDirection, Vec3 targetDelta, float heightError,
237 float distance, std::int64_t stableTieBreak = 0) {
238 const Vec3 forward = normalizedHorizontal(pose.forward);
239 targetDirection = normalizedHorizontal(targetDirection);
240 if (lengthSquared(targetDirection) <= epsilon) targetDirection = forward;
241
242 Vec3 intentDirection =
243 normalizedHorizontal(pose.inputMode == ClimbingInputMode::Precision ? pose.lookIntent : pose.moveIntent);
244 if (lengthSquared(intentDirection) <= epsilon) intentDirection = forward;
245
246 const float directionDot = std::clamp(forward.x * targetDirection.x + forward.z * targetDirection.z, -1.f, 1.f);
247 const float intentDot =
248 std::clamp(intentDirection.x * targetDirection.x + intentDirection.z * targetDirection.z, -1.f, 1.f);
249 const bool precision = pose.inputMode == ClimbingInputMode::Precision;
250 const float assistScale = 1.f - profile.autoAssistStrength * (precision ? 0.15f : 0.5f);
251 const float directionModeScale = precision ? 1.5f : 0.75f;
252 const float speedModeScale = precision ? 0.5f : 1.5f;
253 const float translationModeScale = precision ? 1.25f : 0.75f;
254 const float rotationModeScale = precision ? 1.5f : 0.75f;
255 const float intentModeScale = precision ? 1.5f : 0.5f;
256 const float horizontalDelta = std::sqrt(targetDelta.x * targetDelta.x + targetDelta.z * targetDelta.z);
257 const float warpTranslation = std::sqrt(horizontalDelta * horizontalDelta + targetDelta.y * targetDelta.y);
258 const float warpRotation = std::acos(directionDot);
259 const float intentMismatch = 1.f - intentDot;
260
261 return static_cast<std::int64_t>(action.selectionBias) * 1000000ll +
262 weightedMillimeters((1.f - directionDot) * assistScale * directionModeScale,
263 profile.scoreWeights.direction) +
264 weightedMillimeters(std::fabs(pose.speed - action.minSpeed) * speedModeScale,
265 profile.scoreWeights.approachSpeed) +
266 weightedMillimeters(heightError, profile.scoreWeights.height) +
267 weightedMillimeters(distance, profile.scoreWeights.distance) +
268 weightedMillimeters(warpTranslation * translationModeScale, profile.scoreWeights.warpTranslation) +
269 weightedMillimeters(warpRotation * rotationModeScale, profile.scoreWeights.warpRotation) +
270 weightedMillimeters(intentMismatch * intentModeScale, profile.scoreWeights.intentMismatch) + stableTieBreak;
271}
