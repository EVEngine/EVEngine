157template <class AuthorityQuery>
158[[nodiscard]] std::optional<ConditionResult> resolveAuthorityAllow(std::string_view name, const Value& arguments,
159 AuthorityQuery&& authorityQuery) {
160 if (name != kAuthorityAllowPolicy) return std::nullopt;
161 auto scope = authorityAllowScope(arguments);
162 if (!scope) {
163 return ConditionResult::failed(ConditionReasonCode::InvalidCondition, {},
164 Value::Object{{"policy", Value(std::string(kAuthorityAllowPolicy))},
165 {"error", Value("authority.allow requires string scope")}});
166 }
167 auto allowed = authorityQuery(std::string_view(*scope));
168 if (!allowed) {
169 return ConditionResult::failed(ConditionReasonCode::AuthorityUnavailable, {},
170 Value::Object{{"scope", Value(*scope)}});
171 }
172 if (*allowed) {
173 return ConditionResult::success(Value(true), Value::Object{{"scope", Value(*scope)}});
174 }
175 return ConditionResult::failed(ConditionReasonCode::AuthorityDenied, Value(false),
176 Value::Object{{"scope", Value(*scope)}});
177}
