38// WebGPU (WGSL) X-ray mesh shaders. Must match the engine's mesh3d Frame UBO
39// (group 0 binding 0, std140 = Mesh3DUBO) and shared bindings (main sampler 7,
40// scene depth 9). X-ray params are packed into texBomb/parallax/clipInfo by the
41// backend flush (see bindMeshUniforms("xray") ordering below).
42const char *kXrayMeshVertWgsl = R"wgsl(
43struct VSIn {
44 @location(0) pos: vec3f,
45 @location(1) normal: vec3f,
46 @location(2) uv: vec2f,
47};
48struct Light3D {
49 posRadius: vec4f,
50 color: vec4f,
51};
52struct Frame {
53 mvp: mat4x4f,
54 model: mat4x4f,
55 lightDir: vec4f,
56 lightColor: vec4f,
57 tint: vec4f,
58 cameraPos: vec4f,
59 ambient: vec4f,
60 lights: array<Light3D, 8>,
61 texBomb: vec4f,
62 parallax: vec4f,
63 view: mat4x4f,
64 clipInfo: vec4f,
65};
66struct VSOut {
67 @builtin(position) pos: vec4f,
68 @location(0) vNormal: vec3f,
69 @location(1) vUV: vec2f,
70 @location(2) vTint: vec4f,
71 @location(3) vWorldPos: vec3f,
72 @location(4) vCameraPos: vec3f,
73};
74@group(0) @binding(0) var<uniform> ubo: Frame;
75fn inverse3x3(m: mat3x3f) -> mat3x3f {
76 let a = m[0].x; let b = m[1].x; let c = m[2].x;
77 let d = m[0].y; let e = m[1].y; let f = m[2].y;
78 let g = m[0].z; let h = m[1].z; let i = m[2].z;
79 let det = a * (e * i - f * h) - b * (d * i - f * g) + c * (d * h - e * g);
80 return mat3x3f(
81 vec3f((e * i - f * h) / det, (f * g - d * i) / det, (d * h - e * g) / det),
82 vec3f((c * h - b * i) / det, (a * i - c * g) / det, (b * g - a * h) / det),
83 vec3f((b * f - c * e) / det, (c * d - a * f) / det, (a * e - b * d) / det),
84 );
85}
86@vertex
87fn vs_main(in: VSIn) -> VSOut {
88 var out: VSOut;
89 out.pos = ubo.mvp * vec4f(in.pos, 1.0);
90 let world = ubo.model * vec4f(in.pos, 1.0);
91 out.vWorldPos = world.xyz;
92 let nrm = transpose(inverse3x3(mat3x3f(ubo.model[0].xyz, ubo.model[1].xyz, ubo.model[2].xyz))) * in.normal;
93 out.vNormal = normalize(nrm);
94 out.vUV = in.uv;
95 out.vTint = ubo.tint;
96 out.vCameraPos = ubo.cameraPos.xyz;
97 return out;
98}
99)wgsl";

101const char *kXrayMeshFragWgsl = R"wgsl(
102struct Light3D {
103 posRadius: vec4f,
104 color: vec4f,
105};
106struct Frame {
107 mvp: mat4x4f,
108 model: mat4x4f,
109 lightDir: vec4f,
110 lightColor: vec4f,
111 tint: vec4f,
112 cameraPos: vec4f,
113 ambient: vec4f,
114 lights: array<Light3D, 8>,
115 texBomb: vec4f,
116 parallax: vec4f,
117 view: mat4x4f,
118 clipInfo: vec4f,
119};
120struct FSIn {
121 @location(0) vNormal: vec3f,
122 @location(1) vUV: vec2f,
123 @location(2) vTint: vec4f,
124 @location(3) vWorldPos: vec3f,
125 @location(4) vCameraPos: vec3f,
126};
127@group(0) @binding(0) var<uniform> ubo: Frame;
128@group(0) @binding(7) var mainSamp: sampler;
129@group(0) @binding(9) var sceneDepth: texture_depth_2d;
130
131struct FSOut {
132 @location(0) color: vec4f,
133};
134
135@fragment
136fn fs_main(in: FSIn, @builtin(position) fragPos: vec4f) -> FSOut {
137 let color = ubo.texBomb.xyz;
138 let alpha = clamp(ubo.texBomb.w, 0.0, 1.0);
139 let bias = max(ubo.parallax.x, 0.0);
140 let screenW = max(ubo.parallax.y, 1.0);
141 let screenH = max(ubo.parallax.z, 1.0);
142 let rimStrength = clamp(ubo.parallax.w, 0.0, 1.0);
143 let rimPower = max(ubo.clipInfo.z, 0.1);
144
145 let uv = fragPos.xy / vec2f(screenW, screenH);
146 let sceneZ = textureSampleLevel(sceneDepth, mainSamp, uv, 0.0);
147 let charZ = fragPos.z;
148 let occ = smoothstep(sceneZ, sceneZ + bias, charZ);
149 if (occ < 0.5) {
150 discard;
151 }
152 let N = normalize(in.vNormal);
153 let V = normalize(in.vCameraPos - in.vWorldPos);
154 let rim = pow(clamp(1.0 - max(dot(N, V), 0.0), 0.0, 1.0), rimPower);
155 let edge = mix(1.0, rim, rimStrength);
156 var out: FSOut;
157 out.color = vec4f(color, alpha * edge * occ);
158 return out;
159}
160)wgsl";

182bool styleSupports(const std::string &style, const std::string &feature) {
183 const StyleDefinition *def = findStyleDefinition(style);
184 if (!def) return false;
185 if (feature == "post") return def->post;
186 if (feature == "mesh") return def->mesh;
187 if (feature == "cpu") return def->cpu;
188 if (feature == "depth") return def->depth;
189 if (feature == "normal") return def->normal;

437graphics::Shader *createPostShader(graphics::Graphics *gfx, const std::string &style) {
438 if (!gfx) throw eve::Exception("createPostShader: null graphics");
439 if (isEffectStyle(style)) return createEffectPostShader(gfx, style);
440 const StyleDefinition *def = findStyleDefinition(style);
441 if (!def) throw eve::Exception("createPostShader: unknown style '%s'", style.c_str());
442 if (!def->post)
443 throw eve::Exception("createPostShader: style '%s' has no post technique", style.c_str());
444
445 if (gfx->getBackendName() == "webgpu") {
446 const char *body = style == "cartoon" ? shaders::kCartoon
447 : style == "watercolor" ? shaders::kWatercolor
448 : style == "ink" ? shaders::kInk
449 : shaders::kPixel;
450 graphics::Shader *sh = gfx->newShaderFromWgsl({}, std::string(shaders::kCommon) + body);
451 if (!sh || !sh->gpuHandle)
452 throw eve::Exception("createPostShader: failed to create '%s' WGSL", style.c_str());
453 bindPostUniforms(sh, style);
454 return sh;
455 }
456
457 std::vector<uint32_t> frag;
458 if (style == "cartoon")
459 frag = copySpv(cartoon_post_frag_spv, cartoon_post_frag_spv_count);
460 else if (style == "watercolor")
461 frag = copySpv(watercolor_post_frag_spv, watercolor_post_frag_spv_count);
462 else if (style == "ink")
463 frag = copySpv(ink_post_frag_spv, ink_post_frag_spv_count);
464 else
465 frag = copySpv(pixel_post_frag_spv, pixel_post_frag_spv_count);
466
467 graphics::Shader *sh = gfx->newShaderFromSpv({}, frag);
468 if (!sh || !sh->gpuHandle)
469 throw eve::Exception("createPostShader: failed to create '%s'", style.c_str());
470 bindPostUniforms(sh, style);
471 return sh;
472}
