38void callScriptHandler(ssq::Function &fn, const std::string &kind, const UIEvent &ev) {
39 HSQUIRRELVM vm = fn.getHandle();
40 if (!vm) return;
41 const SQInteger top = sq_gettop(vm);
42 sq_pushobject(vm, fn.getRaw());
43 sq_pushroottable(vm); // env
44 if (kind == "click") {
45 if (SQ_FAILED(sq_call(vm, 1, SQFalse, SQTrue))) {
46 sq_settop(vm, top);
47 return;
48 }
49 } else {
50 sq_pushstring(vm, kind.c_str(), -1);
51 if (kind == "toggle") sq_pushbool(vm, ev.toggleValue ? SQTrue : SQFalse);
52 else if (kind == "value") sq_pushfloat(vm, ev.floatValue);
53 else sq_pushstring(vm, ev.textValue.c_str(), -1);
54 if (SQ_FAILED(sq_call(vm, 3, SQFalse, SQTrue))) {
55 sq_settop(vm, top);
56 return;
57 }
58 }
59 sq_settop(vm, top);
60}

190void UI::updateHostTweens() {
191 if (hostTweens_.empty()) return;
192 const double now =
193 std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now().time_since_epoch())
194 .count();
195 for (auto &t : hostTweens_) {
196 if (!t.host) continue;
197 auto m = t.host->meta();
198 const double elapsed = now - t.startMs;
199 if (t.durationMs <= 0.0 || elapsed >= t.durationMs) {
200 m->hasPos = true;
201 m->posX = t.toX;
202 m->posY = t.toY;
203 t.host = nullptr; // done; removed below
204 continue;
205 }
206 const float k = float(elapsed / t.durationMs);
207 const float ease = k * k * (3.f - 2.f * k); // smoothstep
208 m->hasPos = true;
209 m->posX = t.fromX + (t.toX - t.fromX) * ease;
210 m->posY = t.fromY + (t.toY - t.fromY) * ease;
211 }
212 hostTweens_.erase(std::remove_if(hostTweens_.begin(), hostTweens_.end(),
213 [](const HostTween &t) { return t.host == nullptr; }),
214 hostTweens_.end());
215}

227void UI::fireScriptHandlers(const UIEvent &ev) {
228 for (const auto &h : scriptHandlers_) {
229 if (h.hostName != ev.hostName || h.nodeId != ev.nodeId) continue;
230 if (h.kind == "click" && ev.kind == "click") {
231 ssq::Function fn = h.fn;
232 callScriptHandler(fn, "click", ev);
233 } else if (h.kind == ev.kind &&
234 (ev.kind == "toggle" || ev.kind == "value" || ev.kind == "text")) {
235 ssq::Function fn = h.fn;
236 callScriptHandler(fn, ev.kind, ev);
237 }
238 }

375FlexAlign parseFlexAlign(const std::string &align) {
376 const std::string a = toLowerCopy(align);
377 if (a == "center") return FlexAlign::Center;
378 if (a == "end" || a == "right" || a == "bottom") return FlexAlign::End;
379 if (a == "stretch") return FlexAlign::Stretch;
380 return FlexAlign::Start;

383FlexJustify parseFlexJustify(const std::string &justify) {

385 if (j == "center") return FlexJustify::Center;

387 if (j == "spacebetween" || j == "space-between" || j == "between")
388 return FlexJustify::SpaceBetween;
389 if (j == "spacearound" || j == "space-around" || j == "around") return FlexJustify::SpaceAround;
390 return FlexJustify::Start;

938void nodeToJson(const UIHost::Tree &tree, const UINode &n, Poco::JSON::Object &o) {
939 o.set("type", nodeTypeName(n.type));
940 if (!n.id.empty()) o.set("id", n.id);
941 if (!n.key.empty()) o.set("key", n.key);
942 if (!n.text.empty()) o.set("text", n.text);
943 if (!n.valueText.empty()) o.set("valueText", n.valueText);
944 if (!n.visible) o.set("visible", false);
945 if (n.checked) o.set("checked", true);
946 if (!n.open) o.set("open", false);
947 if (n.value != 0.f) o.set("value", n.value);
948 if (n.minValue != 0.f) o.set("minValue", n.minValue);
949 if (n.maxValue != 1.f) o.set("maxValue", n.maxValue);
950 if (n.sizeX != 0.f) o.set("sizeX", n.sizeX);
951 if (n.sizeY != 0.f) o.set("sizeY", n.sizeY);
952 if (n.marginL != 0.f || n.marginT != 0.f || n.marginR != 0.f || n.marginB != 0.f)
953 o.set("margin", Poco::Dynamic::Array({n.marginL, n.marginT, n.marginR, n.marginB}));
954 if (n.paddingL != 0.f || n.paddingT != 0.f || n.paddingR != 0.f || n.paddingB != 0.f)
955 o.set("padding", Poco::Dynamic::Array({n.paddingL, n.paddingT, n.paddingR, n.paddingB}));
956 if (n.minSizeX != 0.f) o.set("minSizeX", n.minSizeX);
957 if (n.minSizeY != 0.f) o.set("minSizeY", n.minSizeY);
958 if (n.maxSizeX != 0.f) o.set("maxSizeX", n.maxSizeX);
959 if (n.maxSizeY != 0.f) o.set("maxSizeY", n.maxSizeY);
960 if (n.percentW != 0.f) o.set("percentW", n.percentW);
961 if (n.percentH != 0.f) o.set("percentH", n.percentH);
962 if (n.absolute) {
963 o.set("absolute", true);
964 o.set("anchorX", n.anchorX);
965 o.set("anchorY", n.anchorY);
966 o.set("posX", n.posX);
967 o.set("posY", n.posY);
968 }
969 if (n.wrapWidth != 0.f) o.set("wrapWidth", n.wrapWidth);
970 if (n.flexDirection != FlexDirection::Row)
971 o.set("flexDirection", n.flexDirection == FlexDirection::Column ? "column" : "row");
972 if (n.alignItems != FlexAlign::Start) o.set("alignItems", int(n.alignItems));
973 if (n.justifyContent != FlexJustify::Start) o.set("justifyContent", int(n.justifyContent));
974 if (n.gap >= 0.f) o.set("gap", n.gap);
975 if (n.flexGrow != 0.f) o.set("flexGrow", n.flexGrow);
976 if (n.tintR != 1.f || n.tintG != 1.f || n.tintB != 1.f || n.tintA != 1.f)
977 o.set("tint", Poco::Dynamic::Array({n.tintR, n.tintG, n.tintB, n.tintA}));
978 if (n.borderL != 0.f || n.borderT != 0.f || n.borderR != 0.f || n.borderB != 0.f)
979 o.set("border", Poco::Dynamic::Array({n.borderL, n.borderT, n.borderR, n.borderB}));
980 if (n.cornerRadius != 0.f) o.set("cornerRadius", n.cornerRadius);
981 if (n.uv0x != 0.f || n.uv0y != 0.f || n.uv1x != 1.f || n.uv1y != 1.f)
982 o.set("uv", Poco::Dynamic::Array({n.uv0x, n.uv0y, n.uv1x, n.uv1y}));
983
984 Poco::JSON::Array children;
985 for (int c = n.firstChild; c >= 0; c = tree.nodes[size_t(c)].nextSibling) {
986 Poco::JSON::Object child;
987 nodeToJson(tree, tree.nodes[size_t(c)], child);
988 children.add(child);
989 }
990 if (children.size() > 0) o.set("children", children);
991}

1155ssq::Object UI::callPickHandler() {
1156 Runtime *rt = ModuleManager::runtime();
1157 if (!rt) return {};
1158 HSQUIRRELVM squirrel = rt->handle();
1159 const SQInteger top = sq_gettop(squirrel);
1160 sq_pushroottable(squirrel);
1161 sq_pushstring(squirrel, "eve", -1);
1162 if (SQ_FAILED(sq_get(squirrel, -2)) ||
1163 sq_gettype(squirrel, -1) != OT_TABLE) {
1164 sq_settop(squirrel, top);
1165 return {};
1166 }
1167 sq_pushstring(squirrel, "_inspectorPickHandler", -1);
1168 if (SQ_FAILED(sq_get(squirrel, -2)) ||
1169 (sq_gettype(squirrel, -1) != OT_CLOSURE &&
1170 sq_gettype(squirrel, -1) != OT_NATIVECLOSURE)) {
1171 sq_settop(squirrel, top);
1172 return {};
1173 }
1174 sq_pushroottable(squirrel); // environment
1175 if (SQ_FAILED(sq_call(squirrel, 1, SQTrue, SQTrue))) {
1176 sq_settop(squirrel, top);
1177 return {};
1178 }
1179 if (sq_gettype(squirrel, -1) != OT_INSTANCE) {
1180 sq_settop(squirrel, top);
1181 return {};
1182 }
1183 ssq::Object out(squirrel);
1184 sq_getstackobj(squirrel, -1, &out.getRaw());
1185 sq_addref(squirrel, &out.getRaw());
1186 sq_settop(squirrel, top);
1187 return out;
1188}
