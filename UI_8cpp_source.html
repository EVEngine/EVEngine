43void callScriptHandler(ssq::Function &fn, const std::string &kind, const UIEvent &ev) {
44 HSQUIRRELVM vm = fn.getHandle();
45 if (!vm) return;
46 const SQInteger top = sq_gettop(vm);
47 sq_pushobject(vm, fn.getRaw());
48 sq_pushroottable(vm); // env
49 if (kind == "click") {
50 if (SQ_FAILED(sq_call(vm, 1, SQFalse, SQTrue))) {
51 sq_settop(vm, top);
52 return;
53 }
54 } else {
55 sq_pushstring(vm, kind.c_str(), -1);
56 if (kind == "toggle") sq_pushbool(vm, ev.toggleValue ? SQTrue : SQFalse);
57 else if (kind == "value") sq_pushfloat(vm, ev.floatValue);
58 else sq_pushstring(vm, ev.textValue.c_str(), -1);
59 if (SQ_FAILED(sq_call(vm, 3, SQFalse, SQTrue))) {
60 sq_settop(vm, top);
61 return;
62 }
63 }
64 sq_settop(vm, top);
65}

247void UI::updateHostTweens() {
248 if (hostTweens_.empty()) return;
249 const double now =
250 std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now().time_since_epoch())
251 .count();
252 for (auto &t : hostTweens_) {
253 auto host = UIHost::resolve(t.host);
254 if (!host) continue;
255 auto m = host->get().meta();
256 const double elapsed = now - t.startMs;
257 if (t.durationMs <= 0.0 || elapsed >= t.durationMs) {
258 m->hasPos = true;
259 m->posX = t.toX;
260 m->posY = t.toY;
261 t.host = {}; // done; removed below
262 continue;
263 }
264 const float k = float(elapsed / t.durationMs);
265 const float ease = k * k * (3.f - 2.f * k); // smoothstep
266 m->hasPos = true;
267 m->posX = t.fromX + (t.toX - t.fromX) * ease;
268 m->posY = t.fromY + (t.toY - t.fromY) * ease;
269 }
270 hostTweens_.erase(std::remove_if(hostTweens_.begin(), hostTweens_.end(),
271 [](const HostTween &t) { return !UIHost::resolve(t.host).has_value(); }),
272 hostTweens_.end());
273}

285void UI::fireScriptHandlers(const UIEvent &ev) {
286 for (const auto &h : scriptHandlers_) {
287 if (h.hostName != ev.hostName || h.nodeId != ev.nodeId) continue;
288 if (h.kind == "click" && ev.kind == "click") {
289 ssq::Function fn = h.fn;
290 callScriptHandler(fn, "click", ev);
291 } else if (h.kind == ev.kind &&

293 ssq::Function fn = h.fn;
294 callScriptHandler(fn, ev.kind, ev);

1041UI::NinePatchResource *UI::loadNinePatch(const std::string &path) {
1042 if (auto found = ninePatches_.find(path); found != ninePatches_.end())
1043 return &found->second;
1044 auto *images = eve::ModuleManager::getInstance<eve::image::Image>("Image");
1045 auto *graphics = eve::ModuleManager::getInstance<eve::graphics::Graphics>("Graphics");
1046 if (!images || !graphics) return nullptr;
1047 try {
1048 eve::ref<eve::image::ImageData> source(images->newImageDataFromFile(path));
1049 NinePatchResource resource;
1050 std::string error;
1051 if (!parseNinePatch(*source, resource.info, &error)) {
1052 std::fprintf(stderr, "[ui] invalid .9.png '%s': %s\n", path.c_str(),
1053 error.c_str());
1054 return nullptr;
1055 }
1056 auto cropped = stripNinePatchBorder(*source);
1057 if (!cropped) return nullptr;
1058 resource.texture = graphics->newTexture(cropped.get());
1059 if (!resource.texture) return nullptr;
1060 resource.textureId = registerTexture(resource.texture);
1061 if (resource.textureId == 0) {
1062 graphics->releaseTexture(resource.texture);
1063 return nullptr;
1064 }
1065 auto inserted = ninePatches_.emplace(path, std::move(resource));
1066 return &inserted.first->second;
1067 } catch (const std::exception &e) {
1068 std::fprintf(stderr, "[ui] failed to load .9.png '%s': %s\n", path.c_str(), e.what());
1069 return nullptr;
1070 }
1071}

1073void UI::releaseNinePatches() {
1074 auto *graphics = eve::ModuleManager::getInstance<eve::graphics::Graphics>("Graphics");
1075 for (auto &[path, resource] : ninePatches_) {
1076 (void)path;
1077 if (backend_ && resource.textureId != 0)
1078 backend_->unregisterTexture(resource.textureId);
1079 if (graphics && resource.texture) graphics->releaseTexture(resource.texture);
1080 }
1081 ninePatches_.clear();
1082}
