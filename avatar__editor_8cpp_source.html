40 editor::EditorResult<editor::AutomationOwnedTarget> create(
41 const editor::TargetId& target, std::string_view type, const editor::EditorValue::Object& request) override {
42 (void)type;
43 auto document = std::make_unique<avatar_editing::AvatarDocumentTarget>(target.value());
44 const std::string kind = stringField(request, "kind");
45 const std::string source = stringField(request, "source");
46 if (!kind.empty() || !source.empty()) {
47 auto applied = apply(*document, document->makeSetSource(kind.empty() ? document->kind() : kind, source));
48 if (!applied.ok())
49 return editor::EditorResult<editor::AutomationOwnedTarget>::failure(applied.status());
50 }
51 const std::string layer = stringField(request, "layer");
52 if (!layer.empty()) {
53 avatar_editing::AvatarLayerValue value;
54 value.id = editor::ObjectId(layer);
55 value.name = layer;
56 const std::string texture = stringField(request, "texture");
57 value.textureAsset = texture.empty() ? layer + ".png" : texture;
58 auto applied = apply(*document, document->makeCreateLayer(value));
59 if (!applied.ok())
60 return editor::EditorResult<editor::AutomationOwnedTarget>::failure(applied.status());
61 }
62 const std::string parameter = stringField(request, "parameter");
63 if (!parameter.empty()) {
64 avatar_editing::AvatarParameterValue value;
65 value.id = editor::ObjectId(parameter);
66 value.name = parameter;
67 auto applied = apply(*document, document->makeCreateParameter(value));
68 if (!applied.ok())
69 return editor::EditorResult<editor::AutomationOwnedTarget>::failure(applied.status());
70 }
71 editor::AutomationOwnedTarget owned;
72 owned.target = std::move(document);
73 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
74 }
75};
