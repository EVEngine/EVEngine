42 editor::EditorResult<editor::AutomationOwnedTarget> create(
43 const editor::TargetId& target, std::string_view type, const editor::EditorValue::Object& request) override {
44 (void)type;
45 auto document = std::make_unique<avatar_editing::AvatarDocumentTarget>(target.value());
46 const std::string kind = stringField(request, "kind");
47 const std::string source = stringField(request, "source");
48 if (!kind.empty() || !source.empty()) {
49 auto applied = apply(*document, document->makeSetSource(kind.empty() ? document->kind() : kind, source));
50 if (!applied.ok())
51 return editor::EditorResult<editor::AutomationOwnedTarget>::failure(applied.status());
52 }
53 const std::string layer = stringField(request, "layer");
54 if (!layer.empty()) {
55 avatar_editing::AvatarLayerValue value;
56 value.id = editor::ObjectId(layer);
57 value.name = layer;
58 const std::string texture = stringField(request, "texture");
59 value.textureAsset = texture.empty() ? layer + ".png" : texture;
60 auto applied = apply(*document, document->makeCreateLayer(value));
61 if (!applied.ok())
62 return editor::EditorResult<editor::AutomationOwnedTarget>::failure(applied.status());
63 }
64 const std::string parameter = stringField(request, "parameter");
65 if (!parameter.empty()) {
66 avatar_editing::AvatarParameterValue value;
67 value.id = editor::ObjectId(parameter);
68 value.name = parameter;
69 auto applied = apply(*document, document->makeCreateParameter(value));
70 if (!applied.ok())
71 return editor::EditorResult<editor::AutomationOwnedTarget>::failure(applied.status());
72 }
73 editor::AutomationOwnedTarget owned;
74 owned.target = std::move(document);
75 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
76 }
77};
