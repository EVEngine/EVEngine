32void PlacementSystem::clearOccupancy(PlacementWorld &world, int instanceId) {
33 // 默认通道（""）占用存于 occupancy_，附加通道存于 allChannels_，两者都要清。
34 for (int &v : world.occupancy()) {
35 if (v == instanceId) v = 0;
36 }
37 for (auto &kv : world.allChannels()) {
38 for (int &v : kv.second) {
39 if (v == instanceId) v = 0;
40 }
41 }
42 for (auto &kv : world.horizontalEdgeChannels_) {
43 for (int &v : kv.second)
44 if (v == instanceId) v = 0;
45 }
46 for (auto &kv : world.verticalEdgeChannels_) {
47 for (int &v : kv.second)
48 if (v == instanceId) v = 0;
49 }
50 for (auto &kv : world.cornerChannels_) {
51 for (int &v : kv.second)
52 if (v == instanceId) v = 0;
53 }
54 for (auto &level : world.cellChannelsByLevel_)
55 for (auto &channel : level.second)
56 for (int &value : channel.second)
57 if (value == instanceId) value = 0;
58 for (auto &levels : {&world.horizontalEdgesByLevel_, &world.verticalEdgesByLevel_})
59 for (auto &level : *levels)
60 for (auto &channel : level.second)
61 for (int &value : channel.second)
62 if (value == instanceId) value = 0;
63 for (auto &level : world.cornersByLevel_)
64 for (auto &channel : level.second)
65 for (int &value : channel.second)
66 if (value == instanceId) value = 0;
67}

69int PlacementSystem::placeAt(PlacementWorld *world, const std::string &buildingId, int cellX, int cellY,
70 float rotationDeg) {
71 ensureBuiltins();
72 if (!world) return 0;
73 const BuildingDefinition *def = BuildingRegistry::find(buildingId);
74 if (!def) return 0;
75 if (def->placementKind != "cell") return 0;
76 const float rot = normalizeRotation(buildingId, rotationDeg);
77 if (!canPlace(world, buildingId, cellX, cellY, rot, 0, nullptr)) return 0;
78
79 PlacedBuilding pb;
80 pb.instanceId = nextInstanceId(*world);
81 pb.buildingId = buildingId;
82 pb.originCellX = cellX;
83 pb.originCellY = cellY;
84 pb.level = world->getActiveLevel();
85 world->cellToWorldPlane(cellX, cellY, pb.worldX, pb.worldY);
86 pb.elevation = 0.f;
87 pb.rotationDeg = rot;
88 pb.channel = def->channel;
89 pb.tags = def->tags;
90 PlacementQuery supportQuery;
91 supportQuery.cellX = pb.originCellX;
92 supportQuery.cellY = pb.originCellY;
93 supportQuery.level = pb.level;
94 supportQuery.rotationDeg = pb.rotationDeg;
95 pb.supportInstanceIds = collectStructuralSupports(*world, *def, supportQuery);
96
97 writeOccupancy(*world, *def, pb, pb.instanceId);
98 world->buildings()[pb.instanceId] = pb;
99 world->instanceOrder_.push_back(pb.instanceId);
100
101 BuildingChangeEvent ev;
102 ev.action = "place";
103 ev.worldId = world->getId();
104 ev.buildingId = buildingId;
105 ev.instanceId = pb.instanceId;
106 ev.cellX = cellX;
107 ev.cellY = cellY;
108 ev.rotationDeg = rot;
109 ev.worldX = pb.worldX;
110 ev.worldY = pb.worldY;
111 ev.elevation = pb.elevation;
112 ev.channel = pb.channel;
113 if (world->publishEvents_) emit(std::move(ev));
114 return pb.instanceId;
115}

117int PlacementSystem::placeAtWorld(PlacementWorld *world, const std::string &buildingId, float worldX, float worldY,
118 float rotationDeg) {
119 ensureBuiltins();
120 if (!world) return 0;
121 const SnapResult s = snap(*world, buildingId, worldX, worldY);
122 const int id = placeAt(world, buildingId, s.cellX, s.cellY, rotationDeg);
123 if (id <= 0) return 0;
124 // Preserve free / custom snap world pose (cell placement alone snaps to cell origin).
125 auto it = world->buildings().find(id);
126 if (it != world->buildings().end()) {
127 it->second.worldX = s.worldX;
128 it->second.worldY = s.worldY;
129 }
130 return id;
131}
