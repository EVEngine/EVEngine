38Result<void> NavigationTaskService::start(const TaskContext& context, const TaskSpec& spec,
39 std::string& inOutMemoryJson) {
40 const auto taskKey = key(context, spec);
41 if (active_.contains(taskKey))
42 return Result<void>::failure(Diagnostic::error(DiagnosticCode::AlreadyExists,
43 "navigation task already owns an active ticket", {}, {},
44 "npc_ai.navigation"));
45 auto request = requestFactory_->create(context, spec);
46 if (!request.ok()) return Result<void>::failure(request.status());
47 if (request.value().agent != context.agent || !finitePosition(request.value().start) ||
48 !finitePosition(request.value().destination) || !std::isfinite(request.value().agentRadius) ||
49 !std::isfinite(request.value().acceptanceRadius) || request.value().agentRadius <= 0.0 ||
50 request.value().acceptanceRadius < 0.0 || request.value().requestedTick != context.simulationTick)
51 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
52 "navigation request factory returned an invalid request", {}, {},
53 "npc_ai.navigation"));
54 auto ticket = provider_->begin(request.value());
55 if (!ticket.ok()) return Result<void>::failure(ticket.status());
56 if (!ticket.value().isValid()) {
57 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
58 "navigation provider returned an invalid ticket", {}, {},
59 "npc_ai.navigation"));
60 }
61 active_.emplace(taskKey, ticket.value());
62 inOutMemoryJson = "{\"phase\":\"pending\"}";
63 return Result<void>::success(Status::success(StatusCode::Applied));
64}

66Result<TaskStatus> NavigationTaskService::tick(const TaskContext& context, const TaskSpec& spec,
67 std::string& inOutMemoryJson) {
68 const auto found = active_.find(key(context, spec));
69 if (found == active_.end())
70 return Result<TaskStatus>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
71 "navigation task has no active ticket", {}, {},
72 "npc_ai.navigation"));
73 auto progress = provider_->poll(found->second, context.simulationTick);
74 if (!progress.ok()) return Result<TaskStatus>::failure(progress.status());
75 if (!finitePosition(progress.value().desiredVelocity) || !std::isfinite(progress.value().remainingDistance) ||
76 progress.value().remainingDistance < 0.0)
77 return Result<TaskStatus>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
78 "navigation provider returned invalid progress", {}, {},
79 "npc_ai.navigation"));
80 switch (progress.value().phase) {
81 case NavigationPhase::Pending:
82 inOutMemoryJson = "{\"phase\":\"pending\"}";
83 return Result<TaskStatus>::success(TaskStatus::Running);
84 case NavigationPhase::Moving:
85 inOutMemoryJson = "{\"phase\":\"moving\"}";
86 return Result<TaskStatus>::success(TaskStatus::Running);
87 case NavigationPhase::Arrived:
88 active_.erase(found);
89 inOutMemoryJson = "{\"phase\":\"arrived\"}";
90 return Result<TaskStatus>::success(TaskStatus::Succeeded);
91 case NavigationPhase::Unreachable:
92 active_.erase(found);
93 inOutMemoryJson = "{\"phase\":\"unreachable\"}";
94 return Result<TaskStatus>::success(TaskStatus::Failed);
95 case NavigationPhase::Cancelled:
96 active_.erase(found);
97 inOutMemoryJson = "{\"phase\":\"cancelled\"}";
98 return Result<TaskStatus>::success(TaskStatus::Failed);
99 }
100 return Result<TaskStatus>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
101 "navigation provider returned an unknown phase", {}, {},
102 "npc_ai.navigation"));
103}
