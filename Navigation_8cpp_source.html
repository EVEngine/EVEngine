42Result<void> NavigationTaskService::start(const TaskContext& context, const TaskSpec& spec,
43 std::string& inOutMemoryJson) {
44 const auto taskKey = key(context, spec);
45 if (active_.contains(taskKey))
46 return navigationFailure<void>(DiagnosticCode::AlreadyExists, "navigation task already owns an active ticket");
47 auto request = requestFactory_->create(context, spec);
48 if (!request.ok()) return Result<void>::failure(request.status());
49 if (request.value().agent != context.agent || !finitePosition(request.value().start) ||
50 !finitePosition(request.value().destination) || !std::isfinite(request.value().agentRadius) ||
51 !std::isfinite(request.value().acceptanceRadius) || request.value().agentRadius <= 0.0 ||
52 request.value().acceptanceRadius < 0.0 || request.value().requestedTick != context.simulationTick)
53 return navigationFailure<void>(DiagnosticCode::InvalidArgument,
54 "navigation request factory returned an invalid request");
55 auto ticket = provider_->begin(request.value());
56 if (!ticket.ok()) return Result<void>::failure(ticket.status());
57 if (!ticket.value().isValid()) {
58 return navigationFailure<void>(DiagnosticCode::InvariantViolation,
59 "navigation provider returned an invalid ticket");
60 }
61 active_.emplace(taskKey, ticket.value());
62 inOutMemoryJson = "{\"phase\":\"pending\"}";
63 return Result<void>::success(Status::success(StatusCode::Applied));
64}

66Result<TaskStatus> NavigationTaskService::tick(const TaskContext& context, const TaskSpec& spec,
67 std::string& inOutMemoryJson) {
68 const auto found = active_.find(key(context, spec));
69 if (found == active_.end())
70 return navigationFailure<TaskStatus>(DiagnosticCode::PreconditionViolation,
71 "navigation task has no active ticket");
72 auto progress = provider_->poll(found->second, context.simulationTick);
73 if (!progress.ok()) return Result<TaskStatus>::failure(progress.status());
74 if (!finitePosition(progress.value().desiredVelocity) || !std::isfinite(progress.value().remainingDistance) ||
75 progress.value().remainingDistance < 0.0)
76 return navigationFailure<TaskStatus>(DiagnosticCode::InvariantViolation,
77 "navigation provider returned invalid progress");
78 switch (progress.value().phase) {
79 case NavigationPhase::Pending:
80 inOutMemoryJson = "{\"phase\":\"pending\"}";
81 return Result<TaskStatus>::success(TaskStatus::Running);
82 case NavigationPhase::Moving:
83 inOutMemoryJson = "{\"phase\":\"moving\"}";
84 return Result<TaskStatus>::success(TaskStatus::Running);
85 case NavigationPhase::Arrived:
86 active_.erase(found);
87 inOutMemoryJson = "{\"phase\":\"arrived\"}";
88 return Result<TaskStatus>::success(TaskStatus::Succeeded);
89 case NavigationPhase::Unreachable:
90 active_.erase(found);
91 inOutMemoryJson = "{\"phase\":\"unreachable\"}";
92 return Result<TaskStatus>::success(TaskStatus::Failed);
93 case NavigationPhase::Cancelled:
94 active_.erase(found);
95 inOutMemoryJson = "{\"phase\":\"cancelled\"}";
96 return Result<TaskStatus>::success(TaskStatus::Failed);
97 }
98 return navigationFailure<TaskStatus>(DiagnosticCode::InvariantViolation,
99 "navigation provider returned an unknown phase");
100}
