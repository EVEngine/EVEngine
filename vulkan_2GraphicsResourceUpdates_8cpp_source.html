49bool Graphics::reloadTextureFromFile(const std::string &filename) {
50 if (filename.empty()) return false;
51 const std::string key = normalizeTexPath(filename);
52 auto it = texturesByPath.find(key);
53 if (it == texturesByPath.end() || !it->second) return false;
54
55 ensureFileTexturesReady();
56 applyPendingResourceChanges();
57 // The provider hands back a cache-owned ImageData; the pin keeps it alive until
58 // the pixels have been copied out of it.
59 image::ImageData *data = nullptr;
60 eve::ResourcePin keepAlive;
61 try {
62 auto reloaded = eve::ResourceManager::getInstance().reload(filename);
63 if (!reloaded.ok() || !reloaded.value()) return false;
64 auto *imgMod = image::Image::create();
65 data = imgMod->newImageDataFromFile(filename);
66 if (data != nullptr) {
67 auto pinned = eve::ResourceManager::getInstance().pin(data);
68 if (!pinned.ok()) return false;
69 keepAlive = std::move(pinned).takeValue();
70 // The pin is the authority from here on; the borrowed pointer may have gone
71 // stale before the pin was taken.
72 data = static_cast<image::ImageData *>(keepAlive.get());
73 }
74 } catch (...) {
75 return false;
76 }
77 if (!data) return false;
78 return replaceTexturePixels(it->second, data);
79}
