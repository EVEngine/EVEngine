51Result<void> PcgLightingAmbientPhotoMode::applyPhotoModeField(
52 const PhotoModeAssignment &assignment) {
53 const auto fail = [&](DiagnosticCode code, const char *message) {
54 return Result<void>::failure(Diagnostic::error(
55 code, message, assignment.field, {}, "daynight.pcgLightingAmbient"));
56 };
57 if (!target_)
58 return fail(DiagnosticCode::Unsupported, "day-night target is unavailable");
59 if (!accepts(assignment.field))
60 return fail(DiagnosticCode::InvalidArgument, "unsupported Pcg ambient field");
61
62 PcgAmbientLightState candidate = state_;
63 if (assignment.field == "m_ambientSkyColor" ||
64 assignment.field == "m_ambientEquatorColor" ||
65 assignment.field == "m_ambientGroundColor") {
66 const auto *value = std::get_if<PhotoModeColor>(&assignment.value);
67 if (!value || !std::isfinite(value->r) || !std::isfinite(value->g) ||
68 !std::isfinite(value->b) || value->r < 0.f || value->g < 0.f ||
69 value->b < 0.f)
70 return fail(DiagnosticCode::InvalidArgument,
71 "ambient color requires finite non-negative RGB channels");
72 float *red = &candidate.skyRed;
73 float *green = &candidate.skyGreen;
74 float *blue = &candidate.skyBlue;
75 if (assignment.field == "m_ambientEquatorColor") {
76 red = &candidate.equatorRed;
77 green = &candidate.equatorGreen;
78 blue = &candidate.equatorBlue;
79 } else if (assignment.field == "m_ambientGroundColor") {
80 red = &candidate.groundRed;
81 green = &candidate.groundGreen;
82 blue = &candidate.groundBlue;
83 }
84 *red = value->r;
85 *green = value->g;
86 *blue = value->b;
87 } else {
88 const auto *value = std::get_if<float>(&assignment.value);
89 if (!value || !std::isfinite(*value))
90 return fail(DiagnosticCode::InvalidArgument,
91 "ambient setting requires a finite float");
92 if (assignment.field == "m_ambientIntensity")
93 candidate.intensity = *value;
94 else
95 candidate.globalLightMultiplier = *value;
96 }
97
98 auto result = target_->setPcgAmbientLight(candidate);
99 if (!result.ok()) return result;
100 state_ = candidate;
101 return Result<void>::success();
102}
