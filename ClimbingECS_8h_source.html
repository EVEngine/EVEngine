257class ClimbingMotionSystem {
258public:
263 template <class EntityRoot>
264 [[nodiscard]] static eve::Result<std::size_t> step(physics::World3D& world,
265 const eve::SimulationStep& step,
266 const ClimbingMotionInput& motion = {}) {
267 if (step.delta.nanoseconds() <= 0)
268 return detail::climbingEcsFailure<std::size_t>(eve::DiagnosticCode::InvalidArgument,
269 "climbing motion delta must be positive",
270 "step.delta");
271 std::size_t processed = 0;
272 auto view = ecs::View<EntityRoot, ClimbingBody, ClimbingIntent, ClimbingState, ClimbingLinks>();
273 for (auto it = view.begin(); it != view.end(); ++it) {
274 auto [body, intent, state, links] = *it;
275 auto runtime = Climbing::resolve(state->runtime);
276 if (!runtime.isBound())
277 return detail::climbingEcsFailure<std::size_t>(eve::DiagnosticCode::StaleHandle,
278 "climbing runtime link is stale",
279 "state.runtime");
280 auto linkedBody = links->physicsBody.resolve(world);
281 if (!linkedBody) return eve::Result<std::size_t>::failure(linkedBody.status());
282 if (!detail::activeClimbingPhase(runtime->phase())) {
283 const auto ordinaryStart = std::chrono::steady_clock::now();
284 if (!(body->capsuleRadius > 0.f) ||
285 !(body->capsuleHeight > body->capsuleRadius * 2.f) || body->skin < 0.f ||
286 body->groundSnap < 0.f || body->stepHeight < 0.f ||
287 !std::isfinite(body->maxSlopeRadians) || body->maxSlopeRadians < 0.f ||
288 body->maxSlopeRadians >= 1.57079633f)
289 return detail::climbingEcsFailure<std::size_t>(eve::DiagnosticCode::InvalidArgument,
290 "ordinary locomotion capsule is invalid",
291 "body.capsule");
292 const ClimbingLocomotionPolicy policy = runtime->locomotionPolicy();
293 const float deltaSeconds = static_cast<float>(step.delta.seconds());
294 const float moveLength = std::sqrt(intent->move.x * intent->move.x +
295 intent->move.z * intent->move.z);
296 const Vec3 desiredVelocity{intent->move.x, 0.f, intent->move.z};
297 const float acceleration = body->grounded
298 ? (moveLength > 1e-6f ? policy.groundAcceleration
299 : policy.groundBraking)
300 : policy.groundAcceleration * policy.airControl;
301 const float maxVelocityChange = acceleration * deltaSeconds;
302 const Vec3 horizontalError{desiredVelocity.x - body->velocity.x, 0.f,
303 desiredVelocity.z - body->velocity.z};
304 const float errorLength = std::sqrt(horizontalError.x * horizontalError.x +
305 horizontalError.z * horizontalError.z);
306 if (errorLength > maxVelocityChange && errorLength > 1e-6f) {
307 const float scale = maxVelocityChange / errorLength;
308 body->velocity.x += horizontalError.x * scale;
309 body->velocity.z += horizontalError.z * scale;
310 } else {
311 body->velocity.x = desiredVelocity.x;
312 body->velocity.z = desiredVelocity.z;
313 }
314
315 const auto jump = ClimbingInputSystem::peek(*intent, ClimbingCommand::Jump, step.tick);
316 const bool freshJump = jump && (!body->hasOrdinaryJump ||
317 jump->pressedTick != body->lastOrdinaryJumpPressedTick);
318 if (freshJump &&
319 (body->grounded ||
320 ClimbingInputSystem::coyoteWindowState(step.tick, body->lastGroundedTick,
321 policy.coyoteTicks) == ClimbingCoyoteState::Eligible)) {
322 body->velocity.y = policy.jumpSpeed;
323 body->grounded = false;
324 body->lastOrdinaryJumpPressedTick = jump->pressedTick;
325 body->hasOrdinaryJump = true;
326 } else if (!body->grounded) {
327 body->velocity.y -= policy.gravity * deltaSeconds;
328 } else {
329 body->velocity.y = 0.f;
330 }
331
332 physics::QueryFilter3D filter = policy.queryFilter;
333 filter.ignoredBodyId = linkedBody.value()->getId();
334 const physics::CapsuleMovePolicy3D moverPolicy{
335 body->up.x, body->up.y, body->up.z, body->maxSlopeRadians};
336 const float lowerY = body->feet.y + body->capsuleRadius + body->skin;
337 const float upperY = body->feet.y + body->capsuleHeight - body->capsuleRadius + body->skin;
338 const float snap = body->grounded && !freshJump ? body->groundSnap : 0.f;
339 auto moved = world.moveCapsuleOwned(
340 body->feet.x, lowerY, body->feet.z, body->feet.x, upperY, body->feet.z,
341 body->capsuleRadius, body->velocity.x * deltaSeconds,
342 body->velocity.y * deltaSeconds - snap, body->velocity.z * deltaSeconds,
343 filter, moverPolicy);
344 if (!moved) return eve::Result<std::size_t>::failure(moved.status());
345 physics::CapsuleMove3D movement = std::move(moved).takeValue();
346 std::uint32_t queryCount = 1;
347 std::uint32_t moverIterations = static_cast<std::uint32_t>(movement.iterations);
348 const float desiredX = body->velocity.x * deltaSeconds;
349 const float desiredZ = body->velocity.z * deltaSeconds;
350 const float desiredHorizontal = std::sqrt(desiredX * desiredX + desiredZ * desiredZ);
351 const float directHorizontal = std::sqrt(movement.deltaX * movement.deltaX +
352 movement.deltaZ * movement.deltaZ);
353 if (body->grounded && !freshJump && body->stepHeight > 0.f &&
354 desiredHorizontal > 1e-6f && directHorizontal + body->skin < desiredHorizontal) {
355 auto raised = world.moveCapsuleOwned(
356 body->feet.x, lowerY, body->feet.z, body->feet.x, upperY, body->feet.z,
357 body->capsuleRadius, 0.f, body->stepHeight, 0.f, filter, moverPolicy);
358 if (!raised) return eve::Result<std::size_t>::failure(raised.status());
359 ++queryCount;
360 const physics::CapsuleMove3D rise = std::move(raised).takeValue();
361 moverIterations += static_cast<std::uint32_t>(rise.iterations);
362 if (rise.deltaY + body->skin >= body->stepHeight) {
363 auto crossed = world.moveCapsuleOwned(
364 body->feet.x, lowerY + rise.deltaY, body->feet.z,
365 body->feet.x, upperY + rise.deltaY, body->feet.z,
366 body->capsuleRadius, desiredX, 0.f, desiredZ, filter, moverPolicy);
367 if (!crossed) return eve::Result<std::size_t>::failure(crossed.status());
368 ++queryCount;
369 const physics::CapsuleMove3D across = std::move(crossed).takeValue();
370 moverIterations += static_cast<std::uint32_t>(across.iterations);
371 const float steppedHorizontal = std::sqrt(across.deltaX * across.deltaX +
372 across.deltaZ * across.deltaZ);
373 auto lowered = world.moveCapsuleOwned(
374 body->feet.x + across.deltaX, lowerY + rise.deltaY,
375 body->feet.z + across.deltaZ, body->feet.x + across.deltaX,
376 upperY + rise.deltaY, body->feet.z + across.deltaZ,
377 body->capsuleRadius, 0.f,
378 -(rise.deltaY + body->groundSnap + body->skin), 0.f, filter, moverPolicy);
379 if (!lowered) return eve::Result<std::size_t>::failure(lowered.status());
380 ++queryCount;
381 const physics::CapsuleMove3D down = std::move(lowered).takeValue();
382 moverIterations += static_cast<std::uint32_t>(down.iterations);
383 if (down.grounded && steppedHorizontal > directHorizontal + body->skin) {
384 movement.constrained = rise.constrained || across.constrained || down.constrained;
385 movement.grounded = true;
386 movement.deltaX = across.deltaX;
387 movement.deltaY = rise.deltaY + down.deltaY;
388 movement.deltaZ = across.deltaZ;
389 movement.normalX = down.normalX;
390 movement.normalY = down.normalY;
391 movement.normalZ = down.normalZ;
392 movement.planeCount = rise.planeCount + across.planeCount + down.planeCount;
393 movement.iterations = rise.iterations + across.iterations + down.iterations;
394 }
395 }
396 }
397 body->feet = {body->feet.x + movement.deltaX, body->feet.y + movement.deltaY,
398 body->feet.z + movement.deltaZ};
399 body->grounded = movement.grounded && body->velocity.y <= 0.f;
400 body->mode = body->grounded ? ClimbingMovementMode::Grounded
401 : ClimbingMovementMode::Airborne;
402 body->velocity.x = movement.deltaX / deltaSeconds;
403 body->velocity.z = movement.deltaZ / deltaSeconds;
404 body->velocity.y = body->grounded ? 0.f : movement.deltaY / deltaSeconds;
405 if (body->grounded) {
406 body->groundNormal = {movement.normalX, movement.normalY, movement.normalZ};
407 body->lastStableFeet = body->feet;
408 body->lastGroundedTick = step.tick;
409 }
410 if (moveLength > 1e-6f)
411 body->forward = {intent->move.x / moveLength, 0.f, intent->move.z / moveLength};
412 linkedBody.value()->setPosition(body->feet.x,
413 body->feet.y + body->capsuleHeight * 0.5f,
414 body->feet.z);
415 const auto ordinaryElapsed = std::chrono::steady_clock::now() - ordinaryStart;
416 runtime->recordOrdinaryTick(
417 step.tick, queryCount, moverIterations,
418 static_cast<std::uint64_t>(std::chrono::duration_cast<std::chrono::nanoseconds>(
419 ordinaryElapsed)
420 .count()));
421 ++processed;
422 continue;
423 }
424 const Vec3 previous = body->feet;
425 auto advanced = runtime->advance(world, step, motion);
426 if (!advanced) return eve::Result<std::size_t>::failure(advanced.status());
427 state->lastAdvance = std::move(advanced).takeValue();
428 state->lastAdvanceTick = step.tick;
429 body->feet = state->lastAdvance.feet;
430 const float inverseDelta = static_cast<float>(1.0 / step.delta.seconds());
431 body->velocity = {(body->feet.x - previous.x) * inverseDelta,
432 (body->feet.y - previous.y) * inverseDelta,
433 (body->feet.z - previous.z) * inverseDelta};
434 if (state->lastAdvance.hasTerminalVelocity)
435 body->velocity = state->lastAdvance.terminalVelocity;
436 body->grounded = state->lastAdvance.grounded;
437 body->mode = detail::activeClimbingPhase(state->lastAdvance.phase)
438 ? ClimbingMovementMode::Climbing
439 : (body->grounded ? ClimbingMovementMode::Grounded
440 : ClimbingMovementMode::Airborne);
441 if (body->grounded) {
442 body->lastStableFeet = body->feet;
443 body->lastGroundedTick = step.tick;
444 }
445 linkedBody.value()->setPosition(body->feet.x, body->feet.y + body->capsuleHeight * 0.5f,
446 body->feet.z);
447 ++processed;
448 }
449 return eve::Result<std::size_t>::success(processed);
450 }
