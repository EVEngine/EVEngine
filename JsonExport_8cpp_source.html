11std::string escapeJson(const std::string &s) {
12 std::string out;
13 out.reserve(s.size() + 8);
14 for (char c : s) {
15 switch (c) {
16 case '\\': out += "\\\\"; break;
17 case '"': out += "\\\""; break;
18 case '\n': out += "\\n"; break;
19 case '\r': out += "\\r"; break;
20 case '\t': out += "\\t"; break;
21 default: out += c; break;
22 }
23 }
24 return out;
25}
