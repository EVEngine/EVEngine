57Result<void> MaterialEditor::setProperties(const Value::Object& properties) {
58 auto candidate = target_;
59 std::vector<DomainOperation> operations;
60 const auto metadata = target_.schema(selection());
61 for (const auto& [path, input] : properties) {
62 auto value = input;
63 const auto descriptor = metadata.find(PropertyPath(path));
64 if (descriptor && descriptor->type == PropertyType::Float) {
65 if (const auto* integer = value.getIf<std::int64_t>()) value = static_cast<double>(*integer);
66 }
67 if (descriptor && (descriptor->type == PropertyType::Color || descriptor->type == PropertyType::Vec2 ||
68 descriptor->type == PropertyType::Vec3 || descriptor->type == PropertyType::Vec4)) {
69 if (auto* components = value.getIf<Value::Array>()) {
70 for (auto& component : *components)
71 if (const auto* integer = component.getIf<std::int64_t>())
72 component = static_cast<double>(*integer);
73 }
74 }
75 auto current = target_.read(selection(), PropertyPath(path));
76 if (current.state == PropertyReadState::Value && current.value == value) continue;
77 auto operation = candidate.makeSet(selection(), PropertyPath(path), value, PropertySetMode::Absolute);
78 if (!operation.ok()) return Result<void>::failure(operation.status());
79 auto staged = candidate.applyDomainOperation(operation.value());
80 if (!staged.ok()) return staged;
81 operations.push_back(std::move(operation).takeValue());
82 }
83 const auto diagnostics = candidate.validate();
84 for (const auto& diagnostic : diagnostics) {
85 if (diagnostic.severity() == DiagnosticSeverity::Error)
86 return Result<void>::failure(eve::Status(Status::Rejected, diagnostics));
87 }
88 if (operations.empty()) return noOp(diagnostics);
89 editor::TransactionSpec spec;
90 spec.id = TransactionId("material.tx." + std::to_string(++sequence_));
91 spec.label = "Edit material";
92 spec.target = target_.targetId();
93 spec.baseRevision = target_.revision();
94 auto begun = transactions_.begin(std::move(spec));
95 if (!begun.ok()) return Result<void>::failure(begun.status());
96 for (auto& operation : operations) {
97 auto appended = transactions_.append(std::move(operation));
98 if (!appended.ok()) {
99 auto discarded = transactions_.discard();
100 if (!discarded.ok()) return discarded;
101 return appended;
102 }
103 }
104 auto committed = transactions_.commit();
105 if (!committed.ok()) {
106 auto discarded = transactions_.discard();
107 if (!discarded.ok()) return discarded;
108 return Result<void>::failure(committed.status());
109 }
110 return applied<void>(diagnostics);
111}

121Value MaterialEditor::schema() const {
122 const auto metadata = target_.schema(selection());
123 Value::Array properties;
124 for (const auto& field : metadata.properties) {
125 Value::Object item{{"path", field.path.value()},
126 {"label", field.displayNameKey},
127 {"category", field.category},
128 {"type", static_cast<std::int64_t>(field.type)},
129 {"defaultValue", field.defaultValue}};
130 Value::Array choices;
131 for (const auto& choice : field.enumItems) choices.emplace_back(choice);
132 item["choices"] = std::move(choices);
133 if (field.numeric.minimum) item["minimum"] = *field.numeric.minimum;
134 if (field.numeric.maximum) item["maximum"] = *field.numeric.maximum;
135 properties.emplace_back(std::move(item));
136 }
137 return Value::Object{{"typeId", metadata.typeId},
138 {"version", static_cast<std::int64_t>(metadata.version)},
139 {"properties", std::move(properties)}};
140}
