19float smoothstep(float edge0, float edge1, float x) {
20 if (edge0 == edge1) return x < edge0 ? 0.f : 1.f;
21 const float t = std::clamp((x - edge0) / (edge1 - edge0), 0.f, 1.f);
22 return t * t * (3.f - 2.f * t);
23}

26float voronoiDist(const NoiseField &n, float x, float y) {
27 float best = 1e18f;
28 const int cx0 = int(std::floor(x));
29 const int cy0 = int(std::floor(y));
30 for (int oy = -1; oy <= 1; ++oy) {
31 for (int ox = -1; ox <= 1; ++ox) {
32 const int cx = cx0 + ox;
33 const int cy = cy0 + oy;
34 const float px = float(cx) + n.hash01(cx, cy);
35 const float py = float(cy) + n.hash01(cx * 7 + 3, cy * 13 + 5);
36 const float dx = x - px;
37 const float dy = y - py;
38 const float d = dx * dx + dy * dy;
39 if (d < best) best = d;
40 }
41 }
42 return std::sqrt(best);
43}
