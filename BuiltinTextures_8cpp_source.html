23float smoothstep(float edge0, float edge1, float x) {
24 if (edge0 == edge1) return x < edge0 ? 0.f : 1.f;
25 const float t = std::clamp((x - edge0) / (edge1 - edge0), 0.f, 1.f);
26 return t * t * (3.f - 2.f * t);
27}

30float voronoiDist(const NoiseField &n, float x, float y) {
31 float best = 1e18f;
32 const int cx0 = int(std::floor(x));
33 const int cy0 = int(std::floor(y));
34 for (int oy = -1; oy <= 1; ++oy) {
35 for (int ox = -1; ox <= 1; ++ox) {
36 const int cx = cx0 + ox;
37 const int cy = cy0 + oy;
38 const float px = float(cx) + n.hash01(cx, cy);
39 const float py = float(cy) + n.hash01(cx * 7 + 3, cy * 13 + 5);
40 const float dx = x - px;
41 const float dy = y - py;
42 const float d = dx * dx + dy * dy;
43 if (d < best) best = d;
44 }
45 }
46 return std::sqrt(best);
47}

559void buildBlueNoiseCardStamps(std::uint32_t seed, int count, std::vector<LeafStamp> &stamps) {
560 stamps.clear();
561 count = std::clamp(count, 8, 16);
562 // Bridson on a fine grid, then normalize to the unit square with a margin.
563 const float minDist01 = std::sqrt(0.42f / float(count));
564 constexpr int kSide = 64;
565 const float radius = std::max(1.5f, minDist01 * float(kSide));
566 const PointSet pts = poissonDiskPoints(kSide, kSide, radius, seed, count + 12);
567 std::mt19937 rng(seed ^ 0x9e3779b9u);
568 std::uniform_real_distribution<float> unit(0.f, 1.f);
569 constexpr float kMargin = 0.07f;
570 int placed = 0;
571 for (const ProcgenPoint &p : pts.points()) {
572 if (placed >= count) break;
573 LeafStamp s;
574 s.cx = kMargin + (1.f - 2.f * kMargin) * (p.x / float(kSide));
575 s.cy = kMargin + (1.f - 2.f * kMargin) * (p.z / float(kSide));
576 s.ang = unit(rng) * 6.28318530718f;
577 s.halfW = 0.050f + 0.028f * unit(rng);
578 s.halfH = 0.070f + 0.036f * unit(rng);
579 stamps.push_back(s);
580 ++placed;
581 }
582 // Top up with dart throws if Bridson under-filled (small textures / tight radius).
583 int guard = 0;
584 while (placed < count && guard++ < count * 100) {
585 LeafStamp s;
586 s.cx = kMargin + (1.f - 2.f * kMargin) * unit(rng);
587 s.cy = kMargin + (1.f - 2.f * kMargin) * unit(rng);
588 bool ok = true;
589 for (const LeafStamp &o : stamps) {
590 const float dx = s.cx - o.cx, dy = s.cy - o.cy;
591 if (dx * dx + dy * dy < minDist01 * minDist01) {
592 ok = false;
593 break;
594 }
595 }
596 if (!ok) continue;
597 s.ang = unit(rng) * 6.28318530718f;
598 s.halfW = 0.050f + 0.028f * unit(rng);
599 s.halfH = 0.070f + 0.036f * unit(rng);
600 stamps.push_back(s);
601 ++placed;
602 }
603}

605void sampleOvateLeafCard(const NoiseField &noise, float u, float v, const std::vector<LeafStamp> &stamps,
606 float &cover, float &shade) {
607 cover = 0.f;
608 shade = 0.f;
609 for (const LeafStamp &s : stamps) {
610 float dx = u - s.cx;
611 float dy = v - s.cy;
612 const float ca = std::cos(s.ang), sa = std::sin(s.ang);
613 const float rx = dx * ca - dy * sa;
614 const float ry = dx * sa + dy * ca;
615 // Ovate / lanceolate: pointed tip along +Y, wider shoulders below centre.
616 const float halfW = s.halfW * (1.f - 0.42f * std::max(0.f, ry / std::max(1e-4f, s.halfH)));
617 const float d = (rx * rx) / std::max(1e-5f, halfW * halfW) +
618 (ry * ry) / std::max(1e-5f, s.halfH * s.halfH);
619 const float mask = 1.f - smoothstep(0.78f, 1.02f, d);
620 if (mask <= cover) continue;
621 cover = mask;
622 // Flat fill — no veins/mottle; slight per-stamp tone so leaves are not identical.
623 shade = 0.48f + 0.12f * noise.hash01(int(s.cx * 97.f), int(s.cy * 53.f));
624 }
625}

631 std::vector<std::vector<LeafStamp>> &panels) {

644void sampleLeafCardAtlas(const NoiseField &noise, float fu, float v,
645 const std::vector<std::vector<LeafStamp>> &panels, float &cover, float &shade) {
646 const int col = std::clamp(int(fu * float(kLeafCardCols)), 0, kLeafCardCols - 1);
647 const int row = std::clamp(int(v * float(kLeafCardRows)), 0, kLeafCardRows - 1);
648 const float localU = fu * float(kLeafCardCols) - float(col);
649 const float localV = v * float(kLeafCardRows) - float(row);

657void sampleFoliageFill(const NoiseField &noise, float u, float v, float scale, float &cover, float &shade) {
658 cover = 0.f;
659 shade = 0.f;
660 const float su = u * scale;
661 const float sv = v * scale;
662 for (int oy = -1; oy <= 1; ++oy) {
663 for (int ox = -1; ox <= 1; ++ox) {
664 const int cx = int(std::floor(su * 2.2f)) + ox;
665 const int cy = int(std::floor(sv * 2.2f)) + oy;
666 const float px = float(cx) + noise.hash01(cx, cy);
667 const float py = float(cy) + noise.hash01(cx * 7 + 3, cy * 13 + 5);
668 float dx = su * 2.2f - px;
669 float dy = sv * 2.2f - py;
670 const float ang = (noise.hash01(cx * 3 + 1, cy * 5 + 2) - 0.5f) * 1.4f;
671 const float ca = std::cos(ang), sa = std::sin(ang);
672 const float rx = dx * ca - dy * sa;
673 const float ry = dx * sa + dy * ca;
674 const float halfW = 0.48f * (1.f - 0.4f * std::max(0.f, ry));
675 const float d = (rx * rx) / std::max(1e-4f, halfW * halfW) + (ry * ry) / 0.58f;
676 const float mask = 1.f - smoothstep(0.55f, 1.05f, d);
677 if (mask <= cover) continue;
678 cover = mask;
679 const float vein = std::pow(
680 std::fabs(std::sin(ry * 8.f + noise.valueNoise(px * 0.35f, py * 0.35f) * 2.5f)), 3.f);
681 const float mott = noise.fbm(px * 0.28f + u * 1.6f, py * 0.28f + v * 1.6f, 3);
682 shade = std::clamp(0.30f + mott * 0.42f + (1.f - vein) * 0.28f, 0.f, 1.f);
683 }
684 }
685 // Keep blob fill nearly opaque so ellipsoid lobes do not go hollow.
686 cover = std::clamp(cover * 0.55f + 0.45f, 0.f, 1.f);
687}

692Rgba8 sampleBarkColor(const NoiseField &noise, float bu, float v, float scale, int colors) {
693 ColorRamp barkRamp;
694 barkRamp.add(0.00f, 28, 18, 12);
695 barkRamp.add(0.22f, 58, 38, 24);
696 barkRamp.add(0.48f, 102, 72, 44);
697 barkRamp.add(0.72f, 148, 110, 70);
698 barkRamp.add(1.00f, 68, 46, 28);
699 const float warp = noise.warp(bu * scale, v * scale, 2.1f, 3);
700 const float grain = 0.5f + 0.5f * std::sin((bu * 9.f + warp * 6.f) * 6.28318f);
701 const float ridge = noise.ridged(bu * 1.6f * scale, v * 0.28f * scale, 5);
702 const float crack =
703 std::pow(std::fabs(noise.valueNoise(bu * 1.4f * scale, v * 4.2f * scale) - 0.5f) * 2.f, 1.55f);
704 const float flake = noise.fbm(bu * 3.2f * scale + 1.7f, v * 0.9f * scale, 3);
705 const float h =
706 std::clamp(grain * 0.28f + ridge * 0.38f + (1.f - crack) * 0.22f + flake * 0.12f, 0.f, 1.f);
707 return barkRamp.sampleBanded(h, colors);
