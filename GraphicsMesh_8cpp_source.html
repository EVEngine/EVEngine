51Mesh *Graphics::newMeshFromAssimp(const ::aiMesh &mesh) {
52 ASSERT(initialized);
53 if (!initialized) throw Exception("newMeshFromAssimp: graphics not initialized");
54 if (mesh.mNumVertices == 0 || mesh.mNumFaces == 0)
55 throw Exception("newMeshFromAssimp: empty mesh");
56
57 std::vector<MeshVertex> verts;
58 verts.reserve(mesh.mNumVertices);
59 std::vector<float> basePos;
60 std::vector<float> baseNrm;
61 std::vector<float> baseUv;
62 basePos.reserve(mesh.mNumVertices * 3);
63 baseNrm.reserve(mesh.mNumVertices * 3);
64 baseUv.reserve(mesh.mNumVertices * 2);
65 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
66 MeshVertex v{};
67 v.pos = {mesh.mVertices[i].x, mesh.mVertices[i].y, mesh.mVertices[i].z};
68 if (mesh.HasNormals())
69 v.normal = {mesh.mNormals[i].x, mesh.mNormals[i].y, mesh.mNormals[i].z};
70 else
71 v.normal = {0.f, 1.f, 0.f};
72 if (mesh.HasTextureCoords(0))
73 v.uv = {mesh.mTextureCoords[0][i].x, mesh.mTextureCoords[0][i].y};
74 else
75 v.uv = {0.f, 0.f};
76 verts.push_back(v);
77 basePos.push_back(v.pos.x);
78 basePos.push_back(v.pos.y);
79 basePos.push_back(v.pos.z);
80 baseNrm.push_back(v.normal.x);
81 baseNrm.push_back(v.normal.y);
82 baseNrm.push_back(v.normal.z);
83 baseUv.push_back(v.uv.x);
84 baseUv.push_back(v.uv.y);
85 }
86
87 std::vector<uint32_t> indices;
88 indices.reserve(mesh.mNumFaces * 3);
89 for (unsigned f = 0; f < mesh.mNumFaces; ++f) {
90 const aiFace &face = mesh.mFaces[f];
91 if (face.mNumIndices != 3) continue;
92 indices.push_back(face.mIndices[0]);
93 indices.push_back(face.mIndices[1]);
94 indices.push_back(face.mIndices[2]);
95 }
96 if (indices.empty()) throw Exception("newMeshFromAssimp: no triangle faces");
97
98 std::unique_ptr<GpuMesh> gpu;
99 if (mesh.mNumVertices <= 65535u) {
100 std::vector<uint16_t> idx16;
101 idx16.reserve(indices.size());
102 for (uint32_t i : indices) idx16.push_back(uint16_t(i));
103 gpu = uploadGpuMesh16(device, frameToken(), verts, idx16);
104 } else {
105 gpu = uploadGpuMesh(device, frameToken(), verts, indices);
106 }
107 auto handle = makeMeshHandle(*gpu);
108 // Retain the CPU morph base pose only when the mesh actually has morphs;
109 // otherwise the base pos/nrm/uv copies would linger at ~32B/vertex for no reason.
110 if (mesh.mNumAnimMeshes > 0) {
111 handle->initMorphBase(int(mesh.mNumVertices), basePos.data(), baseNrm.data(), baseUv.data());
112 }
113 // Assimp morph targets (VRM / glTF blend shapes often land here).
114 for (unsigned m = 0; m < mesh.mNumAnimMeshes; ++m) {
115 const aiAnimMesh *am = mesh.mAnimMeshes[m];
116 if (!am || !am->mVertices || am->mNumVertices != mesh.mNumVertices) continue;
117 std::string morphName =
118 am->mName.length ? am->mName.C_Str() : ("morph" + std::to_string(m));
119 std::vector<float> absPos(size_t(am->mNumVertices) * 3u);
120 for (unsigned i = 0; i < am->mNumVertices; ++i) {
121 absPos[size_t(i) * 3u + 0] = am->mVertices[i].x;
122 absPos[size_t(i) * 3u + 1] = am->mVertices[i].y;
123 absPos[size_t(i) * 3u + 2] = am->mVertices[i].z;
124 }
125 handle->addMorphTargetAbsolute(morphName, absPos.data());
126 }
127 handle->markMorphClean();
128 Mesh *raw = handle.get();
129 assignMeshBounds(raw, verts);
130 registerMeshRecord(gpu.get());
131 ownedGpuMeshes.push_back(std::move(gpu));
132 ownedMeshes.push_back(std::move(handle));
133 return raw;
134}

136Mesh *Graphics::newMeshFromAssimp(const ::aiMesh &mesh, const aiMatrix4x4 &worldTransform) {
137 ASSERT(initialized);
138 if (!initialized) throw Exception("newMeshFromAssimp: graphics not initialized");
139 if (mesh.mNumVertices == 0 || mesh.mNumFaces == 0)
140 throw Exception("newMeshFromAssimp: empty mesh");
141
142 std::vector<aiVector3D> positions(mesh.mNumVertices);
143 std::vector<aiVector3D> normals(mesh.mNumVertices);
144 aiMatrix3x3 nmat(worldTransform);
145 const float ndet = nmat.Determinant();
146 if (std::fabs(ndet) > 1e-8f) {
147 nmat.Inverse();
148 nmat.Transpose();
149 }
150 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
151 positions[i] = worldTransform * mesh.mVertices[i];
152 if (mesh.HasNormals()) {
153 normals[i] = nmat * mesh.mNormals[i];
154 normals[i].Normalize();
155 } else {
156 normals[i] = aiVector3D(0.f, 1.f, 0.f);
157 }
158 }
159
160 // Negative determinant mirrors the mesh: winding flips while inverse-transpose
161 // keeps normals consistent, so back-face cull would hide the visible side.
162 const bool flipWinding = worldTransform.Determinant() < 0.f;
163 std::vector<aiFace> flippedFaces;
164 std::vector<unsigned> flippedIdx;
165 if (flipWinding) {
166 flippedFaces.resize(mesh.mNumFaces);
167 flippedIdx.resize(size_t(mesh.mNumFaces) * 3u);
168 for (unsigned f = 0; f < mesh.mNumFaces; ++f) {
169 const aiFace &src = mesh.mFaces[f];
170 aiFace &dst = flippedFaces[f];
171 dst.mNumIndices = src.mNumIndices;
172 if (src.mNumIndices == 3 && src.mIndices) {
173 unsigned *idx = flippedIdx.data() + size_t(f) * 3u;
174 idx[0] = src.mIndices[0];
175 idx[1] = src.mIndices[2];
176 idx[2] = src.mIndices[1];
177 dst.mIndices = idx;
178 } else {
179 dst.mIndices = src.mIndices;
180 }
181 }
182 }
183
184 // Non-owning view — do not let aiMesh destructor free borrowed pointers.
185 aiMesh tmp{};
186 tmp.mPrimitiveTypes = mesh.mPrimitiveTypes;
187 tmp.mNumVertices = mesh.mNumVertices;
188 tmp.mVertices = positions.data();
189 tmp.mNormals = normals.data();
190 tmp.mNumFaces = mesh.mNumFaces;
191 tmp.mFaces = flipWinding ? flippedFaces.data() : mesh.mFaces;
192 tmp.mMaterialIndex = mesh.mMaterialIndex;
193 tmp.mNumAnimMeshes = mesh.mNumAnimMeshes;
194 tmp.mAnimMeshes = mesh.mAnimMeshes;
195 if (mesh.HasTextureCoords(0)) {
196 tmp.mTextureCoords[0] = mesh.mTextureCoords[0];
197 tmp.mNumUVComponents[0] = mesh.mNumUVComponents[0];
198 }
199 Mesh *out = newMeshFromAssimp(tmp);
200 tmp.mVertices = nullptr;
201 tmp.mNormals = nullptr;
202 tmp.mFaces = nullptr;
203 tmp.mTextureCoords[0] = nullptr;
204 tmp.mAnimMeshes = nullptr;
205 return out;
206}

208Mesh *Graphics::newMeshFromArrays(const float *posXYZ, const float *nrmXYZ, const float *uvST,
209 int vertexCount, const uint32_t *indices, int indexCount) {
210 ASSERT(initialized);
211 if (!initialized) throw Exception("newMeshFromArrays: graphics not initialized");
212 if (!posXYZ || vertexCount <= 0) throw Exception("newMeshFromArrays: empty positions");
213 if (!indices || indexCount < 3) throw Exception("newMeshFromArrays: empty indices");
214 if (indexCount % 3 != 0) throw Exception("newMeshFromArrays: indexCount must be multiple of 3");
215
216 std::vector<MeshVertex> verts(static_cast<size_t>(vertexCount));
217 for (int i = 0; i < vertexCount; ++i) {
218 MeshVertex &v = verts[static_cast<size_t>(i)];
219 v.pos = {posXYZ[size_t(i) * 3u], posXYZ[size_t(i) * 3u + 1u], posXYZ[size_t(i) * 3u + 2u]};
220 if (nrmXYZ)
221 v.normal = {nrmXYZ[size_t(i) * 3u], nrmXYZ[size_t(i) * 3u + 1u],
222 nrmXYZ[size_t(i) * 3u + 2u]};
223 else
224 v.normal = {0.f, 1.f, 0.f};
225 if (uvST)
226 v.uv = {uvST[size_t(i) * 2u], uvST[size_t(i) * 2u + 1u]};
227 else
228 v.uv = {0.f, 0.f};
229 }
230
231 std::vector<uint32_t> idx(indices, indices + indexCount);
232 for (uint32_t id : idx) {
233 if (int(id) >= vertexCount) throw Exception("newMeshFromArrays: index out of range");
234 }
235
236 auto gpu = uploadGpuMesh(device, frameToken(), verts, idx);
237 auto handle = makeMeshHandle(*gpu);
238 Mesh *raw = handle.get();
239 raw->computeBounds(posXYZ, vertexCount);
240 registerMeshRecord(gpu.get());
241 ownedGpuMeshes.push_back(std::move(gpu));
242 ownedMeshes.push_back(std::move(handle));
243 return raw;
244}

246bool Graphics::bakeMeshMorph(Mesh *mesh) {
247 if (!mesh || !mesh->gpuHandle || !mesh->hasMorphData() || !mesh->isMorphDirty()) return false;
248 if (!initialized) return false;
249
250 std::vector<float> pos;
251 std::vector<float> nrm;
252 mesh->computeMorphedPositions(pos, nrm);
253 const int vc = mesh->getVertexCount();
254 if (vc <= 0 || int(pos.size()) < vc * 3) return false;
255 mesh->computeBounds(pos.data(), vc);
256
257 std::vector<MeshVertex> verts(static_cast<size_t>(vc));
258 const auto &uv = mesh->baseUv();
259 for (int i = 0; i < vc; ++i) {
260 MeshVertex &v = verts[static_cast<size_t>(i)];
261 v.pos = {pos[size_t(i) * 3u + 0], pos[size_t(i) * 3u + 1], pos[size_t(i) * 3u + 2]};
262 if (int(nrm.size()) >= (i + 1) * 3)
263 v.normal = {nrm[size_t(i) * 3u + 0], nrm[size_t(i) * 3u + 1], nrm[size_t(i) * 3u + 2]};
264 else
265 v.normal = {0.f, 1.f, 0.f};
266 if (int(uv.size()) >= (i + 1) * 2)
267 v.uv = {uv[size_t(i) * 2u + 0], uv[size_t(i) * 2u + 1]};
268 else
269 v.uv = {0.f, 0.f};
270 }
271
272 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
273 // Ring-buffered host-visible VBO: the next copy is kDynamicVertexCopies
274 // frames old, so overwriting it never races with in-flight draws — no
275 // device-wide wait (see writeDynamicMesh).
276 ensureDynamicRing(*gpu);
277 writeDynamicMesh(*gpu, verts, getDevice(), frameToken(), nullptr, 0);
278 mesh->markMorphClean();
279 return true;
280}

282bool Graphics::updateMeshVertices(Mesh *mesh, const float *posXYZ, const float *nrmXYZ,
283 const float *uvST, int vertexCount, const uint32_t *indices,
284 int indexCount) {
285 if (!initialized || !mesh || !mesh->gpuHandle) return false;
286 if (!posXYZ || vertexCount <= 0) return false;
287 if (indexCount > 0 && (indexCount % 3 != 0 || !indices)) return false;
288
289 std::vector<MeshVertex> verts(static_cast<size_t>(vertexCount));
290 for (int i = 0; i < vertexCount; ++i) {
291 MeshVertex &v = verts[static_cast<size_t>(i)];
292 v.pos = {posXYZ[size_t(i) * 3u], posXYZ[size_t(i) * 3u + 1u], posXYZ[size_t(i) * 3u + 2u]};
293 if (nrmXYZ)
294 v.normal = {nrmXYZ[size_t(i) * 3u], nrmXYZ[size_t(i) * 3u + 1u],
295 nrmXYZ[size_t(i) * 3u + 2u]};
296 else
297 v.normal = {0.f, 1.f, 0.f};
298 if (uvST)
299 v.uv = {uvST[size_t(i) * 2u], uvST[size_t(i) * 2u + 1u]};
300 else
301 v.uv = {0.f, 0.f};
302 }
303
304 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
305 mesh->computeBounds(posXYZ, vertexCount);
306 // Same ring-buffer approach as bakeMeshMorph: never wait on in-flight
307 // frames, just write the next copy.
308 ensureDynamicRing(*gpu);
309 writeDynamicMesh(*gpu, verts, getDevice(), frameToken(), indices, indexCount);
310 mesh->indexCount = int(gpu->indexCount);
311 return true;
312}

314bool Graphics::releaseMesh(Mesh *mesh) {
315 if (!mesh || !mesh->gpuHandle) return false;
316
317 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
318 auto gpuIt = std::find_if(ownedGpuMeshes.begin(), ownedGpuMeshes.end(),
319 [&](const std::unique_ptr<GpuMesh> &g) {
320 return g.get() == gpu;
321 });
322 if (gpuIt == ownedGpuMeshes.end()) return false;
323
324 auto meshIt = std::find_if(ownedMeshes.begin(), ownedMeshes.end(),
325 [&](const std::unique_ptr<Mesh> &m) {
326 return m.get() == mesh;
327 });
328 if (meshIt == ownedMeshes.end()) return false;
329
330 // An in-flight draw may still read the vertex/index buffers; drain first.
331 waitForSharedGpuResources();
332 mesh->gpuHandle = nullptr;
333 ownedGpuMeshes.erase(gpuIt);
334 // Transfer the CPU facade to the caller instead of destroying it.
335 (void)meshIt->release();
336 ownedMeshes.erase(meshIt);
337 return true;
338}

340Mesh *Graphics::newMeshSphere(int slices, int stacks) {
341 ASSERT(initialized);
342 if (!initialized) throw Exception("newMeshSphere: graphics not initialized");
343 if (slices < 3) slices = 3;
344 if (stacks < 2) stacks = 2;
345 if (slices > 256) slices = 256;
346 if (stacks > 128) stacks = 128;
347
348 constexpr float kPi = 3.14159265358979323846f;
349 constexpr float kTwoPi = kPi * 2.f;
350
351 // Layout (stride = slices+1, seam column duplicated for continuous U):
352 // row 0: north pole verts (same pos, unique U)
353 // row 1..stacks-1: latitude rings
354 // row stacks: south pole verts
355 // One pole vertex per longitude avoids a single-fan UV singularity and
356 // keeps every triangle non-degenerate.
357 const int stride = slices + 1;
358 const int rows = stacks + 1; // includes both pole rows
359 std::vector<MeshVertex> verts;
360 verts.reserve(size_t(stride) * size_t(rows));
361
362 auto pushVert = [&](float px, float py, float pz, float u, float v) {
363 MeshVertex vert{};
364 vert.pos = {px, py, pz};
365 vert.normal = {px, py, pz}; // unit sphere
366 vert.uv = {u, v};
367 verts.push_back(vert);
368 };
369
370 for (int y = 0; y <= stacks; ++y) {
371 const float fv = float(y) / float(stacks);
372 const float phi = fv * kPi;
373 const float sinPhi = std::sin(phi);
374 const float cosPhi = std::cos(phi);
375 for (int x = 0; x <= slices; ++x) {
376 const float u = float(x) / float(slices);
377 const float theta = u * kTwoPi;
378 // Exact poles: collapse ring to a point but keep per-slice UVs.
379 if (y == 0)
380 pushVert(0.f, 1.f, 0.f, u, 0.f);
381 else if (y == stacks)
382 pushVert(0.f, -1.f, 0.f, u, 1.f);
383 else
384 pushVert(sinPhi * std::cos(theta), cosPhi, sinPhi * std::sin(theta), u, fv);
385 }
386 }
387
388 std::vector<uint32_t> indices;
389 indices.reserve(size_t(slices) * size_t(stacks) * 6u);
390
391 // Quads between consecutive rows. Winding must be consistent for outward
392 // faces (object-space CCW; mesh pipelines use Clockwise frontFace after the
393 // Vulkan Y flip in perspectiveVulkanRH_ZO). Use the same winding that
394 // closed the south pole in-game: rowA[x], rowB[x], rowA[x+1] /
395 // rowA[x+1], rowB[x], rowB[x+1] — derived from ring→next with
396 // (i0,i2,i1)+(i1,i2,i3) which equals (lon,lat)->(lon,lat+1)->(lon+1,lat).
397 for (int y = 0; y < stacks; ++y) {
398 const uint32_t row0 = uint32_t(y * stride);
399 const uint32_t row1 = uint32_t((y + 1) * stride);
400 for (int x = 0; x < slices; ++x) {
401 const uint32_t i0 = row0 + uint32_t(x);
402 const uint32_t i1 = row0 + uint32_t(x + 1);
403 const uint32_t i2 = row1 + uint32_t(x);
404 const uint32_t i3 = row1 + uint32_t(x + 1);
405 // Outward for RH Y-up (verified against south-cap fix): i0,i2,i1 + i1,i2,i3
406 indices.push_back(i0);
407 indices.push_back(i2);
408 indices.push_back(i1);
409 indices.push_back(i1);
410 indices.push_back(i2);
411 indices.push_back(i3);
412 }
413 }
414
415 auto gpu = uploadGpuMesh(device, frameToken(), verts, indices);
416 auto handle = makeMeshHandle(*gpu);
417 Mesh *raw = handle.get();
418 assignMeshBounds(raw, verts);
419 registerMeshRecord(gpu.get());
420 ownedGpuMeshes.push_back(std::move(gpu));
421 ownedMeshes.push_back(std::move(handle));
422 return raw;
423}

425Mesh *Graphics::newMeshCylinder(int slices, int stacks, bool caps) {
426 ASSERT(initialized);
427 if (!initialized) throw Exception("newMeshCylinder: graphics not initialized");
428 if (slices < 3) slices = 3;
429 if (stacks < 1) stacks = 1;
430 if (slices > 256) slices = 256;
431 if (stacks > 128) stacks = 128;
432
433 constexpr float kPi = 3.14159265358979323846f;
434 constexpr float kTwoPi = kPi * 2.f;
435 constexpr float kRadius = 1.f;
436 constexpr float kHalfH = 1.f; // height 2, Y from -1..1
437
438 const int stride = slices + 1; // duplicated seam for continuous U
439 const int sideRows = stacks + 1;
440 std::vector<MeshVertex> verts;
441 verts.reserve(size_t(stride) * size_t(sideRows) + size_t(caps ? 2 * (slices + 2) : 0));
442
443 auto pushVert = [&](float px, float py, float pz, float nx, float ny, float nz, float u,
444 float v) {
445 MeshVertex vert{};
446 vert.pos = {px, py, pz};
447 vert.normal = {nx, ny, nz};
448 vert.uv = {u, v};
449 verts.push_back(vert);
450 };
451
452 // Side wall: outward normals in XZ.
453 for (int y = 0; y <= stacks; ++y) {
454 const float fv = float(y) / float(stacks);
455 const float py = kHalfH - fv * (2.f * kHalfH);
456 for (int x = 0; x <= slices; ++x) {
457 const float u = float(x) / float(slices);
458 const float theta = u * kTwoPi;
459 const float cx = std::cos(theta);
460 const float sz = std::sin(theta);
461 pushVert(kRadius * cx, py, kRadius * sz, cx, 0.f, sz, u, fv);
462 }
463 }
464
465 std::vector<uint32_t> indices;
466 indices.reserve(size_t(slices) * size_t(stacks) * 6u +
467 size_t(caps ? slices * 2 * 3 : 0));
468
469 for (int y = 0; y < stacks; ++y) {
470 const uint32_t row0 = uint32_t(y * stride);
471 const uint32_t row1 = uint32_t((y + 1) * stride);
472 for (int x = 0; x < slices; ++x) {
473 const uint32_t i0 = row0 + uint32_t(x);
474 const uint32_t i1 = row0 + uint32_t(x + 1);
475 const uint32_t i2 = row1 + uint32_t(x);
476 const uint32_t i3 = row1 + uint32_t(x + 1);
477 indices.push_back(i0);
478 indices.push_back(i2);
479 indices.push_back(i1);
480 indices.push_back(i1);
481 indices.push_back(i2);
482 indices.push_back(i3);
483 }
484 }
485
486 if (caps) {
487 // Top cap (y = +1, normal +Y) — fan from center.
488 const uint32_t topCenter = uint32_t(verts.size());
489 pushVert(0.f, kHalfH, 0.f, 0.f, 1.f, 0.f, 0.5f, 0.5f);
490 const uint32_t topRing = uint32_t(verts.size());
491 for (int x = 0; x <= slices; ++x) {
492 const float u = float(x) / float(slices);
493 const float theta = u * kTwoPi;
494 const float cx = std::cos(theta);
495 const float sz = std::sin(theta);
496 pushVert(kRadius * cx, kHalfH, kRadius * sz, 0.f, 1.f, 0.f, 0.5f + 0.5f * cx,
497 0.5f + 0.5f * sz);
498 }
499 for (int x = 0; x < slices; ++x) {
500 indices.push_back(topCenter);
501 indices.push_back(topRing + uint32_t(x));
502 indices.push_back(topRing + uint32_t(x + 1));
503 }
504
505 // Bottom cap (y = -1, normal -Y).
506 const uint32_t botCenter = uint32_t(verts.size());
507 pushVert(0.f, -kHalfH, 0.f, 0.f, -1.f, 0.f, 0.5f, 0.5f);
508 const uint32_t botRing = uint32_t(verts.size());
509 for (int x = 0; x <= slices; ++x) {
510 const float u = float(x) / float(slices);
511 const float theta = u * kTwoPi;
512 const float cx = std::cos(theta);
513 const float sz = std::sin(theta);
514 pushVert(kRadius * cx, -kHalfH, kRadius * sz, 0.f, -1.f, 0.f, 0.5f + 0.5f * cx,
515 0.5f + 0.5f * sz);
516 }
517 for (int x = 0; x < slices; ++x) {
518 // CW when viewed from below so outward (-Y) faces are CCW from outside.
519 indices.push_back(botCenter);
520 indices.push_back(botRing + uint32_t(x + 1));
521 indices.push_back(botRing + uint32_t(x));
522 }
523 }
524
525 auto gpu = uploadGpuMesh(device, frameToken(), verts, indices);
526 auto handle = makeMeshHandle(*gpu);
527 Mesh *raw = handle.get();
528 assignMeshBounds(raw, verts);
529 registerMeshRecord(gpu.get());
530 ownedGpuMeshes.push_back(std::move(gpu));
531 ownedMeshes.push_back(std::move(handle));
532 return raw;
533}

539void Graphics::drawMeshShader(Mesh *mesh, const glm::mat4 &model, Texture *texture, const Color &tint,
540 Shader *shader) {
541 ASSERT(initialized);
542 ASSERT(mesh != nullptr);
543 if (!initialized) throw Exception("drawMesh: graphics not initialized");
544 if (!mesh || !mesh->gpuHandle) throw Exception("drawMesh: null mesh");
545 if (!swapchainPassOpen && !offscreen3DPassOpen)
546 throw Exception("drawMesh: call begin3DFrame first");
547 if (!mesh3dPipeline) throw Exception("drawMesh: mesh3d pipeline missing");
548
549 if (shader) {
550 if (shader->getKind() != Shader::Kind::eMesh3D)
551 throw Exception("drawMesh: shader is not a Mesh3D shader (use newMeshShader*)");
552 if (!shader->gpuHandle) throw Exception("drawMesh: shader has no GPU pipeline");
553 }
554
555 auto *gpuMesh = static_cast<GpuMesh *>(mesh->gpuHandle);
556 Texture *tex = texture ? texture : whiteTexture;
557 if (!tex || !tex->gpuHandle) throw Exception("drawMesh: missing texture");
558 auto *gpuTex = static_cast<GpuTexture *>(tex->gpuHandle);
559
560 ensureFlatNormalTexture3D();
561 Texture *ntex = mesh3dNormalTexture ? mesh3dNormalTexture : flatNormalTexture3D;
562 if (!ntex || !ntex->gpuHandle) throw Exception("drawMesh: missing normal texture");
563 auto *gpuNormal = static_cast<GpuTexture *>(ntex->gpuHandle);
564
565 ensureFlatHeightTexture3D();
566 Texture *htex = mesh3dHeightTexture ? mesh3dHeightTexture : flatHeightTexture3D;
567 if (!htex || !htex->gpuHandle) throw Exception("drawMesh: missing height texture");
568 auto *gpuHeight = static_cast<GpuTexture *>(htex->gpuHandle);
569
570 Texture *depthTex = mesh3dSceneDepthTexture ? mesh3dSceneDepthTexture : whiteTexture;
571 if (!depthTex || !depthTex->gpuHandle) throw Exception("drawMesh: missing scene depth texture");
572 auto *gpuDepth = static_cast<GpuTexture *>(depthTex->gpuHandle);
573
574 ensureDefaultEnvCubemap();
575 Texture *envTex = mesh3dEnvTexture ? mesh3dEnvTexture : defaultEnvCubemap;
576 if (!envTex || !envTex->gpuHandle) throw Exception("drawMesh: missing env cubemap");
577 auto *gpuEnv = static_cast<GpuTexture *>(envTex->gpuHandle);
578 if (!gpuEnv->isCube) throw Exception("drawMesh: env texture is not a cubemap");
579 const float envIntensity = (mesh3dEnvTexture && mesh3dEnvIntensity > 0.f) ? mesh3dEnvIntensity : 0.f;
580
581 const bool useClustered = mesh3dClusteredActive && !shader && mesh3dClusteredPipeline;
582 auto &cb = currentPresentCb();
583
584 auto makeShadowUbo = [&]() {
585 ShadowUBO s = mesh3dShadows.ubo;
586 if (!mesh3dShadows.active) {
587 s.bias.y = 0.f;
588 s.splits.w = 0.f;
589 }
590 s.bias.z = mesh3dShadowReceive ? 1.f : 0.f;
591 return s;
592 };
593
594 if (useClustered) {
595 Mesh3DClusteredUBO ubo{};
596 ubo.model = model;
597 ubo.mvp = mesh3dFrameUbo.mvp * model;
598 ubo.view = mesh3dClustered.view;
599 ubo.lightDir = mesh3dClustered.primaryDir;
600 ubo.lightColor = glm::vec4(glm::vec3(mesh3dClustered.primaryColor), envIntensity);
601 ubo.tint = glm::vec4(tint.r, tint.g, tint.b, tint.a);
602 ubo.cameraPos = glm::vec4(glm::vec3(mesh3dFrameUbo.cameraPos), mesh3dRoughness);
603 ubo.ambient = glm::vec4(glm::vec3(mesh3dClustered.ambient), mesh3dMetallic);
604 ubo.gridInfo = mesh3dClustered.gridInfo;
605 ubo.clipInfo = mesh3dClustered.clipInfo;
606 ubo.texBomb = glm::vec4(mesh3dTexBombScale, mesh3dTexBombStrength, mesh3dTexBombRot, 0.f);
607 ubo.parallax =
608 glm::vec4(mesh3dParallaxScale, mesh3dParallaxMinLayers, mesh3dParallaxMaxLayers, 0.f);
609
610 auto &cfslots = currentMesh3dClusteredFrameSlots();
611 if (cfslots.drawIndex >= cfslots.capacity) {
612 std::fprintf(stderr,
613 "[vulkan] clustered mesh3d UBO ring exhausted (%zu draws); draw skipped\n",
614 cfslots.capacity);
615 return;
616 }
617 const size_t slot = cfslots.drawIndex++;
618 ensureMesh3dStrides();
619 const uint32_t uboOffset = uint32_t(slot) * mesh3dClusteredUboStride;
620 const uint32_t shadowOffset = uint32_t(slot) * shadowUboStride;
621 updateRingLocal(cfslots.uboRing, uboOffset, &ubo, sizeof(ubo));
622 const ShadowUBO shadow = makeShadowUbo();
623 updateRingLocal(cfslots.shadowRing, shadowOffset, &shadow, sizeof(shadow));
624 vk::DescriptorSet set =
625 mesh3dClusteredSetFor(gpuTex, gpuNormal, gpuEnv, gpuHeight, cfslots);
626 const uint32_t dynOffsets[2] = {uboOffset, shadowOffset};
627
628 if (mesh3dClusteredPipeline != lastMesh3dClusteredPipeline) {
629 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, mesh3dClusteredPipeline);
630 lastMesh3dClusteredPipeline = mesh3dClusteredPipeline;
631 }
632 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, mesh3dClusteredPipelineLayout, 0, 1,
633 &set, 2, dynOffsets);
634 drawIndexedMesh(cb, *gpuMesh);
635 return;
636 }
637
638 Mesh3DUBO ubo = mesh3dFrameUbo;
639 ubo.model = model;
640 ubo.mvp = mesh3dFrameUbo.mvp * model;
641 ubo.tint = glm::vec4(tint.r, tint.g, tint.b, tint.a);
642 ubo.ambient = glm::vec4(glm::vec3(mesh3dLighting.ambient), mesh3dMetallic);
643 const int lightCount = std::max(0, std::min(mesh3dLighting.count, Lighting3DPack::kMaxLights));
644 ubo.lightDir.w = float(lightCount);
645 ubo.cameraPos.w = mesh3dRoughness;
646 ubo.lightColor.w = envIntensity;
647 ubo.texBomb = glm::vec4(mesh3dTexBombScale, mesh3dTexBombStrength, mesh3dTexBombRot, 0.f);
648 ubo.parallax =
649 glm::vec4(mesh3dParallaxScale, mesh3dParallaxMinLayers, mesh3dParallaxMaxLayers, 0.f);
650 for (int i = 0; i < lightCount; ++i) ubo.lights[i] = mesh3dLighting.lights[i];
651 int dirI = -1;
652 for (int i = 0; i < lightCount; ++i) {
653 if (mesh3dLighting.lights[i].posRadius.w <= 0.f) {
654 dirI = i;
655 break;
656 }
657 }
658 if (dirI >= 0) {
659 glm::vec3 d(mesh3dLighting.lights[dirI].posRadius);
660 if (glm::length(d) < 1e-6f) d = glm::vec3(0.f, 1.f, 0.f);
661 else d = glm::normalize(d);
662 ubo.lightDir = glm::vec4(d, float(lightCount));
663 ubo.lightColor = glm::vec4(glm::vec3(mesh3dLighting.lights[dirI].color), envIntensity);
664 } else {
665 ubo.lightDir = glm::vec4(0.f, 1.f, 0.f, float(lightCount));
666 ubo.lightColor = glm::vec4(0.f, 0.f, 0.f, envIntensity);
667 }
668
669 auto &fslots = currentMesh3dFrameSlots();
670 if (fslots.drawIndex >= fslots.capacity) {
671 std::fprintf(stderr, "[vulkan] mesh3d UBO ring exhausted (%zu draws); draw skipped\n",
672 fslots.capacity);
673 return;
674 }
675 const size_t slot = fslots.drawIndex++;
676 ensureMesh3dStrides();
677 const uint32_t uboOffset = uint32_t(slot) * mesh3dUboStride;
678 const uint32_t shadowOffset = uint32_t(slot) * shadowUboStride;
679 updateRingLocal(fslots.uboRing, uboOffset, &ubo, sizeof(ubo));
680 const ShadowUBO shadow = makeShadowUbo();
681 updateRingLocal(fslots.shadowRing, shadowOffset, &shadow, sizeof(shadow));
682 vk::DescriptorSet set = mesh3dSetFor(gpuTex, gpuNormal, gpuEnv, gpuHeight, gpuDepth, fslots);
683 const uint32_t dynOffsets[2] = {uboOffset, shadowOffset};
684
685 if (shader) {
686 if (offscreen3DPassOpen)
687 throw Exception("drawMeshShader: custom mesh shader in offscreen 3D pass is unsupported");
688 auto *gs = static_cast<GpuShader *>(shader->gpuHandle);
689 vk::Pipeline activePipeline = gs->mesh3dPipeline;
690 if (shader->isXray()) {
691 // X-ray silhouette pass: depth test/write off + alpha blend so the
692 // occluded part paints over the building. The pipeline is created
693 // with the shader (see newMeshShaderFromSpv); do not compile it
694 // here — a render pass is already open.
695 activePipeline = gs->mesh3dXrayPipeline;
696 }
697 if (!activePipeline) return;
698 if (activePipeline != lastMesh3dPipeline) {
699 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, activePipeline);
700 lastMesh3dPipeline = activePipeline;
701 }
702 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, mesh3dShaderPipelineLayout, 0, 1, &set,
703 2, dynOffsets);
704 cb.pushConstants(mesh3dShaderPipelineLayout,
705 vk::ShaderStageFlagBits::eVertex | vk::ShaderStageFlagBits::eFragment, 0,
706 Shader::kPushConstantBytes, shader->pushConstantData());
707 } else {
708 const vk::Pipeline pipe = offscreen3DPassOpen ? offscreen3DMeshPipeline : mesh3dPipeline;
709 if (pipe != lastMesh3dPipeline) {
710 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, pipe);
711 lastMesh3dPipeline = pipe;
712 }
713 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, mesh3dPipelineLayout, 0, 1, &set, 2,
714 dynOffsets);
715 }
716 drawIndexedMesh(cb, *gpuMesh);
717}
