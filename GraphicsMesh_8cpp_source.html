52Mesh *Graphics::newMeshFromAssimp(const ::aiMesh &mesh) {
53 ASSERT(initialized);
54 if (!initialized) throw Exception("newMeshFromAssimp: graphics not initialized");
55 if (mesh.mNumVertices == 0 || mesh.mNumFaces == 0)
56 throw Exception("newMeshFromAssimp: empty mesh");
57
58 std::vector<MeshVertex> verts;
59 verts.reserve(mesh.mNumVertices);
60 std::vector<float> basePos;
61 std::vector<float> baseNrm;
62 std::vector<float> baseUv;
63 if (mesh.mNumAnimMeshes > 0) {
64 basePos.reserve(mesh.mNumVertices * 3);
65 baseNrm.reserve(mesh.mNumVertices * 3);
66 baseUv.reserve(mesh.mNumVertices * 2);
67 }
68 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
69 MeshVertex v{};
70 v.pos = {mesh.mVertices[i].x, mesh.mVertices[i].y, mesh.mVertices[i].z};
71 if (mesh.HasNormals())
72 v.normal = {mesh.mNormals[i].x, mesh.mNormals[i].y, mesh.mNormals[i].z};
73 else
74 v.normal = {0.f, 1.f, 0.f};
75 if (mesh.HasTangentsAndBitangents()) {
76 const auto &tangent = mesh.mTangents[i];
77 const auto &bitangent = mesh.mBitangents[i];
78 const glm::vec3 t(tangent.x, tangent.y, tangent.z);
79 const glm::vec3 b(bitangent.x, bitangent.y, bitangent.z);
80 v.tangent = glm::vec4(t, glm::dot(glm::cross(v.normal, t), b) < 0.f ? -1.f : 1.f);
81 }
82 if (mesh.HasTextureCoords(0))
83 v.uv = {mesh.mTextureCoords[0][i].x, mesh.mTextureCoords[0][i].y};
84 else
85 v.uv = {0.f, 0.f};
86 verts.push_back(v);
87 if (mesh.mNumAnimMeshes > 0) {
88 basePos.push_back(v.pos.x);
89 basePos.push_back(v.pos.y);
90 basePos.push_back(v.pos.z);
91 baseNrm.push_back(v.normal.x);
92 baseNrm.push_back(v.normal.y);
93 baseNrm.push_back(v.normal.z);
94 baseUv.push_back(v.uv.x);
95 baseUv.push_back(v.uv.y);
96 }
97 }
98
99 std::vector<uint32_t> indices;
100 indices.reserve(mesh.mNumFaces * 3);
101 for (unsigned f = 0; f < mesh.mNumFaces; ++f) {
102 const aiFace &face = mesh.mFaces[f];
103 if (face.mNumIndices != 3) continue;
104 indices.push_back(face.mIndices[0]);
105 indices.push_back(face.mIndices[1]);
106 indices.push_back(face.mIndices[2]);
107 }
108 if (indices.empty()) throw Exception("newMeshFromAssimp: no triangle faces");
109
110 std::unique_ptr<GpuMesh> gpu;
111 if (mesh.mNumVertices <= 65535u) {
112 std::vector<uint16_t> idx16;
113 idx16.reserve(indices.size());
114 for (uint32_t i : indices) idx16.push_back(uint16_t(i));
115 gpu = uploadGpuMesh16(device, frameToken(), verts, idx16);
116 } else {
117 gpu = uploadGpuMesh(device, frameToken(), verts, indices);
118 }
119 auto handle = makeMeshHandle(*gpu);
120 handle->captureImportedAttributes(mesh);
121 // Retain the CPU morph base pose only when the mesh actually has morphs;
122 // otherwise the base pos/nrm/uv copies would linger at ~32B/vertex for no reason.
123 if (mesh.mNumAnimMeshes > 0) {
124 handle->initMorphBase(int(mesh.mNumVertices), basePos.data(), baseNrm.data(), baseUv.data());
125 }
126 // Assimp morph targets (VRM / glTF blend shapes often land here).
127 for (unsigned m = 0; m < mesh.mNumAnimMeshes; ++m) {
128 const aiAnimMesh *am = mesh.mAnimMeshes[m];
129 if (!am || !am->mVertices || am->mNumVertices != mesh.mNumVertices) continue;
130 std::string morphName =
131 am->mName.length ? am->mName.C_Str() : ("morph" + std::to_string(m));
132 std::vector<float> absPos(size_t(am->mNumVertices) * 3u);
133 for (unsigned i = 0; i < am->mNumVertices; ++i) {
134 absPos[size_t(i) * 3u + 0] = am->mVertices[i].x;
135 absPos[size_t(i) * 3u + 1] = am->mVertices[i].y;
136 absPos[size_t(i) * 3u + 2] = am->mVertices[i].z;
137 }
138 handle->addMorphTargetAbsolute(morphName, absPos.data());
139 }
140 handle->markMorphClean();
141 Mesh *raw = handle.get();
142 assignMeshBounds(raw, verts);
143 registerMeshRecord(gpu.get(), &verts, &indices);
144 ownedGpuMeshes.push_back(std::move(gpu));
145 ownedMeshes.push_back(std::move(handle));
146 return raw;
147}

149Mesh *Graphics::newMeshFromAssimp(const ::aiMesh &mesh, const aiMatrix4x4 &worldTransform) {
150 ASSERT(initialized);
151 if (!initialized) throw Exception("newMeshFromAssimp: graphics not initialized");
152 if (mesh.mNumVertices == 0 || mesh.mNumFaces == 0)
153 throw Exception("newMeshFromAssimp: empty mesh");
154
155 std::vector<aiVector3D> positions(mesh.mNumVertices);
156 std::vector<aiVector3D> normals(mesh.mNumVertices);
157 std::vector<aiVector3D> tangents;
158 std::vector<aiVector3D> bitangents;
159 if (mesh.HasTangentsAndBitangents()) {
160 tangents.resize(mesh.mNumVertices);
161 bitangents.resize(mesh.mNumVertices);
162 const aiMatrix3x3 linear(worldTransform);
163 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
164 tangents[i] = linear * mesh.mTangents[i];
165 bitangents[i] = linear * mesh.mBitangents[i];
166 tangents[i].Normalize();
167 bitangents[i].Normalize();
168 }
169 }
170 aiMatrix3x3 nmat(worldTransform);
171 const float ndet = nmat.Determinant();
172 if (std::fabs(ndet) > 1e-8f) {
173 nmat.Inverse();
174 nmat.Transpose();
175 }
176 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
177 positions[i] = worldTransform * mesh.mVertices[i];
178 if (mesh.HasNormals()) {
179 normals[i] = nmat * mesh.mNormals[i];
180 normals[i].Normalize();
181 } else {
182 normals[i] = aiVector3D(0.f, 1.f, 0.f);
183 }
184 }
185
186 // Negative determinant mirrors the mesh: winding flips while inverse-transpose
187 // keeps normals consistent, so back-face cull would hide the visible side.
188 const bool flipWinding = worldTransform.Determinant() < 0.f;
189 std::vector<aiFace> flippedFaces;
190 std::vector<unsigned> flippedIdx;
191 if (flipWinding) {
192 flippedFaces.resize(mesh.mNumFaces);
193 flippedIdx.resize(size_t(mesh.mNumFaces) * 3u);
194 for (unsigned f = 0; f < mesh.mNumFaces; ++f) {
195 const aiFace &src = mesh.mFaces[f];
196 aiFace &dst = flippedFaces[f];
197 dst.mNumIndices = src.mNumIndices;
198 if (src.mNumIndices == 3 && src.mIndices) {
199 unsigned *idx = flippedIdx.data() + size_t(f) * 3u;
200 idx[0] = src.mIndices[0];
201 idx[1] = src.mIndices[2];
202 idx[2] = src.mIndices[1];
203 dst.mIndices = idx;
204 } else {
205 dst.mIndices = src.mIndices;
206 }
207 }
208 }
209
210 // Non-owning view — do not let aiMesh destructor free borrowed pointers.
211 aiMesh tmp{};
212 tmp.mPrimitiveTypes = mesh.mPrimitiveTypes;
213 tmp.mNumVertices = mesh.mNumVertices;
214 tmp.mVertices = positions.data();
215 tmp.mNormals = normals.data();
216 tmp.mTangents = tangents.empty() ? nullptr : tangents.data();
217 tmp.mBitangents = bitangents.empty() ? nullptr : bitangents.data();
218 tmp.mNumFaces = mesh.mNumFaces;
219 tmp.mFaces = flipWinding ? flippedFaces.data() : mesh.mFaces;
220 tmp.mMaterialIndex = mesh.mMaterialIndex;
221 tmp.mNumAnimMeshes = mesh.mNumAnimMeshes;
222 tmp.mAnimMeshes = mesh.mAnimMeshes;
223 if (mesh.HasTextureCoords(0)) {
224 tmp.mTextureCoords[0] = mesh.mTextureCoords[0];
225 tmp.mNumUVComponents[0] = mesh.mNumUVComponents[0];
226 }
227 const auto detach = [&]() {
228 tmp.mVertices = nullptr;
229 tmp.mNormals = nullptr;
230 tmp.mTangents = nullptr;
231 tmp.mBitangents = nullptr;
232 tmp.mFaces = nullptr;
233 tmp.mTextureCoords[0] = nullptr;
234 tmp.mAnimMeshes = nullptr;
235 };
236 Mesh *out = nullptr;
237 try {
238 out = newMeshFromAssimp(tmp);
239 } catch (...) {
240 detach();
241 throw;
242 }
243 detach();
244 return out;
245}

247Mesh *Graphics::newMeshFromArrays(const float *posXYZ, const float *nrmXYZ, const float *uvST,
248 int vertexCount, const uint32_t *indices, int indexCount) {
249 ASSERT(initialized);
250 if (!initialized) throw Exception("newMeshFromArrays: graphics not initialized");
251 if (!posXYZ || vertexCount <= 0) throw Exception("newMeshFromArrays: empty positions");
252 if (!indices || indexCount < 3) throw Exception("newMeshFromArrays: empty indices");
253 if (indexCount % 3 != 0) throw Exception("newMeshFromArrays: indexCount must be multiple of 3");
254
255 std::vector<MeshVertex> verts(static_cast<size_t>(vertexCount));
256 for (int i = 0; i < vertexCount; ++i) {
257 MeshVertex &v = verts[static_cast<size_t>(i)];
258 v.pos = {posXYZ[size_t(i) * 3u], posXYZ[size_t(i) * 3u + 1u], posXYZ[size_t(i) * 3u + 2u]};
259 if (nrmXYZ)
260 v.normal = {nrmXYZ[size_t(i) * 3u], nrmXYZ[size_t(i) * 3u + 1u],
261 nrmXYZ[size_t(i) * 3u + 2u]};
262 else
263 v.normal = {0.f, 1.f, 0.f};
264 if (uvST)
265 v.uv = {uvST[size_t(i) * 2u], uvST[size_t(i) * 2u + 1u]};
266 else
267 v.uv = {0.f, 0.f};
268 }
269
270 std::vector<uint32_t> idx(indices, indices + indexCount);
271 for (uint32_t id : idx) {
272 if (int(id) >= vertexCount) throw Exception("newMeshFromArrays: index out of range");
273 }
274
275 auto gpu = uploadGpuMesh(device, frameToken(), verts, idx);
276 auto handle = makeMeshHandle(*gpu);
277 Mesh *raw = handle.get();
278 raw->computeBounds(posXYZ, vertexCount);
279 registerMeshRecord(gpu.get());
280 ownedGpuMeshes.push_back(std::move(gpu));
281 ownedMeshes.push_back(std::move(handle));
282 return raw;
283}

297bool Graphics::bakeMeshMorph(Mesh *mesh) {
298 if (!mesh || !mesh->gpuHandle || !mesh->hasMorphData() || !mesh->isMorphDirty()) return false;
299 if (!initialized) return false;
300
301 std::vector<float> pos;
302 std::vector<float> nrm;
303 mesh->computeMorphedPositions(pos, nrm);
304 const int vc = mesh->getVertexCount();
305 if (vc <= 0 || int(pos.size()) < vc * 3) return false;
306 mesh->computeBounds(pos.data(), vc);
307
308 std::vector<MeshVertex> verts(static_cast<size_t>(vc));
309 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
310 if (mesh->hasGpuSkinning()) {
311 // Morphs change positions/normals only; retain the authored skin influences.
312 auto &source = meshDrawVertices(*gpu);
313 void *mapped = source.map();
314 if (!mapped) return false;
315 std::memcpy(verts.data(), mapped, verts.size() * sizeof(MeshVertex));
316 source.unmap();
317 }
318 const auto &uv = mesh->baseUv();
319 for (int i = 0; i < vc; ++i) {
320 MeshVertex &v = verts[static_cast<size_t>(i)];
321 v.pos = {pos[size_t(i) * 3u + 0], pos[size_t(i) * 3u + 1], pos[size_t(i) * 3u + 2]};
322 if (int(nrm.size()) >= (i + 1) * 3)
323 v.normal = {nrm[size_t(i) * 3u + 0], nrm[size_t(i) * 3u + 1], nrm[size_t(i) * 3u + 2]};
324 else
325 v.normal = {0.f, 1.f, 0.f};
326 if (int(uv.size()) >= (i + 1) * 2)
327 v.uv = {uv[size_t(i) * 2u + 0], uv[size_t(i) * 2u + 1]};
328 else
329 v.uv = {0.f, 0.f};
330 }
331
332 // Ring-buffered host-visible VBO: the next copy is kDynamicVertexCopies
333 // frames old, so overwriting it never races with in-flight draws — no
334 // device-wide wait (see writeDynamicMesh).
335 ensureDynamicRing(*gpu);
336 writeDynamicMesh(*gpu, verts, getDevice(), frameToken(), nullptr, 0);
337 mesh->markMorphClean();
338 return true;
339}

341bool Graphics::updateMeshVertices(Mesh *mesh, const float *posXYZ, const float *nrmXYZ,
342 const float *uvST, int vertexCount, const uint32_t *indices,
343 int indexCount) {
344 if (!initialized || !mesh || !mesh->gpuHandle) return false;
345 if (!posXYZ || vertexCount <= 0 || indexCount < 0) return false;
346 if (indexCount > 0 && (indexCount % 3 != 0 || !indices)) return false;
347 for (int i = 0; i < indexCount; ++i) {
348 if (indices[i] >= uint32_t(vertexCount)) return false;
349 }
350
351 std::vector<MeshVertex> verts(static_cast<size_t>(vertexCount));
352 for (int i = 0; i < vertexCount; ++i) {
353 MeshVertex &v = verts[static_cast<size_t>(i)];
354 v.pos = {posXYZ[size_t(i) * 3u], posXYZ[size_t(i) * 3u + 1u], posXYZ[size_t(i) * 3u + 2u]};
355 if (nrmXYZ)
356 v.normal = {nrmXYZ[size_t(i) * 3u], nrmXYZ[size_t(i) * 3u + 1u],
357 nrmXYZ[size_t(i) * 3u + 2u]};
358 else
359 v.normal = {0.f, 1.f, 0.f};
360 if (uvST)
361 v.uv = {uvST[size_t(i) * 2u], uvST[size_t(i) * 2u + 1u]};
362 else
363 v.uv = {0.f, 0.f};
364 }
365
366 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
367 mesh->computeBounds(posXYZ, vertexCount);
368 // Same ring-buffer approach as bakeMeshMorph: never wait on in-flight
369 // frames, just write the next copy.
370 ensureDynamicRing(*gpu);
371 writeDynamicMesh(*gpu, verts, getDevice(), frameToken(), indices, indexCount);
372 mesh->gpuVertexCount = int(gpu->vertexCount);
373 mesh->indexCount = int(gpu->indexCount);
374 return true;
375}

377bool Graphics::setMeshSkinningData(Mesh *mesh, const uint16_t *joints4, const float *weights4,
378 int vertexCount) {
379 if (!initialized || !mesh || !mesh->gpuHandle || !joints4 || !weights4) return false;
380 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
381 if (vertexCount <= 0 || uint32_t(vertexCount) != gpu->vertexCount) return false;
382
383 std::vector<MeshVertex> verts(static_cast<size_t>(vertexCount));
384 auto &source = meshDrawVertices(*gpu);
385 void *mapped = source.map();
386 if (!mapped) return false;
387 std::memcpy(verts.data(), mapped, verts.size() * sizeof(MeshVertex));
388 source.unmap();
389 for (int i = 0; i < vertexCount; ++i) {
390 const size_t base = static_cast<size_t>(i) * 4u;
391 verts[static_cast<size_t>(i)].joints =
392 glm::u16vec4(joints4[base], joints4[base + 1], joints4[base + 2], joints4[base + 3]);
393 verts[static_cast<size_t>(i)].weights =
394 glm::vec4(weights4[base], weights4[base + 1], weights4[base + 2], weights4[base + 3]);
395 }
396 ensureDynamicRing(*gpu);
397 writeDynamicMesh(*gpu, verts, getDevice(), frameToken(), nullptr, 0);
398 mesh->markGpuSkinned(true);
399 return true;
400}

402bool Graphics::releaseMesh(Mesh *mesh) {
403 if (!mesh || !mesh->gpuHandle) return false;
404
405 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
406 auto gpuIt = std::find_if(ownedGpuMeshes.begin(), ownedGpuMeshes.end(),
407 [&](const std::unique_ptr<GpuMesh> &g) {
408 return g.get() == gpu;
409 });
410 if (gpuIt == ownedGpuMeshes.end()) return false;
411
412 auto meshIt = std::find_if(ownedMeshes.begin(), ownedMeshes.end(),
413 [&](const std::unique_ptr<Mesh> &m) {
414 return m.get() == mesh;
415 });
416 if (meshIt == ownedMeshes.end()) return false;
417
418 // An in-flight draw may still read the vertex/index buffers; drain first.
419 waitForSharedGpuResources();
420 mesh->gpuHandle = nullptr;
421 ownedGpuMeshes.erase(gpuIt);
422 // Transfer the CPU facade to the caller instead of destroying it.
423 (void)meshIt->release();
424 ownedMeshes.erase(meshIt);
425 return true;
426}

428Mesh *Graphics::newMeshSphere(int slices, int stacks) {
429 ASSERT(initialized);
430 if (!initialized) throw Exception("newMeshSphere: graphics not initialized");
431 if (slices < 3) slices = 3;
432 if (stacks < 2) stacks = 2;
433 if (slices > 256) slices = 256;
434 if (stacks > 128) stacks = 128;
435
436 constexpr float kPi = 3.14159265358979323846f;
437 constexpr float kTwoPi = kPi * 2.f;
438
439 // Layout (stride = slices+1, seam column duplicated for continuous U):
440 // row 0: north pole verts (same pos, unique U)
441 // row 1..stacks-1: latitude rings
442 // row stacks: south pole verts
443 // One pole vertex per longitude avoids a single-fan UV singularity and
444 // keeps every triangle non-degenerate.
445 const int stride = slices + 1;
446 const int rows = stacks + 1; // includes both pole rows
447 std::vector<MeshVertex> verts;
448 verts.reserve(size_t(stride) * size_t(rows));
449
450 auto pushVert = [&](float px, float py, float pz, float u, float v) {
451 MeshVertex vert{};
452 vert.pos = {px, py, pz};
453 vert.normal = {px, py, pz}; // unit sphere
454 vert.uv = {u, v};
455 verts.push_back(vert);
456 };
457
458 for (int y = 0; y <= stacks; ++y) {
459 const float fv = float(y) / float(stacks);
460 const float phi = fv * kPi;
461 const float sinPhi = std::sin(phi);
462 const float cosPhi = std::cos(phi);
463 for (int x = 0; x <= slices; ++x) {
464 const float u = float(x) / float(slices);
465 const float theta = u * kTwoPi;
466 // Exact poles: collapse ring to a point but keep per-slice UVs.
467 if (y == 0)
468 pushVert(0.f, 1.f, 0.f, u, 0.f);
469 else if (y == stacks)
470 pushVert(0.f, -1.f, 0.f, u, 1.f);
471 else
472 pushVert(sinPhi * std::cos(theta), cosPhi, sinPhi * std::sin(theta), u, fv);
473 }
474 }
475
476 std::vector<uint32_t> indices;
477 indices.reserve(size_t(slices) * size_t(stacks) * 6u);
478
479 // Quads between consecutive rows use outward object-space CCW winding.
480 for (int y = 0; y < stacks; ++y) {
481 const uint32_t row0 = uint32_t(y * stride);
482 const uint32_t row1 = uint32_t((y + 1) * stride);
483 for (int x = 0; x < slices; ++x) {
484 const uint32_t i0 = row0 + uint32_t(x);
485 const uint32_t i1 = row0 + uint32_t(x + 1);
486 const uint32_t i2 = row1 + uint32_t(x);
487 const uint32_t i3 = row1 + uint32_t(x + 1);
488 indices.push_back(i0);
489 indices.push_back(i1);
490 indices.push_back(i2);
491 indices.push_back(i1);
492 indices.push_back(i3);
493 indices.push_back(i2);
494 }
495 }
496
497 auto gpu = uploadGpuMesh(device, frameToken(), verts, indices);
498 auto handle = makeMeshHandle(*gpu);
499 Mesh *raw = handle.get();
500 assignMeshBounds(raw, verts);
501 registerMeshRecord(gpu.get());
502 ownedGpuMeshes.push_back(std::move(gpu));
503 ownedMeshes.push_back(std::move(handle));
504 return raw;

507Mesh *Graphics::newMeshCylinder(int slices, int stacks, bool caps) {
508 ASSERT(initialized);
509 if (!initialized) throw Exception("newMeshCylinder: graphics not initialized");
510 if (slices < 3) slices = 3;
511 if (stacks < 1) stacks = 1;
512 if (slices > 256) slices = 256;
513 if (stacks > 128) stacks = 128;
514
515 constexpr float kPi = 3.14159265358979323846f;
516 constexpr float kTwoPi = kPi * 2.f;
517 constexpr float kRadius = 1.f;
518 constexpr float kHalfH = 1.f; // height 2, Y from -1..1
519
520 const int stride = slices + 1; // duplicated seam for continuous U
521 const int sideRows = stacks + 1;
522 std::vector<MeshVertex> verts;
523 verts.reserve(size_t(stride) * size_t(sideRows) + size_t(caps ? 2 * (slices + 2) : 0));
524
525 auto pushVert = [&](float px, float py, float pz, float nx, float ny, float nz, float u,
526 float v) {
527 MeshVertex vert{};
528 vert.pos = {px, py, pz};
529 vert.normal = {nx, ny, nz};
530 vert.uv = {u, v};
531 verts.push_back(vert);
532 };
533
534 // Side wall: outward normals in XZ.
535 for (int y = 0; y <= stacks; ++y) {
536 const float fv = float(y) / float(stacks);
537 const float py = kHalfH - fv * (2.f * kHalfH);
538 for (int x = 0; x <= slices; ++x) {
539 const float u = float(x) / float(slices);
540 const float theta = u * kTwoPi;
541 const float cx = std::cos(theta);
542 const float sz = std::sin(theta);
543 pushVert(kRadius * cx, py, kRadius * sz, cx, 0.f, sz, u, fv);
544 }
545 }
546
547 std::vector<uint32_t> indices;
548 indices.reserve(size_t(slices) * size_t(stacks) * 6u +
549 size_t(caps ? slices * 2 * 3 : 0));
550
551 for (int y = 0; y < stacks; ++y) {
552 const uint32_t row0 = uint32_t(y * stride);
553 const uint32_t row1 = uint32_t((y + 1) * stride);
554 for (int x = 0; x < slices; ++x) {
555 const uint32_t i0 = row0 + uint32_t(x);
556 const uint32_t i1 = row0 + uint32_t(x + 1);
557 const uint32_t i2 = row1 + uint32_t(x);
558 const uint32_t i3 = row1 + uint32_t(x + 1);
559 indices.push_back(i0);
560 indices.push_back(i1);
561 indices.push_back(i2);
562 indices.push_back(i1);
563 indices.push_back(i3);
564 indices.push_back(i2);
565 }
566 }
567
568 if (caps) {
569 // Top cap (y = +1, normal +Y) — fan from center.
570 const uint32_t topCenter = uint32_t(verts.size());
571 pushVert(0.f, kHalfH, 0.f, 0.f, 1.f, 0.f, 0.5f, 0.5f);
572 const uint32_t topRing = uint32_t(verts.size());
573 for (int x = 0; x <= slices; ++x) {
574 const float u = float(x) / float(slices);
575 const float theta = u * kTwoPi;
576 const float cx = std::cos(theta);
577 const float sz = std::sin(theta);
578 pushVert(kRadius * cx, kHalfH, kRadius * sz, 0.f, 1.f, 0.f, 0.5f + 0.5f * cx,
579 0.5f + 0.5f * sz);
580 }
581 for (int x = 0; x < slices; ++x) {
582 indices.push_back(topCenter);
583 indices.push_back(topRing + uint32_t(x + 1));
584 indices.push_back(topRing + uint32_t(x));
585 }
586
587 // Bottom cap (y = -1, normal -Y).
588 const uint32_t botCenter = uint32_t(verts.size());
589 pushVert(0.f, -kHalfH, 0.f, 0.f, -1.f, 0.f, 0.5f, 0.5f);
590 const uint32_t botRing = uint32_t(verts.size());
591 for (int x = 0; x <= slices; ++x) {
592 const float u = float(x) / float(slices);
593 const float theta = u * kTwoPi;
594 const float cx = std::cos(theta);
595 const float sz = std::sin(theta);
596 pushVert(kRadius * cx, -kHalfH, kRadius * sz, 0.f, -1.f, 0.f, 0.5f + 0.5f * cx,
597 0.5f + 0.5f * sz);
598 }
599 for (int x = 0; x < slices; ++x) {
600 indices.push_back(botCenter);
601 indices.push_back(botRing + uint32_t(x));
602 indices.push_back(botRing + uint32_t(x + 1));
603 }
604 }
605
606 auto gpu = uploadGpuMesh(device, frameToken(), verts, indices);
607 auto handle = makeMeshHandle(*gpu);
608 Mesh *raw = handle.get();
609 assignMeshBounds(raw, verts);
610 registerMeshRecord(gpu.get());
611 ownedGpuMeshes.push_back(std::move(gpu));
612 ownedMeshes.push_back(std::move(handle));
613 return raw;
614}

626 uint32_t first, uint32_t count) {
627 auto fail = [](const char *message) {
628 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, message, "graphics.instances"));
629 };
630 auto ownedShader = std::find_if(ownedGpuShaders.begin(), ownedGpuShaders.end(),
631 [&](const auto &item) { return item->owner == &shader; });
632 auto ownedMesh =

641 for (int col = 0; col < 4; ++col)

643 if (!std::isfinite(model[col][row])) return fail("Nonfinite model matrix");
644 for (int i = 0; i < 4; ++i)
645 if (!std::isfinite(tint[i])) return fail("Nonfinite instance tint");
646 if (!count) return Result<void>::success();
647 try {
648 drawMeshShaderRange(&mesh, model, nullptr, tint, &shader, first, count);
649 return Result<void>::success();
650 } catch (const std::exception &error) {
651 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Failed, error.what(), "graphics.instances"));
