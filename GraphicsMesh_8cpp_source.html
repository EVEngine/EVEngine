51Mesh *Graphics::newMeshFromAssimp(const ::aiMesh &mesh) {
52 ASSERT(initialized);
53 if (!initialized) throw Exception("newMeshFromAssimp: graphics not initialized");
54 if (mesh.mNumVertices == 0 || mesh.mNumFaces == 0)
55 throw Exception("newMeshFromAssimp: empty mesh");
56
57 std::vector<MeshVertex> verts;
58 verts.reserve(mesh.mNumVertices);
59 std::vector<float> basePos;
60 std::vector<float> baseNrm;
61 std::vector<float> baseUv;
62 basePos.reserve(mesh.mNumVertices * 3);
63 baseNrm.reserve(mesh.mNumVertices * 3);
64 baseUv.reserve(mesh.mNumVertices * 2);
65 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
66 MeshVertex v{};
67 v.pos = {mesh.mVertices[i].x, mesh.mVertices[i].y, mesh.mVertices[i].z};
68 if (mesh.HasNormals())
69 v.normal = {mesh.mNormals[i].x, mesh.mNormals[i].y, mesh.mNormals[i].z};
70 else
71 v.normal = {0.f, 1.f, 0.f};
72 if (mesh.HasTextureCoords(0))
73 v.uv = {mesh.mTextureCoords[0][i].x, mesh.mTextureCoords[0][i].y};
74 else
75 v.uv = {0.f, 0.f};
76 verts.push_back(v);
77 basePos.push_back(v.pos.x);
78 basePos.push_back(v.pos.y);
79 basePos.push_back(v.pos.z);
80 baseNrm.push_back(v.normal.x);
81 baseNrm.push_back(v.normal.y);
82 baseNrm.push_back(v.normal.z);
83 baseUv.push_back(v.uv.x);
84 baseUv.push_back(v.uv.y);
85 }
86
87 std::vector<uint32_t> indices;
88 indices.reserve(mesh.mNumFaces * 3);
89 for (unsigned f = 0; f < mesh.mNumFaces; ++f) {
90 const aiFace &face = mesh.mFaces[f];
91 if (face.mNumIndices != 3) continue;
92 indices.push_back(face.mIndices[0]);
93 indices.push_back(face.mIndices[1]);
94 indices.push_back(face.mIndices[2]);
95 }
96 if (indices.empty()) throw Exception("newMeshFromAssimp: no triangle faces");
97
98 std::unique_ptr<GpuMesh> gpu;
99 if (mesh.mNumVertices <= 65535u) {
100 std::vector<uint16_t> idx16;
101 idx16.reserve(indices.size());
102 for (uint32_t i : indices) idx16.push_back(uint16_t(i));
103 gpu = uploadGpuMesh16(device, frameToken(), verts, idx16);
104 } else {
105 gpu = uploadGpuMesh(device, frameToken(), verts, indices);
106 }
107 auto handle = makeMeshHandle(*gpu);
108 handle->captureImportedAttributes(mesh);
109 // Retain the CPU morph base pose only when the mesh actually has morphs;
110 // otherwise the base pos/nrm/uv copies would linger at ~32B/vertex for no reason.
111 if (mesh.mNumAnimMeshes > 0) {
112 handle->initMorphBase(int(mesh.mNumVertices), basePos.data(), baseNrm.data(), baseUv.data());
113 }
114 // Assimp morph targets (VRM / glTF blend shapes often land here).
115 for (unsigned m = 0; m < mesh.mNumAnimMeshes; ++m) {
116 const aiAnimMesh *am = mesh.mAnimMeshes[m];
117 if (!am || !am->mVertices || am->mNumVertices != mesh.mNumVertices) continue;
118 std::string morphName =
119 am->mName.length ? am->mName.C_Str() : ("morph" + std::to_string(m));
120 std::vector<float> absPos(size_t(am->mNumVertices) * 3u);
121 for (unsigned i = 0; i < am->mNumVertices; ++i) {
122 absPos[size_t(i) * 3u + 0] = am->mVertices[i].x;
123 absPos[size_t(i) * 3u + 1] = am->mVertices[i].y;
124 absPos[size_t(i) * 3u + 2] = am->mVertices[i].z;
125 }
126 handle->addMorphTargetAbsolute(morphName, absPos.data());
127 }
128 handle->markMorphClean();
129 Mesh *raw = handle.get();
130 assignMeshBounds(raw, verts);
131 registerMeshRecord(gpu.get());
132 ownedGpuMeshes.push_back(std::move(gpu));
133 ownedMeshes.push_back(std::move(handle));
134 return raw;
135}

137Mesh *Graphics::newMeshFromAssimp(const ::aiMesh &mesh, const aiMatrix4x4 &worldTransform) {
138 ASSERT(initialized);
139 if (!initialized) throw Exception("newMeshFromAssimp: graphics not initialized");
140 if (mesh.mNumVertices == 0 || mesh.mNumFaces == 0)
141 throw Exception("newMeshFromAssimp: empty mesh");
142
143 std::vector<aiVector3D> positions(mesh.mNumVertices);
144 std::vector<aiVector3D> normals(mesh.mNumVertices);
145 aiMatrix3x3 nmat(worldTransform);
146 const float ndet = nmat.Determinant();
147 if (std::fabs(ndet) > 1e-8f) {
148 nmat.Inverse();
149 nmat.Transpose();
150 }
151 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
152 positions[i] = worldTransform * mesh.mVertices[i];
153 if (mesh.HasNormals()) {
154 normals[i] = nmat * mesh.mNormals[i];
155 normals[i].Normalize();
156 } else {
157 normals[i] = aiVector3D(0.f, 1.f, 0.f);
158 }
159 }
160
161 // Negative determinant mirrors the mesh: winding flips while inverse-transpose
162 // keeps normals consistent, so back-face cull would hide the visible side.
163 const bool flipWinding = worldTransform.Determinant() < 0.f;
164 std::vector<aiFace> flippedFaces;
165 std::vector<unsigned> flippedIdx;
166 if (flipWinding) {
167 flippedFaces.resize(mesh.mNumFaces);
168 flippedIdx.resize(size_t(mesh.mNumFaces) * 3u);
169 for (unsigned f = 0; f < mesh.mNumFaces; ++f) {
170 const aiFace &src = mesh.mFaces[f];
171 aiFace &dst = flippedFaces[f];
172 dst.mNumIndices = src.mNumIndices;
173 if (src.mNumIndices == 3 && src.mIndices) {
174 unsigned *idx = flippedIdx.data() + size_t(f) * 3u;
175 idx[0] = src.mIndices[0];
176 idx[1] = src.mIndices[2];
177 idx[2] = src.mIndices[1];
178 dst.mIndices = idx;
179 } else {
180 dst.mIndices = src.mIndices;
181 }
182 }
183 }
184
185 // Non-owning view — do not let aiMesh destructor free borrowed pointers.
186 aiMesh tmp{};
187 tmp.mPrimitiveTypes = mesh.mPrimitiveTypes;
188 tmp.mNumVertices = mesh.mNumVertices;
189 tmp.mVertices = positions.data();
190 tmp.mNormals = normals.data();
191 tmp.mNumFaces = mesh.mNumFaces;
192 tmp.mFaces = flipWinding ? flippedFaces.data() : mesh.mFaces;
193 tmp.mMaterialIndex = mesh.mMaterialIndex;
194 tmp.mNumAnimMeshes = mesh.mNumAnimMeshes;
195 tmp.mAnimMeshes = mesh.mAnimMeshes;
196 if (mesh.HasTextureCoords(0)) {
197 tmp.mTextureCoords[0] = mesh.mTextureCoords[0];
198 tmp.mNumUVComponents[0] = mesh.mNumUVComponents[0];
199 }
200 Mesh *out = newMeshFromAssimp(tmp);
201 tmp.mVertices = nullptr;
202 tmp.mNormals = nullptr;
203 tmp.mFaces = nullptr;
204 tmp.mTextureCoords[0] = nullptr;
205 tmp.mAnimMeshes = nullptr;
206 return out;
207}

209Mesh *Graphics::newMeshFromArrays(const float *posXYZ, const float *nrmXYZ, const float *uvST,
210 int vertexCount, const uint32_t *indices, int indexCount) {
211 ASSERT(initialized);
212 if (!initialized) throw Exception("newMeshFromArrays: graphics not initialized");
213 if (!posXYZ || vertexCount <= 0) throw Exception("newMeshFromArrays: empty positions");
214 if (!indices || indexCount < 3) throw Exception("newMeshFromArrays: empty indices");
215 if (indexCount % 3 != 0) throw Exception("newMeshFromArrays: indexCount must be multiple of 3");
216
217 std::vector<MeshVertex> verts(static_cast<size_t>(vertexCount));
218 for (int i = 0; i < vertexCount; ++i) {
219 MeshVertex &v = verts[static_cast<size_t>(i)];
220 v.pos = {posXYZ[size_t(i) * 3u], posXYZ[size_t(i) * 3u + 1u], posXYZ[size_t(i) * 3u + 2u]};
221 if (nrmXYZ)
222 v.normal = {nrmXYZ[size_t(i) * 3u], nrmXYZ[size_t(i) * 3u + 1u],
223 nrmXYZ[size_t(i) * 3u + 2u]};
224 else
225 v.normal = {0.f, 1.f, 0.f};
226 if (uvST)
227 v.uv = {uvST[size_t(i) * 2u], uvST[size_t(i) * 2u + 1u]};
228 else
229 v.uv = {0.f, 0.f};
230 }
231
232 std::vector<uint32_t> idx(indices, indices + indexCount);
233 for (uint32_t id : idx) {
234 if (int(id) >= vertexCount) throw Exception("newMeshFromArrays: index out of range");
235 }
236
237 auto gpu = uploadGpuMesh(device, frameToken(), verts, idx);
238 auto handle = makeMeshHandle(*gpu);
239 Mesh *raw = handle.get();
240 raw->computeBounds(posXYZ, vertexCount);
241 registerMeshRecord(gpu.get());
242 ownedGpuMeshes.push_back(std::move(gpu));
243 ownedMeshes.push_back(std::move(handle));
244 return raw;
245}

259bool Graphics::bakeMeshMorph(Mesh *mesh) {
260 if (!mesh || !mesh->gpuHandle || !mesh->hasMorphData() || !mesh->isMorphDirty()) return false;
261 if (!initialized) return false;
262
263 std::vector<float> pos;
264 std::vector<float> nrm;
265 mesh->computeMorphedPositions(pos, nrm);
266 const int vc = mesh->getVertexCount();
267 if (vc <= 0 || int(pos.size()) < vc * 3) return false;
268 mesh->computeBounds(pos.data(), vc);
269
270 std::vector<MeshVertex> verts(static_cast<size_t>(vc));
271 const auto &uv = mesh->baseUv();
272 for (int i = 0; i < vc; ++i) {
273 MeshVertex &v = verts[static_cast<size_t>(i)];
274 v.pos = {pos[size_t(i) * 3u + 0], pos[size_t(i) * 3u + 1], pos[size_t(i) * 3u + 2]};
275 if (int(nrm.size()) >= (i + 1) * 3)
276 v.normal = {nrm[size_t(i) * 3u + 0], nrm[size_t(i) * 3u + 1], nrm[size_t(i) * 3u + 2]};
277 else
278 v.normal = {0.f, 1.f, 0.f};
279 if (int(uv.size()) >= (i + 1) * 2)
280 v.uv = {uv[size_t(i) * 2u + 0], uv[size_t(i) * 2u + 1]};
281 else
282 v.uv = {0.f, 0.f};
283 }
284
285 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
286 // Ring-buffered host-visible VBO: the next copy is kDynamicVertexCopies
287 // frames old, so overwriting it never races with in-flight draws — no
288 // device-wide wait (see writeDynamicMesh).
289 ensureDynamicRing(*gpu);
290 writeDynamicMesh(*gpu, verts, getDevice(), frameToken(), nullptr, 0);
291 mesh->markMorphClean();
292 return true;
293}

295bool Graphics::updateMeshVertices(Mesh *mesh, const float *posXYZ, const float *nrmXYZ,
296 const float *uvST, int vertexCount, const uint32_t *indices,
297 int indexCount) {
298 if (!initialized || !mesh || !mesh->gpuHandle) return false;
299 if (!posXYZ || vertexCount <= 0 || indexCount < 0) return false;
300 if (indexCount > 0 && (indexCount % 3 != 0 || !indices)) return false;
301 for (int i = 0; i < indexCount; ++i) {
302 if (indices[i] >= uint32_t(vertexCount)) return false;
303 }
304
305 std::vector<MeshVertex> verts(static_cast<size_t>(vertexCount));
306 for (int i = 0; i < vertexCount; ++i) {
307 MeshVertex &v = verts[static_cast<size_t>(i)];
308 v.pos = {posXYZ[size_t(i) * 3u], posXYZ[size_t(i) * 3u + 1u], posXYZ[size_t(i) * 3u + 2u]};
309 if (nrmXYZ)
310 v.normal = {nrmXYZ[size_t(i) * 3u], nrmXYZ[size_t(i) * 3u + 1u],
311 nrmXYZ[size_t(i) * 3u + 2u]};
312 else
313 v.normal = {0.f, 1.f, 0.f};
314 if (uvST)
315 v.uv = {uvST[size_t(i) * 2u], uvST[size_t(i) * 2u + 1u]};
316 else
317 v.uv = {0.f, 0.f};
318 }
319
320 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
321 mesh->computeBounds(posXYZ, vertexCount);
322 // Same ring-buffer approach as bakeMeshMorph: never wait on in-flight
323 // frames, just write the next copy.
324 ensureDynamicRing(*gpu);
325 writeDynamicMesh(*gpu, verts, getDevice(), frameToken(), indices, indexCount);
326 mesh->gpuVertexCount = int(gpu->vertexCount);
327 mesh->indexCount = int(gpu->indexCount);
328 return true;
329}

331bool Graphics::setMeshSkinningData(Mesh *mesh, const uint16_t *joints4, const float *weights4,
332 int vertexCount) {
333 if (!initialized || !mesh || !mesh->gpuHandle || !joints4 || !weights4) return false;
334 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
335 if (vertexCount <= 0 || uint32_t(vertexCount) != gpu->vertexCount) return false;
336
337 std::vector<MeshVertex> verts(static_cast<size_t>(vertexCount));
338 auto &source = meshDrawVertices(*gpu);
339 void *mapped = source.map();
340 if (!mapped) return false;
341 std::memcpy(verts.data(), mapped, verts.size() * sizeof(MeshVertex));
342 source.unmap();
343 for (int i = 0; i < vertexCount; ++i) {
344 const size_t base = static_cast<size_t>(i) * 4u;
345 verts[static_cast<size_t>(i)].joints =
346 glm::u16vec4(joints4[base], joints4[base + 1], joints4[base + 2], joints4[base + 3]);
347 verts[static_cast<size_t>(i)].weights =
348 glm::vec4(weights4[base], weights4[base + 1], weights4[base + 2], weights4[base + 3]);
349 }
350 ensureDynamicRing(*gpu);
351 writeDynamicMesh(*gpu, verts, getDevice(), frameToken(), nullptr, 0);
352 mesh->markGpuSkinned(true);
353 return true;
354}

356bool Graphics::releaseMesh(Mesh *mesh) {
357 if (!mesh || !mesh->gpuHandle) return false;
358
359 auto *gpu = static_cast<GpuMesh *>(mesh->gpuHandle);
360 auto gpuIt = std::find_if(ownedGpuMeshes.begin(), ownedGpuMeshes.end(),
361 [&](const std::unique_ptr<GpuMesh> &g) {
362 return g.get() == gpu;
363 });
364 if (gpuIt == ownedGpuMeshes.end()) return false;
365
366 auto meshIt = std::find_if(ownedMeshes.begin(), ownedMeshes.end(),
367 [&](const std::unique_ptr<Mesh> &m) {
368 return m.get() == mesh;
369 });
370 if (meshIt == ownedMeshes.end()) return false;
371
372 // An in-flight draw may still read the vertex/index buffers; drain first.
373 waitForSharedGpuResources();
374 mesh->gpuHandle = nullptr;
375 ownedGpuMeshes.erase(gpuIt);
376 // Transfer the CPU facade to the caller instead of destroying it.
377 (void)meshIt->release();
378 ownedMeshes.erase(meshIt);
379 return true;
380}

382Mesh *Graphics::newMeshSphere(int slices, int stacks) {
383 ASSERT(initialized);
384 if (!initialized) throw Exception("newMeshSphere: graphics not initialized");
385 if (slices < 3) slices = 3;
386 if (stacks < 2) stacks = 2;
387 if (slices > 256) slices = 256;
388 if (stacks > 128) stacks = 128;
389
390 constexpr float kPi = 3.14159265358979323846f;
391 constexpr float kTwoPi = kPi * 2.f;
392
393 // Layout (stride = slices+1, seam column duplicated for continuous U):
394 // row 0: north pole verts (same pos, unique U)
395 // row 1..stacks-1: latitude rings
396 // row stacks: south pole verts
397 // One pole vertex per longitude avoids a single-fan UV singularity and
398 // keeps every triangle non-degenerate.
399 const int stride = slices + 1;
400 const int rows = stacks + 1; // includes both pole rows
401 std::vector<MeshVertex> verts;
402 verts.reserve(size_t(stride) * size_t(rows));
403
404 auto pushVert = [&](float px, float py, float pz, float u, float v) {
405 MeshVertex vert{};
406 vert.pos = {px, py, pz};
407 vert.normal = {px, py, pz}; // unit sphere
408 vert.uv = {u, v};
409 verts.push_back(vert);
410 };
411
412 for (int y = 0; y <= stacks; ++y) {
413 const float fv = float(y) / float(stacks);
414 const float phi = fv * kPi;
415 const float sinPhi = std::sin(phi);
416 const float cosPhi = std::cos(phi);
417 for (int x = 0; x <= slices; ++x) {
418 const float u = float(x) / float(slices);
419 const float theta = u * kTwoPi;
420 // Exact poles: collapse ring to a point but keep per-slice UVs.
421 if (y == 0)
422 pushVert(0.f, 1.f, 0.f, u, 0.f);
423 else if (y == stacks)
424 pushVert(0.f, -1.f, 0.f, u, 1.f);
425 else
426 pushVert(sinPhi * std::cos(theta), cosPhi, sinPhi * std::sin(theta), u, fv);
427 }
428 }
429
430 std::vector<uint32_t> indices;
431 indices.reserve(size_t(slices) * size_t(stacks) * 6u);
432
433 // Quads between consecutive rows use outward object-space CCW winding.
434 for (int y = 0; y < stacks; ++y) {
435 const uint32_t row0 = uint32_t(y * stride);
436 const uint32_t row1 = uint32_t((y + 1) * stride);
437 for (int x = 0; x < slices; ++x) {
438 const uint32_t i0 = row0 + uint32_t(x);
439 const uint32_t i1 = row0 + uint32_t(x + 1);
440 const uint32_t i2 = row1 + uint32_t(x);
441 const uint32_t i3 = row1 + uint32_t(x + 1);
442 indices.push_back(i0);
443 indices.push_back(i1);
444 indices.push_back(i2);
445 indices.push_back(i1);
446 indices.push_back(i3);
447 indices.push_back(i2);
448 }
449 }
450
451 auto gpu = uploadGpuMesh(device, frameToken(), verts, indices);
452 auto handle = makeMeshHandle(*gpu);
453 Mesh *raw = handle.get();
454 assignMeshBounds(raw, verts);
455 registerMeshRecord(gpu.get());
456 ownedGpuMeshes.push_back(std::move(gpu));
457 ownedMeshes.push_back(std::move(handle));
458 return raw;
459}

461Mesh *Graphics::newMeshCylinder(int slices, int stacks, bool caps) {
462 ASSERT(initialized);
463 if (!initialized) throw Exception("newMeshCylinder: graphics not initialized");
464 if (slices < 3) slices = 3;
465 if (stacks < 1) stacks = 1;
466 if (slices > 256) slices = 256;
467 if (stacks > 128) stacks = 128;
468
469 constexpr float kPi = 3.14159265358979323846f;
470 constexpr float kTwoPi = kPi * 2.f;
471 constexpr float kRadius = 1.f;
472 constexpr float kHalfH = 1.f; // height 2, Y from -1..1
473
474 const int stride = slices + 1; // duplicated seam for continuous U
475 const int sideRows = stacks + 1;
476 std::vector<MeshVertex> verts;
477 verts.reserve(size_t(stride) * size_t(sideRows) + size_t(caps ? 2 * (slices + 2) : 0));
478
479 auto pushVert = [&](float px, float py, float pz, float nx, float ny, float nz, float u,
480 float v) {
481 MeshVertex vert{};
482 vert.pos = {px, py, pz};
483 vert.normal = {nx, ny, nz};
484 vert.uv = {u, v};
485 verts.push_back(vert);
486 };
487
488 // Side wall: outward normals in XZ.
489 for (int y = 0; y <= stacks; ++y) {
490 const float fv = float(y) / float(stacks);
491 const float py = kHalfH - fv * (2.f * kHalfH);
492 for (int x = 0; x <= slices; ++x) {
493 const float u = float(x) / float(slices);
494 const float theta = u * kTwoPi;
495 const float cx = std::cos(theta);
496 const float sz = std::sin(theta);
497 pushVert(kRadius * cx, py, kRadius * sz, cx, 0.f, sz, u, fv);
498 }
499 }
500
501 std::vector<uint32_t> indices;
502 indices.reserve(size_t(slices) * size_t(stacks) * 6u +
503 size_t(caps ? slices * 2 * 3 : 0));
504
505 for (int y = 0; y < stacks; ++y) {
506 const uint32_t row0 = uint32_t(y * stride);
507 const uint32_t row1 = uint32_t((y + 1) * stride);
508 for (int x = 0; x < slices; ++x) {
509 const uint32_t i0 = row0 + uint32_t(x);
510 const uint32_t i1 = row0 + uint32_t(x + 1);
511 const uint32_t i2 = row1 + uint32_t(x);
512 const uint32_t i3 = row1 + uint32_t(x + 1);
513 indices.push_back(i0);
514 indices.push_back(i1);
515 indices.push_back(i2);
516 indices.push_back(i1);
517 indices.push_back(i3);
518 indices.push_back(i2);
519 }
520 }
521
522 if (caps) {
523 // Top cap (y = +1, normal +Y) — fan from center.
524 const uint32_t topCenter = uint32_t(verts.size());
525 pushVert(0.f, kHalfH, 0.f, 0.f, 1.f, 0.f, 0.5f, 0.5f);
526 const uint32_t topRing = uint32_t(verts.size());
527 for (int x = 0; x <= slices; ++x) {
528 const float u = float(x) / float(slices);
529 const float theta = u * kTwoPi;
530 const float cx = std::cos(theta);
531 const float sz = std::sin(theta);
532 pushVert(kRadius * cx, kHalfH, kRadius * sz, 0.f, 1.f, 0.f, 0.5f + 0.5f * cx,
533 0.5f + 0.5f * sz);
534 }
535 for (int x = 0; x < slices; ++x) {
536 indices.push_back(topCenter);
537 indices.push_back(topRing + uint32_t(x + 1));
538 indices.push_back(topRing + uint32_t(x));
539 }
540
541 // Bottom cap (y = -1, normal -Y).
542 const uint32_t botCenter = uint32_t(verts.size());
543 pushVert(0.f, -kHalfH, 0.f, 0.f, -1.f, 0.f, 0.5f, 0.5f);
544 const uint32_t botRing = uint32_t(verts.size());
545 for (int x = 0; x <= slices; ++x) {
546 const float u = float(x) / float(slices);
547 const float theta = u * kTwoPi;
548 const float cx = std::cos(theta);
549 const float sz = std::sin(theta);
550 pushVert(kRadius * cx, -kHalfH, kRadius * sz, 0.f, -1.f, 0.f, 0.5f + 0.5f * cx,
551 0.5f + 0.5f * sz);
552 }
553 for (int x = 0; x < slices; ++x) {
554 indices.push_back(botCenter);
555 indices.push_back(botRing + uint32_t(x));
556 indices.push_back(botRing + uint32_t(x + 1));
557 }
558 }
559
560 auto gpu = uploadGpuMesh(device, frameToken(), verts, indices);
561 auto handle = makeMeshHandle(*gpu);
562 Mesh *raw = handle.get();
563 assignMeshBounds(raw, verts);
564 registerMeshRecord(gpu.get());
565 ownedGpuMeshes.push_back(std::move(gpu));
566 ownedMeshes.push_back(std::move(handle));
567 return raw;
568}
