18std::vector<RoadPrefabVertex> clip(const std::vector<RoadPrefabVertex>& polygon, float plane, bool above) {
19 std::vector<RoadPrefabVertex> out;
20 if (polygon.empty()) return out;
21 auto previous = polygon.back();
22 bool wasInside = above ? previous.z >= plane : previous.z <= plane;
23 for (const auto& current : polygon) {
24 const bool inside = above ? current.z >= plane : current.z <= plane;
25 if (inside != wasInside) out.push_back(mix(previous, current, (plane - previous.z) / (current.z - previous.z)));
26 if (inside) out.push_back(current);
27 previous = current;
28 wasInside = inside;
29 }
30 return out;
