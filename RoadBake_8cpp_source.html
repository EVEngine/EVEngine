19V3 operator+(V3 a, V3 b) { return {a.x + b.x, a.y + b.y, a.z + b.z}; }
20V3 operator-(V3 a, V3 b) { return {a.x - b.x, a.y - b.y, a.z - b.z}; }
21V3 operator*(V3 a, float s) { return {a.x * s, a.y * s, a.z * s}; }
22float dot(V3 a, V3 b) { return a.x * b.x + a.y * b.y + a.z * b.z; }
23V3 cross(V3 a, V3 b) {
24 return {a.y * b.z - a.z * b.y, a.z * b.x - a.x * b.z, a.x * b.y - a.y * b.x};
25}
26float length(V3 a) { return std::sqrt(dot(a, a)); }
27V3 normalize(V3 a) {
28 const float l = length(a);
29 return l > 1e-8f ? a * (1.f / l) : V3{0.f, 1.f, 0.f};

32Result<SplinePath> edgeToSpline(const RoadEdge& edge) {
33 SplinePath path;
34 auto kind = path.setKindResult("catmullRom");
35 if (!kind.ok()) return Result<SplinePath>::failure(kind.status());
36 path.setClosed(false);
37 for (const auto& p : edge.controlPoints) {
38 SplinePoint sp;
39 sp.x = p.x;
40 sp.y = p.y;
41 sp.z = p.z;
42 auto added = path.addPointResult(sp);
43 if (!added.ok()) return Result<SplinePath>::failure(added.status());
44 }
45 return Result<SplinePath>::success(std::move(path));
46}

53void appendBox(MeshBuild& mesh, V3 center, V3 side, V3 up, V3 forward, float hx, float hy, float hz,
54 RoadMaterial material) {
55 mesh.setActiveGroup(roadMaterialGroup(material));
56 const V3 corners[8] = {
57 center + side * -hx + up * -hy + forward * -hz, center + side * hx + up * -hy + forward * -hz,
58 center + side * hx + up * hy + forward * -hz, center + side * -hx + up * hy + forward * -hz,
59 center + side * -hx + up * -hy + forward * hz, center + side * hx + up * -hy + forward * hz,
60 center + side * hx + up * hy + forward * hz, center + side * -hx + up * hy + forward * hz,
61 };
62 const int faces[6][4] = {{0, 1, 2, 3}, {4, 7, 6, 5}, {0, 4, 5, 1}, {3, 2, 6, 7}, {0, 3, 7, 4}, {1, 5, 6, 2}};
63 const V3 normals[6] = {forward * -1.f, forward, up * -1.f, up, side * -1.f, side};
64 for (int f = 0; f < 6; ++f) {
65 const auto base = static_cast<std::uint32_t>(mesh.getVertexCount());
66 for (int c = 0; c < 4; ++c) {
67 const V3& p = corners[faces[f][c]];
68 mesh.addVertex(p.x, p.y, p.z, normals[f].x, normals[f].y, normals[f].z, static_cast<float>(c & 1),
69 static_cast<float>((c >> 1) & 1));
70 }
71 mesh.addTriangle(base, base + 1, base + 2);
72 mesh.addTriangle(base, base + 2, base + 3);
73 }
74}

112Result<void> loftProfileClean(MeshBuild& mesh, const std::vector<SplineFrameSample>& frames,
113 const RoadProfile& profile, float uvMeters) {
114 if (frames.size() < 2 || profile.points.size() < 2)
115 return Result<void>::failure(Diagnostic::error(
116 DiagnosticCode::InvalidArgument, "loft needs >=2 frames and profile points", {}, {}, "procgen.road"));
117
118 const int ringCount = static_cast<int>(frames.size());
119 const int profileCount = static_cast<int>(profile.points.size());
120 std::vector<V3> ringPositions(static_cast<std::size_t>(ringCount * profileCount));
121
122 for (int ring = 0; ring < ringCount; ++ring) {
123 const auto& frame = frames[static_cast<std::size_t>(ring)];
124 const V3 origin{frame.sample.x, frame.sample.y, frame.sample.z};
125 const V3 side{frame.sideX, frame.sideY, frame.sideZ};
126 const V3 up{frame.upX, frame.upY, frame.upZ};
127 for (int i = 0; i < profileCount; ++i) {
128 const auto& pp = profile.points[static_cast<std::size_t>(i)];
129 ringPositions[static_cast<std::size_t>(ring * profileCount + i)] = origin + side * pp.side + up * pp.up;
130 }
131 }
132
133 for (int ring = 0; ring < ringCount - 1; ++ring) {
134 const float u0 = static_cast<float>(ring) / static_cast<float>(ringCount - 1) *
135 (uvMeters > 0.f ? uvMeters : 1.f);
136 const float u1 = static_cast<float>(ring + 1) / static_cast<float>(ringCount - 1) *
137 (uvMeters > 0.f ? uvMeters : 1.f);
138 for (int i = 0; i < profileCount - 1; ++i) {
139 const auto& a = profile.points[static_cast<std::size_t>(i)];
140 const auto& b = profile.points[static_cast<std::size_t>(i + 1)];
141 // Prefer asphalt for near-horizontal driving strips (endpoint material alone
142 // would tag the asphalt span as curb because the left corner is a curb drop).
143 RoadMaterial mat = a.material;
144 if (std::fabs(a.up - b.up) <= 1e-3f &&
145 (a.material == RoadMaterial::Asphalt || b.material == RoadMaterial::Asphalt)) {
146 mat = RoadMaterial::Asphalt;
147 }
148 const V3& p00 = ringPositions[static_cast<std::size_t>(ring * profileCount + i)];
149 const V3& p01 = ringPositions[static_cast<std::size_t>(ring * profileCount + i + 1)];
150 const V3& p10 = ringPositions[static_cast<std::size_t>((ring + 1) * profileCount + i)];
151 const V3& p11 = ringPositions[static_cast<std::size_t>((ring + 1) * profileCount + i + 1)];
152 const V3 normal = normalize(cross(p10 - p00, p01 - p00));
153 const float v0 =
154 profile.halfWidth > 1e-5f
155 ? (a.side / profile.halfWidth) * 0.5f + 0.5f
156 : 0.f;
157 const float v1 =
158 profile.halfWidth > 1e-5f
159 ? (b.side / profile.halfWidth) * 0.5f + 0.5f
160 : 1.f;
161 appendStripQuad(mesh, p00, p10, p11, p01, normal, u0, u1, v0, v1, mat);
162 }
163 }
164 return Result<void>::success();
165}

168void capProfileRing(MeshBuild& mesh, const SplineFrameSample& frame, const RoadProfile& profile, bool outward) {
169 if (profile.points.size() < 3) return;
170 const V3 origin{frame.sample.x, frame.sample.y, frame.sample.z};
171 const V3 side{frame.sideX, frame.sideY, frame.sideZ};
172 const V3 up{frame.upX, frame.upY, frame.upZ};
173 V3 fwd{frame.forwardX, frame.forwardY, frame.forwardZ};
174 if (!outward) fwd = fwd * -1.f;
175 const V3 nrm = normalize(fwd);
176 std::vector<V3> ring;
177 ring.reserve(profile.points.size());

183 for (std::size_t i = 1; i + 1 < ring.size(); ++i) {

185 else

197void capProfileShoulders(MeshBuild& mesh, const SplineFrameSample& frame, const RoadProfile& profile, bool outward) {
198 if (profile.points.size() < 6) return;
199 const V3 origin{frame.sample.x, frame.sample.y, frame.sample.z};
200 const V3 side{frame.sideX, frame.sideY, frame.sideZ};
201 const V3 up{frame.upX, frame.upY, frame.upZ};
202 V3 fwd{frame.forwardX, frame.forwardY, frame.forwardZ};
203 if (!outward) fwd = fwd * -1.f;
204 const V3 nrm = normalize(fwd);
205
206 // First Asphalt point marks the deck span start; shoulders are everything else.
207 std::size_t asphaltBegin = profile.points.size();
208 for (std::size_t i = 0; i < profile.points.size(); ++i) {
209 if (profile.points[i].material == RoadMaterial::Asphalt) {
210 asphaltBegin = i;
211 break;
212 }
213 }
214 if (asphaltBegin == 0 || asphaltBegin >= profile.points.size()) return;
215
216 auto emitFan = [&](std::size_t begin, std::size_t end, RoadMaterial mat) {
217 if (end < begin + 3) return;
218 mesh.setActiveGroup(roadMaterialGroup(mat));
219 const auto base = static_cast<std::uint32_t>(mesh.getVertexCount());
220 for (std::size_t i = begin; i < end; ++i) {
221 const auto& pp = profile.points[i];
222 const V3 p = origin + side * pp.side + up * pp.up;
223 mesh.addVertex(p.x, p.y, p.z, nrm.x, nrm.y, nrm.z, 0.f, 0.f);
224 }
225 const auto count = static_cast<std::uint32_t>(end - begin);
226 for (std::uint32_t i = 1; i + 1 < count; ++i) {
227 if (outward) mesh.addTriangle(base, base + i, base + i + 1);
228 else
229 mesh.addTriangle(base, base + i + 1, base + i);
230 }
231 };
232
233 // Left shoulder: outer sidewalk → curb down to asphalt lip (excludes asphalt point).
234 emitFan(0, asphaltBegin, RoadMaterial::Sidewalk);
235 // Right shoulder: asphalt lip → curb → outer sidewalk.
236 emitFan(asphaltBegin, profile.points.size(), RoadMaterial::Sidewalk);
237}

239Result<void> addDeckAndPiers(MeshBuild& mesh, const std::vector<SplineFrameSample>& frames, const RoadStyle& style,
240 float halfWidth, bool includePiers) {
241 if (frames.size() < 2) return Result<void>::success();
242 // Deck underside strip.
243 for (int ring = 0; ring < static_cast<int>(frames.size()) - 1; ++ring) {
244 const auto& f0 = frames[static_cast<std::size_t>(ring)];
245 const auto& f1 = frames[static_cast<std::size_t>(ring + 1)];
246 const V3 o0{f0.sample.x, f0.sample.y, f0.sample.z};
247 const V3 o1{f1.sample.x, f1.sample.y, f1.sample.z};
248 const V3 s0{f0.sideX, f0.sideY, f0.sideZ};
249 const V3 s1{f1.sideX, f1.sideY, f1.sideZ};
250 const V3 u0{f0.upX, f0.upY, f0.upZ};
251 const V3 u1{f1.upX, f1.upY, f1.upZ};
252 const V3 a = o0 + s0 * -halfWidth + u0 * -style.deckThickness;
253 const V3 b = o0 + s0 * halfWidth + u0 * -style.deckThickness;
254 const V3 c = o1 + s1 * halfWidth + u1 * -style.deckThickness;
255 const V3 d = o1 + s1 * -halfWidth + u1 * -style.deckThickness;
256 appendStripQuad(mesh, a, b, c, d, normalize((u0 + u1) * -0.5f), 0.f, 1.f, 0.f, 1.f, RoadMaterial::Deck);
257 }
258
259 if (!includePiers || style.pierSpacing <= 1e-3f) return Result<void>::success();
260
261 float traveled = 0.f;
262 float nextPier = style.pierSpacing * 0.5f;
263 for (std::size_t i = 1; i < frames.size(); ++i) {
264 const auto& a = frames[i - 1];
265 const auto& b = frames[i];
266 const V3 pa{a.sample.x, a.sample.y, a.sample.z};
267 const V3 pb{b.sample.x, b.sample.y, b.sample.z};
268 const float seg = length(pb - pa);
269 if (seg <= 1e-5f) continue;
270 while (nextPier <= traveled + seg + 1e-4f) {
271 const float t = std::clamp((nextPier - traveled) / seg, 0.f, 1.f);
272 const V3 pos = pa + (pb - pa) * t;
273 if (pos.y > style.pierClearance) {
274 // World-up pier: never reuse Frenet side (can flip / tilt and skew the box).
275 V3 fwdFlat{pb.x - pa.x, 0.f, pb.z - pa.z};
276 const float fwdLen = length(fwdFlat);
277 fwdFlat = fwdLen > 1e-5f ? fwdFlat * (1.f / fwdLen) : V3{1.f, 0.f, 0.f};
278 const V3 up{0.f, 1.f, 0.f};
279 const V3 side = normalize(cross(up, fwdFlat));
280 const float pierH = pos.y - style.deckThickness;
281 if (pierH > 0.2f) {
282 const V3 center{pos.x, pierH * 0.5f, pos.z};
283 appendBox(mesh, center, side, up, fwdFlat, style.pierWidth * 0.5f, pierH * 0.5f,
284 style.pierDepth * 0.5f, RoadMaterial::Pier);
285 }
286 }
287 nextPier += style.pierSpacing;
288 if (nextPier > 1.0e7f) break;
289 }
290 traveled += seg;
291 }
292 return Result<void>::success();
293}

295Result<void> addLaneMarkings(MeshBuild& mesh, const std::vector<SplineFrameSample>& frames, const RoadEdge& edge) {
296 if (frames.size() < 2) return Result<void>::success();
297 const auto& style = edge.style;
298 const float asphaltHalf =
299 0.5f * style.laneWidth * static_cast<float>(edge.lanesForward + edge.lanesBackward);
300 auto paintLine = [&](float lateral, bool dashed, RoadMaterial material) {
301 float traveled = 0.f;
302 for (std::size_t i = 1; i < frames.size(); ++i) {
303 const auto& a = frames[i - 1];
304 const auto& b = frames[i];
305 const V3 pa{a.sample.x, a.sample.y, a.sample.z};
306 const V3 pb{b.sample.x, b.sample.y, b.sample.z};
307 const V3 sa{a.sideX, a.sideY, a.sideZ};
308 const V3 sb{b.sideX, b.sideY, b.sideZ};
309 const V3 ua{a.upX, a.upY, a.upZ};
310 const V3 ub{b.upX, b.upY, b.upZ};
311 const float seg = length(pb - pa);
312 const bool draw =
313 !dashed || (std::fmod(traveled, style.dashLength + style.dashGap) < style.dashLength);
314 if (draw && seg > 1e-5f) {
315 const float hw = style.markingWidth * 0.5f;
316 const V3 a0 = pa + sa * (lateral - hw) + ua * 0.035f;
317 const V3 a1 = pa + sa * (lateral + hw) + ua * 0.035f;
318 const V3 b1 = pb + sb * (lateral + hw) + ub * 0.035f;
319 const V3 b0 = pb + sb * (lateral - hw) + ub * 0.035f;
320 appendStripQuad(mesh, a0, b0, b1, a1, normalize(ua + ub), 0.f, 1.f, 0.f, 1.f, material);
321 }
322 traveled += seg;
323 }
324 };
325
326 // White edge lines.
327 paintLine(-asphaltHalf + style.markingWidth, false, RoadMaterial::Marking);
328 paintLine(asphaltHalf - style.markingWidth, false, RoadMaterial::Marking);
329 // Yellow dashed lane dividers (reference centerline look).
330 for (int lane = 1; lane < edge.lanesForward; ++lane) {
331 const float lateral = -asphaltHalf + style.laneWidth * static_cast<float>(lane);
332 paintLine(lateral, true, RoadMaterial::MarkingYellow);
333 }
334 return Result<void>::success();
335}

358Result<void> bakeEdgeGeometry(MeshBuild& mesh, RoadOverlay& overlay, const RoadNetwork& network, const RoadEdge& edge,
359 const RoadBakeOptions& options) {
360 auto spline = edgeToSpline(edge);
361 if (!spline.ok()) return Result<void>::failure(spline.status());
362 auto pathLength = spline.value().lengthResult(24);
363 if (!pathLength.ok()) return Result<void>::failure(pathLength.status());
364
365 auto fromNode = network.nodeResult(edge.from);
366 auto toNode = network.nodeResult(edge.to);
367 if (!fromNode.ok() || !toNode.ok())
368 return Result<void>::failure(
369 Diagnostic::error(DiagnosticCode::NotFound, "edge endpoints missing during bake", {}, {}, "procgen.road"));
370
371 // Honour full junctionRadius so filleted corners meet the arm tips with G1
372 // tangents (square stubs from early trim read as sharp turn corners).
373 const float maxTrim = std::max(0.f, pathLength.value() * 0.5f - 0.5f);
374 // Honour full junctionRadius so filleted corners meet the arm tips with G1
375 // tangents (square stubs from early trim read as sharp turn corners).
376 const float trimStart = std::min(std::max(0.f, fromNode.value().junctionRadius), maxTrim);
377 const float trimEnd = std::min(std::max(0.f, toNode.value().junctionRadius), maxTrim);
378 if (trimStart + trimEnd >= pathLength.value() - 0.5f)
379 return Result<void>::success(); // fully inside junction; skip strip
380
381 const int segments = std::max(4, options.pathSegmentsPerEdge);
382 auto frames = spline.value().sampleFramesResult(segments, true, 0.f, 24);
383 if (!frames.ok()) return Result<void>::failure(frames.status());
384
385 // Keep mid frames, then force exact trim endpoints so the loft meets the
386 // junction apron (uniform samples alone leave ~segment-sized salmon gaps).
387 std::vector<SplineFrameSample> trimmed;
388 trimmed.reserve(frames.value().size() + 2);
389 if (trimStart > 1e-3f) {
390 auto tip = spline.value().travelFrameResult(trimStart, "clamp", 24);
391 if (!tip.ok()) return Result<void>::failure(tip.status());
392 trimmed.push_back(std::move(tip).takeValue());
393 }
394 const float endDist = pathLength.value() - trimEnd;
395 for (const auto& frame : frames.value()) {
396 const float d = frame.sample.normalizedDistance * pathLength.value();
397 if (d > trimStart + 1e-3f && d < endDist - 1e-3f) trimmed.push_back(frame);
398 }
399 if (trimEnd > 1e-3f) {
400 auto tip = spline.value().travelFrameResult(endDist, "clamp", 24);
401 if (!tip.ok()) return Result<void>::failure(tip.status());
402 trimmed.push_back(std::move(tip).takeValue());
403 }
404 if (trimmed.size() < 2) return Result<void>::success();
405
406 auto profile = makeRoadProfile(edge.style, edge.lanesForward, edge.lanesBackward);
407 if (!profile.ok()) return Result<void>::failure(profile.status());
408
409 auto lofted = loftProfileClean(mesh, trimmed, profile.value(), edge.style.uvMeters);
410 if (!lofted.ok()) return lofted;
411 // Cap stub ends fully. On large junction trims, only close curb/sidewalk
412 // shoulders — a full U-cap draws a dark asphalt tip bar that reads as a gap.
413 if (trimStart > 0.05f) {
414 if (trimStart < 2.5f) capProfileRing(mesh, trimmed.front(), profile.value(), false);
415 else
416 capProfileShoulders(mesh, trimmed.front(), profile.value(), false);
417 }
418 if (trimEnd > 0.05f) {
419 if (trimEnd < 2.5f) capProfileRing(mesh, trimmed.back(), profile.value(), true);
420 else
421 capProfileShoulders(mesh, trimmed.back(), profile.value(), true);
422 }
423 auto deck = addDeckAndPiers(mesh, trimmed, edge.style, profile.value().halfWidth, options.includePiers);
424 if (!deck.ok()) return deck;
425 if (options.includeMarkings) {
426 auto marks = addLaneMarkings(mesh, trimmed, edge);
427 if (!marks.ok()) return marks;
428 }
429
430 if (options.includeNavigation) {
431 for (int lane = 0; lane < edge.lanesForward; ++lane) {
432 const float lateral = laneCenterOffset(edge.style, edge.lanesForward, lane);
433 RoadPolyline poly;
434 poly.r = 0.15f;
435 poly.g = 0.9f;
436 poly.b = 1.f;
437 poly.width = options.navRibbonHalfWidth * 2.f;
438 float traveled = 0.f;
439 float nextArrow = options.arrowSpacing * 0.5f;
440 V3 prev{};
441 bool hasPrev = false;
442 for (const auto& frame : trimmed) {
443 const V3 origin{frame.sample.x, frame.sample.y, frame.sample.z};
444 const V3 side{frame.sideX, frame.sideY, frame.sideZ};
445 const V3 up{frame.upX, frame.upY, frame.upZ};
446 if (!std::isfinite(origin.x) || !std::isfinite(origin.y) || !std::isfinite(origin.z)) continue;
447 const V3 pos = origin + side * lateral + up * 0.08f;
448 poly.xyz.push_back(pos.x);
449 poly.xyz.push_back(pos.y);
450 poly.xyz.push_back(pos.z);
451 if (hasPrev) {
452 const float step = length(pos - prev);
453 if (std::isfinite(step) && step > 1e-5f && step < 1.0e4f) {
454 traveled += step;
455 const V3 fwd = normalize(pos - prev);
456 const V3 lat = normalize(cross(up, fwd));
457 const float hw = options.navRibbonHalfWidth;
458 appendStripQuad(mesh, prev + lat * -hw, pos + lat * -hw, pos + lat * hw, prev + lat * hw, up,
459 0.f, 1.f, 0.f, 1.f, RoadMaterial::Nav);
460 int arrowGuard = 0;
461 while (traveled >= nextArrow && arrowGuard++ < 64) {
462 appendArrow(mesh, pos, fwd, up, 0.85f);
463 nextArrow += std::max(options.arrowSpacing, 0.5f);
464 }
465 if (arrowGuard >= 64) nextArrow = traveled + std::max(options.arrowSpacing, 0.5f);
466 }
467 }
468 prev = pos;
469 hasPrev = true;
470 }
471 overlay.lanes.push_back(std::move(poly));
472 }
473 }
474 return Result<void>::success();
475}

477Result<void> bakeJunction(MeshBuild& mesh, const RoadNetwork& network, const RoadNode& node) {
478 struct ArmTip {
479 V3 origin;
480 float asphaltHalf = 0.f;
481 float halfWidth = 0.f;
482 float curbHeight = 0.f;
483 float sidewalkH = 0.f;
484 };
485 std::vector<ArmTip> arms;
486 float maxHalfWidth = 0.f;
487 float maxAsphalt = 0.f;
488 for (const auto& edge : network.edges()) {
489 if (edge.to != node.id && edge.from != node.id) continue;
490 auto spline = edgeToSpline(edge);
491 if (!spline.ok()) continue;
492 auto len = spline.value().lengthResult(16);
493 if (!len.ok()) continue;
494 const float d = edge.to == node.id ? std::max(0.f, len.value() - node.junctionRadius)
495 : std::min(node.junctionRadius, len.value());
496 auto frame = spline.value().travelFrameResult(d, "clamp", 16);
497 if (!frame.ok()) continue;
498 auto profile = makeRoadProfile(edge.style, edge.lanesForward, edge.lanesBackward);
499 if (!profile.ok()) continue;
500 const float asphaltHalf =
501 0.5f * edge.style.laneWidth * static_cast<float>(edge.lanesForward + edge.lanesBackward);
502 const auto& f = frame.value();
503 ArmTip tip;
504 tip.origin = V3{f.sample.x, f.sample.y, f.sample.z};
505 tip.asphaltHalf = asphaltHalf;
506 tip.halfWidth = profile.value().halfWidth;
507 tip.curbHeight = edge.style.curbHeight;
508 tip.sidewalkH = edge.style.sidewalkHeight;
509 arms.push_back(tip);
510 maxHalfWidth = std::max(maxHalfWidth, tip.halfWidth);
511 maxAsphalt = std::max(maxAsphalt, tip.asphaltHalf);
512 }
513 if (static_cast<int>(arms.size()) < 2) return Result<void>::success();
514
515 const float y = node.y;
516 const float asphaltY = y + 0.01f;
517 const float jr = node.junctionRadius;
518
519 // Detect an axis-aligned 4-way cross (the simple debug scene).
520 bool axisCross = arms.size() == 4;
521 if (axisCross) {
522 int axisHits = 0;
523 for (const auto& tip : arms) {
524 const float dx = std::fabs(tip.origin.x - node.x);
525 const float dz = std::fabs(tip.origin.z - node.z);
526 if ((dx < 0.35f && dz > jr * 0.4f) || (dz < 0.35f && dx > jr * 0.4f)) ++axisHits;
527 }
528 axisCross = axisHits == 4;
529 }
530
531 // Asphalt fill. Fan from a hub vertex — a single large strip-quad can vanish
532 // under Lavapipe even when the CPU triangles cover the origin.
533 if (axisCross) {
534 // Curb-return fillet: circle centers sit in the outer corner (jr,jr), so the
535 // asphalt edge is concave and sidewalks wrap toward the property corner.
536 const float ah = maxAsphalt;
537 const float cornerR = std::max(0.75f, jr - ah);
538 const int segs = 16;
539 const V3 upN{0.f, 1.f, 0.f};
540
541 // Hub fan along the filleted outline (arm tips + concave corner arcs).
542 {
543 std::vector<std::pair<float, float>> rim;
544 rim.reserve(static_cast<std::size_t>(4 * (segs + 3)));
545 auto pushPt = [&](float x, float z) {
546 if (!rim.empty() && std::fabs(rim.back().first - x) < 1e-4f &&
547 std::fabs(rim.back().second - z) < 1e-4f)
548 return;
549 rim.emplace_back(x, z);
550 };
551 auto pushCornerArc = [&](int sx, int sz, bool reverse) {
552 const float sxf = static_cast<float>(sx);
553 const float szf = static_cast<float>(sz);
554 const float cx = node.x + sxf * jr;
555 const float cz = node.z + szf * jr;
556 for (int i = 0; i <= segs; ++i) {
557 const int ii = reverse ? (segs - i) : i;
558 const float t = static_cast<float>(ii) / static_cast<float>(segs);
559 const float th = t * 1.5707963f;
560 pushPt(cx - sxf * cornerR * std::sin(th), cz - szf * cornerR * std::cos(th));
561 }
562 };
563 pushPt(node.x + jr, node.z - ah);
564 pushPt(node.x + jr, node.z + ah);
565 pushCornerArc(+1, +1, false);
566 pushPt(node.x - ah, node.z + jr);
567 pushCornerArc(-1, +1, true);
568 pushPt(node.x - jr, node.z - ah);
569 pushCornerArc(-1, -1, false);
570 pushPt(node.x + ah, node.z - jr);
571 pushCornerArc(+1, -1, true);
572
573 const V3 hub{node.x, asphaltY, node.z};
574 for (std::size_t i = 0; i < rim.size(); ++i) {
575 const auto& p0 = rim[i];
576 const auto& p1 = rim[(i + 1) % rim.size()];
577 appendOrientedTri(mesh, hub, V3{p0.first, asphaltY, p0.second},
578 V3{p1.first, asphaltY, p1.second}, upN, RoadMaterial::Asphalt);
579 }
580
581 // Thin seal strips under each arm tip — hides residual sub-cm seams.
582 const float seal = 0.12f;
583 const float ySeal = asphaltY + 0.002f;
584 auto addSeal = [&](float x0, float z0, float x1, float z1, float x2, float z2, float x3,
585 float z3) {
586 appendOrientedQuad(mesh, V3{x0, ySeal, z0}, V3{x1, ySeal, z1}, V3{x2, ySeal, z2}, V3{x3, ySeal, z3},
587 upN, 0.f, 1.f, 0.f, 1.f, RoadMaterial::Asphalt);
588 };
589 addSeal(node.x - ah, node.z + jr - seal, node.x + ah, node.z + jr - seal, node.x + ah, node.z + jr + seal,
590 node.x - ah, node.z + jr + seal);
591 addSeal(node.x - ah, node.z - jr - seal, node.x + ah, node.z - jr - seal, node.x + ah, node.z - jr + seal,
592 node.x - ah, node.z - jr + seal);
593 addSeal(node.x + jr - seal, node.z - ah, node.x + jr + seal, node.z - ah, node.x + jr + seal, node.z + ah,
594 node.x + jr - seal, node.z + ah);
595 addSeal(node.x - jr - seal, node.z - ah, node.x - jr + seal, node.z - ah, node.x - jr + seal, node.z + ah,
596 node.x - jr - seal, node.z + ah);
597 }
598
599 const float curbW = 0.35f;
600 const float walkW = std::max(0.4f, maxHalfWidth - ah - curbW);
601 float curbH = 0.45f;
602 float walkH = 0.14f;
603 if (!arms.empty()) {
604 curbH = arms.front().curbHeight;
605 walkH = arms.front().sidewalkH;
606 }
607 const float walkY = y + walkH;
608 const float curbTop = y + curbH;
609 const float r0 = cornerR;
610 const float r1 = std::max(0.2f, cornerR - curbW);
611 const float r2 = std::max(0.15f, cornerR - curbW - walkW);
612 for (int sx : {-1, 1}) {
613 for (int sz : {-1, 1}) {
614 const float sxf = static_cast<float>(sx);
615 const float szf = static_cast<float>(sz);
616 const float cx = node.x + sxf * jr;
617 const float cz = node.z + szf * jr;
618 auto arcPt = [&](float radius, float t) {
619 const float th = t * 1.5707963f;
620 return V3{cx - sxf * radius * std::sin(th), 0.f, cz - szf * radius * std::cos(th)};
621 };
622 auto wall = [&](V3 p0, V3 p1, float y0, float y1, V3 n, RoadMaterial mat) {
623 appendOrientedQuad(mesh, V3{p0.x, y0, p0.z}, V3{p1.x, y0, p1.z}, V3{p1.x, y1, p1.z},
624 V3{p0.x, y1, p0.z}, n, 0.f, 1.f, 0.f, 1.f, mat);
625 appendOrientedQuad(mesh, V3{p0.x, y0, p0.z}, V3{p1.x, y0, p1.z}, V3{p1.x, y1, p1.z},
626 V3{p0.x, y1, p0.z}, n * -1.f, 0.f, 1.f, 0.f, 1.f, mat);
627 };
628 for (int i = 0; i < segs; ++i) {
629 const float t0 = static_cast<float>(i) / static_cast<float>(segs);
630 const float t1 = static_cast<float>(i + 1) / static_cast<float>(segs);
631 const V3 a0 = arcPt(r0, t0);
632 const V3 a1 = arcPt(r0, t1);
633 const V3 b0 = arcPt(r1, t0);
634 const V3 b1 = arcPt(r1, t1);
635 const V3 s0 = arcPt(r2, t0);
636 const V3 s1 = arcPt(r2, t1);
637 // Sidewalk top slightly past r2 toward the property to cover the skirt seam.
638 const V3 c0 = arcPt(std::max(0.05f, r2 - 0.06f), t0);
639 const V3 c1 = arcPt(std::max(0.05f, r2 - 0.06f), t1);
640 const V3 b0i = arcPt(r1 - 0.02f, t0);
641 const V3 b1i = arcPt(r1 - 0.02f, t1);
642 const V3 b0o = arcPt(r1 + 0.02f, t0);
643 const V3 b1o = arcPt(r1 + 0.02f, t1);
644 const V3 inToCorner = normalize(V3{cx - a0.x, 0.f, cz - a0.z});
645 const V3 outToRoad = inToCorner * -1.f;
646
647 appendOrientedQuad(mesh, V3{a0.x, curbTop, a0.z}, V3{b0o.x, curbTop, b0o.z},
648 V3{b1o.x, curbTop, b1o.z}, V3{a1.x, curbTop, a1.z}, upN, 0.f, 1.f, 0.f, 1.f,
649 RoadMaterial::Curb);
650 appendOrientedQuad(mesh, V3{b0i.x, walkY, b0i.z}, V3{c0.x, walkY, c0.z}, V3{c1.x, walkY, c1.z},
651 V3{b1i.x, walkY, b1i.z}, upN, 0.f, 1.f, 0.f, 1.f, RoadMaterial::Sidewalk);
652
653 wall(a0, a1, y, curbTop, outToRoad, RoadMaterial::Curb);
654 wall(b0, b1, walkY, curbTop, outToRoad, RoadMaterial::Curb);
655 wall(s0, s1, y, walkY, outToRoad, RoadMaterial::Sidewalk);
656 }
657 }
658 }
659 } else {
660 const float radius =
661 std::sqrt(jr * jr + maxHalfWidth * maxHalfWidth) + 0.35f;
662 const int segs = 32;
663 const V3 hub{node.x, asphaltY, node.z};
664 const V3 upN{0.f, 1.f, 0.f};
665 for (int i = 0; i < segs; ++i) {
666 const float a0 = static_cast<float>(i) * 6.2831853f / static_cast<float>(segs);
667 const float a1 = static_cast<float>(i + 1) * 6.2831853f / static_cast<float>(segs);
668 appendOrientedTri(mesh, hub,
669 V3{node.x + std::cos(a0) * radius, asphaltY, node.z + std::sin(a0) * radius},
670 V3{node.x + std::cos(a1) * radius, asphaltY, node.z + std::sin(a1) * radius}, upN,
671 RoadMaterial::Asphalt);
672 }
673 }
674
675 // Zebra stripes on every arm tip that meets this hub.
676 for (const auto& edge : network.edges()) {
677 if (edge.to != node.id && edge.from != node.id) continue;
678 auto spline = edgeToSpline(edge);
679 if (!spline.ok()) continue;
680 auto len = spline.value().lengthResult(16);
681 if (!len.ok() || len.value() < node.junctionRadius + 1.5f) continue;
682 const float alongDist =
683 edge.to == node.id ? (len.value() - node.junctionRadius - 1.6f)
684 : (node.junctionRadius + 1.6f);
685 auto frame = spline.value().travelFrameResult(alongDist, "clamp", 16);
686 if (!frame.ok()) continue;
687 const auto& f = frame.value();
688 const V3 origin{f.sample.x, f.sample.y, f.sample.z};
689 const V3 side{f.sideX, f.sideY, f.sideZ};
690 // Stripe direction: toward the hub from this tip.
691 V3 fwd{f.forwardX, f.forwardY, f.forwardZ};
692 if (edge.from == node.id) fwd = fwd * -1.f;
693 const V3 nrm{0.f, 1.f, 0.f};
694 const float stripeW = edge.style.laneWidth * static_cast<float>(edge.lanesForward) * 0.45f;
695 for (int s = 0; s < 4; ++s) {
696 const float along = static_cast<float>(s) * 0.5f;
697 const V3 c = origin - fwd * along + nrm * 0.04f;
698 const V3 a = c + side * -stripeW;
699 const V3 b = c + side * stripeW;
700 const V3 d = a - fwd * 0.25f;
701 const V3 e = b - fwd * 0.25f;
702 appendStripQuad(mesh, a, b, e, d, nrm, 0.f, 1.f, 0.f, 1.f, RoadMaterial::Marking);
703 }
704 }
705 return Result<void>::success();
706}

708Result<void> bakeTurnOverlays(MeshBuild& mesh, RoadOverlay& overlay, const RoadNetwork& network,
709 const RoadBakeOptions& options) {
710 std::unordered_map<std::uint32_t, SplinePath> paths;
711 for (const auto& edge : network.edges()) {
712 auto spline = edgeToSpline(edge);
713 if (!spline.ok()) return Result<void>::failure(spline.status());
714 paths.emplace(edge.id, std::move(spline).takeValue());
715 }
716
717 for (const auto& link : network.laneLinks()) {
718 auto inEdge = network.edgeResult(link.inEdge);
719 auto outEdge = network.edgeResult(link.outEdge);
720 if (!inEdge.ok() || !outEdge.ok()) continue;
721 auto node = network.nodeResult(inEdge.value().to);
722 if (!node.ok()) continue;
723
724 auto& inPath = paths[link.inEdge];
725 auto& outPath = paths[link.outEdge];
726 auto inLen = inPath.lengthResult(16);
727 auto outLen = outPath.lengthResult(16);
728 if (!inLen.ok() || !outLen.ok()) continue;
729
730 const float inDist = std::max(0.f, inLen.value() - node.value().junctionRadius);
731 const float outDist = std::min(node.value().junctionRadius, outLen.value());
732 auto inFrame = inPath.travelFrameResult(inDist, "clamp", 16);
733 auto outFrame = outPath.travelFrameResult(outDist, "clamp", 16);
734 if (!inFrame.ok() || !outFrame.ok()) continue;
735
736 const float inLat = laneCenterOffset(inEdge.value().style, inEdge.value().lanesForward, link.inLane);
737 const float outLat = laneCenterOffset(outEdge.value().style, outEdge.value().lanesForward, link.outLane);
738
739 const auto& fi = inFrame.value();
740 const auto& fo = outFrame.value();
741 const V3 p0{fi.sample.x, fi.sample.y, fi.sample.z};
742 const V3 p1{fo.sample.x, fo.sample.y, fo.sample.z};
743 const V3 si{fi.sideX, fi.sideY, fi.sideZ};
744 const V3 so{fo.sideX, fo.sideY, fo.sideZ};
745 const V3 ui{fi.upX, fi.upY, fi.upZ};
746 const V3 uo{fo.upX, fo.upY, fo.upZ};
747 const V3 ti{fi.forwardX, fi.forwardY, fi.forwardZ};
748 const V3 to{fo.forwardX, fo.forwardY, fo.forwardZ};
749 const V3 a = p0 + si * inLat + ui * 0.09f;
750 const V3 d = p1 + so * outLat + uo * 0.09f;
751 const float handle = std::max(2.f, node.value().junctionRadius * 0.65f);
752 const V3 b = a + ti * handle;
753 const V3 c = d - to * handle;
754
755 RoadPolyline poly;
756 poly.r = 0.25f;
757 poly.g = 0.95f;
758 poly.b = 0.45f;
759 poly.width = options.navRibbonHalfWidth * 2.f;
760 const int samples = std::max(4, options.turnSamples);
761 V3 prev{};
762 bool hasPrev = false;
763 for (int i = 0; i <= samples; ++i) {
764 const float t = static_cast<float>(i) / static_cast<float>(samples);
765 const float u = 1.f - t;
766 const V3 p = a * (u * u * u) + b * (3.f * u * u * t) + c * (3.f * u * t * t) + d * (t * t * t);
767 poly.xyz.push_back(p.x);
768 poly.xyz.push_back(p.y);
769 poly.xyz.push_back(p.z);
770 if (hasPrev) {
771 const V3 fwd = normalize(p - prev);
772 const V3 up = normalize(ui + uo);
773 const V3 lat = normalize(cross(up, fwd));
774 const float hw = options.navRibbonHalfWidth * 0.9f;
775 appendStripQuad(mesh, prev + lat * -hw, p + lat * -hw, p + lat * hw, prev + lat * hw, up, 0.f, 1.f, 0.f,
776 1.f, RoadMaterial::Nav);
777 }
778 if (i == samples / 2) {
779 const V3 fwd = normalize(d - a);
780 appendArrow(mesh, p, fwd, normalize(ui + uo), 0.7f);
781 }
782 prev = p;
783 hasPrev = true;
784 }
785 overlay.turns.push_back(std::move(poly));
786 }
787 return Result<void>::success();
788}

798MaterialColor colorForGroup(const std::string& name) {
799 if (name == "asphalt") return {0.28f, 0.28f, 0.30f, 1.f};
800 if (name == "curb") return {0.78f, 0.78f, 0.76f, 1.f};

803 if (name == "pier") return {0.70f, 0.68f, 0.64f, 1.f};

810Result<void> paintGroupVertexColors(MeshBuild& mesh) {
811 const int verts = mesh.getVertexCount();
812 if (verts <= 0) return Result<void>::success();
813 std::vector<float> colors(static_cast<std::size_t>(verts) * 4u, 1.f);
814 const int triCount = mesh.getIndexCount() / 3;
815 for (int t = 0; t < triCount; ++t) {
816 const int group = mesh.getTriangleGroup(t);
817 const MaterialColor c =
818 group >= 0 ? colorForGroup(mesh.getGroupName(group)) : MaterialColor{0.5f, 0.5f, 0.5f, 1.f};
819 for (int k = 0; k < 3; ++k) {
820 const int vi = mesh.getIndex(t * 3 + k);
821 if (vi < 0 || vi >= verts) continue;
822 const auto base = static_cast<std::size_t>(vi) * 4u;
823 colors[base + 0] = c.r;
824 colors[base + 1] = c.g;
825 colors[base + 2] = c.b;
826 colors[base + 3] = c.a;
827 }
828 }
829 return mesh.setVertexColors(std::move(colors));
830}

834Result<RoadBakeResult> bakeRoadNetwork(const RoadNetwork& network, const RoadBakeOptions& options) {
835 if (options.pathSegmentsPerEdge < 2 || options.turnSamples < 2)
836 return Result<RoadBakeResult>::failure(
837 Diagnostic::error(DiagnosticCode::InvalidArgument, "segment counts must be >= 2", {}, {}, "procgen.road"));
838 if (network.edgeCount() == 0)
839 return Result<RoadBakeResult>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
840 "road network has no edges", {}, {}, "procgen.road"));
841
842 RoadBakeResult result;
843 for (const auto& edge : network.edges()) {
844 auto baked = bakeEdgeGeometry(result.mesh, result.overlay, network, edge, options);
845 if (!baked.ok()) return Result<RoadBakeResult>::failure(baked.status());
846 }
847 if (options.includeJunctions) {
848 for (const auto& node : network.nodes()) {
849 auto junction = bakeJunction(result.mesh, network, node);
850 if (!junction.ok()) return Result<RoadBakeResult>::failure(junction.status());
851 }
852 }
853 if (options.includeNavigation) {
854 auto turns = bakeTurnOverlays(result.mesh, result.overlay, network, options);
855 if (!turns.ok()) return Result<RoadBakeResult>::failure(turns.status());
856 }
857
858 auto painted = paintGroupVertexColors(result.mesh);
859 if (!painted.ok()) return Result<RoadBakeResult>::failure(painted.status());
860
861 result.mesh.setMeta("generator", "procgen.road");
862 result.mesh.setMeta("schema", "eve.procgen.roadNetwork");
863 result.mesh.setMeta("schemaVersion", "1");
864 result.mesh.setMeta("edges", std::to_string(network.edgeCount()));
865 result.mesh.setMeta("nodes", std::to_string(network.nodeCount()));
866 result.mesh.setMeta("laneLinks", std::to_string(network.laneLinkCount()));
867 if (result.mesh.getVertexCount() < 3)
868 return Result<RoadBakeResult>::failure(Diagnostic::error(
869 DiagnosticCode::InvariantViolation, "bake produced an empty mesh", {}, {}, "procgen.road"));
870 return Result<RoadBakeResult>::success(std::move(result));
871}
