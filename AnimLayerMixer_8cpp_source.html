46bool AnimBoneMask::setBoneAndChildren(const std::string& boneName, float weight) {
47 if (!skeleton_) return false;
48 const int root = skeleton_->findBone(boneName);
49 if (root < 0) return false;
50 const float clamped = clampf(weight, 0.f, 1.f);
51 for (int bone = 0; bone < skeleton_->getBoneCount(); ++bone) {
52 int current = bone;
53 while (current >= 0 && current != root) current = skeleton_->getParent(current);
54 if (current == root) weights_[static_cast<size_t>(bone)] = clamped;
55 }
56 return true;
57}

128void AnimLayerMixer::applyOverride(const Layer& layer) {
129 AnimPose* layerPose = layer.player->getPose();
130 for (int bone = 0; bone < pose_.getBoneCount(); ++bone) {
131 const float weight = layerBoneWeight(layer.mask, layer.weight, bone);
132 if (weight <= 0.f) continue;
133 pose_.local(bone) = blendTRS(pose_.local(bone), layerPose->local(bone), weight);
134 }

137void AnimLayerMixer::applyAdditive(const Layer& layer) {
138 AnimPose* layerPose = layer.player->getPose();
139 for (int bone = 0; bone < pose_.getBoneCount(); ++bone) {
140 const float weight = layerBoneWeight(layer.mask, layer.weight, bone);
141 if (weight <= 0.f) continue;
142 TransformTRS& base = pose_.local(bone);
143 const TransformTRS& sample = layerPose->local(bone);
144 const TransformTRS& reference = skeleton_->bindLocal(bone);
145 base.px += (sample.px - reference.px) * weight;
146 base.py += (sample.py - reference.py) * weight;
147 base.pz += (sample.pz - reference.pz) * weight;
148 base.sx *= lerpf(1.f, std::fabs(reference.sx) > 1e-8f ? sample.sx / reference.sx : 1.f, weight);
149 base.sy *= lerpf(1.f, std::fabs(reference.sy) > 1e-8f ? sample.sy / reference.sy : 1.f, weight);
150 base.sz *= lerpf(1.f, std::fabs(reference.sz) > 1e-8f ? sample.sz / reference.sz : 1.f, weight);
151
152 float dx, dy, dz, dw;
153 multiplyQuat(sample.qx, sample.qy, sample.qz, sample.qw, -reference.qx, -reference.qy, -reference.qz,
154 reference.qw, dx, dy, dz, dw);
155 float ax, ay, az, aw;
156 slerpQuat(0.f, 0.f, 0.f, 1.f, dx, dy, dz, dw, weight, ax, ay, az, aw);
157 float qx, qy, qz, qw;
158 multiplyQuat(base.qx, base.qy, base.qz, base.qw, ax, ay, az, aw, qx, qy, qz, qw);
159 base.qx = qx;
160 base.qy = qy;
161 base.qz = qz;
162 base.qw = qw;
163 base.normalizeRotation();
164 }

167void AnimLayerMixer::compose() {
168 events_.clear();
169 if (basePlayer_) {
170 pose_.copyFrom(basePlayer_->getPose());
171 collectEvents("base", basePlayer_);
172 } else {
173 skeleton_->applyBindPose(&pose_);
174 }
175 for (const Layer& layer : layers_) {
176 if (!layer.enabled || !layer.player) continue;
177 if (layer.weight <= 0.f) continue;
178 collectEvents(layer.name, layer.player);
179 if (layer.additive)
180 applyAdditive(layer);
181 else
182 applyOverride(layer);
183 }
184 pose_.computeWorld(skeleton_);
185}

187eve::Result<void> AnimLayerMixer::advance(const eve::SimulationStep& step) {
188 auto seconds = detail::secondsForStep(step, hasLastTick_, lastTick_, "AnimLayerMixer");
189 if (!seconds) return eve::Result<void>::failure(seconds.status());
190 (void)std::move(seconds).takeValue();
191
192 // The mixer owns the evaluation boundary: all referenced players consume
193 // the same injected step before their poses are combined.
194 if (basePlayer_ && basePlayer_->hasCurrentTick() && step.tick <= basePlayer_->currentTick())
195 return eve::Result<void>::failure(eve::Diagnostic::error(
196 eve::DiagnosticCode::Conflict, "animation mixer base player already consumed this tick"));
197 for (const Layer& layer : layers_) {
198 if (layer.enabled && layer.player && layer.player->hasCurrentTick() && step.tick <= layer.player->currentTick())
199 return eve::Result<void>::failure(eve::Diagnostic::error(
200 eve::DiagnosticCode::Conflict, "animation mixer layer player already consumed this tick"));
201 }
202 if (basePlayer_) {
203 auto result = basePlayer_->advance(step);
204 if (!result) return eve::Result<void>::failure(result.status());
205 }
206 for (const Layer& layer : layers_) {
207 if (!layer.enabled || !layer.player) continue;
208 auto result = layer.player->advance(step);
209 if (!result) return eve::Result<void>::failure(result.status());
210 }
211 compose();
212 lastTick_ = step.tick;
213 hasLastTick_ = true;
214 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
215}
