54bool AnimBoneMask::setBoneAndChildren(const std::string& boneName, float weight) {
55 if (!skeleton_) return false;
56 const int root = skeleton_->findBone(boneName);
57 if (root < 0) return false;
58 const float clamped = clampf(weight, 0.f, 1.f);
59 for (int bone = 0; bone < skeleton_->getBoneCount(); ++bone) {
60 int current = bone;
61 while (current >= 0 && current != root) current = skeleton_->getParent(current);
62 if (current == root) weights_[static_cast<size_t>(bone)] = clamped;
63 }
64 return true;

103 if (!source) {
104 baseSource_ = nullptr;

112 if (source->getSkeleton() != skeleton_)

115 for (const Layer& layer : layers_)

152eve::Result<int> AnimLayerMixer::addPoseLayer(const std::string& name, IAnimPoseSource* source, AnimBoneMask* mask,
153 const std::string& mode) {
154 if (name.empty())
155 return eve::Result<int>::failure(
156 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "animation mixer layer name is empty"));
157 if (findLayer(name))
158 return eve::Result<int>::failure(
159 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "animation mixer layer name already exists"));
160 if (mode != "override" && mode != "additive")
161 return eve::Result<int>::failure(eve::Diagnostic::error(
162 eve::DiagnosticCode::InvalidArgument, "animation mixer layer mode must be \"override\" or \"additive\""));
163 if (mask && mask->getSkeleton() != skeleton_)
164 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
165 "animation mixer layer mask skeleton mismatch"));
166 auto attached = attachSource(source, "layer");
167 if (!attached) return eve::Result<int>::failure(attached.status());
168
169 layers_.push_back({name, source, mask, 1.f, mode == "additive", true, AnimAdditiveReference::BindPose});
170 return eve::Result<int>::success(static_cast<int>(layers_.size()) - 1);
171}

292void AnimLayerMixer::applyOverride(const Layer& layer) {
293 AnimPose* layerPose = layer.source->getPose();
294 for (int bone = 0; bone < pose_.getBoneCount(); ++bone) {
295 const float weight = layerBoneWeight(layer.mask, layer.weight, bone);
296 if (weight <= 0.f) continue;
297 pose_.local(bone) = blendTRS(pose_.local(bone), layerPose->local(bone), weight);
298 }
299}

301void AnimLayerMixer::applyAdditive(const Layer& layer) {
302 AnimPose* layerPose = layer.source->getPose();
303 for (int bone = 0; bone < pose_.getBoneCount(); ++bone) {
304 const float weight = layerBoneWeight(layer.mask, layer.weight, bone);
305 if (weight <= 0.f) continue;
306 const TransformTRS& reference = layer.additiveReference == AnimAdditiveReference::Identity
307 ? TransformTRS::identity()
308 : skeleton_->bindLocal(bone);
309 applyAdditiveTRS(pose_.local(bone), layerPose->local(bone), reference, weight);
310 }

313void AnimLayerMixer::compose() {
314 events_.clear();
315 if (baseSource_) {
316 pose_.copyFrom(baseSource_->getPose());
317 collectEvents("base", baseSource_);
318 } else {
319 skeleton_->applyBindPose(&pose_);
320 }
321 for (const Layer& layer : layers_) {
322 if (!layer.enabled || !layer.source) continue;
323 if (layer.weight <= 0.f) continue;
324 collectEvents(layer.name, layer.source);
325 if (layer.additive)
326 applyAdditive(layer);
327 else
328 applyOverride(layer);
329 }
330 pose_.computeWorld(skeleton_);
331}

333eve::Result<void> AnimLayerMixer::advance(const eve::SimulationStep& step) {
334 auto seconds = detail::secondsForStep(step, hasLastTick_, lastTick_, "AnimLayerMixer");
335 if (!seconds) return eve::Result<void>::failure(seconds.status());
336 (void)std::move(seconds).takeValue();
337
338 // The mixer owns the evaluation boundary: every attached source consumes the
339 // same injected step (including disabled layers, so their clocks stay in sync).
340 if (baseSource_ && baseSource_->hasCurrentTick() && step.tick <= baseSource_->currentTick())
341 return eve::Result<void>::failure(eve::Diagnostic::error(
342 eve::DiagnosticCode::Conflict, "animation mixer base source already consumed this tick"));
343 for (const Layer& layer : layers_) {
344 if (layer.source && layer.source->hasCurrentTick() && step.tick <= layer.source->currentTick())
345 return eve::Result<void>::failure(eve::Diagnostic::error(
346 eve::DiagnosticCode::Conflict, "animation mixer layer source already consumed this tick"));
347 }
348 if (baseSource_) {
349 auto result = baseSource_->advance(step);
350 if (!result) return eve::Result<void>::failure(result.status());
351 }
352 for (const Layer& layer : layers_) {
353 if (!layer.source) continue;
354 auto result = layer.source->advance(step);
355 if (!result) return eve::Result<void>::failure(result.status());
356 }
357 compose();
358 lastTick_ = step.tick;
359 hasLastTick_ = true;
360 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
361}
