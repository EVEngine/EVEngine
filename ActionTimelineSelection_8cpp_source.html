28EditorResult<TimelineSelectionRange> ActionTimelineEditor::selectionRange() const {
29 std::optional<TimelineSelectionRange> range;
30 const auto include = [&](Duration start, Duration end) {
31 if (!range) {
32 range = TimelineSelectionRange{start, end};
33 return;
34 }
35 range->start = std::min(range->start, start);
36 range->end = std::max(range->end, end);
37 };
38 for (const auto& section : target_.timeline().animationSections)
39 if (selection_.contains(section.id.format())) include(section.start, section.end);
40 for (const auto& track : target_.timeline().tracks) {
41 for (const auto& notify : track.notifies)
42 if (selection_.contains(notify.id.format())) include(notify.time, notify.time);
43 for (const auto& state : track.states)
44 if (selection_.contains(state.id.format())) include(state.start, state.end);
45 }
46 if (!range)
47 return eve::editing::failed<TimelineSelectionRange>(EditorStatus::Rejected,
48 RuleId("editor.action.timeline.selection-empty"),
49 "No timeline items are selected");
50 return eve::editing::applied<TimelineSelectionRange>(*range);
51}

53EditorResult<void> ActionTimelineEditor::moveSelection(Duration delta) {
54 if (selection_.empty()) return rejected("editor.action.timeline.selection-empty", "No timeline items are selected");
55 action::ActionTimeline candidate = target_.timeline();
56 const auto move = [&](Duration value) { return value.tryAdd(delta); };
57 for (auto& section : candidate.animationSections) {
58 if (!selection_.contains(section.id.format())) continue;
59 auto start = move(section.start);
60 auto end = move(section.end);
61 if (!start || !end)
62 return rejected("editor.action.timeline.time-overflow", "Selected animation-section time overflowed");
63 section.start = start.value();
64 section.end = end.value();
65 }
66 for (auto& track : candidate.tracks) {
67 const bool selectedNotify = std::any_of(track.notifies.begin(), track.notifies.end(),
68 [&](const auto& item) { return selection_.contains(item.id.format()); });
69 const bool selectedState = std::any_of(track.states.begin(), track.states.end(),
70 [&](const auto& item) { return selection_.contains(item.id.format()); });
71 if (track.locked && (selectedNotify || selectedState))
72 return rejected("editor.action.timeline.track-locked", "Selection contains an item on a locked track");
73 for (auto& notify : track.notifies) {
74 if (!selection_.contains(notify.id.format())) continue;
75 auto time = move(notify.time);
76 if (!time) return rejected("editor.action.timeline.time-overflow", "Selected notify time overflowed");
77 notify.time = time.value();
78 }
79 for (auto& state : track.states) {
80 if (!selection_.contains(state.id.format())) continue;
81 auto start = move(state.start);
82 auto end = move(state.end);
83 if (!start || !end)
84 return rejected("editor.action.timeline.time-overflow", "Selected notify-state time overflowed");
85 state.start = start.value();
86 state.end = end.value();
87 }
88 sortTrack(track);
89 }
90 sortAnimationSections(candidate);
91 return commit(std::move(candidate), "Move action timeline selection", "action.timeline.selection.move");
92}

178EditorResult<void> ActionTimelineEditor::scaleSelection(Duration start, Duration end) {
179 auto range = selectionRange();
180 if (!range.ok()) return EditorResult<void>::failure(range.status());
181 if (start < Duration::zero() || end <= start || end > target_.timeline().duration)
182 return rejected("editor.action.timeline.selection-scale-range",
183 "Scaled selection range must be ordered and inside the timeline");
184 const auto oldSpan = range.value().end.nanoseconds() - range.value().start.nanoseconds();
185 if (oldSpan <= 0)
186 return rejected("editor.action.timeline.selection-scale-point",
187 "A point-only selection cannot be proportionally scaled");
188 const double rate = static_cast<double>(end.nanoseconds() - start.nanoseconds()) /
189 static_cast<double>(oldSpan);
190 const auto map = [&](Duration value) -> Result<Duration> {
191 const Duration offset = Duration::fromNanoseconds(value.nanoseconds() - range.value().start.nanoseconds());
192 auto scaled = offset.scaled(rate);
193 if (!scaled) return scaled;
194 return start.tryAdd(scaled.value());
195 };
196 action::ActionTimeline candidate = target_.timeline();
197 for (auto& section : candidate.animationSections) {
198 if (!selection_.contains(section.id.format())) continue;
199 auto mappedStart = map(section.start);
200 auto mappedEnd = map(section.end);
201 auto blend = section.blendIn.scaled(rate);
202 if (!mappedStart || !mappedEnd || !blend)
203 return rejected("editor.action.timeline.selection-scale-overflow", "Scaled selection time overflowed");
204 section.start = mappedStart.value();
205 section.end = mappedEnd.value();
206 section.blendIn = blend.value();
207 }
208 for (auto& track : candidate.tracks) {
209 const bool touchesLocked = track.locked &&
210 (std::any_of(track.notifies.begin(), track.notifies.end(), [&](const auto& item) {
211 return selection_.contains(item.id.format());
212 }) ||
213 std::any_of(track.states.begin(), track.states.end(), [&](const auto& item) {
214 return selection_.contains(item.id.format());
215 }));
216 if (touchesLocked)
217 return rejected("editor.action.timeline.track-locked", "Selection contains an item on a locked track");
218 for (auto& notify : track.notifies) {
219 if (!selection_.contains(notify.id.format())) continue;
220 auto mapped = map(notify.time);
221 if (!mapped)
222 return rejected("editor.action.timeline.selection-scale-overflow", "Scaled notify time overflowed");
223 notify.time = mapped.value();
224 }
225 for (auto& state : track.states) {
226 if (!selection_.contains(state.id.format())) continue;
227 auto mappedStart = map(state.start);
228 auto mappedEnd = map(state.end);
229 if (!mappedStart || !mappedEnd)
230 return rejected("editor.action.timeline.selection-scale-overflow",
231 "Scaled notify-state time overflowed");
232 state.start = mappedStart.value();
233 state.end = mappedEnd.value();
234 }
235 sortTrack(track);
236 }
237 sortAnimationSections(candidate);
238 return commit(std::move(candidate), "Scale action timeline selection", "action.timeline.selection.scale");
239}
