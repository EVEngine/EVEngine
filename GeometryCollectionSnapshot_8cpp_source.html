126eve::Result<void> GeometryCollectionInstanceSnapshot::validate() const {
127 if (bones.empty())
128 return eve::Result<void>::failure(eve::Diagnostic::error(
129 eve::DiagnosticCode::InvalidArgument, "instance-snapshot requires at least one bone", "bones"));
130 if (bones.size() > 4096 || edges.size() > 16384)
131 return eve::Result<void>::failure(eve::Diagnostic::error(
132 eve::DiagnosticCode::InvalidArgument, "instance-snapshot exceeds size limits", "snapshot"));
133 if (!std::isfinite(originX) || !std::isfinite(originY) || !std::isfinite(originZ))
134 return eve::Result<void>::failure(eve::Diagnostic::error(
135 eve::DiagnosticCode::InvalidArgument, "instance-snapshot origin must be finite", "origin"));
136 if (lastTick < 0)
137 return eve::Result<void>::failure(eve::Diagnostic::error(
138 eve::DiagnosticCode::InvalidArgument, "instance-snapshot lastTick must be non-negative", "lastTick"));
139 for (const auto& bone : bones) {
140 if (bone.state > 2)
141 return eve::Result<void>::failure(eve::Diagnostic::error(
142 eve::DiagnosticCode::InvalidArgument, "instance-snapshot bone state must be 0..2", "bones"));
143 if (bone.clusterId < 0 || bone.fractureLevel < 0)
144 return eve::Result<void>::failure(eve::Diagnostic::error(
145 eve::DiagnosticCode::InvalidArgument, "instance-snapshot cluster fields must be non-negative",
146 "bones"));
147 if (!(bone.halfExtentX > 0.f) || !(bone.halfExtentY > 0.f) || !(bone.halfExtentZ > 0.f) ||
148 !std::isfinite(bone.halfExtentX) || !std::isfinite(bone.halfExtentY) || !std::isfinite(bone.halfExtentZ))
149 return eve::Result<void>::failure(eve::Diagnostic::error(
150 eve::DiagnosticCode::InvalidArgument, "instance-snapshot half-extents must be finite and positive",
151 "bones"));
152 }
153 for (const auto& edge : edges) {
154 if (edge.boneA < 0 || edge.boneB < 0 || static_cast<std::size_t>(edge.boneA) >= bones.size() ||
155 static_cast<std::size_t>(edge.boneB) >= bones.size() || edge.boneA == edge.boneB)
156 return eve::Result<void>::failure(eve::Diagnostic::error(
157 eve::DiagnosticCode::InvalidArgument, "instance-snapshot edge indices invalid", "edges"));
158 if (!std::isfinite(edge.strainThreshold) || edge.strainThreshold <= 0.f ||
159 !std::isfinite(edge.accumulatedStrain) || edge.accumulatedStrain < 0.f)
160 return eve::Result<void>::failure(eve::Diagnostic::error(
161 eve::DiagnosticCode::InvalidArgument, "instance-snapshot edge strain values invalid", "edges"));
162 }
163 return eve::Result<void>::success();
164}
