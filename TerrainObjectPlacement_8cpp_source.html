16public:
17 explicit PcgRandom(std::int32_t seed) {
18 const auto value = seed == 0 ? 1U : static_cast<std::uint32_t>(seed);
19 a_ = UINT64_C(181353) * value;
20 b_ = UINT64_C(7) * value;
21 }
22 float next() {
23 auto x = a_, y = b_;
24 a_ = y;
25 x ^= x << 23;
26 x ^= x >> 17;
27 x ^= y ^ (y >> 26);
28 b_ = x;
29 return static_cast<float>(x + y) / static_cast<float>(std::numeric_limits<std::uint64_t>::max());
30 }
31 float next(float minimum, float maximum) { return minimum + next() * (maximum - minimum); }

33 if (minimum == maximum) return minimum;

35 }

60Normal normalAt(const Heightmap& heights, double u, double v, double width, double depth, double heightScale) {
61 const double du = heights.getWidth() > 1 ? 1.0 / (heights.getWidth() - 1) : 1;
62 const double dv = heights.getHeight() > 1 ? 1.0 / (heights.getHeight() - 1) : 1;
63 const double dx = (raster_detail::sample(heights, std::min(1.0, u + du), v) -
64 raster_detail::sample(heights, std::max(0.0, u - du), v)) *
65 heightScale;
66 const double dz = (raster_detail::sample(heights, u, std::min(1.0, v + dv)) -
67 raster_detail::sample(heights, u, std::max(0.0, v - dv))) *
68 heightScale;
69 const double sx = std::max(du * width * 2, std::numeric_limits<double>::epsilon());
70 const double sz = std::max(dv * depth * 2, std::numeric_limits<double>::epsilon());
71 Normal n{-dx / sx, 1, -dz / sz};
72 const double length = std::sqrt(n.x * n.x + n.y * n.y + n.z * n.z);
73 n.x /= length;
74 n.y /= length;
75 n.z /= length;
76 return n;
77}

191 if (instance.scaleMode == TerrainObjectScaleMode::Random) {

231 } else
232 finalY += yOffset;

235 random.next(instance.minimumRotationY + direction, instance.maximumRotationY + direction);
236 double roll = random.next(instance.minimumRotationZ, instance.maximumRotationZ);

268 point.boundsMaxY = float(sy);
269 point.seed = static_cast<std::uint32_t>(mix(point.id ^ static_cast<std::uint32_t>(s.seed)));

271 auto asset = next.trySetStringAttribute(row, "asset", instance.asset);
272 if (!asset.ok()) return Result<int>::failure(asset.status());
273 auto prototype = next.trySetStringAttribute(row, "objectPrototype", s.prototype);

279 if (!candidateAttr.ok()) return Result<int>::failure(candidateAttr.status());

281 next.trySetIntAttribute(row, "objectResource", static_cast<std::int64_t>(resource));

283 ++emitted;
284 }
285 }
286 if (emitted > emittedBefore) acceptedCenters.emplace_back(centerX, centerZ);
287 }
288 }
289 static_assert(std::is_nothrow_move_assignable_v<PointSet>);
290 output = std::move(next);
291 return Result<int>::success(emitted);
292}

295 const TerrainObjectPlacementSettings& s, float removalStrength) {

297 if (!validRaster(fitness) || !std::all_of(fitness.data().begin(), fitness.data().end(), normalized) ||
298 !std::isfinite(s.originX) || !std::isfinite(s.originZ) || !positive(s.width) || !positive(s.depth) ||
