30vk::Pipeline createPbrPipeline(vkb::Device& device, vk::PipelineLayout layout, const vkb::BuiltRenderPass& pass,
31 vk::SampleCountFlagBits samples, BlendMode blend, bool depthWrite, bool doubleSided) {
32 ShaderModulePair shaders(device, embeddedSpirv(pbr_surface_vert_spv), embeddedSpirv(pbr_surface_frag_spv));
33 auto input = vkb::VertexInputStateBuilder();
34 input.addInputBinding<MeshVertex>().addAttributeDescription<MeshVertex>();
35 // Five mesh attributes plus eleven material UV streams fit Vulkan's minimum
36 // sixteen vertex attributes. Each role can select any canonical mesh UV set.
37 for (uint32_t i = 0; i < 11; ++i) {
38 input.input_bindings.emplace_back(i + 1, uint32_t(2 * sizeof(float)), vk::VertexInputRate::eVertex);
39 input.input_attributes.emplace_back(i + 5, i + 1, vk::Format::eR32G32Sfloat, 0);
40 }
41 auto attachment = makeBlendAttachment(blend);
42 attachment.blendEnable = blend != BlendMode::Opaque;
43 vk::PipelineColorBlendStateCreateInfo color{};
44 color.attachmentCount = 1;
45 color.pAttachments = &attachment;
46 return device.createPipeline()
47 .useClassicPipeline(shaders.vert, shaders.frag)
48 .setPipelineLayout(layout)
49 .setVertexInputState(input)
50 .setDynamicStatesViewportScissor()
51 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
52 doubleSided ? vk::CullModeFlagBits::eNone : vk::CullModeFlagBits::eBack,
53 vk::FrontFace::eCounterClockwise)
54 .setMultisampler(false, samples)
55 .setDepthStencil(true, depthWrite, vk::CompareOp::eLess)
56 .setColorBlending(color)
57 .build(pass);
58}
