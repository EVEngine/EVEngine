40EditorResult<void> UiDocumentTarget::applyDomainOperation(const DomainOperation& operation) {
41 if (operation.target != TargetId(id_))
42 return uiError<void>(EditorStatus::Rejected, "editor.ui.target-mismatch",
43 "UI operation targets another document");
44 if (operation.type == "ui.widget.create.v1") {
45 auto parsed = parseWidget(operation.payload);
46 if (!parsed.ok())
47 return uiError<void>(EditorStatus::Rejected, "editor.ui.create-payload", "UI create payload is invalid");
48 const UiWidgetSnapshot& value = parsed.value();
49 if (value.id.empty() || widgets_.contains(value.id))
50 return uiError<void>(EditorStatus::Conflict, "editor.ui.widget-exists",
51 "UI widget id is empty or already exists");
52 if (!value.parent.empty() && !widgets_.contains(value.parent))
53 return uiError<void>(EditorStatus::NotFound, "editor.ui.parent-not-found",
54 "UI widget parent does not exist");
55 widgets_.emplace(value.id, value);
56 } else if (operation.type == "ui.widget.delete.v1") {
57 auto parsed = parseWidget(operation.payload);
58 if (!parsed.ok() || !widgets_.contains(parsed.value().id))
59 return uiError<void>(EditorStatus::NotFound, "editor.ui.widget-not-found", "UI widget does not exist");
60 if (!children(parsed.value().id).empty())
61 return uiError<void>(EditorStatus::Rejected, "editor.ui.widget-has-children",
62 "UI widget with children cannot be deleted");
63 widgets_.erase(parsed.value().id);
64 } else if (operation.type == "ui.widget.replace.v1") {
65 auto parsed = parseWidget(operation.payload);
66 if (!parsed.ok())
67 return uiError<void>(EditorStatus::Rejected, "editor.ui.replace-payload",
68 "UI replacement payload is invalid");
69 auto current = widgets_.find(parsed.value().id);
70 if (current == widgets_.end())
71 return uiError<void>(EditorStatus::NotFound, "editor.ui.widget-not-found", "UI widget does not exist");
72 if (!parsed.value().parent.empty() && !widgets_.contains(parsed.value().parent))
73 return uiError<void>(EditorStatus::NotFound, "editor.ui.parent-not-found",
74 "UI widget parent does not exist");
75 if (wouldCycle(parsed.value().id, parsed.value().parent, widgets_))
76 return uiError<void>(EditorStatus::Rejected, "editor.ui.hierarchy-cycle",
77 "UI reparenting would create a hierarchy cycle");
78 current->second = parsed.value();
79 } else if (operation.type == "ui.widget.multi-replace.v1") {
80 const auto* entries = operation.payload.getIf<EditorValue::Array>();
81 if (!entries)
82 return uiError<void>(EditorStatus::Rejected, "editor.ui.multi-payload",
83 "UI multi-edit requires an array payload");
84 auto candidate = widgets_;
85 for (const EditorValue& entry : *entries) {
86 auto parsed = parseWidget(entry);
87 if (!parsed.ok() || !candidate.contains(parsed.value().id))
88 return uiError<void>(EditorStatus::Rejected, "editor.ui.multi-widget",
89 "UI multi-edit references an invalid widget");
90 candidate[parsed.value().id] = parsed.value();
91 }
92 for (const auto& [widgetId, value] : candidate) {
93 if ((!value.parent.empty() && !candidate.contains(value.parent)) ||
94 wouldCycle(widgetId, value.parent, candidate))
95 return uiError<void>(EditorStatus::Rejected, "editor.ui.multi-hierarchy",
96 "UI multi-edit produces an invalid hierarchy");
97 }
98 widgets_ = std::move(candidate);
99 } else {
100 return uiError<void>(EditorStatus::Unsupported, "editor.ui.operation-unsupported",
101 "Unsupported UI operation: " + operation.type);
102 }
103 ++revision_;
104 dirty_.include(0, 0);
105 return eve::editing::applied<void>();
106}

272std::vector<EditorDiagnostic> UiDocumentTarget::validate() const {
273 std::vector<EditorDiagnostic> diagnostics;
274 int roots = 0;
275 for (const auto& [id, value] : widgets_) {
276 if (value.parent.empty()) ++roots;
277 if (value.layout.width == 0.0 || value.layout.height == 0.0)
278 diagnostics.push_back(uiDiagnostic("editor.ui.zero-size-widget", DiagnosticSeverity::Warning,
279 "Widget has zero preview size: " + id.value()));
280 if (value.style.cornerRadius * 2.0 > std::min(value.layout.width, value.layout.height) &&
281 value.layout.width > 0.0 && value.layout.height > 0.0)
282 diagnostics.push_back(uiDiagnostic("editor.ui.excessive-corner-radius", DiagnosticSeverity::Warning,
283 "Corner radius exceeds half of widget size: " + id.value()));
284 if (!value.text.empty() && value.content.fontAsset.empty())
285 diagnostics.push_back(uiDiagnostic("editor.ui.missing-font", DiagnosticSeverity::Warning,
286 "Text widget has no font asset: " + id.value()));
287 }
288 if (!widgets_.empty() && roots == 0)
289 diagnostics.push_back(uiDiagnostic("editor.ui.missing-root", DiagnosticSeverity::Error,
290 "UI document has no root widget"));
291 if (roots > 1)
292 diagnostics.push_back(uiDiagnostic("editor.ui.multiple-roots", DiagnosticSeverity::Warning,
293 "UI document contains multiple root widgets"));
294 return diagnostics;
295}

312bool UiDocumentTarget::wouldCycle(const ObjectId& id, const ObjectId& parent,
313 const std::map<ObjectId, UiWidgetSnapshot>& widgets) const {
314 ObjectId ancestor = parent;
315 while (!ancestor.empty()) {
316 if (ancestor == id) return true;
317 const auto current = widgets.find(ancestor);
318 if (current == widgets.end()) return false;
319 ancestor = current->second.parent;
320 }
321 return false;
322}
