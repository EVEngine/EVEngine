51 if (index < 0 || index >= chamberCount() || !std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z) ||

56 }
57 auto& chamber = state_->chambers[size_t(index)];
58 chamber.center = {x, y, z};
59 chamber.radii = {rx, ry, rz};
60 state_->breakdownChambers[size_t(index)] = {x, y, z, rx, ry, rz};
61 state_->breakdownSet = createCaveBreakdown(state_->breakdownChambers, state_->breakdownEvents, state_->breakdown,

63 state_->obstacleScourSites =
64 createCaveObstacleScourSites(state_->breakdownSet, state_->spine, state_->hydrology.trunk, state_->sedimentLoad,
65 state_->multiscaleRoughness);
66 if (index == state_->primaryChamber) state_->primaryChamberVerticalRadius = ry;
67 state_->skeletonEdited = true;
