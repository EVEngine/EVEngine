35template <class T>
36bool readObjectMember(const eve::Value::Object& object, std::string_view key, T& result) {
37 const auto it = object.find(std::string(key));
38 if (it == object.end()) return false;
39 if constexpr (std::is_same_v<T, std::string>) {
40 if (!it->second.isString()) return false;
41 result = it->second.asString();
42 return true;
43 } else if constexpr (std::is_same_v<T, std::int64_t>) {
44 if (!it->second.isInt64()) return false;
45 result = it->second.asInt();
46 return true;
47 } else if constexpr (std::is_same_v<T, double>) {
48 if (!it->second.isDouble() && !it->second.isInt64()) return false;
49 result = it->second.isDouble() ? it->second.asDouble() : static_cast<double>(it->second.asInt());
50 return std::isfinite(result);
51 } else {
52 static_assert(std::is_same_v<T, void>, "unsupported snapshot member type");
53 }
54}

56bool readCounter(const eve::Value::Object& object, std::string_view key, std::uint64_t& result) {
57 std::string encoded;
58 if (!readObjectMember(object, key, encoded) || encoded.empty()) return false;
59 const auto* first = encoded.data();
60 const auto* last = first + encoded.size();
61 std::uint64_t value = 0;
62 const auto parsed = std::from_chars(first, last, value);
63 if (parsed.ec != std::errc{} || parsed.ptr != last) return false;
64 result = value;
65 return true;
66}

103bool generatedOrderId(std::string_view id, std::uint64_t& number) {
104 constexpr std::string_view prefix = "order-";
105 if (!id.starts_with(prefix) || id.size() == prefix.size()) return false;
106 const auto digits = id.substr(prefix.size());
107 for (const char digit : digits)

109 const auto parsed = std::from_chars(digits.data(), digits.data() + digits.size(), number);
110 return parsed.ec == std::errc{} && parsed.ptr == digits.data() + digits.size() && number != 0;

178std::string CommandQueue::create(const std::string& kind, int priority, double timeoutSeconds, bool activateNow) {
179 if (kind.empty()) return {};
180 Order order;
181 std::ostringstream id;
182 id << "order-" << std::setw(16) << std::setfill('0') << nextId_++;
183 order.id = id.str();
184 order.kind = kind;
185 order.priority = priority;
186 order.timeoutSeconds = std::max(0.0, timeoutSeconds);
187 orders_.push_back(std::move(order));
188 const size_t index = orders_.size() - 1;
189 events_.push_back(
190 {{nextSequence_++, {}}, orders_[index].id, orders_[index].kind, OrderState::Queued, OrderState::Queued});
191 if (activateNow) {
192 active_ = index;
193 transition(orders_[index], OrderState::Active, {});
194 } else {
195 queued_.push_back(index);
196 std::stable_sort(queued_.begin(), queued_.end(),
197 [this](size_t a, size_t b) { return orders_[a].priority > orders_[b].priority; });
198 if (active_ == noIndex) activateNext();
199 }
200 return orders_[index].id;
201}

221bool CommandQueue::transition(Order& order, OrderState state, const std::string& reason) {
222 if (order.state == OrderState::Completed || order.state == OrderState::Failed ||
223 order.state == OrderState::Cancelled || order.state == state) {
224 return false;
225 }
226 const OrderState from = order.state;
227 order.state = state;
228 order.reason = reason;
229 events_.push_back({{nextSequence_++, reason}, order.id, order.kind, from, state});
230 return true;
231}

256bool CommandQueue::cancelUnchecked(const std::string& id, const std::string& reason) {
257 if (active_ != noIndex && orders_[active_].id == id) {
258 if (!transition(orders_[active_], OrderState::Cancelled, reason)) return false;
259 active_ = noIndex;
260 activateNext();
261 return true;
262 }
263 const auto it =
264 std::find_if(queued_.begin(), queued_.end(), [this, &id](size_t index) { return orders_[index].id == id; });
265 if (it == queued_.end()) return false;
266 const size_t index = *it;
267 queued_.erase(it);
268 return transition(orders_[index], OrderState::Cancelled, reason);
269}

412eve::Result<std::string> CommandQueue::snapshot() const {
413 eve::Value::Object root;
414 root.emplace("active", active_ == noIndex ? eve::Value() : eve::Value(static_cast<std::int64_t>(active_)));
415 root.emplace("nextId", eve::Value(std::to_string(nextId_)));
416 root.emplace("nextSequence", eve::Value(std::to_string(nextSequence_)));
417 root.emplace("schema", eve::Value(std::string(snapshotSchema)));
418 root.emplace("version", eve::Value(snapshotVersion));
419
420 eve::Value::Array queued;
421 queued.reserve(queued_.size());
422 for (const size_t index : queued_) {
423 if (index >= orders_.size() || index > static_cast<size_t>(std::numeric_limits<std::int64_t>::max()))
424 return snapshotFailure<std::string>("queue contains an invalid queued index", "queued");
425 queued.emplace_back(static_cast<std::int64_t>(index));
426 }
427 root.emplace("queued", eve::Value(std::move(queued)));
428
429 eve::Value::Array orders;
430 orders.reserve(orders_.size());
431 for (const auto& order : orders_) {
432 eve::Value::Object encoded;
433 encoded.emplace("elapsedSeconds", eve::Value(order.elapsedSeconds));
434 encoded.emplace("id", eve::Value(order.id));
435 encoded.emplace("kind", eve::Value(order.kind));
436 encoded.emplace("payload", eve::Value(order.payload.values_));
437 encoded.emplace("priority", eve::Value(order.priority));
438 encoded.emplace("reason", eve::Value(order.reason));
439 encoded.emplace("state", eve::Value(stateName(order.state)));
440 encoded.emplace("timeoutSeconds", eve::Value(order.timeoutSeconds));
441 orders.emplace_back(std::move(encoded));
442 }
443 root.emplace("orders", eve::Value(std::move(orders)));
444
445 eve::Value::Array events;
446 events.reserve(events_.size());
447 for (const auto& event : events_) {
448 eve::Value::Object encoded;
449 encoded.emplace("from", eve::Value(stateName(event.from)));
450 encoded.emplace("kind", eve::Value(event.kind));
451 encoded.emplace("orderId", eve::Value(event.orderId));
452 encoded.emplace("reason", eve::Value(event.reason));
453 encoded.emplace("sequence", eve::Value(std::to_string(event.sequence)));
454 encoded.emplace("to", eve::Value(stateName(event.to)));
455 events.emplace_back(std::move(encoded));
456 }
457 root.emplace("events", eve::Value(std::move(events)));
458
459 auto encoded = eve::Value(std::move(root)).toJson();
460 if (!encoded.ok()) return eve::Result<std::string>::failure(encoded.status());
461 return eve::Result<std::string>::success(std::move(encoded).takeValue());
462}

464eve::Result<void> CommandQueue::restore(std::string_view json) {
465 auto parsed = eve::Value::fromJson(json);
466 if (!parsed.ok()) return eve::Result<void>::failure(parsed.status());
467 eve::Value snapshotValue = std::move(parsed).takeValue();
468 const auto* root = snapshotValue.getIf<eve::Value::Object>();
469 if (!root) return snapshotFailure<void>("command queue snapshot must be an object");
470
471 std::string schema;
472 std::int64_t version = 0;
473 if (!readObjectMember(*root, "schema", schema) || schema != snapshotSchema ||
474 !readObjectMember(*root, "version", version) || version != snapshotVersion)
475 return snapshotFailure<void>("unsupported command queue schema or version", "schema");
476
477 const auto ordersIt = root->find("orders");
478 const auto queuedIt = root->find("queued");
479 const auto activeIt = root->find("active");
480 const auto eventsIt = root->find("events");
481 if (ordersIt == root->end() || queuedIt == root->end() || activeIt == root->end() || eventsIt == root->end() ||
482 !ordersIt->second.isArray() || !queuedIt->second.isArray() || !eventsIt->second.isArray())
483 return snapshotFailure<void>("command queue snapshot is missing required arrays");
484
485 CommandQueue candidate;
486 if (!readCounter(*root, "nextId", candidate.nextId_) ||
487 !readCounter(*root, "nextSequence", candidate.nextSequence_) || candidate.nextId_ == 0 ||
488 candidate.nextSequence_ == 0)
489 return snapshotFailure<void>("command queue snapshot has invalid counters", "counters");
490
491 const auto& encodedOrders = *ordersIt->second.getIf<eve::Value::Array>();
492 std::set<std::string> ids;
493 std::uint64_t largestId = 0;
494 for (const auto& encoded : encodedOrders) {
495 const auto* object = encoded.getIf<eve::Value::Object>();
496 if (!object) return snapshotFailure<void>("order entry must be an object", "orders");
497
498 Order order;
499 std::int64_t priority = 0;
500 if (!readObjectMember(*object, "id", order.id) || !readObjectMember(*object, "kind", order.kind) ||
501 !readObjectMember(*object, "priority", priority) || !readOrderState(*object, "state", order.state) ||
502 !readObjectMember(*object, "timeoutSeconds", order.timeoutSeconds) ||
503 !readObjectMember(*object, "elapsedSeconds", order.elapsedSeconds) ||
504 !readObjectMember(*object, "reason", order.reason))
505 return snapshotFailure<void>("order entry has an invalid field", "orders");
506 const auto payloadIt = object->find("payload");
507 if (payloadIt == object->end() || !payloadIt->second.isObject())
508 return snapshotFailure<void>("order payload must be an object", "orders.payload");
509 if (priority < std::numeric_limits<int>::min() || priority > std::numeric_limits<int>::max() ||
510 order.id.empty() || order.kind.empty() || order.timeoutSeconds < 0.0 || order.elapsedSeconds < 0.0 ||
511 !ids.insert(order.id).second)
512 return snapshotFailure<void>("order entry violates an invariant", "orders");
513
514 std::uint64_t idNumber = 0;
515 if (!generatedOrderId(order.id, idNumber) || idNumber >= candidate.nextId_)
516 return snapshotFailure<void>("order id does not match the queue counter", "orders.id");
517 largestId = std::max(largestId, idNumber);
518 order.priority = static_cast<int>(priority);
519 order.payload.values_ = *payloadIt->second.getIf<eve::Value::Object>();
520 candidate.orders_.push_back(std::move(order));
521 }
522 if (!candidate.orders_.empty() && largestId >= candidate.nextId_)
523 return snapshotFailure<void>("nextId must follow every restored order", "nextId");
524
525 std::vector<bool> queuedMembership(candidate.orders_.size(), false);
526 const auto& encodedQueued = *queuedIt->second.getIf<eve::Value::Array>();
527 for (const auto& encodedIndex : encodedQueued) {
528 size_t index = 0;
529 if (!readIndex(encodedIndex, candidate.orders_.size(), index) || queuedMembership[index] ||
530 candidate.orders_[index].state != OrderState::Queued)
531 return snapshotFailure<void>("queued membership violates an invariant", "queued");
532 if (!candidate.queued_.empty()) {
533 const size_t previous = candidate.queued_.back();
534 if (candidate.orders_[previous].priority < candidate.orders_[index].priority ||
535 (candidate.orders_[previous].priority == candidate.orders_[index].priority && previous > index))
536 return snapshotFailure<void>("queued order is not deterministically sorted", "queued");
537 }
538 queuedMembership[index] = true;
539 candidate.queued_.push_back(index);
540 }
541
542 if (activeIt->second.isNull()) {
543 candidate.active_ = noIndex;
544 } else if (!readIndex(activeIt->second, candidate.orders_.size(), candidate.active_) ||
545 candidate.orders_[candidate.active_].state != OrderState::Active) {
546 return snapshotFailure<void>("active order violates an invariant", "active");
547 }
548 for (size_t index = 0; index < candidate.orders_.size(); ++index) {
549 const auto state = candidate.orders_[index].state;
550 if ((state == OrderState::Queued) != queuedMembership[index] ||
551 (state == OrderState::Active) != (candidate.active_ == index))
552 return snapshotFailure<void>("order state and queue membership disagree", "orders");
553 }
554
555 const auto& encodedEvents = *eventsIt->second.getIf<eve::Value::Array>();
556 std::uint64_t previousSequence = 0;
557 for (const auto& encoded : encodedEvents) {
558 const auto* object = encoded.getIf<eve::Value::Object>();
559 if (!object) return snapshotFailure<void>("event entry must be an object", "events");
560 OrderEvent event;
561 std::string from;
562 std::string to;
563 if (!readCounter(*object, "sequence", event.sequence) || !readObjectMember(*object, "orderId", event.orderId) ||
564 !readObjectMember(*object, "kind", event.kind) || !readObjectMember(*object, "from", from) ||
565 !readObjectMember(*object, "to", to) || !readObjectMember(*object, "reason", event.reason) ||
566 !parseOrderState(from, event.from) || !parseOrderState(to, event.to) || event.sequence == 0 ||
567 event.sequence <= previousSequence || event.sequence >= candidate.nextSequence_ || event.kind.empty() ||
568 !ids.contains(event.orderId))
569 return snapshotFailure<void>("event entry violates an invariant", "events");
570 previousSequence = event.sequence;
571 candidate.events_.push_back(std::move(event));
572 }
573
574 *this = std::move(candidate);
575 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
576}
