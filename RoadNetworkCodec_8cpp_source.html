20bool withinBudget(const eve::Value& value, std::size_t depth, std::size_t& elements, std::size_t& bytes) {
21 if (depth > 64 || ++elements > 262144) return false;
22 if (const auto* text = value.getIf<std::string>()) bytes += text->size();
23 if (const auto* array = value.getIf<eve::Value::Array>()) {
24 if (array->size() > 262144 - elements) return false;
25 for (const auto& item : *array)
26 if (!withinBudget(item, depth + 1, elements, bytes)) return false;
27 }
28 if (const auto* object = value.getIf<eve::Value::Object>()) {
29 if (object->size() > 262144 - elements) return false;
30 for (const auto& [key, item] : *object) {
31 bytes += key.size();
32 if (bytes > 8u * 1024u * 1024u || !withinBudget(item, depth + 1, elements, bytes)) return false;
33 }
34 }
35 return bytes <= 8u * 1024u * 1024u;
36}
