44int ClassRegistry::loadFromJson(const std::string &json, std::string *error) {
45 std::string err;
46 const eve::json::Document doc = eve::json::Document::parse(json, &err);
47 if (!doc.valid()) {
48 if (error) *error = err.empty() ? "invalid json" : err;
49 return 0;
50 }
51 const Value root = doc.root();
52 int n = 0;
53 const auto parseOne = [&](Value o) {
54 if (!o.isObject()) return;
55 ClassDefinition def;
56 def.id = o.getString("id");
57 if (def.id.empty()) return;
58 def.displayName = o.getString("displayName");
59 def.tags = o.getStringArray("tags");
60 def.extra = o.getStringMap("extra");
61 def.traits = o.getStringArray("traits");
62 const Value learn = o.get("learnSkills");
63 for (size_t i = 0; i < learn.size(); ++i) {
64 const Value e = learn.at(i);
65 ClassLearnSkill ls;
66 ls.skillId = e.getString("skillId");
67 ls.level = e.getInt("level", 1);
68 if (ls.level <= 0) ls.level = 1;
69 if (ls.skillId.empty()) continue;
70 def.learnSkills.push_back(std::move(ls));
71 }
72 classTable()[def.id] = std::move(def);
73 ++n;
74 };
75 if (root.isArray()) {
76 for (size_t i = 0; i < root.size(); ++i) parseOne(root.at(i));
77 } else if (root.isObject()) {
78 parseOne(root);
79 }
80 return n;
81}

98bool ClassSystem::setClass(RPGActor *actor, const std::string &classId) {
99 if (!actor) return false;
100 const ClassDefinition *def = ClassRegistry::find(classId);
101 if (!def) return false;
102 auto &ci = *actor->classInfo();
103 // 撤销旧职业特征
104 if (!ci.classId.empty()) {
105 TraitSystem::removeBySource(actor, "class:" + ci.classId);
106 }
107 ci.classId = classId;
108 ci.skillsSyncedUpTo = 0;
109 // 施加新职业特征
110 for (const auto &traitId : def->traits) {
111 TraitSystem::apply(actor, traitId, "class:" + classId);
112 }
113 checkLevelSkills(actor);
114 return true;
115}

123int ClassSystem::checkLevelSkills(RPGActor *actor) {
124 if (!actor) return 0;
125 const ClassDefinition *def = ClassRegistry::find(actor->classInfo()->classId);
126 if (!def) return 0;
127 auto &ci = *actor->classInfo();
128 const int level = LevelSystem::getLevel(actor);
129 int learned = 0;
130 for (const auto &ls : def->learnSkills) {
131 if (ls.level <= ci.skillsSyncedUpTo) continue; // 已处理
132 if (ls.level <= level) {
133 if (!SkillSystem::knows(actor, ls.skillId)) {
134 SkillSystem::learn(actor, ls.skillId);
135 ++learned;
136 }
137 }
138 }
139 ci.skillsSyncedUpTo = std::max(ci.skillsSyncedUpTo, level);
140 return learned;
141}

197eve::Result<void> LevelSystem::restoreProgression(RPGActor *actor, int level, double xp, double xpToNext) {
198 if (!actor)
199 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
200 "progression restore requires an actor", "actor"));
201 if (level < 1)
202 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
203 "progression level must be positive", "level"));
204 if (!std::isfinite(xp) || xp < 0.0)

211 auto candidate = *actor->progression();
212 candidate.level = level;
213 candidate.xp = xp;
214 candidate.xpToNext = xpToNext;
215 *actor->progression() = candidate;
216 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
217}

226eve::Result<PreparedProgressionGain> LevelSystem::prepareGainXp(RPGActor *actor, double amount, double xpGrowth) {
227 if (!actor)
228 return eve::Result<PreparedProgressionGain>::failure(
229 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "XP gain requires an actor", "actor"));
230 if (!std::isfinite(amount) || amount <= 0.0)
231 return eve::Result<PreparedProgressionGain>::failure(eve::Diagnostic::error(
232 eve::DiagnosticCode::InvalidArgument, "XP gain must be finite and positive", "amount"));
233 if (!std::isfinite(xpGrowth) || xpGrowth < 1.0)
234 return eve::Result<PreparedProgressionGain>::failure(eve::Diagnostic::error(
235 eve::DiagnosticCode::InvalidArgument, "XP growth must be finite and at least one", "xpGrowth"));
236 const auto &current = *actor->progression();
237 if (current.level < 1 || !std::isfinite(current.xp) || current.xp < 0.0 || !std::isfinite(current.xpToNext) ||
238 current.xpToNext <= 0.0 || current.xp >= current.xpToNext)
239 return eve::Result<PreparedProgressionGain>::failure(eve::Diagnostic::error(
240 eve::DiagnosticCode::InvariantViolation, "actor progression is not valid for XP gain", "progression"));
241
242 PreparedProgressionGain prepared;
243 prepared.actor_ = actor;
244 prepared.previousLevel_ = current.level;
245 prepared.previousXp_ = current.xp;
246 prepared.previousXpToNext_ = current.xpToNext;
247 prepared.nextLevel_ = current.level;
248 prepared.nextXp_ = current.xp + amount;
249 prepared.nextXpToNext_ = current.xpToNext;
250 if (!std::isfinite(prepared.nextXp_))
251 return eve::Result<PreparedProgressionGain>::failure(eve::Diagnostic::error(
252 eve::DiagnosticCode::InvalidArgument, "XP gain would overflow progression", "amount"));
253 constexpr int maxLevelsPerGain = 10000;
254 int levels = 0;
255 while (prepared.nextXp_ >= prepared.nextXpToNext_) {
256 if (++levels > maxLevelsPerGain)
257 return eve::Result<PreparedProgressionGain>::failure(
258 eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
259 "XP gain exceeds the maximum levels allowed in one transaction", "amount"));
260 prepared.nextXp_ -= prepared.nextXpToNext_;
261 ++prepared.nextLevel_;
262 prepared.nextXpToNext_ *= xpGrowth;
263 if (!std::isfinite(prepared.nextXpToNext_) || prepared.nextXpToNext_ <= 0.0)
264 return eve::Result<PreparedProgressionGain>::failure(eve::Diagnostic::error(
265 eve::DiagnosticCode::InvalidArgument, "XP growth would overflow progression", "xpGrowth"));
266 }
267 return eve::Result<PreparedProgressionGain>::success(std::move(prepared),
268 eve::Status::success(eve::StatusCode::Pending));
269}

271eve::Result<int> LevelSystem::commitGainXp(PreparedProgressionGain prepared) {
272 if (!prepared.actor_)
273 return eve::Result<int>::failure(
274 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "prepared XP gain is empty", "prepared"));
275 auto &current = *prepared.actor_->progression();
276 if (current.level != prepared.previousLevel_ || current.xp != prepared.previousXp_ ||
277 current.xpToNext != prepared.previousXpToNext_)
278 return eve::Result<int>::failure(eve::Diagnostic::error(
279 eve::DiagnosticCode::Conflict, "actor progression changed after XP preparation", "progression"));
280 current.level = prepared.nextLevel_;
281 current.xp = prepared.nextXp_;
282 current.xpToNext = prepared.nextXpToNext_;
283 const int levels = prepared.nextLevel_ - prepared.previousLevel_;
284 for (int offset = 0; offset < levels; ++offset) {
285 LevelUpEvent event;
286 event.actor = prepared.actor_;
287 event.previousLevel = prepared.previousLevel_ + offset;
288 event.newLevel = event.previousLevel + 1;
289 levelUpEvents().push_back(std::move(event));
290 }
291 return eve::Result<int>::success(levels, eve::Status::success(eve::StatusCode::Applied));
292}
