39bool convexPolygon(const std::vector<float>& vertices) {
40 if (vertices.size() < 6 || vertices.size() > 16 || vertices.size() % 2 != 0) return false;
41 const size_t count = vertices.size() / 2;
42 double sign = 0.0;
43 for (size_t index = 0; index < count; ++index) {
44 const size_t next = (index + 1) % count;
45 const size_t after = (index + 2) % count;
46 const double ax = vertices[next * 2] - vertices[index * 2];
47 const double ay = vertices[next * 2 + 1] - vertices[index * 2 + 1];
48 const double bx = vertices[after * 2] - vertices[next * 2];
49 const double by = vertices[after * 2 + 1] - vertices[next * 2 + 1];
50 const double cross = ax * by - ay * bx;
51 if (std::abs(cross) <= 1e-8) return false;
52 if (sign == 0.0) sign = cross;
53 else if ((cross > 0.0) != (sign > 0.0)) return false;
54 }
55 return true;
56}
