23// Validate the backend's single-precision mass accumulation before its assertion-bearing ResetMassData.
24Result<void> validateMassChange(Fixture& changed, float density) {
25 const auto* body = changed.getBody()->raw();
26 if (body->GetType() != b2_dynamicBody) return Result<void>::success();
27 float mass = 0, inertia = 0;
28 b2Vec2 center(0, 0);
29 for (const auto* fixture = body->GetFixtureList(); fixture; fixture = fixture->GetNext()) {
30 const float candidateDensity = fixture == changed.raw() ? density : fixture->GetDensity();
31 if (candidateDensity == 0) continue;
32 b2MassData data;
33 fixture->GetShape()->ComputeMass(&data, candidateDensity);
34 mass += data.mass;
35 inertia += data.I;
36 center += data.mass * data.center;
37 if (!std::isfinite(mass) || !std::isfinite(inertia) || !center.IsValid() || data.mass < 0 || data.I < 0)
38 return invalid("Fixture density and geometry overflow body mass properties");
39 }
40 if (mass > 0) {
41 const float inverse = 1.f / mass;
42 if (!std::isfinite(inverse)) return invalid("Body inverse mass is not representable");
43 center *= inverse;
44 }
45 if (inertia > 0 && !body->IsFixedRotation()) {
46 const float centered = inertia - mass * b2Dot(center, center);
47 if (!std::isfinite(centered) || centered <= 0 || !std::isfinite(1.f / centered))
48 return invalid("Body rotational inertia is not representable");
49 }
50 return Result<void>::success();
51}
