16 std::vector<Vec2> o;

18 std::string p;

41 if (!ok(slow) || !ok(stop) || slow <= stop || stop < 0) return {};

46Vec2 Steering::separation(float x, float y, const std::string& n, float r, float m) {
47 if (r <= 0 || m <= 0) return {};
48 Vec2 sum{};
49 for (auto p : parse(n)) {
50 float dx = x - p.x, dy = y - p.y, d = std::hypot(dx, dy);
51 if (d > 0 && d < r) {
52 float w = (r - d) / (r * d);
53 sum.x += dx * w;
54 sum.y += dy * w;
55 }
56 }
57 float l = std::hypot(sum.x, sum.y);
58 return l > m ? scaled(sum.x, sum.y, m) : sum;
59}
60int Steering::pathTarget(float x, float y, const std::string& p, int cur, float tol) {
61 auto v = parse(p);
62 if (v.empty()) return -1;
63 cur = std::clamp(cur, 0, int(v.size() - 1));
64 while (cur + 1 < int(v.size()) && std::hypot(v[cur].x - x, v[cur].y - y) <= std::max(0.f, tol)) ++cur;
65 return cur;
66}
67Vec2 Steering::avoid(float x, float y, float vx, float vy, float ox, float oy, float radius, float ahead, float m) {
68 if (radius <= 0 || ahead < 0 || m <= 0) return {};
69 Vec2 dir = scaled(vx, vy, ahead);
70 float px = x + dir.x, py = y + dir.y;
71 if (std::hypot(px - ox, py - oy) > radius) return {};
72 return scaled(px - ox, py - oy, m);
73}
