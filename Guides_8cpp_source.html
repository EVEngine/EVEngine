20glm::vec3 sampleCurveAtU(const StrandsDatas &datas, size_t curveIndex, float u) {
21 const auto pts = datas.curvePoints(curveIndex);
22 if (pts.empty()) return {};
23 if (pts.size() == 1) return pts[0].position;
24 const float t = std::clamp(u, 0.f, 1.f);
25 // Prefer explicit per-point u; fall back to index parameterization.
26 size_t i1 = 1;
27 while (i1 + 1 < pts.size() && pts[i1].u < t) ++i1;
28 const size_t i0 = i1 - 1;
29 const float u0 = pts[i0].u;
30 const float u1 = pts[i1].u;
31 float alpha = 0.f;
32 if (std::fabs(u1 - u0) > 1e-8f) {
33 alpha = (t - u0) / (u1 - u0);
34 } else {
35 alpha = float(i1 - i0) > 0.f ? 1.f : 0.f;
36 }
37 alpha = std::clamp(alpha, 0.f, 1.f);
38 return pts[i0].position * (1.f - alpha) + pts[i1].position * alpha;
39}

97Result<std::vector<StrandGuideWeights>> buildGuideWeights(const StrandsDatas &strands, const StrandsDatas &guides,
98 int maxInfluences, InterpolationMode mode) {
99 auto sOk = strands.validate();
100 if (!sOk.ok()) return Result<std::vector<StrandGuideWeights>>::failure(sOk.status());
101 auto gOk = guides.validate();
102 if (!gOk.ok()) return Result<std::vector<StrandGuideWeights>>::failure(gOk.status());
103 if (guides.curveCount() == 0) {
104 return Result<std::vector<StrandGuideWeights>>::failure(
105 Diagnostic::error(DiagnosticCode::InvalidArgument, "buildGuideWeights: no guides", "hair.guides"));
106 }
107
108 const int k = std::clamp(maxInfluences, 1, StrandGuideWeights::kMaxInfluences);
109 const float exp = weightExponent(mode);
110 std::vector<StrandGuideWeights> out(strands.curveCount());
111
112 for (size_t si = 0; si < strands.curveCount(); ++si) {
113 const glm::vec3 root = rootOf(strands, si);
114 struct Cand {
115 uint32_t index;
116 float dist2;
117 };
118 std::vector<Cand> cands;
119 cands.reserve(guides.curveCount());
120 for (size_t gi = 0; gi < guides.curveCount(); ++gi) {
121 const glm::vec3 d = rootOf(guides, gi) - root;
122 cands.push_back({uint32_t(gi), glm::dot(d, d)});
123 }
124 const size_t take = std::min(size_t(k), cands.size());
125 std::partial_sort(cands.begin(), cands.begin() + std::ptrdiff_t(take), cands.end(),
126 [](const Cand &a, const Cand &b) { return a.dist2 < b.dist2; });
127
128 StrandGuideWeights row;
129 float wSum = 0.f;
130 for (size_t i = 0; i < take; ++i) {
131 const float dist = std::sqrt(std::max(cands[i].dist2, 0.f));
132 const float w = 1.f / std::pow(std::max(dist, 1e-4f), exp);
133 row.influencers[row.count] = {cands[i].index, w};
134 wSum += w;
135 ++row.count;
136 }
137 if (wSum > 0.f) {
138 for (int i = 0; i < row.count; ++i) row.influencers[i].weight /= wSum;
139 } else if (row.count > 0) {
140 row.influencers[0].weight = 1.f;
141 for (int i = 1; i < row.count; ++i) row.influencers[i].weight = 0.f;
142 }
143 out[si] = row;
144 }
145 return Result<std::vector<StrandGuideWeights>>::success(std::move(out));
146}

148Result<StrandsDatas> interpolateStrands(const StrandsDatas &strandsRest, const StrandsDatas &guidesRest,
149 const StrandsDatas &guidesDeformed,
150 const std::vector<StrandGuideWeights> &weights, InterpolationMode mode) {
151 auto sOk = strandsRest.validate();
152 if (!sOk.ok()) return Result<StrandsDatas>::failure(sOk.status());
153 auto g0 = guidesRest.validate();
154 if (!g0.ok()) return Result<StrandsDatas>::failure(g0.status());
155 auto g1 = guidesDeformed.validate();
156 if (!g1.ok()) return Result<StrandsDatas>::failure(g1.status());
157 if (guidesRest.curveCount() != guidesDeformed.curveCount()) {
158 return Result<StrandsDatas>::failure(
159 Diagnostic::error(DiagnosticCode::InvariantViolation,
160 "interpolateStrands: guide rest/deformed curve count mismatch", "hair.guides"));
161 }
162 if (weights.size() != strandsRest.curveCount()) {
163 return Result<StrandsDatas>::failure(Diagnostic::error(
164 DiagnosticCode::InvariantViolation, "interpolateStrands: weights size mismatch", "hair.guides.weights"));
165 }
166
167 std::vector<StrandPoint> points(strandsRest.points().begin(), strandsRest.points().end());
168 std::vector<StrandCurve> curves(strandsRest.curves().begin(), strandsRest.curves().end());
169
170 for (size_t si = 0; si < strandsRest.curveCount(); ++si) {
171 const StrandCurve &curve = curves[si];
172 const StrandGuideWeights &row = weights[si];
173 if (curve.pointCount == 0 || row.count <= 0) continue;
174
175 if (mode == InterpolationMode::Rigid) {
176 glm::vec3 delta(0.f);
177 for (int i = 0; i < row.count; ++i) {
178 const auto &inf = row.influencers[i];
179 const glm::vec3 rest = rootOf(guidesRest, inf.guideIndex);
180 const glm::vec3 def = rootOf(guidesDeformed, inf.guideIndex);
181 delta += (def - rest) * inf.weight;
182 }
183 for (uint32_t pi = 0; pi < curve.pointCount; ++pi) {
184 points[curve.pointOffset + pi].position += delta;
185 }
186 continue;
187 }
188
189 // Offset / Smooth: per-point guide deltas at matching u.
190 for (uint32_t pi = 0; pi < curve.pointCount; ++pi) {
191 StrandPoint &pt = points[curve.pointOffset + pi];
192 glm::vec3 delta(0.f);
193 for (int i = 0; i < row.count; ++i) {
194 const auto &inf = row.influencers[i];
195 const glm::vec3 restP = sampleCurveAtU(guidesRest, inf.guideIndex, pt.u);
196 const glm::vec3 defP = sampleCurveAtU(guidesDeformed, inf.guideIndex, pt.u);
197 delta += (defP - restP) * inf.weight;
198 }
199 pt.position += delta;
200 }
201 }
202
203 StrandsDatas out;
204 out.setPoints(std::move(points));
205 out.setCurves(std::move(curves));
206 auto valid = out.validate();
207 if (!valid.ok()) return Result<StrandsDatas>::failure(valid.status());
208 return Result<StrandsDatas>::success(std::move(out));
209}

221Result<void> GuideSimulator::reset(const StrandsDatas &guides, const GuideSimParams &params) {
222 auto ok = guides.validate();
223 if (!ok.ok()) return Result<void>::failure(ok.status());

226 "GuideSimulator::reset: empty guides", "hair.sim.guides"));
227 }
228
229 params_ = params;
230 params_.damping = std::clamp(params_.damping, 0.f, 1.f);
231 params_.iterations = std::clamp(params_.iterations, 1, 16);
232 params_.compliance = std::max(0.f, params_.compliance);
233 params_.maxDt = std::max(1e-4f, params_.maxDt);
234
235 restPoints_.assign(guides.points().begin(), guides.points().end());
236 curves_.clear();
237 curves_.reserve(guides.curveCount());
238 segments_.clear();
239 pos_.clear();
240 prev_.clear();
241 pinned_.clear();
242
243 pos_.reserve(guides.pointCount());
244 prev_.reserve(guides.pointCount());
245 pinned_.reserve(guides.pointCount());
246
247 for (size_t ci = 0; ci < guides.curveCount(); ++ci) {
248 const auto pts = guides.curvePoints(ci);
249 if (pts.size() < 2) {

251 "GuideSimulator::reset: guide curve needs >= 2 points",
252 "hair.sim.curve"));
253 }
254 CurveSpan span;
255 span.pointOffset = uint32_t(pos_.size());
256 span.pointCount = uint32_t(pts.size());
257 float len = 0.f;
258 for (size_t pi = 0; pi < pts.size(); ++pi) {
259 const glm::vec3 p = pts[pi].position;
260 pos_.push_back(p);
261 prev_.push_back(p);
262 pinned_.push_back(pi == 0 ? uint8_t(1) : uint8_t(0));
263 if (pi > 0) {
264 const float seg = glm::length(pts[pi].position - pts[pi - 1].position);
265 len += seg;
266 Segment s;
267 s.i0 = span.pointOffset + uint32_t(pi - 1);
268 s.i1 = span.pointOffset + uint32_t(pi);
269 s.restLength = std::max(seg, kEpsilon);
270 segments_.push_back(s);
271 }
272 }
273 span.length = len;
274 curves_.push_back(span);
275 }
276
277 return Result<void>::success();

325void GuideSimulator::solveDistances(float dt) {
326 const float dt2 = dt * dt;
327 const float alpha = params_.compliance / std::max(dt2, kEpsilon);
328
329 for (int it = 0; it < params_.iterations; ++it) {
330 for (const Segment &seg : segments_) {
331 glm::vec3 &p0 = pos_[seg.i0];
332 glm::vec3 &p1 = pos_[seg.i1];
333 const float w0 = pinned_[seg.i0] ? 0.f : 1.f;
334 const float w1 = pinned_[seg.i1] ? 0.f : 1.f;
335 const float wSum = w0 + w1;
336 if (wSum < kEpsilon) continue;
337
338 glm::vec3 delta = p1 - p0;
339 float len = glm::length(delta);
340 if (len < kEpsilon) {
341 delta = glm::vec3(0.f, kEpsilon, 0.f);
342 len = kEpsilon;
343 }
344 const glm::vec3 n = delta / len;
345 const float C = len - seg.restLength;
346 const float corr = C / (wSum + alpha);
347 p0 += n * (corr * w0);
348 p1 -= n * (corr * w1);
349 }
350 }

365Result<void> GuideSimulator::step(float dt) {
366 if (!isReady()) {
367 return Result<void>::failure(
368 Diagnostic::error(DiagnosticCode::InvalidArgument, "GuideSimulator::step: not ready", "hair.sim"));
369 }
370 if (!std::isfinite(dt) || dt <= 0.f) {
371 return Result<void>::failure(Diagnostic::error(
372 DiagnosticCode::InvalidArgument, "GuideSimulator::step: dt must be positive finite", "hair.sim.dt"));
373 }
374
375 const float stepDt = std::min(dt, params_.maxDt);
376 const float damp = std::pow(std::clamp(params_.damping, 0.f, 1.f), stepDt);
377 const glm::vec3 accel = params_.gravity + params_.wind;
378
379 for (size_t i = 0; i < pos_.size(); ++i) {
380 if (pinned_[i]) {
381 prev_[i] = pos_[i];
382 continue;
383 }
384 const glm::vec3 cur = pos_[i];
385 const glm::vec3 vel = (cur - prev_[i]) * damp;
386 pos_[i] = cur + vel + accel * (stepDt * stepDt);
387 prev_[i] = cur;
388 }
389
390 solveDistances(stepDt);
391 resolveCollisions();
392
393 for (size_t i = 0; i < pos_.size(); ++i) {
394 if (pinned_[i]) pos_[i] = prev_[i];
395 }
396 return Result<void>::success();
397}
