92Result<std::uint64_t> MeshAdhereLive::evaluateResult(bool force) {
93 if (!active_)
94 return Result<std::uint64_t>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
95 "mesh adhere session is not active", "evaluate", {},
96 "procgen.mesh.adhere"));
97 if (!dirty_ && !force && !derived_.empty()) return Result<std::uint64_t>::success(revision_);
98 auto blended = meshContactBlendAgainstSurfaceResult(source_, surface_, params_);
99 if (!blended.ok()) return Result<std::uint64_t>::failure(blended.status());
100 derived_ = std::move(blended).takeValue();
101 dirty_ = false;
102 ++revision_;
103 return Result<std::uint64_t>::success(revision_);
104}
