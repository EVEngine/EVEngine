83std::string quote(const std::string& value) {
84 std::ostringstream out;
85 out << '"';
86 for (unsigned char c : value) {
87 switch (c) {
88 case '"': out << "\\\""; break;
89 case '\\': out << "\\\\"; break;
90 case '\b': out << "\\b"; break;
91 case '\f': out << "\\f"; break;
92 case '\n': out << "\\n"; break;
93 case '\r': out << "\\r"; break;
94 case '\t': out << "\\t"; break;
95 default:
96 if (c < 0x20)
97 out << "\\u" << std::hex << std::setw(4) << std::setfill('0') << int(c) << std::dec;
98 else
99 out << static_cast<char>(c);
100 }
101 }
102 return out.str() + '"';
103}

128std::string canonicalJson(const eve::json::Value& value) {
129 if (value.isNull()) return "null";
130 if (value.isBool()) return value.asBool() ? "true" : "false";
131 if (value.isNumber()) return value.asString();
132 if (value.isString()) return quote(value.asString());
133 if (value.isArray()) {
134 std::string out = "[";
135 for (size_t i = 0; i < value.size(); ++i) {
136 if (i) out += ',';
137 out += canonicalJson(value.at(i));
138 }
139 return out + ']';
140 }
141 if (value.isObject()) {
142 auto keys = value.keys();
143 std::sort(keys.begin(), keys.end());
144 std::string out = "{";
145 for (size_t i = 0; i < keys.size(); ++i) {
146 if (i) out += ',';
147 out += quote(keys[i]) + ':' + canonicalJson(value.get(keys[i].c_str()));
148 }
149 return out + '}';
150 }
151 return {};
152}

260eve::Result<EventSequence> GameEventLog::append(GameEvent envelope) {
261 const auto failure = [](eve::DiagnosticCode code, std::string message) {
262 return eve::Result<EventSequence>::failure(eve::Diagnostic::error(code, std::move(message)));
263 };
264 if (envelope.type.empty()) return failure(eve::DiagnosticCode::InvalidArgument, "event type must not be empty");
265 if (!envelope.schemaId.isValid())
266 return failure(eve::DiagnosticCode::InvalidArgument, "event schemaId must be a valid logical ID");
267 if (envelope.schemaVersion.isZero())
268 return failure(eve::DiagnosticCode::InvalidArgument, "event schemaVersion must be non-zero");
269 auto payloadDocument = eve::json::Document::parse(envelope.payload);
270 if (!payloadDocument.valid()) return failure(eve::DiagnosticCode::ParseError, "payload must be valid JSON");
271 const auto next = nextSequence_.incremented();
272 if (!next) return failure(eve::DiagnosticCode::Failed, "event sequence exhausted");
273 if (!revision_.incremented()) return failure(eve::DiagnosticCode::Failed, "event stream revision exhausted");
274
275 if (envelope.eventId.isNil()) {
276 if (eventIdGenerator_) {
277 const auto generated = eventIdGenerator_->generate();
278 if (!generated) return failure(eve::DiagnosticCode::Failed, "event ID generation failed");
279 envelope.eventId = EventId::fromUuid(*generated);
280 } else {
281 return failure(eve::DiagnosticCode::InvalidArgument,
282 "eventId is required when no UUIDv7 generator is configured");
283 }
284 }
285 if (!envelope.eventId.isNil()) {
286 for (const auto& existing : events_) {
287 if (existing.eventId == envelope.eventId)
288 return failure(eve::DiagnosticCode::Conflict, "eventId already exists in this stream");
289 }
290 }
291
292 envelope.sequence = nextSequence_;
293 envelope.payload = canonicalJson(payloadDocument.root());
294 events_.push_back(std::move(envelope));
295 nextSequence_ = *next;
296 revision_ = *revision_.incremented();
297 if (events_.back().tick > snapshotTick_) snapshotTick_ = events_.back().tick;
298 return eve::Result<EventSequence>::success(events_.back().sequence);
299}

402eve::Result<void> GameEventLog::restore(std::string_view json) {
403 std::string error;
404 auto document = eve::json::Document::parse(std::string(json), &error);
405 const auto root = document.root();
406 const auto invalid = [&](eve::DiagnosticCode code, std::string message) {
407 return eve::Result<void>::failure(eve::Diagnostic::error(code, std::move(message)));
408 };
409 const int snapshotVersion = root.getInt("version");
410 if (!document.valid() || !root.isObject() || (snapshotVersion != 1 && snapshotVersion != 2)) {
411 return invalid(eve::DiagnosticCode::ParseError, error.empty() ? "invalid event stream snapshot" : error);
412 }
413 if (snapshotVersion == 2) {
414 auto schemaValidation = validateCurrentEventStream(json);
415 if (!schemaValidation.ok()) return schemaValidation;
416 }
417 uint64_t restoredNext = 0;
418 if (!parseU64(root.get("nextSequence"), restoredNext) || restoredNext == 0) {
419 return invalid(eve::DiagnosticCode::ParseError, "invalid nextSequence");
420 }
421 uint64_t restoredRevision = restoredNext - 1;
422 if (!root.get("revision").isNull() && !parseU64(root.get("revision"), restoredRevision))
423 return invalid(eve::DiagnosticCode::ParseError, "invalid revision");
424 const auto values = root.get("events");
425 if (!values.isArray()) {
426 return invalid(eve::DiagnosticCode::ParseError, "events must be an array");
427 }
428 std::deque<GameEvent> restored;
429 uint64_t previous = 0;
430 std::vector<std::string> restoredEventIds;
431 for (size_t i = 0; i < values.size(); ++i) {
432 const auto value = values.at(i);
433 GameEvent event;
434 int64_t restoredTick = 0;
435 uint64_t restoredSequence = 0;
436 if (!value.isObject() || !parseU64(value.get("sequence"), restoredSequence) || restoredSequence <= previous ||
437 !parseI64(value.get("tick"), restoredTick) || !value.get("flags").isNumber() || !value.get("payload")) {
438 return invalid(eve::DiagnosticCode::ParseError, "invalid event at index " + std::to_string(i));
439 }
440 event.sequence = EventSequence(restoredSequence);
441 if (restoredTick < 0) {
442 return invalid(eve::DiagnosticCode::ParseError, "invalid tick at index " + std::to_string(i));
443 }
444 event.tick = SimulationTick(static_cast<uint64_t>(restoredTick));
445 event.type = value.getString("type");
446 if (event.type.empty()) {
447 return invalid(eve::DiagnosticCode::ParseError, "event type must not be empty");
448 }
449 const std::string eventIdText = value.getString("eventId");
450 if (eventIdText.empty()) {
451 event.eventId = EventId::nil();
452 } else {
453 const auto parsedId = EventId::parse(eventIdText);
454 if (!parsedId)
455 return invalid(eve::DiagnosticCode::ParseError, "invalid eventId at index " + std::to_string(i));
456 event.eventId = *parsedId;
457 if (!event.eventId.isNil()) {
458 const auto formattedId = event.eventId.format();
459 if (std::find(restoredEventIds.begin(), restoredEventIds.end(), formattedId) != restoredEventIds.end())
460 return invalid(eve::DiagnosticCode::Conflict,
461 "duplicate non-nil eventId at index " + std::to_string(i));
462 restoredEventIds.push_back(formattedId);
463 }
464 }
465 event.source = value.getString("source");
466 event.subject = value.getString("subject");
467 const std::string causationText = value.getString("causation");
468 const std::string correlationText = value.getString("correlation");
469 const std::string causationKind =
470 snapshotVersion == 1 ? (causationText.empty() ? "none" : "legacy") : value.getString("causationKind");
471 const std::string correlationKind =
472 snapshotVersion == 1 ? (correlationText.empty() ? "none" : "legacy") : value.getString("correlationKind");
473 if (causationKind != "none" && causationKind != "event" && causationKind != "command" &&
474 causationKind != "legacy")
475 return invalid(eve::DiagnosticCode::ParseError, "invalid causation kind at index " + std::to_string(i));
476 if (correlationKind != "none" && correlationKind != "id" && correlationKind != "legacy")
477 return invalid(eve::DiagnosticCode::ParseError, "invalid correlation kind at index " + std::to_string(i));
478 if ((causationKind == "none" && !causationText.empty()) || (causationKind != "none" && causationText.empty()) ||
479 (correlationKind == "none" && !correlationText.empty()) ||
480 (correlationKind != "none" && correlationText.empty()))
481 return invalid(eve::DiagnosticCode::ParseError,
482 "causal reference value/kind mismatch at index " + std::to_string(i));
483 event.causation = causationFromSnapshot(causationKind, causationText);
484 event.correlation = correlationFromSnapshot(correlationKind, correlationText);
485 if ((causationKind == "event" || causationKind == "command") &&
486 event.causation.kind() == CausationRef::Kind::None)
487 return invalid(eve::DiagnosticCode::ParseError, "invalid causation ID at index " + std::to_string(i));
488 if (correlationKind == "id" && event.correlation.kind() == CorrelationId::Kind::None)
489 return invalid(eve::DiagnosticCode::ParseError, "invalid correlation ID at index " + std::to_string(i));
490 const std::string schemaIdText = value.getString("schemaId");
491 if (schemaIdText.empty()) {
492 event.schemaId = eve::LogicalId::fromParts("game_event", event.type).value_or(eve::LogicalId());
493 } else {
494 const auto parsedSchema = eve::LogicalId::parse(schemaIdText);
495 if (!parsedSchema)
496 return invalid(eve::DiagnosticCode::ParseError, "invalid schemaId at index " + std::to_string(i));
497 event.schemaId = *parsedSchema;
498 }
499 uint64_t restoredSchemaVersion = snapshotVersion == 1 ? 1 : 0;
500 if (snapshotVersion == 2 &&
501 (!parseU64(value.get("schemaVersion"), restoredSchemaVersion) || restoredSchemaVersion == 0)) {
502 return invalid(eve::DiagnosticCode::ParseError, "invalid schemaVersion at index " + std::to_string(i));
503 }
504 event.schemaVersion = SchemaVersion(restoredSchemaVersion);
505 const double flags = value.getDouble("flags", -1.0);
506 if (flags < 0 || flags > std::numeric_limits<uint32_t>::max() || std::floor(flags) != flags) {
507 return invalid(eve::DiagnosticCode::ParseError, "invalid flags at index " + std::to_string(i));
508 }
509 event.flags = static_cast<uint32_t>(flags);
510 event.payload = canonicalJson(value.get("payload"));
511 previous = event.sequence.value();
512 restored.push_back(std::move(event));
513 }
514 if (!restored.empty() && restored.back().sequence.value() >= restoredNext) {
515 return invalid(eve::DiagnosticCode::ParseError, "nextSequence must exceed retained events");
516 }
517 events_ = std::move(restored);
518 nextSequence_ = EventSequence(restoredNext);
519 revision_ = eve::Revision(restoredRevision);
520 snapshotTick_ = eve::SimulationTick::zero();
521 for (const auto& event : events_)
522 if (event.tick > snapshotTick_) snapshotTick_ = event.tick;
523 query_.clear();
524 consumers_.clear();
525 return eve::Result<void>::success();
526}

535eve::Result<void> GameEventLog::restoreSnapshot(const eve::SnapshotEnvelope& source,
536 const eve::SnapshotHashProvider& hashProvider) {
537 if (source.type != "game_event.stream" || source.schema != eventStreamSchema())
538 return snapshotFailure<void>(eve::DiagnosticCode::InvalidArgument,
539 "snapshot does not belong to game_event::GameEventLog");
540 if (!instanceId_.isNil() && source.instanceId != instanceId_)
541 return snapshotFailure<void>(eve::DiagnosticCode::Conflict,
542 "snapshot instanceId does not match game_event::GameEventLog");
543 auto migrated = eventStreamMigrations().migrate(source, eve::SchemaVersion(1), hashProvider);
544 if (!migrated.ok()) return eve::Result<void>::failure(migrated.status());
545 auto metadata = eve::validateSnapshotPayloadMetadata(migrated.value().payload, migrated.value().revision,
546 migrated.value().tick);
547 if (!metadata.ok()) return eve::Result<void>::failure(metadata.status());
548 auto payload = migrated.value().payload.toJson();
549 if (!payload.ok()) return eve::Result<void>::failure(payload.status());
550
551 GameEventLog candidate(instanceId_);
552 candidate.eventIdGenerator_ = eventIdGenerator_;
553 auto restored = candidate.restore(std::move(payload).takeValue());
554 if (!restored.ok()) return eve::Result<void>::failure(restored.status());
555 if (candidate.snapshotTick_ != migrated.value().tick)
556 return snapshotFailure<void>(eve::DiagnosticCode::Conflict,
557 "event stream payload tick disagrees with snapshot envelope");
558 candidate.instanceId_ = migrated.value().instanceId;
559 candidate.revision_ = migrated.value().revision;
560 candidate.snapshotTick_ = migrated.value().tick;
561 // The UUID generator is a runtime capability, not persisted state. Keep
562 // it across the atomic payload replacement and invalidate old cursors.
563 eventIdGenerator_ = std::move(candidate.eventIdGenerator_);
564 instanceId_ = candidate.instanceId_;
565 revision_ = candidate.revision_;
566 snapshotTick_ = candidate.snapshotTick_;
567 nextSequence_ = candidate.nextSequence_;
568 events_ = std::move(candidate.events_);
569 query_.clear();
570 consumers_.clear();
571 return eve::Result<void>::success();
572}
