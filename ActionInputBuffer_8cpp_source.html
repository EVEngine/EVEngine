41Result<BufferedInput> ActionInputBuffer::peek(SubjectRef subject, const std::vector<std::string>& allows) const {
42 const BufferedInput* best = nullptr;
43 for (const auto& entry : entries_) {
44 if (entry.subject != subject) continue;
45 if (!matchesAllow(entry.input, allows)) continue;
46 if (!best || entry.priority > best->priority ||
47 (entry.priority == best->priority && entry.tick.value() < best->tick.value()))
48 best = &entry;
49 }
50 if (!best)
51 return Result<BufferedInput>::failure(
52 Diagnostic::error(DiagnosticCode::NotFound, "no buffered input matched", "input"));
53 return Result<BufferedInput>::success(*best);
54}

56Result<BufferedInput> ActionInputBuffer::consume(SubjectRef subject, const std::vector<std::string>& allows) {
57 auto peeked = peek(subject, allows);
58 if (!peeked) return peeked;
59 for (auto it = entries_.begin(); it != entries_.end(); ++it) {
60 if (it->subject == peeked.value().subject && it->input == peeked.value().input &&
61 it->tick == peeked.value().tick && it->priority == peeked.value().priority) {
62 BufferedInput value = *it;
63 entries_.erase(it);
64 return Result<BufferedInput>::success(std::move(value), Status::success(StatusCode::Applied));
65 }
66 }
67 return Result<BufferedInput>::failure(
68 Diagnostic::error(DiagnosticCode::NotFound, "buffered input disappeared", "input"));
69}
