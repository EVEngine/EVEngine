35inline eve::Result<std::uint64_t> runtimeSnapshotUint(const eve::Value& value, std::string_view path) {
36 const auto text = value.getIf<std::string>();
37 if (text == nullptr || text->empty())
38 return eve::Result<std::uint64_t>::failure(
39 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "runtime snapshot integer must be a decimal string",
40 std::string(path), {}, "common.definitions.snapshot"));
41 std::uint64_t result = 0;
42 const auto [end, error] = std::from_chars(text->data(), text->data() + text->size(), result);
43 if (error != std::errc{} || end != text->data() + text->size())
44 return eve::Result<std::uint64_t>::failure(
45 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "runtime snapshot integer is out of range",
46 std::string(path), {}, "common.definitions.snapshot"));
47 return eve::Result<std::uint64_t>::success(result);
48}

50inline eve::Result<void> runtimeSnapshotShape(const eve::Value::Object& object) {
51 static const std::set<std::string> fields = {"active", "definition", "definitionGeneration", "instanceId", "state"};
52 for (const auto& [name, unused] : object) {
53 (void)unused;
54 if (!fields.contains(name))
55 return eve::Result<void>::failure(eve::Diagnostic::error(
56 eve::DiagnosticCode::ParseError, "runtime snapshot payload contains an unknown field", "state." + name,
57 {}, "common.definitions.snapshot"));
58 }
59 for (const auto& name : fields) {
60 if (!object.contains(name))
61 return eve::Result<void>::failure(eve::Diagnostic::error(
62 eve::DiagnosticCode::ParseError, "runtime snapshot payload is missing a required field",

64 }

84 const RuntimeInstance<State>& runtime, std::string type, eve::LogicalId schema, eve::SchemaVersion schemaVersion,
85 eve::Revision revision, eve::SimulationTick tick, const eve::SnapshotHashProvider& hashProvider,
86 const RuntimeStateEncoder<State>& encode) {
87 if (!encode)
88 return eve::Result<eve::SnapshotEnvelope>::failure(
89 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "runtime snapshot encoder is required",
90 "encode", {}, "common.definitions.snapshot"));
91 auto encoded = encode(runtime.state());
92 if (!encoded) return eve::Result<eve::SnapshotEnvelope>::failure(encoded.status());
93 eve::Value::Object payload;
94 payload.emplace("active", eve::Value(runtime.isActive()));
95 payload.emplace("definition", eve::Value(runtime.identity().definition.format()));
96 payload.emplace("definitionGeneration",
97 eve::Value(std::to_string(runtime.identity().definitionGeneration.value())));
98 payload.emplace("instanceId", eve::Value(runtime.identity().instanceId.format()));
99 payload.emplace("state", std::move(encoded).takeValue());
100 return eve::makeSnapshotEnvelope(std::move(type), std::move(schema), schemaVersion, runtime.identity().instanceId,
101 revision, tick, eve::Value(std::move(payload)), hashProvider);
102}

116template <class State>
117[[nodiscard]] eve::Result<void> restoreRuntimeInstance(
118 RuntimeInstance<State>& runtime, const eve::SnapshotEnvelope& snapshot, std::string_view expectedType,
119 const eve::LogicalId& expectedSchema, eve::SchemaVersion expectedVersion,
120 const eve::SnapshotHashProvider& hashProvider, const RuntimeStateDecoder<State>& decode) {
121 if (snapshot.type != expectedType || snapshot.schema != expectedSchema)
122 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
123 "runtime snapshot type or schema does not match",
124 "schema", {}, "common.definitions.snapshot"));
125 if (snapshot.schemaVersion != expectedVersion)
126 return eve::Result<void>::failure(eve::Diagnostic::error(
127 snapshot.schemaVersion > expectedVersion ? eve::DiagnosticCode::UnknownVersion
128 : eve::DiagnosticCode::Unsupported,
129 "runtime snapshot schema version is not supported", "schemaVersion", {}, "common.definitions.snapshot"));
130 auto verified = eve::verifySnapshotEnvelope(snapshot, hashProvider);
131 if (!verified) return verified;
132 auto metadata = eve::validateSnapshotPayloadMetadata(snapshot.payload, snapshot.revision, snapshot.tick);
133 if (!metadata) return metadata;
134 const auto object = snapshot.payload.getIf<eve::Value::Object>();
135 if (object == nullptr) {
136 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::ParseError,
137 "runtime snapshot payload must be an object",
138 "payload", {}, "common.definitions.snapshot"));
139 }
140 auto shape = detail::runtimeSnapshotShape(*object);
141 if (!shape) return shape;
142
143 const auto active = object->at("active").getIf<bool>();
144 const auto definitionText = object->at("definition").getIf<std::string>();
145 const auto instanceText = object->at("instanceId").getIf<std::string>();
146 if (active == nullptr || definitionText == nullptr || instanceText == nullptr)
147 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::ParseError,
148 "runtime snapshot identity fields have invalid types",
149 "payload", {}, "common.definitions.snapshot"));
150 const auto instanceId = eve::PersistentId::parse(*instanceText);
151 if (!instanceId)
152 return eve::Result<void>::failure(
153 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "runtime snapshot instanceId is invalid",
154 "payload.instanceId", {}, "common.definitions.snapshot"));
155 auto definition = eve::DefinitionRef::parse(*definitionText);
156 if (!definition) return eve::Result<void>::failure(definition.status());
157 auto generation = detail::runtimeSnapshotUint(object->at("definitionGeneration"), "payload.definitionGeneration");
158 if (!generation) return eve::Result<void>::failure(generation.status());
159 const eve::definition::InstanceIdentity identity{*instanceId, std::move(definition).takeValue(),
160 eve::Generation(std::move(generation).takeValue())};
161 if (snapshot.instanceId != identity.instanceId || identity.instanceId != runtime.identity().instanceId)
162 return eve::Result<void>::failure(eve::Diagnostic::error(
163 eve::DiagnosticCode::Conflict, "runtime snapshot instanceId does not match the destination",
164 "payload.instanceId", {}, "common.definitions.snapshot"));
165 if (identity.definition != runtime.identity().definition)
166 return eve::Result<void>::failure(eve::Diagnostic::error(
167 eve::DiagnosticCode::Conflict, "runtime snapshot definition reference does not match", "payload.definition",
168 {}, "common.definitions.snapshot"));
169 if (identity.definitionGeneration != runtime.identity().definitionGeneration)
170 return eve::Result<void>::failure(eve::Diagnostic::error(
171 eve::DiagnosticCode::StaleHandle, "runtime snapshot definition generation does not match the destination",
172 "payload.definitionGeneration", {}, "common.definitions.snapshot"));
173 if (!decode)
174 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
175 "runtime snapshot decoder is required", "decode", {},
176 "common.definitions.snapshot"));
177 auto candidate = decode(object->at("state"));
178 if (!candidate) return eve::Result<void>::failure(candidate.status());
179 return runtime.restoreExact(identity, std::move(candidate).takeValue(), *active);
180}
