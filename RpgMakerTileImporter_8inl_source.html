37TileLayer::Tileset::Visual rpgMakerVisual(int gid) {
38 TileLayer::Tileset::Visual visual;
39 visual.gid = gid;
40 constexpr int tile = 48, quarter = 24, a1 = 2048, a2 = 2816, a3 = 4352, a4 = 5888;
41 if (gid < a1) {
42 visual.x = ((gid / 128 % 2) * 8 + gid % 8) * tile;
43 visual.y = ((gid % 256 / 8) % 16) * tile;
44 visual.width = tile;
45 visual.height = tile;
46 return visual;
47 }
48 const int kind = (gid - a1) / 48;
49 const int shape = (gid - a1) % 48;
50 const int tx = kind % 8, ty = kind / 8;
51 int bx = 0, by = 0;
52 const Quarter *table = nullptr;
53 int frameCount = 1;
54 bool waterSurface = false, waterfall = false;
55 if (gid < a2) {
56 if (kind == 0) {
57 bx = 0;
58 by = 0;
59 waterSurface = true;
60 } else if (kind == 1) {
61 bx = 0;
62 by = 3;
63 waterSurface = true;
64 } else if (kind == 2) {
65 bx = 6;
66 by = 0;
67 } else if (kind == 3) {
68 bx = 6;
69 by = 3;
70 } else {
71 bx = (tx / 4) * 8;
72 by = ty * 6 + (tx / 2 % 2) * 3;
73 if (kind % 2 == 0)
74 waterSurface = true;
75 else {
76 bx += 6;
77 waterfall = true;
78 }
79 }
80 if (waterSurface) frameCount = 4;
81 if (waterfall) frameCount = 3;
82 table = waterfall ? &kRpgWaterfall[size_t(shape % 4)] : &kRpgFloor[size_t(shape)];
83 } else if (gid < a3) {
84 bx = tx * 2;
85 by = (ty - 2) * 3;
86 table = &kRpgFloor[size_t(shape)];
87 } else if (gid < a4) {
88 bx = tx * 2;
89 by = (ty - 6) * 2;
90 table = &kRpgWall[size_t(shape % 16)];
91 } else {
92 bx = tx * 2;
93 by = int((ty - 10) * 2.5f + (ty % 2 == 1 ? 0.5f : 0.f));
94 table = ty % 2 == 1 ? &kRpgWall[size_t(shape % 16)] : &kRpgFloor[size_t(shape)];
95 }
96 const int waterIndices[4] = {0, 1, 2, 1};
97 for (int frame = 0; frame < frameCount; ++frame) {
98 TileLayer::Tileset::Visual::SubtileFrame output;
99 output.durationMs = 500;
100 int frameBx = bx, frameBy = by;
101 if (waterSurface) frameBx += waterIndices[frame] * 2;
102 if (waterfall) frameBy += frame;
103 for (int part = 0; part < 4; ++part) {
104 output.parts.push_back({(frameBx * 2 + (*table)[size_t(part * 2)]) * quarter,
105 (frameBy * 2 + (*table)[size_t(part * 2 + 1)]) * quarter, quarter, quarter,
106 float((part % 2) * quarter), float((part / 2) * quarter)});
107 }
108 visual.subtileFrames.push_back(std::move(output));
109 }
110 return visual;
111}
