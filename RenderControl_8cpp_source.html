94Result<void> RenderControl::setLightingMode(LightingMode mode) {
95 if (mode != LightingMode::ForwardPlus && mode != LightingMode::Hybrid) {
96 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "unknown LightingMode value",
97 "graphics.RenderControl.lightingMode"));
98 }
99 if (lightingMode_ == mode) return Result<void>::success();
100 lightingMode_ = mode;
101 syncClusteredDeferredFeature();
102 if (mode == LightingMode::Hybrid) {
103 features_["gbuffer"] = true;
104 features_["gbufferAlbedo"] = true;
105 }
106 dirty_ = true;
107 return Result<void>::success();
108}

142void RenderControl::setFeature(const std::string &feature, bool enabled) {
143 if (!isKnownFeature(feature)) return;
144
145 if (feature == "clusteredDeferred") {
146 auto setMode = setLightingMode(enabled ? LightingMode::Hybrid : LightingMode::ForwardPlus);
147 setMode.ignore();
148 return;
149 }
150
151 auto it = features_.find(feature);
152 const bool cur = it == features_.end() ? false : it->second;
153 if (cur == enabled) return;
154 features_[feature] = enabled;
155 if (gfx_ && (feature == "ssr" || feature == "rtgi" || feature == "reflectionChain" || feature == "rtx")) {
156 gfx_->pipelineScreenSpaceReflection()->invalidateHistory();
157 gfx_->pipelineGlobalIllumination()->invalidateHistory();
158 }
159 if (gfx_ && (feature == "taa" || feature == "reflectionChain"))
160 gfx_->pipelineAntiAliasing()->invalidateTemporalHistory();
161 if (feature == "gbufferAlbedo" && enabled) features_["gbuffer"] = true;
162 if (feature == "ao" && enabled) features_["gbuffer"] = true;
163 if (feature == "outline" && enabled) features_["gbuffer"] = true;
164 if (feature == "decal" && enabled) features_["gbuffer"] = true;
165 // Stage 2 GPU cull needs the GBuffer depth as its HZB source.
166 if (feature == "gpuDriven" && enabled) features_["gbuffer"] = true;
167 if ((feature == "ssr" || feature == "rtgi" || feature == "reflectionChain" || feature == "rtx") && enabled)
168 features_["gbuffer"] = true;
169 if (feature == "rtx" && enabled) {
170 // Prefer hardware RT when available; keep SSR as the portable path
171 // that RenderSystem3D can still run when IRayTracing is absent.
172 features_["gbuffer"] = true;
173 }
174 if (feature == "reflectionChain" && enabled) {
175 features_["aa"] = true;
176 features_["taa"] = true;
177 features_["msaa"] = false;
178 features_["rtgi"] = true;
179 features_["ssr"] = true;
180 }
181 if (feature == "reflectionChain" && !enabled) {
182 features_["taa"] = false;
183 features_["rtgi"] = false;
184 features_["ssr"] = false;
185 if (gfx_) gfx_->pipelineScreenSpaceReflection()->setEnabled(false);
186 }
187 if ((feature == "taa" || feature == "rtgi" || feature == "ssr") && !enabled)
188 features_["reflectionChain"] = false;
189 if (feature == "taa" && enabled) {
190 features_["aa"] = true;
191 features_["msaa"] = false;
192 }
193 if ((feature == "volumetricFog" || feature == "fogLocalVolumes" ||
194 feature == "fogTemporal") && enabled) {
195 features_["atmosphere"] = true;
196 features_["volumetricFog"] = true;
197 features_["gbuffer"] = true;
198 }
199 if (feature == "volumetricFog" && !enabled) {
200 features_["fogLocalVolumes"] = false;
201 features_["fogTemporal"] = false;
202 }
203 if (feature == "gi" && enabled) {
204 features_["gbuffer"] = true;
205 features_["gbufferAlbedo"] = true;
206 }
207 if (feature == "gbuffer" && !enabled) {
208 features_["gbufferAlbedo"] = false;
209 features_["ao"] = false;
210 features_["gi"] = false;
211 features_["rtgi"] = false;
212 features_["ssr"] = false;
213 features_["rtx"] = false;
214 features_["reflectionChain"] = false;
215 features_["outline"] = false;
216 features_["decal"] = false;
217 // Hybrid needs GBuffer; keep the request but compile() will fall back.
218 dirty_ = true;
219 return;
220 }
221 dirty_ = true;
222}

241void RenderControl::compile() {
242 passes_.clear();
243
244 const bool wantHybrid = lightingMode_ == LightingMode::Hybrid;
245 const bool gbufferOn = isEnabled("gbuffer") || isEnabled("gbufferAlbedo") || isEnabled("ao") || isEnabled("gi");
246 const bool canHybrid = wantHybrid && isDeferredLightingAvailable() && gbufferOn;
247 effectiveLightingMode_ = canHybrid ? LightingMode::Hybrid : LightingMode::ForwardPlus;
248 hybridLightingFallback_ = wantHybrid && !canHybrid;
249
250 // Phase C: deferred lighting samples 1x GBuffer into scene color. MSAA scene
251 // color would need resolve-before-light; force MSAA off while Hybrid is live.
252 if (canHybrid) features_["msaa"] = false;
253
254 if (isEnabled("shadow")) passes_.push_back("shadow");
255 if (gbufferOn || canHybrid) passes_.push_back("gbuffer");
256 if (isEnabled("decal")) passes_.push_back("decal");
257 if (canHybrid) passes_.push_back("deferredLighting");
258 if (isEnabled("forward")) passes_.push_back("forward");
259 if (isEnabled("atmosphere")) passes_.push_back("atmosphere");
260 if (isEnabled("volumetricFog")) {
261 passes_.push_back("fogMedia");
262 passes_.push_back("fogLighting");
263 if (isEnabled("fogTemporal")) passes_.push_back("fogTemporal");
264 passes_.push_back("fogIntegrate");
265 passes_.push_back("fogComposite");
266 }
267 if (isEnabled("hair")) passes_.push_back("hair");
268 dirty_ = false;
269 compiled_ = true;
270}
