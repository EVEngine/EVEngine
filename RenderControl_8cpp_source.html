61void RenderControl::setFeature(const std::string &feature, bool enabled) {
62 if (!isKnownFeature(feature)) return;
63 auto it = features_.find(feature);
64 const bool cur = it == features_.end() ? false : it->second;
65 if (cur == enabled) return;
66 features_[feature] = enabled;
67 if (gfx_ && (feature == "ssr" || feature == "rtgi" || feature == "reflectionChain")) {
68 gfx_->pipelineScreenSpaceReflection()->invalidateHistory();
69 gfx_->pipelineGlobalIllumination()->invalidateHistory();
70 }
71 if (gfx_ && (feature == "taa" || feature == "reflectionChain"))
72 gfx_->pipelineAntiAliasing()->invalidateTemporalHistory();
73 if (feature == "gbufferAlbedo" && enabled) features_["gbuffer"] = true;
74 if (feature == "ao" && enabled) features_["gbuffer"] = true;
75 if (feature == "outline" && enabled) features_["gbuffer"] = true;
76 if (feature == "decal" && enabled) features_["gbuffer"] = true;
77 // Stage 2 GPU cull needs the GBuffer depth as its HZB source.
78 if (feature == "gpuDriven" && enabled) features_["gbuffer"] = true;
79 if ((feature == "ssr" || feature == "rtgi" || feature == "reflectionChain") && enabled)
80 features_["gbuffer"] = true;
81 if (feature == "reflectionChain" && enabled) {
82 features_["aa"] = true;
83 features_["taa"] = true;
84 features_["msaa"] = false;
85 features_["rtgi"] = true;
86 features_["ssr"] = true;
87 }
88 if (feature == "reflectionChain" && !enabled) {
89 features_["taa"] = false;
90 features_["rtgi"] = false;
91 features_["ssr"] = false;
92 if (gfx_) gfx_->pipelineScreenSpaceReflection()->setEnabled(false);
93 }
94 if ((feature == "taa" || feature == "rtgi" || feature == "ssr") && !enabled)
95 features_["reflectionChain"] = false;
96 if (feature == "taa" && enabled) {
97 features_["aa"] = true;
98 features_["msaa"] = false;
99 }
100 if ((feature == "volumetricFog" || feature == "fogLocalVolumes" ||
101 feature == "fogTemporal") && enabled) {
102 features_["atmosphere"] = true;
103 features_["volumetricFog"] = true;
104 features_["gbuffer"] = true;
105 }
106 if (feature == "volumetricFog" && !enabled) {
107 features_["fogLocalVolumes"] = false;
108 features_["fogTemporal"] = false;
109 }
110 if (feature == "gi" && enabled) {
111 features_["gbuffer"] = true;
112 features_["gbufferAlbedo"] = true;
113 }
114 if (feature == "gbuffer" && !enabled) {
115 features_["gbufferAlbedo"] = false;
116 features_["ao"] = false;
117 features_["gi"] = false;
118 features_["rtgi"] = false;
119 features_["ssr"] = false;
120 features_["reflectionChain"] = false;
121 features_["outline"] = false;
122 features_["decal"] = false;
123 }
124 dirty_ = true;
125}

143void RenderControl::compile() {
144 passes_.clear();
145 if (isEnabled("shadow")) passes_.push_back("shadow");
146 if (isEnabled("gbuffer") || isEnabled("gbufferAlbedo") || isEnabled("ao") || isEnabled("gi"))
147 passes_.push_back("gbuffer");
148 if (isEnabled("decal")) passes_.push_back("decal");
149 if (isEnabled("forward")) passes_.push_back("forward");
150 if (isEnabled("atmosphere")) passes_.push_back("atmosphere");
151 if (isEnabled("volumetricFog")) {
152 passes_.push_back("fogMedia");
153 passes_.push_back("fogLighting");
154 if (isEnabled("fogTemporal")) passes_.push_back("fogTemporal");
155 passes_.push_back("fogIntegrate");
156 passes_.push_back("fogComposite");
157 }
158 if (isEnabled("hair")) passes_.push_back("hair");
159 dirty_ = false;
160 compiled_ = true;
161}
