46void RenderControl::setFeature(const std::string &feature, bool enabled) {
47 if (!isKnownFeature(feature)) return;
48 auto it = features_.find(feature);
49 const bool cur = it == features_.end() ? false : it->second;
50 if (cur == enabled) return;
51 features_[feature] = enabled;
52 if (feature == "gbufferAlbedo" && enabled) features_["gbuffer"] = true;
53 if (feature == "ao" && enabled) features_["gbuffer"] = true;
54 if (feature == "outline" && enabled) features_["gbuffer"] = true;
55 // Stage 2 GPU cull needs the GBuffer depth as its HZB source.
56 if (feature == "gpuDriven" && enabled) features_["gbuffer"] = true;
57 if (feature == "gi" && enabled) {
58 features_["gbuffer"] = true;
59 features_["gbufferAlbedo"] = true;
60 }
61 if (feature == "gbuffer" && !enabled) {
62 features_["gbufferAlbedo"] = false;
63 features_["ao"] = false;
64 features_["gi"] = false;
65 features_["outline"] = false;
66 }
67 dirty_ = true;
68}
