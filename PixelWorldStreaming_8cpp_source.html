21eve::Result<PixelChunkStreamUpdate> PixelChunkStreamCursor::capture(
22 const eve::pixelworld::PixelWorld& source, eve::pixelworld::PixelChunkRegion interest) {
23 auto all = source.snapshotChunksInRegion(interest, 0);
24 if (!all) return eve::Result<PixelChunkStreamUpdate>::failure(all.status());
25
26 const auto link = source.worldLink();
27 const bool fullResync = source_.world == 0 || source_ != link;
28 std::set<Coord> nextKnown;
29 std::set<Coord> emitted;
30 std::vector<eve::pixelworld::PixelChunkSnapshot> output;
31 output.reserve(all.value().size() + known_.size());
32 PixelChunkStreamUpdate update;
33 update.fullResync = fullResync;
34
35 for (const auto& chunk : all.value()) {
36 const Coord key = coord(chunk);
37 if (chunk.removed) {
38 if (fullResync || known_.contains(key)) {
39 output.push_back(chunk);
40 emitted.insert(key);
41 }
42 continue;
43 }
44 nextKnown.insert(key);
45 if (fullResync || !known_.contains(key) || chunk.revision > revision_) {
46 output.push_back(chunk);
47 emitted.insert(key);
48 if (!known_.contains(key)) ++update.chunksEntered;
49 }
50 }
51
52 for (const Coord& key : known_) {
53 if (nextKnown.contains(key) || emitted.contains(key)) continue;
54 ++update.chunksEvicted;
55 if (fullResync || contains(interest, key)) continue;
56 const auto [y, x] = key;
57 output.push_back({x, y, source.revision(), true, {}});
58 }
59 std::sort(output.begin(), output.end(), [](const auto& left, const auto& right) {
60 return coord(left) < coord(right);
61 });
62
63 update.batch.catalogFingerprint = source.materialCatalogFingerprint();
64 update.batch.sourceSeed = source.seed();
65 update.batch.sourceRevision = source.revision();
66 update.batch.sourceTick = eve::SimulationTick(source.tickValue());
67 update.batch.sourceLastEditSequence = source.lastEditSequence();
68 update.batch.fullResync = fullResync;
69 update.batch.chunks = std::move(output);
70
71 source_ = link;
72 revision_ = source.revision();
73 known_ = std::move(nextKnown);
74 return eve::Result<PixelChunkStreamUpdate>::success(std::move(update));
75}
