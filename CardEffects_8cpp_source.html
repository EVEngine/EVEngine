56 eve::Result<settlement::PreparedApply> prepareApply(const settlement::SettlementContext& context) override {
57 const CardEffectTarget before = target_;
58 CardEffectTarget candidate = target_;
59 const int applied = static_cast<int>(context.magnitude());
60 const int absorbed = static_cast<int>(barrierAbsorbed_);
61 if (context.request().kind == "damage") {
62 candidate.barrier -= absorbed;
63 candidate.health = std::max(0, candidate.health - applied);
64 if (candidate.health == 0 && before.health > 0) ++candidate.deathTriggers;
65 } else if (context.request().kind == "healing") {
66 candidate.health = std::min(candidate.maxHealth, candidate.health + applied);
67 } else {
68 candidate.barrier += applied;
69 }
70 return eve::Result<settlement::PreparedApply>::success(settlement::PreparedApply(
71 [this, candidate]() {
72 target_ = candidate;
73 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
74 },
75 [this, before]() { target_ = before; }));
76 }

78 eve::Result<std::vector<settlement::SettlementRequest>> prepareTrigger(
79 const settlement::SettlementContext& context, const settlement::SettlementResult& result) override {
80 std::vector<settlement::SettlementRequest> transitions;
81 const auto append = [&](std::string key) {
82 settlement::SettlementRequest transition = context.request();
83 transition.kind = "trigger";
84 transition.resource.clear();
85 transition.magnitude = 0.0;
86 transition.decisions.clear();
87 transition.trigger = std::move(key);
88 transition.chain = {};
89 transition.tags.push_back("trigger:" + transition.trigger);
90 transitions.push_back(std::move(transition));
91 };
92 if (target_.barrier > 0 && barrierAbsorbed_ >= static_cast<double>(target_.barrier)) append("shield_break");
93 const bool died = context.request().kind == "damage" && target_.health > 0 &&
94 result.applied >= static_cast<double>(target_.health);
95 if (died) {
96 append("death");
97 if (context.request().source.isValid() && context.request().source != context.request().target)
98 append("kill");
99 }
100 return eve::Result<std::vector<settlement::SettlementRequest>>::success(std::move(transitions));
101 }

130 const effects::EffectInstance& effect,
131 const effects::EffectContainer& activeEffects) const {
132 if (target.health < 0 || target.maxHealth <= 0 || target.health > target.maxHealth || target.barrier < 0)
133 return invalid("card effect target state is invalid", "target");
134 const bool damage = hasKind(effect, CardEffectKind::Damage);
135 const bool heal = hasKind(effect, CardEffectKind::Heal);
136 const bool shield = hasKind(effect, CardEffectKind::Shield);
137 if (!shield && ((!damage && !heal) || effect.period > 0.0)) return eve::Result<void>::success();
138 if (effect.magnitude < 0.0 || effect.magnitude > static_cast<double>(std::numeric_limits<int>::max()))
139 return invalid("card immediate magnitude is outside the target range", "magnitude");

141 if (!parsedSubject) return invalid("card shield subject is invalid", "subject");
142 settlement::SettlementRequest request;
143 request.source = eve::SubjectRef::fromPersistentId(eve::PersistentId::fromUuid(effect.identity));
144 request.target = eve::SubjectRef::fromPersistentId(*parsedSubject);
145 request.kind = shield ? "shield" : heal ? "healing" : "damage";
146 request.magnitude = effect.magnitude;
147 request.tags = effect.tags;

150 auto activeRules = settlement::projectEffectRules(activeEffects, "card.active");
151 if (!activeRules) return eve::Result<void>::failure(activeRules.status());
152 auto installed = activeRules.value().install(activePipeline);
153 if (!installed) return installed;
154 auto settled = activePipeline.settle(request, policy);
155 if (!settled) return eve::Result<void>::failure(settled.status());
156 const auto outcome = std::move(settled).takeValue();
157 (void)outcome;

161eve::Result<CardEffectUpdate> CardEffectExecutor::settle(CardEffectTarget& target,
162 effects::EffectUpdateSummary lifecycle,
163 const effects::EffectContainer& activeEffects) const {
164 if (target.health < 0 || target.maxHealth <= 0 || target.health > target.maxHealth || target.barrier < 0)
165 return eve::Result<CardEffectUpdate>::failure(eve::Diagnostic::error(
166 eve::DiagnosticCode::InvalidArgument, "card effect target state is invalid", "target"));
167 CardEffectUpdate result;
168 result.lifecycle = std::move(lifecycle);
169 auto activePipeline = settlement_;
170 auto activeRules = settlement::projectEffectRules(activeEffects, "card.active");
171 if (!activeRules) return eve::Result<CardEffectUpdate>::failure(activeRules.status());
172 auto installed = activeRules.value().install(activePipeline);
173 if (!installed) return eve::Result<CardEffectUpdate>::failure(installed.status());
174 for (const auto& tick : result.lifecycle.periodicTicks) {
175 if (tick.magnitude < 0.0 || tick.stackCount == 0)
176 return eve::Result<CardEffectUpdate>::failure(eve::Diagnostic::error(
177 eve::DiagnosticCode::InvalidArgument, "card periodic magnitude or stack is invalid", "periodic"));
178 const auto amount = tick.magnitude * static_cast<double>(tick.stackCount);
179 if (!std::isfinite(amount) || amount > static_cast<double>(std::numeric_limits<int>::max()))
180 return eve::Result<CardEffectUpdate>::failure(eve::Diagnostic::error(
181 eve::DiagnosticCode::InvalidArgument, "card periodic settlement overflows", "periodic"));
182 if (tick.effectId.empty())
183 return eve::Result<CardEffectUpdate>::failure(
184 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
185 "card periodic trigger has no effect identity", "periodic.effectId"));
186 const auto tagged = [&tick](const char* value) {
187 return std::find(tick.tags.begin(), tick.tags.end(), value) != tick.tags.end();
188 };
189 {
190 const auto parsedSubject = eve::PersistentId::parse(tick.subject);
191 if (!parsedSubject)
192 return eve::Result<CardEffectUpdate>::failure(eve::Diagnostic::error(
193 eve::DiagnosticCode::InvalidArgument, "card periodic subject is invalid", "periodic.subject"));
194 settlement::SettlementRequest request;
195 request.source = eve::SubjectRef::fromPersistentId(
196 eve::PersistentId::fromUuid(tick.effectIdentity));
197 request.target = eve::SubjectRef::fromPersistentId(*parsedSubject);
198 request.kind = tagged("card:shield") ? "shield" : tagged("card:heal") ? "healing" : "damage";
199 request.magnitude = amount;
200 request.tags = tick.tags;
201 request.tick = tick.tick;
202 CardSettlementPolicy policy(target);
203 auto settled = activePipeline.settle(request, policy);
204 if (!settled) return eve::Result<CardEffectUpdate>::failure(settled.status());
205 const auto status = settled.status();
206 auto outcome = std::move(settled).takeValue();
207 result.absorbed += static_cast<std::uint32_t>(outcome.absorbed);
208 if (tagged("card:death-trigger") &&
209 std::any_of(outcome.derived.begin(), outcome.derived.end(),
210 [](const auto& derived) { return derived.trigger == "death"; }))
211 result.deathTriggered = true;
212 settlement::SettlementBatchItemResult audit;
213 audit.index = result.settlements.size();
214 audit.request = std::move(request);
215 audit.status = status;
216 audit.result = std::move(outcome);
217 result.settlements.push_back(std::move(audit));
218 }
219 ++result.settled;
220 }
221 return eve::Result<CardEffectUpdate>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
222}

233 if (!subject.isValid())
