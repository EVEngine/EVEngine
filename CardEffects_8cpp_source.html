35eve::Result<void> CardEffectExecutor::applyImmediate(CardEffectTarget& target,

37 if (target.health < 0 || target.maxHealth <= 0 || target.health > target.maxHealth || target.barrier < 0)

49eve::Result<CardEffectUpdate> CardEffectExecutor::settle(CardEffectTarget& target,
50 effects::EffectUpdateSummary lifecycle) const {
51 if (target.health < 0 || target.maxHealth <= 0 || target.health > target.maxHealth || target.barrier < 0)
52 return eve::Result<CardEffectUpdate>::failure(eve::Diagnostic::error(
53 eve::DiagnosticCode::InvalidArgument, "card effect target state is invalid", "target"));
54 CardEffectUpdate result;
55 result.lifecycle = std::move(lifecycle);
56 for (const auto& tick : result.lifecycle.periodicTicks) {
57 if (tick.magnitude < 0.0 || tick.stackCount == 0)
58 return eve::Result<CardEffectUpdate>::failure(eve::Diagnostic::error(
59 eve::DiagnosticCode::InvalidArgument, "card periodic magnitude or stack is invalid", "periodic"));
60 const auto amount = tick.magnitude * static_cast<double>(tick.stackCount);
61 if (!std::isfinite(amount) || amount > static_cast<double>(std::numeric_limits<int>::max()))
62 return eve::Result<CardEffectUpdate>::failure(eve::Diagnostic::error(
63 eve::DiagnosticCode::InvalidArgument, "card periodic settlement overflows", "periodic"));
64 if (tick.effectId.empty())
65 return eve::Result<CardEffectUpdate>::failure(
66 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
67 "card periodic trigger has no effect identity", "periodic.effectId"));
68 const auto tagged = [&tick](const char* value) {
69 return std::find(tick.tags.begin(), tick.tags.end(), value) != tick.tags.end();
70 };
71 if (tagged("card:shield")) {
72 const auto shield = static_cast<int>(amount);
73 if (shield > std::numeric_limits<int>::max() - target.barrier)
74 return eve::Result<CardEffectUpdate>::failure(
75 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
76 "card periodic barrier addition overflows", "target.barrier"));
77 target.barrier += shield;
78 } else if (tagged("card:heal")) {
79 target.health = std::min(target.maxHealth, target.health + static_cast<int>(amount));
80 } else {
81 const auto damage = static_cast<int>(amount);
82 const auto absorbed = std::min(target.barrier, damage);
83 target.barrier -= absorbed;
84 target.health = std::max(0, target.health - (damage - absorbed));
85 result.absorbed += static_cast<std::uint32_t>(absorbed);
86 if (target.health == 0) {
87 ++target.deathTriggers;
88 result.deathTriggered = true;
89 }
90 }
91 ++result.settled;
92 }
93 return eve::Result<CardEffectUpdate>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
94}

97 eve::SubjectRef subject) {
98 if (definition.id.empty())

101 if (!subject.isValid())

103 eve::DiagnosticCode::InvalidArgument, "card effect subject must be valid", "subject"));
104 effects::EffectDefinition common;
105 common.id = definition.id;
106 common.stackKey = definition.id;
107 common.duration = definition.duration;
108 common.period = definition.period;
109 common.magnitude = definition.magnitude;
110 common.policy = definition.policy;
111 common.tags = definition.tags;
112 common.tags.push_back(definition.kind == CardEffectKind::Damage ? "card:damage"
113 : definition.kind == CardEffectKind::Heal ? "card:heal"
114 : "card:shield");
115 if (definition.deathTrigger) common.tags.push_back("card:death-trigger");
116 auto valid = common.validate();
117 if (!valid) return eve::Result<effects::EffectHandle>::failure(valid.status());
118
119 auto candidate = container_.clone();
120 auto applied = candidate.apply(common, subject.format(), definition.source);
121 if (!applied) return eve::Result<effects::EffectHandle>::failure(applied.status());
122 const std::string id = std::move(applied).takeValue();
123 auto instance = candidate.find(id);
124 if (!instance)
125 return eve::Result<effects::EffectHandle>::failure(eve::Diagnostic::error(
126 eve::DiagnosticCode::InvariantViolation, "card effect candidate disappeared", "effect"));
127 CardEffectTarget targetCandidate = target_;
128 auto immediate = executor_.applyImmediate(targetCandidate, *instance);
129 if (!immediate) return eve::Result<effects::EffectHandle>::failure(immediate.status());
130 auto handle = candidate.handleFor(id);
131 if (!handle) return eve::Result<effects::EffectHandle>::failure(handle.status());
132 target_ = std::move(targetCandidate);
133 container_ = std::move(candidate);
134 return eve::Result<effects::EffectHandle>::success(std::move(handle).takeValue(),
135 eve::Status::success(eve::StatusCode::Applied));
136}

144eve::Result<CardEffectUpdate> CardEffectAdapter::advance(const eve::SimulationStep& step) {
145 auto candidate = container_.clone();
146 auto lifecycle = effects::EffectExecutor{}.advance(candidate, step);
147 if (!lifecycle) return eve::Result<CardEffectUpdate>::failure(lifecycle.status());
148 auto targetCandidate = target_;
149 auto settled = executor_.settle(targetCandidate, std::move(lifecycle).takeValue());
150 if (!settled) return settled;
151 auto result = std::move(settled).takeValue();
152 target_ = std::move(targetCandidate);
153 container_ = std::move(candidate);
154 return eve::Result<CardEffectUpdate>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
155}
