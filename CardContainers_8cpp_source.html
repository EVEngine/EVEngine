21[[nodiscard]] bool sameLayout(const eve::container::ContainerSnapshot& lhs,
22 const eve::container::ContainerSnapshot& rhs) {
23 if (lhs.id != rhs.id || lhs.revision != rhs.revision || lhs.entries.size() != rhs.entries.size()) return false;
24 for (std::size_t index = 0; index < lhs.entries.size(); ++index) {
25 const auto& left = lhs.entries[index];
26 const auto& right = rhs.entries[index];
27 if (left.membership.object != right.membership.object || left.membership.slot != right.membership.slot ||
28 left.membership.generation != right.membership.generation || left.object.id != right.object.id ||

31 }

37class CardContainerAdapter::PreparedState final : public eve::container::IContainer::PreparedState {
38public:
39 PreparedState(CardContainerAdapter& owner, std::vector<CardData*> staged, eve::Revision revision)
40 : owner_(owner), staged_(std::move(staged)), revision_(revision) {}
41
42 void commit() noexcept override {
43 if (committed_) return;
44 auto* live = owner_.cards();
45 if (live == nullptr) std::terminate();
46 live->swap(staged_);
47 owner_.revision_ = revision_;
48 for (auto* card : *live) owner_.updateState(card);
49 committed_ = true;
50 }
51
52 void rollback() noexcept override {
53 if (!committed_) staged_.clear();
54 }
55
56private:
57 CardContainerAdapter& owner_;
58 std::vector<CardData*> staged_;
59 eve::Revision revision_;
60 bool committed_ = false;
61};

138eve::Result<eve::container::ContainerSnapshot> CardContainerAdapter::snapshot() const {
139 const auto* values = cards();
140 if (!descriptor_.id.isValid() || values == nullptr)
141 return eve::Result<eve::container::ContainerSnapshot>::failure(
142 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "card container adapter is not bound"));
143 if (!descriptor_.capacity.isUnlimited() && values->size() > descriptor_.capacity.value())
144 return eve::Result<eve::container::ContainerSnapshot>::failure(eve::Diagnostic::error(
145 eve::DiagnosticCode::InvariantViolation, "card container exceeds configured capacity"));
146 eve::container::ContainerSnapshot snapshot{descriptor_.id, revision_, {}};
147 std::unordered_set<eve::container::MembershipId> ids;
148 snapshot.entries.reserve(values->size());
149 for (std::size_t i = 0; i < values->size(); ++i) {
150 if (i > static_cast<std::size_t>(std::numeric_limits<std::int32_t>::max()))
151 return eve::Result<eve::container::ContainerSnapshot>::failure(eve::Diagnostic::error(
152 eve::DiagnosticCode::InvariantViolation, "card container has too many indexed memberships"));
153 CardData* card = (*values)[i];
154 if (card == nullptr || card->identity()->id.empty())
155 return eve::Result<eve::container::ContainerSnapshot>::failure(eve::Diagnostic::error(
156 eve::DiagnosticCode::InvariantViolation, "card container contains null or unidentified card"));
157 const auto id = objectId(*card);
158 if (!ids.insert(id).second)
159 return eve::Result<eve::container::ContainerSnapshot>::failure(eve::Diagnostic::error(
160 eve::DiagnosticCode::Conflict, "card container contains duplicate card identity"));
161 snapshot.entries.push_back(
162 {{id, eve::container::SlotIndex(static_cast<std::int32_t>(i)), eve::Generation(1)}, describe(card)});
163 }
164 return eve::Result<eve::container::ContainerSnapshot>::success(std::move(snapshot));
165}

167eve::Result<void> CardContainerAdapter::validateInsert(
168 const eve::container::ContainerObject& object, std::optional<eve::container::SlotIndex> destination,
169 std::optional<eve::container::MembershipId> ignoredObject) const {
170 const auto* values = cards();
171 if (values == nullptr || !descriptor_.id.isValid())
172 return error(eve::DiagnosticCode::InvalidArgument, "card container adapter is not bound");
173 if (!object.id.isValid() || !object.payload)
174 return error(eve::DiagnosticCode::InvalidArgument, "card transfer object is incomplete");
175 if (object.quantity != 1u)
176 return error(eve::DiagnosticCode::InvalidArgument, "a card container membership must have quantity one");
177 const auto* payload = dynamic_cast<const CardContainerObject*>(object.payload.get());
178 auto* card = payload == nullptr ? nullptr : static_cast<CardData*>(ecs::try_get(payload->card));
179 if (payload == nullptr || card == nullptr || card->identity()->id != object.id.value() ||
180 card->identity()->kind != object.type)
181 return error(eve::DiagnosticCode::StaleHandle, "card transfer payload is stale");
182 if (hasId(*values, object.id) && (!ignoredObject || *ignoredObject != object.id))
183 return error(eve::DiagnosticCode::Conflict, "destination already contains card");
184 const std::size_t effectiveSize = values->size() - ((ignoredObject && hasId(*values, *ignoredObject)) ? 1u : 0u);
185 if (destination) {
186 if (!destination->isValid() || static_cast<std::size_t>(destination->value()) > effectiveSize)
187 return error(eve::DiagnosticCode::InvalidArgument, "destination card slot is out of range");
188 }
189 auto accepted = descriptor_.filter.evaluate(object);
190 if (!accepted) return accepted;
191 if (!descriptor_.capacity.isUnlimited() && effectiveSize >= descriptor_.capacity.value())
192 return error(eve::DiagnosticCode::Conflict, "card container capacity is full");
193 return eve::Result<void>::success();
194}

196eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>> CardContainerAdapter::prepare(
197 const eve::container::ContainerSnapshot& expected, const eve::container::ContainerSnapshot& candidate) {
198 const auto* values = cards();
199 if (values == nullptr || !descriptor_.id.isValid())
200 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
201 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "card container adapter is not bound"));
202 if (expected.id != descriptor_.id || candidate.id != descriptor_.id)
203 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
204 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "card prepare targets another container"));
205 auto currentResult = snapshot();
206 if (!currentResult)
207 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(currentResult.status());
208 if (!sameLayout(currentResult.value(), expected))
209 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
210 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "card prepare expected snapshot is stale"));
211 const auto nextRevision = expected.revision.incremented();
212 if (!nextRevision || candidate.revision != *nextRevision)
213 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(eve::Diagnostic::error(
214 eve::DiagnosticCode::InvariantViolation, "card candidate revision is not the next revision"));
215 if (!descriptor_.capacity.isUnlimited() && candidate.entries.size() > descriptor_.capacity.value())
216 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
217 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "card candidate exceeds container capacity"));
218
219 std::vector<const eve::container::MembershipEntry*> ordered;
220 ordered.reserve(candidate.entries.size());
221 for (const auto& entry : candidate.entries) ordered.push_back(&entry);
222 std::sort(ordered.begin(), ordered.end(),
223 [](const auto* lhs, const auto* rhs) { return lhs->membership.slot < rhs->membership.slot; });
224 std::vector<CardData*> staged;
225 staged.reserve(ordered.size());
226 std::unordered_set<eve::container::MembershipId> ids;
227 for (std::size_t index = 0; index < ordered.size(); ++index) {
228 const auto* entry = ordered[index];
229 if (!entry->membership.object.isValid() || !entry->membership.slot.isValid() ||
230 entry->membership.slot.value() != static_cast<std::int32_t>(index) ||
231 entry->membership.generation.isZero() || entry->object.id != entry->membership.object ||
232 entry->object.quantity != 1u || !ids.insert(entry->membership.object).second)
233 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
234 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
235 "card candidate contains invalid membership"));
236 const auto* payload = dynamic_cast<const CardContainerObject*>(entry->object.payload.get());
237 auto* card = payload == nullptr ? nullptr : static_cast<CardData*>(ecs::try_get(payload->card));
238 if (payload == nullptr || card == nullptr || card->identity()->id != entry->membership.object.value() ||
239 entry->object.type != card->identity()->kind)
240 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::failure(
241 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "card candidate payload is stale"));
242 staged.push_back(card);
243 }
244 std::unique_ptr<eve::container::IContainer::PreparedState> prepared =
245 std::make_unique<PreparedState>(*this, std::move(staged), *nextRevision);
246 return eve::Result<std::unique_ptr<eve::container::IContainer::PreparedState>>::success(std::move(prepared));
247}
