21int TraitSystem::apply(RPGActor *actor, const std::string &traitId, const std::string &source) {
22 if (!actor) return 0;
23 const TraitDefinition *def = findDef(traitId);
24 if (!def) return 0;
25 auto &ts = *actor->traits();
26 RPGActor::TraitInstance inst;
27 inst.instanceId = ts.nextInstanceId++;
28 inst.traitId = traitId;
29 inst.source = source;
30 // paramRate → 属性修改器（mulAdd 倍率），source="trait:<instanceId>"
31 for (const auto &spec : def->traits) {
32 if (spec.kind != "paramRate" || spec.target.empty()) continue;
33 AttributeSystem::addModifier(actor, spec.target,
34 "trait:" + std::to_string(inst.instanceId), "mulAdd",
35 spec.value - 1.0, 0);
36 }
37 ts.active.push_back(std::move(inst));
38 return inst.instanceId;
39}
