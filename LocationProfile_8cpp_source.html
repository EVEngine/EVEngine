36bool number(const Value::Object& object, const char* name, float& output) {
37 const auto found = object.find(name);
38 if (found == object.end() || !found->second.isNumeric()) return false;
39 output = static_cast<float>(found->second.isDouble() ? found->second.asDouble() : found->second.asInt());
40 return std::isfinite(output);
41}
42bool parsePose(const Value& value, LocationPose& output) {
43 const auto* object = value.getIf<Value::Object>();
44 if (!object || !onlyFields(*object, {"x", "y", "z", "qx", "qy", "qz", "qw"})) return false;
45 LocationPose candidate;
46 if (!number(*object,"x",candidate.x) || !number(*object,"y",candidate.y) ||
47 !number(*object,"z",candidate.z) || !number(*object,"qx",candidate.qx) ||
48 !number(*object,"qy",candidate.qy) || !number(*object,"qz",candidate.qz) ||
49 !number(*object,"qw",candidate.qw)) return false;
50 output = candidate;
51 return true;
52}
53bool parseSnapshot(const Value& value, LocationSnapshot& output) {
54 const auto* object = value.getIf<Value::Object>();
55 if (!object || !onlyFields(*object, {"camera", "player"})) return false;
56 const auto camera = object->find("camera"), player = object->find("player");
57 if (camera == object->end() || player == object->end()) return false;
58 LocationSnapshot candidate;
59 if (!parsePose(camera->second, candidate.camera)) return false;
60 if (!player->second.isNull()) {
61 LocationPose pose;
62 if (!parsePose(player->second, pose)) return false;
63 candidate.player = pose;
64 }
65 output = candidate;
66 return true;
67}

172Result<void> LocationProfile::restoreJson(std::string_view json) {
173 auto decoded = Value::fromJson(json);
174 if (!decoded) return Result<void>::failure(decoded.status());
175 const auto* root = decoded.value().getIf<Value::Object>();
176 const auto fail = [] { return invalid<void>("invalid location profile schema", DiagnosticCode::ParseError); };
177 if (!root || !onlyFields(*root, {"schemaId", "schemaVersion", "saved", "bookmarks"})) return fail();
178 const auto schema = root->find("schemaId"), version = root->find("schemaVersion");
179 const auto saved = root->find("saved"), bookmarks = root->find("bookmarks");
180 if (schema == root->end() || !schema->second.isString() ||
181 schema->second.asString() != "eve.scene.location-profile" || version == root->end() ||
182 !version->second.isInt64() || version->second.asInt() != 1 || saved == root->end() ||
183 bookmarks == root->end() || !bookmarks->second.isArray()) return fail();
184 LocationProfile candidate;
185 if (!saved->second.isNull()) {
186 LocationSnapshot snapshot;
187 if (!parseSnapshot(saved->second, snapshot)) return fail();
188 candidate.saved_ = snapshot;
189 }
190 for (const auto& value : *bookmarks->second.getIf<Value::Array>()) {
191 const auto* object = value.getIf<Value::Object>();
192 if (!object || !onlyFields(*object, {"name", "camera", "player", "controller", "scene"})) return fail();
193 const auto name=object->find("name"), camera=object->find("camera"), player=object->find("player");
194 const auto controller=object->find("controller"), scene=object->find("scene");
195 if (name==object->end() || !name->second.isString() || camera==object->end() ||
196 player==object->end() || controller==object->end() || !controller->second.isString() ||
197 scene==object->end() || !scene->second.isString()) return fail();
198 LocationBookmark bookmark;
199 bookmark.name=name->second.asString(); bookmark.controller=controller->second.asString();
200 bookmark.scene=scene->second.asString();
201 if (!parsePose(camera->second,bookmark.camera)) return fail();
202 if (!player->second.isNull()) { LocationPose pose; if (!parsePose(player->second,pose)) return fail(); bookmark.player=pose; }
203 auto added=candidate.addBookmark(std::move(bookmark));
204 if (!added) return Result<void>::failure(added.status());
205 }
206 *this = std::move(candidate);
207 return Result<void>::success();
208}
