11Result<void> mergeImport(PreparedAssetImport& output, PreparedAssetImport input, const UnitySourceAsset& source,
12 const AssetImportLimits& limits) {
13 if (input.manifest.assets.size() > limits.maximumAssets ||
14 output.manifest.assets.size() > limits.maximumAssets - input.manifest.assets.size())
15 return detail::failure<void>(DiagnosticCode::InvalidArgument, "Unity imported asset count exceeds budget");
16 for (auto& asset : input.manifest.assets) output.manifest.assets.push_back(std::move(asset));
17 for (auto& dependency : input.manifest.dependencies) output.manifest.dependencies.push_back(std::move(dependency));
18 for (auto& entry : input.entries)
19 if (entry.path != "reports/import.json") output.entries.push_back(std::move(entry));
20 for (auto& mapping : input.sourceMappings) {
21 mapping.sourceObject = "unity:" + source.guid + "/" + mapping.sourceObject;
22 output.sourceMappings.push_back(std::move(mapping));
23 }
24 for (auto& finding : input.findings) {
25 finding.sourcePath = source.path;
26 output.findings.push_back(std::move(finding));
27 }
28 for (const auto& [name, entry] : input.manifest.entrypoints)
29 output.manifest.entrypoints.emplace(source.path + (name == "default" ? "" : "#" + name), entry);
30 return Result<void>::success();
31}

33Result<PreparedAssetImport> importSource(const UnityProjectImportRequest& request, const UnitySourceAsset& source) {
34 if (source.guid.empty())
35 return detail::failure<PreparedAssetImport>(
36 DiagnosticCode::Unsupported, "a .meta GUID is required for stable collection asset identity", source.path);
37 const auto ext = detail::extension(source.path);
38 if (ext == "anim") return prepareUnitySpriteAnimation(request, source);
39 if (ext == "asset") return prepareUnityNativeMesh(request, source);
40 if (ext == "fbx") return prepareUnityFbx(request, source);
41 if (source.kind == UnitySourceKind::Material) return prepareUnityMaterial(request, source);
42 auto identity = request.package;
43 identity.packageId = request.package.packageId.child("unity:" + source.guid);
44 if (ext == "png" || ext == "jpg" || ext == "jpeg") {
45 const auto& meta = request.files.at(source.path + ".meta");
46 const std::string text(meta.begin(), meta.end());
47 const std::regex srgb(R"((?:^|\n)\s*sRGBTexture:\s*0(?:\s|$))");
48 const std::regex normal(R"((?:^|\n)\s*textureType:\s*1(?:\s|$))");
49 const bool normalMap = std::regex_search(text, normal);
50 const bool linear = normalMap || std::regex_search(text, srgb);
51 auto result = prepareImageImport({identity, "unity:" + source.guid, request.files.at(source.path),
52 linear ? ImageColorSpace::Linear : ImageColorSpace::Srgb,
53 normalMap ? "normal" : "color", request.limits});
54 if (!result) return result;
55 if (std::regex_search(text, std::regex(R"((?:^|\n)\s*spriteMode:\s*[12](?:\s|$))")))
56 result.value().findings.push_back({source.path, "TextureImporter.sprite", ImportDisposition::Unsupported,
57 "standalone Sprite/UI assets are not emitted; converted sprite "
58 "animation clips retain their referenced slicing and pivots"});
59 if (normalMap)
60 result.value().findings.push_back(
61 {source.path, "TextureImporter.normal", ImportDisposition::Unsupported,
62 "source normal pixels are linear; Unity normal processing settings are not baked"});
63 return result;
64 }
65 if (ext == "gltf" || ext == "glb") {
66 std::map<std::string, std::vector<std::uint8_t>> external;
67 const auto slash = source.path.find_last_of('/');
68 const auto prefix = slash == std::string::npos ? std::string{} : source.path.substr(0, slash + 1);
69 for (const auto& [path, bytes] : request.files)
70 if (path.starts_with(prefix) && path != source.path && !path.ends_with(".meta"))
71 external.emplace(path.substr(prefix.size()), bytes);
72 auto result = prepareGltfImport(
73 {identity, "unity:" + source.guid, request.files.at(source.path), std::move(external), request.limits});
74 if (!result) return result;
75 result.value().findings.push_back(
76 {source.path, "Model.rendering", ImportDisposition::Unsupported,
77 "mesh primitives imported; materials, skinning, animation and scene binding are not converted"});
78 return result;
79 }
80 if (source.kind == UnitySourceKind::Prefab) {
81 return prepareUnityPrefab(request, source.path);
82 }
83 return detail::failure<PreparedAssetImport>(
84 DiagnosticCode::Unsupported,
85 "source retained; this Unity resource type does not yet have a canonical converter", source.path);
86}

89Result<PreparedAssetImport> prepareUnityCollection(const UnityProjectImportRequest& request,
90 const UnitySourceIndex& index) {
91 auto manifest = detail::baseManifest(request.package, "eve.unity-collection/1");
92 if (!manifest) return Result<PreparedAssetImport>::failure(manifest.status());
93 PreparedAssetImport out;
94 out.manifest = std::move(manifest).takeValue();
95 out.findings = index.findings;
96 UnityProjectImportRequest resolved = request;
97 std::uint64_t expandedBytes = 0;
98 for (const auto& source : index.assets) {
99 if (source.kind != UnitySourceKind::Prefab) continue;
100 const auto& bytes = request.files.at(source.path);
101 if (std::string(bytes.begin(), bytes.end()).find("!u!1001 ") == std::string::npos) continue;
102 auto expanded = expandUnityPrefab(request, index, source);
103 if (!expanded) return Result<PreparedAssetImport>::failure(expanded.status());
104 out.findings.insert(out.findings.end(), expanded.value().findings.begin(), expanded.value().findings.end());
105 if (expanded.value().bytes.size() > request.limits.maximumDecodedBytes ||
106 expandedBytes > request.limits.maximumDecodedBytes - expanded.value().bytes.size())
107 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
108 "aggregate Prefab expansion budget exceeded");
109 expandedBytes += expanded.value().bytes.size();
110 resolved.files[source.path] = std::move(expanded).takeValue().bytes;
111 }
112 for (const auto& source : index.assets) {
113 if (source.kind == UnitySourceKind::Folder) continue;
114 auto imported = importSource(resolved, source);
115 if (!imported) {
116 const auto* error = imported.status().primaryDiagnostic();
117 if (!error || error->code() != DiagnosticCode::Unsupported)
118 return Result<PreparedAssetImport>::failure(imported.status());
119 out.findings.push_back(
120 {source.path, "resource.conversion", ImportDisposition::Unsupported, error->message()});
121 continue;
122 }
123 auto merged = mergeImport(out, std::move(imported).takeValue(), source, request.limits);
124 if (!merged) return Result<PreparedAssetImport>::failure(merged.status());
125 }
126 if (out.manifest.assets.empty())
127 return detail::failure<PreparedAssetImport>(
128 DiagnosticCode::Unsupported,
129 "Unity collection has no convertible assets; use asset scan to inspect all resource types");
130 auto bindings = bindUnityRenderers(resolved, index, out);
131 if (!bindings) return Result<PreparedAssetImport>::failure(bindings.status());
132 Value::Object sourceHashes;
133 for (const auto& [path, data] : request.files) {
134 out.entries.push_back({"sources/unity/" + path, data});
135 sourceHashes[path] = Value(detail::sha256(data));
136 if (!path.ends_with(".meta"))
137 out.findings.push_back(
138 {path, "source.archive", ImportDisposition::PreservedSource,
139 "original source and available .meta retained under sources/unity; not runtime conversion"});
140 }
141 out.manifest.provenance["sourceCoordinateSystem"] = Value("left-handed-x-right-y-up-z-forward");
142 std::uint64_t total = 0;
143 for (const auto& entry : out.entries) {
144 if (entry.bytes.size() > request.limits.maximumDecodedBytes ||
145 total > request.limits.maximumDecodedBytes - entry.bytes.size())
146 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
147 "Unity prepared output exceeds byte budget");
148 total += entry.bytes.size();
149 }
150 auto report = detail::finalizeImportReport(out, request.package, "unity", "source-files",
151 {{"sourceHashes", Value(std::move(sourceHashes))}});
152 if (!report) return Result<PreparedAssetImport>::failure(report.status());
153 if (out.entries.back().bytes.size() > request.limits.maximumDecodedBytes ||
154 total > request.limits.maximumDecodedBytes - out.entries.back().bytes.size())
155 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
156 "Unity import report exceeds prepared output budget");
157 return Result<PreparedAssetImport>::success(std::move(out));
158}
