19void pushBody2D(const SceneNode &n, void *target) {
20 auto *b = static_cast<Body *>(target);
21 b->setPosition(n.x, n.y);
22 b->setAngle(n.roll);
23}

25void pullBody2D(SceneNode &n, void *target) {

32bool aliveBody2D(const void *target) {

36void pushBody3D(const SceneNode &n, void *target) {
37 auto *b = static_cast<Body3D *>(target);
38 b->setPosition(n.x, n.y, n.z);
39 glm::quat q = glm::angleAxis(n.yaw, glm::vec3(0.f, 1.f, 0.f)) *
40 glm::angleAxis(n.pitch, glm::vec3(1.f, 0.f, 0.f)) *
41 glm::angleAxis(n.roll, glm::vec3(0.f, 0.f, 1.f));
42 b->setRotation(q.x, q.y, q.z, q.w);
43}
44
45void pullBody3D(SceneNode &n, void *target) {
46 auto *b = static_cast<Body3D *>(target);
47 n.x = b->getX();
48 n.y = b->getY();
49 n.z = b->getZ();
50 glm::quat q(b->getRotW(), b->getRotX(), b->getRotY(), b->getRotZ());
51 glm::vec3 e = glm::eulerAngles(q); // pitch(x), yaw(y), roll(z)
52 n.pitch = e.x;
53 n.yaw = e.y;
54 n.roll = e.z;
55}
56
57bool aliveBody3D(const void *target) {
58 return static_cast<const Body3D *>(target)->isValid();
59}

64 eve::scene::registerLinkKind({"physics3d", &pushBody3D, &pullBody3D, &aliveBody3D});
65 }
66} g_register;
