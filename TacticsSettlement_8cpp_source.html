48Result<std::vector<settlement::SettlementResult>> TacticsSettlementRuntime::settleAtomic(
49 std::span<const AbilitySettlementRequest> requests, std::span<settlement::ISettlementPolicy*> policies,
50 game_event::GameEventLog* eventLog) const {
51 std::vector<settlement::SettlementRequest> common;
52 common.reserve(requests.size());
53 for (const auto& request : requests) {
54 auto projected = makeSettlementRequest(request);
55 if (!projected) return Result<std::vector<settlement::SettlementResult>>::failure(projected.status());
56 common.push_back(std::move(projected).takeValue());
57 }
58 return settlement_.settleAtomic(common, policies, eventLog);
