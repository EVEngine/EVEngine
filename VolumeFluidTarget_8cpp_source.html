59std::vector<EditorDiagnostic> validateSettings(const VolumeFluidAuthoringSettings& s) {
60 std::vector<EditorDiagnostic> out;
61 const double finite[]{s.spacing, s.gravityX, s.gravityY, s.gravityZ, s.minimumX,
62 s.minimumY, s.minimumZ, s.maximumX, s.maximumY, s.maximumZ};
63 if (std::any_of(std::begin(finite), std::end(finite), [](double v) { return !std::isfinite(v); })) {

83EditorResult<VolumeFluidAuthoringSettings> parseSettings(const EditorValue& value) {
84 const auto* object = value.getIf<EditorValue::Object>();
85 if (!object || object->size() != 7)
86 return targetError<VolumeFluidAuthoringSettings>(EditorStatus::Rejected, "editor.volume-fluid.fields",
87 "Volume fluid settings require exactly seven fields");
88 const auto* cv = field(value, "capacity");
89 const auto* pv = field(value, "previewParticles");
90 const auto* sv = field(value, "spacing");
91 const auto* iv = field(value, "iterations");
92 const auto* capacity = cv ? cv->getIf<std::int64_t>() : nullptr;
93 const auto* preview = pv ? pv->getIf<std::int64_t>() : nullptr;
94 const auto* spacing = sv ? sv->getIf<double>() : nullptr;
95 const auto* iterations = iv ? iv->getIf<std::int64_t>() : nullptr;
96 if (!capacity || !preview || !spacing || !iterations || *capacity < 0 || *preview < 0 || *iterations < 0 ||
97 *capacity > 1000000 || *preview > 1000000 || *iterations > 20)
98 return targetError<VolumeFluidAuthoringSettings>(EditorStatus::Rejected, "editor.volume-fluid.types",
99 "Volume fluid scalar fields are invalid");
100 VolumeFluidAuthoringSettings s;
101 s.capacity = static_cast<std::uint32_t>(*capacity);
102 s.previewParticles = static_cast<std::uint32_t>(*preview);
103 s.spacing = *spacing;
104 s.iterations = static_cast<std::uint32_t>(*iterations);
105 if (!readVec3(value, "gravity", s.gravityX, s.gravityY, s.gravityZ) ||
106 !readVec3(value, "minimum", s.minimumX, s.minimumY, s.minimumZ) ||
107 !readVec3(value, "maximum", s.maximumX, s.maximumY, s.maximumZ))
108 return targetError<VolumeFluidAuthoringSettings>(EditorStatus::Rejected, "editor.volume-fluid.vectors",
109 "Gravity and bounds must be finite Vec3 values");
110 const auto diagnostics = validateSettings(s);
111 if (std::any_of(diagnostics.begin(), diagnostics.end(),
112 [](const auto& d) { return d.severity() == DiagnosticSeverity::Error; })) {
113 return targetError<VolumeFluidAuthoringSettings>(EditorStatus::Rejected, "editor.volume-fluid.invalid",
114 "Volume fluid settings are invalid");
115 }
116 return eve::editing::applied<VolumeFluidAuthoringSettings>(s);

278EditorResult<void> VolumeFluidRuntimeApplier::apply(const VolumeFluidTarget& target,
279 fluids::VolumeFluid* simulation) const {
280 if (!simulation) {
281 return targetError<void>(EditorStatus::Rejected, "editor.volume-fluid.runtime-required",
282 "Volume fluid publication requires a live simulation");
283 }
284 for (const auto& d : target.validate()) {
285 if (d.severity() == DiagnosticSeverity::Error) {

287 }
288 }
289 auto snapshot = simulation->snapshot();
290 const auto s = target.settings();
291 snapshot.settings.capacity = s.capacity;
292 snapshot.settings.spacing = static_cast<float>(s.spacing);
293 snapshot.settings.gravity = {static_cast<float>(s.gravityX), static_cast<float>(s.gravityY),
294 static_cast<float>(s.gravityZ)};
295 snapshot.settings.minimum = {static_cast<float>(s.minimumX), static_cast<float>(s.minimumY),
296 static_cast<float>(s.minimumZ)};
297 snapshot.settings.maximum = {static_cast<float>(s.maximumX), static_cast<float>(s.maximumY),
298 static_cast<float>(s.maximumZ)};
299 snapshot.settings.iterations = s.iterations;
300 auto restored = simulation->restore(snapshot);
301 if (!restored.ok()) {
302 return targetError<void>(
303 EditorStatus::Conflict, "editor.volume-fluid.runtime-rejected",
304 restored.error() ? restored.error()->message() : "Volume fluid runtime rejected authored settings");
305 }
306 return eve::editing::applied<void>();
