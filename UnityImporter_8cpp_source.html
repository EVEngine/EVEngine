13Result<std::string> textFile(const UnityProjectImportRequest& request, const std::string& path) {
14 const auto found = request.files.find(path);
15 if (found == request.files.end())
16 return detail::failure<std::string>(DiagnosticCode::NotFound, "Unity source file was not supplied", path);
17 if (found->second.size() > request.limits.maximumSourceBytes)
18 return detail::failure<std::string>(DiagnosticCode::InvalidArgument, "Unity source exceeds budget", path);
19 if (std::find(found->second.begin(), found->second.end(), std::uint8_t(0)) != found->second.end())
20 return detail::failure<std::string>(DiagnosticCode::Unsupported,
21 "Unity adapter requires Force Text serialization", path);
22 return Result<std::string>::success({found->second.begin(), found->second.end()});
23}

25std::optional<std::string> firstMatch(std::string_view text, const std::regex& expression,
26 std::size_t group = 1) {
27 std::match_results<std::string_view::const_iterator> match;
28 if (!std::regex_search(text.begin(), text.end(), match, expression) || group >= match.size()) return std::nullopt;
29 return std::string(match[group].first, match[group].second);

67std::vector<std::string> sectionEntries(std::string_view yaml, std::string_view heading) {
68 const auto start = yaml.find(heading);
69 if (start == std::string_view::npos) return {};
70 const auto lineEnd = yaml.find('\n', start);
71 if (lineEnd == std::string_view::npos) return {};
72 std::vector<std::string> entries;
73 std::string current;
74 std::istringstream lines(std::string(yaml.substr(lineEnd + 1)));
75 std::string line;
76 while (std::getline(lines, line)) {
77 if (line.starts_with(" m_") && !line.starts_with(" ")) break;
78 if (line.starts_with(" - ") || line.starts_with(" - ")) {
79 if (!current.empty()) entries.push_back(std::move(current));
80 current = line + "\n";
81 } else if (!current.empty()) {
82 current += line + "\n";
83 }
84 }
85 if (!current.empty()) entries.push_back(std::move(current));
86 return entries;
87}

89Result<CanonicalTerrainInput> parseTerrain(const UnityProjectImportRequest& request,
90 const std::map<std::string, std::string>& paths,
91 std::vector<ImportFinding>& findings) {
92 auto textResult = textFile(request, request.terrainDataPath);
93 if (!textResult) return Result<CanonicalTerrainInput>::failure(textResult.status());
94 const std::string yaml = std::move(textResult).takeValue();
95 auto resolutionText = firstMatch(yaml, std::regex(R"((?:m_HeightmapResolution|m_Resolution):\s*([0-9]+))"));
96 auto scaleMatchX = firstMatch(yaml, std::regex(R"(m_HeightmapScale:\s*\{x:\s*([^,]+),\s*y:\s*([^,]+),\s*z:\s*([^}]+)\})"), 1);
97 auto scaleMatchY = firstMatch(yaml, std::regex(R"(m_HeightmapScale:\s*\{x:\s*([^,]+),\s*y:\s*([^,]+),\s*z:\s*([^}]+)\})"), 2);
98 auto scaleMatchZ = firstMatch(yaml, std::regex(R"(m_HeightmapScale:\s*\{x:\s*([^,]+),\s*y:\s*([^,]+),\s*z:\s*([^}]+)\})"), 3);
99 auto heightsHex = firstMatch(yaml, std::regex(R"(m_Heights:\s*([0-9a-fA-F]+))"));
100 if (!resolutionText || !scaleMatchX || !scaleMatchY || !scaleMatchZ || !heightsHex)
101 return detail::failure<CanonicalTerrainInput>(DiagnosticCode::ParseError,
102 "Unity TerrainData heightmap fields are incomplete",
103 request.terrainDataPath);
104 std::uint64_t resolution = 0;
105 for (char digit : *resolutionText) {
106 if (resolution > (std::numeric_limits<std::uint32_t>::max() - (digit - '0')) / 10)
107 return detail::failure<CanonicalTerrainInput>(DiagnosticCode::InvalidArgument,
108 "Unity terrain resolution overflows");
109 resolution = resolution * 10 + (digit - '0');
110 }
111 if (resolution < 2 || resolution > request.limits.maximumVerticesPerPrimitive ||
112 resolution > std::numeric_limits<std::uint64_t>::max() / resolution)
113 return detail::failure<CanonicalTerrainInput>(DiagnosticCode::InvalidArgument,
114 "Unity terrain resolution is outside limits");
115 const std::uint64_t samples = resolution * resolution;
116 if (heightsHex->size() != samples * 4)
117 return detail::failure<CanonicalTerrainInput>(DiagnosticCode::ParseError,
118 "Unity m_Heights must contain one little-endian UInt16 per sample");
119 auto spacingX = parseFloat(*scaleMatchX, "m_HeightmapScale.x");
120 auto scaleY = parseFloat(*scaleMatchY, "m_HeightmapScale.y");
121 auto spacingZ = parseFloat(*scaleMatchZ, "m_HeightmapScale.z");
122 if (!spacingX || !scaleY || !spacingZ)
123 return detail::failure<CanonicalTerrainInput>(DiagnosticCode::ParseError,
124 "Unity terrain scale is invalid");
125 const auto nibble = [](char value) -> int {
126 if (value >= '0' && value <= '9') return value - '0';
127 if (value >= 'a' && value <= 'f') return value - 'a' + 10;
128 if (value >= 'A' && value <= 'F') return value - 'A' + 10;
129 return -1;
130 };
131 std::vector<std::uint16_t> sourceHeights(samples);
132 for (std::size_t index = 0; index < samples; ++index) {
133 const std::size_t at = index * 4;
134 sourceHeights[index] = static_cast<std::uint16_t>((nibble((*heightsHex)[at]) << 4) |
135 nibble((*heightsHex)[at + 1]) |
136 (nibble((*heightsHex)[at + 2]) << 12) |
137 (nibble((*heightsHex)[at + 3]) << 8));
138 }
139 CanonicalTerrainInput terrain;
140 terrain.width = terrain.height = static_cast<std::uint32_t>(resolution);
141 terrain.spacingX = spacingX.value(); terrain.spacingZ = spacingZ.value();
142 terrain.heightsMeters.resize(samples);
143 // Unity rows advance +Z; canonical forward is -Z, so reverse rows while preserving winding.
144 for (std::uint32_t row = 0; row < terrain.height; ++row)
145 for (std::uint32_t column = 0; column < terrain.width; ++column)
146 terrain.heightsMeters[std::size_t(row) * terrain.width + column] =
147 float(sourceHeights[std::size_t(terrain.height - 1 - row) * terrain.width + column]) /
148 65535.0f * scaleY.value();
149 terrain.sourceTransform = {{"source", Value("unity-left-handed-x-right-y-up-z-forward")},
150 {"conversion", Value("reflect-z-and-reverse-height-rows")},
151 {"heightScaleMeters", Value(double(scaleY.value()))}};
152
153 for (const auto& entry : sectionEntries(yaml, "m_TerrainLayers:")) {
154 auto guid = firstMatch(entry, std::regex(R"(guid:\s*([0-9a-fA-F]{32}))"));
155 if (!guid) continue;
156 CanonicalTerrainLayer layer;
157 const auto resolved = paths.find(*guid);
158 layer.name = resolved == paths.end() ? *guid : resolved->second;
159 if (resolved != paths.end()) {
160 auto layerText = textFile(request, resolved->second);
161 if (layerText) {
162 if (auto diffuse = firstMatch(layerText.value(), std::regex(R"(m_DiffuseTexture:.*guid:\s*([0-9a-fA-F]{32}))")))
163 layer.diffuseSource = "unity-guid:" + *diffuse;
164 if (auto normal = firstMatch(layerText.value(), std::regex(R"(m_NormalMapTexture:.*guid:\s*([0-9a-fA-F]{32}))")))
165 layer.normalSource = "unity-guid:" + *normal;
166 if (auto tile = firstMatch(layerText.value(), std::regex(R"(m_TileSize:\s*\{x:\s*([^,]+),)"))) {
167 auto parsed = parseFloat(*tile, resolved->second + ".m_TileSize.x");
168 if (parsed) layer.tileSizeMeters = parsed.value();
169 }
170 }
171 }
172 terrain.layers.push_back(std::move(layer));
173 }
174
175 std::vector<std::string> prototypes;
176 for (const auto& entry : sectionEntries(yaml, "m_TreePrototypes:")) {
177 auto guid = firstMatch(entry, std::regex(R"(guid:\s*([0-9a-fA-F]{32}))"));
178 prototypes.push_back(guid ? "unity-guid:" + *guid : "unity-missing-prototype");
179 }
180 for (const auto& entry : sectionEntries(yaml, "m_TreeInstances:")) {
181 auto position = std::regex(R"(position:\s*\{x:\s*([^,]+),\s*y:\s*([^,]+),\s*z:\s*([^}]+)\})");
182 auto x = firstMatch(entry, position, 1); auto y = firstMatch(entry, position, 2); auto z = firstMatch(entry, position, 3);
183 auto prototype = firstMatch(entry, std::regex(R"(prototypeIndex:\s*([0-9]+))"));
184 if (!x || !y || !z || !prototype) continue;
185 const auto prototypeValue = parseUnsignedText(*prototype);
186 if (!prototypeValue || *prototypeValue >= prototypes.size())
187 return detail::failure<CanonicalTerrainInput>(DiagnosticCode::ParseError,
188 "Unity tree prototype index is invalid");
189 const std::size_t prototypeIndex = static_cast<std::size_t>(*prototypeValue);
190 auto px = parseFloat(*x, "tree.position.x"); auto py = parseFloat(*y, "tree.position.y");
191 auto pz = parseFloat(*z, "tree.position.z");
192 if (!px || !py || !pz) return detail::failure<CanonicalTerrainInput>(DiagnosticCode::ParseError,
193 "Unity tree position is invalid");
194 CanonicalTerrainInstance instance;
195 instance.prototype = prototypes[prototypeIndex];
196 instance.position[0] = px.value() * spacingX.value() * float(resolution - 1);
197 instance.position[1] = py.value() * scaleY.value();
198 instance.position[2] = -pz.value() * spacingZ.value() * float(resolution - 1);
199 if (auto rotation = firstMatch(entry, std::regex(R"(rotation:\s*([^\r\n]+))"))) {
200 auto value = parseFloat(*rotation, "tree.rotation");
201 if (value) { instance.rotation[1] = std::sin(-value.value() * 0.5f);
202 instance.rotation[3] = std::cos(-value.value() * 0.5f); }
203 }
204 if (auto width = firstMatch(entry, std::regex(R"(widthScale:\s*([^\r\n]+))"))) {
205 auto value = parseFloat(*width, "tree.widthScale");
206 if (value) instance.scale[0] = instance.scale[2] = value.value();
207 }
208 if (auto height = firstMatch(entry, std::regex(R"(heightScale:\s*([^\r\n]+))"))) {
209 auto value = parseFloat(*height, "tree.heightScale");
210 if (value) instance.scale[1] = value.value();
211 }
212 terrain.instances.push_back(std::move(instance));
213 }
214 if (yaml.find("m_DetailPrototypes:") != std::string::npos)
215 findings.push_back({request.terrainDataPath, "TerrainData.detail-prototypes",
216 ImportDisposition::PreservedSource,
217 "detail prototype metadata is retained but detail density maps require a later adapter"});
218 findings.push_back({request.terrainDataPath, "TerrainData.heightmap",
219 ImportDisposition::Translated, "UInt16 heights converted to metres and canonical Z"});
220 return Result<CanonicalTerrainInput>::success(std::move(terrain));
221}

229std::vector<UnityObject> documents(std::string_view yaml) {
230 const std::regex header(R"(^---\s*!u!([0-9]+)\s*&(-?[0-9]+).*$)", std::regex::multiline);
231 struct Start { std::size_t header; std::size_t body; UnityObject object; };
232 std::vector<Start> starts;
233 for (std::cregex_iterator it(yaml.data(), yaml.data() + yaml.size(), header), end; it != end; ++it) {
234 const std::string signedText = (*it)[2].str();
235 std::int64_t signedId = 0;
236 const auto parsedId = std::from_chars(signedText.data(), signedText.data() + signedText.size(), signedId);
237 const auto classText = (*it)[1].str();
238 const auto classId = parseUnsignedText(classText);
239 if (parsedId.ec != std::errc{} || parsedId.ptr != signedText.data() + signedText.size() ||
240 signedId < 0 || !classId || *classId > std::numeric_limits<std::uint32_t>::max()) continue;
241 starts.push_back({static_cast<std::size_t>((*it).position()),
242 static_cast<std::size_t>((*it).position() + (*it).length()),
243 {static_cast<std::uint64_t>(signedId), static_cast<std::uint32_t>(*classId), {}}});
244 }
245 std::vector<UnityObject> result;
246 for (std::size_t index = 0; index < starts.size(); ++index) {
247 const std::size_t end = index + 1 < starts.size() ? starts[index + 1].header : yaml.size();
248 starts[index].object.body = std::string(yaml.substr(starts[index].body, end - starts[index].body));
249 result.push_back(std::move(starts[index].object));
250 }
251 return result;
252}

254Result<void> appendPrefab(const UnityProjectImportRequest& request, PreparedAssetImport& output,
255 std::vector<ImportFinding>& findings) {
256 if (request.prefabPath.empty()) return Result<void>::success();
257 auto text = textFile(request, request.prefabPath);
258 if (!text) return Result<void>::failure(text.status());
259 const auto prefabGuid = guidFromMeta(request, request.prefabPath);
260 if (!prefabGuid)
261 return Result<void>::failure(Diagnostic::error(DiagnosticCode::NotFound,
262 "Unity prefab .meta GUID is required",
263 request.prefabPath + ".meta", {}, "asset.import.unity"));
264 const auto objects = documents(text.value());
265 std::map<std::uint64_t, std::string> gameNames;
266 for (const auto& object : objects) {
267 if (object.classId == 1) {
268 auto name = firstMatch(object.body, std::regex(R"(m_Name:\s*([^\r\n]+))"));
269 gameNames[object.fileId] = name.value_or("GameObject");
270 } else if (object.classId == 114) {
271 findings.push_back({request.prefabPath, "MonoBehaviour:" + std::to_string(object.fileId),
272 ImportDisposition::Unsupported,
273 "C# behaviour is not executed or translated by the data importer"});
274 }
275 }
276 Value::Array nodes;
277 for (const auto& object : objects) {
278 if (object.classId != 4 && object.classId != 224) continue;
279 auto game = firstMatch(object.body, std::regex(R"(m_GameObject:\s*\{fileID:\s*(-?[0-9]+)\})"));
280 auto father = firstMatch(object.body, std::regex(R"(m_Father:\s*\{fileID:\s*(-?[0-9]+)\})"));
281 const std::regex position(R"(m_LocalPosition:\s*\{x:\s*([^,]+),\s*y:\s*([^,]+),\s*z:\s*([^}]+)\})");
282 const std::regex rotation(R"(m_LocalRotation:\s*\{x:\s*([^,]+),\s*y:\s*([^,]+),\s*z:\s*([^,]+),\s*w:\s*([^}]+)\})");
283 const std::regex scale(R"(m_LocalScale:\s*\{x:\s*([^,]+),\s*y:\s*([^,]+),\s*z:\s*([^}]+)\})");
284 if (!game) continue;
285 auto vector = [&](const std::regex& regex, std::size_t count, float* target, bool reflect) -> Result<void> {
286 for (std::size_t component = 0; component < count; ++component) {
287 auto field = firstMatch(object.body, regex, component + 1);
288 if (!field) return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError,
289 "Unity transform field is missing",
290 request.prefabPath, {}, "asset.import.unity"));
291 auto parsed = parseFloat(*field, request.prefabPath);
292 if (!parsed) return Result<void>::failure(parsed.status());
293 target[component] = parsed.value();
294 }
295 if (reflect) target[2] = -target[2];
296 return Result<void>::success();
297 };
298 float p[3], q[4], s[3];
299 auto pResult = vector(position, 3, p, true); if (!pResult) return pResult;
300 auto qResult = vector(rotation, 4, q, false); if (!qResult) return qResult;
301 q[0] = -q[0]; q[1] = -q[1];
302 auto sResult = vector(scale, 3, s, false); if (!sResult) return sResult;
303 const auto gameIdValue = parseUnsignedText(*game);
304 const auto parentIdValue = father ? parseUnsignedText(*father) : std::optional<std::uint64_t>(0);
305 if (!gameIdValue || !parentIdValue)
306 return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError,
307 "Unity transform fileID overflows",
308 request.prefabPath, {}, "asset.import.unity"));
309 const auto gameId = *gameIdValue;
310 const std::uint64_t parentId = *parentIdValue;
311 Value::Object node;
312 node["objectId"] = Value(request.package.packageId.child("unity:" + *prefabGuid + ":" +
313 std::to_string(object.fileId)).format());
314 node["sourceFileId"] = Value(static_cast<std::int64_t>(object.fileId));
315 node["name"] = Value(gameNames.contains(gameId) ? gameNames[gameId] : "GameObject");
316 node["parentSourceFileId"] = Value(static_cast<std::int64_t>(parentId));
317 node["position"] = Value(Value::Array{Value(double(p[0])), Value(double(p[1])), Value(double(p[2]))});
318 node["rotation"] = Value(Value::Array{Value(double(q[0])), Value(double(q[1])), Value(double(q[2])), Value(double(q[3]))});
319 node["scale"] = Value(Value::Array{Value(double(s[0])), Value(double(s[1])), Value(double(s[2]))});
320 nodes.emplace_back(std::move(node));
321 }
322 const PersistentId sceneId = request.package.packageId.child("unity-prefab:" + *prefabGuid);
323 Value::Object definition;
324 definition["schema"] = Value("eve.scene-template"); definition["schemaVersion"] = Value(std::int64_t(1));
325 definition["sourceGuid"] = Value(*prefabGuid); definition["nodes"] = Value(std::move(nodes));
326 definition["coordinateSystem"] = Value("right-handed-x-right-y-up-minus-z-forward");
327 auto encoded = Value(std::move(definition)).toJson();
328 if (!encoded) return Result<void>::failure(encoded.status());
329 std::string definitionText = std::move(encoded).takeValue();
330 const std::string path = "assets/" + sceneId.format() + "/asset.json";
331 auto reference = detail::assetRef(sceneId);
332 if (!reference) return Result<void>::failure(reference.status());
333 output.manifest.assets.push_back({std::move(reference).takeValue(), "eve.scene-template", SchemaVersion(1), path,
334 detail::sha256(std::span<const std::uint8_t>(
335 reinterpret_cast<const std::uint8_t*>(definitionText.data()),
336 definitionText.size())), {"scene", "source:unity-prefab"}});
337 output.entries.push_back({path, {definitionText.begin(), definitionText.end()}});
338 if (output.manifest.entrypoints.empty()) {
339 auto entrypoint = detail::assetRef(sceneId);
340 if (!entrypoint) return Result<void>::failure(entrypoint.status());
341 output.manifest.entrypoints.emplace("default", std::move(entrypoint).takeValue());
342 }
343 findings.push_back({request.prefabPath, "Prefab.GameObject.Transform",
344 ImportDisposition::Translated, "GUID/fileID hierarchy converted to canonical TRS"});
345 return Result<void>::success();
346}

350Result<PreparedAssetImport> prepareUnityProjectImport(const UnityProjectImportRequest& request) {
351 if (request.files.empty() || request.package.packageId.isNil())
352 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
353 "Unity project files and package identity are required");
354 std::uint64_t totalBytes = 0;
355 for (const auto& [path, bytes] : request.files) {
356 if (path.empty() || path.front() == '/' || path.find("..") != std::string::npos ||
357 bytes.size() > request.limits.maximumSourceBytes ||
358 totalBytes > request.limits.maximumSourceBytes - bytes.size())
359 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
360 "Unity project path or total source budget is invalid", path);
361 totalBytes += bytes.size();
362 }
363 std::vector<ImportFinding> findings;
364 PreparedAssetImport output;
365 if (!request.terrainDataPath.empty()) {
366 auto terrain = parseTerrain(request, guidPaths(request), findings);
367 if (!terrain) return Result<PreparedAssetImport>::failure(terrain.status());
368 auto prepared = prepareCanonicalTerrainImport(request.package, terrain.value(), "eve.unity-terrain/1",
369 request.limits);
370 if (!prepared) return Result<PreparedAssetImport>::failure(prepared.status());
371 output = std::move(prepared).takeValue();
372 } else {
373 auto manifest = detail::baseManifest(request.package, "eve.unity-prefab/1");
374 if (!manifest) return Result<PreparedAssetImport>::failure(manifest.status());
375 output.manifest = std::move(manifest).takeValue();
376 }
377 auto prefab = appendPrefab(request, output, findings);
378 if (!prefab) return Result<PreparedAssetImport>::failure(prefab.status());
379 output.manifest.provenance["sourceEngine"] = Value("unity");
380 output.manifest.provenance["sourceCoordinateSystem"] = Value("left-handed-x-right-y-up-z-forward");
381 output.findings.insert(output.findings.end(), std::make_move_iterator(findings.begin()),
382 std::make_move_iterator(findings.end()));
383 for (const auto& asset : output.manifest.assets) {
384 std::string sourceObject = request.terrainDataPath + "#" + asset.type;
385 if (asset.type == "eve.scene-template") sourceObject = request.prefabPath + "#Transform";
386 output.sourceMappings.push_back({std::move(sourceObject), asset.asset});
387 }
388 auto report = detail::finalizeImportReport(output, request.package, "unity", "force-text",
389 {{"serialization", Value("force-text")}});
390 if (!report) return Result<PreparedAssetImport>::failure(report.status());
391 return Result<PreparedAssetImport>::success(std::move(output));
392}
