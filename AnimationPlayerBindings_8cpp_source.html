15std::string formatLockAxes(RootMotionLockAxes mask) {
16 const auto bits = static_cast<std::uint8_t>(mask);
17 if (bits == 0) return "none";
18 if (bits == static_cast<std::uint8_t>(RootMotionLockAxes::All)) return "xyz";
19 if (bits == static_cast<std::uint8_t>(RootMotionLockAxes::HorizontalXZ)) return "xz";
20 if (bits == static_cast<std::uint8_t>(RootMotionLockAxes::VerticalY)) return "y";
21 if (bits == static_cast<std::uint8_t>(RootMotionLockAxes::X)) return "x";
22 if (bits == static_cast<std::uint8_t>(RootMotionLockAxes::Z)) return "z";
23 if (bits == static_cast<std::uint8_t>(RootMotionLockAxes::X | RootMotionLockAxes::Y)) return "xy";
24 if (bits == static_cast<std::uint8_t>(RootMotionLockAxes::Y | RootMotionLockAxes::Z)) return "yz";
25 return "xyz";
26}
