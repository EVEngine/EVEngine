16inline glm::vec3 faceDirection(uint32_t face, float u, float v) {
17 switch (face) {
18 case 0: return glm::normalize(glm::vec3(1.f, -v, -u));
19 case 1: return glm::normalize(glm::vec3(-1.f, -v, u));
20 case 2: return glm::normalize(glm::vec3(u, 1.f, v));
21 case 3: return glm::normalize(glm::vec3(u, -1.f, -v));
22 case 4: return glm::normalize(glm::vec3(u, -v, 1.f));
23 default: return glm::normalize(glm::vec3(-u, -v, -1.f));
24 }
25}

27inline void directionFaceUv(const glm::vec3 &d, uint32_t &face, float &u, float &v) {
28 const glm::vec3 a = glm::abs(d);
29 if (a.x >= a.y && a.x >= a.z) {
30 if (d.x >= 0.f) {
31 face = 0;
32 u = -d.z / a.x;
33 v = -d.y / a.x;
34 } else {
35 face = 1;
36 u = d.z / a.x;
37 v = -d.y / a.x;
38 }
39 } else if (a.y >= a.z) {
40 if (d.y >= 0.f) {
41 face = 2;
42 u = d.x / a.y;
43 v = d.z / a.y;
44 } else {
45 face = 3;
46 u = d.x / a.y;
47 v = -d.z / a.y;
48 }
49 } else if (d.z >= 0.f) {
50 face = 4;
51 u = d.x / a.z;
52 v = -d.y / a.z;
53 } else {
54 face = 5;
55 u = -d.x / a.z;
56 v = -d.y / a.z;
57 }
58}

60inline glm::vec4 sampleBase(const uint8_t *faces, uint32_t size, const glm::vec3 &direction) {
61 uint32_t face = 0;
62 float u = 0.f;
63 float v = 0.f;
64 directionFaceUv(direction, face, u, v);
65 const float x = (u * 0.5f + 0.5f) * float(size) - 0.5f;
66 const float y = (v * 0.5f + 0.5f) * float(size) - 0.5f;
67 const int x0 = int(std::floor(x));
68 const int y0 = int(std::floor(y));
69 const int x1 = x0 + 1;
70 const int y1 = y0 + 1;
71 const float tx = x - float(x0);
72 const float ty = y - float(y0);
73 auto load = [&](int px, int py) {
74 const float tapU = (2.f * (float(px) + 0.5f) / float(size)) - 1.f;
75 const float tapV = (2.f * (float(py) + 0.5f) / float(size)) - 1.f;
76 const glm::vec3 tapDirection = faceDirection(face, tapU, tapV);
77 uint32_t tapFace = 0;
78 float resolvedU = 0.f;
79 float resolvedV = 0.f;
80 directionFaceUv(tapDirection, tapFace, resolvedU, resolvedV);
81 const uint32_t resolvedX = uint32_t(std::clamp(
82 int((resolvedU * 0.5f + 0.5f) * float(size)), 0, int(size) - 1));
83 const uint32_t resolvedY = uint32_t(std::clamp(
84 int((resolvedV * 0.5f + 0.5f) * float(size)), 0, int(size) - 1));
85 const size_t faceBase = size_t(tapFace) * size_t(size) * size_t(size) * 4u;
86 const size_t i = faceBase + (size_t(resolvedY) * size + resolvedX) * 4u;
87 return glm::vec4(faces[i], faces[i + 1u], faces[i + 2u], faces[i + 3u]);
88 };
89 return glm::mix(glm::mix(load(x0, y0), load(x1, y0), tx),
90 glm::mix(load(x0, y1), load(x1, y1), tx), ty);
91}

93inline float radicalInverse(uint32_t bits) {
94 bits = (bits << 16u) | (bits >> 16u);
95 bits = ((bits & 0x55555555u) << 1u) | ((bits & 0xAAAAAAAAu) >> 1u);
96 bits = ((bits & 0x33333333u) << 2u) | ((bits & 0xCCCCCCCCu) >> 2u);
97 bits = ((bits & 0x0F0F0F0Fu) << 4u) | ((bits & 0xF0F0F0F0u) >> 4u);
98 bits = ((bits & 0x00FF00FFu) << 8u) | ((bits & 0xFF00FF00u) >> 8u);
99 return float(bits) * 2.3283064365386963e-10f;
100}

102inline glm::vec3 importanceSample(float x, float y, const glm::vec3 &n, float roughness) {
103 constexpr float kTau = 6.2831853071795864769f;
104 const float a = std::max(roughness * roughness, 0.001f);
105 const float a2 = a * a;
106 const float phi = kTau * x;
107 const float cosTheta = std::sqrt((1.f - y) / std::max(1.f + (a2 - 1.f) * y, 1e-6f));
108 const float sinTheta = std::sqrt(std::max(1.f - cosTheta * cosTheta, 0.f));
109 const glm::vec3 hT(std::cos(phi) * sinTheta, std::sin(phi) * sinTheta, cosTheta);
110 const glm::vec3 up = std::abs(n.z) < 0.999f ? glm::vec3(0.f, 0.f, 1.f)
111 : glm::vec3(1.f, 0.f, 0.f);
112 const glm::vec3 tangent = glm::normalize(glm::cross(up, n));
113 const glm::vec3 bitangent = glm::cross(n, tangent);
114 return glm::normalize(tangent * hT.x + bitangent * hT.y + n * hT.z);
115}

117inline glm::vec3 cosineSample(float x, float y, const glm::vec3 &n) {
118 constexpr float kTau = 6.2831853071795864769f;
119 const float phi = kTau * x;
120 const float radius = std::sqrt(y);
121 const glm::vec3 local(radius * std::cos(phi), radius * std::sin(phi),
122 std::sqrt(std::max(1.f - y, 0.f)));
123 const glm::vec3 up = std::abs(n.z) < 0.999f ? glm::vec3(0.f, 0.f, 1.f)
124 : glm::vec3(1.f, 0.f, 0.f);
125 const glm::vec3 tangent = glm::normalize(glm::cross(up, n));
126 const glm::vec3 bitangent = glm::cross(n, tangent);
127 return glm::normalize(tangent * local.x + bitangent * local.y + n * local.z);
128}

140inline std::vector<uint8_t> buildGgxCubemapMipChain(const uint8_t *rgbaFaces, uint32_t faceSize,
141 uint32_t mipLevels,
142 uint32_t sampleCount = 32u) {
143 if (!rgbaFaces || faceSize == 0u || mipLevels == 0u) return {};
144 sampleCount = std::max(sampleCount, 1u);
145 struct CacheEntry {
146 uint64_t key = 0;
147 uint32_t faceSize = 0;
148 uint32_t mipLevels = 0;
149 uint32_t sampleCount = 0;
150 std::vector<uint8_t> bytes;
151 };
152 static std::mutex cacheMutex;
153 static std::vector<CacheEntry> cache;
154 const size_t sourceBytes = size_t(faceSize) * faceSize * 4u * 6u;
155 uint64_t key = 1469598103934665603ull;
156 for (size_t i = 0; i < sourceBytes; ++i) {
157 key ^= uint64_t(rgbaFaces[i]);
158 key *= 1099511628211ull;
159 }
160 key ^= uint64_t(faceSize) | (uint64_t(mipLevels) << 32u);
161 key *= 1099511628211ull;
162 key ^= uint64_t(sampleCount);
163 if (mipLevels > 1u) {
164 std::lock_guard<std::mutex> lock(cacheMutex);
165 for (const CacheEntry &entry : cache) {
166 if (entry.key == key && entry.faceSize == faceSize &&
167 entry.mipLevels == mipLevels && entry.sampleCount == sampleCount)
168 return entry.bytes;
169 }
170 }
171
172 std::vector<uint8_t> packed;
173 size_t totalBytes = 0;
174 for (uint32_t level = 0, size = faceSize; level < mipLevels;
175 ++level, size = std::max(size >> 1u, 1u))
176 totalBytes += size_t(size) * size * 4u * 6u;
177 packed.reserve(totalBytes);
178 packed.insert(packed.end(), rgbaFaces, rgbaFaces + sourceBytes);
179 if (mipLevels <= 1u) return packed;
180
181 for (uint32_t level = 1, size = std::max(faceSize >> 1u, 1u); level < mipLevels;
182 ++level, size = std::max(size >> 1u, 1u)) {
183 const float roughness = float(level) / float(mipLevels - 1u);
184 const bool diffuseIrradiance = mipLevels >= 3u && level + 1u == mipLevels;
185 const uint32_t levelSamples =
186 diffuseIrradiance
187 ? std::max(64u, sampleCount * 2u)
188 : std::max(8u, uint32_t(std::ceil(float(sampleCount) * roughness)));
189 for (uint32_t face = 0; face < 6u; ++face) {
190 for (uint32_t y = 0; y < size; ++y) {
191 for (uint32_t x = 0; x < size; ++x) {
192 const float u = (2.f * (float(x) + 0.5f) / float(size)) - 1.f;
193 const float v = (2.f * (float(y) + 0.5f) / float(size)) - 1.f;
194 const glm::vec3 n = cubemap_prefilter_detail::faceDirection(face, u, v);
195 glm::vec4 sum(0.f);
196 float weight = 0.f;
197 for (uint32_t i = 0; i < levelSamples; ++i) {
198 const float xiX = (float(i) + 0.5f) / float(levelSamples);
199 const float xiY = cubemap_prefilter_detail::radicalInverse(i);
200 if (diffuseIrradiance) {
201 const glm::vec3 l =
202 cubemap_prefilter_detail::cosineSample(xiX, xiY, n);
203 sum += cubemap_prefilter_detail::sampleBase(rgbaFaces, faceSize, l);
204 weight += 1.f;
205 continue;
206 }
207 const glm::vec3 h = cubemap_prefilter_detail::importanceSample(
208 xiX, xiY, n, roughness);
209 const glm::vec3 l = glm::normalize(2.f * glm::dot(n, h) * h - n);
210 const float noL = std::max(glm::dot(n, l), 0.f);
211 if (noL <= 0.f) continue;
212 sum += cubemap_prefilter_detail::sampleBase(rgbaFaces, faceSize, l) * noL;
213 weight += noL;
214 }
215 const glm::vec4 c = sum / std::max(weight, 1e-6f);
216 for (int channel = 0; channel < 4; ++channel)
217 packed.push_back(static_cast<uint8_t>(
218 std::clamp(std::lround(c[channel]), 0l, 255l)));
219 }
220 }
221 }
222 }
223 {
224 std::lock_guard<std::mutex> lock(cacheMutex);
225 if (cache.size() >= 2u) cache.erase(cache.begin());
226 cache.push_back(CacheEntry{key, faceSize, mipLevels, sampleCount, packed});
227 }
228 return packed;
229}
