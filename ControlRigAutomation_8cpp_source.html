53Result<editing::DomainOperation> ControlRigEditor::Impl::LiveTarget::prepare(const editing::Value& payload) const {
54 if (!payload.isWithinLimits(32, 2000000, 16000000))
55 return invalid<editing::DomainOperation>("Rig payload exceeds limits");
56 auto value = Value::fromJson(editing::editorValueToJson(payload));
57 if (!value.ok()) return Result<editing::DomainOperation>::failure(value.status());
58 auto decoded = animation::decodeControlRig(value.value());
59 if (!decoded.ok()) return Result<editing::DomainOperation>::failure(decoded.status());
60 auto valid = owner.validate(decoded.value());
61 if (!valid.ok()) return Result<editing::DomainOperation>::failure(valid.status());
62 return owner.document.replace(decoded.value());

70Result<void> ControlRigEditor::Impl::LiveTarget::commitDomainState(
71 std::unique_ptr<editing::IDomainOperationTarget> candidate) {
72 auto* document = dynamic_cast<animation_editing::ControlRigDocument*>(candidate.get());
73 if (!document || document->targetId() != targetId()) return invalid<void>("Wrong Rig candidate");
74 if (!owner.skeleton) return invalid<void>("initialize skeleton first");
75 auto compiled = animation::ControlRig::compile(document->definition(), *owner.skeleton);
76 if (!compiled.ok()) return Result<void>::failure(compiled.status());
77 animation::AnimPose preview;
78 const double time = std::min(owner.playhead, document->definition().duration);
79 auto evaluated = compiled.value()->evaluate(time, owner.input, preview);
80 if (!evaluated.ok()) return evaluated;
81 auto committed = owner.document.commitDomainState(std::move(candidate));
82 if (!committed.ok()) return committed;
83 owner.rig = std::move(compiled).takeValue();
84 owner.pose = std::move(preview);
85 owner.playhead = time;
86 return Result<void>::success();
87}
