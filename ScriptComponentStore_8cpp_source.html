47Result<Value> scriptComponentField(const ScriptComponentStoreKey& key, const std::string& field) {

60 Store& store = scriptComponentStore();

88void reconcileScriptComponentValues(const std::string& name, SchemaVersion version,
89 const std::vector<std::string>& fields, const std::vector<Value>& defaults) {
90 Store& store = scriptComponentStore();
91 bool touched = false;
92 for (auto& [key, values] : store) {
93 if (key.name != name || key.version != version) continue;
94 // 逐字段重建：旧的留着、新的取默认值、不再声明的丢掉。原地改 value 而不是插删键，
95 // 迭代器因此不会失效。
96 Value::Object rebuilt;
97 for (std::size_t index = 0; index < fields.size(); ++index) {
98 const Value* existing = values.isObject() ? values.find(fields[index]) : nullptr;
99 if (existing != nullptr) {
100 rebuilt.emplace(fields[index], *existing);
101 } else {
102 rebuilt.emplace(fields[index], index < defaults.size() ? defaults[index] : Value::null());
103 }
104 }
105 values = Value::object(std::move(rebuilt));
106 touched = true;
107 }
108 if (touched) bumpScriptComponentRevision();
109}

116 it = store.erase(it);

119 ++it;
120 }
121 }
