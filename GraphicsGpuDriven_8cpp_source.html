76void Graphics::createGpuDrivenVisResources(int width, int height) {
77 if (!gpuDrivenCaps_.gpuDrivenAvailable()) return;
78 if (gbufferVisRenderPass || gbufferSlots.empty()) return;
79 const uint32_t w = uint32_t(width);
80 const uint32_t h = uint32_t(height);
81 const vk::Format colorFmt = pickGBufferColorFormat(device);
82 const vk::Format depthFmt = vk::Format::eD32Sfloat;
83 const vk::Format visIDFmt = vk::Format::eR32G32Uint;
84 const vk::Format visBaryFmt = vk::Format::eR16G16Sfloat;
85
86 gbufferVisRenderPass =
87 device.createRenderPass()
88 .addSampledColorAttachment(colorFmt)
89 .addSampledColorAttachment(colorFmt)
90 .addSampledColorAttachment(colorFmt)
91 .addSampledColorAttachment(visIDFmt)
92 .addSampledColorAttachment(visBaryFmt)
93 .addSampledDepthAttachment(depthFmt)
94 .addSubpass(vkb::SubpassBuilder()
95 .addAttachmentRef(0, vk::ImageLayout::eColorAttachmentOptimal)
96 .addAttachmentRef(1, vk::ImageLayout::eColorAttachmentOptimal)
97 .addAttachmentRef(2, vk::ImageLayout::eColorAttachmentOptimal)
98 .addAttachmentRef(3, vk::ImageLayout::eColorAttachmentOptimal)
99 .addAttachmentRef(4, vk::ImageLayout::eColorAttachmentOptimal)
100 .setDepthStencilAttachment(
101 5, vk::ImageLayout::eDepthStencilAttachmentOptimal))
102 .addExternalShaderReadDependencies()
103 .build();
104 for (auto &slot : gbufferSlots) {
105 slot.visFramebuffer = gbufferVisRenderPass.createFramebuffer(
106 device, w, h,
107 {slot.normal.asAttachment(), slot.depthColor.asAttachment(), slot.albedo.asAttachment(),
108 slot.visID.asAttachment(), slot.visBary.asAttachment(), slot.depth.asAttachment()});
109 }
110
111 if (!mesh3dGpuDrivenPipelineLayout) return;
112 // Instance vis pass: no vertex input (fetches from the pooled buffers via
113 // gl_VertexIndex); draws non-indexed with the cull chain's NI commands.
114 {
115 std::vector<uint32_t> visVert(mesh3d_gbuffer_vis_vert_spv,
116 mesh3d_gbuffer_vis_vert_spv +
117 mesh3d_gbuffer_vis_vert_spv_count);
118 std::vector<uint32_t> visFrag(mesh3d_gbuffer_vis_frag_spv,
119 mesh3d_gbuffer_vis_frag_spv +
120 mesh3d_gbuffer_vis_frag_spv_count);
121 vk::ShaderModule visVertModule =
122 vkb::PipelineBuilder::createShaderModule(device.instance, visVert);
123 vk::ShaderModule visFragModule =
124 vkb::PipelineBuilder::createShaderModule(device.instance, visFrag);
125 gbufferVisPipeline =
126 device.createPipeline()
127 .useClassicPipeline(visVertModule, visFragModule)
128 .setPipelineLayout(mesh3dGpuDrivenPipelineLayout)
129 .setDynamicStatesViewportScissor()
130 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
131 vk::CullModeFlagBits::eNone, vk::FrontFace::eClockwise)
132 .setDepthStencil(true, true, vk::CompareOp::eLess)
133 .setColorAttachmentCount(5)
134 .build(gbufferVisRenderPass);
135 device->destroyShaderModule(visVertModule);
136 device->destroyShaderModule(visFragModule);
137 }
138 // VG variant: same vis attachments, cluster-stream fetch (no vertex input).
139 {
140 std::vector<uint32_t> vgVisVert(mesh3d_gbuffer_vgvis_vert_spv,
141 mesh3d_gbuffer_vgvis_vert_spv +
142 mesh3d_gbuffer_vgvis_vert_spv_count);
143 std::vector<uint32_t> vgVisFrag(mesh3d_gbuffer_vgvis_frag_spv,
144 mesh3d_gbuffer_vgvis_frag_spv +
145 mesh3d_gbuffer_vgvis_frag_spv_count);
146 vk::ShaderModule vgVisVertModule =
147 vkb::PipelineBuilder::createShaderModule(device.instance, vgVisVert);
148 vk::ShaderModule vgVisFragModule =
149 vkb::PipelineBuilder::createShaderModule(device.instance, vgVisFrag);
150 gbufferVgVisPipeline =
151 device.createPipeline()
152 .useClassicPipeline(vgVisVertModule, vgVisFragModule)
153 .setPipelineLayout(mesh3dGpuDrivenPipelineLayout)
154 .setDynamicStatesViewportScissor()
155 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
156 vk::CullModeFlagBits::eNone, vk::FrontFace::eClockwise)
157 .setDepthStencil(true, true, vk::CompareOp::eLess)
158 .setColorAttachmentCount(5)
159 .build(gbufferVisRenderPass);
160 device->destroyShaderModule(vgVisVertModule);
161 device->destroyShaderModule(vgVisFragModule);
162 }
163}

421void Graphics::growGpuVertexPool(uint32_t needVertices, uint32_t needIndices) {
422 // Pool growth reallocates the buffers; a pending frame may still be reading
423 // them, so drain the GPU first (rare path: first-time mesh registration).
424 device->waitIdle();
425 const uint32_t oldVerts = gpuVertexPool_.vertexCount;
426 const uint32_t oldInds = gpuVertexPool_.indexCount;
427 const auto hostMem = kHostVisibleCoherent;
428 auto copyInto = [&](vkb::GenericBuffer &dst, vkb::GenericBuffer &src, vk::DeviceSize oldBytes,
429 vk::DeviceSize newBytes) {
430 vkb::GenericBuffer grown(device, vk::BufferUsageFlagBits::eStorageBuffer, newBytes,
431 hostMem);
432 if (oldBytes > 0) {
433 void *srcMap = src.map();
434 void *dstMap = grown.map();
435 std::memcpy(dstMap, srcMap, oldBytes);
436 grown.unmap();
437 src.unmap();
438 }
439 src.release();
440 dst = std::move(grown);
441 };
442 const uint32_t newVerts = std::max(needVertices, oldVerts * 2u);
443 const uint32_t newInds = std::max(needIndices, oldInds * 2u);
444 copyInto(gpuVertexPool_.positions, gpuVertexPool_.positions,
445 vk::DeviceSize(oldVerts) * sizeof(glm::vec4),
446 vk::DeviceSize(newVerts) * sizeof(glm::vec4));
447 copyInto(gpuVertexPool_.normals, gpuVertexPool_.normals,
448 vk::DeviceSize(oldVerts) * sizeof(glm::vec4),
449 vk::DeviceSize(newVerts) * sizeof(glm::vec4));
450 copyInto(gpuVertexPool_.uvs, gpuVertexPool_.uvs,
451 vk::DeviceSize(oldVerts) * sizeof(glm::vec2),
452 vk::DeviceSize(newVerts) * sizeof(glm::vec2));
453 copyInto(gpuVertexPool_.indices, gpuVertexPool_.indices,
454 vk::DeviceSize(oldInds) * sizeof(uint32_t),
455 vk::DeviceSize(newInds) * sizeof(uint32_t));
456 bindGpuVertexPoolBindless();
457}

488 if (!gpu.vertices.buffer || !gpu.indices.buffer) return;
489 ensureGpuVertexPool();
490 const uint32_t nVerts = gpu.record.vertexCount;
491 const uint32_t nInds = gpu.record.indexCount;
492 if (nVerts == 0 || nInds == 0) return;
493 const uint32_t newVerts = gpuVertexPool_.vertexCount + nVerts;
494 const uint32_t newInds = gpuVertexPool_.indexCount + nInds;
495 const uint32_t capVerts =
496 uint32_t(gpuVertexPool_.positions.size / sizeof(glm::vec4));
497 const uint32_t capInds = uint32_t(gpuVertexPool_.indices.size / sizeof(uint32_t));
498 if (newVerts > capVerts || newInds > capInds) {
499 growGpuVertexPool(newVerts, newInds);
500 }
501
502 void *vMap = gpu.vertices.map();
503 void *iMap = gpu.indices.map();
504 if (!vMap || !iMap) return;
505 auto *verts = static_cast<const MeshVertex *>(vMap);
506 auto *posDst = static_cast<glm::vec4 *>(gpuVertexPool_.positions.map());
507 auto *nrmDst = static_cast<glm::vec4 *>(gpuVertexPool_.normals.map());
508 auto *uvDst = static_cast<glm::vec2 *>(gpuVertexPool_.uvs.map());
509 auto *idxDst = static_cast<uint32_t *>(gpuVertexPool_.indices.map());
510 if (!posDst || !nrmDst || !uvDst || !idxDst) {
511 gpu.vertices.unmap();
512 gpu.indices.unmap();
513 return;
514 }
515 posDst += gpuVertexPool_.vertexCount;
516 nrmDst += gpuVertexPool_.vertexCount;
517 uvDst += gpuVertexPool_.vertexCount;
518 idxDst += gpuVertexPool_.indexCount;
519 for (uint32_t i = 0; i < nVerts; ++i) {
520 posDst[i] = glm::vec4(verts[i].pos, 0.f);
521 nrmDst[i] = glm::vec4(verts[i].normal, 0.f);
522 uvDst[i] = verts[i].uv;
523 }
524 if (gpu.indexType == vk::IndexType::eUint16) {
525 const auto *src16 = static_cast<const uint16_t *>(iMap);
526 for (uint32_t i = 0; i < nInds; ++i) idxDst[i] = uint32_t(src16[i]);
527 } else {
528 const auto *src32 = static_cast<const uint32_t *>(iMap);
529 for (uint32_t i = 0; i < nInds; ++i) idxDst[i] = src32[i];
530 }
531 gpuVertexPool_.positions.unmap();
532 gpuVertexPool_.normals.unmap();
533 gpuVertexPool_.uvs.unmap();
534 gpuVertexPool_.indices.unmap();
535 gpu.vertices.unmap();
536 gpu.indices.unmap();
537
538 // Pool offsets are vertex/index counts, resolved by the vis shaders.
539 gpu.record.vertexOffset = gpuVertexPool_.vertexCount;
540 gpu.record.indexOffset = gpuVertexPool_.indexCount;
541 gpu.record.firstIndex = 0; // vis pass draws non-indexed from the pool
542 gpu.record.vertexBase = 0;
543 gpuVertexPool_.vertexCount = newVerts;
544 gpuVertexPool_.indexCount = newInds;
545}

679void Graphics::growVgBuffers(uint32_t needClusters, uint32_t needVertices,
680 uint32_t needTriangles) {
681 device->waitIdle();
682 const auto hostMem = kHostVisibleCoherent;
683 auto growBuffer = [&](vkb::GenericBuffer &dst, uint32_t elementSize, uint32_t oldElems,
684 uint32_t newElems) {
685 if (newElems <= oldElems) return;
686 vkb::GenericBuffer grown(device, vk::BufferUsageFlagBits::eStorageBuffer,
687 vk::DeviceSize(newElems) * elementSize, hostMem);
688 if (oldElems > 0) {
689 void *srcMap = dst.map();
690 void *dstMap = grown.map();
691 std::memcpy(dstMap, srcMap, vk::DeviceSize(oldElems) * elementSize);
692 grown.unmap();
693 dst.unmap();
694 }
695 dst.release();
696 dst = std::move(grown);
697 };
698 const uint32_t curClusters = uint32_t(vgGpu_.clusters.size / sizeof(GpuVgCluster));
699 const uint32_t curVertFloats = uint32_t(vgGpu_.positions.size / sizeof(float));
700 const uint32_t curTriangles = uint32_t(vgGpu_.triangles.size / sizeof(uint32_t));
701 growBuffer(vgGpu_.clusters, uint32_t(sizeof(GpuVgCluster)), curClusters,
702 std::max(needClusters, curClusters * 2u));
703 growBuffer(vgGpu_.clusterAssets, uint32_t(sizeof(uint32_t)), curClusters,
704 std::max(needClusters, curClusters * 2u));
705 growBuffer(vgGpu_.positions, uint32_t(sizeof(float)), curVertFloats,
706 std::max(needVertices, curVertFloats * 2u));
707 growBuffer(vgGpu_.triangles, uint32_t(sizeof(uint32_t)), curTriangles,
708 std::max(needTriangles, curTriangles * 2u));
709 bindVgPoolBindless();
710}

769uint32_t Graphics::gpuDrivenVgUpload(const GpuVgAssetUpload &asset) {
770 if (!asset.positions || asset.vertexCount <= 0 || !asset.triangles ||
771 asset.triangleCount <= 0 || !asset.clusters || asset.clusterCount <= 0)
772 return kInvalidBindlessSlot;
773 if (vgAssetCount_ >= kMaxVgAssets) return kInvalidBindlessSlot;
774 ensureVgBuffers();
775
776 const uint32_t assetId = vgAssetCount_;
777 const uint32_t vertBase = vgVertexCount_; // global vertex index base
778 const uint32_t triBase = vgTriangleCount_; // global triangle (u32) base
779 const uint32_t clusterBase = vgClusterCount_;
780 const uint32_t needClusters = clusterBase + uint32_t(asset.clusterCount);
781 const uint32_t needVerts = uint32_t(vertBase + uint32_t(asset.vertexCount)) * 3;
782 const uint32_t needTris = triBase + uint32_t(asset.triangleCount);
783 const uint32_t curClusters = uint32_t(vgGpu_.clusters.size / sizeof(GpuVgCluster));
784 const uint32_t curVerts = uint32_t(vgGpu_.positions.size / sizeof(float));
785 const uint32_t curTris = uint32_t(vgGpu_.triangles.size / sizeof(uint32_t));
786 if (needClusters > curClusters || needVerts > curVerts || needTris > curTris)
787 growVgBuffers(needClusters, needVerts, needTris);
788
789 void *posMap = vgGpu_.positions.map();
790 void *triMap = vgGpu_.triangles.map();
791 void *clMap = vgGpu_.clusters.map();
792 void *claMap = vgGpu_.clusterAssets.map();
793 if (!posMap || !triMap || !clMap || !claMap) {
794 if (posMap) vgGpu_.positions.unmap();
795 if (triMap) vgGpu_.triangles.unmap();
796 if (clMap) vgGpu_.clusters.unmap();
797 if (claMap) vgGpu_.clusterAssets.unmap();
798 return kInvalidBindlessSlot;
799 }
800 std::memcpy(static_cast<char *>(posMap) + size_t(vertBase) * 3 * sizeof(float), asset.positions,
801 size_t(asset.vertexCount) * 3 * sizeof(float));
802 {
803 auto *dst = static_cast<uint32_t *>(triMap) + triBase;
804 for (int i = 0; i < asset.triangleCount; ++i) dst[i] = asset.triangles[i] + vertBase;
805 }
806 std::memcpy(static_cast<char *>(clMap) + vgClusterCount_ * sizeof(GpuVgCluster),
807 asset.clusters, size_t(asset.clusterCount) * sizeof(GpuVgCluster));
808 {
809 auto *dst = static_cast<GpuVgCluster *>(clMap) + vgClusterCount_;
810 for (int i = 0; i < asset.clusterCount; ++i) {
811 dst[i].u1[0] += triBase; // triStart -> global triangle stream
812 }
813 }
814 auto *cla = static_cast<uint32_t *>(claMap);
815 for (int i = 0; i < asset.clusterCount; ++i) cla[vgClusterCount_ + uint32_t(i)] = assetId;
816 vgGpu_.positions.unmap();
817 vgGpu_.triangles.unmap();
818 vgGpu_.clusters.unmap();
819 vgGpu_.clusterAssets.unmap();
820
821 vgClusterCount_ = needClusters;
822 vgVertexCount_ = vertBase + uint32_t(asset.vertexCount);
823 vgTriangleCount_ = needTris;
824 vgAssetCount_ = assetId + 1;
825 // Default identity model per slot; vgSetInstance overwrites the current slot.
826 {
827 void *m = vgGpu_.assetModels.map();
828 auto *models = static_cast<glm::mat4 *>(m);
829 for (size_t s = 0; s < kAsyncResourceCopies; ++s)
830 models[s * kMaxVgAssets + assetId] = glm::mat4(1.f);
831 vgGpu_.assetModels.unmap();
832 }
833 return assetId;
834}

881uint32_t Graphics::registerMeshRecord(GpuMesh *gpu) {
882 if (!gpu) return kInvalidBindlessSlot;
883 if (gpu->gpuRecordIndex != kInvalidBindlessSlot) return gpu->gpuRecordIndex;
884 if (meshTableRecords_.size() >= meshTableCapacity_) return kInvalidBindlessSlot;
885 // dev's mesh factories do not populate GpuMeshRecord; build it on first
886 // registration from the host-visible buffers (bounds + ranges).
887 if (gpu->record.vertexCount == 0 && gpu->vertices.buffer) {
888 const uint32_t vertexCount = uint32_t(gpu->vertices.size / sizeof(MeshVertex));
889 gpu->record.vertexCount = vertexCount;
890 gpu->record.indexCount = gpu->indexCount;
891 gpu->record.indexType = gpu->indexType == vk::IndexType::eUint16 ? 0u : 1u;
892 void *map = gpu->vertices.map();
893 if (map && vertexCount > 0) {
894 const auto *verts = static_cast<const MeshVertex *>(map);
895 glm::vec3 minv(1e30f), maxv(-1e30f);
896 for (uint32_t i = 0; i < vertexCount; ++i) {
897 minv = glm::min(minv, verts[i].pos);
898 maxv = glm::max(maxv, verts[i].pos);
899 }
900 const glm::vec3 center = (minv + maxv) * 0.5f;
901 float radius = 0.f;
902 for (uint32_t i = 0; i < vertexCount; ++i)
903 radius = std::max(radius, glm::length(verts[i].pos - center));
904 gpu->record.boundsCenterRadius = glm::vec4(center, radius);
905 }
906 if (map) gpu->vertices.unmap();
907 }
908 // Stage 3: lazily pool the mesh's vertices/indices so the vis resolve can
909 // fetch attributes by (pool offset + triangle + barycentric).
910 appendGpuMeshToPool(*gpu);
911 const uint32_t idx = uint32_t(meshTableRecords_.size());
912 meshTableRecords_.push_back(gpu->record);
913 meshRecordOwners_.push_back(gpu);
914 gpu->gpuRecordIndex = idx;
915 syncMeshTable();
916 return idx;
917}

925GpuMaterialRecord Graphics::buildMaterialRecord(Material *material) {
926 GpuMaterialRecord rec{};
927 if (!material) return rec;
928 rec.tint = glm::vec4(material->getTintR(), material->getTintG(), material->getTintB(),
929 material->getTintA());
930 rec.pbr = glm::vec4(material->getMetallic(), material->getRoughness(),
931 material->getReceiveShadow() ? 1.f : 0.f,
932 material->getReceiveLight() ? 1.f : 0.f);
933 rec.texBomb = glm::vec4(material->getTexCellBombScale(), material->getTexCellBombStrength(),
934 material->getTexCellBombRotation(), 0.f);
935 rec.parallax = glm::vec4(material->getParallaxScale(), material->getParallaxMinLayers(),
936 material->getParallaxMaxLayers(), 0.f);
937 auto slotOf = [](Texture *t) {
938 if (!t || !t->gpuHandle) return kInvalidBindlessSlot;
939 return static_cast<GpuTexture *>(t->gpuHandle)->bindlessIndex2D;
940 };
941 rec.textureSlots[0] = slotOf(material->getAlbedoTexture());
942 rec.textureSlots[1] = slotOf(material->getNormalTexture());
943 rec.textureSlots[2] = slotOf(material->getHeightTexture());
944 rec.textureSlots[3] = kInvalidBindlessSlot; // env is camera state, not material state
945 const std::string model = material->getShadingModel();
946 rec.shadingModel = (model == "unlit") ? 1u : (model == "hair") ? 2u
947 : (model == "custom") ? 3u
948 : 0u;
949 if (material->getCastShadow()) rec.flags |= 1u;
950 if (material->getCastOcclusion()) rec.flags |= 2u;
951 return rec;
952}

991bool Graphics::gpuDrivenSubmitOpaque(const GpuInstance *instances, uint32_t instanceCount) {
992 if (!gpuDrivenEnabled() || !gpuDrivenCaps_.gpuDrivenAvailable()) return false;
993 if (!mesh3dGpuDrivenPipeline || bindlessSets_.empty() || !meshTableBuffer_.buffer) return false;
994 if (!instances || instanceCount == 0) return false;
995
996 // Sort by (material, mesh) and merge buckets using the GPU mesh table.
997 eve::graphics::IndirectBuilder builder;
998 for (uint32_t i = 0; i < instanceCount; ++i) {
999 builder.add(i, instances[i].meshId, instances[i].materialId, 0);
1000 }
1001 const uint32_t drawCount = builder.build(meshTableRecords_);
1002 if (drawCount == 0) return false;
1003 lastGpuDrivenDrawCount_ = drawCount;
1004 const auto &cmds = builder.commands();
1005 const auto &order = builder.sortedInstanceOrder();
1006
1007 // Upload instances in the sorted order so each bucket is a contiguous range.
1008 std::vector<GpuInstance> sorted(instanceCount);
1009 for (uint32_t i = 0; i < instanceCount; ++i) sorted[i] = instances[order[i]];
1010
1011 auto &arena = currentFrameArena();
1012 FrameArena::Alloc instAlloc = arena.alloc(instanceCount * sizeof(GpuInstance), 16);
1013 FrameArena::Alloc cmdAlloc = arena.alloc(drawCount * sizeof(GpuIndirectCommand), 16);
1014 if (!instAlloc.mapped || !cmdAlloc.mapped) return false; // arena overflow: caller falls back
1015 std::memcpy(instAlloc.mapped, sorted.data(), instanceCount * sizeof(GpuInstance));
1016 std::memcpy(cmdAlloc.mapped, cmds.data(), drawCount * sizeof(GpuIndirectCommand));
1017
1018 // Bind the arena instance buffer as bindless binding 4 (update before bind,
1019 // on the current frame slot's set only; the other slot's pending command
1020 // buffers keep pointing at their own frame's arena).
1021 const vk::DescriptorSet bindless = bindlessSetForFrame();
1022 if (!bindless) return false;
1023 vk::DescriptorBufferInfo instInfo{arena.buffer(), instAlloc.offset, instAlloc.size};
1024 vk::WriteDescriptorSet instWrite{};
1025 instWrite.dstSet = bindless;
1026 instWrite.dstBinding = 4;
1027 instWrite.descriptorCount = 1;
1028 instWrite.descriptorType = vk::DescriptorType::eStorageBuffer;
1029 instWrite.pBufferInfo = &instInfo;
1030 device->updateDescriptorSets(1, &instWrite, 0, nullptr);
1031
1032 // Per-frame set0 through the shared ring (dynamic UBO offsets).
1033 const Graphics::GpuDrivenFrameSet0 s0 = gpuDrivenFrameSet0();
1034 if (!s0.set) return false;
1035 const uint32_t dynOffsets[2] = {s0.uboOffset, s0.shadowOffset};
1036
1037 auto &cb = currentPresentCb();
1038 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, mesh3dGpuDrivenPipeline);
1039 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, mesh3dGpuDrivenPipelineLayout, 0, 1,
1040 &s0.set, 2, dynOffsets);
1041 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, mesh3dGpuDrivenPipelineLayout, 1, 1,
1042 &bindless, 0, nullptr);
1043 const vk::DeviceSize stride = sizeof(GpuIndirectCommand);
1044 // Stage 1 keeps one host buffer pair per mesh (no pool yet), so each draw
1045 // group must bind the owning mesh's vertex/index buffers. The builder sorts
1046 // by (pipeline, material, mesh), so commands sharing a mesh are contiguous;
1047 // bind only when the mesh changes.
1048 GpuMesh *boundMesh = nullptr;
1049 for (uint32_t i = 0; i < drawCount; ++i) {
1050 const GpuInstance &first = sorted[cmds[i].firstInstance];
1051 if (first.meshId >= meshRecordOwners_.size()) {
1052 EV_ASSERT(false, "indirect draw references an unregistered mesh record");
1053 continue;
1054 }
1055 GpuMesh *mesh = meshRecordOwners_[first.meshId];
1056 if (mesh != boundMesh) {
1057 const vk::DeviceSize vbOffset = 0;
1058 cb.bindVertexBuffers(0, 1, mesh->vertices, &vbOffset);
1059 cb.bindIndexBuffer(mesh->indices.buffer, 0, mesh->indexType);
1060 boundMesh = mesh;
1061 }
1062 cb.drawIndexedIndirect(arena.buffer(), cmdAlloc.offset + vk::DeviceSize(i) * stride, 1,
1063 stride);
1064 }
1065 return true;
1066}

1087 if (len > 1e-8f) planes[i] /= len;

1103void Graphics::ensureGpuDrivenCullResources(int width, int height) {
1104 if (!gpuDrivenCaps_.gpuDrivenCullAvailable() || width <= 0 || height <= 0) return;
1105 if (gpuDrivenCullReady_ && gpuDrivenCullWidth == width && gpuDrivenCullHeight == height)
1106 return;
1107 destroyGpuDrivenCullResources();
1108
1109 gpuDrivenCullWidth = width;
1110 gpuDrivenCullHeight = height;
1111
1112 // HZB word layout per slot: [16 header uints][mip0][mip1]...
1113 uint32_t mipOffsets[kMaxHzbMips] = {};
1114 uint32_t totalWords = kHZBHeaderWords;
1115 uint32_t maxMip = 0;
1116 {
1117 uint32_t w = uint32_t(width);
1118 uint32_t h = uint32_t(height);
1119 for (uint32_t m = 0; m < kMaxHzbMips; ++m) {
1120 mipOffsets[m] = totalWords;
1121 const uint32_t mw = std::max(w >> m, 1u);
1122 const uint32_t mh = std::max(h >> m, 1u);
1123 totalWords += mw * mh;
1124 maxMip = m;
1125 if (mw == 1 && mh == 1) break;
1126 }
1127 }
1128 gpuDrivenCullMaxMip = maxMip;
1129
1130 const vk::DeviceSize flagBytes = kMaxGpuDrivenInstances * sizeof(uint32_t);
1131 const vk::DeviceSize compactBytes = kMaxGpuDrivenInstances * sizeof(GpuInstance);
1132 const vk::DeviceSize indirectBytes = kMaxGpuDrivenBuckets * sizeof(GpuIndirectCommand);
1133 const vk::DeviceSize indirectNIBytes = kMaxGpuDrivenBuckets * sizeof(glm::uvec4);
1134 const vk::DeviceSize counterBytes = kMaxGpuDrivenBuckets * sizeof(uint32_t);
1135 const vk::DeviceSize hzbBytes = totalWords * sizeof(uint32_t);
1136
1137 gpuDrivenCullSlots_.resize(frameSlotCount());
1138 for (auto &slot : gpuDrivenCullSlots_) {
1139 slot.visibleFlags =
1140 vkb::GenericBuffer(device, vk::BufferUsageFlagBits::eStorageBuffer, flagBytes,
1141 kHostVisibleCoherent);
1142 slot.compacted = vkb::GenericBuffer(device, vk::BufferUsageFlagBits::eStorageBuffer,
1143 compactBytes, kHostVisibleCoherent);
1144 slot.indirect = vkb::GenericBuffer(device, vk::BufferUsageFlagBits::eStorageBuffer |
1145 vk::BufferUsageFlagBits::eIndirectBuffer,
1146 indirectBytes, kHostVisibleCoherent);
1147 slot.indirectNI = vkb::GenericBuffer(device, vk::BufferUsageFlagBits::eStorageBuffer |
1148 vk::BufferUsageFlagBits::eIndirectBuffer,
1149 indirectNIBytes, kHostVisibleCoherent);
1150 slot.bucketCounters =
1151 vkb::GenericBuffer(device, vk::BufferUsageFlagBits::eStorageBuffer, counterBytes,
1152 kHostVisibleCoherent);
1153 slot.hzb = vkb::GenericBuffer(device, vk::BufferUsageFlagBits::eStorageBuffer, hzbBytes,
1154 kHostVisibleCoherent);
1155 slot.cullParams =
1156 vkb::GenericBuffer(device, vk::BufferUsageFlagBits::eUniformBuffer,
1157 sizeof(GpuCullParams), kHostVisibleCoherent);
1158 // Header offsets + zero depth (0 = near plane -> everything visible).
1159 void *hzbMap = slot.hzb.map();
1160 auto *u32 = static_cast<uint32_t *>(hzbMap);
1161 for (uint32_t m = 0; m <= maxMip; ++m) u32[m] = mipOffsets[m];
1162 for (uint32_t m = maxMip + 1; m < kHZBHeaderWords; ++m) u32[m] = 0;
1163 std::memset(static_cast<char *>(hzbMap) + kHZBHeaderWords * sizeof(uint32_t), 0,
1164 (totalWords - kHZBHeaderWords) * sizeof(uint32_t));
1165 slot.hzb.unmap();
1166 }
1167
1168 // Compute pipeline layout: set 1 = bindless (set 0 = empty placeholder so
1169 // the shaders' `layout(set = 1, ...)` bindings resolve to index 1).
1170 vk::DescriptorSetLayoutCreateInfo emptyInfo{};
1171 gpuDrivenComputeEmptyLayout_ = device->createDescriptorSetLayout(emptyInfo);
1172 vk::PushConstantRange pcr{vk::ShaderStageFlagBits::eCompute, 0, 20};
1173 vk::PipelineLayoutCreateInfo pli{};
1174 std::array<vk::DescriptorSetLayout, 2> computeLayouts{gpuDrivenComputeEmptyLayout_,
1175 bindlessSetLayout_};
1176 pli.setLayoutCount = uint32_t(computeLayouts.size());
1177 pli.pSetLayouts = computeLayouts.data();
1178 pli.pushConstantRangeCount = 1;
1179 pli.pPushConstantRanges = &pcr;
1180 gpuDrivenComputeLayout = device->createPipelineLayout(pli);
1181
1182 const auto spvOf = [](const uint32_t *p, size_t n) {
1183 return std::vector<uint32_t>(p, p + n);
1184 };
1185 hzbBuildPass_.create(device, gpuDrivenComputeLayout,
1186 spvOf(hzb_build_comp_spv, hzb_build_comp_spv_count));
1187 cullPass_.create(device, gpuDrivenComputeLayout,
1188 spvOf(gpu_cull_comp_spv, gpu_cull_comp_spv_count));
1189 emitPass_.create(device, gpuDrivenComputeLayout,
1190 spvOf(gpu_emit_comp_spv, gpu_emit_comp_spv_count));
1191 vgCullPass_.create(device, gpuDrivenComputeLayout,
1192 spvOf(vg_main_cull_comp_spv, vg_main_cull_comp_spv_count));
1193 gpuDrivenCullReady_ =
1194 hzbBuildPass_.pipeline() && cullPass_.pipeline() && emitPass_.pipeline();
1195}

1225void Graphics::recordGpuDrivenHzbBuild() {
1226 if (!gpuDrivenCullReady_) return;
1227 auto *slot = currentGBufferSlot();
1228 if (!slot || !slot->depthGpu.sampler || !slot->depthGpu.imageView()) return;
1229 auto &cull = currentGpuDrivenCullSlot();
1230 auto &cb = currentPresentCb();
1231
1232 // Depth attachment write -> compute shader read.
1233 vk::ImageMemoryBarrier imb{};
1234 imb.image = slot->depth.image();
1235 imb.oldLayout = vk::ImageLayout::eShaderReadOnlyOptimal;
1236 imb.newLayout = vk::ImageLayout::eShaderReadOnlyOptimal;
1237 imb.srcAccessMask = vk::AccessFlagBits::eDepthStencilAttachmentWrite;
1238 imb.dstAccessMask = vk::AccessFlagBits::eShaderRead;
1239 imb.subresourceRange = {vk::ImageAspectFlagBits::eDepth, 0, 1, 0, 1};
1240 cb.pipelineBarrier(vk::PipelineStageFlagBits::eEarlyFragmentTests |
1241 vk::PipelineStageFlagBits::eLateFragmentTests,
1242 vk::PipelineStageFlagBits::eComputeShader, {}, 0, nullptr, 0, nullptr, 1,
1243 &imb);
1244
1245 uint32_t mipOffsets[kMaxHzbMips] = {};
1246 uint32_t totalWords = kHZBHeaderWords;
1247 {
1248 uint32_t w = uint32_t(gpuDrivenCullWidth);
1249 uint32_t h = uint32_t(gpuDrivenCullHeight);
1250 for (uint32_t m = 0; m <= gpuDrivenCullMaxMip; ++m) {
1251 mipOffsets[m] = totalWords;
1252 const uint32_t mw = std::max(w >> m, 1u);
1253 const uint32_t mh = std::max(h >> m, 1u);
1254 totalWords += mw * mh;
1255 }
1256 }
1257 for (uint32_t m = 0; m <= gpuDrivenCullMaxMip; ++m) {
1258 if (m > 0) {
1259 // Previous mip write -> this mip read.
1260 vk::BufferMemoryBarrier bmb{};
1261 bmb.buffer = cull.hzb.buffer;
1262 bmb.size = VK_WHOLE_SIZE;
1263 bmb.srcAccessMask = vk::AccessFlagBits::eShaderWrite;
1264 bmb.dstAccessMask = vk::AccessFlagBits::eShaderRead;
1265 cb.pipelineBarrier(vk::PipelineStageFlagBits::eComputeShader,
1266 vk::PipelineStageFlagBits::eComputeShader, {}, 0, nullptr, 1, &bmb,
1267 0, nullptr);
1268 }
1269 const uint32_t mw = std::max(uint32_t(gpuDrivenCullWidth) >> m, 1u);
1270 const uint32_t mh = std::max(uint32_t(gpuDrivenCullHeight) >> m, 1u);
1271 struct HzbPush {
1272 uint32_t mip;
1273 uint32_t width;
1274 uint32_t height;
1275 uint32_t slotBase;
1276 uint32_t prevOffset;
1277 } push{m, mw, mh, 0u, mipOffsets[m > 0 ? m - 1 : 0]};
1278 cb.pushConstants(gpuDrivenComputeLayout, vk::ShaderStageFlagBits::eCompute, 0,
1279 sizeof(push), &push);
1280 hzbBuildPass_.record(cb, (mw + 7) / 8, (mh + 7) / 8, 1);
1281 }

1284bool Graphics::gpuDrivenCullBegin(const GpuInstance *instances, uint32_t instanceCount) {
1285 if (!gpuDrivenCullEnabled() || !gpuDrivenCullReady_) return false;
1286 if (!instances || instanceCount == 0) return false;
1287 instanceCount = std::min(instanceCount, kMaxGpuDrivenInstances);
1288
1289 eve::graphics::IndirectBuilder builder;
1290 for (uint32_t i = 0; i < instanceCount; ++i) {
1291 builder.add(i, instances[i].meshId, instances[i].materialId, 0);
1292 }
1293 const uint32_t bucketCount = builder.build(meshTableRecords_);
1294 if (bucketCount == 0 || bucketCount > kMaxGpuDrivenBuckets) return false;
1295 const auto &cmds = builder.commands();
1296 const auto &order = builder.sortedInstanceOrder();
1297
1298 std::vector<GpuInstance> sorted(instanceCount);
1299 for (uint32_t i = 0; i < instanceCount; ++i) sorted[i] = instances[order[i]];
1300
1301 gpuDrivenBucketIds_.assign(instanceCount, 0);
1302 gpuDrivenBucketOffsets_.resize(bucketCount);
1303 gpuDrivenBucketMeshIds_.resize(bucketCount);
1304 for (uint32_t b = 0; b < bucketCount; ++b) {
1305 gpuDrivenBucketOffsets_[b] = cmds[b].firstInstance;
1306 gpuDrivenBucketMeshIds_[b] = sorted[cmds[b].firstInstance].meshId;
1307 }
1308 {
1309 uint32_t b = 0;
1310 for (uint32_t j = 0; j < instanceCount; ++j) {
1311 while (b + 1 < bucketCount && cmds[b + 1].firstInstance <= j) ++b;
1312 gpuDrivenBucketIds_[j] = b;
1313 }
1314 }
1315
1316 auto &arena = currentFrameArena();
1317 // Storage-buffer descriptor offsets must honor minStorageBufferOffsetAlignment
1318 // (64 on this Intel driver); align conservatively to 64.
1319 gpuDrivenInstAlloc_ = arena.alloc(instanceCount * sizeof(GpuInstance), 64);
1320 gpuDrivenBucketIdAlloc_ = arena.alloc(instanceCount * sizeof(uint32_t), 64);
1321 gpuDrivenBucketOffAlloc_ = arena.alloc(bucketCount * sizeof(uint32_t), 64);
1322 if (!gpuDrivenInstAlloc_.mapped || !gpuDrivenBucketIdAlloc_.mapped ||
1323 !gpuDrivenBucketOffAlloc_.mapped) {
1324 gpuDrivenCullInstanceCount_ = 0;
1325 return false;
1326 }
1327 std::memcpy(gpuDrivenInstAlloc_.mapped, sorted.data(),
1328 instanceCount * sizeof(GpuInstance));
1329 std::memcpy(gpuDrivenBucketIdAlloc_.mapped, gpuDrivenBucketIds_.data(),
1330 instanceCount * sizeof(uint32_t));
1331 std::memcpy(gpuDrivenBucketOffAlloc_.mapped, gpuDrivenBucketOffsets_.data(),
1332 bucketCount * sizeof(uint32_t));
1333
1334 gpuDrivenBucketCount_ = bucketCount;
1335 gpuDrivenCullInstanceCount_ = instanceCount;
1336 lastGpuDrivenDrawCount_ = bucketCount; // debug counter: bucket draws the cull path emits
1337
1338 // Cull source buffers: sorted instances (17), bucket ids (12), offsets (13).
1339 const vk::DescriptorSet bindless = bindlessSetForFrame();
1340 if (!bindless) {
1341 gpuDrivenCullInstanceCount_ = 0;
1342 return false;
1343 }
1344 auto bufWrite = [&](uint32_t binding, vk::Buffer buffer, vk::DeviceSize offset,
1345 vk::DeviceSize size) {
1346 vk::DescriptorBufferInfo info{buffer, offset, size};
1347 vk::WriteDescriptorSet w{};
1348 w.dstSet = bindless;
1349 w.dstBinding = binding;
1350 w.descriptorCount = 1;
1351 w.descriptorType = vk::DescriptorType::eStorageBuffer;
1352 w.pBufferInfo = &info;
1353 device->updateDescriptorSets(1, &w, 0, nullptr);
1354 };
1355 bufWrite(17, arena.buffer(), gpuDrivenInstAlloc_.offset, gpuDrivenInstAlloc_.size);
1356 bufWrite(12, arena.buffer(), gpuDrivenBucketIdAlloc_.offset, gpuDrivenBucketIdAlloc_.size);
1357 bufWrite(13, arena.buffer(), gpuDrivenBucketOffAlloc_.offset, gpuDrivenBucketOffAlloc_.size);
1358 return true;
1359}

1391 w[0].pImageInfo = &visIDInfo;

1473 bmb.size = VK_WHOLE_SIZE;
1474 bmb.srcAccessMask = vk::AccessFlagBits::eShaderWrite;
1475 bmb.dstAccessMask = vk::AccessFlagBits::eShaderRead;

1490 gpuDrivenLastCullSlot_ = uint32_t(currentFrameSlot());

1492 auto &cb = currentPresentCb();
1493 // This frame's own bindless set: the previous frame that owned this slot
1494 // completed at acquireForFrame()'s fence wait, so rewriting it is safe.
1495 const vk::DescriptorSet bindless = bindlessSetForFrame();
1496 if (!bindless) return;
1497
1498 // CPU reset of GPU-owned per-slot state (slot's previous frame is complete).
1499 {
1500 void *flagsMap = slot.visibleFlags.map();
1501 std::memset(flagsMap, 0, kMaxGpuDrivenInstances * sizeof(uint32_t));
1502 slot.visibleFlags.unmap();

1505 slot.bucketCounters.unmap();

1507 // commands so a stale instanceCount from an earlier frame cannot make

1516 const uint32_t groups = (gpuDrivenCullInstanceCount_ + 63u) / 64u;
1517 // Arena host writes + previous-frame HZB shader writes visible to cull.
1518 {
1519 vk::BufferMemoryBarrier bmb{};
1520 bmb.buffer = currentFrameArena().buffer();
1521 bmb.size = VK_WHOLE_SIZE;
1522 bmb.srcAccessMask = vk::AccessFlagBits::eHostWrite | vk::AccessFlagBits::eShaderWrite;
1523 bmb.dstAccessMask = vk::AccessFlagBits::eShaderRead;
1524 cb.pipelineBarrier(vk::PipelineStageFlagBits::eHost |
1525 vk::PipelineStageFlagBits::eComputeShader,
1526 vk::PipelineStageFlagBits::eComputeShader, {}, 0, nullptr, 1, &bmb, 0,
1527 nullptr);
1528 }
1529 cullPass_.record(cb, groups, 1, 1);

1531 // Cull flags write -> emit read.

1533 vk::BufferMemoryBarrier bmb{};

1535 bmb.size = VK_WHOLE_SIZE;
1536 bmb.srcAccessMask = vk::AccessFlagBits::eShaderWrite;
1537 bmb.dstAccessMask = vk::AccessFlagBits::eShaderRead;
1538 cb.pipelineBarrier(vk::PipelineStageFlagBits::eComputeShader,
1539 vk::PipelineStageFlagBits::eComputeShader, {}, 0, nullptr, 1, &bmb, 0,
1540 nullptr);
1541 }
1542 emitPass_.record(cb, groups, 1, 1);
1543
1544 // Emit writes (compacted + commands) -> vertex read + indirect read.
1545 {
1546 vk::BufferMemoryBarrier bmb[2]{};
1547 bmb[0].buffer = slot.compacted.buffer;
1548 bmb[0].size = VK_WHOLE_SIZE;
1549 bmb[0].srcAccessMask = vk::AccessFlagBits::eShaderWrite;
1550 bmb[0].dstAccessMask = vk::AccessFlagBits::eShaderRead;
1551 bmb[1].buffer = slot.indirect.buffer;
1552 bmb[1].size = VK_WHOLE_SIZE;
1553 bmb[1].srcAccessMask = vk::AccessFlagBits::eShaderWrite;
1554 bmb[1].dstAccessMask = vk::AccessFlagBits::eIndirectCommandRead;

1556 vk::PipelineStageFlagBits::eVertexShader |
1557 vk::PipelineStageFlagBits::eDrawIndirect,
1558 {}, 0, nullptr, 2, bmb, 0, nullptr);
1559 }
