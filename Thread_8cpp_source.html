20class PoolExecutor final : public eve::caps::IAsyncWorkExecutor {
21public:
22 explicit PoolExecutor(Thread *owner) : owner_(owner) {}
23
24 eve::Result<void> submit(std::function<void()> work) override {
25 if (!work) {
26 return eve::Result<void>::failure(eve::Diagnostic::error(
27 eve::DiagnosticCode::InvalidArgument, "IAsyncWorkExecutor: null work"));
28 }
29 try {
30 std::unique_ptr<Task> task(owner_->getPool()->submit(std::move(work)));
31 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
32 } catch (const eve::Exception &ex) {
33 return eve::Result<void>::failure(
34 eve::Diagnostic::error(eve::DiagnosticCode::Failed, ex.what()));
35 }
36 }
37
38private:
39 Thread *owner_ = nullptr;
40};

84Channel *Thread::getChannel(std::string channelName) {
85 if (channelName.empty())
86 throw eve::Exception("Thread::getChannel: name must not be empty");
87 std::lock_guard<std::mutex> lock(mu_);
88 auto it = namedChannels_.find(channelName);
89 if (it != namedChannels_.end())
90 return it->second.get();
91 auto ch = std::make_unique<Channel>(channelName);
92 Channel *raw = ch.get();
93 namedChannels_.emplace(std::move(channelName), std::move(ch));
94 return raw;
95}
