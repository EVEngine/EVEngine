101 [[nodiscard]] eve::Result<void> prepare(const eve::transaction::TransactionContext&) override {
102 if (prepared_ || committed_) return cardLifecycleConflict("card transfer participant is already in flight");
103 if (&source_ == &destination_)
104 return cardFailure<void>(eve::DiagnosticCode::Conflict, "card play source and destination must differ");
105
106 auto sourceResult = source_.snapshot();
107 if (!sourceResult) return eve::Result<void>::failure(sourceResult.status());
108 auto destinationResult = destination_.snapshot();
109 if (!destinationResult) return eve::Result<void>::failure(destinationResult.status());
110 beforeSource_ = std::move(sourceResult).takeValue();
111 beforeDestination_ = std::move(destinationResult).takeValue();
112
113 const eve::container::MembershipId object(card_.identity()->id);
114 const auto sourceIt = std::find_if(beforeSource_.entries.begin(), beforeSource_.entries.end(),
115 [&object](const auto& entry) { return entry.membership.object == object; });
116 if (sourceIt == beforeSource_.entries.end())
117 return cardFailure<void>(eve::DiagnosticCode::NotFound, "card is not present in the source container",
118 "source");
119 if (sourceSlot_ && *sourceSlot_ != sourceIt->membership.slot)
120 return cardFailure<void>(eve::DiagnosticCode::StaleHandle, "card source slot is stale", "sourceSlot");
121
122 auto accepted = destination_.validateInsert(sourceIt->object, destinationSlot_, std::nullopt);
123 if (!accepted) return accepted;
124 if (destinationSlot_ && (destinationSlot_->value() < 0 || static_cast<std::size_t>(destinationSlot_->value()) >
125 beforeDestination_.entries.size()))
126 return cardFailure<void>(eve::DiagnosticCode::InvalidArgument, "card destination slot is out of range",
127 "destinationSlot");
128
129 afterSource_ = beforeSource_;
130 afterSource_.entries.erase(afterSource_.entries.begin() + (sourceIt - beforeSource_.entries.begin()));
131 renumber(afterSource_);
132 afterSource_.revision = nextRevision(beforeSource_.revision);
133 if (afterSource_.revision == beforeSource_.revision)
134 return cardFailure<void>(eve::DiagnosticCode::InvariantViolation, "source container revision is exhausted");
135
136 afterDestination_ = beforeDestination_;
137 auto moved = *sourceIt;
138 const std::size_t insertion =
139 destinationSlot_ ? static_cast<std::size_t>(destinationSlot_->value()) : afterDestination_.entries.size();
140 if (insertion > afterDestination_.entries.size())
141 return cardFailure<void>(eve::DiagnosticCode::InvalidArgument, "card destination slot is out of range",
142 "destinationSlot");
143 afterDestination_.entries.insert(afterDestination_.entries.begin() + insertion, std::move(moved));
144 renumber(afterDestination_);
145 afterDestination_.revision = nextRevision(beforeDestination_.revision);
146 if (afterDestination_.revision == beforeDestination_.revision)
147 return cardFailure<void>(eve::DiagnosticCode::InvariantViolation,
148 "destination container revision is exhausted");
149
150 auto sourcePrepared = source_.prepare(beforeSource_, afterSource_);
151 if (!sourcePrepared) return eve::Result<void>::failure(sourcePrepared.status());
152 sourceStage_ = std::move(sourcePrepared).takeValue();
153 if (!sourceStage_)
154 return cardFailure<void>(eve::DiagnosticCode::InvariantViolation,
155 "source container returned an empty stage");
156 auto destinationPrepared = destination_.prepare(beforeDestination_, afterDestination_);
157 if (!destinationPrepared) {
158 sourceStage_->rollback();
159 sourceStage_.reset();
160 return eve::Result<void>::failure(destinationPrepared.status());
161 }
162 destinationStage_ = std::move(destinationPrepared).takeValue();
163 if (!destinationStage_) {
164 sourceStage_->rollback();
165 sourceStage_.reset();
166 return cardFailure<void>(eve::DiagnosticCode::InvariantViolation,
167 "destination container returned an empty stage");
168 }
169 prepared_ = true;
170 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
171 }

173 [[nodiscard]] eve::Result<void> commit(const eve::transaction::TransactionContext&) override {
174 if (!prepared_ || committed_) return cardLifecycleConflict("card transfer has no prepared stage");
175 destinationStage_->commit();
176 sourceStage_->commit();
177 committed_ = true;
178 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
179 }

181 [[nodiscard]] eve::Result<void> rollback(const eve::transaction::TransactionContext&) override {
182 if (committed_) return cardLifecycleConflict("committed card transfer requires compensation");
183 if (destinationStage_) destinationStage_->rollback();
184 if (sourceStage_) sourceStage_->rollback();
185 sourceStage_.reset();
186 destinationStage_.reset();
187 prepared_ = false;
188 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
189 }

191 [[nodiscard]] eve::Result<void> compensate(const eve::transaction::TransactionContext&) override {
192 if (!committed_) return cardLifecycleConflict("card transfer has no committed state to compensate");
193
194 auto currentSource = source_.snapshot();
195 if (!currentSource) return eve::Result<void>::failure(currentSource.status());
196 auto currentDestination = destination_.snapshot();
197 if (!currentDestination) return eve::Result<void>::failure(currentDestination.status());
198 if (currentSource.value().revision != afterSource_.revision ||
199 currentDestination.value().revision != afterDestination_.revision)
200 return cardFailure<void>(eve::DiagnosticCode::StaleHandle, "card containers changed before compensation");
201 auto sourceCandidate = beforeSource_;
202 auto destinationCandidate = beforeDestination_;
203 sourceCandidate.revision = nextRevision(currentSource.value().revision);
204 destinationCandidate.revision = nextRevision(currentDestination.value().revision);
205 if (sourceCandidate.revision == currentSource.value().revision ||
206 destinationCandidate.revision == currentDestination.value().revision)
207 return cardFailure<void>(eve::DiagnosticCode::InvariantViolation,
208 "card transfer compensation revision is exhausted");
209
210 auto sourceRestore = source_.prepare(currentSource.value(), sourceCandidate);
211 if (!sourceRestore) return eve::Result<void>::failure(sourceRestore.status());
212 auto sourceStage = std::move(sourceRestore).takeValue();
213 if (!sourceStage)
214 return cardFailure<void>(eve::DiagnosticCode::InvariantViolation,
215 "source compensation returned an empty stage");
216 auto destinationRestore = destination_.prepare(currentDestination.value(), destinationCandidate);
217 if (!destinationRestore) {
218 sourceStage->rollback();
219 return eve::Result<void>::failure(destinationRestore.status());
220 }
221 auto destinationStage = std::move(destinationRestore).takeValue();
222 if (!destinationStage) {
223 sourceStage->rollback();
224 return cardFailure<void>(eve::DiagnosticCode::InvariantViolation,
225 "destination compensation returned an empty stage");
226 }
227 destinationStage->commit();
228 sourceStage->commit();
229 sourceStage_.reset();
230 destinationStage_.reset();
231 prepared_ = false;
232 committed_ = false;
233 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
234 }

333eve::Result<eve::transaction::TransactionReceipt> CardPlayPaymentAdapter::play(CardPlayRequest request) {
334 if (request.card == nullptr || request.definition == nullptr)
335 return cardFailure<eve::transaction::TransactionReceipt>(eve::DiagnosticCode::InvalidArgument,
336 "card play requires a card and definition", "request");
337 auto& card = *request.card;
338 const auto& definition = *request.definition;
339 if ((request.composition.source == nullptr) != (request.composition.destination == nullptr))
340 return cardFailure<eve::transaction::TransactionReceipt>(
341 eve::DiagnosticCode::InvalidArgument, "card play requires both source and destination containers",
342 "composition");
343
344 const auto condition = CardPlayConditionAdapter::evaluate(&card, definition, definition.playCondition,
345 request.composition.conditionQueries);
346 if (!condition.passed()) {
347 DiagnosticDetails details;
348 details.emplace_back("reason", decision::conditionReasonCodeName(condition.reasonCode()));
349 return eve::Result<eve::transaction::TransactionReceipt>::failure(
350 eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation, "card play condition was rejected",
351 "condition", std::move(details)));
352 }
353
354 auto cost = manaCost(definition);
355 if (!cost) return eve::Result<eve::transaction::TransactionReceipt>::failure(cost.status());
356 if (cost.value() && request.playerAccount == nullptr)
357 return cardFailure<eve::transaction::TransactionReceipt>(eve::DiagnosticCode::InvalidArgument,
358 "a mana-bearing card play requires a player account",
359 "playerAccount");
360 if (request.transactionId.empty()) request.transactionId = "card.play." + definition.id;
361
362 CardPlayParticipant state(card);
363 std::unique_ptr<CardTransferParticipant> transfer;
364 std::vector<eve::transaction::ITransactionParticipant*> participants;
365 participants.reserve(request.composition.effect ? 3u : 2u);
366 if (request.composition.source != nullptr) {
367 transfer = std::make_unique<CardTransferParticipant>(
368 card, *request.composition.source, *request.composition.destination, request.composition.sourceSlot,
369 request.composition.destinationSlot);
370 participants.push_back(transfer.get());
371 }
372 participants.push_back(&state);
373 if (request.composition.effect != nullptr) participants.push_back(request.composition.effect);
374
375 eve::transaction::TransactionContext context(std::move(request.transactionId));
376 if (cost.value())
377 return eve::transaction::AtomicResourcePayment::execute(
378 context, *request.playerAccount, *cost.value(),
379 std::span<eve::transaction::ITransactionParticipant*>(participants));
380 return eve::transaction::AtomicResourcePayment::execute(
381 context, std::span<eve::transaction::ITransactionParticipant*>(participants));
382}
