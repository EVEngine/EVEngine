51EditorResult<CrowdPathRecord> parsePath(const EditorValue& value) {
52 const auto* idEntry = field(value, "id"); const auto* nameEntry = field(value, "name");
53 const auto* loopEntry = field(value, "loop"); const auto* pointsEntry = field(value, "points");
54 const auto* id = idEntry ? idEntry->getIf<std::string>() : nullptr;
55 const auto* name = nameEntry ? nameEntry->getIf<std::string>() : nullptr;
56 const auto* loop = loopEntry ? loopEntry->getIf<bool>() : nullptr;
57 const auto* points = pointsEntry ? pointsEntry->getIf<EditorValue::Array>() : nullptr;
58 if (!id || id->empty() || !name || name->empty() || !loop || !points || points->empty())
59 return failure<CrowdPathRecord>(EditorStatus::Rejected, "editor.crowd.invalid-path",
60 "Path requires identity, name and at least one waypoint");
61 CrowdPathRecord result{StableId(*id), *name, *loop, {}};
62 std::set<StableId> ids;
63 for (const auto& entry : *points) {
64 auto point = parsePoint(entry);
65 if (!point.ok() || !ids.insert(point.value().id).second)
66 return failure<CrowdPathRecord>(EditorStatus::Rejected, "editor.crowd.duplicate-waypoint",
67 "Path waypoint ids must be unique and valid");
68 result.points.push_back(std::move(point).value());
69 }
70 if (result.loop && result.points.size() < 2)
71 return failure<CrowdPathRecord>(EditorStatus::Rejected, "editor.crowd.loop-point-count",
72 "Looping paths require at least two waypoints");

166 if (op.target != TargetId(id_)) return failure<void>(EditorStatus::Rejected, "editor.crowd.wrong-target", "Operation targets another crowd document");
167 auto apply = [&](const std::string& type, const EditorValue& payload) -> EditorResult<void> {

175 } else return failure<void>(EditorStatus::Rejected, "editor.crowd.unsupported-operation", "Unsupported crowd document operation");
176 ++revision_; dirty_.include(0, 0); return eve::editing::applied<void>();
177 };
