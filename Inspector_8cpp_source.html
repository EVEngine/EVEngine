55void Inspector::refresh() {
56 classNames_.clear();
57 if (Runtime* rt = runtime()) {
58 rt->scanClasses(); // picks up dofile()/compilestring() defined classes
59 for (const ReflectedClass& cls : rt->reflectedClasses())
60 classNames_.push_back(cls.name);
61 }
62 // Drop a selection whose class vanished (e.g. script unloaded).
63 if (!selectedClass_.empty() &&
64 std::find(classNames_.begin(), classNames_.end(), selectedClass_) ==
65 classNames_.end()) {
66 selectedClass_.clear();
67 instances_.clear();
68 selectedInstance_ = -1;
69 }
70 if (selectedClass_.empty() && !classNames_.empty())
71 selectClass(classNames_.front());
72 rebuildPropertyModel();
73 rebuildHost();
74}

97bool Inspector::selectClass(const std::string& name) {
98 Runtime* rt = runtime();
99 if (!rt ||
100 std::find(classNames_.begin(), classNames_.end(), name) == classNames_.end())
101 return false;
102 navStack_.clear();
103 selectedClass_ = name;
104 instances_.clear();
105 selectedInstance_ = -1;
106 try {
107 ssq::Object instance = rt->createInstance(name);
108 InstanceEntry entry;
109 entry.label = name + " #1";
110 entry.object = instance;
111 instances_.push_back(std::move(entry));
112 selectedInstance_ = 0;
113 } catch (...) {
114 // Keep the class selected; the panel shows "no instances".
115 }
116 rebuildPropertyModel();
117 rebuildHost();
118 return selectedInstance_ >= 0;
119}

402WidgetDesc Inspector::build() {
403 std::vector<WidgetDesc> children;
404 if (classNames_.empty()) {
405 children.push_back(text(
406 "No reflected script classes. Run a script that defines classes, "
407 "then call ui.inspectRefresh().",
408 "hint"));
409 } else {
410 std::vector<WidgetDesc> toolbar = {
411 text("Class", "lbl_class"),
412 spacer("class_spacer"),
413 combo("##class", classNames_, currentClassIndex(), "class",
414 [this](float index) {
415 const int idx = static_cast<int>(index);
416 if (idx >= 0 && idx < int(classNames_.size()))
417 selectClass(classNames_[size_t(idx)]);
418 }),
419 };
420 if (pickScene_) {
421 toolbar.push_back(
422 button("Pick##inspector_pick", "inspector_pick",
423 [this]() {
424 if (!pickScene_) return;
425 const ssq::Object picked = pickScene_();
426 if (picked.getType() == ssq::Type::INSTANCE)
427 inspectObject(picked);
428 }));
429 }
430 if (!navStack_.empty()) {
431 toolbar.insert(toolbar.begin(),
432 button("<- Back##inspector_back", "inspector_back",
433 [this]() { back(); }));
434 }
435 children.push_back(row(
436 std::move(toolbar),
437 "classrow"));
438
439 std::vector<std::string> labels;
440 labels.reserve(instances_.size());
441 for (const InstanceEntry& entry : instances_) labels.push_back(entry.label);
442 children.push_back(row(
443 {
444 text("Instance", "lbl_instance"),
445 spacer("instance_spacer"),
446 combo("##instance", labels, selectedInstance_, "instance",
447 [this](float index) { selectInstance(static_cast<int>(index)); }),
448 button("+", "add_instance", [this]() { addInstance(); }),
449 },
450 "instancerow"));
451 children.push_back(separator("inspector_sep"));
452
453 Runtime* rt = runtime();
454 if (rt && selectedInstance_ >= 0 &&
455 size_t(selectedInstance_) < instances_.size()) {
456 const ssq::Object& instance =
457 instances_[size_t(selectedInstance_)].object;
458 // Inheritance chain: own class first, then bases (parent props are
459 // grouped under their owning class header).
460 std::vector<std::string> chain;
461 std::string current = selectedClass_;
462 while (!current.empty()) {
463 chain.push_back(current);
464 const ReflectedClass* cls = rt->reflectedClass(current);
465 if (!cls || cls->base.empty()) break;
466 current = cls->base;
467 }
468 for (const std::string& className : chain) {
469 const ReflectedClass* cls = rt->reflectedClass(className);
470 if (!cls) continue;
471 std::vector<WidgetDesc> props;
472 for (const ReflectedMember& member : cls->members) {
473 if (member.method) continue;
474 props.push_back(propertyWidget(
475 className, member, rt->readProperty(instance, member.name),
476 instance));
477 }
478 const bool own = className == selectedClass_;
479 const std::string headerLabel =
480 className + (own ? "" : " (base)") + "##cls_" + className;
481 if (props.empty()) {
482 children.push_back(
483 text(className + " (no editable properties)", "cls_" + className));
484 } else {
485 children.push_back(collapsingHeader(
486 headerLabel, std::move(props), "cls_" + className, own));
487 }
488 }
489 } else if (rt) {
490 children.push_back(text("Instance creation failed for " + selectedClass_,
491 "instance_error"));
492 }
493 }
494 return window("Inspector", std::move(children), "root");
495}
