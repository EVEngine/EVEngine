55void Inspector::refresh() {
56 classNames_.clear();
57 if (Runtime* rt = runtime()) {
58 rt->scanClasses(); // picks up dofile()/compilestring() defined classes
59 for (const ReflectedClass& cls : rt->reflectedClasses())
60 classNames_.push_back(cls.name);
61 }
62 // Drop a selection whose class vanished (e.g. script unloaded).
63 if (!selectedClass_.empty() &&
64 std::find(classNames_.begin(), classNames_.end(), selectedClass_) ==
65 classNames_.end()) {
66 selectedClass_.clear();
67 instances_.clear();
68 selectedInstance_ = -1;
69 }
70 if (selectedClass_.empty() && !classNames_.empty())
71 selectClass(classNames_.front());
72 rebuildPropertyModel();
73 rebuildHost();
74}

97bool Inspector::selectClass(const std::string& name) {
98 Runtime* rt = runtime();
99 if (!rt ||
100 std::find(classNames_.begin(), classNames_.end(), name) == classNames_.end())
101 return false;
102 navStack_.clear();
103 selectedClass_ = name;
104 instances_.clear();
105 selectedInstance_ = -1;
106 try {
107 ssq::Object instance = rt->createInstance(name);
108 InstanceEntry entry;
109 entry.label = name + " #1";
110 entry.object = instance;
111 instances_.push_back(std::move(entry));
112 selectedInstance_ = 0;
113 } catch (...) {
114 // Keep the class selected; the panel shows "no instances".
115 }
116 rebuildPropertyModel();
117 rebuildHost();
118 return selectedInstance_ >= 0;
119}

249WidgetDesc Inspector::build() {
250 std::vector<WidgetDesc> children;
251 if (classNames_.empty()) {
252 children.push_back(text(
253 "No reflected script classes. Run a script that defines classes, "
254 "then call ui.inspectRefresh().",
255 "hint"));
256 } else {
257 std::vector<WidgetDesc> toolbar = {
258 text("Class", "lbl_class"),
259 spacer("class_spacer"),
260 combo("##class", classNames_, currentClassIndex(), "class",
261 [this](float index) {
262 const int idx = static_cast<int>(index);
263 if (idx >= 0 && idx < int(classNames_.size()))
264 selectClass(classNames_[size_t(idx)]);
265 }),
266 };
267 if (pickScene_) {
268 toolbar.push_back(
269 button("Pick##inspector_pick", "inspector_pick",
270 [this]() {
271 if (!pickScene_) return;
272 const ssq::Object picked = pickScene_();
273 if (picked.getType() == ssq::Type::INSTANCE)
274 inspectObject(picked);
275 }));
276 }
277 if (!navStack_.empty()) {
278 toolbar.insert(toolbar.begin(),
279 button("<- Back##inspector_back", "inspector_back",
280 [this]() { back(); }));
281 }
282 children.push_back(row(
283 std::move(toolbar),
284 "classrow"));
285
286 std::vector<std::string> labels;
287 labels.reserve(instances_.size());
288 for (const InstanceEntry& entry : instances_) labels.push_back(entry.label);
289 children.push_back(row(
290 {
291 text("Instance", "lbl_instance"),
292 spacer("instance_spacer"),
293 combo("##instance", labels, selectedInstance_, "instance",
294 [this](float index) { selectInstance(static_cast<int>(index)); }),
295 button("+", "add_instance", [this]() { addInstance(); }),
296 },
297 "instancerow"));
298 children.push_back(separator("inspector_sep"));
299
300 Runtime* rt = runtime();
301 if (rt && selectedInstance_ >= 0 &&
302 size_t(selectedInstance_) < instances_.size()) {
303 const ssq::Object& instance =
304 instances_[size_t(selectedInstance_)].object;
305 // Inheritance chain: own class first, then bases (parent props are
306 // grouped under their owning class header).
307 std::vector<std::string> chain;
308 std::string current = selectedClass_;
309 while (!current.empty()) {
310 chain.push_back(current);
311 const ReflectedClass* cls = rt->reflectedClass(current);
312 if (!cls || cls->base.empty()) break;
313 current = cls->base;
314 }
315 for (const std::string& className : chain) {
316 const ReflectedClass* cls = rt->reflectedClass(className);
317 if (!cls) continue;
318 std::vector<WidgetDesc> props;
319 for (const ReflectedMember& member : cls->members) {
320 if (member.method) continue;
321 props.push_back(propertyWidget(
322 className, member, rt->readProperty(instance, member.name),
323 instance));
324 }
325 const bool own = className == selectedClass_;
326 const std::string headerLabel =
327 className + (own ? "" : " (base)") + "##cls_" + className;
328 if (props.empty()) {
329 children.push_back(
330 text(className + " (no editable properties)", "cls_" + className));
331 } else {
332 children.push_back(collapsingHeader(
333 headerLabel, std::move(props), "cls_" + className, own));
334 }
335 }
336 } else if (rt) {
337 children.push_back(text("Instance creation failed for " + selectedClass_,
338 "instance_error"));
339 }
340 }
341 return window("Inspector", std::move(children), "root");
342}
