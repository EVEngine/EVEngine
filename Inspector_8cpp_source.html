15bool sameValue(const ReflectedValue& a, const ReflectedValue& b) {
16 if (a.kind != b.kind) return false;
17 switch (a.kind) {
18 case ReflectedValueKind::Bool: return a.boolean == b.boolean;
19 case ReflectedValueKind::Integer: return a.integer == b.integer;
20 case ReflectedValueKind::Float: return a.floating == b.floating;
21 case ReflectedValueKind::String: return a.text == b.text;
22 default: return true;
23 }
24}

40EditorKind editorKind(const ReflectedMember& member, const ReflectedValue& value) {
41 const std::string editor = member.attrString("editor");
42 if (value.kind == ReflectedValueKind::Bool)
43 return editor == "checkbox" ? EditorKind::Checkbox : EditorKind::Checkbox;
44 if (editor == "combo" && !member.attrOptions("options").empty())
45 return EditorKind::Combo;
46 if (editor == "slider" &&
47 (value.kind == ReflectedValueKind::Integer ||
48 value.kind == ReflectedValueKind::Float))
49 return EditorKind::Slider;
50 if (value.kind == ReflectedValueKind::Array ||
51 value.kind == ReflectedValueKind::Table ||
52 value.kind == ReflectedValueKind::Instance ||
53 value.kind == ReflectedValueKind::None ||
54 value.kind == ReflectedValueKind::Other)
55 return EditorKind::ReadOnly;
56 return EditorKind::Input;
57}

99void Inspector::refresh() {
100 classNames_.clear();
101 if (Runtime* rt = runtime()) {
102 rt->scanClasses(); // picks up dofile()/compilestring() defined classes
103 for (const ReflectedClass& cls : rt->reflectedClasses())
104 classNames_.push_back(cls.name);
105 }
106 // Drop a selection whose class vanished (e.g. script unloaded).
107 if (!selectedClass_.empty() &&
108 std::find(classNames_.begin(), classNames_.end(), selectedClass_) ==
109 classNames_.end()) {
110 selectedClass_.clear();
111 instances_.clear();
112 selectedInstance_ = -1;
113 }
114 if (selectedClass_.empty() && !classNames_.empty())
115 selectClass(classNames_.front());
116 cachedMembers_.clear();
117 rebuildHost();
118}

262WidgetDesc Inspector::propertyWidget(const std::string& ownerClass,
263 const ReflectedMember& member,
264 const ReflectedValue& value,
265 const ssq::Object& instance) {
266 const std::string id = "prop_" + member.name;
267 const std::string label = memberLabel(ownerClass, member.name);
268 switch (editorKind(member, value)) {
269 case EditorKind::Checkbox:
270 return checkbox(label, value.asBool(), id,
271 [this, name = member.name](bool v) {
272 ReflectedValue out;
273 out.kind = ReflectedValueKind::Bool;
274 out.boolean = v;
275 writeProperty(name, std::move(out));
276 });
277 case EditorKind::Slider: {
278 const float minV = member.attrFloat("min", 0.f);
279 const float maxV = member.attrFloat("max", 1.f);
280 const float cur = value.kind == ReflectedValueKind::Integer
281 ? float(value.integer)
282 : float(value.floating);
283 return slider(label, cur, minV, maxV, id,
284 [this, name = member.name](float v) {
285 ReflectedValue out;
286 out.kind = ReflectedValueKind::Float;
287 out.floating = v;
288 writeProperty(name, std::move(out));
289 });
290 }
291 case EditorKind::Combo: {
292 const std::vector<std::string> options = member.attrOptions("options");
293 int index = 0;
294 const std::string current =
295 value.kind == ReflectedValueKind::String ? value.text
296 : valueText(value);
297 const auto it = std::find(options.begin(), options.end(), current);
298 if (it != options.end()) index = int(it - options.begin());
299 return combo(label, options, index, id,
300 [this, name = member.name, kind = value.kind, options](float i) {
301 const int idx = static_cast<int>(i);
302 ReflectedValue out;
303 if (idx < 0 || idx >= int(options.size())) return;
304 const std::string& option = options[size_t(idx)];
305 if (kind == ReflectedValueKind::Integer) {
306 out.kind = ReflectedValueKind::Integer;
307 out.integer = std::strtoll(option.c_str(), nullptr, 10);
308 } else if (kind == ReflectedValueKind::Float) {
309 out.kind = ReflectedValueKind::Float;
310 out.floating = std::strtod(option.c_str(), nullptr);
311 } else {
312 out.kind = ReflectedValueKind::String;
313 out.text = option;
314 }
315 writeProperty(name, std::move(out));
316 });
317 }
318 case EditorKind::Input:
319 return inputText(label, valueText(value), id,
320 [this, name = member.name](const std::string& text) {
321 ReflectedValue out;
322 out.kind = ReflectedValueKind::String;
323 out.text = text;
324 writeProperty(name, std::move(out));
325 });
326 case EditorKind::ReadOnly: {
327 if (value.kind == ReflectedValueKind::Array)
328 return arrayWidget(ownerClass, member, instance);
329 if (value.kind == ReflectedValueKind::Table)
330 return tableWidget(ownerClass, member, instance);
331 if (value.kind == ReflectedValueKind::Instance) {
332 const std::string openId = "open_" + ownerClass + "_" + member.name;
333 return button("open " + member.name + "##" + openId, openId,
334 [this, memberName = member.name]() {
335 Runtime* rt = runtime();
336 if (!rt) return;
337 const ssq::Object* inst = currentInstance();
338 if (!inst) return;
339 const ssq::Object nested =
340 rt->readObjectProperty(*inst, memberName);
341 if (nested.getType() != ssq::Type::INSTANCE) return;
342 const std::string nestedClass =
343 rt->classNameOf(nested);
344 if (nestedClass.empty()) return;
345 openNested(nestedClass, nested);
346 });
347 }
348 std::string shown = "null";
349 return text(member.name + " = " + shown, id);
350 }
351 }
352 return text(member.name, id);
353}

507WidgetDesc Inspector::build() {
508 std::vector<WidgetDesc> children;
509 if (classNames_.empty()) {
510 children.push_back(text(
511 "No reflected script classes. Run a script that defines classes, "
512 "then call ui.inspectRefresh().",
513 "hint"));
514 } else {
515 std::vector<WidgetDesc> toolbar = {
516 text("Class", "lbl_class"),
517 spacer("class_spacer"),
518 combo("##class", classNames_, currentClassIndex(), "class",
519 [this](float index) {
520 const int idx = static_cast<int>(index);
521 if (idx >= 0 && idx < int(classNames_.size()))
522 selectClass(classNames_[size_t(idx)]);
523 }),
524 };
525 if (pickScene_) {
526 toolbar.push_back(
527 button("Pick##inspector_pick", "inspector_pick",
528 [this]() {
529 if (!pickScene_) return;
530 const ssq::Object picked = pickScene_();
531 if (picked.getType() == ssq::Type::INSTANCE)
532 inspectObject(picked);
533 }));
534 }
535 if (!navStack_.empty()) {
536 toolbar.insert(toolbar.begin(),
537 button("<- Back##inspector_back", "inspector_back",
538 [this]() { back(); }));
539 }
540 children.push_back(row(
541 std::move(toolbar),
542 "classrow"));
543
544 std::vector<std::string> labels;
545 labels.reserve(instances_.size());
546 for (const InstanceEntry& entry : instances_) labels.push_back(entry.label);
547 children.push_back(row(
548 {
549 text("Instance", "lbl_instance"),
550 spacer("instance_spacer"),
551 combo("##instance", labels, selectedInstance_, "instance",
552 [this](float index) { selectInstance(static_cast<int>(index)); }),
553 button("+", "add_instance", [this]() { addInstance(); }),
554 },
555 "instancerow"));
556 children.push_back(separator("inspector_sep"));
557
558 Runtime* rt = runtime();
559 if (rt && selectedInstance_ >= 0 &&
560 size_t(selectedInstance_) < instances_.size()) {
561 const ssq::Object& instance =
562 instances_[size_t(selectedInstance_)].object;
563 // Inheritance chain: own class first, then bases (parent props are
564 // grouped under their owning class header).
565 std::vector<std::string> chain;
566 std::string current = selectedClass_;
567 while (!current.empty()) {
568 chain.push_back(current);
569 const ReflectedClass* cls = rt->reflectedClass(current);
570 if (!cls || cls->base.empty()) break;
571 current = cls->base;
572 }
573 for (const std::string& className : chain) {
574 const ReflectedClass* cls = rt->reflectedClass(className);
575 if (!cls) continue;
576 std::vector<WidgetDesc> props;
577 for (const ReflectedMember& member : cls->members) {
578 if (member.method) continue;
579 props.push_back(propertyWidget(
580 className, member, rt->readProperty(instance, member.name),
581 instance));
582 }
583 const bool own = className == selectedClass_;
584 const std::string headerLabel =
585 className + (own ? "" : " (base)") + "##cls_" + className;
586 if (props.empty()) {
587 children.push_back(
588 text(className + " (no editable properties)", "cls_" + className));
589 } else {
590 children.push_back(collapsingHeader(
591 headerLabel, std::move(props), "cls_" + className, own));
592 }
593 }
594 } else if (rt) {
595 children.push_back(text("Instance creation failed for " + selectedClass_,
596 "instance_error"));
597 }
598 }
599 return window("Inspector", std::move(children), "root");
600}

607void Inspector::sync() {
608 if (!host_ || !host_->meta()->visible) return;
609 Runtime* rt = runtime();
610 if (!rt || selectedInstance_ < 0 ||
611 size_t(selectedInstance_) >= instances_.size())
612 return;
613 const ssq::Object& instance = instances_[size_t(selectedInstance_)].object;
614 // Members may have changed shape after a script reload; refresh the cache.
615 if (cachedMembers_.empty())
616 cachedMembers_ = rt->reflectInstance(instance);
617 for (ReflectedMember& cached : cachedMembers_) {
618 const ReflectedValue live = rt->readProperty(instance, cached.name);
619 if (sameValue(live, cached.value)) continue;
620 cached.value = live;
621 const std::string id = "prop_" + cached.name;
622 switch (editorKind(cached, cached.value)) {
623 case EditorKind::Checkbox:
624 host_->setCheckedById(id, live.asBool());
625 break;
626 case EditorKind::Slider:
627 host_->setValueById(
628 id, live.kind == ReflectedValueKind::Integer
629 ? float(live.integer)
630 : float(live.floating));
631 break;
632 case EditorKind::Combo: {
633 const std::vector<std::string> options =
634 cached.attrOptions("options");
635 const std::string current =
636 live.kind == ReflectedValueKind::String ? live.text
637 : valueText(live);
638 const auto it = std::find(options.begin(), options.end(), current);
639 host_->setValueById(id,
640 it == options.end() ? 0.f
641 : float(it - options.begin()));
642 break;
643 }
644 case EditorKind::Input:
645 host_->setValueTextById(id, valueText(live));
646 break;
647 case EditorKind::ReadOnly:
648 break;
649 }
650 }
651}
