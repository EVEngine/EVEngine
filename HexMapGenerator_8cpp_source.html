34class HexRandom {
35public:
37 explicit HexRandom(std::uint32_t seed) noexcept : state_(seed == 0u ? 0x9E3779B9u : seed) {}
38
43 [[nodiscard]] float unit() noexcept {
44 state_ ^= state_ << 13u;
45 state_ ^= state_ >> 17u;
46 state_ ^= state_ << 5u;
47 // 24 significant bits keep the conversion exact in single precision.
48 return static_cast<float>(state_ >> 8) * (1.f / 16777216.f);
49 }
50
56 [[nodiscard]] std::int32_t index(std::int32_t count) noexcept {
57 if (count <= 0) return 0;
58 const auto span = static_cast<std::uint32_t>(count);
59 return static_cast<std::int32_t>(static_cast<std::uint32_t>(unit() * static_cast<float>(span))) % count;
60 }
61
68 [[nodiscard]] std::int32_t range(std::int32_t minimum, std::int32_t maximumExclusive) noexcept {
69 if (maximumExclusive <= minimum) return minimum;
70 return minimum + index(maximumExclusive - minimum);
71 }
72
78 [[nodiscard]] bool chance(float probability) noexcept { return unit() < probability; }
79
80private:
81 std::uint32_t state_;
82};

90[[nodiscard]] std::uint32_t mix(std::uint32_t x, std::uint32_t y, std::uint32_t z) noexcept {
91 std::uint32_t h = x * 0x9E3779B1u;
92 h ^= y * 0x85EBCA77u;
93 h = (h ^ (h >> 15)) * 0xC2B2AE3Du;
94 h ^= z * 0x27D4EB2Fu;
95 h = (h ^ (h >> 13)) * 0x165667B1u;
96 return h ^ (h >> 16);
97}

101 const std::uint32_t hashed = mix(static_cast<std::uint32_t>(ix), static_cast<std::uint32_t>(iz), channel);

117[[nodiscard]] float cellNoise(std::int32_t x, std::int32_t z, std::uint32_t channel) noexcept {
118 constexpr float kScale = 0.1f;
119 const float fx = static_cast<float>(x) * kScale;
120 const float fz = static_cast<float>(z) * kScale;
121 const float bx = std::floor(fx);
122 const float bz = std::floor(fz);
123 const auto ix = static_cast<std::int32_t>(bx);
124 const auto iz = static_cast<std::int32_t>(bz);
125 const float tx = fx - bx;
126 const float tz = fz - bz;
127 const float sx = tx * tx * (3.f - 2.f * tx);
128 const float sz = tz * tz * (3.f - 2.f * tz);
129
130 const float c00 = lattice(ix, iz, channel);
131 const float c10 = lattice(ix + 1, iz, channel);
132 const float c01 = lattice(ix, iz + 1, channel);
133 const float c11 = lattice(ix + 1, iz + 1, channel);
134 const float top = c00 + (c10 - c00) * sx;
135 const float bottom = c01 + (c11 - c01) * sx;
136 return top + (bottom - top) * sz;
137}

267void createRegions(std::int32_t count, std::int32_t cellCountX, std::int32_t cellCountZ,
268 const NormalizedSettings& settings, std::vector<MapRegion>& out) {
269 out.clear();
270 const std::int32_t borderX = settings.mapBorderX;
271 const std::int32_t borderZ = settings.mapBorderZ;
272 const std::int32_t midX = cellCountX / 2;
273 const std::int32_t midZ = cellCountZ / 2;
274 const std::int32_t thirdX = cellCountX / 3;
275 const std::int32_t border = settings.regionBorder;
276
277 switch (count) {
278 case 2: {
279 MapRegion first;
280 MapRegion second;
281 first.zMin = borderZ;
282 first.zMax = cellCountZ - borderZ;
283 second.zMin = borderZ;
284 second.zMax = cellCountZ - borderZ;
285 first.xMin = borderX;
286 first.xMax = midX - border;
287 second.xMin = midX + border;
288 second.xMax = cellCountX - borderX;
289 out.push_back(first);
290 out.push_back(second);
291 break;
292 }
293 case 3: {
294 MapRegion first;
295 MapRegion second;
296 MapRegion third;
297 first.zMin = borderZ;
298 first.zMax = cellCountZ - borderZ;
299 second.zMin = borderZ;
300 second.zMax = cellCountZ - borderZ;
301 third.zMin = borderZ;
302 third.zMax = cellCountZ - borderZ;
303 first.xMin = borderX;
304 first.xMax = thirdX - border;
305 second.xMin = thirdX + border;
306 second.xMax = cellCountX * 2 / 3 - border;
307 third.xMin = cellCountX * 2 / 3 + border;
308 third.xMax = cellCountX - borderX;
309 out.push_back(first);
310 out.push_back(second);
311 out.push_back(third);
312 break;
313 }
314 case 4: {
315 MapRegion first;
316 MapRegion second;
317 MapRegion third;
318 MapRegion fourth;
319 first.xMin = borderX;
320 first.zMin = borderZ;
321 first.xMax = midX - border;
322 first.zMax = midZ - border;
323 second.xMin = midX + border;
324 second.zMin = borderZ;
325 second.xMax = cellCountX - borderX;
326 second.zMax = midZ - border;
327 third.xMin = midX + border;
328 third.zMin = midZ + border;
329 third.xMax = cellCountX - borderX;
330 third.zMax = cellCountZ - borderZ;
331 fourth.xMin = borderX;
332 fourth.zMin = midZ + border;
333 fourth.xMax = midX - border;
334 fourth.zMax = cellCountZ - borderZ;
335 out.push_back(first);
336 out.push_back(second);
337 out.push_back(third);
338 out.push_back(fourth);
339 break;
340 }
341 default: {
342 MapRegion only;
343 only.xMin = borderX;
344 only.xMax = cellCountX - borderX;
345 only.zMin = borderZ;
346 only.zMax = cellCountZ - borderZ;
347 out.push_back(only);
348 break;
349 }

477std::int32_t raiseTerrain(HexMap& map, GeneratorScratch& scratch, HexRandom& random, const NormalizedSettings& settings,
478 const MapRegion& region, std::int32_t chunkSize, std::int32_t budget) {
479 const std::int32_t phase = scratch.search.beginPhase();
480 const std::int32_t first = randomCellIndex(map, region, random);
481 scratch.search.data(first).searchPhase = phase;
482 scratch.search.enqueue(first);
483 const HexCoordinates center = map.coordinatesAt(first);
484
485 const std::int32_t rise = random.chance(settings.highRiseProbability) ? 2 : 1;
486 std::int32_t size = 0;
487 while (size < chunkSize) {
488 std::int32_t index = -1;
489 if (scratch.search.dequeue(index) != HexSearchPop::Cell) break;
490
491 const std::int32_t originalElevation = map.elevation(map.coordinatesAt(index));
492 const std::int32_t newElevation = originalElevation + rise;
493 if (newElevation > settings.elevationMaximum) continue;
494
495 const HexCoordinates coordinates = map.coordinatesAt(index);
496 writeElevation(map, coordinates, newElevation);
497 if (originalElevation < settings.waterLevel && newElevation >= settings.waterLevel) {
498 --budget;
499 if (budget == 0) break;
500 }
501 ++size;
502
503 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
504 HexCoordinates neighbour{};
505 if (!map.getNeighbor(coordinates, static_cast<HexDirection>(i), neighbour)) continue;
506 const std::int32_t neighbourIndex = map.indexOf(neighbour);
507 if (neighbourIndex < 0) continue;
508 if (scratch.search.data(neighbourIndex).searchPhase >= phase) continue;
509 HexSearchData& record = scratch.search.data(neighbourIndex);
510 record.searchPhase = phase;
511 record.distance = neighbour.distanceTo(center);
512 record.heuristic = random.chance(settings.jitterProbability) ? 1 : 0;
513 scratch.search.enqueue(neighbourIndex);
514 }
515 }
516 return budget;
517}

535std::int32_t sinkTerrain(HexMap& map, GeneratorScratch& scratch, HexRandom& random, const NormalizedSettings& settings,
536 const MapRegion& region, std::int32_t chunkSize, std::int32_t budget) {
537 const std::int32_t phase = scratch.search.beginPhase();
538 const std::int32_t first = randomCellIndex(map, region, random);
539 scratch.search.data(first).searchPhase = phase;
540 scratch.search.enqueue(first);
541 const HexCoordinates center = map.coordinatesAt(first);
542
543 const std::int32_t sink = random.chance(settings.highRiseProbability) ? 2 : 1;
544 std::int32_t size = 0;
545 while (size < chunkSize) {
546 std::int32_t index = -1;
547 if (scratch.search.dequeue(index) != HexSearchPop::Cell) break;
548
549 const HexCoordinates coordinates = map.coordinatesAt(index);
550 const std::int32_t originalElevation = map.elevation(coordinates);
551 const std::int32_t newElevation = originalElevation - sink;
552 if (newElevation < settings.elevationMinimum) continue;
553
554 writeElevation(map, coordinates, newElevation);
555 if (originalElevation >= settings.waterLevel && newElevation < settings.waterLevel) ++budget;
556 ++size;
557
558 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
559 HexCoordinates neighbour{};
560 if (!map.getNeighbor(coordinates, static_cast<HexDirection>(i), neighbour)) continue;
561 const std::int32_t neighbourIndex = map.indexOf(neighbour);
562 if (neighbourIndex < 0) continue;
563 if (scratch.search.data(neighbourIndex).searchPhase >= phase) continue;
564 HexSearchData& record = scratch.search.data(neighbourIndex);
565 record.searchPhase = phase;
566 record.distance = neighbour.distanceTo(center);
567 record.heuristic = random.chance(settings.jitterProbability) ? 1 : 0;
568 scratch.search.enqueue(neighbourIndex);
569 }
570 }
571 return budget;
572}

588void createLand(HexMap& map, GeneratorScratch& scratch, HexRandom& random, const NormalizedSettings& settings,
589 const std::vector<MapRegion>& regions) {
590 const std::int32_t cellCount = map.cellCount();
591 const float cellShare = static_cast<float>(cellCount) * 0.01f;
592 std::int32_t landBudget = roundToInt(cellShare * static_cast<float>(settings.landPercentage));
593 scratch.landCells = landBudget;
594
595 for (std::int32_t guard = 0; guard < 10000; ++guard) {
596 const bool sink = random.chance(settings.sinkProbability);
597 for (const MapRegion& region : regions) {
598 const std::int32_t chunkSize = random.range(settings.chunkSizeMin, settings.chunkSizeMax - 1);
599 if (sink) {
600 landBudget = sinkTerrain(map, scratch, random, settings, region, chunkSize, landBudget);
601 } else {
602 landBudget = raiseTerrain(map, scratch, random, settings, region, chunkSize, landBudget);
603 if (landBudget == 0) return;
604 }
605 }
606 }
607 if (landBudget > 0) {
608 // The reference logs a warning here; the port records the shortfall in the
609 // land count instead, which is what the river budget is derived from.
610 scratch.landCells -= landBudget;
611 }
612}

624[[nodiscard]] bool isErodible(const HexMap& map, HexCoordinates coordinates) {
625 const std::int32_t threshold = map.elevation(coordinates) - 2;
626 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
627 HexCoordinates neighbour{};
628 if (!map.getNeighbor(coordinates, static_cast<HexDirection>(i), neighbour)) continue;
629 if (map.elevation(neighbour) <= threshold) return true;
630 }
631 return false;
632}

645[[nodiscard]] std::int32_t erosionTarget(const HexMap& map, HexCoordinates coordinates) {
646 const std::int32_t threshold = map.elevation(coordinates) - 2;
647 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
648 HexCoordinates neighbour{};
649 if (!map.getNeighbor(coordinates, static_cast<HexDirection>(i), neighbour)) continue;
650 if (map.elevation(neighbour) <= threshold) return map.indexOf(neighbour);
651 }
652 return -1;
653}

683void removeErodible(GeneratorScratch& scratch, std::int32_t index) {
684 if (index < 0 || static_cast<std::size_t>(index) >= scratch.erodibleFlag.size()) return;

725void erodeOnce(HexMap& map, GeneratorScratch& scratch, std::int32_t cellIndex) {
726 if (cellIndex < 0 || cellIndex >= map.cellCount()) return;
727 const HexCoordinates cellCoordinates = map.coordinatesAt(cellIndex);
728 const std::int32_t cellElevation = map.elevation(cellCoordinates);
729 const std::int32_t targetIndex = erosionTarget(map, cellCoordinates);
730 if (targetIndex < 0) {
731 removeErodible(scratch, cellIndex);
732 return;
733 }
734 const HexCoordinates targetCoordinates = map.coordinatesAt(targetIndex);
735 const std::int32_t targetElevation = map.elevation(targetCoordinates);
736
737 // Erosion moves land around rather than removing it, so the cell itself stays
738 // land; the running count only has to follow the erosion it performs.
739 --scratch.landCells;
740 writeElevation(map, cellCoordinates, cellElevation - 1);
741 writeElevation(map, targetCoordinates, targetElevation + 1);
742 requeueErodible(scratch, cellIndex, cellElevation);
743 requeueErodible(scratch, targetIndex, targetElevation);
744
745 if (!isErodible(map, cellCoordinates)) removeErodible(scratch, cellIndex);
746
747 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
748 HexCoordinates neighbour{};
749 if (!map.getNeighbor(cellCoordinates, static_cast<HexDirection>(i), neighbour)) continue;
750 const std::int32_t neighbourIndex = map.indexOf(neighbour);
751 if (neighbourIndex < 0) continue;
752 const std::int32_t neighbourElevation = map.elevation(neighbour);

754 addErodible(scratch, neighbourIndex, neighbourElevation);

824[[nodiscard]] float determineTemperature(const HexMap& map, HexCoordinates coordinates,
825 const NormalizedSettings& settings, std::int32_t jitterChannel) noexcept {
826 const float latitude = static_cast<float>(coordinates.z) / static_cast<float>(map.cellCountZ());
827 float temperature = settings.lowTemperature + (settings.highTemperature - settings.lowTemperature) * latitude;
828
829 const std::int32_t viewElevation = map.values(coordinates).viewElevation();
830 const float span = static_cast<float>(settings.elevationMaximum - settings.waterLevel + 1);
831 const float elevationFactor =
832 1.f - static_cast<float>(viewElevation - settings.waterLevel) / (span == 0.f ? 1.f : span);
833 temperature *= elevationFactor;
834
835 const HexVec3 position = map.cellGroundPosition(coordinates);
836 const auto channel = static_cast<std::uint32_t>(clampInt(jitterChannel, 0, 3)) * 0x9E3779B9u;
837 const float jitter =
838 cellNoise(static_cast<std::int32_t>(position.x), static_cast<std::int32_t>(position.z), channel);
839 return temperature + (jitter * 2.f - 1.f) * settings.temperatureJitter;
840}

854void evolveClimate(const HexMap& map, GeneratorScratch& scratch, const NormalizedSettings& settings,
855 std::int32_t cellIndex) {
856 const HexCoordinates coordinates = map.coordinatesAt(cellIndex);
857 const HexValues values = map.values(coordinates);
858 ClimateData cellClimate = scratch.climate[static_cast<std::size_t>(cellIndex)];
859
860 if (values.isUnderwater()) {
861 cellClimate.moisture = 1.f;
862 cellClimate.clouds += settings.evaporationFactor;
863 } else {
864 const float evaporation = cellClimate.moisture * settings.evaporationFactor;
865 cellClimate.moisture -= evaporation;
866 cellClimate.clouds += evaporation;
867 }
868
869 const float precipitation = cellClimate.clouds * settings.precipitationFactor;
870 cellClimate.clouds -= precipitation;
871 cellClimate.moisture += precipitation;
872
873 const float maximumElevation = static_cast<float>(settings.elevationMaximum);
874 const float cloudMaximum = 1.f - static_cast<float>(values.viewElevation()) / (maximumElevation + 1.f);
875 if (cellClimate.clouds > cloudMaximum) {
876 cellClimate.moisture += cellClimate.clouds - cloudMaximum;
877 cellClimate.clouds = cloudMaximum;
878 }
879
880 const HexDirection mainDispersalDirection = opposite(settings.windDirection);
881 const float cloudDispersal = cellClimate.clouds * (1.f / (5.f + settings.windStrength));
882 const float runoff = cellClimate.moisture * settings.runoffFactor * (1.f / 6.f);
883 const float seepage = cellClimate.moisture * settings.seepageFactor * (1.f / 6.f);
884
885 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
886 const auto direction = static_cast<HexDirection>(i);
887 HexCoordinates neighbour{};
888 if (!map.getNeighbor(coordinates, direction, neighbour)) continue;
889 const std::int32_t neighbourIndex = map.indexOf(neighbour);
890 if (neighbourIndex < 0) continue;
891
892 ClimateData neighbourClimate = scratch.nextClimate[static_cast<std::size_t>(neighbourIndex)];
893 if (direction == mainDispersalDirection) {
894 neighbourClimate.clouds += cloudDispersal * settings.windStrength;
895 } else {
896 neighbourClimate.clouds += cloudDispersal;
897 }
898
899 const std::int32_t elevationDelta = map.values(neighbour).viewElevation() - values.viewElevation();
900 if (elevationDelta < 0) {
901 cellClimate.moisture -= runoff;
902 neighbourClimate.moisture += runoff;
903 } else if (elevationDelta == 0) {
904 cellClimate.moisture -= seepage;
905 neighbourClimate.moisture += seepage;
906 }
907 scratch.nextClimate[static_cast<std::size_t>(neighbourIndex)] = neighbourClimate;
908 }
909
910 ClimateData nextCell = scratch.nextClimate[static_cast<std::size_t>(cellIndex)];
911 nextCell.moisture += cellClimate.moisture;
912 if (nextCell.moisture > 1.f) nextCell.moisture = 1.f;
913 scratch.nextClimate[static_cast<std::size_t>(cellIndex)] = nextCell;
914 scratch.climate[static_cast<std::size_t>(cellIndex)] = ClimateData{};
915}

927void createClimate(const HexMap& map, GeneratorScratch& scratch, const NormalizedSettings& settings) {
928 const auto cellCount = static_cast<std::size_t>(map.cellCount());
929 ClimateData initial;
930 initial.moisture = settings.startingMoisture;
931 scratch.climate.assign(cellCount, initial);
932 scratch.nextClimate.assign(cellCount, ClimateData{});
933
934 for (std::int32_t cycle = 0; cycle < 40; ++cycle) {
935 for (std::int32_t index = 0; index < map.cellCount(); ++index) evolveClimate(map, scratch, settings, index);
936 std::swap(scratch.climate, scratch.nextClimate);
937 }

966 scratch.riverOrigins.push_back(index);

989std::int32_t createRiver(HexMap& map, GeneratorScratch& scratch, HexRandom& random, const NormalizedSettings& settings,
990 std::int32_t originIndex) {
991 std::int32_t length = 1;
992 HexCoordinates cellCoordinates = map.coordinatesAt(originIndex);
993 HexValues cellValues = map.values(cellCoordinates);
994 bool cellUnderwater = cellValues.isUnderwater();
995 // The direction taken on the previous step; it is only read once the river is
996 // longer than one cell, which is exactly when the reference starts using it.
997 HexDirection direction = HexDirection::NE;
998
999 while (!cellUnderwater) {
1000 std::int32_t minNeighborElevation = std::numeric_limits<std::int32_t>::max();
1001 scratch.flowDirections.clear();
1002 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
1003 const auto candidate = static_cast<HexDirection>(i);
1004 HexCoordinates neighbour{};
1005 if (!map.getNeighbor(cellCoordinates, candidate, neighbour)) continue;
1006 const std::int32_t neighbourIndex = map.indexOf(neighbour);
1007 if (neighbourIndex < 0) continue;
1008 const HexCellData* neighbourCell = map.cellAt(neighbourIndex);
1009 if (neighbourCell == nullptr) continue;
1010
1011 const std::int32_t neighbourElevation = neighbourCell->values.elevation();
1012 if (neighbourElevation < minNeighborElevation) minNeighborElevation = neighbourElevation;
1013
1014 if (neighbourIndex == originIndex || neighbourCell->flags.hasAnyRiverIn()) continue;
1015
1016 const std::int32_t delta = neighbourElevation - cellValues.elevation();
1017 if (delta > 0) continue;
1018
1019 if (neighbourCell->flags.hasAnyRiverOut()) {
1020 map.setOutgoingRiver(cellCoordinates, candidate).ignore("river join is validated by the map");
1021 return length;
1022 }
1023
1024 if (delta < 0) {
1025 scratch.flowDirections.push_back(candidate);
1026 scratch.flowDirections.push_back(candidate);
1027 scratch.flowDirections.push_back(candidate);
1028 }
1029 // A reversal is discouraged once the river has a direction; on the first
1030 // step there is none, so every candidate keeps its weight.
1031 if (length == 1 || (candidate != next2(direction) && candidate != previous2(direction))) {
1032 scratch.flowDirections.push_back(candidate);
1033 }
1034 scratch.flowDirections.push_back(candidate);
1035 }
1036
1037 if (scratch.flowDirections.empty()) {
1038 if (length == 1) return 0;
1039 if (minNeighborElevation >= cellValues.elevation()) {
1040 writeWaterLevel(map, cellCoordinates, minNeighborElevation);
1041 if (minNeighborElevation == cellValues.elevation()) {
1042 writeElevation(map, cellCoordinates, minNeighborElevation - 1);
1043 }
1044 }
1045 break;
1046 }
1047
1048 const auto chosen = scratch.flowDirections[static_cast<std::size_t>(
1049 random.index(static_cast<std::int32_t>(scratch.flowDirections.size())))];
1050 const std::int32_t outgoingIndex = map.indexOf(cellCoordinates.step(chosen));
1051 if (outgoingIndex < 0) break;
1052
1053 map.setOutgoingRiver(cellCoordinates, chosen).ignore("flow direction was selected downhill");
1054 direction = chosen;
1055 ++length;
1056
1057 if (minNeighborElevation >= cellValues.elevation() && random.chance(settings.extraLakeProbability)) {
1058 writeWaterLevel(map, cellCoordinates, cellValues.elevation());
1059 writeElevation(map, cellCoordinates, cellValues.elevation() - 1);
1060 }
1061
1062 cellCoordinates = map.coordinatesAt(outgoingIndex);
1063 cellValues = map.values(cellCoordinates);
1064 cellUnderwater = cellValues.isUnderwater();
1065 }
1066 return length;
1067}

1081void createRivers(HexMap& map, GeneratorScratch& scratch, HexRandom& random, const NormalizedSettings& settings) {
1082 collectRiverOrigins(map, scratch, settings);
1083 std::int32_t riverBudget =
1084 roundToInt(static_cast<float>(scratch.landCells) * static_cast<float>(settings.riverPercentage) * 0.01f);
1085 while (riverBudget > 0 && !scratch.riverOrigins.empty()) {
1086 const auto position =
1087 static_cast<std::size_t>(random.index(static_cast<std::int32_t>(scratch.riverOrigins.size())));
1088 const std::int32_t originIndex = scratch.riverOrigins[position];
1089 scratch.riverOrigins[position] = scratch.riverOrigins.back();
1090 scratch.riverOrigins.pop_back();
1091
1092 const HexCellData* origin = map.cellAt(originIndex);
1093 if (origin == nullptr || origin->flags.hasRiver()) continue;
1094 bool validOrigin = true;
1095 for (std::int32_t i = 0; i < kHexDirectionCount && validOrigin; ++i) {
1096 HexCoordinates neighbour{};
1097 if (!map.getNeighbor(map.coordinatesAt(originIndex), static_cast<HexDirection>(i), neighbour)) continue;
1098 const HexCellData* neighbourCell = map.cell(neighbour);
1099 if (neighbourCell == nullptr) continue;
1100 if (neighbourCell->flags.hasRiver() || neighbourCell->values.isUnderwater()) validOrigin = false;
1101 }
1102 if (!validOrigin) continue;
1103 riverBudget -= createRiver(map, scratch, random, settings, originIndex);
1104 }
1105}

1123[[nodiscard]] std::int32_t underwaterTerrain(const HexMap& map, HexCoordinates coordinates,
1124 const NormalizedSettings& settings) {
1125 const std::int32_t elevation = map.elevation(coordinates);
1126 if (elevation == settings.waterLevel - 1) {
1127 const std::int32_t water = waterLevelOf(map, coordinates);
1128 std::int32_t cliffs = 0;
1129 std::int32_t slopes = 0;
1130 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
1131 HexCoordinates neighbour{};
1132 if (!map.getNeighbor(coordinates, static_cast<HexDirection>(i), neighbour)) continue;
1133 const std::int32_t delta = map.elevation(neighbour) - water;
1134 if (delta == 0) {
1135 ++slopes;
1136 } else if (delta > 0) {
1137 ++cliffs;
1138 }
1139 }
1140 if (cliffs + slopes > 3) return 1;
1141 if (cliffs > 0) return 3;
1142 if (slopes > 0) return 0;
1143 return 1;
1144 }
1145 if (elevation >= settings.waterLevel) return 1;
1146 if (elevation < 0) return 3;
1147 return 2;
1148}

1163void setTerrainType(HexMap& map, GeneratorScratch& scratch, HexRandom& random, const NormalizedSettings& settings) {
1164 const std::int32_t jitterChannel = random.index(4);
1165 const std::int32_t rockDesertElevation =
1166 settings.elevationMaximum - (settings.elevationMaximum - settings.waterLevel) / 2;
1167
1168 for (std::int32_t index = 0; index < map.cellCount(); ++index) {
1169 const HexCoordinates coordinates = map.coordinatesAt(index);
1170 HexValues values = map.values(coordinates);
1171 const float temperature = determineTemperature(map, coordinates, settings, jitterChannel);
1172 const float moisture = scratch.climate[static_cast<std::size_t>(index)].moisture;
1173
1174 if (!values.isUnderwater()) {
1175 std::int32_t temperatureBand = 0;
1176 for (; temperatureBand < 3; ++temperatureBand) {
1177 if (temperature < kTemperatureBands[temperatureBand]) break;
1178 }
1179 std::int32_t moistureBand = 0;
1180 for (; moistureBand < 3; ++moistureBand) {
1181 if (moisture < kMoistureBands[moistureBand]) break;
1182 }
1183 Biome biome = kBiomes[static_cast<std::size_t>(temperatureBand * 4 + moistureBand)];
1184
1185 if (biome.terrain == 0) {
1186 if (values.elevation() >= rockDesertElevation) biome.terrain = 3;
1187 } else if (values.elevation() == settings.elevationMaximum) {
1188 biome.terrain = 4;
1189 }
1190
1191 if (biome.terrain == 4) {
1192 biome.plant = 0;
1193 } else if (biome.plant < 3 && map.hasRiver(coordinates)) {
1194 ++biome.plant;
1195 }
1196 values = values.withTerrainType(biome.terrain).withPlantLevel(biome.plant);
1197 } else {
1198 std::int32_t terrain = underwaterTerrain(map, coordinates, settings);
1199 if (terrain == 1 && temperature < kTemperatureBands[0]) terrain = 2;
1200 values = values.withTerrainType(terrain);
1201 }
1202 map.setCellState(coordinates, values, map.flags(coordinates))
1203 .ignore("terrain classification writes the values it just read back");
1204 }
