43Result<ProviderHandle> ExtensionProviderRegistry::registerProvider(
44 ExtensionDescriptor descriptor, std::shared_ptr<IEditingExtensionProvider> provider) {
45 if (descriptor.id.empty() || descriptor.schemaVersion == 0 || !provider)
46 return failure<ProviderHandle>(Status::Rejected, "editing.extension.invalid-provider",
47 "Extension provider requires id, schema version and implementation");
48 {
49 std::scoped_lock lock(impl_->mutex);
50 if (impl_->entries.contains(descriptor.id))
51 return failure<ProviderHandle>(Status::Conflict, "editing.extension.duplicate-provider",
52 "Extension provider id is already published");
53 }
54 auto activated = provider->activate();
55 if (!activated.ok())
56 return failure<ProviderHandle>(Status::Failed, "editing.extension.activation-failed",
57 "Extension provider activation failed");
58 ProviderHandle handle;
59 bool duplicate = false;
60 {
61 std::scoped_lock lock(impl_->mutex);
62 if (impl_->entries.contains(descriptor.id)) {
63 duplicate = true;
64 } else {
65 const std::string providerId = descriptor.id;
66 handle = {providerId, ++impl_->generations[providerId]};
67 impl_->entries.emplace(providerId, Impl::Entry{std::move(descriptor), handle, provider});
68 }
69 }
70 if (duplicate) {
71 provider->deactivate();
72 return failure<ProviderHandle>(Status::Conflict, "editing.extension.duplicate-provider",
73 "Extension provider id was concurrently published");
74 }
75 return eve::editing::applied<ProviderHandle>(handle);
76}

78Result<ProviderLease> ExtensionProviderRegistry::acquire(const ProviderHandle& handle) const {
79 std::scoped_lock lock(impl_->mutex);
80 const auto found = impl_->entries.find(handle.id);
81 if (found == impl_->entries.end()) {
82 const auto generation = impl_->generations.find(handle.id);
83 return failure<ProviderLease>(
84 generation == impl_->generations.end() ? Status::Unsupported : Status::Conflict,
85 generation == impl_->generations.end() ? "editing.extension.provider-absent"
86 : "editing.extension.stale-handle",
87 generation == impl_->generations.end() ? "Editing extension provider is not available"
88 : "Editing extension handle is stale");
89 }
90 if (found->second.handle != handle)
91 return failure<ProviderLease>(Status::Conflict, "editing.extension.stale-handle",
92 "Editing extension handle is stale");
93 return eve::editing::applied<ProviderLease>(ProviderLease(handle, found->second.provider));
94}

105Result<void> ExtensionProviderRegistry::unload(const ProviderHandle& handle) {
106 std::shared_ptr<IEditingExtensionProvider> provider;
107 {
108 std::scoped_lock lock(impl_->mutex);
109 const auto found = impl_->entries.find(handle.id);
110 if (found == impl_->entries.end() || found->second.handle != handle)
111 return failure<void>(Status::Conflict, "editing.extension.stale-handle",
112 "Editing extension handle is stale");
113 provider = found->second.provider;
114 }
115 auto stopping = provider->beginUnload();
116 if (!stopping.ok())
117 return failure<void>(Status::Failed, "editing.extension.unload-rejected",
118 "Editing extension refused to stop accepting work");
119 {
120 std::scoped_lock lock(impl_->mutex);
121 const auto found = impl_->entries.find(handle.id);
122 if (found == impl_->entries.end() || found->second.handle != handle)
123 return failure<void>(Status::Conflict, "editing.extension.stale-handle",
124 "Editing extension handle is stale");
125 impl_->entries.erase(found);
126 }
127 provider->deactivate();
128 return eve::editing::applied<void>();
129}

145 Result<void> activate() override {
146 accepting_.store(true, std::memory_order_release);
147 return eve::editing::applied<void>();

149 Result<void> beginUnload() override {
150 accepting_.store(false, std::memory_order_release);
151 return eve::editing::applied<void>();
152 }
153 void deactivate() noexcept override { accepting_.store(false, std::memory_order_release); }
154 void* query(const CapabilityId& capability) noexcept override {
155 return accepting_.load(std::memory_order_acquire) && capability == capability_
156 ? implementation_.get()
157 : nullptr;
158 }

160private:

162 std::shared_ptr<void> implementation_;
163 std::atomic_bool accepting_{false};
