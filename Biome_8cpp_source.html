12uint32_t mixBiome(uint32_t value) {
13 value += 0x9e3779b9u;
14 value = (value ^ (value >> 16u)) * 0x21f0aaadu;
15 value = (value ^ (value >> 15u)) * 0x735a2d97u;
16 return value ^ (value >> 15u);
17}

90PointSet* BiomeRules::generate(SpatialData* domain, float spacing, uint32_t seed, float jitter) {
91 error_.clear();
92 lastCandidateCount_ = 0;
93 lastOutputCount_ = 0;
94 if (!domain || spacing <= 0.f) {
95 error_ = "generate: requires domain and positive spacing";
96 return nullptr;
97 }
98 if (layers_.empty()) {
99 error_ = "generate: no biome layers";
100 return nullptr;
101 }
102 PointSet candidates = domain->sample(spacing, seed, jitter);
103 lastCandidateCount_ = candidates.getCount();
104 PointSet output;
105 for (size_t candidateIndex = 0; candidateIndex < candidates.points().size(); ++candidateIndex) {
106 const auto& candidate = candidates.points()[candidateIndex];
107 bool excluded = false;
108 for (const auto& exclusion : exclusions_) {
109 if (exclusion->contains(candidate.x, candidate.y, candidate.z)) {
110 excluded = true;
111 break;
112 }
113 }
114 if (excluded) continue;
115
116 const BiomeLayerRule* selected = nullptr;
117 for (const auto& layer : layers_) {
118 if (!layer.spatial->contains(candidate.x, candidate.y, candidate.z)) continue;
119 if (!selected || layer.priority > selected->priority) selected = &layer;
120 }
121 if (!selected || selected->assets.empty()) continue;
122 const uint32_t densitySeed = mixBiome(seed ^ candidate.seed ^ 0x243f6a88u);
123 if (biomeUnit(densitySeed) >= selected->density) continue;
124
125 float totalWeight = 0.f;
126 for (const auto& asset : selected->assets) totalWeight += asset.weight;
127 float choice = biomeUnit(densitySeed ^ 0xb7e15162u) * totalWeight;
128 const BiomeAssetRule* selectedAsset = &selected->assets.back();
129 for (const auto& asset : selected->assets) {
130 choice -= asset.weight;
131 if (choice <= 0.f) {
132 selectedAsset = &asset;
133 break;
134 }
135 }
136
137 ProcgenPoint point = candidate;
138 point.density = selected->density;
139 const float scaleT = biomeUnit(densitySeed ^ 0x9e3779b9u);
140 const float scale = selectedAsset->minScale + (selectedAsset->maxScale - selectedAsset->minScale) * scaleT;
141 point.scaleX = point.scaleY = point.scaleZ = scale;
142 if (selectedAsset->randomYaw) point.yaw = biomeUnit(densitySeed ^ 0xdeadbeefu) * 360.f;
143 const int outputIndex =
144 std::move(output.appendPointFrom(candidates, candidateIndex)).expect("biome output attribute schema");
145 output.mutablePoint(size_t(outputIndex)) = std::move(point);
146 output.trySetStringAttribute(outputIndex, "biome", selected->name).expect("biome metadata schema");
147 output.trySetStringAttribute(outputIndex, "asset", selectedAsset->asset).expect("biome asset metadata schema");
148 }
149 lastOutputCount_ = output.getCount();
150 return new PointSet(std::move(output));
151}
