60Result<StepReport> Crowd::advance(float dt) {
61 if (!std::isfinite(dt) || dt < 0.f)
62 return Result<StepReport>::failure(
63 Diagnostic::error(DiagnosticCode::InvalidArgument, "Crowd dt must be finite and nonnegative", "dt"));
64 auto duration = Duration::fromSeconds(dt);
65 if (!duration) return Result<StepReport>::failure(duration.status());
66 SimulationTick tick = impl_->tick;
67 if (!duration.value().isZero()) {
68 const auto next = tick.incremented();
69 if (!next)
70 return Result<StepReport>::failure(
71 Diagnostic::error(DiagnosticCode::PreconditionViolation, "Crowd tick exhausted", "tick"));
72 tick = *next;
73 }
74 return advanceInternal({tick, duration.value(), {}}, ClockAuthority::Compatibility);
75}

83Result<void> Crowd::checkClock(const CrowdStep& step, ClockAuthority authority) const {
84 const auto& d = *impl_;
85 const auto fail = [](const char* message) {
86 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Conflict, message, "crowdStep.tick"));
87 };
88 if (d.budgetActive) return fail("Crowd advancement cannot reenter");
89 if (step.delta.nanoseconds() < 0)
90 return Result<void>::failure(
91 Diagnostic::error(DiagnosticCode::InvalidArgument, "Negative Crowd duration", "crowdStep.delta"));
92 if (step.delta.isZero()) {
93 if (step.tick != d.tick) return fail("Zero-duration step must reconcile the current tick");
94 return Result<void>::success();
95 }
96 const auto next = d.tick.incremented();
97 if (!next || step.tick != *next) return fail("Crowd step must consume the next contiguous tick");
98 if (d.clock && *d.clock != authority) return fail("Cannot mix compatibility and external Crowd clocks");
99 auto elapsed = d.elapsed.tryAdd(step.delta);
100 if (!elapsed) return Result<void>::failure(elapsed.status());
101 return Result<void>::success();
102}
