72template <typename Base, typename Comp>
73int packViewComponent(ShaderSystem &sys, int binding, EcsGpuWorkspace<Comp> &workspace) {
74 const int floatsPer = componentFloatCount<Comp>();
75 int n = 0;
76 {
77 auto view = ecs::View<Base, Comp>();
78 for (auto it = view.begin(); it != view.end(); ++it)
79 ++n;
80 }
81 if (n <= 0) return 0;
82
83 workspace.floats_.resize(size_t(n) * size_t(floatsPer));
84 int i = 0;
85 auto view = ecs::View<Base, Comp>();
86 for (auto it = view.begin(); it != view.end(); ++it) {
87 auto [comp] = *it;
88 std::memcpy(workspace.floats_.data() + size_t(i) * size_t(floatsPer), static_cast<const void *>(comp),
89 sizeof(Comp));
90 ++i;
91 }
92 sys.upload(binding, workspace.floats_.data(), n * floatsPer);
93 return n;
94}

133template <typename Base, typename Comp>
134int packViewComponentRange(ShaderSystem &sys, int binding, int firstEntity, int entityCount,
135 EcsGpuWorkspace<Comp> &workspace) {
136 if (firstEntity < 0 || entityCount <= 0) return 0;
137 const int floatsPer = componentFloatCount<Comp>();
138 workspace.floats_.resize(size_t(entityCount) * size_t(floatsPer));
139 int viewIndex = 0;
140 int packed = 0;
141 auto view = ecs::View<Base, Comp>();
142 for (auto it = view.begin(); it != view.end() && packed < entityCount; ++it, ++viewIndex) {
143 if (viewIndex < firstEntity) continue;
144 auto [comp] = *it;
145 std::memcpy(workspace.floats_.data() + size_t(packed) * size_t(floatsPer), static_cast<const void *>(comp),
146 sizeof(Comp));
147 ++packed;
148 }
149 if (packed > 0) sys.uploadRange(binding, workspace.floats_.data(), packed * floatsPer, firstEntity * floatsPer);
150 return packed;
151}

154template <typename Base, typename Comp>
155int unpackViewComponentRange(ShaderSystem &sys, int binding, int firstEntity, int entityCount,
156 EcsGpuWorkspace<Comp> &workspace) {
157 if (firstEntity < 0 || entityCount <= 0) return 0;
158 const int available = std::max(0, countViewEntities<Base, Comp>() - firstEntity);
159 const int count = std::min(entityCount, available);
160 if (count <= 0) return 0;
161 const int floatsPer = componentFloatCount<Comp>();
162 workspace.floats_.resize(size_t(count) * size_t(floatsPer));
163 sys.downloadRange(binding, workspace.floats_.data(), count * floatsPer, firstEntity * floatsPer);
164 int viewIndex = 0;
165 int unpacked = 0;
166 auto view = ecs::View<Base, Comp>();
167 for (auto it = view.begin(); it != view.end() && unpacked < count; ++it, ++viewIndex) {
168 if (viewIndex < firstEntity) continue;
169 auto [comp] = *it;
170 std::memcpy(static_cast<void *>(comp), workspace.floats_.data() + size_t(unpacked) * size_t(floatsPer),
171 sizeof(Comp));
172 ++unpacked;
173 }
174 return unpacked;
175}
