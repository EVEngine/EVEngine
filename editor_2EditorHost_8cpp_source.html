43Result<EditorHostMountReport> EditorHost::mount(
44 const std::vector<std::string>& owners, const std::unordered_map<std::string, EditorHostMountOverride>& overrides) {
45 EditorHostMountReport report;
46 if (!owners.empty() && eve::cap::listenerCount<editing::IEditorPackageProvider>() == 0)
47 return eve::editing::failed<EditorHostMountReport>(EditorStatus::NotFound, RuleId("editor.host.no-providers"),
48 "No IEditorPackageProvider capabilities are loaded");
49
50 eve::cap::forEach<editing::IEditorPackageProvider>([&](editing::IEditorPackageProvider* provider) {
51 if (!provider) return;
52 const editing::EditorPackage& package = provider->package();
53 if (!ownerAllowed(owners, package.ownerModule)) return;
54
55 EditorHostMountOverride override;
56 if (const auto found = overrides.find(package.ownerModule); found != overrides.end())
57 override = found->second;
58 else
59 override.undoPolicy = package.undoPolicy;
60
61 auto entry = mountPackage(package, override);
62 if (!entry.ok()) {
63 EditorHostMountEntry skipped;
64 skipped.ownerModule = package.ownerModule;
65 skipped.skipped = true;
66 skipped.skipReason = entry.status().describe();
67 report.entries.push_back(std::move(skipped));
68 return;
69 }
70 if (!entry.value().skipped) ++report.mountedPackages;
71 report.entries.push_back(std::move(entry).takeValue());
72 });
73 return eve::editing::applied<EditorHostMountReport>(std::move(report));
74}

76Result<void> EditorHost::unmount(std::string_view ownerModule) {
77 const auto found = mounted_.find(std::string(ownerModule));
78 if (found == mounted_.end()) return eve::editing::noOp();
79 for (const std::string& panelId : found->second.panelIds) (void)workspace_.removePanel(panelId);
80 mounted_.erase(found);
81 hasPrimaryViewport_ = false;
82 for (const auto& [_, mounted] : mounted_)
83 for (const editing::EditorPanelSpec& panel : mounted.package.panels)
84 if (panel.primaryViewport) hasPrimaryViewport_ = true;
85 return eve::editing::applied<void>();
86}

128Result<EditorHostMountEntry> EditorHost::mountPackage(const editing::EditorPackage& package,
129 const EditorHostMountOverride& override) {
130 auto valid = editing::validateEditorPackage(package, override.allowIncomplete);
131 if (!valid.ok()) return Result<EditorHostMountEntry>::failure(valid.status());
132
133 if (mounted_.contains(package.ownerModule))
134 return eve::editing::failed<EditorHostMountEntry>(EditorStatus::Conflict, RuleId("editor.host.already-mounted"),
135 "Editor package already mounted: " + package.ownerModule);
136
137 for (const editing::EditorPanelSpec& panel : package.panels) {
138 if (panel.primaryViewport && hasPrimaryViewport_)
139 return eve::editing::failed<EditorHostMountEntry>(
140 EditorStatus::Conflict, RuleId("editor.host.primary-viewport"),
141 "Only one primaryViewport panel may be mounted on a host");
142 }
143
144 std::vector<std::string> panelIds;
145 auto installed = installPanels(package, panelIds);
146 if (!installed.ok()) {
147 for (const std::string& panelId : panelIds) (void)workspace_.removePanel(panelId);
148 return Result<EditorHostMountEntry>::failure(installed.status());
149 }
150
151 for (const editing::EditorPanelSpec& panel : package.panels)
152 if (panel.primaryViewport) hasPrimaryViewport_ = true;
153
154 MountedPackage mounted;
155 mounted.package = package;
156 mounted.package.undoPolicy = override.undoPolicy;
157 mounted.panelIds = std::move(panelIds);
158 mounted_.emplace(package.ownerModule, std::move(mounted));
159
160 EditorHostMountEntry entry;
161 entry.ownerModule = package.ownerModule;
162 entry.panelCount = package.panels.size();
163 entry.commandCount = package.commands.size();
164 entry.targetCount = package.targetTypes.size();
165 return eve::editing::applied<EditorHostMountEntry>(std::move(entry));
166}

169 for (const editing::EditorPanelSpec& panel : package.panels) {

172 descriptor.title = panel.title;
173 descriptor.region = panel.region;
174 descriptor.capability = panel.capability;
175 descriptor.context = panel.context;
176 descriptor.order = panel.order;

178 if (!registered.ok())
179 return eve::editing::failed<void>(registered.code(), RuleId("editor.host.panel-conflict"),
180 "Could not install package panel: " + panel.id);
181 panelIds.push_back(panel.id);
182 }
183 return eve::editing::applied<void>();

186Result<EditorValue> EditorHost::exportCommandCatalog() const {
187 if (!commands_)
188 return eve::editing::failed<EditorValue>(EditorStatus::Failed, RuleId("editor.host.no-commands"),
189 "EditorHost has no command service");
190 const auto owners = mountedOwnerNames();
191 const bool filter = !owners.empty();
192 EditorValue::Array commands;
193 for (const CommandDescriptor& command : commands_->commands(options_.profile)) {
194 if (filter && std::find(owners.begin(), owners.end(), command.ownerModule) == owners.end()) continue;
195 EditorValue::Object item;
196 item["id"] = command.id.value();
197 item["displayName"] = command.displayName;
198 item["category"] = command.category;
199 item["owner"] = command.ownerModule;
200 item["automationAllowed"] = command.automationAllowed;
201 if (!command.panelCapability.empty()) item["panelCapability"] = command.panelCapability;
202 if (!command.panelContext.empty()) item["panelContext"] = command.panelContext;
203 if (!command.targetKinds.empty()) {
204 EditorValue::Array kinds;
205 for (const std::string& kind : command.targetKinds) kinds.emplace_back(kind);
206 item["targetKinds"] = std::move(kinds);
207 }
208 auto schema = editing::propertySchemaToJsonSchema(command.payloadSchema, command.id.value());
209 if (schema.ok()) item["inputSchema"] = std::move(schema).takeValue();
210 commands.emplace_back(std::move(item));
211 }
212 return eve::editing::applied<EditorValue>(EditorValue::Object{{"commands", std::move(commands)}});
213}

215Result<EditorValue> EditorHost::exportTargetCatalog() const {
216 std::unordered_map<std::string, editing::PropertySchema> createSchemas;
217 for (const auto& [_, mounted] : mounted_) {
218 for (const editing::EditorTargetTypeSpec& target : mounted.package.targetTypes)
219 createSchemas[target.type] = target.createSchema;
220 }
221
222 const auto allowed = mountedTargetTypeNames();
223 const bool filter = !allowed.empty();
224 EditorValue::Array types;
225 eve::cap::forEach<IEditorAutomationTargetFactory>([&](IEditorAutomationTargetFactory* factory) {
226 if (!factory) return;
227 for (const std::string_view type : factory->types()) {
228 if (filter && std::find(allowed.begin(), allowed.end(), type) == allowed.end()) continue;
229 EditorValue::Object item;
230 item["type"] = std::string(type);
231 if (const auto found = createSchemas.find(std::string(type)); found != createSchemas.end()) {
232 auto schema = editing::propertySchemaToJsonSchema(found->second, type);
233 if (schema.ok()) item["createSchema"] = std::move(schema).takeValue();
234 }
235 types.emplace_back(std::move(item));
236 }
237 });
238 std::sort(types.begin(), types.end(), [](const EditorValue& left, const EditorValue& right) {
239 const auto* leftObject = left.getIf<EditorValue::Object>();
240 const auto* rightObject = right.getIf<EditorValue::Object>();
241 auto typeOf = [](const EditorValue::Object* object) -> std::string {
242 if (!object) return {};
243 const auto found = object->find("type");
244 if (found == object->end()) return {};
245 const auto* value = found->second.getIf<std::string>();
246 return value ? *value : std::string{};
247 };
248 return typeOf(leftObject) < typeOf(rightObject);
249 });
250 return eve::editing::applied<EditorValue>(EditorValue::Object{{"supportedTypes", std::move(types)}});
251}
