77#if !defined(ZEROERR_NO_ASSERT)
78 ~ResultObservation() noexcept(false) {
79 const bool observationSatisfied = !mustObserve_ || observed_;
80 EV_DETAIL_RESULT_ASSERT(observationSatisfied,
81 "Result destroyed without checking; inspect it or call ignore() explicitly");

92 template <bool AssertionsEnabled>
93 void assertCanBeOverwritten() const {
94 if constexpr (AssertionsEnabled) {
95 const bool observationSatisfied = !mustObserve_ || observed_;
96 checkResultInvariant(observationSatisfied, "!mustObserve_ || observed_", __FILE__, __LINE__,
97 "move assignment would overwrite an unchecked Result");
98 }
99 }

102 void adoptObservationFrom(ResultObservation& other) noexcept {
103 observed_ = other.observed_;
104 mustObserve_ = other.mustObserve_;
105 other.mustObserve_ = false;
106 }
107
109 void disarmObservation() noexcept { mustObserve_ = false; }

112 void requireObservation() noexcept {
113 observed_ = false;
114 mustObserve_ = true;
115 }
