43// Load a unique sibling image: the original can be rebuilt on Windows, and the
44// loader cannot reuse the previous image by filename/inode. Keeping the sibling
45// in the same directory preserves $ORIGIN / @loader_path dependency resolution.
46std::filesystem::path shadowCopy(const std::filesystem::path& source) {
47#if defined(_WIN32)
48 const auto process = GetCurrentProcessId();
49#else
50 const auto process = getpid();
51#endif
52 for (;;) {
53 const auto name = ".eve-" + std::to_string(process) + "-" + std::to_string(++images().serial) + "-" +
54 source.filename().string();
55 const auto shadow = source.parent_path() / name;
56 std::error_code error;
57 if (std::filesystem::copy_file(source, shadow, std::filesystem::copy_options::none, error)) return shadow;
58 if (error == std::errc::file_exists) continue;
59 const auto message = error.message();
60 std::filesystem::remove(shadow, error);
61 throw Exception("plugins: cannot copy library '%s': %s", source.string().c_str(), message.c_str());
62 }
63}

74Result<void> Plugins::install(const std::string& path, bool replace) {
75 bool registering = false;
76 try {
77 const std::string key = identity(path);
78 auto& state = images();
79 if (replace && !state.loaded.count(key)) return failure(DiagnosticCode::NotFound, "plugin is not loaded", key);
80 if (!replace && state.loaded.count(key)) return Result<void>::success(Status::success(StatusCode::NoOp));
81 // Allocate bookkeeping before native initialization can publish code pointers.
82 state.resident.reserve(state.resident.size() + 1);
83 auto registration = ModuleManager::beginPluginRegistration(key);
84 if (!registration.ok()) return registration;
85 registering = true;
86 const auto shadow = shadowCopy(key);
87#if defined(_WIN32)
88 HMODULE native = LoadLibraryExW(shadow.c_str(), nullptr,
89 LOAD_LIBRARY_SEARCH_DLL_LOAD_DIR | LOAD_LIBRARY_SEARCH_DEFAULT_DIRS);
90 void* mod = native;
91 const auto loadError = GetLastError();
92 if (!mod) {
93 std::error_code ignored;
94 std::filesystem::remove(shadow, ignored);
95 throw Exception("plugins: LoadLibrary failed (err=%lu)", loadError);
96 }
97 auto init = reinterpret_cast<PluginInitFn>(GetProcAddress(native, "eve_plugin_init"));
98#else
99 void* mod = dlopen(shadow.c_str(), RTLD_NOW | RTLD_LOCAL);
100 const std::string loadError = mod ? "" : dlerror();
101 std::error_code ignored;
102 std::filesystem::remove(shadow, ignored);
103 if (!mod) throw Exception("plugins: dlopen failed: %s", loadError.c_str());
104 auto init = reinterpret_cast<PluginInitFn>(dlsym(mod, "eve_plugin_init"));
105#endif
106 // Even a failed exposer can leave native closures in the VM's type map.
107 // Never unmap an initialized image while any VM may still refer to it.
108 state.resident.push_back(mod);
109 if (!init) throw Exception("plugins: missing eve_plugin_init");
110 const int rc = init();
111 if (rc != 0) throw Exception("plugins: eve_plugin_init returned %d", rc);
112 auto committed = ModuleManager::finishPluginRegistration(true);
113 registering = false;
114 if (!committed.ok()) return committed;
115 state.loaded.insert(key);
116 return Result<void>::success(Status::success(StatusCode::Applied));
117 } catch (const std::exception& error) {
118 if (registering) {
119 auto rollback = ModuleManager::finishPluginRegistration(false);
120 rollback.ignore("plugin candidate failed; preserve the previous generation");
121 }
122 return failure(DiagnosticCode::InvalidArgument, error.what(), path);
123 } catch (...) {
124 if (registering) {
125 auto rollback = ModuleManager::finishPluginRegistration(false);
126 rollback.ignore("plugin initializer threw; preserve the previous generation");
127 }
128 return failure(DiagnosticCode::InvalidArgument, "plugin initializer threw", path);
129 }
130}
