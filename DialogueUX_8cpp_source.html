54std::string DialogueUX::plainText(const std::string& richText) const {
55 std::string out;
56 out.reserve(richText.size());
57 for (size_t i = 0; i < richText.size();) {
58 if (richText[i] == '[') {
59 const size_t close = richText.find(']', i + 1);
60 if (close != std::string::npos && isControlTag(richText.substr(i + 1, close - i - 1))) {
61 i = close + 1;
62 continue;
63 }
64 }
65 out.push_back(richText[i++]);
66 }
67 return out;
68}

153bool DialogueVoice::registerClip(const std::string& lineId, const std::string& locale, float duration,
154 const std::string& envelope) {
155 if (lineId.empty() || duration < 0.f) return false;
156 Clip& clip = clips_[key(lineId, locale)];
157 clip.lineId = lineId;
158 clip.locale = locale;
159 clip.duration = duration;
160 clip.envelope.clear();
161 std::istringstream input(envelope);
162 std::string item;
163 while (std::getline(input, item, ',')) {
164 char* end = nullptr;
165 const float value = std::strtof(item.c_str(), &end);
166 if (end == item.c_str()) return false;
167 clip.envelope.push_back(std::clamp(value, 0.f, 1.f));
168 }
169 return true;
170}
