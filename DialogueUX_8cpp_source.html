44 const size_t equal = tag.find('=');
45 const std::string name = tag.substr(0, equal);

52std::string DialogueUX::plainText(const std::string& richText) const {
53 std::string out;
54 out.reserve(richText.size());
55 for (size_t i = 0; i < richText.size();) {
56 if (richText[i] == '[') {
57 const size_t close = richText.find(']', i + 1);
58 if (close != std::string::npos && isControlTag(richText.substr(i + 1, close - i - 1))) {
59 i = close + 1;
60 continue;
61 }
62 }
63 out.push_back(richText[i++]);
64 }
65 return out;
66}

68int DialogueUX::getTextActionCount(const std::string& richText) const {
69 int count = 0;
70 for (size_t i = 0; i < richText.size();) {
71 if (richText[i] == '[') {
72 const size_t close = richText.find(']', i + 1);
73 if (close != std::string::npos) {
74 if (isControlTag(richText.substr(i + 1, close - i - 1))) ++count;
75 i = close + 1;
76 continue;
77 }

80 }
