54Result<void> GroomAsset::validate() const {
55 if (groups_.empty()) {
56 return Result<void>::failure(Diagnostic::error(
57 DiagnosticCode::InvalidArgument, "GroomAsset: no groups", "hair.groom.groups"));
58 }
59 for (size_t i = 0; i < groups_.size(); ++i) {
60 auto ok = groups_[i].strands.validate();
61 if (!ok.ok()) return Result<void>::failure(ok.status());
62 if (groups_[i].guides.curveCount() > 0) {
63 auto guidesOk = groups_[i].guides.validate();
64 if (!guidesOk.ok()) return Result<void>::failure(guidesOk.status());
65 }
66 if (groups_[i].lods.empty()) {
67 return Result<void>::failure(Diagnostic::error(
68 DiagnosticCode::InvariantViolation, "GroomAsset: group has empty LOD table",
69 "hair.groom.groups[" + std::to_string(i) + "].lods"));
70 }
71 }
72 return Result<void>::success();
73}

75size_t selectLodIndex(const std::vector<GroomLod> &lods, float screenSize) {
76 if (lods.empty()) return 0;
77 // Ordered finest → coarsest. Pick the last entry whose threshold is still
78 // >= the current screen size (i.e. the coarsest LOD that still applies).
79 size_t best = 0;
80 for (size_t i = 0; i < lods.size(); ++i) {
81 if (screenSize <= lods[i].screenSize) best = i;
82 }
83 return best;
84}
