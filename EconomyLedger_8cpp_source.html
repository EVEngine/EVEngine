9int EconomyLedger::credit(const std::string& type, int amount) {
10 if (amount <= 0) return 0;
11 const auto* def = ResourceTypeRegistry::find(type);
12 const int cap = def ? def->stockMax : 0;
13
14 int& current = current_[type];
15 int accepted = amount;
16 if (cap > 0 && current + amount > cap) accepted = std::max(0, cap - current);
17 current += accepted;
18 income_[type] += accepted;
19 if (accepted < amount) wasted_[type] += amount - accepted;
20 return accepted;
21}

23bool EconomyLedger::debit(const std::string& type, int amount) {
24 if (amount <= 0) return true;
25 auto it = current_.find(type);
26 const int current = it == current_.end() ? 0 : it->second;
27 if (current < amount) return false;
28 it->second = current - amount;
29 expense_[type] += amount;
30 return true;
31}
