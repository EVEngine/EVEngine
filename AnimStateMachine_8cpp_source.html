31class AnimStateProvider : public eve::caps::IStateProvider {
32public:
33 const char *stateKind() const override { return "anim"; }
34
35 Result<Value> capture() override {
36 StateValue arr = StateValue::array();
37 for (AnimStateMachine *m : machines()) {
38 StateValue sv;
39 if (!m->captureState(sv))
40 return Result<Value>::failure(Diagnostic::error(DiagnosticCode::Failed, "animation capture failed"));
41 arr.pushBack(std::move(sv));
42 }
43 return Result<Value>::success(toCanonicalStateValue(arr));
44 }
45
46 Result<void> restore(const Value &value) override {
47 const StateValue in = toLegacyStateValue(value);
48 if (!in.isArray()) {
49 return Result<void>::failure(
50 Diagnostic::error(DiagnosticCode::InvalidArgument, "anim: expected array of state machines"));
51 }
52 const size_t n = std::min(in.arraySize(), machines().size());
53 for (size_t i = 0; i < n; ++i) {
54 std::string error;
55 if (!machines()[i]->restoreState(in.at(i), &error))
56 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, error));
57 }
58 return Result<void>::success();
59 }
60
61 Result<void> resetToDefaults() override {
62 for (AnimStateMachine *m : machines()) {
63 if (!m->resetToDefaults())
64 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Failed, "animation reset failed"));
65 }
66 return Result<void>::success();
67 }
68};

214bool AnimStateMachine::conditionsMet(const Transition &tr) const {
215 for (const Condition &c : tr.conditions) {
216 switch (c.kind) {
217 case CondKind::Float: {
218 const float v = getFloat(c.param);
219 bool ok = false;
220 switch (c.op) {
221 case FloatOp::Gt: ok = v > c.threshold; break;
222 case FloatOp::Ge: ok = v >= c.threshold; break;
223 case FloatOp::Lt: ok = v < c.threshold; break;
224 case FloatOp::Le: ok = v <= c.threshold; break;
225 case FloatOp::Eq: ok = std::fabs(v - c.threshold) < 1e-5f; break;
226 case FloatOp::Ne: ok = std::fabs(v - c.threshold) >= 1e-5f; break;
227 }
228 if (!ok) return false;
229 break;
230 }
231 case CondKind::Bool: {
232 if (getBool(c.param) != c.boolValue) return false;
233 break;
234 }
235 case CondKind::Trigger: {
236 auto it = triggers_.find(c.param);
237 if (it == triggers_.end() || !it->second) return false;
238 break;
239 }
240 }
241 }
242 return true;
243}

245bool AnimStateMachine::tryTransition() {
246 if (currentState_.empty() || blending_) return false;
247 const State &cur = states_.at(currentState_);
248 const float dur = cur.clip ? cur.clip->getDuration() : 0.f;
249 const float norm = dur > 1e-8f ? clampf(stateTime_ / dur, 0.f, 1.f) : 1.f;
250
251 for (const Transition &tr : transitions_) {
252 if (tr.from != currentState_) continue;
253 if (tr.hasExitTime && norm < tr.exitTime) continue;
254 if (!conditionsMet(tr)) continue;
255
256 // Consume triggers used by this transition.
257 for (const Condition &c : tr.conditions) {
258 if (c.kind == CondKind::Trigger) triggers_[c.param] = false;
259 }
260
261 nextState_ = tr.to;
262 blendDuration_ = tr.blendSeconds;
263 blendElapsed_ = 0.f;
264 stateTime_ = 0.f;
265 if (blendDuration_ <= 1e-6f) {
266 currentState_ = nextState_;
267 blending_ = false;
268 states_.at(currentState_).clip->sample(0.f, &pose_, skeleton_);
269 } else {
270 // Switch logical state immediately; pose cross-fades from previous.
271 fromPose_.copyFrom(&pose_);
272 currentState_ = nextState_;
273 blending_ = true;
274 states_.at(currentState_).clip->sample(0.f, &toPose_, skeleton_);
275 pose_.blendFrom(&fromPose_, &toPose_, 0.f);
276 }
277 return true;
278 }
279 return false;
280}

282void AnimStateMachine::updateUnchecked(float dt) {
283 if (dt < 0.f) throw Exception("AnimStateMachine.update: dt must be >= 0");
284 if (!started_) {
285 if (!stateOrder_.empty())
286 setEntry(stateOrder_.front());
287 else
288 return;
289 }
290 if (currentState_.empty()) return;
291
292 if (blending_) {
293 blendElapsed_ += dt;
294 stateTime_ += dt;
295 const State &dst = states_.at(currentState_);
296 dst.clip->sample(stateTime_, &toPose_, skeleton_);
297 float t = blendDuration_ > 1e-8f ? blendElapsed_ / blendDuration_ : 1.f;
298 if (t >= 1.f) {
299 blending_ = false;
300 pose_.copyFrom(&toPose_);
301 } else {
302 // Keep fromPose frozen at transition start for stable crossfade.
303 pose_.blendFrom(&fromPose_, &toPose_, t);
304 }
305 // Allow chained transitions after blend completes next frame.
306 if (!blending_) tryTransition();
307 return;
308 }
309
310 stateTime_ += dt;
311 const State &st = states_.at(currentState_);
312 st.clip->sample(stateTime_, &pose_, skeleton_);
313 tryTransition();
314}

316eve::Result<void> AnimStateMachine::advance(const eve::SimulationStep &step) {
317 if (step.delta.nanoseconds() < 0)
318 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
319 "animation state-machine delta must be non-negative"));
320 if (hasLastTick_ && step.tick <= lastTick_)
321 return eve::Result<void>::failure(eve::Diagnostic::error(
322 eve::DiagnosticCode::Conflict, "animation state-machine tick must advance monotonically"));
323
324 const float dt = static_cast<float>(step.delta.seconds());
325 if (!std::isfinite(dt))
326 return eve::Result<void>::failure(eve::Diagnostic::error(
327 eve::DiagnosticCode::InvalidArgument, "animation state-machine delta is outside float range"));
328 updateUnchecked(dt);
329 lastTick_ = step.tick;
330 hasLastTick_ = true;
331 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
332}

343 "animation state-machine simulation tick overflow"))

345 return;

351 out = StateValue::object();

372 out.set("triggers", std::move(triggers));

376bool AnimStateMachine::restoreState(const StateValue &in, std::string *err) {
377 if (!in.isObject()) {
378 if (err) *err = "anim: state is not an object";
379 return false;
380 }
381 const StateValue *current = in.find("currentState");
382 const StateValue *started = in.find("started");
383 if (!current || !current->isString() || !started || !started->isBool()) {
384 if (err) *err = "anim: missing currentState/started";
385 return false;
386 }
387 const std::string stateName = current->asString();
388 const bool wasStarted = started->asBool();
389 if (wasStarted && !stateName.empty() && !hasState(stateName)) {
390 if (err) *err = "anim: unknown state '" + stateName + "'";
391 return false;
392 }
393
394 eve::SimulationTick restoredTick = eve::SimulationTick::zero();
395 bool restoredHasTick = false;
396 if (const StateValue *hasTick = in.find("hasSimulationTick"); hasTick && hasTick->isBool())
397 restoredHasTick = hasTick->asBool();
398 if (restoredHasTick) {
399 const StateValue *tick = in.find("simulationTick");
400 if (!tick || !tick->isString()) {
401 if (err) *err = "anim: missing simulationTick";
402 return false;
403 }
404 try {
405 std::size_t parsed = 0;
406 const std::uint64_t value = std::stoull(tick->asString(), &parsed);
407 if (parsed != tick->asString().size()) throw std::invalid_argument("trailing characters");
408 restoredTick = eve::SimulationTick(value);
409 } catch (...) {
410 if (err) *err = "anim: invalid simulationTick";
411 return false;
412 }
413 }
414
415 floats_.clear();
416 bools_.clear();
417 triggers_.clear();
418
419 if (const StateValue *floats = in.find("floats"); floats && floats->isObject()) {
420 for (const auto &key : floats->keys()) {
421 const StateValue *v = floats->find(key);
422 if (v && (v->isInt() || v->isFloat())) floats_[key] = static_cast<float>(stateNumber(*v));
423 }
424 }
425 if (const StateValue *bools = in.find("bools"); bools && bools->isObject()) {
426 for (const auto &key : bools->keys()) {
427 const StateValue *v = bools->find(key);
428 if (v && v->isBool()) bools_[key] = v->asBool();
429 }
430 }
431 if (const StateValue *triggers = in.find("triggers"); triggers && triggers->isObject()) {
432 for (const auto &key : triggers->keys()) {
433 const StateValue *v = triggers->find(key);
434 if (v && v->isBool()) triggers_[key] = v->asBool();
435 }
436 }
437
438 currentState_ = stateName;
439 if (const StateValue *ns = in.find("nextState"); ns && ns->isString()) nextState_ = ns->asString();
440 if (const StateValue *st = in.find("stateTime"); st && (st->isInt() || st->isFloat()))
441 stateTime_ = static_cast<float>(stateNumber(*st));
442 if (const StateValue *bd = in.find("blendDuration"); bd && (bd->isInt() || bd->isFloat()))
443 blendDuration_ = static_cast<float>(stateNumber(*bd));
444 if (const StateValue *be = in.find("blendElapsed"); be && (be->isInt() || be->isFloat()))
445 blendElapsed_ = static_cast<float>(stateNumber(*be));
446 if (const StateValue *bl = in.find("blending"); bl && bl->isBool()) blending_ = bl->asBool();
447 started_ = wasStarted;
448 lastTick_ = restoredTick;
449 hasLastTick_ = restoredHasTick;
450
451 if (started_ && !currentState_.empty()) {
452 states_.at(currentState_).clip->sample(stateTime_, &pose_, skeleton_);
453 }
454 return true;
455}

492eve::Result<void> AnimSyncGroup::advance(const eve::SimulationStep &step) {
493 auto seconds = detail::secondsForStep(step, hasLastTick_, lastTick_, "AnimSyncGroup");
494 if (!seconds) return eve::Result<void>::failure(seconds.status());
495 (void)std::move(seconds).takeValue();
496 usedMarkerSync_ = false;
497 if (entries_.empty()) {
498 lastTick_ = step.tick;
499 hasLastTick_ = true;
500 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
501 }
502 for (const Entry &entry : entries_) {
503 if (entry.player->hasCurrentTick() && step.tick <= entry.player->currentTick())
504 return eve::Result<void>::failure(eve::Diagnostic::error(
505 eve::DiagnosticCode::Conflict, "animation sync-group player already consumed this tick"));
506 }
507 Entry &leader = entries_[static_cast<size_t>(leader_)];
508 auto leaderResult = leader.player->advance(step);
509 if (!leaderResult) return eve::Result<void>::failure(leaderResult.status());
510 AnimClip *leaderClip = leader.player->getClip();
511 if (!leaderClip || leaderClip->getDuration() <= 1e-8f) {
512 lastTick_ = step.tick;
513 hasLastTick_ = true;
514 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
515 }
516 phase_ = leaderClip->wrapTime(leader.player->getTime()) / leaderClip->getDuration();
517 for (int i = 0; i < getCount(); ++i) {
518 if (i == leader_) continue;
519 AnimPlayer *player = entries_[static_cast<size_t>(i)].player;
520 auto playerResult = player->advance(step);
521 if (!playerResult) return eve::Result<void>::failure(playerResult.status());
522 AnimClip *clip = player->getClip();
523 if (!clip || clip->getDuration() <= 1e-8f) continue;
524 float mappedTime;
525 if (leaderClip->hasCompatibleSyncMarkers(clip)) {
526 mappedTime = leaderClip->mapSyncTimeTo(leader.player->getTime(), clip);
527 usedMarkerSync_ = true;
528 } else {
529 mappedTime = phase_ * clip->getDuration();
530 }
531 mappedTime += entries_[static_cast<size_t>(i)].phaseOffset * clip->getDuration();
532 player->setTime(clip->wrapTime(mappedTime));
533 }
534 lastTick_ = step.tick;
535 hasLastTick_ = true;
536 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
537}

642 if (boneIndex < 0 || boneIndex >= pose->getBoneCount()) {

650 pushPoint(px, py, plane == "xy" || plane.empty() ? wz : (plane == "xz" ? wy : wx));

653void AnimTrail::pushPoint(float x, float y, float z) {
654 if (!points_.empty() && minDistance_ > 0.f) {
655 const Point &last = points_.back();
656 const float dx = x - last.x;
657 const float dy = y - last.y;
658 const float dz = z - last.z;
659 if (dx * dx + dy * dy + dz * dz < minDistance_ * minDistance_) return;
660 }
661 Point p;
662 p.x = x;
663 p.y = y;
664 p.z = z;
665 p.age = 0.f;
666 points_.push_back(p);
667 while (static_cast<int>(points_.size()) > capacity_) {
668 points_.erase(points_.begin());
669 }
670}

674void AnimTrail::updateUnchecked(float dt) {
675 if (dt < 0.f) dt = 0.f;
676 for (Point &p : points_) p.age += dt;
677 if (duration_ > 0.f) {
678 auto it = points_.begin();
679 while (it != points_.end()) {
680 if (it->age > duration_) {
681 it = points_.erase(it);
682 } else {
683 ++it;
684 }
685 }
686 }

758void AnimTrail::drawSegment(graphics::Graphics *gfx, float x0, float y0, float x1, float y1, float a0, float a1) const {
759 if (!gfx) return;
760 float sx0 = 0.f, sy0 = 0.f, sx1 = 0.f, sy1 = 0.f;
761 toScreen(x0, y0, &sx0, &sy0);
762 toScreen(x1, y1, &sx1, &sy1);
763 const float dx = sx1 - sx0;
764 const float dy = sy1 - sy0;
765 const float len = std::sqrt(dx * dx + dy * dy);
766 if (len < 1e-4f) {
767 drawDot(gfx, x1, y1, a1);
768 return;
769 }
770 const float step = std::max(1.f, width_ * 0.75f);
771 const int steps = std::max(1, int(len / step));
772 const float half = width_ * 0.5f;
773 for (int i = 0; i <= steps; ++i) {
774 const float t = float(i) / float(steps);
775 const float a = a0 + (a1 - a0) * t;
776 if (a <= 0.f) continue;
777 const float px = sx0 + dx * t;
778 const float py = sy0 + dy * t;
779 gfx->drawSolidRect(px - half, py - half, width_, width_, Color(colorR_, colorG_, colorB_, a));
780 }
781}

783void AnimTrail::draw(graphics::Graphics *gfx) {
784 if (!gfx || points_.empty()) return;
785
786 if (style_ == Style::Points) {
787 for (const Point &p : points_) {
788 drawDot(gfx, p.x, p.y, alphaFor(p));
789 }
790 return;
791 }
792
793 // Line: connect oldest → newest with per-vertex fade.
794 for (size_t i = 1; i < points_.size(); ++i) {
795 const Point &a = points_[i - 1];
796 const Point &b = points_[i];
797 drawSegment(gfx, a.x, a.y, b.x, b.y, alphaFor(a), alphaFor(b));
798 }
799 // Emphasize the head.
800 if (!points_.empty()) {
801 const Point &head = points_.back();
802 drawDot(gfx, head.x, head.y, alphaFor(head));
803 }
804}
