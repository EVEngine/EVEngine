39Result<void> CombatActionWindowState::enter(const action::ActionStateWindowBinding& binding,
40 const action::ActionTimelineEvent& event,
41 const action::ActionNotifyContext& context) {
42 if (!supports(binding.kind))
43 return failure(DiagnosticCode::Unsupported, "combat does not own this state-window kind", "kind");
44 if (!resolver_) return failure(DiagnosticCode::NotFound, "window subject resolver is unavailable", "resolver");
45 std::optional<ecs::EntityHandle> handle;
46 if (binding.targetIndex) {
47 if (*binding.targetIndex >= context.targets.size())
48 return failure(DiagnosticCode::NotFound, "state-window target index is unavailable", "targetIndex");
49 handle = context.targets[*binding.targetIndex];
50 } else {
51 handle = context.source;
52 }
53 if (!handle) return failure(DiagnosticCode::NotFound, "state window has no source or target", "subject");
54 auto subject = resolver_(*handle);
55 if (!subject) return Result<void>::failure(subject.status());
56 if (!subject.value().isValid())
57 return failure(DiagnosticCode::InvalidArgument, "state-window subject is nil", "subject");
58
59 ActiveKey key{context.executionId, event.itemId.format()};
60 const auto found = active_.find(key);
61 if (found != active_.end()) {
62 const bool same = found->second.subject == subject.value() &&
63 found->second.binding.kind == binding.kind &&
64 found->second.binding.resource == binding.resource;
65 if (same) return Result<void>::success(Status::success(StatusCode::NoOp));
66 return failure(DiagnosticCode::Conflict, "state-window key is already active with different data", "itemId");
67 }
68 active_.emplace(std::move(key), ActiveWindow{binding, subject.value()});
69 return Result<void>::success(Status::success(StatusCode::Applied));
70}

72Result<void> CombatActionWindowState::exit(const action::ActionStateWindowBinding& binding,
73 const action::ActionTimelineEvent& event,
74 const action::ActionNotifyContext& context) {
75 if (!supports(binding.kind))
76 return failure(DiagnosticCode::Unsupported, "combat does not own this state-window kind", "kind");
77 const ActiveKey key{context.executionId, event.itemId.format()};
78 const auto found = active_.find(key);
79 if (found == active_.end()) return Result<void>::success(Status::success(StatusCode::NoOp));
80 if (found->second.binding.kind != binding.kind)
81 return failure(DiagnosticCode::Conflict, "state-window exit kind does not match active key", "kind");
82 active_.erase(found);
83 return Result<void>::success(Status::success(StatusCode::Applied));
84}
