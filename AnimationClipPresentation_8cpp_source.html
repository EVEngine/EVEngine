79editor::Result<void> AnimationClipPresentation::write(const editor::PropertyPath& path,
80 const editor::EditorValue& value) {
81 const auto descriptor = schema().find(path);
82 if (!descriptor || editor::hasPropertyFlag(descriptor->flags, editor::PropertyFlag::ReadOnly))
83 return editing::failed<void>(editor::EditorStatus::Rejected, editor::RuleId("editor.animation.property"),
84 "Property is unavailable");
85 auto valid = editor::validatePropertyValue(*descriptor, value);
86 if (!valid.ok()) return valid;
87 const auto& p = path.value();
88 if (p == "loop") return source_.setLoop(*value.getIf<bool>());
89 const double v = value.getIf<double>() ? *value.getIf<double>() : static_cast<double>(*value.getIf<std::int64_t>());
90 if (p == "duration") return source_.setDuration(v);
91 if (p == "sample-rate") return source_.setSampleRate(v);
92 if (p == "mask") return source_.setMaskWeight(v);
93 if (p == "key-time") return source_.moveSelectedKey(v);
94 if (p.starts_with("position-"))
95 return source_.setSelectedPosition(p == "position-x" ? v : source_.selectedPositionX(),
96 p == "position-y" ? v : source_.selectedPositionY(),
97 p == "position-z" ? v : source_.selectedPositionZ());
98 if (p.starts_with("rotation-"))
99 return source_.setSelectedRotation(p == "rotation-x" ? v : source_.selectedRotationX(),
100 p == "rotation-y" ? v : source_.selectedRotationY(),
101 p == "rotation-z" ? v : source_.selectedRotationZ());
102 if (p.starts_with("scale-"))
103 return source_.setSelectedScale(p == "scale-x" ? v : source_.selectedScaleX(),
104 p == "scale-y" ? v : source_.selectedScaleY(),
105 p == "scale-z" ? v : source_.selectedScaleZ());
106 return editing::failed<void>(editor::EditorStatus::NotFound, editor::RuleId("editor.animation.property"),
107 "Unknown property");
108}
109editor::Result<void> AnimationClipPresentation::action(std::string_view id) {
110 if (id == "undo" || id == "redo") {
111 auto result = id == "undo" ? source_.undo() : source_.redo();
112 if (!result.ok()) return editor::Result<void>::failure(result.status());
113 return editing::applied<void>();
114 }
115 if (id == "set-key") return source_.keySelectedBone();
116 if (id == "delete-key") return source_.deleteSelectedKey();
117 if (id == "play-pause") {

121 source_.play();

123 source_.stop();
124 else

126 "Unknown animation action");
