87eve::Result<ProfilerFrameSnapshot> Profiler::captureFrame() const {
88 if (!hasFrame()) {
89 return eve::Result<ProfilerFrameSnapshot>::failure(eve::Diagnostic::error(
90 eve::DiagnosticCode::NotFound, "No completed profiler frame is available",
91 "profiler.frame", {}, "profiler.capture-frame"));
92 }
93 ProfilerFrameSnapshot snapshot;
94 snapshot.sequence = captureSequence_;
95 snapshot.cpuFrameMs = frameMs_;
96 if (auto* timer = eve::cap::query<eve::service::IGpuTimer>()) {
97 snapshot.gpuTimingAvailable = timer->gpuTimingAvailable();
98 if (snapshot.gpuTimingAvailable) snapshot.gpuFrameMs = timer->gpuFrameMs();
99 }
100 const auto& samples = eve::prof::Profiler::lastFrame();
101 snapshot.zones.reserve(samples.size());
102 for (const auto& sample : samples) {
103 snapshot.zones.push_back({sample.module, sample.name, sample.thread, sample.selfMs,
104 sample.totalMs, sample.count, sample.minDepth});
105 }
106 return eve::Result<ProfilerFrameSnapshot>::success(std::move(snapshot));
107}
