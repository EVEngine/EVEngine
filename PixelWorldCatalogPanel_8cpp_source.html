47 Integer value{};
48 const auto parsed = std::from_chars(text.data(), text.data() + text.size(), value);

53std::optional<std::uint32_t> parseRgba(std::string_view text) {
54 if (!text.empty() && text.front() == '#') text.remove_prefix(1);
55 if (text.size() != 8) return std::nullopt;
56 std::uint32_t value = 0;
57 const auto parsed = std::from_chars(text.data(), text.data() + text.size(), value, 16);
58 if (parsed.ec != std::errc{} || parsed.ptr != text.data() + text.size()) return std::nullopt;
59 return value;
60}

68std::vector<std::string> splitTags(std::string_view value) {
69 std::vector<std::string> result;
70 while (!value.empty()) {
71 const auto comma = value.find(',');
72 std::string tag(value.substr(0, comma));
73 const auto first = tag.find_first_not_of(" \t");
74 const auto last = tag.find_last_not_of(" \t");
75 if (first != std::string::npos) result.push_back(tag.substr(first, last - first + 1));
76 if (comma == std::string_view::npos) break;
77 value.remove_prefix(comma + 1);
78 }
79 std::sort(result.begin(), result.end());
80 result.erase(std::unique(result.begin(), result.end()), result.end());
81 return result;
82}

84std::string joinTags(const std::vector<std::string>& tags) {
85 std::string result;
86 for (const auto& tag : tags) {
87 if (!result.empty()) result += ", ";
88 result += tag;
89 }
90 return result;
91}

196eve::Result<CatalogDraftReceipt> PixelWorldCatalogPanel::commitCandidate(
197 std::vector<MaterialDefinition> candidateDefinitions,
198 std::vector<MaterialReactionRule> candidateReactions,
199 std::vector<MaterialPhaseRule> candidatePhases) {
200 auto candidate = MaterialCatalog::create(std::move(candidateDefinitions),
201 std::move(candidateReactions),
202 std::move(candidatePhases));
203 if (!candidate.ok())
204 return eve::Result<CatalogDraftReceipt>::failure(candidate.status());
205 CatalogDraftReceipt receipt{revision_, revision_ + 1, candidate.value().fingerprint()};
206 draft_ = std::move(candidate).takeValue();
207 revision_ = receipt.revisionAfter;
208 if (selectedMaterial_ >= draft_.definitions().size()) selectedMaterial_ = 0;
209 if (draft_.reactions().empty()) selectedReaction_.reset();
210 else if (!selectedReaction_ || *selectedReaction_ >= draft_.reactions().size())
211 selectedReaction_ = draft_.reactions().size() - 1;
212 if (draft_.phaseRules().empty()) selectedPhase_.reset();
213 else if (!selectedPhase_ || *selectedPhase_ >= draft_.phaseRules().size())
214 selectedPhase_ = draft_.phaseRules().size() - 1;
215 return eve::Result<CatalogDraftReceipt>::success(receipt);
216}

218eve::Result<CatalogDraftReceipt> PixelWorldCatalogPanel::replaceMaterial(
219 MaterialDefinition definition) {
220 const auto index = static_cast<std::size_t>(definition.id);
221 if (index >= draft_.definitions().size())
222 return fail<CatalogDraftReceipt>(eve::DiagnosticCode::InvalidArgument,
223 "material id is outside the Catalog", "material.id");
224 auto candidateDefinitions = definitions(draft_);
225 candidateDefinitions[index] = std::move(definition);
226 return commitCandidate(std::move(candidateDefinitions), reactions(draft_), phases(draft_));
227}

239eve::Result<CatalogDraftReceipt> PixelWorldCatalogPanel::addReaction(
240 MaterialReactionRule rule) {
241 const std::string selectedId = rule.id;
242 auto candidate = reactions(draft_);
243 candidate.push_back(std::move(rule));
244 auto committed = commitCandidate(definitions(draft_), std::move(candidate), phases(draft_));
245 if (committed.ok()) {
246 for (std::size_t index = 0; index < draft_.reactions().size(); ++index)
247 if (draft_.reactions()[index].id == selectedId) selectedReaction_ = index;
248 }
249 return committed;
250}
