62inline float silPomCoverageSoft(float displacedU, float displacedV, float feather) {
63 const float f = std::max(feather, 1e-4f);
64 auto edge = [f](float t) {
65 if (t <= 0.f) return 0.f;
66 if (t >= f) return 1.f;
67 const float x = t / f;
68 return x * x * (3.f - 2.f * x); // smoothstep
69 };
70 const float cx = std::min(edge(displacedU), edge(1.f - displacedU));
71 const float cy = std::min(edge(displacedV), edge(1.f - displacedV));
72 return cx * cy;
73}

104inline void ssdmScreenOffset(float height01, float viewNdcX, float viewNdcY, float scale,
105 float &outX, float &outY) {
106 if (scale < 1e-5f) {
107 outX = 0.f;
108 outY = 0.f;
109 return;
110 }
111 // Raised texels push toward the viewer in screen space (opposite to POM's UV dig).
112 const float h = std::clamp(height01, 0.f, 1.f);
113 const float len = std::sqrt(viewNdcX * viewNdcX + viewNdcY * viewNdcY);
114 if (len < 1e-5f) {
115 outX = 0.f;
116 outY = 0.f;
117 return;
118 }
119 outX = (viewNdcX / len) * h * scale;
120 outY = (viewNdcY / len) * h * scale;
121}
