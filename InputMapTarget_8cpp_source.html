53bool InputMapTarget::matches(const SelectionSnapshot& s) const {

92PropertyReadResult InputMapTarget::read(const SelectionSnapshot& s, const PropertyPath& p) const {
93 if (!matches(s) || !schema(s).find(p)) return {};
94 std::optional<EditorValue> common;
95 for (const auto& i : s.items) {
96 EditorValue v;
97 if (i.type == "input.action") {
98 const auto& a = *std::find_if(actions_.begin(), actions_.end(),
99 [&](const auto& x) { return x.id.value() == i.item.value(); });
100 v = p == PropertyPath("action.name") ? EditorValue(a.name) : EditorValue(a.kind);
101 } else {
102 const auto& b = *std::find_if(bindings_.begin(), bindings_.end(),
103 [&](const auto& x) { return x.id.value() == i.item.value(); });
104 if (p == PropertyPath("binding.action"))
105 v = b.action.value();
106 else if (p == PropertyPath("binding.device"))
107 v = b.device;
108 else if (p == PropertyPath("binding.control"))
109 v = b.control;
110 else if (p == PropertyPath("binding.scale"))
111 v = b.scale;
112 else if (p == PropertyPath("binding.deadzone"))
113 v = b.deadzone;
114 else
115 v = b.invert;
116 }
117 if (!common)
118 common = v;
119 else if (*common != v)
120 return {PropertyReadState::Mixed, {}, {}};
121 }
122 return {PropertyReadState::Value, *common, {}};

216std::vector<EditorDiagnostic> InputMapTarget::validate() const {
217 std::vector<EditorDiagnostic> d;
218 std::set<std::string> ids, names, controls;
219 for (const auto& a : actions_) {
220 if (a.id.empty() || a.name.empty() || !ids.insert(a.id.value()).second || !names.insert(a.name).second ||
221 (a.kind != "button" && a.kind != "axis1d"))
222 d.push_back(eve::editing::ruleDiagnostic(
223 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.input.action"),
224 DiagnosticSeverity::Error, "Action IDs/names must be unique and kinds supported"));
225 }
226 for (const auto& b : bindings_) {
227 const bool target =
228 std::any_of(actions_.begin(), actions_.end(), [&](const auto& a) { return a.id == b.action; });
229 if (b.id.empty() || !ids.insert(b.id.value()).second || !target || b.control.empty() ||
230 !std::isfinite(b.scale) || b.scale == 0 || std::abs(b.scale) > 100 || !std::isfinite(b.deadzone) ||
231 b.deadzone < 0 || b.deadzone >= 1)
232 d.push_back(eve::editing::ruleDiagnostic(
233 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.input.binding"),
234 DiagnosticSeverity::Error, "Binding identity, reference or response is invalid"));
235 const std::string key = b.device + ":" + b.control;
236 if (!controls.insert(key).second)
237 d.push_back(eve::editing::ruleDiagnostic(
238 eve::DiagnosticCode::Conflict, RuleId("editor.input.conflict"),
239 DiagnosticSeverity::Warning, "Control is bound more than once: " + key));
240 }
241 if (actions_.size() > 1024 || bindings_.size() > 8192)
242 d.push_back(eve::editing::ruleDiagnostic(
243 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.input.budget"),
244 DiagnosticSeverity::Error, "Input map exceeds action or binding budget"));
245 return d;
246}

314EditorResult<std::map<std::string, double>> InputMapEvaluator::evaluate(
315 const InputMapTarget& t, const std::vector<InputControlSample>& samples) const {
316 if (errors(t.validate()))
317 return fail<std::map<std::string, double>>(EditorStatus::Rejected, "editor.input.invalid",
318 "Cannot evaluate invalid input map");
319 std::map<std::string, double> out;
320 for (const auto& a : t.actions()) out[a.name] = 0;
321 for (const auto& b : t.bindings()) {
322 auto s = std::find_if(samples.begin(), samples.end(),
323 [&](const auto& x) { return x.device == b.device && x.control == b.control; });
324 if (s == samples.end()) continue;
325 double value = std::clamp(s->value, -1.0, 1.0);
326 if (std::abs(value) <= b.deadzone)
327 value = 0;
328 else
329 value = std::copysign((std::abs(value) - b.deadzone) / (1 - b.deadzone), value);
330 if (b.invert) value = -value;
331 value *= b.scale;
332 auto a = std::find_if(t.actions().begin(), t.actions().end(), [&](const auto& x) { return x.id == b.action; });
333 if (a->kind == "button")
334 out[a->name] = std::max(out[a->name], std::abs(value) >= .5 ? 1.0 : 0.0);
335 else
336 out[a->name] = std::clamp(out[a->name] + value, -1.0, 1.0);
337 }
338 return eve::editing::applied<std::map<std::string, double>>(std::move(out));
339}
