28RPGActor *RPGActor::createActor() {
29 RPGActor *actor = RPGActor::create();
30 // Touch each component once so accessors are cheap/valid immediately.
31 actor->attributes();
32 actor->statuses();
33 actor->skills();
34 actor->progression();
35 actor->vitals();
36 actor->traits();
37 actor->classInfo();
38 registryStorage().push_back(ecs::handle_of(actor));
39 return actor;
40}

42const std::vector<RPGActor *> &RPGActor::liveActors() {
43 static std::vector<RPGActor *> resolved;
44 auto &handles = registryStorage();
45
46 handles.erase(std::remove_if(handles.begin(), handles.end(),
47 [](const ecs::EntityHandle &h) { return ecs::try_get(h) == nullptr; }),
48 handles.end());
49
50 resolved.clear();
51 resolved.reserve(handles.size());
52 for (const ecs::EntityHandle &h : handles)
53 resolved.push_back(static_cast<RPGActor *>(ecs::try_get(h)));
54 return resolved;
55}
