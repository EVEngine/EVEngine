26 HostFeature::BuildCook | HostFeature::SourceControl | HostFeature::DebugOverride;
27 if ((manifest.features & forbidden) != HostFeature::None)
28 return manifestError("editor.runtime.forbidden-feature",
29 "Runtime editor manifest requests a developer-only feature");
30
31 RuntimeEditorPackage package;
32 package.profile = HostProfile::runtimeBuilder();
33 package.profile.setFeatures(manifest.features);
34 package.profile.setMaxPayloadBytes(manifest.maxPayloadBytes);
35 for (const CommandId& id : manifest.commands) {
36 const CommandDescriptor* descriptor = commands.find(id);
37 if (!descriptor)
38 return manifestError("editor.runtime.command-not-found",
39 "Manifest command is not registered: " + id.value());
40 if (!descriptor->automationAllowed)
41 return manifestError("editor.runtime.command-not-publishable",
42 "Manifest command is not publishable: " + id.value());

45 "Manifest omits a feature required by command: " + id.value());
46 package.profile.allowCommand(id);
47 package.commands.push_back(*descriptor);
48 }
49
50 HostProfile audienceProfile = HostProfile::runtimeBuilder();
51 audienceProfile.setAllowAllCommands(true);
52 for (const auto& descriptor : extensions.tools(audienceProfile))
53 if (selected(manifest.tools, descriptor.id)) package.tools.push_back(descriptor);
54 for (const ToolId& id : manifest.tools)
55 if (std::none_of(package.tools.begin(), package.tools.end(), [&](const auto& value) { return value.id == id; }))
56 return manifestError("editor.runtime.tool-not-found", "Manifest tool is unavailable: " + id.value());
57 for (const auto& descriptor : extensions.palettes(audienceProfile))
58 if (selected(manifest.palettes, descriptor.id)) package.palettes.push_back(descriptor);

61 [&](const auto& value) { return value.id == id; }))

63 for (const auto& descriptor : extensions.rules(audienceProfile))
64 if (selected(manifest.rules, descriptor.id)) package.rules.push_back(descriptor);
65 for (const RuleId& id : manifest.rules)
66 if (std::none_of(package.rules.begin(), package.rules.end(), [&](const auto& value) { return value.id == id; }))
67 return manifestError("editor.runtime.rule-not-found", "Manifest rule is unavailable: " + id.value());
68
69 std::deque<AssetGuid> pending(manifest.rootAssets.begin(), manifest.rootAssets.end());
70 std::unordered_set<AssetGuid, StrongEditorIdHash<AssetGuid>> visited;
71 while (!pending.empty()) {
72 AssetGuid current = pending.front();
73 pending.pop_front();
74 if (!visited.emplace(current).second) continue;
75 if (!assets.find(current).ok())
76 return manifestError("editor.runtime.asset-not-found", "Manifest asset is unavailable: " + current.value());

78 for (const AssetDependency& dependency : assets.dependencies(current))
79 if (dependency.kind != DependencyKind::EditorOnly && dependency.kind != DependencyKind::Source)
80 pending.push_back(dependency.to);
81 }
82 std::sort(package.assetClosure.begin(), package.assetClosure.end());
83 return eve::editing::applied<RuntimeEditorPackage>(std::move(package));
