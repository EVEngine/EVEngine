18EditorResult<void> AvatarDocumentRuntime::publish(const AvatarDocumentTarget& document,
19 const IAvatarTextureResolver* textures) {
20 const auto diagnostics = document.validate();
21 for (const auto& d : diagnostics)
22 if (d.severity() == DiagnosticSeverity::Error)
23 return EditorResult<void>::failure(eve::Status(EditorStatus::Rejected, diagnostics));
24 std::vector<graphics::Texture*> resolved;
25 resolved.reserve(document.layers().size());
26 for (const auto& layer : document.layers()) {
27 if (!textures)
28 return fail<void>(EditorStatus::Rejected, "editor.avatar.textures",
29 "Image layers require an Avatar texture resolver");
30 auto texture = textures->texture(layer.textureAsset);
31 if (!texture.ok()) return EditorResult<void>::failure(texture.status());
32 if (!texture.value())
33 return fail<void>(EditorStatus::NotFound, "editor.avatar.texture",
34 "Avatar texture resolver returned no texture");
35 resolved.push_back(texture.value());
36 }
37 auto candidate = std::make_unique<avatar::AvatarInstance>(document.kind());
38 if (document.kind() == "live2d" && !candidate->loadLive2DModel(document.sourceAsset()))
39 return fail<void>(EditorStatus::Failed, "editor.avatar.live2d", "Live2D backend rejected the Avatar model");
40 if (document.kind() == "vroid" && !candidate->loadVroidModelPath(document.sourceAsset()))
41 return fail<void>(EditorStatus::Failed, "editor.avatar.vroid", "VRoid backend rejected the Avatar model path");
42 for (std::size_t i = 0; i < document.layers().size(); ++i) {
43 const auto& layer = document.layers()[i];
44 if (!candidate->addLayer(layer.name, resolved[i], layer.zIndex) ||
45 !candidate->setLayerVisible(layer.name, layer.visible) ||
46 !candidate->setLayerOffset(layer.name, layer.offset[0], layer.offset[1]) ||
47 !candidate->setLayerSize(layer.name, layer.size[0], layer.size[1]) ||
48 !candidate->setLayerColor(layer.name, layer.color[0], layer.color[1], layer.color[2], layer.color[3]))
49 return fail<void>(EditorStatus::Failed, "editor.avatar.layer",
50 "Avatar runtime rejected a validated image layer");
51 }
52 for (const auto& parameter : document.parameters()) {
53 if (!candidate->defineParameter(parameter.name, parameter.defaultValue, parameter.minimum, parameter.maximum))
54 return fail<void>(EditorStatus::Failed, "editor.avatar.parameter",
55 "Avatar runtime rejected validated parameter metadata");
56 candidate->setParameter(parameter.name, parameter.value);
57 }
58 for (const auto& expression : document.expressions()) {
59 std::ostringstream spec;
60 bool first = true;
61 for (const auto& [name, value] : expression.channels) {
62 if (!first) spec << ';';
63 first = false;
64 spec << name << '=' << value;
65 }
66 if (!candidate->defineExpression(expression.name, spec.str()))
67 return fail<void>(EditorStatus::Failed, "editor.avatar.expression",
68 "Avatar runtime rejected a validated expression");
69 }
70 instance_ = std::move(candidate);
71 revision_ = document.revision();
72 return eve::editing::applied<void>(diagnostics);
73}
