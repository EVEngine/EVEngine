19void VehicleOrderQueueAdapter::pruneTerminalPayloads() {
20 for (auto it = impl_->payloads.begin(); it != impl_->payloads.end();) {
21 auto recordRef = impl_->queue.find(it->first);
22 const orders::Order* record = recordRef ? &recordRef->get() : nullptr;
23 if (record == nullptr ||
24 (record->state != orders::OrderState::Queued && record->state != orders::OrderState::Active)) {

137void VehicleOrderQueueAdapter::syncCompatibility(VehicleEntity::Orders& legacy) const {
138 legacy.queue.clear();
139 legacy.current = -1;
140 for (int i = 0; i < impl_->queue.orderCount(); ++i) {
141 auto recordRef = impl_->queue.orderAt(i);
142 const orders::Order* record = recordRef ? &recordRef->get() : nullptr;
143 if (record == nullptr ||
144 (record->state != orders::OrderState::Queued && record->state != orders::OrderState::Active)) {
145 continue;
146 }
147 const auto it = impl_->payloads.find(record->id);
148 if (it == impl_->payloads.end()) continue;
149 if (record->state == orders::OrderState::Active) legacy.current = static_cast<int>(legacy.queue.size());
150 legacy.queue.push_back(it->second);
151 }
152}
