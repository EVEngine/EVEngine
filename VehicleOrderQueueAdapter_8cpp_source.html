25void VehicleOrderQueueAdapter::pruneTerminalPayloads() {
26 for (auto it = impl_->payloads.begin(); it != impl_->payloads.end();) {
27 auto recordRef = impl_->queue.find(it->first);
28 const orders::Order* record = recordRef ? &recordRef->get() : nullptr;
29 if (record == nullptr ||
30 (record->state != orders::OrderState::Queued && record->state != orders::OrderState::Active)) {
31 it = impl_->payloads.erase(it);
32 } else {
33 ++it;
34 }
35 }
36}

141void VehicleOrderQueueAdapter::syncCompatibility(VehicleEntity::Orders& legacy) const {
142 legacy.queue.clear();
143 legacy.current = -1;
144 for (int i = 0; i < impl_->queue.orderCount(); ++i) {
145 auto recordRef = impl_->queue.orderAt(i);
146 const orders::Order* record = recordRef ? &recordRef->get() : nullptr;
147 if (record == nullptr ||
148 (record->state != orders::OrderState::Queued && record->state != orders::OrderState::Active)) {
149 continue;
150 }
151 const auto it = impl_->payloads.find(record->id);
152 if (it == impl_->payloads.end()) continue;
153 if (record->state == orders::OrderState::Active) legacy.current = static_cast<int>(legacy.queue.size());
154 legacy.queue.push_back(it->second);
155 }
156}
