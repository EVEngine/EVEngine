157bool UIHost::moveFocus(FocusDirection direction, bool wrap) {
158 auto t = tree();
159
160 UINode *current = nullptr;
161 for (UINode &node : t->nodes) {
162 if (node.focused) {
163 current = &node;
164 break;
165 }
166 if (!current && node.focusRequested) current = &node;
167 }
168
169 if (current) {
170 const std::string &explicitId = neighbor(*current, direction);
171 if (!explicitId.empty()) {
172 auto explicitTarget = findById(explicitId);
173 if (explicitTarget && eligible(explicitTarget->get(), true)) return requestFocusById(explicitId);
174 }
175 }
176
177 std::vector<std::pair<int, std::size_t>> order;
178 order.reserve(t->nodes.size());
179 for (std::size_t index = 0; index < t->nodes.size(); ++index) {
180 const UINode &node = t->nodes[index];
181 if (eligible(node, true) && node.tabIndex >= 0)
182 order.emplace_back(node.tabIndex, index);
183 }
184 if (order.empty()) return false;
185 std::stable_sort(order.begin(), order.end());
186
187 const bool backwards = direction == FocusDirection::Previous ||
188 direction == FocusDirection::Left ||
189 direction == FocusDirection::Up;
190 std::size_t targetOrder = backwards ? order.size() - 1 : 0;
191 if (current) {
192 const auto found = std::find_if(order.begin(), order.end(),
193 [current, &t](const auto &entry) {
194 return &t->nodes[entry.second] == current;
195 });
196 if (found != order.end()) {
197 const std::size_t position = static_cast<std::size_t>(found - order.begin());
198 if (backwards) {
199 if (position == 0 && !wrap) return false;
200 targetOrder = position == 0 ? order.size() - 1 : position - 1;
201 } else {
202 if (position + 1 == order.size() && !wrap) return false;
203 targetOrder = position + 1 == order.size() ? 0 : position + 1;
204 }
205 }
206 }
207 return requestFocusById(t->nodes[order[targetOrder].second].id);
208}
