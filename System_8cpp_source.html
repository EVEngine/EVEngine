60Result<void> System::applyPhotoModeField(const PhotoModeAssignment& assignment) {
61 const auto* value = std::get_if<int64_t>(&assignment.value);
62 const auto fail = [&](const char* message, DiagnosticCode code = DiagnosticCode::InvalidArgument) {
63 return Result<void>::failure(
64 Diagnostic::error(code, message, assignment.field, {}, "system.pcgPhotoMode"));
65 };
66 if (!value) return fail("system photo-mode field requires an integer");
67
68 if (assignment.field == "m_vSync") {
69 if (*value < 0 || *value > 2) return fail("VSync count must be in [0,2]");
70 auto* presentation = cap::query<IFramePresentation>();
71 if (!presentation)
72 return fail("frame presentation capability is unavailable", DiagnosticCode::Unsupported);
73 presentation->setVSyncCount(static_cast<int>(*value));
74 photoModeVSync_ = static_cast<int>(*value);
75 lastFrameCounter_ = 0;
76 } else if (assignment.field == "m_targetFPS") {
77 if (*value < -1 || *value > 240) return fail("target FPS must be in [-1,240]");
78 photoModeTargetFPS_ = static_cast<int>(*value);
79 lastFrameCounter_ = 0;
80 } else {
81 return fail("unsupported system photo-mode field");
82 }
83 return Result<void>::success();
84}

86void System::limitFrame() {
87 const uint64_t now = SDL_GetPerformanceCounter();
88 const uint64_t frequency = SDL_GetPerformanceFrequency();
89 if (lastFrameCounter_ == 0 || frequency == 0) {
90 lastFrameCounter_ = now;
91 return;
92 }
93 if (photoModeVSync_ == 0 && photoModeTargetFPS_ > 0) {
94 const double target = static_cast<double>(frequency) / static_cast<double>(photoModeTargetFPS_);
95 const double elapsed = static_cast<double>(now - lastFrameCounter_);
96 if (elapsed < target) {
97 const uint32_t milliseconds =
98 static_cast<uint32_t>((target - elapsed) * 1000.0 / static_cast<double>(frequency));
99 if (milliseconds > 0) SDL_Delay(milliseconds);
100 }
101 }
102 lastFrameCounter_ = SDL_GetPerformanceCounter();
103}

143int System::getProcessMemoryMB() const {
144#if defined(EVENGINE_WINDOWS) || defined(_WIN32)
145 PROCESS_MEMORY_COUNTERS pmc{};
146 if (GetProcessMemoryInfo(GetCurrentProcess(), &pmc, sizeof(pmc)))
147 return static_cast<int>(pmc.WorkingSetSize / (1024ull * 1024ull));
148 return 0;
149#elif defined(__APPLE__)
150 task_vm_info_data_t info{};
151 mach_msg_type_number_t count = TASK_VM_INFO_COUNT;
152 if (task_info(mach_task_self(), TASK_VM_INFO, reinterpret_cast<task_info_t>(&info), &count) ==
153 KERN_SUCCESS) {
154 return static_cast<int>(info.phys_footprint / (1024ull * 1024ull));
155 }
156 return 0;
157#elif defined(__linux__) || defined(EVENGINE_ANDROID) || defined(EVENGINE_LINUX)
158 std::ifstream in("/proc/self/status");
159 if (!in) return 0;
160 std::string key;
161 while (in >> key) {
162 if (key == "VmRSS:") {
163 long kb = 0;
164 in >> kb;
165 return static_cast<int>(kb / 1024);
166 }
167 std::string rest;
168 std::getline(in, rest);
169 }
170 return 0;
171#else
172 return 0;
173#endif
174}
