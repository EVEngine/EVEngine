69voxel_editing::EditorResult<void> VoxelCatalogEditor::configureWorkspace(editor::EditorWorkspace& workspace) const {
70 editor::EditorWorkspace candidate = workspace;
71 struct Panel {
72 const char* id;
73 const char* title;
74 const char* region;
75 const char* context;
76 int order;
77 };
78 constexpr Panel panels[] = {

81 {"voxel.tools", "Tools", "right", "inspector", 100},
82 {"voxel.inspector", "Model", "bottom", "inspector", 100},
83 };
84 for (const auto& panel : panels) {
85 if (!candidate.registerPanel(panel.id, panel.title, panel.region, panel.order) ||
86 !candidate.setPanelCapability(panel.id, "voxel.sculpt") ||

129voxel_editing::EditorResult<void> VoxelCatalogEditor::commit(
130 voxel_editing::EditorResult<voxel_editing::DomainOperation> operation, std::string label) {
131 if (!operation.ok()) return voxel_editing::EditorResult<void>::failure(operation.status());
132 editor::TransactionSpec spec;
133 spec.id = editor::TransactionId("voxel.sculpt.tx." + std::to_string(++txSequence_));
134 spec.label = std::move(label);
135 spec.target = editor::TargetId(target_.targetId());
136 spec.baseRevision = target_.revision();
137 auto begun = transactions_.begin(std::move(spec));
138 if (!begun.ok())
139 return editorError(begun.code(), "editor.voxel.begin", "Could not begin the voxel transaction");
140 auto appended = transactions_.append(std::move(operation).takeValue());
141 if (!appended.ok()) {
142 auto discarded = transactions_.discard();
143 if (!discarded.ok()) discarded.ignore("pending voxel transaction already inactive");
144 return voxel_editing::EditorResult<void>::failure(appended.status());
145 }
146 auto committed = transactions_.commit();
147 if (!committed.ok()) return voxel_editing::EditorResult<void>::failure(committed.status());
148 return refreshPreview();
149}

151void VoxelCatalogEditor::cameraAxes(float& fx, float& fy, float& fz, float& rx, float& ry, float& rz, float& ux,
152 float& uy, float& uz) const {
153 const float cy = std::cos(yaw_);
154 const float sy = std::sin(yaw_);
155 const float cp = std::cos(pitch_);
156 const float sp = std::sin(pitch_);
157 fx = sy * cp;
158 fy = sp;
159 fz = cy * cp;
160 rx = cy;
161 ry = 0.0f;
162 rz = -sy;
163 ux = -sy * sp;
164 uy = cp;
165 uz = -cy * sp;
166}

168void VoxelCatalogEditor::rebuildScreen() {
169 screen_.clear();
170 const auto* model = selectedModel();
171 if (!model) return;
172 float fx, fy, fz, rx, ry, rz, ux, uy, uz;
173 cameraAxes(fx, fy, fz, rx, ry, rz, ux, uy, uz);
174 const float cx = static_cast<float>(model->sizeX) * 0.5f;
175 const float cy = static_cast<float>(model->sizeY) * 0.5f;
176 const float cz = static_cast<float>(model->sizeZ) * 0.5f;
177 const float extent =
178 std::max(1.0f, static_cast<float>(std::max(model->sizeX, std::max(model->sizeY, model->sizeZ))));
179 const float scale = std::min(viewportW_, viewportH_) / (extent * 1.8f);
180 for (const auto& voxel : model->voxels) {
181 const float px = static_cast<float>(voxel.x) + 0.5f - cx;
182 const float py = static_cast<float>(voxel.y) + 0.5f - cy;
183 const float pz = static_cast<float>(voxel.z) + 0.5f - cz;
184 ScreenVoxel box;
185 box.depth = px * fx + py * fy + pz * fz;
186 box.w = scale * 0.9f;
187 box.h = scale * 0.9f;
188 box.x = viewportW_ * 0.5f + (px * rx + py * ry + pz * rz) * scale - box.w * 0.5f;
189 box.y = viewportH_ * 0.5f - (px * ux + py * uy + pz * uz) * scale - box.h * 0.5f;
190 screen_.push_back(box);
191 }
192 std::sort(screen_.begin(), screen_.end(), [](const ScreenVoxel& a, const ScreenVoxel& b) { return a.depth > b.depth; });
193}

272voxel_editing::EditorResult<void> VoxelCatalogEditor::pointerDown(float x, float y) {
273 const auto* model = selectedModel();
274 if (!model)
275 return editorError(voxel_editing::EditorStatus::Rejected, "editor.voxel.selection",
276 "A model must be selected to sculpt");
277 if (!std::isfinite(x) || !std::isfinite(y))
278 return editorError(voxel_editing::EditorStatus::Rejected, "editor.voxel.pointer",
279 "Pointer coordinates must be finite");
280 float fx, fy, fz, rx, ry, rz, ux, uy, uz;
281 cameraAxes(fx, fy, fz, rx, ry, rz, ux, uy, uz);
282 const float cx = static_cast<float>(model->sizeX) * 0.5f;
