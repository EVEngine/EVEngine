514 const auto document = eve::json::Document::parse(text);
515 const auto root = document.root();
516 if (!root.isObject() || root.getInt("version", 0) != 1) return false;
517 std::vector<Rig> newRigs;
518 const auto rigs = root.get("rigs");
519 if (rigs && !rigs.isArray()) return false;
520 for (size_t i = 0; i < rigs.size(); ++i) {
521 const auto item = rigs.at(i);
522 if (!item.isObject()) return false;
523 Rig rig;
524 rig.name = item.getString("name");
525 rig.mode = item.getString("mode");
526 if (rig.name.empty() || !validMode(rig.mode)) return false;
527 rig.priority = item.getInt("priority");
528 rig.enabled = item.getBool("enabled", true);
529 rig.target = jsonVec3(item.get("target"), rig.target);
530 rig.offset = jsonVec3(item.get("offset"), rig.offset);
531 rig.lookAhead = jsonVec3(item.get("lookAhead"), rig.lookAhead);
532 const auto composition = item.get("composition");
533 if (composition.isArray() && composition.size() == 2)
534 rig.composition = {composition.at(0).asFloat(), composition.at(1).asFloat()};
535 rig.radius = item.getFloat("radius", rig.radius);
536 rig.azimuth = item.getFloat("azimuth", rig.azimuth);
537 rig.elevation = item.getFloat("elevation", rig.elevation);
538 rig.yaw = item.getFloat("yaw", rig.yaw);
539 rig.pitch = item.getFloat("pitch", rig.pitch);
540 rig.fov = item.getFloat("fov", rig.fov);
541 rig.smooth = item.getFloat("smooth", rig.smooth);
542 rig.maxSpeed = item.getFloat("maxSpeed", rig.maxSpeed);
543 newRigs.push_back(std::move(rig));
544 }
545 clearTimeline();
546 rigs_ = std::move(newRigs);
547 activeRig_.clear();
548 directorEnabled_ = !rigs_.empty();
549 const auto cuts = root.get("cuts");
550 if (cuts && !cuts.isArray()) return false;
551 for (size_t i = 0; i < cuts.size(); ++i) {
552 const auto item = cuts.at(i);
553 if (!item.isObject() || !addTimelineCut(item.getFloat("time", -1.f), item.getString("rig"),
554 item.getFloat("blend", 0.f)))

556 }

559 for (size_t i = 0; i < events.size(); ++i) {

568 const auto item = floats.at(i);
569 if (!item.isObject() || !addTimelineFloat(item.getFloat("time", -1.f), item.getString("property"),
570 item.getFloat("value")))
571 return false;
572 }
573 return true;
574}

581void CameraController::evaluateTimelineFloats() {
582 for (const std::string property : {"fov", "radius", "smooth", "compositionX", "compositionY"}) {
583 const TimelineFloat* before = nullptr;
584 const TimelineFloat* after = nullptr;
585 for (const auto& key : timelineFloats_) {
586 if (key.property != property) continue;
587 if (key.time <= timelineTime_) before = &key;
588 if (key.time >= timelineTime_) {
589 after = &key;
590 break;
591 }
592 }
593 if (!before && !after) continue;
594 if (!before) before = after;
595 if (!after) after = before;
596 const float span = after->time - before->time;
597 const float t = span > 1e-6f ? (timelineTime_ - before->time) / span : 0.f;
598 const float value = glm::mix(before->value, after->value, glm::clamp(t, 0.f, 1.f));
599 if (property == "fov") setFov(value);
600 else if (property == "radius") setRadius(value);
601 else if (property == "smooth") setSmooth(value);
602 else if (property == "compositionX") setComposition(value, composition_.y);
603 else setComposition(composition_.x, value);
604 }

635void CameraController::updateTimeline(float dt) {
636 if (!timelinePlaying_) return;
637 float remaining = std::max(0.f, dt);
638 if (timelineDuration_ <= 0.f) remaining = 0.f;
639 timelineTime_ += remaining;
640 for (auto& cut : timelineCuts_) {
641 if (!cut.fired && cut.time <= timelineTime_) {
642 cut.fired = true;
643 activateRig(cut.rig, cut.blend);
644 }
645 }
646 for (auto& marker : timelineEvents_) {
647 if (!marker.fired && marker.time <= timelineTime_) {
648 marker.fired = true;
649 emitTimelineEvent(marker);
650 }
651 }
652 evaluateTimelineFloats();
653 if (timelineTime_ < timelineDuration_) return;
654 if (!timelineLoop_) {
655 timelinePlaying_ = false;

659 for (auto& cut : timelineCuts_) cut.fired = false;
660 for (auto& marker : timelineEvents_) marker.fired = false;
661 for (auto& cut : timelineCuts_)
662 if (cut.time <= timelineTime_) {
663 cut.fired = true;

665 }
666 for (auto& marker : timelineEvents_)
667 if (marker.time <= timelineTime_) {
668 marker.fired = true;
669 emitTimelineEvent(marker);
670 }
671 evaluateTimelineFloats();
672}

674void CameraController::applyCollision(View& v, float dt) {
675 obstructed_ = false;
676 collisionBodyId_ = -1;
677 const glm::vec3 ray = v.eye - v.target;
678 const float desiredDistance = glm::length(ray);
679 if (!collisionEnabled_ || desiredDistance < 1e-5f) {
680 collisionDistance_ = -1.f;
681 return;
682 }
683 const glm::vec3 dir = ray / desiredDistance;
684 float nearest = desiredDistance;
685 for (const auto& box : collisionBoxes_) {
686 const glm::vec3 bmin = box.min - glm::vec3(collisionRadius_);
687 const glm::vec3 bmax = box.max + glm::vec3(collisionRadius_);
688 float tmin = 0.f;
689 float tmax = desiredDistance;
690 bool hit = true;
691 for (int axis = 0; axis < 3; ++axis) {
692 if (std::abs(dir[axis]) < 1e-6f) {
693 if (v.target[axis] < bmin[axis] || v.target[axis] > bmax[axis]) {
694 hit = false;
695 break;
696 }
697 } else {
698 float a = (bmin[axis] - v.target[axis]) / dir[axis];
699 float b = (bmax[axis] - v.target[axis]) / dir[axis];
700 if (a > b) std::swap(a, b);
701 tmin = std::max(tmin, a);
702 tmax = std::min(tmax, b);
703 if (tmin > tmax) {
704 hit = false;
705 break;
706 }
707 }
708 }
709 if (hit && tmin >= 0.f && tmin < nearest) nearest = tmin;
710 }
711 if (auto* query = eve::cap::query<eve::ICameraObstructionQuery>()) {
712 eve::CameraObstructionHit hit;
713 if (query->sphereCast(v.target.x, v.target.y, v.target.z, v.eye.x, v.eye.y, v.eye.z,
714 collisionRadius_, collisionMask_, collisionIgnoredBody_, &hit) &&
715 hit.hit) {
716 const float distance = desiredDistance * glm::clamp(hit.fraction, 0.f, 1.f);
717 if (distance < nearest) {
718 nearest = distance;
719 collisionBodyId_ = hit.bodyId;
720 }
721 }
722 }
723 obstructed_ = nearest < desiredDistance;
724 const float safeDistance = obstructed_ ? std::max(0.05f, nearest - 0.02f) : desiredDistance;
725 if (collisionDistance_ < 0.f || safeDistance < collisionDistance_)
726 collisionDistance_ = safeDistance;
727 else {
728 const float k = 1.f - std::exp(-collisionRecovery_ * std::max(0.f, dt));
729 collisionDistance_ += (safeDistance - collisionDistance_) * k;
730 }
731 collisionDistance_ = std::min(collisionDistance_, desiredDistance);
732 v.eye = v.target + dir * collisionDistance_;
733}

735void CameraController::applyModifiers(View& v, float dt) {
736 for (auto& impulse : impulses_) {
737 impulse.age += std::max(0.f, dt);
738 const float life = glm::clamp(1.f - impulse.age / impulse.duration, 0.f, 1.f);
739 const float phase = impulse.age * 37.f + static_cast<float>(impulse.seed % 997u);
740 const glm::vec3 noise(std::sin(phase * 1.13f), std::sin(phase * 1.71f + 2.f), std::sin(phase * 2.07f + 4.f));
741 v.eye += noise * (impulse.positionAmplitude * life);
742 const glm::vec3 forward = glm::normalize(v.target - v.eye);
743 glm::vec3 right = glm::cross(forward, glm::vec3(0.f, 1.f, 0.f));
744 if (glm::length(right) > 1e-5f)
745 right = glm::normalize(right);
746 else
747 right = glm::vec3(1.f, 0.f, 0.f);
748 const glm::vec3 up = glm::normalize(glm::cross(right, forward));
749 const float angular = std::tan(glm::radians(impulse.rotationAmplitude * life));
750 v.target += (right * noise.x + up * noise.y) * angular * glm::length(v.target - v.eye);
751 v.fov = glm::clamp(v.fov + impulse.fovAmplitude * life, 1.f, 179.f);
752 }
753 impulses_.erase(
754 std::remove_if(impulses_.begin(), impulses_.end(), [](const Impulse& i) { return i.age >= i.duration; }),
755 impulses_.end());
756}

758CameraController::View CameraController::desired() const {
759 View v;
760 v.fov = fovDeg_;
761 if (blending_) {
762 float t = blendDur_ > 0.f ? glm::clamp(blendT_ / blendDur_, 0.f, 1.f) : 1.f;
763 t = t * t * (3.f - 2.f * t);
764 v.eye = glm::mix(blendFrom_.eye, blendTo_.eye, t);
765 v.target = glm::mix(blendFrom_.target, blendTo_.target, t);
766 v.fov = glm::mix(blendFrom_.fov, blendTo_.fov, t);
767 } else if (mode_ == "orbit") {
768 const float az = glm::radians(azimuthDeg_);
769 const float el = glm::radians(elevationDeg_);
770 glm::vec3 dir(std::cos(el) * std::sin(az), std::sin(el), std::cos(el) * std::cos(az));
771 v.eye = target_ + dir * radius_;
772 v.target = target_;
773 } else if (mode_ == "topdown") {
774 v.eye = target_ + glm::vec3(0.f, radius_, 0.f);
775 v.target = target_;
776 } else if (mode_ == "firstperson") {
777 v.eye = target_;
778 const float yaw = glm::radians(yawDeg_);
779 const float pitch = glm::radians(pitchDeg_);
780 glm::vec3 dir(std::cos(pitch) * std::sin(yaw), std::sin(pitch), std::cos(pitch) * std::cos(yaw));
781 v.target = v.eye + dir * 100.f;
782 } else if (mode_ == "cinematic") {
783 if (viewIndex_ >= 0 && viewIndex_ < static_cast<int>(views_.size())) {
784 v = views_[viewIndex_];
785 } else {
786 v.target = target_ + lookAhead_;
787 v.eye = v.target + offset_;
788 }
789 } else { // follow (默认)
790 v.target = target_ + lookAhead_;
791 v.eye = v.target + offset_;
792 }
793 const glm::vec3 forward = glm::normalize(v.target - v.eye);
794 glm::vec3 right = glm::cross(forward, glm::vec3(0.f, 1.f, 0.f));
795 if (glm::length(right) < 1e-5f)
796 right = glm::vec3(1.f, 0.f, 0.f);
797 else
798 right = glm::normalize(right);
799 glm::vec3 up = glm::normalize(glm::cross(right, forward));
800 const float distance = glm::length(v.target - v.eye);
801 v.target += (right * composition_.x + up * composition_.y) * distance;
802 return v;
803}

805void CameraController::applyView(const View& v) {
806 if (!cam_) return;
807 cam_->setEye(v.eye.x, v.eye.y, v.eye.z);
808 cam_->setTarget(v.target.x, v.target.y, v.target.z);
809 const glm::vec3 forward = glm::normalize(v.target - v.eye);
810 glm::vec3 up(0.f, 1.f, 0.f);
811 if (std::abs(glm::dot(forward, up)) > 0.999f) up = glm::vec3(0.f, 0.f, -1.f);
812 glm::vec3 right=glm::normalize(glm::cross(forward,up));
813 up=glm::normalize(glm::cross(right,forward));
814 const float roll=glm::radians(cameraRollDeg_);
815 up=up*std::cos(roll)+right*std::sin(roll);
816 cam_->setUp(up.x, up.y, up.z);
817 cam_->setFov(v.fov);
818}

820void CameraController::easeToward(const View& v, float dt) {
821 const float eyeK = 1.f - std::exp(-positionSmooth_ * dt);
822 const float targetK = 1.f - std::exp(-targetSmooth_ * dt);
823 glm::vec3 deye = (v.eye - cur_.eye) * eyeK;
824 glm::vec3 dtgt = (v.target - cur_.target) * targetK;
825 if (maxSpeed_ > 0.f && dt > 0.f) {
826 const float maxStep = maxSpeed_ * dt;
827 const float le = glm::length(deye);
828 if (le > maxStep && le > 1e-6f) deye *= maxStep / le;
829 const float lt = glm::length(dtgt);
830 if (lt > maxStep && lt > 1e-6f) dtgt *= maxStep / lt;
831 }
832 cur_.eye += deye;
833 cur_.target += dtgt;
834 cur_.fov += (v.fov - cur_.fov) * eyeK;
835}

843void CameraController::update(float dt) {
844 if (!cam_) return;
845
846 dt = std::max(0.f, dt);
847 updateTrackedTarget();
848 updateTimeline(dt);
849 updateDirector();
850
851 if (mode_ == "cinematic" && playing_ && !views_.empty()) {
852 timer_ += dt;
853 if (!blending_ && timer_ >= stepTime_) {
854 const int next = (viewIndex_ + 1) % static_cast<int>(views_.size());
855 switchTo(views_[next].name, blendDur_);
856 timer_ = 0.f;
857 }
858 }
859
860 if (blending_) {
861 blendT_ += dt;
862 if (blendT_ >= blendDur_) {
863 blending_ = false;
864 blendT_ = blendDur_;
865 }
866 }
867
868 // orbit 自动盘旋：不断累积方位角
869 if (mode_ == "orbit") {
870 azimuthDeg_ += orbitSpeedDeg_ * dt;
871 }
872
873 View d = desired();
874 applyCollision(d, dt);
875 applyModifiers(d, dt);
876
877 if (!initialized_) {
878 cur_ = d;
879 initialized_ = true;
880 }
881 easeToward(d, dt);
882 applyView(cur_);
883}
