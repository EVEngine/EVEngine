21std::string jsonString(const std::string& value) {
22 std::ostringstream out;
23 out << '"';
24 for (const unsigned char c : value) {
25 switch (c) {
26 case '"': out << "\\\""; break;
27 case '\\': out << "\\\\"; break;
28 case '\b': out << "\\b"; break;
29 case '\f': out << "\\f"; break;
30 case '\n': out << "\\n"; break;
31 case '\r': out << "\\r"; break;
32 case '\t': out << "\\t"; break;
33 default:
34 if (c < 0x20)
35 out << "\\u00" << "0123456789abcdef"[c >> 4] << "0123456789abcdef"[c & 0xf];
36 else
37 out << static_cast<char>(c);
38 }
39 }
40 out << '"';
41 return out.str();
42}

421bool CameraController::deserializeAsset(const std::string& text) {
422 const auto document = eve::json::Document::parse(text);
423 const auto root = document.root();
424 if (!root.isObject() || root.getInt("version", 0) != 1) return false;
425 std::vector<Rig> newRigs;
426 const auto rigs = root.get("rigs");
427 if (rigs && !rigs.isArray()) return false;
428 for (size_t i = 0; i < rigs.size(); ++i) {
429 const auto item = rigs.at(i);
430 if (!item.isObject()) return false;
431 Rig rig;
432 rig.name = item.getString("name");
433 rig.mode = item.getString("mode");
434 if (rig.name.empty() || !validMode(rig.mode)) return false;
435 rig.priority = item.getInt("priority");
436 rig.enabled = item.getBool("enabled", true);
437 rig.target = jsonVec3(item.get("target"), rig.target);
438 rig.offset = jsonVec3(item.get("offset"), rig.offset);
439 rig.lookAhead = jsonVec3(item.get("lookAhead"), rig.lookAhead);
440 const auto composition = item.get("composition");
441 if (composition.isArray() && composition.size() == 2)
442 rig.composition = {composition.at(0).asFloat(), composition.at(1).asFloat()};
443 rig.radius = item.getFloat("radius", rig.radius);
444 rig.azimuth = item.getFloat("azimuth", rig.azimuth);
445 rig.elevation = item.getFloat("elevation", rig.elevation);
446 rig.yaw = item.getFloat("yaw", rig.yaw);
447 rig.pitch = item.getFloat("pitch", rig.pitch);
448 rig.fov = item.getFloat("fov", rig.fov);
449 rig.smooth = item.getFloat("smooth", rig.smooth);
450 rig.maxSpeed = item.getFloat("maxSpeed", rig.maxSpeed);
451 newRigs.push_back(std::move(rig));
452 }
453 clearTimeline();
454 rigs_ = std::move(newRigs);
455 activeRig_.clear();
456 directorEnabled_ = !rigs_.empty();
457 const auto cuts = root.get("cuts");
458 if (cuts && !cuts.isArray()) return false;
459 for (size_t i = 0; i < cuts.size(); ++i) {
460 const auto item = cuts.at(i);
461 if (!item.isObject() || !addTimelineCut(item.getFloat("time", -1.f), item.getString("rig"),
462 item.getFloat("blend", 0.f)))
463 return false;
464 }
465 const auto events = root.get("events");
466 if (events && !events.isArray()) return false;
467 for (size_t i = 0; i < events.size(); ++i) {
468 const auto item = events.at(i);
469 if (!item.isObject() || !addTimelineEvent(item.getFloat("time", -1.f), item.getString("name"),
470 item.getString("data")))
471 return false;
472 }
473 const auto floats = root.get("floats");
474 if (floats && !floats.isArray()) return false;
475 for (size_t i = 0; i < floats.size(); ++i) {
476 const auto item = floats.at(i);
477 if (!item.isObject() || !addTimelineFloat(item.getFloat("time", -1.f), item.getString("property"),
478 item.getFloat("value")))
479 return false;
480 }
481 return true;
482}

489void CameraController::evaluateTimelineFloats() {
490 for (const std::string property : {"fov", "radius", "smooth", "compositionX", "compositionY"}) {
491 const TimelineFloat* before = nullptr;
492 const TimelineFloat* after = nullptr;
493 for (const auto& key : timelineFloats_) {
494 if (key.property != property) continue;
495 if (key.time <= timelineTime_) before = &key;
496 if (key.time >= timelineTime_) {
497 after = &key;
498 break;
499 }
500 }
501 if (!before && !after) continue;
502 if (!before) before = after;
503 if (!after) after = before;
504 const float span = after->time - before->time;
505 const float t = span > 1e-6f ? (timelineTime_ - before->time) / span : 0.f;
506 const float value = glm::mix(before->value, after->value, glm::clamp(t, 0.f, 1.f));
507 if (property == "fov") setFov(value);
508 else if (property == "radius") setRadius(value);
509 else if (property == "smooth") setSmooth(value);
510 else if (property == "compositionX") setComposition(value, composition_.y);
511 else setComposition(composition_.x, value);
512 }
513}

515void CameraController::updateTrackedTarget() {
516 if (targetNodeHost_.empty() || targetNodeId_.empty()) return;
517 const scene::SceneNodeRef node(targetNodeHost_, targetNodeId_);
518 if (!node.isValid()) return;
519 const glm::vec3 observed(node.getWorldPositionX(), node.getWorldPositionY(), node.getWorldPositionZ());
520 if (!deadZoneInitialized_ || deadZone_ <= 0.f) {
521 deadZoneTarget_ = observed;
522 deadZoneInitialized_ = true;
523 } else {
524 const glm::vec3 delta = observed - deadZoneTarget_;
525 const float length = glm::length(delta);
526 if (length > deadZone_ && length > 1e-6f) deadZoneTarget_ += delta * ((length - deadZone_) / length);
527 }
528 target_ = deadZoneTarget_;
529}

531void CameraController::updateDirector() {
532 if (!directorEnabled_ || timelinePlaying_) return;
533 const Rig* best = nullptr;
534 const Rig* active = nullptr;
535 for (const auto& rig : rigs_) {
536 if (rig.name == activeRig_ && rig.enabled) active = &rig;
537 if (rig.enabled && (!best || rig.priority > best->priority)) best = &rig;
538 }
539 if (active && best && active->priority >= best->priority) best = active;
540 if (best && best->name != activeRig_) activateRig(best->name, initialized_ ? 0.25f : 0.f);

544 if (!timelinePlaying_) return;

582void CameraController::applyCollision(View& v, float dt) {
583 obstructed_ = false;
584 collisionBodyId_ = -1;
585 const glm::vec3 ray = v.eye - v.target;
586 const float desiredDistance = glm::length(ray);
587 if (!collisionEnabled_ || desiredDistance < 1e-5f) {
588 collisionDistance_ = -1.f;
589 return;
590 }
591 const glm::vec3 dir = ray / desiredDistance;
592 float nearest = desiredDistance;
593 for (const auto& box : collisionBoxes_) {
594 const glm::vec3 bmin = box.min - glm::vec3(collisionRadius_);
595 const glm::vec3 bmax = box.max + glm::vec3(collisionRadius_);
596 float tmin = 0.f;
597 float tmax = desiredDistance;
598 bool hit = true;
599 for (int axis = 0; axis < 3; ++axis) {
600 if (std::abs(dir[axis]) < 1e-6f) {
601 if (v.target[axis] < bmin[axis] || v.target[axis] > bmax[axis]) {
602 hit = false;
603 break;
604 }
605 } else {
606 float a = (bmin[axis] - v.target[axis]) / dir[axis];
607 float b = (bmax[axis] - v.target[axis]) / dir[axis];
608 if (a > b) std::swap(a, b);
609 tmin = std::max(tmin, a);
610 tmax = std::min(tmax, b);
611 if (tmin > tmax) {
612 hit = false;
613 break;
614 }
615 }
616 }
617 if (hit && tmin >= 0.f && tmin < nearest) nearest = tmin;
618 }
619 if (auto* query = eve::cap::query<eve::ICameraObstructionQuery>()) {
620 eve::CameraObstructionHit hit;
621 if (query->sphereCast(v.target.x, v.target.y, v.target.z, v.eye.x, v.eye.y, v.eye.z,
622 collisionRadius_, collisionMask_, collisionIgnoredBody_, &hit) &&
623 hit.hit) {
624 const float distance = desiredDistance * glm::clamp(hit.fraction, 0.f, 1.f);
625 if (distance < nearest) {
626 nearest = distance;
627 collisionBodyId_ = hit.bodyId;
628 }
629 }
630 }
631 obstructed_ = nearest < desiredDistance;
632 const float safeDistance = obstructed_ ? std::max(0.05f, nearest - 0.02f) : desiredDistance;
633 if (collisionDistance_ < 0.f || safeDistance < collisionDistance_)
634 collisionDistance_ = safeDistance;
635 else {
636 const float k = 1.f - std::exp(-collisionRecovery_ * std::max(0.f, dt));
637 collisionDistance_ += (safeDistance - collisionDistance_) * k;
638 }
639 collisionDistance_ = std::min(collisionDistance_, desiredDistance);
640 v.eye = v.target + dir * collisionDistance_;
641}

643void CameraController::applyModifiers(View& v, float dt) {
644 for (auto& impulse : impulses_) {
645 impulse.age += std::max(0.f, dt);
646 const float life = glm::clamp(1.f - impulse.age / impulse.duration, 0.f, 1.f);
647 const float phase = impulse.age * 37.f + static_cast<float>(impulse.seed % 997u);
648 const glm::vec3 noise(std::sin(phase * 1.13f), std::sin(phase * 1.71f + 2.f), std::sin(phase * 2.07f + 4.f));
649 v.eye += noise * (impulse.positionAmplitude * life);
650 const glm::vec3 forward = glm::normalize(v.target - v.eye);
651 glm::vec3 right = glm::cross(forward, glm::vec3(0.f, 1.f, 0.f));
652 if (glm::length(right) > 1e-5f)
653 right = glm::normalize(right);
654 else
655 right = glm::vec3(1.f, 0.f, 0.f);
656 const glm::vec3 up = glm::normalize(glm::cross(right, forward));
657 const float angular = std::tan(glm::radians(impulse.rotationAmplitude * life));
658 v.target += (right * noise.x + up * noise.y) * angular * glm::length(v.target - v.eye);
659 v.fov = glm::clamp(v.fov + impulse.fovAmplitude * life, 1.f, 179.f);
660 }
661 impulses_.erase(
662 std::remove_if(impulses_.begin(), impulses_.end(), [](const Impulse& i) { return i.age >= i.duration; }),
663 impulses_.end());
664}

666CameraController::View CameraController::desired() const {
667 View v;
668 v.fov = fovDeg_;
669 if (blending_) {
670 float t = blendDur_ > 0.f ? glm::clamp(blendT_ / blendDur_, 0.f, 1.f) : 1.f;
671 t = t * t * (3.f - 2.f * t);
672 v.eye = glm::mix(blendFrom_.eye, blendTo_.eye, t);
673 v.target = glm::mix(blendFrom_.target, blendTo_.target, t);
674 v.fov = glm::mix(blendFrom_.fov, blendTo_.fov, t);
675 } else if (mode_ == "orbit") {
676 const float az = glm::radians(azimuthDeg_);
677 const float el = glm::radians(elevationDeg_);
678 glm::vec3 dir(std::cos(el) * std::sin(az), std::sin(el), std::cos(el) * std::cos(az));
679 v.eye = target_ + dir * radius_;
680 v.target = target_;
681 } else if (mode_ == "topdown") {
682 v.eye = target_ + glm::vec3(0.f, radius_, 0.f);
683 v.target = target_;
684 } else if (mode_ == "firstperson") {
685 v.eye = target_;
686 const float yaw = glm::radians(yawDeg_);
687 const float pitch = glm::radians(pitchDeg_);
688 glm::vec3 dir(std::cos(pitch) * std::sin(yaw), std::sin(pitch), std::cos(pitch) * std::cos(yaw));
689 v.target = v.eye + dir * 100.f;
690 } else if (mode_ == "cinematic") {
691 if (viewIndex_ >= 0 && viewIndex_ < static_cast<int>(views_.size())) {
692 v = views_[viewIndex_];
693 } else {
694 v.target = target_ + lookAhead_;
695 v.eye = v.target + offset_;
696 }
697 } else { // follow (默认)
698 v.target = target_ + lookAhead_;
699 v.eye = v.target + offset_;
700 }
701 const glm::vec3 forward = glm::normalize(v.target - v.eye);
702 glm::vec3 right = glm::cross(forward, glm::vec3(0.f, 1.f, 0.f));
703 if (glm::length(right) < 1e-5f)
704 right = glm::vec3(1.f, 0.f, 0.f);
705 else
706 right = glm::normalize(right);
707 glm::vec3 up = glm::normalize(glm::cross(right, forward));
708 const float distance = glm::length(v.target - v.eye);
709 v.target += (right * composition_.x + up * composition_.y) * distance;
710 return v;
711}

713void CameraController::applyView(const View& v) {
714 if (!cam_) return;
715 cam_->setEye(v.eye.x, v.eye.y, v.eye.z);
716 cam_->setTarget(v.target.x, v.target.y, v.target.z);
717 const glm::vec3 forward = glm::normalize(v.target - v.eye);
718 glm::vec3 up(0.f, 1.f, 0.f);
719 if (std::abs(glm::dot(forward, up)) > 0.999f) up = glm::vec3(0.f, 0.f, -1.f);
720 cam_->setUp(up.x, up.y, up.z);
721 cam_->setFov(v.fov);
722}

724void CameraController::easeToward(const View& v, float dt) {
725 const float eyeK = 1.f - std::exp(-positionSmooth_ * dt);
726 const float targetK = 1.f - std::exp(-targetSmooth_ * dt);
727 glm::vec3 deye = (v.eye - cur_.eye) * eyeK;
728 glm::vec3 dtgt = (v.target - cur_.target) * targetK;
729 if (maxSpeed_ > 0.f && dt > 0.f) {
730 const float maxStep = maxSpeed_ * dt;
731 const float le = glm::length(deye);
732 if (le > maxStep && le > 1e-6f) deye *= maxStep / le;
733 const float lt = glm::length(dtgt);
734 if (lt > maxStep && lt > 1e-6f) dtgt *= maxStep / lt;
735 }
736 cur_.eye += deye;
737 cur_.target += dtgt;
738 cur_.fov += (v.fov - cur_.fov) * eyeK;
739}

747void CameraController::update(float dt) {
748 if (!cam_) return;
749
750 dt = std::max(0.f, dt);
751 updateTrackedTarget();
752 updateTimeline(dt);
753 updateDirector();
754
755 if (mode_ == "cinematic" && playing_ && !views_.empty()) {
756 timer_ += dt;
757 if (!blending_ && timer_ >= stepTime_) {
758 const int next = (viewIndex_ + 1) % static_cast<int>(views_.size());
759 switchTo(views_[next].name, blendDur_);
760 timer_ = 0.f;
761 }
762 }
763
764 if (blending_) {
765 blendT_ += dt;
766 if (blendT_ >= blendDur_) {
767 blending_ = false;
768 blendT_ = blendDur_;
769 }
770 }
771
772 // orbit 自动盘旋：不断累积方位角
773 if (mode_ == "orbit") {
774 azimuthDeg_ += orbitSpeedDeg_ * dt;
775 }
776
777 View d = desired();
778 applyCollision(d, dt);
779 applyModifiers(d, dt);
780
781 if (!initialized_) {
782 cur_ = d;
783 initialized_ = true;
784 }
785 easeToward(d, dt);
786 applyView(cur_);
787}
