43procgen_editing::EditorResult<void> MeshModifierEditor::activateTool(editor::EditorWorkspace& workspace,
44 std::string tool) {
45 std::string panel;
46 if (tool == "graph") panel = "meshModifier.graph";
47 else if (tool == "spline") panel = "meshModifier.spline";
48 else if (tool == "sculpt") panel = "meshModifier.sculpt";
49 else if (tool == "uvPaint") panel = "meshModifier.uvPaint";
50 else return rejected("editor.mesh-modifier.tool", "Unknown mesh modifier tool");
51 editor::EditorWorkspace candidate = workspace;
52 auto activated = candidate.activatePanel(editor::StableId(panel));
53 if (!activated.ok()) return rejected("editor.mesh-modifier.tool-panel", "Mesh modifier tool panel is unavailable");
54 auto mode = candidate.setModeId(editor::StableId("meshModifier." + tool));
55 if (!mode.ok()) return rejected("editor.mesh-modifier.tool-mode", "Could not set mesh modifier tool mode");
56 auto focus = candidate.focusItem("meshModifier", editor::StableId(panel), editor::StableId(targetId_));
57 if (!focus.ok()) return rejected("editor.mesh-modifier.tool-focus", "Could not focus mesh modifier tool");
58 workspace = std::move(candidate); activeTool_ = std::move(tool);
59 return eve::editing::applied<void>();
60}

62procgen_editing::EditorResult<void> MeshModifierEditor::observeRevision(std::string document, std::uint64_t revision) {
63 std::uint64_t* destination = nullptr;
64 if (document == "graph") destination = &graphRevision_;
65 else if (document == "spline") destination = &splineRevision_;
66 else if (document == "mesh") destination = &meshRevision_;
67 else if (document == "paint") destination = &paintRevision_;
68 else return rejected("editor.mesh-modifier.document", "Unknown mesh modifier document");
69 if (revision < *destination) return rejected("editor.mesh-modifier.stale-revision", "Document revision cannot move backwards");
70 *destination = revision;
71 return eve::editing::applied<void>();
72}
