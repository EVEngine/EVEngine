51eve::Result<eve::Value> runGenerator(HSQUIRRELVM vm, Procgen& procgen, const ssq::Table& generator,
52 const ssq::Object& params, const std::string& systemName,
53 std::uint32_t seed) {
54 if (systemName.empty())
55 return hostFailure<eve::Value>(eve::DiagnosticCode::InvalidArgument,
56 "script generator system name must not be empty", "system");
57 if (generator.isEmpty())
58 return hostFailure<eve::Value>(eve::DiagnosticCode::InvalidArgument,
59 "script generator module must be a table", "generator");
60 if (params.isEmpty())
61 return hostFailure<eve::Value>(eve::DiagnosticCode::InvalidArgument,
62 "script generator Params must not be empty", "params");
63
64 thread_local std::unordered_set<std::string> activeSystems;
65 ActiveSystemScope active(activeSystems, systemName);
66 if (!active.inserted())
67 return hostFailure<eve::Value>(eve::DiagnosticCode::Conflict,
68 "script generator cannot recursively rebuild the same system", "system");
69
70 std::optional<ssq::Function> generate;
71 try {
72 generate.emplace(generator.findFunc("generate"));
73 } catch (const std::exception& error) {
74 return hostFailure<eve::Value>(eve::DiagnosticCode::InvalidArgument,
75 std::string("script generator requires generate(params, ctx): ") + error.what(),
76 "generator.generate");
77 }
78
79 auto contextResult = Procgen::beginSystemHandle(systemName, seed);
80 if (!contextResult.ok()) return eve::Result<eve::Value>::failure(contextResult.status());
81 const ProcgenContextHandleRef contextRef = std::move(contextResult).takeValue();
82 auto contextView = Procgen::resolve(contextRef);
83 if (!contextView.isBound()) {
84 Procgen::release(contextRef).ignore("release unresolved script generator context");
85 return hostFailure<eve::Value>(eve::DiagnosticCode::StaleHandle,
86 "script generator context became stale before execution", "context");
87 }
88
89 auto contextObject = eve::script::makeOwnedSquirrelInstance<ScriptProcgenContext>(
90 vm, std::make_unique<ScriptProcgenContext>(contextRef));
91 if (!contextObject.ok()) {
92 const eve::Status status = contextObject.status();
93 contextObject.ignore("script generator context proxy creation failed");
94 procgen.abortSystem(contextRef).ignore("abort script generator after proxy creation failure");
95 Procgen::release(contextRef).ignore("release script generator context after proxy creation failure");
96 return eve::Result<eve::Value>::failure(status);
97 }
98 ssq::Object context = std::move(contextObject).takeValue();
99
100 const SQInteger top = sq_gettop(vm);
101 eve::script::clearLastScriptError(vm);
102 sq_pushobject(vm, generate->getRaw());
103 sq_pushobject(vm, generator.getRaw());
104 sq_pushobject(vm, params.getRaw());
105 sq_pushobject(vm, context.getRaw());
106 const bool called = SQ_SUCCEEDED(sq_call(vm, 3, SQFalse, SQTrue));
107 sq_settop(vm, top);
108
109 if (!called) {
110 std::string error = eve::script::formatLastScriptError(vm);
111 if (error.empty()) error = "script generator callback failed";
112 procgen.abortSystem(contextRef).ignore("abort failed script generator callback");
113 Procgen::release(contextRef).ignore("release failed script generator context");
114 return hostFailure<eve::Value>(eve::DiagnosticCode::Failed, std::move(error), "generator.generate");
115 }
116
117 auto committed = procgen.commitSystem(contextRef);
118 const eve::Status commitStatus = committed.status();
119 if (!committed.ok()) {
120 Procgen::release(contextRef).ignore("release rejected script generator context");
121 return eve::Result<eve::Value>::failure(commitStatus);
122 }
123 committed.ignore("script generator commit observed by host");
124
125 eve::Value::Array outputs;
126 const int outputCount = procgen.getSystemOutputCount(systemName);
127 outputs.reserve(static_cast<std::size_t>(outputCount));
128 for (int index = 0; index < outputCount; ++index)
129 outputs.emplace_back(procgen.getSystemOutputName(systemName, index));
130
131 eve::Value receipt(eve::Value::Object{
132 {"system", eve::Value(systemName)},
133 {"seed", eve::Value(static_cast<std::int64_t>(procgen.getSystemSeed(systemName)))},
134 {"revision", eve::Value(static_cast<std::int64_t>(procgen.getSystemRevision(systemName)))},
135 {"outputs", eve::Value(std::move(outputs))},
136 });
137 Procgen::release(contextRef).ignore("release committed script generator context");
138 return eve::Result<eve::Value>::success(std::move(receipt), commitStatus);
139}
