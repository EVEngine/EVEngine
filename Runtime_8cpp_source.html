199std::vector<ReflectedMember> collectClassMembersFromStack(HSQUIRRELVM squirrel,
200 SQInteger classIndex) {
201 std::vector<ReflectedMember> members;
202 const SQInteger top = sq_gettop(squirrel);
203 sq_push(squirrel, classIndex); // duplicate the class for iteration
204 sq_pushnull(squirrel);
205 while (SQ_SUCCEEDED(sq_next(squirrel, -2))) {
206 if (sq_gettype(squirrel, -2) != OT_STRING) {
207 sq_pop(squirrel, 2);
208 continue;
209 }
210 const SQChar* memberName = nullptr;
211 if (!SQ_SUCCEEDED(sq_getstring(squirrel, -2, &memberName)) || !memberName) {
212 sq_pop(squirrel, 2);
213 continue;
214 }
215 ReflectedMember member;
216 member.name = memberName;
217 member.type = static_cast<ssq::Type>(sq_gettype(squirrel, -1));
218 member.method = member.type == ssq::Type::CLOSURE ||
219 member.type == ssq::Type::NATIVECLOSURE;
220
221 // sq_getattributes consumes the key at the stack top and replaces it
222 // with the attribute table (or null).
223 const SQInteger memberTop = sq_gettop(squirrel);
224 sq_push(squirrel, -2);
225 if (SQ_SUCCEEDED(sq_getattributes(squirrel, -5)) &&
226 sq_gettype(squirrel, -1) == OT_TABLE) {
227 sq_pushnull(squirrel);
228 while (SQ_SUCCEEDED(sq_next(squirrel, -2))) {
229 if (sq_gettype(squirrel, -2) == OT_STRING) {
230 const SQChar* attributeName = nullptr;
231 if (SQ_SUCCEEDED(sq_getstring(squirrel, -2, &attributeName)) &&
232 attributeName) {
233 ReflectedAttribute attribute;
234 attribute.name = attributeName;
235 attribute.type =
236 static_cast<ssq::Type>(sq_gettype(squirrel, -1));
237 attribute.value = valueString(squirrel, -1);
238 member.attributes.push_back(std::move(attribute));
239 }
240 }
241 sq_pop(squirrel, 2);
242 }
243 }
244 sq_settop(squirrel, memberTop);
245 std::sort(member.attributes.begin(), member.attributes.end(),
246 [](const ReflectedAttribute& a, const ReflectedAttribute& b) {
247 return a.name < b.name;
248 });
249 members.push_back(std::move(member));
250 sq_pop(squirrel, 2);
251 }
252 sq_settop(squirrel, top);
253 std::sort(members.begin(), members.end(),
254 [](const ReflectedMember& a, const ReflectedMember& b) {
255 return a.name < b.name;
256 });
257 return members;
258}

357std::vector<std::string> ReflectedMember::attrOptions(const std::string& name) const {
358 std::vector<std::string> options;
359 const ReflectedAttribute* attribute = findAttribute(name);
360 if (!attribute) return options;
361 std::string rest = attribute->value;
362 while (true) {
363 const size_t comma = rest.find(',');
364 std::string piece = comma == std::string::npos ? rest : rest.substr(0, comma);
365 const size_t first = piece.find_first_not_of(" \t\r\n");
366 const size_t last = piece.find_last_not_of(" \t\r\n");
367 if (first != std::string::npos)
368 options.push_back(piece.substr(first, last - first + 1));
369 if (comma == std::string::npos) break;
370 rest = rest.substr(comma + 1);
371 }
372 return options;
373}

515const ScriptInfo& Runtime::execute(ScriptId id) {
516 auto found = scripts_.find(id);
517 if (found == scripts_.end())
518 throw ScriptException(ScriptStage::Execute, {}, id, "unknown script");
519 ScriptRecord& record = *found->second;
520 if (!record.compiled || record.info.state == ScriptState::Unloaded)
521 throw ScriptException(ScriptStage::Execute, record.info.source, id, "script is unloaded");
522
523 auto scope = enter();
524 auto stack = guard();
525 const auto before = rootClasses();
526 record.info.state = ScriptState::Running;
527 record.info.error.clear();
528 notifyLifecycle(record.info);
529 try {
530 // Drive the Squirrel call directly instead of ssq::VM::run(): when a
531 // DevTool hook replaces the VM's error handler, ssq never populates its
532 // stored RuntimeException and would dereference a null unique_ptr on
533 // failure. The raw call reports through the installed handler (which
534 // captures the live stack) and lets us throw an enriched ScriptException.
535 HSQUIRRELVM vm = handle();
536 sq_pushobject(vm, record.compiled->getRaw());
537 sq_pushroottable(vm);
538 const SQRESULT result = sq_call(vm, 1, SQFalse, SQTrue);
539 if (SQ_FAILED(result)) {
540 script::ScriptErrorContext ctx = script::takeLastScriptError(vm);
541 if (ctx.empty()) ctx.message = "script error";
542 try {
543 discoverClasses(record, before);
544 } catch (...) {
545 }
546 record.info.state = ScriptState::Failed;
547 record.info.error = script::formatScriptError(ctx);
548 notifyLifecycle(record.info);
549 fail(ScriptStage::Execute, record.info.source, id, std::move(ctx));
550 }
551 discoverClasses(record, before);
552 record.info.state = ScriptState::Loaded;
553 notifyLifecycle(record.info);
554 return record.info;
555 } catch (const ScriptException&) {
556 throw;
557 } catch (const std::exception& error) {
558 try {
559 discoverClasses(record, before);
560 } catch (...) {
561 }
562 record.info.state = ScriptState::Failed;
563 record.info.error = error.what();
564 notifyLifecycle(record.info);
565 fail(ScriptStage::Execute, record.info.source, id, error);
566 }
567}

595ssq::Object Runtime::createInstance(const std::string& name, const std::string& source) {
596 const bool validName = !name.empty();
597 EV_PARAM_CHECK(validName, "instance class name must not be empty");
598 auto scope = enter();
599 auto stack = guard();
600 const std::string label = source.empty() ? name : source;
601 HSQUIRRELVM squirrel = handle();
602 const SQInteger top = sq_gettop(squirrel);
603 sq_pushroottable(squirrel); // [root]
604 sq_pushstring(squirrel, name.c_str(), -1); // [root, name]
605 if (SQ_FAILED(sq_get(squirrel, -2)) ||
606 sq_gettype(squirrel, -1) != OT_CLASS) {
607 sq_settop(squirrel, top);
608 script::ScriptErrorContext ctx;
609 ctx.message = "class not found: " + name;
610 fail(ScriptStage::Reflect, label, 0, std::move(ctx));
611 }
612 sq_createinstance(squirrel, -1); // [root, class, instance]
613 ssq::Object instance(squirrel);
614 sq_getstackobj(squirrel, -1, &instance.getRaw());
615 sq_addref(squirrel, &instance.getRaw()); // root the instance
616 // Call the default constructor with `this` = the new instance.
617 sq_pushobject(squirrel, instance.getRaw()); // [.., instance, instance]
618 sq_pushstring(squirrel, "constructor", -1); // [.., instance, instance, "constructor"]
619 if (SQ_SUCCEEDED(sq_get(squirrel, -2)) &&
620 (sq_gettype(squirrel, -1) == OT_CLOSURE ||
621 sq_gettype(squirrel, -1) == OT_NATIVECLOSURE)) {
622 // sq_call expects [closure, this, args...] with params including `this`.
623 sq_pushobject(squirrel, instance.getRaw()); // [.., instance, ctor, this]
624 if (SQ_FAILED(sq_call(squirrel, 1, SQTrue, SQTrue))) {
625 script::ScriptErrorContext ctx = script::takeLastScriptError(squirrel);
626 if (ctx.empty()) ctx.message = "constructor failed";
627 sq_settop(squirrel, top);
628 fail(ScriptStage::Reflect, label, 0, std::move(ctx));
629 }
630 }
631 sq_settop(squirrel, top);
632 return instance;
633}

635std::vector<ReflectedMember> Runtime::reflectInstance(const ssq::Object& instance) const {
636 std::vector<ReflectedMember> members;
637 if (instance.getType() != ssq::Type::INSTANCE) return members;
638 Runtime* self = const_cast<Runtime*>(this);
639 auto scope = self->enter();
640 auto stack = self->guard();
641 HSQUIRRELVM squirrel = handle();
642 const SQInteger top = sq_gettop(squirrel);
643
644 // Walk the instance's class chain (derived → base) and merge member
645 // metadata; derived members win when a name is overridden.
646 sq_pushobject(squirrel, instance.getRaw());
647 if (SQ_FAILED(sq_getclass(squirrel, -1)) || sq_gettype(squirrel, -1) != OT_CLASS) {
648 sq_settop(squirrel, top);
649 return members;
650 }
651 while (sq_gettype(squirrel, -1) == OT_CLASS) {
652 const SQInteger classTop = sq_gettop(squirrel);
653 for (ReflectedMember& member : collectClassMembersFromStack(squirrel, -1)) {
654 if (!member.method) {
655 const SQInteger readTop = sq_gettop(squirrel);
656 sq_pushobject(squirrel, instance.getRaw());
657 sq_pushstring(squirrel, member.name.c_str(), -1);
658 if (SQ_SUCCEEDED(sq_get(squirrel, -2)))
659 member.value = valueFromStack(squirrel, -1);
660 sq_settop(squirrel, readTop);
661 }
662 auto existing =
663 std::find_if(members.begin(), members.end(),
664 [&](const ReflectedMember& m) { return m.name == member.name; });
665 if (existing != members.end())
666 *existing = std::move(member);
667 else
668 members.push_back(std::move(member));
669 }
670 sq_settop(squirrel, classTop);
671 // Move to the base class; a missing base ends the chain.
672 if (!SQ_SUCCEEDED(sq_getbase(squirrel, -1)) ||
673 sq_gettype(squirrel, -1) != OT_CLASS) {
674 sq_settop(squirrel, top);
675 break;
676 }
677 sq_remove(squirrel, -2); // drop the derived class, keep the base on top
678 }
679 return members;
680}

701bool Runtime::writeProperty(const ssq::Object& instance, const std::string& name,
702 const ReflectedValue& value) const {
703 if (name.empty() || value.empty()) return false;
704 Runtime* self = const_cast<Runtime*>(this);
705 auto scope = self->enter();
706 auto stack = self->guard();
707 HSQUIRRELVM squirrel = handle();
708 const SQInteger top = sq_gettop(squirrel);
709 sq_pushobject(squirrel, instance.getRaw()); // [instance]
710 sq_pushstring(squirrel, name.c_str(), -1); // [instance, name]
711 if (SQ_FAILED(sq_get(squirrel, -2))) { // [instance, current]
712 sq_settop(squirrel, top);
713 return false;
714 }
715 const SQObjectType currentType = sq_gettype(squirrel, -1);
716 sq_pop(squirrel, 1); // [instance]
717 if (currentType == OT_CLOSURE || currentType == OT_NATIVECLOSURE ||
718 currentType == OT_CLASS) {
719 sq_settop(squirrel, top);
720 return false;
721 }
722 sq_pushstring(squirrel, name.c_str(), -1); // [instance, name]
723 switch (currentType) {
724 case OT_BOOL:
725 sq_pushbool(squirrel, valueToBool(value) ? SQTrue : SQFalse);
726 break;
727 case OT_INTEGER:
728 sq_pushinteger(squirrel, static_cast<SQInteger>(valueToInteger(value)));
729 break;
730 case OT_FLOAT:
731 sq_pushfloat(squirrel, static_cast<SQFloat>(valueToDouble(value)));
732 break;
733 case OT_STRING: {
734 const std::string text = valueToText(value);
735 sq_pushstring(squirrel, text.c_str(), static_cast<SQInteger>(text.size()));
736 break;
737 }
738 default: {
739 // Null/unsupported slot: take the type of the incoming value.
740 switch (value.kind) {
741 case ReflectedValueKind::Bool:
742 sq_pushbool(squirrel, value.asBool() ? SQTrue : SQFalse);
743 break;
744 case ReflectedValueKind::Integer:
745 sq_pushinteger(squirrel, SQInteger(value.asInt()));
746 break;
747 case ReflectedValueKind::Float:
748 sq_pushfloat(squirrel, SQFloat(value.asFloat()));
749 break;
750 case ReflectedValueKind::String:
751 sq_pushstring(squirrel, value.asString().c_str(), -1);
752 break;
753 default: sq_pushnull(squirrel); break;
754 }
755 break;
756 }
757 }
758 // [instance, name, value] — sq_set pops the value and assigns the slot.
759 const bool ok = SQ_SUCCEEDED(sq_set(squirrel, -3));
760 sq_settop(squirrel, top);
761 return ok;
762}

1031bool Runtime::unload(ScriptId id) {
1032 auto found = scripts_.find(id);
1033 if (found == scripts_.end()) return false;
1034 ScriptRecord& record = *found->second;
1035 if (record.info.state == ScriptState::Unloaded) return false;
1036 auto scope = enter();
1037 auto stack = guard();
1038 record.info.state = ScriptState::Unloading;
1039 notifyLifecycle(record.info);
1040 try {
1041 for (const auto& pair : record.class_objects) {
1042 auto owner = class_owners_.find(pair.first);
1043 if (owner == class_owners_.end() || owner->second != id) continue;
1044 ssq::Object currentObject = vm_->find(pair.first.c_str());
1045 if (currentObject.getType() == ssq::Type::CLASS &&
1046 objectIdentity(currentObject.getRaw()) == objectIdentity(pair.second.getRaw())) {
1047 sq_pushroottable(handle());
1048 sq_pushstring(handle(), pair.first.c_str(), static_cast<SQInteger>(pair.first.size()));
1049 sq_rawdeleteslot(handle(), -2, SQFalse);
1050 sq_pop(handle(), 1);
1051 }
1052 classes_.erase(pair.first);
1053 class_owners_.erase(owner);
1054 class_identities_.erase(pair.first);
1055 }
1056 record.class_objects.clear();
1057 record.compiled.reset();
1058 record.info.state = ScriptState::Unloaded;
1059 notifyLifecycle(record.info);
1060 return true;
1061 } catch (const std::exception& error) {
1062 record.info.state = ScriptState::Failed;
1063 record.info.error = error.what();
1064 notifyLifecycle(record.info);
1065 fail(ScriptStage::Unload, record.info.source, id, error);
1066 }
1067}

1115size_t Runtime::scanClasses() {
1116 auto scope = enter();
1117 auto stack = guard();
1118 const auto after = rootClasses();
1119 size_t scanned = 0;
1120 for (const auto& pair : after) {
1121 auto old = class_identities_.find(pair.first);
1122 if (old != class_identities_.end() && old->second == pair.second) continue;
1123 try {
1124 ssq::Class cls = vm_->findClass(pair.first.c_str());
1125 std::string source;
1126 auto clsInfo = classes_.find(pair.first);
1127 if (clsInfo != classes_.end()) source = clsInfo->second.source;
1128 if (source.empty()) source = "<runtime scan>";
1129 classes_[pair.first] = inspectClass(pair.first, cls, source);
1130 class_identities_[pair.first] = pair.second;
1131 if (!class_owners_.count(pair.first)) class_owners_[pair.first] = 0;
1132 ++scanned;
1133 } catch (...) {
1134 // Not a script class (e.g. native class): skip.
1135 }
1136 }
1137 // Resolve base names against the current root set (also covers classes
1138 // replaced by hot reload whose base relationship changed).
1139 for (const auto& pair : after) {
1140 auto info = classes_.find(pair.first);
1141 if (info == classes_.end()) continue;
1142 try {
1143 const ssq::Class cls = vm_->findClass(pair.first.c_str());
1144 StackGuard stackGuard(*this);
1145 sq_pushobject(handle(), cls.getRaw());
1146 if (SQ_SUCCEEDED(sq_getbase(handle(), -1)) &&
1147 sq_gettype(handle(), -1) == OT_CLASS) {
1148 HSQOBJECT baseObject;
1149 if (SQ_SUCCEEDED(sq_getstackobj(handle(), -1, &baseObject))) {
1150 const SQUserPointer identity = objectIdentity(baseObject);
1151 for (const auto& candidate : after) {
1152 if (candidate.second == identity) {
1153 info->second.base = candidate.first;
1154 break;
1155 }
1156 }
1157 }
1158 }
1159 } catch (...) {
1160 }
1161 }
1162 return scanned;
1163}

1259void Runtime::discoverClasses(
1260 ScriptRecord& record, const std::unordered_map<std::string, SQUserPointer>& before) {
1261 const auto after = rootClasses();
1262 for (const auto& pair : after) {
1263 auto old = before.find(pair.first);
1264 if (old != before.end() && old->second == pair.second) continue;
1265 ssq::Class cls = vm_->findClass(pair.first.c_str());
1266 record.class_objects.insert_or_assign(pair.first, cls);
1267 if (std::find(record.info.classes.begin(), record.info.classes.end(), pair.first) ==
1268 record.info.classes.end())
1269 record.info.classes.push_back(pair.first);
1270 classes_[pair.first] = inspectClass(pair.first, cls, record.info.source);
1271 class_owners_[pair.first] = record.info.id;
1272 class_identities_[pair.first] = pair.second;
1273 }
1274 std::sort(record.info.classes.begin(), record.info.classes.end());
1275
1276 // Resolve base class names after every newly defined class is registered.
1277 for (const std::string& name : record.info.classes) {
1278 auto info = classes_.find(name);
1279 auto cls = record.class_objects.find(name);
1280 if (info == classes_.end() || cls == record.class_objects.end()) continue;
1281 StackGuard stack(*this);
1282 sq_pushobject(handle(), cls->second.getRaw());
1283 if (SQ_SUCCEEDED(sq_getbase(handle(), -1)) && sq_gettype(handle(), -1) == OT_CLASS) {
1284 HSQOBJECT baseObject;
1285 if (SQ_SUCCEEDED(sq_getstackobj(handle(), -1, &baseObject))) {
1286 const SQUserPointer identity = objectIdentity(baseObject);
1287 for (const auto& candidate : after) {
1288 if (candidate.second == identity) {
1289 info->second.base = candidate.first;
1290 break;
1291 }
1292 }
1293 }
1294 }
