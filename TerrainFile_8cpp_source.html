42eve::Result<DecodedTerrainFile> decodeEvtrn(std::span<const std::uint8_t> bytes) {
43 if (bytes.size() < kEvtrnHeader)
44 return terrainFileFailure(eve::DiagnosticCode::ParseError, "EVTRN payload is truncated", "evtrn");
45 const std::uint32_t width = littleU32(bytes.data() + 8);
46 const std::uint32_t height = littleU32(bytes.data() + 12);
47 const float spacingX = littleF32(bytes.data() + 16);
48 const float spacingZ = littleF32(bytes.data() + 20);
49 const std::uint64_t samples = std::uint64_t(width) * std::uint64_t(height);
50 if (width == 0 || height == 0 || width > kMaximumDimension || height > kMaximumDimension)
51 return terrainFileFailure(eve::DiagnosticCode::InvalidArgument, "EVTRN dimensions are invalid", "evtrn");
52 if (!std::isfinite(spacingX) || !std::isfinite(spacingZ) || spacingX <= 0.f || spacingZ <= 0.f)
53 return terrainFileFailure(eve::DiagnosticCode::InvalidArgument, "EVTRN spacing is invalid", "evtrn");
54 if (samples > (std::numeric_limits<std::uint64_t>::max() - kEvtrnHeader) / 4 ||
55 samples * 4 + kEvtrnHeader != bytes.size())
56 return terrainFileFailure(eve::DiagnosticCode::InvalidArgument,
57 "EVTRN payload size does not match its dimensions", "evtrn");
58
59 DecodedTerrainFile decoded;
60 decoded.heightmap.resize(int(width), int(height));
61 decoded.spacingX = spacingX;
62 decoded.spacingZ = spacingZ;
63 decoded.hasSpacing = true;
64 decoded.format = "evtrn";
65 float minimum = std::numeric_limits<float>::max();
66 float maximum = std::numeric_limits<float>::lowest();
67 std::size_t cursor = kEvtrnHeader;
68 for (std::uint32_t y = 0; y < height; ++y) {
69 for (std::uint32_t x = 0; x < width; ++x) {
70 const float value = littleF32(bytes.data() + cursor);
71 cursor += 4;
72 if (!std::isfinite(value))
73 return terrainFileFailure(eve::DiagnosticCode::ParseError,
74 "EVTRN contains a non-finite height", "evtrn");
75 decoded.heightmap.setHeight(int(x), int(y), value);
76 minimum = std::min(minimum, value);
77 maximum = std::max(maximum, value);
78 }
79 }
80 decoded.minHeight = minimum;
81 decoded.maxHeight = maximum;
82 return eve::Result<DecodedTerrainFile>::success(std::move(decoded));
83}

86eve::Result<DecodedTerrainFile> decodeEvtr(std::span<const std::uint8_t> bytes) {
87 TerrainAsset asset;
88 std::string error;
89 if (!asset.open(bytes.data(), bytes.size(), &error))
90 return terrainFileFailure(eve::DiagnosticCode::ParseError,
91 error.empty() ? "EVTR archive is invalid" : error, "evtr");
92
93 DecodedTerrainFile decoded;
94 decoded.heightmap.resize(asset.getWidth(), asset.getHeight());
95 decoded.minHeight = asset.getMinHeight();
96 decoded.maxHeight = asset.getMaxHeight();
97 // EVTR stores no metres-per-cell, so the referencing level owns the spacing.
98 decoded.hasSpacing = false;
99 decoded.format = "evtr";
100
101 const int chunkSize = asset.getChunkSize();
102 for (const TerrainChunkEntry &entry : asset.chunks()) {
103 TerrainChunkData chunk;
104 error.clear();
105 if (!asset.loadChunk(entry.chunkX, entry.chunkY, chunk, &error))
106 return terrainFileFailure(eve::DiagnosticCode::ParseError,
107 error.empty() ? "EVTR chunk failed to decode" : error, "evtr");
108 for (int y = 0; y < chunk.height; ++y) {
109 for (int x = 0; x < chunk.width; ++x) {
110 const int gx = entry.chunkX * chunkSize + x;
111 const int gy = entry.chunkY * chunkSize + y;
112 if (!decoded.heightmap.inBounds(gx, gy)) continue;
113 decoded.heightmap.setHeight(gx, gy, chunk.heights.height(x, y));
114 }
115 }
116 }
117 return eve::Result<DecodedTerrainFile>::success(std::move(decoded));
118}

122TerrainFileFormat parseTerrainFileFormat(std::string_view name) noexcept {
123 if (name.empty()) return TerrainFileFormat::Auto;
124 std::string lowered(name);
125 std::transform(lowered.begin(), lowered.end(), lowered.begin(),
126 [](unsigned char c) { return char(std::tolower(c)); });
127 if (lowered == "evtr") return TerrainFileFormat::Evtr;
128 if (lowered == "evtrn") return TerrainFileFormat::Evtrn;
129 return TerrainFileFormat::Auto;
130}

132eve::Result<DecodedTerrainFile> decodeTerrainFile(std::span<const std::uint8_t> bytes,
133 TerrainFileFormat format) {
134 if (bytes.empty())
135 return terrainFileFailure(eve::DiagnosticCode::InvalidArgument, "terrain file is empty", "terrain");
136
137 if (format == TerrainFileFormat::Auto) {
138 if (startsWith(bytes, kEvtrnMagic, sizeof(kEvtrnMagic))) format = TerrainFileFormat::Evtrn;
139 else if (startsWith(bytes, reinterpret_cast<const std::uint8_t *>("EVTR"), 4)) format = TerrainFileFormat::Evtr;
140 else
141 return terrainFileFailure(eve::DiagnosticCode::Unsupported,
142 "unrecognised terrain file magic (expected EVTR or EVTRN)", "terrain");
143 }
144 if (format == TerrainFileFormat::Evtrn) {
145 if (!startsWith(bytes, kEvtrnMagic, sizeof(kEvtrnMagic)))
146 return terrainFileFailure(eve::DiagnosticCode::ParseError,
147 "EVTRN magic does not match the requested format", "evtrn");
148 return decodeEvtrn(bytes);
149 }
150 return decodeEvtr(bytes);
151}

162 return decodeTerrainFile(std::span<const std::uint8_t>(bytes.data(), bytes.size()), format);
