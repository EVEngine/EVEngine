13EditorResult<PropertyEditIntent> makeIntent(const PropertySchema& schema, const SelectionSnapshot& selection,
14 const PropertyPath& path, const EditorValue& value, PropertySetMode mode,
15 bool runtime) {
16 auto descriptor = schema.find(path);
17 if (!descriptor)
18 return eve::editing::failed<PropertyEditIntent>(EditorStatus::NotFound, RuleId("editor.property.not-found"),
19 "Property is not present in the schema: " + path.value());
20 if (runtime && (!hasPropertyFlag(descriptor->flags, PropertyFlag::Runtime) ||
21 hasPropertyFlag(descriptor->flags, PropertyFlag::EditorOnly)))
22 return eve::editing::failed<PropertyEditIntent>(EditorStatus::Rejected, RuleId("editor.property.runtime-hidden"),
23 "Property is unavailable in runtime editing");
24 EditorResult<void> validation = validatePropertyValue(*descriptor, value);
25 if (!validation.ok()) return EditorResult<PropertyEditIntent>::failure(validation.status());
26
27 EditorValue::Object payload;
28 payload["path"] = path.value();
29 payload["value"] = value;
30 payload["selection"] = selectionValue(selection);
31 switch (mode) {
32 case PropertySetMode::Absolute: payload["mode"] = "absolute"; break;
33 case PropertySetMode::Relative: payload["mode"] = "relative"; break;
34 case PropertySetMode::Reset: payload["mode"] = "reset"; break;
35 }
36 return eve::editing::applied<PropertyEditIntent>(
37 PropertyEditIntent{CommandId("editor.property.set"), EditorValue(std::move(payload))});
38}

63 const IPropertyProvider& provider,

66 presentation.typeId = schema.typeId;
67 presentation.schemaVersion = schema.version;

70 if (!hasPropertyFlag(descriptor.flags, PropertyFlag::Runtime) ||
