25Result<PropertyEditIntent> makeIntent(const PropertySchema& schema, const SelectionSnapshot& selection,
26 const PropertyPath& path, const EditorValue& value, PropertySetMode mode,
27 bool runtime) {
28 auto descriptor = schema.find(path);
29 if (!descriptor)
30 return eve::editing::failed<PropertyEditIntent>(EditorStatus::NotFound, RuleId("editor.property.not-found"),
31 "Property is not present in the schema: " + path.value());
32 if (runtime && (!hasPropertyFlag(descriptor->flags, PropertyFlag::Runtime) ||
33 hasPropertyFlag(descriptor->flags, PropertyFlag::EditorOnly)))
34 return eve::editing::failed<PropertyEditIntent>(EditorStatus::Rejected,
35 RuleId("editor.property.runtime-hidden"),
36 "Property is unavailable in runtime editing");
37 Result<void> validation = validatePropertyValue(*descriptor, value);
38 if (!validation.ok()) return Result<PropertyEditIntent>::failure(validation.status());
39
40 EditorValue::Object payload;
41 payload["path"] = path.value();
42 payload["value"] = value;
43 payload["selection"] = selectionValue(selection);
44 switch (mode) {
45 case PropertySetMode::Absolute: payload["mode"] = "absolute"; break;
46 case PropertySetMode::Relative: payload["mode"] = "relative"; break;
47 case PropertySetMode::Reset: payload["mode"] = "reset"; break;
48 }
49 return eve::editing::applied<PropertyEditIntent>(
50 PropertyEditIntent{CommandId("editor.property.set"), EditorValue(std::move(payload))});
51}

119Result<RuntimeEditorPackage> RuntimeEditorPublisher::publish(const RuntimeEditorManifest& manifest,
120 const EditorCommandService& commands,
121 const EditorExtensionRegistry& extensions,
122 const MemoryAssetDatabase& assets) const {
123 constexpr HostFeature forbidden = HostFeature::SourceAssets | HostFeature::ArbitraryScript |
124 HostFeature::BuildCook | HostFeature::SourceControl | HostFeature::DebugOverride;
125 if ((manifest.features & forbidden) != HostFeature::None)
126 return manifestError("editor.runtime.forbidden-feature",
127 "Runtime editor manifest requests a developer-only feature");
128
129 RuntimeEditorPackage package;
130 package.profile = HostProfile::runtimeBuilder();
131 package.profile.setFeatures(manifest.features);
132 package.profile.setMaxPayloadBytes(manifest.maxPayloadBytes);
133 for (const CommandId& id : manifest.commands) {
134 const CommandDescriptor* descriptor = commands.find(id);
135 if (!descriptor)
136 return manifestError("editor.runtime.command-not-found",
137 "Manifest command is not registered: " + id.value());
138 if (!descriptor->automationAllowed)
139 return manifestError("editor.runtime.command-not-publishable",
140 "Manifest command is not publishable: " + id.value());
141 if (!package.profile.hasFeatures(descriptor->requiredFeatures))
142 return manifestError("editor.runtime.command-feature-missing",
143 "Manifest omits a feature required by command: " + id.value());
144 package.profile.allowCommand(id);
145 package.commands.push_back(*descriptor);
146 }
147
148 HostProfile audienceProfile = HostProfile::runtimeBuilder();
149 audienceProfile.setAllowAllCommands(true);
150 for (const auto& descriptor : extensions.tools(audienceProfile))
151 if (selected(manifest.tools, descriptor.id)) package.tools.push_back(descriptor);
152 for (const ToolId& id : manifest.tools)
153 if (std::none_of(package.tools.begin(), package.tools.end(), [&](const auto& value) { return value.id == id; }))
154 return manifestError("editor.runtime.tool-not-found", "Manifest tool is unavailable: " + id.value());
155 for (const auto& descriptor : extensions.palettes(audienceProfile))
156 if (selected(manifest.palettes, descriptor.id)) package.palettes.push_back(descriptor);
157 for (const StableId& id : manifest.palettes)
158 if (std::none_of(package.palettes.begin(), package.palettes.end(),
159 [&](const auto& value) { return value.id == id; }))
160 return manifestError("editor.runtime.palette-not-found", "Manifest palette is unavailable: " + id.value());
161 for (const auto& descriptor : extensions.rules(audienceProfile))
162 if (selected(manifest.rules, descriptor.id)) package.rules.push_back(descriptor);
163 for (const RuleId& id : manifest.rules)
164 if (std::none_of(package.rules.begin(), package.rules.end(), [&](const auto& value) { return value.id == id; }))
165 return manifestError("editor.runtime.rule-not-found", "Manifest rule is unavailable: " + id.value());
166
167 std::deque<AssetGuid> pending(manifest.rootAssets.begin(), manifest.rootAssets.end());
168 std::unordered_set<AssetGuid, StrongEditorIdHash<AssetGuid>> visited;
169 while (!pending.empty()) {
170 AssetGuid current = pending.front();
171 pending.pop_front();
172 if (!visited.emplace(current).second) continue;
173 if (!assets.find(current).ok())
174 return manifestError("editor.runtime.asset-not-found", "Manifest asset is unavailable: " + current.value());
175 package.assetClosure.push_back(current);
176 for (const AssetDependency& dependency : assets.dependencies(current))
177 if (dependency.kind != DependencyKind::EditorOnly && dependency.kind != DependencyKind::Source)
178 pending.push_back(dependency.to);
179 }
180 std::sort(package.assetClosure.begin(), package.assetClosure.end());
181 return eve::editing::applied<RuntimeEditorPackage>(std::move(package));
182}

355Result<EditorValue> EditorSession::executeCommand(const CommandId& id, const EditorValue& payload,
356 CommandSource source) {
357 if (!commandService_)
358 return eve::editing::failed<EditorValue>(EditorStatus::Failed, RuleId("editor.command.missing-service"),
359 "Editor session has no command service");
360
361 const CommandDescriptor* descriptor = commandService_->find(id);
362 const bool ownsTransaction = descriptor && descriptor->createsTransaction && !transactions_.isActive();
363 if (ownsTransaction) {
364 auto begun =
365 transactions_.beginTransaction(descriptor->displayName.empty() ? id.value() : descriptor->displayName);
366 if (!begun.ok()) return Result<EditorValue>::failure(begun.status());
367 }
368
369 CommandContext commandContext;
370 commandContext.session = this;
371 commandContext.profile = &hostProfile_;
372 commandContext.source = source;
373 if (IEditableTarget* currentTarget = target()) {
374 commandContext.target = TargetId(currentTarget->targetId());
375 commandContext.targetRevision = currentTarget->revision();
376 }
377
378 Result<EditorValue> result = commandService_->execute(id, commandContext, payload);
379 if (ownsTransaction) {
380 if (result.ok()) {
381 auto committed = transactions_.commitTransaction();
382 if (!committed.ok()) {
383 auto discarded = transactions_.rollbackTransaction();
384 if (!discarded.ok()) {
385 std::vector<EditorDiagnostic> diagnostics = committed.diagnostics();
386 appendProjectedDiagnostics(diagnostics, discarded.status());
387 return Result<EditorValue>::failure(eve::Status(EditorStatus::Failed, std::move(diagnostics)));
388 }
389 return Result<EditorValue>::failure(committed.status());
390 }
391 } else {
392 auto discarded = transactions_.rollbackTransaction();
393 if (!discarded.ok()) {
394 std::vector<EditorDiagnostic> diagnostics = result.diagnostics();
395 appendProjectedDiagnostics(diagnostics, discarded.status());
396 return Result<EditorValue>::failure(eve::Status(EditorStatus::Failed, std::move(diagnostics)));
397 }
398 }
399 }
400 return result;
401}

432 if (!commandService_)

448Result<TransactionReceipt> EditorSession::executeCommandReceipt(const CommandId& id, const EditorValue& payload,
449 CommandSource source) {
450 TransactionReceipt receipt;
451 receipt.id = TransactionId((sessionId_.empty() ? std::string("editor.session") : sessionId_.value()) +
452 ".transaction." + std::to_string(++receiptSequence_));
453 IEditableTarget* currentTarget = target();
454 receipt.beforeRevision = currentTarget ? currentTarget->revision() : 0;
455 Result<EditorValue> command = executeCommand(id, payload, source);
456 currentTarget = target();
457 receipt.afterRevision = currentTarget ? currentTarget->revision() : receipt.beforeRevision;
458 receipt.diagnostics = command.diagnostics();
459 receipt.state = command.ok() ? TransactionState::Committed
460 : (command.code() == EditorStatus::Conflict ? TransactionState::Conflicted
461 : TransactionState::Rejected);
462 if (!command.ok()) return Result<TransactionReceipt>::failure(command.status());
463 return Result<TransactionReceipt>::success(std::move(receipt), eve::Status(command.code(), command.diagnostics()));
464}

646bool EditorSession::activateTool(const std::string& id) {
647 if (id.empty()) {
648 deactivateCurrent();
649 return true;
650 }
651 IEditorTool* next = findTool(id);
652 if (!next) return false;
653 if (next == activeTool_) return true;
654 deactivateCurrent();
655 next = findTool(id);
656 if (!next) return false;
657 activeTool_ = next;
658 ++toolGeneration_;
659 try {
660 activeTool_->activate(context_);
661 } catch (...) {
662 if (activeTool_ == next) {
663 activeTool_ = nullptr;
664 capturedPointerId_ = -1;
665 ++toolGeneration_;
666 }
667 throw;
668 }
669 return true;

674ToolResponse EditorSession::dispatchPointer(const EditorPointerEvent& event) {
675 if (!activeTool_) return ToolResponse::ignored();
676 if (capturedPointerId_ >= 0 && event.pointerId != capturedPointerId_) {
677 return ToolResponse::ignored();
678 }
679 IEditorTool* const dispatchedTool = activeTool_;
680 const std::uint64_t generation = toolGeneration_;
681 ToolResponse response = dispatchedTool->pointerEvent(context_, event);
682 if (activeTool_ != dispatchedTool || toolGeneration_ != generation) return {response.handled, false, false};
683 if (response.capturePointer && capturedPointerId_ < 0) capturedPointerId_ = event.pointerId;
684 if (response.releasePointer && capturedPointerId_ == event.pointerId) capturedPointerId_ = -1;
685 if (event.phase == EditorPointerEvent::Phase::Cancel && capturedPointerId_ == event.pointerId) {
686 capturedPointerId_ = -1;
687 }
688 return response;
689}

695void EditorSession::update(float dt) {
696 if (activeTool_) activeTool_->update(context_, dt);
697 if (!documentService_ || activeDocument_.empty() || dt <= 0.f) return;
698
699 if (externalPollInterval_ > 0.f) {
700 externalPollElapsed_ += dt;
701 if (externalPollElapsed_ >= externalPollInterval_) (void)pollDocumentChanges();
702 }
703
704 Result<DocumentSnapshot> snapshot = documentService_->snapshot(activeDocument_);
705 if (!snapshot.ok() || !snapshot.value().dirty()) {
706 autosaveElapsed_ = 0.f;
707 return;
708 }
709 if (!autosaveService_ || autosaveInterval_ <= 0.f || snapshot.value().revision.edit == lastAutosavedRevision_)
710 return;
711 autosaveElapsed_ += dt;
712 if (autosaveElapsed_ >= autosaveInterval_) (void)autosaveDocument();
713}

728void EditorSession::deactivateCurrent() {
729 IEditorTool* current = activeTool_;
730 activeTool_ = nullptr;
731 capturedPointerId_ = -1;
732 if (!current) return;
733 ++toolGeneration_;
734
735 std::exception_ptr failure;
736 try {
737 current->cancel(context_);
738 } catch (...) {
739 failure = std::current_exception();
740 }
741 try {
742 current->deactivate(context_);
743 } catch (...) {
744 if (!failure) failure = std::current_exception();
745 }
746 if (failure) std::rethrow_exception(failure);
747}
