21Result<VegetationSceneAssociation> VegetationSceneAssociationLoader::load(
22 const AssetRef& sceneTemplate, const asset::EvpackCapabilities& capabilities) const {
23 auto scene = asset_scene::EvpackSceneTemplateLoader(reader_).load(sceneTemplate, capabilities);
24 if (!scene) return Result<VegetationSceneAssociation>::failure(scene.status());
25 if (scene.value().sourceGuid.empty())
26 return fail<VegetationSceneAssociation>(DiagnosticCode::Unsupported,
27 "legacy Scene Template has no source GUID for manager association",
28 sceneTemplate.format());
29 auto candidates = reader_.listAssets("eve.vegetation-scene/1", capabilities, 4096);
30 if (!candidates) return Result<VegetationSceneAssociation>::failure(candidates.status());
31 std::optional<LoadedVegetationScene> match;
32 const auto expected = folded(scene.value().sourceGuid);
33 for (const auto& candidate : candidates.value()) {
34 auto manager = EvpackVegetationSceneLoader(reader_).load(candidate, capabilities);
35 if (!manager) return Result<VegetationSceneAssociation>::failure(manager.status());
36 if (folded(manager.value().sourceGuid) != expected) continue;
37 if (match)
38 return fail<VegetationSceneAssociation>(DiagnosticCode::Conflict,
39 "Scene Template matches multiple vegetation managers",
40 sceneTemplate.format());
41 match.emplace(std::move(manager).takeValue());
42 }
43 if (!match)
44 return fail<VegetationSceneAssociation>(DiagnosticCode::NotFound,
45 "Scene Template has no vegetation manager",
46 sceneTemplate.format());
47 auto masks = loadVegetationSceneElementMasks(reader_, *match, capabilities);
48 if (!masks) return Result<VegetationSceneAssociation>::failure(masks.status());
49 return Result<VegetationSceneAssociation>::success(
50 {std::move(scene).takeValue(), std::move(*match), std::move(masks).takeValue()});
51}

53Result<std::unique_ptr<VegetationSceneInstance>> VegetationSceneInstance::create(
54 const asset::EvpackResourceReader& reader, graphics::IResourceFactory& factory,
55 const AssetRef& sceneTemplate, const asset::EvpackCapabilities& capabilities,
56 graphics::PbrSurface baseSurface, graphics::VegetationMotion baseMotion,
57 const VegetationSceneGpuBuild& build, std::uint64_t owner) {
58 auto association = VegetationSceneAssociationLoader(reader).load(sceneTemplate, capabilities);
59 if (!association) return Result<std::unique_ptr<VegetationSceneInstance>>::failure(association.status());
60 auto runtime = VegetationSceneGpuRuntime::create(factory, baseSurface, baseMotion, association.value().manager,
61 association.value().masks, build);
62 if (!runtime) return Result<std::unique_ptr<VegetationSceneInstance>>::failure(runtime.status());
63 auto live = VegetationSceneLiveElements::create(association.value().manager, owner);
64 if (!live) return Result<std::unique_ptr<VegetationSceneInstance>>::failure(live.status());
65 try {
66 auto result = std::unique_ptr<VegetationSceneInstance>(new VegetationSceneInstance(
67 std::move(association).takeValue(), std::move(baseSurface), std::move(baseMotion),
68 std::move(runtime).takeValue(), std::move(live).takeValue()));
69 return Result<std::unique_ptr<VegetationSceneInstance>>::success(std::move(result));
70 } catch (const std::bad_alloc&) {
71 return fail<std::unique_ptr<VegetationSceneInstance>>(
72 DiagnosticCode::Failed, "vegetation scene instance allocation failed");
73 }
74}
