39EditorResult<void> EditorProfilerModel::ingest(EditorProfilerFrame frame) {
40 if (frame.sequence == 0 || !finiteNonNegative(frame.cpuFrameMs) ||
41 !finiteNonNegative(frame.gpuFrameMs) || frame.zones.size() > 100000) {
42 return failure<void>(EditorStatus::Rejected, "editor.profiler.invalid-frame",
43 "Profiler frame shape or timing is invalid");
44 }
45 if (!history_.empty() && frame.sequence <= history_.back().sequence) {
46 return failure<void>(EditorStatus::Conflict, "editor.profiler.stale-frame",
47 "Profiler frame sequence is stale");
48 }
49 for (const auto& zone : frame.zones) {
50 if (zone.module.empty() || zone.name.empty() || zone.thread.empty() ||
51 !finiteNonNegative(zone.selfMs) || !finiteNonNegative(zone.totalMs) ||
52 zone.selfMs > zone.totalMs || zone.count < 1 || zone.depth < 0) {
53 return failure<void>(EditorStatus::Rejected, "editor.profiler.invalid-zone",
54 "Profiler zone identity or timing is invalid");
55 }
56 }
57 history_.push_back(std::move(frame));
58 while (history_.size() > budgets_.historyFrames) history_.pop_front();
59 return eve::editing::applied<void>();
60}

64std::vector<EditorProfilerZone> EditorProfilerModel::query(
65 const EditorProfilerQuery& options) const {
66 std::vector<EditorProfilerZone> result;
67 if (history_.empty() || options.limit == 0) return result;
68 for (const auto& zone : history_.back().zones) {
69 if (contains(zone.module, options.module) && contains(zone.thread, options.thread) &&
70 (contains(zone.name, options.text) || contains(zone.module, options.text))) {
71 result.push_back(zone);
72 }
73 }
74 const auto less = [&](const EditorProfilerZone& left, const EditorProfilerZone& right) {
75 switch (options.sort) {
76 case EditorProfilerSort::SelfTime: return left.selfMs < right.selfMs;
77 case EditorProfilerSort::TotalTime: return left.totalMs < right.totalMs;
78 case EditorProfilerSort::Count: return left.count < right.count;
79 case EditorProfilerSort::Name:
80 return std::tie(left.module, left.name, left.thread) <
81 std::tie(right.module, right.name, right.thread);
82 }
83 return false;
84 };
85 std::stable_sort(result.begin(), result.end(), [&](const auto& left, const auto& right) {
86 return options.descending ? less(right, left) : less(left, right);
87 });
88 if (result.size() > options.limit) result.resize(options.limit);
89 return result;
90}

92std::vector<EditorProfilerModuleSummary> EditorProfilerModel::moduleSummary() const {
93 std::map<std::string, EditorProfilerModuleSummary> grouped;
94 if (!history_.empty()) {
95 for (const auto& zone : history_.back().zones) {
96 auto& summary = grouped[zone.module];
97 summary.module = zone.module;
98 summary.selfMs += zone.selfMs;
99 summary.totalMs += zone.totalMs;
100 summary.count += zone.count;
101 ++summary.zones;
102 }
103 }
104 std::vector<EditorProfilerModuleSummary> result;
105 for (auto& [name, summary] : grouped) result.push_back(std::move(summary));
106 std::stable_sort(result.begin(), result.end(), [](const auto& left, const auto& right) {
107 if (left.selfMs != right.selfMs) return left.selfMs > right.selfMs;
108 return left.module < right.module;
109 });
110 return result;

113std::vector<EditorDiagnostic> EditorProfilerModel::diagnostics() const {
114 std::vector<EditorDiagnostic> result;
115 if (history_.empty()) return result;
116 const auto& frame = history_.back();
117 if (budgets_.cpuFrameMs > 0.0 && frame.cpuFrameMs > budgets_.cpuFrameMs) {

120 DiagnosticSeverity::Warning, "CPU frame time exceeds the editor budget"));
121 }
122 if (frame.gpuTimingAvailable && budgets_.gpuFrameMs > 0.0 &&
123 frame.gpuFrameMs > budgets_.gpuFrameMs) {
124 result.push_back(eve::editing::ruleDiagnostic(
125 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.profiler.gpu-budget"),
126 DiagnosticSeverity::Warning, "GPU frame time exceeds the editor budget"));
127 }
128 if (budgets_.zoneSelfMs > 0.0) {
129 for (const auto& zone : frame.zones) {
130 if (zone.selfMs > budgets_.zoneSelfMs) {
131 result.push_back(eve::editing::ruleDiagnostic(
132 eve::DiagnosticCode::PreconditionViolation,
133 RuleId("editor.profiler.zone-budget"), DiagnosticSeverity::Warning,
134 "Profiler zone exceeds its self-time budget: " + zone.module + "/" +
135 zone.name));
136 }
137 }
138 }
139 return result;
