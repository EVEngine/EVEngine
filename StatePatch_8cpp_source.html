71std::string quote(const std::string& value) {
72 std::ostringstream out;
73 out << '"';
74 for (unsigned char c : value) {
75 switch (c) {
76 case '"': out << "\\\""; break;
77 case '\\': out << "\\\\"; break;
78 case '\b': out << "\\b"; break;
79 case '\f': out << "\\f"; break;
80 case '\n': out << "\\n"; break;
81 case '\r': out << "\\r"; break;
82 case '\t': out << "\\t"; break;
83 default:
84 if (c < 0x20)
85 out << "\\u" << std::hex << std::setw(4) << std::setfill('0') << int(c) << std::dec;
86 else
87 out << static_cast<char>(c);
88 }
89 }
90 return out.str() + '"';
91}

93std::string canonicalJson(const eve::json::Value& value) {
94 if (value.isNull()) return "null";
95 if (value.isBool()) return value.asBool() ? "true" : "false";
96 if (value.isNumber()) return value.asString();
97 if (value.isString()) return quote(value.asString());
98 if (value.isArray()) {
99 std::string out = "[";
100 for (size_t i = 0; i < value.size(); ++i) {
101 if (i) out += ',';
102 out += canonicalJson(value.at(i));
103 }
104 return out + ']';
105 }
106 if (value.isObject()) {
107 auto keys = value.keys();
108 std::sort(keys.begin(), keys.end());
109 std::string out = "{";
110 for (size_t i = 0; i < keys.size(); ++i) {
111 if (i) out += ',';
112 out += quote(keys[i]) + ':' + canonicalJson(value.get(keys[i].c_str()));
113 }
114 return out + '}';
115 }
116 return {};
117}

232bool Store::commit(PatchBatch* batch) {
233 if (!batch) return false;
234 batch->result_ = {};
235 batch->result_.revisionBefore = revision_;
236 batch->result_.revisionAfter = revision_;
237 Values staged = values_;
238 for (size_t i = 0; i < batch->operations_.size(); ++i) {
239 const auto& operation = batch->operations_[i];
240 auto fail = [&](const std::string& code, const std::string& message) {
241 batch->result_.errors.push_back({static_cast<int>(i), operation.subject, operation.key, code, message});
242 };
243 if (operation.subject.empty()) fail("invalid_subject", "subject must not be empty");
244 if (operation.key.empty()) fail("invalid_key", "key must not be empty");
245 if (!operation.inputError.empty()) fail("invalid_json", operation.inputError);
246 if (!batch->result_.errors.empty() && batch->result_.errors.back().operationIndex == static_cast<int>(i))
247 continue;
248 const auto subjectIt = staged.find(operation.subject);
249 const auto valueIt = subjectIt == staged.end() ? std::map<std::string, Value>::const_iterator{}
250 : subjectIt->second.find(operation.key);
251 const bool exists = subjectIt != staged.end() && valueIt != subjectIt->second.end();
252 if (operation.expected && (!exists || valueIt->second.json != *operation.expected)) {
253 fail("conflict", "current value does not match expected JSON");
254 continue;
255 }
256 if (operation.remove) {
257 if (exists) {
258 staged[operation.subject].erase(operation.key);
259 if (staged[operation.subject].empty()) staged.erase(operation.subject);
260 }
261 } else {
262 staged[operation.subject][operation.key].json = operation.value;
263 }
264 }
265 if (!batch->result_.errors.empty()) return false;

267 struct Pending {
268 std::string subject;
269 std::string key;
270 std::string oldJson;
271 std::string newJson;
272 bool removed;
273 };
274 std::vector<Pending> changes;
275 std::set<std::pair<std::string, std::string>> keys;
276 for (const auto& [subject, entries] : values_)
277 for (const auto& [key, value] : entries) keys.emplace(subject, key);
278 for (const auto& [subject, entries] : staged)
279 for (const auto& [key, value] : entries) keys.emplace(subject, key);
280 for (const auto& [subject, key] : keys) {
281 const auto oldSubject = values_.find(subject);
282 const auto newSubject = staged.find(subject);
283 const auto oldValue =
284 oldSubject == values_.end() ? std::map<std::string, Value>::const_iterator{} : oldSubject->second.find(key);
285 const auto newValue =
286 newSubject == staged.end() ? std::map<std::string, Value>::const_iterator{} : newSubject->second.find(key);
287 const bool had = oldSubject != values_.end() && oldValue != oldSubject->second.end();
288 const bool has = newSubject != staged.end() && newValue != newSubject->second.end();
289 if (had != has || (had && oldValue->second.json != newValue->second.json))
290 changes.push_back({subject, key, had ? oldValue->second.json : "", has ? newValue->second.json : "", !has});
291 }
292 if (!changes.empty()) {
293 if (revision_ == std::numeric_limits<uint64_t>::max() ||
294 nextSequence_ > std::numeric_limits<uint64_t>::max() - changes.size()) {
295 batch->result_.errors.push_back({-1, {}, {}, "revision_exhausted", "revision or event sequence exhausted"});
296 return false;
297 }
298 ++revision_;
299 for (const auto& change : changes) {
300 if (!change.removed) staged[change.subject][change.key].revision = revision_;
301 dirty_.emplace(change.subject, change.key);
302 events_.push_back({nextSequence_++, revision_, change.subject, change.key, change.oldJson, change.newJson,
303 change.removed});
304 }
305 values_ = std::move(staged);
306 }
307 batch->result_.success = true;
308 batch->result_.changedCount = static_cast<int>(changes.size());
309 batch->result_.revisionAfter = revision_;
310 return true;
311}

404eve::Result<void> Store::restoreJson(const std::string& json) {
405 std::string parseError;
406 auto document = eve::json::Document::parse(json, &parseError);
407 if (!document.valid())
408 return restoreFailure(parseError.empty() ? "invalid state patch snapshot" : std::move(parseError), "$");
409
410 const auto root = document.root();
411 if (!root.isObject()) return restoreFailure("snapshot root must be an object", "$");
412
413 uint64_t restoredRevision = 0;
414 uint64_t restoredSequence = 0;
415 if (root.getInt("version") != 1) return restoreFailure("unsupported state patch snapshot version", "$.version");
416 if (!parseU64(root.get("revision"), restoredRevision))
417 return restoreFailure("snapshot revision must be a decimal string", "$.revision");
418 if (!parseU64(root.get("nextSequence"), restoredSequence) || restoredSequence == 0)
419 return restoreFailure("snapshot nextSequence must be a non-zero decimal string", "$.nextSequence");
420
421 const auto values = root.get("values");
422 const auto dirty = root.get("dirty");
423 if (!values.isArray() || !dirty.isArray()) {
424 return restoreFailure("values and dirty must be arrays", "$");
425 }
426 Values restoredValues;
427 std::set<std::pair<std::string, std::string>> restoredDirty;
428 for (size_t i = 0; i < values.size(); ++i) {
429 const auto item = values.at(i);
430 uint64_t valueRevision = 0;
431 if (!item.isObject() || item.getString("subject").empty() || item.getString("key").empty() ||
432 !parseU64(item.get("revision"), valueRevision) || valueRevision > restoredRevision || !item.get("value")) {
433 return restoreFailure("invalid value at index " + std::to_string(i), "$.values[" + std::to_string(i) + "]");
434 }
435 const auto subject = item.getString("subject");
436 const auto key = item.getString("key");
437 if (restoredValues[subject].contains(key)) {
438 return restoreFailure("duplicate subject and key", "$.values[" + std::to_string(i) + "]");
439 }
440 restoredValues[subject][key] = {canonicalJson(item.get("value")), valueRevision};
441 }
442 for (size_t i = 0; i < dirty.size(); ++i) {
443 const auto item = dirty.at(i);
444 if (!item.isArray() || item.size() != 2 || !item.at(0).isString() || !item.at(1).isString() ||
445 item.at(0).asString().empty() || item.at(1).asString().empty()) {
446 return restoreFailure("invalid dirty key at index " + std::to_string(i),
447 "$.dirty[" + std::to_string(i) + "]");
448 }
449 restoredDirty.emplace(item.at(0).asString(), item.at(1).asString());
450 }
451 values_ = std::move(restoredValues);
452 dirty_ = std::move(restoredDirty);
453 revision_ = restoredRevision;
454 nextSequence_ = restoredSequence;
455 events_.clear();
456 query_.clear();
457 dirtyQuery_.clear();
458 batches_.clear();
459 return eve::Result<void>::success();
460}

469eve::Result<void> Store::restoreSnapshot(const eve::SnapshotEnvelope& source,
470 const eve::SnapshotHashProvider& hashProvider) {
471 if (source.type != "statepatch.store" || source.schema != statePatchSchema())
472 return snapshotFailure<void>(eve::DiagnosticCode::InvalidArgument,
473 "snapshot does not belong to statepatch::Store");
474 if (!instanceId_.isNil() && source.instanceId != instanceId_)
475 return snapshotFailure<void>(eve::DiagnosticCode::Conflict,
476 "snapshot instanceId does not match statepatch::Store");
477 auto migrated = statePatchMigrations().migrate(source, eve::SchemaVersion(1), hashProvider);
478 if (!migrated.ok()) return eve::Result<void>::failure(migrated.status());
479 const auto& candidateEnvelope = migrated.value();
480 auto metadata = eve::validateSnapshotPayloadMetadata(candidateEnvelope.payload, candidateEnvelope.revision,
481 candidateEnvelope.tick);
482 if (!metadata.ok()) return eve::Result<void>::failure(metadata.status());
483 auto payload = candidateEnvelope.payload.toJson();
484 if (!payload.ok()) return eve::Result<void>::failure(payload.status());
485
486 Store candidate(instanceId_);
487 auto restored = candidate.restoreJson(std::move(payload).takeValue());
488 if (!restored.ok()) return eve::Result<void>::failure(restored.status());
489 if (candidate.revision_ != candidateEnvelope.revision.value())
490 return snapshotFailure<void>(eve::DiagnosticCode::Conflict,
491 "state patch payload revision disagrees with snapshot envelope");
492 candidate.instanceId_ = candidateEnvelope.instanceId;
493 candidate.tick_ = candidateEnvelope.tick;
494 *this = std::move(candidate);
495 return eve::Result<void>::success();
496}

549eve::Result<void> patchParticipantFailure(const PatchResult& result) {
550 eve::StatusCode statusCode = eve::StatusCode::Failed;
551 eve::DiagnosticCode diagnosticCode = eve::DiagnosticCode::Failed;
552 std::vector<eve::Diagnostic> diagnostics;
553 for (const auto& error : result.errors) {
554 if (error.code == "conflict") {
555 statusCode = eve::StatusCode::Conflict;
556 diagnosticCode = eve::DiagnosticCode::Conflict;
557 } else if (statusCode == eve::StatusCode::Failed && error.code.starts_with("invalid_")) {
558 statusCode = eve::StatusCode::Rejected;
559 diagnosticCode = eve::DiagnosticCode::InvalidArgument;
560 }
561 diagnostics.push_back(eve::Diagnostic::error(
562 diagnosticCode, error.message.empty() ? "state patch batch was rejected" : error.message,
563 "statepatch[" + std::to_string(error.operationIndex) + "]"));
564 }
565 if (diagnostics.empty())
566 diagnostics.push_back(eve::Diagnostic::error(eve::DiagnosticCode::Failed,
567 "state patch batch could not be prepared", "statepatch"));
568 return eve::Result<void>::failure(eve::Status(statusCode, std::move(diagnostics)));
569}

592eve::Result<void> StoreTransactionParticipant::prepare(const transaction::TransactionContext& context) {
593 if (phase_ != Phase::Idle) return lifecycleFailure("prepare");
594 if (context.transactionId().empty())
595 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
596 "transaction id must not be empty", "transactionId"));
597
598 transactionId_ = context.transactionId();
599 correlationId_ = context.correlationId();
600 causationId_ = context.causationId();
601 before_ = std::make_unique<Store>();
602 before_->copyTransactionStateFrom(store_);
603 prepared_ = std::make_unique<Store>();
604 prepared_->copyTransactionStateFrom(*before_);
605
606 if (!prepared_->commit(&batch_)) {
607 auto result = patchParticipantFailure(batch_.result());
608 before_.reset();
609 prepared_.reset();
610 phase_ = Phase::Failed;
611 return result;
612 }
613
614 expectedAfter_ = std::make_unique<Store>();
615 expectedAfter_->copyTransactionStateFrom(*prepared_);
616 phase_ = Phase::Prepared;
617 return eve::Result<void>::success();
618}
619
620eve::Result<void> StoreTransactionParticipant::commit(const transaction::TransactionContext& context) {
621 if (phase_ != Phase::Prepared) return lifecycleFailure("commit");
622 if (!contextMatches(context)) return contextFailure(context);
623 if (!before_ || !prepared_ || !expectedAfter_ || !store_.transactionStateEquals(*before_))
624 return eve::Result<void>::failure(eve::Diagnostic::error(
625 eve::DiagnosticCode::Conflict, "state patch store changed after prepare", "statepatch.store"));
626
627 store_.swapTransactionState(*prepared_);
628 phase_ = Phase::Committed;
629 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
630}
631
632eve::Result<void> StoreTransactionParticipant::rollback(const transaction::TransactionContext& context) {
633 if (phase_ != Phase::Prepared) return lifecycleFailure("rollback");
634 if (!contextMatches(context)) return contextFailure(context);
635 before_.reset();
636 prepared_.reset();
637 expectedAfter_.reset();
638 phase_ = Phase::RolledBack;
639 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
640}
641
642eve::Result<void> StoreTransactionParticipant::compensate(const transaction::TransactionContext& context) {
643 if (phase_ != Phase::Committed) return lifecycleFailure("compensate");
644 if (!contextMatches(context)) return contextFailure(context);
645 if (!prepared_ || !expectedAfter_ || !store_.transactionStateEquals(*expectedAfter_))
646 return eve::Result<void>::failure(eve::Diagnostic::error(
647 eve::DiagnosticCode::Conflict, "state patch store changed before compensation", "statepatch.store"));
648
649 store_.swapTransactionState(*prepared_);
650 expectedAfter_.reset();
651 phase_ = Phase::Compensated;
652 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
653}
