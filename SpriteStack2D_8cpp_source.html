119void SpriteStack2D::draw(graphics::Graphics *gfx, graphics::Camera2D *camera) const {
120 if (!visible_ || layerCount_ <= 0) return;
121 float cx = x_;
122 float baseY = y_;
123 float scale = 1.f;
124 if (camera) {
125 const float viewW = float(gfx->getWidth());
126 const float viewH = float(gfx->getHeight());
127 cx = camera->worldToScreenX(x_, y_, viewW, viewH);
128 baseY = camera->worldToScreenY(x_, y_, viewW, viewH);
129 scale = camera->getZoom();
130 }
131 const auto queueLayer = [&](const Layer &layer, float x, float y, float w, float h,
132 const graphics::Color &tint) {
133 gfx->drawTexturedRectShaderUVRotated(layer.texture, nullptr, x, y, w, h, rotation_,
134 layer.u0, layer.v0, layer.u1, layer.v1, tint);
135 };
136 if (shadowEnabled_) {
137 const graphics::Color shadow(0.f, 0.f, 0.f, shadowOpacity_ * tintA_);
138 for (int i = 0; i < layerCount_; ++i) {
139 const Layer &layer = layers_[size_t(i)];
140 if (!layer.texture) continue;
141 queueLayer(layer, cx + shadowOffsetX_ * scale,
142 baseY + shadowOffsetY_ * scale - float(i) * thickness_ * scale,
143 width_ * scale, height_ * scale, shadow);
144 }
145 }
146 if (outlineWidth_ > 0.f) {
147 const graphics::Color outline(outlineR_, outlineG_, outlineB_, tintA_);
148 for (int i = 0; i < layerCount_; ++i) {
149 const Layer &layer = layers_[size_t(i)];
150 if (!layer.texture) continue;
151 queueLayer(layer, cx, baseY - float(i) * thickness_ * scale,
152 width_ * scale + outlineWidth_ * 2.f,
153 height_ * scale + outlineWidth_ * 2.f, outline);
154 }
155 }
156 const graphics::Color tint(tintR_, tintG_, tintB_, tintA_);
157 for (int i = 0; i < layerCount_; ++i) {
158 const Layer &layer = layers_[size_t(i)];
159 if (!layer.texture) continue;
160 queueLayer(layer, cx, baseY - float(i) * thickness_ * scale,
161 width_ * scale, height_ * scale, tint);
162 }
163}
