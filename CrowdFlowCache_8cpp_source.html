13Result<std::vector<int>> canonicalFlowCoordinates(std::span<const FlowGoal> goals, const CrowdField& field) {
14 if (!field.valid() || goals.empty() || goals.size() > 1024)
15 return cacheFailure<std::vector<int>>(DiagnosticCode::InvalidArgument,
16 "Expected 1..1024 goals and a configured field");
17 for (auto goal : goals)
18 if (goal.x < 0 || goal.y < 0 || goal.x >= field.getWidth() || goal.y >= field.getHeight())
19 return cacheFailure<std::vector<int>>(DiagnosticCode::InvalidArgument, "Goal outside grid");
20 std::vector<FlowGoal> sorted(goals.begin(), goals.end());
21 std::sort(sorted.begin(), sorted.end());
22 sorted.erase(std::unique(sorted.begin(), sorted.end()), sorted.end());
23 std::vector<int> result;
24 result.reserve(sorted.size() * 2);
25 for (auto goal : sorted) {
26 result.push_back(goal.x);
27 result.push_back(goal.y);
28 }
29 return Result<std::vector<int>>::success(std::move(result));
30}
31bool isFlowPointReachable(const CrowdField& field, float x, float y) noexcept {
32 if (!field.valid() || !field.isBuilt() || !std::isfinite(x) || !std::isfinite(y)) return false;
33 const double cx = std::floor((double(x) - field.getOriginX()) / field.getCellSize());
34 const double cy = std::floor((double(y) - field.getOriginY()) / field.getCellSize());
35 return cx >= 0 && cy >= 0 && cx < field.getWidth() && cy < field.getHeight() &&
36 field.isReachable(static_cast<int>(cx), static_cast<int>(cy));
37}

45 const auto prefix = [](const auto& entry) {

53Result<std::vector<Crowd::Impl::FlowCacheEntry>> Crowd::Impl::stageFlowCache(std::uint64_t budget,
54 std::uint64_t addedBytes,
55 size_t addedCount) const {
56 if (addedBytes > budget)
57 return cacheFailure<std::vector<FlowCacheEntry>>(DiagnosticCode::PreconditionViolation,
58 "Field exceeds cache storage budget");
59 std::vector<bool> evicted(flowCache.size(), false);
60 size_t count = flowCache.size();
61 std::uint64_t bytes = 0;
62 for (const auto& entry : flowCache) bytes += entry.bytes;
63 while (bytes > budget - addedBytes || count + addedCount > 10000) {
64 size_t victim = flowCache.size();
65 for (size_t i = 0; i < flowCache.size(); ++i) {
66 if (evicted[i] || !flowCache[i].pins.expired()) continue;
67 if (victim == flowCache.size() || flowCache[i].access < flowCache[victim].access ||
68 (flowCache[i].access == flowCache[victim].access && isFlowKeyBefore(flowCache[i], flowCache[victim])))
69 victim = i;
70 }
71 if (victim == flowCache.size())
72 return cacheFailure<std::vector<FlowCacheEntry>>(DiagnosticCode::PreconditionViolation,
73 "Cache pressure: all remaining fields are pinned");
74 evicted[victim] = true;
75 bytes -= flowCache[victim].bytes;
76 --count;
77 }
78 std::vector<FlowCacheEntry> staged;
79 staged.reserve(count + addedCount);
80 for (size_t i = 0; i < flowCache.size(); ++i)
81 if (!evicted[i]) staged.push_back(flowCache[i]);
82 return Result<std::vector<FlowCacheEntry>>::success(std::move(staged));
83}

110Result<FlowLease> Crowd::acquireFlowField(FlowBuildPlan&& plan) try {
111 if (!plan.field_ || plan.owner_ != impl_->owner || plan.revision_ != impl_->field.inputRevision() ||
112 (plan.state_ != FlowBuildState::Ready && plan.state_ != FlowBuildState::Unreachable))
113 return cacheFailure<FlowLease>(DiagnosticCode::Conflict, "Incomplete, stale or foreign field publication");
114 const auto surface = getFieldStatus().surface;
115 for (auto& entry : impl_->flowCache) {
116 if (entry.input == plan.revision_ && entry.surface == surface && entry.algorithm == flowBuildAlgorithmVersion &&
117 entry.field->goals_ == plan.field_->goals_) {
118 auto lease = impl_->leaseFor(entry);
119 plan.field_.reset();
120 plan.state_ = FlowBuildState::Committed;
121 return Result<FlowLease>::success(std::move(lease));
122 }
123 }
124 if (impl_->nextFlowGeneration == std::numeric_limits<std::uint64_t>::max())
125 return cacheFailure<FlowLease>(DiagnosticCode::Conflict, "Flow generation exhausted");
126 const auto bytes = Impl::retainedFlowBytes(*plan.field_);
127 auto staged = impl_->stageFlowCache(impl_->flowCacheBudget, bytes, 1);
128 if (!staged) return Result<FlowLease>::failure(staged.status());
129 auto entries = std::move(staged).takeValue();
130 static_assert(std::is_nothrow_move_constructible_v<CrowdField>);
131 static_assert(std::is_nothrow_move_constructible_v<Impl::FlowCacheEntry>);
132 // All metadata allocation and eviction validation precede the no-throw storage move.
133 auto pin = std::make_shared<const unsigned char>(0);
134 auto data = std::make_shared<const CrowdField>(std::move(*plan.field_));
135 entries.push_back({std::move(data), pin, surface, plan.revision_, impl_->nextFlowGeneration, bytes, impl_->tick,
136 flowBuildAlgorithmVersion});
137 auto lease = impl_->leaseFor(entries.back());
138 impl_->flowCache.swap(entries);

140 plan.field_.reset();
141 plan.state_ = FlowBuildState::Committed;
142 return Result<FlowLease>::success(std::move(lease));
143} catch (const std::exception& error) {
