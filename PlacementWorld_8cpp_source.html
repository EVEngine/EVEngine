83void PlacementWorld::setGridFromLayer(map::TileLayer *layer) {
84 if (!layer) return;
85 const auto &c = *layer->config();
86 switch (c.orientation) {
87 case map::MapOrientation::Orthogonal:
88 grid_.layout = grid::GridLayout::Rectangle;
89 break;
90 case map::MapOrientation::Isometric:
91 grid_.layout = grid::GridLayout::Isometric;
92 break;
93 case map::MapOrientation::Staggered:
94 grid_.layout = grid::GridLayout::Staggered;
95 break;
96 case map::MapOrientation::Hexagonal:
97 grid_.layout = grid::GridLayout::Hexagon;
98 break;
99 }
100 grid_.cellW = c.tileW;
101 grid_.cellH = c.tileH;
102 grid_.originX = c.originX;
103 grid_.originY = c.originY;
104 grid_.staggerAxis = c.staggerAxis == map::StaggerAxis::Y ? grid::StaggerAxis::Y
105 : grid::StaggerAxis::X;
106 grid_.staggerIndex = c.staggerIndex == map::StaggerIndex::Odd ? grid::StaggerIndex::Odd
107 : grid::StaggerIndex::Even;
108 grid_.hexSideLength = c.hexSideLength;
109}

244void PlacementWorld::setTerrainGidMapJson(const std::string &json) {
245 auto *dm = eve::data::DataModule::create();
246 std::string err;
247 std::unique_ptr<data::JsonDocument> doc(dm->decodeJson(json, &err));
248 if (!doc || !doc->isObject()) return;
249 auto obj = doc->object();
250 if (!obj) return;
251 std::vector<std::string> names;
252 obj->getNames(names);
253 for (const auto &n : names) {
254 try {
255 const int gid = std::atoi(n.c_str());
256 const int semantic = obj->getValue<int>(n);
257 if (gid >= 0) terrainGidMap_[gid] = semantic;
258 } catch (...) {
259 }
260 }
261}
