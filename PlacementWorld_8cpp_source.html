95void PlacementWorld::setGridFromLayer(map::TileLayer *layer) {
96 if (!layer) return;
97 const auto &c = *layer->config();
98 switch (c.orientation) {
99 case map::MapOrientation::Orthogonal:
100 grid_->layout = grid::GridLayout::Rectangle;
101 break;
102 case map::MapOrientation::Isometric:
103 grid_->layout = grid::GridLayout::Isometric;
104 break;
105 case map::MapOrientation::Staggered:
106 grid_->layout = grid::GridLayout::Staggered;
107 break;
108 case map::MapOrientation::Hexagonal:
109 grid_->layout = grid::GridLayout::Hexagon;
110 break;
111 }
112 grid_->cellW = c.tileW;
113 grid_->cellH = c.tileH;
114 grid_->originX = c.originX;
115 grid_->originY = c.originY;
116 grid_->staggerAxis = c.staggerAxis == map::StaggerAxis::Y ? grid::StaggerAxis::Y
117 : grid::StaggerAxis::X;
118 grid_->staggerIndex = c.staggerIndex == map::StaggerIndex::Odd ? grid::StaggerIndex::Odd
119 : grid::StaggerIndex::Even;
120 grid_->hexSideLength = c.hexSideLength;
121}

472int PlacementWorld::getEdgeOccupantAtLevel(const std::string &channel, int cellX, int cellY,
473 const std::string &direction, int level) const {
474 auto canonical = PlacementSystem::canonicalEdge(cellX, cellY, direction);
475 if (!canonical.ok()) return 0;
476 const EdgeAddress edge = std::move(canonical).takeValue();
477 const bool inBounds = edge.axis == EdgeAxis::Horizontal
478 ? edge.x >= 0 && edge.x < width_ && edge.y >= 0 && edge.y <= height_
479 : edge.x >= 0 && edge.x <= width_ && edge.y >= 0 && edge.y < height_;
480 if (!inBounds) return 0;
481 const ChannelMap *channels = nullptr;
482 if (level == 0) {
483 channels = edge.axis == EdgeAxis::Horizontal ? &horizontalEdgeChannels_
484 : &verticalEdgeChannels_;
485 } else {
486 const auto &levels = edge.axis == EdgeAxis::Horizontal ? horizontalEdgesByLevel_
487 : verticalEdgesByLevel_;
488 const auto levelIt = levels.find(level);
489 if (levelIt == levels.end()) return 0;
490 channels = &levelIt->second;
491 }
492 const auto it = channels->find(channel);
493 if (it == channels->end()) return 0;
494 const size_t index = edge.axis == EdgeAxis::Horizontal
495 ? size_t(edge.y) * size_t(width_) + size_t(edge.x)
496 : size_t(edge.y) * size_t(width_ + 1) + size_t(edge.x);
497 return index < it->second.size() ? it->second[index] : 0;
498}
