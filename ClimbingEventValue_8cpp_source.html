52Value climbingEventValue(const ClimbingEvent& event) {
53 Value::Array metadata;
54 metadata.reserve(event.metadata.size());
55 for (const std::string& value : event.metadata) metadata.emplace_back(value);
56 return Value::object({

58 {"actionId", event.actionId},
59 {"tick", std::to_string(event.tick.value())},
60 {"executionId", std::to_string(event.executionId.value())},
61 {"metadata", Value(std::move(metadata))},
