48Result<action::ActionPreviewFrame> ActionPreviewController::buildFrame(
49 action::ActionPreviewReason reason, Duration previous, Duration current,
50 const std::vector<action::ActionTimelineEvent>& events) const {
51 action::ActionPreviewFrame frame;
52 frame.reason = reason;
53 frame.animationUri = editor_.target().timeline().animationUri;
54 frame.previous = previous;
55 frame.current = current;
56 frame.cues.reserve(events.size());
57 for (const auto& event : events)
58 frame.cues.push_back({cueKind(event), event.itemId, event.type, event.time, event.payload});
59 if (rootMotion_) {
60 auto path = rootMotion_->sampleRootMotion(frame.animationUri, editor_.target().timeline().duration,
61 rootMotionSampleCount_);
62 if (!path) return Result<action::ActionPreviewFrame>::failure(path.status());
63 frame.rootMotionState = action::RootMotionPreviewState::Available;
64 frame.rootMotionPath = std::move(path).takeValue();
65 }
66 return Result<action::ActionPreviewFrame>::success(std::move(frame));
67}

69EditorResult<void> ActionPreviewController::refresh() {
70 auto frame = buildFrame(action::ActionPreviewReason::Refresh, editor_.previewTime(), editor_.previewTime(), {});
71 if (!frame)
72 return previewError(EditorStatus::Rejected, "editor.action.preview.root-motion",
73 diagnosticMessage(frame.status(), "Could not sample root motion"));
74 auto prepared = sink_.prepare(frame.value());
75 if (!prepared)
76 return previewError(EditorStatus::Rejected, "editor.action.preview.prepare",
77 diagnosticMessage(prepared.status(), "Preview host rejected the frame"));
78 sink_.present(frame.value());
79 lastFrame_ = std::move(frame).takeValue();
80 return eve::editing::applied<void>();
81}

83EditorResult<void> ActionPreviewController::seek(Duration time) {
84 if (time < Duration::zero() || time > editor_.target().timeline().duration)
85 return previewError(EditorStatus::Rejected, "editor.action.preview.seek-range",
86 "Preview seek is outside the timeline");
87 auto frame = buildFrame(action::ActionPreviewReason::Seek, editor_.previewTime(), time, {});
88 if (!frame)
89 return previewError(EditorStatus::Rejected, "editor.action.preview.root-motion",
90 diagnosticMessage(frame.status(), "Could not sample root motion"));
91 auto prepared = sink_.prepare(frame.value());
92 if (!prepared)
93 return previewError(EditorStatus::Rejected, "editor.action.preview.prepare",
94 diagnosticMessage(prepared.status(), "Preview host rejected the frame"));
95 auto changed = editor_.seek(time);
96 if (!changed.ok()) {
97 sink_.discardPrepared();
98 return changed;
99 }
100 sink_.present(frame.value());
101 lastFrame_ = std::move(frame).takeValue();
102 return eve::editing::applied<void>();

105EditorResult<std::size_t> ActionPreviewController::update(Duration delta) {
106 auto plan = editor_.planPreview(delta);
107 if (!plan.ok())
108 return previewErrorValue<std::size_t>(plan.code(), "editor.action.preview.plan",
109 "Could not prepare timeline preview advance");
110 if (plan.code() == EditorStatus::NoOp)
111 return EditorResult<std::size_t>::success(0, Status::success(EditorStatus::NoOp));
112 auto frame = buildFrame(action::ActionPreviewReason::Advance, plan.value().previous,
113 plan.value().current, plan.value().events);
114 if (!frame)
115 return previewErrorValue<std::size_t>(EditorStatus::Rejected, "editor.action.preview.root-motion",
116 diagnosticMessage(frame.status(), "Could not sample root motion"));
117 auto prepared = sink_.prepare(frame.value());
118 if (!prepared)
119 return previewErrorValue<std::size_t>(EditorStatus::Rejected, "editor.action.preview.prepare",
120 diagnosticMessage(prepared.status(), "Preview host rejected the frame"));
121 auto advanced = editor_.applyPreviewPlan(std::move(plan).takeValue());
122 if (!advanced.ok()) {
123 sink_.discardPrepared();
124 return advanced;
125 }
126 sink_.present(frame.value());
127 lastFrame_ = std::move(frame).takeValue();
128 return advanced;
129}

131void ActionPreviewController::drawRootMotion(IEditorOverlay& overlay, OverlayPoint origin, float scale) const {
132 if (!lastFrame_ || lastFrame_->rootMotionState != action::RootMotionPreviewState::Available ||
133 !std::isfinite(scale) || scale <= 0.0f)
134 return;
135 for (std::size_t index = 1; index < lastFrame_->rootMotionPath.size(); ++index) {
136 const auto& previous = lastFrame_->rootMotionPath[index - 1];
137 const auto& current = lastFrame_->rootMotionPath[index];
138 overlay.line({origin.x + previous.x * scale, origin.y + previous.y * scale, origin.z + previous.z * scale},
139 {origin.x + current.x * scale, origin.y + current.y * scale, origin.z + current.z * scale},
140 {0x55d6beffU, 2.0f, false});
141 }
142}
