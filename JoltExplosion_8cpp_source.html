11float projectedArea(const ShapeState& shape, JPH::Vec3Arg direction) {
12 const auto& g = shape.geometry;
13 switch (g.kind) {
14 case ShapeKind3D::Sphere: return JPH::JPH_PI * g.a * g.a;
15 case ShapeKind3D::Capsule:
16 return JPH::JPH_PI * g.b * g.b + 4 * g.a * g.b * JPH::Vec3::sAxisY().Cross(direction).Length();
17 case ShapeKind3D::Box:
18 return 4 * (g.b * g.c * std::abs(direction.GetX()) + g.a * g.c * std::abs(direction.GetY()) +
19 g.a * g.b * std::abs(direction.GetZ()));
20 case ShapeKind3D::ConvexHull: {
21 const auto& hull = static_cast<const JPH::ConvexHullShape&>(*shape.native);
22 float area = 0;
23 for (JPH::uint face = 0; face < hull.GetNumFaces(); ++face) {
24 JPH::uint indices[256];
25 const auto count = hull.GetFaceVertices(face, 256, indices);
26 const auto origin = hull.GetPoint(indices[0]);
27 for (JPH::uint i = 1; i + 1 < count; ++i)
28 area += .5f * std::max(0.f, (hull.GetPoint(indices[i]) - origin)
29 .Cross(hull.GetPoint(indices[i + 1]) - origin)
30 .Dot(direction));
31 }
32 return area;
33 }
34 default: return 0;
35 }
36}

38eve::Result<int> WorldBackend::explode(float x, float y, float z, float radius, float falloff, float strength,
39 int mask) {
40 if (!isValid()) return rejected<int>("World expired", eve::DiagnosticCode::StaleHandle);
41 if (!isFinite(x, y, z) || !isFinite(radius, falloff, strength) || radius < 0 || falloff < 0 ||
42 !std::isfinite(radius + falloff))
43 return rejected<int>("Invalid explosion", eve::DiagnosticCode::InvalidArgument);
44 owner->explosionResults_.clear();
45 struct Impulse {
46 JPH::RVec3 point;
47 JPH::Vec3 impulse;
48 std::shared_ptr<BodyState> body;
49 };
50 std::vector<Impulse> impulses;
51 const JPH::RVec3 center(x, y, z);
52 // Prepare and validate every impulse before mutating any body.
53 for (const auto& [id, body] : state->bodies) {
54 if (!body->active || !body->owner || body->api().GetMotionType(body->id) != JPH::EMotionType::Dynamic) continue;
55 for (const auto& shape : body->shapes) {
56 if (!shape->owner || shape->native->GetType() != JPH::EShapeType::Convex ||
57 !(shape->filter.categoryBits & uint32_t(mask)) || shape->owner->getExplosionScale() == 0)
58 continue;
59 auto point = closestConvexPoint(*shape, center);
60 const float distance = float((point - center).Length());
61 if (distance > radius + falloff) continue;
62 const auto& g = shape->geometry;
63 const auto rotation =
64 body->api().GetRotation(body->id) * JPH::Quat(g.localQx, g.localQy, g.localQz, g.localQw);
65 if (distance <= 1.e-6f)
66 point = body->api().GetWorldTransform(body->id) *
67 (JPH::Vec3(g.localX, g.localY, g.localZ) +
68 JPH::Quat(g.localQx, g.localQy, g.localQz, g.localQw) * shape->native->GetCenterOfMass());
69 const auto direction = JPH::Vec3(point - center).NormalizedOr(JPH::Vec3::sAxisX());
70 const float scale =
71 distance > radius && falloff > 0 ? std::clamp((radius + falloff - distance) / falloff, 0.f, 1.f) : 1.f;
72 const float magnitude = strength * projectedArea(*shape, rotation.Conjugated() * direction) * scale *
73 shape->owner->getExplosionScale();
74 if (!std::isfinite(magnitude))
75 return rejected<int>("Explosion impulse overflow", eve::DiagnosticCode::InvalidArgument);
76 impulses.push_back({point, direction * magnitude, body});
77 }
78 }
79 struct Before {
80 std::shared_ptr<BodyState> body;
81 JPH::Vec3 linear, angular;
82 };
83 std::vector<Before> before;
84 for (const auto& impulse : impulses) {
85 auto body = impulse.body;
86 if (std::none_of(before.begin(), before.end(), [&](const Before& entry) { return entry.body == body; }))
87 before.push_back({body, body->api().GetLinearVelocity(body->id), body->api().GetAngularVelocity(body->id)});
88 body->api().AddImpulse(body->id, impulse.impulse, impulse.point);
89 }
90 for (const auto& entry : before) {
91 const auto dv = entry.body->api().GetLinearVelocity(entry.body->id) - entry.linear;
92 const auto dw = entry.body->api().GetAngularVelocity(entry.body->id) - entry.angular;
93 if (dv.LengthSq() <= 1.e-12f && dw.LengthSq() <= 1.e-12f) continue;
94 owner->explosionResults_.push_back(
95 {entry.body->owner->getId(), dv.GetX(), dv.GetY(), dv.GetZ(), dw.GetX(), dw.GetY(), dw.GetZ()});
96 }
97 std::sort(owner->explosionResults_.begin(), owner->explosionResults_.end(),
98 [](const auto& a, const auto& b) { return a.bodyId < b.bodyId; });
99 return eve::Result<int>::success(int(owner->explosionResults_.size()));
100}
