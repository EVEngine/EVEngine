105bool isSubclassOf(HSQUIRRELVM vm, HSQOBJECT child, HSQOBJECT base) {
106 if (child._type != OT_CLASS || base._type != OT_CLASS) return false;
107 const SQInteger top = sq_gettop(vm);
108 sq_pushobject(vm, child);
109 for (int guard = 0; guard < 64; ++guard) {
110 if (sq_gettype(vm, -1) != OT_CLASS) break;
111 HSQOBJECT cur;
112 sq_getstackobj(vm, -1, &cur);
113 if (cur._unVal.pClass == base._unVal.pClass) {
114 sq_settop(vm, top);
115 return true;
116 }
117 if (SQ_FAILED(sq_getbase(vm, -1))) break;
118 sq_remove(vm, -2); // drop current, keep base
119 if (sq_gettype(vm, -1) != OT_CLASS) break;
120 }
121 sq_settop(vm, top);
122 return false;
123}

126void collectClassFields(HSQUIRRELVM vm, HSQOBJECT clsObj, HSQOBJECT outTable) {
127 const SQInteger top = sq_gettop(vm);
128 sq_pushobject(vm, outTable); // out
129 sq_pushobject(vm, clsObj); // out, cls
130 sq_pushnull(vm); // out, cls, iter
131 while (SQ_SUCCEEDED(sq_next(vm, -2))) {
132 // out, cls, iter, key, value
133 if (sq_gettype(vm, -2) == OT_STRING) {
134 const SQChar* key = nullptr;
135 sq_getstring(vm, -2, &key);
136 if (key && key[0] != '_') {
137 const SQObjectType vt = sq_gettype(vm, -1);
138 if (vt != OT_CLOSURE && vt != OT_NATIVECLOSURE && vt != OT_GENERATOR) {
139 sq_push(vm, -5); // out
140 sq_push(vm, -3); // key
141 sq_push(vm, -3); // value
142 sq_rawset(vm, -3);
143 sq_pop(vm, 1); // pop out
144 }
145 }
146 }
147 sq_pop(vm, 2); // key, value
148 }
149 sq_settop(vm, top);

170std::vector<DeclaredField> declaredFields(HSQOBJECT clsObj) {
171 std::vector<std::pair<SQInteger, DeclaredField>> ordered;
172 SQClass* klass = _class(clsObj);
173 if (klass == nullptr || klass->_members == nullptr) return {};
174
175 SQObjectPtr key;
176 SQObjectPtr value;
177 SQObjectPtr refpos; // `Next` 的游标是它自己返回的整数下标，不是 key（传 key 会命中 TranslateIndex 的断言）
178 SQInteger index = 0;
179 while ((index = klass->_members->Next(false, refpos, key, value)) != -1) {
180 refpos = SQObjectPtr(index);
181 if (sq_type(key) != OT_STRING || !_isfield(value)) continue;
182 const SQChar* name = _stringval(key);
183 if (name == nullptr || name[0] == '_') continue;
184 ordered.emplace_back(static_cast<SQInteger>(_member_idx(value)), DeclaredField{name, key});
185 }
186 std::sort(ordered.begin(), ordered.end(), [](const auto& a, const auto& b) { return a.first < b.first; });
187
188 std::vector<DeclaredField> out;
189 out.reserve(ordered.size());
190 for (auto& entry : ordered) out.push_back(std::move(entry.second));
191 return out;
192}

430ssq::Object bindComponentValues(std::uint64_t entityId, ssq::Object instance) {
431 HSQUIRRELVM vm = instance.getHandle();
432 ssq::Object none(vm);
433 if (instance.getType() != ssq::Type::INSTANCE) return none;
434
435 // 类对象只**借用**：实例自己持有类引用（`SQInstance::SQInstance` 里 `__ObjAddRef(_class)`），
436 // 所以查完注册表、把类弹出栈之后，注册表里那份引用仍然有效。
437 const SQInteger top = sq_gettop(vm);
438 sq_pushobject(vm, instance.getRaw());
439 HSQOBJECT cls{};
440 if (SQ_FAILED(sq_getclass(vm, -1))) {
441 sq_settop(vm, top);
442 return none;
443 }
444 sq_getstackobj(vm, -1, &cls);
445
446 ScriptComponentSchema schema;
447 const bool known = schemaOfComponentClass(cls, &schema);
448 sq_settop(vm, top);
449 if (!known) return none;
450
451 const ScriptComponentStoreKey key{entityId, schema.record.name, schema.record.version};
452 const auto stamp = [&](const char* name, const auto& push) {
453 const SQInteger inner = sq_gettop(vm);
454 sq_pushobject(vm, instance.getRaw());
455 sq_pushstring(vm, name, -1);
456 push();
457 sq_rawset(vm, -3); // raw：`_set` 是字段接缝，控制槽位不能从那里走
458 sq_settop(vm, inner);
459 };
460 stamp("_ownerEid", [&] { sq_pushinteger(vm, static_cast<SQInteger>(entityId)); });
461 stamp("_componentName", [&] { sq_pushstring(vm, schema.record.name.c_str(), -1); });
462 stamp("_componentVersion", [&] { sq_pushinteger(vm, static_cast<SQInteger>(schema.record.version.value())); });
463
464 if (findScriptComponentValues(key) == nullptr) {
465 Value::Object values;
466 const std::vector<Value>& declared = *schema.defaults;
467 const std::vector<std::string>& fields = schema.record.fields;
468 for (std::size_t i = 0; i < fields.size(); ++i)
469 values.emplace(fields[i], i < declared.size() ? declared[i] : Value::null());
470 setScriptComponentValues(key, Value::object(std::move(values)))
471 .ignore("a component whose values cannot be stored would silently lose its data if this succeeded");
472 }
473
474 ssq::Array names(vm);
475 for (const std::string& field : schema.record.fields) names.push(field);
476 return names;
477}

536//
537// `declared` 是引擎侧给的字段名数组（登记组件）或 null（未登记组件）。登记组件的声明默认值

546 }

552 || cur == "number" || cur == "boolean" || cur == "string") {
553 c[name] = this._ecsResolveMarker(cur)
554 resolved[name] <- c[name]
555 }
556 }
557 return resolved

562 // 先盖归属、按 schema 建引擎侧数据条目，**再**写默认值：已登记组件的字段槽位在登记时就被摘掉、
563 // `_set` 已装好，所以下面那些 `c[name] = val` 实际上是**写进引擎存储**，必须有个已绑定的条目
564 // 接着（顺序反了就会报 "not bound to an entity"）。

592 }

675function eve::_ecsCollectInstances(cls, out) {
676 foreach (key, info in this._ecsTypes) {
677 if (info.cls == cls || this.isSubclass(info.cls, cls)) {
678 foreach (e in info.instances) {

683 // 稳定顺序：按实体 id 升序（= 创建顺序）。
684 //
685 // `_ecsUnregisterInstance` 用 swap-remove 保持 O(1) 销毁，所以 instances 数组本身的
686 // 顺序在删除后会变——实测 [1,2,3,4] 删掉 2 得到 [1,4,3]。而 GPU System 按**视图顺序**
687 // 把实体打包成 buffer 的行，回读时又按同一顺序写回（`unpackEcsFloats(ents, …)`），
688 // `requestUploadRange/requestReadbackRange(first, count)` 更是直接按行号寻址：顺序一变，
689 // "第 i 行是哪个实体"就变了，回读的值会落到另一个实体上。
690 //
691 // id 由 `eve.allocEntityId()` 单调分配且**不复用**，所以按 id 排序给出一个跨子类也
692 // 成立的全序；`_ecsTypes` 的散列遍历顺序因此不再影响结果。排序只发生在缓存重建时
693 // （结构性变更后第一次取视图），每帧取视图仍然是 O(n) 拷贝。
694 out.sort(function(a, b) { return a._eid - b._eid })

710 local declared = null
711 try { declared = this.inspectClassFields(old.getclass()) } catch (e) { return 0 }
712 local copied = 0
713 foreach (name, value in declared) {
714 try { fresh[name] = old[name]; copied += 1 } catch (e) { }
715 }
716 return copied

720function eve::_ecsRepointComponent(e, slotName, oldComp, compClass, fieldSlots) {
721 local fresh = compClass()
722 local declared = eve._ecsBindComponentValues(e._eid, fresh)

759function eve::_ecsMigrateAfterReload(oldBindings, rootTable) {
760 local replacements = {};
761 if (rootTable == null) return { entities = 0, components = 0, unresolved = 0 };
762 foreach (name, oldClass in oldBindings) {
763 if (typeof oldClass != "class" || !(name in rootTable)) continue;
764 local newClass = rootTable[name];
765 if (typeof newClass != "class" || newClass == oldClass) continue;
766 if (!this.isSubclass(newClass, this.Component) && !this.isSubclass(newClass, this.Entity)) continue;
767 replacements[oldClass] <- newClass;
768 }
769
770 // 第 1 步：换键。旧项一律删掉，否则热重载每来一次就多一组看不见的实例（§9.42.5 第 5 点）。
771 foreach (oldClass, newClass in replacements) {
772 if (oldClass in this._ecsTypes) {
773 this._ecsTypes[newClass] <- this._ecsTypes[oldClass];
774 this._ecsTypes[newClass].cls = newClass;
775 delete this._ecsTypes[oldClass];
776 }
777 if (oldClass in this._ecsViewCache) delete this._ecsViewCache[oldClass];
778 if (oldClass in this._ecsSlotsCache) delete this._ecsSlotsCache[oldClass];
779 if (oldClass in this._ecsFieldSlotsCache) delete this._ecsFieldSlotsCache[oldClass];
780 if (oldClass in this._ecsCompDefaults) delete this._ecsCompDefaults[oldClass];
781 }
782
783 // 第 2 步：接实例。
784 local entities = 0;
785 local components = 0;
786 local unresolved = 0;
787 foreach (etype, info in this._ecsTypes) {
788 local slots = this._ecsCollectSlots(etype); // 按新类重算：它引用的是新组件类
789 local fieldSlots = this._ecsCollectFieldSlots(etype);
790 foreach (e in info.instances) {
791 if (e == null || !e.isAlive()) continue;
792 e._etype = etype;
793 e._slots = slots;
794 foreach (slotName, compClass in slots) {
795 if (!(slotName in e._comps)) continue;
796 local old = e._comps[slotName];
797 if (old == null) continue;
798 if (compClass in replacements) {
799 // 实体类没被重写：槽位还指向**旧**组件类，引擎改不了类的字段默认值。
800 unresolved += 1;
801 continue;
802 }
803 if (typeof old == "instance" && old.getclass() == compClass) continue;
804 if (this._ecsRepointComponent(e, slotName, old, compClass, fieldSlots)) components += 1;
805 else unresolved += 1;
806 }
807 entities += 1;
808 }
809 this._ecsInvalidateViews(etype);
810 }
811 if (entities > 0) this._ecsRevision += 1;
812 return { entities = entities, components = components, unresolved = unresolved };
813}
