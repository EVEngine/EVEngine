63bool isSubclassOf(HSQUIRRELVM vm, HSQOBJECT child, HSQOBJECT base) {
64 if (child._type != OT_CLASS || base._type != OT_CLASS) return false;
65 const SQInteger top = sq_gettop(vm);
66 sq_pushobject(vm, child);
67 for (int guard = 0; guard < 64; ++guard) {
68 if (sq_gettype(vm, -1) != OT_CLASS) break;
69 HSQOBJECT cur;
70 sq_getstackobj(vm, -1, &cur);
71 if (cur._unVal.pClass == base._unVal.pClass) {
72 sq_settop(vm, top);
73 return true;
74 }
75 if (SQ_FAILED(sq_getbase(vm, -1))) break;
76 sq_remove(vm, -2); // drop current, keep base
77 if (sq_gettype(vm, -1) != OT_CLASS) break;
78 }
79 sq_settop(vm, top);
80 return false;
81}

84void collectClassFields(HSQUIRRELVM vm, HSQOBJECT clsObj, HSQOBJECT outTable) {
85 const SQInteger top = sq_gettop(vm);
86 sq_pushobject(vm, outTable); // out
87 sq_pushobject(vm, clsObj); // out, cls
88 sq_pushnull(vm); // out, cls, iter
89 while (SQ_SUCCEEDED(sq_next(vm, -2))) {
90 // out, cls, iter, key, value
91 if (sq_gettype(vm, -2) == OT_STRING) {
92 const SQChar* key = nullptr;
93 sq_getstring(vm, -2, &key);
94 if (key && key[0] != '_') {
95 const SQObjectType vt = sq_gettype(vm, -1);
96 if (vt != OT_CLOSURE && vt != OT_NATIVECLOSURE && vt != OT_GENERATOR) {
97 sq_push(vm, -5); // out
98 sq_push(vm, -3); // key
99 sq_push(vm, -3); // value
100 sq_rawset(vm, -3);
101 sq_pop(vm, 1); // pop out
102 }
103 }
104 }
105 sq_pop(vm, 2); // key, value
106 }
107 sq_settop(vm, top);
108}

209function eve::_ecsCollectSlots(cls) {
210 if (cls in this._ecsSlotsCache) return this._ecsSlotsCache[cls]
211 // { fieldName = ComponentClass } walking base classes (Entity → … → cls).
212 local slots = {}
213 local cur = cls
214 local guard = 0
215 while (cur != null && guard < 64) {
216 if (!this.isSubclass(cur, this.Entity)) break
217
218 local declared = null
219 try { declared = cur.components } catch (e) { declared = null }
220 if (typeof declared == "table") {
221 foreach (k, v in declared) {
222 if (!(k in slots) && this._ecsIsComponentClass(v))
223 slots[k] <- v
224 }
225 }
226
227 local fields = this.inspectClassFields(cur)
228 foreach (k, v in fields) {
229 if (k == "components") continue
230 if (!(k in slots) && this._ecsIsComponentClass(v))
231 slots[k] <- v
232 }
233
234 cur = this.getClassBase(cur)
235 guard += 1
236 }
237 this._ecsSlotsCache[cls] <- slots
238 return slots
239}

449 // Typed binding schema: { slot, fields, scalar="f32", access="read_write" }.
450 // write-only bindings allocate storage without an unnecessary CPU upload.
451 function bindSchema(binding, schema, uploadEachUpdate = false, readback = false) {
452 if (typeof schema != "table" || !("slot" in schema) || !("fields" in schema))
453 throw "eve.ShaderSystem.bindSchema: schema requires slot and fields"
454 if (typeof schema.slot != "string" || typeof schema.fields != "array" ||
455 schema.fields.len() <= 0)
456 throw "eve.ShaderSystem.bindSchema: invalid slot or fields"

458 if (typeof field != "string")
459 throw "eve.ShaderSystem.bindSchema: every field must be a string"

463 local access = "access" in schema ? schema.access : "read_write"

545 // Re-upload a stable contiguous entity range. Repeated requests are coalesced.
546 function requestUploadRange(binding, firstEntity, entityCount) {
547 if (firstEntity < 0 || entityCount <= 0)
548 throw "eve.ShaderSystem.requestUploadRange: invalid entity range"
549 foreach (b in _bindings) {
550 if (b.binding != binding) continue
551 if (b.source != null)
552 throw "eve.ShaderSystem.requestUploadRange: shared binding is GPU-owned"
553 if (!b.uploadRequested || b.uploadCount == 0) {
554 b.uploadFirst = firstEntity
555 b.uploadCount = entityCount
556 } else if (b.uploadCount >= 0) {
557 local oldEnd = b.uploadFirst + b.uploadCount
558 local newEnd = firstEntity + entityCount
559 local end = oldEnd > newEnd ? oldEnd : newEnd
560 b.uploadFirst = b.uploadFirst < firstEntity ? b.uploadFirst : firstEntity
561 b.uploadCount = end - b.uploadFirst
562 }
563 b.uploadRequested = true
564 }
565 return this
566 }

672 // authoritative and promises to initialize every accessed record.
673 b.uploadedRevision = eve._ecsRevision

781 HSQUIRRELVM vm = cls.getHandle();

783 sq_pushobject(vm, cls.getRaw());

796 break;
