47float smoothNoise2(float x, float y) {
48 const int ix = int(std::floor(x));
49 const int iy = int(std::floor(y));
50 const float fx = x - float(ix);
51 const float fy = y - float(iy);
52 const float sx = fx * fx * (3.f - 2.f * fx);
53 const float sy = fy * fy * (3.f - 2.f * fy);
54 const float n00 = hashNoise2(ix, iy);
55 const float n10 = hashNoise2(ix + 1, iy);
56 const float n01 = hashNoise2(ix, iy + 1);
57 const float n11 = hashNoise2(ix + 1, iy + 1);
58 const float a = n00 + (n10 - n00) * sx;
59 const float b = n01 + (n11 - n01) * sx;
60 return a + (b - a) * sy;
61}

110void sampleEmissionOffset(ParticleEmitter::Config &cfg, ParticleEmitter::Sim &sim, float &ox,
111 float &oy) {
112 ox = 0.f;
113 oy = 0.f;
114 if (cfg.areaX <= 0.f && cfg.areaY <= 0.f) return;
115 if (cfg.areaType == "ellipse") {
116 const float a = randRange(sim.rng, 0.f, kPi * 2.f);
117 const float r = std::sqrt(randRange(sim.rng, 0.f, 1.f));
118 ox = std::cos(a) * cfg.areaX * r;
119 oy = std::sin(a) * cfg.areaY * r;
120 } else if (cfg.areaType == "rect") {

128 const float rx = cfg.areaX > 0.f ? cfg.areaX : 1.f;
129 const float ry = cfg.areaY > 0.f ? cfg.areaY : rx;
130 ox = std::cos(a) * rx;
131 oy = std::sin(a) * ry;

135void rebuildSkinCandidates(ParticleEmitter::SkinSource &src) {
136 src.candidates.clear();
137 src.candidatesDirty = false;
138 if (!src.skin || src.skin->getVertexCount() <= 0) return;
139
140 const int n = src.skin->getVertexCount();
141 const int influences = src.skin->getInfluenceCount();
142 if (src.filterBone < 0) {
143 src.candidates.reserve(static_cast<size_t>(n));
144 for (int v = 0; v < n; ++v) src.candidates.push_back(v);
145 return;
146 }
147
148 src.candidates.reserve(static_cast<size_t>(n / 4 + 1));
149 for (int v = 0; v < n; ++v) {
150 float w = 0.f;
151 for (int i = 0; i < influences; ++i) {
152 if (src.skin->getVertexBone(v, i) == src.filterBone) {
153 w += src.skin->getVertexWeight(v, i);
154 }
155 }
156 if (w >= src.minWeight) src.candidates.push_back(v);
157 }

307void fireSubEmitter(ParticleEmitter::Config &cfg, const std::string &trigger, float x, float y,
308 float vx, float vy) {
309 static int g_subDepth = 0;
310 if (g_subDepth >= 8) return; // cycle guard (A→B→A chains)
311 for (const auto &se : cfg.subEmitters) {
312 if (!se.target || se.trigger != trigger) continue;
313 // Self-referencing sub-emitters would corrupt the compaction loop.
314 if (se.target == cfg.entity) continue;
315 auto tc = se.target->config();
316 auto ts = se.target->sim();
317 if (ts->alive >= int(ts->particles.size())) continue;
318 ++g_subDepth;
319 const bool spawned = spawnParticleAt(*tc, *ts, x, y);
320 --g_subDepth;
321 if (spawned && se.inheritVelocity > 0.f) {
322 Particle &p = ts->particles[size_t(ts->alive - 1)];
323 p.vx += vx * se.inheritVelocity;
324 p.vy += vy * se.inheritVelocity;
325 }
326 }
327}

414void stepEmitterSim(ParticleEmitter::Config &cfg, ParticleEmitter::Sim &sim, float dt) {
415 if (sim.paused || dt <= 0.f) return;
416 if (cfg.maxDeltaTime > 0.f && dt > cfg.maxDeltaTime) dt = cfg.maxDeltaTime;
417
418 // Emitter velocity for inheritVelocity (before lastX/lastY refresh).
419 float emitVx = 0.f, emitVy = 0.f;
420 if (cfg.inheritVelocity > 0.f && sim.hasLastPos) {
421 emitVx = (cfg.x - sim.lastX) / dt;
422 emitVy = (cfg.y - sim.lastY) / dt;
423 }
424 const bool localSpace = cfg.simSpace == "local";
425 const float localDx = localSpace && sim.hasLastPos ? cfg.x - sim.lastX : 0.f;
426 const float localDy = localSpace && sim.hasLastPos ? cfg.y - sim.lastY : 0.f;
427
428 const float damp = cfg.damping > 0.f ? (cfg.damping > 1.f ? 1.f : cfg.damping) : 0.f;
429 const float dampFactor = damp > 0.f ? std::max(0.f, 1.f - damp * dt) : 1.f;
430 const float noiseFreq = cfg.noiseFrequency > 0.f ? cfg.noiseFrequency : 1.f;
431 const bool hasCollision =
432 cfg.collisionMode != "none" && (cfg.worldCollision || cfg.collisionBoundsEnabled);
433 const bool bounce = cfg.collisionMode == "bounce";
434 const bool killMode = cfg.collisionMode == "kill";
435 const bool stopMode = cfg.collisionMode == "stop";
436 const float cRadius = cfg.collisionRadius;
437 const float lifeLoss = cfg.collisionLifetimeLoss;
438
439 int write = 0;
440 for (int i = 0; i < sim.alive; ++i) {
441 Particle &p = sim.particles[size_t(i)];
442 p.life -= dt;
443 if (p.life <= 0.f) {
444 fireSubEmitter(cfg, "death", p.x, p.y, p.vx, p.vy);
445 continue;
446 }
447
448 if (localSpace && (localDx != 0.f || localDy != 0.f)) {
449 p.x += localDx;
450 p.y += localDy;
451 }
452
453 float ax = p.ax + cfg.gravityX;
454 float ay = p.ay + cfg.gravityY;
455 const float dx = p.x - cfg.x;
456 const float dy = p.y - cfg.y;
457 const float len2 = dx * dx + dy * dy;
458 if (len2 > kEps) {
459 const float inv = 1.f / std::sqrt(len2);
460 const float rdx = dx * inv;
461 const float rdy = dy * inv;
462 ax += rdx * p.radial - rdy * p.tangential;
463 ay += rdy * p.radial + rdx * p.tangential;
464 }
465
466 if (cfg.noiseStrength != 0.f) {
467 const float t = sim.emitterAge * cfg.noiseSpeed;
468 ax += smoothNoise2(p.x * noiseFreq + t, p.y * noiseFreq + p.noisePhase) *
469 cfg.noiseStrength;
470 ay += smoothNoise2(p.y * noiseFreq + t, p.x * noiseFreq - p.noisePhase) *
471 cfg.noiseStrength;
472 }
473
474 // Radial force fields (strength > 0 attract, < 0 repel).
475 for (const auto &f : cfg.forceFields) {
476 if (f.radius <= 0.f || f.strength == 0.f) continue;
477 const float fdx = f.x - p.x;
478 const float fdy = f.y - p.y;
479 const float dist = std::sqrt(fdx * fdx + fdy * fdy);
480 if (dist >= f.radius) continue;
481 const float fall = std::pow(1.f - dist / f.radius, f.falloff > 0.f ? f.falloff : 1.f);
482 const float inv = dist > kEps ? 1.f / dist : 0.f;
483 ax += fdx * inv * f.strength * fall;
484 ay += fdy * inv * f.strength * fall;
485 }
486
487 p.vx += ax * dt;
488 p.vy += ay * dt;
489
490 // Velocity-over-lifetime multiplier (applied as a smooth ratio).
491 if (!cfg.velocityCurve.empty() && p.lifetime > 0.f) {
492 const float tNew = 1.f - (p.life / p.lifetime);
493 const float tOld = tNew - dt / p.lifetime;
494 const float vOld = cfg.velocityCurve.sample(tOld, 1.f);
495 const float vNew = cfg.velocityCurve.sample(tNew, 1.f);
496 if (vOld > 1e-4f) {
497 const float ratio = vNew / vOld;
498 p.vx *= ratio;
499 p.vy *= ratio;
500 }
501 }
502
503 if (dampFactor != 1.f) {
504 p.vx *= dampFactor;
505 p.vy *= dampFactor;
506 }
507 if (cfg.limitVelocity > 0.f) {
508 const float sp2 = p.vx * p.vx + p.vy * p.vy;
509 const float max2 = cfg.limitVelocity * cfg.limitVelocity;
510 if (sp2 > max2) {
511 const float s = std::sqrt(max2 / sp2);
512 p.vx *= s;
513 p.vy *= s;
514 }
515 }
516
517 p.x += p.vx * dt;
518 p.y += p.vy * dt;
519 p.rot += p.spin * dt;
520 p.frame += cfg.frameRate * dt;
521
522 if (hasCollision) {
523 const float scale = p.size > 0.f ? p.size : 1.f;
524 const float rad =
525 cRadius > 0.f ? cRadius : std::max(cfg.particleW, cfg.particleH) * 0.5f * scale;
526 float nx = 0.f, ny = 0.f;
527 bool hit = false;
528 if (cfg.worldCollision) {
529 if (WorldCollisionFn fn = getWorldCollisionResolver())
530 if (fn(p.x, p.y, rad, nx, ny)) hit = true;
531 }
532 if (!hit && cfg.collisionBoundsEnabled) {
533 if (p.x - rad < cfg.boundsMinX) {
534 nx += 1.f;
535 p.x = cfg.boundsMinX + rad;
536 hit = true;
537 } else if (p.x + rad > cfg.boundsMaxX) {
538 nx += -1.f;
539 p.x = cfg.boundsMaxX - rad;
540 hit = true;
541 }
542 if (p.y - rad < cfg.boundsMinY) {
543 ny += 1.f;
544 p.y = cfg.boundsMinY + rad;
545 hit = true;
546 } else if (p.y + rad > cfg.boundsMaxY) {
547 ny += -1.f;
548 p.y = cfg.boundsMaxY - rad;
549 hit = true;
550 }
551 const float nlen = std::sqrt(nx * nx + ny * ny);
552 if (nlen > kEps) {
553 nx /= nlen;
554 ny /= nlen;
555 }
556 }
557 if (hit) {
558 fireSubEmitter(cfg, "collision", p.x, p.y, p.vx, p.vy);
559 if (killMode) continue;
560 if (stopMode) {
561 p.vx = 0.f;
562 p.vy = 0.f;
563 } else if (bounce && (nx != 0.f || ny != 0.f)) {
564 const float dot = p.vx * nx + p.vy * ny;
565 p.vx = (p.vx - 2.f * dot * nx) * cfg.collisionRestitution;
566 p.vy = (p.vy - 2.f * dot * ny) * cfg.collisionRestitution;
567 }
568 if (lifeLoss > 0.f) {
569 p.life -= p.lifetime * lifeLoss;
570 if (p.life <= 0.f) continue;
571 }
572 }
573 }
574
575 if (write != i) sim.particles[size_t(write)] = p;
576 ++write;
577 }
578 sim.alive = write;
579
580 if (!sim.active) {
581 sim.lastX = cfg.x;
582 sim.lastY = cfg.y;
583 sim.hasLastPos = true;
584 return;
585 }
586
587 sim.emitterAge += dt;
588
589 // Timed bursts fire once while the emitter is active.
590 auto spawnWithInherit = [&]() {
591 const bool spawned = spawnParticle(cfg, sim);
592 if (spawned && cfg.inheritVelocity > 0.f && sim.alive > 0) {
593 Particle &np = sim.particles[size_t(sim.alive - 1)];
594 np.vx += emitVx * cfg.inheritVelocity;
595 np.vy += emitVy * cfg.inheritVelocity;
596 }
597 return spawned;
598 };
599 for (auto &b : cfg.bursts) {
600 if (!b.emitted && b.count > 0 && sim.emitterAge >= b.time) {
601 b.emitted = true;
602 for (int k = 0; k < b.count; ++k) {
603 if (sim.alive >= int(sim.particles.size())) break;
604 if (!spawnWithInherit()) break;
605 }
606 }
607 }
608
609 const float emissionScale = cfg.resolvedParameterScale("emission");
610 if (cfg.emissionRate > 0.f && emissionScale > 0.f) {
611 sim.emitAccum += cfg.emissionRate * emissionScale * dt;
612 while (sim.emitAccum >= 1.f) {
613 if (sim.alive >= int(sim.particles.size())) {
614 if (cfg.overflowMode == "pause") break; // keep accum; retry next frame
615 if (cfg.overflowMode == "warn" && !sim.overflowWarned) {
616 sim.overflowWarned = true;
617 std::fprintf(stderr,
618 "[particles] buffer overflow (emissionRate=%.1f, buffer=%d)\n",
619 cfg.emissionRate, int(sim.particles.size()));
620 }
621 sim.emitAccum = 0.f;
622 break;
623 }
624 if (!spawnWithInherit()) {
625 sim.emitAccum = 0.f;
626 break;
627 }
628 sim.emitAccum -= 1.f;
629 }
630 }
631
632 // Rate over distance is accumulated in particle units and spawned at
633 // evenly spaced points along the emitter's travelled segment. This avoids
634 // the frame-rate-dependent clumps produced by spawning every item at the
635 // current endpoint (important for weapon trails and wheel dust).
636 const float distanceRate = cfg.emissionRateOverDistance * emissionScale;
637 if (distanceRate > 0.f && sim.hasLastPos) {
638 const float moveX = cfg.x - sim.lastX;
639 const float moveY = cfg.y - sim.lastY;
640 const float distance = std::sqrt(moveX * moveX + moveY * moveY);
641 const float prior = sim.distanceEmitAccum;
642 const float total = prior + distance * distanceRate;
643 const int wanted = int(std::floor(total));
644 sim.distanceEmitAccum = total - float(wanted);
645 const float firstDistance = prior > 0.f ? (1.f - prior) / distanceRate : 1.f / distanceRate;
646 for (int i = 0; i < wanted; ++i) {
647 if (sim.alive >= int(sim.particles.size())) {
648 if (cfg.overflowMode == "pause")
649 sim.distanceEmitAccum += float(wanted - i);
650 else if (cfg.overflowMode == "warn" && !sim.overflowWarned) {
651 sim.overflowWarned = true;
652 std::fprintf(stderr, "[particles] distance emission overflow (rate=%.2f, buffer=%d)\n",
653 cfg.emissionRateOverDistance, int(sim.particles.size()));
654 }
655 break;
656 }
657 float spawnX = cfg.x;
658 float spawnY = cfg.y;
659 if (distance > kEps && !localSpace) {
660 const float along = firstDistance + float(i) / distanceRate;
661 const float alpha = std::min(1.f, along / distance);
662 spawnX = sim.lastX + moveX * alpha;
663 spawnY = sim.lastY + moveY * alpha;
664 }
665 const bool spawned = spawnParticleAt(cfg, sim, spawnX, spawnY);
666 if (!spawned) break;
667 if (cfg.inheritVelocity > 0.f && sim.alive > 0) {
668 Particle& np = sim.particles[size_t(sim.alive - 1)];
669 np.vx += emitVx * cfg.inheritVelocity;
670 np.vy += emitVy * cfg.inheritVelocity;
671 }
672 }
673 }
674
675 // Expire AFTER this frame's emission so a short-lived emitter still
676 // releases the particles due during its final frame.
677 if (cfg.emitterLife >= 0.f && sim.emitterAge >= cfg.emitterLife) {
678 if (cfg.looping && cfg.emitterLife > 0.f) {
679 sim.emitterAge = std::fmod(sim.emitterAge, cfg.emitterLife);
680 for (auto& b : cfg.bursts) b.emitted = false;
681 } else {
682 sim.active = false;
683 sim.emitAccum = 0.f;
684 sim.distanceEmitAccum = 0.f;
685 }
686 }
687
688 sim.lastX = cfg.x;
689 sim.lastY = cfg.y;
690 sim.hasLastPos = true;
691}

707 (void)e->gpuSim();
