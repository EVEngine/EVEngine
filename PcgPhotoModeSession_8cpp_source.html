23 PcgPhotoModeValues working = captured;

25 if (loadSaved && savedEver && currentPipeline == savedPipeline && !savedJson.empty()) {

32 if (!savedScene.ok()) return Result<void>::failure(savedScene.status());
33 if (!savedLighting.ok()) return Result<void>::failure(savedLighting.status());
34 if (!savedPcg.ok()) return Result<void>::failure(savedPcg.status());
35 const bool lightingMatches = savedLighting.value() == lightingProfile &&
36 savedPcg.value() == (lightingProfile >= 0);
37 if (savedScene.value() == sceneName && lightingMatches) {
38 working = std::move(saved);
39 decision = PcgPhotoModeLoadDecision::Saved;
40 } else {
41 decision = PcgPhotoModeLoadDecision::SaveCurrent;
42 }
43 }
44

46 working_ = working;
47 output_ = working_;
48 loadDecision_ = decision;
49 active_ = true;
50 restoreRequested_ = false;
51 removePhotoCameraRequested_ = false;
52 unfreezePlayerRequested_ = false;
53 ++revision_;

68Result<void> PcgPhotoModeSession::end(bool resetOnDisable, bool applicationPlaying) {
69 if (!active_) return fail("photo-mode session is not active");
70 removePhotoCameraRequested_ = true;
71 unfreezePlayerRequested_ = true;
72 restoreRequested_ = resetOnDisable && applicationPlaying;
