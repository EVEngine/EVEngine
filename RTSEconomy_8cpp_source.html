66eve::Result<RTSEffectUpdate> RTSEffectExecutor::settle(RTSEffectTarget& target,
67 effects::EffectUpdateSummary lifecycle) const {
68 auto valid = validate(target);
69 if (!valid) return eve::Result<RTSEffectUpdate>::failure(valid.status());
70 RTSEffectUpdate result;
71 result.lifecycle = std::move(lifecycle);
72 for (const auto& tick : result.lifecycle.periodicTicks) {
73 if (tick.stackCount == 0 || !std::isfinite(tick.magnitude) || tick.magnitude < 0.0)
74 return eve::Result<RTSEffectUpdate>::failure(eve::Diagnostic::error(
75 eve::DiagnosticCode::InvalidArgument, "RTS periodic effect value is invalid", "periodic"));
76 const auto amount = tick.magnitude * static_cast<double>(tick.stackCount);
77 if (!std::isfinite(amount))
78 return eve::Result<RTSEffectUpdate>::failure(eve::Diagnostic::error(
79 eve::DiagnosticCode::InvalidArgument, "RTS periodic effect overflows", "periodic"));
80 const auto tagged = [&tick](const char* tag) {
81 return std::find(tick.tags.begin(), tick.tags.end(), tag) != tick.tags.end();
82 };
83 if (tagged("rts:morale")) {
84 target.morale = std::max(0.0, target.morale - amount);
85 if (target.morale == 0.0) ++target.commandInterrupts;
86 } else if (tagged("rts:suppression")) {
87 if (target.suppression > std::numeric_limits<std::uint32_t>::max() - tick.stackCount)
88 return eve::Result<RTSEffectUpdate>::failure(eve::Diagnostic::error(
89 eve::DiagnosticCode::InvalidArgument, "RTS suppression count overflows", "target.suppression"));
90 target.suppression += tick.stackCount;
91 } else if (tagged("rts:production-lock")) {
92 target.productionLocked = true;
93 ++result.commandInterrupts;
94 }
95 ++result.settled;
96 }
97 return eve::Result<RTSEffectUpdate>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
98}

100eve::Result<effects::EffectHandle> RTSEffectAdapter::apply(const RTSEffectDefinition& definition,
101 eve::SubjectRef subject) {
102 if (definition.id.empty() || !subject.isValid())
103 return eve::Result<effects::EffectHandle>::failure(eve::Diagnostic::error(
104 eve::DiagnosticCode::InvalidArgument, "RTS effect requires an id and valid subject", "effect"));
105 effects::EffectDefinition common;
106 common.id = definition.id;
107 common.stackKey = definition.id;
108 common.duration = definition.duration;
109 common.period = definition.period;
110 common.magnitude = definition.magnitude;
111 common.policy = definition.policy;
112 common.payload.setNumber("speedMultiplier", definition.speedMultiplier);
113 common.payload.setNumber("damageMultiplier", definition.damageMultiplier);
114 common.payload.setNumber("incomingDamageMultiplier", definition.incomingDamageMultiplier);
115 common.payload.setNumber("healingPerSecond", definition.healingPerSecond);
116 common.tags = definition.tags;
117 common.tags.push_back(kindTag(definition.kind));
118 auto valid = common.validate();
119 if (!valid) return eve::Result<effects::EffectHandle>::failure(valid.status());
120 auto candidate = container_.clone();
121 auto applied = candidate.apply(common, subject.format(), definition.source);
122 if (!applied) return eve::Result<effects::EffectHandle>::failure(applied.status());
123 const std::string id = std::move(applied).takeValue();
124 const auto* instance = candidate.find(id);
125 if (!instance)
126 return eve::Result<effects::EffectHandle>::failure(eve::Diagnostic::error(
127 eve::DiagnosticCode::InvariantViolation, "RTS effect candidate disappeared", "effect"));
128 auto targetCandidate = target_;
129 auto targetValid = executor_.applyImmediate(targetCandidate, *instance);
130 if (!targetValid) return eve::Result<effects::EffectHandle>::failure(targetValid.status());
131 auto handle = candidate.handleFor(id);
132 if (!handle) return eve::Result<effects::EffectHandle>::failure(handle.status());
133 target_ = std::move(targetCandidate);
134 container_ = std::move(candidate);
135 return eve::Result<effects::EffectHandle>::success(std::move(handle).takeValue(),
136 eve::Status::success(eve::StatusCode::Applied));
137}

145eve::Result<RTSEffectUpdate> RTSEffectAdapter::advance(const eve::SimulationStep& step) {
146 auto candidate = container_.clone();
147 auto lifecycle = effects::EffectExecutor{}.advance(candidate, step);
148 if (!lifecycle) return eve::Result<RTSEffectUpdate>::failure(lifecycle.status());
149 auto targetCandidate = target_;
150 auto settled = executor_.settle(targetCandidate, std::move(lifecycle).takeValue());
151 if (!settled) return settled;
152 auto result = std::move(settled).takeValue();
153 target_ = std::move(targetCandidate);
154 container_ = std::move(candidate);
155 return eve::Result<RTSEffectUpdate>::success(std::move(result), eve::Status::success(eve::StatusCode::Applied));
156}

175double RTSEffectAdapter::multiplier(std::string_view field) const noexcept {
176 double result = 1.0;
177 for (int index = 0; index < container_.effectCount(); ++index) {
178 const auto* effect = container_.effectAt(index);
179 if (effect == nullptr) continue;
180 const auto found = effect->payload.object().find(std::string(field));
181 if (found == effect->payload.object().end()) continue;
182 if (const auto* real = found->second.getIf<double>())
183 result *= *real;
184 else if (const auto* integer = found->second.getIf<std::int64_t>())
185 result *= static_cast<double>(*integer);
186 }
187 return result;
188}
