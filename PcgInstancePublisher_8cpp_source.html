41 if (batchId.empty()) return failOp("editor.pcg.publish.batch-id", "publish requires a non-empty batch id");

43 editing::DomainOperation operation;
44 operation.type = kPublishType;
45 operation.inverseType = kRemoveType;
46 operation.hasInverse = true;
47 operation.mergeKey = std::string("pcg-instances:") + batchId;
48
49 EditorValue::Object payload;
50 payload["batchId"] = EditorValue(batchId);
51 payload["assetAttribute"] = EditorValue(assetAttribute.empty() ? std::string("mesh") : assetAttribute);
52 payload["defaultAsset"] = EditorValue(defaultAsset);
53 operation.payload = EditorValue(std::move(payload));

55 EditorValue::Object inverse;

57 operation.inverse = EditorValue(std::move(inverse));

89EditorResult<void> PcgInstancePublisher::apply(procgen::Procgen& procgen,
90 const editing::DomainOperation& operation) const {
91 editing::DomainOperation removeOp = operation;
92 if (operation.type == kPublishType && operation.hasInverse) {
93 removeOp.type = kRemoveType;
94 removeOp.payload = operation.inverse;
95 }
96 if (removeOp.type != kRemoveType)
97 return failVoid("editor.pcg.operation", "unsupported PCG instance operation type");
98 const std::string batchId = stringField(removeOp.payload, "batchId");
99 if (batchId.empty()) return failVoid("editor.pcg.remove.batch-id", "remove payload is missing batchId");
100 auto removed = procgen.removeInstances(batchId);
101 if (!removed.ok()) return failVoid("editor.pcg.remove.failed", statusText(removed.status()));
102 return eve::editing::applied();
103}
