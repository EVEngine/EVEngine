15float resolveParam(const eve::stylize::AttackVfxLayerStartRequest& request, const char* key,
16 float fallback) {
17 if (request.layer) {
18 const auto found = request.layer->floatParams.find(key);
19 if (found != request.layer->floatParams.end()) return found->second;
20 }
21 if (request.skin) {
22 const auto found = request.skin->shakeProfile.find(key);
23 if (found != request.skin->shakeProfile.end()) return found->second;
24 }

29public:
30 eve::stylize::AttackVfxLayerRole role() const noexcept override {

34 eve::Result<eve::stylize::AttackVfxLayerHandle> start(
35 const eve::stylize::AttackVfxLayerStartRequest& request) override {
36 if (!request.layer)
37 return eve::Result<eve::stylize::AttackVfxLayerHandle>::failure(eve::Diagnostic::error(
38 eve::DiagnosticCode::InvalidArgument, "AttackVfx camera layer is null", "layer"));
39
40 eve::action::ActionCameraCueBinding binding;
41 if (!request.layer->uri.empty()) {
42 auto cue = eve::LogicalId::parse(request.layer->uri);
43 if (!cue)
44 return eve::Result<eve::stylize::AttackVfxLayerHandle>::failure(eve::Diagnostic::error(
45 eve::DiagnosticCode::InvalidArgument,
46 "camera layer uri must be a LogicalId cue when set", "uri"));
47 binding.cue = std::move(*cue);
48 } else {
49 auto cue = eve::LogicalId::parse("attackvfx:camera");
50 if (!cue)
51 return eve::Result<eve::stylize::AttackVfxLayerHandle>::failure(eve::Diagnostic::error(
52 eve::DiagnosticCode::InvariantViolation, "default camera cue id is invalid", "cue"));
53 binding.cue = std::move(*cue);
54 }
55
56 const float posAmp = resolveParam(request, "posAmp", static_cast<float>(binding.positionAmplitude));
57 const float rotAmp = resolveParam(request, "rotAmp", static_cast<float>(binding.rotationAmplitude));
58 const float fovAmp = resolveParam(request, "fovAmp", static_cast<float>(binding.fovAmplitude));
59 float duration = resolveParam(request, "duration", 0.0f);
60 if (duration <= 0.0f) duration = resolveParam(request, "durationSeconds", 0.25f);
61 if (posAmp < 0.0f || rotAmp < 0.0f || duration <= 0.0f)
62 return eve::Result<eve::stylize::AttackVfxLayerHandle>::failure(eve::Diagnostic::error(
63 eve::DiagnosticCode::InvalidArgument, "camera layer amplitudes/duration are invalid",
64 "floatParams"));
65
66 binding.positionAmplitude = posAmp;
67 binding.rotationAmplitude = rotAmp;
68 binding.fovAmplitude = fovAmp;
69 auto parsedDuration = eve::Duration::fromSeconds(duration);
70 if (!parsedDuration)
71 return eve::Result<eve::stylize::AttackVfxLayerHandle>::failure(parsedDuration.status());
72 binding.duration = std::move(parsedDuration).takeValue();
73 if (request.playRequest)
74 binding.seed = request.playRequest->sourceId ^ (request.playRequest->targetId << 16);
75
76 eve::action::ActionNotifyContext context;
77 std::optional<eve::Result<void>> result;
78 eve::cap::forEachUntil<eve::action::IActionCameraCueSink>([&](eve::action::IActionCameraCueSink* sink) {
79 if (!sink->supports(binding.cue)) return false;
80 result.emplace(sink->trigger(binding, context));
81 return true;
82 });
83 if (!result)
84 return eve::Result<eve::stylize::AttackVfxLayerHandle>::failure(eve::Diagnostic::error(
85 eve::DiagnosticCode::NotFound, "no camera sink accepts the AttackVfx cue",
86 binding.cue.format()));
87 if (!*result) return eve::Result<eve::stylize::AttackVfxLayerHandle>::failure(result->status());
88
89 const auto id = ++nextId_;
90 live_.emplace(id, true);
91 return eve::Result<eve::stylize::AttackVfxLayerHandle>::success(
92 eve::stylize::AttackVfxLayerHandle{id});
93 }

95 eve::Result<void> update(eve::stylize::AttackVfxLayerHandle handle, double,
96 const eve::stylize::AttackVfxLayerStartRequest&) override {
97 if (!live_.contains(handle.id))
98 return eve::Result<void>::failure(eve::Diagnostic::error(
99 eve::DiagnosticCode::StaleHandle, "camera AttackVfx layer handle is stale", "handle"));
100 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
101 }
102
103 eve::Result<void> stop(eve::stylize::AttackVfxLayerHandle handle,
104 eve::stylize::AttackVfxStopBehavior) override {
105 const auto found = live_.find(handle.id);
106 if (found == live_.end())
107 return eve::Result<void>::failure(eve::Diagnostic::error(
108 eve::DiagnosticCode::StaleHandle, "camera AttackVfx layer handle is stale", "handle"));
109 // IActionCameraCueSink has no cancel API yet; impulses run for their authored duration.
110 // Clearing the local handle still makes the AttackVfx layer identity stale immediately.
111 live_.erase(found);
112 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
113 }

115private:

117 std::unordered_map<std::uint64_t, bool> live_;
118};
