20 if (!model) throw eve::Exception("SpriteStack.sliceModel: null model");
21 std::vector<float> pos, nrm, rgb;
22 std::vector<uint32_t> idx;
23 const int meshCount = model->getMeshCount();
24 for (int m = 0; m < meshCount; ++m) {
25 const aiMesh *am = model->getMesh(m);
26 if (!am) continue;
27 const uint32_t base = uint32_t(pos.size() / 3);
28 for (unsigned v = 0; v < am->mNumVertices; ++v) {
29 const aiVector3D &p = am->mVertices[v];
30 pos.insert(pos.end(), {p.x, p.y, p.z});
31 if (am->mNormals) {
32 const aiVector3D &n = am->mNormals[v];
33 nrm.insert(nrm.end(), {n.x, n.y, n.z});
34 }
35 if (am->mColors[0] != nullptr) {
36 const aiColor4D &c = am->mColors[0][v];
37 rgb.insert(rgb.end(), {c.r, c.g, c.b});
38 }
39 }
40 for (unsigned f = 0; f < am->mNumFaces; ++f) {
41 const auto &face = am->mFaces[f];
42 if (face.mNumIndices != 3) continue;
43 idx.insert(idx.end(),
44 {base + face.mIndices[0], base + face.mIndices[1], base + face.mIndices[2]});
45 }
46 }
47 SliceInput in{};
48 in.posXYZ = pos.data();
49 in.nrmXYZ = nrm.empty() ? nullptr : nrm.data();
50 in.rgb = rgb.empty() ? nullptr : rgb.data();
51 in.vertexCount = int(pos.size() / 3);
52 in.indices = idx.data();
53 in.indexCount = int(idx.size());
54 return sliceMeshToLayers(in, opt);
