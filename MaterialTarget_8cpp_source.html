57EditorResult<void> MaterialDocumentTarget::applyDomainOperation(const DomainOperation& operation) {
58 if (operation.target != TargetId(id_))
59 return materialError<void>(EditorStatus::Rejected, "editor.material.target-mismatch",
60 "Material operation targets another document");
61 if (operation.type != "material.property.set.v1")
62 return materialError<void>(EditorStatus::Unsupported, "editor.material.operation-unsupported",
63 "Unsupported material operation: " + operation.type);
64 const EditorValue* pathValue = objectField(operation.payload, "path");
65 const EditorValue* value = objectField(operation.payload, "value");
66 const auto* path = pathValue ? pathValue->getIf<std::string>() : nullptr;
67 if (!path || !value)
68 return materialError<void>(EditorStatus::Rejected, "editor.material.operation-payload",
69 "Material operation requires path and value");
70 auto descriptor = materialSchema().find(PropertyPath(*path));
71 if (!descriptor)
72 return materialError<void>(EditorStatus::Unsupported, "editor.material.property-unsupported",
73 "Unknown material property: " + *path);
74 auto valid = validateAssignment(*descriptor, *value);
75 if (!valid.ok()) return valid;
76 values_[*path] = *value;
77 ++revision_;
78 dirty_.include(0, 0);
79 return eve::editing::applied<void>();
80}

114EditorResult<DomainOperation> MaterialDocumentTarget::makeSet(const SelectionSnapshot& selection,
115 const PropertyPath& path,
116 const EditorValue& value,
117 PropertySetMode mode) const {
118 if (!selectionMatches(selection))
119 return materialError<DomainOperation>(EditorStatus::Rejected, "editor.material.selection",
120 "Selection does not belong to this material");
121 if (mode == PropertySetMode::Reset) return makeReset(selection, path);
122 if (mode != PropertySetMode::Absolute)
123 return materialError<DomainOperation>(EditorStatus::Unsupported, "editor.material.set-mode",
124 "Material properties require absolute assignment");
125 auto descriptor = materialSchema().find(path);
126 if (!descriptor)
127 return materialError<DomainOperation>(EditorStatus::Unsupported, "editor.material.property-unsupported",
128 "Unknown material property: " + path.value());
129 auto valid = validateAssignment(*descriptor, value);
130 if (!valid.ok()) return EditorResult<DomainOperation>::failure(valid.status());
131 const auto previous = values_.find(path.value());
132 if (previous == values_.end())
133 return materialError<DomainOperation>(EditorStatus::NotFound, "editor.material.property-missing",
134 "Material property has no current value");
135 auto payload = [&](const EditorValue& assigned) {
136 EditorValue::Object object;
137 object["path"] = path.value();
138 object["value"] = assigned;
139 return EditorValue(std::move(object));
140 };
141 DomainOperation operation;
142 operation.type = "material.property.set.v1";
143 operation.target = TargetId(id_);
144 operation.payload = payload(value);
145 operation.inverse = payload(previous->second);
146 operation.hasInverse = true;
147 operation.affectedProperties.push_back(path.value());
148 operation.mergeKey = "material:" + id_ + ":" + path.value();
149 return eve::editing::applied<DomainOperation>(std::move(operation));
150}

170EditorResult<void> MaterialDocumentTarget::loadSnapshot(const EditorValue& snapshot) {
171 const EditorValue* versionValue = objectField(snapshot, "schemaVersion");
172 const EditorValue* propertiesValue = objectField(snapshot, "properties");
173 const auto* version = versionValue ? versionValue->getIf<int64_t>() : nullptr;
174 const auto* properties = propertiesValue ? propertiesValue->getIf<EditorValue::Object>() : nullptr;
175 if (!version || *version != 1 || !properties)
176 return materialError<void>(EditorStatus::Rejected, "editor.material.snapshot-format",
177 "Material snapshot requires schemaVersion 1 and properties");
178 auto candidate = defaults();
179 const PropertySchema schemaValue = materialSchema();
180 for (const auto& [path, value] : *properties) {
181 auto descriptor = schemaValue.find(PropertyPath(path));
182 if (!descriptor)
183 return materialError<void>(EditorStatus::Unsupported, "editor.material.snapshot-property",
184 "Material snapshot contains unknown property: " + path);
185 auto valid = validateAssignment(*descriptor, value);
186 if (!valid.ok()) return valid;
187 candidate[path] = value;
188 }
189 values_ = std::move(candidate);
190 ++revision_;
191 dirty_.clear();
192 return eve::editing::applied<void>();
193}

195std::vector<EditorDiagnostic> MaterialDocumentTarget::validate() const {
196 std::vector<EditorDiagnostic> diagnostics;
197 const auto stringValue = [&](const char* path) -> std::string {
198 const auto found = values_.find(path);
199 if (found == values_.end()) return {};
200 const auto* value = found->second.getIf<std::string>();
201 return value ? *value : std::string{};
202 };
203 if (stringValue("shading.model") == "custom" && stringValue("textures.shader").empty())
204 diagnostics.push_back(eve::editing::ruleDiagnostic(
205 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.material.custom-shader-required"),
206 DiagnosticSeverity::Error, "Custom shading requires a shader asset"));
207 if (stringValue("surface.mode") == "masked" && stringValue("textures.albedo").empty())
208 diagnostics.push_back(eve::editing::ruleDiagnostic(
209 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.material.mask-without-albedo"),
210 DiagnosticSeverity::Warning, "Masked material has no albedo texture providing alpha"));
211 if (!stringValue("textures.height").empty()) {
212 const auto* scale = values_.at("parallax.scale").getIf<double>();
213 if (scale && *scale == 0.0)
214 diagnostics.push_back(eve::editing::ruleDiagnostic(
215 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.material.height-without-parallax"),
216 DiagnosticSeverity::Info, "Height texture is assigned while parallax scale is zero"));
217 }
218 return diagnostics;
219}

221PropertySchema MaterialDocumentTarget::materialSchema() {
222 PropertySchema schema;
223 schema.typeId = "graphics.material";
224 schema.version = 1;
225 auto shading = property("shading.model", "editor.material.shading-model", "shading",
226 PropertyType::Enum, "pbr");

228 schema.properties.push_back(std::move(shading));
229 auto tint = property("shading.tint", "editor.material.tint", "shading", PropertyType::Color,
230 EditorValue::Array{1.0, 1.0, 1.0, 1.0});
231 schema.properties.push_back(std::move(tint));
232 auto numeric = [&](const char* path, const char* label, double value, double minimum, double maximum) {
233 auto descriptor = property(path, label, "shading", PropertyType::Float, value);
234 descriptor.numeric.minimum = minimum;
235 descriptor.numeric.maximum = maximum;
236 descriptor.numeric.step = 0.01;
237 schema.properties.push_back(std::move(descriptor));
238 };
239 numeric("shading.metallic", "editor.material.metallic", 0.0, 0.0, 1.0);
240 numeric("shading.roughness", "editor.material.roughness", 0.45, 0.04, 1.0);
241 for (const auto& [path, label] : {std::pair{"textures.albedo", "editor.material.albedo"},
242 {"textures.normal", "editor.material.normal"},
243 {"textures.height", "editor.material.height"},
244 {"textures.shader", "editor.material.shader"}}) {
245 auto descriptor = property(path, label, "textures", PropertyType::AssetRef, "");
246 descriptor.assetTypeFilters = path == std::string("textures.shader")
247 ? std::vector<std::string>{"shader"}
248 : std::vector<std::string>{"texture"};
249 schema.properties.push_back(std::move(descriptor));
250 }
251 auto surface = property("surface.mode", "editor.material.surface-mode", "surface",
252 PropertyType::Enum, "opaque");
253 surface.enumItems = {"opaque", "masked", "transparent"};
254 schema.properties.push_back(std::move(surface));
255 auto blend = property("surface.blend", "editor.material.blend-mode", "surface",
256 PropertyType::Enum, "alpha");
257 blend.enumItems = {"alpha", "premultiplied", "additive", "multiply"};
258 schema.properties.push_back(std::move(blend));
259 numeric("surface.alpha-cutoff", "editor.material.alpha-cutoff", 0.5, 0.0, 1.0);
260 numeric("parallax.scale", "editor.material.parallax-scale", 0.0, 0.0, 0.25);
261 for (const auto& [path, label, value] : {
262 std::tuple{"surface.depth-write", "editor.material.depth-write", false},
263 {"surface.double-sided", "editor.material.double-sided", false},
264 {"lighting.receive", "editor.material.receive-light", true},
265 {"shadow.cast", "editor.material.cast-shadow", true},
266 {"shadow.receive", "editor.material.receive-shadow", true}})
267 schema.properties.push_back(property(path, label, "surface", PropertyType::Bool, value));
268 return schema;
269}
