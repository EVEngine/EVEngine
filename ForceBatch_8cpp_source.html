40Result<Sums> aggregate(W& world, std::span<const ForceLoad> loads, bool spatial) {
41 Sums sums;
42 for (const auto& load : loads) {
43 auto resolved = load.target.resolve(world);
44 if (!resolved) return Result<Sums>::failure(resolved.status());
45 auto& body = *resolved.value();
46 if (!body.isActive() || body.getType() != "dynamic" || !fits(load.force) || !fits(load.point) ||
47 !fits(load.torque) ||
48 (!spatial && (load.force.z != 0 || load.point.z != 0 || load.torque.x != 0 || load.torque.y != 0)))
49 return invalid<Sums>();
50 const auto torque = add(load.torque, cross(add(load.point, scale(center(body), -1)), load.force));
51 if (!fits(torque)) return invalid<Sums>();
52 auto& sum = sums[load.target.body];
53 sum.force = add(sum.force, load.force);
54 sum.torque = add(sum.torque, torque);
55 if (!fits(sum.force) || !fits(sum.torque)) return invalid<Sums>();
56 }
57 return Result<Sums>::success(std::move(sums));
58}
