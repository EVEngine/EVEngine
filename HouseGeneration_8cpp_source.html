18Result<void> house_detail::State::plan() {
19 if (planned) return Result<void>::success();
20 if (!isPolygon && z > 0 && inset == 0 && r.width > 4 && r.depth > 4 && (rng() & 3u) == 0u) inset = 1;
21 auto mask = makeMask(inset);
22 if (!previousMask.empty()) {
23 for (size_t i = 0; i < mask.size(); ++i) mask[i] = uint8_t(mask[i] && previousMask[i]);
24 }
25 procgen::Grid2D grid;
26 grid.resize(r.width, r.depth);
27 for (int y = 0; y < r.depth; ++y)
28 for (int x = 0; x < r.width; ++x) grid.setCell(x, y, mask[size_t(y * r.width + x)]);
29 planned = std::move(grid);
30 return Result<void>::success();
31}

53Result<void> HouseGeneration::emitFloor(const procgen::Grid2D& grid, const std::vector<std::string>& rooms) {
54 if (!state_ || !state_->planned || state_->z >= state_->r.floors)
55 return invalidStage("prepare nextFloorGrid before emitting a floor");
56 const auto& r = state_->r;
57 if (grid.getWidth() != r.width || grid.getHeight() != r.depth || rooms.empty() ||
58 std::any_of(rooms.begin(), rooms.end(), [](const auto& name) { return name.empty(); }))
59 return invalidStage("floor grid dimensions and nonempty room types must match the request");
60 if (std::none_of(grid.cells().begin(), grid.cells().end(), [](auto cell) { return cell != 0; }))
61 return invalidStage("house floor must contain occupied cells");
62 if (!state_->previousMask.empty()) {
63 for (size_t i = 0; i < grid.cells().size(); ++i)
64 if (grid.cells()[i] && !state_->previousMask[i]) return invalidStage("upper floor lacks direct support");
65 }
66 if (rooms.size() > 1 && (state_->library_.byCategory("interior_wall", r.style).empty() ||
67 state_->library_.byCategory("interior_door", r.style).empty()))
68 return invalidStage("multi-room floor requires interior wall and door components");
69 auto candidate = std::make_unique<house_detail::State>(*state_);
70 if (candidate->z == 0) {
71 std::transform(grid.cells().begin(), grid.cells().end(), candidate->baseMask.begin(),
72 [](auto cell) { return uint8_t(cell != 0); });
73 if (candidate->stairwell) {
74 candidate->stairwell = house_detail::findInteriorCell(candidate->baseMask, r.width, r.depth);
75 if (!candidate->stairwell) return invalidStage("edited floor has no interior stair cell");
76 }
77 } else if (candidate->stairwell) {
78 auto [x, y] = *candidate->stairwell;
79 if (!grid.getCell(x, y)) return invalidStage("upper floor removes the stair column");
80 }
81 auto result = candidate->floor(grid, rooms);
82 if (!result.ok()) return result;
83 state_ = std::move(candidate);
84 return Result<void>::success();
85}
