47Result<int> TerrainCollisionMaskStack::apply(Heightmap& target, const Heightmap& input, const Heightmap& curve,
48 TerrainMaskBlend mode) const {
49 if (!impl_ || !validRaster(target) || !validRaster(input) || !validRaster(curve) ||
50 target.getWidth() != input.getWidth() || target.getHeight() != input.getHeight() ||
51 curve.getHeight() != 1 || mode < TerrainMaskBlend::Multiply || mode > TerrainMaskBlend::Subtract)
52 return invalid("terrain.collisionMask.apply: compatible finite target, input, curve and mode required");
53 std::vector<float> output(target.data().size());
54 for (int z = 0; z < target.getHeight(); ++z) {
55 for (int x = 0; x < target.getWidth(); ++x) {
56 const double u = (x + 0.5) / target.getWidth();
57 const double v = (z + 0.5) / target.getHeight();
58 double collisionFree = 1;
59 for (const auto& layer : impl_->layers) {
60 if (!layer.active) continue;
61 double baked = textureSample(layer.mask, u, v);
62 const bool layerType = layer.type == TerrainCollisionMaskType::LayerGameObject ||
63 layer.type == TerrainCollisionMaskType::LayerTree;
64 if (layerType ? !layer.invert : layer.invert) baked = 1 - baked;
65 collisionFree = std::min(collisionFree, baked);
66 }
67 const double filtered = curveSample(curve, collisionFree);
68 const std::size_t index = static_cast<std::size_t>(z) * target.getWidth() + x;
69 const double old = input.data()[index];
70 double result = old;
71 switch (mode) {
72 case TerrainMaskBlend::Multiply: result = old * filtered; break;
73 case TerrainMaskBlend::Maximum: result = std::max(old, filtered); break;
74 case TerrainMaskBlend::Minimum: result = std::min(old, filtered); break;
75 case TerrainMaskBlend::Add: result = old + filtered; break;
76 case TerrainMaskBlend::Subtract: result = old - filtered; break;
77 }
78 if (!isRepresentable(result))
79 return invalid("terrain.collisionMask.apply: output exceeds finite float range");
80 output[index] = static_cast<float>(result);
81 }
82 }
83 return publish(target, std::move(output));
84}
