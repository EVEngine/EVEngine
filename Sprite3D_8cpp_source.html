59void Sprite3D::updateFrameUv() {
60 if (!gfx_ || !quad_) return;
61 std::vector<float> uv(8);
62 for (int i = 0; i < 4; ++i) {
63 float u = (i == 0 || i == 3) ? u0_ : u1_;
64 float v = (i < 2) ? v0_ : v1_;
65 if (flipX_) u = (u0_ + u1_) - u;
66 if (flipY_) v = (v0_ + v1_) - v;
67 uv[size_t(i) * 2u] = u;
68 uv[size_t(i) * 2u + 1u] = v;
69 }
70 // Positions are required by updateMeshVertices; pass the unchanged base.
71 gfx_->updateMeshVertices(quad_, kQuadPos.data(), kQuadNrm.data(), uv.data(),
72 int(kQuadPos.size() / 3), nullptr, 0);
73}

137void Sprite3D::update(float dt) {
138 if (anim_.playing && anim_.fps > 0.f) {
139 anim_.clock += dt;
140 const float frameDur = 1.f / anim_.fps;
141 if (anim_.clock >= frameDur) {
142 int steps = int(anim_.clock / frameDur);
143 anim_.clock -= float(steps) * frameDur;
144 const int span = anim_.end - anim_.start + 1;
145 if (span > 0) {
146 frameIndex_ = anim_.start + (frameIndex_ - anim_.start + steps) % span;
147 setFrameIndex(frameIndex_);
148 }
149 }
150 }

159void Sprite3D::orientToCamera() {
160 if (!renderable_ || !camera_) return;
161 const auto &cam = *camera_->data();
162 const glm::vec3 eye(cam.eyeX, cam.eyeY, cam.eyeZ);
163 const glm::vec3 pos(x_, y_, z_);
164 // The billboard quad's front face is -Z, so it faces the camera when its
165 // +Z points away from the eye. Cylindrical billboard: rotate around world Y
166 // only, keeping the sprite upright (no pitch) so it faces the camera
167 // squarely while standing vertical.
168 glm::vec3 away = pos - eye;
169 if (glm::dot(away, away) < 1e-6f) away = glm::vec3(0.f, 0.f, 1.f);
170 away = glm::normalize(away);
171 const float yaw = std::atan2(away.x, away.z);
172 renderable_->setRotation(yaw, 0.f, 0.f);
173}
