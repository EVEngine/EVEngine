25Result<void> HitFeelRuntime::apply(HitFeelRequest request) {
26 auto valid = request.validate();
27 if (!valid) return valid;
28 auto& attacker = states_[request.attacker.format()];
29 attacker.subject = request.attacker;
30 attacker.hitstopRemaining = maxDuration(attacker.hitstopRemaining, request.attackerHitstop);
31 attacker.timeFrozen = attacker.hitstopRemaining > Duration::zero();
32
33 auto& victim = states_[request.victim.format()];
34 victim.subject = request.victim;
35 victim.hitstopRemaining = maxDuration(victim.hitstopRemaining, request.victimHitstop);
36 victim.hitstunRemaining = maxDuration(victim.hitstunRemaining, request.victimHitstun);
37 victim.timeFrozen = victim.hitstopRemaining > Duration::zero();
38 return Result<void>::success(Status::success(StatusCode::Applied));
39}

41Result<void> HitFeelRuntime::applyFromOutcome(const DamageOutcome& outcome) {
42 HitFeelRequest request;
43 request.attacker = outcome.source;
44 request.victim = outcome.target;
45 switch (outcome.reaction) {
46 case HitReaction::None:
47 request.attackerHitstop = Duration::fromNanoseconds(16000000);
48 request.victimHitstop = Duration::fromNanoseconds(16000000);
49 break;
50 case HitReaction::Flinch:
51 request.attackerHitstop = Duration::fromNanoseconds(33000000);
52 request.victimHitstop = Duration::fromNanoseconds(33000000);
53 request.victimHitstun = Duration::fromNanoseconds(120000000);
54 break;
55 case HitReaction::Stagger:
56 request.attackerHitstop = Duration::fromNanoseconds(50000000);
57 request.victimHitstop = Duration::fromNanoseconds(50000000);
58 request.victimHitstun = Duration::fromNanoseconds(350000000);
59 break;
60 case HitReaction::Knockdown:
61 request.attackerHitstop = Duration::fromNanoseconds(66000000);
62 request.victimHitstop = Duration::fromNanoseconds(66000000);
63 request.victimHitstun = Duration::fromNanoseconds(800000000);
64 break;
65 case HitReaction::Death:
66 request.attackerHitstop = Duration::fromNanoseconds(80000000);
67 request.victimHitstop = Duration::fromNanoseconds(100000000);
68 request.victimHitstun = Duration::fromNanoseconds(1000000000);
69 break;
70 }
71 request.reaction = outcome.reaction;
72 return apply(std::move(request));
73}

75Result<HitFeelAdvance> HitFeelRuntime::advance(const SimulationStep& step) {
76 if (step.tick.value() < lastTick_.value())
77 return Result<HitFeelAdvance>::failure(
78 Diagnostic::error(DiagnosticCode::InvalidArgument, "hit-feel tick must be non-decreasing", "tick"));
79 if (step.delta < Duration::zero())
80 return Result<HitFeelAdvance>::failure(
81 Diagnostic::error(DiagnosticCode::InvalidArgument, "hit-feel delta is invalid", "delta"));
82
83 HitFeelAdvance advance;
84 advance.tick = step.tick;
85 for (auto it = states_.begin(); it != states_.end();) {
86 auto& state = it->second;
87 const bool hadHitstop = state.hitstopRemaining > Duration::zero();
88 const bool hadHitstun = state.hitstunRemaining > Duration::zero();
89 if (state.hitstopRemaining > Duration::zero()) {
90 state.hitstopRemaining = state.hitstopRemaining < step.delta
91 ? Duration::zero()
92 : Duration::fromNanoseconds(state.hitstopRemaining.nanoseconds() -
93 step.delta.nanoseconds());
94 }
95 if (state.hitstunRemaining > Duration::zero()) {
96 state.hitstunRemaining = state.hitstunRemaining < step.delta
97 ? Duration::zero()
98 : Duration::fromNanoseconds(state.hitstunRemaining.nanoseconds() -
99 step.delta.nanoseconds());
100 }
101 state.timeFrozen = state.hitstopRemaining > Duration::zero();
102 if (hadHitstop && state.hitstopRemaining.isZero())
103 advance.events.push_back({state.subject, HitFeelEventKind::HitstopEnded, step.tick});
104 if (hadHitstun && state.hitstunRemaining.isZero())
105 advance.events.push_back({state.subject, HitFeelEventKind::HitstunEnded, step.tick});
106 if (state.hitstopRemaining.isZero() && state.hitstunRemaining.isZero())
107 it = states_.erase(it);
108 else
109 ++it;
110 }
111 lastTick_ = step.tick;
112 return Result<HitFeelAdvance>::success(std::move(advance), Status::success(StatusCode::Applied));
113}
