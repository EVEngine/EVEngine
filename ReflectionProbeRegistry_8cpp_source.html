60int ReflectionProbeRegistry::queueCaptureAABB(float minX, float minY, float minZ, float maxX,
61 float maxY, float maxZ, int changedMask) {
62 const float changedMin[3] = {std::min(minX, maxX), std::min(minY, maxY),
63 std::min(minZ, maxZ)};
64 const float changedMax[3] = {std::max(minX, maxX), std::max(minY, maxY),
65 std::max(minZ, maxZ)};
66 const uint32_t mask = static_cast<uint32_t>(changedMask);
67 int queued = 0;
68 for (Entry &entry : entries_) {
69 ReflectionProbeCapture *probe = entry.probe;
70 if (!probe || (static_cast<uint32_t>(probe->getCaptureMask()) & mask) == 0u) continue;
71 const float center[3] = {probe->getCenterX(), probe->getCenterY(), probe->getCenterZ()};
72 float distance2 = 0.f;
73 for (int axis = 0; axis < 3; ++axis) {
74 const float nearest = std::clamp(center[axis], changedMin[axis], changedMax[axis]);
75 const float delta = center[axis] - nearest;
76 distance2 += delta * delta;
77 }
78 const float captureRadius = probe->getCaptureFarDistance();
79 if (distance2 > captureRadius * captureRadius) continue;
80 probe->queueCapture();
81 ++queued;
82 }
83 return queued;
84}

86int ReflectionProbeRegistry::tick(int faceBudget, int filterBudget, int filterSamples) {
87 lastCapturedFaceCount_ = 0;
88 lastPublishedCount_ = 0;
89 faceBudget = std::max(faceBudget, 0);
90 filterBudget = std::max(filterBudget, 0);
91 filterSamples = std::clamp(filterSamples, 8, 512);
92
93 for (Entry &entry : entries_) {
94 if (!entry.probe) continue;
95 entry.probe->advanceRefreshPolicy();
96 const bool waiting = entry.probe->isCapturePending() ||
97 entry.probe->getPublishedRevision() != entry.probe->getRevision();
98 if (waiting && entry.waitAge < std::numeric_limits<uint32_t>::max()) ++entry.waitAge;
99 }
100
101 auto better = [](const Entry *a, const Entry *b) {
102 const int priorityA = a->probe->getInfluencePriority();
103 const int priorityB = b->probe->getInfluencePriority();
104 if (priorityA != priorityB) return priorityA > priorityB;
105 if (a->waitAge != b->waitAge) return a->waitAge > b->waitAge;
106 return a->order < b->order;
107 };
108
109 while (faceBudget-- > 0) {
110 Entry *selected = nullptr;
111 for (Entry &entry : entries_) {
112 if (!entry.probe || !entry.probe->isCapturePending()) continue;
113 if (!selected || better(&entry, selected)) selected = &entry;
114 }
115 if (!selected) break;
116 selected->probe->update(1);
117 selected->waitAge = 0;
118 lastCapturedFaceCount_ += selected->probe->getLastCapturedFaceCount();
119 }
120
121 while (filterBudget-- > 0) {
122 Entry *selected = nullptr;
123 for (Entry &entry : entries_) {
124 if (!entry.probe || entry.probe->isCapturePending() ||
125 entry.probe->getRevision() == entry.probe->getPublishedRevision())
126 continue;
127 if (!selected || better(&entry, selected)) selected = &entry;
128 }
129 if (!selected) break;
130 if (selected->probe->stageCapturedFaces() &&
131 selected->probe->filterAndPublish(filterSamples)) {
132 selected->waitAge = 0;
133 ++lastPublishedCount_;
134 } else {
135 break;
136 }
137 }
138 return lastPublishedCount_;
139}

142 lastSelectedCount_ = 0;
143 lastCandidateCount_ = 0;
144 if (!camera) {
145 lastSelected_.clear();
146 return 0;
147 }

151 struct Candidate {
152 ReflectionProbeCapture *probe = nullptr;
153 int priority = 0;
154 float distance2 = 0.f;
155 bool retained = false;
156 uint64_t order = 0;
157 };
158 std::vector<Candidate> candidates;
159 candidates.reserve(entries_.size());
160 const auto cameraData = camera->data();
161 const float eye[3] = {cameraData->eyeX, cameraData->eyeY, cameraData->eyeZ};
162 const float target[3] = {cameraData->targetX, cameraData->targetY, cameraData->targetZ};
163 const float viewSegment[3] = {target[0] - eye[0], target[1] - eye[1],
164 target[2] - eye[2]};
165 const float viewSegmentLength2 = viewSegment[0] * viewSegment[0] +
166 viewSegment[1] * viewSegment[1] +
167 viewSegment[2] * viewSegment[2];
168 for (const Entry &entry : entries_) {
169 ReflectionProbeCapture *probe = entry.probe;
170 if (!probe || !probe->getActiveCubemap() || probe->getPublishedRevision() == 0 ||
171 probe->getInfluenceIntensity() <= 0.f)
172 continue;
173 const float center[3] = {probe->getCenterX(), probe->getCenterY(), probe->getCenterZ()};
174 const float extent[3] = {probe->getInfluenceExtentX(), probe->getInfluenceExtentY(),
175 probe->getInfluenceExtentZ()};
176 float closestT = 0.f;
177 if (viewSegmentLength2 > 1e-8f) {
178 const float toCenter[3] = {center[0] - eye[0], center[1] - eye[1],
179 center[2] - eye[2]};
180 closestT = std::clamp((toCenter[0] * viewSegment[0] +
181 toCenter[1] * viewSegment[1] +
182 toCenter[2] * viewSegment[2]) /
183 viewSegmentLength2,
184 0.f, 1.f);
185 }
186 const float closestViewPoint[3] = {
187 eye[0] + viewSegment[0] * closestT,
188 eye[1] + viewSegment[1] * closestT,
189 eye[2] + viewSegment[2] * closestT,
190 };
191 float distance2 = 0.f;
192 for (int axis = 0; axis < 3; ++axis) {
193 const float outside =
194 std::max(std::abs(closestViewPoint[axis] - center[axis]) - extent[axis], 0.f);
195 distance2 += outside * outside;
196 }
197 const bool retained =
198 std::find(lastSelected_.begin(), lastSelected_.end(), probe) != lastSelected_.end();
199 const float distance = std::sqrt(distance2);
200 const float biasedDistance =
201 retained ? std::max(distance - selectionHysteresis_, 0.f) : distance;
202 candidates.push_back(Candidate{probe, probe->getInfluencePriority(),
203 biasedDistance * biasedDistance, retained, entry.order});
204 }
205 lastCandidateCount_ = static_cast<int>(candidates.size());
206 std::stable_sort(candidates.begin(), candidates.end(),
207 [](const Candidate &a, const Candidate &b) {
208 if (a.priority != b.priority) return a.priority > b.priority;
209 if (a.distance2 != b.distance2) return a.distance2 < b.distance2;
210 if (a.retained != b.retained) return a.retained;
211 return a.order < b.order;
212 });
213
214 const int selected = std::min(static_cast<int>(candidates.size()),
215 Camera3D::Data::kMaxReflectionProbes);
216 lastSelected_.clear();
217 lastSelected_.reserve(static_cast<size_t>(selected));
218 for (int slot = 0; slot < selected; ++slot) {
219 ReflectionProbeCapture *probe = candidates[static_cast<size_t>(slot)].probe;
220 if (probe->applyConfiguredToCamera(camera, slot)) {
221 lastSelected_.push_back(probe);
222 ++lastSelectedCount_;
223 }
224 }
225 return lastSelectedCount_;
226}
