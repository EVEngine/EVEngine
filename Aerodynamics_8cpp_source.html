40Result<WingSample> ForceModels::wing(const WingModel& m, ForceVector velocity) {
41 constexpr double pi = std::numbers::pi;
42 if (!isFinite(velocity) || !isFinite(m.chord) || !isFinite(m.normal) || std::abs(length(m.chord) - 1) > 1e-5 ||
43 std::abs(length(m.normal) - 1) > 1e-5 || std::abs(dot(m.chord, m.normal)) > 1e-5 || !isNonnegative(m.density) ||
44 !isPositive(m.area) || !isNonnegative(m.liftSlope) || !isFinite(m.zeroLiftAngle) ||
45 std::abs(m.zeroLiftAngle) >= pi / 2 || !isPositive(m.stallAngle) || m.stallAngle >= pi / 2 ||
46 !isNonnegative(m.baseDrag) || !isPositive(m.aspectRatio) || !isPositive(m.efficiency) || m.efficiency > 1)
47 return invalid<WingSample>();
48 // Section model: spanwise flow creates no section lift/drag.
49 const double forward = dot(velocity, m.chord), up = dot(velocity, m.normal);
50 const auto sectionVelocity = add(scale(m.chord, forward), scale(m.normal, up));
51 const double speed = std::hypot(forward, up);
52 WingSample out;
53 if (speed == 0) return Result<WingSample>::success(out);
54 out.angleOfAttack = std::atan2(-up, forward);
55 const double alpha = std::remainder(out.angleOfAttack - m.zeroLiftAngle, 2 * pi);
56 const double magnitude = std::abs(alpha);
57 // Continuous falloff after stall, zero lift at 90 degrees and in reverse flow.
58 const double lift = magnitude <= m.stallAngle ? m.liftSlope * magnitude
59 : m.liftSlope * m.stallAngle *
60 std::max(0.0, (pi / 2 - magnitude) / (pi / 2 - m.stallAngle));
61 out.liftCoefficient = std::copysign(lift, alpha);
62 const double stallDrag = magnitude > m.stallAngle ? 2 * std::pow(std::sin(alpha), 2) : 0;
63 // Blend added pressure drag from zero at the stall threshold.
64 const double blend = std::clamp((magnitude - m.stallAngle) / (pi / 2 - m.stallAngle), 0.0, 1.0);
65 out.dragCoefficient = m.baseDrag + out.liftCoefficient * out.liftCoefficient / (pi * m.aspectRatio * m.efficiency) +
66 stallDrag * blend;
67 const auto flow = scale(sectionVelocity, 1 / speed);
68 const auto liftDirection = add(scale(m.normal, forward / speed), scale(m.chord, -up / speed));
69 const double qArea = 0.5 * m.density * speed * speed * m.area;
70 out.force = add(scale(liftDirection, qArea * out.liftCoefficient), scale(flow, -qArea * out.dragCoefficient));
71 if (!isFinite(out.force) || !isFinite(out.dragCoefficient) || !isFinite(out.liftCoefficient))
72 return invalid<WingSample>();
73 return Result<WingSample>::success(out);
74}
