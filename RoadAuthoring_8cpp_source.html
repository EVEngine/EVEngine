20 return std::fabs(area(a, b, p)) < 1e-5f && p.side >= std::min(a.side, b.side) - 1e-5f &&
21 p.side <= std::max(a.side, b.side) + 1e-5f && p.up >= std::min(a.up, b.up) - 1e-5f &&

35std::vector<T> clip(const std::vector<T>& intervals, float low, float high, float pathLength) {
36 std::vector<T> result;
37 for (const auto& original : intervals) {
38 const float a = std::max(low, original.start), b = std::min(high, original.end);
39 if (b - a <= 1e-5f) continue;
40 auto item = original;
41 item.start = (a - low) / (high - low);
42 item.end = (b - low) / (high - low);
43 if constexpr (std::is_same_v<T, RoadContinuousFacility>) {
44 if (item.prefab.automaticSpacing && item.prefab.repeatSourceLength == 0.f)
45 item.prefab.repeatSourceLength = pathLength * (original.end - original.start) /
46 (original.prefab.sourceEnd - original.prefab.sourceStart);
47 item.capStart = original.capStart && a == original.start;
48 item.capEnd = original.capEnd && b == original.end;
49 if (!original.prefab.vertices.empty()) {
50 item.prefab.sourceStart = std::lerp(original.prefab.sourceStart, original.prefab.sourceEnd,
51 (a - original.start) / (original.end - original.start));
52 item.prefab.sourceEnd = std::lerp(original.prefab.sourceStart, original.prefab.sourceEnd,
53 (b - original.start) / (original.end - original.start));
54 }
55 }
56 if constexpr (std::is_same_v<T, RoadStructureInterval>) {
57 item.portalStartLength = std::max(0.f, original.portalStartLength - (a - original.start) * pathLength);
58 item.portalEndLength = std::max(0.f, original.portalEndLength - (original.end - b) * pathLength);
59 }
60 result.push_back(std::move(item));
61 }
62 return result;
63}

65void reverseSectionGeometry(std::vector<RoadProfilePoint>& section, bool closed) {
66 if (section.empty()) return;
67 const auto old = section;
68 for (std::size_t i = 0; i < section.size(); ++i) {
69 const std::size_t source = section.size() - 1 - i;
70 section[i] = old[source];
71 section[i].side = -section[i].side;
72 if (source > 0)
73 section[i].material = old[source - 1].material;
74 else if (closed)
75 section[i].material = old.back().material;
76 }

78Result<void> validateSection(const std::vector<RoadProfilePoint>& points, bool closed) {
79 if (points.size() < (closed ? 3u : 2u) || points.size() > 128u)
80 return invalid("section needs 2..128 points (3 for closed)");
81 for (const auto& p : points)
82 if (!std::isfinite(p.side) || !std::isfinite(p.up) || std::fabs(p.side) > 10000.f ||
83 std::fabs(p.up) > 10000.f || p.material > RoadMaterial::TunnelPortalAsphalt)
84 return invalid("section has invalid coordinates or materials");
85 for (std::size_t i = closed ? 0u : 1u; i < (closed ? points.size() : points.size() - 1); ++i) {
86 const auto& a = points[(i + points.size() - 1) % points.size()];
87 const auto& b = points[i];
88 const auto& c = points[(i + 1) % points.size()];
89 if (std::fabs(area(a, b, c)) < 1e-5f &&
90 (b.side - a.side) * (c.side - b.side) + (b.up - a.up) * (c.up - b.up) < 0.f)
91 return invalid("section has overlapping adjacent segments");
92 }
93 const std::size_t count = closed ? points.size() : points.size() - 1;
94 float signedArea = 0.f;
95 for (std::size_t i = 0; i < count; ++i) {
96 const auto& a = points[i];
97 const auto& b = points[(i + 1) % points.size()];
98 if (std::hypot(a.side - b.side, a.up - b.up) < 1e-5f) return invalid("section has a zero-length segment");
99 signedArea += a.side * b.up - a.up * b.side;
100 for (std::size_t j = i + 2; j < count; ++j) {
101 if (closed && i == 0 && j + 1 == count) continue;
102 if (isIntersecting(a, b, points[j], points[(j + 1) % points.size()]))
103 return invalid("section self-intersects");
104 }
105 }
106 if (closed && std::fabs(signedArea) < 1e-5f) return invalid("closed section has zero area");
107 return Result<void>::success();
108}

171 }

211Result<std::pair<RoadStyle, RoadStyle>> splitRoadAuthoring(const RoadEdge& edge, std::size_t pointIndex) {
212 SplinePath path;
213 auto kind = path.setKindResult("catmullRom");
214 if (!kind.ok()) return Result<std::pair<RoadStyle, RoadStyle>>::failure(kind.status());
215 for (const auto& p : edge.controlPoints) {
216 SplinePoint point;
217 point.alignStraightStart = p.alignStraightStart;
218 point.alignStraightEnd = p.alignStraightEnd;
219 point.circularFromEnd = p.circularFromEnd;
220 point.spanControl = static_cast<eve::procgen::SplineSpanControl>(p.pathControl);
221 point.spanStrength = p.splineStrength;
222 point.x = p.x;
223 point.y = p.y;
224 point.z = p.z;
225 auto added = path.addPointResult(point);
226 if (!added.ok()) return Result<std::pair<RoadStyle, RoadStyle>>::failure(added.status());
227 }
228 const int count = path.segmentCount() * 24;
229 float total = 0.f, firstLength = 0.f;
230 SplineSample previous;
231 for (int i = 0; i <= count; ++i) {
232 auto sample = path.evaluateResult(static_cast<float>(i) / static_cast<float>(count));
233 if (!sample.ok()) return Result<std::pair<RoadStyle, RoadStyle>>::failure(sample.status());
234 if (i > 0)
235 total +=
236 std::sqrt(std::pow(sample.value().x - previous.x, 2.f) + std::pow(sample.value().y - previous.y, 2.f) +
237 std::pow(sample.value().z - previous.z, 2.f));
238 if (i == static_cast<int>(pointIndex) * 24) firstLength = total;
239 previous = sample.value();
240 }
241 const float fraction = firstLength / total;
242 if (!std::isfinite(fraction) || fraction <= 1e-5f || fraction >= 1.f - 1e-5f)
243 return Result<std::pair<RoadStyle, RoadStyle>>::failure(
244 Diagnostic::error(DiagnosticCode::InvalidArgument, "split has a degenerate arc interval"));
245 auto first = edge.style, second = edge.style;
246 if (!edge.style.structures.empty()) first.automaticStructures = second.automaticStructures = false;
247 first.structures = clip(edge.style.structures, 0.f, fraction, total);
248 second.structures = clip(edge.style.structures, fraction, 1.f, total);
249 first.facilities = clip(edge.style.facilities, 0.f, fraction, total);
250 second.facilities = clip(edge.style.facilities, fraction, 1.f, total);
251 return Result<std::pair<RoadStyle, RoadStyle>>::success({std::move(first), std::move(second)});
252}
253float roadSurfaceHeight(const RoadStyle& style, float lateral) {
254 if (style.crossSection.empty()) return 0.f;
255 const std::size_t count = style.crossSectionClosed ? style.crossSection.size() : style.crossSection.size() - 1;
256 for (std::size_t i = 0; i < count; ++i) {
257 const auto& a = style.crossSection[i];
258 const auto& b = style.crossSection[(i + 1) % style.crossSection.size()];
259 if (a.material != RoadMaterial::Asphalt || std::fabs(a.side - b.side) < 1e-5f ||
260 lateral < std::min(a.side, b.side) - 1e-5f || lateral > std::max(a.side, b.side) + 1e-5f)
261 continue;
262 return std::lerp(a.up, b.up, std::clamp((lateral - a.side) / (b.side - a.side), 0.f, 1.f));
263 }
264 return 0.f;
265}
