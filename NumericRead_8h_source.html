49[[nodiscard]] inline Result<std::optional<int>> readOptionalInt(const Value::Object& object, std::string_view name,
50 std::string_view path, std::string_view source) {
51 const auto found = object.find(std::string(name));
52 if (found == object.end()) return Result<std::optional<int>>::success(std::nullopt);
53 const auto failure = [&](std::string message) {
54 return Result<std::optional<int>>::failure(Diagnostic::error(
55 DiagnosticCode::InvalidArgument, std::move(message), std::string(path), {}, std::string(source)));
56 };
57 if (const auto* integer = found->second.getIf<std::int64_t>()) {
58 if (*integer < std::numeric_limits<int>::min() || *integer > std::numeric_limits<int>::max())
59 return failure("Integer out of range");
60 return Result<std::optional<int>>::success(static_cast<int>(*integer));
61 }
62 if (const auto* number = found->second.getIf<double>()) {
63 if (!std::isfinite(*number) || *number != std::floor(*number)) return failure("Expected integer");
64 if (*number < static_cast<double>(std::numeric_limits<int>::min()) ||
65 *number > static_cast<double>(std::numeric_limits<int>::max()))
66 return failure("Integer out of range");
67 return Result<std::optional<int>>::success(static_cast<int>(*number));
68 }
69 return failure("Expected integer");
70}
