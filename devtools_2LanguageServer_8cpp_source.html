42 std::string line;
43 size_t length = 0;
44 while (std::getline(input, line)) {
45 if (!line.empty() && line.back() == '\r') line.pop_back();
46 if (line.empty()) break;
47 constexpr std::string_view prefix = "Content-Length:";
48 if (line.rfind(prefix, 0) == 0) {

193std::string percentDecode(std::string path) {
194 for (size_t cursor = 0; cursor + 2 < path.size();) {
195 if (path[cursor] != '%') {
196 ++cursor;
197 continue;
198 }
199 const auto hex = [](char value) -> int {
200 if (value >= '0' && value <= '9') return value - '0';
201 if (value >= 'a' && value <= 'f') return value - 'a' + 10;
202 if (value >= 'A' && value <= 'F') return value - 'A' + 10;
203 return -1;
204 };
205 const int high = hex(path[cursor + 1]);
206 const int low = hex(path[cursor + 2]);
207 if (high < 0 || low < 0) {
208 ++cursor;
209 continue;
210 }
211 path.replace(cursor, 3, 1, static_cast<char>((high << 4) | low));
212 ++cursor;
213 }
214 return path;
215}

244 const size_t newline = source.find('\n', offset);

248 size_t begin = offset;
249 size_t end = offset;
250 while (begin > 0 && (std::isalnum(static_cast<unsigned char>(source[begin - 1])) != 0 || source[begin - 1] == '_'))
251 --begin;
252 while (end < source.size() && (std::isalnum(static_cast<unsigned char>(source[end])) != 0 || source[end] == '_'))
253 ++end;

256 size_t scan = begin;
257 if (result.word.empty() && offset > 0 && source[offset - 1] == '.') scan = offset;
258 while (scan > 0 && std::isspace(static_cast<unsigned char>(source[scan - 1])) != 0) --scan;

266 --qualBegin;

268 }

344std::vector<ExprTok> tokenizeExpr(std::string_view source, size_t begin, size_t end) {
345 std::vector<ExprTok> result;
346 for (size_t i = begin; i < end && i < source.size();) {
347 const char value = source[i];
348 const char next = i + 1 < source.size() ? source[i + 1] : 0;
349 if (std::isspace(static_cast<unsigned char>(value)) != 0) {
350 ++i;
351 continue;
352 }
353 if (value == '/' && next == '/') {
354 while (i < end && i < source.size() && source[i] != '\n') ++i;
355 continue;
356 }
357 if (value == '/' && next == '*') {
358 i += 2;
359 while (i + 1 < source.size() && i < end && !(source[i] == '*' && source[i + 1] == '/')) ++i;
360 i = std::min(i + 2, end);
361 continue;
362 }
363 if (value == '"' || value == '\'') {
364 const char quote = value;
365 ++i;
366 while (i < end && i < source.size() && source[i] != quote) {
367 if (source[i] == '\\' && i + 1 < source.size()) i += 2;
368 else
369 ++i;
370 }
371 if (i < end) ++i;
372 continue;
373 }
374 if (identifierStart(value)) {
375 size_t start = i++;
376 while (i < end && i < source.size() && identifierPart(source[i])) ++i;
377 result.push_back({ExprTokKind::Ident, std::string(source.substr(start, i - start))});
378 continue;
379 }
380 if (value == '.') {
381 result.push_back({ExprTokKind::Dot, "."});
382 ++i;
383 continue;
384 }
385 if (value == '(') {
386 result.push_back({ExprTokKind::LParen, "("});
387 ++i;
388 continue;
389 }
390 if (value == ')') {
391 result.push_back({ExprTokKind::RParen, ")"});
392 ++i;
393 continue;
394 }
395 ++i;
396 }
397 return result;
398}

442std::string typeOfTokens(const std::vector<ExprTok>& tokens, const SlotClassMap& slots,
443 const script::BindingContractRegistry& bindings,
444 const std::function<std::string(std::string_view)>& localType) {
445 std::string type;
446 for (size_t i = 0; i < tokens.size();) {
447 if (tokens[i].kind == ExprTokKind::Dot) {
448 ++i;
449 continue;
450 }
451 if (tokens[i].kind != ExprTokKind::Ident) {
452 ++i;
453 continue;
454 }
455 const std::string name = tokens[i].text;
456 ++i;
457 const bool called = i < tokens.size() && tokens[i].kind == ExprTokKind::LParen;
458 if (called) i = skipCallTokens(tokens, i);
459 type = type.empty() ? applyRootIdent(name, called, slots, bindings, localType)
460 : applyMember(std::move(type), name, called, bindings);
461 if (type.empty()) return {};
462 }
463 return type;

476std::string inferAssignedType(std::string_view source, size_t offset, std::string_view name, const SlotClassMap& slots,
477 const script::BindingContractRegistry& bindings) {
478 size_t match = std::string::npos;
479 for (size_t i = 0; i + name.size() <= offset;) {
480 if (!identifierStart(source[i]) || source.compare(i, name.size(), name) != 0) {
481 ++i;
482 continue;
483 }
484 const size_t after = i + name.size();
485 if ((i > 0 && identifierPart(source[i - 1])) || (after < source.size() && identifierPart(source[after]))) {
486 ++i;
487 continue;
488 }
489 size_t cursor = after;
490 while (cursor < offset && std::isspace(static_cast<unsigned char>(source[cursor])) != 0) ++cursor;
491 if (cursor < offset && source[cursor] == ':') {
492 ++cursor;
493 while (cursor < offset && std::isspace(static_cast<unsigned char>(source[cursor])) != 0) ++cursor;
494 size_t typeEnd = cursor;
495 while (typeEnd < offset && (identifierPart(source[typeEnd]) || source[typeEnd] == '?' || source[typeEnd] == '.'))
496 ++typeEnd;
497 const std::string annotated = trimTypeName(source.substr(cursor, typeEnd - cursor));
498 if (bindings.hasScriptClass(annotated)) {
499 match = i;
500 i = typeEnd;
501 continue;
502 }
503 cursor = typeEnd;
504 while (cursor < offset && std::isspace(static_cast<unsigned char>(source[cursor])) != 0) ++cursor;
505 }
506 const bool assign = cursor < offset && (source[cursor] == '=' ||
507 (source[cursor] == '<' && cursor + 1 < offset && source[cursor + 1] == '-'));
508 if (!assign || (source[cursor] == '=' && cursor + 1 < offset && source[cursor + 1] == '=')) {
509 ++i;
510 continue;
511 }
512 match = i;
513 i = cursor + 1;
514 }
515 if (match == std::string::npos) return {};
516 size_t cursor = match + name.size();
517 while (cursor < offset && std::isspace(static_cast<unsigned char>(source[cursor])) != 0) ++cursor;
518 if (cursor < offset && source[cursor] == ':') {
519 ++cursor;
520 while (cursor < offset && std::isspace(static_cast<unsigned char>(source[cursor])) != 0) ++cursor;
521 size_t typeEnd = cursor;
522 while (typeEnd < offset && identifierPart(source[typeEnd])) ++typeEnd;
523 const std::string annotated = trimTypeName(source.substr(cursor, typeEnd - cursor));
524 if (bindings.hasScriptClass(annotated)) return annotated;
525 cursor = typeEnd;
526 while (cursor < offset && std::isspace(static_cast<unsigned char>(source[cursor])) != 0) ++cursor;
527 }
528 if (cursor < offset && source[cursor] == '<' && cursor + 1 < offset && source[cursor + 1] == '-') cursor += 2;
529 else if (cursor < offset && source[cursor] == '=')
530 ++cursor;
531 while (cursor < offset && std::isspace(static_cast<unsigned char>(source[cursor])) != 0) ++cursor;
532 size_t rhsEnd = cursor;
533 int depth = 0;
534 while (rhsEnd < offset) {
535 const char value = source[rhsEnd];
536 if (value == '(' || value == '[' || value == '{') ++depth;
537 else if (value == ')' || value == ']' || value == '}') {
538 if (depth == 0) break;
539 --depth;
540 } else if (depth == 0 && (value == ';' || value == '\n'))
541 break;
542 ++rhsEnd;
543 }
544 const auto noopLocal = [](std::string_view) { return std::string{}; };
545 return typeOfTokens(tokenizeExpr(source, cursor, rhsEnd), slots, bindings, noopLocal);
546}

588 source.canonicalUri = std::string(canonicalUri);
589 source.utf8Source.assign(std::istreambuf_iterator<char>(input), std::istreambuf_iterator<char>());

592 }

693 std::vector<script::ScriptCompletion> complete(std::string_view uri, lsp::Position position) const {
694 const std::string* source = sourceFor(uri);
695 CursorWord cursor;
696 if (source)
697 cursor = wordAt(*source, position.line, position.character);
698 const std::string ident = identity(uri);
699 std::vector<script::ScriptCompletion> result;
700 if (cursor.memberAccess) {
701 size_t wordBegin = 0;
702 if (source) {
703 size_t offset = 0;
704 for (size_t current = 0; current < position.line && offset < source->size(); ++current) {
705 const size_t newline = source->find('\n', offset);
706 offset = newline == std::string_view::npos ? source->size() : newline + 1;
707 }
708 offset = std::min(offset + position.character, source->size());
709 wordBegin = offset;
710 while (wordBegin > 0 && identifierPart((*source)[wordBegin - 1])) --wordBegin;
711 }
712 size_t dot = wordBegin;
713 if (source) {
714 while (dot > 0 && std::isspace(static_cast<unsigned char>((*source)[dot - 1])) != 0) --dot;
715 if (dot > 0 && (*source)[dot - 1] == '.') --dot;
716 }
717 const auto localType = [&](std::string_view name) -> std::string {
718 if (const auto unit = runtime_.scriptCompiler().metadata(ident)) {
719 for (const script::ScriptSymbolMetadata& symbol : unit->get().symbols) {
720 if (symbol.name != name) continue;
721 const std::string annotated = trimTypeName(symbol.erasedType);
722 if (runtime_.scriptCompiler().bindings().hasScriptClass(annotated)) return annotated;
723 }
724 }
725 return source ? inferAssignedType(*source, dot, name, slotClass_, runtime_.scriptCompiler().bindings())
726 : std::string{};
727 };
728 const std::string receiver =
729 source ? typeOfTokens(tokenizeExpr(*source, statementStart(*source, dot), dot), slotClass_,
730 runtime_.scriptCompiler().bindings(), localType)
731 : std::string{};
732 const std::string expectedClass = !receiver.empty()
733 ? receiver
734 : (slotClass_.count(cursor.qualifier) ? slotClass_.at(cursor.qualifier)
735 : cursor.qualifier);
736 if (expectedClass == "eve") {
737 std::unordered_map<std::string, bool> seen;
738 for (const script::BindingContract& contract : runtime_.scriptCompiler().bindings().snapshot()) {
739 if (contract.scriptClass.empty() || contract.scriptClass.rfind(cursor.word, 0) != 0) continue;
740 if (!seen.emplace(contract.scriptClass, true).second) continue;
741 result.push_back({contract.scriptClass, "class", "eve." + contract.scriptClass, contract.scriptClass});
742 }
743 return result;
744 }
745 for (const script::BindingContract& contract : runtime_.scriptCompiler().bindings().snapshot()) {
746 if (contract.scriptClass != expectedClass) continue;
747 if (contract.method.rfind(cursor.word, 0) != 0) continue;
748 result.push_back(completionFromContract(contract));
749 }
750 return result;
751 }
752
753 result = runtime_.scriptCompiler().completions(ident, cursor.word);
754 static constexpr const char* keywords[] = {"import", "export", "persist", "match", "async", "await",
755 "local", "function", "class", "const", "foreach", "return",
756 "if", "else", "while", "for", "switch", "case",
757 "break", "continue", "try", "catch", "throw", "null",
758 "true", "false", "this", "base", "typeof", "instanceof",
759 "resume", "yield", "clone", "delete"};
760 for (const char* keyword : keywords) {
761 if (std::string_view(keyword).rfind(cursor.word, 0) != 0) continue;
762 result.push_back({keyword, "keyword", "EveScript keyword", keyword});
763 }
764 for (const std::string& slot : activeModules_) {
765 if (slot.rfind(cursor.word, 0) != 0) continue;
766 const auto cls = slotClass_.find(slot);
767 result.push_back({slot, "slot", cls == slotClass_.end() ? "module slot" : cls->second, slot});
768 }
769 std::sort(result.begin(), result.end(), [](const script::ScriptCompletion& a, const script::ScriptCompletion& b) {
770 if (a.label != b.label) return a.label < b.label;
771 return a.kind < b.kind;
772 });
773 result.erase(std::unique(result.begin(), result.end(),
774 [](const script::ScriptCompletion& a, const script::ScriptCompletion& b) {
775 return a.label == b.label && a.kind == b.kind;
776 }),
777 result.end());
778 return result;
779 }

782 const std::string* source = sourceFor(uri);
783 if (!source) return std::nullopt;

785 if (cursor.word.empty()) return std::nullopt;
786 const std::string ident = identity(uri);

788 const auto slot = slotClass_.find(cursor.word);
789 if (slot != slotClass_.end()) {
790 script::ScriptHover hover;
791 hover.symbol = cursor.word;
792 hover.markdown = "`" + cursor.word + "` — `" + slot->second + "` module slot";
793 return hover;
794 }
795 if (const auto location = index_.definition(ident, position)) {
796 script::ScriptHover hover;
797 hover.symbol = cursor.word;
798 hover.markdown = "`" + cursor.word + "` defined in `" + location->uri + "`";
799 return hover;

801 return std::nullopt;

804 std::optional<LanguageSignatureHelp> signatureHelp(std::string_view uri, lsp::Position position) const {
805 const std::string* source = sourceFor(uri);
806 if (!source) return std::nullopt;
807 size_t offset = 0;
808 for (size_t current = 0; current < position.line && offset < source->size(); ++current) {
809 const size_t newline = source->find('\n', offset);
810 offset = newline == std::string_view::npos ? source->size() : newline + 1;

813 const std::string callee = calleeBeforeParen(*source, offset);

818 help.activeParameter = activeParameterAt(*source, offset);
819 help.documentation = contract->documentationId;
820 help.label = contract->scriptClass + "." + contract->method + "(";
821 for (size_t i = 0; i < contract->parameters.size(); ++i) {
822 if (i != 0) help.label += ", ";
823 help.label += contract->parameters[i].name + ": " + contract->parameters[i].type;
824 help.parameters.push_back({contract->parameters[i].name + ": " + contract->parameters[i].type, {}});
825 }
826 help.label += ") -> " + contract->returnType;
827 if (help.activeParameter >= static_cast<int>(help.parameters.size()))
828 help.activeParameter = help.parameters.empty() ? 0 : static_cast<int>(help.parameters.size()) - 1;
829 return help;

970std::vector<lsp::SemanticToken> LanguageServer::semanticTokens(std::string_view uri) const {
971 std::vector<lsp::SemanticToken> tokens = impl_->index_.semanticTokens(impl_->identity(uri));
972 const auto& bindings = impl_->runtime_.scriptCompiler().bindings();
973 const auto metadata = impl_->runtime_.scriptCompiler().metadata(impl_->identity(uri));
974 std::unordered_map<std::string, uint32_t> extra;
975 if (metadata) {
976 for (const script::ScriptSymbolMetadata& symbol : metadata->get().symbols) {
977 if (symbol.kind == "class") extra[symbol.name] = lsp::SemanticTypes::Class;
978 else if (symbol.kind == "function" || symbol.kind == "async")
979 extra[symbol.name] = lsp::SemanticTypes::Function;
980 }
981 }
982 for (lsp::SemanticToken& token : tokens) {
983 if (token.type == lsp::SemanticTypes::Keyword) continue;
984 if (token.type == lsp::SemanticTypes::Method || token.type == lsp::SemanticTypes::Property) {
985 if (bindings.hasScriptClass(token.name)) token.type = lsp::SemanticTypes::Class;
986 continue;
987 }
988 if (impl_->slotClass_.count(token.name) != 0) {
989 token.type = lsp::SemanticTypes::Namespace;
990 token.modifiers |= lsp::SemanticMods::DefaultLibrary;
991 continue;
992 }
993 if (bindings.hasScriptClass(token.name)) {
994 token.type = lsp::SemanticTypes::Class;
995 continue;
996 }
997 const auto found = extra.find(token.name);
998 if (found != extra.end() && token.type == lsp::SemanticTypes::Variable) token.type = found->second;
999 }
1000 return tokens;
1001}

1032 const int line = clampNonNegative(static_cast<int>(diagnostic.position.line) - 1);
1033 const int col = clampNonNegative(static_cast<int>(diagnostic.position.column) - 1);
1034 lsp::Range range{{static_cast<size_t>(line), static_cast<size_t>(col)},
1035 {static_cast<size_t>(line), static_cast<size_t>(col + 1)}};
1036 Poco::JSON::Object::Ptr item = new Poco::JSON::Object();
1037 item->set("range", rangeObject(range));
1038 item->set("severity", diagnostic.severity == script::ScriptDiagnosticSeverity::Error ? 1 : 2);
1039 item->set("code", diagnostic.code);

1073 capabilities->set("definitionProvider", true);

1077 capabilities->set("signatureHelpProvider", signature);
1078 capabilities->set("documentFormattingProvider", true);
1079 capabilities->set("documentRangeFormattingProvider", true);
1080 capabilities->set("foldingRangeProvider", true);

1082 semantic->set("legend", semanticTokensLegend());
1083 semantic->set("full", true);
1084 semantic->set("range", false);
1085 capabilities->set("semanticTokensProvider", semantic);

1087 diagnosticProvider->set("interFileDependencies", true);
1088 diagnosticProvider->set("workspaceDiagnostics", false);

1091 result->set("capabilities", capabilities);

1093 serverInfo->set("name", "evescript");
1094 serverInfo->set("version", "1");
1095 result->set("serverInfo", serverInfo);
1096 writeFrame(output, rpcResponse(id, Poco::Dynamic::Var(result)));
1097 } else if (method == "initialized" || method == "textDocument/didSave" || method.rfind("$/", 0) == 0) {
1098 return LspDispatch::Continue;
1099 } else if (method == "shutdown") {
1100 impl_->shutdown_ = true;
1101 writeFrame(output, rpcResponse(id, Poco::Dynamic::Var()));
1102 } else if (method == "exit") {
1103 return LspDispatch::Exit;
