48eve::Result<std::optional<eve::resource::CostSpec>> weaponCost(const WeaponDefinition& weapon) {
49 if (weapon.resource.infinite || weapon.resource.kind == ResourceKind::None)
50 return eve::Result<std::optional<eve::resource::CostSpec>>::success(std::nullopt);
51
52 double amount = static_cast<double>(weapon.resource.cost);
53 // Ranged WeaponSystem consumes one magazine unit per trigger even when
54 // the legacy template leaves resource.cost at its zero default.
55 if (weapon.resource.kind == ResourceKind::Ammo && amount <= 0.0) amount = 1.0;
56 if (!std::isfinite(amount) || amount <= 0.0 || std::floor(amount) != amount || amount >= std::ldexp(1.0, 63))
57 return failure<std::optional<eve::resource::CostSpec>>(eve::DiagnosticCode::InvalidArgument,
58 "weapon resource cost must be a finite positive integer",
59 "resource.cost");
60
61 auto cost = eve::resource::CostSpec::single(resourceName(weapon.resource.kind), static_cast<std::int64_t>(amount));
62 if (!cost) return eve::Result<std::optional<eve::resource::CostSpec>>::failure(cost.status());
63 return eve::Result<std::optional<eve::resource::CostSpec>>::success(std::move(cost).takeValue());
64}

80eve::Result<eve::transaction::TransactionReceipt> executeActionFire(
81 const WeaponDefinition& weapon, const AttackRequest& attack, eve::resource::IResourceAccount& account,
82 eve::transaction::ITransactionParticipant& effect, eve::SimulationTick tick, std::string transactionId) {
83 auto definition = WeaponActionAdapter::makeDefinition(weapon, attack);
84 if (!definition) return eve::Result<eve::transaction::TransactionReceipt>::failure(definition.status());
85 auto request = WeaponActionAdapter::makeRequest(weapon, attack, std::nullopt, tick);
86 if (!request) return eve::Result<eve::transaction::TransactionReceipt>::failure(request.status());
87
88 auto actionDefinition = std::move(definition).takeValue();
89 auto actionRequest = std::move(request).takeValue();
90 if (transactionId.empty()) transactionId = "weapon.fire." + weapon.id;
91 actionRequest.transactionId = std::move(transactionId);
92
93 action::ActionServices services;
94 services.transactionEffect = &effect;
95 services.transactionAccount = &account;
96 action::ActionRuntime runtime(services);
97 auto submitted = runtime.submit(std::move(actionDefinition), std::move(actionRequest));
98 if (!submitted) return eve::Result<eve::transaction::TransactionReceipt>::failure(submitted.status());
99 const auto executionId = std::move(submitted).takeValue();
100 const auto* submittedExecution = runtime.find(executionId);
101 if (submittedExecution == nullptr)
102 return eve::Result<eve::transaction::TransactionReceipt>::failure(
103 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
104 "weapon Action execution disappeared after submit", "weapon.fire"));
105
106 auto total = submittedExecution->definition().timing.windup.tryAdd(submittedExecution->definition().timing.active);
107 if (!total) return eve::Result<eve::transaction::TransactionReceipt>::failure(total.status());
108 total = std::move(total).takeValue().tryAdd(submittedExecution->definition().timing.recover);
109 if (!total) return eve::Result<eve::transaction::TransactionReceipt>::failure(total.status());
110
111 auto advanced = runtime.advance(executionId, tick, std::move(total).takeValue());
112 if (!advanced) return eve::Result<eve::transaction::TransactionReceipt>::failure(advanced.status());
113 const auto* execution = runtime.find(executionId);
114 if (execution == nullptr || execution->phase() != action::ActionPhase::Completed ||
115 execution->transactionReceipt() == nullptr)
116 return eve::Result<eve::transaction::TransactionReceipt>::failure(
117 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
118 "weapon Action execution completed without a transaction receipt", "weapon.fire"));
119 return eve::Result<eve::transaction::TransactionReceipt>::success(*execution->transactionReceipt(),
120 advanced.status());
121}

138eve::Result<action::ActionDefinition> WeaponActionAdapter::makeDefinition(const WeaponDefinition& weapon,
139 const AttackRequest& request) {
140 auto id = weaponActionId(weapon);
141 if (!id) return eve::Result<action::ActionDefinition>::failure(id.status());
142 auto windup = seconds(weapon.stages.windupTime, "stages.windupTime");
143 if (!windup) return eve::Result<action::ActionDefinition>::failure(windup.status());
144 auto active = seconds(weapon.stages.activeTime, "stages.activeTime");
145 if (!active) return eve::Result<action::ActionDefinition>::failure(active.status());
146 auto recover = seconds(weapon.stages.recoverTime, "stages.recoverTime");
147 if (!recover) return eve::Result<action::ActionDefinition>::failure(recover.status());
148 auto cost = resourceCost(weapon);
149 if (!cost) return eve::Result<action::ActionDefinition>::failure(cost.status());
150
151 action::ActionDefinition definition;
152 definition.id = std::move(id).takeValue();
153 definition.timing.windup = std::move(windup).takeValue();
154 definition.timing.active = std::move(active).takeValue();
155 definition.timing.recover = std::move(recover).takeValue();
156 definition.targetingMode = request.hasTarget ? action::TargetingMode::Explicit : action::TargetingMode::None;
157 definition.cost = std::move(cost).takeValue();
158 // Weapon logic is an Active operation even if its effect ids are resolved
159 // by the weapon logic/event sink rather than the effects registry.
160 definition.activeExecutionRequired = true;
161 definition.metadata.emplace("weaponId", eve::Value(weapon.id));
162 definition.metadata.emplace("logic", eve::Value(weapon.logic));
163 definition.metadata.emplace("kind", eve::Value(static_cast<int>(weapon.kind)));
164
165 auto valid = definition.validate();
166 if (!valid) return eve::Result<action::ActionDefinition>::failure(valid.status());
167 return eve::Result<action::ActionDefinition>::success(std::move(definition));
168}

170eve::Result<action::ActionRequest> WeaponActionAdapter::makeRequest(const WeaponDefinition& weapon,
171 const AttackRequest& attack,
172 std::optional<ecs::EntityHandle> source,
173 eve::SimulationTick requestedTick) {
174 auto definition = makeDefinition(weapon, attack);
175 if (!definition) return eve::Result<action::ActionRequest>::failure(definition.status());
176 auto checkedDefinition = std::move(definition).takeValue();
177
178 action::ActionRequest request;
179 request.actionId = checkedDefinition.id;
180 request.source = std::move(source);
181 request.requestedTick = requestedTick;
182 if (attack.hasTarget) request.targetEntities.push_back(attack.targetHandle);
183
184 // The canonical request owns deterministic attack parameters. Weapon
185 // fields remain in WeaponDefinition/WeaponEntity; this is only a
186 // provider-neutral transport envelope for the Active executor.
187 put(request.parameters, "targetX", attack.targetX);
188 put(request.parameters, "targetY", attack.targetY);
189 put(request.parameters, "targetZ", attack.targetZ);
190 request.parameters.emplace("hasTarget", eve::Value(attack.hasTarget));
191 put(request.parameters, "muzzleX", attack.muzzleX);
192 put(request.parameters, "muzzleY", attack.muzzleY);
193 put(request.parameters, "muzzleZ", attack.muzzleZ);
194 put(request.parameters, "yaw", attack.yaw);
195 put(request.parameters, "pitch", attack.pitch);
196 put(request.parameters, "shooterId", attack.shooterId);
197 put(request.parameters, "arcAngle", attack.arcAngle);
198 put(request.parameters, "aoeRadius", attack.aoeRadius);
199 put(request.parameters, "spread", attack.spread);
200 put(request.parameters, "pelletIndex", attack.pelletIndex);
201 put(request.parameters, "pelletCount", attack.pelletCount);
202
203 auto valid = request.validate(checkedDefinition);
204 if (!valid) return eve::Result<action::ActionRequest>::failure(valid.status());
205 return eve::Result<action::ActionRequest>::success(std::move(request));
206}
