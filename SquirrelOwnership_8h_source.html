319 [[nodiscard]] eve::Result<void> erase(Ref ref) {
320 if (!ref.handle.isValid()) {
321 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
322 "cannot erase an invalid runtime handle", {}, {},
323 "runtime.registry"));
324 }
325 if (ref.ownerEpoch != ownerEpoch_) return staleFailure<void>();
326 const auto slotIndex = findSlotIndex(ref);
327 if (!slotIndex) return staleFailure<void>();
328 Slot& slot = slots_[*slotIndex];
329 if (!slot.object || slot.generation != ref.handle.generation()) return staleFailure<void>();
330
331 const auto next = Handle::nextGeneration(slot.generation);
332 if (next) {
333 try {
334 freeSlots_.push_back(ref.handle.index());
335 } catch (const std::exception& error) {
336 return eve::Result<void>::failure(
337 eve::Diagnostic::error(eve::DiagnosticCode::Failed,
338 std::string("runtime registry release bookkeeping failed: ") + error.what(),
339 {}, {}, "runtime.registry"));

342 "runtime registry release bookkeeping failed",
343 {}, {}, "runtime.registry"));
344 }
345 }
346 slot.object.reset();
347 if (next)
348 slot.generation = *next;
349 else
350 slot.retired = true;

397 [[nodiscard]] bool isStale(Ref ref) const noexcept {
398 if (!ref.handle.isValid()) return false;
399 if (ref.ownerEpoch != ownerEpoch_) return true;
400 const auto slotIndex = findSlotIndex(ref);
401 if (!slotIndex) return true;
402 const Slot& slot = slots_[*slotIndex];

407 void clear() {
408 // Reserve before mutating live slots so free-list bookkeeping cannot
409 // fail after ownership has already been released.
410 freeSlots_.reserve(slots_.size());
411 freeSlots_.clear();
412 for (std::uint32_t index = 0; index < slots_.size(); ++index) {
413 Slot& slot = slots_[index];
414 slot.object.reset();
415 const auto next = Handle::nextGeneration(slot.generation);
416 if (next) {
417 slot.generation = *next;
418 freeSlots_.push_back(index);
419 } else {
420 slot.retired = true;
421 }
422 }

492template <class T>
493[[nodiscard]] eve::Result<ssq::Object> makeOwnedSquirrelInstance(HSQUIRRELVM vm, Owned<T> object) {
494 if (!vm || !object) {
495 return eve::Result<ssq::Object>::failure(eve::Diagnostic::error(
496 eve::DiagnosticCode::InvalidArgument, "owned Squirrel instance requires a VM and non-null object", {}, {},
497 "squirrel.ownership"));
498 }
499
500 const SQInteger top = sq_gettop(vm);
501 bool transferred = false;
502 try {
503 const size_t hashCode = detail::squirrelTypeHash<T*>();
504 const HSQOBJECT& classObject = ssq::detail::getClassObj(vm, hashCode);
505 sq_pushobject(vm, classObject);
506 if (SQ_FAILED(sq_createinstance(vm, -1)))
507 throw ssq::RuntimeException("failed to create owned Squirrel instance");
508 sq_remove(vm, -2);
509 if (SQ_FAILED(sq_setinstanceup(vm, -1, static_cast<SQUserPointer>(object.get()))))
510 throw ssq::RuntimeException("failed to attach owned Squirrel instance");
511 sq_settypetag(vm, -1, reinterpret_cast<SQUserPointer>(hashCode));
512 sq_setreleasehook(vm, -1, &detail::ownedInstanceReleaseHook<T>);
513 object.release();
514 transferred = true;
515
516 ssq::Object result(vm);
517 if (SQ_FAILED(sq_getstackobj(vm, -1, &result.getRaw())))
518 throw ssq::RuntimeException("failed to root owned Squirrel instance");
519 sq_addref(vm, &result.getRaw());
520 sq_settop(vm, top);
521 return eve::Result<ssq::Object>::success(std::move(result));
522 } catch (const std::exception& error) {
523 sq_settop(vm, top);
524 if (!transferred) {
525 return eve::Result<ssq::Object>::failure(eve::Diagnostic::error(
526 eve::DiagnosticCode::Failed, std::string("owned Squirrel instance creation failed: ") + error.what(),
527 {}, {}, "squirrel.ownership"));
528 }
529 return eve::Result<ssq::Object>::failure(eve::Diagnostic::error(
530 eve::DiagnosticCode::Failed, std::string("owned Squirrel instance rooting failed: ") + error.what(), {}, {},
531 "squirrel.ownership"));
532 } catch (...) {
533 sq_settop(vm, top);
534 return eve::Result<ssq::Object>::failure(eve::Diagnostic::error(
535 eve::DiagnosticCode::Failed, "owned Squirrel instance creation failed", {}, {}, "squirrel.ownership"));
536 }
