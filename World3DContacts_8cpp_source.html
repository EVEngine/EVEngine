24int World3D::queryBodyContacts(int bodyId, int maxPoints) {
25 contactPoints_.clear();
26 if (bodyId <= 0) throw eve::Exception("World3D.queryBodyContacts: bodyId must be a positive stable id");
27 if (maxPoints < 1 || maxPoints > 4096)
28 throw eve::Exception("World3D.queryBodyContacts: maxPoints must be in [1, 4096]");
29 const auto *body = findBodyById(bodyId);
30 if (!body) return 0;
31 auto result = queryBodyContactsOwned(body->runtimeHandle(), maxPoints);
32 if (!result) throw eve::Exception("World3D.queryBodyContacts: %s", result.status().describe().c_str());
33 contactPoints_ = std::move(result).takeValue().points;
34 return static_cast<int>(contactPoints_.size());
35}
