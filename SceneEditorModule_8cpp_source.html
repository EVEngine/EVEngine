46class SceneEditorModule::TargetFactory final : public editor::IEditorAutomationTargetFactory {
47public:
48 TargetFactory(scene_editing::SceneComponentPayloadRegistry* payloads, std::map<std::string, std::string>* hosts)
49 : payloads_(payloads), hosts_(hosts) {}
50
51 std::vector<std::string_view> types() const override { return {"scene", "scene-host", "scene-authoring"}; }
52
53 editor::Result<editor::AutomationOwnedTarget> create(const editor::TargetId& target, std::string_view type,
54 const editor::EditorValue::Object& request) override {
55 editor::AutomationOwnedTarget owned;
56 if (type == "scene-authoring") {
57 owned.target = std::make_unique<scene_editing::SceneAuthoringDocumentTarget>(target.value());
58 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
59 }
60 if (type == "scene") {
61 auto sceneTarget = std::make_unique<scene_editing::SceneDocumentTarget>(target.value());
62 sceneTarget->bindComponentPayloads(payloads_);
63 const std::string object = stringField(request, "object");
64 if (!object.empty()) {
65 scene_editing::CreateSceneObjectRequest create;
66 create.id = editor::ObjectId(object);
67 create.name = object;
68 auto operation = sceneTarget->makeCreate(create);
69 if (!operation.ok()) return editor::Result<editor::AutomationOwnedTarget>::failure(operation.status());
70 auto applied = sceneTarget->applyDomainOperation(operation.value());
71 if (!applied.ok()) return editor::Result<editor::AutomationOwnedTarget>::failure(applied.status());
72 }
73 owned.target = std::move(sceneTarget);
74 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
75 }
76
77 const std::string host = stringField(request, "host");
78 if (host.empty())
79 return eve::editing::failed<editor::AutomationOwnedTarget>(
80 editor::EditorStatus::Rejected, editor::RuleId("editor.automation.scene-host-name"),
81 "Live scene target requires a host name");
82 auto* sceneModule = eve::ModuleManager::getInstance<eve::scene::Scene>("Scene");
83 if (!sceneModule)
84 return eve::editing::failed<editor::AutomationOwnedTarget>(
85 editor::EditorStatus::Unsupported, editor::RuleId("editor.automation.scene-module-unavailable"),
86 "Scene module is not available");
87 auto hostResult = sceneModule->findHost(host);
88 if (!hostResult.ok() || !hostResult.value())
89 return eve::editing::failed<editor::AutomationOwnedTarget>(
90 editor::EditorStatus::NotFound, editor::RuleId("editor.automation.scene-host-not-found"),
91 "Live SceneHost does not exist: " + host);
92 auto liveTarget = std::make_unique<scene_editing::SceneHostEditorTarget>(target.value(), hostResult.value());
93 liveTarget->bindComponentPayloads(payloads_);
94 // 目标 id 是调用方选的，宿主名只有这里知道：记下来给组件 payload 提供方做映射
95 // （`ScriptComponentPayloadProvider::HostResolver`）。目标关闭时留着这一条也无害——
96 // 同名目标再创建会覆盖它，而查询一个不存在的宿主只会得到空组件表。
97 (*hosts_)[target.value()] = host;
98 owned.target = std::move(liveTarget);
99 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
100 }
101
102private:
103 scene_editing::SceneComponentPayloadRegistry* payloads_ = nullptr;
104 std::map<std::string, std::string>* hosts_ = nullptr;
105};

109SceneEditorModule::SceneEditorModule()
110 : payloads_(std::make_unique<scene_editing::SceneComponentPayloadRegistry>()),
111 scriptComponents_(std::make_unique<ScriptComponentPayloadProvider>([this](const scene_editing::TargetId& t) {
112 const auto found = targetHosts_.find(t.value());
113 return found == targetHosts_.end() ? std::string{} : found->second;
114 })),
115 factory_(std::make_unique<TargetFactory>(payloads_.get(), &targetHosts_)) {
116 auto* registry = eve::cap::query<editing::IEditingCommandRegistry>();
117 if (!registry || !scene_editing::registerEditingCommands(*registry).ok())
118 throw std::runtime_error("Failed to register scene editing commands");
119 payloads_->registerProvider(scriptComponents_.get())
120 .ignore("this module owns both the registry and the provider; a duplicate can only come from a bug here");
121 eve::cap::addListener<editor::IEditorAutomationTargetFactory>(factory_.get());
122}

124SceneEditorModule::~SceneEditorModule() {
125 payloads_->unregisterProvider(scriptComponents_.get());
126 eve::cap::removeListener<editor::IEditorAutomationTargetFactory>(factory_.get());
127 if (auto* registry = eve::cap::query<editing::IEditingCommandRegistry>())
128 registry->unregisterOwner("scene_editing").ignore("scene editor adapter shutdown");
129}

153const editing::Value* field(const editing::Value& value, const char* key) {
154 const auto* object = value.getIf<editing::Value::Object>();

160bool number(const editing::Value& value, double& output) {
161 if (const auto* real = value.getIf<double>())
162 output = *real;
163 else if (const auto* integer = value.getIf<std::int64_t>())
164 output = static_cast<double>(*integer);
165 else
166 return false;
167 return true;
168}

170editing::Result<PhysicsPlacementCollider> placementCollider(const editing::Value& value) {
171 PhysicsPlacementCollider result;
172 const auto* shapeValue = field(value, "shape");
173 const auto* shape = shapeValue ? shapeValue->getIf<std::string>() : nullptr;
174 if (!shape || (*shape != "box" && *shape != "sphere" && *shape != "capsule" && *shape != "convex"))
175 return editing::failed<PhysicsPlacementCollider>(editing::Status::Rejected,
176 editing::RuleId("scene.physics-placement.script-part-shape"),
177 "Collider part shape is unsupported");
178 if (*shape == "sphere")
179 result.shape = PhysicsPlacementShape::Sphere;
180 else if (*shape == "capsule")
181 result.shape = PhysicsPlacementShape::Capsule;
182 else if (*shape == "convex")
183 result.shape = PhysicsPlacementShape::ConvexHull;
184 if (const auto* sourceValue = field(value, "source")) {
185 const auto* source = sourceValue->getIf<std::string>();
186 if (!source || (*source != "existing" && *source != "generated"))
187 return editing::failed<PhysicsPlacementCollider>(
188 editing::Status::Rejected, editing::RuleId("scene.physics-placement.script-part-source"),
189 "Collider part source must be existing or generated");
190 result.source = *source == "existing" ? PhysicsPlacementColliderSource::Existing
191 : PhysicsPlacementColliderSource::Generated;
192 }
193 const auto parseVector = [&](const char* key, double& x, double& y, double& z, bool required) {
194 const auto* candidate = field(value, key);
195 if (!candidate) return !required;
196 const auto* array = candidate->getIf<editing::Value::Array>();
197 return array && array->size() == 3 && number((*array)[0], x) && number((*array)[1], y) &&
198 number((*array)[2], z);
199 };
200 if (!parseVector("halfExtents", result.halfExtentX, result.halfExtentY, result.halfExtentZ, true) ||
201 !parseVector("localPosition", result.localX, result.localY, result.localZ, false) ||
202 !parseVector("localRotation", result.localRotationX, result.localRotationY, result.localRotationZ, false))
203 return editing::failed<PhysicsPlacementCollider>(editing::Status::Rejected,
204 editing::RuleId("scene.physics-placement.script-part-vector"),
205 "Collider part vectors require three numbers");
206 if (const auto* verticesValue = field(value, "vertices")) {
207 const auto* vertices = verticesValue->getIf<editing::Value::Array>();
208 if (!vertices)
209 return editing::failed<PhysicsPlacementCollider>(
210 editing::Status::Rejected, editing::RuleId("scene.physics-placement.script-part-vertices"),
211 "Collider part vertices must be an array");
212 for (const auto& component : *vertices) {
213 double numeric = 0.0;
214 if (!number(component, numeric))
215 return editing::failed<PhysicsPlacementCollider>(
216 editing::Status::Rejected, editing::RuleId("scene.physics-placement.script-part-vertex"),
217 "Collider part vertices must be numeric");
218 result.convexVertices.push_back(static_cast<float>(numeric));
219 }
220 }
221 if (const auto* maximumValue = field(value, "maxHullVertices")) {
222 const auto* maximum = maximumValue->getIf<std::int64_t>();
223 if (!maximum)
224 return editing::failed<PhysicsPlacementCollider>(
225 editing::Status::Rejected, editing::RuleId("scene.physics-placement.script-part-budget"),
226 "Collider part maxHullVertices must be an integer");
227 result.convexMaxVertices = static_cast<int>(*maximum);
228 }
229 return editing::applied<PhysicsPlacementCollider>(std::move(result));
230}

251 *mode != "align" && *mode != "point"))

261 else if (*mode == "point")

467 editing::RuleId("scene.physics-placement.script-layer"),

469 entry.layerBits = static_cast<std::uint64_t>(*layer);
470 }
471 for (const auto& [key, destination] : std::initializer_list<std::pair<const char*, bool*>>{
472 {"enabled", &entry.enabled}, {"locked", &entry.locked}}) {
473 if (const auto* candidate = field(value, key)) {
474 const auto* boolean = candidate->getIf<bool>();
475 if (!boolean)
476 return editing::failed<PhysicsPlacementRequest>(
477 editing::Status::Rejected, editing::RuleId("scene.physics-placement.script-object-flag"),
478 "Placement object flags must be boolean");
479 *destination = *boolean;
480 }
481 }
482 if (const auto* tagsValue = field(value, "tags")) {
483 const auto* tags = tagsValue->getIf<editing::Value::Array>();
484 if (!tags)
485 return editing::failed<PhysicsPlacementRequest>(editing::Status::Rejected,
486 editing::RuleId("scene.physics-placement.script-tags"),
487 "Placement tags must be an array");
488 for (const auto& candidate : *tags) {
489 const auto* tag = candidate.getIf<std::string>();
490 if (!tag)
491 return editing::failed<PhysicsPlacementRequest>(
492 editing::Status::Rejected, editing::RuleId("scene.physics-placement.script-tag"),
493 "Placement tags must contain strings");
494 entry.tags.push_back(*tag);
495 }
496 }
497 if (const auto* collidersValue = field(value, "colliders")) {
498 const auto* colliders = collidersValue->getIf<editing::Value::Array>();
499 if (!colliders)
500 return editing::failed<PhysicsPlacementRequest>(
501 editing::Status::Rejected, editing::RuleId("scene.physics-placement.script-colliders"),
502 "Placement colliders must be an array");
503 for (const auto& colliderValue : *colliders) {
504 auto collider = placementCollider(colliderValue);
505 if (!collider.ok()) return editing::Result<PhysicsPlacementRequest>::failure(collider.status());
506 entry.colliders.push_back(std::move(collider.value()));
507 }
508 }
509 request.objects.push_back(std::move(entry));
510 }
511 return editing::applied<PhysicsPlacementRequest>(std::move(request));
512}

514editing::Value placementValue(const PhysicsPlacementFrame& frame) {
515 editing::Value::Array objects;
516 for (const auto& entry : frame.objects) {
517 const auto& value = entry.transform;
518 objects.emplace_back(editing::Value::Object{
519 {"object", entry.object.value()},
520 {"selected", entry.selected},
521 {"position", editing::Value::Array{value.x, value.y, value.z}},
522 {"rotation", editing::Value::Array{value.rotationX, value.rotationY, value.rotationZ}},
523 {"scale", editing::Value::Array{value.scaleX, value.scaleY, value.scaleZ}},
524 });
525 }
526 return editing::Value::Object{
527 {"tick", static_cast<std::int64_t>(frame.tick)},
528 {"colliding", frame.colliding},
529 {"settled", frame.settled},
530 {"surfaceAligned", frame.surfaceAligned},
531 {"surfaceNormal", editing::Value::Array{frame.surfaceNormalX, frame.surfaceNormalY, frame.surfaceNormalZ}},
532 {"objects", std::move(objects)}};
533}
