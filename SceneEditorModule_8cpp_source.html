31 editor::EditorResult<editor::AutomationOwnedTarget> create(
32 const editor::TargetId& target, std::string_view type,
33 const editor::EditorValue::Object& request) override {
34 editor::AutomationOwnedTarget owned;
35 if (type == "scene") {
36 auto sceneTarget = std::make_unique<scene_editing::SceneDocumentTarget>(target.value());
37 const std::string object = stringField(request, "object");
38 if (!object.empty()) {
39 scene_editing::CreateSceneObjectRequest create;
40 create.id = editor::ObjectId(object);
41 create.name = object;
42 auto operation = sceneTarget->makeCreate(create);
43 if (!operation.ok())
44 return editor::EditorResult<editor::AutomationOwnedTarget>::failure(
45 operation.status());
46 auto applied = sceneTarget->applyDomainOperation(operation.value());
47 if (!applied.ok())
48 return editor::EditorResult<editor::AutomationOwnedTarget>::failure(
49 applied.status());
50 }
51 owned.target = std::move(sceneTarget);
52 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
53 }
54
55 const std::string host = stringField(request, "host");
56 if (host.empty())
57 return eve::editing::failed<editor::AutomationOwnedTarget>(
58 editor::EditorStatus::Rejected, editor::RuleId("editor.automation.scene-host-name"),
59 "Live scene target requires a host name");
60 auto* sceneModule = eve::ModuleManager::getInstance<eve::scene::Scene>("Scene");
61 if (!sceneModule)
62 return eve::editing::failed<editor::AutomationOwnedTarget>(
63 editor::EditorStatus::Unsupported, editor::RuleId("editor.automation.scene-module-unavailable"),
64 "Scene module is not available");
65 auto hostResult = sceneModule->findHost(host);
66 if (!hostResult.ok() || !hostResult.value())
67 return eve::editing::failed<editor::AutomationOwnedTarget>(
68 editor::EditorStatus::NotFound, editor::RuleId("editor.automation.scene-host-not-found"),
69 "Live SceneHost does not exist: " + host);
70 owned.target = std::make_unique<scene_editing::SceneHostEditorTarget>(target.value(), hostResult.value());
71 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
72 }
73};
