28bool number(const Value& value, float& result) {
29 if (!value.isNumeric()) return false;
30 const double decoded = value.isInt64() ? double(value.asInt()) : value.asDouble();
31 if (!std::isfinite(decoded) || decoded < -std::numeric_limits<float>::max() ||
32 decoded > std::numeric_limits<float>::max())
33 return false;
34 result = static_cast<float>(decoded);
35 return true;
36}

39bool vector(const Value* value, std::array<float, N>& result) {

55Result<LoadedSceneTemplate> EvpackSceneTemplateLoader::load(
56 const AssetRef& sceneTemplate, const asset::EvpackCapabilities& capabilities,
57 const SceneTemplateLoadLimits& limits) const {
58 auto payload = reader_.read(sceneTemplate, "eve.scene-template/1", capabilities,
59 limits.maximumDecodedBytes);
60 if (!payload) return Result<LoadedSceneTemplate>::failure(payload.status());
61 const asset::RuntimeAssetChunk* definition = nullptr;
62 for (const auto& chunk : payload.value().chunks) {
63 if (chunk.kind != asset::EvpackChunkKind::Definition) continue;
64 if (definition)
65 return failure<LoadedSceneTemplate>(DiagnosticCode::Conflict,
66 "scene template has duplicate definitions");
67 definition = &chunk;
68 }
69 if (!definition)
70 return failure<LoadedSceneTemplate>(DiagnosticCode::NotFound,
71 "scene template definition is missing");
72 asset::RuntimeDefinitionLimits definitionLimits;
73 definitionLimits.maximumBytes = limits.maximumDecodedBytes;
74 auto metadata = asset::decodeRuntimeDefinition(definition->bytes, definitionLimits);
75 if (!metadata) return Result<LoadedSceneTemplate>::failure(metadata.status());
76 const auto* root = metadata.value().getIf<Value::Object>();
77 const Value* schema = root ? field(*root, "schema") : nullptr;
78 const Value* version = root ? field(*root, "schemaVersion") : nullptr;
79 const Value* coordinate = root ? field(*root, "coordinateSystem") : nullptr;
80 const Value* nodesValue = root ? field(*root, "nodes") : nullptr;
81 const auto* nodes = nodesValue ? nodesValue->getIf<Value::Array>() : nullptr;
82 if (!schema || !schema->isString() || schema->asString() != "eve.scene-template" ||
83 !version || !version->isInt64() || version->asInt() != 1 || !coordinate ||
84 !coordinate->isString() ||
85 coordinate->asString() != "right-handed-x-right-y-up-minus-z-forward" || !nodes ||
86 nodes->empty() || nodes->size() > limits.maximumNodes)
87 return failure<LoadedSceneTemplate>(DiagnosticCode::ParseError,
88 "scene-template metadata is invalid");
89 std::map<std::uint64_t, FlatNode> flat;
90 constexpr float radiansToDegrees = 57.29577951308232f;
91 for (std::size_t i = 0; i < nodes->size(); ++i) {
92 const auto* node = (*nodes)[i].getIf<Value::Object>();
93 const Value* sourceId = node ? field(*node, "sourceFileId") : nullptr;
94 const Value* parentId = node ? field(*node, "parentSourceFileId") : nullptr;
95 const Value* objectId = node ? field(*node, "objectId") : nullptr;
96 const Value* name = node ? field(*node, "name") : nullptr;
97 std::array<float, 3> position{}, scale{};
98 std::array<float, 4> rotation{};
99 if (!sourceId || !sourceId->isInt64() || sourceId->asInt() <= 0 || !parentId ||
100 !parentId->isInt64() || parentId->asInt() < 0 || !objectId || !objectId->isString() ||
101 !name || !name->isString() || !vector(node ? field(*node, "position") : nullptr, position) ||
102 !vector(node ? field(*node, "rotation") : nullptr, rotation) ||
103 !vector(node ? field(*node, "scale") : nullptr, scale))
104 return failure<LoadedSceneTemplate>(DiagnosticCode::ParseError,
105 "scene node is malformed",
106 "$.nodes[" + std::to_string(i) + "]");
107 auto persistentId = SceneObjectId::parse(objectId->asString());
108 const float length = std::sqrt(rotation[0] * rotation[0] + rotation[1] * rotation[1] +
109 rotation[2] * rotation[2] + rotation[3] * rotation[3]);
110 if (!persistentId || persistentId->isNil() || length < 0.999f || length > 1.001f ||
111 scale[0] == 0.f || scale[1] == 0.f || scale[2] == 0.f)
112 return failure<LoadedSceneTemplate>(DiagnosticCode::InvalidArgument,
113 "scene identity, rotation, or scale is invalid",
114 "$.nodes[" + std::to_string(i) + "]");
115 const float x = rotation[0] / length, y = rotation[1] / length;
116 const float z = rotation[2] / length, w = rotation[3] / length;
117 const float pitch = std::atan2(2.f * (w * x + y * z), 1.f - 2.f * (x * x + y * y));
118 const float yaw = std::asin(std::clamp(2.f * (w * y - z * x), -1.f, 1.f));
119 const float roll = std::atan2(2.f * (w * z + x * y), 1.f - 2.f * (y * y + z * z));
120 scene::NodeDesc desc;
121 desc.id = std::to_string(sourceId->asInt());
122 desc.key = desc.id;
123 desc.name = name->asString();
124 desc.persistentId = *persistentId;
125 desc.withPosition(position[0], position[1], position[2])
126 .withRotation(yaw * radiansToDegrees, pitch * radiansToDegrees,
127 roll * radiansToDegrees)
128 .withScaleXYZ(scale[0], scale[1], scale[2]);
129 const std::uint64_t id = static_cast<std::uint64_t>(sourceId->asInt());
130 if (!flat.emplace(id, FlatNode{id, static_cast<std::uint64_t>(parentId->asInt()),
131 std::move(desc)}).second)
132 return failure<LoadedSceneTemplate>(DiagnosticCode::Conflict,
133 "scene contains duplicate sourceFileId");
134 }
135 for (const auto& [id, node] : flat)
136 if (node.parentId != 0 && !flat.contains(node.parentId))
137 return failure<LoadedSceneTemplate>(DiagnosticCode::NotFound,
138 "scene parent does not exist", std::to_string(id));
139 std::set<std::uint64_t> visiting, completed;
140 std::function<Result<scene::NodeDesc>(std::uint64_t, std::uint32_t)> build =
141 [&](std::uint64_t id, std::uint32_t depth) -> Result<scene::NodeDesc> {
142 if (depth > limits.maximumDepth)
143 return failure<scene::NodeDesc>(DiagnosticCode::InvalidArgument,
144 "scene hierarchy exceeds depth budget");
145 if (!visiting.emplace(id).second)
146 return failure<scene::NodeDesc>(DiagnosticCode::Conflict,
147 "scene hierarchy contains a cycle");
148 scene::NodeDesc result = flat.at(id).desc;
149 for (const auto& [childId, child] : flat) {
150 if (child.parentId != id) continue;
151 auto built = build(childId, depth + 1);
152 if (!built) return built;
153 result.children.push_back(std::move(built).takeValue());
154 }
155 visiting.erase(id);
156 completed.emplace(id);
157 return Result<scene::NodeDesc>::success(std::move(result));
158 };
159 scene::NodeDesc resultRoot;
160 resultRoot.id = "asset-root";
161 resultRoot.key = "asset-root";
162 resultRoot.name = "Asset Scene Template";
163 for (const auto& [id, node] : flat) {
164 if (node.parentId != 0) continue;
165 auto built = build(id, 1);
166 if (!built) return Result<LoadedSceneTemplate>::failure(built.status());
167 resultRoot.children.push_back(std::move(built).takeValue());
168 }
169 if (resultRoot.children.empty() || completed.size() != flat.size())
170 return failure<LoadedSceneTemplate>(DiagnosticCode::Conflict,
171 "scene hierarchy has no valid root");
172 return Result<LoadedSceneTemplate>::success(
173 {sceneTemplate, std::move(resultRoot), std::move(payload).takeValue().variant});
174}
