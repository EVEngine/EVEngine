11inline float smoothstepClamp(float e0, float e1, float x) {
12 if (e0 == e1) return x < e0 ? 0.f : 1.f;
13 const float t = std::clamp((x - e0) / (e1 - e0), 0.f, 1.f);
14 return t * t * (3.f - 2.f * t);
15}

39float CloudField::coverageAt(float x, float z, float time) const {
40 float vx = 0.f, vz = 0.f;
41 windVelocity(vx, vz);
42
43 const float cell = params_.worldScale / float(kLattice);
44 // Drift with the wind; sample in lattice units (seamless wrap at kLattice).
45 const float px = (x - vx * time) / cell;
46 const float pz = (z - vz * time) / cell;
47
48 NoiseField noise;
49 noise.seed = params_.seed;
50 if (params_.seamless) {
51 noise.periodX = kLattice;
52 noise.periodY = kLattice;
53 }
54
55 // iq / noisecreater nested domain warp on sample coordinates, then
56 // Perlin–Worley coverage (NoiseCreator Cloud / Perlin-Worley hybrid).
57 const int warpOct = std::max(2, params_.octaves - 1);
58 const float amp = params_.warp;
59 const float qx = noise.fbm(px, pz, warpOct, params_.lacunarity, params_.gain) - 0.5f;
60 const float qz = noise.fbm(px + 5.2f, pz + 1.3f, warpOct, params_.lacunarity, params_.gain) - 0.5f;
61 const float rx =
62 noise.fbm(px + amp * qx + 1.7f, pz + amp * qz + 9.2f, warpOct, params_.lacunarity, params_.gain) - 0.5f;
63 const float rz =
64 noise.fbm(px + amp * qx + 8.3f, pz + amp * qz + 2.8f, warpOct, params_.lacunarity, params_.gain) - 0.5f;
65 const float sx = px + amp * rx;
66 const float sz = pz + amp * rz;
67
68 const float base = noise.perlinWorley(sx, sz, params_.octaves);
69 float c = smoothstepClamp(params_.coverage - params_.softness, params_.coverage + params_.softness, base);
70
71 // High-frequency wisps erode dense regions for a textured top.
72 if (params_.detail > 0.f) {
73 const float d = noise.fbmRotated(sx * 3.f + 11.7f, sz * 3.f + 5.3f, 2, params_.lacunarity, params_.gain);
74 c = c * (1.f - params_.detail) + (c * d) * params_.detail;
75 }
76 return std::clamp(c, 0.f, 1.f);
77}

79void CloudField::sample(float *out, int width, int height, float time, float x0, float z0, float extent) const {
80 if (!out || width <= 0 || height <= 0) return;
81 for (int y = 0; y < height; ++y) {
82 const float z = z0 + extent * (float(y) / float(std::max(1, height - 1)));
83 for (int x = 0; x < width; ++x) {
84 const float px = x0 + extent * (float(x) / float(std::max(1, width - 1)));
85 out[size_t(y * width + x)] = coverageAt(px, z, time);
86 }
