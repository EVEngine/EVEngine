10inline float smoothstepClamp(float e0, float e1, float x) {
11 if (e0 == e1) return x < e0 ? 0.f : 1.f;
12 const float t = std::clamp((x - e0) / (e1 - e0), 0.f, 1.f);
13 return t * t * (3.f - 2.f * t);
14}

16inline float fade(float t) { return t * t * (3.f - 2.f * t); }

18struct Lat {
19 int periodX = CloudField::kLattice;
20 int periodY = CloudField::kLattice;
21 uint32_t seed = 1;

23 int wrap(int i, int period) const {
24 if (period <= 0) return i;
25 int m = i % period;
26 return m < 0 ? m + period : m;
27 }

29 float hash01(int ix, int iy) const {
30 ix = wrap(ix, periodX);
31 iy = wrap(iy, periodY);
32 uint32_t h = seed;
33 h ^= uint32_t(ix) * 374761393u;
34 h ^= uint32_t(iy) * 668265263u;
35 h = (h ^ (h >> 13)) * 1274126177u;
36 h ^= h >> 16;
37 return float(h & 0x00FFFFFFu) / float(0x00FFFFFFu);
38 }

40 float valueNoise(float x, float y) const {
41 const int x0 = int(std::floor(x));
42 const int y0 = int(std::floor(y));
43 const float fx = fade(x - float(x0));
44 const float fy = fade(y - float(y0));
45 const float n00 = hash01(x0, y0);
46 const float n10 = hash01(x0 + 1, y0);
47 const float n01 = hash01(x0, y0 + 1);
48 const float n11 = hash01(x0 + 1, y0 + 1);
49 return n00 + (n10 - n00) * fx + (n01 - n00) * fy + (n00 - n10 - n01 + n11) * fx * fy;
50 }
51
52 float fbm(float x, float y, int octaves, float lacunarity, float gain) const {
53 octaves = std::max(1, octaves);
54 float sum = 0.f, amp = 1.f, norm = 0.f, freq = 1.f;
55 for (int i = 0; i < octaves; ++i) {
56 sum += valueNoise(x * freq, y * freq) * amp;
57 norm += amp;
58 amp *= gain;
59 freq *= lacunarity;
60 }
61 return norm > 0.f ? sum / norm : 0.f;
62 }
63};

87float CloudField::coverageAt(float x, float z, float time) const {
88 float vx = 0.f, vz = 0.f;
89 windVelocity(vx, vz);
90
91 const float cell = params_.worldScale / float(kLattice);
92 // Drift with the wind; sample in lattice units (seamless wrap at kLattice).
93 const float px = (x - vx * time) / cell;
94 const float pz = (z - vz * time) / cell;
95
96 Lat lat;
97 lat.seed = params_.seed;
98 if (!params_.seamless) {
99 lat.periodX = 0;
100 lat.periodY = 0;
101 }
102
103 // Domain warp → organic, billowy shapes.
104 const float w = lat.fbm(px + 19.1f, pz + 7.3f, params_.octaves, params_.lacunarity, params_.gain) - 0.5f;
105 const float wx = px + w * params_.warp * 0.5f;
106 const float wz = pz + (lat.fbm(px + 5.2f, pz + 31.7f, params_.octaves, params_.lacunarity, params_.gain) - 0.5f) *
107 params_.warp * 0.5f;
108
109 float base = lat.fbm(wx, wz, params_.octaves, params_.lacunarity, params_.gain);
110 // Threshold into puffs with soft edges.
111 float c = smoothstepClamp(params_.coverage - params_.softness, params_.coverage + params_.softness, base);
112
113 // High-frequency wisps erode dense regions for a textured top.
114 if (params_.detail > 0.f) {
115 const float d = lat.fbm(wx * 3.f + 11.7f, wz * 3.f + 5.3f, 2, params_.lacunarity, params_.gain);
116 c = c * (1.f - params_.detail) + (c * d) * params_.detail;
117 }
118 return std::clamp(c, 0.f, 1.f);
119}
