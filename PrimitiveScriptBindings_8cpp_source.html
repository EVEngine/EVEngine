30eve::Result<PrimitiveDescriptor3D> primitiveDescriptor(ScriptPrimitive3D* value) {
31 if (!value)
32 return primitiveBindingFailure<PrimitiveDescriptor3D>(eve::DiagnosticCode::InvalidArgument,
33 "primitive proxy must not be null", "primitive");
34 auto scene = value->scene.lock();
35 if (!scene)
36 return primitiveBindingFailure<PrimitiveDescriptor3D>(eve::DiagnosticCode::StaleHandle,
37 "primitive scene owner no longer exists", "primitive");
38 const PrimitiveDescriptor3D* descriptor = scene->tryGet(value->handle);
39 if (!descriptor)
40 return primitiveBindingFailure<PrimitiveDescriptor3D>(eve::DiagnosticCode::StaleHandle,
41 "primitive handle is stale", "primitive");
42 return eve::Result<PrimitiveDescriptor3D>::success(*descriptor);
43}

45template <class Mutator>
46eve::Result<PrimitiveUpdateStatus> mutatePrimitive(ScriptPrimitive3D* value, Mutator&& mutator) {
47 auto descriptor = primitiveDescriptor(value);
48 if (!descriptor) return eve::Result<PrimitiveUpdateStatus>::failure(descriptor.status());
49 PrimitiveDescriptor3D copy = std::move(descriptor).takeValue();
50 std::invoke(std::forward<Mutator>(mutator), copy);
51 auto scene = value->scene.lock();
52 if (!scene)
53 return primitiveBindingFailure<PrimitiveUpdateStatus>(eve::DiagnosticCode::StaleHandle,
54 "primitive scene owner no longer exists", "primitive");
55 return scene->update(value->handle, std::move(copy));
56}

58ssq::Table makePrimitiveProxy(HSQUIRRELVM vm, const std::shared_ptr<PrimitiveScene>& scene,
59 eve::Result<PrimitiveHandle>&& result) {
60 if (!result) return eve::script::projectStatusResult(vm, result.status(), false, false);
61 const PrimitiveHandle handle = std::move(result).takeValue();
62 auto object = eve::script::makeOwnedSquirrelInstance<ScriptPrimitive3D>(
63 vm, std::make_unique<ScriptPrimitive3D>(scene, handle));
64 if (!object) {
65 const eve::Status status = object.status();
66 object.ignore("failed to create owned primitive proxy");
67 scene->remove(handle).ignore("rollback failed primitive proxy allocation");
68 return eve::script::projectStatusResult(vm, status, false, false);
69 }
70 auto projected = eve::script::projectStatusResult(vm, eve::Status::success(eve::StatusCode::Applied), true, false);
71 projected.set("value", std::move(object).takeValue());
72 projected.set("ownership", std::string("owned"));
73 projected.set("owner", static_cast<std::int64_t>(handle.owner()));
74 projected.set("index", static_cast<std::int64_t>(handle.index()));
75 projected.set("generation", static_cast<std::int64_t>(handle.generation()));
76 return projected;
77}

88eve::Result<std::vector<glm::vec3>> scriptPoints(ssq::Array values, std::size_t minimum, std::size_t maximum) {
89 if (values.size() % 3 != 0 || values.size() / 3 < minimum || values.size() / 3 > maximum)
90 return primitiveBindingFailure<std::vector<glm::vec3>>(eve::DiagnosticCode::InvalidArgument,
91 "expected a flat xyz array with the required point count", "points");
92 std::vector<glm::vec3> points;
93 points.reserve(values.size() / 3);
94 try {
95 for (std::size_t i = 0; i < values.size(); i += 3)
96 points.push_back({values.get<float>(i), values.get<float>(i+1), values.get<float>(i+2)});
97 } catch (const std::exception& error) {
98 return primitiveBindingFailure<std::vector<glm::vec3>>(eve::DiagnosticCode::InvalidArgument,
99 error.what(), "points");
100 }
101 return eve::Result<std::vector<glm::vec3>>::success(std::move(points));
102}
