73std::string applyOneAction(Poco::JSON::Object::Ptr action, Poco::JSON::Object::Ptr receipt) {
74 if (!action || !action->has("op")) return "action.op is required";
75 std::string op;
76 try {
77 op = action->get("op").convert<std::string>();
78 } catch (...) {
79 return "action.op must be a string";
80 }
81
82 auto* input = eve::cap::query<eve::IInputInjection>();
83 auto* ui = eve::cap::query<eve::IUIAutomation>();
84
85 if (op == "wait") {
86 const int frames = std::max(0, getArgInt(action, "frames", 1));
87 receipt->set("waitFrames", receipt->has("waitFrames")
88 ? receipt->getValue<int>("waitFrames") + frames
89 : frames);
90 return {};
91 }
92
93 if (op == "key") {
94 if (!input) return std::string(eve::IInputInjection::capabilityName) + " unavailable";
95 const std::string key = getArgString(action, "key");
96 const bool down = getArgBool(action, "down", true);
97 const bool repeat = getArgBool(action, "repeat", false);
98 auto result = input->injectKey(key, down, repeat);
99 if (!result) return diagnosticMessage(result.status());
100 return {};
101 }
102 if (op == "mouse_move") {
103 if (!input) return std::string(eve::IInputInjection::capabilityName) + " unavailable";
104 auto result = input->injectMouseMove(getArgFloat(action, "x"), getArgFloat(action, "y"),
105 getArgFloat(action, "dx"), getArgFloat(action, "dy"));
106 if (!result) return diagnosticMessage(result.status());
107 return {};
108 }
109 if (op == "mouse_button") {
110 if (!input) return std::string(eve::IInputInjection::capabilityName) + " unavailable";
111 auto result = input->injectMouseButton(getArgInt(action, "button", 1), getArgBool(action, "down", true),
112 getArgFloat(action, "x"), getArgFloat(action, "y"));
113 if (!result) return diagnosticMessage(result.status());
114 return {};
115 }
116 if (op == "wheel") {
117 if (!input) return std::string(eve::IInputInjection::capabilityName) + " unavailable";
118 auto result = input->injectWheel(getArgFloat(action, "x"), getArgFloat(action, "y", -1.f));
119 if (!result) return diagnosticMessage(result.status());
120 return {};
121 }
122 if (op == "text") {
123 if (!input) return std::string(eve::IInputInjection::capabilityName) + " unavailable";
124 auto result = input->injectText(getArgString(action, "text"));
125 if (!result) return diagnosticMessage(result.status());
126 return {};
127 }
128 if (op == "ui_click") {
129 if (!ui) return std::string(eve::IUIAutomation::capabilityName) + " unavailable";
130 const std::string host = getArgString(action, "host");
131 const std::string widget = getArgString(action, "widget");
132 if (widget.empty()) return "ui_click requires widget";
133 const std::string out = ui->click(host, widget);
134 if (out.rfind("error:", 0) == 0) return out.substr(6);
135 return {};
136 }
137 return "unknown action.op '" + op + "' (key|mouse_move|mouse_button|wheel|text|ui_click|wait)";
138}

140void finishCapture(PendingBatch& batch) {
141 Poco::JSON::Object::Ptr out = batch.result ? batch.result : Poco::JSON::Object::Ptr(new Poco::JSON::Object());
142 out->set("id", static_cast<Poco::Int64>(batch.id));
143 out->set("status", "ready");
144 if (!batch.wantCapture) {
145 out->set("ok", true);
146 batch.result = out;
147 batch.done = true;
148 return;
149 }
150 auto* capture = eve::cap::query<eve::IRenderCapture>();
151 if (!capture) {
152 out->set("ok", false);
153 out->set("error", std::string(eve::IRenderCapture::capabilityName) + " unavailable");
154 out->set("hint", "Wait until graphics has presented a frame, then eve_screenshot / eve_input_poll");
155 Poco::JSON::Array::Ptr next = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
156 next->add("eve_render_status");
157 next->add("eve_screenshot");
158 out->set("nextTools", next);
159 batch.result = out;
160 batch.done = true;
161 return;
162 }
163 std::string path = batch.capturePath.empty() ? "mcp_input_capture.png" : batch.capturePath;
164 auto saved = capture->savePng(path);
165 if (!saved) {
166 const auto& status = saved.status();
167 out->set("ok", false);
168 out->set("retryable", true);
169 out->set("reason", "no-presented-frame");
170 out->set("error", diagnosticMessage(status));
171 out->set("hint", "Call eve_input_poll again after one rendered frame, or use eve_screenshot");
172 Poco::JSON::Array::Ptr next = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
173 next->add("eve_input_poll");
174 next->add("eve_screenshot");
175 out->set("nextTools", next);
176 batch.result = out;
177 batch.done = true;
178 return;
179 }
180 out->set("ok", true);
181 out->set("captured", true);
182 out->set("path", path);
183 out->set("width", saved.value().width);
184 out->set("height", saved.value().height);
185 batch.result = out;
186 batch.done = true;
187}

201std::string inputPoll(Poco::JSON::Object::Ptr args) {
202 const std::int64_t id = getArgInt64(args, "id", 0);
203 std::lock_guard lock(g_mu);
204 if (id > 0) {
205 if (auto receipt = findReceipt(id)) return textPayload(receipt);
206 for (auto& batch : g_pending) {
207 if (batch.id == id) {
208 Poco::JSON::Object::Ptr out = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
209 out->set("ok", true);
210 out->set("status", "pending");
211 out->set("id", static_cast<Poco::Int64>(id));
212 out->set("framesLeft", batch.framesLeft);
213 out->set("hint", "Keep calling eve_input_poll each frame until status=ready");
214 Poco::JSON::Array::Ptr next = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
215 next->add("eve_input_poll");
216 out->set("nextTools", next);
217 return textPayload(out);
218 }
219 }
220 return errorPayload("unknown input batch id " + std::to_string(id),
221 "Call eve_input_batch first and reuse the returned id", {"eve_input_batch"});
222 }
223 if (!g_receipts.empty()) return textPayload(g_receipts.back());
224 if (!g_pending.empty()) {
225 Poco::JSON::Object::Ptr out = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
226 out->set("ok", true);
227 out->set("status", "pending");
228 out->set("id", static_cast<Poco::Int64>(g_pending.back().id));
229 out->set("framesLeft", g_pending.back().framesLeft);
230 out->set("hint", "Keep calling eve_input_poll each frame until status=ready");
231 return textPayload(out);
232 }
233 return errorPayload("no input batch to poll", "Submit actions with eve_input_batch", {"eve_input_batch"});
234}

236std::string inputBatch(Poco::JSON::Object::Ptr args) {
237 if (!args || !args->has("actions")) {
238 return errorPayload("actions array is required",
239 "Pass actions:[{op:key|mouse_move|mouse_button|wheel|text|ui_click|wait},...]",
240 {"eve_input_status"});
241 }
242 Poco::JSON::Array::Ptr actions;
243 try {
244 actions = args->getArray("actions");
245 } catch (...) {
246 return errorPayload("actions must be a JSON array");
247 }
248 if (!actions || actions->size() == 0) return errorPayload("actions must not be empty");
249
250 if (!eve::cap::query<eve::IInputInjection>() && !eve::cap::query<eve::IUIAutomation>()) {
251 return unavailablePayload(eve::IInputInjection::capabilityName);
252 }
253
254 Poco::JSON::Object::Ptr receipt = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
255 receipt->set("injected", 0);
256 Poco::JSON::Array::Ptr steps = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
257 int waitFrames = 0;
258
259 for (std::size_t i = 0; i < actions->size(); ++i) {
260 Poco::JSON::Object::Ptr action;
261 try {
262 action = actions->getObject(static_cast<unsigned>(i));
263 } catch (...) {
264 return errorPayload("actions[" + std::to_string(i) + "] must be an object");
265 }
266 const int beforeWait = receipt->has("waitFrames") ? receipt->getValue<int>("waitFrames") : 0;
267 const std::string err = applyOneAction(action, receipt);
268 Poco::JSON::Object::Ptr step = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
269 step->set("index", static_cast<int>(i));
270 if (action && action->has("op")) step->set("op", action->get("op"));
271 if (!err.empty()) {
272 step->set("ok", false);
273 step->set("error", err);
274 steps->add(step);
275 receipt->set("ok", false);
276 receipt->set("error", err);
277 receipt->set("steps", steps);
278 receipt->set("hint", "Fix the failing action; key names use SDL_GetKeyName spelling");
279 Poco::JSON::Array::Ptr next = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
280 next->add("eve_input_status");
281 receipt->set("nextTools", next);
282 return textPayload(receipt, true);
283 }
284 step->set("ok", true);
285 steps->add(step);
286 const int afterWait = receipt->has("waitFrames") ? receipt->getValue<int>("waitFrames") : 0;
287 if (afterWait == beforeWait) receipt->set("injected", receipt->getValue<int>("injected") + 1);
288 waitFrames = afterWait;
289 }
290
291 const bool wantCapture = getArgBool(args, "capture", false);
292 std::string capturePath = getArgString(args, "capturePath", "mcp_input_capture.png");
293 // Top-level waitFrames adds to action waits (Godot-style batch trailer).
294 waitFrames += std::max(0, getArgInt(args, "waitFrames", 0));
295
296 receipt->set("steps", steps);
297 receipt->set("waitFrames", waitFrames);
298 receipt->set("capture", wantCapture);
299
300 if (waitFrames <= 0 && !wantCapture) {
301 receipt->set("ok", true);
302 receipt->set("status", "ready");
303 receipt->set("hint", "Events are queued for the next PlatformEvent::pump; compose with eve_play step "
304 "or waitFrames/capture for a closed loop");
305 Poco::JSON::Array::Ptr next = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
306 next->add("eve_play");
307 next->add("eve_screenshot");
308 receipt->set("nextTools", next);
309 return textPayload(receipt);
310 }
311
312 PendingBatch batch;
313 {
314 std::lock_guard lock(g_mu);
315 batch.id = g_nextId++;
316 batch.framesLeft = waitFrames;
317 batch.wantCapture = wantCapture;
318 batch.capturePath = std::move(capturePath);
319 batch.result = receipt;
320 if (batch.framesLeft <= 0) {
321 finishCapture(batch);
322 g_receipts.push_back(batch.result);
323 return textPayload(batch.result);
324 }
325 g_pending.push_back(batch);
326 }
327
328 receipt->set("ok", true);
329 receipt->set("status", "pending");
330 receipt->set("id", static_cast<Poco::Int64>(batch.id));
331 receipt->set("hint", "Call eve_input_poll after each game frame until status=ready");
332 Poco::JSON::Array::Ptr next = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
333 next->add("eve_input_poll");
334 receipt->set("nextTools", next);
335 return textPayload(receipt);
336}

355void pollMcpInputBatches() {
356 std::vector<PendingBatch> finished;
357 {
358 std::lock_guard lock(g_mu);
359 for (auto& batch : g_pending) {
360 if (batch.framesLeft > 0) --batch.framesLeft;
361 if (batch.framesLeft <= 0) {
362 finishCapture(batch);
363 finished.push_back(batch);
364 }
365 }
366 g_pending.erase(std::remove_if(g_pending.begin(), g_pending.end(),
367 [](const PendingBatch& b) { return b.framesLeft <= 0 && b.done; }),
368 g_pending.end());
369 for (auto& batch : finished) g_receipts.push_back(batch.result);
370 constexpr std::size_t kMaxReceipts = 32;
371 if (g_receipts.size() > kMaxReceipts) {
372 g_receipts.erase(g_receipts.begin(), g_receipts.end() - static_cast<std::ptrdiff_t>(kMaxReceipts));
373 }
374 }
375}
