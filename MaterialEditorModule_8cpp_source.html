48editor::EditorValue runtimeMaterialSnapshot(const graphics::Renderable3D::MeshRenderer& renderer) {
49 material_editing::MaterialDocumentTarget defaults("material.runtime.seed");
50 editor::EditorValue snapshot = defaults.snapshotValue();
51 auto* root = snapshot.getIf<editor::EditorValue::Object>();
52 auto* properties = root ? root->at("properties").getIf<editor::EditorValue::Object>() : nullptr;
53 (*properties)["shading.tint"] = editor::EditorValue::Array{double(renderer.r), double(renderer.g),
54 double(renderer.b), double(renderer.a)};
55 (*properties)["shading.metallic"] = double(renderer.metallic);
56 (*properties)["shading.roughness"] = double(renderer.roughness);
57 (*properties)["parallax.scale"] = double(renderer.parallaxScale);
58 (*properties)["lighting.receive"] = renderer.receiveLight;
59 (*properties)["shadow.cast"] = renderer.castShadow;
60 (*properties)["shadow.receive"] = renderer.receiveShadow;
61 return snapshot;
62}

66class MaterialEditorModule::TargetFactory final : public editor::IEditorAutomationTargetFactory {
67public:
68 bool supports(std::string_view type) const override {
69 return type == "material" || type == "material-renderable3d";
70 }
71
72 editor::EditorResult<editor::AutomationOwnedTarget> create(
73 const editor::TargetId& target, std::string_view type,
74 const editor::EditorValue::Object& request) override {
75 editor::AutomationOwnedTarget owned;
76 if (type == "material") {
77 owned.target = std::make_unique<material_editing::MaterialDocumentTarget>(target.value());
78 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
79 }
80 const auto* id = integerField(request, "entityId");
81 const auto* generation = integerField(request, "generation");
82 constexpr auto maxId = static_cast<std::int64_t>(std::numeric_limits<std::uint32_t>::max());
83 if (!id || !generation || *id < 0 || *generation < 0 || *id > maxId || *generation > maxId)
84 return eve::editing::failed<editor::AutomationOwnedTarget>(
85 editor::EditorStatus::Rejected, editor::RuleId("editor.automation.material-handle"),
86 "Live material target requires uint32 entityId and generation");
87 auto* renderable = findRenderable(static_cast<std::uint32_t>(*id), static_cast<std::uint32_t>(*generation));
88 if (!renderable)
89 return eve::editing::failed<editor::AutomationOwnedTarget>(
90 editor::EditorStatus::Conflict, editor::RuleId("editor.automation.material-stale"),
91 "Renderable3D handle is missing or stale");
92 auto renderer = renderable->meshRenderer();
93 if (renderer->material || renderer->texture || renderer->normalTexture || renderer->heightTexture ||
94 renderer->shader)
95 return eve::editing::failed<editor::AutomationOwnedTarget>(
96 editor::EditorStatus::Unsupported, editor::RuleId("editor.automation.material-assets"),
97 "Live material binding currently supports field-backed materials without asset overrides");
98 static const AutomationMaterialAssetResolver assets;
99 auto sink = std::make_shared<material_editing::Renderable3DMaterialRuntimeSink>(renderable, &assets);
100 auto publishing = std::make_unique<material_editing::MaterialPublishingTarget>(target.value(), sink.get());
101 auto loaded = publishing->authoringTarget().loadSnapshot(runtimeMaterialSnapshot(*renderer));
102 if (!loaded.ok()) {
103 return editor::EditorResult<editor::AutomationOwnedTarget>::failure(loaded.status());
104 }
105 owned.support = std::move(sink);
106 owned.target = std::move(publishing);
107 return eve::editing::applied<editor::AutomationOwnedTarget>(std::move(owned));
108 }
109};
