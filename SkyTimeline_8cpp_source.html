49SkyFrame SkyTimeline::frame() const noexcept {
50 SkyFrame result;
51 result.elapsedSeconds = elapsed_.seconds();
52 // Reduce before scaling so long-running clocks retain sub-day precision.
53 result.hour = clock_.hoursPerSecond == 0
54 ? clock_.initialHour
55 : std::fmod(clock_.initialHour +
56 std::fmod(elapsed_.seconds(), 24 / clock_.hoursPerSecond) * clock_.hoursPerSecond,
57 24);
58 const double age = Duration::fromNanoseconds(elapsed_.nanoseconds() - transitionStart_.nanoseconds()).seconds();
59 const double duration = transitionDuration_.seconds();
60 const double ramp = std::min(age, duration);
61 const double fraction = duration > 0 ? ramp / duration : 1;
62 const auto mix = [fraction](double a, double b) { return std::lerp(a, b, fraction); };
63 result.weather.cloudCoverage = mix(from_.cloudCoverage, to_.cloudCoverage);
64 result.weather.fog = mix(from_.fog, to_.fog);
65 result.weather.rain = mix(from_.rain, to_.rain);
66 result.weather.snow = mix(from_.snow, to_.snow);
67 for (size_t axis = 0; axis < 2; ++axis) {
68 result.weather.windXZ[axis] = mix(from_.windXZ[axis], to_.windXZ[axis]);
69 // Integrate the linear ramp plus constant-speed tail analytically. A large
70 // frame crossing the transition endpoint must equal many smaller frames.
71 result.cloudTravelXZ[axis] = travelOrigin_[axis] +
72 ramp * (from_.windXZ[axis] + result.weather.windXZ[axis]) * .5 +
73 (age - ramp) * to_.windXZ[axis];
74 }
75 return result;
76}
