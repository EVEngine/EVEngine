42Result<std::optional<double>> optionalDouble(const Value::Object& object, const char* name, std::string_view path) {
43 const auto* value = field(object, name);
44 if (!value) return Result<std::optional<double>>::success(std::nullopt);
45 if (const auto* number = value->getIf<double>()) {
46 if (!std::isfinite(*number))
47 return failT<std::optional<double>>(DiagnosticCode::InvalidArgument, "Expected finite number", path);
48 return Result<std::optional<double>>::success(*number);
49 }
50 if (const auto* integer = value->getIf<std::int64_t>()) {
51 const double converted = static_cast<double>(*integer);

53 return failT<std::optional<double>>(DiagnosticCode::InvalidArgument, "Expected finite number", path);
54 return Result<std::optional<double>>::success(converted);
55 }
56 return failT<std::optional<double>>(DiagnosticCode::InvalidArgument, "Expected number", path);

59Result<std::optional<int>> optionalInt(const Value::Object& object, const char* name, std::string_view path) {
60 const auto* value = field(object, name);
61 if (!value) return Result<std::optional<int>>::success(std::nullopt);
62 if (const auto* integer = value->getIf<std::int64_t>()) {
63 if (*integer < std::numeric_limits<int>::min() || *integer > std::numeric_limits<int>::max())
64 return failT<std::optional<int>>(DiagnosticCode::InvalidArgument, "Integer out of range", path);
65 return Result<std::optional<int>>::success(static_cast<int>(*integer));
66 }
67 if (const auto* number = value->getIf<double>()) {
68 if (!std::isfinite(*number) || *number != std::floor(*number))
69 return failT<std::optional<int>>(DiagnosticCode::InvalidArgument, "Expected integer", path);
70 if (*number < static_cast<double>(std::numeric_limits<int>::min()) ||
71 *number > static_cast<double>(std::numeric_limits<int>::max()))
72 return failT<std::optional<int>>(DiagnosticCode::InvalidArgument, "Integer out of range", path);
73 return Result<std::optional<int>>::success(static_cast<int>(*number));
74 }
75 return failT<std::optional<int>>(DiagnosticCode::InvalidArgument, "Expected integer", path);

339Result<CarrierRecipe> decodePreset(const Value::Object& object) {
340 CarrierRecipe recipe;
341 auto id = readLogicalId(object, "id", "id");
342 if (!id.ok()) return Result<CarrierRecipe>::failure(id.status());
343 recipe.id = id.value();
344
345 auto lifetime = readSeconds(object, "lifetime", "lifetime");
346 if (!lifetime.ok()) return Result<CarrierRecipe>::failure(lifetime.status());
347 recipe.lifetime = lifetime.value();
348
349 auto speedField = optionalDouble(object, "speed", "speed");
350 if (!speedField.ok()) return Result<CarrierRecipe>::failure(speedField.status());
351 if (!speedField.value().has_value() || !(*speedField.value() > 0.0))
352 return failT<CarrierRecipe>(DiagnosticCode::InvalidArgument, "Preset requires positive speed", "speed");
353 recipe.speed = *speedField.value();
354
355 double damage = 0.0;
356 if (auto assigned = assignOptionalDouble(object, "damage", damage, "damage"); !assigned)
357 return Result<CarrierRecipe>::failure(assigned.status());
358 std::string damageType;
359 std::string element;
360 (void)readString(object, "damageType", damageType);
361 (void)readString(object, "element", element);
362
363 double gravity = 0.0;
364 double homing = 0.0;
365 double accelerate = 0.0;
366 if (auto assigned = assignOptionalDouble(object, "gravity", gravity, "gravity"); !assigned)
367 return Result<CarrierRecipe>::failure(assigned.status());
368 if (auto assigned = assignOptionalDouble(object, "homing", homing, "homing"); !assigned)
369 return Result<CarrierRecipe>::failure(assigned.status());
370 if (homing == 0.0) {
371 if (auto assigned = assignOptionalDouble(object, "maxTurnRateDegrees", homing, "maxTurnRateDegrees"); !assigned)
372 return Result<CarrierRecipe>::failure(assigned.status());
373 }
374 if (auto assigned = assignOptionalDouble(object, "accelerate", accelerate, "accelerate"); !assigned)
375 return Result<CarrierRecipe>::failure(assigned.status());
376
377 int pierce = 0;
378 int bounce = 0;
379 if (auto assigned = assignOptionalInt(object, "pierce", pierce, "pierce"); !assigned)
380 return Result<CarrierRecipe>::failure(assigned.status());
381 if (auto assigned = assignOptionalInt(object, "bounce", bounce, "bounce"); !assigned)
382 return Result<CarrierRecipe>::failure(assigned.status());
383
384 double fuse = 0.0;
385 double splash = 0.0;
386 if (auto assigned = assignOptionalDouble(object, "fuse", fuse, "fuse"); !assigned)
387 return Result<CarrierRecipe>::failure(assigned.status());
388 if (auto assigned = assignOptionalDouble(object, "splash", splash, "splash"); !assigned)
389 return Result<CarrierRecipe>::failure(assigned.status());
390
391 std::string motionPreset = "linear";
392 (void)readString(object, "motion", motionPreset);
393
394 if (homing > 0.0 || motionPreset == "homing") {
395 CarrierMotionOp op;
396 op.kind = CarrierMotionOpKind::SteerHoming;
397 op.maxTurnRateDegrees = homing > 0.0 ? homing : 90.0;
398 recipe.motionOps.push_back(op);
399 }
400 if (gravity > 0.0 || motionPreset == "ballistic") {
401 CarrierMotionOp op;
402 op.kind = CarrierMotionOpKind::ApplyGravity;
403 op.gravity = gravity > 0.0 ? gravity : 9.8;
404 recipe.motionOps.push_back(op);
405 }
406 if (accelerate > 0.0) {
407 CarrierMotionOp op;
408 op.kind = CarrierMotionOpKind::Accelerate;
409 op.acceleration = accelerate;
410 recipe.motionOps.push_back(op);
411 }
412 recipe.motionOps.push_back({CarrierMotionOpKind::IntegrateLinear});
413
414 recipe.triggers.push_back({CarrierTriggerKind::OnExpire});
415 recipe.triggers.push_back({CarrierTriggerKind::OnHit});
416 if (fuse > 0.0) {
417 CarrierTrigger trigger;
418 trigger.kind = CarrierTriggerKind::OnFuse;
419 auto duration = Duration::fromSeconds(fuse);
420 if (!duration.ok()) return Result<CarrierRecipe>::failure(duration.status());
421 trigger.fuse = duration.value();
422 recipe.triggers.push_back(trigger);
423 }
424
425 if (damage > 0.0 || !element.empty() || !damageType.empty()) {
426 CarrierImpact hit;
427 hit.kind = CarrierImpactKind::EmitHit;
428 hit.on = CarrierTriggerKind::OnHit;
429 hit.damage = damage;
430 hit.damageType = damageType;
431 hit.element = element;
432 recipe.impacts.push_back(hit);
433 }
434 if (pierce > 0) {
435 CarrierImpact impact;
436 impact.kind = CarrierImpactKind::Pierce;
437 impact.on = CarrierTriggerKind::OnHit;
438 impact.pierceCount = pierce;
439 recipe.impacts.push_back(impact);
440 }
441 if (bounce > 0) {
442 CarrierImpact impact;
443 impact.kind = CarrierImpactKind::Bounce;
444 impact.on = CarrierTriggerKind::OnHit;
445 impact.bounceCount = bounce;
446 recipe.impacts.push_back(impact);
447 }
448 recipe.impacts.push_back({CarrierImpactKind::Release, CarrierTriggerKind::OnHit});
449
450 if (fuse > 0.0) {
451 CarrierImpact splashImpact;
452 splashImpact.kind = splash > 0.0 ? CarrierImpactKind::Splash : CarrierImpactKind::EmitHit;
453 splashImpact.on = CarrierTriggerKind::OnFuse;
454 splashImpact.damage = damage;
455 splashImpact.splashRadius = splash;
456 splashImpact.damageType = damageType;
457 splashImpact.element = element;
458 recipe.impacts.push_back(splashImpact);
459 recipe.impacts.push_back({CarrierImpactKind::Release, CarrierTriggerKind::OnFuse});
460 }
461 if (splash > 0.0 && fuse <= 0.0) {
462 CarrierImpact splashImpact;
463 splashImpact.kind = CarrierImpactKind::Splash;

479Result<CarrierRecipe> decodeFull(const Value::Object& object) {
480 CarrierRecipe recipe;
481 auto id = readLogicalId(object, "id", "id");
482 if (!id.ok()) return Result<CarrierRecipe>::failure(id.status());
483 recipe.id = id.value();
484
485 auto lifetime = readSeconds(object, "lifetime", "lifetime");
486 if (!lifetime.ok()) return Result<CarrierRecipe>::failure(lifetime.status());
487 recipe.lifetime = lifetime.value();
488
489 auto speedField = optionalDouble(object, "speed", "speed");
490 if (!speedField.ok()) return Result<CarrierRecipe>::failure(speedField.status());
491 if (!speedField.value().has_value() || !(*speedField.value() > 0.0))
492 return failT<CarrierRecipe>(DiagnosticCode::InvalidArgument, "Recipe requires positive speed", "speed");
493 recipe.speed = *speedField.value();
494
495 const auto* motion = field(object, "motion");
496 if (!motion) motion = field(object, "motionOps");
497 if (!motion || !motion->isArray())
498 return failT<CarrierRecipe>(DiagnosticCode::InvalidArgument, "Full recipe requires motion array", "motion");
499 for (std::size_t i = 0; i < motion->arraySize(); ++i) {
500 auto op = decodeMotionOp(motion->at(i), "motion[" + std::to_string(i) + "]");
501 if (!op.ok()) return Result<CarrierRecipe>::failure(op.status());
502 recipe.motionOps.push_back(op.value());
503 }
504
505 const auto* triggers = field(object, "triggers");
506 if (!triggers || !triggers->isArray())
507 return failT<CarrierRecipe>(DiagnosticCode::InvalidArgument, "Full recipe requires triggers array", "triggers");
508 for (std::size_t i = 0; i < triggers->arraySize(); ++i) {
509 auto trigger = decodeTrigger(triggers->at(i), "triggers[" + std::to_string(i) + "]");
510 if (!trigger.ok()) return Result<CarrierRecipe>::failure(trigger.status());
511 recipe.triggers.push_back(trigger.value());
512 }
513
514 const auto* impacts = field(object, "impacts");
515 if (!impacts || !impacts->isArray())
516 return failT<CarrierRecipe>(DiagnosticCode::InvalidArgument, "Full recipe requires impacts array", "impacts");
517 for (std::size_t i = 0; i < impacts->arraySize(); ++i) {
518 auto impact = decodeImpact(impacts->at(i), "impacts[" + std::to_string(i) + "]");
519 if (!impact.ok()) return Result<CarrierRecipe>::failure(impact.status());
520 recipe.impacts.push_back(impact.value());
521 }
522
523 auto valid = recipe.validate();
524 if (!valid) return Result<CarrierRecipe>::failure(valid.status());
525 return Result<CarrierRecipe>::success(std::move(recipe));
526}

530Result<CarrierRecipe> decodeCarrierRecipe(const Value& value) {
531 const auto* object = value.getIf<Value::Object>();
532 if (!object) return failT<CarrierRecipe>(DiagnosticCode::InvalidArgument, "Carrier recipe must be an object", "");
533 if (field(*object, "motionOps") || field(*object, "triggers") || field(*object, "impacts")) {
534 if (const auto* motion = field(*object, "motion"); motion && motion->isString()) {
535 // Preset with convenience motion string plus optional explicit arrays is ambiguous —
536 // prefer full form only when motion is an array or motionOps is present.
537 }
538 if (field(*object, "motionOps") || (field(*object, "motion") && field(*object, "motion")->isArray()))
539 return decodeFull(*object);
540 }
541 if (field(*object, "motion") && field(*object, "motion")->isArray() && field(*object, "triggers") &&
542 field(*object, "impacts"))
543 return decodeFull(*object);
544 return decodePreset(*object);
545}

547Result<Value> encodeCarrierRecipe(const CarrierRecipe& recipe) {
548 auto valid = recipe.validate();
549 if (!valid) return Result<Value>::failure(valid.status());
550
551 Value::Object object;
552 object.emplace("id", Value(recipe.id.format()));
553 object.emplace("lifetime", Value(recipe.lifetime.seconds()));
554 object.emplace("speed", Value(recipe.speed));
555
556 Value::Array motion;
557 for (const auto& op : recipe.motionOps) {
558 Value::Object entry;
559 entry.emplace("kind", Value(std::string(motionKindName(op.kind))));
560 if (op.gravity != 0.0) entry.emplace("gravity", Value(op.gravity));
561 if (op.maxTurnRateDegrees != 0.0) entry.emplace("maxTurnRateDegrees", Value(op.maxTurnRateDegrees));
562 if (op.acceleration != 0.0) entry.emplace("acceleration", Value(op.acceleration));
563 if (op.avoidLookAhead != 0.0) entry.emplace("avoidLookAhead", Value(op.avoidLookAhead));
564 if (op.avoidStrength != 1.0) entry.emplace("avoidStrength", Value(op.avoidStrength));
565 if (op.avoidRadiusPadding != 0.0) entry.emplace("avoidRadiusPadding", Value(op.avoidRadiusPadding));
566 if (op.curveAmplitude != 0.0) entry.emplace("curveAmplitude", Value(op.curveAmplitude));
567 if (op.curveFrequencyHz != 0.0) entry.emplace("curveFrequencyHz", Value(op.curveFrequencyHz));
568 motion.push_back(Value(std::move(entry)));
569 }
570 object.emplace("motion", Value(std::move(motion)));
571
572 Value::Array triggers;
573 for (const auto& trigger : recipe.triggers) {
574 Value::Object entry;
575 entry.emplace("kind", Value(std::string(triggerKindName(trigger.kind))));
576 if (trigger.fuse > Duration::zero()) entry.emplace("fuse", Value(trigger.fuse.seconds()));
577 if (trigger.interval > Duration::zero()) entry.emplace("interval", Value(trigger.interval.seconds()));
578 if (trigger.proximityRadius > 0.0) entry.emplace("proximityRadius", Value(trigger.proximityRadius));
579 triggers.push_back(Value(std::move(entry)));
580 }
581 object.emplace("triggers", Value(std::move(triggers)));
582
583 Value::Array impacts;
584 for (const auto& impact : recipe.impacts) {
585 Value::Object entry;
586 entry.emplace("kind", Value(std::string(impactKindName(impact.kind))));
587 entry.emplace("on", Value(std::string(triggerKindName(impact.on))));
588 if (impact.damage != 0.0) entry.emplace("damage", Value(impact.damage));
589 if (impact.splashRadius != 0.0) entry.emplace("splashRadius", Value(impact.splashRadius));
590 if (impact.pierceCount != 0) entry.emplace("pierceCount", Value(static_cast<std::int64_t>(impact.pierceCount)));
591 if (impact.bounceCount != 0) entry.emplace("bounceCount", Value(static_cast<std::int64_t>(impact.bounceCount)));
592 if (impact.restitution != 1.0) entry.emplace("restitution", Value(impact.restitution));
593 if (!impact.damageType.empty()) entry.emplace("damageType", Value(impact.damageType));
594 if (!impact.element.empty()) entry.emplace("element", Value(impact.element));
595 if (impact.childRecipeId.isValid()) entry.emplace("childRecipeId", Value(impact.childRecipeId.format()));
596 impacts.push_back(Value(std::move(entry)));
597 }
598 object.emplace("impacts", Value(std::move(impacts)));
599 return Result<Value>::success(Value(std::move(object)));
600}
