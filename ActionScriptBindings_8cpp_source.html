32ssq::Table newRuntime(const script::BindContext& bind) {
33 auto object =
34 script::makeOwnedSquirrelInstance<ScriptActionRuntime>(bind.vm(), std::make_unique<ScriptActionRuntime>());
35 if (!object) return script::projectStatusResult(bind.vm(), object.status());
36 auto result = script::projectStatusResult(bind.vm(), Status::success(StatusCode::Applied));
37 result.set("value", std::move(object).takeValue());
38 result.set("ownership", std::string("owned"));
39 return result;
