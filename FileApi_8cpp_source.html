23bool readViaPhysFS(const char* path, std::string& out) {
24 if (!PHYSFS_isInit() || !PHYSFS_exists(path)) return false;
25 PHYSFS_file* f = PHYSFS_openRead(path);
26 if (!f) return false;
27 const PHYSFS_uint64 len = PHYSFS_fileLength(f);
28 out.resize(static_cast<size_t>(len));
29 if (len > 0 && PHYSFS_readBytes(f, out.data(), len) != static_cast<PHYSFS_sint64>(len)) {
30 PHYSFS_close(f);
31 out.clear();
32 return false;
33 }
34 PHYSFS_close(f);
35 return true;
36}

51// `load.nut` loads the game with `dofile("main.nut")`. A raw `sq_compilebuffer`
52// compiles `import` edges but never instantiates them, so the import opcode
53// fails with "module was not instantiated". Give the chunk a `game:/` URI and
54// run the same ScriptCompiler + instantiate path EditorHost uses.
55std::string scriptUriFromPath(const char* path) {
56 std::string uri = path ? path : "";
57 for (char& ch : uri) {
58 if (ch == '\\') ch = '/';
59 }
60 const size_t scheme = uri.find(":/");
61 const bool drive = uri.size() >= 2 && uri[1] == ':';
62 if (scheme != std::string::npos && !drive) return uri;
63 while (uri.rfind("./", 0) == 0) uri.erase(0, 2);
64 while (!uri.empty() && uri.front() == '/') uri.erase(0, 1);
65 return "game:/" + uri;
66}

69SQInteger compileAndMaybeRun(HSQUIRRELVM vm, bool run) {
70 const SQChar* path = nullptr;
71 if (SQ_FAILED(sq_getstring(vm, 2, &path)) || !path)
72 return sq_throwerror(vm, "expected a script filename");
73
74 std::string content;
75 if (!readFileContent(path, content))
76 return sq_throwerror(vm, "cannot open script");
77
78 // Squirrel's sq_readscripfile skips a UTF-8 BOM; sq_compilebuffer does not.
79 if (content.size() >= 3 && (unsigned char)content[0] == 0xEF &&
80 (unsigned char)content[1] == 0xBB && (unsigned char)content[2] == 0xBF)
81 content.erase(0, 3);
82
83 const std::string uri = scriptUriFromPath(path);
84 if (SQ_FAILED(eve::script::ScriptCompiler::compileBuffer(vm, content.c_str(),
85 static_cast<SQInteger>(content.size()),
86 uri.c_str(), SQTrue)))
87 return SQ_ERROR;
88
89 Runtime* rt = Runtime::current();
90 if (!rt) rt = ModuleManager::runtime();
91 if (rt) {
92 try {
93 rt->scriptModules().instantiateDependencies(uri);
94 } catch (const std::exception& error) {
95 sq_pop(vm, 1);
96 return sq_throwerror(vm, error.what());
97 }
98 }
99
100 if (!run) return 1; // leave the closure on the stack
101
102 sq_pushroottable(vm);
103 if (SQ_FAILED(sq_call(vm, 1, SQFalse, SQTrue))) return SQ_ERROR;
104 sq_pop(vm, 1);
105 return 0;
106}

165SQInteger sq_file(HSQUIRRELVM vm) {
166 const SQChar* path = nullptr;
167 const SQChar* mode = nullptr;
168 if (SQ_FAILED(sq_getstring(vm, 2, &path)) || SQ_FAILED(sq_getstring(vm, 3, &mode)))
169 return sq_throwerror(vm, "file: expected (path, mode)");
170
171 std::string m(mode ? mode : "");
172 const bool writing = m.find('w') != std::string::npos || m.find('a') != std::string::npos;
173
174 std::string content;
175 if (!writing) {
176 // A missing file must fail here, otherwise `file_exists()` (which tries
177 // file(path, "r")) returns an empty handle and reports true.
178 if (!readFileContent(path, content))
179 return sq_throwerror(vm, "cannot open file");
180 }
181
182 sq_newtable(vm); // the handle table (becomes 'this')
183 // Squirrel 的 sq_newslot(-3) 需要 [table, key, value] 三层栈；
184 // 只推一个值会把目标错指到调用帧的 mode 参数上（非表对象会被静默跳过），
185 // 导致句柄永远没有 _path/_data，read()/getSize() 报 "file not readable"。
186 sq_pushstring(vm, "_path", -1); // key
187 sq_pushstring(vm, path, -1); // value
188 sq_newslot(vm, -3, SQFalse);
189 sq_pushstring(vm, "_data", -1); // key
190 sq_pushstring(vm, content.c_str(), static_cast<SQInteger>(content.size())); // value
191 sq_newslot(vm, -3, SQFalse);
192
193 struct { const SQChar* name; SQFUNCTION fn; } methods[] = {
194 {_SC("close"), sq_f_close},
195 {_SC("read"), sq_f_read},
196 {_SC("readString"), sq_f_read},
197 {_SC("getSize"), sq_f_getSize},
198 {_SC("isEOF"), sq_f_isEOF},
199 {_SC("tell"), sq_f_tell},
200 {_SC("seek"), sq_f_seek},
201 {_SC("getFilename"), sq_f_getFilename},
202 };
203 for (const auto& mth : methods) {
204 sq_pushstring(vm, mth.name, -1);
205 sq_newclosure(vm, mth.fn, 0);
206 sq_newslot(vm, -3, SQFalse);
207 }
208 return 1;
209}

214 HSQUIRRELVM h = vm.getHandle();

217 struct { const SQChar* name; SQFUNCTION fn; } globals[] = {
