282eve::Result<void> BodyBackend::setTargetTransform(float x, float y, float z, float qx, float qy, float qz, float qw,
283 float timeStep) {
284 if (!isValid()) return rejected<void>("Body expired", eve::DiagnosticCode::StaleHandle);
285 JPH::Quat q(qx, qy, qz, qw);
286 if (!isFinite(x, y, z) || !std::isfinite(timeStep) || timeStep <= 0 || !std::isfinite(q.LengthSq()) ||
287 q.LengthSq() < 1.e-12f)
288 return rejected<void>("Invalid target transform", eve::DiagnosticCode::InvalidArgument);
289 if (body->api().GetMotionType(body->id) != JPH::EMotionType::Kinematic)
290 return rejected<void>("Target transform requires a kinematic body", eve::DiagnosticCode::InvalidArgument);
291 body->api().MoveKinematic(body->id, {x, y, z}, q.Normalized(), timeStep);
292 return success();
293}

406eve::Result<void> BodyBackend::setMassProperties(float mass, float cx, float cy, float cz, float xx, float yy, float zz,
407 float xy, float xz, float yz) {
408 if (!isValid()) return rejected<void>("Body expired", eve::DiagnosticCode::StaleHandle);
409 if (body->api().GetMotionType(body->id) != JPH::EMotionType::Dynamic || !std::isfinite(mass) || mass <= 0 ||
410 !isFinite(cx, cy, cz) || !isFinite(xx, yy, zz) || !isFinite(xy, xz, yz))
411 return rejected<void>("Invalid mass properties", eve::DiagnosticCode::InvalidArgument);
412 JPH::MassProperties properties;
413 properties.mMass = mass;
414 properties.mInertia =
415 JPH::Mat44(JPH::Vec4(xx, xy, xz, 0), JPH::Vec4(xy, yy, yz, 0), JPH::Vec4(xz, yz, zz, 0), JPH::Vec4(0, 0, 0, 1));
416 JPH::Mat44 rotation;
417 JPH::Vec3 diagonal;
418 if (!properties.DecomposePrincipalMomentsOfInertia(rotation, diagonal) ||
419 !isFinite(diagonal.GetX(), diagonal.GetY(), diagonal.GetZ()) || diagonal.ReduceMin() <= 0)
420 return rejected<void>("Inertia must be positive definite", eve::DiagnosticCode::InvalidArgument);
421 auto original = body->api().GetShape(body->id);
422 const auto previous = original->GetCenterOfMass();
423 if (original->GetSubType() == JPH::EShapeSubType::OffsetCenterOfMass)
424 original = static_cast<const JPH::OffsetCenterOfMassShape*>(original.GetPtr())->GetInnerShape();
425 JPH::RefConst<JPH::Shape> adjusted =
426 new JPH::OffsetCenterOfMassShape(original, JPH::Vec3(cx, cy, cz) - original->GetCenterOfMass());
427 body->api().SetShape(body->id, adjusted, false, JPH::EActivation::Activate);
428 auto* native = body->world.lock()->system.GetBodyLockInterfaceNoLock().TryGetBody(body->id);
429 native->GetMotionPropertiesUnchecked()->SetMassProperties(body->dofs, properties);
430 body->mass = properties;
431 notifyJointShapeChanged(*body->world.lock(), body->id, adjusted->GetCenterOfMass() - previous);
432 return success();
433}
