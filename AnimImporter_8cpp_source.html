43std::string trim(const std::string &s) {
44 size_t a = 0;
45 while (a < s.size() && (s[a] == ' ' || s[a] == '\t' || s[a] == '\r')) ++a;
46 size_t b = s.size();
47 while (b > a && (s[b - 1] == ' ' || s[b - 1] == '\t' || s[b - 1] == '\r')) --b;
48 return s.substr(a, b - a);

53void AnimImporter::collectBones(const aiNode *node, int parent, AnimSkeleton *skeleton,
54 std::vector<const aiNode *> &order) {
55 if (!node) return;
56 const std::string name = node->mName.C_Str();
57 const std::string boneName = name.empty() ? ("node_" + std::to_string(order.size())) : name;
58 const int id = skeleton->addBone(boneName, parent);
59 order.push_back(node);
60
61 float px, py, pz, qx, qy, qz, qw, sx, sy, sz;
62 decomposeBind(node->mTransformation, px, py, pz, qx, qy, qz, qw, sx, sy, sz);
63 skeleton->setBindPosition(id, px, py, pz);
64 skeleton->setBindRotation(id, qx, qy, qz, qw);
65 skeleton->setBindScale(id, sx, sy, sz);
66
67 for (unsigned i = 0; i < node->mNumChildren; ++i) {
68 collectBones(node->mChildren[i], id, skeleton, order);
69 }
70}

86AnimClip *AnimImporter::loadClip(const aiScene *scene, const AnimSkeleton *skeleton,
87 int animIndex) {
88 if (!scene || !skeleton) {
89 throw Exception("AnimImporter.loadClip: null scene/skeleton");
90 }
91 if (animIndex < 0 || animIndex >= getAnimationCount(scene)) {
92 throw Exception("AnimImporter.loadClip: invalid anim index %d", animIndex);
93 }
94 const aiAnimation *anim = scene->mAnimations[animIndex];
95 auto *clip = new AnimClip(anim->mName.length ? anim->mName.C_Str() : "anim");
96 clip->setDuration(ticksToSeconds(anim, anim->mDuration));
97 clip->setLoop(true);
98 clip->setSampleRate(30.f);
99
100 for (unsigned c = 0; c < anim->mNumChannels; ++c) {
101 const aiNodeAnim *channel = anim->mChannels[c];
102 if (!channel) continue;
103 const int bone = skeleton->findBone(channel->mNodeName.C_Str());
104 if (bone < 0) continue;
105
106 for (unsigned i = 0; i < channel->mNumPositionKeys; ++i) {
107 const aiVectorKey &k = channel->mPositionKeys[i];
108 clip->addPositionKey(bone, ticksToSeconds(anim, k.mTime), k.mValue.x, k.mValue.y,
109 k.mValue.z);
110 }
111 for (unsigned i = 0; i < channel->mNumRotationKeys; ++i) {
112 const aiQuatKey &k = channel->mRotationKeys[i];
113 clip->addRotationKey(bone, ticksToSeconds(anim, k.mTime), k.mValue.x, k.mValue.y,
114 k.mValue.z, k.mValue.w);
115 }
116 for (unsigned i = 0; i < channel->mNumScalingKeys; ++i) {
117 const aiVectorKey &k = channel->mScalingKeys[i];
118 clip->addScaleKey(bone, ticksToSeconds(anim, k.mTime), k.mValue.x, k.mValue.y,
119 k.mValue.z);
120 }
121 }
122 return clip;
123}

181void AnimImporter::importAnimationFixtureText(const std::string &text, AnimSkeleton **skeletonOut,
182 AnimClip **clipOut) {
183 if (!skeletonOut || !clipOut) throw Exception("AnimImporter.importAnimationFixtureText: null out");
184 *skeletonOut = nullptr;
185 *clipOut = nullptr;
186
187 auto *skeleton = new AnimSkeleton();
188 AnimClip *clip = nullptr;
189 int curBone = -1;
190
191 std::istringstream in(text);
192 std::string line;
193 try {
194 while (std::getline(in, line)) {
195 line = trim(line);
196 if (line.empty() || line[0] == '#') continue;
197 std::istringstream ls(line);
198 std::string tag;
199 ls >> tag;
200 if (tag == "EVA") {
201 int ver = 0;
202 ls >> ver;
203 if (ver != 1) throw Exception("AnimImporter.importAnimationFixtureText: unsupported version");
204 } else if (tag == "skeleton") {
205 // informational
206 } else if (tag == "bone") {
207 int idx = -1, parent = -1;
208 std::string name;
209 ls >> idx >> parent;
210 std::getline(ls, name);
211 name = trim(name);
212 if (name.empty()) throw Exception("AnimImporter.importAnimationFixtureText: empty bone name");
213 const int id = skeleton->addBone(name, parent);
214 if (id != idx) {
215 throw Exception("AnimImporter.importAnimationFixtureText: bone index mismatch");
216 }
217 } else if (tag == "bind") {
218 int idx = -1;
219 float px, py, pz, qx, qy, qz, qw, sx, sy, sz;
220 ls >> idx >> px >> py >> pz >> qx >> qy >> qz >> qw >> sx >> sy >> sz;
221 skeleton->setBindPosition(idx, px, py, pz);
222 skeleton->setBindRotation(idx, qx, qy, qz, qw);
223 skeleton->setBindScale(idx, sx, sy, sz);
224 } else if (tag == "clip") {
225 std::string name;
226 std::getline(ls, name);
227 name = trim(name);
228 clip = new AnimClip(name.empty() ? "clip" : name);
229 } else if (tag == "duration") {
230 if (!clip) throw Exception("AnimImporter.importAnimationFixtureText: duration before clip");
231 float d = 0.f;
232 ls >> d;
233 clip->setDuration(d);
234 } else if (tag == "loop") {
235 if (!clip) throw Exception("AnimImporter.importAnimationFixtureText: loop before clip");
236 int v = 1;
237 ls >> v;
238 clip->setLoop(v != 0);
239 } else if (tag == "rate") {
240 if (!clip) throw Exception("AnimImporter.importAnimationFixtureText: rate before clip");
241 float r = 30.f;
242 ls >> r;
243 clip->setSampleRate(r);
244 } else if (tag == "sync") {
245 if (!clip) throw Exception("AnimImporter.importAnimationFixtureText: sync marker before clip");
246 float t = 0.f;
247 std::string name;
248 ls >> t;
249 std::getline(ls, name);
250 name = trim(name);
251 clip->addSyncMarker(t, name);
252 } else if (tag == "track") {
253 if (!clip) throw Exception("AnimImporter.importAnimationFixtureText: track before clip");
254 int np = 0, nr = 0, ns = 0;
255 ls >> curBone >> np >> nr >> ns;
256 (void)np;
257 (void)nr;
258 (void)ns;
259 } else if (tag == "p") {
260 float t, x, y, z;
261 ls >> t >> x >> y >> z;
262 clip->addPositionKey(curBone, t, x, y, z);
263 } else if (tag == "r") {
264 float t, x, y, z, w;
265 ls >> t >> x >> y >> z >> w;
266 clip->addRotationKey(curBone, t, x, y, z, w);
267 } else if (tag == "s") {
268 float t, x, y, z;
269 ls >> t >> x >> y >> z;
270 clip->addScaleKey(curBone, t, x, y, z);
271 } else if (tag == "end") {
272 break;
273 }
274 }
275 if (!clip) throw Exception("AnimImporter.importAnimationFixtureText: missing clip");
276 *skeletonOut = skeleton;
277 *clipOut = clip;
278 } catch (...) {
279 delete skeleton;
280 delete clip;
281 throw;
282 }
283}
