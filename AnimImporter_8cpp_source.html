14std::string trim(const std::string &s) {
15 size_t a = 0;
16 while (a < s.size() && (s[a] == ' ' || s[a] == '\t' || s[a] == '\r')) ++a;
17 size_t b = s.size();
18 while (b > a && (s[b - 1] == ' ' || s[b - 1] == '\t' || s[b - 1] == '\r')) --b;
19 return s.substr(a, b - a);
20}

69void AnimImporter::importAnimationFixtureText(const std::string &text, AnimSkeleton **skeletonOut,
70 AnimClip **clipOut) {
71 if (!skeletonOut || !clipOut) throw Exception("AnimImporter.importAnimationFixtureText: null out");
72 *skeletonOut = nullptr;
73 *clipOut = nullptr;
74
75 auto *skeleton = new AnimSkeleton();
76 AnimClip *clip = nullptr;
77 int curBone = -1;
78
79 std::istringstream in(text);
80 std::string line;
81 try {
82 while (std::getline(in, line)) {
83 line = trim(line);
84 if (line.empty() || line[0] == '#') continue;
85 std::istringstream ls(line);
86 std::string tag;
87 ls >> tag;
88 if (tag == "EVA") {
89 int ver = 0;
90 ls >> ver;
91 if (ver != 1) throw Exception("AnimImporter.importAnimationFixtureText: unsupported version");
92 } else if (tag == "skeleton") {
93 // informational
94 } else if (tag == "bone") {
95 int idx = -1, parent = -1;
96 std::string name;
97 ls >> idx >> parent;
98 std::getline(ls, name);
99 name = trim(name);
100 if (name.empty()) throw Exception("AnimImporter.importAnimationFixtureText: empty bone name");
101 const int id = skeleton->addBone(name, parent);
102 if (id != idx) {
103 throw Exception("AnimImporter.importAnimationFixtureText: bone index mismatch");
104 }
105 } else if (tag == "bind") {
106 int idx = -1;
107 float px, py, pz, qx, qy, qz, qw, sx, sy, sz;
108 ls >> idx >> px >> py >> pz >> qx >> qy >> qz >> qw >> sx >> sy >> sz;
109 skeleton->setBindPosition(idx, px, py, pz);
110 skeleton->setBindRotation(idx, qx, qy, qz, qw);
111 skeleton->setBindScale(idx, sx, sy, sz);
112 } else if (tag == "clip") {
113 std::string name;
114 std::getline(ls, name);
115 name = trim(name);
116 clip = new AnimClip(name.empty() ? "clip" : name);
117 } else if (tag == "duration") {
118 if (!clip) throw Exception("AnimImporter.importAnimationFixtureText: duration before clip");
119 float d = 0.f;
120 ls >> d;
121 clip->setDuration(d);
122 } else if (tag == "loop") {
123 if (!clip) throw Exception("AnimImporter.importAnimationFixtureText: loop before clip");
124 int v = 1;
125 ls >> v;
126 clip->setLoop(v != 0);
127 } else if (tag == "rate") {
128 if (!clip) throw Exception("AnimImporter.importAnimationFixtureText: rate before clip");
129 float r = 30.f;
130 ls >> r;
131 clip->setSampleRate(r);
132 } else if (tag == "sync") {
133 if (!clip) throw Exception("AnimImporter.importAnimationFixtureText: sync marker before clip");
134 float t = 0.f;
135 std::string name;
136 ls >> t;
137 std::getline(ls, name);
138 name = trim(name);
139 clip->addSyncMarker(t, name);
140 } else if (tag == "track") {
141 if (!clip) throw Exception("AnimImporter.importAnimationFixtureText: track before clip");
142 int np = 0, nr = 0, ns = 0;
143 ls >> curBone >> np >> nr >> ns;
144 (void)np;
145 (void)nr;
146 (void)ns;
147 } else if (tag == "p") {
148 float t, x, y, z;
149 ls >> t >> x >> y >> z;
150 clip->addPositionKey(curBone, t, x, y, z);
151 } else if (tag == "r") {
152 float t, x, y, z, w;
153 ls >> t >> x >> y >> z >> w;
154 clip->addRotationKey(curBone, t, x, y, z, w);
155 } else if (tag == "s") {
156 float t, x, y, z;
157 ls >> t >> x >> y >> z;
158 clip->addScaleKey(curBone, t, x, y, z);
159 } else if (tag == "end") {
160 break;
161 }
162 }
163 if (!clip) throw Exception("AnimImporter.importAnimationFixtureText: missing clip");
164 *skeletonOut = skeleton;
165 *clipOut = clip;
166 } catch (...) {
167 delete skeleton;
168 delete clip;
169 throw;
170 }
171}
