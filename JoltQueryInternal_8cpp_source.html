9eve::Result<QueryVolume> capsuleVolume(float ax, float ay, float az, float bx, float by, float bz, float radius) {
10 if (!isFinite(ax, ay, az) || !isFinite(bx, by, bz) || !std::isfinite(radius) || radius <= 0)
11 return rejected<QueryVolume>("Invalid capsule", eve::DiagnosticCode::InvalidArgument);
12 const JPH::Vec3 axis(bx - ax, by - ay, bz - az);
13 const float length = axis.Length();
14 if (!std::isfinite(length)) return rejected<QueryVolume>("Capsule overflow", eve::DiagnosticCode::InvalidArgument);
15 QueryVolume result;
16 result.shape = length < 1.e-6f ? JPH::RefConst<JPH::Shape>(new JPH::SphereShape(radius))
17 : JPH::RefConst<JPH::Shape>(new JPH::CapsuleShape(length * .5f, radius));
18 auto rotation = length < 1.e-6f ? JPH::Quat::sIdentity() : JPH::Quat::sFromTo(JPH::Vec3::sAxisY(), axis / length);
19 result.transform =
20 JPH::RMat44::sRotationTranslation(rotation, (JPH::RVec3(ax, ay, az) + JPH::RVec3(bx, by, bz)) * .5);
21 return eve::Result<QueryVolume>::success(std::move(result));
22}
23bool isAcceptedHit(const State& state, JPH::BodyID id, const JPH::SubShapeID& sub, QueryFilter3D filter) {
24 const auto* native = state.system.GetBodyLockInterfaceNoLock().TryGetBody(id);
25 if (!native) return false;
26 const auto* shape = child(*native, sub);
27 if (!shape || !shape->owner || shape->sensor) return false;
28 auto body = shape->body.lock();
29 return body && body->owner && body->owner->getId() != filter.ignoredBodyId &&
30 shape->owner->getId() != filter.ignoredShapeId && (shape->filter.categoryBits & filter.maskBits) &&
31 (shape->filter.maskBits & filter.categoryBits);
32}
33RayHit3D queryHit(const State& state, JPH::BodyID id, const JPH::SubShapeID& sub, JPH::RVec3Arg point,
34 JPH::Vec3Arg normal, float fraction) {
35 const auto& native = *state.system.GetBodyLockInterfaceNoLock().TryGetBody(id);
36 const auto& shape = *child(native, sub);
37 const auto& body = *shape.body.lock()->owner;
38 RayHit3D result;
39 result.hit = true;
40 result.body = body.runtimeHandle();
41 result.shape = shape.owner->runtimeHandle();
42 result.bodyId = body.getId();
43 result.shapeId = shape.owner->getId();
44 result.shapeTag = shape.owner->getTag();
45 result.materialId = static_cast<int>(contactMaterial(native, sub, shape).userMaterialId & 0xffffffffu);
46 result.triangleIndex = triangleIndex(native, sub);
47 result.x = float(point.GetX());
48 result.y = float(point.GetY());
49 result.z = float(point.GetZ());
50 result.normalX = normal.GetX();
51 result.normalY = normal.GetY();
52 result.normalZ = normal.GetZ();
53 result.fraction = fraction;
54 return result;
55}
