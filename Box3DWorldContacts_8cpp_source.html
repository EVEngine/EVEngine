22eve::Result<BodyContacts3D> WorldBackend::do_queryBodyContactsOwned(PhysicsBodyHandle body, int maxPoints) const {
23 if (maxPoints < 1 || maxPoints > 4096)
24 return eve::Result<BodyContacts3D>::failure(eve::Diagnostic::error(
25 eve::DiagnosticCode::InvalidArgument, "maxPoints must be in [1,4096]", "physics.contacts.maxPoints"));
26 const auto *target = owner_->findBody(body);
27 if (!target)
28 return eve::Result<BodyContacts3D>::failure(eve::Diagnostic::error(
29 eve::DiagnosticCode::StaleHandle, "Body does not belong to this live world", "physics.contacts.body"));
30 BodyContacts3D result;
31 result.body = body;
32 auto &points = result.points;
33 constexpr int maxContactPairs = 4096;
34 const int availablePairs = b3Body_GetContactCapacity(box3d::Access::bodyId(*target));
35 const int capacity = std::min(availablePairs, maxContactPairs + 1);
36 if (capacity <= 0) return eve::Result<BodyContacts3D>::success(std::move(result));
37 std::vector<b3ContactData> contacts(static_cast<size_t>(capacity));
38 const int contactCount = b3Body_GetContactData(box3d::Access::bodyId(*target), contacts.data(), capacity);
39 if (contactCount > maxContactPairs) result.coverage = ContactQueryCoverage3D::Truncated;
40 points.reserve(static_cast<size_t>(std::min(maxPoints, contactCount * 2)));
41
42 for (int contactIndex = 0; contactIndex < std::min(contactCount, maxContactPairs); ++contactIndex) {
43 const b3ContactData &contact = contacts[static_cast<size_t>(contactIndex)];
44 Shape3D *shapeA = shapeFromShape(contact.shapeIdA);
45 Shape3D *shapeB = shapeFromShape(contact.shapeIdB);
46 Body3D *bodyA = bodyFromShape(contact.shapeIdA);
47 Body3D *bodyB = bodyFromShape(contact.shapeIdB);
48 if (!shapeA || !shapeB || !bodyA || !bodyB) {
49 result.coverage = ContactQueryCoverage3D::Truncated;
50 continue;
51 }
52 const bool targetIsA = bodyA == target;
53 if (!targetIsA && bodyB != target) continue;
54 Shape3D *ownShape = targetIsA ? shapeA : shapeB;
55 Shape3D *otherShape = targetIsA ? shapeB : shapeA;
56 Body3D *otherBody = targetIsA ? bodyB : bodyA;
57 const b3Pos centerOfMass = b3Body_GetWorldCenterOfMass(box3d::Access::bodyId(*target));
58
59 for (int manifoldIndex = 0; manifoldIndex < contact.manifoldCount; ++manifoldIndex) {
60 const b3Manifold &manifold = contact.manifolds[manifoldIndex];
61 const b3Vec3 normal = targetIsA ? manifold.normal : -manifold.normal;
62 for (int pointIndex = 0; pointIndex < manifold.pointCount; ++pointIndex) {
63 if (static_cast<int>(points.size()) >= maxContactPairs) {
64 result.coverage = ContactQueryCoverage3D::Truncated;
65 break;
66 }
67 const b3ManifoldPoint &point = manifold.points[pointIndex];
68 const b3Pos worldPoint = b3OffsetPos(centerOfMass, targetIsA ? point.anchorA : point.anchorB);
69 points.push_back({ownShape->getId(), ownShape->getTag(), otherBody->getId(), otherShape->getId(),
70 otherShape->getTag(), point.featureId, static_cast<float>(worldPoint.x),
71 static_cast<float>(worldPoint.y), static_cast<float>(worldPoint.z), normal.x,
72 normal.y, normal.z, point.separation, point.normalImpulse, point.totalNormalImpulse,
73 point.normalVelocity, point.persisted});
74 }
75 }
76 }
77
78 std::sort(points.begin(), points.end(), [](const ContactPointResult &a, const ContactPointResult &b) {
79 if (a.shapeId != b.shapeId) return a.shapeId < b.shapeId;
80 if (a.otherShapeId != b.otherShapeId) return a.otherShapeId < b.otherShapeId;
81 if (a.featureId != b.featureId) return a.featureId < b.featureId;
82 if (a.x != b.x) return a.x < b.x;
83 if (a.y != b.y) return a.y < b.y;
84 return a.z < b.z;
85 });
86 if (static_cast<int>(points.size()) > maxPoints) {
87 points.resize(static_cast<size_t>(maxPoints));
88 result.coverage = ContactQueryCoverage3D::Truncated;
89 }
90 return eve::Result<BodyContacts3D>::success(std::move(result));
91}

97eve::Result<void> WorldBackend::collectEvents() {
98 if (!isValid())
99 return eve::Result<void>::failure(
100 eve::Diagnostic::error(eve::DiagnosticCode::StaleHandle, "World is no longer valid"));
101
102
103 b3ContactEvents contacts = b3World_GetContactEvents(worldId_);
104 for (int i = 0; i < contacts.beginCount; ++i) {
105 EventShape a = do_eventShapeFrom(contacts.beginEvents[i].shapeIdA);
106 EventShape b = do_eventShapeFrom(contacts.beginEvents[i].shapeIdB);
107 if (a.shapeId == 0 || b.shapeId == 0) continue;
108 if (a.shapeId > b.shapeId) std::swap(a, b);
109 owner_->beginContacts_.push_back({a, b});
110 }
111 for (int i = 0; i < contacts.endCount; ++i) {
112 EventShape a = do_eventShapeFrom(contacts.endEvents[i].shapeIdA);
113 EventShape b = do_eventShapeFrom(contacts.endEvents[i].shapeIdB);
114 if (a.shapeId == 0 || b.shapeId == 0) continue;
115 if (a.shapeId > b.shapeId) std::swap(a, b);
116 owner_->endContacts_.push_back({a, b});
117 }
118
119 for (int i = 0; i < contacts.hitCount; ++i) {
120 const b3ContactHitEvent &source = contacts.hitEvents[i];
121 EventShape a = do_eventShapeFrom(source.shapeIdA);
122 EventShape b = do_eventShapeFrom(source.shapeIdB);
123 if (a.shapeId == 0 || b.shapeId == 0) continue;
124
125 float normalImpulse = 0.f;
126 if (b3Contact_IsValid(source.contactId)) {
127 const b3ContactData data = b3Contact_GetData(source.contactId);
128 for (int manifoldIndex = 0; manifoldIndex < data.manifoldCount; ++manifoldIndex) {
129 const b3Manifold &manifold = data.manifolds[manifoldIndex];
130 for (int pointIndex = 0; pointIndex < manifold.pointCount; ++pointIndex)
131 normalImpulse += manifold.points[pointIndex].totalNormalImpulse;
132 }
133 }
134
135 b3Vec3 normal = source.normal;
136 if (a.shapeId > b.shapeId) {
137 std::swap(a, b);
138 normal = -normal;
139 }
140 owner_->hits_.push_back({a, b, static_cast<float>(source.point.x), static_cast<float>(source.point.y),
141 static_cast<float>(source.point.z), normal.x, normal.y, normal.z, source.approachSpeed,
142 normalImpulse});
143 }
144
145 b3SensorEvents sensors = b3World_GetSensorEvents(worldId_);
146 for (int i = 0; i < sensors.beginCount; ++i) {
147 EventShape sensor = do_eventShapeFrom(sensors.beginEvents[i].sensorShapeId);
148 EventShape visitor = do_eventShapeFrom(sensors.beginEvents[i].visitorShapeId);
149 if (sensor.shapeId == 0 || visitor.shapeId == 0) continue;
150 owner_->beginTriggers_.push_back({sensor, visitor});
151 }
152 for (int i = 0; i < sensors.endCount; ++i) {
153 EventShape sensor = do_eventShapeFrom(sensors.endEvents[i].sensorShapeId);
154 EventShape visitor = do_eventShapeFrom(sensors.endEvents[i].visitorShapeId);
155 if (sensor.shapeId == 0 || visitor.shapeId == 0) continue;
156 owner_->endTriggers_.push_back({sensor, visitor});
157 }
158
159 const b3JointEvents jointEvents = b3World_GetJointEvents(worldId_);
160 for (int i = 0; i < jointEvents.count; ++i) {
161 const b3JointEvent &source = jointEvents.jointEvents[i];
162 auto *joint = static_cast<Joint3D *>(source.userData);
163 if (!joint || owner_->joints_.find(joint) == owner_->joints_.end() || !joint->isValid() ||
164 !b3Joint_IsValid(source.jointId) || b3Joint_GetUserData(source.jointId) != joint)
165 continue;
166 const b3Vec3 force = b3Joint_GetConstraintForce(source.jointId);
167 const b3Vec3 torque = b3Joint_GetConstraintTorque(source.jointId);
168 owner_->jointStressEvents_.push_back({joint->getId(), joint->getBodyAId(), joint->getBodyBId(),
169 static_cast<int>(joint->kind_), force.x, force.y, force.z, torque.x,
170 torque.y, torque.z});
171 }
172
173 // Invalid handles are retained only until the step that can report their end event.
174 // This keeps destruction-safe metadata without allowing the history map to grow forever.
175 for (auto it = owner_->shapeRecords_.begin(); it != owner_->shapeRecords_.end();) {
176 if (!b3Shape_IsValid(b3LoadShapeId(it->first)))
177 it = owner_->shapeRecords_.erase(it);
178 else
179 ++it;
180 }
181 for (auto it = owner_->shapeRawHandles_.begin(); it != owner_->shapeRawHandles_.end();) {
182 if (!b3Shape_IsValid(b3LoadShapeId(it->first)))
183 it = owner_->shapeRawHandles_.erase(it);
184 else
185 ++it;
186 }
187 for (auto it = owner_->shapeHandles_.begin(); it != owner_->shapeHandles_.end();) {
188 if (!it->second || !it->second->isValid())
189 it = owner_->shapeHandles_.erase(it);
190 else
191 ++it;
192 }
193
194 return eve::Result<void>::success();
195}
