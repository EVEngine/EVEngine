28Result<std::unique_ptr<VegetationAsset>> VegetationAsset::fromCanonical(const asset::CanonicalMeshData& mesh) {
29 using Output = std::unique_ptr<VegetationAsset>;
30 const auto count = mesh.positions.size() / 3;
31 if (!count || count > 4'000'000 || mesh.positions.size() % 3 || mesh.normals.size() != count * 3 ||
32 mesh.indices.empty() || mesh.indices.size() > 12'000'000 || mesh.indices.size() % 3)
33 return failure<Output>(DiagnosticCode::InvalidArgument, "invalid vegetation mesh geometry");
34 auto stream = [&](const char* name) -> const asset::CanonicalMeshAttribute* {
35 auto found = mesh.attributes.find(name);
36 return found != mesh.attributes.end() && found->second.components == 4 &&
37 found->second.values.size() == count * 4
38 ? &found->second
39 : nullptr;
40 };
41 const auto* color = stream("COLOR_0");
42 const auto* uv0 = stream("_UNITY_UV0");
43 const auto* uv1 = stream("_UNITY_UV1");
44 const auto* uv3 = stream("_UNITY_UV3");
45 const auto* tangent = stream("TANGENT");
46 if (mesh.attributes.contains("TANGENT") && !tangent)
47 return failure<Output>(DiagnosticCode::InvalidArgument, "malformed vegetation tangent stream");
48 if (!color || !uv0 || !uv1 || !uv3)
49 return failure<Output>(DiagnosticCode::Unsupported, "TVE requires float4 color and complete Unity UV0/UV1/UV4");
50 const auto renderUv = mesh.texcoords.find(0);
51 if (renderUv == mesh.texcoords.end() || renderUv->second.size() != count * 2)
52 return failure<Output>(DiagnosticCode::InvalidArgument, "vegetation render UV0 is absent or malformed");
53 for (auto f : renderUv->second)
54 if (!std::isfinite(f))
55 return failure<Output>(DiagnosticCode::InvalidArgument, "nonfinite vegetation render UV");
56 for (auto index : mesh.indices)
57 if (index >= count) return failure<Output>(DiagnosticCode::InvalidArgument, "vegetation index out of range");
58 try {
59 std::vector<graphics::TvePackedVertex> packed(count);
60 auto four = [](const auto& a, size_t at) { return glm::vec4(a[at], a[at + 1], a[at + 2], a[at + 3]); };
61 for (size_t v = 0; v < count; ++v) {
62 auto& p = packed[v];
63 p.position = {mesh.positions[v * 3], mesh.positions[v * 3 + 1], mesh.positions[v * 3 + 2]};
64 p.normal = {mesh.normals[v * 3], mesh.normals[v * 3 + 1], mesh.normals[v * 3 + 2]};
65 const auto length = glm::length(p.normal);
66 if (!std::isfinite(length) || length < 1e-6f)
67 return failure<Output>(DiagnosticCode::InvalidArgument, "invalid vegetation normal");
68 p.color = four(color->values, v * 4);
69 p.texcoord0 = four(uv0->values, v * 4);
70 p.texcoord1 = four(uv1->values, v * 4);
71 p.texcoord3 = four(uv3->values, v * 4);
72 }
73 auto decoded = graphics::decodeTveVegetationVertices(packed);
74 if (!decoded) return Result<Output>::failure(decoded.status());
75 auto impl = std::make_unique<Impl>();
76 impl->vertices = std::move(decoded).takeValue();
77 for (size_t v = 0; v < count; ++v) {
78 auto& vertex = impl->vertices[v];
79 vertex.pivot.z = -vertex.pivot.z;
80 vertex.texcoord = {renderUv->second[v * 2], renderUv->second[v * 2 + 1]};
81 vertex.secondaryTexcoord.y = 1.f - vertex.secondaryTexcoord.y;
82 if (tangent) {
83 vertex.tangent = four(tangent->values, v * 4);
84 const auto t = glm::vec3(*vertex.tangent);
85 if (!std::isfinite(t.x) || !std::isfinite(t.y) || !std::isfinite(t.z) ||
86 (vertex.tangent->w != 1.f && vertex.tangent->w != -1.f) ||
87 glm::length(glm::cross(vertex.normal, t)) < 1e-6f)
88 return failure<Output>(DiagnosticCode::InvalidArgument, "invalid vegetation tangent frame");
89 }
90 }
91 impl->uv = renderUv->second;
92 impl->indices = mesh.indices;
93 return Result<Output>::success(Output(new VegetationAsset(std::move(impl))));
94 } catch (const std::bad_alloc&) {
95 return failure<Output>(DiagnosticCode::Failed, "vegetation asset allocation failed");
96 }

99 const AssetRef& ref,
100 const asset::EvpackCapabilities& capabilities,
101 const asset::CanonicalMeshLimits& limits) {
102 using Output = std::unique_ptr<VegetationAsset>;
103 auto payload = reader.read(ref, "eve.mesh/3", capabilities, limits.maximumDecodedBytes);
104 if (!payload) return Result<Output>::failure(payload.status());
105 const asset::RuntimeAssetChunk* bulk = nullptr;
106 for (const auto& chunk : payload.value().chunks) {
107 if (chunk.kind != asset::EvpackChunkKind::Bulk) continue;
108 if (bulk) return failure<Output>(DiagnosticCode::ParseError, "vegetation mesh has multiple bulk chunks");
109 bulk = &chunk;
110 }
111 if (!bulk) return failure<Output>(DiagnosticCode::NotFound, "vegetation mesh bulk is missing");
112 auto decoded = asset::decodeCanonicalMesh(bulk->bytes, limits);
113 if (!decoded) return Result<Output>::failure(decoded.status());
114 return fromCanonical(decoded.value());

150Result<graphics::Mesh*> VegetationAsset::createGpuFieldMesh(graphics::Graphics& graphics) const {
151 try {
152 std::vector<float> positions, normals, tangents, bitangents, shading, deformation;
153 positions.reserve(impl_->vertices.size() * 3);
154 normals.reserve(impl_->vertices.size() * 3);
155 shading.reserve(impl_->vertices.size() * 5);
156 deformation.reserve(impl_->vertices.size() * 9);
157 const bool authored = !impl_->vertices.empty() && impl_->vertices.front().tangent.has_value();
158 if (authored) {
159 tangents.reserve(impl_->vertices.size() * 3);
160 bitangents.reserve(impl_->vertices.size() * 3);
161 }
162 for (const auto& vertex : impl_->vertices) {
163 positions.insert(positions.end(), {vertex.position.x, vertex.position.y, vertex.position.z});
164 normals.insert(normals.end(), {vertex.normal.x, vertex.normal.y, vertex.normal.z});
165 shading.insert(shading.end(), {vertex.variation, vertex.occlusion, vertex.detail, vertex.detailCoord.x,
166 vertex.detailCoord.y});
167 deformation.insert(deformation.end(),
168 {vertex.pivot.x, vertex.pivot.y, vertex.pivot.z, vertex.bending, vertex.branch,
169 vertex.flutter, vertex.variation, vertex.boundsHeight, vertex.boundsRadius});
170 if (authored) {
171 const auto normal = glm::normalize(vertex.normal);
172 const auto tangent = glm::normalize(glm::vec3(*vertex.tangent) -
173 normal * glm::dot(normal, glm::vec3(*vertex.tangent)));
174 const auto bitangent = glm::cross(normal, tangent) * vertex.tangent->w;
175 tangents.insert(tangents.end(), {tangent.x, tangent.y, tangent.z});
176 bitangents.insert(bitangents.end(), {bitangent.x, bitangent.y, bitangent.z});
177 }
178 }
179 auto* mesh = graphics.newMeshFromArrays(positions.data(), normals.data(), impl_->uv.data(),
180 int(impl_->vertices.size()), impl_->indices.data(),
181 int(impl_->indices.size()));
182 if (!mesh) return failure<graphics::Mesh*>(DiagnosticCode::Failed, "vegetation rest mesh upload failed");
183 auto rollback = [&](Result<void> status) -> Result<graphics::Mesh*> {
184 if (!graphics.releaseMesh(mesh))
185 return failure<graphics::Mesh*>(DiagnosticCode::Failed, "vegetation rest mesh rollback failed");
186 return Result<graphics::Mesh*>::failure(status.status());
187 };
188 auto attached = mesh->adoptTangentFrame(std::move(tangents), std::move(bitangents));
189 if (!attached) return rollback(std::move(attached));
190 auto shadingAttached = mesh->adoptVegetationFactors(std::move(shading));
191 if (!shadingAttached) return rollback(std::move(shadingAttached));
192 auto deformationAttached = mesh->adoptVegetationDeformationFactors(std::move(deformation));
193 if (!deformationAttached) return rollback(std::move(deformationAttached));
194 return Result<graphics::Mesh*>::success(mesh);
195 } catch (const std::bad_alloc&) {
196 return failure<graphics::Mesh*>(DiagnosticCode::Failed, "vegetation rest mesh allocation failed");
197 }
198}
