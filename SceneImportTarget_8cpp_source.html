87EditorResult<DomainOperation> SceneImportTarget::makeSet(const SelectionSnapshot& s, const PropertyPath& p,
88 const EditorValue& v, PropertySetMode mode) const {
89 if (mode == PropertySetMode::Reset) return makeReset(s, p);
90 auto d = schema(s).find(p);
91 if (!matches(s) || !d || mode != PropertySetMode::Absolute || !validatePropertyValue(*d, v).ok())
92 return fail<DomainOperation>(EditorStatus::Rejected, "editor.scene-import.set",
93 "Scene import property edit is invalid");
94 EditorValue payload = contentValue();
95 auto& object = *payload.getIf<EditorValue::Object>();
96 object[p.value()] = v;
97 if (p == PropertyPath("preset")) {
98 const auto& name = *v.getIf<std::string>();
99 if (name != "custom") {
100 const bool raw = name == "raw", mobile = name == "mobile";
101 object["triangulate"] = true;
102 object["generateNormals"] = !raw;
103 object["joinVertices"] = !raw;
104 object["flipUvs"] = !raw;
105 object["improveCache"] = !raw;
106 object["sharedMeshes"] = !raw;
107 object["mipmaps"] = !raw;
108 object["importLights"] = !raw && !mobile;
109 object["importCameras"] = false;
110 object["importAnimations"] = !raw && !mobile;
111 }
112 } else
113 object["preset"] = "custom";
114 DomainOperation op;
115 op.type = "scene-import.replace.v1";
116 op.inverseType = op.type;
117 op.target = TargetId(id_);
118 op.payload = std::move(payload);
119 op.inverse = contentValue();
120 op.hasInverse = true;
121 op.affectedProperties.push_back(p.value());
122 op.mergeKey = "scene-import:" + id_ + ":" + p.value();
123 return eve::editing::applied<DomainOperation>(std::move(op));
124}
125EditorResult<DomainOperation> SceneImportTarget::makeReset(const SelectionSnapshot& s, const PropertyPath& p) const {
126 auto d = schema(s).find(p);
127 if (!d)
128 return fail<DomainOperation>(EditorStatus::Unsupported, "editor.scene-import.property",
129 "Unknown scene import property");

132std::vector<EditorDiagnostic> SceneImportTarget::validate() const {
133 std::vector<EditorDiagnostic> d;
134 if (value_.sourceAsset.empty())
135 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument,
136 RuleId("editor.scene-import.source"), DiagnosticSeverity::Error, "Scene import requires a source asset"));
137 const auto presets = std::vector<std::string>{"quality", "balanced", "mobile", "raw", "custom"};
138 if (std::find(presets.begin(), presets.end(), value_.preset) == presets.end())
139 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument,
140 RuleId("editor.scene-import.preset"), DiagnosticSeverity::Error, "Unknown scene import preset"));
141 if (!value_.triangulate)
142 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::Unsupported,
143 RuleId("editor.scene-import.non-triangles"), DiagnosticSeverity::Warning,
144 "Disabled triangulation may produce meshes unsupported by render backends"));
145 if (!value_.generateNormals)
146 d.push_back(eve::editing::ruleDiagnostic(eve::DiagnosticCode::PreconditionViolation,
147 RuleId("editor.scene-import.normals"), DiagnosticSeverity::Warning,
148 "Missing source normals will remain missing"));
149 return d;
150}
151EditorResult<void> SceneImportTarget::applyDomainOperation(const DomainOperation& op) {
152 if (op.target != TargetId(id_) || op.type != "scene-import.replace.v1" || !op.payload.isWithinLimits(3, 32, 4096))
153 return fail<void>(EditorStatus::Rejected, "editor.scene-import.operation", "Scene import operation is invalid");
154 SceneImportValue v;
155 const auto * s = field(op.payload, "source"), *p = field(op.payload, "preset");
156 auto* ss = s ? s->getIf<std::string>() : nullptr;
157 auto* ps = p ? p->getIf<std::string>() : nullptr;
158 if (!ss || !ps)
159 return fail<void>(EditorStatus::Rejected, "editor.scene-import.payload",
160 "Scene import source or preset is invalid");
161 v.sourceAsset = *ss;
162 v.preset = *ps;
163 bool* dst[] = {&v.triangulate, &v.generateNormals, &v.joinVertices, &v.flipUvs, &v.improveCache,
164 &v.sharedMeshes, &v.mipmaps, &v.importLights, &v.importCameras, &v.importAnimations};
165 const char* keys[] = {"triangulate", "generateNormals", "joinVertices", "flipUvs", "improveCache",
166 "sharedMeshes", "mipmaps", "importLights", "importCameras", "importAnimations"};
167 for (int i = 0; i < 10; ++i) {
168 auto* x = field(op.payload, keys[i]);
169 auto* b = x ? x->getIf<bool>() : nullptr;
170 if (!b)
171 return fail<void>(EditorStatus::Rejected, "editor.scene-import.boolean", "Scene import option is invalid");
172 *dst[i] = *b;
173 }
174 SceneImportTarget c(id_);
175 c.value_ = v;
176 if (errors(c.validate()))
177 return fail<void>(EditorStatus::Rejected, "editor.scene-import.invalid",
178 "Scene import settings validation failed");
179 value_ = std::move(v);
180 ++revision_;
181 dirty_.include(0, 0);
182 return eve::editing::applied<void>();
183}

197EditorResult<void> SceneImportTarget::loadSnapshot(const EditorValue& s) {
198 auto *v = field(s, "schemaVersion"), *c = field(s, "content");
199 auto* n = v ? v->getIf<int64_t>() : nullptr;
200 if (!n || *n != 1 || !c)
201 return fail<void>(EditorStatus::Unsupported, "editor.scene-import.snapshot",
202 "Unsupported scene import snapshot");
203 DomainOperation op;
204 op.target = TargetId(id_);
205 op.type = "scene-import.replace.v1";
206 op.payload = *c;

208 if (result.ok()) dirty_.clear();
209 return result;
