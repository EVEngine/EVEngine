33Result<std::string> stringParameter(const Value &parameters, std::string_view name, bool optional = false) {
34 const auto *object = parameters.getIf<Value::Object>();
35 if (!object)
36 return Result<std::string>::failure(Diagnostic::error(
37 DiagnosticCode::InvalidArgument, "RPG gameplay parameters must be an object", "parameters"));
38 const auto found = object->find(std::string(name));
39 if (found == object->end() && optional) return Result<std::string>::success({});
40 if (found == object->end() || !found->second.isString())
41 return Result<std::string>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
42 "RPG gameplay parameter must be a string",
43 "parameters." + std::string(name)));
44 return Result<std::string>::success(found->second.asString());
45}

59Result<void> BattleControl::bindParticipant(SubjectRef subject, RPGActor *actor) {
60 if (!instance_.isValid() || !subject.isValid() || !actor)
61 return Result<void>::failure(Diagnostic::error(
62 DiagnosticCode::InvalidArgument, "battle, subject, and actor identities must be valid", "participant"));
63 bool participant = false;
64 for (int index = 0; index < battle_.getActorCount(); ++index)
65 if (battle_.getActor(index) == actor) participant = true;
66 if (!participant)
67 return Result<void>::failure(
68 Diagnostic::error(DiagnosticCode::NotFound, "actor is not a participant in the adapted battle", "actor"));
69 const auto existing = actors_.find(subject);
70 if (existing != actors_.end() && existing->second != actor)
71 return Result<void>::failure(
72 Diagnostic::error(DiagnosticCode::Conflict, "subject is already bound to another participant", "subject"));
73 if (const auto reverse = subjectOf(actor); reverse.isValid() && reverse != subject)
74 return Result<void>::failure(
75 Diagnostic::error(DiagnosticCode::Conflict, "actor is already bound to another subject", "actor"));
76 actors_[subject] = actor;
77 ++revision_;
78 return Result<void>::success(Status::success(StatusCode::Applied));
79}

107Result<GameplayObservation> BattleControl::observeGameplay(const GameplaySession &session, SubjectRef instance) const {
108 if (instance != instance_ || !instance_.isValid())
109 return Result<GameplayObservation>::failure(
110 Diagnostic::error(DiagnosticCode::NotFound, "RPG battle gameplay instance was not found", "instance"));
111 bool ownsParticipant = false;
112 Value::Array participants;
113 for (int index = 0; index < battle_.getActorCount(); ++index) {
114 RPGActor *actor = battle_.getActor(index);
115 const auto subject = subjectOf(actor);
116 if (!subject.isValid()) continue;
117 const bool controlled = controls(session, subject);
118 ownsParticipant = ownsParticipant || controlled;
119 participants.emplace_back(Value::Object{
120 {"alive", Value(battle_.isActorAlive(actor))},
121 {"controlled", Value(controlled)},
122 {"hp", Value(actor->getCurrent("hp"))},
123 {"maxHp", Value(actor->getMax("hp"))},
124 {"side", Value(battle_.getSide(index))},
125 {"subject", Value(subject.format())},
126 });
127 }
128 if (session.access == GameplayAccess::PlayerEquivalent && !ownsParticipant)
129 return Result<GameplayObservation>::failure(
130 Diagnostic::error(DiagnosticCode::PreconditionViolation,
131 "session controls no participant in this RPG battle", "session.controlledSubjects"));
132 GameplayObservation observation;
133 observation.domain = gameplayId("gameplay:rpg-battle");
134 observation.instance = instance_;
135 observation.tick = tick_;
136 observation.revision = revision_;
137 observation.state = Value(Value::Object{
138 {"defeat", Value(battle_.isDefeat())},
139 {"finished", Value(battle_.isFinished())},
140 {"participants", Value(std::move(participants))},
141 {"turn", Value(battle_.getTurn())},
142 {"victory", Value(battle_.isVictory())},
143 });
144 return Result<GameplayObservation>::success(std::move(observation));
145}

171Result<GameplayCommandReceipt> BattleControl::submitGameplay(const GameplaySession &session, SubjectRef instance,
172 const GameplayCommand &command) {
173 if (instance != instance_)
174 return Result<GameplayCommandReceipt>::failure(
175 Diagnostic::error(DiagnosticCode::NotFound, "RPG battle gameplay instance was not found", "instance"));
176 RPGActor *actor = resolve(command.subject);
177 if (!actor)
178 return Result<GameplayCommandReceipt>::failure(
179 Diagnostic::error(DiagnosticCode::NotFound, "RPG battle participant was not found", "command.subject"));
180 if (!controls(session, command.subject))
181 return Result<GameplayCommandReceipt>::failure(Diagnostic::error(
182 DiagnosticCode::PreconditionViolation, "session does not control this RPG participant", "command.subject"));
183 if (command.id.empty())
184 return Result<GameplayCommandReceipt>::failure(
185 Diagnostic::error(DiagnosticCode::InvalidArgument, "command id must not be empty", "command.id"));
186 if (command.observedTick != tick_ || command.expectedRevision != revision_)
187 return Result<GameplayCommandReceipt>::failure(Diagnostic::error(
188 DiagnosticCode::Conflict, "RPG command was based on a stale observation", "command.expectedRevision"));
189 auto targetText = stringParameter(command.parameters, "target");
190 if (!targetText) return Result<GameplayCommandReceipt>::failure(targetText.status());
191 const auto targetId = PersistentId::parse(targetText.value());
192 RPGActor *target = targetId ? resolve(SubjectRef::fromPersistentId(*targetId)) : nullptr;
193 if (!target)
194 return Result<GameplayCommandReceipt>::failure(
195 Diagnostic::error(DiagnosticCode::NotFound, "RPG command target was not found", "parameters.target"));
196 std::string skillId;
197 if (command.action == gameplayId("rpg:skill")) {
198 auto skill = stringParameter(command.parameters, "skillId");
199 if (!skill) return Result<GameplayCommandReceipt>::failure(skill.status());
200 skillId = std::move(skill).takeValue();
201 } else if (command.action != gameplayId("rpg:attack")) {
202 return Result<GameplayCommandReceipt>::failure(
203 Diagnostic::error(DiagnosticCode::Unsupported, "unsupported RPG gameplay action", "command.action"));
204 }
205 auto queued = battle_.setActionChecked(actor, skillId, target);
206 if (!queued) return Result<GameplayCommandReceipt>::failure(queued.status());
207 ++revision_;
208 lastCommandByActor_[actor] = command.id;
209 GameplayEvent accepted;
210 accepted.sequence = nextEventSequence_++;
211 accepted.tick = tick_;
212 accepted.type = "rpg.command.accepted";
213 accepted.subject = command.subject;
214 accepted.causationCommandId = command.id;
215 accepted.correlationId = command.id;
216 accepted.payload = Value(Value::Object{{"action", Value(command.action.format())}});
217 events_.push_back(std::move(accepted));
218
219 GameplayCommandReceipt receipt;
220 receipt.commandId = command.id;
221 receipt.executionId = "rpg-round-" + std::to_string(battle_.getTurn() + 1);
222 receipt.acceptedTick = tick_;
223 receipt.resultingRevision = revision_;
224 receipt.details = Value(Value::Object{{"skillId", Value(skillId)}, {"target", Value(targetText.value())}});
225 return Result<GameplayCommandReceipt>::success(std::move(receipt), Status::success(StatusCode::Applied));
226}

228Result<GameplayObservation> BattleControl::advanceGameplay(const GameplaySession &session, SubjectRef instance,
229 const SimulationStep &step) {
230 if (instance != instance_)
231 return Result<GameplayObservation>::failure(
232 Diagnostic::error(DiagnosticCode::NotFound, "RPG battle gameplay instance was not found", "instance"));
233 if (step.tick <= tick_)
234 return Result<GameplayObservation>::failure(
235 Diagnostic::error(DiagnosticCode::Conflict, "RPG simulation tick must increase", "step.tick"));
236 auto authorized = observeGameplay(session, instance);
237 if (!authorized) return Result<GameplayObservation>::failure(authorized.status());
238 std::move(authorized).takeValue();
239 battle_.autoEnemyActions();
240 battle_.startRound();
241 while (battle_.executeNextAction() && !battle_.isFinished()) {
242 }
243 battle_.pollEvents();
244 tick_ = step.tick;
245 ++revision_;
246 for (int index = 0; index < battle_.getEventCount(); ++index) {
247 RPGActor *caster = battle_.getEventCaster(index);
248 GameplayEvent event;
249 event.sequence = nextEventSequence_++;
250 event.tick = tick_;
251 event.type = "rpg.battle." + battle_.getEventAction(index);
252 event.subject = subjectOf(caster ? caster : battle_.getEventTarget(index));
253 if (const auto found = lastCommandByActor_.find(caster); found != lastCommandByActor_.end()) {
254 event.causationCommandId = found->second;
255 event.correlationId = found->second;
256 }
257 event.payload = Value(Value::Object{
258 {"amount", Value(battle_.getEventAmount(index))},
259 {"critical", Value(battle_.getEventCrit(index))},
260 {"skillId", Value(battle_.getEventSkillId(index))},
261 {"target", Value(subjectOf(battle_.getEventTarget(index)).format())},
262 });
263 events_.push_back(std::move(event));
264 }
265 return observeGameplay(session, instance);
266}

323// ---- 极简表达式求值器：支持数字、a.<param>、b.<param>、+ - * / ( ) ----
324struct FormulaParser {
325 const std::string &s;
326 size_t pos = 0;
327 RPGActor *a;
328 RPGActor *b;
329 bool validateOnly;
330 std::optional<eve::Diagnostic> error;
331
332 FormulaParser(const std::string &str, RPGActor *atk, RPGActor *tgt, bool syntaxOnly)
333 : s(str), a(atk), b(tgt), validateOnly(syntaxOnly) {}
334
335 void skipSpace() {
336 while (pos < s.size() && std::isspace(static_cast<unsigned char>(s[pos]))) ++pos;
337 }
338
339 double parseExpression() {
340 double lhs = parseTerm();
341 while (true) {
342 skipSpace();
343 if (pos >= s.size()) break;
344 const char c = s[pos];
345 if (c == '+') {
346 ++pos;
347 lhs += parseTerm();
348 } else if (c == '-') {
349 ++pos;
350 lhs -= parseTerm();
351 } else {
352 break;
353 }
354 }
355 return lhs;
356 }
357
358 double parseTerm() {
359 double lhs = parseFactor();
360 while (true) {
361 skipSpace();
362 if (pos >= s.size()) break;
363 const char c = s[pos];
364 if (c == '*') {
365 ++pos;
366 lhs *= parseFactor();
367 } else if (c == '/') {
368 ++pos;
369 const double rhs = parseFactor();
370 if (validateOnly) {
371 lhs = 1.0;
372 } else if (rhs == 0.0) {
373 if (!error)
374 error = eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
375 "damage formula cannot divide by zero",
376 "formula[" + std::to_string(pos) + "]");
377 lhs = 0.0;
378 } else {
379 lhs /= rhs;
380 }
381 } else {
382 break;
383 }
384 }
385 return lhs;
386 }
387
388 double parseFactor() {
389 skipSpace();
390 if (pos < s.size() && (s[pos] == '+' || s[pos] == '-')) {
391 const bool negate = s[pos++] == '-';
392 const double value = parseFactor();
393 return negate ? -value : value;
394 }
395 if (pos < s.size() && s[pos] == '(') {
396 ++pos;
397 double v = parseExpression();
398 skipSpace();
399 if (pos >= s.size() || s[pos] != ')') {
400 fail("damage formula is missing a closing parenthesis");
401 return 0.0;
402 }
403 ++pos;
404 return v;
405 }
406 if (pos < s.size() && (s[pos] == 'a' || s[pos] == 'b') && pos + 1 < s.size() && s[pos + 1] == '.') {
407 const bool isA = s[pos] == 'a';
408 pos += 2;
409 std::string name;
410 while (pos < s.size() && (std::isalnum(static_cast<unsigned char>(s[pos])) || s[pos] == '_')) {
411 name.push_back(s[pos++]);
412 }
413 if (name.empty()) {
414 fail("damage formula attribute name must not be empty");
415 return 0.0;
416 }
417 RPGActor *actor = isA ? a : b;
418 return actor ? AttributeSystem::getFinal(actor, name) : 0.0;
419 }
420 if (pos < s.size() && (std::isdigit(static_cast<unsigned char>(s[pos])) || s[pos] == '.')) {
421 std::string num;
422 while (pos < s.size() && (std::isdigit(static_cast<unsigned char>(s[pos])) || s[pos] == '.')) {
423 num.push_back(s[pos++]);
424 }
425 char *end = nullptr;
426 const double value = std::strtod(num.c_str(), &end);
427 if (end == num.c_str() || *end != '\0' || !std::isfinite(value)) {
428 fail("damage formula contains an invalid number");
429 return 0.0;
430 }
431 return value;
432 }
433 fail("damage formula expected a number, attribute, unary sign, or parenthesized expression");
434 return 0.0;
435 }
436
437 void fail(std::string message) {
438 if (error) return;
439 error =
440 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, message, "formula[" + std::to_string(pos) + "]");
441 }
442};
443
444eve::Result<double> parseFormula(const std::string &formula, RPGActor *attacker, RPGActor *target, bool validateOnly) {
445 if (formula.empty())
446 return eve::Result<double>::failure(
447 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "damage formula must not be empty", "formula"));
448 FormulaParser parser(formula, attacker, target, validateOnly);
449 const double value = parser.parseExpression();
450 parser.skipSpace();
451 if (parser.error) return eve::Result<double>::failure(std::move(*parser.error));
452 if (parser.pos != formula.size())
453 return eve::Result<double>::failure(eve::Diagnostic::error(eve::DiagnosticCode::ParseError,
454 "damage formula contains trailing input",
455 "formula[" + std::to_string(parser.pos) + "]"));
456 if (!std::isfinite(value))
457 return eve::Result<double>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
458 "damage formula result must be finite", "formula"));
459 return eve::Result<double>::success(value);
460}

473DamageResult BattleSystem::resolveHit(RPGActor *attacker, RPGActor *target, const SkillDamageSpec &spec,
474 unsigned seed) {
475 DamageResult result;
476 if (!attacker || !target) return result;
477 std::mt19937 rng(seed);
478 std::uniform_real_distribution<double> dist(0.0, 1.0);
479
480 // 命中判定：攻击方 hit + (100 - 目标 evasion)
481 double attackerHit = 100.0 + TraitSystem::getExParam(attacker, "hit");
482 double targetEvasion = TraitSystem::getExParam(target, "evasion");
483 double hitChance = spec.hitChance + attackerHit - targetEvasion;
484 if (dist(rng) * 100.0 >= hitChance) {
485 result.hit = false;
486 return result;
487 }
488
489 // 公式（默认物理：a.atk - b.def，至少 1）
490 double amount;
491 if (spec.formula.empty()) {
492 amount = AttributeSystem::getFinal(attacker, "attack") - AttributeSystem::getFinal(target, "defense");
493 } else {
494 amount = evaluateFormula(spec.formula, attacker, target);
495 }
496 if (amount <= 0.0) amount = 1.0;
497
498 // 元素耐性
499 if (!spec.element.empty()) {
500 result.elementRate = TraitSystem::getElementRate(target, spec.element);
501 amount *= result.elementRate;
502 }
503
504 // 暴击（RPG Maker 默认 3x）
505 double critChance = spec.critChance + TraitSystem::getExParam(attacker, "critRate");
506 if (critChance > 0.0 && dist(rng) < critChance) {
507 result.crit = true;
508 amount *= 3.0;
509 }
510
511 result.amount = amount;
512 return result;
513}

564 const auto root = document.root();

566 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
567 "battle tactics catalogue must be a non-empty array",
568 "$", {}, "rpg.battle-tactics"));
569 const std::unordered_set<std::string> definitionFields = {"id", "rules"};
570 const std::unordered_set<std::string> ruleFields = {"skillId", "targetPolicy", "conditionResource", "belowRatio"};
571 std::unordered_map<std::string, BattleTacticsDefinition> proposed;
572 for (std::size_t index = 0; index < root.size(); ++index) {
573 const auto object = root.at(index);
574 const std::string path = "$[" + std::to_string(index) + "]";
575 if (!object.isObject())
576 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
577 "battle tactics definition must be an object", path,
578 {}, "rpg.battle-tactics"));
579 for (const auto &key : object.keys())
580 if (!definitionFields.contains(key))
581 return eve::Result<int>::failure(eve::Diagnostic::error(
582 eve::DiagnosticCode::InvalidArgument, "battle tactics definition contains an unknown field",
583 path + "." + key, {}, "rpg.battle-tactics"));
584 const auto id = object.get("id");
585 const auto rules = object.get("rules");
586 if (!id.isString() || !validId(id.asString()))
587 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
588 "battle tactics id must be a stable non-empty id",
589 path + ".id", {}, "rpg.battle-tactics"));
590 if (proposed.contains(id.asString()))
591 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::AlreadyExists,
592 "duplicate battle tactics id", path + ".id", {},
593 "rpg.battle-tactics"));
594 if (!rules.isArray() || rules.size() == 0 || rules.size() > 16)
595 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
596 "rules must contain between 1 and 16 entries",
597 path + ".rules", {}, "rpg.battle-tactics"));
598 BattleTacticsDefinition definition;
599 definition.id = id.asString();
600 for (std::size_t ruleIndex = 0; ruleIndex < rules.size(); ++ruleIndex) {
601 const auto rule = rules.at(ruleIndex);
602 const std::string rulePath = path + ".rules[" + std::to_string(ruleIndex) + "]";
603 if (!rule.isObject())
604 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
605 "battle tactic rule must be an object",
606 rulePath, {}, "rpg.battle-tactics"));
607 for (const auto &key : rule.keys())
608 if (!ruleFields.contains(key))
609 return eve::Result<int>::failure(eve::Diagnostic::error(
610 eve::DiagnosticCode::InvalidArgument, "battle tactic rule contains an unknown field",
611 rulePath + "." + key, {}, "rpg.battle-tactics"));
612 const auto skillId = rule.get("skillId");
613 const auto targetPolicy = rule.get("targetPolicy");
614 const auto conditionResource = rule.get("conditionResource");
615 const auto belowRatio = rule.get("belowRatio");
616 if (!skillId.isString() || !validId(skillId.asString(), true) ||
617 (!skillId.asString().empty() && !SkillRegistry::find(skillId.asString())))
618 return eve::Result<int>::failure(eve::Diagnostic::error(
619 eve::DiagnosticCode::NotFound, "skillId must be empty or reference a registered skill",
620 rulePath + ".skillId", {}, "rpg.battle-tactics"));
621 BattleTargetPolicy parsedPolicy = BattleTargetPolicy::Auto;
622 if (!targetPolicy.isString() || !parsePolicy(targetPolicy.asString(), parsedPolicy))
623 return eve::Result<int>::failure(
624 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "targetPolicy is invalid",
625 rulePath + ".targetPolicy", {}, "rpg.battle-tactics"));
626 const std::string targetType =
627 skillId.asString().empty() ? "enemySingle" : SkillRegistry::find(skillId.asString())->targetType;
628 if (!policyMatchesTargetType(parsedPolicy, targetType))
629 return eve::Result<int>::failure(eve::Diagnostic::error(
630 eve::DiagnosticCode::InvalidArgument, "targetPolicy does not match the skill target type",
631 rulePath + ".targetPolicy", {}, "rpg.battle-tactics"));
632 if (!conditionResource.isString() || !validId(conditionResource.asString(), true) ||
633 !belowRatio.isNumber() || !std::isfinite(belowRatio.asDouble()) || belowRatio.asDouble() < 0.0 ||
634 belowRatio.asDouble() > 1.0)
635 return eve::Result<int>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
636 "conditionResource and belowRatio are invalid",
637 rulePath, {}, "rpg.battle-tactics"));
638 if (parsedPolicy == BattleTargetPolicy::Auto && !conditionResource.asString().empty())
639 return eve::Result<int>::failure(eve::Diagnostic::error(
640 eve::DiagnosticCode::InvalidArgument, "conditional rules require an explicit target policy",
641 rulePath, {}, "rpg.battle-tactics"));
642 BattleTacticRule parsed;
643 parsed.skillId = skillId.asString();
644 parsed.targetPolicy = parsedPolicy;
645 parsed.conditionResource = conditionResource.asString();
646 parsed.belowRatio = belowRatio.asDouble();
647 definition.rules.push_back(std::move(parsed));
648 }
649 if (!definition.rules.back().conditionResource.empty())
650 return eve::Result<int>::failure(eve::Diagnostic::error(
651 eve::DiagnosticCode::InvalidArgument, "the final battle tactic rule must be unconditional",
652 path + ".rules", {}, "rpg.battle-tactics"));
653 proposed.emplace(definition.id, std::move(definition));
654 }
655 definitions() = std::move(proposed);
656 return eve::Result<int>::success(static_cast<int>(definitions().size()));
657}

662eve::Result<std::string> BattleTacticsCatalogue::queueAction(Battle *battle, RPGActor *actor,
663 const std::string &tacticsId) {
664 if (!battle || !actor)
665 return eve::Result<std::string>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
666 "battle and actor must not be null", "battle",
667 {}, "rpg.battle-tactics"));
668 const auto found = definitions().find(tacticsId);
669 if (found == definitions().end())
670 return eve::Result<std::string>::failure(eve::Diagnostic::error(eve::DiagnosticCode::NotFound,
671 "battle tactics id is not registered",
672 "tacticsId", {}, "rpg.battle-tactics"));
673 for (const auto &rule : found->second.rules) {
674 if (!rule.conditionResource.empty()) {
675 const int side = battle->sideOf(actor);
676 RPGActor *target =
677 rule.targetPolicy == BattleTargetPolicy::Self
678 ? actor
679 : battle->lowestHealthTarget(side, rule.targetPolicy == BattleTargetPolicy::LowestHealthAlly);
680 if (!target) continue;
681 const double maximum = target->getMax(rule.conditionResource);
682 if (maximum <= 0.0 || target->getCurrent(rule.conditionResource) / maximum >= rule.belowRatio) continue;
683 }
684 auto queued = battle->setActionByPolicyChecked(actor, rule.skillId, rule.targetPolicy);
685 if (!queued.ok()) return eve::Result<std::string>::failure(queued.status());
686 return eve::Result<std::string>::success(rule.skillId);
687 }
688 return eve::Result<std::string>::failure(eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
689 "battle tactics has no applicable default rule",
690 "rules", {}, "rpg.battle-tactics"));
691}

725eve::Result<BattleVictoryReceipt> BattleVictory::settle(RPGActor &actor, GameState &gameState, Tracker &tracker,
726 const BattleVictoryRequest &request) {
727 if (request.requiredQuestId.empty() || tracker.getState(request.requiredQuestId) != "active")
728 return victoryFailure(eve::DiagnosticCode::PreconditionViolation,
729 "required quest must be active before victory settlement", "requiredQuestId", request);
730 if (request.notifyTopic.empty() || request.notifyTarget.empty() || request.notifyAmount <= 0)
731 return victoryFailure(eve::DiagnosticCode::InvalidArgument,
732 "victory quest notification must be complete and positive", "notification", request);
733 if (request.attributeId.empty() || !std::isfinite(request.attributeAmount) || request.attributeAmount < 0.0 ||
734 !finiteAddition(gameState, request.attributeId, request.attributeAmount))
735 return victoryFailure(eve::DiagnosticCode::InvalidArgument, "victory attribute reward is invalid or non-finite",
736 "attribute", request);
737 if (request.defeatCounterId.empty() || request.defeatCounterAmount <= 0 ||
738 !finiteAddition(gameState, request.defeatCounterId, static_cast<double>(request.defeatCounterAmount)))
739 return victoryFailure(eve::DiagnosticCode::InvalidArgument, "victory counter increment is invalid",
740 "defeatCounter", request);
741 if (request.levelPointAttributeId.empty() || request.pointsPerLevel < 0)
742 return victoryFailure(eve::DiagnosticCode::InvalidArgument, "level-point reward configuration is invalid",
743 "levelPoints", request);
744
745 auto preparedXp = LevelSystem::prepareGainXp(&actor, request.xpAmount, request.xpGrowth);
746 if (!preparedXp.ok()) return eve::Result<BattleVictoryReceipt>::failure(preparedXp.status());
747 PreparedProgressionGain xp = std::move(preparedXp).takeValue();
748 const int levelsGained = xp.levelsGained();
749 const double levelPoints = static_cast<double>(levelsGained) * request.pointsPerLevel;
750 if (!finiteAddition(gameState, request.levelPointAttributeId, levelPoints))
751 return victoryFailure(eve::DiagnosticCode::InvalidArgument,
752 "level-point reward would produce a non-finite value", "levelPoints", request);
753
754 auto gameStateBefore = gameState.snapshotJson();
755 if (!gameStateBefore.ok()) return eve::Result<BattleVictoryReceipt>::failure(gameStateBefore.status());
756 std::string gameStateSnapshot = std::move(gameStateBefore).takeValue();
757 GameState candidateState;
758 auto candidateRestore = candidateState.restoreSnapshotJson(gameStateSnapshot);
759 if (!candidateRestore.ok()) return eve::Result<BattleVictoryReceipt>::failure(candidateRestore.status());
760 WorldState candidateWorld(candidateState);
761 if (candidateWorld.isObjectConsumed(request.mapId, request.objectId))
762 return victoryFailure(eve::DiagnosticCode::Conflict, "encounter was already consumed", "objectId", request);
763 auto candidateConsume = candidateWorld.consumeObject(request.mapId, request.objectId);
764 if (!candidateConsume.ok()) return eve::Result<BattleVictoryReceipt>::failure(candidateConsume.status());
765
766 TrackerRollback trackerBefore{tracker.entries, tracker.order, tracker.pending, tracker.polled};
767 gameState.addVariable(request.attributeId, request.attributeAmount);
768 gameState.addVariable(request.defeatCounterId, request.defeatCounterAmount);
769 if (levelPoints > 0.0) gameState.addVariable(request.levelPointAttributeId, levelPoints);
770 tracker.notify(request.notifyTopic, request.notifyTarget, request.notifyAmount);
771 WorldState world(gameState);
772 auto consumed = world.consumeObject(request.mapId, request.objectId);
773 if (!consumed.ok() || consumed.code() != eve::StatusCode::Applied) {
774 auto restored = gameState.restoreSnapshotJson(gameStateSnapshot);
775 const bool restoredOk = restored.ok();
776 restoreTracker(tracker, std::move(trackerBefore));
777 if (!restoredOk)
778 return victoryFailure(eve::DiagnosticCode::InvariantViolation,
779 "victory rollback could not restore GameState", "rollback", request);
780 return victoryFailure(eve::DiagnosticCode::Conflict, "encounter changed before victory commit", "objectId",
781 request);
782 }
783
784 auto committedXp = LevelSystem::commitGainXp(std::move(xp));
785 if (!committedXp.ok()) {
786 auto restored = gameState.restoreSnapshotJson(gameStateSnapshot);
787 const bool restoredOk = restored.ok();
788 restoreTracker(tracker, std::move(trackerBefore));
789 if (!restoredOk)
790 return victoryFailure(eve::DiagnosticCode::InvariantViolation,
791 "victory rollback could not restore GameState", "rollback", request);
792 return eve::Result<BattleVictoryReceipt>::failure(committedXp.status());
793 }
794
795 BattleVictoryReceipt receipt;
796 receipt.levelsGained = std::move(committedXp).takeValue();
797 receipt.skillsLearned = ClassSystem::checkLevelSkills(&actor);
798 return eve::Result<BattleVictoryReceipt>::success(std::move(receipt),
799 eve::Status::success(eve::StatusCode::Applied));
800}
