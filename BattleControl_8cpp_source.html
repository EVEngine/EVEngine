19Result<std::string> stringParameter(const Value& parameters, std::string_view name,
20 bool optional = false) {
21 const auto* object = parameters.getIf<Value::Object>();
22 if (!object)
23 return failure<std::string>(DiagnosticCode::InvalidArgument,
24 "RPG gameplay parameters must be an object", "parameters");
25 const auto found = object->find(std::string(name));
26 if (found == object->end() && optional) return Result<std::string>::success({});
27 if (found == object->end() || !found->second.isString())
28 return failure<std::string>(DiagnosticCode::InvalidArgument,
29 "RPG gameplay parameter must be a string",
30 "parameters." + std::string(name));
31 return Result<std::string>::success(found->second.asString());
32}

43Result<void> BattleControl::bindParticipant(SubjectRef subject, RPGActor* actor) {
44 if (!instance_.isValid() || !subject.isValid() || !actor)
45 return failure<void>(DiagnosticCode::InvalidArgument,
46 "battle, subject, and actor identities must be valid", "participant");
47 bool participant = false;
48 for (int index = 0; index < battle_.getActorCount(); ++index)
49 if (battle_.getActor(index) == actor) participant = true;
50 if (!participant)
51 return failure<void>(DiagnosticCode::NotFound,
52 "actor is not a participant in the adapted battle", "actor");
53 const auto existing = actors_.find(subject);
54 if (existing != actors_.end() && existing->second != actor)
55 return failure<void>(DiagnosticCode::Conflict,
56 "subject is already bound to another participant", "subject");
57 if (const auto reverse = subjectOf(actor); reverse.isValid() && reverse != subject)
58 return failure<void>(DiagnosticCode::Conflict,
59 "actor is already bound to another subject", "actor");
60 actors_[subject] = actor;
61 ++revision_;
62 return Result<void>::success(Status::success(StatusCode::Applied));
63}

84Result<GameplayObservation> BattleControl::observeGameplay(const GameplaySession& session,
85 SubjectRef instance) const {
86 if (instance != instance_ || !instance_.isValid())
87 return failure<GameplayObservation>(DiagnosticCode::NotFound,
88 "RPG battle gameplay instance was not found", "instance");
89 bool ownsParticipant = false;
90 Value::Array participants;
91 for (int index = 0; index < battle_.getActorCount(); ++index) {
92 RPGActor* actor = battle_.getActor(index);
93 const auto subject = subjectOf(actor);
94 if (!subject.isValid()) continue;
95 const bool controlled = controls(session, subject);
96 ownsParticipant = ownsParticipant || controlled;
97 participants.emplace_back(Value::Object{
98 {"alive", Value(battle_.isActorAlive(actor))},
99 {"controlled", Value(controlled)},
100 {"hp", Value(actor->getCurrent("hp"))},
101 {"maxHp", Value(actor->getMax("hp"))},
102 {"side", Value(battle_.getSide(index))},
103 {"subject", Value(subject.format())},
104 });
105 }
106 if (session.access == GameplayAccess::PlayerEquivalent && !ownsParticipant)
107 return failure<GameplayObservation>(DiagnosticCode::PreconditionViolation,
108 "session controls no participant in this RPG battle",
109 "session.controlledSubjects");
110 GameplayObservation observation;
111 observation.domain = gameplayId("gameplay:rpg-battle");
112 observation.instance = instance_;
113 observation.tick = tick_;
114 observation.revision = revision_;
115 observation.state = Value(Value::Object{
116 {"defeat", Value(battle_.isDefeat())},
117 {"finished", Value(battle_.isFinished())},
118 {"participants", Value(std::move(participants))},
119 {"turn", Value(battle_.getTurn())},
120 {"victory", Value(battle_.isVictory())},
121 });
122 return Result<GameplayObservation>::success(std::move(observation));
123}

148Result<GameplayCommandReceipt> BattleControl::submitGameplay(const GameplaySession& session,
149 SubjectRef instance,
150 const GameplayCommand& command) {
151 if (instance != instance_)
152 return failure<GameplayCommandReceipt>(DiagnosticCode::NotFound,
153 "RPG battle gameplay instance was not found", "instance");
154 RPGActor* actor = resolve(command.subject);
155 if (!actor)
156 return failure<GameplayCommandReceipt>(DiagnosticCode::NotFound,
157 "RPG battle participant was not found", "command.subject");
158 if (!controls(session, command.subject))
159 return failure<GameplayCommandReceipt>(DiagnosticCode::PreconditionViolation,
160 "session does not control this RPG participant",
161 "command.subject");
162 if (command.id.empty())
163 return failure<GameplayCommandReceipt>(DiagnosticCode::InvalidArgument,
164 "command id must not be empty", "command.id");
165 if (command.observedTick != tick_ || command.expectedRevision != revision_)
166 return failure<GameplayCommandReceipt>(DiagnosticCode::Conflict,
167 "RPG command was based on a stale observation",
168 "command.expectedRevision");
169 auto targetText = stringParameter(command.parameters, "target");
170 if (!targetText) return Result<GameplayCommandReceipt>::failure(targetText.status());
171 const auto targetId = PersistentId::parse(targetText.value());
172 RPGActor* target = targetId ? resolve(SubjectRef::fromPersistentId(*targetId)) : nullptr;
173 if (!target)
174 return failure<GameplayCommandReceipt>(DiagnosticCode::NotFound,
175 "RPG command target was not found", "parameters.target");
176 std::string skillId;
177 if (command.action == gameplayId("rpg:skill")) {
178 auto skill = stringParameter(command.parameters, "skillId");
179 if (!skill) return Result<GameplayCommandReceipt>::failure(skill.status());
180 skillId = std::move(skill).takeValue();
181 } else if (command.action != gameplayId("rpg:attack")) {
182 return failure<GameplayCommandReceipt>(DiagnosticCode::Unsupported,
183 "unsupported RPG gameplay action", "command.action");
184 }
185 auto queued = battle_.setActionChecked(actor, skillId, target);
186 if (!queued) return Result<GameplayCommandReceipt>::failure(queued.status());
187 ++revision_;
188 lastCommandByActor_[actor] = command.id;
189 GameplayEvent accepted;
190 accepted.sequence = nextEventSequence_++;
191 accepted.tick = tick_;
192 accepted.type = "rpg.command.accepted";
193 accepted.subject = command.subject;
194 accepted.causationCommandId = command.id;
195 accepted.correlationId = command.id;
196 accepted.payload = Value(Value::Object{{"action", Value(command.action.format())}});
197 events_.push_back(std::move(accepted));
198
199 GameplayCommandReceipt receipt;
200 receipt.commandId = command.id;
201 receipt.executionId = "rpg-round-" + std::to_string(battle_.getTurn() + 1);
202 receipt.acceptedTick = tick_;
203 receipt.resultingRevision = revision_;
204 receipt.details = Value(Value::Object{{"skillId", Value(skillId)},
205 {"target", Value(targetText.value())}});
206 return Result<GameplayCommandReceipt>::success(std::move(receipt),
207 Status::success(StatusCode::Applied));
208}

210Result<GameplayObservation> BattleControl::advanceGameplay(const GameplaySession& session,
211 SubjectRef instance,
212 const SimulationStep& step) {
213 if (instance != instance_)
214 return failure<GameplayObservation>(DiagnosticCode::NotFound,
215 "RPG battle gameplay instance was not found", "instance");
216 if (step.tick <= tick_)
217 return failure<GameplayObservation>(DiagnosticCode::Conflict,
218 "RPG simulation tick must increase", "step.tick");
219 auto authorized = observeGameplay(session, instance);
220 if (!authorized) return Result<GameplayObservation>::failure(authorized.status());
221 std::move(authorized).takeValue();
222 battle_.autoEnemyActions();
223 battle_.startRound();
224 while (battle_.executeNextAction() && !battle_.isFinished()) {}
225 battle_.pollEvents();
226 tick_ = step.tick;
227 ++revision_;
228 for (int index = 0; index < battle_.getEventCount(); ++index) {
229 RPGActor* caster = battle_.getEventCaster(index);
230 GameplayEvent event;
231 event.sequence = nextEventSequence_++;
232 event.tick = tick_;
233 event.type = "rpg.battle." + battle_.getEventAction(index);
234 event.subject = subjectOf(caster ? caster : battle_.getEventTarget(index));
235 if (const auto found = lastCommandByActor_.find(caster); found != lastCommandByActor_.end()) {
236 event.causationCommandId = found->second;
237 event.correlationId = found->second;
238 }
239 event.payload = Value(Value::Object{
240 {"amount", Value(battle_.getEventAmount(index))},
241 {"critical", Value(battle_.getEventCrit(index))},
242 {"skillId", Value(battle_.getEventSkillId(index))},
243 {"target", Value(subjectOf(battle_.getEventTarget(index)).format())},
244 });
245 events_.push_back(std::move(event));
246 }
247 return observeGameplay(session, instance);
248}

250Result<std::vector<GameplayEvent>> BattleControl::gameplayEvents(const GameplaySession& session,
251 SubjectRef instance,
252 std::uint64_t afterSequence) const {
253 auto authorized = observeGameplay(session, instance);
254 if (!authorized) return Result<std::vector<GameplayEvent>>::failure(authorized.status());
255 std::move(authorized).takeValue();
256 std::vector<GameplayEvent> result;
257 for (const auto& event : events_)
258 if (event.sequence > afterSequence) result.push_back(event);
259 const bool empty = result.empty();
260 return Result<std::vector<GameplayEvent>>::success(
261 std::move(result), Status::success(empty ? StatusCode::NoOp : StatusCode::Applied));
262}
