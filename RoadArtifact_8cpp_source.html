53template <class IncludeGroup>
54Collider roadCollider(const MeshBuild& mesh, IncludeGroup&& includeGroup) {
55 Collider result;
56 std::unordered_map<std::uint32_t, std::uint32_t> remap;
57 const int triangleCount = mesh.getIndexCount() / 3;
58 result.indices.reserve(mesh.indices().size());
59 for (int triangle = 0; triangle < triangleCount; ++triangle) {
60 const int group = mesh.getTriangleGroup(triangle);
61 if (group < 0 || !includeGroup(mesh.getGroupName(group))) continue;
62 for (int corner = 0; corner < 3; ++corner) {
63 const auto source = static_cast<std::uint32_t>(mesh.getIndex(triangle * 3 + corner));
64 const auto found = remap.find(source);
65 std::uint32_t destination = 0;
66 if (found == remap.end()) {
67 destination = static_cast<std::uint32_t>(result.vertices.size() / 3u);
68 remap.emplace(source, destination);
69 const int sourceIndex = static_cast<int>(source);
70 const float x = mesh.getPositionX(sourceIndex);
71 const float y = mesh.getPositionY(sourceIndex);
72 const float z = mesh.getPositionZ(sourceIndex);
73 result.vertices.insert(result.vertices.end(), {x, y, z});
74 result.bounds.include(x, y, z);
75 } else {
76 destination = found->second;
77 }
78 result.indices.push_back(destination);
79 }
80 }
81 return result;
82}

101std::uint64_t hashChunkMesh(const MeshBuild& mesh) {
102 std::uint64_t hash = 14695981039346656037ull;
103 auto append = [&](std::uint32_t value) {
104 for (int byte = 0; byte < 4; ++byte) {
105 hash ^= static_cast<std::uint8_t>((value >> (byte * 8)) & 0xffu);
106 hash *= 1099511628211ull;
107 }
108 };
109 for (float value : mesh.positions()) append(std::bit_cast<std::uint32_t>(value));
110 for (float value : mesh.normals()) append(std::bit_cast<std::uint32_t>(value));
111 for (float value : mesh.uvs()) append(std::bit_cast<std::uint32_t>(value));
112 for (float value : mesh.colors()) append(std::bit_cast<std::uint32_t>(value));
113 for (std::uint32_t value : mesh.indices()) append(value);
114 for (int value : mesh.triangleGroups()) append(static_cast<std::uint32_t>(value));
115 for (const auto& name : mesh.groupNames())
116 for (const unsigned char byte : name) {
117 hash ^= byte;
118 hash *= 1099511628211ull;
119 }
120 return hash;
121}

123Result<std::map<RoadChunkCoordinate, MeshBuild>> splitRoadMeshIntoChunks(const MeshBuild& source, float chunkSize,
124 int lod) {
125 std::map<RoadChunkCoordinate, RoadChunkAccumulator> accumulators;
126 const int triangleCount = source.getIndexCount() / 3;
127 for (int triangle = 0; triangle < triangleCount; ++triangle) {
128 const int ia = source.getIndex(triangle * 3);
129 const int ib = source.getIndex(triangle * 3 + 1);
130 const int ic = source.getIndex(triangle * 3 + 2);
131 const float centerX = (source.getPositionX(ia) + source.getPositionX(ib) + source.getPositionX(ic)) / 3.f;
132 const float centerZ = (source.getPositionZ(ia) + source.getPositionZ(ib) + source.getPositionZ(ic)) / 3.f;
133 const RoadChunkCoordinate coordinate{static_cast<int>(std::floor(centerX / chunkSize)),
134 static_cast<int>(std::floor(centerZ / chunkSize)), lod};
135 auto& accumulator = accumulators[coordinate];
136 const int sourceGroup = source.getTriangleGroup(triangle);
137 accumulator.mesh.setActiveGroup(sourceGroup >= 0 ? source.getGroupName(sourceGroup) : "default");
138 std::uint32_t destination[3]{};
139 const int sourceIndices[3] = {ia, ib, ic};
140 for (int corner = 0; corner < 3; ++corner) {
141 const auto sourceIndex = static_cast<std::uint32_t>(sourceIndices[corner]);
142 const auto found = accumulator.remap.find(sourceIndex);
143 if (found != accumulator.remap.end()) {
144 destination[corner] = found->second;
145 continue;
146 }
147 destination[corner] = static_cast<std::uint32_t>(accumulator.mesh.getVertexCount());
148 accumulator.remap.emplace(sourceIndex, destination[corner]);
149 const int vertex = static_cast<int>(sourceIndex);
150 accumulator.mesh.addVertex(source.getPositionX(vertex), source.getPositionY(vertex),
151 source.getPositionZ(vertex), source.getNormalX(vertex),
152 source.getNormalY(vertex), source.getNormalZ(vertex), source.getUvU(vertex),
153 source.getUvV(vertex));
154 if (source.hasVertexColors())
155 for (int component = 0; component < 4; ++component)
156 accumulator.colors.push_back(source.getColor(vertex, component));
157 }
158 accumulator.mesh.addTriangle(destination[0], destination[1], destination[2]);
159 }
160
161 std::map<RoadChunkCoordinate, MeshBuild> chunks;
162 for (auto& [coordinate, accumulator] : accumulators) {
163 if (!accumulator.colors.empty()) {
164 auto colors = accumulator.mesh.setVertexColors(std::move(accumulator.colors));
165 if (!colors.ok()) return Result<std::map<RoadChunkCoordinate, MeshBuild>>::failure(colors.status());
166 }
167 chunks.emplace(coordinate, std::move(accumulator.mesh));
168 }
169 return Result<std::map<RoadChunkCoordinate, MeshBuild>>::success(std::move(chunks));
170}

191float edgeSign(float ax, float az, float bx, float bz, float px, float pz) {
192 return (px - bx) * (az - bz) - (ax - bx) * (pz - bz);

195bool pointInTriangleXZ(float px, float pz, float ax, float az, float bx, float bz, float cx, float cz) {
196 const bool negative = edgeSign(ax, az, bx, bz, px, pz) < 0.f || edgeSign(bx, bz, cx, cz, px, pz) < 0.f ||
197 edgeSign(cx, cz, ax, az, px, pz) < 0.f;
198 const bool positive = edgeSign(ax, az, bx, bz, px, pz) > 0.f || edgeSign(bx, bz, cx, cz, px, pz) > 0.f ||
199 edgeSign(cx, cz, ax, az, px, pz) > 0.f;
200 return !(negative && positive);
201}

203Grid2D roadFootprint(const MeshBuild& mesh, const Bounds& bounds, float requestedCellSize, float padding = 0.f) {
204 Grid2D grid;
205 if (!bounds.isValid()) return grid;
206 const float originX = bounds.minX - padding;
207 const float originZ = bounds.minZ - padding;
208 const float extentX = std::max(requestedCellSize, bounds.maxX - bounds.minX + padding * 2.f);
209 const float extentZ = std::max(requestedCellSize, bounds.maxZ - bounds.minZ + padding * 2.f);
210 const float cellSize = std::max(requestedCellSize, std::max(extentX, extentZ) / 512.f);
211 const int width = std::clamp(static_cast<int>(std::ceil(extentX / cellSize)), 1, 512);
212 const int height = std::clamp(static_cast<int>(std::ceil(extentZ / cellSize)), 1, 512);
213 grid.resize(width, height);
214 grid.fill(0);
215 grid.setMeta("originX", std::to_string(originX));
216 grid.setMeta("originZ", std::to_string(originZ));
217 grid.setMeta("cellSize", std::to_string(cellSize));
218 grid.setMeta("semantics", "road_footprint");
219
220 const int triangleCount = mesh.getIndexCount() / 3;
221 for (int triangle = 0; triangle < triangleCount; ++triangle) {
222 const int group = mesh.getTriangleGroup(triangle);
223 if (group < 0) continue;
224 const std::string name = mesh.getGroupName(group);
225 if (name != "asphalt" && name != "asphalt.bridge" && name != "asphalt.tunnel" &&
226 name != "asphalt.tunnel.portal" && name != "sidewalk" && name != "curb")
227 continue;
228 const int ia = mesh.getIndex(triangle * 3);
229 const int ib = mesh.getIndex(triangle * 3 + 1);
230 const int ic = mesh.getIndex(triangle * 3 + 2);
231 const float ax = mesh.getPositionX(ia), az = mesh.getPositionZ(ia);
232 const float bx = mesh.getPositionX(ib), bz = mesh.getPositionZ(ib);
233 const float cx = mesh.getPositionX(ic), cz = mesh.getPositionZ(ic);
234 const int minX =
235 std::clamp(static_cast<int>(std::floor((std::min({ax, bx, cx}) - originX) / cellSize)), 0, width - 1);
236 const int maxX =
237 std::clamp(static_cast<int>(std::floor((std::max({ax, bx, cx}) - originX) / cellSize)), 0, width - 1);
238 const int minZ =
239 std::clamp(static_cast<int>(std::floor((std::min({az, bz, cz}) - originZ) / cellSize)), 0, height - 1);
240 const int maxZ =
241 std::clamp(static_cast<int>(std::floor((std::max({az, bz, cz}) - originZ) / cellSize)), 0, height - 1);
242 for (int z = minZ; z <= maxZ; ++z) {
243 for (int x = minX; x <= maxX; ++x) {
244 const float px = originX + (static_cast<float>(x) + 0.5f) * cellSize;
245 const float pz = originZ + (static_cast<float>(z) + 0.5f) * cellSize;
246 if (pointInTriangleXZ(px, pz, ax, az, bx, bz, cx, cz)) grid.setCell(x, z, 1);
247 }
248 }
249 }
250 return grid;
251}

253Grid2D roadSurfaceWeights(Grid2D grid, float cellSize, float blendDistance) {
254 const int width = grid.getWidth(), height = grid.getHeight();
255 const float infinity = static_cast<float>(width + height + 1);
256 std::vector<float> distance(static_cast<std::size_t>(width * height), infinity);
257 const auto index = [width](int x, int z) { return static_cast<std::size_t>(z * width + x); };
258 for (int z = 0; z < height; ++z)
259 for (int x = 0; x < width; ++x)
260 if (grid.getCell(x, z) != 0) distance[index(x, z)] = 0.f;
261
262 constexpr float diagonal = 1.41421356f;
263 auto relax = [&](int x, int z, int nx, int nz, float cost) {
264 if (nx < 0 || nz < 0 || nx >= width || nz >= height) return;
265 distance[index(x, z)] = std::min(distance[index(x, z)], distance[index(nx, nz)] + cost);
266 };
267 for (int z = 0; z < height; ++z) {
268 for (int x = 0; x < width; ++x) {
269 relax(x, z, x - 1, z, 1.f);
270 relax(x, z, x, z - 1, 1.f);
271 relax(x, z, x - 1, z - 1, diagonal);
272 relax(x, z, x + 1, z - 1, diagonal);
273 }
274 }
275 for (int z = height; z-- > 0;) {
276 for (int x = width; x-- > 0;) {
277 relax(x, z, x + 1, z, 1.f);
278 relax(x, z, x, z + 1, 1.f);
279 relax(x, z, x + 1, z + 1, diagonal);
280 relax(x, z, x - 1, z + 1, diagonal);
281 }
282 }
283 for (int z = 0; z < height; ++z) {
284 for (int x = 0; x < width; ++x) {
285 const float metres = distance[index(x, z)] * cellSize;
286 const float weight = distance[index(x, z)] == 0.f
287 ? 1.f
288 : (blendDistance > 0.f ? std::clamp(1.f - metres / blendDistance, 0.f, 1.f) : 0.f);
289 grid.setDetail(x, z, static_cast<int>(std::lround(weight * 255.f)));
290 }
291 }
292 grid.setMeta("semantics", "road_surface_weight");
293 grid.setMeta("weightChannel", "detail");
294 grid.setMeta("weightScale", "255");
295 grid.setMeta("blendDistance", std::to_string(blendDistance));
296 return grid;
297}

309Result<void> appendNavigationParts(const road::RoadOverlay& overlay, ArtifactId root, const BuildKey& rootKey,
310 std::vector<ArtifactPart>& parts) {
311 auto append = [&](const road::RoadPolyline& line, std::string role) -> Result<void> {
312 // Fully junction-trimmed lanes legitimately have no visible samples.
313 // They carry no artifact data and must not invalidate the whole build.
314 if (line.xyz.empty()) return Result<void>::success();
315 if (line.xyz.size() % 3u != 0u)
316 return Result<void>::failure(Diagnostic::error(
317 DiagnosticCode::InvariantViolation, "road overlay coordinates are not packed xyz triples", role));
318 if (!std::all_of(line.xyz.begin(), line.xyz.end(), [](float value) { return std::isfinite(value); }))
319 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
320 "road overlay contains non-finite coordinates", role));
321 PointSet points;
322 points.reserve(line.xyz.size() / 3);
323 for (std::size_t i = 0; i + 2 < line.xyz.size(); i += 3)
324 points.add(line.xyz[i], line.xyz[i + 1], line.xyz[i + 2]);
325 auto metadata = setPointDataIntAttribute(points, "in_edge", line.inEdge);
326 if (!metadata.ok()) return metadata;
327 metadata = setPointDataIntAttribute(points, "out_edge", line.outEdge);
328 if (!metadata.ok()) return metadata;
329 metadata = setPointDataIntAttribute(points, "in_lane", line.inLane);
330 if (!metadata.ok()) return metadata;
331 metadata = setPointDataIntAttribute(points, "out_lane", line.outLane);
332 if (!metadata.ok()) return metadata;
333 metadata = setPointDataIntAttribute(points, "in_direction", static_cast<int>(line.inDirection));
334 if (!metadata.ok()) return metadata;
335 metadata = setPointDataIntAttribute(points, "out_direction", static_cast<int>(line.outDirection));
336 if (!metadata.ok()) return metadata;
337 metadata = setPointDataIntAttribute(points, "traffic_priority", line.trafficPriority);
338 if (!metadata.ok()) return metadata;
339 auto speed = setPointDataFloatAttribute(points, "speed_limit_mps", line.speedLimitMps);
340 if (!speed.ok()) return speed;
341 // Compute before moving `points`: function argument evaluation order
342 // must not decide whether the bounds see a moved-from PointSet.
343 const Bounds bounds = pointBounds(points);
344 auto part =
345 roadPart(role, root, rootKey, ArtifactType::PointSet, bounds, std::move(points), {root.child("mesh")});
346 if (!part.ok()) return Result<void>::failure(part.status());
347 parts.push_back(std::move(part).takeValue());
348 return Result<void>::success();
349 };
350 for (std::size_t i = 0; i < overlay.lanes.size(); ++i) {
351 auto added = append(overlay.lanes[i], "navigation/lane/" + std::to_string(i));
352 if (!added.ok()) return added;
353 }
354 for (std::size_t i = 0; i < overlay.turns.size(); ++i) {
355 auto added = append(overlay.turns[i], "navigation/turn/" + std::to_string(i));
356 if (!added.ok()) return added;
357 }
358 return Result<void>::success();
359}

363Result<GeneratedArtifact> generateRoadNetworkArtifact(const Params& params, ArtifactId id) {
364 if (id.isNil())
365 return Result<GeneratedArtifact>::failure(
366 Diagnostic::error(DiagnosticCode::InvalidArgument, "road artifact identity must not be nil", "id"));
367 auto key = BuildKey::forRecipe("mesh.roadNetwork", params);
368 if (!key)
369 return Result<GeneratedArtifact>::failure(
370 Diagnostic::error(DiagnosticCode::InvalidArgument, "road parameters cannot form a build key", "params"));
371 auto baked = road::bakeRoadNetworkRecipe(params);
372 if (!baked.ok()) return Result<GeneratedArtifact>::failure(baked.status());
373
374 road::RoadBakeResult result = std::move(baked).takeValue();
375 auto sourcePartition = road::PreparedRoadGeometrySources::prepare(result.mesh);
376 if (!sourcePartition.ok()) return Result<GeneratedArtifact>::failure(sourcePartition.status());
377 const auto sourceSpanCount = sourcePartition.value().spans().size();
378 const Bounds bounds = meshBounds(result.mesh);
379 Collider collider = roadCollider(result.mesh, isStructuralColliderGroup);
380 Collider drivableCollider = roadCollider(result.mesh, isDrivableColliderGroup);
381 const float chunkSize = params.getFloat("chunkSize", 64.f);
382 const int lodCount = params.getInt("lodCount", 3);
383 if (!std::isfinite(chunkSize) || chunkSize < 4.f || chunkSize > 1024.f || lodCount < 1 || lodCount > 4)
384 return Result<GeneratedArtifact>::failure(Diagnostic::error(
385 DiagnosticCode::InvalidArgument, "road chunkSize must be in [4,1024] and lodCount in [1,4]", "chunks"));
386 std::vector<std::map<RoadChunkCoordinate, MeshBuild>> lodChunks;
387 lodChunks.reserve(static_cast<std::size_t>(lodCount));
388 auto primaryChunks = splitRoadMeshIntoChunks(result.mesh, chunkSize, 0);
389 if (!primaryChunks.ok()) return Result<GeneratedArtifact>::failure(primaryChunks.status());
390 lodChunks.push_back(std::move(primaryChunks).takeValue());
391 const int basePathSegments = std::max(4, params.getInt("pathSegments", 48));
392 const int baseTurnSamples = std::max(4, params.getInt("turnSamples", 12));
393 for (int lod = 1; lod < lodCount; ++lod) {
394 Params lodParams = params;
395 lodParams.setInt("pathSegments", std::max(4, basePathSegments >> lod));
396 lodParams.setInt("turnSamples", std::max(4, baseTurnSamples >> lod));
397 lodParams.setFloat("samplingScale", params.getFloat("samplingScale", 1.f) * static_cast<float>(1 << lod));
398 lodParams.setFloat("junctionChordError",
399 std::min(1.f, params.getFloat("junctionChordError", 0.10f) * static_cast<float>(1 << lod)));
400 lodParams.setBool("navigation", false);
401 lodParams.setBool("placements", false);
402 auto lodBake = road::bakeRoadNetworkRecipe(lodParams);
403 if (!lodBake.ok()) return Result<GeneratedArtifact>::failure(lodBake.status());
404 auto chunks = splitRoadMeshIntoChunks(lodBake.value().mesh, chunkSize, lod);
405 if (!chunks.ok()) return Result<GeneratedArtifact>::failure(chunks.status());
406 lodChunks.push_back(std::move(chunks).takeValue());
407 }
408 const float maskCellSize = std::max(0.1f, params.getFloat("maskCellSize", 1.f));
409 const float terrainBlendDistance = params.getFloat("terrainBlendDistance", 2.f);
410 if (!std::isfinite(terrainBlendDistance) || terrainBlendDistance < 0.f || terrainBlendDistance > 64.f)
411 return Result<GeneratedArtifact>::failure(
412 Diagnostic::error(DiagnosticCode::InvalidArgument,
413 "road terrain blend distance must be finite and in [0,64]", "terrainBlendDistance"));
414 Grid2D footprint = roadFootprint(result.mesh, bounds, maskCellSize);
415 Grid2D surfaceWeights = roadFootprint(result.mesh, bounds, maskCellSize, terrainBlendDistance + maskCellSize);
416 const float surfaceCellSize = std::stof(surfaceWeights.getMeta("cellSize", "1"));
417 surfaceWeights = roadSurfaceWeights(std::move(surfaceWeights), surfaceCellSize, terrainBlendDistance);
418 if (!bounds.isValid() || !collider.isValid() || !drivableCollider.isValid() || footprint.getWidth() <= 0 ||
419 footprint.getHeight() <= 0 || surfaceWeights.getWidth() <= 0 || surfaceWeights.getHeight() <= 0)
420 return Result<GeneratedArtifact>::failure(
421 Diagnostic::error(DiagnosticCode::InvariantViolation, "road bake produced incomplete artifact data"));
422
423 std::vector<ArtifactPart> parts;
424 std::size_t chunkCount = 0;
425 for (const auto& chunks : lodChunks) chunkCount += chunks.size();
426 parts.reserve(6 + chunkCount + result.overlay.lanes.size() + result.overlay.turns.size());
427 auto meshPart = roadPart("mesh", id, *key, ArtifactType::MeshData, bounds, std::move(result.mesh));
428 if (!meshPart.ok()) return Result<GeneratedArtifact>::failure(meshPart.status());
429 parts.push_back(std::move(meshPart).takeValue());
