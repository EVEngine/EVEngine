37Body::~Body() {
38 if (body_ && world_ && world_->raw()) {
39 // Invalidate fixture wrappers before DestroyBody frees b2Fixtures.
40 b2Fixture *f = body_->GetFixtureList();
41 while (f) {
42 b2Fixture *next = f->GetNext();
43 auto *wrap = static_cast<Fixture *>(f->GetUserData());
44 if (wrap) {
45 world_->forgetFixture(wrap);
46 wrap->invalidate();
47 }
48 f = next;
49 }
50 body_->SetUserData(nullptr);
51 world_->raw()->DestroyBody(body_);
52 world_->forgetBody(this);
53 }
54 body_ = nullptr;
55 world_ = nullptr;
56 runtimeHandle_ = PhysicsBodyHandle::invalid();
57}

265Fixture *Body::newPolygonFixture(const std::vector<float> &vertices, float density,
266 float friction, float restitution) {
267 if (!body_ || !world_) throw eve::Exception("Body.newPolygonFixture: body destroyed");
268 if (vertices.size() < 6 || vertices.size() > 16 || vertices.size() % 2 != 0)
269 throw eve::Exception("Body.newPolygonFixture: expected 3..8 packed XY vertices");
270 std::vector<b2Vec2> points;
271 points.reserve(vertices.size() / 2);
272 for (size_t index = 0; index < vertices.size(); index += 2) {
273 if (!std::isfinite(vertices[index]) || !std::isfinite(vertices[index + 1]))
274 throw eve::Exception("Body.newPolygonFixture: vertices must be finite");
275 points.emplace_back(world_->toMeters(vertices[index]), world_->toMeters(vertices[index + 1]));
276 }
277 b2PolygonShape shape;
278 shape.Set(points.data(), static_cast<int32>(points.size()));
279 if (shape.GetVertexCount() < 3)
280 throw eve::Exception("Body.newPolygonFixture: vertices do not form a convex polygon");
281 b2FixtureDef def;
282 def.shape = &shape;
283 def.density = density;
284 def.friction = friction;
285 def.restitution = restitution;
286 b2Fixture *raw = body_->CreateFixture(&def);
287 if (!raw) throw eve::Exception("Body.newPolygonFixture: Box2D rejected the fixture");
288 auto *fixture = new Fixture(world_, this, raw);
289 raw->SetUserData(fixture);
290 world_->fixtures_.insert(fixture);
291 return fixture;
292}

294Fixture *Body::newChainFixture(const std::vector<float> &vertices, bool loop, float friction,
295 float restitution) {
296 if (!body_ || !world_) throw eve::Exception("Body.newChainFixture: body destroyed");
297 const size_t count = vertices.size() / 2;
298 if (vertices.size() % 2 != 0 || count < (loop ? 3U : 2U) || count > 100000U)
299 throw eve::Exception("Body.newChainFixture: invalid packed XY vertex count");
300 std::vector<b2Vec2> points;
301 points.reserve(count);
302 for (size_t index = 0; index < vertices.size(); index += 2) {
303 if (!std::isfinite(vertices[index]) || !std::isfinite(vertices[index + 1]))
304 throw eve::Exception("Body.newChainFixture: vertices must be finite");
305 const b2Vec2 point(world_->toMeters(vertices[index]), world_->toMeters(vertices[index + 1]));
306 if (!points.empty() && point == points.back())
307 throw eve::Exception("Body.newChainFixture: consecutive vertices must be distinct");
308 points.push_back(point);
309 }
310 if (loop && points.front() == points.back())
311 throw eve::Exception("Body.newChainFixture: loop endpoint is implicit and must not be repeated");
312 b2ChainShape shape;
313 if (loop)
314 shape.CreateLoop(points.data(), static_cast<int32>(points.size()));
315 else
316 shape.CreateChain(points.data(), static_cast<int32>(points.size()));
317 b2FixtureDef def;
318 def.shape = &shape;
319 def.friction = friction;
320 def.restitution = restitution;
321 b2Fixture *raw = body_->CreateFixture(&def);
322 if (!raw) throw eve::Exception("Body.newChainFixture: Box2D rejected the fixture");
323 auto *fixture = new Fixture(world_, this, raw);
324 raw->SetUserData(fixture);
325 world_->fixtures_.insert(fixture);
326 return fixture;
327}
