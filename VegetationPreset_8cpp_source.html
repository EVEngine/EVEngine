98 Result<void> statements(const Value::Array& values, const std::string& sourcePath) {
99 for (const auto& value : values) {
100 const auto* object = value.getIf<Value::Object>();
101 const auto* kind = object ? member(*object, "kind") : nullptr;
102 if (!object || !kind || !kind->isString())
103 return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError,
104 "preset statement is malformed", sourcePath, {},
105 "asset.import.vegetation-preset"));
106 if (kind->asString() == "command") {
107 const auto* domain = member(*object, "domain");
108 const auto* operation = member(*object, "operation");
109 auto arguments = strings(member(*object, "arguments"), sourcePath);
110 if (!domain || !domain->isString() || !operation || !operation->isString() || !arguments)
111 return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError,
112 "preset command is malformed", sourcePath, {},
113 "asset.import.vegetation-preset"));
114 if (domain->asString() == "Include") {
115 auto include = evaluate(joined(operation->asString(), arguments.value()));
116 if (!include) return include;
117 } else {
118 commands.push_back(
119 {domain->asString(), operation->asString(), std::move(arguments).takeValue(), sourcePath});
120 }
121 continue;
122 }
123 if (kind->asString() != "condition")
124 return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError,
125 "unknown preset statement kind", sourcePath, {},
126 "asset.import.vegetation-preset"));
127 const auto* name = member(*object, "predicate");
128 const auto* negated = member(*object, "negated");
129 const auto* body = member(*object, "statements");
130 auto arguments = strings(member(*object, "arguments"), sourcePath);
131 if (!name || !name->isString() || !negated || !negated->isBool() || !body || !body->isArray() || !arguments)
132 return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError,
133 "preset condition is malformed", sourcePath, {},
134 "asset.import.vegetation-preset"));
135 auto result = predicate(name->asString(), arguments.value(), context);
136 if (!result) return Result<void>::failure(result.status());
137 if (result.value() != negated->asBool()) {
138 auto nested = statements(*body->getIf<Value::Array>(), sourcePath);
139 if (!nested) return nested;
140 }
141 }
142 return Result<void>::success();
143 }

145 Result<void> evaluate(const std::string& name) {
146 auto found = definitions.find(name);
147 if (found == definitions.end() && name == "Use Default Flower Masks")
148 found = definitions.find("Use Default Flowers Masks");
149 if (found == definitions.end() && name == "Use Default Flower Settings")
150 found = definitions.find("Use Default Flowers Settings");
151 if (found == definitions.end())
152 return Result<void>::failure(Diagnostic::error(DiagnosticCode::NotFound, "TVE preset include is missing",
153 name, {}, "asset.import.vegetation-preset"));
154 if (!active.insert(name).second)
155 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Conflict, "TVE preset include cycle", name,
156 {}, "asset.import.vegetation-preset"));
157 const auto* root = found->second.getIf<Value::Object>();
158 const auto* schema = root ? member(*root, "schema") : nullptr;
159 const auto* version = root ? member(*root, "schemaVersion") : nullptr;
160 const auto* source = root ? member(*root, "sourcePath") : nullptr;
161 const auto* body = root ? member(*root, "statements") : nullptr;
162 if (!root || !schema || !schema->isString() || schema->asString() != "eve.vegetation-conversion-preset" ||
163 !version || !version->isInt64() || version->asInt() != 1 || !source || !source->isString() || !body ||
164 !body->isArray())
165 return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError,
166 "TVE preset definition is malformed", name, {},
167 "asset.import.vegetation-preset"));
168 auto result = statements(*body->getIf<Value::Array>(), source->asString());
169 active.erase(name);
170 return result;
171 }
