96T* checkedReceiver(HSQUIRRELVM vm, T* receiver) {
97 SQUserPointer tag = nullptr;
98 if (sq_gettype(vm, 1) != OT_INSTANCE || SQ_FAILED(sq_gettypetag(vm, 1, &tag)) ||

100 return nullptr;
