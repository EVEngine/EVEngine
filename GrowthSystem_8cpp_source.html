25 const std::uint64_t mixed = splitmix64(seed ^ (stream + 0x9e3779b97f4a7c15ULL) ^ key);

104Result<void> GrowthWorld::advanceDays(double days) {
105 if (!std::isfinite(days) || days < 0.0) {
106 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "days must be finite and >= 0",
107 "growth.world.advanceDays"));
108 }
109 if (days == 0.0) return Result<void>::success();
110
111 double remaining = days;
112 while (remaining > 0.0) {
113 const double toEdge = 1.0 - dayFraction_;
114 const double slice = remaining < toEdge ? remaining : toEdge;
115 const std::int64_t dayForHash = calendar_.dayIndex();
116 std::vector<Organism*> batch = Organism::liveOrganisms();
117 for (Organism* organism : batch) {
118 if (!organism || organism->isDead()) continue;
119 applyDaySlice(*organism, slice, dayForHash);
120 }
121 dayFraction_ += slice;
122 remaining -= slice;
123 if (dayFraction_ + 1e-12 >= 1.0) {
124 dayFraction_ = 0.0;
125 calendar_.advanceWholeDays(1);
126 }
127 }
128 return Result<void>::success();
129}

143double GrowthWorld::growthMultiplier(Organism& organism, const SpeciesDef& species) {
144 if (organism.isDead()) return 0.0;
145 const auto& needs = *organism.vitalNeeds();
146 for (const NeedMeterDef& need : species.needs) {
147 if (need.requiredForGrowth && meterValue(needs, need.id) < need.minPermille) return 0.0;
148 }
149
150 double mul = 1.0;
151 const std::string seasonId = calendar_.seasonId();
152 if (!species.seasonMulById.empty()) {
153 auto it = species.seasonMulById.find(seasonId);
154 if (it != species.seasonMulById.end()) mul = it->second;
155 } else if (!species.seasonMul.empty()) {
156 const int index = calendar_.seasonIndex();
157 if (index >= 0 && index < static_cast<int>(species.seasonMul.size())) {
158 mul = species.seasonMul[static_cast<std::size_t>(index)];
159 } else {
160 mul = species.seasonMul.back();
161 }
162 }
163 if (organism.isDiseased()) mul *= organism.health()->diseaseGrowthMul;
164 if (!(mul > 0.0) || !std::isfinite(mul)) return 0.0;
165 return mul;
166}

168void GrowthWorld::advanceGrowth(Organism& organism, const SpeciesDef& species, double days) {
169 auto& growth = *organism.growth();
170 if (growth.dead || growth.mature || growth.harvested) return;
171 if (growth.stageIndex < 0 || growth.stageIndex >= static_cast<int>(species.stages.size())) return;
172
173 const double mul = growthMultiplier(organism, species);
174 if (mul <= 0.0) return;
175
176 double remaining = days * mul;
177 while (remaining > 0.0 && !growth.mature && !growth.dead) {
178 const GrowthStageDef& stage = species.stages[static_cast<std::size_t>(growth.stageIndex)];
179 const double need = stage.durationDays - growth.stageProgressDays;
180 if (remaining + 1e-12 < need) {
181 growth.stageProgressDays += remaining;
182 remaining = 0.0;
183 break;
184 }
185 remaining -= need;
186 growth.stageProgressDays = 0.0;
187 if (growth.stageIndex + 1 >= static_cast<int>(species.stages.size())) {
188 growth.mature = true;
189 emit(GrowthEventKind::Matured, organism, stage.id);
190 break;
191 }
192 ++growth.stageIndex;
193 emit(GrowthEventKind::StageChanged, organism, species.stages[static_cast<std::size_t>(growth.stageIndex)].id);
194 }
195}

197void GrowthWorld::advanceDisease(Organism& organism, const SpeciesDef& species, double days,
198 std::int64_t dayIndexForHash) {
199 auto& health = *organism.health();
200 if (!health.activeDiseaseId.empty()) {
201 health.diseaseDaysLeft -= days;
202 if (health.diseaseDaysLeft <= 0.0) {
203 health.activeDiseaseId.clear();
204 health.diseaseDaysLeft = 0.0;
205 health.diseaseGrowthMul = 1.0;
206 emit(GrowthEventKind::Cured, organism, {});
207 }
208 return;
209 }
210
211 if (species.diseases.empty() || days <= 0.0) return;
212 if (!isStressed(*organism.vitalNeeds(), species)) return;
213
214 const double resist = std::clamp(species.baseDiseaseResist, 0.0, 1.0);
215 for (std::size_t i = 0; i < species.diseases.size(); ++i) {
216 const DiseaseSusceptibility& disease = species.diseases[i];
217 const double chance = std::clamp(disease.chancePerDay * days * (1.0 - resist), 0.0, 1.0);
218 const double roll = unitHash(
219 rngSeed_, 0xD15EA5EULL,
220 organism.stableId() ^ (static_cast<std::uint64_t>(dayIndexForHash) << 8) ^ static_cast<std::uint64_t>(i));
221 if (roll < chance) {
222 health.activeDiseaseId = disease.diseaseId;
223 health.diseaseDaysLeft = disease.durationDays;
224 health.diseaseGrowthMul = disease.growthMul;
225 emit(GrowthEventKind::Diseased, organism, disease.diseaseId);
226 break;
227 }
228 }

231void GrowthWorld::advanceProduce(Organism& organism, const SpeciesDef& species, double days) {
232 if (!organism.isMature() || organism.isDead()) return;

238 produce.daysSinceProduce -= species.produceIntervalDays;

243Result<void> GrowthWorld::spawnOffspring(Organism& parent, const SpeciesDef& species, int count) {
244 if (count <= 0) return Result<void>::success();
245 const std::string offspringId = species.reproduction.offspringSpeciesId.empty()
246 ? species.id
247 : species.reproduction.offspringSpeciesId;
248 for (int i = 0; i < count; ++i) {
249 auto child = Organism::createOrganism(offspringId);
250 if (!child) {
251 emit(GrowthEventKind::Died, parent, "offspring_failed");
252 return Result<void>::failure(child.status());
253 }
254 emit(GrowthEventKind::Offspring, *child.value(), offspringId);
255 }
256
257 auto& growth = *parent.growth();
258 if (species.reproduction.parentDiesOnBirth) {
259 growth.dead = true;
260 emit(GrowthEventKind::Died, parent, "fission");
261 } else if (species.reproduction.parentResetsStagesOnBirth) {

263 growth.stageProgressDays = 0.0;

265 growth.harvested = false;

271void GrowthWorld::advanceReproduction(Organism& organism, const SpeciesDef& species, double days) {
272 auto& repro = *organism.reproduction();
273 const ReproductionDef& def = species.reproduction;
274
275 if (def.mode == ReproductionMode::PairGestation) {
276 if (!repro.pregnant) return;
277 repro.gestationProgressDays += days;
278 if (repro.gestationProgressDays + 1e-12 < def.gestationDays) return;
279 repro.pregnant = false;
280 repro.gestationProgressDays = 0.0;
281 const int count =
282 def.offspringCountMin + (def.offspringCountMax - def.offspringCountMin) / 2;
283 spawnOffspring(organism, species, std::max(1, count))
284 .ignore("gestation completion already emitted offspring_failed on error");
285 return;
286 }
287
288 if (def.mode != ReproductionMode::SelfBud && def.mode != ReproductionMode::Fission &&
289 def.mode != ReproductionMode::SporeBurst)
290 return;
291 if (!organism.isMature() || organism.isDead()) return;
292 if (!(def.intervalDays > 0.0)) return;
293
294 repro.daysSinceReproduce += days;
295 while (repro.daysSinceReproduce + 1e-12 >= def.intervalDays && !organism.isDead()) {
296 repro.daysSinceReproduce -= def.intervalDays;
297 const int count =
298 def.offspringCountMin + (def.offspringCountMax - def.offspringCountMin) / 2;
299 emit(GrowthEventKind::Bred, organism, "auto:" + species.kindId);
300 spawnOffspring(organism, species, std::max(1, count))
301 .ignore("interval reproduction already emitted offspring_failed on error");
302 if (organism.isDead() || def.parentResetsStagesOnBirth) break;
303 }

306void GrowthWorld::applyDaySlice(Organism& organism, double days, std::int64_t dayIndexForHash) {
307 const SpeciesDef* species = SpeciesRegistry::find(organism.speciesId());
308 if (!species || organism.isDead()) return;

318Result<void> GrowthWorld::tryBreed(Organism& carrier, Organism& partner) {
319 if (&carrier == &partner) {
320 return Result<void>::failure(Diagnostic::error(
321 DiagnosticCode::InvalidArgument, "cannot breed an organism with itself", "growth.world.tryBreed"));
322 }
323 if (carrier.isDead() || partner.isDead()) {
324 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
325 "dead organisms cannot breed", "growth.world.tryBreed"));
326 }
327 if (!carrier.isMature() || !partner.isMature()) {
328 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
329 "both organisms must be mature", "growth.world.tryBreed"));
330 }
331 const SpeciesDef* species = SpeciesRegistry::find(carrier.speciesId());
332 if (!species) {
333 return Result<void>::failure(
334 Diagnostic::error(DiagnosticCode::NotFound, "unknown species", "growth.world.tryBreed"));
335 }
336 const ReproductionDef& def = species->reproduction;
337 if (def.mode != ReproductionMode::PairGestation || !def.requiresPartner) {
338 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
339 "species reproduction.mode is not pair_gestation",
340 "growth.world.tryBreed"));
341 }
342 if (def.partnerMustSameSpecies && carrier.speciesId() != partner.speciesId()) {
343 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "partners must share a species",
344 "growth.world.tryBreed"));
345 }
346 if (!(def.gestationDays > 0.0)) {
347 return Result<void>::failure(
348 Diagnostic::error(DiagnosticCode::InvalidArgument, "species cannot breed", "growth.world.tryBreed"));
349 }
350 if (carrier.isPregnant() || partner.isPregnant()) {
351 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
352 "a partner is already pregnant", "growth.world.tryBreed"));
353 }
354
355 auto& repro = *carrier.reproduction();
356 repro.pregnant = true;
357 repro.gestationProgressDays = 0.0;
358 repro.partnerStableId = partner.stableId();
359 emit(GrowthEventKind::Bred, carrier, partner.speciesId());
360 return Result<void>::success();
361}
