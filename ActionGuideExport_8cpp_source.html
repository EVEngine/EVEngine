56Result<std::vector<uint8_t>> encodeRgbaPng(const ActionGuideAovBuffer& aov) {
57 if (!aov.rgba8 || aov.width == 0 || aov.height == 0)
58 return fail<std::vector<uint8_t>>(DiagnosticCode::InvalidArgument, "AOV buffer is empty", aov.aov);
59 if (uint64_t(aov.width) * aov.height * 4 != aov.rgba8->size())
60 return fail<std::vector<uint8_t>>(DiagnosticCode::InvalidArgument, "AOV RGBA size mismatch", aov.aov);
61 FrameCaptureBuffer buffer;
62 buffer.attachment = FrameAttachment::Final;
63 buffer.format = FramePixelFormat::Rgba8;
64 buffer.width = aov.width;
65 buffer.height = aov.height;
66 buffer.bytes = aov.rgba8;
67 FramePreviewOptions options;
68 options.view = "rawChannels";
69 options.maxWidth = aov.width;
70 options.maxHeight = aov.height;
71 options.maxBytes = size_t(aov.width) * aov.height * 4 + 64u * 1024u;
72 auto preview = frameCapturePreview(buffer, options);
73 if (!preview) return Result<std::vector<uint8_t>>::failure(preview.status());
74 return Result<std::vector<uint8_t>>::success(std::move(preview.value().png));
75}

79Result<std::string> actionGuideAovFileStem(std::string_view aov) {
80 if (aov == "silhouette" || aov == "guide.silhouette") return Result<std::string>::success("silhouette");
81 if (aov == "depth" || aov == "guide.depth") return Result<std::string>::success("depth");
82 if (aov == "bone" || aov == "guide.bone") return Result<std::string>::success("bone");
83 if (aov == "partId" || aov == "guide.partId") return Result<std::string>::success("partId");
84 if (aov == "normal" || aov == "guide.normal") return Result<std::string>::success("normal");
85 if (aov == "final" || aov == "whitebox_final") return Result<std::string>::success("whitebox_final");
86 return fail<std::string>(DiagnosticCode::InvalidArgument, "unsupported guide AOV name", std::string(aov));
87}

89Result<Value> writeActionGuideArtifactPack(const ActionGuideCaptureSnapshot& snapshot,
90 const std::filesystem::path& root) {
91 if (snapshot.phase != ActionGuideCapturePhase::Ready)
92 return fail<Value>(DiagnosticCode::InvalidArgument, "capture is not ready for export");
93 if (snapshot.shots.empty())
94 return fail<Value>(DiagnosticCode::InvalidArgument, "ready capture has no shot artifacts");
95 try {
96 std::filesystem::create_directories(root);
97 } catch (const std::exception& e) {
98 return fail<Value>(DiagnosticCode::Failed, std::string("create export root: ") + e.what());
99 }
100
101 Value::Array shotIndex;
102 Value::Array files;
103
104 try {
105 for (size_t i = 0; i < snapshot.shots.size(); ++i) {
106 const auto& shot = snapshot.shots[i];
107 const auto folder = padShotFolder(i + 1);
108 const auto shotDir = root / folder;
109 std::filesystem::create_directories(shotDir);
110
111 Value::Array aovNames;
112 Value::Object shotFiles;
113 for (const auto& aov : shot.aovs) {
114 auto stem = actionGuideAovFileStem(aov.aov);
115 if (!stem) return Result<Value>::failure(stem.status());
116 auto png = encodeRgbaPng(aov);
117 if (!png) return Result<Value>::failure(png.status());
118 const auto fileName = stem.value() + ".png";
119 const auto relPath = folder + "/" + fileName;
120 const auto tmpPath = shotDir / (fileName + ".tmp");
121 const auto dstPath = shotDir / fileName;
122 writeBytes(tmpPath, reinterpret_cast<const char*>(png.value().data()), png.value().size());
123 std::filesystem::rename(tmpPath, dstPath);
124 aovNames.emplace_back(aov.aov);
125 shotFiles.emplace(stem.value(), Value(relPath));
126 files.emplace_back(relPath);
127 }
128
129 Value::Array tags;
130 for (const auto& tag : shot.tags) tags.emplace_back(tag);
131
132 Value::Object meta;
133 meta.emplace("shotId", Value(shot.shotId));
134 meta.emplace("timeSeconds", Value(shot.timeSeconds));
135 if (shot.sampleIndex) meta.emplace("sampleIndex", Value(*shot.sampleIndex));
136 meta.emplace("view", Value(shot.view));
137 meta.emplace("aovs", Value(std::move(aovNames)));
138 meta.emplace("tags", Value(std::move(tags)));
139 meta.emplace("canvas", canvasValue(snapshot.pack.canvas));
140 meta.emplace("camera", cameraValue(snapshot.pack.camera));
141 meta.emplace("pixelsPerUnit", Value(snapshot.pack.canvas.pixelsPerUnit));
142 meta.emplace("depthEncoding", Value("normalized-gray8-near-far"));
143 meta.emplace("files", Value(std::move(shotFiles)));
144
145 const auto metaJson = Value(std::move(meta)).toJson();
146 if (!metaJson) return Result<Value>::failure(metaJson.status());
147 const auto metaTmp = shotDir / "meta.json.tmp";
148 const auto metaDst = shotDir / "meta.json";
149 writeBytes(metaTmp, metaJson.value().data(), metaJson.value().size());
150 std::filesystem::rename(metaTmp, metaDst);
151 files.emplace_back(folder + "/meta.json");
152
153 shotIndex.emplace_back(Value::object({{"shotId", Value(shot.shotId)}, {"folder", Value(folder)}}));
154 }
155
156 Value manifest = Value::object({{"schemaId", Value("eve.action.guide-pack.artifact")},
157 {"schemaVersion", Value(static_cast<std::int64_t>(1))},
158 {"revision", Value(static_cast<std::int64_t>(snapshot.revision))},
159 {"backend", Value(snapshot.backend)},
160 {"token", Value(snapshot.token)},
161 {"shotCount", Value(static_cast<std::int64_t>(snapshot.shots.size()))},
162 {"shots", Value(std::move(shotIndex))}});
163 const auto manifestJson = manifest.toJson();
164 if (!manifestJson) return Result<Value>::failure(manifestJson.status());
165 const auto manifestTmp = root / "manifest.json.tmp";
166 const auto manifestDst = root / "manifest.json";
167 writeBytes(manifestTmp, manifestJson.value().data(), manifestJson.value().size());
168 std::filesystem::rename(manifestTmp, manifestDst);
169
170 return Result<Value>::success(
171 Value::object({{"exportRoot", Value(root.generic_string())},
172 {"manifest", Value("manifest.json")},
173 {"files", Value(std::move(files))},
174 {"shotCount", Value(static_cast<std::int64_t>(snapshot.shots.size()))},
175 {"revision", Value(static_cast<std::int64_t>(snapshot.revision))},
176 {"backend", Value(snapshot.backend)}}));
177 } catch (const std::exception& e) {
178 return fail<Value>(DiagnosticCode::Failed, std::string("guide export failed: ") + e.what());
179 }
180}
