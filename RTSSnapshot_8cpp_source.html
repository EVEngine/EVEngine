14template <typename T>
15std::vector<SubjectRef> subjectsOf(const std::vector<T>& handles) {
16 std::vector<SubjectRef> result;
17 result.reserve(handles.size());
18 for (const auto& handle : handles) {
19 const auto subject = subjectOf(handle);
20 if (subject.isValid()) result.push_back(subject);
21 }
22 return result;
23}

205 value.combat.target = {};
206 value.durability = *unit->durability();
207 value.shield = *unit->shield();
208 value.veterancy = *unit->veterancy();
209 value.command = *unit->command();

213 value.command.uplink = {};
214 value.abilities = *unit->abilities();
215 if (value.abilities.channel) {
216 value.abilityTarget = subjectOf(value.abilities.channel->target);
217 value.abilities.channel->target = {};

219 value.capture = *unit->capture();
220 value.containmentCapacity = unit->containment()->capacity;

227 value.supply.convoyLeader = {};

297 value.orders = std::move(orders).takeValue();

299 result.buildings.push_back(std::move(value));
300 }
301 for (const auto& handle : resourceNodes_) {
302 if (handle.table != currentTable) continue;
303 auto* node = dynamic_cast<ResourceNode*>(ecs::try_get(handle));
304 if (node == nullptr) continue;
305 result.resourceNodes.push_back({node->identity()->subject, node->identity()->displayName, *node->position(),
306 *node->stock(), node->harvest()->capacity,
307 subjectsOf(node->harvest()->workers)});
308 }
309 for (const auto& handle : players_) {
310 if (handle.table != currentTable) continue;
311 auto* player = dynamic_cast<Player*>(ecs::try_get(handle));
312 if (player == nullptr) continue;
313 result.players.push_back({player->identity()->subject, player->identity()->displayName,
314 subjectsOf(player->selection()->units), subjectsOf(player->selection()->buildings)});
315 }
316 for (const auto& handle : factions_) {
317 if (handle.table != currentTable) continue;
318 auto* faction = dynamic_cast<Faction*>(ecs::try_get(handle));
319 if (faction == nullptr) continue;
320 result.factions.push_back({faction->identity()->subject, faction->identity()->displayName,
321 subjectsOf(faction->members()->units), subjectsOf(faction->members()->buildings),
322 *faction->strategy(), *faction->workforce(), *faction->productionPolicy(),
323 *faction->intel(), *faction->technology()});
324 }
325 for (const auto& handle : matches_) {
326 if (handle.table != currentTable) continue;
327 auto* match = dynamic_cast<Match*>(ecs::try_get(handle));
328 if (match == nullptr) continue;
329 RTSMatchSnapshot value;
330 value.subject = match->identity()->subject;
331 value.rules = *match->rules();
332 value.state = *match->state();
333 value.events = *match->events();
334 for (const auto& participant : match->participants()->entries)
335 value.participants.push_back({subjectOf(participant.faction.resolve()), participant.team,
336 participant.eliminated, participant.surrendered, participant.reason});
337 result.matches.push_back(std::move(value));
338 }
339 result.projectiles = projectiles_.snapshot();
340 return Result<RTSStateSnapshot>::success(std::move(result), Status::success(StatusCode::Applied));
341}
