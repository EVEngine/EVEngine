306 participant.eliminated, participant.surrendered, participant.reason});
307 result.matches.push_back(std::move(value));
308 }
309 result.projectiles = projectiles_.snapshot();
310 return Result<RTSStateSnapshot>::success(std::move(result), Status::success(StatusCode::Applied));
311}

670Result<void> RTS::restoreState(const RTSStateSnapshot& snapshot) {
671 if (snapshot.version != 1 || snapshot.units.size() != unitCount() ||
672 snapshot.buildings.size() != buildingCount() || snapshot.resourceNodes.size() != resourceNodeCount() ||
673 snapshot.players.size() != playerCount() || snapshot.factions.size() != factionCount() ||
674 snapshot.matches.size() != matchCount())
675 return snapshotFailure<void>(DiagnosticCode::Conflict,
676 "RTS snapshot topology/version does not match this module", "snapshot");
677
678 auto resolve = [&](SubjectRef subject) -> ecs::Entity* {
679 if (!subject.isValid()) return nullptr;
680 if (auto* value = findUnit(subject)) return value;
681 if (auto* value = findBuilding(subject)) return value;
682 if (auto* value = findResourceNode(subject)) return value;
683 for (const auto& handle : players_) if (subjectOf(handle) == subject) return ecs::try_get(handle);
684 for (const auto& handle : factions_) if (subjectOf(handle) == subject) return ecs::try_get(handle);
685 for (const auto& handle : matches_) if (subjectOf(handle) == subject) return ecs::try_get(handle);
686 return nullptr;
687 };
688 auto findPlayer = [&](SubjectRef subject) -> Player* {
689 for (const auto& value : players_) if (subjectOf(value) == subject)
690 return dynamic_cast<Player*>(ecs::try_get(value));
691 return nullptr;
692 };
693 auto findFaction = [&](SubjectRef subject) -> Faction* {
694 for (const auto& value : factions_) if (subjectOf(value) == subject)
695 return dynamic_cast<Faction*>(ecs::try_get(value));
696 return nullptr;
697 };
698 auto findMatch = [&](SubjectRef subject) -> Match* {
699 for (const auto& value : matches_) if (subjectOf(value) == subject)
700 return dynamic_cast<Match*>(ecs::try_get(value));
701 return nullptr;
702 };
703 std::unordered_set<SubjectRef> seen;
704 auto requireUnique = [&](SubjectRef subject) {
705 return subject.isValid() && resolve(subject) != nullptr && seen.insert(subject).second;
706 };
707 for (const auto& value : snapshot.units) if (!requireUnique(value.subject) || findUnit(value.subject) == nullptr)
708 return snapshotFailure<void>(DiagnosticCode::Conflict, "RTS snapshot contains an unknown/duplicate unit", "units.subject");
709 for (const auto& value : snapshot.buildings) if (!requireUnique(value.subject) || findBuilding(value.subject) == nullptr)
710 return snapshotFailure<void>(DiagnosticCode::Conflict, "RTS snapshot contains an unknown/duplicate building", "buildings.subject");
711 for (const auto& value : snapshot.resourceNodes) if (!requireUnique(value.subject) || findResourceNode(value.subject) == nullptr)
712 return snapshotFailure<void>(DiagnosticCode::Conflict, "RTS snapshot contains an unknown/duplicate resource node", "resourceNodes.subject");
713 for (const auto& value : snapshot.players) if (!requireUnique(value.subject) || findPlayer(value.subject) == nullptr)
714 return snapshotFailure<void>(DiagnosticCode::Conflict, "RTS snapshot contains an unknown/duplicate player", "players.subject");
715 for (const auto& value : snapshot.factions) if (!requireUnique(value.subject) || findFaction(value.subject) == nullptr)
716 return snapshotFailure<void>(DiagnosticCode::Conflict, "RTS snapshot contains an unknown/duplicate faction", "factions.subject");
717 for (const auto& value : snapshot.matches) if (!requireUnique(value.subject) || findMatch(value.subject) == nullptr)
718 return snapshotFailure<void>(DiagnosticCode::Conflict, "RTS snapshot contains an unknown/duplicate match", "matches.subject");
719
720 auto validType = [&](SubjectRef subject, auto* tag) {
721 using T = std::remove_pointer_t<decltype(tag)>;
722 return !subject.isValid() || dynamic_cast<T*>(resolve(subject)) != nullptr;
723 };
724 auto validUnits = [&](const std::vector<SubjectRef>& subjects) {
725 return std::all_of(subjects.begin(), subjects.end(), [&](SubjectRef subject) {
726 return validType(subject, static_cast<Unit*>(nullptr));
727 });
728 };
729 for (const auto& value : snapshot.units) {
730 if (!validType(value.faction, static_cast<Faction*>(nullptr)) ||
731 !validType(value.worker.resourceNode, static_cast<ResourceNode*>(nullptr)) ||
732 !validType(value.worker.dropoff, static_cast<Building*>(nullptr)) ||
733 (value.container.isValid() && !validType(value.container, static_cast<Unit*>(nullptr)) &&
734 !validType(value.container, static_cast<Building*>(nullptr))) || !validUnits(value.occupants) ||
735 !validType(value.supplyTarget, static_cast<Unit*>(nullptr)) ||
736 !validType(value.fireSupportRequester, static_cast<Unit*>(nullptr)))
737 return snapshotFailure<void>(DiagnosticCode::Conflict,
738 "RTS unit snapshot contains a relationship of the wrong type", "units.relationships");
739 }
740 for (const auto& value : snapshot.buildings) {
741 if (!validType(value.faction, static_cast<Faction*>(nullptr)) || !validUnits(value.builders) ||
742 !validType(value.capturingFaction, static_cast<Faction*>(nullptr)) ||
743 (value.rallyCommandTarget.isValid() && resolve(value.rallyCommandTarget) == nullptr) ||
744 !validType(value.rallyTransport, static_cast<Unit*>(nullptr)) || !validUnits(value.reinforcements) ||
745 !validUnits(value.occupants))
746 return snapshotFailure<void>(DiagnosticCode::Conflict,
747 "RTS building snapshot contains a relationship of the wrong type", "buildings.relationships");
748 }
749 for (const auto& value : snapshot.resourceNodes) if (!validUnits(value.workers))
750 return snapshotFailure<void>(DiagnosticCode::Conflict,
751 "RTS resource snapshot contains a non-unit worker", "resourceNodes.workers");
752 for (const auto& value : snapshot.players)
753 if (!validUnits(value.units) || !std::all_of(value.buildings.begin(), value.buildings.end(), [&](SubjectRef id) {
754 return validType(id, static_cast<Building*>(nullptr));
755 }))
756 return snapshotFailure<void>(DiagnosticCode::Conflict,
757 "RTS player snapshot contains an invalid selection", "players.selection");
758 for (const auto& value : snapshot.factions)
759 if (!validUnits(value.units) || !std::all_of(value.buildings.begin(), value.buildings.end(), [&](SubjectRef id) {
760 return validType(id, static_cast<Building*>(nullptr));
761 }))
762 return snapshotFailure<void>(DiagnosticCode::Conflict,
763 "RTS faction snapshot contains invalid membership", "factions.members");
764 for (const auto& value : snapshot.matches)
765 if (!std::all_of(value.participants.begin(), value.participants.end(), [&](const auto& participant) {
766 return validType(participant.faction, static_cast<Faction*>(nullptr));
767 }))
768 return snapshotFailure<void>(DiagnosticCode::Conflict,
769 "RTS match snapshot contains an invalid participant", "matches.participants");
770
771 RTSProjectileSystem stagedProjectiles;
772 if (!snapshot.projectiles.runtime.slots.empty() || !snapshot.projectiles.payloads.empty()) {
773 auto validProjectiles = stagedProjectiles.restore(snapshot.projectiles, resolve);
774 if (!validProjectiles) return validProjectiles;
775 }

777 auto prepareOrders = [&](const OrderComponent::Snapshot& source,
778 const std::map<std::string, SubjectRef>& targets) -> Result<OrderComponent::Snapshot> {
779 auto candidate = source;
780 for (const auto& [id, subject] : targets) {

782 auto* target = resolve(subject);
783 if (found == candidate.extended.end() || target == nullptr)
784 return snapshotFailure<OrderComponent::Snapshot>(DiagnosticCode::Conflict,
785 "RTS snapshot order target cannot be rebound", "orders.target");

787 }
788 OrderComponent validator;
789 auto valid = validator.restoreState(candidate);
790 if (!valid) return Result<OrderComponent::Snapshot>::failure(valid.status());

792 };
793 std::vector<OrderComponent::Snapshot> unitOrders;
794 std::vector<OrderComponent::Snapshot> buildingOrders;
795 unitOrders.reserve(snapshot.units.size());
796 buildingOrders.reserve(snapshot.buildings.size());
797 for (const auto& value : snapshot.units) {
798 auto prepared = prepareOrders(value.orders, value.orderTargets);
799 if (!prepared) return Result<void>::failure(prepared.status());
800 auto* unit = findUnit(value.subject);
801 auto attributes = value.attributes;
802 attributes.owner = unit->identity()->self;
803 AttributeComponent attributeValidator = unit->attributes()->values;
804 auto currentAttributes = RTSUnitAttributeAdapter::snapshot(*unit);
805 if (!currentAttributes) return Result<void>::failure(currentAttributes.status());
806 auto validAttributes = attributeValidator.restore(attributes,
807 currentAttributes.value().capturedRevision);
808 if (!validAttributes) return validAttributes;
809 TagSet tagValidator;
810 for (const auto& tag : value.tags) {
811 auto validTag = tagValidator.add(tag); if (!validTag) return validTag;
812 }
813 RTSEffectAdapter effectValidator;
814 auto validEffects = effectValidator.restore(value.effects);
815 if (!validEffects) return validEffects;
816 unitOrders.push_back(std::move(prepared).takeValue());
817 }
818 for (const auto& value : snapshot.buildings) {
819 auto prepared = prepareOrders(value.orders, value.orderTargets);
820 if (!prepared) return Result<void>::failure(prepared.status());
821 TagSet tagValidator;
822 for (const auto& tag : value.tags) {
823 auto validTag = tagValidator.add(tag); if (!validTag) return validTag;
824 }
825 RTSEffectAdapter effectValidator;
826 auto validEffects = effectValidator.restore(value.effects);
827 if (!validEffects) return validEffects;
828 ProductionComponent validator;
829 auto valid = validator.restore(value.productionJson);
830 if (!valid) return valid;
831 buildingOrders.push_back(std::move(prepared).takeValue());
832 }

834 auto handle = [&](SubjectRef subject) { auto* entity = resolve(subject); return entity ? ecs::handle_of(entity) : ecs::EntityHandle{}; };
835 for (std::size_t index = 0; index < snapshot.units.size(); ++index) {

942 if (unitCount() != 0 || buildingCount() != 0 || resourceNodeCount() != 0 || playerCount() != 0 ||
943 factionCount() != 0 || matchCount() != 0)
944 return snapshotFailure<void>(DiagnosticCode::Conflict,
945 "RTS snapshot topology can only be rebuilt into an empty module", "snapshot");
946 if (snapshot.version != 1)
947 return snapshotFailure<void>(DiagnosticCode::Conflict, "unsupported RTS snapshot version", "snapshot.version");
948
949 // A topology rebuild can follow destruction performed while an ECS view was
950 // deferred. Publish those tombstones before allocating replacement roots so
951 // component storage (notably effect timelines) cannot be reused half-staged.
952 ecs::commit();
953
954 std::unordered_set<SubjectRef> subjects;
955 auto reserveSubject = [&](SubjectRef subject) {
956 return subject.isValid() && subjects.insert(subject).second;
957 };
958 for (const auto& value : snapshot.units) if (!reserveSubject(value.subject))
959 return snapshotFailure<void>(DiagnosticCode::Conflict, "invalid/duplicate unit subject", "units.subject");
960 for (const auto& value : snapshot.buildings) if (!reserveSubject(value.subject))
961 return snapshotFailure<void>(DiagnosticCode::Conflict, "invalid/duplicate building subject", "buildings.subject");
962 for (const auto& value : snapshot.resourceNodes) if (!reserveSubject(value.subject))
963 return snapshotFailure<void>(DiagnosticCode::Conflict, "invalid/duplicate resource subject", "resourceNodes.subject");

965 return snapshotFailure<void>(DiagnosticCode::Conflict, "invalid/duplicate player subject", "players.subject");
966 for (const auto& value : snapshot.factions) if (!reserveSubject(value.subject))
967 return snapshotFailure<void>(DiagnosticCode::Conflict, "invalid/duplicate faction subject", "factions.subject");
968 for (const auto& value : snapshot.matches) if (!reserveSubject(value.subject))
969 return snapshotFailure<void>(DiagnosticCode::Conflict, "invalid/duplicate match subject", "matches.subject");
970
971 auto failAndClear = [&](const Status& status) {
972 clearOwnedRoots();
973 return Result<void>::failure(status);
974 };
975 for (const auto& value : snapshot.factions) {
976 auto created = newFaction(value.subject);
977 if (!created) return failAndClear(created.status());
978 }
979 for (const auto& value : snapshot.units) {
980 auto created = newUnit(value.subject, value.definition);
