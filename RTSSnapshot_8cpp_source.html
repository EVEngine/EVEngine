851Result<void> RTS::restoreState(const RTSStateSnapshot& snapshot) {
852 if ((snapshot.version != 1 && snapshot.version != 2) || snapshot.units.size() != unitCount() ||
853 snapshot.buildings.size() != buildingCount() || snapshot.resourceNodes.size() != resourceNodeCount() ||
854 snapshot.players.size() != playerCount() || snapshot.factions.size() != factionCount() ||
855 snapshot.matches.size() != matchCount())
856 return Result<void>::failure(Diagnostic::error(
857 DiagnosticCode::Conflict, "RTS snapshot topology/version does not match this module", "snapshot"));
858
859 auto resolve = [&](SubjectRef subject) -> ecs::Entity* {
860 if (!subject.isValid()) return nullptr;
861 if (auto* value = findUnit(subject)) return value;
862 if (auto* value = findBuilding(subject)) return value;
863 if (auto* value = findResourceNode(subject)) return value;
864 for (const auto& handle : players_)
865 if (subjectOf(handle) == subject) return ecs::try_get(handle);
866 for (const auto& handle : factions_)
867 if (subjectOf(handle) == subject) return ecs::try_get(handle);
868 for (const auto& handle : matches_)
869 if (subjectOf(handle) == subject) return ecs::try_get(handle);
870 return nullptr;
871 };
872 auto findPlayer = [&](SubjectRef subject) -> Player* {
873 for (const auto& value : players_)
874 if (subjectOf(value) == subject) return dynamic_cast<Player*>(ecs::try_get(value));
875 return nullptr;
876 };
877 auto findFaction = [&](SubjectRef subject) -> Faction* {
878 for (const auto& value : factions_)
879 if (subjectOf(value) == subject) return dynamic_cast<Faction*>(ecs::try_get(value));
880 return nullptr;
881 };
882 auto findMatch = [&](SubjectRef subject) -> Match* {
883 for (const auto& value : matches_)
884 if (subjectOf(value) == subject) return dynamic_cast<Match*>(ecs::try_get(value));
885 return nullptr;
886 };
887 std::unordered_set<SubjectRef> seen;
888 auto requireUnique = [&](SubjectRef subject) {
889 return subject.isValid() && resolve(subject) != nullptr && seen.insert(subject).second;
890 };
891 for (const auto& value : snapshot.units)
892 if (!requireUnique(value.subject) || findUnit(value.subject) == nullptr)
893 return Result<void>::failure(Diagnostic::error(
894 DiagnosticCode::Conflict, "RTS snapshot contains an unknown/duplicate unit", "units.subject"));
895 for (const auto& value : snapshot.buildings)
896 if (!requireUnique(value.subject) || findBuilding(value.subject) == nullptr)
897 return Result<void>::failure(Diagnostic::error(
898 DiagnosticCode::Conflict, "RTS snapshot contains an unknown/duplicate building", "buildings.subject"));
899 for (const auto& value : snapshot.resourceNodes)
900 if (!requireUnique(value.subject) || findResourceNode(value.subject) == nullptr)
901 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Conflict,
902 "RTS snapshot contains an unknown/duplicate resource node",
903 "resourceNodes.subject"));
904 for (const auto& value : snapshot.players)
905 if (!requireUnique(value.subject) || findPlayer(value.subject) == nullptr)
906 return Result<void>::failure(Diagnostic::error(
907 DiagnosticCode::Conflict, "RTS snapshot contains an unknown/duplicate player", "players.subject"));
908 for (const auto& value : snapshot.factions)
909 if (!requireUnique(value.subject) || findFaction(value.subject) == nullptr)
910 return Result<void>::failure(Diagnostic::error(
911 DiagnosticCode::Conflict, "RTS snapshot contains an unknown/duplicate faction", "factions.subject"));
912 for (const auto& value : snapshot.matches)
913 if (!requireUnique(value.subject) || findMatch(value.subject) == nullptr)
914 return Result<void>::failure(Diagnostic::error(
915 DiagnosticCode::Conflict, "RTS snapshot contains an unknown/duplicate match", "matches.subject"));
916
917 auto validType = [&](SubjectRef subject, auto* tag) {
918 using T = std::remove_pointer_t<decltype(tag)>;
919 return !subject.isValid() || dynamic_cast<T*>(resolve(subject)) != nullptr;
920 };
921 auto validUnits = [&](const std::vector<SubjectRef>& subjects) {
922 return std::all_of(subjects.begin(), subjects.end(),
923 [&](SubjectRef subject) { return validType(subject, static_cast<Unit*>(nullptr)); });
924 };
925 for (const auto& value : snapshot.units) {
926 if (!validType(value.faction, static_cast<Faction*>(nullptr)) ||
927 !validType(value.worker.resourceNode, static_cast<ResourceNode*>(nullptr)) ||
928 !validType(value.worker.dropoff, static_cast<Building*>(nullptr)) ||
929 (value.container.isValid() && !validType(value.container, static_cast<Unit*>(nullptr)) &&
930 !validType(value.container, static_cast<Building*>(nullptr))) ||
931 !validUnits(value.occupants) || !validType(value.supplyTarget, static_cast<Unit*>(nullptr)) ||
932 !validType(value.fireSupportRequester, static_cast<Unit*>(nullptr)))
933 return Result<void>::failure(Diagnostic::error(
934 DiagnosticCode::Conflict, "RTS unit snapshot contains a relationship of the wrong type",
935 "units.relationships"));
936 }
937 for (const auto& value : snapshot.buildings) {
938 if (!validType(value.faction, static_cast<Faction*>(nullptr)) || !validUnits(value.builders) ||
939 !validType(value.capturingFaction, static_cast<Faction*>(nullptr)) ||
940 (value.rallyCommandTarget.isValid() && resolve(value.rallyCommandTarget) == nullptr) ||
941 !validType(value.rallyTransport, static_cast<Unit*>(nullptr)) || !validUnits(value.reinforcements) ||
942 !validUnits(value.occupants))
943 return Result<void>::failure(Diagnostic::error(
944 DiagnosticCode::Conflict, "RTS building snapshot contains a relationship of the wrong type",
945 "buildings.relationships"));
946 }
947 for (const auto& value : snapshot.resourceNodes)
948 if (!validUnits(value.workers))
949 return Result<void>::failure(Diagnostic::error(
950 DiagnosticCode::Conflict, "RTS resource snapshot contains a non-unit worker", "resourceNodes.workers"));
951 for (const auto& value : snapshot.players)
952 if (!validUnits(value.units) ||
953 !std::all_of(value.buildings.begin(), value.buildings.end(),
954 [&](SubjectRef id) { return validType(id, static_cast<Building*>(nullptr)); }))
955 return Result<void>::failure(Diagnostic::error(
956 DiagnosticCode::Conflict, "RTS player snapshot contains an invalid selection", "players.selection"));
957 for (const auto& value : snapshot.factions)
958 if (!validUnits(value.units) ||
959 !std::all_of(value.buildings.begin(), value.buildings.end(),
960 [&](SubjectRef id) { return validType(id, static_cast<Building*>(nullptr)); }))
961 return Result<void>::failure(Diagnostic::error(
962 DiagnosticCode::Conflict, "RTS faction snapshot contains invalid membership", "factions.members"));
963 for (const auto& value : snapshot.matches)
964 if (!std::all_of(value.participants.begin(), value.participants.end(), [&](const auto& participant) {
965 return validType(participant.faction, static_cast<Faction*>(nullptr));
966 }))
967 return Result<void>::failure(Diagnostic::error(DiagnosticCode::Conflict,
968 "RTS match snapshot contains an invalid participant",
969 "matches.participants"));
