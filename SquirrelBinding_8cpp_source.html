40void reportFailureText(const char* what, const std::string& text) {
41 ScriptCallFailureLog& log = scriptCallFailureLog();
42 std::lock_guard<std::mutex> lock(log.mutex);
43 ++log.count;
44 if (text == log.lastMessage) return;
45 log.lastMessage = text;

57SQRESULT compileAndRunChunk(HSQUIRRELVM vm, const char* name, const char* source) {
58 if (!vm || !source) return SQ_ERROR;
59 const SQInteger top = sq_gettop(vm);
60 if (SQ_FAILED(sq_compilebuffer(vm, source, static_cast<SQInteger>(std::strlen(source)), name, SQTrue))) {
61 const ScriptErrorContext context = captureCompileError(vm);
62 reportFailureText(name, context.empty() ? std::string("unknown compile error") : formatScriptError(context));
63 sq_settop(vm, top);
64 return SQ_ERROR;
65 }
66 HSQOBJECT chunk;
67 sq_getstackobj(vm, -1, &chunk); // borrowed: still rooted by `top` until the call ends
68 const SQRESULT result = callFunction(vm, chunk, rootTable);
69 sq_settop(vm, top);
70 return result;
71}

103std::string memberPath(std::string_view parent, std::string_view key) {
104 if (!key.empty() && (std::isalpha(static_cast<unsigned char>(key.front())) != 0 || key.front() == '_') &&
105 std::all_of(key.begin() + 1, key.end(),
106 [](unsigned char value) { return std::isalnum(value) != 0 || value == '_'; })) {
107 return std::string(parent) + "." + std::string(key);
108 }
109 return std::string(parent) + "[\"" + std::string(key) + "\"]";
110}

122bool enterContainer(const HSQOBJECT& object, ConversionContext& context, Diagnostic& diagnostic,
123 const std::string& path) {
124 if (context.activeContainers.size() >= context.options.maxDepth) {
125 diagnostic = conversionError(context, DiagnosticCode::InvalidArgument,
126 "Squirrel value exceeds maximum conversion depth", path);
127 return false;
128 }
129 const void* identity = nullptr;
130 if (object._type == OT_ARRAY)
131 identity = static_cast<const void*>(object._unVal.pArray);
132 else if (object._type == OT_TABLE)
133 identity = static_cast<const void*>(object._unVal.pTable);
134 if (identity && std::find(context.activeContainers.begin(), context.activeContainers.end(), identity) !=
135 context.activeContainers.end()) {
136 diagnostic = conversionError(context, DiagnosticCode::InvalidArgument,
137 "cyclic Squirrel container cannot become an owning Value", path);
138 return false;
139 }
140 if (identity) context.activeContainers.push_back(identity);
141 return true;
142}
143
144void leaveContainer(const HSQOBJECT& object, ConversionContext& context) {
145 if ((object._type != OT_ARRAY && object._type != OT_TABLE) || context.activeContainers.empty()) return;
146 context.activeContainers.pop_back();
147}

159bool convertAt(HSQUIRRELVM vm, SQInteger index, ConversionContext& context, const std::string& path, Value& output,
160 Diagnostic& diagnostic) {
161 if (!countElement(context, diagnostic, path)) return false;
162
163 const SQObjectType type = sq_gettype(vm, index);
164 switch (type) {
165 case OT_NULL: output = Value(); return true;
166 case OT_BOOL: {
167 SQBool value = SQFalse;
168 if (SQ_FAILED(sq_getbool(vm, index, &value))) {
169 diagnostic =
170 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel boolean", path);
171 return false;
172 }
173 output = Value(value != SQFalse);
174 return true;
175 }
176 case OT_INTEGER: {
177 SQInteger value = 0;
178 if (SQ_FAILED(sq_getinteger(vm, index, &value))) {
179 diagnostic =
180 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel integer", path);
181 return false;
182 }
183 output = Value(static_cast<std::int64_t>(value));
184 return true;
185 }
186 case OT_FLOAT: {
187 SQFloat value = 0;
188 if (SQ_FAILED(sq_getfloat(vm, index, &value))) {
189 diagnostic =
190 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel number", path);
191 return false;
192 }
193 const double converted = static_cast<double>(value);
194 if (!std::isfinite(converted)) {
195 diagnostic = conversionError(context, DiagnosticCode::InvalidArgument,
196 "non-finite Squirrel number is not representable", path);
197 return false;
198 }
199 output = Value(converted);
200 return true;
201 }
202 case OT_STRING: {
203 const SQChar* value = nullptr;
204 SQInteger size = 0;
205 if (SQ_FAILED(sq_getstringandsize(vm, index, &value, &size)) || !value || size < 0) {
206 diagnostic =
207 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel string", path);
208 return false;
209 }
210 output = Value(std::string(value, static_cast<std::size_t>(size)));
211 return true;
212 }
213 case OT_ARRAY:
214 case OT_TABLE: break;
215 default:
216 diagnostic = conversionError(context, DiagnosticCode::Unsupported,
217 "Squirrel value kind is not part of eve::Value", path);
218 return false;
219 }
220
221 HSQOBJECT object{};
222 if (!takeObject(vm, index, object, diagnostic, context, path)) return false;
223 if (!enterContainer(object, context, diagnostic, path)) return false;
224
225 bool converted = true;
226 if (type == OT_ARRAY) {
227 const SQInteger size = sq_getsize(vm, index);
228 if (size < 0) {
229 diagnostic =
230 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel array size", path);
231 converted = false;
232 } else {
233 Value::Array values;
234 values.reserve(static_cast<std::size_t>(size));
235 const SQInteger absolute = index > 0 ? index : sq_gettop(vm) + index + 1;
236 for (SQInteger i = 0; converted && i < size; ++i) {
237 sq_pushinteger(vm, i);
238 if (SQ_FAILED(sq_get(vm, absolute))) {
239 diagnostic =
240 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel array element",
241 childPath(path, static_cast<std::size_t>(i)));
242 converted = false;
243 break;
244 }
245 Value element;
246 converted =
247 convertAt(vm, -1, context, childPath(path, static_cast<std::size_t>(i)), element, diagnostic);
248 sq_pop(vm, 1);
249 if (converted) values.push_back(std::move(element));
250 }
251 if (converted) output = Value(std::move(values));
252 }
253 } else {
254 Value::Object fields;
255 const SQInteger absolute = index > 0 ? index : sq_gettop(vm) + index + 1;
256 sq_pushnull(vm);
257 while (converted && SQ_SUCCEEDED(sq_next(vm, absolute))) {
258 if (sq_gettype(vm, -2) != OT_STRING) {
259 diagnostic = conversionError(context, DiagnosticCode::Unsupported,
260 "Squirrel table keys must be strings", path + "[<non-string-key>]");
261 converted = false;
262 sq_pop(vm, 2);
263 break;
264 }
265 const SQChar* key = nullptr;
266 SQInteger keySize = 0;
267 if (SQ_FAILED(sq_getstringandsize(vm, -2, &key, &keySize)) || !key || keySize < 0) {
268 diagnostic =
269 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel table key", path);
270 converted = false;
271 sq_pop(vm, 2);
272 break;
273 }
274 const std::string keyText(key, static_cast<std::size_t>(keySize));
275 const std::string field = memberPath(path, keyText);
276 Value value;
277 converted = convertAt(vm, -1, context, field, value, diagnostic);
278 sq_pop(vm, 2);
279 if (converted) fields.emplace(keyText, std::move(value));
280 }
281 sq_pop(vm, 1); // iterator
282 if (converted) output = Value(std::move(fields));
283 }
284 leaveContainer(object, context);
285 return converted;
286}

288bool validateValue(const Value& value, const SquirrelValueOptions& options, std::size_t depth, std::size_t& elements,
289 const std::string& path, Diagnostic& diagnostic) {
290 if (elements >= options.maxElements) {
291 diagnostic = Diagnostic::error(DiagnosticCode::InvalidArgument, "Value exceeds maximum Squirrel element count",
292 path, {}, options.source);
293 return false;
294 }
295 ++elements;
296 if (value.isDouble() && !std::isfinite(value.asDouble())) {
297 diagnostic = Diagnostic::error(DiagnosticCode::InvalidArgument, "non-finite Value cannot be pushed to Squirrel",
298 path, {}, options.source);
299 return false;
300 }
301 if (!value.isArray() && !value.isObject()) return true;
302 if (depth >= options.maxDepth) {
303 diagnostic = Diagnostic::error(DiagnosticCode::InvalidArgument,
304 "Value exceeds maximum Squirrel conversion depth", path, {}, options.source);
305 return false;
306 }
307 if (value.isArray()) {
308 for (std::size_t index = 0; index < value.arraySize(); ++index) {
309 if (!validateValue(value.at(index), options, depth + 1, elements, childPath(path, index), diagnostic))
310 return false;
311 }
312 return true;
313 }
314 for (const std::string& key : value.keys()) {
315 const Value* member = value.find(key);
316 if (!member) continue;
317 if (!validateValue(*member, options, depth + 1, elements, memberPath(path, key), diagnostic)) return false;
318 }
319 return true;
320}

322bool pushImpl(HSQUIRRELVM vm, const Value& value, const SquirrelValueOptions& options, std::size_t depth,
323 const std::string& path, Diagnostic& diagnostic) {
324 if (value.isNull()) {
325 sq_pushnull(vm);
326 return true;
327 }
328 if (value.isBool()) {
329 sq_pushbool(vm, value.asBool() ? SQTrue : SQFalse);
330 return true;
331 }
332 if (value.isInt64()) {
333 const auto integer = value.asInt();
334 if (integer < static_cast<std::int64_t>(std::numeric_limits<SQInteger>::min()) ||
335 integer > static_cast<std::int64_t>(std::numeric_limits<SQInteger>::max())) {
336 diagnostic = Diagnostic::error(DiagnosticCode::InvalidArgument, "Int64 value does not fit Squirrel integer",
337 path, {}, options.source);
338 return false;
339 }
340 sq_pushinteger(vm, static_cast<SQInteger>(integer));
341 return true;
342 }
343 if (value.isDouble()) {
344 sq_pushfloat(vm, static_cast<SQFloat>(value.asDouble()));
345 return true;
346 }
347 if (value.isString()) {
348 const std::string& string = value.asString();
349 sq_pushstring(vm, string.c_str(), static_cast<SQInteger>(string.size()));
350 return true;
351 }
352 if (depth >= options.maxDepth) {
353 diagnostic = Diagnostic::error(DiagnosticCode::InvalidArgument,
354 "Value exceeds maximum Squirrel conversion depth", path, {}, options.source);
355 return false;
356 }
357 if (value.isArray()) {
358 sq_newarray(vm, 0);
359 for (std::size_t index = 0; index < value.arraySize(); ++index) {
360 if (!pushImpl(vm, value.at(index), options, depth + 1, childPath(path, index), diagnostic)) return false;
361 if (SQ_FAILED(sq_arrayappend(vm, -2))) {
362 diagnostic =
363 Diagnostic::error(DiagnosticCode::Failed, "failed to append Value array element to Squirrel",
364 childPath(path, index), {}, options.source);
365 return false;
366 }
367 }
368 return true;
369 }
370 sq_newtable(vm);
371 for (const std::string& key : value.keys()) {
372 const Value* member = value.find(key);
373 if (!member) continue;
374 sq_pushstring(vm, key.c_str(), static_cast<SQInteger>(key.size()));
375 if (!pushImpl(vm, *member, options, depth + 1, memberPath(path, key), diagnostic)) return false;
376 if (SQ_FAILED(sq_newslot(vm, -3, SQFalse))) {
377 diagnostic = Diagnostic::error(DiagnosticCode::Failed, "failed to add Value object member to Squirrel",
378 memberPath(path, key), {}, options.source);
379 return false;
380 }
381 }
382 return true;
383}

415 if (!vm)
416 return Result<Value>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "Squirrel VM must not be null",

441Result<void> pushValue(HSQUIRRELVM vm, const Value& value, const SquirrelValueOptions& options) {
442 if (!vm)
443 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "Squirrel VM must not be null",
444 "$", {}, options.source));
445 std::size_t elements = 0;
446 Diagnostic diagnostic;
447 if (!validateValue(value, options, 0, elements, "$", diagnostic))
448 return Result<void>::failure(std::move(diagnostic));
449
450 const SQInteger top = sq_gettop(vm);
451 if (!pushImpl(vm, value, options, 0, "$", diagnostic)) {
452 sq_settop(vm, top);
453 return Result<void>::failure(std::move(diagnostic));
454 }
455 return Result<void>::success();
456}

547bool ignoreResult(const ssq::Object& result, const std::string& reason) {
548 if (result.getType() != ssq::Type::TABLE || reason.empty()) return false;
549 HSQUIRRELVM vm = result.getHandle();
550 if (!vm) return false;
551 const SQInteger top = sq_gettop(vm);
552 sq_pushobject(vm, result.getRaw());
553 sq_pushstring(vm, "ignored", -1);
554 sq_pushbool(vm, SQTrue);
555 if (SQ_FAILED(sq_newslot(vm, -3, SQFalse))) {
556 sq_settop(vm, top);
557 return false;
558 }
559 sq_pushstring(vm, "ignoreReason", -1);
560 sq_pushstring(vm, reason.c_str(), static_cast<SQInteger>(reason.size()));
561 if (SQ_FAILED(sq_newslot(vm, -3, SQFalse))) {
562 sq_settop(vm, top);
563 return false;
564 }
565 // A result table is already consumed by C++; this flag makes the script
566 // intent explicit without pretending that ignored means successful.
567 sq_pushstring(vm, "checked", -1);
568 sq_pushbool(vm, SQTrue);
569 if (SQ_FAILED(sq_newslot(vm, -3, SQFalse))) {
570 sq_settop(vm, top);
571 return false;
572 }
573 sq_settop(vm, top);
574 return true;
575}

594void exposeProviderBindings(ssq::Table& eveTable) {
595 ssq::Table provider = eveTable.addTable("provider");
596 HSQUIRRELVM vm = eveTable.getHandle();
597
598 provider.addFunc("publish", [vm](const std::string& name, const ssq::Object& instance) {
599 return projectResult(vm, ScriptProviderRegistry::instance().publish(name, instance), providerHandleValue);
600 });
601
602 provider.addFunc("publishFor", [vm](const std::string& capability, const std::string& name,
603 const ssq::Object& instance) {

608 // Script addresses a provider by name: the registry keeps exactly one live
609 // generation per name, so a script-held numeric generation would add no safety
610 // over the registry's own check. Native callers keep the generation-qualified
611 // handle, which is what protects them across a supersede.
612 provider.addFunc("unload", [vm](const std::string& name) {
613 ScriptProviderRegistry& registry = ScriptProviderRegistry::instance();
614 return projectResult(vm, registry.unload(ScriptProviderHandle{name, registry.generationOf(name)}));
615 });

622 });

625 // binding's parameters from the lambda's own parameter list, so `[] { ... }` is

633 });

638 ssq::Table entry(vm);
