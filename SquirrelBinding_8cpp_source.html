50bool enterContainer(const HSQOBJECT& object, ConversionContext& context, Diagnostic& diagnostic,
51 const std::string& path) {
52 if (context.activeContainers.size() >= context.options.maxDepth) {
53 diagnostic = conversionError(context, DiagnosticCode::InvalidArgument,
54 "Squirrel value exceeds maximum conversion depth", path);
55 return false;
56 }
57 const void* identity = nullptr;
58 if (object._type == OT_ARRAY)
59 identity = static_cast<const void*>(object._unVal.pArray);
60 else if (object._type == OT_TABLE)
61 identity = static_cast<const void*>(object._unVal.pTable);
62 if (identity && std::find(context.activeContainers.begin(), context.activeContainers.end(), identity) !=
63 context.activeContainers.end()) {
64 diagnostic = conversionError(context, DiagnosticCode::InvalidArgument,
65 "cyclic Squirrel container cannot become an owning Value", path);
66 return false;
67 }
68 if (identity) context.activeContainers.push_back(identity);
69 return true;
70}

87bool convertAt(HSQUIRRELVM vm, SQInteger index, ConversionContext& context, const std::string& path, Value& output,
88 Diagnostic& diagnostic) {
89 if (!countElement(context, diagnostic, path)) return false;
90
91 const SQObjectType type = sq_gettype(vm, index);
92 switch (type) {
93 case OT_NULL: output = Value(); return true;
94 case OT_BOOL: {
95 SQBool value = SQFalse;
96 if (SQ_FAILED(sq_getbool(vm, index, &value))) {
97 diagnostic =
98 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel boolean", path);
99 return false;
100 }
101 output = Value(value != SQFalse);
102 return true;
103 }
104 case OT_INTEGER: {
105 SQInteger value = 0;
106 if (SQ_FAILED(sq_getinteger(vm, index, &value))) {
107 diagnostic =
108 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel integer", path);
109 return false;
110 }
111 output = Value(static_cast<std::int64_t>(value));
112 return true;
113 }
114 case OT_FLOAT: {
115 SQFloat value = 0;
116 if (SQ_FAILED(sq_getfloat(vm, index, &value))) {
117 diagnostic =
118 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel number", path);
119 return false;
120 }
121 const double converted = static_cast<double>(value);
122 if (!std::isfinite(converted)) {
123 diagnostic = conversionError(context, DiagnosticCode::InvalidArgument,
124 "non-finite Squirrel number is not representable", path);
125 return false;
126 }
127 output = Value(converted);
128 return true;
129 }
130 case OT_STRING: {
131 const SQChar* value = nullptr;
132 SQInteger size = 0;
133 if (SQ_FAILED(sq_getstringandsize(vm, index, &value, &size)) || !value || size < 0) {
134 diagnostic =
135 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel string", path);
136 return false;
137 }
138 output = Value(std::string(value, static_cast<std::size_t>(size)));
139 return true;
140 }
141 case OT_ARRAY:
142 case OT_TABLE: break;
143 default:
144 diagnostic = conversionError(context, DiagnosticCode::Unsupported,
145 "Squirrel value kind is not part of eve::Value", path);
146 return false;
147 }
148
149 HSQOBJECT object{};
150 if (!takeObject(vm, index, object, diagnostic, context, path)) return false;
151 if (!enterContainer(object, context, diagnostic, path)) return false;
152
153 bool converted = true;
154 if (type == OT_ARRAY) {
155 const SQInteger size = sq_getsize(vm, index);
156 if (size < 0) {
157 diagnostic =
158 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel array size", path);
159 converted = false;
160 } else {
161 Value::Array values;
162 values.reserve(static_cast<std::size_t>(size));
163 const SQInteger absolute = index > 0 ? index : sq_gettop(vm) + index + 1;
164 for (SQInteger i = 0; converted && i < size; ++i) {
165 sq_pushinteger(vm, i);
166 if (SQ_FAILED(sq_get(vm, absolute))) {
167 diagnostic =
168 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel array element",
169 childPath(path, static_cast<std::size_t>(i)));
170 converted = false;
171 break;
172 }
173 Value element;
174 converted =
175 convertAt(vm, -1, context, childPath(path, static_cast<std::size_t>(i)), element, diagnostic);
176 sq_pop(vm, 1);
177 if (converted) values.push_back(std::move(element));
178 }
179 if (converted) output = Value(std::move(values));
180 }
181 } else {
182 Value::Object fields;
183 const SQInteger absolute = index > 0 ? index : sq_gettop(vm) + index + 1;
184 sq_pushnull(vm);
185 while (converted && SQ_SUCCEEDED(sq_next(vm, absolute))) {
186 if (sq_gettype(vm, -2) != OT_STRING) {
187 diagnostic = conversionError(context, DiagnosticCode::Unsupported,
188 "Squirrel table keys must be strings", path + "[<non-string-key>]");
189 converted = false;
190 sq_pop(vm, 2);
191 break;
192 }
193 const SQChar* key = nullptr;
194 SQInteger keySize = 0;
195 if (SQ_FAILED(sq_getstringandsize(vm, -2, &key, &keySize)) || !key || keySize < 0) {
196 diagnostic =
197 conversionError(context, DiagnosticCode::ParseError, "failed to read Squirrel table key", path);
198 converted = false;
199 sq_pop(vm, 2);
200 break;
201 }
202 const std::string keyText(key, static_cast<std::size_t>(keySize));
203 const std::string field = memberPath(path, keyText);
204 Value value;
205 converted = convertAt(vm, -1, context, field, value, diagnostic);
206 sq_pop(vm, 2);
207 if (converted) fields.emplace(keyText, std::move(value));
208 }
209 sq_pop(vm, 1); // iterator
210 if (converted) output = Value(std::move(fields));
211 }
212 leaveContainer(object, context);
213 return converted;
214}

216bool validateValue(const Value& value, const SquirrelValueOptions& options, std::size_t depth, std::size_t& elements,
217 const std::string& path, Diagnostic& diagnostic) {
218 if (elements >= options.maxElements) {
219 diagnostic = Diagnostic::error(DiagnosticCode::InvalidArgument, "Value exceeds maximum Squirrel element count",
220 path, {}, options.source);
221 return false;
222 }
223 ++elements;
224 if (value.isDouble() && !std::isfinite(value.asDouble())) {
225 diagnostic = Diagnostic::error(DiagnosticCode::InvalidArgument, "non-finite Value cannot be pushed to Squirrel",
226 path, {}, options.source);
227 return false;
228 }
229 if (!value.isArray() && !value.isObject()) return true;
230 if (depth >= options.maxDepth) {
231 diagnostic = Diagnostic::error(DiagnosticCode::InvalidArgument,
232 "Value exceeds maximum Squirrel conversion depth", path, {}, options.source);
233 return false;
234 }
235 if (value.isArray()) {
236 for (std::size_t index = 0; index < value.arraySize(); ++index) {
237 if (!validateValue(value.at(index), options, depth + 1, elements, childPath(path, index), diagnostic))
238 return false;
239 }
240 return true;
241 }
242 for (const std::string& key : value.keys()) {
243 const Value* member = value.find(key);
244 if (!member) continue;
245 if (!validateValue(*member, options, depth + 1, elements, memberPath(path, key), diagnostic)) return false;
246 }
247 return true;
248}

250bool pushImpl(HSQUIRRELVM vm, const Value& value, const SquirrelValueOptions& options, std::size_t depth,
251 const std::string& path, Diagnostic& diagnostic) {
252 if (value.isNull()) {
253 sq_pushnull(vm);
254 return true;
255 }
256 if (value.isBool()) {
257 sq_pushbool(vm, value.asBool() ? SQTrue : SQFalse);
258 return true;
259 }
260 if (value.isInt64()) {
261 const auto integer = value.asInt();
262 if (integer < static_cast<std::int64_t>(std::numeric_limits<SQInteger>::min()) ||
263 integer > static_cast<std::int64_t>(std::numeric_limits<SQInteger>::max())) {
264 diagnostic = Diagnostic::error(DiagnosticCode::InvalidArgument, "Int64 value does not fit Squirrel integer",
265 path, {}, options.source);
266 return false;
267 }
268 sq_pushinteger(vm, static_cast<SQInteger>(integer));
269 return true;
270 }
271 if (value.isDouble()) {
272 sq_pushfloat(vm, static_cast<SQFloat>(value.asDouble()));
273 return true;
274 }
275 if (value.isString()) {
276 const std::string& string = value.asString();
277 sq_pushstring(vm, string.c_str(), static_cast<SQInteger>(string.size()));
278 return true;
279 }
280 if (depth >= options.maxDepth) {
281 diagnostic = Diagnostic::error(DiagnosticCode::InvalidArgument,
282 "Value exceeds maximum Squirrel conversion depth", path, {}, options.source);
283 return false;
284 }
285 if (value.isArray()) {
286 sq_newarray(vm, 0);
287 for (std::size_t index = 0; index < value.arraySize(); ++index) {
288 if (!pushImpl(vm, value.at(index), options, depth + 1, childPath(path, index), diagnostic)) return false;
289 if (SQ_FAILED(sq_arrayappend(vm, -2))) {
290 diagnostic =
291 Diagnostic::error(DiagnosticCode::Failed, "failed to append Value array element to Squirrel",
292 childPath(path, index), {}, options.source);
293 return false;
294 }
295 }
296 return true;
297 }
298 sq_newtable(vm);
299 for (const std::string& key : value.keys()) {
300 const Value* member = value.find(key);
301 if (!member) continue;
302 sq_pushstring(vm, key.c_str(), static_cast<SQInteger>(key.size()));
303 if (!pushImpl(vm, *member, options, depth + 1, memberPath(path, key), diagnostic)) return false;
304 if (SQ_FAILED(sq_newslot(vm, -3, SQFalse))) {
305 diagnostic = Diagnostic::error(DiagnosticCode::Failed, "failed to add Value object member to Squirrel",
306 memberPath(path, key), {}, options.source);
307 return false;
308 }
309 }
310 return true;
311}

369Result<void> pushValue(HSQUIRRELVM vm, const Value& value, const SquirrelValueOptions& options) {
370 if (!vm)
371 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "Squirrel VM must not be null",
372 "$", {}, options.source));
373 std::size_t elements = 0;
374 Diagnostic diagnostic;
375 if (!validateValue(value, options, 0, elements, "$", diagnostic))
376 return Result<void>::failure(std::move(diagnostic));

379 if (!pushImpl(vm, value, options, 0, "$", diagnostic)) {

382 }

444bool ignoreResult(const ssq::Object& result, const std::string& reason) {
445 if (result.getType() != ssq::Type::TABLE || reason.empty()) return false;
446 HSQUIRRELVM vm = result.getHandle();
447 if (!vm) return false;
448 const SQInteger top = sq_gettop(vm);
449 sq_pushobject(vm, result.getRaw());
450 sq_pushstring(vm, "ignored", -1);
451 sq_pushbool(vm, SQTrue);
452 if (SQ_FAILED(sq_newslot(vm, -3, SQFalse))) {
453 sq_settop(vm, top);
454 return false;
455 }
456 sq_pushstring(vm, "ignoreReason", -1);
457 sq_pushstring(vm, reason.c_str(), static_cast<SQInteger>(reason.size()));
458 if (SQ_FAILED(sq_newslot(vm, -3, SQFalse))) {
459 sq_settop(vm, top);
460 return false;
461 }
462 // A result table is already consumed by C++; this flag makes the script
463 // intent explicit without pretending that ignored means successful.
464 sq_pushstring(vm, "checked", -1);
465 sq_pushbool(vm, SQTrue);
466 if (SQ_FAILED(sq_newslot(vm, -3, SQFalse))) {
467 sq_settop(vm, top);
468 return false;
469 }
470 sq_settop(vm, top);
471 return true;
472}
