213class Reader {
214public:
215 explicit Reader(std::span<const uint8_t> data) : data_(data) {}
216 bool empty() const { return offset_ == data_.size(); }
217 uint64_t varint() {
218 uint64_t n = 0;
219 for (int shift = 0; shift < 70; shift += 7) {
220 if (offset_ == data_.size()) throw Failure("Truncated protobuf varint", DiagnosticCode::ParseError);
221 const uint8_t b = data_[offset_++];
222 if (shift == 63 && b > 1) throw Failure("Overflowing protobuf varint", DiagnosticCode::ParseError);
223 n |= static_cast<uint64_t>(b & 127) << shift;
224 if (!(b & 128)) return n;
225 }
226 throw Failure("Invalid protobuf varint", DiagnosticCode::ParseError);
227 }
228 std::span<const uint8_t> take(uint64_t n) {
229 if (n > data_.size() - offset_) throw Failure("Truncated protobuf field", DiagnosticCode::ParseError);
230 auto result = data_.subspan(offset_, static_cast<size_t>(n));
231 offset_ += static_cast<size_t>(n);
232 return result;
233 }
234 Field next() {
235 const uint64_t tag = varint();
236 if ((tag >> 3) == 0 || (tag >> 3) > 0x1fffffff)
237 throw Failure("Invalid protobuf tag", DiagnosticCode::ParseError);
238 Field f;
239 f.id = static_cast<int>(tag >> 3);
240 f.wire = tag & 7;
241 switch (f.wire) {
242 case 0: f.number = varint(); break;
243 case 1: f.bytes = take(8); break;
244 case 2: f.bytes = take(varint()); break;
245 case 5: f.bytes = take(4); break;
246 default: throw Failure("Unsupported protobuf wire type", DiagnosticCode::ParseError);
247 }
248 return f;
249 }
250
251private:
252 std::span<const uint8_t> data_;
253 size_t offset_ = 0;
254};

276std::pair<std::string, RuntimeTensor> tensor(std::span<const uint8_t> data) {
277 Reader r(data);
278 RuntimeTensor v;
279 std::string name;
280 std::vector<int64_t> iv;
281 std::vector<float> fv;
282 bool raw = false, type = false;
283 while (!r.empty()) {
284 auto f = r.next();
285 switch (f.id) {
286 case 1: integers(f, v.shape); break;
287 case 2:
288 wire(f, 0);
289 if (f.number > 9) throw Failure("Unsupported tensor element type", DiagnosticCode::Unsupported);
290 v.element = static_cast<OnnxElement>(f.number);
291 type = true;
292 break;
293 case 4: reals(f, fv); break;
294 case 5:
295 case 7: integers(f, iv); break;
296 case 8:
297 wire(f, 2);
298 name = f.text();
299 break;
300 case 9:
301 wire(f, 2);
302 if (raw) throw Failure("Duplicate raw tensor data", DiagnosticCode::ParseError);
303 v.bytes.assign(f.bytes.begin(), f.bytes.end());
304 raw = true;
305 break;
306 case 14:
307 wire(f, 0);
308 if (f.number != 0) throw Failure("External tensor data unsupported", DiagnosticCode::Unsupported);
309 break;
310 case 3:
311 case 6:
312 case 10:
313 case 11:
314 case 13: throw Failure("External, segmented or unsupported tensor data", DiagnosticCode::Unsupported);
315 default: break;
316 }
317 }
318 if (!type) throw Failure("Tensor has no dtype", DiagnosticCode::ParseError);
319 const size_t n = count(v.shape), size = elementSize(v.element);
320 if (raw && (!iv.empty() || !fv.empty())) throw Failure("Ambiguous tensor payload", DiagnosticCode::ParseError);
321 if (!raw) {
322 if (v.element == OnnxElement::Float32) {
323 if (fv.size() != n || !iv.empty())
324 throw Failure("Invalid float tensor payload", DiagnosticCode::ParseError);
325 v.bytes.resize(n * size);
326 if (n) std::memcpy(v.bytes.data(), fv.data(), v.bytes.size());
327 } else {
328 if (iv.size() != n || !fv.empty())
329 throw Failure("Invalid integer tensor payload", DiagnosticCode::ParseError);
330 v.bytes.resize(n * size);
331 for (size_t i = 0; i < n; ++i) {
332 const auto x = iv[i];
333 if ((v.element == OnnxElement::UInt8 && (x < 0 || x > 255)) ||
334 (v.element == OnnxElement::Int8 && (x < -128 || x > 127)) ||
335 (v.element == OnnxElement::Bool && x != 0 && x != 1) ||
336 (v.element == OnnxElement::Int32 && (x < INT32_MIN || x > INT32_MAX)))
337 throw Failure("Integer initializer out of range", DiagnosticCode::ParseError);
338 std::memcpy(v.bytes.data() + i * size, &x, size);
339 }
340 }
341 }
342 validate(v);
343 return {std::move(name), std::move(v)};
344}

423 Reader r(bytes);
424 std::string name;
425 Input value;
426 while (!r.empty()) {
427 auto f = r.next();

429 wire(f, 2);

433 wire(f, 2);

435 while (!type.empty()) {
436 auto t = type.next();

440 }

444 while (!tt.empty()) {
445 auto p = tt.next();

447 wire(p, 0);

449 }
450 if (p.id == 2) {
451 wire(p, 2);

458 int64_t v = -1;
459 while (!dim.empty()) {
460 auto x = dim.next();
461 if (x.id == 1) {
462 wire(x, 0);

464 }
465 }
466 value.shape.push_back(v);
467 }
468 }
469 }
470 }
471 }
472 }

477 return {name, value};

479ModelData parseGraph(std::span<const uint8_t> bytes, size_t depth) {
480 ModelData model;
481 if (depth > 16) throw Failure("ONNX graph nesting limit exceeded", DiagnosticCode::ParseError);
482 Reader g(bytes);
483 while (!g.empty()) {
484 auto f = g.next();
485 if (f.id == 1) {
486 wire(f, 2);
487 if (model.nodes.size() >= 100000) throw Failure("Node limit exceeded", DiagnosticCode::ParseError);
488 model.nodes.push_back(node(f.bytes, depth));
489 }
490 if (f.id == 5) {
491 wire(f, 2);
492 auto t = tensor(f.bytes);
493 t.second.bytes.retainAcrossRuns();
494 if (t.first.empty() || !model.constants.emplace(std::move(t)).second)
495 throw Failure("Duplicate/empty initializer", DiagnosticCode::ParseError);
496 }
497 if (f.id == 11) {
498 wire(f, 2);
499 auto v = input(f.bytes);
500 model.info.inputs.push_back(v.first);
501 if (!model.inputs.emplace(std::move(v)).second)
502 throw Failure("Duplicate graph input", DiagnosticCode::ParseError);
503 }
504 if (f.id == 12) {
505 wire(f, 2);
506 model.info.outputs.push_back(input(f.bytes).first);
507 }
508 if (f.id == 15) throw Failure("Sparse initializers unsupported", DiagnosticCode::Unsupported);
509 }
510 return model;
511}

514ModelData parse(std::span<const uint8_t> bytes) {
515 if constexpr (std::endian::native != std::endian::little)
516 throw Failure("ONNX requires little-endian host", DiagnosticCode::Unsupported);
517 if (bytes.empty() || bytes.size() > 512u * 1024u * 1024u)
518 throw Failure("Model size exceeds import limit", DiagnosticCode::ParseError);
519 ModelData model;
520 Reader r(bytes);
521 std::span<const uint8_t> graph;
522 while (!r.empty()) {
523 auto f = r.next();
524 if (f.id == 1) {
525 wire(f, 0);
526 model.info.irVersion = static_cast<int64_t>(f.number);
527 }
528 if (f.id == 7) {
529 wire(f, 2);
530 if (!graph.empty()) throw Failure("Duplicate graph", DiagnosticCode::ParseError);
531 graph = f.bytes;
532 }
533 if (f.id == 8) {
534 wire(f, 2);
535 Reader ops(f.bytes);
536 std::string domain;
537 int64_t version = 0;
538 while (!ops.empty()) {
539 auto x = ops.next();
540 if (x.id == 1) {
541 wire(x, 2);
542 domain = x.text();
543 }
544 if (x.id == 2) {
545 wire(x, 0);
546 version = x.number;
547 }
548 }
549 if (!model.opsets.emplace(domain, version).second)
550 throw Failure("Duplicate opset", DiagnosticCode::ParseError);
551 }
552 if (f.id == 20 || f.id == 25 || f.id == 26)
553 throw Failure("Training/functions/device configuration unsupported", DiagnosticCode::Unsupported);
554 }
555 if (model.info.irVersion < 3 || model.info.irVersion > 10 || !model.opsets.contains("") ||
556 model.opsets.at("") < 13 || model.opsets.at("") > 17)
557 throw Failure("Unsupported ONNX IR/opset version", DiagnosticCode::UnknownVersion);
558 if (model.opsets.contains("com.microsoft") && model.opsets.at("com.microsoft") != 1)
559 throw Failure("Unsupported Microsoft opset", DiagnosticCode::UnknownVersion);
560 if (graph.empty()) throw Failure("Missing ONNX graph", DiagnosticCode::ParseError);
561 auto parsed = parseGraph(graph, 0);
562 model.nodes = std::move(parsed.nodes);
563 model.constants = std::move(parsed.constants);
564 model.inputs = std::move(parsed.inputs);
565 model.info.inputs = std::move(parsed.info.inputs);
566 model.info.outputs = std::move(parsed.info.outputs);
567 std::set<std::string> names;
568 for (const auto& [name, v] : model.constants) {
569 names.insert(name);
570 model.info.initializerBytes += v.bytes.size();
571 }
572 for (const auto& [name, v] : model.inputs) names.insert(name);
573 for (const auto& n : model.nodes) {
574 if (!model.opsets.contains(n.domain)) throw Failure("Node domain has no opset", DiagnosticCode::ParseError);
575 for (const auto& name : n.inputs)
576 if (!name.empty() && !names.contains(name))
577 throw Failure("Input not in topological order: " + name, DiagnosticCode::ParseError);
578 for (const auto& name : n.outputs)
579 if (!name.empty() && !names.insert(name).second)
580 throw Failure("Duplicate graph value: " + name, DiagnosticCode::ParseError);
581 if (!isSupported(n)) model.info.unsupportedNodes.push_back(n.name + " [" + n.domain + "::" + n.op + "]");
582 }
583 for (const auto& out : model.info.outputs)
584 if (!names.contains(out)) throw Failure("Undefined graph output", DiagnosticCode::ParseError);
585 size_t recursiveNodes = 0;
586 std::function<void(const ModelData&, const std::set<std::string>&, bool)> validateNested;
587 validateNested = [&](const ModelData& g, const std::set<std::string>& outer, bool root) {
588 recursiveNodes += g.nodes.size();
589 if (recursiveNodes > 100000) throw Failure("Recursive node limit exceeded", DiagnosticCode::ParseError);
590 auto available = outer;
591 std::set<std::string> locals;
592 for (const auto& [name, t] : g.constants) locals.insert(name);
593 for (const auto& [name, t] : g.inputs) locals.insert(name);
594 for (const auto& n : g.nodes)
595 for (const auto& out : n.outputs)
596 if (!out.empty() && !locals.insert(out).second)
597 throw Failure("Duplicate nested graph value: " + out, DiagnosticCode::ParseError);
598 available.insert(locals.begin(), locals.end());
599 for (const auto& n : g.nodes) {
600 if (!model.opsets.contains(n.domain))
601 throw Failure("Nested node domain has no opset", DiagnosticCode::ParseError);
602 for (const auto& input : n.inputs)
603 if (!input.empty() && !available.contains(input))
604 throw Failure("Undefined lexical capture: " + input, DiagnosticCode::ParseError);
605 if (!root && !isSupported(n))
606 model.info.unsupportedNodes.push_back(n.name + " [" + n.domain + "::" + n.op + "]");
607 for (const auto& [key, a] : n.attrs)
608 if (a.graph) validateNested(*a.graph, available, false);
609 }
610 for (const auto& out : g.info.outputs)
611 if (!available.contains(out)) throw Failure("Undefined nested output: " + out, DiagnosticCode::ParseError);
612 };
613 validateNested(model, {}, true);
614 model.info.nodeCount = model.nodes.size();
615 return model;
616}

641Value hold(RuntimeTensor t, Context& c) {
642 validate(t);
643 if (c.finite && t.element == OnnxElement::Float32)
644 for (size_t i = 0; i < count(t.shape); ++i)
645 if (!std::isfinite(read<float>(t, i))) throw Failure("Nonfinite float output");
646 const auto size = t.bytes.size();
647 if (size > 512u * 1024u * 1024u - c.liveBytes) throw Failure("ONNX live tensor memory limit exceeded");
648 // Context encloses every runtime Value; escaping public outputs are owning copies.
649 auto* raw = new RuntimeTensor(std::move(t));
650 c.liveBytes += size;
651 return {nullptr,
652 std::shared_ptr<const RuntimeTensor>(raw,
653 [&c, size](const RuntimeTensor* p) {
654 delete p;
655 c.liveBytes -= size;
656 }),
657 {}};
658}

695std::vector<size_t> plan(const ModelData& g, const std::vector<std::string>& requested) {
696 std::unordered_map<std::string, size_t> producers;
697 for (size_t i = 0; i < g.nodes.size(); ++i)
698 for (const auto& name : g.nodes[i].outputs)
699 if (!name.empty() && !producers.emplace(name, i).second) throw Failure("Duplicate graph output");
700 std::vector<bool> selected(g.nodes.size(), false);
701 std::vector<std::string> pending = requested;
702 while (!pending.empty()) {
703 auto name = std::move(pending.back());
704 pending.pop_back();
705 auto p = producers.find(name);
706 if (p == producers.end() || selected[p->second]) continue;
707 selected[p->second] = true;
708 admit(g.nodes[p->second]);
709 auto deps = dependencies(g.nodes[p->second]);
710 pending.insert(pending.end(), deps.begin(), deps.end());
711 }
712 std::vector<size_t> indegree(g.nodes.size());
713 std::vector<std::vector<size_t>> users(g.nodes.size());
714 size_t total = 0;
715 std::priority_queue<size_t, std::vector<size_t>, std::greater<size_t>> ready;
716 for (size_t i = 0; i < g.nodes.size(); ++i)
717 if (selected[i]) {
718 ++total;
719 std::set<size_t> parents;
720 for (const auto& name : dependencies(g.nodes[i])) {
721 auto p = producers.find(name);
722 if (p != producers.end()) parents.insert(p->second);
723 }
724 indegree[i] = parents.size();
725 for (auto p : parents) users[p].push_back(i);
726 if (parents.empty()) ready.push(i);
727 }
728 std::vector<size_t> order;
729 while (!ready.empty()) {
730 size_t i = ready.top();
731 ready.pop();
732 order.push_back(i);
733 for (auto user : users[i])
734 if (--indegree[user] == 0) ready.push(user);
735 }
736 if (order.size() != total) throw Failure("ONNX graph has cyclic lexical dependencies", DiagnosticCode::ParseError);
737 return order;
738}

740std::vector<Value> control(const Node& n, const std::vector<Value>& in, const Values& values, Context& c,
741 size_t depth) {
742 auto scalar = [&](size_t i) {
743 const auto& t = in.at(i).tensor();
744 if (count(t.shape) != 1) throw Failure("Control input must contain one element");
745 return integer(t);
746 };
747 auto seq = [&](size_t i) -> const std::vector<Value>& {
748 if (!in.at(i).sequence) throw Failure("Expected sequence input");
749 return *in[i].sequence;
750 };
751 auto sequence = [](std::vector<Value> v, OnnxElement type) {
752 return Value{nullptr, {}, std::make_shared<std::vector<Value>>(std::move(v)), type};
753 };
754 if (n.op == "SequenceEmpty") {
755 auto type = static_cast<OnnxElement>(attr(n, "dtype", 1));
756 elementSize(type);
757 return {sequence({}, type)};
758 }
759 if (n.op == "SequenceAt") {
760 const auto& s = seq(0);
761 int64_t i = scalar(1);
762 if (i < 0) i += static_cast<int64_t>(s.size());
763 if (i < 0 || static_cast<size_t>(i) >= s.size()) throw Failure("SequenceAt out of bounds");
764 return {s[i]};
765 }
766 if (n.op == "SequenceInsert") {
767 auto s = seq(0);
768 if (s.size() >= 100000) throw Failure("Sequence length limit exceeded");
769 int64_t i = in.size() > 2 ? scalar(2) : static_cast<int64_t>(s.size());
770 if (i < 0) i += static_cast<int64_t>(s.size());
771 if (i < 0 || static_cast<size_t>(i) > s.size()) throw Failure("SequenceInsert out of bounds");
772 const auto& t = in.at(1).tensor();
773 if (in[0].sequenceElement != t.element) throw Failure("Sequence element dtype mismatch");
774 s.insert(s.begin() + i, in[1]);
775 return {sequence(std::move(s), in[0].sequenceElement)};
776 }
777 if (n.op == "SplitToSequence") {
778 const auto& x = in.at(0).tensor();
779 int a = axis(attr(n, "axis", 0), x.shape.size());
780 std::vector<int64_t> lengths;
781 if (in.size() > 1 && (in[1].borrowed || in[1].owned)) {
782 const auto& split = in[1].tensor();
783 lengths = ints(split);
784 if (split.shape.empty()) {
785 if (lengths[0] <= 0) throw Failure("Invalid split size");
786 int64_t width = lengths[0];
787 lengths.clear();
788 for (int64_t i = 0; i < x.shape[a]; i += width) {
789 if (lengths.size() >= 100000) throw Failure("Sequence length limit exceeded");
790 lengths.push_back(std::min(width, x.shape[a] - i));
791 }
792 }
793 } else
794 lengths.assign(x.shape[a], 1);
795 if (lengths.size() > 100000) throw Failure("Sequence length limit exceeded");
796 int64_t total = 0;
797 for (auto size : lengths) {
798 if (size < 0 || size > x.shape[a] - total) throw Failure("Invalid split lengths");
799 total += size;
800 }
801 if (total != x.shape[a]) throw Failure("Split lengths do not cover axis");
802 size_t outer = 1, inner = elementSize(x.element);
803 for (int j = 0; j < a; ++j) outer *= x.shape[j];
804 for (size_t j = a + 1; j < x.shape.size(); ++j) inner *= x.shape[j];
805 std::vector<Value> s;
806 size_t offset = 0;
807 for (auto length : lengths) {
808 auto shape = x.shape;
809 shape[a] = length;
810 if (!attr(n, "keepdims", 1)) {
811 if (in.size() > 1 && (in[1].borrowed || in[1].owned))
812 throw Failure("keepdims=0 with explicit split unsupported", DiagnosticCode::Unsupported);
813 shape.erase(shape.begin() + a);
814 }
815 RuntimeTensor out{x.element, shape, std::vector<uint8_t>(count(shape) * elementSize(x.element))};
816 for (size_t o = 0; o < outer; ++o)
817 if (length)
818 std::memcpy(out.bytes.data() + o * length * inner,
819 x.bytes.data() + (o * x.shape[a] + offset) * inner, length * inner);
820 s.push_back(hold(std::move(out), c));
821 offset += length;
822 }
823 return {sequence(std::move(s), x.element)};
824 }
825 if (n.op == "ConcatFromSequence") {
826 const auto& s = seq(0);
827 if (s.empty()) throw Failure("Cannot concatenate empty sequence");
828 std::vector<RuntimeTensor> tensors;
829 std::vector<const RuntimeTensor*> inputs;
830 for (const auto& v : s) {
831 tensors.push_back(v.tensor());
832 if (attr(n, "new_axis", 0)) {
833 const int a = axis(attr(n, "axis", 0), tensors.back().shape.size() + 1);
834 tensors.back().shape.insert(tensors.back().shape.begin() + a, 1);
835 }
836 }
837 for (const auto& t : tensors) inputs.push_back(&t);
838 Node concat = n;
839 concat.op = "Concat";
840 auto out = executeShape(concat, inputs);
841 if (!out) throw Failure("ConcatFromSequence dispatch failed");
842 return {hold(std::move(*out), c)};
843 }
844 if (n.op == "If") {
845 if (in.at(0).tensor().element != OnnxElement::Bool) throw Failure("If condition must be boolean");
846 auto key = scalar(0) ? "then_branch" : "else_branch";
847 const auto& g = n.attrs.at(key).graph;
848 if (!g) throw Failure("Missing If branch");
849 return run(*g, values, g->info.outputs, c, depth + 1);
850 }
851 if (n.op == "Loop") {
852 const auto& body = n.attrs.at("body").graph;
853 if (!body) throw Failure("Missing Loop body");
854 if (in.size() < 2 || body->info.inputs.size() != in.size() || body->info.outputs.size() != in.size() - 1 ||
855 n.outputs.size() != in.size() - 2)
856 throw Failure("Loop scan outputs unsupported", DiagnosticCode::Unsupported);
857 const int64_t trips = (in[0].borrowed || in[0].owned) ? scalar(0) : 10000;
858 if (trips < 0 || trips > 10000) throw Failure("Loop trip budget exceeded");
859 bool condition = (in[1].borrowed || in[1].owned) ? scalar(1) != 0 : true;
860 std::vector<Value> state(in.begin() + 2, in.end());
861 for (int64_t i = 0; i < trips && condition; ++i) {
862 auto scope = values;
863 scope[body->info.inputs[0]] = hold(make(OnnxElement::Int64, {}, std::vector<int64_t>{i}), c);
864 scope[body->info.inputs[1]] =
865 hold(make(OnnxElement::Bool, {}, std::vector<uint8_t>{uint8_t(condition)}), c);
866 for (size_t j = 0; j < state.size(); ++j) scope[body->info.inputs[j + 2]] = state[j];
867 auto result = run(*body, std::move(scope), body->info.outputs, c, depth + 1);
868 const auto& cond = result[0].tensor();
869 if (cond.element != OnnxElement::Bool || count(cond.shape) != 1)
870 throw Failure("Loop returned invalid condition");
871 condition = integer(cond) != 0;
872 state.assign(result.begin() + 1, result.end());
873 }
874 return state;
875 }
876 if (n.op == "RandomUniformLike" || n.op == "RandomNormalLike") {
877 const auto& x = in.at(0).tensor();
878 if (attr(n, "dtype", static_cast<int>(x.element)) != 1)
879 throw Failure("Random output requires FP32", DiagnosticCode::Unsupported);
880 uint64_t seed = c.seed;
881 for (unsigned char ch : n.name) seed = (seed ^ ch) * 1099511628211ull;
882 if (n.attrs.contains("seed")) {
883 uint32_t bits;
884 std::memcpy(&bits, &n.attrs.at("seed").real, 4);
885 seed ^= bits;
886 }
887 auto [stream, inserted] = c.randomStreams.try_emplace(&n, static_cast<uint32_t>(seed ^ (seed >> 32)));
888 auto& rng = stream->second;
889 auto uniform = [&]() { return (double(rng()) + .5) / 4294967296.; };
890 std::vector<float> v(count(x.shape));
891 const float low = n.attrs.contains("low") ? n.attrs.at("low").real : 0,
892 high = n.attrs.contains("high") ? n.attrs.at("high").real : 1,
893 mean = n.attrs.contains("mean") ? n.attrs.at("mean").real : 0,
894 scale = n.attrs.contains("scale") ? n.attrs.at("scale").real : 1;
895 if (!std::isfinite(low) || !std::isfinite(high) || !std::isfinite(mean) || !std::isfinite(scale) ||
896 high < low || scale < 0)
897 throw Failure("Invalid random distribution parameters");
898 for (auto& f : v)
899 f = n.op == "RandomUniformLike" ? static_cast<float>(low + (high - low) * uniform())
900 : static_cast<float>(mean + scale * std::sqrt(-2 * std::log(uniform())) *
901 std::cos(6.283185307179586 * uniform()));
902 return {hold(make(OnnxElement::Float32, x.shape, v), c)};
903 }
904 std::vector<const RuntimeTensor*> tensors;
905 for (const auto& v : in) tensors.push_back(v.borrowed || v.owned ? &v.tensor() : nullptr);
906 auto outputs = execute(n, tensors, c.compute);
907 std::vector<Value> result;
908 for (auto& t : outputs) result.push_back(hold(std::move(t), c));
909 return result;
910}
911std::vector<Value> run(const ModelData& g, Values values, const std::vector<std::string>& requested, Context& c,
912 size_t depth) {
913 if (depth > 16) throw Failure("Execution graph nesting limit exceeded");
914 for (const auto& [name, t] : g.constants)
915 if (!g.inputs.contains(name) || !values.contains(name)) values[name] = {&t, {}, {}};
916 auto order = plan(g, requested);
917 std::unordered_map<std::string, size_t> uses;
918 for (auto i : order)
919 for (const auto& dep : dependencies(g.nodes[i])) ++uses[dep];
920 for (const auto& name : requested) ++uses[name];
921 for (auto i : order) {
922 const auto& n = g.nodes[i];
923 if (++c.steps > 1000000) throw Failure("ONNX execution step budget exceeded");
924 try {
925 std::vector<Value> in;
926 for (const auto& name : n.inputs) {
927 if (name.empty()) {
928 in.push_back({});
929 continue;
930 }
931 auto it = values.find(name);
932 if (it == values.end()) throw Failure("Missing feed or lexical capture: " + name);
933 in.push_back(it->second);
934 }
935 auto outputs = control(n, in, values, c, depth);
936 if (outputs.size() < n.outputs.size()) throw Failure("Operator output arity mismatch");
937 for (size_t j = 0; j < n.outputs.size(); ++j)
938 if (!n.outputs[j].empty() && uses[n.outputs[j]]) values[n.outputs[j]] = std::move(outputs[j]);
939 for (const auto& dep : dependencies(n))
940 if (--uses[dep] == 0) values.erase(dep);
941 } catch (const Failure& e) {
942 throw Failure(n.name + ": " + e.what(), e.code, e.path.empty() ? n.name : e.path);
943 }
944 }
945 std::vector<Value> out;
946 for (const auto& name : requested) {
947 auto it = values.find(name);
948 if (it == values.end()) throw Failure("Missing output: " + name, DiagnosticCode::NotFound);
949 out.push_back(it->second);
950 }
951 return out;
952}

954std::vector<OnnxNamedTensor> evaluate(const ModelData& model, std::span<const OnnxNamedTensor> feeds,
955 const std::vector<std::string>& requested, OnnxCompute* compute,
956 OnnxRunOptions options) {
957 Context c{compute, options.seed, options.requireFinite};
958 Values inputs;
959 for (const auto& f : feeds) inputs[f.name] = hold({f.tensor.element, f.tensor.shape, f.tensor.bytes}, c);
960 auto outputs = run(model, std::move(inputs), requested, c, 0);
961 std::vector<OnnxNamedTensor> result;
962 size_t bytes = c.liveBytes;
963 for (size_t i = 0; i < outputs.size(); ++i) {
964 const auto& t = outputs[i].tensor();
965 if (t.bytes.size() > 512u * 1024u * 1024u - bytes) throw Failure("ONNX output memory limit exceeded");
966 bytes += t.bytes.size();
967 result.push_back({requested[i], {t.element, t.shape, static_cast<std::vector<uint8_t>>(t.bytes)}});
968 }
969 return result;
970}

975std::vector<int64_t> broadcast(const std::vector<int64_t>& a, const std::vector<int64_t>& b) {
976 std::vector<int64_t> out(std::max(a.size(), b.size()), 1);
977 for (size_t i = 0; i < out.size(); ++i) {
978 const auto x = i < a.size() ? a[a.size() - 1 - i] : 1, y = i < b.size() ? b[b.size() - 1 - i] : 1;
979 if (x != y && x != 1 && y != 1) throw Failure("Broadcast dimension mismatch");
980 out[out.size() - 1 - i] = x == 1 ? y : x;
981 }
982 count(out);
983 return out;
984}
985size_t broadcastIndex(size_t i, const std::vector<int64_t>& shape, const std::vector<int64_t>& output) {
986 size_t index = 0, stride = 1;
987 for (size_t j = 0; j < output.size(); ++j) {
988 const auto coordinate = i % output[output.size() - 1 - j];
989 i /= output[output.size() - 1 - j];
990 if (j < shape.size()) {
991 const auto d = shape[shape.size() - 1 - j];
992 if (d != 1) index += coordinate * stride;
993 stride *= d;
994 }
995 }
996 return index;
997}
998std::optional<RuntimeTensor> executeShape(const Node& n, const std::vector<const RuntimeTensor*>& in) {
999 const auto& x = required(in, 0);
1000 if (n.op == "Slice") {
1001 auto starts = ints(required(in, 1)), ends = ints(required(in, 2));
1002 auto axes = in.size() > 3 && in[3] ? ints(*in[3]) : std::vector<int64_t>(starts.size());
1003 if (in.size() <= 3 || !in[3]) std::iota(axes.begin(), axes.end(), 0);
1004 auto steps = in.size() > 4 && in[4] ? ints(*in[4]) : std::vector<int64_t>(starts.size(), 1);
