40 1);

43 throw Failure("Nonfinite quantization input/range");

45 auto zeroPoint = make(OnnxElement::UInt8, {}, std::vector<uint8_t>{static_cast<uint8_t>(values[1])});

47 std::string source =

49 "x[];};layout(std430,binding=1)readonly buffer P{float p[];};layout(std430,binding=2)writeonly buffer "
50 "Y{uint y[];};void main(){uint word=gl_GlobalInvocationID.x;if(word>=" +
51 std::to_string((total + 3) / 4) +
52 "u)return;uint packed=0u;for(uint c=0u;c<4u;++c){uint j=word*4u+c;if(j<" + std::to_string(total) +
53 "u){float v=x[j];uint "
54 "q=isnan(v)?0u:uint(clamp(roundEven(v/p[0])+p[1],0.0,255.0));packed|=q<<(c*8u);}}y[word]=packed;}";
55 const std::vector<OnnxBuffer> inputs{x.bytes.buffer(), params.bytes.buffer()};
56 auto q = checked(compute->enqueue(source, inputs, total, static_cast<uint32_t>((total + 3) / 4)));
57 return {{OnnxElement::UInt8, x.shape, std::move(q)}, std::move(scale), std::move(zeroPoint)};

59 auto input = floats(x);
60 // ORT CPU DynamicQuantizeLinear ignores NaN for range estimation and stores code 0.
61 // Exported Kokoro phase graphs produce 0/0 at zero-amplitude STFT bins.
62 std::vector<size_t> nanPositions;
63 for (size_t i = 0; i < input.size(); ++i)
64 if (std::isnan(input[i])) {
65 nanPositions.push_back(i);
66 input[i] = 0;
67 }
68 auto q = checked(affine::dynamicQuantize(input));
69 for (auto i : nanPositions) q.values[i] = 0;
70 return {{OnnxElement::UInt8, x.shape, std::move(q.values)},
71 make(OnnxElement::Float32, {}, std::vector<float>{q.scale}),
72 make(OnnxElement::UInt8, {}, std::vector<uint8_t>{q.zeroPoint})};
73 }

92 }
93 if (n.op == "DequantizeLinear")

95 const auto values = floats(x);

100 type == OnnxElement::Int8));

102 }

104 }

177 w.element == OnnxElement::Int8, s, zero(in, 2, x), wz));

179 if (spatial == 2) outputShape.push_back(extent.height);

181 return {{OnnxElement::Int32, outputShape,
182 gpuConvResident(*compute, x.bytes.buffer(), w.bytes.buffer(), x.element == OnnxElement::Int8,

184 }

193 }
194 throw Failure("Unsupported quantized operation", DiagnosticCode::Unsupported);
