24std::string quoteIdentifier(const std::string& name) {
25 if (name.empty()) throw std::invalid_argument("database identifier must not be empty");
26 std::string quoted = "\"";
27 for (char c : name) {
28 if (c == '\0') throw std::invalid_argument("database identifier contains NUL");
29 quoted += c == '"' ? "\"\"" : std::string(1, c);
30 }
31 return quoted + "\"";
32}

35std::string qualifyColumn(const std::string& name) {
36 if (name.empty()) throw std::invalid_argument("database column must not be empty");
37 const auto dot = name.find('.');
38 if (dot == std::string::npos) return quoteIdentifier(name);
39 if (dot == 0 || dot + 1 >= name.size() || name.find('.', dot + 1) != std::string::npos)
40 throw std::invalid_argument("database column must be name or table.name");
41 return quoteIdentifier(name.substr(0, dot)) + "." + quoteIdentifier(name.substr(dot + 1));
42}

44bool isSimpleAlias(const std::string& alias) {
45 if (alias.empty()) return false;
46 if (!(std::isalpha(static_cast<unsigned char>(alias[0])) || alias[0] == '_')) return false;
47 for (char c : alias) {

50 return true;

54std::string validateSelectExpression(const std::string& expression) {
55 static const std::string kCountStar = "COUNT(*)";
56 std::string trimmed;
57 trimmed.reserve(expression.size());
58 for (char c : expression) {
59 if (!std::isspace(static_cast<unsigned char>(c))) trimmed.push_back(c);
60 }
61 if (trimmed == kCountStar) return kCountStar;
62
63 auto parseAgg = [&](const char* fn) -> std::string {
64 const std::string prefix = std::string(fn) + "(";
65 if (trimmed.size() < prefix.size() + 2 || trimmed.compare(0, prefix.size(), prefix) != 0 ||
66 trimmed.back() != ')')
67 return {};
68 const std::string inner = trimmed.substr(prefix.size(), trimmed.size() - prefix.size() - 1);
69 if (inner == "*") {
70 if (std::string(fn) != "COUNT") return {};
71 return std::string(fn) + "(*)";
72 }
73 return std::string(fn) + "(" + qualifyColumn(inner) + ")";
74 };
75
76 if (auto expr = parseAgg("COUNT"); !expr.empty()) return expr;
77 if (auto expr = parseAgg("SUM"); !expr.empty()) return expr;
78 if (auto expr = parseAgg("AVG"); !expr.empty()) return expr;
79 if (auto expr = parseAgg("MIN"); !expr.empty()) return expr;
80 if (auto expr = parseAgg("MAX"); !expr.empty()) return expr;
81 throw std::invalid_argument("unsupported select expression: " + expression);
82}

96std::string literal(const Value& value) {
97 if (std::holds_alternative<std::monostate>(value)) return "NULL";
98 if (auto p = std::get_if<std::int64_t>(&value)) return std::to_string(*p);
99 if (auto p = std::get_if<double>(&value)) return std::to_string(*p);
100 if (auto p = std::get_if<bool>(&value)) return *p ? "1" : "0";

243std::vector<Row> Connection::query(const std::string& sql) {
244 if (!isConnected()) throw std::runtime_error("database connection is closed");
245 Poco::Data::Statement statement(*session_);
246 statement << sql;
247 statement.execute();
248 Poco::Data::RecordSet records(statement);
249 std::vector<Row> result;
250 for (std::size_t rowIndex = 0; rowIndex < records.rowCount(); ++rowIndex) {
251 Row row;
252 for (std::size_t column = 0; column < records.columnCount(); ++column)
253 row[records.columnName(column)] = fromDynamic(records.value(column, rowIndex));
254 result.push_back(std::move(row));
255 }
256 return result;

357Query* Query::selectJson(const std::string& columnsJson) {
358 auto array = Poco::JSON::Parser().parse(columnsJson).extract<Poco::JSON::Array::Ptr>();
359 if (!array || array->empty()) throw std::invalid_argument("selectJson requires a non-empty JSON array");
360 for (std::size_t i = 0; i < array->size(); ++i) {
361 auto entry = array->get(static_cast<unsigned>(i));
362 if (entry.isString()) {
363 select(entry.convert<std::string>());
364 continue;
365 }
366 auto object = entry.extract<Poco::JSON::Object::Ptr>();
367 if (!object) throw std::invalid_argument("selectJson objects require column or expr");
368 const std::string alias = object->has("as") ? object->getValue<std::string>("as") : std::string{};
369 if (object->has("expr")) {
370 selectExpr(object->getValue<std::string>("expr"), alias);
371 continue;
372 }
373 if (object->has("column")) {
374 if (alias.empty())
375 select(object->getValue<std::string>("column"));
376 else
377 selectAs(object->getValue<std::string>("column"), alias);
378 continue;
379 }
380 throw std::invalid_argument("selectJson objects require column or expr");
381 }
382 return this;
383}

533std::string Query::selectSql(bool countOnly, int forcedLimit) const {
534 if (!havings_.empty() && groups_.empty())
535 throw std::invalid_argument("HAVING requires at least one GROUP BY column");
536
537 std::string from = quoteIdentifier(table_);
538 for (const auto& join : joins_) {
539 from += " " + join.type + " JOIN " + quoteIdentifier(join.table) + " ON " + join.left + " = " + join.right;
540 }

542 auto appendPredicates = [&](std::string& sql) {
543 for (std::size_t i = 0; i < predicates_.size(); ++i) {
544 const auto& predicate = predicates_[i];
545 if (i == 0)
546 sql += " WHERE ";
547 else
548 sql += predicate.combineWithOr ? " OR " : " AND ";
549 sql += predicate.sql;
550 }
551 };
552
553 if (countOnly) {
554 if (!groups_.empty()) {
555 std::string inner = "SELECT 1 FROM " + from;
556 appendPredicates(inner);
557 inner += " GROUP BY ";
558 for (std::size_t i = 0; i < groups_.size(); ++i) inner += (i ? "," : "") + groups_[i];
