20std::string quoteIdentifier(const std::string& name) {
21 if (name.empty()) throw std::invalid_argument("database identifier must not be empty");
22 std::string quoted = "\"";
23 for (char c : name) {
24 if (c == '\0') throw std::invalid_argument("database identifier contains NUL");
25 quoted += c == '"' ? "\"\"" : std::string(1, c);
26 }
27 return quoted + "\"";
28}

30std::string literal(const Value& value) {
31 if (std::holds_alternative<std::monostate>(value)) return "NULL";
32 if (auto p = std::get_if<std::int64_t>(&value)) return std::to_string(*p);
33 if (auto p = std::get_if<double>(&value)) return std::to_string(*p);
34 if (auto p = std::get_if<bool>(&value)) return *p ? "1" : "0";
35 std::string escaped;
36 for (char c : std::get<std::string>(value)) escaped += c == '\'' ? "''" : std::string(1, c);
37 return "'" + escaped + "'";
38}

56Value parseScalar(const std::string& json) {
57 auto wrapper = Poco::JSON::Parser().parse("[" + json + "]").extract<Poco::JSON::Array::Ptr>();
58 if (wrapper->size() != 1) throw std::invalid_argument("ORM id must be one JSON scalar");
59 auto value = wrapper->get(0);
60 if (!value.isEmpty() && !value.isBoolean() && !value.isNumeric() && !value.isString())
61 throw std::invalid_argument("ORM id must be a JSON scalar");
62 return fromDynamic(value);
63}

65std::string rowsToJson(const std::vector<Row>& source) {
66 Poco::JSON::Array rows;
67 for (const auto& row : source) {
68 Poco::JSON::Object object;
69 for (const auto& [key, value] : row) object.set(key, toDynamic(value));
70 rows.add(object);
71 }
72 std::ostringstream output;
73 Poco::JSON::Stringifier::stringify(rows, output);
74 return output.str();
75}

103std::vector<Row> Connection::query(const std::string& sql) {
104 if (!isConnected()) throw std::runtime_error("database connection is closed");
105 Poco::Data::Statement statement(*session_);
106 statement << sql;
107 statement.execute();
108 Poco::Data::RecordSet records(statement);
109 std::vector<Row> result;
110 for (std::size_t rowIndex = 0; rowIndex < records.rowCount(); ++rowIndex) {
111 Row row;
112 for (std::size_t column = 0; column < records.columnCount(); ++column)
113 row[records.columnName(column)] = fromDynamic(records.value(column, rowIndex));
114 result.push_back(std::move(row));
115 }
116 return result;
117}

121int Connection::insert(const std::string& table, const Row& values) {
122 if (values.empty()) throw std::invalid_argument("insert values must not be empty");
123 std::string columns, data;
124 for (const auto& [key, value] : values) {
125 if (!columns.empty()) {
126 columns += ',';
127 data += ',';
128 }
129 columns += quoteIdentifier(key);
130 data += literal(value);
131 }
132 return execute("INSERT INTO " + quoteIdentifier(table) + " (" + columns + ") VALUES (" + data + ")");
133}

135int Connection::update(const std::string& table, const Row& values, const std::string& whereClause) {
136 if (values.empty()) throw std::invalid_argument("update values must not be empty");
137 std::string assignments;
138 for (const auto& [key, value] : values) {
139 if (!assignments.empty()) assignments += ',';
140 assignments += quoteIdentifier(key) + '=' + literal(value);
141 }
142 return execute("UPDATE " + quoteIdentifier(table) + " SET " + assignments +
143 (whereClause.empty() ? "" : " WHERE " + whereClause));
144}

208std::string Query::selectSql(bool countOnly, int forcedLimit) const {
209 std::string sql = countOnly ? "SELECT COUNT(*) AS count FROM " : "SELECT * FROM ";
210 sql += quoteIdentifier(table_);
211 for (std::size_t i = 0; i < predicates_.size(); ++i) sql += (i == 0 ? " WHERE " : " AND ") + predicates_[i];
212 if (!countOnly && !ordering_.empty()) {
213 sql += " ORDER BY ";
214 for (std::size_t i = 0; i < ordering_.size(); ++i) sql += (i ? "," : "") + ordering_[i];
215 }
216 int effectiveLimit = forcedLimit >= 0 ? forcedLimit : limit_;
217 if (!countOnly && effectiveLimit >= 0)
218 sql += " LIMIT " + std::to_string(effectiveLimit);
219 else if (!countOnly && offset_ > 0)
220 sql += " LIMIT -1";
221 if (!countOnly && offset_ > 0) sql += " OFFSET " + std::to_string(offset_);
222 return sql;
223}

228int Query::count() {
229 auto rows = connection_->query(selectSql(true));
230 if (rows.empty()) return 0;
231 const Value& value = rows.front().at("count");
232 if (auto integer = std::get_if<std::int64_t>(&value)) return static_cast<int>(*integer);
233 if (auto number = std::get_if<double>(&value)) return static_cast<int>(*number);
234 if (auto text = std::get_if<std::string>(&value)) return std::stoi(*text);
235 throw std::runtime_error("database COUNT returned a non-numeric value");
236}
