75 editor::EditorWorkspace& workspace) const {
76 editor::EditorWorkspace candidate = workspace;
77 struct Panel {
78 const char* id;

90 for (const auto& panel : panels) {

132animation_editing::EditorResult<void> AnimationClipEditor::pointerDown(float x, float y) {
133 if (!std::isfinite(x) || !std::isfinite(y))
134 return editorError(animation_editing::EditorStatus::Rejected, "editor.animation.pointer",
135 "Pointer coordinates must be finite");
136 const auto tracks = target_.tracks();
137 if (tracks.empty()) return seekX(x);
138 const int row = static_cast<int>(std::floor(y / rowHeight_));
139 if (x < labelWidth_ && row >= 0 && static_cast<std::size_t>(row) < tracks.size())
140 return selectBone(tracks[static_cast<std::size_t>(row)].bone);
141
142 const auto keys = flattenKeys();
143 int best = -1;
144 float bestDistance = 8.0f;
145 for (int i = 0; i < static_cast<int>(keys.size()); ++i) {
146 const float dx = x - timeToX(keys[static_cast<std::size_t>(i)].time);
147 const float dy = y - (static_cast<float>(keys[static_cast<std::size_t>(i)].row) * rowHeight_ + rowHeight_ * 0.5f);
148 const float distance = std::hypot(dx, dy);
149 if (distance < bestDistance) {
150 bestDistance = distance;
151 best = i;
152 }
153 }
154 if (best >= 0) {
155 selectedKeyTrack_ = keys[static_cast<std::size_t>(best)].trackId;
156 selectedKeyIndex_ = keys[static_cast<std::size_t>(best)].keyIndex;
157 return selectBone(keys[static_cast<std::size_t>(best)].bone);
158 }
159 selectedKeyIndex_ = -1;
160 return seekX(x);
161}

187animation_editing::EditorResult<void> AnimationClipEditor::moveSelectedKey(double time) {
188 if (selectedKeyIndex_ < 0)
189 return editorError(animation_editing::EditorStatus::NotFound, "editor.animation.no-selected-key",
190 "No key is selected");
191 if (!std::isfinite(time) || time < 0.0 || time > target_.duration())
192 return editorError(animation_editing::EditorStatus::Rejected, "editor.animation.key-time",
193 "Key time must lie inside the clip duration");
194 auto tracks = target_.tracks();
195 for (auto& track : tracks) {
196 if (track.id != selectedKeyTrack_) continue;
197 if (selectedKeyIndex_ >= static_cast<int>(track.keys.size()))
198 return editorError(animation_editing::EditorStatus::NotFound, "editor.animation.key-missing",
199 "Selected key is no longer on the track");
200 track.keys[static_cast<std::size_t>(selectedKeyIndex_)].time = time;
201 return commit(target_.makeSetTrack(track), "Move key");
202 }
203 return editorError(animation_editing::EditorStatus::NotFound, "editor.animation.track-missing",
204 "Selected key track was not found");
205}

207animation_editing::EditorResult<void> AnimationClipEditor::commit(
208 animation_editing::EditorResult<animation_editing::DomainOperation> operation, std::string label) {
209 if (!operation.ok())
210 return animation_editing::EditorResult<void>::failure(operation.status());
211 editor::TransactionSpec spec;
212 spec.id = editor::TransactionId("animation.clip.tx." + std::to_string(++txSequence_));
213 spec.label = std::move(label);
214 spec.target = editor::TargetId(target_.targetId());
215 spec.baseRevision = target_.revision();
216 auto begun = transactions_.begin(std::move(spec));
217 if (!begun.ok())
218 return editorError(begun.code(), "editor.animation.begin", "Could not begin the animation clip transaction");
219 auto appended = transactions_.append(std::move(operation).takeValue());
220 if (!appended.ok()) {
221 auto discarded = transactions_.discard();
222 if (!discarded.ok()) discarded.ignore("pending animation clip transaction already inactive");
223 return animation_editing::EditorResult<void>::failure(appended.status());
224 }
225 auto committed = transactions_.commit();
226 if (!committed.ok())
227 return animation_editing::EditorResult<void>::failure(committed.status());
228 return refreshPreview();
229}

256animation_editing::EditorResult<void> AnimationClipEditor::update(double deltaSeconds) {
257 if (playing_) {
258 if (!std::isfinite(deltaSeconds) || deltaSeconds < 0.0)
259 return editorError(animation_editing::EditorStatus::Rejected, "editor.animation.dt",
260 "Playback delta must be a non-negative finite time");
261 playhead_ += deltaSeconds;
262 const double duration = target_.duration();
263 if (duration <= 0.0) {
264 playhead_ = 0.0;
265 } else if (target_.isLooping()) {
266 playhead_ = std::fmod(playhead_, duration);
267 if (playhead_ < 0.0) playhead_ += duration;
268 } else if (playhead_ >= duration) {
269 playhead_ = duration;
270 playing_ = false;
271 }
272 }
273 return refreshPreview();
274}

276animation_editing::EditorResult<void> AnimationClipEditor::refreshPreview() {
277 const auto sampled = target_.preview(playhead_, skeletonBones());
278 if (sampled.documentRevision != target_.revision()) {
279 return editorError(animation_editing::EditorStatus::Conflict, "editor.animation.stale-preview",
280 "Clip preview revision does not match the document");
281 }
282 if (sampled.status != animation_editing::EditorStatus::Applied)
283 return editorError(sampled.status, "editor.animation.preview", "Could not sample the animation clip pose");
284
285 std::vector<animation_editing::SkeletonOverlayBone> bones;
286 bones.reserve(sampled.bones.size());
287 for (const auto& sample : sampled.bones) {
288 animation_editing::SkeletonOverlayBone bone;
289 bone.id = animation_editing::StableId(sample.bone);
290 bone.parent = sample.bone == "Spine" ? animation_editing::StableId("Hips")
291 : animation_editing::StableId();
292 bone.name = sample.bone;
293 bone.position = {sample.positionX, sample.positionY, sample.positionZ};
294 bone.rotation = {sample.rotationX, sample.rotationY, sample.rotationZ, sample.rotationW};
295 bone.selected = sample.bone == selectedBone_;
296 bone.maskWeight = sample.maskWeight;
297 bones.push_back(std::move(bone));
298 }
299 animation_editing::SkeletonOverlayOptions options;
300 options.showAxes = true;
301 options.axesForSelectedOnly = true;
302 options.showConstraints = false;
303 auto overlay = overlayBuilder_.build(target_.targetId().value(), animation_editing::Revision(target_.revision()),
304 bones, options);
305 if (overlay.status != eve::editing::Status::Applied)
306 return editorError(overlay.status, "editor.animation.overlay",
307 "Could not rebuild the skeleton overlay from the sampled pose");
308 preview_ = sampled;
309 overlay_ = std::move(overlay);
310 return eve::editing::applied<void>();
311}

446std::vector<AnimationClipEditor::TimelineKey> AnimationClipEditor::flattenKeys() const {
447 std::vector<TimelineKey> keys;
448 const auto tracks = target_.tracks();
449 for (int row = 0; row < static_cast<int>(tracks.size()); ++row) {
450 const auto& track = tracks[static_cast<std::size_t>(row)];
451 for (int i = 0; i < static_cast<int>(track.keys.size()); ++i) {
452 TimelineKey key;
453 key.trackId = track.id;
454 key.bone = track.bone;
455 key.keyIndex = i;
456 key.time = track.keys[static_cast<std::size_t>(i)].time;
457 key.row = row;
458 keys.push_back(std::move(key));
459 }
460 }
461 return keys;
