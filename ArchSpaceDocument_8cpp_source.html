35double polygonArea(const std::vector<Vec2>& poly) {
36 if (poly.size() < 3) return 0.0;
37 double acc = 0.0;
38 for (std::size_t i = 0; i < poly.size(); ++i) {
39 const Vec2& a = poly[i];
40 const Vec2& b = poly[(i + 1) % poly.size()];
41 acc += a.x * b.z - b.x * a.z;
42 }
43 return 0.5 * acc;
44}

46bool polygonValid(const std::vector<Vec2>& poly) {
47 if (poly.size() < 3 || poly.size() > 256) return false;
48 for (const auto& p : poly)
49 if (!isFinite(p)) return false;
50 return std::abs(polygonArea(poly)) > kEps;

53void appendBox(MeshBake& bake, const Vec3& a, const Vec3& b, double thickness, double height, const std::string& id) {
54 Vec2 dir{b.x - a.x, b.z - a.z};
55 const double len = std::sqrt(dir.x * dir.x + dir.z * dir.z);
56 if (len <= kEps || thickness <= kEps || height <= kEps) return;
57 dir.x /= len;
58 dir.z /= len;
59 const Vec2 n{-dir.z * thickness * 0.5, dir.x * thickness * 0.5};
60 const std::uint32_t base = static_cast<std::uint32_t>(bake.positions.size());
61 const Vec3 corners[8] = {
62 {a.x + n.x, a.y, a.z + n.z}, {a.x - n.x, a.y, a.z - n.z},
63 {b.x - n.x, a.y, b.z - n.z}, {b.x + n.x, a.y, b.z + n.z},
64 {a.x + n.x, a.y + height, a.z + n.z}, {a.x - n.x, a.y + height, a.z - n.z},
65 {b.x - n.x, a.y + height, b.z - n.z}, {b.x + n.x, a.y + height, b.z + n.z},
66 };
67 for (const auto& c : corners) bake.positions.push_back(c);
68 static constexpr std::uint32_t faces[12][3] = {{0, 1, 2}, {0, 2, 3}, {4, 6, 5}, {4, 7, 6}, {0, 3, 7}, {0, 7, 4},
69 {1, 5, 6}, {1, 6, 2}, {0, 4, 5}, {0, 5, 1}, {3, 2, 6}, {3, 6, 7}};
70 for (const auto& f : faces) {
71 bake.indices.push_back(base + f[0]);
72 bake.indices.push_back(base + f[1]);
73 bake.indices.push_back(base + f[2]);
74 }
75 bake.primitiveIds.push_back(id);
76}

78void appendSlab(MeshBake& bake, const std::vector<Vec2>& poly, double y, double thickness, const std::string& id) {
79 if (poly.size() < 3 || thickness <= kEps) return;
80 const std::uint32_t base = static_cast<std::uint32_t>(bake.positions.size());
81 for (const auto& p : poly) bake.positions.push_back(Vec3{p.x, y, p.z});
82 for (const auto& p : poly) bake.positions.push_back(Vec3{p.x, y + thickness, p.z});
83 const std::uint32_t n = static_cast<std::uint32_t>(poly.size());
84 for (std::uint32_t i = 1; i + 1 < n; ++i) {
85 bake.indices.push_back(base);
86 bake.indices.push_back(base + i);
87 bake.indices.push_back(base + i + 1);
88 bake.indices.push_back(base + n);
89 bake.indices.push_back(base + n + i + 1);
90 bake.indices.push_back(base + n + i);
91 }
92 bake.primitiveIds.push_back(id);
93}

124bool Document::validateNode(const Node& node) const {
125 if (node.id.empty() || node.id.size() > 128 || node.name.size() > 256) return false;
126 if (node.kind == NodeKind::Site) {
127 if (!node.parentId.empty()) return false;
128 } else if (!canParent(node.kind, node.parentId)) {
129 return false;
130 }
131 switch (node.kind) {
132 case NodeKind::Site:
133 case NodeKind::Building: return true;
134 case NodeKind::Level: return isFinite(node.elevation) && isFinite(node.height) && node.height > kEps;
135 case NodeKind::Wall:
136 return isFinite(node.start) && isFinite(node.end) && isFinite(node.thickness) && isFinite(node.height) &&
137 node.thickness > kEps && node.height > kEps && distance(node.start, node.end) > kEps;
138 case NodeKind::Slab:
139 return polygonValid(node.polygon) && isFinite(node.slabThickness) && node.slabThickness > kEps;
140 case NodeKind::Zone: return polygonValid(node.polygon);
141 case NodeKind::Item:
142 return isFinite(node.position) && isFinite(node.yawDegrees) && node.catalogId.size() <= 256;
143 case NodeKind::Opening:
144 return isFinite(node.t) && node.t >= 0.0 && node.t <= 1.0 && isFinite(node.width) && node.width > kEps &&
145 isFinite(node.openingHeight) && node.openingHeight > kEps && isFinite(node.sill) && node.sill >= 0.0;
146 }
147 return false;
148}

198eve::Result<void> Document::eraseCascade(const std::string& id) {
199 if (!find(id)) return fail<void>(eve::DiagnosticCode::NotFound, "ArchSpace node is missing", "id");
200 std::vector<std::string> order;
201 collectDescendants(id, order);
202 std::reverse(order.begin(), order.end());
203 order.push_back(id);
204 for (const auto& victim : order) {
205 unlinkFromParent(victim);
206 nodes_.erase(victim);
207 }
208 if (rootId_ == id) rootId_.clear();
209 return ok();

348std::vector<std::string> Document::diagnostics() const {
349 std::vector<std::string> out;
350 if (rootId_.empty() && !nodes_.empty()) out.push_back("document has nodes but no root site");
351 if (!rootId_.empty() && !find(rootId_)) out.push_back("root id is missing from node map");
352 std::unordered_map<std::string, std::size_t> referenced;
353 for (const auto& [id, node] : nodes_) {
354 if (!validateNode(node)) out.push_back("node failed validation: " + id);
355 for (const auto& child : node.children) {
356 ++referenced[child];
357 const Node* childNode = find(child);
358 if (!childNode)
359 out.push_back("missing child " + child + " under " + id);
360 else if (childNode->parentId != id)
361 out.push_back("child parent mismatch: " + child);
362 }
363 if (node.kind != NodeKind::Site) {
364 const Node* parent = find(node.parentId);
365 if (!parent)
366 out.push_back("missing parent for " + id);
367 else if (!canParent(node.kind, node.parentId))
368 out.push_back("illegal parent kind for " + id);
369 }
370 }
371 for (const auto& [id, node] : nodes_) {
372 if (node.kind == NodeKind::Site) continue;
373 if (referenced.find(id) == referenced.end()) out.push_back("orphaned node: " + id);
374 }
375 return out;
376}

378MeshBake Document::bakeMesh() const {
379 MeshBake bake;
380
381 for (const auto& [id, node] : nodes_) {
382 if (node.kind == NodeKind::Wall) {
383 const Node* level = find(node.parentId);
384 const double y = level ? level->elevation : 0.0;
385 std::vector<const Node*> openings;
386 openings.reserve(node.children.size());
387 for (const auto& childId : node.children) {
388 const Node* child = find(childId);
389 if (child && child->kind == NodeKind::Opening) openings.push_back(child);
390 }
391 appendWallWithOpenings(bake, node, y, openings);
392 } else if (node.kind == NodeKind::Slab) {
393 const Node* level = find(node.parentId);
394 const double y = level ? level->elevation : 0.0;
395 appendSlab(bake, node.polygon, y, node.slabThickness, id);
396 } else if (node.kind == NodeKind::Item) {
397 appendCatalogItem(bake, id, node.catalogId, node.position, node.yawDegrees);
398 }
399 }
400 return bake;
