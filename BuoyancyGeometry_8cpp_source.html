24Result<ImmersionSample> BuoyancyGeometry::sphere3D(ForceVector center, double radius, double surfaceY) {
25 if (!inputs(center, surfaceY) || !isPositive(radius)) return invalid<ImmersionSample>();
26 const double total = (4 * pi / 3) * radius * radius * radius;
27 const double h = std::clamp((surfaceY - center.y) / radius + 1, 0.0, 2.0);
28 // h is cap height/radius. Integrating pi*(r^2-y^2) gives volume and first moment.
29 const double fraction = h * h * (3 - h) / 4;
30 const double offset = -radius * (3 * (2 - h) * (2 - h) / (4 * (3 - h)));
31 return finish(center, total, fraction, offset);
32}
33Result<ImmersionSample> BuoyancyGeometry::circle2D(ForceVector center, double radius, double thickness,
34 double surfaceY) {
35 if (!inputs(center, surfaceY) || center.z != 0 || !isPositive(radius) || !isPositive(thickness))
36 return invalid<ImmersionSample>();
37 const double total = pi * radius * radius * thickness;
38 const double h = std::clamp((surfaceY - center.y) / radius + 1, 0.0, 2.0);
39 if (h == 0 || h == 2) return finish(center, total, h / 2, 0);
40 // Evaluate the smaller segment to avoid cancellation near full immersion.
41 const double small = std::min(h, 2 - h);
42 const double theta = 2 * std::asin(std::sqrt(small / 2));
43 const double sine = std::sqrt(small * (2 - small));
44 double area = theta - (1 - small) * sine;
45 if (theta < .01) {
46 const double q = theta * theta;
47 area = (2.0 / 3) * theta * q * (1 - q / 5 + 2 * q * q / 105 - q * q * q / 945);
48 }
49 const double wetArea = h <= 1 ? area : pi - area;
50 if (wetArea == 0) return finish(center, total, 0, 0);
51 const double offset = -radius * ((2.0 / 3) * sine * sine * sine / wetArea);
52 return finish(center, total, wetArea / pi, offset);
53}
54Result<ImmersionSample> BuoyancyGeometry::box3D(ForceVector center, ForceVector size, double surfaceY) {
55 if (!inputs(center, surfaceY) || !isPositive(size.x) || !isPositive(size.y) || !isPositive(size.z))
56 return invalid<ImmersionSample>();
57 const double total = size.x * size.y * size.z;
58 const double fraction = std::clamp((surfaceY - center.y) / size.y + .5, 0.0, 1.0);
59 return finish(center, total, fraction, -size.y * .5 * (1 - fraction));
60}
