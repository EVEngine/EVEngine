10JointResult createJoint(WorldBackend& world, Body3D& a, Body3D& b, JointKind3D kind, JPH::RVec3Arg anchorA,
11 JPH::RVec3Arg anchorB, JPH::Vec3Arg axis, JPH::Vec3Arg wheelAxis, float length, bool collide) {
12 if (!world.isValid())
13 return rejected<std::unique_ptr<IJointBackend3D>>("World expired", eve::DiagnosticCode::StaleHandle);
14 if (&a == &b || !a.isValid() || !b.isValid() || a.getWorld() != world.owner || b.getWorld() != world.owner ||
15 !isFinite(float(anchorA.GetX()), float(anchorA.GetY()), float(anchorA.GetZ())) ||
16 !isFinite(float(anchorB.GetX()), float(anchorB.GetY()), float(anchorB.GetZ())) ||
17 !std::isfinite(axis.LengthSq()) || axis.LengthSq() < 1.e-12f || !std::isfinite(length) || length < 0 ||
18 !std::isfinite(wheelAxis.LengthSq()) || wheelAxis.LengthSq() < 1.e-12f)
19 return rejected<std::unique_ptr<IJointBackend3D>>("Invalid joint definition",
20 eve::DiagnosticCode::InvalidArgument);
21 auto record = std::make_shared<JointState>();
22 for (const auto& [id, body] : world.state->bodies) {
23 if (body->owner == &a) record->a = body;
24 if (body->owner == &b) record->b = body;
25 }
26 auto ba = record->a.lock(), bb = record->b.lock();
27 if (!ba || !bb)
28 return rejected<std::unique_ptr<IJointBackend3D>>("Bodies are not owned by this provider",
29 eve::DiagnosticCode::InvalidArgument);
30 auto& na = *world.state->system.GetBodyLockInterfaceNoLock().TryGetBody(ba->id);
31 auto& nb = *world.state->system.GetBodyLockInterfaceNoLock().TryGetBody(bb->id);
32 record->world = world.state;
33 record->kind = kind;
34 record->length = length;
35 record->collide = true;
36 const auto direction = axis.Normalized();
37 // Portable frames keep the existing axis conventions used by snapshot reconstruction.
38 auto portableFrame = [&](const JPH::Body& body, JPH::RVec3Arg anchor, JPH::Vec3Arg localAxis,
39 JPH::Vec3Arg worldAxis) {
40 auto p = body.GetWorldTransform().InversedRotationTranslation() * anchor;
41 auto q = JPH::Quat::sFromTo(localAxis, body.GetRotation().Conjugated() * worldAxis);
42 if (kind == JointKind3D::Weld) q = body.GetRotation().Conjugated();
43 if (kind == JointKind3D::Distance || kind == JointKind3D::Motor || kind == JointKind3D::Filter)
44 q = JPH::Quat::sIdentity();
45 return std::array<float, 7>{float(p.GetX()), float(p.GetY()), float(p.GetZ()), q.GetX(),
46 q.GetY(), q.GetZ(), q.GetW()};
47 };
48 record->frames.bodyA = portableFrame(
49 na, anchorA,
50 kind == JointKind3D::Prismatic || kind == JointKind3D::Wheel ? JPH::Vec3::sAxisX() : JPH::Vec3::sAxisZ(),
51 direction);
52 record->frames.bodyB =
53 portableFrame(nb, anchorB, kind == JointKind3D::Prismatic ? JPH::Vec3::sAxisX() : JPH::Vec3::sAxisZ(),
54 kind == JointKind3D::Wheel ? wheelAxis.Normalized() : direction);
55 if (kind == JointKind3D::Distance) {
56 JPH::DistanceConstraintSettings settings;
57 settings.mPoint1 = anchorA;
58 settings.mPoint2 = anchorB;
59 settings.mMinDistance = length;
60 settings.mMaxDistance = length;
61 record->native = createDistanceConstraint(na, nb, settings);
62 } else if (kind != JointKind3D::Filter) {
63 JPH::SixDOFConstraintSettings settings;
64 using A = JPH::SixDOFConstraintSettings::EAxis;
65 for (auto& motor : settings.mMotorSettings) {
66 motor.mSpringSettings = {JPH::ESpringMode::FrequencyAndDamping, 0, 0};
67 if (kind == JointKind3D::Motor) {
68 motor.SetForceLimit(0);
69 motor.SetTorqueLimit(0);
70 }
71 }
72 settings.mPosition1 = anchorA;
73 settings.mPosition2 = anchorB;
74 settings.mAxisX1 = settings.mAxisX2 = direction;
75 settings.mAxisY1 = settings.mAxisY2 = direction.GetNormalizedPerpendicular();
76 if (kind == JointKind3D::Weld || kind == JointKind3D::Motor) {
77 settings.mAxisX1 = settings.mAxisX2 = JPH::Vec3::sAxisX();
78 settings.mAxisY1 = settings.mAxisY2 = JPH::Vec3::sAxisY();
79 }
80 if (kind == JointKind3D::Wheel) {
81 auto spin = wheelAxis.Normalized();
82 if (std::abs(spin.Dot(direction)) > .999f)
83 return rejected<std::unique_ptr<IJointBackend3D>>("Wheel axes are parallel",
84 eve::DiagnosticCode::InvalidArgument);
85 settings.mAxisX1 = settings.mAxisX2 = (spin - direction * spin.Dot(direction)).Normalized();
86 settings.mAxisY1 = settings.mAxisY2 = direction;
87 settings.mSwingType = JPH::ESwingType::Pyramid;
88 }
89 for (int i = 0; i < 6; ++i) settings.MakeFixedAxis(static_cast<A>(i));
90 if (kind == JointKind3D::Revolute) settings.MakeFreeAxis(A::RotationX);
91 if (kind == JointKind3D::Prismatic) settings.MakeFreeAxis(A::TranslationX);
92 if (kind == JointKind3D::Spherical)
93 for (int i = 3; i < 6; ++i) settings.MakeFreeAxis(static_cast<A>(i));
94 if (kind == JointKind3D::Motor)
95 for (int i = 0; i < 6; ++i) settings.MakeFreeAxis(static_cast<A>(i));
96 if (kind == JointKind3D::Parallel) {
97 for (int i = 0; i < 3; ++i) settings.MakeFreeAxis(static_cast<A>(i));
98 settings.MakeFreeAxis(A::RotationX);
99 }
100 if (kind == JointKind3D::Wheel) {
101 settings.MakeFreeAxis(A::TranslationY);
102 settings.MakeFreeAxis(A::RotationX);
103 settings.MakeFreeAxis(A::RotationY);
104 }
105 record->native = settings.Create(na, nb);
106 if (kind == JointKind3D::Revolute || kind == JointKind3D::Prismatic || kind == JointKind3D::Motor) {
107 for (int i = 0; i < 6; ++i) settings.MakeFreeAxis(static_cast<A>(i));
108 record->velocity = static_cast<JPH::SixDOFConstraint*>(settings.Create(na, nb));
109 }
110 }
111 if (record->native) world.state->system.AddConstraint(record->native);
112 if (record->velocity) world.state->system.AddConstraint(record->velocity);
113 auto provider = std::make_unique<JointBackend>(record);
114 auto configured = provider->setCollideConnected(collide);
115 if (!configured) return JointResult::failure(configured.status());
116 std::erase_if(world.state->joints, [](const auto& weak) { return weak.expired(); });
117 world.state->joints.push_back(record);
118 if (kind == JointKind3D::Wheel) {
119 auto defaults = provider->setWheelSuspension(true, 1, .7f);
120 if (!defaults) return JointResult::failure(defaults.status());
121 }
122 if (kind == JointKind3D::Parallel) {
123 auto defaults = provider->setParallelSpring(1, 1, FLT_MAX);
124 if (!defaults) return JointResult::failure(defaults.status());
125 }
126 return JointResult::success(std::move(provider));
127}
