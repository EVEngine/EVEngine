227void ProcgenScriptDocumentTarget::applySpec(ProcgenScriptModuleSpec spec) {
228 EditorValue::Object next;
229 for (const auto& param : spec.params) {
230 const auto existing = values_.find(param.key);
231 if (existing != values_.end()) {
232 editing::PropertyDescriptor descriptor;
233 descriptor.path = editing::PropertyPath("param." + param.key);
234 descriptor.type = propertyType(param.kind);
235 descriptor.defaultValue = defaultValue(param);
236 descriptor.enumItems = param.choices;
237 if (param.hasMinimum) descriptor.numeric.minimum = param.minimum;
238 if (param.hasMaximum) descriptor.numeric.maximum = param.maximum;
239 if (editing::validatePropertyValue(descriptor, existing->second).ok()) {
240 next[param.key] = existing->second;
241 continue;
242 }
243 }
244 next[param.key] = defaultValue(param);
245 }
246 uri_ = std::move(spec.uri);
247 moduleId_ = std::move(spec.id);
248 displayName_ = std::move(spec.displayName);
249 kind_ = std::move(spec.kind);
250 params_ = std::move(spec.params);
251 values_ = std::move(next);
252}

254EditorResult<ProcgenScriptModuleSpec> ProcgenScriptDocumentTarget::parseSpec(std::string uri, std::string id,
255 std::string displayName, std::string kind,
256 const EditorValue& schema) {
257 if (id.empty())
258 return fail<ProcgenScriptModuleSpec>(EditorStatus::Rejected, "editor.procgen-script.id",
259 "Generator module id must not be empty");
260 if (kind.empty()) kind = "points";
261 if (kind != "points")
262 return fail<ProcgenScriptModuleSpec>(EditorStatus::Unsupported, "editor.procgen-script.kind",
263 "Only kind=points generators are supported in this editor version");
264 const auto* rows = schema.getIf<EditorValue::Array>();
265 if (!rows)
266 return fail<ProcgenScriptModuleSpec>(EditorStatus::Rejected, "editor.procgen-script.schema",
267 "Generator schema must be an array");
268 if (rows->size() > 128)
269 return fail<ProcgenScriptModuleSpec>(EditorStatus::Rejected, "editor.procgen-script.schema-size",
270 "Generator schema exceeds 128 parameters");
271
272 ProcgenScriptModuleSpec spec;
273 spec.uri = std::move(uri);
274 spec.id = std::move(id);
275 spec.displayName = displayName.empty() ? spec.id : std::move(displayName);
276 spec.kind = std::move(kind);
277 spec.params.reserve(rows->size());
278 for (const auto& rowValue : *rows) {
279 const auto* row = rowValue.getIf<EditorValue::Object>();
280 if (!row)
281 return fail<ProcgenScriptModuleSpec>(EditorStatus::Rejected, "editor.procgen-script.param",
282 "Each schema entry must be an object");
283 procgen::ParamDescriptor param;
284 param.key = requireString(rowValue, "key");
285 if (param.key.empty())
286 return fail<ProcgenScriptModuleSpec>(EditorStatus::Rejected, "editor.procgen-script.param-key",
287 "Parameter key must not be empty");
288 for (const auto& existing : spec.params)
289 if (existing.key == param.key)
290 return fail<ProcgenScriptModuleSpec>(EditorStatus::Rejected, "editor.procgen-script.param-dup",
291 "Duplicate parameter key: " + param.key);
292 param.displayName = requireString(rowValue, "displayName", requireString(rowValue, "label", param.key));
293 param.description = requireString(rowValue, "description");
294 param.category = requireString(rowValue, "category", "Generator");
295 param.kind = parseKind(requireString(rowValue, "kind", "string"));
296 param.advanced = false;
297 if (const auto* advanced = field(rowValue, "advanced"))
298 if (const auto* flag = advanced->getIf<bool>()) param.advanced = *flag;
299
300 double numeric = 0.0;
301 if (readNumber(rowValue, "min", numeric)) {
302 param.hasMinimum = true;
303 param.minimum = numeric;
304 }
305 if (readNumber(rowValue, "max", numeric)) {
306 param.hasMaximum = true;
307 param.maximum = numeric;
308 }
309 if (readNumber(rowValue, "step", numeric)) param.step = numeric;
310
311 if (const EditorValue* choices = field(rowValue, "choices")) {
312 if (const auto* array = choices->getIf<EditorValue::Array>()) {
313 for (const auto& choice : *array) {
314 const auto* text = choice.getIf<std::string>();
315 if (!text)
316 return fail<ProcgenScriptModuleSpec>(EditorStatus::Rejected,
317 "editor.procgen-script.choice",
318 "Choice values must be strings");
319 param.choices.push_back(*text);
320 }
321 }
322 }
323
324 EditorValue def;
325 if (const EditorValue* defaultField = field(rowValue, "defaultValue")) {
326 def = *defaultField;
327 } else if (const EditorValue* defaultField = field(rowValue, "default")) {
328 def = *defaultField;
329 } else {
330 switch (param.kind) {
331 case procgen::ParamKind::Integer:
332 def = std::int64_t{0};
333 break;
334 case procgen::ParamKind::Float:
335 def = 0.0;
336 break;
337 case procgen::ParamKind::Boolean:
338 def = false;
339 break;
340 case procgen::ParamKind::String:
341 case procgen::ParamKind::Choice:
342 def = param.choices.empty() ? std::string{} : param.choices.front();
343 break;
344 }
345 }
346 param.defaultValue = encodeDefault(param, def);
347 spec.params.push_back(std::move(param));
348 }
349 return eve::editing::applied(std::move(spec));
350}

352EditorResult<editing::DomainOperation> ProcgenScriptDocumentTarget::makeLoadModule(
353 ProcgenScriptModuleSpec spec) const {
354 if (spec.id.empty())
355 return fail<editing::DomainOperation>(EditorStatus::Rejected, "editor.procgen-script.id",
356 "Generator module id must not be empty");
357 if (spec.kind != "points")
358 return fail<editing::DomainOperation>(EditorStatus::Unsupported, "editor.procgen-script.kind",
359 "Only kind=points generators are supported in this editor version");
360 if (spec.params.size() > 128)
361 return fail<editing::DomainOperation>(EditorStatus::Rejected, "editor.procgen-script.schema-size",
362 "Generator schema exceeds 128 parameters");
363
364 ProcgenScriptDocumentTarget candidate = *this;
365 candidate.applySpec(std::move(spec));
366 if (hasErrors(candidate.validate()))
367 return fail<editing::DomainOperation>(EditorStatus::Rejected, "editor.procgen-script.invalid",
368 "Generator module validation failed");
369
370 editing::DomainOperation operation;
371 operation.type = "procgen.script.replace.v1";
372 operation.inverseType = operation.type;
373 operation.target = editing::TargetId(id_);
374 operation.payload = candidate.contentValue();
375 operation.inverse = contentValue();
376 operation.hasInverse = true;
377 operation.mergeKey = "procgen-script:" + id_ + ":module";
378 return eve::editing::applied(std::move(operation));

417std::vector<EditorDiagnostic> ProcgenScriptDocumentTarget::validate() const {
418 std::vector<EditorDiagnostic> diagnostics;
419 if (!kind_.empty() && kind_ != "points")
420 diagnostics.push_back(diagnostic("editor.procgen-script.kind", editing::DiagnosticSeverity::Error,
421 "Only kind=points generators are supported"));
422 const auto propertySchema = schema({});
423 for (const auto& param : params_) {

425 const auto descriptor = propertySchema.find(editing::PropertyPath("param." + param.key));

437EditorResult<void> ProcgenScriptDocumentTarget::applyDomainOperation(const editing::DomainOperation& operation) {
438 if (operation.target != editing::TargetId(id_) || operation.type != "procgen.script.replace.v1")
439 return fail<void>(EditorStatus::Rejected, "editor.procgen-script.operation",
440 "Generator operation is invalid");
441 const auto parsed = parseSpec(requireString(operation.payload, "uri"), requireString(operation.payload, "id"),
442 requireString(operation.payload, "displayName"),
443 requireString(operation.payload, "kind", "points"),
444 field(operation.payload, "schema") ? *field(operation.payload, "schema")
445 : EditorValue{});
446 if (!parsed.ok()) return EditorResult<void>::failure(parsed.status());
447 const EditorValue* valuesField = field(operation.payload, "values");
448 const auto* values = valuesField ? valuesField->getIf<EditorValue::Object>() : nullptr;
449 if (!values || values->size() > 128)
450 return fail<void>(EditorStatus::Rejected, "editor.procgen-script.payload",
451 "Generator payload is invalid");
452
453 ProcgenScriptDocumentTarget candidate = *this;
454 candidate.applySpec(parsed.value());
455 candidate.values_ = *values;
456 if (hasErrors(candidate.validate()))
457 return fail<void>(EditorStatus::Rejected, "editor.procgen-script.invalid",
458 "Generator validation failed");
459 *this = std::move(candidate);
460 ++revision_;
461 dirty_.include(0, 0);
462 return eve::editing::applied<void>();
463}
