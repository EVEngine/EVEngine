31eve::Result<std::string> canonicalJson(const SkillDefinition& def) {
32 eve::Value::Object object;
33 object.emplace("id", eve::Value(def.id));
34 object.emplace("cooldown", eve::Value(static_cast<double>(def.cooldown)));
35 object.emplace("castTime", eve::Value(static_cast<double>(def.castTime)));
36 object.emplace("targetType", eve::Value(def.targetType));
37
38 auto condition = encodeSkillCondition(def.castCondition);
39 if (!condition) return eve::Result<std::string>::failure(condition.status());
40 object.emplace("castCondition", std::move(condition).takeValue());
41
42 eve::Value::Array effects;
43 for (const auto& effect : def.grantedEffects) effects.emplace_back(effect);
44 object.emplace("grantedEffects", eve::Value(std::move(effects)));
45
46 eve::Value::Array tags;
47 for (const auto& tag : def.tags) tags.emplace_back(tag);
48 object.emplace("tags", eve::Value(std::move(tags)));
49
50 eve::Value::Object extra;
51 for (const auto& [key, value] : def.extra) extra.emplace(key, eve::Value(value));
52 object.emplace("extra", eve::Value(std::move(extra)));
53
54 eve::Value::Array costs;
55 if (def.cost) {
56 for (const auto& item : def.cost->items()) {
57 costs.emplace_back(eve::Value(eve::Value::Object{
58 {"attribute", eve::Value(item.resource.value())},
59 {"amount", eve::Value(item.amount.value())},
60 }));
61 }
62 }
63 object.emplace("costs", eve::Value(std::move(costs)));
64 return eve::Value(std::move(object)).toJson();
65}

80void SkillRegistry::registerSkill(const SkillDefinition& def) {
81 if (def.id.empty()) return;
82 auto json = canonicalJson(def);
83 if (!json) {
84 json.ignore("legacy skill registration cannot serialize invalid definition");
85 return;
86 }
87 auto& registry = canonicalSkillRegistry();
88 auto existing = registry.resolve("rpg.skill", def.id);
89 eve::Result<eve::definition::DefinitionHandle> result =
90 existing.ok() ? registry.replace("rpg.skill", def.id, 1, std::move(json).takeValue())
91 : existing.status().code() == eve::StatusCode::NotFound
92 ? registry.insert("rpg.skill", def.id, 1, std::move(json).takeValue())
93 : eve::Result<eve::definition::DefinitionHandle>::failure(existing.status());
94 if (!result) return;
95 table()[def.id] = def;
96}

126SkillDefinition parseSkillObject(Value o, std::string* error) {
127 SkillDefinition def;
128 if (!o.isObject()) return def;
129 def.id = o.getString("id");
130 def.cooldown = o.getFloat("cooldown", 0.f);
131 def.castTime = o.getFloat("castTime", 0.f);
132 def.targetType = o.getString("targetType", "self");
133 def.grantedEffects = o.getStringArray("grantedEffects");
134 def.tags = o.getStringArray("tags");
135
136 const Value costs = o.get("costs");
137 std::vector<eve::resource::ResourceCost> costItems;
138 for (size_t i = 0; i < costs.size(); ++i) {
139 const Value co = costs.at(i);
140 const std::string resource = co.getString("attribute");
141 const double amount = co.getDouble("amount", 0.0);
142 // Keep the legacy JSON shape, but do not silently truncate a value
143 // into the canonical integral resource Amount type.
144 if (resource.empty()) continue;
145 if (!std::isfinite(amount) || amount <= 0.0 || std::floor(amount) != amount || amount >= std::ldexp(1.0, 63)) {
146 if (error) *error = "skill cost must be a finite positive integer: " + resource;
147 return {};
148 }
149 auto item = eve::resource::ResourceCost::create(resource, static_cast<std::int64_t>(amount));
150 if (!item) {
151 if (error) *error = item.status().describe();
152 return {};
153 }
154 costItems.push_back(std::move(item).takeValue());
155 }
156 if (!costItems.empty()) {
157 auto canonical = eve::resource::CostSpec::create(std::move(costItems));
158 if (!canonical) {
159 if (error) *error = canonical.status().describe();
160 return {};
161 }
162 def.cost = std::move(canonical).takeValue();
163 }
164
165 def.extra = o.getStringMap("extra");
166 return def;
167}

171int SkillRegistry::loadFromJson(const std::string& json, std::string* error) {
172 std::string err;
173 const eve::json::Document doc = eve::json::Document::parse(json, &err);
174 if (!doc.valid()) {
175 if (error) *error = err.empty() ? "invalid json" : err;
176 return 0;
177 }
178
179 const Value root = doc.root();
180 int n = 0;
181 if (root.isArray()) {
182 for (size_t i = 0; i < root.size(); ++i) {
183 SkillDefinition def = parseSkillObject(root.at(i), error);
184 if (def.id.empty()) continue;
185 registerSkill(def);
186 ++n;
187 }
188 } else if (root.isObject()) {
189 SkillDefinition def = parseSkillObject(root, error);
190 if (!def.id.empty()) {
191 registerSkill(def);
192 ++n;
193 }
194 }
195 return n;
196}

230eve::Result<action::ActionDefinition> SkillActionAdapter::makeDefinition(const SkillDefinition& skill) {
231 auto id = skillActionId(skill);
232 if (!id) return eve::Result<action::ActionDefinition>::failure(id.status());
233 auto windup = seconds(skill.castTime, "skill.castTime");
234 if (!windup) return eve::Result<action::ActionDefinition>::failure(windup.status());
235 auto cooldown = seconds(skill.cooldown, "skill.cooldown");
236 if (!cooldown) return eve::Result<action::ActionDefinition>::failure(cooldown.status());
237
238 action::ActionDefinition definition;
239 definition.id = std::move(id).takeValue();
240 definition.timing.windup = std::move(windup).takeValue();
241 definition.condition = skill.castCondition;
242 definition.targetingMode =
243 skill.targetType == "self" ? action::TargetingMode::None : action::TargetingMode::Explicit;
244 definition.cost = skill.cost;
245 definition.effectIds = skill.grantedEffects;
246 // A skill is an Active operation even when it currently has no effect id;
247 // the injected executor may emit a settlement/event for that operation.
248 definition.activeExecutionRequired = true;
249 definition.metadata.emplace("cooldown", eve::Value(cooldown.value().seconds()));
250 definition.metadata.emplace("targetType", eve::Value(skill.targetType));
251 definition.metadata.emplace("skillId", eve::Value(skill.id));
252
253 auto valid = definition.validate();
254 if (!valid) return eve::Result<action::ActionDefinition>::failure(valid.status());
255 return eve::Result<action::ActionDefinition>::success(std::move(definition));
256}

258eve::Result<action::ActionRequest> SkillActionAdapter::makeRequest(const SkillDefinition& skill,
259 std::optional<ecs::EntityHandle> source,
260 std::optional<ecs::EntityHandle> target,
261 eve::SimulationTick requestedTick) {
262 auto definition = makeDefinition(skill);
263 if (!definition) return eve::Result<action::ActionRequest>::failure(definition.status());
264 auto checkedDefinition = std::move(definition).takeValue();
265
266 if (checkedDefinition.targetingMode == action::TargetingMode::None && target)
267 return eve::Result<action::ActionRequest>::failure(eve::Diagnostic::error(
268 eve::DiagnosticCode::InvalidArgument, "self skill request cannot carry an explicit target", "target"));
269 if (checkedDefinition.targetingMode == action::TargetingMode::Explicit && !target)
270 return eve::Result<action::ActionRequest>::failure(
271 eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
272 "non-self skill request requires an explicit target", "target"));
273
274 action::ActionRequest request;
275 request.actionId = checkedDefinition.id;
276 request.source = std::move(source);
277 request.requestedTick = requestedTick;
278 if (target) request.targetEntities.push_back(*target);
279 auto valid = request.validate(checkedDefinition);
280 if (!valid) return eve::Result<action::ActionRequest>::failure(valid.status());
281 return eve::Result<action::ActionRequest>::success(std::move(request));
282}

302std::optional<eve::Value> skillValue(const RPGActor* actor, const SkillDefinition& definition, std::string_view key) {
303 if (!actor) return std::nullopt;
304 auto* mutableActor = const_cast<RPGActor*>(actor);
305 const auto* runtime = [&]() -> const SkillRuntime* {
306 const auto it = mutableActor->skills()->known.find(definition.id);
307 return it == mutableActor->skills()->known.end() ? nullptr : &it->second;
308 }();
309 if (key == "skill.id") return eve::Value(definition.id);
310 if (key == "skill.targetType") return eve::Value(definition.targetType);
311 if (key == "skill.known") return eve::Value(runtime != nullptr);
312 if (key == "skill.casting") return eve::Value(mutableActor->skills()->casting.active);
313 if (key == "skill.cooldown") return eve::Value(runtime ? runtime->cooldownRemaining : 0.0);
314 if (key == "skill.phase") return eve::Value(mutableActor->skills()->casting.active ? "casting" : "idle");
315 return std::nullopt;
316}

324std::optional<eve::Value> SkillConditionContext::value(std::string_view key) const {
325 if (auto tag = condition::stripPrefix(key, condition::kTagPrefix)) {
326 if (!actor_) return std::nullopt;
327 if (definition_->hasTag(std::string(*tag))) return eve::Value(true);
328 return eve::Value(const_cast<RPGActor*>(actor_)->hasStatusTag(std::string(*tag)));
329 }
330 auto attributeOf = [this](std::string_view name) -> std::optional<eve::Value> {
331 if (!actor_ || name.empty()) return std::nullopt;
332 auto* mutableActor = const_cast<RPGActor*>(actor_);
333 if (!AttributeSystem::hasAttribute(mutableActor, std::string(name))) return std::nullopt;
334 return eve::Value(AttributeSystem::getFinal(mutableActor, std::string(name)));
335 };
336 if (auto attr = condition::stripPrefix(key, condition::kAttrPrefix)) return attributeOf(*attr);
337 if (auto resource = condition::stripPrefix(key, condition::kResourcePrefix)) {
338 // RPG resources are attributes by design (mana, stamina, and game-defined
339 // resources use the same canonical AttributeSet source).
340 return attributeOf(*resource);
341 }
342 if (auto state = condition::stripPrefix(key, condition::kStatePrefix)) {
343 return skillValue(actor_, *definition_, *state);
344 }
345 return skillValue(actor_, *definition_, key);
346}

348std::optional<condition::ConditionResult> SkillConditionContext::policy(std::string_view name,
349 const eve::Value& arguments) const {
350 if (auto authority = condition::resolveAuthorityAllow(name, arguments, [this](std::string_view scope) {

354 return authority;
