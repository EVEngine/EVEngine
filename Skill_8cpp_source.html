23eve::Result<std::string> canonicalJson(const SkillDefinition& def) {
24 eve::Value::Object object;
25 object.emplace("id", eve::Value(def.id));
26 object.emplace("cooldown", eve::Value(static_cast<double>(def.cooldown)));
27 object.emplace("castTime", eve::Value(static_cast<double>(def.castTime)));
28 object.emplace("targetType", eve::Value(def.targetType));
29
30 auto condition = encodeSkillCondition(def.castCondition);
31 if (!condition) return eve::Result<std::string>::failure(condition.status());
32 object.emplace("castCondition", std::move(condition).takeValue());
33
34 eve::Value::Array effects;
35 for (const auto& effect : def.grantedEffects) effects.emplace_back(effect);
36 object.emplace("grantedEffects", eve::Value(std::move(effects)));
37
38 eve::Value::Array tags;
39 for (const auto& tag : def.tags) tags.emplace_back(tag);
40 object.emplace("tags", eve::Value(std::move(tags)));
41
42 eve::Value::Object extra;
43 for (const auto& [key, value] : def.extra) extra.emplace(key, eve::Value(value));
44 object.emplace("extra", eve::Value(std::move(extra)));
45
46 eve::Value::Array costs;
47 if (def.cost) {
48 for (const auto& item : def.cost->items()) {
49 costs.emplace_back(eve::Value(eve::Value::Object{
50 {"attribute", eve::Value(item.resource.value())},
51 {"amount", eve::Value(item.amount.value())},
52 }));
53 }
54 }
55 object.emplace("costs", eve::Value(std::move(costs)));
56 return eve::Value(std::move(object)).toJson();
57}

72void SkillRegistry::registerSkill(const SkillDefinition &def) {
73 if (def.id.empty()) return;
74 auto json = canonicalJson(def);
75 if (!json) {
76 json.ignore("legacy skill registration cannot serialize invalid definition");
77 return;
78 }
79 auto& registry = canonicalSkillRegistry();
80 auto existing = registry.resolve("rpg.skill", def.id);
81 eve::Result<eve::definition::DefinitionHandle> result =
82 existing.ok() ? registry.replace("rpg.skill", def.id, 1, std::move(json).takeValue())
83 : existing.status().code() == eve::StatusCode::NotFound
84 ? registry.insert("rpg.skill", def.id, 1, std::move(json).takeValue())
85 : eve::Result<eve::definition::DefinitionHandle>::failure(existing.status());
86 if (!result) return;
87 table()[def.id] = def;
88}

118SkillDefinition parseSkillObject(Value o, std::string* error) {
119 SkillDefinition def;
120 if (!o.isObject()) return def;
121 def.id = o.getString("id");
122 def.cooldown = o.getFloat("cooldown", 0.f);
123 def.castTime = o.getFloat("castTime", 0.f);
124 def.targetType = o.getString("targetType", "self");
125 def.grantedEffects = o.getStringArray("grantedEffects");
126 def.tags = o.getStringArray("tags");
127
128 const Value costs = o.get("costs");
129 std::vector<eve::resource::ResourceCost> costItems;
130 for (size_t i = 0; i < costs.size(); ++i) {
131 const Value co = costs.at(i);
132 const std::string resource = co.getString("attribute");
133 const double amount = co.getDouble("amount", 0.0);
134 // Keep the legacy JSON shape, but do not silently truncate a value
135 // into the canonical integral resource Amount type.
136 if (resource.empty()) continue;
137 if (!std::isfinite(amount) || amount <= 0.0 || std::floor(amount) != amount || amount >= std::ldexp(1.0, 63)) {
138 if (error) *error = "skill cost must be a finite positive integer: " + resource;
139 return {};
140 }
141 auto item = eve::resource::ResourceCost::create(resource, static_cast<std::int64_t>(amount));
142 if (!item) {
143 if (error) *error = item.status().describe();
144 return {};
145 }
146 costItems.push_back(std::move(item).takeValue());
147 }
148 if (!costItems.empty()) {
149 auto canonical = eve::resource::CostSpec::create(std::move(costItems));
150 if (!canonical) {
151 if (error) *error = canonical.status().describe();
152 return {};
153 }
154 def.cost = std::move(canonical).takeValue();
155 }
156
157 def.extra = o.getStringMap("extra");
158 return def;
159}

163int SkillRegistry::loadFromJson(const std::string &json, std::string *error) {
164 std::string err;
165 const eve::json::Document doc = eve::json::Document::parse(json, &err);
166 if (!doc.valid()) {
167 if (error) *error = err.empty() ? "invalid json" : err;
168 return 0;
169 }
170
171 const Value root = doc.root();
172 int n = 0;
173 if (root.isArray()) {
174 for (size_t i = 0; i < root.size(); ++i) {
175 SkillDefinition def = parseSkillObject(root.at(i), error);
176 if (def.id.empty()) continue;
177 registerSkill(def);
178 ++n;
179 }
180 } else if (root.isObject()) {
181 SkillDefinition def = parseSkillObject(root, error);
182 if (!def.id.empty()) {
183 registerSkill(def);
184 ++n;
185 }
186 }
187 return n;
188}
