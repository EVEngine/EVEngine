42Result<void> PcgCarCameraSetup::apply(CameraController* c, graphics::Camera3D* camera,
43 scene::SceneNodeRef* focus, float initialYaw, float initialPitch) {
44 if (!c || !camera || !focus) return invalid("controller, camera and focus are required", "camera.pcgCar.apply");
45 if (!finite(initialYaw) || !finite(initialPitch)) return invalid("initial angles must be finite", "camera.pcgCar.apply");
46 if (!configured_) {
47 auto result = configure(profile_);
48 if (!result) return result;
49 }
50 auto limits = c->setRadiusLimits(profile_.minimumDistance, profile_.maximumDistance);
51 if (!limits) return limits;
52 yaw_ = initialYaw;
53 pitch_ = std::clamp(initialPitch, profile_.minimumPitch, profile_.maximumPitch);
54 c->setCamera(camera);
55 c->setTargetNode(focus);
56 c->setMode("orbit");
57 c->setLookAhead(0.f, profile_.targetHeight, 0.f);
58 c->setRadius(profile_.distance);
59 c->setAzimuth(yaw_);
60 c->setElevation(pitch_);
61 c->setTargetSmooth(profile_.rotationDamping);
62 c->setPositionSmooth(profile_.zoomDamping);
63 c->setCollisionEnabled(true);
64 c->setCollisionRadius(profile_.offsetFromWall);
65 c->setCollisionRecovery(profile_.zoomDamping);
66 c->setCollisionMask(profile_.collisionLayers);
67 return Result<void>::success();

70Result<void> PcgCarCameraSetup::update(CameraController* c, float mouseX, float mouseY,
71 float scroll, float dt, bool targetIsMoving, float targetYaw) {
72 if (!c) return invalid("controller is required", "camera.pcgCar.update");
73 if (!finite(mouseX) || !finite(mouseY) || !finite(scroll) || !finite(dt) || !finite(targetYaw) || dt < 0.f)
74 return invalid("camera input and dt must be finite", "camera.pcgCar.update");
75 if (!configured_) return invalid("camera setup has not been configured", "camera.pcgCar.update");
76 if (profile_.allowMouseHorizontal) yaw_ += mouseX * profile_.horizontalSpeed * 0.02f;
77 if (profile_.allowMouseVertical)
78 pitch_ = std::clamp(pitch_ - mouseY * profile_.verticalSpeed * 0.02f,
79 profile_.minimumPitch, profile_.maximumPitch);
80 if ((profile_.lockToRearOfTarget || targetIsMoving) && std::abs(mouseX) <= 1e-6f) {
81 const float blend = 1.f - std::exp(-profile_.rotationDamping * dt);
82 yaw_ += shortestAngle(yaw_, targetYaw) * blend;
83 }
84 c->setAzimuth(yaw_);
85 c->setElevation(pitch_);
86 const float zoomDelta = -scroll * dt * profile_.zoomRate * std::abs(c->getRadius());
87 c->addInput(0.f, 0.f, zoomDelta);
88 return Result<void>::success();
89}
