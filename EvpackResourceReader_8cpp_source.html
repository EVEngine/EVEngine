18 if (!pack_ || asset.id().isNil() || expectedType.empty())
19 return fail<RuntimeAssetPayload>(DiagnosticCode::InvalidArgument,
20 "pack, asset and expected type are required");
21 auto selected = selectEvpackVariant(*pack_, capabilities);
22 if (!selected) return Result<RuntimeAssetPayload>::failure(selected.status());
23 RuntimeAssetPayload result{asset, {}, {}, selected.value(), {}};
24 bool identityFound = false;
25 bool typeFound = false;
26 std::uint64_t decodedTotal = 0;
27 for (std::size_t index = 0; index < pack_->chunks().size(); ++index) {
28 const auto& chunk = pack_->chunks()[index];
29 if (chunk.assetId != asset.id()) continue;
30 identityFound = true;
31 const std::string actualType = chunk.type + "/" + std::to_string(chunk.schemaVersion.value());
32 if (actualType != expectedType) continue;
33 typeFound = true;
34 if (chunk.variantIndex != selected.value().index) continue;
35 if (decodedTotal > maximumDecodedBytes || chunk.decodedSize > maximumDecodedBytes - decodedTotal)
36 return fail<RuntimeAssetPayload>(DiagnosticCode::InvalidArgument,
37 "asset payload exceeds decoded budget", asset.format());
38 auto bytes = pack_->decodeChunk(index, maximumDecodedBytes - decodedTotal);
39 if (!bytes) return Result<RuntimeAssetPayload>::failure(bytes.status());
40 decodedTotal += bytes.value().size();
41 result.type = chunk.type;
42 result.schemaVersion = chunk.schemaVersion;
43 result.chunks.push_back({chunk.kind, chunk.chunkId, std::move(bytes).takeValue()});
44 }
45 if (result.chunks.empty()) {
46 if (identityFound && !typeFound)
47 return fail<RuntimeAssetPayload>(DiagnosticCode::TypeMismatch,
48 "asset does not provide the expected type/version",
49 asset.format());
50 if (typeFound)
51 return fail<RuntimeAssetPayload>(DiagnosticCode::Unsupported,
52 "asset has no chunk for the selected variant", asset.format());
53 return fail<RuntimeAssetPayload>(DiagnosticCode::NotFound, "asset is not present", asset.format());
54 }
55 std::sort(result.chunks.begin(), result.chunks.end(), [](const auto& left, const auto& right) {
56 return std::tie(left.kind, left.chunkId) < std::tie(right.kind, right.chunkId);
57 });
58 if (result.chunks.front().kind != EvpackChunkKind::Definition)
59 return fail<RuntimeAssetPayload>(DiagnosticCode::InvariantViolation,
60 "asset has no definition chunk", asset.format());
61 return Result<RuntimeAssetPayload>::success(std::move(result));
62}
