8struct EditRegion {
9 int minX = 0, minY = 0, maxX = -1, maxY = -1;
10 bool empty() const { return maxX < minX || maxY < minY; }
11 void clear() { *this = {}; }
12 void include(int x, int y) {
13 if (empty()) {
14 minX = maxX = x;
15 minY = maxY = y;
16 return;
17 }
18 if (x < minX) minX = x;
19 if (x > maxX) maxX = x;

22 }
