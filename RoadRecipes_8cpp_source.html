31std::unique_ptr<image::ImageData> generateRoadMarkingsTexture(const Params& params, std::string& error) {
32 const int width = std::clamp(params.getWidth() > 0 ? params.getWidth() : params.getInt("deckWidth", 256), 8, 2048);
33 const int height = std::clamp(params.getHeight() > 0 ? params.getHeight() : params.getInt("height", 64), 8, 2048);
34 const float dashPeriod = std::max(0.05f, params.getFloat("dashPeriod", 0.18f));
35 const float dashDuty = std::clamp(params.getFloat("dashDuty", 0.45f), 0.05f, 0.95f);
36 const bool zebra = params.getBool("zebra", true);
37
38 auto image = std::make_unique<image::ImageData>(width, height, "RGBA8");
39 for (int y = 0; y < height; ++y) {
40 const float v = (static_cast<float>(y) + 0.5f) / static_cast<float>(height);
41 for (int x = 0; x < width; ++x) {
42 const float u = (static_cast<float>(x) + 0.5f) / static_cast<float>(width);
43 float r = 0.18f, g = 0.18f, b = 0.20f;
44
45 // White edge lines near v=0 and v=1.
46 if (v < 0.08f || v > 0.92f) {
47 r = g = b = 0.92f;
48 }
49 // Yellow dashed center.
50 else if (std::fabs(v - 0.5f) < 0.035f) {
51 const float phase = std::fmod(u, dashPeriod) / dashPeriod;
52 if (phase < dashDuty) {
53 r = 0.95f;
54 g = 0.75f;
55 b = 0.12f;
56 }
57 }
58 // Lane divider dashes at 1/4 and 3/4.
59 else if (std::fabs(v - 0.25f) < 0.02f || std::fabs(v - 0.75f) < 0.02f) {
60 const float phase = std::fmod(u + 0.05f, dashPeriod) / dashPeriod;
61 if (phase < dashDuty) r = g = b = 0.88f;
62 }
63
64 // Zebra band near u ends (intersection approach).
65 if (zebra && (u < 0.08f || u > 0.92f)) {
66 const int stripe = static_cast<int>(v * 12.f);
67 if ((stripe & 1) == 0) r = g = b = 0.95f;
68 }
69
70 // Direction arrow glyph in the middle third.
71 const float au = (u - 0.42f) / 0.16f;
72 const float av = (v - 0.5f) / 0.18f;
73 if (au >= 0.f && au <= 1.f && std::fabs(av) <= (1.f - au) * 0.55f + 0.08f) {
74 r = 0.95f;
75 g = 0.45f;
76 b = 0.08f;
77 }
78
79 image->setPixel(x, y, image::ImageData::Colorf{r, g, b, 1.f});
80 }
81 }
82 (void)error;
83 return image;
84}
