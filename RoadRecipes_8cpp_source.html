56std::unique_ptr<image::ImageData> generateRoadMarkingsTexture(const Params& params, std::string& error) {
57 const int width = std::clamp(params.getWidth() > 0 ? params.getWidth() : params.getInt("width", 256), 8, 2048);
58 const int height = std::clamp(params.getHeight() > 0 ? params.getHeight() : params.getInt("height", 64), 8, 2048);
59 const float dashPeriod = std::max(0.05f, params.getFloat("dashPeriod", 0.18f));
60 const float dashDuty = std::clamp(params.getFloat("dashDuty", 0.45f), 0.05f, 0.95f);
61 const bool zebra = params.getBool("zebra", true);
62
63 auto image = std::make_unique<image::ImageData>(width, height, "RGBA8");
64 for (int y = 0; y < height; ++y) {
65 const float v = (static_cast<float>(y) + 0.5f) / static_cast<float>(height);
66 for (int x = 0; x < width; ++x) {
67 const float u = (static_cast<float>(x) + 0.5f) / static_cast<float>(width);
68 float r = 0.18f, g = 0.18f, b = 0.20f;
69
70 // White edge lines near v=0 and v=1.
71 if (v < 0.08f || v > 0.92f) {
72 r = g = b = 0.92f;
73 }
74 // Yellow dashed center.
75 else if (std::fabs(v - 0.5f) < 0.035f) {
76 const float phase = std::fmod(u, dashPeriod) / dashPeriod;
77 if (phase < dashDuty) {
78 r = 0.95f;
79 g = 0.75f;
80 b = 0.12f;
81 }
82 }
83 // Lane divider dashes at 1/4 and 3/4.
84 else if (std::fabs(v - 0.25f) < 0.02f || std::fabs(v - 0.75f) < 0.02f) {
85 const float phase = std::fmod(u + 0.05f, dashPeriod) / dashPeriod;
86 if (phase < dashDuty) r = g = b = 0.88f;
87 }
88
89 // Zebra band near u ends (intersection approach).
90 if (zebra && (u < 0.08f || u > 0.92f)) {
91 const int stripe = static_cast<int>(v * 12.f);
92 if ((stripe & 1) == 0) r = g = b = 0.95f;
93 }
94
95 // Direction arrow glyph in the middle third.
96 const float au = (u - 0.42f) / 0.16f;
97 const float av = (v - 0.5f) / 0.18f;
98 if (au >= 0.f && au <= 1.f && std::fabs(av) <= (1.f - au) * 0.55f + 0.08f) {
99 r = 0.95f;
100 g = 0.45f;
101 b = 0.08f;
102 }
103
104 image->setPixel(x, y, image::ImageData::Colorf{r, g, b, 1.f});
105 }
106 }
107 (void)error;
108 return image;
109}
