26const BoardState* boundBoard(ecs::EntityHandle battle, bool bound) {
27 if (!bound) return nullptr;
28 // Resolving by handle each call is what makes a released battle safe: the ECS refuses a stale
29 // generation, so a dangling board is never read.
30 auto* resolved = dynamic_cast<Battle*>(ecs::try_get(battle));
31 return resolved != nullptr ? &resolved->board()->value : nullptr;
32}

35Result<Cell> toCell(const sensing::TargetLocation& location) {
36 return std::visit(
37 [](const auto& point) -> Result<Cell> {
38 using Point = std::decay_t<decltype(point)>;
39 if constexpr (std::is_same_v<Point, sensing::GridPoint>) {
40 // A 2D grid point is layer zero; a 3D one keeps its z as the tactical layer, so a
41 // layered board is reachable without converting between spaces.
42 if (point.space() == sensing::CoordinateSpace::Grid2D) return Result<Cell>::success(Cell{point.x(), point.y(), 0});
43 if (point.space() == sensing::CoordinateSpace::Grid3D)
44 return Result<Cell>::success(Cell{point.x(), point.y(), point.z()});
45 return Result<Cell>::failure(Diagnostic::error(DiagnosticCode::Unsupported,
46 "tactics line of sight answers grid coordinates only",
47 "lineOfSight.space"));
48 } else {
49 return Result<Cell>::failure(Diagnostic::error(DiagnosticCode::Unsupported,
50 "tactics line of sight answers grid coordinates only",
51 "lineOfSight.space"));
52 }
53 },
54 location);
55}

64Result<void> TacticsLineOfSightAdapter::bindBattle(ecs::EntityHandle battle) {
65 if (boundBoard(battle, true) == nullptr)
66 return Result<void>::failure(Diagnostic::error(DiagnosticCode::StaleHandle,
67 "line-of-sight battle binding is not a live battle", "battle"));
68 if (bound_ && sameHandle(battle_, battle)) return Result<void>::success(Status::success(StatusCode::NoOp));
69 if (bound_)
70 // One board at a time on purpose: with two bound boards the answer would depend on which
71 // was bound last, exactly the order-dependence the router exists to remove.
72 return Result<void>::failure(Diagnostic::error(
73 DiagnosticCode::Conflict, "another battle is already bound for grid line of sight", "battle"));
74 battle_ = battle;
75 bound_ = true;
76 return Result<void>::success(Status::success(StatusCode::Applied));
77}

79Result<void> TacticsLineOfSightAdapter::unbindBattle(ecs::EntityHandle battle) {

87Result<sensing::LineOfSightResult> TacticsLineOfSightAdapter::query(const sensing::TargetLocation& from,
88 const sensing::TargetLocation& to) const {
89 const BoardState* board = boundBoard(battle_, bound_);
90 if (board == nullptr)
91 return Result<sensing::LineOfSightResult>::failure(Diagnostic::error(
92 DiagnosticCode::Unsupported, "no tactics battle is bound for grid line of sight", "lineOfSight.battle"));
93 auto fromCell = toCell(from);
94 if (!fromCell) return Result<sensing::LineOfSightResult>::failure(fromCell.status());
95 auto toCellValue = toCell(to);
96 if (!toCellValue) return Result<sensing::LineOfSightResult>::failure(toCellValue.status());
97
98 // The same policy the module's own range query uses: a registered sight policy when a project
99 // provided one, otherwise the built-in grid rule. One authority for "what blocks sight".
100 const ILineOfSightPolicy* registered = cap::query<ILineOfSightPolicy>();
101 const ILineOfSightPolicy& policy = registered != nullptr ? *registered : *gridLineOfSightPolicy();
102 auto visible = policy.visible(*board, fromCell.value(), toCellValue.value());
103 if (!visible) return Result<sensing::LineOfSightResult>::failure(visible.status());
104
105 sensing::LineOfSightResult result;
106 result.visible = visible.value();
107 // `blocker` is left empty on purpose: the grid policy answers visibility without naming what
108 // blocked it, and reporting the target's occupant instead would be a different fact wearing
109 // the same field. A policy that can identify the blocker is free to fill it in.
110 return Result<sensing::LineOfSightResult>::success(std::move(result));
111}

113Result<void> registerTacticsLineOfSightProvider() {
114 auto router = sensing::ensureLineOfSightRouter();
115 if (!router) return Result<void>::failure(router.status());
116 TacticsLineOfSightAdapter& adapter = tacticsLineOfSightAdapter();
117 auto grid2d = router.value()->addProvider(sensing::CoordinateSpace::Grid2D, &adapter);
118 if (!grid2d) return Result<void>::failure(grid2d.status());
119 auto grid3d = router.value()->addProvider(sensing::CoordinateSpace::Grid3D, &adapter);
120 if (!grid3d) return Result<void>::failure(grid3d.status());
121 return Result<void>::success(Status::success(StatusCode::Applied));
122}
