37editing::Result<void> registerEditingCommands(editing::IEditingCommandRegistry& registry) {
38 editing::EditingCommandDescriptor descriptor;
39 descriptor.id = editing::CommandId("audio.source.property.set.v1");
40 descriptor.ownerModule = "audio_editing";
41 descriptor.displayName = "Set audio source property";
42 descriptor.category = "Audio";
43 descriptor.automationAllowed = true;
44 return registry.registerPlannedCommand(std::move(descriptor), [](editing::IEditableTarget& target,
45 const editing::CommandRequest& request) {
46 auto* properties =
47 static_cast<editing::IPropertyProvider*>(target.queryCapability(AudioSourceTarget::editingCapabilityId()));
48 const editing::Value* pathValue = field(request.payload, "path");
49 const auto* path = pathValue ? pathValue->getIf<std::string>() : nullptr;
50 const editing::Value* value = field(request.payload, "value");
51 if (!properties || !path || !value)
52 return eve::editing::failed<editing::CommandPlan>(
53 editing::Status::Rejected, editing::RuleId("audio.editing.property-payload"),
54 "Audio property requires a source target, path and value");
55 auto operation = properties->makeSet(selectionFor(target), editing::PropertyPath(*path), *value,
56 editing::PropertySetMode::Absolute);
57 if (!operation.ok())
58 return eve::editing::failed<editing::CommandPlan>(operation.code(),
59 editing::RuleId("audio.editing.property-operation"),
60 "Audio source rejected the property value");
61 editing::CommandPlan plan;
62 plan.operations.push_back(std::move(operation.value()));
63 plan.summary = editing::Value::Object{{"path", *path}};
64 return eve::editing::applied<editing::CommandPlan>(std::move(plan));
65 });
66}

96AudioImportInspectionResult AudioImportDiagnosticsService::inspect(const AudioImportInspectionRequest& request,
97 const EditorAudioPcm& pcm) const {
98 AudioImportInspectionResult result;
99 result.asset = request.asset;
100 result.sourceRevision = request.sourceRevision;
101 if (request.asset.empty() || request.sourceRevision == 0 || request.codec.empty() || pcm.sampleRate < 8000 ||
102 pcm.sampleRate > 384000 || pcm.channels < 1 || pcm.channels > 8 || pcm.samples.empty() ||
103 pcm.samples.size() % static_cast<std::size_t>(pcm.channels) != 0 ||
104 pcm.samples.size() > request.maximumSamples || request.maximumSamples == 0 ||
105 !std::isfinite(request.clippingThreshold) || request.clippingThreshold <= 0.0 ||
106 request.clippingThreshold > 1.0 || !std::isfinite(request.silenceThreshold) || request.silenceThreshold < 0.0 ||
107 request.silenceThreshold >= request.clippingThreshold) {
108 result.status = EditorStatus::Rejected;
109 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
110 eve::DiagnosticCode::InvalidArgument, RuleId("editor.audio.invalid-import-inspection"),
111 DiagnosticSeverity::Error, "Audio inspection requires valid identity, format, thresholds and bounded PCM"));
112 return result;
113 }
114 const std::size_t frames = pcm.samples.size() / static_cast<std::size_t>(pcm.channels);
115 result.duration = static_cast<double>(frames) / static_cast<double>(pcm.sampleRate);
116 if (pcm.samples.size() > std::numeric_limits<std::size_t>::max() / sizeof(float)) {
117 result.status = EditorStatus::Rejected;
118 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
119 eve::DiagnosticCode::InvalidArgument, RuleId("editor.audio.decoded-size-overflow"),
120 DiagnosticSeverity::Error, "Decoded audio size exceeds the host range"));
121 return result;
122 }
123 result.decodedBytes = pcm.samples.size() * sizeof(float);
124 result.compressionRatio = request.encodedBytes > 0
125 ? static_cast<double>(result.decodedBytes) / static_cast<double>(request.encodedBytes)
126 : 0.0;
127 result.recommendStreaming = result.decodedBytes > request.staticMemoryBudget || result.duration > 10.0;
128 result.channels.resize(static_cast<std::size_t>(pcm.channels));
129 std::vector<long double> squares(result.channels.size(), 0.0L);
130 std::vector<long double> sums(result.channels.size(), 0.0L);
131 for (std::size_t frame = 0; frame < frames; ++frame) {
132 for (int channel = 0; channel < pcm.channels; ++channel) {
133 const float sample =
134 pcm.samples[frame * static_cast<std::size_t>(pcm.channels) + static_cast<std::size_t>(channel)];
135 if (!std::isfinite(sample)) {
136 result.status = EditorStatus::Rejected;
137 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
138 eve::DiagnosticCode::InvalidArgument, RuleId("editor.audio.nonfinite-sample"),
139 DiagnosticSeverity::Error, "Decoded PCM contains a non-finite sample"));
140 return result;
141 }
142 auto& statistics = result.channels[static_cast<std::size_t>(channel)];
143 const double absolute = std::abs(static_cast<double>(sample));
144 statistics.peak = std::max(statistics.peak, absolute);
145 if (absolute >= request.clippingThreshold) ++statistics.clippedSamples;
146 sums[static_cast<std::size_t>(channel)] += sample;
147 squares[static_cast<std::size_t>(channel)] += static_cast<long double>(sample) * sample;
148 }
149 }
150 bool silent = true;
151 for (std::size_t channel = 0; channel < result.channels.size(); ++channel) {
152 auto& statistics = result.channels[channel];
153 statistics.dcOffset = static_cast<double>(sums[channel] / frames);
154 statistics.rms = std::sqrt(static_cast<double>(squares[channel] / frames));
155 silent = silent && statistics.peak <= request.silenceThreshold;
156 if (statistics.clippedSamples > 0)
157 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
158 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.audio.clipping"),
159 DiagnosticSeverity::Warning, "Audio channel " + std::to_string(channel) + " contains clipped samples"));
160 if (std::abs(statistics.dcOffset) > 0.01)
161 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
162 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.audio.dc-offset"),
163 DiagnosticSeverity::Warning,
164 "Audio channel " + std::to_string(channel) + " has significant DC offset"));
165 }
166 if (silent)
167 result.diagnostics.push_back(
168 eve::editing::ruleDiagnostic(eve::DiagnosticCode::PreconditionViolation, RuleId("editor.audio.silence"),
169 DiagnosticSeverity::Warning, "Decoded audio is effectively silent"));
170 if (result.recommendStreaming && !request.streaming)
171 result.diagnostics.push_back(eve::editing::ruleDiagnostic(
172 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.audio.streaming-recommended"),
173 DiagnosticSeverity::Warning, "Long or memory-heavy audio should use streaming playback"));
174 if (!result.recommendStreaming && request.streaming)
175 result.diagnostics.push_back(
176 eve::editing::ruleDiagnostic(eve::DiagnosticCode::None, RuleId("editor.audio.static-recommended"),
177 DiagnosticSeverity::Info, "Short audio can avoid streaming overhead"));
178 result.status = EditorStatus::Applied;
179 return result;
180}

215Result<void> AudioSourcePublishingTarget::commitDomainState(std::unique_ptr<IDomainOperationTarget> candidate) {
216 auto* typed = dynamic_cast<AudioSourcePublishingTarget*>(candidate.get());
217 if (!typed || typed->targetId() != targetId() || typed->sink_ != sink_ || !typed->staging_)
218 return eve::editing::failed<void>(EditorStatus::Conflict, RuleId("editor.audio.publishing-candidate-mismatch"),
219 "Audio publishing candidate belongs to another live target");
220 if (!sink_)

222 "Audio publishing target requires a live runtime sink");
223 Result<void> published = sink_->publish(typed->document_);
224 if (!published.ok()) return published;
225 document_ = typed->document_;
226 return eve::editing::applied<void>();
