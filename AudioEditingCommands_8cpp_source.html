37editing::Result<void> registerEditingCommands(editing::IEditingCommandRegistry& registry) {
38 editing::EditingCommandDescriptor descriptor;
39 descriptor.id = editing::CommandId("audio.source.property.set.v1");
40 descriptor.ownerModule = "audio_editing";
41 descriptor.displayName = "Set audio source property";
42 descriptor.category = "Audio";
43 descriptor.automationAllowed = true;
44 return registry.registerPlannedCommand(
45 std::move(descriptor), [](editing::IEditableTarget& target, const editing::CommandRequest& request) {
46 auto* properties = static_cast<editing::IPropertyProvider*>(
47 target.queryCapability(AudioSourceTarget::editingCapabilityId()));
48 const editing::Value* pathValue = field(request.payload, "path");
49 const auto* path = pathValue ? pathValue->getIf<std::string>() : nullptr;
50 const editing::Value* value = field(request.payload, "value");
51 if (!properties || !path || !value)
52 return error<editing::CommandPlan>(editing::Status::Rejected, "audio.editing.property-payload",
53 "Audio property requires a source target, path and value");
54 auto operation = properties->makeSet(selectionFor(target), editing::PropertyPath(*path), *value,
55 editing::PropertySetMode::Absolute);
56 if (!operation.ok())
57 return error<editing::CommandPlan>(operation.code(), "audio.editing.property-operation",
58 "Audio source rejected the property value");
59 editing::CommandPlan plan;
60 plan.operations.push_back(std::move(operation.value()));
61 plan.summary = editing::Value::Object{{"path", *path}};
62 return eve::editing::applied<editing::CommandPlan>(std::move(plan));
63 });
64}
