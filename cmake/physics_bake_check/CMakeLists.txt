# Link whole manifest-selected modules, not a hand-picked source subset.
set(_eve_bake_expected common physics_bake physics_bake_collapse)
if(EVENGINE_PROFILE STREQUAL "physics-bake-script-only")
    list(APPEND _eve_bake_expected physics_bake_script)
endif()
if(NOT "${EVE_ENABLED_MODULES}" STREQUAL "${_eve_bake_expected}")
    message(FATAL_ERROR "${EVENGINE_PROFILE} must contain exactly ${_eve_bake_expected}")
endif()
if(NOT "${EVE_THIRDPARTY_GROUPS}" STREQUAL "squirrel")
    message(FATAL_ERROR "physics-bake-only acquired an unexpected third-party group")
endif()
eve_thirdparty_libs(_eve_bake_libraries ${EVE_THIRDPARTY_GROUPS})
add_executable(eve_bake_playback_check "${CMAKE_SOURCE_DIR}/test/profile/physics_bake/manifest.cpp")
add_executable(eve_bake_measure "${CMAKE_SOURCE_DIR}/test/profile/physics_bake/measure.cpp")
add_executable(eve_bake_collapse_inspect "${CMAKE_SOURCE_DIR}/test/profile/physics_bake/inspect.cpp")
set(_eve_bake_targets eve_bake_playback_check eve_bake_measure eve_bake_collapse_inspect)
if(EVENGINE_PROFILE STREQUAL "physics-bake-script-only")
    add_executable(eve_bake_script_check "${CMAKE_SOURCE_DIR}/test/profile/physics_bake/script_manifest.cpp")
    list(APPEND _eve_bake_targets eve_bake_script_check)
endif()
foreach(_eve_bake_target IN LISTS _eve_bake_targets)
    target_compile_features(${_eve_bake_target} PRIVATE cxx_std_20)
    target_link_libraries(${_eve_bake_target} PRIVATE ${EVE_LINK_TARGETS}
        zeroerr eve_engine_includes ${_eve_bake_libraries} ${CMAKE_DL_LIBS})
    set_target_properties(${_eve_bake_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/profile")
endforeach()
