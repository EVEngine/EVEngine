15std::string hexEncode(std::string_view value) {
16 constexpr char digits[] = "0123456789abcdef";
17 std::string output;
18 output.reserve(value.size() * 2);
19 for (const unsigned char byte : value) {
20 output.push_back(digits[byte >> 4U]);
21 output.push_back(digits[byte & 15U]);
22 }
23 return output;
24}

26Result<std::string> hexDecode(std::string_view value) {
27 if (value.size() % 2 != 0)
28 return Result<std::string>::failure(Diagnostic::error(
29 DiagnosticCode::ParseError, "odd embedded definition size", {}, {}, "procgen.buildLayers"));
30 const auto nibble = [](char c) -> int {
31 if (c >= '0' && c <= '9') return c - '0';
32 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
33 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
34 return -1;
35 };
36 std::string output(value.size() / 2, '\0');
37 for (std::size_t index = 0; index < value.size(); index += 2) {
38 const int high = nibble(value[index]);
39 const int low = nibble(value[index + 1]);
40 if (high < 0 || low < 0)
41 return Result<std::string>::failure(Diagnostic::error(
42 DiagnosticCode::ParseError, "invalid embedded definition", {}, {}, "procgen.buildLayers"));
43 output[index / 2] = char((high << 4) | low);
44 }
45 return Result<std::string>::success(std::move(output));
46}

150Result<BuildLayerExecution> BuildLayerStack::executeRegion(const Grid2D& grid, const PointSet& points,
151 const BuildLayerRegion& region,
152 const PointSet* orientation) const {
153 if (region.minCellX < 0 || region.minCellY < 0 || region.maxCellX < region.minCellX ||
154 region.maxCellY < region.minCellY || region.maxCellX > grid.getWidth() || region.maxCellY > grid.getHeight() ||
155 !(region.minWorldX < region.maxWorldX) || !(region.minWorldZ < region.maxWorldZ))
156 return Result<BuildLayerExecution>::failure(Diagnostic::error(
157 DiagnosticCode::InvalidArgument, "invalid build-layer region", {}, {}, "procgen.buildLayers"));
158 PointSet regionPoints;
159 regionPoints.reserve(points.points().size());
160 for (std::size_t index = 0; index < points.points().size(); ++index) {
161 const auto& point = points.points()[index];
162 if (point.x < region.minWorldX || point.x >= region.maxWorldX || point.z < region.minWorldZ ||
163 point.z >= region.maxWorldZ)
164 continue;
165 auto appended = regionPoints.appendPointFrom(points, index);
166 if (!appended.ok()) return Result<BuildLayerExecution>::failure(appended.status());
167 }
168 BuildLayerExecution execution;
169 execution.artifacts_.reserve(layers_.size());
170 for (const auto& layer : layers_) {
171 if (!layer.enabled) continue;
172 if (const auto* tile = std::get_if<TileLayer>(&layer.definition)) {
173 MeshGraph graph;
174 auto result = graph.addNode("grid", "grid.input");
175 if (result.ok()) result = graph.addNode("tiles", "mesh.grid_tiles");
176 if (result.ok()) result = graph.setNodeGrid("grid", grid);
177 if (result.ok()) result = graph.setNodeFloat("tiles", "cellSize", tile->cellSize);
178 if (result.ok()) result = graph.setNodeFloat("tiles", "height", tile->height);
179 if (result.ok()) result = graph.setNodeFloat("tiles", "minX", float(region.minCellX));
180 if (result.ok()) result = graph.setNodeFloat("tiles", "minY", float(region.minCellY));
181 if (result.ok()) result = graph.setNodeFloat("tiles", "maxX", float(region.maxCellX));
182 if (result.ok()) result = graph.setNodeFloat("tiles", "maxY", float(region.maxCellY));
183 if (result.ok()) result = graph.setNodeString("tiles", "group", tile->group);
184 if (result.ok()) result = graph.connect("grid", "tiles");
185 if (!result.ok()) return Result<BuildLayerExecution>::failure(result.status());
186 auto mesh = graph.execute("tiles");
187 if (!mesh.ok()) return Result<BuildLayerExecution>::failure(mesh.status());
188 execution.artifacts_.push_back({layer.id, std::move(mesh).takeValue()});
189 } else {
190 auto built = std::get<ObjectBuildLayer>(layer.definition).build(regionPoints, orientation);
191 if (!built.ok()) return Result<BuildLayerExecution>::failure(built.status());
192 execution.artifacts_.push_back({layer.id, std::move(built).takeValue()});
193 }
194 }
195 return Result<BuildLayerExecution>::success(std::move(execution));
196}

198std::string BuildLayerStack::serializeDefinition() const {
199 std::ostringstream out;
200 out << "EVPCG_BUILD_LAYERS 1\n" << std::setprecision(9);
201 for (const auto& layer : layers_) {
202 if (const auto* tile = std::get_if<TileLayer>(&layer.definition)) {
203 out << "TILE " << std::quoted(layer.id) << ' ' << int(layer.enabled) << ' ' << tile->cellSize << ' '
204 << tile->height << ' ' << std::quoted(tile->group) << '\n';
205 } else {
206 const auto& object = std::get<ObjectBuildLayer>(layer.definition);
207 out << "OBJECT " << std::quoted(layer.id) << ' ' << int(layer.enabled) << ' '
208 << hexEncode(object.serializeDefinition()) << '\n';
209 }
210 }
211 out << "END\n";
212 return out.str();
213}

215Result<void> BuildLayerStack::deserializeDefinition(std::string_view definition) {
216 BuildLayerStack replacement;
217 std::istringstream input{std::string(definition)};
218 std::string magic;
219 int version = 0;
220 if (!(input >> magic >> version) || magic != "EVPCG_BUILD_LAYERS" || version != 1)
221 return Result<void>::failure(
222 Diagnostic::error(DiagnosticCode::ParseError, "invalid build-layer header", {}, {}, "procgen.buildLayers"));
223 std::string line;
224 std::getline(input, line);
225 bool ended = false;
226 while (std::getline(input, line)) {
227 if (line.empty()) continue;
228 std::istringstream record(line);
229 std::string kind;
230 record >> kind;
231 if (kind == "END") {
232 ended = true;
233 break;
234 }
235 std::string id;
236 int enabled = 0;
237 if (kind == "TILE") {
238 float cellSize = 0.f, height = 0.f;
239 std::string group;
240 if (!(record >> std::quoted(id) >> enabled >> cellSize >> height >> std::quoted(group)) ||
241 (enabled != 0 && enabled != 1))
242 return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError, "invalid TILE record", {},
243 {}, "procgen.buildLayers"));
244 auto applied = replacement.addTileLayer(std::move(id), enabled != 0, cellSize, height, std::move(group));
245 if (!applied.ok()) return applied;
246 } else if (kind == "OBJECT") {
247 std::string encoded;
248 if (!(record >> std::quoted(id) >> enabled >> encoded) || (enabled != 0 && enabled != 1))
249 return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError, "invalid OBJECT record", {},
250 {}, "procgen.buildLayers"));
251 auto decoded = hexDecode(encoded);
252 if (!decoded.ok()) return Result<void>::failure(decoded.status());
253 ObjectBuildLayer object;
254 auto applied = object.deserializeDefinition(decoded.value());
255 if (!applied.ok()) return applied;
256 auto added = replacement.addObjectLayer(std::move(id), enabled != 0, object);
257 if (!added.ok()) return added;
258 } else {
259 return Result<void>::failure(Diagnostic::error(
260 DiagnosticCode::ParseError, "unknown build-layer record: " + kind, {}, {}, "procgen.buildLayers"));
261 }
262 record >> std::ws;
263 if (!record.eof())
264 return Result<void>::failure(Diagnostic::error(
265 DiagnosticCode::ParseError, "trailing build-layer record data", {}, {}, "procgen.buildLayers"));
266 }
267 if (!ended)
268 return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError, "build-layer END record is missing",
269 {}, {}, "procgen.buildLayers"));
270 *this = std::move(replacement);
271 return Result<void>::success();
272}

299std::uint64_t mix(std::uint64_t value) {
300 value += 0x9e3779b97f4a7c15ULL;
301 value = (value ^ (value >> 30U)) * 0xbf58476d1ce4e5b9ULL;

306struct RandomStream {
307 std::uint64_t state;
308 std::uint64_t next() {
309 state = mix(state);
310 return state;
311 }
312 float unit() { return float(next() >> 40U) * (1.f / 16777216.f); }
313 float between(float minimum, float maximum) { return minimum + (maximum - minimum) * unit(); }
314};

316std::uint64_t pointKey(const ProcgenPoint& point, std::size_t index, std::uint32_t seed) {
317 if (point.id != 0) return mix(point.id ^ (std::uint64_t(seed) << 32U));
318 const auto qx = std::int64_t(std::llround(double(point.x) * 1000.0));
319 const auto qy = std::int64_t(std::llround(double(point.y) * 1000.0));
320 const auto qz = std::int64_t(std::llround(double(point.z) * 1000.0));
321 return mix(std::uint64_t(qx) ^ (mix(std::uint64_t(qy)) << 1U) ^ (mix(std::uint64_t(qz)) << 2U) ^
322 std::uint64_t(index) ^ (std::uint64_t(seed) << 32U));
323}

325float orientedYaw(const ProcgenPoint& point, const PointSet& orientation, float cellSize, bool invert) {
326 struct Direction {
327 float x;
328 float z;
329 float yaw;
330 };
331 const Direction cardinal[] = {
332 {0.f, 1.f, invert ? 90.f : 270.f}, {1.f, 0.f, 0.f}, {0.f, -1.f, invert ? 270.f : 90.f}, {-1.f, 0.f, 180.f}};
333 const float tolerance = std::max(0.0001f, cellSize * 0.01f);
334 const auto matches = [&](float x, float z) {
335 return std::any_of(orientation.points().begin(), orientation.points().end(), [&](const ProcgenPoint& other) {
336 return std::abs(other.x - x) <= tolerance && std::abs(other.z - z) <= tolerance;
337 });
338 };
339 for (const auto& direction : cardinal)
340 if (matches(point.x + direction.x * cellSize, point.z + direction.z * cellSize)) return direction.yaw;
341 const Direction diagonal[] = {{1.f, 1.f, 90.f}, {1.f, -1.f, 0.f}, {-1.f, -1.f, 180.f}, {-1.f, 1.f, 180.f}};
342 for (const auto& direction : diagonal)
343 if (matches(point.x + direction.x * cellSize, point.z + direction.z * cellSize)) return direction.yaw;
344 return 0.f;
345}

347void applyRandomTransform(ProcgenPoint& point, const ObjectTransformRange& range, RandomStream& random) {
348 point.pitch += random.between(range.minPitch, range.maxPitch);
349 point.yaw += random.between(range.minYaw, range.maxYaw);
350 point.roll += random.between(range.minRoll, range.maxRoll);
351 if (range.uniformScale) {
352 const float scale = random.between(range.minScaleX, range.maxScaleX);
353 point.scaleX *= scale;
354 point.scaleY *= scale;
355 point.scaleZ *= scale;
356 } else {
357 point.scaleX *= random.between(range.minScaleX, range.maxScaleX);
358 point.scaleY *= random.between(range.minScaleY, range.maxScaleY);
359 point.scaleZ *= random.between(range.minScaleZ, range.maxScaleZ);
360 }
361}

502Result<PointSet> ObjectBuildLayer::build(const PointSet& source, const PointSet* orientation) const {
503 if (assets_.empty())
504 return Result<PointSet>::failure(Diagnostic::error(
505 DiagnosticCode::InvalidArgument, "object build layer has no assets", "assets", {}, "procgen.objectBuild"));
506 if (orient_ && !orientation)
507 return Result<PointSet>::failure(Diagnostic::error(
508 DiagnosticCode::InvalidArgument, "orientation is enabled but no orientation layer was supplied", {}, {},
509 "procgen.objectBuild"));
510 float totalWeight = 0.f;
511 for (const auto& asset : assets_) totalWeight += asset.weight;
512 if (!std::isfinite(totalWeight) || totalWeight <= 0.f)
513 return Result<PointSet>::failure(Diagnostic::error(
514 DiagnosticCode::InvalidArgument, "total asset weight is invalid", {}, {}, "procgen.objectBuild"));
515
516 std::size_t childCount = 0;
517 for (const auto& child : children_) childCount += std::size_t(child.count);
518 PointSet output;
519 output.reserve(source.points().size() * (childCount + 1));
520 for (std::size_t sourceIndex = 0; sourceIndex < source.points().size(); ++sourceIndex) {
521 const auto& input = source.points()[sourceIndex];
522 RandomStream random{pointKey(input, sourceIndex, seed_)};
523 float selection = random.unit() * totalWeight;
524 const Asset* selected = &assets_.back();
525 for (const auto& asset : assets_) {
526 selection -= asset.weight;
527 if (selection < 0.f) {
528 selected = &asset;
529 break;
530 }
531 }
532
533 auto appended = output.appendPointFrom(source, sourceIndex);
534 if (!appended.ok()) return Result<PointSet>::failure(appended.status());
535 const int parentIndex = appended.value();
536 auto& parent = output.mutablePoint(std::size_t(parentIndex));
537 if (positionRadius_ > 0.f) {
538 const float angle = random.unit() * 6.283185307179586f;
539 const float radius = std::sqrt(random.unit()) * positionRadius_;
540 parent.x += std::cos(angle) * radius;
541 parent.z += std::sin(angle) * radius;
542 }
543 parent.x += offsetX_;
544 parent.y += offsetY_;
545 parent.z += offsetZ_;
546 if (placeOnTop_) {
547 const char* attribute = useLowest_ ? "surface_lowest_y" : "surface_highest_y";
548 if (source.hasFloatAttribute(int(sourceIndex), attribute))
549 parent.y = source.getFloatAttribute(int(sourceIndex), attribute, parent.y) + topOffset_ + offsetY_;
550 }
551 applyRandomTransform(parent, transform_, random);
552 parent.scaleX *= scaleX_;
553 parent.scaleY *= scaleY_;
554 parent.scaleZ *= scaleZ_;
555 if (orient_)
556 parent.yaw = orientedYaw(input, *orientation, orientationCellSize_, invertOrientation_) +
557 orientationYawOffset_ + random.between(transform_.minYaw, transform_.maxYaw);
558 auto tagged = tagPoint(output, parentIndex, selected->id, "parent", -1);
559 if (!tagged.ok()) return Result<PointSet>::failure(tagged.status());
560
561 for (std::size_t ruleIndex = 0; ruleIndex < children_.size(); ++ruleIndex) {
562 const auto& rule = children_[ruleIndex];
563 RandomStream childRandom{mix(pointKey(input, sourceIndex, seed_) ^ (std::uint64_t(ruleIndex + 1) << 32U))};
564 for (int childOrdinal = 0; childOrdinal < rule.count; ++childOrdinal) {
565 ProcgenPoint child = parent;
566 child.id = derivePointId(parent.id != 0 ? parent.id : pointKey(input, sourceIndex, seed_),
567 std::uint64_t(ruleIndex) * 1000000ULL + std::uint64_t(childOrdinal + 1));
568 const float angle = childRandom.unit() * 6.283185307179586f;
569 const float radius = std::sqrt(childRandom.unit()) * rule.radius;
570 child.x += std::cos(angle) * radius;
571 child.z += std::sin(angle) * radius;
572 child.pitch = 0.f;
573 child.yaw = 0.f;
574 child.roll = 0.f;
575 child.scaleX = 1.f;
576 child.scaleY = 1.f;
577 child.scaleZ = 1.f;
578 applyRandomTransform(child, rule.transform, childRandom);
579 const int childIndex = output.appendPoint(child);
580 tagged = tagPoint(output, childIndex, rule.asset, "child", parentIndex);
581 if (!tagged.ok()) return Result<PointSet>::failure(tagged.status());
582 }
583 }
584 }
585 return Result<PointSet>::success(std::move(output));
586}

614Result<void> ObjectBuildLayer::deserializeDefinition(std::string_view definition) {
615 ObjectBuildLayer replacement;
616 std::istringstream input{std::string(definition)};
617 std::string magic;
618 int version = 0;
619 if (!(input >> magic >> version) || magic != "EVPCG_OBJECT_LAYER" || version != 1)
620 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "invalid object layer header",
621 {}, {}, "procgen.objectBuild"));
622 std::string line;
623 std::getline(input, line);
624 bool ended = false;
625 while (std::getline(input, line)) {
626 if (line.empty()) continue;
627 std::istringstream record(line);
628 std::string kind;
629 record >> kind;
630 if (kind == "END") {
631 ended = true;
632 break;
633 }
634 if (kind == "SEED") {
635 std::uint32_t seed = 0;
636 if (!(record >> seed))
637 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "invalid SEED record",
638 {}, {}, "procgen.objectBuild"));
639 replacement.setSeed(seed);
640 } else if (kind == "OFFSET") {
641 float x = 0.f, y = 0.f, z = 0.f;
642 if (!(record >> x >> y >> z) || !std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z))
643 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "invalid OFFSET record",
644 {}, {}, "procgen.objectBuild"));
645 replacement.setLayerOffset(x, y, z);
646 } else if (kind == "SCALE") {
647 float x = 0.f, y = 0.f, z = 0.f;
648 if (!(record >> x >> y >> z))
649 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "invalid SCALE record",
650 {}, {}, "procgen.objectBuild"));
651 auto applied = replacement.setLayerScale(x, y, z);
652 if (!applied.ok()) return applied;
653 } else if (kind == "POSITION_RADIUS") {
654 float radius = 0.f;
655 if (!(record >> radius))
656 return Result<void>::failure(Diagnostic::error(
657 DiagnosticCode::InvalidArgument, "invalid POSITION_RADIUS record", {}, {}, "procgen.objectBuild"));
658 auto applied = replacement.setPositionRadius(radius);
659 if (!applied.ok()) return applied;
660 } else if (kind == "RANDOM") {
661 ObjectTransformRange range;
662 int uniform = 0;
663 if (!(record >> range.minPitch >> range.maxPitch >> range.minYaw >> range.maxYaw >> range.minRoll >>
664 range.maxRoll >> range.minScaleX >> range.maxScaleX >> range.minScaleY >> range.maxScaleY >>
665 range.minScaleZ >> range.maxScaleZ >> uniform) ||
666 (uniform != 0 && uniform != 1))
667 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "invalid RANDOM record",
668 {}, {}, "procgen.objectBuild"));
669 range.uniformScale = uniform != 0;
670 auto applied = replacement.setRandomTransform(range);
671 if (!applied.ok()) return applied;
672 } else if (kind == "ORIENTATION") {
673 int enabled = 0, invert = 0;
674 float cell = 0.f, offset = 0.f;
675 if (!(record >> enabled >> cell >> offset >> invert) || (enabled != 0 && enabled != 1) ||
676 (invert != 0 && invert != 1))
677 return Result<void>::failure(Diagnostic::error(
678 DiagnosticCode::InvalidArgument, "invalid ORIENTATION record", {}, {}, "procgen.objectBuild"));
679 if (enabled) {
680 auto applied = replacement.setOrientation(cell, offset, invert != 0);
681 if (!applied.ok()) return applied;
682 }
683 } else if (kind == "PLACE") {
684 int enabled = 0, lowest = 0;
685 float offset = 0.f;
686 if (!(record >> enabled >> lowest >> offset) || (enabled != 0 && enabled != 1) ||
687 (lowest != 0 && lowest != 1) || !std::isfinite(offset))
688 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "invalid PLACE record",
689 {}, {}, "procgen.objectBuild"));
690 replacement.setPlaceOnTop(enabled != 0, lowest != 0, offset);
691 } else if (kind == "ASSET") {
692 std::string asset;
693 float weight = 0.f;
694 if (!(record >> std::quoted(asset) >> weight))
695 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "invalid ASSET record",
696 {}, {}, "procgen.objectBuild"));
697 auto applied = replacement.addAsset(std::move(asset), weight);
698 if (!applied.ok()) return applied;
699 } else if (kind == "CHILD") {
700 std::string asset;
701 int count = 0, uniform = 0;
702 float radius = 0.f;
703 ObjectTransformRange range;
704 if (!(record >> std::quoted(asset) >> count >> radius >> range.minPitch >> range.maxPitch >> range.minYaw >>
705 range.maxYaw >> range.minRoll >> range.maxRoll >> range.minScaleX >> range.maxScaleX >>
706 range.minScaleY >> range.maxScaleY >> range.minScaleZ >> range.maxScaleZ >> uniform) ||
707 (uniform != 0 && uniform != 1))
708 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "invalid CHILD record",
709 {}, {}, "procgen.objectBuild"));
710 range.uniformScale = uniform != 0;
711 auto applied = replacement.addChildRule(std::move(asset), count, radius, range);
712 if (!applied.ok()) return applied;
713 } else {
714 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
715 "unknown object layer record: " + kind, {}, {},
716 "procgen.objectBuild"));
717 }
718 record >> std::ws;
719 if (!record.eof())
720 return Result<void>::failure(Diagnostic::error(
721 DiagnosticCode::InvalidArgument, "trailing object layer record data", {}, {}, "procgen.objectBuild"));
722 }
723 if (!ended)
724 return Result<void>::failure(Diagnostic::error(
725 DiagnosticCode::InvalidArgument, "object layer END record is missing", {}, {}, "procgen.objectBuild"));
726 *this = std::move(replacement);
727 return Result<void>::success();
728}
