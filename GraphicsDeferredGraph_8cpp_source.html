12void Graphics::resetDeferredFrameGraphs() {
13 // Graph command pools/framebuffers borrow these targets. Retire every
14 // submitted slot before destroying graphs, then destroy target images.
15 if (device.instance) device->waitIdle();
16 for (auto &graph : deferredFrameGraphs_) graph.reset();
17 deferredGraphRecorded_ = false;
18 deferredGraphRecordedSlot_ = 0;
19}

26void Graphics::buildDeferredFrameGraphs() {
27 // One FrameGraph per in-flight slot imports the engine-owned targets and
28 // owns the deferred passes: the 3 CSM cascades (per-layer views of the
29 // shadow array) + the G-buffer fill share one dependency-free layer, so the
30 // JobSystem executor records all four command buffers concurrently (see
31 // recordDeferredFrameGraph). The engine keeps image ownership so
32 // renderEntityIdMask / readGBufferToImageData and the postFX wrappers are
33 // unaffected. Each graph is only used on its slot's frames, so its command
34 // buffer is reused two frames later — by then the present slot fence
35 // guarantees the previous graph submit completed (same queue, submitted
36 // before the present command buffer).
37 const vk::Format depthFmt = vk::Format::eD32Sfloat;
38 const vk::Format colorFmt = pickGBufferColorFormat(device);
39 const uint32_t mapSize = uint32_t(ShadowConfig::kMapSize);
40 const uint32_t shadowLayers = uint32_t(ShadowConfig::kCascades);
41 const uint32_t w = gbufferWidth > 0 ? uint32_t(gbufferWidth) : 1u;
42 const uint32_t h = gbufferHeight > 0 ? uint32_t(gbufferHeight) : 1u;
43
44 for (size_t i = 0; i < deferredFrameGraphs_.size(); ++i) {
45 auto graph = std::make_unique<vkb::FrameGraph>(&device, 1);
46
47 vkb::TextureDesc shadowDesc;
48 shadowDesc.format = depthFmt;
49 shadowDesc.extent = vk::Extent3D{mapSize, mapSize, 1};
50 shadowDesc.arrayLayers = shadowLayers;
51 shadowDesc.aspect = vk::ImageAspectFlagBits::eDepth;
52 shadowDesc.usage = vk::ImageUsageFlagBits::eSampled | vk::ImageUsageFlagBits::eDepthStencilAttachment;
53 // Shadow descriptors sample the depth array in SHADER_READ_ONLY, matching
54 // the legacy shadow pass and the layout of every imported cascade.
55 shadowDesc.afterLayout = vk::ImageLayout::eShaderReadOnlyOptimal;
56 vk::ClearValue shadowClear{};
57 shadowClear.depthStencil = vk::ClearDepthStencilValue{1.0f, 0};
58 const bool haveShadowSlot = i < shadowMaps.size() && shadowMaps[i].image.layerCount() >= shadowLayers;
59 if (haveShadowSlot) {
60 const vk::Image shadowImage = shadowMaps[i].image.image();
61 for (uint32_t c = 0; c < shadowLayers; ++c) {
62 auto shadowH = graph->importTexture("shadowCascade" + std::to_string(c), shadowImage,
63 shadowMaps[i].image.layerView(c), shadowDesc);
64 graph->addPass("shadow" + std::to_string(c))
65 .depthAttachment(shadowH, vkb::AttachmentOp::clear(shadowClear))
66 .record([this, c](vkb::FrameGraphPassContext &ctx) { recordShadowCascadePass(ctx, int(c)); });
67 }
68 }
69
70 if (i < gbufferSlots.size() && gbufferWidth > 0 && gbufferHeight > 0) {
71 auto &slot = gbufferSlots[i];
72 vkb::TextureDesc colorDesc;
73 colorDesc.format = colorFmt;
74 colorDesc.extent = vk::Extent3D{w, h, 1};
75 colorDesc.usage = vk::ImageUsageFlagBits::eSampled | vk::ImageUsageFlagBits::eColorAttachment |
76 vk::ImageUsageFlagBits::eTransferSrc;
77 colorDesc.afterLayout = vk::ImageLayout::eShaderReadOnlyOptimal;
78 auto normalH = graph->importTexture("gbNormal", slot.normal.image(), slot.normal.imageView(), colorDesc);
79 auto depthColorH =
80 graph->importTexture("gbDepthColor", slot.depthColor.image(), slot.depthColor.imageView(), colorDesc);
81 auto albedoH = graph->importTexture("gbAlbedo", slot.albedo.image(), slot.albedo.imageView(), colorDesc);
82
83 vkb::TextureDesc depthDesc;
84 depthDesc.format = depthFmt;
85 depthDesc.extent = vk::Extent3D{w, h, 1};
86 depthDesc.aspect = vk::ImageAspectFlagBits::eDepth;
87 depthDesc.usage = vk::ImageUsageFlagBits::eSampled | vk::ImageUsageFlagBits::eDepthStencilAttachment;
88 depthDesc.afterLayout = vk::ImageLayout::eShaderReadOnlyOptimal;
89 auto depthH = graph->importTexture("gbHwDepth", slot.depth.image(), slot.depth.imageView(), depthDesc);
90
91 std::array<vk::ClearValue, 4> clears{};
92 clears[0].color = vk::ClearColorValue(std::array<float, 4>{0, 0, 0, 0});
93 clears[1].color = vk::ClearColorValue(std::array<float, 4>{1, 1, 1, 1});
94 clears[2].color = vk::ClearColorValue(std::array<float, 4>{0, 0, 0, 0});
95 clears[3].depthStencil = vk::ClearDepthStencilValue{1.0f, 0};
96 graph->addPass("gbuffer")
97 .colorAttachment(normalH, vkb::AttachmentOp::clear(clears[0]))
98 .colorAttachment(depthColorH, vkb::AttachmentOp::clear(clears[1]))
99 .colorAttachment(albedoH, vkb::AttachmentOp::clear(clears[2]))
100 .depthAttachment(depthH, vkb::AttachmentOp::clear(clears[3]))
101 .record([this](vkb::FrameGraphPassContext &ctx) { recordGBufferPassDraws(ctx); });
102 }
103 graph->compile();
104 deferredFrameGraphs_[i] = std::move(graph);
105 }
106}

108void Graphics::recordShadowCascadePass(vkb::FrameGraphPassContext &ctx, int cascade) {
109 // Runs inside the FrameGraph's "shadow<cascade>" render-pass instance
110 // (already begun with a depth clear); only draw commands go here. The pass
111 // may be recorded on a JobSystem worker, so everything below must be
112 // read-only: shadowCascadeDraws was captured by endShadowPass on the main
113 // thread before the graph records.
114 auto &cb = ctx.commandBuffer();
115 const vk::Extent2D extent = ctx.extent();
116 const uint32_t size = extent.width ? extent.width : uint32_t(ShadowConfig::kMapSize);
117 setViewportAndScissor(cb, size, size);
118 vk::Pipeline boundPipeline{};
119 for (const auto &d : shadowCascadeDraws[cascade]) {
120 if (!d.mesh || !d.mesh->gpuHandle) continue;
121 const bool wantAlpha = d.alphaTest && shadowAlphaPipeline;
122 const bool skinned = d.skinSet && d.mesh->hasGpuSkinning();
123 vk::Pipeline wanted = skinned ? (wantAlpha ? shadowSkinAlphaPipeline : shadowSkinPipeline)
124 : (wantAlpha ? shadowAlphaPipeline : shadowPipeline);
125 if (wanted != boundPipeline) {
126 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, wanted);
127 boundPipeline = wanted;
128 }
129 auto *gpuMesh = static_cast<GpuMesh *>(d.mesh->gpuHandle);
130 if (skinned) {
131 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, skinPassPipelineLayout, 0, 1, &d.skinSet, 1,
132 &d.skinUboOffset);
133 } else if (wantAlpha) {
134 Texture *alb = d.albedo ? d.albedo : whiteTexture;
135 if (alb && alb->gpuHandle && texSetLayout) {
136 auto *gpuTex = static_cast<GpuTexture *>(alb->gpuHandle);
137 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, shadowAlphaPipelineLayout, 0, 1,
138 gpuTex->descriptorSet.ptr(), 0, nullptr);
139 }
140 }
141 if (!skinned)
142 cb.pushConstants(wantAlpha ? shadowAlphaPipelineLayout : shadowPipelineLayout,
143 vk::ShaderStageFlagBits::eVertex, 0, sizeof(glm::mat4), &d.mvp);
144 drawIndexedMesh(cb, *gpuMesh);
145 }
146}

148void Graphics::recordGBufferPassDraws(vkb::FrameGraphPassContext &ctx) {
149 // Runs inside the FrameGraph's "gbuffer" render-pass instance (already
150 // begun with the planned clear values); only draw commands go here. The
151 // pass may be recorded on a JobSystem worker, so everything below must be
152 // read-only: gbufferPassDraws was captured on the main thread.
153 auto &cb = ctx.commandBuffer();
154 const vk::Extent2D extent = ctx.extent();
155 const uint32_t w = extent.width ? extent.width : uint32_t(gbufferWidth);
156 const uint32_t h = extent.height ? extent.height : uint32_t(gbufferHeight);
157 setViewportAndScissor(cb, w, h);
158 vk::Pipeline boundPipeline{};
159 for (const auto &d : gbufferPassDraws) {
160 if (!d.mesh || !d.mesh->gpuHandle) continue;
161 const bool wantAlpha = d.alphaTest && gbufferAlphaPipeline;
162 const bool skinned = d.skinSet && d.mesh->hasGpuSkinning();
163 vk::Pipeline wanted = skinned ? (wantAlpha ? gbufferSkinAlphaPipeline : gbufferSkinPipeline)
164 : (wantAlpha ? gbufferAlphaPipeline : gbufferPipeline);
165 if (wanted != boundPipeline) {
166 cb.bindPipeline(vk::PipelineBindPoint::eGraphics, wanted);
167 boundPipeline = wanted;
168 }
169 auto *gpuMesh = static_cast<GpuMesh *>(d.mesh->gpuHandle);
170 Texture *alb = d.albedo ? d.albedo : whiteTexture;
171 if (skinned) {
172 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, skinPassPipelineLayout, 0, 1, &d.skinSet, 1,
173 &d.skinUboOffset);
174 } else if (alb && alb->gpuHandle && texSetLayout) {
175 auto *gpuTex = static_cast<GpuTexture *>(alb->gpuHandle);
176 cb.bindDescriptorSets(vk::PipelineBindPoint::eGraphics, gbufferPipelineLayout, 0, 1,
177 gpuTex->descriptorSet.ptr(), 0, nullptr);
178 }
179 if (!skinned)
180 cb.pushConstants(gbufferPipelineLayout,
181 vk::ShaderStageFlagBits::eVertex | vk::ShaderStageFlagBits::eFragment, 0,
182 sizeof(GBufferPush), &d.push);
183 drawIndexedMesh(cb, *gpuMesh);
184 }

187void Graphics::recordDeferredFrameGraph() {
188 // Voxel / script begin3DFrame paths never fill G-buffer or CSM. Submitting
189 // the empty four-pass graph every frame still waits on the JobSystem and
190 // can hitch the orbit camera (MAILBOX then shows a newer pose, then an
191 // older in-flight image — a short reverse jump).
192 if (!gbufferPending && shadowPendingMask == 0) {
193 dropPendingOffscreenPasses();
194 return;
195 }
196 const size_t slot = currentFrameSlot();
197 if (deferredGraphRecorded_ && deferredGraphRecordedSlot_ == slot) {
198 // render3D can be called several times per script frame (e.g. the
199 // render tests call it 3x before present). The deferred graph's command
200 // buffers must only be recorded once per slot per frame — re-recording
201 // them while the previous submit is still in flight would reset
202 // in-use command buffers (UB, GPU hang).
203 return;
204 }
205 auto *graph = currentDeferredFrameGraph();
206 if (!graph && (!shadowMaps.empty() || !gbufferSlots.empty())) {
207 // Shadows can be enabled without a G-buffer pass (or vice versa); build
208 // the deferred graphs on demand from whatever targets exist today.
209 buildDeferredFrameGraphs();
210 graph = currentDeferredFrameGraph();
211 }
212 if (!graph || !gbufferPipeline || !gbufferRenderPass || !shadowPipeline) {
213 dropPendingOffscreenPasses();
214 return;
215 }
216 // Record the declarative deferred passes (3 CSM cascades + G-buffer, one
217 // independent layer) with the JobSystem executor — the four command
218 // buffers are recorded concurrently on workers — then submit them on the
219 // graphics queue before the swapchain pass begins. Layout transitions and
220 // the render-pass instances are planned by the FrameGraph. Same-queue
221 // submission order plus the present slot fence (waited in Present::begin)
222 // keep this slot's graph command buffers safe to reuse two frames later.
223 auto *jobs = thread::Thread::create()->getJobSystem();
224 jobs->beginFrame(); // idempotent wait; recycles the per-frame arena
225 // Re-plan every frame (cheap; device objects are cached) so the graph is
226 // in the compiled phase for this record cycle — vkb::FrameGraph enforces
227 // build -> compile -> record -> submit and record() exactly once per
228 // compile.
229 graph->compile();
230 // Parallel executor: each pass owns a dedicated command pool (one pool per
231 // frame slot per pass), so the workers never share a pool while recording
232 // concurrently — the Vulkan external-synchronization rule for command
233 // pools is satisfied structurally.
234 recordFrameGraphWithJobSystem(*graph, jobs);
235 graph->submit();
236 if (!shadowMaps.empty()) currentShadowMap().image.setCurrentLayout(vk::ImageLayout::eDepthStencilReadOnlyOptimal);
237 jobs->endFrame();
238 for (auto &d : shadowCascadeDraws) d.clear();
239 gbufferPassDraws.clear();
240 shadowPendingMask = 0;
241 gbufferPending = false;
242 deferredGraphRecorded_ = true;
243 deferredGraphRecordedSlot_ = slot;
244}
