27eve::Result<void> ShapeBackend::setDensity(float density, bool updateMass) {
28 if (!isValid()) return rejected<void>("Shape expired", eve::DiagnosticCode::StaleHandle);
29 if (!std::isfinite(density) || density < 0)
30 return rejected<void>("Invalid density", eve::DiagnosticCode::InvalidArgument);
31 if (density == shape->density) return success();
32 auto cooked = cook(shape->geometry, density, reinterpret_cast<JPH::uint64>(shape.get()));
33 if (!cooked) return eve::Result<void>::failure(cooked.status());
34 auto previous = shape->native;
35 shape->native = std::move(cooked).takeValue();
36 if (!updateMass) {
37 // Native compounds retain the old leaf until the next explicit mass/topology rebuild.
38 // Geometry and attribution are unchanged; only the pending density differs.
39 shape->density = density;
40 return success();
41 }
42 auto body = shape->body.lock();
43 auto replaced = replaceShapes(*body, body->shapes);
44 if (!replaced) {
45 shape->native = previous;
46 return replaced;
47 }
48 shape->density = density;
49 return success();
50}

83eve::Result<void> validateMaterial(SurfaceMaterial3D value) {
84 if (!isFinite(value.friction, value.restitution, value.rollingResistance) || value.friction < 0 ||
85 value.restitution < 0 || value.restitution > 1 || value.rollingResistance < 0)
86 return rejected<void>("Invalid material", eve::DiagnosticCode::InvalidArgument);
87 if (!isFinite(value.tangentVelocity.x, value.tangentVelocity.y, value.tangentVelocity.z) ||
88 ((value.userMaterialId >> 32) & 7) > 4 || ((value.userMaterialId >> 35) & 7) > 4)
89 return rejected<void>("Invalid material policy", eve::DiagnosticCode::InvalidArgument);
90 return success();
91}

155eve::Result<bool> ShapeBackend::testPoint(float x, float y, float z) const {
156 if (!isValid()) return rejected<bool>("Shape expired", eve::DiagnosticCode::StaleHandle);
157 if (!isFinite(x, y, z)) return rejected<bool>("Invalid point", eve::DiagnosticCode::InvalidArgument);
158 const auto body = shape->body.lock();
159 const auto& g = shape->geometry;
160 const auto pose = body->api().GetWorldTransform(body->id) *
161 JPH::RMat44::sRotationTranslation(JPH::Quat(g.localQx, g.localQy, g.localQz, g.localQw),
162 JPH::RVec3(g.localX, g.localY, g.localZ));
163 const auto local =
164 JPH::Vec3(pose.InversedRotationTranslation() * JPH::RVec3(x, y, z)) - shape->native->GetCenterOfMass();
165 JPH::AllHitCollisionCollector<JPH::CollidePointCollector> hits;
166 shape->native->CollidePoint(local, JPH::SubShapeIDCreator(), hits);
167 return eve::Result<bool>::success(!hits.mHits.empty());
168}
