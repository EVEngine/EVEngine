320Result<int> ImageData::scalePcg(int newWidth, int newHeight, ImageScaleFilter filter) {
321 const auto invalid = [](const char* message) {
322 return Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, message, "image.scalePcg"));
323 };
324 if (newWidth <= 0 || newHeight <= 0) return invalid("Pcg texture scale requires positive destination dimensions");
325 if (filter != ImageScaleFilter::Point && filter != ImageScaleFilter::Bilinear)
326 return invalid("Pcg texture scale requires point or bilinear filtering");
327 if (getWidth() <= 0 || getHeight() <= 0) return invalid("Pcg texture scale requires a non-empty source image");
328 if (filter == ImageScaleFilter::Bilinear && (getWidth() < 2 || getHeight() < 2))
329 return invalid("Pcg bilinear scale requires both source dimensions to be at least two");
330 const auto count = std::uint64_t(newWidth) * std::uint64_t(newHeight);
331 if (count > std::uint64_t(std::numeric_limits<int>::max()))
332 return invalid("Pcg texture scale destination exceeds the supported pixel count");
333
334 ImageData candidate(newWidth, newHeight, getFormat());
335 const float ratioX = filter == ImageScaleFilter::Bilinear ? float(getWidth() - 1) / float(newWidth)
336 : float(getWidth()) / float(newWidth);
337 const float ratioY = filter == ImageScaleFilter::Bilinear ? float(getHeight() - 1) / float(newHeight)
338 : float(getHeight()) / float(newHeight);
339 const auto lerp = [](const Colorf& first, const Colorf& second, float amount) {
340 return Colorf{first.r + (second.r - first.r) * amount, first.g + (second.g - first.g) * amount,
341 first.b + (second.b - first.b) * amount, first.a + (second.a - first.a) * amount};
342 };
343 for (int y = 0; y < newHeight; ++y) {
344 if (filter == ImageScaleFilter::Point) {
345 const int sourceY = int(ratioY * float(y));
346 for (int x = 0; x < newWidth; ++x) candidate.setPixel(x, y, getPixel(int(ratioX * float(x)), sourceY));
347 continue;
348 }
349 const float sourceY = float(y) * ratioY;
350 const int yFloor = int(std::floor(sourceY));
351 const float yLerp = sourceY - float(yFloor);
352 for (int x = 0; x < newWidth; ++x) {
353 const float sourceX = float(x) * ratioX;
354 const int xFloor = int(std::floor(sourceX));
355 const float xLerp = sourceX - float(xFloor);
356 const auto topLeft = getPixel(xFloor, yFloor);
357 const auto topRight = getPixel(xFloor + 1, yFloor);
358 const auto bottomLeft = getPixel(xFloor, yFloor + 1);
359 const auto bottomRight = getPixel(xFloor + 1, yFloor + 1);
360 const auto top = lerp(topLeft, topRight, xLerp);
361 const auto bottom = lerp(bottomLeft, bottomRight, xLerp);
362 candidate.setPixel(x, y, lerp(top, bottom, yLerp));
363 }
364 }
365 adopt(candidate);
366 return Result<int>::success(int(count));
367}
