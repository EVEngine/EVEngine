91std::vector<eve::script::Owned<ImageData>> Image::newCubeFaces(ImageData* src) {
92 // The faces array is always ordered +x, -x, +y, -y, +z, -z.
93 std::vector<eve::script::Owned<ImageData>> faces;
94
95 int totalW = src->getWidth();
96 int totalH = src->getHeight();
97
98 if (totalW % 3 == 0 && totalH % 4 == 0 && totalW / 3 == totalH / 4) {
99 // +y
100 // +z +x -z
101 // -y
102 // -x
103
104 int w = totalW / 3;
105 int h = totalH / 4;
106
107 faces.emplace_back(newPastedImageData(src, 1 * w, 1 * h, w, h));
108 faces.emplace_back(newPastedImageData(src, 1 * w, 3 * h, w, h));
109 faces.emplace_back(newPastedImageData(src, 1 * w, 0 * h, w, h));
110 faces.emplace_back(newPastedImageData(src, 1 * w, 2 * h, w, h));
111 faces.emplace_back(newPastedImageData(src, 0 * w, 1 * h, w, h));
112 faces.emplace_back(newPastedImageData(src, 2 * w, 1 * h, w, h));
113 } else if (totalW % 4 == 0 && totalH % 3 == 0 && totalW / 4 == totalH / 3) {
114 // +y
115 // -x +z +x -z
116 // -y
117
118 int w = totalW / 4;
119 int h = totalH / 3;
120
121 faces.emplace_back(newPastedImageData(src, 2 * w, 1 * h, w, h));
122 faces.emplace_back(newPastedImageData(src, 0 * w, 1 * h, w, h));
123 faces.emplace_back(newPastedImageData(src, 1 * w, 0 * h, w, h));
124 faces.emplace_back(newPastedImageData(src, 1 * w, 2 * h, w, h));
125 faces.emplace_back(newPastedImageData(src, 1 * w, 1 * h, w, h));
126 faces.emplace_back(newPastedImageData(src, 3 * w, 1 * h, w, h));
127 } else if (totalH % 6 == 0 && totalW == totalH / 6) {
128 // +x
129 // -x
130 // +y
131 // -y
132 // +z
133 // -z
134
135 int w = totalW;
136 int h = totalH / 6;
137
138 for (int i = 0; i < 6; i++) faces.emplace_back(newPastedImageData(src, 0, i * h, w, h));
139 } else if (totalW % 6 == 0 && totalW / 6 == totalH) {
140 // +x -x +y -y +z -z
141
142 int w = totalW / 6;
143 int h = totalH;
144
145 for (int i = 0; i < 6; i++) faces.emplace_back(newPastedImageData(src, i * w, 0, w, h));
146 } else
147 throw eve::Exception("Unknown cubemap image dimensions!");
148
149 return faces;
150}

152std::vector<eve::script::Owned<ImageData>> Image::newVolumeLayers(ImageData* src) {
153 std::vector<eve::script::Owned<ImageData>> layers;
154
155 int totalW = src->getWidth();
156 int totalH = src->getHeight();
157
158 if (totalW % totalH == 0) {
159 for (int i = 0; i < totalW / totalH; i++)
160 layers.emplace_back(newPastedImageData(src, i * totalH, 0, totalH, totalH));
161 } else if (totalH % totalW == 0) {
162 for (int i = 0; i < totalH / totalW; i++)
163 layers.emplace_back(newPastedImageData(src, 0, i * totalW, totalW, totalW));
164 } else
165 throw eve::Exception("Cannot extract volume layers from source ImageData.");
166
167 return layers;
168}
