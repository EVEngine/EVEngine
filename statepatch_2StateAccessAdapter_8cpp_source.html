67eve::Result<eve::MutationReceipt> StatePatchStateAdapter::apply(std::span<const eve::StateMutation> mutations,
68 const eve::MutationContext& context) {
69 if (context.transactionId.empty())
70 return failure(eve::DiagnosticCode::InvalidArgument, "persistent dialogue mutations require a transaction id",
71 "transactionId");
72
73 using Key = std::pair<std::string, std::string>;
74 std::map<Key, std::optional<eve::Value>> pending;
75 auto batchReference = store_.newBatch();
76 if (!batchReference) return eve::Result<eve::MutationReceipt>::failure(batchReference.status());
77 const PatchBatchHandleRef batchHandle = std::move(batchReference).takeValue();
78 auto batch = store_.resolveBatch(batchHandle);
79 if (!batch.isBound())
80 return failure(eve::DiagnosticCode::StaleHandle, "StatePatch patch batch became stale during allocation",
81 "batch");
82 struct BatchLease {
83 Store& store;
84 PatchBatchHandleRef handle;
85 ~BatchLease() { store.releaseBatch(handle).ignore("release StatePatch adapter batch"); }
86 } batchLease{store_, batchHandle};
87
88 for (std::size_t index = 0; index < mutations.size(); ++index) {
89 const auto& mutation = mutations[index];
90 const std::string path = "mutations[" + std::to_string(index) + "]";
91 if (!mutation.persistent)
92 return failure(eve::DiagnosticCode::Unsupported, "StatePatch adapter accepts persistent mutations only",
93 path);
94 if (mutation.subject.empty() || mutation.key.empty())
95 return failure(eve::DiagnosticCode::InvalidArgument, "persistent mutation requires subject and key", path);
96
97 const std::string sourceKey = mutation.key;
98 const std::string key =
99 mutation.kind == eve::MutationKind::AddTag || mutation.kind == eve::MutationKind::RemoveTag
100 ? tagKey(sourceKey)
101 : sourceKey;
102 const Key pendingKey{mutation.subject, key};
103 auto current = [&]() -> std::optional<eve::Value> {
104 const auto found = pending.find(pendingKey);
105 if (found != pending.end()) return found->second;
106 if (!store_.has(mutation.subject, key)) return std::nullopt;
107 auto existing = readJsonValue(store_, mutation.subject, key);
108 if (!existing) return std::nullopt;
109 return std::move(existing).takeValue();
110 };
111
112 bool appended = false;
113 switch (mutation.kind) {
114 case eve::MutationKind::Set: {
115 auto json = mutation.value.toJson();
116 if (!json)
117 return failure(json.error() ? json.error()->code() : eve::DiagnosticCode::SerializationError,
118 json.error() ? json.error()->message() : "state value cannot be serialized",
119 path + ".value");
120 appended = batch->set(mutation.subject, key, json.value());
121 if (appended) pending[pendingKey] = mutation.value;
122 break;
123 }
124 case eve::MutationKind::Remove:
125 appended = batch->remove(mutation.subject, key);
126 if (appended) pending[pendingKey] = std::nullopt;
127 break;
128 case eve::MutationKind::AddTag:
129 appended = batch->set(mutation.subject, key, "true");
130 if (appended) pending[pendingKey] = eve::Value(true);
131 break;
132 case eve::MutationKind::RemoveTag:
133 appended = batch->remove(mutation.subject, key);
134 if (appended) pending[pendingKey] = std::nullopt;
135 break;
136 case eve::MutationKind::AddNumber: {
137 double delta = 0.0;
138 if (!numeric(mutation.value, delta))
139 return failure(eve::DiagnosticCode::InvalidArgument, "AddNumber requires a finite numeric value",
140 path + ".value");
141 const auto existing = current();
142 double base = 0.0;
143 if (existing && !numeric(*existing, base))
144 return failure(eve::DiagnosticCode::InvalidArgument,
145 "AddNumber cannot add to a non-numeric state value", path);
146 const double total = base + delta;
147 if (!std::isfinite(total))
148 return failure(eve::DiagnosticCode::InvalidArgument, "AddNumber result must be finite", path);
149 const eve::Value result(total);
150 auto json = result.toJson();
151 if (!json)
152 return failure(eve::DiagnosticCode::SerializationError, "AddNumber result cannot be serialized",
153 path);
154 appended = batch->set(mutation.subject, key, json.value());
155 if (appended) pending[pendingKey] = result;
156 break;
157 }
158 }
159 if (!appended) return failure(eve::DiagnosticCode::Failed, "StatePatch rejected a mutation", path);
160 }
161
162 eve::transaction::TransactionContext transaction(context.transactionId, context.correlationId, context.causationId);
163 StoreTransactionParticipant participant(store_, *batch);
164 std::array<eve::transaction::ITransactionParticipant*, 1> participants{&participant};
165 eve::transaction::Coordinator coordinator;
166 auto committed = coordinator.execute(transaction, participants);
167 if (!committed) return eve::Result<eve::MutationReceipt>::failure(committed.status());
168
169 const PatchResult& result = batch->result();
170 return eve::Result<eve::MutationReceipt>::success(
171 eve::MutationReceipt{context.transactionId, static_cast<std::size_t>(result.changedCount)},
172 eve::Status::success(result.changedCount == 0 ? eve::StatusCode::NoOp : eve::StatusCode::Applied));
173}
