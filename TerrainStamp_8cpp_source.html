17 bool invert) {
18 if (!validRaster(target) || !validRaster(source) || target.getWidth() != source.getWidth() ||
19 target.getHeight() != source.getHeight())
20 return invalid("terrain.mask: finite nonempty rasters with matching dimensions required");
21 if (!std::isfinite(strength) || strength < 0 || strength > 1 || mode < TerrainMaskBlend::Multiply ||
22 mode > TerrainMaskBlend::Subtract)
23 return invalid("terrain.mask: invalid blend mode or strength");
24 std::vector<float> candidate(target.data().size());
25 for (size_t i = 0; i < candidate.size(); ++i) {
26 const double a = target.data()[i], b = invert ? 1.0 - source.data()[i] : source.data()[i];
27 double value = a;
28 switch (mode) {
29 case TerrainMaskBlend::Multiply: value = a * b; break;
30 case TerrainMaskBlend::Maximum: value = std::max(a, b); break;
31 case TerrainMaskBlend::Minimum: value = std::min(a, b); break;
32 case TerrainMaskBlend::Add: value = a + b; break;
33 case TerrainMaskBlend::Subtract: value = a - b; break;
34 }
35 value = std::lerp(a, value, double(strength));
36 if (!isRepresentable(value)) return invalid("terrain.mask: result exceeds finite height range");
37 candidate[i] = float(value);
38 }
39 return publish(target, std::move(candidate));

42Result<int> applyTerrainStamp(Heightmap& target, const Heightmap& stamp, const TerrainStampSettings& s,
43 const Heightmap& localMask, const Heightmap& globalMask) {
44 if (!validRaster(target) || !validRaster(stamp) || !validRaster(localMask) || !validRaster(globalMask))
45 return invalid("terrain.stamp: finite nonempty rasters required");
46 for (double v : {s.originX, s.originZ, s.spacingX, s.spacingZ, s.centerX, s.centerZ, s.width, s.depth, s.rotation,
47 double(s.amplitude), double(s.baseHeight), double(s.blendStrength), double(s.smoothWidth),
48 double(s.edgeFade)})
49 if (!std::isfinite(v)) return invalid("terrain.stamp: settings must be finite");
50 if (s.spacingX <= 0 || s.spacingZ <= 0 || s.width <= 0 || s.depth <= 0 || s.blendStrength < 0 ||
51 s.blendStrength > 1 || s.operation < TerrainStampOperation::Raise ||
52 s.operation > TerrainStampOperation::SmoothRaise || s.smoothWidth < 0 || s.edgeFade < 0)
53 return invalid("terrain.stamp: invalid operation, extent, spacing or blend strength");
54
55 std::vector<float> candidate = target.data();
56 const double c = std::cos(s.rotation), sn = std::sin(s.rotation);
57 for (int y = 0; y < target.getHeight(); ++y) {
58 for (int x = 0; x < target.getWidth(); ++x) {
59 const double dx = (s.originX + x * s.spacingX) - s.centerX;
60 const double dz = (s.originZ + y * s.spacingZ) - s.centerZ;
61 double u = (c * dx + sn * dz) / s.width + 0.5;
62 double v = (-sn * dx + c * dz) / s.depth + 0.5;
63 if (!std::isfinite(u) || !std::isfinite(v))
64 return invalid("terrain.stamp: world-to-stamp coordinates overflow");
65 // A quarter-turn must not lose an edge sample to trigonometric roundoff.
66 constexpr double edgeTolerance = 8 * std::numeric_limits<double>::epsilon();
67 if (u < -edgeTolerance || u > 1 + edgeTolerance || v < -edgeTolerance || v > 1 + edgeTolerance) continue;
68 u = std::clamp(u, 0.0, 1.0);
69 v = std::clamp(v, 0.0, 1.0);
70 double weight = std::clamp(sample(globalMask, u, v), 0.0, 1.0);
71 if (s.operation == TerrainStampOperation::SmoothRaise && s.edgeFade > 0) {
72 const auto fade = [](double distance, double width) {
73 const double t = std::clamp(distance / width, 0.0, 1.0);
74 return t * t * (3.0 - 2.0 * t);
75 };
76 weight *=
77 fade(std::min(u, 1.0 - u) * s.width, s.edgeFade) * fade(std::min(v, 1.0 - v) * s.depth, s.edgeFade);
78 }
79 if (weight == 0) continue;
80 const double old = target.height(x, y);
81 const double level = s.baseHeight + s.amplitude * sample(stamp, u, v) * sample(localMask, u, v);
82 double next = old;
83 switch (s.operation) {
84 case TerrainStampOperation::Raise: next = std::max(old, level); break;
85 case TerrainStampOperation::Lower: next = std::min(old, level); break;
86 case TerrainStampOperation::Set: next = level; break;
87 case TerrainStampOperation::Blend: next = std::lerp(old, level, double(s.blendStrength)); break;
88 case TerrainStampOperation::Add: next = old + level; break;
89 case TerrainStampOperation::Subtract: next = old - level; break;
90 case TerrainStampOperation::SmoothRaise:
91 if (s.smoothWidth == 0 || std::abs(old - level) >= s.smoothWidth) {
92 next = std::max(old, level);
93 break;
94 }
95 if (!isRepresentable(level))
96 return invalid("terrain.stamp: stamp level exceeds finite height range");
97 next = eve::math::smoothMax(float(old), float(level), s.smoothWidth);
98 break;
99 }
100 next = std::lerp(old, next, weight);
101 if (!isRepresentable(next)) return invalid("terrain.stamp: result exceeds finite height range");
102 candidate[size_t(y) * size_t(target.getWidth()) + size_t(x)] = float(next);
103 }
104 }
105 return publish(target, std::move(candidate));
106}
