24Result<AuthorityPlan> LocalWorldAuthority::preflight(const TransactionSpec& transaction,
25 std::span<const DomainOperation> operations) {
26 if (!target_)
27 return authorityError<AuthorityPlan>(Status::Failed, "editor.authority.missing-target",
28 "Local authority has no target");
29 if (transaction.id.empty() || transaction.target.empty())
30 return authorityError<AuthorityPlan>(Status::Rejected, "editor.authority.invalid-transaction",
31 "Transaction id and target are required");
32 if (transaction.target != target_->targetId())
33 return authorityError<AuthorityPlan>(Status::Rejected, "editor.authority.target-mismatch",
34 "Transaction target does not match the authority target");
35 if (transaction.baseRevision != target_->revision())
36 return authorityError<AuthorityPlan>(Status::Conflict, "editor.authority.revision-conflict",
37 "Target changed since the transaction was planned");
38 if (committed_.contains(transaction.id))
39 return authorityError<AuthorityPlan>(Status::Conflict, "editor.authority.duplicate-transaction",
40 "Transaction id was already committed");
41 for (const DomainOperation& operation : operations) {
42 if (operation.type.empty())
43 return authorityError<AuthorityPlan>(Status::Rejected, "editor.operation.missing-type",
44 "Domain operation type is required");
45 if (operation.target != transaction.target)
46 return authorityError<AuthorityPlan>(Status::Rejected, "editor.operation.target-mismatch",
47 "Domain operation target does not match the transaction");
48 }
49
50 AuthorityPlan plan;
51 plan.transaction = transaction;
52 plan.validatedRevision = target_->revision();
53 plan.operations.assign(operations.begin(), operations.end());
54 return eve::editing::applied<AuthorityPlan>(std::move(plan));
55}

57Result<TransactionReceipt> LocalWorldAuthority::commit(const AuthorityPlan& plan) {
58 if (!target_)
59 return authorityError<TransactionReceipt>(Status::Failed, "editor.authority.missing-target",
60 "Local authority has no target");
61 if (plan.validatedRevision != target_->revision())
62 return authorityError<TransactionReceipt>(Status::Conflict, "editor.authority.revision-conflict",
63 "Target changed after authority preflight");
64
65 TransactionReceipt receipt;
66 receipt.id = plan.transaction.id;
67 receipt.state = TransactionState::PendingAuthority;
68 receipt.beforeRevision = target_->revision();
69
70 std::size_t appliedCount = 0;
71 try {
72 for (const DomainOperation& operation : plan.operations) {
73 Result<void> result = target_->applyDomainOperation(operation);
74 if (!result.ok()) {
75 Result<void> rollback = rollbackApplied(plan.operations, appliedCount);
76 receipt.state = rollback.ok() ? TransactionState::Rejected : TransactionState::Failed;
77 receipt.afterRevision = target_->revision();
78 receipt.diagnostics = result.diagnostics();
79 const auto& rollbackDiagnostics = rollback.diagnostics();
80 receipt.diagnostics.insert(receipt.diagnostics.end(), rollbackDiagnostics.begin(),
81 rollbackDiagnostics.end());
82 const Status status = rollback.ok() ? result.code() : Status::Failed;
83 return Result<TransactionReceipt>::failure(eve::Status(status, std::move(receipt.diagnostics)));
84 }
85 ++appliedCount;
86 appendAffected(receipt, operation);
87 }
88 } catch (const std::exception& exception) {
89 Result<void> rollback = rollbackApplied(plan.operations, appliedCount);
90 return authorityError<TransactionReceipt>(Status::Failed, "editor.authority.target-exception",
91 exception.what());
92 } catch (...) {
93 Result<void> rollback = rollbackApplied(plan.operations, appliedCount);
94 (void)rollback;
95 return authorityError<TransactionReceipt>(Status::Failed, "editor.authority.target-exception",
96 "Domain operation target threw an unknown exception");
97 }
98
99 receipt.state = TransactionState::Committed;
100 receipt.afterRevision = target_->revision();
101 receipt.authorityReceipt = "local:" + std::to_string(++receiptSequence_);
102 committed_.emplace(receipt.id, CommittedEntry{receipt, plan.operations});
103 commitOrder_.push_back(receipt.id);
104 return eve::editing::applied<TransactionReceipt>(std::move(receipt));
105}

107Result<TransactionReceipt> LocalWorldAuthority::compensate(const TransactionReceipt& receipt) {
108 if (!target_)
109 return authorityError<TransactionReceipt>(Status::Failed, "editor.authority.missing-target",
110 "Local authority has no target");
111 auto entry = committed_.find(receipt.id);
112 if (entry == committed_.end())
113 return authorityError<TransactionReceipt>(Status::NotFound, "editor.authority.receipt-not-found",
114 "Committed transaction is not available for compensation");
115 if (commitOrder_.empty() || commitOrder_.back() != receipt.id)
116 return authorityError<TransactionReceipt>(Status::Conflict, "editor.authority.compensation-order",
117 "Only the latest committed transaction can be compensated");
118 for (const DomainOperation& operation : entry->second.operations) {
119 if (!operation.hasInverse)
120 return authorityError<TransactionReceipt>(Status::Unsupported,
121 "editor.authority.operation-not-reversible",
122 "Transaction contains an operation without an inverse");
123 }
124
125 if (entry->second.receipt.afterRevision != target_->revision())
126 return authorityError<TransactionReceipt>(Status::Conflict, "editor.authority.revision-conflict",
127 "Target changed after the committed transaction");
128
129 auto* staging = dynamic_cast<IDomainOperationTargetStaging*>(target_);
130 if (!staging)
131 return authorityError<TransactionReceipt>(Status::Unsupported, "editor.authority.staging-unavailable",
132 "Target cannot stage a complete compensation candidate");
133
134 TransactionReceipt compensation;
135 compensation.id = TransactionId(receipt.id.value() + ".undo." + std::to_string(++receiptSequence_));
136 compensation.state = TransactionState::PendingAuthority;
137 compensation.beforeRevision = target_->revision();
138
139 const auto makeFailure = [&](Status status, std::string rule, std::string message) {
140 compensation.state = TransactionState::Failed;
141 compensation.afterRevision = target_->revision();
142 if (compensation.diagnostics.empty())
143 compensation.diagnostics.push_back(
144 ruleDiagnostic(diagnosticCodeForStatus(status), RuleId(std::move(rule)),
145 DiagnosticSeverity::Error, std::move(message)));
146 return Result<TransactionReceipt>::failure(eve::Status(status, compensation.diagnostics));
147 };
148
149 std::unique_ptr<IDomainOperationTarget> candidate;
150 try {
151 candidate = staging->cloneDomainState();
152 } catch (const std::exception& exception) {
153 return makeFailure(Status::Failed, "editor.authority.candidate-exception",
154 std::string("Could not clone compensation candidate: ") + exception.what());
155 } catch (...) {
156 return makeFailure(Status::Failed, "editor.authority.candidate-exception",
157 "Could not clone compensation candidate");
158 }
159 if (!candidate)
160 return makeFailure(Status::Unsupported, "editor.authority.staging-unavailable",
161 "Target did not provide a compensation candidate");
162 if (candidate->targetId() != target_->targetId())
163 return makeFailure(Status::Conflict, "editor.authority.candidate-mismatch",
164 "Compensation candidate belongs to another target");
165
166 for (auto operation = entry->second.operations.rbegin(); operation != entry->second.operations.rend();
167 ++operation) {
168 try {
169 Result<void> result = candidate->applyDomainOperation(inverseOf(*operation));
170 if (!result.ok()) {
171 compensation.diagnostics = result.diagnostics();
172 return makeFailure(result.code(), "editor.authority.compensation-candidate-rejected",
173 "Compensation candidate rejected an inverse operation");
174 }
175 } catch (const std::exception& exception) {
176 compensation.diagnostics.push_back(ruleDiagnostic(
177 eve::DiagnosticCode::Failed, RuleId("editor.authority.candidate-exception"),
178 DiagnosticSeverity::Error,
179 std::string("Compensation candidate threw: ") + exception.what()));
180 return makeFailure(Status::Failed, "editor.authority.candidate-exception", "Compensation candidate threw");
181 } catch (...) {
182 compensation.diagnostics.push_back(ruleDiagnostic(
183 eve::DiagnosticCode::Failed, RuleId("editor.authority.candidate-exception"),
184 DiagnosticSeverity::Error, "Compensation candidate threw an unknown exception"));
185 return makeFailure(Status::Failed, "editor.authority.candidate-exception",
186 "Compensation candidate threw an unknown exception");
187 }
188 appendAffected(compensation, *operation);
189 }
190
191 try {
192 Result<void> published = staging->commitDomainState(std::move(candidate));
193 if (!published.ok()) {
194 compensation.diagnostics = published.diagnostics();
195 return makeFailure(published.code(), "editor.authority.candidate-publish-rejected",
196 "Target rejected the compensation candidate");
197 }
198 } catch (const std::exception& exception) {
199 return makeFailure(Status::Failed, "editor.authority.candidate-publish-exception",
200 std::string("Could not publish compensation candidate: ") + exception.what());
201 } catch (...) {
202 return makeFailure(Status::Failed, "editor.authority.candidate-publish-exception",
203 "Could not publish compensation candidate");
204 }
205 compensation.state = TransactionState::Committed;
206 compensation.afterRevision = target_->revision();
207 compensation.authorityReceipt = "local:" + std::to_string(receiptSequence_);
208 committed_.erase(entry);
209 commitOrder_.pop_back();
210 if (!commitOrder_.empty()) committed_.at(commitOrder_.back()).receipt.afterRevision = compensation.afterRevision;
211 return eve::editing::applied<TransactionReceipt>(std::move(compensation));
212}

223Result<void> LocalWorldAuthority::rollbackApplied(std::span<const DomainOperation> operations,
224 std::size_t count) {
225 while (count > 0) {
226 const DomainOperation& operation = operations[--count];
227 if (!operation.hasInverse)
228 return eve::editing::failed<void>(Status::Failed, RuleId("editor.authority.rollback-impossible"),
229 "Applied operation has no inverse");
230 Result<void> result = target_->applyDomainOperation(inverseOf(operation));
231 if (!result.ok()) return result;
232 }
233 return eve::editing::applied<void>();
234}
