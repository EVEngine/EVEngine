23Result<void> GrowthCalendar::setSeasonLayout(std::vector<SeasonSlotDef> seasons) noexcept {
24 if (seasons.empty()) {
25 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
26 "season layout must be non-empty", "growth.calendar.layout"));
27 }
28 for (const SeasonSlotDef& slot : seasons) {

30 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
31 "each season needs a non-empty id and days > 0",
32 "growth.calendar.layout"));
33 }
34 }
35 seasons_ = std::move(seasons);
36 return Result<void>::success();

39Result<void> GrowthCalendar::setSeasonLayoutFromJson(std::string_view json) {
40 auto parsed = Value::fromJson(json);
41 if (!parsed) return Result<void>::failure(parsed.status());
42 Value root = std::move(parsed).takeValue();
43 const auto* array = root.getIf<Value::Array>();
44 if (!array || array->empty()) {
45 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
46 "season layout JSON must be a non-empty array",
47 "growth.calendar.layout"));
48 }
49 std::vector<SeasonSlotDef> seasons;
50 seasons.reserve(array->size());
51 for (const Value& entry : *array) {
52 const auto* object = entry.getIf<Value::Object>();
53 if (!object) {
54 return Result<void>::failure(Diagnostic::error(
55 DiagnosticCode::InvalidArgument, "season entry must be an object", "growth.calendar.layout"));
56 }
57 SeasonSlotDef slot;
58 auto idIt = object->find("id");
59 if (idIt == object->end() || !idIt->second.getIf<std::string>()) {
60 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
61 "season.id must be a string", "growth.calendar.layout"));
62 }
63 slot.id = *idIt->second.getIf<std::string>();
64 slot.days = 28;
65 auto daysIt = object->find("days");
66 if (daysIt != object->end()) {
67 if (const auto* integer = daysIt->second.getIf<std::int64_t>()) {
68 slot.days = static_cast<int>(*integer);
69 } else if (const auto* number = daysIt->second.getIf<double>()) {
70 slot.days = static_cast<int>(*number);
71 } else {
72 return Result<void>::failure(Diagnostic::error(
73 DiagnosticCode::InvalidArgument, "season.days must be an integer", "growth.calendar.layout"));
74 }
75 }
76 seasons.push_back(std::move(slot));
77 }
78 return setSeasonLayout(std::move(seasons));
79}
