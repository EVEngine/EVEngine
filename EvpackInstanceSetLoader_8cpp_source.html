14std::uint32_t little32(std::span<const std::uint8_t> bytes, std::size_t offset) {
15 return std::uint32_t(bytes[offset]) | (std::uint32_t(bytes[offset + 1]) << 8) |
16 (std::uint32_t(bytes[offset + 2]) << 16) | (std::uint32_t(bytes[offset + 3]) << 24);
17}
