48struct Reader {
49 std::string_view bytes;
50 std::size_t offset = 0;
51 template <class T>
52 bool scalar(T& value) {
53 if constexpr (std::is_enum_v<T>) {
54 std::underlying_type_t<T> raw{};
55 if (!scalar(raw)) return false;
56 value = static_cast<T>(raw);
57 return true;
58 } else if constexpr (std::is_same_v<T, bool>) {
59 std::uint8_t raw{};
60 if (!scalar(raw) || raw > 1) return false;
61 value = raw != 0;
62 return true;
63 } else if constexpr (std::is_same_v<T, float>) {
64 std::uint32_t raw{};
65 if (!scalar(raw)) return false;
66 value = std::bit_cast<float>(raw);
67 return std::isfinite(value);
68 } else if constexpr (std::is_same_v<T, double>) {
69 std::uint64_t raw{};
70 if (!scalar(raw)) return false;
71 value = std::bit_cast<double>(raw);
72 return std::isfinite(value);
73 } else {
74 if (bytes.size() - offset < sizeof(T)) return false;
75 using Unsigned = std::make_unsigned_t<T>;
76 Unsigned bits = 0;
77 for (std::size_t i = 0; i < sizeof(T); ++i)
78 bits |= static_cast<Unsigned>(static_cast<unsigned char>(bytes[offset++])) << (i * 8);
79 value = static_cast<T>(bits);
80 return true;
81 }
82 }
83 bool text(std::string& value) {
84 std::uint32_t size{};
85 if (!scalar(size) || size > bytes.size() - offset) return false;
86 value.assign(bytes.substr(offset, size));
87 offset += size;
88 return true;
89 }

103Result<std::string> hexDecode(std::string_view text) {
104 if ((text.size() & 1U) != 0 || text.size() / 2 > kMaximumSnapshotBytes)
105 return malformed<std::string>("terrain.spawnPlan.restore: invalid payload size");
106 auto nibble = [](char c) -> int {
107 if (c >= '0' && c <= '9') return c - '0';
108 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
109 return -1;
110 };
111 std::string result(text.size() / 2, '\0');
112 for (std::size_t i = 0; i < result.size(); ++i) {
113 const int high = nibble(text[i * 2]), low = nibble(text[i * 2 + 1]);
114 if (high < 0 || low < 0) return malformed<std::string>("terrain.spawnPlan.restore: invalid payload hex");
115 result[i] = static_cast<char>((high << 4) | low);
116 }
117 return Result<std::string>::success(std::move(result));
118}

279 writer.scalar(rule.mode);

286 std::uint8_t kind{};
287 std::string id;
288 bool enabled = true;
289 Heightmap fitness;
290 TerrainStampSettings operation;
291 const std::uint8_t maximumKind = version >= 3 ? 5 : (version >= 2 ? 4 : 3);
292 if (!reader.scalar(kind) || kind > maximumKind || !reader.text(id) || id.empty() ||
293 (version >= 1 && !reader.scalar(enabled)) || !readHeightmap(reader, fitness))
294 return false;
295 Heightmap localMask, globalMask;
296 if (kind == 4 && (!readHeightmap(reader, localMask) || !readHeightmap(reader, globalMask))) return false;
297 if (!readStamp(reader, operation)) return false;
298 if (kind == 3) {
299 TerrainSplatSpawnRule rule;
300 rule.ruleId = std::move(id); rule.enabled = enabled; rule.paint = std::move(fitness); rule.operation = operation;
301 if (!reader.scalar(rule.targetLayer) || !validEnums(rule)) return false;
302 output = std::move(rule);
303 } else if (kind == 0) {
304 TerrainDetailSpawnRule rule;
305 rule.ruleId = std::move(id); rule.enabled = enabled; rule.fitness = std::move(fitness); rule.operation = operation;
306 if (!reader.scalar(rule.mode) || !readDetail(reader, rule.settings) || !reader.scalar(rule.seed) || !validEnums(rule)) return false;
307 output = std::move(rule);
308 } else if (kind == 1) {
309 TerrainTreeSpawnRule rule;
310 rule.ruleId = std::move(id); rule.enabled = enabled; rule.fitness = std::move(fitness); rule.operation = operation;
311 if (!reader.scalar(rule.mode) || !readTree(reader, rule.settings) || !validEnums(rule)) return false;
312 output = std::move(rule);
313 } else if (kind == 2) {
314 TerrainObjectSpawnRule rule;
315 rule.ruleId = std::move(id); rule.enabled = enabled; rule.fitness = std::move(fitness); rule.operation = operation;
316 if (!reader.scalar(rule.mode) || !readObject(reader, rule.settings) || !validEnums(rule)) return false;
317 output = std::move(rule);
318 } else if (kind == 4) {
319 TerrainModifierStampSpawnRule rule;
320 rule.ruleId = std::move(id); rule.enabled = enabled; rule.stamp = std::move(fitness);
321 rule.localMask = std::move(localMask); rule.globalMask = std::move(globalMask); rule.operation = operation;
322 if (!validEnums(rule)) return false;
323 output = std::move(rule);
324 } else {
325 TerrainProbeSpawnRule rule;
326 rule.ruleId = std::move(id); rule.enabled = enabled; rule.fitness = std::move(fitness);
327 rule.operation = operation;
328 if (!reader.scalar(rule.mode) || !readProbe(reader, rule.settings) || !validEnums(rule)) return false;
329 output = std::move(rule);
330 }
331 return true;
332}

358Result<void> TerrainSpawnPlan::restoreJson(const std::string& json) {
359 if (json.size() > kMaximumSnapshotBytes * 2 + 1024)
360 return malformed<void>("terrain.spawnPlan.restore: JSON exceeds size limit");
361 auto decoded = Value::fromJson(json);
362 if (!decoded.ok()) return Result<void>::failure(decoded.status());
363 const auto* root = decoded.value().getIf<Value::Object>();
364 if (!root || root->size() != 3 || !root->contains("schema") || !root->contains("version") ||
365 !root->contains("payload"))
366 return malformed<void>("terrain.spawnPlan.restore: missing or unknown root fields");
367 const auto* schema = root->at("schema").getIf<std::string>();
368 const auto* version = root->at("version").getIf<std::int64_t>();
369 const auto* payload = root->at("payload").getIf<std::string>();
370 if (!schema || *schema != "eve.procgen.terrain-spawn-plan" || !version ||
371 (*version != 0 && *version != 1 && *version != 2 && *version != 3) ||
372 !payload)
373 return malformed<void>("terrain.spawnPlan.restore: unsupported schema or version");
374 auto bytes = hexDecode(*payload);
375 if (!bytes.ok()) return Result<void>::failure(bytes.status());
376 Reader reader{bytes.value()};
377 std::uint32_t count{};
378 if (!reader.scalar(count) || count > kMaximumRules)
379 return malformed<void>("terrain.spawnPlan.restore: invalid rule count");
380 TerrainSpawnPlan candidate;
381 candidate.rules_.reserve(count);
382 for (std::uint32_t i = 0; i < count; ++i) {
383 TerrainSpawnRule rule;
384 if (!readRule(reader, static_cast<int>(*version), rule))
385 return malformed<void>("terrain.spawnPlan.restore: invalid rule payload");
386 const auto id = std::visit([](const auto& value) { return value.ruleId; }, rule);
387 if (candidate.contains(id)) return malformed<void>("terrain.spawnPlan.restore: duplicate rule ID");
388 candidate.rules_.push_back(std::move(rule));
389 }
390 if (reader.offset != reader.bytes.size())
391 return malformed<void>("terrain.spawnPlan.restore: trailing payload data");
392 rules_.swap(candidate.rules_);
393 return Result<void>::success();
394}
