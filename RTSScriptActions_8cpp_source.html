227Result<RTSBuildReceipt> RTS::queueScriptUnit(Building& producer, SubjectRef unitSubject, LogicalId unitDefinition,
228 int priority, std::span<transaction::ITransactionParticipant*> effects,
229 std::optional<resource::CostSpec> additionalCost) {
230 auto prepared = prepareScriptUnit(producer, unitSubject, unitDefinition, priority);
231 if (!prepared) return Result<RTSBuildReceipt>::failure(prepared.status());
232 auto stage = std::move(prepared).takeValue();
233 auto* faction = dynamic_cast<Faction*>(producer.faction()->link.resolve());
234 auto account = scriptAccount(*faction);
235 if (!account) return Result<RTSBuildReceipt>::failure(account.status());
236 auto cost = stage.cost;
237 if (additionalCost) {
238 auto items = cost.items();
239 items.insert(items.end(), additionalCost->items().begin(), additionalCost->items().end());
240 auto merged = resource::CostSpec::create(std::move(items));
241 if (!merged) return Result<RTSBuildReceipt>::failure(merged.status());
242 cost = std::move(merged).takeValue();
243 }
244 std::vector<transaction::ITransactionParticipant*> parts;
245 for (auto& effect : stage.stage.effects) parts.push_back(effect.get());
246 parts.insert(parts.end(), effects.begin(), effects.end());

248 transaction::TransactionContext("rts.script.production." + unitSubject.format()), account.value().get(), cost,
249 parts);
250 if (!receipt) return Result<RTSBuildReceipt>::failure(receipt.status());
