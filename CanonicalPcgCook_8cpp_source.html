40Result<CookedCanonicalPcgGraph> cookCanonicalPcgGraph(
41 std::span<const std::uint8_t> definition, std::uint32_t maximumRules,
42 std::uint64_t maximumPlanBytes) {
43 auto parsed = Value::fromJson(std::string_view(reinterpret_cast<const char*>(definition.data()),
44 definition.size()));
45 if (!parsed) return Result<CookedCanonicalPcgGraph>::failure(parsed.status());
46 const auto* root = parsed.value().getIf<Value::Object>();
47 const Value* schema = root ? field(*root, "schema") : nullptr;
48 const Value* version = root ? field(*root, "schemaVersion") : nullptr;
49 const Value* algorithm = root ? field(*root, "algorithm") : nullptr;
50 const Value* rulesValue = root ? field(*root, "rules") : nullptr;
51 const auto* rules = rulesValue ? rulesValue->getIf<Value::Array>() : nullptr;
52 if (!schema || !schema->isString() || schema->asString() != "eve.pcg-graph" ||
53 !version || !version->isInt64() || version->asInt() != 1 || !algorithm ||
54 !algorithm->isString() || algorithm->asString() != "terrain-layer-scatter-v1" || !rules)
55 return failure<CookedCanonicalPcgGraph>(DiagnosticCode::Unsupported,
56 "unsupported eve.pcg-graph/1 definition");
57 if (rules->empty() || rules->size() > maximumRules)
58 return failure<CookedCanonicalPcgGraph>(DiagnosticCode::InvalidArgument,
59 "PCG rule count exceeds Cook limits", "$.rules");
60
61 std::ostringstream plan;
62 plan << "EVPCG_POINT_GRAPH 1\n" << std::setprecision(9);
63 std::vector<std::string> outputs;
64 Value::Array slots;
65 constexpr double radiansToDegrees = 57.29577951308232;
66 for (std::size_t i = 0; i < rules->size(); ++i) {
67 const auto* rule = (*rules)[i].getIf<Value::Object>();
68 const Value* id = rule ? field(*rule, "id") : nullptr;
69 const Value* prototype = rule ? field(*rule, "prototype") : nullptr;
70 const Value* layer = rule ? field(*rule, "layer") : nullptr;
71 const Value* seed = rule ? field(*rule, "seed") : nullptr;
72 double density = 0, minimumSlope = 0, maximumSlope = 0;
73 if (!id || !id->isString() || id->asString().empty() || !prototype ||
74 !prototype->isString() || prototype->asString().empty() || !layer || !layer->isString() ||
75 !seed || !seed->isInt64() || seed->asInt() < 0 ||
76 !number(rule ? field(*rule, "densityPerSquareMeter") : nullptr, density) || density < 0 ||
77 !number(rule ? field(*rule, "minimumSlopeRadians") : nullptr, minimumSlope) ||
78 !number(rule ? field(*rule, "maximumSlopeRadians") : nullptr, maximumSlope) ||
79 minimumSlope < 0 || maximumSlope < minimumSlope || maximumSlope > 3.141592653589793)
80 return failure<CookedCanonicalPcgGraph>(DiagnosticCode::InvalidArgument,
81 "PCG scatter rule is invalid",
82 "$.rules[" + std::to_string(i) + "]");
83 const std::string suffix = std::to_string(i);
84 const std::string sample = "sample." + suffix;
85 const std::string project = "project." + suffix;
86 const std::string slope = "slope." + suffix;
87 const std::string taggedPrototype = "prototype." + suffix;
88 const std::string taggedLayer = "layer." + suffix;
89 node(plan, sample, "spatial.sample");
90 plan << "FLOAT " << std::quoted(sample) << " \"spacing\" "
91 << (density > 0 ? std::sqrt(1.0 / density) : 1.0) << '\n';
92 plan << "INT " << std::quoted(sample) << " \"seed\" "
93 << (seed->asInt() & 0x7fffffff) << '\n';
94 node(plan, project, "spatial.project"); edge(plan, sample, project);
95 node(plan, slope, "filter.slope"); edge(plan, project, slope);
96 plan << "FLOAT " << std::quoted(slope) << " \"minDegrees\" "
97 << minimumSlope * radiansToDegrees << '\n';
98 plan << "FLOAT " << std::quoted(slope) << " \"maxDegrees\" "
99 << maximumSlope * radiansToDegrees << '\n';
100 std::string scatterOutput = slope;
101 if (density == 0) {
102 const std::string culled = "cull." + suffix;
103 node(plan, culled, "density.cull"); edge(plan, slope, culled);
104 plan << "FLOAT " << std::quoted(culled) << " \"multiplier\" 0\n";
105 scatterOutput = culled;
106 }
107 node(plan, taggedPrototype, "attribute.set.string"); edge(plan, scatterOutput, taggedPrototype);
108 plan << "STRING " << std::quoted(taggedPrototype) << " \"attribute\" \"prototype\"\n"
109 << "STRING " << std::quoted(taggedPrototype) << " \"value\" "
110 << std::quoted(prototype->asString()) << '\n';
111 node(plan, taggedLayer, "attribute.set.string"); edge(plan, taggedPrototype, taggedLayer);
112 plan << "STRING " << std::quoted(taggedLayer) << " \"attribute\" \"layer\"\n"
113 << "STRING " << std::quoted(taggedLayer) << " \"value\" "
114 << std::quoted(layer->asString()) << '\n';
115 outputs.push_back(taggedLayer);
116 slots.emplace_back(Value::Object{{"node", Value(sample)}, {"role", Value("terrain")}});
117 slots.emplace_back(Value::Object{{"node", Value(project)}, {"role", Value("terrain")}});
118 }
119 std::string output = outputs.front();
120 for (std::size_t i = 1; i < outputs.size(); ++i) {
121 const std::string merged = "merge." + std::to_string(i);
122 node(plan, merged, "merge"); edge(plan, output, merged);
123 plan << "EDGE " << std::quoted(outputs[i]) << ' ' << std::quoted(merged) << " 1\n";
124 output = merged;
125 }
126 plan << "END\n";
127 const std::string encodedPlan = plan.str();
128 if (encodedPlan.size() > maximumPlanBytes)
129 return failure<CookedCanonicalPcgGraph>(DiagnosticCode::InvalidArgument,
130 "compiled PCG plan exceeds Cook budget");
131 Value::Object runtime{{"schema", Value("eve.pcg-graph")},
132 {"schemaVersion", Value(std::int64_t(1))},
133 {"executionPlan", Value("chunk:1")},
134 {"executionPlanFormat", Value("EVPCG_POINT_GRAPH/1")},
135 {"outputNode", Value(output)}, {"spatialSlots", Value(std::move(slots))}};
136 auto encodedDefinition = Value(std::move(runtime)).toJson();
137 if (!encodedDefinition)
138 return Result<CookedCanonicalPcgGraph>::failure(encodedDefinition.status());
139 return Result<CookedCanonicalPcgGraph>::success(
140 {{encodedDefinition.value().begin(), encodedDefinition.value().end()},
141 {encodedPlan.begin(), encodedPlan.end()}});
142}
