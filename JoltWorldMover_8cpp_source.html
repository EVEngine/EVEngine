17eve::Result<CapsuleMove3D> WorldBackend::moveCapsuleOwned(float ax, float ay, float az, float bx, float by, float bz,
18 float radius, float dx, float dy, float dz,
19 QueryFilter3D filter, CapsuleMovePolicy3D policy) {
20 if (!isValid()) return rejected<CapsuleMove3D>("World expired", eve::DiagnosticCode::StaleHandle);
21 auto cooked = capsuleVolume(ax, ay, az, bx, by, bz, radius);
22 if (!cooked) return eve::Result<CapsuleMove3D>::failure(cooked.status());
23 JPH::Vec3 up(policy.upX, policy.upY, policy.upZ), desired(dx, dy, dz);
24 if (!isFinite(dx, dy, dz) || !std::isfinite(desired.LengthSq()) || !isFinite(policy.upX, policy.upY, policy.upZ) ||
25 !std::isfinite(up.LengthSq()) || up.LengthSq() < 1.e-12f || !std::isfinite(policy.maxSlopeRadians) ||
26 policy.maxSlopeRadians < 0 || policy.maxSlopeRadians >= JPH::JPH_PI * .5f)
27 return rejected<CapsuleMove3D>("Invalid mover policy or displacement", eve::DiagnosticCode::InvalidArgument);
28 up = up.Normalized();
29 auto volume = std::move(cooked).takeValue();
30 const auto start = volume.transform.GetTranslation();
31 const auto target = start + desired;
32 CapsuleMove3D result;
33 result.world = owner->runtimeHandle();
34 JPH::Vec3 bestNormal = JPH::Vec3::sZero();
35 float bestDot = 0.f, groundDot = -1.f;
36 auto record = [&](JPH::Vec3Arg normal) {
37 ++result.planeCount;
38 result.constrained = true;
39 groundDot = std::max(groundDot, normal.Dot(up));
40 if (normal.Dot(desired) < bestDot) {
41 bestDot = normal.Dot(desired);
42 bestNormal = normal;
43 }
44 };
45 // Query-only controller: no body is created, no impulses or callbacks are dispatched.
46 // Each pass resolves initial penetration, clips against nearby planes, then sweeps continuously.
47 for (int pass = 0; pass < 8; ++pass) {
48 result.iterations = pass + 1;
49 auto overlaps = overlapVolume(*state, volume, .002f);
50 JPH::Vec3 correction = JPH::Vec3::sZero();
51 auto remaining = JPH::Vec3(target - volume.transform.GetTranslation());
52 for (const auto& hit : overlaps.mHits) {
53 if (!isAcceptedHit(*state, hit.mBodyID2, hit.mSubShapeID2, filter)) continue;
54 const auto normal = -hit.mPenetrationAxis.NormalizedOr(JPH::Vec3::sAxisY());
55 record(normal);
56 const float needed = hit.mPenetrationDepth + .0001f - correction.Dot(normal);
57 if (needed > 0) correction += needed * normal;
58 const float inward = remaining.Dot(normal);
59 if (inward < 0) remaining -= inward * normal;
60 }
61 volume.transform.SetTranslation(volume.transform.GetTranslation() + correction);
62 if (remaining.LengthSq() < 1.e-10f) break;
63 auto hits = sweepVolume(*state, volume, remaining);
64 float fraction = 1.f;
65 JPH::Vec3 normal = JPH::Vec3::sZero();
66 for (const auto& hit : hits.mHits) {
67 if (!isAcceptedHit(*state, hit.mBodyID2, hit.mSubShapeID2, filter)) continue;
68 auto n = -hit.mPenetrationAxis.NormalizedOr(JPH::Vec3::sAxisY());
69 if (n.Dot(remaining) >= -1.e-7f) continue;
70 fraction = std::max(0.f, hit.mFraction - .0001f / remaining.Length());
71 normal = n;
72 record(n);
73 break;
74 }
75 volume.transform.SetTranslation(volume.transform.GetTranslation() + remaining * fraction);
76 if (fraction >= 1.f) break;
77 // The next overlap includes the swept surface and clips the remaining displacement along it.
78 if (fraction == 0 && correction.LengthSq() < 1.e-12f &&
79 std::abs(normal.Dot(remaining)) >= remaining.Length() - .0001f)
80 break;
81 }
82 // Classify ground at the final position too, including a purely tangential movement.
83 for (const auto& hit : overlapVolume(*state, volume, .002f).mHits)
84 if (isAcceptedHit(*state, hit.mBodyID2, hit.mSubShapeID2, filter))
85 groundDot = std::max(groundDot, -hit.mPenetrationAxis.NormalizedOr(JPH::Vec3::sAxisY()).Dot(up));
86 auto delta = JPH::Vec3(volume.transform.GetTranslation() - start);
87 result.deltaX = delta.GetX();
88 result.deltaY = delta.GetY();
89 result.deltaZ = delta.GetZ();
90 result.normalX = bestNormal.GetX();
91 result.normalY = bestNormal.GetY();
92 result.normalZ = bestNormal.GetZ();
93 result.grounded = groundDot >= std::cos(policy.maxSlopeRadians);
94 owner->moverGroundDot_ = groundDot;
95 return eve::Result<CapsuleMove3D>::success(result);
96}
97eve::Result<bool> WorldBackend::moveCapsule(float ax, float ay, float az, float bx, float by, float bz, float radius,

112 owner->moverIterations_ = r.iterations;

115}
