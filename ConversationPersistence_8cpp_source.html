18Poco::Dynamic::Var stateToVar(const StateValue& state) {
19 switch (state.kind()) {
20 case StateValue::Kind::Null: return {};
21 case StateValue::Kind::Int: return Poco::Dynamic::Var(static_cast<Poco::Int64>(state.asInt()));
22 case StateValue::Kind::Float: return Poco::Dynamic::Var(state.asDouble());
23 case StateValue::Kind::Bool: return Poco::Dynamic::Var(state.asBool());
24 case StateValue::Kind::String: return Poco::Dynamic::Var(state.asString());
25 case StateValue::Kind::Array: {
26 Poco::JSON::Array::Ptr array(new Poco::JSON::Array());
27 for (size_t i = 0; i < state.arraySize(); ++i) array->add(stateToVar(state.at(i)));
28 return Poco::Dynamic::Var(array);
29 }
30 case StateValue::Kind::Object: {
31 Poco::JSON::Object::Ptr object(new Poco::JSON::Object());
32 for (const auto& key : state.keys()) object->set(key, stateToVar(*state.find(key)));
33 return Poco::Dynamic::Var(object);
34 }
35 }
36 return {};
37}

39StateValue varToState(const Poco::Dynamic::Var& value) {
40 if (value.isEmpty()) return StateValue::null();
41 if (value.isBoolean()) return StateValue::boolean(value.convert<bool>());
42 if (value.isInteger()) return StateValue::integer(value.convert<Poco::Int64>());
43 if (value.isNumeric()) return StateValue::number(value.convert<double>());
44 if (value.isString()) return StateValue::string(value.convert<std::string>());
45 if (value.type() == typeid(Poco::JSON::Array::Ptr)) {
46 StateValue result = StateValue::array();
47 Poco::JSON::Array::Ptr array = value.extract<Poco::JSON::Array::Ptr>();
48 if (array)
49 for (size_t i = 0; i < array->size(); ++i)
50 result.pushBack(varToState(array->get(static_cast<unsigned int>(i))));
51 return result;
52 }
53 if (value.type() == typeid(Poco::JSON::Object::Ptr)) {
54 StateValue result = StateValue::object();
55 Poco::JSON::Object::Ptr object = value.extract<Poco::JSON::Object::Ptr>();
56 if (object)
57 for (const auto& key : object->getNames()) result.set(key, varToState(object->get(key)));
58 return result;
59 }
60 return StateValue::null();
61}

63std::string trim(std::string text) {
64 const auto first = std::find_if_not(text.begin(), text.end(), [](unsigned char c) { return std::isspace(c); });
65 const auto last = std::find_if_not(text.rbegin(), text.rend(), [](unsigned char c) { return std::isspace(c); });
66 return first >= last.base() ? std::string{} : std::string(first, last.base());

101bool ConversationSaveMigrations::registerMigration(const std::string& assetId, int fromVersion,
102 const std::string& currentAssetId, const std::string& nodeMap,
103 std::string* error) {
104 if (assetId.empty() || currentAssetId.empty() || fromVersion < 0)
105 return fail(error, "conversation: migration requires asset IDs and a non-negative version");
106 Rule rule{assetId, fromVersion, currentAssetId, {}};
107 std::istringstream mappings(nodeMap);
108 for (std::string item; std::getline(mappings, item, ',');) {
109 item = trim(std::move(item));
110 const size_t pos = item.find(':');
111 if (item.empty()) continue;
112 if (pos == std::string::npos || trim(item.substr(0, pos)).empty() || trim(item.substr(pos + 1)).empty())
113 return fail(error, "conversation: node migration must use old:new pairs");
114 rule.nodes[trim(item.substr(0, pos))] = trim(item.substr(pos + 1));
115 }
116 auto existing = std::find_if(rules_.begin(), rules_.end(), [&](const Rule& candidate) {
117 return candidate.assetId == assetId && candidate.fromVersion == fromVersion;
118 });
119 if (existing == rules_.end())
120 rules_.push_back(std::move(rule));
121 else
122 *existing = std::move(rule);
123 return true;
124}

126bool ConversationSaveMigrations::migrate(StateValue& state, const Resolver& resolve, std::string* error) const {
127 StateValue candidate = state;
128 auto migrateFrame = [&](StateValue& frame) {
129 StateValue* asset = frame.find("asset");
130 StateValue* version = frame.find("version");
131 StateValue* node = frame.find("node");
132 if (!asset || !asset->isString() || !version || !version->isInt() || !node || !node->isString())
133 return fail(error, "conversation: saved frame is malformed");
134 const ConversationAsset* current = resolve(asset->asString());
135 if (current && current->version == version->asInt()) return true;
136 const auto rule = std::find_if(rules_.begin(), rules_.end(), [&](const Rule& item) {
137 return item.assetId == asset->asString() && item.fromVersion == version->asInt();
138 });
139 if (rule == rules_.end())
140 return fail(error, "conversation: no migration for '" + asset->asString() + "' version " +
141 std::to_string(version->asInt()));
142 current = resolve(rule->currentAssetId);
143 if (!current) return fail(error, "conversation: migration target '" + rule->currentAssetId + "' is missing");
144 std::string currentNode = node->asString();
145 if (const auto renamed = rule->nodes.find(currentNode); renamed != rule->nodes.end())
146 currentNode = renamed->second;
147 if (!current->findNode(currentNode))
148 return fail(error, "conversation: migrated node '" + currentNode + "' is missing");
149 frame.set("asset", StateValue::string(current->id));
150 frame.set("version", StateValue::integer(current->version));
151 frame.set("node", StateValue::string(std::move(currentNode)));
152 return true;
153 };
154
155 StateValue* active = candidate.find("active");
156 if (!active || !active->isBool()) return fail(error, "conversation: state is missing active");
157 if (active->asBool()) {
158 StateValue* current = candidate.find("current");
159 StateValue* stack = candidate.find("stack");
160 if (!current || !current->isObject() || !stack || !stack->isArray())
161 return fail(error, "conversation: state frames are malformed");
162 if (!migrateFrame(*current)) return false;
163 for (size_t i = 0; i < stack->arraySize(); ++i)
164 if (!migrateFrame(stack->at(i))) return false;
165 }
166 state = std::move(candidate);
167 return true;
168}
