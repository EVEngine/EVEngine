18Poco::Dynamic::Var stateToVar(const StateValue& state) {
19 switch (state.kind()) {
20 case StateValue::Kind::Null: return {};
21 case StateValue::Kind::Int: return Poco::Dynamic::Var(static_cast<Poco::Int64>(state.asInt()));
22 case StateValue::Kind::Float: return Poco::Dynamic::Var(state.asDouble());
23 case StateValue::Kind::Bool: return Poco::Dynamic::Var(state.asBool());
24 case StateValue::Kind::String: return Poco::Dynamic::Var(state.asString());
25 case StateValue::Kind::Array: {
26 Poco::JSON::Array::Ptr array(new Poco::JSON::Array());
27 for (size_t i = 0; i < state.arraySize(); ++i) array->add(stateToVar(state.at(i)));
28 return Poco::Dynamic::Var(array);
29 }
30 case StateValue::Kind::Object: {
31 Poco::JSON::Object::Ptr object(new Poco::JSON::Object());
32 for (const auto& key : state.keys()) object->set(key, stateToVar(*state.find(key)));
33 return Poco::Dynamic::Var(object);
34 }
35 }
36 return {};
37}

39StateValue varToState(const Poco::Dynamic::Var& value) {
40 if (value.isEmpty()) return StateValue::null();
41 if (value.isBoolean()) return StateValue::boolean(value.convert<bool>());
42 if (value.isInteger()) return StateValue::integer(value.convert<Poco::Int64>());
43 if (value.isNumeric()) return StateValue::number(value.convert<double>());
44 if (value.isString()) return StateValue::string(value.convert<std::string>());
45 if (value.type() == typeid(Poco::JSON::Array::Ptr)) {
46 StateValue result = StateValue::array();
47 Poco::JSON::Array::Ptr array = value.extract<Poco::JSON::Array::Ptr>();
48 if (array)
49 for (size_t i = 0; i < array->size(); ++i)
50 result.pushBack(varToState(array->get(static_cast<unsigned int>(i))));
51 return result;
52 }
53 if (value.type() == typeid(Poco::JSON::Object::Ptr)) {
54 StateValue result = StateValue::object();
55 Poco::JSON::Object::Ptr object = value.extract<Poco::JSON::Object::Ptr>();
56 if (object)
57 for (const auto& key : object->getNames()) result.set(key, varToState(object->get(key)));
58 return result;
59 }
60 return StateValue::null();
61}

63std::string trim(std::string text) {
64 const auto first = std::find_if_not(text.begin(), text.end(), [](unsigned char c) { return std::isspace(c); });
65 const auto last = std::find_if_not(text.rbegin(), text.rend(), [](unsigned char c) { return std::isspace(c); });
66 return first >= last.base() ? std::string{} : std::string(first, last.base());
67}

103eve::Result<void> ConversationSaveMigrations::registerMigration(const std::string& assetId, int fromVersion,
104 const std::string& currentAssetId,
105 const std::string& nodeMap) {
106 if (assetId.empty() || currentAssetId.empty() || fromVersion < 0)
107 return persistenceFailure(eve::DiagnosticCode::InvalidArgument,
108 "conversation: migration requires asset IDs and a non-negative version",
109 "migration");
110 Rule rule{assetId, fromVersion, currentAssetId, {}};
111 std::istringstream mappings(nodeMap);
112 for (std::string item; std::getline(mappings, item, ',');) {
113 item = trim(std::move(item));
114 const size_t pos = item.find(':');
115 if (item.empty()) continue;
116 if (pos == std::string::npos || trim(item.substr(0, pos)).empty() || trim(item.substr(pos + 1)).empty())
117 return persistenceFailure(eve::DiagnosticCode::InvalidArgument,
118 "conversation: node migration must use old:new pairs", "migration.nodeMap");
119 rule.nodes[trim(item.substr(0, pos))] = trim(item.substr(pos + 1));
120 }
121 auto existing = std::find_if(rules_.begin(), rules_.end(), [&](const Rule& candidate) {
122 return candidate.assetId == assetId && candidate.fromVersion == fromVersion;
123 });
124 if (existing == rules_.end())
125 rules_.push_back(std::move(rule));
126 else
127 *existing = std::move(rule);
128 return eve::Result<void>::success();
129}

131eve::Result<StateValue> ConversationSaveMigrations::migrate(const StateValue& state, const Resolver& resolve) const {
132 StateValue candidate = state;
133 std::string error;
134 const auto failFrame = [&](std::string message) {
135 error = std::move(message);
136 return false;
137 };
138 auto migrateFrame = [&](StateValue& frame) {
139 StateValue* asset = frame.find("asset");
140 StateValue* version = frame.find("version");
141 StateValue* node = frame.find("node");
142 if (!asset || !asset->isString() || !version || !version->isInt() || !node || !node->isString())
143 return failFrame("conversation: saved frame is malformed");
144 const ConversationAsset* current = resolve(asset->asString());
145 if (current && current->version == version->asInt()) return true;
146 const auto rule = std::find_if(rules_.begin(), rules_.end(), [&](const Rule& item) {
147 return item.assetId == asset->asString() && item.fromVersion == version->asInt();
148 });
149 if (rule == rules_.end())
150 return failFrame("conversation: no migration for '" + asset->asString() + "' version " +
151 std::to_string(version->asInt()));
152 current = resolve(rule->currentAssetId);
153 if (!current) {
154 error = "conversation: migration target '" + rule->currentAssetId + "' is missing";
155 return false;
156 }
157 std::string currentNode = node->asString();
158 if (const auto renamed = rule->nodes.find(currentNode); renamed != rule->nodes.end())
159 currentNode = renamed->second;
160 if (!current->findNode(currentNode))
161 return failFrame("conversation: migrated node '" + currentNode + "' is missing");
162 frame.set("asset", StateValue::string(current->id));
163 frame.set("version", StateValue::integer(current->version));
164 frame.set("node", StateValue::string(std::move(currentNode)));
165 return true;
166 };
167
168 StateValue* active = candidate.find("active");
169 if (!active || !active->isBool())
170 return persistenceFailure<StateValue>(eve::DiagnosticCode::SerializationError,
171 "conversation: state is missing active", "save.active");
172 if (active->asBool()) {
173 StateValue* current = candidate.find("current");
174 StateValue* stack = candidate.find("stack");
175 if (!current || !current->isObject() || !stack || !stack->isArray())
176 return persistenceFailure<StateValue>(eve::DiagnosticCode::SerializationError,
177 "conversation: state frames are malformed", "save.frames");
178 if (!migrateFrame(*current))
179 return persistenceFailure<StateValue>(eve::DiagnosticCode::UnknownVersion, error, "save.current");
180 for (size_t i = 0; i < stack->arraySize(); ++i)
181 if (!migrateFrame(stack->at(i)))
182 return persistenceFailure<StateValue>(eve::DiagnosticCode::UnknownVersion, error, "save.stack");
183 }
184 return eve::Result<StateValue>::success(std::move(candidate));
185}
