128Poco::JSON::Array::Ptr expandRecipeActions(const RecipeDef& recipe, Poco::JSON::Object::Ptr params) {
129 Poco::JSON::Array::Ptr actions = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
130 if (std::string_view(recipe.id) == "empty.clear") {
131 actions->add(actionObject("reset", "", {}));
132 return actions;
133 }
134 if (std::string_view(recipe.id) == "ruins.moonlit") {
135 const int seed = paramInt(params, "seed", 7);
136 int trees = paramInt(params, "trees", 3);
137 if (trees < 0) trees = 0;
138 if (trees > 12) trees = 12;
139 actions->add(actionObject("reset", "", {}));
140 actions->add(actionObject("spawn", "", spawnParams("ground", "ground", 0.0, 0.0, 0.0, 12.0, 1.0, 12.0, seed)));
141 actions->add(
142 actionObject("spawn", "", spawnParams("ruin_core", "box", 0.0, 0.6, 0.0, 2.2, 1.4, 2.2, seed + 1)));
143 {
144 Poco::JSON::Object::Ptr light = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
145 light->set("ambient", 0.18);
146 light->set("moonlit", true);
147 actions->add(actionObject("lighting", "", light));
148 }
149 uint32_t rng = static_cast<uint32_t>(seed) * 2654435761u + 1u;
150 for (int i = 0; i < trees; ++i) {
151 const double angle = lcgUnit(rng) * 6.283185307179586;
152 const double radius = 3.5 + lcgUnit(rng) * 3.0;
153 const double x = std::cos(angle) * radius;
154 const double z = std::sin(angle) * radius;
155 actions->add(actionObject(
156 "spawn", "",
157 spawnParams("tree_" + std::to_string(i), "tree", x, 0.0, z, 1.0, 1.0, 1.0, seed + 10 + i)));
158 }
159 return actions;
160 }
161 if (std::string_view(recipe.id) == "plaza.sunlit") {
162 const int seed = paramInt(params, "seed", 11);
163 int posts = paramInt(params, "posts", 4);
164 if (posts < 0) posts = 0;
165 if (posts > 8) posts = 8;
166 actions->add(actionObject("reset", "", {}));
167 actions->add(
168 actionObject("spawn", "", spawnParams("plaza_ground", "ground", 0.0, 0.0, 0.0, 10.0, 1.0, 10.0, seed)));
169 actions->add(
170 actionObject("spawn", "", spawnParams("plaza_marker", "box", 0.0, 0.4, 0.0, 1.0, 0.8, 1.0, seed + 2)));
171 for (int i = 0; i < posts; ++i) {
172 const double angle = (6.283185307179586 * static_cast<double>(i)) / static_cast<double>(posts);
173 const double x = std::cos(angle) * 4.0;
174 const double z = std::sin(angle) * 4.0;
175 actions->add(actionObject(
176 "spawn", "",
177 spawnParams("post_" + std::to_string(i), "box", x, 0.7, z, 0.25, 1.4, 0.25, seed + 20 + i)));
178 }
179 return actions;
180 }
181 return actions;
182}

212std::string recipeApply(Poco::JSON::Object::Ptr args) {
213 if (!args || !args->has("id")) {
214 return errorPayload("id is required", "Pass {id:\"ruins.moonlit\", params:{seed,trees}} from eve_recipe_list");
215 }
216 const std::string id = getArgString(args, "id");
217 const RecipeDef* recipe = findRecipe(id);
218 if (!recipe) {
219 return errorPayload("unknown recipe id '" + id + "'",
220 "Call eve_recipe_list; ids are registry strings, not free-form geometry");
221 }
222
223 Debugger& dbg = Debugger::instance();
224 HSQUIRRELVM vm = dbg.vm();
225 if (!vm) return errorPayload("no VM", "Attach a live eve run --mcp-port session first");
226
227 std::string err;
228 if (!ensureSceneDirectorInstalled(vm, &err, /*allowHeavyWork=*/false)) {
229 if (err == "deferred-to-update") {
230 return errorPayload("deferred-to-update",
231 "scene_director install was queued for the next eve.dev.poll; retry eve_recipe_apply");
232 }
233 return errorPayload(err.empty() ? "scene_director kit unavailable" : err);
234 }
235
236 Poco::JSON::Object::Ptr params;
237 if (args->has("params")) {
238 try {
239 params = args->getObject("params");
240 } catch (...) {
241 return errorPayload("params must be a JSON object");
242 }
243 }
244
245 Poco::JSON::Array::Ptr actions = expandRecipeActions(*recipe, params);
246 Poco::JSON::Object::Ptr options = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
247 options->set("stopOnError", getArgBool(args, "stopOnError", true));
248 auto actionsLiteral = sceneLiteral(Poco::Dynamic::Var(actions));
249 if (!actionsLiteral.ok()) return errorPayload(actionsLiteral.status().describe());
250 auto optionsLiteral = sceneLiteral(Poco::Dynamic::Var(options));
251 if (!optionsLiteral.ok()) return errorPayload(optionsLiteral.status().describe());
252
253 err.clear();
254 const std::string snippet =
255 "return ::scene_director.batch(" + actionsLiteral.value() + "," + optionsLiteral.value() + ");";
256 const std::string sceneOut = callSceneDirectorReturn(vm, snippet, &err);
257 if (!err.empty()) return errorPayload(err);
258
259 Poco::JSON::Object::Ptr report = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
260 report->set("schemaId", "eve.recipe.apply-report");
261 report->set("schemaVersion", 1);
262 report->set("id", id);
263 auto sceneObj = parseJsonObject(sceneOut);
264 if (sceneObj) {
265 report->set("scene", sceneObj);
266 const bool ok = sceneObj->optValue<bool>("ok", false);
267 report->set("ok", ok);
268 if (sceneObj->has("revision")) report->set("revision", sceneObj->get("revision"));
269 if (!ok) {
270 report->set("error", "recipe apply failed");
271 report->set("hint", "Inspect scene.results; recipe apply is transactional when stopOnError=true");
272 return textPayload(report, true);
273 }
274 } else {
275 report->set("ok", false);
276 report->set("error", "scene batch returned non-object JSON");
277 report->set("raw", sceneOut);
278 return textPayload(report, true);
279 }
280
281 if (args->has("expectedRevision")) {
282 const int expected = getArgInt(args, "expectedRevision", -1);
283 const int actual = sceneObj->optValue<int>("revision", -1);
284 if (expected >= 0 && actual >= 0 && expected != actual) {
285 report->set("ok", false);
286 report->set("error", "revision mismatch");
287 report->set("expectedRevision", expected);
288 report->set("revision", actual);
289 return textPayload(report, true);
290 }
291 }
292
293 err.clear();
294 const std::string infoOut = callSceneDirectorReturn(vm, "return ::scene_director.info();", &err);
295 if (err.empty()) {
296 if (auto info = parseJsonObject(infoOut)) report->set("info", info);
297 }
298 report->set("hint", "Recipe applied; escalate to verify.play for reachable/dialogue criteria");
299 return textPayload(report);
300}
