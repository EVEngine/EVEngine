28bool compareValues(const Value& left, CompareOperator op, const Value& right) {
29 double leftNumber = 0.0;
30 double rightNumber = 0.0;
31 bool numeric = number(left, leftNumber) && number(right, rightNumber);
32 if (numeric) {
33 switch (op) {
34 case CompareOperator::Equal: return leftNumber == rightNumber;
35 case CompareOperator::NotEqual: return leftNumber != rightNumber;
36 case CompareOperator::Less: return leftNumber < rightNumber;
37 case CompareOperator::LessEqual: return leftNumber <= rightNumber;
38 case CompareOperator::Greater: return leftNumber > rightNumber;
39 case CompareOperator::GreaterEqual: return leftNumber >= rightNumber;
40 }
41 }
42 if (op == CompareOperator::Equal) return left == right;
43 if (op == CompareOperator::NotEqual) return !(left == right);
44 if (left.isString() && right.isString()) {
45 const auto& l = *left.getIf<std::string>();
46 const auto& r = *right.getIf<std::string>();
47 switch (op) {
48 case CompareOperator::Less: return l < r;
49 case CompareOperator::LessEqual: return l <= r;
50 case CompareOperator::Greater: return l > r;
51 case CompareOperator::GreaterEqual: return l >= r;
52 default: break;
53 }
54 }
55 return false;
56}

207bool Condition::isValid() const noexcept {
208 switch (kind_) {
209 case ConditionKind::All:
210 case ConditionKind::Any:
211 for (const auto& child : children_)
212 if (!child.isValid()) return false;
213 return true;
214 case ConditionKind::Not: return children_.size() == 1 && children_.front().isValid();
215 case ConditionKind::Compare:
216 case ConditionKind::StateEquals: return !key_.empty();
217 case ConditionKind::HasTag:
218 case ConditionKind::HasAttribute:
219 case ConditionKind::HasResource:
220 case ConditionKind::AuthorityCheck: return !key_.empty();
221 case ConditionKind::PolicyCall:
222 return !key_.empty() &&
223 (!scriptDeclaration_ || (scriptDeclaration_->name.empty() || scriptDeclaration_->name == key_));
224 }
225 return false;

228ConditionResult Condition::evaluate(const EvaluationContext& context) const {
229 if (!isValid()) return ConditionResult::failed(ConditionReasonCode::InvalidCondition);
230
231 switch (kind_) {
232 case ConditionKind::All: {
233 std::vector<ConditionResult> results;
234 results.reserve(children_.size());
235 for (const auto& child : children_) {
236 auto result = child.evaluate(context);
237 if (!result.passed()) {
238 const auto reason = result.reasonCode();
239 results.push_back(std::move(result));
240 return ConditionResult::failed(
241 reason == ConditionReasonCode::InvalidCondition ? reason : ConditionReasonCode::ChildFailed,
242 results.back().evidence(),
243 object({{"kind", Value("all")}, {"children", childDetails(results)}}));
244 }
245 results.push_back(std::move(result));
246 }
247 return ConditionResult::success({}, object({{"kind", Value("all")}, {"children", childDetails(results)}}));
248 }
249 case ConditionKind::Any: {
250 std::vector<ConditionResult> results;
251 results.reserve(children_.size());
252 for (const auto& child : children_) {
253 auto result = child.evaluate(context);
254 if (result.passed())
255 return ConditionResult::success(
256 result.evidence(), object({{"kind", Value("any")}, {"children", childDetails(results)}}));
257 results.push_back(std::move(result));
258 }
259 return ConditionResult::failed(ConditionReasonCode::NoChildPassed, {},
260 object({{"kind", Value("any")}, {"children", childDetails(results)}}));
261 }
262 case ConditionKind::Not: {
263 auto result = children_.front().evaluate(context);
264 if (result.passed())
265 return ConditionResult::failed(ConditionReasonCode::Negated, result.evidence(),
266 object({{"kind", Value("not")}, {"child", result.details()}}));
267 return ConditionResult::success(result.evidence(),
268 object({{"kind", Value("not")}, {"child", result.details()}}));
269 }
270 case ConditionKind::Compare: {
271 auto actual = context.value(key_);
272 if (!actual) return unavailable(ConditionReasonCode::MissingValue, "compare", key_);
273 const bool equal = compareValues(*actual, compare_, expected_);
274 return equal ? ConditionResult::success(*actual, object({{"key", Value(key_)},
275 {"operator", Value(compareOperatorName(compare_))},
276 {"expected", expected_}}))
277 : ConditionResult::failed(ConditionReasonCode::ValueMismatch, *actual,
278 object({{"key", Value(key_)},
279 {"operator", Value(compareOperatorName(compare_))},
280 {"expected", expected_}}));
281 }
282 case ConditionKind::HasTag: {
283 auto actual = context.hasTag(key_);
284 if (!actual) return unavailable(ConditionReasonCode::TagUnavailable, "has_tag", key_);
285 return *actual ? ConditionResult::success(Value(true), object({{"tag", Value(key_)}}))
286 : ConditionResult::failed(ConditionReasonCode::TagMissing, Value(false),
287 object({{"tag", Value(key_)}}));
288 }
289 case ConditionKind::HasAttribute: {
290 auto actual = context.attribute(key_);
291 if (!actual) return unavailable(ConditionReasonCode::AttributeMissing, "has_attribute", key_);
292 return ConditionResult::success(*actual, object({{"attribute", Value(key_)}}));
293 }
294 case ConditionKind::HasResource: {
295 auto actual = context.resource(key_);
296 if (!actual) return unavailable(ConditionReasonCode::ResourceMissing, "has_resource", key_);
297 return ConditionResult::success(*actual, object({{"resource", Value(key_)}}));
298 }
299 case ConditionKind::StateEquals: {
300 auto actual = context.state(key_);
301 if (!actual) return unavailable(ConditionReasonCode::StateMissing, "state_equals", key_);
302 return compareValues(*actual, CompareOperator::Equal, expected_)
303 ? ConditionResult::success(*actual, object({{"state", Value(key_)}, {"expected", expected_}}))
304 : ConditionResult::failed(ConditionReasonCode::StateMismatch, *actual,
305 object({{"state", Value(key_)}, {"expected", expected_}}));
306 }
307 case ConditionKind::AuthorityCheck: {
308 auto allowed = context.authority(key_);
309 if (!allowed) return unavailable(ConditionReasonCode::AuthorityUnavailable, "authority_check", key_);
310 return *allowed ? ConditionResult::success(Value(true), object({{"scope", Value(key_)}}))
311 : ConditionResult::failed(ConditionReasonCode::AuthorityDenied, Value(false),
312 object({{"scope", Value(key_)}}));
313 }
314 case ConditionKind::PolicyCall: {
315 auto result = context.policy(key_, arguments_);
316 if (!result) return unavailable(ConditionReasonCode::PolicyUnavailable, "policy_call", key_);
317 if (result->passed()) return std::move(*result);
318 return ConditionResult::failed(result->reasonCode(), result->evidence(),
319 object({{"policy", Value(key_)}, {"details", result->details()}}));
320 }
321 }
322 return ConditionResult::failed(ConditionReasonCode::InvalidCondition);
323}
