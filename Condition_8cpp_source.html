28bool compareValues(const Value& left, CompareOperator op, const Value& right) {
29 double leftNumber = 0.0;
30 double rightNumber = 0.0;
31 bool numeric = number(left, leftNumber) && number(right, rightNumber);
32 if (numeric) {
33 switch (op) {
34 case CompareOperator::Equal: return leftNumber == rightNumber;
35 case CompareOperator::NotEqual: return leftNumber != rightNumber;
36 case CompareOperator::Less: return leftNumber < rightNumber;
37 case CompareOperator::LessEqual: return leftNumber <= rightNumber;
38 case CompareOperator::Greater: return leftNumber > rightNumber;
39 case CompareOperator::GreaterEqual: return leftNumber >= rightNumber;
40 case CompareOperator::Exists: return true;
41 }
42 }
43 if (op == CompareOperator::Equal) return left == right;
44 if (op == CompareOperator::NotEqual) return !(left == right);
45 if (op == CompareOperator::Exists) return true;
46 if (left.isString() && right.isString()) {
47 const auto& l = *left.getIf<std::string>();
48 const auto& r = *right.getIf<std::string>();
49 switch (op) {
50 case CompareOperator::Less: return l < r;
51 case CompareOperator::LessEqual: return l <= r;
52 case CompareOperator::Greater: return l > r;
53 case CompareOperator::GreaterEqual: return l >= r;
54 default: break;
55 }
56 }
57 return false;
58}

187ConditionResult Condition::evaluate(const EvaluationContext& context) const {
188 if (!isValid()) return ConditionResult::failed(ConditionReasonCode::InvalidCondition);
189
190 switch (kind_) {
191 case ConditionKind::All: {
192 std::vector<ConditionResult> results;
193 results.reserve(children_.size());
194 for (const auto& child : children_) {
195 auto result = child.evaluate(context);
196 if (!result.passed()) {
197 const auto reason = result.reasonCode();
198 results.push_back(std::move(result));
199 return ConditionResult::failed(
200 reason == ConditionReasonCode::InvalidCondition ? reason : ConditionReasonCode::ChildFailed,
201 results.back().evidence(),
202 object({{"kind", Value("all")}, {"children", childDetails(results)}}));
203 }
204 results.push_back(std::move(result));
205 }
206 return ConditionResult::success({}, object({{"kind", Value("all")}, {"children", childDetails(results)}}));
207 }
208 case ConditionKind::Any: {
209 std::vector<ConditionResult> results;
210 results.reserve(children_.size());
211 for (const auto& child : children_) {
212 auto result = child.evaluate(context);
213 if (result.passed())
214 return ConditionResult::success(
215 result.evidence(), object({{"kind", Value("any")}, {"children", childDetails(results)}}));
216 results.push_back(std::move(result));
217 }
218 return ConditionResult::failed(ConditionReasonCode::NoChildPassed, {},
219 object({{"kind", Value("any")}, {"children", childDetails(results)}}));
220 }
221 case ConditionKind::Not: {
222 auto result = children_.front().evaluate(context);
223 if (result.passed())
224 return ConditionResult::failed(ConditionReasonCode::Negated, result.evidence(),
225 object({{"kind", Value("not")}, {"child", result.details()}}));
226 return ConditionResult::success(result.evidence(),
227 object({{"kind", Value("not")}, {"child", result.details()}}));
228 }
229 case ConditionKind::Compare: {
230 auto actual = context.value(key_);
231 if (!actual) return unavailable(ConditionReasonCode::MissingValue, "compare", key_);
232 if (compare_ == CompareOperator::Exists) {
233 return ConditionResult::success(
234 *actual, object({{"key", Value(key_)}, {"operator", Value(compareOperatorName(compare_))}}));
235 }
236 const bool equal = compareValues(*actual, compare_, expected_);
237 return equal ? ConditionResult::success(*actual, object({{"key", Value(key_)},
238 {"operator", Value(compareOperatorName(compare_))},
239 {"expected", expected_}}))
240 : ConditionResult::failed(ConditionReasonCode::ValueMismatch, *actual,
241 object({{"key", Value(key_)},
242 {"operator", Value(compareOperatorName(compare_))},
243 {"expected", expected_}}));
244 }
245 case ConditionKind::PolicyCall: {
246 auto result = context.policy(key_, arguments_);
247 if (!result) return unavailable(ConditionReasonCode::PolicyUnavailable, "policy_call", key_);
248 if (result->passed()) return std::move(*result);
249 return ConditionResult::failed(result->reasonCode(), result->evidence(),
250 object({{"policy", Value(key_)}, {"details", result->details()}}));
251 }
252 }
253 return ConditionResult::failed(ConditionReasonCode::InvalidCondition);
254}
