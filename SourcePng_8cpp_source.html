167 const auto at = std::size_t((y * width + x) * 4);
168 for (unsigned c = 0; c < 3; ++c) output[at + c] = std::uint8_t(color[(colorIndices >> (2 * i)) & 3][c]);

197std::vector<std::uint8_t> lzw(std::span<const std::uint8_t> source, std::size_t expected) {
198 std::array<std::uint16_t, 4096> prefix{};
199 std::array<std::uint8_t, 4096> suffix{}, stack{};
200 std::vector<std::uint8_t> out;
201 out.reserve(expected);
202 std::size_t bit = 0;
203 unsigned width = 9, next = 258, previous = 4096;
204 std::uint8_t first = 0;
205 bool cleared = false;
206 while (true) {
207 if (bit > source.size() * 8 || width > source.size() * 8 - bit) throw Invalid{};
208 unsigned code = 0;
209 for (unsigned i = 0; i < width; ++i, ++bit) code = (code << 1) | ((source[bit / 8] >> (7 - bit % 8)) & 1);
210 if (code == 256) {
211 width = 9;
212 next = 258;
213 previous = 4096;
214 cleared = true;
215 continue;
216 }
217 if (!cleared) throw Invalid{};
218 if (code == 257) {
219 if (out.size() != expected) throw Invalid{};
220 return out;
221 }
222 const unsigned original = code;
223 std::size_t length = 0;
224 if (code == next && previous != 4096) {
225 stack[length++] = first;
226 code = previous;
227 } else if (code >= next)
228 throw Invalid{};
229 while (code >= 258) {
230 if (code >= next || length >= stack.size()) throw Invalid{};
231 stack[length++] = suffix[code];
232 code = prefix[code];
233 }
234 if (code >= 256 || length >= stack.size()) throw Invalid{};
235 first = std::uint8_t(code);
236 stack[length++] = first;
237 if (length > expected - out.size()) throw Invalid{};
238 while (length) out.push_back(stack[--length]);
239 if (previous != 4096 && next < 4096) {
240 prefix[next] = std::uint16_t(previous);
241 suffix[next] = first;
242 ++next;
243 // TIFF uses early-change code widths, unlike GIF's late-change LZW.
244 if (width < 12 && next == (1u << width) - 1) ++width;
245 }
246 previous = original;
247 }
