99Result<HexSphereTopology> HexSphereTopology::build(std::int32_t subdivision) {
100 if (subdivision < 0 || subdivision > kMaxHexSphereSubdivision) {
101 return sphereFailure("hex sphere subdivision out of range");
102 }
103
104 HexSphereTopology topology;
105 topology.subdivision_ = subdivision;
106 topology.frequency_ = std::int32_t{1} << subdivision;
107
108 // --- primal polyhedron --------------------------------------------------
109 std::vector<Vec> vertices;
110 vertices.reserve(12u);
111 for (const auto& vertex : kIcosahedronVertices) vertices.push_back(normalize(Vec{vertex[0], vertex[1], vertex[2]}));
112
113 std::vector<Triangle> triangles;
114 triangles.reserve(20u);
115 for (const auto& face : kIcosahedronFaces) triangles.push_back(Triangle{face[0], face[1], face[2]});
116 for (Triangle& triangle : triangles) {
117 const Vec normal =
118 cross(vertices[triangle.b] - vertices[triangle.a], vertices[triangle.c] - vertices[triangle.a]);
119 const Vec outward = vertices[triangle.a] + vertices[triangle.b] + vertices[triangle.c];
120 if (dot(normal, outward) < 0.f) std::swap(triangle.b, triangle.c);
121 }
122
123 // --- midpoint subdivision, frequency 2^subdivision -----------------------
124 //
125 // Splitting every triangle into four and welding the shared midpoints by edge
126 // key keeps one vertex per physical point, so vertex ids are stable and the
127 // original twelve keep their ids 0..11 -- which is what makes the twelve
128 // pentagons addressable as cells [0, 12).
129 for (std::int32_t level = 0; level < subdivision; ++level) {
130 std::unordered_map<std::uint64_t, std::int32_t> midpoints;
131 midpoints.reserve(vertices.size() * 2u);
132 std::vector<Triangle> refined;
133 refined.reserve(triangles.size() * 4u);
134
135 const auto midpoint = [&](std::int32_t i, std::int32_t j) {
136 const std::uint64_t key = edgeKey(i, j);
137 const auto found = midpoints.find(key);
138 if (found != midpoints.end()) return found->second;
139 const std::int32_t index = static_cast<std::int32_t>(vertices.size());
140 vertices.push_back(normalize(vertices[i] + vertices[j]));
141 midpoints.emplace(key, index);
142 return index;
143 };
144
145 for (const Triangle& triangle : triangles) {
146 const std::int32_t ab = midpoint(triangle.a, triangle.b);
147 const std::int32_t bc = midpoint(triangle.b, triangle.c);
148 const std::int32_t ca = midpoint(triangle.c, triangle.a);
149 refined.push_back(Triangle{triangle.a, ab, ca});
150 refined.push_back(Triangle{ab, triangle.b, bc});
151 refined.push_back(Triangle{ca, bc, triangle.c});
152 refined.push_back(Triangle{ab, bc, ca});
153 }
154 triangles.swap(refined);
155 }
156
157 const std::int32_t cellCount = static_cast<std::int32_t>(vertices.size());
158 topology.directions_ = std::move(vertices);
159
160 // --- dual corners: one per primal face ----------------------------------
161 std::vector<Vec> cornerDirections(triangles.size());
162 for (std::size_t i = 0; i < triangles.size(); ++i) {
163 const Triangle& triangle = triangles[i];
164 cornerDirections[i] = normalize(topology.directions_[triangle.a] + topology.directions_[triangle.b] +
165 topology.directions_[triangle.c]);
166 }
167
168 // --- vertex -> incident face incidence, in CSR form ----------------------
169 const std::int32_t faceCount = static_cast<std::int32_t>(triangles.size());
170 std::vector<std::int32_t> incidenceCount(static_cast<std::size_t>(cellCount), 0);
171 for (const Triangle& triangle : triangles) {
172 ++incidenceCount[triangle.a];
173 ++incidenceCount[triangle.b];
174 ++incidenceCount[triangle.c];
175 }
176 std::vector<std::int32_t> incidenceOffset(static_cast<std::size_t>(cellCount) + 1u, 0);
177 for (std::int32_t cell = 0; cell < cellCount; ++cell) {
178 incidenceOffset[cell + 1] = incidenceOffset[cell] + incidenceCount[cell];
179 }
180 std::vector<std::int32_t> incidence(static_cast<std::size_t>(incidenceOffset[cellCount]), -1);
181 {
182 std::vector<std::int32_t> cursor(incidenceOffset.begin(), incidenceOffset.end() - 1);
183 for (std::int32_t face = 0; face < faceCount; ++face) {
184 const Triangle& triangle = triangles[face];
185 incidence[cursor[triangle.a]++] = face;
186 incidence[cursor[triangle.b]++] = face;
187 incidence[cursor[triangle.c]++] = face;
188 }
189 }
190
191 // --- edge -> its one or two faces ---------------------------------------
192 std::unordered_map<std::uint64_t, EdgeFaces> edgeFaces;
193 edgeFaces.reserve(triangles.size() * 3u);
194 for (std::int32_t face = 0; face < faceCount; ++face) {
195 const Triangle& triangle = triangles[face];
196 const std::int32_t pairs[3][2] = {{triangle.a, triangle.b}, {triangle.b, triangle.c}, {triangle.c, triangle.a}};
197 for (const auto& pair : pairs) {
198 EdgeFaces& entry = edgeFaces[edgeKey(pair[0], pair[1])];
199 if (entry.first < 0)
200 entry.first = face;
201 else if (entry.second < 0)
202 entry.second = face;
203 }
204 }
205
206 const auto slotOf = [&](std::int32_t face, std::int32_t vertex) {
207 const Triangle& triangle = triangles[face];
208 if (triangle.a == vertex) return 0;
209 if (triangle.b == vertex) return 1;
210 return 2;
211 };
212 const auto vertexAt = [&](std::int32_t face, std::int32_t slot) {
213 const Triangle& triangle = triangles[face];
214 if (slot == 0) return triangle.a;
215 if (slot == 1) return triangle.b;
216 return triangle.c;
217 };
218 const auto nextVertexInFace = [&](std::int32_t face, std::int32_t vertex) {
219 return vertexAt(face, (slotOf(face, vertex) + 1) % 3);
220 };
221 const auto otherFaceOfEdge = [&](std::int32_t face, std::int32_t i, std::int32_t j) {
222 const auto found = edgeFaces.find(edgeKey(i, j));
223 if (found == edgeFaces.end()) return -1;
224 if (found->second.first == face) return found->second.second;
225 if (found->second.second == face) return found->second.first;
226 return -1;
227 };
228
229 // --- rotate around every vertex to get its cyclic corner and edge order ---
230 topology.cellOffsets_.assign(static_cast<std::size_t>(cellCount) + 1u, 0);
231 topology.neighbors_.assign(static_cast<std::size_t>(incidenceOffset[cellCount]), 0);
232 topology.corners_.assign(topology.neighbors_.size(), 0);
233
234 std::vector<std::int32_t> orderedFaces(6, -1);
235 for (std::int32_t cell = 0; cell < cellCount; ++cell) {
236 const std::int32_t begin = incidenceOffset[cell];
237 const std::int32_t degree = incidenceOffset[cell + 1] - begin;
238 if (degree < 5 || degree > 6) return sphereFailure("hex sphere cell has an unsupported edge count");
239 topology.cellOffsets_[cell] = begin;
240
241 const std::int32_t firstFace = incidence[begin];
242 std::int32_t current = firstFace;
243 for (std::int32_t direction = 0; direction < degree; ++direction) {
244 orderedFaces[direction] = current;
245 const std::int32_t nextFace = otherFaceOfEdge(current, cell, nextVertexInFace(current, cell));
246 if (nextFace < 0) return sphereFailure("hex sphere half-edge walk left the surface");
247 current = nextFace;
248 }
249 if (current != firstFace) return sphereFailure("hex sphere half-edge walk did not close");
250
251 for (std::int32_t direction = 0; direction < degree; ++direction) {
252 topology.corners_[begin + direction] = orderedFaces[direction];
253 topology.neighbors_[begin + direction] = nextVertexInFace(orderedFaces[direction], cell);
254 }
255 }
256 topology.cellOffsets_[cellCount] = static_cast<std::int32_t>(topology.neighbors_.size());
257 topology.cornerDirections_ = std::move(cornerDirections);
258
259 // --- handedness ---------------------------------------------------------
260 //
261 // Walking "the other face of the edge to the next vertex" is a uniform rule,
262 // so every cell comes out with the same handedness and the only question is
263 // which one. The shared edge of two adjacent cells must expose the same two
264 // corners from both sides, in reversed order; if it does not, reversing every
265 // cell's cyclic order flips the handedness of all of them at once.
266 const auto cornersAgree = [&topology]() {
267 for (std::int32_t cell = 0; cell < topology.cellCount(); ++cell) {
268 const std::int32_t degree = topology.neighborCount(cell);
269 for (std::int32_t direction = 0; direction < degree; ++direction) {
270 const HexSphereCell other = topology.neighbor(cell, direction);
271 const std::int32_t back = topology.directionOf(other, cell);
272 if (back < 0) return false;
273 const std::int32_t otherDegree = topology.neighborCount(other);
274 const std::int32_t here = topology.corners_[topology.cellOffsets_[cell] + direction];
275 const std::int32_t hereNext = topology.corners_[topology.cellOffsets_[cell] + (direction + 1) % degree];
276 const std::int32_t there = topology.corners_[topology.cellOffsets_[other] + back];
277 const std::int32_t thereNext =
278 topology.corners_[topology.cellOffsets_[other] + (back + 1) % otherDegree];
279 if (here != thereNext || hereNext != there) return false;
280 }
281 }
282 return true;
283 };
284
285 if (!cornersAgree()) {
286 for (std::int32_t cell = 0; cell < cellCount; ++cell) {
287 const auto begin = topology.neighbors_.begin() + topology.cellOffsets_[cell];
288 const auto end = topology.neighbors_.begin() + topology.cellOffsets_[cell + 1];
289 const auto cornerBegin = topology.corners_.begin() + topology.cellOffsets_[cell];
290 const auto cornerEnd = topology.corners_.begin() + topology.cellOffsets_[cell + 1];
291 std::reverse(begin, end);
292 std::reverse(cornerBegin, cornerEnd);
293 }
294 if (!cornersAgree()) return sphereFailure("hex sphere winding is inconsistent");
295 }
296
297 // --- Euler characteristic ------------------------------------------------
298 if (topology.cellCount() - topology.edgeCount() + topology.cornerCount() != 2) {
299 return sphereFailure("hex sphere does not close into a sphere");
300 }
301
302 return Result<HexSphereTopology>::success(std::move(topology));
303}

315std::int32_t HexSphereTopology::directionOf(HexSphereCell cell, HexSphereCell other) const noexcept {
316 if (!contains(cell) || !contains(other)) return kNoHexSphereCell;
317 const std::int32_t begin = cellOffsets_[static_cast<std::size_t>(cell)];
318 const std::int32_t degree = neighborCount(cell);
319 for (std::int32_t direction = 0; direction < degree; ++direction) {
320 if (neighbors_[static_cast<std::size_t>(begin + direction)] == other) return direction;
321 }
322 return kNoHexSphereCell;
323}

346HexSphereCell HexSphereTopology::cellAt(HexVec3 unitDirection) const noexcept {
347 if (empty()) return kNoHexSphereCell;
348 const Vec target = normalize(unitDirection);
349
350 HexSphereCell current = 0;
351 float best = dot(directions_[0], target);
352 const std::int32_t seeds = cellCount() < 12 ? cellCount() : 12;
353 for (std::int32_t cell = 1; cell < seeds; ++cell) {
354 const float score = dot(directions_[static_cast<std::size_t>(cell)], target);
355 if (score > best) {
356 best = score;
357 current = cell;
358 }
359 }
360
361 for (;;) {
362 bool improved = false;
363 const std::int32_t degree = neighborCount(current);
364 for (std::int32_t direction = 0; direction < degree; ++direction) {
365 const HexSphereCell candidate =
366 neighbors_[static_cast<std::size_t>(cellOffsets_[static_cast<std::size_t>(current)] + direction)];
367 const float score = dot(directions_[static_cast<std::size_t>(candidate)], target);
368 if (score > best + 1e-7f) {
369 best = score;
370 current = candidate;
371 improved = true;
372 }
373 }
374 if (!improved) break;
375 }
376 return current;
377}

412[[nodiscard]] std::string packCells(const std::vector<HexCellData>& cells) {
413 std::string packed(cells.size() * 8u, '\0');
414 auto* bytes = reinterpret_cast<unsigned char*>(packed.data());
415 for (std::size_t i = 0; i < cells.size(); ++i) {
416 writeU32LE(bytes + i * 8u, cells[i].values.raw());
417 writeU32LE(bytes + i * 8u + 4u, cells[i].flags.raw());
418 }
419 return packed;
420}

422[[nodiscard]] Result<std::vector<HexCellData>> unpackCells(std::string_view packed, std::size_t expected) {
423 if (packed.size() != expected * 8u)
424 return Result<std::vector<HexCellData>>::failure(
425 Diagnostic::error(DiagnosticCode::InvalidArgument,
426 "hex terrain bake cell payload length does not match the cell count", "hexmap"));
427 std::vector<HexCellData> cells(expected);
428 const auto* bytes = reinterpret_cast<const unsigned char*>(packed.data());
429 for (std::size_t i = 0; i < expected; ++i) {
430 cells[i].values = HexValues{readU32LE(bytes + i * 8u)};
431 cells[i].flags = HexFlags{readU32LE(bytes + i * 8u + 4u)};
432 }
433 return Result<std::vector<HexCellData>>::success(std::move(cells));
434}

505Result<HexTerrainBake> hexTerrainBakeFromValue(const Value& value) {
506 if (!value.isObject()) return invalidBake("hex terrain bake must be an object");
507 const Value* kindField = value.find("kind");
508 if (kindField == nullptr || !kindField->isString()) return invalidBake("hex terrain bake is missing string 'kind'");
509 const std::string& kind = kindField->asString();
510
511 HexTerrainBake bake;
512 if (kind == "hex.sphere") {
513 bake.kind = HexTerrainBake::Kind::Sphere;
514 auto subdivision = readInt32(value, "subdivision");
515 if (!subdivision) return Result<HexTerrainBake>::failure(subdivision.status());
516 bake.subdivision = subdivision.value();
517 const Value* radiusField = value.find("radius");
518 if (radiusField == nullptr || !radiusField->isNumeric())
519 return invalidBake("hex sphere bake is missing numeric 'radius'");
520 bake.radius = radiusField->isDouble() ? static_cast<float>(radiusField->asDouble())
521 : static_cast<float>(radiusField->asInt());
522 auto cellCount = readInt32(value, "cellCount");
523 if (!cellCount) return Result<HexTerrainBake>::failure(cellCount.status());
524 bake.cellCount = cellCount.value();
525 } else if (kind == "hex.terrain") {
526 bake.kind = HexTerrainBake::Kind::Planar;
527 auto cellCountX = readInt32(value, "cellCountX");
528 auto cellCountZ = readInt32(value, "cellCountZ");
529 if (!cellCountX) return Result<HexTerrainBake>::failure(cellCountX.status());
530 if (!cellCountZ) return Result<HexTerrainBake>::failure(cellCountZ.status());
531 bake.cellCountX = cellCountX.value();
532 bake.cellCountZ = cellCountZ.value();
533 bake.cellCount = bake.cellCountX * bake.cellCountZ;
534 } else {
535 return invalidBake("hex terrain bake kind must be 'hex.terrain' or 'hex.sphere'");
536 }
537
538 auto seed = readUInt32(value, "seed");
539 if (!seed) return Result<HexTerrainBake>::failure(seed.status());
540 bake.seed = seed.value();
541
542 const Value* cellsField = value.find("cells");
543 if (cellsField == nullptr || !cellsField->isString())
544 return invalidBake("hex terrain bake is missing packed 'cells'");
545 if (bake.cellCount < 0) return invalidBake("hex terrain bake cell count must not be negative");
546 auto cells = unpackCells(cellsField->asString(), static_cast<std::size_t>(bake.cellCount));
547 if (!cells) return Result<HexTerrainBake>::failure(cells.status());
548 bake.cells = std::move(cells).takeValue();
549 return Result<HexTerrainBake>::success(std::move(bake));
550}

552Result<void> applyHexTerrain(HexMap& map, const HexTerrainBake& bake) {
553 if (bake.kind != HexTerrainBake::Kind::Planar)
554 return Result<void>::failure(
555 Diagnostic::error(DiagnosticCode::InvalidArgument, "applyHexTerrain requires a planar bake", "hexmap"));
556 if (bake.cellCountX <= 0 || bake.cellCountZ <= 0)
557 return Result<void>::failure(
558 Diagnostic::error(DiagnosticCode::InvalidArgument, "planar bake size must be positive", "hexmap"));
559 if (static_cast<std::int32_t>(bake.cells.size()) != bake.cellCountX * bake.cellCountZ)
560 return Result<void>::failure(Diagnostic::error(
561 DiagnosticCode::InvalidArgument, "planar bake cell count does not match width and height", "hexmap"));
562
563 auto reset = map.reset(bake.cellCountX, bake.cellCountZ, bake.seed);
564 if (!reset) return reset;
565 for (std::int32_t index = 0; index < map.cellCount(); ++index) {
566 const HexCellData& cell = bake.cells[static_cast<std::size_t>(index)];
567 auto written = map.setCellState(map.coordinatesAt(index), cell.values, cell.flags);
568 if (!written) return written;
569 }
570 map.markAllChunksDirty();
571 return Result<void>::success();
572}

574Result<void> applyHexSphereTerrain(HexSphereMap& map, const HexTerrainBake& bake) {
575 if (bake.kind != HexTerrainBake::Kind::Sphere)
576 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
577 "applyHexSphereTerrain requires a sphere bake", "hexmap"));
578 if (map.empty())
579 return Result<void>::failure(Diagnostic::error(
580 DiagnosticCode::InvalidArgument, "applyHexSphereTerrain requires an existing sphere topology", "hexmap"));
581 if (bake.cellCount <= 0 || static_cast<std::int32_t>(bake.cells.size()) != bake.cellCount)
582 return Result<void>::failure(
583 Diagnostic::error(DiagnosticCode::InvalidArgument, "sphere bake cell count is inconsistent", "hexmap"));
584 if (map.cellCount() != bake.cellCount || map.subdivision() != bake.subdivision)
585 return Result<void>::failure(Diagnostic::error(
586 DiagnosticCode::InvalidArgument,
587 "sphere bake does not match the live topology; call newSphere with the same subdivision", "hexmap"));
588
589 for (std::int32_t cell = 0; cell < bake.cellCount; ++cell) {
590 const HexCellData& record = bake.cells[static_cast<std::size_t>(cell)];
591 auto written = map.setCellState(cell, record.values, record.flags);
592 if (!written) return written;
593 }
594 map.markAllDirty();
595 return Result<void>::success();
596}

617[[nodiscard]] bool isDegenerate(HexVec3 a, HexVec3 b, HexVec3 c) noexcept {
618 const HexVec3 ab = b - a;
619 const HexVec3 ac = c - a;
620 const HexVec3 normal{ab.y * ac.z - ab.z * ac.y, ab.z * ac.x - ab.x * ac.z, ab.x * ac.y - ab.y * ac.x};
621 return (normal.x * normal.x + normal.y * normal.y + normal.z * normal.z) <= 1e-8f;
622}

634void emitTriangle(HexMeshData& out, const SurfaceVertex& a, const SurfaceVertex& b, const SurfaceVertex& c,
635 const HexNoise& noise) {
636 const HexVec3 pa = noise.perturb(a.position);
637 const HexVec3 pb = noise.perturb(b.position);
638 const HexVec3 pc = noise.perturb(c.position);
639 if (isDegenerate(pa, pb, pc)) return;
640 out.addTriangle(out.addVertex(pa, a.u, a.v), out.addVertex(pb, b.u, b.v), out.addVertex(pc, c.u, c.v));
641}

652 const SurfaceVertex& r3, const HexNoise& noise) {

654 emitTriangle(out, r1, r3, r2, noise);

687void appendOpenWater(const HexMap& map, HexCoordinates coordinates, HexDirection direction, HexVec3 centre,
688 const HexNoise& noise, HexMeshData& out) {
689 const SurfaceVertex c1{centre + HexMetrics::firstWaterCorner(direction), 0.f, 0.f};
690 const SurfaceVertex c2{centre + HexMetrics::secondWaterCorner(direction), 0.f, 0.f};
691 emitTriangle(out, SurfaceVertex{centre, 0.f, 0.f}, c1, c2, noise);
692
693 HexCoordinates neighbour{};
694 if (static_cast<std::int32_t>(direction) > static_cast<std::int32_t>(HexDirection::SE) ||
695 !map.getNeighbor(coordinates, direction, neighbour)) {
696 // Only the NE, E and SE edges bridge outward: for a water/water boundary the
697 // other side would emit the same quad, so exactly one chunk owns it. An
698 // off-grid neighbour has nothing to bridge to.
699 return;
700 }
701
702 const HexVec3 offset = HexMetrics::waterBridge(direction);
703 const float farY = waterHeightOf(map, neighbour, centre.y);
704 const SurfaceVertex e1{{c1.position.x + offset.x, farY, c1.position.z + offset.z}, 0.f, 0.f};
705 const SurfaceVertex e2{{c2.position.x + offset.x, farY, c2.position.z + offset.z}, 0.f, 0.f};
706 emitQuad(out, c1, c2, e2, e1, noise);
707
708 if (static_cast<std::int32_t>(direction) > static_cast<std::int32_t>(HexDirection::E)) return;
709 HexCoordinates nextNeighbour{};
710 if (!map.getNeighbor(coordinates, next(direction), nextNeighbour) || !isWater(map, nextNeighbour)) return;
711
712 // The third cell of the junction is water too, so its own wedge stops at the
713 // water corner and this triangle fills the gap between the two bridges.
714 HexVec3 corner = map.cellGroundPosition(nextNeighbour) + HexMetrics::firstWaterCorner(previous(direction));
715 corner.y = farY;
716 emitTriangle(out, c2, e2, SurfaceVertex{corner, 0.f, 0.f}, noise);
717}
718
740void appendWaterShore(const HexMap& map, HexCoordinates coordinates, HexDirection direction, HexVec3 centre,
741 const HexNoise& noise, HexMeshData& out) {
742 const EdgeVertices near{centre + HexMetrics::firstWaterCorner(direction),
743 centre + HexMetrics::secondWaterCorner(direction)};
744
745 const SurfaceVertex nearSamples[5] = {SurfaceVertex{near.v1, 0.f, 0.f}, SurfaceVertex{near.v2, 0.f, 0.f},
746 SurfaceVertex{near.v3, 0.f, 0.f}, SurfaceVertex{near.v4, 0.f, 0.f},
747 SurfaceVertex{near.v5, 0.f, 0.f}};
748 const SurfaceVertex wedgeCentre{centre, 0.f, 0.f};
749 for (std::int32_t i = 0; i < 4; ++i) {
750 emitTriangle(out, wedgeCentre, nearSamples[i], nearSamples[i + 1], noise);
751 }
752
753 const HexCoordinates neighbour = coordinates.step(direction);
754 const HexVec3 neighbourGround = map.cellGroundPosition(neighbour);
755 const HexVec3 bank{neighbourGround.x, centre.y, neighbourGround.z};
756 // Second corner first, so sample `i` of the far edge lines up with sample `i` of
757 // the near edge across the strip - the reference orders them the same way.
758 const EdgeVertices far{bank + HexMetrics::secondSolidCorner(opposite(direction)),
759 bank + HexMetrics::firstSolidCorner(opposite(direction))};
760
761 const SurfaceVertex farSamples[5] = {SurfaceVertex{far.v1, 1.f, 0.f}, SurfaceVertex{far.v2, 1.f, 0.f},
762 SurfaceVertex{far.v3, 1.f, 0.f}, SurfaceVertex{far.v4, 1.f, 0.f},
763 SurfaceVertex{far.v5, 1.f, 0.f}};
764 for (std::int32_t i = 0; i < 4; ++i) {
765 emitQuad(out, nearSamples[i], nearSamples[i + 1], farSamples[i + 1], farSamples[i], noise);
766 }
767
768 // Close the junction with the next cell around: its water edge when it is flooded,
769 // its bank corner when it is land. Either way the vertex stays at the water height.
770 HexCoordinates nextNeighbour{};
771 if (!map.getNeighbor(coordinates, next(direction), nextNeighbour)) return;
772 const HexVec3 corner =
773 isWater(map, nextNeighbour)
774 ? map.cellGroundPosition(nextNeighbour) + HexMetrics::firstWaterCorner(previous(direction))
775 : map.cellGroundPosition(nextNeighbour) + HexMetrics::firstSolidCorner(previous(direction));
776 HexVec3 capped = corner;
777 capped.y = centre.y;
778 emitTriangle(out, nearSamples[4], farSamples[4], SurfaceVertex{capped, 0.f, 0.f}, noise);
779}

783void buildWaterMesh(const HexMap& map, std::int32_t chunkIndex, HexMeshData& out) {
784 out.clear();
785 if (chunkIndex < 0 || chunkIndex >= map.chunkCount()) {
786 out.finalize();
787 return;
788 }
789
790 const HexNoise& noise = map.noise();
791
792 for (std::int32_t row = 0; row < HexMetrics::kChunkSizeZ; ++row) {
793 for (std::int32_t column = 0; column < HexMetrics::kChunkSizeX; ++column) {
794 const HexCoordinates coordinates = map.chunkCell(chunkIndex, column, row);
795 if (map.cell(coordinates) == nullptr || !map.isUnderwater(coordinates)) continue;
796
797 const HexVec3 ground = map.cellGroundPosition(coordinates);
798 const HexVec3 centre{ground.x, HexMetrics::waterSurfaceY(map.waterLevel(coordinates)), ground.z};
799
800 // Dispatch per direction, exactly like the reference: a flooded direction
801 // gets the open-water wedge, a land direction gets the shore wedge and its
802 // strip. Emitting a full-cell fan up front instead would double the centre
803 // triangles of every shoreline direction and leave the strip to fight it.
804 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
805 const auto direction = static_cast<HexDirection>(i);
806 HexCoordinates neighbour{};
807 const bool hasNeighbour = map.getNeighbor(coordinates, direction, neighbour);
808 const bool neighbourWater = hasNeighbour && isWater(map, neighbour);
809 if (hasNeighbour && !neighbourWater) {
810 appendWaterShore(map, coordinates, direction, centre, noise, out);
811 } else {
812 appendOpenWater(map, coordinates, direction, centre, noise, out);
813 }
814 }
815 }
816 }
817
818 out.finalize();
819}
