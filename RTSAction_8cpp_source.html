62Result<ActionExecutionResult> ActionAdapter::execute(Unit& unit, const OrderRecord& order, const SimulationStep& step) {
63 if (step.delta.nanoseconds() < 0)
64 return failure<ActionExecutionResult>(DiagnosticCode::InvalidArgument,
65 "RTS action step delta must be non-negative", "step.delta");
66 if (!impl_)
67 return failure<ActionExecutionResult>(DiagnosticCode::InvariantViolation,
68 "RTS action adapter is not initialized", "adapter");
69
70 const ecs::EntityHandle unitHandle = ecs::handle_of(&unit);
71 auto pendingForUnit =
72 std::find_if(impl_->pending.begin(), impl_->pending.end(),
73 [&unitHandle](const Impl::Pending& pending) { return sameHandle(pending.unit, unitHandle); });
74
75 if (pendingForUnit != impl_->pending.end() && pendingForUnit->orderId != order.id) {
76 const auto* oldExecution = impl_->runtime.find(pendingForUnit->execution);
77 if (oldExecution != nullptr && !terminal(oldExecution->phase())) {
78 auto cancelled = impl_->runtime.cancel(pendingForUnit->execution, step.tick);
79 if (!cancelled) cancelled.ignore("RTS order replacement could not cancel old action");
80 }
81 impl_->pending.erase(pendingForUnit);
82 pendingForUnit = impl_->pending.end();
83 }
84
85 action::ActionExecutionId executionId;
86 if (pendingForUnit == impl_->pending.end()) {
87 const auto logicalId = actionIdFor(order);
88 if (!logicalId)
89 return failure<ActionExecutionResult>(
90 DiagnosticCode::InvalidArgument, "RTS order definition is not a valid LogicalId", "order.definitionId");
91
92 action::ActionDefinition definition;
93 definition.id = *logicalId;
94
95 action::ActionRequest request;
96 request.actionId = definition.id;
97 request.source = unitHandle;
98 request.requestedTick = step.tick;
99
100 auto submitted = impl_->runtime.submit(std::move(definition), std::move(request));
101 if (!submitted) return failureFrom<ActionExecutionResult>(submitted.status());
102 executionId = std::move(submitted).takeValue();
103 impl_->pending.push_back({unitHandle, order.id, executionId});
104 } else {
105 executionId = pendingForUnit->execution;
106 }
107
108 auto advanced = impl_->runtime.advance(executionId, step.tick, step.delta);
109 if (!advanced) {
110 impl_->pending.erase(
111 std::remove_if(impl_->pending.begin(), impl_->pending.end(),
112 [&executionId](const Impl::Pending& pending) { return pending.execution == executionId; }),
113 impl_->pending.end());
114 return failureFrom<ActionExecutionResult>(advanced.status());
115 }
116
117 const auto summary = std::move(advanced).takeValue();
118 bool completed = summary.phase == action::ActionPhase::Completed;
119 if (order.kind == OrderKind::Move && !unit.motion()->arrived) completed = false;
120 if (completed) {
121 impl_->pending.erase(
122 std::remove_if(impl_->pending.begin(), impl_->pending.end(),
123 [&executionId](const Impl::Pending& pending) { return pending.execution == executionId; }),
124 impl_->pending.end());
125 }
126
127 const ActionExecutionResult result{completed ? ActionDisposition::Completed : ActionDisposition::Pending};
128 return Result<ActionExecutionResult>::success(
129 result, Status::success(completed ? StatusCode::Applied : StatusCode::Pending));
130}
