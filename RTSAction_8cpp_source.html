56Result<ActionExecutionResult> ActionAdapter::execute(Unit& unit, const OrderRecord& order, const SimulationStep& step) {
57 if (step.delta.nanoseconds() < 0)
58 return Result<ActionExecutionResult>::failure(Diagnostic::error(
59 DiagnosticCode::InvalidArgument, "RTS action step delta must be non-negative", "step.delta"));
60 if (!impl_)
61 return Result<ActionExecutionResult>::failure(
62 Diagnostic::error(DiagnosticCode::InvariantViolation, "RTS action adapter is not initialized", "adapter"));
63
64 const ecs::EntityHandle unitHandle = ecs::handle_of(&unit);
65 auto pendingForUnit =
66 std::find_if(impl_->pending.begin(), impl_->pending.end(),
67 [&unitHandle](const Impl::Pending& pending) { return sameHandle(pending.unit, unitHandle); });
68
69 if (pendingForUnit != impl_->pending.end() && pendingForUnit->orderId != order.id) {
70 const auto* oldExecution = impl_->runtime.find(pendingForUnit->execution);
71 if (oldExecution != nullptr && !terminal(oldExecution->phase())) {
72 auto cancelled = impl_->runtime.cancel(pendingForUnit->execution, step.tick);
73 if (!cancelled) cancelled.ignore("RTS order replacement could not cancel old action");
74 }
75 impl_->pending.erase(pendingForUnit);
76 pendingForUnit = impl_->pending.end();
77 }
78
79 action::ActionExecutionId executionId;
80 if (pendingForUnit == impl_->pending.end()) {
81 const auto logicalId = actionIdFor(order);
82 if (!logicalId)
83 return Result<ActionExecutionResult>::failure(
84 Diagnostic::error(DiagnosticCode::InvalidArgument, "RTS order definition is not a valid LogicalId",
85 "order.definitionId"));
86
87 action::ActionDefinition definition;
88 definition.id = *logicalId;
89
90 action::ActionRequest request;
91 request.actionId = definition.id;
92 request.source = unitHandle;
93 request.requestedTick = step.tick;
94
95 auto submitted = impl_->runtime.submit(std::move(definition), std::move(request));
96 if (!submitted) return Result<ActionExecutionResult>::failure(submitted.status());
97 executionId = std::move(submitted).takeValue();
98 impl_->pending.push_back({unitHandle, order.id, executionId});
99 } else {
100 executionId = pendingForUnit->execution;
101 }
102
103 auto advanced = impl_->runtime.advance(executionId, step.tick, step.delta);
104 if (!advanced) {
105 impl_->pending.erase(
106 std::remove_if(impl_->pending.begin(), impl_->pending.end(),
107 [&executionId](const Impl::Pending& pending) { return pending.execution == executionId; }),
108 impl_->pending.end());
109 return Result<ActionExecutionResult>::failure(advanced.status());
110 }
111
112 const auto summary = std::move(advanced).takeValue();
113 bool completed = summary.phase == action::ActionPhase::Completed;
114 if (order.kind == OrderKind::Move && !unit.motion()->arrived) completed = false;
115 if (completed) {
116 impl_->pending.erase(
117 std::remove_if(impl_->pending.begin(), impl_->pending.end(),
118 [&executionId](const Impl::Pending& pending) { return pending.execution == executionId; }),
119 impl_->pending.end());
120 }
121
122 const ActionExecutionResult result{completed ? ActionDisposition::Completed : ActionDisposition::Pending};
123 return Result<ActionExecutionResult>::success(
124 result, Status::success(completed ? StatusCode::Applied : StatusCode::Pending));
125}

138Result<Value::Object> definitionObject(definitions::DefinitionRegistry& registry, std::string_view type,
139 const LogicalId& id) {
140 if (!id.isValid() || id.name().empty())
141 return Result<Value::Object>::failure(
142 Diagnostic::error(DiagnosticCode::InvalidArgument, "RTS root requires a logical definition",
143 std::string(type) + ".definition", {}, "rts.archetype"));
144 auto resolved = registry.resolve(std::string(type), std::string(id.name()));
145 if (!resolved) return Result<Value::Object>::failure(resolved.status());
146 auto parsed = Value::fromJson(resolved.value().get().json);
147 if (!parsed) return Result<Value::Object>::failure(parsed.status());
148 auto value = std::move(parsed).takeValue();
149 const auto* object = value.getIf<Value::Object>();
150 if (object == nullptr)
151 return Result<Value::Object>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
152 "RTS definition must be a JSON object",
153 std::string(type) + ".json", {}, "rts.archetype"));
154 return Result<Value::Object>::success(*object);
155}

157Result<float> number(const Value::Object& object, std::string_view name, float fallback,
158 float minimum = -std::numeric_limits<float>::max()) {
159 const Value* source = field(object, name);
160 double value = fallback;
161 if (source != nullptr) {
162 if (const auto* real = source->getIf<double>())
163 value = *real;
164 else if (const auto* integer = source->getIf<std::int64_t>())
165 value = static_cast<double>(*integer);
166 else
167 return Result<float>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
168 "RTS archetype numeric field must be a number",
169 std::string(name), {}, "rts.archetype"));
170 }
171 if (!std::isfinite(value) || value < minimum || value > std::numeric_limits<float>::max())
172 return Result<float>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
173 "RTS archetype numeric field is outside its valid range",
174 std::string(name), {}, "rts.archetype"));
175 return Result<float>::success(static_cast<float>(value));
176}

218Result<TagSet> tags(const Value::Object& object, std::string_view name) {
219 TagSet result;
220 const Value* source = field(object, name);
221 if (source == nullptr) return Result<TagSet>::success(std::move(result));
222 const auto* array = source->getIf<Value::Array>();
223 if (array == nullptr)
224 return Result<TagSet>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
225 "RTS archetype tags must be an array", std::string(name), {},
226 "rts.archetype"));
227 for (std::size_t index = 0; index < array->size(); ++index) {
228 const auto* value = (*array)[index].getIf<std::string>();
229 if (value == nullptr)
230 return Result<TagSet>::failure(
231 Diagnostic::error(DiagnosticCode::InvalidArgument, "RTS archetype tag must be a string",
232 std::string(name) + "[" + std::to_string(index) + "]", {}, "rts.archetype"));
233 auto added = result.add(*value);
234 if (!added) return Result<TagSet>::failure(added.status());
235 }
236 return Result<TagSet>::success(std::move(result));
237}
