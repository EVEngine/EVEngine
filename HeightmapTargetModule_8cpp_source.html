57Result<std::string> HeightmapTargetModule::encodeDocument(const procgen::Heightmap* heightmap, float spacingX,
58 float spacingZ) const {
59 if (!heightmap)
60 return Result<std::string>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
61 "terrain document requires a heightmap", {}, {},
62 "heightmap-target.document"));
63 auto encoded = encodeTerrainDocument(*heightmap, spacingX, spacingZ);
64 if (!encoded) return Result<std::string>::failure(encoded.status());
65 return Result<std::string>::success(editor::editorValueToJson(encoded.value()));
66}

68Result<void> HeightmapTargetModule::decodeDocument(std::string_view json, procgen::Heightmap* heightmap) const {
69 if (!heightmap)
70 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
71 "terrain document requires a destination heightmap", {}, {},
72 "heightmap-target.document"));
73 auto parsed = editor::editorValueFromJson(std::string(json));
74 if (!parsed) return Result<void>::failure(parsed.status());
75 auto decoded = decodeTerrainDocument(parsed.value());
76 if (!decoded) return Result<void>::failure(decoded.status());
77 *heightmap = std::move(decoded).takeValue().heightmap;
78 return Result<void>::success();
79}
