196 static void unpackKey(uint64_t k, int &cx, int &cy, int &cz) {
197 auto signExtend21 = [](uint32_t v) -> int {
198 if (v & 0x100000u) return int(v | 0xFFE00000u);
199 return int(v);
200 };
201 cx = signExtend21(uint32_t(k & 0x1FFFFFu));
202 cy = signExtend21(uint32_t((k >> 21) & 0x1FFFFFu));
203 cz = signExtend21(uint32_t((k >> 42) & 0x1FFFFFu));
204 }
