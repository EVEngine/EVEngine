217 static uint64_t key(int cx, int cy, int cz) {
218 const uint64_t ux = uint64_t(uint32_t(cx) & 0x1FFFFFu);
219 const uint64_t uy = uint64_t(uint32_t(cy) & 0x1FFFFFu);
220 const uint64_t uz = uint64_t(uint32_t(cz) & 0x1FFFFFu);
221 return ux | (uy << 21) | (uz << 42);

224 static void unpackKey(uint64_t k, int &cx, int &cy, int &cz) {
225 auto signExtend21 = [](uint32_t v) -> int {
226 if (v & 0x100000u) return int(v | 0xFFE00000u);
227 return int(v);
228 };
229 cx = signExtend21(uint32_t(k & 0x1FFFFFu));
230 cy = signExtend21(uint32_t((k >> 21) & 0x1FFFFFu));
231 cz = signExtend21(uint32_t((k >> 42) & 0x1FFFFFu));
232 }
