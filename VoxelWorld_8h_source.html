245 static uint64_t key(int cx, int cy, int cz) {
246 const uint64_t ux = uint64_t(uint32_t(cx) & 0x1FFFFFu);
247 const uint64_t uy = uint64_t(uint32_t(cy) & 0x1FFFFFu);

252 static void unpackKey(uint64_t k, int &cx, int &cy, int &cz) {
253 auto signExtend21 = [](uint32_t v) -> int {
254 if (v & 0x100000u) return int(v | 0xFFE00000u);
255 return int(v);
256 };
257 cx = signExtend21(uint32_t(k & 0x1FFFFFu));
258 cy = signExtend21(uint32_t((k >> 21) & 0x1FFFFFu));
259 cz = signExtend21(uint32_t((k >> 42) & 0x1FFFFFu));
260 }
