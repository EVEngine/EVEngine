27// ---- tiny deterministic RNG (no dependency on <random>) ----
28struct Lcg {
29 uint32_t s = 0x9e3779b9u;
30 uint32_t next() {
31 s = s * 1664525u + 1013904223u;
32 return s;
33 }
34 float unit() { return float(next() % 10000u) / 9999.f; } // [0,1]
35 float range(float a, float b) { return a + (b - a) * unit(); }
36};

53// Vertical rain streak: bright core, alpha falloff toward the edges/ends.
54void genRainTexture(std::vector<uint8_t> &rgba) {
55 const int w = 8, h = 32;
56 rgba.resize(size_t(w) * h * 4);
57 for (int y = 0; y < h; ++y) {
58 const float v = (float(y) + 0.5f) / float(h); // 0 top -> 1 bottom
59 const float endFade = 1.f - 2.f * std::fabs(v - 0.5f); // 0 at ends, 1 center
60 for (int x = 0; x < w; ++x) {
61 const float u = (float(x) + 0.5f) / float(w);
62 const float edge = 1.f - 2.f * std::fabs(u - 0.5f); // 0 at sides
63 const float alpha = std::pow(edge, 2.f) * std::pow(endFade, 2.5f);
64 const int i = (y * w + x) * 4;
65 rgba[i + 0] = 255;
66 rgba[i + 1] = 250;
67 rgba[i + 2] = 255;
68 rgba[i + 3] = uint8_t(std::min(255.f, alpha * 255.f));
69 }
70 }
71}

73// Soft round snowflake.
74void genSnowTexture(std::vector<uint8_t> &rgba) {
75 const int w = 16, h = 16;
76 rgba.resize(size_t(w) * h * 4);
77 const float cx = (float(w) - 1.f) * 0.5f;
78 const float cy = (float(h) - 1.f) * 0.5f;
79 for (int y = 0; y < h; ++y) {
80 for (int x = 0; x < w; ++x) {
81 const float dx = (float(x) - cx) / cx;
82 const float dy = (float(y) - cy) / cy;
83 const float r = std::sqrt(dx * dx + dy * dy);
84 float alpha = 1.f - r;
85 if (alpha < 0.f) alpha = 0.f;
86 alpha = std::pow(alpha, 2.2f);
87 const int i = (y * w + x) * 4;
88 rgba[i + 0] = 255;
89 rgba[i + 1] = 255;
90 rgba[i + 2] = 255;
91 rgba[i + 3] = uint8_t(std::min(255.f, alpha * 255.f));
92 }
93 }
94}

100// A field of billboarded quads. Each quad shares an anchor in inPos; inUV
101// encodes the local offset. `count` quads, `len` = quad height, `width` = width.
102graphics::Mesh *buildFieldMesh(graphics::Graphics *gfx, Lcg &rng, int count, float len,
103 float width) {
104 (void)len;
105 (void)width;
106 const int vertCount = count * 4;
107 const int idxCount = count * 6;
108 std::vector<float> pos(vertCount * 3);
109 std::vector<float> nrm(vertCount * 3, 0.f);
110 std::vector<float> uv(vertCount * 2);
111 std::vector<uint32_t> idx(idxCount);
112
113 for (int i = 0; i < count; ++i) {
114 const float ax = rng.range(-kBoxXZ, kBoxXZ);
115 const float ay = rng.range(0.f, kBoxY);
116 const float az = rng.range(-kBoxXZ, kBoxXZ);
117 const int v0 = i * 4;
118 const int b = v0 * 3, t = v0 * 2;
119
120 // bottom-left / top-left / top-right / bottom-right (uv-driven billboard).
121 pos[b + 0] = ax; pos[b + 1] = ay; pos[b + 2] = az;
122 pos[b + 3] = ax; pos[b + 4] = ay; pos[b + 5] = az;
123 pos[b + 6] = ax; pos[b + 7] = ay; pos[b + 8] = az;
124 pos[b + 9] = ax; pos[b + 10] = ay; pos[b + 11] = az;
125
126 // Pack per-particle variation in the otherwise unused normal.
127 const float lengthScale = rng.range(0.55f, 1.45f);
128 const float widthScale = rng.range(0.65f, 1.25f);
129 const float speedScale = rng.range(0.82f, 1.18f);
130 for (int j = 0; j < 4; ++j) {
131 nrm[b + j * 3 + 0] = lengthScale;
132 nrm[b + j * 3 + 1] = widthScale;
133 nrm[b + j * 3 + 2] = speedScale;
134 }
135
136 // uv.x = horizontal offset, uv.y = vertical offset (in [0,1]).
137 uv[t + 0] = -0.5f; uv[t + 1] = 0.f;
138 uv[t + 2] = -0.5f; uv[t + 3] = 1.f;
139 uv[t + 4] = 0.5f; uv[t + 5] = 1.f;
140 uv[t + 6] = 0.5f; uv[t + 7] = 0.f;
141
142 const uint32_t base = uint32_t(v0);
143 idx[i * 6 + 0] = base + 0;
144 idx[i * 6 + 1] = base + 1;
145 idx[i * 6 + 2] = base + 2;
146 idx[i * 6 + 3] = base + 0;
147 idx[i * 6 + 4] = base + 2;
148 idx[i * 6 + 5] = base + 3;
149 }
150
151 return gfx->newMeshFromArrays(pos.data(), nrm.data(), uv.data(), vertCount, idx.data(),
152 idxCount);
153}

155struct Pt {
156 float x = 0, y = 0, z = 0;

159void displacePath(std::vector<Pt> &path, Lcg &rng, float amount) {
160 std::vector<Pt> refined;
161 refined.reserve(path.size() * 2 - 1);
162 for (size_t i = 0; i + 1 < path.size(); ++i) {
163 const Pt &a = path[i];
164 const Pt &b = path[i + 1];
165 refined.push_back(a);
166 refined.push_back({(a.x + b.x) * 0.5f + rng.range(-amount, amount),
167 (a.y + b.y) * 0.5f,
168 (a.z + b.z) * 0.5f + rng.range(-amount, amount)});
169 }
170 refined.push_back(path.back());
171 path.swap(refined);
172}

174// Fractal midpoint displacement is the classic inexpensive lightning model.
175// Crossed tapered ribbons keep the channel visible from every camera angle.
176graphics::Mesh *buildBoltMesh(graphics::Graphics *gfx, Lcg &rng, Pt top, Pt ground) {
177 std::vector<Pt> pts{top, ground};
178 float displacement = 3.2f;
179 for (int level = 0; level < 5; ++level) {
180 displacePath(pts, rng, displacement);
181 displacement *= 0.48f;
182 }
183
184 std::vector<std::vector<Pt>> paths{pts};
185 const int branchStarts[] = {7, 12, 18, 23};
186 for (int start : branchStarts) {
187 if (start >= int(pts.size()) - 2) continue;
188 const Pt origin = pts[start];
189 const float side = (rng.unit() < 0.5f) ? -1.f : 1.f;
190 Pt tip{origin.x + side * rng.range(2.5f, 5.0f),
191 origin.y - rng.range(2.5f, 5.5f),
192 origin.z + rng.range(-3.5f, 3.5f)};
193 if (tip.y < 0.5f) tip.y = 0.5f;
194 std::vector<Pt> branch{origin, tip};
195 float branchDisp = 1.1f;
196 for (int level = 0; level < 3; ++level) {
197 displacePath(branch, rng, branchDisp);
198 branchDisp *= 0.45f;
199 }
200 paths.push_back(branch);
201 }
202
203 std::vector<Pt> all;
204 std::vector<float> allUv;
205 auto emitStrip = [&](const std::vector<Pt> &path, float wTop, float wBottom, int plane) {
206 const int n = int(path.size()) - 1;
207 for (int i = 0; i < n; ++i) {
208 Pt a = path[i], b = path[i + 1];
209 float sx = plane == 0 ? 1.f : 0.f;
210 float sz = plane == 0 ? 0.f : 1.f;
211 float w = wBottom + (wTop - wBottom) * (float(i) / float(n));
212 Pt p0 = {a.x + sx * w, a.y, a.z + sz * w};
213 Pt p1 = {a.x - sx * w, a.y, a.z - sz * w};
214 Pt p2 = {b.x - sx * w, b.y, b.z - sz * w};
215 Pt p3 = {b.x + sx * w, b.y, b.z + sz * w};
216 all.push_back(p0);
217 all.push_back(p1);
218 all.push_back(p2);
219 all.push_back(p3);
220 allUv.insert(allUv.end(), {-1.f, float(i) / n, 1.f, float(i) / n,
221 1.f, float(i + 1) / n, -1.f, float(i + 1) / n});
222 }
223 };
224
225 const float wMain = 0.20f;
226 for (size_t p = 0; p < paths.size(); ++p) {
227 const float scale = p == 0 ? 1.f : 0.55f;
228 emitStrip(paths[p], wMain * scale, wMain * 0.10f * scale, 0);
229 emitStrip(paths[p], wMain * scale, wMain * 0.10f * scale, 1);
230 }
231
232 const int vertCount = int(all.size());
233 const int idxCount = vertCount / 4 * 6;
234 std::vector<float> pos(vertCount * 3);
235 std::vector<float> nrm(vertCount * 3, 0.f);
236 std::vector<float> uv(vertCount * 2, 0.f);
237 std::vector<uint32_t> idx(idxCount);
238
239 for (int i = 0; i < vertCount; ++i) {
240 pos[i * 3 + 0] = all[i].x;
241 pos[i * 3 + 1] = all[i].y;
242 pos[i * 3 + 2] = all[i].z;
243 nrm[i * 3 + 1] = 1.f;
244 uv[i * 2 + 0] = allUv[i * 2 + 0];
245 uv[i * 2 + 1] = allUv[i * 2 + 1];
246 }
247 for (int q = 0; q < vertCount / 4; ++q) {
248 const uint32_t base = uint32_t(q * 4);
249 idx[q * 6 + 0] = base + 0;
250 idx[q * 6 + 1] = base + 1;
251 idx[q * 6 + 2] = base + 2;
252 idx[q * 6 + 3] = base + 0;
253 idx[q * 6 + 4] = base + 2;
254 idx[q * 6 + 5] = base + 3;
255 }
256
257 return gfx->newMeshFromArrays(pos.data(), nrm.data(), uv.data(), vertCount, idx.data(),
258 idxCount);
259}

349void Weather::init(graphics::Graphics *gfx) {
350 if (impl_->built) return;
351 impl_->built = true;
352 impl_->gfx = gfx;
353 if (!gfx) return;
354
355 // ---- textures ----
356 std::vector<uint8_t> rainRgba, snowRgba;
357 genRainTexture(rainRgba);
358 genSnowTexture(snowRgba);
359 graphics::Texture *rainTex = gfx->newTexture(8, 32, rainRgba.data(), true, true);
360 graphics::Texture *snowTex = gfx->newTexture(16, 16, snowRgba.data(), true, true);
361
362 // ---- shaders (precompiled SPIR-V; runtime GLSL compile is not available on Windows) ----
363 std::vector<uint32_t> wv(weather_vert_spv, weather_vert_spv + weather_vert_spv_count);
364 std::vector<uint32_t> wf(weather_frag_spv, weather_frag_spv + weather_frag_spv_count);
365 std::vector<uint32_t> bv(bolt_vert_spv, bolt_vert_spv + bolt_vert_spv_count);
366 std::vector<uint32_t> bf(bolt_frag_spv, bolt_frag_spv + bolt_frag_spv_count);
367 graphics::Shader *weatherVert = nullptr;
368 graphics::Shader *boltShader = nullptr;
369 if (gfx->getBackendName() == "webgpu") {
370 weatherVert = gfx->newMeshShaderFromWgsl(shaders::kWeatherVertWgsl,
371 shaders::kWeatherFragWgsl);
372 boltShader = gfx->newMeshShaderFromWgsl(shaders::kBoltVertWgsl,
373 shaders::kBoltFragWgsl);
374 } else {
375 weatherVert = gfx->newMeshShaderFromSpv(wv, wf);
376 boltShader = gfx->newMeshShaderFromSpv(bv, bf);
377 }
378 declareWeatherParams(weatherVert);
379 declareWeatherParams(boltShader);
380
381 // ---- rain ----
382 graphics::Mesh *rainMesh = buildFieldMesh(gfx, impl_->rng, kRainCount, 0.82f, 0.024f);
383 impl_->rain = graphics::Renderable3D::create();
384 impl_->rain->setMesh(rainMesh);
385 impl_->rain->setTexture(rainTex);
386 impl_->rainMat = gfx->newMaterial();
387 impl_->rainMat->setShadingModel("unlit");
388 impl_->rainMat->setReceiveLight(false);
389 impl_->rainMat->setReceiveShadow(false);
390 impl_->rainMat->setCastShadow(false);
391 impl_->rainMat->setTint(0.38f, 0.52f, 0.68f, 1.0f);
392 impl_->rainMat->setShader(weatherVert);
393 impl_->rain->setMaterial(impl_->rainMat);
394 impl_->rain->setVisible(false);
395
396 // ---- snow ----
397 graphics::Mesh *snowMesh = buildFieldMesh(gfx, impl_->rng, kSnowCount, 0.22f, 0.10f);
398 impl_->snow = graphics::Renderable3D::create();
399 impl_->snow->setMesh(snowMesh);
400 impl_->snow->setTexture(snowTex);
401 impl_->snowMat = gfx->newMaterial();
402 impl_->snowMat->setShadingModel("unlit");
403 impl_->snowMat->setReceiveLight(false);
404 impl_->snowMat->setReceiveShadow(false);
405 impl_->snowMat->setCastShadow(false);
406 impl_->snowMat->setTint(1.0f, 1.0f, 1.0f, 1.0f);
407 impl_->snowMat->setShader(weatherVert);
408 impl_->snow->setMaterial(impl_->snowMat);
409 impl_->snow->setVisible(false);
410
411 // ---- lightning bolts ----
412 for (int i = 0; i < kBoltCount; ++i) {
413 Pt top{impl_->rng.range(-4.f, 4.f), kBoxY, impl_->rng.range(-4.f, 4.f)};
414 Pt ground{impl_->rng.range(-3.f, 3.f), 0.f, impl_->rng.range(-3.f, 3.f)};
415 graphics::Mesh *m = buildBoltMesh(gfx, impl_->rng, top, ground);
416 graphics::Renderable3D *b = graphics::Renderable3D::create();
417 b->setMesh(m);
418 graphics::Material *mm = gfx->newMaterial();
419 mm->setShadingModel("unlit");
420 mm->setReceiveLight(false);
421 mm->setReceiveShadow(false);
422 mm->setCastShadow(false);
423 mm->setTint(1.0f, 1.0f, 1.0f, 1.0f);
424 mm->setShader(boltShader);
425 b->setMaterial(mm);
426 b->setVisible(false);
427 impl_->bolts.push_back(b);
428 impl_->boltMats.push_back(mm);
429 }
430
431 pushWeatherParams(impl_->rainMat, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f);
432 pushWeatherParams(impl_->snowMat, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f);
433}

507void Weather::update(float dt, graphics::Graphics *gfx) {
508 if (!impl_->built) init(gfx);
509 if (!impl_->gfx) return;
510 impl_->time += dt;
511
512 // Smooth intensity toward target.
513 const float target = impl_->intensity;
514 impl_->intensityCur += (target - impl_->intensityCur) * std::min(1.f, dt * 3.f);
515 if (std::fabs(target - impl_->intensityCur) < 0.002f) impl_->intensityCur = target;
516
517 // Wind vector in world space (dirDeg 0 -> +Z, 90 -> -X).
518 const float rad = impl_->windDirDeg * 0.0174532925f;
519 const float windX = -std::sin(rad) * impl_->windSpeed;
520 const float windZ = std::cos(rad) * impl_->windSpeed;
521
522 const float time = impl_->time;
523 const float fogR = impl_->fogR, fogG = impl_->fogG, fogB = impl_->fogB;
524 const float fogD = impl_->fogDensity;
525 const float intensity = impl_->intensityCur;
526
527 // Rain visible for drizzle/rain/storm.
528 const bool rainOn = impl_->preset >= 1 && impl_->preset <= 3;
529 impl_->rain->setVisible(rainOn && intensity > 0.01f);
530 if (rainOn) {
531 const float speed = 26.f;
532 pushWeatherParams(impl_->rainMat, time, windX, windZ, speed, 0.82f, 0.024f, intensity,
533 fogR, fogG, fogB, fogD, 0.f);
534 }
535
536 // Snow.
537 const bool snowOn = impl_->preset == 4;
538 impl_->snow->setVisible(snowOn && intensity > 0.01f);
539 if (snowOn) {
540 pushWeatherParams(impl_->snowMat, time, windX, windZ, 2.2f, 0.22f, 0.10f, intensity,
541 fogR, fogG, fogB, fogD, 0.f);
542 }
543
544 // Lightning during storm.
545 const bool storm = impl_->preset == 3;
546 if (storm && impl_->lightningEnabled) {
547 impl_->nextStrike -= dt;
548 if (impl_->nextStrike <= 0.f && impl_->activeBolt < 0) strike();
549 }
550
551 // Drive the active bolt's flash decay.
552 if (impl_->activeBolt >= 0) {
553 impl_->boltLife += dt;
554 impl_->flashTimer -= dt;
555 // Return-stroke profile: bright leader, dark gap, weaker second pulse.
556 float f;
557 if (impl_->boltLife < 0.055f) {
558 f = 1.f - impl_->boltLife * 4.f;
559 } else if (impl_->boltLife < 0.095f) {
560 f = 0.06f;
561 } else if (impl_->boltLife < 0.19f) {
562 f = 0.72f * (1.f - (impl_->boltLife - 0.095f) / 0.095f);
563 } else {
564 f = std::max(0.f, 0.22f * (1.f - (impl_->boltLife - 0.19f) / 0.31f));
565 }
566 impl_->flash = std::clamp(f, 0.f, 1.f);
567
568 const int bi = impl_->activeBolt;
569 impl_->bolts[bi]->setVisible(impl_->flash > 0.02f);
570 pushWeatherParams(impl_->boltMats[bi], time, windX, windZ, 0.f, 0.f, 0.f, 1.f, fogR, fogG,
571 fogB, fogD, impl_->flash);
572
573 if (impl_->boltLife > 0.50f) {
574 impl_->bolts[bi]->setVisible(false);
575 impl_->activeBolt = -1;
576 impl_->flash = 0.f;
577 }
578 } else {
579 impl_->flash *= std::max(0.f, 1.f - dt * 8.f);
580 }
581
582 // Push ambient/sky mood onto the graphics state each frame.
583 if (gfx && impl_->environmentEnabled) {
584 const float dark = 1.f - 0.65f * impl_->intensityCur;
585 const float skyFlash = impl_->flash * 0.55f;
586 gfx->setBackgroundColorRGBA(
587 std::min(1.f, impl_->skyR * dark + skyFlash * 0.68f),
588 std::min(1.f, impl_->skyG * dark + skyFlash * 0.78f),
589 std::min(1.f, impl_->skyB * dark + skyFlash), 1.f);
590 const float flashLight = impl_->flash * 1.35f;
591 gfx->setDirectionalLight(-0.4f, 0.75f, 0.5f,
592 impl_->sunIntensity * (1.f - 0.6f * impl_->intensityCur) + flashLight * 0.72f,
593 impl_->sunIntensity * (1.f - 0.6f * impl_->intensityCur) * 0.95f + flashLight * 0.84f,
594 impl_->sunIntensity * (1.f - 0.6f * impl_->intensityCur) * 0.88f + flashLight);
595 }
596}
