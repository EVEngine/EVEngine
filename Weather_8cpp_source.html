26// ---- tiny deterministic RNG (no dependency on <random>) ----
27struct Lcg {
28 uint32_t s = 0x9e3779b9u;
29 uint32_t next() {
30 s = s * 1664525u + 1013904223u;
31 return s;
32 }
33 float unit() { return float(next() % 10000u) / 9999.f; } // [0,1]
34 float range(float a, float b) { return a + (b - a) * unit(); }
35};

52// Vertical rain streak: bright core, alpha falloff toward the edges/ends.
53void genRainTexture(std::vector<uint8_t> &rgba) {
54 const int w = 8, h = 32;
55 rgba.resize(size_t(w) * h * 4);
56 for (int y = 0; y < h; ++y) {
57 const float v = (float(y) + 0.5f) / float(h); // 0 top -> 1 bottom
58 const float endFade = 1.f - 2.f * std::fabs(v - 0.5f); // 0 at ends, 1 center
59 for (int x = 0; x < w; ++x) {
60 const float u = (float(x) + 0.5f) / float(w);
61 const float edge = 1.f - 2.f * std::fabs(u - 0.5f); // 0 at sides
62 const float alpha = std::pow(edge, 2.f) * std::pow(endFade, 2.5f);
63 const int i = (y * w + x) * 4;
64 rgba[i + 0] = 255;
65 rgba[i + 1] = 250;
66 rgba[i + 2] = 255;
67 rgba[i + 3] = uint8_t(std::min(255.f, alpha * 255.f));
68 }
69 }
70}

72// Soft round snowflake.
73void genSnowTexture(std::vector<uint8_t> &rgba) {
74 const int w = 16, h = 16;
75 rgba.resize(size_t(w) * h * 4);
76 const float cx = (float(w) - 1.f) * 0.5f;
77 const float cy = (float(h) - 1.f) * 0.5f;
78 for (int y = 0; y < h; ++y) {
79 for (int x = 0; x < w; ++x) {
80 const float dx = (float(x) - cx) / cx;
81 const float dy = (float(y) - cy) / cy;
82 const float r = std::sqrt(dx * dx + dy * dy);
83 float alpha = 1.f - r;
84 if (alpha < 0.f) alpha = 0.f;
85 alpha = std::pow(alpha, 2.2f);
86 const int i = (y * w + x) * 4;
87 rgba[i + 0] = 255;
88 rgba[i + 1] = 255;
89 rgba[i + 2] = 255;
90 rgba[i + 3] = uint8_t(std::min(255.f, alpha * 255.f));
91 }
92 }
93}

99// A field of billboarded quads. Each quad shares an anchor in inPos; inUV
100// encodes the local offset. `count` quads, `len` = quad height, `width` = width.
101graphics::Mesh *buildFieldMesh(graphics::Graphics *gfx, Lcg &rng, int count, float len,
102 float width) {
103 (void)len;
104 (void)width;
105 const int vertCount = count * 4;
106 const int idxCount = count * 6;
107 std::vector<float> pos(vertCount * 3);
108 std::vector<float> nrm(vertCount * 3, 0.f);
109 std::vector<float> uv(vertCount * 2);
110 std::vector<uint32_t> idx(idxCount);
111
112 for (int i = 0; i < count; ++i) {
113 const float ax = rng.range(-kBoxXZ, kBoxXZ);
114 const float ay = rng.range(0.f, kBoxY);
115 const float az = rng.range(-kBoxXZ, kBoxXZ);
116 const int v0 = i * 4;
117 const int b = v0 * 3, t = v0 * 2;
118
119 // bottom-left / top-left / top-right / bottom-right (uv-driven billboard).
120 pos[b + 0] = ax; pos[b + 1] = ay; pos[b + 2] = az;
121 pos[b + 3] = ax; pos[b + 4] = ay; pos[b + 5] = az;
122 pos[b + 6] = ax; pos[b + 7] = ay; pos[b + 8] = az;
123 pos[b + 9] = ax; pos[b + 10] = ay; pos[b + 11] = az;
124
125 nrm[b + 1] = nrm[b + 4] = nrm[b + 7] = nrm[b + 10] = 1.f;
126
127 // uv.x = horizontal offset, uv.y = vertical offset (in [0,1]).
128 uv[t + 0] = -0.5f; uv[t + 1] = 0.f;
129 uv[t + 2] = -0.5f; uv[t + 3] = 1.f;
130 uv[t + 4] = 0.5f; uv[t + 5] = 1.f;
131 uv[t + 6] = 0.5f; uv[t + 7] = 0.f;
132
133 const uint32_t base = uint32_t(v0);
134 idx[i * 6 + 0] = base + 0;
135 idx[i * 6 + 1] = base + 1;
136 idx[i * 6 + 2] = base + 2;
137 idx[i * 6 + 3] = base + 0;
138 idx[i * 6 + 4] = base + 2;
139 idx[i * 6 + 5] = base + 3;
140 }
141
142 return gfx->newMeshFromArrays(pos.data(), nrm.data(), uv.data(), vertCount, idx.data(),
143 idxCount);
144}

150// Build a jagged lightning bolt as a tapered quad strip plus short branches.
151graphics::Mesh *buildBoltMesh(graphics::Graphics *gfx, Lcg &rng, Pt top, Pt ground) {
152 std::vector<Pt> pts;
153 // Main trunk with recursive midpoint displacement.
154 const int segs = 24;
155 pts.push_back(top);
156 float y = top.y;
157 float stepY = (top.y - ground.y) / float(segs);
158 for (int i = 1; i < segs; ++i) {
159 y -= stepY;
160 float f = float(i) / float(segs);
161 float jitter = (1.f - f) * 0.9f; // more violent near the top
162 Pt p;
163 p.x = ground.x + (top.x - ground.x) * (1.f - f) + rng.range(-jitter, jitter) * 1.2f;
164 p.y = y;
165 p.z = ground.z + (top.z - ground.z) * (1.f - f) + rng.range(-jitter, jitter) * 1.2f;
166 pts.push_back(p);
167 }
168 pts.push_back(ground);
169
170 // One branch from the upper third.
171 std::vector<Pt> branch;
172 int bStart = int(float(segs) * 0.35f);
173 branch.push_back(pts[bStart]);
174 Pt bp = pts[bStart];
175 for (int i = 1; i <= 8; ++i) {
176 bp.x += rng.range(-0.5f, 0.5f);
177 bp.y -= (pts[bStart].y - ground.y) * 0.12f;
178 bp.z += rng.range(-0.5f, 0.5f);
179 branch.push_back(bp);
180 }
181
182 std::vector<Pt> all;
183 auto emitStrip = [&](const std::vector<Pt> &path, float wTop, float wBottom) {
184 const int n = int(path.size()) - 1;
185 for (int i = 0; i < n; ++i) {
186 Pt a = path[i], b = path[i + 1];
187 // Side vector (perpendicular to the segment in the XZ plane).
188 float dx = b.x - a.x, dz = b.z - a.z;
189 float len = std::sqrt(dx * dx + dz * dz) + 1e-4f;
190 float sx = -dz / len, sz = dx / len;
191 float w = wBottom + (wTop - wBottom) * (float(i) / float(n));
192 Pt p0 = {a.x + sx * w, a.y, a.z + sz * w};
193 Pt p1 = {a.x - sx * w, a.y, a.z - sz * w};
194 Pt p2 = {b.x - sx * w, b.y, b.z - sz * w};
195 Pt p3 = {b.x + sx * w, b.y, b.z + sz * w};
196 all.push_back(p0);
197 all.push_back(p1);
198 all.push_back(p2);
199 all.push_back(p3);
200 }
201 };
202
203 const float wMain = 0.07f;
204 emitStrip(pts, wMain, wMain * 0.15f);
205 emitStrip(branch, wMain * 0.8f, wMain * 0.15f);
206
207 const int vertCount = int(all.size());
208 const int idxCount = vertCount / 4 * 6;
209 std::vector<float> pos(vertCount * 3);
210 std::vector<float> nrm(vertCount * 3, 0.f);
211 std::vector<float> uv(vertCount * 2, 0.f);
212 std::vector<uint32_t> idx(idxCount);
213
214 for (int i = 0; i < vertCount; ++i) {
215 pos[i * 3 + 0] = all[i].x;
216 pos[i * 3 + 1] = all[i].y;
217 pos[i * 3 + 2] = all[i].z;
218 nrm[i * 3 + 1] = 1.f;
219 uv[i * 2 + 1] = float(i / 4) * 0.25f;
220 }
221 for (int q = 0; q < vertCount / 4; ++q) {
222 const uint32_t base = uint32_t(q * 4);
223 idx[q * 6 + 0] = base + 0;
224 idx[q * 6 + 1] = base + 1;
225 idx[q * 6 + 2] = base + 2;
226 idx[q * 6 + 3] = base + 0;
227 idx[q * 6 + 4] = base + 2;
228 idx[q * 6 + 5] = base + 3;
229 }
230
231 return gfx->newMeshFromArrays(pos.data(), nrm.data(), uv.data(), vertCount, idx.data(),
232 idxCount);
233}

322void Weather::init(graphics::Graphics *gfx) {
323 if (impl_->built) return;
324 impl_->built = true;
325 impl_->gfx = gfx;
326 if (!gfx) return;
327
328 // ---- textures ----
329 std::vector<uint8_t> rainRgba, snowRgba;
330 genRainTexture(rainRgba);
331 genSnowTexture(snowRgba);
332 graphics::Texture *rainTex = gfx->newTexture(8, 32, rainRgba.data(), true, true);
333 graphics::Texture *snowTex = gfx->newTexture(16, 16, snowRgba.data(), true, true);
334
335 // ---- shaders (precompiled SPIR-V; runtime GLSL compile is not available on Windows) ----
336 std::vector<uint32_t> wv(weather_vert_spv, weather_vert_spv + weather_vert_spv_count);
337 std::vector<uint32_t> wf(weather_frag_spv, weather_frag_spv + weather_frag_spv_count);
338 std::vector<uint32_t> bv(bolt_vert_spv, bolt_vert_spv + bolt_vert_spv_count);
339 std::vector<uint32_t> bf(bolt_frag_spv, bolt_frag_spv + bolt_frag_spv_count);
340 graphics::Shader *weatherVert = gfx->newMeshShaderFromSpv(wv, wf);
341 graphics::Shader *boltShader = gfx->newMeshShaderFromSpv(bv, bf);
342 declareWeatherParams(weatherVert);
343 declareWeatherParams(boltShader);
344
345 // ---- rain ----
346 graphics::Mesh *rainMesh = buildFieldMesh(gfx, impl_->rng, kRainCount, 1.4f, 0.07f);
347 impl_->rain = graphics::Renderable3D::create();
348 impl_->rain->setMesh(rainMesh);
349 impl_->rain->setTexture(rainTex);
350 impl_->rainMat = gfx->newMaterial();
351 impl_->rainMat->setShadingModel("unlit");
352 impl_->rainMat->setReceiveLight(false);
353 impl_->rainMat->setReceiveShadow(false);
354 impl_->rainMat->setCastShadow(false);
355 impl_->rainMat->setTint(0.8f, 0.9f, 1.0f, 1.0f);
356 impl_->rainMat->setShader(weatherVert);
357 impl_->rain->setMaterial(impl_->rainMat);
358 impl_->rain->setVisible(false);
359
360 // ---- snow ----
361 graphics::Mesh *snowMesh = buildFieldMesh(gfx, impl_->rng, kSnowCount, 0.22f, 0.10f);
362 impl_->snow = graphics::Renderable3D::create();
363 impl_->snow->setMesh(snowMesh);
364 impl_->snow->setTexture(snowTex);
365 impl_->snowMat = gfx->newMaterial();
366 impl_->snowMat->setShadingModel("unlit");
367 impl_->snowMat->setReceiveLight(false);
368 impl_->snowMat->setReceiveShadow(false);
369 impl_->snowMat->setCastShadow(false);
370 impl_->snowMat->setTint(1.0f, 1.0f, 1.0f, 1.0f);
371 impl_->snowMat->setShader(weatherVert);
372 impl_->snow->setMaterial(impl_->snowMat);
373 impl_->snow->setVisible(false);
374
375 // ---- lightning bolts ----
376 for (int i = 0; i < kBoltCount; ++i) {
377 Pt top{impl_->rng.range(-4.f, 4.f), kBoxY, impl_->rng.range(-4.f, 4.f)};
378 Pt ground{impl_->rng.range(-3.f, 3.f), 0.f, impl_->rng.range(-3.f, 3.f)};
379 graphics::Mesh *m = buildBoltMesh(gfx, impl_->rng, top, ground);

382 graphics::Material *mm = gfx->newMaterial();

384 mm->setReceiveLight(false);

392 impl_->boltMats.push_back(mm);
393 }
394
395 pushWeatherParams(impl_->rainMat, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f);
396 pushWeatherParams(impl_->snowMat, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f, 0.f);
397}

468void Weather::update(float dt, graphics::Graphics *gfx) {
469 if (!impl_->built) init(gfx);
470 if (!impl_->gfx) return;
471 impl_->time += dt;
472
473 // Smooth intensity toward target.
474 const float target = impl_->intensity;
475 impl_->intensityCur += (target - impl_->intensityCur) * std::min(1.f, dt * 3.f);
476 if (std::fabs(target - impl_->intensityCur) < 0.002f) impl_->intensityCur = target;
477
478 // Wind vector in world space (dirDeg 0 -> +Z, 90 -> -X).
479 const float rad = impl_->windDirDeg * 0.0174532925f;
480 const float windX = -std::sin(rad) * impl_->windSpeed;
481 const float windZ = std::cos(rad) * impl_->windSpeed;
482
483 const float time = impl_->time;
484 const float fogR = impl_->fogR, fogG = impl_->fogG, fogB = impl_->fogB;
485 const float fogD = impl_->fogDensity;
486 const float intensity = impl_->intensityCur;
487
488 // Rain visible for drizzle/rain/storm.
489 const bool rainOn = impl_->preset >= 1 && impl_->preset <= 3;
490 impl_->rain->setVisible(rainOn && intensity > 0.01f);
491 if (rainOn) {
492 const float speed = 26.f;
493 pushWeatherParams(impl_->rainMat, time, windX, windZ, speed, 1.4f, 0.07f, intensity,
494 fogR, fogG, fogB, fogD, 0.f);
495 }
496
497 // Snow.
498 const bool snowOn = impl_->preset == 4;
499 impl_->snow->setVisible(snowOn && intensity > 0.01f);
500 if (snowOn) {
501 pushWeatherParams(impl_->snowMat, time, windX, windZ, 2.2f, 0.22f, 0.10f, intensity,
502 fogR, fogG, fogB, fogD, 0.f);
503 }
504
505 // Lightning during storm.
506 const bool storm = impl_->preset == 3;
507 if (storm && impl_->lightningEnabled) {
508 impl_->nextStrike -= dt;
509 if (impl_->nextStrike <= 0.f && impl_->activeBolt < 0) strike();
510 }
511
512 // Drive the active bolt's flash decay.
513 if (impl_->activeBolt >= 0) {
514 impl_->boltLife += dt;
515 impl_->flashTimer -= dt;
516 // Flicker: rapid on/off at the start, then decay.
517 float f;
518 if (impl_->boltLife < 0.05f) {
519 f = 1.f;
520 } else if (impl_->boltLife < 0.20f) {
521 // Fast flicker.
522 f = 0.25f + 0.75f * std::fabs(std::sin(impl_->boltLife * 120.f));
523 } else {
524 f = std::max(0.f, 1.f - (impl_->boltLife - 0.20f) / 0.15f);
525 }
526 impl_->flash = std::clamp(f, 0.f, 1.f);
527
528 const int bi = impl_->activeBolt;
529 impl_->bolts[bi]->setVisible(impl_->flash > 0.02f);
530 pushWeatherParams(impl_->boltMats[bi], time, windX, windZ, 0.f, 0.f, 0.f, 1.f, fogR, fogG,
531 fogB, fogD, impl_->flash);
532
533 if (impl_->boltLife > 0.35f) {
534 impl_->bolts[bi]->setVisible(false);
535 impl_->activeBolt = -1;
536 impl_->flash = 0.f;
537 }
538 } else {
539 impl_->flash *= std::max(0.f, 1.f - dt * 8.f);
540 }
541
542 // Push ambient/sky mood onto the graphics state each frame.
543 if (gfx) {
544 const float dark = 1.f - 0.65f * impl_->intensityCur;
545 gfx->setBackgroundColorRGBA(impl_->skyR * dark, impl_->skyG * dark,
546 impl_->skyB * dark, 1.f);
547 gfx->setDirectionalLight(-0.4f, 0.75f, 0.5f,
548 impl_->sunIntensity * (1.f - 0.6f * impl_->intensityCur) * 1.0f,
549 impl_->sunIntensity * (1.f - 0.6f * impl_->intensityCur) * 0.95f,
550 impl_->sunIntensity * (1.f - 0.6f * impl_->intensityCur) * 0.88f);
551 }
552}
