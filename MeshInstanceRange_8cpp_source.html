26bool meshInstanceRangeVisible(const MeshInstanceRange& range, const glm::mat4& model, const glm::mat4& vp,
27 const glm::vec3& eye) {
28 if (!range.count) return false;
29 glm::vec3 localCenter, half;
30 for (int i = 0; i < 3; ++i) {
31 localCenter[i] = range.minimum[i] * .5f + range.maximum[i] * .5f;
32 half[i] = range.maximum[i] * .5f - range.minimum[i] * .5f;
33 }
34 const glm::vec3 center(model * glm::vec4(localCenter, 1));
35 const glm::vec3 axes[]{glm::vec3(model[0]) * half.x, glm::vec3(model[1]) * half.y, glm::vec3(model[2]) * half.z};
36 const auto row = [&](int i) { return glm::vec4(vp[0][i], vp[1][i], vp[2][i], vp[3][i]); };
37 const glm::vec4 planes[]{row(3) + row(0), row(3) - row(0), row(3) + row(1),
38 row(3) - row(1), row(2), row(3) - row(2)};
39 for (const auto& p : planes) {
40 const glm::vec3 n(p);
41 const float radius =
42 std::abs(glm::dot(n, axes[0])) + std::abs(glm::dot(n, axes[1])) + std::abs(glm::dot(n, axes[2]));
43 if (glm::dot(n, center) + p.w < -radius) return false;
44 }
45 if (range.maximumHorizontalDistance > 0) {
46 const glm::vec3 extent = glm::abs(axes[0]) + glm::abs(axes[1]) + glm::abs(axes[2]);
47 const double limit = double(range.maximumHorizontalDistance) + std::max(extent.x, extent.z);
48 const double dx = double(center.x) - eye.x, dz = double(center.z) - eye.z;
49 if (dx * dx + dz * dz > limit * limit) return false;
50 }
51 return true;
52}

133Result<void> Mesh::validateTangentFrame(std::span<const float> tangents, std::span<const float> bitangents) const {
134 auto invalid = [] {
135 return Result<void>::failure(
136 Diagnostic::error(DiagnosticCode::InvalidArgument,
137 "tangent frame must contain finite independent XYZ vectors for every mesh vertex"));
138 };
139 if (tangents.empty() && bitangents.empty()) {
140 return Result<void>::success();
141 }
142 if (gpuVertexCount <= 0 || tangents.size() != size_t(gpuVertexCount) * 3 || bitangents.size() != tangents.size())
143 return invalid();
144 for (size_t i = 0; i < tangents.size(); i += 3) {
145 double tt = 0, bb = 0, tb = 0;
146 for (unsigned c = 0; c < 3; ++c) {
147 const double t = tangents[i + c], b = bitangents[i + c];
148 if (!std::isfinite(t) || !std::isfinite(b)) return invalid();
149 tt += t * t;
150 bb += b * b;
151 tb += t * b;
152 }
153 if (tt <= 1e-20 || bb <= 1e-20 || tt * bb - tb * tb <= tt * bb * 1e-12) return invalid();
154 }
155 return Result<void>::success();
156}
