36Result<ForceBatch> prepareFields(W& world, std::span<const FieldDefinition> fields,
37 std::span<const FieldTarget> targets) {
38 std::set<PhysicsBodyHandle> seen;
39 std::vector<ForceLoad> loads;
40 loads.reserve(targets.size());
41 for (const auto& target : targets) {
42 auto resolved = target.body.resolve(world);
43 if (!resolved) return Result<ForceBatch>::failure(resolved.status());
44 auto& body = *resolved.value();
45 if (!body.isActive() || body.getType() != "dynamic" || !seen.insert(target.body.body).second)
46 return invalid<ForceBatch>();
47 const auto point = center(body);
48 const double mass = body.getMass();
49 if (!isFinite(point) || !isPositive(mass)) return invalid<ForceBatch>();
50 ForceVector sum{};
51 for (const auto& field : fields) {
52 if (!(target.categories & field.filter.includeAny) || (target.categories & field.filter.excludeAny))
53 continue;
54 auto value = sample(field, point, mass);
55 if (!value) return Result<ForceBatch>::failure(value.status());
56 sum = add(sum, value.value());
57 if (!isFinite(sum)) return invalid<ForceBatch>();
58 }
59 // Keep zero loads in the batch so their target identities are also revalidated at commit.
60 loads.push_back({target.body, sum, point, {}});
61 }
62 return ForceBatch::prepare(world, loads);
63}
