240Value captureJoint3DParameters(b3JointId id) {
241 const auto definition = fields(b3Joint_GetType(id));
242 Value::Object out;
243 for (const auto& f : definition.flags) out.emplace(f.name, Value(f.read(id)));
244 for (const auto& f : definition.numbers) {
245 Value::Array values;
246 for (float x : f.read(id)) values.emplace_back(x);
247 out.emplace(f.name, Value(std::move(values)));
248 }
249 return Value(std::move(out));
250}
251Result<void> restoreJoint3DParameters(b3JointId id, const Value& value) {
252 const auto* object = value.getIf<Value::Object>();
253 const auto definition = fields(b3Joint_GetType(id));
254 if (!object || object->size() != definition.flags.size() + definition.numbers.size()) return invalid("fields");
255 std::vector<bool> flags;
256 std::vector<Numbers> numbers;
257 for (const auto& f : definition.flags) {
258 auto parsed = readBool(*object, f.name);
259 if (!parsed) return Result<void>::failure(parsed.status());
260 flags.push_back(parsed.value());
261 }
262 for (const auto& f : definition.numbers) {
263 auto parsed = readFloatArray(*object, f.name);
264 if (!parsed) return Result<void>::failure(parsed.status());
265 const auto& n = parsed.value();
266 if (n.size() != f.count) return invalid(f.name);
267 for (float x : n)
268 if (x < f.minimum || x > f.maximum) return invalid(f.name);
269 if (f.ordered && n[0] > n[1]) return invalid(f.name);
270 if (f.quaternion && !b3IsNormalizedQuat({{n[0], n[1], n[2]}, n[3]})) return invalid(f.name);
271 numbers.push_back(std::move(parsed).takeValue());
272 }
273 // No setter runs until every field has passed validation.
274 for (std::size_t i = 0; i < definition.numbers.size(); ++i) definition.numbers[i].write(id, numbers[i]);
275 for (std::size_t i = 0; i < definition.flags.size(); ++i) definition.flags[i].write(id, flags[i]);
276 return Result<void>::success();
277}
