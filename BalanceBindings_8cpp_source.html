84 return script::projectResult(vm, self->configure(problem.value()));
85 });
86 cls.addFunc("run", [vm](BalanceSearch* self, ssq::Table evaluator) {
87 Result<Value> result = [&]() -> Result<Value> {
88 if (!self->problem_)
89 return Result<Value>::failure(
90 Diagnostic::error(DiagnosticCode::InvalidArgument, "Configure the balance search before run"));
91 ScriptEvaluator adapter(evaluator, *self->problem_);
92 auto report = self->run(adapter);
93 if (!report) return Result<Value>::failure(report.status());
94 return Result<Value>::success(encodeBalanceReport(report.value()));
95 }();
96 if (result) {

98 auto checked = script::pushValue(vm, result.value());
99 if (!checked)
100 return script::projectResult(vm, Result<Value>::failure(checked.status()), [](Value v) { return v; });
101 }
102 return script::projectResult(vm, std::move(result), [](Value v) { return v; });
103 });
