37float ReflectionProbeVisualizer::coordinate(int corner, int component) const {
38 if (!probe_ || corner < 0 || corner >= 8 || component < 0 || component >= 3) return 0.f;
39 const float center[3] = {probe_->getCenterX(), probe_->getCenterY(), probe_->getCenterZ()};
40 const float extent[3] = {probe_->getInfluenceExtentX(), probe_->getInfluenceExtentY(),
41 probe_->getInfluenceExtentZ()};
42 return center[component] + ((corner & (1 << component)) ? extent[component] : -extent[component]);
43}
