158// The push-constant layout index of a param equals its position in the param
159// array, so bind in array order.
160void bindParamArray(graphics::Shader* shader, const std::string& style, bool mesh) {
161 if (!shader) throw eve::Exception("bindParamArray: null shader");
162 int n = mesh ? meshParamCount(style) : postParamCount(style);
163 for (int i = 0; i < n; ++i) {
164 const StyleParameterDesc* d = mesh ? meshParamAt(style, i) : postParamAt(style, i);
165 if (!d) break;
166 shader->declareFloat(d->id);
167 shader->sendFloat(d->id, d->defaultValue);
168 }
169}

234graphics::Shader* createEffectPostShader(graphics::Graphics* gfx, const std::string& style) {
235 if (!gfx) throw eve::Exception("createEffectPostShader: null graphics");
236 std::vector<uint32_t> frag;
237 if (style == "vignette")
238 frag = copySpv(vignette_post_frag_spv, vignette_post_frag_spv_count);
239 else if (style == "chromatic")
240 frag = copySpv(chromatic_post_frag_spv, chromatic_post_frag_spv_count);
241 else if (style == "grain")
242 frag = copySpv(grain_post_frag_spv, grain_post_frag_spv_count);
243 else
244 throw eve::Exception("createEffectPostShader: not an effect post style '%s'", style.c_str());
245
246 graphics::Shader* sh = gfx->newShaderFromSpv({}, frag);
247 if (!sh || !sh->gpuHandle) throw eve::Exception("createEffectPostShader: failed to create '%s'", style.c_str());
248 bindEffectPostUniforms(sh, style);
249 return sh;
250}

252graphics::Shader* createEffectMeshShader(graphics::Graphics* gfx, const std::string& style) {
253 if (!gfx) throw eve::Exception("createEffectMeshShader: null graphics");

259 else if (style == "hologram")

261 else if (style == "snow")
262 frag = copySpv(snow_mesh_frag_spv, snow_mesh_frag_spv_count);
263 else if (style == "slash")
264 frag = copySpv(slash_mesh_frag_spv, slash_mesh_frag_spv_count);
265 else if (style == "ember")
266 frag = copySpv(ember_mesh_frag_spv, ember_mesh_frag_spv_count);
267 else if (style == "aura")
268 frag = copySpv(aura_mesh_frag_spv, aura_mesh_frag_spv_count);
269 else
270 throw eve::Exception("createEffectMeshShader: not an effect mesh style '%s'", style.c_str());
271
272 auto vert = copySpv(mesh3d_toon_vert_spv, mesh3d_toon_vert_spv_count);
273 graphics::Shader* sh = gfx->newMeshShaderFromSpv(vert, frag);
274 if (!sh || !sh->gpuHandle) throw eve::Exception("createEffectMeshShader: failed to create '%s'", style.c_str());
275 bindEffectMeshUniforms(sh, style);

277}
