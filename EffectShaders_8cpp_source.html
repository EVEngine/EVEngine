252// The push-constant layout index of a param equals its position in the param
253// array, so bind in array order.
254void bindParamArray(graphics::Shader* shader, const std::string& style, bool mesh) {
255 if (!shader) throw eve::Exception("bindParamArray: null shader");
256 int n = mesh ? meshParamCount(style) : postParamCount(style);
257 for (int i = 0; i < n; ++i) {
258 const StyleParameterDesc* d = mesh ? meshParamAt(style, i) : postParamAt(style, i);
259 if (!d) break;
260 shader->declareFloat(d->id);
261 shader->sendFloat(d->id, d->defaultValue);
262 }
263}

312void bindEffectPostUniforms(graphics::Shader* shader, const std::string& style) {
313 bindParamArray(shader, style, /*mesh=*/false);
314 // Chromatic / wave glitch need framebuffer size; StylePass::uploadScreenUniforms
315 // refreshes these each apply.
316 if (style == "chromatic" || style == "glitchWave" || style == "neonEdge") {
317 shader->declareFloat("screenW");
318 shader->declareFloat("screenH");
319 shader->sendFloat("screenW", 256.f);
320 shader->sendFloat("screenH", 256.f);
321 }
322 if (style == "kuwahara" || style == "kuwaharaAniso" || style == "dog" || style == "sharpen") {
323 shader->declareFloat("texelW");
324 shader->declareFloat("texelH");
325 shader->sendFloat("texelW", 1.f / 256.f);
326 shader->sendFloat("texelH", 1.f / 256.f);
327 }
328}

334graphics::Shader* createEffectPostShader(graphics::Graphics* gfx, const std::string& style) {
335 if (!gfx) throw eve::Exception("createEffectPostShader: null graphics");
336 std::vector<uint32_t> frag;
337 if (style == "vignette")
338 frag = copySpv(vignette_post_frag_spv, vignette_post_frag_spv_count);
339 else if (style == "chromatic")
340 frag = copySpv(chromatic_post_frag_spv, chromatic_post_frag_spv_count);
341 else if (style == "grain")
342 frag = copySpv(grain_post_frag_spv, grain_post_frag_spv_count);
343 else if (style == "glitchAnalog")
344 frag = copySpv(glitch_analog_post_frag_spv, glitch_analog_post_frag_spv_count);
345 else if (style == "glitchBlock")
346 frag = copySpv(glitch_block_post_frag_spv, glitch_block_post_frag_spv_count);
347 else if (style == "glitchWave")
348 frag = copySpv(glitch_wave_post_frag_spv, glitch_wave_post_frag_spv_count);
349 else if (style == "kuwahara")
350 frag = copySpv(kuwahara_post_frag_spv, kuwahara_post_frag_spv_count);
351 else if (style == "kuwaharaAniso")
352 frag = copySpv(kuwahara_aniso_post_frag_spv, kuwahara_aniso_post_frag_spv_count);
353 else if (style == "dog")
354 frag = copySpv(dog_post_frag_spv, dog_post_frag_spv_count);
355 else if (style == "sharpen")
356 frag = copySpv(sharpen_post_frag_spv, sharpen_post_frag_spv_count);
357 else if (style == "neonEdge")
358 frag = copySpv(neon_edge_post_frag_spv, neon_edge_post_frag_spv_count);
359 else
360 throw eve::Exception("createEffectPostShader: not an effect post style '%s'", style.c_str());
361
362 graphics::Shader* sh = gfx->newShaderFromSpv({}, frag);
363 if (!sh || !sh->gpuHandle) throw eve::Exception("createEffectPostShader: failed to create '%s'", style.c_str());
364 bindEffectPostUniforms(sh, style);
