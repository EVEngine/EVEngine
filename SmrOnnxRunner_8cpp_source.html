54Result<std::vector<float>> SmrOnnxRunner::run(const SmrFeatureBatch& features) const {
55 if (!model_) {
56 return Result<std::vector<float>>::failure(
57 Diagnostic::error(DiagnosticCode::PreconditionViolation, "SmrOnnxRunner: no model"));
58 }
59 if (features.frames <= 0 || features.joints <= 0) {
60 return Result<std::vector<float>>::failure(
61 Diagnostic::error(DiagnosticCode::InvalidArgument, "SmrOnnxRunner: empty features"));
62 }
63 const int64_t T = features.frames;
64 const int64_t J = features.joints;
65 const int64_t S = std::max(features.sensors, 1);
66 const int64_t P = std::max(features.pairs, 1);
67
68 std::vector<tensor::OnnxNamedTensor> feeds;
69 feeds.push_back(makeF32("source_rot6d", {1, T, J, 6}, features.sourceRot6d));
70 feeds.push_back(makeF32("source_geom", {1, S, 7}, features.sourceGeom));
71 feeds.push_back(makeF32("target_geom", {1, S, 7}, features.targetGeom));
72 feeds.push_back(makeF32("source_dmi", {1, T, P, 10}, features.sourceDmi));
73
74 const std::string outName = "target_rot6d";
75 auto ran = model_->run(feeds, std::span<const std::string>(&outName, 1));
76 if (!ran.ok()) return Result<std::vector<float>>::failure(ran.status());
77 if (ran.value().empty()) {
78 return Result<std::vector<float>>::failure(
79 Diagnostic::error(DiagnosticCode::Failed, "SmrOnnxRunner: missing output"));
80 }
81 const auto& bytes = ran.value()[0].tensor.bytes;
82 std::vector<float> out(bytes.size() / sizeof(float));
83 if (!out.empty()) std::memcpy(out.data(), bytes.data(), out.size() * sizeof(float));
84 return Result<std::vector<float>>::success(std::move(out));
85}
