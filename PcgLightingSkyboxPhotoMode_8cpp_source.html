37Result<void> PcgLightingSkyboxPhotoMode::applyPhotoModeField(
38 const PhotoModeAssignment &assignment) {
39 const auto fail = [&](DiagnosticCode code, const char *message) {
40 return Result<void>::failure(Diagnostic::error(
41 code, message, assignment.field, {}, "daynight.pcgLightingSkybox"));
42 };
43 if (!target_)
44 return fail(DiagnosticCode::Unsupported, "day-night target is unavailable");
45
46 PcgSkyboxState candidate = state_;
47 if (assignment.field == "m_skyboxOverride") {
48 const auto *value = std::get_if<bool>(&assignment.value);
49 if (!value)
50 return fail(DiagnosticCode::InvalidArgument,
51 "skybox override requires a bool");
52 candidate.enabled = *value;
53 } else if (assignment.field == "m_skyboxTint") {
54 const auto *value = std::get_if<PhotoModeColor>(&assignment.value);
55 if (!value || !std::isfinite(value->r) || !std::isfinite(value->g) ||
56 !std::isfinite(value->b) || value->r < 0.f || value->g < 0.f ||
57 value->b < 0.f)
58 return fail(DiagnosticCode::InvalidArgument,
59 "skybox tint requires finite non-negative RGB channels");
60 candidate.tintRed = value->r;
61 candidate.tintGreen = value->g;
62 candidate.tintBlue = value->b;
63 } else {
64 const auto *value = std::get_if<float>(&assignment.value);
65 if (!value || !std::isfinite(*value))
66 return fail(DiagnosticCode::InvalidArgument,
67 "skybox field requires a finite float");
68 if (assignment.field == "m_skyboxRotation")
69 candidate.rotationDegrees = *value;
70 else if (assignment.field == "m_skyboxExposure")
71 candidate.exposure = *value;
72 else
73 return fail(DiagnosticCode::InvalidArgument,
74 "unsupported Pcg skybox field");
75 }
