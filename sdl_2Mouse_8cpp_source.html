120void Mouse::setPosition(double x, double y)
121{
122 // Avoid `auto window = getModInst(window, ...)` — local name shadows namespace `window`.
123 auto *winMod = eve::ModuleManager::getInstance<eve::window::Window>("Window");
124
125 SDL_Window *handle = nullptr;
126 if (winMod)
127 handle = (SDL_Window *) winMod->getHandle();
128
129 DPIToWindowCoords(&x, &y);
130 if (handle) {
131 SDL_WarpMouseInWindow(handle, (int) x, (int) y);
132 // SDL_WarpMouse doesn't directly update SDL's internal mouse state on
133 // Linux/Windows; pump so the next getPosition sees the new coords.
134 SDL_PumpEvents();
135 }

153bool Mouse::isDown(const std::vector<int> &buttons) const
154{
155 Uint32 buttonstate = SDL_GetMouseState(nullptr, nullptr);
156
157 for (int button : buttons)
158 {
159 if (button <= 0)
160 continue;
161
162 // We use button index 2 to represent the right mouse button, but SDL
163 // uses 2 to represent the middle mouse button.
164 switch (button)
165 {
166 case 2:
167 button = SDL_BUTTON_RIGHT;
168 break;
169 case 3:
170 button = SDL_BUTTON_MIDDLE;
171 break;
172 }
173
174 if (buttonstate & SDL_BUTTON(button))
175 return true;
176 }
177
178 return false;
179}
