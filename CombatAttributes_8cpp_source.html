20Result<void> CombatAttributeDefinition::validate() const {
21 if (!tags::isValidGameplayTagName(tag))
22 return failure<void>(DiagnosticCode::InvalidArgument, "Combat attribute tag is invalid", "tag");
23 if (!std::isfinite(initialValue) || !std::isfinite(minimumValue) || !std::isfinite(maximumValue) ||
24 !std::isfinite(regenerationPerSecond))
25 return failure<void>(DiagnosticCode::InvalidArgument, "Combat attribute values must be finite", tag);
26 if (minimumValue > maximumValue)
27 return failure<void>(DiagnosticCode::InvalidArgument, "Combat attribute minimum exceeds maximum", tag);
28 if (initialValue < minimumValue || initialValue > maximumValue)
29 return failure<void>(DiagnosticCode::InvalidArgument, "Combat attribute initial value is outside bounds", tag);
30 if (regenerationPerSecond < 0.0)
31 return failure<void>(DiagnosticCode::InvalidArgument, "Combat attribute regeneration must be non-negative",
32 tag);
33 return Result<void>::success();
34}

36Result<void> CombatAttributeRuntime::registerAttribute(CombatAttributeDefinition definition) {
37 auto valid = definition.validate();
38 if (!valid) return valid;
39 if (definitions_.contains(definition.tag))

54Result<std::vector<CombatAttributeEvent>> CombatAttributeRuntime::commit(
55 const CombatAttributeDefinition& definition, double newValue) {
56 if (!std::isfinite(newValue))
57 return failure<std::vector<CombatAttributeEvent>>(DiagnosticCode::InvalidArgument,
58 "Combat attribute value must be finite", definition.tag);
59 const double previous = values_.getBase(definition.tag);
60 const double clamped = std::clamp(newValue, definition.minimumValue, definition.maximumValue);
61 if (clamped == previous)
62 return Result<std::vector<CombatAttributeEvent>>::success({}, Status::success(StatusCode::NoOp));
63
64 values_.setBase(definition.tag, clamped);
65 std::vector<CombatAttributeEvent> events;
66 const auto kind = clamped > previous ? CombatAttributeEventKind::Gain : CombatAttributeEventKind::Loss;
67 events.push_back({definition.tag, kind, previous, clamped, clamped - previous});
68 if (clamped == definition.minimumValue && previous > definition.minimumValue)
69 events.push_back({definition.tag, CombatAttributeEventKind::Depleted, previous, clamped, clamped - previous});
70 return Result<std::vector<CombatAttributeEvent>>::success(std::move(events),
71 Status::success(StatusCode::Applied));
72}

93Result<CombatAttributeAdvance> CombatAttributeRuntime::advance(Duration delta) {
94 if (delta < Duration::zero())
95 return failure<CombatAttributeAdvance>(DiagnosticCode::InvalidArgument,
96 "Combat attribute delta must be non-negative", "delta");
97 if (delta.isZero())
98 return Result<CombatAttributeAdvance>::success({}, Status::success(StatusCode::NoOp));
99
100 struct Candidate {
101 const CombatAttributeDefinition* definition = nullptr;
102 double value = 0.0;
103 };
104 std::vector<Candidate> candidates;
105 candidates.reserve(definitions_.size());
106 for (const auto& [tag, definition] : definitions_) {
107 const double amount = definition.regenerationPerSecond * delta.seconds();
108 if (!std::isfinite(amount))
109 return failure<CombatAttributeAdvance>(DiagnosticCode::InvalidArgument,
110 "Combat attribute regeneration overflowed", tag);
111 candidates.push_back({&definition, values_.getBase(tag) + amount});
112 }
113
114 CombatAttributeAdvance result;
115 for (const Candidate& candidate : candidates) {
116 auto committed = commit(*candidate.definition, candidate.value);
117 if (!committed) return Result<CombatAttributeAdvance>::failure(committed.status());
118 auto events = std::move(committed).takeValue();
119 result.events.insert(result.events.end(), std::make_move_iterator(events.begin()),
120 std::make_move_iterator(events.end()));
121 }
122 return Result<CombatAttributeAdvance>::success(std::move(result));
123}
