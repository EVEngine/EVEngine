31void unlinkFromParent(SceneHost::Tree &tree, int childIndex) {
32 SceneNode &child = tree.nodes[size_t(childIndex)];
33 const int parent = child.parent;
34 if (parent < 0) return;
35 SceneNode &p = tree.nodes[size_t(parent)];
36 if (p.firstChild == childIndex) {
37 p.firstChild = child.nextSibling;
38 } else {
39 for (int c = p.firstChild; c >= 0; c = tree.nodes[size_t(c)].nextSibling) {
40 if (tree.nodes[size_t(c)].nextSibling == childIndex) {
41 tree.nodes[size_t(c)].nextSibling = child.nextSibling;
42 break;
43 }
44 }
45 }
46 child.nextSibling = -1;
47 child.parent = -1;
48}

50void linkAsLastChild(SceneHost::Tree &tree, int parentIndex, int childIndex) {
51 SceneNode &parent = tree.nodes[size_t(parentIndex)];
52 SceneNode &child = tree.nodes[size_t(childIndex)];
53 child.parent = parentIndex;
54 child.nextSibling = -1;
55 if (parent.firstChild < 0) {
56 parent.firstChild = childIndex;
57 return;
58 }
59 int c = parent.firstChild;
60 while (tree.nodes[size_t(c)].nextSibling >= 0) c = tree.nodes[size_t(c)].nextSibling;
61 tree.nodes[size_t(c)].nextSibling = childIndex;
62}

99void walkBreadthFirstImpl(SceneHost *host, int nodeIndex, const SceneHost::NodeVisitFn &fn) {
100 if (!host || nodeIndex < 0 || !fn) return;
101 auto t = host->tree();
102 if (nodeIndex >= int(t->nodes.size())) return;
103 std::vector<int> queue;
104 queue.push_back(nodeIndex);
105 for (size_t i = 0; i < queue.size(); ++i) {
106 const int idx = queue[i];
107 fn(host, idx, t->nodes[size_t(idx)]);
108 for (int c = t->nodes[size_t(idx)].firstChild; c >= 0;
109 c = t->nodes[size_t(c)].nextSibling) {

154SceneMutationStatus SceneHost::appendNode(SceneNode node, const std::string &parentId) {
155 if (node.id.empty() || hasNode(node.id)) return SceneMutationStatus::Rejected;
156 const int parentIndex = parentId.empty() ? -1 : findIndexById(parentId);
157 if (!parentId.empty() && parentIndex < 0) return SceneMutationStatus::Rejected;
158 node.parent = -1;
159 node.firstChild = -1;
160 node.nextSibling = -1;
161 node.localDirty = true;
162 node.subtreeDirty = true;
163 auto value = tree();
164 const int index = static_cast<int>(value->nodes.size());
165 value->nodes.push_back(std::move(node));
166 if (parentIndex >= 0)
167 linkAsLastChild(*value, parentIndex, index);
168 else if (value->root < 0)
169 value->root = index;
170 value->dirty = true;
171 value->transformDirty = true;
172 value->indexValid = false;
173 fireEvent("node_added", value->nodes[size_t(index)].id, parentId);
174 return SceneMutationStatus::Applied;
175}

177SceneMutationStatus SceneHost::removeLeaf(const std::string &nodeId) {
178 const int index = findIndexById(nodeId);
179 auto value = tree();
180 if (index < 0 || value->nodes[size_t(index)].firstChild >= 0) return SceneMutationStatus::Rejected;
181 const int oldParent = value->nodes[size_t(index)].parent;
182 const std::string parentId = oldParent >= 0 ? value->nodes[size_t(oldParent)].id : std::string{};
183 unlinkFromParent(*value, index);
184 value->nodes.erase(value->nodes.begin() + index);
185 for (SceneNode &node : value->nodes) {
186 if (node.parent > index) --node.parent;
187 if (node.firstChild > index) --node.firstChild;
188 if (node.nextSibling > index) --node.nextSibling;
189 }
190 if (value->root == index)
191 value->root = value->nodes.empty() ? -1 : 0;
192 else if (value->root > index)
193 --value->root;
194 value->dirty = true;
195 value->transformDirty = true;
196 value->indexValid = false;
197 fireEvent("node_removed", nodeId, parentId);
198 return SceneMutationStatus::Applied;
199}

302int SceneHost::findIndexByPath(const std::string &path) {
303 auto parts = splitPath(path);
304 if (parts.empty()) return -1;
305 auto t = tree();
306 if (t->root < 0) return -1;
307
308 int idx = t->root;
309 size_t start = 0;
310 if (t->nodes[size_t(idx)].id == parts[0] || t->nodes[size_t(idx)].name == parts[0]) {
311 start = 1;
312 }
313 for (size_t p = start; p < parts.size(); ++p) {
314 const std::string &want = parts[p];
315 int match = -1;
316 for (int child = t->nodes[size_t(idx)].firstChild; child >= 0;
317 child = t->nodes[size_t(child)].nextSibling) {
318 const SceneNode &n = t->nodes[size_t(child)];
319 if (n.id == want || n.name == want) {
320 match = child;
321 break;
322 }
323 }
324 if (match < 0) return -1;
325 idx = match;
326 }
327 return idx;
328}

453bool SceneHost::setParent(int childIndex, int parentIndex) {
454 auto t = tree();
455 if (childIndex < 0 || childIndex >= int(t->nodes.size())) return false;
456 if (parentIndex < -1 || parentIndex >= int(t->nodes.size())) return false;
457 if (childIndex == parentIndex) return false;
458 if (parentIndex >= 0 && isAncestor(*t, childIndex, parentIndex)) return false;
459
460 unlinkFromParent(*t, childIndex);
461 if (parentIndex < 0) {
462 t->nodes[size_t(childIndex)].parent = -1;
463 t->nodes[size_t(childIndex)].nextSibling = -1;
464 markSubtreeDirty(childIndex);
465 fireEvent("node_moved", t->nodes[size_t(childIndex)].id, "");
466 return true;
467 }
468 linkAsLastChild(*t, parentIndex, childIndex);
469 markSubtreeDirty(childIndex);
470 markSubtreeDirty(parentIndex);
471 fireEvent("node_moved", t->nodes[size_t(childIndex)].id,
472 t->nodes[size_t(parentIndex)].id);
473 return true;
474}
