40Result<MontageHandle> MontageCoordinator::play(std::size_t layer, action::ActionTimeline timeline,
41 std::vector<MontageClipAsset>&& clips,
42 action::ActionExecutionId executionId, SimulationTick tick) {
43 if (layer > static_cast<std::size_t>(std::numeric_limits<MontageHandle::index_type>::max() / 2U))
44 return coordinatorError<MontageHandle>(DiagnosticCode::InvalidArgument, "montage layer is too large", "layer");
45 ensureLayer(layer);
46 Layer& entry = layers_[layer];
47 const std::size_t nextIndex = 1U - entry.active;
48 Slot& next = entry.slots[nextIndex];
49 if (next.retired)
50 return coordinatorError<MontageHandle>(DiagnosticCode::Failed, "montage slot generation is exhausted", "slot");
51 if (next.player) retireSlot(next);
52 if (next.retired)
53 return coordinatorError<MontageHandle>(DiagnosticCode::Failed, "montage slot generation is exhausted", "slot");
54
55 auto candidate = std::make_unique<MontagePlayer>(skeleton_);
56 const Duration blendOut = timeline.montage.defaultBlendIn;
57 auto prepared = candidate->prepare(std::move(timeline), std::move(clips));
58 if (!prepared) return Result<MontageHandle>::failure(prepared.status());
59 auto started = candidate->play(executionId);
60 if (!started) return Result<MontageHandle>::failure(started.status());
61 Slot& previous = entry.slots[entry.active];
62 if (previous.player && previous.player->isPlaying()) {
63 auto fading = previous.player->beginBlendOut(blendOut, tick);
64 if (!fading) return Result<MontageHandle>::failure(fading.status());
65 }
66 next.player = std::move(candidate);
67 entry.active = nextIndex;
68 return Result<MontageHandle>::success(
69 MontageHandle(static_cast<MontageHandle::index_type>(handleIndex(layer, nextIndex)), next.generation),
70 Status::success(StatusCode::Applied));
71}

73MontageCoordinator::Slot* MontageCoordinator::slotFor(MontageHandle handle) noexcept {
74 if (!handle.isValid()) return nullptr;
75 const std::size_t layer = handle.index() / 2U;
76 const std::size_t slot = handle.index() % 2U;
77 if (layer >= layers_.size()) return nullptr;

79 return candidate.player && candidate.generation == handle.generation() ? &candidate : nullptr;

109Result<void> MontageCoordinator::advanceBlendOuts(Duration delta, SimulationTick tick, MontageHandle except) {
110 for (Layer& layer : layers_) {
111 for (Slot& slot : layer.slots) {
112 if (!slot.player || !slot.player->isBlendingOut()) continue;
113 const MontageHandle handle(static_cast<MontageHandle::index_type>(&layer - layers_.data()) * 2U +
114 static_cast<MontageHandle::index_type>(&slot - layer.slots),
115 slot.generation);
116 if (handle == except) continue;
117 auto advanced = slot.player->advanceBlendOut(delta, tick);
118 if (!advanced) return Result<void>::failure(advanced.status());
119 }
120 }
121 collectFinished();
122 return Result<void>::success(Status::success(StatusCode::Applied));
123}

131Result<std::reference_wrapper<AnimPose>> MontageCoordinator::pose(std::size_t layerIndex) {
132 if (layerIndex >= layers_.size())
133 return coordinatorError<std::reference_wrapper<AnimPose>>(DiagnosticCode::NotFound,
134 "montage layer does not exist", "layer");
135 Layer& layer = layers_[layerIndex];
136 skeleton_.applyBindPose(layer.pose.get());
137 for (const Slot& slot : layer.slots) {
138 if (!slot.player || slot.player->weight() <= 0.0) continue;
139 AnimPose base;
140 base.copyFrom(layer.pose.get());
141 layer.pose->blendFrom(&base, &slot.player->pose(), static_cast<float>(slot.player->weight()));
142 }
143 return Result<std::reference_wrapper<AnimPose>>::success(std::ref(*layer.pose));
144}
