49Result<MontageHandle> MontageCoordinator::play(std::size_t layer, action::ActionTimeline timeline,
50 std::vector<MontageClipAsset>&& clips,
51 action::ActionExecutionId executionId, SimulationTick tick) {
52 if (layer > static_cast<std::size_t>(std::numeric_limits<MontageHandle::index_type>::max() / 2U))
53 return coordinatorError<MontageHandle>(DiagnosticCode::InvalidArgument, "montage layer is too large", "layer");
54 ensureLayer(layer);
55 Layer& entry = layers_[layer];
56 const std::size_t nextIndex = 1U - entry.active;
57 Slot& next = entry.slots[nextIndex];
58 if (next.retired)
59 return coordinatorError<MontageHandle>(DiagnosticCode::Failed, "montage slot generation is exhausted", "slot");
60 if (next.player) retireSlot(next);
61 if (next.retired)
62 return coordinatorError<MontageHandle>(DiagnosticCode::Failed, "montage slot generation is exhausted", "slot");
63
64 auto candidate = std::make_unique<MontagePlayer>(skeleton_);
65 if (entry.rootMotionReceiver) candidate->setRootMotionReceiver(*entry.rootMotionReceiver);
66 const Duration blendOut = timeline.montage.defaultBlendIn;
67 auto prepared = candidate->prepare(std::move(timeline), std::move(clips));
68 if (!prepared) return Result<MontageHandle>::failure(prepared.status());
69 auto started = candidate->play(executionId);
70 if (!started) return Result<MontageHandle>::failure(started.status());
71 Slot& previous = entry.slots[entry.active];
72 if (previous.player && (previous.player->isPlaying() || previous.player->isFinished())) {
73 auto fading = previous.player->beginBlendOut(blendOut, tick);
74 if (!fading) return Result<MontageHandle>::failure(fading.status());
75 }
76 next.player = std::move(candidate);
77 entry.active = nextIndex;
78 return Result<MontageHandle>::success(
79 MontageHandle(static_cast<MontageHandle::index_type>(handleIndex(layer, nextIndex)), next.generation),
80 Status::success(StatusCode::Applied));
81}

119Result<void> MontageCoordinator::advanceBlendOuts(Duration delta, SimulationTick tick, MontageHandle except) {
120 for (Layer& layer : layers_) {
121 for (Slot& slot : layer.slots) {
122 if (!slot.player || !slot.player->isBlendingOut()) continue;
123 const MontageHandle handle(static_cast<MontageHandle::index_type>(&layer - layers_.data()) * 2U +
124 static_cast<MontageHandle::index_type>(&slot - layer.slots),
125 slot.generation);
126 if (handle == except) continue;
127 auto advanced = slot.player->advanceBlendOut(delta, tick);
128 if (!advanced) return Result<void>::failure(advanced.status());
129 }
130 }
131 collectFinished();
132 return Result<void>::success(Status::success(StatusCode::Applied));
133}

238double MontageCoordinator::evaluateLayerPose(Layer& layer) const {
239 const Slot* first = nullptr;
240 const Slot* second = nullptr;
241 double total = 0.0;
242 for (const Slot& slot : layer.slots) {
243 if (!slot.player || slot.player->weight() <= 0.0) continue;
244 total += slot.player->weight();
245 if (!first) first = &slot;
246 else second = &slot;
247 }
248 if (!first) {
249 skeleton_.applyBindPose(layer.rawPose.get());
250 return 0.0;
251 }
252
253 layer.rawPose->copyFrom(&first->player->pose());
254 if (second) {
255 if (total > 0.0) {
256 layer.blendScratch->copyFrom(layer.rawPose.get());
257 layer.rawPose->blendFrom(layer.blendScratch.get(), &second->player->pose(),
258 static_cast<float>(second->player->weight() / total));
259 }
260 }
261 return total;
262}

264Result<std::reference_wrapper<AnimPose>> MontageCoordinator::pose(std::size_t layerIndex) {
265 if (layerIndex >= layers_.size())
266 return coordinatorError<std::reference_wrapper<AnimPose>>(DiagnosticCode::NotFound,
267 "montage layer does not exist", "layer");
268 Layer& layer = layers_[layerIndex];
269 (void)evaluateLayerPose(layer);
270 skeleton_.applyBindPose(layer.pose.get());
271 if (layer.boneMask.empty()) {
272 layer.pose->copyFrom(layer.rawPose.get());
273 } else {
274 for (int bone = 0; bone < layer.pose->getBoneCount(); ++bone) {
275 const float weight = layer.boneMask[static_cast<std::size_t>(bone)];
276 layer.pose->local(bone) = blendTRS(layer.pose->local(bone), layer.rawPose->local(bone), weight);
277 }
278 }
279 return Result<std::reference_wrapper<AnimPose>>::success(std::ref(*layer.pose));
280}

282Result<std::reference_wrapper<AnimPose>> MontageCoordinator::compose(const AnimPose& basePose) {
283 if (basePose.getBoneCount() != skeleton_.getBoneCount())
284 return coordinatorError<std::reference_wrapper<AnimPose>>(
285 DiagnosticCode::InvalidArgument, "montage base pose must match the coordinator skeleton", "basePose");
286 composedPose_->copyFrom(&basePose);
287 for (Layer& layer : layers_) {
288 const double rawWeight = evaluateLayerPose(layer);
289 const float layerWeight = clampf(static_cast<float>(rawWeight), 0.0f, 1.0f) * layer.weight;
290 if (layerWeight <= 0.0f) continue;
291 for (int bone = 0; bone < composedPose_->getBoneCount(); ++bone) {
292 const float maskWeight = layer.boneMask.empty() ? 1.0f : layer.boneMask[static_cast<std::size_t>(bone)];
293 const float weight = clampf(layerWeight * maskWeight, 0.0f, 1.0f);
294 if (weight <= 0.0f) continue;
295 if (layer.additive) {
296 applyAdditiveTRS(composedPose_->local(bone), layer.rawPose->local(bone), skeleton_.bindLocal(bone),
297 weight);
298 } else {
299 composedPose_->local(bone) = blendTRS(composedPose_->local(bone), layer.rawPose->local(bone), weight);
300 }
301 }
302 }
303 composedPose_->computeWorld(&skeleton_);
304 return Result<std::reference_wrapper<AnimPose>>::success(std::ref(*composedPose_));
305}
