155void AnimGraph::applyAdditive(AnimPose& out, const AnimPose& base, const AnimPose& delta, float weight,
156 const std::vector<float>* mask) const {
157 out.resize(base.getBoneCount());
158 for (int i = 0; i < base.getBoneCount(); ++i) {
159 const float w = clampf(weight * (mask ? (*mask)[static_cast<size_t>(i)] : 1.f), 0.f, 1.f);
160 const TransformTRS& a = base.local(i);
161 const TransformTRS& d = delta.local(i);
162 TransformTRS r = a;
163 r.px += d.px * w;
164 r.py += d.py * w;
165 r.pz += d.pz * w;
166 r.sx *= lerpf(1.f, d.sx, w);
167 r.sy *= lerpf(1.f, d.sy, w);
168 r.sz *= lerpf(1.f, d.sz, w);
169 float qx, qy, qz, qw;
170 slerpQuat(0.f, 0.f, 0.f, 1.f, d.qx, d.qy, d.qz, d.qw, w, qx, qy, qz, qw);
171 r.qx = a.qw * qx + a.qx * qw + a.qy * qz - a.qz * qy;
172 r.qy = a.qw * qy - a.qx * qz + a.qy * qw + a.qz * qx;
173 r.qz = a.qw * qz + a.qx * qy - a.qy * qx + a.qz * qw;
174 r.qw = a.qw * qw - a.qx * qx - a.qy * qy - a.qz * qz;
175 r.normalizeRotation();
176 out.local(i) = r;
177 }

180const AnimPose& AnimGraph::evaluateBlendSpace(Node& node, bool twoDimensional) {
181 if (node.points.empty()) {
182 skeleton_->applyBindPose(&node.cache);
183 return node.cache;
184 }
185 if (node.points.size() == 1) {
186 node.cache.copyFrom(&evaluate(node.points.front().child));
187 return node.cache;
188 }
189 if (!twoDimensional) {
190 if (node.x <= node.points.front().x) {
191 node.cache.copyFrom(&evaluate(node.points.front().child));
192 return node.cache;
193 }
194 if (node.x >= node.points.back().x) {
195 node.cache.copyFrom(&evaluate(node.points.back().child));
196 return node.cache;
197 }
198 for (size_t i = 0; i + 1 < node.points.size(); ++i) {
199 const Point &a = node.points[i], &b = node.points[i + 1];
200 if (node.x >= a.x && node.x <= b.x) {
201 blendMasked(node.cache, evaluate(a.child), evaluate(b.child), (node.x - a.x) / (b.x - a.x), nullptr);
202 return node.cache;
203 }
204 }
205 }
206 // Inverse-distance weights are stable outside the convex hull and degrade
207 // gracefully for authoring-time sparse 2D spaces.
208 std::vector<float> weights(node.points.size());
209 float sum = 0.f;
210 for (size_t i = 0; i < node.points.size(); ++i) {
211 const float dx = node.x - node.points[i].x, dy = node.y - node.points[i].y;
212 weights[i] = 1.f / std::max(1e-6f, dx * dx + dy * dy);
213 sum += weights[i];
214 }
215 node.cache.copyFrom(&evaluate(node.points.front().child));
216 float accumulated = weights[0] / sum;
217 for (size_t i = 1; i < node.points.size(); ++i) {
218 const float w = weights[i] / sum;
219 const float relative = w / (accumulated + w);
220 node.scratch.copyFrom(&node.cache);
221 blendMasked(node.cache, node.scratch, evaluate(node.points[i].child), relative, nullptr);
222 accumulated += w;
223 }
224 return node.cache;
225}

227const AnimPose& AnimGraph::evaluate(int id) {
228 Node& node = requireNode(id);
229 if (node.cacheGeneration == generation_) return node.cache;
230 node.cacheGeneration = generation_;
231 switch (node.kind) {
232 case Kind::Clip: node.clip->sample(node.time, &node.cache, skeleton_); break;
233 case Kind::Blend: blendMasked(node.cache, evaluate(node.a), evaluate(node.b), node.weight, nullptr); break;
234 case Kind::Layer: blendMasked(node.cache, evaluate(node.a), evaluate(node.b), node.weight, &node.mask); break;
235 case Kind::Additive:
236 applyAdditive(node.cache, evaluate(node.a), evaluate(node.b), node.weight, &node.mask);
237 break;
238 case Kind::OneShot: {
239 const AnimPose& base = evaluate(node.a);
240 if (!node.active)
241 node.cache.copyFrom(&base);
242 else {
243 const float duration = requireNode(node.b).clip ? requireNode(node.b).clip->getDuration() : 0.f;
244 float w = 1.f;
245 if (node.fadeIn > 0.f) w = std::min(w, node.time / node.fadeIn);
246 if (node.fadeOut > 0.f) w = std::min(w, (duration - node.time) / node.fadeOut);
247 blendMasked(node.cache, base, evaluate(node.b), clampf(w, 0.f, 1.f), &node.mask);
248 }
249 break;
250 }
251 case Kind::BlendSpace1D: return evaluateBlendSpace(node, false);
252 case Kind::BlendSpace2D: return evaluateBlendSpace(node, true);
253 }
254 return node.cache;
255}

257void AnimGraph::updateUnchecked(float dt) {
258 if (dt < 0.f) throw Exception("AnimGraph.update: dt must be >= 0");
259 if (root_ < 0) throw Exception("AnimGraph.update: root is not set");
260 for (Node& node : nodes_) {
261 if (node.kind == Kind::Clip) node.time = node.clip->wrapTime(node.time + dt * node.speed);
262 if (node.kind == Kind::OneShot && node.active) {
263 node.time += dt;
264 const Node& shot = requireNode(node.b);
265 if (shot.clip && node.time >= shot.clip->getDuration()) node.active = false;
266 }
267 }
268 if (++generation_ == 0) {
269 generation_ = 1;
270 for (Node& n : nodes_) n.cacheGeneration = 0;
271 }
272 output_.copyFrom(&evaluate(root_));
273}

275eve::Result<void> AnimGraph::advance(const eve::SimulationStep& step) {
276 auto seconds = detail::secondsForStep(step, hasLastTick_, lastTick_, "AnimGraph");
