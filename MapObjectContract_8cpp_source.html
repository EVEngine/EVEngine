80eve::Result<void> validateProperty(const std::string& value, const PropertyRule& rule,
81 const std::string& path) {
82 double numeric = 0.0;
83 if (rule.kind == "string") {
84 if (!rule.choices.empty() && !rule.choices.contains(value))
85 return fail(eve::DiagnosticCode::InvalidArgument, "property value is outside its enum", path);
86 } else if (rule.kind == "int") {
87 std::int64_t integer = 0;
88 if (!parseInteger(value, integer))
89 return fail(eve::DiagnosticCode::TypeMismatch, "property must be an integer", path);
90 numeric = static_cast<double>(integer);
91 } else if (rule.kind == "number") {
92 if (!parseNumber(value, numeric))
93 return fail(eve::DiagnosticCode::TypeMismatch, "property must be a finite number", path);
94 } else if (rule.kind == "bool") {
95 if (value != "true" && value != "false")
96 return fail(eve::DiagnosticCode::TypeMismatch, "property must be true or false", path);
97 } else {
98 return fail(eve::DiagnosticCode::Unsupported, "unsupported property kind", path);
99 }
100 if (rule.minimum && numeric < *rule.minimum)
101 return fail(eve::DiagnosticCode::InvalidArgument, "property is below its minimum", path);
102 if (rule.maximum && numeric > *rule.maximum)
103 return fail(eve::DiagnosticCode::InvalidArgument, "property exceeds its maximum", path);
104 return eve::Result<void>::success();
105}

109eve::Result<void> validateMapObjects(std::span<const MapObject> objects, std::string_view contractJson) {
110 auto parsed = eve::Value::fromJson(contractJson);
111 if (!parsed.ok()) return eve::Result<void>::failure(parsed.status());
112 const auto* root = parsed.value().getIf<eve::Value::Object>();
113 if (!root) return fail(eve::DiagnosticCode::TypeMismatch, "contract root must be an object", "$contract");
114
115 const eve::Value* schema = member(*root, "schema");
116 const eve::Value* version = member(*root, "version");
117 const eve::Value* typesValue = member(*root, "types");
118 if (!schema || !schema->isString() || schema->asString() != "eve.map.object-contract")
119 return fail(eve::DiagnosticCode::InvalidArgument, "unsupported contract schema", "$contract.schema");
120 if (!version || !version->isInt64() || version->asInt() != 1)
121 return fail(eve::DiagnosticCode::UnknownVersion, "unsupported contract version", "$contract.version");
122 const auto* types = typesValue ? typesValue->getIf<eve::Value::Array>() : nullptr;
123 if (!types) return fail(eve::DiagnosticCode::TypeMismatch, "types must be an array", "$contract.types");
124
125 bool requireUniqueNames = true;
126 bool allowUnknownTypes = false;
127 if (!booleanMember(*root, "requireUniqueNames", true, requireUniqueNames))
128 return fail(eve::DiagnosticCode::TypeMismatch, "requireUniqueNames must be boolean",
129 "$contract.requireUniqueNames");
130 if (!booleanMember(*root, "allowUnknownTypes", false, allowUnknownTypes))
131 return fail(eve::DiagnosticCode::TypeMismatch, "allowUnknownTypes must be boolean",
132 "$contract.allowUnknownTypes");
133
134 std::map<std::string, TypeRule> rules;
135 for (std::size_t typeIndex = 0; typeIndex < types->size(); ++typeIndex) {
136 const std::string base = "$contract.types[" + std::to_string(typeIndex) + "]";
137 const auto* definition = (*types)[typeIndex].getIf<eve::Value::Object>();
138 if (!definition) return fail(eve::DiagnosticCode::TypeMismatch, "type rule must be an object", base);
139 const eve::Value* type = member(*definition, "type");
140 if (!type || !type->isString() || type->asString().empty())
141 return fail(eve::DiagnosticCode::InvalidArgument, "type must be a non-empty string", base + ".type");
142 TypeRule typeRule;
143 if (!booleanMember(*definition, "allowUnknownProperties", false, typeRule.allowUnknownProperties))
144 return fail(eve::DiagnosticCode::TypeMismatch, "allowUnknownProperties must be boolean",
145 base + ".allowUnknownProperties");
146 const eve::Value* propertiesValue = member(*definition, "properties");
147 const auto* properties = propertiesValue ? propertiesValue->getIf<eve::Value::Array>() : nullptr;
148 if (propertiesValue && !properties)
149 return fail(eve::DiagnosticCode::TypeMismatch, "properties must be an array", base + ".properties");
150 if (properties) {
151 for (std::size_t propertyIndex = 0; propertyIndex < properties->size(); ++propertyIndex) {
152 const std::string propertyPath = base + ".properties[" + std::to_string(propertyIndex) + "]";
153 const auto* definitionObject = (*properties)[propertyIndex].getIf<eve::Value::Object>();
154 if (!definitionObject)
155 return fail(eve::DiagnosticCode::TypeMismatch, "property rule must be an object", propertyPath);
156 const eve::Value* name = member(*definitionObject, "name");
157 const eve::Value* kind = member(*definitionObject, "kind");
158 if (!name || !name->isString() || name->asString().empty())
159 return fail(eve::DiagnosticCode::InvalidArgument, "property name must be non-empty",
160 propertyPath + ".name");
161 if (!kind || !kind->isString())
162 return fail(eve::DiagnosticCode::TypeMismatch, "property kind must be a string",
163 propertyPath + ".kind");
164 PropertyRule propertyRule{name->asString(), kind->asString()};
165 if (!booleanMember(*definitionObject, "required", false, propertyRule.required))
166 return fail(eve::DiagnosticCode::TypeMismatch, "required must be boolean",
167 propertyPath + ".required");
168 for (std::string_view field : {std::string_view("min"), std::string_view("max")}) {
169 if (const eve::Value* bound = member(*definitionObject, field)) {
170 double number = 0.0;
171 if (!numericValue(*bound, number))
172 return fail(eve::DiagnosticCode::TypeMismatch, std::string(field) + " must be numeric",
173 propertyPath + "." + std::string(field));
174 (field == "min" ? propertyRule.minimum : propertyRule.maximum) = number;
175 }
176 }
177 if (propertyRule.minimum && propertyRule.maximum && *propertyRule.minimum > *propertyRule.maximum)
178 return fail(eve::DiagnosticCode::InvalidArgument, "min must not exceed max", propertyPath);
179 if (const eve::Value* enumValue = member(*definitionObject, "enum")) {
180 const auto* choices = enumValue->getIf<eve::Value::Array>();
181 if (!choices)
182 return fail(eve::DiagnosticCode::TypeMismatch, "enum must be an array", propertyPath + ".enum");
183 for (const eve::Value& choice : *choices) {
184 if (!choice.isString())
185 return fail(eve::DiagnosticCode::TypeMismatch, "enum values must be strings",
186 propertyPath + ".enum");
187 if (!propertyRule.choices.emplace(choice.asString()).second)
188 return fail(eve::DiagnosticCode::AlreadyExists, "duplicate enum value",
189 propertyPath + ".enum");
190 }
191 }
192 if (!typeRule.properties.emplace(propertyRule.name, std::move(propertyRule)).second)
193 return fail(eve::DiagnosticCode::AlreadyExists, "duplicate property rule", propertyPath + ".name");
194 }
195 }
196 if (!rules.emplace(type->asString(), std::move(typeRule)).second)
197 return fail(eve::DiagnosticCode::AlreadyExists, "duplicate type rule", base + ".type");
198 }
199
200 std::set<std::string> names;
201 for (std::size_t objectIndex = 0; objectIndex < objects.size(); ++objectIndex) {
202 const MapObject& object = objects[objectIndex];
203 const std::string base = "$objects[" + std::to_string(objectIndex) + "]";
204 if (!std::isfinite(object.x) || !std::isfinite(object.y) || !std::isfinite(object.width) ||
205 !std::isfinite(object.height) || object.width < 0.f || object.height < 0.f)
206 return fail(eve::DiagnosticCode::InvalidArgument, "object geometry must be finite and non-negative", base);
207 if (requireUniqueNames && (!stableName(object.name) || !names.emplace(object.name).second))
208 return fail(eve::DiagnosticCode::Conflict,
209 "object names must be unique, bounded, non-empty, and control-free", base + ".name");
210 const auto type = rules.find(object.type);
211 if (type == rules.end()) {
212 if (!allowUnknownTypes)
213 return fail(eve::DiagnosticCode::Unsupported, "object type is not admitted", base + ".type");
214 continue;
215 }
216 for (const auto& [name, rule] : type->second.properties) {
217 const auto property = object.properties.find(name);
218 if (property == object.properties.end()) {
219 if (rule.required)
220 return fail(eve::DiagnosticCode::NotFound, "required property is missing",
221 base + ".properties." + name);
222 continue;
223 }
224 auto valid = validateProperty(property->second, rule, base + ".properties." + name);
225 if (!valid.ok()) return valid;
226 }
227 if (!type->second.allowUnknownProperties) {
228 for (const auto& [name, value] : object.properties) {
229 (void)value;
230 if (!type->second.properties.contains(name))
231 return fail(eve::DiagnosticCode::Unsupported, "property is not admitted",
232 base + ".properties." + name);
233 }
234 }
235 }
236 return eve::Result<void>::success();
237}
