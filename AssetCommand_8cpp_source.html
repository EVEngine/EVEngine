27Result<std::vector<std::uint8_t>> readFile(const std::filesystem::path& path,
28 std::uint64_t maximumBytes) {
29 std::error_code ec;
30 const auto size = std::filesystem::file_size(path, ec);
31 if (ec || size > maximumBytes || size > std::numeric_limits<std::size_t>::max())
32 return Result<std::vector<std::uint8_t>>::failure(Diagnostic::error(
33 DiagnosticCode::InvalidArgument, "asset input size is outside limits", path.string(), {},
34 "cmd.asset"));
35 std::ifstream input(path, std::ios::binary);
36 if (!input) return Result<std::vector<std::uint8_t>>::failure(Diagnostic::error(
37 DiagnosticCode::NotFound, "cannot open asset input", path.string(), {}, "cmd.asset"));
38 std::vector<std::uint8_t> bytes(static_cast<std::size_t>(size));
39 if (!bytes.empty()) input.read(reinterpret_cast<char*>(bytes.data()), static_cast<std::streamsize>(bytes.size()));
40 if (!input) return Result<std::vector<std::uint8_t>>::failure(Diagnostic::error(
41 DiagnosticCode::Failed, "cannot read complete asset input", path.string(), {}, "cmd.asset"));
42 return Result<std::vector<std::uint8_t>>::success(std::move(bytes));
43}

70Result<std::map<std::string, std::vector<std::uint8_t>>> readTree(
71 const std::filesystem::path& root, std::uint64_t maximumBytes) {
72 if (!std::filesystem::is_directory(root))
73 return Result<std::map<std::string, std::vector<std::uint8_t>>>::failure(Diagnostic::error(
74 DiagnosticCode::NotFound, "asset project root is not a directory", root.string(), {}, "cmd.asset"));
75 std::map<std::string, std::vector<std::uint8_t>> files;
76 std::uint64_t total = 0;
77 std::error_code ec;
78 for (std::filesystem::recursive_directory_iterator it(
79 root, std::filesystem::directory_options::skip_permission_denied, ec), end;
80 it != end; it.increment(ec)) {
81 if (ec) return Result<std::map<std::string, std::vector<std::uint8_t>>>::failure(Diagnostic::error(
82 DiagnosticCode::Failed, "cannot traverse asset project: " + ec.message(), root.string(), {},
83 "cmd.asset"));
84 if (!it->is_regular_file()) continue;
85 const auto size = it->file_size(ec);
86 if (ec || size > maximumBytes || total > maximumBytes - size)
87 return Result<std::map<std::string, std::vector<std::uint8_t>>>::failure(Diagnostic::error(
88 DiagnosticCode::InvalidArgument, "asset project exceeds source budget", it->path().string(), {},
89 "cmd.asset"));
90 auto bytes = readFile(it->path(), maximumBytes);
91 if (!bytes) return Result<std::map<std::string, std::vector<std::uint8_t>>>::failure(bytes.status());
92 auto relative = std::filesystem::relative(it->path(), root, ec).generic_string();
93 if (ec || relative.empty())
94 return Result<std::map<std::string, std::vector<std::uint8_t>>>::failure(Diagnostic::error(
95 DiagnosticCode::Failed, "cannot canonicalize project-relative asset path", it->path().string(), {},
96 "cmd.asset"));
97 total += size;
98 files.emplace(std::move(relative), std::move(bytes).takeValue());
99 }
100 return Result<std::map<std::string, std::vector<std::uint8_t>>>::success(std::move(files));
101}

165 int runImport() {
166 auto identity = packageIdentity(packageId, packageName, packageVersion);
167 if (!identity) { printFailure(identity.status()); return 2; }
168 Result<PreparedAssetImport> prepared = Result<PreparedAssetImport>::failure(
169 Diagnostic::error(DiagnosticCode::Unsupported, "unknown importer", from, {}, "cmd.asset"));
170 asset_import::AssetImportLimits limits;
171 if (from == "image" || from == "gltf") {
172 auto source = readFile(input, limits.maximumSourceBytes);
173 if (!source) { prepared.ignore(); printFailure(source.status()); return 2; }
174 if (from == "image") {
175 if (colorSpace != "srgb" && colorSpace != "linear") {
176 prepared.ignore(); std::cerr << "--color-space must be srgb or linear\n"; return 2;
177 }
178 prepared.ignore();
179 prepared = asset_import::prepareImageImport(
180 {identity.value(), std::filesystem::path(input).filename().string(),
181 std::move(source).takeValue(),
182 colorSpace == "srgb" ? asset_import::ImageColorSpace::Srgb
183 : asset_import::ImageColorSpace::Linear,
184 usage, limits});
185 } else {
186 std::map<std::string, std::vector<std::uint8_t>> external;
187 std::error_code ec;
188 const auto parent = std::filesystem::path(input).parent_path();
189 for (std::filesystem::directory_iterator it(parent.empty() ? "." : parent, ec), end;
190 !ec && it != end; it.increment(ec)) {
191 if (!it->is_regular_file() || it->path() == std::filesystem::path(input)) continue;
192 auto value = readFile(it->path(), limits.maximumSourceBytes);
193 if (value) external.emplace(it->path().filename().generic_string(), std::move(value).takeValue());
194 }
195 prepared.ignore();
196 prepared = asset_import::prepareGltfImport(
197 {identity.value(), std::filesystem::path(input).filename().string(),
198 std::move(source).takeValue(), std::move(external), limits});
199 }
200 } else if (from == "unity" || from == "ue5") {
201 auto files = readTree(input, limits.maximumSourceBytes);
202 if (!files) { prepared.ignore(); printFailure(files.status()); return 2; }
203 prepared.ignore();
204 if (from == "unity")
205 prepared = asset_import::prepareUnityProjectImport(
206 {identity.value(), std::move(files).takeValue(), terrain, prefab, limits});
207 else
208 prepared = asset_import::prepareUnrealM4Import(
209 {identity.value(), std::move(files).takeValue(), descriptor, limits});
210 }
211 if (!prepared) { printFailure(prepared.status()); return 2; }
212 asset::AtomicAssetPackageStore store;
213 auto receipt = store.publishEva(output, prepared.value().manifest, prepared.value().entries);
214 if (!receipt) { printFailure(receipt.status()); return 3; }
215 std::cout << "published " << receipt.value().destination << " package="
216 << receipt.value().packageId.format() << " bytes=" << receipt.value().byteSize << "\n";
217 for (const auto& finding : prepared.value().findings)
218 std::cout << "finding " << static_cast<unsigned>(finding.disposition) << " "
219 << finding.feature << " " << finding.message << "\n";
220 return 0;
221 }

223 int runValidate(bool inspect) {
224 asset::EvaArchiveLimits evaLimits;
225 asset::EvpackLimits packLimits;
226 auto source = readFile(input, std::max(evaLimits.maximumArchiveBytes, packLimits.maximumPackageBytes));
227 if (!source) { printFailure(source.status()); return 2; }
228 if (source.value().size() >= 8 &&
229 std::string_view(reinterpret_cast<const char*>(source.value().data()), 6) == "EVPACK") {
230 auto parsed = asset::parseEvpack(source.value(), packLimits);
231 if (!parsed) { printFailure(parsed.status()); return 2; }
232 if (inspect) std::cout << "evpack package=" << parsed.value().packageId().format()
233 << " build=" << parsed.value().buildId().format()
234 << " variants=" << parsed.value().variants().size()
235 << " chunks=" << parsed.value().chunks().size() << "\n";
236 } else {
237 auto parsed = asset::parseEvaArchive(source.value(), evaLimits);
238 if (!parsed) { printFailure(parsed.status()); return 2; }
239 if (inspect) std::cout << "eva package=" << parsed.value().manifest.packageId.format()
240 << " name=" << parsed.value().manifest.packageName
241 << " assets=" << parsed.value().manifest.assets.size()
242 << " entries=" << parsed.value().entries.size() << "\n";
243 }
244 if (!inspect) std::cout << "valid " << input << "\n";
245 return 0;
246 }

248 int runCook() {
249 asset::EvaArchiveLimits evaLimits;
250 auto source = readFile(input, evaLimits.maximumArchiveBytes);
251 if (!source) { printFailure(source.status()); return 2; }
252 auto archive = asset::parseEvaArchive(source.value(), evaLimits);
253 if (!archive) { printFailure(archive.status()); return 2; }
254 auto profile = asset::assetCookProfileForTarget(target);
255 if (!profile) { printFailure(profile.status()); return 2; }
256 auto cooked = asset::cookEvaToEvpack(archive.value(), profile.value());
257 if (!cooked) { printFailure(cooked.status()); return 3; }
258 asset::AtomicAssetPackageStore store;
259 auto receipt = store.publishEvpack(output, cooked.value().bytes);
260 if (!receipt) { printFailure(receipt.status()); return 3; }
261 std::cout << "published " << receipt.value().destination << " build="
262 << receipt.value().buildId->format() << " chunks=" << cooked.value().chunkCount << "\n";
263 return 0;
264 }

302 int runDiff() {
303 asset::EvaArchiveLimits limits;
304 auto beforeBytes = readFile(input, limits.maximumArchiveBytes);
305 auto afterBytes = readFile(secondInput, limits.maximumArchiveBytes);
306 if (!beforeBytes) { printFailure(beforeBytes.status()); return 2; }
307 if (!afterBytes) { printFailure(afterBytes.status()); return 2; }
308 auto before = asset::parseEvaArchive(beforeBytes.value(), limits);
309 auto after = asset::parseEvaArchive(afterBytes.value(), limits);
310 if (!before) { printFailure(before.status()); return 2; }
311 if (!after) { printFailure(after.status()); return 2; }
312 auto difference = asset::diffEvaManifests(before.value().manifest, after.value().manifest);
313 if (!difference) { printFailure(difference.status()); return 3; }
314 for (const auto& change : difference.value().assets) {
315 const char* kind = change.kind == asset::EvaAssetChangeKind::Added ? "added" :
316 change.kind == asset::EvaAssetChangeKind::Removed ? "removed" : "changed";
317 std::cout << kind << " asset://" << change.assetId.format();
318 if (!change.beforeType.empty()) std::cout << " before=" << change.beforeType << "/" << change.beforeVersion.value();
319 if (!change.afterType.empty()) std::cout << " after=" << change.afterType << "/" << change.afterVersion.value();
320 std::cout << "\n";
321 }
322 std::cout << "dependencies +" << difference.value().addedDependencies << " -"
323 << difference.value().removedDependencies << " entrypoints="
324 << (difference.value().entrypointsChanged ? "changed" : "same") << "\n";
325 return 0;
326 }
