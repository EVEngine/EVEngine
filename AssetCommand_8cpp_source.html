29Result<std::vector<std::uint8_t>> readFile(const std::filesystem::path& path,
30 std::uint64_t maximumBytes) {
31 std::error_code ec;
32 const auto size = std::filesystem::file_size(path, ec);
33 if (ec || size > maximumBytes || size > std::numeric_limits<std::size_t>::max())
34 return Result<std::vector<std::uint8_t>>::failure(Diagnostic::error(
35 DiagnosticCode::InvalidArgument, "asset input size is outside limits", path.string(), {},
36 "cmd.asset"));
37 std::ifstream input(path, std::ios::binary);
38 if (!input) return Result<std::vector<std::uint8_t>>::failure(Diagnostic::error(
39 DiagnosticCode::NotFound, "cannot open asset input", path.string(), {}, "cmd.asset"));
40 std::vector<std::uint8_t> bytes(static_cast<std::size_t>(size));
41 if (!bytes.empty()) input.read(reinterpret_cast<char*>(bytes.data()), static_cast<std::streamsize>(bytes.size()));
42 if (!input) return Result<std::vector<std::uint8_t>>::failure(Diagnostic::error(
43 DiagnosticCode::Failed, "cannot read complete asset input", path.string(), {}, "cmd.asset"));
44 return Result<std::vector<std::uint8_t>>::success(std::move(bytes));
45}

72Result<asset_import::UnitySourceFiles> readTree(const std::filesystem::path& root, std::uint64_t maximumBytes,
73 std::uint64_t maximumFiles = 200000) {
74 std::error_code ec;
75 if (!std::filesystem::is_directory(root, ec))
76 return Result<asset_import::UnitySourceFiles>::failure(
77 Diagnostic::error(DiagnosticCode::NotFound, "asset project root is not a readable directory", root.string(),
78 {}, "cmd.asset"));
79 asset_import::UnitySourceFiles files;
80 std::uint64_t total = 0;
81 for (std::filesystem::recursive_directory_iterator it(root, std::filesystem::directory_options::none, ec), end;
82 !ec && it != end; it.increment(ec)) {
83 if (it->is_symlink(ec) || ec)
84 return Result<asset_import::UnitySourceFiles>::failure(
85 Diagnostic::error(DiagnosticCode::InvalidArgument, "asset source links cannot be followed",
86 it->path().string(), {}, "cmd.asset"));
87 const bool regular = it->is_regular_file(ec);
88 if (ec) break;
89 if (!regular) continue;
90 const auto size = it->file_size(ec);
91 if (ec) break;
92 if (size > maximumBytes || total > maximumBytes - size || files.size() >= maximumFiles)
93 return Result<asset_import::UnitySourceFiles>::failure(
94 Diagnostic::error(DiagnosticCode::InvalidArgument, "asset project exceeds source budget",
95 it->path().string(), {}, "cmd.asset"));
96 auto bytes = readFile(it->path(), maximumBytes);
97 if (!bytes) return Result<asset_import::UnitySourceFiles>::failure(bytes.status());
98 const auto utf8Path = it->path().lexically_relative(root).generic_u8string();
99 const std::string relative(reinterpret_cast<const char*>(utf8Path.data()), utf8Path.size());
100 if (relative.empty())
101 return Result<asset_import::UnitySourceFiles>::failure(Diagnostic::error(
102 DiagnosticCode::Failed, "cannot resolve asset path", it->path().string(), {}, "cmd.asset"));
103 total += size;
104 files.emplace(relative, std::move(bytes).takeValue());
105 }
106 if (ec)
107 return Result<asset_import::UnitySourceFiles>::failure(Diagnostic::error(
108 DiagnosticCode::Failed, "cannot traverse asset project: " + ec.message(), root.string(), {}, "cmd.asset"));
109 return Result<asset_import::UnitySourceFiles>::success(std::move(files));
110}

112Result<asset_import::UnitySourceFiles> readUnityInput(const std::string& input,
113 const asset_import::AssetImportLimits& limits) {
114 std::error_code ec;
115 const auto sourcePath = std::filesystem::path(std::u8string(input.begin(), input.end()));
116 if (std::filesystem::is_regular_file(sourcePath, ec)) {
117 auto bytes = readFile(sourcePath, limits.maximumSourceBytes);
118 if (!bytes) return Result<asset_import::UnitySourceFiles>::failure(bytes.status());
119 return asset_import::readUnityPackage(bytes.value(), limits);
120 }
121 const auto assetsRoot = sourcePath / "Assets";
122 if (!std::filesystem::is_directory(assetsRoot, ec))
123 return readTree(sourcePath, limits.maximumSourceBytes, std::uint64_t(limits.maximumAssets) * 2);
124 auto files = readTree(assetsRoot, limits.maximumSourceBytes, std::uint64_t(limits.maximumAssets) * 2);
125 if (!files) return files;
126 asset_import::UnitySourceFiles prefixed;
127 for (auto& [path, bytes] : files.value()) prefixed.emplace("Assets/" + path, std::move(bytes));
128 return Result<asset_import::UnitySourceFiles>::success(std::move(prefixed));
129}

221 int runScan() {
222 auto files = readUnityInput(input, {});
223 if (!files) {
224 printFailure(files.status());
225 return 2;
226 }
227 auto index = asset_import::indexUnitySources(files.value());
228 if (!index) {
229 printFailure(index.status());
230 return 2;
231 }
232 Value::Array assets, findings;
233 for (const auto& entry : index.value().assets) {
234 Value::Array references;
235 for (const auto& reference : entry.references)
236 references.emplace_back(
237 Value::Object{{"guid", Value(reference.guid)}, {"fileID", Value(reference.fileId)}});
238 assets.emplace_back(Value::Object{{"path", Value(entry.path)},
239 {"guid", Value(entry.guid)},
240 {"kind", Value(std::string(unityKindName(entry.kind)))},
241 {"importer", Value(entry.importer)},
242 {"references", Value(std::move(references))}});
243 }

246 {"feature", Value(finding.feature)},
247 {"message", Value(finding.message)}});
248 auto report = Value(Value::Object{{"schema", Value("eve.unity-source-index")},
249 {"schemaVersion", Value(std::int64_t(1))},
250 {"assets", Value(std::move(assets))},
251 {"findings", Value(std::move(findings))}})
252 .toJson();
253 if (!report) {
254 printFailure(report.status());
255 return 2;
256 }
257 std::cout << report.value() << "\n";
258 return 0;

261 int runImport() {
262 auto identity = packageIdentity(packageId, packageName, packageVersion);
263 if (!identity) { printFailure(identity.status()); return 2; }
264 Result<PreparedAssetImport> prepared = Result<PreparedAssetImport>::failure(
265 Diagnostic::error(DiagnosticCode::Unsupported, "unknown importer", from, {}, "cmd.asset"));
266 asset_import::AssetImportLimits limits;
267 if (from == "shader") {
268 auto source = readFile(input, 8 * 1024 * 1024);
269 if (!source) {
270 prepared.ignore();
271 printFailure(source.status());
272 return 2;
273 }
274 prepared.ignore();
275 prepared = asset_import::prepareShaderImport(
276 identity.value(),
277 std::string_view(reinterpret_cast<const char*>(source.value().data()), source.value().size()));
278 } else if (from == "image" || from == "gltf") {
279 auto source = readFile(input, limits.maximumSourceBytes);
280 if (!source) { prepared.ignore(); printFailure(source.status()); return 2; }
281 if (from == "image") {
282 if (colorSpace != "srgb" && colorSpace != "linear") {
283 prepared.ignore(); std::cerr << "--color-space must be srgb or linear\n"; return 2;
284 }
285 prepared.ignore();
286 prepared = asset_import::prepareImageImport(
287 {identity.value(), std::filesystem::path(input).filename().string(),
288 std::move(source).takeValue(),
289 colorSpace == "srgb" ? asset_import::ImageColorSpace::Srgb
290 : asset_import::ImageColorSpace::Linear,
291 usage, limits});
292 } else {
293 std::map<std::string, std::vector<std::uint8_t>> external;
294 std::error_code ec;
295 const auto parent = std::filesystem::path(input).parent_path();
296 for (std::filesystem::directory_iterator it(parent.empty() ? "." : parent, ec), end;
297 !ec && it != end; it.increment(ec)) {
298 if (!it->is_regular_file() || it->path() == std::filesystem::path(input)) continue;
299 auto value = readFile(it->path(), limits.maximumSourceBytes);
300 if (value) external.emplace(it->path().filename().generic_string(), std::move(value).takeValue());
301 }
302 prepared.ignore();
303 prepared = asset_import::prepareGltfImport(
304 {identity.value(), std::filesystem::path(input).filename().string(), std::move(source).takeValue(),
305 std::move(external), limits,
306 strict ? asset_import::GltfImportMode::Strict : asset_import::GltfImportMode::Permissive});
307 }
308 } else if (from == "unity" || from == "ue5") {
309 auto files = from == "unity" ? readUnityInput(input, limits) : readTree(input, limits.maximumSourceBytes);
310 if (!files) { prepared.ignore(); printFailure(files.status()); return 2; }
311 prepared.ignore();
312 if (from == "unity")
313 prepared = asset_import::prepareUnityProjectImport(
314 {identity.value(), std::move(files).takeValue(), terrain, prefab, limits});
315 else
316 prepared = asset_import::prepareUnrealM4Import(
317 {identity.value(), std::move(files).takeValue(), descriptor, limits});
318 }
319 if (!prepared) { printFailure(prepared.status()); return 2; }
320 asset::AtomicAssetPackageStore store;
321 auto receipt = store.publishEva(output, prepared.value().manifest, prepared.value().entries);
322 if (!receipt) { printFailure(receipt.status()); return 3; }
323 std::cout << "published " << receipt.value().destination << " package="
324 << receipt.value().packageId.format() << " bytes=" << receipt.value().byteSize << "\n";
325 for (const auto& finding : prepared.value().findings)
326 std::cout << (finding.severity == asset_import::ImportSeverity::Warning ? "warning " : "finding ")
327 << static_cast<unsigned>(finding.disposition) << " " << finding.feature << " " << finding.message
328 << "\n";
329 return 0;
330 }

332 int runValidate(bool inspect) {
333 asset::EvaArchiveLimits evaLimits;
334 asset::EvpackLimits packLimits;

349 << " name=" << parsed.value().manifest.packageName

351 << " entries=" << parsed.value().entries.size() << "\n";
352 }
353 if (!inspect) std::cout << "valid " << input << "\n";
354 return 0;

411 int runDiff() {
412 asset::EvaArchiveLimits limits;
413 auto beforeBytes = readFile(input, limits.maximumArchiveBytes);
414 auto afterBytes = readFile(secondInput, limits.maximumArchiveBytes);
415 if (!beforeBytes) { printFailure(beforeBytes.status()); return 2; }
416 if (!afterBytes) { printFailure(afterBytes.status()); return 2; }
417 auto before = asset::parseEvaArchive(beforeBytes.value(), limits);
418 auto after = asset::parseEvaArchive(afterBytes.value(), limits);
419 if (!before) { printFailure(before.status()); return 2; }
420 if (!after) { printFailure(after.status()); return 2; }
421 auto difference = asset::diffEvaManifests(before.value().manifest, after.value().manifest);
422 if (!difference) { printFailure(difference.status()); return 3; }
423 for (const auto& change : difference.value().assets) {
424 const char* kind = change.kind == asset::EvaAssetChangeKind::Added ? "added" :
425 change.kind == asset::EvaAssetChangeKind::Removed ? "removed" : "changed";
426 std::cout << kind << " asset://" << change.assetId.format();
427 if (!change.beforeType.empty()) std::cout << " before=" << change.beforeType << "/" << change.beforeVersion.value();
428 if (!change.afterType.empty()) std::cout << " after=" << change.afterType << "/" << change.afterVersion.value();
429 std::cout << "\n";
430 }
431 std::cout << "dependencies +" << difference.value().addedDependencies << " -"
432 << difference.value().removedDependencies << " entrypoints="
433 << (difference.value().entrypointsChanged ? "changed" : "same") << "\n";
434 return 0;
435 }
