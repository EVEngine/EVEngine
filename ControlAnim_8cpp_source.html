132void ControlAnim::updateUnchecked(float dt) {
133 if (dt <= 0.f) return;
134 const float omega = hzToOmega(frequencyHz_);
135
136 for (auto &kv : channels_) {
137 Channel &ch = kv.second;
138 float xd = 0.f;
139 if (ch.hasXd) {
140 xd = ch.xdExplicit;
141 } else if (ch.hasPrevX && dt > 1e-8f) {
142 xd = (ch.x - ch.xp) / dt;
143 }
144 ch.xp = ch.x;
145 ch.hasPrevX = true;
146
147 switch (integrator_) {
148 case Integrator::SecondOrder: stepSecondOrder(dt, ch.x, xd, coeffs_, ch.y, ch.yd); break;
149 case Integrator::Spring: stepDampedSpring(dt, ch.x, omega, dampingZeta_, ch.y, ch.yd); break;
150 case Integrator::Pd: stepPd(dt, ch.x, xd, kp_, kd_, ch.y, ch.yd); break;
151 }
152 }

252void ControlPose::readTargetIntoState(const AnimPose *target, bool snap) {
253 if (!target) return;
254 const int n = target->getBoneCount();
255 ensureBones(n);
256 for (int i = 0; i < n; ++i) {
257 const TransformTRS &t = target->local(i);
258 BoneState &b = bones_[static_cast<size_t>(i)];
259
260 auto assign = [snap](ScalarState &s, float value) {
261 s.x = value;
262 if (!s.hasPrev || snap) {
263 s.y = value;
264 s.yd = 0.f;
265 s.xp = value;
266 s.hasPrev = true;
267 }
268 };
269
270 assign(b.px, t.px);
271 assign(b.py, t.py);
272 assign(b.pz, t.pz);
273
274 // Shortest-path quaternion: flip target if needed relative to current.
275 float tqx = t.qx, tqy = t.qy, tqz = t.qz, tqw = t.qw;
276 if (b.qw.hasPrev) {
277 const float dot = b.qx.y * tqx + b.qy.y * tqy + b.qz.y * tqz + b.qw.y * tqw;
278 if (dot < 0.f) {
279 tqx = -tqx;
280 tqy = -tqy;
281 tqz = -tqz;
282 tqw = -tqw;
283 }
284 }
285 assign(b.qx, tqx);
286 assign(b.qy, tqy);
287 assign(b.qz, tqz);
288 assign(b.qw, tqw);
289
290 assign(b.sx, t.sx);
291 assign(b.sy, t.sy);
292 assign(b.sz, t.sz);
293 }

328void ControlPose::writePoseFromState() {
329 const int n = static_cast<int>(bones_.size());
330 pose_.resize(n);
331 for (int i = 0; i < n; ++i) {
332 const BoneState &b = bones_[static_cast<size_t>(i)];
333 TransformTRS trs;
334 trs.px = b.px.y;
335 trs.py = b.py.y;
336 trs.pz = b.pz.y;
337 trs.qx = b.qx.y;
338 trs.qy = b.qy.y;
339 trs.qz = b.qz.y;
340 trs.qw = b.qw.y;
341 trs.sx = b.sx.y;
342 trs.sy = b.sy.y;
343 trs.sz = b.sz.y;
344 trs.normalizeRotation();
345
346 if (b.weight < 1.f - 1e-6f) {
347 // Blend dynamics output toward hard target.
348 TransformTRS hard = target_.local(i);
349 // Shortest path for hard target vs dynamics result.
350 const float dot = trs.qx * hard.qx + trs.qy * hard.qy + trs.qz * hard.qz + trs.qw * hard.qw;
351 if (dot < 0.f) {
352 hard.qx = -hard.qx;
353 hard.qy = -hard.qy;
354 hard.qz = -hard.qz;
355 hard.qw = -hard.qw;
356 }
357 trs = blendTRS(hard, trs, b.weight);
358 }
359 pose_.local(i) = trs;
360 }

363void ControlPose::updateUnchecked(float dt) {
364 if (dt <= 0.f || !hasTarget_) return;
365
366 // Refresh targets from stored target_ (weights / integrator may have changed).
367 const int n = target_.getBoneCount();
368 ensureBones(n);
369 for (int i = 0; i < n; ++i) {
370 const TransformTRS &t = target_.local(i);
371 BoneState &b = bones_[static_cast<size_t>(i)];
372
373 auto prep = [](ScalarState &s, float value) {
374 if (!s.hasPrev) {
375 s.y = value;
376 s.yd = 0.f;
377 s.xp = value;
378 s.hasPrev = true;
379 }
380 s.x = value;
381 };
382
383 prep(b.px, t.px);
384 prep(b.py, t.py);
385 prep(b.pz, t.pz);
386
387 float tqx = t.qx, tqy = t.qy, tqz = t.qz, tqw = t.qw;
388 const float dot = b.qx.y * tqx + b.qy.y * tqy + b.qz.y * tqz + b.qw.y * tqw;
389 if (dot < 0.f) {
390 tqx = -tqx;
391 tqy = -tqy;
392 tqz = -tqz;
393 tqw = -tqw;
394 }
395 prep(b.qx, tqx);
396 prep(b.qy, tqy);
397 prep(b.qz, tqz);
398 prep(b.qw, tqw);
399
400 prep(b.sx, t.sx);
401 prep(b.sy, t.sy);
402 prep(b.sz, t.sz);
403
404 auto advance = [&](ScalarState &s) {
405 float xd = 0.f;
406 if (s.hasPrev && dt > 1e-8f) xd = (s.x - s.xp) / dt;
407 s.xp = s.x;
408 stepScalar(dt, xd, s);
409 };
410
411 if (b.weight <= 1e-6f) {
412 // Hard follow
413 auto hard = [](ScalarState &s) {
414 s.y = s.x;
415 s.yd = 0.f;
416 s.xp = s.x;
417 };
418 hard(b.px);
419 hard(b.py);
420 hard(b.pz);
421 hard(b.qx);
422 hard(b.qy);
423 hard(b.qz);
424 hard(b.qw);
425 hard(b.sx);
426 hard(b.sy);
427 hard(b.sz);
428 } else {
429 advance(b.px);
430 advance(b.py);
431 advance(b.pz);
432 advance(b.qx);
433 advance(b.qy);
434 advance(b.qz);
435 advance(b.qw);
436 // Renormalize quaternion state after component integration.
437 {
438 float len = std::sqrt(b.qx.y * b.qx.y + b.qy.y * b.qy.y + b.qz.y * b.qz.y + b.qw.y * b.qw.y);
439 if (len > 1e-8f) {
440 const float inv = 1.f / len;
441 b.qx.y *= inv;
442 b.qy.y *= inv;
443 b.qz.y *= inv;
444 b.qw.y *= inv;
445 } else {
446 b.qx.y = 0.f;
447 b.qy.y = 0.f;
448 b.qz.y = 0.f;
449 b.qw.y = 1.f;
450 }
451 }
452 advance(b.sx);
453 advance(b.sy);
454 advance(b.sz);
455 }
456 }
457 writePoseFromState();
458}
