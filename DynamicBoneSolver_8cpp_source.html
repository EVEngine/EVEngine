25Quat fromTo(float ax, float ay, float az, float bx, float by, float bz) {
26 const float al=length(ax,ay,az), bl=length(bx,by,bz);
27 if (al<=kEpsilon || bl<=kEpsilon) return {};

240 const int steps=std::clamp(static_cast<int>(std::ceil(dt/preferred)),1,8);

242 for (auto &chain:chains_) {
