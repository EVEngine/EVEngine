66bool assign(UiWidgetSnapshot& widget, const PropertyPath& path, const EditorValue& value) {
67 if (path == PropertyPath("content.fontAsset") || path == PropertyPath("content.textureAsset")) {
68 const auto* asset = value.getIf<std::string>(); if (!asset) return false;
69 if (path == PropertyPath("content.fontAsset")) widget.content.fontAsset = *asset;
70 else widget.content.textureAsset = *asset;
71 return true;
72 }
73 if (path == PropertyPath("content.fontSize")) {
74 const auto* number = value.getIf<double>();
75 if (!number || *number <= 0.0 || *number > 512.0) return false;
76 widget.content.fontSize = *number; return true;
77 }
78 if (path == PropertyPath("content.clip")) {
79 const auto* flag = value.getIf<bool>(); if (!flag) return false;
80 widget.content.clip = *flag; return true;
81 }
82 if (path == PropertyPath("content.horizontalAlign") ||
83 path == PropertyPath("content.verticalAlign") || path == PropertyPath("content.imageFit")) {
84 const auto* text = value.getIf<std::string>(); if (!text) return false;
85 if (path == PropertyPath("content.imageFit")) {
86 if (*text != "stretch" && *text != "contain" && *text != "cover") return false;
87 widget.content.imageFit = *text;
88 } else {
89 if (*text != "start" && *text != "center" && *text != "end") return false;
90 if (path == PropertyPath("content.horizontalAlign")) widget.content.horizontalAlign = *text;
91 else widget.content.verticalAlign = *text;
92 }
93 return true;
94 }
95 if (path == PropertyPath("widget.name") || path == PropertyPath("widget.text")) {
96 const auto* text = value.getIf<std::string>();
97 if (!text || (path == PropertyPath("widget.name") && text->empty())) return false;
98 if (path == PropertyPath("widget.name")) widget.name = *text;
99 else widget.text = *text;
100 return true;
101 }
102 if (path == PropertyPath("widget.visible") || path == PropertyPath("widget.enabled")) {
103 const auto* flag = value.getIf<bool>();
104 if (!flag) return false;
105 if (path == PropertyPath("widget.visible")) widget.visible = *flag;
106 else widget.enabled = *flag;
107 return true;
108 }
109 if (path == PropertyPath("style.cornerRadius") || path == PropertyPath("style.gap") ||
110 path == PropertyPath("style.flexGrow")) {
111 const auto* number = value.getIf<double>();
112 if (!number || *number < 0.0) return false;
113 if (path == PropertyPath("style.cornerRadius")) widget.style.cornerRadius = *number;
114 else if (path == PropertyPath("style.gap")) widget.style.gap = *number;
115 else widget.style.flexGrow = *number;
116 return true;
117 }
118 if (path == PropertyPath("style.direction") || path == PropertyPath("style.align") ||
119 path == PropertyPath("style.justify")) {
120 const auto* text = value.getIf<std::string>();
121 if (!text) return false;
122 if (path == PropertyPath("style.direction")) {
123 if (*text != "row" && *text != "column") return false;
124 widget.style.direction = *text;
125 } else if (path == PropertyPath("style.align")) {
126 if (*text != "start" && *text != "center" && *text != "end" && *text != "stretch") return false;
127 widget.style.align = *text;
128 } else {
129 if (*text != "start" && *text != "center" && *text != "end" &&
130 *text != "space-between" && *text != "space-around") return false;
131 widget.style.justify = *text;
132 }
133 return true;
134 }
135 const auto* tuple = value.getIf<EditorValue::Array>();
136 if ((path == PropertyPath("style.margin") || path == PropertyPath("style.padding") ||
137 path == PropertyPath("style.tint") || path == PropertyPath("content.textColor"))) {
138 if (!tuple || tuple->size() != 4) return false;
139 double values[4];
140 for (int i = 0; i < 4; ++i) { const auto* number = (*tuple)[i].getIf<double>(); if (!number || *number < 0.0) return false; values[i] = *number; }
141 if (path == PropertyPath("style.tint") || path == PropertyPath("content.textColor")) {
142 for (double number : values) if (number > 1.0) return false;
143 if (path == PropertyPath("style.tint")) {
144 widget.style.tintR = values[0]; widget.style.tintG = values[1]; widget.style.tintB = values[2]; widget.style.tintA = values[3];
145 } else {
146 widget.content.textR = values[0]; widget.content.textG = values[1]; widget.content.textB = values[2]; widget.content.textA = values[3];
147 }
148 } else if (path == PropertyPath("style.margin")) {
149 widget.style.marginLeft = values[0]; widget.style.marginTop = values[1]; widget.style.marginRight = values[2]; widget.style.marginBottom = values[3];
150 } else {
151 widget.style.paddingLeft = values[0]; widget.style.paddingTop = values[1]; widget.style.paddingRight = values[2]; widget.style.paddingBottom = values[3];
152 }
153 return true;
154 }
155 if (!tuple || tuple->size() != 2) return false;
156 const auto* first = (*tuple)[0].getIf<double>();
157 const auto* second = (*tuple)[1].getIf<double>();
158 if (!first || !second) return false;
159 if (path == PropertyPath("layout.position")) {
160 widget.layout.x = *first;
161 widget.layout.y = *second;
162 } else if (path == PropertyPath("layout.size")) {
163 if (*first < 0.0 || *second < 0.0) return false;
164 widget.layout.width = *first;
165 widget.layout.height = *second;
166 } else if (path == PropertyPath("layout.anchor")) {
167 if (*first < 0.0 || *first > 1.0 || *second < 0.0 || *second > 1.0) return false;
168 widget.layout.anchorX = *first;
169 widget.layout.anchorY = *second;
170 } else if (path == PropertyPath("layout.pivot")) {
171 if (*first < 0.0 || *first > 1.0 || *second < 0.0 || *second > 1.0) return false;
172 widget.layout.pivotX = *first;
173 widget.layout.pivotY = *second;
174 } else {
175 return false;
176 }
177 return true;
178}

235PropertyReadResult UiDocumentTarget::read(const SelectionSnapshot& selection,
236 const PropertyPath& path) const {
237 std::optional<EditorValue> common;
238 for (const SelectionItem& item : selection.items) {
239 if (item.target != TargetId(id_)) return {};
240 const auto found = widgets_.find(ObjectId(item.item.value()));
241 if (found == widgets_.end()) return {};
242 EditorValue value = component(found->second, path);
243 if (value.type() == EditorValue::Type::Null) return {};
244 if (!common) common = value;
245 else if (*common != value) return {PropertyReadState::Mixed, {}, {}};
246 }
247 return common ? PropertyReadResult{PropertyReadState::Value, *common, {}} : PropertyReadResult{};
248}

250EditorResult<DomainOperation> UiDocumentTarget::makeSet(const SelectionSnapshot& selection,
251 const PropertyPath& path,
252 const EditorValue& value,
253 PropertySetMode mode) const {
254 if (mode == PropertySetMode::Reset) return makeReset(selection, path);
255 if (mode != PropertySetMode::Absolute || selection.items.empty())
256 return propertyError<DomainOperation>(EditorStatus::Unsupported, "editor.ui.property-mode",
257 "UI properties require a non-empty absolute assignment");
258 if (!schema(selection).find(path))
259 return propertyError<DomainOperation>(EditorStatus::Unsupported, "editor.ui.property-path",
260 "Unknown UI property: " + path.value());
261 EditorValue::Array payload;
262 EditorValue::Array inverse;
263 DomainOperation operation;
264 for (const SelectionItem& item : selection.items) {
265 if (item.target != TargetId(id_))
266 return propertyError<DomainOperation>(EditorStatus::Rejected, "editor.ui.property-target",
267 "UI selection spans another target");
268 const auto found = widgets_.find(ObjectId(item.item.value()));
269 if (found == widgets_.end())
270 return propertyError<DomainOperation>(EditorStatus::NotFound, "editor.ui.widget-not-found",
271 "Selected UI widget does not exist");
272 UiWidgetSnapshot changed = found->second;
273 if (!assign(changed, path, value))
274 return propertyError<DomainOperation>(EditorStatus::Rejected, "editor.ui.property-value",
275 "UI property value is invalid for " + path.value());
276 payload.push_back(widgetValue(changed));
277 inverse.push_back(widgetValue(found->second));
278 operation.affectedObjects.push_back({TargetId(id_), item.item.value(), 0});
279 }
280 operation.type = "ui.widget.multi-replace.v1";
281 operation.target = TargetId(id_);
282 operation.payload = EditorValue(std::move(payload));
283 operation.inverse = EditorValue(std::move(inverse));
284 operation.hasInverse = true;
285 operation.affectedProperties.push_back(path.value());
286 operation.mergeKey = "ui.selection:" + path.value();
287 return eve::editing::applied<DomainOperation>(std::move(operation));
288}
