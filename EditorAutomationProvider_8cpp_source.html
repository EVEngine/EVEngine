59EditorValue resultValue(EditorStatus status, const std::vector<EditorDiagnostic>& diagnostics) {
60 EditorValue::Object result;
61 result["status"] = statusName(status);

68EditorValue receiptValue(const EditorResult<TransactionReceipt>& result) {

132std::string EditorAutomationProvider::invoke(const std::string& operation, const std::string& requestJson) {
133 refreshProfile();
134 EditorResult<EditorValue> parsed = editorValueFromJson(requestJson.empty() ? "{}" : requestJson);
135 if (!parsed.ok() || parsed.value().type() != EditorValue::Type::Object)
136 return errorJson(EditorStatus::Rejected, "editor.automation.invalid-json", "Request must be a JSON object");
137 const auto& request = *parsed.value().getIf<EditorValue::Object>();
138 if (operation == "commands") return commandsJson();
139 if (operation == "target-create") return createTarget(request);
140 if (operation == "target-close") return closeTarget(request);
141 if (operation == "observe-start") return startObservation(request);
142 if (operation == "observe-describe") return describeObservation(request);
143 if (operation == "observe-publish") return publishObservation(request);
144 if (operation == "observe-poll") return pollObservation(request);
145 if (operation == "observe-close") return closeObservation(request);
146 if (operation == "inspect") {
147 auto result = targets_->inspect(TargetId(stringField(request, "target")));
148 lastDiagnostics_ = result.diagnostics();
149 EditorValue response = resultValue(result.code(), result.diagnostics());
150 if (result.ok()) (*response.getIf<EditorValue::Object>())["target"] = result.value();
151 return editorValueToJson(response);
152 }
153 if (operation == "plan" || operation == "execute") {
154 auto bound = bindRequestedTarget(request);
155 if (!bound.ok()) return editorValueToJson(resultValue(bound.code(), bound.diagnostics()));
156 }
157 if (operation == "plan") {
158 const std::string command = stringField(request, "command");
159 std::optional<Revision> expected;
160 if (const auto* value = integerField(request, "expectedRevision")) expected = static_cast<Revision>(*value);
161 auto result = session_.retainPlan(CommandId(command), valueField(request, "payload"),
162 CommandSource::Automation, expected);
163 lastDiagnostics_ = result.diagnostics();
164 EditorValue response = resultValue(result.code(), result.diagnostics());
165 if (result.ok()) (*response.getIf<EditorValue::Object>())["planId"] = result.value().value();
166 return editorValueToJson(response);
167 }
168 if (operation == "commit") {
169 auto result = session_.executeRetainedPlan(PlanId(stringField(request, "planId")), CommandSource::Automation);
170 lastDiagnostics_ = result.diagnostics();
171 return editorValueToJson(receiptValue(result));
172 }
173 if (operation == "execute") {
174 const CommandId command(stringField(request, "command"));
175 std::optional<EditorResult<TransactionReceipt>> result;
176 if (commands_->supportsPlanning(command)) {
177 auto planned = session_.planCommand(command, valueField(request, "payload"), CommandSource::Automation);
178 if (!planned.ok()) {
179 result.emplace(EditorResult<TransactionReceipt>::failure(planned.status()));
180 } else {
181 result.emplace(session_.executePlan(planned.value(), valueField(request, "payload"),
182 CommandSource::Automation));
183 }
184 } else {
185 result.emplace(session_.executeCommandReceipt(command, valueField(request, "payload"),
186 CommandSource::Automation));
187 }
188 lastDiagnostics_ = result->diagnostics();
189 return editorValueToJson(receiptValue(*result));
190 }
191 if (operation == "cancel") {
192 auto result = session_.cancelRetainedPlan(PlanId(stringField(request, "planId")));
193 lastDiagnostics_ = result.diagnostics();
194 return editorValueToJson(resultValue(result.code(), result.diagnostics()));
195 }
196 if (operation == "undo" || operation == "redo") {
197 const TargetId target(stringField(request, "target"));
198 if (!target.empty()) {
199 auto result = operation == "undo" ? targets_->undo(target) : targets_->redo(target);
200 lastDiagnostics_ = result.diagnostics();
201 return editorValueToJson(receiptValue(result));
202 }
203 const bool changed = operation == "undo" ? session_.transactions().undo() : session_.transactions().redo();
204 EditorValue response = resultValue(changed ? EditorStatus::Applied : EditorStatus::NoOp, {});
205 (*response.getIf<EditorValue::Object>())["changed"] = changed;
206 return editorValueToJson(response);
207 }
208 if (operation == "diagnostics")
209 return editorValueToJson(resultValue(EditorStatus::Applied, lastDiagnostics_));
210 return errorJson(EditorStatus::Unsupported, "editor.automation.unsupported-operation",
211 "Unsupported editor automation operation: " + operation);
212}

214std::string EditorAutomationProvider::startObservation(const EditorValue::Object& request) {
215 const EditorValue* descriptorValue = findField(request, "descriptor");
216 const EditorValue* eventValue = findField(request, "event");
217 const auto* descriptor = descriptorValue ? descriptorValue->getIf<EditorValue::Object>() : nullptr;
218 const auto* event = eventValue ? eventValue->getIf<EditorValue::Object>() : nullptr;
219 if (!descriptor || !event)
220 return errorJson(EditorStatus::Rejected, "editor.automation.observe-start",
221 "Observation descriptor and initial event must be JSON objects");
222
223 const std::string id = "editor.observe." + std::to_string(nextObservationSession_++);
224 auto session = std::make_unique<ObservationSession>();
225 session->descriptor = *descriptor;
226 session->distinct.reset(session->source.distinctUntilChanged());
227 ObservationSession* observed = session.get();
228 session->subscription = session->distinct->subscribe(
229 [observed](const std::string& value) { observed->pending.push_back(value); });
230 session->source.onNext(editorValueToJson(*eventValue));
231 // startObservation returns the initial event directly. Retain it only as
232 // the distinct baseline so the first poll does not deliver it twice.
233 session->pending.clear();
234 observationSessions_.emplace(id, std::move(session));
235
236 EditorValue response = resultValue(EditorStatus::Applied, {});
237 auto* object = response.getIf<EditorValue::Object>();
238 (*object)["sessionId"] = id;
239 (*object)["event"] = *eventValue;
240 return editorValueToJson(response);
241}

295std::string EditorAutomationProvider::createTarget(const EditorValue::Object& request) {
296 const TargetId target(stringField(request, "target"));
297 const std::string type = stringField(request, "type");
298 if (target.empty())
299 return errorJson(EditorStatus::Rejected, "editor.automation.target-id", "Target id must not be empty");
300 if (ownedTargets_.contains(target))
301 return errorJson(EditorStatus::Conflict, "editor.automation.target-exists", "Target already exists");
302
303 std::optional<EditorResult<AutomationOwnedTarget>> created;
304 const bool handled = eve::cap::forEachUntil<IEditorAutomationTargetFactory>(
305 [&](IEditorAutomationTargetFactory* factory) {
306 if (!factory->supports(type)) return false;
307 created.emplace(factory->create(target, type, request));
308 return true;
309 });
310 if (!handled) {
311 return errorJson(EditorStatus::Unsupported, "editor.automation.target-type",
312 "No loaded editor adapter supports target type: " + type);
313 }
314 lastDiagnostics_ = created->diagnostics();
315 if (!created->ok() || !created->value().target)
316 return editorValueToJson(resultValue(created->code(), created->diagnostics()));
317
318 auto registered = targets_->registerTarget(*created->value().target);
319 lastDiagnostics_ = registered.diagnostics();
320 if (!registered.ok()) return editorValueToJson(resultValue(registered.code(), registered.diagnostics()));
321 ownedTargets_.emplace(target, std::move(*created).takeValue());
322 EditorValue response = resultValue(EditorStatus::Applied, {});
323 auto* value = response.getIf<EditorValue::Object>();
324 (*value)["target"] = target.value();
325 (*value)["type"] = type;
326 return editorValueToJson(response);
327}

329std::string EditorAutomationProvider::closeTarget(const EditorValue::Object& request) {
330 const TargetId target(stringField(request, "target"));
331 auto found = ownedTargets_.find(target);
332 if (found == ownedTargets_.end())
333 return errorJson(EditorStatus::NotFound, "editor.automation.target-not-found",
334 "Owned automation target does not exist");
335 auto unregistered = targets_->unregisterTarget(target);
336 lastDiagnostics_ = unregistered.diagnostics();
337 if (!unregistered.ok())
338 return editorValueToJson(resultValue(unregistered.code(), unregistered.diagnostics()));
339 targetUnregistered(target);
340 ownedTargets_.erase(found);
341 return editorValueToJson(resultValue(EditorStatus::Applied, {}));
342}
