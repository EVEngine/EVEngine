55bool vec3At(const Value::Object& object, const char* name, glm::vec3& output) {
56 const auto found = object.find(name);
57 if (found == object.end()) return false;
58 const auto* values = found->second.getIf<Value::Array>();
59 if (!values || values->size() != 3) return false;
60 for (int i = 0; i < 3; ++i) {
61 if (!(*values)[i].isDouble() && !(*values)[i].isInt64()) return false;
62 output[i] = static_cast<float>((*values)[i].isDouble() ? (*values)[i].asDouble() : (*values)[i].asInt());
63 if (!std::isfinite(output[i])) return false;
64 }
65 return true;
66}

80Result<void> WaterStyleConfig::validate() const {
81 if (!color(deepColor) || !color(shallowColor) || !color(foamColor) || !color(reflectionTint))
82 return invalid("water colors must be finite and in [0, 8]", "$.colors");
83 if (!std::isfinite(waveSpeed) || waveSpeed < -20.0F || waveSpeed > 20.0F ||
84 !std::isfinite(waveAmplitude) || waveAmplitude < 0.0F || waveAmplitude > 4.0F ||
85 !std::isfinite(waveScale) || waveScale < 0.01F || waveScale > 256.0F ||
86 !std::isfinite(waveSharpness) || waveSharpness < 0.1F || waveSharpness > 8.0F ||
87 !std::isfinite(rippleAmplitude) || rippleAmplitude < 0.0F || rippleAmplitude > 4.0F ||
88 rippleCount < 0 || rippleCount > 8 || !std::isfinite(rippleInterval) || rippleInterval < 0.05F ||
89 rippleInterval > 60.0F)
90 return invalid("invalid wave or ripple parameter", "$.waves");
91 if (!std::isfinite(depthDistance) || depthDistance <= 0.0F || depthDistance > 10000.0F ||
92 !std::isfinite(opacity) || opacity < 0.0F || opacity > 1.0F)
93 return invalid("invalid depth parameter", "$.depth");
94 if (!std::isfinite(foamWidth) || foamWidth <= 0.0F || foamWidth > 1000.0F ||
95 !std::isfinite(foamSoftness) || foamSoftness < 0.0F || foamSoftness > 1.0F ||
96 !std::isfinite(foamStrength) || foamStrength < 0.0F || foamStrength > 4.0F)
97 return invalid("invalid foam parameter", "$.foam");
98 if (!std::isfinite(fresnelPower) || fresnelPower < 0.01F || fresnelPower > 32.0F ||
99 !std::isfinite(reflectionIntensity) || reflectionIntensity < 0.0F || reflectionIntensity > 4.0F ||
100 !std::isfinite(sunIntensity) || sunIntensity < 0.0F || sunIntensity > 8.0F ||
101 !std::isfinite(screenSpaceReflectionStrength) || screenSpaceReflectionStrength < 0.0F ||
102 screenSpaceReflectionStrength > 4.0F)
103 return invalid("invalid lighting parameter", "$.lighting");
104 if (!std::isfinite(refractionStrength) || refractionStrength < 0.0F || refractionStrength > 0.25F)
105 return invalid("invalid refraction strength", "$.refraction.strength");
106 if (!std::isfinite(causticsStrength) || causticsStrength < 0.0F || causticsStrength > 4.0F ||
107 !std::isfinite(causticsScale) || causticsScale < 0.01F || causticsScale > 256.0F)
108 return invalid("invalid caustics parameter", "$.caustics");
109 return Result<void>::success();
110}
