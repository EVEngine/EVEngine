9Result<void> bind(RTS& rts, transaction::AtomicBatch& batch, Faction& faction) {
10 auto account = rts.scriptAccount(faction);
11 if (!account) return batch.reject(account.status());
12 const auto handle = ecs::handle_of(&faction);
13 const auto tick = rts.scriptTick();
14 const auto expected = reinterpret_cast<std::uintptr_t>(&account.value().get());
15 auto guarded = batch.guard([&rts, handle, tick, expected]() -> Result<void> {
16 auto* faction = handle.table == ecs::current() ? dynamic_cast<Faction*>(ecs::try_get(handle)) : nullptr;
17 if (!faction || rts.scriptTick() != tick)
18 return Result<void>::failure(
19 Diagnostic::error(DiagnosticCode::StaleHandle, "batch simulation context changed", "account"));
20 auto current = rts.scriptAccount(*faction);
21 if (!current) return Result<void>::failure(current.status());
22 if (reinterpret_cast<std::uintptr_t>(&current.value().get()) != expected)
23 return Result<void>::failure(
24 Diagnostic::error(DiagnosticCode::StaleHandle, "batch account was replaced", "account"));
25 return Result<void>::success();
26 });
27 if (!guarded) return guarded;
28 return batch.bindAccount(account.value().get(), [&rts, handle](std::string_view name) -> std::optional<Value> {
29 auto* faction = handle.table == ecs::current() ? dynamic_cast<Faction*>(ecs::try_get(handle)) : nullptr;
30 if (!faction) return std::nullopt;
31 auto balance = rts.scriptResource(*faction, name);
32 if (!balance) {
33 balance.ignore("condition resource is unavailable");
34 return std::nullopt;
35 }
36 return Value(balance.value());
37 });
