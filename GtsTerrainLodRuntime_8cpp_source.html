39Result<std::uint64_t> GtsTerrainLodRuntime::replace(const GtsTerrainLodSet& lods, Procgen& procgen,
40 graphics::Graphics& gfx, float diameter, float fov, float ox,
41 float oy, float oz) {
42 std::vector<ecs::EntityHandle> candidate(size_t(lods.getTileCount()));
43 for (int i = 0; i < lods.getTileCount(); ++i) {
44 auto* tile = lods.tileAt(i);
45 if (!tile || tile->levels.empty() || tile->levels.front().empty()) continue;
46 auto* entity = graphics::Renderable3D::create();
47 if (!entity) {
48 destroy(candidate);
49 return Result<std::uint64_t>::failure(Diagnostic::error(DiagnosticCode::Failed,
50 "failed to create GTS terrain tile renderable",
51 "procgen.gtsTerrainLodRuntime"));
52 }
53 entity->setVisible(false);
54 candidate[size_t(i)] = ecs::handle_of(entity);
55 auto configured = procgen.configureGtsTerrainTileLods(lods, i, *entity, gfx, diameter, fov, ox, oy, oz);
56 if (!configured) {
57 destroy(candidate);
58 return Result<std::uint64_t>::failure(*configured.error());
59 }
60 }
61 for (auto& h : impl_->tiles)
62 if (auto* entity = resolve(h)) entity->setVisible(false);
63 for (auto& h : candidate)
64 if (auto* entity = resolve(h)) entity->setVisible(true);
65 destroy(impl_->tiles);
66 impl_->tiles = std::move(candidate);
67 ++impl_->revision;
68 if (impl_->revision == 0) ++impl_->revision;
69 return Result<std::uint64_t>::success(impl_->revision);
70}
71Result<std::uint64_t> GtsTerrainLodRuntime::replaceAndHideSource(const GtsTerrainLodSet& lods, Procgen& procgen,
72 graphics::Graphics& gfx,
73 graphics::Renderable3D& source, float diameter,
74 float fov, float ox, float oy, float oz) {
75 for (const auto& handle : impl_->tiles)
76 if (resolve(handle) == &source)
77 return Result<std::uint64_t>::failure(
78 Diagnostic::error(DiagnosticCode::InvalidArgument,
79 "GTS source terrain cannot be one of the managed LOD tiles", "sourceTerrain"));
80 const auto sourceHandle = ecs::handle_of(&source);
81 const bool sameSource = impl_->hasSource && resolve(impl_->sourceTerrain) == &source;
82 const bool enteringVisibility = sameSource ? impl_->sourceWasVisible : source.getVisible();
83 auto replaced = replace(lods, procgen, gfx, diameter, fov, ox, oy, oz);
84 if (!replaced) return replaced;
85 if (!sameSource) restoreSource(impl_->sourceTerrain, impl_->sourceWasVisible, impl_->hasSource);
86 impl_->sourceTerrain = sourceHandle;
87 impl_->sourceWasVisible = enteringVisibility;
88 impl_->hasSource = true;
89 source.setVisible(false);
90 return replaced;
91}
