132Result<void> buildGtsTerrainBaseMesh(MeshBuild& output, const Heightmap& heightmap, GtsTerrainSaveResolution resolution,
133 float sizeX, float sizeY, float sizeZ) {
134 const int exponent = static_cast<int>(resolution);
135 const int sourceWidth = heightmap.getWidth(), sourceHeight = heightmap.getHeight();
136 if (exponent < 0 || exponent > 4 || sourceWidth < 2 || sourceHeight < 2 || !std::isfinite(sizeX) || sizeX <= 0.f ||
137 !std::isfinite(sizeY) || sizeY <= 0.f || !std::isfinite(sizeZ) || sizeZ <= 0.f)
138 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
139 "GTS terrain heightmap or export dimensions are invalid",
140 "procgen.gtsTerrainLod"));
141 const int step = 1 << exponent;
142 if ((sourceWidth - 1) % step != 0 || (sourceHeight - 1) % step != 0)
143 return Result<void>::failure(
144 Diagnostic::error(DiagnosticCode::InvalidArgument,
145 "GTS terrain heightmap intervals must be divisible by the save-resolution stride",
146 "procgen.gtsTerrainLod"));
147 const int columns = (sourceWidth - 1) / step + 1, rows = (sourceHeight - 1) / step + 1;
148 const std::uint64_t vertexCount = std::uint64_t(columns) * std::uint64_t(rows);
149 const std::uint64_t indexCount = std::uint64_t(columns - 1) * std::uint64_t(rows - 1) * 6u;
150 if (vertexCount > 8u * 1024u * 1024u || indexCount > 48u * 1024u * 1024u ||
151 vertexCount > std::uint64_t(std::numeric_limits<int>::max()) ||
152 indexCount > std::uint64_t(std::numeric_limits<int>::max()))
153 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
154 "GTS terrain base mesh exceeds the native export budget",
155 "procgen.gtsTerrainLod"));
156 for (float value : heightmap.data())
157 if (!std::isfinite(value))
158 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
159 "GTS terrain heightmap contains a non-finite sample",
160 "procgen.gtsTerrainLod"));
161 MeshBuild candidate;
162 candidate.reserve(static_cast<int>(vertexCount), static_cast<int>(indexCount));
163 candidate.setActiveGroup("terrain");
164 for (int z = 0; z < rows; ++z)
165 for (int x = 0; x < columns; ++x) {
166 const int gx = x * step, gz = z * step;
167 candidate.addVertex(sizeX * float(gx) / float(sourceWidth - 1), heightmap.height(gx, gz) * sizeY,
168 sizeZ * float(gz) / float(sourceHeight - 1), 0.f, 0.f, 0.f,
169 float(gx) / float(sourceWidth - 1), float(gz) / float(sourceHeight - 1));
170 }
171 for (int z = 0; z + 1 < rows; ++z)
172 for (int x = 0; x + 1 < columns; ++x) {
173 const std::uint32_t a = std::uint32_t(z * columns + x), b = a + 1, c = a + std::uint32_t(columns),
174 d = c + 1;
175 candidate.addTriangle(a, c, b);
176 candidate.addTriangle(b, c, d);
177 }
178 auto& normals = candidate.normals();
179 const auto& positions = candidate.positions();
180 const auto& indices = candidate.indices();
181 for (size_t i = 0; i < indices.size(); i += 3) {
182 const auto a = indices[i], b = indices[i + 1], c = indices[i + 2];
183 const float abx = positions[b * 3] - positions[a * 3], aby = positions[b * 3 + 1] - positions[a * 3 + 1],
184 abz = positions[b * 3 + 2] - positions[a * 3 + 2];
185 const float acx = positions[c * 3] - positions[a * 3], acy = positions[c * 3 + 1] - positions[a * 3 + 1],
186 acz = positions[c * 3 + 2] - positions[a * 3 + 2];
187 const float nx = aby * acz - abz * acy, ny = abz * acx - abx * acz, nz = abx * acy - aby * acx;
188 for (auto vertex : {a, b, c}) {
189 normals[vertex * 3] += nx;
190 normals[vertex * 3 + 1] += ny;
191 normals[vertex * 3 + 2] += nz;
192 }
193 }
194 for (size_t i = 0; i < normals.size(); i += 3) {
195 const float length =
196 std::sqrt(normals[i] * normals[i] + normals[i + 1] * normals[i + 1] + normals[i + 2] * normals[i + 2]);
197 if (!(length > 0.f) || !std::isfinite(length))
198 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
199 "GTS terrain base mesh contains a degenerate vertex normal",
200 "procgen.gtsTerrainLod"));
201 normals[i] /= length;
202 normals[i + 1] /= length;
203 normals[i + 2] /= length;
204 }
205 candidate.setMeta("kind", "gts.terrain.base");
206 candidate.setMeta("saveResolution", std::to_string(exponent));
207 candidate.setMeta("sourceWidth", std::to_string(sourceWidth));
208 candidate.setMeta("sourceHeight", std::to_string(sourceHeight));
209 output = std::move(candidate);
210 return Result<void>::success();
211}

370Result<GtsTerrainLodSet> buildGtsTerrainLods(const MeshBuild& source, int xSplits, int zSplits, GtsMeshPivot pivot,
371 const std::vector<GtsTerrainLodLevelSettings>& levels) {
372 if (levels.empty())
373 return Result<GtsTerrainLodSet>::failure(Diagnostic::error(
374 DiagnosticCode::InvalidArgument, "at least one terrain LOD level is required", "procgen.gtsTerrainLod"));
375 float previousTransition = 1.0f;
376 for (const auto& level : levels) {
377 if (!std::isfinite(level.quality) || level.quality < 0.0f || level.quality > 1.0f ||
378 !std::isfinite(level.screenRelativeTransitionHeight) || level.screenRelativeTransitionHeight < 0.0f ||
379 level.screenRelativeTransitionHeight > 1.0f || level.screenRelativeTransitionHeight >= previousTransition) {
380 return Result<GtsTerrainLodSet>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
381 "LOD quality or transition sequence is invalid",
382 "procgen.gtsTerrainLod"));
383 }
384 previousTransition = level.screenRelativeTransitionHeight;
385 }
386
387 auto split = splitGtsMesh(source, xSplits, zSplits, pivot);
388 if (!split) return Result<GtsTerrainLodSet>::failure(*split.error());
389
390 GtsTerrainLodSet result;
391 result.columns_ = split.value().getColumnCount();
392 result.rows_ = split.value().getRowCount();
393 result.settings_ = levels;
394 result.tiles_.reserve(split.value().getTileCount());
395 for (int tileIndex = 0; tileIndex < split.value().getTileCount(); ++tileIndex) {
396 const auto* sourceTile = split.value().tileAt(tileIndex);
397 GtsTerrainLodTile tile;
398 tile.offsetX = sourceTile->offsetX;
399 tile.offsetZ = sourceTile->offsetZ;
400 tile.levels.reserve(levels.size());
401 MeshBuild previous = sourceTile->mesh;
402 for (const auto& level : levels) {
403 if (previous.empty()) {
404 tile.levels.emplace_back();
405 continue;
406 }
407 MeshBuild simplified;
408 auto simplification = simplifyGtsMesh(simplified, previous, level.quality, level.simplification);
409 if (!simplification) return Result<GtsTerrainLodSet>::failure(*simplification.error());
410 tile.levels.push_back(simplified);
411 previous = std::move(simplified);
412 }
413 result.tiles_.push_back(std::move(tile));
414 }
415 return Result<GtsTerrainLodSet>::success(std::move(result));
416}

511Value encodeMesh(const MeshBuild& m) {
512 Value::Array p, n, u, c, i, g;
513 p.reserve(m.positions().size());
514 n.reserve(m.normals().size());
515 u.reserve(m.uvs().size());
516 c.reserve(m.colors().size());
517 i.reserve(m.indices().size());
518 g.reserve(m.getIndexCount() / 3);
519 for (float x : m.positions()) p.emplace_back(x);
520 for (float x : m.normals()) n.emplace_back(x);
521 for (float x : m.uvs()) u.emplace_back(x);
522 for (float x : m.colors()) c.emplace_back(x);
523 for (auto x : m.indices()) i.emplace_back(int64_t(x));
524 for (int t = 0; t < m.getIndexCount() / 3; ++t) {
525 int group = m.getTriangleGroup(t);
526 g.emplace_back(group >= 0 ? m.getGroupName(group) : std::string{});
527 }
528 return Value::Object{{"colors", std::move(c)}, {"groups", std::move(g)}, {"indices", std::move(i)},
529 {"normals", std::move(n)}, {"positions", std::move(p)}, {"uvs", std::move(u)}};

531bool decodeFloatArray(const Value& v, std::vector<float>& out, std::size_t max) {
532 auto* a = v.getIf<Value::Array>();
533 if (!a || a->size() > max) return false;
534 out.reserve(a->size());
535 for (auto& x : *a) {
536 auto* d = num(x);
537 if (!d || !std::isfinite(*d) || *d < -std::numeric_limits<float>::max() ||
538 *d > std::numeric_limits<float>::max())
539 return false;
540 out.push_back(float(*d));
541 }
542 return true;
543}
544bool decodeMesh(const Value& v, MeshBuild& out, bool legacy) {
545 auto* o = v.getIf<Value::Object>();
546 if (!o || !(legacy ? exact(*o, {"groups", "indices", "normals", "positions", "uvs"})
547 : exact(*o, {"colors", "groups", "indices", "normals", "positions", "uvs"})))
548 return false;
549 std::vector<float> p, n, u, c;
550 if (!decodeFloatArray(o->at("positions"), p, kMaxVertices * 3) ||
551 !decodeFloatArray(o->at("normals"), n, kMaxVertices * 3) ||
552 !decodeFloatArray(o->at("uvs"), u, kMaxVertices * 2) ||
553 (!legacy && !decodeFloatArray(o->at("colors"), c, kMaxVertices * 4)) || p.size() % 3 || u.size() % 2 ||
554 n.size() != p.size() || u.size() / 2 != p.size() / 3 || (!c.empty() && c.size() / 4 != p.size() / 3))
555 return false;
556 auto* indices = o->at("indices").getIf<Value::Array>();
557 auto* groups = o->at("groups").getIf<Value::Array>();
558 if (!indices || indices->size() % 3 || indices->size() > kMaxVertices * 6 || !groups ||
559 groups->size() != indices->size() / 3)
560 return false;
561 MeshBuild m;
562 for (size_t x = 0; x < p.size() / 3; ++x)
563 m.addVertex(p[x * 3], p[x * 3 + 1], p[x * 3 + 2], n[x * 3], n[x * 3 + 1], n[x * 3 + 2], u[x * 2], u[x * 2 + 1]);
564 if (!c.empty() && !m.setVertexColors(std::move(c)).ok()) return false;
565 for (size_t t = 0; t < groups->size(); ++t) {
566 auto* name = (*groups)[t].getIf<std::string>();
567 if (!name || name->size() > 1024) return false;
568 uint32_t ids[3];
569 for (int corner = 0; corner < 3; ++corner) {
570 auto* id = (*indices)[t * 3 + corner].getIf<int64_t>();
571 if (!id || *id < 0 || *id >= int64_t(p.size() / 3)) return false;
572 ids[corner] = uint32_t(*id);
573 }
574 if (ids[0] == ids[1] || ids[1] == ids[2] || ids[0] == ids[2]) return false;
575 m.setActiveGroup(*name);
576 m.addTriangle(ids[0], ids[1], ids[2]);
577 }
578 out = std::move(m);
579 return true;
580}

1336Result<std::string> encodeGtsMaskedTerrainObj(const Heightmap& heightmap, const Heightmap& maskmap,
1337 GtsTerrainSaveResolution resolution, float sizeX, float sizeY,
1338 float sizeZ, GtsTerrainObjFaceMode faceMode, float threshold,
1339 bool invert) {
1340 if (!enumRange(int(faceMode), 1) || !std::isfinite(threshold) || threshold < 0.f || threshold > 1.f)
1341 return Result<std::string>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
1342 "GTS masked terrain OBJ settings are invalid",
1343 "procgen.gtsTerrainExportSettings"));
1344 MeshBuild validated;
1345 auto built = buildGtsTerrainBaseMesh(validated, heightmap, resolution, sizeX, sizeY, sizeZ);
1346 if (!built) return Result<std::string>::failure(built.status());
1347 if (maskmap.getWidth() < 1 || maskmap.getHeight() < 1)
1348 return Result<std::string>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
1349 "GTS masked terrain OBJ mask is empty",
1350 "procgen.gtsTerrainExportSettings"));
1351 const int terrainResolution = heightmap.getWidth(), stride = 1 << int(resolution);
1352 std::vector<int> vertexIndices(std::size_t(terrainResolution) * std::size_t(terrainResolution), -1);
1353 struct Vertex {
1354 float x, y, z, u, v;
1355 };
1356 std::vector<Vertex> vertices;
1357 std::vector<int> indices;
1358 auto sampleNormalized = [](const Heightmap& map, float x, float z) {
1359 return map.sampleBilinear(x * float(map.getWidth()), z * float(map.getHeight()));
1360 };
1361 auto vertexIndex = [&](int x, int z, float height) {
1362 auto& index = vertexIndices[std::size_t(z) * std::size_t(terrainResolution) + std::size_t(x)];
1363 if (index < 0) {
1364 index = int(vertices.size());
1365 vertices.push_back({-float(x) * sizeX / float(terrainResolution - 1), height * sizeY,
1366 float(z) * sizeZ / float(terrainResolution - 1), float(x) / float(terrainResolution),
1367 float(z) / float(terrainResolution)});
1368 }
1369 return index;
1370 };
1371 auto addTriangle = [&](int a, int b, int c) {
1372 indices.push_back(c);
1373 indices.push_back(b);
1374 indices.push_back(a);
1375 };
1376 auto addQuad = [&](int a, int b, int c, int d) {
1377 indices.push_back(a);
1378 indices.push_back(c);
1379 indices.push_back(d);
1380 indices.push_back(b);
