57void ensureScriptEcsInTable(ssq::Table& table) {
58 HSQUIRRELVM vm = table.getHandle();
59 sq_pushobject(vm, table.getRaw());
60 const bool hasSystem = eveRawHas(vm, -1, "System");
61 const bool ready = eveRawHas(vm, -1, "Component") && eveRawHas(vm, -1, "Entity") && hasSystem;
62 if (!ready && hasSystem) {
63 // The host-information module historically owns eve.System until the
64 // script ECS is injected. Remove that one namespace collision so the
65 // canonical script System class can be created exactly once.
66 sq_pushstring(vm, "System", -1);
67 sq_deleteslot(vm, -2, SQFalse);
68 }
69 sq_pop(vm, 1);
70 if (!ready) exposeECS(table);
71}

74void runExposer(Info& info, ssq::Table& table, const char* name) {
75 if (info.exposed || !info.exposer) {
76 info.exposed = true;
77 return;
78 }
79 info.exposed = true;
80 const auto t0 = std::chrono::steady_clock::now();
81 info.exposer(table);
82 const double ms =
83 std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - t0).count();
84 if (ms >= 20.0)
85 std::fprintf(stderr, "[startup] bind %s: %.1f ms\n", name ? name : "?", ms);
86 flushPostEcsHooks(table);
87}

96void bindUntilKeyExists(HSQUIRRELVM vm, SQInteger tableIdx, const char* key, ssq::Table& table) {
97 if (eveRawHas(vm, tableIdx, key)) return;
98 auto& modules = ModuleBindAccess::modules();
99 std::vector<std::string> pending;
100 pending.reserve(modules.size());
101 for (auto& entry : modules) {
102 if (!entry.second.exposed && entry.second.exposer) pending.push_back(entry.first);
103 }
104 std::sort(pending.begin(), pending.end());
105 for (const std::string& name : pending) {
106 auto it = modules.find(name);
107 if (it == modules.end() || it->second.exposed) continue;
108 runExposer(it->second, table, name.c_str());
109 if (eveRawHas(vm, tableIdx, key)) return;
110 }
111 // A name can be provided by a post-ECS hook rather than by any module exposer
112 // (eve.SceneEntity is: the Scene module registers a hook at static-init that
113 // creates it). Hooks only run once the script ECS base exists -- flushPostEcsHooks
114 // is gated on it -- so a first lazy access to such a name scanned every module
115 // without ever injecting that base and then failed with "the index
116 // 'SceneEntity' does not exist", which is exactly the natural opening line
117 // `class MyThing extends eve.SceneEntity`. Inject the base script as the last
118 // step: every module has had its chance by now, so ensureScriptEcsInTable()'s
119 // precedence rules (it replaces a module-provided eve.System with the canonical
120 // one) are applied in the right order.
121 if (!eveRawHas(vm, tableIdx, key)) ensureScriptEcsInTable(table);
122}

124void bindOnDemand(HSQUIRRELVM vm, SQInteger tableIdx, const char* key) {
125 if (!key || eveRawHas(vm, tableIdx, key)) return;
126 ssq::Table table = eveTableFromStack(vm, tableIdx);
127 if (isScriptEcsName(key)) {
128 ensureScriptEcsInTable(table);
129 return;
130 }
131 bindNamedModule(key, table);
132 if (eveRawHas(vm, tableIdx, key)) return;
133 // Nested types (WindowSettings, PlacementWorld, …) live on a parent
134 // module's exposer. Scan remaining PascalCase misses only — lowercase
135 // optional keys such as "dev" must not pull in every native class.
136 if (isPascalCaseName(key)) bindUntilKeyExists(vm, tableIdx, key, table);
137}

139SQInteger eveTableGet(HSQUIRRELVM vm) {
140 if (sq_gettop(vm) < 2 || sq_gettype(vm, 2) != OT_STRING)
141 return sq_throwerror(vm, _SC("the index doesn't exist"));
142 const SQChar* key = nullptr;
143 if (SQ_FAILED(sq_getstring(vm, 2, &key)) || !key)
144 return sq_throwerror(vm, _SC("the index doesn't exist"));
145 const std::string name(key);
146 // Functions declared as `function eve::name` execute with the eve table as
147 // their environment. Preserve the historical root lookup of `eve` itself
148 // instead of treating it as a native-class miss in the lazy delegate.
149 if (name == "eve") {
150 sq_push(vm, 1);
151 return 1;
152 }
153 try {
154 bindOnDemand(vm, 1, name.c_str());
155 } catch (const std::exception& error) {
156 return sq_throwerror(vm, error.what());
157 }
158 sq_push(vm, 1);
159 sq_pushstring(vm, name.c_str(), static_cast<SQInteger>(name.size()));
160 if (SQ_FAILED(sq_rawget(vm, -2))) {
161 const std::string message = "the index '" + name + "' does not exist";
162 return sq_throwerror(vm, message.c_str());
163 }
164 return 1;
165}

181void ModuleManager::exposeVM(ssq::VM& vm) {
182 for (auto& entry : inst().state_->registered_modules) entry.second.exposed = false;
183 prepareEcsScriptInjection();
184
185 ssq::Table table;
186 // Runtime::initialize() guarantees expose() runs once, so addTable never
187 // sees an existing "eve" slot. Avoid find()+catch here: ASYNCIFY-wrapped
188 // frames can drop the catch and leak NotFoundException to the JS boundary.
189 table = vm.addTable("eve");
190 // Result/Status/Diagnostic projection is a common script contract, not a
191 // domain-module convenience. Expose it before module constructors run so
192 // every checked binding observes the same `eve.result` helper.
193 script::exposeResultBindings(table);
194 // Script-side provider publication is the same kind of common contract: it is
195 // how user script answers a native extension point, and it must exist before
196 // any module constructor declares a capability seam.
197 script::exposeProviderBindings(table);
198 installLazyClassGet(table);
199 table.addFunc("_bindAllNativeClasses", []() { return ModuleManager::exposeAllForCompatibility(); });
200 table.addFunc("_ensureScriptECS", []() { return ModuleManager::ensureScriptEcs(); });
201}

218int ModuleManager::expose_pending() {
219 Runtime* active = inst().state_->active_runtime_;
220 if (!active)
221 return -1;
222 int count = 0;
223 try {
224 auto scope = active->enter();
225 auto stack = active->guard();
226 ssq::Table table = active->table("eve");
227 for (auto& D : inst().state_->registered_modules) {
228 if (D.second.exposed || !D.second.exposer)
229 continue;
230 D.second.exposer(table);
231 D.second.exposed = true;
232 flushPostEcsHooks(table);
233 ++count;
234 }
235 } catch (...) {
236 return -1;
237 }
238 return count;
239}
