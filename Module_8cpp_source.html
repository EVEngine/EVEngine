37void ModuleManager::register_module(const char* name, creator_t c, exposer_t e) {
38 EV_PARAM_CHECK(name != nullptr, "module name must not be null");
39 const bool hasCreator = c != nullptr; // function pointers can't be printed by zeroerr
40 EV_PARAM_CHECK(hasCreator, "module creator must not be null");
41 auto p = inst().registered_modules.find(name);
42 if (p == inst().registered_modules.end()) {
43 inst().registered_modules.insert(
44 std::make_pair(std::string(name),
45 ModuleInfo{c, e, nullptr}));
46 } else {
47 p->second = {c, e, nullptr};
48 }
49}

51void ModuleManager::exposeVM(ssq::VM& vm) {
52 ssq::Table table;
53 // Runtime::initialize() guarantees expose() runs once, so addTable never
54 // sees an existing "eve" slot. Avoid find()+catch here: ASYNCIFY-wrapped
55 // frames can drop the catch and leak NotFoundException to the JS boundary.
56 table = vm.addTable("eve");
57 for (auto& D : inst().registered_modules) {
58 if (D.second.exposer) D.second.exposer(table);
59 D.second.exposed = true;
60 }
61 // After modules: script ECS owns eve.Component / Entity / System / view.
62 exposeECS(table);
63}

86int ModuleManager::expose_pending() {
87 Runtime* active = inst().active_runtime_;
88 if (!active)
89 return -1;
90 int count = 0;
91 try {
92 auto scope = active->enter();
93 auto stack = active->guard();
94 ssq::Table table = active->table("eve");
95 for (auto& D : inst().registered_modules) {
96 if (D.second.exposed || !D.second.exposer)
97 continue;
98 D.second.exposer(table);
99 D.second.exposed = true;
100 ++count;
101 }
102 } catch (...) {
103 return -1;
104 }
105 return count;
106}
