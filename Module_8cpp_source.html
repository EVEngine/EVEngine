56void ensureScriptEcsInTable(ssq::Table& table) {
57 HSQUIRRELVM vm = table.getHandle();
58 sq_pushobject(vm, table.getRaw());
59 const bool hasSystem = eveRawHas(vm, -1, "System");
60 const bool ready = eveRawHas(vm, -1, "Component") && eveRawHas(vm, -1, "Entity") && hasSystem;
61 if (!ready && hasSystem) {
62 // The host-information module historically owns eve.System until the
63 // script ECS is injected. Remove that one namespace collision so the
64 // canonical script System class can be created exactly once.
65 sq_pushstring(vm, "System", -1);
66 sq_deleteslot(vm, -2, SQFalse);
67 }
68 sq_pop(vm, 1);
69 if (!ready) exposeECS(table);
70}

72template <typename Info>
73void runExposer(Info& info, ssq::Table& table, const char* name) {
74 if (info.exposed || !info.exposer) {
75 info.exposed = true;
76 return;
77 }
78 info.exposed = true;
79 const auto t0 = std::chrono::steady_clock::now();
80 info.exposer(table);
81 const double ms =
82 std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - t0).count();
83 if (ms >= 20.0)
84 std::fprintf(stderr, "[startup] bind %s: %.1f ms\n", name ? name : "?", ms);
85 flushPostEcsHooks(table);
86}

95void bindUntilKeyExists(HSQUIRRELVM vm, SQInteger tableIdx, const char* key, ssq::Table& table) {
96 if (eveRawHas(vm, tableIdx, key)) return;
97 auto& modules = ModuleBindAccess::modules();
98 std::vector<std::string> pending;
99 pending.reserve(modules.size());
100 for (auto& entry : modules) {
101 if (!entry.second.exposed && entry.second.exposer) pending.push_back(entry.first);
102 }
103 std::sort(pending.begin(), pending.end());
104 for (const std::string& name : pending) {
105 auto it = modules.find(name);
106 if (it == modules.end() || it->second.exposed) continue;
107 runExposer(it->second, table, name.c_str());
108 if (eveRawHas(vm, tableIdx, key)) return;
109 }
110}

112void bindOnDemand(HSQUIRRELVM vm, SQInteger tableIdx, const char* key) {
113 if (!key || eveRawHas(vm, tableIdx, key)) return;
114 ssq::Table table = eveTableFromStack(vm, tableIdx);
115 if (isScriptEcsName(key)) {
116 ensureScriptEcsInTable(table);
117 return;
118 }
119 bindNamedModule(key, table);
120 if (eveRawHas(vm, tableIdx, key)) return;
121 // Nested types (WindowSettings, PlacementWorld, …) live on a parent
122 // module's exposer. Scan remaining PascalCase misses only — lowercase
123 // optional keys such as "dev" must not pull in every native class.
124 if (isPascalCaseName(key)) bindUntilKeyExists(vm, tableIdx, key, table);
125}

127SQInteger eveTableGet(HSQUIRRELVM vm) {
128 if (sq_gettop(vm) < 2 || sq_gettype(vm, 2) != OT_STRING)
129 return sq_throwerror(vm, _SC("the index doesn't exist"));
130 const SQChar* key = nullptr;
131 if (SQ_FAILED(sq_getstring(vm, 2, &key)) || !key)
132 return sq_throwerror(vm, _SC("the index doesn't exist"));
133 const std::string name(key);
134 // Functions declared as `function eve::name` execute with the eve table as
135 // their environment. Preserve the historical root lookup of `eve` itself
136 // instead of treating it as a native-class miss in the lazy delegate.
137 if (name == "eve") {
138 sq_push(vm, 1);
139 return 1;
140 }
141 try {
142 bindOnDemand(vm, 1, name.c_str());
143 } catch (const std::exception& error) {
144 return sq_throwerror(vm, error.what());
145 }
146 sq_push(vm, 1);
147 sq_pushstring(vm, name.c_str(), static_cast<SQInteger>(name.size()));
148 if (SQ_FAILED(sq_rawget(vm, -2))) {
149 const std::string message = "the index '" + name + "' does not exist";
150 return sq_throwerror(vm, message.c_str());
151 }
152 return 1;
153}

207void ModuleManager::register_module(const char* name, creator_t c, exposer_t e) {
208 EV_PARAM_CHECK(name != nullptr, "module name must not be null");
209 const bool hasCreator = c != nullptr; // function pointers can't be printed by zeroerr
210 EV_PARAM_CHECK(hasCreator, "module creator must not be null");
211 auto p = inst().registered_modules.find(name);
212 if (inst().plugin_registration_active_ && p != inst().registered_modules.end()) {
213 if (inst().plugin_registration_error_.empty())
214 inst().plugin_registration_error_ =
215 std::string("module name already registered: ") + name;
216 return;
217 }
218 if (p == inst().registered_modules.end()) {
219 inst().registered_modules.insert(
220 std::make_pair(std::string(name), ModuleInfo{c, e, nullptr}));
221 if (inst().plugin_registration_active_)
222 inst().plugin_registration_added_.emplace_back(name);
223 } else {
224 p->second = {c, e, nullptr};
225 }

239eve::Result<void> ModuleManager::finishPluginRegistration(bool commit) {
240 auto& manager = inst();
241 if (!manager.plugin_registration_active_)
242 return eve::Result<void>::failure(eve::Diagnostic::error(
243 eve::DiagnosticCode::Conflict, "no plugin registration transaction is active", {}, {}, "module"));
244
245 const std::string error = manager.plugin_registration_error_;
246 if (!commit || !error.empty()) {
247 for (const auto& name : manager.plugin_registration_added_) {
248 auto it = manager.registered_modules.find(name);
249 if (it == manager.registered_modules.end())
250 continue;
251 delete it->second.instance;
252 manager.registered_modules.erase(it);
253 }
254 }
255 manager.plugin_registration_active_ = false;
256 manager.plugin_registration_added_.clear();
257 manager.plugin_registration_error_.clear();
258 if (!error.empty())
259 return eve::Result<void>::failure(
260 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, error, {}, {}, "module"));
261 return eve::Result<void>::success(eve::Status::success(commit ? eve::StatusCode::Applied : eve::StatusCode::NoOp));
262}

264void ModuleManager::exposeVM(ssq::VM& vm) {
265 for (auto& entry : inst().registered_modules) entry.second.exposed = false;
266 prepareEcsScriptInjection();
267
268 ssq::Table table;
269 // Runtime::initialize() guarantees expose() runs once, so addTable never
270 // sees an existing "eve" slot. Avoid find()+catch here: ASYNCIFY-wrapped
271 // frames can drop the catch and leak NotFoundException to the JS boundary.
272 table = vm.addTable("eve");
273 // Result/Status/Diagnostic projection is a common script contract, not a
274 // domain-module convenience. Expose it before module constructors run so
275 // every checked binding observes the same `eve.result` helper.
276 script::exposeResultBindings(table);
277 installLazyClassGet(table);
278 table.addFunc("_bindAllNativeClasses", []() { return ModuleManager::exposeAllForCompatibility(); });
279 table.addFunc("_ensureScriptECS", []() { return ModuleManager::ensureScriptEcs(); });
280}

303int ModuleManager::expose_pending() {
304 Runtime* active = inst().active_runtime_;
305 if (!active)
306 return -1;
307 int count = 0;
308 try {
309 auto scope = active->enter();
310 auto stack = active->guard();
311 ssq::Table table = active->table("eve");
312 for (auto& D : inst().registered_modules) {
313 if (D.second.exposed || !D.second.exposer)
314 continue;
315 D.second.exposer(table);
316 D.second.exposed = true;
317 flushPostEcsHooks(table);
318 ++count;
319 }
320 } catch (...) {
321 return -1;
322 }
323 return count;
324}
