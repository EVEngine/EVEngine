40Result<void> PcgGraphicsPhotoModeAuthority::applyPhotoModeField(const PhotoModeAssignment& assignment) {
41 auto fail = [&](const char* message) {
42 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, message, assignment.field,
43 {}, "graphics.pcgPhotoMode"));
44 };
45 auto next = state_;
46 const auto* value = std::get_if<float>(&assignment.value);
47 const auto* mode = std::get_if<int64_t>(&assignment.value);
48 if (assignment.field == "m_lodBias") {
49 if (!value || !std::isfinite(*value) || *value <= 0) return fail("LOD bias must be positive and finite");
50 next.lodBias = *value;
51 } else if (assignment.field == "m_antiAliasing") {
52 if (!mode || *mode < 0 || *mode > 3) return fail("anti-aliasing mode must be in [0,3]");
53 next.antiAliasing = *mode;
54 } else if (assignment.field == "m_shadowDistance") {
55 if (!value || !std::isfinite(*value) || *value < 0)
56 return fail("shadow distance must be finite and non-negative");
57 next.shadowDistance = *value;
58 } else if (assignment.field == "m_shadowResolution") {
59 if (!mode || *mode < 0 || *mode > 3) return fail("shadow resolution must be in [0,3]");
60 next.shadowResolution = *mode;
61 } else if (assignment.field == "m_shadowCascades") {
62 if (!mode || *mode < 0 || *mode > 4) return fail("shadow cascades must be in [0,4]");
63 next.shadowCascades = *mode;
64 } else if (assignment.field == "m_globalShadowDistanceMultiplier") {
65 if (!value || !std::isfinite(*value) || *value < 0 || *value > 5)
66 return fail("global shadow distance multiplier must be in [0,5]");
67 next.globalShadowDistanceMultiplier = *value;
68 } else {
69 return fail("unsupported graphics photo-mode field");
70 }
71 state_ = next;
72 project();
73 return Result<void>::success();
74}

76void PcgGraphicsPhotoModeAuthority::project() {
77 if (camera_) {
78 camera_->data()->lodBias = state_.lodBias;
79 const float shadowDistance = state_.shadowDistance * state_.globalShadowDistanceMultiplier;
80 for (int layer = 0; layer < 32; ++layer) camera_->setShadowLayerCullDistance(layer, shadowDistance);
81 }
82 if (!graphics_ || !control_) return;
83 auto* antiAliasing = graphics_->pipelineAntiAliasing();
84 const int mode = int(state_.antiAliasing);
85 if (mode == 0) {
86 control_->disable("aa");
87 control_->disable("taa");
88 } else {
89 control_->enable("aa");
90 control_->enable(mode == 3 ? "taa" : "aa");
91 if (mode != 3) control_->disable("taa");
92 antiAliasing->setMode(mode == 2 ? "smaa" : mode == 3 ? "taa" : "fxaa");
93 }
94 control_->compile();
95}
