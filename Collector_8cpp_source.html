42void Collector::tick(int player) {
43 if (state_ == State::Idle) return;
44 if (node_ == nullptr) {
45 state_ = State::Idle;
46 return;
47 }
48 switch (state_) {
49 case State::Idle:
50 break;
51 case State::MovingToNode:
52 if (--travelRemaining_ <= 0) state_ = State::Gathering;
53 break;
54 case State::Gathering: {
55 const int freeSpace = carryCapacity_ - cargo_;
56 const int take = std::min({freeSpace, node_->amount(), gatherRatePerTick_});
57 if (take > 0) cargo_ += node_->extract(take);
58 if (cargo_ >= carryCapacity_ || node_->depleted()) startMoveToDrop();
59 break;
60 }
61 case State::MovingToDrop:
62 if (--travelRemaining_ <= 0) state_ = State::Depositing;
63 break;
64 case State::Depositing:
65 deposit(player);
66 break;
67 }
68}
