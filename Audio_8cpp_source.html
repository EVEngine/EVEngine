63void Audio::workerMain() {
64 while (running.load()) {
65 std::unique_lock<std::mutex> lock(mutex);
66 cv.wait_for(lock, std::chrono::milliseconds(10),
67 [this] { return !running.load() || !streamSources.empty(); });
68 if (!running.load())
69 break;
70 // Keep Audio::mutex held while filling. Source::~Source() removes the
71 // source from streamSources under the same mutex, so a source cannot be
72 // destroyed (and its decoderMutex/pendingMutex freed) while the worker
73 // is mid-fill. A snapshot-then-release pattern would let the destructor
74 // finish between the snapshot and the fill; the worker would then lock
75 // mutexes that no longer exist (std::system_error: mutex lock failed).
76 for (Source *s : streamSources) {
77 if (s)
78 s->fillPendingFromDecoder();
79 }
80 }
81}

107Source *Audio::newSourceFromDecoder(sound::Decoder *decoder, std::string type) {
108 if (type != "static" && type != "stream")
109 throw eve::Exception("Invalid source type '%s'", type.c_str());
110 if (type == "static") {
111 auto *sound = sound::Sound::create();
112 auto *sd = sound->newSoundDataFromDecoder(decoder);
113 auto *s = newSource(sd);
114 // SoundData owned by caller eventually; keep sd alive via Source holding pointer only —
115 // Source does not own SoundData. Caller must keep SoundData alive.
116 // For static-from-decoder convenience, leak prevention: Source should own or we document.
117 // Spec: squirrel manages SoundData. Here C++ path: attach as dependency by not deleting.
118 // Store sd on Source via staticData without ownership — caller must delete sd after Source.
119 // Better: Source holds SoundData* without delete; document. For this factory, transfer:
120 // We'll let Source keep staticData pointer; user deletes Source then SoundData.
121 return s;
122 }
123 auto *s = new Source(this, decoder, true);
124 std::lock_guard<std::mutex> lock(mutex);
125 allSources.push_back(s);
126 return s;
127}
