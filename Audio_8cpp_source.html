60void Audio::workerMain() {
61 while (running.load()) {
62 std::unique_lock<std::mutex> lock(mutex);
63 cv.wait_for(lock, std::chrono::milliseconds(10),
64 [this] { return !running.load() || !streamSources.empty(); });
65 if (!running.load())
66 break;
67 // Keep Audio::mutex held while filling. Source::~Source() removes the
68 // source from streamSources under the same mutex, so a source cannot be
69 // destroyed (and its decoderMutex/pendingMutex freed) while the worker
70 // is mid-fill. A snapshot-then-release pattern would let the destructor
71 // finish between the snapshot and the fill; the worker would then lock
72 // mutexes that no longer exist (std::system_error: mutex lock failed).
73 for (Source *s : streamSources) {
74 if (s)
75 s->fillPendingFromDecoder();
76 }
77 }
78}

104Source *Audio::newSourceFromDecoder(sound::Decoder *decoder, std::string type) {
105 if (type != "static" && type != "stream")
106 throw eve::Exception("Invalid source type '%s'", type.c_str());
107 if (type == "static") {
108 auto *sound = sound::Sound::create();
109 auto *sd = sound->newSoundDataFromDecoder(decoder);
110 auto *s = newSource(sd);
111 // SoundData owned by caller eventually; keep sd alive via Source holding pointer only —
112 // Source does not own SoundData. Caller must keep SoundData alive.
113 // For static-from-decoder convenience, leak prevention: Source should own or we document.
114 // Spec: squirrel manages SoundData. Here C++ path: attach as dependency by not deleting.
115 // Store sd on Source via staticData without ownership — caller must delete sd after Source.
116 // Better: Source holds SoundData* without delete; document. For this factory, transfer:
117 // We'll let Source keep staticData pointer; user deletes Source then SoundData.
118 return s;
119 }
120 auto *s = new Source(this, decoder, true);
121 std::lock_guard<std::mutex> lock(mutex);
122 allSources.push_back(s);
123 return s;
124}

153void Audio::stopAll() {
154 // Hold the mutex while stopping: Source::~Source() → unregisterSource()
155 // blocks on the same mutex, so a concurrently destroyed source cannot be
156 // touched here after it has been removed from allSources.
157 std::lock_guard<std::mutex> lock(mutex);
158 for (Source *s : allSources) {

160 s->stop();
