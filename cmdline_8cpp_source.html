45 bool version = false;

48
49 std::vector<Handler*> handles;
50 for (auto cmd : handers()) {
51 auto* p = cmd();
52 handles.push_back(p);
53 p->setup(app, formatter);
54 }
55
56 try {
57 this->argc = argcIn;
58 this->argv = argvIn;
59 app.parse(argcIn, argvIn);
60 } catch (const CLI::ParseError& e) {
61 for (auto* h : handles) delete h;
62 int res = app.exit(e);
63 return res;
64 }

66 int res = 0;
67 for (auto* h : handles) {
68 if (int r = h->parse(app, *this); r != -1) {
69 res = r;
70 break;
71 }
72 }
73 if (version) cout << eve::buildInfo() << endl;
74
75 for (auto* h : handles) delete h;
76 return res;
