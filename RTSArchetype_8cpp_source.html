25Result<Value::Object> definitionObject(definitions::DefinitionRegistry& registry, std::string_view type,
26 const LogicalId& id) {
27 if (!id.isValid() || id.name().empty())
28 return invalid<Value::Object>("RTS root requires a logical definition", std::string(type) + ".definition");
29 auto resolved = registry.resolve(std::string(type), std::string(id.name()));
30 if (!resolved) return Result<Value::Object>::failure(resolved.status());
31 auto parsed = Value::fromJson(resolved.value().get().json);
32 if (!parsed) return Result<Value::Object>::failure(parsed.status());
33 auto value = std::move(parsed).takeValue();
34 const auto* object = value.getIf<Value::Object>();
35 if (object == nullptr)
36 return invalid<Value::Object>("RTS definition must be a JSON object", std::string(type) + ".json");
37 return Result<Value::Object>::success(*object);
38}

40Result<float> number(const Value::Object& object, std::string_view name, float fallback,
41 float minimum = -std::numeric_limits<float>::max()) {
42 const Value* source = field(object, name);
43 double value = fallback;
44 if (source != nullptr) {
45 if (const auto* real = source->getIf<double>()) value = *real;
46 else if (const auto* integer = source->getIf<std::int64_t>()) value = static_cast<double>(*integer);
47 else return invalid<float>("RTS archetype numeric field must be a number", std::string(name));
48 }
49 if (!std::isfinite(value) || value < minimum || value > std::numeric_limits<float>::max())
50 return invalid<float>("RTS archetype numeric field is outside its valid range", std::string(name));
51 return Result<float>::success(static_cast<float>(value));
52}

82Result<TagSet> tags(const Value::Object& object, std::string_view name) {
83 TagSet result;
84 const Value* source = field(object, name);
85 if (source == nullptr) return Result<TagSet>::success(std::move(result));
86 const auto* array = source->getIf<Value::Array>();
87 if (array == nullptr) return invalid<TagSet>("RTS archetype tags must be an array", std::string(name));
88 for (std::size_t index = 0; index < array->size(); ++index) {
89 const auto* value = (*array)[index].getIf<std::string>();
90 if (value == nullptr)
91 return invalid<TagSet>("RTS archetype tag must be a string",
92 std::string(name) + "[" + std::to_string(index) + "]");
93 auto added = result.add(*value);
94 if (!added) return Result<TagSet>::failure(added.status());
95 }
96 return Result<TagSet>::success(std::move(result));
