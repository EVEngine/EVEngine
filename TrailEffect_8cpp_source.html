47TrailAppendResult TrailEmitter::append(glm::vec3 root, glm::vec3 tip) {
48 if (!finite(root) || !finite(tip)) throw eve::Exception("TrailEmitter.append: endpoints must be finite");
49 TrailAppendResult result = TrailAppendResult::Added;
50 if (!samples_.empty() && !forceBreak_) {
51 const float distance = endpointDistance(samples_.back(), root, tip);
52 if (distance < settings_.minSampleDistance) return TrailAppendResult::SkippedTooClose;
53 if (distance > settings_.teleportDistance) {
54 ++segment_;
55 result = TrailAppendResult::StartedSegment;
56 }
57 } else {
58 if (!samples_.empty()) ++segment_;
59 result = TrailAppendResult::StartedSegment;
60 }
61 samples_.push_back({root, tip, 0.f, segment_});
62 forceBreak_ = false;
63 while (samples_.size() > settings_.maxSamples) samples_.pop_front();
64 return result;

75TrailMeshSnapshot TrailEmitter::buildMesh() const {
76 TrailMeshSnapshot mesh;
77 mesh.vertices.reserve(samples_.size() * 2);
78 if (samples_.size() < 2) return mesh;
79 std::size_t segmentStart = 0;
80 while (segmentStart < samples_.size()) {
81 std::size_t segmentEnd = segmentStart + 1;
82 while (segmentEnd < samples_.size() && samples_[segmentEnd].segment == samples_[segmentStart].segment)
83 ++segmentEnd;
84 const std::size_t segmentSize = segmentEnd - segmentStart;
85 for (std::size_t i = segmentStart; i < segmentEnd; ++i) {
86 const float u = segmentSize > 1 ? float(i - segmentStart) / float(segmentSize - 1) : 0.f;
87 const float alpha = std::clamp(1.f - samples_[i].age / settings_.lifetime, 0.f, 1.f);
88 mesh.vertices.push_back({samples_[i].root, {u, 0.f}, alpha});
89 mesh.vertices.push_back({samples_[i].tip, {u, 1.f}, alpha});
90 if (i > segmentStart) {
91 const std::uint32_t previous = static_cast<std::uint32_t>(mesh.vertices.size() - 4);
92 const std::uint32_t current = static_cast<std::uint32_t>(mesh.vertices.size() - 2);
93 mesh.indices.insert(mesh.indices.end(), {previous, previous + 1, current + 1,
94 previous, current + 1, current});
95 }
96 }
97 segmentStart = segmentEnd;
98 }
99 return mesh;
