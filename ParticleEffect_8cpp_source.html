110ParticleEffect* ParticleEffect::fromText(const std::string& json, const std::string& sourcePath, std::string* error) {
111 auto fail = [&](const std::string& message) -> ParticleEffect* {
112 if (error) *error = message;
113 return nullptr;
114 };
115
116 auto* dataModule = eve::data::DataModule::create();
117 std::string parseError;
118 std::unique_ptr<data::JsonDocument> document(dataModule->decodeJson(json, &parseError));
119 if (!document || !document->isObject())
120 return fail(parseError.empty() ? "effect root must be an object" : parseError);
121
122 auto root = document->object();
123 if (!root || !root->has("type") || asString(root->get("type")) != "eve.particle-effect")
124 return fail("effect type must be eve.particle-effect");
125 const int version = root->has("version") ? asInt(root->get("version"), 0) : 0;
126 if (version != kParticleEffectVersion)
127 return fail("unsupported particle effect version: " + std::to_string(version));
128
129 Poco::JSON::Array::Ptr emitterArray;
130 try {
131 emitterArray = root->getArray("emitters");
132 } catch (...) {
133 }
134 if (!emitterArray || emitterArray->size() == 0) return fail("effect emitters must be a non-empty array");
135
136 auto effect = std::unique_ptr<ParticleEffect>(new ParticleEffect());
137 effect->version_ = version;
138 effect->sourcePath_ = sourcePath;
139
140 Poco::JSON::Object::Ptr globalParameters;
141 if (root->has("parameters")) {
142 try {
143 globalParameters = root->getObject("parameters");
144 } catch (...) {
145 }
146 if (!globalParameters) return fail("effect parameters must be an object");
147 for (const auto& parameter : *globalParameters)
148 effect->parameters_[parameter.first] = asFloat(parameter.second, 0.f);
149 }
150
151 std::unordered_set<std::string> names;
152 for (std::size_t i = 0; i < emitterArray->size(); ++i) {
153 Poco::JSON::Object::Ptr layerObject;
154 try {
155 layerObject = emitterArray->getObject(static_cast<int>(i));
156 } catch (...) {
157 }
158 if (!layerObject) return fail("emitter layer " + std::to_string(i) + " must be an object");
159
160 const std::string name = layerObject->has("name") ? asString(layerObject->get("name")) : std::string();
161 if (name.empty()) return fail("emitter layer " + std::to_string(i) + " requires a name");
162 if (!names.insert(name).second) return fail("duplicate emitter layer name: " + name);
163
164 Poco::JSON::Object::Ptr embeddedConfig;
165 if (layerObject->has("emitter")) {
166 try {
167 embeddedConfig = layerObject->getObject("emitter");
168 } catch (...) {
169 }
170 if (!embeddedConfig) return fail("emitter layer " + name + " has an invalid emitter object");
171 }
172 const std::string configPath =
173 layerObject->has("config") ? asString(layerObject->get("config")) : std::string();
174 if (embeddedConfig && !configPath.empty())
175 return fail("emitter layer " + name + " cannot define both emitter and config");
176 if (!embeddedConfig && configPath.empty()) return fail("emitter layer " + name + " requires emitter or config");
177
178 int buffer = layerObject->has("buffer") ? asInt(layerObject->get("buffer"), 1000) : 1000;
179 buffer = objectBuffer(embeddedConfig, buffer);
180 auto* emitter = ParticleEmitter::createEmitter(std::max(1, buffer));
181 std::string configError;
182 bool configured = false;
183 if (embeddedConfig) {
184 configured = applyConfigText(emitter, objectJson(embeddedConfig), &configError);
185 } else {
186 configured = loadConfigFile(emitter, resolveAssetPath(sourcePath, configPath), &configError);
187 }
188 if (!configured) {
189 emitter->release();
190 return fail("emitter layer " + name + ": " + configError);
191 }
192
193 applyParameters(emitter, globalParameters);
194 if (layerObject->has("parameters")) {
195 Poco::JSON::Object::Ptr layerParameters;
196 try {
197 layerParameters = layerObject->getObject("parameters");
198 } catch (...) {
199 }
200 if (!layerParameters) {
201 emitter->release();
202 return fail("emitter layer " + name + " parameters must be an object");
203 }
204 applyParameters(emitter, layerParameters);
205 }
206
207 Layer layer;
208 layer.name = name;
209 layer.emitter = emitter;
210 readOffset(layerObject, layer.offsetX, layer.offsetY);
211 const float localRotation = layerObject->has("rotation") ? asFloat(layerObject->get("rotation"), 0.f) : 0.f;
212 layer.baseDirection = emitter->getDirection() + localRotation;
213 layer.baseLayer = emitter->getLayer();
214 layer.enabled = layerObject->has("enabled") ? asBool(layerObject->get("enabled"), true) : true;
215 emitter->setVisible(layer.enabled);
216 if (!layer.enabled) emitter->stop();
217 effect->layers_.push_back(std::move(layer));
218 }
219
220 effect->syncTransform();
221 effect->syncLayer();
222 if (error) error->clear();
223 return effect.release();
224}

226ParticleEffect* ParticleEffect::fromFile(const std::string& path, std::string* error) {
227 if (path.empty()) {
228 if (error) *error = "empty effect path";
229 return nullptr;
230 }
231 auto* filesystem = eve::ModuleManager::getInstance<eve::filesystem::Filesystem>("Filesystem");
232 if (!filesystem) filesystem = eve::filesystem::Filesystem::create();
233 std::unique_ptr<eve::filesystem::FileData> data;
234 try {
235 data.reset(filesystem->read(path));
236 } catch (...) {
237 if (error) *error = "read failed: " + path;
238 return nullptr;
239 }
240 if (!data || data->getSize() == 0) {
241 if (error) *error = "empty file: " + path;
242 return nullptr;
243 }
244 return fromText(std::string(static_cast<const char*>(data->getData()), data->getSize()), path, error);
245}

264void ParticleEffect::syncTransform() {
265 const float c = std::cos(rotation_);
266 const float s = std::sin(rotation_);
267 for (auto& layer : layers_) {
268 const float ox = layer.offsetX * scale_;
269 const float oy = layer.offsetY * scale_;
270 layer.emitter->setPosition(x_ + ox * c - oy * s, y_ + ox * s + oy * c);
271 layer.emitter->setDirection(layer.baseDirection + rotation_);
272 }
273}
