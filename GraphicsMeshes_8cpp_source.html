63Mesh* Graphics::newMeshFromArraysColored(const float* posXYZ, const float* nrmXYZ, const float* uvST,
64 const float* colorRGBA, int vertexCount, const uint32_t* indices,
65 int indexCount) {
66 if (vertexCount <= 0 || !posXYZ) throw Exception("newMeshFromArrays: invalid vertex data");
67 if (indexCount < 0 || (indexCount > 0 && (!indices || indexCount % 3 != 0)))
68 throw Exception("newMeshFromArrays: invalid index data");
69 for (int i = 0; i < indexCount; ++i) {
70 if (indices[i] >= uint32_t(vertexCount)) throw Exception("newMeshFromArrays: index out of range");
71 }
72
73 std::vector<MeshVertex> verts(static_cast<size_t>(vertexCount));
74 for (int i = 0; i < vertexCount; ++i) {
75 auto& v = verts[static_cast<size_t>(i)];
76 v.pos = {posXYZ[i * 3 + 0], posXYZ[i * 3 + 1], posXYZ[i * 3 + 2]};
77 if (nrmXYZ) {
78 v.normal = {nrmXYZ[i * 3 + 0], nrmXYZ[i * 3 + 1], nrmXYZ[i * 3 + 2]};
79 } else {
80 v.normal = {0.f, 0.f, 1.f};
81 }
82 if (uvST) {
83 v.uv = {uvST[i * 2 + 0], uvST[i * 2 + 1]};
84 } else {
85 v.uv = {0.f, 0.f};
86 }
87 if (colorRGBA) v.color = {colorRGBA[i * 4], colorRGBA[i * 4 + 1], colorRGBA[i * 4 + 2], colorRGBA[i * 4 + 3]};
88 }
89
90 auto gpu = std::make_unique<GpuMesh>();
91 gpu->vertexCount = uint32_t(vertexCount);
92 gpu->indexCount = indexCount > 0 ? uint32_t(indexCount) : 0;
93 gpu->vertexStride = sizeof(MeshVertex);
94 gpu->cpuVertices = verts;
95
96 WGPUBufferDescriptor vbd{};
97 vbd.label = sv("eve_mesh_vb");
98 vbd.size = verts.size() * sizeof(MeshVertex);
99 vbd.usage = WGPUBufferUsage_CopyDst | WGPUBufferUsage_Vertex | WGPUBufferUsage_Storage;
100 vbd.mappedAtCreation = false;
101 gpu->vertexBuffer = device.CreateBuffer(reinterpret_cast<const wgpu::BufferDescriptor*>(&vbd));
102 queue.WriteBuffer(gpu->vertexBuffer, 0, verts.data(), vbd.size);
103 gpu->vertexCapacity = vbd.size;
104
105 if (indexCount > 0) {
106 // 16-bit index format halves index memory for meshes with <= 65535 verts.
107 if (vertexCount <= 65535) {
108 std::vector<uint16_t> idx16;
109 idx16.reserve(indexCount);
110 for (int i = 0; i < indexCount; ++i) idx16.push_back(uint16_t(indices[i]));
111 WGPUBufferDescriptor ibd{};
112 ibd.label = sv("eve_mesh_ib");
113 if (idx16.size() % 2 != 0) idx16.push_back(0); // WriteBuffer size must align to 4.
114 ibd.size = uint64_t(idx16.size()) * sizeof(uint16_t);
115 ibd.usage = WGPUBufferUsage_CopyDst | WGPUBufferUsage_Index | WGPUBufferUsage_Storage;
116 ibd.mappedAtCreation = false;
117 gpu->indexBuffer = device.CreateBuffer(reinterpret_cast<const wgpu::BufferDescriptor*>(&ibd));
118 queue.WriteBuffer(gpu->indexBuffer, 0, idx16.data(), ibd.size);
119 gpu->indexFormat = wgpu::IndexFormat::Uint16;
120 gpu->indexCapacity = ibd.size;
121 } else {
122 WGPUBufferDescriptor ibd{};
123 ibd.label = sv("eve_mesh_ib");
124 ibd.size = uint64_t(indexCount) * sizeof(uint32_t);
125 ibd.usage = WGPUBufferUsage_CopyDst | WGPUBufferUsage_Index | WGPUBufferUsage_Storage;
126 ibd.mappedAtCreation = false;
127 gpu->indexBuffer = device.CreateBuffer(reinterpret_cast<const wgpu::BufferDescriptor*>(&ibd));
128 queue.WriteBuffer(gpu->indexBuffer, 0, indices, ibd.size);
129 gpu->indexFormat = wgpu::IndexFormat::Uint32;
130 gpu->indexCapacity = ibd.size;
131 }
132 }
133
134 auto* mesh = new Mesh();
135 mesh->indexCount = indexCount;
136 mesh->gpuVertexCount = vertexCount;
137 mesh->gpuHandle = gpu.get();
138 mesh->computeBounds(posXYZ, vertexCount);
139 ownedGpuMeshes.push_back(std::move(gpu));
140 ownedMeshes.push_back(std::unique_ptr<Mesh>(mesh));
141 return mesh;
142}

161Mesh* Graphics::newMeshFromAssimp(const ::aiMesh& mesh, const glm::mat4& worldTransform) {
162 // glm is column-major; Assimp aiMatrix4x4 is row-major.
163 const aiMatrix4x4 world(worldTransform[0][0], worldTransform[1][0], worldTransform[2][0], worldTransform[3][0],
164 worldTransform[0][1], worldTransform[1][1], worldTransform[2][1], worldTransform[3][1],
165 worldTransform[0][2], worldTransform[1][2], worldTransform[2][2], worldTransform[3][2],
166 worldTransform[0][3], worldTransform[1][3], worldTransform[2][3], worldTransform[3][3]);
167
168 std::vector<float> pos, nrm, uv;
169 std::vector<uint32_t> idx;
170 pos.reserve(mesh.mNumVertices * 3);
171 nrm.reserve(mesh.mNumVertices * 3);
172 uv.reserve(mesh.mNumVertices * 2);
173 idx.reserve(mesh.mNumFaces * 3);
174
175 for (unsigned i = 0; i < mesh.mNumVertices; ++i) {
176 aiVector3D p = world * mesh.mVertices[i];
177 pos.push_back(p.x);
178 pos.push_back(p.y);
179 pos.push_back(p.z);
180 if (mesh.mNormals) {
181 aiMatrix3x3 rot(world);
182 aiVector3D n = rot * mesh.mNormals[i];
183 n.Normalize();
184 nrm.push_back(n.x);
185 nrm.push_back(n.y);
186 nrm.push_back(n.z);
187 } else {
188 nrm.push_back(0.f);
189 nrm.push_back(0.f);
190 nrm.push_back(1.f);
191 }
192 if (mesh.mTextureCoords[0]) {
193 uv.push_back(mesh.mTextureCoords[0][i].x);
194 uv.push_back(mesh.mTextureCoords[0][i].y);
195 } else {
196 uv.push_back(0.f);
197 uv.push_back(0.f);
198 }
199 }
200 for (unsigned f = 0; f < mesh.mNumFaces; ++f) {
201 for (unsigned v = 0; v < mesh.mFaces[f].mNumIndices; ++v) idx.push_back(mesh.mFaces[f].mIndices[v]);
202 }
203 Mesh* m = newMeshFromArrays(pos.data(), nrm.data(), uv.data(), int(mesh.mNumVertices), idx.data(), int(idx.size()));
204 if (m) m->captureImportedAttributes(mesh);
205 if (m && mesh.mNumAnimMeshes > 0) {
206 m->initMorphBase(int(mesh.mNumVertices), pos.data(), nrm.data(), uv.data());
207 for (unsigned a = 0; a < mesh.mNumAnimMeshes; ++a) {
208 const aiAnimMesh* am = mesh.mAnimMeshes[a];
209 std::vector<float> absPos(am->mNumVertices * 3);
210 for (unsigned i = 0; i < am->mNumVertices; ++i) {
211 absPos[i * 3 + 0] = am->mVertices[i].x;
212 absPos[i * 3 + 1] = am->mVertices[i].y;
213 absPos[i * 3 + 2] = am->mVertices[i].z;
214 }
215 std::string name = am->mName.C_Str();
216 if (name.empty()) name = "morph" + std::to_string(a);
217 m->addMorphTargetAbsolute(name, absPos.data());
218 if (am->mNormals) {
219 std::vector<float> deltas(am->mNumVertices * 3u);
220 for (unsigned i = 0; i < am->mNumVertices; ++i) {
221 deltas[i * 3] = am->mNormals[i].x - nrm[i * 3];
222 deltas[i * 3 + 1] = am->mNormals[i].y - nrm[i * 3 + 1];
223 deltas[i * 3 + 2] = am->mNormals[i].z - nrm[i * 3 + 2];
224 }
225 auto normals = m->setMorphNormalDeltas(name, deltas);
226 if (!normals.ok()) throw Exception("Invalid imported morph normals: %s", name.c_str());
227 }
228 }
229 }
230 return m;
231}

233bool Graphics::bakeMeshMorph(Mesh* mesh) {
234 if (!mesh || !mesh->hasMorphData() || !mesh->isMorphDirty()) return false;
235 auto* gpu = static_cast<GpuMesh*>(mesh->gpuHandle);
236 if (!gpu || !gpu->vertexBuffer) return false;
237 std::vector<float> pos, nrm;
238 mesh->computeMorphedPositions(pos, nrm);
239 if (pos.empty()) return false;
240 mesh->computeBounds(pos.data(), mesh->getVertexCount());
241 auto& verts = gpu->cpuVertices;
242 if (verts.size() != static_cast<size_t>(mesh->getVertexCount())) return false;
243 const auto& uvs = mesh->baseUv();
244 for (int i = 0; i < mesh->getVertexCount(); ++i) {
245 auto& v = verts[static_cast<size_t>(i)];
246 v.pos = {pos[i * 3 + 0], pos[i * 3 + 1], pos[i * 3 + 2]};
247 if (nrm.size() >= size_t((i + 1) * 3)) {
248 v.normal = {nrm[i * 3 + 0], nrm[i * 3 + 1], nrm[i * 3 + 2]};
249 } else {
250 v.normal = {0.f, 0.f, 1.f};
251 }
252 if (uvs.size() >= size_t((i + 1) * 2)) {
253 v.uv = {uvs[i * 2 + 0], uvs[i * 2 + 1]};
254 } else {
255 v.uv = {0.f, 0.f};
256 }
257 }
258 queue.WriteBuffer(gpu->vertexBuffer, 0, verts.data(), verts.size() * sizeof(MeshVertex));
259 mesh->markMorphClean();
260 return true;
261}

264bool Graphics::setMeshSkinningData(Mesh* mesh, const uint16_t* joints4, const float* weights4, int vertexCount) {
265 if (!mesh || !mesh->gpuHandle || !joints4 || !weights4) return false;
266 auto* gpu = static_cast<GpuMesh*>(mesh->gpuHandle);
267 if (vertexCount <= 0 || gpu->cpuVertices.size() != static_cast<size_t>(vertexCount)) return false;
268 for (int i = 0; i < vertexCount; ++i) {
269 const size_t base = static_cast<size_t>(i) * 4u;
270 auto& v = gpu->cpuVertices[static_cast<size_t>(i)];
271 v.joints = glm::u16vec4(joints4[base], joints4[base + 1], joints4[base + 2], joints4[base + 3]);
272 v.weights = glm::vec4(weights4[base], weights4[base + 1], weights4[base + 2], weights4[base + 3]);
273 }
274 queue.WriteBuffer(gpu->vertexBuffer, 0, gpu->cpuVertices.data(), gpu->cpuVertices.size() * sizeof(MeshVertex));
275 mesh->markGpuSkinned(true);
276 return true;
277}

279bool Graphics::releaseMesh(Mesh* mesh) {
280 if (!mesh || !mesh->gpuHandle) return false;
281
282 auto* gpu = static_cast<GpuMesh*>(mesh->gpuHandle);
283 auto gpuIt = std::find_if(ownedGpuMeshes.begin(), ownedGpuMeshes.end(),
284 [&](const std::unique_ptr<GpuMesh>& g) { return g.get() == gpu; });
285 if (gpuIt == ownedGpuMeshes.end()) return false;
286
287 auto meshIt = std::find_if(ownedMeshes.begin(), ownedMeshes.end(),
288 [&](const std::unique_ptr<Mesh>& m) { return m.get() == mesh; });
289 if (meshIt == ownedMeshes.end()) return false;
290
291 mesh->gpuHandle = nullptr;
292 ownedGpuMeshes.erase(gpuIt);
293 // Transfer the CPU facade to the caller instead of destroying it.
294 (void)meshIt->release();
295 ownedMeshes.erase(meshIt);
296 return true;
297}

299Mesh* Graphics::newMeshSphere(int slices, int stacks) {
300 std::vector<float> pos, nrm, uv;
301 std::vector<uint32_t> idx;
302 for (int y = 0; y <= stacks; ++y) {
303 for (int x = 0; x <= slices; ++x) {
304 float u = float(x) / float(slices);
305 float v = float(y) / float(stacks);
306 float theta = u * 2.f * glm::pi<float>();
307 float phi = v * glm::pi<float>();
308 float sx = std::sin(phi) * std::cos(theta);
309 float sy = std::cos(phi);
310 float sz = std::sin(phi) * std::sin(theta);
311 pos.push_back(sx);
312 pos.push_back(sy);
313 pos.push_back(sz);
314 nrm.push_back(sx);
315 nrm.push_back(sy);
316 nrm.push_back(sz);
317 uv.push_back(u);
318 uv.push_back(v);
319 }
320 }
321 for (int y = 0; y < stacks; ++y) {
322 for (int x = 0; x < slices; ++x) {
323 int a = y * (slices + 1) + x;
324 int b = a + slices + 1;
325 idx.push_back(a);
326 idx.push_back(a + 1);
327 idx.push_back(b);
328 idx.push_back(b);
329 idx.push_back(a + 1);
330 idx.push_back(b + 1);
331 }
332 }
