80[[nodiscard]] ssq::Table projectBake(HSQUIRRELVM vm, ScriptProcgenParams* params,
81 Result<Value> (*generate)(const Params&), const char* entry) {
82 if (params == nullptr)
83 return script::projectStatusResult(
84 vm, Result<Value>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
85 std::string(entry) + " params proxy must not be null",
86 "params", {}, "procgen.squirrel"))
87 .status());
88 auto view = Procgen::resolve(params->reference);
89 if (!view.isBound())
90 return script::projectStatusResult(
91 vm, Result<Value>::failure(Diagnostic::error(DiagnosticCode::StaleHandle,
92 std::string(entry) + " parameters handle is stale", "params",
93 {}, "procgen.squirrel"))
94 .status());
95 return script::projectResult(vm, generate(*view), [](Value value) { return value; });
96}

146Result<Value> generateHexTerrain(const Params& params) {
147 hexmap::HexMap map;
148 auto reset = map.reset(params.getWidth(), params.getHeight(), params.getSeed());
149 if (!reset) return Result<Value>::failure(reset.status());
150 auto generated = hexmap::generateHexMap(map, planarSettings(params));
151 if (!generated) return Result<Value>::failure(generated.status());
152 return Result<Value>::success(hexmap::hexTerrainBakeToValue(hexmap::snapshotHexTerrain(map)));
153}

155Result<Value> generateHexSphere(const Params& params) {
156 const std::int32_t subdivision = params.getInt("subdivision", 4);
157 const float radius = params.getFloat("radius", 100.f);
158 hexmap::HexSphereMap map;
159 auto reset = map.reset(subdivision, radius, params.getSeed());
160 if (!reset) return Result<Value>::failure(reset.status());
161 auto generated = hexmap::generateSphereMap(map, sphereSettings(params));
162 if (!generated) return Result<Value>::failure(generated.status());
163 return Result<Value>::success(hexmap::hexTerrainBakeToValue(hexmap::snapshotHexSphereTerrain(map)));
164}
