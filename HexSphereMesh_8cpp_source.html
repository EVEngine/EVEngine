45[[nodiscard]] HexVec3 slerpDirection(HexVec3 a, HexVec3 b, float t) noexcept {
46 const float cosine = std::clamp(dotOf(a, b), -1.f, 1.f);
47 const float angle = std::acos(cosine);
48 if (angle < 1e-5f) {
49 return normalize(HexVec3{a.x + (b.x - a.x) * t, a.y + (b.y - a.y) * t, a.z + (b.z - a.z) * t});
50 }
51 const float sine = std::sin(angle);
52 const float wa = std::sin((1.f - t) * angle) / sine;
53 const float wb = std::sin(t * angle) / sine;
54 return HexVec3{a.x * wa + b.x * wb, a.y * wa + b.y * wb, a.z * wa + b.z * wb};
55}

64[[nodiscard]] EdgeVertices sphereEdge(HexVec3 a, HexVec3 b) noexcept {
65 const HexVec3 dirA = normalize(a);
66 const HexVec3 dirB = normalize(b);
67 const float ra = lengthOf(a);
68 const float rb = lengthOf(b);
69
70 EdgeVertices edge;
71 HexVec3* samples[5] = {&edge.v1, &edge.v2, &edge.v3, &edge.v4, &edge.v5};
72 for (std::int32_t i = 0; i < 5; ++i) {
73 const float t = static_cast<float>(i) * 0.25f;
74 const float radius = ra + (rb - ra) * t;
75 *samples[i] = slerpDirection(dirA, dirB, t) * radius;
76 }
77 return edge;
78}

88[[nodiscard]] HexVec3 sphereTerraceLerp(HexVec3 a, HexVec3 b, std::int32_t step) noexcept {
89 const float horizontal = static_cast<float>(step) * HexMetrics::kHorizontalTerraceStepSize;
90 const float vertical = static_cast<float>((step + 1) / 2) * HexMetrics::kVerticalTerraceStepSize;
91 const HexVec3 direction = slerpDirection(normalize(a), normalize(b), horizontal);
92 const float ra = lengthOf(a);
93 const float rb = lengthOf(b);
94 return direction * (ra + (rb - ra) * vertical);
95}

122[[nodiscard]] HexVec3 tangentPerturb(const HexNoise& noise, HexVec3 position, float strength) noexcept {
123 const float radius = lengthOf(position);
124 if (!(radius > 1e-6f)) return position;
125 const HexVec3 up = position * (1.f / radius);
126 // A tangent frame that stays well conditioned over the whole sphere: the world axis
127 // least aligned with `up` is never parallel to it.
128 const HexVec3 axis = std::fabs(up.y) < 0.9f ? HexVec3{0.f, 1.f, 0.f} : HexVec3{1.f, 0.f, 0.f};
129 const HexVec3 tangentX = normalize(crossOf(axis, up));
130 const HexVec3 tangentY = crossOf(up, tangentX);
131
132 const HexVec4 first = noise.sample(position.x, position.z);
133 const HexVec4 second = noise.sample(position.y + 37.f, position.x - 11.f);
134 const float alongX = (first.x * 2.f - 1.f) * strength;
135 const float alongY = (second.z * 2.f - 1.f) * strength;
136 return normalize(position + tangentX * alongX + tangentY * alongY) * radius;
137}

172public:
173 SphereMesher(const HexSphereMap& map, HexMeshData& out) noexcept
174 : map_(map), out_(out),
175 // The planar mesher perturbs by a fixed 40% of a cell's outer radius. The
176 // same fraction of the mean cell spacing keeps the wobble scale-invariant
177 // across subdivision levels and radii, which a fixed world-unit figure
178 // would not be.
179 perturbStrength_((HexMetrics::kCellPerturbStrength / HexMetrics::kOuterRadius) * map.cellSpacing()),
180 // River beds drop by the planar stream-bed offset expressed in elevation
181 // steps, so the channel keeps the same depth relative to the relief.
182 streamBedDrop_((HexMetrics::kStreamBedElevationOffset / HexMetrics::kElevationStep) * map.elevationStep()) {}
183
185 void run() {
186 for (HexSphereCell cell = 0; cell < map_.cellCount(); ++cell) buildCell(cell);
187 }

194 const HexVec3 mixed =

201 const std::int32_t edges = map_.neighborCount(cell);
202 const std::int32_t nextD = (d + 1) % edges;

213 [[nodiscard]] EdgeVertices farEdgeOf(HexSphereCell cell, std::int32_t d, HexSphereCell neighbour) const noexcept {
214 const std::int32_t back = map_.oppositeDirection(cell, d);
215 const std::int32_t nEdges = map_.neighborCount(neighbour);
216 if (back < 0 || nEdges <= 0) return EdgeVertices{};
217 return sphereEdge(solidCornerOf(neighbour, (back + 1) % nEdges), solidCornerOf(neighbour, back));

221 [[nodiscard]] bool riverThrough(HexSphereCell cell, std::int32_t d, const HexCellData* cellData) const noexcept {
222 if (cellData != nullptr && cellData->flags.hasRiverThrough(static_cast<HexDirection>(d))) return true;
223 const HexSphereCell neighbour = map_.neighbor(cell, d);
224 if (neighbour == kNoHexSphereCell) return false;
225 const HexCellData* other = map_.cellAt(neighbour);
226 const std::int32_t back = map_.oppositeDirection(cell, d);
227 return other != nullptr && back >= 0 && other->flags.hasRiverThrough(static_cast<HexDirection>(back));
228 }

233 void buildCell(HexSphereCell cell) {
234 const HexCellData* cellData = map_.cellAt(cell);
235 if (cellData == nullptr) return;
236
237 const HexVec3 center = map_.cellPosition(cell);
238 const std::int32_t edges = map_.neighborCount(cell);
239 for (std::int32_t d = 0; d < edges; ++d) {
240 EdgeVertices near = solidEdgeOf(cell, d);
241 if (riverThrough(cell, d, cellData)) {
242 // Carve a shallow channel along the river centreline (the middle sample only).
243 near.v3 = normalize(near.v3) * (lengthOf(near.v3) + streamBedDrop_);
244 }
245 appendEdgeFan(center, near, cellData);
246
247 const HexSphereCell neighbour = map_.neighbor(cell, d);
248 if (neighbour == kNoHexSphereCell) continue;
249 const HexCellData* neighbourData = map_.cellAt(neighbour);
250 if (neighbourData == nullptr) continue;
251 appendConnection(cell, d, near, cellData, neighbour, neighbourData);
252 }
253 }

256 void appendConnection(HexSphereCell cell, std::int32_t d, const EdgeVertices& near, const HexCellData* cellData,
257 HexSphereCell neighbour, const HexCellData* neighbourData) {
258 // Edge ownership rule. The planar mesher leans on the chunk grid here; a sphere
259 // has none, so the lower cell id owns the edge instead. That is a total order
260 // over the same set of shared edges, so every boundary is still emitted exactly
261 // once, and never by both of its cells.
262 if (cell > neighbour) return;
263
264 const EdgeVertices far = farEdgeOf(cell, d, neighbour);
265
266 const HexTerrainWeights nearWeights = HexTerrainWeights::primary();
267 const HexTerrainWeights farWeights = HexTerrainWeights::blend(1.f);
268
269 switch (map_.edgeTypeTo(cell, neighbour)) {
270 case HexEdgeType::Flat:
271 appendBlendStrip(near, far, nearWeights, farWeights, cellData, neighbourData);
272 break;
273 case HexEdgeType::Slope:
274 appendEdgeTerraces(near, nearWeights, far, farWeights, cellData, neighbourData);
275 break;
276 case HexEdgeType::Cliff:
277 // The blend strip would already be a radial wall here, but its winding
278 // flips with the sign of the elevation step; the vertical winding is
279 // correct in both directions. Split into the same four sub-quads as a
280 // flat strip so the wall does not leave a T-junction against the fan.
281 for (std::int32_t i = 0; i < 4; ++i) {
282 emitQuadVertical(sampleAt(near, i), sampleAt(near, i + 1), sampleAt(far, i), sampleAt(far, i + 1),
283 nearWeights, nearWeights, farWeights, farWeights, cellData, neighbourData,
284 cellData);
285 }
286 break;
287 }
288
289 appendCorner(cell, d, cellData, neighbour, neighbourData);
290 }

303 void appendCorner(HexSphereCell cell, std::int32_t d, const HexCellData* cellData, HexSphereCell neighbour,
304 const HexCellData* neighbourData) {
305 const std::int32_t edges = map_.neighborCount(cell);
306 const std::int32_t nextD = (d + 1) % edges;
307 const HexSphereCell nextCell = map_.neighbor(cell, nextD);
308 if (nextCell == kNoHexSphereCell) return;
309 const HexCellData* nextData = map_.cellAt(nextCell);
310 if (nextData == nullptr) return;
311
312 // The corner between edge `d` and edge `d + 1` of `cell` is corner `d + 1`. Each
313 // neighbour reaches that same corner from the far end of its own shared edge,
314 // which the topology stores in the opposite order.
315 const std::int32_t back = map_.oppositeDirection(cell, d);
316 const std::int32_t back2 = map_.oppositeDirection(cell, nextD);
317 const std::int32_t nextEdges = map_.neighborCount(nextCell);
318 if (back < 0 || back2 < 0 || nextEdges <= 0) return;
319
320 CornerCells corner{};
321 corner.up = solidCornerOf(cell, nextD);
322 corner.right = solidCornerOf(neighbour, back);
323 corner.left = solidCornerOf(nextCell, (back2 + 1) % nextEdges);
324 corner.upCell = cellData;
325 corner.rightCell = neighbourData;
326 corner.leftCell = nextData;
327 appendCornerTriangles(corner);
328 }

331 void appendCornerTriangles(const CornerCells& corner) {
332 const std::array<HexVec3, 3> positions{corner.up, corner.left, corner.right};
333 const std::array<const HexCellData*, 3> cells{corner.upCell, corner.leftCell, corner.rightCell};
334 std::array<std::int32_t, 3> order{0, 1, 2};
335 for (std::int32_t i = 0; i < 2; ++i) {
336 for (std::int32_t j = i + 1; j < 3; ++j) {
337 if (elevationOf(cells[static_cast<std::size_t>(order[static_cast<std::size_t>(j)])]) <
338 elevationOf(cells[static_cast<std::size_t>(order[static_cast<std::size_t>(i)])])) {
339 std::swap(order[static_cast<std::size_t>(i)], order[static_cast<std::size_t>(j)]);
340 }
341 }
342 }
343
344 const HexVec3 bottom = positions[static_cast<std::size_t>(order[0])];
345 const HexVec3 low = positions[static_cast<std::size_t>(order[1])];
346 const HexVec3 high = positions[static_cast<std::size_t>(order[2])];
347 const HexCellData* bottomCell = cells[static_cast<std::size_t>(order[0])];
348 const HexCellData* lowCell = cells[static_cast<std::size_t>(order[1])];
349 const HexCellData* highCell = cells[static_cast<std::size_t>(order[2])];
350
351 const HexEdgeType lowEdge = edgeType(elevationOf(bottomCell), elevationOf(lowCell));
352 const HexEdgeType highEdge = edgeType(elevationOf(bottomCell), elevationOf(highCell));
353
354 if (lowEdge == HexEdgeType::Slope && highEdge == HexEdgeType::Slope) {
355 cornerTerraces(bottom, low, high, bottomCell, lowCell, highCell);
356 } else if (lowEdge == HexEdgeType::Slope && highEdge == HexEdgeType::Flat) {
357 cornerTerraces(low, high, bottom, lowCell, highCell, bottomCell);
358 } else if (lowEdge == HexEdgeType::Flat && highEdge == HexEdgeType::Slope) {
359 cornerTerraces(high, bottom, low, highCell, bottomCell, lowCell);
360 } else if (lowEdge == HexEdgeType::Slope && highEdge == HexEdgeType::Cliff) {
361 cornerTerracesCliff(bottom, low, high, bottomCell, lowCell, highCell, lowEdge);
362 } else if (lowEdge == HexEdgeType::Cliff && highEdge == HexEdgeType::Slope) {
363 cornerCliffTerraces(bottom, low, high, bottomCell, lowCell, highCell, highEdge);
364 } else if (edgeType(elevationOf(lowCell), elevationOf(highCell)) == HexEdgeType::Slope) {
365 if (elevationOf(lowCell) < elevationOf(highCell)) {
366 cornerCliffTerraces(high, bottom, low, highCell, bottomCell, lowCell, highEdge);
367 } else {
368 cornerTerracesCliff(low, high, bottom, lowCell, highCell, bottomCell, lowEdge);
369 }
370 } else {
371 // Mirror of the reference winding: this engine's front face is the opposite
372 // handedness, so the flat corner fan is emitted high-before-low.
373 emitTriangle(bottom, high, low, HexTerrainWeights::primary(), bottomCell, lowCell, highCell);
374 }
375 }

378 void cornerTerraces(const HexVec3& bottom, const HexVec3& left, const HexVec3& right, const HexCellData* bottomCell,
379 const HexCellData* leftCell, const HexCellData* rightCell) {
380 const HexTerrainWeights bottomWeights = HexTerrainWeights::primary();
381 const HexTerrainWeights leftWeights = HexTerrainWeights::primary();
382 const HexTerrainWeights rightWeights = HexTerrainWeights::primary();
383
384 HexVec3 lastLeft = bottom;
385 HexVec3 lastRight = bottom;
386 HexTerrainWeights lastLeftWeights = bottomWeights;
387 HexTerrainWeights lastRightWeights = bottomWeights;
388 for (std::int32_t step = 1; step < HexMetrics::kTerracesPerSlope * 2; ++step) {
389 const float t = static_cast<float>(step) * HexMetrics::kHorizontalTerraceStepSize;
390 const HexTerrainWeights wl = HexTerrainWeights::lerp(bottomWeights, leftWeights, t);
391 const HexTerrainWeights wr = HexTerrainWeights::lerp(bottomWeights, rightWeights, t);
392 const HexVec3 boundaryLeft = sphereTerraceLerp(bottom, left, step);
393 const HexVec3 boundaryRight = sphereTerraceLerp(bottom, right, step);
394
395 emitQuadForward(boundaryLeft, boundaryRight, lastLeft, lastRight, wl, wr, lastLeftWeights, lastRightWeights,
396 bottomCell, leftCell, rightCell);
397 lastLeft = boundaryLeft;
398 lastRight = boundaryRight;
399 lastLeftWeights = wl;
400 lastRightWeights = wr;
401 }
402 emitQuadForward(lastLeft, lastRight, left, right, lastLeftWeights, lastRightWeights, leftWeights, rightWeights,
403 bottomCell, leftCell, rightCell);
404 }

412 void appendBoundaryTriangle(HexVec3 bottom, const HexTerrainWeights& bottomWeights, HexVec3 left,
413 const HexTerrainWeights& leftWeights, HexVec3 right,
414 const HexTerrainWeights& rightWeights, const HexCellData* bottomCell,
415 const HexCellData* leftCell, const HexCellData* rightCell) {
416 HexVec3 leftSide = sphereTerraceLerp(bottom, left, 1);
417 HexVec3 rightSide = sphereTerraceLerp(bottom, right, 1);
418 for (std::int32_t step = 1; step < HexMetrics::kTerracesPerSlope * 2; ++step) {
419 const float t = static_cast<float>(step) * HexMetrics::kHorizontalTerraceStepSize;

433 void cornerTerracesCliff(const HexVec3& bottom, const HexVec3& low, const HexVec3& high,

435 HexEdgeType lowEdge) {
436 const HexTerrainWeights bottomWeights = HexTerrainWeights::primary();
437 const HexTerrainWeights lowWeights = HexTerrainWeights::primary();
438 const HexTerrainWeights highWeights = HexTerrainWeights::primary();
439
440 const float span = static_cast<float>(elevationOf(highCell) - elevationOf(bottomCell));
441 const float blend = span > 1e-6f ? 1.f / span : 1.f;
442 const HexVec3 boundary = lerp(bottom, high, blend);

450 if (lowEdge == HexEdgeType::Slope) {
451 appendBoundaryTriangle(boundary, boundaryWeights, bottom, bottomWeights, high, highWeights, bottomCell,

453 } else {

520 void emitQuadForward(const HexVec3& p0, const HexVec3& p1, const HexVec3& p2, const HexVec3& p3,
521 const HexTerrainWeights& w0, const HexTerrainWeights& w1, const HexTerrainWeights& w2,

622class SphereWaterMesher {
623public:
624 SphereWaterMesher(const HexSphereMap& map, HexMeshData& out) noexcept : map_(map), out_(out) {}
625
627 void run() {
628 for (HexSphereCell cell = 0; cell < map_.cellCount(); ++cell) buildCell(cell);
629 }
630
631private:
639 [[nodiscard]] float waterRadius(std::int32_t waterLevel) const noexcept {
640 const float offset = HexMetrics::kWaterElevationOffset / HexMetrics::kElevationStep;
641 return map_.sphereRadius() + (static_cast<float>(waterLevel) + offset) * map_.elevationStep();
642 }
643
644 void buildCell(HexSphereCell cell) {
645 const HexCellData* data = map_.cellAt(cell);
646 if (data == nullptr || !data->values.isUnderwater()) return;
647
648 // The shore parameter the water shader expects: 1 where the water ends against
649 // land, 0 in open ocean.
650 float shore = 0.f;
651 const std::int32_t edges = map_.neighborCount(cell);
652 for (std::int32_t d = 0; d < edges; ++d) {
653 const HexCellData* neighbour = map_.cellAt(map_.neighbor(cell, d));
654 if (neighbour == nullptr || !neighbour->values.isUnderwater()) {
655 shore = 1.f;
656 break;
657 }
658 }
659
660 const float radius = waterRadius(data->values.waterLevel());
661 const HexVec3 center = map_.direction(cell) * radius;
662 const std::int32_t corners = map_.cornerCountOf(cell);
663 // Same winding as the terrain fan: centre, then the corners in increasing index.
664 for (std::int32_t k = 0; k < corners; ++k) {
665 const HexVec3 a = map_.cornerDirection(cell, k) * radius;
666 const HexVec3 b = map_.cornerDirection(cell, (k + 1) % corners) * radius;
667 const std::uint32_t i0 = static_cast<std::uint32_t>(out_.vertexCount());
668 out_.addVertex(center, shore, 0.f);
669 out_.addVertex(a, shore, 0.f);
670 out_.addVertex(b, shore, 0.f);
671 out_.addTriangle(i0, i0 + 1u, i0 + 2u);
672 }
673 }
674
675 const HexSphereMap& map_;
676 HexMeshData& out_;
677};
