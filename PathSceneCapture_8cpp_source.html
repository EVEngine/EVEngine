17struct TextureKey {
18 Texture* texture;
19 std::array<uint32_t, 9> parameters;
20 bool operator==(const TextureKey&) const = default;
21};
22struct TextureHash {
23 size_t operator()(const TextureKey& key) const {
24 auto hash = std::hash<Texture*>{}(key.texture);
25 for (auto value : key.parameters) hash ^= size_t(value) + size_t(0x9e3779b9u) + (hash << 6) + (hash >> 2);
26 return hash;
27 }
28};
29struct GeometryKey {
30 Mesh* mesh;
31 std::array<uint32_t, 16> material;
32 bool operator==(const GeometryKey&) const = default;
33};
34struct GeometryHash {
35 size_t operator()(const GeometryKey& key) const {
36 auto hash = std::hash<Mesh*>{}(key.mesh);
37 for (auto value : key.material) hash ^= size_t(value) + size_t(0x9e3779b9u) + (hash << 6) + (hash >> 2);
38 return hash;
39 }
40};
41GeometryKey geometryKey(Mesh* mesh, const PathMaterial& material) {
42 const auto bits = [](float value) { return std::bit_cast<uint32_t>(value == 0.f ? 0.f : value); };
43 return {mesh,
44 {bits(material.baseColor.x), bits(material.baseColor.y), bits(material.baseColor.z),
45 bits(material.emission.x), bits(material.emission.y), bits(material.emission.z), bits(material.roughness),
46 bits(material.metallic), material.baseColorTexture, bits(material.opacity), bits(material.alphaCutoff),
47 uint32_t(material.alphaMode), material.metallicRoughnessTexture, material.emissionTexture,
48 material.normalTexture, bits(material.normalScale)}};
49}

51Result<PathSceneToken> preparePathWorld(ecs::Table& world, Graphics& graphics, IRayTracing& provider) {
52 EV_PROFILE_MODULE("graphics", "Path.WorldCapture");
53 const auto unsupported = [](const char* message) {
54 return Result<PathSceneToken>::failure(
55 Diagnostic::error(DiagnosticCode::Unsupported, message, "graphics.path_scene"));
56 };
57 std::vector<PathGeometry> geometry;
58 std::vector<PathMaterial> materials;
59 std::vector<PathInstance> instances;
60 std::vector<PathLight> lights;
61 std::vector<PathColorTexture> textures;
62 std::unordered_map<TextureKey, uint32_t, TextureHash> textureIndices;
63 std::unordered_map<Material*, PathMaterial> materialValues;
64 const auto registerTexture = [&](const PathColorTexture& input) {
65 const auto bits = [](float value) { return std::bit_cast<uint32_t>(value == 0.f ? 0.f : value); };
66 TextureKey key{input.sourceTexture,
67 {uint32_t(input.srgbDecode), bits(input.offset.x), bits(input.offset.y), bits(input.scale.x),
68 bits(input.scale.y), bits(input.rotation), uint32_t(input.filter), uint32_t(input.addressU),
69 uint32_t(input.addressV)}};
70 auto [entry, inserted] = textureIndices.try_emplace(key, uint32_t(textures.size()));
71 if (inserted) textures.push_back(input);
72 return entry->second;
73 };
74 const auto registerBasicTexture = [&](Texture* texture, bool srgbDecode) {
75 if (!texture) return UINT32_MAX;
76 PathColorTexture input;
77 input.sourceTexture = texture;
78 input.srgbDecode = srgbDecode;
79 return registerTexture(input);
80 };
81 // Keys live only during capture. Equal material values share geometry even
82 // when authored through distinct Material objects or legacy component fields.
83 std::unordered_map<GeometryKey, uint32_t, GeometryHash> geometryIndices;
84 // Scope: Renderable3D descendants with Transform3D+MeshRenderer, and Light3D
85 // descendants with Data. Read only; no structural mutation/events/RNG/time.
86 // Phase: explicit preparation before frame recording. Service: IRayTracing.
87 auto renderables = ecs::View<Renderable3D, Renderable3D::Transform3D, Renderable3D::MeshRenderer>(world);
88 for (auto it = renderables.begin(); it != renderables.end(); ++it) {
89 auto [transform, renderer] = *it;
90 if (!renderer->visible) continue;
91 if (renderer->instances || renderer->isHair)
92 return unsupported("Path world requires supported instance and hair geometry");
93 const int count = renderer->partCount > 0 ? renderer->partCount : 1;
94 if (count > Renderable3D::MeshRenderer::kMaxParts)
95 return unsupported("Path world part count exceeds renderer storage");
96 for (int part = 0; part < count; ++part) {
97 auto* mesh =
98 renderer->partCount > 0
99 ? renderer->parts[part].mesh
100 : (renderer->lodCount > 0 && renderer->lodMeshes[0] ? renderer->lodMeshes[0] : renderer->mesh);
101 if (!mesh) continue;
102 auto* material = renderer->partCount > 0 && renderer->parts[part].material ? renderer->parts[part].material
103 : renderer->material;
104 PathMaterial value;
105 if (material && materialValues.contains(material)) {
106 value = materialValues.at(material);
107 } else if (material) {
108 if (material->virtualTextureMode() != MaterialVirtualTextureMode::Conventional ||
109 material->getShader() || material->getHeightTexture() ||
110 (material->surfaceMode() == SurfaceMode::Transparent &&
111 material->blendMode() != BlendMode::Alpha) ||
112 material->getShadingModel() != "pbr")
113 return unsupported("Path world material conversion is unavailable for this surface");
114 value.baseColor = {material->getTintR(), material->getTintG(), material->getTintB()};
115 value.opacity = material->getTintA();
116 value.alphaCutoff = material->getAlphaCutoff();
117 value.alphaMode = material->surfaceMode() == SurfaceMode::Masked ? PathAlphaMode::Masked
118 : material->surfaceMode() == SurfaceMode::Transparent ? PathAlphaMode::Blend
119 : PathAlphaMode::Opaque;
120 value.roughness = material->getRoughness();
121 value.metallic = material->getMetallic();
122 if (material->hasPbrSurface()) {
123 auto pbr = material->pbrSurface();
124 value.emission =
125 glm::vec3(pbr.emissive[0], pbr.emissive[1], pbr.emissive[2]) * pbr.emissiveStrength;
126 pbr.emissive = {0.f, 0.f, 0.f};
127 pbr.emissiveStrength = 1.f;
128 value.normalScale = pbr.normalScale;
129 pbr.normalScale = 1.f;
130 for (auto slot : {PbrTextureSlot::BaseColor, PbrTextureSlot::MetallicRoughness,
131 PbrTextureSlot::Emissive, PbrTextureSlot::Normal}) {
132 auto& binding = pbr.textures[size_t(slot)];
133 if (binding.texture) {
134 if (binding.texcoord != 0)
135 return unsupported(
136 "Path world material requires UV0; additional UV sets are not captured yet");
137 PathColorTexture input;
138 input.sourceTexture = binding.texture;
139 input.srgbDecode = binding.srgbDecode;
140 input.offset = {binding.offset[0], binding.offset[1]};
141 input.scale = {binding.scale[0], binding.scale[1]};
142 input.rotation = binding.rotation;
143 input.filter =
144 binding.magFilter == 9728 ? PathTextureFilter::Nearest : PathTextureFilter::Linear;
145 const auto address = [](uint32_t wrap) {
146 return wrap == 10497
147 ? PathTextureAddress::Repeat
148 : (wrap == 33648 ? PathTextureAddress::Mirror : PathTextureAddress::Clamp);
149 };
150 input.addressU = address(binding.wrapS);
151 input.addressV = address(binding.wrapT);
152 auto index = registerTexture(input);
153 if (slot == PbrTextureSlot::BaseColor) value.baseColorTexture = index;
154 if (slot == PbrTextureSlot::MetallicRoughness) value.metallicRoughnessTexture = index;
155 if (slot == PbrTextureSlot::Emissive) value.emissionTexture = index;
156 if (slot == PbrTextureSlot::Normal) value.normalTexture = index;
157 }
158 binding = {};
159 }
160 // Defaulted value comparison also covers newly added extension fields:
161 // an unimplemented non-default extension must never disappear silently.
162 if (!(pbr == PbrSurface{})) return unsupported("Path world PBR extension is not yet supported");
163 } else {
164 value.baseColorTexture = registerBasicTexture(material->getAlbedoTexture(), true);
165 value.normalTexture = registerBasicTexture(material->getNormalTexture(), false);
166 }
167 materialValues.emplace(material, value);
168 } else {
169 if (renderer->heightTexture || renderer->shader || !renderer->receiveLight)
170 return unsupported("Path world legacy material conversion is unavailable for this surface");
171 value.baseColor = {renderer->r, renderer->g, renderer->b};
172 value.opacity = renderer->a;
173 value.alphaMode = renderer->a == 1.f ? PathAlphaMode::Opaque : PathAlphaMode::Blend;
174 value.roughness = renderer->roughness;
175 value.metallic = renderer->metallic;
176 value.baseColorTexture = registerBasicTexture(renderer->texture, true);
177 value.normalTexture = registerBasicTexture(renderer->normalTexture, false);
178 }
179 const auto [entry, inserted] =
180 geometryIndices.try_emplace(geometryKey(mesh, value), uint32_t(geometry.size()));
181 if (inserted) {
182 PathGeometry input;
183 input.sourceMesh = mesh;
184 input.material = uint32_t(materials.size());
185 geometry.push_back(input);
186 materials.push_back(value);
187 }
188 instances.push_back({entry->second, detail::modelFromTransform(*transform)});
189 }
190 }
191 auto lightView = ecs::View<Light3D, Light3D::Data>(world);
192 for (auto it = lightView.begin(); it != lightView.end(); ++it) {
193 auto [light] = *it;
194 if (!light->enabled) continue;
195 if (light->volumetric || light->volumetricOnly ||
196 (light->type != "point" && light->type != "dir" && light->type != "spot"))
197 return unsupported("Path world light conversion requires point, directional or spot surface lights");
198 PathLight value;
199 value.kind = light->type == "dir" ? PathLightKind::Directional : PathLightKind::Point;
200 if (light->type == "spot") {
201 value.kind = PathLightKind::Spot;
202 value.beamDirection = {light->dx, light->dy, light->dz};
203 lightSpotCosines(light->spotAngleDeg, light->spotSoftness, value.cosOuter, value.cosInner);
204 value.cosInner = std::min(value.cosInner, 1.f);
205 }
206 value.position = {light->x, light->y, light->z};
207 value.directionToLight = {light->dx, light->dy, light->dz};
208 value.strength = glm::vec3(light->r, light->g, light->b) * light->intensity;
209 lights.push_back(value);
210 }
211 return provider.preparePathScene(&graphics, {geometry, instances, materials, lights, textures});
212}
