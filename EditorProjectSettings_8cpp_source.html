73EditorResult<void> ProjectSettingsTarget::applyDomainOperation(const DomainOperation& domainOperation) {
74 if (domainOperation.target != TargetId(id_) || domainOperation.type != "project.setting.set.v1")
75 return settingsError<void>(EditorStatus::Rejected, "editor.settings.operation",
76 "Unsupported or mismatched settings operation");
77 const auto* pathEntry = field(domainOperation.payload, "path");
78 const auto* value = field(domainOperation.payload, "value");
79 const auto* path = pathEntry ? pathEntry->getIf<std::string>() : nullptr;
80 const auto* setting = path ? descriptor(PropertyPath(*path)) : nullptr;
81 if (!setting || !value)
82 return settingsError<void>(EditorStatus::Rejected, "editor.settings.payload",
83 "Settings operation payload is incomplete");
84 auto valid = validatePropertyValue(setting->property, *value);
85 if (!valid.ok()) return valid;
86 if (setting->sensitive) {
87 const auto* reference = value->getIf<std::string>();
88 if (!reference || !reference->starts_with("secret://"))
89 return settingsError<void>(
90 EditorStatus::Rejected, "editor.settings.raw-secret",
91 "Sensitive settings accept secret:// references only; raw credentials are never stored");
92 }
93 if (values_.at(*path) == *value) return eve::editing::noOp();
94 values_[*path] = *value;
95 if (setting->requiresRestart) restartDirty_[*path] = true;
96 ++revision_;
97 dirty_.include(0, 0);
98 return eve::editing::applied<void>();
99}

116PropertyReadResult ProjectSettingsTarget::read(const SelectionSnapshot& selection, const PropertyPath& path) const {
117 if (!selectionMatches(selection))
118 return {PropertyReadState::Error,
119 {},
120 {eve::editing::ruleDiagnostic(eve::DiagnosticCode::InvalidArgument, RuleId("editor.settings.selection"),
121 DiagnosticSeverity::Error, "Selection does not match settings target")}};
122 const auto found = values_.find(path.value());
123 if (found == values_.end()) return {};
124 const auto* setting = descriptor(path);
125 if (setting && setting->sensitive) {
126 const auto* reference = found->second.getIf<std::string>();
127 if (!reference)
128 return {PropertyReadState::Error,
129 {},
130 {eve::editing::ruleDiagnostic(
131 eve::DiagnosticCode::PreconditionViolation, RuleId("editor.settings.invalid-sensitive-schema"),
132 DiagnosticSeverity::Error, "Sensitive setting schema must use a string value")}};
133 if (!reference->empty()) return {PropertyReadState::Value, EditorValue("secret://redacted"), {}};
134 }
135 return {PropertyReadState::Value, found->second, {}};
136}

138EditorResult<DomainOperation> ProjectSettingsTarget::makeSet(const SelectionSnapshot& selection,
139 const PropertyPath& path, const EditorValue& value,
140 PropertySetMode mode) const {
141 if (!selectionMatches(selection))
142 return settingsError<DomainOperation>(EditorStatus::Rejected, "editor.settings.selection",
143 "Selection does not match settings target");
144 const auto* setting = descriptor(path);
145 if (!setting)
146 return settingsError<DomainOperation>(EditorStatus::Unsupported, "editor.settings.property",
147 "Unknown project setting: " + path.value());
148 const EditorValue candidate = mode == PropertySetMode::Reset ? setting->property.defaultValue : value;
149 auto valid = validatePropertyValue(setting->property, candidate);
150 if (!valid.ok())
151 return settingsError<DomainOperation>(valid.code(), "editor.settings.value",
152 "Project setting value is invalid");
153 if (setting->sensitive) {
154 const auto* reference = candidate.getIf<std::string>();
155 if (!reference || (!reference->empty() && !reference->starts_with("secret://")))
156 return settingsError<DomainOperation>(
157 EditorStatus::Rejected, "editor.settings.raw-secret",
158 "Sensitive settings accept an empty value or secret:// reference only");
159 }
160 return eve::editing::applied<DomainOperation>(operation(id_, path, candidate, values_.at(path.value())));
161}

175EditorValue ProjectSettingsTarget::snapshotValue() const {
176 EditorValue::Object values;
177 for (const auto& setting : schema_.settings) {
178 const auto& value = values_.at(setting.property.path.value());
179 const auto* reference = setting.sensitive ? value.getIf<std::string>() : nullptr;
180 if (!setting.sensitive || (reference && (reference->empty() || reference->starts_with("secret://"))))
181 values.emplace(setting.property.path.value(), value);
182 }
183 return EditorValue::Object{
184 {"schema", schema_.typeId}, {"schemaVersion", int64_t{schema_.version}}, {"values", std::move(values)}};
185}

187EditorResult<void> ProjectSettingsTarget::loadSnapshot(const EditorValue& snapshot) {
188 const auto* schemaEntry = field(snapshot, "schema");
189 const auto* versionEntry = field(snapshot, "schemaVersion");
190 const auto* valuesEntry = field(snapshot, "values");
191 const auto* schema = schemaEntry ? schemaEntry->getIf<std::string>() : nullptr;
192 const auto* version = versionEntry ? versionEntry->getIf<int64_t>() : nullptr;
193 const auto* values = valuesEntry ? valuesEntry->getIf<EditorValue::Object>() : nullptr;
194 if (!schema || *schema != schema_.typeId || !version || *version != schema_.version || !values)
195 return settingsError<void>(EditorStatus::Rejected, "editor.settings.snapshot",
196 "Settings snapshot schema does not match this target");
197 ProjectSettingsTarget candidate(id_, schema_);
198 for (const auto& [path, value] : *values) {
199 const auto* setting = candidate.descriptor(PropertyPath(path));
200 if (!setting)
201 return settingsError<void>(EditorStatus::Rejected, "editor.settings.unknown-snapshot-key",
202 "Settings snapshot contains an unknown key");
203 auto valid = validatePropertyValue(setting->property, value);
204 if (!valid.ok())
205 return settingsError<void>(EditorStatus::Rejected, "editor.settings.invalid-snapshot-value",
206 "Settings snapshot contains an invalid value");
207 if (setting->sensitive) {
208 const auto* reference = value.getIf<std::string>();
209 if (!reference || (!reference->empty() && !reference->starts_with("secret://")))
210 return settingsError<void>(EditorStatus::Rejected, "editor.settings.raw-secret",
211 "Settings snapshot contains a raw sensitive value");
212 }
213 candidate.values_[path] = value;
214 }
215 values_ = std::move(candidate.values_);
216 restartDirty_.clear();
217 ++revision_;
218 dirty_.include(0, 0);
219 return eve::editing::applied<void>();
220}
