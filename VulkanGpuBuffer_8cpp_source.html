28 if (!device_ || !buffer_ || !src || nbytes == 0) return;
29 if (dstOffset + nbytes > size_)
30 throw Exception("GpuBuffer.write: out of range (offset=%llu size=%llu capacity=%llu)",
31 (unsigned long long)dstOffset, (unsigned long long)nbytes,
32 (unsigned long long)size_);
33
34 auto *vkg = requireVulkanGraphics();
35 auto queue = computeQueue(vkg);
36 auto pool = computeCommandPool(vkg);
37
38 if (hostVisible_) {
39 void *ptr = (*device_)->mapMemory(memory_, dstOffset, nbytes, vk::MemoryMapFlags{});
40 std::memcpy(ptr, src, size_t(nbytes));
41 (*device_)->unmapMemory(memory_);
42 return;
43 }
44
45 using buf = vk::BufferUsageFlagBits;

50 vkb::executeImmediately(device_->instance, pool, queue, [&](vk::CommandBuffer cb) {
