27std::uint32_t little32(std::span<const std::uint8_t> bytes, std::size_t offset) {
28 return std::uint32_t(bytes[offset]) | (std::uint32_t(bytes[offset + 1]) << 8) |
29 (std::uint32_t(bytes[offset + 2]) << 16) | (std::uint32_t(bytes[offset + 3]) << 24);

32void put32(std::vector<std::uint8_t>& bytes, std::uint32_t value) {

43Result<Document> parseDocument(const GltfImportRequest& request) {
44 if (request.documentBytes.empty() || request.documentBytes.size() > request.limits.maximumSourceBytes)
45 return detail::failure<Document>(DiagnosticCode::InvalidArgument, "glTF document size is outside limits");
46 std::string json;
47 std::vector<std::uint8_t> binary;
48 const auto bytes = std::span<const std::uint8_t>(request.documentBytes);
49 if (bytes.size() >= 12 && little32(bytes, 0) == 0x46546c67) {
50 if (little32(bytes, 4) != 2 || little32(bytes, 8) != bytes.size() || bytes.size() < 20)
51 return detail::failure<Document>(DiagnosticCode::ParseError, "GLB header is invalid");
52 std::size_t cursor = 12;
53 const std::uint32_t jsonSize = little32(bytes, cursor);
54 const std::uint32_t jsonType = little32(bytes, cursor + 4);
55 cursor += 8;
56 if (jsonType != 0x4e4f534a || jsonSize > bytes.size() - cursor)
57 return detail::failure<Document>(DiagnosticCode::ParseError, "GLB JSON chunk is invalid");
58 json.assign(reinterpret_cast<const char*>(bytes.data() + cursor), jsonSize);
59 while (!json.empty() && (json.back() == ' ' || json.back() == '\0')) json.pop_back();
60 cursor += jsonSize;
61 if (cursor < bytes.size()) {
62 if (bytes.size() - cursor < 8)
63 return detail::failure<Document>(DiagnosticCode::ParseError, "GLB trailing chunk is truncated");
64 const std::uint32_t binarySize = little32(bytes, cursor);
65 const std::uint32_t binaryType = little32(bytes, cursor + 4);
66 cursor += 8;
67 if (binaryType != 0x004e4942 || binarySize > bytes.size() - cursor || cursor + binarySize != bytes.size())
68 return detail::failure<Document>(DiagnosticCode::ParseError, "GLB BIN chunk is invalid");
69 binary.assign(bytes.begin() + cursor, bytes.end());
70 }
71 } else {
72 json.assign(request.documentBytes.begin(), request.documentBytes.end());
73 }
74 auto parsed = Value::fromJson(json);
75 if (!parsed) return Result<Document>::failure(parsed.status());
76 return Result<Document>::success({std::move(parsed).takeValue(), std::move(binary)});
77}

79bool safeUri(std::string_view uri, const AssetImportLimits& limits) {
80 if (uri.empty() || uri.size() > limits.maximumStringBytes || uri.front() == '/' ||
81 uri.find('\\') != std::string_view::npos || uri.starts_with("data:") || uri.find(':') != std::string_view::npos)
82 return false;
83 std::size_t start = 0;
84 for (std::size_t index = 0; index <= uri.size(); ++index) {
85 if (index != uri.size() && uri[index] != '/') continue;
86 const auto segment = uri.substr(start, index - start);
87 if (segment.empty() || segment == "." || segment == "..") return false;
88 start = index + 1;
89 }
90 return true;
91}

93Result<std::vector<std::span<const std::uint8_t>>> resolveBuffers(
94 const Value::Object& root, const Document& document, const GltfImportRequest& request) {
95 const Value* buffersValue = member(root, "buffers");
96 const auto* buffers = buffersValue ? buffersValue->getIf<Value::Array>() : nullptr;
97 if (!buffers || buffers->empty())
98 return detail::failure<std::vector<std::span<const std::uint8_t>>>(DiagnosticCode::ParseError,
99 "glTF buffers are required", "$.buffers");
100 std::vector<std::span<const std::uint8_t>> result;
101 result.reserve(buffers->size());
102 std::uint64_t totalBytes = 0;
103 for (std::size_t index = 0; index < buffers->size(); ++index) {
104 const auto* object = (*buffers)[index].getIf<Value::Object>();
105 if (!object)
106 return detail::failure<std::vector<std::span<const std::uint8_t>>>(DiagnosticCode::ParseError,
107 "glTF buffer must be an object");
108 auto declared = unsignedValue(member(*object, "byteLength"),
109 "$.buffers[" + std::to_string(index) + "].byteLength");
110 if (!declared) return Result<std::vector<std::span<const std::uint8_t>>>::failure(declared.status());
111 std::span<const std::uint8_t> data;
112 const Value* uriValue = member(*object, "uri");
113 if (uriValue) {
114 if (!uriValue->isString() || !safeUri(uriValue->asString(), request.limits))
115 return detail::failure<std::vector<std::span<const std::uint8_t>>>(DiagnosticCode::Unsupported,
116 "external glTF buffer URI is unsafe or unsupported");
117 const auto found = request.externalResources.find(uriValue->asString());
118 if (found == request.externalResources.end())
119 return detail::failure<std::vector<std::span<const std::uint8_t>>>(DiagnosticCode::NotFound,
120 "external glTF buffer was not supplied",
121 uriValue->asString());
122 data = found->second;
123 } else {
124 if (index != 0 || document.binaryChunk.empty())
125 return detail::failure<std::vector<std::span<const std::uint8_t>>>(DiagnosticCode::ParseError,
126 "buffer without URI requires the first GLB BIN chunk");
127 data = document.binaryChunk;
128 }
129 if (declared.value() > data.size())
130 return detail::failure<std::vector<std::span<const std::uint8_t>>>(DiagnosticCode::ParseError,
131 "glTF buffer is shorter than byteLength");
132 if (declared.value() > request.limits.maximumSourceBytes ||
133 totalBytes > request.limits.maximumSourceBytes - declared.value())
134 return detail::failure<std::vector<std::span<const std::uint8_t>>>(DiagnosticCode::InvalidArgument,
135 "glTF buffer budget is exceeded");
136 totalBytes += declared.value();
137 result.push_back(data.first(static_cast<std::size_t>(declared.value())));
138 }
139 return Result<std::vector<std::span<const std::uint8_t>>>::success(std::move(result));
140}

158Result<Accessor> accessorAt(const Value::Object& root,
159 const std::vector<std::span<const std::uint8_t>>& buffers,
160 std::uint64_t accessorIndex, const AssetImportLimits& limits) {
161 const auto* accessors = member(root, "accessors") ? member(root, "accessors")->getIf<Value::Array>() : nullptr;
162 const auto* views = member(root, "bufferViews") ? member(root, "bufferViews")->getIf<Value::Array>() : nullptr;
163 if (!accessors || !views || accessorIndex >= accessors->size())
164 return detail::failure<Accessor>(DiagnosticCode::ParseError, "glTF accessor index is invalid");
165 const auto* object = (*accessors)[accessorIndex].getIf<Value::Object>();
166 if (!object || member(*object, "sparse"))
167 return detail::failure<Accessor>(DiagnosticCode::Unsupported, "sparse or malformed glTF accessor is unsupported");
168 auto viewIndex = unsignedValue(member(*object, "bufferView"), "accessor.bufferView");
169 auto count = unsignedValue(member(*object, "count"), "accessor.count");
170 auto component = unsignedValue(member(*object, "componentType"), "accessor.componentType");
171 auto accessorOffset = unsignedValue(member(*object, "byteOffset"), "accessor.byteOffset", false, 0);
172 if (!viewIndex || !count || !component || !accessorOffset)
173 return detail::failure<Accessor>(DiagnosticCode::ParseError, "glTF accessor fields are invalid");
174 if (viewIndex.value() >= views->size() || count.value() == 0 ||
175 count.value() > std::max(limits.maximumVerticesPerPrimitive, limits.maximumIndicesPerPrimitive))
176 return detail::failure<Accessor>(DiagnosticCode::InvalidArgument, "glTF accessor exceeds limits");
177 const Value* typeValue = member(*object, "type");
178 if (!typeValue || !typeValue->isString())
179 return detail::failure<Accessor>(DiagnosticCode::ParseError, "glTF accessor type is missing");
180 std::uint32_t components = 0;
181 if (typeValue->asString() == "SCALAR") components = 1;
182 else if (typeValue->asString() == "VEC2") components = 2;
183 else if (typeValue->asString() == "VEC3") components = 3;
184 else if (typeValue->asString() == "VEC4") components = 4;
185 else return detail::failure<Accessor>(DiagnosticCode::Unsupported, "glTF accessor shape is unsupported");
186 const auto* view = (*views)[viewIndex.value()].getIf<Value::Object>();
187 if (!view) return detail::failure<Accessor>(DiagnosticCode::ParseError, "glTF bufferView is malformed");
188 auto bufferIndex = unsignedValue(member(*view, "buffer"), "bufferView.buffer");
189 auto viewOffset = unsignedValue(member(*view, "byteOffset"), "bufferView.byteOffset", false, 0);
190 auto viewLength = unsignedValue(member(*view, "byteLength"), "bufferView.byteLength");
191 auto byteStride = unsignedValue(member(*view, "byteStride"), "bufferView.byteStride", false, 0);
192 if (!bufferIndex || !viewOffset || !viewLength || !byteStride || bufferIndex.value() >= buffers.size())
193 return detail::failure<Accessor>(DiagnosticCode::ParseError, "glTF bufferView fields are invalid");
194 const std::uint32_t elementSize = componentSize(static_cast<std::uint32_t>(component.value())) * components;
195 const std::uint64_t stride = byteStride.value() == 0 ? elementSize : byteStride.value();
196 if (elementSize == 0 || stride < elementSize || stride > 252 ||
197 viewOffset.value() > buffers[bufferIndex.value()].size() ||
198 viewLength.value() > buffers[bufferIndex.value()].size() - viewOffset.value() ||
199 accessorOffset.value() > viewLength.value())
200 return detail::failure<Accessor>(DiagnosticCode::ParseError, "glTF accessor layout is invalid");
201 const std::uint64_t required = (count.value() - 1) * stride + elementSize;
202 if (required > viewLength.value() - accessorOffset.value())
203 return detail::failure<Accessor>(DiagnosticCode::ParseError, "glTF accessor range is truncated");
204 return Result<Accessor>::success({buffers[bufferIndex.value()],
205 static_cast<std::size_t>(viewOffset.value() + accessorOffset.value()),
206 static_cast<std::size_t>(stride), static_cast<std::uint32_t>(count.value()),
207 static_cast<std::uint32_t>(component.value()), components});
208}

210Result<float> readFloat(const Accessor& accessor, std::uint32_t element, std::uint32_t component) {
211 if (accessor.componentType != 5126 || component >= accessor.components)
212 return detail::failure<float>(DiagnosticCode::Unsupported, "mesh attribute must use FLOAT components");
213 const std::size_t offset = accessor.offset + std::size_t(element) * accessor.stride + component * 4;
214 const float value = std::bit_cast<float>(little32(accessor.buffer, offset));
215 if (!std::isfinite(value))
216 return detail::failure<float>(DiagnosticCode::ParseError, "mesh attribute contains non-finite values");
217 return Result<float>::success(value);
218}

220Result<std::uint32_t> readIndex(const Accessor& accessor, std::uint32_t element) {
221 if (accessor.components != 1)
222 return detail::failure<std::uint32_t>(DiagnosticCode::ParseError, "index accessor must be SCALAR");
223 const std::size_t offset = accessor.offset + std::size_t(element) * accessor.stride;
224 if (accessor.componentType == 5121) return Result<std::uint32_t>::success(accessor.buffer[offset]);
225 if (accessor.componentType == 5123)
226 return Result<std::uint32_t>::success(std::uint32_t(accessor.buffer[offset]) |
227 (std::uint32_t(accessor.buffer[offset + 1]) << 8));
228 if (accessor.componentType == 5125) return Result<std::uint32_t>::success(little32(accessor.buffer, offset));
229 return detail::failure<std::uint32_t>(DiagnosticCode::Unsupported,
230 "indices must use UNSIGNED_BYTE, UNSIGNED_SHORT or UNSIGNED_INT");
231}

243Result<PrimitiveOutput> decodePrimitive(const Value::Object& root, const Value::Object& primitive,
244 const std::vector<std::span<const std::uint8_t>>& buffers,
245 const AssetImportLimits& limits) {
246 auto mode = unsignedValue(member(primitive, "mode"), "primitive.mode", false, 4);
247 if (!mode) return Result<PrimitiveOutput>::failure(mode.status());
248 if (mode.value() != 4)
249 return detail::failure<PrimitiveOutput>(DiagnosticCode::Unsupported,
250 "only glTF TRIANGLES primitives are supported");
251 const auto* attributesValue = member(primitive, "attributes");
252 const auto* attributes = attributesValue ? attributesValue->getIf<Value::Object>() : nullptr;
253 if (!attributes)
254 return detail::failure<PrimitiveOutput>(DiagnosticCode::ParseError, "primitive attributes are required");
255 auto positionIndex = unsignedValue(member(*attributes, "POSITION"), "primitive.attributes.POSITION");
256 if (!positionIndex) return Result<PrimitiveOutput>::failure(positionIndex.status());
257 auto positions = accessorAt(root, buffers, positionIndex.value(), limits);
258 if (!positions) return Result<PrimitiveOutput>::failure(positions.status());
259 if (positions.value().components != 3 || positions.value().componentType != 5126 ||
260 positions.value().count > limits.maximumVerticesPerPrimitive)
261 return detail::failure<PrimitiveOutput>(DiagnosticCode::Unsupported,
262 "POSITION must be a bounded FLOAT VEC3 accessor");
263 std::optional<Accessor> normals;
264 if (const Value* normal = member(*attributes, "NORMAL")) {
265 auto normalIndex = unsignedValue(normal, "primitive.attributes.NORMAL");
266 if (!normalIndex) return Result<PrimitiveOutput>::failure(normalIndex.status());
267 auto decoded = accessorAt(root, buffers, normalIndex.value(), limits);
268 if (!decoded) return Result<PrimitiveOutput>::failure(decoded.status());
269 if (decoded.value().components != 3 || decoded.value().componentType != 5126 ||
270 decoded.value().count != positions.value().count)
271 return detail::failure<PrimitiveOutput>(DiagnosticCode::Unsupported, "NORMAL must match POSITION");
272 normals = std::move(decoded).takeValue();
273 }
274 std::optional<Accessor> texcoords;
275 if (const Value* texcoord = member(*attributes, "TEXCOORD_0")) {
276 auto texcoordIndex = unsignedValue(texcoord, "primitive.attributes.TEXCOORD_0");
277 if (!texcoordIndex) return Result<PrimitiveOutput>::failure(texcoordIndex.status());
278 auto decoded = accessorAt(root, buffers, texcoordIndex.value(), limits);
279 if (!decoded) return Result<PrimitiveOutput>::failure(decoded.status());
280 if (decoded.value().components != 2 || decoded.value().componentType != 5126 ||
281 decoded.value().count != positions.value().count)
282 return detail::failure<PrimitiveOutput>(DiagnosticCode::Unsupported, "TEXCOORD_0 must match POSITION");
283 texcoords = std::move(decoded).takeValue();
284 }
285 std::vector<std::uint32_t> indices;
286 if (const Value* indicesValue = member(primitive, "indices")) {
287 auto index = unsignedValue(indicesValue, "primitive.indices");
288 if (!index) return Result<PrimitiveOutput>::failure(index.status());
289 auto accessor = accessorAt(root, buffers, index.value(), limits);
290 if (!accessor) return Result<PrimitiveOutput>::failure(accessor.status());
291 if (accessor.value().count > limits.maximumIndicesPerPrimitive || accessor.value().count % 3 != 0)
292 return detail::failure<PrimitiveOutput>(DiagnosticCode::InvalidArgument, "triangle index count is invalid");
293 indices.reserve(accessor.value().count);
294 for (std::uint32_t item = 0; item < accessor.value().count; ++item) {
295 auto value = readIndex(accessor.value(), item);
296 if (!value) return Result<PrimitiveOutput>::failure(value.status());
297 if (value.value() >= positions.value().count)
298 return detail::failure<PrimitiveOutput>(DiagnosticCode::ParseError, "mesh index exceeds vertex count");
299 indices.push_back(value.value());
300 }
301 } else {
302 if (positions.value().count % 3 != 0)
303 return detail::failure<PrimitiveOutput>(DiagnosticCode::ParseError,
304 "unindexed triangle vertex count must be divisible by three");
305 indices.resize(positions.value().count);
306 for (std::uint32_t index = 0; index < indices.size(); ++index) indices[index] = index;
307 }
308 const std::uint64_t decodedSize = 24 + std::uint64_t(positions.value().count) *
309 (12 + (normals ? 12 : 0) + (texcoords ? 8 : 0)) + std::uint64_t(indices.size()) * 4;
310 if (decodedSize > limits.maximumDecodedBytes || decodedSize > std::numeric_limits<std::size_t>::max())
311 return detail::failure<PrimitiveOutput>(DiagnosticCode::InvalidArgument, "canonical mesh exceeds decoded budget");
312 PrimitiveOutput output;
313 output.vertexCount = positions.value().count;
314 output.indexCount = static_cast<std::uint32_t>(indices.size());
315 output.hasNormals = normals.has_value();
316 output.hasTexcoords = texcoords.has_value();
317 output.blob.insert(output.blob.end(), {'E', 'V', 'M', 'E', 'S', 'H', 0, 1});
318 put32(output.blob, output.vertexCount); put32(output.blob, output.indexCount);
319 put32(output.blob, (output.hasNormals ? 1u : 0u) | (output.hasTexcoords ? 2u : 0u)); put32(output.blob, 0);
320 for (std::uint32_t vertex = 0; vertex < output.vertexCount; ++vertex) {
321 for (std::uint32_t axis = 0; axis < 3; ++axis) {
322 auto value = readFloat(positions.value(), vertex, axis);
323 if (!value) return Result<PrimitiveOutput>::failure(value.status());
324 if (vertex == 0) output.minimum[axis] = output.maximum[axis] = value.value();
325 else { output.minimum[axis] = std::min(output.minimum[axis], value.value());
326 output.maximum[axis] = std::max(output.maximum[axis], value.value()); }
327 putFloat(output.blob, value.value());
328 }
329 if (normals) for (std::uint32_t axis = 0; axis < 3; ++axis) {
330 auto value = readFloat(*normals, vertex, axis);
331 if (!value) return Result<PrimitiveOutput>::failure(value.status());
332 putFloat(output.blob, value.value());
333 }
334 if (texcoords) for (std::uint32_t axis = 0; axis < 2; ++axis) {
335 auto value = readFloat(*texcoords, vertex, axis);
336 if (!value) return Result<PrimitiveOutput>::failure(value.status());
337 putFloat(output.blob, value.value());
338 }
339 }
340 for (const auto index : indices) put32(output.blob, index);
341 return Result<PrimitiveOutput>::success(std::move(output));
342}

350Result<PreparedAssetImport> prepareGltfImport(const GltfImportRequest& request) {
351 if (request.sourceName.empty() || request.sourceName.size() > request.limits.maximumStringBytes)
352 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument, "glTF source name is invalid");
353 auto document = parseDocument(request);
354 if (!document) return Result<PreparedAssetImport>::failure(document.status());
355 const auto* root = document.value().root.getIf<Value::Object>();
356 if (!root) return detail::failure<PreparedAssetImport>(DiagnosticCode::ParseError, "glTF root must be an object");
357 const auto* assetValue = member(*root, "asset");
358 const auto* asset = assetValue ? assetValue->getIf<Value::Object>() : nullptr;
359 const Value* version = asset ? member(*asset, "version") : nullptr;
360 if (!version || !version->isString() || !version->asString().starts_with("2."))
361 return detail::failure<PreparedAssetImport>(DiagnosticCode::UnknownVersion, "only glTF 2.x is supported",
362 "$.asset.version");
363 auto buffers = resolveBuffers(*root, document.value(), request);
364 if (!buffers) return Result<PreparedAssetImport>::failure(buffers.status());
365 const Value* meshesValue = member(*root, "meshes");
366 const auto* meshes = meshesValue ? meshesValue->getIf<Value::Array>() : nullptr;
367 if (!meshes || meshes->empty())
368 return detail::failure<PreparedAssetImport>(DiagnosticCode::ParseError, "glTF contains no meshes", "$.meshes");
369 auto manifestResult = detail::baseManifest(request.package, "eve.gltf2");
370 if (!manifestResult) return Result<PreparedAssetImport>::failure(manifestResult.status());
371 PreparedAssetImport result;
372 result.manifest = std::move(manifestResult).takeValue();
373 std::uint32_t assetCount = 0;
374 std::uint64_t decodedTotal = 0;
375 for (std::size_t meshIndex = 0; meshIndex < meshes->size(); ++meshIndex) {
376 const auto* mesh = (*meshes)[meshIndex].getIf<Value::Object>();
377 const Value* primitivesValue = mesh ? member(*mesh, "primitives") : nullptr;
378 const auto* primitives = primitivesValue ? primitivesValue->getIf<Value::Array>() : nullptr;
379 if (!primitives || primitives->empty())
380 return detail::failure<PreparedAssetImport>(DiagnosticCode::ParseError, "glTF mesh has no primitives");
381 for (std::size_t primitiveIndex = 0; primitiveIndex < primitives->size(); ++primitiveIndex) {
382 if (++assetCount > request.limits.maximumAssets)
383 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
384 "glTF primitive asset count exceeds limits");
385 const auto* primitive = (*primitives)[primitiveIndex].getIf<Value::Object>();
386 if (!primitive)
387 return detail::failure<PreparedAssetImport>(DiagnosticCode::ParseError, "glTF primitive is malformed");
388 auto decoded = decodePrimitive(*root, *primitive, buffers.value(), request.limits);
389 if (!decoded) return Result<PreparedAssetImport>::failure(decoded.status());
390 if (decoded.value().blob.size() > request.limits.maximumDecodedBytes ||
391 decodedTotal > request.limits.maximumDecodedBytes - decoded.value().blob.size())
392 return detail::failure<PreparedAssetImport>(DiagnosticCode::InvalidArgument,
393 "total canonical mesh budget is exceeded");
394 decodedTotal += decoded.value().blob.size();
395 const PersistentId id = request.package.packageId.child(
396 "gltf:mesh:" + std::to_string(meshIndex) + ":primitive:" + std::to_string(primitiveIndex));
397 auto reference = detail::assetRef(id);
398 if (!reference) return Result<PreparedAssetImport>::failure(reference.status());
399 const std::string base = "assets/" + id.format() + "/";
400 const std::string definitionPath = base + "asset.json";
401 const std::string blobPath = base + "mesh.bin";
402 Value::Object definition;
