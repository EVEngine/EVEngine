95Result<Value> encodeControlRig(const RigDefinition& d) {
96 auto valid = validateRigDefinition(d);
97 if (!valid.ok()) return Result<Value>::failure(valid.status());
98 Value controls = Value::array({}), nodes = Value::array({});
99 for (const auto& c : d.controls) {
100 Value keys = Value::array({});
101 for (const auto& k : c.keys) {
102 auto projected = constrainRigControl(c.limits, k.transform);
103 if (!projected.ok()) return Result<Value>::failure(projected.status());
104 keys.pushBack(Value::object({{"time", k.time}, {"transform", transformValue(projected.value())}}));
105 }
106 auto projected = constrainRigControl(c.limits, c.initial);
107 if (!projected.ok()) return Result<Value>::failure(projected.status());
108 controls.pushBack(Value::object({{"id", c.id},
109 {"parent", c.parent},
110 {"initial", transformValue(projected.value())},
111 {"limits", limitsValue(c.limits)},
112 {"shapeSize", c.shapeSize},
113 {"keys", std::move(keys)}}));
114 }
115 for (const auto& n : d.nodes) {
116 Value after = Value::array({});
117 for (const auto& id : n.after) after.pushBack(id);
118 const std::string kind = n.operation == RigOperation::FK ? "fk"
119 : n.operation == RigOperation::Aim ? "aim"
120 : "twoBoneIK";
121 nodes.pushBack(Value::object({{"id", n.id},
122 {"control", n.control},
123 {"bone", n.bone},
124 {"middle", n.middle},
125 {"tip", n.tip},
126 {"operation", kind},
127 {"weight", n.weight},
128 {"after", std::move(after)}}));
129 }
130 return Result<Value>::success(Value::object({{"schema", "eve.control-rig"},
131 {"version", std::int64_t{2}},
132 {"duration", d.duration},
133 {"controls", std::move(controls)},
134 {"nodes", std::move(nodes)}}));

137Result<RigDefinition> decodeControlRig(const Value& value) {
138 auto text = value.toJson();
139 if (!text.ok()) return Result<RigDefinition>::failure(text.status());
140 auto parsed = json::Document::parse(text.value());
141 if (!parsed.valid()) return error<RigDefinition>("invalid rig JSON representation");
142 const auto v = parsed.root();
143 if (!fields(v, {"schema", "version", "duration", "controls", "nodes"}))
144 return error<RigDefinition>("unexpected or missing document fields");
145 if (!v.get("schema").isString() || v.get("schema").asString() != "eve.control-rig" || !v.get("version").isInt64() ||
146 (v.get("version").asInt() != 1 && v.get("version").asInt() != 2))
147 return error<RigDefinition>("unsupported schema/version");
148 RigDefinition d;
149 if (!number(v.get("duration"), d.duration) || !v.get("controls").isArray() || !v.get("nodes").isArray())
150 return error<RigDefinition>("invalid document data");
151 if (v.get("controls").size() > 4096 || v.get("nodes").size() > 4096)
152 return error<RigDefinition>("rig capacity exceeded");
153 for (size_t ci = 0; ci < v.get("controls").size(); ++ci) {
154 const auto item = v.get("controls").at(ci);
155 RigControl c;
156 double size = 0;
157 const bool legacy = v.get("version").asInt() == 1;
158 if (!(legacy ? fields(item, {"id", "parent", "initial", "shapeSize", "keys"})
159 : fields(item, {"id", "parent", "initial", "shapeSize", "keys", "limits"})) ||
160 !string(item, "id", c.id) || !string(item, "parent", c.parent) ||
161 !transform(item.get("initial"), c.initial) || !number(item.get("shapeSize"), size) ||
162 !item.get("keys").isArray())
163 return error<RigDefinition>("invalid control fields");
164 c.shapeSize = static_cast<float>(size);
165 c.limits.reference = c.initial;
166 if (!legacy && !limits(item.get("limits"), c.limits)) return error<RigDefinition>("invalid limits fields");
167 if (item.get("keys").size() > 100000) return error<RigDefinition>("key capacity exceeded");
168 for (size_t ki = 0; ki < item.get("keys").size(); ++ki) {
169 const auto key = item.get("keys").at(ki);
170 RigKey k;
171 if (!fields(key, {"time", "transform"}) || !number(key.get("time"), k.time) ||
172 !transform(key.get("transform"), k.transform))
173 return error<RigDefinition>("invalid key fields");
174 c.keys.push_back(k);
175 }
176 d.controls.push_back(std::move(c));
177 }
178 for (size_t ni = 0; ni < v.get("nodes").size(); ++ni) {
179 const auto item = v.get("nodes").at(ni);
180 RigNode n;
181 std::string kind;
182 double weight = 0;
183 if (!fields(item, {"id", "control", "bone", "middle", "tip", "operation", "weight", "after"}) ||
184 !string(item, "id", n.id) || !string(item, "control", n.control) || !string(item, "bone", n.bone) ||
185 !string(item, "middle", n.middle) || !string(item, "tip", n.tip) || !string(item, "operation", kind) ||
186 !number(item.get("weight"), weight) || !item.get("after").isArray())
187 return error<RigDefinition>("invalid node fields");
188 if (kind == "fk")
189 n.operation = RigOperation::FK;
190 else if (kind == "aim")
191 n.operation = RigOperation::Aim;
192 else if (kind == "twoBoneIK")
193 n.operation = RigOperation::TwoBoneIK;
194 else
195 return error<RigDefinition>("unknown operation");
196 n.weight = static_cast<float>(weight);
197 if (item.get("after").size() > 4096) return error<RigDefinition>("dependency capacity exceeded");
198 for (size_t ai = 0; ai < item.get("after").size(); ++ai) {
199 const auto dependency = item.get("after").at(ai);
200 if (!dependency.isString()) return error<RigDefinition>("dependency must be a node ID");
201 n.after.push_back(dependency.asString());
202 }
203 d.nodes.push_back(std::move(n));
204 }
205 auto valid = validateRigDefinition(d);
206 if (!valid.ok()) return Result<RigDefinition>::failure(valid.status());
207 for (auto& c : d.controls) {
208 auto initial = constrainRigControl(c.limits, c.initial);
209 if (!initial.ok()) return Result<RigDefinition>::failure(initial.status());
210 c.initial = initial.value();
211 for (auto& k : c.keys) {
212 auto projected = constrainRigControl(c.limits, k.transform);
213 if (!projected.ok()) return Result<RigDefinition>::failure(projected.status());
214 k.transform = projected.value();
215 }
216 }
217 return Result<RigDefinition>::success(std::move(d));
218}
