73 const ISplineBindingTransformSource& source) const {
74 if (document.targetId().value() != documentTarget_)
75 return eve::editing::failed<void>(EditorStatus::Rejected, RuleId("editor.spline.binding-target-mismatch"),
76 "Spline binder targets another document");

78 const auto points = document.points();
79 for (const auto& [id, binding] : bindings_) {

128Result<void> SplinePathBinder::loadSnapshot(const EditorValue& snapshot) {
129 const auto* schemaValue = field(snapshot, "schema");
130 const auto* versionValue = field(snapshot, "version");
131 const auto* documentValue = field(snapshot, "document");
132 const auto* bindingsValue = field(snapshot, "bindings");
133 const auto* schema = schemaValue ? schemaValue->getIf<std::string>() : nullptr;
134 const auto* version = versionValue ? versionValue->getIf<std::int64_t>() : nullptr;
135 const auto* document = documentValue ? documentValue->getIf<std::string>() : nullptr;

145 const auto* pointValue = field(value, "point");
146 const auto* hostValue = field(value, "host");
147 const auto* objectValue = field(value, "object");
148 const auto* point = pointValue ? pointValue->getIf<std::string>() : nullptr;

177void addLine(editing::GizmoSnapshot& output, std::string id, const std::array<double, 3>& from,
178 const std::array<double, 3>& to, std::array<double, 4> color, bool dashed = false) {
179 std::array<double, 3> direction{to[0] - from[0], to[1] - from[1], to[2] - from[2]};
180 const double length =
181 std::sqrt(direction[0] * direction[0] + direction[1] * direction[1] + direction[2] * direction[2]);
182 if (length <= 1e-12) return;
183 for (double& value : direction) value /= length;
184 output.primitives.push_back({std::move(id), "line", from, {}, direction, color, 0.0, length, dashed});
185}

187bool normalizeRay(double x, double y, double z, std::array<double, 3>& output) {
188 const double length = std::sqrt(x * x + y * y + z * z);
189 if (!std::isfinite(length) || length < 1e-9) return false;
190 output = {x / length, y / length, z / length};
191 return true;
192}

196editing::GizmoSnapshot SplinePathGizmoBuilder::build(const SplinePathDocument& document, int sampleCount,
197 int maximumSamples, SplineGizmoStyle style) const {
198 if (sampleCount < 2 || maximumSamples < 2 || sampleCount > maximumSamples)
199 return failedSnapshot(document, "editor.spline.gizmo-budget",
200 "Spline gizmo sample count exceeds its bounded budget");
201 const auto validColor = [](const auto& color) {
202 return std::all_of(color.begin(), color.end(),
203 [](double value) { return std::isfinite(value) && value >= 0.0 && value <= 1.0; });
204 };
205 if (!std::isfinite(style.nodeSize) || !std::isfinite(style.handleSize) || style.nodeSize <= 0.0 ||
206 style.handleSize <= 0.0 || !validColor(style.lineColor) || !validColor(style.textColor))
207 return failedSnapshot(document, "editor.spline.gizmo-style-invalid",
208 "Spline gizmo sizes must be positive and colors must be finite normalized values");
209 editing::GizmoSnapshot result;
210 result.target = document.targetId().value();
211 result.targetRevision = document.revision();
212 for (const auto& point : document.points()) {
213 const auto anchor = position(point);
214 result.primitives.push_back(
215 {"spline.anchor." + point.id.value(), "sphere", anchor, {}, {}, {1.0, 0.72, 0.18, 1.0}, style.nodeSize});
216 if (style.showPointLabels) {
217 editing::GizmoPrimitive label;
218 label.id = "spline.label." + point.id.value();
219 label.kind = "text";
220 label.position = anchor;
221 label.color = style.textColor;
222 label.text = point.id.value();
223 result.primitives.push_back(std::move(label));
224 }
225 if (document.settings().kind != "bezier") continue;
226 const std::array<double, 3> incoming{point.x + point.inX, point.y + point.inY, point.z + point.inZ};
227 const std::array<double, 3> outgoing{point.x + point.outX, point.y + point.outY, point.z + point.outZ};
228 addLine(result, "spline.in-line." + point.id.value(), anchor, incoming, {0.95, 0.4, 0.3, 0.9}, true);
229 addLine(result, "spline.out-line." + point.id.value(), anchor, outgoing, {0.35, 0.9, 0.45, 0.9}, true);
230 result.primitives.push_back(
231 {"spline.in." + point.id.value(), "sphere", incoming, {}, {}, {0.95, 0.4, 0.3, 1.0}, style.handleSize});
232 result.primitives.push_back(
233 {"spline.out." + point.id.value(), "sphere", outgoing, {}, {}, {0.35, 0.9, 0.45, 1.0}, style.handleSize});
234 }
235 auto path = document.compilePath();
236 if (!path.ok()) {
237 if (document.points().size() >= 2)
238 return failedSnapshot(document, "editor.spline.gizmo-incomplete", path.status().describe());
239 result.diagnostics.push_back(editing::ruleDiagnostic(
240 DiagnosticCode::PreconditionViolation, RuleId("editor.spline.gizmo-incomplete"),
241 DiagnosticSeverity::Warning, "Add at least two points to preview the spline curve"));
242 result.status = EditorStatus::Applied;
243 return result;
244 }
245 auto polyline = path.value().polylineResult(sampleCount, false);
246 if (!polyline.ok()) return failedSnapshot(document, "editor.spline.gizmo-evaluate", polyline.status().describe());
247 for (int chunk = 0; chunk < polyline.value().chunkCount(); ++chunk) {
248 auto previous = polyline.value().chunkPointResult(chunk, 0);
249 if (!previous.ok())
250 return failedSnapshot(document, "editor.spline.gizmo-evaluate", previous.status().describe());
251 for (int sample = 1; sample < polyline.value().chunkPointCount(chunk); ++sample) {
252 auto current = polyline.value().chunkPointResult(chunk, sample);
253 if (!current.ok())
254 return failedSnapshot(document, "editor.spline.gizmo-evaluate", current.status().describe());
255 addLine(result, "spline.chunk." + std::to_string(chunk) + ".segment." + std::to_string(sample - 1),
256 {previous.value().x, previous.value().y, previous.value().z},
257 {current.value().x, current.value().y, current.value().z}, style.lineColor);
258 previous = std::move(current);
259 }
260 }
261 result.status = EditorStatus::Applied;
262 return result;
263}

288Result<SplinePathDragPreview> SplinePathDragSession::beginDrag(double rayOriginX, double rayOriginY, double rayOriginZ,
289 double rayDirectionX, double rayDirectionY,
290 double rayDirectionZ) {
291 cancelDrag();
292 if (!document_)
293 return eve::editing::failed<SplinePathDragPreview>(
294 EditorStatus::Rejected, RuleId("editor.spline.drag-target-required"), "Spline drag target is unavailable");
295 std::array<double, 3> direction;
296 if (!normalizeRay(rayDirectionX, rayDirectionY, rayDirectionZ, direction))
297 return eve::editing::failed<SplinePathDragPreview>(EditorStatus::Rejected,
298 RuleId("editor.spline.drag-ray-invalid"),
299 "Spline drag requires a finite non-zero pointer ray");
300 const std::array<double, 3> origin{rayOriginX, rayOriginY, rayOriginZ};
301 const auto gizmo = SplinePathGizmoBuilder().build(*document_, sampleCount_);
302 if (gizmo.status != EditorStatus::Applied)
303 return eve::editing::failed<SplinePathDragPreview>(EditorStatus::Rejected,
304 RuleId("editor.spline.drag-gizmo-invalid"),
305 "Spline gizmo is not available for picking");
306 double closest = std::numeric_limits<double>::infinity();
307 std::string pickedId;
308 for (const auto& primitive : gizmo.primitives) {
309 if (primitive.kind != "sphere" || primitive.id.rfind("spline.", 0) != 0) continue;
310 const std::array<double, 3> offset{origin[0] - primitive.position[0], origin[1] - primitive.position[1],
311 origin[2] - primitive.position[2]};
312 const double b = offset[0] * direction[0] + offset[1] * direction[1] + offset[2] * direction[2];
313 const double c =
314 offset[0] * offset[0] + offset[1] * offset[1] + offset[2] * offset[2] - primitive.radius * primitive.radius;
315 const double discriminant = b * b - c;
316 if (discriminant < 0.0) continue;
317 const double nearHit = -b - std::sqrt(discriminant);
318 const double farHit = -b + std::sqrt(discriminant);
319 const double hit = nearHit >= 0.0 ? nearHit : farHit;
320 if (hit < 0.0 || hit >= closest) continue;
321 closest = hit;
322 pickedId = primitive.id;
323 dragPlanePoint_ = primitive.position;
324 }
325 if (pickedId.empty())
326 return eve::editing::failed<SplinePathDragPreview>(
327 EditorStatus::NotFound, RuleId("editor.spline.handle-not-hit"), "Pointer ray did not hit a spline handle");
328 const auto roleEnd = pickedId.find('.', 7);
329 activeHandle_ = pickedId.substr(7, roleEnd - 7);
330 activePoint_ = StableId(pickedId.substr(roleEnd + 1));
331 const auto points = document_->points();
332 const auto found =
333 std::find_if(points.begin(), points.end(), [&](const auto& point) { return point.id == activePoint_; });
334 if (found == points.end())
335 return eve::editing::failed<SplinePathDragPreview>(
336 EditorStatus::NotFound, RuleId("editor.spline.point-not-found"), "Picked spline point no longer exists");
337 draft_ = *found;
338 dragPlaneNormal_ = direction;
339 baseRevision_ = document_->revision();
340 dragging_ = true;
341 return previewCurrent();
342}

344Result<SplinePathDragPreview> SplinePathDragSession::updateDrag(double rayOriginX, double rayOriginY, double rayOriginZ,
345 double rayDirectionX, double rayDirectionY,
346 double rayDirectionZ) {
347 if (!dragging_ || !document_)
348 return eve::editing::failed<SplinePathDragPreview>(
349 EditorStatus::Rejected, RuleId("editor.spline.drag-not-active"), "Spline drag has not begun");
350 if (document_->revision() != baseRevision_) {
351 cancelDrag();
352 return eve::editing::failed<SplinePathDragPreview>(EditorStatus::Conflict, RuleId("editor.spline.drag-stale"),
353 "Spline document changed while dragging");
354 }
355 std::array<double, 3> direction;
356 if (!normalizeRay(rayDirectionX, rayDirectionY, rayDirectionZ, direction))
357 return eve::editing::failed<SplinePathDragPreview>(EditorStatus::Rejected,
358 RuleId("editor.spline.drag-ray-invalid"),
359 "Spline drag requires a finite non-zero pointer ray");
360 const std::array<double, 3> origin{rayOriginX, rayOriginY, rayOriginZ};
361 const double denominator =
362 direction[0] * dragPlaneNormal_[0] + direction[1] * dragPlaneNormal_[1] + direction[2] * dragPlaneNormal_[2];
363 if (std::abs(denominator) < 1e-9)
364 return eve::editing::failed<SplinePathDragPreview>(EditorStatus::Rejected,
365 RuleId("editor.spline.drag-ray-parallel"),
366 "Pointer ray is parallel to the spline drag plane");
367 const double distance = ((dragPlanePoint_[0] - origin[0]) * dragPlaneNormal_[0] +
368 (dragPlanePoint_[1] - origin[1]) * dragPlaneNormal_[1] +
369 (dragPlanePoint_[2] - origin[2]) * dragPlaneNormal_[2]) /
370 denominator;
371 if (distance < 0.0)
372 return eve::editing::failed<SplinePathDragPreview>(EditorStatus::Rejected,
373 RuleId("editor.spline.drag-behind-camera"),
374 "Spline drag plane is behind the pointer ray origin");
375 const std::array<double, 3> world{origin[0] + direction[0] * distance, origin[1] + direction[1] * distance,
376 origin[2] + direction[2] * distance};
377 if (activeHandle_ == "anchor") {
378 draft_.x = world[0];
379 draft_.y = world[1];
380 draft_.z = world[2];
381 } else if (activeHandle_ == "in") {
382 draft_.inX = world[0] - draft_.x;
383 draft_.inY = world[1] - draft_.y;
384 draft_.inZ = world[2] - draft_.z;
385 } else {
386 draft_.outX = world[0] - draft_.x;
387 draft_.outY = world[1] - draft_.y;
388 draft_.outZ = world[2] - draft_.z;
389 }
390 return previewCurrent();
391}
