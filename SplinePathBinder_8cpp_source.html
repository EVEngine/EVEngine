45EditorResult<void> SplinePathBinder::bindResult(const SplinePathDocument& document, const StableId& point,
46 std::string host, std::string object,
47 const ISplineBindingTransformSource& source) {
48 if (document.targetId().value() != documentTarget_)
49 return binderError<void>(EditorStatus::Rejected, "editor.spline.binding-target-mismatch",
50 "Spline binder targets another document");
51 const auto points = document.points();
52 if (point.empty() ||
53 std::none_of(points.begin(), points.end(), [&](const auto& value) { return value.id == point; }))
54 return binderError<void>(EditorStatus::NotFound, "editor.spline.binding-point-missing",
55 "Bound spline point does not exist");
56 if (host.empty() || object.empty())
57 return binderError<void>(EditorStatus::Rejected, "editor.spline.binding-identity-empty",
58 "Spline binding requires non-empty host and object identities");
59 auto resolved = source.resolve(host, object);
60 if (!resolved.ok())
61 return binderError<void>(resolved.code(), "editor.spline.binding-source-stale", resolved.status().describe());
62 bindings_[point] = {point, std::move(host), std::move(object)};
63 return eve::editing::applied<void>();

73EditorResult<void> SplinePathBinder::refreshResult(SplinePathDocument& document,
74 const ISplineBindingTransformSource& source) const {
75 if (document.targetId().value() != documentTarget_)
76 return binderError<void>(EditorStatus::Rejected, "editor.spline.binding-target-mismatch",
77 "Spline binder targets another document");
78 auto candidate = document;
79 const auto points = document.points();
80 for (const auto& [id, binding] : bindings_) {
81 const auto found =
82 std::find_if(points.begin(), points.end(), [&](const auto& value) { return value.id == id; });
83 if (found == points.end())
84 return binderError<void>(EditorStatus::Conflict, "editor.spline.binding-point-stale",
85 "Bound spline point was removed");
86 auto pose = source.resolve(binding.host, binding.object);
87 if (!pose.ok())
88 return binderError<void>(pose.code(), "editor.spline.binding-source-stale", pose.status().describe());
89 if (!std::isfinite(pose.value().x) || !std::isfinite(pose.value().y) || !std::isfinite(pose.value().z))
90 return binderError<void>(EditorStatus::Rejected, "editor.spline.binding-pose-invalid",
91 "Bound transform returned non-finite coordinates");
92 auto updated = *found;
93 updated.x = pose.value().x;
94 updated.y = pose.value().y;
95 updated.z = pose.value().z;
96 auto operation = candidate.makeSetPoint(updated);
97 if (!operation.ok())
98 return binderError<void>(operation.code(), "editor.spline.binding-operation-failed",
99 operation.status().describe());
100 auto applied = candidate.applyDomainOperation(operation.value());
101 if (!applied.ok()) return applied;
102 }
103 return document.commitDomainState(std::make_unique<SplinePathDocument>(std::move(candidate)));
104}

128EditorResult<void> SplinePathBinder::loadSnapshot(const EditorValue& snapshot) {
129 const auto* schemaValue = field(snapshot, "schema");
130 const auto* versionValue = field(snapshot, "version");
131 const auto* documentValue = field(snapshot, "document");
132 const auto* bindingsValue = field(snapshot, "bindings");
133 const auto* schema = schemaValue ? schemaValue->getIf<std::string>() : nullptr;
134 const auto* version = versionValue ? versionValue->getIf<std::int64_t>() : nullptr;
135 const auto* document = documentValue ? documentValue->getIf<std::string>() : nullptr;
136 const auto* values = bindingsValue ? bindingsValue->getIf<EditorValue::Array>() : nullptr;
137 if (!schema || *schema != "eve.procgen.splineBinder" || !version || *version != 1)
138 return binderError<void>(EditorStatus::Unsupported, "editor.spline.binding-schema-unsupported",
139 "Unsupported spline binder snapshot schema");
140 if (!document || *document != documentTarget_ || !values)
141 return binderError<void>(EditorStatus::Rejected, "editor.spline.binding-snapshot-invalid",
142 "Spline binder snapshot targets another document or has invalid bindings");
143 std::map<StableId, SplinePointBinding> candidate;
144 for (const auto& value : *values) {
145 const auto* pointValue = field(value, "point");
146 const auto* hostValue = field(value, "host");
147 const auto* objectValue = field(value, "object");
148 const auto* point = pointValue ? pointValue->getIf<std::string>() : nullptr;
149 const auto* host = hostValue ? hostValue->getIf<std::string>() : nullptr;
150 const auto* object = objectValue ? objectValue->getIf<std::string>() : nullptr;
151 if (!point || point->empty() || !host || host->empty() || !object || object->empty() ||
152 !candidate.emplace(StableId(*point), SplinePointBinding{StableId(*point), *host, *object}).second)
153 return binderError<void>(EditorStatus::Rejected, "editor.spline.binding-snapshot-invalid",
154 "Spline binder snapshot contains invalid or duplicate links");
155 }
156 bindings_ = std::move(candidate);
157 return eve::editing::applied<void>();
158}
