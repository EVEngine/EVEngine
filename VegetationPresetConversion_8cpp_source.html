13 const auto found = result.candidate.outputDirectives.find(domain);
14 return found == result.candidate.outputDirectives.end() || found->second.size() != 1 ? nullptr

16 };
17 const auto invalid = [](std::string message) {
18 return Result<void>::failure(Diagnostic::error(DiagnosticCode::ParseError, message, {}, {},

70Result<void> transformToWorld(asset::CanonicalMeshData& mesh, const std::array<float, 16>& point,
71 const std::array<float, 9>& direction) {
72 for (const float value : point)
73 if (!std::isfinite(value))
74 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
75 "source-to-world point matrix contains nonfinite values", {},
76 {}, "asset.import.vegetation-preset.conversion"));
77 for (const float value : direction)
78 if (!std::isfinite(value))
79 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
80 "source-to-world direction matrix contains nonfinite values",
81 {}, {}, "asset.import.vegetation-preset.conversion"));
82 const float determinant = direction[0] * (direction[4] * direction[8] - direction[7] * direction[5]) -
83 direction[3] * (direction[1] * direction[8] - direction[7] * direction[2]) +
84 direction[6] * (direction[1] * direction[5] - direction[4] * direction[2]);
85 const auto dot = [&](unsigned a, unsigned b) {
86 return direction[a] * direction[b] + direction[a + 1] * direction[b + 1] + direction[a + 2] * direction[b + 2];
87 };
88 if (std::abs(determinant - 1.f) > 1e-4f || std::abs(dot(0, 0) - 1.f) > 1e-4f || std::abs(dot(3, 3) - 1.f) > 1e-4f ||
89 std::abs(dot(6, 6) - 1.f) > 1e-4f || std::abs(dot(0, 3)) > 1e-4f || std::abs(dot(0, 6)) > 1e-4f ||
90 std::abs(dot(3, 6)) > 1e-4f)
91 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
92 "source-to-world direction matrix is not a proper rotation", {},
93 {}, "asset.import.vegetation-preset.conversion"));
94 const auto transformDirection = [&](float& x, float& y, float& z) {
95 const float ox = direction[0] * x + direction[3] * y + direction[6] * z;
96 const float oy = direction[1] * x + direction[4] * y + direction[7] * z;
97 const float oz = direction[2] * x + direction[5] * y + direction[8] * z;
98 const float length = std::sqrt(ox * ox + oy * oy + oz * oz);
99 if (length < 1e-8f) return false;
100 x = ox / length;
101 y = oy / length;
102 z = oz / length;
103 return true;
104 };
105 for (std::size_t i = 0; i < mesh.positions.size(); i += 3) {
106 const float x = mesh.positions[i], y = mesh.positions[i + 1], z = mesh.positions[i + 2];
107 mesh.positions[i] = point[0] * x + point[4] * y + point[8] * z + point[12];
108 mesh.positions[i + 1] = point[1] * x + point[5] * y + point[9] * z + point[13];
109 mesh.positions[i + 2] = point[2] * x + point[6] * y + point[10] * z + point[14];
110 if (!transformDirection(mesh.normals[i], mesh.normals[i + 1], mesh.normals[i + 2]))
111 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
112 "world transform produced a zero mesh normal", {}, {},
113 "asset.import.vegetation-preset.conversion"));
114 }
115 if (auto tangent = mesh.attributes.find("TANGENT"); tangent != mesh.attributes.end()) {
116 if (tangent->second.components != 4 || tangent->second.values.size() / 4 != mesh.positions.size() / 3)
117 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
118 "world transform tangent stream is malformed", {}, {},
119 "asset.import.vegetation-preset.conversion"));
120 for (std::size_t i = 0; i < tangent->second.values.size(); i += 4)
121 if (!transformDirection(tangent->second.values[i], tangent->second.values[i + 1],
122 tangent->second.values[i + 2]))
123 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
124 "world transform produced a zero mesh tangent", {}, {},
125 "asset.import.vegetation-preset.conversion"));
126 }
127 return Result<void>::success();
128}

131Result<VegetationConversionResult> executeVegetationConversion(
132 VegetationConversionCandidate source, const std::vector<VegetationPresetCommand>& commands,
133 std::optional<asset::CanonicalMeshData> mesh, const std::map<std::string, VegetationPresetImage>& textures,
134 float variationSeed, const std::optional<std::array<float, 16>>& sourceToWorld,
135 const std::optional<std::array<float, 9>>& sourceDirectionToWorld) {
136 try {
137 auto applied = applyVegetationPreset(std::move(source), commands);
138 if (!applied) return Result<VegetationConversionResult>::failure(applied.status());
139 VegetationConversionResult result;
140 result.candidate = std::move(applied).takeValue();
141 auto outputPlan = applyOutputPlan(result);
142 if (!outputPlan) return Result<VegetationConversionResult>::failure(outputPlan.status());
143 if (mesh && result.meshOutput != VegetationMeshOutputMode::Off &&
144 result.transformOutput == VegetationTransformOutputMode::TransformToWorldSpace) {
145 if (sourceToWorld.has_value() != sourceDirectionToWorld.has_value())
146 return Result<VegetationConversionResult>::failure(Diagnostic::error(
147 DiagnosticCode::InvalidArgument, "world conversion requires both point and direction matrices", {},
148 {}, "asset.import.vegetation-preset.conversion"));
149 if (sourceToWorld) {
150 auto transformed = transformToWorld(*mesh, *sourceToWorld, *sourceDirectionToWorld);
151 if (!transformed) return Result<VegetationConversionResult>::failure(transformed.status());
152 }
153 }
154 if (mesh && result.meshOutput != VegetationMeshOutputMode::Off) {
155 auto converted = executeVegetationMeshRules(result.candidate, std::move(*mesh), textures, variationSeed);
156 if (!converted) return Result<VegetationConversionResult>::failure(converted.status());
157 result.mesh = std::move(converted).takeValue();
158 const auto count = result.mesh->positions.size() / 3;
159 float radius = 0, minimumY = result.mesh->positions[1], maximumY = minimumY;
160 for (std::size_t i = 0; i < count; ++i) {
161 radius = std::max(radius, std::max(std::abs(result.mesh->positions[i * 3]),
162 std::abs(result.mesh->positions[i * 3 + 2])));
163 minimumY = std::min(minimumY, result.mesh->positions[i * 3 + 1]);
164 maximumY = std::max(maximumY, result.mesh->positions[i * 3 + 1]);
165 }
166 if (auto bounds = result.candidate.meshRules.find("SetBounds");
167 bounds != result.candidate.meshRules.end()) {
168 if (bounds->second.size() != 2 || bounds->second[0] != "GET_BOUNDS_PROCEDURAL")
169 return Result<VegetationConversionResult>::failure(
170 Diagnostic::error(DiagnosticCode::Unsupported, "unsupported vegetation bounds rule", {}, {},
171 "asset.import.vegetation-preset.conversion"));
172 const auto& option = bounds->second[1];
173 const float expand = option == "0" ? 1.2f
174 : option == "1" ? 1.4f
175 : option == "2" ? 1.6f
176 : option == "3" ? 2.f
177 : 0.f;
178 if (expand == 0)
179 return Result<VegetationConversionResult>::failure(
180 Diagnostic::error(DiagnosticCode::ParseError, "invalid vegetation bounds option", {}, {},
181 "asset.import.vegetation-preset.conversion"));
182 result.meshBoundsMinimum = std::array{-radius * expand, minimumY, -radius * expand};
183 result.meshBoundsMaximum = std::array{radius * expand, maximumY * 1.25f, radius * expand};
184 }
185 if (auto readable = result.candidate.meshRules.find("SetReadWrite");
186 readable != result.candidate.meshRules.end()) {
187 if (readable->second.size() != 1 || (readable->second[0] != "MARK_MESHES_AS_READABLE" &&
188 readable->second[0] != "MARK_MESHES_AS_NON_READABLE"))
189 return Result<VegetationConversionResult>::failure(
190 Diagnostic::error(DiagnosticCode::Unsupported, "unsupported vegetation mesh readability rule",
191 {}, {}, "asset.import.vegetation-preset.conversion"));
192 result.meshCpuReadable = readable->second[0] == "MARK_MESHES_AS_READABLE";
193 }
194 } else if (!mesh && !result.candidate.meshRules.empty() && result.meshOutput != VegetationMeshOutputMode::Off) {
195 return Result<VegetationConversionResult>::failure(
196 Diagnostic::error(DiagnosticCode::NotFound, "vegetation preset requires a source mesh", {}, {},
197 "asset.import.vegetation-preset.conversion"));
198 }
199 if (result.materialOutput != VegetationMaterialOutputMode::Off) {
200 auto packed = result.mesh ? executeVegetationTexturePacks(result.candidate, textures, *result.mesh)
201 : executeVegetationTexturePacks(result.candidate, textures);
202 if (!packed) return Result<VegetationConversionResult>::failure(packed.status());
203 result.textures = std::move(packed).takeValue();
204 }
205 return Result<VegetationConversionResult>::success(std::move(result));
206 } catch (const std::bad_alloc&) {
207 return Result<VegetationConversionResult>::failure(
208 Diagnostic::error(DiagnosticCode::Failed, "vegetation conversion allocation failed", {}, {},
209 "asset.import.vegetation-preset.conversion"));
210 }
211}

278 const VegetationConversionCandidate& candidate, std::uint64_t maximumBytes) {

280 auto shader = shaderGuid(candidate.materialShader);
281 if (!shader) return Result<std::vector<std::uint8_t>>::failure(shader.status());
282 std::map<std::string, std::array<double, 4>> vectors = candidate.materialVectors;
283 for (const auto& [name, value] : candidate.materialColors) {
284 if (const auto found = vectors.find(name); found != vectors.end() && found->second != value)
285 return Result<std::vector<std::uint8_t>>::failure(
286 Diagnostic::error(DiagnosticCode::Conflict, "material vector and color values conflict", name, {},
287 "asset.import.vegetation-preset.material"));
288 vectors[name] = value;
289 }
290 for (const auto& [name, value] : candidate.materialFloats)
291 if (!propertyName(name) || !std::isfinite(value))
292 return Result<std::vector<std::uint8_t>>::failure(
293 Diagnostic::error(DiagnosticCode::InvalidArgument, "invalid vegetation material float", name, {},
294 "asset.import.vegetation-preset.material"));

297 return Result<std::vector<std::uint8_t>>::failure(
