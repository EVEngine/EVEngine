60Result<OrderRecord> projectOrder(const orders::Order& order, const CommandSpec* extended = nullptr) {
61 const auto kind = parseOrderKind(order.kind);
62 if (!kind) {
63 return failure<OrderRecord>(DiagnosticCode::InvariantViolation, "generic order contains an unknown RTS kind",
64 "order.kind");
65 }
66
67 const std::string json = order.payload.toJson();
68 std::string errorText;
69 auto document = json::Document::parse(json, &errorText);
70 if (!document.valid() || !document.root().isObject()) {
71 return failure<OrderRecord>(DiagnosticCode::ParseError, "generic order payload is not a JSON object",
72 "order.payload");
73 }
74
75 const auto root = document.root();
76 OrderRecord result;
77 result.id = order.id;
78 result.kind = *kind;
79 result.target.x = root.get("x").asFloat(0.0f);
80 result.target.y = root.get("y").asFloat(0.0f);
81 result.definitionId = root.get("definition").asString();
82 result.formationSlot = root.get("formationSlot").asInt(-1);
83 result.secondaryTarget.x = root.get("secondaryX").asFloat(0.0f);
84 result.secondaryTarget.y = root.get("secondaryY").asFloat(0.0f);
85 result.radius = root.get("radius").asFloat(0.0f);
86 result.append = root.get("append").asBool(false);
87 if (extended != nullptr) {
88 result.targetEntity = extended->targetEntity;
89 }
90 if (!finitePosition(result.target)) {
91 return failure<OrderRecord>(DiagnosticCode::InvariantViolation,
92 "generic order payload contains a non-finite target", "order.payload");
93 }
94 return Result<OrderRecord>::success(std::move(result));
95}

249Result<void> TagSet::add(std::string_view tag) {
250 if (tag.empty()) return failure(DiagnosticCode::InvalidArgument, "RTS tag must not be empty", "tag");
251 auto it = std::lower_bound(values_.begin(), values_.end(), tag,
252 [](const std::string& value, std::string_view key) { return value < key; });
253 if (it != values_.end() && *it == tag) return Result<void>::success(Status::success(StatusCode::NoOp));
254 values_.insert(it, std::string(tag));
255 return Result<void>::success(Status::success(StatusCode::Applied));
256}

258Result<void> TagSet::remove(std::string_view tag) {
259 if (tag.empty()) return failure(DiagnosticCode::InvalidArgument, "RTS tag must not be empty", "tag");
260 auto it = std::lower_bound(values_.begin(), values_.end(), tag,
261 [](const std::string& value, std::string_view key) { return value < key; });
262 if (it == values_.end() || *it != tag) return failure(DiagnosticCode::NotFound, "RTS tag is not present", "tag");
263 values_.erase(it);
264 return Result<void>::success(Status::success(StatusCode::Applied));
265}

378Result<std::string> OrderComponent::enqueue(const CommandSpec& command, int formationSlot) {
379 auto valid = command.validate();
380 if (!valid) return failureFrom<std::string>(valid.status());
381 if (!impl_) impl_ = std::make_unique<Impl>();
382 auto queued = impl_->queue.append(orderKindName(command.kind), command.priority, command.timeoutSeconds);
383 if (!queued) return failureFrom<std::string>(queued.status());
384 const std::string id = std::move(queued).takeValue();
385 auto order = impl_->queue.find(id);
386 if (!order)
387 return failure<std::string>(DiagnosticCode::InvariantViolation,
388 "generic order queue lost its newly appended command", "order");
389 order->get().payload.setNumber("x", command.target.x);
390 order->get().payload.setNumber("y", command.target.y);
391 order->get().payload.setString("definition", command.definitionId);
392 order->get().payload.setNumber("formationSlot", static_cast<double>(formationSlot));
393 order->get().payload.setNumber("secondaryX", command.secondaryTarget.x);
394 order->get().payload.setNumber("secondaryY", command.secondaryTarget.y);
395 order->get().payload.setNumber("radius", command.radius);
396 order->get().payload.setBool("append", command.append);
397 impl_->extended[id] = command;
398 return Result<std::string>::success(id, Status::success(StatusCode::Applied));
399}

425Result<OrderRecord> OrderComponent::current() const {
426 if (!impl_ || !impl_->queue.current())
427 return failure<OrderRecord>(DiagnosticCode::NotFound, "RTS order queue is idle", "order");
428 const auto& order = impl_->queue.current()->get();

501Result<void> OrderComponent::restoreState(const Snapshot& snapshotValue) {
502 orders::CommandQueue queue;
503 auto restored = queue.restore(snapshotValue.queueJson);
504 if (!restored) return restored;
505 for (const auto& [id, command] : snapshotValue.extended) {
506 auto valid = command.validate();
507 if (!valid) return valid;
508 if (!queue.find(id))
509 return failure(DiagnosticCode::Conflict,
510 "RTS order extension references an absent canonical queue record", "orders.extended");
511 }
512 auto candidate = std::make_unique<Impl>();
513 candidate->queue = std::move(queue);
514 candidate->extended = snapshotValue.extended;
515 impl_ = std::move(candidate);
516 return Result<void>::success(Status::success(StatusCode::Applied));
517}
