92bool readCurveArray(Poco::JSON::Array::Ptr arr, ParticleCurve &curve) {
93 if (!arr) return false;
94 bool any = false;
95 for (size_t i = 0; i < arr->size(); ++i) {
96 try {
97 if (arr->isObject(int(i))) {
98 auto obj = arr->getObject(int(i));
99 if (obj) {
100 curve.add(asFloat(obj->get("t"), 0.f), asFloat(obj->get("v"), 0.f));
101 any = true;
102 }
103 } else if (arr->isArray(int(i))) {
104 auto sub = arr->getArray(int(i));
105 if (sub && sub->size() >= 2) {
106 curve.add(asFloat(sub->get(0), 0.f), asFloat(sub->get(1), 0.f));
107 any = true;
108 }
109 }
110 } catch (...) {
111 }
112 }
113 return any;
114}

116bool readGradientArray(Poco::JSON::Array::Ptr arr, ParticleGradient &gradient) {
117 if (!arr) return false;
118 bool any = false;
119 for (size_t i = 0; i < arr->size(); ++i) {
120 float t = 0.f, r = 1.f, g = 1.f, b = 1.f, a = 1.f;
121 try {
122 if (arr->isObject(int(i))) {
123 auto obj = arr->getObject(int(i));
124 if (obj) {
125 t = asFloat(obj->get("t"), t);
126 r = asFloat(obj->get("r"), r);
127 g = asFloat(obj->get("g"), g);
128 b = asFloat(obj->get("b"), b);
129 a = obj->has("a") ? asFloat(obj->get("a"), a) : a;
130 gradient.add(t, r, g, b, a);
131 any = true;
132 }
133 } else if (arr->isArray(int(i))) {
134 auto sub = arr->getArray(int(i));
135 if (sub && sub->size() >= 5) {
136 t = asFloat(sub->get(0), t);
137 r = asFloat(sub->get(1), r);
138 g = asFloat(sub->get(2), g);
139 b = asFloat(sub->get(3), b);
140 a = asFloat(sub->get(4), a);
141 gradient.add(t, r, g, b, a);
142 any = true;
143 }
144 }
145 } catch (...) {
146 }
147 }
148 return any;
149}

152 if (path.empty()) return;

157 if (normalMap)
158 emitter->setNormalTexture(tex);
159 else
160 emitter->setTexture(tex);
161 if (auto *hot = eve::ModuleManager::getInstance<eve::filesystem::HotReload>("HotReload"))
162 hot->bind(path, "texture");
163 } catch (...) {
164 // Leave existing / null texture; config still applied.
165 }

606 float rs = obj->has("frameRandomStart") ? asFloat(obj->get("frameRandomStart"), 0.f)

609 }
610

612 emitter->setLayer(static_cast<int>(asFloat(obj->get("layer"), float(emitter->getLayer()))));
613 if (obj->has("visible"))
614 emitter->setVisible(asBool(obj->get("visible"), emitter->isVisible()));
615

617 std::string texPath = asString(obj->get("texture"));

619 tryLoadTexture(emitter, texPath);
620 }
621
622 if (obj->has("autoReload"))
623 emitter->resource()->autoReload = asBool(obj->get("autoReload"), true);
624
625 if (obj->has("autoStart") && asBool(obj->get("autoStart"), false))
626 emitter->start();
627
628 return true;

646bool loadConfigFile(ParticleEmitter *emitter, const std::string &path, std::string *error) {
647 if (!emitter || path.empty()) {
648 if (error) *error = "empty path";
649 return false;
650 }
651 auto *fs = eve::ModuleManager::getInstance<eve::filesystem::Filesystem>("Filesystem");
652 if (!fs) fs = eve::filesystem::Filesystem::create();
653
654 std::unique_ptr<eve::filesystem::FileData> data;
655 try {
656 data.reset(fs->read(path));
657 } catch (...) {
658 if (error) *error = "read failed: " + path;
659 return false;
660 }
661 if (!data || data->getSize() == 0) {
662 if (error) *error = "empty file: " + path;
663 return false;
664 }
665
666 std::string text(static_cast<const char *>(data->getData()), data->getSize());
667 if (!applyConfigText(emitter, text, error)) return false;
668
669 auto res = emitter->resource();
670 res->path = path;
671 res->modtime = fileModtime(path);
672 // Prefer OS watch; mtime remains a fallback in ParticleConfigSystem.
673 fs->watch(path);
674 if (auto *hot = eve::ModuleManager::getInstance<eve::filesystem::HotReload>("HotReload"))
675 hot->bind(path, "particle");
676 return true;
677}

679bool reloadConfigFile(ParticleEmitter *emitter, std::string *error) {

681 const std::string &path = emitter->resource()->path;
682 if (path.empty()) {
683 if (error) *error = "no config path";
684 return false;
685 }
686 return loadConfigFile(emitter, path, error);
