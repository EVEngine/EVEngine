57EditorResult<std::vector<std::size_t>> MaterialBatchTarget::selectedIndices(const SelectionSnapshot& selection) const {
58 if (selection.items.empty())
59 return editing::failed<std::vector<std::size_t>>(EditorStatus::Rejected,
60 RuleId("editor.material-batch.empty-selection"),
61 "Material batch selection is empty");
62 std::vector<std::size_t> result;
63 std::set<std::size_t> unique;
64 for (const auto& item : selection.items) {
65 if (item.target != targetId())
66 return editing::failed<std::vector<std::size_t>>(EditorStatus::Rejected,
67 RuleId("editor.material-batch.target-mismatch"),
68 "Material batch selection targets another document");
69 const auto found = std::find_if(materials_.begin(), materials_.end(), [&](const auto& material) {
70 return material.targetId().value() == item.item.value();
71 });
72 if (found == materials_.end())
73 return editing::failed<std::vector<std::size_t>>(EditorStatus::NotFound,
74 RuleId("editor.material-batch.material-not-found"),
75 "Selected material is not in this batch");
76 const auto index = static_cast<std::size_t>(found - materials_.begin());
77 if (!unique.insert(index).second)
78 return editing::failed<std::vector<std::size_t>>(EditorStatus::Rejected,
79 RuleId("editor.material-batch.duplicate-selection"),
80 "Material batch selection contains a duplicate");
81 result.push_back(index);
82 }
83 return editing::applied<std::vector<std::size_t>>(std::move(result));
84}

96PropertyReadResult MaterialBatchTarget::read(const SelectionSnapshot& selection, const PropertyPath& path) const {
97 const auto indices = selectedIndices(selection);
98 if (!indices.ok()) return {PropertyReadState::Error, {}, indices.status().diagnostics()};
99 std::optional<EditorValue> common;
100 for (const auto index : indices.value()) {
101 const auto value = materials_[index].read(one(materials_[index]), path);
102 if (value.state != PropertyReadState::Value) return value;
103 if (!common)
104 common = value.value;
105 else if (*common != value.value)
106 return {PropertyReadState::Mixed, {}, {}};
107 }
108 return {PropertyReadState::Value, *common, {}};
109}

125EditorResult<DomainOperation> MaterialBatchTarget::makeSet(const SelectionSnapshot& selection, const PropertyPath& path,
126 const EditorValue& value, PropertySetMode mode) const {
127 if (mode == PropertySetMode::Reset) return makeReset(selection, path);
128 const auto indices = selectedIndices(selection);
129 if (!indices.ok()) return EditorResult<DomainOperation>::failure(indices.status());
130 auto candidates = materials_;
131 for (const auto index : indices.value()) {
132 auto operation = candidates[index].makeSet(one(candidates[index]), path, value, mode);
133 if (!operation.ok()) return EditorResult<DomainOperation>::failure(operation.status());
134 auto applied = candidates[index].applyDomainOperation(operation.value());
135 if (!applied.ok()) return EditorResult<DomainOperation>::failure(applied.status());
136 }
137 return replacement(std::move(candidates), path);
138}

162EditorResult<void> MaterialBatchTarget::applyDomainOperation(const DomainOperation& operation) {
163 if (operation.target != targetId() || operation.type != "material.batch.replace.v1")
164 return editing::failed<void>(EditorStatus::Rejected, RuleId("editor.material-batch.operation"),
165 "Material batch operation mismatch");
166 const auto* values = field(operation.payload, "materials");
167 const auto* array = values ? values->getIf<EditorValue::Array>() : nullptr;
168 if (!array || array->size() != materials_.size() || !operation.payload.isWithinLimits(16, 200000, 16 * 1024 * 1024))
169 return editing::failed<void>(EditorStatus::Rejected, RuleId("editor.material-batch.payload"),
170 "Material batch payload is incomplete or exceeds limits");
171 std::vector<MaterialDocumentTarget> candidates;
172 candidates.reserve(materials_.size());
173 std::set<std::string> ids;
174 for (const auto& item : *array) {
175 const auto* idValue = field(item, "id");
176 const auto* snapshot = field(item, "snapshot");
177 const auto* id = idValue ? idValue->getIf<std::string>() : nullptr;
178 if (!id || !snapshot || !ids.insert(*id).second)
179 return editing::failed<void>(EditorStatus::Rejected, RuleId("editor.material-batch.entry"),
180 "Material batch entry is invalid or duplicated");
181 const auto original = std::find_if(materials_.begin(), materials_.end(),
182 [&](const auto& material) { return material.targetId().value() == *id; });
183 if (original == materials_.end())
184 return editing::failed<void>(EditorStatus::Conflict, RuleId("editor.material-batch.identity"),
185 "Material batch identities changed");
186 MaterialDocumentTarget candidate(*id);
187 auto loaded = candidate.loadSnapshot(*snapshot);
188 if (!loaded.ok()) return EditorResult<void>::failure(loaded.status());
189 candidates.push_back(std::move(candidate));
190 }
191 EditRegion dirty;
192 dirty.include(0, 0);
193 return publishAndAdopt(std::move(candidates), revisionValue() + 1, dirty);
194}
