59Result<std::shared_ptr<const FileData>> readPreparedFile(const std::string& path, size_t limit) {
60 using R = Result<std::shared_ptr<const FileData>>;
61 try {
62 if (!limit || limit > maximumBytes)
63 return R::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "Invalid prepared file size limit"));
64 auto result = ResourceManager::getInstance().waitFor(keyFor(path));
65 if (!result) return R::failure(result.status());
66 auto bytes = static_cast<FileSnapshot&>(result.value().get()).bytes;
67 if (bytes->getSize() > limit)
68 return R::failure(
69 Diagnostic::error(DiagnosticCode::InvalidArgument, "Prepared file exceeds caller size limit", path));
70 return R::success(std::move(bytes));
71 } catch (const std::exception& e) {
72 return R::failure(Diagnostic::error(DiagnosticCode::Failed, e.what(), path));
73 }
