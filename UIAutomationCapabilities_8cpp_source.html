132Poco::JSON::Object::Ptr nodeJson(const UIHost::Tree& tree, int index) {
133 Poco::JSON::Object::Ptr out(new Poco::JSON::Object());
134 if (index < 0 || index >= static_cast<int>(tree.nodes.size())) return out;
135
136 const UINode& node = tree.nodes[static_cast<size_t>(index)];
137 out->set("type", nodeTypeName(node.type));
138 out->set("id", node.id);
139 if (!node.key.empty()) out->set("key", node.key);
140 out->set("text", node.text);
141 out->set("visible", node.visible);
142 out->set("enabled", node.enabled);
143 out->set("clickable", isClickable(node));
144 out->set("focusMode", focusModeName(node.focusMode));
145 out->set("focused", node.focused);
146 out->set("tabIndex", node.tabIndex);
147 out->set("mouseFilter", mouseFilterName(node.mouseFilter));
148 out->set("theme", themePresetName(node.themePreset));
149 if (!node.focusPrevious.empty()) out->set("focusPrevious", node.focusPrevious);
150 if (!node.focusNext.empty()) out->set("focusNext", node.focusNext);
151 if (!node.focusLeft.empty()) out->set("focusLeft", node.focusLeft);
152 if (!node.focusRight.empty()) out->set("focusRight", node.focusRight);
153 if (!node.focusUp.empty()) out->set("focusUp", node.focusUp);
154 if (!node.focusDown.empty()) out->set("focusDown", node.focusDown);
155 Poco::JSON::Object::Ptr accessibility(new Poco::JSON::Object());
156 accessibility->set("role", accessibilityRoleName(node.accessibilityRole));
157 accessibility->set("name", node.accessibilityName.empty() ? node.text
158 : node.accessibilityName);
159 if (!node.accessibilityDescription.empty())
160 accessibility->set("description", node.accessibilityDescription);
161 out->set("accessibility", accessibility);
162 if (node.type == NodeType::Checkbox || node.type == NodeType::Switch ||
163 node.type == NodeType::MenuItem)
164 out->set("checked", node.checked);
165 if (node.type == NodeType::Slider || node.type == NodeType::Progress ||
166 node.type == NodeType::Combo || node.type == NodeType::SplitPane) {
167 out->set("value", node.value);
168 out->set("min", node.minValue);
169 out->set("max", node.maxValue);
170 }
171 if (node.type == NodeType::InputText || node.type == NodeType::SearchField ||
172 node.type == NodeType::Combo || node.type == NodeType::MenuItem)
173 out->set("valueText", node.valueText);
174 if (!node.tooltip.empty()) out->set("tooltip", node.tooltip);
175
176 Poco::JSON::Array::Ptr children(new Poco::JSON::Array());
177 for (int child = node.firstChild; child >= 0; child = tree.nodes[static_cast<size_t>(child)].nextSibling) {
178 if (child >= static_cast<int>(tree.nodes.size())) break;
179 children->add(nodeJson(tree, child));
180 }
181 if (children->size() != 0) out->set("children", children);
182 return out;
183}

216std::vector<WidgetMatch> findWidgets(const std::string& hostName, const std::string& widgetId) {
217 std::vector<WidgetMatch> result;
218 for (UIHostHandle handle : hosts()) {
219 auto host = UIHost::resolve(handle);
220 if (!host || (!hostName.empty() && host->get().getName() != hostName)) continue;
221 auto node = host->get().findById(widgetId);
222 if (!node) continue;
223 const auto tree = host->get().tree();
224 result.push_back({handle, static_cast<int>(&node->get() - tree->nodes.data())});
225 }
226 return result;
227}

238class UIAutomationImpl final : public eve::IUIAutomation {
239public:
240 std::string tree(const std::string& hostName) const override {
241 Poco::JSON::Array::Ptr items(new Poco::JSON::Array());
242 for (UIHostHandle handle : hosts()) {
243 auto host = UIHost::resolve(handle);
244 if (!host || (!hostName.empty() && host->get().getName() != hostName)) continue;
245 items->add(hostJson(host->get()));
246 }
247 if (!hostName.empty() && items->size() == 0) return "error: UI host not found: " + hostName;
248
249 Poco::JSON::Object::Ptr out(new Poco::JSON::Object());
250 out->set("hostCount", static_cast<int>(items->size()));
251 out->set("hosts", items);
252 return stringify(Poco::Dynamic::Var(out));
253 }
254
255 std::string get(const std::string& hostName, const std::string& widgetId) const override {
256 const std::vector<WidgetMatch> matches = findWidgets(hostName, widgetId);
257 if (matches.size() != 1) return matchError(hostName, widgetId, matches.size());
258 auto host = UIHost::resolve(matches[0].host);
259 if (!host) return "error: UI host became stale";
260 Poco::JSON::Object::Ptr out = nodeJson(*host->get().tree(), matches[0].nodeIndex);
261 out->set("host", host->get().getName());
262 out->set("hostVisible", host->get().meta()->visible);
263 return stringify(Poco::Dynamic::Var(out));
264 }
265
266 std::string click(const std::string& hostName, const std::string& widgetId) override {
267 const std::vector<WidgetMatch> matches = findWidgets(hostName, widgetId);
268 if (matches.size() != 1) return matchError(hostName, widgetId, matches.size());
269 auto host = UIHost::resolve(matches[0].host);
270 if (!host) return "error: UI host became stale";
271 UIHost& resolvedHost = host->get();
272 const auto tree = resolvedHost.tree();
273 if (matches[0].nodeIndex < 0 || matches[0].nodeIndex >= static_cast<int>(tree->nodes.size()))
274 return "error: UI widget became stale";
275 UINode& node = tree->nodes[static_cast<std::size_t>(matches[0].nodeIndex)];
276 if (!resolvedHost.meta()->visible || !node.visible)
277 return "error: widget is not visible: " + resolvedHost.getName() + "/" + widgetId;
278 if (!isClickable(node)) return "error: widget is not clickable: " + resolvedHost.getName() + "/" + widgetId;
279
280 UIEvent event;
281 event.host = matches[0].host;
282 event.hostName = resolvedHost.getName();
283 event.nodeId = node.id;
284 event.nodeIndex = matches[0].nodeIndex;
285 event.kind = "click";
286 event.handlerIndex = node.handlerClick;
287 UISystem::pendingEvents().push_back(std::move(event));
288
289 Poco::JSON::Object::Ptr out(new Poco::JSON::Object());
290 out->set("queued", true);
291 out->set("host", resolvedHost.getName());
292 out->set("widget", widgetId);
293 out->set("event", "click");
294 return stringify(Poco::Dynamic::Var(out));
295 }

297private:
298 static int nodeIndex(const UIHost::Tree& tree, const UINode* node) {
299 if (!node || tree.nodes.empty()) return -1;
300 return static_cast<int>(node - tree.nodes.data());
301 }
