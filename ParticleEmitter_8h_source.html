607 [[nodiscard]] eve::SimulationTick currentSimulationTick() const noexcept {
608 // ECS.hpp's generated accessor has no const overload and would create
609 // a missing component. Query the existing table buffer directly so a
610 // read-only tick query neither casts away constness nor mutates ECS.
611 auto *const buffer = storage ? storage->getComponentBuffer<Sim>() : nullptr;
612 if (!buffer || id >= buffer->size()) return eve::SimulationTick::zero();
613 // The size guard makes this existing-slot read non-mutating; ECS.hpp's
614 // const overload currently attempts to resize its deque.
615 return buffer->get(id).simulationTick;
616 }
