32inline std::vector<double> layer(const std::vector<double>& x, const Policy& p, std::size_t offset, std::size_t outputs,
33 bool activate) {
34 std::vector<double> y(outputs);
35 for (std::size_t j = 0; j < outputs; ++j) {
36 const auto start = offset + j * (x.size() + 1);
37 double sum = p.weights[start + x.size()];
38 for (std::size_t i = 0; i < x.size(); ++i) sum += x[i] * p.weights[start + i];
39 y[j] = activate ? std::tanh(sum) : sum;
40 }
41 return y;
42}

44inline std::vector<double> softmax(std::vector<double> logits, const Observation& o) {
45 double maximum = -std::numeric_limits<double>::infinity();
46 for (auto a : o.legalActions) maximum = std::max(maximum, logits[a]);
47 std::vector<double> probabilities(logits.size());
48 double total = 0;
49 for (auto a : o.legalActions) total += probabilities[a] = std::exp(logits[a] - maximum);
50 for (auto a : o.legalActions) probabilities[a] /= total;
51 return probabilities;
52}

63inline Policy makePolicy(const Config& c) {
64 Policy p;
65 p.featureCount = c.featureCount;
66 p.actionCount = c.actionCount;
67 p.hiddenWidth = c.hiddenWidth;
68 p.weights.resize(weightCount(c.featureCount, c.hiddenWidth, c.actionCount));
69 Random random(c.learningSeed);
70 for (auto& w : p.weights) w = (random.unit() * 2 - 1) / std::sqrt(double(c.hiddenWidth));
71 return p;
72}

74inline void train(Policy& p, const Observation& o, std::uint32_t action, double rate) {
75 std::vector<double> x(o.features.begin(), o.features.end());
76 auto h1 = layer(x, p, 0, p.hiddenWidth, true);
77 const auto second = p.hiddenWidth * (p.featureCount + 1);
78 auto h2 = layer(h1, p, second, p.hiddenWidth, true);
79 const auto third = second + p.hiddenWidth * (p.hiddenWidth + 1);
80 auto d3 = softmax(layer(h2, p, third, p.actionCount, false), o);
81 d3[action] -= 1;
82 std::vector<double> d2(p.hiddenWidth), d1(p.hiddenWidth);
83 for (std::size_t i = 0; i < p.hiddenWidth; ++i) {
84 for (std::size_t j = 0; j < p.actionCount; ++j) d2[i] += d3[j] * p.weights[third + j * (p.hiddenWidth + 1) + i];
85 d2[i] *= 1 - h2[i] * h2[i];
86 }
87 for (std::size_t i = 0; i < p.hiddenWidth; ++i) {
88 for (std::size_t j = 0; j < p.hiddenWidth; ++j)
89 d1[i] += d2[j] * p.weights[second + j * (p.hiddenWidth + 1) + i];
90 d1[i] *= 1 - h1[i] * h1[i];
91 }
92 auto update = [&](std::size_t offset, const auto& inputs, const auto& gradient) {
93 for (std::size_t j = 0; j < gradient.size(); ++j) {
94 auto start = offset + j * (inputs.size() + 1);
95 for (std::size_t i = 0; i < inputs.size(); ++i)
96 p.weights[start + i] -= rate * std::clamp(gradient[j] * inputs[i], -1.0, 1.0);
97 p.weights[start + inputs.size()] -= rate * std::clamp(gradient[j], -1.0, 1.0);
98 }
99 };
100 update(third, h2, d3);
101 update(second, h1, d2);
102 update(0, x, d1);
103}
