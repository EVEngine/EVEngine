98Result<void> Organism::refillNeed(const std::string& needId, double permille) {
99 const SpeciesDef* species = SpeciesRegistry::find(growth()->speciesId);
100 if (!species) {
101 return Result<void>::failure(
102 Diagnostic::error(DiagnosticCode::NotFound, "unknown species", "growth.organism.refillNeed"));
103 }
104 if (!species->hasNeed(needId)) {
105 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
106 "species does not declare need '" + needId + "'",
107 "growth.organism.refillNeed"));
108 }
109 if (growth()->dead) {
110 return Result<void>::failure(
111 Diagnostic::error(DiagnosticCode::PreconditionViolation, "organism is dead", "growth.organism.refillNeed"));
112 }
113 if (!std::isfinite(permille) || permille < 0.0) {
114 return Result<void>::failure(Diagnostic::error(
115 DiagnosticCode::InvalidArgument, "need amount must be finite and >= 0", "growth.organism.refillNeed"));
116 }
117 auto& meters = vitalNeeds()->meters;
118 meters[needId] = clampPermille(meterOrZero(*vitalNeeds(), needId) + permille);
119 if (needId == "water") {
120 GrowthWorld::instance().emit(GrowthEventKind::Watered, *this, std::to_string(permille));
121 } else if (needId == "food") {
122 GrowthWorld::instance().emit(GrowthEventKind::Fed, *this, std::to_string(permille));
123 } else if (needId == "fertility") {
124 GrowthWorld::instance().emit(GrowthEventKind::Fertilized, *this, std::to_string(permille));
125 } else {
126 GrowthWorld::instance().emit(GrowthEventKind::Watered, *this, needId + ":" + std::to_string(permille));
127 }
128 return Result<void>::success();
129}

137Result<std::vector<YieldReceipt>> Organism::harvest() {
138 if (growth()->dead) {
139 return Result<std::vector<YieldReceipt>>::failure(
140 Diagnostic::error(DiagnosticCode::PreconditionViolation, "organism is dead", "growth.organism.harvest"));
141 }
142 if (!growth()->mature) {
143 return Result<std::vector<YieldReceipt>>::failure(Diagnostic::error(
144 DiagnosticCode::PreconditionViolation, "organism is not mature", "growth.organism.harvest"));
145 }
146 if (growth()->harvested) {
147 return Result<std::vector<YieldReceipt>>::failure(Diagnostic::error(
148 DiagnosticCode::PreconditionViolation, "organism already harvested", "growth.organism.harvest"));
149 }
150 const SpeciesDef* species = SpeciesRegistry::find(growth()->speciesId);
151 if (!species) {
152 return Result<std::vector<YieldReceipt>>::failure(
153 Diagnostic::error(DiagnosticCode::NotFound, "unknown species", "growth.organism.harvest"));
154 }
155 if (species->harvestYield.empty()) {
156 return Result<std::vector<YieldReceipt>>::failure(Diagnostic::error(
157 DiagnosticCode::InvalidArgument, "species has no harvestYield", "growth.organism.harvest"));
158 }
159
160 std::vector<YieldReceipt> yields;
161 yields.reserve(species->harvestYield.size());
162 for (const YieldDef& def : species->harvestYield) {
163 const int count = def.minCount + (def.maxCount - def.minCount) / 2;
164 yields.push_back(YieldReceipt{def.itemId, count});
165 }
166 growth()->harvested = true;
167 GrowthWorld::instance().emit(GrowthEventKind::Harvested, *this, std::to_string(static_cast<int>(yields.size())));
168 return Result<std::vector<YieldReceipt>>::success(std::move(yields));
169}
