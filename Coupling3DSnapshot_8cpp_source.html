34Result<Value> WorldSnapshotAccess::captureCouplings3D(const World3D& world) {
35 Value::Array records;
36 for (auto [id, coupling] : world.couplings3D_) {
37 const auto& d = *coupling->data_;
38 auto row = d.row();
39 if (!row) return Result<Value>::failure(row.status());
40 const auto& limits = d.limits();
41 Value::Object object{{"id", Value(std::to_string(id))},
42 {"kind", Value(d.gear ? "gear" : "pulley")},
43 {"enabled", Value(d.enabled)},
44 {"impulse", Value(d.lastImpulse)},
45 {"saturated", Value(d.saturated)},
46 {"limits", Value(Value::Array{Value(limits.positionCorrection),
47 Value(limits.maxCorrectionSpeed), Value(limits.maxForce)})}};
48 if (d.gear) {
49 for (int i = 0; i < 2; ++i) {
50 auto* source = world.findJoint(d.sources[i]);
51 if (!source)
52 return Result<Value>::failure(
53 Diagnostic::error(DiagnosticCode::StaleHandle, "Missing gear source"));
54 object.emplace(i == 0 ? "sourceA" : "sourceB", Value(std::to_string(source->getId())));
55 }
56 object.emplace("ratio", Value(d.gearSettings.ratio));
57 object.emplace("reference", Value(d.reference));
58 object.emplace("coordinateA", Value(row.value().coordinate[0]));
59 object.emplace("coordinateB", Value(row.value().coordinate[1]));
60 } else {
61 for (int i = 0; i < 2; ++i) {
62 auto* body = world.findBody(d.bodies[i]);
63 if (!body)
64 return Result<Value>::failure(
65 Diagnostic::error(DiagnosticCode::StaleHandle, "Missing pulley body"));
66 object.emplace(i == 0 ? "sourceA" : "sourceB", Value(std::to_string(body->getId())));
67 }
68 const auto& p = d.pulleySettings;
69 object.emplace("ratio", Value(p.ratio));
70 object.emplace("length", Value(p.length));
71 object.emplace("tensionOnly", Value(p.tensionOnly));
72 object.emplace("localAnchorA", point(p.localAnchorA));
73 object.emplace("localAnchorB", point(p.localAnchorB));
74 object.emplace("groundA", point(p.groundA));
75 object.emplace("groundB", point(p.groundB));
76 }
77 records.emplace_back(std::move(object));
78 }
79 return Result<Value>::success(Value(std::move(records)));
80}
81Result<void> WorldSnapshotAccess::restoreCouplings3D(World3D& world, const Value::Array& values) {
82 using namespace snapshot_detail;
83 try {
84 std::map<int, const Value::Object*> ordered;
85 for (const auto& value : values) {
86 const auto* o = value.getIf<Value::Object>();
87 if (!o) return invalidCoupling("Transmission must be an object");
88 auto id = readInt64(*o, "id");
89 if (!id) return Result<void>::failure(id.status());
90 if (id.value() <= 0 || id.value() >= std::numeric_limits<int>::max() ||
91 !ordered.emplace(static_cast<int>(id.value()), o).second)
92 return invalidCoupling("Invalid or duplicate transmission id");
93 }
94 std::map<int, Joint3D*> joints;
95 for (auto* joint : world.joints_) joints.emplace(joint->getId(), joint);
96 for (auto [id, o] : ordered) {
97 auto kind = readString(*o, "kind");
98 if (!kind) return Result<void>::failure(kind.status());
99 const bool gear = kind.value() == "gear";
100 if (!gear && kind.value() != "pulley") return invalidCoupling("Unknown transmission kind");
101 if (!(gear ? hasExactFields(*o, {"id", "kind", "enabled", "impulse", "saturated", "limits", "sourceA",
102 "sourceB", "ratio", "reference", "coordinateA", "coordinateB"})
103 : hasExactFields(*o, {"id", "kind", "enabled", "impulse", "saturated", "limits", "sourceA",
104 "sourceB", "ratio", "length", "tensionOnly", "localAnchorA",
105 "localAnchorB", "groundA", "groundB"})))
106 return invalidCoupling("Unknown or missing transmission fields");
107 auto sourceA = readInt64(*o, "sourceA");
108 if (!sourceA) return Result<void>::failure(sourceA.status());
109 auto sourceB = readInt64(*o, "sourceB");
110 if (!sourceB) return Result<void>::failure(sourceB.status());
111 if (sourceA.value() <= 0 || sourceA.value() >= std::numeric_limits<int>::max() || sourceB.value() <= 0 ||
112 sourceB.value() >= std::numeric_limits<int>::max())
113 return invalidCoupling("Source id out of range");
114 auto enabled = readBool(*o, "enabled");
115 if (!enabled) return Result<void>::failure(enabled.status());
116 auto saturated = readBool(*o, "saturated");
117 if (!saturated) return Result<void>::failure(saturated.status());
118 auto impulse = number(*o, "impulse");
119 if (!impulse) return Result<void>::failure(impulse.status());
120 auto ratio = readFloat(*o, "ratio");
121 if (!ratio) return Result<void>::failure(ratio.status());
122 auto limits = readFloatArray(*o, "limits");
123 if (!limits) return Result<void>::failure(limits.status());
124 if (limits.value().size() != 3) return invalidCoupling("Transmission limits need three values");
125 CouplingLimits3D policy{limits.value()[0], limits.value()[1], limits.value()[2]};
126 world.nextCoupling3DId_ = id;
127 std::unique_ptr<Coupling3D> owned;
128 if (gear) {
129 auto reference = number(*o, "reference");
130 if (!reference) return Result<void>::failure(reference.status());
131 auto a = number(*o, "coordinateA");
132 if (!a) return Result<void>::failure(a.status());
133 auto b = number(*o, "coordinateB");
134 if (!b) return Result<void>::failure(b.status());
135 auto first = joints.find(static_cast<int>(sourceA.value())),
136 second = joints.find(static_cast<int>(sourceB.value()));
137 if (first == joints.end() || second == joints.end()) return invalidCoupling("Missing gear source");
138 auto created = Coupling3D::createGear(world, *first->second, *second->second, {ratio.value(), policy});
139 if (!created) return Result<void>::failure(created.status());
140 owned = std::move(created).takeValue();
141 owned->data_->reference = reference.value();
142 owned->data_->coordinate = {a.value(), b.value()};
143 } else {
144 auto length = readFloat(*o, "length");
145 if (!length) return Result<void>::failure(length.status());
146 auto tension = readBool(*o, "tensionOnly");
147 if (!tension) return Result<void>::failure(tension.status());
148 auto la = readPoint(*o, "localAnchorA");
149 if (!la) return Result<void>::failure(la.status());
150 auto lb = readPoint(*o, "localAnchorB");
151 if (!lb) return Result<void>::failure(lb.status());
152 auto ga = readPoint(*o, "groundA");
153 if (!ga) return Result<void>::failure(ga.status());
154 auto gb = readPoint(*o, "groundB");
155 if (!gb) return Result<void>::failure(gb.status());
156 auto* first = world.findBodyById(static_cast<int>(sourceA.value()));
157 auto* second = world.findBodyById(static_cast<int>(sourceB.value()));
158 if (!first || !second) return invalidCoupling("Missing pulley body");
159 auto created = Coupling3D::createPulley(world, *first, *second,
160 {la.value(), lb.value(), ga.value(), gb.value(), ratio.value(),
161 length.value(), tension.value(), policy});
162 if (!created) return Result<void>::failure(created.status());
163 owned = std::move(created).takeValue();
164 }
165 owned->data_->enabled = enabled.value();
166 owned->data_->lastImpulse = impulse.value();
167 owned->data_->saturated = saturated.value();
168 auto row = owned->data_->row();
169 if (!row) return Result<void>::failure(row.status());
170 world.restoredCouplings3D_.push_back(std::move(owned));
171 }
172 return Result<void>::success();
173 } catch (const std::exception& e) {
174 return Result<void>::failure(Diagnostic::error(
175 DiagnosticCode::Failed, std::string("Transmission preparation failed: ") + e.what(), "payload.couplings"));
176 }
