1// Provides the scene camera projection capability.
2//
3// scene/Scene.cpp used to implement pickScreenAt / collectFrustumIdsAt, which
4// forced scene (L1) to include graphics headers (L3) -- an upward dependency.
5// The camera math those entry points need (screen ray, clip matrix) is still
6// graphics' business, but defining scene's members here also made scene's own
7// object files reference a symbol only graphics defines, which a per-layer DLL
8// split cannot link. So scene keeps the entry points and this file provides
9// ISceneCameraProjection instead; scene queries it through common/Capability.h.
10//
11// The file is excluded from the build when scene is disabled (same rule as
12// SceneLinks.cpp), so graphics keeps working without scene.
13

28public:
29 std::optional<eve::scene::ScreenRay> screenRay(Camera3D &cam, float screenX, float screenY, float viewW,
30 float viewH) const override {
31 cam.screenToRay(screenX, screenY, viewW, viewH);
32 eve::scene::ScreenRay ray;
33 ray.origin = {cam.getScreenRayOriginX(), cam.getScreenRayOriginY(), cam.getScreenRayOriginZ()};
34 ray.direction = {cam.getScreenRayDirX(), cam.getScreenRayDirY(), cam.getScreenRayDirZ()};
35 return ray;
36 }

38 std::optional<eve::scene::CameraClip> clipForViewport(Camera3D &cam, float viewW, float viewH) const override {
39 if (!(viewW > 0.f) || !(viewH > 0.f)) return std::nullopt;
40 auto d = cam.data();
41 const glm::mat4 view =
42 glm::lookAtRH(glm::vec3(d->eyeX, d->eyeY, d->eyeZ), glm::vec3(d->targetX, d->targetY, d->targetZ),
43 glm::vec3(d->upX, d->upY, d->upZ));
44 const glm::mat4 projection = graphics::cameraProjectionVulkanRH_ZO(
45 d->orthographic, glm::radians(d->fovYDeg), d->orthoHeight, viewW / viewH, d->nearZ, d->farZ);
46 eve::scene::CameraClip clip;
47 clip.clip = projection * view;
48 clip.inverse = glm::inverse(clip.clip);
49 return clip;
50 }
51};
