101template <class T>
102eve::Value arrayValue(const std::vector<T>& values) {
103 eve::Value::Array encoded;
104 encoded.reserve(values.size());
105 for (const auto value : values) {
106 if constexpr (std::is_same_v<T, float>)
107 encoded.emplace_back(value);
108 else
109 encoded.emplace_back(std::int64_t(value));
110 }
111 return eve::Value(std::move(encoded));

114template <class T>
115bool readArray(const eve::Value* encoded, std::vector<T>& output) {
116 const auto* array = encoded ? encoded->getIf<eve::Value::Array>() : nullptr;
117 if (!array) return false;
118 output.clear();
119 output.reserve(array->size());
120 for (const eve::Value& value : *array) {
121 if constexpr (std::is_same_v<T, float>) {
122 if (const auto* number = value.getIf<double>()) {
123 if (!std::isfinite(*number)) return false;
124 const float converted = static_cast<float>(*number);
125 if (!std::isfinite(converted)) return false;
126 output.push_back(converted);
127 } else if (const auto* integer = value.getIf<std::int64_t>()) {
128 const float converted = static_cast<float>(*integer);
129 if (!std::isfinite(converted)) return false;
130 output.push_back(converted);
131 } else {
132 return false;
133 }
134 } else {
135 const auto* integer = value.getIf<std::int64_t>();
136 if (!integer || *integer < 0 || static_cast<std::uint64_t>(*integer) > std::numeric_limits<T>::max())
137 return false;
138 output.push_back(static_cast<T>(*integer));
139 }
140 }
141 return true;
142}

155bool validGeometry(const std::vector<float>& vertices, const std::vector<std::uint32_t>& indices) {
156 if (vertices.empty() || vertices.size() % 3u != 0u || indices.empty() || indices.size() % 3u != 0u ||
157 vertices.size() / 3u > static_cast<std::size_t>(std::numeric_limits<std::int32_t>::max()))
158 return false;
159 for (const float value : vertices)
160 if (!std::isfinite(value)) return false;
161 for (const auto index : indices)
162 if (index >= vertices.size() / 3u ||
163 index > static_cast<std::uint32_t>(std::numeric_limits<std::int32_t>::max()))
164 return false;
165 return true;
166}

190 PhysicsArtifactStage(PhysicsArtifactProvider& owner, PhysicsArtifactCollider collider,
191 std::unique_ptr<World3D> world, std::unique_ptr<Body3D> body, std::unique_ptr<Shape3D> shape)

225eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>> PhysicsArtifactProvider::prepare(
226 const eve::artifact::PublicationView& publication) {
227 if (failPrepare_)
228 return prepareFailure(eve::DiagnosticCode::Failed, "physics artifact prepare failure was injected");
229 if (publication.schemaVersion != 1 || publication.id.isNil() || publication.buildKey.empty())
230 return prepareFailure(
231 publication.schemaVersion != 1 ? eve::DiagnosticCode::UnknownVersion : eve::DiagnosticCode::InvalidArgument,
232 "physics publication requires schema version 1, identity and build key");
233 if (find(publication.id) != nullptr)
234 return prepareFailure(eve::DiagnosticCode::Conflict, "physics artifact identity is already published");
235
236 const eve::artifact::PartView* colliderPart = nullptr;
237 for (const auto& part : publication.parts) {
238 if (part.kind == eve::artifact::PartKind::Collider && (part.role == "collider" || colliderPart == nullptr))
239 colliderPart = &part;
240 }
241 if (!colliderPart)
242 return prepareFailure(eve::DiagnosticCode::PreconditionViolation,
243 "physics publication requires a collider part");
244 if (colliderPart->schemaVersion != 1)
245 return prepareFailure(eve::DiagnosticCode::UnknownVersion, "physics collider schema version is unsupported");
246
247 PhysicsArtifactCollider collider;
248 collider.id = publication.id;
249 collider.buildKey = std::string(publication.buildKey);
250 collider.shape = "triangle_mesh";
251 collider.backend = "box3d";
252 collider.bounds = colliderPart->bounds;
253 try {
254 collider.vertices.assign(colliderPart->positions.begin(), colliderPart->positions.end());
255 collider.indices.assign(colliderPart->indices.begin(), colliderPart->indices.end());
256 if (!validBounds(collider.bounds) || !validGeometry(collider.vertices, collider.indices))
257 return prepareFailure(eve::DiagnosticCode::PreconditionViolation,
258 "physics publication requires finite valid triangle geometry");
259 if (nextIndex_ == PhysicsArtifactHandle::invalidIndex)
260 return prepareFailure(eve::DiagnosticCode::Failed, "physics artifact handle index exhausted");
261
262 // Allocate the artifact handle during prepare. A rolled-back stage
263 // burns this slot intentionally; reusing it would make a stale handle
264 // resolve to a different collider.
265 collider.handle = PhysicsArtifactHandle(nextIndex_, 1u);
266 ++nextIndex_;
267 state_->records.reserve(state_->records.size() + state_->pendingStages + 1u);
268
269 auto world = std::make_unique<World3D>(0.f, 0.f, 0.f, true);
270 auto body = std::unique_ptr<Body3D>(world->newBody("static", 0.f, 0.f, 0.f));
271 if (!body)
272 return prepareFailure(eve::DiagnosticCode::Failed, "Box3D did not create the generated collider body");
273 auto mesh = signedIndices(collider.indices);
274 auto shape = std::unique_ptr<Shape3D>(body->newTriangleMeshShape(collider.vertices, mesh));
275 if (!body || !shape)
276 return prepareFailure(eve::DiagnosticCode::Failed, "Box3D did not create the generated collider");
277 auto stage = std::make_unique<PhysicsArtifactStage>(*this, std::move(collider), std::move(world),
278 std::move(body), std::move(shape));
279 ++state_->pendingStages;
280 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::success(std::move(stage));
281 } catch (const std::exception& error) {
282 return prepareFailure(eve::DiagnosticCode::Failed,
283 std::string("Box3D collider construction failed: ") + error.what());
284 } catch (...) {
285 return prepareFailure(eve::DiagnosticCode::Failed, "Box3D collider construction failed");
286 }
287}

300void PhysicsArtifactProvider::release(PhysicsArtifactCollider& collider, std::unique_ptr<World3D>& world,
301 std::unique_ptr<Body3D>& body, std::unique_ptr<Shape3D>& shape) noexcept {
302 // Explicitly tear down the backend before wrapper ownership is released.
303 // This also makes rollback behavior independent of Box3D's wrapper order.
304 if (state_->pendingStages > 0) --state_->pendingStages;
305 if (shape) shape->destroy();
306 shape.reset();
307 if (body) body->destroy();
308 body.reset();
309 if (world) world->destroy();
310 world.reset();
311 collider = {};
312}

331PhysicsArtifactRayHit PhysicsArtifactProvider::rayCast(eve::PersistentId id, float ox, float oy, float oz, float dx,
332 float dy, float dz) const noexcept {
333 PhysicsArtifactRayHit result;
334 const auto found = std::find_if(state_->records.begin(), state_->records.end(),
335 [id](const auto& record) { return record.descriptor.id == id; });
336 if (found == state_->records.end()) return result;
337 const auto& runtime = *found;
338 if (!runtime.world || !runtime.body || !runtime.shape || !runtime.world->isValid() || !runtime.body->isValid() ||
339 !runtime.shape->isValid())
340 return result;
341 if (!std::isfinite(ox) || !std::isfinite(oy) || !std::isfinite(oz) || !std::isfinite(dx) || !std::isfinite(dy) ||
342 !std::isfinite(dz))
343 return result;
344 const float ex = ox + dx;
345 const float ey = oy + dy;
346 const float ez = oz + dz;
347 if (!std::isfinite(ex) || !std::isfinite(ey) || !std::isfinite(ez)) return result;
348 try {
349 const int bodyId = runtime.world->rayCast(ox, oy, oz, ex, ey, ez);
350 if (bodyId < 0 || runtime.world->getRayResultCount() <= 0) return result;
351 result.hit = true;
352 result.fraction = runtime.world->getRayResultFraction(0);
353 result.x = runtime.world->getRayResultX(0);
354 result.y = runtime.world->getRayResultY(0);
355 result.z = runtime.world->getRayResultZ(0);
356 result.normalX = runtime.world->getRayResultNormalX(0);
357 result.normalY = runtime.world->getRayResultNormalY(0);
358 result.normalZ = runtime.world->getRayResultNormalZ(0);
359 } catch (...) {
360 // The public query is a noexcept probe. Invalid input or a backend
361 // query failure is represented as a miss; the collider remains intact.
362 return PhysicsArtifactRayHit{};
363 }
364 return result;
365}

392eve::Result<eve::Value> PhysicsArtifactProvider::snapshotState() const {
393 try {
394 eve::Value::Array colliders;
395 colliders.reserve(state_->records.size());
396 for (const auto& runtime : state_->records) {
397 const auto& collider = runtime.descriptor;
398 eve::Value::Object item;
399 item.emplace("artifactId", eve::Value(collider.id.format()));
400 item.emplace("buildKey", eve::Value(collider.buildKey));
401 // Kept for diagnostics and compatibility. restoreState deliberately
402 // regenerates this process-local handle rather than reviving it.
403 item.emplace("handle", eve::Value(std::to_string(collider.handle.packed())));
404 item.emplace("shape", eve::Value(collider.shape));
405 item.emplace("backend", eve::Value(collider.backend));
406 item.emplace("bounds", boundsValue(collider.bounds));
407 item.emplace("vertices", arrayValue(collider.vertices));
408 item.emplace("indices", arrayValue(collider.indices));
409 colliders.emplace_back(std::move(item));
410 }
411 eve::Value::Object state;
412 state.emplace("provider", eve::Value("physics.box3d-collider"));
413 state.emplace("version", eve::Value(std::int64_t(1)));
414 state.emplace("colliders", eve::Value(std::move(colliders)));
415 return eve::Result<eve::Value>::success(eve::Value(std::move(state)));
416 } catch (...) {
417 return eve::Result<eve::Value>::failure(
418 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "physics artifact snapshot allocation failed"));
419 }

422eve::Result<void> PhysicsArtifactProvider::restoreState(const eve::Value& state) {
423 if (state_->pendingStages != 0)
424 return restoreFailure(eve::DiagnosticCode::Conflict,
425 "cannot restore physics artifacts while a publication is staged");
426 const auto* object = state.getIf<eve::Value::Object>();
427 const auto* provider = object ? member(*object, "provider") : nullptr;
428 const auto* providerName = provider ? provider->getIf<std::string>() : nullptr;
429 const auto* encodedColliders = object ? member(*object, "colliders") : nullptr;
430 const auto* colliders = encodedColliders ? encodedColliders->getIf<eve::Value::Array>() : nullptr;
431 if (!object || !providerName || *providerName != "physics.box3d-collider" || !hasSupportedVersion(*object) ||
432 !colliders)
433 return restoreFailure(eve::DiagnosticCode::ParseError, "invalid physics artifact provider state");
434
435 std::vector<State::RuntimeCollider> candidate;
436 std::unordered_set<eve::PersistentId> identities;
437 std::uint32_t next = nextIndex_;
438 try {
439 candidate.reserve(colliders->size());
440 for (const eve::Value& encoded : *colliders) {
441 const auto* item = encoded.getIf<eve::Value::Object>();
442 std::string idText;
443 std::string buildKey;
444 std::string shape;
445 std::string backend;
446 std::uint64_t packed = 0;
447 PhysicsArtifactCollider collider;
448 if (!item || !readString(*item, "artifactId", idText) || !readString(*item, "buildKey", buildKey) ||
449 !readString(*item, "shape", shape) || !readString(*item, "backend", backend) ||
450 !readU64(*item, "handle", packed) || !readBounds(*item, collider.bounds) ||
451 !readArray(member(*item, "vertices"), collider.vertices) ||
452 !readArray(member(*item, "indices"), collider.indices))
453 return restoreFailure(eve::DiagnosticCode::ParseError, "invalid physics artifact collider state");
454 const auto parsedId = eve::PersistentId::parse(idText);
455 const auto persistedHandle = PhysicsArtifactHandle::fromPacked(packed);
456 if (!parsedId || parsedId->isNil() || persistedHandle.isInvalid() || backend != "box3d" ||
457 shape != "triangle_mesh" || !identities.emplace(*parsedId).second ||
458 !validGeometry(collider.vertices, collider.indices))
459 return restoreFailure(eve::DiagnosticCode::Unsupported,
460 "physics artifact state has unsupported backend or geometry");
461
462 if (next == PhysicsArtifactHandle::invalidIndex)
463 return restoreFailure(eve::DiagnosticCode::Failed,
464 "physics artifact handle index exhausted during restore");
465 collider.id = *parsedId;
466 collider.buildKey = std::move(buildKey);
467 collider.shape = std::move(shape);
468 collider.backend = std::move(backend);
469 // Runtime handles are deliberately regenerated. This preserves
470 // stale detection across clear/restore and keeps process-local
471 // coordinates out of the persistent contract.
472 collider.handle = PhysicsArtifactHandle(next++, 1u);
473
474 auto world = std::make_unique<World3D>(0.f, 0.f, 0.f, true);
475 auto body = std::unique_ptr<Body3D>(world->newBody("static", 0.f, 0.f, 0.f));
476 if (!body)
477 return restoreFailure(eve::DiagnosticCode::Failed, "Box3D did not recreate the restored collider body");
478 const auto mesh = signedIndices(collider.indices);
479 auto shapeObject = std::unique_ptr<Shape3D>(body->newTriangleMeshShape(collider.vertices, mesh));
480 if (!body || !shapeObject)
481 return restoreFailure(eve::DiagnosticCode::Failed, "Box3D did not recreate a restored collider");
482 State::RuntimeCollider runtime;
483 runtime.descriptor = std::move(collider);
484 runtime.world = std::move(world);
485 runtime.body = std::move(body);
486 runtime.shape = std::move(shapeObject);
487 candidate.push_back(std::move(runtime));
488 }
489 } catch (const std::exception& error) {
490 return restoreFailure(eve::DiagnosticCode::Failed, std::string("Box3D restore failed: ") + error.what());
491 } catch (...) {
492 return restoreFailure(eve::DiagnosticCode::Failed, "Box3D restore failed");
493 }
494
495 state_->records.swap(candidate);
496 nextIndex_ = next;
497 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
498}
