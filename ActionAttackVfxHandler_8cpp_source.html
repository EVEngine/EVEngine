26eve::Result<eve::LogicalId> ensureRecipe(eve::stylize::AttackVfxRuntime& runtime,
27 const eve::action::ActionAttackVfxBinding& binding) {
28 if (binding.recipeId.isValid()) {
29 if (!runtime.findRecipe(binding.recipeId))
30 return eve::Result<eve::LogicalId>::failure(eve::Diagnostic::error(
31 eve::DiagnosticCode::NotFound, "AttackVfx recipeId is not registered", "recipeId"));
32 return eve::Result<eve::LogicalId>::success(binding.recipeId);
33 }
34
35 std::string json;
36 if (binding.uri.rfind("json:", 0) == 0) {
37 json = binding.uri.substr(5);
38 } else {
39 auto* fs = eve::ModuleManager::getInstance<eve::filesystem::Filesystem>("Filesystem");
40 if (!fs)
41 return eve::Result<eve::LogicalId>::failure(eve::Diagnostic::error(
42 eve::DiagnosticCode::NotFound, "Filesystem is required to load AttackVfx uri", "uri"));
43 json = fs->readText(binding.uri);
44 if (json.empty())
45 return eve::Result<eve::LogicalId>::failure(eve::Diagnostic::error(
46 eve::DiagnosticCode::NotFound, "AttackVfx recipe uri could not be read", "uri"));
47 }
48 auto parsed = eve::stylize::AttackVfxRecipe::fromJson(json);
49 if (!parsed) return eve::Result<eve::LogicalId>::failure(parsed.status());
50 auto recipe = std::move(parsed).takeValue();
51 const eve::LogicalId id = recipe.id;
52 if (!runtime.findRecipe(id)) {
53 auto registered = runtime.registerRecipe(recipe);
54 if (!registered) return eve::Result<eve::LogicalId>::failure(registered.status());
55 }
56 return eve::Result<eve::LogicalId>::success(id);
57}

78class ActionAttackVfxHandler final : public eve::action::IActionNotifyHandler {
79public:
80 eve::Result<void> handle(const eve::action::ActionTimelineEvent& event,
81 const eve::action::ActionNotifyContext& context) override {
82 auto* runtime = moduleRuntime();
83 if (!runtime)
84 return eve::Result<void>::failure(eve::Diagnostic::error(
85 eve::DiagnosticCode::NotFound, "StylizeAction module is unavailable", "stylizeAction"));
86
87 const ActiveKey key{context.executionId, event.itemId.format()};
88 if (event.kind == eve::action::ActionTimelineEventKind::StateExit) {
89 const auto found = active_.find(key);
90 if (found == active_.end())
91 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
92 auto stopped = runtime->stop(found->second.handle, eve::stylize::AttackVfxStopMode::StopEmitting);
93 active_.erase(found);
94 if (!stopped) {
95 // Recipe may already have completed and invalidated the handle.
96 if (stopped.code() == eve::StatusCode::Rejected ||
97 stopped.code() == eve::StatusCode::NotFound)
98 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
99 return eve::Result<void>::failure(stopped.status());
100 }
101 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
102 }
103
104 const bool instant = event.kind == eve::action::ActionTimelineEventKind::Notify;
105 if (!instant && event.kind != eve::action::ActionTimelineEventKind::StateEnter)
106 return eve::Result<void>::failure(eve::Diagnostic::error(
107 eve::DiagnosticCode::InvalidArgument,
108 "AttackVfx state handler requires enter or exit boundary", "event.kind"));
109 if (!instant && active_.contains(key))
110 return eve::Result<void>::failure(eve::Diagnostic::error(
111 eve::DiagnosticCode::Conflict, "AttackVfx state is already active", "itemId"));
112
113 const auto shape =
114 instant ? eve::action::ActionAttackVfxShape::Instant : eve::action::ActionAttackVfxShape::State;
115 auto binding = eve::action::ActionAttackVfxBinding::fromPayload(event.payload, shape);
116 if (!binding) return eve::Result<void>::failure(binding.status());
117
118 auto recipeId = ensureRecipe(*runtime, binding.value());
119 if (!recipeId) return eve::Result<void>::failure(recipeId.status());
120
121 auto request = makeRequest(binding.value(), context);
122 auto played = runtime->play(recipeId.value(), request);
123 if (!played) return eve::Result<void>::failure(played.status());
124
125 ActiveInstance owned;
126 owned.handle = std::move(played).takeValue();
127 owned.binding = std::move(binding).takeValue();
128 owned.startTime = context.time;
129 owned.executionId = context.executionId;
130 owned.nextCue = 0;
131 owned.lastAge = 0.0;
132 executionClocks_[context.executionId] = context.time.seconds();
133
134 if (instant) {
135 auto duration = eve::Duration::fromSeconds(owned.binding.lifetimeSeconds);
136 if (!duration) return eve::Result<void>::failure(duration.status());
137 auto end = context.time.tryAdd(duration.value());
138 if (!end) return eve::Result<void>::failure(end.status());
139 owned.endTime = std::move(end).takeValue();
140 transients_.push_back(std::move(owned));
141 } else {
142 active_.emplace(key, std::move(owned));
143 }
144 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
145 }

147 eve::Result<void> update(const eve::action::ActionActiveBlock& block,
148 const eve::action::ActionNotifyContext& context) override {
149 auto* runtime = moduleRuntime();
150 if (!runtime)
151 return eve::Result<void>::failure(eve::Diagnostic::error(
152 eve::DiagnosticCode::NotFound, "StylizeAction module is unavailable", "stylizeAction"));
153 const auto found = active_.find({context.executionId, block.itemId.format()});
154 if (found == active_.end())
155 return eve::Result<void>::failure(eve::Diagnostic::error(
156 eve::DiagnosticCode::NotFound, "Active AttackVfx state has no instance", "itemId"));
157 return deliverCues(*runtime, found->second, context.time);
158 }

160 eve::Result<void> advance(const eve::action::ActionNotifyContext& context) override {
161 auto* runtime = moduleRuntime();
162 if (!runtime)
163 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::NoOp));
164
165 bool changed = false;
166 const double now = context.time.seconds();
167
168 // Per-execution clocks: concurrent actions must not share one lastAdvance.
169 auto clockIt = executionClocks_.find(context.executionId);
170 if (clockIt == executionClocks_.end()) {
171 clockIt = executionClocks_.emplace(context.executionId, now).first;
172 }
173 double dt = now - clockIt->second;
174 if (context.scrubbing || dt < 0.0) dt = 0.0;
175 clockIt->second = now;
176 if (dt > 0.0) {
177 auto frame = runtime->advance(dt);
178 if (!frame) return eve::Result<void>::failure(frame.status());
179 changed = !frame.value().advanced.empty() || !frame.value().stopped.empty() ||
180 !frame.value().events.empty();
181 }
182
183 for (auto& active : active_) {
184 if (active.second.executionId != context.executionId) continue;
185 auto cues = deliverCues(*runtime, active.second, context.time);
186 if (!cues) return cues;
187 changed = changed || cues.code() == eve::StatusCode::Applied;
188 }
189
190 for (auto it = transients_.begin(); it != transients_.end();) {
191 if (it->executionId != context.executionId) {
192 ++it;
193 continue;
194 }
195 auto cues = deliverCues(*runtime, *it, context.time);
196 if (!cues) return cues;
197 if (context.time < it->endTime) {
198 ++it;
199 continue;
200 }
201 auto stopped = runtime->stop(it->handle, eve::stylize::AttackVfxStopMode::ClearImmediately);
202 if (!stopped) {
203 // Instant recipes often complete before lifetime ends; ignore stale stops.
204 if (stopped.code() != eve::StatusCode::Rejected &&
205 stopped.code() != eve::StatusCode::NotFound)
206 return eve::Result<void>::failure(stopped.status());
207 }
208 it = transients_.erase(it);
209 changed = true;
210 }
211
212 return eve::Result<void>::success(
213 eve::Status::success(changed ? eve::StatusCode::Applied : eve::StatusCode::NoOp));
214 }

219 struct ActiveInstance {
220 eve::stylize::AttackVfxHandle handle{};
221 eve::action::ActionAttackVfxBinding binding;
222 eve::Duration startTime = eve::Duration::zero();
223 eve::Duration endTime = eve::Duration::zero();
224 eve::action::ActionExecutionId executionId{};
225 std::size_t nextCue = 0;
226 double lastAge = 0.0;
227 };

229 static eve::Result<void> deliverCues(eve::stylize::AttackVfxRuntime& runtime, ActiveInstance& active,
230 eve::Duration now) {
231 const double age =
232 static_cast<double>(now.nanoseconds() - active.startTime.nanoseconds()) / 1000000000.0;
233 bool changed = false;
234 while (active.nextCue < active.binding.cues.size()) {
235 const auto& cue = active.binding.cues[active.nextCue];
236 if (cue.offsetSeconds > age) break;
237 if (cue.offsetSeconds + 1e-9 >= active.lastAge) {
238 auto signaled = runtime.signal(active.handle, cue.cue);
239 if (!signaled) {
240 if (signaled.code() == eve::StatusCode::Rejected ||
241 signaled.code() == eve::StatusCode::NotFound) {
242 ++active.nextCue;
243 continue;
244 }
245 return eve::Result<void>::failure(signaled.status());
246 }
247 changed = true;
248 }
249 ++active.nextCue;
250 }
251 active.lastAge = age;
252 return eve::Result<void>::success(
253 eve::Status::success(changed ? eve::StatusCode::Applied : eve::StatusCode::NoOp));
254 }
255
256 std::map<ActiveKey, ActiveInstance> active_;
257 std::vector<ActiveInstance> transients_;
258 std::map<eve::action::ActionExecutionId, double> executionClocks_;
259};
