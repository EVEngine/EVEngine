38void dualGridHalfOffset(const TileLayer::Config &cfg, float &offX, float &offY) {
39 switch (cfg.orientation) {
40 case MapOrientation::Isometric:
41 // tileToWorld(tx-0.5, ty-0.5) with same iso formula ⇒ origin (ox, oy - th/2).
42 offX = 0.f;
43 offY = -cfg.tileH * 0.5f;
44 break;
45 case MapOrientation::Staggered:
46 case MapOrientation::Hexagonal:
47 if (cfg.staggerAxis == StaggerAxis::Y) {
48 offX = -cfg.tileW * 0.5f;
49 offY = -staggerPitchY(cfg) * 0.5f;
50 } else {
51 offX = -staggerPitchX(cfg) * 0.5f;
52 offY = -cfg.tileH * 0.5f;
53 }
54 break;
55 case MapOrientation::Orthogonal:
56 default:
57 offX = -cfg.tileW * 0.5f;
58 offY = -cfg.tileH * 0.5f;
59 break;
60 }
61}

80bool resolveDualGrid(TileLayer *logic, TileLayer *display, const DualGridOptions &opts,
81 std::string *error) {
82 if (!logic || !display) {
83 if (error) *error = "resolveDualGrid: logic and display layers required";
84 return false;
85 }
86 if (logic == display) {
87 if (error) *error = "resolveDualGrid: logic and display must be different layers";
88 return false;
89 }
90
91 const int logicW = logic->getMapWidth();
92 const int logicH = logic->getMapHeight();
93 if (logicW <= 0 || logicH <= 0) {
94 if (error) *error = "resolveDualGrid: logic layer has empty size";
95 return false;
96 }
97
98 const float tileW = logic->getTileWidth();
99 const float tileH = logic->getTileHeight();
100 display->setTileSize(tileW, tileH);
101 display->resize(logicW + 1, logicH + 1);
102
103 auto lc = logic->config();
104 auto dc = display->config();
105 dc->orientation = lc->orientation;
106 dc->staggerAxis = lc->staggerAxis;
107 dc->staggerIndex = lc->staggerIndex;
108 dc->hexSideLength = lc->hexSideLength;
109
110 if (opts.applyHalfOffset) {
111 float offX = 0.f, offY = 0.f;
112 dualGridHalfOffset(*lc, offX, offY);
113 display->setOrigin(logic->getX() + offX, logic->getY() + offY);
114 } else {
115 display->setOrigin(logic->getX(), logic->getY());
116 }
117
118 // Keep draw bookkeeping in sync when display is freshly created.
119 display->setLayer(logic->getLayer());
120 display->setCamera(logic->draw()->camera);
121 display->setCanvas(logic->draw()->canvas);
122
123 const int firstGid = resolveFirstGid(display, opts.firstDisplayGid);
124 const int filledGid = opts.filledGid;
125
126 for (int dy = 0; dy < logicH + 1; ++dy) {
127 for (int dx = 0; dx < logicW + 1; ++dx) {
128 const int mask = dualGridMaskAt(*logic, dx, dy, filledGid);
129 int frame = -1;
130 if (opts.useDefaultFrameTable) {
131 frame = dualGridDefaultFrame(mask);
132 } else if (mask != 0) {
133 frame = mask;
134 }
135 const int gid = (frame < 0) ? 0 : firstGid + frame;
136 display->setTile(dx, dy, gid);
137 }
138 }
139
140 if (opts.hideLogic) logic->setVisible(false);
141 display->setVisible(true);
142
143 if (error) error->clear();
144 return true;
145}
