62editing::Result<std::string> EditorHistory::undoAction() {
63 if (undoStack_.empty())
64 return eve::editing::failed<std::string>(editing::Status::NotFound, editing::RuleId("editor.history.nothing-to-undo"),
65 "Editor history has no action to undo");
66 Action a = undoStack_.back();
67 undoStack_.pop_back();
68 redoStack_.push_back(a);
69 lastApplied_ = a;
70 hasLast_ = true;
71 lastWasUndo_ = true;
72 return eve::editing::applied<std::string>(a.name);
73}

92editing::Result<void> EditorHistory::applyLastToBufferChecked(TileBuffer& buffer) {
93 if (!hasLast_)
94 return eve::editing::failed<void>(editing::Status::NotFound, editing::RuleId("editor.history.no-last-action"),
95 "Editor history has no applied action");
96 if (lastApplied_.kind != "tiles")
97 return eve::editing::failed<void>(editing::Status::Unsupported, editing::RuleId("editor.history.action-not-tiles"),
98 "The latest history action does not contain tile changes");
99 for (const auto &t : lastApplied_.tiles) {
100 if (!buffer.containsCell(t.x, t.y)) continue;
101 buffer.setGid(t.x, t.y, lastWasUndo_ ? t.oldGid : t.newGid);
102 }
103 return eve::editing::applied<void>();
104}

106bool EditorHistory::applyLastToBuffer(TileBuffer *buffer) {
