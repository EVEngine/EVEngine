21Result<void> finalizeImportReport(PreparedAssetImport& prepared,
22 const ImportPackageIdentity& package,
23 std::string_view sourceEngine,
24 std::string_view sourceVersion,
25 Value::Object options) {
26 if (sourceEngine.empty() || prepared.manifest.assets.empty())
27 return failure<void>(DiagnosticCode::InvalidArgument,
28 "import report requires source engine and canonical assets");
29 std::sort(prepared.sourceMappings.begin(), prepared.sourceMappings.end(),
30 [](const auto& left, const auto& right) {
31 if (left.sourceObject != right.sourceObject)
32 return left.sourceObject < right.sourceObject;
33 return left.asset.id() < right.asset.id();
34 });
35 for (std::size_t index = 1; index < prepared.sourceMappings.size(); ++index) {
36 if (prepared.sourceMappings[index - 1].sourceObject ==
37 prepared.sourceMappings[index].sourceObject &&
38 prepared.sourceMappings[index - 1].asset == prepared.sourceMappings[index].asset)
39 return failure<void>(DiagnosticCode::AlreadyExists,
40 "duplicate import source mapping",
41 prepared.sourceMappings[index].sourceObject);
42 }
43 std::sort(prepared.findings.begin(), prepared.findings.end(), [](const auto& left, const auto& right) {
44 return std::tie(left.sourcePath, left.feature, left.disposition, left.message) <
45 std::tie(right.sourcePath, right.feature, right.disposition, right.message);
46 });

48 Value::Object report;
49 report["schema"] = Value("eve.asset-import-report");
50 report["schemaVersion"] = Value(std::int64_t(1));
51 report["sourceEngine"] = Value(std::string(sourceEngine));
52 report["sourceVersion"] = Value(std::string(sourceVersion));
53 report["packageName"] = Value(package.packageName);
54 report["packageVersion"] = Value(package.packageVersion);
55 report["importer"] = prepared.manifest.provenance.contains("importer")
56 ? prepared.manifest.provenance.at("importer")
57 : Value("unknown");
58 report["importerVersion"] = prepared.manifest.provenance.contains("importerVersion")
59 ? prepared.manifest.provenance.at("importerVersion")
60 : Value(std::int64_t(0));
61 report["options"] = Value(std::move(options));
62
63 Value::Array mappings;
64 for (const auto& mapping : prepared.sourceMappings)
65 mappings.emplace_back(Value::Object{{"sourceObject", Value(mapping.sourceObject)},
66 {"asset", Value(mapping.asset.format())}});
67 report["sourceObjects"] = Value(std::move(mappings));
68
69 Value::Object counts{{"translated", Value(std::int64_t(0))},
70 {"baked", Value(std::int64_t(0))},
71 {"preserved-source", Value(std::int64_t(0))},
72 {"unsupported", Value(std::int64_t(0))}};
73 Value::Array findings;
74 for (const auto& finding : prepared.findings) {
75 const std::string disposition = dispositionName(finding.disposition);
76 counts[disposition] = Value(counts.at(disposition).asInt() + 1);
77 findings.emplace_back(Value::Object{{"sourcePath", Value(finding.sourcePath)},
78 {"feature", Value(finding.feature)},
79 {"disposition", Value(disposition)},
80 {"message", Value(finding.message)}});
81 }
82 report["counts"] = Value(std::move(counts));
83 report["findings"] = Value(std::move(findings));
84
85 Value::Array assets;
86 for (const auto& asset : prepared.manifest.assets)
87 assets.emplace_back(Value::Object{{"asset", Value(asset.asset.format())},
88 {"type", Value(asset.type + "/" +
89 std::to_string(asset.schemaVersion.value()))},
90 {"contentHash", Value(asset.contentHash)}});
91 report["canonicalAssets"] = Value(std::move(assets));
92 auto keySeed = Value(report).toJson();
93 if (!keySeed) return Result<void>::failure(keySeed.status());
94 const std::string importKey = sha256(std::span<const std::uint8_t>(
95 reinterpret_cast<const std::uint8_t*>(keySeed.value().data()), keySeed.value().size()));
96 report["importKey"] = Value(importKey);
97 auto encoded = Value(std::move(report)).toJson();
98 if (!encoded) return Result<void>::failure(encoded.status());
99 prepared.entries.push_back({"reports/import.json",
100 {encoded.value().begin(), encoded.value().end()}});
101 prepared.manifest.provenance["path"] = Value("reports/import.json");
102 prepared.manifest.provenance["importKey"] = Value(importKey);
103 prepared.manifest.provenance["sourceEngine"] = Value(std::string(sourceEngine));
104 prepared.manifest.provenance["sourceVersion"] = Value(std::string(sourceVersion));
105 return Result<void>::success();
106}
