187void accumulateRootMotion(animation::TransformTRS& total, const animation::TransformTRS& delta) {
188 total.px += delta.px;
189 total.py += delta.py;
190 total.pz += delta.pz;
191 const float tx = total.qx;
192 const float ty = total.qy;
193 const float tz = total.qz;
194 const float tw = total.qw;
195 total.qx = tw * delta.qx + tx * delta.qw + ty * delta.qz - tz * delta.qy;
196 total.qy = tw * delta.qy - tx * delta.qz + ty * delta.qw + tz * delta.qx;
197 total.qz = tw * delta.qz + tx * delta.qy - ty * delta.qx + tz * delta.qw;
198 total.qw = tw * delta.qw - tx * delta.qx - ty * delta.qy - tz * delta.qz;
199}

549 while (true) {
550 Duration remainingTimeline =
551 Duration::fromNanoseconds(runtimeTimeline_->duration.nanoseconds() - montage->time().nanoseconds());
552 if (runtimeTimeline_->montage.looping && remainingTimeline.isZero()) {
553 auto restarted = restartExecution();
554 if (!restarted) return Result<animation::MontageAdvance>::failure(restarted.status());
555 montage = activeMontage();
556 remainingTimeline = runtimeTimeline_->duration;
557 }
558 const Duration slice =
559 runtimeTimeline_->montage.looping ? std::min(remainingDelta, remainingTimeline) : remainingDelta;
560 tick_ = SimulationTick(tick_.value() + 1);
561 auto actionAdvance = runtime_->advance(executionId_, tick_, slice);
562 if (!actionAdvance) return Result<animation::MontageAdvance>::failure(actionAdvance.status());
563 auto routed = routeAvailableBlocks(actionAdvance.value());
564 if (!routed) return Result<animation::MontageAdvance>::failure(routed.status());
565 auto presented = coordinator_->present(montageHandle_, actionAdvance.value(), tick_);
566 if (!presented) return Result<animation::MontageAdvance>::failure(presented.status());
567 auto faded = coordinator_->advanceBlendOuts(slice, tick_, montageHandle_);
568 if (!faded) return Result<animation::MontageAdvance>::failure(faded.status());
569 auto value = std::move(presented).takeValue();
570 if (firstSlice) {
571 combined = std::move(value);
572 firstSlice = false;
573 } else {
574 combined.current = value.current;
575 combined.sectionId = value.sectionId;
576 combined.events.insert(combined.events.end(), std::make_move_iterator(value.events.begin()),
577 std::make_move_iterator(value.events.end()));
578 combined.activeBlocks = std::move(value.activeBlocks);
579 accumulateRootMotion(combined.rootMotion, value.rootMotion);
580 combined.weight = value.weight;
581 combined.completed = value.completed;
582 }
583 if (!runtimeTimeline_->montage.looping || remainingDelta <= slice) break;
584
585 remainingDelta = Duration::fromNanoseconds(remainingDelta.nanoseconds() - slice.nanoseconds());
586 auto restarted = restartExecution();
587 if (!restarted) return Result<animation::MontageAdvance>::failure(restarted.status());
588 }
589 if (runtimeTimeline_->montage.looping && combined.completed) {
590 auto restarted = restartExecution();
591 if (!restarted) return Result<animation::MontageAdvance>::failure(restarted.status());
592 combined.current = Duration::zero();
593 combined.sectionId.reset();
594 combined.activeBlocks.clear();
595 combined.completed = false;
596 montage = activeMontage();
597 combined.weight = montage ? montage->weight() : 0.0;
598 }
599 runtimeAdvance_ = combined;
600 return Result<animation::MontageAdvance>::success(std::move(combined), Status::success(StatusCode::Pending));
601 }

899 const bool hasActiveMontage = activeMontage() != nullptr;
900 const auto previousHandle = montageHandle_;
901 const auto previousLayer = activeLayer_;
902 if (!executionId_.isZero()) {

905 if (!interrupted) return interrupted;

909 action::ActionRequest request;
910 request.actionId = definition.id;
911 auto submitted = runtime->submit(std::move(definition), std::move(request));
912 if (!submitted) return Result<void>::failure(submitted.status());
913 if (playMontage) {
914 if (!coordinator_)
915 return Result<void>::failure(
916 Diagnostic::error(DiagnosticCode::Conflict, "montage coordinator has not been started", "runtime"));
917 if (hasActiveMontage) tick_ = SimulationTick(tick_.value() + 1);
918 const std::size_t layer = runtimeTimeline_->montage.animationLayer;
919 auto played = coordinator_->play(layer, *runtimeTimeline_, cloneRuntimeClips(), submitted.value(), tick_);
920 if (!played) return Result<void>::failure(played.status());
921 montageHandle_ = played.value();
922 activeLayer_ = layer;
923 if (hasActiveMontage && previousLayer != layer) {
924 auto previous = coordinator_->resolve(previousHandle);
925 if (previous) previous.value().get().stop();
926 coordinator_->collectFinished();
927 }
928 }
929 runtime_ = std::move(runtime);
930 executionId_ = submitted.value();
931 return Result<void>::success(Status::success(StatusCode::Applied));
932 }
