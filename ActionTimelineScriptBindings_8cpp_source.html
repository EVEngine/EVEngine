99void accumulateRootMotion(animation::TransformTRS& total, const animation::TransformTRS& delta) {
100 total.px += delta.px;
101 total.py += delta.py;
102 total.pz += delta.pz;
103 const float tx = total.qx;
104 const float ty = total.qy;
105 const float tz = total.qz;
106 const float tw = total.qw;
107 total.qx = tw * delta.qx + tx * delta.qw + ty * delta.qz - tz * delta.qy;
108 total.qy = tw * delta.qy - tx * delta.qz + ty * delta.qw + tz * delta.qx;
109 total.qz = tw * delta.qz + tx * delta.qy - ty * delta.qx + tz * delta.qw;
110 total.qw = tw * delta.qw - tx * delta.qx - ty * delta.qy - tz * delta.qz;
111}

127 [[nodiscard]] Result<void> registerRuntimeClip(std::string uri, model3d::ModelData& model,
128 animation::AnimSkeleton& skeleton, int animationIndex) {
129 if (uri.empty())
130 return Result<void>::failure(
131 Diagnostic::error(DiagnosticCode::InvalidArgument, "montage clip URI must not be empty", "uri"));
132 if (animationIndex < 0 || animationIndex >= animation::AnimImporter::getAnimationCountFromModel(&model))
133 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
134 "montage animation index is outside the model", "index"));
135 std::unique_ptr<animation::AnimClip> clip(
136 animation::AnimImporter::loadClipFromModel(&model, &skeleton, animationIndex));
137 if (!clip)
138 return Result<void>::failure(
139 Diagnostic::error(DiagnosticCode::Failed, "montage clip import produced no animation", uri));
140 if (montage_) return montage_->replaceClip(uri, std::move(clip));
141 const auto found = std::find_if(runtimeClips_.begin(), runtimeClips_.end(),
142 [&](const auto& asset) { return asset.uri == uri; });
143 if (found == runtimeClips_.end())
144 runtimeClips_.push_back({std::move(uri), std::move(clip)});
145 else
146 found->clip = std::move(clip);
147 return Result<void>::success(Status::success(StatusCode::Applied));
148 }

150 [[nodiscard]] Result<void> beginRuntime(animation::AnimSkeleton& skeleton) {
151 auto montage = std::make_unique<animation::MontagePlayer>(skeleton);
152 auto timeline = editor_.target().timeline();
153 auto prepared = montage->prepare(timeline, std::move(runtimeClips_));
154 if (!prepared) return Result<void>::failure(prepared.status());
155 runtimeTimeline_ = timeline;
156 montage_ = std::move(montage);
157 tick_ = SimulationTick::zero();
158 runtimeAdvance_.reset();
159 runtimeRate_ = timeline.montage.basePlayRate;
160 runtimePaused_ = false;
161 sectionRates_.clear();
162 return restartExecution();
163 }

165 [[nodiscard]] Result<animation::MontageAdvance> advanceRuntime(Duration delta) {
166 if (!runtime_ || !montage_)
167 return Result<animation::MontageAdvance>::failure(
168 Diagnostic::error(DiagnosticCode::Conflict, "montage runtime has not been started", "runtime"));
169 if (montage_->isBlendingOut()) {
170 tick_ = SimulationTick(tick_.value() + 1);
171 auto settled = montage_->advanceBlendOut(delta, tick_);
172 if (settled) runtimeAdvance_ = settled.value();
173 return settled;
174 }
175 if (runtimePaused_) {
176 animation::MontageAdvance paused;
177 paused.previous = paused.current = montage_->time();
178 paused.weight = montage_->weight();
179 return Result<animation::MontageAdvance>::success(std::move(paused), Status::success(StatusCode::NoOp));
180 }
181 double sectionRate = 1.0;
182 auto sectionIndex = montage_->physicalSectionIndex();
183 if (sectionIndex) {
184 const auto found = sectionRates_.find(sectionIndex.value());
185 if (found != sectionRates_.end()) sectionRate = found->second;
186 }
187 auto scaledDelta = Duration::fromSeconds(delta.seconds() * runtimeRate_ * sectionRate);
188 if (!scaledDelta) return Result<animation::MontageAdvance>::failure(scaledDelta.status());
189 Duration remainingDelta = std::move(scaledDelta).takeValue();
190 animation::MontageAdvance combined;
191 combined.previous = montage_->time();
192 bool firstSlice = true;
193 while (true) {
194 Duration remainingTimeline =
195 Duration::fromNanoseconds(runtimeTimeline_->duration.nanoseconds() - montage_->time().nanoseconds());
196 if (runtimeTimeline_->montage.looping && remainingTimeline.isZero()) {
197 auto restarted = restartExecution();
198 if (!restarted) return Result<animation::MontageAdvance>::failure(restarted.status());
199 remainingTimeline = runtimeTimeline_->duration;
200 }
201 const Duration slice =
202 runtimeTimeline_->montage.looping ? std::min(remainingDelta, remainingTimeline) : remainingDelta;
203 tick_ = SimulationTick(tick_.value() + 1);
204 auto actionAdvance = runtime_->advance(executionId_, tick_, slice);
205 if (!actionAdvance) return Result<animation::MontageAdvance>::failure(actionAdvance.status());
206 auto presented = montage_->present(actionAdvance.value(), tick_);
207 if (!presented) return Result<animation::MontageAdvance>::failure(presented.status());
208 auto value = std::move(presented).takeValue();
209 if (firstSlice) {
210 combined = std::move(value);
211 firstSlice = false;
212 } else {
213 combined.current = value.current;
214 combined.sectionId = value.sectionId;
215 combined.events.insert(combined.events.end(), std::make_move_iterator(value.events.begin()),
216 std::make_move_iterator(value.events.end()));
217 combined.activeBlocks = std::move(value.activeBlocks);
218 accumulateRootMotion(combined.rootMotion, value.rootMotion);
219 combined.weight = value.weight;
220 combined.completed = value.completed;
221 }
222 if (!runtimeTimeline_->montage.looping || remainingDelta <= slice) break;
223
224 remainingDelta = Duration::fromNanoseconds(remainingDelta.nanoseconds() - slice.nanoseconds());
225 auto restarted = restartExecution();
226 if (!restarted) return Result<animation::MontageAdvance>::failure(restarted.status());
227 }
228 if (runtimeTimeline_->montage.looping && combined.completed) {
229 auto restarted = restartExecution();
230 if (!restarted) return Result<animation::MontageAdvance>::failure(restarted.status());
231 combined.current = Duration::zero();
232 combined.sectionId.reset();
233 combined.activeBlocks.clear();
234 combined.completed = false;
235 combined.weight = montage_->weight();
236 }
237 runtimeAdvance_ = combined;
238 return Result<animation::MontageAdvance>::success(std::move(combined), Status::success(StatusCode::Pending));
239 }

241 [[nodiscard]] Result<animation::MontageAdvance> jumpRuntime(Duration target) {
242 if (!runtimeTimeline_ || !montage_)
243 return Result<animation::MontageAdvance>::failure(
244 Diagnostic::error(DiagnosticCode::Conflict, "montage runtime has not been started", "runtime"));
245 auto restarted = restartExecution(false);
246 if (!restarted) return Result<animation::MontageAdvance>::failure(restarted.status());
247 auto rebound = montage_->rebindExecution(executionId_);
248 if (!rebound) return Result<animation::MontageAdvance>::failure(rebound.status());
249 tick_ = SimulationTick(tick_.value() + 1);
250 auto advanced = runtime_->advance(executionId_, tick_, target);
251 if (!advanced) return Result<animation::MontageAdvance>::failure(advanced.status());
252 auto jumped = montage_->jumpToTime(executionId_, target, tick_);
253 if (jumped) runtimeAdvance_ = jumped.value();
254 return jumped;
255 }

266 [[nodiscard]] Result<void> syncRuntimeSection(std::size_t index, Duration targetDuration) {
267 if (!runtimeTimeline_)
268 return Result<void>::failure(
269 Diagnostic::error(DiagnosticCode::Conflict, "montage runtime has not been started", "runtime"));
270 if (targetDuration <= Duration::zero())

323 [[nodiscard]] action::ActionDefinition runtimeDefinition() const {
324 action::ActionDefinition definition;
325 definition.id = runtimeTimeline_->actionId;
326 if (runtimeTimeline_->splitTimestamps.size() >= 2) {
327 definition.timing.windup = runtimeTimeline_->splitTimestamps[0];
328 definition.timing.active = Duration::fromNanoseconds(runtimeTimeline_->splitTimestamps[1].nanoseconds() -
329 runtimeTimeline_->splitTimestamps[0].nanoseconds());
330 definition.timing.recover = Duration::fromNanoseconds(runtimeTimeline_->duration.nanoseconds() -
331 runtimeTimeline_->splitTimestamps[1].nanoseconds());
332 } else {
333 definition.timing.active = runtimeTimeline_->duration;
334 }
335 definition.timeline = *runtimeTimeline_;
336 return definition;
337 }

339 [[nodiscard]] Result<void> restartExecution(bool playMontage = true) {
340 auto runtime = std::make_unique<action::ActionRuntime>();
341 auto definition = runtimeDefinition();
342 action::ActionRequest request;
343 request.actionId = definition.id;
344 auto submitted = runtime->submit(std::move(definition), std::move(request));
345 if (!submitted) return Result<void>::failure(submitted.status());
346 if (playMontage) {
347 auto played = montage_->play(submitted.value());
348 if (!played) return Result<void>::failure(played.status());
349 }
350 runtime_ = std::move(runtime);
351 executionId_ = submitted.value();
352 return Result<void>::success(Status::success(StatusCode::Applied));
353 }
