65const action::ActionNotifyState* findState(const action::ActionTimeline& timeline, const std::string& itemId) {
66 for (const auto& track : timeline.tracks) {
67 const auto found = std::find_if(track.states.begin(), track.states.end(),
68 [&](const auto& state) { return state.id.format() == itemId; });
69 if (found != track.states.end()) return &*found;
70 }
71 return nullptr;
72}
