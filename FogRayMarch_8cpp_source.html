16[[nodiscard]] std::optional<FogRayInterval> intersectSlab(float origin, float dir, float minV, float maxV) {
17 if (std::fabs(dir) < 1e-8f) {
18 if (origin < minV || origin > maxV) return std::nullopt;
19 return FogRayInterval{0.f, 1e6f};
20 }
21 float t0 = (minV - origin) / dir;
22 float t1 = (maxV - origin) / dir;
23 if (t0 > t1) std::swap(t0, t1);
24 return FogRayInterval{t0, t1};
25}

27[[nodiscard]] std::optional<FogRayInterval> intersectRayAabb(const glm::vec3& o, const glm::vec3& d,
28 const glm::vec3& mn, const glm::vec3& mx) {
29 auto sx = intersectSlab(o.x, d.x, mn.x, mx.x);
30 auto sy = intersectSlab(o.y, d.y, mn.y, mx.y);
31 auto sz = intersectSlab(o.z, d.z, mn.z, mx.z);
32 if (!sx || !sy || !sz) return std::nullopt;
33 const float tEnter = std::max({sx->tEnter, sy->tEnter, sz->tEnter, 0.f});
34 const float tExit = std::min({sx->tExit, sy->tExit, sz->tExit});
35 if (tExit <= tEnter) return std::nullopt;
36 return FogRayInterval{tEnter, tExit};
37}

41std::optional<FogRayInterval> FogRayMarch::intersectBound(const glm::vec3& origin, const glm::vec3& direction,
42 const FogVolumeBound& bound) {
43 const float len = glm::length(direction);
44 if (!(len > 1e-8f)) return std::nullopt;
45 const glm::vec3 d = direction / len;
46
47 switch (bound.kind) {
48 case FogVolumeBound::Kind::SkyShell: {
49 // Ray vs sphere centered at origin xz / y=0 shell approximation.
50 const glm::vec3 center(origin.x, 0.f, origin.z);
51 const glm::vec3 oc = origin - center;
52 const float b = glm::dot(oc, d);
53 const float c = glm::dot(oc, oc) - bound.skyRadius * bound.skyRadius;
54 const float disc = b * b - c;
55 if (disc < 0.f) return std::nullopt;
56 const float s = std::sqrt(disc);
57 const float t0 = -b - s;
58 const float t1 = -b + s;
59 const float tEnter = std::max(t0, 0.f);
60 if (t1 <= tEnter) return std::nullopt;
61 return FogRayInterval{tEnter, t1};
62 }
63 case FogVolumeBound::Kind::HeightLayer: {
64 auto slab = intersectSlab(origin.y, d.y, bound.heightMin, bound.heightMax);
65 if (!slab) return std::nullopt;
66 const float tEnter = std::max(slab->tEnter, 0.f);
67 if (slab->tExit <= tEnter) return std::nullopt;
68 // Cap far distance so infinite rays remain bounded.
69 return FogRayInterval{tEnter, std::min(slab->tExit, tEnter + 200.f)};
70 }
71 case FogVolumeBound::Kind::Obb: {
72 // Transform ray into OBB local space, then AABB test.
73 const glm::vec3 o = origin - bound.obbCenter;
74 const glm::vec3 localO(glm::dot(o, bound.obbAxisX), glm::dot(o, bound.obbAxisY),
75 glm::dot(o, bound.obbAxisZ));
76 const glm::vec3 localD(glm::dot(d, bound.obbAxisX), glm::dot(d, bound.obbAxisY),
77 glm::dot(d, bound.obbAxisZ));
78 return intersectRayAabb(localO, localD, -bound.obbHalfExtents, bound.obbHalfExtents);
79 }
80 }
81 return std::nullopt;
82}

84Result<FogRayResult> FogRayMarch::integrate(const FogDensityField& field, const FogProfile& profile,
85 const glm::vec3& origin, const glm::vec3& direction,
86 const FogVolumeBound& bound, const glm::vec3& lightDir,
87 const glm::vec3& lightColor, float lightIntensity,
88 const glm::vec3& skyAmbient, const glm::vec3& groundAmbient,
89 int sampleCount, float sceneDepth, const BeerLightCache* beerCache,
90 bool useAnalyticSegments) const {
91 if (sampleCount < 1 || sampleCount > 256) {
92 return Result<FogRayResult>::failure(Diagnostic::error(
93 DiagnosticCode::InvalidArgument, "sampleCount must be in [1,256]", "sampleCount", {}, "graphics.fog"));
94 }
95 const float dirLen = glm::length(direction);
96 if (!(dirLen > 1e-8f)) {
97 return Result<FogRayResult>::failure(Diagnostic::error(
98 DiagnosticCode::InvalidArgument, "direction must be non-zero", "direction", {}, "graphics.fog"));
99 }
100 const glm::vec3 dir = direction / dirLen;
101 auto interval = intersectBound(origin, dir, bound);
102 if (!interval) {
103 FogRayResult empty;
104 return Result<FogRayResult>::success(empty);
105 }
106 if (sceneDepth > 0.f) interval->tExit = std::min(interval->tExit, sceneDepth);
107 if (!interval->valid()) {
108 FogRayResult empty;
109 return Result<FogRayResult>::success(empty);
110 }
111
112 const glm::vec3 L = glm::length(lightDir) > 1e-6f ? glm::normalize(lightDir) : glm::vec3(0.f, 1.f, 0.f);
113 const float cosTheta = glm::dot(-dir, L);
114 const float phase = profile.phase(cosTheta);
115
116 FogRayResult result;
117 float T = 1.f;
118 const float segLen = interval->length() / static_cast<float>(sampleCount);
119 float prevDensity = -1.f;
120
121 for (int i = 0; i < sampleCount && T > 1e-4f; ++i) {
122 const float t = interval->tEnter + (static_cast<float>(i) + 0.5f) * segLen;
123 const glm::vec3 p = origin + dir * t;
124 const float density = field.sampleDensity(p);
125 const float sigmaT = profile.extinctionAt(density);
126 const glm::vec3 sigmaS = profile.scatteringAt(density);
127
128 float stepOptical = sigmaT * segLen;
129 float stepT;
130 if (useAnalyticSegments && prevDensity >= 0.f &&
131 std::fabs(density - prevDensity) < 0.02f * std::max(1.f, prevDensity)) {
132 // Closed-form Beer segment for locally stable media.
133 stepT = std::exp(-stepOptical);
134 } else {
135 stepT = std::exp(-stepOptical);
136 }
137 prevDensity = density;
138
139 float visibility = 1.f;
140 if (beerCache && beerCache->valid()) {
141 visibility = beerCache->sampleTransmittance(field, p);
142 }
143 const glm::vec3 ambient = profile.ambientRadiance(skyAmbient, groundAmbient, density);
144 const glm::vec3 assist = field.sampleLightAssist(p);
145 const glm::vec3 Li = lightColor * lightIntensity * visibility * phase + ambient + assist;
146 // Front-to-back: in-scatter attenuated by current transmittance.
147 const glm::vec3 scatter = sigmaS * Li * ((1.f - stepT) / std::max(sigmaT, 1e-6f));
148 result.inScatter += T * scatter;
149 T *= stepT;
150 result.opticalDepth += stepOptical;
151 ++result.samplesUsed;
152 }
153
154 result.transmittance = T;
155 return Result<FogRayResult>::success(result);
156}

225 const glm::vec3& lightColor, float intensity) {
226 const auto budget = budgetFor(quality_);
227 if (volume.getWidth() != budget.froxelWidth || volume.getHeight() != budget.froxelHeight ||
228 volume.getDepth() != budget.froxelDepth) {
229 volume.resize(budget.froxelWidth, budget.froxelHeight, budget.froxelDepth);
230 volume.setDepthRange(0.1f, std::max(density_.bounds().size().z, 40.f));
231 } else {
232 volume.clear();
233 }
234
235 auto cacheReady = ensureBeerCache(lightDir, lightColor, intensity);
236 if (!cacheReady.ok()) return Result<int>::failure(cacheReady.status());
237
238 auto injected = froxelBridge_.inject(volume, density_, profile_, invViewProj, budget.occupancySkip);
239 if (!injected.ok()) return injected;
240
241 const BeerLightCache* cache = (budget.beerLightCache && beerCache_.valid()) ? &beerCache_ : nullptr;
242 auto integrated =
243 froxelBridge_.integrate(volume, density_, profile_, invViewProj, lightDir, lightColor, intensity, cache);
244 if (!integrated.ok()) return Result<int>::failure(integrated.status());
245
246 if (budget.temporalHistory && frameIndex_ > 1) {
247 // Lightweight self-history: blend with a cleared-then-reintegrated snapshot is
248 // intentionally skipped here; callers may keep an external history volume.
249 (void)budget.historyWeight;
250 }
251 return injected;
252}
