7inline TransformTRS mulTRS(const TransformTRS& parent, const TransformTRS& local) {
8 // world = parent * local (TRS, scale ignored in rotation path for FK positions)
9 TransformTRS out;
10 // Rotate local translation by parent rotation, then add parent translation * scales.
11 const float qx = parent.qx, qy = parent.qy, qz = parent.qz, qw = parent.qw;
12 const float lx = local.px * parent.sx;
13 const float ly = local.py * parent.sy;
14 const float lz = local.pz * parent.sz;
15 // q * v
16 const float ix = qw * lx + qy * lz - qz * ly;
17 const float iy = qw * ly + qz * lx - qx * lz;
18 const float iz = qw * lz + qx * ly - qy * lx;
19 const float iw = -qx * lx - qy * ly - qz * lz;
20 out.px = parent.px + (ix * qw + iw * -qx + iy * -qz - iz * -qy);
21 out.py = parent.py + (iy * qw + iw * -qy + iz * -qx - ix * -qz);
22 out.pz = parent.pz + (iz * qw + iw * -qz + ix * -qy - iy * -qx);
23
24 // Quaternion multiply parent * local
25 out.qw = parent.qw * local.qw - parent.qx * local.qx - parent.qy * local.qy - parent.qz * local.qz;
26 out.qx = parent.qw * local.qx + parent.qx * local.qw + parent.qy * local.qz - parent.qz * local.qy;
27 out.qy = parent.qw * local.qy - parent.qx * local.qz + parent.qy * local.qw + parent.qz * local.qx;
28 out.qz = parent.qw * local.qz + parent.qx * local.qy - parent.qy * local.qx + parent.qz * local.qw;
29 out.normalizeRotation();
30
31 out.sx = parent.sx * local.sx;
32 out.sy = parent.sy * local.sy;
33 out.sz = parent.sz * local.sz;
34 return out;
35}
