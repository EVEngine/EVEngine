63Result<BlackboardPredicate> parsePredicate(const Value& value, const std::string& path) {
64 const auto* object = value.getIf<Value::Object>();
65 if (!object)
66 return Result<BlackboardPredicate>::failure(Diagnostic::error(
67 DiagnosticCode::InvalidArgument, "npc_ai predicate must be an object", path, {}, "npc_ai.squirrel"));
68 BlackboardPredicate predicate;
69 const auto key = object->find("key");
70 const auto op = object->find("op");
71 if (key == object->end() || !key->second.getIf<std::string>())
72 return Result<BlackboardPredicate>::failure(Diagnostic::error(
73 DiagnosticCode::InvalidArgument, "npc_ai predicate key is required", path + ".key", {}, "npc_ai.squirrel"));
74 if (op == object->end() || !op->second.getIf<std::string>())
75 return Result<BlackboardPredicate>::failure(Diagnostic::error(
76 DiagnosticCode::InvalidArgument, "npc_ai predicate op is required", path + ".op", {}, "npc_ai.squirrel"));
77 predicate.key = *key->second.getIf<std::string>();
78 auto parsedOp = parseCompareOp(*op->second.getIf<std::string>());
79 if (!parsedOp) return Result<BlackboardPredicate>::failure(parsedOp.status());
80 predicate.op = parsedOp.value();
81 if (const auto found = object->find("value"); found != object->end()) {
82 auto parsed = parseBlackboardValue(found->second);
83 if (!parsed) return Result<BlackboardPredicate>::failure(parsed.status());
84 predicate.value = std::move(parsed).takeValue();
85 }
86 return Result<BlackboardPredicate>::success(std::move(predicate));
87}

260 if (const auto* integer = priority->second.getIf<std::int64_t>()) {

271 const auto* conditionArray = conditions->second.getIf<Value::Array>();

286 }
287 definition.states.push_back(std::move(state));

292Value blackboardValueToValue(const BlackboardValue& value) {
293 return std::visit(
294 [](const auto& item) -> Value {
295 using T = std::decay_t<decltype(item)>;
296 if constexpr (std::is_same_v<T, bool>)
297 return Value(item);
298 else if constexpr (std::is_same_v<T, std::int64_t>)
299 return Value(item);
300 else if constexpr (std::is_same_v<T, double>)
301 return Value(item);
302 else
303 return Value(item);
304 },
305 value);
306}

455 result["lastTick"] = static_cast<std::int64_t>(snapshot.value().lastTick);
456 Value::Array path;
457 for (const auto& state : snapshot.value().activePath) path.emplace_back(state);
458 result["activePath"] = std::move(path);
459 Value::Object board;
460 for (const auto& [key, value] : snapshot.value().blackboard) board[key] = blackboardValueToValue(value);
461 result["blackboard"] = std::move(board);
462 Value::Array perception;
463 for (const auto& memory : snapshot.value().perception) {
464 Value::Object item;
465 item["subject"] = memory.subject;
466 item["sense"] = memory.sense;
467 item["confidence"] = memory.confidence;
468 item["observedTick"] = static_cast<std::int64_t>(memory.observedTick);
469 item["forgetAfterTicks"] = static_cast<std::int64_t>(memory.forgetAfterTicks);
470 item["payloadJson"] = memory.payloadJson;
471 perception.emplace_back(std::move(item));
472 }
473 result["perception"] = std::move(perception);
474 return Result<Value>::success(Value(std::move(result)));
475 }
