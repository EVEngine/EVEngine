13bool ClassSystem::setClass(RPGActor *actor, const std::string &classId) {
14 if (!actor) return false;
15 const ClassDefinition *def = ClassRegistry::find(classId);
16 if (!def) return false;
17 auto &ci = *actor->classInfo();
18 // 撤销旧职业特征
19 if (!ci.classId.empty()) {
20 TraitSystem::removeBySource(actor, "class:" + ci.classId);
21 }
22 ci.classId = classId;
23 ci.skillsSyncedUpTo = 0;
24 // 施加新职业特征
25 for (const auto &traitId : def->traits) {
26 TraitSystem::apply(actor, traitId, "class:" + classId);
27 }
28 checkLevelSkills(actor);
29 return true;
30}

40int ClassSystem::checkLevelSkills(RPGActor *actor) {
41 if (!actor) return 0;
42 const ClassDefinition *def = ClassRegistry::find(actor->classInfo()->classId);
43 if (!def) return 0;
44 auto &ci = *actor->classInfo();
45 const int level = LevelSystem::getLevel(actor);
46 int learned = 0;
47 for (const auto &ls : def->learnSkills) {
48 if (ls.level <= ci.skillsSyncedUpTo) continue; // 已处理
49 if (ls.level <= level) {
50 if (!SkillSystem::knows(actor, ls.skillId)) {
51 SkillSystem::learn(actor, ls.skillId);
52 ++learned;
53 }
54 }
55 }
56 ci.skillsSyncedUpTo = std::max(ci.skillsSyncedUpTo, level);
57 return learned;
58}
