61void MeshEffectInstance::update(float dtSeconds) {
62 validateDuration(dtSeconds, "dt");
63 if (state_ == MeshEffectState::Stopped || state_ == MeshEffectState::Finished) return;
64 if (stopDuration_ > 0.f) {
65 stopElapsed_ += dtSeconds;
66 elapsed_ += dtSeconds;
67 if (stopElapsed_ >= stopDuration_) state_ = MeshEffectState::Finished;
68 } else {
69 elapsed_ += dtSeconds;
70 const float cycle = playback_.fadeIn + playback_.duration + playback_.fadeOut;
71 if (playback_.loop && cycle > 0.f)
72 elapsed_ = std::fmod(elapsed_, cycle);
73 else if (elapsed_ >= cycle) {
74 elapsed_ = cycle;
75 state_ = MeshEffectState::Finished;
76 }
77 if (state_ != MeshEffectState::Finished) updateTimelineState();
78 }
79 if (style_.hasParam("time")) style_.setFloat("time", elapsed_);
80}

82void MeshEffectInstance::updateTimelineState() noexcept {
83 if (playback_.fadeIn > 0.f && elapsed_ < playback_.fadeIn) {
84 state_ = MeshEffectState::FadingIn;
85 } else if (elapsed_ < playback_.fadeIn + playback_.duration) {
86 state_ = MeshEffectState::Active;
87 } else if (playback_.fadeOut > 0.f) {
88 state_ = MeshEffectState::FadingOut;
89 } else {
90 state_ = playback_.loop ? MeshEffectState::Active : MeshEffectState::Finished;
91 }
92}

94float MeshEffectInstance::intensity() const noexcept {
95 if (state_ == MeshEffectState::Stopped || state_ == MeshEffectState::Finished) return 0.f;
96 if (stopDuration_ > 0.f)
97 return stopStartIntensity_ * std::clamp(1.f - stopElapsed_ / stopDuration_, 0.f, 1.f);
98 if (playback_.fadeIn > 0.f && elapsed_ < playback_.fadeIn)
99 return std::clamp(elapsed_ / playback_.fadeIn, 0.f, 1.f);
100 const float fadeOutStart = playback_.fadeIn + playback_.duration;
101 if (playback_.fadeOut > 0.f && elapsed_ >= fadeOutStart)
102 return std::clamp(1.f - (elapsed_ - fadeOutStart) / playback_.fadeOut, 0.f, 1.f);
103 return 1.f;
104}
