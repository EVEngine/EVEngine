74inline FaceDir rotateFaceY(FaceDir dir, int quarterTurns) {
75 if (dir == FaceDir::PosY || dir == FaceDir::NegY) return dir;
76 static const FaceDir cycle[4] = {FaceDir::PosX, FaceDir::NegZ, FaceDir::NegX, FaceDir::PosZ};
