37Result<std::uint64_t> parseEntityId(const Value& value, std::size_t index) {
38 if (!value.isString())
39 return Result<std::uint64_t>::failure(Diagnostic::error(
40 DiagnosticCode::ParseError, "script component snapshot entity id must be a decimal string",
41 std::to_string(index), {}, "ecs.snapshot"));
42 const std::string& text = value.asString();
43 if (text.empty())
44 return Result<std::uint64_t>::failure(Diagnostic::error(DiagnosticCode::ParseError,
45 "script component snapshot entity id must not be empty",
46 std::to_string(index), {}, "ecs.snapshot"));
47 std::uint64_t parsed = 0;
48 for (const char digit : text) {
49 if (digit < '0' || digit > '9')
50 return Result<std::uint64_t>::failure(Diagnostic::error(
51 DiagnosticCode::ParseError, "script component snapshot entity id must be decimal digits only", text, {},
52 "ecs.snapshot"));
53 const std::uint64_t next = parsed * 10u + static_cast<std::uint64_t>(digit - '0');
54 if (next < parsed)
55 return Result<std::uint64_t>::failure(Diagnostic::error(
56 DiagnosticCode::ParseError, "script component snapshot entity id overflows", text, {}, "ecs.snapshot"));
57 parsed = next;
58 }
59 return Result<std::uint64_t>::success(parsed);
60}

74EVENGINE_API_FOUNDATION Result<SnapshotEnvelope> snapshotScriptComponents(PersistentId instanceId, Revision revision, SimulationTick tick,
75 const SnapshotHashProvider& hashProvider) {
76 Value::Array components;
77 for (const ScriptComponentStoreEntry& entry : scriptComponentStoreEntries()) {
78 Value::Object item;
79 item.emplace(kEntityField, Value::string(entityIdText(entry.key.entityId)));
80 item.emplace(kNameField, Value::string(entry.key.name));
81 item.emplace(kVersionField, Value::integer(static_cast<std::int64_t>(entry.key.version.value())));
82 item.emplace(kFieldsField, entry.values);
83 components.push_back(Value::object(std::move(item)));

86 Value::Object payload;
87 payload.emplace(kComponentsField, Value::array(std::move(components)));

89 SchemaVersion(kSchemaVersion), instanceId, revision, tick,
90 Value::object(std::move(payload)), hashProvider);

93EVENGINE_API_FOUNDATION Result<void> restoreScriptComponents(const SnapshotEnvelope& snapshot, const SnapshotHashProvider& hashProvider) {
94 if (snapshot.type != kScriptComponentSnapshotType || snapshot.schema != scriptComponentSnapshotSchema())
95 return snapshotFailure(DiagnosticCode::InvalidArgument,
96 "snapshot does not belong to the script component store");
97 if (snapshot.schemaVersion.value() != kSchemaVersion)
98 return snapshotFailure(DiagnosticCode::UnknownVersion,
99 "script component snapshot version is not the current one", snapshot.schema.format());
100 auto verified = verifySnapshotEnvelope(snapshot, hashProvider);
101 if (!verified.ok()) return verified;
102
103 const Value* components = snapshot.payload.find(kComponentsField);
104 if (components == nullptr || !components->isArray())
105 return snapshotFailure(DiagnosticCode::ParseError, "script component snapshot has no components array");
106
107 // 先全部校验进候选表，**不碰**现有数据：恢复失败必须什么都不改（仓库规则：restore 不许留下
108 // 部分变更的可观测状态）。
109 std::map<ScriptComponentStoreKey, Value> candidate;
110 for (std::size_t index = 0; index < components->arraySize(); ++index) {
111 const Value& item = components->at(index);
112 if (!item.isObject())
113 return snapshotFailure(DiagnosticCode::ParseError, "script component snapshot entry must be an object",
114 std::to_string(index));
115 const Value* entity = item.find(kEntityField);
116 const Value* name = item.find(kNameField);
117 const Value* version = item.find(kVersionField);
118 const Value* fields = item.find(kFieldsField);
119 if (entity == nullptr || name == nullptr || version == nullptr || fields == nullptr)
120 return snapshotFailure(DiagnosticCode::ParseError,
121 "script component snapshot entry is missing a required field",
122 std::to_string(index));
123 if (!name->isString() || name->asString().empty())
124 return snapshotFailure(DiagnosticCode::ParseError, "script component snapshot entry has an empty name",
125 std::to_string(index));
126 if (!version->isInt64() || version->asInt() <= 0)
127 return snapshotFailure(DiagnosticCode::ParseError, "script component snapshot entry has an invalid version",
128 name->asString());
129 if (!fields->isObject())
130 return snapshotFailure(DiagnosticCode::ParseError,
131 "script component snapshot entry fields must be an object", name->asString());
132
133 auto entityId = parseEntityId(*entity, index);
134 if (!entityId.ok()) return Result<void>::failure(entityId.status());
135
136 const std::string componentName = name->asString();
137 const SchemaVersion componentVersion(static_cast<std::uint64_t>(version->asInt()));
138 const ScriptComponentRecord* record = findScriptComponent(componentName, componentVersion);
139 if (record == nullptr)
140 return snapshotFailure(DiagnosticCode::NotFound,
141 "snapshot describes a script component this build does not know",
142 componentName + "@" + std::to_string(componentVersion.value()));
143 // 未知字段策略：显式拒绝。字段集变过的旧载荷必须走迁移链，不能被静默收下——否则存储里会
144 // 出现没人声明的字段，序列化/编辑器都会看见它。
145 for (const std::string& key : fields->keys()) {
146 bool declared = false;
147 for (const std::string& field : record->fields) {
148 if (field == key) {
149 declared = true;
150 break;
151 }
152 }
153 if (!declared)
154 return snapshotFailure(DiagnosticCode::Unsupported,
155 "script component snapshot field is not in the current schema",
156 componentName + "." + key);
157 }
158
159 const ScriptComponentStoreKey key{entityId.value(), componentName, componentVersion};
160 if (candidate.find(key) != candidate.end())
161 return snapshotFailure(DiagnosticCode::Conflict, "script component snapshot has a duplicate entry",
162 componentName);
163 candidate.emplace(key, *fields);
164 }
165
166 std::vector<ScriptComponentStoreEntry> entries;
167 entries.reserve(candidate.size());
168 for (auto& [key, values] : candidate) entries.push_back(ScriptComponentStoreEntry{key, std::move(values)});
169 replaceScriptComponentValues(std::move(entries));
170 return Result<void>::success(Status::success(StatusCode::Applied));
171}

186EVENGINE_API_FOUNDATION Result<std::vector<ScriptComponentInstanceState>> scriptComponentInstancesOf(const ssq::Object& entityInstance) {
187 if (entityInstance.getType() != ssq::Type::INSTANCE)
188 return Result<std::vector<ScriptComponentInstanceState>>::failure(
189 Diagnostic::error(DiagnosticCode::InvalidArgument, "script component inspection needs an entity instance",
190 {}, {}, "ecs.snapshot"));
191 HSQUIRRELVM vm = entityInstance.getHandle();
192 if (vm == nullptr)
193 return Result<std::vector<ScriptComponentInstanceState>>::failure(
194 Diagnostic::error(DiagnosticCode::InvalidArgument, "script component inspection needs a live script VM", {},
195 {}, "ecs.snapshot"));
196
197 // 实例上的 `_eid` 是实体身份，`_comps`（都是 `eve.Entity` 的声明槽位）是"这个实体有哪些
198 // 组件"的事实来源；每个组件实例上的 `_componentName`/`_componentVersion` 是它登记时的身份
199 // （§9.64 的控制槽位）。两步都用公开 sq API 读取，不碰 Squirrel 内部结构。
200 std::vector<std::pair<std::string, SchemaVersion>> identities;
201 std::uint64_t entityId = 0;
202 const SQInteger top = sq_gettop(vm);
203 sq_pushobject(vm, entityInstance.getRaw());
204
205 sq_pushstring(vm, "_eid", -1);
206 if (SQ_SUCCEEDED(sq_get(vm, -2))) {
207 SQInteger raw = 0;
208 if (SQ_SUCCEEDED(sq_getinteger(vm, -1, &raw)) && raw > 0) entityId = static_cast<std::uint64_t>(raw);
209 }
210 sq_settop(vm, top);
211
212 sq_pushobject(vm, entityInstance.getRaw());
213 sq_pushstring(vm, "_comps", -1);
214 if (SQ_FAILED(sq_get(vm, -2)) || sq_gettype(vm, -1) != OT_TABLE) {
215 sq_settop(vm, top);
216 return Result<std::vector<ScriptComponentInstanceState>>::success({});
217 }
218 sq_pushnull(vm); // 迭代游标
219 while (SQ_SUCCEEDED(sq_next(vm, -2))) {
220 // 栈：instance, comps, cursor, key, value
221 if (sq_gettype(vm, -1) == OT_INSTANCE) {
222 std::string name;
223 std::int64_t version = 0;
224 const auto readComponentSlot = [&](const char* slot, bool asText) -> bool {
225 const SQInteger inner = sq_gettop(vm);
226 sq_push(vm, -1); // 组件实例
227 sq_pushstring(vm, slot, -1);
228 const bool found = SQ_SUCCEEDED(sq_get(vm, -2));
229 bool ok = false;
230 if (found && asText) {
231 const SQChar* raw = nullptr;
232 ok = SQ_SUCCEEDED(sq_getstring(vm, -1, &raw)) && raw != nullptr;
233 if (ok) name = raw;
234 } else if (found) {
235 SQInteger raw = 0;
236 ok = SQ_SUCCEEDED(sq_getinteger(vm, -1, &raw));
237 if (ok) version = static_cast<std::int64_t>(raw);
238 }
239 sq_settop(vm, inner);
240 return ok;
241 };
242 if (readComponentSlot("_componentName", true) && !name.empty() &&
243 readComponentSlot("_componentVersion", false) && version > 0)
244 identities.emplace_back(name, SchemaVersion(static_cast<std::uint64_t>(version)));
245 }
246 sq_pop(vm, 2); // key, value
247 }
248 sq_settop(vm, top);
249
250 std::sort(identities.begin(), identities.end());
251 identities.erase(std::unique(identities.begin(), identities.end()), identities.end());
252
253 std::vector<ScriptComponentInstanceState> out;
254 out.reserve(identities.size());
255 for (const auto& [name, version] : identities) {
256 ScriptComponentInstanceState state;
257 state.entityId = entityId;
258 state.name = name;
259 state.version = version;
260 const ScriptComponentRecord* record = findScriptComponent(name, version);
261 state.generation = record != nullptr ? record->generation : Generation(0);
262
263 const Value* stored = findScriptComponentValues(ScriptComponentStoreKey{entityId, name, version});
264 // 按声明的顺序收集（inspector 需要稳定顺序）；存储里缺的字段给 null，让"有组件但缺数据"
265 // 在工具里可见。
266 const std::vector<std::string> fields = record != nullptr ? record->fields : std::vector<std::string>{};
267 for (const std::string& field : fields) {
268 const Value* value = stored != nullptr ? stored->find(field) : nullptr;
269 state.fields.emplace(field, value != nullptr ? *value : Value::null());
270 }
271 if (record == nullptr && stored != nullptr) {
272 if (const auto* object = stored->getIf<Value::Object>()) state.fields = *object;
273 }
274 out.push_back(std::move(state));
275 }
276 return Result<std::vector<ScriptComponentInstanceState>>::success(std::move(out));
277}

279EVENGINE_API_FOUNDATION Result<std::string> scriptComponentStatesJson() {
280 // 存储本身就是按 (实体, 名字, 版本) 排序的 `std::map`，所以按实体分组只需要顺序扫一遍。
281 Value::Array entities;
282 Value::Array components;
283 std::uint64_t currentId = 0;
284 bool open = false;
285 const auto flush = [&]() {
286 if (!open) return;
287 entities.push_back(Value::object(
288 {{"id", Value::string(entityIdText(currentId))}, {"components", Value::array(std::move(components))}}));
289 components = Value::Array{};
290 open = false;
291 };
292 for (const ScriptComponentStoreEntry& entry : scriptComponentStoreEntries()) {
293 if (!open || entry.key.entityId != currentId) {
294 flush();
295 currentId = entry.key.entityId;
296 open = true;
297 }
298 const ScriptComponentRecord* record = findScriptComponent(entry.key.name, entry.key.version);
299 components.push_back(Value::object(
300 {{"name", Value::string(entry.key.name)},
301 {"version", Value::integer(static_cast<std::int64_t>(entry.key.version.value()))},
302 {"generation",
303 Value::integer(static_cast<std::int64_t>(record != nullptr ? record->generation.value() : 0))},
304 {"fields", entry.values}}));
305 }
306 flush();
307
308 return Value::object({{"entities", Value::array(std::move(entities))}}).toJson();
309}
