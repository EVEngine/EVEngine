44Result<void> GraphDocument::connect(GraphEdgeRecord edge, const GraphConnectionDecision& decision) {
45 if (!decision.allowed) {
46 auto diagnostics = decision.diagnostics;
47 if (diagnostics.empty())
48 diagnostics.push_back(ruleDiagnostic(eve::DiagnosticCode::PreconditionViolation,
49 RuleId("editing.graph.connection-rejected"),
50 DiagnosticSeverity::Error, "Connection was rejected"));
51 return Result<void>::failure(eve::Status(Status::Rejected, std::move(diagnostics)));
52 }
53 if (edge.id.empty() || edges_.contains(edge.id) || !findPin(edge.from) || !findPin(edge.to))
54 return graphError<void>(Status::Rejected, "editing.graph.invalid-edge",
55 "Graph edge id/pins are invalid or duplicated");
56 edges_.emplace(edge.id, std::move(edge));
57 ++revision_;
58 return eve::editing::applied<void>();
