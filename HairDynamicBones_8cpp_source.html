22Result<int> addChain(DynamicBoneSolver &solver, const ChainDesc &desc) {
23 if (desc.rootBone.empty() || desc.tipBone.empty())
24 return invalidInt("hair chain requires rootBone and tipBone", "hair.chain");
25 if (desc.stiffness < 0.f || desc.stiffness > 1.f || desc.damping < 0.f || desc.damping > 1.f ||
26 desc.inertia < 0.f || desc.inertia > 1.f || desc.radius < 0.f || desc.iterations < 1)
27 return invalidInt("hair chain parameters out of range", "hair.chain");
28
29 const int chain =
30 solver.addChainByName(desc.rootBone, desc.tipBone, desc.stiffness, desc.damping,
31 desc.inertia, desc.gravityScale, desc.radius, desc.iterations);
32 if (chain < 0)
33 return invalidInt("hair chain bone path cannot be resolved",
34 desc.rootBone + "->" + desc.tipBone);
35
36 solver.setChainSelfCollision(chain, desc.selfCollision);
37 if (desc.endLength > 0.f) solver.setChainEndLength(chain, desc.endLength);
38 return Result<int>::success(chain);
39}

41Result<void> setupChains(DynamicBoneSolver &solver, const std::vector<ChainDesc> &chains,
42 bool clearFirst) {
43 if (clearFirst) solver.clearChains();
44 for (size_t i = 0; i < chains.size(); ++i) {
45 auto added = addChain(solver, chains[i]);
46 if (!added) {
47 const Diagnostic *diag = added.status().primaryDiagnostic();
48 return invalidVoid(diag ? std::string(diag->message()) : added.status().describe(),
49 "hair.chains[" + std::to_string(i) + "]");
50 }
51 }
52 return Result<void>::success();
53}

55Result<int> addHeadCollider(DynamicBoneSolver &solver, const std::string &boneName, float radius,
56 float offsetX, float offsetY, float offsetZ) {
57 if (boneName.empty()) return invalidInt("head collider bone name is empty", "hair.headBone");
58 if (!(radius > 0.f))
59 return invalidInt("head collider radius must be positive", "hair.headRadius");
60 AnimSkeleton *skeleton = solver.getSkeleton();
61 if (!skeleton) return invalidInt("dynamic bone solver has no skeleton", "hair.skeleton");
62 const int bone = skeleton->findBone(boneName);
63 if (bone < 0) return invalidInt("head collider bone not found", boneName);
64 solver.addBoneColliderSphere(bone, offsetX, offsetY, offsetZ, radius);
65 return Result<int>::success(solver.getColliderCount());
66}
