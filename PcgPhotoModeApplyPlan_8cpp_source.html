89Result<void> PcgPhotoModeApplyPlan::execute(IPhotoModeApplySink& sink) {
90 appliedCount_=0; rollbackComplete_=true;
91 for(size_t i=0;i<commands_.size();++i){
92 auto result=sink.applyPhotoModeAssignment({commands_[i].field,commands_[i].domain,commands_[i].after});
93 if(!result.ok()){
94 bool restored=true;
95 for(size_t j=i;j>0;--j){
96 auto rollback=sink.applyPhotoModeAssignment({commands_[j-1].field,commands_[j-1].domain,commands_[j-1].before});
97 if(!rollback.ok())restored=false;
98 }
99 rollbackComplete_=restored;
100 return fail<void>(restored?"photo-mode assignment failed; applied assignments were rolled back":
101 "photo-mode assignment failed and rollback was incomplete");
102 }
103 ++appliedCount_;
104 }
105 ++revision_; return Result<void>::success();
