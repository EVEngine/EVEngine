29Evaluation evaluate(const OceanSurfaceSettings& s, glm::dvec2 q, double time) {
30 Evaluation e;
31 e.position = {q.x, s.seaLevel, q.y};
32 e.velocity = {s.current.x, 0.0, s.current.y};
33 for (const auto& wave : s.waves) {
34 const double k = tau / wave.wavelength;
35 const double omega = std::sqrt(s.gravity * k);
36 const auto d = wave.direction;
37 const double phase = std::remainder(k * glm::dot(d, q - s.current * time) - omega * time + wave.phase, tau);
38 const double sn = std::sin(phase), cs = std::cos(phase);
39 const double horizontal = wave.choppiness * wave.amplitude;
40 e.position += glm::dvec3(horizontal * d.x * cs, wave.amplitude * sn, horizontal * d.y * cs);
41 e.tangentX += glm::dvec3(-horizontal * k * d.x * d.x * sn, wave.amplitude * k * d.x * cs,
42 -horizontal * k * d.x * d.y * sn);
43 e.tangentZ += glm::dvec3(-horizontal * k * d.x * d.y * sn, wave.amplitude * k * d.y * cs,
44 -horizontal * k * d.y * d.y * sn);
45 // Material derivative follows the uniform current, cancelling its phase advection.
46 e.velocity +=
47 glm::dvec3(horizontal * omega * d.x * sn, -wave.amplitude * omega * cs, horizontal * omega * d.y * sn);
48 }
49 return e;
50}

53Result<OceanSurface> OceanSurface::create(const OceanSurfaceSettings& settings) {
54 if (!std::isfinite(settings.seaLevel) || std::abs(settings.seaLevel) > 1e6 || !std::isfinite(settings.gravity) ||
55 settings.gravity <= 0.0 || settings.gravity > 100.0 || !finite(settings.current) ||
56 glm::length(settings.current) > 100.0 || settings.waves.size() > 64)
57 return invalid<OceanSurface>();
58 auto normalized = settings;
59 double steepness = 0.0;
60 for (auto& w : normalized.waves) {
61 if (!finite(w.direction) || glm::length(w.direction) < 1e-8 || glm::length(w.direction) > 1e6 ||
62 !std::isfinite(w.amplitude) || w.amplitude < 0.0 || w.amplitude > 100.0 || !std::isfinite(w.wavelength) ||
63 w.wavelength < 0.05 || w.wavelength > 1e6 || !std::isfinite(w.phase) || std::abs(w.phase) > 1e9 ||
64 !std::isfinite(w.choppiness) || w.choppiness < 0.0 || w.choppiness > 1.0)
65 return invalid<OceanSurface>();
66 w.direction = glm::normalize(w.direction);
67 steepness += w.choppiness * w.amplitude * tau / w.wavelength;
68 }
69 if (steepness > 0.75) return invalid<OceanSurface>();
70 return Result<OceanSurface>::success(OceanSurface(std::move(normalized)));
71}

78Result<OceanSurfaceSample> OceanSurface::sample(glm::dvec2 worldXZ, double time) const {
79 if (!validQuery(worldXZ, time)) return invalid<OceanSurfaceSample>();
80 auto q = worldXZ;
81 // The admitted displacement is a contraction (Lipschitz <= .75). This
82 // safeguarded fixed-point solve cannot jump to a different folded branch.
83 for (int i = 0; i < 96; ++i) {
84 const auto e = evaluate(settings_, q, time);
85 const auto residual = glm::dvec2(e.position.x, e.position.z) - worldXZ;
86 q -= residual;
87 if (glm::length(residual) < 1e-7) break;
88 }
89 const auto e = evaluate(settings_, q, time);
90 OceanSurfaceSample sample;
91 sample.height = e.position.y;
92 sample.normal = glm::normalize(glm::cross(e.tangentZ, e.tangentX));
93 sample.velocity = e.velocity;
94 sample.parameterPosition = q;
95 sample.time = time;
96 sample.horizontalJacobian = e.tangentX.x * e.tangentZ.z - e.tangentX.z * e.tangentZ.x;
97 return Result<OceanSurfaceSample>::success(sample);
98}

100Result<OceanCellForce> evaluateOceanCellForce(const OceanBuoyancyCell& cell, const OceanSurfaceSample& water,
101 double density, double gravity) {
102 if (!finite(cell.center) || !finite(cell.pointVelocity) || glm::length(cell.center) > 1e9 ||
103 glm::length(cell.pointVelocity) > 1e4 || !std::isfinite(cell.radius) || cell.radius <= 0.0 ||
104 cell.radius > 1e4 || !std::isfinite(cell.volume) || cell.volume <= 0.0 || cell.volume > 1e9 ||
105 !std::isfinite(cell.dragCoefficient) || cell.dragCoefficient < 0.0 || cell.dragCoefficient > 100.0 ||
106 !std::isfinite(water.height) || std::abs(water.height) > 1e9 || !finite(water.velocity) ||
107 glm::length(water.velocity) > 1e4 || !std::isfinite(density) || density <= 0.0 || density > 1e5 ||
108 !std::isfinite(gravity) || gravity <= 0.0 || gravity > 100.0)
109 return invalid<OceanCellForce>();
110 const double depth = std::clamp(water.height - cell.center.y + cell.radius, 0.0, 2.0 * cell.radius);
111 const double h = depth / cell.radius;
112 const double fraction = h * h * (3.0 - h) * 0.25;
113 OceanCellForce result;
114 result.position = cell.center;
115 result.submergedVolume = cell.volume * fraction;
116 const auto relative = cell.pointVelocity - water.velocity;
117 const double area = std::numbers::pi * cell.radius * cell.radius * fraction;
118 result.force = -0.5 * density * cell.dragCoefficient * area * glm::length(relative) * relative;
119 result.force.y += density * gravity * result.submergedVolume;
120 return Result<OceanCellForce>::success(result);
121}
