112SpriteSheet *Animation::newSpriteSheetFromSequence(graphics::Graphics *gfx,
113 const std::string &pattern, int first,
114 int last, int columns) {
115 if (!gfx) throw Exception("Animation.newSpriteSheetFromSequence: gfx is null");
116 const size_t marker = pattern.find("{n}");
117 if (marker == std::string::npos)
118 throw Exception("Animation.newSpriteSheetFromSequence: pattern must contain '{n}'");
119 if (first < 0 || last < first)
120 throw Exception("Animation.newSpriteSheetFromSequence: expected 0 <= first <= last");
121
122 const int count = last - first + 1;
123 if (columns <= 0) columns = static_cast<int>(std::ceil(std::sqrt(float(count))));
124 if (columns <= 0)
125 throw Exception("Animation.newSpriteSheetFromSequence: columns must be > 0");
126 const std::string cacheKey = pattern + "#" + std::to_string(first) + ":" +
127 std::to_string(last) + ":" + std::to_string(columns);
128 auto cached = spriteSequenceCache_.find(cacheKey);
129 if (cached != spriteSequenceCache_.end()) return cached->second->clone();
130 const int rows = (count + columns - 1) / columns;
131
132 image::Image *images = image::Image::create();
133 std::vector<image::ImageData *> frames;
134 struct Bounds { int x = 0, y = 0, w = 1, h = 1; };
135 std::vector<Bounds> bounds;
136 frames.reserve(static_cast<size_t>(count));
137 int frameW = 0, frameH = 0;
138 int packedW = 1, packedH = 1;
139 for (int n = first; n <= last; ++n) {
140 std::string path = pattern;
141 path.replace(marker, 3, std::to_string(n));
142 image::ImageData *frame = images->newImageDataFromFile(path);
143 if (!frame) throw Exception("Animation.newSpriteSheetFromSequence: failed '%s'", path.c_str());
144 if (frame->getFormat() != "RGBA8")
145 throw Exception("Animation.newSpriteSheetFromSequence: '%s' must be RGBA8", path.c_str());
146 if (frames.empty()) {
147 frameW = frame->getWidth();
148 frameH = frame->getHeight();
149 } else if (frame->getWidth() != frameW || frame->getHeight() != frameH) {
150 throw Exception("Animation.newSpriteSheetFromSequence: frame size mismatch at '%s'",
151 path.c_str());
152 }
153 frames.push_back(frame);
154 int minX = frameW, minY = frameH, maxX = -1, maxY = -1;
155 for (int py = 0; py < frameH; ++py) {
156 for (int px = 0; px < frameW; ++px) {
157 if (frame->getPixel(px, py).a <= 1.f / 255.f) continue;
158 minX = std::min(minX, px); minY = std::min(minY, py);
159 maxX = std::max(maxX, px); maxY = std::max(maxY, py);
160 }
161 }
162 Bounds b;
163 if (maxX >= minX && maxY >= minY)
164 b = {minX, minY, maxX - minX + 1, maxY - minY + 1};
165 bounds.push_back(b);
166 packedW = std::max(packedW, b.w);
167 packedH = std::max(packedH, b.h);
168 }
169
170 constexpr int padding = 1;
171 const int cellW = packedW + padding * 2;
172 const int cellH = packedH + padding * 2;
173 std::unique_ptr<image::ImageData> atlas(
174 images->newImageData(columns * cellW, rows * cellH, "RGBA8"));
175 auto *sheet = new SpriteSheet();
176 try {
177 for (int i = 0; i < count; ++i) {
178 const int cellX = (i % columns) * cellW;
179 const int cellY = (i / columns) * cellH;
180 const int x = cellX + padding;
181 const int y = cellY + padding;
182 image::ImageData *frame = frames[static_cast<size_t>(i)];
183 const Bounds &b = bounds[size_t(i)];
184 atlas->paste(frame, x, y, b.x, b.y, b.w, b.h);
185 atlas->paste(frame, x, cellY, b.x, b.y, b.w, 1);
186 atlas->paste(frame, x, y + b.h, b.x, b.y + b.h - 1, b.w, 1);
187 atlas->paste(frame, cellX, y, b.x, b.y, 1, b.h);
188 atlas->paste(frame, x + b.w, y, b.x + b.w - 1, b.y, 1, b.h);
189 atlas->paste(frame, cellX, cellY, b.x, b.y, 1, 1);
190 atlas->paste(frame, x + b.w, cellY, b.x + b.w - 1, b.y, 1, 1);
191 atlas->paste(frame, cellX, y + b.h, b.x, b.y + b.h - 1, 1, 1);
192 atlas->paste(frame, x + b.w, y + b.h, b.x + b.w - 1, b.y + b.h - 1, 1, 1);
193 sheet->addFrameTrimmed(std::to_string(first + i), x, y, b.w, b.h,
194 frameW, frameH, b.x, b.y);
195 }
196 sheet->setTexture(gfx->newTextureFromImageData(atlas.get()));
197 spriteSequenceCache_[cacheKey].reset(sheet->clone());
198 } catch (...) {
199 delete sheet;
200 throw;
201 }
202 return sheet;
203}

216SpriteSheet *Animation::newSpriteSheetFromAtlasJson(graphics::Graphics *gfx,
217 const std::string &texturePath,
218 const std::string &jsonPath) {
219 if (!gfx) throw Exception("Animation.newSpriteSheetFromAtlasJson: gfx is null");
220 auto *fs = ModuleManager::getInstance<filesystem::Filesystem>("Filesystem");
221 if (!fs) fs = filesystem::Filesystem::create();
222 std::unique_ptr<filesystem::FileData> file(fs->read(jsonPath));
223 std::string text(static_cast<const char *>(file->getData()), size_t(file->getSize()));
224 auto *dm = data::DataModule::create();
225 std::string error;
226 std::unique_ptr<data::JsonDocument> doc(dm->decodeJson(text, &error));
227 if (!doc || !doc->isObject()) throw Exception("Atlas JSON: %s", error.c_str());
228 auto root = doc->object();
229 if (!root->has("frames")) throw Exception("Atlas JSON: missing frames");
230 auto frames = root->getObject("frames");
231 if (!frames) throw Exception("Atlas JSON: frames must be an object");
232 auto *sheet = new SpriteSheet();
233 try {
234 for (const auto &name : frames->getNames()) {
235 auto item = frames->getObject(name);
236 if (item->optValue<bool>("rotated", false))
237 throw Exception("Atlas JSON: rotated frames are not supported");
238 auto rect = item->getObject("frame");
239 auto src = item->getObject("spriteSourceSize");
240 auto size = item->getObject("sourceSize");
241 const int w = rect->getValue<int>("w"), h = rect->getValue<int>("h");
242 sheet->addFrameTrimmed(name, rect->getValue<int>("x"), rect->getValue<int>("y"), w, h,
243 size ? size->getValue<int>("w") : w,
244 size ? size->getValue<int>("h") : h,
245 src ? src->getValue<int>("x") : 0,
246 src ? src->getValue<int>("y") : 0);
247 }
248 sheet->setTexture(gfx->newTextureFromFile(texturePath));
249 } catch (...) { delete sheet; throw; }
250 return sheet;
251}

438eve::Result<void> Animation::advance(const eve::SimulationStep &step) {
439 if (step.delta.nanoseconds() < 0)
440 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
441 "animation simulation delta must be non-negative"));
442 if (hasLastTick_ && step.tick <= lastTick_)
443 return eve::Result<void>::failure(eve::Diagnostic::error(
444 eve::DiagnosticCode::Conflict, "animation simulation tick must advance monotonically"));
445
446 const float dt = static_cast<float>(step.delta.seconds());
447 if (!std::isfinite(dt))
448 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
449 "animation simulation delta is outside float range"));
450
451 // Preflight every live child before mutating any of them. A child may also
452 // be driven directly by a caller, so the host must preserve all-or-none
453 // behavior when one child has already consumed this tick.
454 for (Tween *t : tweens_) {
455 if (t && t->isActive() && t->hasCurrentTick() && step.tick <= t->currentTick())
456 return eve::Result<void>::failure(eve::Diagnostic::error(
457 eve::DiagnosticCode::Conflict, "animation Tween child already consumed this tick"));
458 }
459 for (SpriteAnim *a : spriteAnims_) {
460 if (a && a->isPlaying() && a->hasCurrentTick() && step.tick <= a->currentTick())
461 return eve::Result<void>::failure(eve::Diagnostic::error(
462 eve::DiagnosticCode::Conflict, "animation SpriteAnim child already consumed this tick"));
463 }
464 for (SpineAnim *a : spineAnims_) {
465 if (a && a->isPlaying() && a->hasCurrentTick() && step.tick <= a->currentTick())
466 return eve::Result<void>::failure(eve::Diagnostic::error(
467 eve::DiagnosticCode::Conflict, "animation SpineAnim child already consumed this tick"));
468 }
469
470 // Copy pointer lists: destructors during update must not invalidate iteration.
471 std::vector<Tween *> tweenSnap = tweens_;
472 for (Tween *t : tweenSnap) {
473 if (!t) continue;
474 if (t->isActive()) {
475 auto result = t->advance(step);
476 if (!result) return eve::Result<void>::failure(result.status());
477 }
478 }
479 std::vector<SpriteAnim *> spriteSnap = spriteAnims_;
480 for (SpriteAnim *a : spriteSnap) {
481 if (a && a->isPlaying()) {
482 auto result = a->advance(step);
483 if (!result) return eve::Result<void>::failure(result.status());
484 }
485 }
486 std::vector<SpineAnim *> spineSnap = spineAnims_;
487 for (SpineAnim *a : spineSnap) {
488 if (a && a->isPlaying()) {
489 auto result = a->advance(step);
490 if (!result) return eve::Result<void>::failure(result.status());
491 }
492 }
493
494 lastTick_ = step.tick;
495 hasLastTick_ = true;
496 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
497}

499void Animation::update(float dt) {
500 EV_PROFILE_MODULE("animation", "Animation::update");
501 auto duration = eve::Duration::fromSeconds(dt);
502 if (!duration) {
503 duration.ignore("legacy animation update received an invalid duration");
504 return;
505 }
506 auto nextTick = hasLastTick_ ? lastTick_.incremented() : std::optional<eve::SimulationTick>(eve::SimulationTick(1));
507 if (!nextTick) {
508 eve::Result<void>::failure(
509 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation, "animation simulation tick overflow"))
510 .ignore("legacy animation update tick overflow");
511 return;
512 }
513 advance({*nextTick, std::move(duration).takeValue()}).ignore("legacy animation update facade");
514}
