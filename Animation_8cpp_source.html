33void copyFloatArray(ssq::Array arr, std::vector<float> &out) {

261void Animation::update(float dt) {
262 // Copy pointer lists: destructors during update must not invalidate iteration.
263 std::vector<Tween *> tweenSnap = tweens_;
264 for (Tween *t : tweenSnap) {
265 if (!t) continue;
266 if (t->isActive()) t->update(dt);
267 }
268 std::vector<SpriteAnim *> spriteSnap = spriteAnims_;
269 for (SpriteAnim *a : spriteSnap) {
270 if (a && a->isPlaying()) a->update(dt);
271 }
272 std::vector<SpineAnim *> spineSnap = spineAnims_;
273 for (SpineAnim *a : spineSnap) {
274 if (a && a->isPlaying()) a->update(dt);
275 }
276}

295 Tween *t = *it;
