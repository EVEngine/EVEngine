15Result<DamageNumberKind> parseKind(const std::string& kind) {
16 if (kind == "damage") return Result<DamageNumberKind>::success(DamageNumberKind::Damage);
17 if (kind == "critical") return Result<DamageNumberKind>::success(DamageNumberKind::Critical);
18 if (kind == "heal") return Result<DamageNumberKind>::success(DamageNumberKind::Heal);
19 if (kind == "miss") return Result<DamageNumberKind>::success(DamageNumberKind::Miss);
20 return Result<DamageNumberKind>::failure(
21 Diagnostic::error(DiagnosticCode::InvalidArgument, "Unknown damage-number kind", "kind"));
22}
