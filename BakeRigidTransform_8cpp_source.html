13 for (float value : {placement.x, placement.y, placement.z, placement.rotationX, placement.rotationY,

27Result<RigidTrackPose> BakeRigidTransform::apply(RigidTrackPose pose) const {
28 for (float value : {pose.x, pose.y, pose.z, pose.qx, pose.qy, pose.qz, pose.qw})
29 if (!std::isfinite(value)) return Result<RigidTrackPose>::failure(transformError("nonfinite pose"));
30 // Double intermediates avoid overflow before the final representability check.
31 const double qx = placement_.rotationX, qy = placement_.rotationY, qz = placement_.rotationZ,
32 qw = placement_.rotationW;
33 const double norm = qx * qx + qy * qy + qz * qz + qw * qw;
34 const double tx = 2 * (qy * pose.z - qz * pose.y) / norm;
35 const double ty = 2 * (qz * pose.x - qx * pose.z) / norm;
36 const double tz = 2 * (qx * pose.y - qy * pose.x) / norm;
37 const double px = pose.x + qw * tx + qy * tz - qz * ty + placement_.x;
38 const double py = pose.y + qw * ty + qz * tx - qx * tz + placement_.y;
39 const double pz = pose.z + qw * tz + qx * ty - qy * tx + placement_.z;
40 for (double value : {px, py, pz})
41 if (!std::isfinite(value) || std::abs(value) > std::numeric_limits<float>::max())
42 return Result<RigidTrackPose>::failure(transformError("placed sample position overflow"));
43 const double bx = pose.qx, by = pose.qy, bz = pose.qz, bw = pose.qw;
44 const double rx = qw * bx + qx * bw + qy * bz - qz * by;
45 const double ry = qw * by - qx * bz + qy * bw + qz * bx;
46 const double rz = qw * bz + qx * by - qy * bx + qz * bw;
47 const double rw = qw * bw - qx * bx - qy * by - qz * bz;
48 const double length = std::sqrt(rx * rx + ry * ry + rz * rz + rw * rw);
49 if (length == 0) return Result<RigidTrackPose>::failure(transformError("zero pose quaternion"));
50 pose.x = static_cast<float>(px);
51 pose.y = static_cast<float>(py);
52 pose.z = static_cast<float>(pz);
53 pose.qx = static_cast<float>(rx / length);
54 pose.qy = static_cast<float>(ry / length);
55 pose.qz = static_cast<float>(rz / length);
56 pose.qw = static_cast<float>(rw / length);
57 return Result<RigidTrackPose>::success(pose);
58}
