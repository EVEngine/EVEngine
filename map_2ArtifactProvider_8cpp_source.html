69eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>> MapArtifactProvider::prepare(
70 const eve::artifact::PublicationView& publication) {
71 if (failPrepare_)
72 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(
73 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "map artifact prepare failure was injected"));
74 if (publication.id.isNil() || publication.buildKey.empty())
75 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(eve::Diagnostic::error(
76 eve::DiagnosticCode::InvalidArgument, "map publication requires identity and build key"));
77 if (find(publication.id) != nullptr)
78 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(
79 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "map artifact identity is already published"));
80
81 const eve::artifact::PartView* topology = nullptr;
82 for (const auto& part : publication.parts) {
83 if (part.kind == eve::artifact::PartKind::Grid && (part.role == "topology" || topology == nullptr))
84 topology = &part;
85 }
86 if (!topology || topology->width <= 0 || topology->height <= 0 ||
87 topology->cells.size() !=
88 static_cast<std::size_t>(topology->width) * static_cast<std::size_t>(topology->height))
89 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(eve::Diagnostic::error(
90 eve::DiagnosticCode::PreconditionViolation, "map publication requires a complete grid topology"));
91 if (nextIndex_ == MapArtifactHandle::invalidIndex)
92 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(
93 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "map artifact handle index exhausted"));
94
95 MapArtifactRecord record;
96 record.id = publication.id;
97 record.buildKey = std::string(publication.buildKey);
98 record.handle = MapArtifactHandle(nextIndex_, 1);
99 record.width = topology->width;
100 record.height = topology->height;
101 try {
102 record.cells.assign(topology->cells.begin(), topology->cells.end());
103 records_.reserve(records_.size() + 1);
104 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::success(
105 std::make_unique<MapArtifactStage>(*this, std::move(record)));
106 } catch (...) {
107 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(
108 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "map artifact prepare allocation failed"));
109 }
110}

137eve::Result<eve::Value> MapArtifactProvider::snapshotState() const {
138 try {
139 eve::Value::Array records;
140 records.reserve(records_.size());
141 for (const auto& record : records_) {
142 eve::Value::Array cells;
143 cells.reserve(record.cells.size());
144 for (const auto value : record.cells) cells.emplace_back(std::int64_t(value));
145 eve::Value::Object item;
146 item.emplace("artifactId", eve::Value(record.id.format()));
147 item.emplace("buildKey", eve::Value(record.buildKey));
148 item.emplace("handle", eve::Value(std::to_string(record.handle.packed())));
149 item.emplace("width", eve::Value(std::int64_t(record.width)));
150 item.emplace("height", eve::Value(std::int64_t(record.height)));
151 item.emplace("cells", eve::Value(std::move(cells)));
152 records.emplace_back(std::move(item));
153 }
154 eve::Value::Object state;
155 state.emplace("provider", eve::Value("map.cpu-registry"));
156 state.emplace("version", eve::Value(std::int64_t(1)));
157 state.emplace("records", eve::Value(std::move(records)));
158 return eve::Result<eve::Value>::success(eve::Value(std::move(state)));
159 } catch (...) {
160 return eve::Result<eve::Value>::failure(
161 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "map artifact snapshot allocation failed"));
162 }

165eve::Result<void> MapArtifactProvider::restoreState(const eve::Value& state) {
166 if (!records_.empty())
167 return eve::Result<void>::failure(
168 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "map artifact restore requires an empty registry"));
169 const auto* object = state.getIf<eve::Value::Object>();
170 const auto* provider = object ? member(*object, "provider") : nullptr;
171 const auto* providerName = provider ? provider->getIf<std::string>() : nullptr;
172 const auto* encodedRecords = object ? member(*object, "records") : nullptr;
173 const auto* records = encodedRecords ? encodedRecords->getIf<eve::Value::Array>() : nullptr;
174 if (!object || !providerName || *providerName != "map.cpu-registry" || !hasSupportedVersion(*object) || !records)
175 return eve::Result<void>::failure(
176 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "invalid map artifact provider state"));
177
178 std::vector<MapArtifactRecord> candidate;
179 std::unordered_set<eve::PersistentId> identities;
180 std::uint32_t next = 0;
181 try {
182 candidate.reserve(records->size());
183 for (const eve::Value& encoded : *records) {
184 const auto* item = encoded.getIf<eve::Value::Object>();
185 std::string idText;
186 std::string buildKey;
187 std::uint64_t packed = 0;
188 std::int64_t width = 0;
189 std::int64_t height = 0;
190 const auto* encodedCells = item ? member(*item, "cells") : nullptr;
191 const auto* cells = encodedCells ? encodedCells->getIf<eve::Value::Array>() : nullptr;
192 if (!item || !readString(*item, "artifactId", idText) || !readString(*item, "buildKey", buildKey) ||
193 !readU64(*item, "handle", packed) || !readI64(*item, "width", width) ||
194 !readI64(*item, "height", height) || width <= 0 || height <= 0 || !cells)
195 return eve::Result<void>::failure(
196 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "invalid map artifact record"));
197 const auto parsedId = eve::PersistentId::parse(idText);
198 const auto handle = MapArtifactHandle::fromPacked(packed);
199 if (!parsedId || parsedId->isNil() || handle.isInvalid() ||
200 width > std::numeric_limits<std::int32_t>::max() || height > std::numeric_limits<std::int32_t>::max() ||
201 !identities.emplace(*parsedId).second ||
202 static_cast<std::uint64_t>(width) * static_cast<std::uint64_t>(height) != cells->size())
203 return eve::Result<void>::failure(eve::Diagnostic::error(
204 eve::DiagnosticCode::ParseError, "invalid map artifact dimensions or identity"));
205 MapArtifactRecord record;
206 record.id = *parsedId;
207 record.buildKey = std::move(buildKey);
208 record.handle = handle;
209 record.width = static_cast<std::int32_t>(width);
210 record.height = static_cast<std::int32_t>(height);
211 record.cells.reserve(cells->size());
212 for (const eve::Value& value : *cells) {
213 const auto* cell = value.getIf<std::int64_t>();
214 if (!cell || *cell < 0 || *cell > std::numeric_limits<std::uint32_t>::max())
215 return eve::Result<void>::failure(
216 eve::Diagnostic::error(eve::DiagnosticCode::ParseError, "invalid map topology cell"));
217 record.cells.push_back(static_cast<std::uint32_t>(*cell));
218 }
219 next = std::max(next, handle.index() + 1u);
220 candidate.push_back(std::move(record));
221 }
222 } catch (...) {
223 return eve::Result<void>::failure(
224 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "map artifact restore allocation failed"));
225 }
226 records_.swap(candidate);
227 nextIndex_ = next;
228 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
229}
