29 const UnitySourceAsset& source;
30 const AssetImportLimits& limits;

37 Result<Value::Array> block(bool nested) {
38 if (++depth > 64)
39 return Result<Value::Array>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
40 "TVE preset condition nesting exceeds budget",
41 source.path, {}, "asset.import.unity"));
42 Value::Array result;
43 while (at < lines.size()) {
44 auto line = trim(lines[at++]);
45 if (line.empty() || line.starts_with("//") || line.front() == '*') continue;
46 if (line == "}") {
47 --depth;
48 if (!nested)
49 return Result<Value::Array>::failure(Diagnostic::error(DiagnosticCode::ParseError,
50 "unexpected TVE preset closing brace",
51 source.path, {}, "asset.import.unity"));
52 return Result<Value::Array>::success(std::move(result));
53 }
54 if (++statements > limits.maximumAssets * 64ull)
55 return Result<Value::Array>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
56 "TVE preset statement count exceeds budget",
57 source.path, {}, "asset.import.unity"));
58 if (line.starts_with("if ") || line.starts_with("f ")) {
59 const bool recoveredTypo = line.starts_with("f ");
60 if (recoveredTypo) ++recoveredConditionTypos;
61 auto tokens = words(trim(line.substr(recoveredTypo ? 2 : 3)));
62 if (tokens.empty() || at >= lines.size() || trim(lines[at++]) != "{")
63 return Result<Value::Array>::failure(
64 Diagnostic::error(DiagnosticCode::ParseError, "TVE preset condition requires a following block",
65 source.path, {}, "asset.import.unity"));
66 auto body = block(true);
67 if (!body) return body;
68 auto predicate = tokens.front().asString();
69 const bool negated = !predicate.empty() && predicate.front() == '!';
70 if (negated) predicate.erase(predicate.begin());
71 tokens.erase(tokens.begin());
72 result.emplace_back(Value::Object{{"kind", Value("condition")},
73 {"predicate", Value(predicate)},
74 {"negated", Value(negated)},
75 {"arguments", Value(std::move(tokens))},
76 {"statements", Value(std::move(body).takeValue())}});
77 continue;
78 }
79 if (line == "{")
80 return Result<Value::Array>::failure(Diagnostic::error(DiagnosticCode::ParseError,
81 "unexpected TVE preset opening brace",
82 source.path, {}, "asset.import.unity"));
83 const auto split = line.find_first_of(" \t");
84 const auto domain = line.substr(0, split);
85 auto remainder = split == std::string_view::npos ? std::string_view{} : trim(line.substr(split));
86 const auto operationSplit = remainder.find_first_of(" \t");
87 const auto operation = remainder.substr(0, operationSplit);
88 auto arguments = operationSplit == std::string_view::npos ? Value::Array{}
89 : words(trim(remainder.substr(operationSplit)));
90 result.emplace_back(Value::Object{{"kind", Value("command")},
91 {"domain", Value(std::string(domain))},
92 {"operation", Value(std::string(operation))},
93 {"arguments", Value(std::move(arguments))}});
94 }
95 --depth;
96 if (nested)
97 return Result<Value::Array>::failure(Diagnostic::error(DiagnosticCode::ParseError,
98 "unterminated TVE preset condition block",
99 source.path, {}, "asset.import.unity"));
100 return Result<Value::Array>::success(std::move(result));
101 }
