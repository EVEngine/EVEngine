38eve::Result<TransactionReceipt> AtomicResourcePayment::execute(const TransactionContext& context,
39 eve::resource::IResourceAccount& account,
40 const eve::resource::CostSpec& cost,
41 std::span<ITransactionParticipant*> participants) {
42 if (!cost.isValid())
43 return failure<TransactionReceipt>(eve::DiagnosticCode::InvalidArgument,
44 "resource payment requires a validated non-empty CostSpec", "cost");
45
46 ResourceDebitParticipant debit(account, cost, "resource-payment");
47 std::vector<ITransactionParticipant*> all;
48 all.reserve(participants.size() + 1u);
49 for (auto* participant : participants) all.push_back(participant);
50 // The debit is deliberately last. Domain participants are therefore
51 // prepared first, and a failed debit compensates any committed domain
52 // participant without exposing a partially paid operation.
53 all.push_back(&debit);
54 Coordinator coordinator;
55 return coordinator.execute(context, std::span<ITransactionParticipant*>(all));
56}
