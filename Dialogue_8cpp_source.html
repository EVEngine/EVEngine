306bool Dialogue::parseCondition(const DataValue &v, Condition &out, std::string &error) {
307 if (v.kind() == DataValue::Kind::Array) {
308 out.kind = Condition::Kind::All;
309 for (size_t i = 0; i < v.arraySize(); ++i) {
310 const auto &item = v.at(i);
311 Condition child;
312 if (!parseCondition(item, child, error)) return false;
313 out.children.push_back(std::move(child));
314 }
315 return true;
316 }
317 if (v.kind() != DataValue::Kind::Object) {
318 error = "condition must be a table";
319 return false;
320 }
321 if (const DataValue *script = v.find("script")) {
322 if (script->kind() != DataValue::Kind::String) {
323 error = "script condition name must be a string";
324 return false;
325 }
326 out.kind = Condition::Kind::Script;
327 out.script = script->asString();
328 return true;
329 }
330 if (const DataValue *all = v.find("all")) {
331 if (all->kind() != DataValue::Kind::Array) {
332 error = "all must be an array";
333 return false;
334 }
335 out.kind = Condition::Kind::All;
336 for (size_t i = 0; i < all->arraySize(); ++i) {
337 const auto &item = all->at(i);
338 Condition child;
339 if (!parseCondition(item, child, error)) return false;
340 out.children.push_back(std::move(child));
341 }
342 return true;
343 }
344 if (const DataValue *any = v.find("any")) {
345 if (any->kind() != DataValue::Kind::Array) {
346 error = "any must be an array";
347 return false;
348 }
349 out.kind = Condition::Kind::Any;
350 for (size_t i = 0; i < any->arraySize(); ++i) {
351 const auto &item = any->at(i);
352 Condition child;
353 if (!parseCondition(item, child, error)) return false;
354 out.children.push_back(std::move(child));
355 }
356 return true;
357 }
358 if (const DataValue *notc = v.find("not")) {
359 out.kind = Condition::Kind::Not;
360 out.children.emplace_back();
361 return parseCondition(*notc, out.children.back(), error);
362 }
363 if (const DataValue *var = v.find("var")) {
364 const DataValue *op = v.find("op");
365 if (var->kind() != DataValue::Kind::String || !op || op->kind() != DataValue::Kind::String) {
366 error = "var/op condition requires string var and op";
367 return false;
368 }
369 static const char *kOps[] = {"eq", "ne", "gt", "ge", "lt", "le", "has", "missing"};
370 const std::string opStr = op->asString();
371 if (std::find(std::begin(kOps), std::end(kOps), opStr) == std::end(kOps)) {
372 error = "unknown condition op: " + opStr;
373 return false;
374 }
375 out.kind = Condition::Kind::Cmp;
376 out.var = var->asString();
377 out.op = opStr;
378 out.value = VarValue::string("");
379 if (const DataValue *val = v.find("value")) {
380 switch (val->kind()) {
381 case DataValue::Kind::Int: out.value = VarValue::integer(val->asInt()); break;
382 case DataValue::Kind::Float: out.value = VarValue::number(val->asDouble()); break;
383 case DataValue::Kind::Bool: out.value = VarValue::boolean(val->asBool()); break;
384 case DataValue::Kind::String: out.value = VarValue::string(val->asString()); break;
385 default:
386 break;
387 }
388 }
389 return true;
390 }
391 error = "unrecognized condition table";
392 return false;
393}

395bool Dialogue::compareEq(const VarValue &a, const VarValue &b) const {
396 if (a.isNumeric() && b.isNumeric()) {
397 const double x = a.type() == VarValue::Type::Int ? double(a.asInt()) : a.asDouble();
398 const double y = b.type() == VarValue::Type::Int ? double(b.asInt()) : b.asDouble();
399 return x == y;
400 }

402 if (a.type() == VarValue::Type::String && b.type() == VarValue::Type::String) return a.asString() == b.asString();
403 return false;

406bool Dialogue::compareOrder(const std::string &op, const VarValue &a, const VarValue &b) const {
407 if (!a.isNumeric() || !b.isNumeric()) return false;
408 const double x = a.type() == VarValue::Type::Int ? double(a.asInt()) : a.asDouble();
409 const double y = b.type() == VarValue::Type::Int ? double(b.asInt()) : b.asDouble();
410 if (op == "gt") return x > y;
411 if (op == "ge") return x >= y;
412 if (op == "lt") return x < y;
413 if (op == "le") return x <= y;
414 return false;
415}

417bool Dialogue::evalConditionInternal(const Condition &c,
418 const std::unordered_map<std::string, VarValue> &merged,
419 const std::unordered_map<std::string, VarValue> &params,
420 const std::string &lineId) const {
421 switch (c.kind) {
422 case Condition::Kind::Always:
423 return true;
424 case Condition::Kind::Cmp: {
425 const auto it = merged.find(c.var);
426 const bool present = it != merged.end();
427 if (c.op == "has") return present;
428 if (c.op == "missing") return !present;
429 if (!present) return false;
430 if (c.op == "eq") return compareEq(it->second, c.value);
431 if (c.op == "ne") return !compareEq(it->second, c.value);
432 return compareOrder(c.op, it->second, c.value);
433 }
434 case Condition::Kind::All:
435 for (const auto &child : c.children)
436 if (!evalConditionInternal(child, merged, params, lineId)) return false;
437 return true;
438 case Condition::Kind::Any:
439 for (const auto &child : c.children)
440 if (evalConditionInternal(child, merged, params, lineId)) return true;
441 return false;
442 case Condition::Kind::Not:
443 return c.children.empty() ||
444 !evalConditionInternal(c.children.front(), merged, params, lineId);
445 case Condition::Kind::Script:
446 return evalScriptPredicate(c.script, merged, params, lineId);
447 }
448 return false;
449}

458bool Dialogue::evalScriptPredicate(const std::string &name,
459 const std::unordered_map<std::string, VarValue> &merged,
460 const std::unordered_map<std::string, VarValue> &params,
461 const std::string &lineId) const {
462 if (!vm_) return false;
463 const auto it = predicates_.find(name);
464 if (it == predicates_.end()) return false;
465
466 const SQInteger top = sq_gettop(vm_);
467 sq_pushobject(vm_, it->second); // closure first; sq_call expects fn below its args
468 sq_newtable(vm_); // ctx
469
470 sq_pushstring(vm_, "vars", -1);
471 pushVarTable(vm_, merged);
472 sq_newslot(vm_, -3, SQFalse); // fn at -4, key at -2, var table at -1 -> slot into ctx(-3)
473
474 sq_pushstring(vm_, "params", -1);
475 pushVarTable(vm_, params);
476 sq_newslot(vm_, -3, SQFalse);
477
478 sq_pushstring(vm_, "lineId", -1);
479 sq_pushstring(vm_, lineId.c_str(), lineId.size());
480 sq_newslot(vm_, -3, SQFalse);
481
482 if (SQ_FAILED(sq_call(vm_, 1, SQTrue, SQTrue))) {
483 sq_settop(vm_, top);
484 return false;
485 }
486 SQBool result = SQFalse;
487 if (SQ_FAILED(sq_getbool(vm_, -1, &result))) {
488 sq_settop(vm_, top);
489 return false;
490 }
491 sq_settop(vm_, top);
492 return result != 0;
493}

599 continue;
600 }
601 if (const DataValue *nr = kv.second.find("noRepeat")) {
602 if (nr->kind() == DataValue::Kind::Int)
603 pool.noRepeat = int(nr->asInt());
604 else if (nr->kind() == DataValue::Kind::Float)
605 pool.noRepeat = int(nr->asDouble());
606 if (pool.noRepeat < 0) pool.noRepeat = 0;
607 }
608 const DataValue *lines = kv.second.find("lines");
609 if (!lines || lines->kind() != DataValue::Kind::Array) {
610 lastPoolsError_ = "pool '" + kv.first + "': missing lines array";
611 continue;
612 }
613 int lineIndex = 1;
614 for (size_t i = 0; i < lines->arraySize(); ++i) {
615 const auto &lv = lines->at(i);
616 Line line;
617 std::string error;
618 if (!parseLineData(pool.id, lv, lineIndex, line, error)) {
619 if (lastPoolsError_.empty()) lastPoolsError_ = error;
620 continue;
621 }
622 pool.lines.push_back(std::move(line));
623 ++lineIndex;
624 }
625 if (Pool *existing = findPool(pool.id)) *existing = std::move(pool);
626 else pools_.push_back(std::move(pool));
627 ++registered;
628 }
629 return registered;
630}

719std::string Dialogue::interpolate(const std::string &tpl,
720 const std::unordered_map<std::string, VarValue> &vars) const {
721 std::string out;
722 out.reserve(tpl.size());
723 for (size_t i = 0; i < tpl.size();) {
724 if (tpl[i] == '{') {
725 const size_t close = tpl.find('}', i + 1);
726 if (close != std::string::npos) {
727 const std::string name = tpl.substr(i + 1, close - i - 1);
728 const auto it = vars.find(name);
729 if (it != vars.end()) {
730 out += it->second.toString();
731 i = close + 1;
732 continue;
733 }
734 }
735 }
736 out += tpl[i++];
737 }
738 return out;
739}

766
767std::string Dialogue::pickLineWithParams(
768 const std::string &poolId, const std::unordered_map<std::string, VarValue> &params) {
769 Pool *pool = findPool(poolId);
770 if (!pool) return "";
771 const auto merged = mergedVars(params);
772
773 auto collect = [&](std::vector<size_t> &out, bool allowRecent) {
774 for (size_t i = 0; i < pool->lines.size(); ++i) {
775 const Line &line = pool->lines[i];
776 if (line.weight <= 0.0) continue;
777 if (!evalConditionInternal(line.when, merged, params, line.id)) continue;
778 if (!allowRecent && pool->noRepeat > 0 &&
779 std::find(pool->recent.begin(), pool->recent.end(), i) != pool->recent.end())
780 continue;
781 out.push_back(i);
782 }
783 };
784
785 std::vector<size_t> candidates;
786 collect(candidates, false);
787 if (candidates.empty()) collect(candidates, true); // noRepeat must not deadlock
788 if (candidates.empty()) return "";
789
790 double total = 0.0;
791 for (const size_t i : candidates) total += pool->lines[i].weight;
792 double roll = nextUnit() * total;
793 size_t picked = candidates.front();
794 double acc = 0.0;
795 for (const size_t i : candidates) {
796 acc += pool->lines[i].weight;
797 if (roll < acc) {
798 picked = i;
799 break;
800 }
801 }
802 if (pool->noRepeat > 0) {
803 pool->recent.push_back(picked);
804 if (pool->recent.size() > size_t(pool->noRepeat)) pool->recent.erase(pool->recent.begin());
805 }
806 return pool->lines[picked].id;
807}

1150void Dialogue::updateLipSync(float dt) {
1151 if (dt < 0.f) dt = 0.f;
1152 if (lipSyncEnabled_ && phase_ == Phase::Typing && !speakerId_.empty()) {
1153 lipSyncTime_ += dt;
1154 // Simple mouth envelope while characters appear (no audio dependency).
1155 const float wave = std::fabs(std::sin(lipSyncTime_ * 14.f));
1156 lipSyncValue_ = lipSyncAmplitude_ * (0.25f + 0.75f * wave);
1157 } else {

1159 lipSyncValue_ *= std::max(0.f, 1.f - dt * 8.f);
1160 if (lipSyncValue_ < 0.01f) lipSyncValue_ = 0.f;
1161 }
1162 applyLipSyncToSpeaker();
