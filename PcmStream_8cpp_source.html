45Result<std::unique_ptr<PcmStream>> PcmStream::create(int sampleRate, int channels) {
46 if (sampleRate < 8000 || sampleRate > 192000 || (channels != 1 && channels != 2))
47 return Result<std::unique_ptr<PcmStream>>::failure(
48 Diagnostic::error(DiagnosticCode::InvalidArgument, "PCM requires 8000..192000 Hz and one or two channels"));
49 auto state = std::make_shared<PcmStreamState>();
50 state->rate = sampleRate;
51 state->channels = channels;
52 alGetError();
53 alGenSources(1, &state->source);
54 alGenBuffers(blockCount, state->buffers.data());
55 alSourcei(state->source, AL_SOURCE_RELATIVE, AL_TRUE);
56 alSourcef(state->source, AL_ROLLOFF_FACTOR, 0.f);
57 auto result = alResult();
58 if (!result.ok()) {
59 closeState(state);
60 return Result<std::unique_ptr<PcmStream>>::failure(result.status());
61 }
62 return Result<std::unique_ptr<PcmStream>>::success(std::unique_ptr<PcmStream>(new PcmStream(std::move(state))));
63}

65void PcmStream::closeState(const std::shared_ptr<PcmStreamState>& state) {
66 if (state->source) {
67 alSourceStop(state->source);
68 alSourcei(state->source, AL_BUFFER, 0);
69 alDeleteSources(1, &state->source);
70 state->source = 0;
71 }
72 for (auto& buffer : state->buffers) {
73 if (buffer) alDeleteBuffers(1, &buffer);
74 buffer = 0;
75 }
76 state->frames.fill(0);
77}

81void PcmStream::pumpState(const std::shared_ptr<PcmStreamState>& state) {
82 if (!state->source) return;
83 ALint processed = 0;
84 alGetSourcei(state->source, AL_BUFFERS_PROCESSED, &processed);
85 while (processed-- > 0) {
86 ALuint buffer = 0;
87 alSourceUnqueueBuffers(state->source, 1, &buffer);
88 const auto found = std::find(state->buffers.begin(), state->buffers.end(), buffer);
89 if (found != state->buffers.end()) {
90 const auto index = static_cast<size_t>(found - state->buffers.begin());
91 state->played += state->frames[index];
92 state->frames[index] = 0;
93 }
94 }
95 if (state->wantsPlay && !state->paused) {
96 ALint playback = 0, queued = 0;
97 alGetSourcei(state->source, AL_SOURCE_STATE, &playback);
98 alGetSourcei(state->source, AL_BUFFERS_QUEUED, &queued);
99 if (queued > 0 && playback != AL_PLAYING) alSourcePlay(state->source);
100 }
101}

103Result<int> PcmStream::submit(std::span<const float> samples) {
104 auto& state = *state_;
105 if (!state.source) return Result<int>::failure(closedDiagnostic());
106 if (state.ended)
107 return Result<int>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation, "PCM input has ended"));
108 if (samples.empty() || samples.size() % state.channels != 0 ||
109 samples.size() > static_cast<size_t>(maxBlockFrames * state.channels) ||
110 std::any_of(samples.begin(), samples.end(), [](float v) { return !std::isfinite(v) || v < -1.f || v > 1.f; }))
111 return Result<int>::failure(
112 Diagnostic::error(DiagnosticCode::InvalidArgument,
113 "PCM samples must be finite, normalized, channel-aligned and bounded to 4096 frames"));
114 pumpState(state_);
115 const auto free = std::find(state.frames.begin(), state.frames.end(), 0);
116 if (free == state.frames.end())
117 return Result<int>::failure(
118 Diagnostic::error(DiagnosticCode::Conflict, "PCM queue is full; retry this block later"));
119 const auto index = static_cast<size_t>(free - state.frames.begin());
120 std::vector<ALshort> pcm(samples.size());
121 std::transform(samples.begin(), samples.end(), pcm.begin(),
122 [](float v) { return static_cast<ALshort>(std::lround(v < 0.f ? v * 32768.f : v * 32767.f)); });
123 alGetError();
124 alBufferData(state.buffers[index], state.channels == 1 ? AL_FORMAT_MONO16 : AL_FORMAT_STEREO16, pcm.data(),
125 static_cast<ALsizei>(pcm.size() * sizeof(ALshort)), state.rate);
126 auto upload = alResult();
127 if (!upload.ok()) return Result<int>::failure(upload.status());
128 alSourceQueueBuffers(state.source, 1, &state.buffers[index]);
129 auto queued = alResult();
130 if (!queued.ok()) return Result<int>::failure(queued.status());
131 const int accepted = static_cast<int>(samples.size() / state.channels);
132 *free = accepted;
133 pumpState(state_);
134 return Result<int>::success(accepted);
135}

175PcmStreamInfo PcmStream::getInfo() const {
176 PcmStreamInfo info;
177 info.sampleRate = state_->rate;
178 info.channels = state_->channels;
179 info.playedFrames = state_->played;
180 for (int frames : state_->frames) {
181 info.queuedFrames += frames;
182 if (!frames) ++info.availableBlocks;
183 }
184 if (!state_->source) {
185 info.phase = PcmStreamPhase::Closed;
186 info.availableBlocks = 0;
187 } else if (state_->ended && !info.queuedFrames)
188 info.phase = PcmStreamPhase::Finished;
189 else if (state_->paused)
190 info.phase = PcmStreamPhase::Paused;
191 else if (!state_->wantsPlay)
192 info.phase = PcmStreamPhase::Idle;
193 else if (!info.queuedFrames)
194 info.phase = PcmStreamPhase::Starved;
195 else
196 info.phase = state_->ended ? PcmStreamPhase::Draining : PcmStreamPhase::Playing;
197 return info;
198}

200void exposePcmStream(ssq::Table& table, ssq::Class& audioClass) {
201 auto cls = table.addClass<PcmStream>(
202 "PcmStream",
203 std::function<PcmStream*()>([]() -> PcmStream* { throw std::runtime_error("Use audio.newPcmStream"); }), false);
204 const auto vm = table.getHandle();
205 cls.addFunc("submit", [vm](PcmStream* self, ssq::Array samples) {
206 if (samples.size() > PcmStream::maxBlockFrames * 2)
207 return eve::script::projectResult(vm,
208 Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
209 "PCM block exceeds sample limit")),
210 [](int n) { return Value(n); });
211 std::vector<float> values;
212 values.reserve(samples.size());
213 try {
214 for (size_t i = 0; i < samples.size(); ++i) values.push_back(samples.get<float>(i));
215 } catch (const ssq::Exception&) {
216 return eve::script::projectResult(
217 vm,
218 Result<int>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "PCM samples must be numbers")),
219 [](int n) { return Value(n); });
220 }
221 return eve::script::projectResult(vm, self->submit(values), [](int n) { return Value(n); });
222 });
223 cls.addFunc("play", [vm](PcmStream* self) { return eve::script::projectResult(vm, self->play()); });
224 cls.addFunc("pause", [vm](PcmStream* self) { return eve::script::projectResult(vm, self->pause()); });
225 cls.addFunc("finishInput", [vm](PcmStream* self) { return eve::script::projectResult(vm, self->finishInput()); });
226 cls.addFunc("reset", [vm](PcmStream* self) { return eve::script::projectResult(vm, self->reset()); });
227 cls.addFunc("close", &PcmStream::close);
228 cls.addFunc("getInfo", [vm](PcmStream* self) {
229 const auto info = self->getInfo();
230 constexpr std::array names = {"idle", "playing", "paused", "starved", "draining", "finished", "closed"};
231 ssq::Table snapshot(vm);
232 snapshot.set("phase", std::string(names[static_cast<size_t>(info.phase)]));
233 snapshot.set("sampleRate", info.sampleRate);
234 snapshot.set("channels", info.channels);
235 snapshot.set("availableBlocks", info.availableBlocks);
236 snapshot.set("maxBlockFrames", PcmStream::maxBlockFrames);
237 snapshot.set("queuedFrames", info.queuedFrames);
238 snapshot.set("playedFrames", static_cast<std::int64_t>(info.playedFrames));
239 return snapshot;
240 });
241 audioClass.addFunc("newPcmStream", [vm](Audio* self, int rate, int channels) {
242 auto result = self->newPcmStream(rate, channels);
243 if (!result.ok()) return eve::script::projectStatusResult(vm, result.status());
244 auto owned = eve::script::makeOwnedSquirrelInstance(vm, std::move(result).takeValue());
245 if (!owned.ok()) return eve::script::projectStatusResult(vm, owned.status());
246 return eve::script::projectStatusResult(vm, Status::success(), std::move(owned).takeValue());
247 });
248}
