47[[nodiscard]] float neighbourhoodTopY(const HexMap& map, HexCoordinates coordinates) noexcept {
48 float top = cellTopY(map, coordinates);
49 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
50 HexCoordinates neighbour{};
51 if (!map.getNeighbor(coordinates, static_cast<HexDirection>(i), neighbour)) continue;
52 const float candidate = cellTopY(map, neighbour);
53 if (candidate > top) top = candidate;
54 }
55 return top;
56}

80void appendFogColumn(const HexMap& map, HexCoordinates coordinates, float shade, HexMeshData& out) {
81 const HexNoise& noise = map.noise();
82 const HexVec3 ground = map.cellGroundPosition(coordinates);
83 const float top = neighbourhoodTopY(map, coordinates) + kFogCapLift;
84 const float bottom = HexMetrics::elevationY(HexMetrics::kMinElevation) - kFogSkirtDepth;
85
86 HexVec3 ring[kHexDirectionCount];
87 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
88 HexVec3 corner = noise.perturb(ground + HexMetrics::corner(static_cast<HexDirection>(i)));
89 corner.y = top;
90 ring[i] = corner;
91 }
92 HexVec3 center = noise.perturb(HexVec3{ground.x, 0.f, ground.z});
93 center.y = top;
94
95 for (std::int32_t i = 0; i < kHexDirectionCount; ++i) {
96 const std::int32_t next = (i + 1) % kHexDirectionCount;
97 emitTriangle(out, center, ring[i], ring[next], shade);
98
99 HexVec3 topNear = ring[i];
100 HexVec3 topFar = ring[next];
101 HexVec3 botFar = ring[next];
102 HexVec3 botNear = ring[i];
103 botNear.y = bottom;
104 botFar.y = bottom;
105 const auto i0 = static_cast<std::uint32_t>(out.vertexCount());
106 out.addVertex(topNear, shade, 0.f);
107 out.addVertex(topFar, shade, 0.f);
108 out.addVertex(botFar, shade, 0.f);
109 out.addVertex(botNear, shade, 0.f);
110 out.addQuad(i0, i0 + 1u, i0 + 2u, i0 + 3u);
111 }
112}

116void buildFogMesh(const HexMap& map, const HexVisibility& visibility, std::int32_t chunkIndex, HexMeshData& out) {
117 out.clear();
118 if (map.empty() || visibility.cellCount() != map.cellCount() || chunkIndex < 0 ||
119 chunkIndex >= map.chunkCount()) {
120 out.finalize();
121 return;
122 }
123
124 for (std::int32_t row = 0; row < HexMetrics::kChunkSizeZ; ++row) {
125 for (std::int32_t column = 0; column < HexMetrics::kChunkSizeX; ++column) {
126 const HexCoordinates coordinates = map.chunkCell(chunkIndex, column, row);
127 if (!map.contains(coordinates)) continue;
128 const std::int32_t index = map.indexOf(coordinates);
129 if (index < 0) continue;
130
131 float shade = 0.f;
132 if (!map.isExplored(coordinates)) {
133 shade = kFogShadeUnknown;
134 } else if (!visibility.isVisible(index)) {
135 shade = kFogShadeSeen;
136 } else {
137 // Fully visible: no overlay at all.
138 continue;
139 }
140 appendFogColumn(map, coordinates, shade, out);
141 }
142 }
143 out.finalize();
144}
