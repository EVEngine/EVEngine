12NoiseField layer(const NoiseField& source, int frequency, uint32_t stream) {
13 auto noise = source;
14 noise.seed ^= stream;
15 // Scale the lattice period with frequency, preserving the complete UV tile
16 // instead of repeating a tiny noise cell many times around the vessel.
17 noise.periodX *= frequency;
18 noise.periodY *= frequency;
19 return noise;
20}
21float grain(const NoiseField& source, float u, float v, int frequency, uint32_t stream) {
22 const auto noise = layer(source, frequency, stream);
23 return noise.valueNoise(u * frequency, v * frequency);
24}
25float smooth(float low, float high, float x) {
26 const float t = std::clamp((x - low) / (high - low), 0.f, 1.f);
27 return t * t * (3.f - 2.f * t);
28}
29struct Cells {
30 float pore = 0.f;
31 float crack = 0.f;
32};
33Cells cells(const NoiseField& source, float u, float v, int frequency) {
34 const auto noise = layer(source, frequency, 0xa511e9b3u);
35 const float x = u * frequency, y = v * frequency;
36 const int ix = int(std::floor(x)), iy = int(std::floor(y));
37 float first = 100.f, second = 100.f, pore = 0.f;
38 for (int j = -1; j <= 1; ++j) {
39 for (int i = -1; i <= 1; ++i) {
40 const int cx = ix + i, cy = iy + j;
41 const float sx = cx + 0.2f + 0.6f * noise.hash01(cx, cy);
42 const auto offset = layer(source, frequency, 0x63d83595u);
43 const float sy = cy + 0.2f + 0.6f * offset.hash01(cx, cy);
44 const float distance = std::hypot(x - sx, y - sy);
45 if (distance < first) {
46 second = first;
47 first = distance;
48 } else
49 second = std::min(second, distance);
50 const auto pits = layer(source, frequency, 0xb5297a4du);
51 const float selection = pits.hash01(cx, cy);
52 if (selection > 0.58f) {
53 const float radius = 0.07f + 0.10f * selection;
54 pore = std::max(pore, 1.f - smooth(radius * 0.25f, radius, distance));
55 }
56 }
57 }
58 return {pore, 1.f - smooth(0.012f, 0.055f, second - first)};
59}
60float surfaceHeight(Finish finish, float u, float v, const NoiseField& noise) {
61 const float broad = 0.65f * grain(noise, u, v, 1, 0x9e3779b9u) + 0.35f * grain(noise, u, v, 3, 0x68e31da4u);
62 const float fine = grain(noise, u, v, 24, 0x1b56c4e9u) - 0.5f;
63 const float wave = grain(noise, u, v, 2, 0x7f4a7c15u) - 0.5f;
64 const float rings = std::sin(tau * (v * 12.f + wave * 0.12f + noise.hash01(0, 0)));
65 if (finish == Finish::Celadon) {
66 const auto cell = cells(noise, u + wave * 0.035f, v + wave * 0.035f, 4);
67 return std::clamp(0.63f + 0.18f * (broad - 0.5f) + 0.045f * fine - 0.31f * cell.crack + 0.007f * rings, 0.f,
68 1.f);
69 }
70 const auto cell = cells(noise, u, v, 12);
71 switch (finish) {
72 case Finish::Clay:
73 return std::clamp(0.55f + 0.34f * (broad - 0.5f) + 0.23f * fine - 0.30f * cell.pore + 0.008f * rings, 0.f,
74 1.f);
75 case Finish::Glaze: {
76 const float flow = grain(noise, u + wave * 0.08f, v, 6, 0x94d049bbu);
77 return std::clamp(0.53f + 0.55f * (broad - 0.5f) + 0.20f * (flow - 0.5f) + 0.09f * fine -
78 0.055f * cell.pore + 0.01f * rings,
79 0.f, 1.f);
80 }
81 case Finish::Porcelain:
82 return std::clamp(0.60f + 0.12f * (broad - 0.5f) + 0.07f * fine - 0.04f * cell.pore + 0.004f * rings, 0.f,
83 1.f);
84 case Finish::Stoneware:
85 return std::clamp(0.54f + 0.55f * (broad - 0.5f) + 0.26f * fine - 0.25f * cell.pore + 0.012f * rings, 0.f,
86 1.f);
87 case Finish::Celadon: break;
88 }
89 return 0.5f;
90}
