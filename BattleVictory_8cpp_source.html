44eve::Result<BattleVictoryReceipt>
45BattleVictory::settle(RPGActor &actor, GameState &gameState, Tracker &tracker,
46 const BattleVictoryRequest &request) {
47 if (request.requiredQuestId.empty() || tracker.getState(request.requiredQuestId) != "active")
48 return victoryFailure(eve::DiagnosticCode::PreconditionViolation,
49 "required quest must be active before victory settlement",
50 "requiredQuestId", request);
51 if (request.notifyTopic.empty() || request.notifyTarget.empty() || request.notifyAmount <= 0)
52 return victoryFailure(eve::DiagnosticCode::InvalidArgument,
53 "victory quest notification must be complete and positive",
54 "notification", request);
55 if (request.attributeId.empty() || !std::isfinite(request.attributeAmount) ||
56 request.attributeAmount < 0.0 ||
57 !finiteAddition(gameState, request.attributeId, request.attributeAmount))
58 return victoryFailure(eve::DiagnosticCode::InvalidArgument,
59 "victory attribute reward is invalid or non-finite", "attribute", request);
60 if (request.defeatCounterId.empty() || request.defeatCounterAmount <= 0 ||
61 !finiteAddition(gameState, request.defeatCounterId,
62 static_cast<double>(request.defeatCounterAmount)))
63 return victoryFailure(eve::DiagnosticCode::InvalidArgument,
64 "victory counter increment is invalid", "defeatCounter", request);
65 if (request.levelPointAttributeId.empty() || request.pointsPerLevel < 0)
66 return victoryFailure(eve::DiagnosticCode::InvalidArgument,
67 "level-point reward configuration is invalid", "levelPoints", request);
68
69 auto preparedXp = LevelSystem::prepareGainXp(&actor, request.xpAmount, request.xpGrowth);
70 if (!preparedXp.ok())
71 return eve::Result<BattleVictoryReceipt>::failure(preparedXp.status());
72 PreparedProgressionGain xp = std::move(preparedXp).takeValue();
73 const int levelsGained = xp.levelsGained();
74 const double levelPoints = static_cast<double>(levelsGained) * request.pointsPerLevel;
75 if (!finiteAddition(gameState, request.levelPointAttributeId, levelPoints))
76 return victoryFailure(eve::DiagnosticCode::InvalidArgument,
77 "level-point reward would produce a non-finite value", "levelPoints", request);
78
79 auto gameStateBefore = gameState.snapshotJson();
80 if (!gameStateBefore.ok())
81 return eve::Result<BattleVictoryReceipt>::failure(gameStateBefore.status());
82 std::string gameStateSnapshot = std::move(gameStateBefore).takeValue();
83 GameState candidateState;
84 auto candidateRestore = candidateState.restoreSnapshotJson(gameStateSnapshot);
85 if (!candidateRestore.ok())
86 return eve::Result<BattleVictoryReceipt>::failure(candidateRestore.status());
87 WorldState candidateWorld(candidateState);
88 if (candidateWorld.isObjectConsumed(request.mapId, request.objectId))
89 return victoryFailure(eve::DiagnosticCode::Conflict,
90 "encounter was already consumed", "objectId", request);
91 auto candidateConsume = candidateWorld.consumeObject(request.mapId, request.objectId);
92 if (!candidateConsume.ok())
93 return eve::Result<BattleVictoryReceipt>::failure(candidateConsume.status());
94
95 TrackerRollback trackerBefore{tracker.entries, tracker.order, tracker.pending, tracker.polled};
96 gameState.addVariable(request.attributeId, request.attributeAmount);
97 gameState.addVariable(request.defeatCounterId, request.defeatCounterAmount);
98 if (levelPoints > 0.0) gameState.addVariable(request.levelPointAttributeId, levelPoints);
99 tracker.notify(request.notifyTopic, request.notifyTarget, request.notifyAmount);
100 WorldState world(gameState);
101 auto consumed = world.consumeObject(request.mapId, request.objectId);
102 if (!consumed.ok() || consumed.code() != eve::StatusCode::Applied) {
103 auto restored = gameState.restoreSnapshotJson(gameStateSnapshot);
104 const bool restoredOk = restored.ok();
105 restoreTracker(tracker, std::move(trackerBefore));
106 if (!restoredOk)
107 return victoryFailure(eve::DiagnosticCode::InvariantViolation,
108 "victory rollback could not restore GameState", "rollback", request);
109 return victoryFailure(eve::DiagnosticCode::Conflict,
110 "encounter changed before victory commit", "objectId", request);
111 }
112
113 auto committedXp = LevelSystem::commitGainXp(std::move(xp));
114 if (!committedXp.ok()) {
115 auto restored = gameState.restoreSnapshotJson(gameStateSnapshot);
116 const bool restoredOk = restored.ok();
117 restoreTracker(tracker, std::move(trackerBefore));
118 if (!restoredOk)
119 return victoryFailure(eve::DiagnosticCode::InvariantViolation,
120 "victory rollback could not restore GameState", "rollback", request);
121 return eve::Result<BattleVictoryReceipt>::failure(committedXp.status());
122 }
123
124 BattleVictoryReceipt receipt;
125 receipt.levelsGained = std::move(committedXp).takeValue();
126 receipt.skillsLearned = ClassSystem::checkLevelSkills(&actor);
127 return eve::Result<BattleVictoryReceipt>::success(
128 std::move(receipt), eve::Status::success(eve::StatusCode::Applied));
129}
