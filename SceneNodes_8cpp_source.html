314bool Scene::setNodeLookAtAt(const std::string &hostName, const std::string &nodeId,
315 float tx, float ty, float tz) {
316 SceneHost *h = resolveHost(hostName);
317 SceneNode *n = h ? borrowSceneResult(h->findById(nodeId)) : nullptr;
318 if (!h || !n) return false;
319 glm::vec3 pos(n->x, n->y, n->z);
320 glm::vec3 target(tx, ty, tz);
321 glm::vec3 f = target - pos;
322 if (glm::dot(f, f) < 1e-8f) return false;
323 f = glm::normalize(f);
324 glm::vec3 up(0.f, 1.f, 0.f);
325 glm::vec3 right = glm::normalize(glm::cross(up, f));
326 if (glm::dot(right, right) < 1e-8f) right = glm::vec3(1.f, 0.f, 0.f); // f ∥ up
327 glm::vec3 up2 = glm::normalize(glm::cross(f, right));
328 glm::mat3 rot(right, up2, f);
329 glm::quat q = glm::quat_cast(rot);
330 glm::vec3 e = glm::eulerAngles(q);
331 n->yaw = e.y;
332 n->pitch = e.x;
333 n->roll = e.z;
334 h->markSubtreeDirtyById(nodeId);
335 return true;
336}

393bool Scene::setNodeEventHandlerAt(const std::string &hostName, ssq::Object cb) {
394 if (!vm_) return false;
395 SceneHost *h = resolveHost(hostName);
396 if (!h) return false;
397 const std::string name = hostName.empty() ? h->getName() : hostName;
398
399 auto it = eventCbs_.find(name);
400 if (it != eventCbs_.end()) {
401 sq_release(vm_, &it->second);
402 eventCbs_.erase(it);
403 }
404 HSQOBJECT raw = cb.getRaw();
405 if (raw._type == OT_NULL) { // clear handler
406 h->setEventHandler({});
407 return true;
408 }
409 if (raw._type != OT_CLOSURE && raw._type != OT_NATIVECLOSURE) return false;
410 sq_addref(vm_, &raw);
411 eventCbs_[name] = raw;
412 h->setEventHandler(
413 [this, name](SceneHost *, const std::string &action, const std::string &nodeId,
414 const std::string &parentId) {
415 callEventCallback(name, action, nodeId, parentId);
416 });
417 return true;
418}

420void Scene::callEventCallback(const std::string &hostName, const std::string &action,
421 const std::string &nodeId, const std::string &parentId) {
422 if (!vm_) return;
423 auto it = eventCbs_.find(hostName);
424 if (it == eventCbs_.end()) return;
425 const SQInteger top = sq_gettop(vm_);
426 sq_pushobject(vm_, it->second);
427 sq_pushstring(vm_, action.c_str(), -1);
428 sq_pushstring(vm_, nodeId.c_str(), -1);
429 sq_pushstring(vm_, parentId.c_str(), -1);
430 if (SQ_FAILED(sq_call(vm_, 3, SQFalse, SQTrue))) {
431 sq_settop(vm_, top);
432 return;
433 }
434 sq_settop(vm_, top);
435}

472std::string Scene::pickRayAt(const std::string &hostName, float ox, float oy, float oz,
473 float dx, float dy, float dz) const {
474 SceneHost *h = resolveHost(hostName);
475 if (!h) return {};
476 const glm::vec3 origin(ox, oy, oz);
477 glm::vec3 dir(dx, dy, dz);
478 if (glm::dot(dir, dir) < 1e-12f) return {};
479 dir = glm::normalize(dir);
480
481 float bestT = std::numeric_limits<float>::max();
482 std::string best;
483 h->walkDepthFirst([&](SceneHost *, int, SceneNode &n) {
484 if (!n.hasBounds) return;
485 const AABB3f w = worldBoundsOf(n);
486 float t0 = 0.f, t1 = 0.f;
487 if (rayAABB(origin, dir, w, t0, t1) && t1 >= 0.f && t0 < bestT) {
488 bestT = std::max(t0, 0.f);
489 best = n.id;
490 }
491 });
492 return best;
493}
