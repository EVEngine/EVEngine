69std::string normalize(std::string_view value) {
70 std::string result(value);
71 std::transform(result.begin(), result.end(), result.begin(), [](unsigned char c) {
72 if (c == '_' || c == ' ') return '-';
73 return static_cast<char>(std::tolower(c));
74 });
75 return result;
76}
