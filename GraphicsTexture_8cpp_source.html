37template <class TextureImage>
38void recordTextureCopies(vk::CommandBuffer cb, TextureImage &image, vkb::GenericBuffer &staging, uint32_t width,
39 uint32_t height, uint32_t depth, uint32_t mipLevels, uint32_t layers,
40 uint32_t bytesPerPixel = 4) {
41 vk::DeviceSize offset = 0;
42 for (uint32_t mip = 0; mip < mipLevels; ++mip) {
43 const uint32_t mipWidth = std::max(width >> mip, 1u);
44 const uint32_t mipHeight = std::max(height >> mip, 1u);
45 const uint32_t mipDepth = std::max(depth >> mip, 1u);
46 for (uint32_t layer = 0; layer < layers; ++layer) {
47 image.copy(cb, staging.buffer, mip, layer, mipWidth, mipHeight, mipDepth, uint32_t(offset));
48 offset += vk::DeviceSize(mipWidth) * mipHeight * mipDepth * bytesPerPixel;
49 }
50 }
51 image.setLayout(cb, vk::ImageLayout::eShaderReadOnlyOptimal);
52}

54template <class TextureImage>
55void uploadTextureForAllShaderStages(vkb::Device &device, vk::CommandPool commandPool, vk::Queue graphicsQueue,
56 TextureImage &image, uint32_t width, uint32_t height, uint32_t mipLevels,
57 uint32_t layers, const std::vector<uint8_t> &bytes, uint32_t bytesPerPixel = 4,
58 uint32_t depth = 1) {
59 vkb::GenericBuffer staging(
60 device, vk::BufferUsageFlagBits::eTransferSrc, vk::DeviceSize(bytes.size()),
61 vk::MemoryPropertyFlagBits::eHostVisible | vk::MemoryPropertyFlagBits::eHostCoherent);
62 staging.updateLocal(vkb::FrameSlot::gpuIdle(), bytes.data(), vk::DeviceSize(bytes.size()));
63
64 vkb::executeImmediately(device.instance, commandPool, graphicsQueue, [&](vk::CommandBuffer cb) {
65 recordTextureCopies(cb, image, staging, width, height, depth, mipLevels, layers, bytesPerPixel);
66 });
67 staging.release();
68}

70void Graphics::writeCombinedImageDescriptor(GpuTexture *gpu) {
71 if (!gpu || !gpu->descriptorSet || !gpu->sampler) return;
72 vk::ImageView view = gpu->imageView();
73 if (!view) return;
74 vkb::UnboundSet unbound = vkb::UnboundSet::reopenAfterIdle(gpu->descriptorSet);
75 vkb::DescriptorSetUpdater updater;
76 updater.beginDescriptorSet(unbound)
77 .beginImages(0, 0, vk::DescriptorType::eCombinedImageSampler)
78 .image(vkb::SampledImage::forLaterSample(gpu->sampler, view))
79 .beginImages(1, 0, vk::DescriptorType::eCombinedImageSampler)
80 .image(vkb::SampledImage::forLaterSample(gpu->sampler, view))
81 .update(device.instance);
82 gpu->descriptorSet = std::move(unbound).publish();
83}

92Texture *Graphics::newTexture(int w, int h, const uint8_t *rgba, const TextureCreateInfo &rawInfo) {
93 ASSERT(initialized);
94 ASSERT_GT(w, 0);
95 ASSERT_GT(h, 0);
96 ASSERT(rgba != nullptr);
97 if (!initialized) throw Exception("newTexture: graphics not initialized");
98 if (w <= 0 || h <= 0 || !rgba) throw Exception("newTexture: invalid args");
99
100 TextureCreateInfo info = normalizeTextureInfo(rawInfo);
101 const uint32_t mipLevels =
102 info.generateMipmaps ? uint32_t(mipmapCountForSize(w, h)) : 1u;
103
104 auto gpu = std::make_unique<GpuTexture>();
105 gpu->width = w;
106 gpu->height = h;
107 gpu->isCube = false;
108 gpu->mipLevels = mipLevels;
109 gpu->samplerState = info.sampler;
110 gpu->image = vkb::TextureImage2D(device, uint32_t(w), uint32_t(h), mipLevels);
111
112 std::vector<uint8_t> bytes =
113 (mipLevels > 1) ? buildMipChain2D(rgba, uint32_t(w), uint32_t(h), mipLevels)
114 : std::vector<uint8_t>(rgba, rgba + size_t(w) * size_t(h) * 4);
115 uploadTextureForAllShaderStages(device, uploadPool,
116 device.getQueue(vkb::QueueType::graphics), gpu->image,
117 uint32_t(w), uint32_t(h), mipLevels, 1, bytes);
118
119 gpu->sampler = createVkSampler(info.sampler, mipLevels);
120
121 auto sets = vkb::DescriptorSetBuilder().layout(texSetLayout).build(device.instance, descriptorPool);
122
123 gpu->descriptorSet = vkb::BoundSet{sets[0]};
124 writeCombinedImageDescriptor(gpu.get());
125 registerBindlessTexture2D(gpu.get());
126
127 auto tex = std::make_unique<Texture>();
128 tex->width = w;
129 tex->height = h;
130 tex->pixelWidth = w;
131 tex->pixelHeight = h;
132 tex->mipmapCount = int(mipLevels);
133 tex->sampler = info.sampler;
134 tex->gpuHandle = gpu.get();
135
136 Texture *raw = tex.get();
137 ownedTextures.push_back(std::move(tex));
138 ownedGpuTextures.push_back(std::move(gpu));
139 return raw;
140}

142Result<Texture *> Graphics::newTextureMipChain(uint32_t width, uint32_t height, uint32_t levels,
143 std::span<const uint8_t> rgba) {
144 auto fail = [](DiagnosticCode code, std::string message) {
145 return Result<Texture *>::failure(Diagnostic::error(code, std::move(message), {}, {}, "graphics.texture.mips"));
146 };
147 if (!initialized) return fail(DiagnosticCode::Failed, "graphics is not initialized");
148 const auto maximum = device.physical_device.properties.limits.maxImageDimension2D;
149 if (!width || !height || width > maximum || height > maximum)
150 return fail(DiagnosticCode::InvalidArgument, "mip dimensions exceed device limits");
151 uint64_t expected = 0;
152 uint32_t count = 0, w = width, h = height;
153 for (;;) {
154 expected += uint64_t(w) * h * 4;
155 ++count;
156 if (w == 1 && h == 1) break;
157 w = std::max(w / 2, 1u);
158 h = std::max(h / 2, 1u);
159 }
160 if (levels != count || expected != rgba.size() || expected > UINT32_MAX)
161 return fail(DiagnosticCode::InvalidArgument, "mip count, packed bytes or staging offset budget invalid");
162 std::unique_ptr<GpuTexture> gpu;
163 try {
164 // Reserve both registries before allocating GPU state; publication then cannot allocate.
165 ownedTextures.reserve(ownedTextures.size() + 1);
166 ownedGpuTextures.reserve(ownedGpuTextures.size() + 1);
167 auto tex = std::make_unique<Texture>();
168 std::vector<uint8_t> bytes(rgba.begin(), rgba.end());
169 gpu = std::make_unique<GpuTexture>();
170 gpu->width = int(width);
171 gpu->height = int(height);
172 gpu->mipLevels = levels;
173 gpu->samplerState = TextureSampler::linearMipmap();
174 gpu->image = vkb::TextureImage2D(device, width, height, levels);
175 uploadTextureForAllShaderStages(device, uploadPool, device.getQueue(vkb::QueueType::graphics), gpu->image,
176 width, height, levels, 1, bytes);
177 gpu->sampler = createVkSampler(gpu->samplerState, levels);
178 auto sets = vkb::DescriptorSetBuilder().layout(texSetLayout).build(device.instance, descriptorPool);
179 gpu->descriptorSet = vkb::BoundSet{sets[0]};
180 writeCombinedImageDescriptor(gpu.get());
181 tex->width = tex->pixelWidth = int(width);
182 tex->height = tex->pixelHeight = int(height);
183 tex->mipmapCount = int(levels);
184 tex->sampler = gpu->samplerState;
185 tex->gpuHandle = gpu.get();
186 registerBindlessTexture2D(gpu.get());
187 auto *result = tex.get();
188 ownedTextures.push_back(std::move(tex));
189 ownedGpuTextures.push_back(std::move(gpu));
190 return Result<Texture *>::success(result);
191 } catch (const std::exception &error) {
192 if (gpu) {
193 unregisterBindlessTexture(gpu.get());
194 if (gpu->descriptorSet) device->freeDescriptorSets(descriptorPool, {gpu->descriptorSet.handle});
195 if (gpu->sampler) device->destroySampler(gpu->sampler);
196 }
197 return fail(DiagnosticCode::Failed, error.what());
198 }

785bool Graphics::releaseTexture(Texture *texture) {
786 if (!texture) return false;
787 dropDeferredFileTexture(texture);
788 if (!texture->gpuHandle) {
789 auto texIt = std::find_if(ownedTextures.begin(), ownedTextures.end(),
790 [&](const std::unique_ptr<Texture> &t) {
791 return t.get() == texture;
792 });
793 if (texIt == ownedTextures.end()) return false;
794 for (auto it = texturesByPath.begin(); it != texturesByPath.end();) {
795 if (it->second == texture)
796 it = texturesByPath.erase(it);
797 else
798 ++it;
799 }
800 (void)texIt->release();
801 ownedTextures.erase(texIt);
802 return true;
803 }
804 // Renderer-owned fallback textures must never be released by callers.
805 if (texture == whiteTexture || texture == flatNormalTexture || texture == flatNormalTexture3D ||
806 texture == defaultEnvCubemap || texture == defaultExtrasArray || texture == defaultColorsArray ||
807 texture == defaultVertexArray || texture == defaultMotionArray || texture == defaultVegetationFadeNoise)
808 return false;
809
810 auto *gpu = static_cast<GpuTexture *>(texture->gpuHandle);
811 auto gpuIt = std::find_if(ownedGpuTextures.begin(), ownedGpuTextures.end(),
812 [&](const std::unique_ptr<GpuTexture> &g) {
813 return g.get() == gpu;
814 });
815 if (gpuIt == ownedGpuTextures.end()) return false;
816
817 auto texIt = std::find_if(ownedTextures.begin(), ownedTextures.end(),
818 [&](const std::unique_ptr<Texture> &t) {
819 return t.get() == texture;
820 });
821 if (texIt == ownedTextures.end()) return false;
822
823 // Path-cached textures must leave the hot-reload cache once released.
824 for (auto it = texturesByPath.begin(); it != texturesByPath.end();) {
825 if (it->second == texture)
826 it = texturesByPath.erase(it);
827 else
828 ++it;
829 }
830
831 // In-flight frames may still sample the image / sampler; drain first.
832 waitForSharedGpuResources();
833 unregisterBindlessTexture(gpu);
834 if ((*gpuIt)->sampler) device->destroySampler((*gpuIt)->sampler);
835 texture->gpuHandle = nullptr;
836 ownedGpuTextures.erase(gpuIt);
837 // Transfer the CPU facade to the caller instead of destroying it.
838 (void)texIt->release();
839 ownedTextures.erase(texIt);
840 return true;
841}

928eve::Result<void> Graphics::updateTextureRegions(
929 Texture *tex, std::span<const TextureRegionUpload> regions) {
930 if (!tex)
931 return eve::Result<void>::failure(eve::Diagnostic::error(
932 eve::DiagnosticCode::InvalidArgument, "texture must be non-null",
933 "graphics.updateTextureRegions.texture"));
934 if (tex->mipmapCount != 1)
935 return eve::Result<void>::failure(eve::Diagnostic::error(
936 eve::DiagnosticCode::Unsupported, "partial updates require a single-mip texture",
937 "graphics.updateTextureRegions.mipmaps"));
938
939 auto *gpu = static_cast<GpuTexture *>(tex->gpuHandle);
940 const auto owned = std::find_if(ownedGpuTextures.begin(), ownedGpuTextures.end(),
941 [gpu](const std::unique_ptr<GpuTexture> &candidate) {
942 return candidate.get() == gpu;
943 });
944 if (!gpu || owned == ownedGpuTextures.end() || gpu->isCube)
945 return eve::Result<void>::failure(eve::Diagnostic::error(
946 eve::DiagnosticCode::InvalidArgument, "texture is not an owned 2D texture",
947 "graphics.updateTextureRegions.texture"));
948 if (regions.empty()) return eve::Result<void>::success();
949
950 std::vector<std::size_t> packedOffsets;
951 packedOffsets.reserve(regions.size());
952 std::size_t packedSize = 0;
953 for (const TextureRegionUpload &region : regions) {
954 if (region.x < 0 || region.y < 0 || region.width <= 0 || region.height <= 0 ||
955 region.x > tex->width - region.width || region.y > tex->height - region.height)
956 return eve::Result<void>::failure(eve::Diagnostic::error(
957 eve::DiagnosticCode::InvalidArgument, "invalid texture region",
958 "graphics.updateTextureRegions.region"));
959 const std::size_t packedRow = std::size_t(region.width) * 4U;
960 const std::size_t stride = region.bytesPerRow == 0 ? packedRow : region.bytesPerRow;
961 const std::size_t requiredBytes = stride * std::size_t(region.height - 1) + packedRow;
962 if (stride < packedRow || region.rgba.size() < requiredBytes)
963 return eve::Result<void>::failure(eve::Diagnostic::error(
964 eve::DiagnosticCode::InvalidArgument,
965 "source bytes do not cover the texture region",
966 "graphics.updateTextureRegions.bytes"));
967 packedOffsets.push_back(packedSize);
968 packedSize += packedRow * std::size_t(region.height);
969 }
970
971 std::vector<std::uint8_t> packed(packedSize);
972 for (std::size_t index = 0; index < regions.size(); ++index) {
973 const TextureRegionUpload &region = regions[index];
974 const std::size_t packedRow = std::size_t(region.width) * 4U;
975 const std::size_t stride = region.bytesPerRow == 0 ? packedRow : region.bytesPerRow;
976 for (int row = 0; row < region.height; ++row)
977 std::copy_n(region.rgba.data() + std::size_t(row) * stride, packedRow,
978 packed.data() + packedOffsets[index] + std::size_t(row) * packedRow);
979 }

981 device, vk::BufferUsageFlagBits::eTransferSrc, vk::DeviceSize(packed.size()),
982 vk::MemoryPropertyFlagBits::eHostVisible | vk::MemoryPropertyFlagBits::eHostCoherent);

987 device.getQueue(vkb::QueueType::graphics),
988 [&](vk::CommandBuffer cb) {

991 copies.reserve(regions.size());
992 for (std::size_t index = 0; index < regions.size(); ++index) {
993 const TextureRegionUpload &source = regions[index];
994 vk::BufferImageCopy copy{};
995 copy.bufferOffset = packedOffsets[index];
996 copy.imageSubresource = {vk::ImageAspectFlagBits::eColor, 0, 0, 1};
997 copy.imageOffset = vk::Offset3D{source.x, source.y, 0};
998 copy.imageExtent = vk::Extent3D{std::uint32_t(source.width),
999 std::uint32_t(source.height), 1};
1000 copies.push_back(copy);
1001 }
1002 cb.copyBufferToImage(staging.buffer, gpu->image.image(),
1003 vk::ImageLayout::eTransferDstOptimal, copies);
1004
1005 vk::ImageMemoryBarrier barrier{};
1006 barrier.srcAccessMask = vk::AccessFlagBits::eTransferWrite;
1007 barrier.dstAccessMask = vk::AccessFlagBits::eShaderRead;
1008 barrier.oldLayout = vk::ImageLayout::eTransferDstOptimal;
1009 barrier.newLayout = vk::ImageLayout::eShaderReadOnlyOptimal;
1010 barrier.srcQueueFamilyIndex = VK_QUEUE_FAMILY_IGNORED;
1011 barrier.dstQueueFamilyIndex = VK_QUEUE_FAMILY_IGNORED;
1012 barrier.image = gpu->image.image();
1013 barrier.subresourceRange = {vk::ImageAspectFlagBits::eColor, 0, 1, 0, 1};
1014 cb.pipelineBarrier(vk::PipelineStageFlagBits::eTransfer,
1015 vk::PipelineStageFlagBits::eVertexShader |
1016 vk::PipelineStageFlagBits::eFragmentShader |
1017 vk::PipelineStageFlagBits::eComputeShader,
1018 {}, 0, nullptr, 0, nullptr, 1, &barrier);
1019 gpu->image.setCurrentLayout(vk::ImageLayout::eShaderReadOnlyOptimal);
1020 });
1021 staging.release();
1022 return eve::Result<void>::success();

1079 tex->gpuHandle = gpu.get();

1081 tex->width = w;

1084 tex->pixelHeight = h;

1092 ASSERT(!filename.empty());
1093 if (filename.empty()) throw Exception("newTextureFromFile: empty filename");
1094
1095 const std::string key = normalizeTexPath(filename);

1101 requestFileImageDecode(key);

1103 auto waited = eve::ResourceManager::getInstance().waitFor(key);
1104 if (!waited.ok()) throw Exception("%s", waited.status().describe().c_str());
1105 auto *data = dynamic_cast<image::ImageData *>(&waited.value().get());

1115 ownedTextures.push_back(std::move(tex));
1116 texturesByPath[key] = raw;
1117 deferredFileTextures_.push_back({key, raw});

1137 auto *imgMod = image::Image::create();
