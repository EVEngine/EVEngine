25void Graphics::writeCombinedImageDescriptor(GpuTexture *gpu) {
26 if (!gpu || !gpu->descriptorSet || !gpu->sampler) return;
27 vk::ImageView view = gpu->imageView();
28 if (!view) return;
29 vkb::UnboundSet unbound = vkb::UnboundSet::reopenAfterIdle(gpu->descriptorSet);
30 vkb::DescriptorSetUpdater updater;
31 updater.beginDescriptorSet(unbound)
32 .beginImages(0, 0, vk::DescriptorType::eCombinedImageSampler)
33 .image(vkb::SampledImage::forLaterSample(gpu->sampler, view))
34 .beginImages(1, 0, vk::DescriptorType::eCombinedImageSampler)
35 .image(vkb::SampledImage::forLaterSample(gpu->sampler, view))
36 .update(device.instance);
37 gpu->descriptorSet = std::move(unbound).publish();
38}

47Texture *Graphics::newTexture(int w, int h, const uint8_t *rgba, const TextureCreateInfo &rawInfo) {
48 ASSERT(initialized);
49 ASSERT_GT(w, 0);
50 ASSERT_GT(h, 0);
51 ASSERT(rgba != nullptr);
52 if (!initialized) throw Exception("newTexture: graphics not initialized");
53 if (w <= 0 || h <= 0 || !rgba) throw Exception("newTexture: invalid args");
54
55 TextureCreateInfo info = normalizeTextureInfo(rawInfo);
56 const uint32_t mipLevels =
57 info.generateMipmaps ? uint32_t(mipmapCountForSize(w, h)) : 1u;
58
59 auto gpu = std::make_unique<GpuTexture>();
60 gpu->width = w;
61 gpu->height = h;
62 gpu->isCube = false;
63 gpu->mipLevels = mipLevels;
64 gpu->samplerState = info.sampler;
65 gpu->image = vkb::TextureImage2D(device, uint32_t(w), uint32_t(h), mipLevels);
66
67 std::vector<uint8_t> bytes =
68 (mipLevels > 1) ? buildMipChain2D(rgba, uint32_t(w), uint32_t(h), mipLevels)
69 : std::vector<uint8_t>(rgba, rgba + size_t(w) * size_t(h) * 4);
70 gpu->image.upload(uploadPool, device.getQueue(vkb::QueueType::graphics), bytes);
71
72 gpu->sampler = createVkSampler(info.sampler, mipLevels);
73
74 auto sets = vkb::DescriptorSetBuilder().layout(texSetLayout).build(device.instance, descriptorPool);
75
76 gpu->descriptorSet = vkb::BoundSet{sets[0]};
77 writeCombinedImageDescriptor(gpu.get());
78 registerBindlessTexture2D(gpu.get());
79
80 auto tex = std::make_unique<Texture>();
81 tex->width = w;
82 tex->height = h;
83 tex->pixelWidth = w;
84 tex->pixelHeight = h;
85 tex->mipmapCount = int(mipLevels);
86 tex->sampler = info.sampler;
87 tex->gpuHandle = gpu.get();
88
89 Texture *raw = tex.get();
90 ownedTextures.push_back(std::move(tex));
91 ownedGpuTextures.push_back(std::move(gpu));
92 return raw;
93}

100Texture *Graphics::newCubemap(int faceSize, const uint8_t *rgbaFaces,
101 const TextureCreateInfo &rawInfo) {
102 ASSERT(initialized);
103 ASSERT_GT(faceSize, 0);
104 ASSERT(rgbaFaces != nullptr);
105 if (!initialized) throw Exception("newCubemap: graphics not initialized");
106 if (faceSize <= 0 || !rgbaFaces) throw Exception("newCubemap: invalid args");
107
108 TextureCreateInfo info = normalizeTextureInfo(rawInfo);
109 info.sampler.repeatU = false;
110 info.sampler.repeatV = false;
111 info.sampler.repeatW = false;
112 const uint32_t mipLevels =
113 info.generateMipmaps ? uint32_t(mipmapCountForSize(faceSize, faceSize)) : 1u;
114
115 const size_t faceBytes = size_t(faceSize) * size_t(faceSize) * 4u;
116 auto gpu = std::make_unique<GpuTexture>();
117 gpu->width = faceSize;
118 gpu->height = faceSize;
119 gpu->isCube = true;
120 gpu->mipLevels = mipLevels;
121 gpu->samplerState = info.sampler;
122 gpu->cubeImage = vkb::TextureImageCube(device, device.physical_device.memory_properties,
123 uint32_t(faceSize), uint32_t(faceSize), mipLevels);
124
125 std::vector<uint8_t> bytes =
126 (mipLevels > 1) ? buildMipChainCube(rgbaFaces, uint32_t(faceSize), mipLevels)
127 : std::vector<uint8_t>(rgbaFaces, rgbaFaces + faceBytes * 6u);
128 gpu->cubeImage.upload(uploadPool, device.getQueue(vkb::QueueType::graphics), bytes);
129
130 gpu->sampler = createVkSampler(info.sampler, mipLevels);
131 // Cubemap sampled via mesh3d descriptor sets — no 2D texSetLayout binding required here.
132 registerBindlessTextureCube(gpu.get());
133
134 auto tex = std::make_unique<Texture>();
135 tex->width = faceSize;
136 tex->height = faceSize;
137 tex->pixelWidth = faceSize;
138 tex->pixelHeight = faceSize;
139 tex->layers = 6;
140 tex->mipmapCount = int(mipLevels);
141 tex->sampler = info.sampler;
142 tex->gpuHandle = gpu.get();
143
144 Texture *raw = tex.get();
145 ownedTextures.push_back(std::move(tex));
146 ownedGpuTextures.push_back(std::move(gpu));
147 return raw;
148}

169void Graphics::setTextureSampler(Texture *texture, const TextureSampler &sampler) {
170 if (!texture || !texture->gpuHandle || !initialized) return;
171 for (auto &owned : ownedGpuTextures) {
172 if (owned.get() != texture->gpuHandle) continue;
173 // In-flight frames may still be sampling the old sampler / descriptor.
174 waitForSharedGpuResources();
175 if (owned->sampler) device->destroySampler(owned->sampler);
176 owned->samplerState = sampler;
177 owned->sampler = createVkSampler(sampler, owned->mipLevels);
178 texture->sampler = sampler;
179 if (!owned->isCube) writeCombinedImageDescriptor(owned.get());
180 invalidateTextureBindings();
181 return;
182 }
183}

185bool Graphics::releaseTexture(Texture *texture) {
186 if (!texture || !texture->gpuHandle) return false;
187 // Renderer-owned fallback textures must never be released by callers.
188 if (texture == whiteTexture || texture == flatNormalTexture ||
189 texture == flatNormalTexture3D || texture == defaultEnvCubemap)
190 return false;
191
192 auto *gpu = static_cast<GpuTexture *>(texture->gpuHandle);
193 auto gpuIt = std::find_if(ownedGpuTextures.begin(), ownedGpuTextures.end(),
194 [&](const std::unique_ptr<GpuTexture> &g) {
195 return g.get() == gpu;
196 });
197 if (gpuIt == ownedGpuTextures.end()) return false;
198
199 auto texIt = std::find_if(ownedTextures.begin(), ownedTextures.end(),
200 [&](const std::unique_ptr<Texture> &t) {
201 return t.get() == texture;
202 });
203 if (texIt == ownedTextures.end()) return false;
204
205 // Path-cached textures must leave the hot-reload cache once released.
206 for (auto it = texturesByPath.begin(); it != texturesByPath.end();) {
207 if (it->second == texture)
208 it = texturesByPath.erase(it);
209 else
210 ++it;
211 }
212
213 // In-flight frames may still sample the image / sampler; drain first.
214 waitForSharedGpuResources();
215 unregisterBindlessTexture(gpu);
216 if ((*gpuIt)->sampler) device->destroySampler((*gpuIt)->sampler);
217 texture->gpuHandle = nullptr;
218 ownedGpuTextures.erase(gpuIt);
219 // Transfer the CPU facade to the caller instead of destroying it.
220 (void)texIt->release();
221 ownedTextures.erase(texIt);
222 return true;
223}

225bool Graphics::replaceTexturePixels(Texture *tex, image::ImageData *data) {
226 if (!tex || !data) return false;
227 if (data->getFormat() != "RGBA8") return false;
228 const int w = data->getWidth();
229 const int h = data->getHeight();
230 const auto *rgba = static_cast<const uint8_t *>(data->getData());
231 if (w <= 0 || h <= 0 || !rgba) return false;
232
233 TextureCreateInfo info;
234 info.sampler = tex->sampler;
235 info.generateMipmaps = tex->mipmapCount > 1;
236 info = normalizeTextureInfo(info);
237 const uint32_t mipLevels =
238 info.generateMipmaps ? uint32_t(mipmapCountForSize(w, h)) : 1u;
239
240 auto gpu = std::make_unique<GpuTexture>();
241 gpu->width = w;
242 gpu->height = h;
243 gpu->isCube = false;
244 gpu->mipLevels = mipLevels;
245 gpu->samplerState = info.sampler;
246 gpu->image = vkb::TextureImage2D(device, uint32_t(w), uint32_t(h), mipLevels);
247 std::vector<uint8_t> bytes =
248 (mipLevels > 1) ? buildMipChain2D(rgba, uint32_t(w), uint32_t(h), mipLevels)
249 : std::vector<uint8_t>(rgba, rgba + size_t(w) * size_t(h) * 4);
250 gpu->image.upload(uploadPool, device.getQueue(vkb::QueueType::graphics), bytes);
251
252 gpu->sampler = createVkSampler(info.sampler, mipLevels);
253
254 auto sets = vkb::DescriptorSetBuilder().layout(texSetLayout).build(device.instance, descriptorPool);
255
256 gpu->descriptorSet = vkb::BoundSet{sets[0]};
257 writeCombinedImageDescriptor(gpu.get());
258 registerBindlessTexture2D(gpu.get());
259
260 void *oldHandle = tex->gpuHandle;
261 for (auto &owned : ownedGpuTextures) {
262 if (owned.get() != oldHandle) continue;
263 // Destroying the old image/sampler while an in-flight frame still
264 // samples it is a typical TDR. Drain first, then drop cached sets.
265 waitForSharedGpuResources();
266 unregisterBindlessTexture(static_cast<GpuTexture *>(oldHandle));
267 if (owned->sampler) device->destroySampler(owned->sampler);
268 owned = std::move(gpu);
269 tex->gpuHandle = owned.get();
270 tex->width = w;
271 tex->height = h;
272 tex->pixelWidth = w;
273 tex->pixelHeight = h;
274 tex->mipmapCount = int(mipLevels);
275 tex->sampler = info.sampler;
276 registerBindlessTexture2D(owned.get());
277 invalidateTextureBindings();
278 return true;
279 }
280
281 // Texture not in owned list — attach as new ownership.
282 tex->gpuHandle = gpu.get();
283 registerBindlessTexture2D(static_cast<GpuTexture *>(tex->gpuHandle));
284 tex->width = w;
285 tex->height = h;
286 tex->pixelWidth = w;
287 tex->pixelHeight = h;
288 tex->mipmapCount = int(mipLevels);
289 tex->sampler = info.sampler;
290 ownedGpuTextures.push_back(std::move(gpu));
291 return true;
292}

294Texture *Graphics::newTextureFromFile(const std::string &filename) {
295 ASSERT(!filename.empty());
296 if (filename.empty()) throw Exception("newTextureFromFile: empty filename");
297
298 const std::string key = normalizeTexPath(filename);
299 auto *imgMod = image::Image::create();
300 eve::ref<image::ImageData> data(imgMod->newImageDataFromFile(filename));
301
302 auto it = texturesByPath.find(key);
303 if (it != texturesByPath.end() && it->second) {
304 if (!replaceTexturePixels(it->second, data.get()))
305 throw Exception("newTextureFromFile: reload failed '%s'", filename.c_str());
306 return it->second;
307 }
308
309 Texture *tex = newTexture(data.get());
310 texturesByPath[key] = tex;
311 return tex;
312}

314bool Graphics::reloadTextureFromFile(const std::string &filename) {
315 if (filename.empty()) return false;
316 const std::string key = normalizeTexPath(filename);
317 auto it = texturesByPath.find(key);
318 if (it == texturesByPath.end() || !it->second) return false;
319
320 image::ImageData *data = nullptr;
321 try {
322 auto *imgMod = image::Image::create();
323 eve::ref<image::ImageData> cached(imgMod->newImageDataFromFile(filename));
324 data = cached.get();
325 } catch (...) {
326 return false;
327 }
328 if (!data) return false;
329 return replaceTexturePixels(it->second, data);
330}
