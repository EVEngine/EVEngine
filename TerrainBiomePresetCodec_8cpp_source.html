22 entries.reserve(entries_.size());

30 {"autoAssignResources", entry.autoAssignResources},
31 {"entryId", entry.entryId},

36 {"version", 1}})

44Result<void> TerrainBiomePreset::restoreJson(const std::string& json) {
45 if (json.size() > kMaximumPresetJsonBytes)
46 return malformedPreset<void>("terrain.biomePreset.restore: JSON exceeds size limit");

48 if (!decoded.ok()) return Result<void>::failure(decoded.status());
49 const auto* root = decoded.value().getIf<Value::Object>();
50 if (!root || root->size() != 3 || !root->contains("schema") || !root->contains("version") ||
51 !root->contains("entries"))
52 return malformedPreset<void>("terrain.biomePreset.restore: missing or unknown root fields");
53 const auto* schema = root->at("schema").getIf<std::string>();
54 const auto* version = root->at("version").getIf<std::int64_t>();
55 const auto* entries = root->at("entries").getIf<Value::Array>();
56 if (!schema || *schema != "eve.procgen.terrain-biome-preset" || !version ||
57 (*version != 0 && *version != 1) || !entries || entries->size() > kMaximumSpawnerEntries)
58 return malformedPreset<void>("terrain.biomePreset.restore: unsupported schema, version or entry count");
59
60 TerrainBiomePreset candidate;
61 for (const auto& encodedEntry : *entries) {
62 const auto* entry = encodedEntry.getIf<Value::Object>();
63 const std::size_t expectedFields = *version == 0 ? 4 : 5;
64 if (!entry || entry->size() != expectedFields || !entry->contains("entryId") ||
65 !entry->contains("activeInBiome") || !entry->contains("activeInStamper") ||
66 !entry->contains("plan") || (*version >= 1 && !entry->contains("autoAssignResources")))
67 return malformedPreset<void>("terrain.biomePreset.restore: missing or unknown entry fields");
68 const auto* entryId = entry->at("entryId").getIf<std::string>();
69 const auto* activeInBiome = entry->at("activeInBiome").getIf<bool>();
70 const auto* activeInStamper = entry->at("activeInStamper").getIf<bool>();
71 const auto* planJson = entry->at("plan").getIf<std::string>();
72 const bool* autoAssign = nullptr;
73 if (*version >= 1) autoAssign = entry->at("autoAssignResources").getIf<bool>();
74 if (!entryId || entryId->empty() || !activeInBiome || !activeInStamper || !planJson ||
75 (*version >= 1 && !autoAssign))
76 return malformedPreset<void>("terrain.biomePreset.restore: invalid entry values");

79 if (!restored.ok()) return Result<void>::failure(restored.status());

81 *version == 0 ? true : *autoAssign);
82 if (!added.ok()) return Result<void>::failure(added.status());
83 }
