46float pointSegmentDistanceSquared(float px, float py, float pz, const ProcgenPoint& a, const ProcgenPoint& b) {
47 const float vx = b.x - a.x;
48 const float vy = b.y - a.y;
49 const float vz = b.z - a.z;
50 const float l2 = vx * vx + vy * vy + vz * vz;
51 const float t = l2 > 0.f ? std::clamp(((px - a.x) * vx + (py - a.y) * vy + (pz - a.z) * vz) / l2, 0.f, 1.f) : 0.f;
52 const float dx = px - (a.x + vx * t);
53 const float dy = py - (a.y + vy * t);
54 const float dz = pz - (a.z + vz * t);
55 return dx * dx + dy * dy + dz * dz;
56}

58bool pointInPolygon(float x, float z, const std::vector<ProcgenPoint>& polygon) {
59 bool inside = false;

69bool triangleHeightAtXZ(const std::vector<float>& positions, std::uint32_t ia, std::uint32_t ib, std::uint32_t ic,
70 float x, float z, float& y, float& nx, float& ny, float& nz) {
71 const size_t a = size_t(ia) * 3;
72 const size_t b = size_t(ib) * 3;
73 const size_t c = size_t(ic) * 3;
74 if (a + 2 >= positions.size() || b + 2 >= positions.size() || c + 2 >= positions.size()) return false;
75 const float ax = positions[a], ay = positions[a + 1], az = positions[a + 2];
76 const float bx = positions[b], by = positions[b + 1], bz = positions[b + 2];
77 const float cx = positions[c], cy = positions[c + 1], cz = positions[c + 2];
78 const float denominator = (bz - cz) * (ax - cx) + (cx - bx) * (az - cz);
79 if (std::abs(denominator) <= 1e-8f) return false;
80 const float wa = ((bz - cz) * (x - cx) + (cx - bx) * (z - cz)) / denominator;
81 const float wb = ((cz - az) * (x - cx) + (ax - cx) * (z - cz)) / denominator;
82 const float wc = 1.f - wa - wb;
83 constexpr float epsilon = 1e-5f;
84 if (wa < -epsilon || wb < -epsilon || wc < -epsilon) return false;
85 y = wa * ay + wb * by + wc * cy;
86 const float ux = bx - ax, uy = by - ay, uz = bz - az;
87 const float vx = cx - ax, vy = cy - ay, vz = cz - az;
88 nx = uy * vz - uz * vy;
89 ny = uz * vx - ux * vz;
90 nz = ux * vy - uy * vx;
91 const float length = std::sqrt(nx * nx + ny * ny + nz * nz);
92 if (length > 0.f) {
93 nx /= length;
94 ny /= length;
95 nz /= length;
96 }
97 if (ny < 0.f) {
98 nx = -nx;
99 ny = -ny;
100 nz = -nz;
101 }
102 return true;
103}

105bool closestMeshHeight(const MeshBuild& mesh, float x, float z, float referenceY, float& y, float& nx, float& ny,
106 float& nz) {
107 bool found = false;
108 float bestDistance = std::numeric_limits<float>::max();
109 const auto& positions = mesh.positions();
110 const auto& indices = mesh.indices();
111 for (size_t i = 0; i + 2 < indices.size(); i += 3) {
112 float candidateY = 0.f, candidateNx = 0.f, candidateNy = 1.f, candidateNz = 0.f;
113 if (!triangleHeightAtXZ(positions, indices[i], indices[i + 1], indices[i + 2], x, z, candidateY, candidateNx,
114 candidateNy, candidateNz))
115 continue;
116 const float distance = std::abs(referenceY - candidateY);
117 if (distance >= bestDistance) continue;
118 found = true;
119 bestDistance = distance;
120 y = candidateY;
121 nx = candidateNx;
122 ny = candidateNy;
123 nz = candidateNz;
124 }

312bool SpatialData::contains(float x, float y, float z) const {
313 switch (kind_) {
314 case Kind::Points:
315 for (const auto& point : points_.points())
316 if (point.x == x && point.y == y && point.z == z) return true;
317 return false;
318 case Kind::Box:
319 return x >= bounds_.minX && x <= bounds_.maxX && y >= bounds_.minY && y <= bounds_.maxY &&
320 z >= bounds_.minZ && z <= bounds_.maxZ;
321 case Kind::Sphere: {
322 const float dx = x - centerX_;
323 const float dy = y - centerY_;
324 const float dz = z - centerZ_;
325 return dx * dx + dy * dy + dz * dz <= radius_ * radius_;
326 }
327 case Kind::Polygon: return y >= bounds_.minY && y <= bounds_.maxY && pointInPolygon(x, z, points_.points());
328 case Kind::Spline:
329 for (size_t i = 1; i < points_.points().size(); ++i)
330 if (pointSegmentDistanceSquared(x, y, z, points_.points()[i - 1], points_.points()[i]) <=
331 radius_ * radius_)
332 return true;
333 return false;
334 case Kind::Heightfield: {
335 if (x < bounds_.minX || x > bounds_.maxX || z < bounds_.minZ || z > bounds_.maxZ) return false;
336 const float hx = (x - originX_) / cellSize_;
337 const float hz = (z - originZ_) / cellSize_;
338 const float surface = heightmap_->sampleBilinear(hx, hz) * heightScale_;
339 return std::abs(y - surface) <= cellSize_ * 0.5f;
340 }
341 case Kind::TextureMask: {
342 if (x < bounds_.minX || x > bounds_.maxX || y < bounds_.minY || y > bounds_.maxY || z < bounds_.minZ ||
343 z > bounds_.maxZ)
344 return false;
345 const float value = heightmap_->sampleBilinear((x - originX_) / cellSize_, (z - originZ_) / cellSize_);
346 return value >= minValue_ && value <= maxValue_;
347 }
348 case Kind::MeshSurface: {
349 float surfaceY = 0.f, nx = 0.f, ny = 1.f, nz = 0.f;
350 return closestMeshHeight(*mesh_, x, z, y, surfaceY, nx, ny, nz) && std::abs(y - surfaceY) <= radius_;
351 }
352 case Kind::Union: return left_->contains(x, y, z) || right_->contains(x, y, z);
353 case Kind::Intersection: return left_->contains(x, y, z) && right_->contains(x, y, z);
354 case Kind::Difference: return left_->contains(x, y, z) && !right_->contains(x, y, z);
355 default: return false;
356 }
357}

368PointSet SpatialData::sample(float spacing, uint32_t seed, float jitter) const {
369 PointSet output;
370 if (!bounds_.valid || spacing <= 0.f) return output;
371 jitter = std::clamp(jitter, 0.f, 1.f);
372 const float extent = spacing * jitter * 0.5f;
373 uint32_t index = 0;
374 if (kind_ == Kind::Heightfield || kind_ == Kind::MeshSurface) {
375 for (float z = bounds_.minZ; z <= bounds_.maxZ + spacing * 0.001f; z += spacing) {
376 for (float x = bounds_.minX; x <= bounds_.maxX + spacing * 0.001f; x += spacing) {
377 const uint32_t pointSeed = mixSpatialSeed(seed ^ index++);
378 const float sx =
379 std::clamp(x + (unitFloat(pointSeed) * 2.f - 1.f) * extent, bounds_.minX, bounds_.maxX);
380 const float sz = std::clamp(z + (unitFloat(pointSeed ^ 0x02e5be93u) * 2.f - 1.f) * extent, bounds_.minZ,
381 bounds_.maxZ);
382 const int added = output.add(sx, 0.f, sz);
383 output.setPointSeed(added, pointSeed);
384 }
385 }
386 return project(output);
387 }
388 for (float z = bounds_.minZ; z <= bounds_.maxZ + spacing * 0.001f; z += spacing) {
389 for (float y = bounds_.minY; y <= bounds_.maxY + spacing * 0.001f; y += spacing) {
390 for (float x = bounds_.minX; x <= bounds_.maxX + spacing * 0.001f; x += spacing) {
391 const uint32_t pointSeed = mixSpatialSeed(seed ^ index++);
392 const float sx = bounds_.maxX > bounds_.minX ? x + (unitFloat(pointSeed) * 2.f - 1.f) * extent : x;
393 const float sy =
394 bounds_.maxY > bounds_.minY ? y + (unitFloat(pointSeed ^ 0x68bc21ebu) * 2.f - 1.f) * extent : y;
395 const float sz =
396 bounds_.maxZ > bounds_.minZ ? z + (unitFloat(pointSeed ^ 0x02e5be93u) * 2.f - 1.f) * extent : z;
397 if (!contains(sx, sy, sz)) continue;
398 const int added = output.add(sx, sy, sz);
399 output.setPointSeed(added, pointSeed);
400 }
401 }
402 }
403 return output;
404}

416PointSet SpatialData::project(const PointSet& input) const {
417 if (kind_ != Kind::Heightfield && kind_ != Kind::MeshSurface) return input;
418 PointSet output = input;
419 for (size_t index = 0; index < output.points().size(); ++index) {
420 auto& point = output.mutablePoint(index);
421 if (point.x < bounds_.minX || point.x > bounds_.maxX || point.z < bounds_.minZ || point.z > bounds_.maxZ)
422 continue;
423 if (kind_ == Kind::MeshSurface) {
424 float y = 0.f, nx = 0.f, ny = 1.f, nz = 0.f;
425 if (!closestMeshHeight(*mesh_, point.x, point.z, point.y, y, nx, ny, nz)) continue;
426 point.y = y;
427 point.normalX = nx;
428 point.normalY = ny;
429 point.normalZ = nz;
430 continue;
431 }
432 const float hx = (point.x - originX_) / cellSize_;
433 const float hz = (point.z - originZ_) / cellSize_;
434 point.y = heightmap_->sampleBilinear(hx, hz) * heightScale_;
435 const float left = heightmap_->sampleBilinear(hx - 0.5f, hz) * heightScale_;
436 const float right = heightmap_->sampleBilinear(hx + 0.5f, hz) * heightScale_;
437 const float down = heightmap_->sampleBilinear(hx, hz - 0.5f) * heightScale_;
438 const float up = heightmap_->sampleBilinear(hx, hz + 0.5f) * heightScale_;
439 float nx = -(right - left) / cellSize_;
440 float ny = 1.f;
441 float nz = -(up - down) / cellSize_;
442 const float length = std::sqrt(nx * nx + ny * ny + nz * nz);
443 if (length > 0.f) {
444 nx /= length;
445 ny /= length;
446 nz /= length;
447 }
448 point.normalX = nx;
449 point.normalY = ny;
450 point.normalZ = nz;
451 }
452 return output;
