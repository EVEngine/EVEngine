30ui_editing::EditorResult<void> UiThemeEditor::configureWorkspace(editor::EditorWorkspace& workspace) const {
31 editor::EditorWorkspace candidate = workspace;
32 struct Panel {
33 const char* id;
34 const char* title;
35 const char* region;
36 const char* context;
37 int order;
38 };
39 constexpr Panel panels[] = {
40 {"ui.themes", "Themes", "left", "list", 100},
41 {"ui.preview", "Theme Preview", "center", "preview", 100},
42 {"ui.inspector", "Theme Inspector", "right", "inspector", 100},
43 };
44 for (const auto& panel : panels) {
45 if (!candidate.registerPanel(panel.id, panel.title, panel.region, panel.order) ||
46 !candidate.setPanelCapability(panel.id, "ui.theme") ||
47 !candidate.setPanelContext(panel.id, panel.context))
48 return editorError(ui_editing::EditorStatus::Rejected, "editor.ui-theme.workspace-conflict",
49 "Could not install the UI theme workspace composition");
50 }
51 if (!candidate.activatePanel("ui.preview"))
52 return editorError(ui_editing::EditorStatus::Rejected, "editor.ui-theme.workspace-activate",
53 "Could not activate the UI theme preview panel");
54 workspace = std::move(candidate);
55 return eve::editing::applied<void>();
56}

71ui_editing::EditorResult<void> UiThemeEditor::commit(
72 ui_editing::EditorResult<ui_editing::DomainOperation> operation, std::string label) {
73 if (!operation.ok())
74 return ui_editing::EditorResult<void>::failure(operation.status());
75 editor::TransactionSpec spec;
76 spec.id = editor::TransactionId("ui.theme.tx." + std::to_string(++txSequence_));
77 spec.label = std::move(label);
78 spec.target = editor::TargetId(target_.targetId());
79 spec.baseRevision = target_.revision();
80 auto begun = transactions_.begin(std::move(spec));
81 if (!begun.ok())
82 return editorError(begun.code(), "editor.ui-theme.begin", "Could not begin the theme transaction");
83 auto appended = transactions_.append(std::move(operation).takeValue());
84 if (!appended.ok()) {
85 auto discarded = transactions_.discard();
86 if (!discarded.ok()) discarded.ignore("pending theme transaction already inactive");
87 return ui_editing::EditorResult<void>::failure(appended.status());
88 }
89 auto committed = transactions_.commit();
90 if (!committed.ok())
91 return ui_editing::EditorResult<void>::failure(committed.status());
92 return refreshPreview();
93}

174ui_editing::EditorResult<void> UiThemeEditor::applyPreviewHost(const std::string& hostName) {
175 if (hostName.empty())
176 return editorError(ui_editing::EditorStatus::Rejected, "editor.ui-theme.host",
177 "Preview host name must not be empty");
178 if (preview_.status != ui_editing::EditorStatus::Applied || preview_.documentRevision != target_.revision()) {
179 auto previewed = refreshPreview();

186 host->get().setThemeOverride(preview_.theme);
187 return eve::editing::applied<void>();

212float UiThemeEditor::getColorChannel(const std::string& path, int channel) const {
213 auto value = target_.read(selection(), ui_editing::PropertyPath(path));
214 if (value.state != ui_editing::PropertyReadState::Value) return 0.0f;
215 const auto* array = value.value.getIf<ui_editing::EditorValue::Array>();
216 if (!array || channel < 0 || static_cast<std::size_t>(channel) >= array->size()) return 0.0f;
217 const auto* number = (*array)[static_cast<std::size_t>(channel)].getIf<double>();
218 return number ? static_cast<float>(*number) : 0.0f;
219}
