60#else

62 return failure(DiagnosticCode::InvalidArgument,

64 auto read = [&](const std::filesystem::path& path) -> Result<std::vector<std::uint8_t> > {
65 std::error_code ec;
66 const auto size = std::filesystem::file_size(path, ec);
67 if (ec || size > request.limits.maximumSourceBytes || size > std::size_t(-1))
68 return Result<std::vector<std::uint8_t> >::failure(Diagnostic::error(
69 DiagnosticCode::InvalidArgument, "OpenSCAD input/output is missing or exceeds byte budget",
70 path.string(), {}, "asset.import.scad"));
71 std::ifstream stream(path, std::ios::binary);
72 std::vector<std::uint8_t> bytes(static_cast<std::size_t>(size));
73 if (!bytes.empty())
74 stream.read(reinterpret_cast<char*>(bytes.data()), static_cast<std::streamsize>(bytes.size()));
75 if (!stream)
76 return Result<std::vector<std::uint8_t> >::failure(Diagnostic::error(
77 DiagnosticCode::Failed, "cannot read OpenSCAD input/output", path.string(), {}, "asset.import.scad"));
78 return Result<std::vector<std::uint8_t> >::success(std::move(bytes));
79 };
80 try {
81 const auto input = std::filesystem::absolute(
82 std::filesystem::path(std::u8string(request.sourcePath.begin(), request.sourcePath.end())));
83 auto source = read(input);
84 if (!source) return Output::failure(source.status());
85 static std::atomic<std::uint64_t> sequence{0};
86 const auto directory = std::filesystem::temp_directory_path() /
87 ("eve-scad-" + std::to_string(Poco::Process::id()) + "-" +
88 std::to_string(std::chrono::steady_clock::now().time_since_epoch().count()) + "-" +
89 std::to_string(sequence.fetch_add(1)));
90 if (!std::filesystem::create_directory(directory))
91 return failure(DiagnosticCode::Conflict, "cannot create private OpenSCAD staging directory");

93 std::filesystem::path path;

95 std::error_code ec;

97 }
98 } cleanup{directory};
99 auto utf8 = [](const std::filesystem::path& path) {
100 const auto value = path.u8string();
101 return std::string(reinterpret_cast<const char*>(value.data()), value.size());
102 };
103 auto compileOne = [&](const std::filesystem::path& output,
104 const std::vector<std::string>& extraArgs) -> Result<std::vector<std::uint8_t> > {
105 std::vector<std::string> args = {"--hardwarnings", "--export-format", "binstl", "-o", utf8(output)};
106 args.insert(args.end(), extraArgs.begin(), extraArgs.end());
107 args.push_back(utf8(input));
108 auto process = Poco::Process::launch(request.executable, args);
109 const auto deadline = std::chrono::steady_clock::now() + std::chrono::seconds(request.timeoutSeconds);
110 int exit = -1;
111 while ((exit = process.tryWait()) == -1) {
112 if (std::chrono::steady_clock::now() >= deadline) {
113 Poco::Process::kill(process.id());
114 process.wait();
115 return Result<std::vector<std::uint8_t> >::failure(

118 }
119 std::this_thread::sleep_for(std::chrono::milliseconds(20));
120 }
121 if (exit != 0)
122 return Result<std::vector<std::uint8_t> >::failure(Diagnostic::error(
123 DiagnosticCode::Failed, "OpenSCAD compiler exited with code " + std::to_string(exit),
124 request.sourcePath, {}, "asset.import.scad"));
125 return read(output);
126 };

131 const auto stemName =

133 const auto sidecar = input.parent_path() / (stemName + ".rig.json");
134 std::error_code sidecarEc;
135 if (std::filesystem::is_regular_file(sidecar, sidecarEc)) {
136 auto rig = read(sidecar);
137 if (!rig) return Output::failure(rig.status());
138 prepared.rigBytes = std::move(rig).takeValue();

140 if (!parts) return Output::failure(parts.status());
141 for (const auto& part : parts.value()) {

143 if (!stl) return Output::failure(stl.status());
144 prepared.partStls.emplace(part, std::move(stl).takeValue());
145 }
146 } else {
147 auto stl = compileOne(directory / "model.stl", {});
148 if (!stl) return Output::failure(stl.status());

150 }
151 auto unchanged = read(input);
152 if (!unchanged) return Output::failure(unchanged.status());
153 if (unchanged.value() != prepared.sourceBytes)
154 return failure(DiagnosticCode::Conflict, "SCAD source changed during compilation; retry import");

156 } catch (const Poco::Exception& error) {

158 "cannot run OpenSCAD: " + error.displayText() + "; use --openscad <executable>");
159 } catch (const std::filesystem::filesystem_error& error) {

161 }
162#endif
