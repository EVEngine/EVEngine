67std::string quote(std::string_view value) {
68 std::ostringstream out;
69 out << '"';
70 for (unsigned char c : value) {
71 switch (c) {
72 case '"': out << "\\\""; break;
73 case '\\': out << "\\\\"; break;
74 case '\n': out << "\\n"; break;
75 case '\r': out << "\\r"; break;
76 case '\t': out << "\\t"; break;
77 default:
78 if (c < 0x20)
79 out << "\\u" << std::hex << std::setw(4) << std::setfill('0') << int(c) << std::dec;
80 else
81 out << static_cast<char>(c);
82 }
83 }
84 return out.str() + '"';
85}

87std::string canonicalJson(const eve::json::Value& value) {
88 if (value.isNull()) return "null";
89 if (value.isBool()) return value.asBool() ? "true" : "false";
90 if (value.isNumber()) return value.asString();
91 if (value.isString()) return quote(value.asString());
92 if (value.isArray()) {
93 std::string out = "[";
94 for (size_t i = 0; i < value.size(); ++i) {
95 if (i) out += ',';
96 out += canonicalJson(value.at(i));
97 }
98 return out + ']';
99 }
100 auto keys = value.keys();
101 std::sort(keys.begin(), keys.end());
102 std::string out = "{";
103 for (size_t i = 0; i < keys.size(); ++i) {
104 if (i) out += ',';
105 out += quote(keys[i]) + ':' + canonicalJson(value.get(keys[i].c_str()));
106 }
107 return out + '}';
108}

281void WorkQueue::schedule(std::string_view owner) {
282 int available = slotCount(owner) - runningCount(owner);
283 while (available-- > 0) {
284 ProductionTask* best = nullptr;
285 for (auto& candidate : tasks_) {
286 if (candidate->owner != owner || candidate->state != TaskState::Queued) continue;
287 if (!best || candidate->priority > best->priority ||
288 (candidate->priority == best->priority && candidate->enqueueSequence < best->enqueueSequence))
289 best = candidate.get();
290 }
291 if (!best) break;
292 best->state = TaskState::Running;
293 emit(ProductionEventKind::Started, *best);
294 }
295}

297eve::Result<std::string> WorkQueue::enqueue(std::string_view owner, std::string_view kind, std::string_view product,
298 eve::Value context, double duration, int priority) {
299 if (owner.empty() || kind.empty() || product.empty())
300 return productionBindingFailure<std::string>(eve::DiagnosticCode::InvalidArgument,
301 "work task owner, kind and product are required");
302 if (!context.isObject())
303 return productionBindingFailure<std::string>(eve::DiagnosticCode::InvalidArgument,
304 "work task context must be a Value object", "context");
305 if (!std::isfinite(duration) || duration <= 0.0)
306 return productionBindingFailure<std::string>(eve::DiagnosticCode::InvalidArgument,
307 "work task duration must be finite and positive", "duration");
308 auto durationValue = eve::Duration::fromSeconds(duration);
309 if (!durationValue.ok()) return eve::Result<std::string>::failure(durationValue.status());
310 if (durationValue.value().nanoseconds() <= 0)
311 return productionBindingFailure<std::string>(eve::DiagnosticCode::InvalidArgument,
312 "work task duration must be positive", "duration");
313 auto task = std::make_unique<ProductionTask>();
314 std::ostringstream id;
315 id << "task-" << std::setw(16) << std::setfill('0') << nextTaskId_;
316 task->id = id.str();
317 task->owner = std::string(owner);
318 task->kind = std::string(kind);
319 task->product = std::string(product);
320 task->context = std::move(context);
321 task->duration = std::move(durationValue).takeValue();
322 task->priority = priority;
323 task->enqueueSequence = nextEnqueueSequence_++;
324 const std::string result = task->id;
325 tasks_.push_back(std::move(task));
326 ++nextTaskId_;
327 emit(ProductionEventKind::Enqueued, *tasks_.back());
328 schedule(owner);
329 return eve::Result<std::string>::success(result);
330}

404eve::Result<void> WorkQueue::advance(const eve::SimulationStep& step) {
405 if (step.tick <= tick_)
406 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::PreconditionViolation,
407 "production step tick must be strictly newer"));
408 if (step.delta.nanoseconds() < 0)
409 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
410 "production step duration must be non-negative"));
411
412 tick_ = step.tick;
413 const std::int64_t delta = step.delta.nanoseconds();
414 std::set<std::string> owners;
415 for (auto& task : tasks_) {
416 if (task->state != TaskState::Running) continue;
417 const std::int64_t remaining = task->duration.nanoseconds() - task->progress.nanoseconds();
418 const std::int64_t applied = delta >= remaining ? remaining : delta;
419 task->progress = eve::Duration::fromNanoseconds(task->progress.nanoseconds() + applied);
420 if (task->progress >= task->duration) {
421 task->state = TaskState::Completed;
422 emit(ProductionEventKind::Completed, *task);
423 owners.insert(task->owner);
424 }
425 }
426 for (const auto& owner : owners) schedule(owner);
427 return eve::Result<void>::success();
428}

527eve::Result<void> WorkQueue::restore(std::string_view json) {
528 std::string error;
529 auto doc = eve::json::Document::parse(std::string(json), &error);
530 if (!doc.valid() || !doc.root().isObject()) {
531 return productionBindingFailure<void>(eve::DiagnosticCode::ParseError,
532 error.empty() ? "snapshot must be an object" : error);
533 }
534 WorkQueue candidate(instanceId_);
535 const auto root = doc.root();
536 if (root.getInt("version") != 1 || !parseU64(root.get("nextTaskId"), candidate.nextTaskId_) ||
537 !parseU64(root.get("nextEnqueueSequence"), candidate.nextEnqueueSequence_) ||
538 !parseU64(root.get("nextEventSequence"), candidate.nextEventSequence_) || candidate.nextTaskId_ == 0 ||
539 candidate.nextEnqueueSequence_ == 0 || candidate.nextEventSequence_ == 0 || !root.get("slots").isArray() ||
540 !root.get("events").isArray() || !root.get("tasks").isArray()) {
541 return productionBindingFailure<void>(eve::DiagnosticCode::ParseError, "invalid snapshot counters or arrays");
542 }
543 if (!root.get("tick").isNull()) {
544 uint64_t tick = 0;
545 if (!parseU64(root.get("tick"), tick)) {
546 return productionBindingFailure<void>(eve::DiagnosticCode::ParseError, "invalid snapshot tick", "tick");
547 }
548 candidate.tick_ = eve::SimulationTick(tick);
549 }
550 if (!root.get("revision").isNull()) {
551 uint64_t revision = 0;
552 if (!parseU64(root.get("revision"), revision)) {
553 return productionBindingFailure<void>(eve::DiagnosticCode::ParseError, "invalid snapshot revision",
554 "revision");
555 }
556 candidate.revision_ = eve::Revision(revision);
557 } else {
558 candidate.revision_ = eve::Revision(candidate.nextEventSequence_ - 1);
559 }
560 for (size_t i = 0; i < root.get("slots").size(); ++i) {
561 auto value = root.get("slots").at(i);
562 if (!value.isObject() || !value.get("owner").isString() || !value.get("value").isNumber()) {
563 return productionBindingFailure<void>(eve::DiagnosticCode::ParseError, "invalid slot entry", "slots");
564 }
565 int slots = value.get("value").asInt();
566 auto setSlots = candidate.setSlotCount(value.get("owner").asString(), slots);
567 if (!setSlots.ok()) return eve::Result<void>::failure(setSlots.status());
568 }
569 std::set<std::string> ids;
570 for (size_t i = 0; i < root.get("tasks").size(); ++i) {
571 auto value = root.get("tasks").at(i);
572 auto task = std::make_unique<ProductionTask>();
573 uint64_t sequence = 0;
574 TaskState state = TaskState::Queued;
575 if (!value.isObject() || !value.get("id").isString() || !value.get("owner").isString() ||
576 !value.get("kind").isString() || !value.get("product").isString() ||
577 (!value.get("progressNs").isString() && !value.get("progress").isNumber()) ||
578 (!value.get("durationNs").isString() && !value.get("duration").isNumber()) ||
579 !value.get("priority").isNumber() || !value.get("state").isString() || !value.get("reason").isString() ||
580 !parseU64(value.get("enqueueSequence"), sequence) || !parseState(value.get("state").asString(), state) ||
581 value.get("id").asString().empty() || value.get("owner").asString().empty() ||
582 value.get("kind").asString().empty() || value.get("product").asString().empty() ||
583 !ids.insert(value.get("id").asString()).second) {
584 return productionBindingFailure<void>(eve::DiagnosticCode::ParseError, "invalid task entry", "tasks");
585 }
586 task->id = value.get("id").asString();
587 task->owner = value.get("owner").asString();
588 task->kind = value.get("kind").asString();
589 task->product = value.get("product").asString();
590 auto context = eve::Value::fromJson(canonicalJson(value.get("context")));
591 if (!context.ok()) return eve::Result<void>::failure(context.status());
592 if (!context.value().isObject())
593 return productionBindingFailure<void>(eve::DiagnosticCode::ParseError,
594 "work task context must be an object", "tasks.context");
595 task->context = std::move(context).takeValue();
596 if (!parseDuration(value.get("durationNs"), value.get("duration"), task->duration) ||
597 !parseDuration(value.get("progressNs"), value.get("progress"), task->progress)) {
598 return productionBindingFailure<void>(eve::DiagnosticCode::ParseError, "invalid task duration",
599 "tasks.duration");
600 }
601 task->priority = value.get("priority").asInt();
602 task->state = state;
603 task->enqueueSequence = sequence;
604 task->reason = value.get("reason").asString();
605 if (task->duration.nanoseconds() <= 0 || task->progress.nanoseconds() < 0 || task->progress > task->duration) {
606 return productionBindingFailure<void>(eve::DiagnosticCode::ParseError, "invalid task progress",
607 "tasks.progress");
608 }
609 candidate.tasks_.push_back(std::move(task));
610 }
611 uint64_t previousEventSequence = 0;
612 for (size_t i = 0; i < root.get("events").size(); ++i) {
613 auto value = root.get("events").at(i);
614 ProductionEvent event;
615 if (!value.isObject() || !value.get("kind").isString() || !value.get("owner").isString() ||
616 !value.get("product").isString() || !value.get("reason").isString() || !value.get("taskId").isString() ||
617 !value.get("taskKind").isString() || !parseU64(value.get("sequence"), event.sequence) ||
618 !parseEventKind(value.get("kind").asString(), event.kind) || event.sequence <= previousEventSequence ||
619 event.sequence >= candidate.nextEventSequence_) {
620 return productionBindingFailure<void>(eve::DiagnosticCode::ParseError, "invalid event entry", "events");
621 }
622 previousEventSequence = event.sequence;
623 event.owner = value.get("owner").asString();
624 event.product = value.get("product").asString();
625 event.reason = value.get("reason").asString();
626 event.taskId = value.get("taskId").asString();
627 event.taskKind = value.get("taskKind").asString();
628 if (!value.get("tick").isNull()) {
629 uint64_t tick = 0;
630 if (!parseU64(value.get("tick"), tick)) {
631 return productionBindingFailure<void>(eve::DiagnosticCode::ParseError, "invalid event tick",
632 "events.tick");
633 }
634 event.tick = eve::SimulationTick(tick);
635 }
636 candidate.events_.push_back(std::move(event));
637 }
638 *this = std::move(candidate);
639 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
640}

680 candidate.instanceId_ = candidateEnvelope.instanceId;
681 candidate.revision_ = candidateEnvelope.revision;
682 candidate.tick_ = candidateEnvelope.tick;

685}
