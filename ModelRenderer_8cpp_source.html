91Texture* loadModelTexture(IResourceFactory* gfx, ModelData* model, const std::string& path) {
92 // Imported texture paths are relative to the source model, not the VFS root.
93 // Normalize parent segments (OBJ/MTL commonly uses ../textures/foo.png).
94 std::string uri = model->getUri();
95 if (uri.rfind("file://", 0) == 0) uri.erase(0, 7);
96 // ResourceManager cache keys are normalized VFS paths without a scheme.
97 if (!uri.empty() && uri.find("://") == std::string::npos) {
98 uri = uri.substr(0, uri.find('?'));
99 std::string relative = path;
100 std::replace(relative.begin(), relative.end(), '\\', '/');
101 const auto resolved = (std::filesystem::path(uri).parent_path() / relative).lexically_normal();
102 if (auto* texture = loadExternalTexture(gfx, resolved.generic_string())) return texture;
103 }
104 return loadExternalTexture(gfx, path);
105}

130void extendedLook(IResourceFactory* gfx, ModelData* model, int index, TextureLook& look,
131 const ModelRenderOptions& options) {
132 const auto* scene = model->getScene();
133 if (!scene || index < 0 || unsigned(index) >= scene->mNumMaterials) return;
134 const auto* material = scene->mMaterials[index];
135 aiString alpha;
136 if (material->Get(AI_MATKEY_GLTF_ALPHAMODE, alpha) != AI_SUCCESS) return;
137 look.extendedPbr = true;
138 look.ta = model->getMaterialBaseColorA(index);
139 auto& p = look.pbr;
140 auto scalar = [&](const char* key, unsigned type, unsigned slot, float fallback) {
141 float value = fallback;
142 material->Get(key, type, slot, value);
143 if (!std::isfinite(value)) throw eve::Exception("nonfinite imported material factor %s", key);
144 return value;
145 };
146 auto unit = [&](const char* key, unsigned type, unsigned slot, float fallback) {
147 float value = scalar(key, type, slot, fallback);
148 if (value < 0 || value > 1)
149 std::fprintf(stderr, "[model3d] warning %s=%g clamped to [0,1]\n", key, double(value));
150 return std::clamp(value, 0.f, 1.f);
151 };
152 p.specularFactor = unit(AI_MATKEY_SPECULAR_FACTOR, 1);
153 aiColor3D color(1, 1, 1);
154 material->Get(AI_MATKEY_COLOR_SPECULAR, color);
155 p.specularColor = {color.r, color.g, color.b};
156 color = {0, 0, 0};
157 material->Get(AI_MATKEY_COLOR_EMISSIVE, color);
158 p.emissive = {color.r, color.g, color.b};
159 p.emissiveStrength = scalar(AI_MATKEY_EMISSIVE_INTENSITY, 1);
160 p.ior = scalar(AI_MATKEY_REFRACTI, 1.5f);
161 p.clearcoatFactor = unit(AI_MATKEY_CLEARCOAT_FACTOR, 0);
162 p.clearcoatRoughness = unit(AI_MATKEY_CLEARCOAT_ROUGHNESS_FACTOR, 0);
163 p.anisotropyStrength = unit(AI_MATKEY_ANISOTROPY_FACTOR, 0);
164 p.normalScale = scalar(AI_MATKEY_GLTF_TEXTURE_SCALE(aiTextureType_NORMALS, 0), 1);
165 p.clearcoatNormalScale = scalar(AI_MATKEY_GLTF_TEXTURE_SCALE(aiTextureType_CLEARCOAT, 2), 1);
166 p.occlusionStrength = unit("$tex.file.strength", aiTextureType_LIGHTMAP, 0, 1);
167 int shading = 0;
168 material->Get(AI_MATKEY_SHADING_MODEL, shading);
169 p.unlit = shading == aiShadingMode_Unlit;
170 const std::pair<aiTextureType, unsigned> roles[] = {
171 {aiTextureType_BASE_COLOR, 0}, {aiTextureType_UNKNOWN, 0}, {aiTextureType_NORMALS, 0},
172 {aiTextureType_LIGHTMAP, 0}, {aiTextureType_EMISSIVE, 0}, {aiTextureType_SPECULAR, 0},
173 {aiTextureType_SPECULAR, 1}, {aiTextureType_NONE, 0}, {aiTextureType_CLEARCOAT, 0},
174 {aiTextureType_CLEARCOAT, 1}, {aiTextureType_CLEARCOAT, 2}};
175 for (size_t i = 0; i < 11; i++) {
176 if ((i == 0 && !options.importAlbedo) || ((i == 2 || i == 10) && !options.importNormalMaps)) continue;
177 const auto [type, slot] = roles[i];
178 if (type == aiTextureType_NONE) continue;
179 aiString path;
180 unsigned uv = 0;
181 aiTextureMapMode wrap[2] = {aiTextureMapMode_Wrap, aiTextureMapMode_Wrap};
182 if (material->GetTexture(type, slot, &path, nullptr, &uv, nullptr, nullptr, wrap) != AI_SUCCESS) continue;
183 auto& binding = p.textures[i];
184 if (i == 0 && look.albedo)
185 binding.texture = look.albedo;
186 else if (i == 2 && look.normal)
187 binding.texture = look.normal;
188 else if (path.length > 1 && path.C_Str()[0] == '*')
189 binding.texture = loadEmbeddedTexture(gfx, model, std::stoi(path.C_Str() + 1));
190 else
191 binding.texture = loadModelTexture(gfx, model, path.C_Str());
192 if (!binding.texture) {
193 // Model3D preview preserves the legacy factor-only material when an
194 // external resource is absent. Canonical asset admission remains strict.
195 std::fprintf(stderr, "[model3d] warning: missing material texture %s (model %s); using material factors\n",
196 path.C_Str(), model->getUri().c_str());
197 continue;
198 }
199 binding.texcoord = uv;
200 auto wrapEnum = [](aiTextureMapMode mode) {
201 return mode == aiTextureMapMode_Clamp ? 33071u : mode == aiTextureMapMode_Mirror ? 33648u : 10497u;
202 };
203 binding.wrapS = wrapEnum(wrap[0]);
204 binding.wrapT = wrapEnum(wrap[1]);
205 int filter = 9987;
206 material->Get(AI_MATKEY_GLTF_MAPPINGFILTER_MIN(type, slot), filter);
207 binding.minFilter = filter;
208 filter = 9729;
209 material->Get(AI_MATKEY_GLTF_MAPPINGFILTER_MAG(type, slot), filter);
210 binding.magFilter = filter;
211 aiUVTransform transform;
212 if (material->Get(AI_MATKEY_UVTRANSFORM(type, slot), transform) == AI_SUCCESS) {
213 // First undo the optional postprocess, then Assimp's glTF origin/center conversion.
214 if (model->hasFlippedUvs()) {
215 transform.mRotation = -transform.mRotation;
216 transform.mTranslation.y = -transform.mTranslation.y;
217 }
218 const float r = -transform.mRotation, c = std::cos(r), sn = std::sin(r);
219 const float sx = transform.mScaling.x, sy = transform.mScaling.y;
220 binding.rotation = r;
221 binding.scale = {sx, sy};
222 binding.offset = {transform.mTranslation.x - .5f * sx * (-c + sn + 1),
223 1 - sy - transform.mTranslation.y + .5f * sy * (sn + c - 1)};
224 }
225 }
226 look.albedo = p.textures[0].texture;
227 look.normal = p.textures[2].texture;
228}

230TextureLook materialLook(IResourceFactory *gfx, ModelData *model, int matIndex, const ModelRenderOptions &options,
231 std::unordered_map<int, TextureLook> &cache) {
232 auto it = cache.find(matIndex);
233 if (it != cache.end()) return it->second;
234
235 TextureLook look;
236 if (matIndex >= 0) {
237 look.tr = model->getMaterialBaseColorR(matIndex);
238 look.tg = model->getMaterialBaseColorG(matIndex);
239 look.tb = model->getMaterialBaseColorB(matIndex);
240 look.ta = model->getMaterialBaseColorA(matIndex);
241 look.metallic = model->getMaterialMetallicFactor(matIndex);
242 look.roughness = model->getMaterialRoughnessFactor(matIndex);
243 look.ta *= model->getMaterialOpacity(matIndex);
244 look.alphaMode = model->getMaterialAlphaMode(matIndex);
245 look.alphaCutoff = model->getMaterialAlphaCutoff(matIndex);
246 look.doubleSided = model->getMaterialTwoSided(matIndex);
247 if (options.importAlbedo) {
248 look.albedo = loadTextureSlot(gfx, model, matIndex, "base_color");
249 if (!look.albedo) look.albedo = loadTextureSlot(gfx, model, matIndex, "diffuse");
250 }
251 if (options.importNormalMaps)
252 look.normal = loadTextureSlot(gfx, model, matIndex, "normals");
253 if (options.importHeightMaps)
254 look.height = loadTextureSlot(gfx, model, matIndex, "height");
255 }
256 extendedLook(gfx, model, matIndex, look, options);
257 cache.emplace(matIndex, look);
258 return look;
259}

261Renderable3D *makeRenderable(IResourceFactory *gfx, ModelData *model, int meshIndex, const aiMatrix4x4 &world,
262 const ModelRenderOptions &options, std::unordered_map<int, TextureLook> &cache) {
263 const aiScene *scene = model->getScene();
264 if (!scene || meshIndex < 0 || static_cast<unsigned>(meshIndex) >= scene->mNumMeshes)
265 return nullptr;
266 const aiMesh *ai = scene->mMeshes[meshIndex];
267 if (!ai || ai->mNumVertices == 0 || ai->mNumFaces == 0) return nullptr;
268
269 // Skinned vertex positions and inverse-bind matrices share the model's
270 // bind-pose space. Baking the owning node transform into only the vertices
271 // makes the skin palette apply that transform a second time, separating
272 // modular body parts (notably KayKit heads, armour and limbs). Static
273 // meshes still use the established baked scene transform.
274 const bool bakeWorld = options.bakeWorldTransform && !ai->HasBones();
275 Mesh *mesh = bakeWorld ? gfx->newMeshFromAssimp(*ai, world) : gfx->newMeshFromAssimp(*ai);
276 if (!mesh) return nullptr;
277
278 const int matIndex = model->getMaterialIndex(meshIndex);
279 const TextureLook look = materialLook(gfx, model, matIndex, options, cache);
280
281 Renderable3D *ent = Renderable3D::create();
282 ent->meshRenderer()->visible = true;
283 ent->setMesh(mesh);
284 // Preserve the established Renderable3D inspection API. The Material below
285 // is authoritative for drawing, while these mirrored fields keep imported
286 // models compatible with callers that query meshRenderer()/getTexture().
287 if (look.albedo) ent->setTexture(look.albedo);
288 if (look.normal) ent->setNormalTexture(look.normal);
289 if (look.height) ent->setHeightTexture(look.height);
290 ent->setTint(look.tr, look.tg, look.tb, look.ta);
291 ent->setMetallic(look.metallic);
292 ent->setRoughness(look.roughness);
293 // Keep imported surface semantics in one Material instead of losing glTF
294 // alphaMode/alphaCutoff on the legacy renderer fields.
295 Material *material = new Material();
296 material->setAlbedoTexture(look.albedo);
297 material->setNormalTexture(look.normal);
298 material->setHeightTexture(look.height);
299 material->setTint(look.tr, look.tg, look.tb, look.ta);
300 material->setMetallic(look.metallic);
301 material->setRoughness(look.roughness);
302 material->setDoubleSided(look.doubleSided);
303 material->setAlphaCutoff(look.alphaCutoff);
304 if (look.alphaMode == "MASK")
305 material->setSurfaceMode("masked");
306 else if (look.alphaMode == "BLEND")
307 material->setSurfaceMode("transparent");
308 else
309 material->setSurfaceMode("opaque");
310 if (look.extendedPbr) {
311 auto configured = material->setPbrSurface(look.pbr);
312 if (!configured) throw eve::Exception("%s", configured.error()->message().c_str());
313 for (unsigned channel = 0; channel < AI_MAX_NUMBER_OF_TEXTURECOORDS; channel++) {
314 if (!ai->mTextureCoords[channel]) continue;
315 std::vector<float> values;
316 values.reserve(size_t(ai->mNumVertices) * 2);
317 for (unsigned v = 0; v < ai->mNumVertices; v++) {
318 values.push_back(ai->mTextureCoords[channel][v].x);
319 values.push_back(model->hasFlippedUvs() ? ai->mTextureCoords[channel][v].y
320 : 1 - ai->mTextureCoords[channel][v].y);
321 }
322 auto attached = mesh->setTexcoordSet(channel, values);
323 if (!attached) throw eve::Exception("%s", attached.error()->message().c_str());
324 }
325 }
326 ent->setMaterial(material);
327 return ent;
328}

330void walkNodes(const aiNode *node, const aiMatrix4x4 &parent, IResourceFactory *gfx, ModelData *model,
331 const ModelRenderOptions &options, std::vector<Renderable3D *> &out,
332 std::unordered_map<int, TextureLook> &cache) {
333 if (!node) return;
334 const aiMatrix4x4 world = parent * node->mTransformation;
335 for (unsigned i = 0; i < node->mNumMeshes; ++i) {
336 Renderable3D *ent =
337 makeRenderable(gfx, model, static_cast<int>(node->mMeshes[i]), world, options, cache);
338 if (ent) out.push_back(ent);
339 }
340 for (unsigned c = 0; c < node->mNumChildren; ++c)
341 walkNodes(node->mChildren[c], world, gfx, model, options, out, cache);
342}

344bool findMeshTransform(const aiNode *node, const aiMatrix4x4 &parent, unsigned meshIndex,
345 aiMatrix4x4 &out) {
346 if (!node) return false;
347 const aiMatrix4x4 world = parent * node->mTransformation;
348 for (unsigned i = 0; i < node->mNumMeshes; ++i) {
349 if (node->mMeshes[i] == meshIndex) {
350 out = world;
351 return true;
352 }
353 }
354 for (unsigned c = 0; c < node->mNumChildren; ++c)
355 if (findMeshTransform(node->mChildren[c], world, meshIndex, out)) return true;
356 return false;
357}
