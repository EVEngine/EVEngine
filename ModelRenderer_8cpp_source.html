53 auto *fs = filesystem::Filesystem::create();

56 fd.reset(fs->read(path));

58 }

104TextureLook materialLook(IResourceFactory *gfx, ModelData *model, int matIndex, const ModelRenderOptions &options,
105 std::unordered_map<int, TextureLook> &cache) {
106 auto it = cache.find(matIndex);
107 if (it != cache.end()) return it->second;
108
109 TextureLook look;
110 if (matIndex >= 0) {
111 look.tr = model->getMaterialBaseColorR(matIndex);
112 look.tg = model->getMaterialBaseColorG(matIndex);
113 look.tb = model->getMaterialBaseColorB(matIndex);
114 look.ta = model->getMaterialBaseColorA(matIndex);
115 look.metallic = model->getMaterialMetallicFactor(matIndex);
116 look.roughness = model->getMaterialRoughnessFactor(matIndex);
117 look.ta *= model->getMaterialOpacity(matIndex);
118 look.alphaMode = model->getMaterialAlphaMode(matIndex);
119 look.alphaCutoff = model->getMaterialAlphaCutoff(matIndex);
120 look.doubleSided = model->getMaterialTwoSided(matIndex);
121 if (options.importAlbedo) {
122 look.albedo = loadTextureSlot(gfx, model, matIndex, "base_color");
123 if (!look.albedo) look.albedo = loadTextureSlot(gfx, model, matIndex, "diffuse");
124 }
125 if (options.importNormalMaps)
126 look.normal = loadTextureSlot(gfx, model, matIndex, "normals");
127 if (options.importHeightMaps)
128 look.height = loadTextureSlot(gfx, model, matIndex, "height");
129 }
130 cache.emplace(matIndex, look);
131 return look;
132}

134Renderable3D *makeRenderable(IResourceFactory *gfx, ModelData *model, int meshIndex, const aiMatrix4x4 &world,
135 const ModelRenderOptions &options, std::unordered_map<int, TextureLook> &cache) {
136 const aiScene *scene = model->getScene();
137 if (!scene || meshIndex < 0 || static_cast<unsigned>(meshIndex) >= scene->mNumMeshes)
138 return nullptr;
139 const aiMesh *ai = scene->mMeshes[meshIndex];
140 if (!ai || ai->mNumVertices == 0 || ai->mNumFaces == 0) return nullptr;
141
142 // Skinned vertex positions and inverse-bind matrices share the model's
143 // bind-pose space. Baking the owning node transform into only the vertices
144 // makes the skin palette apply that transform a second time, separating
145 // modular body parts (notably KayKit heads, armour and limbs). Static
146 // meshes still use the established baked scene transform.
147 const bool bakeWorld = options.bakeWorldTransform && !ai->HasBones();
148 Mesh *mesh = bakeWorld ? gfx->newMeshFromAssimp(*ai, world) : gfx->newMeshFromAssimp(*ai);
149 if (!mesh) return nullptr;
150
151 const int matIndex = model->getMaterialIndex(meshIndex);
152 const TextureLook look = materialLook(gfx, model, matIndex, options, cache);
153
154 Renderable3D *ent = Renderable3D::create();
155 ent->meshRenderer()->visible = true;
156 ent->setMesh(mesh);
157 // Preserve the established Renderable3D inspection API. The Material below
158 // is authoritative for drawing, while these mirrored fields keep imported
159 // models compatible with callers that query meshRenderer()/getTexture().
160 if (look.albedo) ent->setTexture(look.albedo);
161 if (look.normal) ent->setNormalTexture(look.normal);
162 if (look.height) ent->setHeightTexture(look.height);
163 ent->setTint(look.tr, look.tg, look.tb, look.ta);
164 ent->setMetallic(look.metallic);
165 ent->setRoughness(look.roughness);
166 // Keep imported surface semantics in one Material instead of losing glTF
167 // alphaMode/alphaCutoff on the legacy renderer fields.
168 Material *material = new Material();
169 material->setAlbedoTexture(look.albedo);
170 material->setNormalTexture(look.normal);
171 material->setHeightTexture(look.height);
172 material->setTint(look.tr, look.tg, look.tb, look.ta);
173 material->setMetallic(look.metallic);
174 material->setRoughness(look.roughness);
175 material->setDoubleSided(look.doubleSided);
176 material->setAlphaCutoff(look.alphaCutoff);
177 if (look.alphaMode == "MASK")
178 material->setSurfaceMode("masked");
179 else if (look.alphaMode == "BLEND")
180 material->setSurfaceMode("transparent");
181 else
182 material->setSurfaceMode("opaque");
183 ent->setMaterial(material);
184 return ent;
185}

187void walkNodes(const aiNode *node, const aiMatrix4x4 &parent, IResourceFactory *gfx, ModelData *model,
188 const ModelRenderOptions &options, std::vector<Renderable3D *> &out,
189 std::unordered_map<int, TextureLook> &cache) {
190 if (!node) return;
191 const aiMatrix4x4 world = parent * node->mTransformation;
192 for (unsigned i = 0; i < node->mNumMeshes; ++i) {
193 Renderable3D *ent =
194 makeRenderable(gfx, model, static_cast<int>(node->mMeshes[i]), world, options, cache);
195 if (ent) out.push_back(ent);
196 }
197 for (unsigned c = 0; c < node->mNumChildren; ++c)
198 walkNodes(node->mChildren[c], world, gfx, model, options, out, cache);
199}

201bool findMeshTransform(const aiNode *node, const aiMatrix4x4 &parent, unsigned meshIndex,
202 aiMatrix4x4 &out) {
203 if (!node) return false;
204 const aiMatrix4x4 world = parent * node->mTransformation;
205 for (unsigned i = 0; i < node->mNumMeshes; ++i) {
206 if (node->mMeshes[i] == meshIndex) {
207 out = world;
208 return true;
209 }
210 }
211 for (unsigned c = 0; c < node->mNumChildren; ++c)
212 if (findMeshTransform(node->mChildren[c], world, meshIndex, out)) return true;
213 return false;
214}

218Renderable3D *buildRenderable(IResourceFactory &gfx, ModelData *model, int meshIndex,
219 const ModelRenderOptions &options) {
220 if (!model) return nullptr;
221 const aiScene *scene = model->getScene();
222 if (!scene || !scene->mRootNode) return nullptr;
223
224 aiMatrix4x4 world;
225 if (!findMeshTransform(scene->mRootNode, aiMatrix4x4(), static_cast<unsigned>(meshIndex),
226 world)) {
227 world = aiMatrix4x4();
228 }
229 std::unordered_map<int, TextureLook> cache;
230 return makeRenderable(&gfx, model, meshIndex, world, options, cache);
231}
