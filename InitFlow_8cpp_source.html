32AdapterDone InitFlow::requestAdapter(InstanceDone &&prev, const wgpu::Surface &compatibleSurface) {
33 wgpu::Instance &instance = prev.instance;
34
35 struct Result {
36 wgpu::Adapter adapter;
37 std::string error;
38 std::atomic<bool> received{false};
39 } result;
40
41 WGPURequestAdapterOptions opts{};
42 opts.compatibleSurface = compatibleSurface.Get();
43 opts.powerPreference = WGPUPowerPreference_HighPerformance;
44
45 WGPURequestAdapterCallbackInfo cbInfo{};
46 cbInfo.nextInChain = nullptr;
47 cbInfo.mode = WGPUCallbackMode_AllowProcessEvents;
48 cbInfo.callback = [](WGPURequestAdapterStatus status, WGPUAdapter a, WGPUStringView msg,
49 void *userdata1, void * /*userdata2*/) {
50 auto *r = static_cast<Result *>(userdata1);
51 if (status == WGPURequestAdapterStatus_Success && a) {
52 r->adapter = wgpu::Adapter(a);
53 } else {
54 r->error = msg.data ? std::string(msg.data, msg.length) : "unknown adapter error";
55 }
56 r->received.store(true);
57 };
58 cbInfo.userdata1 = &result;
59 cbInfo.userdata2 = nullptr;
60
61 wgpuInstanceRequestAdapter(instance.Get(), &opts, cbInfo);
62 while (!result.received.load()) {
63#if defined(__EMSCRIPTEN__)
64 // Yield to the browser event loop so the JS requestAdapter promise can
65 // resolve and fire the callback, then flush it with ProcessEvents.
66 emscripten_sleep(0);
67#endif
68 wgpuInstanceProcessEvents(instance.Get());
69 }
70 if (!result.adapter) {
71 throw Exception("WebGPU: no adapter found (%s)", result.error.c_str());
72 }
73 return AdapterDone{std::move(instance), std::move(result.adapter)};
74}
