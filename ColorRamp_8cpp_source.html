14Rgba8 ColorRamp::sample(float t) const {
15 if (stops.empty()) return {};
16 t = std::clamp(t, 0.f, 1.f);
17 if (t <= stops.front().t) return stops.front().c;
18 if (t >= stops.back().t) return stops.back().c;
19 for (size_t i = 1; i < stops.size(); ++i) {
20 if (t <= stops[i].t) {
21 const Stop &a = stops[i - 1];
22 const Stop &b = stops[i];
23 const float u = (t - a.t) / std::max(1e-6f, b.t - a.t);
24 auto lerp8 = [&](uint8_t x, uint8_t y) {
25 return uint8_t(std::lround(float(x) + (float(y) - float(x)) * u));
26 };
27 return {lerp8(a.c.r, b.c.r), lerp8(a.c.g, b.c.g), lerp8(a.c.b, b.c.b),
28 lerp8(a.c.a, b.c.a)};
29 }
30 }
31 return stops.back().c;
32}
