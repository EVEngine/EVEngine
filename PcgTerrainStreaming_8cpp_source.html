36Result<void> PcgTerrainStreamingAuthority::applyPhotoModeField(const PhotoModeAssignment& assignment) {
37 const auto* value = std::get_if<float>(&assignment.value);
38 if (!value || !std::isfinite(*value) || *value < 0.f || *value > 5000.f) {
39 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
40 "streaming range must be finite in [0,5000]", assignment.field, {},
41 "procgen.pcgTerrainStreaming"));
42 }
43 auto next = state_;
44 if (assignment.field == "m_pcgLoadRange") next.regularRange = *value;
45 else if (assignment.field == "m_pcgImpostorRange") next.impostorRange = *value;
46 else {
47 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
48 "unsupported streaming photo-mode field", assignment.field, {}, "procgen.pcgTerrainStreaming"));
49 }
50 state_ = next;
51 return Result<void>::success();
52}

54int PcgTerrainStreamingAuthority::rangeToChunks(float range) const noexcept {
55 if (!cache_) return 0;
56 const int chunkSize = cache_->asset().getChunkSize();
57 if (chunkSize <= 0) return 0;
58 return static_cast<int>(std::ceil(range / (state_.worldUnitsPerSample * static_cast<float>(chunkSize))));
59}
