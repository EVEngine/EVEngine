38Poco::Net::Context::Ptr tlsContext(const HttpPolicy& policy) {
39 try {
40 using Context = Poco::Net::Context;
41 Context::Ptr context =
42 new Context(Context::CLIENT_USE, "", "", "",
43 policy.verifySsl ? Context::VERIFY_STRICT : Context::VERIFY_NONE, 9, false, "DEFAULT");
44 context->setInvalidCertificateHandler(new Poco::Net::RejectCertificateHandler(false));
45 auto* ssl = context->sslContext();
46 if (SSL_CTX_set_min_proto_version(ssl, TLS1_2_VERSION) != 1)
47 throw Poco::Net::SSLContextException("Cannot require TLS 1.2");
48 if (!policy.verifySsl) return context;
49 if (!policy.caLocation.empty()) {
50 if (policy.caLocation.find('\0') != std::string::npos)
51 throw Poco::Net::SSLContextException("Invalid CA location");
52 const bool directory = Poco::File(policy.caLocation).isDirectory();
53 if (SSL_CTX_load_verify_locations(ssl, directory ? nullptr : policy.caLocation.c_str(),
54 directory ? policy.caLocation.c_str() : nullptr) != 1)
55 throw Poco::Net::SSLContextException("Cannot load custom CA location");
56 } else if (std::getenv("SSL_CERT_FILE") || std::getenv("SSL_CERT_DIR")) {
57 if (SSL_CTX_set_default_verify_paths(ssl) != 1)
58 throw Poco::Net::SSLContextException("Cannot load environment CA locations");
59 } else {
60#ifdef _WIN32
61 if (SSL_CTX_load_verify_store(ssl, "org.openssl.winstore:") != 1)
62 throw Poco::Net::SSLContextException("Cannot load Windows certificate roots");
63#else
64 // The bundled OpenSSL prefix does not supply a root bundle.
65 bool loaded = false;
66 for (const char* file :
67 {"/etc/ssl/certs/ca-certificates.crt", "/etc/pki/tls/certs/ca-bundle.crt", "/etc/ssl/cert.pem"}) {
68 if (Poco::File(file).exists()) {
69 if (SSL_CTX_load_verify_locations(ssl, file, nullptr) != 1)
70 throw Poco::Net::SSLContextException("Cannot load system certificate roots");
71 loaded = true;
72 break;
73 }
74 }
75 if (!loaded && SSL_CTX_set_default_verify_paths(ssl) != 1)
76 throw Poco::Net::SSLContextException("Cannot load default certificate roots");
77#endif
78 }
79 return context;
80 } catch (const Poco::Net::SSLException&) {
81 throw;
82 } catch (const Poco::Exception& ex) {
83 throw Poco::Net::SSLContextException("Cannot initialize HTTPS trust", ex.displayText());
84 }
85}

87HttpResult performHttp(const HttpOptions& options) {
88 HttpResult result;
89 auto& c = result.completion;
90 c.kind = NetKind::Http;
91 c.type = NetEvType::Err;
92 try {
93 Poco::URI uri(options.url);
94 if ((uri.getScheme() != "http" && uri.getScheme() != "https") || uri.getHost().empty())
95 throw Poco::InvalidArgumentException("Expected an absolute HTTP(S) URL");
96 std::unique_ptr<Poco::Net::HTTPClientSession> session;
97 if (uri.getScheme() == "https")
98 session = std::make_unique<Poco::Net::HTTPSClientSession>(uri.getHost(), uri.getPort(),
99 tlsContext(options.policy));
100 else
101 session = std::make_unique<Poco::Net::HTTPClientSession>(uri.getHost(), uri.getPort());
102 session->setTimeout(
103 Poco::Timespan(static_cast<Poco::Timespan::TimeDiff>((std::max)(1, options.policy.timeoutMs)) * 1000));
104 std::string path = uri.getPathAndQuery();
105 if (path.empty()) path = "/";
106 Poco::Net::HTTPRequest request(options.method, path, Poco::Net::HTTPMessage::HTTP_1_1);
107 request.setHost(uri.getHost(), uri.getPort());
108 for (const auto& [key, value] : options.headers) request.set(key, value);
109 if (options.body) request.setContentLength(static_cast<std::streamsize>(options.body->size()));
110 auto& output = session->sendRequest(request);
111 if (options.body && !options.body->empty()) {
112 output.write(options.body->data(), static_cast<std::streamsize>(options.body->size()));
113 if (!output) throw Poco::IOException("Cannot send HTTP body");
114 result.sentBytes = options.body->size();
115 }
116 Poco::Net::HTTPResponse response;
117 auto& input = session->receiveResponse(response);
118 std::ostringstream body;
119 const auto copied = Poco::StreamCopier::copyStream(input, body);
120 if (input.bad() || !body) throw Poco::IOException("Cannot read HTTP response body");
121 if (options.method != Poco::Net::HTTPRequest::HTTP_HEAD &&
122 response.getStatus() != Poco::Net::HTTPResponse::HTTP_NO_CONTENT &&
123 response.getStatus() != Poco::Net::HTTPResponse::HTTP_NOT_MODIFIED &&
124 !response.getChunkedTransferEncoding() && response.hasContentLength() &&
125 copied != response.getContentLength64())
126 throw Poco::IOException("Truncated HTTP response body");
127 const auto bytes = body.str();
128 c.type = NetEvType::HttpResp;
129 c.status = static_cast<int>(response.getStatus());
130 c.bytes = std::make_shared<std::vector<char>>(bytes.begin(), bytes.end());
131 } catch (const Poco::Net::SSLException&) {
132 c.reason = "tls";
133 } catch (const Poco::TimeoutException&) {
134 c.reason = "timeout";
135 } catch (const Poco::Net::DNSException&) {
136 c.reason = "dns";
137 } catch (...) {
138 c.reason = "refused";
139 }
140 return result;
141}

175HttpRuntime::SubmitStatus HttpRuntime::submit(void* origin, HttpOptions options) {
176 std::lock_guard lock(mu_);
177 if (closed_) return SubmitStatus::Closed;
178 try {
179 jobs_.reserve(jobs_.size() + 1);
180 const auto address = reinterpret_cast<std::uintptr_t>(origin);
181 auto& lifetime = origins_[address];
182 if (!lifetime) lifetime = std::make_shared<HttpOrigin>(address);
183 auto job = std::async(std::launch::async, [snapshot = std::move(options)] { return performHttp(snapshot); });
184 jobs_.push_back({lifetime, std::move(job)});
185 return SubmitStatus::Accepted;
186 } catch (...) {
187 return SubmitStatus::Unavailable;
188 }
189}

191HttpRuntime::PollStatus HttpRuntime::take(void* origin, HttpResult& result) {
192 std::lock_guard lock(mu_);
193 if (closed_) return PollStatus::Closed;
194 for (auto job = jobs_.begin(); job != jobs_.end(); ++job) {
195 if (job->lifetime->address == reinterpret_cast<std::uintptr_t>(origin) && job->lifetime->active &&
196 job->future.wait_for(std::chrono::seconds(0)) == std::future_status::ready) {
197 result = job->future.get();
198 result.origin_ = job->lifetime;
199 result.completion.handle = origin;
200 jobs_.erase(job);
201 return PollStatus::Ready;
202 }
203 }
204 return PollStatus::Pending;
205}

207void HttpRuntime::drain(std::vector<HttpResult>& out) {
208 std::lock_guard lock(mu_);
209 for (auto job = jobs_.begin(); job != jobs_.end();) {
210 if (job->future.wait_for(std::chrono::seconds(0)) != std::future_status::ready) {
211 ++job;
212 continue;
213 }
214 auto result = job->future.get();
215 if (job->lifetime->active && !closed_) {
216 result.origin_ = job->lifetime;
217 out.push_back(std::move(result));
218 }
219 job = jobs_.erase(job);
220 }
221}

223HttpRuntime::DeliveryStatus HttpRuntime::prepareDelivery(HttpResult& result) {
224 std::lock_guard lock(mu_);
225 if (closed_ || !result.origin_ || !result.origin_->active) return DeliveryStatus::Retired;
226 result.completion.handle = reinterpret_cast<void*>(result.origin_->address);
227 return DeliveryStatus::Ready;

230void HttpRuntime::cancel(void* origin) {
231 std::lock_guard lock(mu_);
232 auto it = origins_.find(reinterpret_cast<std::uintptr_t>(origin));
233 if (it != origins_.end()) {
234 it->second->active = false;
235 origins_.erase(it);
236 }
237}

239void HttpRuntime::close() {
240 std::vector<Job> pending;
241 {
242 std::lock_guard lock(mu_);
243 closed_ = true;
244 for (auto& [origin, lifetime] : origins_) {
245 (void)origin;
246 lifetime->active = false;
247 }
248 origins_.clear();
249 pending.swap(jobs_);
250 }
251 // Future destruction joins outside the lock. Each job owns its SSL session.
252}

308bool HttpRequest::submit() {
309 const auto runtime = runtime_.lock();
310 if (!runtime) return false;
311 HttpOptions options{method_, url_, headers_, body_, runtime->policy()};
312 if (timeoutMs_ >= 0) options.policy.timeoutMs = timeoutMs_;
313 if (verifySsl_ >= 0) options.policy.verifySsl = verifySsl_ != 0;
314 if (caLocation_) options.policy.caLocation = *caLocation_;
315 return runtime->submit(this, std::move(options)) == HttpRuntime::SubmitStatus::Accepted;
