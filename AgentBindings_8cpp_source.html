26// Script references live only for this synchronous adapter call. Every callback
27// restores the VM stack, including missing methods and thrown script errors.
28class ScriptEnvironment final : public IEnvironment {
29public:
30 explicit ScriptEnvironment(const ssq::Table& environment) : environment_(environment) {}
31 Result<Observation> reset(std::uint64_t seed) override { return invoke("reset", seed, 0, 0); }
32 Result<Observation> step(std::uint32_t action, double dt) override { return invoke("step", 0, action, dt); }
33
34private:
35 Result<Observation> invoke(const char* method, std::uint64_t seed, std::uint32_t action, double dt) {
36 auto vm = environment_.getHandle();
37 StackScope stack(vm);
38 sq_pushobject(vm, environment_.getRaw());
39 sq_pushstring(vm, method, -1);
40 if (SQ_FAILED(sq_get(vm, -2)))
41 return bindingError<Observation>(DiagnosticCode::NotFound, "Missing environment method");
42 sq_pushobject(vm, environment_.getRaw());
43 const bool resetting = std::string_view(method) == "reset";
44 if (resetting) {
45 const auto text = std::to_string(seed);
46 sq_pushstring(vm, text.c_str(), SQInteger(text.size()));
47 } else {
48 sq_pushinteger(vm, SQInteger(action));
49 sq_pushfloat(vm, SQFloat(dt));
50 }
51 if (SQ_FAILED(sq_call(vm, resetting ? 2 : 3, SQTrue, SQFalse))) {
52 sq_getlasterror(vm);
53 const SQChar* text = nullptr;
54 const auto converted = sq_getstring(vm, -1, &text);
55 return bindingError<Observation>(DiagnosticCode::CallbackFailure,
56 SQ_SUCCEEDED(converted) && text ? text : "Environment callback failed");
57 }
58 auto value = script::valueFromSquirrel(vm, -1);
59 if (!value) return Result<Observation>::failure(value.status());
60 return decodeObservation(value.value());
61 }
62 ssq::Table environment_;
63};

65ssq::Table project(HSQUIRRELVM vm, Result<Value> result) {
66 // The common projector has a bounded payload. Check first so oversized
67 // reports return a failure instead of an ok result with a null payload.
68 if (result) {
69 StackScope stack(vm);
70 auto checked = script::pushValue(vm, result.value());
71 if (!checked)
72 return script::projectResult(vm, Result<Value>::failure(checked.status()), [](Value v) { return v; });
73 }
74 return script::projectResult(vm, std::move(result), [](Value v) { return v; });
75}
76Result<Value> inferScript(const ssq::Object& policy, const ssq::Object& observation, Backend backend, bool actionOnly) {
77 auto pv = script::valueFromSquirrel(policy);
78 if (!pv) return Result<Value>::failure(pv.status());
79 auto ov = script::valueFromSquirrel(observation);
80 if (!ov) return Result<Value>::failure(ov.status());
81 auto p = decodePolicy(pv.value());
82 if (!p) return Result<Value>::failure(p.status());
83 auto o = decodeObservation(ov.value());
84 if (!o) return Result<Value>::failure(o.status());
85 auto result = infer(p.value(), o.value(), backend);
86 if (!result) return Result<Value>::failure(result.status());
87 if (actionOnly) {
88 const auto& probabilities = result.value();
89 const auto action = *std::max_element(o.value().legalActions.begin(), o.value().legalActions.end(),
90 [&](auto a, auto b) { return probabilities[a] < probabilities[b]; });
91 return Result<Value>::success(Value(std::int64_t(action)));
92 }
93 Value::Array values;
94 for (auto n : result.value()) values.emplace_back(n);
95 return Result<Value>::success(Value(std::move(values)));
96}
