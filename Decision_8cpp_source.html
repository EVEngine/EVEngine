45template <class Ref, class Proxy, class Release>
46ssq::Table makeOwnedProxy(HSQUIRRELVM vm, eve::Result<Ref>&& reference, Release&& release) {
47 if (!reference) {
48 return eve::script::projectStatusResult(vm, reference.status(), false, false);
49 }
50
51 const Ref ref = std::move(reference).takeValue();
52 auto object = eve::script::makeOwnedSquirrelInstance<Proxy>(vm, std::make_unique<Proxy>(ref));
53 if (!object) {
54 const eve::Status status = object.status();
55 object.ignore("failed to create owned decision proxy");
56 std::invoke(std::forward<Release>(release), ref).ignore("rollback failed owned decision allocation");
57 return eve::script::projectStatusResult(vm, status, false, false);
58 }
59 ssq::Object owned = std::move(object).takeValue();
60 auto result = eve::script::projectStatusResult(vm, eve::Status::success(eve::StatusCode::Applied), true, false);
61 result.set("value", owned);
62 result.set("ownership", std::string("owned"));
63 result.set("ownerEpoch", static_cast<std::int64_t>(ref.ownerEpoch));
64 result.set("handle", static_cast<std::int64_t>(ref.packed()));
65 return result;
66}

157eve::Result<bool> DecisionContext::trigger(const std::string& m, const std::string& t) {
158 if (m.empty() || t.empty())
159 return decisionFailure<bool>(eve::DiagnosticCode::InvalidArgument,
160 "FSM machine and trigger names must not be empty", "trigger");
161 auto s = states_.find(m);
162 auto a = transitions_.find(m);
163 if (s == states_.end())
164 return decisionFailure<bool>(eve::DiagnosticCode::NotFound, "FSM machine has no current state", "machine");
165 if (a == transitions_.end()) return decisionTriggered(false);
166 auto i = a->second.find({s->second, t});
167 if (i == a->second.end()) return decisionTriggered(false);
168 s->second = i->second;
169 return decisionTriggered(true);
170}

175float DecisionContext::utility(const std::string& considerationsCsv) {
176 std::stringstream input(considerationsCsv);
177 std::string consideration;
178 double weightedSum = 0;
179 double totalWeight = 0;
180
181 while (std::getline(input, consideration, ',')) {
182 auto separator = consideration.find(':');
183 if (separator == std::string::npos) {
184 continue;
185 }
186
187 float value = 0;
188 float weight = 0;
189 if (parseNumber(consideration.substr(0, separator), value) &&
190 parseNumber(consideration.substr(separator + 1), weight) && weight > 0) {
191 weightedSum += std::clamp(value, 0.f, 1.f) * weight;
192 totalWeight += weight;
193 }
194 }
195
196 return totalWeight ? float(weightedSum / totalWeight) : 0.f;
197}

199std::string DecisionContext::choose(const std::string& options) {
200 std::stringstream input(options);
201 std::string option;
202 std::string bestName;
203 float bestScore = -1;
204
205 while (std::getline(input, option, ';')) {
206 auto separator = option.find('=');
207 if (separator == std::string::npos || separator == 0) {
208 continue;
209 }
210
211 auto name = option.substr(0, separator);
212 float score = utility(option.substr(separator + 1));
213 if (score > bestScore || (score == bestScore && (bestName.empty() || name < bestName))) {
214 bestScore = score;
215 bestName = name;
216 }
217 }
218
219 return bestName;
220}

275float DecisionContext::sample(const std::string& name, float worldX, float worldY, float fallback) const {
276 auto gridIt = grids_.find(name);
277 if (gridIt == grids_.end()) {
278 return fallback;
279 }
280
281 const Grid& grid = gridIt->second;
282 int cellX = int(std::floor((worldX - grid.originX) / grid.cellSize));
283 int cellY = int(std::floor((worldY - grid.originY) / grid.cellSize));

285 return fallback;
286 }
287
288 return grid.values[size_t(cellY) * grid.width + cellX];

290std::string DecisionContext::snapshotJson() const {
291 std::ostringstream o;
292 o << std::setprecision(9) << "{\"version\":1,\"boards\":{";
293 bool a = true;
294 for (auto& [b, vs] : boards_) {
295 if (!a) o << ',';
296 a = false;
297 o << quote(b) << ":{";
298 bool f = true;
299 for (auto& [k, v] : vs) {
300 if (!f) o << ',';
301 f = false;
302 o << quote(k) << ':' << v;
303 }
304 o << '}';
305 }
306 o << "},\"states\":{";
307 a = true;
308 for (auto& [m, s] : states_) {
309 if (!a) o << ',';
310 a = false;
311 o << quote(m) << ':' << quote(s);
312 }
313 o << "},\"transitions\":[";
314 a = true;
315 for (auto& [m, ts] : transitions_)
316 for (auto& [key, to] : ts) {
317 if (!a) o << ',';
318 a = false;
319 o << "[" << quote(m) << ',' << quote(key.first) << ',' << quote(key.second) << ',' << quote(to) << "]";
320 }
321 o << "],\"grids\":[";
322 a = true;
323 for (auto& [n, g] : grids_) {
324 if (!a) o << ',';
325 a = false;
326 o << "{\"name\":" << quote(n) << ",\"w\":" << g.width << ",\"h\":" << g.height << ",\"cell\":" << g.cellSize
327 << ",\"ox\":" << g.originX << ",\"oy\":" << g.originY << ",\"values\":[";
328 for (size_t i = 0; i < g.values.size(); ++i) {
329 if (i) o << ',';
330 o << g.values[i];
331 }
332 o << "]}";
333 }
334 return o.str() + "]}";
335}
336eve::Result<void> DecisionContext::restoreJson(const std::string& j) {
337 std::string e;
338 auto d = eve::json::Document::parse(j, &e);
339 if (!d.valid())
340 return decisionFailure<void>(eve::DiagnosticCode::ParseError, e.empty() ? "invalid decision snapshot JSON" : e,
341 "snapshot");
342 if (!d.root().isObject()) return snapshotFailure("snapshot", "decision snapshot must be a JSON object");
343 const auto version = d.root().get("version");
344 if (!version.isIntegerLiteral() || version.asInt() != 1)
345 return decisionFailure<void>(eve::DiagnosticCode::UnknownVersion, "unsupported decision snapshot version",
346 "version");
347 DecisionContext n;
348 auto b = d.root().get("boards");
349 if (!b.isObject()) return snapshotFailure("boards", "decision snapshot boards must be an object");
350 for (auto& bn : b.keys()) {
351 auto v = b.get(bn.c_str());
352 if (!v.isObject()) return snapshotFailure("boards." + bn, "blackboard must be an object");
353 for (auto& k : v.keys()) {
354 const auto value = v.get(k.c_str());
355 if (!isScalar(value))
356 return snapshotFailure("boards." + bn + "." + k, "blackboard value must be a JSON scalar");
357 auto result = n.set(bn, k, canonicalize(value));
358 if (!result.ok()) return eve::Result<void>::failure(result.status());
359 }
360 }
361 auto st = d.root().get("states");
362 if (!st.isObject()) return snapshotFailure("states", "decision snapshot states must be an object");
363 for (auto& m : st.keys()) {
364 const auto state = st.get(m.c_str());
365 if (!state.isString()) return snapshotFailure("states." + m, "FSM state must be a string");
366 auto result = n.setState(m, state.asString());
367 if (!result.ok()) return eve::Result<void>::failure(result.status());
368 }
369 auto tr = d.root().get("transitions");
370 if (!tr.isArray()) return snapshotFailure("transitions", "decision snapshot transitions must be an array");
371 for (size_t i = 0; i < tr.size(); ++i) {
372 auto v = tr.at(i);
373 if (!v.isArray() || v.size() != 4)
374 return snapshotFailure("transitions[" + std::to_string(i) + "]",
375 "FSM transition must contain four strings");
376 for (size_t field = 0; field < 4; ++field)
377 if (!v.at(field).isString())
378 return snapshotFailure("transitions[" + std::to_string(i) + "]",
379 "FSM transition must contain four strings");
380 auto result = n.addTransition(v.at(0).asString(), v.at(1).asString(), v.at(2).asString(), v.at(3).asString());
381 if (!result.ok()) return eve::Result<void>::failure(result.status());
382 }
383 auto gs = d.root().get("grids");
384 if (!gs.isArray()) return snapshotFailure("grids", "decision snapshot grids must be an array");
385 for (size_t i = 0; i < gs.size(); ++i) {
386 auto v = gs.at(i);
387 if (!v.isObject())
388 return snapshotFailure("grids[" + std::to_string(i) + "]", "influence grid must be an object");
389 int w = v.getInt("w"), h = v.getInt("h");
390 const auto name = v.get("name");
391 const auto cell = v.get("cell");
392 const auto ox = v.get("ox");
393 const auto oy = v.get("oy");
394 if (!name.isString() || !cell.isNumber() || !ox.isNumber() || !oy.isNumber())
395 return snapshotFailure("grids[" + std::to_string(i) + "]", "influence grid fields are malformed");
396 auto grid =
397 n.newGrid(name.asString(), w, h, float(cell.asDouble()), float(ox.asDouble()), float(oy.asDouble()));
398 if (!grid.ok()) return eve::Result<void>::failure(grid.status());
399 auto vals = v.get("values");
400 if (!vals.isArray() || vals.size() != size_t(w) * h)
401 return snapshotFailure("grids[" + std::to_string(i) + "].values",
402 "influence grid values have the wrong size");
403 for (size_t z = 0; z < vals.size(); ++z) {
404 const auto value = vals.at(z);
405 if (!value.isNumber() || !std::isfinite(value.asDouble()))
406 return snapshotFailure("grids[" + std::to_string(i) + "].values[" + std::to_string(z) + "]",
407 "influence grid cell must be a finite number");
408 auto result = n.setCell(name.asString(), int(z % size_t(w)), int(z / size_t(w)), float(value.asDouble()));
409 if (!result.ok()) return eve::Result<void>::failure(result.status());
410 }
411 }
412 boards_ = std::move(n.boards_);
413 states_ = std::move(n.states_);
414 transitions_ = std::move(n.transitions_);
415 grids_ = std::move(n.grids_);
416 return decisionApplied();
417}
