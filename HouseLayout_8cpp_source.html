35graphics::Texture *textureFromEmbedded(graphics::Graphics *gfx, const aiTexture *source) {
36 if (!source || !source->pcData) return nullptr;
37 if (source->mHeight == 0) {
38 data::ByteData bytes(source->pcData, static_cast<size_t>(source->mWidth));
39 std::unique_ptr<image::ImageData> decoded(image::Image::create()->newImageData(&bytes));
40 return gfx->newTexture(decoded.get(), graphics::TextureCreateInfo::withMipmaps(true));
41 }
42 std::vector<uint8_t> rgba(size_t(source->mWidth) * size_t(source->mHeight) * 4);
43 for (size_t pixel = 0; pixel < size_t(source->mWidth) * size_t(source->mHeight); ++pixel) {
44 rgba[pixel * 4 + 0] = source->pcData[pixel].r;
45 rgba[pixel * 4 + 1] = source->pcData[pixel].g;
46 rgba[pixel * 4 + 2] = source->pcData[pixel].b;
47 rgba[pixel * 4 + 3] = source->pcData[pixel].a;
48 }
49 return gfx->newTexture(int(source->mWidth), int(source->mHeight), rgba.data(),
50 graphics::TextureCreateInfo::withMipmaps(true));
51}

53graphics::Texture *textureFromFile(graphics::Graphics *gfx, const std::string &modelPath,
54 const std::string &texturePath) {
55 if (texturePath.empty()) return nullptr;
56 std::filesystem::path resolved(texturePath);
57 if (resolved.is_relative()) resolved = std::filesystem::path(modelPath).parent_path() / resolved;
58 try {
59 if (std::filesystem::is_regular_file(resolved)) {
60 std::ifstream input(resolved, std::ios::binary | std::ios::ate);
61 const std::streamsize size = input.tellg();
62 if (size <= 0) return nullptr;
63 input.seekg(0, std::ios::beg);
64 std::vector<uint8_t> bytes(static_cast<size_t>(size));
65 if (!input.read(reinterpret_cast<char *>(bytes.data()), size)) return nullptr;
66 data::ByteData source(bytes.data(), bytes.size());
67 std::unique_ptr<image::ImageData> decoded(
68 image::Image::create()->newImageData(&source));
69 return gfx->newTexture(decoded.get(), graphics::TextureCreateInfo::withMipmaps(true));
70 }
71 return gfx->newTextureFromFile(resolved.lexically_normal().string());
72 } catch (...) {
73 return nullptr;
74 }
75}

77graphics::Texture *assimpTexture(graphics::Graphics *gfx, const aiScene *scene,
78 const aiMaterial *material, const std::string &modelPath,
79 aiTextureType primary, aiTextureType fallback) {
80 if (!scene || !material) return nullptr;
81 aiString texturePath;
82 if (material->GetTexture(primary, 0, &texturePath) != AI_SUCCESS &&
83 material->GetTexture(fallback, 0, &texturePath) != AI_SUCCESS)
84 return nullptr;
85 const std::string path = texturePath.C_Str();
86 if (path.empty()) return nullptr;
87 if (path.front() == '*') {
88 const int index = std::atoi(path.c_str() + 1);
89 if (index < 0 || static_cast<unsigned>(index) >= scene->mNumTextures) return nullptr;
90 try {
91 return textureFromEmbedded(gfx, scene->mTextures[index]);
92 } catch (...) {
93 return nullptr;
94 }
95 }
96 return textureFromFile(gfx, modelPath, path);
97}

117ModelHandle loadModel(model3d::Model3D *models, const std::string &path) {
118 if (!std::filesystem::is_regular_file(std::filesystem::path(path))) {
119 // The module hands back the cache-owned resource, so pin it: this layout keeps
120 // using the model after the cache entry may have been unloaded.
121 model3d::ModelData *cached = models->newModelDataFromFile(path);
122 if (cached == nullptr) return {};
123 auto pinned = eve::ResourceManager::getInstance().pin(cached);
124 if (!pinned.ok()) return {};
125 ModelHandle handle;
126 handle.pinned = std::move(pinned).takeValue();
127 return handle;
128 }
129 std::ifstream input(path, std::ios::binary | std::ios::ate);
130 if (!input) throw std::runtime_error("cannot open model: " + path);
131 const std::streamsize size = input.tellg();
132 if (size <= 0) throw std::runtime_error("model is empty: " + path);
133 input.seekg(0, std::ios::beg);
134 std::vector<uint8_t> bytes(static_cast<size_t>(size));
135 if (!input.read(reinterpret_cast<char *>(bytes.data()), size))
136 throw std::runtime_error("cannot read model: " + path);
137 data::ByteData source(bytes.data(), bytes.size());
138 ModelHandle handle;
139 handle.owned.reset(models->newModelData(&source, std::filesystem::path(path).extension().string()));
140 return handle;
141}

212eve::Result<void> HouseLayout::validate(const HouseComponentLibrary &library) const {
213 std::unordered_set<std::string> occupied;
214 std::unordered_set<std::string> floorCells, roofCells;
215 std::vector<std::tuple<int, int, int>> floors;
216 auto cellKey = [](int x, int y, int z) {
217 return std::to_string(x) + ":" + std::to_string(y) + ":" + std::to_string(z);
218 };
219 bool entrance = false, roof = false;
220 for (const auto &i : instances) {
221 const auto component = library.find(i.componentId);
222 if (!component)
223 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::NotFound,
224 "unknown component: " + i.componentId,
225 "componentId", {}, "housegen.layout"));
226 const HouseComponent &c = component->get();
227 if (i.rotationDeg % 90 != 0)
228 return eve::Result<void>::failure(eve::Diagnostic::error(
229 eve::DiagnosticCode::InvalidArgument, "non-cardinal rotation", "rotationDeg", {}, "housegen.layout"));
230 const bool quarter = (i.rotationDeg / 90) % 2 != 0;
231 const int w = quarter ? c.depth : c.width, d = quarter ? c.width : c.depth;
232 for (int y = 0; y < d; ++y) for (int x = 0; x < w; ++x) {
233 // Boundary cells legitimately carry two perpendicular wall modules at corners.
234 const std::string orientation = (c.category == "wall" || c.category == "door")
235 ? ":" + std::to_string((i.rotationDeg % 360 + 360) % 360)
236 : "";
237 const std::string key = std::to_string(i.x + x) + ":" + std::to_string(i.y + y) + ":" +
238 std::to_string(i.z) + ":" + c.category + orientation;
239 if (!occupied.insert(key).second)
240 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::Conflict,
241 "overlapping " + c.category + " components",
242 {}, {}, "housegen.layout"));
243 if (c.category == "floor") {
244 floorCells.insert(cellKey(i.x + x, i.y + y, i.z));
245 floors.emplace_back(i.x + x, i.y + y, i.z);
246 } else if (c.category == "roof") {
247 roofCells.insert(cellKey(i.x + x, i.y + y, i.z));
248 }
249 }
250 entrance = entrance || (c.category == "door" && i.z == 0);
251 roof = roof || c.category == "roof";
252 }
253 if (!entrance)
254 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
255 "house has no entrance", {}, {}, "housegen.layout"));
256 if (!roof)
257 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
258 "house has no roof", {}, {}, "housegen.layout"));
259 for (const auto &[x, y, z] : floors) {
260 if (z > 0 && !floorCells.contains(cellKey(x, y, z - 1))) {
261 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
262 "upper floor has no structural support", {}, {},
263 "housegen.layout"));
264 }
265 if (!floorCells.contains(cellKey(x, y, z + 1)) &&
266 !roofCells.contains(cellKey(x, y, z + 1))) {
267 return eve::Result<void>::failure(eve::Diagnostic::error(
268 eve::DiagnosticCode::InvalidArgument, "floor cell has no roof coverage", {}, {}, "housegen.layout"));
269 }
270 }
271 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
272}

275 const HouseComponentLibrary &library) const {
276 std::vector<ecs::EntityHandle> entities;
277 const auto destroyCreated = [&entities]() noexcept {
278 for (const auto &handle : entities) {
279 if (auto *entity = ecs::try_get(handle)) ecs::DestroyEntity(entity);
280 }
281 entities.clear();
282 };
283 std::unordered_map<std::string, ModelHandle> data;
284 struct CachedPart {
285 graphics::Mesh *mesh = nullptr;
286 graphics::Texture *texture = nullptr;
287 graphics::Texture *normalTexture = nullptr;
288 graphics::Texture *heightTexture = nullptr;
289 float r = 1.f, g = 1.f, b = 1.f, a = 1.f;
290 float metallic = 0.f, roughness = 0.72f;
291 float parallaxScale = 0.f, parallaxMinLayers = 8.f, parallaxMaxLayers = 32.f;
292 float cellBombScale = 4.f, cellBombStrength = 0.f, cellBombRotation = 1.f;
293 };
294 std::unordered_map<std::string, std::vector<CachedPart>> parts;
295 try {
296 for (const auto &i : instances) {
297 const auto component = library.find(i.componentId);
298 if (!component) {
299 destroyCreated();
300 return eve::Result<std::vector<ecs::EntityHandle>>::failure(
301 eve::Diagnostic::error(eve::DiagnosticCode::NotFound, "unknown component: " + i.componentId,
302 "componentId", {}, "housegen.layout"));
303 }
304 const HouseComponent &c = component->get();
305 // Loads once per path: the handle either owns the decode or pins the cache entry.
306 auto modelIt = data.find(c.modelPath);
307 if (modelIt == data.end()) modelIt = data.emplace(c.modelPath, loadModel(&models, c.modelPath)).first;
308 model3d::ModelData *model = modelIt->second.get();
309 // Material overrides belong to a component, so two components may safely reuse the
310 // same GLB with different architectural finishes.
311 auto &cached = parts[c.id];
312 if (cached.empty()) {
313 const aiScene *scene = model->getScene();
314 if (!scene || !scene->mRootNode) throw std::runtime_error("model has no scene root: " + c.modelPath);
315 std::function<void(const aiNode *, const aiMatrix4x4 &)> walk =
316 [&](const aiNode *node, const aiMatrix4x4 &parent) {
317 const aiMatrix4x4 world = parent * node->mTransformation;
318 for (unsigned ni = 0; ni < node->mNumMeshes; ++ni) {
319 const unsigned mi = node->mMeshes[ni];
320 if (mi >= scene->mNumMeshes || !scene->mMeshes[mi]) continue;
321 const aiMesh *source = scene->mMeshes[mi];
322 CachedPart part;
323 part.mesh = gfx.newMeshFromAssimp(*source, world);
324 if (scene->mMaterials && source->mMaterialIndex < scene->mNumMaterials) {
325 const aiMaterial *material = scene->mMaterials[source->mMaterialIndex];
326 aiColor4D color(1.f, 1.f, 1.f, 1.f);
327 material->Get(AI_MATKEY_COLOR_DIFFUSE, color);
328 material->Get(AI_MATKEY_BASE_COLOR, color);
329 part.r = color.r; part.g = color.g; part.b = color.b; part.a = color.a;
330 material->Get(AI_MATKEY_METALLIC_FACTOR, part.metallic);
331 material->Get(AI_MATKEY_ROUGHNESS_FACTOR, part.roughness);
332 part.texture = assimpTexture(&gfx, scene, material, c.modelPath,
333 aiTextureType_BASE_COLOR, aiTextureType_DIFFUSE);
334 part.normalTexture = assimpTexture(&gfx, scene, material, c.modelPath,
335 aiTextureType_NORMALS, aiTextureType_NORMAL_CAMERA);
336 part.heightTexture = assimpTexture(&gfx, scene, material, c.modelPath,
337 aiTextureType_HEIGHT, aiTextureType_DISPLACEMENT);
338 }
339 if (c.material.hasBaseColor) {
340 part.r = c.material.baseColorR;
341 part.g = c.material.baseColorG;
342 part.b = c.material.baseColorB;
343 part.a = c.material.baseColorA;
344 }
345 if (!c.material.baseColorTexture.empty())
346 part.texture = textureFromFile(&gfx, c.modelPath, c.material.baseColorTexture);
347 if (!c.material.normalTexture.empty())
348 part.normalTexture = textureFromFile(&gfx, c.modelPath, c.material.normalTexture);
349 if (!c.material.heightTexture.empty())
350 part.heightTexture = textureFromFile(&gfx, c.modelPath, c.material.heightTexture);
351 if (c.material.hasMetallic) part.metallic = c.material.metallic;
352 if (c.material.hasRoughness) part.roughness = c.material.roughness;
353 part.parallaxScale = c.material.parallaxScale;
354 part.parallaxMinLayers = c.material.parallaxMinLayers;
355 part.parallaxMaxLayers = c.material.parallaxMaxLayers;
356 part.cellBombScale = c.material.cellBombScale;
357 part.cellBombStrength = c.material.cellBombStrength;
358 part.cellBombRotation = c.material.cellBombRotation;
359 cached.push_back(part);
360 }
361 for (unsigned child = 0; child < node->mNumChildren; ++child)
362 walk(node->mChildren[child], world);
363 };
364 walk(scene->mRootNode, aiMatrix4x4());
365 }
366 for (const auto &part : cached) {
367 auto *e = graphics::Renderable3D::create();
368 if (!e) {
369 destroyCreated();
370 return eve::Result<std::vector<ecs::EntityHandle>>::failure(
371 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "failed to create Renderable3D entity", {},
372 {}, "housegen.layout"));
373 }
374 e->setMesh(part.mesh);
375 e->setPosition(i.x * moduleSize, i.z * floorHeight, i.y * moduleSize);
376 e->setYaw(float(i.rotationDeg) * 3.14159265358979323846f / 180.f);
377 e->setTint(part.r, part.g, part.b, part.a);
378 if (part.texture) e->setTexture(part.texture);
379 if (part.normalTexture) e->setNormalTexture(part.normalTexture);
380 if (part.heightTexture) e->setHeightTexture(part.heightTexture);
381 e->setMetallic(part.metallic);
382 e->setRoughness(part.roughness);
383 e->setTexCellBomb(part.cellBombScale, part.cellBombStrength,
384 part.cellBombRotation);
385 if (part.heightTexture && part.parallaxScale > 0.f)
386 e->setParallax(part.parallaxScale, part.parallaxMinLayers,
387 part.parallaxMaxLayers);
388 entities.push_back(ecs::handle_of(e));
389 }
390 }
391 } catch (const std::exception &e) {
392 destroyCreated();
393 return eve::Result<std::vector<ecs::EntityHandle>>::failure(
394 eve::Diagnostic::error(eve::DiagnosticCode::Failed, e.what(), {}, {}, "housegen.layout"));
395 } catch (...) {
396 destroyCreated();
397 return eve::Result<std::vector<ecs::EntityHandle>>::failure(eve::Diagnostic::error(
398 eve::DiagnosticCode::Failed, "house layout instantiation failed", {}, {}, "housegen.layout"));
399 }
400 return eve::Result<std::vector<ecs::EntityHandle>>::success(std::move(entities),
401 eve::Status::success(eve::StatusCode::Applied));
402}
