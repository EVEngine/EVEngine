41graphics::Texture *textureFromEmbedded(graphics::Graphics *gfx, const aiTexture *source) {
42 if (!source || !source->pcData) return nullptr;
43 if (source->mHeight == 0) {
44 data::ByteData bytes(source->pcData, static_cast<size_t>(source->mWidth));
45 std::unique_ptr<image::ImageData> decoded(image::Image::create()->newImageData(&bytes));
46 return gfx->newTexture(decoded.get(), graphics::TextureCreateInfo::withMipmaps(true));
47 }
48 std::vector<uint8_t> rgba(size_t(source->mWidth) * size_t(source->mHeight) * 4);
49 for (size_t pixel = 0; pixel < size_t(source->mWidth) * size_t(source->mHeight); ++pixel) {
50 rgba[pixel * 4 + 0] = source->pcData[pixel].r;
51 rgba[pixel * 4 + 1] = source->pcData[pixel].g;
52 rgba[pixel * 4 + 2] = source->pcData[pixel].b;
53 rgba[pixel * 4 + 3] = source->pcData[pixel].a;
54 }
55 return gfx->newTexture(int(source->mWidth), int(source->mHeight), rgba.data(),
56 graphics::TextureCreateInfo::withMipmaps(true));
57}

59graphics::Texture *textureFromFile(graphics::Graphics *gfx, const std::string &modelPath,
60 const std::string &texturePath) {
61 if (texturePath.empty()) return nullptr;
62 std::filesystem::path resolved(texturePath);
63 if (resolved.is_relative()) resolved = std::filesystem::path(modelPath).parent_path() / resolved;
64 try {
65 if (std::filesystem::is_regular_file(resolved)) {
66 std::ifstream input(resolved, std::ios::binary | std::ios::ate);
67 const std::streamsize size = input.tellg();
68 if (size <= 0) return nullptr;
69 input.seekg(0, std::ios::beg);
70 std::vector<uint8_t> bytes(static_cast<size_t>(size));
71 if (!input.read(reinterpret_cast<char *>(bytes.data()), size)) return nullptr;
72 data::ByteData source(bytes.data(), bytes.size());
73 std::unique_ptr<image::ImageData> decoded(
74 image::Image::create()->newImageData(&source));
75 return gfx->newTexture(decoded.get(), graphics::TextureCreateInfo::withMipmaps(true));
76 }
77 return gfx->newTextureFromFile(resolved.lexically_normal().string());
78 } catch (...) {
79 return nullptr;
80 }

83graphics::Texture *assimpTexture(graphics::Graphics *gfx, const aiScene *scene,
84 const aiMaterial *material, const std::string &modelPath,
85 aiTextureType primary, aiTextureType fallback) {
86 if (!scene || !material) return nullptr;
87 aiString texturePath;
88 if (material->GetTexture(primary, 0, &texturePath) != AI_SUCCESS &&
89 material->GetTexture(fallback, 0, &texturePath) != AI_SUCCESS)
90 return nullptr;
91 const std::string path = texturePath.C_Str();
92 if (path.empty()) return nullptr;
93 if (path.front() == '*') {
94 const int index = std::atoi(path.c_str() + 1);
95 if (index < 0 || static_cast<unsigned>(index) >= scene->mNumTextures) return nullptr;
96 try {
97 return textureFromEmbedded(gfx, scene->mTextures[index]);
98 } catch (...) {
99 return nullptr;
100 }
101 }
102 return textureFromFile(gfx, modelPath, path);
103}

137eve::Result<void> HouseLayout::fromJson(std::string_view json) {
138 using eve::json::Value;
139 std::string parseError;
140 const eve::json::Document doc = eve::json::Document::parse(std::string(json), &parseError);
141 if (!doc.valid())
142 return failure<void>(eve::DiagnosticCode::ParseError,
143 parseError.empty() ? "invalid house layout JSON" : parseError);
144 const Value o = doc.root();
145 if (!o.isObject()) return failure<void>(eve::DiagnosticCode::ParseError, "layout must be an object");
146
147 HouseLayout parsed;
148 parsed.seed = static_cast<unsigned>(o.getInt("seed", 1));
149 parsed.moduleSize = o.getFloat("moduleSize", 1.f);
150 parsed.floorHeight = o.getFloat("floorHeight", 3.f);
151 parsed.footprintStyle = o.getString("footprintStyle", "rectangle");
152 parsed.roofStyle = o.getString("roofStyle", "gable");
153 parsed.entranceSide = o.getString("entranceSide", "north");
154
155 const Value instances = o.get("instances");
156 if (!instances.isArray())
157 return failure<void>(eve::DiagnosticCode::ParseError, "layout has no instances", "instances");
158 for (size_t i = 0; i < instances.size(); ++i) {
159 const Value v = instances.at(i);
160 // componentId and the cell coordinates are required, not defaulted.
161 if (!v.has("componentId") || !v.has("x") || !v.has("y") || !v.has("z"))
162 return failure<void>(eve::DiagnosticCode::ParseError, "instance needs componentId, x, y and z",
163 "instances");
164 parsed.instances.push_back({v.getString("componentId"), v.getInt("x"), v.getInt("y"),
165 v.getInt("z"), v.getInt("rotationDeg", 0)});
166 }
167
168 const Value rooms = o.get("rooms");
169 for (size_t i = 0; i < rooms.size(); ++i) {
170 const Value v = rooms.at(i);
171 if (!v.has("type") || !v.has("x") || !v.has("y") || !v.has("width") || !v.has("depth"))
172 return failure<void>(eve::DiagnosticCode::ParseError, "room needs type, x, y, width and depth", "rooms");
173 parsed.rooms.push_back({v.getString("type"), v.getInt("x"), v.getInt("y"),
174 v.getInt("width"), v.getInt("depth")});
175 }
176
177 parsed.diagnostics = o.getStringArray("diagnostics");
178 *this = std::move(parsed);
179 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));

182eve::Result<void> HouseLayout::validate(const HouseComponentLibrary &library) const {
183 std::unordered_set<std::string> occupied;
184 std::unordered_set<std::string> floorCells, roofCells;
185 std::vector<std::tuple<int, int, int>> floors;
186 auto cellKey = [](int x, int y, int z) {
187 return std::to_string(x) + ":" + std::to_string(y) + ":" + std::to_string(z);
188 };
189 bool entrance = false, roof = false;
190 for (const auto &i : instances) {
191 const auto component = library.find(i.componentId);
192 if (!component)
193 return failure<void>(eve::DiagnosticCode::NotFound, "unknown component: " + i.componentId, "componentId");
194 const HouseComponent &c = component->get();
195 if (i.rotationDeg % 90 != 0)
196 return failure<void>(eve::DiagnosticCode::InvalidArgument, "non-cardinal rotation", "rotationDeg");
197 const bool quarter = (i.rotationDeg / 90) % 2 != 0;
198 const int w = quarter ? c.depth : c.width, d = quarter ? c.width : c.depth;
199 for (int y = 0; y < d; ++y) for (int x = 0; x < w; ++x) {
200 // Boundary cells legitimately carry two perpendicular wall modules at corners.
201 const std::string orientation = (c.category == "wall" || c.category == "door")
202 ? ":" + std::to_string((i.rotationDeg % 360 + 360) % 360)
203 : "";
204 const std::string key = std::to_string(i.x + x) + ":" + std::to_string(i.y + y) + ":" +
205 std::to_string(i.z) + ":" + c.category + orientation;
206 if (!occupied.insert(key).second)
207 return failure<void>(eve::DiagnosticCode::Conflict, "overlapping " + c.category + " components");
208 if (c.category == "floor") {
209 floorCells.insert(cellKey(i.x + x, i.y + y, i.z));
210 floors.emplace_back(i.x + x, i.y + y, i.z);
211 } else if (c.category == "roof") {
212 roofCells.insert(cellKey(i.x + x, i.y + y, i.z));
213 }
214 }
215 entrance = entrance || (c.category == "door" && i.z == 0);
216 roof = roof || c.category == "roof";
217 }
218 if (!entrance) return failure<void>(eve::DiagnosticCode::InvalidArgument, "house has no entrance");
219 if (!roof) return failure<void>(eve::DiagnosticCode::InvalidArgument, "house has no roof");
220 for (const auto &[x, y, z] : floors) {
221 if (z > 0 && !floorCells.contains(cellKey(x, y, z - 1))) {
222 return failure<void>(eve::DiagnosticCode::InvalidArgument, "upper floor has no structural support");
223 }
224 if (!floorCells.contains(cellKey(x, y, z + 1)) &&
225 !roofCells.contains(cellKey(x, y, z + 1))) {
226 return failure<void>(eve::DiagnosticCode::InvalidArgument, "floor cell has no roof coverage");
227 }
228 }
229 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
230}

233 const HouseComponentLibrary &library) const {

244 graphics::Texture *texture = nullptr;
245 graphics::Texture *normalTexture = nullptr;
246 graphics::Texture *heightTexture = nullptr;
247 float r = 1.f, g = 1.f, b = 1.f, a = 1.f;
248 float metallic = 0.f, roughness = 0.72f;
249 float parallaxScale = 0.f, parallaxMinLayers = 8.f, parallaxMaxLayers = 32.f;
250 float cellBombScale = 4.f, cellBombStrength = 0.f, cellBombRotation = 1.f;
251 };
252 std::unordered_map<std::string, std::vector<CachedPart>> parts;
253 try {
254 for (const auto &i : instances) {
255 const auto component = library.find(i.componentId);
256 if (!component) {
257 destroyCreated();

259 "unknown component: " + i.componentId, "componentId");
260 }
261 const HouseComponent &c = component->get();
262 // eve::ref cannot represent null, so look up before default-inserting.
263 auto modelIt = data.find(c.modelPath);
264 if (modelIt == data.end()) modelIt = data.emplace(c.modelPath, loadModel(&models, c.modelPath)).first;
265 model3d::ModelData *model = modelIt->second.get();
266 // Material overrides belong to a component, so two components may safely reuse the
267 // same GLB with different architectural finishes.
268 auto &cached = parts[c.id];
269 if (cached.empty()) {
270 const aiScene *scene = model->getScene();
271 if (!scene || !scene->mRootNode) throw std::runtime_error("model has no scene root: " + c.modelPath);
272 std::function<void(const aiNode *, const aiMatrix4x4 &)> walk =
273 [&](const aiNode *node, const aiMatrix4x4 &parent) {
274 const aiMatrix4x4 world = parent * node->mTransformation;
275 for (unsigned ni = 0; ni < node->mNumMeshes; ++ni) {
276 const unsigned mi = node->mMeshes[ni];
277 if (mi >= scene->mNumMeshes || !scene->mMeshes[mi]) continue;
278 const aiMesh *source = scene->mMeshes[mi];
279 CachedPart part;
280 part.mesh = gfx.newMeshFromAssimp(*source, world);
281 if (scene->mMaterials && source->mMaterialIndex < scene->mNumMaterials) {
282 const aiMaterial *material = scene->mMaterials[source->mMaterialIndex];
283 aiColor4D color(1.f, 1.f, 1.f, 1.f);
284 material->Get(AI_MATKEY_COLOR_DIFFUSE, color);
285 material->Get(AI_MATKEY_BASE_COLOR, color);
286 part.r = color.r; part.g = color.g; part.b = color.b; part.a = color.a;
287 material->Get(AI_MATKEY_METALLIC_FACTOR, part.metallic);
288 material->Get(AI_MATKEY_ROUGHNESS_FACTOR, part.roughness);
289 part.texture = assimpTexture(&gfx, scene, material, c.modelPath,
290 aiTextureType_BASE_COLOR, aiTextureType_DIFFUSE);
291 part.normalTexture = assimpTexture(&gfx, scene, material, c.modelPath,
292 aiTextureType_NORMALS, aiTextureType_NORMAL_CAMERA);
293 part.heightTexture = assimpTexture(&gfx, scene, material, c.modelPath,
294 aiTextureType_HEIGHT, aiTextureType_DISPLACEMENT);
295 }
296 if (c.material.hasBaseColor) {
297 part.r = c.material.baseColorR;
298 part.g = c.material.baseColorG;
299 part.b = c.material.baseColorB;
300 part.a = c.material.baseColorA;
301 }
302 if (!c.material.baseColorTexture.empty())
303 part.texture = textureFromFile(&gfx, c.modelPath, c.material.baseColorTexture);
304 if (!c.material.normalTexture.empty())
305 part.normalTexture = textureFromFile(&gfx, c.modelPath, c.material.normalTexture);
306 if (!c.material.heightTexture.empty())
307 part.heightTexture = textureFromFile(&gfx, c.modelPath, c.material.heightTexture);
308 if (c.material.hasMetallic) part.metallic = c.material.metallic;
309 if (c.material.hasRoughness) part.roughness = c.material.roughness;
310 part.parallaxScale = c.material.parallaxScale;
311 part.parallaxMinLayers = c.material.parallaxMinLayers;
312 part.parallaxMaxLayers = c.material.parallaxMaxLayers;
313 part.cellBombScale = c.material.cellBombScale;
314 part.cellBombStrength = c.material.cellBombStrength;
315 part.cellBombRotation = c.material.cellBombRotation;
316 cached.push_back(part);
317 }
318 for (unsigned child = 0; child < node->mNumChildren; ++child)
319 walk(node->mChildren[child], world);
320 };
321 walk(scene->mRootNode, aiMatrix4x4());
322 }
323 for (const auto &part : cached) {
324 auto *e = graphics::Renderable3D::create();
325 if (!e) {
326 destroyCreated();
327 return failure<std::vector<ecs::EntityHandle>>(eve::DiagnosticCode::Failed,
328 "failed to create Renderable3D entity");
329 }
330 e->setMesh(part.mesh);
331 e->setPosition(i.x * moduleSize, i.z * floorHeight, i.y * moduleSize);
332 e->setYaw(float(i.rotationDeg) * 3.14159265358979323846f / 180.f);
333 e->setTint(part.r, part.g, part.b, part.a);
334 if (part.texture) e->setTexture(part.texture);
335 if (part.normalTexture) e->setNormalTexture(part.normalTexture);
336 if (part.heightTexture) e->setHeightTexture(part.heightTexture);
337 e->setMetallic(part.metallic);
338 e->setRoughness(part.roughness);
339 e->setTexCellBomb(part.cellBombScale, part.cellBombStrength,
340 part.cellBombRotation);
341 if (part.heightTexture && part.parallaxScale > 0.f)
342 e->setParallax(part.parallaxScale, part.parallaxMinLayers,
343 part.parallaxMaxLayers);
344 entities.push_back(ecs::handle_of(e));
345 }
346 }
347 } catch (const std::exception &e) {
348 destroyCreated();
349 return failure<std::vector<ecs::EntityHandle>>(eve::DiagnosticCode::Failed, e.what());
350 } catch (...) {
351 destroyCreated();
352 return failure<std::vector<ecs::EntityHandle>>(eve::DiagnosticCode::Failed,
353 "house layout instantiation failed");
354 }
355 return eve::Result<std::vector<ecs::EntityHandle>>::success(std::move(entities),
356 eve::Status::success(eve::StatusCode::Applied));
357}
