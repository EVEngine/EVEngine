29template <class T>
30void appendInteger(std::uint64_t& hash, T value) noexcept {
31 using Unsigned = std::make_unsigned_t<T>;
32 const Unsigned bits = static_cast<Unsigned>(value);
33 for (std::size_t index = 0; index < sizeof(T); ++index)
34 appendByte(hash, static_cast<std::uint8_t>((bits >> (index * 8)) & static_cast<Unsigned>(0xff)));
35}

42std::int64_t quantize(float value) noexcept {
43 if (!std::isfinite(value)) return std::numeric_limits<std::int64_t>::min();
44 const double scaled = std::round(static_cast<double>(value) * 1000.0);
45 if (scaled <= static_cast<double>(std::numeric_limits<std::int64_t>::min()))
46 return std::numeric_limits<std::int64_t>::min();
47 if (scaled >= static_cast<double>(std::numeric_limits<std::int64_t>::max()))
48 return std::numeric_limits<std::int64_t>::max();
49 return static_cast<std::int64_t>(scaled);
50}

345eve::Result<ClimbingPredictionDecision> ClimbingRuntime::tryBeginPredicted(
346 physics::World3D& world, const ClimbingPose& pose, const ClimbingPredictionRequest& request,
347 eve::SimulationTick serverTick, std::uint64_t maxClientTickLead) {
348 ClimbingPredictionDecision decision;
349 decision.sequence = request.sequence;
350 decision.clientTick = request.clientTick;
351 decision.serverTick = serverTick;
352 const auto finish = [this](ClimbingPredictionDecision value)
353 -> eve::Result<ClimbingPredictionDecision> {
354 auto state = snapshot();
355 if (!state) return eve::Result<ClimbingPredictionDecision>::failure(state.status());
356 value.authoritativeSnapshot = std::move(state).takeValue();
357 return eve::Result<ClimbingPredictionDecision>::success(std::move(value));
358 };
359
360 if (request.sequence.isZero() || !validKey(request.candidate)) {
361 decision.disposition = ClimbingPredictionDisposition::Rejected;
362 decision.reason = ClimbingPredictionReason::InvalidRequest;
363 return finish(std::move(decision));
364 }
365 if (request.clientTick > serverTick &&
366 request.clientTick.value() - serverTick.value() > maxClientTickLead) {
367 decision.disposition = ClimbingPredictionDisposition::Rejected;
368 decision.reason = ClimbingPredictionReason::TickTooFarAhead;
369 return finish(std::move(decision));
370 }
371 if (activePhase(phase_)) {
372 decision.disposition = ClimbingPredictionDisposition::Rejected;
373 decision.reason = ClimbingPredictionReason::RuntimeBusy;
374 return finish(std::move(decision));
375 }
376
377 auto candidates = probe(world, pose);
378 if (!candidates) return eve::Result<ClimbingPredictionDecision>::failure(candidates.status());
379 if (candidates.value().empty()) {
380 decision.disposition = ClimbingPredictionDisposition::Rejected;
381 decision.reason = ClimbingPredictionReason::NoCandidate;
382 return finish(std::move(decision));
383 }
384 decision.authoritativeCandidate = makeClimbingCandidateKey(candidates.value().front(), 0);
385 std::size_t match = candidates.value().size();
386 for (std::size_t index = 0; index < candidates.value().size(); ++index) {
387 if (index > std::numeric_limits<std::uint32_t>::max()) break;
388 if (makeClimbingCandidateKey(candidates.value()[index], static_cast<std::uint32_t>(index)) ==
389 request.candidate) {
390 match = index;
391 break;
392 }
393 }
394 if (match == candidates.value().size()) {
395 decision.disposition = ClimbingPredictionDisposition::Corrected;
396 decision.reason = ClimbingPredictionReason::CandidateMismatch;
397 return finish(std::move(decision));
398 }
399
400 decision.authoritativeCandidate = request.candidate;
401 auto prepared = prepareBeginCandidate(world, pose, serverTick, candidates.value()[match]);
402 if (!prepared) return eve::Result<ClimbingPredictionDecision>::failure(prepared.status());
403 auto committed = commitBegin(std::move(prepared).takeValue());
404 if (!committed) return eve::Result<ClimbingPredictionDecision>::failure(committed.status());
405 decision.disposition = ClimbingPredictionDisposition::Accepted;
406 decision.reason = ClimbingPredictionReason::None;
407 return finish(std::move(decision));
408}
