23bool ConversationAsset::validate(std::string* error) const {
24 const auto fail = [&](const std::string& message) {
25 if (error) *error = "conversation '" + id + "': " + message;
26 return false;
27 };
28 if (id.empty()) return fail("missing id");
29 if (entry.empty()) return fail("missing entry node");
30 std::unordered_set<std::string> ids;
31 for (const auto& node : nodes) {
32 if (node.id.empty()) return fail("node has an empty id");
33 if (!ids.insert(node.id).second) return fail("duplicate node id '" + node.id + "'");
34 }
35 if (!findNode(entry)) return fail("entry node '" + entry + "' does not exist");
36 const auto checkRef = [&](const std::string& owner, const std::string& ref) {
37 return ref.empty() || findNode(ref) ? true : fail("node '" + owner + "' references missing node '" + ref + "'");
38 };
39 for (const auto& node : nodes) {
40 if (!checkRef(node.id, node.next)) return false;
41 if (!checkRef(node.id, node.returnNode)) return false;
42 auto paymentValid = node.payment.validate();
43 if (!paymentValid) {
44 const auto* diagnostic = paymentValid.error();
45 return fail(diagnostic ? diagnostic->message() : "invalid node payment");
46 }
47 if (!node.payment.empty() && node.kind != ConversationAsset::Node::Kind::Command)
48 return fail("node '" + node.id + "' payment is only valid for command nodes");
49 if (!node.stateMutations.empty() && node.kind != ConversationAsset::Node::Kind::Command)
50 return fail("node '" + node.id + "' state mutations are only valid for command nodes");
51 for (const auto& route : node.routes) {
52 auto routePaymentValid = route.payment.validate();
53 if (!routePaymentValid) {
54 const auto* diagnostic = routePaymentValid.error();
55 return fail(diagnostic ? diagnostic->message() : "invalid route payment");
56 }
57 if (!route.payment.empty() && node.kind != ConversationAsset::Node::Kind::Choice)
58 return fail("route payment is only valid for choice nodes");
59 if (!route.stateMutations.empty() && node.kind != ConversationAsset::Node::Kind::Choice)
60 return fail("route state mutations are only valid for choice nodes");
61 if (!checkRef(node.id, route.second)) return false;
62 }
63 }
64 return true;
65}

140std::string ConversationRunner::evaluateRoute(const ConversationAsset::Node& node, std::string* error) {
141 for (const auto& route : node.routes) {
142 if (!route.condition.isNull()) {
143 if (!conditionEvaluator_) {
144 fail(error, "conversation: structured route requires a condition evaluator");
145 return {};
146 }
147 eve::decision::ConditionResult result = conditionEvaluator_(route.condition);
148 lastConditionResult_ = result;
149 if (result.passed()) return route.second;
150 continue;
151 }
152 if (route.first.empty() || route.first == "else") return route.second;
153 if (!expressionEvaluator_) {
154 fail(error, "conversation: branch requires an expression evaluator");
155 return {};
156 }
157 const StateValue value = expressionEvaluator_(route.first, bindings_, locals_);
158 if (!value.isBool()) {
159 fail(error, "conversation: expression '" + route.first + "' did not return bool");
160 return {};
161 }
162 if (value.asBool()) return route.second;
163 }
164 return node.next;
165}

167bool ConversationRunner::runUntilBlocked(std::string* error) {
168 int budget = 10000;
169 while (asset_ && budget-- > 0) {
170 const auto* node = currentNode();
171 if (!node) return fail(error, "conversation: invalid execution cursor");
172 switch (node->kind) {
173 case ConversationAsset::Node::Kind::Line:
174 emit(Event::Kind::Line, node);
175 blocked_ = true;
176 return true;
177 case ConversationAsset::Node::Kind::Choice:
178 emit(Event::Kind::Choice, node);
179 blocked_ = true;
180 return true;
181 case ConversationAsset::Node::Kind::Wait:
182 blocked_ = true;
183 return true;
184 case ConversationAsset::Node::Kind::Branch: {
185 const std::string next = evaluateRoute(*node, error);
186 if (next.empty() && error && !error->empty()) return false;
187 if (!enter(next, error)) return false;
188 break;
189 }
190 case ConversationAsset::Node::Kind::End:
191 if (callStack_.empty()) {
192 emit(Event::Kind::Ended, node);
193 stop();
194 return true;
195 } else {
196 Frame frame = std::move(callStack_.back());
197 callStack_.pop_back();
198 asset_ = frame.asset;
199 bindings_ = std::move(frame.bindings);
200 locals_ = std::move(frame.locals);
201 if (!enter(frame.returnNode, error)) return false;
202 }
203 break;
204 case ConversationAsset::Node::Kind::Call: {
205 if (!assetResolver_)
206 return fail(error, "conversation: call node requires an asset resolver");
207 const ConversationAsset* target = assetResolver_(node->target);
208 if (!target)
209 return fail(error, "conversation: missing called asset '" + node->target + "'");
210 std::string validationError;
211 if (!target->validate(&validationError)) return fail(error, validationError);
212 Frame frame;
213 frame.asset = asset_;
214 frame.returnNode = node->returnNode.empty() ? node->next : node->returnNode;
215 frame.bindings = bindings_;
216 frame.locals = locals_;
217 callStack_.push_back(std::move(frame));
218 StateValue targetBindings = node->arguments;
219 targetBindings.mergeDefaults(bindings_);
220 asset_ = target;
221 bindings_ = std::move(targetBindings);
222 locals_ = StateValue::object();
223 if (!enter(target->entry, error)) return false;
224 break;
225 }
226 case ConversationAsset::Node::Kind::Command: {
227 const auto requestIt = commandRequestHandlers_.find(node->target);
228 const auto legacyIt = commandHandlers_.find(node->target);
229 if (requestIt == commandRequestHandlers_.end() && legacyIt == commandHandlers_.end() &&
230 !commandRequestDispatcher_)
231 return fail(error, "conversation: command '" + node->target +
232 "' is not registered");
233 emit(Event::Kind::Command, node, node->target);
234 if (requestIt != commandRequestHandlers_.end() || commandRequestDispatcher_) {
235 CommandRequest request;
236 request.name = node->target;
237 request.kind = node->commandKind;
238 request.arguments = toCanonicalValue(node->arguments);
239 request.bindings = toCanonicalValue(bindings_);
240 request.locals = toCanonicalValue(locals_);
241 request.payment = node->payment;
242 request.stateMutations = node->stateMutations;
243 lastCommandRequest_ = request;
244 const CommandRequestHandler& handler =
245 requestIt != commandRequestHandlers_.end() ? requestIt->second : commandRequestDispatcher_;
246 CommandResponse response = handler(request);
247 if (response.status == CommandResponse::Status::Failed)
248 return fail(error, response.error.empty() ? "conversation: command failed" : response.error);
249 if (response.status == CommandResponse::Status::Blocked) {
250 blocked_ = true;
251 waitingCommand_ = true;
252 return true;
253 }
254 if (!node->expression.empty()) locals_.set(node->expression, toDialogueStateValue(response.value));
255 } else {
256 CommandResult result = legacyIt->second(node->arguments, bindings_, locals_);
257 if (result.status == CommandResult::Status::Failed)
258 return fail(error, result.error.empty() ? "conversation: command failed" : result.error);
259 if (result.status == CommandResult::Status::Blocked) {
260 blocked_ = true;
261 waitingCommand_ = true;
262 return true;
263 }
264 if (!node->expression.empty()) locals_.set(node->expression, std::move(result.value));
265 }
266 if (!enter(node->next, error)) return false;
267 break;
268 }
269 }
270 }
271 return fail(error, "conversation: execution budget exceeded");
272}

297 out = StateValue::object();

300 out.set("current", captureFrame(asset_, nodeId_, bindings_, locals_));

310bool ConversationRunner::restoreState(const StateValue& in, std::string* error) {
311 if (!in.isObject()) return fail(error, "conversation: runner state must be an object");
312 const StateValue* active = in.find("active");
313 if (!active || !active->isBool()) return fail(error, "conversation: state is missing active");
314 if (!active->asBool()) {
315 stop();
316 return true;
317 }
318 if (!assetResolver_) return fail(error, "conversation: restore requires an asset resolver");
319 const StateValue* current = in.find("current");
320 if (!current || !current->isObject()) return fail(error, "conversation: state is missing current");
321 std::string assetId;
322 std::string nodeId;
323 if (!readString(*current, "asset", assetId) || !readString(*current, "node", nodeId))
324 return fail(error, "conversation: current frame is malformed");
325 const ConversationAsset* restoredAsset = assetResolver_(assetId);
326 if (!restoredAsset) return fail(error, "conversation: saved asset '" + assetId + "' is missing");
327 const StateValue* version = current->find("version");
328 if (!version || !version->isInt() || version->asInt() != restoredAsset->version)
329 return fail(error, "conversation: saved asset version does not match '" + assetId + "'");
330 const StateValue* savedBindings = current->find("bindings");
331 const StateValue* savedLocals = current->find("locals");
332 if (!savedBindings || !savedBindings->isObject() || !savedLocals || !savedLocals->isObject())
333 return fail(error, "conversation: current values are malformed");
334
335 std::vector<Frame> restoredStack;
336 const StateValue* stack = in.find("stack");
337 if (!stack || !stack->isArray()) return fail(error, "conversation: state is missing stack");
338 for (size_t i = 0; i < stack->arraySize(); ++i) {
339 const StateValue& saved = stack->at(i);
340 std::string savedAssetId;
341 std::string returnNode;
342 if (!saved.isObject() || !readString(saved, "asset", savedAssetId) ||
343 !readString(saved, "node", returnNode))
344 return fail(error, "conversation: saved call frame is malformed");
345 const ConversationAsset* savedAsset = assetResolver_(savedAssetId);
346 const StateValue* savedVersion = saved.find("version");
347 const StateValue* frameBindings = saved.find("bindings");
348 const StateValue* frameLocals = saved.find("locals");
349 if (!savedAsset || !savedVersion || !savedVersion->isInt() ||
350 savedVersion->asInt() != savedAsset->version || !frameBindings ||
351 !frameBindings->isObject() || !frameLocals || !frameLocals->isObject() ||
352 !savedAsset->findNode(returnNode))
353 return fail(error, "conversation: saved call frame cannot be restored");
354 restoredStack.push_back({savedAsset, returnNode, *frameBindings, *frameLocals});
355 }
356 if (!restoredAsset->findNode(nodeId))
357 return fail(error, "conversation: saved node '" + nodeId + "' is missing");
358 const StateValue* blocked = in.find("blocked");
359 if (!blocked || !blocked->isBool()) return fail(error, "conversation: state is missing blocked");
360 const StateValue* waitingCommand = in.find("waitingCommand");
361 if (waitingCommand && !waitingCommand->isBool())
362 return fail(error, "conversation: waitingCommand is malformed");
363 asset_ = restoredAsset;
364 nodeId_ = std::move(nodeId);
365 bindings_ = *savedBindings;
366 locals_ = *savedLocals;
367 blocked_ = blocked->asBool();
368 waitingCommand_ = waitingCommand && waitingCommand->asBool();
369 callStack_ = std::move(restoredStack);
370 return true;
371}

373eve::Result<void> ConversationRunner::resumeCommand(StateValue result) {
374 const auto* node = currentNode();
375 if (!node || !blocked_ || !waitingCommand_ ||
376 node->kind != ConversationAsset::Node::Kind::Command)
377 return runnerFailure(eve::DiagnosticCode::DialogueNotWaitingForCommand,

407bool ConversationRunner::select(const std::string& routeId, std::string* error) {
408 const auto* node = currentNode();
409 if (node) {

424eve::Result<void> ConversationRunner::selectRouteForTransaction(const std::string& routeId) {
425 const auto* node = currentNode();
426 if (!node || !blocked_ || node->kind != ConversationAsset::Node::Kind::Choice)
427 return runnerFailure(eve::DiagnosticCode::DialogueNotWaitingForChoice,
428 "conversation: runner is not waiting for a choice", "route");
429 for (const auto& route : node->routes) {
430 if (route.first != routeId) continue;
431 if (!route.condition.isNull()) {
432 if (!conditionEvaluator_)
433 return runnerFailure(eve::DiagnosticCode::PreconditionViolation,
434 "conversation: structured choice requires a condition evaluator",
435 "route.condition");
436 eve::decision::ConditionResult result = conditionEvaluator_(route.condition);
437 lastConditionResult_ = result;
438 if (!result.passed())
439 return runnerFailure(eve::DiagnosticCode::DialogueConditionRejected,
440 "conversation: choice condition rejected (" +
441 std::string(eve::decision::conditionReasonCodeName(result.reasonCode())) + ")",
442 "route.condition");
443 }
444
445 StateValue before;
446 if (!captureState(before))
447 return runnerFailure(eve::DiagnosticCode::Failed, "conversation: could not capture choice state", "route");
448 std::string error;
449 if (enter(route.second, &error) && runUntilBlocked(&error))
450 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
451
452 std::string restoreError;
453 if (!restoreState(before, &restoreError) && error.empty()) error = restoreError;
454 return runnerFailure(eve::DiagnosticCode::Failed,
455 error.empty() ? "conversation: choice selection failed" : std::move(error), "route");
456 }
457 return runnerFailure(eve::DiagnosticCode::DialogueRouteNotFound, "conversation: unknown choice '" + routeId + "'",
458 "route");
