24eve::Result<void> ConversationAsset::validate() const {
25 const auto fail = [&](const std::string& message) {
26 return runnerFailure(eve::DiagnosticCode::InvalidArgument,
27 "conversation '" + id + "': " + message, id);
28 };
29 if (id.empty()) return fail("missing id");
30 if (entry.empty()) return fail("missing entry node");
31 std::unordered_set<std::string> parameterIds;
32 for (const auto& parameter : parameters) {
33 if (parameter.name.empty()) return fail("parameter has an empty name");
34 if (!parameterIds.insert(parameter.name).second)
35 return fail("duplicate parameter '" + parameter.name + "'");
36 if (parameter.required && !parameter.defaultValue.isNull())
37 return fail("required parameter '" + parameter.name + "' cannot declare a default");
38 }
39 std::unordered_set<std::string> ids;
40 for (const auto& node : nodes) {
41 if (node.id.empty()) return fail("node has an empty id");
42 if (!ids.insert(node.id).second) return fail("duplicate node id '" + node.id + "'");
43 }
44 if (!findNode(entry)) return fail("entry node '" + entry + "' does not exist");
45 const auto checkRef = [&](const std::string& owner, const std::string& ref) -> eve::Result<void> {
46 if (ref.empty() || findNode(ref)) return eve::Result<void>::success();
47 return fail("node '" + owner + "' references missing node '" + ref + "'");
48 };
49 for (const auto& node : nodes) {
50 auto nextValid = checkRef(node.id, node.next);
51 if (!nextValid) return eve::Result<void>::failure(nextValid.status());
52 auto returnValid = checkRef(node.id, node.returnNode);
53 if (!returnValid) return eve::Result<void>::failure(returnValid.status());
54 auto paymentValid = node.payment.validate();
55 if (!paymentValid) {
56 const auto* diagnostic = paymentValid.error();
57 return fail(diagnostic ? diagnostic->message() : "invalid node payment");
58 }
59 if (!node.payment.empty() && node.kind != ConversationAsset::Node::Kind::Command)
60 return fail("node '" + node.id + "' payment is only valid for command nodes");
61 if (!node.stateMutations.empty() && node.kind != ConversationAsset::Node::Kind::Command)
62 return fail("node '" + node.id + "' state mutations are only valid for command nodes");
63 for (const auto& route : node.routes) {
64 auto routePaymentValid = route.payment.validate();
65 if (!routePaymentValid) {
66 const auto* diagnostic = routePaymentValid.error();
67 return fail(diagnostic ? diagnostic->message() : "invalid route payment");
68 }
69 if (!route.payment.empty() && node.kind != ConversationAsset::Node::Kind::Choice)
70 return fail("route payment is only valid for choice nodes");
71 if (!route.stateMutations.empty() && node.kind != ConversationAsset::Node::Kind::Choice)
72 return fail("route state mutations are only valid for choice nodes");
73 auto routeValid = checkRef(node.id, route.second);
74 if (!routeValid) return eve::Result<void>::failure(routeValid.status());
75 }
76 }
77 return eve::Result<void>::success();
78}

85bool ConversationRunner::startImpl(const ConversationAsset* asset, StateValue bindings,
86 std::string* error) {
87 if (!asset) return fail(error, "conversation: null asset");
88 if (!bindings.isObject()) return fail(error, "conversation: bindings must be an object");
89 auto validated = asset->validate();
90 if (!validated) return fail(error, validated.status().describe());
91 StateValue resolvedBindings = bindings;
92 for (const auto& parameter : asset->parameters) {
93 const StateValue* value = resolvedBindings.find(parameter.name);
94 if (!value && !parameter.defaultValue.isNull()) {
95 resolvedBindings.set(parameter.name, parameter.defaultValue);
96 value = resolvedBindings.find(parameter.name);
97 }
98 if (!value && parameter.required)
99 return fail(error, "conversation '" + asset->id + "': missing required binding '" + parameter.name + "'");
100 if (!value) continue;
101 const bool typeMatches = parameter.type == ConversationAsset::Parameter::Type::Any ||
102 (parameter.type == ConversationAsset::Parameter::Type::String && value->isString()) ||
103 (parameter.type == ConversationAsset::Parameter::Type::Int && value->isInt()) ||
104 (parameter.type == ConversationAsset::Parameter::Type::Float && (value->isFloat() || value->isInt())) ||
105 (parameter.type == ConversationAsset::Parameter::Type::Bool && value->isBool());
106 if (!typeMatches)
107 return fail(error, "conversation '" + asset->id + "': binding '" + parameter.name + "' has the wrong type");
108 }
109 for (const auto& key : bindings.keys()) {
110 if (std::none_of(asset->parameters.begin(), asset->parameters.end(),
111 [&](const auto& parameter) { return parameter.name == key; }))
112 return fail(error, "conversation '" + asset->id + "': undeclared binding '" + key + "'");
113 }
114 asset_ = asset;
115 bindings_ = std::move(resolvedBindings);
116 locals_ = StateValue::object();
117 callStack_.clear();
118 blocked_ = false;
119 waitingCommand_ = false;
120 pendingCommandRequestId_.clear();
121 lastConditionResult_.reset();
122 lastCommandRequest_.reset();
123 emit(Event::Kind::Started);
124 return enter(asset_->entry, error) && runUntilBlockedImpl(error);
125}

201std::string ConversationRunner::evaluateRoute(const ConversationAsset::Node& node, std::string* error) {
202 for (const auto& route : node.routes) {
203 if (!route.condition.isNull()) {
204 if (!conditionEvaluator_) {
205 fail(error, "conversation: structured route requires a condition evaluator");
206 return {};
207 }
208 eve::decision::ConditionResult result = conditionEvaluator_(route.condition);
209 lastConditionResult_ = result;
210 if (result.passed()) return route.second;
211 continue;
212 }
213 if (route.expression.empty()) return route.second;
214 if (!expressionEvaluator_) {
215 fail(error, "conversation: branch requires an expression evaluator");
216 return {};
217 }
218 auto evaluated = expressionEvaluator_(route.expression, bindings_, locals_);
219 if (!evaluated) {
220 fail(error, evaluated.status().describe());
221 return {};
222 }
223 const StateValue value = std::move(evaluated).takeValue();
224 if (!value.isBool()) {
225 fail(error, "conversation: expression '" + route.expression + "' did not return bool");
226 return {};
227 }
228 if (value.asBool()) return route.second;
229 }
230 return node.next;
231}

233bool ConversationRunner::runUntilBlockedImpl(std::string* error) {
234 int budget = 10000;
235 while (asset_ && budget-- > 0) {
236 const auto* node = currentNode();
237 if (!node) return fail(error, "conversation: invalid execution cursor");
238 switch (node->kind) {
239 case ConversationAsset::Node::Kind::Line:
240 emit(Event::Kind::Line, node);
241 blocked_ = true;
242 return true;
243 case ConversationAsset::Node::Kind::Choice:
244 emit(Event::Kind::Choice, node);
245 blocked_ = true;
246 return true;
247 case ConversationAsset::Node::Kind::Wait:
248 blocked_ = true;
249 return true;
250 case ConversationAsset::Node::Kind::Branch: {
251 const std::string next = evaluateRoute(*node, error);
252 if (next.empty() && error && !error->empty()) return false;
253 if (!enter(next, error)) return false;
254 break;
255 }
256 case ConversationAsset::Node::Kind::End:
257 if (callStack_.empty()) {
258 emit(Event::Kind::Ended, node);
259 stop();
260 return true;
261 } else {
262 Frame frame = std::move(callStack_.back());
263 callStack_.pop_back();
264 asset_ = frame.asset;
265 bindings_ = std::move(frame.bindings);
266 locals_ = std::move(frame.locals);
267 if (!enter(frame.returnNode, error)) return false;
268 }
269 break;
270 case ConversationAsset::Node::Kind::Call: {
271 if (!assetResolver_)
272 return fail(error, "conversation: call node requires an asset resolver");
273 const ConversationAsset* target = assetResolver_(node->target);
274 if (!target)
275 return fail(error, "conversation: missing called asset '" + node->target + "'");
276 auto validated = target->validate();
277 if (!validated) return fail(error, validated.status().describe());
278 Frame frame;
279 frame.asset = asset_;
280 frame.returnNode = node->returnNode.empty() ? node->next : node->returnNode;
281 frame.bindings = bindings_;
282 frame.locals = locals_;
283 callStack_.push_back(std::move(frame));
284 StateValue targetBindings = node->arguments;
285 targetBindings.mergeDefaults(bindings_);
286 asset_ = target;
287 bindings_ = std::move(targetBindings);
288 locals_ = StateValue::object();
289 if (!enter(target->entry, error)) return false;
290 break;
291 }
292 case ConversationAsset::Node::Kind::Command: {
293 const auto requestIt = commandRequestHandlers_.find(node->target);
294 const auto legacyIt = commandHandlers_.find(node->target);
295 if (requestIt == commandRequestHandlers_.end() && legacyIt == commandHandlers_.end() &&
296 !commandRequestDispatcher_)
297 return fail(error, "conversation: command '" + node->target +
298 "' is not registered");
299 emit(Event::Kind::Command, node, node->target);
300 if (requestIt != commandRequestHandlers_.end() || commandRequestDispatcher_) {
301 CommandRequest request;
302 request.requestId = asset_->id + ":" + node->id + ":" + std::to_string(commandSequence_++);
303 request.name = node->target;
304 request.kind = node->commandKind;
305 request.arguments = toCanonicalValue(node->arguments);
306 request.bindings = toCanonicalValue(bindings_);
307 request.locals = toCanonicalValue(locals_);
308 request.payment = node->payment;
309 request.stateMutations = node->stateMutations;
310 lastCommandRequest_ = request;
311 const CommandRequestHandler& handler =
312 requestIt != commandRequestHandlers_.end() ? requestIt->second : commandRequestDispatcher_;
313 CommandResponse response = handler(request);
314 if (response.status == CommandResponse::Status::Failed)
315 return fail(error, response.error.empty() ? "conversation: command failed" : response.error);
316 if (response.status == CommandResponse::Status::Blocked) {
317 blocked_ = true;
318 waitingCommand_ = true;
319 pendingCommandRequestId_ = request.requestId;
320 return true;
321 }
322 if (!node->expression.empty()) locals_.set(node->expression, toDialogueStateValue(response.value));
323 } else {
324 CommandResult result = legacyIt->second(node->arguments, bindings_, locals_);
325 if (result.status == CommandResult::Status::Failed)
326 return fail(error, result.error.empty() ? "conversation: command failed" : result.error);
327 if (result.status == CommandResult::Status::Blocked) {
328 blocked_ = true;
329 waitingCommand_ = true;
330 pendingCommandRequestId_ = asset_->id + ":" + node->id + ":" +
331 std::to_string(commandSequence_++);
332 return true;
333 }
334 if (!node->expression.empty()) locals_.set(node->expression, std::move(result.value));
335 }
336 if (!enter(node->next, error)) return false;
337 break;
338 }
339 }
340 }
341 return fail(error, "conversation: execution budget exceeded");
342}

366bool ConversationRunner::captureStateImpl(StateValue& out) const {
367 out = StateValue::object();
368 out.set("schema", StateValue::string(std::string(SaveSchema)));
369 out.set("schemaVersion", StateValue::integer(SaveVersion));
370 out.set("active", StateValue::boolean(asset_ != nullptr));
371 if (!asset_) return true;
372 out.set("current", captureFrame(asset_, nodeId_, bindings_, locals_));
373 out.set("blocked", StateValue::boolean(blocked_));
374 out.set("waitingCommand", StateValue::boolean(waitingCommand_));
375 out.set("pendingCommandRequestId", StateValue::string(pendingCommandRequestId_));
376 out.set("commandSequence", StateValue::integer(static_cast<std::int64_t>(commandSequence_)));
377 StateValue stack = StateValue::array();
378 for (const auto& frame : callStack_)
379 stack.pushBack(captureFrame(frame.asset, frame.returnNode, frame.bindings, frame.locals));
380 out.set("stack", std::move(stack));
381 return true;

384bool ConversationRunner::restoreStateImpl(const StateValue& in, std::string* error) {
385 if (!in.isObject()) return fail(error, "conversation: runner state must be an object");
386 const StateValue* schema = in.find("schema");
387 const StateValue* schemaVersion = in.find("schemaVersion");
388 if (!schema || !schema->isString() || schema->asString() != SaveSchema || !schemaVersion ||
389 !schemaVersion->isInt() || schemaVersion->asInt() != SaveVersion)
390 return fail(error, "conversation: unsupported runner save schema or version");
391 const StateValue* active = in.find("active");
392 if (!active || !active->isBool()) return fail(error, "conversation: state is missing active");
393 if (!active->asBool()) {
394 stop();
395 return true;
396 }
397 if (!assetResolver_) return fail(error, "conversation: restore requires an asset resolver");
398 const StateValue* current = in.find("current");
399 if (!current || !current->isObject()) return fail(error, "conversation: state is missing current");
400 std::string assetId;
401 std::string nodeId;
402 if (!readString(*current, "asset", assetId) || !readString(*current, "node", nodeId))
403 return fail(error, "conversation: current frame is malformed");
404 const ConversationAsset* restoredAsset = assetResolver_(assetId);
405 if (!restoredAsset) return fail(error, "conversation: saved asset '" + assetId + "' is missing");
406 const StateValue* version = current->find("version");
407 if (!version || !version->isInt() || version->asInt() != restoredAsset->version)
408 return fail(error, "conversation: saved asset version does not match '" + assetId + "'");
409 const StateValue* savedBindings = current->find("bindings");
410 const StateValue* savedLocals = current->find("locals");
411 if (!savedBindings || !savedBindings->isObject() || !savedLocals || !savedLocals->isObject())
412 return fail(error, "conversation: current values are malformed");
413
414 std::vector<Frame> restoredStack;
415 const StateValue* stack = in.find("stack");
416 if (!stack || !stack->isArray()) return fail(error, "conversation: state is missing stack");
417 for (size_t i = 0; i < stack->arraySize(); ++i) {
418 const StateValue& saved = stack->at(i);
419 std::string savedAssetId;
420 std::string returnNode;
421 if (!saved.isObject() || !readString(saved, "asset", savedAssetId) ||
422 !readString(saved, "node", returnNode))
423 return fail(error, "conversation: saved call frame is malformed");
424 const ConversationAsset* savedAsset = assetResolver_(savedAssetId);
425 const StateValue* savedVersion = saved.find("version");
426 const StateValue* frameBindings = saved.find("bindings");
427 const StateValue* frameLocals = saved.find("locals");
428 if (!savedAsset || !savedVersion || !savedVersion->isInt() ||
429 savedVersion->asInt() != savedAsset->version || !frameBindings ||
430 !frameBindings->isObject() || !frameLocals || !frameLocals->isObject() ||
431 !savedAsset->findNode(returnNode))
432 return fail(error, "conversation: saved call frame cannot be restored");
433 restoredStack.push_back({savedAsset, returnNode, *frameBindings, *frameLocals});
434 }
435 if (!restoredAsset->findNode(nodeId))
436 return fail(error, "conversation: saved node '" + nodeId + "' is missing");
437 const StateValue* blocked = in.find("blocked");
438 if (!blocked || !blocked->isBool()) return fail(error, "conversation: state is missing blocked");
439 const StateValue* waitingCommand = in.find("waitingCommand");
440 if (waitingCommand && !waitingCommand->isBool())
441 return fail(error, "conversation: waitingCommand is malformed");
442 const StateValue* pendingRequestId = in.find("pendingCommandRequestId");
443 const StateValue* commandSequence = in.find("commandSequence");
444 if (!pendingRequestId || !pendingRequestId->isString() || !commandSequence || !commandSequence->isInt() ||
445 commandSequence->asInt() < 1)
446 return fail(error, "conversation: pending command state is malformed");
447 if ((waitingCommand && waitingCommand->asBool()) != !pendingRequestId->asString().empty())
448 return fail(error, "conversation: pending command identity does not match waiting state");
449 const bool restoredWaitingCommand = waitingCommand && waitingCommand->asBool();
450 const auto* restoredCommand = restoredAsset->findNode(nodeId);
451 if (restoredWaitingCommand &&
452 (!restoredCommand || restoredCommand->kind != ConversationAsset::Node::Kind::Command))
453 return fail(error, "conversation: pending command cursor is not a command node");
454 asset_ = restoredAsset;
455 nodeId_ = std::move(nodeId);
456 bindings_ = *savedBindings;
457 locals_ = *savedLocals;
458 blocked_ = blocked->asBool();
459 waitingCommand_ = restoredWaitingCommand;
460 pendingCommandRequestId_ = pendingRequestId->asString();
461 commandSequence_ = static_cast<std::uint64_t>(commandSequence->asInt());
462 callStack_ = std::move(restoredStack);
463 lastCommandRequest_.reset();
464 if (waitingCommand_) {
465 CommandRequest request;
466 request.requestId = pendingCommandRequestId_;
467 request.name = restoredCommand->target;
468 request.kind = restoredCommand->commandKind;
469 request.arguments = toCanonicalValue(restoredCommand->arguments);
470 request.bindings = toCanonicalValue(bindings_);
471 request.locals = toCanonicalValue(locals_);
472 request.payment = restoredCommand->payment;
473 request.stateMutations = restoredCommand->stateMutations;
474 lastCommandRequest_ = std::move(request);
475 emit(Event::Kind::Command, restoredCommand, restoredCommand->target);
476 }
477 return true;
478}

508eve::Result<void> ConversationRunner::resumeCommand(const std::string& requestId, StateValue result) {
509 const auto* node = currentNode();
510 if (!node || !blocked_ || !waitingCommand_ ||
511 node->kind != ConversationAsset::Node::Kind::Command)
512 return runnerFailure(eve::DiagnosticCode::DialogueNotWaitingForCommand,
513 "conversation: runner is not waiting for a command", "command");
514 if (requestId.empty() || requestId != pendingCommandRequestId_)
515 return runnerFailure(eve::DiagnosticCode::Conflict,
516 "conversation: command request id is stale or does not match", "command.requestId");
517
518 StateValue before;
519 if (!captureStateImpl(before))
520 return runnerFailure(eve::DiagnosticCode::Failed, "conversation: could not capture command state", "command");
521 if (!node->expression.empty()) locals_.set(node->expression, std::move(result));
522 std::string error;
523 if (enter(node->next, &error) && runUntilBlockedImpl(&error))
524 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
525
526 std::string restoreError;
527 if (!restoreStateImpl(before, &restoreError) && error.empty()) error = restoreError;
528 return runnerFailure(eve::DiagnosticCode::Failed,
529 error.empty() ? "conversation: command resume failed" : std::move(error), "command");
530}

536bool ConversationRunner::advanceImpl(std::string* error) {
537 const auto* node = currentNode();
538 if (!node || !blocked_) return fail(error, "conversation: runner is not blocked");
539 if (waitingCommand_) return fail(error, "conversation: resume the pending command instead");
540 if (node->kind == ConversationAsset::Node::Kind::Choice)
541 return fail(error, "conversation: select a choice route instead");
542 const std::string next = node->next;
543 return enter(next, error) && runUntilBlockedImpl(error);
544}

557bool ConversationRunner::selectImpl(const std::string& routeId, std::string* error) {
558 const auto* node = currentNode();
559 if (node) {
560 for (const auto& route : node->routes) {
561 if (route.id == routeId && (!route.payment.empty() || !route.stateMutations.empty()))
562 return fail(error, "conversation: payment-bearing choice requires DialogueFlow integration");
563 }
564 }
565 auto result = selectRouteForTransaction(routeId);
566 if (result.ok()) {
567 if (error) error->clear();
568 return true;
569 }
570 if (error) *error = result.status().describe();
571 return false;
572}

585eve::Result<void> ConversationRunner::selectRouteForTransaction(const std::string& routeId) {
586 const auto* node = currentNode();
587 if (!node || !blocked_ || node->kind != ConversationAsset::Node::Kind::Choice)
588 return runnerFailure(eve::DiagnosticCode::DialogueNotWaitingForChoice,
589 "conversation: runner is not waiting for a choice", "route");
590 for (const auto& route : node->routes) {
591 if (route.id != routeId) continue;
592 if (!route.condition.isNull()) {
593 if (!conditionEvaluator_)
594 return runnerFailure(eve::DiagnosticCode::PreconditionViolation,
595 "conversation: structured choice requires a condition evaluator",
596 "route.condition");
597 eve::decision::ConditionResult result = conditionEvaluator_(route.condition);
598 lastConditionResult_ = result;
599 if (!result.passed())
600 return runnerFailure(eve::DiagnosticCode::DialogueConditionRejected,
601 "conversation: choice condition rejected (" +
602 std::string(eve::decision::conditionReasonCodeName(result.reasonCode())) + ")",
603 "route.condition");
604 }
605
606 StateValue before;
607 if (!captureStateImpl(before))
608 return runnerFailure(eve::DiagnosticCode::Failed, "conversation: could not capture choice state", "route");
609 std::string error;
610 if (enter(route.second, &error) && runUntilBlockedImpl(&error))
611 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
612
613 std::string restoreError;
614 if (!restoreStateImpl(before, &restoreError) && error.empty()) error = restoreError;
615 return runnerFailure(eve::DiagnosticCode::Failed,
616 error.empty() ? "conversation: choice selection failed" : std::move(error), "route");
617 }
618 return runnerFailure(eve::DiagnosticCode::DialogueRouteNotFound, "conversation: unknown choice '" + routeId + "'",
619 "route");
620}
