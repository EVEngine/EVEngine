15Result<structured_detail::MeshPart> decodeBinaryStl(const std::vector<std::uint8_t>& bytes,
16 const AssetImportLimits& limits) {
17 using Part = Result<structured_detail::MeshPart>;
18 auto invalid = [](std::string message) {
19 return Part::failure(
20 Diagnostic::error(DiagnosticCode::ParseError, std::move(message), {}, {}, "asset.import.scad"));
21 };
22 if (bytes.size() < 84 || bytes.size() > limits.maximumSourceBytes)
23 return invalid("binary STL is missing or outside budget");
24 auto word = [&](std::size_t offset) {
25 return std::uint32_t(bytes[offset]) | (std::uint32_t(bytes[offset + 1]) << 8) |
26 (std::uint32_t(bytes[offset + 2]) << 16) | (std::uint32_t(bytes[offset + 3]) << 24);
27 };
28 const auto count = word(80);
29 if (count == 0 || 84ull + std::uint64_t(count) * 50 != bytes.size() ||
30 std::uint64_t(count) * 3 > limits.maximumVerticesPerPrimitive ||
31 std::uint64_t(count) * 3 > limits.maximumIndicesPerPrimitive ||
32 std::uint64_t(count) * 84 + 24 > limits.maximumDecodedBytes)
33 return invalid("binary STL triangle count/length/budget is invalid");
34 structured_detail::MeshPart part;
35 for (std::uint32_t i = 0; i < count; ++i) {
36 const auto offset = std::size_t(84) + std::size_t(i) * 50;
37 if (bytes[offset + 48] != 0 || bytes[offset + 49] != 0)
38 return invalid("coloured/extended STL is unsupported; export plain binary STL");
39 for (std::size_t j = 0; j < 3; ++j)
40 if (!std::isfinite(std::bit_cast<float>(word(offset + j * 4)))) return invalid("nonfinite STL normal");
41 std::array<std::array<double, 3>, 3> triangle{};
42 for (std::size_t vertex = 0; vertex < 3; ++vertex) {
43 const auto base = offset + 12 + vertex * 12;
44 const double x = std::bit_cast<float>(word(base));
45 const double y = std::bit_cast<float>(word(base + 4));
46 const double z = std::bit_cast<float>(word(base + 8));
47 triangle[vertex] = {x * 0.001, z * 0.001, -y * 0.001};
48 }
49 auto added = structured_detail::appendTriangle(part.mesh, triangle, limits);
50 if (!added) return Part::failure(added.status());
51 }
52 return Part::success(std::move(part));
53}

60Result<Value::Object> parseScadRig(const std::vector<std::uint8_t>& bytes, const AssetImportLimits& limits,
61 const std::set<std::string>* requiredPartNames) {
62 using Output = Result<Value::Object>;
63 auto invalid = [](std::string message) {
64 return Output::failure(
65 Diagnostic::error(DiagnosticCode::ParseError, std::move(message), {}, {}, "asset.import.scad"));
66 };
67 if (bytes.empty() || bytes.size() > limits.maximumSourceBytes)
68 return invalid("ScadRig document is missing or outside budget");
69 auto parsed = Value::fromJson(std::string_view(reinterpret_cast<const char*>(bytes.data()), bytes.size()));
70 if (!parsed) return Output::failure(parsed.status());
71 const auto* root = parsed.value().getIf<Value::Object>();
72 if (!root) return invalid("ScadRig root must be a JSON object");
73 const auto* schema = parsed.value().find("schema");
74 const auto* version = parsed.value().find("schemaVersion");
75 const auto* unknown = parsed.value().find("unknownFields");
76 if (!schema || !schema->isString() || schema->asString() != "eve.scad-rig" || !version || !version->isInt64() ||
77 version->asInt() != 1 || !unknown || !unknown->isString() ||
78 (unknown->asString() != "reject" && unknown->asString() != "preserve"))
79 return invalid("ScadRig requires schema eve.scad-rig/1 and unknownFields reject|preserve");
80 if (unknown->asString() == "reject") {
81 static const std::unordered_set<std::string> known = {"schema", "schemaVersion", "unknownFields",
82 "coordinateSystem", "joints", "parts"};
83 for (const auto& key : parsed.value().keys())
84 if (!known.count(key)) return invalid("ScadRig unknown field rejected: " + key);
85 }
86 const auto* coordinate = parsed.value().find("coordinateSystem");
87 if (coordinate && (!coordinate->isString() || coordinate->asString() != "openscad-mm-z-up"))
88 return invalid("ScadRig coordinateSystem must be openscad-mm-z-up when present");
89 const auto* jointsValue = parsed.value().find("joints");
90 const auto* partsValue = parsed.value().find("parts");
91 if (!jointsValue || !jointsValue->isArray() || jointsValue->arraySize() == 0 || !partsValue ||
92 !partsValue->isArray() || partsValue->arraySize() == 0)
93 return invalid("ScadRig requires non-empty joints and parts arrays");
94 if (jointsValue->arraySize() > limits.maximumAssets || partsValue->arraySize() > limits.maximumAssets)
95 return invalid("ScadRig joint/part count exceeds budget");
96
97 std::unordered_map<std::string, std::string> parents;
98 std::unordered_set<std::string> jointNames;
99 std::size_t roots = 0;
100 for (std::size_t i = 0; i < jointsValue->arraySize(); ++i) {
101 const auto& jointValue = jointsValue->at(i);
102 if (!jointValue.isObject()) return invalid("ScadRig joint entries must be objects");
103 if (unknown->asString() == "reject") {
104 static const std::unordered_set<std::string> known = {"name", "parent", "pivot", "axis", "limits"};
105 for (const auto& key : jointValue.keys())
106 if (!known.count(key)) return invalid("ScadRig joint unknown field rejected: " + key);
107 }
108 const auto* name = jointValue.find("name");
109 if (!name || !name->isString() || name->asString().empty() ||
110 name->asString().size() > limits.maximumStringBytes)
111 return invalid("ScadRig joint name is required and bounded");
112 if (!jointNames.insert(name->asString()).second) return invalid("ScadRig joint names must be unique");
113 const auto* parent = jointValue.find("parent");
114 if (!parent || parent->isNull()) {
115 parents.emplace(name->asString(), std::string{});
116 ++roots;
117 } else if (parent->isString() && !parent->asString().empty()) {
118 parents.emplace(name->asString(), parent->asString());
119 } else {
120 return invalid("ScadRig joint parent must be null or a non-empty string");
121 }
122 const auto* pivot = jointValue.find("pivot");
123 const auto* axis = jointValue.find("axis");
124 if (!pivot || !pivot->isArray() || pivot->arraySize() != 3 || !axis || !axis->isArray() ||
125 axis->arraySize() != 3)
126 return invalid("ScadRig joint pivot/axis must be length-3 numeric arrays");
127 for (std::size_t axisIndex = 0; axisIndex < 3; ++axisIndex) {
128 if (!pivot->at(axisIndex).isNumeric() || !std::isfinite(numeric(pivot->at(axisIndex))) ||
129 !axis->at(axisIndex).isNumeric() || !std::isfinite(numeric(axis->at(axisIndex))))
130 return invalid("ScadRig joint pivot/axis values must be finite numbers");
131 }
132 const double ax = numeric(axis->at(0)), ay = numeric(axis->at(1)), az = numeric(axis->at(2));
133 if (ax * ax + ay * ay + az * az <= 0) return invalid("ScadRig joint axis must be non-zero");
134 if (const auto* limitsValue = jointValue.find("limits")) {
135 if (!limitsValue->isArray() || limitsValue->arraySize() != 2 || !limitsValue->at(0).isNumeric() ||
136 !limitsValue->at(1).isNumeric() || !std::isfinite(numeric(limitsValue->at(0))) ||
137 !std::isfinite(numeric(limitsValue->at(1))) ||
138 numeric(limitsValue->at(0)) > numeric(limitsValue->at(1)))
139 return invalid("ScadRig joint limits must be [min,max] degrees with min <= max");
140 }
141 }
142 if (roots != 1) return invalid("ScadRig requires exactly one root joint (parent null)");
143 for (const auto& [name, parent] : parents) {
144 if (parent.empty()) continue;
145 if (!jointNames.count(parent)) return invalid("ScadRig joint parent is unknown: " + parent);
146 std::unordered_set<std::string> seen{name};
147 auto cursor = parent;
148 while (!cursor.empty()) {
149 if (!seen.insert(cursor).second) return invalid("ScadRig joint parent graph contains a cycle");
150 cursor = parents[cursor];
151 }
152 }
153
154 std::set<std::string> partNames;
155 for (std::size_t i = 0; i < partsValue->arraySize(); ++i) {
156 const auto& partValue = partsValue->at(i);
157 if (!partValue.isObject()) return invalid("ScadRig part entries must be objects");
158 if (unknown->asString() == "reject") {
159 static const std::unordered_set<std::string> known = {"name", "joint"};
160 for (const auto& key : partValue.keys())
161 if (!known.count(key)) return invalid("ScadRig part unknown field rejected: " + key);
162 }
163 const auto* name = partValue.find("name");
164 const auto* joint = partValue.find("joint");
165 if (!name || !name->isString() || name->asString().empty() ||
166 name->asString().size() > limits.maximumStringBytes || !joint || !joint->isString() ||
167 !jointNames.count(joint->asString()))
168 return invalid("ScadRig part name/joint binding is invalid");
169 if (!partNames.insert(name->asString()).second) return invalid("ScadRig part names must be unique");
170 }
171 if (requiredPartNames && *requiredPartNames != partNames)
172 return invalid("ScadRig parts must match compiled mesh part names exactly");
173 return Output::success(*root);
174}

176Result<void> attachScadRig(PreparedAssetImport& prepared, const ImportPackageIdentity& identity,
177 const Value::Object& rig) {
178 using Output = Result<void>;
179 const auto rigId = identity.packageId.child("scad:rig");
180 auto reference = detail::assetRef(rigId);
181 if (!reference) return Output::failure(reference.status());

200Result<PreparedAssetImport> prepareScadImport(const ScadImportRequest& request) {
201 using Output = Result<PreparedAssetImport>;
202 auto invalid = [](std::string message) {
203 return Output::failure(
204 Diagnostic::error(DiagnosticCode::ParseError, std::move(message), {}, {}, "asset.import.scad"));
205 };
206 if (request.sourceBytes.empty() || request.sourceBytes.size() > request.limits.maximumSourceBytes)
207 return invalid("SCAD source is missing or outside budget");
208
209 std::map<std::string, std::vector<std::uint8_t>> stls = request.partStls;
210 if (stls.empty()) {
211 if (request.binaryStl.empty()) return invalid("SCAD binary STL is missing");
212 stls.emplace("default", request.binaryStl);
213 } else if (!request.binaryStl.empty()) {
214 return invalid("provide either binaryStl or partStls, not both");
215 }
216 if (stls.size() > request.limits.maximumAssets) return invalid("SCAD part count exceeds budget");
217
218 std::map<std::string, structured_detail::MeshPart> parts;
219 std::set<std::string> partNames;
220 for (const auto& [name, bytes] : stls) {
221 if (name.empty() || name.size() > request.limits.maximumStringBytes) return invalid("SCAD part name is invalid");
222 if (!partNames.insert(name).second) return invalid("SCAD part names must be unique");
223 auto decoded = decodeBinaryStl(bytes, request.limits);
224 if (!decoded) return Output::failure(decoded.status());
225 parts.emplace(name, std::move(decoded).takeValue());
226 }
227
228 std::vector<ImportFinding> findings{
229 {request.sourceName, "SCAD.geometry", ImportDisposition::Baked,
230 "OpenSCAD binary STL triangulated; millimetres/Z-up converted to metres/Y-up"},
231 {request.sourceName, "SCAD.nonGeometry", ImportDisposition::Unsupported,
232 "STL does not retain colours or animation; root source retained, external dependency closure not archived"}};
233
234 std::optional<Value::Object> rig;
235 if (!request.rigBytes.empty()) {
236 const bool multiPart = !(partNames.size() == 1 && partNames.count("default"));
237 auto parsed = parseScadRig(request.rigBytes, request.limits, multiPart ? &partNames : nullptr);
238 if (!parsed) return Output::failure(parsed.status());
239 rig = std::move(parsed).takeValue();
240 findings.push_back({request.sourceName, "SCAD.rig", ImportDisposition::Translated,
241 "eve.scad-rig/1 joint tree retained as metadata; static prefab pose stays identity"});
242 }
243
244 std::map<std::string, std::vector<std::uint8_t>> sources{{request.sourceName, request.sourceBytes}};
245 if (!request.rigBytes.empty()) sources.emplace("scad-rig.json", request.rigBytes);
246
247 auto prepared = structured_detail::packageModel(request.package, request.sourceName, "scad", std::move(parts),
248 sources, std::move(findings), request.limits);
249 if (!prepared) return prepared;
250 if (rig) {
251 prepared.value().entries.erase(
252 std::remove_if(prepared.value().entries.begin(), prepared.value().entries.end(),
253 [](const asset::EvaArchiveEntry& entry) { return entry.path == "reports/import.json"; }),
254 prepared.value().entries.end());
255 auto attached = attachScadRig(prepared.value(), request.package, *rig);
256 if (!attached) return Output::failure(attached.status());
257 auto report = detail::finalizeImportReport(prepared.value(), request.package, "scad", "1");
258 if (!report) return Output::failure(report.status());
259 }
260 return prepared;
261}
