17eve::Result<void> Material::setVirtualTexture(Texture *albedoAtlas, Texture *normalAtlas,
18 Texture *pageTable, int pageCountX, int pageCountY,
19 int atlasSlotsX, int atlasSlotsY,
20 float borderFraction) {
21 if (!albedoAtlas || !normalAtlas || !pageTable)
22 return eve::Result<void>::failure(
23 invalidMaterial("virtual texture requires albedo, normal, and page-table textures"));
24 if (pageCountX <= 0 || pageCountY <= 0 || atlasSlotsX <= 0 || atlasSlotsY <= 0)
25 return eve::Result<void>::failure(
26 invalidMaterial("virtual texture page and atlas dimensions must be positive"));
27 if (!std::isfinite(borderFraction) || borderFraction < 0.f || borderFraction >= 0.5f)
28 return eve::Result<void>::failure(
29 invalidMaterial("virtual texture border fraction must be in [0, 0.5)"));
30 albedo_ = albedoAtlas;
31 normal_ = normalAtlas;
32 height_ = pageTable;
33 virtualPageCountX_ = pageCountX;
34 virtualPageCountY_ = pageCountY;
35 virtualAtlasSlotsX_ = atlasSlotsX;
36 virtualAtlasSlotsY_ = atlasSlotsY;
37 virtualBorderFraction_ = borderFraction;
38 virtualTextureEnabled_ = true;
39 return eve::Result<void>::success();
40}

162void Material::bind(Graphics &gfx) const {
163 const BlendMode effectiveBlend =
164 albedo_ && albedo_->hasPremultipliedAlpha() && blendMode_ == BlendMode::Alpha
165 ? BlendMode::Premultiplied
166 : blendMode_;
167 gfx.setMesh3DSurface(surfaceMode_, effectiveBlend, depthWrite_, doubleSided_, alphaCutoff_,
168 alphaTechnique_);
169 gfx.setMesh3DMaterial(metallic_, roughness_);
170 gfx.setMesh3DTexCellBomb(texBombScale_, texBombStrength_, texBombRot_);
171 gfx.setMesh3DNormalTexture(normal_);
172 gfx.setMesh3DHeightTexture(height_);
173 gfx.setMesh3DVirtualTexture(virtualTextureEnabled_, virtualPageCountX_, virtualPageCountY_,
174 virtualAtlasSlotsX_, virtualAtlasSlotsY_,
175 virtualBorderFraction_);
176 gfx.setMesh3DParallax(parallaxScale_, parallaxMinLayers_, parallaxMaxLayers_);
177 gfx.setMesh3DShadowReceive(receiveShadow_);
178
179 if (!receiveLight_ || shadingModel_ == "unlit") {
180 // Cheap toggle only: never clobber the clustered light table that the
181 // frame uploaded once (see Graphics::setMesh3DClusteredActive).
182 gfx.setMesh3DClusteredActive(false);
183 Lighting3DPack none{};
184 none.count = 0;
185 none.ambient = glm::vec4(1.f, 1.f, 1.f, 0.f);
186 gfx.setMesh3DLighting(none);
187 }
188
189 if (shader_ && !params_.empty()) {
190 for (const auto &kv : params_) {
191 if (shader_->hasUniform(kv.first)) shader_->sendFloat(kv.first, kv.second);
192 }
193 }
194}
