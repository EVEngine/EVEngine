15eve::Result<const eve::Value::Object*> objectWithFields(const eve::Value& value,
16 std::initializer_list<std::string_view> fields,
17 const std::string& path) {
18 const auto* object = value.getIf<eve::Value::Object>();
19 if (object == nullptr)
20 return invalid<const eve::Value::Object*>(eve::DiagnosticCode::TypeMismatch, "expected object", path);
21 if (object->size() != fields.size())
22 return invalid<const eve::Value::Object*>(eve::DiagnosticCode::InvalidArgument,
23 "object contains missing or unknown fields", path);
24 for (const auto field : fields)
25 if (!object->contains(std::string(field)))
26 return invalid<const eve::Value::Object*>(eve::DiagnosticCode::InvalidArgument,
27 "object contains missing or unknown fields", path);
28 return eve::Result<const eve::Value::Object*>::success(object);
29}

58eve::Result<std::vector<std::string>> stringsValue(const eve::Value& value, const std::string& path) {
59 const auto* array = value.getIf<eve::Value::Array>();
60 if (array == nullptr)
61 return invalid<std::vector<std::string>>(eve::DiagnosticCode::TypeMismatch, "expected string array", path);
62 std::vector<std::string> strings;
63 strings.reserve(array->size());
64 for (std::size_t index = 0; index < array->size(); ++index) {
65 const auto* item = (*array)[index].getIf<std::string>();
66 if (item == nullptr)
67 return invalid<std::vector<std::string>>(eve::DiagnosticCode::TypeMismatch, "expected string",
68 path + "[" + std::to_string(index) + "]");
69 strings.push_back(*item);
70 }
71 return eve::Result<std::vector<std::string>>::success(std::move(strings));
72}

102eve::Result<SettlementRule> decodeRule(const eve::Value& value, const std::string& path) {
103 auto object = objectWithFields(value,
104 {"filter", "id", "operation", "priority", "source", "stacks", "stage", "value",
105 "valuePerExtraStack", "when"},
106 path);
107 if (!object) return eve::Result<SettlementRule>::failure(object.status());
108 auto id = stringValue(*object.value(), "id", path);
109 auto source = stringValue(*object.value(), "source", path);
110 auto stage = stringValue(*object.value(), "stage", path);
111 auto operation = stringValue(*object.value(), "operation", path);
112 auto priority = integerValue(*object.value(), "priority", path);
113 auto stacks = integerValue(*object.value(), "stacks", path);
114 auto operand = numberValue(*object.value(), "value", path);
115 auto perStack = numberValue(*object.value(), "valuePerExtraStack", path);
116 auto when = stringValue(*object.value(), "when", path);
117 if (!id) return eve::Result<SettlementRule>::failure(id.status());
118 if (!source) return eve::Result<SettlementRule>::failure(source.status());
119 if (!stage) return eve::Result<SettlementRule>::failure(stage.status());
120 if (!operation) return eve::Result<SettlementRule>::failure(operation.status());
121 if (!priority) return eve::Result<SettlementRule>::failure(priority.status());
122 if (!stacks) return eve::Result<SettlementRule>::failure(stacks.status());
123 if (!operand) return eve::Result<SettlementRule>::failure(operand.status());
124 if (!perStack) return eve::Result<SettlementRule>::failure(perStack.status());
125 if (!when) return eve::Result<SettlementRule>::failure(when.status());
126 if (priority.value() < std::numeric_limits<int>::min() || priority.value() > std::numeric_limits<int>::max())
127 return invalid<SettlementRule>(eve::DiagnosticCode::InvalidArgument, "priority is out of range",
128 path + ".priority");
129 if (stacks.value() <= 0 || stacks.value() > std::numeric_limits<std::uint32_t>::max())

131 path + ".stacks");

133 auto decodedOperation = operationValue(operation.value(), path + ".operation");
134 if (!decodedStage) return eve::Result<SettlementRule>::failure(decodedStage.status());
135 if (!decodedOperation) return eve::Result<SettlementRule>::failure(decodedOperation.status());

138 path + ".filter");

141 auto required = stringsValue(filter.value()->at("requiredTags"), path + ".filter.requiredTags");
142 auto excluded = stringsValue(filter.value()->at("excludedTags"), path + ".filter.excludedTags");
143 if (!kinds) return eve::Result<SettlementRule>::failure(kinds.status());
144 if (!required) return eve::Result<SettlementRule>::failure(required.status());
145 if (!excluded) return eve::Result<SettlementRule>::failure(excluded.status());

147 SettlementRule rule;
148 rule.id = std::move(id).takeValue();
149 rule.source = std::move(source).takeValue();
150 rule.stage = decodedStage.value();
151 rule.priority = static_cast<int>(priority.value());
152 rule.operation = decodedOperation.value();

173eve::Result<void> SettlementRuleSet::configureJson(std::string_view json) {
174 auto parsed = eve::Value::fromJson(json);
175 if (!parsed) return eve::Result<void>::failure(parsed.status());
176 auto root = objectWithFields(parsed.value(), {"conditionLanguageVersion", "rules", "schema", "version"},
177 "document");
178 if (!root) return eve::Result<void>::failure(root.status());
179 auto schema = stringValue(*root.value(), "schema", "document");
180 auto version = integerValue(*root.value(), "version", "document");
181 auto conditionVersion = integerValue(*root.value(), "conditionLanguageVersion", "document");
182 if (!schema) return eve::Result<void>::failure(schema.status());
183 if (!version) return eve::Result<void>::failure(version.status());
184 if (!conditionVersion) return eve::Result<void>::failure(conditionVersion.status());
185 if (schema.value() != "settlement.rules")
186 return invalid<void>(eve::DiagnosticCode::InvalidArgument, "unexpected settlement rules schema",
187 "document.schema");
188 if (version.value() != schemaVersion())
189 return invalid<void>(eve::DiagnosticCode::UnknownVersion, "unsupported settlement rules version",
190 "document.version");
191 if (conditionVersion.value() != 1)
192 return invalid<void>(eve::DiagnosticCode::UnknownVersion, "unsupported condition language version",
193 "document.conditionLanguageVersion");
194 const auto* encodedRules = root.value()->at("rules").getIf<eve::Value::Array>();
195 if (encodedRules == nullptr)
196 return invalid<void>(eve::DiagnosticCode::TypeMismatch, "expected rule array", "document.rules");
197 std::vector<SettlementRule> rules;
198 rules.reserve(encodedRules->size());
199 for (std::size_t index = 0; index < encodedRules->size(); ++index) {
200 auto rule = decodeRule((*encodedRules)[index], "document.rules[" + std::to_string(index) + "]");
201 if (!rule) return eve::Result<void>::failure(rule.status());
202 rules.push_back(std::move(rule).takeValue());
203 }
204 return configure(std::move(rules));
205}
