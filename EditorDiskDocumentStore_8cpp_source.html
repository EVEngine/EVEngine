48std::filesystem::path DiskAtomicDocumentStore::resolve(const std::string& resourceUri) const {
49 if (root_.empty()) return {};
50 std::string relative;
51 if (resourceUri.rfind("content://", 0) == 0)
52 relative = "Content/" + resourceUri.substr(10);
53 else if (resourceUri.rfind("autosave://", 0) == 0)
54 relative = "Saved/Autosave/" + resourceUri.substr(11);
55 else
56 return {};
57 if (relative.empty()) return {};
58
59 std::filesystem::path candidate(relative);
60 if (candidate.is_absolute()) return {};
61 for (const auto& part : candidate)
62 if (part == "..") return {};
63
64 candidate = (root_ / candidate).lexically_normal();
65 const std::filesystem::path scoped = candidate.lexically_relative(root_);
66 if (scoped.empty()) return {};
67 for (const auto& part : scoped)
68 if (part == "..") return {};
69 return candidate;
70}

72EditorResult<StoredDocument> DiskAtomicDocumentStore::read(const std::string& resourceUri) const {
73 const std::filesystem::path path = resolve(resourceUri);
74 if (path.empty()) return failure(EditorStatus::Rejected, "editor.document.invalid-uri", "Rejected resource URI");
75 std::ifstream input(path, std::ios::binary);
76 if (!input)
77 return failure(EditorStatus::NotFound, "editor.document.store-not-found",
78 "Document resource is not persisted: " + resourceUri);
79 const std::string json{std::istreambuf_iterator<char>(input), std::istreambuf_iterator<char>()};
80 EditorResult<EditorValue> parsed = editorValueFromJson(json);
81 if (!parsed.ok() || !parsed.ok())
82 return failure(EditorStatus::Failed, "editor.document.invalid-envelope", "Document JSON envelope is invalid");
83
84 const EditorValue* content = member(parsed.value(), "content");
85 const Revision revision = revisionMember(parsed.value(), "revision");
86 const auto* schema = member(parsed.value(), "schemaVersion");
87 const auto* schemaInteger = schema ? schema->getIf<std::int64_t>() : nullptr;
88 if (!content || !schemaInteger || *schemaInteger != 1)
89 return failure(EditorStatus::Failed, "editor.document.unsupported-schema",
90 "Document envelope schemaVersion must be 1");
91
92 StoredDocument stored;
93 stored.content = *content;
94 stored.revision = revision;
95 stored.contentHash = editorValueContentHash(stored.content);
96 return eve::editing::applied<StoredDocument>(std::move(stored));
97}

99EditorResult<StoredDocument> DiskAtomicDocumentStore::compareAndSwap(const std::string& resourceUri,
100 Revision expectedRevision,
101 const std::string& expectedContentHash,
102 const EditorValue& content) {
103 const std::filesystem::path path = resolve(resourceUri);
104 if (path.empty()) return failure(EditorStatus::Rejected, "editor.document.invalid-uri", "Rejected resource URI");
105
106 StoredDocument current;
107 EditorResult<StoredDocument> existing = read(resourceUri);
108 if (existing.ok())
109 current = existing.value();
110 else if (existing.code() != EditorStatus::NotFound)
111 return existing;
112 else
113 current.contentHash.clear();
114 if (current.revision != expectedRevision || current.contentHash != expectedContentHash)
115 return failure(EditorStatus::Conflict, "editor.document.disk-conflict",
116 "Document content or revision changed before save");
117
118 StoredDocument next;
119 next.content = content;
120 next.revision = current.revision + 1;
121 next.contentHash = editorValueContentHash(content);
122 EditorValue::Object envelope;
123 envelope["schemaVersion"] = EditorValue(1);
124 envelope["revision"] = EditorValue(static_cast<std::int64_t>(next.revision));
125 envelope["contentHash"] = EditorValue(next.contentHash);
126 envelope["content"] = content;
127
128 std::error_code ec;
129 std::filesystem::create_directories(path.parent_path(), ec);
130 if (ec) return failure(EditorStatus::Failed, "editor.document.create-directory", ec.message());
131 const std::filesystem::path temporary = path.string() + ".tmp." + std::to_string(++sequence_);
132 {
133 std::ofstream output(temporary, std::ios::binary | std::ios::trunc);
134 if (!output) return failure(EditorStatus::Failed, "editor.document.temp-open", "Cannot open temp file");
135 output << editorValueToJson(EditorValue(std::move(envelope)));
136 output.flush();
137 if (!output.good()) {
138 output.close();
139 std::filesystem::remove(temporary, ec);
140 return failure(EditorStatus::Failed, "editor.document.temp-write", "Cannot flush temp file");
141 }
142 }
143 if ((beforeReplace_ && !beforeReplace_(path)) || !replace(temporary, path)) {
144 std::filesystem::remove(temporary, ec);
145 return failure(EditorStatus::Failed, "editor.document.atomic-replace", "Atomic replacement failed");
146 }
147 return eve::editing::applied<StoredDocument>(std::move(next));
148}

174EditorResult<StoredDocument> AutosaveService::writeDraft(const DocumentSnapshot& snapshot, const EditorValue& content) {
175 if (!store_ || snapshot.id.empty())
176 return eve::editing::failed<StoredDocument>(EditorStatus::Rejected, RuleId("editor.autosave.invalid"),
177 "Autosave store and document are required");
178 const std::string uri = draftUri(snapshot.id);
179 Revision revision = 0;
180 std::string hash;
181 EditorResult<StoredDocument> existing = store_->read(uri);
182 if (existing.ok()) {
183 revision = existing.value().revision;
184 hash = existing.value().contentHash;
185 } else if (existing.code() != EditorStatus::NotFound) {
186 return existing;
187 }
188 EditorValue::Object draft;
189 draft["document"] = EditorValue(snapshot.id.value());
190 draft["baseDiskHash"] = EditorValue(snapshot.diskContentHash);
191 draft["schemaVersion"] = EditorValue(1);
192 draft["editRevision"] = EditorValue(static_cast<std::int64_t>(snapshot.revision.edit));
193 draft["content"] = content;
194 return store_->compareAndSwap(uri, revision, hash, EditorValue(std::move(draft)));
195}

197EditorResult<AutosaveDraft> AutosaveService::readDraft(const DocumentId& document) const {
198 if (!store_ || document.empty()) return draftFailure("editor.autosave.invalid", "Autosave store is required");
199 const EditorResult<StoredDocument> stored = store_->read(draftUri(document));
200 if (!stored.ok()) return EditorResult<AutosaveDraft>::failure(stored.status());
201 const EditorValue& root = stored.value().content;
202 const EditorValue* content = member(root, "content");
203 if (!content || revisionMember(root, "schemaVersion") != 1)
204 return draftFailure("editor.autosave.unsupported-schema", "Autosave draft schemaVersion must be 1");
205 AutosaveDraft draft;
206 draft.document = DocumentId(stringMember(root, "document"));
207 draft.baseDiskHash = stringMember(root, "baseDiskHash");
208 draft.schemaVersion = 1;
209 draft.editRevision = revisionMember(root, "editRevision");
210 draft.content = *content;
211 if (draft.document != document)
212 return draftFailure("editor.autosave.document-mismatch", "Autosave draft belongs to another document");
213 return eve::editing::applied<AutosaveDraft>(std::move(draft));
214}
