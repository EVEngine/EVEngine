65PolicyGraph makePolicyGraph(int f, int h, int a, bool training) {
66 Builder b;

68 const int weights = b.input(0, 1, n1 + n2 + n3);

70 auto matrix = [&](int start, int count, int rows, int cols) {
71 int flat = b.slice(weights, 1, start, start + count);
72 return b.node(OpType::Reshape, rows, cols, flat);
73 };
