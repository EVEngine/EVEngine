33eve::Result<CraftingRecipe> parseCraftRecipe(const eve::Value& root) {
34 const auto* object = root.getIf<eve::Value::Object>();
35 if (object == nullptr)
36 return failure<CraftingRecipe>(eve::DiagnosticCode::InvalidArgument, "craft recipe must be an object",
37 "recipe");
38 const auto id = object->find("id");
39 const auto process = object->find("process");
40 const auto ingredients = object->find("ingredients");
41 const auto outputs = object->find("outputs");
42 const auto duration = object->find("duration");
43 if (id == object->end() || !id->second.isString() || id->second.asString().empty() ||
44 ingredients == object->end() || outputs == object->end() || !outputs->second.isArray() ||
45 duration == object->end() || !duration->second.isNumeric())
46 return failure<CraftingRecipe>(eve::DiagnosticCode::InvalidArgument,
47 "craft recipe requires id, ingredients, outputs and duration", "recipe");
48 auto ingredientsJson = ingredients->second.toJson();
49 if (!ingredientsJson) return eve::Result<CraftingRecipe>::failure(ingredientsJson.status());
50 auto cost = transaction::AtomicBatch::parseCost(ingredientsJson.value());
51 if (!cost) return eve::Result<CraftingRecipe>::failure(cost.status());
52 auto seconds = Duration::fromSeconds(duration->second.isInt64() ? static_cast<double>(duration->second.asInt())
53 : duration->second.asDouble());
54 if (!seconds) return eve::Result<CraftingRecipe>::failure(seconds.status());
55 CraftingRecipe recipe;
56 recipe.id = id->second.asString();
57 recipe.process = (process != object->end() && process->second.isString() && !process->second.asString().empty())
58 ? process->second.asString()
59 : "craft";
60 recipe.ingredients = std::move(cost).takeValue();
61 recipe.duration = std::move(seconds).takeValue();
62 const auto* outputValues = outputs->second.getIf<eve::Value::Array>();
63 for (const auto& value : *outputValues) {
64 const auto* entry = value.getIf<eve::Value::Object>();
65 if (entry == nullptr)
66 return failure<CraftingRecipe>(eve::DiagnosticCode::InvalidArgument, "craft output must be an object",
67 "outputs");
68 const auto item = entry->find("item");
69 const auto quantity = entry->find("quantity");
70 if (item == entry->end() || !item->second.isString() || item->second.asString().empty() ||
71 quantity == entry->end() || !quantity->second.isInt64() || quantity->second.asInt() <= 0)
72 return failure<CraftingRecipe>(eve::DiagnosticCode::InvalidArgument,
73 "craft outputs require item and positive quantity", "outputs");
74 recipe.outputs.push_back({item->second.asString(), static_cast<int>(quantity->second.asInt())});
75 }
76 if (recipe.outputs.empty())
77 return failure<CraftingRecipe>(eve::DiagnosticCode::InvalidArgument, "craft recipe has no outputs", "outputs");
78 return eve::Result<CraftingRecipe>::success(std::move(recipe));
79}

237eve::Result<int> ForgeBench::settle(std::string_view taskId) {
238 if (bag_ == nullptr) return failure<int>(eve::DiagnosticCode::InvalidArgument, "forge bench is not bound", "bag");
239 auto queueView = queue();
240 if (!queueView) return eve::Result<int>::failure(queueView.status());
241 auto task = queueView.value().get().find(taskId);
242 if (!task) return failure<int>(eve::DiagnosticCode::NotFound, "forge task was not found", "taskId");
243 auto settled = (task->get().kind == "upgrade") ? EquipmentUpgrade::settle(queueView.value().get(), *bag_, taskId)
244 : Crafting::settle(queueView.value().get(), *bag_, taskId);
245 if (settled) (void)syncEquipment();
246 return settled;
247}

249eve::Result<int> ForgeBench::settleReady() {
250 if (bag_ == nullptr) return failure<int>(eve::DiagnosticCode::InvalidArgument, "forge bench is not bound", "bag");
251 auto queueView = queue();
252 if (!queueView) return eve::Result<int>::failure(queueView.status());
253 std::vector<std::string> ready;
254 for (int i = 0; i < queueView.value().get().taskCount(); ++i) {
255 const auto task = queueView.value().get().taskAt(i);
256 if (!task) continue;
257 if (task->get().state == production::TaskState::ReadyToSettle ||
258 task->get().state == production::TaskState::SettlementFailed)
259 ready.push_back(task->get().id);
260 }
261 int applied = 0;
262 for (const std::string& taskId : ready) {
263 auto settled = settle(taskId);
264 if (!settled) return settled;
265 if (settled.status().code() == eve::StatusCode::Applied) ++applied;
266 }
267 return eve::Result<int>::success(applied, eve::Status::success(eve::StatusCode::Applied));
268}
