29std::optional<ItemView> findItem(const action::ActionTimeline& timeline, const LogicalId& itemId) {
30 for (const auto& track : timeline.tracks) {
31 for (const auto& notify : track.notifies)
32 if (notify.id == itemId)
33 return ItemView{track.id, notify.id, notify.type, notify.time,
34 notify.time, notify.payload, false, track.locked};
35 for (const auto& state : track.states)
36 if (state.id == itemId)
37 return ItemView{track.id, state.id, state.type, state.start,
38 state.end, state.payload, true, track.locked};
39 }
40 return std::nullopt;

68float ActionTimelineWidget::timeToX(Duration time) const noexcept {

75Duration ActionTimelineWidget::xToTime(float x) const noexcept {
76 const float clamped = std::clamp(x, labelWidth_, width_);

78 const auto duration = editor_.target().timeline().duration.nanoseconds();
79 return Duration::fromNanoseconds(static_cast<std::int64_t>(std::llround(static_cast<double>(fraction) * duration)));

82TimelineWidgetLayout ActionTimelineWidget::layout() const {
83 TimelineWidgetLayout result;
84 result.width = width_;
85 result.height = rowHeight_ * static_cast<float>(editor_.target().timeline().tracks.size());
86 result.playheadX = timeToX(editor_.previewTime());
87 const auto selected = editor_.selectedItemIds();
88 auto isSelected = [&](const LogicalId& id) {
89 return std::find(selected.begin(), selected.end(), id) != selected.end();
90 };
91 for (std::size_t row = 0; row < editor_.target().timeline().tracks.size(); ++row) {
92 const auto& track = editor_.target().timeline().tracks[row];
93 const float top = static_cast<float>(row) * rowHeight_;
94 for (const auto& notify : track.notifies) {
95 Duration time = notify.time;
96 if (drag_ && drag_->itemId == notify.id) time = drag_->previewStart;
97 const float center = timeToX(time);
98 result.items.push_back({track.id, notify.id, notify.type, false, isSelected(notify.id),
99 center - kNotifyWidth * 0.5f, center + kNotifyWidth * 0.5f, top + 3.0f,
100 top + rowHeight_ - 3.0f});
101 }
102 for (const auto& state : track.states) {
103 Duration start = state.start;
104 Duration end = state.end;
105 if (drag_ && drag_->itemId == state.id) {
106 start = drag_->previewStart;
107 end = drag_->previewEnd;
108 }
109 const float minimum = timeToX(start);
110 const float maximum = std::max(timeToX(end), minimum + 4.0f);
111 result.items.push_back({track.id, state.id, state.type, true, isSelected(state.id), minimum, maximum,
112 top + 3.0f, top + rowHeight_ - 3.0f});
113 }
114 }
115 return result;
116}

144std::optional<TimelineHit> ActionTimelineWidget::hitTest(float x, float y) const {
145 const auto projected = layout();
146 for (auto it = projected.items.rbegin(); it != projected.items.rend(); ++it) {
147 if (y < it->minimumY || y > it->maximumY || x < it->minimumX - kHandleRadius ||
148 x > it->maximumX + kHandleRadius)
149 continue;
150 if (it->state && std::abs(x - it->minimumX) <= kHandleRadius)
151 return TimelineHit{it->itemId, TimelineHitPart::StartHandle};
152 if (it->state && std::abs(x - it->maximumX) <= kHandleRadius)
153 return TimelineHit{it->itemId, TimelineHitPart::EndHandle};
154 if (x >= it->minimumX && x <= it->maximumX) return TimelineHit{it->itemId, TimelineHitPart::Body};
155 }
156 return std::nullopt;
157}

177EditorResult<void> ActionTimelineWidget::updateDrag(float x) {
178 if (!drag_) return widgetError("editor.action.timeline.widget.drag-missing", "No timeline drag is active");
179 const Duration cursor = xToTime(x);
180 const Duration duration = editor_.target().timeline().duration;
181 const Duration delta = difference(cursor, drag_->anchorTime);
182 if (!drag_->state) {
183 auto moved = drag_->originalStart.tryAdd(delta);
184 if (!moved) return widgetError("editor.action.timeline.widget.drag-overflow", "Notify drag overflowed");
185 drag_->previewStart = std::clamp(moved.value(), Duration::zero(), duration);
186 drag_->previewEnd = drag_->previewStart;
187 return eve::editing::applied<void>();
188 }
189 if (drag_->part == TimelineHitPart::StartHandle) {
190 auto moved = drag_->originalStart.tryAdd(delta);
191 if (!moved) return widgetError("editor.action.timeline.widget.drag-overflow", "State start drag overflowed");
192 drag_->previewStart = std::clamp(moved.value(), Duration::zero(), drag_->originalEnd);
193 drag_->previewEnd = drag_->originalEnd;
194 return eve::editing::applied<void>();
195 }
196 if (drag_->part == TimelineHitPart::EndHandle) {
197 auto moved = drag_->originalEnd.tryAdd(delta);
198 if (!moved) return widgetError("editor.action.timeline.widget.drag-overflow", "State end drag overflowed");
199 drag_->previewStart = drag_->originalStart;
200 drag_->previewEnd = std::clamp(moved.value(), drag_->originalStart, duration);
201 return eve::editing::applied<void>();
202 }
203 const auto span = drag_->originalEnd.nanoseconds() - drag_->originalStart.nanoseconds();
204 auto moved = drag_->originalStart.tryAdd(delta);
205 if (!moved) return widgetError("editor.action.timeline.widget.drag-overflow", "State drag overflowed");
206 const auto latestStart = Duration::fromNanoseconds(duration.nanoseconds() - span);
207 drag_->previewStart = std::clamp(moved.value(), Duration::zero(), latestStart);
208 drag_->previewEnd = Duration::fromNanoseconds(drag_->previewStart.nanoseconds() + span);
209 return eve::editing::applied<void>();
210}

226EditorResult<void> ActionTimelineWidget::inspectSelection(IEditorInspector& inspector) {
227 const auto selected = editor_.selectedItemIds();
228 if (selected.size() != 1) return eve::editing::noOp();
229 auto item = findItem(editor_.target().timeline(), selected.front());
230 if (!item)
231 return widgetError("editor.action.timeline.widget.item-missing", "Selected item no longer exists",
232 EditorStatus::Conflict);
233
234 float startSeconds = static_cast<float>(item->start.seconds());
235 float endSeconds = static_cast<float>(item->end.seconds());
236 std::string type = item->type.format();
237 auto payloadJson = Value(item->payload).toJson();
238 if (!payloadJson)
239 return widgetError("editor.action.timeline.widget.payload-encode", "Could not encode item payload");
240 std::string payload = payloadJson.value();
241
242 inspector.beginGroup("action.timeline.item", item->state ? "Notify State" : "Notify");
243 const float maximum = static_cast<float>(editor_.target().timeline().duration.seconds());
244 bool timingChanged = inspector.scalar("start", item->state ? "Start" : "Time", startSeconds, 0.0f, maximum);
245 if (item->state) timingChanged = inspector.scalar("end", "End", endSeconds, 0.0f, maximum) || timingChanged;
246 const bool typeChanged = inspector.string("type", "Type", type);
247 const bool payloadChanged = inspector.string("payload", "Payload JSON", payload);
248 inspector.endGroup();
249
250 if (!timingChanged && !typeChanged && !payloadChanged) return eve::editing::applied<void>();
251 Result<Duration> parsedStart =
252 timingChanged ? Duration::fromSeconds(startSeconds) : Result<Duration>::success(item->start);
253 Result<Duration> parsedEnd =
254 timingChanged ? Duration::fromSeconds(endSeconds) : Result<Duration>::success(item->end);
255 if (!parsedStart || !parsedEnd)
256 return widgetError("editor.action.timeline.widget.time-invalid", "Inspector time is invalid");
257 auto parsedType = LogicalId::parse(type);
258 if (!parsedType) return widgetError("editor.action.timeline.widget.type-invalid", "Notify type is invalid");
259 auto parsedPayload = Value::fromJson(payload);
260 if (!parsedPayload || !parsedPayload.value().getIf<Value::Object>())
261 return widgetError("editor.action.timeline.widget.payload-invalid", "Payload must be a JSON object");
262 auto descriptor = registry_.descriptor(type);
263 if (!descriptor)
264 return widgetError("editor.action.timeline.widget.type-unregistered", "Notify type is not registered");
265 const auto expected = item->state ? action::ActionNotifyShape::State : action::ActionNotifyShape::Instant;
266 if (descriptor.value().shape != expected)
267 return widgetError("editor.action.timeline.widget.shape-mismatch", "Notify type shape does not match item");
268 const auto* object = parsedPayload.value().getIf<Value::Object>();
269 action::ActionTimelineEvent event{
270 item->state ? action::ActionTimelineEventKind::StateEnter : action::ActionTimelineEventKind::Notify,
271 item->trackId,
272 item->itemId,
273 *parsedType,
274 parsedStart.value(),
275 *object};
276 if (!registry_.validate(event))
277 return widgetError("editor.action.timeline.widget.payload-contract", "Payload violates notify contract");
278 return editor_.editItem(item->itemId, parsedStart.value(), item->state ? parsedEnd.value() : parsedStart.value(),
279 std::move(*parsedType), *object);
280}

292EditorResult<void> ActionTimelineWidget::invoke(TimelineWidgetCommand command) {
293 switch (command) {
294 case TimelineWidgetCommand::Copy: {
295 const auto selected = editor_.selectedItemIds();
296 if (selected.empty())
297 return widgetError("editor.action.timeline.selection-empty", "No timeline items are selected");
298 std::optional<Duration> earliest;
299 for (const auto& id : selected) {
300 const auto item = findItem(editor_.target().timeline(), id);
301 if (item && (!earliest || item->start < *earliest)) earliest = item->start;
302 }
303 auto copied = editor_.copySelection();
304 if (copied.ok()) clipboardAnchor_ = earliest;
305 return adapt(copied, "editor.action.timeline.widget.copy", "Could not copy timeline selection");
306 }
307 case TimelineWidgetCommand::Paste: {
308 if (!clipboardAnchor_)
309 return widgetError("editor.action.timeline.clipboard-empty", "Action timeline clipboard is empty");
310 return adapt(editor_.paste(difference(editor_.previewTime(), *clipboardAnchor_)),
311 "editor.action.timeline.widget.paste", "Could not paste timeline selection");
312 }
313 case TimelineWidgetCommand::DeleteSelection: return editor_.deleteSelection();
314 case TimelineWidgetCommand::Undo: {
315 auto result = editor_.undo();
316 if (result.ok()) return eve::editing::applied<void>();
317 return EditorResult<void>::failure(result.status());
318 }
319 case TimelineWidgetCommand::Redo: {
320 auto result = editor_.redo();
321 if (result.ok()) return eve::editing::applied<void>();
322 return EditorResult<void>::failure(result.status());
323 }
324 case TimelineWidgetCommand::PlayPause:
325 if (editor_.playing())
326 editor_.pause();
327 else
328 editor_.play();
329 return eve::editing::applied<void>();
330 }
331 return widgetError("editor.action.timeline.widget.command", "Timeline command is unsupported",
332 EditorStatus::Unsupported);
333}
