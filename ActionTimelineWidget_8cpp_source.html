30std::optional<ItemView> findItem(const action::ActionTimeline& timeline, const LogicalId& itemId) {
31 for (const auto& section : timeline.animationSections) {
32 if (section.id != itemId) continue;
33 return ItemView{
34 *LogicalId::fromParts("action-track", "animation"),
35 section.id,
36 *LogicalId::fromParts("animation", "section"),
37 section.start,
38 section.end,
39 {{"animationUri", Value(section.animationUri)}, {"blendInNs", Value(section.blendIn.nanoseconds())}},
40 true,
41 false,
42 true};
43 }
44 for (const auto& track : timeline.tracks) {
45 for (const auto& notify : track.notifies)
46 if (notify.id == itemId)
47 return ItemView{track.id, notify.id, notify.type, notify.time,
48 notify.time, notify.payload, false, track.locked};
49 for (const auto& state : track.states)
50 if (state.id == itemId)
51 return ItemView{track.id, state.id, state.type, state.start,
52 state.end, state.payload, true, track.locked};
53 }
54 return std::nullopt;
55}

99EditorResult<void> ActionTimelineWidget::zoom(double factor, double normalizedAnchor) {
100 if (!std::isfinite(factor) || factor <= 0.0 || !std::isfinite(normalizedAnchor) || normalizedAnchor < 0.0 ||
101 normalizedAnchor > 1.0)
102 return widgetError("editor.action.timeline.widget.zoom", "Timeline zoom factor or anchor is invalid");
103 const Duration duration = editor_.target().timeline().duration;
104 const Duration start = visibleEnd_ > visibleStart_ ? visibleStart_ : Duration::zero();
105 const Duration end = visibleEnd_ > visibleStart_ ? visibleEnd_ : duration;
106 const double oldSpan = static_cast<double>(end.nanoseconds() - start.nanoseconds());
107 const double minimumSpan = std::max(1.0, static_cast<double>(duration.nanoseconds()) / 10000.0);
108 const auto newSpan = static_cast<std::int64_t>(
109 std::llround(std::clamp(oldSpan / factor, minimumSpan, static_cast<double>(duration.nanoseconds()))));
110 const auto anchor = start.nanoseconds() + static_cast<std::int64_t>(std::llround(oldSpan * normalizedAnchor));
111 auto newStart = anchor - static_cast<std::int64_t>(std::llround(newSpan * normalizedAnchor));
112 newStart = std::clamp<std::int64_t>(newStart, 0, duration.nanoseconds() - newSpan);
113 visibleStart_ = Duration::fromNanoseconds(newStart);
114 visibleEnd_ = Duration::fromNanoseconds(newStart + newSpan);
115 return eve::editing::applied<void>();

130float ActionTimelineWidget::timeToX(Duration time) const noexcept {

140Duration ActionTimelineWidget::xToTime(float x) const noexcept {
141 const float clamped = std::clamp(x, labelWidth_, width_);
142 const float fraction = (clamped - labelWidth_) / (width_ - labelWidth_);
143 const auto assetDuration = editor_.target().timeline().duration;
144 const auto start = visibleEnd_ > visibleStart_ ? visibleStart_ : Duration::zero();
145 const auto end = visibleEnd_ > visibleStart_ ? visibleEnd_ : assetDuration;
146 const auto span = end.nanoseconds() - start.nanoseconds();
147 std::int64_t time =
148 start.nanoseconds() + static_cast<std::int64_t>(std::llround(static_cast<double>(fraction) * span));
149 if (snapInterval_.nanoseconds() > 0) {
150 const double steps = static_cast<double>(time) / static_cast<double>(snapInterval_.nanoseconds());
151 time = static_cast<std::int64_t>(std::llround(steps)) * snapInterval_.nanoseconds();
152 time = std::clamp<std::int64_t>(time, start.nanoseconds(), end.nanoseconds());
153 }
154 return Duration::fromNanoseconds(time);
155}

157TimelineWidgetLayout ActionTimelineWidget::layout() const {
158 TimelineWidgetLayout result;
159 result.width = width_;
160 const bool hasAnimationLane = !editor_.target().timeline().animationSections.empty();
161 const std::size_t rowOffset = hasAnimationLane ? 1U : 0U;
162 result.height = rowHeight_ * static_cast<float>(editor_.target().timeline().tracks.size() + rowOffset);
163 result.playheadX = timeToX(editor_.previewTime());
164 const auto selected = editor_.selectedItemIds();
165 auto isSelected = [&](const LogicalId& id) {
166 return std::find(selected.begin(), selected.end(), id) != selected.end();
167 };
168 if (snapInterval_.nanoseconds() > 0) {
169 const std::int64_t duration = editor_.target().timeline().duration.nanoseconds();
170 const std::int64_t interval = snapInterval_.nanoseconds();
171 const auto visibleStart = visibleEnd_ > visibleStart_ ? visibleStart_.nanoseconds() : 0;
172 const auto visibleEnd = visibleEnd_ > visibleStart_ ? visibleEnd_.nanoseconds() : duration;
173 std::int64_t time = (visibleStart / interval) * interval;
174 if (time < visibleStart) time += interval;
175 for (std::int64_t index = time / interval; time <= visibleEnd; time += interval, ++index) {
176 const Duration tick = Duration::fromNanoseconds(time);
177 result.rulerTicks.push_back({tick, timeToX(tick), index % 5 == 0});
178 if (duration - time < interval) break;
179 }
180 }
181 if (hasAnimationLane) {
182 const auto animationTrack = *LogicalId::fromParts("action-track", "animation");
183 const auto animationType = *LogicalId::fromParts("animation", "section");
184 for (const auto& section : editor_.target().timeline().animationSections) {
185 Duration start = section.start;
186 Duration end = section.end;
187 if (drag_ && drag_->itemId == section.id) {
188 start = drag_->previewStart;
189 end = drag_->previewEnd;
190 }
191 result.items.push_back({animationTrack, section.id, animationType, true, isSelected(section.id),
192 timeToX(start), std::max(timeToX(end), timeToX(start) + 4.0f), 3.0f,
193 rowHeight_ - 3.0f});
194 }
195 }
196 for (std::size_t row = 0; row < editor_.target().timeline().tracks.size(); ++row) {
197 const auto& track = editor_.target().timeline().tracks[row];
198 const float top = static_cast<float>(row + rowOffset) * rowHeight_;
199 for (const auto& notify : track.notifies) {
200 Duration time = notify.time;
201 if (drag_ && drag_->itemId == notify.id) time = drag_->previewStart;
202 const float center = timeToX(time);
203 result.items.push_back({track.id, notify.id, notify.type, false, isSelected(notify.id),
204 center - kNotifyWidth * 0.5f, center + kNotifyWidth * 0.5f, top + 3.0f,
205 top + rowHeight_ - 3.0f});
206 }
207 for (const auto& state : track.states) {
208 Duration start = state.start;
209 Duration end = state.end;
210 if (drag_ && drag_->itemId == state.id) {
211 start = drag_->previewStart;
212 end = drag_->previewEnd;
213 }
214 const float minimum = timeToX(start);
215 const float maximum = std::max(timeToX(end), minimum + 4.0f);
216 result.items.push_back({track.id, state.id, state.type, true, isSelected(state.id), minimum, maximum,
217 top + 3.0f, top + rowHeight_ - 3.0f});
218 }
219 }
220 return result;
221}

258std::optional<TimelineHit> ActionTimelineWidget::hitTest(float x, float y) const {
259 const auto projected = layout();
260 for (auto it = projected.items.rbegin(); it != projected.items.rend(); ++it) {
261 if (y < it->minimumY || y > it->maximumY || x < it->minimumX - kHandleRadius ||
262 x > it->maximumX + kHandleRadius)
263 continue;
264 if (it->state && std::abs(x - it->minimumX) <= kHandleRadius)
265 return TimelineHit{it->itemId, TimelineHitPart::StartHandle};
266 if (it->state && std::abs(x - it->maximumX) <= kHandleRadius)
267 return TimelineHit{it->itemId, TimelineHitPart::EndHandle};
268 if (x >= it->minimumX && x <= it->maximumX) return TimelineHit{it->itemId, TimelineHitPart::Body};
269 }
270 return std::nullopt;
271}

291EditorResult<void> ActionTimelineWidget::updateDrag(float x) {
292 if (!drag_) return widgetError("editor.action.timeline.widget.drag-missing", "No timeline drag is active");
293 const Duration cursor = xToTime(x);
294 const Duration duration = editor_.target().timeline().duration;
295 const Duration delta = difference(cursor, drag_->anchorTime);
296 if (!drag_->state) {
297 auto moved = drag_->originalStart.tryAdd(delta);
298 if (!moved) return widgetError("editor.action.timeline.widget.drag-overflow", "Notify drag overflowed");
299 drag_->previewStart = std::clamp(moved.value(), Duration::zero(), duration);
300 drag_->previewEnd = drag_->previewStart;
301 return eve::editing::applied<void>();
302 }
303 if (drag_->part == TimelineHitPart::StartHandle) {
304 auto moved = drag_->originalStart.tryAdd(delta);
305 if (!moved) return widgetError("editor.action.timeline.widget.drag-overflow", "State start drag overflowed");
306 drag_->previewStart = std::clamp(moved.value(), Duration::zero(), drag_->originalEnd);
307 drag_->previewEnd = drag_->originalEnd;
308 return eve::editing::applied<void>();
309 }
310 if (drag_->part == TimelineHitPart::EndHandle) {
311 auto moved = drag_->originalEnd.tryAdd(delta);
312 if (!moved) return widgetError("editor.action.timeline.widget.drag-overflow", "State end drag overflowed");
313 drag_->previewStart = drag_->originalStart;
314 drag_->previewEnd = std::clamp(moved.value(), drag_->originalStart, duration);
315 return eve::editing::applied<void>();
316 }
317 const auto span = drag_->originalEnd.nanoseconds() - drag_->originalStart.nanoseconds();
318 auto moved = drag_->originalStart.tryAdd(delta);
319 if (!moved) return widgetError("editor.action.timeline.widget.drag-overflow", "State drag overflowed");
320 const auto latestStart = Duration::fromNanoseconds(duration.nanoseconds() - span);
321 drag_->previewStart = std::clamp(moved.value(), Duration::zero(), latestStart);
322 drag_->previewEnd = Duration::fromNanoseconds(drag_->previewStart.nanoseconds() + span);
323 return eve::editing::applied<void>();
324}

468EditorResult<void> ActionTimelineWidget::invoke(TimelineWidgetCommand command) {
469 switch (command) {
470 case TimelineWidgetCommand::Copy: {
471 const auto selected = editor_.selectedItemIds();
472 if (selected.empty())
473 return widgetError("editor.action.timeline.selection-empty", "No timeline items are selected");
474 std::optional<Duration> earliest;
475 for (const auto& id : selected) {
476 const auto item = findItem(editor_.target().timeline(), id);
477 if (item && (!earliest || item->start < *earliest)) earliest = item->start;
478 }
479 auto copied = editor_.copySelection();
480 if (copied.ok()) clipboardAnchor_ = earliest;
481 return adapt(copied, "editor.action.timeline.widget.copy", "Could not copy timeline selection");
482 }
483 case TimelineWidgetCommand::Paste: {
484 if (!clipboardAnchor_)
485 return widgetError("editor.action.timeline.clipboard-empty", "Action timeline clipboard is empty");
486 return adapt(editor_.paste(difference(editor_.previewTime(), *clipboardAnchor_)),
487 "editor.action.timeline.widget.paste", "Could not paste timeline selection");
488 }
489 case TimelineWidgetCommand::DeleteSelection: return editor_.deleteSelection();
490 case TimelineWidgetCommand::Undo: {
491 auto result = editor_.undo();
492 if (result.ok()) return eve::editing::applied<void>();
493 return EditorResult<void>::failure(result.status());
494 }
495 case TimelineWidgetCommand::Redo: {
496 auto result = editor_.redo();
497 if (result.ok()) return eve::editing::applied<void>();
498 return EditorResult<void>::failure(result.status());
499 }
500 case TimelineWidgetCommand::PlayPause:
501 if (editor_.playing())
502 editor_.pause();
503 else
504 editor_.play();
505 return eve::editing::applied<void>();
506 }
507 return widgetError("editor.action.timeline.widget.command", "Timeline command is unsupported",
508 EditorStatus::Unsupported);
509}
