68Result<Value> parsePlayTrace(const Value& recording) {
69 auto root = asObject(recording, "recording");
70 if (!root) return Result<Value>::failure(root.status());
71 auto known = knownFields(*root.value(),
72 {"contractHash", "contractId", "engine", "result", "schemaId", "schemaVersion", "seed",
73 "startCheckpoint", "steps"},
74 "recording");
75 if (!known) return Result<Value>::failure(known.status());
76 auto schema = stringField(*root.value(), "schemaId", "recording");
77 if (!schema) return Result<Value>::failure(schema.status());
78 const auto version = root.value()->find("schemaVersion");
79 if (version == root.value()->end() || !version->second.isInt64() || version->second.asInt() != kTraceVersion)
80 return failure<Value>(DiagnosticCode::Unsupported, "unsupported play-trace schema", "recording.schemaVersion");
81 if (schema.value() != kTraceSchemaId)
82 return failure<Value>(DiagnosticCode::Unsupported, "unsupported play-trace schema", "recording.schemaId");
83 const auto steps = root.value()->find("steps");
84 if (steps == root.value()->end() || !steps->second.isArray())
85 return failure<Value>(DiagnosticCode::ParseError, "play-trace requires steps", "recording.steps");
86 const auto* array = steps->second.getIf<Value::Array>();
87 for (std::size_t index = 0; index < array->size(); ++index) {
88 const std::string itemPath = "recording.steps." + std::to_string(index);
89 auto item = asObject((*array)[index], itemPath);
90 if (!item) return Result<Value>::failure(item.status());
91 auto itemKnown =
92 knownFields(*item.value(), {"capture", "hostFrame", "observationDigest", "request", "seq"}, itemPath);
93 if (!itemKnown) return Result<Value>::failure(itemKnown.status());
94 const auto request = item.value()->find("request");
95 if (request == item.value()->end())
96 return failure<Value>(DiagnosticCode::ParseError, "trace step requires request", itemPath + ".request");
97 }
98 return Result<Value>::success(recording);
99}

104Result<Value> replayPlayTrace(const Value& recording, IPlayHostRuntime& runtime) {
105 auto parsed = parsePlayTrace(recording);
106 if (!parsed) return Result<Value>::failure(parsed.status());
107 auto root = asObject(parsed.value(), "recording");
108 if (!root) return Result<Value>::failure(root.status());
109 const auto checkpoint = root.value()->find("startCheckpoint");
110 if (checkpoint != root.value()->end() && checkpoint->second.isString() &&
111 !checkpoint->second.asString().empty()) {
112 auto restored = runtime.restoreCheckpoint(checkpoint->second.asString());
113 if (!restored) return Result<Value>::failure(restored.status());
114 }
115 PlayReplayGuard guard;
116 const auto* steps = root.value()->find("steps")->second.getIf<Value::Array>();
117 for (std::size_t index = 0; index < steps->size(); ++index) {
118 const auto* step = (*steps)[index].getIf<Value::Object>();
119 if (!step)
120 return failure<Value>(DiagnosticCode::ParseError, "trace step must be an object",
121 "recording.steps." + std::to_string(index));
122 auto replayed = executePlayRequest(step->at("request"), runtime);
123 if (!replayed) return Result<Value>::failure(replayed.status());
124 const auto digest = step->find("observationDigest");
125 if (digest != step->end() && digest->second.isString() && !digest->second.asString().empty()) {
126 const auto* response = replayed.value().getIf<Value::Object>();
127 if (!response)
128 return failure<Value>(DiagnosticCode::Failed, "replay response must be an object",
129 "recording.steps." + std::to_string(index));
130 const auto state = response->find("state");
131 if (state == response->end())
132 return failure<Value>(DiagnosticCode::HashMismatch, "replay observe is missing state",
133 "recording.steps." + std::to_string(index));
134 auto actual = playObservationDigest(state->second);
135 if (!actual) return Result<Value>::failure(actual.status());
136 if (actual.value() != digest->second.asString())
137 return failure<Value>(DiagnosticCode::HashMismatch, "play-trace observation digest mismatch",
138 "recording.steps." + std::to_string(index));
139 }
140 }
141 return Result<Value>::success(Value(Value::Object{
142 {"op", Value("replay")},
143 {"schemaId", Value("evengine.play-response")},
144 {"schemaVersion", Value(std::int64_t{1})},
145 {"status", Value("passed")},
146 {"steps", Value(static_cast<std::int64_t>(steps->size()))},
147 }));
148}

175void PlayTraceBuffer::append(const Value& request, const Value& response, std::uint64_t hostFrame) {
176 if (replaying_) return;
177 const auto* object = request.getIf<Value::Object>();
178 if (!object) return;
179 const auto op = object->find("op");
180 if (op == object->end() || !op->second.isString()) return;
181 const std::string& name = op->second.asString();
182 if (name == "status" || name == "clock" || name == "trace" || name == "replay") return;
183 if (!begun_) begin({}, {}, 0, {});
184 Value::Object step{{"hostFrame", Value(static_cast<std::int64_t>(hostFrame))},
185 {"request", request},
186 {"seq", Value(++stepCount_)}};
187 const auto* responseObject = response.getIf<Value::Object>();
188 if (responseObject) {
189 const auto state = responseObject->find("state");
190 if (state != responseObject->end()) {
191 auto digest = playObservationDigest(state->second);
192 if (digest) step.emplace("observationDigest", Value(std::move(digest).takeValue()));
193 }
194 const auto path = responseObject->find("path");
195 if (path != responseObject->end() && path->second.isString())
196 step.emplace("capture", Value(path->second.asString()));
197 }
198 stepItems_.emplace_back(Value(std::move(step)));
199}
