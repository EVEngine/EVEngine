330void ReflectionProbeCapture::reportGpuDurationMs(float milliseconds) {
331 if (milliseconds <= 0.f || !std::isfinite(milliseconds)) return;
332 smoothedGpuDurationMs_ =
333 smoothedGpuDurationMs_ > 0.f ? smoothedGpuDurationMs_ * 0.8f + milliseconds * 0.2f : milliseconds;
334
335 if (smoothedGpuDurationMs_ > gpuBudgetMs_ * 1.1f) {
336 underBudgetReports_ = 0;
337 if (adaptiveFaceBudget_ > 1)
338 --adaptiveFaceBudget_;
339 else
340 adaptiveFilterSamples_ = std::max(adaptiveFilterSamples_ / 2, 8);
341 return;
342 }
343
344 if (smoothedGpuDurationMs_ < gpuBudgetMs_ * 0.7f) {
345 if (++underBudgetReports_ < 4) return;
346 underBudgetReports_ = 0;
347 if (adaptiveFilterSamples_ < 64)
348 adaptiveFilterSamples_ = std::min(adaptiveFilterSamples_ * 2, 64);
349 else
350 adaptiveFaceBudget_ = std::min(adaptiveFaceBudget_ + 1, 6);
351 } else {
352 underBudgetReports_ = 0;
353 }
354}

464int ReflectionProbeRegistry::queueCaptureAABB(float minX, float minY, float minZ, float maxX, float maxY, float maxZ,
465 int changedMask) {
466 const float changedMin[3] = {std::min(minX, maxX), std::min(minY, maxY), std::min(minZ, maxZ)};
467 const float changedMax[3] = {std::max(minX, maxX), std::max(minY, maxY), std::max(minZ, maxZ)};
468 const uint32_t mask = static_cast<uint32_t>(changedMask);
469 int queued = 0;
470 for (Entry &entry : entries_) {
471 ReflectionProbeCapture *probe = entry.probe;
472 if (!probe || (static_cast<uint32_t>(probe->getCaptureMask()) & mask) == 0u) continue;
473 const float center[3] = {probe->getCenterX(), probe->getCenterY(), probe->getCenterZ()};
474 float distance2 = 0.f;
475 for (int axis = 0; axis < 3; ++axis) {
476 const float nearest = std::clamp(center[axis], changedMin[axis], changedMax[axis]);
477 const float delta = center[axis] - nearest;
478 distance2 += delta * delta;
479 }
480 const float captureRadius = probe->getCaptureFarDistance();
481 if (distance2 > captureRadius * captureRadius) continue;
482 probe->queueCapture();
483 ++queued;
484 }
485 return queued;
486}

488int ReflectionProbeRegistry::tick(int faceBudget, int filterBudget, int filterSamples) {
489 lastCapturedFaceCount_ = 0;
490 lastPublishedCount_ = 0;
491 faceBudget = std::max(faceBudget, 0);
492 filterBudget = std::max(filterBudget, 0);
493 filterSamples = std::clamp(filterSamples, 8, 512);
494
495 for (Entry &entry : entries_) {
496 if (!entry.probe || (distanceCullingEnabled_ && entry.distanceCulled)) continue;
497 entry.probe->advanceRefreshPolicy();
498 const bool waiting =
499 entry.probe->isCapturePending() || entry.probe->getPublishedRevision() != entry.probe->getRevision();
500 if (waiting && entry.waitAge < std::numeric_limits<uint32_t>::max()) ++entry.waitAge;
501 }
502
503 auto better = [](const Entry *a, const Entry *b) {
504 const int priorityA = a->probe->getInfluencePriority();
505 const int priorityB = b->probe->getInfluencePriority();
506 if (priorityA != priorityB) return priorityA > priorityB;
507 if (a->waitAge != b->waitAge) return a->waitAge > b->waitAge;
508 return a->order < b->order;
509 };
510
511 while (faceBudget-- > 0) {
512 Entry *selected = nullptr;
513 for (Entry &entry : entries_) {
514 if (!entry.probe || !entry.probe->isCapturePending()) continue;
515 if (!selected || better(&entry, selected)) selected = &entry;
516 }
517 if (!selected) break;
518 selected->probe->update(1);
519 selected->waitAge = 0;
520 lastCapturedFaceCount_ += selected->probe->getLastCapturedFaceCount();
521 }
522
523 while (filterBudget-- > 0) {
524 Entry *selected = nullptr;
525 for (Entry &entry : entries_) {
526 if (!entry.probe || entry.probe->isCapturePending() ||
527 entry.probe->getRevision() == entry.probe->getPublishedRevision())
528 continue;
529 if (!selected || better(&entry, selected)) selected = &entry;
530 }
531 if (!selected) break;
532 if (selected->probe->stageCapturedFaces() && selected->probe->filterAndPublish(filterSamples)) {
533 selected->waitAge = 0;
534 ++lastPublishedCount_;
535 } else {
536 break;
537 }
538 }
539 return lastPublishedCount_;
540}

552 struct Candidate {
553 ReflectionProbeCapture *probe = nullptr;
554 int priority = 0;
555 float distance2 = 0.f;
556 bool retained = false;
557 uint64_t order = 0;
558 };
559 std::vector<Candidate> candidates;
560 candidates.reserve(entries_.size());
561 const auto cameraData = camera->data();
562 const float eye[3] = {cameraData->eyeX, cameraData->eyeY, cameraData->eyeZ};
563 const float target[3] = {cameraData->targetX, cameraData->targetY, cameraData->targetZ};
564 const float viewSegment[3] = {target[0] - eye[0], target[1] - eye[1], target[2] - eye[2]};
565 const float viewSegmentLength2 =
566 viewSegment[0] * viewSegment[0] + viewSegment[1] * viewSegment[1] + viewSegment[2] * viewSegment[2];
567 for (Entry &entry : entries_) {
568 ReflectionProbeCapture *probe = entry.probe;
569 if (!probe) continue;
570 const float center[3] = {probe->getCenterX(), probe->getCenterY(), probe->getCenterZ()};
571 const float eyeDx = center[0] - eye[0], eyeDy = center[1] - eye[1], eyeDz = center[2] - eye[2];
572 const bool distanceCulled = distanceCullingEnabled_ && eyeDx * eyeDx + eyeDy * eyeDy + eyeDz * eyeDz >=
573 maxRenderDistance_ * maxRenderDistance_;
574 if (entry.distanceCulled && !distanceCulled) probe->queueCapture();
575 entry.distanceCulled = distanceCulled;
576 if (distanceCulled) {
577 ++lastDistanceCulledCount_;
578 continue;
579 }
580 if (!probe->getActiveCubemap() || probe->getPublishedRevision() == 0 || probe->getInfluenceIntensity() <= 0.f)
581 continue;
582 const float extent[3] = {probe->getInfluenceExtentX(), probe->getInfluenceExtentY(),
583 probe->getInfluenceExtentZ()};
584 float closestT = 0.f;
585 if (viewSegmentLength2 > 1e-8f) {
586 const float toCenter[3] = {center[0] - eye[0], center[1] - eye[1], center[2] - eye[2]};
587 closestT = std::clamp(
588 (toCenter[0] * viewSegment[0] + toCenter[1] * viewSegment[1] + toCenter[2] * viewSegment[2]) /
589 viewSegmentLength2,
590 0.f, 1.f);
591 }
592 const float closestViewPoint[3] = {
593 eye[0] + viewSegment[0] * closestT,
594 eye[1] + viewSegment[1] * closestT,
595 eye[2] + viewSegment[2] * closestT,
596 };
597 float distance2 = 0.f;
598 for (int axis = 0; axis < 3; ++axis) {
599 const float outside = std::max(std::abs(closestViewPoint[axis] - center[axis]) - extent[axis], 0.f);
600 distance2 += outside * outside;
601 }
602 const bool retained = std::find(lastSelected_.begin(), lastSelected_.end(), probe) != lastSelected_.end();
603 const float distance = std::sqrt(distance2);
604 const float biasedDistance = retained ? std::max(distance - selectionHysteresis_, 0.f) : distance;
605 candidates.push_back(
606 Candidate{probe, probe->getInfluencePriority(), biasedDistance * biasedDistance, retained, entry.order});
607 }
608 lastCandidateCount_ = static_cast<int>(candidates.size());
609 std::stable_sort(candidates.begin(), candidates.end(), [](const Candidate &a, const Candidate &b) {
610 if (a.priority != b.priority) return a.priority > b.priority;
611 if (a.distance2 != b.distance2) return a.distance2 < b.distance2;
612 if (a.retained != b.retained) return a.retained;
613 return a.order < b.order;
614 });
615
616 const int selected = std::min(static_cast<int>(candidates.size()), Camera3D::Data::kMaxReflectionProbes);
617 lastSelected_.clear();
618 lastSelected_.reserve(static_cast<size_t>(selected));
619 for (int slot = 0; slot < selected; ++slot) {
620 ReflectionProbeCapture *probe = candidates[static_cast<size_t>(slot)].probe;
621 if (probe->applyConfiguredToCamera(camera, slot)) {
622 lastSelected_.push_back(probe);
623 ++lastSelectedCount_;
624 }
625 }
626 return lastSelectedCount_;
627}
