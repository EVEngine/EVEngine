18class PhysicsQueryImpl final : public eve::IPhysicsQuery {
19public:
20 int newWorld(float gravityX, float gravityY) override {
21 auto *ph = eve::ModuleManager::getInstance<Physics>("Physics");
22 if (!ph) return -1;
23 auto *w = ph->newWorld(gravityX, gravityY);
24 if (!w) return -1;
25 worlds_.push_back(w);
26 return static_cast<int>(worlds_.size()) - 1;
27 }
28
29 int worldCount() override {
30 int n = 0;
31 for (auto *w : worlds_)
32 if (w) ++n;
33 return n;
34 }
35
36 bool worldGravity(int id, float *gx, float *gy) override {
37 auto *w = world(id);
38 if (!w) return false;
39 if (gx) *gx = w->getGravityX();
40 if (gy) *gy = w->getGravityY();
41 return true;
42 }
43
44 bool rayCast(int id, float x1, float y1, float x2, float y2, RayHitInfo *out) override {
45 auto *w = world(id);
46 if (!w || !out) return false;
47 w->rayCast(x1, y1, x2, y2);
48 out->hit = w->hasRayHit();
49 if (out->hit) {
50 out->bodyId = w->getRayHitBodyId();
51 out->x = w->getRayHitX();
52 out->y = w->getRayHitY();
53 out->normalX = w->getRayHitNormalX();
54 out->normalY = w->getRayHitNormalY();
55 out->fraction = w->getRayHitFraction();
56 }
57 return true;
58 }
59
60 bool removeWorld(int id) override {
61 if (id < 0 || static_cast<size_t>(id) >= worlds_.size()) return false;
62 delete worlds_[static_cast<size_t>(id)];
63 worlds_[static_cast<size_t>(id)] = nullptr;
64 return true;
65 }
66
67private:
68 World *world(int id) const {
69 if (id < 0 || static_cast<size_t>(id) >= worlds_.size()) return nullptr;
70 return worlds_[static_cast<size_t>(id)];
71 }

73 std::vector<World *> worlds_;
74};

76class CameraObstructionQueryImpl final : public eve::ICameraObstructionQuery {
77public:
78 void add(World3D* world, std::weak_ptr<const void> lifetime) {
79 std::erase_if(worlds_, [](const RegisteredWorld& entry) { return entry.lifetime.expired(); });
80 const auto found = std::find_if(worlds_.begin(), worlds_.end(),
81 [world](const RegisteredWorld& entry) { return entry.world == world; });
82 if (world && found == worlds_.end()) worlds_.push_back({world, std::move(lifetime)});
83 }
84
85 bool sphereCast(float fromX, float fromY, float fromZ, float toX, float toY, float toZ,
86 float radius, uint64_t maskBits, int ignoredBodyId,
87 eve::CameraObstructionHit* out) override {
88 if (out) *out = eve::CameraObstructionHit{};
89 if (!out) return false;
90 for (const RegisteredWorld& entry : worlds_) {
91 auto lifetime = entry.lifetime.lock();
92 World3D* world = entry.world;
93 if (!lifetime || !world || !world->isValid()) continue;
94 CameraSphereHit3D hit;
95 if (!world->sphereCast(fromX, fromY, fromZ, toX, toY, toZ, radius, maskBits,
96 ignoredBodyId, &hit))
97 continue;
98 if (!out->hit || hit.fraction < out->fraction) {
99 out->hit = true;
100 out->bodyId = hit.bodyId;
101 out->fraction = hit.fraction;
102 out->x = hit.x;
103 out->y = hit.y;
104 out->z = hit.z;
105 out->normalX = hit.nx;
106 out->normalY = hit.ny;
107 out->normalZ = hit.nz;
108 }
109 }
110 return out->hit;
111 }

113private:
114 struct RegisteredWorld {
115 World3D* world = nullptr;
116 std::weak_ptr<const void> lifetime;
117 };

120};
