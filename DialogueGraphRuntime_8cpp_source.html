26 const GraphDocumentData& graph) const {
27 DialogueGraphDomain domain;
28 const DialogueGraphCompileResult compiled = domain.compile(graph);
29 if (compiled.status != EditorStatus::Applied)
30 return EditorResult<dialogue::ConversationDocument*>::failure(
31 eve::Status(compiled.status, compiled.diagnostics));
32 const auto* id = field(compiled.definition, "id")->getIf<std::string>();
33 const auto* version = field(compiled.definition, "version")->getIf<int64_t>();
34 const auto* entry = field(compiled.definition, "entry")->getIf<std::string>();
35 const auto* parameters = field(compiled.definition, "parameters")->getIf<EditorValue::Array>();
36 const auto* nodes = field(compiled.definition, "nodes")->getIf<EditorValue::Array>();
37 auto document = std::make_unique<dialogue::ConversationDocument>(*id);
38 document->removeNode("end");
39 if (!document->setVersion(static_cast<int>(*version)))
40 return runtimeError<dialogue::ConversationDocument*>(EditorStatus::Failed,
41 "editor.dialogue.version-rejected",
42 "Conversation version was rejected");
43 for (const EditorValue& parameter : *parameters)
44 document->addParameter(*parameter.getIf<std::string>());
45 for (const EditorValue& nodeValue : *nodes) {
46 const auto* nodeId = field(nodeValue, "id")->getIf<std::string>();
47 const auto* kind = field(nodeValue, "kind")->getIf<std::string>();
48 if (!document->addNode(*nodeId, *kind))
49 return runtimeError<dialogue::ConversationDocument*>(EditorStatus::Failed,
50 "editor.dialogue.node-rejected",
51 "Conversation node was rejected: " + *nodeId);
52 }
53 for (const EditorValue& nodeValue : *nodes) {
54 const auto* nodeId = field(nodeValue, "id")->getIf<std::string>();
55 const auto* kind = field(nodeValue, "kind")->getIf<std::string>();
56 const auto* properties = field(nodeValue, "properties")->getIf<EditorValue::Object>();
57 for (const auto& [key, value] : *properties) {
58 const auto* text = value.getIf<std::string>();
59 if (!text || !document->setField(*nodeId, key, *text))
60 return runtimeError<dialogue::ConversationDocument*>(EditorStatus::Failed,
61 "editor.dialogue.field-rejected",
62 "Conversation field was rejected: " +
63 *nodeId + "." + key);
64 }
65 if (*kind != "branch" && *kind != "choice" && *kind != "end") {
66 const auto* next = field(nodeValue, "next")->getIf<std::string>();
67 if (!document->setField(*nodeId, "next", *next))
68 return runtimeError<dialogue::ConversationDocument*>(EditorStatus::Failed,
69 "editor.dialogue.next-rejected",
70 "Conversation next target was rejected");
71 }
72 const auto* routes = field(nodeValue, "routes")->getIf<EditorValue::Array>();
73 for (const EditorValue& route : *routes) {
74 const auto* label = field(route, "label")->getIf<std::string>();
75 const auto* target = field(route, "target")->getIf<std::string>();
76 if (!document->addRoute(*nodeId, *label, *target))
77 return runtimeError<dialogue::ConversationDocument*>(EditorStatus::Failed,
78 "editor.dialogue.route-rejected",
79 "Conversation route was rejected");
80 }
81 }
82 if (!document->setEntry(*entry) || !document->validate()) {
83 std::vector<EditorDiagnostic> diagnostics;
84 diagnostics.push_back(eve::editing::ruleDiagnostic(
85 eve::DiagnosticCode::Failed, RuleId("editor.dialogue.runtime-validation"), DiagnosticSeverity::Error,
86 "Conversation runtime validation failed"));
87 for (int index = 0; index < document->getDiagnosticCount(); ++index)
88 diagnostics.push_back(eve::editing::ruleDiagnostic(
89 eve::DiagnosticCode::Failed, RuleId("editor.dialogue.runtime-diagnostic"),
90 document->getDiagnosticSeverity(index) == "error" ? DiagnosticSeverity::Error
91 : DiagnosticSeverity::Warning,
92 document->getDiagnosticMessage(index)));
93 return EditorResult<dialogue::ConversationDocument*>::failure(
94 eve::Status(EditorStatus::Failed, std::move(diagnostics)));
95 }
96 return eve::editing::applied<dialogue::ConversationDocument*>(document.release());
97}
