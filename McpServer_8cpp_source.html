68std::string mcpJsonEscape(const std::string& s) {
69 std::string out;
70 out.reserve(s.size() + 8);
71 for (char c : s) {
72 switch (c) {
73 case '"': out += "\\\""; break;
74 case '\\': out += "\\\\"; break;
75 case '\n': out += "\\n"; break;
76 case '\r': out += "\\r"; break;
77 case '\t': out += "\\t"; break;
78 default:
79 if (static_cast<unsigned char>(c) < 0x20) {
80 char buf[8];
81 std::snprintf(buf, sizeof(buf), "\\u%04x", static_cast<unsigned>(c));
82 out += buf;
83 } else {
84 out += c;
85 }
86 break;
87 }
88 }
89 return out;
90}

241// Encode a Poco JSON value as a Squirrel literal (table/array/scalar/null).
242std::string sqLiteralValue(const Poco::Dynamic::Var& v) {
243 if (v.isEmpty()) return "null";
244 if (v.isBoolean()) return v.convert<bool>() ? "true" : "false";
245 if (v.isInteger()) return std::to_string(v.convert<Poco::Int64>());
246 if (v.isNumeric()) {
247 const double d = v.convert<double>();
248 char buf[64];
249 std::snprintf(buf, sizeof(buf), "%g", d);
250 return buf;
251 }
252 if (v.isString()) return std::string("\"") + sqStringLiteralEscape(v.convert<std::string>()) + "\"";
253 if (v.isArray()) {
254 std::string out = "[";
255 try {
256 auto arr = v.extract<Poco::JSON::Array::Ptr>();
257 for (size_t i = 0; i < arr->size(); ++i) {
258 if (i) out += ",";
259 out += sqLiteralValue(arr->get(static_cast<unsigned int>(i)));
260 }
261 } catch (...) {
262 }
263 out += "]";
264 return out;
265 }
266 if (v.isStruct()) {
267 std::string out = "{";
268 bool first = true;
269 try {
270 auto obj = v.extract<Poco::JSON::Object::Ptr>();
271 for (const auto& kv : *obj) {
272 if (!first) out += ",";
273 first = false;
274 out += "\"" + sqStringLiteralEscape(kv.first) + "\"=" + sqLiteralValue(kv.second);
275 }
276 } catch (...) {
277 }
278 out += "}";
279 return out;
280 }
281 return "null";
282}

291// Compile + run a snippet against the live VM (no return value captured).
292bool runVmSnippet(HSQUIRRELVM vm, const std::string& source, std::string* err) {
293 const SQInteger top = sq_gettop(vm);
294 // MCP snippets are transient input, not project scripts. Compile them
295 // directly from memory so the unified project compiler cannot register a
296 // file-shaped source that hot reload may later treat as user content.
297 if (SQ_FAILED(
298 sq_compilebuffer(vm, source.c_str(), static_cast<SQInteger>(source.size()), kMcpSnippetSource, SQTrue))) {
299 sq_settop(vm, top);
300 if (err) *err = snippetErrorText(vm, true);
301 return false;
302 }
303 sq_pushroottable(vm);
304 if (SQ_FAILED(sq_call(vm, 1, SQFalse, SQTrue))) {
305 sq_settop(vm, top);
306 if (err) *err = snippetErrorText(vm, false);
307 return false;
308 }
309 sq_settop(vm, top);
310 return true;
311}

313// Serialize a Squirrel value (at stack idx) to compact JSON.
314std::string sqValueToJson(HSQUIRRELVM vm, SQInteger idx) {
315 if (idx < 0) idx = sq_gettop(vm) + idx + 1; // normalize relative -> absolute
316 switch (sq_gettype(vm, idx)) {
317 case OT_NULL: return "null";
318 case OT_BOOL: {
319 SQBool b = SQFalse;
320 sq_getbool(vm, idx, &b);
321 return b ? "true" : "false";
322 }
323 case OT_INTEGER: {
324 SQInteger i = 0;
325 sq_getinteger(vm, idx, &i);
326 return std::to_string(i);
327 }
328 case OT_FLOAT: {
329 SQFloat f = 0;
330 sq_getfloat(vm, idx, &f);
331 char buf[64];
332 std::snprintf(buf, sizeof(buf), "%g", static_cast<double>(f));
333 return buf;
334 }
335 case OT_STRING: {
336 const SQChar* s = nullptr;
337 sq_getstring(vm, idx, &s);
338 return std::string("\"") + mcpJsonEscape(s ? s : "") + "\"";
339 }
340 case OT_ARRAY: {
341 std::string out = "[";
342 bool first = true;
343 sq_pushnull(vm);
344 while (SQ_SUCCEEDED(sq_next(vm, idx))) {
345 if (!first) out += ",";
346 first = false;
347 out += sqValueToJson(vm, -1);
348 sq_pop(vm, 2);
349 }
350 sq_pop(vm, 1);
351 out += "]";
352 return out;
353 }
354 case OT_TABLE: {
355 std::string out = "{";
356 bool first = true;
357 sq_pushnull(vm);
358 while (SQ_SUCCEEDED(sq_next(vm, idx))) {
359 if (!first) out += ",";
360 first = false;
361 out += sqValueToJson(vm, -2);
362 out += ":";
363 out += sqValueToJson(vm, -1);
364 sq_pop(vm, 2);
365 }
366 sq_pop(vm, 1);
367 out += "}";
368 return out;
369 }
370 default: return "\"<unserializable>\"";
371 }
372}

375// return value (e.g. `return ::scene_director.info();`).
376std::string callSceneDirectorReturn(HSQUIRRELVM vm, const std::string& snippet, std::string* err) {
377 const SQInteger top = sq_gettop(vm);
378 if (SQ_FAILED(sq_compilebuffer(vm, snippet.c_str(), static_cast<SQInteger>(snippet.size()), kMcpSceneDirectorSource,
379 SQTrue))) {
380 sq_settop(vm, top);
381 if (err) *err = snippetErrorText(vm, true);
382 return {};
383 }
384 sq_pushroottable(vm);
385 if (SQ_FAILED(sq_call(vm, 1, SQTrue, SQTrue))) {
386 sq_settop(vm, top);
387 std::string text = snippetErrorText(vm, false);
388 if (err)
389 *err = text == "runtime failed" ? "runtime failed (is the scene_director kit installed?)" : std::move(text);
390 return {};
391 }
392 std::string json = sqValueToJson(vm, -1);
393 sq_settop(vm, top);
394 return json;
395}

567 if (!isJsonObject(actual)) {
568 match->valid = false;
569 match->error = "Expected object at " + path;
570 return;
571 }
572 auto actualObject = actual.extract<Poco::JSON::Object::Ptr>();
573 auto expectedObject = expected.extract<Poco::JSON::Object::Ptr>();
574 for (const auto& item : *expectedObject) {
575 const std::string childPath = path.empty() ? item.first : path + "." + item.first;
576 if (!actualObject->has(item.first)) {
577 match->valid = false;
578 match->error = "Observed value has no field " + childPath;
579 return;
580 }
581 compareExpectation(actualObject->get(item.first), item.second, childPath, tolerance, match);
582 }
583 return;
584 }
585 if (isJsonArray(expected)) {
586 if (!isJsonArray(actual)) {
587 match->valid = false;
588 match->error = "Expected array at " + path;
589 return;
590 }
591 auto actualArray = actual.extract<Poco::JSON::Array::Ptr>();
592 auto expectedArray = expected.extract<Poco::JSON::Array::Ptr>();
593 if (actualArray->size() < expectedArray->size()) {
594 match->valid = false;
595 match->error = "Observed array is shorter than expectation at " + path;
596 return;
597 }
598 for (std::size_t index = 0; index < expectedArray->size(); ++index) {
599 compareExpectation(actualArray->get(static_cast<unsigned>(index)),
600 expectedArray->get(static_cast<unsigned>(index)),
601 path + "[" + std::to_string(index) + "]", tolerance, match);
602 }
603 return;
604 }
605 if (expected.isNumeric()) {
606 if (!actual.isNumeric()) {
607 match->valid = false;
608 match->error = "Expected numeric value at " + path;
609 return;
610 }
611 const double error = std::abs(actual.convert<double>() - expected.convert<double>());
612 match->maxError = std::max(match->maxError, error);
613 if (error > tolerance) match->mismatches.push_back(path);
614 return;
615 }
616 if (expected.isBoolean()) {
617 if (!actual.isBoolean()) {
618 match->valid = false;
619 match->error = "Expected boolean value at " + path;
620 return;
621 }
622 if (actual.convert<bool>() != expected.convert<bool>()) match->mismatches.push_back(path);
623 return;
624 }
625 if (expected.isString()) {
626 if (!actual.isString()) {
627 match->valid = false;
628 match->error = "Expected string value at " + path;
629 return;
630 }
631 if (actual.convert<std::string>() != expected.convert<std::string>()) match->mismatches.push_back(path);
632 return;
633 }
634 if (expected.isEmpty()) {
635 if (!actual.isEmpty()) match->mismatches.push_back(path);
636 return;
637 }
638 match->valid = false;
639 match->error = "Unsupported expectation value at " + path;
640}

642Poco::JSON::Object::Ptr expectationValue(const ExpectationMatch& match) {
643 Poco::JSON::Object::Ptr output = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
644 output->set("valid", match.valid);
645 output->set("converged", match.valid && match.mismatches.empty());
646 output->set("maxError", match.maxError);
647 Poco::JSON::Array::Ptr mismatches = Poco::JSON::Array::Ptr(new Poco::JSON::Array());
648 for (const auto& path : match.mismatches) mismatches->add(path);
649 output->set("mismatches", mismatches);
650 if (!match.error.empty()) output->set("error", match.error);
651 return output;
652}

654Poco::JSON::Object::Ptr observationEvent(Poco::JSON::Object::Ptr descriptor,
655 Poco::JSON::Object::Ptr observation,
656 std::string* error) {
657 Poco::JSON::Object::Ptr event = Poco::JSON::Object::Ptr(new Poco::JSON::Object());
658 event->set("observation", observation);
659 if (!observation || observation->optValue<std::string>("status", "unavailable") != "ok") {
660 event->set("status", "observation-unavailable");
661 event->set("converged", false);
662 return event;
663 }
664 if (!descriptor || !descriptor->has("expect")) return event;
665
666 double tolerance = 0.001;
667 if (descriptor->has("tolerance")) {
668 try {
669 tolerance = descriptor->get("tolerance").convert<double>();
670 } catch (...) {
671 tolerance = -1.0;
672 }
673 }
674 if (!std::isfinite(tolerance) || tolerance < 0.0) {
675 if (error) *error = "tolerance must be a finite non-negative number";
676 return nullptr;
677 }
678 const Poco::Dynamic::Var expected = descriptor->get("expect");
679 if (!isJsonObject(expected)) {
680 if (error) *error = "expect must be a JSON object";
681 return nullptr;
682 }
683 ExpectationMatch match;
684 compareExpectation(Poco::Dynamic::Var(observation), expected, "", tolerance, &match);
685 if (!match.valid) {
686 if (error) *error = match.error;
687 return nullptr;
688 }
689 event->set("expectation", expectationValue(match));
690 event->set("converged", match.mismatches.empty());
691 event->set("maxError", match.maxError);
692 return event;
693}
