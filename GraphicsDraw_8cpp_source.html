66void Graphics::ensureUboArena(UboArena& arena, uint64_t bytes) {
67 if (arena.buffer && arena.capacity >= bytes) return;
68 uint64_t size = arena.capacity;
69 while (size < bytes) size = size ? size * 2 : (64u << 10);
70 WGPUBufferDescriptor bd{};
71 bd.label = sv("eve_ubo_arena");
72 bd.size = size;
73 bd.usage = WGPUBufferUsage_CopyDst | WGPUBufferUsage_Uniform;
74 bd.mappedAtCreation = false;
75 arena.buffer = device.CreateBuffer(reinterpret_cast<const wgpu::BufferDescriptor*>(&bd));
76 arena.capacity = size;
77 arena.used = 0;
78}

142void Graphics::drawPrimitiveCanvas(const PrimitiveCanvas2D& canvas) {
143 const int targetWidth = activeCanvas ? activeCanvas->getWidth() : getWidth();
144 const int targetHeight = activeCanvas ? activeCanvas->getHeight() : getHeight();
145 const auto triangles = resolvePrimitiveStrokes2D(canvas, {targetWidth, targetHeight});
146 if (triangles.vertices.empty()) return;
147 auto logicalPoint = [targetWidth, targetHeight](const PrimitiveTriangleVertex& vertex) {
148 const glm::vec2 ndc = glm::vec2(vertex.clipPosition) / vertex.clipPosition.w;
149 return glm::vec2((ndc.x + 1.f) * 0.5f * static_cast<float>(targetWidth),
150 (ndc.y + 1.f) * 0.5f * static_cast<float>(targetHeight));
151 };
152 for (const ResolvedPrimitiveBatch2D& resolvedBatch : triangles.batches2D) {
153 auto it = std::find_if(solidBatches.begin(), solidBatches.end(),
154 [&](const SolidBatch& batch) { return batch.blend == resolvedBatch.blend; });
155 if (it == solidBatches.end()) {
156 solidBatches.push_back(SolidBatch{resolvedBatch.blend, Batcher{}});
157 it = solidBatches.end() - 1;
158 }
159 for (std::size_t i = resolvedBatch.firstVertex; i < resolvedBatch.firstVertex + resolvedBatch.vertexCount;
160 i += 3) {
161 it->batch.addTriangle(logicalPoint(triangles.vertices[i]), logicalPoint(triangles.vertices[i + 1]),
162 logicalPoint(triangles.vertices[i + 2]), triangles.vertices[i].color,
163 triangles.vertices[i + 1].color, triangles.vertices[i + 2].color);
164 }
165 const auto batchIndex = static_cast<std::uint32_t>(it - solidBatches.begin());
166 const std::uint32_t count = static_cast<std::uint32_t>(resolvedBatch.vertexCount);
167 const std::uint32_t end = static_cast<std::uint32_t>(it->batch.vertices().size());
168 overlaySpans.push_back({OverlayKind::Solid, batchIndex, end - count, count});
169 }
170}

334void Graphics::flush2D(wgpu::RenderPassEncoder pass, int viewW, int viewH, WGPUTextureFormat format) {
335 ensureFileTexturesReady();
336 applyPendingResourceChanges();
337 auto spans = std::move(overlaySpans);
338 const bool offscreen = uint32_t(format) != uint32_t(surfaceFormat);
339
340 struct SolidUpload {
341 BlendMode blend = BlendMode::Alpha;
342 uint64_t offset = 0;
343 uint64_t bytes = 0;
344 };
345 std::vector<SolidUpload> solidUploads;
346 solidUploads.reserve(solidBatches.size());
347 for (const auto& sb : solidBatches) {
348 if (sb.batch.empty()) {
349 solidUploads.push_back(SolidUpload{sb.blend, 0, 0});
350 continue;
351 }
352 Batcher ndc = sb.batch;
353 ndc.toNDC(viewW, viewH);
354 auto verts = ndc.vertices();
355 // Batcher::toNDC targets Vulkan's Y-down NDC ((-1,-1)=top-left), but
356 // WebGPU's clip space is Y-up ((-1,-1)=bottom-left). Flip Y so logical
357 // top-left maps to the swapchain's top-left.
358 for (auto& v : verts) v.pos.y = -v.pos.y;
359 std::vector<float> data;
360 data.reserve(verts.size() * 6);
361 for (const auto& v : verts) {
362 data.push_back(v.pos.x);
363 data.push_back(v.pos.y);
364 data.push_back(v.color.r);
365 data.push_back(v.color.g);
366 data.push_back(v.color.b);
367 data.push_back(v.color.a);
368 }
369 const uint64_t bytes = data.size() * sizeof(float);
370 auto& arena = currentVertexArena();
371 ensureVertexArena(arena, arena.used + bytes);
372 const uint64_t offset = arena.alloc(bytes);
373 queue.WriteBuffer(arena.buffer, offset, data.data(), bytes);
374 solidUploads.push_back(SolidUpload{sb.blend, offset, bytes});
375 }
376
377 auto solidPipe = [&](BlendMode mode) -> wgpu::RenderPipeline { return get2DColorPipeline(mode, offscreen); };
378
379 auto drawSolid = [&](uint32_t batchIndex, uint32_t first, uint32_t count) {
380 if (batchIndex >= solidUploads.size()) return;
381 const SolidUpload& u = solidUploads[batchIndex];
382 if (u.bytes == 0 || count == 0) return;
383 pass.SetPipeline(solidPipe(u.blend));
384 pass.SetVertexBuffer(0, currentVertexArena().buffer, u.offset, u.bytes);
385 pass.Draw(count, 1, first, 0);
386 };
387
388 if (!spans.empty()) {
389 for (const auto& sp : spans) {
390 if (sp.kind == OverlayKind::Solid)
391 drawSolid(sp.index, sp.vertBegin, sp.vertCount);
392 else if (sp.kind == OverlayKind::Textured && sp.index < texturedBatches.size())
393 drawTexturedBatch(pass, texturedBatches[sp.index], viewW, viewH, format, offscreen);
394 else if (sp.kind == OverlayKind::Lit && sp.index < litBatches.size())
395 drawLitBatch(pass, litBatches[sp.index], viewW, viewH, format);
396 }
397 } else {
398 for (size_t i = 0; i < solidUploads.size(); ++i) {
399 if (solidUploads[i].bytes > 0) drawSolid(uint32_t(i), 0, uint32_t(solidBatches[i].batch.vertices().size()));
400 }
401 for (auto& tb : texturedBatches) {
402 if (tb.batch.empty()) continue;
403 drawTexturedBatch(pass, tb, viewW, viewH, format, offscreen);
404 }
405 for (auto& lb : litBatches) {
406 if (lb.batch.empty()) continue;
407 drawLitBatch(pass, lb, viewW, viewH, format);
408 }
409 }
410 clear2DBatches();
411}

413void Graphics::drawTexturedBatch(wgpu::RenderPassEncoder pass, TexturedBatch& tb, int viewW, int viewH,
414 WGPUTextureFormat format, bool offscreen) {
415 Batcher ndc = tb.batch;
416 ndc.toNDC(viewW, viewH);
417 auto verts = ndc.vertices();
418 for (auto& v : verts) v.pos.y = -v.pos.y;
419 if (verts.empty()) return;
420
421 std::vector<float> data;
422 data.reserve(verts.size() * 8);
423 for (const auto& v : verts) {
424 data.push_back(v.pos.x);
425 data.push_back(v.pos.y);
426 data.push_back(v.color.r);
427 data.push_back(v.color.g);
428 data.push_back(v.color.b);
429 data.push_back(v.color.a);
430 data.push_back(v.uv.x);
431 data.push_back(v.uv.y);
432 }
433 uint64_t bytes = data.size() * sizeof(float);
434 auto& arena = currentVertexArena();
435 ensureVertexArena(arena, arena.used + bytes);
436 uint64_t vtxOffset = arena.alloc(bytes);
437 queue.WriteBuffer(arena.buffer, vtxOffset, data.data(), bytes);
438
439 GpuTexture* gpu = gpuForTexture(tb.texture);
440 GpuTexture* depthGpu = gpuForTexture(tb.depth);
441 GpuTexture* motionGpu = gpuForTexture(tb.motion);
442 GpuTexture* extraGpu = gpuForTexture(tb.extra);
443 GpuTexture* specularGpu = gpuForTexture(tb.specular);
444 auto& uboArena = currentUboArena();
445 ensureUboArena(uboArena, uboArena.used + 256);
446
447 // Push-constant (Externals) block for custom shaders.
448 uint32_t pushOffset = 0;
449 if (tb.shader && tb.shader->pushConstantSize() > 0) {
450 pushOffset = uboArena.alloc(Shader::kPushConstantBytes, 256);
451 queue.WriteBuffer(uboArena.buffer, pushOffset, tb.shader->pushConstantData(), Shader::kPushConstantBytes);
452 }
453
454 wgpu::BindGroup bg = makeTex2DBindGroup(gpu, depthGpu, motionGpu, extraGpu, specularGpu);
455 uint32_t offsets[1] = {pushOffset};
456
457 wgpu::RenderPipeline pipe;
458 if (tb.shader && tb.shader->gpuHandle) {
459 auto* gs = static_cast<GpuShader*>(tb.shader->gpuHandle);
460 const bool hdr = format == WGPUTextureFormat_RGBA16Float;
461 if (tb.blend == BlendMode::Opaque)
462 pipe = hdr ? gs->hdrOffscreenOpaquePipeline
463 : (offscreen ? gs->offscreenOpaquePipeline : gs->swapchainOpaquePipeline);
464 else
465 pipe = hdr ? gs->hdrOffscreenPipeline : (offscreen ? gs->offscreenPipeline : gs->swapchainPipeline);
466 } else {
467 switch (tb.blend) {
468 case BlendMode::Additive:
469 pipe = offscreen ? offscreenTexturedAdditivePipeline : texturedAdditivePipeline;
470 break;
471 case BlendMode::Premultiplied:
472 pipe = offscreen ? offscreenTexturedPremultipliedPipeline : texturedPremultipliedPipeline;
473 break;
474 case BlendMode::Multiply:
475 pipe = offscreen ? offscreenTexturedMultiplyPipeline : texturedMultiplyPipeline;
476 break;
477 case BlendMode::Opaque:
478 pipe = format == WGPUTextureFormat_RGBA16Float
479 ? hdrOffscreenTexturedOpaquePipeline
480 : (offscreen ? offscreenTexturedOpaquePipeline : texturedOpaquePipeline);
481 break;
482 case BlendMode::Alpha:
483 default:
484 pipe = format == WGPUTextureFormat_RGBA16Float
485 ? hdrOffscreenTexturedPipeline
486 : (offscreen ? offscreenTexturedPipeline : texturedPipeline);
487 break;
488 }
489 }
490 if (!pipe) return;
491 pass.SetPipeline(pipe);
492 pass.SetBindGroup(0, bg, 1, offsets);
493 pass.SetVertexBuffer(0, arena.buffer, vtxOffset, bytes);
494 pass.Draw(static_cast<uint32_t>(verts.size()), 1, 0, 0);
495}

497void Graphics::drawLitBatch(wgpu::RenderPassEncoder pass, LitBatch& lb, int viewW, int viewH,
498 WGPUTextureFormat format) {
499 Batcher ndc = lb.batch;
500 ndc.toNDC(viewW, viewH);
501 auto verts = ndc.vertices();
502 for (auto& v : verts) v.pos.y = -v.pos.y;
503 if (verts.empty()) return;
504
505 std::vector<float> data;
506 data.reserve(verts.size() * 8);
507 for (const auto& v : verts) {
508 data.push_back(v.pos.x);
509 data.push_back(v.pos.y);
510 data.push_back(v.color.r);
511 data.push_back(v.color.g);
512 data.push_back(v.color.b);
513 data.push_back(v.color.a);
514 data.push_back(v.uv.x);
515 data.push_back(v.uv.y);
516 }
517 uint64_t bytes = data.size() * sizeof(float);
518 auto& arena = currentVertexArena();
519 ensureVertexArena(arena, arena.used + bytes);
520 uint64_t vtxOffset = arena.alloc(bytes);
521 queue.WriteBuffer(arena.buffer, vtxOffset, data.data(), bytes);
522
523 GpuTexture* albedoGpu = gpuForTextureOrWhite(lb.albedo);
524 // Null / missing normal → flat +Z (128,128,255), not white (which is a
525 // bright wrong normal and washes out lit sprites without a map).
526 GpuTexture* normalGpu = gpuForTexture(lb.normal);
527 if (!normalGpu) normalGpu = flatNormalTexture ? flatNormalTexture : whiteTexture;
528
529 auto& uboArena = currentUboArena();
530 ensureUboArena(uboArena, uboArena.used + 512);
531 uint32_t uboOffset = uboArena.alloc(sizeof(Lighting2DUBO), 256);
532 queue.WriteBuffer(uboArena.buffer, uboOffset, &lighting2dFrame, sizeof(Lighting2DUBO));
533
534 wgpu::BindGroup bg = makeTex2DBindGroup(albedoGpu, normalGpu);
535 uint32_t offsets[1] = {uboOffset};
536 wgpu::RenderPipeline pipe = get2DLitPipeline(lb.blend, format);
537 if (!pipe) return;
538 pass.SetPipeline(pipe);
539 pass.SetBindGroup(0, bg, 1, offsets);
540 pass.SetVertexBuffer(0, arena.buffer, vtxOffset, bytes);
541 pass.Draw(static_cast<uint32_t>(verts.size()), 1, 0, 0);
542}

580void Graphics::end3DFrameToCanvas() {
581 OffscreenCanvas* canvas = active3DCanvas;
582 if (!canvas || !device) return;
583 auto& uboArena = currentUboArena();
584 uboArena.reset();
585 ensureUboArena(uboArena, 4096);
586 auto& vertexArena = currentVertexArena();
587 vertexArena.reset();
588
589 wgpuInstanceProcessEvents(instance.Get());
590 OffscreenTimestampReadback* timestampSlot = nullptr;
591 if (offscreenTimestampSupported && offscreenTimestampQuerySet && offscreenTimestampResolveBuffer) {
592 for (size_t candidate = 0; candidate < offscreenTimestampReadbacks.size(); ++candidate) {
593 const size_t index = (offscreenTimestampReadbackCursor + candidate) % offscreenTimestampReadbacks.size();
594 if (!offscreenTimestampReadbacks[index].pending) {
595 timestampSlot = &offscreenTimestampReadbacks[index];
596 offscreenTimestampReadbackCursor = (index + 1) % offscreenTimestampReadbacks.size();
597 break;
598 }
599 }
600 }
601
602 wgpu::CommandEncoder encoder = device.CreateCommandEncoder();
603 recordDirectionalShadowPasses(encoder);
604 WGPURenderPassColorAttachment color{};
605 color.view = canvas->colorView.Get();
606 color.depthSlice = WGPU_DEPTH_SLICE_UNDEFINED;
607 color.loadOp = WGPULoadOp_Clear;
608 color.storeOp = WGPUStoreOp_Store;
609 const Color canvasClear = canvas->clearRequested ? canvas->clearColor : clearColor;
610 color.clearValue = {canvasClear.r, canvasClear.g, canvasClear.b, canvasClear.a};
611 WGPURenderPassDepthStencilAttachment depth{};
612 depth.view = canvas->depthView.Get();
613 depth.depthClearValue = 1.f;
614 depth.depthLoadOp = WGPULoadOp_Clear;
615 depth.depthStoreOp = WGPUStoreOp_Store;
616 depth.stencilLoadOp = WGPULoadOp_Undefined;
617 depth.stencilStoreOp = WGPUStoreOp_Undefined;
618 WGPURenderPassDescriptor descriptor{};
619 descriptor.colorAttachmentCount = 1;
620 descriptor.colorAttachments = &color;
621 descriptor.depthStencilAttachment = &depth;
622 WGPUPassTimestampWrites timestampWrites{};
623 if (timestampSlot) {
624 timestampWrites.querySet = offscreenTimestampQuerySet.Get();
625 timestampWrites.beginningOfPassWriteIndex = 0;
626 timestampWrites.endOfPassWriteIndex = 1;
627 descriptor.timestampWrites = &timestampWrites;
628 }
629 wgpu::RenderPassEncoder pass =
630 encoder.BeginRenderPass(reinterpret_cast<const wgpu::RenderPassDescriptor*>(&descriptor));
631 flushMesh3D(pass, canvas->isHDR() ? WGPUTextureFormat_RGBA16Float : WGPUTextureFormat_RGBA8Unorm, true);
632 flushPrimitive3D(pass, canvas->isHDR() ? WGPUTextureFormat_RGBA16Float : WGPUTextureFormat_RGBA8Unorm, 1);
633 pass.End();
634 if (timestampSlot) {
635 encoder.ResolveQuerySet(offscreenTimestampQuerySet, 0, 2, offscreenTimestampResolveBuffer, 0);
636 encoder.CopyBufferToBuffer(offscreenTimestampResolveBuffer, 0, timestampSlot->buffer, 0, sizeof(uint64_t) * 2);
637 }
638 wgpu::CommandBuffer command = encoder.Finish();
639 queue.Submit(1, &command);
640 if (timestampSlot) {
641 timestampSlot->pending = true;
642 WGPUBufferMapCallbackInfo callback{};
643 callback.mode = WGPUCallbackMode_AllowProcessEvents;
644 callback.callback = [](WGPUMapAsyncStatus status, WGPUStringView, void* userdata1, void*) {
645 auto* slot = static_cast<OffscreenTimestampReadback*>(userdata1);
646 if (status == WGPUMapAsyncStatus_Success && slot && slot->owner) {
647 const auto* ticks =
648 static_cast<const uint64_t*>(slot->buffer.GetConstMappedRange(0, sizeof(uint64_t) * 2));
649 if (ticks && ticks[1] >= ticks[0]) {
650 const float milliseconds = float(ticks[1] - ticks[0]) * 1.0e-6f;
651 slot->owner->completedOffscreenTimestampMs.fetch_add(milliseconds);
652 }
653 slot->buffer.Unmap();
654 }
655 if (slot) slot->pending = false;
656 };
657 callback.userdata1 = timestampSlot;
658 wgpuBufferMapAsync(timestampSlot->buffer.Get(), WGPUMapMode_Read, 0, sizeof(uint64_t) * 2, callback);
659 }
660 canvas->clearRequested = false;
661 active3DCanvas = nullptr;
662 frame3DStarted = false;
663 frameHad3DThisFrame = false;
664 frameHad3D = false;
665}

789 if (!device) return;

791 auto ensure = [&](wgpu::Buffer& buf, uint64_t& cap, uint64_t need) {
792 if (need == 0) need = 4;
793 if (buf && cap >= need) return;

795 bd.label = sv("eve_clustered_ssbo");
796 bd.size = std::max(need, cap ? cap * 2 : need);
797 bd.usage = WGPUBufferUsage_CopyDst | WGPUBufferUsage_Storage;
798 bd.mappedAtCreation = false;

800 cap = bd.size;
801 };

803 const uint64_t tableBytes = std::max<uint64_t>(1, upload.clusterTable.size()) * sizeof(ClusterTableEntry);
804 const uint64_t indicesBytes = std::max<uint64_t>(1, upload.lightIndices.size()) * sizeof(uint32_t);
805 ensure(st.lights, st.lightsCap, lightsBytes);
806 ensure(st.table, st.tableCap, tableBytes);
807 ensure(st.indices, st.indicesCap, indicesBytes);
808
809 ClusteredLightGpu zero{};
810 if (!upload.lights.empty())
811 queue.WriteBuffer(st.lights, 0, upload.lights.data(), lightsBytes);
812 else
813 queue.WriteBuffer(st.lights, 0, &zero, sizeof(zero));
814 if (!upload.clusterTable.empty()) queue.WriteBuffer(st.table, 0, upload.clusterTable.data(), tableBytes);
815 if (!upload.lightIndices.empty()) queue.WriteBuffer(st.indices, 0, upload.lightIndices.data(), indicesBytes);
