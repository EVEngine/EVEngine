14 const std::string& buildingId, const std::vector<BuildingEdgeCurvePoint>& controlPoints, int subdivisions) const {

16 runtimeControls.reserve(controlPoints.size());

18 runtimeControls.push_back({static_cast<float>(point.x), static_cast<float>(point.y)});
19 auto vertices = building::PlacementSystem::sampleEdgeCubicBezier(runtimeControls, subdivisions);
20 if (!vertices.ok())
21 return eve::editing::failed<DomainOperation>(EditorStatus::Rejected,
22 RuleId("editor.building.edge-curve-rejected"),
23 "Cubic edge curve sampling was rejected");

157editing::GizmoSnapshot BuildingPlacementTarget::edgeCubicBezierGizmo(
158 const std::string& buildingId, const std::vector<BuildingEdgeCurvePoint>& controlPoints, int subdivisions,
159 int level, int replacingMemberInstanceId, const std::string& surfaceName) const {
160 editing::GizmoSnapshot gizmo;
161 gizmo.target = id_;
162 gizmo.targetRevision = revisionValue();
163 const BuildingEdgeCurvePreview preview =
164 previewEdgeCubicBezier(buildingId, controlPoints, subdivisions, replacingMemberInstanceId, surfaceName);
165 gizmo.status = preview.status;
166 gizmo.diagnostics = preview.diagnostics;
167 if (!world_) return gizmo;
168 const auto project = [&](double gx, double gy) {
169 float ox = 0.f, oy = 0.f, xx = 0.f, xy = 0.f, yx = 0.f, yy = 0.f;
170 world_->cellToWorldPlane(0, 0, ox, oy);
171 world_->cellToWorldPlane(1, 0, xx, xy);
172 world_->cellToWorldPlane(0, 1, yx, yy);
173 const double px = ox + gx * (xx - ox) + gy * (yx - ox);
174 const double py = oy + gx * (xy - oy) + gy * (yy - oy);
175 const double elevation = static_cast<double>(level) * world_->getFloorHeight();
176 if (!surfaceName.empty()) {
177 auto hit = building::PlacementSystem::sampleSurface(*world_, surfaceName, static_cast<float>(px),
178 static_cast<float>(py));
179 if (hit.ok()) return std::array<double, 3>{hit.value().worldX, hit.value().worldY, hit.value().worldZ};
180 }
181 return world_->getGridPlaneName() == "xz" ? std::array<double, 3>{px, elevation, py}
182 : std::array<double, 3>{px, py, elevation};
183 };
184 const auto addLine = [&](const std::string& id, const std::array<double, 3>& a, const std::array<double, 3>& b,
185 const std::array<double, 4>& color, bool dashed) {
186 editing::GizmoPrimitive line;
187 line.id = id;
188 line.kind = "line";
189 for (size_t axis = 0; axis < 3; ++axis) {
190 line.position[axis] = (a[axis] + b[axis]) * 0.5;
191 line.direction[axis] = b[axis] - a[axis];
192 }
193 line.length = std::sqrt(line.direction[0] * line.direction[0] + line.direction[1] * line.direction[1] +
194 line.direction[2] * line.direction[2]);
195 if (line.length > 0.0)
196 for (double& component : line.direction) component /= line.length;
197 line.color = color;
198 line.dashed = dashed;
199 gizmo.primitives.push_back(std::move(line));
200 };
201 const std::array<double, 4> accepted{0.2, 0.95, 0.35, 1.0};
202 const std::array<double, 4> rejected{0.95, 0.2, 0.2, 1.0};
203 const auto& pathColor = preview.status == EditorStatus::Applied ? accepted : rejected;
204 for (size_t index = 0; index < controlPoints.size(); ++index) {
205 editing::GizmoPrimitive handle;
206 handle.id = "curve.control." + std::to_string(index);
207 handle.kind = "sphere";
208 handle.position = project(controlPoints[index].x, controlPoints[index].y);
209 handle.radius = world_->getCellSize() * 0.15;
210 handle.color = index == 0 || index + 1 == controlPoints.size() ? std::array<double, 4>{0.2, 0.65, 1.0, 1.0}
211 : std::array<double, 4>{1.0, 0.7, 0.15, 1.0};
212 gizmo.primitives.push_back(std::move(handle));
213 if (index > 0)
214 addLine("curve.tangent." + std::to_string(index - 1),
215 project(controlPoints[index - 1].x, controlPoints[index - 1].y),
216 project(controlPoints[index].x, controlPoints[index].y), {0.65, 0.65, 0.7, 0.8}, true);
217 }
218 if (!preview.surfaceSamples.empty()) {
219 for (size_t index = 1; index < preview.surfaceSamples.size(); ++index) {
220 const auto& a = preview.surfaceSamples[index - 1];
221 const auto& b = preview.surfaceSamples[index];
222 addLine("curve.path." + std::to_string(index - 1), {a.worldX, a.worldY, a.worldZ},
223 {b.worldX, b.worldY, b.worldZ}, pathColor, false);
224 }
225 } else {
226 for (size_t index = 1; index < preview.sampledVertices.size(); ++index)
227 addLine("curve.path." + std::to_string(index - 1),
228 project(preview.sampledVertices[index - 1].x, preview.sampledVertices[index - 1].y),
229 project(preview.sampledVertices[index].x, preview.sampledVertices[index].y), pathColor, false);
230 }
231 return gizmo;
232}

287Result<BuildingEdgeCurvePoint> BuildingPlacementTarget::curveLogicalPointFromWorld(double worldX, double worldY,
288 double worldZ) const {
289 if (!world_ || !std::isfinite(worldX) || !std::isfinite(worldY) || !std::isfinite(worldZ))
290 return eve::editing::failed<BuildingEdgeCurvePoint>(
291 EditorStatus::Rejected, RuleId("editor.building.curve-world-point-invalid"),
292 "Curve drag requires a finite world-space point and an available world");
293 float ox = 0.f, oy = 0.f, xx = 0.f, xy = 0.f, yx = 0.f, yy = 0.f;
294 world_->cellToWorldPlane(0, 0, ox, oy);
295 world_->cellToWorldPlane(1, 0, xx, xy);
296 world_->cellToWorldPlane(0, 1, yx, yy);
297 const double bxX = xx - ox;
298 const double bxY = xy - oy;
299 const double byX = yx - ox;
300 const double byY = yy - oy;
301 const double determinant = bxX * byY - bxY * byX;
302 if (std::abs(determinant) < 1e-9)
303 return eve::editing::failed<BuildingEdgeCurvePoint>(EditorStatus::Unsupported,
304 RuleId("editor.building.curve-grid-basis-singular"),
305 "Curve drag requires an invertible affine grid basis");
306 const double planeX = worldX;
307 const double planeY = world_->getGridPlaneName() == "xz" ? worldZ : worldY;
308 const double dx = planeX - ox;
309 const double dy = planeY - oy;
310 BuildingEdgeCurvePoint point;
311 point.x = (dx * byY - dy * byX) / determinant;
312 point.y = (bxX * dy - bxY * dx) / determinant;
313 return eve::editing::applied<BuildingEdgeCurvePoint>(point);
314}
