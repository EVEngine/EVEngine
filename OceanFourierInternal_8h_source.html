15inline void inverse1D(std::vector<Complex>& a) {
16 const auto n = a.size();
17 for (std::size_t i = 1, j = 0; i < n; ++i) {
18 auto bit = n >> 1;
19 for (; j & bit; bit >>= 1) j ^= bit;
20 j ^= bit;
21 if (i < j) std::swap(a[i], a[j]);
22 }
23 for (std::size_t length = 2; length <= n; length <<= 1) {
24 const auto step = std::polar(1.0, tau / static_cast<double>(length));
25 for (std::size_t start = 0; start < n; start += length) {
26 Complex w{1.0, 0.0};
27 for (std::size_t j = 0; j < length / 2; ++j) {
28 const auto even = a[start + j], odd = a[start + j + length / 2] * w;
29 a[start + j] = even + odd;
30 a[start + j + length / 2] = even - odd;
31 w *= step;
32 }
33 }
