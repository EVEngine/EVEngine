59class ScriptCircleHitProbe final : public ICarrierHitProbe {

62 ecs::EntityHandle id{};

67 [[nodiscard]] Result<std::vector<Contact>> query(CarrierHandle, const CarrierMotion& previous,
68 const CarrierMotion& current) const override {
69 std::vector<Contact> contacts;
70 for (const auto& target : targets_) {
71 const double ax = previous.position.x - target.center.x;
72 const double ay = previous.position.y - target.center.y;
73 const double az = previous.position.z - target.center.z;
74 const double bx = current.position.x - target.center.x;
75 const double by = current.position.y - target.center.y;
76 const double bz = current.position.z - target.center.z;
77 const double abx = bx - ax;
78 const double aby = by - ay;
79 const double abz = bz - az;
80 const double ab2 = abx * abx + aby * aby + abz * abz;
81 const double radius2 = target.radius * target.radius;
82 const double startDistance = ax * ax + ay * ay + az * az;
83 const double projection = ax * abx + ay * aby + az * abz;
84 if (ab2 <= 1e-12) continue;
85 double t = 0.0;
86 if (startDistance >= radius2 - 1e-9) {
87 // At a surface while departing: do not bounce again on the next frame.
88 if (projection >= 0.0) continue;
89 const double discriminant = projection * projection - ab2 * (startDistance - radius2);
90 if (discriminant < 0.0) continue;
91 t = (-projection - std::sqrt(discriminant)) / ab2;
92 if (t < -1e-9 || t > 1.0) continue;
93 t = std::max(0.0, t);
94 }
95 const double cx = ax + abx * t;
96 const double cy = ay + aby * t;
97 const double cz = az + abz * t;
98 const double dist2 = cx * cx + cy * cy + cz * cz;
99 if (dist2 <= radius2 + 1e-9 * std::max(1.0, radius2)) {
100 Contact contact;
101 contact.fraction = t;
102 contact.target = target.id;
103 contact.point = {target.center.x + cx, target.center.y + cy, target.center.z + cz};
104 const double len = std::sqrt(dist2);
105 if (len > 1e-9) {
106 contact.normal = {cx / len, cy / len, cz / len};
107 } else {
108 contact.normal = {-1.0, 0.0, 0.0};
109 }
110 contacts.push_back(contact);
111 }
112 }
113 return Result<std::vector<Contact>>::success(std::move(contacts));
114 }

116 void add(Target target) { targets_.push_back(target); }
117 void clear() { targets_.clear(); }
118 [[nodiscard]] bool empty() const noexcept { return targets_.empty(); }

120private:
121 std::vector<Target> targets_;

223class ScriptMotionFunction final : public ICarrierMotionFunction {
224public:
225 explicit ScriptMotionFunction(ssq::Function callback) : callback_(std::move(callback)) {}
226 Result<CarrierMotion> evaluate(const CarrierState& before, const CarrierMotion& proposed,
227 Duration step) const override {
228 const auto vm = callback_.getHandle();
229 const auto top = sq_gettop(vm);
230 struct StackGuard {
231 HSQUIRRELVM vm;
232 SQInteger top;
233 ~StackGuard() { sq_settop(vm, top); }
234 } guard{vm, top};
235 auto table = [vm](const CarrierMotion& motion) {
236 ssq::Table result(vm);
237 result.set("x", static_cast<float>(motion.position.x));
238 result.set("y", static_cast<float>(motion.position.y));
239 result.set("z", static_cast<float>(motion.position.z));
240 result.set("vx", static_cast<float>(motion.velocity.x));
241 result.set("vy", static_cast<float>(motion.velocity.y));
242 result.set("vz", static_cast<float>(motion.velocity.z));
243 return result;
244 };
245 auto state = table(before.motion);
246 state.set("age", static_cast<float>(before.age.seconds()));
247 state.set("lifetime", static_cast<float>(before.lifetime.seconds()));
248 state.set("distanceTravelled", static_cast<float>(before.distanceTravelled));
249 state.set("recipeId", before.recipeId.format());
250 state.set("slot", static_cast<std::int64_t>(before.handle.slot));
251 state.set("generation", static_cast<std::int64_t>(before.handle.generation));
252 if (SQ_FAILED(script::callFunctionResult(vm, callback_.getRaw(), script::rootTable, state, table(proposed),
253 static_cast<float>(step.seconds()))))
254 return Result<CarrierMotion>::failure(Diagnostic::error(
255 DiagnosticCode::Failed, "Custom motion callback threw; see script diagnostic", "motionFunction"));
256 if (sq_gettype(vm, -1) == OT_NULL) return Result<CarrierMotion>::success(proposed);
257 auto value = script::valueFromSquirrel(vm, -1);
258 if (!value) return Result<CarrierMotion>::failure(value.status());
259 const auto* object = value.value().getIf<Value::Object>();
260 if (!object)
261 return Result<CarrierMotion>::failure(
262 Diagnostic::error(DiagnosticCode::InvalidArgument,
263 "Motion function must return null or a position/velocity table", "motionFunction"));
264 CarrierMotion output = proposed;
265 int positions = 0, velocities = 0;
266 for (const auto& [name, field] : *object) {
267 double number = 0.0;
268 if (const auto* real = field.getIf<double>())
269 number = *real;
270 else if (const auto* integer = field.getIf<std::int64_t>())
271 number = static_cast<double>(*integer);
272 else
273 return Result<CarrierMotion>::failure(
274 Diagnostic::error(DiagnosticCode::InvalidArgument, "Motion coordinate must be a number", name));
275 if (name == "x") {
276 output.position.x = number;
277 ++positions;
278 } else if (name == "y") {
279 output.position.y = number;
280 ++positions;
281 } else if (name == "z") {
282 output.position.z = number;
283 ++positions;
284 } else if (name == "vx") {
285 output.velocity.x = number;
286 ++velocities;
287 } else if (name == "vy") {
288 output.velocity.y = number;
289 ++velocities;
290 } else if (name == "vz") {
291 output.velocity.z = number;
292 ++velocities;
293 } else
294 return Result<CarrierMotion>::failure(
295 Diagnostic::error(DiagnosticCode::InvalidArgument, "Unknown motion output field", name));
296 }
297 if ((positions != 0 && positions != 3) || (velocities != 0 && velocities != 3) ||
298 (positions == 0 && velocities == 0))
299 return Result<CarrierMotion>::failure(Diagnostic::error(
300 DiagnosticCode::InvalidArgument, "Return complete x/y/z and/or vx/vy/vz groups", "motionFunction"));
301 // Returning the projected proposal is an identity operation despite the VM's float precision.
302 if (positions == 3 && velocities == 3 && output.position.x == static_cast<float>(proposed.position.x) &&
303 output.position.y == static_cast<float>(proposed.position.y) &&
304 output.position.z == static_cast<float>(proposed.position.z) &&
305 output.velocity.x == static_cast<float>(proposed.velocity.x) &&
306 output.velocity.y == static_cast<float>(proposed.velocity.y) &&
307 output.velocity.z == static_cast<float>(proposed.velocity.z))
308 return Result<CarrierMotion>::success(proposed);
309 const double dt = step.seconds();
310 if (positions == 0)
311 output.position = {before.motion.position.x + output.velocity.x * dt,
312 before.motion.position.y + output.velocity.y * dt,
313 before.motion.position.z + output.velocity.z * dt};
314 if (velocities == 0)
315 output.velocity = {(output.position.x - before.motion.position.x) / dt,
316 (output.position.y - before.motion.position.y) / dt,
317 (output.position.z - before.motion.position.z) / dt};
318 return Result<CarrierMotion>::success(output);
319 }
320
321private:
322 ssq::Function callback_;
323};
