48void DistanceConstraint::configure() {
49 const bool soft = springEnabled && (!limitsEnabled || minimum < maximum);
50 solveSpring = !soft || hertz > 0;
51 solveLimits = soft && limitsEnabled;
52 solveMotor = soft && motorEnabled;
53 const float rest = !soft && limitsEnabled && minimum == maximum ? minimum : length;
54 spring.SetDistance(rest, rest);
55 spring.SetLimitsSpringSettings({JPH::ESpringMode::FrequencyAndDamping, soft ? hertz : 0.f, damping});
56 limits.SetDistance(minimum, maximum);
57 ResetWarmStart();
58}

91void DistanceConstraint::SetupVelocityConstraint(float dt) {
92 if (solveSpring) spring.SetupVelocityConstraint(dt);
93 if (solveLimits) limits.SetupVelocityConstraint(dt);
94 if (solveMotor) {
95 const auto p1 = mBody1->GetCenterOfMassTransform() * spring.GetConstraintToBody1Matrix().GetTranslation();
96 const auto p2 = mBody2->GetCenterOfMassTransform() * spring.GetConstraintToBody2Matrix().GetTranslation();
97 axis = JPH::Vec3(p2 - p1).NormalizedOr(JPH::Vec3::sAxisX());
98 motor.CalculateConstraintProperties(*mBody1, JPH::Vec3(p1 - mBody1->GetCenterOfMassPosition()), *mBody2,
99 JPH::Vec3(p2 - mBody2->GetCenterOfMassPosition()), axis, -speed);
100 } else
101 motor.Deactivate();
102}
103void DistanceConstraint::WarmStartVelocityConstraint(float ratio) {
104 if (solveSpring) spring.WarmStartVelocityConstraint(ratio);
105 if (solveLimits) limits.WarmStartVelocityConstraint(ratio);
106 if (solveMotor) motor.WarmStart(*mBody1, *mBody2, axis, ratio);
107}
108bool DistanceConstraint::SolveVelocityConstraint(float dt) {
109 bool changed = false;
110 if (solveMotor) changed = motor.SolveVelocityConstraint(*mBody1, *mBody2, axis, -dt * maxForce, dt * maxForce);
111 if (solveSpring) changed = spring.SolveVelocityConstraint(dt) || changed;
112 if (solveLimits) changed = limits.SolveVelocityConstraint(dt) || changed;
113 return changed;
114}
115bool DistanceConstraint::SolvePositionConstraint(float dt, float baumgarte) {
116 bool changed = false;
117 if (solveSpring) changed = spring.SolvePositionConstraint(dt, baumgarte);
118 if (solveLimits) changed = limits.SolvePositionConstraint(dt, baumgarte) || changed;
119 return changed;
120}
