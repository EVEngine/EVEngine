43void DistanceConstraint::configure() {
44 const bool soft = springEnabled && (!limitsEnabled || minimum < maximum);
45 solveSpring = !soft || hertz > 0;
46 solveLimits = soft && limitsEnabled;
47 solveMotor = soft && motorEnabled;
48 const float rest = !soft && limitsEnabled && minimum == maximum ? minimum : length;
49 spring.SetDistance(rest, rest);
50 spring.SetLimitsSpringSettings({JPH::ESpringMode::FrequencyAndDamping, soft ? hertz : 0.f, damping});
51 limits.SetDistance(minimum, maximum);
52 ResetWarmStart();
53}

86void DistanceConstraint::SetupVelocityConstraint(float dt) {
87 if (solveSpring) spring.SetupVelocityConstraint(dt);
88 if (solveLimits) limits.SetupVelocityConstraint(dt);
89 if (solveMotor) {
90 const auto p1 = mBody1->GetCenterOfMassTransform() * spring.GetConstraintToBody1Matrix().GetTranslation();
91 const auto p2 = mBody2->GetCenterOfMassTransform() * spring.GetConstraintToBody2Matrix().GetTranslation();
92 axis = JPH::Vec3(p2 - p1).NormalizedOr(JPH::Vec3::sAxisX());
93 motor.CalculateConstraintProperties(*mBody1, JPH::Vec3(p1 - mBody1->GetCenterOfMassPosition()), *mBody2,
94 JPH::Vec3(p2 - mBody2->GetCenterOfMassPosition()), axis, -speed);
95 } else
96 motor.Deactivate();
97}
98void DistanceConstraint::WarmStartVelocityConstraint(float ratio) {
99 if (solveSpring) spring.WarmStartVelocityConstraint(ratio);
100 if (solveLimits) limits.WarmStartVelocityConstraint(ratio);
101 if (solveMotor) motor.WarmStart(*mBody1, *mBody2, axis, ratio);
102}
103bool DistanceConstraint::SolveVelocityConstraint(float dt) {
104 bool changed = false;
105 if (solveMotor) changed = motor.SolveVelocityConstraint(*mBody1, *mBody2, axis, -dt * maxForce, dt * maxForce);
106 if (solveSpring) changed = spring.SolveVelocityConstraint(dt) || changed;
107 if (solveLimits) changed = limits.SolveVelocityConstraint(dt) || changed;
108 return changed;
109}
110bool DistanceConstraint::SolvePositionConstraint(float dt, float baumgarte) {
111 bool changed = false;
112 if (solveSpring) changed = spring.SolvePositionConstraint(dt, baumgarte);
113 if (solveLimits) changed = limits.SolvePositionConstraint(dt, baumgarte) || changed;
114 return changed;
115}
