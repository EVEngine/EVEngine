60float combineMaterialValue(float a, uint64_t materialA, float b, uint64_t materialB,
61 int shift, bool friction) {
62 const uint64_t modeA = (materialA >> shift) & combineModeMask;
63 const uint64_t modeB = (materialB >> shift) & combineModeMask;
64 const uint64_t mode = std::max(modeA, modeB);
65 switch (mode) {
66 case 1: return 0.5f * (a + b);
67 case 2: return std::min(a, b);
68 case 3: return a * b;
69 case 4: return std::max(a, b);
70 default: return friction ? std::sqrt(a * b) : std::max(a, b);
71 }
72}

256int World3D::explode(float x, float y, float z, float radius, float falloff,
257 float impulsePerArea, int maskBits) {
258 explosionResults_.clear();
259 if (!std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z))
260 throw eve::Exception("World3D.explode: center must be finite");
261 if (!std::isfinite(radius) || radius < 0.f || !std::isfinite(falloff) || falloff < 0.f)
262 throw eve::Exception("World3D.explode: radius and falloff must be finite and >= 0");
263 if (!std::isfinite(impulsePerArea))
264 throw eve::Exception("World3D.explode: impulsePerArea must be finite");
265 if (!isValid() || impulsePerArea == 0.f) return 0;
266
267 struct Before {
268 Body3D *body = nullptr;
269 b3Vec3 linear{};
270 b3Vec3 angular{};
271 };
272 std::vector<Before> before;
273 before.reserve(bodies_.size());
274 for (Body3D *body : bodies_) {
275 if (!body || !body->isValid() || b3Body_GetType(body->raw()) != b3_dynamicBody)
276 continue;
277 before.push_back({body, b3Body_GetLinearVelocity(body->raw()),
278 b3Body_GetAngularVelocity(body->raw())});
279 }
280
281 b3ExplosionDef def = b3DefaultExplosionDef();
282 def.position = b3Pos{x, y, z};
283 def.radius = radius;
284 def.falloff = falloff;
285 def.impulsePerArea = impulsePerArea;
286 def.maskBits = static_cast<uint32_t>(maskBits);
287 b3World_Explode(worldId_, &def);
288
289 for (const Before &entry : before) {
290 if (!entry.body->isValid()) continue;
291 const b3Vec3 linear = b3Body_GetLinearVelocity(entry.body->raw());
292 const b3Vec3 angular = b3Body_GetAngularVelocity(entry.body->raw());
293 const b3Vec3 dv = linear - entry.linear;
294 const b3Vec3 dw = angular - entry.angular;
295 if (b3LengthSquared(dv) <= 1e-12f && b3LengthSquared(dw) <= 1e-12f) continue;
296 explosionResults_.push_back({entry.body->getId(), dv.x, dv.y, dv.z,
297 dw.x, dw.y, dw.z});
298 }
299 std::sort(explosionResults_.begin(), explosionResults_.end(),
300 [](const ExplosionResult &a, const ExplosionResult &b) {
301 return a.bodyId < b.bodyId;
302 });
303 return static_cast<int>(explosionResults_.size());
304}

423bool World3D::sphereCast(float x1, float y1, float z1, float x2, float y2, float z2,
424 float radius, uint64_t maskBits, int ignoredBodyId,
425 CameraSphereHit3D* out) const {
426 if (out) *out = CameraSphereHit3D{};
427 if (!out || !isValid() || radius < 0.f) return false;
428
429 struct Collector {
430 CameraSphereHit3D* out = nullptr;
431 int ignoredBodyId = -1;
432 static float callback(b3ShapeId shapeId, b3Pos point, b3Vec3 normal, float fraction,
433 uint64_t, int, int, void* context) {
434 auto* self = static_cast<Collector*>(context);
435 Body3D* body = bodyFromShape(shapeId);
436 if (!body || body->getId() == self->ignoredBodyId) return -1.f;
437 if (!self->out->hit || fraction < self->out->fraction) {
438 self->out->hit = true;
439 self->out->bodyId = body->getId();
440 self->out->fraction = fraction;
441 self->out->x = static_cast<float>(point.x);
442 self->out->y = static_cast<float>(point.y);
443 self->out->z = static_cast<float>(point.z);
444 self->out->nx = normal.x;
445 self->out->ny = normal.y;
446 self->out->nz = normal.z;
447 }
448 return fraction;
449 }
450 } collector{out, ignoredBodyId};
451
452 const b3Vec3 point{0.f, 0.f, 0.f};
453 const b3ShapeProxy proxy{&point, 1, radius};
454 b3QueryFilter filter = b3DefaultQueryFilter();
455 filter.maskBits = maskBits;
456 b3World_CastShape(worldId_, b3Pos{x1, y1, z1}, &proxy,
457 b3Vec3{x2 - x1, y2 - y1, z2 - z1}, filter,
458 &Collector::callback, &collector);
459 return out->hit;
460}

958bool World3D::preSolveCallback(b3ShapeId shapeIdA, b3ShapeId shapeIdB, b3Pos point,
959 b3Vec3 normal, void *context) {
960 auto *world = static_cast<World3D *>(context);
961 if (!world) return true;
962 auto findShape = [world](b3ShapeId id) -> Shape3D * {
963 const auto raw = world->shapeRawHandles_.find(b3StoreShapeId(id));
964 if (raw == world->shapeRawHandles_.end()) return nullptr;
965 const auto found = world->shapeHandles_.find(raw->second);
966 return found == world->shapeHandles_.end() ? nullptr : found->second;
967 };
968 Shape3D *shapeA = findShape(shapeIdA);
969 Shape3D *shapeB = findShape(shapeIdB);
970 if (shapeA && shapeA->isOneWay() &&
971 !shapeA->allowsOneWayContact(point.x, point.y, point.z, normal.x, normal.y, normal.z))
972 return false;
973 if (shapeB && shapeB->isOneWay() &&
974 !shapeB->allowsOneWayContact(point.x, point.y, point.z, -normal.x, -normal.y,
975 -normal.z))
976 return false;
977 return true;
978}

1007int World3D::queryBodyContacts(int bodyId, int maxPoints) {
1008 contactPoints_.clear();
1009 if (bodyId <= 0)
1010 throw eve::Exception("World3D.queryBodyContacts: bodyId must be a positive stable id");
1011 constexpr int maxContactPoints = 4096;
1012 if (maxPoints < 1 || maxPoints > maxContactPoints)
1013 throw eve::Exception("World3D.queryBodyContacts: maxPoints must be in [1, 4096]");
1014 if (!isValid()) return 0;
1015
1016 Body3D *target = nullptr;
1017 for (Body3D *body : bodies_) {
1018 if (body && body->isValid() && body->getId() == bodyId) {
1019 target = body;
1020 break;
1021 }
1022 }
1023 if (!target) return 0;
1024

1080 if (a.x != b.x) return a.x < b.x;
1081 if (a.y != b.y) return a.y < b.y;

1084 if (static_cast<int>(contactPoints_.size()) > maxPoints)

1754 closestNormalX_ = closestNormalY_ = closestNormalZ_ = 0.f;

1757 !(maxDistance >= 0.f) || !std::isfinite(maxDistance))
1758 throw eve::Exception(
1759 "World3D.closestPoint: target must be finite; maxDistance finite and >= 0");
1760 constexpr double floatLimit = static_cast<double>(FLT_MAX);
1761 const double distance = static_cast<double>(maxDistance);
1762 if (distance * distance > floatLimit ||
1763 std::fabs(static_cast<double>(x)) + distance > floatLimit ||
1764 std::fabs(static_cast<double>(y)) + distance > floatLimit ||

1770 World3D *world = nullptr;

1777
1778 static bool callback(b3ShapeId shapeId, void *context) {

1781 Body3D *body = bodyFromShape(shapeId);
1782 Shape3D *shape = shapeFromShape(shapeId);

1787 const int bodyId = body->getId();
1788 const int stableShapeId = shape->getId();

1790 if (distanceSquared < self->bestDistanceSquared - tieTolerance ||
1791 (std::fabs(distanceSquared - self->bestDistanceSquared) <= tieTolerance &&
1792 (self->bodyId < 0 || bodyId < self->bodyId ||
1793 (bodyId == self->bodyId && stableShapeId < self->stableShapeId)))) {
1794 self->bestDistanceSquared = distanceSquared;
1795 self->bodyId = bodyId;
1796 self->stableShapeId = stableShapeId;

1799 }
1800 return true;
1801 }
1802 } collector;
1803 collector.world = this;

1862 struct PlaneCollector {
1863 World3D *world = nullptr;
1864 b3CollisionPlane planes[planeCapacity]{};
1865 int count = 0;
1866 b3Vec3 desired = b3Vec3_zero;
1867 b3Vec3 up = b3Vec3_axisY;
1868 b3Vec3 bestNormal = b3Vec3_zero;
1869 float bestDot = 0.f;
1870 float maxUpDot = -1.f;
1871
1872 static bool callback(b3ShapeId shapeId, const b3PlaneResult *results, int resultCount,
1873 void *context) {
1874 auto *self = static_cast<PlaneCollector *>(context);
1875 if (self->world->shouldIgnoreQueryShape(shapeId)) return true;
1876 for (int i = 0; i < resultCount && self->count < planeCapacity; ++i) {
1877 const b3Plane &plane = results[i].plane;
1878 self->planes[self->count++] = b3CollisionPlane{
1879 .plane = plane, .pushLimit = FLT_MAX, .push = 0.f, .clipVelocity = true};
1880 const float d = b3Dot(plane.normal, self->desired);
1881 if (d < self->bestDot) {
1882 self->bestDot = d;
1883 self->bestNormal = plane.normal;
1884 }
1885 self->maxUpDot = std::max(self->maxUpDot, b3Dot(plane.normal, self->up));
1886 }
1887 return true;
1888 }
1889 } collector;
1890 collector.world = this;
1891 collector.up = moverUp_;
1892
1893 const b3Pos start{ax, ay, az};
1894 b3Pos current = start;
1895 const b3Pos target{ax + dx, ay + dy, az + dz};
1896 b3Capsule mover{b3Vec3_zero, b3Vec3{bx - ax, by - ay, bz - az}, radius};
1897 b3QueryFilter filter = makeQueryFilter();
1898 constexpr float toleranceSquared = 1e-8f;
1899 bool collided = false;
1900
1901 for (int pass = 0; pass < 5; ++pass) {
1902 collector.count = 0;
1903 collector.desired = target - current;
1904 b3World_CollideMover(worldId_, current, &mover, filter, &PlaneCollector::callback,
1905 &collector);
1906 moverPlaneCount_ += collector.count;
1907 collided = collided || collector.count > 0;
1908
1909 const b3Vec3 targetDelta = target - current;
1910 b3PlaneSolverResult solved = b3SolvePlanes(targetDelta, collector.planes, collector.count);
1911 b3Vec3 delta = solved.delta;

1913 World3D *world = nullptr;
1914 static bool callback(b3ShapeId shapeId, void *context) {
1915 auto *self = static_cast<MoverFilter *>(context);

1920 &MoverFilter::callback, &moverFilter);
1921 if (fraction < 1.f) collided = true;
1922 delta = fraction * delta;
1923 current = current + delta;
1924 moverIterations_ = pass + 1;
1925
1926 if (b3LengthSquared(delta) <= toleranceSquared) break;
1927 if (b3LengthSquared(target - current) <= toleranceSquared) break;
1928 }
1929
1930 const b3Vec3 total = current - start;
1931 moverDeltaX_ = total.x;
1932 moverDeltaY_ = total.y;
1933 moverDeltaZ_ = total.z;
1934 moverNormalX_ = collector.bestNormal.x;
1935 moverNormalY_ = collector.bestNormal.y;
1936 moverNormalZ_ = collector.bestNormal.z;
1937 moverGroundDot_ = collector.maxUpDot;
1938 moverGrounded_ = moverGroundDot_ >= moverSlopeCos_;
1939 return collided;
1940}
