32uint64_t stablePairKey(int idA, int idB) {
33 const uint32_t low = static_cast<uint32_t>(std::min(idA, idB));
34 const uint32_t high = static_cast<uint32_t>(std::max(idA, idB));
35 return (static_cast<uint64_t>(high) << 32) | low;
36}

38bool pairContains(uint64_t key, int id) {
39 const uint32_t value = static_cast<uint32_t>(id);
40 return static_cast<uint32_t>(key) == value || static_cast<uint32_t>(key >> 32) == value;
41}

81World3D::World3D(float gravityX, float gravityY, float gravityZ, bool sleep, eve::PersistentId instanceId,
82 bool registerQueries, RigidBackend3D backend)
83 : rigidBackendKind_(backend), instanceId_(instanceId) {
84 runtimeHandle_ = detail::allocatePhysicsWorldHandle();
85 if (instanceId_.isNil()) instanceId_ = detail::makePhysicsWorldPersistentId(runtimeHandle_);
86#if EVENGINE_HAS_JOLT
87 if (backend == RigidBackend3D::Jolt)
88 rigidBackend_ = worldValue(jolt::makeWorld(*this, gravityX, gravityY, gravityZ, sleep));
89 else
90#endif
91 rigidBackend_ = worldValue(box3d::makeWorld(*this, gravityX, gravityY, gravityZ, sleep));
92 auto selection = rigidBackend_->selectSimulation();
93 if (!selection) {
94 const eve::Status status = selection.status();
95 throw eve::Exception("World3D: cannot select a simulation backend: %s", status.describe().c_str());
96 }
97 backendSelectionStatus_ = selection.status();
98 auto selected = std::move(selection).takeValue();
99 backendFallback_ = selected.usedFallback;
100 simulation_ = std::move(selected.backend);
101 if (registerQueries) registerCameraObstructionWorld(this);
102 auto targetingRegistration = registerQueries ? registerTargetingLineOfSightWorld(this)
103 : eve::Result<void>::success();
104 if (!registerQueries) {
105 targetingRegistration.ignore("detached topology candidate does not publish targeting queries");
106 } else if (!targetingRegistration) {
107 // Physics supports multiple worlds; targeting LOS deliberately exposes
108 // only one, so an explicitly consumed Conflict leaves this world valid
109 // without making its query context ambiguous.
110 targetingRegistration.ignore("another World3D is already the active targeting LOS world");
111 }
112 auto& registry = worldRegistry();
113 std::scoped_lock registryLock(registry.mutex);
114 registry.worlds[runtimeHandle_.index()] = this;
115}

241void World3D::destroy() {
242 if (destroyed_) return;
243 cancelForceStep();
244 {
245 auto& registry = worldRegistry();
246 std::scoped_lock registryLock(registry.mutex);
247 auto found = registry.worlds.find(runtimeHandle_.index());
248 if (found != registry.worlds.end() && found->second == this) registry.worlds.erase(found);
249 }
250 invalidateCouplings3D();
251 restoredCouplings3D_.clear();
252 destroyed_ = true;
253 // Invalidate borrowed query registrations without touching any global
254 // registry. This keeps teardown safe for every static destruction order.
255 queryLifetime_.reset();
256
257 std::vector<Mechanism3D *> mechanisms(mechanisms_.begin(), mechanisms_.end());
258 for (Mechanism3D *mechanism : mechanisms) {
259 if (mechanism) mechanism->invalidate();
260 }
261 mechanisms_.clear();
262
263 std::vector<Joint3D *> joints(joints_.begin(), joints_.end());
264 for (Joint3D *joint : joints) {
265 if (joint) joint->invalidate();
266 }
267 joints_.clear();
268 restoredJoints_.clear();
269 disabledBodyPairs_.clear();
270 disabledShapePairs_.clear();
271
272 std::vector<Body3D *> bodies(bodies_.begin(), bodies_.end());
273 for (Body3D *b : bodies) {
274 if (b) b->invalidate();
275 }
276 bodies_.clear();
277
278 std::vector<Shape3D *> shapes(shapes_.begin(), shapes_.end());
279 for (Shape3D *s : shapes) {
280 if (s) s->invalidate();
281 }
282 shapes_.clear();
283 restoredShapes_.clear();
284 restoredBodies_.clear();
285
286 simulation_.reset();
287 if (rigidBackend_) rigidBackend_->destroy();
288 clearContactEvents();
289 shapeRecords_.clear();
290 shapeHandles_.clear();
291 shapeRawHandles_.clear();
292 jointHandles_.clear();
293 runtimeHandle_ = PhysicsWorldHandle::invalid();
294}

341void World3D::setBodyPairCollisionEnabled(Body3D *bodyA, Body3D *bodyB, bool enabled) {
342 if (!bodyA || !bodyB || bodyA == bodyB || !bodyA->isValid() || !bodyB->isValid() ||
343 bodyA->getWorld() != this || bodyB->getWorld() != this)
344 throw eve::Exception(
345 "World3D.setBodyPairCollisionEnabled: distinct live bodies must belong to this world");
346 const uint64_t key = stablePairKey(bodyA->getId(), bodyB->getId());
347 for (Shape3D *shape : shapes_) {
348 if (shape && (shape->getBody() == bodyA || shape->getBody() == bodyB))
349 shape->enableContactOverrideFiltering();
350 }
351 if (enabled) disabledBodyPairs_.erase(key); else disabledBodyPairs_.insert(key);
352 for (Shape3D *shape : shapes_) {
353 if (shape && (shape->getBody() == bodyA || shape->getBody() == bodyB))
354 refreshContactFilter(shape);
355 }
356}

545void World3D::indexShapeHandle(Shape3D *shape) {
546 const auto handle = shape->runtimeHandle();
547 const uint64_t rawKey = rigidBackend_->shapeKey(*shape);

562void World3D::emitContactEvents() {
563 if (!isValid()) return;
564 const auto beginStart = beginContacts_.size(), endStart = endContacts_.size();
565 const auto triggerBeginStart = beginTriggers_.size(), triggerEndStart = endTriggers_.size();
566 const auto hitStart = hits_.size(), stressStart = jointStressEvents_.size();
567 worldValue(rigidBackend_->collectEvents());
568
569 auto *ev = eve::ModuleManager::getInstance<eve::platform_event::PlatformEvent>("PlatformEvent");
570 if (!ev) return;
571 const auto publishPair = [&](const char *name, const EventShape &a, const EventShape &b) {
572 std::vector<eve::platform_event::Variant> args = {
573 eve::platform_event::Variant::makeInt(a.bodyId), eve::platform_event::Variant::makeInt(b.bodyId),
574 eve::platform_event::Variant::makeInt(a.shapeId), eve::platform_event::Variant::makeInt(b.shapeId),
575 eve::platform_event::Variant::makeInt(a.shapeTag), eve::platform_event::Variant::makeInt(b.shapeTag)};
576 ev->push(new eve::platform_event::Message(name, args));
577 };
578 for (auto i = beginStart; i < beginContacts_.size(); ++i)
579 publishPair("begincontact3d", beginContacts_[i].shapeA, beginContacts_[i].shapeB);
580 for (auto i = endStart; i < endContacts_.size(); ++i)
581 publishPair("endcontact3d", endContacts_[i].shapeA, endContacts_[i].shapeB);
582 for (auto i = hitStart; i < hits_.size(); ++i) publishPair("hit3d", hits_[i].shapeA, hits_[i].shapeB);
583 for (auto i = triggerBeginStart; i < beginTriggers_.size(); ++i)
584 publishPair("begintrigger3d", beginTriggers_[i].sensor, beginTriggers_[i].visitor);
585 for (auto i = triggerEndStart; i < endTriggers_.size(); ++i)
586 publishPair("endtrigger3d", endTriggers_[i].sensor, endTriggers_[i].visitor);
587 for (auto i = stressStart; i < jointStressEvents_.size(); ++i) {
588 const auto &joint = jointStressEvents_[i];
589 std::vector<eve::platform_event::Variant> args = {
590 eve::platform_event::Variant::makeInt(joint.jointId), eve::platform_event::Variant::makeInt(joint.bodyAId),
591 eve::platform_event::Variant::makeInt(joint.bodyBId), eve::platform_event::Variant::makeInt(joint.kind)};
592 ev->push(new eve::platform_event::Message("jointstress3d", args));
593 }
594}
