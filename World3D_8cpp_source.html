121void World3D::emitContactEvents() {
122 if (!isValid()) return;
123
124 auto *ev = eve::ModuleManager::getInstance<eve::event::Event>("Event");
125 if (!ev) return;
126
127 b3ContactEvents contacts = b3World_GetContactEvents(worldId_);
128 for (int i = 0; i < contacts.beginCount; ++i) {
129 Body3D *a = bodyFromShape(contacts.beginEvents[i].shapeIdA);
130 Body3D *b = bodyFromShape(contacts.beginEvents[i].shapeIdB);
131 if (!a || !b) continue;
132 std::vector<eve::event::Variant> args = {eve::event::Variant::makeInt(a->getId()),
133 eve::event::Variant::makeInt(b->getId())};
134 ev->push(new eve::event::Message("begincontact3d", args));
135 }
136 for (int i = 0; i < contacts.endCount; ++i) {
137 Body3D *a = bodyFromShape(contacts.endEvents[i].shapeIdA);
138 Body3D *b = bodyFromShape(contacts.endEvents[i].shapeIdB);
139 if (!a || !b) continue;
140 std::vector<eve::event::Variant> args = {eve::event::Variant::makeInt(a->getId()),
141 eve::event::Variant::makeInt(b->getId())};
142 ev->push(new eve::event::Message("endcontact3d", args));
143 }

177int World3D::queryAABB(float minX, float minY, float minZ, float maxX, float maxY, float maxZ) {
178 queryBodyIds_.clear();
179 if (!isValid()) return 0;
180
181 struct Collector {
182 std::vector<int> *ids = nullptr;
183 std::unordered_set<int> seen;
184
185 static bool callback(b3ShapeId shapeId, void *context) {
186 auto *self = static_cast<Collector *>(context);
187 Body3D *b = bodyFromShape(shapeId);
188 if (!b) return true;
189 int id = b->getId();
190 if (self->seen.insert(id).second) self->ids->push_back(id);
191 return true;
192 }
193 } cb;
194 cb.ids = &queryBodyIds_;
195
196 b3AABB aabb;
197 aabb.lowerBound = b3Vec3{std::min(minX, maxX), std::min(minY, maxY), std::min(minZ, maxZ)};
198 aabb.upperBound = b3Vec3{std::max(minX, maxX), std::max(minY, maxY), std::max(minZ, maxZ)};
199 b3QueryFilter filter = b3DefaultQueryFilter();
200 b3World_OverlapAABB(worldId_, aabb, filter, &Collector::callback, &cb);
201 return static_cast<int>(queryBodyIds_.size());
202}
