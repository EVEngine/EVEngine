99Result<MarketConfig> parseMarketObject(const Value::Object& object) {
100 MarketConfig config;
101 if (auto ok = readString(object, "id", config.id, true); !ok) return fail<MarketConfig>(ok.status());
102
103 std::string scopeText;
104 if (auto ok = readString(object, "scope", scopeText, true); !ok) return fail<MarketConfig>(ok.status());
105 auto scope = parseMarketScope(scopeText);
106 if (!scope) return fail<MarketConfig>(scope.status());
107 config.scope = scope.value();
108
109 if (auto ok = readString(object, "regionId", config.regionId, false); !ok) return fail<MarketConfig>(ok.status());
110 if (auto ok = readString(object, "nationId", config.nationId, false); !ok) return fail<MarketConfig>(ok.status());
111 if (auto ok = readString(object, "currencyType", config.currencyType, false); !ok)
112 return fail<MarketConfig>(ok.status());
113 if (auto ok = readInt(object, "tariffBps", config.tariffBps, false); !ok) return fail<MarketConfig>(ok.status());
114 if (auto ok = readBool(object, "enabled", config.enabled); !ok) return fail<MarketConfig>(ok.status());
115
116 if (const Value* listingsValue = objectField(object, "listings")) {
117 const auto* listings = listingsValue->getIf<Value::Array>();
118 if (!listings) {
119 return Result<MarketConfig>::failure(Diagnostic::error(
120 DiagnosticCode::InvalidArgument, "listings must be an array", "economy.market.listings"));
121 }
122 config.listings.reserve(listings->size());
123 for (const Value& entry : *listings) {
124 const auto* listingObject = entry.getIf<Value::Object>();
125 if (!listingObject) {
126 return Result<MarketConfig>::failure(Diagnostic::error(
127 DiagnosticCode::InvalidArgument, "listing entries must be objects", "economy.market.listings"));
128 }
129 auto listing = parseListing(*listingObject);
130 if (!listing) return fail<MarketConfig>(listing.status());
131 config.listings.push_back(std::move(listing).takeValue());
132 }
133 }
134
135 if (const Value* schema = objectField(object, "schemaId")) {
136 const auto* text = schema->getIf<std::string>();
137 if (!text || *text != "eve.economy.market") {
138 return Result<MarketConfig>::failure(Diagnostic::error(
139 DiagnosticCode::InvalidArgument, "schemaId must be eve.economy.market", "economy.market.schemaId"));
140 }
141 }
142 if (const Value* version = objectField(object, "schemaVersion")) {
143 int schemaVersion = 0;
144 if (const auto* integer = version->getIf<std::int64_t>()) {
145 schemaVersion = static_cast<int>(*integer);
146 } else if (const auto* number = version->getIf<double>()) {
147 schemaVersion = static_cast<int>(*number);
148 } else {
149 return Result<MarketConfig>::failure(Diagnostic::error(
150 DiagnosticCode::InvalidArgument, "schemaVersion must be an integer", "economy.market.schemaVersion"));
151 }
152 if (schemaVersion != 1) {
153 return Result<MarketConfig>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
154 "unsupported economy.market schemaVersion",
155 "economy.market.schemaVersion"));
156 }
157 }
158
159 return Result<MarketConfig>::success(std::move(config));
160}

187Result<void> normalizeMarketConfig(MarketConfig& config) {
188 if (config.id.empty()) {
189 return Result<void>::failure(
190 Diagnostic::error(DiagnosticCode::InvalidArgument, "market id is required", "economy.market.id"));
191 }
192 if (config.currencyType.empty()) config.currencyType = "gold";
193 if (config.tariffBps < 0) {
194 return Result<void>::failure(
195 Diagnostic::error(DiagnosticCode::InvalidArgument, "tariffBps must be >= 0", "economy.market.tariffBps"));
196 }
197
198 if (config.scope == MarketScope::Regional) {
199 if (config.regionId.empty()) {
200 return Result<void>::failure(Diagnostic::error(
201 DiagnosticCode::InvalidArgument, "regional market requires regionId", "economy.market.regionId"));
202 }
203 } else if (config.scope == MarketScope::National) {
204 if (config.nationId.empty()) {
205 return Result<void>::failure(Diagnostic::error(
206 DiagnosticCode::InvalidArgument, "national market requires nationId", "economy.market.nationId"));
207 }
208 if (!config.regionId.empty()) {
209 return Result<void>::failure(Diagnostic::error(
210 DiagnosticCode::InvalidArgument, "national market must not set regionId", "economy.market.regionId"));
211 }
212 } else {
213 return Result<void>::failure(
214 Diagnostic::error(DiagnosticCode::InvalidArgument, "unknown market scope", "economy.market.scope"));
215 }
216
217 std::unordered_set<std::string> seenTypes;
218 for (const MarketListing& listing : config.listings) {
219 if (listing.resourceType.empty()) {
220 return Result<void>::failure(Diagnostic::error(
221 DiagnosticCode::InvalidArgument, "listing.resourceType is required", "economy.market.listings"));
222 }
223 if (listing.basePrice <= 0) {
224 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
225 "listing.basePrice must be > 0", "economy.market.listings"));
226 }
227 if (listing.buyPremiumBps < 0) {
228 return Result<void>::failure(Diagnostic::error(
229 DiagnosticCode::InvalidArgument, "listing.buyPremiumBps must be >= 0", "economy.market.listings"));
230 }
231 if (listing.sellDiscountBps < 0 || listing.sellDiscountBps > 10000) {
232 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
233 "listing.sellDiscountBps must be in [0,10000]",
234 "economy.market.listings"));
235 }
236 if (listing.stockCap < 0) {
237 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
238 "listing.stockCap must be >= 0", "economy.market.listings"));
239 }
240 if (!seenTypes.insert(listing.resourceType).second) {
241 return Result<void>::failure(Diagnostic::error(
242 DiagnosticCode::Conflict, "duplicate listing.resourceType in one market", "economy.market.listings"));
243 }
244 }
245 return Result<void>::success();
246}
