44bool readNumber(const eve::Value::Object& object, std::string_view name, double& result) {
45 const auto* value = field(object, name);
46 if (const auto* number = value ? value->getIf<double>() : nullptr) {
47 result = *number;
48 return std::isfinite(result);
49 }
50 if (const auto* integer = value ? value->getIf<std::int64_t>() : nullptr) {
51 result = static_cast<double>(*integer);
52 return std::isfinite(result);
53 }
54 return false;
55}

57eve::Result<std::vector<std::string>> readStrings(const eve::Value::Object& object, std::string_view name) {
58 const auto* value = field(object, name);
59 if (value == nullptr) return eve::Result<std::vector<std::string>>::success({});
60 const auto* array = value->getIf<eve::Value::Array>();
61 if (array == nullptr) return invalid<std::vector<std::string>>("skill field must be an array", std::string(name));
62 std::vector<std::string> result;
63 result.reserve(array->size());
64 for (std::size_t index = 0; index < array->size(); ++index) {
65 const auto* text = (*array)[index].getIf<std::string>();
66 if (text == nullptr || text->empty())
67 return invalid<std::vector<std::string>>("skill array item must be a non-empty string",
68 std::string(name) + "[" + std::to_string(index) + "]");
69 result.push_back(*text);
70 }
71 return eve::Result<std::vector<std::string>>::success(std::move(result));
72}

74eve::Result<SkillDefinition> parseDefinition(const eve::definitions::Definition& source) {
75 auto owned = eve::Value::fromJson(source.json);
76 if (!owned) return eve::Result<SkillDefinition>::failure(owned.status());
77 const auto* payload = owned.value().getIf<eve::Value::Object>();
78 if (payload == nullptr) return invalid<SkillDefinition>("skill definition payload must be an object", "json");
79
80 SkillDefinition result;
81 result.id = source.id;
82 const auto* target = field(*payload, "targetType");
83 if (target != nullptr) {
84 const auto* text = target->getIf<std::string>();
85 if (text == nullptr || text->empty())
86 return invalid<SkillDefinition>("skill targetType must be a non-empty string", "targetType");
87 result.targetType = *text;
88 }
89 double number = 0.0;
90 if (field(*payload, "cooldown") != nullptr) {
91 if (!readNumber(*payload, "cooldown", number) || number < 0.0 || number > std::numeric_limits<float>::max())
92 return invalid<SkillDefinition>("skill cooldown must be finite and non-negative", "cooldown");
93 result.cooldown = static_cast<float>(number);
94 }
95 if (field(*payload, "castTime") != nullptr) {
96 if (!readNumber(*payload, "castTime", number) || number < 0.0 || number > std::numeric_limits<float>::max())
97 return invalid<SkillDefinition>("skill castTime must be finite and non-negative", "castTime");
98 result.castTime = static_cast<float>(number);
99 }
100 auto effects = readStrings(*payload, "grantedEffects");
101 if (!effects) return eve::Result<SkillDefinition>::failure(effects.status());
102 result.grantedEffects = std::move(effects).takeValue();
103 auto tags = readStrings(*payload, "tags");
104 if (!tags) return eve::Result<SkillDefinition>::failure(tags.status());
105 result.tags = std::move(tags).takeValue();
106 if (const auto* condition = field(*payload, "castCondition")) {
107 auto decoded = decodeSkillCondition(*condition);
108 if (!decoded) return eve::Result<SkillDefinition>::failure(decoded.status());
109 result.castCondition = std::move(decoded).takeValue();
110 }
111 auto extras = field(*payload, "extra");
112 if (extras != nullptr) {
113 const auto* extraObject = extras->getIf<eve::Value::Object>();
114 if (extraObject == nullptr) return invalid<SkillDefinition>("skill extra must be an object", "extra");
115 for (const auto& [key, value] : *extraObject) {
116 const auto* text = value.getIf<std::string>();
117 if (text == nullptr) return invalid<SkillDefinition>("skill extra values must be strings", "extra." + key);
118 result.extra.emplace(key, *text);
119 }
120 }
121 const auto* costs = field(*payload, "costs");
122 if (costs != nullptr) {
123 const auto* array = costs->getIf<eve::Value::Array>();
124 if (array == nullptr) return invalid<SkillDefinition>("skill costs must be an array", "costs");
125 std::vector<eve::resource::ResourceCost> items;
126 for (std::size_t index = 0; index < array->size(); ++index) {
127 const auto* item = (*array)[index].getIf<eve::Value::Object>();
128 if (item == nullptr)
129 return invalid<SkillDefinition>("skill cost must be an object", "costs[" + std::to_string(index) + "]");
130 const auto* resource = field(*item, "attribute");
131 const auto* amount = field(*item, "amount");
132 const auto* name = resource ? resource->getIf<std::string>() : nullptr;
133 const auto* integer = amount ? amount->getIf<std::int64_t>() : nullptr;
134 if (name == nullptr || name->empty() || integer == nullptr || *integer <= 0)
135 return invalid<SkillDefinition>("skill cost requires a positive attribute amount",
136 "costs[" + std::to_string(index) + "]");
137 auto cost = eve::resource::ResourceCost::create(*name, *integer);
138 if (!cost) return eve::Result<SkillDefinition>::failure(cost.status());
139 items.push_back(std::move(cost).takeValue());
140 }
141 if (!items.empty()) {
142 auto cost = eve::resource::CostSpec::create(std::move(items));
143 if (!cost) return eve::Result<SkillDefinition>::failure(cost.status());
144 result.cost = std::move(cost).takeValue();
145 }
146 }
147 return eve::Result<SkillDefinition>::success(std::move(result));
148}

157eve::Result<SkillRuntimeState> decodeState(const eve::Value& value) {
158 const auto* object = value.getIf<eve::Value::Object>();
159 if (object == nullptr) return invalid<SkillRuntimeState>("skill runtime state must be an object", "state");
160 static const std::set<std::string> fields = {"cooldownRemaining", "learned"};
161 for (const auto& [name, unused] : *object) {
162 (void)unused;
163 if (!fields.contains(name))
164 return invalid<SkillRuntimeState>("unknown skill runtime state field", "state." + name);
165 }
166 if (!object->contains("cooldownRemaining") || !object->contains("learned"))
167 return invalid<SkillRuntimeState>("skill runtime state is missing a required field", "state");
168 const auto* cooldown = object->at("cooldownRemaining").getIf<double>();
169 const auto* learned = object->at("learned").getIf<bool>();
170 if (cooldown == nullptr || learned == nullptr || !std::isfinite(*cooldown) || *cooldown < 0.0 ||
171 *cooldown > std::numeric_limits<float>::max())
172 return invalid<SkillRuntimeState>("skill cooldown state is invalid", "state.cooldownRemaining");
173 return eve::Result<SkillRuntimeState>::success(SkillRuntimeState{static_cast<float>(*cooldown), *learned});
174}

178eve::Result<SkillDefinitionRuntime> SkillDefinitionRuntime::create(eve::definitions::DefinitionRegistry& registry,
179 eve::DefinitionRef definition,
180 eve::PersistentId instanceId,
181 eve::definition::ReloadPolicy policy) {
182 if (!definition.id().isValid() || definition.id().namespaceName() != "rpg.skill")
183 return invalid<SkillDefinitionRuntime>("skill definition reference must use rpg.skill namespace", "definition");
184 auto handle = currentHandle(registry, definition);
185 if (!handle) return eve::Result<SkillDefinitionRuntime>::failure(handle.status());
186 auto resolved = registry.resolveHandle(handle.value());
187 if (!resolved) return eve::Result<SkillDefinitionRuntime>::failure(resolved.status());
188 auto typed = parseDefinition(resolved.value().get());
189 if (!typed) return eve::Result<SkillDefinitionRuntime>::failure(typed.status());
190 auto runtime = eve::definition::RuntimeInstance<SkillRuntimeState>::create(
191 instanceId, std::move(definition), handle.value().generation, SkillRuntimeState{});
192 if (!runtime) return eve::Result<SkillDefinitionRuntime>::failure(runtime.status());
193 return eve::Result<SkillDefinitionRuntime>::success(
194 SkillDefinitionRuntime(registry, std::move(runtime).takeValue(), policy));
195}

232 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,

234 "actor", {}, "rpg.skill.definition_runtime"));
235 auto candidate = *actor->skills().operator->();
236 const std::string id(identity().definition.id().name());
237 if (state().learned)
238 candidate.known[id] = SkillRuntime{state().cooldownRemaining};

243 return eve::Result<void>::success();

246eve::Result<eve::definition::ReloadOutcome> SkillDefinitionRuntime::reload(eve::definition::ReloadPolicy policy) {
247 if (registry_ == nullptr)
248 return invalid<eve::definition::ReloadOutcome>("skill definition registry is not bound", "registry");
249 auto next = currentHandle(*registry_, identity().definition);
250 if (!next) return eve::Result<eve::definition::ReloadOutcome>::failure(next.status());
251 auto resolved = registry_->resolveHandle(next.value());
252 if (!resolved) return eve::Result<eve::definition::ReloadOutcome>::failure(resolved.status());
253 auto typed = parseDefinition(resolved.value().get());
254 if (!typed) return eve::Result<eve::definition::ReloadOutcome>::failure(typed.status());
255 const float nextCooldown = typed.value().cooldown;
256 SkillRuntimeState defaults{};
257 auto rebuild = [nextCooldown](const SkillRuntimeState& oldState, const eve::definition::InstanceIdentity&,
258 const eve::definition::DefinitionHandle&) {
259 SkillRuntimeState rebuilt = oldState;
260 rebuilt.cooldownRemaining = std::clamp(oldState.cooldownRemaining, 0.f, nextCooldown);
261 return eve::Result<SkillRuntimeState>::success(rebuilt);
262 };
263 auto result = runtime_.reload(next.value(), policy, defaults, std::move(rebuild));
264 if (result.ok()) policy_ = policy;
265 return result;
266}

286 const eve::SnapshotHashProvider& hashProvider) {
