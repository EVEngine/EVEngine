44bool readNumber(const eve::Value::Object &object, std::string_view name, double &result) {
45 const auto *value = field(object, name);
46 if (const auto *number = value ? value->getIf<double>() : nullptr) {
47 result = *number;
48 return std::isfinite(result);
49 }
50 if (const auto *integer = value ? value->getIf<std::int64_t>() : nullptr) {
51 result = static_cast<double>(*integer);
52 return std::isfinite(result);
53 }
54 return false;

57eve::Result<std::vector<std::string>> readStrings(const eve::Value::Object &object, std::string_view name) {
58 const auto *value = field(object, name);
59 if (value == nullptr) return eve::Result<std::vector<std::string>>::success({});
60 const auto *array = value->getIf<eve::Value::Array>();
61 if (array == nullptr)
62 return eve::Result<std::vector<std::string>>::failure(
63 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "skill field must be an array",
64 std::string(name), {}, "rpg.skill.definition_runtime"));
65 std::vector<std::string> result;
66 result.reserve(array->size());
67 for (std::size_t index = 0; index < array->size(); ++index) {
68 const auto *text = (*array)[index].getIf<std::string>();
69 if (text == nullptr || text->empty())
70 return eve::Result<std::vector<std::string>>::failure(eve::Diagnostic::error(
71 eve::DiagnosticCode::InvalidArgument, "skill array item must be a non-empty string",
72 std::string(name) + "[" + std::to_string(index) + "]", {}, "rpg.skill.definition_runtime"));
73 result.push_back(*text);
74 }
75 return eve::Result<std::vector<std::string>>::success(std::move(result));

78eve::Result<SkillDefinition> parseDefinition(const eve::definitions::Definition &source) {
79 auto owned = eve::Value::fromJson(source.json);
80 if (!owned) return eve::Result<SkillDefinition>::failure(owned.status());
81 const auto *payload = owned.value().getIf<eve::Value::Object>();
82 if (payload == nullptr)
83 return eve::Result<SkillDefinition>::failure(
84 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "skill definition payload must be an object",
85 "json", {}, "rpg.skill.definition_runtime"));
86
87 SkillDefinition result;
88 result.id = source.id;
89 const auto *target = field(*payload, "targetType");
90 if (target != nullptr) {
91 const auto *text = target->getIf<std::string>();
92 if (text == nullptr || text->empty())
93 return eve::Result<SkillDefinition>::failure(eve::Diagnostic::error(
94 eve::DiagnosticCode::InvalidArgument, "skill targetType must be a non-empty string", "targetType", {},
95 "rpg.skill.definition_runtime"));
96 result.targetType = *text;
97 }
98 double number = 0.0;
99 if (field(*payload, "cooldown") != nullptr) {
100 if (!readNumber(*payload, "cooldown", number) || number < 0.0 || number > std::numeric_limits<float>::max())
101 return eve::Result<SkillDefinition>::failure(eve::Diagnostic::error(
102 eve::DiagnosticCode::InvalidArgument, "skill cooldown must be finite and non-negative", "cooldown", {},
103 "rpg.skill.definition_runtime"));
104 result.cooldown = static_cast<float>(number);
105 }
106 if (field(*payload, "castTime") != nullptr) {
107 if (!readNumber(*payload, "castTime", number) || number < 0.0 || number > std::numeric_limits<float>::max())
108 return eve::Result<SkillDefinition>::failure(eve::Diagnostic::error(
109 eve::DiagnosticCode::InvalidArgument, "skill castTime must be finite and non-negative", "castTime", {},
110 "rpg.skill.definition_runtime"));
111 result.castTime = static_cast<float>(number);
112 }
113 auto effects = readStrings(*payload, "grantedEffects");
114 if (!effects) return eve::Result<SkillDefinition>::failure(effects.status());
115 result.grantedEffects = std::move(effects).takeValue();
116 auto tags = readStrings(*payload, "tags");
117 if (!tags) return eve::Result<SkillDefinition>::failure(tags.status());
118 result.tags = std::move(tags).takeValue();
119 if (const auto *condition = field(*payload, "castCondition")) {
120 auto decoded = decodeSkillCondition(*condition);
121 if (!decoded) return eve::Result<SkillDefinition>::failure(decoded.status());
122 result.castCondition = std::move(decoded).takeValue();
123 }
124 auto extras = field(*payload, "extra");
125 if (extras != nullptr) {
126 const auto *extraObject = extras->getIf<eve::Value::Object>();
127 if (extraObject == nullptr)
128 return eve::Result<SkillDefinition>::failure(
129 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "skill extra must be an object", "extra",
130 {}, "rpg.skill.definition_runtime"));
131 for (const auto &[key, value] : *extraObject) {
132 const auto *text = value.getIf<std::string>();
133 if (text == nullptr)
134 return eve::Result<SkillDefinition>::failure(
135 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "skill extra values must be strings",
136 "extra." + key, {}, "rpg.skill.definition_runtime"));
137 result.extra.emplace(key, *text);
138 }
139 }
140 const auto *costs = field(*payload, "costs");
141 if (costs != nullptr) {
142 const auto *array = costs->getIf<eve::Value::Array>();
143 if (array == nullptr)
144 return eve::Result<SkillDefinition>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
145 "skill costs must be an array", "costs",
146 {}, "rpg.skill.definition_runtime"));
147 std::vector<eve::resource::ResourceCost> items;
148 for (std::size_t index = 0; index < array->size(); ++index) {
149 const auto *item = (*array)[index].getIf<eve::Value::Object>();
150 if (item == nullptr)
151 return eve::Result<SkillDefinition>::failure(
152 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "skill cost must be an object",
153 "costs[" + std::to_string(index) + "]", {}, "rpg.skill.definition_runtime"));
154 const auto *resource = field(*item, "attribute");
155 const auto *amount = field(*item, "amount");
156 const auto *name = resource ? resource->getIf<std::string>() : nullptr;
157 const auto *integer = amount ? amount->getIf<std::int64_t>() : nullptr;
158 if (name == nullptr || name->empty() || integer == nullptr || *integer <= 0)
159 return eve::Result<SkillDefinition>::failure(eve::Diagnostic::error(
160 eve::DiagnosticCode::InvalidArgument, "skill cost requires a positive attribute amount",
161 "costs[" + std::to_string(index) + "]", {}, "rpg.skill.definition_runtime"));
162 auto cost = eve::resource::ResourceCost::create(*name, *integer);
163 if (!cost) return eve::Result<SkillDefinition>::failure(cost.status());
164 items.push_back(std::move(cost).takeValue());
165 }
166 if (!items.empty()) {
167 auto cost = eve::resource::CostSpec::create(std::move(items));
168 if (!cost) return eve::Result<SkillDefinition>::failure(cost.status());
169 result.cost = std::move(cost).takeValue();
170 }
171 }
172 return eve::Result<SkillDefinition>::success(std::move(result));
173}

182eve::Result<SkillRuntimeState> decodeState(const eve::Value &value) {
183 const auto *object = value.getIf<eve::Value::Object>();
184 if (object == nullptr)
185 return eve::Result<SkillRuntimeState>::failure(
186 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "skill runtime state must be an object",
187 "state", {}, "rpg.skill.definition_runtime"));
188 static const std::set<std::string> fields = {"cooldownRemaining", "learned"};
189 for (const auto &[name, unused] : *object) {
190 (void)unused;
191 if (!fields.contains(name))
192 return eve::Result<SkillRuntimeState>::failure(
193 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "unknown skill runtime state field",
194 "state." + name, {}, "rpg.skill.definition_runtime"));
195 }
196 if (!object->contains("cooldownRemaining") || !object->contains("learned"))
197 return eve::Result<SkillRuntimeState>::failure(eve::Diagnostic::error(
198 eve::DiagnosticCode::InvalidArgument, "skill runtime state is missing a required field", "state", {},
199 "rpg.skill.definition_runtime"));
200 const auto *cooldown = object->at("cooldownRemaining").getIf<double>();
201 const auto *learned = object->at("learned").getIf<bool>();
202 if (cooldown == nullptr || learned == nullptr || !std::isfinite(*cooldown) || *cooldown < 0.0 ||
203 *cooldown > std::numeric_limits<float>::max())
204 return eve::Result<SkillRuntimeState>::failure(
205 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "skill cooldown state is invalid",
206 "state.cooldownRemaining", {}, "rpg.skill.definition_runtime"));
207 return eve::Result<SkillRuntimeState>::success(SkillRuntimeState{static_cast<float>(*cooldown), *learned});
208}

213 eve::DefinitionRef definition,
214 eve::PersistentId instanceId,
215 eve::definition::ReloadPolicy policy) {

227 instanceId, std::move(definition), handle.value().generation, SkillRuntimeState{});
228 if (!runtime) return eve::Result<SkillDefinitionRuntime>::failure(runtime.status());
229 return eve::Result<SkillDefinitionRuntime>::success(
230 SkillDefinitionRuntime(registry, std::move(runtime).takeValue(), policy));
231}

272eve::Result<void> SkillDefinitionRuntime::applyTo(RPGActor *actor) const {
273 if (actor == nullptr)
274 return eve::Result<void>::failure(eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
275 "skill runtime cannot project to a null RPGActor",
276 "actor", {}, "rpg.skill.definition_runtime"));
277 auto candidate = *actor->skills().operator->();
278 const std::string id(identity().definition.id().name());
279 if (state().learned)

283 using std::swap;

285 return eve::Result<void>::success();

288eve::Result<eve::definition::ReloadOutcome> SkillDefinitionRuntime::reload(eve::definition::ReloadPolicy policy) {
289 if (registry_ == nullptr)
290 return eve::Result<eve::definition::ReloadOutcome>::failure(
291 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "skill definition registry is not bound",
292 "registry", {}, "rpg.skill.definition_runtime"));
293 auto next = currentHandle(*registry_, identity().definition);
294 if (!next) return eve::Result<eve::definition::ReloadOutcome>::failure(next.status());
295 auto resolved = registry_->resolveHandle(next.value());
296 if (!resolved) return eve::Result<eve::definition::ReloadOutcome>::failure(resolved.status());
297 auto typed = parseDefinition(resolved.value().get());
298 if (!typed) return eve::Result<eve::definition::ReloadOutcome>::failure(typed.status());
299 const float nextCooldown = typed.value().cooldown;
300 SkillRuntimeState defaults{};
301 auto rebuild = [nextCooldown](const SkillRuntimeState &oldState, const eve::definition::InstanceIdentity &,
302 const eve::definition::DefinitionHandle &) {
303 SkillRuntimeState rebuilt = oldState;
304 rebuilt.cooldownRemaining = std::clamp(oldState.cooldownRemaining, 0.f, nextCooldown);
305 return eve::Result<SkillRuntimeState>::success(rebuilt);
306 };
307 auto result = runtime_.reload(next.value(), policy, defaults, std::move(rebuild));
308 if (result.ok()) policy_ = policy;
309 return result;
310}

431bool SkillSystem::canCast(RPGActor *actor, const std::string &skillId, std::string *reason) {
432 auto fail = [&](const std::string &r) {
433 if (reason) *reason = r;
434 return false;
435 };
436 if (!actor) return fail("no_actor");
437 const SkillDefinition *def = SkillRegistry::find(skillId);
438 if (!def) return fail("unknown_skill");
439 if (!knows(actor, skillId)) return fail("not_learned");
440 if (isCasting(actor)) return fail("already_casting");
441 if (getCooldownRemaining(actor, skillId) > 0.f) return fail("on_cooldown");
442 if (!checkCostsAffordable(actor, *def)) return fail("insufficient_cost");
443
444 for (auto &kv : conditions()) {
445 std::string r;
446 if (!kv.second(actor, *def, r)) return fail(r.empty() ? kv.first : r);
447 }
448 const auto condition = SkillConditionAdapter::evaluate(actor, *def, def->castCondition);
449 if (!condition.passed()) return fail(condition::conditionReasonCodeName(condition.reasonCode()));
450 return true;
451}

453bool SkillSystem::beginCast(RPGActor *actor, const std::string &skillId, RPGActor *target, std::string *reason) {
454 if (!canCast(actor, skillId, reason)) return false;
455 const SkillDefinition *def = SkillRegistry::find(skillId);
456
457 deductCosts(actor, *def);
458 setCooldownRemaining(actor, skillId, def->cooldown);
459
460 if (def->castTime <= 0.f) {
461 resolveCast(actor, *def, target);
462 return true;
463 }
464
465 CastingState &cs = actor->skills()->casting;
466 cs.active = true;
467 cs.skillId = skillId;
468 cs.remaining = def->castTime;
469 cs.totalCastTime = def->castTime;
470 cs.target = target;
471 return true;
472}

485float SkillSystem::getCastProgress(RPGActor *actor) {
486 if (!actor) return 0.f;
487 const CastingState &cs = actor->skills()->casting;
488 if (!cs.active || cs.totalCastTime <= 0.f) return 0.f;

493void SkillSystem::update(float dt) {
494 for (RPGActor *actor : RPGActor::liveActors()) {
495 for (auto &kv : actor->skills()->known)
496 if (kv.second.cooldownRemaining > 0.f) kv.second.cooldownRemaining -= dt;
497
498 CastingState &cs = actor->skills()->casting;
499 if (!cs.active) continue;
500 cs.remaining -= dt;
501 if (cs.remaining <= 0.f) {
