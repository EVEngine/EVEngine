141bool readColor(const EditorValue& value, float dest[4]) {
142 const auto* array = value.getIf<EditorValue::Array>();
143 if (!array || array->size() < 4) return false;
144 for (int i = 0; i < 4; ++i) {
145 const auto* number = (*array)[static_cast<std::size_t>(i)].getIf<double>();
146 if (!number || !std::isfinite(*number) || *number < 0.0 || *number > 1.0) return false;
147 dest[i] = static_cast<float>(*number);
148 }
149 return true;
150}

214EditorResult<ui::Theme> parseThemeTokens(const EditorValue& value) {
215 if (!value.getIf<EditorValue::Object>())
216 return tokenError<ui::Theme>(EditorStatus::Rejected, "editor.ui-theme.tokens",
217 "Theme tokens must be an object");
218 ui::Theme theme;
219 for (const ColorTok& token : kColors) {
220 const EditorValue* entry = field(value, token.path);
221 if (!entry || !readColor(*entry, theme.*(token.member)))
222 return tokenError<ui::Theme>(EditorStatus::Rejected, "editor.ui-theme.color",
223 std::string("Theme color is missing or invalid: ") + token.path);
224 }
225 for (const FloatTok& token : kFloats) {
226 const EditorValue* entry = field(value, token.path);
227 const auto* number = entry ? entry->getIf<double>() : nullptr;
228 if (!number || !std::isfinite(*number))
229 return tokenError<ui::Theme>(EditorStatus::Rejected, "editor.ui-theme.float",
230 std::string("Theme number is missing or invalid: ") + token.path);
231 theme.*(token.member) = static_cast<float>(*number);
232 }
233 for (const LayoutTok& token : kLayout) {
234 const EditorValue* entry = field(value, token.path);
235 const auto* number = entry ? entry->getIf<double>() : nullptr;
236 if (!number || !std::isfinite(*number))
237 return tokenError<ui::Theme>(EditorStatus::Rejected, "editor.ui-theme.layout",
238 std::string("Theme layout number is missing or invalid: ") + token.path);
239 theme.layout.*(token.member) = static_cast<float>(*number);
240 }
241 const EditorValue* keyboard = field(value, "nav.keyboard");
242 const EditorValue* gamepad = field(value, "nav.gamepad");
243 const auto* keyboardFlag = keyboard ? keyboard->getIf<bool>() : nullptr;
244 const auto* gamepadFlag = gamepad ? gamepad->getIf<bool>() : nullptr;
245 if (!keyboardFlag || !gamepadFlag)
246 return tokenError<ui::Theme>(EditorStatus::Rejected, "editor.ui-theme.nav",
247 "Theme navigation flags are required");
248 theme.navEnableKeyboard = *keyboardFlag;
249 theme.navEnableGamepad = *gamepadFlag;
250 auto diagnostics = validateThemeTokens(theme);
251 if (!diagnostics.empty())
252 return tokenError<ui::Theme>(EditorStatus::Rejected, "editor.ui-theme.range",
253 diagnostics.front().message());
254 return eve::editing::applied<ui::Theme>(theme);
255}

257EditorResult<void> assignThemeToken(ui::Theme& theme, const PropertyPath& path, const EditorValue& value) {
258 for (const ColorTok& token : kColors) {
259 if (path != PropertyPath(token.path)) continue;
260 if (!readColor(value, theme.*(token.member)))
261 return tokenError<void>(EditorStatus::Rejected, "editor.ui-theme.token-color",
262 std::string("Theme color token is invalid: ") + token.path);
263 return eve::editing::applied<void>();
264 }
265 for (const FloatTok& token : kFloats) {
266 if (path != PropertyPath(token.path)) continue;
267 const auto* number = value.getIf<double>();
268 if (!number || !inRange(static_cast<float>(*number), token.minimum, token.maximum))
269 return tokenError<void>(EditorStatus::Rejected, "editor.ui-theme.token-float",
270 std::string("Theme number token is invalid: ") + token.path);
271 theme.*(token.member) = static_cast<float>(*number);
272 return eve::editing::applied<void>();
273 }
274 for (const LayoutTok& token : kLayout) {
275 if (path != PropertyPath(token.path)) continue;
276 const auto* number = value.getIf<double>();
277 if (!number || !inRange(static_cast<float>(*number), token.minimum, token.maximum))
278 return tokenError<void>(EditorStatus::Rejected, "editor.ui-theme.token-layout",
279 std::string("Theme layout token is invalid: ") + token.path);
280 theme.layout.*(token.member) = static_cast<float>(*number);
281 return eve::editing::applied<void>();
282 }
283 if (path == PropertyPath("nav.keyboard")) {
284 const auto* flag = value.getIf<bool>();
285 if (!flag)
286 return tokenError<void>(EditorStatus::Rejected, "editor.ui-theme.token-nav",
287 "Theme navigation token is invalid");
288 theme.navEnableKeyboard = *flag;
289 return eve::editing::applied<void>();
290 }
291 if (path == PropertyPath("nav.gamepad")) {
292 const auto* flag = value.getIf<bool>();
293 if (!flag)
294 return tokenError<void>(EditorStatus::Rejected, "editor.ui-theme.token-nav",
295 "Theme navigation token is invalid");
296 theme.navEnableGamepad = *flag;
297 return eve::editing::applied<void>();
298 }
299 return tokenError<void>(EditorStatus::NotFound, "editor.ui-theme.token",
300 "Unknown theme token: " + path.value());
301}

303EditorValue readThemeToken(const ui::Theme& theme, const PropertyPath& path) {
304 for (const ColorTok& token : kColors) {
305 if (path == PropertyPath(token.path)) return colorValue(theme.*(token.member));
306 }
307 for (const FloatTok& token : kFloats) {
308 if (path == PropertyPath(token.path)) return static_cast<double>(theme.*(token.member));
309 }
310 for (const LayoutTok& token : kLayout) {
311 if (path == PropertyPath(token.path)) return static_cast<double>(theme.layout.*(token.member));
312 }
313 if (path == PropertyPath("nav.keyboard")) return theme.navEnableKeyboard;
314 if (path == PropertyPath("nav.gamepad")) return theme.navEnableGamepad;
315 return {};
316}

349std::vector<EditorDiagnostic> validateThemeTokens(const ui::Theme& theme) {
350 std::vector<EditorDiagnostic> diagnostics;
351 for (const ColorTok& token : kColors) {
352 const float* color = theme.*(token.member);
353 for (int i = 0; i < 4; ++i) {
354 if (!inRange(color[i], 0.f, 1.f)) {
355 diagnostics.push_back(tokenDiagnostic("editor.ui-theme.color-range",
356 std::string("Theme color channel is out of range: ") + token.path));
357 break;
358 }
359 }
360 }
361 for (const FloatTok& token : kFloats) {
362 if (!inRange(theme.*(token.member), token.minimum, token.maximum))
363 diagnostics.push_back(tokenDiagnostic("editor.ui-theme.float-range",
364 std::string("Theme number is out of range: ") + token.path));
365 }
366 for (const LayoutTok& token : kLayout) {
367 if (!inRange(theme.layout.*(token.member), token.minimum, token.maximum))
368 diagnostics.push_back(tokenDiagnostic("editor.ui-theme.layout-range",
369 std::string("Theme layout number is out of range: ") + token.path));
370 }
371 return diagnostics;
372}
