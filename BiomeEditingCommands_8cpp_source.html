13const editing::Value* field(const editing::Value& value, const char* key) {
14 const auto* object = value.getIf<editing::Value::Object>();
15 if (!object) return nullptr;
16 const auto found = object->find(key);
17 return found == object->end() ? nullptr : &found->second;
18}

25bool readNumber(const editing::Value& value, const char* key, double& output, bool required) {
26 const editing::Value* entry = field(value, key);
27 if (!entry) return !required;
28 if (const auto* real = entry->getIf<double>()) {
29 output = *real;
30 return std::isfinite(output);
31 }
32 if (const auto* integer = entry->getIf<std::int64_t>()) {
33 output = static_cast<double>(*integer);
34 return true;
35 }
36 return false;
37}

39bool readInt(const editing::Value& value, const char* key, int& output, bool required) {
40 if (!field(value, key)) return !required;
41 double parsed = 0.0;
42 if (!readNumber(value, key, parsed, true) || parsed < -100000.0 || parsed > 100000.0) return false;
43 output = static_cast<int>(parsed);
44 return true;
45}

103editing::Result<void> registerEditingCommands(editing::IEditingCommandRegistry& registry) {
104 auto registered = addCommand(
105 registry, "biome.layer.create.v1", "Create biome layer",
106 [](editing::IEditableTarget& target, const editing::CommandRequest& request) {
107 auto* biome = asBiome(target);
108 const std::string* id = stringField(request.payload, "id");
109 const std::string* spatial = stringField(request.payload, "spatial");
110 if (!biome || !id || id->empty() || !spatial || spatial->empty())
111 return error<editing::CommandPlan>(editing::Status::Rejected, "biome.editing.layer-payload",
112 "Biome layer create requires id and spatial asset");
113 const std::string* name = stringField(request.payload, "name");
114 BiomeLayerValue layer;
115 layer.id = editing::ObjectId(*id);
116 layer.name = name && !name->empty() ? *name : *id;
117 layer.spatialAsset = *spatial;
118 layer.priority = 0;
119 layer.density = 1.f;
120 double density = 1.0;
121 if (!readInt(request.payload, "priority", layer.priority, false) ||
122 !readNumber(request.payload, "density", density, false))
123 return error<editing::CommandPlan>(editing::Status::Rejected, "biome.editing.layer-number",
124 "Biome layer priority and density must be numbers");
125 layer.density = static_cast<float>(density);
126 return planFrom(biome->makeCreateLayer(layer), editing::Value::Object{{"id", *id}});
127 });
128 if (!registered.ok()) return registered;
129
130 registered = addCommand(
131 registry, "biome.asset.create.v1", "Create biome asset",
132 [](editing::IEditableTarget& target, const editing::CommandRequest& request) {
133 auto* biome = asBiome(target);
134 const std::string* layer = stringField(request.payload, "layer");
135 const std::string* id = stringField(request.payload, "id");
136 const std::string* asset = stringField(request.payload, "asset");
137 if (!biome || !layer || layer->empty() || !id || id->empty() || !asset || asset->empty())
138 return error<editing::CommandPlan>(editing::Status::Rejected, "biome.editing.asset-payload",
139 "Biome asset create requires layer, id and asset");
140 BiomeAssetValue entry;
141 entry.id = editing::ObjectId(*id);
142 entry.asset = *asset;
143 double weight = 1.0, minScale = 1.0, maxScale = 1.0;
144 if (!readNumber(request.payload, "weight", weight, false) ||
145 !readNumber(request.payload, "minScale", minScale, false) ||
146 !readNumber(request.payload, "maxScale", maxScale, false) ||
147 !readBool(request.payload, "randomYaw", entry.randomYaw, false))
148 return error<editing::CommandPlan>(editing::Status::Rejected, "biome.editing.asset-number",
149 "Biome asset weight, scale and yaw must be valid");
150 entry.weight = static_cast<float>(weight);
151 entry.minScale = static_cast<float>(minScale);
152 entry.maxScale = static_cast<float>(maxScale);
153 return planFrom(biome->makeCreateAsset(editing::ObjectId(*layer), entry),
154 editing::Value::Object{{"id", *id}, {"layer", *layer}});
155 });
156 if (!registered.ok()) return registered;
157
158 return addCommand(

170 "Biome property requires item, type, path and value");
171 return planFrom(properties->makeSet(selectionFor(*biome, *item, *type), editing::PropertyPath(*path),
172 *value, editing::PropertySetMode::Absolute),
173 editing::Value::Object{{"item", *item}, {"path", *path}});
174 });
