18std::map<ObjectId, SceneObjectSnapshot> collect(const SceneTargetBase& target) {
19 std::map<ObjectId, SceneObjectSnapshot> result;
20 const EditorValue snapshot = target.snapshotValue();
21 const auto* root = snapshot.getIf<EditorValue::Object>();
22 if (!root) return result;
23 const auto found = root->find("objects");
24 if (found == root->end()) return result;
25 const auto* objects = found->second.getIf<EditorValue::Array>();
26 if (!objects) return result;
27 for (const EditorValue& value : *objects) {
28 const auto* object = value.getIf<EditorValue::Object>();
29 if (!object) continue;
30 const auto* id = object->at("id").getIf<std::string>();
31 auto snapshot = target.sceneObject(ObjectId(*id));
32 if (snapshot.ok()) result.emplace(snapshot.value().id, snapshot.value());
33 }
34 return result;
35}

57bool hostMatches(scene::SceneHost* host, const SceneObjectSnapshot& object) {
58 const int index = host->findIndexById(object.id.value());
59 scene::SceneNode* node = host->getNode(index);
60 if (!node) return false;
61 const std::string parent = node->parent >= 0 ? host->getNode(node->parent)->id : std::string{};
62 const auto close = [](double left, double right) { return std::abs(left - right) <= 1e-5; };
63 return parent == object.parent.value() && node->name == object.name &&
64 close(node->x, object.transform.x) && close(node->y, object.transform.y) &&
65 close(node->z, object.transform.z) && close(node->pitch, object.transform.rotationX) &&
66 close(node->yaw, object.transform.rotationY) && close(node->roll, object.transform.rotationZ) &&
67 close(node->sx, object.transform.scaleX) && close(node->sy, object.transform.scaleY) &&
68 close(node->sz, object.transform.scaleZ);
69}

73SceneHostEditorTarget::SceneHostEditorTarget(std::string id, scene::SceneHost* host)
74 : SceneTargetBase(std::move(id), "runtime-scene-host"), host_(host) {
75 if (!host_) return;
76 const auto tree = host_->tree();
77 for (const scene::SceneNode& node : tree->nodes) {
78 CreateSceneObjectRequest request;
79 request.id = ObjectId(node.id);
80 request.parent = node.parent >= 0 ? ObjectId(tree->nodes[size_t(node.parent)].id) : ObjectId{};
81 request.name = node.name;
82 request.transform = {node.x, node.y, node.z, node.pitch, node.yaw, node.roll,
83 node.sx, node.sy, node.sz};
84 auto operation = makeCreate(request);
85 if (operation.ok()) {
86 auto applied = SceneTargetBase::applyDomainOperation(operation.value());
87 applied.ignore();
88 }
89 }
90 clearDirtyRegion();
91}

99EditorResult<std::vector<SceneComponentLinkSnapshot>> SceneHostEditorTarget::componentLinks(
100 const ObjectId& object) const {
101 if (!host_)
102 return liveError<std::vector<SceneComponentLinkSnapshot>>(EditorStatus::Unsupported,
103 "editor.scene.live-host-required", "Component links are unavailable on a staging target");
104 auto nodeResult = host_->findById(object.value());
105 if (!nodeResult.ok() || !nodeResult.value())
106 return liveError<std::vector<SceneComponentLinkSnapshot>>(EditorStatus::NotFound,
107 "editor.scene.component-object-not-found", "Live scene object does not exist: " + object.value());
108 std::vector<SceneComponentLinkSnapshot> result;
109 for (const scene::SceneLink& link : nodeResult.value()->links) {
110 const scene::LinkOps* operations = scene::linkOps(link.kind);
111 const bool alive = link.target && (!operations || !operations->alive || operations->alive(link.target));
112 result.push_back({scene::linkKindName(link.kind), link.syncMode, alive});
113 }
114 std::sort(result.begin(), result.end(), [](const auto& a, const auto& b) { return a.kind < b.kind; });
115 return eve::editing::applied<std::vector<SceneComponentLinkSnapshot>>(std::move(result));
116}

136EditorResult<void> SceneHostEditorTarget::synchronizeHost(const SceneTargetBase& desiredTarget) {
137 if (!host_)
138 return liveError<void>(EditorStatus::Rejected, "editor.scene.live-host-required",
139 "Live scene host is unavailable");
140 const auto current = collect(*this);
141 const auto desired = collect(desiredTarget);
142 for (const auto& [id, object] : current) {
143 static_cast<void>(object);
144 if (!hostMatches(host_, object))
145 return liveError<void>(EditorStatus::Conflict, "editor.scene.live-host-diverged",
146 "Live SceneHost changed outside this editor target: " + id.value());
147 if (!desired.contains(id) && host_->getChildCountById(id.value()) != 0)
148 return liveError<void>(EditorStatus::Conflict, "editor.scene.live-delete-not-leaf",
149 "Live SceneHost object gained children before deletion: " + id.value());
150 }
151 for (const auto& [id, object] : desired) {
152 if (!current.contains(id) && host_->hasNode(id.value()))
153 return liveError<void>(EditorStatus::Conflict, "editor.scene.live-id-conflict",
154 "Live SceneHost already contains newly-created id: " + id.value());
155 if (object.transform.scaleX == 0.0 || object.transform.scaleY == 0.0 || object.transform.scaleZ == 0.0)
156 return liveError<void>(EditorStatus::Rejected, "editor.scene.live-zero-scale",
157 "Live SceneHost rejects zero scale: " + id.value());
158 }
159
160 for (const auto& [id, object] : current) {
161 static_cast<void>(object);
162 if (!desired.contains(id) && host_->removeLeaf(id.value()) != scene::SceneMutationStatus::Applied)
163 return liveError<void>(EditorStatus::Conflict, "editor.scene.live-delete-failed",
164 "Live SceneHost rejected leaf deletion: " + id.value());
165 }
166 std::set<ObjectId> pending;
167 for (const auto& [id, object] : desired)
168 if (!current.contains(id)) pending.insert(id);
169 while (!pending.empty()) {
170 bool progressed = false;
171 for (auto iterator = pending.begin(); iterator != pending.end();) {
172 const SceneObjectSnapshot& object = desired.at(*iterator);
173 if (!object.parent.empty() && !host_->hasNode(object.parent.value())) {
174 ++iterator;
175 continue;
176 }
177 if (host_->appendNode(runtimeNode(object), object.parent.value()) != scene::SceneMutationStatus::Applied)
178 return liveError<void>(EditorStatus::Conflict, "editor.scene.live-create-failed",
179 "Live SceneHost rejected object creation: " + object.id.value());
180 iterator = pending.erase(iterator);
181 progressed = true;
182 }
183 if (!progressed)
184 return liveError<void>(EditorStatus::Conflict, "editor.scene.live-parent-order",
185 "Live SceneHost could not resolve created object parents");
186 }
187 for (const auto& [id, object] : desired) {
188 const auto old = current.find(id);
189 if (old != current.end() && old->second.parent != object.parent &&
190 !host_->setParentById(id.value(), object.parent.value()))
191 return liveError<void>(EditorStatus::Conflict, "editor.scene.live-reparent-failed",
192 "Live SceneHost rejected object reparent: " + id.value());
193 if (old == current.end() || old->second.name != object.name)
194 if (host_->renameNode(id.value(), object.name) != scene::SceneMutationStatus::Applied)
195 return liveError<void>(EditorStatus::Conflict, "editor.scene.live-rename-failed",
196 "Live SceneHost rejected object rename: " + id.value());
197 if (old == current.end() || !sameTransform(old->second.transform, object.transform)) {
198 const SceneTransformValue& transform = object.transform;
199 if (host_->setLocalTransform(id.value(), static_cast<float>(transform.x),
200 static_cast<float>(transform.y), static_cast<float>(transform.z),
201 static_cast<float>(transform.rotationY),
202 static_cast<float>(transform.rotationX),
203 static_cast<float>(transform.rotationZ),
204 static_cast<float>(transform.scaleX),
205 static_cast<float>(transform.scaleY),
206 static_cast<float>(transform.scaleZ)) !=
207 scene::SceneMutationStatus::Applied)
208 return liveError<void>(EditorStatus::Conflict, "editor.scene.live-transform-failed",
209 "Live SceneHost rejected object transform: " + id.value());
210 }
211 }
212 return eve::editing::applied<void>();
213}
