16 float& outNx, float& outNy, float& outNz) {
17 if (!std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z) || !(epsilon > 0.f))
18 return SdfField3DGradientStatus::Undefined;
19 const float dx = field.sample(x + epsilon, y, z) - field.sample(x - epsilon, y, z);
20 const float dy = field.sample(x, y + epsilon, z) - field.sample(x, y - epsilon, z);
21 const float dz = field.sample(x, y, z + epsilon) - field.sample(x, y, z - epsilon);
22 const float len = std::sqrt(dx * dx + dy * dy + dz * dz);
23 if (!std::isfinite(len) || len <= 1e-8f) return SdfField3DGradientStatus::Undefined;
24 outNx = dx / len;
25 outNy = dy / len;
26 outNz = dz / len;
27 return SdfField3DGradientStatus::Defined;
28}
