22 std::shared_ptr<VehicleWheelRuntime>& runtime, const SimulationStep& step) {
23 using namespace physics;
24 auto pending = submissionState(runtime, step);
25 if (!pending || pending.value() != VehiclePhysicsStatus::Unavailable) return pending;
26 const auto* definition = vehicle.definition()->def;
27 auto* world = body.getWorld();
28 if (!definition || !world || !body.isValid()) return reject("Vehicle definition or planar body is unavailable");
29 const auto& config = definition->suspension;
30 const double dt = step.delta.seconds(), meter = world->getMeter(), mass = body.getMass();
31 const auto input = *vehicle.input();
32 if (!positive(dt) || dt > .05 || !positive(meter) || !positive(mass) || !nonnegative(definition->maxSpeed) ||
33 !nonnegative(definition->accel) || !nonnegative(config.supportAcceleration2D) ||
34 !nonnegative(config.driveForce) || !nonnegative(config.brakeTorque) || !nonnegative(config.maxSteerAngle) ||
35 config.maxSteerAngle > 1.4 || !std::isfinite(input.throttle) || !std::isfinite(input.steer) ||
36 !std::isfinite(input.brake))
37 return reject("Invalid vehicle tire input, units or fixed duration");
38 auto wheels = config.wheels;
39 if (wheels.empty()) {
40 if (!positive(definition->radius)) return reject("Default wheel layout requires a positive chassis radius");
41 SuspensionWheel front, rear;
42 front.x = static_cast<float>(definition->radius / meter * .7);
43 rear.x = -front.x;
44 front.radius = rear.radius = static_cast<float>(std::clamp(definition->radius / meter * .5, .05, .5));
45 rear.steer = false;
46 wheels = {front, rear};
47 }
48 std::vector<VehicleEntity::SuspensionState::WheelState> candidate(wheels.size());
49 const auto& previous = vehicle.suspension()->wheels;
50 if (previous.size() == wheels.size()) candidate = previous;
51 const double wheelCount = static_cast<double>(wheels.size());
52 std::vector<ForceLoad> loads;
53 loads.reserve(wheels.size());
54 auto link = PhysicsLink::fromBody(body);
55 if (!link) return Result<VehiclePhysicsStatus>::failure(link.status());
56 auto* raw = body.raw();
57 const auto localCenter = raw->GetLocalCenter();
58 const double inertia = raw->GetInertia() - mass * b2Dot(localCenter, localCenter);
59 const double inverseInertia = raw->IsFixedRotation() ? 0 : (inertia > 0 ? 1 / inertia : 0);
60 const auto center = raw->GetWorldCenter();
61 const std::size_t driven =
62 std::count_if(wheels.begin(), wheels.end(), [](const auto& wheel) { return wheel.drive; });
63 for (std::size_t i = 0; i < wheels.size(); ++i) {
64 const auto& wheel = wheels[i];
65 if (!std::isfinite(wheel.x) || !std::isfinite(wheel.y) || wheel.z != 0 || !positive(wheel.radius))
66 return reject("2D tire mounts require finite XY meters, zero Z and positive radius");
67 const auto point = raw->GetWorldPoint({wheel.x, wheel.y});
68 const double steer = wheel.steer ? std::clamp(double(input.steer), -1.0, 1.0) * config.maxSteerAngle : 0;
69 const auto forward = raw->GetWorldVector({float(std::cos(steer)), float(std::sin(steer))});
70 const b2Vec2 lateral{-forward.y, forward.x};
71 const auto velocity = raw->GetLinearVelocityFromWorldPoint(point);
72 const auto lever = point - center;
73 const double a = b2Cross(lever, forward), b = b2Cross(lever, lateral);
74 const double crossMass = std::abs(a * b * inverseInertia);
75 auto tireModel = model(config, wheel.radius);
76 TireInput sampleInput;
77 sampleInput.longitudinalVelocity = b2Dot(velocity, forward);
78 sampleInput.lateralVelocity = b2Dot(velocity, lateral);
79 sampleInput.angularVelocity = candidate[i].angularVelocity;
80 sampleInput.normalLoad = mass * config.supportAcceleration2D / wheelCount;
81 driveInputs(sampleInput, config, input, wheel, driven, definition->maxSpeed / meter,
82 std::min(double(config.driveForce), mass * definition->accel / meter));
83 // A diagonal majorant bounds cross-axis and all-wheel chassis response (Cauchy-Schwarz).
84 // It adds implicit damping, rather than pretending off-center contacts have diagonal inverse mass.
85 sampleInput.longitudinalInverseMass = wheelCount * (1 / mass + a * a * inverseInertia + crossMass);
86 sampleInput.lateralInverseMass = wheelCount * (1 / mass + b * b * inverseInertia + crossMass);
87 auto sampled = ForceModels::tire(tireModel, sampleInput, dt);
88 if (!sampled) return Result<VehiclePhysicsStatus>::failure(sampled.status());
89 const auto& value = sampled.value();
90 auto& state = candidate[i];
91 state.angularVelocity = value.angularVelocity;
92 state.longitudinalForce = value.longitudinalForce;
93 state.lateralForce = value.lateralForce;
94 state.slipRatio = value.slipRatio;
95 state.slipAngle = value.slipAngle;
96 state.normalLoad = sampleInput.normalLoad;
97 state.grounded = sampleInput.normalLoad > 0;
98 const ForceVector force{forward.x * value.longitudinalForce + lateral.x * value.lateralForce,
99 forward.y * value.longitudinalForce + lateral.y * value.lateralForce, 0};
100 loads.push_back({link.value(), force, {point.x, point.y, 0}, {}});
101 }
102 auto batch = ForceBatch::prepare(*world, loads);
103 if (!batch) return Result<VehiclePhysicsStatus>::failure(batch.status());
104 return commit(*world, batch.value(), runtime, std::move(candidate), step);
105}
