47eve::Result<std::unique_ptr<GeometryCollectionInstance>> GeometryCollectionInstance::createFractured(
48 World3D& world, const GeometryCollectionAsset& asset, const FractureGeometry& geometry, float x, float y, float z) {
49 auto valid = geometry.validate();
50 if (!valid) return eve::Result<std::unique_ptr<GeometryCollectionInstance>>::failure(valid.status());
51 if (geometry.fragments.size() != asset.bones.size())
52 return eve::Result<std::unique_ptr<GeometryCollectionInstance>>::failure(eve::Diagnostic::error(
53 eve::DiagnosticCode::InvalidArgument, "fragment count does not match bones", "geometry"));
54 for (std::size_t i = 0; i < geometry.fragments.size(); ++i) {
55 const auto& b = asset.bones[i];
56 const auto& f = geometry.fragments[i];
57 for (std::size_t v = 0; v < f.positions.size(); v += 3)
58 if (std::abs(f.positions[v]) > b.halfExtentX + 1e-5f ||
59 std::abs(f.positions[v + 1]) > b.halfExtentY + 1e-5f ||
60 std::abs(f.positions[v + 2]) > b.halfExtentZ + 1e-5f)
61 return eve::Result<std::unique_ptr<GeometryCollectionInstance>>::failure(eve::Diagnostic::error(
62 eve::DiagnosticCode::InvalidArgument, "fragment vertices exceed declared bounds", "geometry"));
63 }
64 return createInternal(world, asset, std::make_shared<const FractureGeometry>(geometry), x, y, z);
65}
66eve::Result<std::unique_ptr<GeometryCollectionInstance>> GeometryCollectionInstance::createInternal(
67 World3D& world, const GeometryCollectionAsset& asset, std::shared_ptr<const FractureGeometry> geometry,
68 float originX, float originY, float originZ) {
69 auto valid = asset.validate();
70 if (!valid) return eve::Result<std::unique_ptr<GeometryCollectionInstance>>::failure(valid.status());
71 if (!std::isfinite(originX) || !std::isfinite(originY) || !std::isfinite(originZ))
72 return eve::Result<std::unique_ptr<GeometryCollectionInstance>>::failure(
73 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument, "instance origin must be finite", "origin"));
74 if (!world.isValid())
75 return eve::Result<std::unique_ptr<GeometryCollectionInstance>>::failure(eve::Diagnostic::error(
76 eve::DiagnosticCode::StaleHandle, "geometry-collection create requires a live World3D", "world"));
77
78 auto instance = std::unique_ptr<GeometryCollectionInstance>(
79 new GeometryCollectionInstance(world.runtimeHandle(), originX, originY, originZ));
80 instance->geometry_ = std::move(geometry);
81 instance->bones_.reserve(asset.bones.size());
82 instance->edges_.reserve(asset.edges.size());
83
84 std::vector<Body3D*> createdBodies;
85 createdBodies.reserve(asset.bones.size());
86 auto rollback = [&]() {
87 for (Body3D* body : createdBodies) {
88 if (body && body->isValid()) body->destroy();
89 }
90 createdBodies.clear();
91 instance->bones_.clear();
92 };
93
94 for (const auto& boneDef : asset.bones) {
95 const float x = originX + boneDef.localX;
96 const float y = originY + boneDef.localY;
97 const float z = originZ + boneDef.localZ;
98 // Intact Attached bones stay static so connection-graph topology (not
99 // independent dynamic bodies) keeps the collection from falling apart
100 // before any edge breaks. Detach sync promotes free islands to dynamic.
101 Body3D* body = world.newBody("static", x, y, z);
102 if (!body || !body->isValid()) {
103 rollback();
104 return eve::Result<std::unique_ptr<GeometryCollectionInstance>>::failure(eve::Diagnostic::error(
105 eve::DiagnosticCode::Failed, "geometry-collection failed to create a bone body", "body"));
106 }
107 try {
108 if (instance->geometry_) {
109 const auto& fragment = instance->geometry_->fragments[instance->bones_.size()];
110 body->newConvexHullShape(fragment.positions, 254, boneDef.density, boneDef.friction,
111 boneDef.restitution);
112 } else {
113 body->newBoxShape(boneDef.halfExtentX * 2.f, boneDef.halfExtentY * 2.f, boneDef.halfExtentZ * 2.f,
114 boneDef.density, boneDef.friction, boneDef.restitution);
115 }
116 } catch (const std::exception& error) {
117 body->destroy();
118 rollback();
119 return eve::Result<std::unique_ptr<GeometryCollectionInstance>>::failure(
120 eve::Diagnostic::error(eve::DiagnosticCode::Failed, error.what(), "fragmentCollider"));
121 }
122 auto link = PhysicsLink::fromBody(*body);
123 if (!link) {
124 rollback();
125 if (body->isValid()) body->destroy();
126 return eve::Result<std::unique_ptr<GeometryCollectionInstance>>::failure(link.status());
127 }
128 BoneRuntime bone;
129 bone.state = BoneRuntimeState::Attached;
130 bone.anchored = boneDef.anchoredDefault;
131 bone.link = link.value();
132 bone.halfExtentX = boneDef.halfExtentX;
133 bone.halfExtentY = boneDef.halfExtentY;
134 bone.halfExtentZ = boneDef.halfExtentZ;
135 bone.clusterId = boneDef.clusterId;
136 bone.fractureLevel = boneDef.fractureLevel;
137 instance->bones_.push_back(bone);
138 createdBodies.push_back(body);
139 }
140
141 for (const auto& edgeDef : asset.edges) {
142 EdgeRuntime edge;
143 edge.boneA = edgeDef.boneA;
144 edge.boneB = edgeDef.boneB;
145 edge.strainThreshold = edgeDef.strainThreshold;
146 instance->edges_.push_back(edge);
147 }
148
149 if (instance->geometry_) {
150 auto constraints = instance->stageConstraints(instance->edges_);
151 if (!constraints) {
152 rollback();
153 return eve::Result<std::unique_ptr<GeometryCollectionInstance>>::failure(constraints.status());
154 }
155 instance->ownedJoints_ = std::move(constraints).takeValue();
156 for (std::size_t i = 0; i < instance->edges_.size(); ++i)
157 instance->edges_[i].joint =
158 instance->ownedJoints_[i] ? instance->ownedJoints_[i]->runtimeHandle() : PhysicsJointHandle::invalid();
159 }
160 return eve::Result<std::unique_ptr<GeometryCollectionInstance>>::success(std::move(instance));
161}

182float GeometryCollectionInstance::falloffAt(const DestructionField& field, float x, float y, float z) const {
183 const float dx = x - field.centerX;
184 const float dy = y - field.centerY;
185 const float dz = z - field.centerZ;
186 const float dist = std::sqrt(dx * dx + dy * dy + dz * dz);
187 if (!(field.radius > 0.f) || !std::isfinite(field.radius)) return 0.f;
188 if (dist > field.radius) return 0.f;
189 if (field.falloff == DestructionFieldFalloff::None) return 1.f;
190 return 1.f - dist / field.radius;
191}

193eve::Result<FieldApplicationReceipt> GeometryCollectionInstance::applyField(const DestructionField& field) {
194 if (!std::isfinite(field.centerX) || !std::isfinite(field.centerY) || !std::isfinite(field.centerZ) ||
195 !std::isfinite(field.radius) || field.radius < 0.f || !std::isfinite(field.magnitude))
196 return eve::Result<FieldApplicationReceipt>::failure(eve::Diagnostic::error(
197 eve::DiagnosticCode::InvalidArgument, "destruction field parameters must be finite", "field"));
198 if (field.kind == DestructionFieldKind::Impulse &&
199 (!std::isfinite(field.dirX) || !std::isfinite(field.dirY) || !std::isfinite(field.dirZ)))
200 return eve::Result<FieldApplicationReceipt>::failure(eve::Diagnostic::error(
201 eve::DiagnosticCode::InvalidArgument, "impulse field direction must be finite", "field"));
202 World3D* world = resolveWorld();
203 if (!world) {
204 orphanedPhysics_ = true;
205 return eve::Result<FieldApplicationReceipt>::failure(eve::Diagnostic::error(
206 eve::DiagnosticCode::StaleHandle, "geometry-collection world is gone", "world"));
207 }
208
209 FieldApplicationReceipt receipt;
210 int sleepsThisCall = 0;
211 bool needGraphSync = false;
212 for (std::size_t i = 0; i < bones_.size(); ++i) {
213 auto bodyResult = bones_[i].link.resolve(*world);
214 if (!bodyResult) continue;
215 Body3D* body = bodyResult.value();
216 const float weight = falloffAt(field, body->getX(), body->getY(), body->getZ());
217 if (weight <= 0.f) continue;
218 switch (field.kind) {
219 case DestructionFieldKind::Anchor: {
220 const bool wasSleeping = bones_[i].state == BoneRuntimeState::Sleeping;
221 bones_[i].anchored = true;
222 bones_[i].state = BoneRuntimeState::Attached;
223 body->setType("static");
224 if (wasSleeping) ++sleepBatchRevision_;
225 ++receipt.bonesAffected;
226 needGraphSync = true;
227 break;
228 }
229 case DestructionFieldKind::Impulse: {
230 // Anchored foundations and still-Attached graph members stay put.
231 if (bones_[i].anchored || bones_[i].state == BoneRuntimeState::Attached) break;
232 float dx = field.dirX, dy = field.dirY, dz = field.dirZ;
233 const float len = std::sqrt(dx * dx + dy * dy + dz * dz);
234 if (len <= 1e-6f) {
235 dx = 0.f;
236 dy = 1.f;
237 dz = 0.f;
238 } else {
239 dx /= len;
240 dy /= len;
241 dz /= len;
242 }
243 const float impulse = field.magnitude * weight;
244 const bool wasSleeping = bones_[i].state == BoneRuntimeState::Sleeping;
245 body->setType("dynamic");
246 body->applyLinearImpulse(dx * impulse, dy * impulse, dz * impulse);
247 bones_[i].state = BoneRuntimeState::Detached;
248 if (wasSleeping) ++sleepBatchRevision_;
249 ++receipt.bonesAffected;
250 break;
251 }
252 case DestructionFieldKind::Sleep: {
253 if (bones_[i].state != BoneRuntimeState::Detached) break;
254 const float vx = body->getLinearVelocityX();
255 const float vy = body->getLinearVelocityY();
256 const float vz = body->getLinearVelocityZ();
257 const float speed = std::sqrt(vx * vx + vy * vy + vz * vz);
258 if (speed > kSleepSpeed) break;
259 if (budget_.maxSleepsPerStep > 0 && sleepsThisCall >= budget_.maxSleepsPerStep) {
260 ++receipt.sleepsDeferred;
261 break;
262 }
263 body->setType("static");
264 if (bones_[i].state != BoneRuntimeState::Sleeping) ++sleepBatchRevision_;
265 bones_[i].state = BoneRuntimeState::Sleeping;
266 ++receipt.bonesAffected;
267 ++sleepsThisCall;
268 break;
269 }
270 case DestructionFieldKind::Strain:
271 break;
272 }
273 }
274 if (needGraphSync) syncBoneActivationFromGraph();
275
276 if (field.kind == DestructionFieldKind::Strain) {
277 for (auto& edge : edges_) {
278 if (edge.broken) continue;
279 if (edge.boneA < 0 || edge.boneB < 0 || edge.boneA >= static_cast<int>(bones_.size()) ||
280 edge.boneB >= static_cast<int>(bones_.size()))
281 continue;
282 auto bodyA = bones_[static_cast<std::size_t>(edge.boneA)].link.resolve(*world);
283 auto bodyB = bones_[static_cast<std::size_t>(edge.boneB)].link.resolve(*world);
284 if (!bodyA || !bodyB) continue;
285 const float midX = 0.5f * (bodyA.value()->getX() + bodyB.value()->getX());
286 const float midY = 0.5f * (bodyA.value()->getY() + bodyB.value()->getY());
287 const float midZ = 0.5f * (bodyA.value()->getZ() + bodyB.value()->getZ());
288 const float weight = falloffAt(field, midX, midY, midZ);
289 if (weight <= 0.f) continue;
290 edge.accumulatedStrain += field.magnitude * weight;
291 ++receipt.edgesAffected;
292 }
293 }
294 return eve::Result<FieldApplicationReceipt>::success(receipt);
295}

297void GeometryCollectionInstance::breakEdge(int edgeIndex, std::uint64_t tick, DestructionStepReceipt& receipt) {
298 auto& edge = edges_[static_cast<std::size_t>(edgeIndex)];
299 if (edge.broken) return;
300 edge.broken = true;
301 if (static_cast<std::size_t>(edgeIndex) < ownedJoints_.size())
302 ownedJoints_[static_cast<std::size_t>(edgeIndex)].reset();
303 edge.joint = PhysicsJointHandle::invalid();
304 ++receipt.edgesBroken;
305
306 BoneDetachEvent event;
307 event.boneA = edge.boneA;
308 event.boneB = edge.boneB;
309 event.tick = static_cast<std::int64_t>(tick);
310 detachEvents_.push_back(event);
311
312 if (edge.boneA >= 0 && edge.boneB >= 0 && edge.boneA < static_cast<int>(bones_.size()) &&
313 edge.boneB < static_cast<int>(bones_.size())) {
314 const int clusterA = bones_[static_cast<std::size_t>(edge.boneA)].clusterId;
315 const int clusterB = bones_[static_cast<std::size_t>(edge.boneB)].clusterId;
316 if (clusterA != clusterB) {
317 ClusterBreakEvent clusterEvent;
318 clusterEvent.clusterA = clusterA;
319 clusterEvent.clusterB = clusterB;
320 clusterEvent.boneA = edge.boneA;
321 clusterEvent.boneB = edge.boneB;
322 clusterEvent.tick = static_cast<std::int64_t>(tick);
323 clusterBreakEvents_.push_back(clusterEvent);
324 ++receipt.clusterBreaks;
325 }
326 }

329void GeometryCollectionInstance::syncBoneActivationFromGraph() {
330 const int boneCount = static_cast<int>(bones_.size());
331 if (boneCount == 0) return;
332
333 std::vector<std::vector<int>> adjacency(static_cast<std::size_t>(boneCount));
334 for (const auto& edge : edges_) {
335 if (edge.broken) continue;
336 if (edge.boneA < 0 || edge.boneB < 0 || edge.boneA >= boneCount || edge.boneB >= boneCount) continue;
337 adjacency[static_cast<std::size_t>(edge.boneA)].push_back(edge.boneB);
338 adjacency[static_cast<std::size_t>(edge.boneB)].push_back(edge.boneA);
339 }
340
341 std::vector<int> component(static_cast<std::size_t>(boneCount), -1);
342 int componentCount = 0;
343 bool anyAnchored = false;
344 for (int i = 0; i < boneCount; ++i) {
345 if (bones_[static_cast<std::size_t>(i)].anchored) anyAnchored = true;
346 if (component[static_cast<std::size_t>(i)] >= 0) continue;
347 std::queue<int> queue;
348 queue.push(i);
349 component[static_cast<std::size_t>(i)] = componentCount;
350 while (!queue.empty()) {
351 const int cur = queue.front();
352 queue.pop();
353 for (int next : adjacency[static_cast<std::size_t>(cur)]) {
354 if (component[static_cast<std::size_t>(next)] >= 0) continue;
355 component[static_cast<std::size_t>(next)] = componentCount;
356 queue.push(next);
357 }
358 }
359 ++componentCount;
360 }
361
362 std::vector<bool> componentHasAnchor(static_cast<std::size_t>(componentCount), false);
363 for (int i = 0; i < boneCount; ++i) {
364 if (bones_[static_cast<std::size_t>(i)].anchored)
365 componentHasAnchor[static_cast<std::size_t>(component[static_cast<std::size_t>(i)])] = true;
366 }
367
368 World3D* world = resolveWorld();
369 for (int i = 0; i < boneCount; ++i) {
370 auto& bone = bones_[static_cast<std::size_t>(i)];
371 if (bone.state == BoneRuntimeState::Sleeping) continue; // Sleep is sticky until Impulse/Anchor
372 const bool stayAttached =
373 bone.anchored ||
374 componentHasAnchor[static_cast<std::size_t>(component[static_cast<std::size_t>(i)])] ||
375 (!anyAnchored && componentCount == 1);
376 const BoneRuntimeState desired =
377 stayAttached ? BoneRuntimeState::Attached : BoneRuntimeState::Detached;
378 if (bone.state == desired) {
379 // Ensure Attached bones remain static even if a prior path made them dynamic.
380 if (desired == BoneRuntimeState::Attached && world) {
381 auto body = bone.link.resolve(*world);
382 if (body) setBoneBodyType(*body.value(), desired);
383 }
384 continue;
385 }
386 bone.state = desired;
387 if (!world) continue;
388 auto body = bone.link.resolve(*world);
389 if (body) setBoneBodyType(*body.value(), desired);
390 }
391}

393eve::Result<DestructionStepReceipt> GeometryCollectionInstance::step(SimulationStep step) {
394 DestructionStepReceipt receipt;
395 if (step.delta.nanoseconds() < 0)
396 return eve::Result<DestructionStepReceipt>::failure(eve::Diagnostic::error(
397 eve::DiagnosticCode::InvalidArgument, "geometry-collection step delta must be non-negative", "step"));
398 const std::uint64_t tick = step.tick.value();
399 if (lastTick_ != 0 && tick <= lastTick_)
400 return eve::Result<DestructionStepReceipt>::failure(eve::Diagnostic::error(
401 eve::DiagnosticCode::InvalidArgument, "geometry-collection step tick must advance", "step"));
402 World3D* world = resolveWorld();
403 if (!world) {
404 orphanedPhysics_ = true;
405 return eve::Result<DestructionStepReceipt>::failure(eve::Diagnostic::error(
406 eve::DiagnosticCode::StaleHandle, "geometry-collection world is gone", "world"));
407 }
408 orphanedPhysics_ = false;
409 detachEvents_.clear();
410 clusterBreakEvents_.clear();
411
412 std::vector<int> toBreak = pendingEdgeBreaks_;
413 pendingEdgeBreaks_.clear();
414 for (std::size_t i = 0; i < edges_.size(); ++i) {
415 const auto& edge = edges_[i];
416 if (edge.broken) continue;
417 if (edge.accumulatedStrain >= edge.strainThreshold) toBreak.push_back(static_cast<int>(i));
418 }
419 std::sort(toBreak.begin(), toBreak.end());
420 toBreak.erase(std::unique(toBreak.begin(), toBreak.end()), toBreak.end());
421
422 const int budget =
423 budget_.maxEdgeBreaksPerStep > 0 ? budget_.maxEdgeBreaksPerStep : static_cast<int>(toBreak.size());
424 int brokenThisStep = 0;
425 for (int edgeIndex : toBreak) {
426 if (edgeIndex < 0 || edgeIndex >= static_cast<int>(edges_.size())) continue;
427 if (edges_[static_cast<std::size_t>(edgeIndex)].broken) continue;
428 if (brokenThisStep >= budget) {
429 pendingEdgeBreaks_.push_back(edgeIndex);
430 ++receipt.edgesDeferred;
431 continue;
432 }
433 breakEdge(edgeIndex, tick, receipt);
434 ++brokenThisStep;
435 }
436 if (receipt.edgesBroken > 0) syncBoneActivationFromGraph();
437 lastTick_ = tick;
438 return eve::Result<DestructionStepReceipt>::success(receipt);
439}

546eve::Result<void> GeometryCollectionInstance::restoreSnapshot(const GeometryCollectionInstanceSnapshot& snapshot) {
547 auto valid = snapshot.validate();
548 if (!valid) return eve::Result<void>::failure(valid.status());
549 if (snapshot.bones.size() != bones_.size() || snapshot.edges.size() != edges_.size())
550 return eve::Result<void>::failure(eve::Diagnostic::error(
551 eve::DiagnosticCode::InvalidArgument,
552 "instance-snapshot bone/edge counts must match the live instance", "snapshot"));
553 World3D* world = resolveWorld();
554 if (!world) {
555 orphanedPhysics_ = true;
556 return eve::Result<void>::failure(eve::Diagnostic::error(
557 eve::DiagnosticCode::StaleHandle, "geometry-collection world is gone", "world"));
558 }
559
560 // Stage + resolve every body before mutating World3D so failures leave both
561 // the instance topology and body types unchanged.
562 std::vector<BoneRuntime> stagedBones = bones_;
563 std::vector<EdgeRuntime> stagedEdges = edges_;
564 std::vector<Body3D*> bodies(snapshot.bones.size(), nullptr);
565 for (std::size_t i = 0; i < snapshot.bones.size(); ++i) {
566 const auto& src = snapshot.bones[i];
567 if (geometry_ && (src.halfExtentX != bones_[i].halfExtentX || src.halfExtentY != bones_[i].halfExtentY ||
568 src.halfExtentZ != bones_[i].halfExtentZ || src.clusterId != bones_[i].clusterId ||
569 src.fractureLevel != bones_[i].fractureLevel))
570 return eve::Result<void>::failure(
571 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
572 "snapshot cannot replace immutable fragment metadata", "snapshot"));
573 stagedBones[i].state = static_cast<BoneRuntimeState>(src.state);
574 stagedBones[i].anchored = src.anchored;
575 stagedBones[i].clusterId = src.clusterId;
576 stagedBones[i].fractureLevel = src.fractureLevel;
577 stagedBones[i].halfExtentX = src.halfExtentX;
578 stagedBones[i].halfExtentY = src.halfExtentY;
579 stagedBones[i].halfExtentZ = src.halfExtentZ;
580 auto body = stagedBones[i].link.resolve(*world);
581 if (!body) return eve::Result<void>::failure(body.status());
582 bodies[i] = body.value();
583 }
584 for (std::size_t i = 0; i < snapshot.edges.size(); ++i) {
585 const auto& src = snapshot.edges[i];
586 if (src.boneA != stagedEdges[i].boneA || src.boneB != stagedEdges[i].boneB)
587 return eve::Result<void>::failure(eve::Diagnostic::error(
588 eve::DiagnosticCode::InvalidArgument, "instance-snapshot edge topology mismatch", "edges"));
589 stagedEdges[i].strainThreshold = src.strainThreshold;
590 stagedEdges[i].accumulatedStrain = src.accumulatedStrain;
591 stagedEdges[i].broken = src.broken;
592 }
593
594 if (geometry_) {
595 auto stagedJoints = stageConstraints(stagedEdges);
596 if (!stagedJoints) return eve::Result<void>::failure(stagedJoints.status());
597 releaseConstraints();
598 ownedJoints_ = std::move(stagedJoints).takeValue();
599 for (std::size_t i = 0; i < stagedEdges.size(); ++i)
600 stagedEdges[i].joint = ownedJoints_[i] ? ownedJoints_[i]->runtimeHandle() : PhysicsJointHandle::invalid();
601 }
602 for (std::size_t i = 0; i < stagedBones.size(); ++i) {
603 Body3D* body = bodies[i];
604 switch (stagedBones[i].state) {
605 case BoneRuntimeState::Attached:
606 body->setType(stagedBones[i].anchored ? "static" : "dynamic");
607 break;
608 case BoneRuntimeState::Detached:
609 body->setType("dynamic");
610 body->setAwake(true);
611 break;
612 case BoneRuntimeState::Sleeping:
613 body->setType("static");
614 break;
615 }
616 }
617
618 bones_ = std::move(stagedBones);
619 edges_ = std::move(stagedEdges);
620 originX_ = snapshot.originX;
621 originY_ = snapshot.originY;
622 originZ_ = snapshot.originZ;
623 lastTick_ = static_cast<std::uint64_t>(snapshot.lastTick);
624 sleepBatchRevision_ = snapshot.sleepBatchRevision;
625 detachEvents_.clear();
626 clusterBreakEvents_.clear();
627 pendingEdgeBreaks_.clear();
628 orphanedPhysics_ = false;
629 return eve::Result<void>::success();
630}
