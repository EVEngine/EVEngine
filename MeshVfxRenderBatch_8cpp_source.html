29MeshVfxRenderQueue MeshVfxRenderBatchPlanner::build(std::span<const MeshVfxRenderItem> items,
30 std::span<const MeshVfxLodDecision> lodDecisions,
31 std::uint64_t frameIndex) const {
32 MeshVfxRenderQueue queue;
33 queue.stats.inputItems = static_cast<std::uint32_t>(items.size());
34
35 std::unordered_map<std::uint64_t, const MeshVfxLodDecision*> decisions;
36 decisions.reserve(lodDecisions.size());
37 for (const auto& decision : lodDecisions) {
38 decisions.try_emplace(decision.stableInstanceId, &decision);
39 }
40
41 std::vector<PlannedDraw> additive;
42 std::vector<PlannedDraw> alpha;
43 additive.reserve(items.size());
44 alpha.reserve(items.size());
45 for (const auto& item : items) {
46 const auto found = decisions.find(item.stableInstanceId);
47 if (found == decisions.end()) {
48 ++queue.stats.missingDecisions;
49 continue;
50 }
51 const auto& lod = *found->second;
52 if (lod.tier == MeshVfxLodTier::Culled) {
53 ++queue.stats.culledDraws;
54 continue;
55 }
56
57 const auto interval = std::max(1u, lod.meshUpdateInterval);
58 const auto refresh = frameIndex % interval == item.stableInstanceId % interval;
59 PlannedDraw planned{
60 item.key,
61 std::isfinite(item.viewDepth) ? item.viewDepth : 0.0f,
62 {item.stableInstanceId, lod.tier, refresh ? MeshVfxMeshUpdate::Refresh : MeshVfxMeshUpdate::Reuse,
63 std::max(1u, lod.trailSampleStride)}};
64 if (refresh) {
65 ++queue.stats.dynamicMeshRefreshes;
66 }
67 (item.key.blend == MeshVfxBatchBlend::Additive ? additive : alpha).push_back(std::move(planned));
68 }
69
70 std::sort(additive.begin(), additive.end(), [](const PlannedDraw& lhs, const PlannedDraw& rhs) {
71 if (lhs.key != rhs.key) {
72 return keyLess(lhs.key, rhs.key);
73 }
74 return lhs.draw.stableInstanceId < rhs.draw.stableInstanceId;
75 });
76 std::sort(alpha.begin(), alpha.end(), [](const PlannedDraw& lhs, const PlannedDraw& rhs) {
77 if (lhs.viewDepth != rhs.viewDepth) {
78 return lhs.viewDepth > rhs.viewDepth;
79 }
80 if (lhs.key != rhs.key) {
81 return keyLess(lhs.key, rhs.key);
82 }
83 return lhs.draw.stableInstanceId < rhs.draw.stableInstanceId;
84 });
85
86 queue.batches.reserve(additive.size() + alpha.size());
87 for (auto& planned : additive) {
88 appendBatch(queue.batches, std::move(planned));
89 ++queue.stats.queuedDraws;
90 }
91 for (auto& planned : alpha) {
92 appendBatch(queue.batches, std::move(planned));
93 ++queue.stats.queuedDraws;
94 }
95 return queue;
96}

104MeshVfxLodTier requestedTier(const MeshVfxLodCandidate& candidate, const MeshVfxScalabilityPolicy& policy) {
105 if (!candidate.visible || !std::isfinite(candidate.distance) || candidate.distance < 0.0f ||
106 candidate.distance > policy.maximumDistance || !std::isfinite(candidate.projectedRadiusPixels)) {
107 return MeshVfxLodTier::Culled;
108 }
109 if (candidate.projectedRadiusPixels >= policy.fullQualityMinPixels) {
110 return MeshVfxLodTier::Full;
111 }
112 if (candidate.projectedRadiusPixels >= policy.reducedQualityMinPixels) {
113 return MeshVfxLodTier::Reduced;
114 }
115 return MeshVfxLodTier::Minimal;
116}

118MeshVfxLodDecision makeDecision(std::uint64_t id, MeshVfxLodTier tier, const MeshVfxScalabilityPolicy& policy) {
119 MeshVfxLodDecision decision;
120 decision.stableInstanceId = id;
121 decision.tier = tier;
122 switch (tier) {
123 case MeshVfxLodTier::Full: decision.workUnits = policy.fullWorkUnits; break;
124 case MeshVfxLodTier::Reduced:
125 decision.workUnits = policy.reducedWorkUnits;
126 decision.trailSampleStride = std::max(1u, policy.reducedTrailSampleStride);
127 decision.meshUpdateInterval = std::max(1u, policy.reducedMeshUpdateInterval);
128 break;
129 case MeshVfxLodTier::Minimal:
130 decision.workUnits = policy.minimalWorkUnits;
131 decision.trailSampleStride = std::max(1u, policy.minimalTrailSampleStride);
132 decision.meshUpdateInterval = std::max(1u, policy.minimalMeshUpdateInterval);
133 break;
134 case MeshVfxLodTier::Culled: break;
135 }
136 return decision;
137}

139MeshVfxLodTier degrade(MeshVfxLodTier tier) {
140 switch (tier) {
141 case MeshVfxLodTier::Full: return MeshVfxLodTier::Reduced;
142 case MeshVfxLodTier::Reduced: return MeshVfxLodTier::Minimal;
143 case MeshVfxLodTier::Minimal:
144 case MeshVfxLodTier::Culled: return MeshVfxLodTier::Culled;
145 }
146 return MeshVfxLodTier::Culled;
147}

153std::vector<MeshVfxLodDecision> MeshVfxScalabilityPlanner::plan(std::span<const MeshVfxLodCandidate> candidates) const {
154 std::vector<MeshVfxLodDecision> decisions;
155 decisions.reserve(candidates.size());
156 for (const auto& candidate : candidates) {
157 decisions.push_back(makeDecision(candidate.stableInstanceId, MeshVfxLodTier::Culled, policy_));
158 }
159
160 std::vector<std::size_t> order(candidates.size());
161 std::iota(order.begin(), order.end(), 0);
162 std::stable_sort(order.begin(), order.end(), [&](std::size_t lhs, std::size_t rhs) {
163 const auto& a = candidates[lhs];
164 const auto& b = candidates[rhs];
165 if (a.priority != b.priority) {
166 return a.priority > b.priority;
167 }
168 if (a.distance != b.distance) {
169 return a.distance < b.distance;
170 }
171 return a.stableInstanceId < b.stableInstanceId;
172 });
173
174 std::uint32_t remaining = policy_.workUnitBudget;
175 for (const auto index : order) {
176 auto tier = requestedTier(candidates[index], policy_);
177 auto decision = makeDecision(candidates[index].stableInstanceId, tier, policy_);
178 while (decision.workUnits > remaining && tier != MeshVfxLodTier::Culled) {
179 tier = degrade(tier);
180 decision = makeDecision(candidates[index].stableInstanceId, tier, policy_);
181 }
182 remaining -= decision.workUnits;
183 decisions[index] = decision;
184 }
185 return decisions;
186}
