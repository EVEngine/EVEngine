11vk::Pipeline createSolidColorPipeline(vkb::Device &device, const vkb::BuiltRenderPass &renderPass,
12 vk::PipelineLayout layout,
13 BlendMode mode) {
14 if (mode == BlendMode::Additive || mode == BlendMode::Premultiplied ||
15 mode == BlendMode::Multiply) {
16 // cbs/attachments must outlive build(); the builder must stay a single
17 // expression — copying the builder into a named local leaves its
18 // shader-stage pName pointers dangling into the temporary's storage.
19 std::vector<vk::PipelineColorBlendAttachmentState> attachments(1,
20 makeBlendAttachment(mode));
21 vk::PipelineColorBlendStateCreateInfo cbs{};
22 cbs.logicOpEnable = false;
23 cbs.attachmentCount = 1;
24 cbs.pAttachments = attachments.data();
25 return device.createPipeline()
26 .useClassicPipeline(embeddedSpirv(color_vert_spv), embeddedSpirv(color_frag_spv))
27 .setPipelineLayout(layout)
28 .setVertexInputState(vkb::VertexInputStateBuilder()
29 .addInputBinding<ColorVertex>()
30 .addAttributeDescription<ColorVertex>())
31 .setDynamicStatesViewportScissor()
32 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
33 vk::CullModeFlagBits::eNone, vk::FrontFace::eCounterClockwise)
34 .setColorBlending(cbs)
35 .build(renderPass);
36 }
37 if (mode == BlendMode::Alpha) {
38 return device.createPipeline()
39 .useClassicPipeline(embeddedSpirv(color_vert_spv), embeddedSpirv(color_frag_spv))
40 .setPipelineLayout(layout)
41 .setVertexInputState(vkb::VertexInputStateBuilder()
42 .addInputBinding<ColorVertex>()
43 .addAttributeDescription<ColorVertex>())
44 .setDynamicStatesViewportScissor()
45 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f,
46 vk::CullModeFlagBits::eNone, vk::FrontFace::eCounterClockwise)
47 .setAlphaBlending(1)
48 .build(renderPass);
49 }
50 // Opaque: keep the original single-expression chain (no explicit blend state).
51 return device.createPipeline()
52 .useClassicPipeline(embeddedSpirv(color_vert_spv), embeddedSpirv(color_frag_spv))
53 .setPipelineLayout(layout)
54 .setVertexInputState(vkb::VertexInputStateBuilder()
55 .addInputBinding<ColorVertex>()
56 .addAttributeDescription<ColorVertex>())
57 .setDynamicStatesViewportScissor()
58 .setRasterizer(vk::PolygonMode::eFill, false, false, 1.0f, vk::CullModeFlagBits::eNone,
59 vk::FrontFace::eCounterClockwise)
60 .build(renderPass);
61}

65vkb::BuiltRenderPass persistentColorPass(vkb::Device& device, vk::Format format) {
66 vk::AttachmentDescription color{};
67 color.format = format;
68 color.samples = vk::SampleCountFlagBits::e1;
69 color.loadOp = vk::AttachmentLoadOp::eLoad;
70 color.storeOp = vk::AttachmentStoreOp::eStore;
71 color.stencilLoadOp = vk::AttachmentLoadOp::eDontCare;
72 color.stencilStoreOp = vk::AttachmentStoreOp::eDontCare;
73 color.initialLayout = vk::ImageLayout::eColorAttachmentOptimal;
74 color.finalLayout = vk::ImageLayout::eShaderReadOnlyOptimal;
75 auto pass = device.createRenderPass().addAttachment(color)
76 .addSubpass(vkb::SubpassBuilder().addAttachmentRef(0, vk::ImageLayout::eColorAttachmentOptimal))
77 .addDependency(VK_SUBPASS_EXTERNAL, 0,
78 vk::PipelineStageFlagBits::eFragmentShader | vk::PipelineStageFlagBits::eColorAttachmentOutput,
79 vk::PipelineStageFlagBits::eColorAttachmentOutput,
80 vk::AccessFlagBits::eShaderRead | vk::AccessFlagBits::eColorAttachmentWrite,
81 vk::AccessFlagBits::eColorAttachmentRead | vk::AccessFlagBits::eColorAttachmentWrite)
82 .addDependency(0, VK_SUBPASS_EXTERNAL, vk::PipelineStageFlagBits::eColorAttachmentOutput,
83 vk::PipelineStageFlagBits::eFragmentShader, vk::AccessFlagBits::eColorAttachmentWrite,
84 vk::AccessFlagBits::eShaderRead).build();
85 // addAttachment preserves the explicit initial layout; supply its role metadata.
86 pass.colorAttachmentCount = 1;
87 pass.colorsSampledAfter = true;
88 pass.colorFinalLayout = vk::ImageLayout::eShaderReadOnlyOptimal;
89 return pass;
90}

135void Graphics::ensureShaderOffscreenPipeline(Shader *shader) {
136 if (!shader || !shader->gpuHandle || !offscreenRenderPass) return;
137 // Mesh3D SPIR-V / layout is incompatible with the 2D textured offscreen pass.
138 if (shader->getKind() == Shader::Kind::eMesh3D) return;
139 auto *gpu = static_cast<GpuShader *>(shader->gpuHandle);
140 if (gpu->isMesh3D || gpu->offscreenPipeline) return;
141 gpu->offscreenPipeline = createTexturedStylePipeline(shader->vertexSpirv(), shader->fragmentSpirv(),
142 offscreenRenderPass, shaderPipelineLayout);
143 gpu->offscreenOpaquePipeline = createTexturedStylePipeline(
144 shader->vertexSpirv(), shader->fragmentSpirv(), offscreenRenderPass,
145 shaderPipelineLayout, BlendMode::Opaque);
146}
