43 bool merge(Type& into, const Type& from) {
44 if (from.name.empty() || from.name == "null") return false;
45 auto before = into;
46 if (into.name.empty() || into.name == "null")
47 into.name = from.name;
48 else if (into.name != from.name)
49 into.name = "dynamic";
50 into.values.insert(from.values.begin(), from.values.end());
51 into.identities.insert(from.identities.begin(), from.identities.end());
52 into.hosts.insert(from.hosts.begin(), from.hosts.end());
53 if (into.values.size() > 8 || into.identities.size() > 8 || into.hosts.size() > 8) {
54 into.name = "dynamic";
55 into.values.clear();
56 into.identities.clear();
57 into.hosts.clear();
58 }
59 return before.name != into.name || before.values != into.values || before.identities != into.identities ||
60 before.hosts != into.hosts;

62 const Contract* contract(const Expression& expression, const std::string& file, const std::string& scope,
63 unsigned depth = 0) const {
64 if (depth >= 64 || expression.kind != "call" || expression.operands.empty()) return nullptr;
65 const auto& callee = expression.operands[0];
66 if (callee.kind != "member" || callee.operands.empty()) return nullptr;
67 auto owner = type(callee.operands[0], file, scope, depth + 1);
68 if (owner.name.starts_with("module:")) return registry.find(owner.name.substr(7) + "/" + callee.value);
69 auto found = methods.find(owner.name + '\0' + callee.value);
70 return found == methods.end() ? nullptr : found->second;
71 }
72 Type type(const Expression& expression, const std::string& file, const std::string& scope,
73 unsigned depth = 0) const {
74 if (depth >= 64) return {};
75 if (expression.kind == "string") return {"string", {expression.value}, {}, {}};
76 if (expression.kind == "number") return {"number", {}, {}, {}};
77 if (expression.kind == "identifier") {
78 if (expression.value == "null") return {"null", {}, {}, {}};
79 return variable(file, scope, expression.value);
80 }
81 if (expression.kind != "call" || expression.operands.empty()) return {};
82 const auto& callee = expression.operands[0];
83 if (callee.kind == "member" && !callee.operands.empty() && callee.operands[0].kind == "identifier" &&
84 callee.operands[0].value == "eve" && !variables.contains(localKey(file, scope, "eve")) &&
85 !variables.contains("global:eve")) {
86 auto factory = factories.find(callee.value);
87 if (factory != factories.end() && expression.operands.size() == 1)
88 return {"module:" + factory->second, {}, {}, {}};
89 }
90 const auto* api = contract(expression, file, scope, depth + 1);
91 if (!api) return {};
92 Type value;
93 value.name =
94 api->semantics && !api->semantics->resultClass.empty() ? api->semantics->resultClass : api->returnType;
95 if (value.name == "dynamic") value.name.clear();
96 if (api->semantics && api->semantics->role == "scene_ref") {
97 value.identities = argument(expression, api->semantics->identityArgument, file, scope).values;
98 value.hosts = argument(expression, api->semantics->relatedArgument, file, scope).values;
99 }
100 return value;
101 }

130 if (!snapshot.nodes.contains(from) || !snapshot.nodes.contains(to) ||

132 snapshot.nodes.at(to)->getValue<std::string>("status") != "resolved")
133 return;
134 const auto key = from + '\0' + to + '\0' + relation;
135 if (auto found = edgeKeys.find(key); found != edgeKeys.end()) {

170 auto item = node("class:" + value.name, "script_class", source(path, value.line));
171 if (item->has("fields") || item->getValue<std::string>("status") == "ambiguous") {
172 item->set("status", "ambiguous");

184 if (binding.scope == value.name + ".") {

187 field->set("type", binding.declaredType.empty() ? "dynamic" : binding.declaredType);
188 field->set("initializerKind", binding.expression.kind);
189 fields->add(field);

191 auto indexed = node(fieldId, "script_field", source(path, binding.expression.line));

193 indexed->set("initializerKind", binding.expression.kind);
194 if (binding.expression.kind == "string" || binding.expression.kind == "number")
195 indexed->set("defaultValue", binding.expression.value);
196 edge("class:" + value.name, fieldId, "contains", source(path, binding.expression.line));
197 }
198 item->set("fields", fields);

201 functions[function.name].push_back({path, &function});
202 for (const auto& parameter : function.parameters)
203 variables[localKey(path, function.name, parameter.name)].name = parameter.type;
204 }
205 for (const auto& binding : file->structure.bindings) {
206 auto& value = variables[bindingKey(path, binding)];
207 if (!binding.declaredType.empty()) value.name = binding.declaredType;

211 bool changed = false;
212 for (const auto& [path, file] : snapshot.files) {
213 if (!file->error.empty()) continue;
214 for (const auto& binding : file->structure.bindings)
215 changed |=
216 merge(variables[bindingKey(path, binding)], type(binding.expression, path, binding.scope));
217 for (const auto& call : file->structure.calls) {
218 const auto& expression = call.expression;
219 if (expression.operands.empty() || expression.operands[0].kind != "identifier") continue;
220 auto function = functions.find(expression.operands[0].value);
221 if (function == functions.end() || function->second.size() != 1) continue;
222 const auto& target = function->second[0];
223 for (size_t p = 0; p < target.syntax->parameters.size() && p + 1 < expression.operands.size(); ++p)
224 changed |= merge(
225 variables[localKey(target.file, target.syntax->name, target.syntax->parameters[p].name)],
226 type(expression.operands[p + 1], path, call.scope));

228 }

275 void schemaObjects(const Poco::Dynamic::Var& value, const std::string& path, const std::string& pointer,
276 Object::Ptr schema, const std::string& apiKey) {
277 if (value.type() == typeid(Object::Ptr)) {
278 auto object = value.extract<Object::Ptr>();
279 const auto identity = schema->getValue<std::string>("identityField");
280 bool matches = object->has(identity) && object->get(identity).isString();
281 for (const auto& field : *schema->getObject("fields"))
282 matches &= object->has(field.first) &&
283 jsonType(object->get(field.first)) == field.second.convert<std::string>();
284 if (matches) {
285 auto id = schema->getValue<std::string>("kind") + ":" + object->getValue<std::string>(identity);
286 auto item = node(id, schema->getValue<std::string>("kind"), source(path, 0, pointer));
287 if (item->has("definition")) {
288 item->set("status", "ambiguous");
289 item->remove("definition");
290 Object::Ptr diagnostic = new Object();
291 diagnostic->set("code", "ambiguous_schema_identity");
292 diagnostic->set("id", id);
293 snapshot.diagnostics.push_back(diagnostic);
294 } else if (item->getValue<std::string>("status") != "ambiguous")
295 item->set("definition", object);
296 item->set("schemaContract", apiKey);
297 item->set("evidence", "engine-schema-shape");
298 }
299 for (const auto& field : *object) {
300 std::string escaped;
301 for (char c : field.first) escaped += c == '~' ? "~0" : c == '/' ? "~1" : std::string(1, c);
302 schemaObjects(field.second, path, pointer + "/" + escaped, schema, apiKey);
303 }
304 } else if (value.type() == typeid(Array::Ptr)) {
305 auto array = value.extract<Array::Ptr>();
306 for (size_t i = 0; i < array->size(); ++i)
307 schemaObjects(array->get(static_cast<unsigned>(i)), path, pointer + "/" + std::to_string(i), schema,
308 apiKey);
309 }

314 for (const auto& api : registry.snapshot()) {

316 version += api.key() + '\0' + api.toJson() + '\0';

325 std::map<std::string, std::set<std::string>> nativeByFunction, calledFunctions;
326 for (const auto& [path, file] : snapshot.files) {
327 if (!file->error.empty()) continue;
328 std::map<std::string, std::set<std::string>> filters, subjects;
329 for (const auto& call : file->structure.calls) {
330 const auto& expression = call.expression;
331 if (expression.operands.empty()) continue;
332 auto from = functionId(call.scope);
333 auto origin = source(path, expression.line);
334 const auto& callee = expression.operands[0];
335 if (callee.kind == "identifier") {
336 auto target = functionId(callee.value);
337 if (!from.empty() && !target.empty()) {
338 edge(from, target, "calls", origin);
339 calledFunctions[from].insert(target);
340 }
341 }
342 const auto* api = contract(expression, path, call.scope);
343 if (!api && !from.empty() && callee.kind == "member" &&
344 !type(expression, path, call.scope).name.starts_with("module:")) {
345 auto item = snapshot.nodes.at(from);
346 auto unknowns = item->getArray("unknownCalls");
347 if (!unknowns) {
348 unknowns = new Array();

350 }
351 if (unknowns->size() < 16) {

353 unknown->set("method", callee.value);
354 unknown->set("source", origin);
355 unknown->set("reason", "receiver type unresolved or method ambiguous");
356 unknowns->add(unknown);

358 item->set("unknownCallsTruncated", true);
359 }

363 Poco::JSON::Parser parser;
364 item->set("contract", parser.parse(api->toJson()));
365 item->set("evidence", "engine-build-contract");
366 if (!from.empty()) {
367 edge(from, apiId, "uses", origin);
368 nativeByFunction[from].insert(apiId);
369 }
370 if (!api->semantics || api->semantics->role.empty()) continue;
371 const auto& rule = *api->semantics;
372 auto identity = argument(expression, rule.identityArgument, path, call.scope);
373 if (rule.role == "scene_ref" && identity.values.size() == 1) {
374 auto host = argument(expression, rule.relatedArgument, path, call.scope);
375 if (host.values.size() == 1)
376 node("entity:" + *host.values.begin() + "/" + *identity.values.begin(), "scene_entity", origin);
377 } else if (rule.role == "scene_write" && callee.kind == "member" && !callee.operands.empty()) {
378 auto target = type(callee.operands[0], path, call.scope);
379 if (target.identities.size() == 1 && target.hosts.size() == 1) {
380 auto entityId = "entity:" + *target.hosts.begin() + "/" + *target.identities.begin();
381 node(entityId, "scene_entity", origin);
382 edge(from, entityId, "updates", origin);
383 edge(entityId, apiId, "uses", origin);
384 } else
385 item->set("targetCoverage", "dynamic-or-ambiguous");
386 } else if (rule.role == "event_emit" && identity.values.size() == 1) {
387 auto eventId = "event:" + *identity.values.begin();
388 node(eventId, "event", origin);
389 edge(from, eventId, "emits", origin);
390 auto subject = argument(expression, rule.relatedArgument, path, call.scope);
391 if (subject.values.size() == 1)
392 edge("quest:" + *subject.values.begin(), eventId, "subject_of", origin);
393 } else if ((rule.role == "event_type" || rule.role == "event_subject") && !call.comparison.empty() &&
394 call.comparedValue.kind == "string") {
395 if (rule.role == "event_type") {
396 auto id = "event:" + call.comparedValue.value;
397 node(id, "event", origin);
398 edge(id, from, "filtered_by", origin);
399 filters[call.scope].insert(id);
400 } else
401 subjects[call.scope].insert(call.comparedValue.value);
402 } else if (rule.role == "quest_read" || rule.role == "quest_complete") {
403 if (identity.values.size() == 1)

406 }
407 }
408 for (const auto& [scope, events] : filters)
409 for (const auto& event : events)

412 }
413 // Test references come from source-directory ownership and actual direct
414 // call reachability, not from a function name substring.
415 for (const auto& [name, locations] : functions) {
416 if (locations.size() != 1 || !locations[0].file.starts_with("tests/")) continue;
417 std::set<std::string> reached{functionId(name)}, checked;
418 for (int depth = 0; depth < 16; ++depth) {
419 auto before = reached.size();

424 }
425 if (reached.size() == before) break;
426 }
427 for (const auto& api : checked)

429 }
