27[[nodiscard]] bool isDegenerate(HexVec3 a, HexVec3 b, HexVec3 c) noexcept {
28 const HexVec3 ab = b - a;
29 const HexVec3 ac = c - a;
30 const HexVec3 normal{ab.y * ac.z - ab.z * ac.y, ab.z * ac.x - ab.x * ac.z, ab.x * ac.y - ab.y * ac.x};
31 return (normal.x * normal.x + normal.y * normal.y + normal.z * normal.z) <= 1e-8f;
32}

43void emitTriangle(HexMeshData& out, const SurfaceVertex& a, const SurfaceVertex& b, const SurfaceVertex& c,
44 const HexNoise& noise) {
45 const HexVec3 pa = noise.perturb(a.position);
46 const HexVec3 pb = noise.perturb(b.position);

65[[nodiscard]] HexVec3 acrossDirection(HexVec3 forward) noexcept {
66 const float length = std::sqrt(forward.x * forward.x + forward.z * forward.z);
67 if (length <= 1e-6f) return HexVec3{0.f, 0.f, 0.f};
68 return HexVec3{forward.z / length, 0.f, -forward.x / length};

102void buildRoadMesh(const HexMap& map, std::int32_t chunkIndex, HexMeshData& out) {
103 out.clear();
104 if (chunkIndex < 0 || chunkIndex >= map.chunkCount()) {
105 out.finalize();
106 return;
107 }
108
109 const HexNoise& noise = map.noise();
110 const float halfWidth = HexMetrics::innerRadius() * HexMetrics::kSolidFactor * 0.375f;
111 // Lift the surface off the terrain so the two coplanar layers cannot z-fight.
112 constexpr float kRoadLift = 0.05f;
113
114 for (std::int32_t row = 0; row < HexMetrics::kChunkSizeZ; ++row) {
115 for (std::int32_t column = 0; column < HexMetrics::kChunkSizeX; ++column) {
116 const HexCoordinates coordinates = map.chunkCell(chunkIndex, column, row);
117 const HexCellData* data = map.cell(coordinates);
118 if (data == nullptr || !map.hasRoad(coordinates)) continue;
119
120 const HexVec3 start = map.cellPosition(coordinates);
121 const HexVec3 ground = map.cellGroundPosition(coordinates);
122
123 // Emit each edge once, from the chunk that owns that border.
124 for (std::int32_t i = 0; i <= static_cast<std::int32_t>(HexDirection::SE); ++i) {
125 const HexDirection direction = static_cast<HexDirection>(i);
126 if (!data->flags.hasRoad(direction)) continue;
127 HexCoordinates neighbour{};
128 if (!map.getNeighbor(coordinates, direction, neighbour)) continue;
129
130 const HexVec3 neighbourCentre = map.cellPosition(neighbour);
131 const HexVec3 edgeMiddle = HexMetrics::solidEdgeMiddle(direction);
132 // The crossing sits halfway between the two surfaces so the ribbon bends with
133 // the slope instead of stepping at the border.
134 const HexVec3 crossing{ground.x + edgeMiddle.x, (start.y + neighbourCentre.y) * 0.5f,
135 ground.z + edgeMiddle.z};
136
137 const StripStation from{
138 {start.x, start.y + kRoadLift, start.z}, acrossDirection(crossing - start), halfWidth, 0.f};
139 const StripStation middle{{crossing.x, crossing.y + kRoadLift, crossing.z},
140 acrossDirection(neighbourCentre - start),
141 halfWidth,
142 0.5f};
143 const StripStation finish{{neighbourCentre.x, neighbourCentre.y + kRoadLift, neighbourCentre.z},
144 acrossDirection(neighbourCentre - crossing),
145 halfWidth,
146 1.f};
147 emitSegment(out, from, middle, noise);
148 emitSegment(out, middle, finish, noise);
149 }
150 }
151 }
152
153 out.finalize();
154}
