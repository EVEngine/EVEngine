32std::string describeSqValue(HSQUIRRELVM vm, SQInteger idx) {
33 const SQObjectType t = sq_gettype(vm, idx);
34 switch (t) {
35 case OT_NULL:
36 return "null";
37 case OT_INTEGER: {
38 SQInteger v = 0;
39 sq_getinteger(vm, idx, &v);
40 return "i:" + std::to_string(static_cast<long long>(v));
41 }
42 case OT_FLOAT: {
43 SQFloat v = 0;
44 sq_getfloat(vm, idx, &v);
45 return "f:" + std::to_string(static_cast<double>(v));
46 }
47 case OT_BOOL: {
48 SQBool v = SQFalse;
49 sq_getbool(vm, idx, &v);
50 return v ? "b:1" : "b:0";
51 }
52 case OT_STRING: {
53 const SQChar* s = nullptr;
54 sq_getstring(vm, idx, &s);
55 return std::string("s:") + (s ? s : "");
56 }
57 case OT_TABLE:
58 return "table";
59 case OT_ARRAY:
60 return "array";
61 case OT_USERDATA:
62 return "userdata";
63 case OT_CLOSURE:
64 return "closure";
65 case OT_NATIVECLOSURE:
66 return "native";
67 case OT_GENERATOR:
68 return "generator";
69 case OT_USERPOINTER:
70 return "userpointer";
71 case OT_THREAD:
72 return "thread";
73 case OT_CLASS:
74 return "class";
75 case OT_INSTANCE:
76 return "instance";
77 case OT_WEAKREF:
78 return "weakref";
79 default:
80 return "other";
81 }

92SQInteger runtimeErrorHook(HSQUIRRELVM v) {
93 if (!g_active) return 0;
94 // Capture the full context (message + live call stack) so Runtime::execute
95 // can enrich the ScriptException it throws once the call unwinds.
96 eve::script::ScriptErrorContext ctx = eve::script::captureScriptError(v);
97 const std::string msg = ctx.empty() ? std::string("script error")
98 : eve::script::formatScriptError(ctx);
99 try {
100 g_active->notifyError(msg);
101 } catch (...) {
102 }
103 ctx.reported = true;
104 eve::script::setLastScriptError(v, std::move(ctx));
105 return 0;
106}

491void DevTool::handleDebugEvent(HSQUIRRELVM vm, int type, const char* source, int line,
492 const char* funcname) {
493 SourceLoc loc;
494 loc.source = source ? source : "";
495 loc.line = line;
496 loc.function = funcname ? funcname : "";
497
498 // Squirrel passes 'l' / 'c' / 'r' as event type characters.
499 switch (type) {
500 case 'c':
501 loc.function = Debugger::instance().onCall(loc);
502 localSnap_.erase(static_cast<int>(graph_.currentStack().size()));
503 profileCall(loc.function);
504 break;
505 case 'r':
506 Debugger::instance().onReturn(loc);
507 // Drop snapshot for the frame that just returned.
508 localSnap_.erase(static_cast<int>(graph_.currentStack().size()) + 1);
509 profileReturn();
510 break;
511 case 'l':
512 graph_.onLine(loc);
513 profileLine(loc.function);
514 if (sampleLocals_) sampleFrameLocals(vm, loc);
515 if (Debugger::instance().onScriptLine(loc)) {
516 dap().notifyStopped(Debugger::instance().lastPauseReason(), loc);
517 Debugger::instance().refreshWatches();
518 Debugger::instance().waitWhilePaused([this]() { pumpWhilePaused(); });
519 }
520 break;
521 default:
522 break;
523 }
524}

526void DevTool::handleDebugHotkey(const std::string& key) {
527 if (key.empty()) return;
528 Debugger& dbg = debugger();
529 if (key == "F5") {
530 if (!dbg.isPaused()) return;
531 dbg.resume();
532 dap().notifyContinued();
533 return;
534 }
535 if (key == "F10") {
536 if (!dbg.isPaused()) return;
537 dbg.stepOver();
538 dap().notifyContinued();
539 return;
540 }
541 if (key == "F11") {
542 if (!dbg.isPaused()) return;
543 dbg.stepInto();
544 dap().notifyContinued();
545 return;
546 }
547 if (key == "F8") {
548 if (!dbg.isPaused()) return;
549 dbg.stepFrame();
550 dap().notifyContinued();
551 return;
552 }
553 if (key == "Pause") {
554 // Already inside a script pause — Pause resumes (toggle).
555 if (!dbg.isPaused()) return;
556 dbg.resume();
557 dap().notifyContinued();
558 }

561void DevTool::pumpWhilePaused() {
562 poll(); // DAP continue / next / pause
563
564 auto* ev = eve::ModuleManager::getInstance<eve::platform_event::PlatformEvent>("PlatformEvent");
565 if (!ev) return;
566 ev->pump();
567
568 // Consume debug hotkeys so F5/F8/F10/F11 work while blocked in the line hook;
569 // re-queue everything else for the main loop after resume.
570 std::vector<std::unique_ptr<eve::platform_event::Message>> keep;
571 while (auto msg = ev->pollOwned()) {
572 bool handled = false;
573 if (msg->name == "keypressed" && !msg->args.empty() &&
574 msg->args[0].type == eve::platform_event::Variant::Type::String) {
575 const std::string& key = msg->args[0].s;
576 if (key == "F5" || key == "F8" || key == "F10" || key == "F11" || key == "Pause") {
577 handleDebugHotkey(key);
578 handled = true;
579 }
580 }
581 if (!handled) keep.push_back(std::move(msg));
582 }
583 for (auto& msg : keep) ev->push(std::move(msg));
584}

586void DevTool::sampleFrameLocals(HSQUIRRELVM vm, const SourceLoc& loc) {
587 if (!vm) return;
588 const int depth = static_cast<int>(graph_.currentStack().size());
589 auto& prev = localSnap_[depth];
590 std::unordered_map<std::string, std::string> cur;
591
592 // Level 0 = current script frame. The native debug hook is invoked as a
593 // direct C call (no CallInfo pushed), so the top of the Squirrel call
594 // stack is the script frame that triggered the line event.
595 const SQUnsignedInteger level = 0;
596 for (SQUnsignedInteger n = 0;; ++n) {
597 const SQInteger top = sq_gettop(vm);
598 const SQChar* name = sq_getlocal(vm, level, n);
599 if (!name) {
600 sq_settop(vm, top);
601 break;
602 }
603 const std::string key(name);
604 // Skip temporaries / this binding noise if desired; keep "this" for OO flow.
605 const std::string val = describeSqValue(vm, -1);
606 sq_settop(vm, top);
607
608 cur[key] = val;

610 if (it == prev.end() || it->second != val) {
611 // First sighting or value change ⇒ definition for the data-flow graph.
612 graph_.onDef(loc, key);
613 }
614 }
615 prev.swap(cur);

643 }
644 }

675 const std::vector<std::string>& hintVars) {
676 if (!vm_) return;
677 SourceLoc site = loc;
678 if (site.empty()) {
679 SQStackInfos si;
680 if (SQ_SUCCEEDED(sq_stackinfos(vm_, 1, &si))) {

712std::string DevTool::notifyError(const std::string& errorMessage,
713 const std::vector<std::string>& hintVars) {
714 SourceLoc site;
715 if (vm_) {
716 // When called from the uncaught-error hook, level 0 is the native hook
717 // itself and level 1 is the throwing script frame, so this already
718 // lands on the exact throw site (before the stack unwinds). When called
719 // from a script catch (eve.dev.reportError), level 1 is the catch
720 // statement that reported the error (the game script when load.nut
721 // calls the native reporter directly).
722 SQStackInfos si;
723 if (SQ_SUCCEEDED(sq_stackinfos(vm_, 1, &si))) {
724 if (si.source) site.source = si.source;
725 site.line = static_cast<int>(si.line);
726 if (si.funcname) site.function = si.funcname;
727 }
728 }
729 markErrorUses(site, hintVars);
730 // Ensure the error is marked on the render flow even if Exception ctor
731 // already did (idempotent append of another Error node is fine).
732 if (renderTraceEnabled_) renderFlow_.error(errorMessage.c_str());
733
734 std::string report;
735 if (vm_ || !graph_.events().empty()) report += formatError(errorMessage, hintVars);
736 if (renderTraceEnabled_ && !renderFlow_.events().empty()) {
737 if (!report.empty()) report += "\n";
738 report += renderFlow_.formatErrorReport(errorMessage);
739 }
740 if (report.empty()) report = std::string("Error: ") + errorMessage + "\n";
741 lastReport_ = report;
742 lastError_ = errorMessage;
743 lastSlice_ = analyzeError(errorMessage, hintVars);
744
745 // If a scenario is being recorded, pair the failure report + site with it so
746 // the dumped scenario becomes a reproducible baseline + input sequence.
747 ScenarioRecorder::instance().setErrorInfo(report, site.source + ":" + std::to_string(site.line));
748
749 RenderVision::instance().notifyPending("error", site.source, site.line);
750 if (debugger().breakOnError()) {
751 // Godot "Break on Error": stop at the reported site. Block inside the
752 // hook so the IDE sees a stable frame instead of the next executed line.
753 debugger().pauseAt(PauseReason::Exception, site);
754 dap().notifyStopped(PauseReason::Exception, site, errorMessage);
755 debugger().waitWhilePaused([this]() { pumpWhilePaused(); });
756 }
757
758 return lastReport_;
759}
