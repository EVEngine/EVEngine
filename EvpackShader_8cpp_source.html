34Result<std::unique_ptr<EvpackShader>> EvpackShader::load(const asset::EvpackResourceReader& reader,
35 const AssetRef& asset,
36 const asset::EvpackCapabilities& capabilities,
37 graphics::Graphics& graphics) {
38 if (graphics.getBackendName() != "vulkan" || capabilities.graphics != graphics.getBackendName())
39 return failure<std::unique_ptr<EvpackShader>>(DiagnosticCode::Unsupported, "Shader backend mismatch");
40 auto decoded = asset::loadShaderAsset(reader, asset, capabilities);
41 if (!decoded) return Result<std::unique_ptr<EvpackShader>>::failure(decoded.status());
42 auto validated = validateShaderAssetGpu(decoded.value());
43 if (!validated) return Result<std::unique_ptr<EvpackShader>>::failure(validated.status());
44 auto result = std::unique_ptr<EvpackShader>(
45 new EvpackShader(std::make_unique<Impl>(Impl{graphics, std::move(decoded).takeValue()})));
46 try {
47 auto& impl = *result->impl_;
48 impl.shader = impl.definition.interface == asset::ShaderAssetInterface::Mesh3D
49 ? graphics.newMeshShaderFromSpv(impl.definition.vertex, impl.definition.fragment)
50 : graphics.newShaderFromSpv(impl.definition.vertex, impl.definition.fragment);
51 if (!impl.shader)
52 return failure<std::unique_ptr<EvpackShader>>(DiagnosticCode::Failed,
53 "Shader upload returned no allocation");
54 for (const auto& parameter : impl.definition.parameters) {
55 switch (parameter.defaults.size()) {
56 case 1: impl.shader->declareFloat(parameter.name); break;
57 case 2: impl.shader->declareVec2(parameter.name); break;
58 case 3: impl.shader->declareVec3(parameter.name); break;
59 case 4: impl.shader->declareVec4(parameter.name); break;
60 default:
61 return failure<std::unique_ptr<EvpackShader>>(DiagnosticCode::InvalidArgument,
62 "Invalid parameter width");
63 }
64 impl.shader->sendToVar(parameter.name, parameter.defaults.data(),
65 parameter.defaults.size() * sizeof(float));
66 }
67 // Initialize/upload the entire ABI block, including unused trailing slots.
68 unsigned padding = 0;
69 while (impl.shader->usedFloats() < graphics::Shader::kMaxFloats) {
70 const auto name = "__eve_padding_" + std::to_string(padding++);
71 if (!impl.shader->hasUniform(name)) impl.shader->declareFloat(name);
72 }
73 return Result<std::unique_ptr<EvpackShader>>::success(std::move(result));
74 } catch (const std::exception& error) {
75 return failure<std::unique_ptr<EvpackShader>>(DiagnosticCode::Failed, error.what());
76 }
77}
