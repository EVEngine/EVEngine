62LevelChange LevelDocument::moveLayer(int from, int to) {
63 if (!layer(from) || to < 0 || to >= int(layers_.size()) || from == to) return LevelChange::Unchanged;
64 auto moving = std::move(layers_[from]);
65 layers_.erase(layers_.begin() + from);
66 layers_.insert(layers_.begin() + to, std::move(moving));
67 return LevelChange::Changed;
68}
