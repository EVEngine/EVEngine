42void NetRpc::call(uint16_t msgId, const void* data, size_t n, bool reliable) {
43 if (!link_ || (data == nullptr && n > 0)) return;
44 std::vector<char> env;
45 env.push_back(static_cast<char>(msgId & 0xff));
46 env.push_back(static_cast<char>((msgId >> 8) & 0xff));
47 const char* p = static_cast<const char*>(data);
48 env.insert(env.end(), p, p + n);
49 link_->send(reliable ? UdpLink::MsgType::Reliable : UdpLink::MsgType::Unreliable, 0,
50 env.data(), env.size());
51}

68void NetRpc::onLinkMessage(UdpLink::MsgType type, uint8_t channel,
69 const char* data, size_t n) {
70 (void)type;
71 if (channel != 0 || n < 2) return;
72 const uint16_t msgId = static_cast<uint16_t>(uint8_t(data[0])) |
73 (static_cast<uint16_t>(uint8_t(data[1])) << 8);
74 const char* payload = data + 2;
75 const size_t payloadLen = n - 2;
76
77 auto sit = scriptHandlers_.find(msgId);
78 if (sit != scriptHandlers_.end()) {
79 callScriptFn(sit->second, std::string(payload, payloadLen));
80 return;
81 }
82 auto it = handlers_.find(msgId);
83 if (it == handlers_.end()) return;
84 NetReader reader(payload, payloadLen);
85 it->second(reader);
86}
