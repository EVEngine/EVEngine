442 output.actionId, step.tick, execution.executionId});
443 } else if (execution.candidate.kind == ClimbingActionKind::LedgeGrab) {
444 phase_ = ClimbingPhase::Hanging;
445 execution.branchWindowOpen = true;
446 output.branchWindowOpen = true;
447 enqueueEvent({ClimbingEventKind::Hanging, output.actionId, step.tick, execution.executionId});
448 } else {

450 terminalCode_ = "climbing.completed";

452 enqueueEvent({ClimbingEventKind::Landed, output.actionId, step.tick, execution.executionId});

454 if (graphBound) {
455 auto released = releaseAnchorReservation();
456 released.ignore("completed graph dismount releases occupancy");
457 }
458 }

461 output.terminalVelocity =
462 terminalVelocityFor(execution.action, actual, static_cast<float>(1.0 / step.delta.seconds()));
463 output.hasTerminalVelocity = true;
464 execution_.reset();
465 }
466 }
467 return eve::Result<ClimbingAdvance>::success(std::move(output));
468}
