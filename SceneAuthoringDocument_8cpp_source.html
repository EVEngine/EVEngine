17bool number(const Value& value, double& out) {
18 if (const auto* v = value.getIf<double>())
19 out = *v;
20 else if (const auto* v = value.getIf<std::int64_t>())
21 out = static_cast<double>(*v);
22 else

26bool vector(const Value* value, int count) {
27 const auto* a = value ? value->getIf<Value::Array>() : nullptr;
28 if (!a || a->size() != static_cast<std::size_t>(count)) return false;

35Result<void> validate(const std::string& section, const Value& value) {
36 const auto reject = [](const char* text) {
37 return failed<void>(Status::Rejected, RuleId("scene.authoring.schema"), text);
38 };
39 const auto* object = value.getIf<Value::Object>();
40 if (!object) return reject("Authoring payload must be an object");
41 if (object->empty() && section != "sources") return applied<void>();
42 std::set<std::string> keys;
43 if (section == "sources")
44 keys = {"kind", "seed", "mesh_params", "path", "revision", "tint",
45 "roughness", "metallic", "visible", "cast_shadow", "receive_shadow"};
46 else if (section == "camera")
47 keys = {"eye", "target", "up", "fov", "near", "far", "exposure", "ambient"};
48 else if (section == "lighting")
49 keys = {"background", "direction", "color"};
50 else
51 return reject("Unknown authoring payload section");
52 for (const auto& [key, entry] : *object) {
53 if (!keys.contains(key)) return reject("Unknown authoring payload field");
54 if (key == "kind" || key == "path" || key == "revision") {
55 const auto* text = entry.getIf<std::string>();
56 if (!text || text->empty()) return reject("Source identifiers must be nonempty strings");
57 } else if (key == "seed") {
58 if (!entry.getIf<std::int64_t>()) return reject("Seed must be an integer");
59 } else if (key == "mesh_params") {
60 const auto* params = entry.getIf<Value::Object>();
61 if (!params) return reject("Mesh parameters must be an object");
62 for (const auto& [name, parameter] : *params) {
63 (void)name;
64 double numeric;
65 if (!parameter.getIf<std::string>() && !number(parameter, numeric))
66 return reject("Recipe parameters must be finite numbers or strings");
67 }
68 } else if (key == "visible" || key == "cast_shadow" || key == "receive_shadow") {
69 if (!entry.getIf<bool>()) return reject("Presentation flags must be booleans");
70 } else if (key == "eye" || key == "target" || key == "up" || key == "ambient" || key == "tint" ||
71 key == "direction" || key == "background" || key == "color") {
72 if (!vector(&entry, 3)) return reject("Vectors require three finite numbers");
73 } else {
74 double numeric;
75 if (!number(entry, numeric)) return reject("Scalar must be finite");
76 if ((key == "roughness" || key == "metallic") && (numeric < 0 || numeric > 1))
77 return reject("Material scalars must be in [0,1]");
78 if (key == "fov" && (numeric <= 0 || numeric >= 179)) return reject("Vertical FOV must be in (0,179)");
79 if ((key == "near" || key == "far") && numeric <= 0) return reject("Clip distances must be positive");
80 }
81 }
82 if (section == "sources") {
83 const auto* kind = field(value, "kind");
84 if (!kind) return reject("Source kind is required");
85 if (*kind->getIf<std::string>() == "asset" && (!field(value, "path") || !field(value, "revision")))
86 return reject("Imported sources require a path and immutable revision");
87 } else if (section == "camera") {
88 for (const auto* key : {"eye", "target", "up", "fov", "near", "far", "exposure", "ambient"})
89 if (!field(value, key)) return reject("Camera payload must contain all camera fields");
90 double nearClip, farClip;
91 if (!number(*field(value, "near"), nearClip) || !number(*field(value, "far"), farClip) || nearClip >= farClip)
92 return reject("Far clip must exceed near clip");
93 const auto components = [&](const char* key) {
94 const auto& a = *field(value, key)->getIf<Value::Array>();
95 std::array<double, 3> out{};
96 for (std::size_t i = 0; i < 3; ++i) number(a[i], out[i]);
97 return out;
98 };
99 const auto eye = components("eye"), target = components("target"), up = components("up");
100 const double x = target[0] - eye[0], y = target[1] - eye[1], z = target[2] - eye[2];
101 const double cx = y * up[2] - z * up[1], cy = z * up[0] - x * up[2], cz = x * up[1] - y * up[0];
102 if (cx * cx + cy * cy + cz * cz < 1e-12) return reject("Camera basis is degenerate");
103 } else {
104 for (const auto* key : {"background", "direction", "color"})
105 if (!field(value, key)) return reject("Lighting requires background, direction and color");
106 double lengthSquared = 0;
107 for (const auto& component : *field(value, "direction")->getIf<Value::Array>()) {
108 double numeric;
109 number(component, numeric);
110 lengthSquared += numeric * numeric;
111 }
112 if (lengthSquared < 1e-12) return reject("Light direction is degenerate");
113 }
114 return applied<void>();
115}

202Result<void> SceneAuthoringDocumentTarget::applyDomainOperation(const DomainOperation& operation) {
203 if (operation.type != "scene.authoring.restore.v1") return SceneTargetBase::applyDomainOperation(operation);
204 if (operation.target != targetId())

210 // Prepare all owning copies before changing the hierarchy.
