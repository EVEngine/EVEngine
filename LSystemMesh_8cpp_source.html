30void basisFor(V3 axis, V3& right, V3& forward) {
31 axis = norm(axis);
32 const V3 helper = std::fabs(axis.y) < 0.92f ? V3{0.f, 1.f, 0.f} : V3{1.f, 0.f, 0.f};
33 right = norm(cross(helper, axis));
34 forward = norm(cross(axis, right));

37void addTaperedCylinder(MeshBuild& out, V3 a, V3 b, float r0, float r1, int sides) {
38 const V3 axis = norm(sub(b, a));
39 V3 right, forward;
40 basisFor(axis, right, forward);
41 const uint32_t base = uint32_t(out.getVertexCount());
42 for (int ring = 0; ring < 2; ++ring) {
43 const V3 center = ring ? b : a;
44 const float radius = ring ? r1 : r0;
45 for (int i = 0; i < sides; ++i) {
46 const float t = float(i) / float(sides);
47 const float angle = t * 2.f * kPi;
48 const V3 radial = add(mul(right, std::cos(angle)), mul(forward, std::sin(angle)));
49 const V3 p = add(center, mul(radial, radius));
50 // Bark occupies the left side of the texture atlas; foliage the right.
51 out.addVertex(p.x, p.y, p.z, radial.x, radial.y, radial.z, t * 0.45f, float(ring));
52 }
53 }
54 for (int i = 0; i < sides; ++i) {
55 const uint32_t n = uint32_t((i + 1) % sides);
56 const uint32_t i0 = base + uint32_t(i), i1 = base + n;
57 const uint32_t i2 = base + uint32_t(sides) + uint32_t(i);
58 const uint32_t i3 = base + uint32_t(sides) + n;
59 out.addTriangle(i0, i2, i1);
60 out.addTriangle(i1, i2, i3);
61 }
62}

64void addLeafCard(MeshBuild& out, V3 c, V3 direction, float size, float twist) {
65 V3 right, up;
66 basisFor(norm(direction), right, up);
67 right = add(mul(right, std::cos(twist)), mul(up, std::sin(twist)));
68 up = norm(cross(norm(direction), right));
69 const V3 r = mul(right, size * 0.5f), u = mul(up, size);
70 const V3 normal = norm(cross(right, up));
71 const uint32_t base = uint32_t(out.getVertexCount());
72 const V3 points[4] = {sub(sub(c, r), u), add(sub(c, u), r), add(add(c, r), u), add(sub(c, r), u)};
73 const float uv[4][2] = {{0, 0}, {1, 0}, {1, 1}, {0, 1}};
74 for (int i = 0; i < 4; ++i)
75 out.addVertex(points[i].x, points[i].y, points[i].z, normal.x, normal.y, normal.z, 0.55f + uv[i][0] * 0.45f,
76 uv[i][1]);
77 out.addTriangle(base, base + 1, base + 2);
78 out.addTriangle(base, base + 2, base + 3);
79 out.addTriangle(base + 2, base + 1, base);
80 out.addTriangle(base + 3, base + 2, base);
81}

83void configurePreset(LSystem& ls, const std::string& style) {
84 ls.setLeafSymbols("L");
85 if (style == "fern") {
86 ls.setAxiom("X");
87 ls.clearRules();
88 ls.addRule('X', "F-[[X]+X]+F[+FX]-X");
89 ls.addRule('F', "FF");
90 ls.setAngle(25.f);
91 ls.setInitialHeading(0.f, 1.f, 0.f);
92 ls.setBranchRadius(0.05f);
93 ls.setBranchRadiusFalloff(0.8f);
94 } else if (style == "plant") {
95 ls.setAxiom("F");
96 ls.clearRules();
97 ls.addRule('F', "F[+F]F[-F]F");
98 ls.setAngle(26.f);
99 ls.setInitialHeading(0.f, 1.f, 0.f);
100 ls.setBranchRadius(0.09f);
101 ls.setBranchRadiusFalloff(0.72f);
102 } else if (style == "weed") {
103 ls.setAxiom("X");
104 ls.clearRules();
105 ls.addRule('X', "F[-X][+X]FX");
106 ls.addRule('X', "F[-X]FX");
107 ls.addRule('X', "F[&X]F^X");
108 ls.setAngle(28.f);
109 ls.setInitialHeading(0.f, 1.f, 0.f);
110 ls.setBranchRadius(0.06f);
111 ls.setBranchRadiusFalloff(0.7f);
112 } else { // tree (default)
113 ls.setAxiom("A");
114 ls.clearRules();
115 ls.addRules('A', {"F[&L A]F[&L A]^F[&L A]F L A", "F[&L A]F L A", "F L A"},
116 {2.f, 1.f, 1.f});
117 ls.setAngle(22.f);
118 ls.setInitialHeading(0.f, 1.f, 0.f);
119 ls.setBranchRadius(0.16f);
120 ls.setBranchRadiusFalloff(0.62f);
121 }

126bool generateLSystemMesh(const Params& params, MeshBuild& out, std::string& error) {
127 const std::string style = params.getString("style", "tree");
128 if (style != "tree" && style != "fern" && style != "plant" && style != "weed") {
129 error = "mesh.lsystem: unknown style '" + style + "'";
130 return false;
131 }
132 const std::string leafMode = params.getString("leafMode", "cards");
133 if (leafMode != "cards" && leafMode != "none") {
134 error = "mesh.lsystem: unknown leafMode '" + leafMode + "'";
135 return false;
136 }
137
138 LSystem ls;
139 configurePreset(ls, style);
140 ls.setSeed(params.getSeed());
141 ls.setIterations(params.getInt("iterations", 5));
142 ls.setAngle(params.getFloat("angle", 22.f));
143 ls.setBranchRadius(params.getFloat("branchRadius", 0.16f));
144 ls.setBranchRadiusFalloff(params.getFloat("radiusFalloff", 0.62f));
145 ls.setLeafSize(params.getFloat("leafSize", 0.4f));
146 ls.setTropism(0.f, params.getFloat("tropism", 0.f), 0.f);
147 const int branchSegments = std::clamp(params.getInt("branchSegments", 6), 3, 24);
148 const int leavesPerMarker = std::max(1, params.getInt("leavesPerMarker", 1));
149
150 LSystemResult result;
151 ls.generate(result);
152 if (result.segments.empty()) {
153 error = "mesh.lsystem: generated no geometry";
154 return false;
155 }
156
157 out.reserve(int(result.segments.size() * 12), int(result.segments.size() * 24));
158 for (const LSystemSegment& seg : result.segments) {
159 if (seg.leaf) {
160 if (leafMode == "none") continue;
161 const V3 c = {seg.ex, seg.ey, seg.ez};
162 const V3 dir = {seg.dx, seg.dy, seg.dz};
163 for (int i = 0; i < leavesPerMarker; ++i) {
164 const float twist = float(i) * 2.39996323f; // golden angle
165 addLeafCard(out, c, dir, seg.leafSize, twist);
166 }
167 continue;
168 }
169 const V3 a = {seg.sx, seg.sy, seg.sz};
170 const V3 b = {seg.ex, seg.ey, seg.ez};
171 addTaperedCylinder(out, a, b, std::max(1e-4f, seg.r0), std::max(1e-4f, seg.r1), branchSegments);
172 }
173
174 out.setMeta("recipe", "mesh.lsystem");
175 out.setMeta("style", style);
176 out.setMeta("leafMode", leafMode);
177 out.setMeta("seed", std::to_string(params.getSeed()));
178 if (out.empty()) {
179 error = "mesh.lsystem: generated an empty mesh";
180 return false;
181 }
182 return true;
183}
