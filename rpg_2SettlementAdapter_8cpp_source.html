114eve::Result<void> RPGSettlementAdapter::validate(settlement::SettlementContext& context) {
115 if (!(context.request().target == targetRef_))
116 return failure<void>(eve::DiagnosticCode::Conflict,
117 "RPG settlement request target does not match adapter target", "target");
118 const bool damage = isDamage(context);
119 const bool healing = isHealing(context);
120 if (!damage && !healing)
121 return failure<void>(eve::DiagnosticCode::Unsupported,
122 "RPG settlement adapter supports damage and healing kinds only", "kind");
123
124 auto health = readBase(config_.healthAttribute);
125 const bool healthOk = health.ok();
126 if (!healthOk) {
127 const auto status = health.status();
128 return failure<void>(status);
129 }
130 auto maximum = readBase(config_.maxHealthAttribute);
131 const bool maximumOk = maximum.ok();
132 if (!maximumOk) {
133 const auto status = maximum.status();
134 return failure<void>(status);
135 }
136 if (health.value() > maximum.value())
137 return failure<void>(eve::DiagnosticCode::InvariantViolation, "RPG health exceeds max health", "health");
138 context.setStageDetail("policy", "rpg");
139 context.setStageDetail("health_attribute", config_.healthAttribute);
140 context.setStageDetail("kind", context.request().kind);
141 (void)damage;
142 return eve::Result<void>::success();
143}

145eve::Result<void> RPGSettlementAdapter::sourceModifiers(settlement::SettlementContext& context) {
146 double multiplier = 1.0;
147 if (const auto value = numberValue(contextValue(context, "source_multiplier"))) {
148 multiplier = *value;
149 } else {
150 auto sourceMultiplier = readSourceFinal(config_.sourceMultiplierAttribute, 1.0);
151 const bool sourceMultiplierOk = sourceMultiplier.ok();
152 if (!sourceMultiplierOk) {
153 const auto status = sourceMultiplier.status();
154 return failure<void>(status);
155 }
156 multiplier = sourceMultiplier.value();
157 }
158 if (!std::isfinite(multiplier) || multiplier < 0.0)
159 return failure<void>(eve::DiagnosticCode::InvalidArgument,
160 "RPG source multiplier must be finite and non-negative", "source_multiplier");
161
162 auto scaled = context.setMagnitude(context.magnitude() * multiplier);
163 const bool scaledOk = scaled.ok();
164 if (!scaledOk) {
165 const auto status = scaled.status();
166 return failure<void>(status);
167 }
168
169 bool critical = boolValue(contextValue(context, "critical")).value_or(false);
170 if (!critical) {
171 const auto roll = numberValue(contextValue(context, "critical_roll"));
172 auto chance = readSourceFinal(config_.criticalChanceAttribute, 0.0);
173 const bool chanceOk = chance.ok();
174 if (!chanceOk) {
175 const auto status = chance.status();
176 return failure<void>(status);
177 }
178 if (roll && std::isfinite(*roll)) {
179 const double probability = std::clamp(chance.value(), 0.0, 1.0);
180 critical = *roll >= 0.0 && *roll < probability;
181 }
182 }
183
184 double criticalMultiplier = 1.0;
185 if (critical) {
186 if (const auto value = numberValue(contextValue(context, "critical_multiplier"))) {
187 criticalMultiplier = *value;
188 } else {
189 auto configured = readSourceFinal(config_.criticalMultiplierAttribute, 2.0);
190 const bool configuredOk = configured.ok();
191 if (!configuredOk) {
192 const auto status = configured.status();
193 return failure<void>(status);
194 }
195 criticalMultiplier = configured.value();
196 }
197 if (!std::isfinite(criticalMultiplier) || criticalMultiplier < 1.0)
198 return failure<void>(eve::DiagnosticCode::InvalidArgument,
199 "RPG critical multiplier must be finite and at least one", "critical_multiplier");
200 auto critScaled = context.setMagnitude(context.magnitude() * criticalMultiplier);
201 const bool critScaledOk = critScaled.ok();
202 if (!critScaledOk) {
203 const auto status = critScaled.status();
204 return failure<void>(status);
205 }
206 }
207 context.setCritical(critical);
208 context.setStageDetail("source_multiplier", multiplier);
209 context.setStageDetail("critical", critical);
210 if (critical) context.setStageDetail("critical_multiplier", criticalMultiplier);
211 return eve::Result<void>::success();
212}

214eve::Result<void> RPGSettlementAdapter::targetMitigation(settlement::SettlementContext& context) {
215 if (!isDamage(context)) return eve::Result<void>::success();
216
217 std::string element = stringValue(contextValue(context, "element")).value_or("damage");
218 const std::string resistanceAttribute = config_.resistancePrefix + element;
219 auto resistance = readFinal(resistanceAttribute, 0.0);
220 const bool resistanceOk = resistance.ok();
221 if (!resistanceOk) {
222 const auto status = resistance.status();
223 return failure<void>(status);
224 }
225 const double factor = 1.0 - std::clamp(resistance.value(), 0.0, 1.0);
226 const double before = context.magnitude();
227 auto changed = context.setMagnitude(before * factor);
228 const bool changedOk = changed.ok();
229 if (!changedOk) {
230 const auto status = changed.status();
231 return failure<void>(status);
232 }
233 auto recorded = context.addResisted(before - context.magnitude());
234 const bool recordedOk = recorded.ok();
235 if (!recordedOk) {
236 const auto status = recorded.status();
237 return failure<void>(status);
238 }
239 context.setStageDetail("element", element);
240 context.setStageDetail("resistance", resistance.value());
241 return eve::Result<void>::success();
242}

244eve::Result<void> RPGSettlementAdapter::armorShield(settlement::SettlementContext& context) {
245 if (!isDamage(context)) return eve::Result<void>::success();
246
247 auto armor = readFinal(config_.armorAttribute, 0.0);
248 const bool armorOk = armor.ok();
249 if (!armorOk) {
250 const auto status = armor.status();
251 return failure<void>(status);
252 }
253 const double penetration = std::max(0.0, numberValue(contextValue(context, "penetration")).value_or(0.0));
254 const double effectiveArmor = std::max(0.0, armor.value() - penetration);
255 const double armorFactor = 100.0 / (100.0 + effectiveArmor);
256 const double beforeArmor = context.magnitude();
257 auto armorChanged = context.setMagnitude(beforeArmor * armorFactor);
258 const bool armorChangedOk = armorChanged.ok();
259 if (!armorChangedOk) {
260 const auto status = armorChanged.status();
261 return failure<void>(status);
262 }
263 auto armorLoss = context.addResisted(beforeArmor - context.magnitude());
264 const bool armorLossOk = armorLoss.ok();
265 if (!armorLossOk) {
266 const auto status = armorLoss.status();
267 return failure<void>(status);
268 }
269
270 auto shield = readBase(config_.shieldAttribute);
271 double shieldValue = 0.0;
272 if (shield.ok()) {
273 shieldValue = shield.value();
274 } else if (shield.code() != eve::StatusCode::NotFound) {
275 const auto status = shield.status();
276 return failure<void>(status);
277 }
278 const double absorbed = std::min(shieldValue, context.magnitude());
279 auto shieldChanged = context.setMagnitude(context.magnitude() - absorbed);
280 const bool shieldChangedOk = shieldChanged.ok();
281 if (!shieldChangedOk) {
282 const auto status = shieldChanged.status();
283 return failure<void>(status);
284 }
285 auto absorbedRecorded = context.addAbsorbed(absorbed);
286 const bool absorbedRecordedOk = absorbedRecorded.ok();
287 if (!absorbedRecordedOk) {
288 const auto status = absorbedRecorded.status();
289 return failure<void>(status);
290 }
291 context.setStageDetail("armor", armor.value());
292 context.setStageDetail("penetration", penetration);
293 context.setStageDetail("shield", shieldValue);
294 context.setStageDetail("absorbed", absorbed);
295 return eve::Result<void>::success();
296}

322eve::Result<settlement::PreparedApply> RPGSettlementAdapter::prepareApply(
323 const settlement::SettlementContext& context) {
324 auto health = readBase(config_.healthAttribute);
325 const bool healthOk = health.ok();
326 if (!healthOk) {
327 const auto status = health.status();
328 return failure<settlement::PreparedApply>(status);
329 }
330 auto maximum = readBase(config_.maxHealthAttribute);
331 const bool maximumOk = maximum.ok();
332 if (!maximumOk) {
333 const auto status = maximum.status();
334 return failure<settlement::PreparedApply>(status);
335 }
336
337 auto candidate = target_.attributes()->values;
338 const double nextHealth =
339 isDamage(context) ? health.value() - context.magnitude() : health.value() + context.magnitude();
340 if (!std::isfinite(nextHealth) || nextHealth < 0.0 || nextHealth > maximum.value())
341 return failure<settlement::PreparedApply>(eve::DiagnosticCode::InvariantViolation,
342 "RPG settlement candidate health is outside its validated bounds",
343 "health");
344 candidate.setBase(config_.healthAttribute, nextHealth);
345
346 if (isDamage(context) && !config_.shieldAttribute.empty() && context.absorbed() > 0.0) {
347 auto shield = readBase(config_.shieldAttribute);
348 const bool shieldOk = shield.ok();
349 if (!shieldOk) {
350 const auto status = shield.status();
351 if (status.code() != eve::StatusCode::NotFound) return failure<settlement::PreparedApply>(status);
352 } else {
353 if (context.absorbed() > shield.value())
354 return failure<settlement::PreparedApply>(
355 eve::DiagnosticCode::InvariantViolation,
356 "RPG shield absorption exceeds the authoritative shield balance", "shield");
357 candidate.setBase(config_.shieldAttribute, shield.value() - context.absorbed());
358 }
359 }
