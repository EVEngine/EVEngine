50eve::Result<settlement::SettlementRequest> makeStatusTickSettlementRequest(
51 const StatusTickEvent& tick, SubjectRef source, SubjectRef target, SimulationTick simulationTick) {
52 if (tick.actor == nullptr || tick.effectId.empty() || tick.instanceId <= 0 || tick.stacks <= 0)
53 return requestFailure(eve::DiagnosticCode::InvalidArgument, "RPG status tick identity is invalid", "tick");
54 if (!target.isValid())
55 return requestFailure(eve::DiagnosticCode::InvalidArgument, "RPG status tick target must be valid", "target");
56 const EffectDefinition* definition = EffectRegistry::find(tick.effectId);
57 if (definition == nullptr)
58 return requestFailure(eve::DiagnosticCode::NotFound, "RPG status tick definition was not found",
59 "tick.effectId");
60 const auto encoded = definition->extra.find("settlement.tick");
61 if (encoded == definition->extra.end())
62 return requestFailure(eve::DiagnosticCode::NotFound, "RPG effect has no settlement.tick configuration",
63 "effect.extra.settlement.tick");
64 auto parsed = eve::Value::fromJson(encoded->second);
65 if (!parsed) return requestFailure(parsed.status());
66 const auto* object = parsed.value().getIf<eve::Value::Object>();
67 if (object == nullptr)
68 return requestFailure(eve::DiagnosticCode::InvalidArgument, "settlement.tick must be a JSON object",
69 "effect.extra.settlement.tick");
70 static const std::set<std::string> fields = {"context", "kind", "magnitude", "resource", "tags"};
71 for (const auto& [name, value] : *object) {
72 (void)value;
73 if (!fields.contains(name))
74 return requestFailure(eve::DiagnosticCode::InvalidArgument, "unknown settlement.tick field",
75 "effect.extra.settlement.tick." + name);
76 }
77 const auto* kindValue = objectField(*object, "kind");
78 const auto* resourceValue = objectField(*object, "resource");
79 const auto* magnitudeValue = objectField(*object, "magnitude");
80 const auto* kind = kindValue == nullptr ? nullptr : kindValue->getIf<std::string>();
81 const auto* resource = resourceValue == nullptr ? nullptr : resourceValue->getIf<std::string>();
82 if (kind == nullptr || kind->empty() || resource == nullptr || resource->empty() || magnitudeValue == nullptr)
83 return requestFailure(eve::DiagnosticCode::InvalidArgument,
84 "settlement.tick requires kind, resource, and magnitude",
85 "effect.extra.settlement.tick");
86 double magnitude = 0.0;
87 if (const auto* real = magnitudeValue->getIf<double>())
88 magnitude = *real;
89 else if (const auto* integer = magnitudeValue->getIf<std::int64_t>())
90 magnitude = static_cast<double>(*integer);
91 else
92 return requestFailure(eve::DiagnosticCode::InvalidArgument, "settlement.tick magnitude must be a number",
93 "effect.extra.settlement.tick.magnitude");
94 magnitude *= static_cast<double>(tick.stacks);
95 if (!std::isfinite(magnitude) || magnitude < 0.0)
96 return requestFailure(eve::DiagnosticCode::InvalidArgument,
97 "settlement.tick magnitude must be finite and non-negative",
98 "effect.extra.settlement.tick.magnitude");
99
100 settlement::SettlementRequest request;
101 request.source = source;
102 request.target = target;
103 request.kind = *kind;
104 request.resource = *resource;
105 request.magnitude = magnitude;
106 request.tick = simulationTick;
107 request.tags = definition->tags;
108 request.tags.push_back("rpg:status-tick");
109 if (const auto* tagsValue = objectField(*object, "tags")) {
110 const auto* tags = tagsValue->getIf<eve::Value::Array>();
111 if (tags == nullptr)
112 return requestFailure(eve::DiagnosticCode::InvalidArgument, "settlement.tick tags must be an array",
113 "effect.extra.settlement.tick.tags");
114 for (std::size_t index = 0; index < tags->size(); ++index) {
115 const auto* tag = (*tags)[index].getIf<std::string>();
116 if (tag == nullptr || tag->empty())
117 return requestFailure(eve::DiagnosticCode::InvalidArgument,
118 "settlement.tick tag must be a non-empty string",
119 "effect.extra.settlement.tick.tags[" + std::to_string(index) + "]");
120 request.tags.push_back(*tag);
121 }
122 }
123 std::sort(request.tags.begin(), request.tags.end());
124 request.tags.erase(std::unique(request.tags.begin(), request.tags.end()), request.tags.end());
125 eve::Value::Object context;
126 if (const auto* contextValue = objectField(*object, "context")) {
127 const auto* configured = contextValue->getIf<eve::Value::Object>();
128 if (configured == nullptr)
129 return requestFailure(eve::DiagnosticCode::InvalidArgument, "settlement.tick context must be an object",
130 "effect.extra.settlement.tick.context");
131 context = *configured;
132 }
133 context["effect_id"] = tick.effectId;
134 context["status_instance_id"] = static_cast<std::int64_t>(tick.instanceId);
135 context["stacks"] = static_cast<std::int64_t>(tick.stacks);
136 request.context = eve::Value(std::move(context));
137 return eve::Result<settlement::SettlementRequest>::success(std::move(request));
138}

214eve::Result<void> RPGSettlementAdapter::validate(settlement::SettlementContext& context) {
215 if (!(context.request().target == targetRef_))
216 return eve::Result<void>::failure(eve::Diagnostic::error(
217 eve::DiagnosticCode::Conflict, "RPG settlement request target does not match adapter target", "target"));
218 const bool damage = isDamage(context);
219 const bool healing = isHealing(context);
220 const bool trigger = context.request().kind == "trigger";
221 if (!damage && !healing && !trigger)
222 return eve::Result<void>::failure(eve::Diagnostic::error(
223 eve::DiagnosticCode::Unsupported, "RPG settlement adapter supports damage, healing and trigger kinds only",
224 "kind"));
225
226 auto health = readBase(config_.healthAttribute);
227 const bool healthOk = health.ok();
228 if (!healthOk) {
229 const auto status = health.status();
230 return eve::Result<void>::failure(status);
231 }
232 auto maximum = readBase(config_.maxHealthAttribute);
233 const bool maximumOk = maximum.ok();
234 if (!maximumOk) {
235 const auto status = maximum.status();
236 return eve::Result<void>::failure(status);
237 }
238 if (health.value() > maximum.value())
239 return eve::Result<void>::failure(
240 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation, "RPG health exceeds max health", "health"));
241 context.setStageDetail("policy", "rpg");
242 context.setStageDetail("health_attribute", config_.healthAttribute);
243 context.setStageDetail("kind", context.request().kind);
244 (void)damage;
245 return eve::Result<void>::success();
246}

248eve::Result<void> RPGSettlementAdapter::sourceModifiers(settlement::SettlementContext& context) {
249 if (context.request().kind == "trigger") return context.setMagnitude(0.0);
250 double multiplier = 1.0;
251 if (const auto value = numberValue(contextValue(context, "source_multiplier"))) {
252 multiplier = *value;
253 } else {
254 auto sourceMultiplier = readSourceFinal(config_.sourceMultiplierAttribute, 1.0);
255 const bool sourceMultiplierOk = sourceMultiplier.ok();
256 if (!sourceMultiplierOk) {
257 const auto status = sourceMultiplier.status();
258 return eve::Result<void>::failure(status);
259 }
260 multiplier = sourceMultiplier.value();
261 }
262 if (!std::isfinite(multiplier) || multiplier < 0.0)
263 return eve::Result<void>::failure(
264 eve::Diagnostic::error(eve::DiagnosticCode::InvalidArgument,
265 "RPG source multiplier must be finite and non-negative", "source_multiplier"));
266
267 auto scaled = context.setMagnitude(context.magnitude() * multiplier);
268 const bool scaledOk = scaled.ok();
269 if (!scaledOk) {
270 const auto status = scaled.status();
271 return eve::Result<void>::failure(status);
272 }
273
274 bool critical = boolValue(contextValue(context, "critical")).value_or(false);
275 if (!critical) {
276 const auto roll = numberValue(contextValue(context, "critical_roll"));
277 auto chance = readSourceFinal(config_.criticalChanceAttribute, 0.0);
278 const bool chanceOk = chance.ok();
279 if (!chanceOk) {
280 const auto status = chance.status();
281 return eve::Result<void>::failure(status);
282 }
283 if (roll && std::isfinite(*roll)) {
284 const double probability = std::clamp(chance.value(), 0.0, 1.0);
285 critical = *roll >= 0.0 && *roll < probability;
286 }
287 }
288
289 double criticalMultiplier = 1.0;
290 if (critical) {
291 if (const auto value = numberValue(contextValue(context, "critical_multiplier"))) {
292 criticalMultiplier = *value;
293 } else {
294 auto configured = readSourceFinal(config_.criticalMultiplierAttribute, 2.0);
295 const bool configuredOk = configured.ok();
296 if (!configuredOk) {
297 const auto status = configured.status();
298 return eve::Result<void>::failure(status);
299 }
300 criticalMultiplier = configured.value();
301 }
302 if (!std::isfinite(criticalMultiplier) || criticalMultiplier < 1.0)
303 return eve::Result<void>::failure(eve::Diagnostic::error(
304 eve::DiagnosticCode::InvalidArgument, "RPG critical multiplier must be finite and at least one",
305 "critical_multiplier"));
306 auto critScaled = context.setMagnitude(context.magnitude() * criticalMultiplier);
307 const bool critScaledOk = critScaled.ok();
308 if (!critScaledOk) {
309 const auto status = critScaled.status();
310 return eve::Result<void>::failure(status);
311 }
312 }
313 context.setCritical(critical);
314 context.setStageDetail("source_multiplier", multiplier);
315 context.setStageDetail("critical", critical);
316 if (critical) context.setStageDetail("critical_multiplier", criticalMultiplier);
317 return eve::Result<void>::success();
318}

320eve::Result<void> RPGSettlementAdapter::targetMitigation(settlement::SettlementContext& context) {
321 if (!isDamage(context)) return eve::Result<void>::success();
322
323 std::string element = stringValue(contextValue(context, "element")).value_or("damage");
324 const std::string resistanceAttribute = config_.resistancePrefix + element;
325 auto resistance = readFinal(resistanceAttribute, 0.0);
326 const bool resistanceOk = resistance.ok();
327 if (!resistanceOk) {
328 const auto status = resistance.status();
329 return eve::Result<void>::failure(status);
330 }
331 const double factor = 1.0 - std::clamp(resistance.value(), 0.0, 1.0);
332 const double before = context.magnitude();
333 auto changed = context.setMagnitude(before * factor);
334 const bool changedOk = changed.ok();
335 if (!changedOk) {
336 const auto status = changed.status();
337 return eve::Result<void>::failure(status);
338 }
339 auto recorded = context.addResisted(before - context.magnitude());
340 const bool recordedOk = recorded.ok();
341 if (!recordedOk) {
342 const auto status = recorded.status();
343 return eve::Result<void>::failure(status);
344 }
345 context.setStageDetail("element", element);
346 context.setStageDetail("resistance", resistance.value());
347 return eve::Result<void>::success();
348}

350eve::Result<void> RPGSettlementAdapter::armorShield(settlement::SettlementContext& context) {
351 if (!isDamage(context)) return eve::Result<void>::success();
352
353 auto armor = readFinal(config_.armorAttribute, 0.0);
354 const bool armorOk = armor.ok();
355 if (!armorOk) {
356 const auto status = armor.status();
357 return eve::Result<void>::failure(status);
358 }
359 const double penetration = std::max(0.0, numberValue(contextValue(context, "penetration")).value_or(0.0));
360 const double effectiveArmor = std::max(0.0, armor.value() - penetration);
361 const double armorFactor = 100.0 / (100.0 + effectiveArmor);
362 const double beforeArmor = context.magnitude();
363 auto armorChanged = context.setMagnitude(beforeArmor * armorFactor);
364 const bool armorChangedOk = armorChanged.ok();
365 if (!armorChangedOk) {
366 const auto status = armorChanged.status();
367 return eve::Result<void>::failure(status);
368 }
369 auto armorLoss = context.addResisted(beforeArmor - context.magnitude());
370 const bool armorLossOk = armorLoss.ok();
371 if (!armorLossOk) {
372 const auto status = armorLoss.status();
373 return eve::Result<void>::failure(status);
374 }
375
376 auto shield = readBase(config_.shieldAttribute);
377 double shieldValue = 0.0;
378 if (shield.ok()) {
379 shieldValue = shield.value();
380 } else if (shield.code() != eve::StatusCode::NotFound) {
381 const auto status = shield.status();
382 return eve::Result<void>::failure(status);
383 }
384 const double absorbed = std::min(shieldValue, context.magnitude());
385 auto shieldChanged = context.setMagnitude(context.magnitude() - absorbed);
386 const bool shieldChangedOk = shieldChanged.ok();
387 if (!shieldChangedOk) {
388 const auto status = shieldChanged.status();
389 return eve::Result<void>::failure(status);
390 }
391 auto absorbedRecorded = context.addAbsorbed(absorbed);
392 const bool absorbedRecordedOk = absorbedRecorded.ok();
393 if (!absorbedRecordedOk) {
394 const auto status = absorbedRecorded.status();
395 return eve::Result<void>::failure(status);
396 }
397 context.setStageDetail("armor", armor.value());
398 context.setStageDetail("penetration", penetration);
399 context.setStageDetail("shield", shieldValue);
400 context.setStageDetail("absorbed", absorbed);
401 return eve::Result<void>::success();
402}

429 const settlement::SettlementContext& context) {
430 auto health = readBase(config_.healthAttribute);
431 const bool healthOk = health.ok();
432 if (!healthOk) {
433 const auto status = health.status();
434 return eve::Result<settlement::PreparedApply>::failure(status);
435 }
436 auto maximum = readBase(config_.maxHealthAttribute);
437 const bool maximumOk = maximum.ok();
438 if (!maximumOk) {
439 const auto status = maximum.status();
440 return eve::Result<settlement::PreparedApply>::failure(status);
441 }
442
443 auto candidate = target_.attributes()->values;
444 const double nextHealth =
445 isDamage(context) ? health.value() - context.magnitude() : health.value() + context.magnitude();
446 if (!std::isfinite(nextHealth) || nextHealth < 0.0 || nextHealth > maximum.value())
447 return eve::Result<settlement::PreparedApply>::failure(
448 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
449 "RPG settlement candidate health is outside its validated bounds", "health"));
450 candidate.setBase(config_.healthAttribute, nextHealth);
451
452 if (isDamage(context) && !config_.shieldAttribute.empty() && context.absorbed() > 0.0) {
453 auto shield = readBase(config_.shieldAttribute);
454 const bool shieldOk = shield.ok();
455 if (!shieldOk) {
456 const auto status = shield.status();
457 if (status.code() != eve::StatusCode::NotFound)
458 return eve::Result<settlement::PreparedApply>::failure(status);
459 } else {
460 if (context.absorbed() > shield.value())
461 return eve::Result<settlement::PreparedApply>::failure(
462 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
463 "RPG shield absorption exceeds the authoritative shield balance", "shield"));
464 candidate.setBase(config_.shieldAttribute, shield.value() - context.absorbed());
465 }

487eve::Result<std::vector<settlement::SettlementRequest>> RPGSettlementAdapter::prepareTrigger(
488 const settlement::SettlementContext& context, const settlement::SettlementResult& result) {
489 std::vector<settlement::SettlementRequest> transitions;
490 if (!isDamage(context))
491 return eve::Result<std::vector<settlement::SettlementRequest>>::success(std::move(transitions));
492
493 const auto append = [&](std::string key) {
494 settlement::SettlementRequest transition = context.request();
495 transition.kind = "trigger";
496 transition.resource.clear();
497 transition.magnitude = 0.0;
498 transition.decisions.clear();
499 transition.trigger = std::move(key);
500 transition.chain = {};
501 transition.tags.push_back("trigger:" + transition.trigger);
502 transitions.push_back(std::move(transition));
503 };
504
505 auto shield = readBase(config_.shieldAttribute);
506 if (shield.ok() && shield.value() > 0.0 && result.absorbed >= shield.value()) append("shield_break");
507 if (!shield.ok() && shield.code() != eve::StatusCode::NotFound)
508 return eve::Result<std::vector<settlement::SettlementRequest>>::failure(shield.status());
509
510 auto health = readBase(config_.healthAttribute);
511 if (!health) return eve::Result<std::vector<settlement::SettlementRequest>>::failure(health.status());
512 const bool died = config_.emitLifeTransitions && health.value() > 0.0 && result.applied >= health.value();
513 if (died) {
514 append("death");
515 if (context.request().source.isValid() && context.request().source != context.request().target) append("kill");
516 }
517 return eve::Result<std::vector<settlement::SettlementRequest>>::success(std::move(transitions));
518}
