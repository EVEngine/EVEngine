41 Result<void> prepare(const transaction::TransactionContext&) override {
42 if (prepared_ || committed_) return failure("hand stage is already in flight");
43 auto* hand = resolveHand();
44 auto* card = resolveCard();
45 if (!hand || !card) return failure("hand or card is stale");
46 const auto& values = hand->membership()->cards;
47 if (std::count(values.begin(), values.end(), card) != 1)
48 return failure("card membership is absent or duplicated");
49 before_ = values;
50 dragBefore_ = *hand->drag();
51 prepared_ = true;
52 return Result<void>::success();
53 }
54 Result<void> commit(const transaction::TransactionContext&) override {
55 if (!prepared_ || committed_) return failure("hand stage is not prepared");
56 auto* hand = resolveHand();
57 auto* card = resolveCard();
58 if (!hand || !card) return failure("hand or card is stale");
59 if (hand->membership()->cards != before_) return failure("hand changed before commit");
60 auto& values = hand->membership()->cards;
61 values.erase(std::find(values.begin(), values.end(), card));
62 if (hand->drag()->card == card) hand->drag()->card = nullptr;
63 if (hand->drag()->press == card) hand->drag()->press = nullptr;
64 committed_ = true;
65 return Result<void>::success();
66 }
67 Result<void> rollback(const transaction::TransactionContext&) override {
68 if (!prepared_ || committed_) return failure("hand stage is not prepared");
69 prepared_ = false;
70 return Result<void>::success();
71 }
72 Result<void> compensate(const transaction::TransactionContext&) override {
73 if (!committed_) return failure("hand stage is not committed");
74 auto* hand = resolveHand();
75 auto* card = resolveCard();
76 if (!hand || !card) return failure("hand or card is stale");
77 auto expected = before_;
78 expected.erase(std::find(expected.begin(), expected.end(), card));
79 if (hand->membership()->cards != expected) return failure("hand changed before compensation");
80 hand->membership()->cards.swap(before_);
81 *hand->drag() = dragBefore_;
82 committed_ = false;
83 prepared_ = false;
84 return Result<void>::success();
85 }
