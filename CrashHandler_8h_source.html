37inline LONG WINAPI crashHandler(EXCEPTION_POINTERS *ep) {
38 // DbgHelp must be initialized before StackWalk64, otherwise the walk
39 // produces garbage frames. Ignore the "already initialized" failure.
40 SymInitialize(GetCurrentProcess(), nullptr, TRUE);
41 std::ostringstream report;
42 report << "[crash] code=0x" << std::hex << std::setw(8) << std::setfill('0')
43 << ep->ExceptionRecord->ExceptionCode << std::dec << std::setfill(' ')
44 << " at " << ep->ExceptionRecord->ExceptionAddress << "\n";
45 try {
46 backward::StackTrace st;
47 // Walk from the handler's own frame: the exception dispatch ran on the
48 // crashing thread's stack, so the crash site is still in the chain.
49 // Walking from ep->ContextRecord produced garbage frames (0xCC) with
50 // DbgHelp StackWalk64 on this setup.
51 st.load_here(64);
52 backward::Printer p;
53 p.snippet = false;
54 p.color_mode = backward::ColorMode::never;
55 p.print(st, report);
56 } catch (...) {
57 report << "[crash] backtrace unavailable\n";
58 }
59 const std::string text = report.str();
60 std::fputs(text.c_str(), stderr);
61 std::fflush(stderr);
62 eve::recordCrashEvent(text);
63 return EXCEPTION_CONTINUE_SEARCH;
64}
65
67inline void installCrashHandler() {
68 SetUnhandledExceptionFilter(&crashHandler);
69}
