61void VulkanSequence::destroy() {
62 if (status == SequenceStatus::Submitted) {
63 try {
64 (void)vulkanSequenceWait(this);
65 } catch (...) {
66 }
67 }
68 auto *dev = vkg ? &vkg->getDevice() : nullptr;
69 if (dev && static_cast<VkDevice>(dev->instance)) {
70 if (commandBuffer) {
71 (*dev)->freeCommandBuffers(pool, commandBuffer);
72 commandBuffer = vk::CommandBuffer{};

105void vulkanSequenceRecordUpload(VulkanSequence *seq, GpuBuffer *dst,
106 const void *src, uint64_t nbytes,
107 uint64_t dstOffset) {
108 if (!seq || !dst || !src || nbytes == 0) return;
109 seq->ensureCommandBuffer();
110 auto *vkbDst = dynamic_cast<VulkanGpuBuffer *>(dst);
111 if (!vkbDst || !vkbDst->buffer_)
112 throw Exception("Gpgpu.Sequence.recordUpload: destination is not a Vulkan storage buffer");
113 if (dstOffset + nbytes > uint64_t(vkbDst->size_))
114 throw Exception("Gpgpu.Sequence.recordUpload: out of range");
115
116 auto &device = seq->vkg->getDevice();
117 vkb::GenericBuffer *staging = nullptr;
118 for (size_t i = seq->stagingUsed; i < seq->stagingPool.size(); ++i) {
119 if (seq->stagingPool[i].capacity >= vk::DeviceSize(nbytes)) {
120 staging = &seq->stagingPool[i];
121 std::swap(seq->stagingPool[seq->stagingUsed], seq->stagingPool[i]);
122 break;
123 }
124 }
125 if (!staging) {
126 seq->stagingPool.emplace_back(device, buf::eTransferSrc, vk::DeviceSize(nbytes),
127 pfb::eHostVisible | pfb::eHostCoherent);
128 staging = &seq->stagingPool.back();
129 }
130 ++seq->stagingUsed;
131 staging->updateLocal(vkb::FrameSlot::gpuIdle(), src, vk::DeviceSize(nbytes));
132
133 vk::BufferCopy bc{0, vk::DeviceSize(dstOffset), vk::DeviceSize(nbytes)};
134 seq->commandBuffer.copyBuffer(staging->buffer, vkbDst->buffer_, bc);
135 fullMemoryBarrier(seq);
136}

138void vulkanSequenceRecordDownload(VulkanSequence *seq, GpuBuffer *src,
139 GpuBuffer *staging, uint64_t nbytes,
140 uint64_t srcOffset) {
141 if (!seq || !src || !staging || nbytes == 0) return;
142 seq->ensureCommandBuffer();
143 auto *vkbSrc = dynamic_cast<VulkanGpuBuffer *>(src);
144 auto *vkbStaging = dynamic_cast<VulkanGpuBuffer *>(staging);
145 if (!vkbSrc || !vkbSrc->buffer_)
146 throw Exception("Gpgpu.Sequence.recordDownload: source is not a Vulkan storage buffer");
147 if (!vkbStaging || !vkbStaging->buffer_ || !vkbStaging->hostVisible_)
148 throw Exception("Gpgpu.Sequence.recordDownload: target must be a \"staging\" buffer");
149 if (srcOffset + nbytes > uint64_t(vkbSrc->size_))
150 throw Exception("Gpgpu.Sequence.recordDownload: out of range");
151 if (nbytes > uint64_t(vkbStaging->size_))
152 throw Exception("Gpgpu.Sequence.recordDownload: staging buffer too small");
153
154 vk::BufferCopy bc{vk::DeviceSize(srcOffset), 0, vk::DeviceSize(nbytes)};
155 seq->commandBuffer.copyBuffer(vkbSrc->buffer_, vkbStaging->buffer_, bc);
156 fullMemoryBarrier(seq);
157}

159void vulkanSequenceRecordDispatch(VulkanSequence *seq, ComputeShader *shader,
160 int groupsX, int groupsY, int groupsZ) {
161 if (!seq || !shader) return;
162 seq->ensureCommandBuffer();
163 auto *vs = dynamic_cast<VulkanComputeShader *>(shader);
164 if (!vs || !vs->pipeline_) return;
165 if (groupsX <= 0) groupsX = 1;
166 if (groupsY <= 0) groupsY = 1;
167 if (groupsZ <= 0) groupsZ = 1;
168
169 auto &device = seq->vkg->getDevice();
170 vs->flushDescriptors(device);
171 if (std::find(seq->usedShaders.begin(), seq->usedShaders.end(), vs) == seq->usedShaders.end()) {
172 vs->beginSequence();
173 seq->usedShaders.push_back(vs);
174 }
175
176 seq->commandBuffer.bindPipeline(vk::PipelineBindPoint::eCompute, vs->pipeline_);
177 if (vs->descriptorSet_) {
178 seq->commandBuffer.bindDescriptorSets(vk::PipelineBindPoint::eCompute,
179 vs->pipelineLayout_, 0, vs->descriptorSet_,
180 nullptr);
181 }
182 seq->commandBuffer.pushConstants(vs->pipelineLayout_, vk::ShaderStageFlagBits::eCompute,
183 0, ComputeShader::kPushConstantBytes,
184 vs->pushConstantData());
185 seq->commandBuffer.dispatch(uint32_t(groupsX), uint32_t(groupsY), uint32_t(groupsZ));
186 fullMemoryBarrier(seq);
187}

195SequenceStatus vulkanSequenceSubmitAsync(VulkanSequence *seq) {
196 if (!seq) return SequenceStatus::Failed;
197 seq->ensureReady();
198 if (!seq->recording || !seq->commandBuffer)
199 throw Exception("Gpgpu.Sequence.submit: nothing recorded");
200 auto &device = seq->vkg->getDevice();
201
202 vk::CommandBuffer cb = seq->commandBuffer;
203 seq->commandBuffer = vk::CommandBuffer{};
204 seq->recording = false;
205 cb.end();
206
207 device->resetFences(seq->fence);
208 vk::SubmitInfo submit;
209 submit.commandBufferCount = 1;
210 submit.pCommandBuffers = &cb;
211 seq->queue.submit(submit, seq->fence);
212 seq->submittedCommandBuffer = cb;
213 seq->status = SequenceStatus::Submitted;
214 return seq->status;
215}

219SequenceStatus retireSubmission(VulkanSequence *seq, SequenceStatus terminal) {
220 auto &device = seq->vkg->getDevice();
221 for (ComputeShader *shader : seq->usedShaders) {
222 auto *vs = dynamic_cast<VulkanComputeShader *>(shader);
223 if (vs) {
224 vs->endSequence();
225 vs->releasePendingDescriptors(device);
226 }
227 }
228 seq->usedShaders.clear();
229 if (seq->submittedCommandBuffer) {
230 device->freeCommandBuffers(seq->pool, seq->submittedCommandBuffer);
231 seq->submittedCommandBuffer = vk::CommandBuffer{};
232 }
233 seq->status = terminal;
234 return terminal;
235}

239SequenceStatus vulkanSequencePoll(VulkanSequence *seq) {
240 if (!seq) return SequenceStatus::Failed;
241 if (seq->status != SequenceStatus::Submitted) return seq->status;
242 const vk::Result result = seq->vkg->getDevice()->getFenceStatus(seq->fence);
243 if (result == vk::Result::eNotReady) return SequenceStatus::Submitted;
244 return retireSubmission(seq, result == vk::Result::eSuccess ? SequenceStatus::Complete : SequenceStatus::Failed);
245}

247SequenceStatus vulkanSequenceWait(VulkanSequence *seq) {
248 if (!seq) return SequenceStatus::Failed;
249 if (seq->status != SequenceStatus::Submitted) return seq->status;
250 const vk::Result result = seq->vkg->getDevice()->waitForFences(1, &seq->fence, VK_TRUE, UINT64_MAX);
251 return retireSubmission(seq, result == vk::Result::eSuccess ? SequenceStatus::Complete : SequenceStatus::Failed);
252}
