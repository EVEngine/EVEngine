80bool mcpResolveTaskProfile(std::string_view taskKind, std::string_view profile, std::string* outProfile,
81 std::string* outTaskKind, std::string* errorCode, std::string* errorMessage) {
82 std::string kind = std::string(taskKind);
83 std::string prof = std::string(profile);
84 if (kind.empty() && prof.empty()) {
85 kind = "verify";
86 prof = "verify.play";
87 } else if (kind.empty()) {
88 if (!isKnownProfile(prof)) {
89 if (errorCode) *errorCode = "invalid-profile";
90 if (errorMessage) *errorMessage = "profile must be a known task profile id";
91 return false;
92 }
93 kind = taskKindForProfile(prof);
94 } else if (prof.empty()) {
95 if (!isKnownTaskKind(kind)) {
96 if (errorCode) *errorCode = "invalid-task-kind";
97 if (errorMessage) *errorMessage = "taskKind must be content, script, verify, fix, or workshop";
98 return false;
99 }
100 prof = defaultProfileForKind(kind);
101 } else {
102 if (!isKnownTaskKind(kind)) {
103 if (errorCode) *errorCode = "invalid-task-kind";
104 if (errorMessage) *errorMessage = "taskKind must be content, script, verify, fix, or workshop";
105 return false;
106 }
107 if (!isKnownProfile(prof)) {
108 if (errorCode) *errorCode = "invalid-profile";
109 if (errorMessage) *errorMessage = "profile must be a known task profile id";
110 return false;
111 }
112 if (taskKindForProfile(prof) != kind) {
113 if (errorCode) *errorCode = "profile-task-mismatch";
114 if (errorMessage) *errorMessage = "profile does not belong to the given taskKind";
115 return false;
116 }
117 }
118 if (outProfile) *outProfile = std::move(prof);
119 if (outTaskKind) *outTaskKind = std::move(kind);
120 return true;
121}

143bool mcpProfileAllowsTool(std::string_view profile, std::string_view tool) {
144 if (profile.empty()) return true;
145 if (!isKnownProfile(profile)) return false;
146 // Session / agent tools stay reachable so the Agent can escalate, status, and complete.
147 if (isAgentSessionTool(tool) || tool == "eve_status" || tool == "eve_game_world") return true;
148
149 if (isDangerEvalTool(tool)) {
150 // Only the fix profile may touch eval/run_script; still not valid required evidence.
151 return profile == "fix.compile";
152 }
153 // Stable facade: never leak into content/verify via shared scene/play tags.
154 if (tool == "eve_workshop") return profile == "workshop.rpg";
155
156 const auto tagList = mcpProfileToolsetTags(profile);
157 std::unordered_set<std::string> tags(tagList.begin(), tagList.end());
158
159 if (profile == "script.bind") {
160 if (equalsAny(tool, {"eve_api_search", "eve_api_get", "eve_console_read", "eve_console_write",
161 "eve_console_clear", "eve_error_slice", "eve_crash_report"}))
162 return true;
163 // Keep debug observations, but not breakpoints / snapshots / eval (eval already rejected).
164 if (equalsAny(tool, {"eve_stack", "eve_locals", "eve_list_breakpoints"})) return true;
165 return false;
166 }
167
168 if (profile == "verify.play") {
169 if (equalsAny(tool, {"eve_screenshot", "eve_screenshot_image", "eve_render_status", "eve_console_read"}))
170 return true;
171 return tagsAllow(tool, tags);
172 }
173
174 if (profile == "fix.compile") {
175 if (isCoreDiscoverTool(tool) || tool == "eve_restart" || tool == "eve_error_slice" ||
176 tool == "eve_api_search" || tool == "eve_api_get")
177 return true;
178 return tagsAllow(tool, tags);
179 }
180
181 if (profile == "content.scene") {
182 if (startsWith(tool, "eve_procgen_") || startsWith(tool, "eve_recipe_") || tool == "eve_scene_info" ||
183 tool == "eve_scene_director_status")
184 return true;
185 return tagsAllow(tool, tags);
186 }
187
188 if (profile == "content.map") {
189 if (tool == "eve_console_read") return true;
190 return tagsAllow(tool, tags);
191 }
192
193 if (profile == "workshop.rpg") {
194 if (tool == "eve_workshop" || tool == "eve_screenshot" || tool == "eve_screenshot_image" ||
195 tool == "eve_console_read" || tool == "eve_api_search" || tool == "eve_api_get")
196 return true;
197 return tagsAllow(tool, tags);
198 }
199
200 return tagsAllow(tool, tags);
201}

203std::optional<McpProfileEscalate> mcpProfileEscalateHint(std::string_view profile, std::string_view tool) {
204 if (profile.empty() || mcpProfileAllowsTool(profile, tool)) return std::nullopt;
205
206 McpProfileEscalate hint;
207 hint.errorCode = "profile-denied";
208
209 if (isDangerEvalTool(tool) && profile != "fix.compile") {
210 hint.to = "fix.compile";
211 hint.reason = "eval and run_script are limited to fix.compile and cannot satisfy required criteria";
212 return hint;
213 }
214 if ((startsWith(tool, "eve_scene_") || startsWith(tool, "eve_camera_") || startsWith(tool, "eve_procgen_") ||
215 startsWith(tool, "eve_recipe_") || tool == "eve_whitebox_playtest") &&
216 (profile == "verify.play" || profile == "script.bind" || profile == "fix.compile")) {
217 hint.to = "content.scene";
218 hint.reason = "scene, recipe, or procgen mutation requires content.scene";
219 return hint;
220 }
221 if (tool == "eve_workshop" && profile != "workshop.rpg") {
222 hint.to = "workshop.rpg";
223 hint.reason = "workshop facade is limited to workshop.rpg";
224 return hint;
225 }
226 if ((startsWith(tool, "eve_editor_") || startsWith(tool, "eve_host_")) &&
227 (profile == "verify.play" || profile == "script.bind" || profile == "content.scene")) {
228 hint.to = profile == "content.scene" ? "content.map" : "workshop.rpg";
229 hint.reason = "editor/host tools require content.map or workshop.rpg";
230 return hint;
231 }
232 if ((tool == "eve_play" || tool == "eve_gameplay" || startsWith(tool, "eve_input_") ||
233 startsWith(tool, "eve_ui_")) &&
234 (profile == "content.scene" || profile == "content.map" || profile == "script.bind" ||
235 profile == "fix.compile")) {
236 hint.to = "verify.play";
237 hint.reason = "play/observe verification requires verify.play";
238 return hint;
239 }
240 if ((tool == "eve_api_search" || tool == "eve_api_get") &&
241 (profile == "content.scene" || profile == "content.map" || profile == "verify.play")) {
242 hint.to = "script.bind";
243 hint.reason = "binding-contract lookup is opened by script.bind";
244 return hint;
245 }
246 if ((tool == "eve_restart" || tool == "eve_error_slice") && profile != "fix.compile") {
247 hint.to = "fix.compile";
248 hint.reason = "compile/runtime repair tools require fix.compile";
249 return hint;
250 }
251 return std::nullopt;
252}
