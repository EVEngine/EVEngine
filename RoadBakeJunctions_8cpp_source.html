421 const auto walkRing = offsetRing(walkDistances);
422 for (std::size_t i = 0; i < rim.size(); ++i) {
423 const std::size_t next = (i + 1) % rim.size();
424 if (isMouth(rim[i], rim[next])) continue;
425 const V3 outward = edgeOutward(i);
426 const V3 upA = rimUps[i];
427 const V3 upB = rimUps[next];
428 const auto& styleA = rimStyles[i];
429 const auto& styleB = rimStyles[next];
430 const V3 innerA{rim[i].x, rim[i].y, rim[i].z};
431 const V3 innerB{rim[next].x, rim[next].y, rim[next].z};
432 const V3 surfaceA = innerA;
433 const V3 surfaceB = innerB;
434 const V3 innerTopA = surfaceA + upA * styleA.curbHeight;
435 const V3 innerTopB = surfaceB + upB * styleB.curbHeight;
436 const V3 curbTopA = curbRing[i] + upA * styleA.curbHeight;
437 const V3 curbTopB = curbRing[next] + upB * styleB.curbHeight;
438 const V3 curbWalkA = curbRing[i] + upA * styleA.sidewalkHeight;
439 const V3 curbWalkB = curbRing[next] + upB * styleB.sidewalkHeight;
440 const V3 walkA = walkRing[i] + upA * styleA.sidewalkHeight;
441 const V3 walkB = walkRing[next] + upB * styleB.sidewalkHeight;
442 if (styleA.curbWidth > 1e-4f || styleB.curbWidth > 1e-4f) {
443 appendOrientedQuad(mesh, innerA, innerB, innerTopB, innerTopA, outward * -1.f, 0.f, 1.f, 0.f, 1.f,
444 RoadMaterial::Curb);
445 appendOrientedQuad(mesh, innerTopA, innerTopB, curbTopB, curbTopA, upN, 0.f, 1.f, 0.f, 1.f,
446 RoadMaterial::Curb);
447 appendOrientedQuad(mesh, curbWalkA, curbWalkB, curbTopB, curbTopA, outward, 0.f, 1.f, 0.f, 1.f,
448 RoadMaterial::Curb);
449 }
450 if (styleA.sidewalkWidth > 1e-4f || styleB.sidewalkWidth > 1e-4f) {
451 appendOrientedQuad(mesh, curbWalkA, curbWalkB, walkB, walkA, upN, 0.f, 1.f, 0.f, 1.f,
452 RoadMaterial::Sidewalk);
453 appendOrientedQuad(mesh, walkRing[i], walkRing[next], walkB, walkA, outward, 0.f, 1.f, 0.f, 1.f,
454 RoadMaterial::Sidewalk);
455 }
456 }
457 }
458
459 if (!options.includeMarkings || arms.size() == 2u) return Result<void>::success();
460
461 // Zebra stripes on every arm tip that meets this hub.
462 for (const auto& edge : network.edges()) {
463 if (edge.to != node.id && edge.from != node.id) continue;
464 auto spline = edgeToSpline(edge);
465 if (!spline.ok()) continue;
466 auto len = spline.value().lengthResult(16);
467 if (!len.ok()) continue;
468 const float trim = junctionSocketDistance(network, node, edge, len.value());
469 if (len.value() < trim + 1.5f) continue;
470 const float alongDist = edge.to == node.id ? (len.value() - trim - 1.6f) : (trim + 1.6f);
471 auto frame = spline.value().travelFrameResult(alongDist, "clamp", 16);
472 if (!frame.ok()) continue;
473 const auto& f = frame.value();
474 const V3 origin{f.sample.x, f.sample.y, f.sample.z};
475 const V3 side{f.sideX, f.sideY, f.sideZ};
476 // Stripe direction: toward the hub from this tip.
477 V3 fwd{f.forwardX, f.forwardY, f.forwardZ};
478 if (edge.from == node.id) fwd = fwd * -1.f;
479 const V3 nrm = normalize(V3{f.upX, f.upY, f.upZ});
480 const float stripeW = edge.style.laneWidth * static_cast<float>(edge.lanesForward + edge.lanesBackward) *
481 frame.value().sample.scaleX * 0.45f;
482 for (int s = 0; s < 4; ++s) {
483 const float along = static_cast<float>(s) * 0.5f;
484 const V3 c = origin - fwd * along + nrm * 0.04f;
485 const V3 a = c + side * -stripeW;
486 const V3 b = c + side * stripeW;
487 const V3 d = a - fwd * 0.25f;
488 const V3 e = b - fwd * 0.25f;
489 appendOrientedQuad(mesh, a, b, e, d, nrm, 0.f, 1.f, 0.f, 1.f, RoadMaterial::Marking);
490 }
491 }
492 return Result<void>::success();
