54audio_editing::Result<void> AudioSourceEditor::requestPreview() {
55 auto provider = previewSession_->provider;
56 if (!provider)
57 return editing::failed<void>(editing::Status::Unsupported, editing::RuleId("editor.audio.preview-provider"),
58 "Waveform provider is unavailable");
59 audio_editing::AudioWaveformRequest request;
60 request.asset = "asset://preview/tone.sine";
61 request.sourceRevision = auditionRevision();
62 request.pixelWidth = static_cast<int>(std::lround(requestedViewportWidth_));
63 auto readLoop = [this](const char* path) -> editing::Result<double> {
64 const auto property = read(path);
65 if (const auto* value = property.value.getIf<double>()) return editing::applied(*value);
66 if (const auto* value = property.value.getIf<std::int64_t>())
67 return editing::applied(static_cast<double>(*value));
68 return editing::rejected<double>(editing::RuleId("editor.audio.preview-loop"), "Loop property is not numeric");
69 };
70 auto start = readLoop("play.loop-start");
71 if (!start.ok()) return editing::Result<void>::failure(start.status());
72 auto end = readLoop("play.loop-end");
73 if (!end.ok()) return editing::Result<void>::failure(end.status());
74 request.loopStart = start.value();
75 request.loopEnd = end.value();
76 auto pcm = pcm_;
77 const auto revision = target_.revision();
78 const auto width = requestedViewportWidth_;
79 return previewSession_->async.request([provider = std::move(provider), request = std::move(request),
80 pcm = std::move(pcm), revision,
81 width](const editing::TaskContext& context) -> editing::Result<Publication> {
82 auto result = provider->generate(request, pcm, context);
83 if (!result.ok()) return editing::Result<Publication>::failure(result.status());
84 return editing::applied(Publication{std::move(result).takeValue(), revision, width});
85 });
86}
87audio_editing::Result<void> AudioSourceEditor::pollPreview() {
88 auto result = previewSession_->async.poll();
89 if (!result) return previewState().phase == editor::PreviewPhase::Pending ? editing::pending() : editing::noOp();
90 if (!result->ok()) return editing::Result<void>::failure(result->status());
91 auto publication = std::move(*result).takeValue();
92 if (publication.revision != revision() || publication.waveform.sourceRevision != auditionRevision() ||
93 publication.width != requestedViewportWidth_ || publication.waveform.asset != "asset://preview/tone.sine" ||
94 publication.waveform.status != editing::Status::Applied) {
95 previewSession_->async.rejectPublication();
96 return editing::failed<void>(editing::Status::Conflict, editing::RuleId("editor.audio.stale-preview"),
97 "Waveform inputs changed before publication");
98 }
99 waveform_ = std::move(publication.waveform);
100 viewportWidth_ = publication.width;
101 return editing::applied<void>();
102}
