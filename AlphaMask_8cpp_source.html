16AlphaMask::AlphaMask(Graphics *graphics) : graphics_(graphics) {
17 if (!graphics_) throw eve::Exception("AlphaMask: null graphics");
18 if (graphics_->getBackendName() == "webgpu") {
19 static const std::string fragment = R"(
20struct Externals { data: array<f32, 32>, };
21@group(0) @binding(0) var mainTex: texture_2d<f32>;
22@group(0) @binding(1) var maskTex: texture_2d<f32>;
23@group(0) @binding(2) var mainSampler: sampler;
24@group(0) @binding(3) var maskSampler: sampler;
25@group(0) @binding(4) var<uniform> ext: Externals;
26struct In { @location(0) color: vec4<f32>, @location(1) uv: vec2<f32>, };
27@fragment fn fs_main(input: In) -> @location(0) vec4<f32> {
28 let color = textureSample(mainTex, mainSampler, input.uv) * input.color;
29 let ms = textureSample(maskTex, maskSampler, input.uv);
30 var m = ms.r;
31 if (ext.data[2] > 0.5) { m = 1.0 - m; }
32 let width = max(ext.data[1], 0.0001);
33 let coverage = smoothstep(ext.data[0] - width, ext.data[0] + width, m);
34 return vec4<f32>(color.rgb, color.a * coverage);
35})";
36 shader_ = graphics_->newShaderFromWgsl({}, fragment);
37 } else {
38 std::vector<uint32_t> fragment(alpha_mask_frag_spv,
39 alpha_mask_frag_spv + alpha_mask_frag_spv_count);
40 shader_ = graphics_->newShaderFromSpv({}, fragment);
41 }
42 if (!shader_) throw eve::Exception("AlphaMask: shader creation failed");
43 shader_->declareFloat("threshold");
44 shader_->declareFloat("softness");
45 shader_->declareFloat("inverted");
46 syncUniforms();
47}
