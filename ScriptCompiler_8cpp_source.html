59std::string maskComments(std::string_view source) {
60 std::string result(source);
61 char quote = 0;
62 bool lineComment = false;
63 bool blockComment = false;
64 for (size_t i = 0; i < result.size(); ++i) {
65 const char c = result[i];
66 const char next = i + 1 < result.size() ? result[i + 1] : 0;
67 if (lineComment) {
68 if (c == '\n')
69 lineComment = false;
70 else
71 result[i] = ' ';
72 } else if (blockComment) {
73 if (c == '*' && next == '/') {
74 result[i] = result[i + 1] = ' ';
75 ++i;
76 blockComment = false;
77 } else if (c != '\n') {
78 result[i] = ' ';
79 }
80 } else if (quote != 0) {
81 if (c == '\\')
82 ++i;
83 else if (c == quote)
84 quote = 0;
85 } else if (c == '"' || c == '\'') {
86 quote = c;
87 } else if (c == '/' && next == '/') {
88 result[i] = result[i + 1] = ' ';
89 ++i;
90 lineComment = true;
91 } else if (c == '/' && next == '*') {
92 result[i] = result[i + 1] = ' ';
93 ++i;
94 blockComment = true;
95 }
96 }
97 return result;
98}

115std::vector<std::string> splitArguments(std::string_view arguments) {
116 std::vector<std::string> result;
117 size_t begin = 0;
118 char quote = 0;
119 for (size_t i = 0; i <= arguments.size(); ++i) {
120 const char c = i == arguments.size() ? ',' : arguments[i];
121 if ((c == '"' || c == '\'') && (i == 0 || arguments[i - 1] != '\\'))
122 quote = quote == 0 ? c : (quote == c ? 0 : quote);
123 if (c == ',' && quote == 0) {
124 result.push_back(trim(arguments.substr(begin, i - begin)));
125 begin = i + 1;
126 }
127 }
128 if (result.size() == 1 && result.front().empty()) result.clear();
129 return result;
130}

206bool sourceIdentityMatches(std::string_view registered, std::string_view requested) {
207 std::string left(registered);
208 std::string right(requested);
209 std::replace(left.begin(), left.end(), '\\', '/');
210 std::replace(right.begin(), right.end(), '\\', '/');
211 if (left == right) return true;
212 const size_t scheme = left.find(":/");
213 if (scheme != std::string::npos) left.erase(0, scheme + 2);
214 while (!left.empty() && left.front() == '/') left.erase(left.begin());
215 return right.size() >= left.size() && right.compare(right.size() - left.size(), left.size(), left) == 0 &&
216 (right.size() == left.size() || right[right.size() - left.size() - 1] == '/');
217}

311 bindings_ = generatedBindingContracts();
312 const SQInteger top = sq_gettop(vm_->getHandle());
313 sq_pushroottable(vm_->getHandle());
314 sq_pushnull(vm_->getHandle());
315 while (SQ_SUCCEEDED(sq_next(vm_->getHandle(), -2))) {
316 if (sq_gettype(vm_->getHandle(), -2) == OT_STRING) {
317 const SQChar* name = nullptr;
318 if (SQ_SUCCEEDED(sq_getstring(vm_->getHandle(), -2, &name)) && name != nullptr) nativeGlobals_.insert(name);
319 }
320 sq_pop(vm_->getHandle(), 2);
321 }
322 sq_settop(vm_->getHandle(), top);
323 {
324 std::lock_guard lock(compilerRegistryMutex);
325 compilerRegistry[vm_->getHandle()] = this;
326 }
327 sq_setnamedargresolver(
328 vm_->getHandle(),
329 [](HSQUIRRELVM, const SQChar* callee, SQInteger index, const SQChar** type, SQBool* nullable,
330 const SQChar** unit, const SQChar** choices, SQUserPointer user) -> const SQChar* {
331 const auto& self = *static_cast<ScriptCompiler*>(user);
332 if (self.nativeGlobals_.find(callee) != self.nativeGlobals_.end()) return nullptr;
333 const BindingContract* contract = self.bindings_.findMethod(callee);
334 if (contract == nullptr || index < 0 || static_cast<size_t>(index) >= contract->parameters.size())
335 return nullptr;
336 const BindingParameterContract& parameter = contract->parameters[static_cast<size_t>(index)];
337 if (type != nullptr) *type = parameter.type.c_str();
338 if (nullable != nullptr) *nullable = parameter.nullable ? SQTrue : SQFalse;
339 static const char* units[] = {nullptr, "seconds", "milliseconds", "radians", "degrees", "pixels", "meters"};
340 if (unit != nullptr) *unit = units[static_cast<size_t>(parameter.unit)];
341 if (choices != nullptr) {
342 static thread_local std::string choiceBuffer;
343 choiceBuffer.clear();
344 for (size_t i = 0; i < parameter.choices.size(); ++i) {
345 if (i != 0) choiceBuffer += ',';
346 choiceBuffer += parameter.choices[i];

348 *choices = choiceBuffer.empty() ? nullptr : choiceBuffer.c_str();
349 }
350 return parameter.name.c_str();
351 },
352 this);
353 sq_setannotationresolver(
354 vm_->getHandle(),
355 [](HSQUIRRELVM, const SQChar* annotation, SQUserPointer user) -> SQBool {
356 const auto& names = static_cast<ScriptCompiler*>(user)->annotations_;
357 return names.find(annotation) == names.end() ? SQFalse : SQTrue;
358 },
359 this);

372 const std::string uri(sourceName);
373 modules_->beginCompilation(uri);
374 ScriptMetadata next = analyze(source, uri);

377 modules_->prepareDependencies(uri);

386 metadata_[uri] = std::move(next);
387 throw std::runtime_error(*failure);
388 }
389 metadata_[uri] = std::move(next);
390 throw;
391 }

461std::vector<ScriptCompletion> ScriptCompiler::completions(std::string_view canonicalUri,
462 std::string_view prefix) const {
463 std::vector<ScriptCompletion> result;
464 const auto matches = [prefix](std::string_view value) { return value.rfind(prefix, 0) == 0; };
465 if (const auto unit = metadata(canonicalUri)) {
466 for (const ScriptSymbolMetadata& symbol : unit->get().symbols) {
467 if (matches(symbol.name)) result.push_back({symbol.name, symbol.kind, symbol.erasedType, symbol.name});
468 }
469 for (const std::string& name : unit->get().exports) {
470 if (matches(name)) result.push_back({name, "export", "module export", name});
471 }
472 }
473 for (const BindingContract& contract : bindings_.snapshot()) {
474 if (!matches(contract.method)) continue;
475 std::string signature = contract.method + "(";
476 std::string insertion = contract.method + "(";
477 for (size_t i = 0; i < contract.parameters.size(); ++i) {
478 if (i != 0) {
479 signature += ", ";
480 insertion += ", ";
481 }
482 signature += contract.parameters[i].name + ": " + contract.parameters[i].type;
483 insertion += contract.parameters[i].name + ": ";
484 }
485 signature += ") -> " + contract.returnType;
486 insertion += ')';
487 result.push_back({contract.method, "function", std::move(signature), std::move(insertion)});
488 }
489 std::sort(result.begin(), result.end(), [](const ScriptCompletion& a, const ScriptCompletion& b) {
490 if (a.label != b.label) return a.label < b.label;
491 return a.kind < b.kind;
492 });
493 result.erase(std::unique(result.begin(), result.end(),
494 [](const ScriptCompletion& a, const ScriptCompletion& b) {
495 return a.label == b.label && a.kind == b.kind;
496 }),
497 result.end());
498 return result;
499}

548SQRESULT ScriptCompiler::compileBuffer(HSQUIRRELVM vm, const SQChar* source, SQInteger size, const SQChar* sourceName,
549 SQBool raiseError) {
550 ScriptCompiler* compiler = nullptr;
551 {
552 std::lock_guard lock(compilerRegistryMutex);
553 const auto found = compilerRegistry.find(vm);
554 if (found != compilerRegistry.end()) compiler = found->second;
555 }
556 if (compiler == nullptr) return sq_compilebuffer(vm, source, size, sourceName, raiseError);
557
558 const std::string_view text(source, static_cast<size_t>(size));
559 const std::string uri = sourceName == nullptr ? "buffer" : sourceName;
560 ScriptMetadata next = analyze(text, uri);
561 compiler->modules_->beginCompilation(uri);
562 const SQInteger top = sq_gettop(vm);
563 const SQRESULT result = sq_compilebuffer(vm, source, size, sourceName, raiseError);
564 if (SQ_FAILED(result)) {
565 const ScriptErrorContext context = captureCompileError(vm);
566 ScriptDiagnostic diagnostic;
567 diagnostic.code = diagnosticCode(context.message);
568 diagnostic.message = context.message.empty() ? "script compilation failed" : context.message;
569 diagnostic.canonicalUri = context.source.empty() ? uri : context.source;
570 diagnostic.position.line = static_cast<uint32_t>(std::max(context.line, 1));
571 diagnostic.position.column = static_cast<uint32_t>(std::max(context.column, 1));
572 next.diagnostics.push_back(std::move(diagnostic));
573 compiler->metadata_[uri] = std::move(next);
574 return result;
575 }
576 try {
577 compiler->modules_->prepareDependencies(uri);
578 } catch (const std::exception& error) {
579 sq_settop(vm, top);
580 next.diagnostics.push_back(makeDiagnostic(error.what(), uri));
581 compiler->metadata_[uri] = std::move(next);
582 return sq_throwerror(vm, error.what());
583 }
584 next.dependencies = compiler->modules_->dependencies(uri);
585 compiler->metadata_[uri] = std::move(next);
586 compiler->refreshDependencyMetadata();
587 return SQ_OK;
588}

593 const std::vector<std::string>& activeModules) {

596 static const std::regex listPattern(R"(\b(modules|optionalModules)\s*=\s*\[([^\]]*)\])");
597 static const std::regex valuePattern(R"([\"']([^\"']+)[\"'])");

604 const bool active = std::find(activeModules.begin(), activeModules.end(), name) != activeModules.end();

608 diagnostic.message = known ? "required module is absent from the selected profile: " + name

611 diagnostic.position = positionAt(source, listOffset + static_cast<size_t>(list.position(2)) + valueOffset);
612 diagnostic.related = name;
613 diagnostic.fix = known ? "enable the module in the selected build profile"
614 : "use a root slot from the generated Module Contract";
615 result.push_back(std::move(diagnostic));
616 });
617 });
618 return result;

621ScriptMetadata ScriptCompiler::analyze(std::string_view source, std::string_view canonicalUri) {
622 ScriptMetadata result;
623 result.sourceHash = hashSource(source);
624 result.canonicalUri = std::string(canonicalUri);
625 result.providerOrigin = std::string(canonicalUri);
626 result.sourceMap.canonicalUri = result.canonicalUri;
627 result.sourceMap.entries.push_back({{1, 1}, {1, 1}});
628 const std::string scanned = maskComments(source);
629
630 static const std::regex importPattern(
631 R"(\bimport\s+(?:\*\s+as\s+[A-Za-z_]\w*|\{[^}]*\})\s+from\s+[\"']([^\"']+)[\"'])");
632 static const std::regex exportPattern(R"(\bexport\s+(?:function|class|const)\s+([A-Za-z_]\w*))");
633 static const std::regex persistPattern(R"(\bpersist\s+([A-Za-z_]\w*)\s*(?::[^=]+)?=)");
634 static const std::regex symbolPattern(R"(\b(local|function|class)\s+([A-Za-z_]\w*)\s*(?::\s*([^=,\)\{\n]+))?)");
635 static const std::regex asyncPattern(R"(\basync\s+function\s+([A-Za-z_]\w*))");
636 static const std::regex awaitPattern(R"(\bawait\b)");
637 static const std::regex propertyPattern(
638 R"(((?:[ \t]*@[A-Za-z_]\w*\‍([^\r\n]*\)[ \t]*\r?\n)+)[ \t]*([A-Za-z_]\w*)[ \t]*:[ \t]*([^=\r\n]+)[ \t]*=)");
639 static const std::regex annotationPattern(R"(@([A-Za-z_]\w*)\‍(([^)]*)\))");
640
641 forMatches(scanned, importPattern, [&](const std::smatch& match, size_t) {
642 appendUnique(result.imports, match[1].str());
643 appendUnique(result.moduleReferences, match[1].str());
644 });
645 forMatches(scanned, exportPattern,
646 [&](const std::smatch& match, size_t) { appendUnique(result.exports, match[1].str()); });
647 forMatches(scanned, persistPattern, [&](const std::smatch& match, size_t offset) {
648 appendUnique(result.persistRoots, match[1].str());
649 result.symbols.push_back({match[1].str(), "persist", "dynamic", positionAt(source, offset)});
650 });
651 forMatches(scanned, symbolPattern, [&](const std::smatch& match, size_t offset) {
652 result.symbols.push_back({match[2].str(), match[1].str(), match[3].matched ? match[3].str() : "dynamic",
653 positionAt(source, offset)});
654 });
655 forMatches(scanned, asyncPattern,
656 [&](const std::smatch& match, size_t) { appendUnique(result.asyncFunctions, match[1].str()); });
657 forMatches(scanned, awaitPattern,
658 [&](const std::smatch&, size_t offset) { result.awaitLocations.push_back(positionAt(source, offset)); });
659 forMatches(scanned, propertyPattern, [&](const std::smatch& match, size_t offset) {
660 ScriptPropertyMetadata property;
661 property.name = match[2].str();
662 property.erasedType = trim(match[3].str());
663 property.position = positionAt(source, offset + static_cast<size_t>(match.position(2)));
664 const std::string annotations = match[1].str();
665 forMatches(annotations, annotationPattern, [&](const std::smatch& annotation, size_t) {
666 const std::string name = annotation[1].str();
667 const std::vector<std::string> args = splitArguments(annotation[2].str());
668 for (size_t i = 0; i < args.size(); ++i) {
669 const size_t colon = args[i].find(':');
670 if (i == 0 && colon == std::string::npos)
671 property.attributes[name] = attributeValue(args[i]);
672 else if (colon != std::string::npos)
673 property.attributes[trim(std::string_view(args[i]).substr(0, colon))] =
674 attributeValue(std::string_view(args[i]).substr(colon + 1));
675 }
676 });
677 property.choices = typeChoices(property.erasedType);
678 const auto editor = property.attributes.find("editor");
679 if (editor != property.attributes.end() && editor->second == "combo" && !property.choices.empty()) {
680 std::string options;
681 for (size_t i = 0; i < property.choices.size(); ++i) {
682 if (i != 0) options += ',';
683 options += property.choices[i];
684 }
685 property.attributes["options"] = std::move(options);
686 }
687 result.properties.push_back(std::move(property));
688 });
689 return result;
690}
