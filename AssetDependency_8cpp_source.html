20bool validFallback(const EvaDependency& dependency) {
21 if (dependency.kind != EvaDependencyKind::RuntimeOptional)
22 return dependency.fallback.empty();
23 const auto behavior = dependency.fallback.find("behavior");
24 const auto observable = dependency.fallback.find("observableCode");
25 if (behavior == dependency.fallback.end() || !behavior->second.isString() ||
26 observable == dependency.fallback.end() || !observable->second.isString() ||
27 observable->second.asString().empty())
28 return false;
29 const std::string& name = behavior->second.asString();
30 const auto asset = dependency.fallback.find("asset");
31 if (name == "use-asset")
32 return asset != dependency.fallback.end() && asset->second.isString() &&
33 AssetRef::parse(asset->second.asString()).ok();
34 return (name == "omit-feature" || name == "use-default") &&
35 asset == dependency.fallback.end();
36}

41 using TypeIdentity = std::pair<std::string, SchemaVersion>;
42 std::map<PersistentId, TypeIdentity> types;
43 std::set<PersistentId> local;
44 for (const auto& asset : manifest.assets) {
45 types.emplace(asset.asset.id(), TypeIdentity{asset.type, asset.schemaVersion});
46 local.emplace(asset.asset.id());
47 }
48 for (const auto& external : available) {
49 if (external.asset.id().isNil() || external.type.empty() || external.schemaVersion.isZero())
50 return fail<EvaDependencyValidation>(DiagnosticCode::InvalidArgument,
51 "external dependency fact is invalid");
52 const auto [found, inserted] = types.emplace(
53 external.asset.id(), TypeIdentity{external.type, external.schemaVersion});
54 if (!inserted && found->second != TypeIdentity{external.type, external.schemaVersion})
55 return fail<EvaDependencyValidation>(DiagnosticCode::TypeMismatch,
56 "dependency resolver reports conflicting types",
57 external.asset.format());
58 }
59 for (const auto& [name, entrypoint] : manifest.entrypoints)
60 if (!local.contains(entrypoint.id()))
61 return fail<EvaDependencyValidation>(DiagnosticCode::NotFound,
62 "entrypoint does not resolve to a local asset", name);

64 using EdgeKey = std::tuple<PersistentId, PersistentId, EvaDependencyKind, std::string>;
65 std::set<EdgeKey> edges;
66 std::map<PersistentId, std::vector<PersistentId>> graph;
67 EvaDependencyValidation result;
68 for (const auto& dependency : manifest.dependencies) {
69 if (!validFallback(dependency))
70 return fail<EvaDependencyValidation>(DiagnosticCode::InvalidArgument,
71 "dependency fallback policy is invalid",
72 dependency.path);
73 if (!local.contains(dependency.from.id()))
74 return fail<EvaDependencyValidation>(DiagnosticCode::NotFound,
75 "dependency source is not locally authoritative",
76 dependency.from.format());

78 dependency.path).second)

80 "duplicate dependency edge", dependency.path);
81 const auto target = types.find(dependency.to.id());
82 if (target == types.end()) {

93 return fail<EvaDependencyValidation>(DiagnosticCode::TypeMismatch,

101 enum class Mark : std::uint8_t { Visiting, Complete };
102 std::map<PersistentId, Mark> marks;
103 std::function<Result<void>(const PersistentId&)> visit = [&](const PersistentId& node) -> Result<void> {
104 const auto marked = marks.find(node);
105 if (marked != marks.end()) {
106 if (marked->second == Mark::Visiting)
107 return Result<void>::failure(Diagnostic::error(
108 DiagnosticCode::Conflict, "runtime-required dependency cycle detected",
109 node.format(), {}, "asset.dependency"));
110 return Result<void>::success();
111 }
112 marks.emplace(node, Mark::Visiting);
113 auto& targets = graph[node];
114 std::sort(targets.begin(), targets.end());
115 for (const auto& target : targets) {
116 auto nested = visit(target);
117 if (!nested) return nested;
118 }
119 marks[node] = Mark::Complete;
120 return Result<void>::success();
121 };
122 for (const auto& id : local) {
123 auto visited = visit(id);
124 if (!visited) return Result<EvaDependencyValidation>::failure(visited.status());
125 }
126 std::sort(result.presentAssets.begin(), result.presentAssets.end());
127 result.presentAssets.erase(std::unique(result.presentAssets.begin(), result.presentAssets.end()),
128 result.presentAssets.end());
129 std::sort(result.omittedOptionalAssets.begin(), result.omittedOptionalAssets.end());
130 result.omittedOptionalAssets.erase(

132 result.omittedOptionalAssets.end());
