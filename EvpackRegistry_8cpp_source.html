19std::vector<Diagnostic> dispatchCallbacks(const std::vector<EvpackRegistry::Callback>& callbacks,
20 const EvpackRegistryEvent& event) {
21 std::vector<Diagnostic> diagnostics;
22 for (const auto& callback : callbacks) {
23 try {
24 auto notified = callback(event);
25 if (!notified && notified.error()) diagnostics.push_back(*notified.error());
26 } catch (const std::exception& error) {
27 diagnostics.push_back(Diagnostic::error(DiagnosticCode::CallbackFailure,
28 "registry callback threw: " + std::string(error.what()),
29 event.handle.packageId.format(), {}, "asset.evpack.registry"));
30 } catch (...) {
31 diagnostics.push_back(Diagnostic::error(DiagnosticCode::CallbackFailure,
32 "registry callback threw a non-standard exception",
33 event.handle.packageId.format(), {}, "asset.evpack.registry"));
34 }
35 }
36 return diagnostics;
37}

75Result<EvpackMountReceipt> EvpackRegistry::commit(PreparedEvpackMount candidate) {
76 if (!candidate.pack_)
77 return Result<EvpackMountReceipt>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
78 "mount candidate has already been consumed", {},
79 {}, "asset.evpack.registry"));
80 const PersistentId packageId = candidate.pack_->packageId();
81 bool replaced = false;
82 EvpackHandle handle;
83 std::vector<Callback> callbacks;
84 {
85 std::lock_guard lock(mutex_);
86 std::set<PersistentId> available;
87 std::set<PersistentId> candidateAssets;
88 for (const auto& chunk : candidate.pack_->chunks()) {
89 available.emplace(chunk.assetId);
90 candidateAssets.emplace(chunk.assetId);
91 }
92 for (const auto& [mountedId, entry] : packages_) {
93 if (mountedId == packageId) continue;
94 for (const auto& chunk : entry.pack->chunks()) {
95 if (candidateAssets.contains(chunk.assetId))
96 return Result<EvpackMountReceipt>::failure(Diagnostic::error(
97 DiagnosticCode::Conflict, "runtime asset identity already has another mounted provider",
98 chunk.assetId.format(), {}, "asset.evpack.registry"));
99 available.emplace(chunk.assetId);
100 }
101 }
102 for (const auto& chunk : candidate.pack_->chunks()) {
103 for (const auto& dependency : chunk.dependencies) {
104 if (!available.contains(dependency))
105 return Result<EvpackMountReceipt>::failure(
106 Diagnostic::error(DiagnosticCode::NotFound, "required runtime dependency is not mounted",
107 dependency.format(), {}, "asset.evpack.registry"));
108 }
109 }
110 for (const auto& [mountedId, entry] : packages_) {
111 if (mountedId == packageId) continue;
112 for (const auto& chunk : entry.pack->chunks()) {
113 for (const auto& dependency : chunk.dependencies) {
114 if (!available.contains(dependency))
115 return Result<EvpackMountReceipt>::failure(Diagnostic::error(
116 DiagnosticCode::NotFound, "replacement would invalidate a mounted package dependency",
117 dependency.format(), {}, "asset.evpack.registry"));
118 }
119 }
120 }
121 replaced = packages_.contains(packageId);
122 std::uint64_t& next = nextGenerations_[packageId];
123 if (next == std::numeric_limits<std::uint64_t>::max())
124 return Result<EvpackMountReceipt>::failure(
125 Diagnostic::error(DiagnosticCode::InvariantViolation, "package generation is exhausted",
126 packageId.format(), {}, "asset.evpack.registry"));
127 ++next;
128 auto pack = std::move(candidate.pack_);
129 handle = {packageId, pack->buildId(), next};
130 packages_.insert_or_assign(packageId, Entry{std::move(pack), next});
131 callbacks.reserve(callbacks_.size());
132 for (const auto& [id, callback] : callbacks_) {
133 (void)id;
134 callbacks.push_back(callback);
135 }
136 }
137 const EvpackRegistryEvent event{replaced ? EvpackRegistryEventKind::Replaced : EvpackRegistryEventKind::Mounted,
138 handle};
139 auto callbackDiagnostics = dispatchCallbacks(callbacks, event);
140 return Result<EvpackMountReceipt>::success({handle, replaced, std::move(callbackDiagnostics)});
141}

157Result<EvpackAssetHandle> EvpackRegistry::resolveAsset(const AssetRef& assetRef, std::string_view expectedType,
158 const EvpackCapabilities& capabilities) const {
159 if (expectedType.empty())
160 return Result<EvpackAssetHandle>::failure(Diagnostic::error(
161 DiagnosticCode::InvalidArgument, "expected runtime asset type is empty", {}, {}, "asset.evpack.registry"));
162 std::lock_guard lock(mutex_);
163 bool identityFound = false;
164 bool typeFound = false;
165 for (const auto& [packageId, entry] : packages_) {
166 const EvpackChunk* representative = nullptr;
167 for (const auto& chunk : entry.pack->chunks()) {
168 if (chunk.assetId != assetRef.id()) continue;
169 identityFound = true;
170 representative = &chunk;
171 const std::string actual = chunk.type + "/" + std::to_string(chunk.schemaVersion.value());
172 if (actual == expectedType)
173 typeFound = true;
174 else
175 return Result<EvpackAssetHandle>::failure(Diagnostic::error(
176 DiagnosticCode::TypeMismatch, "runtime asset provider has a different canonical type",
177 assetRef.format(), {}, "asset.evpack.registry"));
178 }
179 if (!representative) continue;
180 auto selected = selectEvpackVariant(*entry.pack, capabilities);
181 if (!selected) continue;
182 return Result<EvpackAssetHandle>::success(
183 {{packageId, entry.pack->buildId(), entry.generation}, assetRef, std::string(expectedType), capabilities});
184 }
185 if (identityFound && typeFound)
186 return Result<EvpackAssetHandle>::failure(Diagnostic::error(DiagnosticCode::Unsupported,
187 "asset provider has no compatible variant",
188 assetRef.format(), {}, "asset.evpack.registry"));
189 return Result<EvpackAssetHandle>::failure(Diagnostic::error(DiagnosticCode::NotFound,
190 "runtime asset has no mounted provider",
191 assetRef.format(), {}, "asset.evpack.registry"));
192}

257Result<RuntimeAssetPayload> EvpackResourceReader::read(const AssetRef& asset, std::string_view expectedType,
258 const EvpackCapabilities& capabilities,
259 std::uint64_t maximumDecodedBytes) const {
260 if (!pack_ || asset.id().isNil() || expectedType.empty())
261 return Result<RuntimeAssetPayload>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
262 "pack, asset and expected type are required", {},
263 {}, "asset.evpack.reader"));
264 auto selected = selectEvpackVariant(*pack_, capabilities);
265 if (!selected) return Result<RuntimeAssetPayload>::failure(selected.status());
266 RuntimeAssetPayload result{asset, {}, {}, selected.value(), {}};
267 bool identityFound = false;
268 bool typeFound = false;
269 std::uint64_t decodedTotal = 0;
270 for (std::size_t index = 0; index < pack_->chunks().size(); ++index) {
271 const auto& chunk = pack_->chunks()[index];
272 if (chunk.assetId != asset.id()) continue;
273 identityFound = true;
274 const std::string actualType = chunk.type + "/" + std::to_string(chunk.schemaVersion.value());
275 if (actualType != expectedType) continue;
276 typeFound = true;
277 if (chunk.variantIndex != selected.value().index) continue;
278 if (decodedTotal > maximumDecodedBytes || chunk.decodedSize > maximumDecodedBytes - decodedTotal)
279 return Result<RuntimeAssetPayload>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
280 "asset payload exceeds decoded budget",
281 asset.format(), {}, "asset.evpack.reader"));
282 auto bytes = pack_->decodeChunk(index, maximumDecodedBytes - decodedTotal);
283 if (!bytes) return Result<RuntimeAssetPayload>::failure(bytes.status());
284 decodedTotal += bytes.value().size();
285 result.type = chunk.type;
286 result.schemaVersion = chunk.schemaVersion;
287 result.chunks.push_back({chunk.kind, chunk.chunkId, std::move(bytes).takeValue()});
288 }
289 if (result.chunks.empty()) {
290 if (identityFound && !typeFound)
291 return Result<RuntimeAssetPayload>::failure(
292 Diagnostic::error(DiagnosticCode::TypeMismatch, "asset does not provide the expected type/version",
293 asset.format(), {}, "asset.evpack.reader"));
294 if (typeFound)
295 return Result<RuntimeAssetPayload>::failure(Diagnostic::error(DiagnosticCode::Unsupported,
296 "asset has no chunk for the selected variant",
297 asset.format(), {}, "asset.evpack.reader"));
298 return Result<RuntimeAssetPayload>::failure(Diagnostic::error(DiagnosticCode::NotFound, "asset is not present",
299 asset.format(), {}, "asset.evpack.reader"));
300 }
301 std::sort(result.chunks.begin(), result.chunks.end(), [](const auto& left, const auto& right) {
302 return std::tie(left.kind, left.chunkId) < std::tie(right.kind, right.chunkId);
303 });
304 if (result.chunks.front().kind != EvpackChunkKind::Definition)
305 return Result<RuntimeAssetPayload>::failure(Diagnostic::error(DiagnosticCode::InvariantViolation,
306 "asset has no definition chunk", asset.format(),
307 {}, "asset.evpack.reader"));
308 return Result<RuntimeAssetPayload>::success(std::move(result));
309}

311Result<std::vector<AssetRef>> EvpackResourceReader::listAssets(std::string_view expectedType,
312 const EvpackCapabilities& capabilities,
313 std::uint32_t maximumAssets) const {
314 if (!pack_ || expectedType.empty() || maximumAssets == 0)
315 return Result<std::vector<AssetRef>>::failure(
316 Diagnostic::error(DiagnosticCode::InvalidArgument, "pack, expected type and asset budget are required", {},
317 {}, "asset.evpack.reader"));
318 auto selected = selectEvpackVariant(*pack_, capabilities);
319 if (!selected) return Result<std::vector<AssetRef>>::failure(selected.status());
320 try {
321 std::vector<PersistentId> identities;
322 for (const auto& chunk : pack_->chunks()) {
323 const std::string actualType = chunk.type + "/" + std::to_string(chunk.schemaVersion.value());
324 if (chunk.variantIndex != selected.value().index || actualType != expectedType) continue;
325 identities.push_back(chunk.assetId);
326 }
327 std::sort(identities.begin(), identities.end());
328 identities.erase(std::unique(identities.begin(), identities.end()), identities.end());
329 if (identities.size() > maximumAssets)
330 return Result<std::vector<AssetRef>>::failure(Diagnostic::error(
331 DiagnosticCode::InvalidArgument, "asset listing exceeds result budget", {}, {}, "asset.evpack.reader"));
332 std::vector<AssetRef> result;
333 result.reserve(identities.size());
334 for (const auto& identity : identities) {
335 auto reference = AssetRef::fromId(identity);
336 if (!reference) return Result<std::vector<AssetRef>>::failure(reference.status());
337 result.push_back(std::move(reference).takeValue());
338 }
339 return Result<std::vector<AssetRef>>::success(std::move(result));
340 } catch (const std::bad_alloc&) {
341 return Result<std::vector<AssetRef>>::failure(Diagnostic::error(
342 DiagnosticCode::Failed, "asset listing allocation failed", {}, {}, "asset.evpack.reader"));
343 }
344}
