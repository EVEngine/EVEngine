20std::vector<Diagnostic> dispatchCallbacks(
21 const std::vector<EvpackRegistry::Callback>& callbacks,
22 const EvpackRegistryEvent& event) {
23 std::vector<Diagnostic> diagnostics;
24 for (const auto& callback : callbacks) {
25 try {
26 auto notified = callback(event);
27 if (!notified && notified.error()) diagnostics.push_back(*notified.error());
28 } catch (const std::exception& error) {
29 diagnostics.push_back(Diagnostic::error(DiagnosticCode::CallbackFailure,
30 "registry callback threw: " + std::string(error.what()),
31 event.handle.packageId.format(), {},
32 "asset.evpack.registry"));
33 } catch (...) {
34 diagnostics.push_back(Diagnostic::error(DiagnosticCode::CallbackFailure,
35 "registry callback threw a non-standard exception",
36 event.handle.packageId.format(), {},
37 "asset.evpack.registry"));
38 }
39 }
40 return diagnostics;
41}

79Result<EvpackMountReceipt> EvpackRegistry::commit(PreparedEvpackMount candidate) {
80 if (!candidate.pack_)
81 return Result<EvpackMountReceipt>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
82 "mount candidate has already been consumed", {},
83 {}, "asset.evpack.registry"));
84 const PersistentId packageId = candidate.pack_->packageId();
85 bool replaced = false;
86 EvpackHandle handle;
87 std::vector<Callback> callbacks;
88 {
89 std::lock_guard lock(mutex_);
90 std::set<PersistentId> available;
91 std::set<PersistentId> candidateAssets;
92 for (const auto& chunk : candidate.pack_->chunks()) {
93 available.emplace(chunk.assetId);
94 candidateAssets.emplace(chunk.assetId);
95 }
96 for (const auto& [mountedId, entry] : packages_) {
97 if (mountedId == packageId) continue;
98 for (const auto& chunk : entry.pack->chunks()) {
99 if (candidateAssets.contains(chunk.assetId))
100 return Result<EvpackMountReceipt>::failure(Diagnostic::error(
101 DiagnosticCode::Conflict, "runtime asset identity already has another mounted provider",
102 chunk.assetId.format(), {}, "asset.evpack.registry"));
103 available.emplace(chunk.assetId);
104 }
105 }
106 for (const auto& chunk : candidate.pack_->chunks()) {
107 for (const auto& dependency : chunk.dependencies) {
108 if (!available.contains(dependency))
109 return Result<EvpackMountReceipt>::failure(
110 Diagnostic::error(DiagnosticCode::NotFound, "required runtime dependency is not mounted",
111 dependency.format(), {}, "asset.evpack.registry"));
112 }
113 }
114 for (const auto& [mountedId, entry] : packages_) {
115 if (mountedId == packageId) continue;
116 for (const auto& chunk : entry.pack->chunks()) {
117 for (const auto& dependency : chunk.dependencies) {
118 if (!available.contains(dependency))
119 return Result<EvpackMountReceipt>::failure(Diagnostic::error(
120 DiagnosticCode::NotFound, "replacement would invalidate a mounted package dependency",
121 dependency.format(), {}, "asset.evpack.registry"));
122 }
123 }
124 }
125 replaced = packages_.contains(packageId);
126 std::uint64_t& next = nextGenerations_[packageId];
127 if (next == std::numeric_limits<std::uint64_t>::max())
128 return Result<EvpackMountReceipt>::failure(
129 Diagnostic::error(DiagnosticCode::InvariantViolation, "package generation is exhausted",
130 packageId.format(), {}, "asset.evpack.registry"));
131 ++next;
132 auto pack = std::move(candidate.pack_);
133 handle = {packageId, pack->buildId(), next};
134 packages_.insert_or_assign(packageId, Entry{std::move(pack), next});
135 callbacks.reserve(callbacks_.size());
136 for (const auto& [id, callback] : callbacks_) {
137 (void)id;
138 callbacks.push_back(callback);
139 }
140 }
141 const EvpackRegistryEvent event{replaced ? EvpackRegistryEventKind::Replaced
142 : EvpackRegistryEventKind::Mounted,
143 handle};
144 auto callbackDiagnostics = dispatchCallbacks(callbacks, event);
145 return Result<EvpackMountReceipt>::success({handle, replaced, std::move(callbackDiagnostics)});
146}

148Result<std::shared_ptr<const Evpack>> EvpackRegistry::resolve(const EvpackHandle& handle) const {

162Result<EvpackAssetHandle> EvpackRegistry::resolveAsset(
163 const AssetRef& assetRef, std::string_view expectedType,
164 const EvpackCapabilities& capabilities) const {
165 if (expectedType.empty())
166 return Result<EvpackAssetHandle>::failure(Diagnostic::error(
167 DiagnosticCode::InvalidArgument, "expected runtime asset type is empty", {}, {}, "asset.evpack.registry"));
168 std::lock_guard lock(mutex_);
169 bool identityFound = false;
170 bool typeFound = false;
171 for (const auto& [packageId, entry] : packages_) {
172 const EvpackChunk* representative = nullptr;
173 for (const auto& chunk : entry.pack->chunks()) {
174 if (chunk.assetId != assetRef.id()) continue;
175 identityFound = true;
176 representative = &chunk;
177 const std::string actual = chunk.type + "/" +
178 std::to_string(chunk.schemaVersion.value());
179 if (actual == expectedType) typeFound = true;
180 else
181 return Result<EvpackAssetHandle>::failure(Diagnostic::error(
182 DiagnosticCode::TypeMismatch, "runtime asset provider has a different canonical type",
183 assetRef.format(), {}, "asset.evpack.registry"));
184 }
185 if (!representative) continue;
186 auto selected = selectEvpackVariant(*entry.pack, capabilities);
187 if (!selected) continue;
188 return Result<EvpackAssetHandle>::success(
189 {{packageId, entry.pack->buildId(), entry.generation}, assetRef,
190 std::string(expectedType), capabilities});
191 }
192 if (identityFound && typeFound)
193 return Result<EvpackAssetHandle>::failure(Diagnostic::error(DiagnosticCode::Unsupported,
194 "asset provider has no compatible variant",
195 assetRef.format(), {}, "asset.evpack.registry"));
196 return Result<EvpackAssetHandle>::failure(Diagnostic::error(DiagnosticCode::NotFound,
197 "runtime asset has no mounted provider",
198 assetRef.format(), {}, "asset.evpack.registry"));
199}
