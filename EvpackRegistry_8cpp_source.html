26std::vector<Diagnostic> dispatchCallbacks(
27 const std::vector<EvpackRegistry::Callback>& callbacks,
28 const EvpackRegistryEvent& event) {
29 std::vector<Diagnostic> diagnostics;
30 for (const auto& callback : callbacks) {
31 try {
32 auto notified = callback(event);
33 if (!notified && notified.error()) diagnostics.push_back(*notified.error());
34 } catch (const std::exception& error) {
35 diagnostics.push_back(Diagnostic::error(DiagnosticCode::CallbackFailure,
36 "registry callback threw: " + std::string(error.what()),
37 event.handle.packageId.format(), {},
38 "asset.evpack.registry"));
39 } catch (...) {
40 diagnostics.push_back(Diagnostic::error(DiagnosticCode::CallbackFailure,
41 "registry callback threw a non-standard exception",
42 event.handle.packageId.format(), {},
43 "asset.evpack.registry"));
44 }
45 }
46 return diagnostics;
47}

60Result<EvpackRegistrySubscription> EvpackRegistry::subscribe(Callback callback) {
61 if (!callback)
62 return registryFailure<EvpackRegistrySubscription>(DiagnosticCode::InvalidArgument,
63 "registry callback is empty");
64 std::lock_guard lock(mutex_);

84Result<EvpackMountReceipt> EvpackRegistry::commit(PreparedEvpackMount candidate) {
85 if (!candidate.pack_)
86 return registryFailure<EvpackMountReceipt>(DiagnosticCode::InvalidArgument,
87 "mount candidate has already been consumed");
88 const PersistentId packageId = candidate.pack_->packageId();
89 bool replaced = false;
90 EvpackHandle handle;
91 std::vector<Callback> callbacks;
92 {
93 std::lock_guard lock(mutex_);
94 std::set<PersistentId> available;
95 std::set<PersistentId> candidateAssets;
96 for (const auto& chunk : candidate.pack_->chunks()) {
97 available.emplace(chunk.assetId);
98 candidateAssets.emplace(chunk.assetId);
99 }
100 for (const auto& [mountedId, entry] : packages_) {
101 if (mountedId == packageId) continue;
102 for (const auto& chunk : entry.pack->chunks()) {
103 if (candidateAssets.contains(chunk.assetId))
104 return registryFailure<EvpackMountReceipt>(
105 DiagnosticCode::Conflict,
106 "runtime asset identity already has another mounted provider",
107 chunk.assetId.format());
108 available.emplace(chunk.assetId);
109 }
110 }
111 for (const auto& chunk : candidate.pack_->chunks()) {
112 for (const auto& dependency : chunk.dependencies) {
113 if (!available.contains(dependency))
114 return registryFailure<EvpackMountReceipt>(
115 DiagnosticCode::NotFound, "required runtime dependency is not mounted",
116 dependency.format());
117 }
118 }
119 for (const auto& [mountedId, entry] : packages_) {
120 if (mountedId == packageId) continue;
121 for (const auto& chunk : entry.pack->chunks()) {
122 for (const auto& dependency : chunk.dependencies) {
123 if (!available.contains(dependency))
124 return registryFailure<EvpackMountReceipt>(
125 DiagnosticCode::NotFound,
126 "replacement would invalidate a mounted package dependency",
127 dependency.format());
128 }
129 }
130 }
131 replaced = packages_.contains(packageId);
132 std::uint64_t& next = nextGenerations_[packageId];
133 if (next == std::numeric_limits<std::uint64_t>::max())
134 return registryFailure<EvpackMountReceipt>(DiagnosticCode::InvariantViolation,
135 "package generation is exhausted", packageId.format());
136 ++next;
137 auto pack = std::move(candidate.pack_);
138 handle = {packageId, pack->buildId(), next};
139 packages_.insert_or_assign(packageId, Entry{std::move(pack), next});
140 callbacks.reserve(callbacks_.size());
141 for (const auto& [id, callback] : callbacks_) {
142 (void)id;
143 callbacks.push_back(callback);
144 }
145 }
146 const EvpackRegistryEvent event{replaced ? EvpackRegistryEventKind::Replaced
147 : EvpackRegistryEventKind::Mounted,
148 handle};
149 auto callbackDiagnostics = dispatchCallbacks(callbacks, event);
150 return Result<EvpackMountReceipt>::success({handle, replaced, std::move(callbackDiagnostics)});
151}

153Result<std::shared_ptr<const Evpack>> EvpackRegistry::resolve(const EvpackHandle& handle) const {
154 std::lock_guard lock(mutex_);
155 const auto found = packages_.find(handle.packageId);
156 if (found == packages_.end())
157 return registryFailure<std::shared_ptr<const Evpack>>(DiagnosticCode::NotFound,
158 "runtime package is not mounted",
159 handle.packageId.format());
160 if (!sameGeneration(handle, found->second))
161 return registryFailure<std::shared_ptr<const Evpack>>(DiagnosticCode::StaleHandle,
162 "runtime package handle is stale",
163 handle.packageId.format());
164 return Result<std::shared_ptr<const Evpack>>::success(found->second.pack);
165}

167Result<EvpackAssetHandle> EvpackRegistry::resolveAsset(
168 const AssetRef& assetRef, std::string_view expectedType,
169 const EvpackCapabilities& capabilities) const {
170 if (expectedType.empty())
171 return registryFailure<EvpackAssetHandle>(DiagnosticCode::InvalidArgument,
172 "expected runtime asset type is empty");
173 std::lock_guard lock(mutex_);
174 bool identityFound = false;
175 bool typeFound = false;
176 for (const auto& [packageId, entry] : packages_) {
177 const EvpackChunk* representative = nullptr;
178 for (const auto& chunk : entry.pack->chunks()) {
179 if (chunk.assetId != assetRef.id()) continue;
180 identityFound = true;
181 representative = &chunk;
182 const std::string actual = chunk.type + "/" +
183 std::to_string(chunk.schemaVersion.value());
184 if (actual == expectedType) typeFound = true;
185 else
186 return registryFailure<EvpackAssetHandle>(
187 DiagnosticCode::TypeMismatch,
188 "runtime asset provider has a different canonical type", assetRef.format());
189 }
190 if (!representative) continue;
191 auto selected = selectEvpackVariant(*entry.pack, capabilities);
192 if (!selected) continue;
193 return Result<EvpackAssetHandle>::success(
194 {{packageId, entry.pack->buildId(), entry.generation}, assetRef,
195 std::string(expectedType), capabilities});
196 }
197 if (identityFound && typeFound)
198 return registryFailure<EvpackAssetHandle>(DiagnosticCode::Unsupported,
199 "asset provider has no compatible variant",
200 assetRef.format());
201 return registryFailure<EvpackAssetHandle>(DiagnosticCode::NotFound,
202 "runtime asset has no mounted provider",
203 assetRef.format());
204}

215Result<EvpackChunkHandle> EvpackRegistry::resolveChunk(const EvpackHandle& handle,
216 const PersistentId& assetId, std::string_view type,
217 EvpackChunkKind kind,
218 std::uint32_t chunkId,
219 std::uint32_t variantIndex) const {
220 auto resolved = resolve(handle);
221 if (!resolved) return Result<EvpackChunkHandle>::failure(resolved.status());
222 const auto pack = std::move(resolved).takeValue();
223 for (std::size_t index = 0; index < pack->chunks().size(); ++index) {
224 const auto& chunk = pack->chunks()[index];
225 if (chunk.assetId == assetId && chunk.type == type && chunk.kind == kind && chunk.chunkId == chunkId &&
226 chunk.variantIndex == variantIndex)
227 return Result<EvpackChunkHandle>::success({handle, static_cast<std::uint32_t>(index)});
228 }
229 return registryFailure<EvpackChunkHandle>(DiagnosticCode::NotFound, "runtime chunk is not present",
230 assetId.format());
231}
