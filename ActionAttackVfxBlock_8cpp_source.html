11Result<double> numeric(const Value::Object& payload, std::string_view key, double fallback) {
12 const auto found = payload.find(std::string(key));
13 if (found == payload.end()) return Result<double>::success(fallback);
14 double value = 0.0;
15 if (const auto* integer = found->second.getIf<std::int64_t>())
16 value = static_cast<double>(*integer);
17 else if (const auto* decimal = found->second.getIf<double>())
18 value = *decimal;
19 else
20 return Result<double>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
21 "AttackVfx field must be numeric", std::string(key)));
22 if (!std::isfinite(value))
23 return Result<double>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
24 "AttackVfx field must be finite", std::string(key)));
25 return Result<double>::success(value);
