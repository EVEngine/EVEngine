14eve::Result<std::unordered_map<std::string, double>> parseNumbers(const eve::Value *value,
15 const std::string &path) {
16 if (!value || !value->isObject())
17 return gameStateFailure<std::unordered_map<std::string, double>>(
18 eve::DiagnosticCode::ParseError, "game state field must be an object", path);
19 std::unordered_map<std::string, double> result;
20 for (const auto &key : value->keys()) {
21 const eve::Value *item = value->find(key);
22 if (!item || !item->isNumeric())
23 return gameStateFailure<std::unordered_map<std::string, double>>(
24 eve::DiagnosticCode::ParseError, "game state variable must be numeric", path + "." + key);
25 const double number = item->isInt64() ? static_cast<double>(item->asInt()) : item->asDouble();
26 if (!std::isfinite(number))
27 return gameStateFailure<std::unordered_map<std::string, double>>(
28 eve::DiagnosticCode::InvalidArgument, "game state variable must be finite", path + "." + key);
29 result.emplace(key, number);
30 }
31 return eve::Result<std::unordered_map<std::string, double>>::success(std::move(result));
32}

83eve::Result<std::string> GameState::snapshotJson() const {
84 eve::Value::Object switches;
85 for (const auto &[name, on] : switches_) switches.emplace(name, eve::Value(on));
86 eve::Value::Object variables;
87 for (const auto &[name, value] : variables_) variables.emplace(name, eve::Value(value));
88 eve::Value::Object scoped;
89 for (const auto &[scope, values] : selfVariables_) {
90 eve::Value::Object encoded;
91 for (const auto &[name, value] : values) encoded.emplace(name, eve::Value(value));
92 scoped.emplace(scope, eve::Value(std::move(encoded)));
93 }
94 eve::Value::Object root;
95 root.emplace("schema", eve::Value("eve.rpg.game-state"));
96 root.emplace("selfVariables", eve::Value(std::move(scoped)));
97 root.emplace("switches", eve::Value(std::move(switches)));
98 root.emplace("variables", eve::Value(std::move(variables)));
99 root.emplace("version", eve::Value(1));
100 return eve::Value(std::move(root)).toJson();
101}

103eve::Result<void> GameState::restoreSnapshotJson(std::string_view json) {
104 auto parsed = eve::Value::fromJson(json);
105 if (!parsed.ok()) return eve::Result<void>::failure(parsed.status());
106 const eve::Value &root = parsed.value();
107 if (!root.isObject())
108 return gameStateFailure<void>(eve::DiagnosticCode::ParseError, "game state root must be an object", "$.");
109 const eve::Value *schema = root.find("schema");
110 if (!schema || !schema->isString() || schema->asString() != "eve.rpg.game-state")
111 return gameStateFailure<void>(eve::DiagnosticCode::InvalidArgument,
112 "snapshot does not belong to RPG GameState", "$.schema");
113 const eve::Value *version = root.find("version");
114 if (!version || !version->isInt64() || version->asInt() != 1)
115 return gameStateFailure<void>(eve::DiagnosticCode::UnknownVersion,
116 "unsupported RPG GameState snapshot version", "$.version");
117
118 const eve::Value *switches = root.find("switches");
119 if (!switches || !switches->isObject())
120 return gameStateFailure<void>(eve::DiagnosticCode::ParseError,
121 "game state switches must be an object", "$.switches");
122 std::unordered_map<std::string, bool> candidateSwitches;
123 for (const auto &key : switches->keys()) {
124 const eve::Value *item = switches->find(key);
125 if (!item || !item->isBool())
126 return gameStateFailure<void>(eve::DiagnosticCode::ParseError,
127 "game state switch must be boolean", "$.switches." + key);
128 candidateSwitches.emplace(key, item->asBool());
129 }
130
131 auto candidateVariables = parseNumbers(root.find("variables"), "$.variables");
132 if (!candidateVariables.ok()) return eve::Result<void>::failure(candidateVariables.status());
133 const eve::Value *scoped = root.find("selfVariables");
134 if (!scoped || !scoped->isObject())
135 return gameStateFailure<void>(eve::DiagnosticCode::ParseError,
136 "game state selfVariables must be an object", "$.selfVariables");
137 std::unordered_map<std::string, std::unordered_map<std::string, double>> candidateScoped;
138 for (const auto &scope : scoped->keys()) {
139 auto values = parseNumbers(scoped->find(scope), "$.selfVariables." + scope);
140 if (!values.ok()) return eve::Result<void>::failure(values.status());
141 candidateScoped.emplace(scope, std::move(values).takeValue());
142 }
143
144 switches_ = std::move(candidateSwitches);
145 variables_ = std::move(candidateVariables).takeValue();
146 selfVariables_ = std::move(candidateScoped);
147 return eve::Result<void>::success();
148}
