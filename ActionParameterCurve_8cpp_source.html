67Result<ActionParameterCurveBinding> ActionParameterCurveBinding::fromPayload(const Value::Object& payload) {
68 ActionParameterCurveBinding candidate;
69 const auto target = payload.find("target");
70 if (target == payload.end() || !target->second.getIf<std::string>())
71 return invalid<ActionParameterCurveBinding>("parameter target must be text", "target");
72 auto parsedTarget = LogicalId::parse(*target->second.getIf<std::string>());
73 if (!parsedTarget)
74 return invalid<ActionParameterCurveBinding>("parameter target must be a LogicalId", "target");
75 candidate.target = std::move(*parsedTarget);
76
77 if (const auto found = payload.find("operation"); found != payload.end()) {
78 const auto* text = found->second.getIf<std::string>();
79 if (!text || !parseOperation(*text))
80 return invalid<ActionParameterCurveBinding>("parameter operation is unknown", "operation");
81 candidate.operation = *parseOperation(*text);
82 }
83
84 const auto foundKeys = payload.find("keys");
85 const auto* keys = foundKeys == payload.end() ? nullptr : foundKeys->second.getIf<Value::Array>();
86 if (!keys || keys->size() < 2 || keys->size() > 1024)
87 return invalid<ActionParameterCurveBinding>("parameter curve requires 2 to 1024 keys", "keys");
88 candidate.keys.reserve(keys->size());
89 for (std::size_t index = 0; index < keys->size(); ++index) {
90 const auto* object = (*keys)[index].getIf<Value::Object>();
91 const std::string path = "keys[" + std::to_string(index) + "]";
92 if (!object) return invalid<ActionParameterCurveBinding>("parameter key must be an object", path);
93 const auto time = object->find("time");
94 const auto value = object->find("value");
95 if (time == object->end() || value == object->end())
96 return invalid<ActionParameterCurveBinding>("parameter key requires time and value", path);
97 auto parsedTime = number(time->second, path + ".time");
98 if (!parsedTime) return Result<ActionParameterCurveBinding>::failure(parsedTime.status());
99 auto parsedValue = number(value->second, path + ".value");
100 if (!parsedValue) return Result<ActionParameterCurveBinding>::failure(parsedValue.status());
101 if (parsedTime.value() < 0.0 || parsedTime.value() > 1.0 ||
102 (!candidate.keys.empty() && parsedTime.value() <= candidate.keys.back().time))
103 return invalid<ActionParameterCurveBinding>("parameter key times must be strictly ordered in [0, 1]",
104 path + ".time");
105 ActionParameterKey key;
106 key.time = parsedTime.value();
107 key.value = parsedValue.value();
108 if (const auto tangent = object->find("inTangent"); tangent != object->end()) {
109 auto parsed = number(tangent->second, path + ".inTangent");
110 if (!parsed) return Result<ActionParameterCurveBinding>::failure(parsed.status());
111 key.inTangent = parsed.value();
112 }
113 if (const auto tangent = object->find("outTangent"); tangent != object->end()) {
114 auto parsed = number(tangent->second, path + ".outTangent");
115 if (!parsed) return Result<ActionParameterCurveBinding>::failure(parsed.status());
116 key.outTangent = parsed.value();
117 }
118 if (const auto mode = object->find("interpolation"); mode != object->end()) {
119 const auto* text = mode->second.getIf<std::string>();
120 if (!text || !parseInterpolation(*text))
121 return invalid<ActionParameterCurveBinding>("parameter interpolation is unknown",
122 path + ".interpolation");
123 key.interpolation = *parseInterpolation(*text);
124 }
125 candidate.keys.push_back(key);
126 }
127 if (candidate.keys.front().time != 0.0 || candidate.keys.back().time != 1.0)
128 return invalid<ActionParameterCurveBinding>("parameter curve endpoints must be at 0 and 1", "keys");
129 return Result<ActionParameterCurveBinding>::success(std::move(candidate));
130}

148double ActionParameterCurveBinding::sample(double progress) const noexcept {
149 if (keys.empty()) return 0.0;
150 progress = std::clamp(progress, 0.0, 1.0);
151 const auto upper = std::upper_bound(keys.begin(), keys.end(), progress,
152 [](double value, const ActionParameterKey& key) { return value < key.time; });
153 if (upper == keys.begin()) return keys.front().value;
154 if (upper == keys.end()) return keys.back().value;
155 const auto& left = *(upper - 1);
156 const auto& right = *upper;
157 if (left.interpolation == ActionParameterInterpolation::Step) return left.value;
158 const double span = right.time - left.time;
159 const double t = (progress - left.time) / span;
160 if (left.interpolation == ActionParameterInterpolation::Linear)
161 return left.value + (right.value - left.value) * t;
162 const double t2 = t * t;
163 const double t3 = t2 * t;
164 return (2.0 * t3 - 3.0 * t2 + 1.0) * left.value +
165 (t3 - 2.0 * t2 + t) * left.outTangent * span +
166 (-2.0 * t3 + 3.0 * t2) * right.value +
167 (t3 - t2) * right.inTangent * span;
168}
