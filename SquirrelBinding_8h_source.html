211template <typename Receiver, typename... Args>
212SQRESULT callFunction(HSQUIRRELVM vm, const HSQOBJECT& fn, const Receiver& receiver, Args&&... args) {
213 const SQInteger top = sq_gettop(vm);
214 sq_pushobject(vm, fn);
215 pushReceiver(vm, receiver);
216 (ssq::detail::pushValue(vm, scriptArgument(args)), ...);
217 // Everything above the callable: the receiver plus the arguments. The callable itself is not
218 // part of the count (sq_call takes the callee at -(params)-1).
219 const SQInteger params = sq_gettop(vm) - top - 1;
220 const SQRESULT result = sq_call(vm, params, SQFalse, SQTrue);
221 if (SQ_FAILED(result)) reportScriptCallFailure(vm, "callFunction");
222 sq_settop(vm, top);
223 return result;
224}
