24int bufferFromJson(const std::string &json, int fallback) {
25 auto *dm = eve::data::DataModule::create();
26 std::string err;
27 std::unique_ptr<data::JsonDocument> doc(dm->decodeJson(json, &err));
28 if (!doc || !doc->isObject()) return fallback;
29 auto obj = doc->object();
30 if (!obj || !obj->has("buffer")) return fallback;
31 try {
32 int n = obj->getValue<int>("buffer");
33 return n > 0 ? n : fallback;
34 } catch (...) {
35 return fallback;
36 }
37}

47ParticleEmitter *Particles::newEmitterFromFile(const std::string &path) {
48 auto *fs = eve::ModuleManager::getInstance<eve::filesystem::Filesystem>("Filesystem");
49 if (!fs) fs = eve::filesystem::Filesystem::create();
50
51 std::unique_ptr<eve::filesystem::FileData> data;
52 try {
53 data.reset(fs->read(path));
54 } catch (...) {
55 return nullptr;
56 }
57 if (!data || data->getSize() == 0) return nullptr;
58
59 std::string text(static_cast<const char *>(data->getData()), data->getSize());
60 int buffer = bufferFromJson(text, 1000);
61 ParticleEmitter *e = ParticleEmitter::createEmitter(buffer);
62 if (!loadConfigFile(e, path, nullptr)) {
63 // Entity stays in ECS registry; still return for inspection, or null?
64 // Prefer null on failed apply after create — leave entity (ECS has no destroy).
65 return e; // path may have failed; emitter still usable
66 }
67 return e;
68}
