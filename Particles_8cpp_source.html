27int bufferFromJson(const std::string &json, int fallback) {
28 auto *dm = eve::data::DataModule::create();
29 std::string err;
30 std::unique_ptr<data::JsonDocument> doc(dm->decodeJson(json, &err));
31 if (!doc || !doc->isObject()) return fallback;
32 auto obj = doc->object();
33 if (!obj || !obj->has("buffer")) return fallback;
34 try {
35 int n = obj->getValue<int>("buffer");
36 return n > 0 ? n : fallback;
37 } catch (...) {
38 return fallback;
39 }
40}

50ParticleEmitter *Particles::newEmitterFromFile(const std::string &path) {
51 auto *fs = eve::ModuleManager::getInstance<eve::filesystem::Filesystem>("Filesystem");
52 if (!fs) fs = eve::filesystem::Filesystem::create();
53
54 std::unique_ptr<eve::filesystem::FileData> data;
55 try {
56 data.reset(fs->read(path));
57 } catch (...) {
58 return nullptr;
59 }
60 if (!data || data->getSize() == 0) return nullptr;
61
62 std::string text(static_cast<const char *>(data->getData()), data->getSize());
63 int buffer = bufferFromJson(text, 1000);
64 ParticleEmitter *e = ParticleEmitter::createEmitter(buffer);
65 if (!loadConfigFile(e, path, nullptr)) {
66 // Entity stays in ECS registry; still return for inspection, or null?
67 // Prefer null on failed apply after create — leave entity (ECS has no destroy).
68 return e; // path may have failed; emitter still usable
69 }
70 return e;
71}

90eve::Result<void> Particles::advance(const eve::SimulationStep &step) {
91 // Config polling is an asset-side concern. Simulation itself is driven
92 // exclusively by the injected scheduler step below.
93 ParticleConfigSystem::poll();
94 auto simulation = ParticleSimSystem::advance(step);
95 if (!simulation) return eve::Result<void>::failure(simulation.status());
96 ParticleLightSystem::update();
97 return simulation;
