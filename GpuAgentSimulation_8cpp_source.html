36Result<void> GpuAgentSimulation::configure(const EffectProfile& profile) {
37 if (profile.base().maxAgents <= 0) {
38 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "maxAgents must be positive",
39 "maxAgents", {}, "gpuagents"));
40 }
41 if (profile.base().fixedDt <= 0.f) {
42 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "fixedDt must be positive",
43 "fixedDt", {}, "gpuagents"));
44 }
45 profile_ = profile;
46 capacity_ = profile.base().maxAgents;
47 buffers_[0].assign(static_cast<size_t>(capacity_), AgentState{});
48 buffers_[1].assign(static_cast<size_t>(capacity_), AgentState{});
49 front_ = 0;

59 return Result<void>::success();

75Result<void> GpuAgentSimulation::initialize(int count, const std::function<AgentState(int)>& spawnFn) {
76 if (!configured_) {
77 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
78 "simulation not configured", "configure", {}, "gpuagents"));
79 }
80 if (!spawnFn) {
81 return Result<void>::failure(
82 Diagnostic::error(DiagnosticCode::InvalidArgument, "spawnFn is empty", "spawnFn", {}, "gpuagents"));
83 }
84 if (count < 0 || count > capacity_) {
85 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
86 "initialize count out of range", "count", {}, "gpuagents"));
87 }
88 auto& buf = buffers_[front_];
89 for (int i = 0; i < capacity_; ++i) {
90 if (i < count) {
91 AgentState s = spawnFn(i);
92 s.alive = 1;
93 buf[static_cast<size_t>(i)] = s;
94 } else {
95 buf[static_cast<size_t>(i)] = AgentState{};
96 }
97 }
98 buffers_[1 - front_] = buf;
99 accumulator_ = 0.f;
100 return Result<void>::success();
101}

121Result<void> GpuAgentSimulation::step(float dt, const EnvironmentSnapshot& env) {
122 if (!configured_ || !solver_) {
123 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
124 "simulation not configured", "configure", {}, "gpuagents"));
125 }
126 EV_PARAM_CHECK(dt >= 0.f);
127 accumulator_ += dt;
128 const float fixed = profile_.base().fixedDt;
129 int guard = 0;
130 while (accumulator_ + 1e-8f >= fixed && guard < 8) {
131 const int read = front_;
132 const int write = 1 - front_;
133 solver_->step(buffers_[read], buffers_[write], profile_, env, fixed);
134 front_ = write;
135 accumulator_ -= fixed;
136 simTime_ += fixed;
137 ++stepCount_;
138 ++guard;
139 }
140 return Result<void>::success();
141}
