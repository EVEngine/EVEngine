34Result<void> GpuAgentSimulation::configure(const EffectProfile& profile) {
35 if (profile.base().maxAgents <= 0) {
36 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "maxAgents must be positive",
37 "maxAgents", {}, "gpuagents"));
38 }
39 if (profile.base().fixedDt <= 0.f) {
40 return Result<void>::failure(
41 Diagnostic::error(DiagnosticCode::InvalidArgument, "fixedDt must be positive", "fixedDt", {}, "gpuagents"));
42 }
43 profile_ = profile;
44 capacity_ = profile.base().maxAgents;
45 buffers_[0].assign(static_cast<size_t>(capacity_), AgentState{});
46 buffers_[1].assign(static_cast<size_t>(capacity_), AgentState{});
47 front_ = 0;
48 accumulator_ = 0.f;
49 simTime_ = 0.f;
50 stepCount_ = 0;
51 configured_ = true;
52 auto solverRes = makeDefaultSolver();
53 if (!solverRes.ok()) {
54 configured_ = false;
55 return solverRes;
56 }
57 return Result<void>::success();

73Result<void> GpuAgentSimulation::initialize(int count, const std::function<AgentState(int)>& spawnFn) {
74 if (!configured_) {
75 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
76 "simulation not configured", "configure", {}, "gpuagents"));
77 }
78 if (!spawnFn) {
79 return Result<void>::failure(
80 Diagnostic::error(DiagnosticCode::InvalidArgument, "spawnFn is empty", "spawnFn", {}, "gpuagents"));
81 }
82 if (count < 0 || count > capacity_) {
83 return Result<void>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument, "initialize count out of range",
84 "count", {}, "gpuagents"));
85 }
86 auto& buf = buffers_[front_];
87 for (int i = 0; i < capacity_; ++i) {
88 if (i < count) {
89 AgentState s = spawnFn(i);
90 s.alive = 1;
91 buf[static_cast<size_t>(i)] = s;
92 } else {
93 buf[static_cast<size_t>(i)] = AgentState{};
94 }
95 }
96 buffers_[1 - front_] = buf;
97 accumulator_ = 0.f;
98 return Result<void>::success();
99}

119Result<void> GpuAgentSimulation::step(float dt, const EnvironmentSnapshot& env) {
120 if (!configured_ || !solver_) {
121 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
122 "simulation not configured", "configure", {}, "gpuagents"));
123 }
124 EV_PARAM_CHECK(dt >= 0.f);
125 accumulator_ += dt;
126 const float fixed = profile_.base().fixedDt;
127 int guard = 0;
128 while (accumulator_ + 1e-8f >= fixed && guard < 8) {
129 const int read = front_;
130 const int write = 1 - front_;
131 solver_->step(buffers_[read], buffers_[write], profile_, env, fixed);
132 front_ = write;
133 accumulator_ -= fixed;
134 simTime_ += fixed;
135 ++stepCount_;
136 ++guard;
137 }
138 return Result<void>::success();
139}

211glm::vec3 unitFromIndex(int i, std::uint32_t seed) {
212 const float a = static_cast<float>((static_cast<std::uint32_t>(i) * 1103515245u + seed) & 0xFFFFu) / 65535.f;
213 const float b = static_cast<float>(((static_cast<std::uint32_t>(i) * 12345u) ^ seed) & 0xFFFFu) / 65535.f;
214 const float theta = a * 6.2831853f;
215 const float u = b * 2.f - 1.f;
216 const float r = std::sqrt(std::max(0.f, 1.f - u * u));
217 return glm::vec3(r * std::cos(theta), u, r * std::sin(theta));
218}

244Result<void> GpuAgents::spawnCloud(EffectBackend* backend, int count, float centerX, float centerY, float centerZ,
245 float spread) const {
246 if (!backend) {
247 return Result<void>::failure(
248 Diagnostic::error(DiagnosticCode::InvalidArgument, "backend is null", "backend", {}, "gpuagents"));
249 }
250 if (!backend->simulation().isConfigured()) {
251 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation, "backend not configured",
252 "configure", {}, "gpuagents"));
253 }
254 const glm::vec3 center(centerX, centerY, centerZ);
255 const auto& base = backend->simulation().profile().base();
256 const EffectKind kind = backend->kind();
257 return backend->simulation().initialize(count, [&](int i) {
258 AgentState s;
259 const glm::vec3 dir = unitFromIndex(i, base.seed);
260 s.position = center + dir * spread;
261 switch (kind) {
262 case EffectKind::Fish: s.velocity = dir * 2.f; break;
263 case EffectKind::Bird: s.velocity = glm::normalize(dir + glm::vec3(0.f, 0.2f, 0.f)) * 6.f; break;
264 case EffectKind::Petal:
265 s.velocity = glm::vec3(dir.x, -1.f, dir.z);
266 s.customData[1] = 0.75f + 0.5f * (static_cast<float>(i % 10) / 10.f);
267 s.rotation = lookRotation(glm::vec3(0.f, -1.f, 0.f));
268 break;
269 case EffectKind::LifeNetwork:
270 s.position.y = centerY;
271 s.velocity = glm::normalize(glm::vec3(dir.x, 0.f, dir.z) + glm::vec3(1e-3f, 0.f, 0.f)) * 1.5f;
272 break;
273 }
274 s.rotation = lookRotation(glm::length(s.velocity) > 1e-4f ? s.velocity : glm::vec3(0.f, 0.f, 1.f));
275 return s;
276 });
277}

280 const std::vector<std::uint32_t>& indices, float originX, float originY,
281 float originZ, float worldSize, int resolution) const {
282 if (!world) {
