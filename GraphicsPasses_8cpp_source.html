51void Graphics::createSceneColorResources(int width, int height) {
52 if (!device) return;
53 // Keep the main WebGPU scene single-sampled. Dawn/Metal does not complete
54 // the current MSAA resolve followed by synchronous scene readback chain.
55 // Temporal AA remains available; Vulkan owns the multisampled main path.
56 const uint32_t want = 1u;
57 if (sceneColorWidth == width && sceneColorHeight == height && sceneColorSamples == want && !sceneColorSlots.empty())
58 return;
59
60 destroySceneColorResources();
61 sceneColorWidth = width;
62 sceneColorHeight = height;
63 const uint32_t oldSamples = sceneColorSamples;
64 const bool samplesChanged = oldSamples != want;
65 sceneColorSamples = want;
66 sceneColorSlots.reserve(kFramesInFlight);
67 for (int s = 0; s < int(kFramesInFlight); ++s) {
68 sceneColorSlots.emplace_back();
69 SceneColorSlot& slot = sceneColorSlots.back();
70 slot.sampleCount = sceneColorSamples;
71
72 WGPUTextureDescriptor cd{};
73 cd.label = sv("eve_scene_color");
74 cd.dimension = WGPUTextureDimension_2D;
75 cd.size = {static_cast<uint32_t>(width), static_cast<uint32_t>(height), 1};
76 // Single-sample resolve target: composited to the swapchain and used
77 // as the frame-readback source. The multisampled color lives in
78 // msaaColor (created below) and is resolved into this texture.
79 cd.sampleCount = 1;
80 cd.format = sceneColorFormat;
81 cd.mipLevelCount = 1;
82 cd.usage = WGPUTextureUsage_TextureBinding | WGPUTextureUsage_RenderAttachment | WGPUTextureUsage_CopySrc;
83 slot.color = device.CreateTexture(reinterpret_cast<const wgpu::TextureDescriptor*>(&cd));
84 slot.colorView = slot.color.CreateView();
85
86 if (sceneColorSamples > 1) {
87 // Multisampled color target; the scene pass resolves it into the
88 // single-sample `color` above for compositing / readback.
89 WGPUTextureDescriptor mcd{};
90 mcd.label = sv("eve_scene_color_msaa");
91 mcd.dimension = WGPUTextureDimension_2D;
92 mcd.size = {static_cast<uint32_t>(width), static_cast<uint32_t>(height), 1};
93 mcd.sampleCount = sceneColorSamples;
94 mcd.format = sceneColorFormat;
95 mcd.mipLevelCount = 1;
96 mcd.usage = WGPUTextureUsage_RenderAttachment;
97 slot.msaaColor = device.CreateTexture(reinterpret_cast<const wgpu::TextureDescriptor*>(&mcd));
98 slot.msaaView = slot.msaaColor.CreateView();
99 }
100
101 WGPUTextureDescriptor dd{};
102 dd.label = sv("eve_scene_depth");
103 dd.dimension = WGPUTextureDimension_2D;
104 dd.size = {static_cast<uint32_t>(width), static_cast<uint32_t>(height), 1};
105 dd.sampleCount = sceneColorSamples;
106 dd.format = WGPUTextureFormat_Depth32Float;
107 dd.mipLevelCount = 1;
108 dd.usage = WGPUTextureUsage_RenderAttachment;
109 slot.depth = device.CreateTexture(reinterpret_cast<const wgpu::TextureDescriptor*>(&dd));
110 slot.depthView = slot.depth.CreateView();
111
112 slot.colorGpu.texture = slot.color;
113 slot.colorGpu.view = slot.colorView;
114 slot.colorGpu.width = width;
115 slot.colorGpu.height = height;
116 slot.colorGpu.sampler = createLinearSampler(device);
117 slot.colorGpu.samplerState = TextureSampler::linearMipmap();
118
119 slot.colorTex.gpuHandle = &slot.colorGpu;
120 slot.colorTex.width = width;
121 slot.colorTex.height = height;
122 slot.colorTex.mipmapCount = 1;
123 }
124 sceneColorTexture = &sceneColorSlots[0].colorTex;
125 // 3D pipelines that render into the scene target must match its sample
126 // count. Invalidate only the affected cached variants; they are rebuilt
127 // individually if a later draw actually needs them.
128 if (samplesChanged) {
129 mesh3dPipelines = {};
130 mesh3dPipeline = {};
131 mesh3dTransparentPipeline = {};
132 mesh3dClusteredPipeline = {};
133 voxelRectPipeline = {};
134 // Deferred lighting is 1x-only; recreate when scene sample count changes.
135 destroyDeferredLightingResources();
136 }
137 if (samplesChanged && gpuDrivenCullPipeline_) {
138 // The forward indirect pipeline inherits the scene target sample
139 // count. Recreate the GPU-driven layouts/pipelines lazily on the next
140 // cull after an MSAA toggle.
141 gpuDrivenCullPipeline_ = {};
142 gpuDrivenRenderPipeline_ = {};
143 gpuDrivenCanvasPipeline_ = {};
144 gpuDrivenComputePipelineLayout_ = {};
145 gpuDrivenRenderPipelineLayout_ = {};
146 gpuDrivenComputeSetLayout_ = {};
147 gpuDrivenRenderSetLayout_ = {};
148 gpuDrivenComputeBindGroup_ = {};
149 gpuDrivenRenderBindGroup_ = {};
150 }
151}

669void Graphics::flushGbufferPass(wgpu::RenderPassEncoder pass) {
670 if (gbufferPassDraws.empty() || gbufferSlots.empty()) return;
671 if (!mesh3dGbufferPipeline) createGbufferPipelines();
672 auto& uboArena = currentUboArena();
673 ensureUboArena(uboArena, uboArena.used + gbufferPassDraws.size() * 4096);
674 for (auto& d : gbufferPassDraws) {
675 auto* gpuMesh = static_cast<GpuMesh*>(d.mesh->gpuHandle);
676 if (!gpuMesh || !gpuMesh->vertexBuffer) continue;
677
678 if (d.pbrDraw) {
679 drawPbrMesh(pass, WGPUTextureFormat_RGBA8Unorm, true, *d.pbrDraw, *gpuMesh);
680 continue;
681 }
682 pass.SetPipeline(d.alphaTest ? mesh3dGbufferAlphaPipeline : mesh3dGbufferPipeline);
683
684 SkinPassUBO ubo;
685 ubo.mvp = d.mvp;
686 ubo.model = d.model;
687 auto u6 = [](float value) { return uint32_t(std::lround(std::clamp(value, 0.f, 1.f) * 63.f)); };
688 // Phase B: tint RGB6 | rough7 | metal7 — bit-cast (exceeds float's 24-bit exact ints).
689 const uint32_t rough7 = uint32_t(std::lround(std::clamp(d.roughness, 0.f, 1.f) * 127.f));
690 const uint32_t metal7 = uint32_t(std::lround(std::clamp(d.metallic, 0.f, 1.f) * 127.f));
691 const uint32_t packedTint =
692 u6(d.tint.r) | (u6(d.tint.g) << 6) | (u6(d.tint.b) << 12) | (rough7 << 18) | (metal7 << 25);
693 auto motion12 = [](float value) {
694 return uint32_t(std::lround(std::clamp(value, -1.f, 1.f) * 2047.f)) + 2047u;
695 };
696 const uint32_t packedMotion = motion12(d.motion.x) | (motion12(d.motion.y) << 12);
697 ubo.clip = glm::vec4(d.nearZ, d.farZ, std::bit_cast<float>(packedTint), float(packedMotion));
698 if (d.mesh->hasGpuSkinning()) ubo.skinInfo.x = static_cast<float>(d.mesh->getSkinPaletteCount());
699 ubo.skinInfo.y = static_cast<float>(d.skinInfluenceLimit);
700 const auto skinBuffer = uploadSkinPalette(d.mesh);
701
702 uint32_t offset = uboArena.alloc(sizeof(SkinPassUBO), 256);
703 queue.WriteBuffer(uboArena.buffer, offset, &ubo, sizeof(ubo));
704
705 GpuTexture* albedo = gpuForTextureOrWhite(d.albedo);
706 WGPUBindGroupEntry entries[4]{};
707 entries[0].binding = 0;
708 entries[0].buffer = uboArena.buffer.Get();
709 entries[0].size = sizeof(SkinPassUBO);
710 entries[1].binding = 1;
711 entries[1].textureView = albedo->view.Get();
712 entries[2].binding = 2;
713 entries[2].sampler = albedo->sampler.Get();
714 entries[3].binding = 3;
715 entries[3].buffer = skinBuffer.Get();
716 entries[3].size = skinBuffer.GetSize();
717 WGPUBindGroupDescriptor bgd{};
718 bgd.layout = gbufferSetLayout.Get();
719 bgd.entryCount = 4;
720 bgd.entries = entries;
721 wgpu::BindGroup bg = device.CreateBindGroup(reinterpret_cast<const wgpu::BindGroupDescriptor*>(&bgd));
722 uint32_t offsets[1] = {offset};
723 pass.SetBindGroup(0, bg, 1, offsets);
724
725 if (gpuMesh->indexBuffer) {
726 pass.SetVertexBuffer(0, gpuMesh->vertexBuffer, 0, uint64_t(gpuMesh->vertexCount) * gpuMesh->vertexStride);
727 const uint64_t indexBytes = gpuMesh->indexFormat == wgpu::IndexFormat::Uint16 ? 2u : 4u;
728 pass.SetIndexBuffer(gpuMesh->indexBuffer, gpuMesh->indexFormat, 0,
729 uint64_t(gpuMesh->indexCount) * indexBytes);
730 pass.DrawIndexed(gpuMesh->indexCount, 1, 0, 0, 0);
731 } else {
732 pass.SetVertexBuffer(0, gpuMesh->vertexBuffer, 0, uint64_t(gpuMesh->vertexCount) * gpuMesh->vertexStride);
733 pass.Draw(gpuMesh->vertexCount, 1, 0, 0);
734 }
735 }
736 gbufferPassDraws.clear();
737 gbufferPassPending = false;
