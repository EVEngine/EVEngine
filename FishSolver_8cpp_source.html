9void FishSolver::step(std::span<const AgentState> read, std::span<AgentState> write, const EffectProfile& profile,
10 const EnvironmentSnapshot& env, float dt) {
11 EV_PARAM_CHECK(profile.kind == EffectKind::Fish);
12 EV_PARAM_CHECK(read.size() == write.size());
13 const FishProfile& p = profile.fish;
14
15 for (size_t i = 0; i < read.size(); ++i) {
16 const AgentState& in = read[i];
17 if (in.alive == 0) {
18 write[i] = in;
19 continue;
20 }
21 glm::vec3 forward = in.velocity;
22 if (glm::length(forward) < 1e-4f) forward = glm::vec3(0.f, 0.f, 1.f);
23 else
24 forward = glm::normalize(forward);
25
26 const auto acc = detail::gatherNeighbors(read, static_cast<int>(i), forward, p.separationRadius,
27 p.cohesionRadius, p.alignmentRadius, p.perceptionFovDegrees,
28 p.maxNeighborSamples);
29
30 glm::vec3 force(0.f);
31 if (acc.sepWeightSum > 0.f) force += (acc.separation / acc.sepWeightSum) * p.separationWeight;
32 if (acc.cohWeightSum > 0.f) {
33 const glm::vec3 center = acc.cohesionPos / acc.cohWeightSum;
34 const glm::vec3 toC = center - in.position;
35 const float len = glm::length(toC);
36 if (len > 1e-4f) force += (toC / len) * p.cohesionWeight;
37 }
38 if (acc.aliWeightSum > 0.f) {
39 const glm::vec3 avgV = acc.alignmentVel / acc.aliWeightSum;
40 const glm::vec3 delta = avgV - in.velocity;
41 force += delta * p.alignmentWeight;
42 }
43
44 force += detail::markerForce(in.position, env.goals, true) * p.goalWeight;
45 force += detail::markerForce(in.position, env.dangers, false) * p.dangerWeight;
46 force += env.waterCurrent * p.currentWeight;
47
48 const float depthErr = (in.position.y - p.preferredDepth) / std::max(p.depthTolerance, 1e-3f);
49 force.y -= depthErr * p.depthWeight;
50
51 AgentState out;
52 detail::integrateBoid(out, in, force, dt, p.maxAcceleration, p.maxSpeed, p.maxHorizontalTurn,
53 p.maxVerticalTurn);
54 if (env.obstacles) {
55 env.obstacles->resolve(out.position, out.velocity, p.base.agentRadius, p.base.obstaclePredictTime);
56 out.rotation = lookRotation(out.velocity);
57 }
58 write[i] = out;
59 }
