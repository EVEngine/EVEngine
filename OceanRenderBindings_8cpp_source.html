49Result<HullMaskScriptPublication> decodeHullMask(const Value& value, std::shared_ptr<OceanRenderer> renderer) {
50 using namespace ocean_codec;
51 try {
52 keys(value, {"origin", "volumes"});
53 HullMaskScriptPublication result;
54 result.renderer = std::move(renderer);
55 result.origin = vector3(field(value, "origin"));
56 const auto& volumes = field(value, "volumes");
57 if (!volumes.isArray() || volumes.arraySize() > 32) throw std::runtime_error("At most 32 hull volumes");
58 auto vector4 = [](const Value& v) {
59 if (!v.isArray() || v.arraySize() != 4) throw std::runtime_error("Four plane/quaternion components");
60 return glm::dvec4(number(v.at(0)), number(v.at(1)), number(v.at(2)), number(v.at(3)));
61 };
62 for (std::size_t i = 0; i < volumes.arraySize(); ++i) {
63 const auto& v = volumes.at(i);
64 keys(v, {"center", "rotation", "halfExtent", "bevels"});
65 OceanHullVolume hull;
66 hull.center = vector3(field(v, "center"));
67 hull.halfExtent = vector3(field(v, "halfExtent"));
68 const auto q = vector4(field(v, "rotation"));
69 hull.rotation = {q.w, q.x, q.y, q.z};
70 const auto& bevels = field(v, "bevels");
71 if (!bevels.isArray() || bevels.arraySize() > 10) throw std::runtime_error("At most ten hull bevels");
72 for (std::size_t j = 0; j < bevels.arraySize(); ++j) hull.bevels.push_back(vector4(bevels.at(j)));
73 result.volumes.push_back(std::move(hull));
74 }
75 return Result<HullMaskScriptPublication>::success(std::move(result));
76 } catch (const std::runtime_error& error) {
77 return Result<HullMaskScriptPublication>::failure(invalid(error.what()));
78 }
79}

87Result<RenderBuilder> decodeSettings(const Value& v) {
88 using namespace ocean_codec;
89 try {
90 keys(v, {"segments", "rings", "firstRadius", "outerRadius", "backend"});
91 const auto& segments = field(v, "segments");
92 const auto& rings = field(v, "rings");
93 const auto& backend = field(v, "backend");
94 if (!segments.isInt64() || segments.asInt() < 16 || segments.asInt() > 1024 || !rings.isInt64() ||
95 rings.asInt() < 2 || rings.asInt() > 512 || !backend.isString() ||
96 (backend.asString() != "gpu" && backend.asString() != "cpu"))
97 return Result<RenderBuilder>::failure(invalid("Invalid render quality/backend"));
98 RenderBuilder out;
99 out.mesh = {static_cast<unsigned>(segments.asInt()), static_cast<unsigned>(rings.asInt()),
100 number(field(v, "firstRadius")), number(field(v, "outerRadius"))};
101 out.backend = backend.asString() == "gpu" ? OceanRenderBackend::GpuResident : OceanRenderBackend::CpuReference;
102 return Result<RenderBuilder>::success(out);
103 } catch (const std::runtime_error& error) {
104 return Result<RenderBuilder>::failure(invalid(error.what()));
105 }
106}
107Result<Placement> decodePlacement(const ssq::Object& object) {
108 auto v = script::valueFromSquirrel(object);
109 if (!v) return Result<Placement>::failure(v.status());
110 try {
111 ocean_codec::keys(v.value(), {"center", "origin"});
112 return Result<Placement>::success({ocean_codec::vector(ocean_codec::field(v.value(), "center")),
113 ocean_codec::vector3(ocean_codec::field(v.value(), "origin"))});
114 } catch (const std::runtime_error& error) {
115 return Result<Placement>::failure(invalid(error.what()));
116 }

118Result<void> publish(const Publication& p) {
119 const auto* wake = p.wake.isNull() ? nullptr : p.wake.to<OceanWake*>();
120 if (p.gpu) {
121 auto b = p.renderer->gpuFrameBuilder(*p.gpu, p.token);
122 b.center = p.placement.center;
123 b.origin = p.placement.origin;
124 b.wake = wake;
125 return b.publish();
126 }
127 const auto* frame = p.frame.to<OceanSpectrumFrame*>();
128 const auto water = wake ? OceanWaterView(*frame, *wake) : OceanWaterView(*frame);
129 auto b = p.renderer->frameBuilder(water);
130 b.center = p.placement.center;
131 b.origin = p.placement.origin;
132 auto prepared = b.build();
133 if (!prepared) return Result<void>::failure(prepared.status());
134 return p.renderer->publish(prepared.value());
135}
