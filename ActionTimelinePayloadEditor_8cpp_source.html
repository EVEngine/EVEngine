27std::optional<PayloadItem> findPayloadItem(const action::ActionTimeline& timeline, const LogicalId& itemId) {
28 for (const auto& track : timeline.tracks) {
29 const auto notify = std::find_if(track.notifies.begin(), track.notifies.end(),
30 [&](const auto& value) { return value.id == itemId; });
31 if (notify != track.notifies.end())
32 return PayloadItem{track.id, notify->type, action::ActionTimelineEventKind::Notify, notify->time,
33 notify->time, notify->payload};
34 const auto state = std::find_if(track.states.begin(), track.states.end(),
35 [&](const auto& value) { return value.id == itemId; });
36 if (state != track.states.end())
37 return PayloadItem{track.id, state->type, action::ActionTimelineEventKind::StateEnter, state->start,
38 state->end, state->payload};
39 }
40 return std::nullopt;
41}

66EditorResult<void> ActionTimelinePayloadEditor::fitBlockToClip(const LogicalId& itemId) {
67 const auto item = findPayloadItem(editor_.target().timeline(), itemId);
68 if (!item)
69 return payloadError("editor.action.timeline.payload-item-missing",
70 "Editable action-block payload was not found");
71 if (item->kind != action::ActionTimelineEventKind::StateEnter ||
72 item->type.format() != "presentation:vfx-state")
73 return payloadError("editor.action.timeline.clip-fit-type", "Only VFX states can fit a block to a clip");
74 auto binding = action::ActionVfxBinding::fromPayload(item->payload, action::ActionVfxShape::State);
75 if (!binding) return EditorResult<void>::failure(binding.status());
76 auto duration = Duration::fromSeconds(binding.value().clipEndTime - binding.value().clipStartTime);
77 if (!duration) return EditorResult<void>::failure(duration.status());
78 auto end = item->time.tryAdd(duration.value());
79 if (!end)
80 return payloadError("editor.action.timeline.clip-fit-overflow", "Fitted VFX block time overflowed");
81 return editor_.editItem(itemId, item->time, end.value(), item->type, item->payload);
82}
