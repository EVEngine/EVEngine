76inline RadialFillMesh radialFillMesh(float fraction, float startDegrees, bool clockwise, float innerRadius) {
77 RadialFillMesh mesh;
78 const float f = hudFraction(fraction);
79 if (f <= 0.f || !std::isfinite(startDegrees)) return mesh;
80 const float inner = hudFraction(innerRadius);
81 if (inner >= 1.f) return mesh;
82 constexpr float pi = 3.14159265358979323846f;
83 const float start = std::fmod(startDegrees, 360.f) * pi / 180.f;
84 const float sweep = f * 2.f * pi * (clockwise ? 1.f : -1.f);
85 mesh.count = std::max(1, int(std::ceil(f * 64.f)));
86 for (int i = 0; i < mesh.count; ++i) {
87 const float a = start + sweep * float(i) / float(mesh.count);
88 const float b = start + sweep * float(i + 1) / float(mesh.count);
89 const float ax = std::cos(a) * .5f, ay = std::sin(a) * .5f;
90 const float bx = std::cos(b) * .5f, by = std::sin(b) * .5f;
91 mesh.quads[std::size_t(i)] = {.5f + ax, .5f + ay, .5f + bx, .5f + by,
92 .5f + ax * inner, .5f + ay * inner, .5f + bx * inner, .5f + by * inner};
93 }
94 return mesh;
95}
