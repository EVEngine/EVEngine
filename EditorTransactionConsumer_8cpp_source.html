53std::optional<std::size_t> coordinatorFailedCommitIndex(const eve::Status& status) noexcept {
54 constexpr std::string_view prefix = "transaction.commit[";
55 for (const auto& diagnostic : status.diagnostics()) {
56 const std::string& path = diagnostic.path();
57 if (!path.starts_with(prefix) || path.back() != ']') continue;
58 std::uint64_t index = 0;
59 const char* first = path.data() + prefix.size();
60 const char* last = path.data() + path.size() - 1;
61 const auto [end, error] = std::from_chars(first, last, index);
62 if (error == std::errc{} && end == last &&
63 index <= static_cast<std::uint64_t>(std::numeric_limits<std::size_t>::max()))
64 return static_cast<std::size_t>(index);
65 }
66 return std::nullopt;
67}

69bool hasPreviouslyCommittedAdditionalParticipant(const eve::Status& status,
70 std::size_t editorParticipantCount) noexcept {
71 const auto failedIndex = coordinatorFailedCommitIndex(status);
72 // Editor-owned participants are rebuilt for every attempt. A borrowed
73 // additional participant that precedes the failing commit is different:
74 // its effect may already have escaped, and retaining only its name is not
75 // enough to prove that a retry is idempotent. Require explicit discard or
76 // reconciliation instead of invoking that participant twice.
77 return failedIndex.has_value() && *failedIndex > editorParticipantCount;
78}

85class CommandParticipant final : public transaction::ITransactionParticipant {
86public:
87 enum class InitialPhase { Idle, Committed };
88
89 explicit CommandParticipant(CommandBatch& batch, InitialPhase initial = InitialPhase::Idle)
90 : batch_(batch), phase_(initial == InitialPhase::Committed ? Phase::Committed : Phase::Idle) {}
91
92 std::string_view name() const noexcept override { return "editor.commands"; }
93
94 [[nodiscard]] eve::Result<void> prepare(const transaction::TransactionContext&) override {
95 if (phase_ != Phase::Idle)
96 return failure<void>(eve::DiagnosticCode::PreconditionViolation, "editor command participant is not idle",
97 "editor.commands.prepare");
98 if (batch_.commands.empty())
99 return failure<void>(eve::DiagnosticCode::InvalidArgument,
100 "editor transaction must contain at least one command", "editor.commands");
101 for (const auto& command : batch_.commands) {
102 if (!command)
103 return failure<void>(eve::DiagnosticCode::InvalidArgument, "editor command must not be null",
104 "editor.commands");
105 }
106 phase_ = Phase::Prepared;
107 return eve::Result<void>::success();
108 }
109
110 [[nodiscard]] eve::Result<void> commit(const transaction::TransactionContext&) override {
111 if (phase_ != Phase::Prepared)
112 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
113 "editor command participant is not prepared", "editor.commands.commit");
114 if (!batch_.previewApplied) {
115 std::size_t applied = 0;
116 try {
117 for (const auto& command : batch_.commands) {
118 if (!command->apply()) {
119 const bool reverted = revertPrefix(applied);
120 if (!reverted) batch_.previewApplied = true;
121 return failure<void>(
122 reverted ? eve::DiagnosticCode::InvalidArgument : eve::DiagnosticCode::Failed,
123 reverted ? "editor command rejected during commit"
124 : "editor command commit failed and compensation was incomplete",
125 "editor.commands.commit");
126 }
127 ++applied;
128 }
129 } catch (const std::exception& exception) {
130 const bool reverted = revertPrefix(applied);
131 if (!reverted) batch_.previewApplied = true;
132 return failure<void>(eve::DiagnosticCode::Failed,
133 std::string("editor command threw during commit: ") + exception.what(),
134 "editor.commands.commit");
135 } catch (...) {
136 const bool reverted = revertPrefix(applied);
137 if (!reverted) batch_.previewApplied = true;
138 return failure<void>(eve::DiagnosticCode::Failed, "editor command threw during commit",
139 "editor.commands.commit");
140 }
141 batch_.previewApplied = true;
142 }
143 phase_ = Phase::Committed;
144 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
145 }
146
147 [[nodiscard]] eve::Result<void> rollback(const transaction::TransactionContext&) override {
148 if (phase_ != Phase::Prepared)
149 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
150 "editor command participant is not prepared", "editor.commands.rollback");
151 if (batch_.previewApplied && !revertAll())
152 return failure<void>(eve::DiagnosticCode::Failed, "editor command preview rollback was incomplete",
153 "editor.commands.rollback");
154 batch_.previewApplied = false;
155 phase_ = Phase::RolledBack;
156 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
157 }
158
159 [[nodiscard]] eve::Result<void> compensate(const transaction::TransactionContext&) override {
160 if (phase_ != Phase::Committed)
161 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
162 "editor command participant is not committed", "editor.commands.compensate");
163 if (!batch_.previewApplied || !revertAll())
164 return failure<void>(eve::DiagnosticCode::Failed, "editor command compensation was incomplete",
165 "editor.commands.compensate");
166 batch_.previewApplied = false;
167 phase_ = Phase::Compensated;
168 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
169 }
170
171private:
172 enum class Phase { Idle, Prepared, Committed, RolledBack, Compensated };
173
174 bool revertPrefix(std::size_t count) noexcept {
175 try {
176 while (count > 0) batch_.commands[--count]->revert();
177 return true;
178 } catch (...) {
179 return false;
180 }
181 }
182
183 bool revertAll() noexcept { return revertPrefix(batch_.commands.size()); }
184
185 CommandBatch& batch_;
186 Phase phase_ = Phase::Idle;
187};

189class AuthorityParticipant final : public transaction::ITransactionParticipant {
190public:
191 AuthorityParticipant(IEditAuthority& authority, TransactionSpec specification,
192 std::vector<DomainOperation> operations)
193 : authority_(&authority), specification_(std::move(specification)), operations_(std::move(operations)) {}
194
195 AuthorityParticipant(IEditAuthority& authority, TransactionSpec specification,
196 std::vector<DomainOperation> operations, TransactionReceipt receipt)
197 : authority_(&authority),
198 specification_(std::move(specification)),
199 operations_(std::move(operations)),
200 receipt_(std::move(receipt)),
201 phase_(Phase::Committed) {}
202
203 std::string_view name() const noexcept override { return "editor.authority"; }
204
205 [[nodiscard]] eve::Result<void> prepare(const transaction::TransactionContext&) override {
206 if (phase_ != Phase::Idle)
207 return failure<void>(eve::DiagnosticCode::PreconditionViolation, "editor authority participant is not idle",
208 "editor.authority.prepare");
209 if (!authority_)
210 return failure<void>(eve::DiagnosticCode::Failed, "editor authority participant has no authority",
211 "editor.authority");
212 auto plan = authority_->preflight(specification_, operations_);
213 if (!plan.ok()) return convertEditorFailure<void>(plan, "editor authority preflight");
214 plan_ = std::move(plan.value());
215 phase_ = Phase::Prepared;
216 return eve::Result<void>::success();
217 }
218
219 [[nodiscard]] eve::Result<void> commit(const transaction::TransactionContext&) override {
220 if (phase_ != Phase::Prepared || !plan_)
221 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
222 "editor authority participant is not prepared", "editor.authority.commit");
223 auto result = authority_->commit(*plan_);
224 if (!result.ok()) return convertEditorFailure<void>(result, "editor authority commit");
225 receipt_ = std::move(result.value());
226 phase_ = Phase::Committed;
227 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
228 }
229
230 [[nodiscard]] eve::Result<void> rollback(const transaction::TransactionContext&) override {
231 if (phase_ != Phase::Prepared)
232 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
233 "editor authority participant is not prepared", "editor.authority.rollback");
234 plan_.reset();
235 phase_ = Phase::RolledBack;
236 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
237 }
238
239 [[nodiscard]] eve::Result<void> compensate(const transaction::TransactionContext&) override {
240 if (phase_ != Phase::Committed || !receipt_)
241 return failure<void>(eve::DiagnosticCode::PreconditionViolation,
242 "editor authority participant has no committed receipt",
243 "editor.authority.compensate");
244 auto result = authority_->compensate(*receipt_);
245 if (!result.ok())
246 return convertEditorFailure<void>(result, "editor authority compensation");
247 // History keeps the original commit receipt. This participant's
248 // latest effect is the compensation receipt, whose afterRevision is
249 // also the base revision required by a later redo.
250 receipt_ = std::move(result.value());
251 phase_ = Phase::Compensated;
252 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
253 }
254
255 [[nodiscard]] const std::optional<TransactionReceipt>& receipt() const noexcept { return receipt_; }
256

258 enum class Phase { Idle, Prepared, Committed, RolledBack, Compensated };

260 IEditAuthority* authority_ = nullptr;
261 TransactionSpec specification_;
262 std::vector<DomainOperation> operations_;
263 std::optional<AuthorityPlan> plan_;
264 std::optional<TransactionReceipt> receipt_;
265 Phase phase_ = Phase::Idle;
266};

344eve::Result<void> EditorTransactionConsumer::append(std::unique_ptr<IEditCommand> command) {
345 if (!impl_->pending)
346 return failure<void>(eve::DiagnosticCode::PreconditionViolation, "no editor transaction is active",
347 "editor.transaction");
348 if (!command)
349 return failure<void>(eve::DiagnosticCode::InvalidArgument, "editor command must not be null", "editor.command");
350 if (!impl_->pending->operations.empty())
351 return failure<void>(eve::DiagnosticCode::Conflict,
352 "commands and authority operations cannot share one editor transaction",

358 impl_->pending->previewMode = false;
359 if (!impl_->pending->commands) impl_->pending->commands = std::make_unique<CommandBatch>();
360 if (!impl_->pending->commands->commands.empty()) {

362 if (impl_->pending->commands->commands.back()->mergeWith(*command)) return eve::Result<void>::success();
363 } catch (const std::exception& exception) {

365 std::string("editor command merge threw: ") + exception.what(),
366 "editor.command.merge");
367 } catch (...) {
368 return failure<void>(eve::DiagnosticCode::Failed, "editor command merge threw", "editor.command.merge");
369 }
370 }
371 impl_->pending->commands->commands.push_back(std::move(command));
372 return eve::Result<void>::success();

375eve::Result<void> EditorTransactionConsumer::appendPreview(std::unique_ptr<IEditCommand> command) {
376 if (!impl_->pending)
377 return failure<void>(eve::DiagnosticCode::PreconditionViolation, "no editor transaction is active",
378 "editor.transaction");
379 if (!command)
380 return failure<void>(eve::DiagnosticCode::InvalidArgument, "editor command must not be null", "editor.command");
381 if (!impl_->pending->operations.empty())
382 return failure<void>(eve::DiagnosticCode::Conflict,
383 "commands and authority operations cannot share one editor transaction",
384 "editor.transaction");
385 if (impl_->pending->commandModeSet && !impl_->pending->previewMode)
386 return failure<void>(eve::DiagnosticCode::Conflict, "a preview command cannot follow a strict command",
387 "editor.transaction");
388 if (!impl_->pending->commands) impl_->pending->commands = std::make_unique<CommandBatch>();
389
390 // A command is applied only after merge evaluation has completed on an
391 // owning candidate copy. The candidate carries the aggregate history but
392 // never publishes to the real target; the incoming command is the only
393 // object that applies the next preview delta.
394 std::unique_ptr<IEditCommand> mergedCandidate;
395 if (!impl_->pending->commands->commands.empty()) {
396 try {
397 mergedCandidate = impl_->pending->commands->commands.back()->clone();
398 } catch (const std::exception& exception) {
399 return failure<void>(eve::DiagnosticCode::Failed,
400 std::string("editor command staging threw: ") + exception.what(),
401 "editor.command.stage");
402 } catch (...) {
403 return failure<void>(eve::DiagnosticCode::Failed, "editor command staging threw", "editor.command.stage");
404 }
405 if (mergedCandidate) {
406 try {
407 if (mergedCandidate->mergeWith(*command)) {
408 // The merge has changed only mergedCandidate. Applying the
409 // incoming delta keeps the live target at the same state
410 // that the command recorded as its `before` value.
411 bool applied = false;
412 try {
413 applied = command->apply();
414 } catch (const std::exception& exception) {
415 try {
416 command->revert();
417 } catch (...) {
418 return failure<void>(eve::DiagnosticCode::Failed,
419 "editor command preview failed and could not be reverted",
420 "editor.command.preview");
421 }
422 return failure<void>(eve::DiagnosticCode::Failed,
423 std::string("editor command preview threw: ") + exception.what(),
424 "editor.command.preview");
425 } catch (...) {
426 try {
427 command->revert();
428 } catch (...) {
429 return failure<void>(eve::DiagnosticCode::Failed,
430 "editor command preview failed and could not be reverted",
431 "editor.command.preview");
432 }
433 return failure<void>(eve::DiagnosticCode::Failed, "editor command preview threw",
434 "editor.command.preview");
435 }
436 if (!applied) {
437 try {
438 command->revert();
439 } catch (...) {
440 return failure<void>(eve::DiagnosticCode::Failed,
441 "editor command preview was rejected and could not be reverted",
442 "editor.command.preview");
443 }
444 return failure<void>(eve::DiagnosticCode::InvalidArgument,
445 "editor command preview was rejected", "editor.command.preview");
446 }
447 impl_->pending->commands->commands.back() = std::move(mergedCandidate);
448 impl_->pending->commands->previewApplied = true;
449 impl_->pending->commandModeSet = true;
450 impl_->pending->previewMode = true;
451 return eve::Result<void>::success();
452 }
453 } catch (const std::exception& exception) {
454 return failure<void>(eve::DiagnosticCode::Failed,
455 std::string("editor command merge threw: ") + exception.what(),
456 "editor.command.merge");
457 } catch (...) {
458 return failure<void>(eve::DiagnosticCode::Failed, "editor command merge threw", "editor.command.merge");
459 }
460 }
461 }
462
463 try {
464 // Reserve before applying so vector growth cannot be the first
465 // observable failure after the target has been changed.
466 impl_->pending->commands->commands.reserve(impl_->pending->commands->commands.size() + 1);
467 } catch (const std::exception& exception) {
468 return failure<void>(eve::DiagnosticCode::Failed,
469 std::string("editor command staging allocation threw: ") + exception.what(),
470 "editor.command.stage");
471 } catch (...) {
472 return failure<void>(eve::DiagnosticCode::Failed, "editor command staging allocation threw",
473 "editor.command.stage");
474 }
475
476 bool applied = false;
477 try {
478 applied = command->apply();
479 } catch (const std::exception& exception) {
480 try {
481 command->revert();
482 } catch (...) {
483 return failure<void>(eve::DiagnosticCode::Failed, "editor command preview failed and could not be reverted",
484 "editor.command.preview");
485 }
486 return failure<void>(eve::DiagnosticCode::Failed,
487 std::string("editor command preview threw: ") + exception.what(),
488 "editor.command.preview");
489 } catch (...) {
490 try {
491 command->revert();
492 } catch (...) {
493 return failure<void>(eve::DiagnosticCode::Failed, "editor command preview failed and could not be reverted",
494 "editor.command.preview");
495 }
496 return failure<void>(eve::DiagnosticCode::Failed, "editor command preview threw", "editor.command.preview");
497 }
498 if (!applied) {
499 try {
500 command->revert();
501 } catch (...) {
502 return failure<void>(eve::DiagnosticCode::Failed,
503 "editor command preview was rejected and could not be reverted",
504 "editor.command.preview");
505 }
506 return failure<void>(eve::DiagnosticCode::InvalidArgument, "editor command preview was rejected",
507 "editor.command.preview");
508 }
509 impl_->pending->commands->commands.push_back(std::move(command));
510 impl_->pending->commands->previewApplied = true;
511 impl_->pending->commandModeSet = true;
512 impl_->pending->previewMode = true;
513 return eve::Result<void>::success();
514}

538eve::Result<EditorDryRunReport> EditorTransactionConsumer::dryRun() const {
539 if (!impl_->pending)
540 return failure<EditorDryRunReport>(eve::DiagnosticCode::PreconditionViolation,
541 "no editor transaction is active", "editor.transaction");
542 const auto& pending = *impl_->pending;
543 const std::size_t commandCount = pending.commands ? pending.commands->commands.size() : 0;
544 if (commandCount == 0 && pending.operations.empty())
545 return failure<EditorDryRunReport>(eve::DiagnosticCode::InvalidArgument, "editor transaction has no work",
546 "editor.transaction");
547 EditorDryRunReport report;
548 report.specification = pending.specification;
549 report.commandCount = commandCount;
550 report.operationCount = pending.operations.size();
551 if (!pending.operations.empty()) {
552 if (!impl_->authority)
553 return failure<EditorDryRunReport>(eve::DiagnosticCode::Unsupported,
554 "authority preflight requires an injected authority",
555 "editor.authority");
556 auto plan = impl_->authority->preflight(pending.specification, pending.operations);
557 if (!plan.ok())
558 return convertEditorFailure<EditorDryRunReport>(plan, "editor authority preflight");
559 report.authorityPlan = std::move(plan.value());
560 }
561 return eve::Result<EditorDryRunReport>::success(std::move(report));
562}

573eve::Result<EditorTransactionRecord> EditorTransactionConsumer::retry(
574 std::span<transaction::ITransactionParticipant*> additional) {
575 if (!impl_->pending)
576 return failure<EditorTransactionRecord>(eve::DiagnosticCode::PreconditionViolation,
577 "no failed editor transaction is pending", "editor.transaction.retry");
578 if (impl_->state != EditorCommitState::FailedRetryable)
579 return failure<EditorTransactionRecord>(eve::DiagnosticCode::PreconditionViolation,
580 "editor transaction is not awaiting retry", "editor.transaction.retry");
581 if (!impl_->retryAllowed)
582 return failure<EditorTransactionRecord>(
583 eve::DiagnosticCode::Conflict,
584 impl_->retryBlockedByCommittedParticipant
585 ? "editor transaction cannot be retried after an additional participant committed; reconcile or "
586 "discard it"
587 : "editor transaction cannot be retried after incomplete cleanup; discard it",
588 "editor.transaction.retry");
589 return commitAttempt(additional, true);

592eve::Result<EditorTransactionRecord> EditorTransactionConsumer::commitAttempt(
593 std::span<transaction::ITransactionParticipant*> additional, bool retryAttempt) {
594 if (!impl_->pending)
595 return failure<EditorTransactionRecord>(eve::DiagnosticCode::PreconditionViolation,
596 "no editor transaction is active", "editor.transaction");
597
598 auto retainFailure = [&](const eve::Status& status, bool retryAllowed) {
599 impl_->state = EditorCommitState::FailedRetryable;
600 impl_->retryAllowed = retryAllowed;
601 impl_->retryBlockedByCommittedParticipant = false;
602 impl_->diagnostics = status.diagnostics();
603 return eve::Result<EditorTransactionRecord>::failure(status);
604 };
605 auto reject = [&](eve::DiagnosticCode code, std::string message, std::string path) {
606 auto result = failure<EditorTransactionRecord>(code, std::move(message), std::move(path));
607 const bool preserveRetryBlock = impl_->state == EditorCommitState::FailedRetryable;
608 impl_->state = EditorCommitState::FailedRetryable;
609 if (!preserveRetryBlock) {
610 impl_->retryAllowed = true;
611 impl_->retryBlockedByCommittedParticipant = false;
612 }
613 impl_->diagnostics = result.diagnostics();
614 return result;
615 };
616
617 if (!retryAttempt && impl_->state == EditorCommitState::FailedRetryable)
618 return reject(eve::DiagnosticCode::Conflict,
619 "editor commit failed; call retry() or discard() before another commit attempt",
620 "editor.transaction.commit");
621 if (retryAttempt && impl_->state != EditorCommitState::FailedRetryable)
622 return reject(eve::DiagnosticCode::PreconditionViolation, "editor transaction is not awaiting retry",
623 "editor.transaction.retry");
624
625 Impl::Pending& pending = *impl_->pending;
626 const std::size_t commandCount = pending.commands ? pending.commands->commands.size() : 0;
627 if (commandCount == 0 && pending.operations.empty())
628 return reject(eve::DiagnosticCode::InvalidArgument, "editor transaction has no work", "editor.transaction");
629 if (!pending.operations.empty() && !impl_->authority)
630 return reject(eve::DiagnosticCode::Unsupported, "authority commit requires an injected authority",
631 "editor.authority");
632
633 for (std::size_t i = 0; i < additional.size(); ++i) {
634 if (!additional[i])
635 return reject(eve::DiagnosticCode::InvalidArgument, "additional participant must not be null",
636 "editor.transaction.participants[" + std::to_string(i) + "]");
637 for (std::size_t previous = 0; previous < i; ++previous) {
638 if (additional[previous] == additional[i])
639 return reject(eve::DiagnosticCode::Conflict,
640 "an additional participant may occur only once in an editor transaction",
641 "editor.transaction.participants[" + std::to_string(i) + "]");
642 }
643 }
644
645 if (!retryAttempt) {
646 pending.additionalNames.clear();
647 pending.additionalNames.reserve(additional.size());
648 for (auto* participant : additional) pending.additionalNames.emplace_back(participant->name());
649 } else {
650 if (additional.size() != pending.additionalNames.size())
651 return reject(eve::DiagnosticCode::Conflict, "retry must provide the same additional participant set",
652 "editor.transaction.retry");
653 for (std::size_t i = 0; i < additional.size(); ++i) {
654 if (additional[i]->name() != pending.additionalNames[i])
655 return reject(eve::DiagnosticCode::Conflict,
656 "retry participant order or identity does not match the failed attempt",
657 "editor.transaction.retry");
658 }
659 }
660
661 std::unique_ptr<AuthorityParticipant> authorityParticipant;
662 std::unique_ptr<CommandParticipant> commandParticipant;
663 std::vector<transaction::ITransactionParticipant*> participants;
664 if (!pending.operations.empty()) {
665 authorityParticipant =
666 std::make_unique<AuthorityParticipant>(*impl_->authority, pending.specification, pending.operations);
667 participants.push_back(authorityParticipant.get());
668 }
669 if (pending.commands) {
670 commandParticipant = std::make_unique<CommandParticipant>(*pending.commands);
671 participants.push_back(commandParticipant.get());
672 }
673 participants.insert(participants.end(), additional.begin(), additional.end());
674 if (participants.empty())
675 return reject(eve::DiagnosticCode::InvalidArgument, "editor transaction has no participant",
676 "editor.transaction");
677
678 if (retryAttempt) {
679 if (pending.attemptSequence == std::numeric_limits<std::uint64_t>::max())
680 return reject(eve::DiagnosticCode::Failed, "editor transaction retry sequence is exhausted",
681 "editor.transaction.retry");
682 ++pending.attemptSequence;
683 }
684 std::string attemptId = pending.specification.id.value();
685 if (retryAttempt) attemptId += ".retry." + std::to_string(pending.attemptSequence);
686 transaction::TransactionContext context(std::move(attemptId));
687 auto coordinated = impl_->coordinator.execute(context, participants);
688 if (!coordinated.ok()) {
689 const eve::Status& status = coordinated.status();
690 const bool committedAdditional = hasPreviouslyCommittedAdditionalParticipant(
691 status, (pending.operations.empty() ? 0u : 1u) + (pending.commands ? 1u : 0u));
692 auto result = retainFailure(status, !hasCoordinatorCleanupFailure(status) && !committedAdditional);
693 impl_->retryBlockedByCommittedParticipant = committedAdditional;
694 return result;
695 }
696 transaction::TransactionReceipt coordinatorReceipt = std::move(coordinated).takeValue();
697
698 Impl::HistoryEntry history;
699 history.specification = pending.specification;
700 history.commands = std::move(pending.commands);
701 history.operations = std::move(pending.operations);
702 if (authorityParticipant) history.authorityReceipt = authorityParticipant->receipt();
703 impl_->undo.push_back(std::move(history));
704 impl_->redo.clear();
705 impl_->pending.reset();
706 impl_->state = EditorCommitState::Committed;
707 impl_->retryAllowed = false;
708 impl_->retryBlockedByCommittedParticipant = false;
709 impl_->diagnostics.clear();
710
711 EditorTransactionRecord record;
712 record.coordinator = std::move(coordinatorReceipt);
713 record.specification = impl_->undo.back().specification;
714 record.authorityReceipt = impl_->undo.back().authorityReceipt;
715 record.commandCount = commandCount;
716 record.operationCount = impl_->undo.back().operations.size();
717 return eve::Result<EditorTransactionRecord>::success(std::move(record));
718}

720eve::Result<void> EditorTransactionConsumer::discard() {
721 if (!impl_->pending)
722 return failure<void>(eve::DiagnosticCode::PreconditionViolation, "no editor transaction is active",
723 "editor.transaction");
724 Impl::Pending& pending = *impl_->pending;
725 if (pending.commands && pending.commands->previewApplied) {
726 try {
727 for (auto it = pending.commands->commands.rbegin(); it != pending.commands->commands.rend(); ++it)
728 (*it)->revert();
729 } catch (const std::exception& exception) {
730 auto result = failure<void>(eve::DiagnosticCode::Failed,
731 std::string("editor preview rollback threw: ") + exception.what(),
732 "editor.transaction.discard");
733 impl_->state = EditorCommitState::FailedRetryable;
734 impl_->retryAllowed = false;
735 impl_->retryBlockedByCommittedParticipant = false;
736 impl_->diagnostics = result.diagnostics();
737 return result;
738 } catch (...) {
739 auto result = failure<void>(eve::DiagnosticCode::Failed, "editor preview rollback threw",
740 "editor.transaction.discard");
741 impl_->state = EditorCommitState::FailedRetryable;
742 impl_->retryAllowed = false;
743 impl_->retryBlockedByCommittedParticipant = false;
744 impl_->diagnostics = result.diagnostics();
745 return result;
746 }
747 pending.commands->previewApplied = false;
748 }
749 impl_->pending.reset();
750 impl_->state = EditorCommitState::Discarded;
751 impl_->retryAllowed = false;
752 impl_->retryBlockedByCommittedParticipant = false;
753 impl_->diagnostics.clear();
754 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
755}

759eve::Result<EditorTransactionRecord> EditorTransactionConsumer::undo() {
760 if (impl_->pending)
761 return failure<EditorTransactionRecord>(eve::DiagnosticCode::PreconditionViolation,
762 "cannot undo while an editor transaction is active",
763 "editor.transaction.undo");
764 if (impl_->undo.empty())
765 return failure<EditorTransactionRecord>(eve::DiagnosticCode::NotFound, "there is no editor transaction to undo",
766 "editor.history.undo");
767 Impl::HistoryEntry history = std::move(impl_->undo.back());
768 impl_->undo.pop_back();
769
770 std::unique_ptr<AuthorityParticipant> authorityParticipant;
771 std::unique_ptr<CommandParticipant> commandParticipant;
772 std::vector<transaction::ITransactionParticipant*> participants;
773 if (history.authorityReceipt && impl_->authority) {
774 authorityParticipant = std::make_unique<AuthorityParticipant>(*impl_->authority, history.specification,
775 history.operations, *history.authorityReceipt);
776 participants.push_back(authorityParticipant.get());
777 }
778 if (history.commands) {
779 commandParticipant =
780 std::make_unique<CommandParticipant>(*history.commands, CommandParticipant::InitialPhase::Committed);
781 participants.push_back(commandParticipant.get());
782 }
783 if (participants.empty()) {
784 impl_->undo.push_back(std::move(history));
785 return failure<EditorTransactionRecord>(eve::DiagnosticCode::Unsupported,
786 "editor history entry has no compensatable participant",
787 "editor.history.undo");
788 }
789 const std::string compensationId =
790 history.specification.id.value() + ".undo." + std::to_string(++impl_->redoSequence);
791 transaction::TransactionContext context(compensationId);
792 auto compensated = impl_->coordinator.compensate(context, participants);
793 if (!compensated.ok()) {
794 impl_->undo.push_back(std::move(history));
795 return eve::Result<EditorTransactionRecord>::failure(compensated.status());
796 }
797 transaction::TransactionReceipt receipt = std::move(compensated).takeValue();
798 if (authorityParticipant && authorityParticipant->receipt())
799 history.specification.baseRevision = authorityParticipant->receipt()->afterRevision;
800 impl_->redo.push_back(std::move(history));
801 impl_->state = EditorCommitState::Discarded;
802 const auto& moved = impl_->redo.back();
803 EditorTransactionRecord record;
804 record.coordinator = std::move(receipt);
805 record.specification = moved.specification;
806 record.authorityReceipt = moved.authorityReceipt;
807 record.commandCount = moved.commands ? moved.commands->commands.size() : 0;
808 record.operationCount = moved.operations.size();
809 return eve::Result<EditorTransactionRecord>::success(std::move(record));
810}

812eve::Result<EditorTransactionRecord> EditorTransactionConsumer::redo() {
813 if (impl_->pending)
814 return failure<EditorTransactionRecord>(eve::DiagnosticCode::PreconditionViolation,
815 "cannot redo while an editor transaction is active",
816 "editor.transaction.redo");
817 if (impl_->redo.empty())
818 return failure<EditorTransactionRecord>(eve::DiagnosticCode::NotFound, "there is no editor transaction to redo",
819 "editor.history.redo");
820 Impl::HistoryEntry history = std::move(impl_->redo.back());
821 impl_->redo.pop_back();
822 if (history.operations.empty() && !history.commands) {
823 impl_->redo.push_back(std::move(history));
824 return failure<EditorTransactionRecord>(eve::DiagnosticCode::Unsupported,
825 "editor history entry has no replayable participant",
826 "editor.history.redo");
827 }
828 if (!history.operations.empty() && !impl_->authority) {
829 impl_->redo.push_back(std::move(history));
830 return failure<EditorTransactionRecord>(eve::DiagnosticCode::Unsupported,
831 "authority redo requires an injected authority", "editor.authority");
832 }
833
834 TransactionSpec specification = history.specification;
835 specification.id = TransactionId(specification.id.value() + ".redo." + std::to_string(++impl_->redoSequence));
836 std::unique_ptr<AuthorityParticipant> authorityParticipant;
837 std::unique_ptr<CommandParticipant> commandParticipant;
838 std::vector<transaction::ITransactionParticipant*> participants;
839 if (!history.operations.empty()) {
840 authorityParticipant =
841 std::make_unique<AuthorityParticipant>(*impl_->authority, specification, history.operations);
842 participants.push_back(authorityParticipant.get());
843 }
844 if (history.commands) {
845 commandParticipant = std::make_unique<CommandParticipant>(*history.commands);
846 participants.push_back(commandParticipant.get());
847 }
848 transaction::TransactionContext context(specification.id.value());
849 auto replayed = impl_->coordinator.execute(context, participants);
850 if (!replayed.ok()) {
851 impl_->redo.push_back(std::move(history));
852 return eve::Result<EditorTransactionRecord>::failure(replayed.status());
853 }
854 transaction::TransactionReceipt receipt = std::move(replayed).takeValue();
855 history.specification = specification;
856 if (authorityParticipant) history.authorityReceipt = authorityParticipant->receipt();
857 impl_->undo.push_back(std::move(history));
858 impl_->state = EditorCommitState::Committed;
859 if (!impl_->redo.empty() && authorityParticipant && authorityParticipant->receipt())
860 impl_->redo.back().specification.baseRevision = authorityParticipant->receipt()->afterRevision;
861 const auto& moved = impl_->undo.back();
862 EditorTransactionRecord record;
863 record.coordinator = std::move(receipt);
864 record.specification = moved.specification;
865 record.authorityReceipt = moved.authorityReceipt;
866 record.commandCount = moved.commands ? moved.commands->commands.size() : 0;
867 record.operationCount = moved.operations.size();
868 return eve::Result<EditorTransactionRecord>::success(std::move(record));
869}
