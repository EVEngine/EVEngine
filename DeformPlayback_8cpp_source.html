38 if (output.size() != scratch_.size())

40 "incorrect deformation output extent", "deformPlayback"));
41 auto sampled = track_.sample(static_cast<float>(clock_.time()), scratch_);
42 if (!sampled) return sampled;
43 for (auto& point : scratch_) {
44 auto placed = transform_.apply({point.x, point.y, point.z, 0, 0, 0, 1});
45 if (!placed) return Result<void>::failure(placed.status());
46 const auto value = placed.value();
