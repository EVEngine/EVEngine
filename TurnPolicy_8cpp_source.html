82Result<void> TurnPolicyRegistry::add(std::shared_ptr<const ITurnPolicy> policy) {
83 if (policy == nullptr || policy->id().empty())
84 return Result<void>::failure(Diagnostic::error(
85 DiagnosticCode::InvalidArgument, "tactics turn policy requires a non-empty stable id", "turnPolicy.id"));
86 const std::string key(policy->id());
87 if (policies_.contains(key))
88 return Result<void>::failure(Diagnostic::error(
89 DiagnosticCode::Conflict, "tactics turn policy id is already registered", "turnPolicy.id"));
90 policies_.emplace(key, std::move(policy));
91 return Result<void>::success(Status::success(StatusCode::Applied));
92}

94Result<const ITurnPolicy*> TurnPolicyRegistry::find(std::string_view id) const {
95 const auto found = policies_.find(id);
96 if (found == policies_.end())
97 return Result<const ITurnPolicy*>::failure(
98 Diagnostic::error(DiagnosticCode::NotFound, "tactics turn policy is not registered", "turnPolicy"));
99 return Result<const ITurnPolicy*>::success(found->second.get());
100}

111TurnPolicyRegistry& TurnPolicyRegistry::builtins() {
112 // Function-local so registration happens once, after static initialisation
113 // order is no longer a question.
114 static TurnPolicyRegistry registry = [] {
115 TurnPolicyRegistry value;
116 // Built-ins are immutable and stateless, so a failed registration is
117 // impossible here; ignoring the result would hide a real duplicate bug.
118 if (!value.add(std::make_shared<const SideAlternatingPolicy>()).ok()) return TurnPolicyRegistry{};
119 if (!value.add(std::make_shared<const InitiativePolicy>()).ok()) return TurnPolicyRegistry{};
120 if (!value.add(std::make_shared<const ChargeTimeBattlePolicy>()).ok()) return TurnPolicyRegistry{};
121 return value;
122 }();
123 return registry;
124}
