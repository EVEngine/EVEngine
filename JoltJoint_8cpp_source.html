22 if (joint->alive) {
23 auto world = joint->world.lock();
24 auto a = joint->a.lock();
25 auto b = joint->b.lock();
26 if (world && world->alive) {
27 if (joint->native) world->system.RemoveConstraint(joint->native);
28 if (joint->velocity) world->system.RemoveConstraint(joint->velocity);
29 if (!joint->collide && a && b) {
30 auto i = world->suppressedPairs.find(pairKey(a->id, b->id));
31 if (i != world->suppressedPairs.end() && --i->second == 0) world->suppressedPairs.erase(i);
32 if (a->valid()) a->api().InvalidateContactCache(a->id);
33 if (b->valid()) b->api().InvalidateContactCache(b->id);
34 }
35 }
36 }

64eve::Result<void> JointBackend::setCollideConnected(bool collide) {
65 if (!isValid()) return rejected<void>("Joint expired", eve::DiagnosticCode::StaleHandle);
66 if (collide == joint->collide) return success();
67 auto world = joint->world.lock();
68 auto a = joint->a.lock();
69 auto b = joint->b.lock();
70 auto key = pairKey(a->id, b->id);
71 if (!collide)
72 ++world->suppressedPairs[key];
73 else {
74 auto i = world->suppressedPairs.find(key);
75 if (i != world->suppressedPairs.end() && --i->second == 0) world->suppressedPairs.erase(i);
76 }
77 joint->collide = collide;
78 a->api().InvalidateContactCache(a->id);
79 b->api().InvalidateContactCache(b->id);
80 if (a->active) a->api().ActivateBody(a->id);
81 if (b->active) b->api().ActivateBody(b->id);
82 return success();
83}

88JPH::Vec3 JointBackend::force(bool torque) const {
89 if (!joint->native) return JPH::Vec3::sZero();
90 const float inverse = joint->world.lock()->inverseStep;
91 if (joint->kind == JointKind3D::Distance) {
92 if (torque) return JPH::Vec3::sZero();
93 const auto axis =
94 JPH::Vec3(frame(true).GetTranslation() - frame(false).GetTranslation()).NormalizedOr(JPH::Vec3::sAxisX());
95 return axis * distanceImpulse(*joint->native) * inverse;
96 }
97 auto value = torque ? six().GetTotalLambdaRotation() + six().GetTotalLambdaMotorRotation()
98 : six().GetTotalLambdaPosition() + six().GetTotalLambdaMotorTranslation();
99 // Fully locked SixDOF point/rotation parts report world-space impulses; individual axes report constraint space.
100 const bool locked = torque ? six().IsFixedAxis(JPH::SixDOFConstraint::EAxis::RotationX) &&
101 six().IsFixedAxis(JPH::SixDOFConstraint::EAxis::RotationY) &&
102 six().IsFixedAxis(JPH::SixDOFConstraint::EAxis::RotationZ)
103 : six().IsFixedAxis(JPH::SixDOFConstraint::EAxis::TranslationX) &&
104 six().IsFixedAxis(JPH::SixDOFConstraint::EAxis::TranslationY) &&
105 six().IsFixedAxis(JPH::SixDOFConstraint::EAxis::TranslationZ);
106 auto world = locked ? value : frame(false).Multiply3x3(value);
107 if (joint->velocity)
108 world += frame(false).Multiply3x3(torque ? drive().GetTotalLambdaMotorRotation()
109 : drive().GetTotalLambdaMotorTranslation());
110 return world * inverse;
111}
