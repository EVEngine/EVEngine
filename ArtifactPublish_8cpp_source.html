37void appendPartView(const ArtifactPart& source, PublicationBuffer& buffer) {
38 eve::artifact::PartView view;
39 view.id = eve::PersistentId::fromUuid(source.id);
40 view.role = source.role;
41 view.kind = partKind(source.type);
42 view.schemaVersion = source.schemaVersion.value();
43 view.buildKey = source.buildKey.format();
44 view.bounds = boundsView(source.bounds);
45 std::visit(
46 [&view, &buffer](const auto& payload) {
47 using T = std::decay_t<decltype(payload)>;
48 if constexpr (std::is_same_v<T, MeshData>) {
49 view.positions = payload.positions();
50 view.normals = payload.normals();
51 view.uvs = payload.uvs();
52 view.indices = payload.indices();
53 } else if constexpr (std::is_same_v<T, Collider>) {
54 view.positions = payload.vertices;
55 view.indices = payload.indices;
56 } else if constexpr (std::is_same_v<T, Grid2D>) {
57 view.width = payload.getWidth();
58 view.height = payload.getHeight();
59 view.cells = payload.cells();
60 } else if constexpr (std::is_same_v<T, PointSet>) {
61 buffer.pointBuffers.emplace_back();
62 auto& points = buffer.pointBuffers.back();
63 points.reserve(payload.points().size());
64 for (const auto& point : payload.points()) points.push_back({point.x, point.y, point.z});
65 view.points = points;
66 }
67 },
68 source.payload);
69 buffer.parts.emplace_back(view);
70}

107PublicationBuffer makePublicationBuffer(const GeneratedArtifact& artifact) {
108 PublicationBuffer buffer;
109 if (artifact.type == ArtifactType::Composite) {
110 const auto& composite = std::get<CompositeArtifact>(artifact.payload);
111 buffer.parts.reserve(composite.children.size());
112 buffer.pointBuffers.reserve(composite.children.size());
113 for (const auto& part : composite.children) appendPartView(part, buffer);
114 } else {
115 buffer.parts.reserve(1);
116 buffer.pointBuffers.reserve(1);
117 appendLeafView(artifact, buffer);
118 }
119 buffer.view.id = eve::PersistentId::fromUuid(artifact.id);
120 buffer.view.schemaVersion = artifact.schemaVersion.value();
121 buffer.view.buildKey = artifact.buildKey.format();
122 buffer.view.bounds = boundsView(artifact.bounds);
123 buffer.view.parts = std::span<const eve::artifact::PartView>(buffer.parts.data(), buffer.parts.size());
124 return buffer;
125}

134 ProviderParticipant(std::string name, eve::artifact::ProviderContract& provider,

138 ~ProviderParticipant() override {

142 std::string_view name() const noexcept override { return name_; }

144 [[nodiscard]] eve::Result<void> prepare(const eve::transaction::TransactionContext&) override {
145 if (phase_ != Phase::Idle)
146 return participantFailure(eve::DiagnosticCode::PreconditionViolation,
147 "artifact provider participant is not idle", name_ + ".prepare");
148 // Do not initialize a placeholder Result and then move-assign over it:
149 // in assertion builds move assignment intentionally rejects overwriting
150 // an unobserved Result. Returning from this lambda constructs the one
151 // authoritative Result directly while still converting provider throws
152 // into the publication diagnostic.
153 auto prepared = [&]() -> eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>> {
154 try {
155 return provider_->prepare(view_);
156 } catch (const std::exception& error) {
157 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(eve::Diagnostic::error(
158 eve::DiagnosticCode::Failed, std::string("artifact provider prepare threw: ") + error.what(),
159 name_ + ".prepare"));
160 } catch (...) {
161 return eve::Result<std::unique_ptr<eve::artifact::PreparedPublication>>::failure(eve::Diagnostic::error(
162 eve::DiagnosticCode::Failed, "artifact provider prepare threw", name_ + ".prepare"));
163 }
164 }();
165 if (!prepared.ok()) return eve::Result<void>::failure(prepared.status());
166 stage_ = std::move(prepared).takeValue();
167 if (!stage_)
168 return participantFailure(eve::DiagnosticCode::InvariantViolation,
169 "artifact provider returned a null prepared stage", name_ + ".prepare");
170 phase_ = Phase::Prepared;
171 return eve::Result<void>::success();
172 }
173
174 [[nodiscard]] eve::Result<void> commit(const eve::transaction::TransactionContext&) override {
175 if (phase_ != Phase::Prepared || !stage_)
176 return participantFailure(eve::DiagnosticCode::PreconditionViolation,
177 "artifact provider participant is not prepared", name_ + ".commit");
178 stage_->commit();
179 stage_.reset();
180 phase_ = Phase::Committed;
181 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
182 }
183
184 [[nodiscard]] eve::Result<void> rollback(const eve::transaction::TransactionContext&) override {
185 if (phase_ != Phase::Prepared)
186 return participantFailure(eve::DiagnosticCode::PreconditionViolation,
187 "artifact provider participant is not prepared", name_ + ".rollback");
188 stage_->rollback();
189 stage_.reset();
190 phase_ = Phase::RolledBack;
191 return eve::Result<void>::success(eve::Status::success(eve::StatusCode::Applied));
192 }
193
194 [[nodiscard]] eve::Result<void> compensate(const eve::transaction::TransactionContext&) override {
195 if (phase_ != Phase::Committed)
196 return participantFailure(eve::DiagnosticCode::PreconditionViolation,
197 "artifact provider participant is not committed", name_ + ".compensate");
198 return participantFailure(eve::DiagnosticCode::Unsupported,
199 "artifact provider publication has no inverse compensation", name_ + ".compensate");
200 }

203 enum class Phase { Idle, Prepared, Committed, RolledBack };

418eve::Result<ArtifactPublishReceipt> ArtifactPublisher::publish(GeneratedArtifact artifact,
419 ArtifactPublishOptions options) {
420 PublicationBuffer publication;
421 try {
422 publication = makePublicationBuffer(artifact);
423 } catch (const std::exception& error) {
424 return eve::Result<ArtifactPublishReceipt>::failure(
425 eve::Diagnostic::error(eve::DiagnosticCode::Failed,
426 std::string("failed to materialize artifact publication view: ") + error.what()));
427 } catch (...) {
428 return eve::Result<ArtifactPublishReceipt>::failure(
429 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "failed to materialize artifact publication view"));
430 }
431
432 std::vector<std::unique_ptr<eve::transaction::ITransactionParticipant>> owners;
433 std::vector<eve::transaction::ITransactionParticipant*> participants;
434 owners.reserve(5);
435 participants.reserve(5);
436 auto scene =
437 addProviderParticipant<ISceneArtifactAdapter>(options.scene, "scene", publication.view, owners, participants);
438 if (!scene.ok()) return eve::Result<ArtifactPublishReceipt>::failure(scene.status());
439 auto map = addProviderParticipant<IMapArtifactAdapter>(options.map, "map", publication.view, owners, participants);
440 if (!map.ok()) return eve::Result<ArtifactPublishReceipt>::failure(map.status());
441 auto graphics = addProviderParticipant<IGraphicsArtifactAdapter>(options.graphics, "graphics", publication.view,
442 owners, participants);
443 if (!graphics.ok()) return eve::Result<ArtifactPublishReceipt>::failure(graphics.status());
444 auto physics = addProviderParticipant<IPhysicsArtifactAdapter>(options.physics, "physics", publication.view, owners,
445 participants);
446 if (!physics.ok()) return eve::Result<ArtifactPublishReceipt>::failure(physics.status());
447
448 try {
449 owners.emplace_back(std::make_unique<StoreParticipant>(store_, std::move(artifact)));
450 participants.push_back(owners.back().get());
451 } catch (...) {
452 return eve::Result<ArtifactPublishReceipt>::failure(
453 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "failed to allocate artifact store participant"));
454 }
455
456 const auto context = publicationContext(publication.view.id, "artifact.publish");
457 eve::transaction::Coordinator coordinator;
458 auto coordinated = coordinator.execute(context, participants);
459 if (!coordinated.ok()) return eve::Result<ArtifactPublishReceipt>::failure(coordinated.status());
460 auto transactionReceipt = std::move(coordinated).takeValue();
461 if (transactionReceipt.state != eve::transaction::CoordinatorState::Committed)
462 return eve::Result<ArtifactPublishReceipt>::failure(
463 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation,
464 "artifact publication coordinator returned a non-committed receipt"));
465
466 ArtifactPublishReceipt receipt;
467 receipt.id = ArtifactId::fromUuid(publication.view.id);
468 receipt.scenePublished = options.scene;
469 receipt.graphicsPublished = options.graphics;
470 receipt.physicsPublished = options.physics;
471 receipt.mapPublished = options.map;
472 return eve::Result<ArtifactPublishReceipt>::success(std::move(receipt));
473}

475eve::Result<eve::SnapshotEnvelope> ArtifactPublisher::snapshot(const ArtifactSnapshotContext& context) const {
476 if (!context.hashProvider)
477 return eve::Result<eve::SnapshotEnvelope>::failure(eve::Diagnostic::error(
478 eve::DiagnosticCode::Unsupported, "artifact snapshot requires an injected hash provider"));
479 const auto schema = eve::LogicalId::fromParts("eve", "artifact-publication");
480 if (!schema)
481 return eve::Result<eve::SnapshotEnvelope>::failure(
482 eve::Diagnostic::error(eve::DiagnosticCode::InvariantViolation, "artifact snapshot schema id is invalid"));
483
484 auto store = store_.snapshotState();
485 if (!store.ok()) return eve::Result<eve::SnapshotEnvelope>::failure(store.status());
486 eve::Value::Object providers;
487 const auto appendProvider = [&providers](const char* name,
488 eve::artifact::ProviderContract* provider) -> eve::Result<void> {
489 if (!provider) return eve::Result<void>::success();
490 auto state = provider->snapshotState();
491 if (!state.ok()) return eve::Result<void>::failure(state.status());
492 providers.emplace(name, std::move(state).takeValue());
493 return eve::Result<void>::success();
494 };
495 auto scene = appendProvider("scene", eve::cap::query<eve::artifact::ISceneArtifactAdapter>());
496 if (!scene.ok()) return eve::Result<eve::SnapshotEnvelope>::failure(scene.status());
497 auto map = appendProvider("map", eve::cap::query<eve::artifact::IMapArtifactAdapter>());
498 if (!map.ok()) return eve::Result<eve::SnapshotEnvelope>::failure(map.status());
499 auto graphics = appendProvider("graphics", eve::cap::query<eve::artifact::IGraphicsArtifactAdapter>());
500 if (!graphics.ok()) return eve::Result<eve::SnapshotEnvelope>::failure(graphics.status());
501 auto physics = appendProvider("physics", eve::cap::query<eve::artifact::IPhysicsArtifactAdapter>());
502 if (!physics.ok()) return eve::Result<eve::SnapshotEnvelope>::failure(physics.status());
503
504 eve::Value::Object payload;
505 payload.emplace("store", std::move(store).takeValue());
506 payload.emplace("providers", eve::Value(std::move(providers)));
507 return eve::makeSnapshotEnvelope("eve.artifact.publication", *schema, eve::SchemaVersion(1), context.instanceId,
508 context.revision, context.tick, eve::Value(std::move(payload)),
509 context.hashProvider);

512eve::Result<void> ArtifactPublisher::restore(const eve::SnapshotEnvelope& snapshot,
513 const eve::SnapshotHashProvider& hashProvider) {
514 auto verified = eve::verifySnapshotEnvelope(snapshot, hashProvider);
515 if (!verified.ok()) return verified;
516 const auto schema = eve::LogicalId::fromParts("eve", "artifact-publication");
517 if (!schema || snapshot.type != "eve.artifact.publication" || snapshot.schema != *schema ||
518 snapshot.schemaVersion.value() != 1)
519 return eve::Result<void>::failure(
520 eve::Diagnostic::error(eve::DiagnosticCode::UnknownVersion, "unsupported artifact publication snapshot"));
521 if (store_.size() != 0)
522 return eve::Result<void>::failure(
523 eve::Diagnostic::error(eve::DiagnosticCode::Conflict, "artifact snapshot restore requires an empty store"));
524
525 const auto* payload = snapshot.payload.getIf<eve::Value::Object>();
526 const auto* storeState = payload ? objectMember(*payload, "store") : nullptr;
527 const auto* providerStateValue = payload ? objectMember(*payload, "providers") : nullptr;
528 const auto* providerStates = providerStateValue ? providerStateValue->getIf<eve::Value::Object>() : nullptr;
529 if (!payload || !storeState || !providerStates)
530 return eve::Result<void>::failure(eve::Diagnostic::error(
531 eve::DiagnosticCode::ParseError, "artifact publication snapshot payload is incomplete"));
532

545 };

552 (lookupProvider("graphics") && !graphics) || (lookupProvider("physics") && !physics))

562 participants.push_back(owners.back().get());
563 const auto addRestore = [&owners, &participants](const char* name, eve::artifact::ProviderContract* provider,
564 const eve::Value* state) {

566 owners.emplace_back(std::make_unique<ProviderRestoreParticipant>(name, *provider, *state));
567 participants.push_back(owners.back().get());
568 };
569 addRestore("scene", scene, lookupProvider("scene"));
570 addRestore("map", map, lookupProvider("map"));
571 addRestore("graphics", graphics, lookupProvider("graphics"));
572 addRestore("physics", physics, lookupProvider("physics"));
573 } catch (...) {

575 eve::Diagnostic::error(eve::DiagnosticCode::Failed, "failed to allocate artifact restore participants"));
576 }

578 const auto context = publicationContext(snapshot.instanceId, "artifact.restore");

581 if (!coordinated.ok()) return eve::Result<void>::failure(coordinated.status());

584 return eve::Result<void>::failure(eve::Diagnostic::error(
585 eve::DiagnosticCode::InvariantViolation, "artifact restore coordinator returned a non-committed receipt"));
