72float random01(std::uint32_t x) {

93Result<void> PcgGrowth::start(float now, std::uint32_t seed) {
94 if (!std::isfinite(now))
95 return Result<void>::failure(
96 Diagnostic::error(DiagnosticCode::InvalidArgument, "start time must be finite", "scene.pcgGrowth.start"));
97 startTime_ = now;
98 actualEnd_ = endScale_ + random01(seed) * variance_;
99 scale_ = startScale_;
100 started_ = true;
101 finished_ = false;
102 dying_ = false;
103 return Result<void>::success();
104}
105Result<float> PcgGrowth::advance(float now) {
106 if (!started_)
107 return Result<float>::failure(
108 Diagnostic::error(DiagnosticCode::InvalidArgument, "growth has not started", "scene.pcgGrowth.advance"));
109 if (!std::isfinite(now) || now < startTime_)
110 return Result<float>::failure(Diagnostic::error(
111 DiagnosticCode::InvalidArgument, "time must be finite and monotonic", "scene.pcgGrowth.advance"));
112 float t = std::min((now - startTime_) / duration_, 1.f);
113 scale_ = startScale_ + t * (actualEnd_ - startScale_);
114 finished_ = t >= 1;
115 return Result<float>::success(scale_);
116}

131Result<bool> applyPcgGrowth(PcgGrowth& g, SceneNodeRef& node, float now) {
132 auto due = g.shouldDestroy(now);
133 if (!due) return Result<bool>::failure(due.status());
134 if (due.value()) {
135 auto* s = node.getScene();
136 if (!s)
137 return Result<bool>::failure(Diagnostic::error(DiagnosticCode::InvalidArgument,
138 "growth node cannot be removed", "scene.pcgGrowth.node"));
139 auto removed = s->removeLeafAt(node.getHostName(), node.getNodeId());
140 if (!removed.ok()) return Result<bool>::failure(removed.status());
141 return Result<bool>::success(true);
142 }
143 auto scale = g.advance(now);
144 if (!scale) return Result<bool>::failure(scale.status());
145 if (!node.setScale(scale.value(), scale.value(), scale.value()))
146 return Result<bool>::failure(
147 Diagnostic::error(DiagnosticCode::InvalidArgument, "growth node cannot be scaled", "scene.pcgGrowth.node"));
148 return Result<bool>::success(false);
149}
