38ssq::Object readMemberValue(const ssq::Object& instance, const std::string& name) {
39 if (instance.getType() != ssq::Type::INSTANCE || name.empty()) return nullObject(instance.getHandle());
40 HSQUIRRELVM vm = instance.getHandle();
41 const SQInteger top = sq_gettop(vm);
42 sq_pushobject(vm, instance.getRaw());
43 sq_pushstring(vm, name.c_str(), static_cast<SQInteger>(name.size()));
44 if (SQ_FAILED(sq_get(vm, -2))) {
45 sq_settop(vm, top);
46 return nullObject(vm);
47 }
48 return objectFromTop(vm, top);
49}

93bool convertScalar(const ssq::Object& value, ReflectedValue& out) {
94 switch (value.getType()) {
95 case ssq::Type::BOOL:
96 out.kind = ReflectedValueKind::Bool;
97 out.boolean = value.toBool();
98 return true;
99 case ssq::Type::INTEGER:
100 out.kind = ReflectedValueKind::Integer;
101 out.integer = value.toInt();
102 return true;
103 case ssq::Type::FLOAT:
104 out.kind = ReflectedValueKind::Float;
105 out.floating = value.toFloat();
106 return true;
107 case ssq::Type::STRING:
108 out.kind = ReflectedValueKind::String;
109 out.text = value.toString();
110 return true;
111 default: return false;
112 }
113}

116ssq::Table memberTable(ssq::VM& vm, const ReflectedMember& member, const ssq::Object* instance) {
117 ssq::Table out = vm.newTable();
118 out.set("name", member.name);
119 out.set("type", std::string(ssq::typeToStr(member.type)));
120 out.set("method", member.method);
121
122 ssq::Table attributes = vm.newTable();
123 for (const ReflectedAttribute& attribute : member.attributes)
124 attributes.set(attribute.name.c_str(), attribute.value);
125 out.set("attributes", attributes);
126
127 if (instance && !member.method)
128 out.set("value", readMemberValue(*instance, member.name));
129 else
130 out.set("value", nullObject(vm.getHandle()));
131 return out;
132}

135ssq::Table classTable(ssq::VM& vm, const ReflectedClass& cls) {
136 ssq::Table out = vm.newTable();
137 out.set("name", cls.name);
138 out.set("source", cls.source);
139 out.set("base", cls.base);
140
141 ssq::Array members = vm.newArray();
142 for (const ReflectedMember& member : cls.members) members.push(memberTable(vm, member, nullptr));
143 out.set("members", members);
144 return out;
145}
