40Result<void> ActionComboWindowState::enter(const ActionStateWindowBinding& binding,
41 const ActionTimelineEvent& event,
42 const ActionNotifyContext& context) {
43 if (!supports(binding.kind))
44 return failure<void>(DiagnosticCode::Unsupported, "input adapter does not own this window kind", "kind");
45 if (!resolver_) return failure<void>(DiagnosticCode::NotFound, "combo subject resolver is unavailable", "resolver");
46 std::optional<ecs::EntityHandle> handle;
47 if (binding.targetIndex) {
48 if (*binding.targetIndex >= context.targets.size())
49 return failure<void>(DiagnosticCode::NotFound, "combo target index is unavailable", "targetIndex");
50 handle = context.targets[*binding.targetIndex];
51 } else {
52 handle = context.source;
53 }
54 if (!handle) return failure<void>(DiagnosticCode::NotFound, "combo window has no source or target", "subject");
55 auto subject = resolver_(*handle);
56 if (!subject) return Result<void>::failure(subject.status());
57 if (!subject.value().isValid())
58 return failure<void>(DiagnosticCode::InvalidArgument, "combo-window subject is nil", "subject");
59
60 ActiveKey key{context.executionId, event.itemId.format()};
61 const auto found = active_.find(key);
62 if (found != active_.end()) {
63 if (found->second.subject == subject.value() && found->second.input == binding.resource)
64 return Result<void>::success(Status::success(StatusCode::NoOp));
65 return failure<void>(DiagnosticCode::Conflict, "combo-window key has different active data", "itemId");
66 }
67 active_.emplace(std::move(key), ActiveCombo{event.itemId, subject.value(), binding.resource});
68 return Result<void>::success(Status::success(StatusCode::Applied));
69}

94Result<ActionComboMatch> ActionComboWindowState::match(SubjectRef subject, std::string_view input) const {
95 if (!subject.isValid())
96 return failure<ActionComboMatch>(DiagnosticCode::InvalidArgument, "combo subject is nil", "subject");
97 if (input.empty()) return failure<ActionComboMatch>(DiagnosticCode::InvalidArgument, "combo input is empty", "input");
98 for (const auto& [key, combo] : active_) {
99 if (combo.subject == subject && combo.input == input)
100 return Result<ActionComboMatch>::success(
101 ActionComboMatch{key.first, combo.itemId, combo.subject, combo.input});
102 }
103 return failure<ActionComboMatch>(DiagnosticCode::NotFound, "no active combo window accepts the input", "input");
104}
