52EditorResult<std::uint64_t> EditorEventTimeline::append(EditorTimelineEvent event,
53 std::size_t maximumPayloadNodes) {
54 if (event.domain.empty() || event.type.empty() || maximumPayloadNodes == 0 ||
55 nodeCount(event.payload, maximumPayloadNodes) > maximumPayloadNodes)
56 return eve::editing::failed<std::uint64_t>(EditorStatus::Rejected,
57 RuleId("editor.timeline.invalid-event"), "Timeline event identity or payload budget is invalid");
58 if (event.sequence == 0) event.sequence = nextSequence_++;
59 else {
60 if (event.sequence < nextSequence_)

62 RuleId("editor.timeline.sequence"), "Timeline event sequence is stale or duplicated");
63 nextSequence_ = event.sequence + 1;

66 if (events_.size() == capacity_) { events_.pop_front(); ++dropped_; }
67 events_.push_back(std::move(event)); ++generation_;
68 return eve::editing::applied<std::uint64_t>(sequence);

71EditorResult<EditorTimelinePage> EditorEventTimeline::query(
72 const EditorTimelineQuery& filter, std::size_t offset, std::size_t limit,
73 std::optional<std::uint64_t> generation) const {
74 if (limit == 0 || limit > 10000)
75 return eve::editing::failed<EditorTimelinePage>(EditorStatus::Rejected,
76 RuleId("editor.timeline.page-size"), "Timeline page size must be between 1 and 10,000");
77 if (generation && *generation != generation_)
78 return eve::editing::failed<EditorTimelinePage>(EditorStatus::Conflict,
79 RuleId("editor.timeline.stale-generation"), "Timeline changed while paging");
80 std::vector<EditorTimelineEvent> matched;
81 for (const auto& event : events_) {
82 if (!includes(filter.domains, event.domain) || !includes(filter.types, event.type) ||
83 (!filter.source.empty() && event.source != filter.source) ||
84 (!filter.correlation.empty() && event.correlation != filter.correlation) ||
85 (filter.minimumSeverity && severityRank(event.severity) < severityRank(*filter.minimumSeverity)) ||
86 event.tick < filter.firstTick || (filter.lastTick > 0 && event.tick > filter.lastTick)) continue;
87 matched.push_back(event);
88 }
89 EditorTimelinePage result; result.generation = generation_; result.droppedEvents = dropped_;
90 const std::size_t begin = std::min(offset, matched.size());
91 const std::size_t end = std::min(begin + limit, matched.size());
92 result.values.assign(matched.begin() + static_cast<std::ptrdiff_t>(begin),
93 matched.begin() + static_cast<std::ptrdiff_t>(end));
94 result.nextOffset = end; result.hasMore = end < matched.size();
95 return eve::editing::applied<EditorTimelinePage>(std::move(result));
96}
