18bool validObservation(const Observation& o, std::size_t features, std::size_t actions) {
19 if (o.features.size() != features || o.legalActions.size() > actions || !std::isfinite(o.reward) ||
20 std::abs(o.reward) > 1e9 || o.coverage.size() > 64 || o.finding.size() > 4096)
21 return false;
22 if (o.outcome != Outcome::Running && o.outcome != Outcome::Success && o.outcome != Outcome::Failure) return false;
23 if (o.outcome == Outcome::Running && o.legalActions.empty()) return false;
24 if (o.outcome == Outcome::Failure && o.finding.empty()) return false;
25 std::set<std::uint32_t> seen;
26 for (auto a : o.legalActions)
27 if (a >= actions || !seen.insert(a).second) return false;
28 for (float f : o.features)
29 if (!std::isfinite(f) || std::abs(f) > 1e6f) return false;
30 for (const auto& key : o.coverage)
31 if (key.empty() || key.size() > 256) return false;
32 return true;
33}

58Result<std::uint32_t> sample(const Policy& policy, const Observation& o, detail::Random& random, bool uniform,
59 Backend backend) {
60 if (uniform) return Result<std::uint32_t>::success(o.legalActions[random.index(o.legalActions.size())]);
61 auto evaluated = infer(policy, o, backend);
62 if (!evaluated) return Result<std::uint32_t>::failure(evaluated.status());
63 const auto& probabilities = evaluated.value();
64 double u = random.unit();
65 for (auto a : o.legalActions) {
66 u -= probabilities[a];
67 if (u <= 0) return Result<std::uint32_t>::success(a);
68 }
69 return Result<std::uint32_t>::success(o.legalActions.back());
70}

72bool matches(const Observation& a, const Observation& b, double tolerance) {
73 if (a.features.size() != b.features.size() || a.legalActions != b.legalActions || a.coverage != b.coverage ||
74 a.outcome != b.outcome || a.finding != b.finding || !std::isfinite(b.reward) ||

76 return false;
77 for (std::size_t i = 0; i < a.features.size(); ++i)
78 if (!std::isfinite(b.features[i]) || std::abs(double(a.features[i]) - b.features[i]) > tolerance) return false;

117Result<std::vector<double>> infer(const Policy& policy, const Observation& observation, Backend backend) {
118 auto validated = validatePolicy(policy);
119 if (!validated) return Result<std::vector<double>>::failure(validated.status());
120 if (!validObservation(observation, policy.featureCount, policy.actionCount) || observation.legalActions.empty())
121 return invalid<std::vector<double>>("Invalid observation or empty legal action mask");
122 if (backend == Backend::Cpu) return Result<std::vector<double>>::success(detail::forward(policy, observation));
123 auto available = backendName(backend);
124 if (!available) return Result<std::vector<double>>::failure(available.status());
125 IPolicyBackend* provider = backend == Backend::Gpu ? cap::query<IGpuPolicyBackend>() : cap::query<IPolicyBackend>();
126 EV_ASSERT(provider, "backend registration must remain stable during inference");
127 auto result = provider->evaluate(policy, observation);
128 if (!result) return result;
129 if (result.value().size() != policy.actionCount) return invalid<std::vector<double>>("Invalid backend shape");
130 double sum = 0;
131 for (std::size_t i = 0; i < result.value().size(); ++i) {
132 const auto value = result.value()[i];
133 if (!std::isfinite(value) || value < 0 ||
134 (std::find(observation.legalActions.begin(), observation.legalActions.end(), i) ==
135 observation.legalActions.end() &&
136 value != 0))
137 return invalid<std::vector<double>>("Invalid backend probability");
138 sum += value;
139 }
140 if (std::abs(sum - 1) > 1e-5) return invalid<std::vector<double>>("Backend probabilities must sum to one");
141 return result;
142}

144Result<Report> run(const Config& c, IEnvironment& environment) {
145 if (!validConfig(c)) return invalid<Report>("Invalid dimensions, budgets, strategy, probabilities or time");
146 auto selectedBackend = backendName(c.backend);
147 if (!selectedBackend) return Result<Report>::failure(selectedBackend.status());
148 Report report;
149 report.policy = detail::makePolicy(c);
150 report.backend = c.strategy == Strategy::Random ? "uniform-random" : selectedBackend.value();
151 report.trainingBackend = c.strategy == Strategy::Random ? "none"
152 : c.backend == Backend::Gpu ? "tensor-gpu-sgd"
153 : "cpu-sgd";
154 report.bestScore = -std::numeric_limits<double>::infinity();
155 detail::Random random(c.searchSeed);
156 std::vector<Candidate> parents;
157 std::set<std::string> coverage, findingKeys;
158 for (std::uint32_t generation = 0; generation < c.generations; ++generation) {
159 std::vector<Candidate> candidates;
160 for (std::uint32_t member = 0; member < c.population; ++member) {
161 Candidate candidate;
162 candidate.trace.environmentSeed = c.environmentSeed;
163 candidate.trace.dt = c.dt;
164 auto reset = environment.reset(c.environmentSeed);
165 if (!reset) return Result<Report>::failure(reset.status());
166 if (!validObservation(reset.value(), c.featureCount, c.actionCount) || reset.value().reward != 0)
167 return invalid<Report>("Environment returned invalid initial observation");
168 candidate.trace.initial = std::move(reset).takeValue();
169 Observation observation = candidate.trace.initial;
170 std::set<std::string> episodeCoverage(observation.coverage.begin(), observation.coverage.end());
171 const auto parent = parents.empty() ? 0 : random.index(parents.size());
172 for (std::uint32_t tick = 0; tick < c.horizon && observation.outcome == Outcome::Running; ++tick) {
173 const bool uniform = c.strategy == Strategy::Random || random.unit() < c.randomProbability;
174 auto sampled = sample(report.policy, observation, random, uniform, c.backend);
175 if (!sampled) return Result<Report>::failure(sampled.status());
176 auto action = sampled.value();
177 if (!uniform && !parents.empty() && tick < parents[parent].trace.steps.size() &&
178 random.unit() >= c.mutationProbability) {
179 const auto inherited = parents[parent].trace.steps[tick].action;
180 if (std::find(observation.legalActions.begin(), observation.legalActions.end(), inherited) !=
181 observation.legalActions.end())
182 action = inherited;
183 }
184 auto stepped = environment.step(action, c.dt);
185 if (!stepped) return Result<Report>::failure(stepped.status());
186 if (!validObservation(stepped.value(), c.featureCount, c.actionCount))
187 return invalid<Report>("Environment returned invalid step observation");
188 observation = std::move(stepped).takeValue();
189 candidate.score += observation.reward;
190 episodeCoverage.insert(observation.coverage.begin(), observation.coverage.end());
191 candidate.trace.steps.push_back({action, observation});
192 ++report.steps;
193 }
194 ++report.episodes;
195 coverage.insert(episodeCoverage.begin(), episodeCoverage.end());
196 if (coverage.size() > 65536)
197 return Result<Report>::failure(Diagnostic::error(
198 DiagnosticCode::Cancelled, "Coverage storage budget exceeded (65536 points)", {}, {}, "agent"));
199 candidate.score += c.coverageWeight * double(episodeCoverage.size());
200 if (observation.outcome == Outcome::Failure) {
201 ++report.failures;
202 candidate.score += c.failureWeight;
203 if (report.findings.size() < c.maxFindings && findingKeys.insert(observation.finding).second)
204 report.findings.push_back(candidate.trace);
205 }
206 if (candidate.score > report.bestScore) {
207 report.bestScore = candidate.score;
208 report.best = candidate.trace;
209 }
210 candidates.push_back(std::move(candidate));
211 }
212 std::stable_sort(candidates.begin(), candidates.end(),
213 [](const auto& a, const auto& b) { return a.score > b.score; });
214 candidates.resize(c.elites);
215 if (c.strategy == Strategy::EvolutionLearning) {
216 for (std::uint32_t epoch = 0; epoch < c.trainingEpochs; ++epoch)
217 for (const auto& elite : candidates) {
218 const Observation* previous = &elite.trace.initial;
219 for (const auto& step : elite.trace.steps) {
220 if (c.backend == Backend::Gpu) {
221 auto* provider = cap::query<IGpuPolicyBackend>();
222 if (!provider)
223 return Result<Report>::failure(Diagnostic::error(
224 DiagnosticCode::Unsupported,
225 "GPU provider was removed during an environment callback", {}, {}, "agent"));
226 auto trained = provider->train(report.policy, *previous, step.action, c.learningRate);
227 if (!trained) return Result<Report>::failure(trained.status());
228 auto valid = validatePolicy(trained.value());
229 if (!valid) return Result<Report>::failure(valid.status());
230 if (trained.value().featureCount != c.featureCount ||
231 trained.value().hiddenWidth != c.hiddenWidth ||
232 trained.value().actionCount != c.actionCount)
233 return invalid<Report>("GPU training changed policy shape");
234 report.policy = std::move(trained).takeValue();
235 } else
236 detail::train(report.policy, *previous, step.action, c.learningRate);
237 previous = &step.observation;
238 ++report.trainingSamples;
239 }
240 }
241 parents = std::move(candidates);
242 }
243 }
244 report.coverage.assign(coverage.begin(), coverage.end());
245 return Result<Report>::success(std::move(report));
246}

248Result<void> replay(const Trace& trace, IEnvironment& environment, double tolerance) {
249 if (trace.schemaId != "evengine.agent.trace" || trace.schemaVersion != 1 || !std::isfinite(trace.dt) ||
250 trace.dt <= 0 || trace.dt > 60 || !std::isfinite(tolerance) || tolerance < 0 || trace.steps.size() > 1024 ||
251 trace.initial.features.empty() || trace.initial.features.size() > 1024 || trace.initial.reward != 0)
252 return invalid<void>("Unsupported trace schema/version, time, length or tolerance");
253 const Observation* previous = &trace.initial;
254 if (!validObservation(*previous, trace.initial.features.size(), 1024))
255 return invalid<void>("Invalid initial trace observation");
256 for (const auto& step : trace.steps) {
257 if (previous->outcome != Outcome::Running ||
258 std::find(previous->legalActions.begin(), previous->legalActions.end(), step.action) ==
259 previous->legalActions.end() ||
260 !validObservation(step.observation, trace.initial.features.size(), 1024))
261 return invalid<void>("Invalid trace action or observation");
262 previous = &step.observation;
263 }
264 auto reset = environment.reset(trace.environmentSeed);
265 if (!reset) return Result<void>::failure(reset.status());
266 if (!matches(trace.initial, reset.value(), tolerance)) return divergence();
267 for (const auto& step : trace.steps) {
268 auto result = environment.step(step.action, trace.dt);
269 if (!result) return Result<void>::failure(result.status());
270 if (!matches(step.observation, result.value(), tolerance)) return divergence();
271 }
272 return Result<void>::success();
273}
