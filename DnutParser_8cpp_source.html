101 void tokenize() {
102 toks_.clear();
103 const std::string &src = source_;
104 const size_t n = src.size();
105 size_t i = 0;
106 int line = 1;
107 const auto add = [&](Tok k, std::string v) {
108 toks_.push_back(Token{k, std::move(v), line});
109 };
110
111 while (i < n) {
112 const char c = src[i];
113 if (c == '\n') {
114 ++line;
115 ++i;
116 continue;
117 }
118 if (c == ' ' || c == '\t' || c == '\r') {
119 ++i;
120 continue;
121 }
122 if (c == '/' && i + 1 < n && src[i + 1] == '/') {
123 while (i < n && src[i] != '\n') ++i;
124 continue;
125 }
126 if (c == '/' && i + 1 < n && src[i + 1] == '*') {
127 const int startLine = line;
128 i += 2;
129 while (i < n && !(src[i] == '*' && i + 1 < n && src[i + 1] == '/')) {
130 if (src[i] == '\n') ++line;
131 ++i;
132 }
133 if (i >= n)
134 throw ParseError(path_ + ":" + std::to_string(startLine) + ": 未闭合的块注释");
135 i += 2;
136 continue;
137 }
138 if (c == '"' || c == '\'') {
139 const char quote = c;
140 const int startLine = line;
141 ++i;
142 std::string s;
143 while (i < n && src[i] != quote) {
144 if (src[i] == '\\' && i + 1 < n) {
145 ++i;
146 const char e = src[i];
147 switch (e) {
148 case 'n': s += '\n'; break;
149 case 't': s += '\t'; break;
150 case 'r': s += '\r'; break;
151 case '"': s += '"'; break;
152 case '\'': s += '\''; break;
153 case '\\': s += '\\'; break;
154 case '{': s += '{'; break;
155 case '}': s += '}'; break;
156 default: s += e; break;
157 }
158 ++i;
159 } else {
160 s += src[i++];
161 }
162 }
163 if (i >= n)
164 throw ParseError(path_ + ":" + std::to_string(startLine) + ": 未闭合的字符串");
165 ++i;
166 add(Tok::Str, std::move(s));
167 continue;
168 }
169 if (std::isdigit(static_cast<unsigned char>(c)) ||
170 (c == '.' && i + 1 < n && std::isdigit(static_cast<unsigned char>(src[i + 1])))) {
171 const size_t start = i;
172 while (i < n) {
173 const char d = src[i];
174 if (d >= '0' && d <= '9') {
175 ++i;
176 } else if (d == '.') {
177 ++i;
178 } else if ((d == 'e' || d == 'E') && i + 1 < n &&
179 ((src[i + 1] >= '0' && src[i + 1] <= '9') ||
180 src[i + 1] == '+' || src[i + 1] == '-')) {
181 i += 2;
182 } else {
183 break;
184 }
185 }
186 add(Tok::Num, src.substr(start, i - start));
187 continue;
188 }
189 if (std::isalpha(static_cast<unsigned char>(c)) || c == '_') {
190 const size_t start = i;
191 while (i < n) {
192 const char d = src[i];
193 if (std::isalnum(static_cast<unsigned char>(d)) || d == '_' || d == '.') ++i;
194 else break;
195 }
196 add(Tok::Ident, src.substr(start, i - start));
197 continue;
198 }
199 const std::string two = (i + 1 < n) ? src.substr(i, 2) : "";
200 if (two == "==" || two == "!=" || two == ">=" || two == "<=" ||
201 two == "&&" || two == "||") {
202 add(Tok::Punct, two);
203 i += 2;
204 continue;
205 }
206 if (c == '{' || c == '}' || c == '(' || c == ')' || c == '[' || c == ']' ||
207 c == ':' || c == ',' || c == '=' || c == '>' || c == '<' || c == '!' ||
208 c == '-') {
209 add(Tok::Punct, std::string(1, c));
210 ++i;
211 continue;
212 }
213 throw ParseError(path_ + ":" + std::to_string(line) + ": 无法识别的字符 '" +
214 std::string(1, c) + "'");
215 }
216 toks_.push_back(Token{Tok::Eof, "", line});
217 }

219 DataValue parseLiteralValue() {
220 const Token &t = cur();
221 if (t.kind == Tok::Str) {
222 adv();
223 return DataValue::string(t.value);
224 }
225 if (t.kind == Tok::Num) {
226 adv();
227 return numValue(t.value);
228 }
229 if (t.kind == Tok::Ident) {
230 if (t.value == "true") {
231 adv();
232 return DataValue::boolean(true);
233 }
234 if (t.value == "false") {
235 adv();
236 return DataValue::boolean(false);
237 }
238 fail("条件字面量只支持字符串/数字/true/false");
239 }
240 if (isPunct("-")) {
241 adv();
242 const Token &tt = cur();
243 if (tt.kind == Tok::Num) {
244 adv();
245 return negNumValue(tt.value);
246 }
247 fail("'-' 后应跟数字");
248 }
249 fail("期望字面量");

252 DataValue parseComparison() {
253 if (!isIdent()) fail("条件左侧应为变量名");
254 const std::string varName = adv().value;
255 const Token &op = cur();
256 if (op.kind != Tok::Punct ||
257 (op.value != "==" && op.value != "!=" && op.value != ">" && op.value != "<" &&
258 op.value != ">=" && op.value != "<="))
259 fail("期望比较运算符（== != > < >= <=）");
260 adv();
261 DataValue value = parseLiteralValue();
262 std::string mapped;
263 if (op.value == "==") mapped = "eq";
264 else if (op.value == "!=") mapped = "ne";
265 else if (op.value == ">") mapped = "gt";
266 else if (op.value == "<") mapped = "lt";
267 else if (op.value == ">=") mapped = "ge";
268 else mapped = "le";
269 return DataValue::object({
270 {"var", DataValue::string(varName)},
271 {"op", DataValue::string(mapped)},
272 {"value", std::move(value)},
273 });
274 }

290 DataValue parseAnd() {
291 DataValue left = parseNot();
292 while (isPunct("&&")) {
293 adv();
294 DataValue right = parseNot();
295 bool appended = false;
296 if (left.kind() == DataValue::Kind::Object) {
297 for (auto &kv : *left.getIf<DataValue::Object>()) {
298 if (kv.first == "all" && kv.second.kind() == DataValue::Kind::Array) {
299 kv.second.pushBack(std::move(right));
300 appended = true;
301 break;
302 }
303 }
304 }
305 if (!appended) {
306 left = DataValue::object(
307 {{"all", DataValue::array({std::move(left), std::move(right)})}});
308 }
309 }
310 return left;
311 }

313 DataValue parseOr() {
314 DataValue left = parseAnd();
315 while (isPunct("||")) {
316 adv();
317 DataValue right = parseAnd();
318 bool appended = false;
319 if (left.kind() == DataValue::Kind::Object) {
320 for (auto &kv : *left.getIf<DataValue::Object>()) {
321 if (kv.first == "any" && kv.second.kind() == DataValue::Kind::Array) {
322 kv.second.pushBack(std::move(right));
323 appended = true;
324 break;
325 }
326 }
327 }
328 if (!appended) {
329 left = DataValue::object(
330 {{"any", DataValue::array({std::move(left), std::move(right)})}});
331 }
332 }
333 return left;

336 void parseAttrs(const int lineNum, DataValue::Object &out) {
337 while (cur().line == lineNum && !isPunct("}") && cur().kind != Tok::Eof) {
338 if (!isIdent()) fail("期望属性名");
339 const std::string name = adv().value;
340 if (name == "meta" && isPunct("(")) {
341 adv();
342 DataValue::Object metaFields;
343 while (!isPunct(")")) {
344 if (!isIdent()) fail("meta 键应为标识符");
345 const std::string k = adv().value;
346 expectPunct("=");
347 DataValue v = parseLiteralValue();
348 if (v.kind() != DataValue::Kind::String && v.kind() != DataValue::Kind::Int &&
349 v.kind() != DataValue::Kind::Float && v.kind() != DataValue::Kind::Bool)
350 fail("meta 值只支持标量");
351 metaFields.emplace(k, DataValue::string(scalarToString(v)));
352 if (isPunct(",")) adv();
353 else if (!isPunct(")")) fail("meta 内期望 ',' 或 ')'");
354 }
355 adv(); // )
356 out.emplace("meta", DataValue::object(std::move(metaFields)));
357 continue;
358 }
359 if (name == "tags") {
360 expectPunct("=");
361 if (!isPunct("[")) fail("tags 后应为 [");
362 adv();
363 std::vector<DataValue> arr;
364 while (!isPunct("]")) {
365 if (cur().kind != Tok::Str) fail("tags 元素应为字符串");
366 arr.emplace_back(DataValue::string(adv().value));
367 if (isPunct(",")) adv();
368 else if (!isPunct("]")) fail("tags 内期望 ',' 或 ']'");
369 }
370 adv(); // ]
371 out.emplace("tags", DataValue::array(std::move(arr)));
372 continue;
373 }
374 expectPunct("=");
375 DataValue v = parseLiteralValue();
376 if (name == "weight" || name == "i18n" || name == "id") {
377 out.emplace(name, std::move(v));
378 } else {
379 fail("未知属性 '" + name + "'");
380 }
381 }

384 DataValue parseLine(const std::string &poolId, int idx, const DataValue *inheritWhen) {
385 const int lineNum = cur().line;
386 std::string speaker;
387 if (isPunct("-")) {
388 adv();
389 } else if (isIdent()) {
390 if (cur().value == "when") fail("when 分组不允许嵌套");
391 speaker = adv().value;
392 expectPunct(":");
393 } else {
394 fail("期望说话人或 '-'");
395 }
396 if (cur().kind != Tok::Str) fail("期望台词字符串");
397 const std::string text = adv().value;
398
399 DataValue::Object fields;
400 fields.emplace("speaker", DataValue::string(speaker));
401 fields.emplace("text", DataValue::string(text));
402 if (inheritWhen) fields.emplace("when", *inheritWhen);
403 DataValue::Object attrs;
404 parseAttrs(lineNum, attrs);
405 bool hasId = false;
406 for (auto &kv : attrs) {
407 if (kv.first == "id") hasId = true;
408 fields.emplace(std::move(kv.first), std::move(kv.second));
409 }
410 if (!hasId) fields.emplace("id", DataValue::string(poolId + "." + std::to_string(idx)));
411 return DataValue::object(std::move(fields));
412 }

414 void parsePool(DataValue::Object &pools) {
415 expectIdent("pool");
416 const std::string poolId = expectIdent("pool 名称");
417 long long noRepeat = -1;
418 while (isIdent() && cur().value == "noRepeat") {
419 adv();
420 expectPunct("=");
421 const Token &t = cur();
422 if (t.kind != Tok::Num) fail("noRepeat 应为数字");
423 noRepeat = std::strtoll(adv().value.c_str(), nullptr, 10);
424 }
425 expectPunct("{");
426
427 std::vector<DataValue> lines;
428 int idx = 1;
429 while (!isPunct("}")) {
430 if (cur().kind == Tok::Eof) fail("未闭合的 pool 块");
431 if (isIdent() && cur().value == "pool") fail("pool 块不允许嵌套");
432 if (isIdent() && cur().value == "when") {
433 adv();
434 DataValue cond = parseOr();
435 expectPunct("{");
436 while (!isPunct("}")) {
437 if (cur().kind == Tok::Eof) fail("未闭合的 when 块");
438 lines.push_back(parseLine(poolId, idx, &cond));
439 ++idx;
440 }
441 adv();
442 continue;
443 }
444 lines.push_back(parseLine(poolId, idx, nullptr));
445 ++idx;
446 }
447 adv(); // }
448
449 DataValue::Object poolFields;
450 poolFields.emplace("lines", DataValue::array(std::move(lines)));
451 if (noRepeat >= 0) poolFields.emplace("noRepeat", DataValue::integer(noRepeat));
452 pools.emplace(poolId, DataValue::object(std::move(poolFields)));
453 }

455 DataValue parsePools() {
456 DataValue::Object pools;
457 while (cur().kind != Tok::Eof) parsePool(pools);
458 return DataValue::object({{"pools", DataValue::object(std::move(pools))}});
