57 bool parseDocument(DnutDocument& out, std::vector<ConversationDiagnostic>& diagnostics) {
58 try {
59 tokenize();
60 if (!isIdent() || cur().value != "schema") fail("期望文件 schema");
61 adv();
62 if (cur().kind != Tok::Str) fail("schema 应为字符串");
63 out.schema = adv().value;
64 if (out.schema != "eve.dnut") fail("不支持的 dnut schema '" + out.schema + "'");
65 if (!isIdent() || cur().value != "version") fail("期望文件 version");
66 adv();
67 if (cur().kind != Tok::Num) fail("version 应为整数");
68 out.version = static_cast<int>(std::strtol(adv().value.c_str(), nullptr, 10));
69 if (out.version != DnutDocument::CurrentVersion) fail("不支持的 dnut version");
70 DataValue::Object pools;
71 while (cur().kind != Tok::Eof) {
72 if (!isIdent()) fail("期望 pool 或 conversation");
73 if (cur().value == "pool") parsePool(pools);
74 else if (cur().value == "conversation") out.conversations.push_back(parseConversation());
75 else fail("未知顶层字段 '" + cur().value + "'");
76 }
77 out.poolRoot = DataValue::object({{"pools", DataValue::object(std::move(pools))}});
78 return true;
79 } catch (const ParseError& e) {
80 const int line = toks_.empty() ? 1 : cur().line;
81 const int column = toks_.empty() ? 1 : cur().column;
82 const std::string message = e.what();
83 const std::string code = message.find("不支持的 dnut") != std::string::npos
84 ? "UnsupportedSchemaVersion"
85 : (message.find("未知") != std::string::npos ? "UnknownField"
86 : "DnutParseError");
87 diagnostics.push_back({ConversationDiagnostic::Severity::Error, path_, line, e.what(),
88 code, column, {}});
89 return false;
90 }

128 void tokenize() {
129 toks_.clear();
130 const std::string &src = source_;
131 const size_t n = src.size();
132 size_t i = 0;
133 size_t lineStart = 0;
134 int line = 1;
135
136 while (i < n) {
137 const int column = static_cast<int>(i - lineStart + 1);
138 const auto add = [&](Tok k, std::string v) {
139 toks_.push_back(Token{k, std::move(v), line, column});
140 };
141 const char c = src[i];
142 if (c == '\n') {
143 ++line;
144 ++i;
145 lineStart = i;
146 continue;
147 }
148 if (c == ' ' || c == '\t' || c == '\r') {
149 ++i;
150 continue;
151 }
152 if (c == '/' && i + 1 < n && src[i + 1] == '/') {
153 while (i < n && src[i] != '\n') ++i;
154 continue;
155 }
156 if (c == '/' && i + 1 < n && src[i + 1] == '*') {
157 const int startLine = line;
158 i += 2;
159 while (i < n && !(src[i] == '*' && i + 1 < n && src[i + 1] == '/')) {
160 if (src[i] == '\n') {
161 ++line;
162 lineStart = i + 1;
163 }
164 ++i;
165 }
166 if (i >= n)
167 throw ParseError(path_ + ":" + std::to_string(startLine) + ": 未闭合的块注释");
168 i += 2;
169 continue;
170 }
171 if (c == '"' || c == '\'') {
172 const char quote = c;
173 const int startLine = line;
174 ++i;
175 std::string s;
176 while (i < n && src[i] != quote) {
177 if (src[i] == '\\' && i + 1 < n) {
178 ++i;
179 const char e = src[i];
180 switch (e) {
181 case 'n': s += '\n'; break;
182 case 't': s += '\t'; break;
183 case 'r': s += '\r'; break;
184 case '"': s += '"'; break;
185 case '\'': s += '\''; break;
186 case '\\': s += '\\'; break;
187 case '{': s += '{'; break;
188 case '}': s += '}'; break;
189 default: s += e; break;
190 }
191 ++i;
192 } else {
193 s += src[i++];
194 }
195 }
196 if (i >= n)
197 throw ParseError(path_ + ":" + std::to_string(startLine) + ": 未闭合的字符串");
198 ++i;
199 add(Tok::Str, std::move(s));
200 continue;
201 }
202 if (std::isdigit(static_cast<unsigned char>(c)) ||
203 (c == '.' && i + 1 < n && std::isdigit(static_cast<unsigned char>(src[i + 1])))) {
204 const size_t start = i;
205 while (i < n) {
206 const char d = src[i];
207 if (d >= '0' && d <= '9') {
208 ++i;
209 } else if (d == '.') {
210 ++i;
211 } else if ((d == 'e' || d == 'E') && i + 1 < n &&
212 ((src[i + 1] >= '0' && src[i + 1] <= '9') ||
213 src[i + 1] == '+' || src[i + 1] == '-')) {
214 i += 2;
215 } else {
216 break;
217 }
218 }
219 add(Tok::Num, src.substr(start, i - start));
220 continue;
221 }
222 if (std::isalpha(static_cast<unsigned char>(c)) || c == '_') {
223 const size_t start = i;
224 while (i < n) {
225 const char d = src[i];
226 if (std::isalnum(static_cast<unsigned char>(d)) || d == '_' || d == '.') ++i;
227 else break;
228 }
229 add(Tok::Ident, src.substr(start, i - start));
230 continue;
231 }
232 const std::string two = (i + 1 < n) ? src.substr(i, 2) : "";
233 if (two == "==" || two == "!=" || two == ">=" || two == "<=" ||
234 two == "&&" || two == "||") {
235 add(Tok::Punct, two);
236 i += 2;
237 continue;
238 }
239 if (c == '{' || c == '}' || c == '(' || c == ')' || c == '[' || c == ']' ||
240 c == ':' || c == ',' || c == '=' || c == '>' || c == '<' || c == '!' ||
241 c == '-') {
242 add(Tok::Punct, std::string(1, c));
243 ++i;
244 continue;
245 }
246 throw ParseError(path_ + ":" + std::to_string(line) + ": 无法识别的字符 '" +
247 std::string(1, c) + "'");
248 }
249 toks_.push_back(Token{Tok::Eof, "", line, static_cast<int>(i - lineStart + 1)});
250 }

253 const Token &t = cur();
254 if (t.kind == Tok::Str) {
255 adv();
256 return DataValue::string(t.value);
257 }
258 if (t.kind == Tok::Num) {

281 }

285 DataValue parseComparison() {
286 if (!isIdent()) fail("条件左侧应为变量名");
287 const std::string varName = adv().value;
288 const Token &op = cur();
289 if (op.kind != Tok::Punct ||
290 (op.value != "==" && op.value != "!=" && op.value != ">" && op.value != "<" &&
291 op.value != ">=" && op.value != "<="))
292 fail("期望比较运算符（== != > < >= <=）");
293 adv();
294 DataValue value = parseLiteralValue();
295 std::string mapped;
296 if (op.value == "==") mapped = "eq";
297 else if (op.value == "!=") mapped = "ne";
298 else if (op.value == ">") mapped = "gt";
299 else if (op.value == "<") mapped = "lt";
300 else if (op.value == ">=") mapped = "ge";
301 else mapped = "le";
302 return DataValue::object({
303 {"var", DataValue::string(varName)},
304 {"op", DataValue::string(mapped)},
305 {"value", std::move(value)},
306 });
307 }

323 DataValue parseAnd() {
324 DataValue left = parseNot();
325 while (isPunct("&&")) {
326 adv();
327 DataValue right = parseNot();
328 bool appended = false;
329 if (left.kind() == DataValue::Kind::Object) {
330 for (auto &kv : *left.getIf<DataValue::Object>()) {
331 if (kv.first == "all" && kv.second.kind() == DataValue::Kind::Array) {
332 kv.second.pushBack(std::move(right));
333 appended = true;
334 break;
335 }
336 }
337 }
338 if (!appended) {
339 left = DataValue::object(
340 {{"all", DataValue::array({std::move(left), std::move(right)})}});
341 }
342 }
343 return left;
344 }

346 DataValue parseOr() {
347 DataValue left = parseAnd();
348 while (isPunct("||")) {
349 adv();
350 DataValue right = parseAnd();
351 bool appended = false;
352 if (left.kind() == DataValue::Kind::Object) {
353 for (auto &kv : *left.getIf<DataValue::Object>()) {
354 if (kv.first == "any" && kv.second.kind() == DataValue::Kind::Array) {
355 kv.second.pushBack(std::move(right));
356 appended = true;
357 break;
358 }
359 }
360 }
361 if (!appended) {
362 left = DataValue::object(
363 {{"any", DataValue::array({std::move(left), std::move(right)})}});
364 }
365 }
366 return left;

369 void parseAttrs(const int lineNum, DataValue::Object &out) {
370 while (cur().line == lineNum && !isPunct("}") && cur().kind != Tok::Eof) {
371 if (!isIdent()) fail("期望属性名");
372 const std::string name = adv().value;
373 if (name == "meta" && isPunct("(")) {
374 adv();
375 DataValue::Object metaFields;
376 while (!isPunct(")")) {
377 if (!isIdent()) fail("meta 键应为标识符");
378 const std::string k = adv().value;
379 expectPunct("=");
380 DataValue v = parseLiteralValue();
381 if (v.kind() != DataValue::Kind::String && v.kind() != DataValue::Kind::Int &&
382 v.kind() != DataValue::Kind::Float && v.kind() != DataValue::Kind::Bool)
383 fail("meta 值只支持标量");
384 metaFields.emplace(k, DataValue::string(scalarToString(v)));
385 if (isPunct(",")) adv();
386 else if (!isPunct(")")) fail("meta 内期望 ',' 或 ')'");
387 }
388 adv(); // )
389 out.emplace("meta", DataValue::object(std::move(metaFields)));
390 continue;
391 }
392 if (name == "tags") {
393 expectPunct("=");
394 if (!isPunct("[")) fail("tags 后应为 [");
395 adv();
396 std::vector<DataValue> arr;
397 while (!isPunct("]")) {
398 if (cur().kind != Tok::Str) fail("tags 元素应为字符串");
399 arr.emplace_back(DataValue::string(adv().value));
400 if (isPunct(",")) adv();
401 else if (!isPunct("]")) fail("tags 内期望 ',' 或 ']'");
402 }
403 adv(); // ]
404 out.emplace("tags", DataValue::array(std::move(arr)));
405 continue;
406 }
407 expectPunct("=");
408 DataValue v = parseLiteralValue();
409 if (name == "weight" || name == "i18n" || name == "id") {
410 out.emplace(name, std::move(v));
411 } else {
412 fail("未知属性 '" + name + "'");
413 }
414 }

418 const int lineNum = cur().line;

447 void parsePool(DataValue::Object &pools) {
448 expectIdent("pool");
449 const std::string poolId = expectIdent("pool 名称");
450 if (pools.find(poolId) != pools.end()) fail("重复 pool ID '" + poolId + "'");
451 long long noRepeat = -1;
452 while (isIdent() && cur().value == "noRepeat") {
453 adv();
454 expectPunct("=");
455 const Token &t = cur();
456 if (t.kind != Tok::Num) fail("noRepeat 应为数字");
457 noRepeat = std::strtoll(adv().value.c_str(), nullptr, 10);
458 }
459 expectPunct("{");
460
461 std::vector<DataValue> lines;
462 int idx = 1;
463 while (!isPunct("}")) {
464 if (cur().kind == Tok::Eof) fail("未闭合的 pool 块");
465 if (isIdent() && cur().value == "pool") fail("pool 块不允许嵌套");
466 if (isIdent() && cur().value == "when") {
467 adv();
468 DataValue cond = parseOr();
469 expectPunct("{");
470 while (!isPunct("}")) {
471 if (cur().kind == Tok::Eof) fail("未闭合的 when 块");
472 lines.push_back(parseLine(poolId, idx, &cond));
473 ++idx;
474 }
475 adv();
476 continue;
477 }
478 lines.push_back(parseLine(poolId, idx, nullptr));
479 ++idx;
480 }
481 adv(); // }
482
483 DataValue::Object poolFields;
484 poolFields.emplace("lines", DataValue::array(std::move(lines)));
485 if (noRepeat >= 0) poolFields.emplace("noRepeat", DataValue::integer(noRepeat));
486 pools.emplace(poolId, DataValue::object(std::move(poolFields)));
487 }

515 eve::StateMutation parseMutation() {
516 DataValue fields = parseObjectArguments();
517 const DataValue* subject = fields.find("subject");
518 const DataValue* key = fields.find("key");
519 const DataValue* kind = fields.find("kind");
520 const DataValue* value = fields.find("value");
521 const DataValue* persistent = fields.find("persistent");
522 for (const auto& field : fields.keys())
523 if (field != "subject" && field != "key" && field != "kind" && field != "value" &&
524 field != "persistent")
525 fail("未知 mutation 字段 '" + field + "'");
526 if (!subject || !subject->isString() || subject->asString().empty())
527 fail("mutation.subject 必须是非空字符串");
528 if (!key || !key->isString() || key->asString().empty())
529 fail("mutation.key 必须是非空字符串");
530 if (!kind || !kind->isString()) fail("mutation.kind 必须是字符串");
531 eve::StateMutation mutation;
532 mutation.subject = subject->asString();
533 mutation.key = key->asString();
534 if (kind->asString() == "set") mutation.kind = eve::MutationKind::Set;
535 else if (kind->asString() == "remove") mutation.kind = eve::MutationKind::Remove;
536 else if (kind->asString() == "addTag") mutation.kind = eve::MutationKind::AddTag;
537 else if (kind->asString() == "removeTag") mutation.kind = eve::MutationKind::RemoveTag;
538 else if (kind->asString() == "addNumber") mutation.kind = eve::MutationKind::AddNumber;
539 else fail("mutation.kind 不受支持 '" + kind->asString() + "'");
540 if (mutation.kind == eve::MutationKind::Set || mutation.kind == eve::MutationKind::AddNumber) {
541 if (!value) fail("set/addNumber mutation 缺少 value");
542 mutation.value = *value;
543 } else if (value) {
544 fail("remove/tag mutation 不允许 value");
545 }
546 if (persistent) {
547 if (!persistent->isBool()) fail("mutation.persistent 必须是 bool");
548 mutation.persistent = persistent->asBool();
549 }
550 return mutation;
551 }

592 const int line = cur().line;
593 const int column = cur().column;
594 expectIdent("route");
595 ConversationRoute route;

598 route.first = expectIdent("稳定 route ID");

603 if (key != "target" && key != "when" && key != "text" && key != "i18n" && key != "payment" &&
604 key != "mutation")
605 fail("未知 route 字段 '" + key + "'");
606 if (key == "payment") {
607 route.payment = parsePayment();
608 continue;
609 }
610 if (key == "mutation") {
611 route.stateMutations.push_back(parseMutation());
612 continue;
613 }
614 expectPunct("=");
615 const std::string value = scalarText();
616 if (key == "target") { route.second = value; hasTarget = true; }
617 else if (key == "when") route.expression = value;
618 else if (key == "text") route.text = value;
619 else if (key == "i18n") route.i18nKey = value;
620 }
621 if (!hasTarget) fail("route 缺少 target");
622 return route;
623 }

625 ConversationAsset::Node parseNode() {
626 const int line = cur().line;
627 const int column = cur().column;
628 expectIdent("node");
629 ConversationAsset::Node node;
630 node.sourceLine = line;
631 node.sourceColumn = column;
632 node.id = expectIdent("node ID");
633 const std::string kind = expectIdent("node 类型");
634 if (kind == "line") node.kind = ConversationAsset::Node::Kind::Line;
635 else if (kind == "branch") node.kind = ConversationAsset::Node::Kind::Branch;
636 else if (kind == "choice") node.kind = ConversationAsset::Node::Kind::Choice;
637 else if (kind == "call") node.kind = ConversationAsset::Node::Kind::Call;
638 else if (kind == "command") node.kind = ConversationAsset::Node::Kind::Command;
639 else if (kind == "wait") node.kind = ConversationAsset::Node::Kind::Wait;
640 else if (kind == "end") node.kind = ConversationAsset::Node::Kind::End;
641 else fail("未知 node 类型 '" + kind + "'");
642 parseNodeAttributes(node, line);
643 if (isPunct("{")) {
644 adv();
645 while (!isPunct("}")) {
646 if (cur().kind == Tok::Eof) fail("未闭合的 node 块");
647 if (!isIdent() || cur().value != "route") fail("node 块只允许 route");
648 node.routes.push_back(parseRoute());
649 }
650 adv();
651 }
652 return node;
653 }

655 ConversationAsset parseConversation() {
656 const int declarationLine = cur().line;
657 const int declarationColumn = cur().column;
658 expectIdent("conversation");
659 ConversationAsset asset;
660 asset.sourceLine = declarationLine;
661 asset.sourceColumn = declarationColumn;
662 asset.id = expectIdent("conversation ID");
663 while (cur().kind != Tok::Eof && cur().line == declarationLine && !isPunct("{")) {
664 const std::string key = expectIdent("conversation 字段");
665 if (key != "entry" && key != "version") fail("未知 conversation 字段 '" + key + "'");
666 expectPunct("=");
667 const std::string value = scalarText();
668 if (key == "entry") asset.entry = value;
669 else asset.version = static_cast<int>(std::strtol(value.c_str(), nullptr, 10));
670 }
671 expectPunct("{");
672 while (!isPunct("}")) {
673 if (cur().kind == Tok::Eof) fail("未闭合的 conversation 块");
674 if (!isIdent()) fail("conversation 内期望 parameter 或 node");
675 if (cur().value == "parameter") {
676 const int line = cur().line;
677 const int column = cur().column;
678 adv();
679 const std::string name = expectIdent("parameter 名称");
680 const std::string type = expectIdent("parameter 类型");
681 ConversationAsset::Parameter parameter;
682 parameter.name = name;
683 parameter.sourceLine = line;
684 parameter.sourceColumn = column;
685 if (type != "string" && type != "int" && type != "float" && type != "bool")
686 fail("未知 parameter 类型 '" + type + "'");
687 if (type == "string") parameter.type = ConversationAsset::Parameter::Type::String;
688 else if (type == "int") parameter.type = ConversationAsset::Parameter::Type::Int;
689 else if (type == "float") parameter.type = ConversationAsset::Parameter::Type::Float;
690 else parameter.type = ConversationAsset::Parameter::Type::Bool;
691 while (cur().line == line && cur().kind != Tok::Eof) {
692 const std::string field = expectIdent("parameter 字段");
693 if (field == "required") parameter.required = true;
694 else if (field == "optional") parameter.required = false;
695 else if (field == "default") {
696 expectPunct("=");
697 parameter.defaultValue = toDialogueStateValue(parseLiteralValue());
698 parameter.required = false;
699 } else fail("未知 parameter 字段 '" + field + "'");
700 }
701 asset.parameters.push_back(std::move(parameter));
702 } else if (cur().value == "node") {
703 asset.nodes.push_back(parseNode());
704 } else {
705 fail("未知 conversation 字段 '" + cur().value + "'");
706 }
707 }
708 adv();
709 return asset;
710 }
