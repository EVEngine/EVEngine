88eve::Result<game_event::GameEvent> makeEvent(const SettlementContext& context) {
89 const auto& request = context.request();
90 const auto& result = context.projectedResult();
91
92 auto schema = eve::LogicalId::parse("settlement:result");
93 if (!schema)
94 return failure<game_event::GameEvent>(eve::DiagnosticCode::InvariantViolation,
95 "settlement result schema id is not a valid LogicalId", "event.schemaId");
96
97 eve::Value::Object payload;
98 payload["kind"] = request.kind;
99 payload["requested"] = result.requested;
100 payload["applied"] = result.applied;
101 payload["absorbed"] = result.absorbed;
102 payload["resisted"] = result.resisted;
103 payload["clamped"] = result.clamped;
104 payload["critical"] = result.critical;
105 payload["tick"] = static_cast<std::int64_t>(result.tick.value());
106 payload["context"] = request.context;
107
108 eve::Value::Array tags;
109 tags.reserve(request.tags.size());
110 for (const auto& tag : request.tags) tags.emplace_back(tag);
111 payload["tags"] = eve::Value(std::move(tags));
112
113 eve::Value::Array stages;
114 stages.reserve(result.stages.size());
115 for (const auto& stage : result.stages) {
116 eve::Value::Object stageValue;
117 stageValue["name"] = stage.name;
118 stageValue["kind"] = stageKindName(stage.kind);
119 stageValue["status"] = std::string(eve::statusCodeName(stage.status));
120 stageValue["before"] = stage.before;
121 stageValue["after"] = stage.after;
122 stageValue["details"] = stage.details;
123 stages.emplace_back(eve::Value(std::move(stageValue)));
124 }
125 payload["stages"] = eve::Value(std::move(stages));
126
127 auto encoded = eve::Value(std::move(payload)).toJson();
128 if (!encoded) {
129 const auto status = encoded.status();
130 return failure<game_event::GameEvent>(status);
131 }
132
133 game_event::GameEvent envelope;
134 envelope.type = "settlement.result";
135 envelope.source = request.source.isValid() ? request.source.format() : std::string{};
136 envelope.subject = request.target.format();
137 envelope.causation = request.causation;
138 envelope.correlation = request.correlation;
139 envelope.schemaId = *schema;
140 envelope.schemaVersion = eve::SchemaVersion(1);
141 envelope.tick = request.tick;
142 envelope.flags = result.critical ? 1u : 0u;
143 envelope.payload = std::move(encoded).takeValue();
144 return eve::Result<game_event::GameEvent>::success(std::move(envelope));
145}

208eve::Result<void> PreparedApply::commit() {
209 if (committed_) return success(eve::StatusCode::Applied);
210 if (!isValid())
211 return failure<void>(eve::DiagnosticCode::InvariantViolation,
212 "settlement mutation is empty or already rolled back", "apply");
213
214 try {
215 auto outcome = commit_();
216 const bool passed = outcome.ok();
217 if (passed) {
218 committed_ = true;
219 return outcome;
220 }
221 const auto status = outcome.status();
222 return failure<void>(status);
223 } catch (const std::exception& exception) {
224 return failure<void>(eve::DiagnosticCode::Failed,
225 std::string("settlement apply commit threw: ") + exception.what(), "apply");
226 } catch (...) {
227 return failure<void>(eve::DiagnosticCode::Failed, "settlement apply commit threw an unknown exception",
228 "apply");
229 }
230}

232void PreparedApply::rollback() noexcept {
233 if (rolledBack_ || !rollback_) return;
234 try {
235 rollback_();
236 } catch (...) {
237 // A rollback exception would make the public failure contract false:
238 // the caller could observe a failure while domain state remained
239 // partially applied. There is no safe recovery path from noexcept
240 // rollback (including the destructor path), so fail closed.
241 std::terminate();
242 }
243 rolledBack_ = true;
244}

343 double next = std::max(0.0, magnitude_);

480eve::Result<SettlementResult> SettlementPipeline::settle(const SettlementRequest& request, ISettlementPolicy& policy,
481 game_event::GameEventLog* events) const {
482 SettlementResult result;
483 result.requested = request.magnitude;
484 result.tick = request.tick;
485 SettlementContext context(request, result, policy);
486
487 std::vector<const StageEntry*> ordered;
488 ordered.reserve(stages_.size());
489 for (const auto& stage : stages_) ordered.push_back(&stage);
490 std::sort(ordered.begin(), ordered.end(), [](const auto* left, const auto* right) {
491 const auto leftKind = static_cast<std::uint8_t>(left->kind);
492 const auto rightKind = static_cast<std::uint8_t>(right->kind);
493 if (leftKind != rightKind) return leftKind < rightKind;
494 // Terminal preparation is an unregistrable phase boundary. It must
495 // sort after every custom stage in the same phase, even when a custom
496 // stage deliberately uses a larger priority or a lexicographically
497 // later name.
498 if (left->terminal != right->terminal) return !left->terminal && right->terminal;
499 if (left->priority != right->priority) return left->priority < right->priority;
500 if (left->name != right->name) return left->name < right->name;
501 return left->registration < right->registration;
502 });
503
504 auto rollback = [&]() noexcept {
505 if (auto* pending = context.pendingApply()) pending->rollback();
506 };
507
508 for (const StageEntry* entry : ordered) {
509 context.beginStage();
510 const double before = context.magnitude();
511 auto outcome = entry->function(context);
512 const bool outcomeOk = outcome.ok();
513 const auto outcomeStatus = outcome.status();
514
515 context.synchronizeResult();
516 auto frame = validateFrame(context);
517 const bool frameOk = frame.ok();
518 const auto frameStatus = frame.status();
519 const auto* violation = context.mutationViolation();
520 const bool violationPresent = violation != nullptr;
521 const eve::Status violationStatus =
522 violationPresent ? eve::Status::failure(*violation) : eve::Status::success();
523
524 SettlementStageResult stageResult;
525 stageResult.kind = entry->kind;
526 stageResult.name = entry->name;
527 stageResult.status = !outcomeOk ? outcomeStatus.code()
528 : !frameOk ? frameStatus.code()
529 : violationPresent ? violationStatus.code()
530 : outcomeStatus.code();
531 stageResult.before = before;
532 stageResult.after = context.magnitude();
533 stageResult.details = Value(context.stageDetails());
534 result.stages.push_back(std::move(stageResult));
535
536 if (!outcomeOk) {
537 rollback();
538 return failure<SettlementResult>(stageFailureStatus(entry->name, outcomeStatus));
539 }
540 if (!frameOk) {
541 rollback();
542 return failure<SettlementResult>(stageFailureStatus(entry->name, frameStatus));
543 }
544 if (violationPresent) {
545 rollback();
546 return failure<SettlementResult>(stageFailureStatus(entry->name, violationStatus));
547 }
548 }
549
550 context.synchronizeResult();
551 auto* pending = context.pendingApply();
552 if (pending == nullptr)
553 return failure<SettlementResult>(eve::DiagnosticCode::InvariantViolation,
554 "settlement pipeline did not prepare an apply mutation", "apply");
555
556 auto committed = pending->commit();
557 const bool committedOk = committed.ok();
558 if (!committedOk) {
559 const auto status = committed.status();
560 rollback();
561 return failure<SettlementResult>(status);
562 }
563
564 if (const auto* preparedEvent = context.pendingEvent()) {
565 game_event::GameEvent envelope = *preparedEvent;
566 result.event = envelope;
567 if (events != nullptr) {
568 try {
569 auto appended = events->append(envelope);
570 const bool appendedOk = appended.ok();
571 if (!appendedOk) {
572 const auto status = appended.status();
573 rollback();
574 return failure<SettlementResult>(status);
575 }
576 const auto sequence = std::move(appended).takeValue();
577 envelope.sequence = sequence;
578 if (const auto* stored = events->find(sequence)) {
579 result.event = *stored;
580 } else {
581 result.event = envelope;
582 }
583 } catch (const std::exception& exception) {
584 rollback();
585 return failure<SettlementResult>(
586 eve::DiagnosticCode::Failed,
587 std::string("settlement result event append threw: ") + exception.what(), "event");
588 } catch (...) {
589 rollback();
590 return failure<SettlementResult>(eve::DiagnosticCode::Failed,
591 "settlement result event append threw an unknown exception", "event");
592 }
593 }
594 }
595
596 const auto status = result.applied == 0.0 ? eve::StatusCode::NoOp : eve::StatusCode::Applied;
597 return eve::Result<SettlementResult>::success(std::move(result), eve::Status::success(status));
598}
