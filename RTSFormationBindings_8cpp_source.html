19Result<CombatStance> parseCombatStance(std::string_view text) {
20 if (text == "passive") return Result<CombatStance>::success(CombatStance::Passive);
21 if (text == "defensive") return Result<CombatStance>::success(CombatStance::Defensive);
22 if (text == "aggressive") return Result<CombatStance>::success(CombatStance::Aggressive);
23 return Result<CombatStance>::failure(Diagnostic::error(
24 DiagnosticCode::InvalidArgument, "RTS combat stance must be passive, defensive, or aggressive", "stance"));
25}

27Result<FormationSpec> parseFormation(const std::string& kind, float spacing, int columns, float rotation) {
28 FormationSpec spec;
29 if (kind == "line")
30 spec.kind = FormationKind::Line;
31 else if (kind == "grid")
32 spec.kind = FormationKind::Grid;
33 else if (kind == "wedge")
34 spec.kind = FormationKind::Wedge;
35 else if (kind == "column")
36 spec.kind = FormationKind::Column;
37 else if (kind == "dispersed")
38 spec.kind = FormationKind::Dispersed;
39 else
40 return Result<FormationSpec>::failure(
41 Diagnostic::error(DiagnosticCode::InvalidArgument, "Unknown formation layout", "formation.kind"));
42 spec.spacing = spacing;
43 spec.columns = columns;
44 spec.rotationRadians = rotation;
45 auto valid = spec.validate();
46 if (!valid) return Result<FormationSpec>::failure(valid.status());
47 return Result<FormationSpec>::success(spec);
48}
