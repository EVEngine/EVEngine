85Grid2D combine(const Grid2D& a, const Grid2D& b, std::string_view operation) {
86 Grid2D out = a;
87 for (int y = 0; y < a.getHeight(); ++y) {
88 for (int x = 0; x < a.getWidth(); ++x) {
89 const bool left = occupied(a, x, y);
90 const bool right = occupied(b, x, y);
91 bool keep = false;
92 if (operation == "grid.union") keep = left || right;
93 if (operation == "grid.intersect") keep = left && right;
94 if (operation == "grid.subtract") keep = left && !right;
95 if (!keep) {
96 out.setCell(x, y, int(Semantic::Empty));
97 out.setDetail(x, y, 0);
98 } else if (!left) {
99 out.setCell(x, y, b.getCell(x, y));
100 out.setDetail(x, y, b.getDetail(x, y));
101 }
102 }
103 }
104 return out;
105}

107Grid2D morphology(const Grid2D& input, int radius, bool expand) {
108 Grid2D out = input;
109 radius = std::max(1, radius);
110 for (int y = 0; y < input.getHeight(); ++y) {
111 for (int x = 0; x < input.getWidth(); ++x) {
112 bool result = expand ? occupied(input, x, y) : true;
113 for (int oy = -radius; oy <= radius && (expand ? !result : result); ++oy) {
114 for (int ox = -radius; ox <= radius; ++ox) {
115 const bool sample = occupied(input, x + ox, y + oy);
116 result = expand ? (result || sample) : (result && sample);
117 if (expand ? result : !result) break;
118 }
119 }
120 if (result) {
121 if (!occupied(out, x, y)) out.setCell(x, y, int(Semantic::Floor));
122 } else {
123 out.setCell(x, y, int(Semantic::Empty));
124 out.setDetail(x, y, 0);
125 }
126 }
127 }
128 return out;
129}

140Result<PointSet> gridToPoints(const Grid2D& grid, int semantic, float cellSize, float originX, float originZ,
141 std::string_view nodeId) {
142 if (!std::isfinite(cellSize) || cellSize <= 0.f)
143 return fail<PointSet>(DiagnosticCode::InvalidArgument, "cellSize must be finite and positive",
144 std::string(nodeId));
145 PointSet points;
146 points.reserve(grid.cells().size());
147 std::uint64_t ordinal = 0;
148 for (int y = 0; y < grid.getHeight(); ++y) {
149 for (int x = 0; x < grid.getWidth(); ++x) {
150 const int cell = grid.getCell(x, y);
151 if (cell == int(Semantic::Empty) || (semantic >= 0 && cell != semantic)) continue;
152 const int index =
153 points.add(originX + (float(x) + 0.5f) * cellSize, 0.f, originZ + (float(y) + 0.5f) * cellSize);
154 auto idResult = points.trySetPointId(index, derivePointId(hashText(nodeId), ++ordinal));
155 if (!idResult.ok()) return Result<PointSet>::failure(idResult.status());
156 auto xResult = points.trySetIntAttribute(index, "cell_x", x);
157 if (!xResult.ok()) return Result<PointSet>::failure(xResult.status());
158 auto yResult = points.trySetIntAttribute(index, "cell_y", y);
159 if (!yResult.ok()) return Result<PointSet>::failure(yResult.status());
160 auto semanticResult = points.trySetIntAttribute(index, "semantic", cell);
161 if (!semanticResult.ok()) return Result<PointSet>::failure(semanticResult.status());
162 auto detailResult = points.trySetIntAttribute(index, "detail", grid.getDetail(x, y));
163 if (!detailResult.ok()) return Result<PointSet>::failure(detailResult.status());
164 }
165 }
166 return Result<PointSet>::success(std::move(points));
167}

171Result<void> GridGraph::addNode(std::string id, std::string operation) {
172 if (id.empty()) return failVoid(DiagnosticCode::InvalidArgument, "node id is empty");
173 if (!specFor(operation)) return failVoid(DiagnosticCode::NotFound, "unknown operation: " + operation, id);
174 if (nodes_.contains(id)) return failVoid(DiagnosticCode::Conflict, "duplicate node id: " + id, id);
175 Node node;
176 node.id = id;
177 node.operation = std::move(operation);
178 nodes_.emplace(id, std::move(node));
179 order_.push_back(std::move(id));
180 ++revision_;
181 return Result<void>::success();
182}

184Result<void> GridGraph::connect(std::string_view fromId, std::string_view toId, int inputIndex) {
185 const auto from = nodes_.find(std::string(fromId));
186 const auto to = nodes_.find(std::string(toId));
187 if (from == nodes_.end() || to == nodes_.end())
188 return failVoid(DiagnosticCode::NotFound, "connection references an unknown node", std::string(toId));
189 const OperationSpec* target = specFor(to->second.operation);
190 if (!target || inputIndex < 0 || inputIndex >= target->inputs)
191 return failVoid(DiagnosticCode::InvalidArgument, "input slot is outside the operation contract",
192 std::string(toId));
193 const auto sourceType = operationOutputType(from->second.operation);
194 if (!sourceType.ok()) return Result<void>::failure(sourceType.status());
195 const GridGraphValueType expected =
196 to->second.operation == "point.subgraph" ? GridGraphValueType::PointSet : GridGraphValueType::Grid;
197 if (sourceType.value() != expected)
198 return failVoid(DiagnosticCode::TypeMismatch, "typed graph ports are incompatible", std::string(toId));
199 to->second.inputs[inputIndex] = std::string(fromId);
200 invalidateFrom(toId);
201 ++revision_;
202 return Result<void>::success();
203}

277ResultRef<const GridGraphValue> GridGraph::evaluate(std::string_view id, std::unordered_map<std::string, int>& states) {
278 const auto found = nodes_.find(std::string(id));
279 if (found == nodes_.end())
280 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::NotFound, "unknown node",
281 std::string(id));
282 Node& node = found->second;
283 if (node.cacheValid) return ResultRef<const GridGraphValue>::success(std::cref(node.cache));
284 if (states[node.id] == 1)
285 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::Conflict, "graph contains a cycle",
286 node.id);
287 states[node.id] = 1;
288 const OperationSpec* spec = specFor(node.operation);
289 std::vector<std::reference_wrapper<const GridGraphValue>> inputs;
290 for (int i = 0; i < spec->inputs; ++i) {
291 if (node.inputs[i].empty())
292 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::NotFound,
293 "required input is disconnected", node.id);
294 auto input = evaluate(node.inputs[i], states);
295 if (!input.ok()) return ResultRef<const GridGraphValue>::failure(input.status());
296 inputs.push_back(input.value());
297 }
298
299 if (node.operation == "grid.input") {
300 if (!node.hasInputGrid)
301 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::NotFound,
302 "grid.input has no bound value", node.id);
303 node.cache = node.inputGrid;
304 } else if (node.operation == "generate.registry") {
305 const auto algorithm = node.strings.find("algorithm");
306 if (algorithm == node.strings.end() || algorithm->second.empty())
307 return fail<std::reference_wrapper<const GridGraphValue>>(
308 DiagnosticCode::InvalidArgument, "generate.registry requires a non-empty algorithm parameter",
309 node.id);
310 Params params;
311 params.setSize(node.ints.contains("width") ? node.ints["width"] : 32,
312 node.ints.contains("height") ? node.ints["height"] : 32);
313 params.setSeed(std::uint32_t(node.ints.contains("seed") ? node.ints["seed"] : 1));
314 for (const auto& [key, value] : node.ints)
315 if (key != "width" && key != "height" && key != "seed") params.setInt(key, value);
316 for (const auto& [key, value] : node.floats) params.setFloat(key, value);
317 for (const auto& [key, value] : node.strings)
318 if (key != "algorithm") params.setString(key, value);
319 auto& registry = GeneratorRegistry::instance();
320 registry.registerBuiltins();
321 if (!registry.has(algorithm->second))
322 return fail<std::reference_wrapper<const GridGraphValue>>(
323 DiagnosticCode::NotFound, "registered generator was not found: " + algorithm->second, node.id);
324 Grid2D generated;
325 std::string error;
326 if (!registry.generate(algorithm->second, params, generated, error))
327 return fail<std::reference_wrapper<const GridGraphValue>>(
328 DiagnosticCode::Failed, error.empty() ? "registered generator failed" : std::move(error), node.id);
329 node.cache = std::move(generated);
330 } else if (node.operation.starts_with("generate.")) {
331 gridgraph::GenerateSettings settings;
332 settings.width = node.ints.contains("width") ? node.ints["width"] : 32;
333 settings.height = node.ints.contains("height") ? node.ints["height"] : 32;
334 settings.semantic = node.ints.contains("semantic") ? node.ints["semantic"] : int(Semantic::Floor);
335 settings.seed = std::uint64_t(node.ints.contains("seed") ? node.ints["seed"] : 1);
336 settings.a = node.ints.contains("a") ? node.ints["a"] : 1;
337 settings.b = node.ints.contains("b") ? node.ints["b"] : 0;
338 settings.c = node.ints.contains("c") ? node.ints["c"] : 1;
339 settings.x = node.floats.contains("x") ? node.floats["x"] : 0.5f;
340 settings.y = node.floats.contains("y") ? node.floats["y"] : 0.f;
341 auto generated = gridgraph::generate(node.operation, settings);
342 if (!generated.ok()) return ResultRef<const GridGraphValue>::failure(generated.status());
343 node.cache = std::move(generated).takeValue();
344 } else if (node.operation == "select.semantic") {
345 const auto* source = std::get_if<Grid2D>(&inputs[0].get());
346 if (!source)
347 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::TypeMismatch,
348 "grid input required", node.id);
349 Grid2D out;
350 out.resize(source->getWidth(), source->getHeight());
351 const int semantic = node.ints.contains("semantic") ? node.ints["semantic"] : int(Semantic::Floor);
352 for (int y = 0; y < source->getHeight(); ++y)
353 for (int x = 0; x < source->getWidth(); ++x)
354 if (source->getCell(x, y) == semantic) {
355 out.setCell(x, y, semantic);
356 out.setDetail(x, y, source->getDetail(x, y));
357 }
358 for (const auto& [key, value] : source->metadata()) out.setMeta(key, value);
359 node.cache = std::move(out);
360 } else if (node.operation == "select.detail_range") {
361 const auto* source = std::get_if<Grid2D>(&inputs[0].get());
362 if (!source)
363 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::TypeMismatch,
364 "grid input required", node.id);
365 Grid2D out;
366 out.resize(source->getWidth(), source->getHeight());
367 const int minimum = node.ints.contains("min") ? node.ints["min"] : 0;
368 const int maximum = node.ints.contains("max") ? node.ints["max"] : 255;
369 for (int y = 0; y < source->getHeight(); ++y)
370 for (int x = 0; x < source->getWidth(); ++x)
371 if (occupied(*source, x, y) && source->getDetail(x, y) >= minimum &&
372 source->getDetail(x, y) <= maximum) {
373 out.setCell(x, y, source->getCell(x, y));
374 out.setDetail(x, y, source->getDetail(x, y));
375 }
376 node.cache = std::move(out);
377 } else if (node.operation.starts_with("select.")) {
378 const auto* source = std::get_if<Grid2D>(&inputs[0].get());
379 if (!source)
380 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::TypeMismatch,
381 "grid input required", node.id);
382 auto selected = gridgraph::select(*source, node.operation, node.ints.contains("mode") ? node.ints["mode"] : 0,
383 node.ints.contains("count") ? node.ints["count"] : 0,
384 node.floats.contains("weight") ? node.floats["weight"] : 0.5f,
385 std::uint64_t(node.ints.contains("seed") ? node.ints["seed"] : 1),
386 node.strings.contains("rule") ? node.strings["rule"] : "*********");
387 if (!selected.ok()) return ResultRef<const GridGraphValue>::failure(selected.status());
388 node.cache = std::move(selected).takeValue();
389 } else if (node.operation == "grid.path") {
390 const auto* navigation = std::get_if<Grid2D>(&inputs[0].get());
391 const auto* starts = std::get_if<Grid2D>(&inputs[1].get());
392 const auto* targets = std::get_if<Grid2D>(&inputs[2].get());
393 if (!navigation || !starts || !targets)
394 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::TypeMismatch,
395 "three grid inputs required", node.id);
396 auto path = gridgraph::findPath(*navigation, *starts, *targets,
397 node.ints.contains("semantic") ? node.ints["semantic"] : int(Semantic::Floor));
398 if (!path.ok()) return ResultRef<const GridGraphValue>::failure(path.status());
399 node.cache = std::move(path).takeValue();
400 } else if (node.operation == "grid.union" || node.operation == "grid.intersect" ||
401 node.operation == "grid.subtract") {
402 const auto* first = std::get_if<Grid2D>(&inputs[0].get());
403 const auto* second = std::get_if<Grid2D>(&inputs[1].get());
404 if (!first || !second || !sameSize(*first, *second))
405 return fail<std::reference_wrapper<const GridGraphValue>>(
406 DiagnosticCode::TypeMismatch, "grid inputs must have equal dimensions", node.id);
407 node.cache = combine(*first, *second, node.operation);
408 } else if (node.operation == "grid.invert") {
409 const auto* source = std::get_if<Grid2D>(&inputs[0].get());
410 if (!source)
411 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::TypeMismatch,
412 "grid input required", node.id);
413 Grid2D out = *source;
414 const int semantic = node.ints.contains("semantic") ? node.ints["semantic"] : int(Semantic::Floor);
415 for (int y = 0; y < out.getHeight(); ++y)
416 for (int x = 0; x < out.getWidth(); ++x)
417 out.setCell(x, y, occupied(*source, x, y) ? int(Semantic::Empty) : semantic);
418 node.cache = std::move(out);
419 } else if (node.operation == "grid.expand" || node.operation == "grid.shrink") {
420 const auto* source = std::get_if<Grid2D>(&inputs[0].get());
421 if (!source)
422 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::TypeMismatch,
423 "grid input required", node.id);
424 node.cache = morphology(*source, node.ints.contains("radius") ? node.ints["radius"] : 1,
425 node.operation == "grid.expand");
426 } else if (node.operation == "grid.smooth") {
427 const auto* source = std::get_if<Grid2D>(&inputs[0].get());
428 if (!source)
429 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::TypeMismatch,
430 "grid input required", node.id);
431 Grid2D out = *source;
432 const int threshold = node.ints.contains("threshold") ? node.ints["threshold"] : 4;
433 for (int y = 0; y < source->getHeight(); ++y) {
434 for (int x = 0; x < source->getWidth(); ++x) {
435 int neighbors = 0;
436 for (int oy = -1; oy <= 1; ++oy)
437 for (int ox = -1; ox <= 1; ++ox)
438 if ((ox != 0 || oy != 0) && occupied(*source, x + ox, y + oy)) ++neighbors;
439 out.setCell(x, y, neighbors >= threshold ? int(Semantic::Floor) : int(Semantic::Empty));
440 }
441 }
442 node.cache = std::move(out);
443 } else if (node.operation == "grid.autotile") {
444 const auto* source = std::get_if<Grid2D>(&inputs[0].get());
445 if (!source)
446 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::TypeMismatch,
447 "grid input required", node.id);
448 Grid2D out = *source;
449 auto autotiled = autotileOccupiedGridInPlace(out);
450 if (!autotiled.ok())
451 return ResultRef<const GridGraphValue>::failure(autotiled.status());
452 node.cache = std::move(out);
453 } else if (node.operation == "convert.grid_to_points") {
454 const auto* source = std::get_if<Grid2D>(&inputs[0].get());
455 if (!source)
456 return fail<std::reference_wrapper<const GridGraphValue>>(DiagnosticCode::TypeMismatch,
457 "grid input required", node.id);
458 auto points = gridToPoints(*source, node.ints.contains("semantic") ? node.ints["semantic"] : -1,
459 node.floats.contains("cellSize") ? node.floats["cellSize"] : 1.f,
460 node.floats.contains("originX") ? node.floats["originX"] : 0.f,
461 node.floats.contains("originZ") ? node.floats["originZ"] : 0.f, node.id);
462 if (!points.ok()) return ResultRef<const GridGraphValue>::failure(points.status());
463 node.cache = std::move(points).takeValue();
464 } else if (node.operation == "point.subgraph") {
465 const auto* source = std::get_if<PointSet>(&inputs[0].get());
466 if (!source || !node.pointSubgraph)
467 return fail<std::reference_wrapper<const GridGraphValue>>(
468 DiagnosticCode::TypeMismatch, "bound PointGraph and PointSet input required", node.id);
469 PointSet copy = *source;
470 if (!node.pointSubgraph->setNodePoints(node.pointInputNode, &copy))
471 return fail<std::reference_wrapper<const GridGraphValue>>(
472 DiagnosticCode::InvalidArgument, "nested PointGraph input binding failed", node.id);
473 auto output = node.pointSubgraph->executeResult(node.pointOutputNode);
474 if (!output.ok()) return ResultRef<const GridGraphValue>::failure(output.status());
475 node.cache = std::move(output).takeValue();
476 }
477 states[node.id] = 2;
478 node.cacheValid = true;
479 return ResultRef<const GridGraphValue>::success(std::cref(node.cache));
480}

489Result<void> GridGraph::validate(std::string_view outputId) const {
490 std::unordered_map<std::string, int> states;
491 const auto visit = [&](const auto& self, std::string_view id) -> Result<void> {
492 const auto found = nodes_.find(std::string(id));
493 if (found == nodes_.end()) return failVoid(DiagnosticCode::NotFound, "unknown node", std::string(id));
494 if (states[found->first] == 1)
495 return failVoid(DiagnosticCode::Conflict, "graph contains a cycle", found->first);
496 if (states[found->first] == 2) return Result<void>::success();
497 states[found->first] = 1;
498 const OperationSpec* spec = specFor(found->second.operation);
499 for (int index = 0; index < spec->inputs; ++index) {
500 if (found->second.inputs[index].empty())
501 return failVoid(DiagnosticCode::NotFound, "required input is disconnected", found->first);
502 auto input = self(self, found->second.inputs[index]);
503 if (!input.ok()) return input;
504 }
505 if (found->second.operation == "grid.input" && !found->second.hasInputGrid)
506 return failVoid(DiagnosticCode::NotFound, "grid.input has no bound value", found->first);
507 if (found->second.operation == "point.subgraph" &&
508 (!found->second.pointSubgraph || found->second.pointInputNode.empty() ||
509 found->second.pointOutputNode.empty()))
510 return failVoid(DiagnosticCode::NotFound, "point.subgraph has no complete binding", found->first);
511 states[found->first] = 2;
512 return Result<void>::success();
513 };
514 return visit(visit, outputId);
515}

521void GridGraph::invalidateFrom(std::string_view id) {
522 std::queue<std::string> pending;
523 std::unordered_set<std::string> visited;
524 pending.push(std::string(id));
525 while (!pending.empty()) {
526 const std::string current = pending.front();
527 pending.pop();
528 if (!visited.insert(current).second) continue;
529 const auto found = nodes_.find(current);
530 if (found != nodes_.end()) found->second.cacheValid = false;
531 for (const auto& [otherId, node] : nodes_)
532 if (node.inputs[0] == current || node.inputs[1] == current || node.inputs[2] == current)
533 pending.push(otherId);
534 }

550std::string GridGraph::serializeDefinition() const {
551 std::ostringstream out;
552 out << "EVPCG_GRID_GRAPH 1\n" << std::setprecision(9);
553 for (const auto& id : order_) {
554 const auto& node = nodes_.at(id);
555 out << "NODE " << std::quoted(id) << ' ' << std::quoted(node.operation) << '\n';
556 std::vector<std::string> keys;
557 for (const auto& [key, value] : node.ints) keys.push_back(key);
558 std::sort(keys.begin(), keys.end());
559 for (const auto& key : keys)
560 out << "INT " << std::quoted(id) << ' ' << std::quoted(key) << ' ' << node.ints.at(key) << '\n';
561 keys.clear();
562 for (const auto& [key, value] : node.floats) keys.push_back(key);
563 std::sort(keys.begin(), keys.end());
564 for (const auto& key : keys)
565 out << "FLOAT " << std::quoted(id) << ' ' << std::quoted(key) << ' ' << node.floats.at(key) << '\n';
566 keys.clear();
567 for (const auto& [key, value] : node.strings) keys.push_back(key);
568 std::sort(keys.begin(), keys.end());
569 for (const auto& key : keys)
570 out << "STRING " << std::quoted(id) << ' ' << std::quoted(key) << ' ' << std::quoted(node.strings.at(key))
571 << '\n';
572 }
573 for (const auto& id : order_) {
574 const auto& node = nodes_.at(id);
575 for (int input = 0; input < 3; ++input)
576 if (!node.inputs[input].empty())
577 out << "EDGE " << std::quoted(node.inputs[input]) << ' ' << std::quoted(id) << ' ' << input << '\n';
578 }
579 out << "END\n";
580 return out.str();
581}

583Result<void> GridGraph::deserializeDefinition(std::string_view definition) {
584 GridGraph replacement;
585 std::istringstream input{std::string(definition)};
586 std::string magic;
587 int version = 0;
588 if (!(input >> magic >> version) || magic != "EVPCG_GRID_GRAPH" || version != 1)
589 return failVoid(DiagnosticCode::ParseError, "invalid grid graph header");
590 std::string line;
591 std::getline(input, line);
592 bool ended = false;
593 while (std::getline(input, line)) {
594 if (line.empty()) continue;
595 std::istringstream record(line);
596 std::string kind;
597 record >> kind;
598 if (kind == "END") {
599 ended = true;
600 break;
601 }
602 std::string id;
603 if (kind == "NODE") {
604 std::string operation;
605 if (!(record >> std::quoted(id) >> std::quoted(operation)))
606 return failVoid(DiagnosticCode::ParseError, "invalid NODE record");
607 auto result = replacement.addNode(std::move(id), std::move(operation));
608 if (!result.ok()) return result;
609 } else if (kind == "EDGE") {
610 std::string target;
611 int slot = -1;
612 if (!(record >> std::quoted(id) >> std::quoted(target) >> slot))
613 return failVoid(DiagnosticCode::ParseError, "invalid EDGE record");
614 auto result = replacement.connect(id, target, slot);
615 if (!result.ok()) return result;
616 } else if (kind == "INT") {
617 std::string key;
618 int value = 0;
619 if (!(record >> std::quoted(id) >> std::quoted(key) >> value))
620 return failVoid(DiagnosticCode::ParseError, "invalid INT record");
621 auto result = replacement.setNodeInt(id, std::move(key), value);
622 if (!result.ok()) return result;
623 } else if (kind == "FLOAT") {
624 std::string key;
625 float value = 0.f;
626 if (!(record >> std::quoted(id) >> std::quoted(key) >> value))
627 return failVoid(DiagnosticCode::ParseError, "invalid FLOAT record");
628 auto result = replacement.setNodeFloat(id, std::move(key), value);
629 if (!result.ok()) return result;
630 } else if (kind == "STRING") {
631 std::string key;
632 std::string value;
633 if (!(record >> std::quoted(id) >> std::quoted(key) >> std::quoted(value)))
634 return failVoid(DiagnosticCode::ParseError, "invalid STRING record");
635 auto result = replacement.setNodeString(id, std::move(key), std::move(value));
636 if (!result.ok()) return result;
637 } else {
638 return failVoid(DiagnosticCode::ParseError, "unknown grid graph record: " + kind);
639 }
640 record >> std::ws;
641 if (!record.eof()) return failVoid(DiagnosticCode::ParseError, "trailing grid graph record data");
642 }
643 if (!ended) return failVoid(DiagnosticCode::ParseError, "grid graph END record is missing");
644 std::unordered_map<std::string, int> states;
645 const auto visit = [&](const auto& self, const std::string& id) -> Result<void> {
646 if (states[id] == 2) return Result<void>::success();
647 if (states[id] == 1) return failVoid(DiagnosticCode::Conflict, "cycle in serialized graph", id);
648 states[id] = 1;
649 const auto found = replacement.nodes_.find(id);
650 if (found == replacement.nodes_.end()) return failVoid(DiagnosticCode::NotFound, "unknown node", id);
651 for (const auto& dependency : found->second.inputs) {
652 if (dependency.empty()) continue;
653 auto result = self(self, dependency);
654 if (!result.ok()) return result;
655 }
656 states[id] = 2;
657 return Result<void>::success();
658 };
659 for (const auto& id : replacement.order_) {
660 auto result = visit(visit, id);
661 if (!result.ok()) return result;
662 }
663 *this = std::move(replacement);
664 return Result<void>::success();
665}
