70Result<void> ActionDefinition::validate() const {
71 if (!id.isValid()) return failure(DiagnosticCode::InvalidArgument, "action definition id is invalid", "id");
72 if (!condition.isValid())
73 return failure(DiagnosticCode::InvalidArgument, "action condition is invalid", "condition");
74 if (timing.windup.nanoseconds() < 0 || timing.active.nanoseconds() < 0 || timing.recover.nanoseconds() < 0)
75 return failure(DiagnosticCode::InvalidArgument, "action phase durations must be non-negative", "timing");
76
77 switch (targetingMode) {
78 case TargetingMode::None:
79 case TargetingMode::Explicit:
80 if (targetingSpec)
81 return failure(DiagnosticCode::InvalidArgument, "targetingSpec is only valid for Query actions",
82 "targetingSpec");
83 break;
84 case TargetingMode::Query:
85 if (!targetingSpec)
86 return failure(DiagnosticCode::InvalidArgument, "Query actions require a targetingSpec",
87 "targetingSpec");
88 {
89 auto valid = targetingSpec->validate();
90 if (!valid) return failureFrom(valid.status());
91 }
92 break;
93 }
94
95 if (cost && !cost->isValid())
96 return failure(DiagnosticCode::InvalidArgument, "action cost must be a validated non-empty CostSpec", "cost");
97 for (const auto& effectId : effectIds) {
98 if (effectId.empty())
99 return failure(DiagnosticCode::InvalidArgument, "action effect ids must not be empty", "effectIds");
100 }
101 if (timeline) {
102 auto timelineValid = timeline->validate();
103 if (!timelineValid) return failureFrom(timelineValid.status());
104 if (timeline->actionId != id)
105 return failure(DiagnosticCode::InvalidArgument, "action timeline id does not match its definition",
106 "timeline.actionId");
107 auto windupAndActive = timing.windup.tryAdd(timing.active);
108 if (!windupAndActive) return failureFrom(windupAndActive.status());
109 auto total = windupAndActive.value().tryAdd(timing.recover);
110 if (!total) return failureFrom(total.status());
111 if (timeline->duration != total.value())
112 return failure(DiagnosticCode::InvalidArgument,
113 "action timeline duration must equal windup + active + recover", "timeline.durationNs");
114 }
115 return Result<void>::success();
116}

118Result<void> ActionRequest::validate(const ActionDefinition& definition) const {
119 auto definitionValid = definition.validate();
120 if (!definitionValid) return failureFrom(definitionValid.status());
121 if (actionId != definition.id)
122 return failure(DiagnosticCode::InvalidArgument, "action request id does not match its definition", "actionId");
123
124 switch (definition.targetingMode) {
125 case TargetingMode::None:
126 if (!targetEntities.empty() || targetingQuery)
127 return failure(DiagnosticCode::InvalidArgument, "a non-targeted action cannot carry target selection",
128 "targets");
129 break;
130 case TargetingMode::Explicit:
131 if (targetEntities.empty())
132 return failure(DiagnosticCode::PreconditionViolation, "an Explicit action requires at least one target",
133 "targetEntities");
134 if (targetingQuery)
135 return failure(DiagnosticCode::InvalidArgument, "Explicit actions cannot carry a targeting query",
136 "targetingQuery");
137 break;
138 case TargetingMode::Query:
139 if (!targetingQuery)
140 return failure(DiagnosticCode::PreconditionViolation, "a Query action requires a targeting query",
141 "targetingQuery");
142 if (!targetEntities.empty())
143 return failure(DiagnosticCode::InvalidArgument, "Query actions cannot carry explicit targets",
144 "targetEntities");
145 {
146 auto valid = targetingQuery->validate();
147 if (!valid) return failureFrom(valid.status());
148 }
149 break;
150 }
151 return Result<void>::success();
152}

200Result<void> ActionRuntime::validateExecution(ActionExecution& execution) {
201 const auto& definition = execution.definition_;
202 const auto& request = execution.request_;
203
204 if (!isEmptyCondition(definition.condition)) {
205 if (services_.conditions == nullptr)
206 return failure(DiagnosticCode::Unsupported, "action condition requires an evaluator",
207 "services.conditions");
208 auto condition = services_.conditions->evaluate(definition, request);
209 if (!condition) return failureFrom(condition.status());
210 auto checked = std::move(condition).takeValue();
211 if (!checked.passed()) {
212 DiagnosticDetails details;
213 details.emplace_back("reason", decision::conditionReasonCodeName(checked.reasonCode()));
214 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
215 "action condition was rejected", "condition",
216 std::move(details)));
217 }
218 }
219
220 if (definition.targetingMode == TargetingMode::Query) {
221 if (services_.targeting == nullptr)
222 return failure(DiagnosticCode::Unsupported, "Query action requires a target resolver",
223 "services.targeting");
224 sensing::TargetingQuery query = *request.targetingQuery;
225 query.spec = *definition.targetingSpec;
226 auto targets = services_.targeting->resolve(query);
227 if (!targets) return failureFrom(targets.status());
228 execution.resolvedTargets_ = std::move(targets).takeValue();
229 }
230
231 if (definition.activeExecutionRequired || !definition.effectIds.empty()) {
232 if (services_.effects == nullptr && services_.transactionEffect == nullptr)
233 return failure(DiagnosticCode::Unsupported, "action Active phase requires an effect executor",
234 "services.effects");
235 }
236
237 if (definition.cost) {
238 const bool transactionBacked =
239 services_.transactionEffect != nullptr && services_.transactionAccount != nullptr;
240 if (services_.resources == nullptr && !transactionBacked)
241 return failure(DiagnosticCode::Unsupported, "cost-bearing action requires a resource provider",
242 "services.resources");
243 auto affordability = services_.resources != nullptr
244 ? services_.resources->canAfford(definition, request, *definition.cost)
245 : services_.transactionAccount->canAfford(*definition.cost);
246 if (!affordability) return failureFrom(affordability.status());
247 auto checked = std::move(affordability).takeValue();
248 if (!checked.affordable) {
249 DiagnosticDetails details;
250 for (const auto& shortfall : checked.shortfalls) {
251 details.emplace_back("resource", shortfall.resource.value());
252 details.emplace_back("required", std::to_string(shortfall.required.value()));
253 details.emplace_back("available", std::to_string(shortfall.available.value()));
254 }
255 return Result<void>::failure(Diagnostic::error(DiagnosticCode::PreconditionViolation,
256 "action resource cost is not affordable", "cost",
257 std::move(details)));
258 }
259 }
260 return Result<void>::success(Status::success(StatusCode::Applied));
261}

263Result<void> ActionRuntime::enterActive(ActionExecution& execution, SimulationTick tick) {
264 const auto& definition = execution.definition_;
265 const auto& request = execution.request_;
266
267 if (services_.transactionEffect != nullptr) {
268 if (definition.cost && services_.transactionAccount == nullptr)
269 return failure(DiagnosticCode::Unsupported, "transaction-backed action cost requires an account",
270 "services.transactionAccount");
271
272 const std::string transactionId = request.transactionId.empty() ? "action." + definition.id.format() + "." +
273 std::to_string(execution.id_.value())
274 : request.transactionId;
275 transaction::TransactionContext context(transactionId);
276 auto transactionResult =
277 definition.cost
278 ? transaction::AtomicResourcePayment::execute(context, *services_.transactionAccount, *definition.cost,
279 *services_.transactionEffect)
280 : transaction::AtomicResourcePayment::execute(context, *services_.transactionEffect);
281 if (!transactionResult) return failureFrom(transactionResult.status());
282 execution.transactionReceipt_ = std::move(transactionResult).takeValue();
283 execution.activeExecuted_ = true;
284 return Result<void>::success(Status::success(StatusCode::Applied));
285 }
286
287 std::unique_ptr<IActionEffectOperation> stagedEffect;
288 if (definition.activeExecutionRequired || !definition.effectIds.empty()) {
289 if (services_.effects == nullptr)
290 return failure(DiagnosticCode::Unsupported, "action Active phase requires an effect executor",
291 "services.effects");
292 auto prepared = services_.effects->prepare(
293 definition, request, execution.resolvedTargets_ ? &*execution.resolvedTargets_ : nullptr, tick);
294 if (!prepared) return failureFrom(prepared.status());
295 stagedEffect = std::move(prepared).takeValue();
296 if (!stagedEffect)
297 return failure(DiagnosticCode::InvariantViolation, "effect executor returned an empty staged operation",
298 "effects");
299 }
300
301 std::optional<resource::Reservation> reservation;
302 if (definition.cost) {
303 if (services_.resources == nullptr) {
304 if (stagedEffect) stagedEffect->rollback();
305 return failure(DiagnosticCode::Unsupported, "activating a cost-bearing action requires a resource provider",
306 "services.resources");
307 }
308 auto reserved = services_.resources->reserve(definition, request, *definition.cost);
309 if (!reserved) {
310 if (stagedEffect) stagedEffect->rollback();
311 return failureFrom(reserved.status());
312 }
313 auto credential = std::move(reserved).takeValue();
314 if (!credential.isValid()) {
315 if (stagedEffect) stagedEffect->rollback();
316 return failure(DiagnosticCode::InvariantViolation, "resource provider returned an invalid reservation",
317 "cost.reservation");
318 }
319 reservation = std::move(credential);
320
321 // The provider routes this operation by reservation.account. The
322 // action runtime never retains an account pointer, so a provider may
323 // change its default/request routing without misdirecting rollback.
324 auto committed = services_.resources->commit(*reservation);
325 if (!committed) {
326 auto rolledBack = services_.resources->rollback(*reservation);
327 if (!rolledBack) {
328 rolledBack.ignore("commit failed and rollback also failed; commit diagnostic retained");
329 }
330 if (stagedEffect) stagedEffect->rollback();
331 return failureFrom(committed.status());
332 }
333 auto receipt = std::move(committed).takeValue();
334 execution.receipt_ = std::move(receipt);
335 }
336
337 if (stagedEffect) {
338 // Resource commit is the only fallible commit boundary. Once it has
339 // succeeded, the staged effect commit is noexcept and cannot leave a
340 // successfully debited action without its Active operation.
341 stagedEffect->commit();
342 }
343 // Mark the whole Active transaction as complete even for a timing-only or
344 // cost-only action. Otherwise a zero-effect action would reserve/commit
345 // its cost again on every subsequent advance while it remains Active.
346 execution.activeExecuted_ = true;
347 return Result<void>::success(Status::success(StatusCode::Applied));
348}

360void ActionRuntime::transition(ActionExecution& execution, ActionPhase next, SimulationTick tick,
361 std::vector<ActionTransition>& transitions) {
362 if (execution.phase_ == next) return;
363 transitions.push_back({execution.phase_, next, tick});
364 execution.phase_ = next;
365 execution.phaseElapsed_ = Duration::zero();
366 if (next == ActionPhase::Completed)
367 execution.status_ = Status::success(StatusCode::Applied);
368 else if (next == ActionPhase::Requested || next == ActionPhase::Validating || next == ActionPhase::Windup ||
369 next == ActionPhase::Active || next == ActionPhase::Recover)
370 execution.status_ = pendingStatus();
371}

385Result<ActionAdvance> ActionRuntime::advance(ActionExecutionId id, SimulationTick tick, Duration delta) {
386 ActionExecution* execution = find(id);
387 if (execution == nullptr) return failureFrom<ActionAdvance>(notFoundStatus());
388 if (delta.nanoseconds() < 0)
389 return failureFrom<ActionAdvance>(invalidStatus("action advance duration must be non-negative", "delta"));
390 if (tick < execution->lastTick_)
391 return failureFrom<ActionAdvance>(invalidStatus("action simulation tick moved backwards", "tick"));
392 execution->lastTick_ = tick;
393 const Duration timelinePrevious = execution->totalElapsed_;
394
395 if (execution->phase_ == ActionPhase::Completed) {
396 ActionAdvance result{id, execution->phase_, {}, execution->phaseElapsed_, execution->totalElapsed_};
397 return Result<ActionAdvance>::success(std::move(result), Status::success(StatusCode::NoOp));
398 }
399 if (execution->phase_ == ActionPhase::Cancelled || execution->phase_ == ActionPhase::Failed)
400 return failureFrom<ActionAdvance>(execution->status());
401
402 Duration remaining = delta;
403 std::vector<ActionTransition> transitions;
404 for (int guard = 0; guard < 8; ++guard) {
405 if (execution->phase_ == ActionPhase::Requested) {
406 transition(*execution, ActionPhase::Validating, tick, transitions);
407 auto valid = validateExecution(*execution);
408 if (!valid) {
409 failExecution(*execution, valid.status(), tick, &transitions);
410 return failureFrom<ActionAdvance>(execution->status());
411 }
412 continue;
413 }
414
415 if (execution->phase_ == ActionPhase::Validating) {
416 if (execution->definition_.timing.windup.isZero())
417 transition(*execution, ActionPhase::Active, tick, transitions);
418 else
419 transition(*execution, ActionPhase::Windup, tick, transitions);
420 continue;
421 }
422
423 if (execution->phase_ == ActionPhase::Windup) {
424 const Duration phase = execution->definition_.timing.windup;
425 const auto needed = phase.nanoseconds() - execution->phaseElapsed_.nanoseconds();
426 if (remaining.nanoseconds() < needed) {
427 auto added = addElapsed(*execution, remaining);
428 if (!added) {
429 failExecution(*execution, added.status(), tick, &transitions);
430 return failureFrom<ActionAdvance>(execution->status());
431 }
432 remaining = Duration::zero();
433 break;
434 }
435 auto added = addElapsed(*execution, Duration::fromNanoseconds(needed));
436 if (!added) {
437 failExecution(*execution, added.status(), tick, &transitions);
438 return failureFrom<ActionAdvance>(execution->status());
439 }
440 remaining = Duration::fromNanoseconds(remaining.nanoseconds() - needed);
441 transition(*execution, ActionPhase::Active, tick, transitions);
442 continue;
443 }
444
445 if (execution->phase_ == ActionPhase::Active) {
446 if (!execution->activeExecuted_) {
447 auto active = enterActive(*execution, tick);
448 if (!active) {
449 failExecution(*execution, active.status(), tick, &transitions);
450 return failureFrom<ActionAdvance>(execution->status());
451 }
452 }
453 const Duration phase = execution->definition_.timing.active;
454 const auto needed = phase.nanoseconds() - execution->phaseElapsed_.nanoseconds();
455 if (needed == 0) {
456 transition(*execution, ActionPhase::Recover, tick, transitions);
457 continue;
458 }
459 if (remaining.nanoseconds() < needed) {
460 auto added = addElapsed(*execution, remaining);
461 if (!added) {
462 failExecution(*execution, added.status(), tick, &transitions);
463 return failureFrom<ActionAdvance>(execution->status());
464 }
465 remaining = Duration::zero();
466 break;
467 }
468 auto added = addElapsed(*execution, Duration::fromNanoseconds(needed));
469 if (!added) {
470 failExecution(*execution, added.status(), tick, &transitions);
471 return failureFrom<ActionAdvance>(execution->status());
472 }
473 remaining = Duration::fromNanoseconds(remaining.nanoseconds() - needed);
474 transition(*execution, ActionPhase::Recover, tick, transitions);
475 continue;
476 }
477
478 if (execution->phase_ == ActionPhase::Recover) {
479 const Duration phase = execution->definition_.timing.recover;
480 const auto needed = phase.nanoseconds() - execution->phaseElapsed_.nanoseconds();
481 if (needed == 0) {
482 transition(*execution, ActionPhase::Completed, tick, transitions);
483 continue;
484 }
485 if (remaining.nanoseconds() < needed) {
486 auto added = addElapsed(*execution, remaining);
487 if (!added) {
488 failExecution(*execution, added.status(), tick, &transitions);
489 return failureFrom<ActionAdvance>(execution->status());
490 }
491 remaining = Duration::zero();
492 break;
493 }
494 auto added = addElapsed(*execution, Duration::fromNanoseconds(needed));
495 if (!added) {
496 failExecution(*execution, added.status(), tick, &transitions);
497 return failureFrom<ActionAdvance>(execution->status());
498 }
499 remaining = Duration::fromNanoseconds(remaining.nanoseconds() - needed);
500 transition(*execution, ActionPhase::Completed, tick, transitions);
501 continue;
502 }
503
504 if (isTerminal(execution->phase_)) break;
505 }
506
507 std::vector<ActionTimelineEvent> timelineEvents;
508 if (execution->definition_.timeline) {
509 auto sampled = execution->definition_.timeline->sample(timelinePrevious, execution->totalElapsed_,
510 !execution->timelineStarted_);
511 if (!sampled) {
512 failExecution(*execution, sampled.status(), tick, &transitions);
513 return failureFrom<ActionAdvance>(execution->status());
514 }
515 timelineEvents = std::move(sampled).takeValue();
516 execution->timelineStarted_ = true;
517 }
518 if (execution->phase_ == ActionPhase::Completed) {
519 ActionAdvance result{id,
520 execution->phase_,
521 std::move(transitions),
522 execution->phaseElapsed_,
523 execution->totalElapsed_,
524 std::move(timelineEvents)};
525 return Result<ActionAdvance>::success(std::move(result), Status::success(StatusCode::Applied));
526 }
527 ActionAdvance result{id,
528 execution->phase_,
529 std::move(transitions),
530 execution->phaseElapsed_,
531 execution->totalElapsed_,
532 std::move(timelineEvents)};
533 return Result<ActionAdvance>::success(std::move(result), pendingStatus());
534}

536Result<void> ActionRuntime::cancel(ActionExecutionId id, SimulationTick tick) {
537 ActionExecution* execution = find(id);
538 if (execution == nullptr) return Result<void>::failure(notFoundStatus());
539 if (execution->phase_ == ActionPhase::Completed)
540 return Result<void>::failure(Status::failure(
541 StatusCode::Conflict,
542 Diagnostic::error(DiagnosticCode::Conflict, "completed action execution cannot be cancelled", "action")));
543 if (execution->phase_ == ActionPhase::Failed) return Result<void>::failure(execution->status_);
544 if (execution->phase_ == ActionPhase::Cancelled) return Result<void>::success(Status::success(StatusCode::NoOp));
545
546 execution->phase_ = ActionPhase::Cancelled;
547 execution->phaseElapsed_ = Duration::zero();
548 execution->lastTick_ = tick;
549 execution->status_ = cancelledStatus();
550 return Result<void>::success(Status::success(StatusCode::Applied));
551}
