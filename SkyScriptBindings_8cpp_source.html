35void exposeSkyScriptBindings(ssq::Table& table, ssq::Class& daynightClass) {
36 const auto vm = table.getHandle();
37 auto cls =
38 table.addClass<SkyRuntime>("SkyRuntime", std::function<SkyRuntime*()>([]() -> SkyRuntime* {
39 throw std::runtime_error("Use daynight.prepareSky(graphics, profileJson)");
40 }),
41 false);
42 daynightClass.addFunc("prepareSky", [vm](DayNight*, graphics::Graphics* graphics, const std::string& profileJson) {
43 if (!graphics || profileJson.size() > 65536)
44 return script::projectResult(
45 vm, invalid("Sky preparation requires graphics and at most 64 KiB of profile JSON"));
46 auto document = Value::fromJson(profileJson);
47 if (!document) return script::projectStatusResult(vm, document.status());
48 auto profile = SkyProfile::decode(document.value());
49 if (!profile) return script::projectStatusResult(vm, profile.status());
50 std::optional<graphics::SkyWispsAsset> wisps;
51 if (profile.value().proceduralSeed() || !profile.value().wispsAsset().empty()) {
52 auto loaded = profile.value().proceduralSeed()
53 ? graphics::SkyWispsAsset::generate(*profile.value().proceduralSeed())
54 : graphics::SkyWispsAsset::load(profile.value().wispsAsset());
55 if (!loaded) return script::projectStatusResult(vm, loaded.status());
56 wisps.emplace(std::move(loaded).takeValue());
57 }
58 auto prepared = SkyRuntime::prepare(*graphics, profile.value().settings(), profile.value().atmosphere(),
59 wisps ? &wisps->layer() : nullptr);
60 if (!prepared) return script::projectStatusResult(vm, prepared.status());
61 auto owned = script::makeOwnedSquirrelInstance(vm, std::move(prepared).takeValue());
62 if (!owned) return script::projectStatusResult(vm, owned.status());
63 auto result = script::projectStatusResult(vm, Status::success(), std::move(owned).takeValue());
64 result.set("ownership", std::string("owned"));
65 result.set("assetSource",
66 std::string(profile.value().proceduralSeed() ? "procedural-v1"
67 : (wisps ? "external-pack" : "atmosphere-only")));
68 return result;
69 });
70 cls.addFunc("attach", [vm](SkyRuntime* sky) {
71 return script::projectResult(vm, sky ? sky->attach() : invalid("Sky runtime must not be null"));
72 });
73 cls.addFunc("detach", [vm](SkyRuntime* sky) {
74 if (!sky) return script::projectResult(vm, invalid("Sky runtime must not be null"));
75 sky->detach();
76 return script::projectResult(vm, Result<void>::success());
77 });
78 cls.addFunc("advanceSeconds", [vm](SkyRuntime* sky, float seconds) {
79 if (!sky) return script::projectResult(vm, invalid("Sky runtime must not be null"));
80 auto duration = Duration::fromSeconds(seconds);
81 if (!duration) return script::projectStatusResult(vm, duration.status());
82 return script::projectResult(vm, sky->advance(duration.value()));
83 });
84 cls.addFunc("frame", [vm](SkyRuntime* sky) {
85 if (!sky) return script::projectResult(vm, invalid("Sky runtime must not be null"));
86 return script::projectStatusResult(vm, Status::success(), frameValue(sky->frame()));
87 });
88}
