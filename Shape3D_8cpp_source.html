21uint64_t parseCombineMode(const std::string &mode, const char *operation) {
22 if (mode == "default") return 0;
23 if (mode == "average") return 1;
24 if (mode == "minimum" || mode == "min") return 2;
25 if (mode == "multiply") return 3;
26 if (mode == "maximum" || mode == "max") return 4;
27 throw eve::Exception("%s: mode must be default|average|minimum|multiply|maximum",
28 operation);
29}

134void Shape3D::setOneWay(float nx, float ny, float nz, float planeOffset, float margin,
135 float minNormalDot) {
136 if (!std::isfinite(nx) || !std::isfinite(ny) || !std::isfinite(nz) ||
137 !std::isfinite(planeOffset) || !std::isfinite(margin) ||
138 !std::isfinite(minNormalDot))
139 throw eve::Exception("Shape3D.setOneWay: all values must be finite");
140 const float lengthSquared = nx * nx + ny * ny + nz * nz;
141 if (!(lengthSquared > 1e-12f))
142 throw eve::Exception("Shape3D.setOneWay: normal must be non-zero");
143 if (margin < 0.f)
144 throw eve::Exception("Shape3D.setOneWay: margin must be >= 0");
145 if (minNormalDot < -1.f || minNormalDot > 1.f)
146 throw eve::Exception("Shape3D.setOneWay: minNormalDot must be in [-1,1]");
147 if (isValid()) shapeValue(backend_->enablePreSolveEvents(!isSensor()));
148 const float inverseLength = 1.f / std::sqrt(lengthSquared);
149 oneWayLocalNormalX_ = nx * inverseLength;
150 oneWayLocalNormalY_ = ny * inverseLength;
151 oneWayLocalNormalZ_ = nz * inverseLength;
152 oneWayPlaneOffset_ = planeOffset;
153 oneWayMargin_ = margin;
154 oneWayMinNormalDot_ = minNormalDot;
155 oneWayEnabled_ = true;
156}

158void Shape3D::disableOneWay() {
159 if (isValid()) shapeValue(backend_->enablePreSolveEvents(false));
160 oneWayEnabled_ = false;
161}

163void Shape3D::refreshOneWayWorldData() {

172 oneWayWorldNormalZ_ = plane.nz;
173}

175bool Shape3D::allowsOneWayContact(double pointX, double pointY, double pointZ, float normalX,
176 float normalY, float normalZ) const {
177 if (!oneWayEnabled_) return true;
178 const double side = (pointX - oneWayWorldPointX_) * oneWayWorldNormalX_ +
179 (pointY - oneWayWorldPointY_) * oneWayWorldNormalY_ +
180 (pointZ - oneWayWorldPointZ_) * oneWayWorldNormalZ_;
181 const float alignment = normalX * oneWayWorldNormalX_ + normalY * oneWayWorldNormalY_ +
182 normalZ * oneWayWorldNormalZ_;
183 return side >= -static_cast<double>(oneWayMargin_) && alignment >= oneWayMinNormalDot_;
184}

298void Shape3D::setTriangleMeshMaterialIndices(
299 const std::vector<int32_t> &materialIndices) {
300 if (geometry_.kind != Kind::TriangleMesh)
301 throw eve::Exception(
302 "Shape3D.setTriangleMeshMaterialIndices: shape is not a triangle mesh");
303 const size_t triangleCount = geometry_.meshIndices.size() / 3;
304 if (materialIndices.size() != triangleCount)
305 throw eve::Exception(
306 "Shape3D.setTriangleMeshMaterialIndices: one index is required per triangle");
307 std::vector<uint8_t> encoded;
308 encoded.reserve(materialIndices.size());
309 int requiredCount = 1;
310 for (int32_t index : materialIndices) {
311 if (index < 0 || index > 254)
312 throw eve::Exception(
313 "Shape3D.setTriangleMeshMaterialIndices: indices must be in [0, 254]");
314 encoded.push_back(static_cast<uint8_t>(index));
315 requiredCount = std::max(requiredCount, static_cast<int>(index) + 1);
316 }
317 if (encoded == geometry_.meshMaterialIndices && requiredCount == static_cast<int>(meshMaterials_.size())) return;
318 const bool sensor = isSensor();
319 const auto oldIndices = geometry_.meshMaterialIndices;
320 const auto oldMaterials = meshMaterials_;
321 geometry_.meshMaterialIndices = std::move(encoded);
322 if (meshMaterials_.empty()) meshMaterials_.push_back(SurfaceMaterial3D{});
323 meshMaterials_.resize(static_cast<size_t>(requiredCount), meshMaterials_.front());
324 meshMaterialsDirty_ = true;
325 try {
326 recreate(sensor);
327 } catch (...) {
328 geometry_.meshMaterialIndices = oldIndices;
329 meshMaterials_ = oldMaterials;
330 meshMaterialsDirty_ = false;
331 throw;
332 }
333}

350void Shape3D::setTriangleMeshMaterial(int slot, float friction, float restitution,
351 float rollingResistance, float tangentX,
352 float tangentY, float tangentZ, int materialId,
353 const std::string &frictionMode,
354 const std::string &restitutionMode) {
355 if (geometry_.kind != Kind::TriangleMesh)
356 throw eve::Exception("Shape3D.setTriangleMeshMaterial: shape is not a triangle mesh");
357 if (slot < 0 || slot >= getTriangleMeshMaterialCount())
358 throw eve::Exception("Shape3D.setTriangleMeshMaterial: slot out of range");
359 if (!std::isfinite(friction) || friction < 0.f || !std::isfinite(restitution) ||
360 restitution < 0.f || !std::isfinite(rollingResistance) || rollingResistance < 0.f ||
361 !std::isfinite(tangentX) || !std::isfinite(tangentY) || !std::isfinite(tangentZ))
362 throw eve::Exception(
363 "Shape3D.setTriangleMeshMaterial: numeric values must be finite and non-negative where required");
364 const uint64_t frictionCode =
365 parseCombineMode(frictionMode, "Shape3D.setTriangleMeshMaterial");
366 const uint64_t restitutionCode =
367 parseCombineMode(restitutionMode, "Shape3D.setTriangleMeshMaterial");
368 SurfaceMaterial3D material = SurfaceMaterial3D{};
369 material.friction = friction;
370 material.restitution = restitution;
371 material.rollingResistance = rollingResistance;
372 material.tangentVelocity = {tangentX, tangentY, tangentZ};
373 material.userMaterialId = static_cast<uint32_t>(materialId) |
374 (frictionCode << frictionModeShift) |
375 (restitutionCode << restitutionModeShift);
376 if (meshMaterials_.empty()) meshMaterials_.push_back(SurfaceMaterial3D{});
377 if (isValid()) shapeValue(backend_->setMeshMaterial(material, slot));
378 meshMaterials_[static_cast<size_t>(slot)] = material;
379}

450void Shape3D::setLocalTransform(float px, float py, float pz, float qx, float qy, float qz,

459 if (!(lengthSquared > 1e-12f))
460 throw eve::Exception("Shape3D.setLocalTransform: quaternion must be non-zero");
461 const float inverseLength = 1.f / std::sqrt(lengthSquared);
462 const float normalizedQx = qx * inverseLength;
463 const float normalizedQy = qy * inverseLength;
464 const float normalizedQz = qz * inverseLength;
465 const float normalizedQw = qw * inverseLength;
466 if (geometry_.kind == Kind::HeightField && (px != 0.f || py != 0.f || pz != 0.f || normalizedQx != 0.f ||
467 normalizedQy != 0.f || normalizedQz != 0.f || normalizedQw != 1.f))
468 throw eve::Exception(
469 "Shape3D.setLocalTransform: height fields use their Body transform and require an identity local transform");
470 if (geometry_.localX == px && geometry_.localY == py && geometry_.localZ == pz &&
471 geometry_.localQx == normalizedQx && geometry_.localQy == normalizedQy && geometry_.localQz == normalizedQz &&
472 geometry_.localQw == normalizedQw)
473 return;
474
475 const bool sensor = isSensor();
476 auto candidate = geometry_;
477 candidate.localX = px;
478 candidate.localY = py;
479 candidate.localZ = pz;
480 candidate.localQx = normalizedQx;

482 candidate.localQz = normalizedQz;
