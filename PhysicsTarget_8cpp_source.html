50EditorResult<void> PhysicsColliderTarget::applyDomainOperation(const DomainOperation& operation) {
51 if (operation.target != TargetId(id_))
52 return physicsError<void>(EditorStatus::Rejected, "editor.physics.target-mismatch",
53 "Physics operation targets another collider");
54 if (operation.type != "physics.collider.property.set.v1")
55 return physicsError<void>(EditorStatus::Unsupported, "editor.physics.operation-unsupported",
56 "Unsupported physics collider operation: " + operation.type);
57 const EditorValue* pathValue = field(operation.payload, "path");
58 const EditorValue* value = field(operation.payload, "value");
59 const auto* path = pathValue ? pathValue->getIf<std::string>() : nullptr;
60 if (!path || !value)
61 return physicsError<void>(EditorStatus::Rejected, "editor.physics.operation-payload",
62 "Physics collider operation requires path and value");
63 auto descriptor = colliderSchema().find(PropertyPath(*path));
64 if (!descriptor)
65 return physicsError<void>(EditorStatus::Unsupported, "editor.physics.property-unsupported",
66 "Unknown physics collider property: " + *path);
67 auto valid = validateAssignment(*descriptor, *value);
68 if (!valid.ok()) return valid;
69 values_[*path] = *value;
70 ++revision_;
71 dirty_.include(0, 0);
72 return eve::editing::applied<void>();
73}

105EditorResult<DomainOperation> PhysicsColliderTarget::makeSet(const SelectionSnapshot& selection,
106 const PropertyPath& path, const EditorValue& value,
107 PropertySetMode mode) const {
108 if (!selectionMatches(selection))
109 return physicsError<DomainOperation>(EditorStatus::Rejected, "editor.physics.selection",
110 "Selection does not belong to this collider");
111 if (mode == PropertySetMode::Reset) return makeReset(selection, path);
112 if (mode != PropertySetMode::Absolute)
113 return physicsError<DomainOperation>(EditorStatus::Unsupported, "editor.physics.property-mode",
114 "Collider properties require absolute assignment");
115 auto descriptor = colliderSchema().find(path);
116 if (!descriptor)
117 return physicsError<DomainOperation>(EditorStatus::Unsupported, "editor.physics.property-unsupported",
118 "Unknown collider property: " + path.value());
119 auto valid = validateAssignment(*descriptor, value);
120 if (!valid.ok()) return EditorResult<DomainOperation>::failure(valid.status());
121 const auto previous = values_.find(path.value());
122 auto payload = [&](const EditorValue& assigned) {
123 EditorValue::Object object;
124 object["path"] = path.value();
125 object["value"] = assigned;
126 return EditorValue(std::move(object));
127 };
128 DomainOperation operation;
129 operation.type = "physics.collider.property.set.v1";
130 operation.target = TargetId(id_);
131 operation.payload = payload(value);
132 operation.inverse = payload(previous->second);
133 operation.hasInverse = true;
134 operation.affectedProperties.push_back(path.value());
135 operation.mergeKey = "physics.collider:" + id_ + ":" + path.value();
136 return eve::editing::applied<DomainOperation>(std::move(operation));
137}

206EditorResult<void> PhysicsColliderTarget::validateAssignment(const PropertyDescriptor& descriptor,
207 const EditorValue& value) const {
208 auto result = validatePropertyValue(descriptor, value);
209 if (!result.ok()) return result;
210 if (descriptor.path == PropertyPath("shape.size") || descriptor.path == PropertyPath("shape.offset")) {
211 const auto* tuple = value.getIf<EditorValue::Array>();
212 const std::size_t expected = dimensions_ == 2 ? 2 : 3;
213 if (!tuple || tuple->size() != expected)
214 return physicsError<void>(EditorStatus::Rejected, "editor.physics.vector-size",
215 "Physics vector has the wrong dimensionality");
216 for (const EditorValue& component : *tuple) {
217 const auto* number = component.getIf<double>();
218 if (!number || (descriptor.path == PropertyPath("shape.size") && *number <= 0.0))
219 return physicsError<void>(EditorStatus::Rejected, "editor.physics.vector-value",
220 "Physics size components must be positive numbers");
221 }
222 }
223 return eve::editing::applied<void>();
224}
