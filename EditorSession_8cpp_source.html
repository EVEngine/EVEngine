113EditorResult<EditorValue> EditorSession::executeCommand(const CommandId& id, const EditorValue& payload,
114 CommandSource source) {
115 if (!commandService_)
116 return eve::editing::failed<EditorValue>(EditorStatus::Failed, RuleId("editor.command.missing-service"),
117 "Editor session has no command service");
118
119 const CommandDescriptor* descriptor = commandService_->find(id);
120 const bool ownsTransaction = descriptor && descriptor->createsTransaction && !transactions_.isActive();
121 if (ownsTransaction) {
122 auto begun = transactions_.beginTransaction(descriptor->displayName.empty() ? id.value() : descriptor->displayName);
123 if (!begun.ok()) return projectCommonFailure<EditorValue>(begun.status());
124 }
125
126 CommandContext commandContext;
127 commandContext.session = this;
128 commandContext.profile = &hostProfile_;
129 commandContext.source = source;
130 if (IEditableTarget* currentTarget = target()) {
131 commandContext.target = TargetId(currentTarget->targetId());
132 commandContext.targetRevision = currentTarget->revision();
133 }
134
135 EditorResult<EditorValue> result = commandService_->execute(id, commandContext, payload);
136 if (ownsTransaction) {
137 if (result.ok()) {
138 auto committed = transactions_.commitTransaction();
139 if (!committed.ok()) {
140 auto discarded = transactions_.rollbackTransaction();
141 if (!discarded.ok()) {
142 std::vector<EditorDiagnostic> diagnostics = committed.diagnostics();
143 appendProjectedDiagnostics(diagnostics, discarded.status());
144 return EditorResult<EditorValue>::failure(
145 eve::Status(EditorStatus::Failed, std::move(diagnostics)));
146 }
147 return projectCommonFailure<EditorValue>(committed.status());
148 }
149 } else {
150 auto discarded = transactions_.rollbackTransaction();
151 if (!discarded.ok()) {
152 std::vector<EditorDiagnostic> diagnostics = result.diagnostics();
153 appendProjectedDiagnostics(diagnostics, discarded.status());
154 return EditorResult<EditorValue>::failure(
155 eve::Status(EditorStatus::Failed, std::move(diagnostics)));
156 }
157 }
158 }
159 return result;
160}

208EditorResult<TransactionReceipt> EditorSession::executeCommandReceipt(const CommandId& id, const EditorValue& payload,
209 CommandSource source) {
210 TransactionReceipt receipt;
211 receipt.id = TransactionId((sessionId_.empty() ? std::string("editor.session") : sessionId_.value()) +
212 ".transaction." + std::to_string(++receiptSequence_));
213 IEditableTarget* currentTarget = target();
214 receipt.beforeRevision = currentTarget ? currentTarget->revision() : 0;
215 EditorResult<EditorValue> command = executeCommand(id, payload, source);
216 currentTarget = target();
217 receipt.afterRevision = currentTarget ? currentTarget->revision() : receipt.beforeRevision;
218 receipt.diagnostics = command.diagnostics();
219 receipt.state = command.ok() ? TransactionState::Committed
220 : (command.code() == EditorStatus::Conflict ? TransactionState::Conflicted
221 : TransactionState::Rejected);
222 if (!command.ok()) return EditorResult<TransactionReceipt>::failure(command.status());
223 return EditorResult<TransactionReceipt>::success(
224 std::move(receipt), eve::Status(command.code(), command.diagnostics()));
225}

413bool EditorSession::activateTool(const std::string& id) {
414 if (id.empty()) {
415 deactivateCurrent();
416 return true;
417 }
418 IEditorTool* next = findTool(id);
419 if (!next) return false;
420 if (next == activeTool_) return true;
421 deactivateCurrent();
422 next = findTool(id);
423 if (!next) return false;
424 activeTool_ = next;
425 ++toolGeneration_;

427 activeTool_->activate(context_);
428 } catch (...) {

435 }

441ToolResponse EditorSession::dispatchPointer(const EditorPointerEvent& event) {
442 if (!activeTool_) return ToolResponse::ignored();
443 if (capturedPointerId_ >= 0 && event.pointerId != capturedPointerId_) {
444 return ToolResponse::ignored();
445 }
446 IEditorTool* const dispatchedTool = activeTool_;
447 const std::uint64_t generation = toolGeneration_;
448 ToolResponse response = dispatchedTool->pointerEvent(context_, event);
449 if (activeTool_ != dispatchedTool || toolGeneration_ != generation)
450 return {response.handled, false, false};
451 if (response.capturePointer && capturedPointerId_ < 0) capturedPointerId_ = event.pointerId;
452 if (response.releasePointer && capturedPointerId_ == event.pointerId) capturedPointerId_ = -1;
453 if (event.phase == EditorPointerEvent::Phase::Cancel && capturedPointerId_ == event.pointerId) {
454 capturedPointerId_ = -1;
455 }
456 return response;
457}

463void EditorSession::update(float dt) {
464 if (activeTool_) activeTool_->update(context_, dt);
465 if (!documentService_ || activeDocument_.empty() || dt <= 0.f) return;
466
467 if (externalPollInterval_ > 0.f) {
468 externalPollElapsed_ += dt;
469 if (externalPollElapsed_ >= externalPollInterval_) (void)pollDocumentChanges();
470 }
471
472 EditorResult<DocumentSnapshot> snapshot = documentService_->snapshot(activeDocument_);
473 if (!snapshot.ok() || !snapshot.value().dirty()) {
474 autosaveElapsed_ = 0.f;
475 return;
476 }
477 if (!autosaveService_ || autosaveInterval_ <= 0.f ||
478 snapshot.value().revision.edit == lastAutosavedRevision_)
479 return;
480 autosaveElapsed_ += dt;
481 if (autosaveElapsed_ >= autosaveInterval_) (void)autosaveDocument();
482}

497void EditorSession::deactivateCurrent() {
498 IEditorTool* current = activeTool_;
499 activeTool_ = nullptr;
500 capturedPointerId_ = -1;
501 if (!current) return;
502 ++toolGeneration_;
503
504 std::exception_ptr failure;
505 try {
506 current->cancel(context_);
507 } catch (...) {
508 failure = std::current_exception();
509 }
510 try {
511 current->deactivate(context_);
512 } catch (...) {
513 if (!failure) failure = std::current_exception();
514 }
515 if (failure) std::rethrow_exception(failure);
