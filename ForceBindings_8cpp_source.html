167Result<std::uint64_t> maskValue(const ssq::Object& value) {
168 if (value.getType() != ssq::Type::STRING) return nullInput<std::uint64_t>();
169 const auto text = value.toString();
170 std::uint64_t mask = 0;
171 const auto parsed = std::from_chars(text.data(), text.data() + text.size(), mask);
172 if (text.empty() || parsed.ec != std::errc{} || parsed.ptr != text.data() + text.size())
173 return Result<std::uint64_t>::failure(
174 Diagnostic::error(DiagnosticCode::InvalidArgument, "Expected unsigned decimal 64-bit mask"));
175 return Result<std::uint64_t>::success(mask);
176}
177template <class T>
178ssq::Table ownedValue(HSQUIRRELVM vm, Result<T> result) {
179 if (!result) return script::projectStatusResult(vm, result.status());
180 auto instance = script::makeOwnedSquirrelInstance(vm, std::make_unique<T>(std::move(result).takeValue()));
181 if (!instance) return script::projectStatusResult(vm, instance.status());
182 return script::projectStatusResult(vm, Status::success(), std::move(instance).takeValue());
183}
184template <class Model>
185Result<void> appendField(const ssq::Object& builderObject, const ssq::Object& modelObject,
186 const ssq::Object& includeObject, const ssq::Object& excludeObject) {
187 auto* builder = borrowed<ScriptFieldBuilder>(builderObject);
188 auto* model = borrowed<Model>(modelObject);
189 if (!builder || !model) return nullInput<void>();
190 auto include = maskValue(includeObject);
191 if (!include) return Result<void>::failure(include.status());
192 auto exclude = maskValue(excludeObject);
193 if (!exclude) return Result<void>::failure(exclude.status());
194 FieldDefinition definition{model->native(), {include.value(), exclude.value()}};
195 auto validated = FieldSet::build(std::span(&definition, 1));
196 if (!validated) return Result<void>::failure(validated.status());
197 builder->fields.push_back(std::move(definition));
198 return Result<void>::success();
199}
200template <class BodyType>
201Result<void> appendTarget(const ssq::Object& targetsObject, const ssq::Object& bodyObject,
202 const ssq::Object& maskObject) {
203 auto* targets = borrowed<ScriptFieldTargets>(targetsObject);
204 auto* body = borrowed<BodyType>(bodyObject);
205 if (!targets || !body) return nullInput<void>();
206 auto mask = maskValue(maskObject);
207 if (!mask) return Result<void>::failure(mask.status());
208 auto link = PhysicsLink::fromBody(*body);
209 if (!link) return Result<void>::failure(link.status());
210 targets->targets.push_back({link.value(), mask.value()});
211 return Result<void>::success();
212}
213template <class W>
214ssq::Table prepareFields(HSQUIRRELVM vm, const ssq::Object& setObject, const ssq::Object& worldObject,
215 const ssq::Object& targetsObject) {
216 auto* set = borrowed<FieldSet>(setObject);
217 auto* world = borrowed<W>(worldObject);
218 auto* targets = borrowed<ScriptFieldTargets>(targetsObject);
219 return ownedValue(vm, set && world && targets ? set->prepare(*world, targets->targets) : nullInput<ForceBatch>());
220}
221Result<SimulationStep> forceStepValue(const ssq::Object& tickObject, const ssq::Object& nsObject) {
222 auto tick = maskValue(tickObject);
223 if (!tick) return Result<SimulationStep>::failure(tick.status());
224 auto ns = maskValue(nsObject);
225 if (!ns) return Result<SimulationStep>::failure(ns.status());
226 if (ns.value() == 0 || ns.value() > 50000000)
227 return Result<SimulationStep>::failure(
228 Diagnostic::error(DiagnosticCode::InvalidArgument, "Force step duration must be 1..50000000 nanoseconds"));
229 return Result<SimulationStep>::success(
230 {SimulationTick{tick.value()}, Duration::fromNanoseconds(static_cast<std::int64_t>(ns.value()))});
231}
232template <class W>
233ssq::Table commitFields(HSQUIRRELVM vm, const ssq::Object& batchObject, const ssq::Object& worldObject,
234 const ssq::Object& tickObject, const ssq::Object& nsObject) {
235 auto* batch = borrowed<ForceBatch>(batchObject);
236 auto* world = borrowed<W>(worldObject);
237 if (!batch || !world) return script::projectResult(vm, nullInput<void>());
238 auto step = forceStepValue(tickObject, nsObject);
239 if (!step) return script::projectStatusResult(vm, step.status());
240 return ownedValue(vm, batch->commit(*world, step.value()));
241}
242template <class W>
243ssq::Table stepForceWorld(HSQUIRRELVM vm, const ssq::Object& worldObject, const ssq::Object& tickObject,
244 const ssq::Object& nsObject, const SimulationSettings& settings) {
245 auto* world = borrowed<W>(worldObject);
246 if (!world) return script::projectResult(vm, nullInput<void>());
247 auto step = forceStepValue(tickObject, nsObject);
248 if (!step) return script::projectStatusResult(vm, step.status());
249 return script::projectResult(vm, world->step(step.value(), settings));
250}
