15PropertyKind propertyKind(const ReflectedMember &member) {
16 if (member.attrString("editor") == "combo" && !member.attrOptions("options").empty())
17 return PropertyKind::Enum;
18 switch (member.value.kind) {
19 case ReflectedValueKind::Bool: return PropertyKind::Bool;
20 case ReflectedValueKind::Integer: return PropertyKind::Integer;
21 case ReflectedValueKind::Float: return PropertyKind::Number;
22 case ReflectedValueKind::String: return PropertyKind::String;

37std::string ownerClass(const Runtime &runtime, const std::string &className,
38 const std::string &memberName) {
39 std::string current = className;
40 while (!current.empty()) {
41 const ReflectedClass *reflected = runtime.reflectedClass(current);
42 if (!reflected) break;
43 const auto found = std::find_if(reflected->members.begin(), reflected->members.end(),
44 [&memberName](const ReflectedMember &member) {
45 return member.name == memberName;
46 });
47 if (found != reflected->members.end()) return current;
48 current = reflected->base;
49 }
50 return className;
51}

53PropertyFlag propertyFlags(const ReflectedMember &member) {
54 PropertyFlag flags = PropertyFlag::Runtime;
55 if (!scalar(member.value.kind) || member.attrBool("read_only") ||
56 member.attrBool("readonly"))
57 flags = flags | PropertyFlag::ReadOnly;
58 if (member.attrBool("advanced")) flags = flags | PropertyFlag::Advanced;
59 if (member.attrBool("editor_only")) flags = flags | PropertyFlag::EditorOnly;
60 if (member.attrBool("transient")) flags = flags | PropertyFlag::Transient;
61 if (member.attrBool("dangerous")) flags = flags | PropertyFlag::Dangerous;
62 return flags;
63}

65ReflectedValue toReflectedValue(const Value &value) {
66 ReflectedValue result;
67 if (const auto *boolean = value.getIf<bool>()) {
68 result.kind = ReflectedValueKind::Bool;
69 result.boolean = *boolean;
70 } else if (const auto *integer = value.getIf<std::int64_t>()) {
71 result.kind = ReflectedValueKind::Integer;
72 result.integer = *integer;
73 } else if (const auto *number = value.getIf<double>()) {
74 result.kind = ReflectedValueKind::Float;
75 result.floating = *number;
76 } else if (const auto *text = value.getIf<std::string>()) {
77 result.kind = ReflectedValueKind::String;
78 result.text = *text;
79 }
80 return result;
81}

117void ReflectedPropertyModel::rebuildSchema() {
118 schema_ = {};
119 cachedValues_.clear();
120 if (!runtime_ || instance_.getType() != ssq::Type::INSTANCE) return;
121 schema_.typeId = runtime_->classNameOf(instance_);
122 for (const ReflectedMember &member : runtime_->reflectInstance(instance_)) {
123 if (member.method) continue;
124 PropertyDescriptor descriptor;
125 descriptor.path = member.name;
126 descriptor.displayName = member.attrString("label", member.name);
127 descriptor.description = member.attrString(
128 "tooltip", member.attrString("description"));
129 descriptor.category = member.attrString(
130 "category", ownerClass(*runtime_, schema_.typeId, member.name));
131 descriptor.kind = propertyKind(member);
132 descriptor.flags = propertyFlags(member);
133 descriptor.defaultValue = convertValue(member.name, member.value);
134 descriptor.numeric.minimum = member.findAttribute("min")
135 ? std::optional<double>(member.attrFloat("min"))
136 : std::nullopt;
137 descriptor.numeric.maximum = member.findAttribute("max")
138 ? std::optional<double>(member.attrFloat("max"))
139 : std::nullopt;
140 descriptor.numeric.step = member.findAttribute("step")
141 ? std::optional<double>(member.attrFloat("step"))
142 : std::nullopt;
143 descriptor.numeric.units = member.attrString("units");
144 descriptor.numeric.precision = static_cast<int>(member.attrFloat("precision", 3.f));
145 descriptor.choices = member.attrOptions("options");
146 descriptor.presenterHint = member.attrString("editor");
147 schema_.properties.push_back(std::move(descriptor));
148 }
149 ++revision_;
150}

152Value ReflectedPropertyModel::convertValue(const std::string &path,
153 const ReflectedValue &value) const {
154 switch (value.kind) {
155 case ReflectedValueKind::Bool: return Value(value.boolean);
156 case ReflectedValueKind::Integer: return Value(value.integer);
157 case ReflectedValueKind::Float: return Value(value.floating);
158 case ReflectedValueKind::String: return Value(value.text);
159 case ReflectedValueKind::Array: {
160 Value::Array result;
161 const std::size_t count = runtime_->arraySize(instance_, path);
162 result.reserve(count);
163 for (std::size_t index = 0; index < count; ++index) {
164 const ReflectedValue entry = runtime_->arrayGet(instance_, path, index);
165 if (scalar(entry.kind)) {
166 switch (entry.kind) {
167 case ReflectedValueKind::Bool: result.emplace_back(entry.boolean); break;
168 case ReflectedValueKind::Integer: result.emplace_back(entry.integer); break;
169 case ReflectedValueKind::Float: result.emplace_back(entry.floating); break;
170 case ReflectedValueKind::String: result.emplace_back(entry.text); break;
171 default: break;
172 }
173 } else {
174 result.emplace_back();
175 }
176 }
177 return Value(std::move(result));
178 }
179 case ReflectedValueKind::Table: {
180 Value::Object result;
181 for (const std::string &key : runtime_->tableKeys(instance_, path)) {
182 const ReflectedValue entry = runtime_->tableGet(instance_, path, key);
183 switch (entry.kind) {
184 case ReflectedValueKind::Bool: result.emplace(key, Value(entry.boolean)); break;
185 case ReflectedValueKind::Integer: result.emplace(key, Value(entry.integer)); break;
186 case ReflectedValueKind::Float: result.emplace(key, Value(entry.floating)); break;
187 case ReflectedValueKind::String: result.emplace(key, Value(entry.text)); break;
188 default: result.emplace(key, Value()); break;
189 }
190 }
191 return Value(std::move(result));
192 }
193 case ReflectedValueKind::Instance: {
194 const ssq::Object nested = runtime_->readObjectProperty(instance_, path);
195 return Value(runtime_->classNameOf(nested));
196 }
197 case ReflectedValueKind::None:
198 case ReflectedValueKind::Other: return {};
199 }
200 return {};
201}

208property_access::WriteResult ReflectedPropertyModel::write(const std::string &path, const Value &value) {
209 auto descriptor = schema_.find(path);
210 if (!descriptor)
211 return property_access::WriteResult::reject("scriptmodel.property.missing", "Property is not reflected");
212 const property_access::WriteResult validation = property_access::validatePropertyValue(descriptor->get(), value);
213 if (!validation.accepted) return validationFailure(validation);
214
215 ReflectedValue reflected = toReflectedValue(value);
216 if (reflected.empty())
217 return property_access::WriteResult::reject("scriptmodel.property.type", "Unsupported property value type");
218 if (!runtime_->writeProperty(instance_, path, reflected))
219 return property_access::WriteResult::reject("scriptmodel.property.write",
220 "Runtime rejected the property write");
221 const Value applied = convertValue(path, runtime_->readProperty(instance_, path));
222 const auto found = cachedValues_.find(path);
223 if (found == cachedValues_.end() || found->second != applied) emit(path, applied);
224 return property_access::WriteResult::success();
225}

227property_access::Subscription ReflectedPropertyModel::subscribe(ChangeCallback callback) {
228 const std::uint64_t id = observers_->nextId++;
229 observers_->entries.push_back({id, std::move(callback)});
230 std::weak_ptr<ObserverState> weak = observers_;
231 return property_access::Subscription([weak, id]() {
232 if (const auto state = weak.lock())
233 std::erase_if(state->entries,
234 [id](const ObserverState::Entry &entry) { return entry.id == id; });
235 });
236}

238void ReflectedPropertyModel::refresh() {
239 for (const PropertyDescriptor &descriptor : schema_.properties) {
240 const std::optional<Value> current = read(descriptor.path);
241 if (!current) continue;
242 const auto found = cachedValues_.find(descriptor.path);
243 if (found == cachedValues_.end() || found->second != *current)
244 emit(descriptor.path, *current);
245 }
246}

248void ReflectedPropertyModel::emit(const std::string &path, const Value &value) {
249 cachedValues_[path] = value;
250 const property_access::PropertyChange change{path, value, ++revision_};
251 const auto snapshot = observers_->entries;
252 for (const ObserverState::Entry &entry : snapshot)
253 if (entry.callback) entry.callback(change);
254}
