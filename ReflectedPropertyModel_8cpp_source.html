12PropertyKind propertyKind(const ReflectedMember &member) {
13 const std::string editor = member.attrString("editor");
14 if (editor == "combo" && !member.attrOptions("options").empty()) return PropertyKind::Enum;
15 if (editor == "color") return PropertyKind::Color;
16 if (editor == "vec2") return PropertyKind::Vec2;
17 if (editor == "vec3") return PropertyKind::Vec3;
18 if (editor == "vec4") return PropertyKind::Vec4;
19 switch (member.value.kind) {
20 case ReflectedValueKind::Bool: return PropertyKind::Bool;
21 case ReflectedValueKind::Integer: return PropertyKind::Integer;
22 case ReflectedValueKind::Float: return PropertyKind::Number;
23 case ReflectedValueKind::String: return PropertyKind::String;

26 case ReflectedValueKind::Instance: return PropertyKind::ObjectRef;
27 case ReflectedValueKind::None:
28 case ReflectedValueKind::Other: return PropertyKind::ReadOnlyText;

30 return PropertyKind::Auto;

63std::string ownerClass(const Runtime &runtime, const std::string &className,
64 const std::string &memberName) {
65 // Prefer the most-base class that still declares the member so inherited
66 // fields group under their defining class even when the leaf also lists them.
67 std::string owner;
68 std::string current = className;
69 while (!current.empty()) {
70 const ReflectedClass *reflected = runtime.reflectedClass(current);
71 if (!reflected) break;
72 const auto found = std::find_if(reflected->members.begin(), reflected->members.end(),
73 [&memberName](const ReflectedMember &member) {
74 return member.name == memberName;
75 });
76 if (found != reflected->members.end()) owner = current;
77 current = reflected->base;
78 }
79 return owner.empty() ? className : owner;
80}

93ReflectedValue toReflectedValue(const Value &value) {
94 ReflectedValue result;
95 if (const auto *boolean = value.getIf<bool>()) {
96 result.kind = ReflectedValueKind::Bool;
97 result.boolean = *boolean;
98 } else if (const auto *integer = value.getIf<std::int64_t>()) {
99 result.kind = ReflectedValueKind::Integer;
100 result.integer = *integer;
101 } else if (const auto *number = value.getIf<double>()) {
102 result.kind = ReflectedValueKind::Float;

155void ReflectedPropertyModel::rebuildSchema() {
156 schema_ = {};
157 cachedValues_.clear();
158 if (!runtime_ || instance_.getType() != ssq::Type::INSTANCE) return;
159 schema_.typeId = runtime_->classNameOf(instance_);
160 for (const ReflectedMember &member : runtime_->reflectInstance(instance_)) {
161 if (member.method) continue;
162 PropertyDescriptor descriptor;
163 descriptor.path = member.name;
164 descriptor.displayName = member.attrString("label", member.name);
165 descriptor.description = member.attrString("tooltip", member.attrString("description"));
166 descriptor.category = member.attrString("category", ownerClass(*runtime_, schema_.typeId, member.name));
167 descriptor.kind = propertyKind(member);
168 descriptor.flags = propertyFlags(member, descriptor.kind);
169 descriptor.defaultValue = convertValue(member.name, member.value);
170 descriptor.numeric.minimum =
171 member.findAttribute("min") ? std::optional<double>(member.attrFloat("min")) : std::nullopt;
172 descriptor.numeric.maximum =
173 member.findAttribute("max") ? std::optional<double>(member.attrFloat("max")) : std::nullopt;
174 descriptor.numeric.step =
175 member.findAttribute("step") ? std::optional<double>(member.attrFloat("step")) : std::nullopt;
176 descriptor.numeric.units = member.attrString("units");
177 descriptor.numeric.precision = static_cast<int>(member.attrFloat("precision", 3.f));
178 descriptor.choices = member.attrOptions("options");
179 descriptor.presenterHint = member.attrString("editor");
180 schema_.properties.push_back(std::move(descriptor));
181 }
182 ++revision_;
183}

185Value ReflectedPropertyModel::convertValue(const std::string &path, const ReflectedValue &value) const {
186 switch (value.kind) {
187 case ReflectedValueKind::Bool: return Value(value.boolean);
188 case ReflectedValueKind::Integer: return Value(value.integer);
189 case ReflectedValueKind::Float: return Value(value.floating);
190 case ReflectedValueKind::String: return Value(value.text);
191 case ReflectedValueKind::Array: {
192 Value::Array result;
193 const std::size_t count = runtime_->arraySize(instance_, path);
194 result.reserve(count);
195 for (std::size_t index = 0; index < count; ++index) {
196 const ReflectedValue entry = runtime_->arrayGet(instance_, path, index);
197 if (scalar(entry.kind))
198 result.push_back(scalarValue(entry));
199 else
200 result.emplace_back();
201 }
202 return Value(std::move(result));
203 }
204 case ReflectedValueKind::Table: {
205 Value::Object result;
206 for (const std::string &key : runtime_->tableKeys(instance_, path)) {
207 const ReflectedValue entry = runtime_->tableGet(instance_, path, key);
208 if (scalar(entry.kind))
209 result.emplace(key, scalarValue(entry));
210 else
211 result.emplace(key, Value());
212 }
213 return Value(std::move(result));
214 }
215 case ReflectedValueKind::Instance: {
216 const ssq::Object nested = runtime_->readObjectProperty(instance_, path);
217 return Value(runtime_->classNameOf(nested));
218 }
219 case ReflectedValueKind::None:
220 case ReflectedValueKind::Other: return {};
221 }
222 return {};
223}

230WriteResult ReflectedPropertyModel::write(const std::string &path, const Value &value) {
231 auto descriptor = schema_.find(path);
232 if (!descriptor) return WriteResult::reject("property_access.script.missing", "Property is not reflected");
233 const WriteResult validation = validatePropertyValue(descriptor->get(), value);
234 if (!validation.accepted) return validationFailure(validation);
235
236 const PropertyKind kind = descriptor->get().kind;
237 if (kind == PropertyKind::Array || kind == PropertyKind::Color || kind == PropertyKind::Vec2 ||
238 kind == PropertyKind::Vec3 || kind == PropertyKind::Vec4) {
239 const auto *items = value.getIf<Value::Array>();
240 if (!items)
241 return WriteResult::reject("property_access.script.type", "Unsupported property value type");
242 std::size_t current = runtime_->arraySize(instance_, path);
243 while (current > items->size()) {
244 if (!runtime_->arrayRemove(instance_, path, current - 1))
245 return WriteResult::reject("property_access.script.write", "Runtime rejected the property write");
246 --current;
247 }
248 for (std::size_t index = 0; index < items->size(); ++index) {
249 const ReflectedValue reflected = toReflectedValue((*items)[index]);
250 if (reflected.empty())
251 return WriteResult::reject("property_access.script.type", "Unsupported property value type");
252 if (index < current) {
253 if (!runtime_->arraySet(instance_, path, index, reflected))
254 return WriteResult::reject("property_access.script.write",
255 "Runtime rejected the property write");
256 } else if (!runtime_->arrayAppend(instance_, path, reflected)) {
257 return WriteResult::reject("property_access.script.write", "Runtime rejected the property write");
258 }
259 }
260 } else if (kind == PropertyKind::Map || kind == PropertyKind::Struct) {
261 const auto *fields = value.getIf<Value::Object>();
262 if (!fields)
263 return WriteResult::reject("property_access.script.type", "Unsupported property value type");
264 const std::vector<std::string> existing = runtime_->tableKeys(instance_, path);
265 for (const std::string &key : existing) {
266 if (fields->find(key) != fields->end()) continue;
267 if (!runtime_->tableRemove(instance_, path, key))
268 return WriteResult::reject("property_access.script.write", "Runtime rejected the property write");
269 }
270 for (const auto &[key, entry] : *fields) {
271 const ReflectedValue reflected = toReflectedValue(entry);
272 if (reflected.empty())
273 return WriteResult::reject("property_access.script.type", "Unsupported property value type");
274 if (!runtime_->tableSet(instance_, path, key, reflected))
275 return WriteResult::reject("property_access.script.write", "Runtime rejected the property write");
276 }
277 } else {
278 ReflectedValue reflected = toReflectedValue(value);
279 if (reflected.empty())
280 return WriteResult::reject("property_access.script.type", "Unsupported property value type");
281 if (!runtime_->writeProperty(instance_, path, reflected))
282 return WriteResult::reject("property_access.script.write", "Runtime rejected the property write");
283 }
284
285 const Value applied = convertValue(path, runtime_->readProperty(instance_, path));
286 const auto found = cachedValues_.find(path);
287 if (found == cachedValues_.end() || found->second != applied) emit(path, applied);
288 return WriteResult::success();
289}

291Subscription ReflectedPropertyModel::subscribe(ChangeCallback callback) {
292 const std::uint64_t id = observers_->nextId++;
293 observers_->entries.push_back({id, std::move(callback)});
294 std::weak_ptr<ObserverState> weak = observers_;
295 return Subscription([weak, id]() {
296 if (const auto state = weak.lock())
297 std::erase_if(state->entries, [id](const ObserverState::Entry &entry) { return entry.id == id; });
298 });
299}

301void ReflectedPropertyModel::refresh() {
302 for (const PropertyDescriptor &descriptor : schema_.properties) {
303 const std::optional<Value> current = read(descriptor.path);
304 if (!current) continue;
305 const auto found = cachedValues_.find(descriptor.path);
306 if (found == cachedValues_.end() || found->second != *current) emit(descriptor.path, *current);
307 }
308}

310void ReflectedPropertyModel::emit(const std::string &path, const Value &value) {
311 cachedValues_[path] = value;
312 const PropertyChange change{path, value, ++revision_};
313 const auto snapshot = observers_->entries;
314 for (const ObserverState::Entry &entry : snapshot)
315 if (entry.callback) entry.callback(change);
