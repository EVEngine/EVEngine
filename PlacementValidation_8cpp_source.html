42bool PlacementSystem::checkTerrain(const PlacementWorld &world, const BuildingDefinition &def, const PlacementQuery &q,
43 std::string *reason) {
44 if (def.requireTerrain.empty() && def.forbidTerrain.empty()) return true;
45 bool ok = true;
46 foreachFootprintCell(def, q.cellX, q.cellY, q.rotationDeg, [&](int cx, int cy) {
47 const int sem = world.getTerrain(cx, cy);
48 if (!def.forbidTerrain.empty()) {
49 if (std::find(def.forbidTerrain.begin(), def.forbidTerrain.end(), sem) != def.forbidTerrain.end()) {
50 if (reason) *reason = "terrain_forbidden";
51 ok = false;
52 return false;

54 }
55 if (!def.requireTerrain.empty()) {
56 if (std::find(def.requireTerrain.begin(), def.requireTerrain.end(), sem) == def.requireTerrain.end()) {
57 if (reason) *reason = "terrain_mismatch";
58 ok = false;
59 return false;
60 }
61 }
62 return true;
63 });
64 return ok;

67bool PlacementSystem::checkAdjacency(const PlacementWorld &world, const BuildingDefinition &def,
68 const PlacementQuery &q, std::string *reason) {
69 const bool needTag = !def.requireAdjacentTag.empty();
70 const bool needTerrain = def.requireAdjacentTerrain >= 0;
71 if (!needTag && !needTerrain) return true;
72
73 // Collect footprint cells + expand by 1 (4-neigh).
74 std::vector<std::pair<int, int>> cells;
75 foreachFootprintCell(def, q.cellX, q.cellY, q.rotationDeg, [&](int cx, int cy) {
76 cells.emplace_back(cx, cy);
77 return true;
78 });
79
80 auto isFootprint = [&](int x, int y) {
81 return std::find(cells.begin(), cells.end(), std::make_pair(x, y)) != cells.end();
82 };
83
84 bool tagOk = !needTag;
85 bool terrainOk = !needTerrain;
86 static const int dx[4] = {1, -1, 0, 0};
87 static const int dy[4] = {0, 0, 1, -1};
88
89 for (const auto &c : cells) {
90 for (int i = 0; i < 4; ++i) {
91 const int nx = c.first + dx[i];
92 const int ny = c.second + dy[i];
93 if (isFootprint(nx, ny)) continue;
94 if (!world.inBounds(nx, ny)) continue;
95
96 if (needTag && !tagOk) {
97 const int occ = world.getAnyOccupantAtLevel(nx, ny, q.level);
98 if (occ != 0 && occ != q.excludeInstanceId) {
99 auto it = world.buildings().find(occ);
100 if (it != world.buildings().end()) {
101 const PlacedBuilding &pb = it->second;
102 bool hit = pb.hasTag(def.requireAdjacentTag);
103 if (!hit) {
104 if (const BuildingDefinition *odef = BuildingRegistry::find(pb.buildingId)) {
105 hit = odef->hasTag(def.requireAdjacentTag);
106 }
107 }
108 if (hit) tagOk = true;
109 }
110 }
111 }
112 if (needTerrain && !terrainOk) {
113 if (world.getTerrain(nx, ny) == def.requireAdjacentTerrain) terrainOk = true;
114 }
115 if (tagOk && terrainOk) return true;
116 }
117 }
118
119 if (needTag && !tagOk) {
120 if (reason) *reason = "adjacency_tag";
121 return false;
122 }
123 if (needTerrain && !terrainOk) {
124 if (reason) *reason = "adjacency_terrain";
125 return false;
126 }
127 return true;
128}

130bool PlacementSystem::runValidate(const PlacementWorld &world, const BuildingDefinition &def, const PlacementQuery &q,
131 std::string *reason) {
132 ensureBuiltins();
133 std::string rule = def.validateRule.empty() ? world.getValidateRule() : def.validateRule;
134 if (rule.empty()) rule = "default";

187bool PlacementSystem::checkStructuralSupport(const PlacementWorld &world, const BuildingDefinition &def,
188 const PlacementQuery &q, std::string *reason) {
189 if (def.supportMode == "corner_below") {
190 if (def.placementKind != "corner") {
191 if (reason) *reason = "support_mode_mismatch";
192 return false;
193 }
194 const int instanceId = world.getAnyCornerOccupantAtLevel(q.cellX, q.cellY, q.level - 1);
195 const auto found = world.buildings().find(instanceId);
196 if (instanceId == 0 || found == world.buildings().end()) {
197 if (reason) *reason = "support_missing";
198 return false;
199 }
200 if (def.supportTag.empty()) return true;
201 const BuildingDefinition *support = BuildingRegistry::find(found->second.buildingId);
202 const bool tagged = found->second.hasTag(def.supportTag) || (support && support->hasTag(def.supportTag));
203 if (!tagged && reason) *reason = "support_missing";
204 return tagged;
205 }
206 if (def.supportMode != "cell_below") return true;
207 bool supported = true;
208 foreachFootprintCell(def, q.cellX, q.cellY, q.rotationDeg, [&](int cellX, int cellY) {
209 const int instanceId = world.getAnyOccupantAtLevel(cellX, cellY, q.level - 1);
210 const auto found = world.buildings().find(instanceId);
211 if (instanceId == 0 || found == world.buildings().end()) {
212 supported = false;
213 } else if (!def.supportTag.empty()) {
214 const PlacedBuilding &placed = found->second;
215 const BuildingDefinition *support = BuildingRegistry::find(placed.buildingId);
216 // Footprint iteration visits every cell even after a callback returns false.
217 // A later tagged support must not erase an earlier missing/invalid support.
218 supported = supported && (placed.hasTag(def.supportTag) || (support && support->hasTag(def.supportTag)));
219 }
220 if (!supported && reason) *reason = "support_missing";
221 return supported;
222 });
223 return supported;
224}

253bool PlacementSystem::replacementPreservesStructuralDependents(const PlacementWorld &world, int instanceId,
254 const BuildingDefinition &replacement) {
255 const auto sourceIt = world.buildings_.find(instanceId);
256 if (sourceIt == world.buildings_.end()) return false;
257 const PlacedBuilding &source = sourceIt->second;
258
259 std::unordered_set<uint64_t> replacementCells;
260 if (replacement.placementKind != "edge") {
261 foreachFootprintCell(replacement, source.originCellX, source.originCellY, source.rotationDeg,
262 [&](int x, int y) {
263 replacementCells.insert((uint64_t(uint32_t(x)) << 32U) | uint32_t(y));
264 return true;
265 });
266 }
267
268 for (const auto &[candidateId, dependent] : world.buildings_) {
269 (void)candidateId;
270 if (std::find(dependent.supportInstanceIds.begin(), dependent.supportInstanceIds.end(), instanceId) ==
271 dependent.supportInstanceIds.end())
272 continue;
273 const BuildingDefinition *dependentDef = BuildingRegistry::find(dependent.buildingId);
274 if (!dependentDef) return false;
275 const bool tagMatches = dependentDef->supportTag.empty() || replacement.hasTag(dependentDef->supportTag);
276 if (!tagMatches) return false;
277 if (dependentDef->supportMode == "corner_below") {
278 if (replacement.placementKind != "corner" || source.corner != dependent.corner) return false;
279 continue;
280 }
281 if (dependentDef->supportMode != "cell_below" || replacement.placementKind != "cell") return false;
282 bool covered = true;
283 foreachFootprintCell(*dependentDef, dependent.originCellX, dependent.originCellY, dependent.rotationDeg,
284 [&](int x, int y) {
285 if (world.getAnyOccupantAtLevel(x, y, dependent.level - 1) != instanceId) return true;
286 const uint64_t key = (uint64_t(uint32_t(x)) << 32U) | uint32_t(y);
287 covered = replacementCells.contains(key);
288 return covered;
289 });
290 if (!covered) return false;
291 }
292 return true;
293}
